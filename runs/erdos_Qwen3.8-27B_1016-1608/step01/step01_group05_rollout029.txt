    """Simulated annealing with paired swaps on binary vectors."""
    h = h_b.copy()
    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())
    best_c = compute_c5(h, n)

        # Pick random one to flip to 0, random zero to flip to 1
        i = rng.choice(sorted(ones)) if ones else rng.randint(n)
        j = rng.choice(sorted(zeros)) if zeros else rng.randint(n)
        if i == j:
            continue

        h[i], h[j] = 0, 1
        ones.discard(i); ones.add(j)
        zeros.discard(j); zeros.add(i)
        
        new_c = compute_c5(h, n)

        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
        else:
            # Undo
            h[i], h[j] = 1, 0
            ones.discard(j); ones.add(i)
            zeros.discard(i); zeros.add(j)
    
    return h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_set(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        hp = np.clip(h - mid, 0.0, 1.0)
        if float(np.sum(hp)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def upsample(h, n_new):
    x_old = np.linspace(0.0, 2.0, len(h))
    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, h)

def to_binary(h, n):
    """Hard threshold: top n/2 values become 1, rest 0."""
    h_b = np.zeros(n)
    idx = np.argsort(-h)
    h_b[idx[:n//2]] = 1.0
    return h_b

def paired_swap_sa(h_b, n, rng, n_iters=30000, T_init=0.03, T_end=0.0005):
    h = h_b.astype(float).copy()
    cur_c = compute_c5(h, n)
    best_c = cur_c
    best_h = h.copy()
    alpha = (T_end / T_init) ** (1.0 / max(n_iters, 1))
    
    for it in range(n_iters):
        T = max(T_init * (alpha ** it), T_end)
        ones_mask = h > 0.5
        zeros_mask = ~ones_mask
        ones_idx = np.where(ones_mask)[0]
        zeros_idx = np.where(zeros_mask)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        h[i], h[j] = 0.0, 1.0
        new_c = compute_c5(h, n)
        delta = new_c - cur_c
        if delta < 0 or (T > 0 and rng.random() < np.exp(-delta / T)):
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
    return best_h, best_c

def coord_descent(h, n, rng, n_sweeps=25, steps=None):
    if steps is None:
        steps = [0.0, 1.0, -0.1, 0.1, -0.05, 0.05, -0.02, 0.02]
    target = n / 2.0
    h = h.copy()
    best = compute_c5(h, n)
    for _ in range(n_sweeps):
        order = rng.permutation(n)
        improved = False
        for j in order:
            old = h[j]
            bc, bv = old, best
            for s in steps:
                c = old + s
                if c < -0.01 or c > 1.01:
                    continue
                h[j] = np.clip(c, 0, 1)
                h = project_to_set(h, target)
                v = compute_c5(h, n)
                if v < bv:
                    bv, bc = v, float(h[j])
            if bc != old:
                h[j] = bc
                h = project_to_set(h, target)
                best = bv
                improved = True
            else:
                h[j] = old
                h = project_to_set(h, target)
        if not improved:
            break
    return h, best

def slsqp_opt(h0, n, maxiter=500, ftol=1e-13):
    target = n / 2.0
    h0 = project_to_set(h0.copy(), target)
    
    def obj(x):
        return compute_c5(x, n)
    
    def grad(x):
        # Analytic gradient of max correlation
        dx = 2.0 / n
        corr = np.correlate(x, 1.0 - x, mode="full")
        k_max = int(np.argmax(corr))
        # For np.correlate(a, b, 'full') with len(a)=len(b)=n:
        # out[k] = sum_i a[i] * b[i + k - (n-1)]
        # d(out[k])/d(a[j]) = b[j + k - (n-1)] (if in range)
        # d(out[k])/d(b[j]) = a[j - k + (n-1)] (if in range)
        # Here a=x, b=1-x
        # d/dx[j] of out[k] = b[j+k-(n-1)] + a[j-(k-(n-1))] * (-1)
        # = (1-x[j+k-(n-1)]) - x[j-k+(n-1)]
        # (with appropriate bounds checking)
        g = np.zeros(n)
        shift = k_max - (n - 1)  # b index offset
        for j in range(n):
            bi = j + shift
            ai = j - shift
            val = 0.0
            if 0 <= bi < n:
                val += (1.0 - x[bi])
            if 0 <= ai < n:
                val -= x[ai]
            g[j] = val * dx
        return g
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol},
                       jac=grad)
        h_out = project_to_set(res.x.copy(), target)
        return h_out, compute_c5(h_out, n)
    except Exception:
        return h0, compute_c5(h0, n)

def make_random_binary(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n//2, replace=False)] = 1.0
    return h

def make_pattern(n, rng):
    """Create diverse structured patterns."""
    strategy = rng.randint(4)
    h = np.zeros(n)
    half = n // 2
    if strategy == 0:  # Random
        h[rng.choice(n, half, replace=False)] = 1.0
    elif strategy == 1:  # Block
        start = rng.randint(0, n)
        for i in range(half):
            h[(start + i) % n] = 1.0
    elif strategy == 2:  # Alternating with noise
        base = (np.arange(n) % 2 == 0).astype(float)
        # Randomly flip some
        flip = rng.choice(n, rng.randint(0, n//4), replace=False)
        base[flip] = 1 - base[flip]
        # Ensure exactly half
        while base.sum() != half:
            if base.sum() > half:
                i = rng.choice(np.where(base==1)[0])
                base[i] = 0
            else:
                i = rng.choice(np.where(base==0)[0])
                base[i] = 1
        h = base
    elif strategy == 3:  # Low-freq sine threshold
        f = rng.uniform(0.5, 3.0)
        p = rng.uniform(0, 2*np.pi)
        sig = np.sin(f * 2 * np.pi * np.arange(n) / n + p)
        h = (sig > 0).astype(float)
        while h.sum() != half:
            if h.sum() > half:
                i = rng.choice(np.where(h==1)[0])
                h[i] = 0
            else:
                i = rng.choice(np.where(h==0)[0])
                h[i] = 1
    return h

def _time_left(start, budget):

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_set(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        hp = np.clip(h - mid, 0.0, 1.0)
        if float(np.sum(hp)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def upsample(h, n_new):
    x_old = np.linspace(0.0, 2.0, len(h))
    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, h)

def to_binary(h, n):
    h_b = np.zeros(n)
    idx = np.argsort(-h)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def paired_swap_sa(h_b, n, rng, n_iters=30000, T_init=0.03, T_end=0.0005):
    h = h_b.astype(float).copy()
    cur_c = compute_c5(h, n)
    best_c = cur_c
    best_h = h.copy()
    alpha = (T_end / T_init) ** (1.0 / max(n_iters, 1))
    for it in range(n_iters):
        T = max(T_init * (alpha ** it), T_end)
        oi = np.where(h > 0.5)[0]
        zi = np.where(h <= 0.5)[0]
        if len(oi) == 0 or len(zi) == 0:
            break
        i = rng.choice(oi)
        j = rng.choice(zi)
        h[i], h[j] = 0.0, 1.0
        new_c = compute_c5(h, n)
        delta = new_c - cur_c
        if delta < 0 or (T > 1e-12 and rng.random() < np.exp(-delta / T)):
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
    return best_h, best_c

def grad_max_corr(x, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    k = int(np.argmax(corr))
    shift = k - (n - 1)
    g = np.ones(n)
    for j in range(n):
        bi = j + shift
        ai = j - shift
        if 0 <= bi < n:
            g[j] -= x[bi]
        if 0 <= ai < n:
            g[j] -= x[ai]
    return g * dx

def slsqp_opt(h0, n, maxiter=600, ftol=1e-13):
    target = n / 2.0
    h0 = project_to_set(h0.copy(), target)

    def obj(x):
        return compute_c5(x, n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h0, jac=lambda x: grad_max_corr(x, n),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_out = project_to_set(res.x.copy(), target)
        return h_out, compute_c5(h_out, n)
    except Exception:
        return h0, compute_c5(h0, n)

def coord_descent(h, n, rng, n_sweeps=25, steps=None):
    if steps is None:
        steps = [0.0, 1.0, -0.1, 0.1, -0.05, 0.05, -0.02, 0.02]
    target = n / 2.0
    h = h.copy()
    best = compute_c5(h, n)
    for _ in range(n_sweeps):
        order = rng.permutation(n)
        improved = False
        for j in order:
            old = h[j]
            bv = best
            bc = old
            for s in steps:
                c = old + s
                if c < -0.01 or c > 1.01:
                    continue
                h[j] = np.clip(c, 0.0, 1.0)
                h = project_to_set(h, target)
                v = compute_c5(h, n)
                if v < bv:
                    bv, bc = v, float(h[j])
            if bc != old:
                best = bv
                improved = True
            else:
                h[j] = old
                h = project_to_set(h, target)
        if not improved:
            break
    return h, best

def make_pattern(n, rng):
    strat = rng.randint(5)
    h = np.zeros(n)
    half = n // 2
    if strat == 0:
        h[rng.choice(n, half, replace=False)] = 1.0
    elif strat == 1:
        start = rng.randint(0, n)
        for i in range(half):
            h[(start + i) % n] = 1.0
    elif strat == 2:
        h = (np.arange(n) % 2 == 0).astype(float)
        nflip = rng.randint(0, n // 3)
        for i in rng.choice(n, nflip, replace=False):
            h[i] = 1.0 - h[i]
        _fix_count(h, half, n, rng)
    elif strat == 3:
        f = rng.uniform(0.5, 3.0)
        p = rng.uniform(0, 2 * np.pi)
        sig = np.sin(f * 2 * np.pi * np.arange(n) / n + p)
        h = (sig > 0).astype(float)
        _fix_count(h, half, n, rng)
    else:
        h = (np.arange(n) % 3 != 0).astype(float)
        _fix_count(h, half, n, rng)
    return h

def _fix_count(h, half, n, rng):
    while h.sum() != half:
        if h.sum() > half:
            idx = np.where(h == 1)[0]
            h[rng.choice(idx)] = 0
        else:
            idx = np.where(h == 0)[0]
            h[rng.choice(idx)] = 1

def _time_left(start, budget):
    return max(0.0, budget - (time.time() - start))

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    budget = float(budget_s)
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = np.inf
    best_n = 0

    def update(h, n):
        nonlocal best_h, best_c5, best_n
        h = project_to_set(np.asarray(h, float), n / 2.0)
        c = compute_c5(h, n)
        if c < best_c5:
            best_h, best_c5, best_n = h.copy(), c, n

    n1 = 90
    try:
        h_init = np.array(initial_h_values, float).copy()
        if len(h_init) != n1:
            h_init = upsample(h_init, n1)
    except Exception:
        h_init = make_pattern(n1, rng)
    h_init = project_to_set(h_init, n1 / 2.0)
    update(h_init, n1)

    # SLSQP from parent
    h_opt, _ = slsqp_opt(h_init, n1, maxiter=800)
    update(h_opt, n1)

    # Binary SA from parent
    h_bin = to_binary(h_opt, n1)
    h_sa, _ = paired_swap_sa(h_bin, n1, rng, n_iters=40000, T_init=0.04, T_end=0.0005)
    update(h_sa, n1)

    # SLSQP from SA result
    h_opt, _ = slsqp_opt(h_sa, n1, maxiter=500)
    update(h_opt, n1)

    # Coordinate polish
    h_pol, _ = coord_descent(h_opt, n1, rng, n_sweeps=25)
    update(h_pol, n1)

    # Multi-start loop
    while _time_left(start, budget) > budget * 0.50:
        h_r = make_pattern(n1, rng)
        h_sa, _ = paired_swap_sa(h_r, n1, rng, n_iters=20000, T_init=0.03, T_end=0.001)
        update(h_sa, n1)
        h_opt, _ = slsqp_opt(h_sa, n1, maxiter=400)
        update(h_opt, n1)
        h_pol, _ = coord_descent(h_opt, n1, rng, n_sweeps=15)
        update(h_pol, n1)

    # Reheated SA on best
    h_bin = to_binary(best_h, best_n)
    h_sa, _ = paired_swap_sa(h_bin, best_n, rng, n_iters=50000, T_init=0.02, T_end=0.0002)
    update(h_sa, best_n)

    # Final SLSQP + CD at n1
    h_opt, _ = slsqp_opt(h_sa, n1, maxiter=600)
    update(h_opt, n1)
    h_pol, _ = coord_descent(h_opt, n1, rng, n_sweeps=30)
    update(h_pol, n1)

    # Phase 2: n=180
    if _time_left(start, budget) > budget * 0.30:
        n2 = 180
        h_up = project_to_set(upsample(best_h, n2), n2 / 2.0)
        update(h_up, n2)
        h_opt, _ = slsqp_opt(h_up, n2, maxiter=500)
        update(h_opt, n2)
        h_bin = to_binary(h_opt, n2)
        h_sa, _ = paired_swap_sa(h_bin, n2, rng, n_iters=30000, T_init=0.02, T_end=0.0005)
        update(h_sa, n2)
        h_opt, _ = slsqp_opt(h_sa, n2, maxiter=400)
        update(h_opt, n2)
        h_pol, _ = coord_descent(h_opt, best_n if best_n == n2 else n2, rng, n_sweeps=20)
        update(h_pol, n2)

    # Phase 3: n=360
    if _time_left(start, budget) > budget * 0.18:
        n3 = 360
        h_up = project_to_set(upsample(best_h, n3), n3 / 2.0)
        update(h_up, n3)
        h_opt, _ = slsqp_opt(h_up, n3, maxiter=300)
        update(h_opt, n3)
        h_pol, _ = coord_descent(h_opt, n3, rng, n_sweeps=15)
        update(h_pol, n3)

    # Final polish
    if _time_left(start, budget) > 15.0:
        h_pol, _ = coord_descent(best_h, best_n, rng, n_sweeps=40,
                                  steps=[0.0, 1.0, -0.05, 0.05, -0.02, 0.02, -0.01, 0.01])
        update(h_pol, best_n)
        h_opt, _ = slsqp_opt(h_pol, best_n, maxiter=300)
        update(h_opt, best_n)

    best_h = project_to_set(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h, best_n)
    return best_h, best_c5, best_n
