            break
        h += d / n
    return np.clip(h, 0.0, 1.0)

def _lse_val_w(c, eps):
    m = np.max(c)
    if eps < 1e-13:
        return m, None
    s = np.clip((c - m) / eps, -60.0, 60.0)
    e = np.exp(s)
    se = np.sum(e)
    return m + eps * np.log(se), e / se

def _obj(x):
    n = _g['n']
    eps = _g['eps']
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    v, _ = _lse_val_w(c, eps)
    return v

def _grad(x):
    n = _g['n']
    eps = _g['eps']
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    _, w = _lse_val_w(c, eps)
    if w is None:
        kstar = int(np.argmax(c))
        grad = np.zeros(n)
        for i in range(n):
            j1 = i + kstar - (n - 1)
            j2 = i - kstar + (n - 1)
            t1 = (1.0 - x[j1]) if 0 <= j1 < n else 0.0
            t2 = x[j2] if 0 <= j2 < n else 0.0
            grad[i] = dx * (t1 - t2)
        return grad
    ks = np.arange(2 * n - 1)
    grad = np.zeros(n)
    for i in range(n):
        idx1 = i + ks - (n - 1)
        idx2 = i - ks + (n - 1)
        m1 = (idx1 >= 0) & (idx1 < n)
        m2 = (idx2 >= 0) & (idx2 < n)
        t1 = np.where(m1, 1.0 - x[np.clip(idx1, 0, n - 1)], 0.0)
        t2 = np.where(m2, x[np.clip(idx2, 0, n - 1)], 0.0)
        grad[i] = dx * np.dot(w, t1 - t2)
    return grad

def _cfun(x):
    return np.sum(x) - _g['n'] / 2.0

def _cjac(x):
    return np.ones(_g['n'])

def do_slsqp(h, n, eps, n_outer, t_end):
    _g['n'] = n
    _g['eps'] = eps
    bds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cfun, 'jac': _cjac}]
    for _ in range(n_outer):
        if time.time() > t_end:
            break
        r = minimize(_obj, h, jac=_grad, method='SLSQP',
                     bounds=bds, constraints=cons,
                     options={'maxiter': 500, 'ftol': 1e-15})
        h = repair(r.x, n)
    return h

def interp_h(h_old, n_new):
    nOld = len(h_old)
    if nOld == n_new:
        return h_old.copy()
    xo = np.linspace(0.0, 2.0, nOld, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return repair(np.interp(xn, xo, h_old), n_new)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = 0.5 * np.ones(80)
    n0 = len(h0)
    h0 = repair(h0, n0)

    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    best_n = n0

    # Phase 1: Polish at original resolution (n0)
    t1 = min(t0 + 80, budget_s * 0.12)
    for eps in [0.01, 0.005, 0.003, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
        if time.time() > t1:
            break
        h0 = do_slsqp(h0, n0, eps, 4, t1)
        c5 = compute_c5(h0)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h0.copy(), n0

    # Phase 2: Higher-resolution optimization
    h_hi = h0.copy()
    n_hi = n0
    for nn in [160, 200]:
        if time.time() > budget_s - 250:
            break
        h_hi = interp_h(h0, nn)
        n_hi = nn
        for eps in [0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.002, 0.001,
                    0.0005, 0.0003, 0.0002, 0.0001, 0.00005]:
            if time.time() > budget_s - 120:
                break
            h_hi = do_slsqp(h_hi, nn, eps, 3, budget_s - 120)
            c5 = compute_c5(h_hi)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_hi.copy(), nn

    # Phase 3: Diverse restarts at the best resolution
    n_r = best_n
    h_r = best_h.copy()
    for i in range(600):
        if time.time() > budget_s - 60:
            break
        r = i % 7
        if r == 0:
            hr = h_r + np.random.randn(n_r) * 0.02
        elif r == 1:
            hr = h_r.copy()
            s = np.random.randint(0, max(1, n_r - 12))
            l = np.random.randint(3, min(15, n_r - s))
            hr[s:s + l] = 1.0 - hr[s:s + l]
        elif r == 2:
            k2 = np.random.randint(1, 6)
            ph = np.random.uniform(0, 2 * np.pi)
            hr = h_r + 0.03 * np.sin(2 * np.pi * k2 * np.arange(n_r) / n_r + ph)
        elif r == 3:
            hr = h_r + np.random.randn(n_r) * 0.05
        elif r == 4:
            hr = h_r.copy()
            idx = np.random.choice(n_r, size=max(1, n_r // 5), replace=False)
            hr[idx] = np.random.uniform(0, 1, size=len(idx))
        elif r == 5:
            hr = h_r.copy()
            for _ in range(np.random.randint(1, 4)):
                idx = np.random.randint(n_r)
                hr[idx] = 1.0 - hr[idx]
        else:
            hr = h_r + np.random.randn(n_r) * 0.008
        hr = repair(hr, n_r)
        hn = do_slsqp(hr, n_r, 0.003, 3, budget_s - 60)
        c5 = compute_c5(hn)
        if c5 < best_c5:
            best_c5, best_h = c5, hn.copy()
            h_r = hn.copy()

    # Phase 4: Final multi-eps polish
    if time.time() < budget_s - 30:
        for eps in [0.0001, 0.00005, 0.00002, 0.00001]:
            if time.time() > budget_s - 15:
                break
            hn = do_slsqp(best_h, best_n, eps, 4, budget_s - 15)
            c5 = compute_c5(hn)
            if c5 < best_c5:
                best_c5, best_h = c5, hn

    # Phase 5: Tiny perturbation search
    if time.time() < budget_s - 5:
        for _ in range(8000):
            if time.time() > budget_s - 2:
                break
            hp = best_h + np.random.randn(best_n) * 0.002
            hp = repair(hp, best_n)
            c5 = compute_c5(hp)
            if c5 < best_c5:
                best_c5, best_h = c5, hp

    fh = repair(best_h, best_n)
    fc = compute_c5(fh)
    return (fh.tolist(), float(fc), best_n)
