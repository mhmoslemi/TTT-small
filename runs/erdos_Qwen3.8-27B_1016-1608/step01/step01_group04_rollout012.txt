```python
def precompute_deltas(h, n):
    """Precompute delta arrays for flipping each position."""
    # delta_flip[i] = change in corr when h[i] flips
    # For h[i]=1 -> 0: delta = -(1-h[.]) at certain lags + h[.] at other lags
    # For h[i]=0 -> 1: delta = +(1-h[.]) at certain lags - h[.] at other lags
    
    # Using the formula:
    # Effect of h[i] 1->0:
    #   For lag k: -(1-h[i-k]) [if valid] + h[i+k] [if valid]
    # Effect of h[i] 0->1:
    #   For lag k: +(1-h[i-k]) [if valid] - h[i+k] [if valid]
    
    ks = np.arange(-(n-1), n)  # all lags, size 2n-1
    n_total = 2*n - 1
    
    delta_ones = np.zeros((n, n_total))  # delta if h[i] goes 1->0
    delta_zeros = np.zeros((n, n_total))  # delta if h[i] goes 0->1
    
    for i in range(n):
        if h[i] == 1:
            # 1->0
            mask_lk = (i - ks >= 0) & (i - ks < n)
            mask_pk = (i + ks >= 0) & (i + ks < n)
            delta_ones[i] = 0
            delta_ones[i][mask_lk] -= (1 - h[i - ks[mask_lk]])
            delta_ones[i][mask_pk] += h[i + ks[mask_pk]]
        else:
            # 0->1
            mask_lk = (i - ks >= 0) & (i - ks < n)
            mask_pk = (i + ks >= 0) & (i + ks < n)
            delta_zeros[i] = 0
            delta_zeros[i][mask_lk] += (1 - h[i - ks[mask_lk]])
            delta_zeros[i][mask_pk] -= h[i + ks[mask_pk]]
    
    dx = 2.0 / n
    delta_ones *= dx
    delta_zeros *= dx
    return delta_ones, delta_zeros
```

def compute_c5(h):

def to_binary(h, n):
    h = np.array(h, dtype=float)
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:n//2]] = 1.0

def greedy_local_search(h, n, max_rounds=100000, time_limit=60.0):
    t0 = time.time()
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cur_max = float(np.max(corr))
    
    for _ in range(max_rounds):
        if time.time() - t0 > time_limit:
            break
        h1 = np.where(h == 1)[0]
        h0 = np.where(h == 0)[0]

            break

        best_corr = None

            h[i] = 0
            for j in h0:

                corr_new = np.correlate(h, 1.0 - h, mode='full') * dx

def sa(h, n, t0_temp, t1_temp, n_iters, time_limit):
    t_start = time.time()
    dx = 2.0/n
    corr = np.correlate(h, 1-h, mode='full') * dx
    best_max = corr.max()
    best_h = h.copy()
    temp = t0_temp
    for it in range(n_iters):
        if time.time() - t_start > time_limit: break
        temp = t0_temp * (t1_temp/t0_temp) ** (it/n_iters)
        h1 = np.flatnonzero(h)
        h0 = np.flatnonzero(1-h)
        i = h1[np.random.randint(len(h1))]
        j = h0[np.random.randint(len(h0))]
        h[i], h[j] = h[j], h[i]
        new_corr = np.correlate(h, 1-h, mode='full') * dx
        new_max = new_corr.max()
        delta = new_max - best_max  # positive = worse
        if delta < 0 or np.random.rand() < np.exp(-delta/temp):
            corr = new_corr
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]  # revert
    return best_h, best_max
