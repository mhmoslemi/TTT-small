def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.array(h, dtype=float).copy()
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        d = n / 2.0 - np.sum(h)
        if abs(d) < 1e-14:
            break
        # Distribute adjustment only to non-bound elements
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += d / np.sum(mask)
        else:
            h += d / n
    return np.clip(h, 0.0, 1.0)
