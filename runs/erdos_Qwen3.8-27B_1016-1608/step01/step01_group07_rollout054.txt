import numpy as np
from scipy.optimize import minimize
import time

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def _fix_sum(h, target):
    n = len(h)
    for _ in range(2000):
        d = target - np.sum(h)
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        k = int(mask.sum())
        if k == 0:
            break
        h = h + (d / k) * mask
        np.clip(h, 0.0, 1.0, out=h)
    return h

def _project(h, target):
    h = np.asarray(h, dtype=float).copy()
    np.clip(h, 0.0, 1.0, out=h)
    return _fix_sum(h, target)

def _corrs(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _obj(h, alpha, n, pen):
    c = _corrs(h)
    m = c.max()
    v = np.clip(c - m, -120.0, 0.0)
    e = np.exp(v)
    se = e.sum()
    f = m + np.log(se) / alpha
    s = np.sum(h) - 0.5 * n
    f += pen * s * s
    return f

def _grad(h, alpha, n, pen):
    c = _corrs(h)
    m = c.max()
    v = np.clip(c - m, -120.0, 0.0)
    e = np.exp(v)
    se = e.sum()
    w = e / se
    active = np.where(w > 1e-12)[0]
    grad = np.zeros(n)
    for m_idx in active:
        s = m_idx - (n - 1)
        wm = w[m_idx]
        if s >= 0:
            if s < n:
                grad[:n - s] += wm * (1.0 - h[s:])
                grad[s:] += wm * (-h[:n - s])
        else:
            ss = -s
            if ss < n:
                grad[ss:] += wm * (1.0 - h[:n - ss])
                grad[:n - ss] += wm * (-h[ss:])
    dx = 2.0 / n
    grad *= dx
    if pen > 0:
        s_val = np.sum(h) - 0.5 * n
        grad += 2.0 * pen * s_val
    return grad

def _objgrad_h(h, alpha, n, pen):
    return _obj(h, alpha, n, pen), _grad(h, alpha, n, pen)

def _lbfgs(h0, n, alpha, pen, maxiter, t0, budget):
    if budget - (time.time() - t0) < 20:
        return None
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_objgrad_h, h0, args=(alpha, n, pen),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-9,
                                'maxls': 30, 'maxfun': maxiter * 3})
        if res.x is not None:
            return _project(res.x, 0.5 * n)
    except Exception:
        pass
    return None

def _momentum(h0, n, alpha, nsteps, lr0, mom, t0, budget):
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c = _c5(h)
    lr = lr0
    for i in range(nsteps):
        if budget - (time.time() - t0) < 5:
            break
        g = _grad(h, alpha, n, 0.0)
        v = mom * v + lr * g
        h = _project(h - v, 0.5 * n)
        c = _c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        lr *= 0.9995
        if i % 80 == 79:
            alpha *= 1.6
    return best_h, best_c

def _interp(h, nn):
    n = len(h)
    xo = np.arange(n, dtype=float)
    xn = np.linspace(0, n - 1, nn)
    return _project(np.interp(xn, xo, h), 0.5 * nn)

def _sinusoidal_perturb(h, n, freq, amp):
    x = np.arange(n) * 2.0 / n
    return _project(h + amp * np.sin(2 * np.pi * freq * x), 0.5 * n)
