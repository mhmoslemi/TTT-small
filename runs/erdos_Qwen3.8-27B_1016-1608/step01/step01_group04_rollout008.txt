import numpy as np
from scipy.optimize import minimize
import time

_opt = {}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair_real(h, n):
    h = np.array(h, dtype=float).copy()
    for _ in range(80):
        h = np.clip(h, 0.0, 1.0)
        d = n / 2.0 - np.sum(h)
        if abs(d) < 1e-14:
            break
        h += d / n
    return np.clip(h, 0.0, 1.0)

def round_binary(h, n):
    h = np.array(h, dtype=float).copy()
    idx = np.argsort(-h)
    out = np.zeros(n)
    out[idx[:n // 2]] = 1.0
    return out

def smax_obj(x, n, eps):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(c)
    if eps < 1e-12:
        return float(m)
    s = np.clip((c - m) / eps, -50.0, 50.0)
    return float(m + eps * np.log(np.sum(np.exp(s))))

def _eq_fun(x):
    return float(np.sum(x) - _opt['n'] / 2.0)

def _eq_jac(x):
    return np.ones(_opt['n'])

def _obj_fun(x):
    return smax_obj(x, _opt['n'], _opt['eps'])

def do_slsqp(h, n, eps, maxiter=300, ftol=1e-12):
    _opt['n'] = n
    _opt['eps'] = eps
    bds = [(0.0, 1.0)] * n
    cs = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    try:
        r = minimize(_obj_fun, h, method='SLSQP', bounds=bds,
                     constraints=cs, options={'maxiter': maxiter, 'ftol': ftol})
        return repair_real(r.x, n)
    except Exception:
        return repair_real(h, n)

def best_improving_search(hb, n, max_passes=200, deadline=None, rng=None):
    bh = hb.copy()
    bc = compute_c5(bh)
    ones = np.where(bh == 1)[0].tolist()
    zeros = np.where(bh == 0)[0].tolist()
    for _ in range(max_passes):
        if deadline and time.time() > deadline:
            break
        best_delta = 0.0
        best_i, best_j = -1, -1
        for i in ones:
            bh[i] = 0.0
            for j in zeros:
                bh[j] = 1.0
                nc = compute_c5(bh)
                d = bc - nc
                if d > best_delta:
                    best_delta = d
                    best_i, best_j = i, j
                bh[j] = 0.0
            bh[i] = 1.0
        if best_i < 0:
            break
        bh[best_i] = 0.0
        bh[best_j] = 1.0
        ones.remove(best_i)
        zeros.remove(best_j)
        ones.append(best_j)
        zeros.append(best_i)
        bc = compute_c5(bh)
    return bh, bc

def random_hill_climb(hb, n, max_iters=5000, deadline=None, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    bh = hb.copy()
    bc = compute_c5(bh)
    ones = np.where(bh == 1)[0]
    zeros = np.where(bh == 0)[0]
    for _ in range(max_iters):
        if deadline and time.time() > deadline:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        bh[i] = 0.0
        bh[j] = 1.0
        nc = compute_c5(bh)
        if nc < bc:
            bc = nc
            ones = np.where(bh == 1)[0]
            zeros = np.where(bh == 0)[0]
        else:
            bh[i] = 1.0
            bh[j] = 0.0
    return bh, bc

def simulated_anneal(hb, n, max_iters=30000, t0=0.008, t1=0.0003, deadline=None, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    bh = hb.copy()
    bc = compute_c5(bh)
    bb = bh.copy()
    bb_c = bc
    ones = np.where(bh == 1)[0]
    zeros = np.where(bh == 0)[0]
    for it in range(max_iters):
        if deadline and time.time() > deadline:
            break
        T = t0 * (t1 / t0) ** (it / max_iters)
        i = rng.choice(ones)
        j = rng.choice(zeros)
        bh[i] = 0.0
        bh[j] = 1.0
        nc = compute_c5(bh)
        delta = nc - bc
        if delta < 0 or (T > 1e-10 and rng.random() < np.exp(-delta / T)):
            bc = nc
            ones = np.where(bh == 1)[0]
            zeros = np.where(bh == 0)[0]
            if bc < bb_c:
                bb_c = bc
                bb = bh.copy()
        else:
            bh[i] = 1.0
            bh[j] = 0.0
    return bb, bb_c

def subgradient_step(h, n, lr, iters=200):
    bh = h.copy()
    for _ in range(iters):
        dx = 2.0 / n
        corr = np.correlate(bh, 1.0 - bh, mode='full') * dx
        kstar = int(np.argmax(corr))
        # Subgradient of corr[kstar] w.r.t. h[i]
        # corr[k] = sum_j h[j]*(1-h[j+k-(n-1)])
        # d/dh[i] of corr[k] = (1-h[i+k-(n-1)]) - h[i-k+(n-1)]  (where valid)
        grad = np.zeros(n)
        off1 = kstar - (n - 1)
        off2 = -(kstar - (n - 1))
        for i in range(n):
            j1 = i + off1
            j2 = i + off2
            if 0 <= j1 < n:
                grad[i] += (1.0 - bh[j1])
            if 0 <= j2 < n:
                grad[i] -= bh[j2]
        bh -= lr * grad
        bh = np.clip(bh, 0.0, 1.0)
        d = n / 2.0 - np.sum(bh)
        bh += d / n
        bh = np.clip(bh, 0.0, 1.0)
    return bh

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = 0.5 * np.ones(80)
    n = len(h0)
    h0 = repair_real(h0, n)
    bh = h0.copy()
    bc = compute_c5(bh)

    # Phase 1: SLSQP from initial (moderate effort)
    for eps in [0.01, 0.003, 0.001]:
        if time.time() - t0 > budget_s - 600:
            break
        hn = do_slsqp(bh, n, eps, 200, 1e-10)
        c5 = compute_c5(hn)
        if c5 < bc:
            bc, bh = c5, hn

    # Phase 2: Subgradient refinement (real-valued)
    if time.time() - t0 < budget_s - 500:
        for lr in [0.002, 0.001, 0.0005]:
            if time.time() - t0 > budget_s - 500:
                break
            hn = subgradient_step(bh, n, lr, iters=300)
            hn = repair_real(hn, n)
            c5 = compute_c5(hn)
            if c5 < bc:
                bc, bh = c5, hn

    # Phase 3: Convert to binary, best-improving local search
    hb = round_binary(bh, n)
    c5_bin = compute_c5(hb)
    if c5_bin < bc:
        bh, bc = hb, c5_bin
    deadline3 = t0 + budget_s - 350
    hb2, bc2 = best_improving_search(hb, n, max_passes=500, deadline=deadline3, rng=rng)
    if bc2 < bc:
        bc, bh = bc2, hb2
    # Random hill climbing from best binary
    deadline3b = t0 + budget_s - 300
    hb3, bc3 = random_hill_climb(hb2, n, max_iters=10000, deadline=deadline3b, rng=rng)
    if bc3 < bc:
        bc, bh = bc3, hb3

    # Phase 4: Simulated annealing (multiple runs)
    deadline4 = t0 + budget_s - 150
    for s in range(3):
        if time.time() - t0 > deadline4:
            break
        sa_start = round_binary(bh, n) if not np.allclose(bh, np.round(bh)) else bh.copy()
        t_hi = 0.01 - s * 0.003
        t_lo = 0.0002
        max_it = 25000 - s * 5000
        hb4, bc4 = simulated_anneal(sa_start, n, max_iters=max_it, t0=t_hi, t1=t_lo,
                                     deadline=deadline4, rng=np.random.RandomState(seed + 1000 + s))
        if bc4 < bc:
            bc, bh = bc4, hb4

    # Phase 5: Best-improving from SA result
    if time.time() - t0 < budget_s - 80:
        deadline5 = t0 + budget_s - 70
        hb5, bc5 = best_improving_search(bh, n, max_passes=300, deadline=deadline5, rng=rng)
        if bc5 < bc:
            bc, bh = bc5, hb5

    # Phase 6: SLSQP polish (real-valued, from best binary)
    if time.time() - t0 < budget_s - 30:
        hn = do_slsqp(bh, n, 0.0005, 300, 1e-12)
        c5 = compute_c5(hn)
        if c5 < bc:
            bc, bh = c5, hn

    # Phase 7: Final fine binary search
    if time.time() - t0 < budget_s - 10:
        hb_f = round_binary(bh, n)
        deadline7 = t0 + budget_s - 5
        hb7, bc7 = best_improving_search(hb_f, n, max_passes=200, deadline=deadline7, rng=rng)
        if bc7 < bc:
            bc, bh = bc7, hb7

    fh = repair_real(bh, n)
    fc = compute_c5(fh)
    return (fh.tolist(), float(fc), n)
