    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-16:
            break
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def sum_eq_fun(h):
    return np.sum(h) - len(h) / 2.0

def sum_eq_jac(h):
    return np.ones(len(h))

def soft_max_obj(h, temp):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    shifted = (c - m) / temp
    ms = shifted.max()
    return m + temp * (ms + np.log(np.sum(np.exp(shifted - ms))))

def pair_move_eval(best_h, n, i, j, d):
    a = best_h[i] + d
    b = best_h[j] - d
    if a < 0 or a > 1 or b < 0 or b > 1:
        return None
    ht = best_h.copy()
    ht[i] = a
    ht[j] = b
    c5t = compute_c5(ht)
    return (ht, c5t)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    constraints = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac}]
    bounds = [(0.0, 1.0)] * n

    # Phase 1: SLSQP with soft-max, many restarts (45% budget)
    t1 = t_start + budget_s * 0.45
    temps = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for temp in temps:
        if time.time() > t1:
            break
        res = minimize(soft_max_obj, best_h.copy(), args=(temp,),
                     method='SLSQP', bounds=bounds, constraints=constraints,
                     options={'maxiter': 10000, 'ftol': 1e-15})
        h_new = project_feasible(res.x, n)
        c5n = compute_c5(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new.copy()
        n_restarts = 12 if temp <= 0.01 else 8
        for r in range(n_restarts):
            if time.time() > t1:
                break
            scale = 0.01 + 0.008 * min(r, 6)
            h_r = project_feasible(best_h + rng.normal(0, scale, n), n)
            res = minimize(soft_max_obj, h_r, args=(temp,),
                         method='SLSQP', bounds=bounds, constraints=constraints,
                         options={'maxiter': 4000, 'ftol': 1e-15})
            h_new = project_feasible(res.x, n)
            c5n = compute_c5(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new.copy()

    # Phase 2: Simulated annealing with pair swaps (20% budget)
    t2 = t_start + budget_s * 0.65
    h_sa = best_h.copy()
    c5_sa = compute_c5(h_sa)
    h_best_sa = h_sa.copy()
    c5_best_sa = c5_sa
    T = 0.06
    while T > 0.0002 and time.time() < t2:
        for _ in range(4):
            i, j = rng.randint(0, n, 2)
            if i == j:
                continue
            d = rng.uniform(-0.3, 0.3)
            a = h_sa[i] + d
            b = h_sa[j] - d
            if a < 0 or a > 1 or b < 0 or b > 1:
                continue
            ht = h_sa.copy()
            ht[i] = a
            ht[j] = b
            c5t = compute_c5(ht)
            dc = c5t - c5_sa
            if dc < 0 or rng.uniform() < np.exp(-dc / T):
                h_sa = ht
                c5_sa = c5t
                if c5_sa < c5_best_sa:
                    c5_best_sa = c5_sa
                    h_best_sa = h_sa.copy()
        T *= 0.9985
    if c5_best_sa < best_c5:
        best_c5 = c5_best_sa
        best_h = h_best_sa.copy()

    # Phase 3: Target SLSQP polishing (18% budget)
    t3 = t_start + budget_s * 0.83
    for temp in [0.00005, 0.00002, 0.00001, 0.000005]:
        if time.time() > t3:
            break
        for r in range(4):
            if time.time() > t3:
                break
            if r == 0:
                h_s = best_h.copy()
            else:
                h_s = project_feasible(best_h + rng.normal(0, 0.004, n), n)
            res = minimize(soft_max_obj, h_s, args=(temp,),
                         method='SLSQP', bounds=bounds, constraints=constraints,
                         options={'maxiter': 15000, 'ftol': 1e-16})
            h_new = project_feasible(res.x, n)
            c5n = compute_c5(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new.copy()

    # Phase 4: Ultra-fine pair-move coordinate descent (14% budget)
    t4 = t_start + budget_s * 0.975
    steps = [0.001, -0.001, 0.002, -0.002, 0.003, -0.003, 0.005, -0.005]
    for outer in range(400):
        if time.time() > t4:
            break
        improved = False
        for i in range(n):
            if time.time() > t4:
                break
            for j in range(n):
                if i == j:
                    continue
                for d in steps:
                    res_pair = pair_move_eval(best_h, n, i, j, d)
                    if res_pair is None:
                        continue
                    ht, c5t = res_pair
                    if c5t < best_c5 - 1e-16:
                        best_c5 = c5t
                        best_h = ht
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break

    # Final feasibility and evaluation
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n)
