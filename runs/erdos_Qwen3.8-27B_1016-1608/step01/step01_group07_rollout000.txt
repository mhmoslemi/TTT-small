Note: out[n-1+k] ≈ out[n-1-k] in some symmetric sense.

    lo, hi = -2.0, 2.0

        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > t:

        else:

    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(c)
    tol = tol_frac * m + 1e-12
    active = np.where(c >= m - tol)[0]

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def _all_c(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(h, t):
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > t:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _fix(h, n):
    return _project(h, 0.5 * n)

def _grad_lag(h, k, n, dx):
    g = np.zeros(n)
    if k >= 0:
        if n - k > 0:
            g[:n-k] += (1.0 - h[k:]) * dx
        if k > 0:
            g[k:] -= h[:n-k] * dx
    else:
        kk = -k
        if n - kk > 0:
            g[:n-kk] -= h[kk:] * dx
        if kk > 0:
            g[kk:] += (1.0 - h[:n-kk]) * dx
    return g

def _pg_step(h, n, eta, tol_frac=0.04):
    dx = 2.0 / n
    c = _all_c(h)
    m = np.max(c)
    tol = tol_frac * m + 1e-12
    active = np.where(c >= m - tol)[0]
    g = np.zeros(n)
    for mi in active:
        k = mi - (n - 1)
        g += _grad_lag(h, k, n, dx)
    g /= len(active)
    h_new = _project(h - eta * g, 0.5 * n)
    return h_new, m

def _lp_solve(h0, n, max_rounds=6):
    dx = 2.0 / n
    c = _all_c(h0)
    active_set = np.argsort(c)[::-1][:20]
    h_cur = h0.copy()
    for _ in range(max_rounds):
        h_var = cp.Variable(n, bounds=(0, 1))
        t_var = cp.Variable(nonneg=True)
        cons = []
        for mi in active_set:
            k = mi - (n - 1)
            g_k = _grad_lag(h0, k, n, dx)
            c_lin = c[mi] - np.dot(g_k, h0) + g_k @ h_var
            cons.append(c_lin <= t_var)
        cons.append(cp.sum(h_var) == 0.5 * n)
        prob = cp.Problem(cp.Minimize(t_var), cons)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, verbose=False)
            except Exception:
                return None
        if h_var.value is None:
            return None
        h_new = np.array(h_var.value, dtype=float)
        h_new = _project(np.clip(h_new, 0, 1), 0.5 * n)
        c_new = _all_c(h_new)
        m_new = np.max(c_new)
        viol = np.where(c_new > m_new - (m_new * 0.02 + 1e-12))[0]
        active_set = np.unique(np.concatenate([active_set[:15], viol]))
        active_set = active_set[np.argsort(c_new[active_set])[::-1]][:40]
        h_cur = h_new
        h0 = h_new.copy()
        c = c_new.copy()
    return h_cur

def _interp(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    return _project(np.clip(h_new, 0, 1), 0.5 * n_new)

def _run_pg(h, n, max_iter, eta0, t0, budget):
    eta = eta0
    c_cur = _c5(h)
    for it in range(max_iter):
        if time.time() - t0 > budget - 20:
            break
        h_new, m = _pg_step(h, n, eta)
        c_new = _c5(h_new)
        if c_new <= c_cur + 1e-12:
            h = h_new
            c_cur = c_new
            eta = min(eta * 1.005, n * 3.0)
        else:
            eta *= 0.6
            if eta < 0.05:
                break
        if it % 1000 == 999:
            eta *= 0.85
    return h, c_cur

def _slsqp_polish(h, n, t0, budget):
    if time.time() - t0 > budget - 10:
        return h
    dx = 2.0 / n
    def _obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = np.max(c)
        e = np.exp(np.clip(c - m, -50.0, 50.0))
        return m + np.log(np.sum(e)) / 500.0
    def _ic(x):
        return np.sum(x) - 0.5 * n
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _ic, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_obj, h.copy(), method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': 300, 'ftol': 1e-12})
        if res.success and res.x is not None:
            return _project(np.clip(res.x, 0, 1), 0.5 * n)
    except Exception:
        pass
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:

    else:

    n_init = len(h_init)

    best_h = h_init.copy()
    best_c = _c5(best_h)

    resolutions = sorted(set([n_init, n_init // 2, max(60, n_init // 3), 100, 150]))
    resolutions = [r for r in resolutions if 40 <= r <= 500]

    for n_target in resolutions:
        if time.time() - t0 > budget_s - 40:
            break
        h = _interp(best_h, n_target)
        c_cur = _c5(h)

        # Phase A: Projected gradient
        h, c_cur = _run_pg(h, n_target, 5000, n_target * 0.8, t0, budget_s)
        if c_cur < best_c:
            best_c = c_cur
            best_h = h.copy()

        # Phase B: LP iterations
        for lp_round in range(30):
            if time.time() - t0 > budget_s - 30:
                break
            h_lp = _lp_solve(h, n_target, max_rounds=5)
            if h_lp is None:
                break
            c_lp = _c5(h_lp)
            if c_lp < c_cur - 1e-8:
                h = h_lp
                c_cur = c_lp
                if c_cur < best_c:
                    best_c = c_cur
                    best_h = h.copy()
            else:
                break

        # Phase C: More PG after LP (refined landscape)
        h, c_cur = _run_pg(h, n_target, 3000, n_target * 0.5, t0, budget_s)
        if c_cur < best_c:
            best_c = c_cur
            best_h = h.copy()

        print(f"n={n_target}: C5={best_c:.6f}, time={time.time()-t0:.0f}s")

    # SLSQP polish at best resolution
    nb = len(best_h)
    best_h = _slsqp_polish(best_h, nb, t0, budget_s)

    # Final fix
    best_h = _project(best_h, 0.5 * len(best_h))
    best_c = _c5(best_h)

    return best_h, best_c, len(best_h)
