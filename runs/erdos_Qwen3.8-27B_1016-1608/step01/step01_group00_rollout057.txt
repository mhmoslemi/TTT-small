```python
def project(h, S):
    # Project h onto {sum(h) = S, 0 <= h <= 1}
    lo, hi = 0.0, 0.0
    # Ensure range covers solution
    for _ in range(60):
        mid = (lo + hi) / 2
        proj = np.clip(h + mid, 0, 1)
        s = np.sum(proj)
        if s < S:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo+hi)/2, 0, 1)
```

```python
        s = np.sum(proj)

        top_idx = np.argsort(corr[active])[-n_active:]
        active = active[top_idx]
```

```python
            break
```

```python
        h = np.clip(h, 0.0, 1.0)

    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        proj = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(proj)
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def grad_at_lag(h, n, d):
    """Compute ∂corr[j]/∂h for lag d (where j = d + (n-1))."""
    s = np.zeros(n)
    t = np.arange(n)
    # Term 1: (1-h[t-d]) for valid t-d
    idx1 = t - d
    m1 = (idx1 >= 0) & (idx1 < n)
    s[m1] += (1.0 - h[idx1[m1]])
    # Term 2: -h[t+d] for valid t+d
    idx2 = t + d
    m2 = (idx2 >= 0) & (idx2 < n)
    s[m2] -= h[idx2[m2]]
    return s

    target = n / 2.0
    bnds = [(0.0, 1.0)] * n
    def obj(x):
        return compute_c5(x)
    def con(x):
        return np.sum(x) - target
    cons = [{'type': 'eq', 'fun': con}]
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.fun < obj(h0) + 1e-12:
            return repair(np.clip(res.x, 0.0, 1.0), n)
    except Exception:
        pass
    return h0

def subgrad_loop(h, n, max_iters, t0, deadline, alpha0=0.04):
    """Run projected subgradient with momentum."""
    beta = 0.85
    v = np.zeros(n)
    best = compute_c5(h)
    h_best = h.copy()
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        C = np.max(corr) * (2.0 / n)

            best = C
            h_best = h.copy()

        threshold = np.max(corr) - 0.5
        active = np.where(corr >= threshold)[0]

            active = active[np.argsort(corr[active])[-8:]]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])

        s = np.zeros(n)
        for j in active:

            s += grad_at_lag(h, n, d)

        h = project_feasible(h + v, n)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
        else:
            mask = h > 1e-13
        if not np.any(mask):
            h[:] = target / n
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        proj = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(proj)
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def grad_at_lag(h, n, d):
    s = np.zeros(n)
    t = np.arange(n)
    idx1 = t - d
    m1 = (idx1 >= 0) & (idx1 < n)
    s[m1] += (1.0 - h[idx1[m1]])
    idx2 = t + d
    m2 = (idx2 >= 0) & (idx2 < n)
    s[m2] -= h[idx2[m2]]
    return s

def slsqp_polish(h0, n, maxiter=500):
    target = n / 2.0
    bnds = [(0.0, 1.0)] * n
    def obj(x):
        return compute_c5(x)
    def con(x):
        return np.sum(x) - target
    cons = [{'type': 'eq', 'fun': con}]
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.fun < obj(h0) + 1e-12:
            return repair(np.clip(res.x, 0.0, 1.0), n)
    except Exception:
        pass
    return h0

def subgrad_loop(h, n, max_iters, deadline, alpha0=0.05, verbose=False):
    beta = 0.82
    v = np.zeros(n)
    best = compute_c5(h)
    h_best = h.copy()
    stagnation = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        C_raw = np.max(corr)
        C = C_raw * (2.0 / n)
        if C < best:
            best = C
            h_best = h.copy()
            stagnation = 0
        else:
            stagnation += 1
        threshold = C_raw - max(0.3, stagnation * 0.0005)
        active = np.where(corr >= threshold)[0]
        if len(active) > 10:
            active = active[np.argsort(corr[active])[-10:]]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        mid = n - 1
        s = np.zeros(n)
        for j in active:
            d = j - mid
            s += grad_at_lag(h, n, d)
        s /= len(active)
        alpha_t = alpha0 / (1.0 + it / 3000) ** 0.6
        v = beta * v - alpha_t * s
        h = project_feasible(h + v, n)
        if it % 3000 == 2999:
            v = np.zeros(n)
        if stagnation > 8000:
            v = np.zeros(n)
            alpha0 = max(0.01, alpha0 * 0.7)
            stagnation = 0
    return h_best, best

def binary_local_search(h, n, max_passes=20):
    hb = np.round(h).clip(0, 1)
    hb = repair(hb, n)
    best = compute_c5(hb)
    improved = True
    for _ in range(max_passes):
        if not improved:
            break
        improved = False
        for i in range(n):
            for val in [0.0, 1.0]:
                if abs(hb[i] - val) < 1e-9:
                    continue
                h_try = h
                h_try[i] = val
                h_try = repair(h_try, n)
                c = compute_c5(h_try)
                if c < best - 1e-12:
                    best = c
                    hb = h_try.copy()
                    improved = True
    return hb, best

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    try:
        h = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h = np.full(70, 0.5)
    n = len(h)
    h = repair(h, n)
    c5_best = compute_c5(h)
    h_best = h.copy()

    # Phase 1: Main subgradient (50% budget)
    dl1 = t0 + budget_s * 0.50
    h1, c1 = subgrad_loop(h, n, 800000, dl1, alpha0=0.06)
    if c1 < c5_best:
        c5_best, h_best = c1, h1

    # Phase 2: SLSQP polish (5%)
    if time.time() < t0 + budget_s * 0.58:
        hp = slsqp_polish(h_best, n, maxiter=800)
        cp = compute_c5(hp)
        if cp < c5_best:
            c5_best, h_best = cp, hp

    # Phase 3: Binary local search
    if time.time() < t0 + budget_s * 0.63:
        hb, cb = binary_local_search(h_best, n, max_passes=30)
        if cb < c5_best:
            c5_best, h_best = cb, hb
            hp = slsqp_polish(h_best, n, maxiter=500)
            cp = compute_c5(hp)
            if cp < c5_best:
                c5_best, h_best = cp, hp

    # Phase 4: Random restarts with subgradient (28%)
    dl4 = t0 + budget_s * 0.90
    scales = [0.02, 0.05, 0.12, 0.08, 0.2, 0.15, 0.03, 0.25]
    idx = 0
    while time.time() < dl4:
        scale = scales[idx % len(scales)]
        h_r = repair(h_best + np.random.randn(n) * scale, n)
        h_s, c_s = subgrad_loop(h_r, n, 60000, dl4, alpha0=0.05)
        if c_s < c5_best:
            c5_best, h_best = c_s, h_s
            if time.time() < dl4:
                hp = slsqp_polish(h_best, n, maxiter=400)
                cp = compute_c5(hp)
                if cp < c5_best:
                    c5_best, h_best = cp, hp
        idx += 1

    # Phase 5: Final SLSQP (2%)
    if time.time() < t0 + budget_s * 0.96:
        h_best = slsqp_polish(h_best, n, maxiter=3000)
        c5_best = compute_c5(h_best)

    # Phase 6: Try other resolutions (3%)
    if time.time() < t0 + budget_s * 0.995:
        for n2 in [84, 56, 98, 112, 63]:
            if time.time() > t0 + budget_s * 0.99:
                break
            xp = np.linspace(0.0, 1.0, n)
            xn = np.linspace(0.0, 1.0, n2)
            h2 = np.interp(xn, xp, h_best)
            h2 = repair(h2, n2)
            h2, c2 = subgrad_loop(h2, n2, 50000, t0 + budget_s * 0.99, alpha0=0.05)
            if c2 < c5_best:
                if time.time() < t0 + budget_s * 0.99:
                    h2 = slsqp_polish(h2, n2, maxiter=500)
                    c2 = compute_c5(h2)
                if c2 < c5_best:
                    c5_best, h_best, n = c2, h2, n2

    # Final validation
    h_best = repair(h_best, n)
    c5_best = compute_c5(h_best)
    return (h_best, c5_best, n)
```
