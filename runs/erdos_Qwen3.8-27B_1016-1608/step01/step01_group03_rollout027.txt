                    h_curr = repair(h_t, tgt)
                    improved = True
                    if compute_c5(h_curr, dx) < best_c5:
                        best_c5 = compute_c5(h_curr, dx)
                        best_h = h_curr.copy()
                        best_n = n
                else:
                    h_t[i] = orig
                    h_t = repair(h_t, tgt)
            if not improved:
                break

    # Final polish
    if time.time() - t0 < budget_s - 5:
        n = best_n
        dx = 2.0 / n
        tgt = n / 2.0
        bnds = [(0.0, 1.0)] * n

        def obj_fin(x):
            return compute_c5(repair(x, tgt), dx)

        for _ in range(5):
            if time.time() - t0 > budget_s - 3:
                break
            try:
                res = minimize(obj_fin, best_h, method='SLSQP', bounds=bnds,
                             options={'maxiter': 500, 'ftol': 1e-15})
                hf = repair(res.x, tgt)
                cf = compute_c5(hf, dx)
                if cf < best_c5:
                    best_c5, best_h = cf, hf
            except Exception:
                break

    best_h = repair(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
