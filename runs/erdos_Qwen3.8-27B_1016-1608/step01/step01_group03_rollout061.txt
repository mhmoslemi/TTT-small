def softmax_gradient(h, dx, n, beta=50.0):
    corr = np.correlate(h, 1.0 - h, mode="full")
    beta_corr = beta * corr
    # Softmax weights
    max_bc = beta_corr.max()
    weights = np.exp(beta_corr - max_bc)
    weights /= weights.sum()
    # Gradient
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if weights[k] < 1e-10:
            continue
        d = k - (n - 1)
        n_lo = max(0, -d)
        n_hi = min(n - 1, n - 1 - d)
        j_range = np.arange(n_lo, n_hi + 1)
        g[j_range] += weights[k]
        v1 = (j_range + d >= 0) & (j_range + d < n)
        g[j_range[v1]] -= weights[k] * h[j_range[v1] + d]
        v2 = (j_range - d >= 0) & (j_range - d < n)
        g[j_range[v2]] -= weights[k] * h[j_range[v2] - d]
    g *= dx
    return g

    w = np.exp(bc)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(corr))] = 1.0
    else:
        w /= ws
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        d = k - (n - 1)
        lo = max(0, -d)
        hi = min(n - 1, n - 1 - d)
        jr = np.arange(lo, hi + 1)
        g[jr] += wk
        v1 = (jr + d >= 0) & (jr + d < n)
        if v1.any():
            g[jr[v1]] -= wk * h[jr[v1] + d]
        v2 = (jr - d >= 0) & (jr - d < n)
        if v2.any():
            g[jr[v2]] -= wk * h[jr[v2] - d]
    g *= dx
    return g

def gen_starts(n, target, h0, n0, rng):
    starts = []
    if n == n0:
        starts.append(repair(h0, target))
    for _ in range(3):
        hb = np.zeros(n)
        idx = rng.choice(n, n // 2, replace=False)
        hb[idx] = 1.0
        starts.append(repair(hb, target))
    for a, w in [(0.4, 0.4), (0.5, 0.8), (0.6, 0.3)]:
        xs = np.linspace(0, 2, n)
        hs = np.zeros(n)
        hs[xs < a] = 1.0
        mid = (xs >= a) & (xs <= a + w)
        if mid.any():
            t = (xs[mid] - a) / max(w, 1e-6)
            hs[mid] = 1.0 / (1.0 + np.exp(6.0 * (t - 0.5)))
        starts.append(repair(hs, target))
    return starts

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.ones(70) * 0.5
    n0 = len(h0)

    best_c5 = 1.0
    best_h = None
    best_n = n0

    n_list = sorted(set([n0, 50, 70, 90, 110]))
    n_list = [x for x in n_list if x >= 30 and x <= 150]

    for n in n_list:
        if time.time() - t0 > budget_s - 100:
            break
        dx = 2.0 / n
        target = n / 2.0
        bnds = [(0.0, 1.0)] * n

        starts = gen_starts(n, target, h0, n0, rng)

        for h_s in starts:
            if time.time() - t0 > budget_s - 60:
                break
            h_cur = h_s.copy()
            c_cur = compute_c5(h_cur, dx)
            if c_cur < best_c5:
                best_c5, best_h, best_n = c_cur, h_cur.copy(), n

            # Soft-max subgradient descent
            lr = 0.06
            beta = 25.0
            stall = 0
            for it in range(600):
                if time.time() - t0 > budget_s - 30:
                    break
                g = softmax_grad(h_cur, dx, n, beta)
                gn = np.linalg.norm(g)
                if gn < 1e-13:
                    break
                h_new = h_cur - lr * g
                h_new = repair(h_new, target)
                c_new = compute_c5(h_new, dx)
                if c_new < c_cur - 1e-15:
                    h_cur = h_new
                    c_cur = c_new
                    if c_cur < best_c5:
                        best_c5, best_h, best_n = c_cur, h_cur.copy(), n
                    lr = min(lr * 1.04, 0.12)
                    stall = 0
                else:
                    lr *= 0.65
                    stall += 1
                    if stall > 60:
                        break
                if it % 150 == 149:
                    beta = min(beta * 1.6, 200.0)

            # SLSQP polish
            if time.time() - t0 < budget_s - 25:
                try:
                    def f_obj(x):
                        return compute_c5(np.clip(x, 0, 1), dx)
                    r = minimize(f_obj, h_cur, method='SLSQP', bounds=bnds,
                               constraints=[{'type': 'eq',
                                             'fun': lambda x: np.sum(x) - target}],
                               options={'maxiter': 600, 'ftol': 1e-15})
                    h_sl = repair(r.x, target)
                    c_sl = compute_c5(h_sl, dx)
                    if c_sl < best_c5:
                        best_c5, best_h, best_n = c_sl, h_sl.copy(), n
                except Exception:
                    pass

    # Final polish loop on global best
    if best_h is not None and time.time() - t0 < budget_s - 10:
        dx = 2.0 / best_n
        target = best_n / 2.0
        bnds = [(0.0, 1.0)] * best_n
        for _ in range(6):
            if time.time() - t0 > budget_s - 3:
                break
            try:
                def f_obj(x):
                    return compute_c5(np.clip(x, 0, 1), dx)
                r = minimize(f_obj, best_h, method='SLSQP', bounds=bnds,
                           constraints=[{'type': 'eq',
                                         'fun': lambda x: np.sum(x) - target}],
                           options={'maxiter': 1000, 'ftol': 1e-16})
                h_f = repair(r.x, target)
                c_f = compute_c5(h_f, dx)
                if c_f < best_c5:
                    best_c5, best_h = c_f, h_f
                else:
                    break
            except Exception:
                break

    best_h = repair(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
