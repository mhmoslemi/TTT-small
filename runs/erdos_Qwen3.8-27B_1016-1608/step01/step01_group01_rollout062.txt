    c_max = c[sorted_idx[0]]

    active = [k for k in sorted_idx if c[k] > 0.95 * c_max]

    grads = []
    m_arr = np.arange(n)
    for k in active:
        shift = k - (n-1)
        grad = np.zeros(n)
        jp = m_arr + shift
        jm = m_arr - shift
        vp = (jp >= 0) & (jp < n)
        vm = (jm >= 0) & (jm < n)
        grad[vp] += (1.0 - h[jp[vp]])
        grad[vm] -= h[jm[vm]]
        grads.append(grad)

    if len(grads) == 1:
        direction = -grads[0]
    else:

        # This is a QP. For 2-3 gradients, solve analytically.
        G = np.array(grads)  # shape (k, n)
        # We want to find w that minimizes ||G^T w||^2 s.t. w >= 0, 1^T w = 1

        # Simple: use uniform weights, then project
        w = np.ones(k_act) / k_act
        # Check if this is a descent direction
        direction = -(G.T @ w)
        
        # Project onto feasible cone (orthogonal to sum constraint)
        direction -= direction.mean()

    # For each shift k (0 to 2n-2), compute the subgradient
    # This gives a matrix of shape (2n-1, n)
    # grad[k, j] = (1 - h[j+s_k]) - h[j-s_k]

    j = np.arange(n)  # shape (n,)
    # For shift s: valid j+s requires 0 <= j+s < n, valid j-s requires 0 <= j-s < n
    # We need to handle boundaries carefully
    
    all_grads = np.zeros((2*n-1, n))
    for idx, s in enumerate(shifts):
        # (1 - h[j+s]) part: valid where 0 <= j+s < n
        jp = j + s
        vp = (jp >= 0) & (jp < n)
        if vp.any():
            all_grads[idx, vp] += (1.0 - h[jp[vp]])
        # -h[j-s] part: valid where 0 <= j-s < n
        jm = j - s
        vm = (jm >= 0) & (jm < n)
        if vm.any():
            all_grads[idx, vm] -= h[jm[vm]]
    return all_grads  # shape (2n-1, n)

            c5_res2 = compute_c5(h_res2)
            if c5_res2 < best_c5:
                best_c5 = c5_res2
                best_h = h_res2.copy()
                h = h_res2.copy()

    # Phase 5: Progressive coordinate descent (25% time)
    t5 = t_start + budget_s * 0.94
    for rnd in range(40):
        if time.time() > t5:
            break
        ss = 0.004 / (1.0 + 0.25 * rnd)
        best_h, best_c5 = coord_descent(best_h, n, ss, max_rounds=3)

    # Phase 6: Final SLSQP squeeze
    if time.time() < t_start + budget_s * 0.995:
        h_res = do_slsqp(best_h, 0.00002, 12000)
        c5_res = compute_c5(h_res)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res.copy()
        h_comp = project_feasible(1.0 - best_h, n)
        h_res2 = do_slsqp(h_comp, 0.00002, 6000)
        c5_res2 = compute_c5(h_res2)
        if c5_res2 < best_c5:
            best_c5 = c5_res2
            best_h = h_res2.copy()

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n)
