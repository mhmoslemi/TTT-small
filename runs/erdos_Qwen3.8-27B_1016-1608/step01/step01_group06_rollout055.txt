def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h = np.clip(h + diff / cnt * mask, 0.0, 1.0)
    return h

def compute_c(h):
    n = len(h)
    dx = 2.0 / n

    return np.max(compute_c(h))
