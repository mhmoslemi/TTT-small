import numpy as np
from scipy.optimize import minimize
import time

def _compute_c5_and_worst(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr)
    return float(corr[idx] * dx), idx

def _compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _project(h, target):
    lo, hi = -1.0, 1.0
    for _ in range(45):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _upsample(h, n_new):
    x_old = np.linspace(0.0, 2.0, len(h), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)

def _random_binary(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, size=n//2, replace=False)] = 1.0
    return h

def _coord_descent(h, n, rng, iters=100):
    h = h.copy()
    target = n / 2.0
    best = _compute_c5(h, n)
    
    for it in range(iters):
        improved = False
        for j in rng.permutation(n):
            old = h[j]
            best_c, best_v = old, best
            for c in (0.0, 1.0, 0.5, max(0.0, old-0.1), min(1.0, old+0.1)):
                if abs(c - old) < 1e-9:
                    continue
                h[j] = c
                h = _project(h, target)
                v = _compute_c5(h, n)
                if v < best_v:
                    best_v, best_c = v, c
            if best_c != old:
                h[j] = best_c
                h = _project(h, target)
                best = best_v
                improved = True
            else:
                h = _project(h, target)
        if not improved:
            break
    return h, best

def _subgrad(h, n, iters=3000):
    h = h.copy()
    target = n / 2.0
    best = _compute_c5(h, n)
    
    for it in range(iters):
        alpha = 0.05 * max(0.01, 1.0 - it / iters)
        c, worst_idx = _compute_c5_and_worst(h, n)
        k = worst_idx - (n - 1)
        
        grad = np.zeros(n)
        # ∂A(k)/∂h[j] = (1-h[j-k]) - h[j+k] (with bounds)
        for j in range(n):
            if 0 <= j - k < n:
                grad[j] += (1.0 - h[j - k])
            if 0 <= j + k < n:
                grad[j] -= h[j + k]
        
        grad -= np.mean(grad)
        h = h - alpha * grad
        h = np.clip(h, 0.0, 1.0)
        h = _project(h, target)
        
        c2 = _compute_c5(h, n)
        if c2 < best:
            best = c2
    return h, best

def _sim_anneal(h, n, rng, n_iter=20000, t0=0.02, t1=0.0005):
    h = h.copy()
    target = n / 2.0
    best_h = h.copy()
    best_c = _compute_c5(h, n)
    cur_c = best_c
    
    for it in range(n_iter):
        t = t0 * (t1/t0)**(it/n_iter)
        move = rng.randint(4)
        if move == 0:
            j = rng.randint(n)
            h[j] = 1.0 - h[j]
        elif move == 1:
            j = rng.randint(n)
            h[j] = 1.0 - h[j]
            k = rng.randint(n)
            h[k] = 1.0 - h[k]
        elif move == 2:
            j = rng.randint(n)
            h[j] += rng.uniform(-0.4, 0.4)
        else:
            j = rng.randint(n-3)
            v = rng.choice([0.0, 1.0])
            h[j:j+3] = v
        
        h = _project(h, target)
        new_c = _compute_c5(h, n)
        
        if new_c < cur_c:
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        elif rng.random() < np.exp((cur_c - new_c) / max(t, 1e-10)):
            cur_c = new_c
    
    return best_h, best_c

def _time_left(budget, start):
    return max(0.0, budget - (time.time() - start))

    t0 = time.time()

def _c5_worst(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    return float(corr[idx] * dx), idx

def _proj(h, tgt):
    lo, hi = -1.0, 1.0
    for _ in range(40):
        m = 0.5 * (lo + hi)
        if float(np.sum(np.clip(h - m, 0.0, 1.0))) > tgt:
            lo = m
        else:
            hi = m
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _upsample(h, nn):
    x0 = np.linspace(0.0, 2.0, len(h), endpoint=False)
    x1 = np.linspace(0.0, 2.0, nn, endpoint=False)
    return np.interp(x1, x0, h)

def _random_bin(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h

def _cd(h, n, rng, iters=80):
    tgt = n / 2.0
    h = h.copy()
    best = _c5(h, n)
    for _ in range(iters):
        imp = False
        for j in rng.permutation(n):
            old = h[j]
            bc, bv = old, best
            for c in (0.0, 1.0, 0.5, max(0.0, old - 0.1), min(1.0, old + 0.1)):
                if abs(c - old) < 1e-9:
                    continue
                h[j] = c
                h = _proj(h, tgt)
                v = _c5(h, n)
                if v < bv:
                    bv, bc = v, c
            if bc != old:
                h[j] = bc
                h = _proj(h, tgt)
                best, imp = bv, True
            else:
                h = _proj(h, tgt)
        if not imp:
            break
    return h, best

def _subgrad(h, n, iters=2000):
    tgt = n / 2.0
    h = h.copy()
    best = _c5(h, n)
    for it in range(iters):
        a = 0.04 * max(0.02, 1.0 - it / float(iters))
        _, wi = _c5_worst(h, n)
        k = wi - (n - 1)
        g = np.zeros(n)
        for j in range(n):
            if 0 <= j - k < n:
                g[j] += 1.0 - h[j - k]
            if 0 <= j + k < n:
                g[j] -= h[j + k]
        g -= np.mean(g)
        h = np.clip(h - a * g, 0.0, 1.0)
        h = _proj(h, tgt)
        c = _c5(h, n)
        if c < best:
            best = c
    return h, best

def _sa(h, n, rng, nit=12000, t0=0.025, t1=0.001):
    tgt = n / 2.0
    h = h.copy()
    bh = h.copy()
    bc = _c5(h, n)
    cc = bc
    for it in range(nit):
        t = t0 * (t1 / t0) ** (it / max(1, nit - 1))
        mv = rng.randint(5)
        if mv == 0:
            h[rng.randint(n)] = 1.0 - h[rng.randint(n)]
        elif mv == 1:
            i, j = rng.randint(n), rng.randint(n)
            h[i] = 1.0 - h[i]
            h[j] = 1.0 - h[j]
        elif mv == 2:
            h[rng.randint(n)] += rng.uniform(-0.5, 0.5)
        elif mv == 3:
            j = rng.randint(n - 4)
            h[j:j + 4] = rng.choice([0.0, 1.0])
        else:
            i, j = rng.choice(n, 2, replace=False)
            d = rng.uniform(-0.3, 0.3)
            h[i] += d
            h[j] -= d
        h = np.clip(h, 0.0, 1.0)
        h = _proj(h, tgt)
        nc = _c5(h, n)
        if nc < cc:
            cc = nc
            if nc < bc:
                bc, bh = nc, h.copy()
        elif rng.random() < np.exp((cc - nc) / max(t, 1e-12)):
            cc = nc
    return bh, bc

def _block_opt(h, n, rng, iters=40):
    """Try setting contiguous blocks to 0 or 1, keeping sum constraint."""
    tgt = n / 2.0
    h = h.copy()
    best = _c5(h, n)
    for _ in range(iters):
        imp = False
        for _bb in range(20):
            L = rng.randint(2, max(3, n // 8))
            j = rng.randint(n - L + 1)
            val = 0.0 if np.mean(h[j:j+L]) > 0.5 else 1.0
            old_block = h[j:j+L].copy()
            h[j:j+L] = val
            h = _proj(h, tgt)
            v = _c5(h, n)
            if v < best:
                best = v
                imp = True
            else:
                h[:] = _proj(h, tgt)
        if not imp:
            break
    return h, best

def _time_left(bud, t0):
    return max(0.0, bud - (time.time() - t0))

def run(seed=42, budget_s=1100, **kwargs):
    bud = float(budget_s)
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h, best_c, best_n = None, np.inf, 0

    def _upd(h, n):
        nonlocal best_h, best_c, best_n
        h = _proj(np.asarray(h, dtype=float), n / 2.0)
        c = _c5(h, n)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, n

    # Initialize from parent
    n0 = 90
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        if len(h0) != n0:
            h0 = _upsample(h0, n0)
    except NameError:
        h0 = _random_bin(n0, rng)
    _upd(h0, n0)

    # Also try n=45
    n_small = 45
    h_small = _proj(_upsample(h0, n_small), n_small / 2.0)
    _upd(h_small, n_small)

    # Phase 1: Coordinate descent at n=90
    if _time_left(bud, t0) > 20:
        _upd(_cd(best_h, best_n, rng, iters=60), best_n)

    # Phase 2: Subgradient
    if _time_left(bud, t0) > 30:
        _upd(_subgrad(best_h, best_n, iters=2000), best_n)

    # Phase 3: SA + CD loop
    while _time_left(bud, t0) > bud * 0.32:
        _upd(_sa(best_h, best_n, rng, nit=10000, t0=0.03, t1=0.001), best_n)
        _upd(_cd(best_h, best_n, rng, iters=30), best_n)
        _upd(_block_opt(best_h, best_n, rng, iters=20), best_n)

    # Phase 4: Try n=45 refinement
    if _time_left(bud, t0) > bud * 0.15 and best_n != 45:
        h45 = _proj(_upsample(best_h, 45), 22.5)
        _upd(h45, 45)
        _upd(_cd(h45, 45, rng, iters=50), 45)
        _upd(_sa(_proj(h45, 22.5), 45, rng, nit=8000, t0=0.02, t1=0.0005), 45)
        _upd(_cd(best_h, best_n, rng, iters=20), best_n)

    # Phase 5: Upsample to 180
    if _time_left(bud, t0) > bud * 0.18:
        n2 = 180
        h2 = _proj(_upsample(best_h, n2), n2 / 2.0)
        _upd(h2, n2)
        _upd(_cd(h2, n2, rng, iters=25), n2)
        _upd(_subgrad(best_h, best_n, iters=1000), best_n)

    # Phase 6: Final refinement at best resolution
    while _time_left(bud, t0) > 8:
        _upd(_cd(best_h, best_n, rng, iters=20), best_n)
        _upd(_subgrad(best_h, best_n, iters=500), best_n)
        _upd(_block_opt(best_h, best_n, rng, iters=10), best_n)

    best_h = _proj(best_h, best_n / 2.0)
    best_c = _c5(best_h, best_n)
    return best_h, best_c, best_n
