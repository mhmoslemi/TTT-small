def _fix(h, n):
    """Project h onto {x in [0,1]^n : sum(x) = n/2}"""
    t = 0.5 * n
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)

        if abs(d) < 1e-12:

        else:

    n = len(h)
    v = _c5_vals(h)

The correlation array v has length 2n-1. Index m corresponds to lag m - (n-1).

The total gradient for the weighted objective is:

    """Compute all lag values."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')

    v_half = c[n-1:] * dx  # length n, v_half[k] = value at lag k
    return v_half

def _grad_half(h, w_half):

    Compute gradient of Σ_k w_half[k] * f_k(h) w.r.t. h
    where f_k = sum_i h[i]*(1-h[i-k]) for valid i
    w_half has length n (for k = 0, ..., n-1)

        ct = _c5(h_new)
        if ct < best_c:
            best_c = ct
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def _v_half(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return c[n-1:] * dx

def _fix(h, n):
    t = 0.5 * n
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        d = t - np.sum(h)
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        k = int(np.sum(mask))
        if k == 0:
            break
        h[mask] += d / k
    return np.clip(h, 0.0, 1.0)

def _grad_half(h, w_half):
    n = len(h)
    g = np.full(n, np.sum(w_half))
    nz = np.nonzero(w_half)[0]
    for k in nz:
        if k == 0:
            continue
        wk = w_half[k]
        g[k:] -= wk * h[:n-k]
        g[:n-k] -= wk * h[k:]
    if w_half[0] != 0:
        g -= 2.0 * w_half[0] * h
    return g

import numpy as np
from scipy.optimize import minimize
import time

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def _v_half(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return c[n - 1:] * dx

def _fix(h, n):
    t = 0.5 * n
    h = h.copy()
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        d = t - np.sum(h)
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        k = int(np.sum(mask))
        if k == 0:
            break
        h[mask] += d / k
    return np.clip(h, 0.0, 1.0)

def _grad_half(h, w_half):
    n = len(h)
    g = np.full(n, np.sum(w_half))
    nz = np.nonzero(w_half)[0]
    for idx in nz:
        k = int(idx)
        if k == 0:
            continue
        wk = w_half[k]
        g[k:] -= wk * h[:n - k]
        g[:n - k] -= wk * h[k:]
    if w_half[0] != 0.0:
        g -= 2.0 * w_half[0] * h
    return g

def _psgd(h, n, max_iters, alpha, K, lr_init, momentum, t0, budget):
    v_mom = np.zeros(n)
    best_h = h.copy()
    best_c = _c5(h)
    h = h.copy()
    for it in range(max_iters):
        if budget - (time.time() - t0) < 3:
            break
        if it > 0 and it % 400 == 0:
            ct = _c5(h)
            if ct < best_c:
                best_c = ct
                best_h = h.copy()
        vh = _v_half(h)
        top_idx = np.argsort(vh)[::-1][:K]
        w_half = np.zeros(n)
        v_top = vh[top_idx]
        w_top = np.exp(alpha * (v_top - v_top[0]))
        w_top /= w_top.sum()
        for i in range(len(top_idx)):
            w_half[top_idx[i]] = w_top[i]
        g = _grad_half(h, w_half)
        g -= np.mean(g)
        lr = lr_init / (1.0 + 0.003 * it)
        v_mom = momentum * v_mom + lr * g
        h = _fix(h - v_mom, n)
    ct = _c5(h)
    if ct < best_c:
        best_c = ct
        best_h = h.copy()
    return best_h, best_c

def _slsqp_polish(h, n, t0, budget):
    if budget - (time.time() - t0) < 30:
        return None
    if n > 250:
        return None
    alpha_s = 40.0

    def _obj(x):
        vh = _v_half(x)
        m = vh.max()
        e = np.exp(np.clip(vh - m, -50, 50) * alpha_s)
        return m + np.log(e.sum()) / (alpha_s * alpha_s)

    def _grad(x):
        vh = _v_half(x)
        m = vh.max()
        e = np.exp(np.clip(vh - m, -50, 50) * alpha_s)
        w = e / e.sum()
        return _grad_half(x, w)

    def _icf(x):
        return np.sum(x) - 0.5 * n

    def _icj(x):
        return np.ones(n)

    try:
        res = minimize(_obj, h, jac=_grad, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': _icf, 'jac': _icj}],
                       options={'maxiter': 300, 'ftol': 1e-12})
        if res.x is not None:
            return _fix(np.clip(res.x, 0.0, 1.0), n)
    except Exception:
        pass
    return None

def _interp_up(h, n_new):
    n_old = len(h)
    xo = np.arange(n_old) * (2.0 / n_old)
    xn = np.arange(n_new) * (2.0 / n_new)
    return _fix(np.clip(np.interp(xn, xo, h), 0.0, 1.0), n_new)

def _make_start(n, seed_val):
    rng2 = np.random.RandomState(seed_val)
    t = np.arange(n)
    h = 0.5 + 0.3 * np.sin(2 * np.pi * t / max(4, n // 8))
    h += 0.15 * np.sin(2 * np.pi * t / max(2, n // 16) + 0.7)
    h += rng2.randn(n) * 0.05
    return _fix(np.clip(h, 0.0, 1.0), n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        h = np.full(45, 0.5)
    n = len(h)
    h = _fix(h, n)
    bh = h.copy()
    bc = _c5(bh)

    # Phase 1: PSD at original resolution with increasing softness
    phases = [
        (5.0, 4, 0.9, 0.85, 4000),
        (12.0, 6, 0.6, 0.85, 4000),
        (25.0, 10, 0.4, 0.88, 4000),
        (50.0, 15, 0.25, 0.90, 3000),
        (100.0, 20, 0.15, 0.92, 3000),
        (200.0, 25, 0.08, 0.93, 3000),
    ]
    for alpha, K, lr, mom, niter in phases:
        if budget_s - (time.time() - t0) < 20:
            break
        ht, ct = _psgd(bh, n, niter, alpha, K, lr, mom, t0, budget_s)
        if ct < bc:
            bh, bc = ht, ct

    # Phase 2: SLSQP polish (only for n <= 250)
    ht = _slsqp_polish(bh, n, t0, budget_s)
    if ht is not None:
        ct = _c5(ht)
        if ct < bc:
            bh, bc = ht, ct

    # Phase 3: Try multiple resolutions
    res_list = [200, 300, 400, 500, 650, 800]
    if n not in res_list:
        res_list.insert(0, n)
    for n_new in res_list:
        if budget_s - (time.time() - t0) < 50:
            break
        hi = _interp_up(bh, n_new)
        ct0 = _c5(hi)
        if ct0 < bc:
            bh, bc = hi, ct0
        for alpha, K, lr, mom, niter in [
            (15.0, 8, 0.5, 0.85, 3000),
            (40.0, 12, 0.3, 0.88, 3000),
            (80.0, 18, 0.15, 0.92, 2500),
        ]:
            if budget_s - (time.time() - t0) < 15:
                break
            ht, ct = _psgd(hi, n_new, niter, alpha, K, lr, mom, t0, budget_s)
            if ct < bc:
                bh, bc = ht, ct
                hi = ht.copy()
        if n_new <= 250:
            ht = _slsqp_polish(bh, n_new, t0, budget_s)
            if ht is not None:
                ct = _c5(ht)
                if ct < bc:
                    bh, bc = ht, ct

    # Phase 4: Structured restarts at lower res
    for n_r in [100, 150, 200]:
        if budget_s - (time.time() - t0) < 35:
            break
        for trial in range(4):
            hr = _make_start(n_r, seed + trial * 7 + 1)
            # Also try perturbation of current best
            hr2 = _interp_up(bh, n_r)
            hr2 = _fix(np.clip(hr2 + rng.randn(n_r) * 0.06, 0, 1), n_r)
            for h_start in [hr, hr2]:
                if budget_s - (time.time() - t0) < 10:
                    break
                for alpha, K, lr, mom, niter in [
                    (20.0, 8, 0.4, 0.85, 2500),
                    (60.0, 15, 0.15, 0.92, 2000),
                ]:
                    ht, ct = _psgd(h_start, n_r, niter, alpha, K, lr, mom, t0, budget_s)
                    if ct < bc:
                        bh, bc = ht, ct
                        h_start = ht.copy()

    # Phase 5: Final fine PSD
    n_final = len(bh)
    for alpha, K, lr, mom, niter in [
        (100.0, 20, 0.05, 0.93, 3000),
        (300.0, 30, 0.02, 0.94, 2500),
    ]:
        if budget_s - (time.time() - t0) < 8:
            break
        ht, ct = _psgd(bh, n_final, niter, alpha, K, lr, mom, t0, budget_s)
        if ct < bc:
            bh, bc = ht, ct

    bh = _fix(bh, len(bh))
    bc = _c5(bh)
    return bh, bc, len(bh)
