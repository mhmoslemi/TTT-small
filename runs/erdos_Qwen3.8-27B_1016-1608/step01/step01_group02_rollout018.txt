```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_c5(h, dx, temp):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    scaled = (corr - m) / temp
    # protect against overflow
    scaled = np.clip(scaled, -500, 500)
    return (m + np.log(np.sum(np.exp(scaled)) / len(corr)) / temp) * dx

def project_to_constraint(h, target):
    h = np.array(h, dtype=float, copy=True)
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        excess = h.sum() - target
        if abs(excess) < 1e-14:
            break
        if excess > 0:
            mask = h > 1e-10
            cnt = mask.sum()
            if cnt == 0:
                break
            step = min(excess / cnt, h[mask].min() - 1e-15)
            if step <= 0:
                break
            h[mask] -= step
        else:
            mask = h < 1.0 - 1e-10
            cnt = mask.sum()
            if cnt == 0:
                break
            step = min(-excess / cnt, (1.0 - h[mask]).min() - 1e-15)
            if step <= 0:
                break
            h[mask] += step
    # Final exact adjustment
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def try_slsqp(h0, dx, target, maxiter=300, ftol=1e-12, use_smooth=False, temp=50.0):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    def eq_fun(x):
        return x.sum() - target
    constraints = [{'type': 'eq', 'fun': eq_fun, 'jac': '2-point'}]
    if use_smooth:
        def obj(x):
            return smooth_c5(x, dx, temp)
    else:
        def obj(x):
            return compute_c5(x, dx)
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_to_constraint(res.x, target)
    except Exception:
        return None
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_eq_target = [0.0]
_sdx = [0.0]
_stemp = [0.0]

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_c5(h, dx, temp):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    scaled = np.clip((corr - m) / temp, -500, 500)
    return (m + np.log(np.sum(np.exp(scaled)) / len(corr)) / temp) * dx

def _eq_fun(x):
    return x.sum() - _eq_target[0]

def _sharp_obj(x):
    return compute_c5(x, _sdx[0])

def _smooth_obj(x):
    return smooth_c5(x, _sdx[0], _stemp[0])

def project_to_constraint(h, target):
    h = np.array(h, dtype=float, copy=True)
    for _ in range(300):
        np.clip(h, 0.0, 1.0, out=h)
        excess = h.sum() - target
        if abs(excess) < 1e-13:
            break
        if excess > 0:
            m = h > 1e-12
            c = int(m.sum())
            if c == 0:
                break
            s = min(excess / c, h[m].min())
            if s <= 1e-14:
                break
            h[m] -= s
        else:
            m = h < 1.0 - 1e-12
            c = int(m.sum())
            if c == 0:
                break
            s = min(-excess / c, 1.0 - h[m].min())
            if s <= 1e-14:
                break
            h[m] += s
    return h

def slsqp_opt(h0, target, use_smooth=False, temp=50.0, maxiter=200, ftol=1e-12):
    n = len(h0)
    _sdx[0] = 2.0 / n
    _stemp[0] = temp
    _eq_target[0] = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun}]
    obj = _smooth_obj if use_smooth else _sharp_obj
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_to_constraint(res.x, target)
    except Exception:
        return None
```
