import numpy as np
from scipy.optimize import minimize
import time

_NG = 400

def c5_eval(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def to_binary(h):
    n = len(h)
    h2 = (h >= 0.5).astype(float)
    n1 = int(h2.sum())
    tgt = n // 2
    if n1 > tgt:
        idx = np.where(h2 == 1)[0]
        order = np.argsort(h[idx])
        for k in range(n1 - tgt):
            h2[idx[order[k]]] = 0.0
    elif n1 < tgt:
        idx = np.where(h2 == 0)[0]
        order = np.argsort(-h[idx])
        for k in range(tgt - n1):
            h2[idx[order[k]]] = 1.0
    return h2

def fix_integral(h):
    n = len(h)
    tgt = 0.5 * n
    for _ in range(30):
        d = tgt - h.sum()
        if abs(d) < 1e-12:
            break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        c = int(m.sum())
        if c == 0:
            break
        h = np.clip(h + (d / c) * m, 0.0, 1.0)
    return h

def interp_h(h, n):
    n0 = len(h)
    xo = np.linspace(0, 2, n0, endpoint=False)
    xn = np.linspace(0, 2, n, endpoint=False)
    return fix_integral(np.clip(np.interp(xn, xo, h), 0, 1))

def sa_binary(h0, n_steps, t0, budget_s, seed, T0, cool):
    rng = np.random.RandomState(seed)
    h = h0.copy()
    n = len(h)
    best = h.copy()
    best_c = c5_eval(best)
    cur_c = best_c
    T = T0
    for s in range(n_steps):
        if time.time() - t0 > budget_s - 25:
            break
        if s > 0 and s % 8000 == 0:
            T *= cool
            if T < 1e-5:
                T = 1e-5
        i = rng.randint(n)
        j = rng.randint(n)
        if h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        nc = c5_eval(h)
        if nc <= cur_c or rng.random() < np.exp(-(nc - cur_c) / T):
            cur_c = nc
            if nc < best_c:
                best_c = nc
                best = h.copy()
        else:
            h[j], h[i] = h[i], h[j]
    return best, best_c

def local_search(h0, max_rounds, t0, budget_s):
    h = h0.copy()
    n = len(h)
    best = h.copy()
    best_c = c5_eval(best)
    for _ in range(max_rounds):
        if time.time() - t0 > budget_s - 30:
            break
        ov = np.correlate(h, 1.0 - h, mode='full')
        imax = int(np.argmax(ov))
        wl = imax - (n - 1)
        if wl >= 0:
            k = wl
        else:
            k = -wl
        if k == 0:
            idx_src = np.where(h < 0.5)[0]
        elif wl > 0:
            src_m = (h[:n-k] > 0.5) & (h[k:] < 0.5)
            idx_src = np.where(src_m)[0]
        else:
            src_m = (h[k:] > 0.5) & (h[:n-k] < 0.5)
            idx_src = np.where(src_m)[0] + k
        idx_tgt = np.where(h < 0.5)[0]
        if len(idx_src) == 0 or len(idx_tgt) == 0:
            break
        improved = False
        cur_max = float(np.max(ov))
        n_try = min(len(idx_src), 30)
        n_tgt = min(len(idx_tgt), 30)
        best_new = cur_max
        best_ij = None
        for ii in range(n_try):
            i = idx_src[ii]
            h[i] = 0.0
            for jj in range(n_tgt):
                j = idx_tgt[jj]
                if j == i:
                    continue
                h[j] = 1.0
                nm = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
                if nm < best_new:
                    best_new = nm
                    best_ij = (i, j)
                h[j] = 0.0
            h[i] = 1.0
        if best_ij is not None:
            i, j = best_ij
            h[i], h[j] = h[j], h[i]
            nc = c5_eval(h)
            if nc < best_c:
                best_c = nc
                best = h.copy()
            improved = True
        else:
            break
    return best, best_c

def _slsqp_obj(x, alpha):
    n = _NG
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = float(np.max(c))
    e = np.exp(np.clip(c - m, -50.0, 50.0))
    return m + np.log(np.sum(e)) / alpha

def _slsqp_f(x):
    return np.sum(x) - _NG * 0.5

def _slsqp_j(x):
    return np.ones(_NG)

def run(seed=42, budget_s=1100, **kwargs):
    global _NG
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_start = np.array(initial_h_values, dtype=float).copy()
    else:
        h_start = np.full(400, 0.5)
    n = len(h_start)
    _NG = n

    hb = to_binary(h_start)
    cb = c5_eval(hb)
    hc = fix_integral(h_start.copy())
    cc = c5_eval(hc)
    best_h = hb.copy() if cb <= cc else hc.copy()
    best_c = min(cb, cc)

    # Phase 1: Local search on binary
    lh, lc = local_search(hb, 80, t0, budget_s)
    if lc < best_c:
        best_c, best_h = lc, lh

    # Phase 2: Simulated annealing - multiple restarts
    n_restarts = 3
    for r in range(n_restarts):
        if time.time() - t0 > budget_s - 80:
            break
        if r == 0:
            start = hb.copy()
        else:
            start = to_binary(fix_integral(best_h + rng.randn(n) * 0.15))
        sh, sc = sa_binary(start, 800000, t0, budget_s, seed + r * 104729, 0.008, 0.99995)
        if sc < best_c:
            best_c, best_h = sc, sh

    # Phase 3: More local search after SA
    lh2, lc2 = local_search(best_h.copy(), 100, t0, budget_s)
    if lc2 < best_c:
        best_c, best_h = lc2, lh2

    # Phase 4: Continuous polish via SLSQP
    if time.time() - t0 < budget_s - 40:
        h_cont = best_h.copy()
        h_cont[rng.randint(n, size=n // 4)] = rng.uniform(0, 1, n // 4)
        h_cont = fix_integral(np.clip(h_cont, 0, 1))
        bnds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _slsqp_f, 'jac': _slsqp_j}]
        for alpha in [200.0, 800.0]:
            if time.time() - t0 > budget_s - 15:
                break
            try:
                res = minimize(_slsqp_obj, h_cont, (alpha,), method='SLSQP',
                               bounds=bnds, constraints=cons,
                               options={'maxiter': 800, 'ftol': 1e-13})
                if res.x is not None:
                    ht = fix_integral(np.clip(res.x, 0, 1))
                    ct = c5_eval(ht)
                    if ct < best_c:
                        best_c, best_h = ct, ht
                        h_cont = ht.copy()
            except Exception:
                pass

    # Phase 5: Try alternate resolution
    for nn in [200, 600]:
        if time.time() - t0 > budget_s - 30:
            break
        _NG = nn
        hi = interp_h(best_h, nn)
        hi_b = to_binary(hi)
        ci = c5_eval(hi_b)
        if ci < best_c:
            hi_b2, ci2 = sa_binary(hi_b, 200000, t0, budget_s, seed + 999, 0.006, 0.9999)
            if ci2 < best_c:
                best_c, best_h = ci2, hi_b2
                _NG = nn
        _NG = n

    # Final: ensure exact integral and re-evaluate
    best_h = fix_integral(best_h)
    best_c = c5_eval(best_h)
    return best_h, best_c, len(best_h)
