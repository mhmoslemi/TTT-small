import numpy as np
from scipy.optimize import minimize
import time

_GN = 400

def _g_sum(x):
    return np.sum(x) - 0.5 * _GN

def _g_jac(x):
    return np.ones(_GN)

def _g_obj(x, alpha):
    n = _GN
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(c)
    return m + np.log(np.sum(np.exp(np.clip(c - m, -50.0, 50.0)))) / alpha

def c5_eval(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def fix_sum(h, n):
    target = 0.5 * n
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        d = target - np.sum(h)
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        k = np.sum(mask)
        if k == 0:
            break
        h[mask] += d / k
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_h(h, n_new):
    n_old = len(h)
    xo = np.arange(n_old) * (2.0 / n_old)
    xn = np.arange(n_new) * (2.0 / n_new)
    hr = np.interp(xn, xo, h)
    return fix_sum(np.clip(hr, 0.0, 1.0), n_new)

def slsqp_opt(h, n, alpha, maxiter, t0, budget_s):
    global _GN
    if time.time() - t0 > budget_s - 25:
        return h
    _GN = n
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _g_sum, 'jac': _g_jac}]
    try:
        res = minimize(_g_obj, h, (alpha,), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None:
            return fix_sum(np.clip(res.x, 0.0, 1.0), n)
    except Exception:
        pass
    return h

def coord_descend(h, n, n_passes=20, t0=None, budget_s=None):
    dx = 2.0 / n
    h_work = h.copy()
    h_flip = h_work[::-1].copy()
    c = np.correlate(h_work, 1.0 - h_work, mode='full') * dx
    for p in range(n_passes):
        if t0 is not None and time.time() - t0 > budget_s - 15:
            break
        for i in range(n):
            old = h_work[i]
            if old < 1e-15 or old > 1.0 - 1e-15:
                continue
            s = np.zeros(2 * n - 1)
            s[i:i+n] += (1.0 - h_flip)
            s[n-1-i:2*n-1-i] -= h_work
            lo = -old
            hi = 1.0 - old
            for _ in range(45):
                mid = (lo + hi) * 0.5
                vals = c + mid * s
                kstar = np.argmax(vals)
                sl = s[kstar]
                if sl > 1e-13:
                    hi = mid
                elif sl < -1e-13:
                    lo = mid
                else:
                    lo = hi = mid
                    break
            bt = (lo + hi) * 0.5
            new_val = np.clip(old + bt, 0.0, 1.0)
            at = new_val - old
            if abs(at) > 1e-15:
                h_work[i] = new_val
                h_flip = h_work[::-1].copy()
                c += at * s
        h_work = fix_sum(h_work, n)
        h_flip = h_work[::-1].copy()
        c = np.correlate(h_work, 1.0 - h_work, mode='full') * dx
    return h_work

def subgrad_polish(h, n, n_iters=200, t0=None, budget_s=None):
    dx = 2.0 / n
    h_work = h.copy()
    for it in range(n_iters):
        if t0 is not None and time.time() - t0 > budget_s - 10:
            break
        c = np.correlate(h_work, 1.0 - h_work, mode='full') * dx
        kstar = int(np.argmax(c))
        m = kstar - (n - 1)
        g = np.zeros(n)
        for i in range(n):
            ip = i - m
            iq = i + m
            val = 0.0
            if 0 <= ip < n:
                val += (1.0 - h_work[ip])
            if 0 <= iq < n:
                val -= h_work[iq]
            g[i] = val
        eta = 0.02 / (np.max(np.abs(g)) + 1e-12)
        h_work = np.clip(h_work - eta * g, 0.0, 1.0)
        h_work = fix_sum(h_work, n)
    return h_work

def run(seed=42, budget_s=1100, **kwargs):
    global _GN
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        h = np.full(400, 0.5)
    n = len(h)
    h = fix_sum(h, n)
    best_h = h.copy()
    best_c = c5_eval(best_h)

    # Phase 1: SLSQP with increasing alpha
    for alpha in [80.0, 300.0, 1000.0, 3000.0]:
        if time.time() - t0 > budget_s - 120:
            break
        h_try = slsqp_opt(best_h, n, alpha, 1500, t0, budget_s)
        if h_try is not None:
            ct = c5_eval(h_try)
            if ct < best_c - 1e-10:
                best_c = ct
                best_h = h_try.copy()

    # Phase 2: Coordinate descent (exact max)
    for rnd in range(5):
        if time.time() - t0 > budget_s - 60:
            break
        h_work = coord_descend(best_h, n, n_passes=40, t0=t0, budget_s=budget_s)
        ct = c5_eval(h_work)
        if ct < best_c - 1e-12:
            best_c = ct
            best_h = h_work

    # Phase 3: Subgradient polish on active lag
    for rnd in range(3):
        if time.time() - t0 > budget_s - 40:
            break
        h_work = subgrad_polish(best_h, n, n_iters=300, t0=t0, budget_s=budget_s)
        ct = c5_eval(h_work)
        if ct < best_c - 1e-12:
            best_c = ct
            best_h = h_work

    # Phase 4: Multi-resolution search
    for nn in [200, 300, 500, 600]:
        if time.time() - t0 > budget_s - 80:
            break
        hi = interp_h(best_h, nn)
        hi = slsqp_opt(hi, nn, 500.0, 800, t0, budget_s)
        hi = coord_descend(hi, nn, n_passes=30, t0=t0, budget_s=budget_s)
        ct = c5_eval(hi)
        if ct < best_c - 1e-10:
            best_c = ct
            best_h = hi.copy()

    # Phase 5: Final CD + subgrad at best resolution
    nb = len(best_h)
    if time.time() - t0 < budget_s - 30:
        h_work = coord_descend(best_h, nb, n_passes=60, t0=t0, budget_s=budget_s)
        ct = c5_eval(h_work)
        if ct < best_c - 1e-12:
            best_c = ct
            best_h = h_work
    if time.time() - t0 < budget_s - 20:
        h_work = subgrad_polish(best_h, nb, n_iters=400, t0=t0, budget_s=budget_s)
        ct = c5_eval(h_work)
        if ct < best_c - 1e-12:
            best_c = ct
            best_h = h_work

    # Final: one more CD pass
    if time.time() - t0 < budget_s - 5:
        h_work = coord_descend(best_h, nb, n_passes=10, t0=t0, budget_s=budget_s)
        ct = c5_eval(h_work)
        if ct < best_c:
            best_c = ct
            best_h = h_work

    best_h = fix_sum(best_h, len(best_h))
    best_c = c5_eval(best_h)
    return best_h, best_c, len(best_h)
