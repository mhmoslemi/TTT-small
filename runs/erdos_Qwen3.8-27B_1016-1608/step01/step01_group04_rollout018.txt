    best_c = float(np.max(corr))
    
    # Precompute factor matrices
    ks = np.arange(2*n-1)  # 0 to 2n-2
    P = np.arange(n)[:, None]  # n×1
    K = ks[None, :]  # 1×(2n-1)
    
    # first_factor[p, k] = 1 - h[p-k+n-1] if valid
    idx1 = P - K + n - 1  # n × (2n-1)
    mask1 = (idx1 >= 0) & (idx1 < n)
    val1 = np.zeros_like(idx1, dtype=float)
    val1[mask1] = 1 - h[idx1[mask1]]
    
    # second_factor[p, k] = h[p+k-n+1] if valid
    idx2 = P + K - n + 1
    mask2 = (idx2 >= 0) & (idx2 < n)
    val2 = np.zeros_like(idx2, dtype=float)
    val2[mask2] = h[idx2[mask2]]
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(max_iters):
        if time.time() - t0 > time_limit - 5:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break

        # delta = val1[j,:] - val1[i,:] + val2[i,:] - val2[j,:]
        # new_corr = corr + delta
        # new_max = max(new_corr)
        
        # Vectorize over all pairs
        # ones: (n1,), zeros: (n2,)
        # val1[j,:] - val1[i,:] for all i in ones, j in zeros: shape (n1, n2, 2n-1)
        # This might be too big for large n, but for n=80: 40*40*159 = 254400, fine
        
        v1_o = val1[ones]  # (n1, 2n-1)
        v1_z = val1[zeros]  # (n2, 2n-1)
        v2_o = val2[ones]
        v2_z = val2[zeros]

        new_corr = corr[None, None, :] + delta
        new_max = np.max(new_corr, axis=2)  # (n1, n2)

        best_val = new_max[best_idx]

            h[j] = 1

            best_c = float(np.max(corr))

            val1[mask1] = 1 - h[idx1[mask1]]
            val2[mask2] = h[idx2[mask2]]

            zeros = np.where(h == 0)[0]
        else:
            # Stuck - try random swap or perturb
            # Just break and rely on external restart
            break
    
    return h, best_c
