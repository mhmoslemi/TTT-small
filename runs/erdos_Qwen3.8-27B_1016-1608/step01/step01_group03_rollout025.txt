def compute_grad(h, weights):
    """
    h: array of length n
    weights: array of length 2n-1, corresponding to shifts -(n-1) to (n-1)
    Returns gradient of sum_k weights[k] * c_k
    """
    n = len(h)
    grad = np.zeros(n)
    for k in range(-(n-1), n):
        w = weights[k + n - 1]
        if abs(w) < 1e-15:
            continue
        # d(c_k)/d(h[i]) = (1-h[i-k])*1[0<=i-k<n] - h[i+k]*1[0<=i+k<n]
        for i in range(n):
            val = 0.0
            if 0 <= i-k < n:
                val += 1.0 - h[i-k]
            if 0 <= i+k < n:
                val -= h[i+k]
            grad[i] += w * val
    return grad

        if abs(w) < 1e-15:

            grad[k:] += w * (1 - h[:n-k])

            grad[:n-k] -= w * h[k:]

            grad[:n-t] += w * (1 - h[t:])
            # i+k = i-t valid: i from t to n-1
            grad[t:] -= w * h[:n-t]
    return grad

np.correlate(h, 1-h, mode="full") computes:
For arrays a=h (length n) and v=1-h (length n):
result[k] = sum_{t} a[t] * v[t + k - (n-1)]

where t ranges over values where both indices are valid (0 to n-1).

So result[k] = sum_{t} h[t] * (1-h[t + k - (n-1)])

Let s = k - (n-1). Then result[n-1+s] = sum_t h[t] * (1-h[t+s])

For this to be valid, we need 0 ≤ t < n and 0 ≤ t+s < n.

d(result[n-1+s])/d(h[i]):
The term h[i] appears:
1. As h[t] when t=i: contributes (1-h[i+s]) if 0≤i+s<n
2. In (1-h[t+s]) when t+s=i, i.e., t=i-s: contributes -h[i-s] if 0≤i-s<n

So d/d(h[i]) = (1-h[i+s])·1[0≤i+s<n] - h[i-s]·1[0≤i-s<n]

In my compute_grad, I'm iterating over shift s and computing:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    h = np.clip(h, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def repair(h, target):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(300):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-13:
            break
        if d > 0:
            m = h < 1.0 - 1e-14
        else:
            m = h > 1e-14
        c = int(m.sum())
        if c == 0:
            break
        h[m] += d / c
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_grad(h, weights):
    n = len(h)
    grad = np.zeros(n)
    for shift in range(-(n - 1), n):
        w = weights[shift + n - 1]
        if abs(w) < 1e-14:
            continue
        if shift >= 0:
            grad[shift:] += w * (1.0 - h[:n - shift])
            grad[:n - shift] -= w * h[shift:]
        else:
            t = -shift
            grad[:n - t] += w * (1.0 - h[t:])
            grad[t:] -= w * h[:n - t]
    return grad

def softmax_weights(corr_raw, beta):
    m = np.max(corr_raw)
    e = np.exp(beta * (corr_raw - m))
    sw = np.sum(e)
    if sw < 1e-300:
        w = np.zeros_like(corr_raw)
        w[np.argmax(corr_raw)] = 1.0
        return w
    return e / sw

def optimize_from(h0, n, dx, target, t0, budget_s, max_iter=500):
    h = repair(h0.copy(), target)
    best_c = compute_c5(h, dx)
    best_h = h.copy()
    beta = 60.0
    alpha = 0.3
    stall = 0
    for it in range(max_iter):
        if time.time() - t0 > budget_s - 25:
            break
        hc = np.clip(h, 0.0, 1.0)
        corr_raw = np.correlate(hc, 1.0 - hc, mode="full")
        w = softmax_weights(corr_raw, beta)
        g = compute_grad(hc, w)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        g = g / gnorm
        improved = False
        a = alpha
        current_c = compute_c5(hc, dx)
        for _ in range(20):
            hn = repair(hc - a * g, target)
            cn = compute_c5(hn, dx)
            if cn < current_c - 1e-16:
                h = hn
                if cn < best_c:
                    best_c = cn
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
                improved = True
                break
            a *= 0.5
        if not improved:
            stall += 3
            h = repair(hc + 0.015 * np.random.randn(n), target)
        beta = max(12.0, beta * 0.997)
        alpha = max(0.03, alpha * 0.9995)
        if stall > 150:
            h = repair(best_h + 0.04 * np.random.randn(n), target)
            stall = 0
    return best_h, best_c
