def projection(h, target):
    h = np.array(h, dtype=float, copy=True)
    for _ in range(1000):
        h = np.clip(h, 0.0, 1.0)
        excess = h.sum() - target
        if abs(excess) < 1e-12:
            return h
        if excess > 0:
            mask = h > 0
            if not mask.any():
                return h
            s = min(excess / mask.sum(), h[mask].min())
            h[mask] -= s
        else:
            mask = h < 1
            if not mask.any():
                return h
            s = min(-excess / mask.sum(), (1.0 - h[mask]).min())
            h[mask] += s
    return h

                improved = False
                for delta in [ss, -ss, ss * 0.5, -ss * 0.5]:
                    h_t = h_cd.copy()
                    h_t[idx] = np.clip(h_t[idx] + delta, 0, 1)
                    h_t = projection(h_t, target)
                    c5_t = compute_c5(h_t, dx)
                    if c5_t < c5_cd - 1e-15:
                        h_cd, c5_cd = h_t, c5_t
                        improved = True
                        if c5_t < best_c5:
                            best_h = h_cd.copy()
                            best_c5 = c5_t
                        break
                if improved:
                    stuck = 0
                else:
                    stuck += 1
                if stuck > 300:
                    break
            # nudge with small random perturbation
            h_cd = projection(h_cd + rng.randn(n) * 0.01, target)
            c5_cd = compute_c5(h_cd, dx)
            if c5_cd < best_c5:
                best_h = h_cd.copy()
                best_c5 = c5_cd

    # Phase 7: Final exact SLSQP polish (multiple attempts)
    for _ in range(4):
        if remaining() < 8:
            break
        h_new = try_slsqp_exact(best_h, dx, target, 400)
        try_update(h_new)

    # Phase 8: One more L-BFGS-B at very high temperature then back down
    if remaining() > 15:
        h_new = try_lbfgs(best_h, dx, target, 3.0, 800)
        try_update(h_new)
        if remaining() > 8:
            h_new = try_lbfgs(best_h, dx, target, 1.0, 500)
            try_update(h_new)

    best_h = projection(best_h, target)
    final_c5 = compute_c5(best_h, dx)
    return (best_h, final_c5, n)
