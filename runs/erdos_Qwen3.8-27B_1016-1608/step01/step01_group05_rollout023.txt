```python
def gilbert_grad(h, n, s):
    grad = np.zeros(n)
    # Term 1: (1-h[j+s]) for valid j
    j_min1 = max(0, -s)
    j_max1 = min(n, n - s)
    if j_min1 < j_max1:
        grad[j_min1:j_max1] += (1.0 - h[j_min1+s:j_max1+s])
    # Term 2: -h[j-s] for valid j
    j_min2 = max(0, s)
    j_max2 = min(n, n + s)
    if j_min2 < j_max2:
        grad[j_min2:j_max2] -= h[j_min2-s:j_max2-s]
    return grad
```

```python
    return np.interp(x_new, x_old, h)


def gilbert_grad(h, n, s):
    grad = np.zeros(n)
    j1, j2 = max(0, -s), min(n, n - s)
    if j1 < j2:
        grad[j1:j2] += (1.0 - h[j1 + s:j2 + s])
    j3, j4 = max(0, s), min(n, n + s)
    if j3 < j4:
        grad[j3:j4] -= h[j3 - s:j4 - s]
    return grad


def gilbert_optimize(h0, n, max_iter=800, top_k=3):
    target = n / 2.0
    h = project_to_set(h0.astype(float).copy(), target)
    best = compute_c5(h, n)
    alpha = 0.04
    for it in range(max_iter):
        corr = compute_corr(h, n)
        idx = np.argsort(corr)[::-1][:top_k]
        grad = np.zeros(n)
        for k in idx:
            s = int(k - (n - 1))
            grad += gilbert_grad(h, n, s)
        grad /= float(top_k)
        h_new = project_to_set(h - alpha * grad, target)
        c_new = compute_c5(h_new, n)
        if c_new < best - 1e-15:
            h = h_new
            best = c_new
            alpha = min(alpha * 1.08, 0.2)
        else:
            alpha *= 0.75
            if alpha < 1e-8:
                break
    return h, best


def lbfbs_smooth(h0, n, beta=35.0, maxiter=120):
    target = n / 2.0
    h0 = project_to_set(h0.astype(float).copy(), target)

    def _obj(x):
        xp = project_to_set(x, target)
        corr = compute_corr(xp, n)
        m = np.max(corr)
        return m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta

    try:
        res = minimize(_obj, h0, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-9})
        h_out = project_to_set(res.x, target)
        return h_out, compute_c5(h_out, n)
    except Exception:
        return h0, compute_c5(h0, n)


def swap_search(h, n, rng, n_iter=30000):
    h = h.copy()
    best = compute_c5(h, n)
    for _ in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        old_i, old_j = h[i], h[j]
        if h[i] >= 0.5:
            max_d = min(1.0 - h[i], h[j])
            if max_d < 1e-10:
                continue
            d = rng.uniform(0.01, max_d)
            h[i] += d
            h[j] -= d
        else:
            max_d = min(h[i], 1.0 - h[j])
            if max_d < 1e-10:
                continue
            d = rng.uniform(0.01, max_d)
            h[i] -= d
            h[j] += d
        c = compute_c5(h, n)
        if c >= best - 1e-15:
            h[i], h[j] = old_i, old_j
        else:
            best = c
    return h, best


def make_random_binary(n, rng):
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h


def make_perturbed(h_base, n, rng, sigma=0.08):
    h = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, len(h_base)), h_base)
    h += rng.normal(0, sigma, n)
    return project_to_set(h, n / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    global _budget, _start_t
    _budget = float(budget_s)
    _start_t = time.time()
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = np.inf
    best_n = 0

    # Phase 1: n=90
    n1 = 90
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        if len(h_init) != n1:
            h_init = upsample(h_init, n1)
    except Exception:
        h_init = make_random_binary(n1, rng)
    h_init = project_to_set(h_init, n1 / 2.0)
    c = compute_c5(h_init, n1)
    best_h, best_c5, best_n = h_init.copy(), c, n1

    # Gilbert with various top_k
    for tk in [1, 3, 5]:
        h_g, c_g = gilbert_optimize(h_init, n1, max_iter=500, top_k=tk)
        if c_g < best_c5:
            best_h, best_c5, best_n = h_g.copy(), c_g, n1

    # L-BFGS-B soft-max
    if _time_left() > _budget * 0.65:
        h_l, c_l = lbfbs_smooth(best_h, n1, beta=35, maxiter=100)
        if c_l < best_c5:
            best_h, best_c5, best_n = h_l.copy(), c_l, n1

    # More Gilbert after L-BFGS
    if _time_left() > _budget * 0.55:
        for tk in [2, 4, 1]:
            h_g, c_g = gilbert_optimize(best_h, n1, max_iter=400, top_k=tk)
            if c_g < best_c5:
                best_h, best_c5, best_n = h_g.copy(), c_g, n1

    # Swap search
    if _time_left() > _budget * 0.40:
        h_s, c_s = swap_search(best_h, n1, rng, n_iter=40000)
        if c_s < best_c5:
            best_h, best_c5, best_n = h_s.copy(), c_s, n1

    # Random restarts at n=90
    restarts = 0
    while _time_left() > _budget * 0.25 and restarts < 12:
        h_r = make_random_binary(n1, rng)
        h_r += rng.normal(0, 0.15, n1)
        h_r = project_to_set(h_r, n1 / 2.0)
        h_o, c_o = gilbert_optimize(h_r, n1, max_iter=300, top_k=3)
        h_o2, c_o2 = gilbert_optimize(h_o, n1, max_iter=200, top_k=1)
        h_s2, c_s2 = swap_search(h_o2, n1, rng, n_iter=8000)
        if c_s2 < best_c5:
            best_h, best_c5, best_n = h_s2.copy(), c_s2, n1
        restarts += 1

    # Perturbed restarts from best
    for sigma in [0.05, 0.1, 0.2]:
        if _time_left() < _budget * 0.25:
            break
        h_p = make_perturbed(best_h, n1, rng, sigma)
        h_o, c_o = gilbert_optimize(h_p, n1, max_iter=400, top_k=3)
        h_s2, c_s2 = swap_search(h_o, n1, rng, n_iter=10000)
        if c_s2 < best_c5:
            best_h, best_c5, best_n = h_s2.copy(), c_s2, n1

    # Phase 2: n=180
    if _time_left() > _budget * 0.22:
        n2 = 180
        h_up = project_to_set(upsample(best_h, n2), n2 / 2.0)
        c_u = compute_c5(h_up, n2)
        if c_u < best_c5:
            best_h, best_c5, best_n = h_up.copy(), c_u, n2
        for tk in [3, 5]:
            if _time_left() < _budget * 0.22:
                break
            h_g, c_g = gilbert_optimize(h_up, n2, max_iter=350, top_k=tk)
            if c_g < best_c5:
                best_h, best_c5, best_n = h_g.copy(), c_g, n2
        if _time_left() > _budget * 0.15:
            h_l, c_l = lbfbs_smooth(best_h, n2, beta=30, maxiter=60)
            if c_l < best_c5:
                best_h, best_c5, best_n = h_l.copy(), c_l, n2
        if _time_left() > _budget * 0.12:
            h_s, c_s = swap_search(best_h, best_n, rng, n_iter=25000)
            if c_s < best_c5:
                best_h, best_c5, best_n = h_s.copy(), c_s, n2

    # Phase 3: n=360 (brief)
    if _time_left() > _budget * 0.12:
        n3 = 360
        h_up3 = project_to_set(upsample(best_h, n3), n3 / 2.0)
        h_g3, c_g3 = gilbert_optimize(h_up3, n3, max_iter=200, top_k=3)
        if c_g3 < best_c5:
            best_h, best_c5, best_n = h_g3.copy(), c_g3, n3

    # Final polish
    if _time_left() > 15.0:
        h_s, c_s = swap_search(best_h, best_n, rng, n_iter=20000)
        if c_s < best_c5:
            best_h, best_c5, best_n = h_s.copy(), c_s, best_n
    if _time_left() > 10.0:
        h_g, c_g = gilbert_optimize(best_h, best_n, max_iter=300, top_k=2)
        if c_g < best_c5:
            best_h, best_c5, best_n = h_g.copy(), c_g, best_n

    best_h = project_to_set(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h, best_n)
    return best_h, best_c5, best_n
```
