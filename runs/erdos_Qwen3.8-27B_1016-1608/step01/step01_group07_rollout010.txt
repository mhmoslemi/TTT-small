            break

        else:

            break

def _grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    # grad[p, s_idx] = dc_s/dh_p * dx where s_idx = s + (n-1)
    grad = np.zeros((n, 2*n-1))

            # p as second index: n=p, n+s=p+s
            if 0 <= p + s < n:
                g -= h[p + s]
            grad[p, s + n - 1] = g * dx
    return grad

def _minimize_max_lines(a, b, lo=0.0, hi=1.0):
    """Minimize max_s (a[s]*x + b[s]) for x in [lo, hi]."""
    g_lo = np.max(a * lo + b)
    g_hi = np.max(a * hi + b)
    xlo, xhi = lo, hi
    for _ in range(50):
        m1 = xlo + (xhi - xlo) / 3.0
        m2 = xhi - (xhi - xlo) / 3.0
        fm1 = np.max(a * m1 + b)
        fm2 = np.max(a * m2 + b)
        if fm1 < fm2:
            xhi = m2
        else:
            xlo = m1
    return (xlo + xhi) / 2.0
