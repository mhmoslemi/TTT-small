    best_h = h.copy()

        t = it / max(1, n_iters - 1)
        T = T_init * (T_end / T_init) ** t
        
        i, j = rng.choice(n, 2, replace=False)
        # Determine max valid delta
        max_pos = min(h[i], 1.0 - h[j])  # move mass from i to j
        max_neg = min(1.0 - h[i], h[j])  # move mass from j to i
        # Random delta in [-max_neg, max_pos]
        d = rng.uniform(-max_neg, max_pos)
        if abs(d) < 1e-8:
            continue
        
        h[i] -= d  # h[i] += -d... wait
        # If d > 0: move from i to j: h[i] -= d, h[j] += d
        # If d < 0: move from j to i: h[i] += |d|, h[j] -= |d|
        # Unify: h[i] -= d, h[j] += d
        # Check: if d > 0, h[i] -= d ≤ h[i] - 0 = h[i], OK if d ≤ h[i]
        #          h[j] += d ≤ h[j] + (1-h[j]) = 1, OK if d ≤ 1-h[j]

        # If d < 0: h[i] -= d = h[i] + |d| ≤ h[i] + (1-h[i]) = 1, OK
        #            h[j] += d = h[j] - |d| ≥ h[j] - h[j] = 0, OK
        # Great! Sum is preserved.
        
        h[i] -= d
        h[j] += d
        
        new_c5 = compute_c5(h, n)
        delta = new_c5 - cur_c5
        
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()

            h[j] -= d
