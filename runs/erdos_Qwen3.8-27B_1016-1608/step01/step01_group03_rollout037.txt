    tgt0 = n0 / 2.0
    dx0 = 2.0 / n0
    h_init = repair(h_init, tgt0)

    best_c5 = compute_c5(h_init, dx0)
    best_h = h_init.copy()
    best_n = n0

    # Phase 1: Annealed smooth max SLSQP on original grid
    h_curr = best_h.copy()
    betas1 = [10, 30, 100, 300, 1000, 3000, 10000]
    for beta in betas1:
        if elapsed() > budget_s - 250:
            break
        try:
            h_new = _slsqp_smooth(h_curr, dx0, tgt0, beta, maxiter=500, use_analytical_jac=True)
            h_new = repair(h_new, tgt0)
            c5_new = compute_c5(h_new, dx0)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_curr = h_new
        except Exception:
            pass

    # Phase 2: Subgradient polishing on exact max
    if elapsed() < budget_s - 200:
        h_sg = best_h.copy()
        for lr in [0.05, 0.02, 0.01, 0.005]:
            if elapsed() > budget_s - 150:
                break
            try:
                h_sg = _subgrad_step(h_sg, dx0, tgt0, lr, n_steps=80)
                c5_sg = compute_c5(h_sg, dx0)
                if c5_sg < best_c5:
                    best_c5 = c5_sg
                    best_h = h_sg.copy()
            except Exception:
                pass

    # Phase 3: Multi-start SLSQP from perturbed best
    if elapsed() < budget_s - 150:
        for i in range(10):
            if elapsed() > budget_s - 130:
                break
            scale = 0.03 * (1.0 - i / 10.0) + 0.005
            h_pert = repair(best_h + np.random.randn(n0) * scale, tgt0)
            try:
                h_ns = _slsqp_smooth(h_pert, dx0, tgt0, 3000, maxiter=400, use_analytical_jac=True)
                h_ns = repair(h_ns, tgt0)
                c5_ns = compute_c5(h_ns, dx0)
                if c5_ns < best_c5:
                    best_c5 = c5_ns
                    best_h = h_ns.copy()
            except Exception:
                pass

    # Phase 4: Interpolate to larger grid and optimize
    n_trials = [int(n0 * 1.5), n0 * 2, n0 + 20]
    n_trials = sorted(set(x for x in n_trials if 50 <= x <= 280))
    for n_trial in n_trials:
        if elapsed() > budget_s - 150:
            break
        nx = n_trial
        tx = 2.0 / nx
        tgt_x = nx / 2.0
        x_old = np.arange(n0) * dx0
        x_new = np.arange(nx) * tx
        # Wrap-around interpolation for periodic structure
        h_ext = np.concatenate([best_h[-5:], best_h, best_h[:5]])
        x_ext = np.concatenate([np.arange(-5, 0) * dx0, x_old, np.arange(n0, n0 + 5) * dx0])
        h_interp = np.interp(x_new, x_ext, h_ext)
        h_interp = repair(h_interp, tgt_x)
        c5_int = compute_c5(h_interp, tx)
        h_curr_x = h_interp.copy()

        if c5_int < best_c5:
            best_c5 = c5_int
            best_h = h_curr_x.copy()
            best_n = nx

        for beta in [50, 200, 1000, 5000]:
            if elapsed() > budget_s - 80:
                break
            try:
                mi = 300 if nx <= 120 else 200
                h_new = _slsqp_smooth(h_curr_x, tx, tgt_x, beta, maxiter=mi, use_analytical_jac=True)
                h_new = repair(h_new, tgt_x)
                c5_new = compute_c5(h_new, tx)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    best_n = nx
                h_curr_x = h_new
            except Exception:
                pass

    # Phase 5: Final intensive polish on best solution
    if elapsed() < budget_s - 30:
        nb = len(best_h)
        dxb = 2.0 / nb
        tgtb = nb / 2.0
        for beta_fin in [10000, 50000, 200000]:
            if elapsed() > budget_s - 20:
                break
            try:
                h_fin = _slsqp_smooth(best_h, dxb, tgtb, beta_fin, maxiter=600, use_analytical_jac=True)
                h_fin = repair(h_fin, tgtb)
                c5_fin = compute_c5(h_fin, dxb)
                if c5_fin < best_c5:
                    best_c5 = c5_fin
                    best_h = h_fin.copy()
                    best_n = nb
            except Exception:
                pass

    # Final validation
    nf = len(best_h)
    best_h = repair(best_h, nf / 2.0)
    best_c5 = compute_c5(best_h, 2.0 / nf)

    return (best_h, best_c5, nf)
