        if time.time() < t_start + budget_s * 0.98:
            h_lp = solve_lp_left(h_snap, n, dx)
            if h_lp is not None:
                h_lp = project_feasible(h_lp, n)
                c5_lp = compute_c5(h_lp)
                if c5_lp < best_c5:
                    best_c5 = c5_lp
                    best_h = h_lp.copy()
        c5_snap = compute_c5(h_snap)
        if c5_snap < best_c5:
            best_c5 = c5_snap
            best_h = h_snap.copy()

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n)
