            g[(h <= 1e-10) & (g < 0)] = 0.0
            g[(h >= 1.0 - 1e-10) & (g > 0)] = 0.0
            mom = beta * mom - lr * g
            h_new = project_feasible(h + mom, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_n if False else c5_new
                best_h = h_new.copy()
                h = h_new
                lr = min(lr * 1.005, 0.04)
            else:
                h = h_new
                if c5_new > best_c5 * 1.006:
                    lr *= 0.72
                    mom *= 0.4
            if it % 1500 == 1499:
                h = best_h + rng.normal(0, 0.012, n)
                h = project_feasible(h, n)
                mom = np.zeros(n)

    # Phase 3: Coordinate descent with ternary search
    if time.time() - t_start < budget_s * 0.88:
        for sweep in range(400):
            if time.time() - t_start > budget_s * 0.88:
                break
            improved = False
            order = rng.permutation(n)
            for i in order:
                lo = max(0.0, best_h[i] - 0.4)
                hi = min(1.0, best_h[i] + 0.4)
                for _ in range(16):
                    m1 = lo + (hi - lo) * 0.382
                    m2 = hi - (hi - lo) * 0.382
                    h1 = best_h.copy()
                    h1[i] = m1
                    h1 = project_feasible(h1, n)
                    h2 = best_h.copy()
                    h2[i] = m2
                    h2 = project_feasible(h2, n)
                    if compute_c5(h1) < compute_c5(h2):
                        hi = m2
                    else:
                        lo = m1
                opt = (lo + hi) / 2.0
                h_n = best_h.copy()
                h_n[i] = opt
                h_n = project_feasible(h_n, n)
                c5_n = compute_c5(h_n)
                if c5_n < best_c5 - 1e-12:
                    best_c5 = c5_n
                    best_h = h_n.copy()
                    improved = True
            if not improved and sweep > 5:
                break

    # Phase 4: Pair-swap local search
    if time.time() - t_start < budget_s * 0.93:
        for sweep in range(30):
            if time.time() - t_start > budget_s * 0.93:
                break
            improved = False
            for i in range(n - 1):
                for j in range(i + 1, n):
                    h_n = best_h.copy()
                    h_n[i], h_n[j] = best_h[j], best_h[i]
                    h_n = project_feasible(h_n, n)
                    c5_n = compute_c5(h_n)
                    if c5_n < best_c5 - 1e-12:
                        best_c5 = c5_n
                        best_h = h_n.copy()
                        improved = True
                        break
                if time.time() - t_start > budget_s * 0.93:
                    break
            if not improved:
                break

    # Phase 5: Binary rounding restart + SLSQP
    if time.time() - t_start < budget_s * 0.96:
        h_bin = np.round(best_h)
        h_bin = np.clip(h_bin, 0.0, 1.0)
        diff = n / 2.0 - h_bin.sum()
        if abs(diff) > 1e-10:
            if diff > 0:
                idx = np.argmin(h_bin)
            else:
                idx = np.argmax(h_bin)
            h_bin[idx] = np.clip(h_bin[idx] + diff, 0.0, 1.0)
        h_bin = project_feasible(h_bin, n)
        x_res = do_slsqp(h_bin, 0.001, bounds, cons, 4000)
        h_n = project_feasible(x_res, n)
        c5_n = compute_c5(h_n)
        if c5_n < best_c5:
            best_c5 = c5_n
            best_h = h_n.copy()

    # Phase 6: Final SLSQP polish with very low temp
    if time.time() - t_start < budget_s * 0.99:
        x_res = do_slsqp(best_h, 1e-5, bounds, cons, 10000)
        h_n = project_feasible(x_res, n)
        c5_n = compute_c5(h_n)
        if c5_n < best_c5:
            best_c5 = c5_n
            best_h = h_n.copy()

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n)
