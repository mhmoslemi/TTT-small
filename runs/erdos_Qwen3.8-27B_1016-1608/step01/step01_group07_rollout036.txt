    c = np.correlate(h, 1.0 - h, mode='full')

    m = np.max(v)
    e = np.exp(np.clip(v - m, -60, 60))

def _smooth_max_grad(h, alpha):

            G2[sh:] += w[k] * x[:n-sh]

    return np.ones(_GN)

    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

    t = 0.5 * n
    for _ in range(200):
        d = t - np.sum(h)
        if abs(d) < 1e-13:
            break
        if d > 0:

        else:
            mask = h > 0.0
        k = int(np.sum(mask))
        if k == 0:
            break
        h = h + (d / k) * mask
        h = np.clip(h, 0.0, 1.0)

        return h.copy()
    h_new = np.interp(np.arange(n) * (2.0 / n), np.arange(n0) * (2.0 / n0), h)

import numpy as np
from scipy.optimize import minimize
import time

_GN = 0
_GA = 1.0
_GMU = 0.0

def _obj(x):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    v = c * dx
    m = np.max(v)
    e = np.exp(np.clip(v - m, -60, 60))
    val = m + np.log(e.sum()) / _GA
    if _GMU > 0.0:
        val += 0.5 * _GMU * (np.sum(x) - 0.5 * n) ** 2
    return val

def _grad(x):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    v = c * dx
    m = np.max(v)
    e = np.exp(np.clip(v - m, -60, 60))
    ee = e.sum()
    w = e / ee
    G1 = np.zeros(n)
    G2 = np.zeros(n)
    for k in range(2 * n - 1):
        sh = k - (n - 1)
        if sh >= 0:
            G1[:n-sh] += w[k] * (1.0 - x[sh:])
            G2[sh:] += w[k] * x[:n-sh]
        else:
            t = -sh
            G1[t:] += w[k] * (1.0 - x[:n-t])
            G2[:n-t] += w[k] * x[t:]
    g = dx * (G1 - G2)
    if _GMU > 0.0:
        g += _GMU * (np.sum(x) - 0.5 * n)
    return g

def _cfun(x):
    return np.sum(x) - 0.5 * _GN

def _cjac(x):
    return np.ones(_GN)

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def _fix(h, n):
    t = 0.5 * n
    for _ in range(300):
        d = t - np.sum(h)
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        k = int(np.sum(mask))
        if k == 0:
            break
        h = h + (d / k) * mask
        h = np.clip(h, 0.0, 1.0)
    return h

def _interp(h, n):
    n0 = len(h)
    if n0 == n:
        return h.copy()
    h_new = np.interp(np.arange(n) * (2.0 / n), np.arange(n0) * (2.0 / n0), h)
    return _fix(np.clip(h_new, 0.0, 1.0), n)

def _try_slsqp(h0, n, alpha, maxiter, t0, budget):
    global _GN, _GA, _GMU
    if budget - (time.time() - t0) < 30:
        return None
    _GN = n
    _GA = alpha
    _GMU = 0.0
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_obj, h0, jac=_grad, method='SLSQP',
                       bounds=bnds,
                       constraints=[{'type': 'eq', 'fun': _cfun, 'jac': _cjac}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None:
            return _fix(np.clip(res.x, 0.0, 1.0), n)
    except Exception:
        pass
    return None

def _try_lbfgs(h0, n, alpha, mu, maxiter, t0, budget):
    global _GN, _GA, _GMU
    if budget - (time.time() - t0) < 20:
        return None
    _GN = n
    _GA = alpha
    _GMU = mu
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_obj, h0, jac=_grad, method='L-BFGS-B',
                       bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10, 'maxls': 50})
        if res.x is not None:
            ht = _fix(np.clip(res.x, 0.0, 1.0), n)
            if abs(np.sum(ht) - 0.5 * n) < 0.1 * n:
                return ht
    except Exception:
        pass
    return None
