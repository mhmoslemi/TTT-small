```python
def smoothed_grad(h, T=5, temp=0.01):
    """Average gradient over top-T lags, weighted by softmax."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    # Get top T lags
    top_idx = np.argsort(corr)[-T:]
    # Softmax weights
    vals = corr[top_idx]
    weights = np.exp((vals - vals.max()) / temp)
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for w, idx in zip(weights, top_idx):
        k = idx - (n - 1)
        if k >= 0:
            grad[:n-k] += w * (1.0 - h[k:])
            grad[k:] -= w * h[:n-k]
        else:
            j = -k
            grad[j:] += w * (1.0 - h[:n-j])
            grad[:n-j] -= w * h[j:]
    
    return grad * dx
```
