def repair(h, n):

    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:

        else:

            h[:] = target / n
            break
        h[mask] += diff / max(1, np.sum(mask))
        h = np.clip(h, 0.0, 1.0)

    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)

    for _ in range(50):

        diff = target - s
        if abs(diff) < 1e-12:

        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)

    lo, hi = -2.0, 2.0
    for _ in range(60):

            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) / 2, 0.0, 1.0)
    # Final fix
    h = repair(h, n)
    return h

    """Local search on binary vectors using swap moves."""
    rng = np.random.RandomState(seed)
    h = np.round(h0)
    h = (h > 0.5).astype(float)
    # Ensure exactly n/2 ones
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    while len(ones) > n // 2:
        idx = rng.choice(ones)
        h[idx] = 0
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
    while len(ones) < n // 2:
        idx = rng.choice(zeros)
        h[idx] = 1
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
    
    best_c5 = compute_c5(h)
    best_h = h.copy()
    t = t_start
    cooling = (t_end / t_start) ** (1.0 / max(1, max_iters))
    
    for it in range(max_iters):
        # Choose a random swap
        i, j = rng.choice(n, size=2, replace=False)
        if h[i] == h[j]:
            continue

        h[i], h[j] = h[j], h[i]
        new_c5 = compute_c5(h)
        if new_c5 < best_c5 or rng.random() < np.exp((best_c5 - new_c5) / t):
            best_c5 = min(best_c5, new_c5)
            best_h = h.copy()
        else:
            # Revert
            h[i], h[j] = h[j], h[i]
        t *= cooling

        return np.sum(x) - target

def subgradient_step(h, n, lr=0.01):
    """One subgradient step on f(h) = max_k g_k(h)."""
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    n_corr = len(corr)  # 2n-1
    # Find the active set (lags achieving max)
    max_val = np.max(corr)
    active = np.where(corr >= max_val - 1e-10)[0]
    
    # Compute subgradient: average gradient of active g_k
    grad = np.zeros(n)
    for l in active:
        d = l - (n - 1)  # actual lag
        # g_l(h) = sum_{i: valid} h[i] * (1-h[i-d])
        # d/grad[j] of g_l:
        # j contributes as h[j] in term h[j]*(1-h[j-d]) if j-d valid: (1-h[j-d])
        # j contributes as h[j-d] in term h[j-d+d]*(1-h[j-d])... wait
        # Let me re-derive
        # g_l = sum_i h[i]*(1-h[i-d]) for valid i
        # d(g_l)/d(h[j]) = (1-h[j-d]) [if j-d valid] - sum_{i: i-d=j} h[i] [if i=j+d valid]
        # = (1-h[j-d]) [if 0≤j-d<n] - h[j+d] [if 0≤j+d<n]
        jd = j - d
        jd2 = j + d
        if 0 <= jd < n:
            grad[j] += (1 - h[jd])
        if 0 <= jd2 < n:
            grad[j] -= h[jd2]
    grad /= len(active)
    
    # Take gradient step
    h_new = h - lr * grad
    h_new = project_to_constraint(h_new, n)
    return h_new, max_val

Instead of directly minimizing the max, minimize:

The gradient of f_T is:

= weighted average of ∇C(k) with weights proportional to exp(T*C(k))

This naturally focuses on the "worst" lags while still considering all of them.

from scipy.optimize import minimize
import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_all_lags(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def smooth_max(h, T):
    lags = compute_all_lags(h)
    # Smooth max approximation
    m = np.max(lags)
    return m + np.log(np.sum(np.exp(T * (lags - m)))) / T

def smooth_max_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    lags = corr * dx
    m = np.max(lags)
    exps = np.exp(T * (lags - m))
    weights = exps / np.sum(exps)
    
    # Gradient of each lag w.r.t. h
    # Lag l (0-indexed in full output) corresponds to d = l - (n-1)
    # C_l = sum_i h[i]*(1-h[i-d]) * dx for valid i
    # d(C_l)/d(h[j]) = dx * [(1-h[j-d]) - h[j+d]] for valid indices
    grad = np.zeros(n)
    for l in range(len(corr)):
        if weights[l] < 1e-15:
            continue
        d = l - (n - 1)
        w = weights[l]
        # For each j:
        # Term 1: h[j] appears as h[i] when i=j, valid if j-d >= 0 and j-d < n
        #   contribution: (1-h[j-d])
        # Term 2: h[j] appears as h[i-d] when i=j+d, valid if j+d >= 0 and j+d < n
        #   contribution: -h[j+d]
        # j-d valid: 0 <= j-d < n => d <= j < n+d
        # j+d valid: 0 <= j+d < n => -d <= j < n-d
        
        jd_low = max(0, d)  # j >= d for j-d >= 0
        jd_high = min(n, n + d)  # j < n+d
        # h[j-d] term: (1-h[j-d])
        if jd_low < jd_high:
            idx = np.arange(jd_low, jd_high)
            grad[idx] += w * (1.0 - h[idx - d])
        
        jd2_low = max(0, -d)  # j >= -d for j+d >= 0
        jd2_high = min(n, n - d)  # j < n-d
        # -h[j+d] term
        if jd2_low < jd2_high:
            idx = np.arange(jd2_low, jd2_high)
            grad[idx] += w * (-h[idx + d])
    
    return m + np.log(np.sum(exps)) / T, grad

def project(p, n):
    """Project p onto {x: 0<=x<=1, sum(x)=n/2}"""
    target = n / 2.0
    # Bisection on lambda
    lo, hi = -max(np.abs(p)) - 1, max(np.abs(p)) + 1
    for _ in range(80):
        mid = (lo + hi) / 2
        x = np.clip(p - mid, 0, 1)
        s = np.sum(x)
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(p - (lo + hi) / 2, 0, 1)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            h[:] = target / n
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def project(p, n):
    target = n / 2.0
    lo, hi = -max(1.0, np.max(np.abs(p))) - 1, max(1.0, np.max(np.abs(p))) + 1
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(p - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(p - (lo + hi) / 2, 0, 1)

def smooth_max_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    lags = corr * dx
    mx = np.max(lags)
    exps = np.exp(T * (lags - mx))
    tot = np.sum(exps)
    fval = mx + np.log(tot) / T
    weights = exps / tot
    d_vals = np.arange(len(corr)) - (n - 1)
    grad = np.zeros(n)
    for l in range(len(corr)):
        w = weights[l]
        if w < 1e-13:
            continue
        d = d_vals[l]
        j1lo = max(0, d)
        j1hi = min(n, n + d)
        if j1lo < j1hi:
            grad[j1lo:j1hi] += w * (1.0 - h[j1lo - d:j1hi - d])
        j2lo = max(0, -d)
        j2hi = min(n, n - d)
        if j2lo < j2hi:
            grad[j2lo:j2hi] += w * (-h[j2lo + d:j2hi + d])
    return fval, grad

def projected_grad_descent(h0, n, T_start=5, T_end=100, n_iter=5000, t_budget=None):
    h = h0.copy()
    t0 = time.time()
    T = T_start
    n_total = n_iter
    for it in range(n_total):
        if t_budget is not None and time.time() - t0 > t_budget:
            break
        T = T_start * (T_end / T_start) ** (it / max(1, n_total - 1))
        fval, grad = smooth_max_grad(h, T)
        # Backtracking line search
        lr = 0.5 / (T + 1)
        for _ in range(50):
            h_new = project(h - lr * grad, n)
            f_new, _ = smooth_max_grad(h_new, T)
            if f_new < fval - 1e-10:
                break
            lr *= 0.5
        else:
            break
        h = h_new
    return h

def slsqp_opt(h0, n, maxiter=500):
    bnds = [(0.0, 1.0)] * n
    target = n / 2.0
    def sc(x):
        return np.sum(x) - target
    cons = [{'type': 'eq', 'fun': sc}]
    try:
        res = minimize(compute_c5, h0, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.fun < compute_c5(h0) + 1e-14:
            return repair(np.clip(res.x, 0, 1), n)
    except Exception:
        pass
    return h0

def to_binary(h, n, seed=0):
    rng = np.random.RandomState(seed)
    h = np.round(h)
    # Ensure exactly n/2 ones
    ones = np.sum(h)
    while ones > n / 2:
        idx = rng.choice(np.where(h == 1)[0])
        h[idx] = 0
        ones -= 1
    while ones < n / 2:
        idx = rng.choice(np.where(h == 0)[0])
        h[idx] = 1
        ones += 1
    return h

def binary_sa(h_bin, n, n_iter=3000, t_hi=1.0, t_lo=0.01, seed=0):
    rng = np.random.RandomState(seed)
    h = h_bin.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t = t_hi
    cool = (t_lo / t_hi) ** (1.0 / max(1, n_iter))
    for it in range(n_iter):
        i, j = rng.choice(n, size=2, replace=False)
        if h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        c5_new = compute_c5(h)
        if c5_new < best_c5 or rng.random() < np.exp(-(c5_new - best_c5) / max(t, 1e-10)):
            best_c5 = min(best_c5, c5_new)
            best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
        t *= cool
    return best_h, best_c5

    try:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            h[:] = target / n
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def project(p, n):
    target = n / 2.0
    lo, hi = -max(2.0, np.max(np.abs(p))) - 1, max(2.0, np.max(np.abs(p))) + 1
    for _ in range(70):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(p - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(p - (lo + hi) / 2, 0, 1)

def smooth_max_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    lags = corr * dx
    mx = np.max(lags)
    exps = np.exp(T * (lags - mx))
    tot = np.sum(exps)
    fval = mx + np.log(tot) / T
    weights = exps / tot
    d_vals = np.arange(len(corr)) - (n - 1)
    grad = np.zeros(n)
    for l in range(len(corr)):
        w = weights[l]
        if w < 1e-13:
            continue
        d = d_vals[l]
        j1lo = max(0, d)
        j1hi = min(n, n + d)
        if j1lo < j1hi:
            grad[j1lo:j1hi] += w * dx * (1.0 - h[j1lo - d:j1hi - d])
        j2lo = max(0, -d)
        j2hi = min(n, n - d)
        if j2lo < j2hi:
            grad[j2lo:j2hi] += w * dx * (-h[j2lo + d:j2hi + d])
    return fval, grad

def projected_grad_descent(h0, n, T_start, T_end, n_iter, t_budget):
    h = h0.copy()
    t0 = time.time()
    for it in range(n_iter):
        if time.time() - t0 > t_budget:
            break
        T = T_start * (T_end / T_start) ** (it / max(1, n_iter - 1))
        fval, grad = smooth_max_grad(h, T)
        lr = 1.0 / (T * dx_factor(n) + 1.0)
        accepted = False
        for _ in range(60):
            h_new = project(h - lr * grad, n)
            f_new, _ = smooth_max_grad(h_new, T)
            if f_new < fval - 1e-12 * max(1, fval):
                h = h_new
                accepted = True
                break
            lr *= 0.5
        if not accepted:
            break
    return h

def dx_factor(n):
    return 2.0 / n

def slsqp_opt(h0, n, maxiter=500):
    bnds = [(0.0, 1.0)] * n
    target = n / 2.0
    def _sc(x):
        return np.sum(x) - target
    cons = [{'type': 'eq', 'fun': _sc}]
    try:
        res = minimize(compute_c5, h0, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.fun < compute_c5(h0) + 1e-14:
            return repair(np.clip(res.x, 0, 1), n)
    except Exception:
        pass
    return h0

def to_binary(h, n, seed=0):
    rng = np.random.RandomState(seed)
    h = np.round(h).astype(float)
    ones = np.sum(h)
    for _ in range(300):
        if ones > n / 2:
            idx = rng.choice(np.where(h == 1)[0])
            h[idx] = 0
            ones -= 1
        elif ones < n / 2:
            idx = rng.choice(np.where(h == 0)[0])
            h[idx] = 1
            ones += 1
        else:
            break
    return h

def binary_sa(h_bin, n, n_iter, t_hi, t_lo, seed):
    rng = np.random.RandomState(seed)
    h = h_bin.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t = t_hi
    cool = (t_lo / t_hi) ** (1.0 / max(1, n_iter - 1))
    for it in range(n_iter):
        i, j = rng.choice(n, size=2, replace=False)
        if h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        c5_new = compute_c5(h)
        delta = c5_new - best_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-10)):
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
        t *= cool
    return best_h, best_c5
