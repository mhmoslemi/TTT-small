grad[tau:] += (1 - h[:n-tau])  # first term
grad[:n-tau] += -h[tau:]       # second term

For tau < 0, l = -tau:
c[-l] = sum_{valid i} h[i] * (1-h[i+l])
- h[j] in h[i] when i=j: need j+l valid (0 ≤ j+l < n), i.e., j < n-l. Value: (1-h[j+l])
- h[j] in h[i+l] when i+l=j, i=j-l: need j-l valid (0 ≤ j-l < n), i.e., j ≥ l. Value: -h[j-l]

So: grad[:n-l] += (1 - h[l:])   # first term
grad[l:] += -h[:n-l]           # second term

Let me verify with n=4, h=[a,b,c,d], tau=-1 (l=1):
c[-1] = h[0]*(1-h[1]) + h[1]*(1-h[2]) + h[2]*(1-h[3])
      = a(1-b) + b(1-c) + c(1-d)

d/dh[0] = (1-b) ✓ (from first term, j=0 < n-l=3, (1-h[0+1])=(1-b))
d/dh[1] = -a + (1-c) ✓ (from second term j=1≥l=1: -h[1-1]=-h[0]=-a; first term j=1<3: (1-h[2])=(1-c))
d/dh[2] = -b + (1-d) ✓
d/dh[3] = -c ✓ (only second term, j=3≥1: -h[3-1]=-h[2]=-c)

So the vectorized gradient for a set of active lags K:
For each tau in K:
  if tau >= 0:
    grad[tau:] += (1.0 - h[:n-tau])
    grad[:n-tau] += -h[tau:]
  else:
    l = -tau
    grad[:n-l] += (1.0 - h[l:])
    grad[l:] += -h[:n-l]

Then multiply by dx.

This is O(n * |K|) per gradient evaluation, which for n=50 and |K| up to ~50 is 2500 operations. Very fast.

    target = n / 2.0

        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h = np.clip(h + diff / cnt * mask, 0.0, 1.0)
    return h

def active_grad(h, n, lags):
    """Gradient of sum of overlaps at given lags, times dx"""
    dx = 2.0 / n
    grad = np.zeros(n)
    for tau in lags:
        if tau >= 0:
            if tau == 0:
                grad[:] += (1.0 - h) - h  # (1-h) + (-h) = 1-2h
            else:
                grad[tau:] += (1.0 - h[:n-tau])
                grad[:n-tau] += -h[tau:]
        else:
            l = -tau
            grad[:n-l] += (1.0 - h[l:])
            grad[l:] += -h[:n-l]
    return grad * dx

    h = project_feasible(h, n)
    score = compute_score(h)
    
    for it in range(max_iter):
        overlaps = compute_overlaps(h)
        # Find active set: lags near the maximum
        max_val = overlaps.max()
        # Use top-k or all within epsilon of max
        active_mask = overlaps > max_val - 0.02
        active_indices = np.where(active_mask)[0]
        lags = active_indices - (n - 1)

        if len(lags) > top_k + 10:

            lags = active_idx_sorted - (n - 1)
        
        # Compute gradient
        g = active_grad(h, n, lags)
        
        # Backtracking line search
        alpha = 0.1
        best_h = h

        score = best_s

        if s_ls < best_score:

    # Phase 3: Multi-restart from structured patterns (25%)
    tl = time_left()
    if tl > 30:
        per = min(tl * 0.25 / 8, 12.0)
        for attempt in range(8):
            if time_left() < 10:
                break
            style = attempt % 4
            h_start = generate_structured(n0, rng, style)
            # Also try perturbed initial
            if attempt >= 4:
                h_start = generate_structured(n0, rng, style - 4)
            h_opt, s_opt = optimize_gradient(h_start, n0, per)
            # Quick local search
            h_ls, s_ls = pair_swap_search(h_opt, n0, max_sweeps=100)
            if s_ls < s_opt:
                s_opt, h_opt = s_ls, h_ls
            if s_opt < best_score:
                best_score, best_h = s_opt, h_opt

    # Phase 4: Refine best with longer gradient (15%)
    tl = time_left()
    if tl > 10:
        h_opt, s_opt = optimize_gradient(best_h, n0, tl * 0.8)
        if s_opt < best_score:
            best_score, best_h = s_opt, h_opt
        h_ls, s_ls = pair_swap_search(best_h, n0, max_sweeps=200)
        if s_ls < best_score:
            best_score, best_h = s_ls, h_ls

    # Phase 5: Try other n values (15%)
    tl = time_left()
    if tl > 20:
        x_old = np.linspace(0, 2, n0, endpoint=False)
        for n_try in [40, 48, 56, 64]:
            if time_left() < 15:
                break
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_try = project_feasible(np.interp(x_new, x_old, best_h), n_try)
            h_opt, s_opt = optimize_gradient(h_try, n_try, time_left() * 0.3)
            h_ls, s_ls = pair_swap_search(h_opt, n_try, max_sweeps=100)
            if s_ls < s_opt:
                s_opt, h_opt = s_ls, h_opt
            if s_opt < best_score:
                best_score, best_h = s_opt, h_opt.copy()

    # Final polish
    if time_left() > 5:
        n_f = len(best_h)
        h_opt, s_opt = optimize_gradient(best_h, n_f, time_left() * 0.7)
        if s_opt < best_score:
            best_score, best_h = s_opt, h_opt

    best_h = project_feasible(best_h, len(best_h))
    best_score = compute_score(best_h)
    return best_h, best_score, len(best_h)
