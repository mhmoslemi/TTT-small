    dx = 2.0 / n
    # Compute correlation with h[i] = 0 and h[i] = 1
    h0 = h.copy()
    h0[i] = 0.0
    h1 = h.copy()
    h1[i] = 1.0

    corr1 = np.correlate(h1, 1.0 - h1, mode='full') * dx

    """Try flipping a pair (i: 1->0, j: 0->1) in the binary version."""

    zeros = np.where(h < 0.5)[0]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def sum_eq(x, n):
    return np.sum(x) - n / 2.0

def repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            h[:] = target / n
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_descent(h, n, n_sweeps=80, grid=40, rng=None, random_order=False):
    best_c = compute_c5(h)
    for sweep in range(n_sweeps):
        improved = False
        order = rng.permutation(n) if random_order else range(n)
        for i in order:
            best_x = h[i]
            for x in np.linspace(0, 1, grid):
                h[i] = x
                c = compute_c5(h)
                if c < best_c - 1e-14:
                    best_c = c
                    best_x = x
            h[i] = best_x
            if abs(h[i] - (h[i] if i >= n else h[i])) > 0:
                pass
        if not improved:
            break
    return h, best_c

def coord_descent_fast(h, n, n_sweeps=80, grid=40, rng=None, random_order=False):

    xs_grid = np.linspace(0, 1, grid)
    for sweep in range(n_sweeps):
        improved = False
        order = rng.permutation(n) if random_order else range(n)
        for i in order:
            old_x = h[i]
            best_x = old_x
            for x in xs_grid:
                h[i] = x
                c = compute_c5(h)
                if c < best_c - 1e-14:
                    best_c = c
                    best_x = x
            h[i] = best_x
            if best_x != old_x:
                improved = True
        if not improved:
            break
    return h, best_c

def binary_sa(h, n, rng, t0, budget_s, frac_limit=0.65):
    hb = (h > 0.5).astype(float).copy()
    n_target = n // 2
    n_ones = int(np.sum(hb))
    if n_ones > n_target:
        ones = np.where(hb == 1)[0]
        for idx in rng.choice(ones, n_ones - n_target, replace=False):
            hb[idx] = 0
    elif n_ones < n_target:
        zeros = np.where(hb == 0)[0]
        for idx in rng.choice(zeros, n_target - n_ones, replace=False):
            hb[idx] = 1
    c_cur = compute_c5(hb)
    c_best = c_cur
    h_best_bin = hb.copy()
    temp = 0.005
    it = 0
    while time.time() - t0 < budget_s * frac_limit:
        ones = np.where(hb > 0.5)[0]
        zeros = np.where(hb < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        hb[i] = 0.0
        hb[j] = 1.0
        c_new = compute_c5(hb)
        if c_new < c_cur or rng.random() < np.exp(-(c_new - c_cur) / max(temp, 1e-12)):
            c_cur = c_new
            if c_new < c_best:
                c_best = c_new
                h_best_bin = hb.copy()
        else:
            hb[i] = 1.0
            hb[j] = 0.0
        if it % 100 == 0:
            temp = max(temp * 0.995, 1e-5)
        it += 1
    return h_best_bin, c_best

def slsqp_refine(h0, n, maxiter=300):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: sum_eq(x, n)}]
    try:
        res = minimize(compute_c5, h0, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.fun < compute_c5(h0) + 1e-14:
            return repair(np.clip(res.x, 0.0, 1.0), n)
    except Exception:
        pass
    return h0
