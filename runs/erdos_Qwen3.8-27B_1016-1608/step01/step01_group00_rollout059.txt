def compute_c5(h):

def repair(h, n):

    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:

        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:

        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)

    return h

        return compute_c5(x)

                       options={'maxiter': maxiter, 'ftol': 1e-14})

            return repair(np.clip(res.x, 0.0, 1.0), n)

        pass

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def sum_constraint(x):
    return float(np.sum(x)) - len(x) / 2.0

def fast_repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - float(np.sum(h))
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-10
    else:
        mask = h > 1e-10
    cnt = int(np.sum(mask))
    if cnt == 0:
        return h
    h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def slsqp_opt(h0, n, maxiter=200):
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(compute_c5, h0, method='SLSQP', bounds=bnds,
                       constraints=[{'type': 'eq', 'fun': sum_constraint}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        c_new = float(res.fun)
        if c_new < compute_c5(h0) - 1e-15:
            return fast_repair(np.clip(res.x, 0.0, 1.0), n)
    except Exception:
        pass
    return h0
