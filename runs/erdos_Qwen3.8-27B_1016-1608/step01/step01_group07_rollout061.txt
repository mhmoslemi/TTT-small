    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def _fix(h, n):
    target = 0.5 * n
    for _ in range(500):
        d = target - np.sum(h)
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        k = int(np.sum(mask))
        if k == 0:
            break
        h = h + (d / k) * mask
        np.clip(h, 0.0, 1.0, out=h)
    return h

    n0 = len(h)
    xo = np.arange(n0) * (2.0 / n0)
    xn = np.arange(n_new) * (2.0 / n_new)
    h_new = np.interp(xn, xo, h)
    return _fix(np.clip(h_new, 0.0, 1.0), n_new)

def _coord_descent(h, sweeps=100):
    n = len(h)
    h = h.astype(float).copy()
    c = np.correlate(h, 1.0 - h, mode='full').copy()
    nm1 = n - 1
    ks = np.arange(2 * n - 1) - nm1
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    prev_c5 = _c5(h)
    
    for sweep in range(sweeps):
        for i in range(n):
            old = h[i]
            if old <= 0.0 or old >= 1.0:
                continue
            va = ((i - ks >= 0) & (i - ks < n)).astype(float)
            vb = ((i + ks >= 0) & (i + ks < n)).astype(float)
            idx_mk = np.clip(i - ks, 0, n - 1)
            idx_pk = np.clip(i + ks, 0, n - 1)
            h_ik = h[idx_mk] * va
            h_ipk = h[idx_pk] * vb
            dcoef = (1.0 - h_ik) * va - h_ipk * vb
            
            lo = -old
            hi = 1.0 - old
            cp = hi - gr * (hi - lo)
            dp = lo + gr * (hi - lo)

                    dp = cp
                    fd = fc

                    cp = dp
                    fc = fd

        ht = _try_trf(bh, n, 500.0, 500, t0, budget_s)
        if ht is not None:
            ct = _c5(ht)
            if ct < bc:
                bh, bc = ht, ct
    # Final CD
    if time.time() - t0 < budget_s - 5:
        ht = _coord_descent(bh, sweeps=200)
        ct = _c5(ht)
        if ct < bc:
            bh, bc = ht, ct
    bh = _fix(bh, n)
    bc = _c5(bh)
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(400, 0.5)
    n0 = len(h0)
    h0 = _fix(h0, n0)

    bh = h0.copy()
    bc = _c5(bh)

    # Phase 1: Optimize parent at native resolution
    bh, bc = _improve(bh, n0, t0, budget_s)

    # Phase 2: Try other resolutions
    for nn in [200, 250, 300, 320, 350, 400, 450, 500]:
        if time.time() - t0 > budget_s - 80:
            break
        if nn == n0:
            continue
        hi = _interp(bh, nn)
        htc = _c5(hi)
        if htc < bc:
            bh, bc = hi, htc
        else:
            bh = hi
        bh, ctmp = _improve(bh, nn, t0, budget_s)
        if ctmp < bc:
            bc = ctmp

    # Phase 3: Try alternative initializations
    for init_fn in [_init_block, lambda n: _init_sin(n, 1), lambda n: _init_sin(n, 2),
                    lambda n: _init_sin(n, 3)]:
        if time.time() - t0 > budget_s - 60:
            break
        hi = _fix(np.clip(rng.rand(n0) * 0.3 + init_fn(n0), 0, 1), n0)
        bh_c, cc = _improve(hi, n0, t0, budget_s)
        if cc < bc:
            bh, bc = bh_c, cc

    # Phase 4: Final high-precision polish
    nb = len(bh)
    if time.time() - t0 < budget_s - 30:
        ht = _try_slsqp(bh, nb, 5000.0, 1500, t0, budget_s)
        if ht is not None:
            ct = _c5(ht)
            if ct < bc:
                bh, bc = ht, ct
    if time.time() - t0 < budget_s - 10:
        ht = _coord_descent(bh, sweeps=300)
        ct = _c5(ht)
        if ct < bc:
            bh, bc = ht, ct

    bh = _fix(bh, len(bh))
    bc = _c5(bh)
    return bh, bc, len(bh)
