import time

def compute_c5(h):

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        d = n / 2.0 - np.sum(h)
        if abs(d) < 1e-15:
            break
        h += d / n
    h = np.clip(h, 0.0, 1.0)
    # Final projection
    for _ in range(100):
        d = n / 2.0 - np.sum(h)
        if abs(d) < 1e-15:
            break
        if d > 0:
            mask = h < 1.0 - 1e-10
            if np.sum(mask) == 0: break
            add = d / np.sum(mask)
            h[mask] = np.minimum(h[mask] + add, 1.0)
        else:
            mask = h > 1e-10
            if np.sum(mask) == 0: break
            sub = -d / np.sum(mask)
            h[mask] = np.maximum(h[mask] - sub, 0.0)
    return h

def coordinate_descent(h, n, n_sweeps=50, n_inner=30):
    h = h.copy()
    dx = 2.0 / n
    for sweep in range(n_sweeps):
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        for j in range(n):

            # c_{n-1+s} = dx * sum_{i=0}^{n-1-s} h[i]*(1-h[i+s]) for s>0

                #   dc/dh[j] = dx*(1-2*h[j])
                # For s < 0, s=-|s|: c = dx*sum_{i=|s|}^{n-1} h[i]*(1-h[i-|s|])
                #   = dx*sum_{j=0}^{n-1-|s|} h[j+|s|]*(1-h[j])
                #   dc/dh[j] = dx*(1-h[j+|s|]) if 0<=j<=n-1-|s| (from h[j+|s|] term)
                #            - dx*h[j-|s|] if |s|<=j<=n-1 (from (1-h[j]) term)

                # c = dx*sum_{j=0}^{n-1-m} h[j+m]*(1-h[j])

                #   h[j] as (1-h[j]): -dx*h[j+m] for j in [0, n-1-m]
                #   h[j] as h[j+m]: dx*(1-h[j-m]) for j in [m, n-1] (since j+m in [0,n-1-m+m]=[m,n-1])
                #   Wait: h[j+m] for j in [0, n-1-m] covers indices [m, n-1]
                #   So h[j] for j in [m, n-1] appears as h[j] = h[(j-m)+m] where (j-m) in [0, n-1-m]

                else:

    idx1 = n - 1 + s_pos[mask1]
    slopes[idx1] += (1.0 - h[j + s_pos[mask1]]) * dx
    # -h[j-s] for j >= s, i.e., s <= j
    mask2 = s_pos <= j
    idx2 = n - 1 + s_pos[mask2]
    slopes[idx2] -= h[j - s_pos[mask2]] * dx
    # s < 0: m = -s from 1 to n-1
    m_pos = np.arange(1, n)
    # -h[j+m] for j+m < n, i.e., m < n-j
    mask3 = m_pos < (n - j)
    idx3 = n - 1 - m_pos[mask3]
    slopes[idx3] -= h[j + m_pos[mask3]] * dx
    # (1-h[j-m]) for j >= m, i.e., m <= j
    mask4 = m_pos <= j
    idx4 = n - 1 - m_pos[mask4]
    slopes[idx4] += (1.0 - h[j - m_pos[mask4]]) * dx
    return slopes
