def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-12:
            break
        # Bisection for uniform shift
        lo, hi = -2.0, 2.0
        if d > 0:
            lo = 0.0 - 1e-6
        else:
            hi = 0.0 + 1e-6
        for _ in range(80):
            mid = (lo + hi) / 2.0
            s_test = np.sum(np.clip(h + mid, 0.0, 1.0))
            if s_test < target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    # Final clip and tiny correction
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) > 1e-14:
        # Distribute remaining error among non-bound elements
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += (target - s) / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h

def make_grad(h, n, eps):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    s = (corr - np.max(corr)) / eps
    w = np.exp(s)
    w /= np.sum(w)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        delta = k - (n - 1)
        for j in range(n):
            g = 0.0
            if 0 <= j + delta < n:
                g += (1.0 - h[j + delta])
            if 0 <= j - delta < n:
                g -= h[j - delta]
            grad[j] += g * dx * w[k]
    return grad

        g = make_grad(h, n, eps)

                break
        # Occasionally increase alpha

    return best_h, best_c

    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        s = np.sum(h)
        d = target - s

        else:
            hi = 0.0

            if np.sum(np.clip(h + mid, 0.0, 1.0)) < target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)
    return h

def make_grad(h, n, eps):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    s = (corr - m) / eps

        # Second term
        v2 = j_arr - delta
        mask2 = (v2 >= 0) & (v2 < n)

    for it in range(max_iter):
        if time.time() > t_limit:
            break
        g = make_grad(h, n, eps)
        g -= np.mean(g)
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        g = g / gn
        improved = False
        for trial in range(15):
            h_new = project_feasible(h - alpha * g, n)
            c_new = compute_c5(h_new)
            if c_new < best_c:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                improved = True
                break
            alpha *= 0.6
        if not improved:
            alpha *= 0.4
            if alpha < 1e-13:
                break
        if it % 200 == 0 and alpha < 0.5:
            alpha = min(alpha * 1.5, 2.0)
    return best_h, best_c

def interp_parent(h_parent, n_new):
    n_old = len(h_parent)
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    h_new = np.interp(x_new, x_old, h_parent, period=1.0)
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    parent_h = None
    parent_n = 80
    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
    except Exception:
        pass

    cands = []
    if parent_h is not None:
        hp = project_feasible(parent_h, parent_n)
        cands.append((hp, compute_c5(hp), parent_n))

    for n in [60, 80, 100, 120, 160]:
        if time.time() - t0 > budget_s - 300:
            break
        for p in [4, 5, 6, 7, 8, 10, 12]:
            if p > n // 2:
                continue
            half = p // 2
            hp = np.array([1.0 if (i % p) < half else 0.0 for i in range(n)])
            hp = project_feasible(hp, n)
            cands.append((hp, compute_c5(hp), n))
        for k in [1, 2, 3, 4, 5, 7]:
            hp = 0.5 + 0.48 * np.cos(2.0 * np.pi * k * np.arange(n) / n)
            hp = project_feasible(hp, n)
            cands.append((hp, compute_c5(hp), n))
        t = np.linspace(0, 2 * np.pi, n, endpoint=False)
        hp = 0.5 - 0.48 * (np.abs(2.0 * np.sin(t) / np.pi - 1.0))
        hp = project_feasible(hp, n)
        cands.append((hp, compute_c5(hp), n))
        for _ in range(4):
            hp = project_feasible(np.random.rand(n), n)
            cands.append((hp, compute_c5(hp), n))

    if parent_h is not None and time.time() - t0 < budget_s - 300:
        for n_new in [60, 100, 120, 160]:
            if n_new == parent_n:
                continue
            hp = interp_parent(parent_h, n_new)
            cands.append((hp, compute_c5(hp), n_new))

    cands.sort(key=lambda x: x[1])

    best_h = cands[0][0].copy()
    best_c = cands[0][1]
    best_n = cands[0][2]

    tried_keys = set()
    max_cands = min(20, len(cands))
    for idx in range(max_cands):
        if time.time() - t0 > budget_s - 200:
            break
        h_c, c_c, n_c = cands[idx]
        key = n_c
        if key in tried_keys and c_c > best_c + 0.002:
            continue
        tried_keys.add(key)
        t_limit = min(time.time() + 50, budget_s - 180)
        for eps_init in [0.02, 0.005]:
            if time.time() > t_limit:
                break
            h_new, c_new = gd_refine(h_c, n_c, eps_init, 8000, t_limit)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
                best_n = n_c
                h_c = h_new.copy()

    if time.time() - t0 < budget_s - 50:
        for eps in [0.003, 0.001, 0.0004, 0.0001]:
            if time.time() - t0 > budget_s - 30:
                break
            h_new, c_new = gd_refine(best_h, best_n, eps, 20000, budget_s - 30)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new

    if time.time() - t0 < budget_s - 15:
        t_limit = budget_s - 5
        for _ in range(8000):
            if time.time() > t_limit:
                break
            h_try = project_feasible(best_h + np.random.randn(best_n) * 0.004, best_n)
            c_try = compute_c5(h_try)
            if c_try < best_c:
                best_c = c_try
                best_h = h_try
        h_new, c_new = gd_refine(best_h, best_n, 0.00005, 30000, t_limit)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h)
    return (best_h.tolist(), float(best_c), best_n)
