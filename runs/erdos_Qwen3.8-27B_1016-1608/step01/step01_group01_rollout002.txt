    w /= w.sum()
    grad = np.zeros(n)
    for k in range(len(c)):

            continue
        shift = k - (n - 1)

        # grad[m_idx] += w[k]*(-h[m_idx-shift]) if m_idx-shift valid
        for i in range(n):

            if 0 <= j < n:
                grad[i] += w[k] * (1.0 - h[j])
            j2 = i - shift
            if 0 <= j2 < n:
                grad[i] += w[k] * (-h[j2])
    # Penalty gradient
    grad += 2.0 * mu * (h.sum() - _glob['target']) * np.ones(n)
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * dx)

def corr_vec(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            avail = np.where(h < 1.0 - 1e-10, 1.0 - h, 0.0)
            total = avail.sum()
            if total < 1e-15:
                break
            h = np.minimum(h + diff * avail / total, 1.0)
        else:
            avail = np.where(h > 1e-10, h, 0.0)
            total = avail.sum()
            if total < 1e-15:
                break
            h = np.maximum(h + diff * avail / total, 0.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h = np.clip(h + diff / n, 0.0, 1.0)
    return h

_gp = {}

def obj_fwd(x):
    temp, mu, tgt = _gp['temp'], _gp['mu'], _gp['tgt']
    c = corr_vec(x)
    m = c.max()
    sm = m + temp * np.log(np.sum(np.exp((c - m) / temp)))
    pen = mu * (x.sum() - tgt) ** 2
    return sm + pen

def obj_bwd(x):
    temp, mu, tgt = _gp['temp'], _gp['mu'], _gp['tgt']
    n = len(x)
    c = corr_vec(x)
    m = c.max()
    w = np.exp((c - m) / temp)
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(len(c)):
        wki = w[k]
        if wki < 1e-13:
            continue
        shift = n - 1 - k
        if shift >= 0:
            nvalid = n - shift
            grad[:nvalid] += wki * (1.0 - x[shift:])
            grad[shift:] += wki * (-x[:nvalid])
        else:
            s = -shift
            nvalid = n - s
            grad[s:] += wki * (1.0 - x[:nvalid])
            grad[:nvalid] += wki * (-x[s:])
    grad += 2.0 * mu * (x.sum() - tgt)
    return grad
