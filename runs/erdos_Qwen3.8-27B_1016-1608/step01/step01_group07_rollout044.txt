```python
import numpy as np
from scipy.optimize import minimize
import time

def c5_eval(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def c5_grad(h, k):
    """Gradient of c_k = correlate(h, 1-h, 'full')[k] w.r.t. h"""
    n = len(h)
    s = k - (n - 1)
    grad = np.zeros(n)
    if s >= 0:
        grad[:n-s] += (1.0 - h[s:])
        grad[s:] -= h[:n-s]
    else:
        grad[-s:] += (1.0 - h[:n+s])
        grad[:n+s] -= h[-s:]
    return grad

def project_feasible(h, n):
    """Project h to [0,1]^n with sum = n/2"""
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    for _ in range(200):
        d = target - np.sum(h)
        if abs(d) < 1e-12:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_step(h, n, lr, momentum_buf, top_m):
    """One subgradient step on the max correlation"""
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    # Get top m correlations
    idx = np.argsort(c)[-top_m:]
    weights = np.exp((c[idx] - c[idx].max()) * 10)
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for i, k in enumerate(idx):
        grad += weights[i] * c5_grad(h, k)
    
    # Momentum
    v = momentum_buf
    v = 0.9 * v - lr * grad
    h_new = h + v
    h_new = project_feasible(h_new, n)
    return h_new, v, c[idx].max()

def interp_h(h, n):
    n0 = len(h)
    xo = np.linspace(0, 2, n0, endpoint=False)
    xn = np.linspace(0, 2, n, endpoint=False)
    h_interp = np.interp(xn, xo, h)
    return project_feasible(h_interp, n)

def random_binary(n, rng):
    h = np.zeros(n)
    ones = n // 2
    idx = rng.choice(n, ones, replace=False)
    h[idx] = 1.0
    return h

def slsqp_polish(h, n, t0, budget):
    if time.time() - t0 > budget - 15:
        return h
    dx = 2.0 / n
    
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = np.max(c)
        # Sharp log-sum-exp
        e = np.exp(np.clip(c - m, -50, 50))
        return m + np.log(np.sum(e)) * 0.01
    
    def icf(x):
        return np.sum(x) - 0.5 * n
    
    def icj(x):
        return np.ones(n)
    
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': icf, 'jac': icj}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': 500, 'ftol': 1e-14})
        if res.x is not None:
            return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        pass
    return h

    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(200, 0.5)
    
    best_h = project_feasible(h0.copy(), len(h0))
    best_c = c5_eval(best_h)
    
    n0 = len(h0)
    
    # Phase 1: Subgradient at original resolution
    n = n0
    h = best_h.copy()
    v = np.zeros(n)
    lr = 0.5
    for i in range(300):
        if time.time() - t0 > budget_s - 100:
            break
        h, v, _ = subgradient_step(h, n, lr, v, top_m=3)
        c = c5_eval(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        # Adaptive learning rate
        if i % 50 == 49:
            lr *= 0.7
    
    # Phase 2: SLSQP polish
    h_pol = slsqp_polish(best_h, n, t0, budget_s)
    c_pol = c5_eval(h_pol)
    if c_pol < best_c:
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def c5_eval(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def c5_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def c5_grad(h, k):
    n = len(h)
    s = k - (n - 1)
    grad = np.zeros(n)
    if s >= 0:
        if n - s > 0:
            grad[:n-s] += (1.0 - h[s:])
        if n - s > 0:
            grad[s:] -= h[:n-s]
    else:
        if n + s > 0:
            grad[-s:] += (1.0 - h[:n+s])
        if n + s > 0:
            grad[:n+s] -= h[-s:]
    return grad

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    for _ in range(300):
        d = target - np.sum(h)
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_h(h, n):
    n0 = len(h)
    xo = np.linspace(0, 2, n0, endpoint=False)
    xn = np.linspace(0, 2, n, endpoint=False)
    return project_feasible(np.interp(xn, xo, h).copy(), n)

def subgradient_opt(h, n, iters, t0, budget, top_m=5):

    lr = 0.8
    h_cur = h.copy()

            break
        c = c5_corr(h_cur)
        idx = np.argsort(c)[-top_m:]
        w = np.exp((c[idx] - c[idx].max()) * 8.0)

        grad = np.zeros(n)

            grad += wi * c5_grad(h_cur, ki)
        v = 0.85 * v - lr * grad
        h_cur = project_feasible(h_cur + v, n)
        if i % 60 == 59:
            lr *= 0.75
    return h_cur

def binary_local_search(h_bin, n, iters, t0, budget, rng):
    h_cur = h_bin.copy()
    best_c = c5_eval(h_cur)
    ones = np.where(h_cur > 0.5)[0]
    zeros = np.where(h_cur <= 0.5)[0]
    for i in range(iters):
        if time.time() - t0 > budget - 15:
            break
        a = ones[rng.integers(len(ones))]
        b = zeros[rng.integers(len(zeros))]
        h_cur[a] = 0.0
        h_cur[b] = 1.0
        c_new = c5_eval(h_cur)
        if c_new < best_c:
            best_c = c_new
            ones = np.where(h_cur > 0.5)[0]
            zeros = np.where(h_cur <= 0.5)[0]
        else:
            h_cur[a] = 1.0
            h_cur[b] = 0.0
    return h_cur, best_c

def slsqp_polish(h, n, t0, budget, maxiter=400, alpha=500.0):
    if time.time() - t0 > budget - 20:
        return h
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = np.max(c)
        e = np.exp(np.clip(c - m, -40.0, 40.0))
        return m + np.log(np.sum(e)) / alpha
    def icf(x):
        return np.sum(x) - 0.5 * n
    def icj(x):
        return np.ones(n)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': icf, 'jac': icj}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None:
            return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        pass
    return h
```
