        h = rng.uniform(0, 1, n)
    elif choice == 4:
        if h_ref is not None and n_ref is not None:
            x_old = np.linspace(0, 2, n_ref, endpoint=False)
            h = np.interp(x, x_old, h_ref, period=2)
        else:
            h[:n // 2] = 1.0
    else:
        for i in range(n):
            h[i] = 1.0 if (i % 3 == 0) else 0.0
    h = project_feasible(h, n)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n_ref = len(h_init)

    best_h = project_feasible(h_init, n_ref)
    best_n = n_ref
    best_c5 = compute_c5(best_h)

    # Phase 1: Polish the given initial solution (n_ref = 47)
    n = n_ref
    h_cur = best_h.copy()
    c5_cur = best_c5

    # SLSQP multi-temp
    h_cur, c5_cur = slsqp_polish(h_cur, n, [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001],
                                  t_start, t_start + budget_s * 0.25)
    if c5_cur < best_c5:
        best_c5 = c5_cur
        best_h = h_cur.copy()

    # Subgradient
    h_cur, c5_cur = subgrad_descent(h_cur, n, t_start, t_start + budget_s * 0.40, alpha0=0.05)
    if c5_cur < best_c5:
        best_c5 = c5_cur
        best_h = h_cur.copy()

    # Coordinate descent
    h_cur, c5_cur = coord_descent(h_cur, n, c5_cur,
                                  [0.1, -0.1, 0.05, -0.05, 0.02, -0.02, 0.01, -0.01],
                                  4, t_start, t_start + budget_s * 0.50)
    if c5_cur < best_c5:
        best_c5 = c5_cur
        best_h = h_cur.copy()

    # SLSQP again after coord descent
    h_cur, c5_cur = slsqp_polish(h_cur, n, [0.0003, 0.0001],
                                  t_start, t_start + budget_s * 0.55)
    if c5_cur < best_c5:
        best_c5 = c5_cur
        best_h = h_cur.copy()

    # Phase 2: Random restarts from best (same n)
    if time.time() - t_start < budget_s * 0.70:
        for r in range(8):
            if time.time() - t_start > budget_s * 0.70:
                break
            scale = 0.05 + 0.03 * (r % 4)
            h_r = best_h + rng.normal(0, scale, best_n)
            h_r = project_feasible(h_r, best_n)
            c5_r = compute_c5(h_r)
            if c5_r > best_c5 + 0.005:
                continue
            h_r, c5_r = slsqp_polish(h_r, best_n, [0.005, 0.001, 0.0003],
                                      t_start, t_start + budget_s * 0.72)
            h_r, c5_r = coord_descent(h_r, best_n, c5_r,
                                      [0.05, -0.05, 0.02, -0.02, 0.01, -0.01],
                                      2, t_start, t_start + budget_s * 0.74)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()

    # Phase 3: Try different n values
    n_candidates = [44, 46, 48, 50, 52, 42, 40, 54, 56]
    for n_try in n_candidates:
        if time.time() - t_start > budget_s * 0.90:
            break
        for s in range(3):
            if time.time() - t_start > budget_s * 0.92:
                break
            h_t = make_init(n_try, seed + 1000 * n_try + s, h_init, n_ref)
            c5_t = compute_c5(h_t)
            if c5_t > best_c5 + 0.02:
                continue
            h_t, c5_t = slsqp_polish(h_t, n_try, [0.005, 0.001, 0.0003],
                                      t_start, t_start + budget_s * 0.93)
            h_t, c5_t = coord_descent(h_t, n_try, c5_t,
                                      [0.05, -0.05, 0.02, -0.02, 0.01, -0.01],
                                      2, t_start, t_start + budget_s * 0.94)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()
                best_n = n_try

    # Phase 4: Final polishing
    n = best_n
    h_cur = best_h.copy()
    h_cur, c5_cur = slsqp_polish(h_cur, n, [0.0002, 0.0001, 0.00005],
                                  t_start, t_start + budget_s * 0.97)
    if c5_cur < best_c5:
        best_c5 = c5_cur
        best_h = h_cur

    h_cur, c5_cur = coord_descent(best_h, best_n, best_c5,
                                  [0.02, -0.02, 0.01, -0.01, 0.005, -0.005, 0.002, -0.002],
                                  3, t_start, t_start + budget_s * 0.985)
    if c5_cur < best_c5:
        best_c5 = c5_cur
        best_h = h_cur

    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
