import time
from scipy.optimize import minimize

_global_n = 50

def constraint_fun(h):
    return np.sum(h) - _global_n / 2.0

def compute_score(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def corr_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def soft_max_obj(h, tau=150.0):
    c = corr_array(h)
    cm = np.max(c)
    return cm + np.log(np.sum(np.exp(tau * (c - cm)))) / tau

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = mask.sum()
        if cnt == 0:
            break
        h = np.clip(h + diff / cnt * mask, 0.0, 1.0)

def make_sinusoidal(n, freq=1.0, phase=0.0, amp=0.5):
    x = np.arange(n, dtype=float)
    h = 0.5 + amp * np.sin(2.0 * np.pi * freq * x / n + phase)
    return project_feasible(h, n)

def make_tanh(n, freq=1.0, k=3.0, phase=0.0):
    x = np.arange(n, dtype=float)
    h = 0.5 + 0.5 * np.tanh(k * np.sin(2.0 * np.pi * freq * x / n + phase))
    return project_feasible(h, n)

def grad_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        idx = np.arange(n - lag)
        g[idx + lag] += (1.0 - h[idx])
        g[idx] += -h[idx + lag]
    else:
        la = -lag
        idx = np.arange(la, n)
        g[idx - la] += (1.0 - h[idx])
        g[idx] += -h[idx - la]
    return g

def multi_lag_grad(h, top_m=3):
    n = len(h)
    c_raw = np.correlate(h, 1.0 - h, mode='full')
    c = c_raw * (2.0 / n)
    order = np.argsort(c)
    g = np.zeros(n)
    weights = np.array([1.0, 0.4, 0.2, 0.1, 0.05, 0.03][:top_m])
    for i, idx in enumerate(order[-top_m:]):
        lag = idx - (n - 1)
        g += weights[i] * grad_lag(h, lag)
    return g * (2.0 / n), c, order

def subgradient_descent(h0, n, steps, lr_init, decay):
    global _global_n
    _global_n = n
    h = h0.copy()
    best = h.copy()
    best_s = compute_score(h)
    lr = lr_init
    for t in range(steps):
        g, c, _ = multi_lag_grad(h, top_m=3)
        h = project_feasible(h - lr * g, n)
        lr *= decay
        s = compute_score(h)
        if s < best_s:
            best_s = s
            best = h.copy()
    return best, best_s

def simulated_annealing(h0, n, iters, t_init, t_final, rng):
    global _global_n
    _global_n = n
    h = h0.copy()
    best = h.copy()
    best_s = compute_score(h)
    for t in range(iters):
        temp = t_init * (t_final / t_init) ** (t / max(1, iters - 1))
        n_pert = rng.randint(1, min(4, n // 10 + 1))
        idx = rng.choice(n, size=n_pert, replace=False)
        h_try = h.copy()
        h_try[idx] += rng.uniform(-0.1, 0.1, size=n_pert)
        h_try = project_feasible(h_try, n)
        s_try = compute_score(h_try)
        delta = s_try - compute_score(h)
        if delta < 0 or rng.rand() < np.exp(-delta / max(temp, 1e-10)):
            h = h_try
        s = compute_score(h)
        if s < best_s:
            best_s = s
            best = h.copy()
        if best_s < 0.375 and temp < t_init * 0.1:
            break
    return best, best_s

def slsqp_polish(h0, n, maxiter=400):
    global _global_n
    _global_n = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun}]
    h_best = h0.copy()
    s_best = compute_score(h0)
    for obj in [soft_max_obj, compute_score]:
        try:
            res = minimize(obj, h_best, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-13})
            hc = project_feasible(res['x'], n)
            sc = compute_score(hc)
            if sc < s_best:
                s_best = sc
                h_best = hc
        except Exception:
            pass
    return h_best, s_best
