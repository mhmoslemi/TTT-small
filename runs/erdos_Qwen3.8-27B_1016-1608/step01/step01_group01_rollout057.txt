                continue
            best_d = 0.0
            best_v = best_c5
            deltas = np.linspace(d_lo, d_hi, num_tries)
            for d in deltas:
                if abs(d) < 1e-12:
                    continue
                ht = h.copy()
                ht[i] = hi + d
                ht[oidx] = h
                ht[i] = hi + d
                ht[i] = h[i] + d
                others_new = h[np.arange(n) != i] - d / (n - 1)
                full = np.empty(n)
                full[:i] = others_new[:i] if i > 0 else np.array([])
                full[i] = hi + d
                full[i+1:] = others_new[i:] if i < n-1 else np.array([])
                if full.min() < -1e-10 or full.max() > 1.0 + 1e-10:
                    continue
                v = compute_c5(full)
                if v < best_v - 1e-13:
                    best_v = v
                    best_d = d
            if best_d != 0.0:
                h[i] = hi + best_d
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                h[mask] -= best_d / (n - 1)
                h = np.clip(h, 0, 1)
                best_c5 = best_v
                improved = True
        if not improved:
            break
    return h, best_c5

def har_init(n, seed, mode):
    rng = np.random.RandomState(seed)
    t = np.linspace(0, 2, n, endpoint=False)
    h = np.zeros(n)
    h += 0.5
    h += 0.3 * np.sin(2 * np.pi * mode * t / 2.0)
    h += 0.15 * np.cos(2 * np.pi * (mode + 1) * t / 2.0)
    h += rng.normal(0, 0.05, n)
    return project_feasible(np.clip(h, 0, 1), n)

def binary_init(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    k = n // 2
    idx = rng.choice(n, k, replace=False)
    h[idx] = 1.0
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    n = len(initial_h_values)
    h = project_feasible(np.array(initial_h_values, dtype=float), n)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    # Phase 1: L-BFGS-B multi-temperature (30%)
    dl1 = t0 + budget_s * 0.30
    for temp in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
        if time.time() > dl1:
            break
        for pen in [1000.0, 5000.0]:
            if time.time() > dl1:
                break
            x = lbfgsb_optimize(best_h, n, temp, pen, 400, dl1)
            if np.all(np.isfinite(x)):
                h2 = project_feasible(x, n)
                c2 = compute_c5(h2)
                if c2 < best_c5:
                    best_c5, best_h = c2, h2.copy()

    # Phase 2: Coordinate descent (15%)
    dl2 = t0 + budget_s * 0.45
    best_h, best_c5 = coord_descent(best_h, n, 400, 40, dl2)

    # Phase 3: L-BFGS-B fine polish (15%)
    dl3 = t0 + budget_s * 0.60
    for temp in [0.0001, 0.00005, 0.00002, 0.00001]:
        if time.time() > dl3:
            break
        for pen in [10000.0, 50000.0]:
            if time.time() > dl3:
                break
            x = lbfgsb_optimize(best_h, n, temp, pen, 600, dl3)
            if np.all(np.isfinite(x)):
                h2 = project_feasible(x, n)
                c2 = compute_c5(h2)
                if c2 < best_c5:
                    best_c5, best_h = c2, h2.copy()

    # Phase 4: Multi-start with diverse initializations (20%)
    dl4 = t0 + budget_s * 0.80
    starts = []
    for r in range(4):
        sc = 0.03 + 0.02 * r
        starts.append(project_feasible(best_h + rng.normal(0, sc, n), n))
    starts.append(binary_init(n, seed + 100))
    starts.append(har_init(n, seed + 200, 1))
    starts.append(har_init(n, seed + 300, 2))
    starts.append(har_init(n, seed + 400, 3))
    starts.append(har_init(n, seed + 500, 5))
    starts.append(np.roll(best_h, n // 5).copy())
    starts.append(np.roll(best_h, -n // 5).copy())
    for hr in starts:
        if time.time() > dl4:
            break
        x = lbfgsb_optimize(hr, n, 0.001, 10000.0, 300, dl4)
        if np.all(np.isfinite(x)):
            h2 = project_feasible(x, n)
            c2 = compute_c5(h2)
            if c2 < best_c5:
                best_c5, best_h = c2, h2.copy()
                best_h, best_c5 = coord_descent(best_h, n, 50, 30,
                                               min(dl4, time.time() + 3))
        # Second pass with lower temp
        if time.time() < dl4:
            x = lbfgsb_optimize(best_h, n, 0.0001, 50000.0, 400, dl4)
            if np.all(np.isfinite(x)):
                h2 = project_feasible(x, n)
                c2 = compute_c5(h2)
                if c2 < best_c5:
                    best_c5, best_h = c2, h2.copy()

    # Phase 5: Final coordinate descent with fine grid (15%)
    dl5 = t0 + budget_s * 0.97
    best_h, best_c5 = coord_descent(best_h, n, 600, 60, dl5)

    # Final projection and verification
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n)
