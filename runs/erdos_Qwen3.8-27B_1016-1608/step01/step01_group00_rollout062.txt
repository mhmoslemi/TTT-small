```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Bisection for alpha
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - proj.sum()
    if abs(diff) > 1e-15:
        if diff > 0:
            mask = proj < 1.0
        else:
            mask = proj > 0.0
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def soft_max_obj(h, beta, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    s = beta * corr
    s_max = s.max()
    exp_s = np.exp(s - s_max)
    obj = (s_max + np.log(exp_s.sum())) / beta * dx
    obj += lam * (np.sum(h) - n / 2.0) ** 2
    return obj

def soft_max_grad(h, beta, lam):
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    s = beta * corr
    s_max = s.max()
    exp_s = np.exp(s - s_max)
    weights = exp_s / exp_s.sum()
    
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        w = weights[k_idx]
        if w < 1e-10:
            continue
        shift = (n - 1) - k_idx
        if shift == 0:
            grad += w * (1.0 - 2.0 * h)
        elif shift > 0:
            grad[:n - shift] += w * one_minus_h[shift:]
            grad[shift:] -= w * h[:n - shift]
        else:
            s_pos = -shift
            grad[s_pos:] += w * one_minus_h[:n - s_pos]
            grad[:n - s_pos] -= w * h[s_pos:]
    grad *= dx
    grad += 2.0 * lam * (np.sum(h) - n / 2.0)
    return grad

def lbfgsb_opt(h0, n, beta=30.0, lam=100.0, maxiter=1000):
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(soft_max_obj, h0, jac=soft_max_grad,
                       method='L-BFGS-B', bounds=bnds,
                       args=(beta, lam),
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
        h = project_feasible(res.x, n)
        return h
    except Exception:
        return h0

def slsqp_opt(h0, n, maxiter=500):
    bnds = [(0.0, 1.0)] * n
    def cons_fun(x):
        return np.sum(x) - n / 2.0
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(compute_c5, h0, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.fun < compute_c5(h0) + 1e-12:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0

def nm_opt(h0, n):
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(compute_c5, h0, method='Nelder-Mead',
                       bounds=bnds, options={'maxiter': 20000, 'xatol': 1e-12, 'fatol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h0

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    # Get initial
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_best = np.full(70, 0.5)
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)
    
    def remaining():
        return budget_s - (time.time() - t0)

    for beta in [15.0, 30.0, 60.0]:

        c = compute_c5(h)
        if c < c5_best:
            h_best, c5_best = h, c
    
    # Phase 2: SLSQP polish
    if remaining() > 60:
        h = slsqp_opt(h_best, n, maxiter=1000)
        c = compute_c5(h)
        if c < c5_best:
            h_best, c5_best = h, c
    
    # Phase 3: Multi-start with L-BFGS-B
    scales = [0.01, 0.03, 0.08, 0.15, 0.3, 0.5]
    attempt = 0
    while remaining() > budget_s * 0.5:
```

```python
def _subgrad(h0, n, steps=10000, lr_init=0.01):
    h = h0.copy()
    v = np.zeros(n)  # momentum
    mu = 0.9  # momentum coefficient
    best_c = compute_c5(h)
    best_h = h.copy()
    for t in range(steps):
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = np.argmax(corr)
        shift = (n - 1) - k_star
        grad = np.zeros(n)
        if shift == 0:
            grad[:] = 1.0 - 2.0 * h
        elif shift > 0:
            grad[:n-shift] += (1.0 - h[shift:])
            grad[shift:] -= h[:n-shift]
        else:
            sp = -shift
            grad[sp:] += (1.0 - h[:n-sp])
            grad[:n-sp] -= h[sp:]
        grad *= 2.0 / n  # dx
        lr = lr_init / (1.0 + t * 1e-4)
        v = mu * v - lr * grad
        h = project_feasible(h + v, n)
        c = compute_c5(h)
        if c < best_c:
```

```python
    dx = 2.0 / n
```
