        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

            if _time_left(start_t, budget) < 30:
                break
            h_opt = softmax_lbfgs(h_up4, n4, beta, maxiter=100)
            update(h_opt, n4)
            h_up4 = h_opt.copy()

    # Final polish: coordinate descent on best
    if _time_left(start_t, budget) > 15:
        h_p = project_to_set(best_h.copy(), best_n / 2.0)
        c_p = compute_c5(h_p, best_n)
        cs_deltas = [0.01, 0.02, 0.05, 0.1, 0.2, 0.5]
        for _ in range(3):
            improved = False
            for j in rng.permutation(best_n):
                orig = h_p[j]
                for d in cs_deltas:
                    for sgn in (1, -1):
                        nv = orig + sgn * d
                        if nv < 0 or nv > 1:
                            continue
                        h_p[j] = nv
                        h_p = project_to_set(h_p, best_n / 2.0)
                        c = compute_c5(h_p, best_n)
                        if c < c_p - 1e-12:
                            c_p = c
                            improved = True
                        else:
                            h_p = project_to_set(h_p, best_n / 2.0)
                            h_p[j] = orig
                if h_p[j] != orig:
                    h_p[j] = orig
                    h_p = project_to_set(h_p, best_n / 2.0)
            if not improved:
                break
        update(h_p, best_n)

    # Ensure final validity
    best_h = project_to_set(np.asarray(best_h, dtype=float), best_n / 2.0)
    best_c5 = compute_c5(best_h, best_n)
    assert np.all(best_h >= -1e-12) and np.all(best_h <= 1 + 1e-12)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h, best_n)
    return best_h, best_c5, best_n
