    return float(np.max(corr)) * dx

def repair(h, target):

    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-12:
            break
        if d > 0:
            m = h < 1.0 - 1e-12
            # Add to non-saturated

                h = np.clip(h + d/len(h), 0, 1)
            else:

                h = np.clip(h, 0, 1)

            else:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def repair(h, target):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-12:
            break
        if d > 0:
            m = h < 1.0 - 1e-12
        else:
            m = h > 1e-12
        cnt = m.sum()
        if cnt == 0:
            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _eval_pair(h, i, j, t, c):
    oi, oj = h[i], h[j]
    h[i] = t
    h[j] = c - t
    corr = np.correlate(h, 1.0 - h, mode="full")
    v = float(np.max(corr))
    h[i] = oi
    h[j] = oj
    return v

def coord_pass(h, n, target, dx, rng, n_pairs=500):
    """One pass of coordinate descent with random pair swaps."""
    for _ in range(n_pairs):
        i, j = int(rng.integers(n)), int(rng.integers(n))
        if i == j:
            continue
        c = h[i] + h[j]
        lo = max(0.0, c - 1.0)
        hi = min(1.0, c)
        if hi - lo < 1e-10:
            continue
        inv_phi = 0.6180339887
        a, b = lo, hi
        d_ = b - a
        c1 = b - inv_phi * d_
        c2 = a + inv_phi * d_
        f1 = _eval_pair(h, i, j, c1, c)
        f2 = _eval_pair(h, i, j, c2, c)
        for _ in range(35):
            if f1 < f2:
                b = c2
                c2 = c1
                f2 = f1
                c1 = b - inv_phi * (b - a)
                f1 = _eval_pair(h, i, j, c1, c)
            else:
                a = c1
                c1 = c2
                f1 = f2
                c2 = a + inv_phi * (b - a)
                f2 = _eval_pair(h, i, j, c2, c)
        t_star = (a + b) / 2
        h[i] = t_star
        h[j] = c - t_star
    h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_pass(h, n, target, dx, n_steps=200, lr_init=0.5):
    """Subgradient descent on the active correlation term."""
    for step in range(n_steps):
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_star = int(np.argmax(corr))
        # Gradient of corr[k_star] wrt h
        # corr[k] = sum_i h[i]*(1-h[i+k]) for valid i
        # d(corr[k])/dh[j] = (1-h[j+k]) + h[j-k]*(-1) for valid indices
        # More carefully:
        grad = np.zeros(n)
        # For the full correlation at index k_star:
        # corr_full[m] = sum_i h[i] * (1-h)[m-i] where 0<=i<n and 0<=m-i<n
        # = sum_i h[i] * (1-h[m-i])
        # d/dh[j]: appears when i=j: (1-h[m-j])
        #           and when m-i=j, i.e., i=m-j: h[m-j]*(-1) = -h[m-j]
        m = k_star
        # i ranges: max(0, m-(n-1)) to min(n-1, m)
        i_lo = max(0, m - (n - 1))
        i_hi = min(n - 1, m)
        for i in range(i_lo, i_hi + 1):
            j = i
            k = m - i
            # term: h[i] * (1-h[k])
            # d/dh[i]: (1-h[k])
            grad[i] += (1.0 - h[k])
            # d/dh[k]: h[i]*(-1)
            grad[k] -= h[i]
        # Step
        lr = lr_init / (1.0 + 0.01 * step)
        h_new = h - lr * grad
        h_new = repair(h_new, target)
        c_old = compute_c5(h, dx)
        c_new = compute_c5(h_new, dx)
        if c_new < c_old:
            h = h_new
        else:
            lr_init *= 0.9
    return h

def slsqp_optimize(h, n, target, dx, maxiter=300):
    """SLSQP local optimization."""
    bnds = [(0.0, 1.0)] * n
    
    def obj(x):
        return compute_c5(x, dx)
    
    def obj_grad(x):
        # Numerical gradient (central difference)
        eps = 1e-7
        g = np.zeros(n)
        base = compute_c5(x, dx)
        for i in range(n):
            xp = x.copy(); xp[i] += eps
            xm = x.copy(); xm[i] -= eps

Actually wait - is the function truly convex in t? Let's see:
f(t) = max_k corr_k(h with h_i=t, h_j=c-t)

Each corr_k is a quadratic in (t, c-t), but when we fix all other variables, it's:
corr_k = sum_i h_i(0)·(1-h_i(k)) + terms involving t and (c-t)

Specifically, for a given k:
- If i is involved: h_i changes to t, so term becomes t·(1-h[k-t_idx])
- If k_idx is involved: h_k changes to c-t, so term becomes h[i-idx]·(1-(c-t))

Each corr_k is linear in t (since each term involving t is either t·const or const·(c-t) which is linear in t). The max of linear functions is convex. So yes, f(t) is convex, and golden section will find the global minimum on [lo, hi].

This is a strong guarantee that makes the coordinate descent very effective.

            best_h = coord_pass(best_h, best_n, target, dx, rng2, n_pairs=300)
            best_h = _slsqp(best_h, best_n, target, dx, maxiter=400, ftol=1e-15)
        best_h = repair(best_h, target)
        best_c5 = compute_c5(best_h, dx)

    # Validate
    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_n / 2.0
    best_h = repair(best_h, target)
    best_c5 = compute_c5(best_h, 2.0 / best_n)

    return (best_h, best_c5, best_n)
