    dx = 2.0 / n

    h = np.clip(h.copy(), 0.0, 1.0)

        diff = target - s

            break
        h[mask] += diff / np.count_nonzero(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    """Round h to binary with exactly n/2 ones."""
    target = n // 2
    h_sorted = np.sort(h)[::-1]
    # Take top n/2 as 1
    threshold = h_sorted[target - 1]
    binary = (h >= threshold).astype(float)
    # Fix if not exactly target ones
    if np.sum(binary) > target:
        # Reduce: set some to 0 that are closest to threshold
        ones = np.where(binary == 1)[0]
        diffs = h[ones] - threshold
        order = np.argsort(diffs)
        excess = int(np.sum(binary)) - target
        for idx in order[:excess]:
            binary[ones[idx]] = 0
    elif np.sum(binary) < target:
        zeros = np.where(binary == 0)[0]
        diffs = threshold - h[zeros]
        order = np.argsort(diffs)
        deficit = target - int(np.sum(binary))
        for idx in order[:deficit]:

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            h[:] = target / n
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):

    idx = np.argsort(h)[::-1]
    binary = np.zeros(n)
    binary[idx[:target]] = 1.0
    return binary

def slsqp_local(h0, n, maxiter=200, ftol=1e-14):
    target = n / 2.0
    bnds = [(0.0, 1.0)] * n
    def obj(x):
        return compute_c5(x)
    def constr(x):
        return np.sum(x) - target
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bnds,
                       constraints=[{'type': 'eq', 'fun': constr}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        x = np.clip(res.x, 0.0, 1.0)
        x = repair(x, n)
        if compute_c5(x) <= obj(h0) + 1e-12:

    except Exception:
        pass
    return h0
