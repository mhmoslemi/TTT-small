def soft_max_c5(h, n, alpha=100.0):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    # Soft max
    m = np.max(corr)
    vals = np.exp(alpha * (corr - m))
    s = vals.sum()
    val = m + np.log(s) / alpha
    grad = vals / s  # These are the weights for the sub-gradient
    return val, corr, grad

Actually, for the binary case with a swap at positions i and j, the change in corr at lag m is:
- h[i] changes from 1 to 0 (or 0 to 1)
- h[j] changes from 0 to 1 (or 1 to 0)

The change in corr[m] = sum of h[a] * (1-h[a+lag_m]) where lag_m = m-(n-1):
- If the swap changes h[i] from a_old to a_new:
  - For each m where i is in the valid range: corr[m] changes by (a_new - a_old) * (1-h[i+lag_m])
  - For each m where i-lag_m is in the valid range (i is the "second" factor): corr[m] changes by -(a_new - a_old) * h[i-lag_m]

This is O(n) per swap (since i affects at most 2n lags, but actually for a fixed i, it affects all lags k such that 0≤i+k<n and 0≤i<n, which is at most n lags for the first factor and n for the second, total 2n but bounded by 2n-1 total lags).

def project_to_set(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        hp = np.clip(h - mid, 0.0, 1.0)
        if float(np.sum(hp)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def upsample_fn(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, h)

def grad_at_lag(h, n, k):
    dx = 2.0 / n
    g = np.zeros(n)
    if k >= 0:
        if k < n:
            g[:n-k] += (1.0 - h[k:n])
            g[k:] -= h[:n-k]
    else:
        kk = -k
        if kk < n:
            g[kk:] += (1.0 - h[:n-kk])
            g[:n-kk] -= h[kk:]
    return g * dx

def binary_sa(h, n, rng, n_iter=200000, T0=0.3, T1=0.005, time_limit=120.0):

    n_ones = int(np.sum(h))
    if n_ones != target:
        h = np.where(h >= 0.5, 1.0, 0.0)
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) > target:
            for idx in rng.choice(ones, len(ones)-target, replace=False):
                h[idx] = 0.0

One more optimization thought: for the binary SA, instead of computing the full correlation each time, I can compute the change incrementally. When I swap h[i] (was 1, now 0) and h[j] (was 0, now 1):

The change in c_k for each k:
- At position i (1→0): 
  - As h[n+k]=h[i] (first factor): n = i-k, c_k changes by -1*(1-h[i-k]) = -(1-h[i-k])
  - In (1-h[n])=(1-h[i]) (second factor): n = i, c_k changes by h[i+k]*0 - 0... wait.
  
  Actually, when h[i] changes from 1 to 0:
  - Terms where i is the first factor h[i] = h[n+k]: c_k loses (1-h[n]) = (1-h[i-k])
  - Terms where i is in (1-h[i]): (1-h[i]) changes from 0 to 1, so c_k gains h[n+k] = h[i+k]

import numpy as np
import time
try:
    import cvxpy as cp
    HAVE_CP = True
except ImportError:
    HAVE_CP = False

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_to_set(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        hp = np.clip(h - mid, 0.0, 1.0)
        if float(np.sum(hp)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def upsample_fn(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, h)

def grad_at_lag(h, n, k):
    dx = 2.0 / n
    g = np.zeros(n)
    if k >= 0:
        if k < n:
            g[k:] += (1.0 - h[:n - k])
            g[:n - k] -= h[k:]
    else:
        kk = -k
        if kk < n:
            g[:n - kk] += (1.0 - h[kk:])
            g[kk:] -= h[:n - kk]
    return g * dx

def binary_sa(h, n, rng, n_iter=200000, T0=0.3, T1=0.003, time_limit=120.0):
    t_s = time.time()
    h = h.astype(float).copy()
    target = n // 2
    n_ones = int(np.sum(h))
    if n_ones != target:
        h = np.where(h >= 0.5, 1.0, 0.0)
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) > target:
            for idx in rng.choice(ones, len(ones) - target, replace=False):
                h[idx] = 0.0
        elif len(ones) < target:
            for idx in rng.choice(zeros, target - len(ones), replace=False):
                h[idx] = 1.0
    best = compute_c5(h, n)
    best_h = h.copy()
    cur = h.copy()
    for it in range(n_iter):
        if time.time() - t_s > time_limit:
            break
        T = T0 * (T1 / T0) ** (it / max(1, n_iter))
        i = rng.randint(n)
        j = rng.randint(n)
        if cur[i] == cur[j]:
            continue
        cur[i], cur[j] = cur[j], cur[i]
        v = compute_c5(cur, n)
        if v < best:
            best = v
            best_h = cur.copy()
        elif v - best > 0 and rng.rand() >= np.exp(-(v - best) / max(T, 1e-8)):
            cur[i], cur[j] = cur[j], cur[i]
    return best_h, best

def soft_max_opt(h, n, n_iter=40000, alpha=200.0, top_m=5, time_limit=60.0):
    t_s = time.time()
    h = h.copy()
    target = n / 2.0
    corr = get_corr(h, n)
    best = float(np.max(corr))
    best_h = h.copy()
    step = 0.02
    for it in range(n_iter):
        if time.time() - t_s > time_limit:
            break
        top_idx = np.argpartition(corr, -top_m)[-top_m:]
        top_vals = corr[top_idx]
        wmax = top_vals.max()
        exps = np.exp(alpha * (top_vals - wmax))
        w = exps / exps.sum()
        g = np.zeros(n)
        for i in range(top_m):
            k = int(top_idx[i]) - (n - 1)
            g += w[i] * grad_at_lag(h, n, k)
        h_new = h - step * g
        h_new = project_to_set(h_new, target)
        corr_new = get_corr(h_new, n)
        v = float(np.max(corr_new))
        if v < best:
            best = v
            best_h = h_new.copy()
            step = min(step * 1.02, 0.08)
        else:
            step *= 0.97
        h = h_new
        corr = corr_new
    return best_h, best

def lp_refine(h0, n, n_outer=15, top_m=30, time_limit=60.0):
    if not HAVE_CP:
        return soft_max_opt(h0, n, n_iter=20000, alpha=200, top_m=5, time_limit=time_limit)
    t_s = time.time()
    h = h0.copy()
    target = n / 2.0
    best = compute_c5(h, n)
    best_h = h.copy()
    for outer in range(n_outer):
        if time.time() - t_s > time_limit:
            break
        corr = get_corr(h, n)
        top_idx = np.argpartition(corr, -top_m)[-top_m:]
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        cons = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
        for idx in top_idx:
            k = int(idx) - (n - 1)
            g0 = corr[idx]
            gk = grad_at_lag(h, n, k)
            cons.append(g0 + gk @ (h_var - h) <= t_var)
        prob = cp.Problem(cp.Minimize(t_var), cons)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, verbose=False)
            except Exception:
                break
        if h_var.value is None:
            break
        h_new = np.clip(h_var.value, 0.0, 1.0)
        h_new = project_to_set(h_new, target)
        v = compute_c5(h_new, n)
        if v < best:
            best = v
            best_h = h_new.copy()
        h = project_to_set(0.5 * h + 0.5 * h_new, target)
    return best_h, best

def coordinate_refine(h, n, rng, n_sweeps=25, time_limit=60.0):
    t_s = time.time()
    target = n / 2.0
    best = compute_c5(h, n)
    best_h = h.copy()
    for _ in range(n_sweeps):
        if time.time() - t_s > time_limit:
            break
        improved = False
        for j in range(n):
            old = h[j]
            cands = [0.0, 1.0, max(0.0, old - 0.15), min(1.0, old + 0.15),
                     max(0.0, old - 0.05), min(1.0, old + 0.05)]
            bc, bv = old, best
            for c in cands:
                if abs(c - old) < 1e-8:
                    continue
                h[j] = c
                h_p = project_to_set(h, target)
                v = compute_c5(h_p, n)
                if v < bv:
                    bv = v
                    bc = c
            if bc != old:
                h = project_to_set(np.array(h, copy=True), target)
                best = bv
                best_h = h.copy()
                improved = True
            else:
                h[j] = old
                h = project_to_set(h, target)
        if not improved:
            break
    return best_h, best

def make_pattern(n, rng, ptype=0):
    h = np.zeros(n)
    if ptype == 0:
        h[rng.choice(n, n // 2, replace=False)] = 1.0
    elif ptype == 1:
        x = np.linspace(0, 2, n)
        h = 0.5 + 0.4 * np.sin(2 * np.pi * x + rng.uniform(0, 2 * np.pi))
        h = project_to_set(h, n / 2.0)
    elif ptype == 2:
        h[:n // 4] = 1.0
        h[3 * n // 4:] = 1.0
        h = h + rng.normal(0, 0.2, n)
        h = project_to_set(h, n / 2.0)
    else:
        half = n // 2
        for p in range(2):
            block = np.zeros(half)
            block[rng.choice(half, half // 2, replace=False)] = 1.0
            h[p * half:(p + 1) * half] = block
    return h

def run(seed=42, budget_s=1100, **kwargs):

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = np.inf
    best_n = 0

    def _elapsed():
        return time.time() - t_start

    def _remain():
        return max(0.0, budget_s - _elapsed())

    def _update(h, n):
        nonlocal best_h, best_c5, best_n
        h = np.asarray(h, dtype=float)
        h = project_to_set(h, n / 2.0)
        c = compute_c5(h, n)
        if c < best_c5:
            best_h = h.copy()
            best_c5 = c
            best_n = n

    n1 = 90
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        if len(h_init) != n1:
            h_init = upsample_fn(h_init, n1)
    except Exception:
        h_init = make_pattern(n1, rng, 0)
    h_init = project_to_set(h_init, n1 / 2.0)
    _update(h_init, n1)

    # Phase 1a: Binary SA from initial
    h_bin = np.where(h_init > 0.5, 1.0, 0.0)
    h_bin, c = binary_sa(h_bin, n1, rng, n_iter=400000, T0=0.25, T1=0.002, time_limit=60.0)
    _update(h_bin, n1)

    # Phase 1b: Continuous refinement from best
    _update(soft_max_opt(best_h, best_n, n_iter=50000, alpha=300, top_m=5, time_limit=50.0)[0], best_n)

    # Phase 1c: LP refinement
    _update(lp_refine(best_h, best_n, n_outer=15, top_m=30, time_limit=40.0)[0], best_n)

    # Phase 1d: Multiple random starts with SA + continuous
    pcount = 0
    while _remain() > 200 and pcount < 10:
        ptype = pcount % 4
        h_r = make_pattern(n1, rng, ptype)
        h_b, c = binary_sa(h_r, n1, rng, n_iter=250000, T0=0.25, T1=0.003, time_limit=35.0)
        _update(h_b, n1)
        if _remain() > 180:
            h_c, c2 = soft_max_opt(h_b.astype(float), n1, n_iter=40000, alpha=250, top_m=5, time_limit=25.0)
            _update(h_c, n1)
        if _remain() > 160:

        _update(h_up, n2)
        _update(soft_max_opt(h_up, n2, n_iter=60000, alpha=400, top_m=6, time_limit=60.0)[0], n2)
        h_bin2 = np.where(h_up > 0.5, 1.0, 0.0)
        h_b2, c = binary_sa(h_bin2, n2, rng, n_iter=300000, T0=0.2, T1=0.003, time_limit=50.0)
        _update(h_b2, n2)
        _update(lp_refine(best_h, best_n, n_outer=12, top_m=30, time_limit=40.0)[0], best_n)
        if _remain() > 80:
            _update(coordinate_refine(best_h, best_n, rng, n_sweeps=35, time_limit=50.0)[0], best_n)
        # Another round of SA at n=180 from best

                                n_iter=200000, T0=0.15, T1=0.002, time_limit=40.0)
            _update(h_b3, best_n)

    # Phase 3: n=360
    if _remain() > 80:
        n3 = 360
        h_up3 = project_to_set(upsample_fn(best_h, n3), n3 / 2.0)
        _update(h_up3, n3)
        _update(soft_max_opt(h_up3, n3, n_iter=40000, alpha=500, top_m=6, time_limit=50.0)[0], n3)

    best_h = project_to_set(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h, best_n)
    return best_h, best_c5, best_n
