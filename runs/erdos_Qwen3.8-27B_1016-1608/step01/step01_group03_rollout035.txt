def coordinate_descent(h, n, target, dx, max_passes=100):

        improved = False

            # Try to optimize h[i] given all other coordinates fixed
            # corr[k] = sum_j h_orig[j]*(1-h_orig[j+m_k]) + adjustment for h[i]

def pair_cd(h, n, target, dx, max_passes=50):
    """Pair coordinate descent."""
    h = h.astype(float).copy()
    rng = np.random.RandomState(42)
    
    for p in range(max_passes):
        improved = False
        # Try all pairs (or a subset)
        pairs = [(i, j) for i in range(n) for j in range(i+1, n)]
        rng.shuffle(pairs)
        
        for i, j in pairs:
            # Find optimal δ for h[i]+=δ, h[j]-=δ
            # δ range: max(-h[i], h[j]-1) <= δ <= min(1-h[i], h[j])
            d_lo = max(-h[i], h[j] - 1.0)
            d_hi = min(1.0 - h[i], h[j])
            if d_hi - d_lo < 1e-10:
                continue
            
            # Binary/ternary search on δ
            best_d = 0.0
            best_c5_p = compute_c5_repair_pair(h, i, j, 0.0, dx)
            
            # Try golden section or just a few points
            for frac in [0.25, 0.4, 0.5, 0.6, 0.75, 0.1, 0.9, 0.05, 0.95]:
                d = d_lo + frac * (d_hi - d_lo)
                c5_d = compute_c5_repair_pair(h, i, j, d, dx)
                if c5_d < best_c5_p:
                    best_c5_p = c5_d
                    best_d = d
            
            # Refine around best_d
            if best_d != 0.0:
                d_span = (d_hi - d_lo) * 0.05
                for frac in np.linspace(0.2, 0.8, 7):
                    d = best_d - d_span + frac * 2 * d_span
                    if d < d_lo or d > d_hi:
                        continue
                    c5_d = compute_c5_repair_pair(h, i, j, d, dx)
                    if c5_d < best_c5_p:
                        best_c5_p = c5_d
                        best_d = d
                
                if best_d != 0.0:
                    h[i] += best_d
                    h[j] -= best_d
                    improved = True
        
        if not improved:
            break
    
    return h

def solve_lp(n, target, dx):

    h = cp.Variable(n)
