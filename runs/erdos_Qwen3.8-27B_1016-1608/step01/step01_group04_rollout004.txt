        c = compute_c5(hn)
        if c < bc:
            bc, bh = c, hn

    # Phase 2: Binarize + swap at original resolution
    if time.time() - t0 < 300:
        hb, cb = swap_search(bh, bn, 100)
        if cb < bc:
            bc, bh, bn = cb, hb.copy(), bn

    # Phase 3: SLSQP refinement after swap
    for eps in [0.002, 0.0005, 0.0001]:
        if time.time() - t0 > 350:
            break
        hn = slsqp(bh, bn, eps, 400, 1e-14)
        c = compute_c5(hn)
        if c < bc:
            bc, bh = c, hn

    # Phase 4: Random restarts at original resolution
    for i in range(60):
        if time.time() - t0 > 450:
            break
        r = i % 5
        if r == 0:
            hp = bh + np.random.randn(bn) * 0.04
        elif r == 1:
            hp = bh.copy()
            s = np.random.randint(0, max(1, bn - 8))
            l = np.random.randint(3, min(9, bn - s))
            hp[s:s + l] = 1.0 - hp[s:s + l]
        elif r == 2:
            k2 = np.random.randint(1, 10)
            ph = np.random.uniform(0, 2 * np.pi)
            hp = bh + 0.06 * np.sin(2 * np.pi * k2 * np.arange(bn) / bn + ph)
        elif r == 3:
            hp = bh.copy()
            idx = np.random.choice(bn, size=max(1, bn // 6), replace=False)
            hp[idx] = 1.0 - hp[idx]
        else:
            hp = bh + np.random.randn(bn) * 0.08
        hp = repair(hp, bn)
        hn = slsqp(hp, bn, 0.005, 300, 1e-12)
        c = compute_c5(hn)
        if c < bc:
            bc, bh = c, hn
        # Also try binarize+swap from this
        hb2, cb2 = swap_search(hn, bn, 30)
        if cb2 < bc:
            bc, bh = cb2, hb2.copy()

    # Phase 5: Try higher resolution (2x)
    n2 = bn * 2
    if time.time() - t0 < 900:
        for start_type in range(4):
            if time.time() - t0 > 900:
                break
            if start_type == 0:
                h2 = interp_up(bh, bn, n2)
            elif start_type == 1:
                h2 = np.repeat(bh, 2)
                h2 = h2[:n2]
                if len(h2) < n2:
                    h2 = np.pad(h2, (0, n2 - len(h2)))
            elif start_type == 2:
                p = np.random.choice([3, 4, 5, 6, 7, 8, 10])
                h2 = np.array([1.0 if (i % p) < p // 2 else 0.0 for i in range(n2)])
            else:
                h2 = 0.5 + 0.45 * np.sin(2 * np.pi * np.random.randint(1, 5) * np.arange(n2) / n2)
            h2 = repair(h2, n2)
            for eps in [0.01, 0.003, 0.001]:
                if time.time() - t0 > 900:
                    break
                h2 = slsqp(h2, n2, eps, 300, 1e-13)
            c2 = compute_c5(h2)
            if c2 < bc:
                bc, bh, bn = c2, h2, n2
            # Swap search at higher res
            if time.time() - t0 < 900:
                hb2, cb2 = swap_search(h2, n2, 40)
                if cb2 < bc:
                    bc, bh, bn = cb2, hb2, n2

    # Phase 6: Try 4x resolution
    n4 = bn0 * 4 if 'bn0' in dir() else n0 * 4
    if time.time() - t0 < 1000 and n4 <= 320:
        h4 = interp_up(bh, bn, n4)
        h4 = repair(h4, n4)
        for eps in [0.01, 0.003, 0.001, 0.0003]:
            if time.time() - t0 > 1000:
                break
            h4 = slsqp(h4, n4, eps, 300, 1e-13)
        c4 = compute_c5(h4)
        if c4 < bc:
            bc, bh, bn = c4, h4, n4
        if time.time() - t0 < 1000:
            hb4, cb4 = swap_search(h4, n4, 30)
            if cb4 < bc:
                bc, bh, bn = cb4, hb4, n4

    # Phase 7: Final polish
    for eps in [0.0001, 0.00005, 0.00002]:
        if time.time() - t0 > budget_s - 30:
            break
        hn = slsqp(bh, bn, eps, 500, 1e-15)
        c = compute_c5(hn)
        if c < bc:
            bc, bh = c, hn

    # Tiny random perturbations
    for _ in range(5000):
        if time.time() - t0 > budget_s - 5:
            break
        hp = bh + np.random.randn(bn) * 0.003
        hp = repair(hp, bn)
        c = compute_c5(hp)
        if c < bc:
            bc, bh = c, hp
        hp = np.clip(bh + np.random.randn(bn) * 0.01, 0, 1)
        hp = repair(hp, bn)
        c = compute_c5(hp)
        if c < bc:
            bc, bh = c, hp

    # Final swap
    if time.time() - t0 < budget_s - 5:
        hb, cb = swap_search(bh, bn, 20)
        if cb < bc:
            bc, bh = cb, hb

    fh = repair(bh, bn)
    fc = compute_c5(fh)
    if fc > bc:
        fh, fc = bh, bc
    return (fh.tolist(), float(fc), bn)
