def compute_score(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return np.max(c) * dx

def soft_max_score(h, tau):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cm = np.max(c)
    return cm + np.log(np.sum(np.exp(tau * (c - cm)))) / tau

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h = np.clip(h + diff / cnt * mask, 0.0, 1.0)
    return h

def paired_cd(h, n_steps, rng):
    n = len(h)
    h = h.copy()
    score = compute_score(h)
    gr = 0.618
    for _ in range(n_steps):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            j = (i + 1) % n
        t_max = min(h[i], 1.0 - h[j])
        t_min = max(-h[i], h[j] - 1.0)
        if t_max - t_min < 1e-14:
            continue
        lo, hi = t_min, t_max
        for _ in range(12):
            t1 = lo + gr * (hi - lo)
            t2 = lo + (1.0 - gr) * (hi - lo)
            h[i] += t1; h[j] -= t1
            s1 = compute_score(h)
            h[i] -= t1; h[j] += t1
            h[i] += t2; h[j] -= t2
            s2 = compute_score(h)
            h[i] -= t2; h[j] += t2
            if s1 < s2:
                hi = t1
            else:
                lo = t2
        t_opt = (lo + hi) / 2.0
        h[i] += t_opt; h[j] -= t_opt
        h = np.clip(h, 0.0, 1.0)
        ns = compute_score(h)
        if ns >= score:
            h[i] -= t_opt; h[j] += t_opt
            h = np.clip(h, 0.0, 1.0)
    return h, compute_score(h)
