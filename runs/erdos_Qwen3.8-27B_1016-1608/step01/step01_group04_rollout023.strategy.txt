def project_to_constraint(h):
    """Adjust h to maintain sum(h) = n/2 while projecting to [0,1]"""
    s = np.sum(h)
    adjustment = (n/2 - s)/(n*0.9)  # Small penalty to ensure movement
    new_h = h + adjustment
    new_h = np.clip(new_h, 0, 1)
    # Final adjustment to meet constraint
    s_final = np.sum(new_h)
    if abs(s_final - n/2) > 1e-6:
        # Gradual adjustment
        while abs(np.sum(new_h) - n/2) > 1e-6:
            if np.sum(new_h) < n/2:
                # Only zeros exist, must add ones
                idx = np.argmax(1 - new_h)
                delta = min(1, 2 * (n/2 - s_final))
                new_h[idx] += delta
            else:
                # Over saturation, must remove ones
                idx = np.argmax(new_h)
                delta = min(1, 2 * (s_final - n/2))
                new_h[idx] -= delta
                if new_h[idx] < 0:
                    new_h[idx] = 0
    return new_h

def evaluate_objective(h):
