import numpy as np
from scipy.optimize import minimize
import time

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def _fix_sum(h, n, tol=1e-14, maxiter=200):
    target = 0.5 * n
    for _ in range(maxiter):
        d = target - np.sum(h)
        if abs(d) < tol:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        k = int(np.sum(mask))
        if k == 0:
            break
        h = h + (d / k) * mask
        h = np.clip(h, 0.0, 1.0)
    return h

def _obj_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(c)
    a = np.clip(alpha * (c - m), -100.0, 100.0)
    e = np.exp(a)
    Z = np.sum(e)
    p = e / Z
    f = m + np.log(Z) / alpha
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = p[k]
        if wk < 1e-16:
            continue
        s = k - (n - 1)
        j1 = max(0, -s)
        j2 = min(n, n - s)
        if j1 < j2:
            grad[j1:j2] += wk * (1.0 - h[j1 + s:j2 + s])
        j3 = max(0, s)
        j4 = min(n, n + s)
        if j3 < j4:
            grad[j3:j4] -= wk * h[j3 - s:j4 - s]
    grad *= dx
    return f, grad

        return None
    bnds = [(0.0, 1.0)] * n
    
    def obj(x):
        return _obj_grad(x, alpha)[0]
    
    def grad(x):
        return _obj_grad(x, alpha)[1]
    
    def con_fun(x):
        return np.sum(x) - 0.5 * n
    
    def con_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None:
            return _fix_sum(np.clip(res.x, 0.0, 1.0), n)
    except Exception:
        pass
    return None

def _project(h, n):
    """Project onto {0<=h<=1, sum(h)=n/2}"""
    h = np.clip(h, 0.0, 1.0)
    return _fix_sum(h, n)

def _subgrad_polish(h, n, n_iter, t0, budget):
    """Projected subgradient method to polish."""
    dx = 2.0 / n
    target = 0.5 * n
    best_h = h.copy()
    best_c = _c5(h)
    for it in range(n_iter):
        if budget - (time.time() - t0) < 2:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        k_star = np.argmax(c)
        s = k_star - (n - 1)
        # subgradient of max at k_star: gradient of c_{k_star}
        g = np.zeros(n)
        j1 = max(0, -s)
        j2 = min(n, n - s)
        g[j1:j2] = (1.0 - h[j1 + s:j2 + s])
        j3 = max(0, s)
        j4 = min(n, n + s)
        g[j3:j4] -= h[j3 - s:j4 - s]
        g *= dx
        
        # step size

        c_new = _c5(h_new)
        if c_new < best_c:

        h = h_new
    return best_h, best_c

def _interp_to(h, n_new):

    xo = np.linspace(0, 2, n_old, endpoint=False)
    xn = np.linspace(0, 2, n_new, endpoint=False)
