def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def compute_all_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        s = np.sum(h)
        d = target_sum - s
        if abs(d) < 1e-15:
            break
        if d > 0:
            mask = h < 1.0 - 1e-13
        else:
            mask = h > 1e-13
        c = np.sum(mask)
        if c == 0:
            break
        h[mask] += d / c
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_gradient(h, n):
    """Compute gradient of all correlations. Returns array shape (2n-1, n)."""
    grad = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        s = m - (n - 1)
        if s >= 0:
            lo1, hi1 = 0, n - 1 - s
            lo2, hi2 = s, n - 1
            for i in range(lo1, hi1 + 1):

            for i in range(lo2, hi2 + 1):
                grad[m, i] += (-h[i - s])

            lo2, hi2 = 0, n - 1 - s
            for i in range(lo1, hi1 + 1):
                grad[m, i] += (1.0 - h[i - s])
            for i in range(lo2, hi2 + 1):
                grad[m, i] += (-h[i + s])
    return grad
