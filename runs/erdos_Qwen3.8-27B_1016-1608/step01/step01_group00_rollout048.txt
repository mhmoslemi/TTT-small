```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def grad_corr(h, lag):
    n = len(h)
    dx = 2.0 / n
    shift = lag - n + 1
    g = np.zeros(n)
    if shift >= 0:
        g[:n - shift] += (1.0 - h[shift:n])
        g[shift:] -= h[:n - shift]
    else:
        s = -shift
        g[s:] += (1.0 - h[:n - s])
        g[:n - s] -= h[s:]
    return g * dx

def repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h < 1.0 - 1e-13) if diff > 0 else (h > 1e-13)
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_topm(h, m):
    c = compute_corr(h)
    idx = np.argpartition(c, -m)[-m:]
    return float(np.sum(c[idx]))

def grad_topm(h, m):
    c = compute_corr(h)
    idx = np.argpartition(c, -m)[-m:]
    g = np.zeros(len(h))
    for lag in idx:
        g += grad_corr(h, lag)
    return g

def obj_max(h, _):
    return compute_c5(h)

def grad_max(h, _):
    c = compute_corr(h)
    lag = int(np.argmax(c))
    return grad_corr(h, lag)

def lbfgs(h0, m, maxiter=3000):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_topm, h0, args=(m,), jac=grad_topm,
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8})
        return repair(res.x, n)
    except Exception:
        return h0

def lbfgs_max(h0, maxiter=5000):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_max, h0, args=(0,), jac=grad_max,
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-9})
        return repair(res.x, n)
    except Exception:
        return h0

def perturb_gauss(h, scale):
    return repair(h + np.random.randn(len(h)) * scale, len(h))

def perturb_flip(h, n_flip):
    h2 = h.copy()
    n = len(h)
    idx = np.random.choice(n, size=n_flip, replace=False)
    h2[idx] = 1.0 - h2[idx]
    return repair(h2, n)

def perturb_block(h, block_sz):
    h2 = h.copy()
    n = len(h)
    start = np.random.randint(0, n - block_sz)
    h2[start:start + block_sz] = 1.0 - h2[start:start + block_sz]
    return repair(h2, n)

    np.random.seed(seed)
    t0 = time.time()

    try:
        h_best = np.array(initial_h_values, dtype=float).copy()

    n = len(h_best)

            h = lbfgs(h_best, m, maxiter=2000)
            c = compute_c5(h)
            if c < c5_best:
                h_best, c5_best = h, c

    # Phase 2: Multi-restart search
    m_values = [3, 5, 8, 4]
    m_idx = 0
    stall = 0
    while time.time() - t0 < budget_s * 0.75:
        m = m_values[m_idx % len(m_values)]

        elif strategy == 1:

            stall = 0

    c5_best = compute_c5(h_best)
    return (h_best, c5_best, n)
```
