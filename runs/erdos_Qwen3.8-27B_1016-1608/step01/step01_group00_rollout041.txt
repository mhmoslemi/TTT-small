import numpy as np
from scipy.optimize import minimize
import time

# Global state for constraint functions
_g_n = [70]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            h[:] = 0.5
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _obj_func(x):
    return compute_c5(x)

def _eq_func(x):
    return np.sum(x) - _g_n[0] * 0.5

def _eq_jac_func(x):
    return np.ones(len(x))

def slsqp_opt(h0, n, maxiter):
    _g_n[0] = n
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_func, 'jac': _eq_jac_func}]
    try:
        res = minimize(_obj_func, h0, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        h_new = repair(np.clip(res.x, 0.0, 1.0), n)
        if compute_c5(h_new) < compute_c5(h0) + 1e-14:
            return h_new
    except Exception:
        pass
    return h0

def coord_desc(h, n, n_sweeps, rng, deadline):
    """2D coordinate descent."""
    h = h.copy()
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        for _ in range(n):
            if time.time() > deadline:
                break
            idx = rng.choice(n, size=2, replace=False)
            i, j = int(idx[0]), int(idx[1])
            s = h[i] + h[j]
            lo = max(0.0, s - 1.0)
            hi = min(1.0, s)
            if hi - lo < 1e-10:
                continue
            # Sample
            ts = np.linspace(lo, hi, 200)
            best_t = lo
            best_c = np.inf
            for t in ts:
                h[i] = t
                h[j] = s - t
                c = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
                if c < best_c:
                    best_c = c
                    best_t = t
            h[i] = best_t
            h[j] = s - best_t
    return h
