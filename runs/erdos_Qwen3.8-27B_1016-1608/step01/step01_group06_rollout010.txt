    w = np.exp(tau * (c - cm))
    wsum = np.sum(w)

    n = len(h)

    """Subgradient descent with adaptive step size."""
    alpha = 0.01
    best_local = h.copy()
    best_score = compute_score(h)
    for it in range(n_iter):
        if time.time() - t0 > budget_frac * 1100:

        cm = np.max(c)

            active = np.array([np.argmax(c)])
            n_act = 1
        # Average subgradient over active lags
        grad = np.zeros(n)
        for ki in active:
            k = int(ki) - (n - 1)
            if k == 0:

            elif k > 0:
                g = np.zeros(n)
                g[:n-k] += (1.0 - h[k:])
                g[k:] -= h[:n-k]

                g = np.zeros(n)

                g[:n-kk] -= h[kk:]

import numpy as np

from scipy.optimize import minimize

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_score(h):
    return float(np.max(compute_corr(h)))

def _corr_grad_one(h, k):
    n = len(h)
    if k == 0:
        return 1.0 - 2.0 * h
    if k > 0:
        g = np.zeros(n)
        g[:n-k] += 1.0 - h[k:]
        g[k:] -= h[:n-k]
        return g
    kk = -k
    g = np.zeros(n)
    g[kk:] += 1.0 - h[:n-kk]
    g[:n-kk] -= h[kk:]
    return g

def _softmax_obj(h, tau):
    c = compute_corr(h)
    cm = np.max(c)
    return cm + float(np.log(np.sum(np.exp(tau * (c - cm))))) / tau

def _softmax_pen(h, tau, mu):
    n = len(h)
    return _softmax_obj(h, tau) + mu * (np.sum(h) - n / 2.0) ** 2

def _softmax_pen_jac(h, tau, mu):
    n = len(h)
    c = compute_corr(h)
    cm = np.max(c)
    w = np.exp(tau * (c - cm))
    w /= np.sum(w)
    grad = np.zeros(n)
    for ki in range(2 * n - 1):
        if w[ki] < 1e-14:
            continue
        grad += w[ki] * _corr_grad_one(h, ki - (n - 1))
    grad += 2.0 * mu * (np.sum(h) - n / 2.0)
    return grad

def project_feasible(h, n):

                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
        return project_feasible(res['x'], n)
    except Exception:
        return h.copy()
