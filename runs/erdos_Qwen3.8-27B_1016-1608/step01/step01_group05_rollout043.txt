def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_set(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        hp = np.clip(h - mid, 0.0, 1.0)
        if float(np.sum(hp)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def upsample_h(h, n_new):
    x_old = np.linspace(0.0, 2.0, len(h), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)

def make_binary(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h

            old = h[j]
            steps = [0.005, 0.01, 0.025, 0.05, 0.1, 0.25]
            cands = [0.0, 1.0]
            for s in steps:
                s2 = s * step_scale
                cands.append(max(0.0, old - s2))
                cands.append(min(1.0, old + s2))
            cands = np.unique(np.array(cands))

            if len(cands) == 0:

            bc, bv = old, best
            for c in cands:
                h[j] = c
                h = project_to_set(h, target)
                v = compute_c5(h, n)
                if v < bv:
                    bv, bc = v, float(c)
            if bc != old:
                h[j] = bc
                best = bv
                improved = True
            else:
                h[j] = old
                h = project_to_set(h, target)
        if not improved:
            break
    h = project_to_set(h, target)
    return h, compute_c5(h, n)

        T *= 0.9998

        h[j] = max(0.0, min(1.0, old + rng.normal(0.0, 0.06)))

        nc5 = compute_c5(h, n)
        if nc5 <= cc5 or rng.random() < np.exp(-(nc5 - cc5) / max(T, 1e-10)):

            if nc5 < bc5:
                bc5, bh = nc5, h.copy()
        else:

    return bh, bc5

def _sm_obj(x, n, target, alpha):
    hp = project_to_set(x, target)
    dx = 2.0 / n
    corr = np.correlate(hp, 1.0 - hp, mode="full") * dx
    mc = np.max(corr)
    a = alpha * (corr - mc)
    a = a - np.max(a)
    return mc + np.log(np.sum(np.exp(a))) / alpha

            h2 = project_to_set(res.x, target)

            if c5 < bc5:

        except Exception:
