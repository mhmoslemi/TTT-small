import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(5000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not np.any(mask):
            h[:] = target / n
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(x, tau):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = corr.max()
    return m + np.log(np.sum(np.exp(tau * (corr - m)))) / tau

def opt_lbfgs(h0, tau, maxiter=400):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    def f(x):
        return smooth_obj(x, tau)
    try:
        r = minimize(f, h0, method='L-BFGS-B', bounds=bnds,
                     options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
        return repair(np.clip(r.x, 0, 1), n)
    except Exception:
        return h0

def sim_anneal(h0, n, n_iter=40000, t0=0.003, t1=0.00005):
    h = h0.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    for it in range(n_iter):
        frac = (it + 1) / n_iter
        temp = t0 * (t1 / t0) ** frac
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        avail = min(1.0 - h[i], h[j])
        if avail < 1e-7:
            continue
        delta = avail * np.random.uniform(0.05, 1.0)
        h[i] += delta
        h[j] -= delta
        c5_new = compute_c5(h)
        if c5_new < c5 or np.random.random() < np.exp(-(c5_new - c5) / (temp + 1e-12)):
            c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            h[i] -= delta
            h[j] += delta
    return best_h, best_c5

def binary_round(h, n):
    """Round to binary maintaining sum constraint via greedy"""
    h_sorted_idx = np.argsort(-h)
    half = n // 2
    b = np.zeros(n)
    for idx in range(half):
        b[h_sorted_idx[idx]] = 1.0
    return b

def binary_hill_climb(h0, n, n_iter=200000):
    """Hill climbing on binary sequences via pair swaps"""
    h = h0.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h < 0.5)[0].tolist()
    for it in range(n_iter):
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        c5_new = compute_c5(h)
        if c5_new < c5 - 1e-15:
            c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
        else:
            h[i] = 1.0
            h[j] = 0.0
            # occasional accept for diversity
            if it % 5000 == 0 and np.random.random() < 0.3:
                ones.remove(i)
                ones.append(j)
                zeros.remove(j)
                zeros.append(i)
            else:
                pass
    return best_h, best_c5
