import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return c.max() * dx

def corr_vec(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(free):
            h[free] += diff / np.sum(free)
            h[free] = np.clip(h[free], 0.0, 1.0)
        else:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        h = np.clip(h + diff / n, 0.0, 1.0)
    return h

def rounded_pattern(h, n):
    """Round to 0-0.5-1 pattern maintaining sum=n/2."""
    h = np.clip(h, 0.0, 1.0)
    # Sort by value to assign 1s to largest, 0.5 to middle, 0s to smallest
    order = np.argsort(-h)
    pat = np.zeros(n)
    pat[order[:n//2]] = 1.0
    pat[order[n//2]] = 0.5
    return pat

def smax_obj(h, args):
    beta, n = args
    c = corr_vec(h)
    m = c.max()
    scaled = beta * (c - m)
    return (m + (1.0 / beta) * np.log(np.sum(np.exp(scaled)))) * (2.0 / n)

def smax_obj_grad(h, args):
    beta, n = args
    c = corr_vec(h)
    m = c.max()
    w = np.exp(beta * (c - m))
    w /= w.sum()
    # Numerical gradient (cheap for n=47)
    npts = len(h)
    grad = np.zeros(npts)
    eps = 1e-7
    dx = 2.0 / npts
    for j in range(npts):
        h_plus = h.copy()
        h_plus[j] = np.clip(h[j] + eps, 0.0, 1.0)
        h_minus = h.copy()
        h_minus[j] = np.clip(h[j] - eps, 0.0, 1.0)
        cp = np.correlate(h_plus, 1.0 - h_plus, mode='full')
        cm = np.correlate(h_minus, 1.0 - h_minus, mode='full')
        mp, mm = cp.max(), cm.max()
        wp = np.exp(beta * (cp - mp))
        wm = np.exp(beta * (cm - mm))
        wp /= wp.sum()
        wm /= wm.sum()
        fp = (mp + (1.0/beta)*np.log(np.sum(wp))) * dx
        fm = (mm + (1.0/beta)*np.log(np.sum(wm))) * dx
        if h[j] > 0 and h[j] < 1:
            grad[j] = (fp - fm) / (2.0 * eps)
        elif h[j] >= 1:
            grad[j] = (fp - fm) / eps
        else:
            grad[j] = (fp - fm) / eps
    return grad

def sum_con(h, args):
    target = args[0]
    return np.sum(h) - target

def sum_con_jac(h, args):
    return np.ones(len(h))

def slsqp_step(h, beta, n, bounds, target, maxiter):
    args_obj = (beta, n)
    con = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (target,)}]
    res = minimize(smax_obj, h.copy(), args=args_obj,
                  method='SLSQP', bounds=bounds, constraints=con,
                  options={'maxiter': maxiter, 'ftol': 1e-14})
    return res.x

def pattern_local_search(pat, n, dx, best_c5, rng, max_iters=5000):
    """Local search on 0-0.5-1 patterns via swaps."""
    c5_curr = compute_c5(pat)
    if c5_curr >= best_c5:
        return pat, best_c5
    best_c5 = c5_curr
    best_pat = pat.copy()
    stall = 0
    for it in range(max_iters):
        moved = False
        r = rng.random()
        if r < 0.6:
            # Swap a 1 with a 0
            ones = np.where(pat == 1.0)[0]
            zeros = np.where(pat == 0.0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                continue
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            trial = best_pat.copy()
            trial[i], trial[j] = 0.0, 1.0
            c5_t = compute_c5(trial)
            if c5_t < best_c5 - 1e-14:
                best_pat = trial
                best_c5 = c5_t
                stall = 0
                moved = True
        else:
            # Move the 0.5
            half_idx = np.where(pat == 0.5)[0]
            if len(half_idx) == 0:
                continue
            ho = half_idx[0]
            j = rng.randint(n)
            if j == ho:
                continue
            trial = best_pat.copy()
            old_j = trial[j]
            trial[ho] = old_j
            trial[j] = 0.5
            c5_t = compute_c5(trial)
            if c5_t < best_c5 - 1e-14:
                best_pat = trial
                best_c5 = c5_t
                stall = 0
                moved = True
        if not moved:
            stall += 1
            if stall > 200:
                break
    return best_pat, best_c5
