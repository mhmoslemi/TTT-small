            best_c5 = c5_new
            best_h = h_new.copy()

        if elapsed() > budget_s * 0.5:
            break
        for _ in range(3):
            h_rand = best_h + rng.normal(0, 0.02, n)
            h_rand = project_feasible(h_rand, n)
            res = minimize(soft_max_obj, h_rand, args=(temp,),
                         method='SLSQP', bounds=bnds, constraints=cons,
                         options={'maxiter': 1500, 'ftol': 1e-13})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                break

    # 2b: Top-k equalization gradient steps
    if elapsed() < budget_s * 0.65:
        alpha_kw = 0.005
        for it in range(2000):
            if elapsed() > budget_s * 0.65:
                break
            grad = topk_equalize(best_h, n, num_top=3)
            gn = np.linalg.norm(grad)
            if gn < 1e-14:
                break
            step = alpha_kw / (gn + 1e-14)
            h_trial = best_h - step * grad
            h_trial = project_feasible(h_trial, n)
            c5_t = compute_c5(h_trial)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_trial.copy()
                alpha_kw = min(alpha_kw * 1.02, 0.02)
            else:
                alpha_kw *= 0.95
                if alpha_kw < 0.001:
                    alpha_kw = 0.005

    # 2c: Coordinate descent - multiple passes with fine steps
    if elapsed() < budget_s * 0.82:
        step_schedule = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
        for ss in step_schedule:
            if elapsed() > budget_s * 0.82:
                break
            for pass_num in range(20):
                if elapsed() > budget_s * 0.82:
                    break
                improved = False
                order = rng.permutation(n)
                for i in order:
                    for d in [ss, -ss, 2*ss, -2*ss]:
                        hv = best_h[i] + d
                        if hv < -0.01 or hv > 1.01:
                            continue
                        h_trial = best_h.copy()
                        h_trial[i] = np.clip(hv, 0.0, 1.0)
                        h_trial = project_feasible(h_trial, n)
                        c5_t = compute_c5(h_trial)
                        if c5_t < best_c5 - 1e-15:
                            best_c5 = c5_t
                            best_h = h_trial.copy()
                            improved = True
                            break
                if not improved:
                    break

    # 2d: Pairwise exchange
    if elapsed() < budget_s * 0.90:
        for it in range(5000):
            if elapsed() > budget_s * 0.90:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            h_trial = best_h.copy()
            h_trial[i], h_trial[j] = h_trial[j], h_trial[i]
            h_trial = project_feasible(h_trial, n)
            c5_t = compute_c5(h_trial)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_trial.copy()

    # Phase 3: Final ultra-fine polishing
    if elapsed() < budget_s * 0.97:
        for temp in [0.00003, 0.00001, 0.000005]:
            if elapsed() > budget_s * 0.97:
                break
            res = minimize(soft_max_obj, best_h.copy(), args=(temp,),
                         method='SLSQP', bounds=bnds, constraints=cons,
                         options={'maxiter': 5000, 'ftol': 1e-15})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

        # Ultra-fine coordinate descent
        if elapsed() < budget_s * 0.97:
            for _ in range(5):
                improved = False
                order = rng.permutation(n)
                for i in order:
                    for d in [0.00005, -0.00005]:
                        hv = best_h[i] + d
                        if hv < 0 or hv > 1:
                            continue
                        h_trial = best_h.copy()
                        h_trial[i] = hv
                        h_trial = project_feasible(h_trial, n)
                        c5_t = compute_c5(h_trial)
                        if c5_t < best_c5 - 1e-16:
                            best_c5 = c5_t
                            best_h = h_trial.copy()
                            improved = True
                            break
                if not improved:
                    break

    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
