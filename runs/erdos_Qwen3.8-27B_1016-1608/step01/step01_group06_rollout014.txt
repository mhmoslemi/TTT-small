import numpy as np
import time
from scipy.optimize import minimize

def compute_score(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def soft_max_obj(h, tau=300.0):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    c = np.correlate(h, omh, mode='full') * dx
    cm = np.max(c)
    shifted = np.clip(tau * (c - cm), -50.0, 50.0)
    es = np.exp(shifted)
    se = np.sum(es)
    score = cm + np.log(se) / tau
    p = es / se
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-9:
            continue
        lag = k - (n - 1)
        wk = p[k] * dx
        a = max(0, lag)
        b = n - 1
        if a <= b:
            grad[a:b+1] += wk * omh[a-lag:b-lag+1]
        a2 = max(0, -lag)
        b2 = min(n - 1, n - 1 - lag)
        if a2 <= b2:
            grad[a2:b2+1] -= wk * h[a2+lag:b2+lag+1]
    return score, grad

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(2000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    return h

def coord_descent(h, n, rounds=10, deadline=np.inf):
    h = h.copy()
    cur_s = compute_score(h)
    for _ in range(rounds):
        if time.time() > deadline:
            break
        imp = False
        for i in range(n):
            if time.time() > deadline:
                return h
            old = h.copy()
            for dv in (0.05, -0.05, 0.1, -0.1, 0.02, -0.02, 0.08, -0.08):
                nv = old[i] + dv
                if nv < -1e-6 or nv > 1.0 + 1e-6:
                    continue
                h_try = old.copy()
                h_try[i] = nv
                h_try = project_feasible(h_try, n)
                s_try = compute_score(h_try)
                if s_try < cur_s - 1e-12:
                    cur_s = s_try
                    h = h_try
                    imp = True
        if not imp:
            break
    return h

def better_of_pair(h):
    s1 = compute_score(h)
    h_r = 1.0 - h
    s2 = compute_score(h_r)
    if s2 < s1:
        return h_r, s2
    return h, s1
