    dx = 2.0 / n
    shifts = _get_shifts(n)  # length 2n-1
    c, _ = np.correlate(h, 1.0 - h, mode='full') * dx, None

            if 0 <= j1 < n:
                g += (1.0 - h[j1]) * dx
            if 0 <= j2 < n:
                g -= h[j2] * dx
            grad[i] += weights[m] * g
    return grad, cm + (max_s + np.log(np.sum(weights))) / tau

import numpy as np
import time
from scipy.optimize import minimize

def compute_score(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(c), c

def soft_max_obj(h, tau):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cm = np.max(c)
    shifted = tau * (c - cm)
    ms = np.max(shifted)
    return cm + (ms + np.log(np.sum(np.exp(shifted - ms)))) / tau

def soft_max_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cm = np.max(c)
    shifted = tau * (c - cm)
    ms = np.max(shifted)
    w = np.exp(shifted - ms)
    w /= np.sum(w)
    grad = np.zeros(n)
    oh = 1.0 - h
    shifts = np.arange(2 * n - 1) - (n - 1)
    for m in range(2 * n - 1):
        s = shifts[m]
        wm = w[m] * dx
        if wm < 1e-15:
            continue
        if s >= 0:
            grad[:n - s] += wm * oh[s:]
            if s > 0:
                grad[s:] -= wm * h[:n - s]
            else:
                grad -= wm * h
        else:
            ns = -s
            grad[ns:] += wm * oh[:n - ns]
            if s < 0:
                grad[:n - ns] -= wm * h[ns:]
    return grad

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(2000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h = np.clip(h + step * mask, 0.0, 1.0)
    return h

def coord_descent(h, n, max_passes=100, grid_points=301):
    dx = 2.0 / n
    shifts = np.arange(2 * n - 1) - (n - 1)
    for _ in range(max_passes):
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        cur_max = np.max(c)
        improved = False
        for i in range(n):
            slopes = np.zeros(2 * n - 1)
            for m in range(2 * n - 1):
                s = shifts[m]
                t = 0.0
                j1 = i + s
                j2 = i - s
                if 0 <= j1 < n:
                    t += 1.0 - h[j1]
                if 0 <= j2 < n:
                    t -= h[j2]
                slopes[m] = t * dx
            c0 = c - h[i] * slopes
            ts = np.linspace(0.0, 1.0, grid_points)
            vals = np.max(c0[:, None] + ts[None, :] * slopes[:, None], axis=0)
            t_best = ts[np.argmin(vals)]
            if abs(t_best - h[i]) > 1e-6:
                h[i] = t_best
                improved = True
        if not improved:
            break
    return h

def subgradient_step(h, tau, step_size, n):
    grad = soft_max_grad(h, tau)
    h_new = h - step_size * grad
    return project_feasible(h_new, n)

def make_fourier_init(n, rng):
    h = np.zeros(n)
    freqs = rng.choice(np.arange(2, n // 2 + 1), size=min(4, n // 4), replace=False)
    phases = rng.uniform(0, 2 * np.pi, size=len(freqs))
    amps = rng.uniform(0.15, 0.35, size=len(freqs))
    for k, (f, p, a) in enumerate(zip(freqs, phases, amps)):
        h += a * np.sin(2 * np.pi * f * np.arange(n) / n + p)
    h = 0.5 + h
    return project_feasible(h, n)

def make_block_init(n, rng, nblocks):
    h = np.zeros(n)
    sizes = np.full(nblocks, n // nblocks)
    sizes[:n % nblocks] += 1
    order = rng.permutation(nblocks)
    idx = 0
    for bi in range(nblocks):
        sz = sizes[bi]
        h[idx:idx + sz] = rng.uniform(0.3, 0.7)
        idx += sz
    return project_feasible(h, n)
