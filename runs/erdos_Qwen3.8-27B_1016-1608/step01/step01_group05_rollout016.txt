import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def project(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def smooth_max_grad(h, n, beta):
    corr = compute_corr(h)
    dx = 2.0 / n
    m = np.max(corr)
    exps = np.exp((corr - m) * beta)
    sw = np.sum(exps)
    w = exps / sw
    
    grad = np.zeros(n)
    for L in range(n):
        if L == 0:
            wL = w[n - 1]
        else:
            wL = w[n - 1 + L] + w[n - 1 - L]
        if wL < 1e-12:
            continue
        grad[L:] += wL * (1.0 - h[:n - L])
        grad[:n - L] -= wL * h[L:]
    
    obj = (m + np.log(sw) / (len(corr) * beta)) * dx
    return obj, grad * dx

def pgd_smooth(h0, n, n_iter, beta0, beta1, lr0, lr1):
    h = h0.copy()
    target = n / 2.0
    h = project(h, target)
    best = compute_c5(h, n)
    best_h = h.copy()
    
    for it in range(n_iter):
        t = it / max(1, n_iter - 1)
        beta = beta0 + (beta1 - beta0) * t
        lr = lr0 + (lr1 - lr0) * t
        
        # Backtracking line search
        for _ in range(20):
            obj, grad = smooth_max_grad(h, n, beta)
            h_new = h - lr * grad
            h_new = project(h_new, target)
            c5_new = compute_c5(h_new, n)
            if c5_new < best or np.allclose(h_new, h):

            lr *= 0.5
            if lr < 1e-10:

        h = project(h_new, target)
        c5_new = compute_c5(h, n)
        if c5_new < best:

            best_h = h.copy()
    
    return best_h, best

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h, n):
    return float(np.max(compute_corr(h)) * (2.0 / n))

def project(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def smooth_grad(h, n, beta):
    corr = compute_corr(h)
    dx = 2.0 / n
    scaled = corr * dx
    m = np.max(scaled)
    e = np.exp(scaled - m)
    w = e / np.sum(e)
    grad = np.zeros(n)
    for L in range(n):
        wL = w[n - 1] if L == 0 else w[n - 1 + L] + w[n - 1 - L]
        if wL < 1e-12:
            continue
        grad[L:] += wL * (1.0 - h[:n - L])
        grad[:n - L] -= wL * h[L:]
    obj = m + np.log(np.sum(e) / len(scaled))
    return obj, grad * dx
