def project(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h = np.clip(h + diff / cnt * mask, 0.0, 1.0)
    return h

import numpy as np
import time

def compute_score(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(c)

def project(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h = np.clip(h + diff / cnt * mask, 0.0, 1.0)
    return h

def compute_grad(h, n, c_vals, tau):
    dx = 2.0 / n
    N = 2 * n - 1
    if tau <= 1.0:
        cm = np.max(c_vals)
        mask = c_vals >= cm - 1e-9
        w = np.zeros(N)
        w[mask] = 1.0 / np.sum(mask)
    else:
        cm = np.max(c_vals)
        shifted = tau * (c_vals - cm)
        shifted -= np.max(shifted)
        ew = np.exp(shifted)
        w = ew / np.sum(ew)
    grad = np.zeros(n)
    for j in range(n):
        s1 = 0.0
        s2 = 0.0
        s3 = 0.0
        for k in range(N):
            wk = w[k]
            if wk < 1e-14:
                continue
            shift = k - (n - 1)
            jm = j - shift
            jp = j + shift
            if 0 <= jm < n:
                s1 += wk
                s2 += wk * h[jm]
            if 0 <= jp < n:
                s3 += wk * h[jp]
        grad[j] = dx * (s1 - s2 - s3)
    return grad

def grad_descent(h, n, tau0=80.0, tau_min=1.0, tau_decay=800, lr0=0.03, lr_decay=2000, max_iter=4000):
    h = h.copy()
    best = compute_score(h)
    for it in range(max_iter):
        dx = 2.0 / n
        c_vals = np.correlate(h, 1.0 - h, mode='full') * dx
        score = np.max(c_vals)
        if score < best:
            best = score
        tau = max(tau_min, tau0 * np.exp(-it / tau_decay))
        g = compute_grad(h, n, c_vals, tau)
        lr = lr0 * max(0.1, 1.0 / (1.0 + it / lr_decay))
        for _ in range(20):
            h_new = project(h - lr * g, n)
            s_new = compute_score(h_new)
            if s_new < score - 1e-15:
                h = h_new
                break
            lr *= 0.5
        if lr < 1e-8:
            break
    return h, best

def finalize(h, n):
    h = project(h, n)
    return h, compute_score(h)
