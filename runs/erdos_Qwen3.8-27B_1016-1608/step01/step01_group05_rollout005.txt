    """Create structured binary patterns."""
    pattern_type = rng.randint(5)
    h = np.zeros(n)
    half = n // 2
    
    if pattern_type == 0:
        # Alternating blocks of varying size
        block = rng.randint(1, 5)
        for i in range(n):

        h[flip] = 1.0

        h[flip] = 0.0
    return h

    cur = best

    zeros = list(np.where(h == 0)[0])

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def binarize(htarget, n):
    idx = np.argsort(htarget)[::-1]
    out = np.zeros(n)
    out[idx[:n // 2]] = 1.0
    return out

def make_binary(n, rng, method=0):
    half = n // 2
    h = np.zeros(n)
    if method == 0:
        h[rng.choice(n, half, replace=False)] = 1.0
    elif method == 1:
        blk = rng.randint(2, 6)
        for i in range(n):
            h[i] = 1.0 if (i // blk) % 2 == 0 else 0.0
        _fix_count(h, n)
    elif method == 2:
        x = np.linspace(0, 2 * np.pi, n, endpoint=False)
        val = np.sin(x) + 0.4 * np.sin(2 * x + 0.7) + 0.2 * np.sin(3 * x)
        h = (val >= np.median(val)).astype(float)
        _fix_count(h, n)
    elif method == 3:
        val = rng.randn(n)
        h = (val >= np.median(val)).astype(float)
    elif method == 4:
        try:
            x_old = np.linspace(0, 2, len(initial_h_values))

    else:
        phase = rng.uniform(0, 2 * np.pi)
        h = ((np.arange(n) + phase) % 7 < 3.5).astype(float)

def _fix_count(h, n):
    half = n // 2
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) > half:
        flip = np.random.choice(ones, len(ones) - half, replace=False)
        h[flip] = 0.0
    elif len(ones) < half:
        flip = np.random.choice(zeros, half - len(ones), replace=False)
        h[flip] = 1.0

def sa_swap(h, n, rng, n_iter=200000, t0=0.03, t1=0.0005):

    cur = compute_c5(h, n)
    best = cur
    best_h = h.copy()
    ones_list = np.where(h == 1)[0].tolist()
    zeros_list = np.where(h == 0)[0].tolist()
    n1, n0 = len(ones_list), len(zeros_list)
    if n1 == 0 or n0 == 0:
        return best_h, best
    ratio = t1 / t0
    for it in range(n_iter):
        t = t0 * ratio ** (it / max(n_iter, 1))
        ii = rng.integers(n1)
        jj = rng.integers(n0)
        i = ones_list[ii]
        j = zeros_list[jj]
        h[i] = 0.0
        h[j] = 1.0
        nc = compute_c5(h, n)

            cur = nc
            ones_list[ii] = j
            zeros_list[jj] = i
            if nc < best:

        else:

    best_h = h.copy()

def upsample_binary(h_old, n_new, rng):
    n_old = len(h_old)
    h = np.zeros(n_new)
    ratio = n_new / n_old
    for i in range(n_new):

        if old_idx < n_old:

    ones = np.where(h == 1)[0]

    if need > 0:
        zeros = np.where(h == 0)[0]
        flip = rng.choice(zeros, need, replace=False)
        h[flip] = 1.0
    elif need < 0:
        flip = rng.choice(ones, -need, replace=False)
        h[flip] = 0.0
    return h

    if _tl() > 30 and best_n <= 200:
        h_gp, c5_gp = greedy_polish(best_h, best_n, rng, max_passes=100)
        _upd(h_gp, best_n)

    # Ensure validity
    best_h = np.clip(best_h, 0, 1)
    s = np.sum(best_h)
    target = best_n / 2.0
    if abs(s - target) > 1e-6:
        # Re-project: nudge values
        diff = target - s
        step = diff / best_n
        best_h = np.clip(best_h + step, 0, 1)
    best_c5 = compute_c5(best_h, best_n)
    return best_h, best_c5, best_n
