    return float(np.max(c) * dx)

    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return c * dx

def _fix_sum(h):
    n = len(h)
    target = 0.5 * n
    for _ in range(300):
        d = target - np.sum(h)
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h = h + (d / cnt) * mask
        h = np.clip(h, 0.0, 1.0)
    return h

def _soft_max_val(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    v = c * dx
    m = np.max(v)
    clipped = np.clip(v - m, -50.0, 50.0)
    e = np.exp(clipped)
    return m + np.log(np.sum(e)) / alpha

def _soft_max_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    v = c * dx
    m = np.max(v)
    ks = int(np.argmax(v))
    exp_v = np.exp(np.clip(v - m, -50.0, 50.0))
    p = exp_v / np.sum(exp_v)
    w = p / alpha
    w[ks] += 1.0
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        lag = k - n + 1
        if lag >= 0:
            # (1-h[i-lag]) valid for i in [lag, n-1]
            idx1 = np.arange(lag, n)
            grad[idx1] += dx * wk * (1.0 - h[idx1 - lag])
            # -h[i+lag] valid for i in [0, n-1-lag]
            idx2 = np.arange(0, n - lag)
            grad[idx2] -= dx * wk * h[idx2 + lag]
        else:
            a = -lag
            # (1-h[i+a]) valid for i in [0, n-1-a]
            idx1 = np.arange(0, n - a)
            grad[idx1] += dx * wk * (1.0 - h[idx1 + a])
            # -h[i-a] valid for i in [a, n-1]
            idx2 = np.arange(a, n)
            grad[idx2] -= dx * wk * h[idx2 - a]
    return grad

def _eq_fun(h):
    return np.sum(h) - 0.5 * len(h)

def _eq_jac(h):
    return np.ones(len(h))

def _optimize_alpha(h0, n, alpha, maxiter, t0, budget):
    if budget - (time.time() - t0) < 15:
        return None
    def obj(x):
        return _soft_max_val(x, alpha)
    def jac(x):
        return _soft_max_grad(x, alpha)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    try:
        res = minimize(obj, h0, jac=jac, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12, 'iprint': 0})
        if res.x is not None:
            return _fix_sum(np.clip(res.x, 0.0, 1.0))
    except Exception:
        pass
    return None

import numpy as np
from scipy.optimize import fmin_l_bfgs_b
import time

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _fix(h):
    n = len(h)
    t = 0.5 * n
    for _ in range(800):
        d = t - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h < 1.0) if d > 0 else (h > 0.0)
        k = int(m.sum())
        if k == 0:
            break
        h = h + (d / k) * m
        np.clip(h, 0, 1, out=h)
    return h

def _sm(h, alpha, lam):
    dx = 2.0 / len(h)
    v = np.correlate(h, 1.0 - h, mode='full') * dx
    m = v.max()
    e = np.exp(np.clip(v - m, -50, 50))
    return m + np.log(e.sum()) / alpha + lam * (h.sum() - 0.5 * len(h))**2

def _do_lbf(h0, alpha, lam, iters):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    try:
        x = fmin_l_bfgs_b(_sm, h0.copy(), args=(alpha, lam), bounds=bnds,
                          maxiter=iters, iprint=0, pgtol=1e-9)
        return _fix(np.clip(x, 0, 1))
    except Exception:
        return None

def _interp(h, n_new):
    n_old = len(h)
    xo = np.linspace(0, 2, n_old, endpoint=False)
    xn = np.linspace(0, 2, n_new, endpoint=False)
    return _fix(np.clip(np.interp(xn, xo, h), 0, 1))

def _perturb(h, rng, scale):
    return _fix(np.clip(h + rng.randn(len(h)) * scale, 0, 1))

def _binary_sa(h, t0, budget, n_it=25000, seed=42):
    n = len(h)
    hb = (h > 0.5).astype(float)
    s = int(hb.sum())
    tgt = int(0.5 * n)
    rng = np.random.RandomState(seed)
    if s > tgt:
        idx = np.where(hb == 1)[0].copy()
        rng.shuffle(idx)
        for i in idx[:s - tgt]:
            hb[i] = 0
    elif s < tgt:
        idx = np.where(hb == 0)[0].copy()
        rng.shuffle(idx)
        for i in idx[:tgt - s]:
            hb[i] = 1
    best = hb.copy()
    bv = _c5(best)
    v = bv
    ones = np.where(hb == 1)[0].tolist()
    zeros = np.where(hb == 0)[0].tolist()
    T = 0.008
    Tmin = 3e-5
    decay = (Tmin / T) ** (1.0 / max(n_it, 1))
    rng2 = np.random.RandomState(seed + 7)
    for it in range(n_it):
        if budget - (time.time() - t0) < 2:
            break
        T *= decay
        i = ones[rng2.randint(len(ones))]
        j = zeros[rng2.randint(len(zeros))]
        hb[i] = 0.0
        hb[j] = 1.0
        nv = _c5(hb)
        if nv < v or rng2.rand() < np.exp(-(nv - v) / max(T, 1e-12)):
            v = nv
            if nv < bv:
                bv = nv
                best = hb.copy()
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
        else:
            hb[i] = 1.0
            hb[j] = 0.0
    return best, bv

def _gen_starts(h_base, n, rng, count=4):
    starts = [h_base.copy()]
    starts.append(_perturb(h_base, rng, 0.08))
    starts.append(_perturb(h_base, rng, 0.15))
    fb = 3.0 + rng.rand() * 4.0
    xs = np.arange(n) * (2.0 / n)
    hsin = 0.5 + 0.35 * np.sin(2.0 * np.pi * fb * xs)
    hsin = _fix(np.clip(hsin, 0, 1))
    starts.append(hsin)
    if count > 3:
        starts.append(_perturb(h_base, rng, 0.25))
    return starts

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        bh = np.array(initial_h_values, dtype=float).copy()
    else:

    bh = _fix(bh)
    bc = _c5(bh)
    n = len(bh)

    # Phase 1: L-BFGS-B at original resolution, multiple alphas and starts
    starts = _gen_starts(bh, n, rng, 4)
    for alpha in [80, 250, 800, 2500]:
        if budget_s - (time.time() - t0) < 40:
            break
        for s in starts:
            if budget_s - (time.time() - t0) < 15:
                break
            ht = _do_lbf(s, alpha, 100.0, 250)
            if ht is not None:
                ct = _c5(ht)
                if ct < bc:
                    bh, bc = ht, ct
        # Re-generate starts from current best
        starts = _gen_starts(bh, n, rng, 3)

    # Phase 2: Binary SA
    if budget_s - (time.time() - t0) > 50:
        hb, cb = _binary_sa(bh, t0, budget_s, n_it=30000, seed=seed)
        if cb < bc:
            bh, bc = hb, cb
        # Refine binary with L-BFGS-B
        if budget_s - (time.time() - t0) > 25:

                    ct = _c5(ht)

    # Phase 3: Resolution sweep
    for nn in [250, 320, 500, 640]:
        if budget_s - (time.time() - t0) < 45:

        ci = _c5(hi)

            bh, bc = hi, ci

                    hi = ht.copy()

    # Phase 4: Final polish at best resolution
    nb = len(bh)
    for alpha in [3000, 8000]:

                bh, bc = ht, ct

    bh = _fix(bh)
    bc = _c5(bh)
    return bh, bc, len(bh)
