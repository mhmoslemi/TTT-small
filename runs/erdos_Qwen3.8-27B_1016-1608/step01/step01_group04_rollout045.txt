import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

_opt = {}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        d = target - np.sum(h)
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not np.any(mask):
            break
        h[mask] += d / np.sum(mask)
    return np.clip(h, 0.0, 1.0)

def smax(x, n, eps):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(c)
    if eps < 1e-12:
        return float(m)
    s = np.clip((c - m) / eps, -50.0, 50.0)
    return float(m + eps * np.log(np.sum(np.exp(s))))

def eqf(x):
    return float(np.sum(x) - _opt['n'] / 2.0)

def eqj(x):
    return np.ones(_opt['n'])

def obf(x):
    return smax(x, _opt['n'], _opt['eps'])

def do_slsqp(h, n, eps, mi, ft):
    _opt['n'] = n
    _opt['eps'] = eps
    bds = [(0.0, 1.0)] * n
    cs = [{'type': 'eq', 'fun': eqf, 'jac': eqj}]
    r = minimize(obf, h, method='SLSQP', bounds=bds, constraints=cs,
                 options={'maxiter': mi, 'ftol': ft})
    return repair(r.x, n)

def upsample(h_old, n_new):
    n_old = len(h_old)
    x_old = np.arange(n_old, dtype=float)
    x_new = np.arange(n_new, dtype=float) * (n_old / n_new)
    h_new = np.interp(x_new, x_old, np.concatenate([h_old, [h_old[0]]]))
    return h_new

def de_objective_factory(n):
    def _de_obj(x):
        h = repair(x, n)
        return compute_c5(h) + 5.0 * (np.sum(h) - n / 2.0) ** 2
    return _de_obj

    np.random.seed(seed)

    best_h = None
    best_c = 1.0

    resolutions = [80, 160, 240]

    for n in resolutions:
        if time.time() - t0 > budget_s - 120:
            break

        # Prepare initial point
        if h_parent is not None:
            if n == len(h_parent):
                h0 = h_parent.copy()
            else:
                h0 = upsample(h_parent, n)
        else:
            h0 = 0.5 * np.ones(n)
        h0 = repair(h0, n)
        c0 = compute_c5(h0)
        if c0 < best_c:
            best_c, best_h = c0, h0.copy()

        # Phase 1: DE global search
        if time.time() - t0 < budget_s - 300:
            de_time_budget = min(120.0, budget_s - 300 - (time.time() - t0))
            popsize = max(20, min(50, n // 4))
            maxit = max(50, int(de_time_budget / (popsize * 2 * 0.0001)))
            maxit = min(maxit, 500)
            bounds = [(0.0, 1.0)] * n
            de_obj = de_objective_factory(n)
            try:
                result = differential_evolution(

                    maxiter=maxit, popsize=popsize,
                    tol=0.01, mutation=(0.4, 1.3),
                    recombination=0.85, polish=False,
                    init='latinhypercube'

                h_de = repair(result.x, n)
                c_de = compute_c5(h_de)

                elif r == 5:
                    hp = base.copy()
                    idx = np.random.choice(n, size=3, replace=False)
                    hp[idx] = 1.0 - hp[idx]
                else:
                    hp = base.copy()
                    idx = np.random.choice(n, size=max(1, n // 4), replace=False)
                    hp[idx] = np.random.uniform(0, 1, size=len(idx))
                hp = repair(hp, n)
                hn = do_slsqp(hp, n, 0.003, 150, 1e-12)
                c5 = compute_c5(hn)
                if c5 < best_c:
                    best_c, best_h = c5, hn

    # Phase 4: Simulated annealing on best solution
    if best_h is not None and time.time() - t0 < budget_s - 40:
        n = len(best_h)
        curr = best_h.copy()
        curr_c = best_c
        T = 0.002
        for it in range(20000):

                curr_c = new_c

    if best_h is not None and time.time() - t0 < budget_s - 15:
        n = len(best_h)
        for eps in [0.001, 0.0005, 0.0002, 0.0001]:
            if time.time() - t0 > budget_s - 10:

            c5 = compute_c5(hn)
            if c5 < best_c:

    # Final tiny-step search
    if best_h is not None and time.time() - t0 < budget_s - 8:
        n = len(best_h)
        for _ in range(5000):
            if time.time() - t0 > budget_s - 5:

            hp = repair(hp, n)
            c5 = compute_c5(hp)
            if c5 < best_c:
                best_c, best_h = c5, hp

    fh = repair(best_h, len(best_h))
    fc = compute_c5(fh)
    return (fh.tolist(), float(fc), len(fh))
