            old = h[j]
            cands = [old]
            for d in [0.05, 0.1, 0.2, 0.5]:
                cands.append(max(0.0, min(1.0, old + d)))
                cands.append(max(0.0, min(1.0, old - d)))

            bv = best
            bc = old
            for c in cands:
                if abs(c - old) < 1e-9:
                    continue

                h = project(h, target)
                v = compute_c5(h, n)
                if v < bv:

                    bc = c

                h[j] = bc
                h = project(h, target)
                best = bv
                improved = True
            else:
                h[j] = old
                h = project(h, target)
        if not improved:
            break
    return h, best

        w = np.exp(-np.arange(top_k) * 0.5)
        w /= w.sum()
        c5 = float(corr[sidx[0]] * 2.0 / n)
        ii = np.arange(n)
        grad = np.zeros(n)
        for wi, si in zip(w, sidx):
            shift = si - (n - 1)
            j1 = ii + shift
            v1 = (j1 >= 0) & (j1 < n)
            grad[v1] += wi * (1.0 - h[j1[v1]])
            j2 = ii - shift
            v2 = (j2 >= 0) & (j2 < n)
            grad[v2] -= wi * h[j2[v2]]
    return c5, grad

def project(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        hp = np.clip(h - mid, 0.0, 1.0)
        if float(np.sum(hp)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def upsample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    return np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, n_old), h)

def subgrad_optimize(h0, n, time_limit, lr0=0.5, top_k=1, rng=None):
    target = n / 2.0
    h = project(h0.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    t0 = time.time()
    lr = lr0
    patience = 0
    best_patience = 0
    
    while time.time() - t0 < time_limit:
        c5, grad = subgrad_step(h, n, top_k)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
        
        h = h - lr * grad
        h = project(h, target)
        lr *= 0.9997
        
        if patience > 300:
            if rng is not None:
                noise = rng.randn(n) * 0.03
                h = h + noise
                h = project(h, target)
            lr = max(lr0 * 0.3, 0.05)
            patience = 0
    
    best_h = project(best_h, target)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5

def coord_descent(h, n, rng, n_sweeps, time_limit):
    target = n / 2.0
    h = h.copy()
    best = compute_c5(h, n)
    t0 = time.time()
    for _ in range(n_sweeps):
        if time.time() - t0 > time_limit:
            break
        order = rng.permutation(n)
        improved = False
        for j in order:
            old = h[j]
            for d in [0.1, 0.05, 0.2, 0.4]:
                for sign in [1, -1]:
                    c = max(0.0, min(1.0, old + sign * d))
                    if abs(c - old) < 1e-9:
                        continue
                    h[j] = c
                    h = project(h, target)
                    v = compute_c5(h, n)
                    if v < best - 1e-12:
                        best = v
                        old = c
                        improved = True
                    else:
                        h[j] = old
                        h = project(h, target)
        if not improved:
            break
    return h, best

def gen_candidates(n, rng, base_h=None):
    cands = []
    # Binary random
    for _ in range(4):
        h = np.zeros(n)
        h[rng.choice(n, size=n//2, replace=False)] = 1.0
        h += rng.normal(0, 0.1, n)
        cands.append(project(h, n/2.0))
    # Smooth random
    for _ in range(3):
        x = np.linspace(0, 2, n)
        h = np.zeros(n)
        for _ in range(rng.randint(2, 5)):
            f = rng.randint(1, 7)
            p = rng.uniform(0, 2*np.pi)
            a = rng.uniform(0.2, 0.5)
            h += a * np.sin(f*np.pi*x + p)
        h = 0.5 + 0.4*h
        cands.append(project(h, n/2.0))
    # From base (upsampled + noise)
    if base_h is not None and len(base_h) != n:
        for temp in [0.03, 0.08, 0.15]:
            h = upsample(base_h, n) + rng.normal(0, temp, n)
            cands.append(project(h, n/2.0))
    elif base_h is not None:
        for temp in [0.03, 0.08]:
            h = base_h + rng.normal(0, temp, n)
            cands.append(project(h, n/2.0))
    return cands

def run(seed=42, budget_s=1100, **kwargs):
    global _budget, _start_t, _best_h, _best_c5, _best_n
    _budget = float(budget_s)
    _start_t = time.time()
    _best_h = None
    _best_c5 = np.inf
    _best_n = 0
    rng = np.random.RandomState(seed)
    
    def tleft():
        return max(0.0, _budget - (time.time() - _start_t))
    
    def update(h, n):
        global _best_h, _best_c5, _best_n
        h = project(np.asarray(h, dtype=float), n/2.0)
        c = compute_c5(h, n)
        if c < _best_c5:
            _best_h = h.copy()
            _best_c5 = c
            _best_n = n
    
    # Phase 1: n=90
    n1 = 90
    base = None
    try:
        base = np.array(initial_h_values, dtype=float).copy()
        if len(base) != n1:
            base = upsample(base, n1)
        base = project(base, n1/2.0)
        update(base, n1)
    except Exception:
        pass
    
    cands = gen_candidates(n1, rng, base)
    # Optimize each with time allocation
    total_cands = len(cands) + 1  # +1 for base
    for ci, h_c in enumerate(cands):
        if tleft() < 20:
            break
        alloc = min(tleft() / (total_cands - ci) * 0.7, 80)
