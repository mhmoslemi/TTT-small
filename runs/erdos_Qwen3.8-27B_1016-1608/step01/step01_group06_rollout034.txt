Actually, a simpler approach: since we just need the minimum of a convex piecewise-linear function on [u_lo, u_hi], I can use a simple bisection on the subgradient. The subgradient of max_j f_j(u) at u is the set {f_j'(u) : j in argmax f_j(u)}. The minimum is where 0 is in the subgradient.

Even simpler: just do a golden-section search on the 1D convex function. It takes ~50 evaluations of the max, each O(n). So O(50n) per coordinate, O(50n²) per sweep, which for n=50 is 125000. Very fast.

```python
import numpy as np
import time

def compute_score(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(c)

def soft_max_score(h, tau):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cm = c.max()
    return cm + np.log(np.sum(np.exp(tau * (c - cm)))) / tau

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h = np.clip(h + diff / cnt * mask, 0.0, 1.0)
    return h

def coord_sweep(h, n, passes=5):
    """Multiple passes of coordinate descent."""
    for p in range(passes):
        improved = False
        for i in range(n):
            old = h[i]
            new_t = optimize_coord(h, i, n)
            if abs(new_t - old) > 1e-10:
                h[i] = new_t
                improved = True
        if not improved:
            break
    return h

def optimize_coord(h, i, n):
    """1D optimization of h[i]."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    d = np.zeros(2*n-1)
    sj_all = np.arange(2*n-1) - (n-1)
    # Vectorized computation of d
    # For term 1: 0 <= i-sj < n → sj in [i-n+1, i], clipped to [-(n-1), n-1]
    # For term 2: 0 <= i+sj < n → sj in [-i, n-1-i], clipped to [-(n-1), n-1)
    m = 2*n-1
    # Term 1: sj from max(i-n+1, -(n-1)) to min(i, n-1)
    s1_lo = max(i-n+1, -(n-1))
    s1_hi = min(i, n-1)
    if s1_lo <= s1_hi:
        j1_lo = s1_lo + (n-1)
        j1_hi = s1_hi + (n-1)
        sj_valid = sj_all[j1_lo:j1_hi+1]
        d[j1_lo:j1_hi+1] += (1.0 - h[i - sj_valid])
    # Term 2: sj from max(-i, -(n-1)) to min(n-1-i, n-1)
    s2_lo = max(-i, -(n-1))
    s2_hi = min(n-1-i, n-1)
    if s2_lo <= s2_hi:
        j2_lo = s2_lo + (n-1)
        j2_hi = s2_hi + (n-1)
        sj_valid = sj_all[j2_lo:j2_hi+1]
        d[j2_lo:j2_hi+1] -= h[i + sj_valid]
    
    u_lo, u_hi = -h[i], 1.0 - h[i]
    if u_hi - u_lo < 1e-14:
        return h[i]
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_, b_ = u_lo, u_hi
    c_ = b_ - gr * (b_ - a_)
    d_ = a_ + gr * (b_ - a_)
    fc = np.max(c + c_ * d)
    fd = np.max(c + d_ * d)
    for _ in range(30):
        if fc <= fd:
            b_ = d_
            d_ = c_
            fd = fc
            c_ = b_ - gr * (b_ - a_)
            fc = np.max(c + c_ * d)
        else:
            a_ = c_
            c_ = d_
            fc = fd
            d_ = a_ + gr * (b_ - a_)
            fd = np.max(c + d_ * d)
    
    best_u = (a_ + b_) / 2.0
    # Check boundaries too
    for u_check in [u_lo, u_hi, (a_+b_)/2]:
        _ = u_check  # already covered by golden section near optimum
    return h[i] + best_u
```
