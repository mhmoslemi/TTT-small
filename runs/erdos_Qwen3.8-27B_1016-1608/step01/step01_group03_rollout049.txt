                          options={'maxiter': 1000, 'ftol': 1e-16})
            hf = repair(res.x, f_target)
            cf = compute_c5(hf, f_dx)
            if cf < best_c5:
                best_c5, best_h = cf, hf
        except Exception:
            break

    best_h = repair(best_h, f_target)
    best_c5 = compute_c5(best_h, f_dx)
    return (best_h, best_c5, best_n)
