import time

def compute_c5_fast(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def make_random_binary(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, size=n//2, replace=False)
    h[idx] = 1.0
    return h

def make_greedy_fill(n, rng):
    """Greedy: place ones to minimize max correlation at each step."""
    h = np.zeros(n)
    ones_needed = n // 2
    # Place first half randomly, then greedily place rest
    h[rng.choice(n, size=n//4, replace=False)] = 1.0
    remaining = ones_needed - n//4
    ones = set(np.where(h==1)[0])
    for _ in range(remaining):
        best_pos = -1
        best_val = np.inf
        for pos in range(n):
            if pos in ones:
                continue
            h[pos] = 1.0
            corr = np.correlate(h, 1.0-h, mode="full")
            val = np.max(corr)
            h[pos] = 0.0
            if val < best_val:
                best_val = val
                best_pos = pos
        h[best_pos] = 1.0
        ones.add(best_pos)
    return h

    best_c5 = compute_c5_fast(h)
    curr_c5 = best_c5
    ones = list(np.where(h==1)[0])
    zeros = list(np.where(h==0)[0])

    T = T_init
    no_improve = 0
    
    for it in range(n_iter):
        T *= alpha
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5_fast(h)
        
        if new_c5 < curr_c5:
            # Accept
            curr_c5 = new_c5
            # Update ones/zeros lists
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            no_improve = 0
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # Maybe accept (annealing)
            delta = new_c5 - curr_c5
            if delta > 0 and np.log(rng.randint(1, 1000000) / 1000000.0) < -delta / max(T, 1e-10):
                # Accept
                curr_c5 = new_c5
                ones.remove(i)
                ones.append(j)
                zeros.remove(j)
                zeros.append(i)
            else:
                # Reject
                h[i] = 1.0
                h[j] = 0.0
            no_improve += 1
        
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def make_random_binary(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h

def make_greedy_fill(n, rng):
    h = np.zeros(n)
    n_ones = n // 2
    h[rng.choice(n, size=n_ones // 3, replace=False)] = 1.0
    rem = n_ones - n_ones // 3
    for _ in range(rem):
        best_p, best_v = -1, np.inf
        for p in range(n):
            if h[p] > 0.5:
                continue
            h[p] = 1.0
            v = float(np.max(np.correlate(h, 1.0 - h, mode="full")))
            h[p] = 0.0
            if v < best_v:
                best_v, best_p = v, p
        h[best_p] = 1.0
    return h

def sa_pair_search(h, n, rng, n_iter, T_init, T_final):
    h = h.copy()
    best_h, best_c5 = h.copy(), compute_c5(h)
    cur_c5 = best_c5
    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())
    alpha = (T_final / T_init) ** (1.0 / max(n_iter, 1))
    T = T_init
    for _ in range(n_iter):
        T *= alpha
        i = list(ones)[rng.randint(len(ones))]
        j = list(zeros)[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        nv = compute_c5(h)
        if nv < cur_c5:
            cur_c5 = nv
            ones.discard(i); ones.add(j)
            zeros.discard(j); zeros.add(i)
            if nv < best_c5:
                best_c5, best_h = nv, h.copy()
        else:
            d = nv - cur_c5
            if d / max(T, 1e-12) < -np.log(max(rng.random(), 1e-15)):
                cur_c5 = nv
                ones.discard(i); ones.add(j)
                zeros.discard(j); zeros.add(i)
            else:
                h[i], h[j] = 1.0, 0.0
    return best_h, best_c5

def fractional_refine(h, n, rng, n_rounds=200):
    h = h.copy()
    target = n / 2.0
    best = compute_c5(h)
    ones = list(np.where(h > 0.3)[0])
    zeros = list(np.where(h < 0.7)[0])
    for rd in range(n_rounds):
        improved = False
        if rd % 20 == 0 and (len(ones) < 5 or len(zeros) < 5):
            ones = list(np.where(h > 0.3)[0])
            zeros = list(np.where(h < 0.7)[0])
        for _ in range(30):
            if not ones or not zeros:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            if i == j:
                continue
            for eps in (0.05, 0.1, 0.2, -0.05, -0.1, -0.2):
                ni, nj = h[i] + eps, h[j] - eps
                if ni < -0.01 or ni > 1.01 or nj < -0.01 or nj > 1.01:
                    continue
                h[i] = np.clip(ni, 0, 1)
                h[j] = np.clip(nj, 0, 1)
                s = np.sum(h)
                if abs(s - target) > 0.5:
                    h[i] -= (s - target) * 0.5
                    h[j] += (s - target) * 0.5
                    h = np.clip(h, 0, 1)
                nv = compute_c5(h)
                if nv < best - 1e-10:
                    best = nv
                    improved = True
                    break
            else:
                continue
            break
        if not improved:
            if rd > 20:
                break
    s = np.sum(h)
    h = np.clip(h - (s - target) / n, 0, 1)
    return h, compute_c5(h)

def project_binary(h, n):
    order = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[order[:n // 2]] = 1.0
    return out

def upsample_linear(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)

def run(seed=42, budget_s=1100, **kwargs):
    _t0 = time.time()
    _budget = float(budget_s)
    rng = np.random.RandomState(seed)
    best_h, best_c5, best_n = None, np.inf, 0

    def tl():
        return max(0.0, _budget - (time.time() - _t0))

    def update(h, n):
        nonlocal best_h, best_c5, best_n
        h = np.clip(h, 0, 1)
        c = compute_c5(h)
        if c < best_c5 - 1e-12:
            best_h, best_c5, best_n = h.copy(), c, n

    # Phase 1: n=90
    n1 = 90
    candidates = []
    try:
        h0 = np.asarray(initial_h_values, dtype=float).copy()
        if len(h0) != n1:
            h0 = upsample_linear(h0, n1)
        candidates.append(h0)
    except Exception:
        pass
    candidates.append(make_random_binary(n1, rng))
    candidates.append(make_greedy_fill(n1, rng))
    for s2 in range(3):
        candidates.append(make_random_binary(n1, np.random.RandomState(seed + 100 + s2)))
        candidates.append(make_greedy_fill(n1, np.random.RandomState(seed + 200 + s2)))

    for ci, hc in enumerate(candidates):
        if tl() < 80:
            break
        hc = project_binary(hc, n1)
        hb, cb = sa_pair_search(hc, n1, rng, 80000, 0.015, 1e-5)
        update(hb, n1)
        hb2, cb2 = sa_pair_search(hb, n1, rng, 40000, 0.005, 1e-6)
        update(hb2, n1)
        hf, cf = fractional_refine(hb2, n1, rng, n_rounds=150)
        update(hf, n1)

    # More SA restarts at n=90
    restarts = 0
    while tl() > 120:
        hr = best_h + rng.normal(0, 0.3, best_n)
        hr = project_binary(hr, best_n)
        hb, cb = sa_pair_search(hr, best_n, rng, 60000, 0.012, 1e-5)
        update(hb, best_n)
        hb2, cb2 = sa_pair_search(hb, best_n, rng, 30000, 0.004, 1e-6)
        update(hb2, best_n)
        hf, cf = fractional_refine(hb2, best_n, rng, n_rounds=100)
        update(hf, best_n)
        restarts += 1

    # Fractional polish on best
    if tl() > 40:
        hf, cf = fractional_refine(best_h, best_n, rng, n_rounds=300)
        update(hf, best_n)

    # Phase 2: n=180
    if tl() > 150:
        n2 = 180
        h_up = project_binary(upsample_linear(best_h, n2), n2)
        update(h_up, n2)
        for s3 in range(2):
            hr = project_binary(upsample_linear(best_h, n2) + rng.normal(0, 0.4, n2), n2)
            hb, cb = sa_pair_search(hr, n2, rng, 30000, 0.012, 1e-5)
            update(hb, n2)
            hb2, cb2 = sa_pair_search(hb, n2, rng, 20000, 0.003, 1e-6)
            update(hb2, n2)
            hf, cf = fractional_refine(hb2, n2, rng, n_rounds=100)
            update(hf, n2)
        if tl() > 40:
            hf, cf = fractional_refine(best_h, best_n, rng, n_rounds=200)
            update(hf, best_n)

    # Phase 3: n=120 (alternative)
    if tl() > 100 and best_n == 90:
        n3 = 120
        h_up = project_binary(upsample_linear(best_h, n3), n3)
        update(h_up, n3)
        hr = project_binary(h_up + rng.normal(0, 0.3, n3), n3)
        hb, cb = sa_pair_search(hr, n3, rng, 30000, 0.010, 1e-5)
        update(hb, n3)
        hb2, cb2 = sa_pair_search(hb, n3, rng, 20000, 0.003, 1e-6)
        update(hb2, n3)
        hf, cf = fractional_refine(hb2, n3, rng, n_rounds=100)
        update(hf, n3)

    # Final polish
    if tl() > 15:
        hf, cf = fractional_refine(best_h, best_n, rng, n_rounds=200)
        update(hf, best_n)

    if best_h is None:
        best_h = np.zeros(90)
        best_h[:45] = 1.0
        best_n = 90
    best_h = np.clip(best_h, 0, 1)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
