from scipy.optimize import minimize
import time

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def _fix(h, n):
    t = 0.5 * n
    for _ in range(200):
        d = t - np.sum(h)
        if abs(d) < 1e-14:
            break
        mask = (h < 1.0) if d > 0 else (h > 0.0)
        k = int(np.sum(mask))
        if k == 0:
            break
        h = np.clip(h + (d / k) * mask, 0.0, 1.0)
    return h

def _interp(h, n_new):
    n0 = len(h)
    xo = np.arange(n0) * (2.0 / n0)
    xn = np.arange(n_new) * (2.0 / n_new)
    h_new = np.interp(xn, xo, h)
    return _fix(h_new, n_new)

def _softmax_obj_and_grad(h, alpha, mu):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    # penalty for integral constraint
    s = np.sum(h) - n * 0.5
    obj = c  # we'll take smooth max
    m = np.max(c)
    log_e = np.clip(alpha * (c - m), -50, 50)
    e = np.exp(log_e)
    p = e / e.sum()
    smooth_max = m + np.log(e.sum()) / alpha
    
    # penalty term
    obj_val = smooth_max + mu * s * s
    
    # gradient
    nshifts = 2 * n - 1
    grad = np.zeros(n)
    for m in range(nshifts):
        s_shift = m - (n - 1)
        w = p[m] * dx
        if w < 1e-15:

        if s_shift >= 0:
            # First index: j in [0, n-s_shift), contribution w*(1-h[j+s_shift])
            grad[:n - s_shift] += w * (1.0 - h[s_shift:])
            # Second index: j in [s_shift, n), contribution -w*h[j-s_shift]
            grad[s_shift:] -= w * h[:n - s_shift]
        else:
            ss = -s_shift
            # First index: j in [ss, n), contribution w*(1-h[j-ss])
            grad[ss:] += w * (1.0 - h[:n - ss])
            # Second index: j in [0, n-ss), contribution -w*h[j+ss]
            grad[:n - ss] -= w * h[ss:]
    
    # Add penalty gradient
    grad += 2 * mu * s * np.ones(n)
    
    return obj_val, grad

    return np.sum(h) - len(h) * 0.5

def _lsq_jac(h):
    return np.ones(len(h))

def _try_lbfgsb(h0, n, alpha, mu, maxiter, t0, budget):

    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def _fix(h, n):
    t = 0.5 * n
    for _ in range(200):
        d = t - np.sum(h)
        if abs(d) < 1e-14:
            break
        mask = (h < 1.0) if d > 0 else (h > 0.0)
        k = int(np.sum(mask))
        if k == 0:
            break
        h = np.clip(h + (d / k) * mask, 0.0, 1.0)
    return h

def _interp(h, n_new):
    n0 = len(h)
    xo = np.arange(n0) * (2.0 / n0)
    xn = np.arange(n_new) * (2.0 / n_new)
    return _fix(np.clip(np.interp(xn, xo, h), 0, 1), n_new)

def _obj_grad(h, alpha, mu):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(c)
    clipped = np.clip(alpha * (c - m), -50, 50)
    e = np.exp(clipped)
    p = e / e.sum()
    val = m + np.log(e.sum()) / alpha
    if mu > 0:
        s = np.sum(h) - 0.5 * n
        val += mu * s * s
    ns = 2 * n - 1
    grad = np.zeros(n)
    for m_i in range(ns):
        s = m_i - (n - 1)
        w = p[m_i] * dx
        if w < 1e-15:
            continue
        if s >= 0:
            if n - s > 0:
                grad[:n - s] += w * (1.0 - h[s:])
                grad[s:] -= w * h[:n - s]
        else:
            ss = -s
            if n - ss > 0:
                grad[ss:] += w * (1.0 - h[:n - ss])
                grad[:n - ss] -= w * h[ss:]
    if mu > 0:
        s_val = np.sum(h) - 0.5 * n
        grad += 2 * mu * s_val
