```python
    dx = 2.0 / n
    grad = np.zeros(n)
    if lag >= 0:
        grad[:n - lag] += (1.0 - h[lag:])
        grad[lag:] -= h[:n - lag]
    else:
        a = -lag
        grad[a:] += (1.0 - h[:n - a])
        grad[:n - a] -= h[a:]
    return dx * grad

        c, lag = corr_max_info(h, n)
        g = corr_grad(h, n, lag)
```

```python
    if time.time() - t0 < budget_s - 20:
        hn = do_slsqp(bh, n, 0.00001, 600, 1e-15)
        c5 = compute_c5(hn)
        if c5 < bc:
            bc, bh = c5, hn

    # Phase 7: Micro local search
    if time.time() - t0 < budget_s - 5:
        for _ in range(10000):
            if time.time() - t0 > budget_s - 3:
                break
            hp = bh + np.random.randn(n) * 0.002
            hp = repair(hp, n)
            c5 = compute_c5(hp)
            if c5 < bc:
                bc, bh = c5, hp

    # Final binary check
    hb = to_binary(bh, n)
    cb = compute_c5(hb)
    if cb < bc:
        bc, bh = cb, hb

    fh = repair(bh, n)
    fc = compute_c5(fh)
    return (fh.tolist(), float(fc), n)
```
