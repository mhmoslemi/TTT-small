        h[fr] += d / c
        h = np.clip(h, 0.0, 1.0)
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _lss(x, alpha):
    n = _N
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = float(np.max(c))
    e = np.exp(np.clip(c - m, -50.0, 50.0))
    return m + np.log(np.sum(e)) / alpha

def _eqfun(x):
    return np.sum(x) - _N * 0.5

def _eqjac(x):
    return np.ones(_N)

def _slsqp(h0, alpha, maxiter):
    global _N
    n = _N
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eqfun, 'jac': _eqjac}]
    try:
        res = minimize(_lss, h0, (alpha,), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None:
            return _proj(np.clip(res.x, 0.0, 1.0), n * 0.5)
    except Exception:
        pass
    return None

def _sg_step(h, alpha):
    n = _N
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    ki = int(np.argmax(c))
    lag = ki - (n - 1)
    m = np.arange(n)
    g = np.zeros(n)
    v1 = (m + lag >= 0) & (m + lag < n)
    g[v1] += (1.0 - h[m[v1] + lag])
    v2 = (m - lag >= 0) & (m - lag < n)
    g[v2] -= h[m[v2] - lag]
    g *= dx
    return _proj(h - alpha * g, n * 0.5), float(np.max(c) * dx)

def _interp_ds(h0, n0, n1):
    x0 = np.arange(n0) * 2.0 / n0
    x1 = np.arange(n1) * 2.0 / n1
    return _proj(np.clip(np.interp(x1, x0, h0), 0.0, 1.0), n1 * 0.5)

def _block_flip(h, n, rng):
    w = rng.randint(20, max(21, n // 4))
    s = rng.randint(0, n - w)
    hf = h.copy()
    hf[s:s + w] = 1.0 - hf[s:s + w]
    return _proj(hf, n * 0.5)

def run(seed=42, budget_s=1100, **kwargs):
    global _N
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        bh = np.array(initial_h_values, dtype=float).copy()
    else:
        bh = np.full(400, 0.5)
    n = len(bh)
    _N = n
    bh = _proj(bh, n * 0.5)
    bc = _eval(bh)

    def _rem():
        return budget_s - (time.time() - t0)

    # Phase 1: SLSQP at starting resolution, increasing alpha
    for alpha in [60.0, 150.0, 400.0, 1000.0, 2500.0]:
        if _rem() < 20:
            break
        mi = min(6000, max(800, int(_rem() * 2.5)))
        ht = _slsqp(bh, alpha, mi)
        if ht is not None:
            ct = _eval(ht)
            if ct < bc:
                bh, bc = ht, ct

    # Phase 2: Subgradient diversification with SLSQP refinement
    for trial in range(12):
        if _rem() < 60:
            break
        # Diversification strategy: mix of subgradient and block flip
        hc = bh.copy()
        if trial % 3 == 0:
            # Random block flip
            hc = _block_flip(bh, n, rng)
        elif trial % 3 == 1:
            # Subgradient walk
            a = 0.3
            for _ in range(400):
                if _rem() < 10:
                    break
                hc, _ = _sg_step(hc, a)
                a *= 0.997
        else:
            # Gaussian perturbation on subset
            idx = rng.choice(n, n // 4, replace=False)
            hc[idx] = np.clip(hc[idx] + rng.randn(n // 4) * 0.15, 0, 1)
            hc = _proj(hc, n * 0.5)

        # SLSQP refine
        if _rem() < 25:
            break
        ht = _slsqp(hc, 300.0, min(3000, max(500, int(_rem() * 2))))
        if ht is not None:
            ct = _eval(ht)
            if ct < bc:
                bh, bc = ht, ct
        # Also try from the subgradient point directly
        hc2 = _proj(np.clip(hc, 0, 1), n * 0.5)
        ct2 = _eval(hc2)
        if ct2 < bc:
            bh, bc = hc2, ct2
            if _rem() > 20:
                ht2 = _slsqp(bh, 600.0, 2000)
                if ht2 is not None:
                    ct3 = _eval(ht2)
                    if ct3 < bc:
                        bh, bc = ht2, ct3

    # Phase 3: Multi-resolution search
    for nn in [300, 350, 450, 500, 550]:
        if _rem() < 70:
            break
        hi = _interp_ds(bh, n, nn)
        Ci = _eval(hi)
        improved = False
        for alpha in [120.0, 500.0, 1500.0]:
            if _rem() < 30:
                break
            _N = nn
            ht = _slsqp(hi, alpha, min(4000, max(600, int(_rem() * 2))))
            if ht is not None:
                ct = _eval(ht)
                if ct < Ci:
                    hi, Ci = ht, ct
                    improved = True
        if Ci < bc:
            bh, bc, n = hi, Ci, nn
            _N = n

    # Phase 4: Coarse-to-fine: optimize at small n, interpolate up
    if _rem() > 120:
        for nc in [150, 200]:
            if _rem() < 60:
                break
            hc = _interp_ds(bh, n, nc)
            _N = nc
            for alpha in [200.0, 1000.0]:
                if _rem() < 20:
                    break
                ht = _slsqp(hc, alpha, 3000)
                if ht is not None:
                    ct = _eval(ht)
                    if ct < _eval(hc):
                        hc = ht
            # Interpolate back up
            hi = _interp_ds(hc, nc, n)
            _N = n
            ht = _slsqp(hi, 800.0, min(3000, max(500, int(_rem() * 1.5))))
            if ht is not None:
                ct = _eval(ht)
                if ct < bc:
                    bh, bc = ht, ct

    # Phase 5: Final high-alpha polish at multiple alphas
    _N = n
    for alpha in [1000.0, 4000.0, 10000.0]:
        if _rem() < 12:
            break
        ht = _slsqp(bh, alpha, 2500)
        if ht is not None:
            ct = _eval(ht)
            if ct < bc:
                bh, bc = ht, ct

    # Final block-flip local search
    for _ in range(50):
        if _rem() < 5:
            break
        hb = _block_flip(bh, n, rng)
        cb = _eval(hb)
        if cb < bc:
            bh, bc = hb, cb

    bh = _proj(bh, n * 0.5)
    bc = _eval(bh)
    return bh, bc, n
