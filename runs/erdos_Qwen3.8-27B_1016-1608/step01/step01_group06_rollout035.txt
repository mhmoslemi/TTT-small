    best_score = compute_score(h_init)
    x_init = np.linspace(0.0, 2.0, n_init, endpoint=False)

    n_list = sorted(set([n_init, 40, 48, 50, 64, 80]))

    for n_p in n_list:
        if time.time() - t0 > budget_s - 100:
            break
        _cur_n = n_p
        x_new = np.linspace(0.0, 2.0, n_p, endpoint=False)
        h0 = np.interp(x_new, x_init, h_init)
        h0 = project_feasible(h0, n_p)
        s0 = compute_score(h0)
        if s0 < best_score:
            best_score = s0
            best_h = h0.copy()

        bounds = [(0.0, 1.0)] * n_p
        cons = [{'type': 'eq', 'fun': sum_con_fun}]

        # Phase 1: SLSQP smooth max - multiple temperatures
        for tau in [30, 60, 150, 400, 1000]:
            if time.time() - t0 > budget_s - 60:
                break
            _tau_cur = tau
            try:
                res = minimize(smooth_max_obj, h0, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 600, 'ftol': 1e-13, 'iprint': 0})
                hc = project_feasible(res['x'], n_p)
                sc = compute_score(hc)
                if sc < best_score:
                    best_score = sc
                    best_h = hc.copy()
                if sc < s0:
                    s0 = sc
                    h0 = hc.copy()
            except Exception:
                pass

        # Phase 2: Subgradient iterations
        if time.time() - t0 < budget_s - 50:
            h_sub = h0.copy()
            for it in range(5000):
                if time.time() - t0 > budget_s - 20:
                    break
                alpha = 0.015 * (1.0 + 0.5 * np.cos(0.02 * it))
                h_sub = subgradient_step(h_sub, alpha)
                h_sub = project_feasible(h_sub, n_p)
                if it % 300 == 0:
                    sc = compute_score(h_sub)
                    if sc < best_score:
                        best_score = sc
                        best_h = h_sub.copy()
                    if sc < s0:
                        s0 = sc
                        h0 = h_sub.copy()
            sc = compute_score(h_sub)
            if sc < best_score:
                best_score = sc
                best_h = h_sub.copy()
            if sc < s0:
                s0 = sc
                h0 = h_sub.copy()

        # Phase 3: Nelder-Mead with topk smoothing
        if time.time() - t0 < budget_s - 40:
            for k_top in [3, 5, 10]:
                if time.time() - t0 > budget_s - 15:
                    break
                _k_top_cur = k_top
                _tau_cur = 60.0
                try:
                    res = minimize(topk_obj, h0, method='Nelder-Mead',
                                   options={'maxiter': 4000, 'xatol': 1e-9, 'fatol': 1e-13})
                    hc = project_feasible(res['x'], n_p)
                    sc = compute_score(hc)
                    if sc < best_score:
                        best_score = sc
                        best_h = hc.copy()
                    if sc < s0:
                        s0 = sc
                        h0 = hc.copy()
                except Exception:
                    pass

        # Phase 4: Direct SLSQP on exact max
        if time.time() - t0 < budget_s - 25:
            try:
                res = minimize(compute_score, h0, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 800, 'ftol': 1e-14, 'iprint': 0})
                hc = project_feasible(res['x'], n_p)
                sc = compute_score(hc)
                if sc < best_score:
                    best_score = sc
                    best_h = hc.copy()
                if sc < s0:
                    s0 = sc
                    h0 = hc.copy()
            except Exception:
                pass

        # Phase 5: Structured random starts + SLSQP
        if time.time() - t0 < budget_s - 60:
            for trial in range(5):
                if time.time() - t0 > budget_s - 20:
                    break
                h_rand = make_structured_init(n_p, trial % 4, rng)
                _tau_cur = 100.0
                try:
                    res = minimize(smooth_max_obj, h_rand, method='SLSQP',
                                   bounds=bounds, constraints=cons,
                                   options={'maxiter': 400, 'ftol': 1e-12, 'iprint': 0})
                    hc = project_feasible(res['x'], n_p)
                    sc = compute_score(hc)
                    if sc < best_score:
                        best_score = sc
                        best_h = hc.copy()
                    if sc < s0:
                        s0 = sc
                        h0 = hc.copy()
                except Exception:
                    pass

    # Final polish on best solution
    if time.time() - t0 < budget_s - 15:
        n_f = len(best_h)
        _cur_n = n_f
        bounds_f = [(0.0, 1.0)] * n_f
        cons_f = [{'type': 'eq', 'fun': sum_con_fun}]

        # Fine subgradient
        h_pol = best_h.copy()
        for it in range(3000):
            if time.time() - t0 > budget_s - 10:
                break
            alpha = 0.004 * (1.0 + 0.5 * np.cos(0.01 * it))
            h_pol = subgradient_step(h_pol, alpha)
            h_pol = project_feasible(h_pol, n_f)
            if it % 300 == 0:
                sc = compute_score(h_pol)
                if sc < best_score:
                    best_score = sc
                    best_h = h_pol.copy()

        # SLSQP final
        if time.time() - t0 < budget_s - 8:
            for tau in [500, 2000]:
                if time.time() - t0 > budget_s - 5:
                    break
                _tau_cur = tau
                try:
                    res = minimize(smooth_max_obj, best_h, method='SLSQP',
                                   bounds=bounds_f, constraints=cons_f,
                                   options={'maxiter': 1000, 'ftol': 1e-14, 'iprint': 0})
                    hf = project_feasible(res['x'], n_f)
                    sf = compute_score(hf)
                    if sf < best_score:
                        best_score = sf
                        best_h = hf.copy()
                except Exception:
                    pass

    return best_h, best_score, len(best_h)
