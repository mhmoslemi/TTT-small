import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def softmax_obj(h, dx, temp):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    return (m + temp * np.log(np.mean(np.exp((corr - m) / temp)))) * dx

def project_h(h, target):
    h = np.array(h, dtype=float).copy()
    np.clip(h, 0.0, 1.0, out=h)
    for _ in range(400):
        ex = h.sum() - target
        if abs(ex) < 1e-15:
            break
        if ex > 0:
            m = h > 1e-14
            if not m.any():
                break
            s = min(ex / m.sum(), 0.999 * h[m].min())
            h[m] -= s
        else:
            m = h < 1.0 - 1e-14
            if not m.any():
                break
            s = min(-ex / m.sum(), 0.999 * (1.0 - h[m]).min())
            h[m] += s
    return h

def slsqp_opt(h0, dx, target, bounds, maxiter, ftol, obj_func=None, opt_args=()):
    if obj_func is None:
        obj_func, opt_args = compute_c5, (dx,)
    try:
        r = minimize(obj_func, h0, args=opt_args, method='SLSQP',
                     bounds=bounds, options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(r.x, target)
    except Exception:
        return None
