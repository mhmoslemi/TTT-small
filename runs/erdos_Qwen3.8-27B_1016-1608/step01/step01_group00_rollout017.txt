            h_bin = binary_project(pop[p] + np.random.randn(n) * 0.05, n)
            blend = np.random.uniform(0.3, 0.8)
            h_new = blend * h_bin + (1.0 - blend) * pop[p]
        else:
            q = (p + 1 + np.random.randint(len(pop) - 1)) % len(pop)
            a = np.random.uniform(0.2, 0.8)
            h_new = a * pop[p] + (1.0 - a) * pop[q]
        h_new = repair(h_new, n)
        c_new = compute_c5(h_new)
        c_old = pop_c[p]
        if c_new < c_old or np.random.random() < np.exp(-(c_new - c_old) / max(T, 1e-10)):
            pop[p] = h_new
            pop_c[p] = c_new
            if c_new < c5_best:
                h_best, c5_best = h_new.copy(), c_new
                # Seed neighbors from best
                for q in range(1, min(3, len(pop))):
                    pop[q] = repair(h_best + np.random.randn(n) * 0.015, n)
                    pop_c[q] = compute_c5(pop[q])
        T = max(T * 0.997, 1e-6)

    # Phase 4: SLSQP + trust-constr polish
    if elapsed() < budget_s * 0.92:
        h = slsqp_opt(h_best, n, maxiter=5000)
        c = compute_c5(h)
        if c < c5_best:
            h_best, c5_best = h, c
    if elapsed() < budget_s * 0.94:
        h = trustcon_opt(h_best, n, maxiter=1500)
        c = compute_c5(h)
        if c < c5_best:
            h_best, c5_best = h, c

    # Phase 5: Resolution sweep
    if elapsed() < budget_s * 0.985:
        for n2 in [n + 8, n - 8, n + 4, n - 4, n + 14, n - 14]:
            if elapsed() > budget_s * 0.98:
                break
            if n2 < 40 or n2 > 200:
                continue
            xp = np.linspace(0.0, 2.0, n, endpoint=False)
            xn = np.linspace(0.0, 2.0, n2, endpoint=False)
            h_interp = np.interp(xn, xp, h_best)
            h_interp = repair(h_interp, n2)
            h2 = slsqp_opt(h_interp, n2, maxiter=1500)
            c2 = compute_c5(h2)
            if c2 < c5_best:
                h_best, c5_best, n = h2, c2, n2

    h_best = repair(h_best, n)
    c5_best = compute_c5(h_best)
    return (h_best, c5_best, n)
