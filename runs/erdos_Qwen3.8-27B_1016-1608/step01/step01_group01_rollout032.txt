The minimum of max_k (c_k + (t-a)*d_k) over t ∈ [t_lo, t_hi] occurs either at a breakpoint or at the boundary. The breakpoints are where two of the linear functions intersect:
c_i + (t-a)*d_i = c_j + (t-a)*d_j
(t-a) = (c_j - c_i) / (d_i - d_j)

So I can:
1. Compute all breakpoints
2. Evaluate the max at each breakpoint and each boundary
3. Pick the best t

    lo, hi = t_lo, t_hi

        mid = 0.5 * (lo + hi)
        # Find active lag
        vals = c_vals + (mid - a) * d_vec
        k_star = np.argmax(vals)
        slope = d_vec[k_star]
        if slope < -1e-14:
            lo = mid  # still decreasing, go right
        elif slope > 1e-14:
            hi = mid  # still increasing, go left
        else:
            break  # at minimum
    t_opt = 0.5 * (lo + hi)
    # Also check boundaries
    cands = [t_lo, t_hi, t_opt]
    best_val = np.inf
    best_t = t_opt
    for tc in cands:
        vals = c_vals + (tc - a) * d_vec
        v = vals.max()
        if v < best_val:
            best_val = v
            best_t = tc
    return best_t, best_val

def compute_d_vec(h, i, j, n):
    # d_k for pair (i, j)
    # d_k = (h[j+sk] - h[i+sk]) + (h[i-sk] - h[j-sk])
    # where sk = k - (n-1)
    # Valid: indices must be in [0, n-1]
    d = np.zeros(2 * n - 1)
    sk = np.arange(-(n-1), n)  # shifts for k=0,...,2n-2
    # h[i+sk] valid when 0 <= i+sk < n
    ip = i + sk
    jp = j + sk
    im = i - sk
    jm = j - sk
    valid_ip = (ip >= 0) & (ip < n)
    valid_jp = (jp >= 0) & (jp < n)
    valid_im = (im >= 0) & (im < n)
    valid_jm = (jm >= 0) & (jm < n)
    d[valid_jp] += h[jp[valid_jp]]
    d[valid_ip] -= h[ip[valid_ip]]
    d[valid_im] += h[im[valid_im]]
    d[valid_jm] -= h[jm[valid_jm]]
