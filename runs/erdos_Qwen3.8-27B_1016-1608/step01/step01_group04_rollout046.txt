        return budget_s - elapsed()
    
    # Phase 1: Multi-eps SLSQP from parent (n=80)
    for eps in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
        if remaining() < 500:
            break
        h = slsqp_pass(h, n, eps, 400, 1e-14)
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h
    
    # Phase 2: Targeted search on worst lags (n=80)
    for _ in range(3):
        if remaining() < 450:
            break
        h = targeted_search(best_h, n, remaining())
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h
        h = slsqp_pass(h, n, 0.0005, 300, 1e-14)
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h
    
    # Phase 3: Coordinate descent / local search (n=80)
    if remaining() > 400:
        h, c = coordinate_descent(best_h, n, iters=200, step=0.01)
        if c < best_c:
            best_c, best_h = c, h
        h, c = coordinate_descent(best_h, n, iters=200, step=0.005)
        if c < best_c:
            best_c, best_h = c, h
    
    # Phase 4: Interpolate to n=200 and optimize
    n2 = 200
    if remaining() > 300:
        x_old = np.linspace(0.0, 2.0, n)
        x_new = np.linspace(0.0, 2.0, n2)
        h2 = np.interp(x_new, x_old, best_h)
        h2 = repair(h2, n2)
        best_h2 = h2.copy()
        best_c2 = compute_c5(h2)
        
        # SLSQP at n=200
        for eps in [0.01, 0.003, 0.001, 0.0003, 0.0001]:
            if remaining() < 150:
                break
            h2 = slsqp_pass(h2, n2, eps, 300, 1e-14)
            c2 = compute_c5(h2)
            if c2 < best_c2:
                best_c2, best_h2 = c2, h2
        
        # Targeted search at n=200
        if remaining() > 200:
            h2 = targeted_search(best_h2, n2, remaining())
            c2 = compute_c5(h2)
            if c2 < best_c2:
                best_c2, best_h2 = c2, h2
            h2 = slsqp_pass(h2, n2, 0.0003, 300, 1e-14)
            c2 = compute_c5(h2)
            if c2 < best_c2:
                best_c2, best_h2 = c2, h2
        
        if best_c2 < best_c:
            best_c, best_h = best_c2, best_h2
            n = n2
    
    # Phase 5: Diverse restarts at current best n (if time allows)
    if remaining() > 200:
        for restart in range(15):
            if remaining() < 80:
                break
            r = restart % 5
            if r == 0:
                hp = best_h + np.random.randn(n) * 0.02
            elif r == 1:
                hp = best_h.copy()
                s = np.random.randint(0, max(1, n - 8))
                l = np.random.randint(3, min(10, n - s))
                hp[s:s+l] = 1.0 - hp[s:s+l]
            elif r == 2:
                k2 = np.random.randint(2, 8)
                ph = np.random.uniform(0, 2*np.pi)
                hp = 0.5 + 0.3*np.sin(2*np.pi*k2*np.arange(n)/n + ph)
                hp += 0.05*(best_h - 0.5)
            elif r == 3:
                hp = best_h.copy()
                idx = np.random.choice(n, size=n//4, replace=False)
                hp[idx] = 1.0 - hp[idx]
            else:
                hp = best_h + np.random.randn(n) * 0.005
            hp = repair(hp, n)
            hn = slsqp_pass(hp, n, 0.003, 200, 1e-13)
            c = compute_c5(hn)
            if c < best_c:
                best_c, best_h = c, hn
            hn = slsqp_pass(hn, n, 0.0005, 200, 1e-14)
            c = compute_c5(hn)
            if c < best_c:
                best_c, best_h = c, hn
    
    # Phase 6: Final fine polish
    if remaining() > 30:
        h = slsqp_pass(best_h, n, 0.00005, 500, 1e-15)
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h
    
    # Phase 7: Tiny random perturbations
    if remaining() > 15:
        for _ in range(5000):
            if elapsed() > budget_s - 5:
                break
            hp = best_h + np.random.randn(n) * 0.003
            hp = repair(hp, n)
            c = compute_c5(hp)
            if c < best_c:
                best_c, best_h = c, hp
    
    fh = repair(best_h, n)
    fc = compute_c5(fh)
    return (fh.tolist(), float(fc), n)
