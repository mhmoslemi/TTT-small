    temps = [50.0, 25.0, 12.0, 6.0, 3.0, 1.5, 0.8, 0.4, 0.2, 0.1, 0.05]
    for T in temps:
        if time_left() < 25:
            break
        h_new = slsqp_softmax(best_h, dx, target, bounds, T, maxiter=400)
        do_update(h_new)

    # Phase 2: Active-set balancing (focus on lags near max)
    if time_left() > 30:
        for it in range(10):
            if time_left() < 15:
                break
            corr = compute_corr(best_h)
            m = np.max(corr)
            active = np.where(corr >= m * 0.998)[0]
            if len(active) < 2:
                active = np.where(corr >= m * 0.99)[0]
            if len(active) < 2:
                active = np.where(corr >= m * 0.95)[0]
            if len(active) < 2:
                continue
            h_new = slsqp_active(best_h, dx, target, bounds, active, 0.05, maxiter=400)
            do_update(h_new)
            # Second pass with tighter threshold
            corr2 = compute_corr(best_h)
            m2 = np.max(corr2)
            active2 = np.where(corr2 >= m2 * 0.999)[0]
            if len(active2) >= 2:
                h_new = slsqp_active(best_h, dx, target, bounds, active2, 0.02, maxiter=300)
                do_update(h_new)

    # Phase 3: Coordinate descent with multiple step sizes
    if time_left() > 20:
        h_cd = best_h.copy()
        c5_cd = best_c5
        for ss in [0.05, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
            if time_left() < 10:
                break
            for sweep in range(6):
                if time_left() < 5:
                    break
                improved = False
                for j in range(n):
                    for delta in [ss, -ss]:
                        h_t = h_cd.copy()
                        h_t[j] = np.clip(h_t[j] + delta, 0.0, 1.0)
                        h_t = projection(h_t, target)
                        c5_t = compute_c5(h_t, dx)
                        if c5_t < c5_cd - 1e-14:
                            h_cd[:] = h_t
                            c5_cd = c5_t
                            improved = True
                            if c5_cd < best_c5:
                                best_h = h_cd.copy()
                                best_c5 = c5_cd
                            break
                if not improved:
                    break

    # Phase 4: Paired swaps (transfer mass between two coordinates)
    if time_left() > 30:
        h_sw = best_h.copy()
        c5_sw = best_c5
        for ss in [0.02, 0.01, 0.005, 0.002]:
            if time_left() < 15:
                break
            interior = np.where((h_sw > 0.05) & (h_sw < 0.95))[0]
            if len(interior) < 2:
                break
            for sweep in range(3):
                if time_left() < 10:
                    break
                for a in range(len(interior)):
                    for b in range(a + 1, len(interior)):
                        j, k = interior[a], interior[b]
                        for delta in [ss, -ss]:
                            h_t = h_sw.copy()
                            h_t[j] += delta
                            h_t[k] -= delta
                            if 0 <= h_t[j] <= 1 and 0 <= h_t[k] <= 1:
                                c5_t = compute_c5(h_t, dx)
                                if c5_t < c5_sw - 1e-14:
                                    h_sw[:] = h_t
                                    c5_sw = c5_t
                                    if c5_sw < best_c5:
                                        best_h = h_sw.copy()
                                        best_c5 = c5_sw
                                    break

    # Phase 5: Random restarts with diversity
    if time_left() > 50:
        n_rest = min(12, int((time_left() - 20) / 35))
        for r in range(n_rest):
            if time_left() < 30:
                break
            mode = r % 4
            if mode == 0:
                h0 = projection(best_h + rng.randn(n) * 0.08, target)
            elif mode == 1:
                h0 = projection(best_h + rng.randn(n) * 0.2, target)
            elif mode == 2:
                t = np.linspace(0, 4 * np.pi, n)
                h0 = 0.5 + 0.35 * np.sin(t + rng.uniform(0, 2 * np.pi))
                h0 = projection(h0, target)
            else:
                h0 = rng.random(n) * 0.8 + 0.1
                h0 = projection(h0, target)
            T_r = 3.0 + 5.0 * rng.random()
            h_new = slsqp_softmax(h0, dx, target, bounds, T_r, maxiter=200)
            do_update(h_new)
            if time_left() > 20 and h_new is not None:
                corr = compute_corr(best_h)
                m = np.max(corr)
                active = np.where(corr >= m * 0.99)[0]
                if len(active) >= 2:
                    h_new2 = slsqp_active(best_h, dx, target, bounds, active, 0.05, maxiter=200)
                    do_update(h_new2)

    # Phase 6: Final polish with very low temperature
    if time_left() > 15:
        h_new = slsqp_softmax(best_h, dx, target, bounds, 0.02, maxiter=500)
        do_update(h_new)
        if time_left() > 10:
            corr = compute_corr(best_h)
            m = np.max(corr)
            active = np.where(corr >= m * 0.999)[0]
            if len(active) >= 2:
                h_new = slsqp_active(best_h, dx, target, bounds, active, 0.01, maxiter=500)
                do_update(h_new)
        if time_left() > 8:
            h_new = slsqp_softmax(best_h, dx, target, bounds, 0.01, maxiter=400)
            do_update(h_new)

    best_h = projection(best_h, target)
    final_c5 = compute_c5(best_h, dx)
    return (best_h, final_c5, n)
