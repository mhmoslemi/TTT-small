import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def to_binary(h):
    n = len(h)
    k = n // 2
    idx = np.argpartition(h, k)[k:]
    out = np.zeros(n)
    out[idx] = 1.0
    return out

def best_improve(h):
    n = len(h)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    best_c = compute_c5(h)
    best_pair = None

        a = ones[i]
        for j in range(len(zeros)):
            b = zeros[j]
            h[a] = 0.0
            h[b] = 1.0
            c = compute_c5(h)
            h[a] = 1.0
            h[b] = 0.0
            if c < best_c - 1e-15:

                best_pair = (a, b)

        h[best_pair[0]] = 0.0
        h[best_pair[1]] = 1.0
