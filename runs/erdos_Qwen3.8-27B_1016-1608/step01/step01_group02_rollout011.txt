def project_feasible(h, target):
    h = np.array(h, dtype=float, copy=True)
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        excess = h.sum() - target
        if abs(excess) < 1e-12:
            break
        if excess > 0:
            mask = h > 0
            if mask.sum() == 0:
                break
            adj = min(excess / mask.sum(), h[mask].min() * 0.999)
            h[mask] -= adj
        else:
            mask = h < 1
            if mask.sum() == 0:

            adj = min(-excess / mask.sum(), (1.0 - h[mask]).min() * 0.999)

def project_feasible(h, target):
    h = np.array(h, dtype=float, copy=True)
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        excess = h.sum() - target
        if abs(excess) < 1e-13:
            break
        if excess > 0:
            mask = h > 0
            if mask.sum() == 0:
                break
            adj = min(excess / mask.sum(), h[mask].min() * 0.999)
            h[mask] -= adj
        else:
            mask = h < 1
            if mask.sum() == 0:
                break
            adj = min(-excess / mask.sum(), (1.0 - h[mask]).min() * 0.999)
