```python
def smooth_max_grad(h, dx, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    n = len(h)
    # Stable log-sum-exp
    mc = np.max(corr)
    exp_c = np.exp(T * (corr - mc))
    p = exp_c / np.sum(exp_c)
    
    # lag_k = k - (n-1) for k = 0..2n-2
    lags = np.arange(2*n-1) - (n-1)  # shape (2n-1), values -(n-1)..(n-1)
    
    grad = np.zeros(n)
    for m in range(n):
        # Term 1: sum_k p[k] * I[0 <= m - lag_k < n]
        valid1 = (m - lags >= 0) & (m - lags < n)
        # Term 2: sum_k p[k] * h[m - lag_k] * I[0 <= m - lag_k < n]
        idx1 = m - lags  # shape (2n-1)
        term2 = np.where(valid1, p * h[np.clip(idx1, 0, n-1)], 0)
        # Term 3: sum_k p[k] * h[m + lag_k] * I[0 <= m + lag_k < n]
        valid3 = (m + lags >= 0) & (m + lags < n)
        idx3 = m + lags
        term3 = np.where(valid3, p * h[np.clip(idx3, 0, n-1)], 0)
        grad[m] = dx * (np.sum(p[valid1]) - np.sum(term2) - np.sum(term3))
    return grad
```

```python
    mc = np.max(corr)
    return float((mc + np.log(np.sum(np.exp(T * (corr - mc)))) / T) * dx)

    n = len(h)
```

```python
import time
```

```python
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

    return np.correlate(h, 1.0 - h, mode="full")

        grad[m] = dx * (t1 - t2 - t3)

    return grad

def project_h(h, target):
    h = np.array(h, dtype=float, copy=True)
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        excess = h.sum() - target
        if abs(excess) < 1e-14:
            return h
        if excess > 0:
            mask = h > 0
            if mask.sum() == 0:
                return h
            step = min(excess / mask.sum(), h[mask].min() - 1e-15)
            if step <= 0:
                return h
            h[mask] -= step
        else:
            mask = h < 1
            if mask.sum() == 0:
                return h
            step = min(-excess / mask.sum(), (1.0 - h[mask]).min() - 1e-15)
            if step <= 0:
                return h
            h[mask] += step
    return h

def subgrad_step(h, dx, alpha):
    corr = compute_corr(h)
    kmax = int(np.argmax(corr))
    lag = kmax - (len(h) - 1)
    n = len(h)
    g = np.zeros(n)

        jr = m + lag
        if 0 <= jr < n:
            v -= h[jr]
        g[m] = v * dx
    return project_h(h - alpha * g, n / 2.0)
```
