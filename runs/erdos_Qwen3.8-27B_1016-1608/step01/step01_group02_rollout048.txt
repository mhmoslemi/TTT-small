        # (1-h[m-k]) for m = 0..n-1 where 0 <= m-k < n
        m_lo = max(0, k)
        m_hi = min(n, n + k)
        if m_lo < m_hi:
            # m ranges from m_lo to m_hi-1
            # (1-h[m-k]) for m in [m_lo, m_hi)
            grad[m_lo:m_hi] += p[j] * dx * one_minus_h[m_lo-k:m_hi-k]
        # -h[m+k] for m = 0..n-1 where 0 <= m+k < n
        m_lo2 = max(0, -k)
        m_hi2 = min(n, n - k)
        if m_lo2 < m_hi2:
            grad[m_lo2:m_hi2] -= p[j] * dx * h[m_lo2+k:m_hi2+k]
    
    return L, grad
