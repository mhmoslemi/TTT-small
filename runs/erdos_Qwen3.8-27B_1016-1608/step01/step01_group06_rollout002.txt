            h_cand = project_feasible(res['x'], n_pts)
            s_cand = compute_score(h_cand)
            if s_cand < best_score:
                best_score = s_cand
                best_h = h_cand
        except Exception:
            pass

    # Phase 5: Tiny final SLSQP with very tight tolerance
    if remaining() > 15:
        n_pts = len(best_h)
        _g_n = n_pts
        bounds = [(0.0, 1.0)] * n_pts
        constraints = [{'type': 'eq', 'fun': _eq_constraint}]
        try:
            res = minimize(soft_max_score, best_h, method='SLSQP',
                           args=(200.0,), bounds=bounds, constraints=constraints,
                           options={'maxiter': 500, 'ftol': 1e-15})
            h_cand = project_feasible(res['x'], n_pts)
            s_cand = compute_score(h_cand)
            if s_cand < best_score:
                best_score = s_cand
                best_h = h_cand
        except Exception:
            pass

    best_h = project_feasible(best_h, len(best_h))
    best_score = compute_score(best_h)
    return best_h, best_score, len(best_h)
