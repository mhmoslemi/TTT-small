import numpy as np
import time

def _corr_max(h):
    """Compute C5 = max(correlate(h, 1-h, 'full') * dx)"""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def _corr_full(h):
    """Return the full correlation * dx"""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return c * dx

def _subgrad(h, lag):
    """Subgradient of the max-correlation at the given lag index.
    lag is the index into the full correlation (0 to 2n-2).
    The actual shift m = lag - (n-1).
    Returns gradient vector g where g[i] = d(c[lag])/d(h[i])
    """
    n = len(h)
    m = lag - (n - 1)
    g = np.ones(n)  # from sum_j h[j] term
    # Subtract h[i+m] where valid
    if m >= 0:
        g[:n-m] -= h[m:]  # h[i+m] for i=0..n-1-m
    else:
        g[-m:] -= h[:n+m]  # h[i+m] for i=-m..n-1
    # Subtract h[i-m] where valid
    if m >= 0:
        g[m:] -= h[:n-m]  # h[i-m] for i=m..n-1
    else:
        g[:n+m] -= h[-m:]  # h[i-m] for i=0..n+m-1
    return g

def _project(h, n):
    """Project onto {h: 0<=h<=1, sum(h)=n/2}"""
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        d = target - np.sum(h)
        if abs(d) < 1e-12:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        k = np.sum(mask)
        if k == 0:
            break
        h[mask] += d / k
        h = np.clip(h, 0.0, 1.0)
    return h

def _soft_max_obj(h, alpha):
    """Smoothed max objective using log-sum-exp"""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(c)
    # log-sum-exp for stability
    v = c - m
    e = np.exp(np.clip(v, -50, 50))
    return m + np.log(np.sum(e)) / alpha

def _random_pattern(n, rng, pattern_type=0):
    """Generate a random starting pattern"""
    h = np.zeros(n)
    if pattern_type == 0:
        # Random binary
        h[rng.choice(n, n//2, replace=False)] = 1.0
    elif pattern_type == 1:
        # Block pattern
        block = rng.randint(4, max(5, n//10))
        for i in range(0, n, 2*block):
            h[i:i+block] = 1.0
        h = _project(h, n)
    elif pattern_type == 2:
        # Periodic with jitter
        period = rng.randint(3, max(4, n//20))
        for i in range(n):
            h[i] = 1.0 if (i % period) < period//2 else 0.0
        h += rng.randn(n) * 0.1
        h = _project(np.clip(h, 0, 1), n)
    elif pattern_type == 3:
        # Smooth transition
        x = np.linspace(0, 2*np.pi*rng.randint(2,6), n)
        h = 0.5 + 0.5*np.sin(x)
        h += rng.randn(n) * 0.05
        h = _project(np.clip(h, 0, 1), n)
    else:
        h = np.full(n, 0.5)
        h += rng.randn(n) * 0.2
        h = _project(np.clip(h, 0, 1), n)
    return h

def _subgrad_descent(h0, n, t0, budget, max_iters=5000):
    """Subgradient descent for the max correlation"""
    h = h0.copy()
    best_h = h.copy()
    best_c = _corr_max(h)
    lr = 0.5 / np.sqrt(n)
    
    for it in range(max_iters):
        if budget - (time.time() - t0) < 5:
            break
        # Compute correlation
        c = _corr_full(h)
        cur = c.max()
        if cur < best_c:
            best_c = cur
            best_h = h.copy()
        
        # Subgradient at argmax
        lag = int(np.argmax(c))
        g = _subgrad(h, lag)
        
        # Generalized subgradient: also consider top few lags
        top_k = np.argsort(c)[-3:]
        g_sum = np.zeros(n)
        for lag_i in top_k:
            g_sum += _subgrad(h, int(lag_i))
        g_sum /= len(top_k)
        
        # Update
        h_new = h - lr * (g_sum + g) / 2
        h_new = _project(h_new, n)
        
        # Check improvement
        c_new = _corr_max(h_new)
        if c_new > best_c * 1.001:  # if got much worse, revert
            h_new = h.copy()
            lr *= 0.95
        else:
            h = h_new
        
        # Decrease learning rate
        lr = max(lr * 0.9995, 0.001 / n)
    
    return best_h, best_c

def _local_search(h, n, t0, budget, n_trials=2000):
    """Local search: flip small blocks"""
    best_h = h.copy()
    best_c = _corr_max(h)
    rng = np.random.RandomState(12345)
    
    for it in range(n_trials):
        if budget - (time.time() - t0) < 3:
            break
        # Pick a random location and try to modify a small neighborhood
        i = rng.randint(0, n - 2)
        width = rng.randint(1, 4)
        
        # Try adding a small perturbation
        delta = rng.uniform(-0.3, 0.3) * np.ones(width)
        h_new = best_h.copy()
        h_new[i:i+width] += delta
        h_new = _project(h_new, n)
        
        c_new = _corr_max(h_new)
        if c_new < best_c - 1e-10:
            best_h = h_new
            best_c = c_new
    
    return best_h, best_c

def _try_resolution(h, new_n, t0, budget):
    """Interpolate to new resolution and optimize briefly"""
    n_orig = len(h)
    x_old = np.arange(n_orig) * 2.0 / n_orig
    x_new = np.arange(new_n) * 2.0 / new_n
    h_new = np.interp(x_new, x_old, h)
    h_new = _project(np.clip(h_new, 0, 1), new_n)
    
    # Quick subgradient descent
    h_new, c_new = _subgrad_descent(h_new, new_n, t0, budget, max_iters=2000)
    return h_new, c_new

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    # Get initial values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(400, 0.5)
    
    n_init = len(h_init)
    h_init = _project(h_init, n_init)
    
    best_h = h_init.copy()
    best_c = _corr_max(h_init)
    
    # Phase 1: Subgradient descent from initial
    bh, bc = _subgrad_descent(h_init, n_init, t0, budget_s, max_iters=3000)
    if bc < best_c:
        best_h, best_c = bh, bc
    
    # Phase 2: Try different starting patterns at same resolution
    for pt in range(5):
        if budget_s - (time.time() - t0) < 30:
            break
        h_rand = _random_pattern(n_init, rng, pt)
        bh, bc = _subgrad_descent(h_rand, n_init, t0, budget_s, max_iters=1500)
        if bc < best_c:
            best_h, best_c = bh, bc
    
    # Phase 3: Try different resolutions
    for nn in [200, 300, 500, 600, 800]:
        if budget_s - (time.time() - t0) < 60:
            break
        h_new, c_new = _try_resolution(best_h, nn, t0, budget_s)
        if c_new < best_c:
            best_h, best_c = h_new, c_new
    
    # Phase 4: Local search at best resolution
    nb = len(best_h)
    best_h, best_c = _local_search(best_h, nb, t0, budget_s, n_trials=3000)
    
    # Phase 5: Final subgradient polish
    if budget_s - (time.time() - t0) > 20:
        best_h, best_c = _subgrad_descent(best_h, nb, t0, budget_s, max_iters=2000)
    
    # Ensure exact integral
    best_h = _project(best_h, len(best_h))
    best_c = _corr_max(best_h)
    
    return best_h, best_c, len(best_h)

def _subgrad(h, i, n):
    """Subgradient of result[i] w.r.t. h"""
    lo = max(0, i - n + 1)
    hi = min(n - 1, i)
    if lo > hi:
        return np.zeros(n)
    g = np.zeros(n)
    indices = np.arange(lo, hi + 1)
    g[indices] = 1.0 - 2.0 * h[i - indices]
    return g

import numpy as np
import time

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def _c5_full(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(h, n):
    target = 0.5 * n
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        d = target - np.sum(h)
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        k = int(np.sum(mask))
        if k == 0:
            break
        h[mask] += d / k
        h = np.clip(h, 0.0, 1.0)
    return h

def _subgrad_at(h, idx, n):
    lo = max(0, idx - n + 1)
    hi_idx = min(n - 1, idx)
    if lo > hi_idx:
        return np.zeros(n)
    g = np.zeros(n)
    idxs = np.arange(lo, hi_idx + 1)
    g[idxs] = 1.0 - 2.0 * h[idx - idxs]
    return g

def _opt_subgrad(h0, n, t0, budget, max_iter=5000, lr0=None):
    if lr0 is None:
        lr0 = 2.0 / np.sqrt(n)
    h = h0.copy()
    bh = h.copy()
    bc = _c5(h)
    lr = lr0
    for it in range(max_iter):
        if budget - (time.time() - t0) < 3:
            break
        c = _c5_full(h)
        cur = float(c.max())
        if cur < bc:
            bc = cur
            bh = h.copy()
        # Top-k lags
        top_k = min(5, len(c))
        top_indices = np.argsort(c)[-top_k:]
        # Weighted subgradient
        w = c[top_indices] - c[top_indices[0]] + 1.0  # shift so min weight is 1
        w = w / w.sum()
        g = np.zeros(n)
        for i, lag in enumerate(top_indices):
            g += w[i] * _subgrad_at(h, int(lag), n)
        # Update
        h_new = h - lr * g
        h_new = _project(h_new, n)
        c_new = _c5(h_new)
        if c_new < bc:
            pass  # accepted
        elif c_new < cur + lr * 0.1:
            h = h_new  # accept small regression
        else:
            h = h_new  # always accept, but decrease lr more
            lr *= 0.98
        lr *= 0.9997
        lr = max(lr, 0.01 / n)
    return bh, bc

def _rand_init(n, rng, typ):
    h = np.zeros(n)
    if typ == 0:
        idx = rng.choice(n, n // 2, replace=False)
        h[idx] = 1.0
    elif typ == 1:
        b = rng.randint(3, max(4, n // 8))
        for i in range(0, n, 2 * b):
            h[i:i + b] = 1.0
    elif typ == 2:
        b = rng.randint(2, max(3, n // 15))
        h = np.tile(0.5 * np.ones(2 * b) * np.array([1.0] * b + [0.0] * b), n // (2 * b) + 1)[:n]
        h += rng.randn(n) * 0.05
    elif typ == 3:
        f = rng.randint(1, max(2, n // 30))
        x = np.arange(n) / n * 2.0
        h = 0.5 + 0.3 * np.sin(2 * np.pi * f * x) + 0.2 * np.cos(4 * np.pi * f * x + rng.rand() * 3)
    else:
        h = np.full(n, 0.5)
        h += rng.randn(n) * 0.15
    return _project(np.clip(h, 0, 1), n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    # Get initial
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = _rand_init(400, rng, 0)
    n0 = len(h0)
    h0 = _project(h0, n0)
    
    bh = h0.copy()
    bc = _c5(bh)
    
    # Phase 1: Subgradient from initial
    h, c = _opt_subgrad(h0, n0, t0, budget_s, max_iter=4000)
    if c < bc:
        bh, bc = h, c
    
    # Phase 2: Random restarts
    remaining = budget_s - (time.time() - t0)
    n_restarts = int(min(8, max(3, remaining / 60)))
    for r in range(n_restarts):
        if budget_s - (time.time() - t0) < 40:
            break
        h_r = _rand_init(n0, rng, r % 5)
        h, c = _opt_subgrad(h_r, n0, t0, budget_s, max_iter=2000)
        if c < bc:
            bh, bc = h, c
    
    # Phase 3: Resolution changes
    for nn in [250, 350, 500, 600, 750, 900]:
        if budget_s - (time.time() - t0) < 50:
            break
        n_cur = len(bh)
        x_old = np.arange(n_cur) * 2.0 / n_cur
        x_new = np.arange(nn) * 2.0 / nn
        h_new = np.interp(x_new, x_old, bh)
        h_new = _project(np.clip(h_new, 0, 1), nn)
        h, c = _opt_subgrad(h_new, nn, t0, budget_s, max_iter=3000)
        if c < bc:
            bh, bc = h, c
    
    # Phase 4: Final polish
    if budget_s - (time.time() - t0) > 20:
        nb = len(bh)
        h, c = _opt_subgrad(bh, nb, t0, budget_s, max_iter=5000, lr0=1.0 / np.sqrt(nb))
        if c < bc:
            bh, bc = h, c
    
    bh = _project(bh, len(bh))
    bc = _c5(bh)
    return bh, bc, len(bh)

Alternative: ignore the sum constraint during the coordinate update, and project at the end of each sweep. This is a common trick.

Or: when we increase h[i] by δ, we decrease h[j] by δ for some other j. This is a "transfer" operation.

Let me simplify: do coordinate descent without the sum constraint (just box constraint), and project to the sum constraint periodically.

For the box-constrained coordinate descent:
- f(δ) = max_k (c_k + δ * g_k) for δ ∈ [-(h[i]), 1-h[i]]
- This is a convex piecewise linear function
- The minimum is at a point where the subgradient changes sign, or at a breakpoint

To find the optimum:
- The subgradient of f at δ is g_k* where k* = argmax_k (c_k + δ*g_k)
- f is decreasing where the subgradient is negative and increasing where positive
- At optimum: 0 ∈ ∂f(δ*), i.e., there exists k* with g_{k*} * sign condition...

Actually, for a max of linear functions, the minimum is at the intersection point where two (or more) lines cross at the minimum. We can find this by:
1. Sort the lines by their slope g_k
2. The lower envelope changes at breakpoints
3. The minimum of the upper envelope is where the slope changes from negative to positive

But this is O(n log n) per coordinate, making the total O(n² log n) per sweep. Might be too slow for many sweeps.

Simpler approach: use a ternary search or golden section search on δ. Since f(δ) is convex, we can find the minimum in O(log(1/ε)) evaluations, each O(n). Total per coordinate: O(n * log(1/ε)). Total per sweep: O(n² * log(1/ε)).

import numpy as np
import time

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def _corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _proj(h, n):
    tgt = 0.5 * n
    h = np.clip(h, 0.0, 1.0)
    for _ in range(80):
        d = tgt - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h < 1.0) if d > 0 else (h > 0.0)
        k = int(m.sum())
        if k == 0:
            break
        h[m] += d / k
        h = np.clip(h, 0.0, 1.0)
    return h

def _sg(h, idx, n):
    """Subgradient of corr[idx] w.r.t. h"""
    lo = max(0, idx - n + 1)
    hi = min(n - 1, idx)
    if lo > hi:
        return None
    idxs = np.arange(lo, hi + 1)
    return idxs, 1.0 - 2.0 * h[idx - idxs]

def _optimize(h0, n, t0, budget, max_it=5000):
    h = h0.copy()
    bh, bc = h.copy(), _c5(h)
    c = _corr(h)
    lr = 1.5 / np.sqrt(n)
    K = 5  # top-k lags
    for it in range(max_it):
        if budget - (time.time() - t0) < 2:
            break
        cur = c.max()
        if cur < bc:
            bc, bh = cur, h.copy()
        # Top-k weighted subgradient
        top = np.argsort(c)[-K:]
        w = c[top]
        w = w - w.min() + 0.5
        w /= w.sum()
        g = np.zeros(n)
        for ii in range(K):
            r = _sg(h, int(top[ii]), n)
            if r is not None:
                idxs, vals = r
                g[idxs] += w[ii] * vals
        h_new = _proj(h - lr * g, n)
        c_new = _corr(h_new)
        h = h_new
        c = c_new
        lr *= 0.9998
        lr = max(lr, 0.005 / n)
    return bh, bc

def _coord_descent(h0, n, t0, budget, n_sweeps=200):
    h = h0.copy()
    bh, bc = h.copy(), _c5(h)
    c = _corr(h)
    rng = np.random.RandomState(777)
    order = np.arange(n)
    for sweep in range(n_sweeps):
        if budget - (time.time() - t0) < 2:
            break
        rng.shuffle(order)
        for i in order:
            # Find optimal delta for h[i]
            lo_k = max(0, i)  # k such that i in S_k: k-n+1 <= i <= k
            hi_k = min(2 * n - 2, i + n - 1)
            # g_k = 1 - 2*h[k-i] for k in [lo_k, hi_k]
            ks = np.arange(lo_k, hi_k + 1)
            gs = 1.0 - 2.0 * h[ks - i]
            # f(delta) = max(c[k] + delta*g[k]) for k in [lo_k, hi_k], and max(c) for k outside
            c_max_outside = np.max(np.concatenate([c[:lo_k], c[hi_k+1:]])) if lo_k > 0 or hi_k < 2*n-2 else -np.inf
            # Binary search for optimal delta in [-(h[i]), 1-h[i]]
            dlo, dhi = -h[i], 1.0 - h[i]
            for _ in range(30):
                dm = 0.5 * (dlo + dhi)
                # Evaluate f at dlo, dm, dhi
                f_lo = max(c_max_outside, np.max(c[lo_k:hi_k+1] + dlo*gs)) if dlo != 0 else c.max()
                f_mid = max(c_max_outside, np.max(c[lo_k:hi_k+1] + dm*gs))
                f_hi = max(c_max_outside, np.max(c[lo_k:hi_k+1] + dhi*gs)) if dhi != 0 else c.max()
                if f_mid <= f_lo and f_mid <= f_hi:
                    break
                elif f_lo < f_mid:
                    dhi = dm
                else:
                    dlo = dm
            delta = 0.5 * (dlo + dhi)
            h[i] += delta
            # Update c incrementally
            c[lo_k:hi_k+1] += delta * gs
        h = _proj(h, n)
        c = _corr(h)
        cur = _c5(h)
        if cur < bc:
            bc, bh = cur, h.copy()
    return bh, bc

def _rand_h(n, rng, typ):
    if typ == 0:
        h = np.zeros(n)
        h[rng.choice(n, n//2, replace=False)] = 1.0
    elif typ == 1:
        b = rng.randint(3, max(4, n//10))
        h = np.zeros(n)
        h[0:n:n+b:(-1)]  # can't do this...
        for i in range(0, n, 2*b):
            h[i:i+b] = 1.0
    elif typ == 2:
        b = rng.randint(3, max(4, n//12))
        h = np.zeros(n)
        for i in range(0, n, b):
            if (i // b) % 2 == 0:
                h[i:i+b] = 1.0
        h += rng.randn(n) * 0.05
    else:
        f = rng.randint(1, max(2, n//40))
        x = np.arange(n) * 2.0 / n
        h = 0.5 + 0.25 * np.sin(2*np.pi*f*x + rng.rand()*6) + 0.15*np.sin(4*np.pi*f*x)
    return _proj(np.clip(h, 0, 1), n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = _rand_h(400, rng, 0)
    n0 = len(h0)
    h0 = _proj(h0, n0)
    
    bh, bc = h0.copy(), _c5(h0)
    
    # Phase 1: Subgradient from initial
    h, c = _optimize(h0, n0, t0, budget_s, 4000)
    if c < bc: bh, bc = h, c
    
    # Phase 2: Coordinate descent polish
    h, c = _coord_descent(bh, n0, t0, budget_s, 100)
    if c < bc: bh, bc = h, c
    
    # Phase 3: Random restarts
    for r in range(6):
        if budget_s - (time.time() - t0) < 50:
            break
        hr = _rand_h(n0, rng, r % 4)
        h, c = _optimize(hr, n0, t0, budget_s, 2000)
        if c < bc: bh, bc = h, c
        h, c = _coord_descent(bh, n0, t0, budget_s, 50)
        if c < bc: bh, bc = h, c
    
    # Phase 4: Try other resolutions
    for nn in [300, 450, 550, 700]:
        if budget_s - (time.time() - t0) < 50:
            break
        nc = len(bh)
        xo = np.arange(nc) * 2.0 / nc
        xn = np.arange(nn) * 2.0 / nn
        hn = _proj(np.clip(np.interp(xn, xo, bh), 0, 1), nn)
        h, c = _optimize(hn, nn, t0, budget_s, 2500)
        if c < bc:
            bh, bc = h, c
            h, c = _coord_descent(bh, nn, t0, budget_s, 80)
            if c < bc: bh, bc = h, c
    
    # Phase 5: Final polish
    if budget_s - (time.time() - t0) > 15:
        nb = len(bh)
        h, c = _coord_descent(bh, nb, t0, budget_s, 150)
        if c < bc: bh, bc = h, c
    
    bh = _proj(bh, len(bh))
    bc = _c5(bh)
    return bh, bc, len(bh)

import numpy as np
import time

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / len(h))

def _proj(h, n):
    tgt = 0.5 * n
    h = np.clip(h, 0.0, 1.0)
    for _ in range(60):
        d = tgt - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h < 1.0) if d > 0 else (h > 0.0)
        k = int(m.sum())
        if k == 0:
            break
        h[m] += d / k
        np.clip(h, 0.0, 1.0, out=h)
    return h

def _sg_comp(h, lag, n):
    """Return (indices, values) for subgradient of corr[lag]"""
    lo = max(0, lag - n + 1)
    hi = min(n - 1, lag)
    if lo > hi:
        return None
    idxs = np.arange(lo, hi + 1)
    return idxs, 1.0 - 2.0 * h[lag - idxs]

def _subgrad_opt(h0, t0, budget, max_it=3000, K=5):
    n = len(h0)
    h = h0.copy()
    c = _corr(h)
    bh, bc = h.copy(), float(c.max())
    lr = 1.5 / np.sqrt(n)
    for it in range(max_it):
        if budget - (time.time() - t0) < 1.5:
            break
        top = np.argsort(c)[-K:]
        w = c[top]
        w = np.maximum(w - w[0] + 0.3, 0.1)
        w /= w.sum()
        g = np.zeros(n)
        for ii in range(K):
            r = _sg_comp(h, int(top[ii]), n)
            if r is not None:
                ix, v = r
                g[ix] += w[ii] * v
        h = _proj(h - lr * g, n)
        c = _corr(h)
        m = float(c.max())
        if m < bc:
            bc, bh = m, h.copy()
        lr = max(lr * 0.9997, 0.003 / n)
    return bh, bc

def _local_search(h0, t0, budget, n_iter=5000):
    n = len(h0)
    h = h0.copy()
    bh, bc = h.copy(), _c5(h)
    rng = np.random.RandomState(999)
    step = 0.1
    for it in range(n_iter):
        if budget - (time.time() - t0) < 1:
            break
        i = rng.randint(n)
        d = rng.uniform(-step, step)
        h[i] = np.clip(h[i] + d, 0.0, 1.0)
        h = _proj(h, n)
        new_c = _c5(h)
        if new_c < bc:
            bc, bh = new_c, h.copy()
        else:
            h[i] = np.clip(bh[i], 0.0, 1.0)
            # Recompute from bh
            h = _proj(bh.copy(), n)
        # Adaptive step
        if it % 200 == 199:
            step *= 0.95
    return bh, bc
