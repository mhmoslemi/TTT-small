from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return compute_corr(h).max() * (2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def golden_section_pair(h, i, j, n, corr):
    """Golden section search for optimal t in h[i]+t, h[j]-t"""
    t_min = max(-h[i], h[j] - 1.0)
    t_max = min(1.0 - h[i], h[j])
    if t_max - t_min < 1e-14:
        return h, corr
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0  # golden ratio
    
    h_trial = h.copy()
    a = t_min
    b = t_max
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    
    h_trial[i] += c
    h_trial[j] -= c
    fc = compute_corr(h_trial).max()
    
    h_trial = h.copy()
    h_trial[i] += d
    h_trial[j] -= d
    fd = compute_corr(h_trial).max()
    
    for _ in range(60):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            h_trial = h.copy()
            h_trial[i] += c
            h_trial[j] -= c
            fc = compute_corr(h_trial).max()
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            h_trial = h.copy()
            h_trial[i] += d
            h_trial[j] -= d
            fd = compute_corr(h_trial).max()
    
    t_opt = (a + b) / 2.0
    h[i] += t_opt
    h[j] -= t_opt
    corr = compute_corr(h)
    return h, corr
