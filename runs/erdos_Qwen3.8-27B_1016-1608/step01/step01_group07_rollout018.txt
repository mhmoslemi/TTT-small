                best[i] = bval
                best = _fix(best, n)
                bc = _c5(best)
                imp = True
        if not imp:
            break
    return best, bc

def _interp(h, nn):
    n0 = len(h)
    xo = np.arange(n0) * (2.0 / n0)
    xn = np.arange(nn) * (2.0 / nn)
    return _fix(np.clip(np.interp(xn, xo, h), 0.0, 1.0), nn)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        bh = np.array(initial_h_values, dtype=float).copy()
    else:
        bh = np.full(45, 0.5)
    nb = len(bh)
    bh = _fix(bh, nb)
    bc = _c5(bh)

    # Phase 1: aggressive n=45 optimization
    n1 = 45
    h1 = _interp(bh, n1)
    c1 = _c5(h1)

    for alpha in [20, 60, 150, 400, 1200, 3000]:
        if budget_s - (time.time() - t0) < 12:
            break
        ht = _slsqp(h1.copy(), n1, alpha, 2000, t0, budget_s)
        if ht is not None:
            ct = _c5(ht)
            if ct < c1:
                h1 = ht
                c1 = ct
        if budget_s - (time.time() - t0) < 12:
            break
        h0 = _fix(np.clip(h1 + rng.randn(n1) * 0.06, 0, 1), n1)
        ht = _slsqp(h0, n1, alpha, 800, t0, budget_s)
        if ht is not None:
            ct = _c5(ht)
            if ct < c1:
                h1 = ht
                c1 = ct

    h1, c1 = _coord_desc(h1, n1, t0, budget_s, rounds=20, ngrid=19)

    # Alternative structures at n=45
    for trial in range(4):
        if budget_s - (time.time() - t0) < 18:
            break
        if trial == 0:
            h_alt = _fix(0.5 + 0.35 * np.sin(2 * np.pi * np.arange(n1) / 7.0), n1)
        elif trial == 1:
            h_alt = _fix(0.5 + 0.32 * np.sin(2 * np.pi * np.arange(n1) / 9.0), n1)
        elif trial == 2:
            h_alt = _fix(0.5 + 0.3 * np.sign(np.sin(2 * np.pi * np.arange(n1) / 5.0) + 0.3), n1)
        else:
            h_alt = _fix(np.clip(0.5 + 0.25 * rng.randn(n1), 0, 1), n1)
        c_alt = _c5(h_alt)
        if c_alt < c1 + 0.06:
            for alpha in [100, 400, 1000]:
                if budget_s - (time.time() - t0) < 10:
                    break
                ht = _slsqp(h_alt.copy(), n1, alpha, 1000, t0, budget_s)
                if ht is not None:
                    ct = _c5(ht)
                    if ct < c_alt:
                        h_alt = ht
                        c_alt = ct
            h_alt, c_alt = _coord_desc(h_alt, n1, t0, budget_s, rounds=12, ngrid=15)
            if c_alt < c1:
                h1 = h_alt
                c1 = c_alt

    # Transfer best low-res to full res
    h1f = _interp(h1, nb)
    c1f = _c5(h1f)
    if c1f < bc:
        bh = h1f
        bc = c1f

    # Phase 2: n=80 refinement
    if budget_s - (time.time() - t0) > 50:
        n2 = 80
        h2 = _interp(bh, n2)
        c2 = _c5(h2)
        if c2 < bc:
            bh = h2.copy()
            bc = c2
        for alpha in [50, 200, 600]:
            if budget_s - (time.time() - t0) < 12:
                break
            ht = _slsqp(bh.copy(), n2, alpha, 600, t0, budget_s)
            if ht is not None:
                ct = _c5(ht)
                if ct < bc:
                    bh = ht
                    bc = ct
        bh, bc = _coord_desc(bh, n2, t0, budget_s, rounds=10, ngrid=13)

    # Phase 3: n=120
    if budget_s - (time.time() - t0) > 50:
        n3 = 120
        h3 = _interp(bh, n3)
        c3 = _c5(h3)
        if c3 < bc:
            bh = h3.copy()
            bc = c3
        for alpha in [80, 300]:
            if budget_s - (time.time() - t0) < 12:
                break
            ht = _slsqp(bh.copy(), n3, alpha, 400, t0, budget_s)
            if ht is not None:
                ct = _c5(ht)
                if ct < bc:
                    bh = ht
                    bc = ct
        bh, bc = _coord_desc(bh, n3, t0, budget_s, rounds=8, ngrid=11)

    # Phase 4: full resolution
    bh = _interp(bh, nb)
    bc = _c5(bh)
    for alpha in [100, 400, 1200]:
        if budget_s - (time.time() - t0) < 25:
            break
        ht = _slsqp(bh.copy(), nb, alpha, 250, t0, budget_s)
        if ht is not None:
            ct = _c5(ht)
            if ct < bc:
                bh = ht
                bc = ct

    # Phase 5: n=200 branch (different path)
    if budget_s - (time.time() - t0) > 45:
        n4 = 200
        h4 = _interp(h1, n4)
        c4 = _c5(h4)
        for alpha in [100, 400, 1000]:
            if budget_s - (time.time() - t0) < 15:
                break
            ht = _slsqp(h4.copy(), n4, alpha, 350, t0, budget_s)
            if ht is not None:

                if ct < c4:

                    c4 = ct
        h4, c4 = _coord_desc(h4, n4, t0, budget_s, rounds=6, ngrid=9)
        h4f = _interp(h4, nb)
        if _c5(h4f) < bc:
            bh = h4f
            bc = _c5(bh)

    # Final polish at n=400 with very sharp alpha
    if budget_s - (time.time() - t0) > 20:
        bh = _fix(bh, nb)
        ht = _slsqp(bh.copy(), nb, 3000, 200, t0, budget_s)
        if ht is not None:
            ct = _c5(ht)
            if ct < bc:
                bh = ht
                bc = ct

    bh = _fix(bh, len(bh))
    bc = _c5(bh)
    return bh, bc, len(bh)
