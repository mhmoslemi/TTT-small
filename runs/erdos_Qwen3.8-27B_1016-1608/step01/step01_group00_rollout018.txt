import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not np.any(mask):
            h[:] = target / n
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_and_grad(beta, h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    exp_c = np.exp(beta * (corr - max_c))
    s = np.sum(exp_c)
    w = exp_c / s
    val = (max_c + np.log(s)) / beta * dx
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        klag = m - (n - 1)
        wm = w[m]
        if wm < 1e-18:
            continue
        j1 = max(0, klag)
        j2 = min(n, n + klag)
        if j1 < j2:
            grad[j1:j2] += wm * (1.0 - h[j1 - klag:j2 - klag])
        j3 = max(0, -klag)
        j4 = min(n, n - klag)
        if j3 < j4:
            grad[j3:j4] -= wm * h[j3 + klag:j4 + klag]
    grad *= dx
    return val, grad

def slsqp_smooth(h0, n, beta, maxiter=400):
    def obj(x):
        return lse_and_grad(beta, x)[0]
    def grad(x):
        return lse_and_grad(beta, x)[1]
    def con_fun(x):
        return np.sum(x) - n / 2.0
    def con_jac(x):
        return np.ones(n)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h0, method='SLSQP', jac=grad, bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        if res.fun < obj(h0) + 1e-12:
            return repair(np.clip(res.x, 0.0, 1.0), n)
    except Exception:
        pass
    return h0

def thue_morse_start(n):
    tm = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1')
        tm[i] = float(c % 2)
    ones = np.sum(tm > 0.5)
    target = n // 2
    if ones > target:
        idx = np.where(tm > 0.5)[0]
        np.random.shuffle(idx)
        tm[idx[:ones - target]] = 0.0
    elif ones < target:
        idx = np.where(tm < 0.5)[0]
        np.random.shuffle(idx)
        tm[idx[:target - ones]] = 1.0
    return repair(tm, n)

def coordinate_polish(h, n, max_rounds=100):
    h = h.copy()
    c5 = compute_c5(h)
    for _ in range(max_rounds):
        improved = False
        for j in range(n):
            for d in [0.02, -0.02, 0.01, -0.01, 0.005, -0.005, 0.002, -0.002]:

                ct = compute_c5(ht)
                if ct < c5 - 1e-13:
                    h, c5 = ht, ct
                    improved = True
                    break
        if not improved:

    return h, c5

            h_p = h_best.copy()
            h_p[start:start+blk] = np.clip(h_p[start:start+blk] + scale * np.random.randn(blk), 0, 1)
        else:
            # Fourier perturbation
            h_p = h_best.copy()
            freq = np.random.randint(1, n // 4)
            phase = np.random.uniform(0, 2 * np.pi)
            h_p += scale * np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
        h_p = repair(h_p, n)
        # Quick smooth optimization from perturbed start
        beta_use = 100 if c5_best > 0.381 else 300
        h = slsqp_smooth(h_p, n, beta_use, maxiter=200)
        c = compute_c5(h)
        if c < c5_best:
            h_best, c5_best = h, c
        attempt += 1

    # Phase 3: Coordinate descent polish
    if time.time() - t0 < budget_s * 0.90:
        h_pol, c_pol = coordinate_polish(h_best, n, max_rounds=80)
        if c_pol < c5_best:
            h_best, c5_best = h_pol, c_pol

    # Phase 4: Final SLSQP with high-beta smoothing
    if time.time() - t0 < budget_s * 0.95:
        h = slsqp_smooth(h_best, n, 5000, maxiter=1000)
        c = compute_c5(h)
        if c < c5_best:
            h_best, c5_best = h, c
        # Another pass with exact-ish SLSQP
        def exact_obj(x):
            return compute_c5(x)
        bnds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}]
        try:
            res = minimize(exact_obj, h_best, method='SLSQP', bounds=bnds,
                           constraints=cons, options={'maxiter': 1000, 'ftol': 1e-16})
            h_r = repair(np.clip(res.x, 0.0, 1.0), n)
            c_r = compute_c5(h_r)
            if c_r < c5_best:
                h_best, c5_best = h_r, c_r
        except Exception:
            pass

    # Phase 5: Try interpolation to slightly different n if time allows
    if time.time() - t0 < budget_s * 0.99:
        for n2 in [n + 10, n - 10, n + 20]:
            if n2 < 20 or time.time() - t0 > budget_s * 0.985:
                continue
            xp = np.linspace(0.0, 2.0, n)
            xn = np.linspace(0.0, 2.0, n2)
            h_interp = np.interp(xn, xp, h_best)
            h_interp = repair(h_interp, n2)
            h2 = slsqp_smooth(h_interp, n2, 200, maxiter=300)
            c2 = compute_c5(h2)
            if c2 < c5_best:
                h_best, c5_best, n = h2, c2, n2

    # Final validation
    h_best = repair(h_best, n)
    c5_best = compute_c5(h_best)
    return (h_best, c5_best, n)
