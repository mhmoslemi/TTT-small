        try:
            r4 = minimize(exact_obj, best_h, args=(dx,), method='SLSQP',
                        bounds=bnds, constraints=cons,
                        options={'maxiter': 150, 'ftol': 1e-15})
            hf = repair(r4.x, tgt)
            cf = compute_c5(hf, dx)
            if cf < best_c5:
                best_c5, best_h = cf, hf
        except Exception:
            break

    # Final validation
    best_h = repair(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
