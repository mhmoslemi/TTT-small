    m = np.max(c5_vals)
    return m + np.log(np.sum(np.exp(alpha * (c5_vals - m)))) / alpha

import numpy as np
from scipy.optimize import minimize
import time

_budget = 1100.0
_start_t = 0.0

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_c5_all(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_to_set(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        hp = np.clip(h - mid, 0.0, 1.0)
        if float(np.sum(hp)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def smooth_max_and_grad(h, n, alpha):
    c5_vals = compute_c5_all(h, n)
    m = np.max(c5_vals)
    shifted = alpha * (c5_vals - m)
    w = np.exp(shifted)
    S = np.sum(w)
    w = w / S
    f = m + np.log(S) / alpha
    dx = 2.0 / n
    grad = np.zeros(n)
    for mi in range(len(c5_vals)):
        if w[mi] < 1e-8:
            continue
        k = mi - (n - 1)
        wm = w[mi] * dx
        if k == 0:
            grad[:] += wm * (1.0 - 2.0 * h)
        elif k > 0:
            grad[:n-k] += wm * (1.0 - h[k:])
            grad[k:] -= wm * h[:n-k]
        else:
            kk = -k
            grad[kk:] += wm * (1.0 - h[:n-kk])
            grad[:n-kk] -= wm * h[kk:]
    return f, grad

def pgd_nesterov(h0, n, n_iter, time_limit, alpha_start=20.0, lr_start=0.5, momentum=0.85):
    target = n / 2.0
    h = project_to_set(h0.astype(float).copy(), target)
    v = np.zeros(n)
    lr = lr_start
    alpha = alpha_start
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    t0 = time.time()
    
    for it in range(n_iter):
        if time.time() - t0 > time_limit:
            break
        f, grad = smooth_max_and_grad(h, n, alpha)
        v = momentum * v + grad
        accepted = False
        for back in range(15):
            h_new = project_to_set(h - lr * v, target)
            c5_new = compute_c5(h_new, n)
            if c5_new < best_c5 + 1e-10:
                h = h_new
                if c5_new < best_c5 - 1e-12:
                    best_c5 = c5_new
                    best_h = h.copy()
                accepted = True
                break
            lr *= 0.7
        if not accepted:
            v *= 0.5
        if it % 100 == 0:
            alpha = min(alpha * 1.2, 300.0)
    return best_h, best_c5

def upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, h)

def coord_desc(h, n, rng, n_sweeps=30, time_limit=60.0):
    target = n / 2.0
    h = project_to_set(h.astype(float).copy(), target)
    best = compute_c5(h, n)
    t0 = time.time()
    for sw in range(n_sweeps):
        if time.time() - t0 > time_limit:
            break
        improved = False
        order = rng.permutation(n)
        for j in order:
            if time.time() - t0 > time_limit:
                break
            old = h[j]
            # Try several candidates
            cands = []
            cands.append(0.0)
            cands.append(1.0)
            cands.append(max(0.0, min(1.0, old - 0.15)))
            cands.append(max(0.0, min(1.0, old + 0.15)))
            cands.append(max(0.0, min(1.0, old - 0.05)))
            cands.append(max(0.0, min(1.0, old + 0.05)))
            bc, bv = old, best
            for c in cands:
                if abs(c - old) < 1e-9:
                    continue
                h[j] = c
                h = project_to_set(h, target)
                v = compute_c5(h, n)
                h[j] = old  # restore
                h = project_to_set(h, target)
                if v < bv - 1e-12:
                    bv = v
                    bc = c
            if bc != old:
                h[j] = bc
                h = project_to_set(h, target)
                best = bv
                improved = True
        if not improved:
            break
    return h, best

def time_left():
    return max(0.0, _budget - (time.time() - _start_t))

def make_random_binary(n, rng):
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, size=n//2, replace=False)] = 1.0

def make_structured(n, rng, mode=0):
    x = np.linspace(0.0, 2.0, n)
    if mode == 0:
        # Block with soft edges
        pos = rng.uniform(0.3, 1.7)
        width = rng.uniform(0.3, 0.8)
        h = 0.5 / (1.0 + np.exp(-(x-pos)/width)) + 0.5 / (1.0 + np.exp((x-pos)/width))
    elif mode == 1:
        # Two blocks
        p1, p2 = sorted(rng.uniform(0.2, 1.8, 2))

        h = 1.0
        h -= 1.0 / (1.0 + np.exp(-(x-p1)/w1)) * 0.0  # just sigmoid barriers

    else:

        phase = rng.uniform(0, 2*np.pi)
        h = 0.5 + 0.45*np.sin(freq*np.pi*x + phase)
    return project_to_set(h, n/2.0)

import numpy as np
from scipy.optimize import minimize
import time

_budget = 1100.0
_start_t = 0.0
_con_target = 0.0
_best_h = None
_best_c5 = np.inf
_best_n = 0

def _integ_fn(x):
    return float(np.sum(x) - _con_target)

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_c5_all(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_to_set(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        hp = np.clip(h - mid, 0.0, 1.0)
        if float(np.sum(hp)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def smooth_max_grad(h, n, alpha):
    c5v = compute_c5_all(h, n)
    m = float(np.max(c5v))
    sh = alpha * (c5v - m)
    w = np.exp(sh)
    S = float(np.sum(w))
    w = w / S
    f = m + np.log(S) / alpha
    dx = 2.0 / n
    grad = np.zeros(n)
    for mi in range(len(c5v)):
        if w[mi] < 1e-9:
            continue
        k = mi - (n - 1)
        wm = w[mi] * dx
        if k == 0:
            grad[:] += wm * (1.0 - 2.0 * h)
        elif k > 0:
            grad[:n-k] += wm * (1.0 - h[k:])
            grad[k:] -= wm * h[:n-k]
        else:
            kk = -k
            grad[kk:] += wm * (1.0 - h[:n-kk])
            grad[:n-kk] -= wm * h[kk:]
    return f, grad

def pgd_nesterov(h0, n, n_iter, time_limit, alpha0=25.0, lr0=0.4, mom=0.85):
    global _con_target
    target = n / 2.0
    _con_target = target
    h = project_to_set(h0.astype(float).copy(), target)
    v = np.zeros(n)
    lr = lr0
    alpha = alpha0
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    t0 = time.time()
    for it in range(n_iter):
        if time.time() - t0 > time_limit:
            break
        f, grad = smooth_max_grad(h, n, alpha)
        v = mom * v + grad
        acc = False
        for bk in range(12):
            h_new = project_to_set(h - lr * v, target)
            c5_new = compute_c5(h_new, n)
            if c5_new < best_c5 + 1e-11:
                h = h_new
                if c5_new < best_c5 - 1e-13:
                    best_c5 = c5_new
                    best_h = h.copy()
                acc = True
                break
            lr *= 0.65
        if not acc:
            v *= 0.4
            h = project_to_set(h - lr * v, target)
        if it % 80 == 0 and it > 0:
            alpha = min(alpha * 1.18, 350.0)
    return best_h, best_c5

def upsample(h, n_new):
    n_old = len(h)
    xo = np.linspace(0.0, 2.0, n_old)
    xn = np.linspace(0.0, 2.0, n_new)
    return np.interp(xn, xo, h)

def coord_desc(h, n, rng, n_sweeps, time_limit):
    global _con_target
    target = n / 2.0
    _con_target = target
    h = project_to_set(h.astype(float).copy(), target)
    best = compute_c5(h, n)
    t0 = time.time()
    for _ in range(n_sweeps):
        if time.time() - t0 > time_limit:
            break
        imp = False
        order = rng.permutation(n)
        for j in order:
            if time.time() - t0 > time_limit:
                break
            old = h[j]
            bv = best
            bc = old
            for c in [0.0, 1.0, max(0.0, old-0.12), min(1.0, old+0.12),
                      max(0.0, old-0.04), min(1.0, old+0.04),
                      max(0.0, old-0.01), min(1.0, old+0.01)]:
                if abs(c - old) < 1e-9:
                    continue
                h[j] = c
                h = project_to_set(h, target)
                v = compute_c5(h, n)
                if v < bv - 1e-12:
                    bv = v
                    bc = c
            if bc != old:
                h[j] = bc
                h = project_to_set(h, target)
                best = bv
                imp = True
            else:
                h[j] = old
                h = project_to_set(h, target)
        if not imp:
            break
    return h, best

def slsqp_opt(h0, n, maxiter):
    global _con_target
    target = n / 2.0
    _con_target = target
    h0 = project_to_set(h0.astype(float).copy(), target)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _integ_fn}]
    try:
        res = minimize(lambda x: compute_c5(x, n), h0, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-11})
        h_out = project_to_set(res.x.copy(), target)
        return h_out, compute_c5(h_out, n)
    except Exception:
        return h0, compute_c5(h0, n)

def make_random_binary(n, rng):
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, size=n//2, replace=False)] = 1.0
    return h

def make_structured(n, rng, mode):
    x = np.linspace(0.0, 2.0, n)

        wd = rng.uniform(0.25, 0.6)

    elif mode == 1:

           + 1.0/(1.0+np.exp(-(x-p2-w2)/0.04)) - 1.0/(1.0+np.exp(-(x-p2+w2)/0.04)))
    else:

        h = 0.5 + 0.42*np.sin(freq*np.pi*x + ph) + 0.08*np.sin(2*freq*np.pi*x + ph*0.7)
    return project_to_set(h, n/2.0)
