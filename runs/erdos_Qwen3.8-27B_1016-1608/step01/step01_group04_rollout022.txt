    """Project h onto {x: sum(x)=target_sum, 0<=x[i]<=1}"""
    lo, hi = -max(h), 1 - min(h)
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target_sum:
            lo = mid
        else:
            hi = mid

        h_new = project_h(h + momentum, target)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
            lr = min(lr * 1.05, 0.08)
        else:
            lr *= 0.9
            momentum *= 0.5
        h = h_new
    return best_h, best_c5

def discrete_local_search(h, n, t0, deadline, max_rounds=50):
    h = project_h(h.copy(), n / 2.0)
    target = n / 2.0
    best_c5 = compute_c5(h)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        one_idx = np.where(h == np.round(h))[0]
        for j in range(n):
            if time.time() > deadline:
                break
            trial = h.copy()
            if trial[j] < 0.5:
                trial[j] = 1.0
            else:
                trial[j] = 0.0
            trial = project_h(trial, target)
            c5 = compute_c5(trial)
            if c5 < best_c5 - 1e-10:
                h = trial
                best_c5 = c5
                improved = True
        if not improved:
            break
    return h, best_c5

def upsample(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_h(h_new, n_new / 2.0)

def slsqp_smooth(h0, n, eps, maxiter, t0, deadline):
    state = {'n': n, 'ep': eps, 'dl': deadline}
    def obj(x):
        if time.time() > state['dl']:
            return 999.0
        v, _ = smooth_max_val_grad(x, state['ep'])
        return v
    def eqf(x):
        return float(np.sum(x) - state['n'] / 2.0)
    def eqj(x):
        return np.ones(state['n'])
    bds = [(0.0, 1.0)] * n
    cs = [{'type': 'eq', 'fun': eqf, 'jac': eqj}]
    r = minimize(obj, h0, method='SLSQP', bounds=bds, constraints=cs,
                 options={'maxiter': maxiter, 'ftol': 1e-14})
    return project_h(r.x, n / 2.0)
