        else:

        # Find the lag that achieves the max
        k_max = int(np.argmax(corr))

        # corr[k_max] = sum_i h[i] * (1-h[i+k_max]) for valid i
        # Actually need to map k_max to actual lag

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corrs(h, n):
    return np.correlate(h, 1.0 - h, mode="full")

def get_gradient_for_lag(h, n, lag):
    """Compute gradient of correlation at given lag."""
    n = len(h)
    grad = np.zeros(n)
    if lag >= 0:
        k = lag
        if k < n:
            grad[k:] += 1.0 - h[:n-k]
            grad[:n-k] -= h[k:]
    else:
        k = -lag
        if k < n:
            grad[:n-k] += 1.0 - h[k:]
            grad[k:] -= h[:n-k]
    return grad

def project_to_set(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        hp = np.clip(h - mid, 0.0, 1.0)
        if float(np.sum(hp)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def subgradient_optimize(h0, n, t_max, top_k=3):
    """Subgradient optimization with top-k smoothing."""
    target = n / 2.0
    dx = 2.0 / n
    h = project_to_set(h0.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    t_start = time.time()
    
    lr = 0.05
    it = 0
    while time.time() - t_start < t_max:
        corr = compute_corrs(h, n)
        # Find top-k lags
        if top_k > 1:
            top_idx = np.argsort(corr)[-top_k:]
            weights = np.exp(corr[top_idx] / 0.01)  # softmax-like
            weights /= weights.sum()
            grad = np.zeros(n)
            for wi, ki in zip(weights, top_idx):
                lag = ki - (n - 1)
                grad += wi * get_gradient_for_lag(h, n, lag)
        else:
            k_max = int(np.argmax(corr))
            lag = k_max - (n - 1)
            grad = get_gradient_for_lag(h, n, lag)
        
        h = h - lr * grad
        h = project_to_set(h, target)
        
        c5 = compute_c5(h, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Adapt learning rate
        lr = 0.05 / (1.0 + 0.005 * it)
        it += 1
    
    return best_h, best_c5

def coordinate_descent(h, n, t_max, rng):
    """Coordinate descent with integral projection."""
    target = n / 2.0
    h = project_to_set(h.copy(), target)
    best_c5 = compute_c5(h, n)
    t_start = time.time()
    
    n_sweeps = 0
    while time.time() - t_start < t_max:
        order = rng.permutation(n)
        improved = False
        for j in order:
            old = h[j]
            cands = np.array([0.0, 1.0, old - 0.15, old + 0.15, old - 0.05, old + 0.05])
            cands = np.clip(cands, 0.0, 1.0)
            bc, bv = old, best_c5
            for c in cands:
                if abs(c - old) < 1e-8:
                    continue
                h[j] = c
                h = project_to_set(h, target)
                v = compute_c5(h, n)
                if v < bv - 1e-10:
                    bv = v
                    bc = float(c)
            if bc != old:
                h[j] = bc
                h = project_to_set(h, target)
                best_c5 = bv
                improved = True
            else:
                h[j] = old
                h = project_to_set(h, target)
        n_sweeps += 1
        if not improved:
            break
    return h, best_c5

def upsample_h(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0.0, 2.0, n_old + 1)
    x_new = np.linspace(0.0, 2.0, n_new + 1)
    h_ext = np.concatenate([h, [h[0]]])  # periodic extension for interp
    return np.interp(x_new, x_old, h_ext)

def make_binary_restart(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, size=n//2, replace=False)
    h[idx] = 1.0
    h += rng.normal(0, 0.03, n)
    return project_to_set(h, n/2.0)

def make_smooth_restart(n, rng):
    x = np.linspace(0.0, 2.0, n)
    h = 0.5 + np.zeros(n)
    for _ in range(rng.randint(2, 5)):
        f = rng.randint(1, 6)
        p = rng.uniform(0, 2*np.pi)
        a = rng.uniform(0.15, 0.4)
        h += a * np.sin(f * np.pi * x + p)
    return project_to_set(h, n/2.0)

def slsqp_step(h, n, maxiter=200):
    target = n / 2.0
    h = project_to_set(h.copy(), target)
    
    def obj(x):
        return compute_c5(x, n)
    
    def con_fun(x):
        return float(np.sum(x) - target)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun}]
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        h_out = project_to_set(res.x.copy(), target)
        return h_out, compute_c5(h_out, n)
    except:
        return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()
    
    def tleft():
        return max(0.0, budget_s - (time.time() - t_start))
    
    best_h = None
    best_c5 = np.inf
    best_n = 0
    
    def update(h, n):
        nonlocal best_h, best_c5, best_n
        h = np.asarray(h, dtype=float)
        h = project_to_set(h, n / 2.0)
        c = compute_c5(h, n)
        if c < best_c5:
            best_h = h.copy()
            best_c5 = c
            best_n = n
    
    # Get initial values
    try:
        h_init = np.asarray(initial_h_values, dtype=float).copy()
    except:
        h_init = make_binary_restart(90, rng)
    n_base = len(h_init)
    update(h_init, n_base)
    
    # Phase 1: Subgradient at base resolution (~250s)
    if tleft() > 250:
        h_sg, c5 = subgradient_optimize(h_init, n_base, t_max=200, top_k=3)
        update(h_sg, n_base)
    
    # Phase 2: Coordinate descent (~200s)
    if tleft() > 200:
        h_cd, c5 = coordinate_descent(best_h, best_n, t_max=180, rng=rng)
        update(h_cd, best_n)
    
    # Phase 3: SLSQP polish (~100s)

                best_h, best_c5 = h, c
                improved_round = True
        hr = mk_binary(nb, rng)
        if tl() > 80:
            h, c = subgrad_h(hr, nb, min(30, tl() * 0.2), 5)
            if c < best_c5:
                best_h, best_c5 = h, c
                improved_round = True
        hr = mk_smooth(nb, rng)
        if tl() > 80:
            h, c = subgrad_h(hr, nb, min(25, tl() * 0.2), 4)
            if c < best_c5:
                best_h, best_c5 = h, c
                improved_round = True
        if not improved_round and tl() < BL * 0.55:
            break

    # Phase 5: Try n=60 (coarser can sometimes be better)
    if tl() > 100 and nb > 60:
        h60 = project(np.interp(np.linspace(0, 2, 60), np.linspace(0, 2, nb), best_h), 30.0)
        c60 = compute_c5(h60)
        if c60 < best_c5 - 0.0005:
            h, c = subgrad_h(h60, 60, min(40, tl() * 0.3), 3)
            if c < best_c5:
                best_h, best_c5, best_n = h, c, 60

    # Phase 6: Higher resolution
    if tl() > 110 and best_n < 360:
        n2 = min(2 * best_n, 360)
        hu = project(upsmppl(best_h, n2), n2 / 2.0)
        c_u = compute_c5(hu)
        h, c = subgrad_h(hu, n2, min(60, tl() * 0.35), 7)
        if c < best_c5:
            best_h, best_c5, best_n = h, c, n2

    # Phase 7: Binary nudge + final coord descent
    if tl() > 40:
        hn, cn = try_binary_nudge(best_h, best_n)
        if cn < best_c5:
            best_h, best_c5 = hn, cn
        h, c = coord_desc(best_h, best_n, min(25, tl() * 0.4), rng)
        if c < best_c5:
            best_h, best_c5 = h, c

    # Phase 8: Final SLSQP
    if tl() > 20:
        h, c = slsqp_opt(best_h, best_n, 120)
        if c < best_c5:
            best_h, best_c5 = h, c

    # Final validation
    best_h = project(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
