    def sc(h):
        return np.sum(h) - n / 2.0

    def sc_jac(h):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': sc, 'jac': sc_jac}]
    bnds = [(0.0, 1.0)] * n

    for temp in [0.01, 0.003, 0.001, 0.0004, 0.0001]:
        if time.time() > slsqp_deadline:
            break
        res = minimize(obj_fn, best_h.copy(), args=(temp,),
                      method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': 3000, 'ftol': 1e-15})
        h_new = project_feasible(res.x, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 5: Fine neighborhood search
    nb_deadline = t_start + budget_s * 0.99
    for scale in [0.02, 0.01, 0.005, 0.002]:
        for trial in range(200):
            if time.time() > nb_deadline:
                break
            h_trial = best_h + rng.normal(0, scale, n)
            h_trial = project_feasible(h_trial, n)
            c5_new = compute_c5(h_trial)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_trial.copy()

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n)
