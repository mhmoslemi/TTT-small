    dx = 2.0 / n
    # Compute all correlation values
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    # Find active set (top values)
    k_max = c.max()
    # Use top 10% as active set for stability
    thresh = k_max - 0.01 * k_max  # or use a fixed number
    active = np.where(c >= thresh)[0]
    
    # Compute subgradient
    g = np.zeros(n)
    for k in active:
        # k in [0, 2n-2], shift = k - (n-1)
        shift = k - (n - 1)
        if shift >= 0:
            # h[i] * (1-h[i+shift]) for i in [0, n-1-shift]
            valid = np.arange(n - shift)
            g[valid] += (1.0 - h[valid + shift]) * dx
