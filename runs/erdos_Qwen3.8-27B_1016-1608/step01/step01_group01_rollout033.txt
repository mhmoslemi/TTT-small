    return corr.max() * dx

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def sum_fun(h):
    return np.sum(h) - len(h) / 2.0

def sum_jac(h):
    return np.ones(len(h))

def get_subgrad(h, topk=3):
    corr = corr_full(h)
    n = len(h)
    top_idx = np.argsort(corr)[-topk:]
    m = np.arange(n)
    grad = np.zeros(n)
    for k_s in top_idx:
        shift = k_s - (n - 1)
        jp = m + shift
        jm = m - shift
        vp = (jp >= 0) & (jp < n)
        vm = (jm >= 0) & (jm < n)
        grad[vp] += (1.0 - h[jp[vp]])
        grad[vm] -= h[jm[vm]]
    return grad / topk

    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        diff = target - h.sum()

            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        nf = np.sum(free)
        if nf == 0:
            break
        h[free] += diff / nf
        h = np.clip(h, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def sum_fun(h):
    return np.sum(h) - len(h) / 2.0

def sum_jac(h):
    return np.ones(len(h))

def get_subgrad(h, topk=3):
    corr = corr_full(h)
    n = len(h)
    top_idx = np.argsort(corr)[-topk:]
    j = np.arange(n)
    grad = np.zeros(n)
    for k_s in top_idx:
        shift = k_s - (n - 1)
        v1 = (j - shift >= 0) & (j - shift < n)
        grad[v1] += (1.0 - h[j[v1] - shift])
        v2 = (j + shift >= 0) & (j + shift < n)
        grad[v2] -= h[j[v2] + shift]
    return grad / topk

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        nf = np.sum(free)
        if nf == 0:
            break
        h[free] += diff / nf
        h = np.clip(h, 0.0, 1.0)
    return h

def softmax_top(h, temp, topk=5):
    c = corr_full(h)
    m = c.max()
    sc = np.sort(c)[-topk:]
    return m + temp * np.log(np.sum(np.exp((sc - m) / temp)))

def top_lag_penalty(h, lags, w):
    c = corr_full(h)
    return c.max() + w * np.sum(c[lags])

def resample_h(h, new_n):
    old_n = len(h)
    old_x = np.linspace(0, 2, old_n, endpoint=False)
    new_x = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h)
    return project_feasible(h_new, new_n)
