import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def repair(h, target):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        d = target - np.sum(h)
        if abs(d) < 1e-15:
            break
        m = (h < 1.0 - 1e-14) if d > 0 else (h > 1e-14)
        c = int(np.count_nonzero(m))
        if c == 0:
            break
        h[m] += d / c
        np.clip(h, 0, 1, out=h)
    return h

def lse_max(corr, tau):
    m = np.max(corr)
    return m + tau * np.log(np.sum(np.exp((corr - m) / tau)))

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.ones(70) * 0.5

    n = len(h_init)
    tgt = n / 2.0
    dx = 2.0 / n
    h = repair(h_init, tgt)
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    best_n = n
    bnds = [(0.0, 1.0)] * n

    def obj_lse(x, tau):
        xc = np.clip(x, 0, 1)
        corr = np.correlate(xc, 1.0 - xc, mode="full")
        return lse_max(corr, tau) * dx + 5.0 * (np.sum(xc) - tgt) ** 2

    # Phase 1: Multi-temperature L-BFGS-B (main optimization)
    taus = [0.1, 0.04, 0.015, 0.006, 0.002, 0.0008, 0.0003, 0.0001]
    for tau in taus:
        if time.time() - t0 > budget_s - 250:
            break
        r = minimize(lambda x: obj_lse(x, tau), best_h, method='L-BFGS-B',
                    bounds=bnds, options={'maxiter': 1000, 'gtol': 1e-15, 'ftol': 1e-16})
        hn = repair(r.x, tgt)
        cn = compute_c5(hn, dx)
        if cn < best_c5:
            best_c5, best_h = cn, hn
        # Also try from complement and shift
        for alt in [np.clip(1.0 - hn, 0, 1), np.roll(hn, 1)]:
            if time.time() - t0 > budget_s - 200:
                break
            alt = repair(alt, tgt)
            r2 = minimize(lambda x: obj_lse(x, tau * 0.5), alt, method='L-BFGS-B',
                         bounds=bnds, options={'maxiter': 300, 'gtol': 1e-14})
            hn2 = repair(r2.x, tgt)
            cn2 = compute_c5(hn2, dx)
            if cn2 < best_c5:
                best_c5, best_h = cn2, hn2

    # Phase 2: Targeted coordinate descent (top lags focus)
    if time.time() - t0 < budget_s - 80:
        h_cd = best_h.copy()
        for sweep in range(200):
            if time.time() - t0 > budget_s - 40:
                break
            any_imp = False
            corr_now = np.correlate(h_cd, 1.0 - h_cd, mode="full")
            top_lags = np.argsort(-corr_now)[:5]
            for i in range(n):
                # Try a few candidate values around current
                cur = h_cd[i]
                cands = [0.0, 1.0, cur, cur - 0.1, cur + 0.1, cur - 0.03, cur + 0.03]
                cands = [c for c in cands if 0.0 <= c <= 1.0]
                best_cv = best_c5
                best_cv_i = cur
                for cv_val in cands:
                    if abs(cv_val - cur) < 1e-10:
                        continue
                    hv = h_cd.copy()
                    hv[i] = cv_val
                    hv = repair(hv, tgt)
                    c_val = compute_c5(hv, dx)
                    if c_val < best_cv:
                        best_cv = c_val
                        best_cv_i = cv_val
                if best_cv < best_c5:
                    h_cd[i] = best_cv_i
                    h_cd = repair(h_cd, tgt)
                    best_c5 = best_cv
                    best_h = h_cd.copy()
                    any_imp = True
            if not any_imp:
                break

    # Phase 3: SLSQP with exact max (tight polish)
    if time.time() - t0 < budget_s - 15:
        _t = tgt
        def sc(x):
            return np.sum(x) - _t
        cons = [{'type': 'eq', 'fun': sc}]
        def fobj(x):
            xc = np.clip(x, 0, 1)
            corr = np.correlate(xc, 1.0 - xc, mode="full")
            return float(np.max(corr)) * dx + 0.2 * (np.sum(xc) - tgt) ** 2
        for _ in range(15):
            if time.time() - t0 > budget_s - 5:
                break
            r = minimize(fobj, best_h, method='SLSQP', bounds=bnds,
                        constraints=cons, options={'maxiter': 3000, 'ftol': 1e-18})
            hn = repair(r.x, tgt)
            cn = compute_c5(hn, dx)
            if cn < best_c5:
                best_c5, best_h = cn, hn
            elif cn < best_c5 + 1e-12:
                best_h = hn
            else:
                break

    # Phase 4: Fine restarts with very small perturbations
    if time.time() - t0 < budget_s - 10:
        for i in range(80):
            if time.time() - t0 > budget_s - 3:
                break
            sc = 0.003 + 0.02 * rng.random()
            hp = repair(best_h + rng.randn(n) * sc, tgt)
            tau_r = 0.0005
            r = minimize(lambda x: obj_lse(x, tau_r), hp, method='L-BFGS-B',
                        bounds=bnds, options={'maxiter': 200, 'gtol': 1e-14})
            hn = repair(r.x, tgt)
            cn = compute_c5(hn, dx)
            if cn < best_c5:
                best_c5, best_h = cn, hn
            # Quick SLSQP polish on any improvement
            if cn < best_c5 + 1e-7 and time.time() - t0 < budget_s - 5:
                r2 = minimize(fobj, hn, method='SLSQP', bounds=bnds,
                             constraints=cons, options={'maxiter': 500, 'ftol': 1e-17})
                hn2 = repair(r2.x, tgt)
                cn2 = compute_c5(hn2, dx)
                if cn2 < best_c5:
                    best_c5, best_h = cn2, hn2

    # Final validation
    best_h = repair(best_h, tgt)
    best_c5 = compute_c5(best_h, dx)
    return (best_h, best_c5, best_n)
