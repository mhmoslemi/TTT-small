  h(x+1) for x∈[0,2]: h(x+1) = 1 if x+1∈[0,1] i.e. x∈[-1,0] → empty in [0,2]
  So h(x+1) = 0 for all x∈[0,2]
  O(1) = ∫₀¹ 1·1 dx + ∫₁² 0·1 dx = 1

So C5 = 1 for the indicator function. That's the worst!

The optimal should spread h more evenly. If h(x) = 1/2 for all x∈[0,2]:
- ∫ = 1 ✓
- O(k) = ∫₀² (1/2)(1-1/2) dx = 1/4 · 2 = 1/2 for all k
- C5 = 1/2

So the constant function gives C5 = 0.5.

Can we do better? The Erdős problem asks for the minimum of C5 over all such h. The known upper bound is around 0.38.

So we need h to be non-constant and structured to minimize the max overlap.

Let me reconsider the algorithm. The key challenge is that:
1. The objective max_k O(k) is non-smooth
2. The feasible set is a polytope (0≤h≤1, sum(h)=n/2)
3. We need to find a good local minimum

My gradient approach should work, but let me refine it:

1. Instead of just the gradient at the worst lag, use a "soft max" approximation:
   C5_smooth = log(sum(exp(O(k)/τ))) · τ
   This gives a smooth approximation of the max.

2. Or alternatively, use the subgradient: the subgradient of max_k O(k) at h is the weighted average of gradients at the active lags.

Let me implement the soft-max approach:

C5_soft(h) = τ · log(sum_k exp(O(k)/τ))

Gradient: dC5_soft/dh[m] = (1/sum_k exp(O(k)/τ)) · sum_k exp(O(k)/τ) · dO(k)/dh[m]
            = sum_k (exp(O(k)/τ) / Z) · dO(k)/dh[m]

where Z = sum_k exp(O(k)/τ).

This is a weighted average of the individual lag gradients, weighted by their "importance" (how close they are to the max).

        # d/dh[j] = (1-h[j+k])

        # i.e., j = j' + k, j' in [0, n-1-k], so j in [k, n-1]

        for j in range(n - kk):
            grad[j + kk] += (1.0 - h[j])
            grad[j] -= h[j + kk]
    return grad

import numpy as np
import time

_budget = 1100.0
_start_t = 0.0

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_lags(h, n):
    return np.correlate(h, 1.0 - h, mode="full")

def project(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        hp = np.clip(h - mid, 0.0, 1.0)
        if float(np.sum(hp)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=True)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=True)
    return np.interp(x_new, x_old, h)

def grad_lag(h, k, n):
    grad = np.zeros(n)
    if k >= 0 and k < n:
        grad[:n-k] += (1.0 - h[k:])
        grad[k:] -= h[:n-k]
    elif k < 0 and -k < n:
        kk = -k
        grad[kk:] += (1.0 - h[:n-kk])
        grad[:n-kk] -= h[kk:]
    return grad

def targeted_gradient(h0, n, n_iter=300, step_init=0.02, top_m=8):
    h = h0.copy()
    target = n / 2.0
    best_c = compute_c5(h, n)
    best_h = h.copy()
    step = step_init
    for it in range(n_iter):
        corr = get_lags(h, n)
        top = np.argsort(corr)[::-1][:top_m]
        c_vals = corr[top]
        c_max = c_vals[0]
        w = np.exp(-(c_max - c_vals) / 0.008)
        w /= np.sum(w)
        g = np.zeros(n)
        for wi, ti in zip(w, top):
            k = int(ti) - (n - 1)
            g += wi * grad_lag(h, k, n)
        gn = np.linalg.norm(g)
        if gn < 1e-10:
            break
        g = g / gn
        h_new = h - step * g
        h_new = project(h_new, target)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            step = min(step * 1.02, 0.05)
        else:
            h_try = h - 0.5 * step * g
            h_try = project(h_try, target)
            c_try = compute_c5(h_try, n)
            if c_try < best_c:
                best_c = c_try
                best_h = h_try.copy()
                h = h_try
            step *= 0.85
            if step < 0.0008:
                break
    return best_h, best_c

def coordinate_descent(h0, n, rng, sweeps=15):
    h = h0.copy()
    target = n / 2.0
    best_c = compute_c5(h, n)
    for sw in range(sweeps):
        order = rng.permutation(n)
        any_imp = False
        for j in order:
            old = h[j]
            candidates = [max(0, old-0.04), max(0, old-0.015), min(1, old+0.015), min(1, old+0.04), 0.0, 1.0]
            bp = old
            bv = best_c
            for p in candidates:
                if p < -1e-9 or p > 1+1e-9 or abs(p - old) < 1e-8:
                    continue
                h[j] = p
                h = project(h, target)
                v = compute_c5(h, n)
                if v < bv:
                    bv = v
                    bp = p
            if bp != old:
                h[j] = bp
                h = project(h, target)
                best_c = bv
                any_imp = True
            else:
                h[j] = old
                h = project(h, target)
        if not any_imp:
            break
    return h, best_c

def sa_explore(h0, n, rng, n_iter=2000, T0=0.04, T1=0.002):
    h = h0.copy()
    target = n / 2.0
    best_c = compute_c5(h, n)
    best_h = h.copy()
    c_cur = best_c
    for it in range(n_iter):
        T = T0 * (T1 / T0) ** (it / max(n_iter, 1))
        np_ = rng.randint(1, 4)
        idxs = rng.choice(n, size=np_, replace=False)
        h_try = h.copy()
        h_try[idxs] += rng.uniform(-0.25, 0.25, np_)
        h_try = project(h_try, target)
        c_try = compute_c5(h_try, n)
        if c_try < c_cur or rng.random() < np.exp((c_cur - c_try) / max(T, 1e-6)):
            h = h_try
            c_cur = c_try
            if c_try < best_c:
                best_c = c_try
                best_h = h.copy()
    return best_h, best_c

def make_binary(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h

def make_quasi(n, rng):
    x = np.linspace(0, 2, n)
    h = 0.5 + 0.25 * np.sin(3 * np.pi * x + rng.uniform(0, 6.28))
    h += 0.15 * np.sin(7 * np.pi * x + rng.uniform(0, 6.28))
    h += rng.normal(0, 0.05, n)
    return project(h, n / 2.0)

    return max(0.0, _budget - (time.time() - _start_t))
