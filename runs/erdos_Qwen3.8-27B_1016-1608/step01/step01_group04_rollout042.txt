import time

    corr = np.correlate(h, 1.0 - h, mode='full')

    bds = [(0.0, 1.0)] * n

                 constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    return repair(r.x, n)

    """Project h to {0<=h<=1, sum(h)=target_sum} via bisection on shift."""
    lo, hi = -1.5, 1.5
    for _ in range(80):

        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid

        grad = np.zeros(n)

        for k in range(2 * n - 1):

                continue
            lag = lags[k]

            if lag >= 0:
                idx = np.arange(0, n - lag)
                grad[idx] += p[k] * (1.0 - h[idx + lag])
            else:
                idx = np.arange(-lag, n)
                grad[idx] += p[k] * (1.0 - h[idx + lag])
            # term 2: -h[j-lag] for valid j-lag in [0,n-1]

            jhi = min(n, n + lag)
            if jhi > jlo:
                idx = np.arange(jlo, jhi)
                grad[idx] -= p[k] * h[idx - lag]
        grad *= dx
        # Projected gradient step

        h_new = constrained_project(h_new, n / 2.0)
        c5_new = compute_c5(h_new)
        if c5_new < best_c:
            best_c = c5_new
            best_h = h_new
            h = h_new.copy()
            # Maybe decrease lr slowly
        else:
            lr *= 0.95
    return best_h, best_c

def upsample(h, new_n):

    old_n = len(h)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    h_new = np.interp(x_new, x_old, h)
    return h_new

    for eps in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
        if rem() < 150:
            break
        h2 = _solve(best, n, eps, 400, 1e-14)
        c2 = _comp(h2)
        if c2 < bestc:
            best, bestc = h2, c2

    # Phase 2: Diverse restarts with SLSQP
    n_restart = 200
    for i in range(n_restart):
        if rem() < 60:
            break
        t = i / max(1, n_restart)
        sig = 0.03 * (1 - t) + 0.005
        m = i % 7
        if m == 0:
            hp = best + np.random.randn(n) * sig
        elif m == 1:
            hp = best.copy()
            a, b2 = np.random.choice(n, 2, replace=False)
            hp[a], hp[b2] = 1.0 - hp[a], 1.0 - hp[b2]
        elif m == 2:
            hp = best.copy()
            s = np.random.randint(0, max(1, n - 5))
            l = np.random.randint(2, 5)
            hp[s:s + l] = 1.0 - hp[s:s + l]
            s2 = (s + l) % n
            if s2 + l <= n:
                hp[s2:s2 + l] = 1.0 - hp[s2:s2 + l]
        elif m == 3:
            f2 = np.random.randint(1, 7)
            ph = np.random.uniform(0, 6.283)
            hp = best + 0.025 * np.sin(2.0 * np.pi * f2 * np.arange(n) / n + ph)
        elif m == 4:
            hp = best.copy()
            kN = np.random.randint(2, 5)
            idx = np.random.choice(n, kN, replace=False)
            for j in idx:
                hp[j] += np.random.uniform(-0.08, 0.08)
        elif m == 5:
            hp = best.copy()
            idx = np.random.choice(n, size=max(3, n // 6), replace=False)
            hp[idx] = np.random.uniform(0, 1, size=len(idx))
        else:
            hp = best + np.random.randn(n) * sig * 0.4
        hp = _fix(hp, n)
        eps_r = 0.003 if i < n_restart // 2 else 0.0008
        h2 = _solve(hp, n, eps_r, 250, 1e-14)
        c2 = _comp(h2)
        if c2 < bestc:
            best, bestc = h2, c2

    # Phase 3: Ultra-tight SLSQP polish
    for eps in [5e-5, 2e-5, 1e-5, 5e-6, 2e-6]:
        if rem() < 25:
            break
        h2 = _solve(best, n, eps, 600, 1e-16)
        c2 = _comp(h2)
        if c2 < bestc:
            best, bestc = h2, c2

    # Phase 4: Try upsampled resolution
    if rem() > 200:
        n2 = n * 2
        h_up = _upsample(best, n2)
        h_up = _fix(h_up, n2)
        for eps in [0.005, 0.002, 0.0005, 0.0001]:
            if rem() < 100:
                break
            h2 = _solve(h_up, n2, eps, 300, 1e-14)
            c2 = _comp(h2)
            if c2 < bestc:
                best, bestc, n = h2, c2, n2
                h_up = best.copy()
        if rem() > 100:
            h2 = _solve(best, n, 5e-5, 400, 1e-16)
            c2 = _comp(h2)
            if c2 < bestc:
                best, bestc = h2, c2

    # Phase 5: Local micro-search (direct evaluation, no SLSQP)
    if rem() > 5:
        ntry = int(rem() * 100)
        for _ in range(ntry):
            if rem() < 2:
                break
            hp = best + np.random.randn(n) * 0.002
            hp = _fix(hp, n)
            c2 = _comp(hp)
            if c2 < bestc:
                best, bestc = hp, c2

    # Phase 6: Targeted pair transfers
    if rem() > 10:
        for _ in range(int(rem() * 30)):
            if rem() < 3:
                break
            hp = best.copy()
            i1, i2 = np.random.choice(n, 2, replace=False)
            d = np.random.uniform(-0.03, 0.03)
            hp[i1] += d
            hp[i2] -= d
            hp = _fix(hp, n)
            c2 = _comp(hp)
            if c2 < bestc:
                best, bestc = hp, c2

    # Final SLSQP polish
    if rem() > 15:
        h2 = _solve(best, n, 1e-5, 500, 1e-16)
        c2 = _comp(h2)
        if c2 < bestc:
            best, bestc = h2, c2

    fh = _fix(best, n)
    fc = _comp(fh)
    return (fh.tolist(), float(fc), n)
