import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return c.max() * dx

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_update(c, h, i, new_val, n):
    old_val = h[i]
    delta = new_val - old_val
    if abs(delta) < 1e-16:
        return c
    c_new = c.copy()
    center = n - 1
    for s in range(-(n - 1), n):
        lag = center + s
        if s == 0:
            c_new[lag] += delta * (1.0 - new_val - old_val)
        else:
            if 0 <= i + s < n:
                c_new[lag] += delta * (1.0 - h[i + s])
            if 0 <= i - s < n:
                c_new[lag] -= h[i - s] * delta
    return c_new

def coord_search(h, i, c, n):
    best_x = h[i]
    best_max = c.max()
    grid = np.linspace(0.0, 1.0, 65)
    for x in grid:
        c_try = corr_update(c, h, i, x, n)
        v = c_try.max()
        if v < best_max:
            best_max = v
            best_x = x
    idx = np.argmin(np.abs(grid - best_x))
    lo = grid[max(0, idx - 2)]
    hi = grid[min(len(grid) - 1, idx + 2)]
    grid2 = np.linspace(lo, hi, 31)

        v = c_try.max()
        if v < best_max:

    m = c.max()
    if temp < 1e-12:
        return m
    shifted = (c - m) / temp
    return m + temp * np.log(np.sum(np.exp(shifted)))
