    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return c.max() * dx

    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    obj = (m + temp * np.log(np.sum(np.exp((c - m) / temp)))) * dx
    sum_viol = np.sum(h) - n / 2.0
    obj += penalty * sum_viol ** 2
    return obj

    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        # Only adjust non-bound variables
        lo = h <= 1e-10
        hi = h >= 1.0 - 1e-10
        free = ~(lo | hi)
        if free.sum() == 0:
            break
        h[free] += d / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    k_star = np.argmax(c)
    shift = k_star - (n - 1)
    m = np.arange(n)

    jm = m - shift
    vp = (jp >= 0) & (jp < n)
    vm = (jm >= 0) & (jm < n)
    grad[vp] += (1.0 - h[jp[vp]])
    grad[vm] -= h[jm[vm]]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return c.max() * dx

def soft_max_obj(h, temp):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    return (m + temp * np.log(np.sum(np.exp((c - m) / temp)))) * dx

    return soft_max_obj(h, temp) + mu * (np.sum(h) - len(h) / 2.0) ** 2

    jm = m - shift
    vp = (jp >= 0) & (jp < n)
