        starts.append(repair(hp, n))
    for i in range(6):
        hp = np.random.uniform(0.0, 1.0, n)
        starts.append(repair(hp, n))

    scored = []
    for s in starts:
        scored.append((compute_c5(s), s))
    scored.sort(key=lambda x: x[0])
    if scored[0][0] < bc:
        bc, bh = scored[0][0], scored[0][1]

    for rank in range(8):
        if time.time() - t0 > budget_s - 350:
            break
        x0 = scored[rank][1].copy()
        try:
            res = differential_evolution(
                _de_fit, bds, args=(n,),
                x0=x0, maxiter=50, popsize=35,
                tol=1e-10, seed=seed + rank,
                polish=False, updating='deferred'
            )
            hx = repair(res.x, n)
            c = compute_c5(hx)
            if c < bc:
                bc, bh = c, hx
        except Exception:
            pass

    eps_list = [0.02, 0.01, 0.005, 0.0025, 0.001, 0.0005, 0.0002]
    mi_list = [200, 300, 300, 400, 400, 500, 600]
    for idx, (eps, mi) in enumerate(zip(eps_list, mi_list)):
        if time.time() - t0 > budget_s - 150:
            break
        try:
            r = minimize(_slsqp_obj, bh, args=(n, eps), method='SLSQP',
                         bounds=bds, options={'maxiter': mi, 'ftol': 1e-15})
            hn = repair(r.x, n)
            c = compute_c5(hn)
            if c < bc:
                bc, bh = c, hn
        except Exception:
            pass

    for i in range(30):
        if time.time() - t0 > budget_s - 60:
            break
        r = i % 6
        if r == 0:
            hp = bh + np.random.randn(n) * 0.04
        elif r == 1:
            hp = bh.copy()
            idx = np.random.choice(n, size=n // 5, replace=False)
            hp[idx] = 1.0 - hp[idx]
        elif r == 2:
            k2 = np.random.randint(1, 12)
            ph = np.random.uniform(0, 2 * np.pi)
            hp = bh + 0.04 * np.sin(2 * np.pi * k2 * np.arange(n) / n + ph)
        elif r == 3:
            hp = bh + np.random.randn(n) * 0.012
        elif r == 4:
            hp = bh.copy()
            s = np.random.randint(0, max(1, n - 30))
            l = np.random.randint(5, min(30, n - s))
            hp[s:s + l] = 1.0 - hp[s:s + l]
        else:
            hp = bh.copy()
            idx = np.random.choice(n, size=max(1, n // 6), replace=False)
            hp[idx] = np.random.uniform(0.0, 1.0, size=len(idx))
        hp = repair(hp, n)
        try:
            r2 = minimize(_slsqp_obj, hp, args=(n, 0.005), method='SLSQP',
                          bounds=bds, options={'maxiter': 150, 'ftol': 1e-13})
            hn = repair(r2.x, n)
            c = compute_c5(hn)
            if c < bc:
                bc, bh = c, hn
        except Exception:
            pass

    if time.time() - t0 < budget_s - 15:
        try:
            r = minimize(_slsqp_obj, bh, args=(n, 0.0001), method='SLSQP',
                         bounds=bds, options={'maxiter': 800, 'ftol': 1e-16})
            hn = repair(r.x, n)
            c = compute_c5(hn)
            if c < bc:
                bc, bh = c, hn
        except Exception:
            pass

    if time.time() - t0 < budget_s - 3:
        for _ in range(5000):
            if time.time() - t0 > budget_s - 1:
                break
            hp = bh + np.random.randn(n) * 0.007
            hp = repair(hp, n)
            c = compute_c5(hp)
            if c < bc:
                bc, bh = c, hp

    fh = repair(bh, n)
    fc = compute_c5(fh)
    return (fh.tolist(), float(fc), n)
