```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def gradient_c5(h, n):
    """Compute gradient of max_k g_k(h) using soft-max"""
    dx = 2.0 / n
    beta = 30.0  # soft-max temperature
    corr = np.correlate(h, 1.0 - h, mode="full")  # length 2n-1
    g = corr * dx  # actual g_k values (unscaled)
    # Soft-max weights
    g_shifted = g - np.max(g)  # numerical stability
    w = np.exp(beta * g_shifted)
    w /= np.sum(w)
    # Gradient: sum_k w_k * dg_k/dh
    # dg_s/dh[j] = dx * [(1-h[j-s]) - h[j+s]] where s = k - (n-1)
    # For shift s, k = s + (n-1)
    grad = np.zeros(n)
    for idx in range(len(w)):
        if w[idx] < 1e-10:
            continue
        s = idx - (n - 1)  # shift
        # dg/dh[j] = dx * [(1-h[j-s]) - h[j+s]]
        # h[out of range] = 0
        for j in range(n):
            hjm = h[j - s] if 0 <= j - s < n else 0.0
            hjp = h[j + s] if 0 <= j + s < n else 0.0
            grad[j] += w[idx] * dx * ((1.0 - hjm) - hjp)
    return grad
```
