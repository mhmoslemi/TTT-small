    best_t = 0.0

    for t in [0.0, 1.0]:
        v = np.max(gamma + t * delta)
        if v < best_val:
            best_val = v
            best_t = t
    
    # Evaluate at all pairwise intersections in [0,1]
    # For n=50, 99 correlations → 99*98/2 ≈ 4851 pairs
    for m1 in range(2*n-1):
        for m2 in range(m1+1, 2*n-1):
            dd = delta[m1] - delta[m2]
            if abs(dd) < 1e-14:
                continue
            t = (gamma[m2] - gamma[m1]) / dd
            if 0.0 < t < 1.0:
                v = np.max(gamma + t * delta)
                if v < best_val:
                    best_val = v
                    best_t = t
    
    return best_t, best_val

def coord_descent_sweep(h, rng, n, max_sweeps=50, time_limit=None, t0=None):

                h[i] = t_new
                improved = True
        if not improved:

    target = n / 2.0

        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:

        if diff > 0:

        else:
            mask = h > 0.0

        if cnt == 0:

        h = np.clip(h + diff / cnt * mask, 0.0, 1.0)
    return h

def coord_optimize(h, i, n):
    """Return optimal h[i] value given h[0..i-1,i+1..n-1] fixed."""
    C = np.correlate(h, 1.0 - h, mode='full')
    delta = np.zeros(2*n - 1)
    for d in range(-(n-1), n):
        m = n - 1 + d
        if 0 <= i + d < n:
            delta[m] += 1.0 - h[i + d]
        if 0 <= i - d < n:
            delta[m] += -h[i - d]
    gamma = C - h[i] * delta
    best_val = np.inf
    best_t = 0.0
    cands = [0.0, 1.0]
    lm = 2*n - 1
    for m1 in range(lm):
        for m2 in range(m1+1, lm):
            dd = delta[m1] - delta[m2]
            if abs(dd) > 1e-14:
                t = (gamma[m2] - gamma[m1]) / dd
                if 0.0 < t < 1.0:
                    cands.append(t)
    for t in cands:
        v = np.max(gamma + t * delta)
        if v < best_val:
            best_val = v
            best_t = t
    return best_t

    n_init = len(h_init)
    best_h = h_init.copy()
    best_score = compute_score(h_init)

    # Phase 1: Coordinate descent on initial n (fast, exact per coord)
    _global_n = n_init
    h_cd = h_init.copy()
    h_cd = coord_descent(h_cd, rng, n_init, t0 + budget_s * 0.15)
    h_cd = project_feasible(h_cd, n_init)
    s = compute_score(h_cd)
    if s < best_score:
        best_score = s
        best_h = h_cd.copy()

    # Phase 2: Multi-n coordinate descent
    n_list = sorted(set([n_init, 24, 30, 35, 40, 45, 50, 55, 60, 70, 80, 100, 120]))
    n_list = [x for x in n_list if x >= 16 and x % 2 == 0]
    x_init = np.linspace(0.0, 2.0, n_init, endpoint=False)
    t_phase2 = t0 + budget_s * 0.55

    for n_pts in n_list:
        if time.time() > t_phase2:
            break
        _global_n = n_pts
        x_new = np.linspace(0.0, 2.0, n_pts, endpoint=False)
        h0 = np.interp(x_new, x_init, best_h)
        h0 = project_feasible(h0, n_pts)

        inits = [h0.copy()]
        inits.append(project_feasible(h0 + rng.randn(n_pts) * 0.08, n_pts))
        if n_pts <= 100:
            inits.append(project_feasible(rng.uniform(0, 1, n_pts), n_pts))

        for h_start in inits:
            if time.time() > t_phase2:
                break
            t_sub = min(t_phase2, time.time() + budget_s * 0.06)
            h_cand = coord_descent(h_start.copy(), rng, n_pts, t_sub)
            h_cand = project_feasible(h_cand, n_pts)
            s_cand = compute_score(h_cand)
            if s_cand < best_score:
                best_score = s_cand
                best_h = h_cand.copy()

    # Phase 3: Binary swap SA on best solution's n
    t_phase3 = t0 + budget_s * 0.78
    n_best = len(best_h)
    _global_n = n_best
    h_bin = (best_h > 0.5).astype(float)
    n_ones = int(round(n_best / 2))
    if np.sum(h_bin) != n_ones:
        idx = np.argsort(h_bin)[::-1][:n_ones]
        h_bin = np.zeros(n_best)
        h_bin[idx] = 1.0
    t_sub3 = min(t_phase3, time.time() + budget_s * 0.12)
    h_sa, s_sa = binary_swap_sa(h_bin, rng, n_best, t_sub3)
    if s_sa < best_score:
        best_score = s_sa
        best_h = h_sa.copy()

    # Also try binary SA from random start
    if time.time() < t_phase3:
        h_rand_bin = np.zeros(n_best)
        h_rand_bin[rng.choice(n_best, n_best // 2, replace=False)] = 1.0
        t_sub3b = min(t_phase3, time.time() + budget_s * 0.08)
        h_sa2, s_sa2 = binary_swap_sa(h_rand_bin, rng, n_best, t_sub3b)
        if s_sa2 < best_score:
            best_score = s_sa2
            best_h = h_sa2.copy()

    # Phase 4: SLSQP polish
    t_phase4 = t0 + budget_s * 0.90
    n_pol = len(best_h)
    _global_n = n_pol
    bounds = [(0.0, 1.0)] * n_pol
    constraints = [{'type': 'eq', 'fun': _con_fun}]
    for tau in [80, 150, 300]:
        if time.time() > t_phase4:
            break
        try:
            res = minimize(lambda x: soft_max_obj(x, tau), best_h.copy(),
                           method='SLSQP', bounds=bounds, constraints=constraints,
                           options={'maxiter': 400, 'ftol': 1e-15})
            h_cand = project_feasible(res['x'], n_pol)
            s_cand = compute_score(h_cand)
            if s_cand < best_score:
                best_score = s_cand
                best_h = h_cand.copy()
        except Exception:
            pass

    # Phase 5: Final coordinate descent polish
    if time.time() < t0 + budget_s * 0.96:
        n_fin = len(best_h)
        _global_n = n_fin
        h_fin = best_h.copy()
        h_fin = coord_descent(h_fin, rng, n_fin, t0 + budget_s * 0.96)
        h_fin = project_feasible(h_fin, n_fin)
        s_fin = compute_score(h_fin)
        if s_fin < best_score:
            best_score = s_fin
            best_h = h_fin.copy()

    # Final validation
    best_h = project_feasible(best_h, len(best_h))
    best_score = compute_score(best_h)
    assert best_h.shape[0] == len(best_h)
    assert np.all(best_h >= -1e-9) and np.all(best_h <= 1.0 + 1e-9)
    best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_score, len(best_h)
