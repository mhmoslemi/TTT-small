    n = len(h)
    corr = np.correlate(h, 1.0-h, mode='full')
    # max index
    k_max = np.argmax(corr)
    # subgradient of corr[k_max] w.r.t. h
    # corr has length 2n-1, index n-1+k corresponds to shift k
    shift = k_max - (n-1)
    grad = np.zeros(n)
    if shift >= 0:
        # f = sum_{i=0}^{n-1-shift} h[i]*(1-h[i+shift])
        for i in range(n-shift):
            grad[i] += (1-h[i+shift])
            grad[i+shift] += -h[i]
    else:

        for i in range(n-shift):

            grad[i] += -h[i+shift]
    # Projected subgradient step
    h_new = h - step_size * grad
    h_new = np.clip(h_new, 0, 1)

            fc = g(c)
        else:
            a = c; c = d; fc = fd
            d = a + gr * (b - a)
            fd = g(d)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_c5(h, dx, temp):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    scaled = temp * (corr - m)
    return (m + np.log(np.sum(np.exp(scaled)) / len(corr)) / temp) * dx

def power_c5(h, dx, p):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.mean(corr**p)**(1.0/p))

def projection(h, target):
    h = np.array(h, dtype=float, copy=True)
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        excess = h.sum() - target
        if abs(excess) < 1e-14:
            return h
        if excess > 0:
            mask = h > 0
            cnt = mask.sum()
            if cnt == 0:
                return h
            s = min(excess / cnt, h[mask].min() - 1e-15)
            if s < 0:
                s = excess / cnt
                h[mask] -= s
            else:
                h[mask] -= s
        else:
            mask = h < 1
            cnt = mask.sum()
            if cnt == 0:
                return h
            s = min(-excess / cnt, 1.0 - h[mask].min() - 1e-15)
            if s < 0:
                s = -excess / cnt
                h[mask] += s
            else:
                h[mask] += s
    return np.clip(h, 0, 1)

def coord_descent(h, dx, target, max_sweeps=2000, tol=1e-12):
    n = len(h)
    h = np.array(h, dtype=float).copy()
    h = projection(h, target)
    best = compute_c5(h, dx)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    stall = 0
    for sweep in range(max_sweeps):
        sweep_improve = 0.0
        for i in range(n):
            h_save = h.copy()
            a, b = 0.0, 1.0
            for _ in range(45):
                if b - a < tol:
                    break
                c = b - gr * (b - a)
                d = a + gr * (b - a)
                h[i] = c
                hc = projection(h, target)
                fc = compute_c5(hc, dx)
                h[i] = d
                hd = projection(h, target)
                fd = compute_c5(hd, dx)
                if fc < fd:
                    b, h[i] = d, b
                else:
                    a, h[i] = c, a
            opt = 0.5 * (a + b)
            h[i] = opt
            h = projection(h, target)
            val = compute_c5(h, dx)
            if val < best - 1e-14:
                best = val
                sweep_improve += best
            else:
                h = h_save
        if sweep_improve < 1e-14:
            stall += 1
            if stall > 3:
                break
        else:
            stall = 0
    return h, best

def periodic_init(n, period, seed=42):
    rng = np.random.RandomState(seed)
    pat = rng.rand(period) * 0.8 + 0.1
    pat = pat / pat.sum() * (n / period) * 0.5
    pat = projection(pat, (n / period) * 0.5)
    h = np.tile(pat, n // period)
    if len(h) < n:
        h = np.concatenate([h, pat[:n - len(h)]])
    return projection(h, n / 2.0)
