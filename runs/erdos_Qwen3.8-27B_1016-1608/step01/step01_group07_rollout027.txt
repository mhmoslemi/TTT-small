from scipy.optimize import minimize
import time

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def _fix(h, n):
    """Project onto box [0,1]^n and sum = n/2."""
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        d = 0.5 * n - np.sum(h)
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        k = mask.sum()
        if k == 0:
            break
        h[mask] += d / k
    return np.clip(h, 0.0, 1.0)

def _project_box_sum(h, target_sum):
    """Faster projection via bisection."""
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0.0, 1.0)

    dx = 2.0 / n
    grad = np.zeros(n)
    j = np.arange(n)

def _projected_subgrad(h, n, iters, lr_init, lr_end, t0, budget):
    """Run projected subgradient method."""
    h = h.copy()
    for it in range(iters):
        if budget - (time.time() - t0) < 5:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(c))
        shift = k_star - (n - 1)
        grad = _subgrad(c, h, n, shift)
        # Decay learning rate
        lr = lr_init * (lr_end / lr_init) ** (it / max(iters-1, 1))
        h = h - lr * grad
        h = _project_box_sum(h, 0.5 * n)
    return h

                    # Actually for k=0: c[0] = sum h[i](1-h[i]), dc[0]/dh[j] = 1 - 2h[j]
