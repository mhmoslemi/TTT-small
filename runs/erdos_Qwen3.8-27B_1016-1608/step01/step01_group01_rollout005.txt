def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return c.max() * dx

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_obj(h, temp, n):
    c = corr_full(h)
    m = c.max()
    return m * 2.0/n + temp * 2.0/n * np.log(np.sum(np.exp((c - m) / temp)))

def sum_con_fun(h):
    return np.sum(h) - len(h) / 2.0

    return np.ones(len(h))

def slsqp_polish(h_init, n, t_start, time_limit, temps):
    """Polish with SLSQP using soft-max objective."""
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    constraints = [{'type': 'eq', 'fun': sum_con_fun, 'jac': sum_con_jac}]

                      method='SLSQP', bounds=bounds, constraints=constraints,
                      options={'maxiter': 2000, 'ftol': 1e-14})

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return c.max() * dx

    return np.correlate(h, 1.0 - h, mode='full')

    target = n / 2.0

def sa_binary_single(n, t_start, time_limit, rng, symmetric=False):
    if not symmetric:
        target = n // 2
        pos = rng.choice(n, target, replace=False)

        h[pos] = 1.0
        is_one = h.astype(bool)
    else:
        half = n // 2
        target_half = half // 2
        pos_half = rng.choice(half, target_half, replace=False)
        h_half = np.zeros(half)
        h_half[pos_half] = 1.0
        h = np.concatenate([h_half, h_half[::-1]])
        is_one = h.astype(bool)
    
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    T = 0.5

                return best_h, best_c5
            one_pos = np.where(is_one)[0]
