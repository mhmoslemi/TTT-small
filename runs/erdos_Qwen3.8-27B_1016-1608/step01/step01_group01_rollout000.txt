    grad = np.zeros(n)
    for idx in top_idx:
        shift = idx - (n - 1)
        m = np.arange(n)
        jp = m + shift
        jm = m - shift
        vp = (jp >= 0) & (jp < n)
        vm = (jm >= 0) & (jm < n)
        grad[vp] += (1.0 - h[jp[vp]])
        grad[vm] -= h[jm[vm]]
    return grad / float(k)

def sum_fun(h):
    return h.sum() - len(h) / 2.0

def sum_jac(h):
    return np.ones(len(h))

def soft_max_obj(h, temp):
    c = corr_values(h)
    m = c.max()
    return m + temp * np.log(np.sum(np.exp((c - m) / temp)))

def soft_max_jac(h, temp):
    c = corr_values(h)
    n = len(h)
    m = c.max()
    w = np.exp((c - m) / temp)
    w /= w.sum()
    grad = np.zeros(n)
    for i in range(len(w)):
        if w[i] < 1e-12:
            continue
        shift = i - (n - 1)
        mm = np.arange(n)
        jp = mm + shift
        jm = mm - shift
        vp = (jp >= 0) & (jp < n)
        vm = (jm >= 0) & (jm < n)
        grad[vp] += w[i] * (1.0 - h[jp[vp]])
        grad[vm] -= w[i] * h[jm[vm]]
    return grad

def make_bounds(n):
    return [(0.0, 1.0)] * n

def make_cons(n):
    return [{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac}]

def optimize_one(h_init, n, budget_end, rng):
    best_h = project_feasible(h_init.copy(), n)
    best_c5 = compute_c5(best_h)
    bounds = make_bounds(n)
    cons = make_cons(n)

    # SA phase
    sa_end = min(budget_end - 60, time.time() + (budget_end - time.time()) * 0.30)
    temp = 0.15
    cooling = 0.99965
    for it in range(200000):
        if time.time() > sa_end:
            break
        h_new = best_h.copy()
        npert = rng.randint(1, 6)
        idx = rng.randint(0, n, npert)
        h_new[idx] += rng.normal(0, 0.25, npert)
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        delta = c5_new - best_c5
        if delta < 0 or (temp > 1e-6 and rng.random() < np.exp(-delta / temp)):
            best_h = h_new
            best_c5 = c5_new
        temp *= cooling

    # SLSQP with soft-max, decreasing temps
    slsqp_end = min(budget_end - 40, time.time() + (budget_end - sa_end) * 0.45)
    for t in [0.03, 0.01, 0.003, 0.001]:
        if time.time() > slsqp_end:
            break
        res = minimize(soft_max_obj, best_h.copy(), args=(t,), jac=lambda x: soft_max_jac(x, t),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': 1500, 'ftol': 1e-13})
        h_new = project_feasible(res.x, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Top-k subgradient with adaptive step
    sg_end = min(budget_end - 30, time.time() + (budget_end - slsqp_end) * 0.50)
    alpha = 0.08
    for k in [1, 2, 3]:
        if time.time() > sg_end:
            break
        patience = 0
        for it in range(5000):
            if time.time() > sg_end:
                break
            grad = subgrad_topk(best_h, k)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-12:
                break
            step = alpha / (gnorm + 1e-12)
            h_trial = project_feasible(best_h - step * grad, n)
            c5_t = compute_c5(h_trial)
            if c5_t < best_c5 - 1e-14:
                best_c5 = c5_t
                best_h = h_trial.copy()
                patience = 0
            else:
                patience += 1
                if patience > 80:
                    alpha *= 0.82
                    patience = 0

    # Coordinate descent with binary search refinement
    cd_end = budget_end
    deltas_coarse = [0.05, -0.05, 0.1, -0.1, 0.02, -0.02]
    deltas_fine = [0.01, -0.01, 0.005, -0.005, 0.002, -0.002]
    for deltas in [deltas_coarse, deltas_fine]:
        improved = True
        while improved and time.time() < cd_end:
            improved = False
            order = rng.permutation(n)
            for i in order:
                if time.time() > cd_end:
                    break
                for d in deltas:
                    h_trial = best_h.copy()
                    h_trial[i] = min(1.0, max(0.0, h_trial[i] + d))
                    h_trial = project_feasible(h_trial, n)
                    c5_t = compute_c5(h_trial)
                    if c5_t < best_c5 - 1e-14:
                        best_c5 = c5_t
                        best_h = h_trial.copy()
                        improved = True
                        break

    # Final SLSQP polish
    if time.time() < cd_end - 5:
        res = minimize(soft_max_obj, best_h.copy(), args=(0.0005,),
                      jac=lambda x: soft_max_jac(x, 0.0005),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': 2000, 'ftol': 1e-14})
        h_new = project_feasible(res.x, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)

    # Try the initial solution directly
    h_init = project_feasible(h0.copy(), n)
    best_h, best_c5 = optimize_one(h_init, n, t_start + budget_s * 0.55, rng)

    # Try a few alternative n values with structured starts
    alt_ns = [n - 2, n + 2, n - 4, n + 4]
    t_remaining = t_start + budget_s * 0.88 - time.time()
    for alt_n in alt_ns:
        if alt_n < 20 or alt_n > 200:
            continue
        budget_per = max(t_remaining / (len(alt_ns)), 5.0)
        if budget_per < 15:
            break
        # Structured start: alternating blocks
        h_alt = np.zeros(alt_n)
        block = max(alt_n // 12, 2)
        for b in range(0, alt_n, block * 2):
            h_alt[b:b + block] = 1.0
        h_alt = project_feasible(h_alt, alt_n)
        # Add randomness
        h_alt += rng.normal(0, 0.15, alt_n)
        h_alt = project_feasible(h_alt, alt_n)

        h_cand, c5_cand = optimize_one(h_alt, alt_n, t_start + budget_s * 0.88, rng)
        if c5_cand < best_c5:
            best_c5 = c5_cand
            best_h = h_cand
            n = alt_n

    # Final fine-tune on the best solution
    if time.time() < t_start + budget_s * 0.96:
        n = len(best_h)
        bounds = make_bounds(n)
        cons = make_cons(n)
        for t_fm in [0.0002, 0.0001]:
            if time.time() > t_start + budget_s * 0.96:
                break
            res = minimize(soft_max_obj, best_h.copy(), args=(t_fm,),
                          jac=lambda x, t=t_fm: soft_max_jac(x, t),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': 1000, 'ftol': 1e-14})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

        # Ultra-fine coord descent
        d_fine = [0.001, -0.001, 0.003, -0.003]
        for _ in range(3):
            if time.time() > t_start + budget_s * 0.96:
                break
            improved = True
            while improved and time.time() < t_start + budget_s * 0.96:
                improved = False
                for i in rng.permutation(n):
                    if time.time() > t_start + budget_s * 0.96:
                        break
                    for d in d_fine:
                        h_trial = best_h.copy()
                        h_trial[i] = min(1.0, max(0.0, h_trial[i] + d))
                        h_trial = project_feasible(h_trial, n)
                        c5_t = compute_c5(h_trial)
                        if c5_t < best_c5 - 1e-14:
                            best_c5 = c5_t
                            best_h = h_trial.copy()
                            improved = True
                            break

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n)
