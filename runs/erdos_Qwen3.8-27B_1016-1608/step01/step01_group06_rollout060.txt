        h_bin[np.argsort(-h)[:target]] = 1.0
    else:
        h_bin[np.argsort(h)[target:]] = 1.0
    return h_bin

def local_search_full(h, n, max_iters, rng):
    """Full-sweep swap local search: try all pairs, pick best improvement."""
    h = project_feasible(h, n)
    best_c = np.max(np.correlate(h, 1.0 - h, mode='full'))
    curr_c = best_c
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    stall = 0
    for it in range(max_iters):
        if len(ones) == 0 or len(zeros) == 0:
            break
        n_ones = len(ones)
        n_zeros = len(zeros)
        best_delta = 0
        best_pair = (-1, -1)
        # Sample swaps if too many
        if n_ones * n_zeros > 5000:
            pairs = [(ones[rng.randint(n_ones)], zeros[rng.randint(n_zeros)]) for _ in range(300)]
        else:
            pairs = [(a, b) for a in ones for b in zeros]
        for a, b in pairs:
            h[a] = 0.0
            h[b] = 1.0
            new_c = np.max(np.correlate(h, 1.0 - h, mode='full'))
            h[a] = 1.0
            h[b] = 0.0
            delta = new_c - curr_c
            if delta < best_delta:
                best_delta = delta
                best_pair = (a, b)
        if best_pair[0] == -1:
            stall += 1
            if stall > 5:
                break
            # Random perturbation
            a = ones[rng.randint(n_ones)]
            b = zeros[rng.randint(n_zeros)]
            h[a], h[b] = 0.0, 1.0
            curr_c = np.max(np.correlate(h, 1.0 - h, mode='full'))
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            continue
        a, b = best_pair
        h[a] = 0.0
        h[b] = 1.0
        curr_c += best_delta
        if curr_c < best_c:
            best_c = curr_c
            stall = 0
        else:
            stall += 1
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
    return h, best_c * (2.0 / n)

def local_search_sa(h, n, max_iters, rng, temp0=0.05):
    """Simple random-swap local search with simulated annealing."""
    h = project_feasible(h, n)
    best_c = np.max(np.correlate(h, 1.0 - h, mode='full'))
    curr_c = best_c
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    temp = temp0
    for it in range(max_iters):
        if len(ones) == 0 or len(zeros) == 0:
            break
        a = ones[rng.randint(len(ones))]
        b = zeros[rng.randint(len(zeros))]
        h[a], h[b] = 0.0, 1.0
        new_c = np.max(np.correlate(h, 1.0 - h, mode='full'))
        delta = new_c - curr_c
        if delta < 0 or (temp > 1e-8 and rng.random() < np.exp(-delta / (temp * 2.0))):
            curr_c = new_c
            if curr_c < best_c:
                best_c = curr_c
        else:
            h[a], h[b] = 1.0, 0.0
        temp *= 0.9995
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
    return h, best_c * (2.0 / n)

def run(seed=42, budget_s=1100, **kwargs):
    global _global_n
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    best_h = h_init.copy()
    best_score = compute_score(h_init)
    x_ref = np.linspace(0.0, 2.0, n_init, endpoint=False)
    h_ref = h_init.copy()

    n_list = sorted(set([n_init, 40, 48, 50, 56, 60, 64, 72, 80, 90, 100, 110, 120]))
    n_list = [n for n in n_list if 20 <= n <= 150]

    for n_pts in n_list:
        if time.time() - t0 > budget_s * 0.80:
            break
        _global_n = n_pts
        x_new = np.linspace(0.0, 2.0, n_pts, endpoint=False)
        h0 = np.interp(x_new, x_ref, h_ref)
        h0 = project_feasible(h0, n_pts)
        s0 = compute_score(h0)
        if s0 < best_score:
            best_score = s0
            best_h = h0.copy()

        bounds = [(0.0, 1.0)] * n_pts
        constraints = [{'type': 'eq', 'fun': _constraint_fun}]

        # SLSQP soft-max
        for tau in [400.0, 1000.0]:
            if time.time() - t0 > budget_s * 0.83:
                break
            try:
                res = minimize(soft_max_obj, h0, args=(tau,), method='SLSQP',
                               bounds=bounds, constraints=constraints,
                               options={'maxiter': 150, 'ftol': 1e-12})
                h_c = project_feasible(res['x'], n_pts)
                s_c = compute_score(h_c)
                if s_c < best_score:
                    best_score = s_c
                    best_h = h_c.copy()
                if s_c < s0:
                    s0 = s_c
                    h0 = h_c.copy()
            except Exception:
                pass

        # SLSQP exact
        if time.time() - t0 < budget_s * 0.85:
            try:
                res = minimize(compute_score, h0, method='SLSQP', bounds=bounds,
                               constraints=constraints,
                               options={'maxiter': 300, 'ftol': 1e-14})
                h_c = project_feasible(res['x'], n_pts)
                s_c = compute_score(h_c)
                if s_c < best_score:
                    best_score = s_c
                    best_h = h_c.copy()
                if s_c < s0:
                    s0 = s_c
                    h0 = h_c.copy()
            except Exception:
                pass

        # Binary local search from rounded start
        if time.time() - t0 < budget_s * 0.82 and n_pts <= 120:
            h_bin = make_binary(h0, n_pts, rng)
            iters = 6000 if n_pts <= 70 else 3000
            h_ls, s_ls = local_search_full(h_bin, n_pts, iters, rng)
            if s_ls < best_score:
                best_score = s_ls
                best_h = h_ls.copy()
            if s_ls < s0:
                s0 = s_ls
                h0 = h_ls.copy()

            # SA local search for diversity
            if time.time() - t0 < budget_s * 0.84:
                h_sa, s_sa = local_search_sa(h_bin, n_pts, 4000, rng, 0.08)
                if s_sa < best_score:
                    best_score = s_sa
                    best_h = h_sa.copy()
                if s_sa < s0:
                    s0 = s_sa
                    h0 = h_sa.copy()

            # Polish best binary with SLSQP
            if time.time() - t0 < budget_s * 0.85:
                try:
                    res = minimize(compute_score, h0, method='SLSQP', bounds=bounds,
                                   constraints=constraints,
                                   options={'maxiter': 200, 'ftol': 1e-14})
                    h_c = project_feasible(res['x'], n_pts)
                    s_c = compute_score(h_c)
                    if s_c < best_score:
                        best_score = s_c
                        best_h = h_c.copy()
                    if s_c < s0:
                        s0 = s_c
                        h0 = h_c.copy()
                except Exception:
                    pass

        # Nelder-Mead for small n
        if time.time() - t0 < budget_s * 0.82 and n_pts <= 64:
            for r in range(2):
                if time.time() - t0 > budget_s * 0.84:
                    break
                h_start = project_feasible(h0 + rng.randn(n_pts) * 0.04, n_pts)
                try:
                    res = minimize(_penalty_obj, h_start, method='Nelder-Mead',
                                   options={'maxiter': 600, 'xatol': 1e-5, 'fatol': 1e-9})
                    h_c = project_feasible(res['x'], n_pts)
                    s_c = compute_score(h_c)
                    if s_c < best_score:
                        best_score = s_c
                        best_h = h_c.copy()
                    if s_c < s0:
                        s0 = s_c
                        h0 = h_c.copy()
                except Exception:
                    pass

        x_ref = x_new
        h_ref = h0.copy()

    # Random binary starts
    if time.time() - t0 < budget_s * 0.60:
        for n_small in [n_init, 50, 64, 80]:
            for trial in range(3):
                if time.time() - t0 > budget_s * 0.78:
                    break
                h_rand = np.zeros(n_small)
                pos = rng.choice(n_small, size=n_small // 2, replace=False)
                h_rand[pos] = 1.0
                h_ls, s_ls = local_search_full(h_rand, n_small, 4000, rng)
                if s_ls < best_score:
                    best_score = s_ls
                    best_h = h_ls.copy()

    # Final polish
    if time.time() - t0 < budget_s * 0.93:
        n_f = len(best_h)
        _global_n = n_f
        bounds = [(0.0, 1.0)] * n_f
        constraints = [{'type': 'eq', 'fun': _constraint_fun}]
        try:
            res = minimize(compute_score, best_h, method='SLSQP', bounds=bounds,
                           constraints=constraints, options={'maxiter': 500, 'ftol': 1e-14})
            h_final = project_feasible(res['x'], n_f)
            s_final = compute_score(h_final)
            if s_final < best_score:
                best_score = s_final
                best_h = h_final
        except Exception:
            pass
        if time.time() - t0 < budget_s * 0.97:
            try:
                res = minimize(soft_max_obj, best_h, args=(2000.0,), method='SLSQP',
                               bounds=bounds, constraints=constraints,
                               options={'maxiter': 300, 'ftol': 1e-14})
                h_c = project_feasible(res['x'], n_f)
                s_c = compute_score(h_c)
                if s_c < best_score:
                    best_score = s_c
                    best_h = h_c
            except Exception:
                pass

    return best_h, best_score, len(best_h)
