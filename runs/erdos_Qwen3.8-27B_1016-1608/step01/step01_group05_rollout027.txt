        res = minimize(obj, h0, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                                     'jac': lambda x: ones_vec}],
                       options={'maxiter': maxiter, 'ftol': 1e-10})
        h_out = project_to_set(res.x.copy(), target)
        return h_out, compute_c5(h_out, n)
    except Exception:
        return h0, compute_c5(h0, n)

def fast_coord(h, n, rng, n_sweeps=100, t_limit=60.0):
    h = h.copy()
    t0 = time.time()
    for sweep in range(n_sweeps):
        if time.time() - t0 > t_limit:
            break
        h = np.clip(project_to_set(h, n / 2.0), 0, 1)
        base_corr = np.correlate(h, 1.0 - h, mode='full')
        base_max = float(np.max(base_corr))
        improved = False
        order = rng.permutation(n)
        for j in order:
            old = h[j]
            best_v = old
            best_m = base_max
            cands = np.array([0.0, 1.0, old - 0.3, old + 0.3, old - 0.1, old + 0.1,
                              old - 0.04, old + 0.04])
            cands = cands[(cands >= 0.0) & (cands <= 1.0) & (np.abs(cands - old) > 1e-8)]
            for cv in cands:
                h[j] = cv
                c = np.correlate(h, 1.0 - h, mode='full')
                mv = float(np.max(c))
                if mv < best_m:
                    best_m = mv
                    best_v = cv
            if best_v != old:
                h[j] = best_v
                base_max = best_m
                improved = True
            else:
                h[j] = old
        if not improved:
            break
    h = project_to_set(h, n / 2.0)
    return h, compute_c5(h, n)

def binary_local_search(h, n, rng, t_limit=60.0):
    h = project_to_set(h, n / 2.0)
    h = (h > 0.5).astype(float)
    if np.sum(h) != n // 2:
        h = make_binary(n, rng)
    t0 = time.time()
    best_c = compute_c5(h, n)
    while time.time() - t0 < t_limit:
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        for _ in range(200):
            j = ones[rng.randint(len(ones))]
            k = zeros[rng.randint(len(zeros))]
            h[j], h[k] = h[k], h[j]
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                improved = True
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
            else:
                h[j], h[k] = h[k], h[j]
        if not improved:
            break
    if np.sum(h) != n // 2:
        h = project_to_set(h, n / 2.0)
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    _tl = [budget_s]

    def tleft():
        return max(0.0, budget_s - (time.time() - t_start))

    rng = np.random.RandomState(seed)
    best_h = [None]
    best_c5 = [np.inf]
    best_n = [0]

    def update(h, n):
        h = project_to_set(np.asarray(h, dtype=float), n / 2.0)
        c = compute_c5(h, n)
        if c < best_c5[0]:
            best_h[0] = h.copy()
            best_c5[0] = c
            best_n[0] = n

    # Get initial
    h_init = None
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        pass

    n1 = 90
    if h_init is not None:
        if len(h_init) == n1:
            pass
        else:
            h_init = upsample(h_init, n1)
    else:
        h_init = make_binary(n1, rng)
    h_init = project_to_set(h_init, n1 / 2.0)
    update(h_init, n1)

    # SLSQP on initial
    if tleft() > 100:
        h_s, c = slsqp_optimize(h_init, n1, maxiter=800)
        update(h_s, n1)

    # Fast coordinate search
    if tleft() > 80:
        h_cs, c = fast_coord(best_h[0], best_n[0], rng, n_sweeps=300, t_limit=min(150, tleft() * 0.35))
        update(h_cs, best_n[0])

    # Binary local search
    if tleft() > 60:
        h_b, c = binary_local_search(best_h[0], n1, rng, t_limit=min(40, tleft() * 0.2))
        update(h_b, n1)
        h_cs, c = fast_coord(h_b, n1, rng, n_sweeps=100, t_limit=min(30, tleft() * 0.15))
        update(h_cs, n1)

    # Random restarts at n=90
    restart_budget = budget_s * 0.35
    while tleft() > restart_budget:
        slot = rng.rand()
        if slot < 0.4:
            h_r = make_binary(n1, rng)
        elif slot < 0.7:
            h_r = make_smooth(n1, rng)
        else:
            h_r = best_h[0].copy() + rng.normal(0, 0.1, n1)
        h_r = project_to_set(h_r, n1 / 2.0)
        h_s, c = slsqp_optimize(h_r, n1, maxiter=300)
        update(h_s, n1)
        h_cs, c = fast_coord(h_s, n1, rng, n_sweeps=80, t_limit=min(25, tleft() * 0.12))
        update(h_cs, n1)

    # Phase 2: n=180
    if tleft() > budget_s * 0.22:
        n2 = 180
        h_up = project_to_set(upsample(best_h[0], n2), n2 / 2.0)
        update(h_up, n2)
        if tleft() > budget_s * 0.18:
            h_s, c = slsqp_optimize(h_up, n2, maxiter=500)
            update(h_s, n2)
        if tleft() > budget_s * 0.14:
            h_cs, c = fast_coord(best_h[0], best_n[0], rng, n_sweeps=150, t_limit=min(80, tleft() * 0.3))
            update(h_cs, best_n[0])
        # Also try binary at n2
        if tleft() > budget_s * 0.10:
            h_b = make_binary(n2, rng)
            h_b = np.clip(project_to_set(h_b, n2 / 2.0), 0, 1)
            h_cs, c = fast_coord(h_b, n2, rng, n_sweeps=100, t_limit=min(40, tleft() * 0.2))
            update(h_cs, n2)

    # Phase 3: n=360
    if tleft() > budget_s * 0.10:
        n3 = 360
        h_up = project_to_set(upsample(best_h[0], n3), n3 / 2.0)
        update(h_up, n3)
        if tleft() > budget_s * 0.05:
            h_s, c = slsqp_optimize(h_up, n3, maxiter=300)
            update(h_s, n3)
        if tleft() > 25:
            h_cs, c = fast_coord(best_h[0], best_n[0], rng, n_sweeps=80, t_limit=min(35, tleft() * 0.5))
            update(h_cs, best_n[0])

    # Final polish
    if tleft() > 15:
        h_cs, c = fast_coord(best_h[0], best_n[0], rng, n_sweeps=200, t_limit=min(25, tleft() * 0.7))
        update(h_cs, best_n[0])

    best_h[0] = project_to_set(best_h[0], best_n[0] / 2.0)
    best_c5[0] = compute_c5(best_h[0], best_n[0])
    return best_h[0], best_c5[0], best_n[0]
