def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_set(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def upsample_h(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, h)

def coord_descent(h, n, n_sweeps, rng, n_cands=16):
    target = n / 2.0
    h = h.copy()
    c5 = compute_c5(h, n)
    grid = np.linspace(0.0, 1.0, n_cands)
    for sweep in range(n_sweeps):
        improved = False
        order = rng.permutation(n)
        for j in order:
            old = h[j]
            best_v = old
            best_c5_local = c5
            # Add old value to grid if not present
            for c in grid:
                if abs(c - old) < 1e-4:
                    continue
                h[j] = c
                h = project_to_set(h, target)
                v = compute_c5(h, n)
                if v < best_c5_local:

                    best_v = c
            if best_v != old:
                h[j] = best_v
                h = project_to_set(h, target)
                c5 = best_c5_local
                improved = True
            else:
                h[j] = old
                h = project_to_set(h, target)
        if not improved:
            break
    return h, c5

            else:

        elif r < 0.85:

            h[j] = np.clip(old + rng.normal(0, 0.2), 0.0, 1.0)
            h = project_to_set(h, target)
            new_c5 = compute_c5(h, n)

def subgradient_step(h, n, lr=0.01, top_m=5, rng=None):

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_set(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def upsample_h(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, h)

def make_binary_h(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h

def coord_descent(h, n, n_sweeps, rng, n_cands=20, t_end=None):
    target = n / 2.0
    h = h.copy()
    c5 = compute_c5(h, n)
    grid = np.unique(np.concatenate([np.linspace(0.0, 1.0, n_cands),
                                      np.array([0.0, 1.0, 0.25, 0.5, 0.75])]))
    for sweep in range(n_sweeps):
        if t_end is not None and time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n)
        for j in order:
            old = h[j]
            best_v = old
            best_local = c5
            # Adaptive: focus near current value
            near = np.clip(np.array([old - 0.1, old - 0.05, old + 0.05, old + 0.1]), 0, 1)
            all_cands = np.concatenate([grid, near])
            all_cands = np.unique(all_cands)
            for c in all_cands:
                if abs(c - old) < 1e-5:
                    continue
                h[j] = c
                h = project_to_set(h, target)
                v = compute_c5(h, n)
                if v < best_local:
                    best_local = v
                    best_v = float(c)
            if best_v != old:
                h[j] = best_v
                h = project_to_set(h, target)
                c5 = best_local
                improved = True
            else:
                h[j] = old
                h = project_to_set(h, target)
        if not improved:
            break
    return h, c5

def sa_search(h0, n, rng, n_iter, T_init, T_min):
    target = n / 2.0
    h = project_to_set(h0.copy(), target)
    c5 = compute_c5(h, n)
    best_h = h.copy()
    best_c5 = c5
    cool = (T_min / T_init) ** (1.0 / max(1, n_iter))
    T = T_init
    for it in range(n_iter):
        r = rng.random()
        if r < 0.55:
            i, j = rng.randint(0, n, size=2)
            if i == j:
                T *= cool
                continue
            h[i], h[j] = h[j], h[i]
            new_c5 = compute_c5(h, n)
            d = new_c5 - c5
            if d > 0 and rng.random() > np.exp(-d / T):
                h[i], h[j] = h[j], h[i]
            else:
                c5 = new_c5
        elif r < 0.85:
            j = rng.randint(0, n)
            old = h[j]
            h[j] = np.clip(old + rng.normal(0, 0.25), 0.0, 1.0)
            h = project_to_set(h, target)
            new_c5 = compute_c5(h, n)
            d = new_c5 - c5
            if d > 0 and rng.random() > np.exp(-d / T):
                h[j] = old
                h = project_to_set(h, target)
                c5 = compute_c5(h, n)
            else:
                c5 = new_c5
        else:
            blk = rng.randint(2, max(3, n // 5))
            i = rng.randint(0, n - blk + 1)
            h[i:i + blk] = 1.0 - h[i:i + blk]
            h = project_to_set(h, target)
            new_c5 = compute_c5(h, n)
            d = new_c5 - c5
            if d > 0 and rng.random() > np.exp(-d / T):
                h[i:i + blk] = 1.0 - h[i:i + blk]
                h = project_to_set(h, target)
                c5 = compute_c5(h, n)
            else:
                c5 = new_c5
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        T *= cool
    return best_h, best_c5

def subgradient_refine(h, n, n_steps, lr_init=0.08, top_m=5):
    target = n / 2.0
    h = h.copy()
    c5 = compute_c5(h, n)
    for step in range(n_steps):
        corr = np.correlate(h, 1.0 - h, mode="full")
        top_idx = np.argsort(corr)[-top_m:]
        grad = np.zeros(n)
        for k_idx in top_idx:
            k = int(k_idx) - (n - 1)
            for j in range(n):
                s = 0.0
                if 0 <= j + k < n:
                    s += 1.0 - h[j + k]
                if 0 <= j - k < n:
                    s -= h[j - k]
                grad[j] += s
        grad /= float(top_m)
        lr = lr_init * (1.0 / (1.0 + 0.01 * step))
        h_new = project_to_set(h - lr * grad, target)
        c5_new = compute_c5(h_new, n)
        if c5_new < c5:
            h = h_new
            c5 = c5_new
        else:
            lr_init *= 0.7
    return h, c5
