import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def repair(h, target):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0 - 1e-13
        else:
            m = h > 1e-13
        c = int(m.sum())
        if c == 0:
            break
        h[m] += d / c
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(h, dx, tau, pen, target):
    hc = np.clip(h, 0.0, 1.0)
    corr = np.correlate(hc, 1.0 - hc, mode="full")
    scaled = corr * dx / tau
    sm = scaled.max() + np.log(np.sum(np.exp(scaled - scaled.max())))
    obj = sm * tau
    s = np.sum(hc)
    obj += pen * (s - target) ** 2
    obj += pen * 0.01 * (np.sum(np.clip(-hc, 0, None) ** 2) + np.sum(np.clip(hc - 1.0, 0, None) ** 2))
    return obj

def sum_sq_obj(h, dx, pen, target):

    obj = np.sum(corr ** 2)
    s = np.sum(hc)
    obj += pen * (s - target) ** 2
