    return np.correlate(h, 1.0 - h, mode='full') * dx

def _grad_v_component(h, s, n, dx):
    """Gradient of v at shift s (negative = left shift) w.r.t. h."""
    g = np.zeros(n)
    if s >= 0:
        # j-s valid: j in [s, n-1]
        j1 = np.arange(s, n)
        g[j1] = dx * (1.0 - h[j1 - s])
        # j+s valid: j in [0, n-1-s]
        if n - s > 0:
            j2 = np.arange(0, n - s)
            g[j2] -= dx * h[j2 + s]
    else:
        sp = -s
        # j-s = j+sp valid: j in [0, n-1-sp]
        if n - sp > 0:
            j1 = np.arange(0, n - sp)
            g[j1] = dx * (1.0 - h[j1 + sp])
        # j+s = j-sp valid: j in [sp, n-1]
        j2 = np.arange(sp, n)
        g[j2] -= dx * h[j2 - sp]
    return g

def _obj_grad(h, alpha):
    n = len(h)
    v = _compute_v(h)
    m = np.max(v)
    exp_v = np.exp(np.clip(v - m, -70.0, 70.0))
    sum_exp = np.sum(exp_v)
    w = exp_v / sum_exp
    obj = m + np.log(sum_exp) / alpha
    
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(len(v)):
        wk = w[k] / alpha
        if wk < 1e-13:
            continue
        s = k - (n - 1)
        grad += wk * _grad_v_component(h, s, n, dx)
    
    return obj, grad

    """Fix integral constraint: sum(h) should be n/2."""
    target = n / 2.0
    for _ in range(200):
        d = target - np.sum(h)
        if abs(d) < 1e-12:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _interp_to(h, n_target):

    x_src = np.arange(n_src) * (2.0 / n_src)
    x_tgt = np.arange(n_target) * (2.0 / n_target)
    h_new = np.interp(x_tgt, x_src, h)
    return _fix_int(np.clip(h_new, 0, 1), n_target)

        if res.x is None:
            return None
        return _fix_int(np.clip(res.x, 0, 1), n)
    except Exception:
        return None
