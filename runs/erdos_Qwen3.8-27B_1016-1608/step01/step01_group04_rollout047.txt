        hp = repair(hp, n_use)
        hn = do_slsqp(hp, n_use, 0.003, 250, 1e-13)
        c = compute_c5(hn)
        if c < best_c:
            best_c, best_h = c, hn

    # Phase 4: Try n=400 if time allows and not already there
    if best_n < 400 and time.time() - t0 < budget_s - 250:
        n4 = 400
        x1 = np.linspace(0.0, 2.0, best_n, endpoint=False)
        x4 = np.linspace(0.0, 2.0, n4, endpoint=False)
        h4 = np.interp(x4, x1, best_h)
        h4 = repair(h4, n4)
        for eps in [0.005, 0.002, 0.001, 0.0005]:
            if time.time() - t0 > budget_s - 150:
                break
            hn4 = do_slsqp(h4, n4, eps, 250, 1e-13)
            c4 = compute_c5(hn4)
            if c4 < best_c:
                best_c, best_h, best_n = c4, hn4, n4

    # Phase 5: Final fine polish at best resolution
    if time.time() - t0 < budget_s - 60:
        for eps in [0.0001, 0.00005, 0.00002, 0.00001]:
            if time.time() - t0 > budget_s - 25:
                break
            hn = do_slsqp(best_h, best_n, eps, 600, 1e-15)
            c = compute_c5(hn)
            if c < best_c:
                best_c, best_h = c, hn

    # Phase 6: Tiny random perturbations
    if time.time() - t0 < budget_s - 10:
        nsteps = 8000
        for _ in range(nsteps):
            if time.time() - t0 > budget_s - 5:
                break
            sigma = 0.002 + 0.003 * max(0.0, 1.0 - (_ / nsteps))
            hp = best_h + rng.randn(best_n) * sigma
            hp = repair(hp, best_n)
            c = compute_c5(hp)
            if c < best_c:
                best_c, best_h = c, hp

    fh = repair(best_h, best_n)
    fc = compute_c5(fh)
    return (fh.tolist(), float(fc), best_n)
