import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def project_constraint(y, target):
    lo, hi = -float(y.max()) - 2.0, 1.0 - float(y.min()) + 2.0
    for _ in range(64):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(y + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(y + 0.5 * (lo + hi), 0.0, 1.0)

def grad_smooth(h, eps, dx):
    C = np.correlate(h, 1.0 - h, mode='full') * dx
    Cn = C - C.max()
    w = np.exp(Cn / eps)
    w /= w.sum()
    g = 1.0 - 2.0 * h
    grad = dx * np.correlate(w, g, mode='valid')
    return grad, C

def pgd_refine(h, n, target, dx, eps0, n_steps, lr_base, deadline):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h_cur = h.copy()
    v = np.zeros(n)
    for i in range(n_steps):
        if time.time() > deadline:
            break
        frac = (i + 1) / n_steps
        eps = eps0 * (0.02 ** frac)
        lr = lr_base * (0.6 + 0.4 * (1.0 - frac))
        grad, C = grad_smooth(h_cur, eps, dx)
        v = 0.8 * v - lr * grad
        h_new = project_constraint(h_cur + v, target)
        c5_check = compute_c5(h_new)
        if c5_check < best_c5:
            best_c5 = c5_check
            best_h = h_new.copy()
        h_cur = h_new
    return best_h, best_c5

def slsqp_polish(h, n, eps, maxiter, ftol):
    dx = 2.0 / n
    target = n / 2.0
    bds = [(0.0, 1.0)] * n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        cn = np.clip((c - m) / eps, -60.0, 60.0)
        return m + eps * np.log(np.sum(np.exp(cn)))
    def obj_grad(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        cn = np.clip((c - m) / eps, -60.0, 60.0)
        w = np.exp(cn); w /= w.sum()
        g = 1.0 - 2.0 * x
        return dx * np.correlate(w, g, mode='valid')
    def con_fun(x):
        return x.sum() - target
    def con_jac(x):
        return np.ones(n)
    r = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bds,
                 constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                 options={'maxiter': maxiter, 'ftol': ftol})
    return project_constraint(r.x, target)

def lbfgsb_opt(h, n, target, dx, smooth_eps, lam, maxiter):
    bds = [(0.0, 1.0)] * n
    def obj(x):
        C = np.correlate(x, 1.0 - x, mode='full') * dx
        m = C.max()
        cn = np.clip((C - m) / smooth_eps, -60.0, 60.0)
        sm = m + smooth_eps * np.log(np.sum(np.exp(cn)))
        return sm + lam * (x.sum() - target) ** 2
    def obj_grad(x):
        C = np.correlate(x, 1.0 - x, mode='full') * dx
        m = C.max()
        cn = np.clip((C - m) / smooth_eps, -60.0, 60.0)
        w = np.exp(cn); w /= w.sum()
        g = 1.0 - 2.0 * x
        gb = dx * np.correlate(w, g, mode='valid')
        gp = 2.0 * lam * (x.sum() - target) * np.ones(n)
        return gb + gp
    r = minimize(obj, h, jac=obj_grad, method='L-BFGS-B', bounds=bds,
                 options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12})
    return project_constraint(r.x, target)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline_main = t0 + budget_s - 250
    deadline_late = t0 + budget_s - 60
    deadline_final = t0 + budget_s - 5

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = 0.5 * np.ones(80)
    n = len(h0)
    target = n / 2
    dx = 2.0 / n
    h0 = project_constraint(h0, target)
    bh = h0.copy()
    bc = compute_c5(bh)

    # Phase 1: PGD with annealing from parent
    for eps0 in [0.08, 0.04, 0.02, 0.01, 0.005]:
        if time.time() > deadline_main:
            break
        hn, cn = pgd_refine(bh, n, target, dx, eps0, 3000, 1.5, deadline_main)
        if cn < bc:
            bh, bc = hn, cn

    # Phase 2: L-BFGS-B with penalty
    if time.time() < deadline_main:
        for lam in [1.0, 10.0, 100.0]:
            if time.time() > deadline_main:
                break
            hn = lbfgsb_opt(bh, n, target, dx, 0.005, lam, 200)
            cn = compute_c5(hn)
            if cn < bc:
                bh, bc = hn, cn

    # Phase 3: SLSQP with analytical gradient
    if time.time() < deadline_late:
        hn = slsqp_polish(bh, n, 0.003, 300, 1e-13)
        cn = compute_c5(hn)
        if cn < bc:
            bh, bc = hn, cn

    # Phase 4: Structured restarts
    for p in [3, 4, 5, 6, 7, 8, 10, 12, 13, 16, 20]:
        if time.time() > deadline_late:
            break
        hp = np.zeros(n)
        for i in range(n):
            hp[i] = 1.0 if (i % p) < p // 2 else 0.0
        hp = project_constraint(hp, target)
        hn, cn = pgd_refine(hp, n, target, dx, 0.03, 800, 1.2, deadline_late)
        if cn < bc:
            bh, bc = hn, cn
        if time.time() < deadline_late - 20:
            hn2 = slsqp_polish(hp, n, 0.002, 150, 1e-12)
            cn2 = compute_c5(hn2)
            if cn2 < bc:
                bh, bc = hn2, cn2

    # Phase 5: Sinusoidal / structured starts
    for k in [1, 2, 3, 4, 5, 6]:
        if time.time() > deadline_late:
            break
        hp = 0.5 + 0.45 * np.sin(2.0 * np.pi * k * np.arange(n) / n)
        hp = project_constraint(hp, target)
        hn, cn = pgd_refine(hp, n, target, dx, 0.02, 600, 1.0, deadline_late)
        if cn < bc:
            bh, bc = hn, cn

    # Phase 6: Random restarts around best
    for i in range(300):
        if time.time() > deadline_final:
            break
        r = i % 6
        if r == 0:
            hp = bh + np.random.randn(n) * 0.04
        elif r == 1:
            hp = bh + np.random.randn(n) * 0.08
        elif r == 2:
            hp = bh.copy()
            s = np.random.randint(0, max(1, n - 8))
            l = np.random.randint(3, min(9, n - s))
            hp[s:s + l] = 1.0 - hp[s:s + l]
        elif r == 3:
            k2 = np.random.randint(1, 7)
            ph = np.random.uniform(0, 2 * np.pi)
            hp = bh + 0.04 * np.sin(2 * np.pi * k2 * np.arange(n) / n + ph)
        elif r == 4:
            hp = bh.copy()
            idx = np.random.choice(n, size=max(2, n // 4), replace=False)
            hp[idx] = 1.0 - hp[idx]
        else:
            hp = bh + np.random.randn(n) * 0.015
        hp = project_constraint(hp, target)
        hn, cn = pgd_refine(hp, n, target, dx, 0.015, 400, 1.0, deadline_final)
        if cn < bc:
            bh, bc = hn, cn
        if cn < bc + 1e-5 and time.time() < deadline_final - 30:
            hn2 = slsqp_polish(hn, n, 0.001, 200, 1e-14)
            cn2 = compute_c5(hn2)
            if cn2 < bc:
                bh, bc = hn2, cn2

    # Final polish
    if time.time() < deadline_final:
        hn = slsqp_polish(bh, n, 0.0005, 400, 1e-14)
        cn = compute_c5(hn)
        if cn < bc:
            bh, bc = hn, cn
        hn = lbfgsb_opt(bh, n, target, dx, 0.0005, 100.0, 200)
        cn = compute_c5(hn)
        if cn < bc:
            bh, bc = hn, cn

    fh = project_constraint(bh, target)
    fc = compute_c5(fh)
    return (fh.tolist(), float(fc), n)
