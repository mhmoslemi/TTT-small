def _eq_jac(x):
    return np.ones(_opt['n'])

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def get_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        d = n / 2.0 - np.sum(h)
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0 - 1e-10
        else:
            m = h > 1e-10
        cnt = int(m.sum())
        if cnt == 0:
            h += d / n
        else:
            h[m] += d / cnt
    return np.clip(h, 0.0, 1.0)

def do_slsqp(h, n, eps, maxiter):
    _opt['n'] = n
    _opt['eps'] = eps
    r = minimize(_smax_obj, h, method='SLSQP',
                 bounds=[(0.0, 1.0)] * n,
                 constraints=[{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}],
                 options={'maxiter': maxiter, 'ftol': 1e-12})
    return project_feasible(r.x, n)

def subgrad_step(h, n, step, bucket):
    c = get_corr(h)
    M = c.max()
    active = c >= M - bucket
    na = max(1, int(active.sum()))
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if not active[k]:
            continue
        m = k - (n - 1)
        lo = max(0, -m)
        hi = min(n, n - m)
        for i in range(lo, hi):
            j1 = i + m
            g[j1] += 1.0 - h[i]
            g[i] -= h[j1]
    g /= na
    return project_feasible(h - step * g, n)

def binary_round(h, n):
    h_b = np.where(h > 0.5, 1.0, 0.0)
    d = n / 2.0 - np.sum(h_b)
    if d > 0:
        order = np.argsort(h)[::-1]
        for idx in order[:int(np.ceil(d))]:
            h_b[idx] = 1.0
    elif d < 0:
        order = np.argsort(h)
        for idx in order[:int(np.ceil(-d))]:
            h_b[idx] = 0.0
    return project_feasible(h_b, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    try:
        h = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h = 0.5 + 0.3 * np.sin(2.0 * np.pi * np.arange(80) / 80)
    n = len(h)
    h = project_feasible(h, n)

    best_h = h.copy()
    best_c = compute_c5(best_h)

    # Phase 1: Multi-eps SLSQP
    for eps in [0.02, 0.008, 0.003, 0.001, 0.0005, 0.0002]:
        if time.time() - t0 > budget_s * 0.25:
            break
        h_new = do_slsqp(h, n, eps, 300)
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c, best_h = c_new, h_new
            h = h_new.copy()

    # Phase 2: Subgradient with adaptive schedule
    h_work = best_h.copy()
    no_improve = 0
    for it in range(12000):
        if time.time() - t0 > budget_s - budget_s * 0.3:
            break
        step = 0.08 / np.sqrt(1.0 + it * 0.01)
        bucket = 0.001 if it % 300 < 200 else 0.005
        h_new = subgrad_step(h_work, n, step, bucket)
        c_new = compute_c5(h_new)
        if c_new < best_c - 1e-13:
            best_c, best_h = c_new, h_new
            h_work = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        h_work = h_new

        if it % 800 == 799:
            h_pol = do_slsqp(h_work, n, 0.001, 150)
            c_pol = compute_c5(h_pol)
            if c_pol < best_c:
                best_c, best_h = c_pol, h_pol
                h_work = h_pol.copy()
                no_improve = 0

        if it % 400 == 399:
            h_work = project_feasible(h_work + np.random.randn(n) * 0.04, n)

    # Phase 3: Coordinate descent with binary targets
    h = best_h.copy()
    for it in range(6000):
        if time.time() - t0 > budget_s - budget_s * 0.15:
            break
        # Focus on non-extreme coordinates first
        non_ext = np.where((h > 0.05) & (h < 0.95))[0]
        if len(non_ext) > 0 and np.random.random() < 0.7:
            i = non_ext[np.random.randint(len(non_ext))]
        else:
            i = np.random.randint(n)
        targets = [0.0, 1.0, 1.0 - h[i], max(0.0, h[i] - 0.15), min(1.0, h[i] + 0.15)]
        improved = False
        for t in targets:
            h_try = h.copy()
            h_try[i] = t
            h_try = project_feasible(h_try, n)
            c_try = compute_c5(h_try)
            if c_try < best_c - 1e-13:
                best_c, best_h = c_try, h_try
                h = h_try.copy()
                improved = True
                break
        if not improved and it % 300 == 299:
            h_pb = binary_round(h, n)
            c_pb = compute_c5(h_pb)
            if c_pb < best_c:
                best_c, best_h = c_pb, h_pb
                h = h_pb.copy()
            h = project_feasible(h + np.random.randn(n) * 0.03, n)

    # Phase 4: Pair coordinate updates (adjust i and partner for worst lag)
    h = best_h.copy()
    for it in range(2000):
        if time.time() - t0 > budget_s - 40:
            break
        c = get_corr(h)
        M = c.max()
        k_worst = int(np.argmax(c))
        m_lag = k_worst - (n - 1)
        i = np.random.randint(n)
        j = (i + m_lag) % n
        for ti in [0.0, 1.0]:
            for tj in [0.0, 1.0]:
                h_try = h.copy()
                h_try[i] = ti
                h_try[j] = tj
                h_try = project_feasible(h_try, n)
                c_try = compute_c5(h_try)
                if c_try < best_c - 1e-13:
                    best_c, best_h = c_try, h_try
                    h = h_try.copy()
                    break
            else:
                continue
            break

    # Phase 5: Final SLSQP polish
    if time.time() - t0 < budget_s - 20:
        for eps in [0.0005, 0.0002, 0.0001]:
            if time.time() - t0 > budget_s - 10:
                break
            h_new = do_slsqp(best_h, n, eps, 300)
            c_new = compute_c5(h_new)
            if c_new < best_c:
                best_c, best_h = c_new, h_new

    # Phase 6: Micro-perturbation
    for _ in range(8000):
        if time.time() - t0 > budget_s - 3:
            break
        h_try = project_feasible(best_h + np.random.randn(n) * 0.002, n)
        c_try = compute_c5(h_try)
        if c_try < best_c:
            best_c, best_h = c_try, h_try

    return (best_h.tolist(), float(best_c), n)
