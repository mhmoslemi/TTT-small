import numpy as np
from scipy.optimize import minimize
import time

_P = {}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def repair(h, n):
    h = np.array(h, dtype=float).copy()
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        d = n / 2.0 - np.sum(h)
        if abs(d) < 1e-14:
            break
        h += d / n
    return np.clip(h, 0.0, 1.0)

def _obj(x):
    n = _P['n']
    eps = _P['eps']
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(c)
    if eps < 1e-15:
        return float(m)
    s = np.clip((c - m) / eps, -50.0, 50.0)
    return float(m + eps * np.log(np.sum(np.exp(s))))

def _grad(x):
    n = _P['n']
    eps = _P['eps']
    dx = 2.0 / n
    if eps < 1e-15:
        return np.zeros(n)
    raw = np.correlate(x, 1.0 - x, mode='full')
    c = raw * dx
    m = np.max(c)
    s = np.clip((c - m) / eps, -50.0, 50.0)
    exp_s = np.exp(s)
    S = np.sum(exp_s)
    w = exp_s / S
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        d = k - (n - 1)
        lo1 = max(0, d)
        hi1 = min(n, n + d)
        if lo1 < hi1:
            grad[lo1:hi1] += wk * (1.0 - x[lo1 - d:hi1 - d])
        lo2 = max(0, -d)
        hi2 = min(n, n - d)
        if lo2 < hi2:
            grad[lo2:hi2] -= wk * x[lo2 + d:hi2 + d]
    return grad

def _eq(x):
    return np.sum(x) - _P['n'] / 2.0

def _eqj(x):
    return np.ones(_P['n'])

def do_opt(h, n, eps, maxiter, ftol):
    _P['n'] = n
    _P['eps'] = eps
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq, 'jac': _eqj}]
    res = minimize(_obj, h, method='SLSQP', jac=_grad, bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    return repair(res.x, n)

def interp_h(h_src, n_src, n_dst):
    if n_dst == n_src:
        return h_src.copy()
    if n_dst < n_src:
        idx = np.linspace(0, n_src, n_dst, dtype=int)
        return h_src[idx]
    x_old = np.linspace(0, 2.0, n_src, endpoint=False)
    x_new = np.linspace(0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def make_block(h_target, n, p):
    return np.array([1.0 if (i % p) < p // 2 else 0.0 for i in range(n)])

def make_sinus(h_target, n, k, ph, amp):
    return 0.5 + amp * np.sin(2.0 * np.pi * k * np.arange(n) / n + ph)
