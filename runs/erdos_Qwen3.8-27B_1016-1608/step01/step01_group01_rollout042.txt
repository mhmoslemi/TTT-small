def subgrad_shift(h, shift, n):
    dx = 2.0 / n
    grad = np.zeros(n)
    if shift >= 0:
        for i in range(n - shift):
            j = i + shift
            grad[i] += (1.0 - h[j])
            if shift > 0:
                grad[j] -= h[i]

            j = i - s
            grad[i] += (1.0 - h[j])
            grad[j] -= h[i]
    return grad * dx

`np.correlate(a, v, mode='full')` computes the cross-correlation. For two sequences a and v of lengths n each, the result has length 2n-1.

The k-th element (0-indexed) of the full correlation is:
sum_i a[i+k] * v[i] (with the appropriate range of i)

More precisely, for `np.correlate(a, v, 'full')[k]`:
- k ranges from 0 to 2n-2
- Let lag = k - (n-1)
- The sum is over i where both i and i+lag are valid indices
- result[k] = sum_{i: 0≤i<n, 0≤i+lag<n} a[i+lag] * v[i]

So for `np.correlate(h, 1-h, 'full')[k]` with lag = k-(n-1):
= sum_{valid i} h[i+lag] * (1-h[i])

This is sum_{valid i} h[i+lag] - sum_{valid i} h[i+lag]*h[i]

Which is the same as sum_{valid j} h[j] - sum_{valid j} h[j]*h[j-lag] (with j = i+lag)

= sum_{valid j} h[j]*(1-h[j-lag])

So the correlation at lag s gives: sum h[i]*(1-h[i+s]) (where valid)

The C₅ = max over all lags of this sum * dx.

def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)

def corr_vals(h):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / len(h))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = m.sum()
        if cnt == 0:
            m = (h < 1.0) if d > 0 else (h > 0.0)
            cnt = m.sum()
        if cnt == 0:
            break
        h[m] = np.clip(h[m] + d / cnt, 0.0, 1.0)
    return h

def smax_grad(h, n, lam):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    w = np.exp((c - c.max()) * lam)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(2 * n - 1)
        w[np.argmax(c)] = 1.0
    else:
        w /= ws
    g = np.zeros(n)
    for s in range(-(n - 1), n):
        wi = w[s + n - 1]
        if wi < 1e-12:
            continue
        if s >= 0:
            ii = np.arange(n - s)
            jj = ii + s
        else:
            jj = np.arange(n + s)
            ii = jj - s
        g[ii] += wi * (1.0 - h[jj]) * dx
        g[jj] -= wi * h[ii] * dx
    return g, c

    c5 = compute_c5(h)

    return x.sum() - len(x) / 2.0
