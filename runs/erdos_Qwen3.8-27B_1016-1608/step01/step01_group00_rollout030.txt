```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_active_lags(h, tol=1e-8):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    active = np.where(corr >= max_val - tol * max_val)[0]
    # Convert to lags
    lags = active - (n - 1)
    return lags

def compute_subgradient(h, active_lags):
    n = len(h)
    grad = np.zeros(n)
    for L in active_lags:
        L2 = abs(int(L))
        if L2 == 0:
            # c_0 = sum h[i]*(1-h[i]) = sum h[i] - sum h[i]^2
            # ∂/∂h[j] = 1 - 2*h[j]
            grad[:] += (1 - 2 * h)
        else:
            # ∂c/∂h[j] = (1-h[j-L2]) * I[j >= L2] - h[j+L2] * I[j <= n-1-L2]
            mask1 = np.arange(n) >= L2
            mask2 = np.arange(n) <= n - 1 - L2
            grad[mask1] += (1 - h[mask1 - L2])
            grad[mask2] -= h[mask2 + L2]
    grad /= len(active_lags)
    return grad

def repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not np.any(mask):
            h[:] = target / n
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_descent(h, n, step_size, n_steps, noise_scale=0.0):
    h = h.copy()
    for step in range(n_steps):
        active = get_active_lags(h, tol=1e-6)
        grad = compute_subgradient(h, active)
        dx = 2.0 / n
        grad *= dx
        h_new = h - step_size * grad
        if noise_scale > 0:
            h_new += noise_scale * np.random.randn(n)
        h = repair(h_new, n)
    return h

def slsqp_refine(h, n, maxiter=500):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0}]
    try:
        res = minimize(compute_c5, h.copy(), method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.fun < compute_c5(h) + 1e-12:
            return repair(np.clip(res.x, 0.0, 1.0), n)
    except:
        pass
    return h
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_weights(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    shifted = beta * corr
    mx = np.max(shifted)
    e = np.exp(shifted - mx)
    s = np.sum(e)
    if s < 1e-300:
        return np.full(len(corr), 1.0 / len(corr))
    return e / s

def weighted_gradient(h, weights):
    n = len(h)
    idx = np.arange(n)
    grad = np.zeros(n)
    for k in range(len(weights)):
        w = weights[k]
        if w < 1e-9:
            continue
        L2 = abs(k - (n - 1))
        j1 = idx[idx >= L2]
        if len(j1) > 0:
            grad[j1] += w * (1.0 - h[j1 - L2])
        j2 = idx[idx <= n - 1 - L2]
        if len(j2) > 0:
            grad[j2] -= w * h[j2 + L2]
    return grad

def repair(h, n):
    target = float(n) / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not np.any(mask):
            h[:] = target / n
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_gradient_descent(h, n, max_steps=10000, beta_init=10.0, beta_end=200.0,
                            lr=0.04, noise_scale=0.02):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    for step in range(max_steps):
        frac = step / max(max_steps - 1, 1)
        beta = beta_init * (beta_end / beta_init) ** frac
        w = get_weights(h, beta)
        g = weighted_gradient(h, w)
        lr_adapt = lr * (1.0 + 4.0 * (1.0 - frac))
        h_new = h - lr_adapt * g
        if noise_scale > 0:
            h_new += noise_scale * (1.0 - frac) * np.random.randn(n)
        h_new = repair(h_new, n)
        if step % 50 == 0 or step == max_steps - 1:
            c = compute_c5(h_new)
            if c < best_c - 1e-14:
                best_h = h_new.copy()
                best_c = c
        h = h_new
    c_final = compute_c5(h)
    if c_final < best_c:
        best_h = h.copy()
        best_c = c_final
    return best_h, best_c

def slsqp_refine(h, n, maxiter=500):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}]
    try:
        res = minimize(compute_c5, h.copy(), method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        c0 = compute_c5(h)
        if res.fun < c0 + 1e-12:
            return repair(np.clip(res.x, 0.0, 1.0), n)
    except Exception:
        pass
    return h

def gen_binary_starts(n, rng, count):
    starts = []
    target = n // 2
    tm = np.zeros(n, dtype=int)
    for i in range(1, n):
        tm[i] = tm[i // 2] ^ (i % 2)
    tm_f = tm.astype(float)
    n_ones = int(np.sum(tm_f))
    if n_ones < target:
        zeros = np.where(tm_f == 0)[0]
        flips = rng.choice(zeros, min(target - n_ones, len(zeros)), replace=False)
        tm_f[flips] = 1.0
    elif n_ones > target:
        ones = np.where(tm_f == 1)[0]
        flips = rng.choice(ones, min(n_ones - target, len(ones)), replace=False)
        tm_f[flips] = 0.0
    starts.append(repair(tm_f, n))
    alt = np.zeros(n)
    alt[::2] = 1.0
    if int(np.sum(alt)) > target:
        alt[1::2] = 0.0
    starts.append(repair(alt, n))
    for _ in range(max(0, count - 2)):
        rb = np.zeros(n)
        if target > 0:
            rb[rng.choice(n, target, replace=False)] = 1.0
        starts.append(repair(rb, n))
    return starts

        if c_c < c5_best:
```
