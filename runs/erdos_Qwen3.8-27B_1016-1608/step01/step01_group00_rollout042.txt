import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    """Project onto {0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    h = h.copy()
    # Clip to [0,1] first
    h = np.clip(h, 0.0, 1.0)
    # Find t such that sum(clamp(h - t, 0, 1)) = target
    # sum(clamp(h-t,0,1)) is decreasing in t
    def sum_clamp(t):
        return np.sum(np.clip(h - t, 0.0, 1.0))
    
    # Bisection for t
    # At t = -1: sum = min(n, sum(h) + n) = n (all at 1)
    # At t = 1: sum = max(0, sum(h) - n) = 0 (all at 0)
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = sum_clamp(mid)
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    return np.clip(h - t, 0.0, 1.0)

def compute_grad_for_shift(h, s, n):
    """Compute gradient of correlation term for shift s."""
    # c = sum_{i: 0<=i<n, 0<=i+s<n} h[i] * (1-h[i+s])
    # d/dh[j] = (1-h[j+s]) * I(0<=j+s<n) - h[j-s] * I(0<=j-s<n)
    grad = np.zeros(n)
    valid_pos = (s >= 0)
    if s >= 0:
        # grad[j] = (1-h[j+s]) for j=0..n-1-s, minus h[j-s] for j=s..n-1
        grad[:n-s] += (1.0 - h[s:n])
        grad[s:] -= h[:n-s]
    else:
        ss = -s
        # grad[j] = (1-h[j+s]) for j=ss..n-1, minus h[j-s] for j=0..n-1-ss
        grad[ss:] += (1.0 - h[:n-ss])
        grad[:n-ss] -= h[ss:]
    return grad

def subgradient_h(h0, n, n_iter=10000, t0_time=60):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    t_start = time.time()
    
    # Compute correlation vector
    dx = 2.0 / n
    
    for it in range(n_iter):
        if time.time() - t_start > t0_time:
            break
        # Compute all correlation terms
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        c_val = c_max * dx
        
        if c_val < best_c:
            best_c = c_val
            best_h = h.copy()
        
        # Active set: terms within 10% of max (or all for simplicity)
        # For speed, use just the argmax and a few neighbors
        threshold = c_max - 0.01 * c_max
        active = np.where(corr >= threshold)[0]

        subgrad = np.zeros(n)
        for m in active:
            s = m - n + 1  # shift
            subgrad += compute_grad_for_shift(h, s, n)
        subgrad /= len(active)
        
        # Step size: alpha_t = c / sqrt(t)
        alpha = 0.5 / (1 + it * 0.01)
        
        h_new = h - alpha * subgrad
        h = project(h_new, n)
    
    return best_h, best_c

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2} using bisection on shift."""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Find t: sum(clip(h - t, 0, 1)) = target
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) * 0.5
    return np.clip(h - t, 0.0, 1.0)

def grad_shift(h, s, n):
    """Gradient of corr term at shift s w.r.t. h."""
    grad = np.zeros(n)
    if s >= 0:
        if s == 0:
            grad = 1.0 - 2.0 * h
        else:
            grad[:n-s] += (1.0 - h[s:])
            grad[s:] -= h[:n-s]
    else:
        ss = -s
        grad[ss:] += (1.0 - h[:n-ss])
        grad[:n-ss] -= h[ss:]
    return grad

    return np.clip(h - t, 0.0, 1.0)

def grad_shift(h, s, n):
    if s == 0:
        return 1.0 - 2.0 * h
    grad = np.zeros(n)
    if s > 0:
        grad[:n - s] += (1.0 - h[s:])
        grad[s:] -= h[:n - s]
    else:
        ss = -s
        grad[ss:] += (1.0 - h[:n - ss])
        grad[:n - ss] -= h[ss:]
    return grad

def smooth_grad(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    g = corr * dx
    gs = g - np.max(g)
    w = np.exp(beta * gs)
    wsum = w.sum()
    if wsum < 1e-300:
        w = np.zeros_like(g)
        w[np.argmax(g)] = 1.0
    else:
        w /= wsum
    active = np.where(w > 1e-10)[0]
    grad = np.zeros(n)
    for m in active:
        s = m - n + 1
        grad += w[m] * grad_shift(h, s, n)
    return grad * dx

def pgd_smooth(h0, n, beta, t_budget):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    t_start = time.time()
    alpha = 0.8 / (beta * 1.5)
    stall = 0
    while time.time() - t_start < t_budget:
        g = smooth_grad(h, n, beta)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        c_cur = compute_c5(h)
        moved = False
        for _ in range(30):
            h_new = project_feasible(h - alpha * g, n)
            c_new = compute_c5(h_new)
            if c_new < c_cur * (1.0 - 1e-10):
                h = h_new
                moved = True
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
                    stall = 0
                break
            alpha *= 0.5
        if not moved:
            stall += 1
            if stall > 40:
                alpha = 0.8 / (beta * 1.5)
                h = project_feasible(h + np.random.randn(n) * 0.008, n)
                stall = 0
        alpha = min(alpha * 1.03, 2.0)
    return best_h, best_c

def slsqp_smooth(h0, n, beta, maxiter=300):
    dx = 2.0 / n

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        g = corr * dx
        gs = g - g.max()
        return g.max() + np.log(np.sum(np.exp(beta * gs))) / beta

    def obj_jac(x):
        return smooth_grad(x, n, beta)

    def sum_fun(x):
        return np.sum(x) - n / 2.0

    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_fun}]
    try:
        res = minimize(obj, h0, jac=obj_jac, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        hx = project_feasible(np.clip(res.x, 0.0, 1.0), n)
        if compute_c5(hx) < compute_c5(h0) + 1e-12:
            return hx
    except Exception:
        pass
    return h0

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_best = np.full(70, 0.5)
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: Beta-annealed projected gradient
    for beta in [3.0, 5.0, 10.0, 20.0, 50.0, 100.0, 200.0]:
        elapsed = time.time() - t0
        remaining = budget_s * 0.42 - elapsed
        if remaining < 3.0:
            break
        h, c = pgd_smooth(h_best, n, beta, min(remaining, 35.0))
        if c < c5_best:
            h_best, c5_best = h, c

    # Phase 2: SLSQP on smoothed objective
    for beta in [20.0, 100.0, 400.0]:
        if time.time() - t0 > budget_s * 0.62:
            break
        h = slsqp_smooth(h_best, n, beta, 350)
        c = compute_c5(h)
        if c < c5_best:
            h_best, c5_best = h, c

    # Phase 3: Multi-scale perturbation + re-optimize
    scales = [0.003, 0.007, 0.012, 0.025, 0.05, 0.1, 0.2]
    idx = 0
    while time.time() - t0 < budget_s * 0.86:
        scale = scales[idx % len(scales)]
        h_p = project_feasible(h_best + np.random.randn(n) * scale, n)
        rem1 = budget_s * 0.86 - (time.time() - t0)
        if rem1 < 2.0:
            break
        h, c = pgd_smooth(h_p, n, 50.0, min(6.0, rem1 * 0.4))
        if c < c5_best:
            h_best, c5_best = h, c
            h = slsqp_smooth(h_best, n, 200.0, 250)
            c = compute_c5(h)
            if c < c5_best:
                h_best, c5_best = h, c
        idx += 1

    # Phase 4: SLSQP on raw (non-smooth) objective
    if time.time() - t0 < budget_s * 0.94:
        bnds = [(0.0, 1.0)] * n

        def sf(x):
            return np.sum(x) - n / 2.0

        cons = [{'type': 'eq', 'fun': sf}]
        try:
            res = minimize(compute_c5, h_best, method='SLSQP', bounds=bnds,
                           constraints=cons, options={'maxiter': 1500, 'ftol': 1e-15})
            h = project_feasible(np.clip(res.x, 0.0, 1.0), n)
            c = compute_c5(h)
            if c < c5_best:
                h_best, c5_best = h, c
        except Exception:
            pass

    # Phase 5: Resolution exploration
    if time.time() - t0 < budget_s * 0.99:
        for n2 in [140, 100, 50]:
            if time.time() - t0 > budget_s * 0.985:
                break
            xp = np.linspace(0.0, 1.0, n)
            xn = np.linspace(0.0, 1.0, n2)
            h_i = np.interp(xn, xp, h_best)
            h_i = project_feasible(h_i, n2)
            rem2 = budget_s * 0.985 - (time.time() - t0)
            if rem2 < 4.0:
                break
            h_i, _ = pgd_smooth(h_i, n2, 30.0, min(6.0, rem2 * 0.4))
            h_i = slsqp_smooth(h_i, n2, 150.0, 200)
            c2 = compute_c5(h_i)
            if c2 < c5_best:
                h_best, c5_best, n = h_i, c2, n2

    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)
    return (h_best, c5_best, n)
