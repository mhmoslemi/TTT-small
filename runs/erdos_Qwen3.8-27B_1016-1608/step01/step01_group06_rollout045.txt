import numpy as np
import time
from scipy.optimize import minimize

_mod_n = 0
_mod_tau = 200.0

def compute_score(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(c)

def compute_shifts(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def soft_max_score(h, tau=200.0):
    c = compute_shifts(h)
    cm = np.max(c)
    shifted = tau * (c - cm)
    shifted = np.clip(shifted, -500, 500)
    return cm + np.log(np.sum(np.exp(shifted))) / tau

def _obj_softmax(h):
    return soft_max_score(h, _mod_tau)

def _constraint_sum(h):
    return np.sum(h) - _mod_n / 2.0

def _penalty_obj(h):
    return compute_score(h) + 500.0 * (np.sum(h) - _mod_n / 2.0) ** 2

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(600):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h = np.clip(h + step * mask, 0.0, 1.0)
    return h

def antisym_project(h, n):
    if n % 2 != 0:
        return project_feasible(h, n)
    half = n // 2
    h2 = np.empty(n)
    for i in range(half):
        avg = (h[i] + (1.0 - h[i + half])) / 2.0
        h2[i] = np.clip(avg, 0.0, 1.0)
        h2[i + half] = np.clip(1.0 - avg, 0.0, 1.0)
    return project_feasible(h2, n)

def coord_descent(h, n, max_sweeps=30, time_limit=None, t_start=None):
    h = h.copy()
    dx = 2.0 / n
    target = n / 2.0
    prev = compute_score(h)
    deltas = [0.02, -0.02, 0.01, -0.01, 0.005, -0.005]
    for sweep in range(max_sweeps):
        if time_limit is not None and time.time() - t_start > time_limit:
            break
        improved = False
        for j in range(n):
            if time_limit is not None and time.time() - t_start > time_limit:
                break
            for d in deltas:
                new_v = h[j] + d
                if new_v < 0 or new_v > 1:
                    continue
                h2 = h.copy()
                h2[j] = new_v
                diff = target - np.sum(h2)
                if abs(diff) < 1e-13:
                    pass
                elif diff > 0:
                    mask = h2 < 1.0
                    mask[j] = False
                else:
                    mask = h2 > 0.0
                    mask[j] = False
                if not np.any(mask):
                    continue
                if abs(diff) > 1e-13:
                    m = int(np.argmax(np.where(mask, 1.0 - h2 if diff > 0 else h2, -1.0)))
                    if not mask[m]:
                        continue
                    adj = diff / np.sum(mask)
                    h2 = np.clip(h2 + adj * mask, 0.0, 1.0)
                if abs(np.sum(h2) - target) > 1e-10:
                    continue
                s2 = compute_score(h2)
                if s2 < prev - 1e-12:
                    h = h2
                    prev = s2
                    improved = True
                    break
        if not improved:
            break
    return h, prev

def run(seed=42, budget_s=1100, **kwargs):
    global _mod_n, _mod_tau
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    best_h = h_init.copy()
    best_score = compute_score(h_init)

    # Try anti-symmetric version
    h_as = antisym_project(h_init, n_init)
    s_as = compute_score(h_as)
    if s_as < best_score:
        best_score = s_as
        best_h = h_as.copy()

    # Multi-resolution optimization
    n_list = [n_init, 64, 80, 100, 128]
    x_ref = np.linspace(0.0, 2.0, len(best_h), endpoint=False)

    for n_pts in n_list:
        if time.time() - t0 > budget_s * 0.82:
            break
        _mod_n = n_pts
        bounds = [(0.0, 1.0)] * n_pts
        cons = [{'type': 'eq', 'fun': _constraint_sum}]
        x_new = np.linspace(0.0, 2.0, n_pts, endpoint=False)

        # Build candidate starts
        starts = []
        h_interp = project_feasible(np.interp(x_new, x_ref, best_h), n_pts)
        starts.append(h_interp)
        starts.append(antisym_project(h_interp, n_pts))

        # Random structured starts
        for _ in range(4):
            hr = project_feasible(rng.uniform(0, 1, n_pts), n_pts)
            starts.append(hr)
        # Anti-symmetric random
        for _ in range(3):
            half = n_pts // 2
            ha = rng.uniform(0, 1, half)
            hf = np.empty(n_pts)
            hf[:half] = ha
            hf[half:] = 1.0 - ha
            hf = project_feasible(hf, n_pts)
            starts.append(hf)
        # Bimodal-ish start
        hb = np.zeros(n_pts)
        active = rng.choice(n_pts, size=n_pts // 2, replace=False)
        hb[active] = 1.0
        hb = project_feasible(hb + rng.randn(n_pts) * 0.15, n_pts)
        starts.append(hb)

        local_best = h_interp.copy()
        local_best_s = compute_score(local_best)

        for h_start in starts:
            if time.time() - t0 > budget_s * 0.84:
                break
            s_start = compute_score(h_start)
            if s_start < best_score:
                best_score = s_start
                best_h = h_start.copy()
            if s_start < local_best_s:
                local_best_s = s_start
                local_best = h_start.copy()

            # SLSQP with soft-max, varying tau
            for tau_val in [150, 400, 1200]:
                if time.time() - t0 > budget_s * 0.85:
                    break
                _mod_tau = tau_val
                try:

                                   bounds=bounds, constraints=cons,

                    hc = project_feasible(res['x'], n_pts)

                        best_h = hc.copy()
                    if sc < local_best_s:
                        local_best_s = sc
                        local_best = hc.copy()
                        h_start = hc.copy()
                except Exception:
                    pass

            # Exact SLSQP
            if time.time() - t0 < budget_s * 0.85:

                    res = minimize(compute_score, h_start, method='SLSQP',
                                   bounds=bounds, constraints=cons,

                    hc = project_feasible(res['x'], n_pts)
                    sc = compute_score(hc)
                    if sc < best_score:
                        best_score = sc
                        best_h = hc.copy()
                    if sc < local_best_s:
                        local_best_s = sc
                        local_best = hc.copy()
                except Exception:
                    pass

        # Nelder-Mead on local best
        if time.time() - t0 < budget_s * 0.80:
            try:
                res = minimize(_penalty_obj, local_best, method='Nelder-Mead',
                               options={'maxiter': 3000, 'xatol': 1e-7, 'fatol': 1e-11})
                hc = project_feasible(res['x'], n_pts)
                sc = compute_score(hc)
                if sc < best_score:
                    best_score = sc
                    best_h = hc.copy()
            except Exception:
                pass

        # Coordinate descent polish
        if time.time() - t0 < budget_s * 0.78:
            hc, sc = coord_descent(local_best, n_pts, max_sweeps=20,
                                   time_limit=budget_s * 0.78 - t0, t_start=t0)
            if sc < best_score:
                best_score = sc
                best_h = hc.copy()

        x_ref = np.linspace(0.0, 2.0, len(best_h), endpoint=False)

    # Try smaller n for potential simpler structure
    if time.time() - t0 < budget_s * 0.6:
        for n_sm in [32, 40, 48]:
            if time.time() - t0 > budget_s * 0.7:
                break
            _mod_n = n_sm
            bounds = [(0.0, 1.0)] * n_sm
            cons = [{'type': 'eq', 'fun': _constraint_sum}]
            x_sm = np.linspace(0.0, 2.0, n_sm, endpoint=False)
            h_sm = project_feasible(np.interp(x_sm, x_ref, best_h), n_sm)
            for trial in range(3):
                if time.time() - t0 > budget_s * 0.72:
                    break
                h_s = h_sm + (rng.randn(n_sm) * 0.1 if trial > 0 else 0)
                h_s = project_feasible(h_s, n_sm)
                _mod_tau = 300
                try:
                    res = minimize(_obj_softmax, h_s, method='SLSQP',
                                   bounds=bounds, constraints=cons,
                                   options={'maxiter': 200, 'ftol': 1e-12})
                    hc = project_feasible(res['x'], n_sm)
                    sc = compute_score(hc)
                    if sc < best_score:
                        best_score = sc
                        best_h = hc.copy()
                except Exception:
                    pass

    # Final intensive polish on best solution
    n_fin = len(best_h)
    _mod_n = n_fin
    bounds = [(0.0, 1.0)] * n_fin
    cons = [{'type': 'eq', 'fun': _constraint_sum}]

    if time.time() - t0 < budget_s * 0.92:
        # Multiple SLSQP passes with decreasing tau
        h_cur = best_h.copy()
        for tau_val in [1000, 500, 200, 100]:
            if time.time() - t0 > budget_s * 0.93:
                break
            _mod_tau = tau_val
            try:
                res = minimize(_obj_softmax, h_cur, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 400, 'ftol': 1e-14})
                hc = project_feasible(res['x'], n_fin)
                sc = compute_score(hc)
                if sc < best_score:
                    best_score = sc
                    best_h = hc.copy()
                h_cur = hc
            except Exception:
                pass

    if time.time() - t0 < budget_s * 0.96:
        # Exact SLSQP final
        try:
            res = minimize(compute_score, best_h, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 500, 'ftol': 1e-14})
            hc = project_feasible(res['x'], n_fin)
            sc = compute_score(hc)
            if sc < best_score:
                best_score = sc
                best_h = hc
        except Exception:
            pass

    if time.time() - t0 < budget_s * 0.97:
        hc, sc = coord_descent(best_h, n_fin, max_sweeps=15,
                               time_limit=budget_s * 0.03, t_start=t0)
        if sc < best_score:
            best_score = sc
            best_h = hc

    best_h = project_feasible(best_h, len(best_h))
    best_score = compute_score(best_h)
    return best_h, best_score, len(best_h)
