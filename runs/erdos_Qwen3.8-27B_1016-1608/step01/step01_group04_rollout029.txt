import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr)), corr

def repair(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = n / 2.0 - np.sum(h)
        if abs(d) < 1e-15:
            break
        h += d / n
        h = np.clip(h, 0.0, 1.0)
    return h

def _c_s_grad(h, n1, t):
    n = n1 + 1
    g = np.zeros(n)
    if t >= 0:
        ia = np.arange(n - t)
        g[ia] += 1.0 - h[ia + t]
        ib = np.arange(t, n)
        g[ib] -= h[ib - t]
    else:
        tt = -t
        ia = np.arange(tt, n)
        g[ia] += 1.0 - h[ia - tt]
        ib = np.arange(n - tt)
        g[ib] -= h[ib + tt]
    return g

def _slsqp_obj(x, n, dx, eps):
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(c)
    s = np.clip((c - m) / eps, -50.0, 50.0)
    return float(m + eps * np.log(np.sum(np.exp(s))))

def _slsqp_grad(x, n, dx, eps):
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(c)
    z = np.exp(np.clip((c - m) / eps, -50.0, 50.0))
    w = z / np.sum(z)
    n1 = n - 1
    grad = np.zeros(n)
    for s in range(2 * n - 1):
        ws = w[s]
        if ws < 1e-9:
            continue
        t = s - n1
        g = _c_s_grad(x, n1, t)
        grad += ws * dx * g
    return grad

def _con_fun(x, n):
    return np.sum(x) - n / 2.0

def _con_jac(x, n):
    return np.ones(n)

def slsqp_solve(h0, n, eps, maxiter):
    dx = 2.0 / n
    bds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq',
             'fun': lambda x: _con_fun(x, n),
             'jac': lambda x: _con_jac(x, n)}]
    r = minimize(lambda x: _slsqp_obj(x, n, dx, eps),
                 h0,
                 jac=lambda x: _slsqp_grad(x, n, dx, eps),
                 method='SLSQP', bounds=bds, constraints=cons,
                 options={'maxiter': maxiter, 'ftol': 1e-14})
    return repair(r.x, n)

def subgrad_optimize(h0, n, t0, deadline):
    dx = 2.0 / n
    h = repair(h0.copy(), n)
    best_h = h.copy()
    best_c, _ = compute_c5(h)
    base_lr = 0.04
    stall = 0
    for it in range(30000):
        if time.time() > deadline:
            break
        c_cur, corr = compute_c5(h)
        c_max = np.max(corr)
        frac = max(0.02, 0.10 * (1.0 - min(it / 8000.0, 1.0)))
        thresh = c_max * (1.0 - frac)
        S = np.where(corr >= thresh)[0]
        if len(S) == 0:
            S = np.array([np.argmax(corr)])
        n1 = n - 1
        grad = np.zeros(n)
        for s in S:
            t = s - n1
            grad += _c_s_grad(h, n1, t)
        grad *= dx / len(S)
        lr = base_lr
        improved = False

            ht = repair(np.clip(h - lr * grad, 0.0, 1.0), n)

            lr *= 0.78

    best_h = h0.copy()
    best_c, _ = compute_c5(best_h)
    for it in range(800000):

                    break
    return best_h, best_c

    return np.array([(i & (i >> 1)).bit_count() % 2 if hasattr(int, 'bit_count') else bin(i & (i >> 1)).count('1') % 2 for i in range(n)], dtype=float)
