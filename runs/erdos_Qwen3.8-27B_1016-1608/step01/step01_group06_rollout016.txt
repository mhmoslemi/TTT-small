def all_lags(h):

    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:

        else:

        cnt = np.sum(mask)
        if cnt == 0:

        h = np.clip(h + diff / cnt * mask, 0.0, 1.0)
    return h
