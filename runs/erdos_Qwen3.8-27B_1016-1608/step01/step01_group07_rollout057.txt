    _G_A = alpha
    _G_MU = mu

        if r.x is not None:

def _coord_descent(h, n, rng, iters):
    dx = 2.0 / n
    tgt = 0.5 * n
    h = np.clip(h, 0.0, 1.0)
    for it in range(iters):
        order = rng.permutation(n)
        improved = 0
        for j in order:
            c0 = _c5(h)
            best_val = h[j]
            best_c = c0
            for tv in [h[j] - 0.05, h[j] - 0.02, h[j] + 0.02, h[j] + 0.05, 0.0, 1.0]:
                tv = np.clip(tv, 0.0, 1.0)
                if tv == h[j]:
                    continue
                h[j] = tv
                h_other_sum = np.sum(h) - tv
                rem = tgt - h_other_sum
                if rem < -1e-10 or rem > (n - 1) * 1.0:
                    h[j] = best_val
                    continue
                h2 = _fix(h, n)
                c2 = _c5(h2)
                if c2 < best_c - 1e-10:
                    best_c = c2
                    best_val = tv
                h[j] = best_val
                h = _fix(h, n)
            if best_val != h[j]:
                improved += 1
        if improved == 0:

    best_h, best_c = _anneal_slsqp(best_h, n, [100.0, 300.0, 800.0, 2000.0, 5000.0], 2000, t0, budget_s)

        ccd = _c5(hcd)

    for nn in [n + 1, n + 7, n - 13, n + 19]:

        if ci < best_c:

    best_c = _c5(best_h)
    return best_h, best_c, len(best_h)
