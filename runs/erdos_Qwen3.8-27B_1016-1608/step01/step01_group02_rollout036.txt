```python
    h = np.asarray(h, dtype=float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo + hi), 0.0, 1.0)

def logsumexp_max(corr, temp):
    m = corr.max()
    return m + temp * np.log(np.sum(np.exp((corr - m) / temp))) / len(corr)

def make_obj(dx, temp, top_k=None):
    def obj(x):
        c = compute_corr(x)
        if top_k is not None and top_k < len(c):
            c = np.sort(c)[-top_k:]
        return logsumexp_max(c, temp) * dx
    return obj
```

```python
    """Compute gradient of corr[k] w.r.t. h."""
    s = k - (n - 1)  # actual shift
    grad = np.zeros(n)
    if s >= 0:
        # First index: j in [0, n-s)
        if n - s > 0:
            grad[0:n-s] += (1.0 - h[s:n])
        # Second index: j in [s, n)
        if s < n:
            grad[s:n] += -h[0:n-s]

        # Second index: j in [0, n-t)
        if n - t > 0:
            grad[0:n-t] += -h[t:n]
```

Let me refine the gradient step. Since we're minimizing max_k corr[k], a good approach is:

At each step:
1. Compute all correlations
2. Take the top-T correlations (T=3 or 5)
3. Average their gradients (weighted by how close they are to the max)
4. Take a projected gradient step in the negative direction
5. Line search to find the best step size

This is like a "smoothed subgradient" that accounts for the top correlations.

Actually, even better: use the softmax-weighted gradient:
w_k = exp(corr[k]/T) / ∑_j exp(corr[j]/T)
grad = ∑_k w_k * ∂corr[k]/∂h

This is exactly the gradient of the log-sum-exp smooth max! And it's cheap to compute (O(n * K) where K is the number of correlations, which is 2n-1=139, but most weights will be near zero).

```python
def smooth_max_grad(h, n, temp):
    """Compute gradient of log-sum-exp max of correlations."""
    corr = compute_corr(h)  # shape (2n-1,)
    m = corr.max()
    w = np.exp((corr - m) / temp)
    w /= w.sum()
    
    # Compute weighted gradient
    grad = np.zeros(n)
    for k in range(len(corr)):
        if w[k] < 1e-10:
            continue
        s = k - (n - 1)
        if s >= 0:
            if n - s > 0:
                grad[0:n-s] += w[k] * (1.0 - h[s:n])
            if s < n:
                grad[s:n] += w[k] * (-h[0:n-s])
        else:
            t = -s
            if n - t > 0:
                grad[t:n] += w[k] * (1.0 - h[0:n-t])
            if n - t > 0:
                grad[0:n-t] += w[k] * (-h[t:n])
    return grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_h(h, target):
    h = np.asarray(h, dtype=float)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def lse_max(corr, temp):
    m = corr.max()
    return m + temp * np.log(np.sum(np.exp((corr - m) / temp))) / len(corr)

def smooth_grad(h, temp):
    n = len(h)
    corr = compute_corr(h)
    m = corr.max()
    w = np.exp((corr - m) / temp)
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-12:
            continue
        s = k - (n - 1)
        if s >= 0:
            L = n - s
            if L > 0:
                grad[:L] += wk * (1.0 - h[s:])
                grad[s:] += wk * (-h[:L])
        else:
            t = -s
            L = n - t
            if L > 0:
                grad[t:] += wk * (1.0 - h[:L])
                grad[:L] += wk * (-h[t:])
    return grad
```

```python
def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_h(h, target):
    h = np.asarray(h, dtype=float)
    lo, hi = -2.0, 2.0
    for _ in range(90):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def lse_max(corr, temp):
    m = corr.max()
    return m + temp * np.log(np.sum(np.exp((corr - m) / temp))) / len(corr)

def smooth_grad(h, temp):
    n = len(h)
    corr = compute_corr(h)
    m = corr.max()
    w = np.exp((corr - m) / temp)

    grad = np.zeros(n)
```
