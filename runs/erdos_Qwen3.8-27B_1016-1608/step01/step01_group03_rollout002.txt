    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def project_feasible(h, target):
    """Project onto {0<=x<=1, sum(x)=target}"""
    # Bisection on lambda
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    return np.clip(h - lam, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def project_feasible(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(64):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def compute_subgrad(h, n, dx, top_k=10):
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[::-1][:top_k]
    grad = np.zeros(n)
    for m in top_idx:
        L = m - (n - 1)
        if L == 0:
            grad += (1.0 - 2.0 * h)
        elif L > 0:
            a = np.zeros(n)
            a[L:] += 1.0 - h[:n - L]
            a[:n - L + 1] += -h[L:]
            grad += a
        else:
            Lg = -L
            a = np.zeros(n)
            a[:n - Lg + 1] += 1.0 - h[Lg:]
            a[Lg:] += -h[:n - Lg]
            grad += a
    grad *= dx / len(top_idx)
    return grad

def nesterov_subgrad(h0, target, n, dx, n_iter=3000, alpha0=0.15):
    h = h0.astype(float).copy()
    h_prev = h0.astype(float).copy()
    t = 1.0
    best_h = h0.copy()
    best_c5 = compute_c5(h0, dx)
    for k in range(n_iter):
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = (t - 1.0) / t_new
        y = h + beta * (h - h_prev)
        g = compute_subgrad(y, n, dx, top_k=10)
        alpha = alpha0 / np.sqrt(float(k + 1))
        h_new = project_feasible(y - alpha * g, target)
        c5v = compute_c5(h_new, dx)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_new.copy()
        h_prev = h
        h = h_new
        t = t_new
    return best_h, best_c5

def slsqp_polish(h0, target, n, dx, maxiter=500):
    tgt = target
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - tgt}]
    def obj(x):
        return compute_c5(np.clip(x, 0.0, 1.0), dx)
    try:
        r = minimize(obj, h0, method='SLSQP', bounds=bnds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-15})
        hn = project_feasible(np.clip(r.x, 0, 1), target)
        return hn, compute_c5(hn, dx)
    except Exception:
        return h0, compute_c5(h0, dx)
