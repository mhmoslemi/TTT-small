import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def projection(h, target):
    h = np.array(h, dtype=float, copy=True)
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        excess = h.sum() - target
        if abs(excess) < 1e-13:
            return h
        if excess > 0:
            mask = h > 1e-15
            cnt = mask.sum()
            if cnt == 0:
                return h
            s = min(excess / cnt, h[mask].min())
            h[mask] -= s
        else:
            mask = h < 1.0 - 1e-15
            cnt = mask.sum()
            if cnt == 0:
                return h
            s = min(-excess / cnt, (1.0 - h[mask]).min())
            h[mask] += s
    return h

def lse_obj_and_grad(h, dx, temp, lam, target):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    scaled = (corr - m) / temp
    exp_scaled = np.exp(scaled)
    s_sum = np.sum(exp_scaled)
    val = (m + temp * np.log(s_sum)) * dx + lam * (h.sum() - target) ** 2
    p = exp_scaled / s_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = p[k]
        if w < 1e-14:
            continue
        pp = k - (n - 1)
        if pp >= 0:
            if pp < n:
                grad[pp:] += w * (1.0 - h[:n - pp])
                grad[:n - pp] -= w * h[pp:]
        else:
            q = -pp
            if q < n:
                grad[:n - q] += w * (1.0 - h[q:])
                grad[q:] -= w * h[:n - q]
    grad *= dx
    grad += 2.0 * lam * (h.sum() - target)
    return val, grad

def slsqp_smooth(h0, dx, temp, lam, target, n, bounds, maxiter=200):
    def obj(x):
        v, _ = lse_obj_and_grad(x, dx, temp, lam, target)
        return v
    def jac(x):
        _, g = lse_obj_and_grad(x, dx, temp, lam, target)
        return g
    try:
        res = minimize(obj, h0, jac=jac, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10, 'maxcor': 20})
        return projection(res.x, target)
    except Exception:
        return None

def slsqp_hard(h0, dx, target, n, bounds, maxiter=300):
    eq = {'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}
    try:
        res = minimize(compute_c5, h0, args=(dx,), method='SLSQP',
                       bounds=bounds, constraints=(eq,),
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return projection(res.x, target)
    except Exception:
        return None

def div_init(rng, n, best_h):
    r = rng.randint(6)
    if r == 0:
        return best_h + rng.randn(n) * 0.1
    elif r == 1:
        return rng.random(n) * 0.8 + 0.1
    elif r == 2:
        h0 = np.zeros(n)
        blk = rng.randint(3, max(4, n // 4))
        for st in range(0, n, 2 * blk):
            h0[st:st + blk] = 1.0
        return h0
    elif r == 3:
        t = np.linspace(0, 4 * np.pi, n)
        return 0.5 + 0.4 * np.sin(t + rng.uniform(0, 2 * np.pi)) * rng.uniform(0.4, 0.8)
    elif r == 4:
        return best_h + rng.randn(n) * rng.uniform(0.2, 0.6)
    else:
        h0 = np.zeros(n)
        for _ in range(rng.randint(n // 5, n // 3)):
            b = rng.randint(2, 9)
            s = rng.randint(0, max(1, n - b))
            h0[s:s + b] = 1.0
        return h0

    rng = np.random.RandomState(seed)

    best_h = projection(best_h, target)
    final_c5 = compute_c5(best_h, dx)
    return (best_h, final_c5, n)
