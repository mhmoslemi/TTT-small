def grad_for_shift(h, shift):
    n = len(h)
    g = np.zeros(n)
    if shift >= 0:
        g[shift:] += 1.0 - h[:n - shift]
        g[:n - shift] -= h[shift:]
    else:
        t = -shift
        g[:n - t] += h[t:] - 1.0
        g[t:] -= h[:n - t]
    return g

def soft_max_grad(h, tau, top_k=None):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0/n)
    cm = c.max()
    # For numerical stability
    w = np.exp(tau * (c - cm))
    if top_k is not None and len(c) > top_k:
        # Zero out weights for non-top terms
        idx = np.argsort(w)[:-top_k]
        w[idx] = 0.0
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.zeros_like(c)
        w[cm.argmax()] = 1.0
        wsum = 1.0
    w = w / wsum
    # grad = sum_k w_k * grad(c_k)
    grad = np.zeros(n)
    for k in range(len(c)):
        if w[k] > 1e-8:
            shift = k - (n - 1)
            grad += w[k] * grad_for_shift(h, shift)
    return grad, cm

        if elapsed_fn() > time_frac_end:
            break

        cmax = c.max()

        if len(active) > 10:
            active = np.argsort(c)[-10:]

            if s < best_s:

def _cons_fun(x, n):
    return np.sum(x) - n / 2.0

def subgrad_descend(h, n, max_iter, base_lr, active_eps, elapsed_fn, time_limit):
    h = h.copy()
    n = len(h)
    momentum = np.zeros(n)
    mu = 0.85
    for it in range(max_iter):
        if elapsed_fn() > time_limit:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        cmax = c.max()
        active = np.where(c >= cmax - active_eps)[0]
        if len(active) > 8:
            active = np.argsort(c)[-8:]
        elif len(active) == 0:
            active = np.array([int(np.argmax(c))])
        grad = np.zeros(n)
        for lag in active:
            shift = lag - (n - 1)
            grad += grad_for_shift(h, shift)
        grad /= len(active)
        lr = base_lr / (1.0 + 0.03 * it)
        momentum = mu * momentum - lr * grad
        h = h + momentum
        h = project_feasible(h, n)
    return h

def soft_max_obj(h, tau):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cm = np.max(c)
    shifted = tau * (c - cm)
    shifted = np.clip(shifted, -50, 50)
    return cm + np.log(np.sum(np.exp(shifted))) / tau

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    best_h = h_init.copy()
    best_score = compute_score(h_init)

    n_list = sorted(set([n_init, 36, 48, 60, 80, 100, 128, 160, 200, 256]))
    current_h = h_init.copy()
    current_n = n_init

    for n_points in n_list:
        if elapsed() > budget_s * 0.80:
            break
        if n_points == current_n:
            h0 = current_h.copy()
        else:
            x_old = np.linspace(0, 2, current_n, endpoint=False)
            x_new = np.linspace(0, 2, n_points, endpoint=False)
            h0 = np.interp(x_new, x_old, current_h)
            h0 = project_feasible(h0, n_points)

        s0 = compute_score(h0)
        if s0 < best_score:
            best_score = s0
            best_h = h0.copy()

        base_lr = 0.06 if n_points <= 100 else 0.04
        n_iter = 5000 if n_points <= 80 else 3000 if n_points <= 160 else 1500
        tl = budget_s * 0.82
        h_out = subgrad_descend(h0, n_points, n_iter, base_lr, 0.002, elapsed, tl)
        s_out = compute_score(h_out)
        if s_out < best_score:
            best_score = s_out
            best_h = h_out.copy()
        if s_out < s0:
            h0 = h_out
            s0 = s_out

        # Second pass with tighter active set
        if elapsed() < tl - 10 and s_out > best_score * 0.999:
            h_out2 = subgrad_descend(h0, n_points, n_iter // 2, base_lr * 0.5, 0.001, elapsed, tl)
            s_out2 = compute_score(h_out2)
            if s_out2 < best_score:
                best_score = s_out2
                best_h = h_out2.copy()
            if s_out2 < s0:
                h0 = h_out2
                s0 = s_out2

        # SLSQP polish with soft-max
        if elapsed() < budget_s * 0.76 and n_points <= 200:
            try:
                cons = [{'type': 'eq', 'fun': _cons_fun, 'args': (n_points,)}]
                res = minimize(soft_max_obj, h0, args=(50.0,), method='SLSQP',
                               bounds=[(0, 1)] * n_points, constraints=cons,
                               options={'maxiter': 200, 'ftol': 1e-12})
                h_sl = project_feasible(res['x'], n_points)
                s_sl = compute_score(h_sl)
                if s_sl < best_score:
                    best_score = s_sl
                    best_h = h_sl.copy()
                if s_sl < s0:
                    h0 = h_sl
                    s0 = s_sl
            except Exception:
                pass

        # SLSQP with exact objective
        if elapsed() < budget_s * 0.76 and n_points <= 128:
            try:
                cons = [{'type': 'eq', 'fun': _cons_fun, 'args': (n_points,)}]
                res = minimize(compute_score, h0, method='SLSQP',
                               bounds=[(0, 1)] * n_points, constraints=cons,
                               options={'maxiter': 400, 'ftol': 1e-13})
                h_sl = project_feasible(res['x'], n_points)
                s_sl = compute_score(h_sl)
                if s_sl < best_score:
                    best_score = s_sl
                    best_h = h_sl.copy()
            except Exception:
                pass

        current_h = h0
        current_n = n_points

    # Multi-start exploration
    if elapsed() < budget_s * 0.60:
        for n_s in [32, 48, 64, 80]:
            if elapsed() > budget_s * 0.70:
                break
            for trial in range(5):
                if elapsed() > budget_s * 0.72:
                    break
                if trial == 0:
                    x_old = np.linspace(0, 2, len(best_h), endpoint=False)
                    x_new = np.linspace(0, 2, n_s, endpoint=False)
                    h_r = np.interp(x_new, x_old, best_h)
                elif trial == 1:
                    h_r = np.sin(np.linspace(0, 2 * np.pi * 3, n_s, endpoint=False)) * 0.3 + 0.5
                elif trial == 2:
                    h_r = np.sin(np.linspace(0, 2 * np.pi * 7, n_s, endpoint=False)) * 0.3 + 0.5
                else:
                    h_r = rng.uniform(0, 1, size=n_s)
                h_r = project_feasible(h_r, n_s)
                h_r = subgrad_descend(h_r, n_s, 3000, 0.05, 0.002, elapsed, budget_s * 0.72)
                s_r = compute_score(h_r)
                if s_r < best_score:
                    best_score = s_r
                    best_h = h_r.copy()

    # Final high-iteration polish on best
    if elapsed() < budget_s * 0.85:
        n_f = len(best_h)
        h_f = subgrad_descend(best_h, n_f, 3000, 0.015, 0.0005, elapsed, budget_s * 0.90)
        s_f = compute_score(h_f)
        if s_f < best_score:
            best_score = s_f
            best_h = h_f.copy()

    # Final SLSQP
    if elapsed() < budget_s * 0.92:
        n_f = len(best_h)
        try:
            cons = [{'type': 'eq', 'fun': _cons_fun, 'args': (n_f,)}]
            res = minimize(compute_score, best_h, method='SLSQP',
                           bounds=[(0, 1)] * n_f, constraints=cons,
                           options={'maxiter': 500, 'ftol': 1e-14})
            h_f = project_feasible(res['x'], n_f)
            s_f = compute_score(h_f)
            if s_f < best_score:
                best_score = s_f
                best_h = h_f
        except Exception:
            pass

    best_h = project_feasible(best_h, len(best_h))
    best_score = compute_score(best_h)
    return best_h, best_score, len(best_h)
