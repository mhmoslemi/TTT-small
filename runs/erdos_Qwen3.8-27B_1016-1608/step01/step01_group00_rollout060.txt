from scipy.optimize import minimize
import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        if not np.any(mask):
            h[:] = target / n
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def thue_morse(n):
    """Generate Thue-Morse binary sequence of length n."""
    tm = np.zeros(n, dtype=int)
    for i in range(1, n):
        tm[i] = np.binary_repr(i).count('1') % 2
    return tm

def slsqp_opt(h0, n, maxiter=500, ftol=1e-14):
    bnds = [(0.0, 1.0)] * n
    def obj(x):
        return float(np.max(np.correlate(x, 1.0 - x, mode='full')))
    def con_fun(x):
        return np.sum(x) - n / 2.0
    cons = [{'type': 'eq', 'fun': con_fun}]
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        if res.fun < np.max(np.correlate(h0, 1.0-h0, mode='full')) + 1e-12:
            return repair(np.clip(res.x, 0.0, 1.0), n)
    except Exception:
        pass
    return h0

def binary_sa(h_bin, n, t0, t_end):
    """Simulated annealing on binary sequence via pair flips."""
    h = h_bin.copy().astype(float)
    best_h = h.copy()
    best_val = float(np.max(compute_corr(h)))
    cur_val = best_val
    
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    
    T = 0.5
    cooling = 0.9995
    
    while time.time() < t_end:
        T *= cooling
        if T < 0.001:
            T = 0.001
            cooling = 1.0  # stay at minimum temperature
        
        if not ones or not zeros:
            break
        
        i1 = ones[np.random.randint(len(ones))]
        i0 = zeros[np.random.randint(len(zeros))]
        
        h[i1] = 0.0
        h[i0] = 1.0
        new_val = float(np.max(compute_corr(h)))

            ones[ones.index(i1)] = i0  # i1 becomes zero

        else:

def repair(h, n):

    out = np.zeros(n, dtype=float)
    for i in range(n):

def golden_seq(n):
    phi = (1 + np.sqrt(5)) / 2
    return np.array([1.0 if (i * phi) % 1.0 < 0.5 else 0.0 for i in range(n)])

def slsqp_opt(h0, n, maxiter=500):
    def obj_func(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        return float(np.max(c))
    def con_func(x):
        return float(np.sum(x) - n / 2.0)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_func}]
    try:
        res = minimize(obj_func, h0, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = np.clip(res.x, 0.0, 1.0)
        if max_corr(h_new) <= max_corr(h0) + 1e-10:
            return repair(h_new, n)
    except Exception:
        pass
    return h0

    # Ensure sum is n/2
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    while len(ones) > n // 2:
        idx = ones.pop(np.random.randint(len(ones)))
        h[idx] = 0
        zeros.append(idx)
    while len(ones) < n // 2:
        idx = zeros.pop(np.random.randint(len(zeros)))
        h[idx] = 1
        ones.append(idx)

    best_val = cur_val
    best_h = h.copy()

        i1 = ones[np.random.randint(len(ones))]
        i0 = zeros[np.random.randint(len(zeros))]

                best_h = h.copy()
        else:

import numpy as np
from scipy.optimize import minimize
import time

def _obj(x):
    return float(np.max(np.correlate(x, 1.0 - x, mode='full')))

def _con(x, n):
    return float(np.sum(x) - n / 2.0)

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')))

def repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        if not np.any(mask):
            h[:] = target / n
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

        i1 = ones[np.random.randint(len(ones))]

        h[i1] = 0.0
        h[i0] = 1.0
        nv = max_corr_val(h)
        d = nv - cur_val
        if d < 0 or np.random.random() < np.exp(-d / T):
            cur_val = nv
            ones[ones.index(i1)] = i0
            zeros[zeros.index(i0)] = i1
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
        else:
            h[i1] = 1.0
            h[i0] = 0.0
    return best_h, best_val

def binary_local_search(h0, n, max_passes=200):
    h = h0.copy().astype(float)
    one_set = list(np.where(h == 1)[0])
    zero_set = list(np.where(h == 0)[0])
    cur_val = max_corr_val(h)
    best_h = h.copy()
    best_val = cur_val
    for _ in range(max_passes):
        improved = False

                    best_h = h.copy()
                    improved = True
                    break
                else:
                    h[i1] = 1.0
                    h[i0] = 0.0
        if not improved:

    return best_h, best_val

    try:

                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = np.clip(res.x, 0.0, 1.0)
        if _obj(h_new) <= _obj(h0) + 1e-10:

    return h0
