```python
    # For each i, grad[i] = dx * sum_k p[k] * [(1-h[i+lags[k]])*valid - h[i-lags[k]]*valid]
    
    # Build matrix: M[i,k] = (1-h[i+lags[k]]) if 0<=i+lags[k]<n else 0
    #              - h[i-lags[k]] if 0<=i-lags[k]<n else 0
    # grad = dx * M @ p
    
    # This is an nx(2n-1) matrix, which for n=400 is 400x799 ≈ 320k entries
    # We can build it efficiently:
    
    i_idx = np.arange(n)[:, None]  # (n, 1)
    lag_idx = lags[None, :]  # (1, 2n-1)
    
    ipl = i_idx + lag_idx  # i + lag
    ipm = i_idx - lag_idx  # i - lag
    
    mask1 = (ipl >= 0) & (ipl < n)
    mask2 = (ipm >= 0) & (ipm < n)
    
    # For term1: we need 1-h[ipl] where mask1, else 0
    ipl_clipped = np.clip(ipl, 0, n-1)
    term1 = np.where(mask1, 1.0 - h[ipl_clipped], 0.0)
    
    # For term2: we need h[ipm] where mask2, else 0
    ipm_clipped = np.clip(ipm, 0, n-1)
    term2 = np.where(mask2, h[ipm_clipped], 0.0)
    
    grad = dx * ((term1 - term2) @ p)
    return grad
```
