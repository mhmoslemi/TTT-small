def c5_grad_with_pen(h, lam):
    """Computes gradient of c5 + lam*(sum(h)-n/2)^2."""
    n = len(h)
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode='full')
    corr = corr_raw * dx
    c5 = float(np.max(corr))
    
    tol = 1e-9 * max(1.0, abs(c5))
    active = np.where(corr >= c5 - tol)[0]
    
    grad = np.zeros(n)
    for jk in active:
        m = int(jk - (n - 1))
        i_lo = max(0, -m)
        i_hi = min(n - 1, n - 1 - m)
        # (1-h[i+m]) for i in [i_lo, i_hi]
        grad[i_lo:i_hi+1] += (1.0 - h[i_lo+m : i_hi+1+m])
        # -h[i] for j=i+m in [max(0,m), min(n-1, i_hi+m)]
        j_lo = i_lo + m
        j_hi = i_hi + m
        grad[j_lo:j_hi+1] -= h[i_lo:i_hi+1]
    
    if len(active) > 0:
        grad /= len(active)
    
    # Penalty gradient
    s = np.sum(h) - n / 2.0
    grad += 2.0 * lam * s
    
    return c5 + lam * s * s, grad

    def tl():
        return budget_s - (time.time() - t0)

    # Phase 1: Softmax warm-start then hard-max L-BFGS-B
    if tl() > 80:
        tau_vals = [0.5, 2.0, 5.0, 15.0, 50.0]
        for tau in tau_vals:
            if tl() < 30:
                break
            h = lbfgsb_opt(h_best, n, maxiter=800, use_softmax=True, tau=tau)
            c = compute_c5(h)
            if c < c5_best:
                h_best, c5_best = h, c
        h = lbfgsb_opt(h_best, n, maxiter=1000, use_softmax=False)
        c = compute_c5(h)
        if c < c5_best:
            h_best, c5_best = h, c

    # Phase 2: Multi-start with diverse perturbations
    styles = [0, 1, 2, 3, 4]
    attempt = 0
    while tl() > 250:
        style = attempt % len(styles)
        h_p = perturb(h_best, n, rng, style)
        # Try both softmax and direct
        if attempt % 2 == 0:
            h = lbfgsb_opt(h_p, n, maxiter=600, use_softmax=True, tau=8.0)
        else:
            h = lbfgsb_opt(h_p, n, maxiter=600, use_softmax=False)
        c = compute_c5(h)
        if c < c5_best:
            h_best, c5_best = h, c
        attempt += 1

    # Phase 3: SLSQP with exact constraint
    if tl() > 80:
        h = slsqp_opt(h_best, n, maxiter=4000)
        c = compute_c5(h)
        if c < c5_best:
            h_best, c5_best = h, c

    # Phase 4: Fine perturbations around best
    while tl() > 50:
        for scale in [0.001, 0.003, 0.008]:
            if tl() < 30:
                break
            h_p = repair(h_best + rng.randn(n) * scale, n)
            h = lbfgsb_opt(h_p, n, maxiter=800, use_softmax=False)
            c = compute_c5(h)
            if c < c5_best:
                h_best, c5_best = h, c

    # Phase 5: SLSQP final polish
    if tl() > 25:
        h = slsqp_opt(h_best, n, maxiter=5000)
        c = compute_c5(h)
        if c < c5_best:
            h_best, c5_best = h, c

    # Phase 6: Try nearby n values via interpolation
    if tl() > 40:
        for n2 in [50, 80]:
            if tl() < 25:
                break
            xp = np.linspace(0.0, 1.0, n)
            xn = np.linspace(0.0, 1.0, n2)
            h_interp = np.interp(xn, xp, h_best)
            h_interp = repair(h_interp, n2)
            h2 = lbfgsb_opt(h_interp, n2, maxiter=600, use_softmax=False)
            c2 = compute_c5(h2)
            if c2 < c5_best:
                h_best, c5_best, n = h2, c2, n2
            if tl() > 30:
                h2 = slsqp_opt(h_best, n, maxiter=3000)
                c2 = compute_c5(h2)
                if c2 < c5_best:
                    h_best, c5_best, n = h2, c2, n

    h_best = repair(h_best, n)
    c5_best = compute_c5(h_best)
    return (h_best, c5_best, n)
