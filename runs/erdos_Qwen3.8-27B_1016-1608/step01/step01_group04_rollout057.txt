from scipy.optimize import minimize
import time

_budget = 1100

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

    h = np.array(h, dtype=float).copy()
    for _ in range(80):
        np.clip(h, 0.0, 1.0, out=h)
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h += d / len(h)
    return np.clip(h, 0.0, 1.0)

def subgrad_vec(h, n, k):
    dx = 2.0 / n
    g = np.zeros(n)
    if k >= 0:
        g[:n-k] += dx
        g[:n-k] -= dx * h[k:]
        g[k:] -= dx * h[:n-k]
    else:
        k2 = -k
        g[k2:] += dx
        g[k2:] -= dx * h[:n-k2]
        g[:n-k2] -= dx * h[k2:]
    return g

def subgrad_opt(h, n, n_iter, step0):
    target = n / 2.0
    h = h.copy()
    best = compute_c5(h)
    best_h = h.copy()
    step = step0
    best_it = 0
    for it in range(n_iter):
        if time.time() - _t0 > _budget - 30:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        ks = int(np.argmax(corr))
        c5 = float(corr[ks]) * (2.0/n)

            best = c5
            best_h = h.copy()
            best_it = it

        h = repair(h, target)
        step = max(step * 0.9997, 0.0001)
        if it - best_it > 3000:

def obj_pn(x, n, p):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    c = np.maximum(c, 0.0)
    return float(np.sum(c**p)**(1.0/p))

                 constraints=cs, options={'maxiter': maxiter, 'ftol': 1e-12})

                best_h = h.copy()
                imp = True
            else:
