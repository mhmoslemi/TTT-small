        h_alt = np.interp(xs_new, xs_old, best_h)
        h_alt = projection(h_alt, tgt_a)
        for temp in [150, 50, 15, 5, 3]:
            if tl() < 30:
                break
            res = minimize(smooth_obj, h_alt, args=(dx_a, temp, tgt_a, 60.0),
                         method='L-BFGS-B', bounds=bnds_a,
                         options={'maxiter': 200, 'ftol': 1e-16})
            h_alt = projection(res.x, tgt_a)
        if tl() > 25:
            cons_a = [{'type': 'eq', 'fun': eq_constraint, 'args': (tgt_a,)}]
            res = minimize(exact_obj, h_alt, args=(dx_a,), method='SLSQP',
                          bounds=bnds_a, constraints=cons_a,
                          options={'maxiter': 400, 'ftol': 1e-16})
            h_alt = projection(res.x, tgt_a)
            c5_alt = compute_c5(h_alt, dx_a)
            if c5_alt < best_c5:
                best_h, best_c5 = h_alt.copy(), c5_alt
                n, dx, target, bounds = n_alt, dx_a, tgt_a, bnds_a

    # Phase 5: Pair-swap coordinate descent
    if tl() > 25:
        h_cd = best_h.copy()
        c5_cd = best_c5
        step = 0.03
        for it in range(8000):
            if tl() < 8:
                break
            idx = rng.randint(n)
            improved = False
            for delta in [step, -step, step * 0.3, -step * 0.3]:
                h_t = h_cd.copy()
                h_t[idx] = np.clip(h_t[idx] + delta, 0.0, 1.0)
                h_t = projection(h_t, target)
                c5_t = compute_c5(h_t, dx)
                if c5_t < c5_cd - 1e-15:
                    h_cd, c5_cd = h_t, c5_t
                    improved = True
                    if c5_cd < best_c5:
                        best_h, best_c5 = h_cd.copy(), c5_cd
                    break
            if not improved and it % 400 == 0:
                step = max(0.001, step * 0.6)

    # Phase 6: Final SLSQP + check
    if tl() > 12:
        cons = [{'type': 'eq', 'fun': eq_constraint, 'args': (target,)}]
        res = minimize(exact_obj, best_h, args=(dx,), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': 600, 'ftol': 1e-16})
        h_p = projection(res.x, target)
        c5_p = compute_c5(h_p, dx)
        if c5_p < best_c5:
            best_h, best_c5 = h_p.copy(), c5_p

    best_h = projection(best_h, target)
    final_c5 = compute_c5(best_h, dx)
    return (best_h, final_c5, n)
