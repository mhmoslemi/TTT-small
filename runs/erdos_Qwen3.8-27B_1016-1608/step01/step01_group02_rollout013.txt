                      bounds=bounds, constraints=(cons,),
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return res.x
    except Exception:
        return None

def try_slsq_hard(h0, dx, target, bounds, maxiter, ftol):
    try:
        cons = {'type': 'eq', 'fun': sum_constraint, 'args': (target,)}
        res = minimize(compute_c5, h0, args=(dx,), method='SLSQP',
                      bounds=bounds, constraints=(cons,),
                      options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return None

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    best_h = np.array(initial_h_values, dtype=float).copy()
    best_c5 = compute_c5(best_h, dx)

    # Phase 1: L-BFGS-B soft-max with increasing temperature
    temps_lbf = [5.0, 10.0, 20.0, 40.0, 80.0, 160.0, 320.0]
    for temp in temps_lbf:
        if budget_s - (time.time() - t0) < 30:
            break
        h_res = try_lbfgs(best_h, dx, target, temp, 500.0, bounds, 400)
        if h_res is not None:
            h_p = projection(h_res, target)
            c5_p = compute_c5(h_p, dx)
            if c5_p < best_c5 - 1e-14:
                best_h, best_c5 = h_p, c5_p

    # Phase 2: SLSQP soft-max refinement
    for temp in [20.0, 80.0, 300.0]:
        if budget_s - (time.time() - t0) < 25:
            break
        h_res = try_slsq_smooth(best_h, dx, target, temp, bounds, 300)
        if h_res is not None:
            h_p = projection(h_res, target)
            c5_p = compute_c5(h_p, dx)
            if c5_p < best_c5 - 1e-14:
                best_h, best_c5 = h_p, c5_p

    # Phase 3: SLSQP hard max from current best
    for attempt in range(4):
        if budget_s - (time.time() - t0) < 20:
            break
        scale = 0.02 * (attempt + 1)
        h0 = projection(best_h + rng.randn(n) * scale, target)
        h_res = try_slsq_hard(h0, dx, target, bounds, 500, 1e-15)
        if h_res is not None:
            h_p = projection(h_res, target)
            c5_p = compute_c5(h_p, dx)
            if c5_p < best_c5 - 1e-14:
                best_h, best_c5 = h_p, c5_p

    # Phase 4: Basin hopping with L-BFGS-B (diverse restarts)
    if budget_s - (time.time() - t0) > 80:
        n_bh = 40
        for bh_i in range(n_bh):
            if budget_s - (time.time() - t0) < 20:
                break
            frac = bh_i / max(1, n_bh)
            scale = 0.3 * (1.0 - frac) + 0.05
            mode = bh_i % 6
            if mode == 0:
                h0 = best_h + rng.randn(n) * scale
            elif mode == 1:
                h0 = np.zeros(n)
                bk = rng.randint(5, 12)
                for st in range(0, n, 2 * bk):
                    h0[st:st + bk] = 1.0
                h0 += rng.randn(n) * 0.1
            elif mode == 2:
                t_arr = np.linspace(0, 4 * np.pi, n)
                h0 = 0.5 + 0.35 * np.sin(t_arr + rng.uniform(0, 6.28)) + 0.15 * np.sin(2 * t_arr + rng.uniform(0, 6.28))
            elif mode == 3:
                h0 = best_h.copy()
                idxs = rng.choice(n, n // 3, replace=False)
                h0[idxs] = rng.random(n // 3)
            elif mode == 4:
                h0 = np.zeros(n)
                for _ in range(n // 4):
                    blk = rng.randint(3, 9)
                    st = rng.randint(0, max(1, n - blk))
                    h0[st:st + blk] = rng.uniform(0.6, 1.0)
            else:
                h0 = best_h + rng.randn(n) * 0.15
                h0 = np.clip(h0, 0, 1)
            h0 = projection(h0, target)
            h_res = try_lbfgs(h0, dx, target, 60.0, 500.0, bounds, 150)
            if h_res is not None:
                h_p = projection(h_res, target)
                c5_p = compute_c5(h_p, dx)
                if c5_p < best_c5 - 1e-14:
                    best_h, best_c5 = h_p, c5_p

    # Phase 5: Multi-coordinate descent with adaptive steps
    if budget_s - (time.time() - t0) > 40:
        h_cd = best_h.copy()
        c5_cd = best_c5
        for phase in range(5):
            base_step = [0.06, 0.03, 0.015, 0.007, 0.003][phase]
            for it in range(4000):
                if budget_s - (time.time() - t0) < 8:
                    break
                grp = rng.randint(2, 5)
                idxs = rng.choice(n, grp, replace=False)
                found = False
                for _ in range(4):
                    h_t = h_cd.copy()
                    for idx in idxs:
                        h_t[idx] += rng.uniform(-base_step, base_step)
                    h_t = projection(h_t, target)
                    c5_t = compute_c5(h_t, dx)
                    if c5_t < c5_cd - 1e-13:
                        h_cd, c5_cd = h_t, c5_t
                        found = True
                        if c5_cd < best_c5:
                            best_h, best_c5 = h_cd.copy(), c5_cd
                        break
                if not found and it % 800 == 799:
                    pass

    # Phase 6: Final SLSQP polish (multiple tries)
    if budget_s - (time.time() - t0) > 15:
        h_res = try_slsq_hard(best_h, dx, target, bounds, 600, 1e-15)
        if h_res is not None:
            h_p = projection(h_res, target)
            c5_p = compute_c5(h_p, dx)
            if c5_p < best_c5 - 1e-14:
                best_h, best_c5 = h_p, c5_p

    # Phase 7: One more L-BFGS-B at very high temp for final tightening
    if budget_s - (time.time() - t0) > 10:
        h_res = try_lbfgs(best_h, dx, target, 500.0, 2000.0, bounds, 300)
        if h_res is not None:
            h_p = projection(h_res, target)
            c5_p = compute_c5(h_p, dx)
            if c5_p < best_c5 - 1e-14:
                best_h, best_c5 = h_p, c5_p

    best_h = projection(best_h, target)
    final_c5 = compute_c5(best_h, dx)
    return (best_h, final_c5, n)
