    Returns J of shape (2n-1, n) where J[k, i] = d c[k] / d h[i]
    """
    n = len(h)
    ncorr = 2 * n - 1
    J = np.zeros((ncorr, n))

            if c5_t < c5_cd - 1e-14:
                h_cd, c5_cd = h_t, c5_t
                if c5_t < best_c5:
                    best_h, best_c5 = h_cd.copy(), c5_t

    # Phase 7: Final cutting-plane + SLSQP polish
    if budget_s - el() > 15:
        h_n, c5_n = upd(cutting_plane(best_h, target, 50))
        if c5_n < best_c5:
            best_h, best_c5 = h_n, c5_n
    if budget_s - el() > 8:
        try:
            res = minimize(compute_c5, best_h, args=(dx,), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 1000, 'ftol': 1e-18})
            h_n, c5_n = upd(res.x)
            if c5_n < best_c5 - 1e-15:
                best_h, best_c5 = h_n, c5_n
        except Exception:
            pass
    if budget_s - el() > 8:
        h_n, c5_n = upd(cutting_plane(best_h, target, 30))
        if c5_n < best_c5:
            best_h, best_c5 = h_n, c5_n

    best_h = project_feasible(best_h, target)
    final_c5 = compute_c5(best_h, dx)
    return (best_h, final_c5, n)
