import numpy as np
import time
from scipy.optimize import minimize

_g_tau = 300.0
_g_n = 50

def _smooth_obj(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cm = np.max(c)
    return cm + np.log(np.sum(np.exp(_g_tau * (c - cm)))) / _g_tau + 0.005 * np.mean(c)

def _score(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def _cons_fun(h):
    return np.sum(h) - _g_n / 2.0

def _cons_jac(h):
    return np.ones(_g_n)

def _project(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        cnt = np.sum(m)
        if cnt == 0:
            break
        h = np.clip(h + diff / cnt * m, 0.0, 1.0)
    return h

def _run_slsqp(h0, n, tau, maxit, ftol):
    global _g_tau, _g_n
    _g_tau = tau
    _g_n = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(_smooth_obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxit, 'ftol': ftol})
        return _project(res['x'], n)
    except Exception:
        return h0

def _run_cobyla(h0, n, rho, maxit):
    try:
        def pen(h):
            return _score(h) + 2000.0 * (np.sum(h) - n / 2.0) ** 2
        res = minimize(pen, h0, method='COBYLA',
                       options={'maxiter': maxit, 'rhobeg': rho, 'catol': 1e-13})
        return _project(res['x'], n)
    except Exception:
        return h0

def _run_nm(h0, n, maxit):
    try:
        def pen(h):
            return _score(h) + 1000.0 * (np.sum(h) - n / 2.0) ** 2
        res = minimize(pen, h0, method='Nelder-Mead',
                       options={'maxiter': maxit, 'xatol': 1e-9, 'fatol': 1e-14})
        return _project(res['x'], n)
    except Exception:
        return h0

def _perturb(h, n, rng, scale):
    return _project(h + rng.randn(n) * scale, n)

def _block_perturb(h, n, rng):
    h2 = h.copy()
    nb = rng.randint(1, max(2, n // 8))
    for _ in range(nb):
        s = rng.randint(0, n - 2)
        ln = rng.randint(2, max(3, n // 6))
        s2 = min(s, n - ln)
        h2[s2:s2 + ln] += rng.uniform(-0.4, 0.4)
    return _project(h2, n)

def _swap_perturb(h, n, rng):
    h2 = h.copy()
    for _ in range(rng.randint(2, 5)):
        i, j = rng.choice(n, 2, replace=False)
        h2[i], h2[j] = h2[j], h2[i]
    return _project(h2, n)
