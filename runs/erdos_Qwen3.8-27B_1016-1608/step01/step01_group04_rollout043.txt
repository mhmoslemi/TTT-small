    off = n - 1
    m = np.arange(n)
    g = np.zeros(n)
    j1 = m + k - off
    mk1 = (j1 >= 0) & (j1 < n)
    g[mk1] += 1.0 - h[j1[mk1]]
    j2 = m - k + off
    mk2 = (j2 >= 0) & (j2 < n)
    g[mk2] -= h[j2[mk2]]
    return g

def subgrad_step(h, target, top_m):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argpartition(c, -top_m)[-top_m:]
    g = np.zeros(n)
    for k in idx:
        g += grad_lag(h, int(k))
    g /= top_m
    return g, float(np.max(c) * 2.0 / n)

def subgrad_descent(h0, target, dur_s, top_m=3):
    h = project_feasible(h0.copy(), target)
    n = len(h)
    bh, bc = h.copy(), compute_c5(h)

def slsqp_polish(h0, n, target, eps, mi):
    def obj(x):

        m = c.max()

                 constraints=[{'type': 'eq', 'fun': eqf, 'jac': eqj}],
                 options={'maxiter': mi, 'ftol': 1e-14})

    n = len(h0)
    tgt = n / 2.0
    bh = project_feasible(h0.copy(), tgt)
    bc = compute_c5(bh)

    # Phase 1: subgradient at base resolution (two passes, different top_m)
    rem = budget_s - (time.time() - t0)
    if rem > 300:
        h1, c1 = subgrad_descent(bh, tgt, min(250, rem - 200), top_m=3)
        bh, bc = _try_update(bh, bc, h1)
    rem = budget_s - (time.time() - t0)
    if rem > 150:
        h1, c1 = subgrad_descent(bh, tgt, min(120, rem - 100), top_m=5)
        bh, bc = _try_update(bh, bc, h1)
    rem = budget_s - (time.time() - t0)
    if rem > 120:
        h1, c1 = subgrad_descent(bh, tgt, min(60, rem - 80), top_m=2)
        bh, bc = _try_update(bh, bc, h1)

    # SLSQP polish (multi-eps)
    for eps in [0.005, 0.002, 0.001, 0.0003, 0.0001]:
        if budget_s - (time.time() - t0) < 80:
            break
        bh, bc = _try_update(bh, bc, slsqp_polish(bh, n, eps, 300))

    # Phase 2: diverse restarts
    for i in range(400):
        if budget_s - (time.time() - t0) < 60:
            break
        r = i % 10
        if r == 0:
            hp = bh + np.random.randn(n) * 0.03
        elif r == 1:
            idx = np.random.choice(n, size=max(1, n // 10), replace=False)
            hp = bh.copy()
            hp[idx] = 1.0 - bh[idx]
        elif r == 2:
            k2 = np.random.randint(1, 8)
            ph = np.random.uniform(0, 6.28)
            hp = bh + 0.04 * np.sin(2.0 * np.pi * k2 * np.arange(n) / n + ph)
        elif r == 3:
            p = np.random.randint(3, 20)
            hp = (np.arange(n) % p < p // 2).astype(float)
        elif r == 4:
            hp = bh + np.random.randn(n) * 0.08
        elif r == 5:
            idx = np.random.choice(n, size=max(1, n // 5), replace=False)
            hp = bh.copy()
            hp[idx] = np.random.uniform(0, 1, size=len(idx))
        elif r == 6:
            kk = np.random.randint(1, 6)
            hp = 0.5 + 0.4 * np.cos(2.0 * np.pi * kk * np.arange(n) / n)
        elif r == 7:
            hp = bh + np.random.randn(n) * 0.01
        elif r == 8:
            s = np.random.randint(0, max(1, n - 5))
            l = np.random.randint(3, min(10, n - s))
            hp = bh.copy()
            hp[s:s + l] = 1.0 - bh[s:s + l]
        else:
            hp = np.random.rand(n) * 0.3 + 0.35
        hp = project_feasible(hp, tgt)
        rem = budget_s - (time.time() - t0)
        dur = max(2, min(10, int((rem - 60) * 0.25)))
        hd, cd = subgrad_descent(hp, tgt, dur, top_m=3)
        bh, bc = _try_update(bh, bc, hd)
        bh, bc = _try_update(bh, bc, slsqp_polish(hd, n, 0.002, 120))

    # Phase 3: higher resolution attempts
    for n_hi in [160, 240, 320, 480]:
        if budget_s - (time.time() - t0) < 80:
            break
        nc = len(bh)
        x0 = np.linspace(0, 2, nc)
        x1 = np.linspace(0, 2, n_hi)
        hh = np.interp(x1, x0, bh)
        hh = project_feasible(hh, n_hi / 2.0)
        rem = budget_s - (time.time() - t0)
        dur = max(5, min(60, int((rem - 80) * 0.4)))
        hh, ch = subgrad_descent(hh, n_hi / 2.0, dur, top_m=3)
        hh2 = slsqp_polish(hh, n_hi, 0.001, 200)
        c5 = compute_c5(hh2)
        if c5 < bc:
            bc, bh, n = c5, hh2, n_hi
            tgt = n / 2.0

    # Final SLSQP polish
    if budget_s - (time.time() - t0) > 25:
        bh, bc = _try_update(bh, bc, slsqp_polish(bh, n, 0.0001, 500))

    # Tiny-step local search
    for _ in range(4000):
        if budget_s - (time.time() - t0) < 3:
            break
        hp = project_feasible(bh + np.random.randn(n) * 0.004, tgt)
        bh, bc = _try_update(bh, bc, hp)

    return (bh.tolist(), float(bc), n)
