For coordinate descent: at each step, for coordinate j, we want to find h[j] that minimizes the max correlation while keeping other coordinates fixed. The correlation at lag k is:
corr[k] = ... + h[j]*(1-h[j_tau]) + h[j+tau]*(1-h[j]) + ... 

where tau = k-(n-1). This is a quadratic in h[j] at most (since h[j] appears linearly in the "linear" part and in products with other h values).

Actually, each corr[k] is linear in h[j] when we fix all other h values:
corr[k] = (terms not involving h[j]) + h[j] * (1-h[j-tau]) + (if different term) h[j+tau]*(1-h[j])

Wait no: corr[k] = sum_i h[i]*(1-h[i-tau]). If we fix all h[i] except h[j]:
- If i=j: term is h[j]*(1-h[j-tau]) = h[j] - h[j]*h[j-tau] (linear in h[j])
- If i-tau=j, i.e., i=j+tau: term is h[j+tau]*(1-h[j]) = h[j+tau] - h[j+tau]*h[j] (linear in h[j])
- Other terms don't involve h[j]

So corr[k] is linear in h[j]! The coefficient of h[j] is:
(1-h[j-tau]) - h[j+tau] (with appropriate boundary conditions)

So the max over k of corr[k] is the max of a set of linear functions of h[j], which is convex in h[j]. We can minimize this by finding where the "kink" is (the point where two constraints cross) or at a boundary.

This makes coordinate descent feasible! For each j, we compute the slopes and intercepts of each corr[k] as a function of h[j], then find the optimal h[j].

```python
def coordinate_optimize(h, n, dx, tgt, max_sweeps=10):
    """Optimize each coordinate to minimize the max correlation"""
    for sweep in range(max_sweeps):
        improved = False
        for j in range(n):
            # Compute correlation for all k, tracking linear dependence on h[j]
            # corr[k] = A_k + B_k * h[j]
            # where A_k = corr[k] - h[j]*(1-h[j-tau_k]) - h[j+tau_k]*(1-h[j]) ... 
            # Actually, let's compute it directly.
            h_old = h[j]
            
            # For each k (lag), corr[k] as function of h[j]:
            # tau_k = k - (n-1)
            # h[j] appears in corr[k] as:
            #   if 0 <= j < n and 0 <= j-tau_k < n: h[j]*(1-h[j-tau_k])
            #   if 0 <= j+tau_k < n and 0 <= j < n: h[j+tau_k]*(1-h[j])
            #   (these are the same term if tau_k = 0)
            
            # We need to compute the new corr for a given h[j] value.
            # Since corr is linear in h[j] for each k, we can find the optimal h[j]
            # as the minimizer of max_k (a_k + b_k * t) where t = h[j]
            
            # Compute a_k and b_k for all k
            # This requires computing corr for h with h[j]=0 and h[j]=1, then:
            # corr_k(t) = corr_k(0) + t * (corr_k(1) - corr_k(0))
            
            h0 = h.copy(); h0[j] = 0.0
            h1 = h.copy(); h1[j] = 1.0
            corr0 = np.correlate(h0, 1.0-h0, mode="full")
            corr1 = np.correlate(h1, 1.0-h1, mode="full")
            a = corr0  # intercept
            b = corr1 - corr0  # slope
            
            # Minimize max_k (a_k + b_k * t) over t in [0,1]
            # This is a 1D convex problem
            # The optimal t is where the "upper envelope" has its minimum
            # We can find it by checking the breakpoints
            
            # For simplicity, use ternary search or bisection on the derivative
            # f(t) = max_k (a_k + b_k * t)
            # f'(t) = b_k* where k* is argmax (subgradient)
            # f'(t) = 0 when there exist k1, k2 with b_k1 < 0 < b_k2 and a_k1 + b_k1*t = a_k2 + b_k2*t
            
            # Simpler: just sample t in [0,1] and find the min
            # Or use the fact that f is piecewise linear and convex, so min is at a breakpoint or boundary
            
            # Let's just do a fine grid search in t
            ts = np.linspace(0, 1, 101)
            vals = np.array([np.max(a + b*t) for t in ts])
            j_min = np.argmin(vals)
            t_new = ts[j_min]
            
            # Refine around t_new
            if 0 < j_min < 100:
                ts_fine = np.linspace(max(0, ts[j_min-1]), min(1, ts[j_min+1]), 101)
                vals_fine = np.array([np.max(a + b*t) for t in ts_fine])
                t_new = ts_fine[np.argmin(vals_fine)]
            
            h[j] = t_new
            improved = True
        
        if not improved:
            break
    
    return h
```

import numpy as np

import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def repair(h, target):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(2000):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-15:
            break
        if d > 0:
            m = h < 1.0 - 1e-14
        else:
            m = h > 1e-14
        c = int(m.sum())
        if c == 0:
            break
        h[m] += d / c
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_optimize(h, n, max_sweeps=20):
    """Coordinate descent minimizing max correlation, with sum repair"""
    tgt = n / 2.0
    for sweep in range(max_sweeps):
        for j in range(n):
            h0 = h.copy(); h0[j] = 0.0
            h1 = h.copy(); h1[j] = 1.0
            corr0 = np.correlate(h0, 1.0 - h0, mode="full")
            corr1 = np.correlate(h1, 1.0 - h1, mode="full")
            a = corr0
            b = corr1 - corr0
            # Minimize max_k(a_k + b_k * t) for t in [0,1]
            # Use fine grid + refinement
            ts = np.linspace(0, 1, 201)
            vals = np.array([np.max(a + b * t) for t in ts])
            j_min = np.argmin(vals)
            t_new = ts[j_min]
            if 0 < j_min < 200:
                lo = ts[j_min - 1]
                hi = ts[j_min + 1]
                ts_f = np.linspace(lo, hi, 201)
                vals_f = np.array([np.max(a + b * t) for t in ts_f])
                t_new = ts_f[np.argmin(vals_f)]
            h[j] = t_new
        # Repair sum
        h = repair(h, tgt)
    return h

        c_trial = compute_c5(h_trial, dx)

            c_best = c_trial

    return h_best, c_best
