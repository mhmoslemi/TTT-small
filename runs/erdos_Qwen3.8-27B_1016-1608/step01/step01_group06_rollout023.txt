    n = len(h)
    dx = 2.0 / n
    c = compute_c(h)
    best_max = np.max(c)
    
    for sweep in range(max_sweeps):
        improved = False
        k_star = np.argmax(c)  # The lag achieving the max
        # For this lag, find the pairs contributing most
        # c[k_star] involves pairs (i, i+k_star-n+1) or similar
        
        # Try all pairwise adjustments
        for i in range(n):
            for j in range(i):
                # Try: decrease h[i], increase h[j] by same amount
                # or swap h[i] and h[j]
                # Simplest: swap
                h[i], h[j] = h[j], h[i]
                c_new = compute_c(h)
                max_new = np.max(c_new)
                if max_new < best_max - 1e-12:

                    c = c_new
                    improved = True
                else:
                    h[i], h[j] = h[j], h[i]  # revert
        
        if not improved:

def compute_c(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_score(h):
    return np.max(compute_c(h))

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h = np.clip(h + diff / cnt * mask, 0.0, 1.0)
    return h

    n = len(h)
    c_shifted = c - np.max(c)
    weights = np.exp(tau * c_shifted)
    weights /= np.sum(weights)
    
    # For each output index m, the shift is s = m - (n-1)

import numpy as np
import time

_rng = None

def compute_c(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_score(h):
    return np.max(compute_c(h))

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h = np.clip(h + diff / cnt * mask, 0.0, 1.0)
    return h

def grad_for_shift(h, s, n):
    grad = np.zeros(n)
    if 0 <= s < n:
        grad[:n - s] += 1.0 - h[s:]
        grad[s:] -= h[:n - s]
    elif -n < s < 0:
        sp = -s
        grad[sp:] += 1.0 - h[:n - sp]
        grad[:n - sp] -= h[sp:]
    return grad

def soft_grad(h, c, tau, n):
    c_shift = c - np.max(c)
    w = np.exp(tau * c_shift)
    w /= w.sum()
    grad = np.zeros(n)
    for m in range(len(c)):
        if w[m] < 1e-12:
            continue
        s = m - (n - 1)
        grad += w[m] * grad_for_shift(h, s, n)
    return grad

def subgrad_descent(h, max_iter, tau_start, alpha0, t_end):
    global _rng
    n = len(h)
    h_curr = h.copy()
    best_h = h.copy()
    best_score = compute_score(h)
    alpha = alpha0
    stall = 0
    tau = tau_start

    for it in range(max_iter):
        if time.time() > t_end:
            break
        c = compute_c(h_curr)
        score = np.max(c)
        if score < best_score - 1e-15:
            best_score = score
            best_h = h_curr.copy()
            stall = 0
        else:
            stall += 1

        switch = max_iter * 0.35
        if it < switch:
            g = soft_grad(h_curr, c, tau, n)
            tau *= 1.002
        else:
            m_star = np.argmax(c)
            s = m_star - (n - 1)
            g = grad_for_shift(h_curr, s, n)

        g -= np.mean(g)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn

        h_new = h_curr - alpha * g
        h_new = project_feasible(h_new, n)
        s_new = compute_score(h_new)
        if s_new < score:
            h_curr = h_new
            if stall > 300:
                alpha = max(1e-5, alpha * 0.85)
                stall = 0
        else:
            alpha *= 0.6
            if alpha < 1e-6:
                break
        if stall > 5000:
            break

    return best_h, best_score

def swap_search(h, max_sweeps, t_end):
    global _rng
    n = len(h)
    h_curr = h.copy()
    best_h = h.copy()
    best_score = compute_score(h)

        improved = False
        c = compute_c(h_curr)
        k_star = np.argmax(c)
        s_star = k_star - (n - 1)

        if s_star >= 0:
            ov = n - s_star
            pen = np.zeros(n)
            pen[:ov] = h_curr[:ov] * h_curr[s_star:s_star + ov]
        else:
            sp = -s_star
            ov = n - sp
            pen = np.zeros(n)
            pen[sp:] = h_curr[sp:] * h_curr[:ov]

        order = np.argsort(-pen)[:25]
        for idx in order:

                    continue

                    improved = True

                improved = True
            else:

            break

    return best_h, best_score

                improved = True
            else:
                h_curr[i] -= d
                h_curr[j] += d
