from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def build_sym(h_half, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = h_half
    h[m:] = 1.0 - h_half[::-1]
    return h

        if not np.any(mask):

            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
