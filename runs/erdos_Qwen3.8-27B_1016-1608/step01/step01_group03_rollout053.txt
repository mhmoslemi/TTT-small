    x = corr * dx
    x_max = np.max(x)
    s = np.sum(np.exp(beta * (x - x_max)))
    return x_max + np.log(s) / beta

    corr = np.correlate(h, 1.0 - h, mode="full")

    best_c5 = compute_c5(h0, dx)
    best_h = h0.copy()

    # Phase 1: Subgradient from parent
    if time.time() - t0 < budget_s - 200:
        h1, c1 = subgradient(h0, target, dx, max_iter=10000, lr0=0.012, momentum=0.82, topk=3)
        if c1 < best_c5:
            best_c5, best_h = c1, h1
        # SLSQP polish
        if time.time() - t0 < budget_s - 150:
            for beta in [30, 80, 200]:
                if time.time() - t0 > budget_s - 150:
                    break
                h2, c2 = slsqp_smooth(best_h, target, dx, beta, maxiter=350)
                if c2 < best_c5:
                    best_c5, best_h = c2, h2

    # Phase 2: Diverse restarts
    n_restarts = 40
    for i in range(n_restarts):
        if time.time() - t0 > budget_s - 120:
            break
        rtype = i % 5
        if rtype == 0:
            hr = best_h + rng.randn(n) * 0.12
        elif rtype == 1:
            hr = _make_binary_random(n, target, rng)
        elif rtype == 2:
            tm = _make_thue_morse(n)
            hr = tm + rng.randn(n) * 0.25
        elif rtype == 3:
            # Alternating block pattern with varying block sizes
            blocks = []
            pos = 0
            while pos < n:
                blen = rng.choice([2, 3, 4, 5, 6])
                val = 1.0 if len(blocks) % 2 == 0 else 0.0
                blocks.append(np.full(min(blen, n - pos), val))
                pos += blen
            hr = np.concatenate(blocks) + rng.randn(n) * 0.15
        else:
            # Cosine-based smooth
            freq = rng.uniform(1.0, 4.0)
            phase = rng.uniform(0, 2 * np.pi)
            xg = np.linspace(0, 2, n)
            hr = 0.5 + 0.4 * np.cos(2 * np.pi * freq * xg / 2.0 + phase)
            hr = hr + rng.randn(n) * 0.1
        hr = repair(hr, target)

        # Subgradient refine
        h3, c3 = subgradient(hr, target, dx, max_iter=3000, lr0=0.008, momentum=0.80, topk=3)
        if c3 < best_c5:
            best_c5, best_h = c3, h3
            # Quick SLSQP polish
            if time.time() - t0 < budget_s - 60:
                h4, c4 = slsqp_smooth(h3, target, dx, 100, maxiter=200)
                if c4 < best_c5:
                    best_c5, best_h = c4, h4

    # Phase 3: Extended subgradient from best
    if time.time() - t0 < budget_s - 80:
        h5, c5 = subgradient(best_h, target, dx, max_iter=15000, lr0=0.005, momentum=0.88, topk=4)
        if c5 < best_c5:
            best_c5, best_h = c5, h5

    # Phase 4: Final SLSQP polish with increasing beta
    if time.time() - t0 < budget_s - 20:
        for beta in [100, 300, 800]:
            if time.time() - t0 > budget_s - 15:
                break
            h6, c6 = slsqp_smooth(best_h, target, dx, beta, maxiter=250)
            if c6 < best_c5:
                best_c5, best_h = c6, h6

    best_h = repair(best_h, target)
    best_c5 = compute_c5(best_h, dx)
    return (best_h, best_c5, n)
