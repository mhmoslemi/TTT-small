import numpy as np
from scipy.optimize import minimize
import time

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def _fix_sum(h, n):
    target = 0.5 * n
    for _ in range(500):
        d = target - np.sum(h)
        if abs(d) < 1e-12 * n:
            break
        if d > 0:
            room = np.maximum(1.0 - h, 0)
            tr = room.sum()
            if tr < 1e-15:
                break
            h = h + min(1.0, d / tr) * room
        else:
            avail = np.maximum(h, 0)
            ta = avail.sum()
            if ta < 1e-15:
                break
            h = h - min(1.0, (-d) / ta) * avail
    return np.clip(h, 0.0, 1.0)

def _interp(h, n):
    n0 = len(h)
    if n0 == n:
        return _fix_sum(h.copy(), n)
    xo = np.arange(n0) * (2.0 / n0)
    xn = np.arange(n) * (2.0 / n)
    return _fix_sum(np.clip(np.interp(xn, xo, h), 0, 1), n)

def _perturb(h, n, rng, scale=0.15):
    h0 = _interp(h, n)
    return _fix_sum(np.clip(h0 + rng.randn(n) * scale, 0, 1), n)

def _topk_smooth(x, alpha, topk):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    k = min(topk, len(c))
    top = np.partition(c, -k)[-k:]
    m = top.max()
    e = np.exp(np.clip(top - m, -50.0, 0.0))
    return m + np.log(np.sum(e)) / alpha

def _slsqp(h0, alpha, topk, maxiter):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    def obj(x):
        return _topk_smooth(x, alpha, topk)
    def jac(x):
        return None
    cons = [{'type': 'eq',
             'fun': lambda x: np.sum(x) - 0.5 * n,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bnds,
                       constraints=cons, jac=jac,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None:
            return _fix_sum(np.clip(res.x, 0, 1), n)
    except Exception:
        pass
    return None

def _grad_topk(h, topk):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    k = min(topk, len(c))
    top_idx = np.argpartition(c, -k)[-k:]
    w = np.exp((c[top_idx] - c[top_idx].max()) / 0.05)
    w /= w.sum()
    grad = np.zeros(n)
    for ti, wt in zip(top_idx, w):
        s = ti - (n - 1)
        if s > 0:
            grad[s:] += wt * (1.0 - h[:n - s])
            grad[:n - s] += -wt * h[s:]
        elif s < 0:
            t = -s
            grad[:n - t] += wt * (1.0 - h[t:])
            grad[t:] += -wt * h[:n - t]
        else:
            grad += wt * (1.0 - 2.0 * h)
    return grad * dx, c

def _momentum(h0, n_iter, lr0, momentum, topk, decay=0.999):
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)
    lr = lr0
    for it in range(n_iter):
        grad, c = _grad_topk(h, topk)
        v = momentum * v - lr * grad
        h = h + v
        h = np.clip(h, 0, 1)
        h = _fix_sum(h, n)
        lr *= decay
    return h

def _coord_descent(h0, n_sweeps, n):
    h = h0.copy()
    dx = 2.0 / n
    for _ in range(n_sweeps):
        improved = False
        for i in range(n):
            old = h[i]
            c_old = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)
            for trial in [0.0, 1.0, 0.25, 0.5, 0.75, old + 0.1, old - 0.1]:
                if trial < 0 or trial > 1:
                    continue
                h[i] = trial
                h[i] = np.clip(h[i], 0, 1)
                h = _fix_sum(h, n)
                c_new = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)
                if c_new < c_old:
                    improved = True
                    c_old = c_new
            h[i] = old
        if not improved:
            break
    return h

def _coord_descent_fast(h0, n_sweeps, n, topk=5):

    dx = 2.0 / n

            h = best_val.copy() if best_val != old else _fix_sum(np.clip(h, 0, 1), n)
            if best_c < c_cur - 1e-10:

        if not any_improve:
            break
