            coarse = np.random.randn(n)
            coarse -= np.mean(coarse)
            h_pert = repair(h_best + 0.02 * coarse + 0.01 * np.random.randn(n), n)
        else:
            scale = np.random.uniform(0.001, 0.008)
            h_pert = repair(h_best + np.random.randn(n) * scale, n)
        
        # L-BFGS-B optimization
        h_new = lbfgsb_opt(h_pert, n, beta, lam, maxiter=3000)
        c = compute_c5(h_new)
        if c < c5_best - 1e-12:
            h_best, c5_best = h_new, c
        iters += 1
        
        # Periodic SLSQP refinement
        if iters % 15 == 14 and time.time() - t0 < budget_s * 0.68:
            h_ref = slsqp_opt(h_best, n, maxiter=1000)
            c = compute_c5(h_ref)
            if c < c5_best:
                h_best, c5_best = h_ref, c

    # Phase 3: Intensive gradient polish with high beta
    if time.time() - t0 < budget_s * 0.85:
        for beta in [500, 1000, 2000]:
            if time.time() - t0 > budget_s * 0.84:
                break
            h_new = lbfgsb_opt(h_best, n, beta, 0.1, maxiter=5000)
            c = compute_c5(h_new)
            if c < c5_best - 1e-12:
                h_best, c5_best = h_new, c
        # SLSQP final
        h_new = slsqp_opt(h_best, n, maxiter=5000)
        c = compute_c5(h_new)
        if c < c5_best:
            h_best, c5_best = h_new, c

    # Phase 4: Try alternative n values
    if time.time() - t0 < budget_s * 0.97:
        for n2 in [80, 60, 90, 50]:
            if time.time() - t0 > budget_s * 0.95:
                break
            x_old = np.linspace(0, 1, n)
            x_new = np.linspace(0, 1, n2)
            h_interp = np.interp(x_new, x_old, h_best)
            h_interp = repair(h_interp, n2)
            h2 = lbfgsb_opt(h_interp, n2, 100, 10.0, maxiter=3000)
            h2 = slsqp_opt(h2, n2, maxiter=2000)
            c2 = compute_c5(h2)
            if c2 < c5_best - 1e-10:
                h_best, c5_best, n = h2, c2, n2

    # Final repair and validation
    h_best = repair(h_best, n)
    c5_best = compute_c5(h_best)
    return (h_best, c5_best, n)
