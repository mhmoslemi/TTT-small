import numpy as np
from scipy.optimize import minimize
import time

_GLOBAL_N = 0
_GLOBAL_ALPHA = 100.0
_GLOBAL_TARGET = 0.0

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def _fix(h, n):
    t = 0.5 * n
    h = np.array(h, dtype=float).copy()
    for _ in range(300):
        d = t - np.sum(h)
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = (h < 1.0)
        else:
            mask = (h > 0.0)
        k = int(np.sum(mask))
        if k == 0:
            break
        h = h + (d / k) * mask
        h = np.clip(h, 0.0, 1.0)
    return h

def _interp(h, n):
    n0 = len(h)
    xo = np.arange(n0) * (2.0 / n0)
    xn = np.arange(n) * (2.0 / n)
    return _fix(np.clip(np.interp(xn, xo, h), 0.0, 1.0), n)

def _obj_fun(x):
    n = _GLOBAL_N
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(c)
    e = np.exp(np.clip(c - m, -80.0, 80.0))
    return m + np.log(np.sum(e)) / _GLOBAL_ALPHA

def _cons_fun(x):
    return np.sum(x) - _GLOBAL_TARGET

def _cons_jac(x):
    return np.ones(_GLOBAL_N)

def _run_slsqp(h0, n, alpha, maxiter, t0, budget_s):
    global _GLOBAL_N, _GLOBAL_ALPHA, _GLOBAL_TARGET
    if budget_s - (time.time() - t0) < 8:
        return None
    _GLOBAL_N = n
    _GLOBAL_ALPHA = float(alpha)
    _GLOBAL_TARGET = 0.5 * n
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(_obj_fun, h0.copy(), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None:
            return _fix(np.clip(res.x, 0.0, 1.0), n)
    except Exception:
        pass
    return None

def _structured_starts(n, rng):
    x = np.linspace(0, 2, n)
    starts = []
    for freq in [1, 2, 3, 4, 5]:
        for amp in [0.25, 0.35, 0.45]:
            for ph in [0.0, np.pi / 2.0, np.pi]:
                h = np.clip(0.5 + amp * np.sin(np.pi * freq * x + ph), 0, 1)
                starts.append(_fix(h, n))
    # Two-block pattern
    for w in [0.3, 0.4, 0.5, 0.6]:
        h = np.zeros(n)
        h[:int(n * w)] = 1.0
        if n * (1 - w) > n * w:
            h = np.clip(h - 0.5, 0, 1) * 2.0
        starts.append(_fix(h, n))
    return starts

def run(seed=42, budget_s=1100, **kwargs):

    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:

    n_par = len(h_par)
    h_par = _fix(h_par, n_par)

    best_h = h_par.copy()
    best_c = _c5(best_h)

    res_list = [r for r in res_list if r <= 500]

    alphas = [30.0, 80.0, 250.0, 800.0, 3000.0, 15000.0]

    for n in res_list:
        if budget_s - (time.time() - t0) < 40:
            break
        # Build starting points
        starts = []
        if n == n_par:
            starts.append(h_par.copy())
        else:
            starts.append(_interp(h_par, n))
        if len(best_h) != n:
            starts.append(_interp(best_h, n))
        else:
            starts.append(best_h.copy())
        # Structured
        for hs in _structured_starts(n, rng):

        for _ in range(3):

            starts.append(hp)

                    best_c = ct
    best_h = _fix(best_h, len(best_h))
    best_c = _c5(best_h)
    return best_h, best_c, len(best_h)
