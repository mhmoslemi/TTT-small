import time
from scipy.optimize import minimize

_g_n = 50
_g_active = None
_g_p = 8

def compute_score(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(c)

def compute_all_scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(2000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.sum(mask).astype(float)
        if cnt == 0:
            break
        h = np.clip(h + diff / cnt * mask, 0.0, 1.0)
    return h

def _constraint_fun(h):
    return np.sum(h) - _g_n / 2.0

def _obj_power(h):
    c = compute_all_scores(h)
    ac = c[_g_active]
    return np.sum(ac ** _g_p) ** (1.0 / _g_p)

def _obj_max_direct(h):
    return compute_score(h)

def active_set_optimize(h, n, p_list, max_iter_each, rng, budget_frac, t0, budget_s):
    global _g_n, _g_active, _g_p
    _g_n = n

    cons = [{'type': 'eq', 'fun': _constraint_fun}]
    
    for p in p_list:

                sc = compute_score(hc)
                if sc < best_s:
                    best_s = sc
                    best = hc
            except Exception:
                pass
