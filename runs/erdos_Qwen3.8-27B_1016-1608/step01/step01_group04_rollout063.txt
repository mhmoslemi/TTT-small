    h = np.array(h, dtype=float).copy()

        d = n / 2.0 - np.sum(h)
        if abs(d) < 1e-14:
            break
        h += d / n
    h = np.clip(h, 0.0, 1.0)

def lag_grad(x, k, n, dx):

    l = k - (n - 1)
    g = np.zeros(n)
    # d(C_k)/dh[j] = dx * [(1-x[j-l]) * I(0≤j-l<n) - x[j+l] * I(0≤j+l<n)]
    # Part 1: j-l in [0,n) => j in [l, l+n) intersect [0,n)
    lo1 = max(0, l)
    hi1 = min(n, l + n)
    if lo1 < hi1:
        j = np.arange(lo1, hi1)
        g[lo1:hi1] += (1.0 - x[j - l])

    lo2 = max(0, -l)
    hi2 = min(n, n - l)
    if lo2 < hi2:
        j = np.arange(lo2, hi2)
        g[lo2:hi2] += (-x[j + l])
    return g * dx

    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    res = minimize(smax_val, h, jac=smax_jac, method='SLSQP',
                   bounds=bds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return repair(res.x, n)

def hinge_obj(x):
    T = _ctx['T']
    dx = 2.0 / _ctx['n']
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    excess = np.maximum(0.0, c - T)
    return 0.5 * np.sum(excess ** 2)

def hinge_jac(x):
    T = _ctx['T']
    n = _ctx['n']
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    active = c > T
    g = np.zeros(n)
    for k in range(len(c)):
        if active[k]:
            g += (c[k] - T) * lag_grad(x, k, n, dx)
    return g

def hinge_opt(h, n, T, maxiter=300, ftol=1e-14):
    _ctx['n'] = n
    _ctx['T'] = T
    bds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    res = minimize(hinge_obj, h, jac=hinge_jac, method='SLSQP',
                   bounds=bds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return repair(res.x, n)

def subgrad_polish(h, n, steps=3000, max_setup=1.5):
    dx = 2.0 / n
    bh = h.copy()
    bc = compute_c5(bh)
    for it in range(steps):
        c = all_corrs(bh)
        m = np.max(c)
        top_k = np.argsort(c)[-6:]
        w = np.ones(len(top_k)) / len(top_k)
        g = np.zeros(n)
        for i, k in enumerate(top_k):
            g += w[i] * lag_grad(bh, k, n, dx)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        step = max_setup / (it + 20.0) ** 0.7
        for alpha in [step, step * 0.5, step * 0.1]:
            h_new = bh - alpha * g / gn
            h_new = repair(h_new, n)
            c_new = compute_c5(h_new)
            if c_new < bc:
                bh = h_new
                bc = c_new
                break
    return bh, bc

def interp_h(h_c, n_fine):
    n_c = len(h_c)
    xc = np.linspace(0.0, 2.0, n_c, endpoint=False)
    xf = np.linspace(0.0, 2.0, n_fine, endpoint=False)
    # Periodic interpolation
    hc_ext = np.concatenate([h_c[-1:], h_c, h_c[:1]])
    xc_ext = np.concatenate([[-2.0/n_c], xc, [2.0 - 2.0/n_c]])
    # Simple linear with wrap
    h_f = np.zeros(n_fine)
    for i in range(n_fine):
        x = xf[i]
        # Find segment in periodic domain
        xi = x
        while xi >= 2.0:
            xi -= 2.0
        j = int(xi / (2.0 / n_c))
        frac = (xi - j * 2.0 / n_c) / (2.0 / n_c)
        h_f[i] = (1 - frac) * h_c[j % n_c] + frac * h_c[(j + 1) % n_c]
    return h_f

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    def elapsed():
        return time.time() - t0

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.tile([1.0, 0.5, 1.0, 0.0, 1.0, 0.0, 0.5, 0.0], 10)
    n = len(h0)
    h0 = repair(h0, n)
    bh = h0.copy()
    bc = compute_c5(bh)

    # Phase 1: SLSQP at native resolution with fine eps schedule
    eps_list = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.00025, 0.0001, 0.00005]
    for eps in eps_list:
        if elapsed() > budget_s - 500:
            break
        hn = slsqp_opt(bh, n, eps, 400, 1e-13)
        c5 = compute_c5(hn)
        if c5 < bc:
            bc, bh = c5, hn

    # Phase 1b: Structured restarts
    patterns = [
        lambda nn: 0.5 + 0.45 * np.sin(2 * np.pi * 3 * np.arange(nn) / nn),
        lambda nn: 0.5 + 0.4 * np.cos(2 * np.pi * 2 * np.arange(nn) / nn + 0.7),
        lambda nn: 0.5 + 0.5 * np.sin(2 * np.pi * np.arange(nn) / nn + np.pi / 4),
        lambda nn: np.tile([1, 0, 1, 0, 0.5, 0, 0.5, 0], nn // 8)[:nn],
        lambda nn: np.tile([0, 1, 0, 1, 1, 0, 1, 0], nn // 8)[:nn],
        lambda nn: 0.5 + 0.35 * np.sin(4 * np.pi * np.arange(nn) / nn) * np.cos(2 * np.pi * np.arange(nn) / nn),
    ]
    for pat in patterns:
        if elapsed() > budget_s - 350:
            break
        hp = repair(pat(n), n)
        hn = slsqp_opt(hp, n, 0.01, 300, 1e-13)
        hn = slsqp_opt(hn, n, 0.001, 200, 1e-13)
        hn = slsqp_opt(hn, n, 0.0003, 200, 1e-13)
        c5 = compute_c5(hn)
        if c5 < bc:
            bc, bh = c5, hn
        # Perturbed best
        if elapsed() > budget_s - 350:
            break
        hp2 = bh + np.random.randn(n) * 0.04
        hp2 = repair(hp2, n)
        hn2 = slsqp_opt(hp2, n, 0.008, 250, 1e-13)
        hn2 = slsqp_opt(hn2, n, 0.001, 200, 1e-13)
        c5_2 = compute_c5(hn2)
        if c5_2 < bc:
            bc, bh = c5_2, hn2

    # Phase 1c: Hinge-loss optimization (target slightly below current best)
    if elapsed() < budget_s - 250:
        for frac in [1.0, 0.999, 0.998, 0.997, 0.996]:
            T = bc * frac
            hn = hinge_opt(bh, n, T, 300, 1e-14)
            c5 = compute_c5(hn)
            if c5 < bc:
                bc, bh = c5, hn

    # Phase 1d: Subgradient polish
    if elapsed() < budget_s - 200:
        bh, bc = subgrad_polish(bh, n, steps=4000, max_setup=1.2)
        # Quick SLSQP after subgradient
        if elapsed() < budget_s - 150:
            bh = slsqp_opt(bh, n, 0.0002, 300, 1e-14)
            bc = compute_c5(bh)
            bh, bc = subgrad_polish(bh, n, steps=3000, max_setup=0.8)

    # Phase 2: Higher resolution (n*2)
    n2 = n * 2
    if elapsed() < budget_s - 300:
        h2 = interp_h(bh, n2)
        h2 = repair(h2, n2)
        b2 = h2.copy()
        bc2 = compute_c5(b2)
        for eps in [0.03, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
            if elapsed() > budget_s - 120:
                break
            b2 = slsqp_opt(b2, n2, eps, 300, 1e-13)
            c5_2 = compute_c5(b2)
            bc2 = min(bc2, c5_2)
        if elapsed() < budget_s - 80:
            T2 = bc2 * 0.999
            b2 = hinge_opt(b2, n2, T2, 300, 1e-14)
            bc2 = compute_c5(b2)
        if elapsed() < budget_s - 40:
            b2, bc2 = subgrad_polish(b2, n2, steps=2000, max_setup=0.7)
        if bc2 < bc:
            bh, bc, n = b2, bc2, n2

    # Phase 3: Final intensive polish
    if elapsed() < budget_s - 30:
        bh = slsqp_opt(bh, n, 0.0001, 500, 1e-15)
        bc = compute_c5(bh)
    if elapsed() < budget_s - 20:
        T = bc * 0.9995
        bh = hinge_opt(bh, n, T, 400, 1e-15)
        bc = compute_c5(bh)
    if elapsed() < budget_s - 10:
        bh, bc = subgrad_polish(bh, n, steps=5000, max_setup=0.5)
    if elapsed() < budget_s - 5:
        bh = slsqp_opt(bh, n, 0.00005, 300, 1e-15)
        bc = compute_c5(bh)

    bh = repair(bh, n)
    bc = compute_c5(bh)
    return (bh.tolist(), float(bc), n)
