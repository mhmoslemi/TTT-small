def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

def compute_c5_smooth(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    # log-sum-exp for numerical stability
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(corr - m))) / T

def project_onto_feasible(h, target):
    # Project h onto {0<=h<=1, sum(h)=target}
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        hp = np.clip(h - mid, 0.0, 1.0)
        if float(np.sum(hp)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def subgradient_step(h, n, lr):
    # Compute all correlation values
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    k_star = int(np.argmax(corr))
    # Gradient of g_{k_star} w.r.t. h
    # g_k(h) = dx * sum_i h[i] * (1-h[i+k]) for shift k in full correlation
    # In full mode, index k corresponds to shift (k - n + 1)
    shift = k_star - (n - 1)
    grad = np.zeros(n)
    for j in range(n):
        g = 0.0
        # h[j] appears as h[i] when i=j: contributes -h[j+shift] to (1-h[i+shift])
        if 0 <= j + shift < n:
            g += -h[j + shift]
        # h[j] appears as h[i+shift] when i+shift=j i.e. i=j-shift: contributes (1-h[i]) = (1-h[j-shift])
        if 0 <= j - shift < n:
            g += (1.0 - h[j - shift])
        grad[j] = dx * g
    return h - lr * grad, k_star

def lbfgs_optimize_smooth(h0, n, T, maxiter=200):

        m = np.max(corr)
        return m + np.log(np.sum(np.exp(corr - m))) / T

        res = minimize(obj, h0.copy(), method='L-BFGS-B',
                      bounds=[(0.0, 1.0)] * n,
                      options={'maxiter': maxiter, 'ftol': 1e-14})

        return h_out, compute_c5(h_out, n)
    except Exception:
        return h0, compute_c5(h0, n)

    best = compute_c5(h, n)

    lr = lr_init

    return best_h, best

def smooth_obj(x, n, T, M):

    target = n / 2.0
    dx = 2.0 / n
    hp = np.clip(x, 0.0, 1.0)

Let me also think about whether the penalty approach for L-BFGS-B is good. The penalty M*(sum(h)-target)^2 with M=1000 should be strong enough. But it might interfere with the optimization. Let me use M=500 and check.

Actually, a cleaner approach: I can parameterize h to automatically satisfy the constraint. Let h = clip(x, 0, 1) and then adjust: h = clip(x - lambda, 0, 1) where lambda is chosen so sum(h) = target. This is exactly what project_feasible does! So I can:
1. Run L-BFGS-B on x (unconstrained except bounds 0-1)
2. After each evaluation, project to feasible set
3. Use the projected value for the objective

This is a "projection-embedded" L-BFGS-B. It's not mathematically clean (the optimizer sees a non-smooth objective due to projection), but in practice it works well because the projection is a smooth operation away from the boundaries.

Let me implement this:

```python
def proj_obj(x, n, T, M):
    target = n / 2.0
    dx = 2.0 / n
    hp = project_feasible(x, target)  # This projects to sum=target, 0<=h<=1
    corr = np.correlate(hp, 1.0 - hp, mode="full") * dx
    m = np.max(corr)
    smooth = m + np.log(np.sum(np.exp(corr - m))) / T
    return smooth
```

But this creates a non-smooth objective for L-BFGS-B. The projection is piecewise-linear (the binary search on shift gives a piecewise-linear result). This might confuse L-BFGS-B.

Alternative: just use the penalty approach. It's simpler and L-BFGS-B handles it fine.

import numpy as np
from scipy.optimize import minimize
import time

_budget = 1100.0
_start_t = 0.0

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def upsample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, h)

def smooth_obj(x, n, T, M):
    target = n / 2.0
    dx = 2.0 / n
    hp = np.clip(x, 0.0, 1.0)
    corr = np.correlate(hp, 1.0 - hp, mode="full") * dx
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(corr - m))) / T
    val += M * (np.sum(hp) - target) ** 2
    return val

def lbfgs_smooth(h0, n, T, maxiter, M=500.0):
    target = n / 2.0
    h0 = project_feasible(h0.copy(), target)
    try:
        res = minimize(smooth_obj, h0.copy(), args=(n, T, M),
                      method='L-BFGS-B',
                      bounds=[(0.0, 1.0)] * n,
                      options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-8})
        h_out = project_feasible(res.x, target)
        return h_out, compute_c5(h_out, n)
    except Exception:
        return h0, compute_c5(h0, n)

def subgradient_descent(h0, n, max_iter, lr_init):
    target = n / 2.0
    dx = 2.0 / n
    h = project_feasible(h0.copy(), target)
    best = compute_c5(h, n)
    best_h = h.copy()
    lr = lr_init
    stag = 0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_star = int(np.argmax(corr))
        s = k_star - (n - 1)
        grad = np.zeros(n)
        if s >= 0:
            grad[s:] += (1.0 - h[:n - s])
            grad[:n - s] += -h[s:]
        else:
            ss = -s
            grad[:n - ss] += (1.0 - h[ss:])
            grad[ss:] += -h[:n - ss]
        grad *= dx
        h = h - lr * grad
        h = project_feasible(h, target)
        c = compute_c5(h, n)
        if c < best - 1e-12:
            best = c
            best_h = h.copy()
            stag = 0
        else:
            stag += 1
        if stag % 80 == 0:
            lr *= 0.82
        if lr < 1e-7:
            break
    return best_h, best

def coord_search(h, n, rng, sweeps):
    target = n / 2.0
    h = h.copy()
    best = compute_c5(h, n)
    for _ in range(sweeps):
        improved = False
        order = rng.permutation(n)
        for j in order:
            ov = h[j]
            shifts = [0.0, 0.1, 0.05, 0.02, 0.01, -0.1, -0.05, -0.02, -0.01, 1.0 - ov, -(1.0 - ov)]
            bv = best
            bh = ov
            for d in shifts:
                c = ov + d
                if c < 0.0 or c > 1.0 or abs(c - ov) < 1e-8:
                    continue
                h[j] = c
                h = project_feasible(h, target)
                v = compute_c5(h, n)
                if v < bv:
                    bv = v
                    bh = c
            h[j] = bh
            h = project_feasible(h, target)
            if bv < best:
                best = bv
                improved = True
        if not improved:
            break
    return h, best
