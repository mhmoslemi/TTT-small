```python
    dx = 2.0/n
    for it in range(max_iter):
        corr = np.correlate(h, 1-h, mode='full') * dx
        k_max = np.argmax(corr)
        f_val = corr[k_max]
        
        s = k_max - n + 1
        # Gradient of f_s w.r.t. h
        grad = np.zeros(n)
        # (1 - h[i+s])*dx for valid i
        if s >= 0:
            valid_i = range(0, n - s)
            grad[valid_i] += (1 - h[s:n]) * dx  # h[i+s] for i in [0, n-s)
        else:
            valid_i = range(-s, n)
            grad[valid_i] += (1 - h[0:n+s]) * dx  # h[i+s] for i in [-s, n)
        
        # -h[i-s]*dx for valid i
        if s >= 0:
            valid_i = range(s, n)
            grad[valid_i] -= h[0:n-s] * dx
        else:
            valid_i = range(0, n+s)
            grad[valid_i] -= h[-s:n] * dx
        
        # Frank-Wolfe: find y that minimizes grad^T y s.t. sum(y)=n/2, 0<=y<=1

        idx = np.argsort(grad)[:n_half]

        y[idx] = 1.0

        t = 2.0 / (it + 3)  # standard FW step
        # Or use line search
        h_new = (1 - t) * h + t * y
        h_new = np.clip(h_new, 0, 1)
        h_new = repair(h_new, n)
        
        # Check improvement
        if compute_c5(h_new) < compute_c5(h):  # or just update
            h = h_new
    
    return h
```

```python
        else:

            break
```

```python
def soft_max_obj(h, beta):
    n = len(h)
    dx = 2.0/n
    corr = np.correlate(h, 1.0-h, mode='full') * dx
    # Soft max
    max_corr = np.max(corr)
    exps = np.exp(beta * (corr - max_corr))
    return max_corr + np.log(np.sum(exps)) / beta

def soft_max_grad(h, beta):
    n = len(h)
    dx = 2.0/n
    corr_raw = np.correlate(h, 1.0-h, mode='full')
    corr = corr_raw * dx
    max_corr = np.max(corr)
    exps = np.exp(beta * (corr - max_corr))
    weights = exps / np.sum(exps)  # softmax weights
    
    # Weighted gradient
    grad = np.zeros(n)
    for k in range(len(corr)):
        if weights[k] < 1e-10:
            continue
        s = k - n + 1
        g = np.zeros(n)

        else:
            g[-s:] += (1.0 - h[:n+s]) * dx
            g[:n+s] -= h[-s:] * dx
        grad += weights[k] * g
    return grad
```

```python
    corr = np.correlate(h, 1.0-h, mode='full') * dx  # f_k values
    
    # Compute A_k = (1-h[j+s_k])*valid1 - h[j-s_k]*valid2 for each k
    A = np.zeros(2*n-1)
    for k, s in enumerate(s_all):
        valid1 = (0 <= j+s < n)
        valid2 = (0 <= j-s < n)
        A[k] = (1-h[j+s]) * valid1 - h[j-s] * valid2
    A *= dx

    # (since current h[j] is the value, f_k =corr[k] = f_k(0) + A_k * h[j])
    f0 = corr - A * h[j]  # f_k when h[j]=0
    
    # We want to minimize max_k (f0[k] + A[k]*v) over v in [0,1]
    # The value at v=0: max_k f0[k]
    # The value at v=1: max_k (f0[k]+A[k])
    val_0 = np.max(f0)
    val_1 = np.max(f0 + A)
    
    # If val_0 <= val_1 and all top-constraint A's > 0, optimum at v=0
    # If val_1 < val_0 and all top-constraint A's < 0, optimum at v=1
    # Otherwise, find intersection
    
    # Simple approach: try v=0, v=1, and the current value
    # Also try the "equalize" point
    # Find k* that argmax at current v
    k_star = np.argmax(corr)
    
    # The optimal v is where the constraint functions are most balanced
    # For 2 constraints: f0[k]+A[k]*v = f0[l]+A[l]*v
    # v = (f0[l]-f0[k])/(A[k]-A[l])
    
    # Top-2 lags
    order = np.argsort(f0 + A * h[j])[::-1]  # at current v, sorted desc
    if len(order) >= 2:
        k1, k2 = order[0], order[1]
        if abs(A[k1] - A[k2]) > 1e-15:
            v_eq = (f0[k2] - f0[k1]) / (A[k1] - A[k2])
            v_eq = np.clip(v_eq, 0, 1)
        else:
            v_eq = h[j]
    else:
        v_eq = h[j]
    
    # Evaluate at candidates
    best_v = h[j]
    best_val = np.max(corr)
    for v_cand in [0.0, 1.0, v_eq, h[j]]:
        val = np.max(f0 + A * v_cand)
        if val < best_val:
            best_val = val
            best_v = v_cand
    
    return best_v
```

```python
        g[valid_m] -= x[idx_m[valid_m]] * dx2
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            h[:] = target / n
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_descent(h, n, t_limit):
    dx = 2.0 / n
    s_arr = np.arange(2 * n - 1) - n + 1
    t0 = time.time()
    n_passes = 0
    while time.time() - t0 < t_limit:
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        improved = False
        for j in range(n):
            idx_p = j + s_arr
            idx_m = j - s_arr
            valid_p = (idx_p >= 0) & (idx_p < n)
            valid_m = (idx_m >= 0) & (idx_m < n)
            A = np.zeros(2 * n - 1)
            if np.any(valid_p):
                A[valid_p] = (1.0 - h[idx_p[valid_p]]) * dx
            if np.any(valid_m):
                A[valid_m] -= h[idx_m[valid_m]] * dx
            f0 = corr - A * h[j]
            best_v = h[j]
            best_val = float(np.max(f0))
            val_1 = float(np.max(f0 + A))
            if val_1 < best_val - 1e-13:
                best_val = val_1
                best_v = 1.0
            cur_vals = f0 + A * h[j]
            top2 = np.argsort(cur_vals)[-2:]
            k1, k2 = int(top2[1]), int(top2[0])
            denom = A[k1] - A[k2]

                best_v = 0.0
            elif best_v > 1.0 - 1e-12:
                best_v = 1.0
            if abs(best_v - h[j]) > 1e-14:
                h[j] = best_v
                improved = True
        h = repair(h, n)
        n_passes += 1
        if not improved and n_passes > 1:
            break
    return h, n_passes

def obj_func(x):
    n2 = len(x)
    dx2 = 2.0 / n2
    corr = np.correlate(x, 1.0 - x, mode='full')
    return float(np.max(corr) * dx2)

def gradient_func(x):
    n2 = len(x)
    dx2 = 2.0 / n2
    corr_raw = np.correlate(x, 1.0 - x, mode='full')
    k_max = int(np.argmax(corr_raw))
    s_k = k_max - n2 + 1
    idx = np.arange(n2) + s_k
    valid_p = (idx >= 0) & (idx < n2)
    idx2 = np.arange(n2) - s_k
    valid_m = (idx2 >= 0) & (idx2 < n2)
    g = np.zeros(n2)
    if np.any(valid_p):
        g[valid_p] += (1.0 - x[idx[valid_p]]) * dx2
    if np.any(valid_m):
        g[valid_m] -= x[idx2[valid_m]] * dx2
    return g

def constraint_func(x):
    return np.sum(x) - len(x) / 2.0

def slsqp_optimize(h0, n, maxiter, ftol):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_func}]
    try:
        res = minimize(obj_func, h0, jac=gradient_func, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return repair(np.clip(res.x, 0.0, 1.0), n)
    except Exception:
        return h0

def soft_max_obj(x, beta):
    n2 = len(x)
    dx2 = 2.0 / n2
    corr = np.correlate(x, 1.0 - x, mode='full') * dx2
    m = np.max(corr)
    return float(m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta)

def soft_max_grad(x, beta):
    n2 = len(x)
    dx2 = 2.0 / n2
    corr = np.correlate(x, 1.0 - x, mode='full') * dx2
    m = np.max(corr)
    w = np.exp(beta * (corr - m))
    w /= np.sum(w)
    grad = np.zeros(n2)
    for k in range(2 * n2 - 1):
        if w[k] < 1e-10:
            continue
        s_k = k - n2 + 1
        idx_p = np.arange(n2) + s_k
        vp = (idx_p >= 0) & (idx_p < n2)
        idx_m = np.arange(n2) - s_k
        vm = (idx_m >= 0) & (idx_m < n2)
        g = np.zeros(n2)
        if np.any(vp):
            g[vp] += (1.0 - x[idx_p[vp]]) * dx2
        if np.any(vm):
            g[vm] -= x[idx_m[vm]] * dx2
        grad += w[k] * g
    return grad

def slsqp_soft(h0, n, beta, maxiter):
    def soft_wrap(x):
        return soft_max_obj(x, beta)
    def soft_grad_wrap(x):
        return soft_max_grad(x, beta)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_func}]
    try:
        res = minimize(soft_wrap, h0, jac=soft_grad_wrap, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        return repair(np.clip(res.x, 0.0, 1.0), n)
    except Exception:
        return h0
```
