        s_lo = x - n + 1
        s_hi = x

def slsqp_polish(h, n, maxiter=1000):

        dx = 2.0 / n
        return float(np.max(np.correlate(x, 1.0 - x, mode='full')) * dx)
    
    def obj_grad(x):

        dx = 2.0 / n
        corr = np.correlate(x, 1.0 - x, mode='full')
        # The active constraint
        k = np.argmax(corr)
        # Gradient of corr[k] w.r.t. x
        g = np.zeros(n)
        s = k - (n - 1)
        for i in range(n):

            hi = min(n - 1, n - 1 + s)

                val = 0.0
                if 0 <= i - s < n:
                    val += (1.0 - x[i - s])
                if 0 <= i + s < n:
                    val -= x[i + s]
                g[i] = val
        return g * dx

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}]
    try:

                       constraints=cons, jac=obj_grad,

        if res.fun < compute_c5(h) + 1e-14:
            return repair(np.clip(res.x, 0.0, 1.0), n)

        pass

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not np.any(mask):
            h[:] = target / n
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_g(h):
    n = len(h)
    g = np.zeros((n, 2 * n - 1))
    for x in range(n):
        s_vals = np.arange(x - n + 1, x + 1)
        idx = s_vals + (n - 1)
        x_ms = np.arange(n - 1, -1, -1)

            g[x, idx_sub] -= h[x_ps]
