                       options={'maxiter': maxiter, 'ftol': 1e-13})
        if res.x is not None:
            return _project(np.clip(res.x, 0, 1), target)
    except Exception:
        pass
    return None

def _interp(h, n_new):
    n_old = len(h)
    xo = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    xn = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(xn, xo, h)
    return _project(np.clip(h_new, 0, 1), 0.5 * n_new)

rng_global = np.random.RandomState(42)

def run(seed=42, budget_s=1100, **kwargs):
    global rng_global
    rng_global = np.random.RandomState(seed)
    t0 = time.time()

    def remaining():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        h = np.full(160, 0.5)
    n = len(h)
    target = 0.5 * n
    h = _project(h, target)
    bh = h.copy()
    bc = _c5(bh)

    # Phase 1: PG at original resolution, increasing alpha
    for alpha in [15.0, 40.0, 100.0, 300.0, 800.0, 2000.0]:
        if remaining() < 25:
            break
        for restart in range(4):
            if remaining() < 15:
                break
            h_cur = bh.copy() if restart == 0 else bh.copy()
            ht = _pg_solve(h_cur, alpha, 4000, t0, budget_s,
                           lr_init=0.08 if alpha < 100 else 0.03,
                           noise=0.03 if restart > 0 else 0.0)
            ct = _c5(ht)
            if ct < bc:
                bh, bc = ht.copy(), ct
            if remaining() > 20 and ct < bc + 0.005:
                hp = _slsqp_opt(ht, alpha, 600, t0, budget_s)
                if hp is not None:
                    ct2 = _c5(hp)
                    if ct2 < bc:
                        bh, bc = hp.copy(), ct2

    # Phase 2: Resolution sweep - try both up and down
    resolutions = sorted(set([max(80, n // 2), n, n + 40, n + 80, min(512, n + 160)]))
    for nn in resolutions:
        if remaining() < 30:
            break
        h_new = _interp(bh, nn)
        ct_new = _c5(h_new)
        if ct_new < bc:
            bh, bc = h_new.copy(), ct_new
        tn = 0.5 * nn
        for alpha in [60.0, 250.0, 1000.0]:
            if remaining() < 15:
                break
            ht = _pg_solve(h_new.copy(), alpha, 3000, t0, budget_s,
                           lr_init=0.06 if alpha < 200 else 0.02)
            ct = _c5(ht)
            if ct < bc:
                bh = ht.copy()
                bc = ct
                h_new = ht.copy()
        if remaining() > 20:
            hp = _slsqp_opt(h_new.copy(), 300.0, 500, t0, budget_s)
            if hp is not None:
                ct = _c5(hp)
                if ct < bc:
                    bh, bc = hp.copy(), ct
                    h_new = hp.copy()

    # Phase 3: Multi-start at best resolution
    nb = len(bh)
    for trial in range(6):
        if remaining() < 10:
            break
        if trial == 0:
            h_start = bh.copy()
        else:
            h_start = _project(bh + rng_global.randn(nb) * rng_global.uniform(0.02, 0.12), 0.5 * nb)
        for alpha in [200.0, 800.0, 3000.0]:
            if remaining() < 8:
                break
            ht = _pg_solve(h_start, alpha, 2500, t0, budget_s, lr_init=0.02)
            ct = _c5(ht)
            if ct < bc:
                bh, bc = ht.copy(), ct
                h_start = ht.copy()

    # Phase 4: Final SLSQP polish
    for alpha in [500.0, 3000.0]:
        if remaining() < 8:
            break
        hp = _slsqp_opt(bh.copy(), alpha, 800, t0, budget_s)
        if hp is not None:
            ct = _c5(hp)
            if ct < bc:
                bh, bc = hp.copy(), ct

    # Final constraint enforcement
    bh = _project(bh, 0.5 * len(bh))
    bc = _c5(bh)
    return bh, bc, len(bh)
