def compute_corr(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not np.any(mask):
            h[:] = target / n
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def sum_constraint(x):
    return np.sum(x) - len(x) / 2.0

    corr_scaled = corr * dx / temp
    m = np.max(corr_scaled)

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not np.any(mask):
            h[:] = target / n
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def sum_eq(x):
    return np.sum(x) - len(x) / 2.0

def softmax_obj(h, temp):
    corr = np.correlate(h, 1.0 - h, mode='full')
    n = len(h)
    dx = 2.0 / n
    v = corr * dx / temp
    m = np.max(v)
    return temp * (m + np.log(np.sum(np.exp(v - m))))

def slsqp_opt(h0, n, maxiter, obj_fn=None):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq}]
    if obj_fn is None:
        obj = compute_c5
    else:
        obj = obj_fn
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return repair(np.clip(res.x, 0.0, 1.0), n)
    except Exception:
        pass
    return h0

def make_softmax_obj(h_ref, temp):
    n = len(h_ref)
    # Can't use lambda, so we use a wrapper class
    class SO:
        def __call__(self, x):
            return softmax_obj(x, temp)

    k_max = np.argmax(corr)
    shift = k_max - n + 1
    h_p = h.copy()
    for _ in range(8):
        i = rng.randint(0, n)
        j = (i + shift) % n
        h_p[i] += rng.uniform(-0.08, 0.08)
        h_p[j] -= rng.uniform(-0.08, 0.08)
    return repair(h_p, n)

def coord_descend(h, n, steps_per_dim=7, max_sweeps=3):
    h = h.copy()
    best_c = compute_c5(h)
    deltas = np.linspace(-0.05, 0.05, steps_per_dim)
    for sweep in range(max_sweeps):
        improved = False
        order = np.random.permutation(n)
        for i in order:
            base = h[i]
            for d in deltas:

                c = compute_c5(h)
                if c < best_c - 1e-14:

                    improved = True
                else:
