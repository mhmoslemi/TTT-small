            # grad[j] += (1 - h[j+s]) for j >= -s, j < n

        # Term 2: -h[j-s] if 0 <= j-s < n
        
        if s >= 0:

def _fix_sum(h, n):
    target = 0.5 * n
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        d = target - np.sum(h)
        if abs(d) < 1e-12:
            break
        if d > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        k = np.sum(mask)
        if k == 0:
            break
        h[mask] += d / k
    return np.clip(h, 0.0, 1.0)

def _interp(h, n_new):
    n_old = len(h)
    xo = np.arange(n_old)
    xn = np.arange(n_new) * n_old / n_new
    h_new = np.interp(xn, xo, h)
    return _fix_sum(h_new, n_new)

def _smooth_obj(h, alpha):
    dx = 2.0 / len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    v = c * dx
    m = np.max(v)
    e = np.exp(np.clip(v - m, -50.0, 50.0))
    return m + np.log(np.sum(e)) / alpha

_N_global = 400

def _icf(x):
    return np.sum(x) - _N_global * 0.5

def _icj(x):
    return np.ones(_N_global)

def _obj_wrap(x, alpha):
    global _N_global
    return _smooth_obj(x, alpha)

def _slsqp_opt(h0, n, alpha, maxiter, t0, budget):
    global _N_global
    if budget - (time.time() - t0) < 15:
        return None
    _N_global = n
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _icf, 'jac': _icj}]
    try:
        res = minimize(_obj_wrap, h0, (alpha,), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is None or not np.isfinite(res.x).all():
            return None
        return _fix_sum(np.clip(res.x, 0.0, 1.0), n)

import numpy as np
from scipy.optimize import minimize
import time

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def _fix_sum(h, n):
    target = 0.5 * n
    for _ in range(1000):
        h = np.clip(h, 0.0, 1.0)
        d = target - np.sum(h)
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        k = np.sum(mask)
        if k == 0:
            break
        h[mask] += d / k
    return np.clip(h, 0.0, 1.0)

def _interp(h, n_new):
    n_old = len(h)
    xo = np.arange(n_old)
    xn = np.arange(n_new) * n_old / n_new
    h_new = np.interp(xn, xo, h)
    return _fix_sum(h_new, n_new)

def _smooth_value(h, alpha):
    dx = 2.0 / len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    v = c * dx
    m = np.max(v)
    e = np.exp(np.clip(v - m, -50.0, 50.0))
    return m + np.log(np.sum(e)) / alpha

_Ng = 400

def _icf(x):
    return np.sum(x) - _Ng * 0.5

def _icj(x):
    return np.ones(_Ng)

def _obj_wrap(x, alpha):
    return _smooth_value(x, alpha)

def _slsqp(h0, n, alpha, maxiter, t0, bgt):
    global _Ng
    if bgt - (time.time() - t0) < 20:
        return None
    _Ng = n
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _icf, 'jac': _icj}]
    try:
        res = minimize(_obj_wrap, h0, (alpha,), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.x is None or not np.isfinite(res.x).all():
            return None
        return _fix_sum(np.clip(res.x, 0.0, 1.0), n)
    except Exception:
        return None

def _worst_shift_subgrad(h, n, n_iter, t0, bgt, n_worst=5):
    dx = 2.0 / n
    best = h.copy()
    best_c = _c5(best)
    lr = 0.08
    for it in range(n_iter):
        if bgt - (time.time() - t0) < 8:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argsort(c)[-n_worst:][::-1]
        moved = False
        for k_star in idx:
            s = int(k_star - (n - 1))
            grad = np.zeros(n)
            if s >= 0:
                grad[:n-s] += (1.0 - h[s:])
                if s > 0:
                    grad[s:] -= h[:n-s]
            else:
                sp = -s
                grad[sp:] += (1.0 - h[:n-sp])
                grad[:n-sp] -= h[sp:]
            grad *= dx
            h_new = _fix_sum(h - lr * grad, n)
            cn = _c5(h_new)
            if cn < best_c:
                best = h_new
                best_c = cn
                h = h_new
                moved = True
        if not moved and it > 50:
            lr *= 0.95
    return best, best_c

def _pairwise_swap(h, n, n_trials, t0, bgt, rng):
    best = h.copy()
    best_c = _c5(best)
    for _ in range(n_trials):
        if bgt - (time.time() - t0) < 8:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(n - 1)) % n
        hi = best[i]
        hj = best[j]
        if abs(hi - hj) < 1e-6:
            continue
        delta = min(0.5 * abs(hi - hj), 0.15)
        if hi > hj:
            best[i] = np.clip(hi - delta, 0, 1)
            best[j] = np.clip(hj + delta, 0, 1)
        else:
            best[i] = np.clip(hi + delta, 0, 1)
            best[j] = np.clip(hj - delta, 0, 1)
        best = _fix_sum(best, n)
        cn = _c5(best)

            best[j] = np.clip(hj, 0, 1)
