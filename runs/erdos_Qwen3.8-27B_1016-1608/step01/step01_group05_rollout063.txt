import time

    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_set(h, target):

    for _ in range(50):
        mid = 0.5 * (lo + hi)
        hp = np.clip(h - mid, 0.0, 1.0)
        if float(np.sum(hp)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def upsample(h, n_new):

    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, h)

def make_binary(h, n, rng):
    target = n // 2
    h = np.round(h).clip(0, 1).astype(float)
    s = int(np.sum(h))
    if s > target:

    elif s < target:
        zeros = np.where(h == 0)[0]
        h[rng.choice(zeros, target - s, replace=False)] = 1.0
    return h

def sa_binary(h, n, rng, n_iters=500000, T0=1.0, T1=1e-5):
    target = n // 2
    h = make_binary(h, n, rng).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    cur_c5 = best_c5
    no_improve = 0
    stall = 15000

        T = max(T1, T0 * (T1 / T0) ** (it / n_iters))
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h, n)
        delta = new_c5 - cur_c5
        if delta < 0 or (T > 0 and rng.random() < np.exp(-delta / T)):
            cur_c5 = new_c5
            no_improve += 1
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                no_improve = 0
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
        if no_improve > stall:
            # Perturb
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            for _ in range(8):
                a = ones[rng.randint(len(ones))]
                b = zeros[rng.randint(len(zeros))]
                h[a], h[b] = h[b], h[a]
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            cur_c5 = compute_c5(h, n)
            no_improve = 0
    return best_h, best_c5

def local_search_binary(h, n, rng, max_rounds=50):
    """Try each pair swap and keep improvements."""
    h = h.copy()
    best_c5 = compute_c5(h, n)
    for rnd in range(max_rounds):
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        # Try random pairs
        n_tries = 500
        for _ in range(n_tries):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i], h[j] = h[j], h[i]
            new_c5 = compute_c5(h, n)
            if new_c5 < best_c5:
                best_c5 = new_c5
                improved = True
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
            else:
                h[i], h[j] = h[j], h[i]
        if not improved:
            break
    return h, best_c5

def slsqp_optimize(h0, n, maxiter=300):
    target = n / 2.0
    h0 = project_to_set(h0.copy(), target)
    
    def obj(x):
        return compute_c5(x, n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: float(np.sum(x) - target)}]
    
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-10})
        h_out = project_to_set(res.x.copy(), target)
        return h_out, compute_c5(h_out, n)
    except:
        return h0, compute_c5(h0, n)

def coord_descent(h, n, rng, n_sweeps=30):
    target = n / 2.0
    h = h.copy()
    best = compute_c5(h, n)
    for _ in range(n_sweeps):
        improved = False
        order = rng.permutation(n)
        for j in order:
            old = h[j]
            for c in [0.0, 1.0, min(1.0, old + 0.1), max(0.0, old - 0.1)]:
                if abs(c - old) < 1e-8:
                    continue
                h[j] = c
                h = project_to_set(h, target)
                v = compute_c5(h, n)
                if v < best:
                    best = v
                    improved = True
                    break
            else:
                h[j] = old
                h = project_to_set(h, target)
        if not improved:
            break
    return h, best

        return time.time() - _start

        return max(0.0, _budget - elapsed())

    return h, best

def run(seed=42, budget_s=1100, **kwargs):
    global _n_g, _con_t
    t0 = time.time()
    budget = float(budget_s)
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = np.inf

    def _tl():
        return max(0.0, budget - (time.time() - t0))

    n1 = 90
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        if len(h0) != n1:
            h0 = upsample(h0, n1)
    except Exception:
        h0 = rng.rand(n1)
    h0 = project_to_set(h0, n1 / 2.0)
    best_h = h0.copy()
    best_c5 = compute_c5(h0, n1)

    # Phase 1: Extensive binary SA at n=90
    h1, c1 = sa(h0, n1, rng, 800000, T0=1.0, T1=1e-5, stall=30000, perturb_n=8)
    if c1 < best_c5:
        best_h, best_c5 = h1.copy(), c1

    h1b, c1b = sa(best_h, n1, rng, 600000, T0=0.3, T1=1e-6, stall=20000, perturb_n=5)
    if c1b < best_c5:
        best_h, best_c5 = h1b.copy(), c1b

    # Another SA from a random start
    if _tl() > budget * 0.55:
        h_rand = to_binary(rng.rand(n1), n1, rng)
        h1c, c1c = sa(h_rand, n1, rng, 400000, T0=1.0, T1=1e-5, stall=25000, perturb_n=10)
        if c1c < best_c5:
            best_h, best_c5 = h1c.copy(), c1c

    # Phase 2: Hill climbing
    if _tl() > budget * 0.45:
        h2, c2 = hill_climb(best_h, n1, rng, n_rounds=150, n_tries=500)
        if c2 < best_c5:
            best_h, best_c5 = h2.copy(), c2

    # Phase 3: Non-binary SLSQP
    if _tl() > budget * 0.35:
        h3, c3 = slsqp(best_h, n1, maxiter=250)
        if c3 < best_c5:
            best_h, best_c5 = h3.copy(), c3

    # Phase 4: Coordinate descent
    if _tl() > budget * 0.30:
        h4, c4 = coord(best_h, n1, rng, n_sweeps=25)
        if c4 < best_c5:
            best_h, best_c5 = h4.copy(), c4

    # Phase 5: n=180
    if _tl() > budget * 0.35:
        n2 = 180
        h_up = project_to_set(upsample(best_h, n2), n2 / 2.0)
        c_up = compute_c5(h_up, n2)
        if c_up < best_c5:
            best_h, best_c5 = h_up.copy(), c_up
        h5, c5v = sa(h_up, n2, rng, 300000, T0=0.8, T1=1e-5, stall=30000, perturb_n=6)
        if c5v < best_c5:
            best_h, best_c5 = h5.copy(), c5v
        if _tl() > budget * 0.15:
            h6, c6 = slsqp(best_h, n2, maxiter=200)
            if c6 < best_c5:
                best_h, best_c5 = h6.copy(), c6
        if _tl() > budget * 0.12:
            h6b, c6b = coord(best_h, n2, rng, n_sweeps=15)
            if c6b < best_c5:
                best_h, best_c5 = h6b.copy(), c6b

    # Phase 6: n=360
    if _tl() > budget * 0.18:
        n3 = 360
        h_up = project_to_set(upsample(best_h, n3), n3 / 2.0)
        c_up = compute_c5(h_up, n3)
        if c_up < best_c5:
            best_h, best_c5 = h_up.copy(), c_up
        h7, c7 = slsqp(h_up, n3, maxiter=150)
        if c7 < best_c5:
            best_h, best_c5 = h7.copy(), c7

    # Final polish at best resolution
    if _tl() > 15:
        nb = len(best_h)
        h8, c8 = coord(best_h, nb, rng, n_sweeps=30)
        if c8 < best_c5:
            best_h, best_c5 = h8.copy(), c8

    if _tl() > 10:
        nb = len(best_h)
        h9, c9 = hill_climb(to_binary(best_h, nb, rng), nb, rng, n_rounds=50, n_tries=300)
        if c9 < best_c5:
            best_h, best_c5 = h9.copy(), c9

    best_h = project_to_set(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h, len(best_h))
    return best_h, best_c5, len(best_h)
