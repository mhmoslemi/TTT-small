def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def projection(h, target):
    h = np.array(h, dtype=float, copy=True)
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        excess = h.sum() - target
        if abs(excess) < 1e-14:
            return h
        if excess > 0:
            mask = h > 0
            cnt = mask.sum()
            if cnt == 0:
                return h
            s = min(excess / cnt, h[mask].min())
            h[mask] -= s
        else:
            mask = h < 1
            cnt = mask.sum()
            if cnt == 0:
                return h
            s = min(-excess / cnt, (1.0 - h[mask]).min())
            h[mask] += s
    return h

def smooth_c5(h, dx, temp):
    corr = get_corr(h)
    m = np.max(corr)
    shifted = temp * (corr - m)
    log_mean = np.log(np.mean(np.exp(shifted)))
    return (m + log_mean / temp) * dx

def slsqp_run(h0, dx, target, maxiter=500, use_smooth=False, temp=50.0):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target}]
    obj = smooth_c5 if use_smooth else compute_c5
    args = (dx, temp) if use_smooth else (dx,)
    try:
        res = minimize(obj, h0, args=args, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return projection(res.x, target)
    except:
        return None

def to_binary(h, n):
    """Project to nearest 0-1 vector with sum = n//2."""
    h = np.array(h, copy=True)
    k = n // 2
    # Sort by value, keep top k as 1

    hb[idx[:k]] = 1.0

    h = hb.astype(float).copy()

        zeros = np.where(h < 0.5)[0]

            break
        best_d = 0.0
        best_ij = None
        for i in zeros:
            hi = h[i]  # 0

                nc = compute_c5(h, dx)
                h[i], h[j] = 0.0, 1.0

                    best_ij = (i, j)

            i, j = best_ij
            h[i], h[j] = 1.0, 0.0
            c5 += best_d
            improved = True
        if not improved:
            break
    return h, c5

def projection(h, target):
    h = np.array(h, dtype=float, copy=True)
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        excess = h.sum() - target
        if abs(excess) < 1e-14:
            return h
        if excess > 0:
            m = h > 0
            c = m.sum()
            if c == 0:
                return h
            s = min(excess / c, h[m].min())
            h[m] -= s
        else:
            m = h < 1
            c = m.sum()
            if c == 0:
                return h
            s = min(-excess / c, (1.0 - h[m]).min())
            h[m] += s
    return h

def proj(h, tgt):
    h = np.array(h, dtype=float, copy=True)

        h = np.clip(h, 0.0, 1.0)
