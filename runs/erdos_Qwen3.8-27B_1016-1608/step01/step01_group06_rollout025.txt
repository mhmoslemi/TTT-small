import numpy as np
import time
from scipy.optimize import minimize

def compute_score(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return np.max(c) * dx

def soft_max_obj(h, tau):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cm = np.max(c)
    if cm < 1e-15:
        return 0.0
    return cm + np.log(np.sum(np.exp(tau * (c - cm)))) / tau

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def local_swap_refine(h, n, rng, max_iters=500):
    """Try pairwise swaps/moves to reduce the max correlation."""
    h = h.copy()
    best_s = compute_score(h)
    dx = 2.0 / n
    for it in range(max_iters):
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        active = np.sum(c > best_s - 1e-6)
        if active <= 0:
            break
        # Pick two random positions and try moving mass
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        # Try small transfer
        for delta in [0.01, -0.01, 0.02, -0.02, 0.05, -0.05]:
            if h[i] + delta < 0 or h[i] + delta > 1:
                continue
            if h[j] - delta < 0 or h[j] - delta > 1:
                continue
            h[i] += delta
            h[j] -= delta
            s_new = compute_score(h)
            if s_new < best_s - 1e-10:
                best_s = s_new
                break
            else:
                h[i] -= delta
                h[j] += delta
        if it % 50 == 0 and compute_score(h) < best_s - 1e-9:
            best_s = compute_score(h)
    return h, best_s

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    h_parent = np.array(initial_h_values, dtype=float)
    n_parent = len(h_parent)
    best_h = h_parent.copy()
    best_score = compute_score(h_parent)
    
    x_parent = np.linspace(0.0, 2.0, n_parent, endpoint=False)
    
    # Grid sizes: prefer small, add a couple larger
    n_list = sorted(set([16, 24, 30, 40, n_parent, 50, 60, 72, 80, 100]))
    
    for n_points in n_list:
        if time.time() - t0 > budget_s * 0.80:
            break
        
        x_new = np.linspace(0.0, 2.0, n_points, endpoint=False)
        h0 = np.interp(x_new, x_parent, h_parent)
        h0 = project_feasible(h0, n_points)
        s0 = compute_score(h0)
        if s0 < best_score:
            best_score = s0
            best_h = h0.copy()
        
        # Number of restarts based on grid size
        n_restarts = 10 if n_points <= 40 else (7 if n_points <= 60 else 4)
        
        for r in range(n_restarts):
            if time.time() - t0 > budget_s * 0.84:
                break
            
            # Diverse initializations
            if r == 0:
                h_start = h0.copy()
            elif r == 1:
                h_start = project_feasible(h0 + rng.randn(n_points) * 0.04, n_points)
            elif r == 2:
                h_start = project_feasible(h0 + rng.randn(n_points) * 0.10, n_points)
            elif r == 3:
                # Block structure
                nb = rng.randint(2, max(3, n_points // 8))
                h_start = np.zeros(n_points)

                    s_i = int(b * n_points / nb)

                    h_cand = project_feasible(res['x'], n_points)

                except Exception:
                    pass
            
            # Stage 2: Higher tau refinement
            if s_cand < best_score + 0.015:

                try:
                    res = minimize(lambda x: soft_max_obj(x, tau2), h_cand,
                                   method='L-BFGS-B',
                                   bounds=[(0,1)]*n_points,
                                   options={'maxiter': 300, 'ftol': 1e-15, 'gtol': 1e-9})
                    h_cand = project_feasible(res['x'], n_points)
                    s_cand = compute_score(h_cand)
                except Exception:
                    pass
            
            # Stage 3: SLSQP polish (handles constraint natively)
            if s_cand < best_score + 0.008:
                try:
                    def _c_fun(x):
                        return np.sum(x) - n_points / 2.0
                    res = minimize(compute_score, h_cand, method='SLSQP',
                                   bounds=[(0,1)]*n_points,
                                   constraints=[{'type':'eq','fun':_c_fun}],
                                   options={'maxiter': 300, 'ftol': 1e-14})
                    h_cand = project_feasible(res['x'], n_points)
                    s_cand = compute_score(h_cand)
                except Exception:
                    pass
            
            # Stage 4: Local swap refinement
            if s_cand < best_score + 0.005 and time.time() - t0 < budget_s * 0.82:
                h_cand, s_cand = local_swap_refine(h_cand, n_points, rng, max_iters=300)
            
            if s_cand < best_score:
                best_score = s_cand
                best_h = h_cand.copy()
            if s_cand < s0:
                s0 = s_cand
                h0 = h_cand.copy()
    
    # Final focused refinement on best solution
    if time.time() - t0 < budget_s * 0.93:
        n_fin = len(best_h)
        # Several rounds of SLSQP + swap
        for rnd in range(3):
            if time.time() - t0 > budget_s * 0.94:
                break
            try:
                def _cf(x):
                    return np.sum(x) - n_fin / 2.0
                res = minimize(compute_score, best_h, method='SLSQP',
                               bounds=[(0,1)]*n_fin,
                               constraints=[{'type':'eq','fun':_cf}],
                               options={'maxiter': 500, 'ftol': 1e-15})
                best_h = project_feasible(res['x'], n_fin)
            except Exception:
                pass
            best_h, s = local_swap_refine(best_h, n_fin, rng, max_iters=400)
            if s < best_score:
                best_score = s
    
    # Try downsample to smaller grid
    if time.time() - t0 < budget_s * 0.97:
        n_cur = len(best_h)
        for n_ds in [max(16, n_cur//2), max(16, n_cur//3)]:
            if n_ds >= n_cur:
                continue

            x_ds = np.linspace(0, 2, n_ds, endpoint=False)
            h_ds = np.interp(x_ds, x_cur, best_h)

                best_score = s_ds
                best_h = h_ds
