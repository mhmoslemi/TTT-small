from scipy.optimize import minimize
import time

_N = 45  # will be set dynamically

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def _fix(h, n):
    t = 0.5 * n
    for _ in range(200):
        d = t - np.sum(h)
        if abs(d) < 1e-13:
            break
        mask = (h < 1.0) if d > 0 else (h > 0.0)
        k = int(np.sum(mask))
        if k == 0:
            break
        h = h + (d / k) * mask
        h = np.clip(h, 0.0, 1.0)
    return h

def _obj(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    v = c * dx
    m = np.max(v)
    # Use log-sum-exp for stability
    shifted = v - m
    shifted = np.clip(shifted, -50, 50)
    return m + np.log(np.sum(np.exp(shifted)) - 1.0) / alpha  # subtract 1 to not double-count max

def _icf(x):
    return np.sum(x) - _N * 0.5

def _icj(x):
    return np.ones(_N)

def _try_slsqp(h0, n, alpha, maxiter, tol):
    global _N
    _N = n
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _icf, 'jac': _icj}]
    try:
        res = minimize(_obj, h0, args=(alpha,), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None:
            return _fix(np.clip(res.x, 0.0, 1.0), n)
    except Exception:
        pass
    return None

def _swap_search(h, n, n_iter, rng):
    """Pairwise swap local search."""
    best_c = _c5(h)
    for _ in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        # Try swapping values (or partial swap)
        delta = h[i] - h[j]
        if abs(delta) < 1e-10:
            continue
        # Swap: h[i] -> h[j], h[j] -> h[i]
        h[i], h[j] = h[j], h[i]
        c = _c5(h)
        if c < best_c:
            best_c = c
        else:
            h[i], h[j] = h[j], h[i]  # revert
    return h, best_c

def _interp(h, n_new):
    n0 = len(h)
    if n0 == n_new:
        return h.copy()
    xo = np.linspace(0, 2, n0, endpoint=False)
    # Actually, the samples are at i*dx for i=0,...,n-1

    xo = (np.arange(n0) + 0.5) * (2.0 / n0)
    xn = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(xn, xo, h)
    return _fix(np.clip(h_new, 0.0, 1.0), n_new)

import numpy as np
from scipy.optimize import minimize
import time

_N = 400

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def _fix(h, n):
    t = 0.5 * n
    for _ in range(200):
        d = t - np.sum(h)
        if abs(d) < 1e-13:
            break
        mask = (h < 1.0) if d > 0 else (h > 0.0)
        k = int(np.sum(mask))
        if k == 0:
            break
        h = h + (d / k) * mask
        h = np.clip(h, 0.0, 1.0)
    return h

def _project(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if np.sum(x) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    d = target - np.sum(x)
    if abs(d) > 1e-14:
        mask = (x > 0) & (x < 1)
        k = np.sum(mask)
        if k > 0:
            x[mask] += d / k
            x = np.clip(x, 0.0, 1.0)
    return x

def _obj_smooth(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    f = c * dx
    fm = np.max(f)
    e = np.exp(np.clip(alpha * (f - fm), -50, 50))
    return fm + np.log(np.sum(e)) / alpha

def _grad_smooth(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    f = c * dx
    fm = np.max(f)
    e = np.exp(np.clip(alpha * (f - fm), -500, 500))
    p = e / np.sum(e)
    
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        pm = p[m]
        if pm < 1e-15:
            continue
        t = m - (n - 1)
        # + (1-h[k-t]) for 0<=k-t<n
        k_lo = max(0, t)
        k_hi = min(n, n + t)
        if k_hi > k_lo:
            grad[k_lo:k_hi] += pm * (1.0 - h[k_lo - t:k_hi - t])
        # - h[k+t] for 0<=k+t<n
        k_lo2 = max(0, -t)
        k_hi2 = min(n, n - t)
        if k_hi2 > k_lo2:
            grad[k_lo2:k_hi2] -= pm * h[k_lo2 + t:k_hi2 + t]
    
    return dx * grad

def _adam_opt(h, n, alpha, steps, lr, seed):
    rng = np.random.RandomState(seed)
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.8, 0.999, 1e-8
    best_h = h.copy()
    best_c = _c5(h)
    for t in range(1, steps + 1):
        g = _grad_smooth(h, alpha)
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)
        # Adaptive lr
        step = lr * np.sqrt(t)  # warmup
        h_new = h - step * m_hat / (np.sqrt(v_hat) + eps)
        h_new = _project(h_new, n)
        c_new = _c5(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c

def _icf(x):
    return np.sum(x) - _N * 0.5

def _icj(x):
    return np.ones(_N)

def _slsqp_obj(x, alpha):
    return _obj_smooth(x, alpha)

def _try_slsqp(h0, n, alpha, maxiter):
    global _N
    _N = n
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _icf, 'jac': _icj}]
    try:
        res = minimize(_slsqp_obj, h0, args=(alpha,), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None:
            return _fix(np.clip(res.x, 0.0, 1.0), n)
    except Exception:
        pass
    return None

def _swap_search(h, n, n_trials, seed):
    rng = np.random.RandomState(seed)
    best_c = _c5(h)
    # Also try small perturbations
    for _ in range(n_trials):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        orig_i, orig_j = h[i], h[j]
        # Try swap
        h[i], h[j] = orig_j, orig_i
        c = _c5(h)
        if c < best_c - 1e-15:
            best_c = c
        else:
            h[i], h[j] = orig_i, orig_j
        # Try nudge
        d = 0.1 * (1 - abs(orig_i - orig_j))
        if d < 0.01:
            d = 0.05
        ni = np.clip(orig_i + d, 0, 1)
        nj = np.clip(orig_j - d, 0, 1)
        if ni != orig_i and nj != orig_j:
            h[i] = ni
            h[j] = nj
            h = _fix(h, n)
            c = _c5(h)
            if c < best_c - 1e-15:
                best_c = c
            else:
                h[i] = orig_i
                h[j] = orig_j
    return h, best_c

def _interp(h, n_new):
    n0 = len(h)
    if n0 == n_new:
        return h.copy()
    xo = (np.arange(n0) + 0.5) * (2.0 / n0)
    xn = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(xn, xo, h)
    return _fix(np.clip(h_new, 0.0, 1.0), n_new)

def run(seed=42, budget_s=1100, **kwargs):
