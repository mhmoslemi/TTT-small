            if not valid_pairs:

            linear_term = sum(h_cur[i] * h_var[j] for i, j in valid_pairs)

        if c5_new < c5_cur:

def compute_c5(h):
    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    target = n / 2.0
