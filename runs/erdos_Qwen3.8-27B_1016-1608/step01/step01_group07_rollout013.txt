import numpy as np
from scipy.optimize import minimize
import time

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _fix(h, n):
    t = 0.5 * n
    for _ in range(100):
        d = t - np.sum(h)
        if abs(d) < 1e-13:
            break
        mask = (h < 1.0) if d > 0 else (h > 0.0)
        k = int(np.sum(mask))
        if k == 0:
            break
        h = h + (d / k) * mask
        h = np.clip(h, 0.0, 1.0)
    return h

def _interp(h, n1):
    n0 = len(h)
    return _fix(np.clip(np.interp(np.arange(n1) * 2.0 / n1, np.arange(n0) * 2.0 / n0, h), 0, 1), n1)

def _obj(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full').astype(float) * dx
    m = c.max()
    return m + np.log(np.sum(np.exp(np.minimum(c - m, 0.0)))) / alpha

def _grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full').astype(float) * dx
    m = c.max()
    e = np.exp(np.minimum(c - m, 0.0))
    S = e.sum()
    w = e / S
    lags = np.arange(2 * n - 1) - (n - 1)
    g = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx] * dx
        if wi < 1e-12:
            continue
        L = int(lags[idx])
        if L == 0:
            g += wi * (1.0 - 2.0 * h)
        elif L > 0:
            g[:n - L] += wi * (1.0 - h[L:])
            g[L:] -= wi * h[:n - L]
        else:
            LL = -L
            g[LL:] += wi * (1.0 - h[:n - LL])
            g[:n - LL] -= wi * h[LL:]
    IM = 1.0 - 1.0 / alpha
    if abs(IM) > 1e-10:
        kmax = int(np.argmax(c))
        L = int(lags[kmax])
        if L == 0:
            g += IM * dx * (1.0 - 2.0 * h)
        elif L > 0:
            g[:n - L] += IM * dx * (1.0 - h[L:])
            g[L:] -= IM * dx * h[:n - L]
        else:
            LL = -L
            g[LL:] += IM * dx * (1.0 - h[:n - LL])
            g[:n - LL] -= IM * dx * h[LL:]
    return g

def _cfun(h, n):
    return np.sum(h) - 0.5 * n

def _cjac(h, n):
    return np.ones(n)

def _slsqp(h0, n, alpha, maxiter):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cfun, 'jac': _cjac, 'args': (n,)}]
    try:
        res = minimize(_obj, h0, (n, alpha), jac=_grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None and hasattr(res, 'fun'):
            return _fix(np.clip(res.x, 0.0, 1.0), n)
    except Exception:
        pass
    return None

def _lbb(h0, n, alpha, pen, maxiter):
    def obj_pen(x):
        dx = 2.0 / n
        c = np.correlate(x, 1.0 - x, mode='full').astype(float) * dx
        m = c.max()
        sm = m + np.log(np.sum(np.exp(np.minimum(c - m, 0.0)))) / alpha
        cv = np.sum(x) - 0.5 * n
        return sm + pen * cv * cv
    try:
        res = minimize(obj_pen, h0, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None:
            return _fix(np.clip(res.x, 0.0, 1.0), n)
    except Exception:
        pass
    return None

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    def rem():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        h = np.full(50, 0.5)
    n = len(h)
    h = _fix(h, n)
    bh = h.copy()
    bc = _c5(bh)

    # Phase A: Optimize at original n with increasing alpha
    for alpha in [20.0, 80.0, 400.0, 1500.0]:
        if rem() < 30:
            break
        ht = _slsqp(bh.copy(), n, alpha, 3000)
        if ht is not None:
            ct = _c5(ht)
            if ct < bc:
                bh, bc = ht, ct

    # Phase B: Multi-start at original n
    for trial in range(6):
        if rem() < 40:
            break
        scale = 0.06 + 0.04 * trial
        h0 = _fix(np.clip(bh + rng.randn(n) * scale, 0, 1), n)
        for alpha in [400.0, 2000.0]:
            if rem() < 25:
                break
            ht = _slsqp(h0, n, alpha, 3000)
            if ht is not None:
                ct = _c5(ht)
                if ct < bc:
                    bh, bc = ht, ct
                    h0 = ht.copy()

    # Phase C: L-BFGS-B with penalty (different optimizer, may find different basin)
    for pen in [100.0, 1000.0]:
        if rem() < 30:
            break
        ht = _lbb(bh.copy(), n, 800.0, pen, 3000)
        if ht is not None:
            ct = _c5(ht)
            if ct < bc:
                bh, bc = ht, ct
        ht = _lbb(_fix(np.clip(bh + rng.randn(n) * 0.1, 0, 1), n), n, 800.0, pen, 2000)
        if ht is not None:
            ct = _c5(ht)
            if ct < bc:
                bh, bc = ht, ct

    # Phase D: Downsample to n/2 for diverse exploration, then upsample
    n_half = max(40, n // 2)
    if rem() > 60:
        hd = _interp(bh, n_half)
        for trial in range(4):
            if rem() < 30:

            for alpha in [100.0, 1000.0]:

                    break
                ht = _slsqp(h0, n_half, alpha, 4000)
                if ht is not None:
                    ct = _c5(ht)
                    if ct < bc:
                        h0 = ht.copy()
        # Upsample best low-res back and refine
        if rem() > 30:
            hu = _interp(h0, n)
            ct = _c5(hu)
            if ct < bc:
                bh, bc = hu, ct
            ht = _slsqp(bh.copy(), n, 1500.0, 3000)
            if ht is not None:
                ct = _c5(ht)
                if ct < bc:
                    bh, bc = ht, ct

    # Phase E: Upsample to higher resolution
    n_up = min(n * 2, 800)
    if rem() > 80:
        bh = _interp(bh, n_up)
        n = n_up
        bc = _c5(bh)
        for alpha in [400.0, 2000.0]:
            if rem() < 40:
                break
            ht = _slsqp(bh.copy(), n, alpha, 3000)
            if ht is not None:
                ct = _c5(ht)

    # Phase F: Final polish at highest alpha
    for alpha in [5000.0, 20000.0]:

        if ht is not None:
            ct = _c5(ht)
            if ct < bc:
                bh, bc = ht, ct

    # Ensure exact integral
    bh = _fix(bh, len(bh))
    bc = _c5(bh)
    return bh, bc, len(bh)
