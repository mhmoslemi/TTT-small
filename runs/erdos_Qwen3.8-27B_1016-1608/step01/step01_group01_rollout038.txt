    cp = np.maximum(c, 0.0)
    val = (np.sum(cp ** p)) ** (1.0 / p)
    s = h.sum() - len(h) / 2.0
    return val + pen * s * s

def subgrad_max(h):
    c = corr_vec(h)
    k_star = int(np.argmax(c))
    n = len(h)
    s = k_star - (n - 1)
    m = np.arange(n)
    jp = m + s
    jm = m - s
    grad = np.zeros(n)
    vp = (jp >= 0) & (jp < n)
    vm = (jm >= 0) & (jm < n)
    grad[vp] += (1.0 - h[m[vp]])
    grad[vm] -= h[m[vm]]
    return grad, c

def coord_pass(h, n, xatol, min_improve):
    improved = False
    for i in range(n):
        try:
            res = minimize_scalar(coord_1d, args=(h, i, n),
                                  bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': xatol})
            if res.fun < h[0] and False:
                pass
            ht = h.copy()
            ht[i] = res.x
            ht = project_feasible(ht, n)
            ct = compute_c5(ht)
            if ct < compute_c5(h) - min_improve:
                h = ht
                improved = True
        except Exception:
            pass
    return h, improved

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    bounds = [(0.0, 1.0)] * n
    cons_eq = [{'type': 'eq', 'fun': sum_fun}]

    elapsed = time.time() - t0
    # Phase 1: SLSQP from parent with decreasing temperature
    for temp in [0.03, 0.01, 0.005, 0.002, 0.001, 0.0005]:
        elapsed = time.time() - t0
        if elapsed > budget_s * 0.15:
            break
        try:
            res = minimize(soft_obj, best_h.copy(), args=(temp,),
                          method='SLSQP', bounds=bounds, constraints=cons_eq,
                          options={'maxiter': 1500, 'ftol': 1e-15})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        except Exception:
            pass

    # Phase 2: 1D coordinate search (multiple passes)
    elapsed = time.time() - t0
    for pnum in range(30):
        elapsed = time.time() - t0
        if elapsed > budget_s * 0.40:
            break
        xatol = 1e-5 if pnum < 10 else 1e-7
        improved = False
        for i in range(n):
            elapsed = time.time() - t0
            if elapsed > budget_s * 0.40:
                break
            try:
                res = minimize_scalar(coord_1d, args=(best_h, i, n),
                                      bounds=(0.0, 1.0), method='bounded',
                                      options={'xatol': xatol})
                ht = best_h.copy()
                ht[i] = res.x
                ht = project_feasible(ht, n)
                ct = compute_c5(ht)
                if ct < best_c5 - 1e-12:
                    best_c5 = ct
                    best_h = ht
                    improved = True
            except Exception:
                pass
        if not improved:
            break

    # Phase 3: Subgradient descent on exact max
    elapsed = time.time() - t0
    if elapsed < budget_s * 0.55:
        alpha = 0.03
        stall = 0
        for it in range(3000):
            elapsed = time.time() - t0
            if elapsed > budget_s * 0.55:
                break
            grad, c = subgrad_max(best_h)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-12:
                break
            step = alpha / gnorm
            h_try = best_h - step * grad
            h_try = project_feasible(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try.copy()
                stall = 0
            else:
                stall += 1
                if stall > 40:
                    alpha *= 0.85
                    stall = 0
                    if alpha < 1e-6:
                        break

    # Phase 4: Diverse restarts with local optimization
    elapsed = time.time() - t0
    n_rest = 0
    while n_rest < 25:
        elapsed = time.time() - t0
        if elapsed > budget_s * 0.78:
            break
        if elapsed - t0 > budget_s - 60:
            break
        n_rest += 1
        scale = 0.04 + 0.22 * rng.random()
        h_r = project_feasible(best_h + rng.normal(0, scale, n), n)
        # SLSQP from restart
        for temp in [0.01, 0.003]:
            elapsed = time.time() - t0
            if elapsed > budget_s * 0.78:
                break
            try:
                res = minimize(soft_obj, h_r.copy(), args=(temp,),
                            method='SLSQP', bounds=bounds, constraints=cons_eq,
                            options={'maxiter': 400, 'ftol': 1e-14})
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h_r = h_new.copy()
            except Exception:
                pass
        # Quick 1D pass from restart
        elapsed = time.time() - t0
        if elapsed > budget_s * 0.78:
            continue
        for i in range(0, n, 3):
            elapsed = time.time() - t0
            if elapsed > budget_s * 0.78:
                break
            try:
                res = minimize_scalar(coord_1d, args=(best_h, i, n),
                                      bounds=(0.0, 1.0), method='bounded',
                                      options={'xatol': 1e-5})
                ht = best_h.copy()
                ht[i] = res.x
                ht = project_feasible(ht, n)
                ct = compute_c5(ht)
                if ct < best_c5 - 1e-12:
                    best_c5 = ct
                    best_h = ht
            except Exception:
                pass

    # Phase 5: L-BFGS-B with p-norm (no gradient, use numerical)
    elapsed = time.time() - t0
    if elapsed < budget_s * 0.88:
        for p in [16, 32, 64]:
            elapsed = time.time() - t0
            if elapsed > budget_s * 0.88:
                break
            try:
                res = minimize(pnorm_obj, best_h.copy(), args=(p, 5.0),
                              method='L-BFGS-B', bounds=bounds,
                              options={'maxiter': 500, 'ftol': 1e-15})
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
            except Exception:
                pass

    # Phase 6: Final fine coordinate search
    elapsed = time.time() - t0
    for pnum in range(15):
        elapsed = time.time() - t0
        if elapsed > budget_s * 0.97:
            break
        improved = False
        for i in range(n):
            elapsed = time.time() - t0
            if elapsed > budget_s * 0.97:
                break
            try:
                res = minimize_scalar(coord_1d, args=(best_h, i, n),
                                      bounds=(0.0, 1.0), method='bounded',
                                      options={'xatol': 1e-9})
                ht = best_h.copy()
                ht[i] = res.x
                ht = project_feasible(ht, n)
                ct = compute_c5(ht)
                if ct < best_c5 - 1e-14:
                    best_c5 = ct
                    best_h = ht
                    improved = True
            except Exception:
                pass
        if not improved:
            break

    # Phase 7: Final SLSQP polish
    elapsed = time.time() - t0
    if elapsed < budget_s * 0.995:
        try:
            res = minimize(soft_obj, best_h.copy(), args=(0.0003,),
                          method='SLSQP', bounds=bounds, constraints=cons_eq,
                          options={'maxiter': 2000, 'ftol': 1e-16})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        except Exception:
            pass

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n)
