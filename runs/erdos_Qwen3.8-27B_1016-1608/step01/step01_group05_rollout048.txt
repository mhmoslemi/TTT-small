    return np.interp(x_new, x_old, h)

def smooth_p_obj(h, n, p):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.sum(np.maximum(corr, 0.0) ** p))

def slsqp_smooth(h0, n, p, maxiter=800):
    target = n / 2.0
    h0 = project_to_set(h0.copy(), target)

    def obj(x):
        return smooth_p_obj(x, n, p)

    def con(x):
        return float(np.sum(x) - target)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con}]
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_set(res.x, target)
    except Exception:
        return h0

def coord_search(h, n, rng, n_sweeps=15, steps=0.05):
    target = n / 2.0
    h = h.copy()
    best = compute_c5(h, n)
    for _ in range(n_sweeps):
        order = rng.permutation(n)
        improved = False
        for j in order:
            old = h[j]
            cands = [0.0, 1.0, max(0.0, old - steps), min(1.0, old + steps),
                     max(0.0, old - steps * 0.3), min(1.0, old + steps * 0.3)]
            bc, bv = old, best
            h[j] = old  # reset
            for c in cands:
                if abs(c - old) < 1e-9:
                    continue
                h[j] = c
                hp = project_to_set(h, target)
                v = compute_c5(hp, n)
                if v < bv:
                    bv = v
                    bc = c
            h[j] = bc
            h = project_to_set(h, target)
            if bc != old:
                best = bv
                improved = True
        if not improved:
            break
    return h, best

def subgrad_refine(h, n, rng, n_iter=200, lr0=0.05):
    target = n / 2.0
    h = h.copy()
    best = compute_c5(h, n)
    dx = 2.0 / n
    for it in range(n_iter):
        lr = lr0 * (0.98 ** it)
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        k_max = int(np.argmax(corr)) - (n - 1)
        # subgradient direction for shift k_max
        grad = np.zeros(n)
        for i in range(n):
            g = dx * 0.0
            j1 = i + k_max
            if 0 <= j1 < n:
                g += dx * 1.0
                g -= dx * h[j1]
            j2 = i - k_max
            if 0 <= j2 < n:
                g -= dx * h[j2]
            grad[i] = g
        h_new = h - lr * grad
        h_new = project_to_set(h_new, target)
        c_new = compute_c5(h_new, n)
        if c_new < best:
            h = h_new
            best = c_new
    return h, best

def make_random_start(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    h += rng.normal(0, 0.1, n)
    return project_to_set(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget = float(budget_s)
    rng = np.random.RandomState(seed)

    def tl():
        return max(0.0, budget - (time.time() - t0))

    best_h = None
    best_c5 = np.inf
    best_n = 0

    def upd(h, n):
        nonlocal best_h, best_c5, best_n
        h = project_to_set(np.asarray(h, dtype=float), n / 2.0)
        c = compute_c5(h, n)
        if c < best_c5:
            best_h = h.copy()
            best_c5 = c
            best_n = n
        return c

    # Phase 1: n=90 from initial
    n0 = 90
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        if len(h0) != n0:
            h0 = upsample(h0, n0)
    except NameError:
        h0 = make_random_start(n0, rng)
    h0 = project_to_set(h0, n0 / 2.0)
    upd(h0, n0)

    # Progressive p-norm SLSQP at n=90
    for p in [2, 4, 8, 16, 32, 64]:
        if tl() < 80:
            break
        h0 = slsqp_smooth(h0, n0, p, maxiter=1000)
        upd(h0, n0)

    # Coordinate search at n=90
    if tl() > 100:
        h0, _ = coord_search(h0, n0, rng, n_sweeps=20, steps=0.05)
        upd(h0, n0)

    # Subgradient refinement
    if tl() > 80:
        h0, _ = subgrad_refine(h0, n0, rng, n_iter=150, lr0=0.03)
        upd(h0, n0)

    # More SLSQP after coord search
    if tl() > 60:
        h0 = slsqp_smooth(h0, n0, 32, maxiter=600)
        upd(h0, n0)
        h0, _ = coord_search(h0, n0, rng, n_sweeps=10, steps=0.03)
        upd(h0, n0)

    # Phase 2: Try nearby resolutions
    for ntry in [88, 92, 96, 84, 100]:
        if tl() < 80:
            break
        h_up = project_to_set(upsample(best_h, ntry), ntry / 2.0)
        upd(h_up, ntry)
        for p in [16, 64]:
            if tl() < 50:
                break
            h_up = slsqp_smooth(h_up, ntry, p, maxiter=600)
            upd(h_up, ntry)
        if tl() > 30:
            h_up, _ = coord_search(h_up, ntry, rng, n_sweeps=10, steps=0.04)
            upd(h_up, ntry)

    # Phase 3: Random restarts at best n
    restart_budget = tl() * 0.3
    if restart_budget > 40:
        nbest = best_n
        for _ in range(5):
            if tl() < restart_budget:
                break
            hr = make_random_start(nbest, rng)
            for p in [8, 64]:
                if tl() < restart_budget:
                    break
                hr = slsqp_smooth(hr, nbest, p, maxiter=500)
            hr, _ = coord_search(hr, nbest, rng, n_sweeps=8, steps=0.05)
            upd(hr, nbest)

    # Phase 4: Final fine polish
    if tl() > 20:
        best_h, _ = coord_search(best_h, best_n, rng, n_sweeps=15, steps=0.02)
        upd(best_h, best_n)
        if tl() > 15:
            best_h = slsqp_smooth(best_h, best_n, 64, maxiter=400)
            upd(best_h, best_n)

    best_h = project_to_set(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h, best_n)
    return best_h, best_c5, best_n
