import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def repair(h, target):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0 - 1e-13
        else:
            m = h > 1e-13
        c = int(m.sum())
        if c == 0:
            break
        h[m] += d / c
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_and_grad(h):
    """Return (corr, subgradient) where subgradient is for the max component."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    n = len(h)
    max_val = np.max(corr)
    active = np.where(corr >= max_val - 1e-12)[0]
    grad = np.zeros(n)
    for idx in active:
        L = idx - (n - 1)
        # grad[i] += (1-h[i-L])_valid - h[i+L]_valid
        for i in range(n):
            v = 0.0
            j1 = i - L
            if 0 <= j1 < n:
                v += 1.0 - h[j1]
            j2 = i + L
            if 0 <= j2 < n:
                v -= h[j2]
            grad[i] += v
    grad /= len(active)
    return corr, max_val, grad

def subgradient_step(h, dx, alpha, target, mu=10.0):
    corr, maxval, grad = corr_and_grad(h)
    # Add sum penalty gradient
    s = np.sum(h)
    grad_p = grad * maxval * dx + 2.0 * mu * (s - target) / len(h)
    h_new = h - alpha * grad_p
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = repair(h_new, target)
    return h_new

def coord_descent(h, dx, target, n_sweeps=50, mu=50.0, seed=42):
    rng = np.random.RandomState(seed)
    n = len(h)
    h = h.copy()
    for sweep in range(n_sweeps):
        order = rng.permutation(n)
        for i in order:
            # Compute the 1D problem for h[i]
            corr = np.correlate(h, 1.0 - h, mode="full")
            # For each lag idx, compute g = d(corr[idx])/d(h[i])
            g = np.zeros(len(corr))
            for idx in range(len(corr)):
                L = idx - (n - 1)
                j1 = i - L
                j2 = i + L
                v = 0.0
                if 0 <= j1 < n:
                    v += 1.0 - h[j1]
                if 0 <= j2 < n:
                    v -= h[j2]
                g[idx] = v
            # f(t) = dx * max_k (g[k]*t + corr[k] - h[i]*g[k]) + mu*(t + rest - target)^2
            rest = np.sum(h) - h[i]
            old = h[i]
            b = corr - old * g  # intercepts
            
            def obj_1d(t):
                vals = g * t + b
                return dx * np.max(vals) + mu * (t + rest - target) ** 2
            
            # Golden section search
            a, bnd = 0.0, 1.0
            gr = (np.sqrt(5) - 1) / 2
            c1 = bnd - gr * (bnd - a)
            c2 = a + gr * (bnd - a)
            f1, f2 = obj_1d(c1), obj_1d(c2)
            for _ in range(60):
                if f1 < f2:
                    bnd, c2, f2 = c2, c1, f1
                    c1 = bnd - gr * (bnd - a)
                    f1 = obj_1d(c1)
                else:
                    a, c1, f1 = c1, c2, f2
                    c2 = a + gr * (bnd - a)
                    f2 = obj_1d(c2)
            h[i] = (a + bnd) / 2
            h[i] = np.clip(h[i], 0.0, 1.0)
    return repair(h, target)

        for idx in active:
            L = idx - (n - 1)
            for i in range(n):
                v = 0.0
                j1 = i - L
                if 0 <= j1 < n:
                    v += 1.0 - h[j1]
                j2 = i + L
                if 0 <= j2 < n:
                    v -= h[j2]
                grad[i] += v
        grad /= len(active)
        alpha = alpha0 / (1.0 + 0.01 * step)
        h_new = h - alpha * maxval * dx * grad
        h_new += alpha * 0.1 * rng.randn(n)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = repair(h_new, target)
        c_new = compute_c5(h_new, dx)
        c_cur = compute_c5(h, dx)
        if c_new < c_cur + 1e-8:
            h = h_new
        else:
            alpha *= 0.5
            h_new = np.clip(h - alpha * maxval * dx * grad, 0.0, 1.0)
            h_new = repair(h_new, target)
            if compute_c5(h_new, dx) < c_cur + 1e-8:
                h = h_new
    return h

def logsumexp_obj(h, dx, target, beta, mu):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta + mu * (np.sum(h) - target) ** 2

def logsumexp_grad(h, dx, target, beta, mu):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    p = exps / np.sum(exps)
    n = len(h)
    grad = np.zeros(n)
    for idx in range(len(corr)):
        if p[idx] < 1e-12:
            continue
        L = idx - (n - 1)
        for i in range(n):
            v = 0.0
            j1 = i - L
            if 0 <= j1 < n:
                v += 1.0 - h[j1]
            j2 = i + L
            if 0 <= j2 < n:
                v -= h[j2]
            grad[i] += p[idx] * v
    grad *= dx
    grad += 2.0 * mu * (np.sum(h) - target) / n
    return grad, logsumexp_obj(h, dx, target, beta, mu)

def slsqp_polish(h, dx, target, maxiter=500):
    n = len(h)
    bnds = [(0.0, 1.0)] * n

    def obj(x):
        return compute_c5(x, dx)

    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        maxval = np.max(corr)
        active = np.where(corr >= maxval - 1e-12)[0]
        g = np.zeros(n)
        for idx in active:
            L = idx - (n - 1)
            for i in range(n):
                v = 0.0
                j1 = i - L
                if 0 <= j1 < n:
                    v += 1.0 - h[j1]
                j2 = i + L
                if 0 <= j2 < n:
                    v -= h[j2]
                g[i] += v
        g /= len(active)
        return g * maxval * dx

    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                    'jac': lambda x: np.ones(n)}]

    def my_obj(x):
        return compute_c5(np.clip(x, 0, 1), dx)

    def my_jac(x):
        x = np.clip(x, 0, 1)
        corr = np.correlate(x, 1.0 - x, mode="full")
        maxval = np.max(corr)
        active = np.where(corr >= maxval - 1e-12)[0]
        g = np.zeros(n)
        for idx in active:
            L = idx - (n - 1)
            for i in range(n):
                v = 0.0
                j1 = i - L
                if 0 <= j1 < n:
                    v += 1.0 - x[j1]
                j2 = i + L
                if 0 <= j2 < n:
                    v -= x[j2]
                g[i] += v
        g /= len(active)
        return g * maxval * dx

    r = minimize(my_obj, h, method='SLSQP', bounds=bnds, constraints=constraints,
                 jac=my_jac, options={'maxiter': maxiter, 'ftol': 1e-16})
    return repair(np.clip(r.x, 0, 1), target)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.ones(70) * 0.5

    n = len(h0)
    target = n / 2.0
    dx = 2.0 / n
    h0 = repair(h0, target)
    best_h = h0.copy()
    best_c5 = compute_c5(h0, dx)

    # Phase 1: Coordinate descent (exact 1D solves)
    if time.time() - t0 < budget_s - 300:
        h_cd = coord_descent(h0, dx, target, n_sweeps=80, mu=5.0, seed=seed)
        c_cd = compute_c5(h_cd, dx)
        if c_cd < best_c5:
            best_h, best_c5 = h_cd, c_cd

    # Phase 2: Logsumexp-smoothed L-BFGS-B with annealing
    if time.time() - t0 < budget_s - 250:
        for beta in [20.0, 50.0, 100.0, 200.0, 400.0]:
            if time.time() - t0 > budget_s - 200:
                break
            h_curr = best_h.copy()
            mu_ls = 2.0
            bnds = [(0.0, 1.0)] * n

            def ls_obj(x):
                return logsumexp_obj(x, dx, target, beta, mu_ls)

            def ls_jac(x):
                g, _ = logsumexp_grad(x, dx, target, beta, mu_ls)
                return g

            try:
                r = minimize(ls_obj, h_curr, method='L-BFGS-B', bounds=bnds,
                             jac=ls_jac, options={'maxiter': 200, 'ftol': 1e-14,
                                                   'gtol': 1e-10, 'maxls': 50})
                h_r = repair(np.clip(r.x, 0, 1), target)
                c_r = compute_c5(h_r, dx)
                if c_r < best_c5:
                    best_h, best_c5 = h_r, c_r
            except Exception:
                pass
            # Follow up with coord descent on the smoothed result
            if time.time() - t0 < budget_s - 180:
                h_cd2 = coord_descent(best_h, dx, target, n_sweeps=40, mu=5.0,
                                      seed=seed + beta)
                c_cd2 = compute_c5(h_cd2, dx)
                if c_cd2 < best_c5:
                    best_h, best_c5 = h_cd2, c_cd2

    # Phase 3: Subgradient + coord descent interleave
    if time.time() - t0 < budget_s - 150:
        for round_i in range(5):
            if time.time() - t0 > budget_s - 130:
                break
            h_sg = subgradient_optimize(best_h, dx, target, n_steps=80,
                                        seed=seed + 100 + round_i)
            c_sg = compute_c5(h_sg, dx)
            if c_sg < best_c5:
                best_h, best_c5 = h_sg, c_sg
            # Refine with coordinate descent
            h_cd3 = coord_descent(best_h, dx, target, n_sweeps=60, mu=5.0,
                                  seed=seed + 200 + round_i)
            c_cd3 = compute_c5(h_cd3, dx)
            if c_cd3 < best_c5:
                best_h, best_c5 = h_cd3, c_cd3

    # Phase 4: Multi-start with diverse perturbations
    if time.time() - t0 < budget_s - 100:
        n_restarts = 15
        for i in range(n_restarts):
            if time.time() - t0 > budget_s - 80:
                break
            scale = 0.03 * (1.0 - i / float(n_restarts)) + 0.005
            hp = best_h + np.random.randn(n) * scale
            hp = repair(hp, target)
            h_rs = coord_descent(hp, dx, target, n_sweeps=50, mu=5.0,
                                 seed=seed + 500 + i)
            c_rs = compute_c5(h_rs, dx)
            if c_rs < best_c5:
                best_h, best_c5 = h_rs, c_rs

    # Phase 5: SLSQP final polish
    if time.time() - t0 < budget_s - 30:
        for _ in range(3):
            if time.time() - t0 > budget_s - 10:
                break
            try:
                h_sp = slsqp_polish(best_h, dx, target, maxiter=300)
                c_sp = compute_c5(h_sp, dx)
                if c_sp < best_c5:
                    best_h, best_c5 = h_sp, c_sp
            except Exception:
                break

    # Phase 6: Final coordinate descent grind
    if time.time() - t0 < budget_s - 10:
        h_final = coord_descent(best_h, dx, target, n_sweeps=100, mu=10.0,
                                seed=seed + 999)
        c_final = compute_c5(h_final, dx)
        if c_final < best_c5:
            best_h, best_c5 = h_final, c_final

    # Final validation
    best_h = repair(best_h, target)
    best_c5 = compute_c5(best_h, dx)
    return (best_h, best_c5, n)
