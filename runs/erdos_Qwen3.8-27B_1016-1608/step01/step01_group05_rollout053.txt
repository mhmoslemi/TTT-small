Let me be more careful about the coordinate descent. The key issue is that after changing one element and projecting, many other elements change. So it's not truly "coordinate" descent. 

A better approach: for coordinate descent, work with the Lagrangian. The constraint sum(h) = target can be handled by a Lagrange multiplier λ. Then the objective becomes:
L(h) = compute_c5(h, n) + λ * (sum(h) - target)

At the optimum, ∂L/∂h[j] = 0 for 0 < h[j] < 1, and the appropriate inequality conditions at the boundaries.

For coordinate descent with Lagrangian:
- Fix h[j] ≠ h_m
- Optimize over h_m ∈ [0,1]
- The objective as a function of h_m is: max_k c_k(h_m) + λ * h_m
- Each c_k is at most linear in h_m (since h_m appears in at most 2 terms of c_k: as h[m] and as (1-h[m]) in the (1-h) factor)

Wait, actually h_m can appear in c_k in multiple ways:
- c_k = sum_i h[i+k] * (1-h[i])
- h_m appears as h[i+k] when i+k = m, i.e., i = m-k
- h_m appears in (1-h[i]) when i = m

So for a given k, h_m appears in at most 2 terms:
1. The term where i = m-k (if valid): h[m] * (1-h[m-k])
2. The term where i = m (if valid): h[m+k] * (1-h[m])

So c_k(h_m) = A_k + B_k * h_m + C_k * (h_m)^2 where:
- C_k is non-zero only if both i=m-k and i=m are valid for the same k (i.e., m-k = m, so k=0, meaning the term h[m]*(1-h[m]))
- Generally C_k = 0 for k ≠ 0

So for k ≠ 0: c_k is linear in h_m
For k = 0: c_m*c_m includes term h[m]*(1-h[m]) which is -h[m]² + h[m], so it's quadratic

Therefore max_k c_k(h_m) is the max of one quadratic and many linear functions. This is convex in h_m. We can find its minimum by:
- Evaluating at critical points (where the derivative changes sign)
- The derivative of max_k c_k at h_m equals the derivative of the active c_k
- We can find the minimum by bisection on the derivative

This would give exact 1D optimization at each coordinate! Let me implement this:

def coord_1d_solve(h, m, n, lam=0.0):

    target = n / 2.0

    # For efficiency, compute c_k at h[m]=0 and h[m]=1, then interpolate
    h0 = h.copy()
    h1 = h.copy()
    h0[m] = 0.0

    c1 = np.correlate(h1, 1.0 - h1, mode="full")
    # c_k(t) = c0[k] + t*(c1[k] - c0[k]) + t²*(quadratic part)
    # Only the k where i=m and i+k=m (self-product) has quadratic part
    # For k=0: the term is h[m]*(1-h[m]) = t*(1-t), contributing -t²+t to c_0
    # Actually, let's just evaluate at several points and find the minimum numerically
    
    dx = 2.0 / n
    best_t = h[m]
    best_val = np.max(
        np.correlate(
            h.copy().__setitem__(m, h[m]) and h,  1.0 - h, mode="full"
        )
    )  # current value
    
    # Sample t in [0,1] and find min of max_k c_k(t) + lam * t
    ts = np.linspace(0, 1, 11)
    best_t = h[m]
    best_v = np.inf

        if v < best_v:

            best_t = t
    return best_t, best_v

import numpy as np
from scipy.optimize import minimize
import time

_start_t = 0.0
_budget = 1100.0

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def upsample(h, n_new):
    n = len(h)
    x_old = np.arange(n, dtype=float)
    x_new = np.linspace(0, n - 1, n_new)
    return np.interp(x_new, x_old, h)

def timing_left():
    return max(0.0, _budget - (time.time() - _start_t))

def slsqp(h0, n, maxiter=300):
    target = n / 2.0
    h0 = project(h0.copy(), target)
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return compute_c5(x, n)
    def con(x):
        return np.sum(x) - target
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_out = project(res.x.copy(), target)
        return h_out, compute_c5(h_out, n)
    except Exception:
        return h0, compute_c5(h0, n)

def nelder_mead(h0, n, maxiter=5000):
    target = n / 2.0
    h0 = project(h0.copy(), target)
    def obj(x):
        hp = project(x, target)
        return compute_c5(hp, n)
    try:
        res = minimize(obj, h0, method='Nelder-Mead',
                       options={'maxiter': maxiter, 'xatol': 1e-9, 'fatol': 1e-12, 'maxfev': maxiter})
        h_out = project(res.x.copy(), target)
        return h_out, compute_c5(h_out, n)
    except Exception:
        return h0, compute_c5(h0, n)

def active_set(h0, n, M=5, n_rounds=12):
    target = n / 2.0
    h = project(h0.copy(), target)
    dx = 2.0 / n
    for _ in range(n_rounds):

        top = np.argsort(corr)[-M:]

            c = np.correlate(x, 1.0 - x, mode="full")
            return float(np.max(c[top]) * dx)
        def con(x):
            return np.sum(x) - target
        try:
            res = minimize(obj, h, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n,
                          constraints=[{'type': 'eq', 'fun': con}],
                          options={'maxiter': 100, 'ftol': 1e-14})
            h = project(res.x.copy(), target)
        except Exception:
            pass
    return h, compute_c5(h, n)

def coord_descend(h0, n, rng, sweeps=50):
    target = n / 2.0
    h = project(h0.copy(), target)
    best_c = compute_c5(h, n)
    for _ in range(sweeps):
        improved = False
        order = rng.permutation(n)
        for j in order:
            old = h[j]
            cands = [0.0, 1.0,
                     max(0.0, old - 0.1), min(1.0, old + 0.1),
                     max(0.0, old - 0.05), min(1.0, old + 0.05),
                     max(0.0, old - 0.02), min(1.0, old + 0.02),
                     max(0.0, old - 0.005), min(1.0, old + 0.005)]
            best_cand = old
            best_v = best_c
            for c in cands:
                if abs(c - old) < 1e-9:
                    continue
                saved = h[j]
                h[j] = c
                h = project(h, target)
                v = compute_c5(h, n)

                    best_cand = c

                best_c = best_v
                improved = True
        if not improved:

    return h, best_c

    h = np.zeros(n)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h

    x = np.linspace(0.0, 2.0, n)
    h = np.zeros(n)
    for _ in range(rng.randint(2, 5)):
        f = rng.randint(1, 10)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.3, 0.9)
        h += a * np.sin(f * np.pi * x + p)
    h = 0.5 + 0.35 * h / (np.max(np.abs(h)) + 1e-10)
    return project(h, n / 2.0)

def make_block(n, rng):
    h = np.zeros(n)
    nb = rng.randint(2, 6)
    blocks = []
    pos = 0
    while pos < n:
        L = rng.randint(3, max(4, n // (nb + 1)))
        if pos + L > n:
            L = n - pos
        blocks.append((pos, pos + L))
        pos += L + rng.randint(1, max(2, n // (nb * 3)))
    for i, (a, b) in enumerate(blocks):
        if i % 2 == 0:
            h[a:b] = 1.0
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones < target:
        zeros = np.where(h == 0)[0]
        picks = rng.choice(zeros, target - n_ones, replace=False)
        h[picks] = 1.0
    elif n_ones > target:
        ones = np.where(h == 1)[0]
        picks = rng.choice(ones, n_ones - target, replace=False)
        h[picks] = 0.0
    return h
