```python
def peak_equalization(h, n, K=5):
    c = get_corr(h)
    top_k = np.argsort(c)[-K:][::-1]
    grad_stack = np.array([grad_corr_at_k(h, k) for k in top_k])
    vals = c[top_k]
    target = np.mean(vals)
    # Solve: grad_stack @ delta ≈ (target - vals)
    # delta = pinv(grad_stack) @ (target - vals)
    b = target - vals
    delta = np.linalg.lstsq(grad_stack, b, rcond=None)[0]
    # Scale delta to be small
    delta_max = np.max(np.abs(delta))
    if delta_max > 1e-3:
        delta *= 1e-3 / delta_max
    h_new = h + delta
    return project_feasible(h_new, n)
```
