import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_sum(h, target):
    """Project h to have sum = target while staying in [0,1]"""
    h = np.array(h, dtype=float, copy=True)
    for _ in range(100):
        h = np.clip(h, 0, 1)
        excess = h.sum() - target
        if abs(excess) < 1e-12:
            break
        if excess > 0:
            mask = h > 0
            if mask.sum() == 0: break
            s = min(excess / mask.sum(), h[mask].min() - 1e-15)
            if s <= 0: break
            h[mask] -= s
        else:
            mask = h < 1
            if mask.sum() == 0: break
            s = min(-excess / mask.sum(), (1 - h[mask]).min() - 1e-15)
            if s <= 0: break
            h[mask] += s
    return h

def coord_descent(h, dx, target, n_iter=50, rng=None):
    """Exact coordinate descent with Lagrange multiplier for sum constraint"""
    n = len(h)
    h = h.copy()
    h = project_sum(h, target)
    lam = 0.0
    
    for it in range(n_iter):
        # Determine coordinate order
        if rng is not None:
            order = rng.permutation(n)
        else:
            order = np.arange(n)
        
        changed = False
        for i in order:
            # Compute a[d], b[d] for each lag d
            corr = np.correlate(h, 1.0 - h, mode="full")
            d_range = np.arange(-(n-1), n)
            a = np.zeros(2*n-1)
            for d_idx, d in enumerate(d_range):
                slope = 0.0
                j1 = i + d
                if 0 <= j1 < n:
                    slope += (1.0 - h[j1])
                j2 = i - d
                if 0 <= j2 < n:
                    slope += (-h[j2])
                a[d_idx] = slope
            b = corr - a * h[i]
            
            # Minimize max_d(a[d]*v + b[d]) + lam*v over v in [0,1]
            # Dense evaluation
            vs = np.linspace(0, 1, 500)
            f = np.max(a[:, None] * vs[None, :] + b[:, None], axis=0) + lam * vs
            best_v = vs[np.argmin(f)]
            
            # Refine
            lo, hi = 0.0, 1.0
            if 0 < best_v < 1:
                # Binary search for the minimum
                for _ in range(20):
                    m1 = lo + (hi - lo) / 3
                    m2 = hi - (hi - lo) / 3
                    f1 = np.max(a * m1 + b) + lam * m1
                    f2 = np.max(a * m2 + b) + lam * m2
                    if f1 < f2:
                        hi = m2
                    else:
                        lo = m1
                best_v = (lo + hi) / 2
            
            if abs(best_v - h[i]) > 1e-10:
                h[i] = best_v
                changed = True
        
        # Update lambda
        excess = h.sum() - target
        if excess > 1e-10:
            lam += 0.1
        elif excess < -1e-10:
            lam -= 0.1
        else:
            lam *= 0.9
        
        if not changed and abs(excess) < 1e-10:
            break
    
    h = project_sum(h, target)
    return h

def project_sum(h, target):
    h = np.array(h, dtype=float, copy=True)
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        excess = h.sum() - target
        if abs(excess) < 1e-13:
            break
        if excess > 0:
            mask = h > 1e-15
            cnt = mask.sum()
            if cnt == 0:
                break
            s = min(excess / cnt, h[mask].min())
            if s <= 0:
                break
            h[mask] -= s
        else:
            mask = h < 1.0 - 1e-15
            cnt = mask.sum()
            if cnt == 0:
                break
            s = min(-excess / cnt, (1.0 - h[mask]).min())
            if s <= 0:
                break
            h[mask] += s
    return h

def coord_optimize(h, target, lam, rng):
    n = len(h)
    h = h.copy()
    order = rng.permutation(n)
    for i in order:
        corr = np.correlate(h, 1.0 - h, mode="full")
        a = np.zeros(2 * n - 1)
        for d_idx in range(2 * n - 1):
            d = d_idx - (n - 1)
            slope = 0.0
            j1 = i + d
            if 0 <= j1 < n:
                slope += (1.0 - h[j1])
            j2 = i - d
            if 0 <= j2 < n:
                slope += (-h[j2])
            a[d_idx] = slope
        b = corr - a * h[i]
        vs = np.linspace(0.0, 1.0, 800)
        obj = np.max(a[:, None] * vs[None, :] + b[:, None], axis=0) + lam * vs
        best_v = vs[np.argmin(obj)]
        lo, hi = 0.0, 1.0
        if 0.0 < best_v < 1.0:
            for _ in range(25):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = np.max(a * m1 + b) + lam * m1
                f2 = np.max(a * m2 + b) + lam * m2
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
            best_v = (lo + hi) / 2.0
        h[i] = best_v
    return h

def run_coord_cd(h0, target, rng, n_sweeps=80, lam_init=0.0):
    h = project_sum(h0, target)
    lam = lam_init
    for _ in range(n_sweeps):
        h = coord_optimize(h, target, lam, rng)
        excess = h.sum() - target
        if excess > 1e-10:
            lam = min(lam + 0.05, 50.0)
        elif excess < -1e-10:
            lam = max(lam - 0.05, -50.0)
        else:
            lam *= 0.85
    h = project_sum(h, target)
    return h

def smooth_c5(h, dx, temp):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    scaled = temp * (corr - m)
    return (m + np.log(np.sum(np.exp(scaled)) / len(corr)) / temp) * dx

def try_slsqp_smooth(h0, dx, target, temp, maxiter=200):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    def eqfun(x):
        return x.sum() - target
    eq_con = ({'type': 'eq', 'fun': eqfun},)
    try:
        res = minimize(smooth_c5, h0, args=(dx, temp), method='SLSQP',
                      bounds=bounds, constraints=eq_con,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_sum(res.x, target)
    except Exception:
        return None

def try_slsqp_raw(h0, dx, target, maxiter=300):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    def eqfun(x):
        return x.sum() - target
    eq_con = ({'type': 'eq', 'fun': eqfun},)
    try:
        res = minimize(compute_c5, h0, args=(dx,), method='SLSQP',
                      bounds=bounds, constraints=eq_con,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_sum(res.x, target)
    except Exception:
        return None

def binary_swap_search(h, dx, target, rng, n_iters=5000):
    n = len(h)
    h = h.copy()
    c5_cur = compute_c5(h, dx)
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    for _ in range(n_iters):
        if len(ones) < 1 or len(zeros) < 1:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h_t = h.copy()
        h_t[i], h_t[j] = h_t[j], h_t[i]
        c5_t = compute_c5(h_t, dx)
        if c5_t < c5_cur - 1e-14:
            h = h_t
            c5_cur = c5_t
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
    return h, c5_cur

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    h_best = np.array(initial_h_values, dtype=float).copy()
    n = len(h_best)
    dx = 2.0 / n
    target = n / 2.0
    c5_best = compute_c5(h_best, dx)
    
    def time_left():
        return budget_s - (time.time() - t0)
    
    def update(h_new):
        nonlocal h_best, c5_best
        if h_new is None:
            return
        h_new = project_sum(h_new, target)
        c5 = compute_c5(h_new, dx)
        if c5 < c5_best - 1e-14:
            h_best = h_new.copy()
            c5_best = c5
    
    # Phase 1: Coordinate descent from initial
    if time_left() > 30:
        h_cd = run_coord_cd(h_best, target, rng, n_sweeps=100, lam_init=0.0)
        update(h_cd)
    
    # Phase 2: Coordinate descent with varying lambda
    for lam_i in [0.5, -0.5, 1.0, -1.0, 2.0, -2.0]:
        if time_left() < 20:
            break
        h_cd = run_coord_cd(h_best, target, rng, n_sweeps=60, lam_init=lam_i)
        update(h_cd)
    
    # Phase 3: Smooth SLSQP from current best
    for temp in [20.0, 50.0, 150.0, 400.0]:
        if time_left() < 15:
            break
        h_s = try_slsqp_smooth(h_best, dx, target, temp, maxiter=200)
        update(h_s)
        if h_s is not None:
            h_r = try_slsqp_raw(h_s, dx, target, maxiter=200)
            update(h_r)
    
    # Phase 4: Random restarts with coordinate descent
    n_restarts = 30
    for ri in range(n_restarts):
        if time_left() < 15:
            break
        style = ri % 6
        if style == 0:
            h0 = rng.random(n)
        elif style == 1:
            h0 = np.zeros(n)
            bk = rng.randint(5, 15)
            for st in range(0, n, 2 * bk):
                h0[st:st + bk] = 1.0
        elif style == 2:
            t = np.linspace(0, 4 * np.pi, n)
            h0 = 0.5 + 0.4 * np.sin(t + rng.uniform(0, 6.28))
        elif style == 3:
            h0 = h_best + rng.randn(n) * rng.uniform(0.15, 0.5)
        elif style == 4:
            h0 = np.zeros(n)
            for _ in range(n // 3):
                blk = rng.randint(2, 8)
                st = rng.randint(0, max(1, n - blk))
                h0[st:st + blk] = 1.0
        else:
            h0 = h_best.copy()
            for _ in range(rng.randint(5, 20)):
                h0[rng.randint(n)] = rng.uniform(0, 1)
        h0 = project_sum(h0, target)
        h_cd = run_coord_cd(h0, target, rng, n_sweeps=80, lam_init=rng.uniform(-2, 2))
        update(h_cd)
    
    # Phase 5: Nelder-Mead with penalty on best
    if time_left() > 40:
        pen = 1000.0
        def pen_obj(x):
            p = (x.sum() - target) ** 2
            c = np.sum(np.maximum(-x, 0.0) ** 2 + np.maximum(x - 1.0, 0.0) ** 2)
            return compute_c5(x, dx) + pen * (p + c)
        for restart in range(3):
            if time_left() < 15:
                break
            x0 = h_best + rng.randn(n) * 0.08
            x0 = project_sum(x0, target)
            try:
                res = minimize(pen_obj, x0, method='Nelder-Mead',
                              options={'maxiter': 5000, 'xatol': 1e-10,
                                       'fatol': 1e-14, 'adaptive': True})
                update(project_sum(res.x, target))
            except Exception:
                pass
    
    # Phase 6: Binary swap local search (discrete refinement)
    if time_left() > 20:
        h_bin = (h_best > 0.5).astype(float)
        # Adjust to exact count
        cnt = h_bin.sum()
        if cnt > target:
            excess = int(cnt - target)
            idx = np.where(h_bin == 1.0)[0]
            for k in range(excess):
                h_bin[idx[rng.randint(len(idx))]] = 0.0
        elif cnt < target:
            need = int(target - cnt)
            idx = np.where(h_bin == 0.0)[0]
            for k in range(need):
                h_bin[idx[rng.randint(len(idx))]] = 1.0
        h_sw, c5_sw = binary_swap_search(h_bin, dx, target, rng, n_iters=30000)
        update(h_sw)
        # Also try continuous version near the binary solution
        if time_left() > 15:
            h_mix = 0.3 * h_sw + 0.7 * h_best
            h_mix = project_sum(h_mix, target)
            h_cd = run_coord_cd(h_mix, target, rng, n_sweeps=50, lam_init=0.0)
            update(h_cd)
    
    # Phase 7: Final coordinate descent + SLSQP polish
    if time_left() > 10:
        h_cd = run_coord_cd(h_best, target, rng, n_sweeps=100, lam_init=0.0)
        update(h_cd)
    if time_left() > 8:
        h_f = try_slsqp_raw(h_best, dx, target, maxiter=500)
        update(h_f)
    if time_left() > 5:
        h_f = try_slsqp_smooth(h_best, dx, target, temp=500.0, maxiter=200)
        update(h_f)
    
    h_best = project_sum(h_best, target)
    c5_final = compute_c5(h_best, dx)
    return (h_best, c5_final, n)
