def smooth_val(h, dx, temp):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    return float((m + np.log(np.sum(np.exp((corr - m) * temp)) / len(corr)) / temp) * dx)

def smooth_grad(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    s = (corr - corr.max()) * temp
    w = np.exp(s - s.max())
    w /= w.sum()
    grad = np.zeros(n)
    for j in range(n):
        a = 0.0
        b = 0.0
        for lo in range(2 * n - 1):
            lag = lo - (n - 1)
            wl = w[lo]
            jp = j + lag
            if 0 <= jp < n:
                a += wl * (1.0 - h[jp])
            jm = j - lag
            if 0 <= jm < n:
                b += wl * h[jm]
        grad[j] = (a - b) * dx
    return grad

def make_cons(n, target):
    global _eq_target
    _eq_target = target
    return [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]

def slsqp_smooth(x, dx, target, temp, maxiter=500):
    global _eq_target
    _eq_target = target
    bounds = [(0.0, 1.0)] * len(x)
    cons = make_cons(len(x), target)
    try:
        res = minimize(smooth_val, x, args=(dx, temp), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target)
    except Exception:
        return None

def lbfgs_smooth(x, dx, target, temp, maxiter=2000):
    global _eq_target
    _eq_target = target
    def obj_wrapper(x_):
        pen = 100.0 * (x_.sum() - _eq_target) ** 2
        return smooth_val(x_, dx, temp) + pen
    def grad_wrapper(x_):
        return smooth_grad(x_, temp) + 200.0 * (x_.sum() - _eq_target)
    try:
        res = minimize(obj_wrapper, x, jac=grad_wrapper, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * len(x),
                       options={'maxiter': maxiter, 'gtol': 1e-9})
        return project_feasible(res.x, target)
    except Exception:
        return None

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - t0)

    best_h = np.array(initial_h_values, dtype=float).copy()
    best_n = len(best_h)
    best_c5 = compute_c5(best_h, 2.0 / best_n)

    def try_update(h, n):
        nonlocal best_h, best_n, best_c5
        if h is None:
            return
        h = project_feasible(h, n / 2.0)
        c5 = compute_c5(h, 2.0 / n)
        if c5 < best_c5 - 1e-15:
            best_h, best_n, best_c5 = h.copy(), n, c5

    # Phase 1: Adam gradient descent with annealing
    n = best_n
    dx = 2.0 / n
    target = n / 2.0

    for temp in [120, 60, 30, 18, 12, 8, 5, 3, 2, 1.5, 1]:
        if tl() < 15:
            break
        h = best_h.copy()
        m_v = np.zeros(n)
        v_v = np.zeros(n)
        lr = 0.003
        for step in range(3000):
            if tl() < 5:
                break
            g = smooth_grad(h, temp)
            m_v = 0.9 * m_v + 0.1 * g
            v_v = 0.999 * v_v + 0.001 * g * g
            m_hat = m_v / (1 - 0.9 ** (step + 1))
            v_hat = v_v / (1 - 0.999 ** (step + 1))
            h_new = h - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
            h_new = project_feasible(h_new, target)
            if h_new is not None and h_new.sum() > 0:
                h = h_new
            if step % 500 == 499:
                c5_chk = compute_c5(h, dx)
                if c5_chk < best_c5:
                    best_h, best_c5 = h.copy(), c5_chk
        if tl() > 25:
            h_sl = slsqp_smooth(h, dx, target, max(1.0, temp * 0.3))
            try_update(h_sl, n)

    # Phase 2: L-BFGS-B with analytic gradient
    if tl() > 60:
        for restart in range(4):
            if tl() < 30:
                break
            amp = 0.08 if restart == 0 else 0.25
            h0 = best_h + rng.randn(n) * amp
            h0 = project_feasible(h0, target)
            h_new = lbfgs_smooth(h0, dx, target, 3.0)
            try_update(h_new, n)
            if tl() > 20:
                h_new2 = slsqp_smooth(h_new if h_new is not None else h0, dx, target, 2.0)
                try_update(h_new2, n)

    # Phase 3: Structured diverse restarts
    structures = [
        'block', 'sine', 'plateau', 'alternating', 'gaussian'
    ]
    for si, sname in enumerate(structures):
        if tl() < 25:
            break
        if sname == 'block':
            h0 = np.zeros(n)
            nb = 3 + si
            for st in range(0, n, 2 * nb):
                h0[st:st + nb] = 1.0
        elif sname == 'sine':
            t = np.linspace(0, 4 * np.pi, n)
            h0 = 0.5 + 0.35 * np.sin(t + 0.5 * si)
        elif sname == 'plateau':
            h0 = np.zeros(n)
            w = n // 4
            for st in range(0, n, w * 2):
                h0[st:st + w] = 0.8
        elif sname == 'alternating':
            h0 = np.where(np.arange(n) % 4 < 2, 1.0, 0.0)
            h0 += rng.randn(n) * 0.1
        else:
            ctr = n // 2 + si * 3
            t = np.linspace(-1, 1, n)
            h0 = 0.5 + 0.4 * np.exp(-((t - 0.1 * si) ** 2) * 8)
        h0 = project_feasible(h0, target)
        h_new = slsqp_smooth(h0, dx, target, 5.0, maxiter=400)
        try_update(h_new, n)
        if h_new is not None and tl() > 15:
            h_new2 = lbfgs_smooth(h_new, dx, target, 3.0)
            try_update(h_new2, n)

    # Phase 4: Lag-equalization heuristic
    if tl() > 40:
        h_c = best_h.copy()
        c5_c = best_c5
        dx_c = 2.0 / best_n
        tgt_c = best_n / 2.0
        step_sz = 0.03
        for it in range(15000):
            if tl() < 5:
                break
            corr = np.correlate(h_c, 1.0 - h_c, mode="full")
            top_lags = np.argsort(corr)[-5:]
            idx = rng.randint(best_n)
            lag_choice = top_lags[rng.randint(len(top_lags))]
            lag = int(lag_choice) - (best_n - 1)
            # Perturb h to reduce overlap at chosen lag
            delta = step_sz * (1.0 if rng.rand() < 0.5 else -1.0)
            h_t = h_c.copy()
            h_t[idx] = np.clip(h_t[idx] + delta, 0, 1)
            h_t = project_feasible(h_t, tgt_c)
            c5_t = compute_c5(h_t, dx_c)
            if c5_t < c5_c - 1e-14:
                h_c, c5_c = h_t, c5_t
                if c5_t < best_c5:
                    best_h, best_c5 = h_c.copy(), c5_t
            if it % 1000 == 999:
                step_sz *= 0.75

    # Phase 5: Multi-grid exploration
    for trial_n in [60, 80, 56, 72, 90]:
        if tl() < 25:
            break
        if trial_n == best_n:
            continue
        old_x = np.linspace(0, 2, best_n)
        new_x = np.linspace(0, 2, trial_n)
        h_interp = np.interp(new_x, old_x, best_h, left=0, right=0)
        tgt_t = trial_n / 2.0
        dx_t = 2.0 / trial_n
        h_interp = project_feasible(h_interp, tgt_t)
        h_new = slsqp_smooth(h_interp, dx_t, tgt_t, 3.0, maxiter=400)
        try_update(h_new, trial_n)

    # Phase 6: Final SLSQP polish
    if tl() > 10:
        n_f = best_n
        dx_f = 2.0 / n_f
        tgt_f = n_f / 2.0
        h_f = slsqp_smooth(best_h, dx_f, tgt_f, 1.5, maxiter=800)
        try_update(h_f, n_f)
        if tl() > 10:
            h_f2 = lbfgs_smooth(best_h, dx_f, tgt_f, 1.0, maxiter=3000)
            try_update(h_f2, n_f)

    best_h = project_feasible(best_h, best_n / 2.0)
    final_c5 = compute_c5(best_h, 2.0 / best_n)
    return (best_h, final_c5, best_n)
