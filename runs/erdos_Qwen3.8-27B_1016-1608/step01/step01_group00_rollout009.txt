import numpy as np
from scipy.optimize import minimize
import time

_N = 70

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c5_smooth(h, sharpness=800.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    if m <= 1e-15:
        return 0.0
    c = (corr - m) * sharpness
    c = c - np.max(c)
    return m + np.log(np.sum(np.exp(c))) / sharpness

def _sum_con(x):
    return np.sum(x) - _N / 2.0

def _sum_jac(x):
    return np.ones(_N)

def repair(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
        else:
            mask = h > 1e-13
        if not np.any(mask):
            h[:] = target / n
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = repair(np.clip(res.x, 0.0, 1.0), n)
