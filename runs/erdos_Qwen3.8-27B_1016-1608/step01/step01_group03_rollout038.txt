For this to be small, we want sum(h[i]^2) to be large, i.e., h should be close to binary.

- Have transitions arranged so that no shift k has many (1,0) pairs

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def repair(h, target):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        d = target - np.sum(h)
        if abs(d) < 1e-14:
            break
        m = (h < 1.0 - 1e-15) if d > 0 else (h > 1e-15)
        c = int(m.sum())
        if c == 0:
            break
        h[m] += d / c
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_cv(v, alpha):
    v = np.asarray(v, dtype=float)
    m = float(v.max())
    if alpha < 1:
        return m
    s = np.clip(v - m, -50.0, 0.0)
    return m + float(np.log(np.sum(np.exp(alpha * s)))) / alpha

class ObjSmooth:
    def __init__(self, dx, alpha):
        self.dx = dx
        self.alpha = alpha
    def __call__(self, x):
        xc = np.clip(x, 0.0, 1.0)
        cv = np.correlate(xc, 1.0 - xc, mode="full") * self.dx
        return smooth_max_cv(cv, self.alpha)

class ObjDirect:
    def __init__(self, dx):
        self.dx = dx
    def __call__(self, x):
        xc = np.clip(x, 0.0, 1.0)
        return compute_c5(xc, self.dx)

def lsqb_smooth(h, dx, alpha, bnds):
    obj = ObjSmooth(dx, alpha)
    try:
        r = minimize(obj, h, method='L-BFGS-B', bounds=bnds,
                     maxiter=500, ftol=1e-18, gtol=1e-14, maxfun=8000)
        return r.x
    except Exception:
        return h.copy()

def slsqp_direct(h, dx, bnds, maxit=400, ftol=1e-15):
    obj = ObjDirect(dx)
    try:
        r = minimize(obj, h, method='SLSQP', bounds=bnds,
                     maxiter=maxit, ftol=ftol)
        return r.x
    except Exception:
        return h.copy()
