    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corrs(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return corr

def repair(h, n):
    h = np.array(h, dtype=float).copy()
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        d = n / 2.0 - np.sum(h)
        if abs(d) < 1e-12:
            break
        h += d / n
    return np.clip(h, 0.0, 1.0)

def dh_lp(h, n, active_lags):
    """LP-based descent step. Returns new h."""
    dx = 2.0 / n
    corrs = compute_corrs(h)
    
    # Get gradients for active lags
    # For lag index k in [0, 2n-2], shift = k - (n-1)
    # grad_k[j] = (1-h[j-shift]) - h[j+shift] (with 0 padding outside [0,n))
    
    A_ub = []
    b_ub = []
    
    for k in active_lags:
        shift = k - (n - 1)
        grad = np.zeros(n)
        # Term 1: (1 - h[j-shift]) for valid j
        lo1 = max(0, shift)
        hi1 = min(n, n + shift)
        if lo1 < hi1:
            grad[lo1:hi1] += (1.0 - h[lo1 - shift:hi1 - shift])
        # Term 2: -h[j+shift] for valid j
        lo2 = max(0, -shift)
        hi2 = min(n, n - shift)
        if lo2 < hi2:
            grad[lo2:hi2] -= h[lo2 + shift:hi2 + shift]
        
        # Constraint: grad · d - t <= -f_k
        row = np.zeros(n + 1)
        row[:n] = grad
        row[n] = -1.0
        A_ub.append(row)
        b_ub.append(-corrs[k])
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds: -h[i] <= d[i] <= 1-h[i], t >= 0 (well, t >= max(corrs) but LP will handle)
    bounds = [(-h[i], 1.0 - h[i])] + [(0, None)]
    
    # Objective: minimize t
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')

        return h
    
    d = res.x[:n]
    h_new = h + d

    s = np.clip((c - m) / eps, -50.0, 50.0)
    return float(m + eps * np.log(np.sum(np.exp(s))))

            bh, bc = hn, c5
    for i in range(150):
        if time.time() - t0 > t_lim:
            break
        r = i % 6
        if r == 0:
            hp = bh + np.random.randn(n) * 0.05
        elif r == 1:
            hp = bh.copy()
            s = np.random.randint(0, n - 4)
            l = np.random.randint(2, 7)
            hp[s:s + l] = 1.0 - hp[s:s + l]
        elif r == 2:
            k2 = np.random.randint(1, 6)
            ph = np.random.uniform(0, 2 * np.pi)
            hp = bh + 0.06 * np.sin(2 * np.pi * k2 * np.arange(n) / n + ph)
        elif r == 3:
            hp = bh.copy()
            for _ in range(np.random.randint(1, 4)):
                idx = np.random.randint(n)
                hp[idx] = 1.0 - hp[idx]
        elif r == 4:
            hp = bh + np.random.randn(n) * 0.02
        else:
            hp = bh.copy()
            idx = np.random.choice(n, size=max(1, n // 4), replace=False)
            hp[idx] = np.random.uniform(0, 1, size=len(idx))
        hp = repair(hp, n)
        active = active_lags_for(hp, 0.02)
        hn = lp_step(hp, n, active)
        c5 = compute_c5(hn)
        if c5 < bc:
            bh, bc = hn, c5
    if time.time() - t0 < t_lim:
        hn = do_slsqp(bh, n, 0.00005, 400)
        c5 = compute_c5(hn)
        if c5 < bc:
            bh, bc = hn, c5
        active = active_lags_for(bh, 0.005)
        hn = lp_step(bh, n, active)
        c5 = compute_c5(hn)
        if c5 < bc:
            bh, bc = hn, c5
    if time.time() - t0 < t_lim:
        for _ in range(3000):
            if time.time() - t0 > t_lim - 5:
                break
            hp = bh + np.random.randn(n) * 0.003
            hp = repair(hp, n)
            c5 = compute_c5(hp)
            if c5 < bc:
                bh, bc = hp, c5
    return repair(bh, n), bc

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = 0.5 * np.ones(80)
    n0 = len(h0)
    h0 = repair(h0, n0)

    bh, bc = optimize_h(h0, n0, t0, budget_s, 300)

    if time.time() - t0 < budget_s - 200:
        n1 = n0 * 2
        x0 = np.linspace(0, 1, n0, endpoint=False) + 0.5 / n0
        x1 = np.linspace(0, 1, n1, endpoint=False) + 0.5 / n1
        h_up = np.interp(x1, x0, h0)
        h_up = repair(h_up, n1)
        bh1, bc1 = optimize_h(h_up, n1, t0, budget_s, 50)
        if bc1 < bc:
            bh, bc = bh1, bc1

    if time.time() - t0 < budget_s - 30:
        hn = do_slsqp(bh, len(bh), 0.00003, 500)
        c5 = compute_c5(hn)
        if c5 < bc:
            bh, bc = hn, c5
        active = active_lags_for(bh, 0.003)
        hn = lp_step(bh, len(bh), active)
        c5 = compute_c5(hn)
        if c5 < bc:
            bh, bc = hn, c5

    fh = repair(bh, len(bh))
    fc = compute_c5(fh)
    return (fh.tolist(), float(fc), len(fh))
