            s = min(-excess / cnt, (1.0 - h[mask]).min())
            h[mask] += s
    return h

def coord_opt(h, target, dx, ncand=80, local_span=0.25):
    n = len(h)
    c5 = compute_c5(h, dx)
    for i in range(n):
        hi = h[i]
        local = np.linspace(max(0.0, hi - local_span), min(1.0, hi + local_span), min(30, ncand // 2 + 1))
        glob = np.linspace(0.0, 1.0, max(1, ncand // 2))
        cands = np.unique(np.concatenate([local, glob, [0.0, 1.0]]))
        cands = cands[(cands >= 0.0) & (canks <= 1.0)] if False else cands[(cands >= 0.0) & (cands <= 1.0)]
        best_c5 = c5
        best_h = h
        for t in cands:
            if abs(t - hi) < 1e-9:
                continue
            h_try = h.copy()
            h_try[i] = t
            h_try = projection(h_try, target)
            c_try = compute_c5(h_try, dx)
            if c_try < best_c5:
                best_c5 = c_try
                best_h = h_try
        if best_c5 < c5:
            h = best_h
            c5 = best_c5
    return h, c5

def pair_opt(h, target, dx, ntrials, rng, dmax=0.08):
    n = len(h)
    c5 = compute_c5(h, dx)
    for _ in range(ntrials):
        i, j = rng.choice(n, 2, replace=False)
        d = rng.uniform(0.005, dmax)
        for sgn in [1, -1]:
            h_try = h.copy()
            h_try[i] += sgn * d
            h_try[j] -= sgn * d
            np.clip(h_try, 0, 1, out=h_try)
            h_try = projection(h_try, target)
            c_try = compute_c5(h_try, dx)
            if c_try < c5:
                h = h_try
                c5 = c_try
    return h, c5

def anneal_step(h, target, dx, temp, rng, n):
    mt = rng.randint(4)
    if mt == 0:
        i = rng.randint(n)
        h_try = h.copy()
        h_try[i] += rng.randn() * 0.04
    elif mt == 1:
        i, j = rng.choice(n, 2, replace=False)
        d = rng.uniform(0.01, 0.1)
        h_try = h.copy()
        h_try[i] += d
        h_try[j] -= d
    elif mt == 2:
        i = rng.randint(max(1, n - 6))
        L = rng.randint(2, 6)
        h_try = h.copy()
        h_try[i:i + L] += rng.randn(L) * 0.03
    else:
        i = rng.randint(max(0, n - 2))
        h_try = h.copy()
        h_try[i + 1] += (h[i] - h[i + 1]) * 0.5
        h_try[i] += (h[i + 1] - h[i]) * 0.5
    np.clip(h_try, 0, 1, out=h_try)
    h_try = projection(h_try, target)
    return h_try

def slsqp(h0, dx, target, maxiter=200):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_diff, 'args': (target,)}]
    try:
        res = minimize(c5_func, h0, args=(dx,), method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return projection(res.x, target)
    except Exception:
        return None

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0

    best_h = projection(initial_h_values.astype(float).copy(), target)
    best_c5 = compute_c5(best_h, dx)

    def time_left():
        return budget_s - (time.time() - t0)

    # Phase 1: Coordinate descent to local minimum
    h = best_h.copy()
    for p in range(40):
        if time_left() < 60:
            break
        h, c5 = coord_opt(h, target, dx, ncand=80, local_span=0.3)
        if c5 < best_c5:
            best_h, best_c5 = h.copy(), c5

    # Phase 2: Alternating annealing + coordinate descent
    cycle = 0
    while time_left() > 80:
        cycle += 1
        temp = 0.004 / (1.0 + 0.2 * cycle)
        h = best_h.copy()
        c5 = best_c5
        n_aneal = min(3000, int(20000 * temp / 0.004))
        for trial in range(n_aneal):
            if time_left() < 80:
                break
            h_try = anneal_step(h, target, dx, temp, rng, n)
            c_try = compute_c5(h_try, dx)
            if c_try < c5 or rng.random() < np.exp(-(c_try - c5) / max(temp, 1e-12)):
                h = h_try
                c5 = c_try
        if c5 < best_c5:
            best_h, best_c5 = h.copy(), c5
        # Local refine after annealing
        if time_left() > 80:
            for p in range(15):
                if time_left() < 80:
                    break
                h, c5 = coord_opt(h, target, dx, ncand=60, local_span=0.2)
                if c5 < best_c5:
                    best_h, best_c5 = h.copy(), c5

    # Phase 3: Extensive fine coordinate descent
    while time_left() > 50:
        h = best_h.copy()
        c5 = best_c5
        improved = True
        passes = 0
        while improved and passes < 60 and time_left() > 50:
            improved = False
            passes += 1
            for i in range(n):
                if time_left() < 50:
                    break
                hi = h[i]
                cands = np.linspace(max(0.0, hi - 0.08), min(1.0, hi + 0.08), 40)
                cands = np.unique(np.concatenate([cands, [0.0, 1.0]]))
                for t in cands:
                    if abs(t - hi) < 1e-9:
                        continue
                    h_try = h.copy()
                    h_try[i] = t
                    h_try = projection(h_try, target)
                    c_try = compute_c5(h_try, dx)
                    if c_try < c5 - 1e-15:
                        h = h_try
                        c5 = c_try
                        hi = t
                        improved = True
                        break
        if c5 < best_c5:
            best_h, best_c5 = h.copy(), c5

    # Phase 4: Pairwise fine moves
    while time_left() > 30:
        h = best_h.copy()
        c5 = best_c5
        h, c5 = pair_opt(h, target, dx, 5000, rng, dmax=0.04)
        if c5 < best_c5:
            best_h, best_c5 = h.copy(), c5
        h, c5 = coord_opt(h, target, dx, ncand=50, local_span=0.1)
        if c5 < best_c5:
            best_h, best_c5 = h.copy(), c5

    # Phase 5: SLSQP polish
    if time_left() > 15:
        h2 = slsqp(best_h, dx, target, maxiter=300)
        if h2 is not None:
            c52 = compute_c5(h2, dx)
            if c52 < best_c5:
                best_h, best_c5 = h2, c52

    # Phase 6: Ultra-fine coordinate scan
    while time_left() > 5:
        improved = False
        for i in range(n):
            if time_left() < 5:
                break
            hi = best_h[i]
            cands = np.linspace(max(0.0, hi - 0.02), min(1.0, hi + 0.02), 25)
            for t in cands:
                if abs(t - hi) < 1e-10:
                    continue
                h_try = best_h.copy()
                h_try[i] = t
                h_try = projection(h_try, target)
                c_try = compute_c5(h_try, dx)
                if c_try < best_c5 - 1e-15:
                    best_h = h_try
                    best_c5 = c_try
                    hi = t
                    improved = True
        if not improved:
            break

    best_h = projection(best_h, target)
    final_c5 = compute_c5(best_h, dx)
    return (best_h, final_c5, n)
