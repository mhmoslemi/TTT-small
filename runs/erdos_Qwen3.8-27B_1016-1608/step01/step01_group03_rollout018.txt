from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def repair(h, target):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-12:
            break
        if d > 0:
            m = h < 1.0 - 1e-10
        else:
            m = h > 1e-10
        c = int(m.sum())
        if c == 0:
            break
        h[m] += d / c
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_max(vals, T):
    m = np.max(vals)
    if T < 1e-12:
        return m
    return T * np.log(np.sum(np.exp((vals - m) / T))) + m

def grad_smooth_max(h, n, T, mu, target):
    """Compute gradient of soft_max(corr, T) + mu*(sum(h)-target)^2"""
    v = 1.0 - h
    corr = np.correlate(h, v, mode="full")
    # Weights
    e = np.exp((corr - np.max(corr)) / T)
    w = e / np.sum(e)
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        s = k - (n - 1)
        wk = w[k]
        # j as first index: j in [max(0,-s), min(n-1, n-1-s)]
        j1_lo = max(0, -s)
        j1_hi = min(n - 1, n - 1 - s)
        if j1_lo <= j1_hi:
            js = np.arange(j1_lo, j1_hi + 1)
            grad[js] += wk * (1.0 - h[js + s])
        # j as second index: j in [max(0,s), min(n-1, n-1+s)]
        j2_lo = max(0, s)
        j2_hi = min(n - 1, n - 1 + s)
        if j2_lo <= j2_hi:
            js = np.arange(j2_lo, j2_hi + 1)
            grad[js] -= wk * h[js - s]
    
    # Penalty gradient
    s_val = np.sum(h)
    grad += 2.0 * mu * (s_val - target)
    return grad
