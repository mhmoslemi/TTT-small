from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def projection(h, target):
    h = np.array(h, dtype=float, copy=True)
    for _ in range(2000):
        h = np.clip(h, 0.0, 1.0)
        excess = h.sum() - target
        if abs(excess) < 1e-14:
            break
        if excess > 0:
            mask = h > 1e-12
            if not mask.any(): break
            s = min(excess / mask.sum(), (h[mask] - 1e-12).min())
            if s <= 0: break
            h[mask] -= s
        else:
            mask = h < 1 - 1e-12
            if not mask.any(): break
            s = min(-excess / mask.sum(), (1 - 1e-12 - h[mask]).min())
            if s <= 0: break
            h[mask] += s
    return h

def smooth_max_obj(h, dx, temp):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(temp * (corr - m))) / len(corr)) / temp) * dx

def smooth_max_grad_h(h, dx, temp):
    """Gradient of smooth max of correlations w.r.t. h."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    # Softmax weights
    log_w = temp * (corr - m) - np.log(len(corr))  # for numerical stability
    log_w -= log_w.max()
    w = np.exp(log_w)
    w /= w.sum()
    
    # shifts: m_idx - n + 1 for m_idx in 0..2n-2
    shifts = np.arange(2*n-1) - (n-1)  # range -(n-1) to n-1
    
    grad = np.zeros(n)
    for j in range(n):
        for m_idx in range(2*n-1):
            s = shifts[m_idx]
            # Contribution from h[j] as first array element: (1 - h[j+s]) if 0 <= j+s < n
            if 0 <= j + s < n:
                grad[j] += w[m_idx] * (1 - h[j+s])
            # Contribution from h[j] as second array element: -h[j-s] if 0 <= j-s < n
            if 0 <= j - s < n:
                grad[j] += w[m_idx] * (-h[j-s])
    return grad * dx

def coordinate_descent_step(h, j, k, dx):
    """Optimize h[j] + h[k] = const by adjusting t: h[j]+t, h[k]-t."""
    n = len(h)
    h_j = h[j]
    h_k = h[k]
    
    # Range of t: 0 <= h_j + t <= 1, 0 <= h_k - t <= 1
    t_min = max(-h_j, h_k - 1)
    t_max = min(1 - h_j, h_k)
    
    # corr_k as function of t:
    # corr_m = sum_i h[i](1-h[i+s_m])
    # Only terms involving h[j] or h[k] change
    # h[j] → h_j + t, h[k] → h_k - t
    # 
    # For a given lag s:
    # corr_s includes h[j]*(1-h[j+s]) and h[j-s]*(1-h[j])
    #           and h[k]*(1-h[k+s]) and h[k-s]*(1-h[k])
    # 
    # After substitution:
    # corr_s(t) = (h_j+t)(1-h[j+s]) + h[j-s](1-h_j-t) + [k terms] + [const]
    #           + (h_k-t)(1-h[k+s]) + h[k-s](1-h_k+t) + [const]
    # = const + t*(1-h[j+s]) + (-h[j-s])*t + (-t)*(1-h[k+s]) + h[k-s]*t
    # = const + t * [(1-h[j+s]) - h[j-s] - (1-h[k+s]) + h[k-s]]
    # = const + t * [h[k+s] - h[j+s] - h[j-s] + h[k-s]]
    
    # So each corr_s is linear in t with slope:
    # slope_s = (1 - h[j+s]) - h[j-s] - (1 - h[k+s]) + h[k-s]
    #         = -(h[j+s]) + h[j-s] - h[k+s] + h[k-s] (with boundary conditions)
    
    # Compute slopes for each lag
    full_corr0 = np.correlate(h, 1-h, mode="full")
    shifts = np.arange(2*n-1) - (n-1)
    
    slopes = np.zeros(2*n-1)
    intercepts = full_corr0  # corr at t=0
    
    for mi, s in enumerate(shifts):
        sl = 0.0
        # h[j] term: (1 - h[j+s]) if valid
        if 0 <= j + s < n: sl += (1 - h[j+s])
        # h[j-s] term: -h[j-s] if valid  
        if 0 <= j - s < n: sl += -h[j-s]
        # h[k] term: -(1 - h[k+s]) if valid
        if 0 <= k + s < n: sl += -(1 - h[k+s])
        # h[k-s] term: +h[k-s] if valid
        if 0 <= k - s < n: sl += h[k-s]
        slopes[mi] = sl
    
    # Now min_t max_m (intercept_m + slopes[m]*t) for t ∈ [t_min, t_max]
    # This is a 1D convex piecewise-linear problem
    # Optimal t is where 0 is in the subgradient, i.e., where slopes change sign around 0
    
    # Collect all breakpoints where two lines cross
    breakpoints = [t_min, t_max]
    for m1 in range(2*n-1):
        for m2 in range(m1+1, 2*n-1):
            if abs(slopes[m1] - slopes[m2]) > 1e-15:
                bp = (intercepts[m2] - intercepts[m1]) / (slopes[m1] - slopes[m2])
                if t_min < bp < t_max:
                    breakpoints.append(bp)
    
    # Evaluate at all breakpoints and endpoints
    best_t = t_min
    best_val = np.max(intercepts + slopes * t_min)
    for t_candidate in breakpoints:
        val = np.max(intercepts + slopes * t_candidate)
        if val < best_val:
            best_val = val
            best_t = t_candidate
    
    return best_t

def projection(h, target):
    h = np.array(h, dtype=float, copy=True)
    for _ in range(3000):
        h = np.clip(h, 0.0, 1.0)
        excess = h.sum() - target
        if abs(excess) < 1e-14:
            break
        if excess > 0:
            mask = h > 1e-12
            if not mask.any(): break
            s = min(excess / mask.sum(), (h[mask] - 1e-12).min())
            if s <= 0: break
            h[mask] -= s
        else:
            mask = h < 1 - 1e-12
            if not mask.any(): break
            s = min(-excess / mask.sum(), (1 - 1e-12 - h[mask]).min())
            if s <= 0: break
            h[mask] += s
    return h

    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(temp*(corr-m)))/len(corr))/temp) * dx

import numpy as np
from scipy.optimize import minimize
import time

_g_target = 0.0

def _c_eq(x):
    return x.sum() - _g_target

def _c_eq_jac(x):
    return np.ones_like(x)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def projection(h, target):
    h = np.array(h, dtype=float, copy=True)
    for _ in range(5000):
        h = np.clip(h, 0.0, 1.0)
        excess = h.sum() - target
        if abs(excess) < 1e-15:
            break
        if excess > 0:
            m = h > 1e-14
            if not m.any():
                break
            s = min(excess / m.sum(), (h[m] - 1e-14).min())
            if s <= 0:
                break
            h[m] -= s
        else:
            m = h < 1.0 - 1e-14
            if not m.any():
                break
            s = min(-excess / m.sum(), (1.0 - 1e-14 - h[m]).min())
            if s <= 0:
                break
            h[m] += s
    return h

def smooth_obj(h, dx, temp):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    e = np.exp(temp * (corr - m))
    return (m + np.log(e.sum() / len(corr)) / temp) * dx

def slsqp_optimize(h0, dx, target, temp, maxiter):
    global _g_target
    _g_target = target
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': _c_eq, 'jac': _c_eq_jac}
    try:
        res = minimize(smooth_obj, h0, args=(dx, temp), method='SLSQP',
                       bounds=bounds, constraints=(cons,),
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return projection(res.x, target)
    except Exception:
        return None

def pair_cd(h, dx, target, n_iter, rng, t_limit):
    n = len(h)
    h = h.copy()
    shifts = np.arange(2 * n - 1) - (n - 1)
    for it in range(n_iter):
        if time.time() > t_limit:
            break
        j = int(rng.randint(n))
        k = int(rng.randint(n))
        if j == k:
            continue
        hj, hk = h[j], h[k]
        t_lo = max(-hj, hk - 1.0)
        t_hi = min(1.0 - hj, hk)
        if t_hi - t_lo < 1e-13:
            continue
        corr0 = np.correlate(h, 1.0 - h, mode="full")
        slopes = np.zeros(2 * n - 1)
        for mi in range(2 * n - 1):
            s = shifts[mi]
            sl = 0.0
            if 0 <= j + s < n:
                sl += (1.0 - h[j + s])
            if 0 <= j - s < n:
                sl -= h[j - s]
            if 0 <= k + s < n:
                sl -= (1.0 - h[k + s])
            if 0 <= k - s < n:
                sl += h[k - s]
            slopes[mi] = sl
        top_k = min(20, 2 * n - 1)
        top_idx = np.argsort(-corr0)[:top_k]
        cands = [t_lo, t_hi, 0.0]
        mid = (t_lo + t_hi) / 2.0
        cands.append(mid)
        for a in range(top_k):
            for b in range(a + 1, top_k):
                ia, ib = int(top_idx[a]), int(top_idx[b])
                ds = slopes[ia] - slopes[ib]
                if abs(ds) > 1e-12:
                    bp = (corr0[ib] - corr0[ia]) / ds
                    if t_lo + 1e-12 < bp < t_hi - 1e-12:
                        cands.append(bp)
        best_t, best_v = 0.0, np.inf
        for tc in cands:
            hc = h.copy()
            hc[j] = hj + tc
            hc[k] = hk - tc
            v = float(np.max(np.correlate(hc, 1.0 - hc, mode="full")))
            if v < best_v:
                best_v, best_t = v, tc
        h[j] = hj + best_t
        h[k] = hk - best_t
    return projection(h, target)

def make_binary_init(n, rng, pattern):
    h = np.zeros(n)
    nh = n // 2
    if pattern == 0:
        blk = max(3, int(rng.randint(3, 10)))
        for i in range(0, n, 2 * blk):
            h[i:i + blk] = 1.0
    elif pattern == 1:
        phi = 0.618033988749895
        pos = set()
        for i in range(nh):
            pos.add(int((i * phi) % 1.0 * n))
        while len(pos) < nh:
            pos.add(int(rng.randint(n)))
        for p in sorted(pos)[:nh]:
            h[p] = 1.0
    elif pattern == 2:
        x = np.linspace(0, 2, n)
        freq = rng.uniform(1.5, 5.0)
        phase = rng.uniform(0, 6.28)
        h = (np.sin(freq * np.pi * x + phase) > 0).astype(float)
    else:
        h[rng.choice(n, nh, replace=False)] = 1.0
    n_ones = int(np.sum(h > 0.5))
    if n_ones > nh:
        ones = np.where(h > 0.5)[0]
        h[rng.choice(ones, n_ones - nh, replace=False)] = 0
    elif n_ones < nh:
        zeros = np.where(h < 0.5)[0]
        h[rng.choice(zeros, nh - n_ones, replace=False)] = 1
    return h

    t0 = time.time()
    rng = np.random.RandomState(seed)
    hard_deadline = t0 + budget_s - 5

    best_h = np.array(initial_h_values, dtype=float).copy()

    for attempt in range(5):
        if time.time() > hard_deadline - 50:
            break
        h0 = projection(best_h + rng.randn(n) * 0.015, target)
        h_new = slsqp_optimize(h0, dx, target, 1000.0, 500)
        if h_new is not None:
            c5 = compute_c5(h_new, dx)

            if time.time() > hard_deadline - 10:
                break
            h_bin = make_binary_init(n, rng, pat)
            h_soft = 0.85 * h_bin + 0.15 * (1.0 - h_bin)
            h_soft = projection(h_soft, target)
            h_new = slsqp_optimize(h_soft, dx, target, 100.0, 400)

                c5 = compute_c5(h_new, dx)
                if c5 < best_c5:
                    best_h, best_c5 = h_new.copy(), c5
            h_new2 = slsqp_optimize(h_bin, dx, target, 30.0, 400)
            if h_new2 is not None:
                c5 = compute_c5(h_new2, dx)
                if c5 < best_c5:
                    best_h, best_c5 = h_new2.copy(), c5

    # Phase 6: Final SLSQP polish + coordinate descent
    if time.time() < hard_deadline - 3:
        h_new = slsqp_optimize(best_h, dx, target, 2000.0, 500)
        if h_new is not None:
            c5 = compute_c5(h_new, dx)
            if c5 < best_c5:
                best_h, best_c5 = h_new.copy(), c5
        h_final = pair_cd(best_h, dx, target, 2000, rng, hard_deadline - 1)
        c5 = compute_c5(h_final, dx)
        if c5 < best_c5:
            best_h, best_c5 = h_final, c5

    best_h = projection(best_h, target)
    final_c5 = compute_c5(best_h, dx)
    return (best_h, final_c5, n)
