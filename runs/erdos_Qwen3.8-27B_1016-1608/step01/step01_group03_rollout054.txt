    """P is a set of positions with h=1, |P|=n/2
    Minimize max_k (n/2 - autocorr(k))
    Equivalently maximize min_k autocorr(k)
    """
    P = set(P)
    m = n // 2  # number of 1s
    
    def compute_c5_binary():
        h = np.zeros(n)
        for i in P:
            h[i] = 1.0
        return compute_c5(h, 2.0/n)
    
    best_c = compute_c5_binary()
    
    for it in range(iterations):
        # Try all pairs of swaps
        best_delta = 0
        best_swap = None
        for i in P:
            for j in list(range(n)):
                if j in P:
                    continue
                # Try swapping i (currently 1) with j (currently 0)
                P.remove(i)
                P.add(j)
                c = compute_c5_binary()
                if c < best_c - 1e-15:

        P.add(j)

    for i in P:

    return h, best_c

    t = _OBJ_TEMP
    return _DX_VAL * (m + np.log(np.sum(np.exp((corr - m) * t))) / t)

def obj_exact_func(x):
    return compute_c5(np.clip(x, 0.0, 1.0), _DX_VAL)

def subgradient_step(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_k = 5
    idx = np.argsort(corr)[::-1][:top_k]
    grad = np.zeros(n)
    for k in idx:
        shift = k - (n - 1)
        for j in range(n):
            g = 0.0
            js = j + shift
            jm = j - shift
            if 0 <= js < n:
                g += (1.0 - h[js])
            if 0 <= jm < n:
                g -= h[jm]
            grad[j] += g
    grad /= float(len(idx))
    return grad

def run(seed=42, budget_s=1100, **kwargs):
    global _SUM_TARGET, _DX_VAL, _OBJ_TEMP
    np.random.seed(seed)
    t0 = time.time()

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.ones(70) * 0.5

    n = len(h0)
    tgt = n / 2.0
    dx = 2.0 / n
    _SUM_TARGET = tgt
    _DX_VAL = dx
    h0 = repair(h0, tgt)

    best_c5 = compute_c5(h0, dx)
    best_h = h0.copy()

    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constr}]

    # Phase 1: Smooth SLSQP with multiple temperatures (annealing)
    for temp in [5, 15, 50, 200, 800]:
        if time.time() - t0 > budget_s - 280:
            break
        _OBJ_TEMP = float(temp)
        for _ in range(3):
            if time.time() - t0 > budget_s - 280:
                break
            try:
                r = minimize(obj_smooth_func, best_h, method='SLSQP',
                           bounds=bnds, constraints=cons,
                           options={'maxiter': 400, 'ftol': 1e-14})
                hn = repair(r.x, tgt)
                cn = compute_c5(hn, dx)
                if cn < best_c5:
                    best_c5, best_h = cn, hn
            except Exception:
                pass

    # Phase 2: Exact SLSQP (multiple passes)
    for _ in range(5):
        if time.time() - t0 > budget_s - 200:
            break
        try:
            r = minimize(obj_exact_func, best_h, method='SLSQP', bounds=bnds,
                        constraints=cons, options={'maxiter': 500, 'ftol': 1e-15})
            hn = repair(r.x, tgt)
            cn = compute_c5(hn, dx)
            if cn < best_c5:
                best_c5, best_h = cn, hn
        except Exception:
            break

    # Phase 3: Subgradient steps + SLSQP polish
    if time.time() - t0 < budget_s - 150:
        for _ in range(20):
            if time.time() - t0 > budget_s - 150:
                break
            g = subgradient_step(best_h)
            found = False
            for alpha in [0.02, 0.01, 0.005, 0.002, 0.001]:
                trial = best_h - alpha * g
                trial = repair(trial, tgt)
                tc = compute_c5(trial, dx)
                if tc < best_c5 - 1e-14:
                    best_h, best_c5 = trial, tc
                    found = True
                    break
            if time.time() - t0 > budget_s - 150:
                break
            try:
                r = minimize(obj_exact_func, best_h, method='SLSQP', bounds=bnds,
                           constraints=cons, options={'maxiter': 200, 'ftol': 1e-15})
                hn = repair(r.x, tgt)
                cn = compute_c5(hn, dx)
                if cn < best_c5:
                    best_c5, best_h = cn, hn
            except Exception:
                pass

    # Phase 4: Many small-perturbation restarts
    n_restarts = 300
    for i in range(n_restarts):
        if time.time() - t0 > budget_s - 60:
            break
        sc = 0.003 + 0.015 * np.exp(-i / 30.0)
        hp = best_h + np.random.randn(n) * sc
        hp = repair(hp, tgt)
        try:
            r = minimize(obj_exact_func, hp, method='SLSQP', bounds=bnds,
                        constraints=cons, options={'maxiter': 80, 'ftol': 1e-14})
            ho = repair(r.x, tgt)
            co = compute_c5(ho, dx)
            if co < best_c5:
                best_c5, best_h = co, ho
        except Exception:
            pass

    # Phase 5: Binary local search via random swaps
    if time.time() - t0 < budget_s - 40:
        h_bin = (best_h > 0.5).astype(float)
        excess = np.sum(h_bin) - tgt
        if excess > 0:
            ones_idx = np.where(h_bin == 1.0)[0]
            for idx in ones_idx[:int(excess + 0.5)]:
                h_bin[idx] = 0.0
        elif excess < 0:
            zeros_idx = np.where(h_bin == 0.0)[0]
            for idx in zeros_idx[:int(-excess + 0.5)]:
                h_bin[idx] = 1.0
        h_bin = repair(h_bin, tgt)
        cb = compute_c5(h_bin, dx)
        for _ in range(4000):
            if time.time() - t0 > budget_s - 40:
                break
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i == j or h_bin[i] == h_bin[j]:
                continue
            h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
            ts = compute_c5(h_bin, dx)
            if ts >= cb:
                h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
            else:
                cb = ts
        if cb < best_c5:
            best_c5, best_h = cb, h_bin.copy()
            if time.time() - t0 < budget_s - 10:
                try:
                    r = minimize(obj_exact_func, best_h, method='SLSQP', bounds=bnds,
                               constraints=cons, options={'maxiter': 300, 'ftol': 1e-15})
                    hn = repair(r.x, tgt)
                    cn = compute_c5(hn, dx)
                    if cn < best_c5:
                        best_c5, best_h = cn, hn
                except Exception:
                    pass

    # Phase 6: Final intensive polish
    for _ in range(20):
        if time.time() - t0 > budget_s - 3:
            break
        try:
            r = minimize(obj_exact_func, best_h, method='SLSQP', bounds=bnds,
                        constraints=cons, options={'maxiter': 500, 'ftol': 1e-16})
            hf = repair(r.x, tgt)
            cf = compute_c5(hf, dx)
            if cf < best_c5:
                best_c5, best_h = cf, hf
        except Exception:
            break

    best_h = repair(best_h, tgt)
    best_c5 = compute_c5(best_h, dx)
    return (best_h, best_c5, n)
