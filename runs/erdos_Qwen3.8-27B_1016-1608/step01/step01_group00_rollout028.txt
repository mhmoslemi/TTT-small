    max_val = np.max(corr_full)
    # Active set: correlations within 10% of max
    active = corr_full > max_val - 0.1 * max_val
    # Subgradient is the average gradient over active set
    grad = np.zeros(n)
    ones_h = 1.0 - h

    grad /= np.sum(active)
    # Take step
    h_new = h - alpha * grad
    h_new = repair(h_new, n)
    return h_new, max_val

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not np.any(mask):
            h[:] = target / n
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    if beta > 1e8:
        return m
    return m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta

def smooth_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    if beta > 1e8:
        # Just return subgradient of the max
        k_star = np.argmax(corr)
        k = k_star - (n - 1)
        ones_h = 1.0 - h
        grad = np.zeros(n)
        if k >= 0:
            grad[k:] += ones_h[:n-k]
            grad[:n-k] -= h[k:]
        else:
            kp = -k
            grad[:n-kp] += ones_h[kp:]
            grad[kp:] -= h[:n-kp]
        return grad * dx
    weights = np.exp(beta * (corr - m))
    weights /= np.sum(weights)
    grad = np.zeros(n)
    ones_h = 1.0 - h
    for m_idx in range(2*n - 1):
        w = weights[m_idx]
        if w < 1e-12:
            continue
        k = m_idx - (n - 1)
        if k >= 0:
            grad[k:] += w * ones_h[:n-k]
            grad[:n-k] -= w * h[k:]
        else:
            kp = -k
            grad[:n-kp] += w * ones_h[kp:]
            grad[kp:] -= w * h[:n-kp]
    return grad * dx

def sum_constraint(x):
    return np.sum(x) - len(x) / 2.0

def slsqp_smooth(h0, n, beta, maxiter=500):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint}]
    try:
        res = minimize(smooth_max_obj, h0, jac=lambda x: smooth_max_grad(x, beta),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.fun < 1e10:
            return repair(np.clip(res.x, 0.0, 1.0), n)
    except Exception:
        pass
    return h0

def slsqp_direct(h0, n, maxiter=500):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint}]
    try:
        res = minimize(compute_c5, h0, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.fun < compute_c5(h0) + 1e-14:
            return repair(np.clip(res.x, 0.0, 1.0), n)
    except Exception:
        pass
    return h0

def subgradient_optimize(h0, n, t_max):
    """Run subgradient method with averaging until t_max is reached."""
    h = h0.copy()
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    
    avg_h = np.zeros(n)
    count = 0
    alpha = 0.05
    no_improve = 0
    
    while time.time() < t_max:
        step_start = time.time()
        # Compute subgradient
        h_new = h - alpha * smooth_max_grad(h, 1e8)  # beta=1e8 means exact subgradient
        h_new = repair(h_new, n)
        c = compute_c5(h_new)
        
        if c < best_c5:
            best_h = h_new.copy()
            best_c5 = c
            no_improve = 0
        else:
            no_improve += 1
        
        h = h_new
        avg_h += h
        count += 1
        
        # Adaptive step size
        if no_improve > 200:
            alpha *= 0.95
            no_improve = 0
        elif no_improve < 20 and alpha < 0.1:
            alpha *= 1.02
        
    # Also check the average
    avg_h /= max(count, 1)
    avg_h = repair(avg_h, n)
    c_avg = compute_c5(avg_h)
    if c_avg < best_c5:
        best_h, best_c5 = avg_h, c_avg
    
    return best_h, best_c5

def simulated_annealing(h0, n, t_max):
    """SA with informed perturbations."""
    h = h0.copy()
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    curr_c5 = best_c5
    
    temp = 0.02
    min_temp = 0.0005
    cool_rate = 0.9997

    while time.time() < t_max and temp > min_temp:

            if time.time() >= t_max:

            # Perturbation: mix of random and gradient-informed

            if r < 0.3:

                delta = delta - np.mean(delta)  # keep sum roughly 0

            c_new = compute_c5(h_new)

                h = h_new
                curr_c5 = c_new
                if c_new < best_c5:
                    best_h = h.copy()
                    best_c5 = c_new
            else:
                h = h  # rejected, keep current
        
        temp *= cool_rate
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(1000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not np.any(mask):
            h[:] = target / n
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def sum_constraint(x):
    return np.sum(x) - len(x) / 2.0

def smooth_max_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    if beta > 1e6:
        return m
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -500, 500)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def exact_subgrad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    k_star = np.argmax(corr)
    k = k_star - (n - 1)
    ones_h = 1.0 - h
    grad = np.zeros(n)
    if k >= 0:
        grad[k:] += ones_h[:n - k]
        grad[:n - k] -= h[k:]
    else:
        kp = -k
        grad[:n - kp] += ones_h[kp:]
        grad[kp:] -= h[:n - kp]
    return grad * dx

def smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    if beta > 1e6:
        return exact_subgrad(h)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -500, 500)
    weights = np.exp(shifted)
    sw = np.sum(weights)
    if sw < 1e-300:
        return exact_subgrad(h)
    weights /= sw
    grad = np.zeros(n)
    ones_h = 1.0 - h
    for m_idx in range(2 * n - 1):
        w = weights[m_idx]
        if w < 1e-10:
            continue
        k = m_idx - (n - 1)
        if k >= 0:
            grad[k:] += w * ones_h[:n - k]
            grad[:n - k] -= w * h[k:]
        else:
            kp = -k
            grad[:n - kp] += w * ones_h[kp:]
            grad[kp:] -= h[:n - kp]
    return grad * dx

def _sgrad(x):
    return smooth_grad(x, _gbeta)

def _dobj(x):
    return compute_c5(x)

def slsqp_smooth(h0, n, beta, maxiter=500):
    global _gbeta
    _gbeta = beta
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint}]
    try:
        res = minimize(_sobj, h0, jac=_sgrad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        if res.fun < 1e10:
            return repair(np.clip(res.x, 0.0, 1.0), n)
    except Exception:
        pass
    return h0

def slsqp_direct(h0, n, maxiter=500):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint}]

    except Exception:

            best_c5 = c

    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    temp = 0.015

                if c_new < best_c5:
                    best_h = h.copy()
                    best_c5 = c_new

    return best_h, best_c5
