```python
def project(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    lo, hi = h.min() - 1, h.max() + 1
    for _ in range(60):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0, 1)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2, 0, 1)
```

```python
def compute_grad(h, p, n):
    """Compute gradient of smoothed max correlation.
    h: array of length n
    p: softmax weights, array of length 2n-1
    Returns: gradient array of length n
    """
    grad = np.zeros(n)
    n_corr = 2 * n - 1
    for k_idx in range(n_corr):
        if p[k_idx] < 1e-12:
            continue
        lag = k_idx - (n - 1)
        w = p[k_idx]
        if lag >= 0:
            valid_i = np.arange(n - lag)
            grad[valid_i] += w * (1.0 - h[valid_i + lag])
            grad[valid_i + lag] += w * (-h[valid_i])
        else:
            lag_neg = -lag
            valid_i = np.arange(lag_neg, n)
            grad[valid_i] += w * (1.0 - h[valid_i - lag_neg])
            grad[valid_i - lag_neg] += w * (-h[valid_i])
    return grad
```

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def gd_grad(h, n, eps):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = corr.max()
    p = np.exp((corr - c_max) / eps)
    ps = p.sum()
    if ps < 1e-30:
        p = np.zeros_like(corr)
        p[np.argmax(corr)] = 1.0
    else:
        p /= ps
    g = np.zeros(n)
    g += p[n - 1] * (1.0 - 2.0 * h)
    for k in range(1, n):
        wp = p[n - 1 + k]
        wn = p[n - 1 - k]
        if wp > 1e-14:
            g[k:] += wp * (1.0 - h[:n - k])
            g[:n - k] += wp * (-h[k:])
        if wn > 1e-14:
            g[:n - k] += wn * (1.0 - h[k:])
            g[k:] += wn * (-h[:n - k])
    return g

def gd_run(h0, n, eps, t_start, t_end, lr0=1.0):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr0
    stall = 0
    while time.time() < t_end:
        g = gd_grad(h, n, eps)
        h_new = project(h - lr * g, n)
        c5 = compute_c5(h_new)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h_new.copy()
            stall = 0
            h = h_new
        else:
            h = h_new  # still take the step
            stall += 1
            if stall > 500:
                lr *= 0.1
                stall = 0
        lr = max(lr * 0.9995, 1e-6)
    return best_h, best_c5
```

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def sp_grad(h, n, eps):
    c = np.correlate(h, 1.0 - h, mode='full')
    cm = c.max()
    p = np.exp((c - cm) / eps)
    s = p.sum()
    if s < 1e-30:
        p = np.zeros_like(c); p[np.argmax(c)] = 1.0
    else:
        p /= s
    g = p[n-1] * (1.0 - 2.0*h)
    for k in range(1, n):
        wp = p[n-1+k]; wn = p[n-1-k]
        if wp > 1e-13:
            g[k:] += wp*(1.0-h[:n-k]); g[:n-k] += -wp*h[k:]
        if wn > 1e-13:
            g[:n-k] += wn*(1.0-h[k:]); g[k:] += -wn*h[:n-k]
    return g

def gd_phase(h, n, eps, t_end, lr=0.3):
    bh = h.copy(); bc = compute_c5(h)
    while time.time() < t_end:
        g = sp_grad(h, n, eps)
        if np.all(np.abs(g) < 1e-12): break
        h2 = project(h - lr*g, n)
        c2 = compute_c5(h2)
        if c2 < bc - 1e-14:
            bc = c2; bh = h2.copy(); h = h2
            lr = min(lr*1.01, 2.0)
        else:
            lr *= 0.85
            if lr < 1e-9: break
    return bh, bc

def sa_phase(h, n, t_end, T0=0.002):
    h = h.copy(); c5 = compute_c5(h)
    bh = h.copy(); bc5 = c5; T = T0
    while time.time() < t_end:
        i = np.random.randint(n); j = np.random.randint(n)
        if i == j: continue
        e = np.random.uniform(0.001, 0.02) * (1 if np.random.random()<0.5 else -1)
        oi, oj = h[i], h[j]
        h[i] += e; h[j] -= e
        if 0<=h[i]<=1 and 0<=h[j]<=1:
            nc = compute_c5(h)
            if nc-c5 < 0 or np.random.random() < np.exp(-max(nc-c5,0)/T):
                c5 = nc
                if c5 < bc5: bc5 = c5; bh = h.copy()
            else: h[i],h[j] = oi,oj
        else: h[i],h[j] = oi,oj
        T = max(T*0.99995, 1e-6)
    return bh, bc5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time(); dl = t0 + budget_s - 15
    try: h0 = np.array(initial_h_values, dtype=float).copy()
    except: h0 = None

    h = project(h0, n)
    bh = h.copy(); bc5 = compute_c5(h)
    
    # Multi-scale GD
    eps_list = [0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    gd_time_total = (dl - t0) * 0.55
    for i, eps in enumerate(eps_list):
        te = t0 + gd_time_total * (i+1) / len(eps_list)
        h, c5 = gd_phase(h, n, eps, min(te, dl-250), lr=0.3)
        if c5 < bc5: bc5 = c5; bh = h
    
    # SA
    if time.time() < dl - 80:
        h, c5 = sa_phase(bh, n, dl-80, T0=0.001)
        if c5 < bc5: bc5 = c5; bh = h
    
    # Final GD
    if time.time() < dl - 5:
        h, c5 = gd_phase(bh, n, 1e-5, dl-2, lr=0.1)
        if c5 < bc5: bc5 = c5; bh = h
    
    fh = project(bh, n)
    fc = compute_c5(fh)
    return (fh.tolist(), float(fc), n)
```
