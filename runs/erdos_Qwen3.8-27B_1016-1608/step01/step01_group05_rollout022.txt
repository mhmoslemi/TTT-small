**Idea: Use the fact that for the max to be small, ALL correlations must be small**

This is a min-max problem. We can use a "soft max" approximation:
softmax_β(corr) = (1/β) * log(sum(exp(β * corr_k)))

And minimize this smooth function. For large β, it approaches the true max. This gives a smoother optimization landscape.

    x_new = np.linspace(0.0, 2.0, n_new)

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return (1.0/beta) * np.log(np.sum(np.exp(beta * corr - beta * np.max(corr)))) + np.max(corr)

                    if v < best:
                        best = v
                        improved = True
                    else:
                        h[i] -= d
                        h[j] += d
        if not improved:
            delta *= 0.55
            if delta < 0.001:
                break
    return h, best

def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if bin(i).count('1') % 2 == 0 else 0.0
    return h

def make_fourier(n, rng):
    x = np.linspace(0, 2, n)
    h = 0.5 * np.ones(n)
    for f in range(1, 8):
        h += rng.uniform(-0.12, 0.12) * np.sin(f * np.pi * x + rng.uniform(0, 2 * np.pi))
    return h

def cobyla_opt(h0, n, maxiters=2000):
    h0 = project_to_set(h0.copy(), n / 2.0)
    target = n / 2.0
    def _cobj(x):
        hp = project_to_set(x, target)
        return compute_c5(hp, n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_cobj, h0, method='COBYLA', bounds=bounds,
                       options={'maxiter': maxiters, 'rhobeg': 0.3, 'catol': 1e-11})
        h_out = project_to_set(res.x.copy(), target)
        return h_out, compute_c5(h_out, n)
    except Exception:
        return h0, compute_c5(h0, n)

def run(seed=42, budget_s=1100, **kwargs):
    global _con_target
    t0 = time.time()
    budget = float(budget_s)
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = np.inf
    best_n = 90

    def tl():
        return max(0.0, budget - (time.time() - t0))

    n = 90
    target = n / 2.0

    # Initialize
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        if len(h0) != n:
            h0 = upsample(h0, n)
    except Exception:
        h0 = make_thue_morse(n)
    h0 = project_to_set(h0, target)
    best_h = project_to_set(h0.copy(), target)
    best_c5 = compute_c5(best_h, n)

    # Phase 1: SLSQP on initial
    if tl() > 150:
        h_s, c_s = slsqp_opt(best_h, n, 1200)
        if c_s < best_c5:
            best_h, best_c5 = h_s, c_s

    # Phase 2: COBYLA
    if tl() > 120:
        h_c, c_c = cobyla_opt(best_h, n, 3000)
        if c_c < best_c5:
            best_h, best_c5 = h_c, c_c

    # Phase 3: Paired CD
    if tl() > 100:
        h_cd, c_cd = paired_cd(best_h, n, rng, 400)
        if c_cd < best_c5:
            best_h, best_c5 = h_cd, c_cd

    # Phase 4: Diverse restarts
    while tl() > 80:
        h_r = None
        mode = int((time.time() - t0) % 9)
        if mode == 0:
            h_r = best_h + rng.normal(0, 0.03, n)
        elif mode == 1:
            h_r = best_h.copy()
            idx = rng.choice(n, size=6, replace=False)
            h_r[idx] = 1.0 - h_r[idx]
        elif mode == 2:
            h_r = np.roll(best_h, rng.randint(1, n))
        elif mode == 3:
            h_r = make_thue_morse(n) + rng.normal(0, 0.02, n)
        elif mode == 4:
            h_r = make_fourier(n, rng)
        elif mode == 5:
            h_r = np.zeros(n)
            h_r[rng.choice(n, size=45, replace=False)] = 1.0
            h_r += rng.normal(0, 0.04, n)
        elif mode == 6:
            h_r = best_h * 1.1 + rng.normal(0, 0.02, n)
        elif mode == 7:
            h_r = np.convolve(best_h, np.ones(5) / 5, mode='same')
            h_r += rng.normal(0, 0.02, n)
        else:
            h_r = best_h.copy()
            h_r[rng.choice(n, size=10, replace=False)] = rng.uniform(0.1, 0.9, 10)

        h_r = project_to_set(h_r, target)
        h_s, c_s = slsqp_opt(h_r, n, 500)
        if c_s < best_c5:
            best_h, best_c5 = h_s, c_s
        if tl() > 20:
            h_cd, c_cd = paired_cd(best_h, n, rng, 100)
            if c_cd < best_c5:
                best_h, best_c5 = h_cd, c_cd

    # Phase 5: Final polish at n=90
    if tl() > 20:
        h_cd, c_cd = paired_cd(best_h, n, rng, 500)
        if c_cd < best_c5:
            best_h, best_c5 = h_cd, c_cd
    if tl() > 15:
        h_s, c_s = slsqp_opt(best_h, n, 800)
        if c_s < best_c5:
            best_h, best_c5 = h_s, c_s

    # Phase 6: Try higher resolution
    if tl() > 40:
        n2 = 180
        h_up = project_to_set(upsample(best_h, n2), n2 / 2.0)
        c_up = compute_c5(h_up, n2)
        if c_up < best_c5:
            best_h, best_c5, best_n = h_up, c_up, n2
        if tl() > 25:
            h_s, c_s = slsqp_opt(best_h, best_n, 600)
            if c_s < best_c5:
                best_h, best_c5 = h_s, c_s
        if tl() > 15:
            h_cd, c_cd = paired_cd(best_h, best_n, rng, 200)
            if c_cd < best_c5:
                best_h, best_c5 = h_cd, c_cd
        if tl() > 15:
            n3 = 360
            h_up3 = project_to_set(upsample(best_h, n3), n3 / 2.0)
            c_up3 = compute_c5(h_up3, n3)
            if c_up3 < best_c5:
                best_h, best_c5, best_n = h_up3, c_up3, n3
            if tl() > 10:
                h_s, c_s = slsqp_opt(best_h, best_n, 400)
                if c_s < best_c5:
                    best_h, best_c5 = h_s, c_s

    best_h = project_to_set(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h, best_n)
    return best_h, best_c5, best_n
