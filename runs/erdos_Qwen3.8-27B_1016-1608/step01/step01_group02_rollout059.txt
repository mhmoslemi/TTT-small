import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def proj_h(y, target):
    n = len(y)
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_val(h, dx, tau):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    return (m + tau * np.log(np.sum(np.exp((corr - m) / tau)))) * dx

def smooth_grad(h, dx, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    w = np.exp((corr - m) / tau)
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        s = k - (n - 1)
        i_lo = max(0, -s)
        i_hi = min(n, n - s)
        idx = np.arange(i_lo, i_hi)
        grad[idx] += w[k] * (1.0 - h[idx + s])
        grad[idx + s] += w[k] * (-h[idx])
    return grad * dx

def pg_optimize(h0, target, dx, tau, max_iter=5000):
    h = h0.copy()
    f_old = smooth_val(h, dx, tau)
    best_f = f_old
    best_h = h.copy()
    alpha = 1.0 / (tau + 1.0)
    stall = 0
    for it in range(max_iter):
        g = smooth_grad(h, dx, tau)
        improved = False
        a = alpha
        for _ in range(25):
            h_try = proj_h(h - a * g, target)
            f_try = smooth_val(h_try, dx, tau)
            if f_try < f_old - 1e-16:
                h = h_try
                f_old = f_try
                if f_try < best_f:

                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
                alpha = min(a * 1.5, 5.0)
                improved = True
                break
            a *= 0.5
        if not improved:
            alpha *= 0.25
            stall += 1
            if alpha < 1e-12 or stall > 200:
                break
    return best_h

def slsqp_smooth(h0, target, dx, tau, maxiter=300):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = ({'type': 'eq',
             'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)},)
    try:
        res = minimize(smooth_val, h0, args=(dx, tau),
                      method='SLSQP',
                      jac=lambda x: smooth_grad(x, dx, tau),
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        return proj_h(res.x, target)
    except Exception:
        return None

def slsqp_exact(h0, target, dx, maxiter=300):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = ({'type': 'eq',
             'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)},)
    try:
        res = minimize(compute_c5, h0, args=(dx,),
                      method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        return proj_h(res.x, target)
    except Exception:
        return None
