from scipy.optimize import minimize
import time

_S = {}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c5_grad(h):
    """Compute gradient of C5 (at the max lag) wrt h."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    k_max = np.argmax(c)
    # k in original indexing: k = k_max - (n-1)
    k = k_max - (n - 1)
    # c_k = sum_{valid n} h[n+k] * (1-h[n])
    # For k >= 0: n from 0 to n-1-k
    # For k < 0: n from -k to n-1
    grad = np.zeros(n)
    if k >= 0:
        for n_idx in range(n - k):
            # h[n_idx+k] * (1-h[n_idx])
            grad[n_idx + k] += (1.0 - h[n_idx])
            grad[n_idx] -= h[n_idx + k]
    else:
        for n_idx in range(-k, n):
            # h[n_idx+k] * (1-h[n_idx])
            grad[n_idx + k] += (1.0 - h[n_idx])
            grad[n_idx] -= h[n_idx + k]
    return grad * dx

def repair(h, n):
    h = np.array(h, dtype=float).copy()
    for _ in range(100):
        np.clip(h, 0.0, 1.0, out=h)
        d = n / 2.0 - np.sum(h)

            break
        h += d / n

        return float(m)
