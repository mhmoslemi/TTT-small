def compute_c5(h, n):

    corr = np.correlate(h, 1.0 - h, mode="full")

    lo, hi = -2.0, 2.0
    for _ in range(60):

        if np.sum(hp) > target:
            lo = mid
        else:

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_set(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        hp = np.clip(h - mid, 0.0, 1.0)
        if float(np.sum(hp)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def make_smooth_init(n, rng, n_tones=3):
    x = np.linspace(0.0, 2.0, n)
    h = np.full(n, 0.5)
    for _ in range(n_tones):
        h += rng.uniform(0.15, 0.45) * np.sin(
            rng.randint(1, 9) * np.pi * x + rng.uniform(0.0, 2.0 * np.pi))
    return project_to_set(h, n / 2.0)

def coord_desc(h, n, rng, n_sweeps=200, n_cands=21):
    target = n / 2.0
    h = h.copy()
    best_c5 = compute_c5(h, n)
    cands_global = np.linspace(0.0, 1.0, n_cands)
    for sweep in range(n_sweeps):
        improved = False
        order = np.arange(n) if sweep == 0 else rng.permutation(n)
        for i in order:
            old = h[i]
            best_val = best_c5
            best_new = old
            for new in cands_global:
                if abs(float(new) - old) < 1e-10:
                    continue
                h[i] = float(new)
                h = project_to_set(h, target)
                val = compute_c5(h, n)
                if val < best_val - 1e-13:
                    best_val = val
                    best_new = float(new)
            if best_new != old:
                h[i] = best_new
                h = project_to_set(h, target)
                best_c5 = best_val
                improved = True
        if not improved:
            break
    return h, best_c5

def coord_desc_fine(h, n, rng, n_sweeps=200):
    target = n / 2.0
    h = h.copy()
    best_c5 = compute_c5(h, n)
    for sweep in range(n_sweeps):
        improved = False
        order = np.arange(n) if sweep == 0 else rng.permutation(n)
        frac = 1.0 - min(1.0, float(sweep) / max(1, n_sweeps))
        half = 0.03 + 0.12 * frac
        order2 = rng.permutation(n) if sweep > 2 else np.arange(n)
        for i in order2:
            old = h[i]
            lo_c = max(0.0, old - half)
            hi_c = min(1.0, old + half)
            cands = np.unique(np.concatenate([
                [0.0, 1.0],
                np.linspace(0.0, 1.0, 7),
                np.linspace(lo_c, hi_c, 13),
            ]))
            best_val = best_c5
            best_new = old
            for new in cands:
                if abs(float(new) - old) < 1e-10:
                    continue
                h[i] = float(new)
                h = project_to_set(h, target)
                val = compute_c5(h, n)
                if val < best_val - 1e-13:
                    best_val = val
                    best_new = float(new)
            if best_new != old:
                h[i] = best_new
                h = project_to_set(h, target)
                best_c5 = best_val
                improved = True
        if not improved and sweep > 3:
            break
    return h, best_c5

def pair_moves(h, n, rng, n_pairs=800):
    h = h.copy()
    best_c5 = compute_c5(h, n)
    deltas = [0.02, 0.05, 0.1, 0.15, 0.2]
    improved_any = False
    for _ in range(n_pairs):
        i, j = rng.randint(0, n, 2)
        if i == j:
            continue
        for d in deltas:
            if h[i] >= d and h[j] <= 1.0 - d:
                h[i] -= d
                h[j] += d
                val = compute_c5(h, n)
                if val < best_c5 - 1e-13:
                    best_c5 = val
                    improved_any = True
                    break
                h[i] += d
                h[j] -= d
            if h[j] >= d and h[i] <= 1.0 - d:
                h[j] -= d
                h[i] += d
                val = compute_c5(h, n)
                if val < best_c5 - 1e-13:
                    best_c5 = val
                    improved_any = True
                    break
                h[j] += d
                h[i] -= d
    return h, best_c5, improved_any

def full_local_search(h, n, rng, time_limit_s=60.0):
    target = n / 2.0
    h = project_to_set(h.copy(), target)
    best_c5 = compute_c5(h, n)
    t0 = time.time()
    while time.time() - t0 < time_limit_s:
        h, c5 = coord_desc(h, n, rng, n_sweeps=80, n_cands=21)
        if c5 < best_c5:
            best_c5 = c5
        h, c5 = coord_desc_fine(h, n, rng, n_sweeps=100)
        if c5 < best_c5:
            best_c5 = c5
        h, c5, imp = pair_moves(h, n, rng, n_pairs=400)
        if c5 < best_c5:
            best_c5 = c5
        if not imp:
            h2, c52 = coord_desc_fine(h, n, rng, n_sweeps=60)
            if c52 < best_c5:
                best_c5 = c52
                h = h2
            else:
                break
    return h, best_c5

    start = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return max(0.0, budget_s - (time.time() - start))

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()

    # Phase 1: Deep local search from initial
    if tleft() > 120:
        h, c5 = full_local_search(h0, n, rng, time_limit_s=min(80.0, tleft() * 0.08))
        if c5 < best_c5:
            best_h, best_c5 = h.copy(), c5

    # Phase 2: Multiple diverse restarts
    n_restart = 0
    stagnation = 0
    while tleft() > 55:
        strategy = n_restart % 5
        if strategy == 0:
            scale = 0.08 + 0.08 * (n_restart % 3)
            h_r = best_h + rng.normal(0, scale, n)
        elif strategy == 1:
            h_r = best_h.copy()
            k = rng.randint(5, n // 3)
            idx = rng.choice(n, size=k, replace=False)
            h_r[idx] = 1.0 - h_r[idx]
        elif strategy == 2:
            h_r = make_smooth_init(n, rng, n_tones=rng.randint(2, 6))
        elif strategy == 3:
            h_r = np.zeros(n)
            h_r[rng.choice(n, n // 2, replace=False)] = 1.0
            h_r += rng.normal(0, 0.06, n)
        else:
            # Mirror + perturb
            h_r = best_h[::-1].copy() + rng.normal(0, 0.05, n)

        h_r = project_to_set(h_r, target)
        alloc = min(50.0, tleft() * 0.04)
        h, c5 = full_local_search(h_r, n, rng, time_limit_s=alloc)
        if c5 < best_c5 - 1e-5:
            best_h, best_c5 = h.copy(), c5
            stagnation = 0
        else:
            stagnation += 1
        n_restart += 1

    # Phase 3: Intensive final polish on best
    if tleft() > 20:
        h, c5 = full_local_search(best_h, n, rng, time_limit_s=min(40.0, tleft() * 0.4))

    best_h = project_to_set(best_h, target)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
