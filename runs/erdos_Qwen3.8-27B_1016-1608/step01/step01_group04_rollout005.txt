import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.array(h, dtype=float).copy()
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        d = n / 2.0 - np.sum(h)
        if abs(d) < 1e-14:
            break
        h += d / n
    return np.clip(h, 0.0, 1.0)

def binarize_keep(h, n):
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:n // 2]] = 1.0
    return b

def sa_binary(h, n, T_init, T_final, n_iter, t_limit):
    h = h.copy()
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    cur = compute_c5(h)
    best = cur
    best_h = h.copy()
    for it in range(n_iter):
        if time.time() > t_limit:
            break
        T = T_init * (T_final / T_init) ** (it / max(1, n_iter - 1))
        pi = np.random.randint(len(ones))
        pj = np.random.randint(len(zeros))
        i = ones[pi]
        j = zeros[pj]
        h[i] = 0.0
        h[j] = 1.0
        new = compute_c5(h)
        if new < cur or np.random.rand() < np.exp(-(new - cur) / max(T, 1e-15)):
            cur = new
            if new < best:
                best = new
                best_h = h.copy()
            ones[pi] = j
            zeros[pj] = i
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best

def greedy_binary(h, n, max_sweeps=50):
    h = h.copy()
    cur = compute_c5(h)
    for _ in range(max_sweeps):
        best_delta = 0.0
        best_pair = None
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                new = compute_c5(h)
                h[i] = 1.0
                h[j] = 0.0
                d = new - cur
                if d < best_delta - 1e-15:
                    best_delta = d
                    best_pair = (i, j)
        if best_pair is None:
            break
        h[best_pair[0]] = 0.0
        h[best_pair[1]] = 1.0
        cur += best_delta
    return h, cur

def slsqp_polish(h, n, maxiter=800, ftol=1e-15):
    n_local = n
    def obj(x):
        return compute_c5(x)
    def eqf(x):
        return float(np.sum(x) - n_local / 2.0)
    def eqj(x):
        return np.ones(n_local)
    bds = [(0.0, 1.0)] * n_local
    cs = [{'type': 'eq', 'fun': eqf, 'jac': eqj}]
    try:
        r = minimize(obj, h, method='SLSQP', bounds=bds, constraints=cs,
                     options={'maxiter': maxiter, 'ftol': ftol})
        return repair(r.x, n_local)
    except Exception:
        return repair(h, n_local)
