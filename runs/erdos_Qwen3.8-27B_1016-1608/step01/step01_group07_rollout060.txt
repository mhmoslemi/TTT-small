Actually, a much simpler approach: since g_k is linear in h[m] for fixed other coordinates, the max of linear functions is convex, and the minimum of a convex piecewise-linear function on [0,1] is at a vertex (0, 1, or a breakpoint). The breakpoints are where two g_k lines cross.

For coordinate m, g_k(h[m]) = a_k * h[m] + b_k where:
a_k = (1-h[m-k])*dx - h[m+k]*dx (with validity)
b_k = g_k(h) - a_k * h[m] (current value minus the m-dependent part)

The max of lines min over h[m] ∈ [0,1] can be found by:
1. Compute all a_k, b_k
2. The upper envelope is the max of all lines
3. Find the minimum of the upper envelope

For a convex piecewise-linear function, the minimum is where the slope changes from negative to positive. We can find this by:
- Sort lines by slope
- The upper envelope transitions from the highest intercept (lowest slope) to the lowest intercept at x=1 (highest slope)
- The minimum is at the transition point

Actually, for 1D, a simpler approach:
- The function f(x) = max_k(a_k*x + b_k) is convex
- f'(x) (subgradient) is positive for x > x*, negative for x < x*
- x* is where the steepest "active" line has slope 0

Even simpler: just evaluate f at 0, 1, and at all pairwise crossings in [0,1]. There are O(K²) crossings but most won't be on the upper envelope.

For practical purposes, ternary search with 30-40 iterations is fine and O(40*K) per coordinate.

from scipy.optimize import minimize
import time

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _fix_sum(h, n):
    target = 0.5 * n
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        d = target - np.sum(h)
        if abs(d) < 1e-15:
            break
        if d > 0:
            mask = h < 0.9999999
        else:
            mask = h > 1e-7
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
    return np.clip(h, 0.0, 1.0)

def _interp(h, n_new):
    n_old = len(h)
    x_old = np.arange(n_old) * (2.0 / n_old)
    x_new = np.arange(n_new) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    return _fix_sum(h_new, n_new)

def _coord_descent(h, n, max_sweeps=50, t0=0, budget=1100, verbose=False):
    h = h.copy()
    dx = 2.0 / n
    best_c = _c5(h)
    for sweep in range(max_sweeps):
        if time.time() - t0 > budget - 20:
            break
        old_c = best_c
        for i in range(n):
            if time.time() - t0 > budget - 20:
                break
            val = h[i]
            lo, hi = 0.0, 1.0
            for _ in range(30):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                h[i] = m1
                c1 = _c5(h)
                h[i] = m2
                c2 = _c5(h)
                if c1 < c2:
                    hi = m2
                else:
                    lo = m1
            h[i] = (lo + hi) / 2.0
            c_new = _c5(h)
            if c_new < best_c:
                best_c = c_new
            else:
                h[i] = val
        if best_c >= old_c - 1e-14:
            break
    h = _fix_sum(h, n)
    return h, _c5(h)

def _slsqp_opt(h0, n, alpha, maxiter, t0, budget):
    if time.time() - t0 > budget - 15:
        return None
    dx = 2.0 / n
    
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = np.max(c)
        return m + np.log(np.sum(np.exp(np.clip(c - m, -50, 50)))) / alpha
    
    def obj_grad(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = np.max(c)
        e = np.exp(np.clip(c - m, -50, 50))
        w = e / np.sum(e) / alpha
        # d(c)/dx is anti-symmetric in a sense
        # dc[i]/dx[j] = -dx*(h[j+i-(n-1)] + ... ) 
        # This is complex; let SLSQP do numerical differentiation
        return None
    
    bnds = [(0.0, 1.0)] * n
    
    def icf(x):
        return np.sum(x) - 0.5 * n
    
    def icj(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': icf, 'jac': icj}]
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None:
            return _fix_sum(np.clip(res.x, 0, 1), n)
    except Exception:
        pass
    return None

    else:

                    best_c = c2

        if ci < best_c:

            best_c = ci
