def smooth_max_objective(h, dx, tau=500.0):
    """Compute smooth max of correlations using log-sum-exp."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr / dx  # scale
    # log-sum-exp for numerical stability
    max_v = np.max(vals)
    log_sum = max_v + np.log(np.sum(np.exp(vals - max_v)))
    return log_sum / tau

def smooth_max_gradient(h, dx, tau=500.0):
    """Compute gradient of smooth max via softmax weights."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr / dx
    max_v = np.max(vals)
    exp_vals = np.exp(vals - max_v)
    weights = exp_vals / exp_vals.sum()
    
    n = len(h)
    grad = np.zeros(n)
    for m_idx, w in enumerate(weights):
        if w < 1e-10:
            continue
        m = m_idx - (n - 1)  # actual lag
        for p in range(n):
            g = 1.0
            if p - m >= 0 and p - m < n:
                g -= h[p - m]
            if p + m >= 0 and p + m < n:
                g -= h[p + m]
            grad[p] += w * g
    
    return grad / tau

def _repair(h, target):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)

def _smooth_grad(h, dx, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    g = corr * dx
    vmax = g.max()
    exp_g = np.exp(g - vmax)

import numpy as np
from scipy.optimize import minimize
import time


def _repair(h, target):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(120):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        c = int(mask.sum())
        if c == 0:
            break
        h[mask] += d / c
        h = np.clip(h, 0.0, 1.0)
    return h


def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full"))) * dx


def _lgrad(h, m):
    n = len(h)
    gc = np.zeros(n)
    if m == 0:
        gc[:] = 1.0 - 2.0 * h
    elif m > 0:
        gc[m:] = 1.0
        gc[m:] -= h[:n - m]
        gc[:n - m] -= h[m:]
    else:
        a = -m
        gc[:n - a] = 1.0
        gc[:n - a] -= h[a:]
        gc[a:] -= h[:n - a]
    return gc


def _smooth_grad(h, dx, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    g = corr * dx
    vmax = g.max()
    exp_g = np.exp(g - vmax)
    w = exp_g / exp_g.sum()
    obj = (vmax + np.log(exp_g.sum())) / tau
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        m = k - (n - 1)
        grad += wk * _lgrad(h, m)
    grad *= dx
    return obj, grad


def _optimize(h_init, n, dx, target, t_end, rng):
    h = _repair(h_init.copy(), target)
    best_c5 = _c5(h, dx)
    best_h = h.copy()
    v = np.zeros(n)
    beta = 0.85
    lr0 = 0.012
    no_improve = 0
    it = 0
    while time.time() < t_end:
        frac = min(1.0, it / 5000.0)
        tau = 50.0 * (8000.0 / 50.0) ** frac
        lr = lr0 * (1.0 - 0.92 * frac)
        if lr < 0.0005:
            lr = 0.0005
        _, grad = _smooth_grad(h, dx, tau)
        v = beta * v + (1.0 - beta) * grad
        h_new = h - lr * v
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = _repair(h_new, target)
        c5_new = _c5(h_new, dx)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        h = h_new
        it += 1
        if no_improve > 3000 and it > 2000:
            h = best_h + rng.randn(n) * 0.02
            h = _repair(h, target)
            v = np.zeros(n)
            no_improve = 0
    return best_h, best_c5


def _optimize_high_tau(h_init, n, dx, target, t_end, rng):
    h = _repair(h_init.copy(), target)
    best_c5 = _c5(h, dx)
    best_h = h.copy()
    v = np.zeros(n)
    beta = 0.7
    it = 0
    while time.time() < t_end:
        lr = 0.003 * (1.0 - 0.7 * min(1.0, it / 8000.0))
        if lr < 0.0003:
            lr = 0.0003
        _, grad = _smooth_grad(h, dx, 10000.0)
        v = beta * v + (1.0 - beta) * grad
        h_new = h - lr * v
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = _repair(h_new, target)
        c5_new = _c5(h_new, dx)
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        it += 1
        if it % 5000 == 0 and it > 2000:
            h = best_h + rng.randn(n) * 0.008
            h = _repair(h, target)
            v = np.zeros(n)
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    try:

    t_end1 = t0 + budget_s * 0.30

    # Phase 4: Final high-tau polish (10% budget)
    t_end4 = t0 + budget_s - 3.0
    bh4, bc4 = _optimize_high_tau(best_h, best_n, 2.0 / best_n, best_n / 2.0, t_end4, rng)

        best_c5, best_h = bc4, bh4

    # Final SLSQP polish

            n_f = best_n

                return _c5(np.clip(x, 0.0, 1.0), dx_f)

                return x.sum() - tgt_f

            res = minimize(_obj, best_h, method='SLSQP', bounds=bnds,
                           constraints=[{'type': 'eq', 'fun': _sum_eq}],
                           options={'maxiter': 200, 'ftol': 1e-15})
            ho = _repair(res.x, tgt_f)
            co = _c5(ho, dx_f)
            if co < best_c5:
                best_c5, best_h = co, ho
        except Exception:
            pass

    best_h = _repair(best_h, best_n / 2.0)
    best_c5 = _c5(best_h, 2.0 / best_n)

    return (best_h, best_c5, best_n)
