            c = np.correlate(h, omh, mode='full')
            E = np.zeros(2 * n - 1)
            for m in range(2 * n - 1):
                lag = m - (n - 1)
                e = 0.0
                if 0 <= i - lag < n:
                    e += (1.0 - h[i - lag])
                if 0 <= i + lag < n:
                    e -= h[i + lag]
                E[m] = e
            D = c - h[i] * E
            lo, hi = 0.0, 1.0
            for _ in range(50):
                mid = (lo + hi) / 2.0
                vals = D + mid * E
                mx = np.max(vals)
                active = E[vals >= mx - 1e-14]
                if active.size == 0:
                    break
                if np.min(active) > 1e-13:
                    hi = mid
                elif np.max(active) < -1e-13:
                    lo = mid
                else:
                    break
            v_star = (lo + hi) / 2.0
            f0 = np.max(D)
            f1 = np.max(D + E)
            fm = np.max(D + v_star * E)
            if f0 <= f1 and f0 <= fm:
                v_best = 0.0
            elif f1 <= f0 and f1 <= fm:
                v_best = 1.0
            else:
                v_best = v_star
            if abs(v_best - h[i]) > tol:
                h[i] = v_best
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h

def binary_local_search(h, n, rng, max_iters=20000):
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:n // 2]] = 1.0
    best_s = compute_score(hb)
    for _ in range(max_iters):
        ones = np.where(hb == 1.0)[0]
        zeros = np.where(hb == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        hb[i], hb[j] = 0.0, 1.0
        s = compute_score(hb)
        if s < best_s:
            best_s = s
        else:
            hb[i], hb[j] = 1.0, 0.0
    return hb, best_s

def run(seed=42, budget_s=1100, **kwargs):
    global _g_n
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        n_init = 50
        h_init = np.zeros(n_init)
        h_init[:n_init // 2] = 1.0

    n = len(h_init)
    _g_n = n
    best_h = project_feasible(h_init, n)
    best_score = compute_score(best_h)

    # Phase 1: SLSQP warm start
    if elapsed() < budget_s * 0.08:
        bounds = [(0.0, 1.0)] * n
        for r in range(3):
            if elapsed() > budget_s * 0.1:
                break
            h_start = best_h.copy() if r == 0 else project_feasible(best_h + rng.randn(n) * 0.04, n)
            try:
                res = minimize(soft_max_score, h_start, args=(400.0,), method='SLSQP',
                               bounds=bounds, constraints=[{'type': 'eq', 'fun': _sum_con}],
                               options={'maxiter': 200, 'ftol': 1e-12})
                hc = project_feasible(res['x'], n)
                s = compute_score(hc)
                if s < best_score:
                    best_score, best_h = s, hc
            except Exception:
                pass

    # Phase 2: Heavy coordinate descent
    if elapsed() < budget_s * 0.55:
        for rnd in range(40):
            if elapsed() > budget_s * 0.55:
                break
            h_cd = coordinate_descent(best_h, n, max_sweeps=60, tol=1e-11)
            h_cd = project_feasible(h_cd, n)
            s = compute_score(h_cd)
            if s < best_score - 1e-14:
                best_score, best_h = s, h_cd
            if rnd > 4 and rnd % 5 == 0:
                h_p = project_feasible(best_h + rng.randn(n) * 0.06, n)
                h_cd2 = coordinate_descent(h_p, n, max_sweeps=40, tol=1e-11)
                h_cd2 = project_feasible(h_cd2, n)
                s2 = compute_score(h_cd2)
                if s2 < best_score:
                    best_score, best_h = s2, h_cd2

    # Phase 3: Binary local search + refinement
    if elapsed() < budget_s * 0.75:
        hb, sb = binary_local_search(best_h, n, rng, max_iters=30000)
        if sb < best_score:
            best_score, best_h = sb, hb.copy()
        # Refine binary solution with CD
        h_ref = coordinate_descent(best_h, n, max_sweeps=80, tol=1e-11)
        h_ref = project_feasible(h_ref, n)
        sr = compute_score(h_ref)
        if sr < best_score:
            best_score, best_h = sr, h_ref
        # Another round of binary search from refined
        hb2, sb2 = binary_local_search(best_h, n, rng, max_iters=30000)
        if sb2 < best_score:
            best_score, best_h = sb2, hb2.copy()

    # Phase 4: Multi-resolution
    if elapsed() < budget_s * 0.9:
        x_mid = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
        for n_try in [n + 10, n + 20, n + 30]:
            if elapsed() > budget_s * 0.9:
                break
            x_new = np.linspace(0.0, 2.0, n_try, endpoint=False) + 1.0 / n_try
            h_try = np.interp(x_new, x_mid, best_h)
            h_try = project_feasible(h_try, n_try)
            _g_n = n_try
            bounds_t = [(0.0, 1.0)] * n_try
            try:
                res = minimize(soft_max_score, h_try, args=(400.0,), method='SLSQP',
                               bounds=bounds_t,
                               constraints=[{'type': 'eq', 'fun': _sum_con}],
                               options={'maxiter': 100, 'ftol': 1e-12})
                h_try = project_feasible(res['x'], n_try)
            except Exception:
                pass
            h_try = coordinate_descent(h_try, n_try, max_sweeps=60, tol=1e-11)
            h_try = project_feasible(h_try, n_try)
            s_try = compute_score(h_try)
            if s_try < best_score:
                best_score, best_h = s_try, h_try.copy()
        _g_n = n

    # Phase 5: Final intensive polish
    n_final = len(best_h)
    _g_n = n_final
    if elapsed() < budget_s * 0.97:
        h_cd_f = coordinate_descent(best_h, n_final, max_sweeps=200, tol=1e-12)
        h_cd_f = project_feasible(h_cd_f, n_final)
        sf = compute_score(h_cd_f)
        if sf < best_score:
            best_score, best_h = sf, h_cd_f
        bounds_f = [(0.0, 1.0)] * n_final
        try:
            res = minimize(compute_score, best_h, method='SLSQP', bounds=bounds_f,
                           constraints=[{'type': 'eq', 'fun': _sum_con}],
                           options={'maxiter': 300, 'ftol': 1e-14})
            hf = project_feasible(res['x'], n_final)
            sf2 = compute_score(hf)
            if sf2 < best_score:
                best_score, best_h = sf2, hf
        except Exception:
            pass
        # One more CD
        h_cd2 = coordinate_descent(best_h, n_final, max_sweeps=100, tol=1e-13)
        h_cd2 = project_feasible(h_cd2, n_final)
        sf3 = compute_score(h_cd2)
        if sf3 < best_score:
            best_score, best_h = sf3, h_cd2

    return best_h, best_score, len(best_h)
