   ```python
   from scipy.optimize import minimize
   import time

   _budget = 1100.0
   _start_t = 0.0

   def compute_c5(h, n):
       dx = 2.0 / n
       corr = np.correlate(h, 1.0 - h, mode="full")
       return float(np.max(corr) * dx)

   def project_to_set(h, target):
       lo, hi = -2.0, 2.0
       for _ in range(60):
           mid = 0.5 * (lo + hi)
           if float(np.sum(np.clip(h - mid, 0.0, 1.0))) > target:
               lo = mid
           else:
               hi = mid
       return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

   def upsample(h, n_new):
       n_old = len(h)
       x_old = np.linspace(0.0, 2.0, n_old)
       x_new = np.linspace(0.0, 2.0, n_new)
       return np.interp(x_new, x_old, h)

   def slsqp_optimize(h0, n, maxiter=300, ftol=1e-10):
       target = n / 2.0
       h0 = project_to_set(h0.copy(), target)
       bounds = [(0.0, 1.0)] * n

       def obj(x):
           hp = np.clip(x, 0, 1)
           return compute_c5(hp, n)

       def grad(x):
           # Finite difference gradient (expensive but correct)
           eps = 1e-7
           g = np.zeros(n)
           for i in range(n):
               xp = x.copy(); xp[i] += eps
               xm = x.copy(); xm[i] -= eps
               g[i] = (obj(xp) - obj(xm)) / (2 * eps)
           return g

       try:
           res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': ftol})
           h_out = project_to_set(res.x.copy(), target)
           return h_out, compute_c5(h_out, n)
       except:
           return h0, compute_c5(h0, n)
   ```
