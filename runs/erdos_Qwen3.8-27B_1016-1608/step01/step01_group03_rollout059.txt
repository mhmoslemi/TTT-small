    dx, beta, lam, tgt = _G['dx'], _G['beta'], _G['lam'], _G['target']
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode="full")
    cs = corr - np.max(corr)
    w = np.exp(beta * cs)
    w /= w.sum()
    g = np.zeros(n)
    shift = n - 1
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-16:
            continue
        s = k - shift
        if s >= 0:
            g[:n - s] += wk * (1.0 - x[s:])
            g[s:] -= wk * x[:n - s]
        else:
            g[-s:] += wk * (1.0 - x[:n + s])
            g[:n + s] -= wk * x[-s:]
    g *= dx
    g += 2.0 * lam * (np.sum(x) - tgt)
    return g

def raw_obj(x):
    return compute_c5(np.clip(x, 0.0, 1.0), _G['dx'])

def sum_con(x):
    return np.sum(x) - _G['target']

def subgrad_step(h, dx, target, step_size):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    cmax = np.max(corr)
    active = np.where(corr > cmax - 0.005)[0]
    g = np.zeros(n)
    shift = n - 1
    for k in active:
        s = k - shift
        if s >= 0:
            g[:n - s] += 1.0 - h[s:]
            g[s:] -= h[:n - s]
        else:
            g[-s:] += 1.0 - h[:n + s]
            g[:n + s] -= h[-s:]
    g *= dx / len(active)
    return repair(h - step_size * g, target)

def make_structured_init(n, freq, amp):
    xs = np.linspace(0, 2, n)
    h = 0.5 + amp * np.sin(np.pi * freq * xs)
    return repair(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.ones(70) * 0.5
    n = len(h0)
    target = n / 2.0
    dx = 2.0 / n
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con}]
    _G['dx'] = dx
    _G['target'] = target
    _G['lam'] = 1000.0

    h0 = repair(h0, target)
    best_c5 = compute_c5(h0, dx)
    best_h = h0.copy()

    # Phase 1: SLSQP from initial
    if time.time() - t0 < budget_s - 130:
        try:
            r = minimize(raw_obj, h0, method='SLSQP', bounds=bnds,
                         constraints=cons, options={'maxiter': 500, 'ftol': 1e-14})
            hn = repair(r.x, target)
            cn = compute_c5(hn, dx)
            if cn < best_c5:
                best_c5, best_h = cn, hn
        except Exception:
            pass

    # Phase 2: L-BFGS-B with LSE beta annealing
    for beta in [3, 10, 30, 100, 300, 1000, 3000, 10000]:
        if time.time() - t0 > budget_s - 90:
            break
        _G['beta'] = beta
        maxit = min(400, 100 + int(300 * np.log10(1 + beta)))
        try:
            r = minimize(lse_obj, best_h, jac=lse_grad, method='L-BFGS-B',
                         bounds=bnds, options={'maxiter': maxit, 'gtol': 1e-11, 'ftol': 1e-16})
            hn = repair(r.x, target)
            cn = compute_c5(hn, dx)
            if cn < best_c5:
                best_c5, best_h = cn, hn
        except Exception:
            pass

    # Phase 3: Multi-start with diverse initializations
    for i in range(30):
        if time.time() - t0 > budget_s - 60:
            break
        if i == 0:
            hp = make_structured_init(n, 2, 0.35)
        elif i == 1:
            hp = make_structured_init(n, 4, 0.30)
        elif i == 2:
            hp = make_structured_init(n, 3, 0.40)
        else:
            sc = 0.02 + 0.15 * ((i - 3) / 27.0)
            hp = repair(best_h + np.random.randn(n) * sc, target)
        _G['beta'] = 100
        try:
            r = minimize(lse_obj, hp, jac=lse_grad, method='L-BFGS-B',
                         bounds=bnds, options={'maxiter': 250, 'gtol': 1e-10})
            hn = repair(r.x, target)
            cn = compute_c5(hn, dx)
            if cn < best_c5:
                best_c5, best_h = cn, hn
        except Exception:
            pass

    # Phase 4: Subgradient descent with annealing
    if time.time() - t0 < budget_s - 25:
        step = 0.004
        for it in range(8000):
            if time.time() - t0 > budget_s - 10:
                break
            best_h = subgrad_step(best_h, dx, target, step)
            cn = compute_c5(best_h, dx)
            if cn < best_c5:
                best_c5 = cn
            if it % 500 == 0 and it > 0:
                step *= 0.82

    # Phase 5: Final SLSQP polish (two passes)
    for _ in range(2):
        if time.time() - t0 > budget_s - 5:
            break
        try:
            r = minimize(raw_obj, best_h, method='SLSQP', bounds=bnds,
                         constraints=cons, options={'maxiter': 500, 'ftol': 1e-15})
            hn = repair(r.x, target)
            cn = compute_c5(hn, dx)
            if cn < best_c5:
                best_c5, best_h = cn, hn
        except Exception:
            break

    best_h = repair(best_h, target)
    best_c5 = compute_c5(best_h, dx)
    return (best_h, best_c5, n)
