            hn = do_slsqp(bh, n, 0.0005, 300, 1e-13)
            c5 = compute_c5(hn)
            if c5 < bc:
                bc, bh = c5, hn

    # Phase 6: Higher resolution
    if time.time() - t0 < budget_s - 180:
        n2 = 2 * n
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n2, endpoint=False)
        h_up = np.interp(x_new, x_old, np.concatenate([bh, [bh[0]]]))
        h_up = repair(h_up, n2)
        bn2 = compute_c5(h_up)
        bh2 = h_up.copy()
        for eps in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
            if time.time() - t0 > budget_s - 80:
                break
            hn2 = do_slsqp(bh2, n2, eps, 300, 1e-13)
            c5_2 = compute_c5(hn2)
            if c5_2 < bn2:
                bn2, bh2 = c5_2, hn2
        for i in range(40):
            if time.time() - t0 > budget_s - 80:
                break
            hp2 = bh2 + np.random.randn(n2) * 0.04
            hp2 = repair(hp2, n2)
            hn2 = do_slsqp(hp2, n2, 0.002, 200, 1e-12)
            c5_2 = compute_c5(hn2)
            if c5_2 < bn2:
                bn2, bh2 = c5_2, hn2
        if time.time() - t0 > budget_s - 60:
            pass
        else:
            t_end6 = min(time.time() + 20, t0 + budget_s - 60)
            bb2, bn2b = binary_ls(bh2, n2, t0, t_end6)
            if bn2b < bn2:
                bn2, bh2 = bn2b, bb2
                hn2 = do_slsqp(bh2, n2, 0.001, 300, 1e-13)
                c5_2 = compute_c5(hn2)
                if c5_2 < bn2:
                    bn2, bh2 = c5_2, hn2
        if bn2 < bc:
            bc, bh, n = bn2, bh2, n2

    # Phase 7: Final fine polish
    if time.time() - t0 < budget_s - 10:
        for eps in [0.0001, 5e-05, 1e-05]:
            if time.time() - t0 > budget_s - 5:
                break
            hn = do_slsqp(bh, n, eps, 600, 1e-14)
            c5 = compute_c5(hn)
            if c5 < bc:
                bc, bh = c5, hn

    fh = repair(bh, n)
    fc = compute_c5(fh)
    return (fh.tolist(), float(fc), n)
