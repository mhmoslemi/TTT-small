import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return c.max() * dx

def corr_vec(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() > 0:
            h[free] += diff / free.sum()
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_one(h, k_idx, n):
    shift = k_idx - (n - 1)
    m = np.arange(n)
    jp = m + shift
    jm = m - shift
    grad = np.zeros(n)
    vp = (jp >= 0) & (jp < n)
    vm = (jm >= 0) & (jm < n)
    grad[vp] += (1.0 - h[jp[vp]])
    grad[vm] -= h[jm[vm]]
    return grad

def subgrad_topk(h, n, k=3):
    c = corr_vec(h)
    idx = np.argsort(c)[-k:]
    w = np.exp(c[idx] - c[idx].max())
    w /= w.sum()
    grad = np.zeros(n)
    for wi, ki in zip(w, idx):
        grad += wi * subgrad_one(h, ki, n)
    return grad

def soft_max_obj(h, temp):
    c = corr_vec(h)
    m = c.max()
    return m + temp * np.log(np.sum(np.exp((c - m) / temp)))

def sum_eq(x):
    return x.sum() - len(x) / 2.0

def sum_eq_jac(x):
    return np.ones(len(x))

def subgrad_optimize(h0, n, time_limit, k=3, base_lr=0.3):
    best = h0.copy()
    best_c5 = compute_c5(h0)
    h = h0.copy()
    h_prev = h0.copy()
    t = 1.0
    stagnation = 0
    
    while time.time() < time_limit:
        y = h + (t - 1) / (t + 2) * (h - h_prev)
        y = project_feasible(y, n)
        grad = subgrad_topk(y, n, k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-10:
            break
        
        lr = base_lr / np.sqrt(t)
        h_prev = h.copy()
        h = project_feasible(h - lr * grad, n)
        t += 1
        
        c5_t = compute_c5(h)
        if c5_t < best_c5:
            best_c5 = c5_t
            best = h.copy()
            stagnation = 0
        else:
            stagnation += 1
            if stagnation > 100:
                base_lr *= 0.8
                stagnation = 0
                if base_lr < 0.001:
                    break
    
    return best, best_c5

def coord_optimize(h0, n, time_limit, step_sizes):
    best = h0.copy()
    best_c5 = compute_c5(h0)
    improved = True
    while improved and time.time() < time_limit:

        for i in range(n):
