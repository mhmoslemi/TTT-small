    return float(np.max(corr) * dx)

def smooth_obj_grad(h, dx, temp):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    s = temp * (corr - m)
    s = np.clip(s, -500, 500)
    exp_s = np.exp(s)
    sum_exp = np.sum(exp_s)
    obj = (m + np.log(sum_exp) / temp) * dx
    w = exp_s / sum_exp
    grad = np.zeros(n)
    n_full = len(corr)
    for l in range(n_full):
        wl = w[l]
        if wl < 1e-10:
            continue
        j = l - (n - 1)
        # First term: (1-h[x-j]) for x in [max(0,j), min(n, n+j))
        x1lo = max(0, j)
        x1hi = min(n, n + j)
        if x1hi > x1lo:
            idx = np.arange(x1lo, x1hi)
            grad[idx] += wl * (1.0 - h[idx - j])
        # Second term: -h[x+j] for x in [max(0,-j), min(n, n-j))
        x2lo = max(0, -j)
        x2hi = min(n, n - j)
        if x2hi > x2lo:
            idx = np.arange(x2lo, x2hi)
            grad[idx] += wl * (-h[idx + j])
    grad *= dx
    return obj, grad

    v, _ = smooth_obj_grad(x, dx, temp)
    return v + pen * (x.sum() - target) ** 2
