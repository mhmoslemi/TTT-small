        h_new = slsqp_exact(best_h, dx, target, maxiter=1000)
        if h_new is not None:
            c5 = compute_c5(h_new, dx)
            if c5 < best_c5:
                best_h, best_c5 = h_new, c5

    # Phase 6: Try slightly different n
    for n_try in [50, 60, 80, 100]:
        if tl() < 30:
            break
        dx_t = 2.0 / n_try
        target_t = n_try / 2.0
        _g_target = target_t
        # Interpolate from best_h
        old_x = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
        new_x = np.linspace(0, 2, n_try, endpoint=False) + 1.0 / n_try
        h_interp = project(np.interp(new_x, old_x, best_h), target_t)
        h_opt, c5_opt = optimize_one(n_try, dx_t, target_t, h_interp, tl, rng)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    _g_target = n / 2.0
    best_h = project(best_h, target)
    # Ensure we return with the right n
    if len(best_h) != n:
        n = len(best_h)
        dx = 2.0 / n
        target = n / 2.0
    best_c5 = compute_c5(best_h, dx)
    return (best_h, best_c5, n)
