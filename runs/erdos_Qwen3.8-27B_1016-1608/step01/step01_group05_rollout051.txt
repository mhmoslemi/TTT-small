def compute_c5(h, n):

    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_c5_and_active(h, n):

    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    idx = int(np.argmax(corr))

    lo, hi = -2.0, 2.0

    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)

Or: minimize max_k g_k(h) + λ * sum_k max(0, g_k(h) - target)²

This gives a more "flat" correlation profile.

**Another idea: Use the structure of the problem**

The correlation at lag 0 is: Σ h[i](1-h[i]) * dx = (Σh[i] - Σh[i]²) * dx = 1 - dx * Σh[i]²

To minimize this, we want to maximize Σh[i]², which for a fixed sum means making h as "spread out" as possible (some 0s and 1s).

For binary h: Σh[i]² = n/2, so lag 0 = (n/2 - n/2) * dx = 0. 

So for binary sequences, the max is always at a non-zero lag. This is a good property.

For the non-zero lags, the correlation is:
g_s = Σ_{valid i} h[i](1-h[i-s]) * dx

For binary h, this counts the number of positions where h has a 1 and h shifted by s has a 0, times dx.

If we have n/2 ones and n/2 zeros, and the shift by s moves m positions from "1 to 0" overlap, then g_s = m * dx = m * 2/n.

To minimize the max, we want to arrange the ones and zeros so that for every shift s, the number of "1 followed by 0 at distance s" is minimized (and uniform).

This is exactly a binary sequence design problem! The optimal binary sequence would be one where the autocorrelation (or rather, the cross-correlation with its complement) is as flat as possible.

For a binary sequence of length n with n/2 ones, the sum of all cross-correlations at all lags is:
Σ_s g_s = Σ_s Σ_i h[i](1-h[i-s]) = Σ_i h[i] * (number of zeros in the window)

import numpy as np
from scipy.optimize import minimize
import time

_budget = 1100.0
_start_t = 0.0

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_c5_with_corr(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    idx = int(np.argmax(corr))
    return float(corr[idx]), idx, corr

def project_to_set(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        hp = np.clip(h - mid, 0.0, 1.0)
        if float(np.sum(hp)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, h)

def time_left():
    return max(0.0, _budget - (time.time() - _start_t))

def to_binary(h, n):
    h = np.asarray(h, dtype=float)
    idx = np.argsort(h)[::-1][:n // 2]
    out = np.zeros(n)
    out[idx] = 1.0
    return out

def binary_sa(h_init, n, rng, T0=0.003, Tf=0.0001, n_steps=20000):
    h = to_binary(h_init, n)
    best_c5, _, _ = compute_c5_with_corr(h, n)
    best_h = h.copy()
    for step in range(n_steps):
        if step % 2000 == 0 and time_left() < 40:
            break
        T = T0 * (Tf / T0) ** (step / float(n_steps))
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        i = int(rng.choice(ones))
        j = int(rng.choice(zeros))
        h[i], h[j] = 0.0, 1.0
        c5_new, _, _ = compute_c5_with_corr(h, n)
        delta = c5_new - best_c5
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-12)):
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
    return best_h, best_c5

def targeted_coord_descent(h, n, rng, n_sweeps=80):
    target = n / 2.0
    h = project_to_set(h.copy(), target)
    best = compute_c5(h, n)
    for sweep in range(n_sweeps):
        if time_left() < 15:
            break
        c5_cur, active_idx, corr = compute_c5_with_corr(h, n)
        s = active_idx - (n - 1)
        order = rng.permutation(n)
        improved = False
        for j in order:
            old = h[j]
            h[j] = 0.0
            h = project_to_set(h, target)
            v0 = compute_c5(h, n)
            h[j] = 1.0
            h = project_to_set(h, target)
            v1 = compute_c5(h, n)
            h[j] = old
            h = project_to_set(h, target)
            if v0 < best - 1e-12:
                h[j] = 0.0
                h = project_to_set(h, target)
                best = v0
                improved = True
            elif v1 < best - 1e-12:
                h[j] = 1.0
                h = project_to_set(h, target)
                best = v1
                improved = True
            else:
                d = 0.05 + 0.1 * rng.rand()
                for sign in [1.0, -1.0]:
                    c = float(np.clip(old + sign * d, 0.0, 1.0))
                    h[j] = c
                    h = project_to_set(h, target)
                    vc = compute_c5(h, n)
                    if vc < best - 1e-12:
                        best = vc
                        improved = True
                        break
                    elif sign > 0 and vc > best:
                        h[j] = old
                        h = project_to_set(h, target)
                        break
                else:
                    h[j] = old
                    h = project_to_set(h, target)
        if not improved:
            break
    return h, best

def perturbation_search(h, n, rng, n_trials=30, scale=0.05):
    target = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    for _ in range(n_trials):
        if time_left() < 20:
            break
        h_p = h + rng.normal(0, scale, n)
        h_p = project_to_set(h_p, target)
        h_c, c5_c = targeted_coord_descent(h_p, n, rng, n_sweeps=15)
        if c5_c < best_c5:
            best_c5 = c5_c
            best_h = h_c.copy()
    return best_h, best_c5

def slsqp_optimize(h0, n, maxiter=400):
    target = n / 2.0
    h0 = project_to_set(h0.copy(), target)
    bounds = [(0.0, 1.0)] * n
    def _obj(x):
        return compute_c5(x, n)
    def _con(x):
        return np.sum(x) - target
    cons = [{'type': 'eq', 'fun': _con}]
    try:
        res = minimize(_obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        h_out = project_to_set(res.x.copy(), target)
        return h_out, compute_c5(h_out, n)
    except Exception:
        return h0, compute_c5(h0, n)

def lse_smoothed_optimize(h0, n, T=50.0, maxiter=300):
    target = n / 2.0
    h0 = project_to_set(h0.copy(), target)
    bounds = [(0.0, 1.0)] * n
    def _obj(x):
        dx = 2.0 / n
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        m = np.max(corr)
        return m + (1.0 / T) * np.log(np.sum(np.exp(T * (corr - m))))
    def _con(x):
        return np.sum(x) - target
    cons = [{'type': 'eq', 'fun': _con}]
    try:
        res = minimize(_obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-13})
        h_out = project_to_set(res.x.copy(), target)
        return h_out, compute_c5(h_out, n)
    except Exception:
        return h0, compute_c5(h0, n)
