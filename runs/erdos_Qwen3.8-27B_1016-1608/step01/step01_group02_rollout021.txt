import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def projection(h, target):
    h = np.array(h, dtype=float, copy=True)
    for _ in range(500):
        np.clip(h, 0.0, 1.0, out=h)
        excess = h.sum() - target
        if abs(excess) < 1e-13:
            break
        if excess > 0:
            mask = h > 1e-15
            cnt = mask.sum()
            if cnt == 0:
                break
            s = min(excess / cnt, h[mask].min())
            h[mask] -= s
        else:
            mask = h < 1.0 - 1e-15
            cnt = mask.sum()
            if cnt == 0:
                break
            s = min(-excess / cnt, (1.0 - h[mask]).min())
            h[mask] += s
    return h

def smooth_max_obj(h, dx, temp):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    scaled = temp * (m - corr)
    return (m - np.log(np.sum(np.exp(scaled)) / len(corr)) / temp) * dx

def smooth_max_grad(h, dx, temp):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    w = np.exp(temp * (m - corr))
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        lo1 = max(0, -lag)
        hi1 = min(n, n - lag)
        lo2 = max(0, lag)
        hi2 = min(n, n + lag)
        g = np.zeros(n)
        if hi1 > lo1:
            g[lo1:hi1] += (1.0 - h[lo1 + lag:hi1 + lag])
        if hi2 > lo2:
            g[lo2:hi2] -= h[lo2 - lag:hi2 - lag]
        grad += w[k] * g
    return grad * dx

def subgrad_max(h, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    k = np.argmax(corr)
    lag = k - (n - 1)
    grad = np.zeros(n)
    lo1 = max(0, -lag)
    hi1 = min(n, n - lag)
    lo2 = max(0, lag)
    hi2 = min(n, n + lag)
    if hi1 > lo1:
        grad[lo1:hi1] += (1.0 - h[lo1 + lag:hi1 + lag])
    if hi2 > lo2:
        grad[lo2:hi2] -= h[lo2 - lag:hi2 - lag]
    return grad * dx

def make_block_init(n, rng, pattern="periodic"):
    h = np.zeros(n)
    if pattern == "periodic":
        period = rng.randint(3, 9)
        fill = rng.randint(1, period)
        for i in range(n):
            if i % period < fill:
                h[i] = 1.0
    elif pattern == "bilateral":
        c = n // 2
        r = n // 3
        h[c - r:c + r + 1] = 1.0
    elif pattern == "random_blocks":
        pos = 0
        while pos < n:
            blen = rng.randint(2, max(3, n // 6))
            if rng.rand() < 0.5:
                h[pos:min(pos + blen, n)] = 1.0
            pos += blen + rng.randint(1, 3)
    h = projection(h, n / 2.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t0)

    best_h = np.array(initial_h_values, dtype=float).copy()
    n = len(best_h)
    dx = 2.0 / n
    target = n / 2.0
    best_c5 = compute_c5(best_h, dx)
    best_n = n

    def try_update(h_cand, n_c, dx_c):
        nonlocal best_h, best_c5, best_n
        if h_cand is None:
            return
        c5 = compute_c5(h_cand, dx_c)
        if c5 < best_c5 - 1e-13:
            best_h = h_cand.copy()
            best_c5 = c5
            best_n = n_c

    # Phase 1: Momentum-based projected gradient on smooth max
    for temp in [50.0, 150.0, 500.0, 1500.0, 5000.0]:
        if time_left() < 30:
            break
        h_curr = best_h.copy()
        momentum = np.zeros(n)
        lr = 0.05
        for it in range(3000):
            if time_left() < 10:
                break
            g = smooth_max_grad(h_curr, dx, temp)
            momentum = 0.9 * momentum - lr * g
            h_new = projection(h_curr + momentum, target)
            # Simulated annealing acceptance
            if it % 200 == 199:
                c5_old = smooth_max_obj(h_curr, dx, temp)
                c5_new = smooth_max_obj(h_new, dx, temp)
                if c5_new < c5_old or rng.rand() < np.exp(-(c5_new - c5_old) * temp * 0.1):
                    h_curr = h_new
                else:
                    h_curr = projection(h_curr + rng.randn(n) * 0.1, target)
                lr = max(0.001, lr * 0.8)
            else:
                h_curr = h_new
        c5_check = compute_c5(h_curr, dx)
        if c5_check < best_c5:
            best_h = h_curr.copy()
            best_c5 = c5_check

    # Phase 2: Subgradient with Polyak step size
    if time_left() > 60:
        h_curr = best_h.copy()
        best_obj = best_c5
        k_iter = 1
        for it in range(5000):
            if time_left() < 10:
                break
            g = subgrad_max(h_curr, dx)
            f_val = compute_c5(h_curr, dx)
            lr_polyak = (best_obj - f_val) / (np.dot(g, g) + 1e-10)
            lr_polyak = max(1e-4, min(lr_polyak, 0.5))
            h_new = projection(h_curr - lr_polyak * g, target)
            f_new = compute_c5(h_new, dx)
            if f_new < best_obj:
                best_obj = f_new
                h_curr = h_new
                if f_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = f_new
            else:
                h_curr = h_new
            if it % 500 == 499 and time_left() > 20:
                h_curr = projection(h_curr + rng.randn(n) * 0.08, target)

    # Phase 3: Try different n values with optimization
    for n_try in [50, 60, 80, 100]:
        if time_left() < 60:
            break
        dx_try = 2.0 / n_try
        target_try = n_try / 2.0
        # Interpolate best_h to new n
        x_old = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
        x_new = np.linspace(0, 2, n_try, endpoint=False) + 1.0 / n_try
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = projection(h_interp, target_try)
        # Also try some structured inits
        inits = [h_interp, make_block_init(n_try, rng, "periodic"), make_block_init(n_try, rng, "random_blocks")]
        for h_init in inits:
            if time_left() < 20:
                break
            h_c = h_init.copy()
            # Quick momentum optimization
            momentum = np.zeros(n_try)
            lr = 0.03
            for temp in [100.0, 500.0, 2000.0]:
                for it in range(1500):
                    if time_left() < 10:
                        break
                    g = smooth_max_grad(h_c, dx_try, temp)
                    momentum = 0.9 * momentum - lr * g
                    h_c = projection(h_c + momentum, target_try)
                    if it % 400 == 399:
                        lr = max(0.001, lr * 0.7)
            c5_try = compute_c5(h_c, dx_try)
            if c5_try < best_c5:
                best_h = h_c.copy()
                best_c5 = c5_try
                best_n = n_try

    # Phase 4: L-BFGS-B on smooth max (with penalty for constraint)
    if time_left() > 40:
        n = best_n
        dx = 2.0 / n
        target = n / 2.0
        for temp in [500.0, 2000.0]:
            if time_left() < 20:
                break
            def obj_pen(x):
                c = smooth_max_obj(x, dx, temp)
                p = (x.sum() - target) ** 2 * 100.0
                p += np.sum(np.maximum(-x, 0) ** 2 + np.maximum(x - 1, 0) ** 2) * 100.0
                return c + p
            def obj_grad(x):
                g = smooth_max_grad(x, dx, temp) + 200.0 * (x.sum() - target)
                g += 200.0 * np.maximum(-x, 0) - 200.0 * np.maximum(x - 1, 0)
                return g
            x0 = best_h + rng.randn(n) * 0.02
            x0 = projection(x0, target)
            try:
                res = minimize(obj_pen, x0, jac=obj_grad, method='L-BFGS-B',
                              options={'maxiter': 500, 'ftol': 1e-16})
                h_new = projection(res.x, target)
                c5_new = compute_c5(h_new, dx)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
            except Exception:
                pass

    # Phase 5: SLSQP polish on exact C5
    if time_left() > 15:
        n = best_n
        dx = 2.0 / n
        target = n / 2.0
        bounds = [(0.0, 1.0)] * n
        eq_con = ({'type': 'eq', 'fun': lambda x: x.sum() - target},)
        for attempt in range(3):
            if time_left() < 10:
                break
            noise = 0.0 if attempt == 0 else rng.randn(n) * rng.uniform(0.02, 0.08)
            h0 = projection(best_h + noise, target)
            try:
                res = minimize(compute_c5, h0, args=(dx,), method='SLSQP',
                              bounds=bounds, constraints=eq_con,
                              options={'maxiter': 800, 'ftol': 1e-15})
                h_new = projection(res.x, target)
                c5_new = compute_c5(h_new, dx)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
            except Exception:
                pass

    # Phase 6: Final coordinate descent polish
    if time_left() > 10:
        n = best_n
        dx = 2.0 / n
        target = n / 2.0
        h_c = best_h.copy()
        c5_c = best_c5
        step = 0.02
        for it in range(20000):
            if time_left() < 3:
                break
            idx = rng.randint(n)
            for delta in [step, -step]:
                h_t = h_c.copy()
                h_t[idx] = np.clip(h_t[idx] + delta, 0, 1)
                h_t = projection(h_t, target)
                c5_t = compute_c5(h_t, dx)
                if c5_t < c5_c - 1e-14:
                    h_c, c5_c = h_t, c5_t
                    if c5_t < best_c5:
                        best_h = h_c.copy()
                        best_c5 = c5_t
                    break
            if it % 1000 == 999:
                step = max(0.001, step * 0.7)

    best_h = projection(best_h, best_n / 2.0)
    final_c5 = compute_c5(best_h, 2.0 / best_n)
    return (best_h, final_c5, best_n)
