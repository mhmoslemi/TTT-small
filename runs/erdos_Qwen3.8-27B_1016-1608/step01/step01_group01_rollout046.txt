import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return c.max() * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_obj(h, temp):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    return m + temp * np.log(np.sum(np.exp((c - m) / temp)))

def sum_eq(h):
    return np.sum(h) - len(h) / 2.0

def sum_eq_jac(h):
    return np.ones(len(h))

def pairwise_cd_fast(h, n, t_end, n_fracs=6):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

            break

                d_hi = min(hi, 1.0 - hj)

                    continue
