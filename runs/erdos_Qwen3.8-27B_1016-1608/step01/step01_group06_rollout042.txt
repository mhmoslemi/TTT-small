def subgradient_step(h, n, lr=0.01):
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full')
    k_star = np.argmax(corrs)
    # Subgradient of corr[k_star] w.r.t. h
    g = np.zeros(n)
    # For shift k_star in 'full' mode:
    # corr[k] = sum_i h[i] * (1-h[i+k-(n-1)])
    # valid i: 0 <= i < n and 0 <= i+k-(n-1) < n
    shift = k_star - (n - 1)
    for i in range(n):
        # h[i] as first factor
        j1 = i + shift
        if 0 <= j1 < n:
            g[i] += (1.0 - h[j1])
        # h[i] appears in (1-h[j]) where j + shift = i, i.e., j = i - shift
        j2 = i - shift
        if 0 <= j2 < n:
            g[i] -= h[j2]
    g *= dx
    # Project to be orthogonal to all-ones (constraint direction)
    g -= np.mean(g)
    # Update
    h_new = h - lr * g
    h_new = project_feasible(h_new, n)
    return h_new

    n = len(h)
    sh = k_star - (n - 1)
    idx = np.arange(n)
    g = np.zeros(n)
    j1 = idx + sh
    v1 = (j1 >= 0) & (j1 < n)
    g[v1] += 1.0 - h[j1[v1]]
    j2 = idx - sh
    v2 = (j2 >= 0) & (j2 < n)
    g[v2] -= h[j2[v2]]

        s = compute_score(h)
        if s < bs:
            bs, bh = s, h.copy()
    return bh, bs

def opt_coord(h, n, rng, rnds, dl):
    h = h.copy()
    bs = compute_score(h)
    bh = h.copy()
    st = 0.04
    for _ in range(rnds):
        if time.time() > dl:
            break
        imp = False
        for i in rng.permutation(n):
            if time.time() > dl:
                break
            oi = h[i]
            bd, bj, bsc = 0, -1, bs
            for d in (st, -st, 0.3 * st, -0.3 * st):
                ni = oi + d
                if ni < 0 or ni > 1:
                    continue
                for off in (1, n // 4, n // 2):
                    j = (i + off) % n
                    if j == i:
                        continue
                    oj = h[j]
                    nj = oj - d
                    if nj < 0 or nj > 1:
                        continue
                    h[i], h[j] = ni, nj
                    sc = compute_score(h)
                    h[i], h[j] = oi, oj
                    if sc < bsc:
                        bsc, bd, bj = sc, d, j
            if bj >= 0:
                h[i] += bd
                h[bj] -= bd
                h = project_feasible(h, n)
                s = compute_score(h)
                if s < bs:
                    bs, bh = s, h.copy()
                imp = True
        if not imp:
            st *= 0.55
            if st < 1e-8:
                break
    return bh, bs

def slsqp_pen(x):
    return compute_score(x) + 1e5 * (x.sum() - len(x) / 2.0) ** 2

def slsqp_soft(x):
    c = compute_corrs(x)
    cm = c.max()
    return cm + np.log(np.exp(500.0 * (c - cm)).sum()) / 500.0 + 1e5 * (x.sum() - len(x) / 2.0) ** 2

def opt_slsqp(h, n, rng, dl):
    bs = compute_score(h)
    bh = h.copy()
    bnds = [(0.0, 1.0)] * n
    starts = [h.copy(), project_feasible(h + rng.randn(n) * 0.03, n)]
    for st in starts:
        for f, mi in [(slsqp_soft, 500), (slsqp_pen, 400)]:
            if time.time() > dl:
                break
            try:
                r = minimize(f, st, method='SLSQP', bounds=bnds,
                             options={'maxiter': mi, 'ftol': 1e-15})
                hc = project_feasible(r['x'], n)
                sc = compute_score(hc)
                if sc < bs:
                    bs, bh, st = sc, hc.copy(), hc.copy()
            except Exception:
                pass
    return bh, bs

def opt_sa(h, n, rng, iters, dl):
    h = h.copy()
    cur_s = compute_score(h)
    bs = cur_s
    bh = h.copy()
    T = 0.005
    for _ in range(iters):
        if time.time() > dl:
            break
        i, j = rng.choice(n, 2, replace=False)
        d = rng.uniform(-0.08, 0.08)
        ho = h.copy()
        h[i] += d
        h[j] -= d
        h = project_feasible(h, n)
        s = compute_score(h)
        if s < cur_s or rng.rand() < np.exp(-max(s - cur_s, 0.0) / max(T, 1e-8)):
            cur_s = s
        else:
            h = ho
        if cur_s < bs:
            bs, bh = cur_s, h.copy()
        T = max(T * 0.9995, 0.0005)
    return bh, bs

def try_inits(n, bh, bs, dl):
    x = np.linspace(0, 2, n, endpoint=False)
    for freq in range(1, 6):
        for amp in (0.2, 0.35, 0.5):
            if time.time() > dl:
                return bh, bs
            hs = project_feasible(0.5 + amp * np.sin(np.pi * freq * x), n)
            s = compute_score(hs)
            if s < bs:
                bh, bs = hs.copy(), s
    for nb in range(2, 7):
        if time.time() > dl:
            return bh, bs
        hb = np.zeros(n)
        for b in range(nb):
            si = int(b * n / nb)
            ei = int((b + 1) * n / nb)
            hb[si:ei] = 1.0 if b % 2 == 0 else 0.0
        hb = project_feasible(hb, n)
        s = compute_score(hb)
        if s < bs:
            bh, bs = hb.copy(), s
    return bh, bs

def upscale(bh, nn):
    xo = np.linspace(0, 2, len(bh), endpoint=False)
    xn = np.linspace(0, 2, nn, endpoint=False)
    return project_feasible(np.interp(xn, xo, bh), nn)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    bh = h0.copy()
    bs = compute_score(h0)

    bh, bs = try_inits(n0, bh, bs, t0 + budget_s * 0.015)

    d1 = t0 + budget_s * 0.48
    h, s = opt_sg(bh, n0, rng, 6000, d1)
    if s < bs:
        bh, bs = h.copy(), s
    h, s = opt_coord(bh, n0, rng, 500, d1)
    if s < bs:
        bh, bs = h.copy(), s
    h, s = opt_sa(bh, n0, rng, 15000, d1)
    if s < bs:
        bh, bs = h.copy(), s
    h, s = opt_slsqp(bh, n0, rng, d1)
    if s < bs:
        bh, bs = h.copy(), s
    h, s = opt_coord(bh, n0, rng, 400, d1)
    if s < bs:
        bh, bs = h.copy(), s

    d2 = t0 + budget_s * 0.68
    if time.time() < d2:
        hi = upscale(bh, 100)
        h, s = opt_sg(hi, 100, rng, 4000, d2)
        if s < bs:
            bh, bs = h.copy(), s
        h, s = opt_coord(bh, 100, rng, 250, d2)
        if s < bs:
            bh, bs = h.copy(), s

    d3 = t0 + budget_s * 0.84
    if time.time() < d3:
        hi = upscale(bh, 200)
        h, s = opt_sg(hi, 200, rng, 3000, d3)
        if s < bs:
            bh, bs = h.copy(), s
        h, s = opt_coord(bh, 200, rng, 150, d3)
        if s < bs:
            bh, bs = h.copy(), s

    d4 = t0 + budget_s * 0.98
    if time.time() < d4:
        nf = len(bh)
        h, s = opt_coord(bh, nf, rng, 500, d4)
        if s < bs:
            bh, bs = h.copy(), s
        h, s = opt_slsqp(bh, nf, rng, d4)
        if s < bs:
            bh, bs = h.copy(), s
        h, s = opt_coord(bh, nf, rng, 300, d4)
        if s < bs:
            bh, bs = h.copy(), s

    return bh, compute_score(bh), len(bh)
