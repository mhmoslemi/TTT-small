def compute_c5(h):

    dx = 2.0 / n

    return float(np.max(corr) * dx)

def repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not np.any(mask):
            h[:] = target / n
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def worst_lag(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return int(np.argmax(corr))

def coordinate_descent_step(h, n, alpha=0.01, iters=100):
    """Move coordinates to reduce the worst-lag correlation."""
    for _ in range(iters):
        lag_idx = worst_lag(h)
        k = lag_idx - (n - 1)  # actual lag
        dx = 2.0 / n
        # Compute gradient-like direction for this lag
        # For lag k > 0: corr = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
        # For lag k < 0: corr = sum_{i=0}^{n-1+k} h[i]*(1-h[i-k])
        # For lag k = 0: corr = sum h[i]*(1-h[i])
        
        grad = np.zeros(n)
        if k >= 0:
            for i in range(n - k):
                # term: h[i+k]*(1-h[i])
                grad[i+k] += (1 - h[i])
                grad[i] -= h[i+k]
        elif k < 0:
            for i in range(n + k):
                # term: h[i]*(1-h[i-k])
                grad[i] += (1 - h[i-k])
                grad[i-k] -= h[i]
        else:
            for i in range(n):
                # term: h[i]*(1-h[i]) = h[i] - h[i]^2
                # d/dh[i] = 1 - 2*h[i]
                grad[i] = 1 - 2 * h[i]

        h_new = h + step

Let me add a "worst-lag focused" perturbation:
- Find the lag k* that gives the max correlation
- Generate a perturbation that specifically tries to reduce corr[k*]
- Accept if the overall max doesn't increase too much

This is a form of "directed" search that could be more efficient than blind SA.

For reducing corr at lag k (positive k):
corr[k] = sum_{i=0}^{n-1-k} h[i] * (1-h[i+k])

To reduce this, for each i, we want h[i] and h[i+k] to be both 0 or both 1. So:
- If h[i] is high and h[i+k] is low: decrease h[i] or increase h[i+k]
- If h[i] is low and h[i+k] is high: increase h[i] or decrease h[i+k]

This gives a gradient direction for each coordinate.

Let me implement this as a "guided perturbation" in the SA:

```python
def guided_perturbation(h, n, temp, rng):
    """Perturb h in a direction that reduces the worst correlation."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = np.argmax(corr) - (n - 1)  # actual lag
    
    h_new = h.copy()
    
    if k_star > 0:
        for i in range(n - k_star):
            pair_prod = 1.0  # We want to reduce h[i]*(1-h[i+k_star])
            # Gradient of h[i]*(1-h[i+k]) w.r.t. h[i] is (1-h[i+k])
            # Gradient w.r.t. h[i+k] is -h[i]
            h_new[i] -= temp * 0.1 * (1 - h[i + k_star])
            h_new[i + k_star] += temp * 0.1 * h[i]
    elif k_star < 0:
        for i in range(n + k_star):
            h_new[i] -= temp * 0.1 * (1 - h[i - k_star])
            h_new[i - k_star] += temp * 0.1 * h[i]
    else:
        # Lag 0: h[i]*(1-h[i]), gradient = 1-2*h[i]
        h_new -= temp * 0.1 * (1 - 2 * h)
    
    # Add random noise
    h_new += temp * 0.05 * rng.randn(n)
    
    return h_new
```

This is a subgradient step in the direction that reduces the worst correlation, with some noise added for exploration.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    target = n / 2.0
    h = h.copy()
    np.clip(h, 0.0, 1.0, out=h)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            h.fill(target / n)
            break
        h[mask] += diff / mask.sum()
        np.clip(h, 0.0, 1.0, out=h)
    return h

def slsqp_opt(h0, n, maxiter=1000):
    bnds = [(0.0, 1.0)] * n
    target_sum = n / 2.0
    def obj(x):
        return compute_c5(x)
    def con_fun(x):
        return np.sum(x) - target_sum
    def con_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    c0 = compute_c5(h0)
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        if res.x is not None:
            h = repair(np.clip(res.x, 0.0, 1.0), n)
            if compute_c5(h) < c0:
                return h
    except Exception:
        pass
    return h0

def guided_step(h, n, temp, rng):

    k_star = int(np.argmax(corr)) - (n - 1)
    h_new = h.copy()
    scale = temp * 0.15

            h_new[i + k_star] += scale * h[i]
