```python
def project_binary(h, n):
    # Project to have exactly n/2 ones (binary)
    # Sort by value, keep top n/2 as 1s
    vals = np.sort(h)[::-1]
    threshold = vals[n//2 - 1]
    out = np.where(h >= threshold, 1.0, 0.0)
    # Handle ties
    n_ones = int(np.sum(out))
    if n_ones > n//2:
        tied = np.where((h == threshold) & (out == 1.0))[0]
        excess = n_ones - n//2
        for idx in tied[:excess]:
            out[idx] = 0.0
    elif n_ones < n//2:
        tied = np.where((h == threshold) & (out == 0.0))[0]
        deficit = n//2 - n_ones
        for idx in tied[:deficit]:
            out[idx] = 1.0
    return out

        else:
            hi = mid
```

For a perfectly uniform h = 1/2 everywhere, the autocorrelation is (1/2)*(1/2)*2 = 1/2 for all shifts (in the limit). Then C₅ = 1 - 1/2 = 1/2. That's the "naive" answer.

But we can do better by making h more concentrated in a structured way. The optimal should have C₅ < 1/2.

The theoretical lower bound for this problem... I recall that for the continuous case, the optimal C₅ is around 0.38 or so, which matches our target.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -5.0, 5.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo+hi), 0.0, 1.0)

def lse_val(h, n, beta):
    c5v = compute_c5_vec(h, n)
    m = np.max(c5v)
    return m + np.log(np.sum(np.exp(beta*(c5v - m)))) / beta

def lse_grad(h, n, beta):
    c5v = compute_c5_vec(h, n)
    m = np.max(c5v)
    eb = np.exp(beta * (c5v - m))
    w = eb / np.sum(eb)
    dx = 2.0 / n
    n_full = 2 * n - 1
    grad = np.zeros(n)
    for j in range(n):
        s1 = 0.0  # sum_k w[k] * 1{j+k valid}
        s2 = 0.0  # sum_k w[k] * h[j+k] * 1{j+k valid}
        s3 = 0.0  # sum_k w[k] * h[j-k] * 1{j-k valid}
        for idx in range(n_full):
            k = idx - (n-1)
            wi = w[idx]
            if wi < 1e-15:
                continue
            i1 = j + k
            if 0 <= i1 < n:
                s1 += wi
                s2 += wi * h[i1]
            i2 = j - k
            if 0 <= i2 < n:
                s3 += wi * h[i2]
        grad[j] = dx * (s1 - s2 - s3)
    return grad

def lse_optimize(h0, n, beta_start=5, beta_end=50, n_iter=300, lr=0.1, rng=None):
    h = h0.copy()
    target = n / 2.0
    h = project_feasible(h, n)
    best = compute_c5(h, n)
    best_h = h.copy()
    
    for it in range(n_iter):
        frac = it / max(1, n_iter - 1)
        beta = beta_start + (beta_end - beta_start) * frac
        g = lse_grad(h, n, beta)
        # Gradient step
        h_new = h - lr * g
        # Project
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new, n)
        if c5_new < best:
            best = c5_new
            best_h = h_new.copy()
        h = h_new
        # Adaptive learning rate
        if c5_new > best * 1.01:
            lr *= 0.7
        elif c5_new < best * 0.99:
            lr = min(lr * 1.1, 0.5)
        if lr < 1e-6:
            break
    return best_h, best

def binary_local_search(h0, n, rng, n_iters=5000, temp=0.01):
    # Work with binary sequences
    target = n // 2
    h = np.zeros(n)
    # Initialize from h0
    thresholds = np.sort(h0)[::-1]
    h[np.argsort(h0)[::-1][:target]] = 1.0
    best_c5 = compute_c5(h, n)
    best_h = h.copy()

    zeros = np.where(h == 0.0)[0]

        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        c5_new = compute_c5(h, n)
        diff = c5_new - best_c5
        if diff < 0 or rng.random() < np.exp(-diff / max(temp, 1e-10)):
            best_c5 = min(best_c5, c5_new)
            if c5_new < best_c5 + 1e-15:
                best_h = h.copy()
                best_c5 = c5_new
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
        else:
            h[i] = 1.0
            h[j] = 0.0
        # Anneal
        temp *= 0.999
        
    return best_h, best_c5
```

So the total change in f_k from swap (i,j) is:

(where h[out_of_bounds] = 0)

```python
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -5.0, 5.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo+hi), 0.0, 1.0)

def upsample(h, n_new):
    x_old = np.linspace(0.0, 2.0, len(h))
    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, h)

def lse_gradient(h, n, beta):
    c5v = compute_c5_vec(h, n)
    m = np.max(c5v)
    eb = np.exp(beta * (c5v - m))
    w = eb / np.sum(eb)
    dx = 2.0 / n
    n_full = 2 * n - 1
    # grad[j] = dx * sum_k w[k] * [(1-h[j+k])*1{valid} - h[j-k]*1{valid}]
    # = dx * [sum_k w[k]*1{j+k valid} - sum_k w[k]*h[j+k]*1{valid} - sum_k w[k]*h[j-k]*1{valid}]
    # Use convolution: 
    # term1[j] = sum_k w[k] * 1{0<=j+k<n} = (w * ones_n)[j] in some sense
    # Let's just do it with the precomputed approach
    shifts = np.arange(-(n-1), n)
    # For each j, we need:
    # S1[j] = sum_idx w[idx] * 1{0<=j+shifts[idx]<n}
    # S2[j] = sum_idx w[idx] * h[j+shifts[idx]] * 1{valid}
    # S3[j] = sum_idx w[idx] * h[j-shifts[idx]] * 1{valid}
    
    # S2[j] = sum_k w[k] * h[j+k] (with h=0 out of bounds)
    # This is the correlation of h with w (padded)
    # np.correlate(h, w_padded) where w_padded is w centered...
    # Actually let's use the direct approach with FFT or just loop over j
    # For n <= 360, a loop with numpy vectorization is fast enough
    
    grad = np.zeros(n)
    for j in range(n):
        s = 0.0
        for idx in range(n_full):
            wi = w[idx]
            if wi < 1e-14:
                continue
            k = shifts[idx]
            i1 = j + k
            if 0 <= i1 < n:
                s += wi * (1.0 - h[i1])
            i2 = j - k
            if 0 <= i2 < n:
                s -= wi * h[i2]
        grad[j] = dx * s
    return grad
```

```python
# M[j, idx] = h[j + shifts[idx]] if 0 <= j+shifts[idx] < n else 0

cols = shifts[None, :]
idx_mat = rows + cols  # shape (n, 2n-1)

M3 = np.where((idx_mat3 >= 0) & (idx_mat3 < n), h[idx_mat3.clip(0, n-1)], 0.0)
S3 = M3.T @ w  # length n
# For S1: count of valid indices
M1 = ((idx_mat >= 0) & (idx_mat < n)).astype(float)
S1 = M1.T @ w  # length n

grad = (S1 - S2 - S3) * dx
```

```python
def binary_hill_climb(h0, n, n_sweeps=100, rng=None):
    target = n // 2
    h = np.zeros(n)
    h[np.argsort(h0)[::-1][:target]] = 1.0
    
    best_c5 = compute_c5(h, n)
    best_h = h.copy()
    
    for sweep in range(n_sweeps):
        c5v = compute_c5_vec(h, n)
        max_c5 = np.max(c5v)
        dx = 2.0 / n
        shifts = np.arange(-(n-1), n)
        
        # Precompute A: A[p] = h[p+shifts] + h[p-shifts] (out-of-bounds = 0)
        A = np.zeros((n, 2*n-1))
        for p in range(n):
            m1 = (p + shifts >= 0) & (p + shifts < n)
            A[p, m1] += h[p + shifts[m1]]
            m2 = (p - shifts >= 0) & (p - shifts < n)
            A[p, m2] += h[p - shifts[m2]]
        
        ones_idx = np.where(h == 1.0)[0]
        zeros_idx = np.where(h == 0.0)[0]
        
        # For all pairs, compute new max
        # delta[i,j] = (A[i] - A[j]) * dx
        # new_c5v[i,j] = c5v + delta[i,j]
        # new_max[i,j] = max(new_c5v[i,j])
        
        A_ones = A[ones_idx]  # (n1, 2n-1)
        A_zeros = A[zeros_idx]  # (n0, 2n-1)
        
        # Compute new_max for all pairs
        # This is (n1, n0, 2n-1) which for n=90 is (45, 45, 179) = 362,475
        diffs = A_ones[:, None, :] - A_zeros[None, :, :]  # (45, 45, 179)
        new_maxes = np.max(c5v[None, None, :] + diffs * dx, axis=2)  # (45, 45)
        
        # Find best improvement
        best_improve = max_c5 - np.min(new_maxes)
        if best_improve < 1e-12:
            break
        flat_idx = np.argmin(new_maxes)
        ii = flat_idx // len(zeros_idx)
        jj = flat_idx % len(zeros_idx)
        i = ones_idx[ii]
        j = zeros_idx[jj]
        
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h, n)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
    
    return best_h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -5.0, 5.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def upsample_h(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, h)

def precompute_A(h, n):
    shifts = np.arange(-(n - 1), n)
    A = np.zeros((n, 2 * n - 1))
    for p in range(n):
        m1 = (p + shifts >= 0) & (p + shifts < n)
        A[p, m1] += h[p + shifts[m1]]
        m2 = (p - shifts >= 0) & (p - shifts < n)
        A[p, m2] += h[p - shifts[m2]]
    return A

def binary_hill_climb(h0, n, n_sweeps=200, rng=None):
    target = n // 2
    h = np.zeros(n)
    h[np.argsort(h0)[::-1][:target]] = 1.0
    best_c5 = compute_c5(h, n)
    best_h = h.copy()
    for sweep in range(n_sweeps):
        c5v = compute_c5_vec(h, n)
        max_c5 = float(np.max(c5v))
        dx = 2.0 / n
        A = precompute_A(h, n)
        ones_idx = np.where(h == 1.0)[0]
        zeros_idx = np.where(h == 0.0)[0]
        A_o = A[ones_idx]
        A_z = A[zeros_idx]
        diffs = A_o[:, None, :] - A_z[None, :, :]
        new_maxes = np.max(c5v[None, None, :] + diffs * dx, axis=2)
        best_val = float(np.min(new_maxes))
        if max_c5 - best_val < 1e-12:
            if rng is not None and sweep < n_sweeps - 10:
                ni = rng.randint(len(ones_idx))
                nj = rng.randint(len(zeros_idx))
                h[ones_idx[ni]] = 0.0
                h[zeros_idx[nj]] = 1.0
                continue
            break
        fi = int(np.argmin(new_maxes))
        ii, jj = fi // len(zeros_idx), fi % len(zeros_idx)
        h[ones_idx[ii]] = 0.0
        h[zeros_idx[jj]] = 1.0
        c5_new = compute_c5(h, n)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
    return best_h, best_c5

def lse_gradient_vec(h, n, beta):
    c5v = compute_c5_vec(h, n)
    m = np.max(c5v)
    eb = np.exp(beta * (c5v - m))
    w = eb / np.sum(eb)
    dx = 2.0 / n
    shifts = np.arange(-(n - 1), n)
    rows = np.arange(n)[:, None]
    cols = shifts[None, :]
    idx1 = rows + cols
    idx2 = rows - cols
    mask1 = (idx1 >= 0) & (idx1 < n)
    mask2 = (idx2 >= 0) & (idx2 < n)
    h1 = np.where(mask1, h[idx1.clip(0, n - 1)], 0.0)
    h2 = np.where(mask2, h[idx2.clip(0, n - 1)], 0.0)
    S1 = mask1.T.astype(float) @ w
    S2 = h1.T @ w
    S3 = h2.T @ w
    return (S1 - S2 - S3) * dx

def lse_optimize(h0, n, t_max, rng=None):
    h = project_feasible(h0.copy(), n)
    best_c5 = compute_c5(h, n)
    best_h = h.copy()
    beta = 3.0
    lr = 0.05
    t_start = time.time()
    it = 0
    while time.time() - t_start < t_max:
        g = lse_gradient_vec(h, n, beta)
        h_new = h - lr * g
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new, n)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            lr = min(lr * 1.05, 0.3)
        else:
            lr *= 0.8
        h = h_new
        it += 1
        if it % 200 == 0:
            beta *= 1.5
            if beta > 200.0:
                beta = 3.0
            lr = 0.05
        if lr < 1e-7:
            beta = max(3.0, beta * 0.5)
            lr = 0.03
    return best_h, best_c5

def multi_start_binary(h0, n, t_max, rng):
    best_h, best_c5 = None, np.inf
    t_start = time.time()
    while time.time() - t_start < t_max:
        h_try = h0.copy()
        if rng.random() < 0.3:
            h_try = h0 + rng.normal(0, 0.3, n)
        h_b, c5_b = binary_hill_climb(h_try, n, n_sweeps=150, rng=rng)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
    return best_h, best_c5

def coordinate_polish(h, n, rng, n_sweeps=30):
    target = n / 2.0
    h = project_feasible(h.copy(), n)
    best = compute_c5(h, n)
    for _ in range(n_sweeps):
        improved = False
        order = rng.permutation(n)
        for j in order:
            old = h[j]
            for c in [0.0, 1.0, max(0.0, old - 0.1), min(1.0, old + 0.1),
                      max(0.0, old - 0.05), min(1.0, old + 0.05)]:
                if abs(c - old) < 1e-8:
                    continue
                h[j] = c
                h = project_feasible(h, n)
                v = compute_c5(h, n)
                if v < best - 1e-12:
                    best = v
                    improved = True
                    break
                else:
                    h[j] = old
                    h = project_feasible(h, n)
        if not improved:
            break
    return h, best

def _time_left(budget, t0):
    return max(0.0, budget - (time.time() - t0))
```
