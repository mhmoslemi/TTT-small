        if it % 100 == 0:
            c = compute_corr(h_cur)
            top_k = np.argsort(c)[::-1][:3]
            top_lags = [int(k - (n - 1)) for k in top_k]
            lr = max(0.001, 0.02 * (1.0 + it / 2000.0) ** (-0.7))
        sg = np.zeros(n)
        for lag in top_lags:
            sg += sg_for_lag(h_cur, lag, n)
        sg /= len(top_lags)
        h_new = project_feasible(h_cur - lr * sg, n)
        if it % 200 == 0:
            s_new = compute_score(h_new)
            if s_new < best_local - 1e-10:
                best_local = s_new
                h_best = h_new.copy()
                stalling = 0
            else:
                stalling += 1
            if stalling > 30:
                lr = max(0.0005, lr * 0.7)
                stalling = 0
        h_cur = h_new
    return h_best, best_local

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    best_h = h_init.copy()
    best_score = compute_score(h_init)

    # Phase 1: Projected subgradient on primary grid
    h_sub, s_sub = subgradient_minimize(best_h, n, 0.25, t0, budget_s)
    if s_sub < best_score:
        best_score = s_sub
        best_h = h_sub

    # Phase 2: SLSQP with soft-max then exact max
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def cons_sum(h):
        return np.sum(h) - target

    constraints = [{'type': 'eq', 'fun': cons_sum, 'jac': lambda h: np.ones(n)}]

    for r in range(10):
        if time.time() - t0 > budget_s * 0.5:
            break
        if r == 0:
            h_start = best_h.copy()
        elif r == 1:
            h_start = project_feasible(best_h + rng.randn(n) * 0.015, n)
        elif r == 2:
            h_start = project_feasible(best_h + rng.randn(n) * 0.06, n)
        elif r == 3:
            blen = max(3, n // 8)
            si = rng.randint(0, n - blen)
            h_start = best_h.copy()
            h_start[si:si + blen] = 1.0 - best_h[si:si + blen]
            h_start = project_feasible(h_start, n)
        elif r == 4:
            x = np.linspace(0, 2, n)
            h_start = 0.5 + 0.28 * np.sin(2 * np.pi * 2.7 * x + 0.4) + 0.18 * np.sin(2 * np.pi * 6.3 * x + 1.9)
            h_start = project_feasible(h_start, n)
        elif r == 5:
            h_start = (rng.rand(n) > 0.5).astype(float)
            h_start = project_feasible(h_start, n)
        elif r == 6:
            # Thue-Morse variant
            tm = np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)
            h_start = 0.5 + 0.35 * (2 * tm - 1)
            h_start = project_feasible(h_start, n)
        elif r == 7:
            # Block structure
            nb = rng.randint(4, 8)
            h_start = np.zeros(n)
            for b in range(nb):
                s = int(b * n / nb)
                e = int((b + 1) * n / nb)
                h_start[s:e] = rng.uniform(0.1, 0.9)
            h_start = project_feasible(h_start, n)
        else:
            h_start = project_feasible(0.5 + 0.3 * (2 * rng.rand(n) - 1), n)

        try:
            res = minimize(soft_max_score, h_start, method='SLSQP',
                           bounds=bounds, constraints=constraints,
                           options={'maxiter': 800, 'ftol': 1e-14, 'maxls': 30})
            h_c = project_feasible(res['x'], n)
            s_c = compute_score(h_c)
            if s_c < best_score:
                best_score = s_c
                best_h = h_c.copy()
        except Exception:
            pass

        if time.time() - t0 > budget_s * 0.55:
            break
        try:
            res2 = minimize(compute_score, best_h, method='SLSQP',
                           bounds=bounds, constraints=constraints,
                           options={'maxiter': 500, 'ftol': 1e-15, 'maxls': 30})
            h_c = project_feasible(res2['x'], n)
            s_c = compute_score(h_c)
            if s_c < best_score:
                best_score = s_c
                best_h = h_c.copy()
        except Exception:
            pass

    # Phase 3: Nelder-Mead with penalty (handles non-smoothness)
    if time.time() - t0 < budget_s * 0.7:
        def penalty_obj(h):
            s = compute_score(h)
            p = 500.0 * (np.sum(h) - target) ** 2
            return s + p

        for r in range(4):
            if time.time() - t0 > budget_s * 0.72:
                break
            h_start = best_h.copy() if r == 0 else project_feasible(best_h + rng.randn(n) * 0.04, n)
            try:
                res = minimize(penalty_obj, h_start, method='Nelder-Mead',
                               options={'maxiter': 8000, 'xatol': 1e-9, 'fatol': 1e-13, 'adaptive': True})
                h_c = project_feasible(res['x'], n)
                s_c = compute_score(h_c)
                if s_c < best_score:
                    best_score = s_c
                    best_h = h_c.copy()
            except Exception:
                pass

    # Phase 4: Subgradient pass on current best (refined)
    if time.time() - t0 < budget_s * 0.82:
        h_sub2, s_sub2 = subgradient_minimize(best_h, n, 0.12, t0, budget_s)
        if s_sub2 < best_score:
            best_score = s_sub2
            best_h = h_sub2

    # Phase 5: Try n=40 grid
    if time.time() - t0 < budget_s * 0.9:
        n40 = 40
        x_old = np.linspace(0, 2, n, endpoint=False)
        x40 = np.linspace(0, 2, n40, endpoint=False)
        h40 = np.interp(x40, x_old, best_h)
        h40 = project_feasible(h40, n40)
        t40 = n40 / 2.0
        bounds40 = [(0.0, 1.0)] * n40

        def cons40(h):
            return np.sum(h) - t40
        constr40 = [{'type': 'eq', 'fun': cons40, 'jac': lambda h: np.ones(n40)}]

        def sm40(h):
            dx40 = 2.0 / n40
            c = np.correlate(h, 1.0 - h, mode='full') * dx40
            cm = np.max(c)
            return cm + np.log1p(np.sum(np.exp(np.clip(350 * (c - cm), -50, 50)) - 1.0)) / 350.0

        for r in range(5):
            if time.time() - t0 > budget_s * 0.92:
                break
            h_s = h40.copy() if r == 0 else project_feasible(h40 + rng.randn(n40) * 0.05, n40)
            try:
                res = minimize(sm40, h_s, method='SLSQP',
                               bounds=bounds40, constraints=constr40,
                               options={'maxiter': 600, 'ftol': 1e-14})
                h_c = project_feasible(res['x'], n40)
                s_c = compute_score(h_c)
                if s_c < best_score:
                    best_score = s_c
                    best_h = h_c.copy()
            except Exception:
                pass
            if time.time() - t0 > budget_s * 0.93:
                break
            try:
                res2 = minimize(compute_score, best_h, method='SLSQP',
                                bounds=bounds40, constraints=constr40,
                                options={'maxiter': 400, 'ftol': 1e-15})
                h_c = project_feasible(res2['x'], n40)
                s_c = compute_score(h_c)
                if s_c < best_score:
                    best_score = s_c
                    best_h = h_c.copy()
            except Exception:
                pass

    # Phase 6: Coordinate descent final polish
    if time.time() - t0 < budget_s * 0.96:
        nf = len(best_h)
        tf = nf / 2.0
        no_improve = 0
        for it in range(20000):
            if time.time() - t0 > budget_s * 0.97:
                break
            if no_improve > 3000:
                break
            i, j = rng.choice(nf, size=2, replace=False)
            delta = rng.uniform(-0.015, 0.015)
            hi = np.clip(best_h[i] + delta, 0.0, 1.0)
            hj = np.clip(best_h[j] - delta, 0.0, 1.0)
            actual_move = (hi - best_h[i]) - (best_h[j] - hj)
            if abs(actual_move) > 1e-9:
                continue
            if abs(hi - best_h[i]) < 1e-14:
                continue
            old_i, old_j = best_h[i], best_h[j]
            best_h[i], best_h[j] = hi, hj
            s_try = compute_score(best_h)
            if s_try < best_score - 1e-12:
                best_score = s_try
                no_improve = 0
            else:
                best_h[i], best_h[j] = old_i, old_j
                no_improve += 1

    # Phase 7: Final SLSQP polish
    if time.time() - t0 < budget_s * 0.985:
        nf = len(best_h)
        tf = nf / 2.0
        bounds_f = [(0.0, 1.0)] * nf

        def cons_f(h):
            return np.sum(h) - tf
        try:
            res = minimize(compute_score, best_h, method='SLSQP',
                           bounds=bounds_f,
                           constraints=[{'type': 'eq', 'fun': cons_f, 'jac': lambda h: np.ones(nf)}],
                           options={'maxiter': 800, 'ftol': 1e-15, 'maxls': 20})
            h_f = project_feasible(res['x'], nf)
            s_f = compute_score(h_f)
            if s_f < best_score:
                best_score = s_f
                best_h = h_f
        except Exception:
            pass

    best_h = project_feasible(best_h, len(best_h))
    best_score = compute_score(best_h)
    return best_h, best_score, len(best_h)
