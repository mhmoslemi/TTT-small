def subgrad_step(h, n):
    """Get subgradient direction to decrease max correlation"""
    c = np.correlate(h, 1.0 - h, mode='full')
    k_star = np.argmax(c)
    # shift = k_star - (n-1)
    shift = k_star - (n - 1)

    grad = np.zeros(n)

    vm = (j_minus >= 0) & (j_minus < n)
    grad[vp] += (1.0 - h[j_plus[vp]])
    grad[vm] -= h[j_minus[vm]]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return c.max() * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if not free.any():
            free = np.ones(n, dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] = np.clip(h[idx] + diff, 0, 1)
    return h

def sum_eq(h):
    return h.sum() - len(h) / 2.0

def sum_eq_jac(h):
    return np.ones(len(h))

def subgrad_max(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    k_star = np.argmax(c)
    shift = k_star - (n - 1)
    m = np.arange(n)
    grad = np.zeros(n)
    jp = m + shift
    jm = m - shift
    vp = (jp >= 0) & (jp < n)
    vm = (jm >= 0) & (jm < n)
    grad[vp] += (1.0 - h[jp[vp]])
    grad[vm] -= h[jm[vm]]
    return grad

    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    return m + temp * np.log(np.sum(np.exp((c - m) / temp)))
