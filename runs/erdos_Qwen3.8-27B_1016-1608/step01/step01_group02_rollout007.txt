            best_h, best_c5 = hc.copy(), c5c

    # Phase 7: Final SLSQP + subgradient
    if remaining() > 10:
        hf = do_slsqp(best_h, best_n, maxiter=200)
        if hf is not None:
            c5f = compute_c5(hf, 2.0 / best_n)
            if c5f < best_c5:
                best_h, best_c5 = hf.copy(), c5f
        hf2, c5f2 = subgradient_polish(best_h, 2.0 / best_n, best_n / 2.0, max_iter=20000)
        if c5f2 < best_c5:
            best_h, best_c5 = hf2.copy(), c5f2

    best_h = projection(best_h, best_n / 2.0)
    final_c5 = compute_c5(best_h, 2.0 / best_n)
    return (best_h, final_c5, best_n)
