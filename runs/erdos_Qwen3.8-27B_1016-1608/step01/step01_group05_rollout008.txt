import numpy as np
from scipy.optimize import minimize
import time

_opt = {}

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def project_to_set(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        hp = np.clip(h - mid, 0.0, 1.0)
        if float(np.sum(hp)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def upsample_func(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, h)

def _obj_smooth(x):
    n = _opt['n']
    target = _opt['target']
    beta = _opt['beta']
    hp = project_to_set(np.clip(x, 0.0, 1.0), target)
    dx = 2.0 / n
    corr = np.correlate(hp, 1.0 - hp, mode="full") * dx
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -40.0, 40.0)
    return m + np.log(np.sum(np.exp(sh))) / beta

def _obj_mixed(x):
    """Max of top-3 lags: balances peak and near-peak."""
    n = _opt['n']
    target = _opt['target']
    hp = project_to_set(np.clip(x, 0.0, 1.0), target)
    dx = 2.0 / n
    corr = np.correlate(hp, 1.0 - hp, mode="full") * dx
    top3 = np.sort(corr)[-3:]
    return float(np.max(top3)) + 0.1 * float(np.mean(top3))

def lbfgs_opt(h0, n, beta, maxiter=300):
    target = n / 2.0
    _opt['n'] = n
    _opt['target'] = target
    _opt['beta'] = beta
    h0 = project_to_set(h0.copy(), target)
    bounds = [(0.0, 1.0)] * n
    res = minimize(_obj_smooth, h0, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12, 'maxls': 50})
    h_out = project_to_set(res.x.copy(), target)
    return h_out, compute_c5(h_out, n)

def lbfgs_mixed(h0, n, maxiter=200):

    h0 = project_to_set(h0.copy(), target)

    h = project_to_set(h.copy(), target)
    best = compute_c5(h, n)
    for sweep in range(n_sweeps):
        improved = False
        order = rng.permutation(n)
        for j in order:
            old = h[j]
            cands = [0.0, 1.0]
            for d in (0.02, 0.05, 0.1, 0.15, 0.25, 0.4, 0.6):
                cands.append(max(0.0, old - d))
                cands.append(min(1.0, old + d))
            cands.append(max(0.0, old * 0.5))
            cands.append(min(1.0, 1.0 - old * 0.5))
            cands = [c for c in cands if abs(c - old) > 1e-7]
            best_c = old
            best_v = best
            for c in cands:
                h[j] = c
                h = project_to_set(h, target)
                v = compute_c5(h, n)
                if v < best_v - 1e-13:
                    best_v = v
                    best_c = c
            if best_c != old:
                h[j] = best_c
                h = project_to_set(h, target)
                best = best_v
                improved = True
            else:
                h[j] = old
                h = project_to_set(h, target)
        if not improved:
            break
    return h, best

def pair_swap(h, n, rng, n_rounds=50):
    target = n / 2.0
    h = project_to_set(h.copy(), target)
    best = compute_c5(h, n)
    for _ in range(n_rounds):
        j, k = rng.choice(n, size=2, replace=False)
        if h[j] == h[k]:
            continue
        # Shift mass from one to the other
        delta = 0.1 * (h[j] - h[k])
        h2 = h.copy()
        h2[j] = h[j] - delta
        h2[k] = h[k] + delta
        h2 = project_to_set(np.clip(h2, 0, 1), target)
        v = compute_c5(h2, n)
        if v < best - 1e-13:
            h = h2
            best = v
    return h, best

def make_random_start(n, rng, binary_frac=0.7):
    h = np.zeros(n)
    nb = int(n * binary_frac)
    idx = rng.choice(n, size=nb, replace=False)
    h[idx] = rng.uniform(0.5, 1.0, nb)
    h += rng.uniform(-0.15, 0.15, n)
    return project_to_set(h, n / 2.0)

def make_two_level(n, rng, frac=0.55):
    """Two-level: value a on measure 1/frac of domain, 0 elsewhere."""
    a = 1.0 / frac
    a = min(a, 1.0)
    h = np.zeros(n)
    n_ones = int(n * frac)
    idx = rng.choice(n, size=n_ones, replace=False)
    h[idx] = a
    return project_to_set(h, n / 2.0)

def _time_left(budget, t0):
    return max(0.0, budget - (time.time() - t0))

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    budget = float(budget_s)

    best_h = None
    best_c5 = np.inf
    best_n = 0

    def try_update(h, n):
        nonlocal best_h, best_c5, best_n
        h = project_to_set(np.asarray(h, dtype=float), n / 2.0)
        c = compute_c5(h, n)
        if c < best_c5:
            best_h = h.copy()
            best_c5 = c
            best_n = n
            return True
        return False

    # Phase 1: n=90, start from parent
    n1 = 90
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        if len(h_init) != n1:
            h_init = upsample_func(h_init, n1)
    except Exception:
        h_init = make_two_level(n1, rng, 0.55)
    h_init = project_to_set(h_init, n1 / 2.0)
    try_update(h_init, n1)

    # L-BFGS-B annealing
    for beta in [5.0, 10.0, 30.0, 80.0, 200.0, 500.0]:
        if _time_left(budget, t0) < budget * 0.30:
            break
        h_opt, c5 = lbfgs_opt(h_init, n1, beta, maxiter=250)
        if try_update(h_opt, n1):
            h_init = h_opt.copy()
        # Quick coord descent between stages
        if best_n == n1:
            h_cd, _ = coord_descent(best_h, n1, rng, n_sweeps=15)
            try_update(h_cd, n1)
            if best_n == n1:
                h_init = best_h.copy()

    # Mixed objective
    if _time_left(budget, t0) > budget * 0.25:
        h_opt, c5 = lbfgs_mixed(best_h, best_n, maxiter=200)
        if try_update(h_opt, n1):
            h_init = h_opt.copy()

    # Aggressive coordinate descent
    if _time_left(budget, t0) > budget * 0.20:
        h_cd, _ = coord_descent(best_h, best_n, rng, n_sweeps=150)
        try_update(h_cd, best_n)

    # Pair swaps
    if _time_left(budget, t0) > budget * 0.18:
        h_ps, _ = pair_swap(best_h, best_n, rng, n_rounds=2000)
        try_update(h_ps, best_n)

    # Random restarts at n=90
    if _time_left(budget, t0) > budget * 0.15:
        for i in range(4):
            if _time_left(budget, t0) < budget * 0.10:
                break
            h_r = make_random_start(n1, rng)
            if i == 0:
                h_r = make_two_level(n1, rng, 0.50)
            elif i == 1:
                h_r = make_two_level(n1, rng, 0.52)
            h_opt, c5 = lbfgs_opt(h_r, n1, 50.0, maxiter=200)
            try_update(h_opt, n1)
            h_opt, c5 = lbfgs_opt(h_r, n1, 200.0, maxiter=150)
            try_update(h_opt, n1)
            h_cd, _ = coord_descent(h_opt, n1, rng, n_sweeps=60)
            try_update(h_cd, n1)

    # Phase 2: n=180 refinement
    if _time_left(budget, t0) > budget * 0.22:
        n2 = 180
        h_up = project_to_set(upsample_func(best_h, n2), n2 / 2.0)
        try_update(h_up, n2)
        for beta in [30.0, 80.0, 200.0, 400.0]:
            if _time_left(budget, t0) < budget * 0.10:
                break
            h_opt, c5 = lbfgs_opt(h_up, n2, beta, maxiter=200)
            if try_update(h_opt, n2):
                h_up = h_opt.copy()
        if _time_left(budget, t0) > budget * 0.08:
            h_cd, _ = coord_descent(best_h, best_n, rng, n_sweeps=50)
            try_update(h_cd, best_n)
        if _time_left(budget, t0) > budget * 0.06:
            h_ps, _ = pair_swap(best_h, best_n, rng, n_rounds=1000)
            try_update(h_ps, best_n)

    # Phase 3: n=360 if time
    if _time_left(budget, t0) > budget * 0.12:
        n3 = 360
        h_up = project_to_set(upsample_func(best_h, n3), n3 / 2.0)
        try_update(h_up, n3)
        for beta in [50.0, 150.0]:
            if _time_left(budget, t0) < budget * 0.04:
                break
            h_opt, c5 = lbfgs_opt(h_up, n3, beta, maxiter=150)
            if try_update(h_opt, n3):
                h_up = h_opt.copy()

    # Final polish
    if _time_left(budget, t0) > 5.0:
        h_cd, _ = coord_descent(best_h, best_n, rng, n_sweeps=60)
        try_update(h_cd, best_n)
        if _time_left(budget, t0) > 3.0:
            h_ps, _ = pair_swap(best_h, best_n, rng, n_rounds=2000)
            try_update(h_ps, best_n)

    best_h = project_to_set(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h, best_n)
    return best_h, best_c5, best_n
