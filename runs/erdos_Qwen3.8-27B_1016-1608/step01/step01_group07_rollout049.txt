        if time.time() - t0 > budget_s * 0.35:
            break
        h_cur = _fix_sum(np.clip(best_h + rng.randn(n) * 0.03, 0, 1))
        h_opt, c5_opt = _proj_grad(h_cur, 2000, t0, budget_s, binary_push=bp)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt

    # Phase 2: More gradient iterations without binary push
    for _ in range(3):
        if time.time() - t0 > budget_s * 0.55:
            break
        h_cur = _fix_sum(np.clip(best_h + rng.randn(n) * 0.02, 0, 1))
        h_opt, c5_opt = _proj_grad(h_cur, 3000, t0, budget_s, binary_push=0.0)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt

    # Phase 3: SLSQP polish at multiple alpha
    n_slsqp = len(best_h)
    for alpha in [300, 1000, 3000, 8000]:
        if time.time() - t0 > budget_s * 0.7:
            break
        h_opt, c5_opt = _slsqp_opt(best_h, alpha, 1200, t0, budget_s)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt

    # Phase 4: Try lower resolution (sometimes coarser is better)
    for n_try in [320, 256, 200]:
        if time.time() - t0 > budget_s * 0.82:
            break
        if n_try >= n:
            continue
        h_down = _interp(best_h, n_try)
        c5_down = _c5(h_down)
        h_opt, c5_opt = _proj_grad(h_down, 1500, t0, budget_s, binary_push=1.0)
        if c5_opt < best_c5:
            h_opt2, c5_opt2 = _slsqp_opt(h_opt, 2000, 800, t0, budget_s)
            if c5_opt2 < c5_opt:
                h_opt, c5_opt = h_opt2, c5_opt2
            best_h = h_opt
            best_c5 = c5_opt
        elif c5_down < best_c5:
            best_h = h_down
            best_c5 = c5_down

    # Phase 5: Try higher resolution
    for n_try in [512, 640]:
        if time.time() - t0 > budget_s * 0.9:
            break
        if n_try <= len(best_h):
            continue
        h_up = _interp(best_h, n_try)
        c5_up = _c5(h_up)
        h_opt, c5_opt = _proj_grad(h_up, 1500, t0, budget_s, binary_push=0.5)
        h_opt2, c5_opt2 = _slsqp_opt(h_opt, 1000, 600, t0, budget_s)
        if c5_opt2 < c5_opt:
            h_opt, c5_opt = h_opt2, c5_opt2
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt

    # Phase 6: Final fine-tune
    if time.time() - t0 < budget_s - 30:
        h_opt, c5_opt = _proj_grad(best_h, 5000, t0, budget_s, binary_push=0.3)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
        h_opt, c5_opt = _slsqp_opt(best_h, 10000, 800, t0, budget_s)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt

    best_h = _fix_sum(best_h)
    best_c5 = _c5(best_h)
    return best_h, best_c5, len(best_h)
