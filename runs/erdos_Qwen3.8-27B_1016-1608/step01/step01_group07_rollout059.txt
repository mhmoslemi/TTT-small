import numpy as np
from scipy.optimize import minimize
import time

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _fix(h, n):
    target = 0.5 * n
    for _ in range(200):
        d = target - np.sum(h)
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        k = int(np.sum(mask))
        if k == 0:
            break
        h = np.where(mask, h + d / k, h)
        h = np.clip(h, 0.0, 1.0)
    return h

def _project(h, n):
    target = 0.5 * n
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def _obj_lse(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(c)
    return m + np.log(np.sum(np.exp(np.clip(c - m, -60, 60)))) / alpha

def _slsqp(h0, n, alpha, maxiter, t0, budget):
    if budget - (time.time() - t0) < 10:
        return None
    bnds = [(0.0, 1.0)] * n
    cons = [
        {'type': 'eq', 'fun': lambda x: np.sum(x) - 0.5*n, 'jac': lambda x: np.ones(n)}
    ]
    try:
        res = minimize(_obj_lse, h0.copy(), (alpha,), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None:
            return _fix(np.clip(res.x, 0, 1), n)
    except:
        pass
    return None

    t0 = time.time()

        return time.time() - t0

    n = len(h)
    h = _fix(h, n)
    best_h = h.copy()
    best_c = _c5(h)
    
    # Phase 1: Subgradient descent (fast, many iterations)
    dx = 2.0 / n
    for it in range(5000):
        if elapsed() > budget_s * 0.2:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        kstar = int(np.argmax(c)) - (n - 1)
        # Gradient of f_{kstar}
        g = np.zeros(n)
        if kstar >= 0:
            g[:n-kstar] += 1.0 - h[kstar:]
            g[kstar:] += -h[:n-kstar]
        else:
            m = -kstar
            g[:n-m] += -h[m:]
            g[m:] += 1.0 - h[:n-m]
        eta = 0.5 / (1.0 + 0.1 * it)
        h = _project(h - eta * g * dx, n)
        c_now = _c5(h)
        if c_now < best_c:
            best_c = c_now
            best_h = h.copy()
    
    h = _fix(h, n)
    best_h = h.copy()
    best_c = _c5(h)
    
    # Phase 2: SLSQP with increasing alpha

        for r in range(3):
