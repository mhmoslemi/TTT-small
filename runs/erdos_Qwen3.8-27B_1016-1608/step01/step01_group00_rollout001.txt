def optimize_direct(h0, n, maxiter):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint}]
    try:
        res = minimize(lambda x: compute_c5(x), h0, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_out = repair(np.clip(res.x, 0.0, 1.0), n)
        if compute_c5(h_out) < compute_c5(h0):
            return h_out
    except Exception:
        pass
    return h0

def optimize_active(h0, n, k_top=5, maxiter=300):
    dx = 2.0 / n
    corr = np.correlate(h0, 1.0 - h0, mode='full') * dx
    top = np.argsort(corr)[-k_top:]
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        return np.mean(c[top])
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint}]
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        h_out = repair(np.clip(res.x, 0.0, 1.0), n)
        if compute_c5(h_out) < compute_c5(h0):
            return h_out
    except Exception:
        pass
    return h0

def nm_local(h0, n):
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(lambda x: compute_c5(x), h0, method='Nelder-Mead',
                       bounds=bnds, options={'maxiter': 30000, 'xatol': 1e-13, 'fatol': 1e-15})
        h_out = repair(np.clip(res.x, 0.0, 1.0), n)
        if compute_c5(h_out) < compute_c5(h0):
            return h_out
    except Exception:
        pass
    return h0

def random_binary_init(n, rng):
    h = np.zeros(n)
    targets = np.random.choice(n, size=n // 2, replace=False)
    h[targets] = 1.0
    return h + rng.normal(0, 0.1, n)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_best = np.full(70, 0.5)
    n = len(h_best)
    c5_best = compute_c5(h_best)

    # Phase 1: Coarse-to-fine smooth SLSQP
    if time.time() - t0 < budget_s * 0.12:
        for alpha in [30.0, 80.0, 200.0]:
            h = optimize_smooth(h_best, n, alpha, maxiter=1000)
            c = compute_c5(h)
            if c < c5_best:
                h_best, c5_best = h, c

    # Phase 2: Active-set refinement (multiple rounds)
    if time.time() - t0 < budget_s * 0.28:
        for _ in range(5):
            if time.time() - t0 > budget_s * 0.28:
                break
            for kt in [10, 5, 3]:
                h = optimize_active(h_best, n, k_top=kt, maxiter=250)
                c = compute_c5(h)
                if c < c5_best:
                    h_best, c5_best = h, c

    # Phase 3: Adaptive perturbation + smooth optimize
    scale = 0.06
    while time.time() - t0 < budget_s * 0.62:
        h_p = repair(h_best + rng.randn(n) * scale, n)
        h = optimize_smooth(h_p, n, alpha=80.0, maxiter=250)
        c = compute_c5(h)
        if c < c5_best:
            h_best, c5_best = h, c
            scale = min(0.25, scale * 1.2)
        else:
            scale = max(0.003, scale * 0.88)

    # Phase 4: Random restart with binary-like initializations
    if time.time() - t0 < budget_s * 0.72:
        for _ in range(5):
            if time.time() - t0 > budget_s * 0.72:
                break
            h_r = repair(random_binary_init(n, rng), n)
            h_r = optimize_smooth(h_r, n, alpha=60.0, maxiter=400)
            c = compute_c5(h_r)
            if c < c5_best:
                h_best, c5_best = h_r, c

    # Phase 5: Direct SLSQP polish
    if time.time() - t0 < budget_s * 0.80:
        h = optimize_direct(h_best, n, maxiter=2000)
        c = compute_c5(h)
        if c < c5_best:
            h_best, c5_best = h, c

    # Phase 6: Nelder-Mead local polish
    if time.time() - t0 < budget_s * 0.89:
        h = nm_local(h_best, n)
        c = compute_c5(h)
        if c < c5_best:
            h_best, c5_best = h, c

    # Phase 7: Higher resolution interpolation
    if time.time() - t0 < budget_s * 0.96:
        for n2 in [100, 140]:
            if time.time() - t0 > budget_s * 0.95:
                break
            xp = np.linspace(0.0, 1.0, n)
            xn = np.linspace(0.0, 1.0, n2)
            h_i = np.interp(xn, xp, h_best)
            h_i = repair(h_i, n2)
            h2 = optimize_smooth(h_i, n2, alpha=60.0, maxiter=500)
            c2 = compute_c5(h2)
            if c2 < c5_best:
                h_best, c5_best, n = h2, c2, n2

    # Final validation
    h_best = repair(h_best, n)
    c5_best = compute_c5(h_best)
    return (h_best, c5_best, n)
