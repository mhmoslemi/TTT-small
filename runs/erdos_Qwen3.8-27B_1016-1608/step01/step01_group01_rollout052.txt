def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return c.max() * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float, copy=True)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if mask.sum() == 0:
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_shift(h, shift):
    n = len(h)
    grad = np.zeros(n)
    i = np.arange(n)
    j1 = i - shift
    j2 = i + shift
    m1 = (j1 >= 0) & (j1 < n)
    m2 = (j2 >= 0) & (j2 < n)
    grad[m1] += (1.0 - h[j1[m1]])
    grad[m2] -= h[j2[m2]]
    return grad

def grad_top_m(h, m=5):
    c = get_corr(h)
    top = np.argsort(c)[-m:]
    g = np.zeros(len(h))
    n = len(h)

        g += grad_shift(h, k - (n - 1))
    return g
