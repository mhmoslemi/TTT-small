    """Smooth max approximation of C5 and its gradient."""
    n = len(h)

    # lag = m - (n-1), range [-(n-1), n-1]
    # d(corr_m)/dh[i] = (1-h[i-lag])*1{i-lag in [0,n-1]} - h[i+lag]*1{i+lag in [0,n-1]}

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_c5_grad(h, dx, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    A = corr * dx
    max_A = np.max(A)
    exps = np.exp(tau * (A - max_A))
    sum_exps = np.sum(exps)
    f = max_A + np.log(sum_exps) / tau
    w = exps / sum_exps
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m] * dx
        if wm < 1e-18:
            continue
        lag = m - (n - 1)
        if lag == 0:
            grad += wm * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += wm * (1.0 - h[:n - lag])
            grad[:n - lag] -= wm * h[lag:]
        else:
            s = -lag
            grad[:n - s] += wm * (1.0 - h[s:])
            grad[s:] -= wm * h[:n - s]
    return f, grad

def pen_obj(x, dx, tau, lam, target):
    f, g = smooth_c5_grad(x, dx, tau)
    p = x.sum() - target
    return f + lam * p * p

def pen_grad(x, dx, tau, lam, target):
    f, g = smooth_c5_grad(x, dx, tau)
    p = x.sum() - target
    return g + 2.0 * lam * p

def project(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    for _ in range(300):
        excess = h.sum() - target
        if abs(excess) < 1e-14:
            break
        if excess > 0:
            mask = h > 0
            cnt = mask.sum()
            if cnt == 0:
                break
            s = min(excess / cnt, h[mask].min())
            h[mask] -= s
        else:
            mask = h < 1
            cnt = mask.sum()
            if cnt == 0:
                break
            s = min(-excess / cnt, (1.0 - h[mask]).min())
            h[mask] += s
    return h

def slsqp_obj(x, dx):
    return compute_c5(x, dx)

def slsqp_con(x, target):
    return x.sum() - target

def try_lbfgsb(h0, dx, target, tau, lam, bounds, maxiter=400):

                       jac=pen_grad, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
        return project(res.x, target)
    except Exception:
        return None

def try_slsqp(h0, dx, target, bounds, maxiter=800):

                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': slsqp_con, 'args': (target,)}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project(res.x, target)
    except Exception:
        return None

    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False) + 1.0 / n_new
    return np.interp(x_new, x_old, h)
