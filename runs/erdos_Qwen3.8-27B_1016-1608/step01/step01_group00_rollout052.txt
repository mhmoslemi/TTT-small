import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        mask = (h < 1.0 - 1e-13) if diff > 0 else (h > 1e-13)
        if not np.any(mask):
            h[:] = target / n
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def sum_con(x):
    return np.sum(x) - len(x) / 2.0

def slsqp_opt(h0, n, maxiter=500, ftol=1e-14):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(compute_c5, h0, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None:
            h = repair(np.clip(res.x, 0.0, 1.0), n)
            if compute_c5(h) <= compute_c5(h0) + 1e-12:
                return h
    except Exception:
        pass
    return h0

def grad_sub(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    max_c = np.max(corr)
    tol = max(n * 1e-16, 1e-18)
    active = np.where(corr >= max_c - tol)[0]
    grad = np.zeros(n)
    for k in active:
        m = k - (n - 1)
        if m == 0:
            grad += 1.0 - 2.0 * h
        elif m > 0:
            grad[m:] += (1.0 - h[:n-m])
            if m < n:
                grad[:n-m] -= h[m:n]
        else:
            mp = -m
            grad[:n-mp] += (1.0 - h[mp:n])
            grad[mp:n] -= h[:n-mp]
    grad /= len(active)
    return grad * dx

def make_perturbation(n, scale, kind, rng):
    if kind == 0:  # Gaussian
        return rng.standard_normal(n) * scale
    elif kind == 1:  # Sinusoidal
        freq = rng.randint(1, n//4 + 1)
        phase = rng.uniform(0, 2*np.pi)
        x = np.arange(n)
        return scale * np.sin(2*np.pi*freq*x/n + phase)
    elif kind == 2:  # Block
        p = np.zeros(n)
        nblocks = rng.randint(1, 5)
        for _ in range(nblocks):
            s = rng.randint(n)
            l = rng.randint(5, n//3)
            p[s:s+l] = rng.choice([-1, 1]) * rng.uniform(0.3, 1.0)
        return p
    elif kind == 3:  # Smooth Gaussian
        center = rng.randint(0, n)
        width = rng.uniform(3, n/3)
        x = np.arange(n)
        return scale * np.exp(-0.5*((x-center)/width)**2) * rng.choice([-1, 1])
    else:
        return rng.standard_normal(n) * scale
