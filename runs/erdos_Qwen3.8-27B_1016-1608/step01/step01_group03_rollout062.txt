def coordinate_step(h, i, n):
    """Find optimal h[i] given all other coordinates fixed."""
    dx = 2.0 / n
    # Compute coeff[m] and const[m] for all correlation indices m
    c_old = np.correlate(h, 1.0 - h, mode="full")
    
    # For each m, compute the coefficient of h[i] in c[m]
    coeff = np.zeros(2*n - 1)
    for m in range(2*n - 1):
        # h[i] as first factor: j=i, need 0<=i<n (always true) and 0<=i+m-n+1<n
        lo = max(0, m - n + 1)
        hi = min(n - 1, m + n - 2)
        c = 0.0
        if lo <= i <= hi:
            j2 = i + m - n + 1
            if 0 <= j2 < n:
                c += (1.0 - h[j2])
        # h[i] in (1-h[...]) part: j+m-n+1=i → j=i-m+n-1
        j1 = i - m + n - 1
        if 0 <= j1 < n and lo <= j1 <= hi:
            c -= h[j1]
        coeff[m] = c
    
    # const[m] = c_old[m] - coeff[m] * h[i]
    const = c_old - coeff * h[i]
    
    # Find v in [0,1] minimizing max_m (coeff[m]*v + const[m])
    best_v = h[i]
    best_val = np.max(c_old)
    for v in np.concatenate([np.linspace(0, 1, 100), [h[i]]]):
        vals = coeff * v + const
        val = np.max(vals)
        if val < best_val:
            best_val = val
            best_v = v
    return best_v
