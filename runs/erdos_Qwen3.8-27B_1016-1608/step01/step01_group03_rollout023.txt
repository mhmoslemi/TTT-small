    k = int(np.argmax(corr))
    return k, float(corr[k]) * dx

def compute_grad(h, dx, k):
    n = len(h)
    shift = k - (n - 1)
    idx = np.arange(n)
    grad = np.zeros(n)
    v1 = (idx + shift)
    m1 = (v1 >= 0) & (v1 < n)
    grad[m1] += (1.0 - h[v1[m1]])
    v2 = (idx - shift)
    m2 = (v2 >= 0) & (v2 < n)
    grad[m2] -= h[v2[m2]]
    return grad * dx

def project_simplex_box(x, target, lo=0.0, hi=1.0):
    """Project x onto {y: lo≤y_i≤hi, sum(y)=target}"""
    n = len(x)
    lo_v = np.full(n, lo)
    hi_v = np.full(n, hi)
    # Bisection on lambda
    lo_l, hi_l = x.min() - 1.0, x.max() + 1.0
    for _ in range(100):
        mid = (lo_l + hi_l) / 2
        y = np.clip(x - mid, lo, hi)
        s = y.sum()
        if s > target:
            lo_l = mid
        else:
            hi_l = mid
    y = np.clip(x - (lo_l + hi_l) / 2, lo, hi)
    # Final correction
    d = target - y.sum()
    if abs(d) > 1e-14:
        # Distribute remaining
        if d > 0:
            m = y < hi - 1e-14
        else:
            m = y > lo + 1e-14
        if m.sum() > 0:
            y[m] += d / m.sum()
            y = np.clip(y, lo, hi)
    return y

                b[i] = 1 - b[i]
                # Check sum constraint: need n/2 zeros

                    partner = rng.randint(n)
                    if partner == i:
                        continue
                    b[i], b[partner] = b[partner], b[i]
                # Evaluate
                h_new = np.where(b == 0, a, 1.0 - a)
                c5_new = compute_c5(h_new, dx)

                    best_h = h_new.copy()
                    best_a = a

        # (We'll do this in a refinement step)

from scipy.optimize import minimize
import time

_DX = 0.0
_TARGET = 0.0

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def project_feasible(x, target):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(x - mid, 0, 1)) > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - (lo+hi)*0.5, 0, 1)
    d = target - y.sum()
    if abs(d) > 1e-14:
        m = (y < 1-1e-13) if d > 0 else (y > 1e-13)
        if m.any():
            y[m] += d / m.sum()
            y = np.clip(y, 0, 1)
    return y

def _slsqp_obj(x):
    return compute_c5(x, _DX)

def _slsqp_eq(x):
    return np.sum(x) - _TARGET

def grad_max_corr(h, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    k = int(np.argmax(corr))
    shift = k - (n - 1)
    idx = np.arange(n)
    g = np.zeros(n)
    v1 = idx + shift
    m1 = (v1 >= 0) & (v1 < n)
    g[m1] += (1.0 - h[v1[m1]])
    v2 = idx - shift
    m2 = (v2 >= 0) & (v2 < n)
    g[m2] -= h[v2[m2]]
    return g * dx

                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        h = project_feasible(r.x, target)
        return h, compute_c5(h, dx)
