import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)

    shifted = (corr - m) / temp
    return float(m + temp * np.log(np.mean(np.exp(shifted))))

def project_to_constraint(h, target):
    """Project h onto {x: 0<=x<=1, sum(x)=target}"""
    h = np.clip(h, 0.0, 1.0)

        if abs(excess) < 1e-13:

        else:
