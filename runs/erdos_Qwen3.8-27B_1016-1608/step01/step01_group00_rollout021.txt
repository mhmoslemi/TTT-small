                    sg_min = np.min(coeff[active])
                    sg_max = np.max(coeff[active])
                    if sg_min <= 0.0 and sg_max >= 0.0:
                        t_lo = t_hi = t_mid
                        break
                    elif sg_max < 0.0:
                        t_hi = t_mid
                    else:
                        t_lo = t_mid
            t_new = 0.5 * (t_lo + t_hi)
            g_new = np.max(d + coeff * t_new)
            if np.max(d) < g_new:
                t_new = 0.0
            if np.max(d + coeff) < g_new:
                t_new = 1.0
            if abs(t_new - h[i]) > 1e-14:
                delta = t_new - h[i]
                corr += delta * coeff
                h[i] = t_new
                improved = True
        if not improved and sweep > 0:
            break
    return h, float(np.max(corr))

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_best = np.full(70, 0.5)
    n = len(h_best)
    h_best = repair(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: Thorough coordinate descent
    h_best, _ = coord_desc(h_best, n, max_sweeps=300)
    h_best = repair(h_best, n)
    c5_best = compute_c5(h_best)

    # Also try complement as a different landscape
    if time.time() - t0 < budget_s * 0.05:
        h_comp = 1.0 - h_best
        h_comp, _ = coord_desc(h_comp, n, max_sweeps=100)
        h_comp = repair(h_comp, n)
        c_comp = compute_c5(h_comp)
        if c_comp < c5_best:
            h_best, c5_best = h_comp, c_comp

    # Phase 2: Main search with diverse perturbations
    scales = [0.02, 0.04, 0.07, 0.12, 0.2, 0.35, 0.55]
    it = 0
    no_improve = 0
    while time.time() - t0 < budget_s * 0.65:
        scale = scales[it % len(scales)]
        if no_improve > 12:
            scale = min(0.9, scale * 2.0)

        pert_type = it % 4
        if pert_type == 0:
            h_p = h_best + np.random.randn(n) * scale
        elif pert_type == 1:
            h_p = h_best.copy()
            idx = np.random.choice(n, size=max(1, n // 5), replace=False)
            h_p[idx] += np.random.randn(len(idx)) * scale * 2
        elif pert_type == 2:
            h_p = h_best.copy()
            block_size = max(2, n // 8)
            start = np.random.randint(0, n - block_size)
            h_p[start:start + block_size] += np.random.randn(block_size) * scale
        else:
            h_p = h_best + np.random.randn(n) * scale
            h_p[np.random.choice(n, size=n // 3, replace=False)] = np.clip(
                h_p[np.random.choice(n, size=n // 3, replace=False)] + np.random.randn(n // 3) * scale * 0.5, 0, 1)

        h_p = repair(h_p, n)
        h_c, _ = coord_desc(h_p, n, max_sweeps=40, random_order=True)
        h_c = repair(h_c, n)
        c_c = compute_c5(h_c)

        if c_c < c5_best - 1e-15:
            h_best, c5_best = h_c, c_c
            no_improve = 0
        else:
            no_improve += 1
        it += 1

    # Phase 3: Moderate polish
    if time.time() - t0 < budget_s * 0.78:
        h_best, _ = coord_desc(h_best, n, max_sweeps=500)
        h_best = repair(h_best, n)
        c5_best = compute_c5(h_best)

    # Phase 4: Fine perturbations
    while time.time() - t0 < budget_s * 0.90:
        for sc in [0.003, 0.008, 0.015, 0.03, 0.06]:
            if time.time() - t0 > budget_s * 0.89:
                break
            h_p = h_best + np.random.randn(n) * sc
            h_p = repair(h_p, n)
            h_c, _ = coord_desc(h_p, n, max_sweeps=60, random_order=True)
            h_c = repair(h_c, n)
            c_c = compute_c5(h_c)
            if c_c < c5_best:
                h_best, c5_best = h_c, c_c

    # Phase 5: Try complement again
    if time.time() - t0 < budget_s * 0.92:
        h_comp = 1.0 - h_best
        h_comp, _ = coord_desc(h_comp, n, max_sweeps=200)
        h_comp = repair(h_comp, n)
        c_comp = compute_c5(h_comp)
        if c_comp < c5_best:
            h_best, c5_best = h_comp, c_comp

    # Phase 6: Final deep polish
    if time.time() - t0 < budget_s * 0.98:
        h_best, _ = coord_desc(h_best, n, max_sweeps=2000)
        h_best = repair(h_best, n)
        c5_best = compute_c5(h_best)

    return (h_best, c5_best, n)
