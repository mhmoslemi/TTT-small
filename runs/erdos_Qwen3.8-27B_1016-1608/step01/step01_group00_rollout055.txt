        c1 = np.correlate(h_w, 1.0 - h_w, mode='full').astype(float)
        h_w[i] = sv
        if rng is not None and rng.random() < noise_prob:
            h_w[i] = rng.random()
        else:
            h_w[i] = coord_opt_t(c0, c1 - c0)
    return h_w

def _sum_constr(x):
    return np.sum(x) - len(x) / 2.0

def slsqp(h, n, maxiter=500):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_constr}]
    try:
        res = minimize(compute_c5, h, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        c_new = compute_c5(np.clip(res.x, 0.0, 1.0))
        c_old = compute_c5(h)
        if c_new <= c_old + 1e-15:
            return repair(np.clip(res.x, 0.0, 1.0), n)
    except Exception:
        pass
    return h

def nm_polish(h, n):
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(compute_c5, h, method='Nelder-Mead', bounds=bnds,
                       options={'maxiter': 30000, 'xatol': 1e-14, 'fatol': 1e-16})
        c_new = compute_c5(np.clip(res.x, 0.0, 1.0))
        if c_new < compute_c5(h) + 1e-15:
            return repair(np.clip(res.x, 0.0, 1.0), n)
    except Exception:
        pass
    return h

def block_swap(h, n, rng):
    h = h.copy()
    i = rng.randint(0, n - 5)
    j = rng.randint(0, n - 5)
    ln = rng.randint(3, 8)
    if i + ln >= n or j + ln >= n:
        return h
    h[i:i+ln], h[j:j+ln] = h[j:j+ln].copy(), h[i:i+ln].copy()
    return repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_best = np.full(70, 0.5)
    n = len(h_best)
    h_best = repair(h_best, n)
    c5_best = compute_c5(h_best)

    def elapsed():
        return time.time() - t0

    # Phase 1: Pure coordinate descent convergence (0–12%)
    t1 = t0 + budget_s * 0.12
    for _ in range(1000):
        if elapsed() > t1:
            break
        h_c = repair(cd_pass(h_best, n, 0.0, None), n)
        c = compute_c5(h_c)
        if c < c5_best - 1e-14:
            h_best, c5_best = h_c, c

    # Phase 2: Diversified search (12–65%)
    t2 = t0 + budget_s * 0.65
    scales = [0.003, 0.007, 0.015, 0.03, 0.06, 0.12, 0.25, 0.5]
    it = 0
    st = 0
    while elapsed() < t2:
        sc = scales[it % len(scales)]
        r = rng.random()
        if r < 0.4:
            h_p = repair(h_best + rng.standard_normal(n) * sc, n)
        elif r < 0.7:
            h_p = h_best.copy()
            k = rng.randint(2, 10)
            idx = rng.choice(n, size=k, replace=False)
            h_p[idx] = 1.0 - h_p[idx]
            h_p = repair(h_p, n)
        else:
            h_p = block_swap(h_best, n, rng)

        for inner in range(8):
            if elapsed() > t2:
                break
            h_c = repair(cd_pass(h_p, n, 0.02, rng), n)
            c = compute_c5(h_c)
            if c < c5_best - 1e-13:
                h_best, c5_best = h_c, c
            h_p = h_c
            st += 1
        if it % 15 == 0 and elapsed() < t2 - 20:
            h_s = slsqp(h_best, n, maxiter=400)
            c = compute_c5(h_s)
            if c < c5_best:
                h_best, c5_best = h_s, c
        it += 1

    # Phase 3: SLSQP deep polish (65–80%)
    t3 = t0 + budget_s * 0.80
    imp = True
    while imp and elapsed() < t3:
        imp = False
        h_s = slsqp(h_best, n, maxiter=5000)
        c = compute_c5(h_s)
        if c < c5_best - 1e-15:
            h_best, c5_best = h_s, c
            imp = True

    # Phase 4: Nelder-Mead polish (80–88%)
    if elapsed() < budget_s * 0.88:
        h_nm = nm_polish(h_best, n)
        c = compute_c5(h_nm)
        if c < c5_best:
            h_best, c5_best = h_nm, c

    # Phase 5: Resolution exploration (88–95%)
    t5 = t0 + budget_s * 0.95
    for n2 in [50, 60, 80, 90, 100, 120]:
        if elapsed() > t5:
            break
        if n2 % 2 != 0 or n2 == n:
            continue
        xp = np.linspace(0.0, 1.0, n, endpoint=False)
        xn = np.linspace(0.0, 1.0, n2, endpoint=False)
        h_i = repair(np.interp(xn, xp, h_best), n2)
        for _ in range(60):
            if elapsed() > t5:
                break
            h_c = repair(cd_pass(h_i, n2, 0.01, rng), n2)
            c2 = compute_c5(h_c)
            if c2 < c5_best - 1e-13:
                h_best, c5_best, n = h_c, c2, n2
            h_i = h_c

    # Phase 6: Final tight CD + SLSQP (95–99.5%)
    while elapsed() < budget_s * 0.995:
        h_c = repair(cd_pass(h_best, n, 0.0, None), n)
        c = compute_c5(h_c)
        if c < c5_best - 1e-15:
            h_best, c5_best = h_c, c
        if elapsed() < budget_s * 0.99:
            h_s = slsqp(h_best, n, maxiter=8000)
            c = compute_c5(h_s)
            if c < c5_best:
                h_best, c5_best = h_s, c
        else:
            break

    h_best = repair(h_best, n)
    c5_best = compute_c5(h_best)
    return (h_best, c5_best, n)
