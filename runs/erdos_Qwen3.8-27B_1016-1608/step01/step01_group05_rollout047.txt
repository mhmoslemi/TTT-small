        noise_scale = 0.08 if rng.rand() < 0.5 else 0.2
        hr = state['h'] + rng.normal(0, noise_scale, n1)
        hr = project_to_set(hr, n1 / 2.0)
        ho, _ = anneal_slsqp(hr, n1, [10.0, 30.0, 80.0, 150.0], 150)
        update_state(state, ho, n1)

    # Random binary starts
    while time_left(budget, t0) > budget * 0.65:
        hr = make_random_binary(n1, rng)
        hr += rng.normal(0, 0.05, n1)
        hr = project_to_set(hr, n1 / 2.0)
        ho, _ = slsqp_smooth(hr, n1, 20.0, 300)
        update_state(state, ho, n1)
        if time_left(budget, t0) > budget * 0.62:
            ho, _ = slsqp_smooth(ho, n1, 80.0, 200)
            update_state(state, ho, n1)

    # Smooth random starts
    while time_left(budget, t0) > budget * 0.72:
        hr = make_smooth_init(n1, rng)
        ho, _ = anneal_slsqp(hr, n1, [10.0, 40.0], 150)
        update_state(state, ho, n1)

    # Coordinate search polish at n=90
    if time_left(budget, t0) > budget * 0.55 and state['n'] == n1:
        hc, _ = coord_search(state['h'], n1, rng, 50)
        update_state(state, hc, n1)

    # Phase 2: n=180
    if time_left(budget, t0) > budget * 0.3:
        n2 = 180
        hu = project_to_set(upsample(state['h'], n2), n2 / 2.0)
        update_state(state, hu, n2)
        if time_left(budget, t0) > budget * 0.22:
            ho, _ = anneal_slsqp(hu, n2, [10.0, 30.0, 80.0], 150)
            update_state(state, ho, n2)
        if time_left(budget, t0) > budget * 0.18:
            ho, _ = slsqp_smooth(state['h'], n2, 150.0, 200)
            update_state(state, ho, n2)
        if time_left(budget, t0) > budget * 0.14 and state['n'] <= n2:
            hc, _ = coord_search(state['h'], state['n'], rng, 25)
            update_state(state, hc, state['n'])

    # Phase 3: n=360
    if time_left(budget, t0) > budget * 0.15:
        n3 = 360
        hu = project_to_set(upsample(state['h'], n3), n3 / 2.0)
        update_state(state, hu, n3)
        if time_left(budget, t0) > budget * 0.1:
            ho, _ = anneal_slsqp(hu, n3, [20.0, 60.0], 100)
            update_state(state, ho, n3)

    # Final coordinate polish
    if time_left(budget, t0) > 15.0 and state['n'] <= 180:
        hc, _ = coord_search(state['h'], state['n'], rng, 40)
        update_state(state, hc, state['n'])

    # One more SLSQP refinement at best resolution
    if time_left(budget, t0) > 10.0 and state['n'] <= 180:
        ho, _ = slsqp_smooth(state['h'], state['n'], 200.0, 100)
        update_state(state, ho, state['n'])

    best_h = project_to_set(state['h'], state['n'] / 2.0)
    best_c5 = compute_c5(best_h, state['n'])
    return best_h, best_c5, state['n']
