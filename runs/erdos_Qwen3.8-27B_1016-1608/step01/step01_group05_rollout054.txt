import numpy as np
from scipy.optimize import minimize
import time

_budget = 1100.0
_start_t = 0.0

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_set(h, target):
    lo, hi = -1.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        hp = np.clip(h - mid, 0.0, 1.0)
        if float(np.sum(hp)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def upsample(h, n_new):
    x_old = np.linspace(0.0, 2.0, len(h))
    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, h)

def time_left():
    return max(0.0, _budget - (time.time() - _start_t))

def fix_binary_count(h, n):
    target = n // 2
    while int(np.sum(h)) > target:
        idx = np.where(h == 1)[0]
        h[idx[np.random.randint(len(idx))]] = 0
    while int(np.sum(h)) < target:
        idx = np.where(h == 0)[0]
        h[idx[np.random.randint(len(idx))]] = 1
    return h

def sa_binary(h, n, rng, n_iters=20000, t0=0.4, t1=0.004):
    h = (h > 0.5).astype(float).copy()
    h = fix_binary_count(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    cur_c5 = best_c5
    for it in range(n_iters):
        t = t0 * (t1 / t0) ** (it / max(n_iters, 1))
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        nc = compute_c5(h, n)
        d = nc - cur_c5
        if d < 0 or rng.random() < np.exp(-d / max(t, 1e-12)):
            cur_c5 = nc
            if nc < best_c5:
                best_c5 = nc
                best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
    return best_h, best_c5

def coord_descent(h, n, rng, n_sweeps=15, step_init=0.12):
    target = n / 2.0
    h = project_to_set(h.copy(), target)
    best = compute_c5(h, n)
    step = step_init
    for _ in range(n_sweeps):
        order = rng.permutation(n)
        improved = False
        for j in order:
            old = h[j]
            cands = [0.0, 1.0, max(0.0, old - step), min(1.0, old + step),
                     max(0.0, old - 0.4 * step), min(1.0, old + 0.4 * step)]
            bc, bv = old, best
            for c in cands:
                if abs(c - old) < 1e-9:
                    continue
                h[j] = c
                h = project_to_set(h, target)
                v = compute_c5(h, n)
                if v < bv:
                    bv, bc = v, c
            if bc != old:
                h[j] = bc
                h = project_to_set(h, target)
                best = bv
                improved = True
            else:
                h[j] = old
                h = project_to_set(h, target)
        if not improved:
            step *= 0.6
            if step < 0.008:
                break
    return h, best

def slsqp_opt(h0, n, maxiter=300):
    target = n / 2.0
    h0 = project_to_set(h0.copy(), target)
    def obj(x):
        return compute_c5(x, n)
    def con(x):
        return float(np.sum(x) - target)
    try:
        res = minimize(obj, h0, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': 1e-11})
        h_out = project_to_set(res.x.copy(), target)
        return h_out, compute_c5(h_out, n)
    except Exception:
        return h0, compute_c5(h0, n)

def block_swap_search(h, n, rng, n_iters=5000):
    target = n / 2.0
    h = h.copy()
    best = compute_c5(h, n)
    for _ in range(n_iters):
        bl = min(rng.randint(1, max(2, n // 8)), n // 3)
        i = rng.randint(0, n - bl)
        j = rng.randint(0, n - bl)
        if abs(i - j) < bl:
            continue
        h[i:i+bl], h[j:j+bl] = h[j:j+bl].copy(), h[i:i+bl].copy()
        h = project_to_set(h, target)
        c = compute_c5(h, n)
        if c < best:
            best = c
        else:
            h[i:i+bl], h[j:j+bl] = h[j:j+bl].copy(), h[i:i+bl].copy()
            h = project_to_set(h, target)
    return h, best

def run(seed=42, budget_s=1100, **kwargs):
    global _budget, _start_t
    _budget = float(budget_s)
    _start_t = time.time()
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = np.inf
    best_n = 0

    def update(h, n):
        nonlocal best_h, best_c5, best_n
        h = project_to_set(np.asarray(h, dtype=float), n / 2.0)
        c = compute_c5(h, n)
        if c < best_c5:
            best_h, best_c5, best_n = h.copy(), c, n

    n1 = 90
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        if len(h0) != n1:
            h0 = upsample(h0, n1)
    except Exception:
        h0 = np.zeros(n1)
        h0[rng.choice(n1, size=n1//2, replace=False)] = 1.0
    h0 = project_to_set(h0, n1 / 2.0)
    update(h0, n1)

    # Phase 1: Binary SA + continuous refinement at n=90
    hb, cb = sa_binary(h0, n1, rng, n_iters=40000, t0=0.5, t1=0.003)
    update(hb, n1)

    hc, cc = coord_descent(h0, n1, rng, n_sweeps=20, step_init=0.15)
    update(hc, n1)

    if time_left() > 300:
        hs, cs = slsqp_opt(best_h, n1, maxiter=500)
        update(hs, n1)

    # Multiple SA restarts with perturbations
    n_rounds = 0
    while time_left() > 350 and n_rounds < 8:
        n_rounds += 1
        noise_scale = max(0.03, 0.15 * 0.7 ** n_rounds)
        hm = best_h + rng.normal(0, noise_scale, n1)
        hm = project_to_set(hm, n1 / 2.0)
        hm_b = (hm > 0.5).astype(float)
        hm_b = fix_binary_count(hm_b, n1)
        ha, ca = sa_binary(hm_b, n1, rng, n_iters=20000, t0=0.3, t1=0.002)
        update(ha, n1)
        if time_left() > 320:
            hc2, _ = coord_descent(best_h if best_n == n1 else upsample(best_h, n1), n1, rng, n_sweeps=8)
            update(hc2, n1)

    # Block swap search
    if time_left() > 320:
        hs, _ = block_swap_search(best_h, n1, rng, n_iters=8000)
        update(hs, n1)

    # Coord descent final pass
    if time_left() > 300:
        hc, _ = coord_descent(best_h, n1, rng, n_sweeps=10, step_init=0.05)
        update(hc, n1)

    # Phase 2: n=180 refinement
    if time_left() > 150:
        n2 = 180
        h_up = project_to_set(upsample(best_h, n2), n2 / 2.0)
        update(h_up, n2)
        hc, _ = coord_descent(h_up, n2, rng, n_sweeps=12)
        update(hc, n2)
        if time_left() > 100:
            hs, _ = slsqp_opt(project_to_set(upsample(best_h, n2), n2/2.0), n2, maxiter=400)
            update(hs, n2)
        if time_left() > 80:
            hc, _ = coord_descent(best_h, best_n, rng, n_sweeps=8, step_init=0.04)
            update(hc, best_n)

    # Phase 3: n=360
    if time_left() > 60:
        n3 = 360
        h_up = project_to_set(upsample(best_h, n3), n3 / 2.0)
        update(h_up, n3)
        hc, _ = coord_descent(h_up, n3, rng, n_sweeps=6, step_init=0.06)
        update(hc, n3)

    # Final polish at best resolution
    if time_left() > 15:
        hc, _ = coord_descent(best_h, best_n, rng, n_sweeps=8, step_init=0.03)
        update(hc, best_n)

    best_h = project_to_set(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h, best_n)
    return best_h, best_c5, best_n
