from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_c5(h, n, top_m=10, temperature=100.0):
    """Smooth approximation of max using softmax of top correlations"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    # Use log-sum-exp of top m values
    idx = np.argsort(corr)[-top_m:]
    top = corr[idx]
    return float(np.log(np.sum(np.exp(temperature * (top - top.max())))) / temperature + top.max())

def project_to_set(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        hp = np.clip(h - mid, 0.0, 1.0)
        if float(np.sum(hp)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

        return float(np.sum(x) - target_val)

                       options={'maxiter': maxiter, 'ftol': 1e-14})

        return h_out, compute_c5(h_out, n)

        return h0, compute_c5(h0, n)

C₅ = max_k ∫ h(x)(1-h(x+k)) dx

For the discrete case: C₅ = max_k Σ_i h[i]·(1-h[i+k]) · (2/n) where the sum is over valid i.

The key shifts that matter:
- k=0: Σ h[i](1-h[i]) = Σh - Σh². To minimize this, we want h to be as binary as possible (h² close to h, i.e., h close to 0 or 1).
- Small k (1, 2, 3, ...): These measure local "misalignment" of h with its shifted version.
- Large k (near n/2): The overlap region is small, so the sum is small.

The maximum is likely at small k (1-5 or so). So we should focus on those.

    best_c5 = compute_c5(best_h, best_n)
    return best_h, best_c5, best_n
