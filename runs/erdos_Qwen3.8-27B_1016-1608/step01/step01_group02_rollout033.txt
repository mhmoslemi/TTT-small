            tk_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * tk * tk))
            y = h_new + ((tk - 1.0) / tk_new) * (h_new - h)
            y = project_constraint(y, target)
            h = h_new
            tk = tk_new
    return h

def slsqp_opt(h0, dx, target, maxiter):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(compute_c5, h0, args=(dx,), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_constraint(res.x, target)
    except Exception:
        return None

def swap_improve(h, dx, target, rng, n_iter):
    n = len(h)
    h = h.copy().astype(float)
    c5_cur = compute_c5(h, dx)
    for it in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        for direction in [1, -1]:
            if direction == 1:
                max_d = min(h[i], 1.0 - h[j])
            else:
                max_d = min(1.0 - h[i], h[j])
            if max_d < 1e-12:
                continue
            # Binary search for best delta
            best_d = 0.0
            best_c5 = c5_cur
            for frac in [0.25, 0.5, 0.75, 1.0]:
                d = frac * max_d
                h[i] -= direction * d
                h[j] += direction * d
                c5_t = compute_c5(h, dx)
                if c5_t < best_c5:
                    best_c5 = c5_t
                    best_d = d
                else:
                    h[i] += direction * d
                    h[j] -= direction * d
                    break
            if best_d > 0:
                h[i] -= direction * best_d
                h[j] += direction * best_d
                c5_cur = best_c5
            else:
                h[i] += direction * max_d * 0  # no change
                h[j] -= direction * max_d * 0
    return h, c5_cur

def make_thue_morse(n):
    seq = [0]
    for i in range(n - 1):
        seq.append(seq[-1] ^ ((i + 1) & -(i + 1)) if False else 0)
    # Better: standard Thue-Morse
    seq = [0]
    for _ in range(int(np.ceil(np.log2(n)))):
        seq = seq + [1 - x for x in seq]
    seq = seq[:n]
    return np.array(seq, dtype=float)

def make_initialization(n, rng, seed_type):
    if seed_type == 0:
        # Thue-Morse based
        seq = make_thue_morse(n)
        h = seq + rng.randn(n) * 0.3
        return np.clip(h, 0, 1)
    elif seed_type == 1:
        # Sine
        t = np.linspace(0, 4 * np.pi, n)
        freq = rng.choice([1, 2, 3, 5, 7])
        h = 0.5 + 0.4 * np.sin(freq * t / 2 + rng.uniform(0, 2 * np.pi))
        return np.clip(h, 0, 1)
    elif seed_type == 2:
        # Random blocks
        h = np.zeros(n)
        pos = 0
        while pos < n:
            blk = rng.randint(3, 10)
            if rng.random() < 0.5:
                h[pos:pos+blk] = 1.0
            pos += blk
        return np.clip(h, 0, 1)
    elif seed_type == 3:
        # Pure random
        return rng.random(n)
    else:
        h = np.zeros(n)
        h[::2] = 1.0  # alternating
        return h + rng.randn(n) * 0.2
