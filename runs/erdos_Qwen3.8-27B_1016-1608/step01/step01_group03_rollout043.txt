    m = np.max(corrs)
    return m + np.log(np.sum(np.exp(beta * (corrs - m)))) / beta

def grad_lse(h):
    n = _n
    corr = np.correlate(h, 1.0 - h, mode="full") * _dx
    m = np.max(corr)
    exp_c = np.exp(_beta * (corr - m))
    w = exp_c / np.sum(exp_c)
    
    grad = np.zeros(n)
    for k in range(2*n-1):
        wk = w[k] * _dx
        if abs(wk) < 1e-18:
            continue
        
        # First term: (1-h[j+n-1-k]) for j in S_k
        lo = max(0, k-n+1)
        hi = min(n-1, k)
        if lo <= hi:
            js = np.arange(lo, hi+1)
            idxs = js + n - 1 - k
            grad[js] += wk * (1.0 - h[idxs])
        
        # Second term: -h[j-n+1+k] for j where i=j-n+1+k in S_k
        if k < n:
            # j in [n-1-k, n-1]
            js2 = np.arange(n-1-k, n)
        else:
            # j in [0, min(n-1, 2n-2-k)]
            hi2 = min(n-1, 2*n-2-k)
            js2 = np.arange(0, hi2+1)
        
        if len(js2) > 0:
            idxs2 = js2 - n + 1 + k
            grad[js2] -= wk * h[idxs2]
    
    return grad
