def coord_best_t(h, i, n):
    m = 2*n - 1
    s_values = np.arange(-(n-1), n)
    idx_p = i + s_values
    idx_m = i - s_values

    vm = (idx_m >= 0) & (idx_m < n)
    B[vp] += (1.0 - h[idx_p[vp]])
    B[vm] -= h[idx_m[vm]]
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    C = corr - B * h[i]
    
    # Vectorized crossing point computation
    # For all pairs (a,b): t_cross = (C[b]-C[a])/(B[a]-B[b])
    C_diff = C[:,None] - C[None,:]  # (m,m)
    B_diff = B[:,None] - B[None,:]  # (m,m)
    
    # Only valid where |B_diff| > eps and 0 < t_cross < 1
    valid = (np.abs(B_diff) > 1e-14)
    t_cross = np.where(valid, C_diff / np.where(valid, B_diff, 1.0), -1.0)
    valid &= (t_cross > 0) & (t_cross < 1)
    
    candidates = t_cross[valid]
    candidates = np.unique(candidates)
    candidates = np.concatenate([[0.0, 1.0], candidates])

    # Values[k, j] = C[k] + B[k] * candidates[j]
    values = C[:,None] + B[:,None] * candidates[None,:]
    max_vals = values.max(axis=0)
    
    best_idx = np.argmin(max_vals)
    return candidates[best_idx], max_vals[best_idx]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return c.max() * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.any():
            h[free] += diff / free.sum()
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.argmin(h)
            h[idx] = min(h[idx] + residual, 1.0)
        else:
            idx = np.argmax(h)
            h[idx] = max(h[idx] + residual, 0.0)
    return h
