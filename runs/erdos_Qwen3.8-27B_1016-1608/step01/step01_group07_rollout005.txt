import time

    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def _fix(h, n):
    t = 0.5 * n
    for _ in range(200):
        d = t - np.sum(h)
        if abs(d) < 1e-14:
            break
        mask = (h < 1.0 - 1e-12) if d > 0 else (h > 1e-12)
        nk = int(np.sum(mask))
        if nk == 0:
            break
        h = h + (d / nk) * mask
        h = np.clip(h, 0.0, 1.0)
    return h

def _interp(h, n):
    n0 = len(h)
    xo = np.linspace(0, 2, n0, endpoint=False)
    xn = np.linspace(0, 2, n, endpoint=False)
    return _fix(np.clip(np.interp(xn, xo, h), 0, 1), n)

def _smooth_obj(x, alpha, M, lam):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    # Top M lags
    idx = np.argpartition(c, -M)[-M:]
    tv = c[idx]
    m = np.max(tv)
    e = np.exp(np.clip(tv - m, -50, 50))
    pen = lam * (np.sum(x) - 0.5 * n) ** 2
    return m + np.log(np.sum(e)) / alpha + pen

def _try_lbfgs(h0, n, alpha, M, lam, maxiter, t0, budget):
    if budget - (time.time() - t0) < 30:
        return None
    bnds = [(0.0, 1.0)] * n
    args = (alpha, M, lam)
    try:
        res = minimize(lambda x: _smooth_obj(x, *args), h0, method='L-BFGS-B',
                       bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12})
        ht = _fix(np.clip(res.x, 0, 1), n)
        return ht
    except Exception:
        return None

def _try_slsqp(h0, n, alpha, M, maxiter, t0, budget):

        return None
    bnds = [(0.0, 1.0)] * n

        return _smooth_obj(x, alpha, M, 0)
    def icf(x):
        return np.sum(x) - 0.5 * n
    def icj(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': icf, 'jac': icj}]
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        ht = _fix(np.clip(res.x, 0, 1), n)
        return ht
    except Exception:
        return None

    h2[idx] = 1.0

    se = np.log(np.sum(e))
    pen = lam * (np.sum(x) - 0.5 * n) ** 2
    return m + se / alpha + pen

                       bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14, 'maxls': 50})
        if res.success or res.fun < 1e6:
            ht = _fix(np.clip(res.x, 0, 1), n)
            return ht
    except Exception:
        pass
    return None

def _run_slsqp(h0, n, alpha, M, maxiter, t0, budget):
    elapsed = time.time() - t0
    if budget - elapsed < 90:
        return None
    
    def obj(x):
        n_loc = len(x)
        dx = 2.0 / n_loc
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        if M < len(c):
            idx = np.argpartition(-c, M)[:M]
            tv = c[idx]
        else:
            tv = c
        m = np.max(tv)
        e = np.exp(np.clip(tv - m, -40, 40))
        return m + np.log(np.sum(e)) / alpha
    
    def con_fun(x):
        return np.sum(x) - 0.5 * n
    
    def con_jac(x):
        return np.ones(n)
    
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        ht = _fix(np.clip(res.x, 0, 1), n)
        return ht
    except Exception:
        return None

    # Threshold to binary with exactly n//2 ones
    half = n // 2

    h2[idx] = 1.0

            h2[oz] = 1.0
            c = _c5(h2)
            if c < rc - 1e-12:
                ones_list[i] = oz
                zeros_list[j] = oi
                rc = c
                improved = True
            else:
                h2[oi] = 1.0
                h2[oz] = 0.0
    return h2, rc
