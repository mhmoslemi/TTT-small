Actually, numpy's correlate(a, v, mode='full') computes:
c[k] = sum_n a[n+k] * conj(v[n]) for k going from -(len(v)-1) to (len(a)-1)

So c[n-1] (the middle) gives shift 0: sum_n a[n] * conj(v[n]) = sum h[i] * (1-h[i])

            for _ in range(6):
                if time.time() > budget_s * 0.99:
                    break
                improved = False
                for i in range(n):
                    for d in deltas:
                        h_t = best_h.copy()
                        h_t[i] = np.clip(h_t[i] + d, 0.0, 1.0)
                        h_t = project_feasible(h_t, n)
                        c5_t = compute_c5(h_t)
                        if c5_t < best_c5 - 1e-13:
                            best_c5 = c5_t
                            best_h = h_t.copy()
                            improved = True
                            break
                if not improved:
                    break

    # Final sanity check
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    if best_c5 > 0.40:
        best_h = project_feasible(np.array(initial_h_values, dtype=float), 47)
        best_c5 = compute_c5(best_h)
        best_n = 47

    return (best_h, best_c5, best_n)
