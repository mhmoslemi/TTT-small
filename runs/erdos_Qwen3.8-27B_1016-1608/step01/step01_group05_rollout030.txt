    for i in range(len(a)):
        for l in range(i+1, len(a)):

        val = np.max(a * t + b)
        if val < best_val:
            best_val = val
            best_t = t
    
    h[j] = best_t
    return best_val

def anneal_coord(h, n, rng, n_iter=2000, T0=0.02):
    h = h.copy()

    cur_c = best_c
