    # Use a known primitive polynomial for small m
    # For m=3: x^3 + x + 1 (polynomial 0b1011 = 11)
    # For m=4: x^4 + x + 1 (polynomial 0b10011 = 19)

    for _ in range(n):

        # Generate next state
        lsb = state & 1
        state >>= 1
        if lsb:

            # Get the lower m bits (excluding the x^m term)
            lower = p & ((1 << m) - 1)
            for b in range(m):
                if lower & (1 << b):
                    taps |= (1 << (m - 1 - b))

    return np.array(seq, dtype=float)
