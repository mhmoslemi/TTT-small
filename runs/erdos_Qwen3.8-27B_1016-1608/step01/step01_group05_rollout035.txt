    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_c5_soft(h, n, beta=50.0):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    if np.max(corr) < 0:
        return 0.0
    shifted = corr - np.max(corr)
    return float(np.max(corr) + np.log(np.sum(np.exp(shifted * beta))) / beta)

def project_to_set(h, target):

                delta = c5 - best
                try_cnt += 1
                if delta < best_delta - 1e-14:
                    best_delta = delta
                    best_pair = (i, j)
        if best_pair is not None:
            i, j = best_pair
            hb[i], hb[j] = 0.0, 1.0
            ones.remove(i)
            zeros.remove(j)
            zeros.append(i)
            ones.append(j)
            best += best_delta
        else:
            # Random perturbation to escape local optimum
            if try_cnt > 0 and rng.random() < 0.3:
                i = int(rng.choice(ones))
                j = int(rng.choice(zeros))
                hb[i], hb[j] = 0.0, 1.0
                c5 = compute_c5(hb, n)
                if c5 < best * 1.01:
                    ones.remove(i)
                    zeros.remove(j)
                    zeros.append(i)
                    ones.append(j)
                    best = c5
                else:
                    hb[i], hb[j] = 1.0, 0.0
            else:
                break
    return hb, best

def coord_desc(h, n, rng, n_sweeps):
    target = n / 2.0
    h = project_to_set(h.copy(), target)
    best = compute_c5(h, n)
    deltas = (0.01, -0.01, 0.02, -0.02, 0.05, -0.05)
    for _ in range(n_sweeps):
        if _time_left() < 5:
            break
        improved = False
        for j in rng.permutation(n):
            old = h[j]
            for d in deltas:
                c = old + d
                if c < 0.0 or c > 1.0:
                    continue
                h[j] = c
                h = project_to_set(h, target)
                v = compute_c5(h, n)
                if v < best - 1e-14:
                    best = v
                    improved = True
                    break
                h[j] = old
            h = project_to_set(h, target)
        if not improved:
            break
    return h, best

def fourier_gen(params, n, K):
    x = np.linspace(0.0, 2.0, n)
    h = np.full(n, params[0])
    for k in range(1, K + 1):
        h += params[1 + 2*(k-1)] * np.cos(k * np.pi * x)
        h += params[2 + 2*(k-1)] * np.sin(k * np.pi * x)
    return np.clip(h, 0.0, 1.0)

def _fourier_obj(params):
    n = _obj_n
    K = int((len(params) - 1) / 2)
    h = fourier_gen(params, n, K)
    h = project_to_set(h, n / 2.0)
    return compute_c5(h, n)

def optimize_fourier(n, K, rng, maxiter):
    global _obj_n
    _obj_n = n
    target = n / 2.0
    p0 = np.zeros(1 + 2 * K)
    p0[0] = 0.5
    p0[1:] = rng.uniform(-0.15, 0.15, 2 * K)
    try:
        res = minimize(_fourier_obj, p0, method='Nelder-Mead',
                       options={'maxiter': maxiter, 'xatol': 1e-9, 'fatol': 1e-11})
        h = project_to_set(fourier_gen(res.x, n, K), target)
    except Exception:
        h = project_to_set(fourier_gen(p0, n, K), target)
    return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    global _budget, _start_t
    _budget = float(budget_s)
    _start_t = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = np.inf
    best_n = 0

    def _upd(h, n):
        nonlocal best_h, best_c5, best_n
        h = project_to_set(np.asarray(h, dtype=float), n / 2.0)
        c = compute_c5(h, n)
        if c < best_c5:
            best_h, best_c5, best_n = h.copy(), c, n

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n1 = len(h_init)
    except Exception:
        n1 = 90
        h_init = np.full(n1, 0.5)
    h_init = project_to_set(h_init, n1 / 2.0)
    _upd(h_init, n1)

    # Phase 1: SLSQP with increasing softmax sharpness at n1
    for beta in (20, 40, 80, 150):
        if _time_left() < _budget * 0.22:
            break
        h_o, c5 = slsqp_refine(best_h, n1, beta, 350)
        _upd(h_o, n1)

    # Phase 2: Binary steepest descent
    if _time_left() > _budget * 0.18:
        hb, cb = binary_steepest(best_h, best_n, rng, 20000)
        _upd(hb, best_n)
        h_cd, cc = coord_desc(hb, best_n, rng, 35)
        _upd(h_cd, best_n)

    # Phase 3: Fourier exploration for diversity
    if _time_left() > _budget * 0.20:
        for K in (3, 5, 7):
            if _time_left() < _budget * 0.10:
                break
            hf, cf = optimize_fourier(n1, K, rng, 300)
            _upd(hf, n1)
            if cf < best_c5 + 0.005:
                ho, c5r = slsqp_refine(hf, n1, 80, 200)
                _upd(ho, n1)

    # Phase 4: Upsample to 180
    if _time_left() > _budget * 0.25:
        n2 = 180
        h_up = project_to_set(upsample(best_h, n2), n2 / 2.0)
        _upd(h_up, n2)
        for beta in (50, 100):
            if _time_left() < _budget * 0.12:
                break
            ho, c5 = slsqp_refine(best_h, n2, beta, 250)
            _upd(ho, n2)
        if _time_left() > _budget * 0.12:
            hb2, cb2 = binary_steepest(best_h, best_n, rng, 12000)
            _upd(hb2, best_n)
        h_cd2, cc2 = coord_desc(best_h, best_n, rng, 30)
        _upd(h_cd2, best_n)

    # Phase 5: Upsample to 360
    if _time_left() > _budget * 0.18:
        n3 = 360
        h_up3 = project_to_set(upsample(best_h, n3), n3 / 2.0)
        _upd(h_up3, n3)
        if _time_left() > _budget * 0.10:
            ho3, c5_3 = slsqp_refine(best_h, n3, 50, 200)
            _upd(ho3, n3)
        h_cd3, cc3 = coord_desc(best_h, best_n, rng, 25)
        _upd(h_cd3, best_n)

    # Final polish
    if _time_left() > 15:
        hf, cf = coord_desc(best_h, best_n, rng, 60)
        _upd(hf, best_n)
        if _time_left() > 10:
            ho, c5f = slsqp_refine(best_h, best_n, 200, 150)
            _upd(ho, best_n)

    best_h = project_to_set(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h, best_n)
    return best_h, best_c5, best_n
