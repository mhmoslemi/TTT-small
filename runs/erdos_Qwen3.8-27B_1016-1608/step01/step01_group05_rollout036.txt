    c_max = np.max(corr)
    corr_shifted = corr - c_max  # for numerical stability
    sm = c_max + np.log(np.sum(np.exp(T * corr_shifted))) / T

def soft_max_grad(h, n, T=50):

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_set(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def soft_max_obj(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    cm = np.max(corr)
    return (cm + np.log(np.sum(np.exp(T * (corr - cm)))) / T) * dx

def soft_max_grad(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    cm = np.max(corr)
    w = np.exp(T * (corr - cm))
    w /= np.sum(w)
    grad = np.zeros(n)
    for m in range(len(corr)):
        if w[m] < 1e-12:
            continue
        s = m - (n - 1)
        j = np.arange(n)
        m1 = (j + s >= 0) & (j + s < n)
        m2 = (j - s >= 0) & (j - s < n)
        grad[m1] += w[m] * (1.0 - h[j[m1] + s])
        grad[m2] += -w[m] * h[j[m2] - s]
    return grad * dx

def lbfgs_smooth(h0, n, T=30, maxiter=300):
    h0 = project_to_set(h0.copy(), n / 2.0)
    
    def obj(x):
        return soft_max_obj(x, n, T)
    
    def grad(x):
        return soft_max_grad(x, n, T)
    
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, jac=grad, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'gtol': 1e-10})
    h_out = project_to_set(res.x.copy(), n / 2.0)
    return h_out, compute_c5(h_out, n)

def coordinate_descent(h, n, rng, n_rounds=50):
    target = n / 2.0
    h = h.copy()
    best = compute_c5(h, n)
    for r in range(n_rounds):
        order = rng.permutation(n)
        improved = False
        for j in order:
            old = h[j]
            # Try binary values and small perturbations
            cands = [0.0, 1.0]
            if old > 0.05 and old < 0.95:
                cands.extend([max(0, old-0.1), min(1, old+0.1), max(0, old-0.05), min(1, old+0.05)])
            bc, bv = old, best
            for c in cands:
                if abs(c - old) < 1e-8:
                    continue
                h[j] = c
                h = project_to_set(h, target)
                v = compute_c5(h, n)
                if v < bv - 1e-12:
                    bv = v
                    bc = c
            if bc != old:
                h[j] = bc
                h = project_to_set(h, target)
                best = bv
                improved = True
            else:
                h[j] = old
                # Need to re-project
                h = project_to_set(h, target)
        if not improved:
            break
    return h, best

def upsample(h, n_new):
    n_old = len(h)
    x = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x, h)

def make_binary_init(n, rng):

import numpy as np
from scipy.optimize import minimize
import time

_budget = 1100.0
_start_t = 0.0

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_set(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def soft_max_obj_grad(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    cm = np.max(corr)
    eT = np.exp(T * (corr - cm))
    sw = np.sum(eT)
    obj = (cm + np.log(sw) / T) * dx
    w = eT / sw
    if np.min(w) < 1e-14:
        w = np.where(w < 1e-14, 0.0, w)
    grad = np.zeros(n)
    for m in range(len(corr)):
        if w[m] < 1e-14:
            continue
        s = m - (n - 1)
        j = np.arange(n)
        m1 = (j + s >= 0) & (j + s < n)
        m2 = (j - s >= 0) & (j - s < n)
        grad[m1] += w[m] * (1.0 - h[j[m1] + s])
        grad[m2] -= w[m] * h[j[m2] - s]
    return obj, grad * dx

def lbfgs_opt(h0, n, T, maxiter=400):
    h0 = project_to_set(h0.copy(), n / 2.0)
    def obj(x):
        return soft_max_obj_grad(x, n, T)[0]
    def grad(x):
        return soft_max_obj_grad(x, n, T)[1]
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, jac=grad, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'gtol': 1e-12, 'maxls': 50})
    return project_to_set(res.x, n / 2.0)

def anneal_T(h0, n, T_list, iters_per_T=200):
    h = h0.copy()
    for T in T_list:
        h = lbfgs_opt(h, n, T, maxiter=iters_per_T)
    return h

def coord_descent(h, n, rng, n_rounds=30):

    h = h.copy()
    best = compute_c5(h, n)
    for _ in range(n_rounds):

        improved = False
        for j in order:
            old = h[j]
            best_v = best
            best_c = old

            for c in [0.0, 1.0]:

                h = project_to_set(h, target)
                best = best_v
                improved = True
            else:
                h = h_saved
        if not improved:

    return np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, len(h)), h)

import numpy as np
from scipy.optimize import minimize
import time

_budget = 1100.0
_start_t = 0.0

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_set(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _sm_obj(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    cm = np.max(corr)
    return (cm + np.log(np.sum(np.exp(T * (corr - cm)))) / T) * dx

def _sm_grad(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    cm = np.max(corr)
    eT = np.exp(T * (corr - cm))
    sw = np.sum(eT)
    w = eT / sw
    w = np.where(w < 1e-14, 0.0, w)
    grad = np.zeros(n)
    j = np.arange(n)
    for m in range(len(corr)):
        if w[m] < 1e-14:
            continue
        s = m - (n - 1)
        m1 = (j + s >= 0) & (j + s < n)
        m2 = (j - s >= 0) & (j - s < n)
        grad[m1] += w[m] * (1.0 - x[j[m1] + s])
        grad[m2] -= w[m] * x[j[m2] - s]
    return grad * dx

def lbfgs_at_T(h0, n, T, maxiter=300):
    h0 = project_to_set(h0.copy(), n / 2.0)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_sm_obj, h0, args=(n, T), jac=_sm_grad,
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'gtol': 1e-11, 'maxls': 40})
        return project_to_set(res.x, n / 2.0)
    except Exception:
        return h0

def anneal_T(h0, n, T_list, iters=250):
    h = h0.copy()
    for T in T_list:
        h = lbfgs_at_T(h, n, T, maxiter=iters)
    return h

def coord_descent(h, n, rng, n_rounds=30):
    target = n / 2.0
    h = h.copy()
    best = compute_c5(h, n)
    for _ in range(n_rounds):
        order = rng.permutation(n)
        improved = False
        for j in order:
            h0 = h.copy()
            h0[j] = 0.0
            h0 = project_to_set(h0, target)
            v0 = compute_c5(h0, n)
            h1 = h.copy()
            h1[j] = 1.0
            h1 = project_to_set(h1, target)
            v1 = compute_c5(h1, n)
            if v0 < best - 1e-12:
                h, best = h0, v0
                improved = True
            elif v1 < best - 1e-12:
                h, best = h1, v1
                improved = True
        if not improved:
            break
    return h, best

def upsample(h, n_new):
    return np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, len(h)), h)

def downsample(h, n_new):
    return np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, len(h)), h)
