    return np.clip(h, 0.0, 1.0)

def coord_step(h, i, n, lam, t0_budget):
    """Optimize h[i] in 1D with sum-constraint penalty."""
    v0 = h[i]
    sum_before = np.sum(h) - v0
    target = n / 2.0
    n_full = 2 * n - 1
    dx = 2.0 / n

    h[i] = 0.0
    base = np.correlate(h, 1.0 - h, mode='full')
    b0 = base[0]

    shifts = np.arange(n_full) - (n - 1)
    coeff = np.zeros(n_full)
    ip = i + shifts
    vp = (ip >= 0) & (ip < n)
    im = i - shifts
    vm = (im >= 0) & (im < n)
    coeff[vp] += (1.0 - h[ip[vp]])
    coeff[vm] -= h[im[vm]]
    h[i] = v0

    def eval_v(v):
        c0 = b0 + v - v * v
        mx = c0
        if n_full > 1:
            r = base[1:] + v * coeff[1:]
            rmx = float(np.max(r))
            if rmx > mx:
                mx = rmx
        return dx * mx + lam * (sum_before + v - target) ** 2

    best_v = v0
    best_f = eval_v(v0)
    for v_c in np.linspace(0.0, 1.0, 41):
        f_c = eval_v(v_c)
        if f_c < best_f:
            best_f, best_v = f_c, v_c
    lo = max(0.0, best_v - 0.04)
    hi = min(1.0, best_v + 0.04)
    for v_c in np.linspace(lo, hi, 81):
        f_c = eval_v(v_c)
        if f_c < best_f:
            best_f, best_v = f_c, v_c
    h[i] = best_v
    return h

def cd_pass(h, n, lam, n_iter, t0, budget):
    for it in range(n_iter):
        if time.time() - t0 > budget - 50:
            break
        h1 = h.copy()
        order = np.random.permutation(n)
        for i in order:
            coord_step(h1, int(i), n, lam, t0, budget)
        h1 = repair(h1, n)
        if compute_c5(h1) < compute_c5(h):
            h = h1
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = 0.5 * np.ones(80)

    n = len(h0)
    h0 = repair(h0, n)
    bh = h0.copy()
    bc = compute_c5(bh)

    # Phase 1: Coordinate descent with increasing penalty
    lam_list = [50.0, 200.0, 1000.0, 5000.0, 20000.0, 100000.0]
    for lam in lam_list:
        if time.time() - t0 > budget_s - 500:
            break
        for _ in range(8):
            if time.time() - t0 > budget_s - 500:
                break
            bh = cd_pass(bh, n, lam, 3, t0, budget_s)
        c5 = compute_c5(bh)
        if c5 < bc:
            bc = c5

    # Phase 2: Simulated annealing + CD
    temp = 0.08
    for it in range(4000):
        if time.time() - t0 > budget_s - 200:
            break
        k_move = np.random.randint(2, 8)
        idx = np.random.choice(n, size=k_move, replace=False)
        hp = bh.copy()
        hp[idx] += np.random.randn(k_move) * temp
        hp = repair(hp, n)
        c5_p = compute_c5(hp)
        accept = c5_p < bc
        if not accept and temp > 1e-4:
            accept = np.random.uniform() < np.exp(-max(0.0, c5_p - bc) / (temp * 0.01))
        if accept:
            bh = hp
            if c5_p < bc:
                bc = c5_p
            if it % 5 == 0:
                bh = cd_pass(bh, n, 20000.0, 2, t0, budget_s)
                c5 = compute_c5(bh)
                if c5 < bc:
                    bc = c5
        temp *= 0.9985

    # Phase 3: Fine CD with very high lambda
    for lam in [50000.0, 200000.0, 500000.0]:
        if time.time() - t0 > budget_s - 80:
            break
        bh = cd_pass(bh, n, lam, 15, t0, budget_s)
        c5 = compute_c5(bh)
        if c5 < bc:
            bc = c5

    # Phase 4: Tiny random perturbations + CD
    for _ in range(2000):
        if time.time() - t0 > budget_s - 20:
            break
        hp = bh + np.random.randn(n) * 0.004
        hp = repair(hp, n)
        c5 = compute_c5(hp)
        if c5 < bc:
            bc, bh = c5, hp
            bh = cd_pass(bh, n, 200000.0, 2, t0, budget_s)
            c5 = compute_c5(bh)
            if c5 < bc:
                bc = c5

    # Phase 5: Block optimization - optimize small groups jointly
    if time.time() - t0 < budget_s - 15:
        for _ in range(100):
            if time.time() - t0 > budget_s - 10:
                break
            bsz = np.random.randint(2, 6)
            s = np.random.randint(0, n - bsz + 1)
            idx = list(range(s, s + bsz))
            for rep in range(3):
                for i in idx:
                    coord_step(bh, i, n, 200000.0, t0, budget_s)
            bh = repair(bh, n)
            c5 = compute_c5(bh)
            if c5 < bc:
                bc = c5

    fh = repair(bh, n)
    fc = compute_c5(fh)
    return (fh.tolist(), float(fc), n)
