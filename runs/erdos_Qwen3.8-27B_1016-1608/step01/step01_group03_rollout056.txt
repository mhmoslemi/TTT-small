from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5_grad(h, n):
    """Compute subgradient of max_k corr_k."""
    corr = compute_corr(h)
    k_max = np.argmax(corr)
    lag = k_max - (n - 1)
    grad = np.zeros(n)
    if lag == 0:
        grad = 1.0 - 2.0 * h
    else:
        d = abs(lag)
        if lag > 0:
            # corr_d = sum_{i=d}^{n-1} h[i]*(1-h[i-d])
            # d/dh[j] = (1-h[j-d]) for j>=d, -h[j+d] for j+d<n
            mask1 = np.arange(n) >= d
            grad[mask1] += (1.0 - h[np.arange(n)[mask1] - d])
            mask2 = np.arange(n) + d < n
            grad[mask2] -= h[np.arange(n)[mask2] + d]
        else:
            # corr_{-d} = sum_{i=0}^{n-1-d} h[i]*(1-h[i+d])
            # d/dh[j] = (1-h[j+d]) for j<=n-1-d, -h[j-d] for j>=d
            mask1 = np.arange(n) < n - d
            grad[mask1] += (1.0 - h[np.arange(n)[mask1] + d])
            mask2 = np.arange(n) >= d
            grad[mask2] -= h[np.arange(n)[mask2] - d]
    return grad, k_max, corr

def project_feasible(h, target):
    """Project h onto {x: 0<=x<=1, sum(x)=target}."""
    h = np.clip(h, 0.0, 1.0)
    # Bisection for lambda
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(projected)
        if s > target:
            lo = mid
        else:
            hi = mid
    projected = np.clip(h - (lo+hi)/2.0, 0.0, 1.0)
    # Fix any remaining numerical issue
    diff = target - np.sum(projected)
    if abs(diff) > 1e-10:
        # Distribute among non-boundary elements
        mask = (projected > 1e-10) & (projected < 1.0 - 1e-10)
        if mask.sum() > 0:
            projected[mask] += diff / mask.sum()
            projected = np.clip(projected, 0.0, 1.0)
    return projected

def soft_max_grad(h, n, tau=100.0):
    """Compute gradient of smooth max of correlations."""
    corr = compute_corr(h)
    # Softmax weights
    corr_shifted = corr - np.max(corr)
    weights = np.exp(tau * corr_shifted)
    weights /= weights.sum()
    
    grad = np.zeros(n)
    idx = np.arange(n)
    for k in range(2*n-1):
        if weights[k] < 1e-12:
            continue
        lag = k - (n-1)
        if lag == 0:
            grad += weights[k] * (1.0 - 2.0*h)
        elif lag > 0:

            m1 = idx >= d
            m2 = idx + d < n
            grad[m1] += weights[k] * (1.0 - h[idx[m1] - d])
            grad[m2] -= weights[k] * h[idx[m2] + d]
        else:
            d = -lag
            m1 = idx < n - d
            m2 = idx >= d
            grad[m1] += weights[k] * (1.0 - h[idx[m1] + d])
            grad[m2] -= weights[k] * h[idx[m2] - d]
    return grad, np.max(corr)
