    tf = t_mat.ravel()
    tf = tf[(tf >= -1e-12) & (tf <= 1.0 + 1e-12)]
    tf = np.clip(tf, 0.0, 1.0)
    if len(tf) < 2:
        f0 = np.max(a)
        f1 = np.max(a + b)
        return 0.0 if f0 <= f1 else 1.0
    tf = np.concatenate([tf, [0.0, 1.0]])
    tf = np.unique(tf)
    fv = np.max(a[np.newaxis, :] + tf[:, np.newaxis] * b[np.newaxis, :], axis=1)
    return float(tf[np.argmin(fv)])

def gauss_seidel(h, t0, max_time, max_sweeps=20000):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode="full")
    c5 = float(np.max(c) * (2.0 / n))
    for sv in range(max_sweeps):
        if time.time() - t0 > max_time:
            break
        mc = 0.0
        for j in range(n):
            if time.time() - t0 > max_time:
                break
            old = h[j]
            to = exact_coord_step(h, c, j, n)
            d = abs(to - old)
            if d > 1e-13:
                h[j] = to
                mc = max(mc, d)
                lags = np.arange(-(n - 1), n)
                vl = (j - lags >= 0) & (j - lags < n)
                vr = (j + lags >= 0) & (j + lags < n)
                sl = np.zeros(2 * n - 1)
                if np.any(vl):
                    sl[vl] += 1.0 - h[j - lags[vl]]
                if np.any(vr):
                    sl[vr] -= h[j + lags[vr]]
                c += (to - old) * sl
        c5 = float(np.max(c) * (2.0 / n))
        if mc < 1e-13:
            break
    return h, c5

def try_slsqp_smooth(h0, dx, target, max_time):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    temp = 300.0
    def smooth_obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = np.max(corr)
        sc = temp * (corr - m)
        sc = np.clip(sc, -500, 500)
        return (m + np.log(np.sum(np.exp(sc)) / len(corr)) / temp) * dx
    _CT = target
    def eq_con(x):
        return x.sum() - _CT
    try:
        res = minimize(smooth_obj, h0, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': eq_con}],
                       options={'maxiter': 800, 'ftol': 1e-15})
        if time.time() - smooth_obj._t0 > max_time:
            return None
        return projection(res.x, target)
    except Exception:
        return None

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    def time_ok(margin=20):
        return elapsed() < budget_s - margin

    best_h = np.array(initial_h_values, dtype=float).copy()
    best_n = len(best_h)
    best_c5 = compute_c5(best_h, 2.0 / best_n)

    # Phase 1: Exact coordinate descent on initial
    h1, c51 = gauss_seidel(best_h.copy(), t0, elapsed() + (budget_s * 0.25), max_sweeps=5000)
    if c51 < best_c5:
        best_h, best_c5 = h1.copy(), c51

    # Phase 2: SLSQP smooth polish
    if time_ok(40):
        hs = try_slsqp_smooth(best_h.copy(), 2.0 / best_n, best_n / 2.0, budget_s - elapsed() - 40)
        if hs is not None:
            c5s = compute_c5(hs, 2.0 / best_n)
            if c5s < best_c5:
                best_h, best_c5 = hs.copy(), c5s
            h2, c52 = gauss_seidel(hs.copy(), t0, budget_s - 40, max_sweeps=5000)
            if c52 < best_c5:
                best_h, best_c5 = h2.copy(), c52

    # Phase 3: Diverse restarts with coordinate descent
    n = best_n
    dx = 2.0 / n
    target = n / 2.0
    n_restarts = 50
    for i in range(n_restarts):
        if not time_ok(60):
            break
        s = i % 7
        if s == 0:
            h0 = best_h + rng.randn(n) * rng.uniform(0.01, 0.04)
        elif s == 1:
            h0 = best_h + rng.randn(n) * rng.uniform(0.05, 0.12)
        elif s == 2:
            h0 = rng.random(n) * 0.7 + 0.15
        elif s == 3:
            h0 = np.zeros(n)
            nb = rng.randint(3, 10)
            for _ in range(nb):
                bl = rng.randint(4, max(5, n // 4))
                st = rng.randint(0, max(1, n - bl))
                h0[st:st + bl] = rng.uniform(0.4, 1.0)
        elif s == 4:
            q = np.linspace(0, 2 * np.pi * rng.randint(2, 6), n)
            h0 = 0.5 + 0.35 * np.sin(q + rng.uniform(0, 2 * np.pi))
        elif s == 5:
            h0 = best_h.copy()
            bl = rng.randint(3, max(4, n // 5))
            st = rng.randint(0, max(1, n - bl))
            h0[st:st + bl] = rng.uniform(0, 1, bl)
        else:
            h0 = 0.5 + 0.3 * rng.randn(n)
        h0 = projection(h0, target)
        tlim = min(budget_s - 60, elapsed() + (budget_s - elapsed()) / (n_restarts - i + 1) * 0.8)
        ht, c5t = gauss_seidel(h0, t0, tlim, max_sweeps=3000)
        if c5t < best_c5:
            best_h, best_c5 = ht.copy(), c5t
            # Polish with SLSQP
            if time_ok(50):
                hr, c5r = gauss_seidel(best_h + rng.randn(n) * 0.01, t0, budget_s - 50, max_sweeps=3000)
                if c5r < best_c5:
                    best_h, best_c5 = hr.copy(), c5r

    # Phase 4: Try larger n
    for try_n in [80, 100, 120, 140]:
        if not time_ok(50):
            break
        dx_t = 2.0 / try_n
        tgt = try_n / 2.0
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, try_n, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = projection(h_interp, tgt)
        h_rand = projection(0.5 + 0.25 * rng.randn(try_n), tgt)
        for h_start in [h_interp, h_rand]:
            if not time_ok(50):
                break
            ht, c5t = gauss_seidel(h_start, t0, budget_s - 50, max_sweeps=5000)
            if c5t < best_c5:
                best_h, best_c5, best_n = ht.copy(), c5t, try_n

    # Phase 5: Final polish
    if time_ok(15):
        best_h, best_c5 = gauss_seidel(best_h.copy(), t0, budget_s - 15, max_sweeps=10000)

    best_h = projection(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return (best_h, final_c5, len(best_h))
