import numpy as np
from scipy.optimize import minimize
import time

_budget = 1100.0
_start_t = 0.0

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr_array(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def project_to_set(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        hp = np.clip(h - mid, 0.0, 1.0)
        if float(np.sum(hp)) > target:
            lo = mid
        else:
            hi = mid
    mid = 0.5 * (lo + hi)
    result = np.clip(h - mid, 0.0, 1.0)
    diff = target - float(np.sum(result))
    if abs(diff) > 1e-9:
        if diff > 0:
            idx = np.argmax(1.0 - result)
            result[idx] = min(1.0, result[idx] + diff)
        else:
            idx = np.argmax(result)
            result[idx] = max(0.0, result[idx] + diff)
    return result

def soft_max_val(corr, alpha):
    m = np.max(corr)
    return m + alpha * np.log(np.sum(np.exp((corr - m) / alpha)))

def smooth_obj(h, n, alpha):
    corr = compute_corr_array(h, n)
    return soft_max_val(corr, alpha)

def smooth_grad_fd(h, n, alpha, eps):
    f0 = smooth_obj(h, n, alpha)
    g = np.zeros(n)
    for j in range(n):
        h[j] += eps
        f1 = smooth_obj(h, n, alpha)
        h[j] -= eps
        g[j] = (f1 - f0) / eps
    return g

def pgd_optimize(h, n, alpha_start, alpha_end, n_iters, lr_base, target):
    h = h.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    for it in range(n_iters):
        alpha = alpha_start * (alpha_end / alpha_start) ** (it / max(1, n_iters - 1))
        eps = max(1e-4, lr_base * 0.1)
        g = smooth_grad_fd(h, n, alpha, eps)
        t = lr_base
        improved = False
        for _ in range(8):
            h_new = project_to_set(h - t * g, target)
            c5_new = compute_c5(h_new, n)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            f_old = smooth_obj(h, n, alpha)
            f_new = smooth_obj(h_new, n, alpha)
            if f_new < f_old or c5_new < best_c5:
                h = h_new
                improved = True
                break
            t *= 0.5
        if not improved:
            h = project_to_set(h - t * g, target)
    return best_h, best_c5

def coord_search(h, n, rng, n_sweeps, target):
    h = h.copy().astype(float)
    best = compute_c5(h, n)
    for _ in range(n_sweeps):
        order = rng.permutation(n)
        improved = False
        for j in order:
            old = h[j]
            cands = np.array([0.0, 1.0, 0.5,
                              max(0.0, old - 0.15), min(1.0, old + 0.15),
                              max(0.0, old - 0.08), min(1.0, old + 0.08),
                              max(0.0, old - 0.04), min(1.0, old + 0.04),
                              max(0.0, old - 0.01), min(1.0, old + 0.01)])
            bc, bv = old, best
            for c in cands:
                if abs(c - old) < 1e-7:
                    continue
                h[j] = c
                hp = project_to_set(h, target)
                v = compute_c5(hp, n)
                if v < bv:
                    bv = v
                    bc = float(c)
                    h = hp
            if abs(bc - old) > 1e-7:
                h[j] = bc
                h = project_to_set(h, target)
                best = bv
                improved = True
            else:
                h[j] = old
                h = project_to_set(h, target)
        if not improved:
            break
    return h, best

def targeted_pairs(h, n, rng, n_rounds, target):
    h = h.copy().astype(float)
    best = compute_c5(h, n)
    for _ in range(n_rounds):
        corr = compute_corr_array(h, n)
        k_idx = int(np.argmax(corr))
        k = k_idx - (n - 1)
        lo_i = max(0, -k)
        hi_i = min(n, n - k)
        i_best, worst = -1, 0.0
        for i in range(lo_i, hi_i):
            j = i + k
            m = h[i] * (1.0 - h[j])
            if m > worst:
                worst = m
                i_best = i
        if i_best < 0 or worst < 0.03:
            continue
        j_best = i_best + k
        for delta in [0.05, 0.1, 0.15, 0.2]:
            h_try = h.copy()
            if h_try[i_best] > h_try[j_best]:
                h_try[i_best] -= delta
                h_try[j_best] += delta
            else:
                h_try[i_best] += delta
                h_try[j_best] -= delta
            h_try = np.clip(h_try, 0.0, 1.0)
            h_try = project_to_set(h_try, target)
            v = compute_c5(h_try, n)
            if v < best:
                best = v
                h = h_try
                break
    return h, best

def upsample(h, n_new):
    x_old = np.linspace(0.0, 2.0, len(h))
    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, h)

    else:

    return project_to_set(h, n / 2.0)
