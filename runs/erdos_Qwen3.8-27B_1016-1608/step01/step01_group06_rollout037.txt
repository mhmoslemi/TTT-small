        if time.time() - t0 > budget_s * 0.85:
            break
        if restart == 0:
            h_r = project_feasible(best_h + rng.randn(n_best) * 0.02, n_best)
        elif restart == 1:
            h_r = best_h.copy()
            blk = max(2, n_best // 8)
            si = rng.randint(0, n_best - blk)
            h_r[si:si+blk] = 1.0 - h_r[si:si+blk]
            h_r = project_feasible(h_r, n_best)
        elif restart == 2:
            h_r = project_feasible(best_h + rng.randn(n_best) * 0.08, n_best)
        else:
            h_r = project_feasible(best_h * rng.uniform(0.3, 1.7, size=n_best), n_best)
        h_opt, s_opt = _optimize_one(h_r, n_best, t0, budget_s * 0.92, rng)
        if s_opt < best_score:
            best_score = s_opt
            best_h = h_opt

    # Phase 4: Final epoch of very tight refinement
    if time.time() - t0 < budget_s * 0.96 and len(best_h) <= 200:
        n_f = len(best_h)
        _global_n = n_f
        _c_global_n = n_f
        bounds = [(0.0, 1.0)] * n_f
        for tau in [0.5, 0.25, 0.1]:
            if time.time() - t0 > budget_s * 0.98:
                break
            _global_tau = tau
            try:
                res = minimize(_soft_max_obj, best_h, jac=_soft_max_grad,
                              method='L-BFGS-B', bounds=bounds,
                              options={'maxiter': 300, 'ftol': 1e-18, 'gtol': 1e-12})
                h_cand = project_feasible(res['x'], n_f)
                s_cand = compute_score(h_cand)
                if s_cand < best_score:
                    best_score = s_cand
                    best_h = h_cand
            except Exception:
                pass
        cons = [{'type': 'eq', 'fun': _constraint_eq}]
        try:
            res = minimize(_exact_obj, best_h, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': 600, 'ftol': 1e-16})
            h_cand = project_feasible(res['x'], n_f)
            s_cand = compute_score(h_cand)
            if s_cand < best_score:
                best_score = s_cand
                best_h = h_cand
        except Exception:
            pass

    # Ensure feasibility
    best_h = project_feasible(best_h, len(best_h))
    best_score = compute_score(best_h)
    return best_h, best_score, len(best_h)
