    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp((corr - m) * beta)) / (2 * n - 1)) / beta
    pen = 200.0 * (np.sum(x) - n / 2.0) ** 2
    return val + pen

def lbfgs_opt(h0, n, beta, maxiter):
    bnds = [(0.0, 1.0)] * n
    best_h = h0.copy()
    best_c = compute_c5(h0)
    try:
        res = minimize(smooth_obj, h0, method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-11, 'maxls': 40})
        h = repair(np.clip(res.x, 0.0, 1.0), n)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c = h, c
    except Exception:
        pass
    return best_h

def slsqp_obj(x, n):
    return compute_c5(x)

def slsqp_con(x, n):
    return np.sum(x) - n / 2.0

def slsqp_opt(h0, n, maxiter):
    bnds = [(0.0, 1.0)] * n
    best_h = h0.copy()
    best_c = compute_c5(h0)
    try:
        res = minimize(slsqp_obj, h0, method='SLSQP', bounds=bnds,
                       constraints=[{'type': 'eq', 'fun': slsqp_con, 'args': (n,)}],
                       args=(n,), options={'maxiter': maxiter, 'ftol': 1e-16})
        h = repair(np.clip(res.x, 0.0, 1.0), n)
        c = compute_c5(h)
        if c < best_c:
            best_h = h
    except Exception:
        pass
    return best_h

def nelder_opt(h0, n, maxiter):
    bnds = [(0.0, 1.0)] * n
    best_h = h0.copy()
    best_c = compute_c5(h0)
    try:
        res = minimize(slsqp_obj, h0, method='Nelder-Mead', bounds=bnds,
                       args=(n,), options={'maxiter': maxiter, 'xatol': 1e-12, 'fatol': 1e-15})
        h = repair(np.clip(res.x, 0.0, 1.0), n)
        c = compute_c5(h)
        if c < best_c:
            best_h = h
    except Exception:
        pass
    return best_h

def gaussian_perturb(h, n, scale):
    return repair(h + np.random.randn(n) * scale, n)

def sparse_perturb(h, n, scale):
    h_c = h.copy()
    k = np.random.randint(3, max(4, n // 5))
    idx = np.random.choice(n, k, replace=False)
    h_c[idx] += np.random.randn(k) * scale
    return repair(h_c, n)

def flip_perturb(h, n):
    h_c = h.copy()
    k = np.random.randint(1, 4)
    idx = np.random.choice(n, k, replace=False)
    h_c[idx] = 1.0 - h_c[idx]
    return repair(h_c, n)

def block_perturb(h, n):
    h_c = h.copy()
    bs = np.random.randint(4, max(5, n // 4))
    start = np.random.randint(0, n - bs)
    t = np.linspace(-np.pi, np.pi, bs)
    amp = np.random.choice([-1.0, 1.0]) * np.random.uniform(0.05, 0.15)
    h_c[start:start + bs] += amp * np.sin(t)
    return repair(h_c, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_best = np.full(70, 0.5)
    n = len(h_best)
    h_best = repair(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: Initial L-BFGS-B polish
    if time.time() - t0 < budget_s - 40:
        h = lbfgs_opt(h_best, n, 8, 400)
        c = compute_c5(h)
        if c < c5_best:
            h_best, c5_best = h, c

    # Phase 2: Main exploration with simulated annealing
    temp = 0.015
    stall = 0
    while time.time() - t0 < budget_s * 0.70:
        r = np.random.random()
        if r < 0.35:
            scale = temp * (0.6 + 0.4 * np.random.random())
            h_cand = gaussian_perturb(h_best, n, scale)
        elif r < 0.60:
            h_cand = sparse_perturb(h_best, n, temp)
        elif r < 0.80:
            h_cand = block_perturb(h_best, n)
        else:
            h_cand = flip_perturb(h_best, n)

        h_opt = lbfgs_opt(h_cand, n, 12, 350)
        c = compute_c5(h_opt)

        if c < c5_best:
            h_best, c5_best = h_opt, c
            stall = 0
            temp = max(0.001, temp * 0.93)
        else:
            gap = c - c5_best
            if gap < 0.05 and np.random.random() < np.exp(-gap / max(temp, 1e-10)):
                h_best = h_opt
            stall += 1
            if stall > 40:
                temp = min(0.12, temp * 1.2)
                stall = 0

    # Phase 3: Targeted SLSQP from small perturbations
    if time.time() - t0 < budget_s * 0.82:
        for sc in [0.003, 0.008, 0.015, 0.03]:
            if time.time() - t0 > budget_s * 0.81:
                break
            h_p = gaussian_perturb(h_best, n, sc)
            h = slsqp_opt(h_p, n, 800)
            c = compute_c5(h)
            if c < c5_best:
                h_best, c5_best = h, c

    # Phase 4: Nelder-Mead local polish
    if time.time() - t0 < budget_s * 0.88:
        h = nelder_opt(h_best, n, 15000)
        c = compute_c5(h)
        if c < c5_best:
            h_best, c5_best = h, c

    # Phase 5: High-beta L-BFGS-B (nearly exact max)
    if time.time() - t0 < budget_s * 0.94:
        h = lbfgs_opt(h_best, n, 40, 2000)
        c = compute_c5(h)
        if c < c5_best:
            h_best, c5_best = h, c

    # Phase 6: Final SLSQP
    if time.time() - t0 < budget_s * 0.975:
        h = slsqp_opt(h_best, n, 3000)
        c = compute_c5(h)
        if c < c5_best:
            h_best, c5_best = h, c

    # Phase 7: Try other resolutions
    if time.time() - t0 < budget_s * 0.99:
        for n2 in [100, 140, 56]:
            if time.time() - t0 > budget_s * 0.99:
                break
            xp = np.linspace(0.0, 1.0, n)
            xn = np.linspace(0.0, 1.0, n2)
            h_interp = np.interp(xn, xp, h_best)
            h_interp = repair(h_interp, n2)
            h2 = lbfgs_opt(h_interp, n2, 15, 800)
            c2 = compute_c5(h2)
            if c2 < c5_best:
                h_best, c5_best, n = h2, c2, n2
            else:
                h2 = slsqp_opt(h_interp, n2, 500)
                c2 = compute_c5(h2)
                if c2 < c5_best:
                    h_best, c5_best, n = h2, c2, n2

    h_best = repair(h_best, n)
    c5_best = compute_c5(h_best)
    return (h_best, c5_best, n)
