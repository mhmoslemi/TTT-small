import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def repair(h, target):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(3000):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0 - 1e-14
        else:
            m = h > 1e-14
        c = int(m.sum())
        if c == 0:
            break
        h[m] += d / c
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(x, alpha, dx, target):
    hc = np.clip(x, 0.0, 1.0)
    corr = np.correlate(hc, 1.0 - hc, mode="full")
    z = alpha * corr * dx
    z = z - np.max(z)
    return float(np.log(np.sum(np.exp(z))) / alpha)

def smooth_obj_sym(x, alpha, dx):
    nh = len(x)
    n = 2 * nh
    x = np.clip(x, 0.0, 1.0)
    h = np.concatenate([x, 1.0 - x])
    corr = np.correlate(h, 1.0 - h, mode="full")
    z = alpha * corr * dx
    z = z - np.max(z)
    return float(np.log(np.sum(np.exp(z))) / alpha)

def full_c5_sym(x, dx):
    nh = len(x)
    h = np.concatenate([x, 1.0 - x])
    return compute_c5(h, dx)
