    return g

def coord_descent(h, n, n_rounds=80, n_cands=8):
    """Coordinate descent minimizing max correlation."""
    c5_cur = np.max(compute_corr(h))
    for rnd in range(n_rounds):
        improved = False
        perm = np.random.permutation(n)
        for i in perm:
            old_val = h[i]
            cands = {0.0, 1.0, old_val}
            cands.update(np.random.uniform(0, 1, n_cands).tolist())
            best_x, best_c = old_val, c5_cur
            for x in cands:
                if abs(x - old_val) < 1e-10:
                    continue
                h[i] = x
                c = np.max(compute_corr(h))
                if c < best_c - 1e-12:
                    best_c = c
                    best_x = x
            h[i] = best_x
            if best_x != old_val:
                improved = True
        h = repair(h, n)
        c5_cur = np.max(compute_corr(h))
        if not improved:
            break
    return h

def lse_obj(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h) * dx
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(tau * (corr - m)))) / tau

def lse_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h) * dx
    m = np.max(corr)
    w = np.exp(tau * (corr - m))
    w /= np.sum(w)
    g = np.zeros(n)
    for k_idx in range(len(corr)):
        if w[k_idx] > 1e-15:
            g += w[k_idx] * corr_grad(h, k_idx, n) * dx
    return g

def slsqp_smooth(h0, n, tau, maxiter=300):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(lse_obj, h0, jac=lse_grad, args=(tau,), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.fun < lse_obj(h0, tau) + 1e-13:
            return repair(np.clip(res.x, 0.0, 1.0), n)
    except Exception:
        pass
    return h0

def subgradient_steps(h, n, n_steps=200, alpha0=0.003):
    alpha = alpha0
    for _ in range(n_steps):
        corr = compute_corr(h)
        max_c = np.max(corr)
        active = np.where(corr > max_c - 1e-10)[0]
        g = np.zeros(n)
        for k_idx in active:
            g += corr_grad(h, k_idx, n)
        g /= len(active)
        h_new = h - alpha * g
        h_new = repair(h_new, n)
        if compute_c5(h_new) < compute_c5(h):
            h = h_new
        alpha *= 0.998
    return h

def perturb_optimize(h_best, n, scale, tau, maxiter):
    h_p = repair(h_best + np.random.randn(n) * scale, n)
    return slsqp_smooth(h_p, n, tau, maxiter)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_best = np.full(70, 0.5)
    n = len(h_best)
    h_best = repair(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: Coordinate descent
    if time.time() - t0 < budget_s * 0.15:
        h = coord_descent(h_best, n, n_rounds=60)
        c = compute_c5(h)
        if c < c5_best:
            h_best, c5_best = h, c

    # Phase 2: Log-sum-exp SLSQP with annealing tau
    taus = [80, 50, 30, 20, 15, 10, 7, 5, 3, 2, 1.5]
    for tau in taus:
        if time.time() - t0 > budget_s * 0.45:
            break
        h = slsqp_smooth(h_best, n, tau, maxiter=400)
        c = compute_c5(h)
        if c < c5_best:
            h_best, c5_best = h, c

    # Phase 3: Subgradient refinement
    if time.time() - t0 < budget_s * 0.6:
        h = subgradient_steps(h_best, n, n_steps=300, alpha0=0.002)
        c = compute_c5(h)
        if c < c5_best:
            h_best, c5_best = h, c

    # Phase 4: Perturbation + smooth SLSQP (main search)
    scales = [0.002, 0.004, 0.008, 0.015, 0.03, 0.06, 0.1]
    idx = 0
    while time.time() - t0 < budget_s * 0.85:
        scale = scales[idx % len(scales)]
        h = perturb_optimize(h_best, n, scale, tau=15, maxiter=250)
        c = compute_c5(h)
        if c < c5_best:
            h_best, c5_best = h, c
            # Local coordinate descent after improvement
            if time.time() - t0 < budget_s * 0.85:
                h = coord_descent(h_best, n, n_rounds=20)
                c = compute_c5(h)
                if c < c5_best:
                    h_best, c5_best = h, c
        idx += 1

    # Phase 5: Final SLSQP with raw max objective
    if time.time() - t0 < budget_s * 0.93:
        bnds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
                 'jac': lambda x: np.ones(n)}]
        try:
            res = minimize(compute_c5, h_best, method='SLSQP', bounds=bnds,
                           constraints=cons, options={'maxiter': 5000, 'ftol': 1e-16})
            h_new = repair(np.clip(res.x, 0, 1), n)
            c = compute_c5(h_new)
            if c < c5_best:
                h_best, c5_best = h_new, c
        except Exception:
            pass

    # Phase 6: Try other resolutions
    if time.time() - t0 < budget_s * 0.98:
        for n2 in [50, 100, 90]:
            if time.time() - t0 > budget_s * 0.97:
                break
            xp = np.linspace(0, 2, n, endpoint=False)
            xn = np.linspace(0, 2, n2, endpoint=False)
            h_interp = np.interp(xn, xp, h_best)
            h_interp = repair(h_interp, n2)
            if compute_c5(h_interp) < c5_best + 0.002:
                h2 = slsqp_smooth(h_interp, n2, tau=20, maxiter=300)
                c2 = compute_c5(h2)
                if c2 < c5_best:
                    h_best, c5_best, n = h2, c2, n2

    h_best = repair(h_best, n)
    c5_best = compute_c5(h_best)
    return (h_best, c5_best, n)
