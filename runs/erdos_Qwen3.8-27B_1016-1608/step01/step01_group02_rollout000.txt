import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h, dx):
    return float(np.max(compute_corr(h)) * dx)

def projection(h, target):
    h = np.array(h, dtype=float, copy=True)
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        excess = h.sum() - target

    corr = compute_corr(h)
    m = np.max(corr)

    nblocks = rng.randint(3, 8)
    for _ in range(nblocks):
        ln = rng.randint(3, n // 4)
        st = rng.randint(0, max(1, n - ln))
        h[st:st + ln] = 1.0
    return projection(h, n / 2.0)

def gen_fourier(rng, n):
    t = np.linspace(0, 2 * np.pi, n, endpoint=False)
    coeffs = []
    for freq in range(1, 8):
        coeffs.append((rng.uniform(0.05, 0.4), freq, rng.uniform(0, 2 * np.pi)))
    h = 0.5 + np.zeros(n)
    for amp, freq, phase in coeffs:
        h += amp * np.sin(freq * t + phase)
    h = np.clip(h, 0.01, 0.99)
    return projection(h, n / 2.0)

def gen_sparse(rng, n):
    h = np.zeros(n)
    n_ones = n // 2
    indices = rng.choice(n, size=n_ones, replace=False)
    h[indices] = 1.0
    return h

def gen_triang(rng, n):
    tri = np.array([min(i, n - 1 - i) / (n / 2.0) for i in range(n)])
    phase = rng.uniform(0, 2 * np.pi)
    h = (1.0 + np.sin(2 * np.pi * np.arange(n) / n * rng.randint(2, 5) + phase)) / 2.0
    h = h * 0.7 + tri * 0.3
    return projection(h, n / 2.0)

def coord_descend(h, dx, target, rng, n_steps=5000):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(h, dx)
    step = 0.05
    for it in range(n_steps):
        if it % 500 == 0 and step > 0.002:
            step *= 0.75
        idx = rng.randint(n)
        h2 = best.copy()
        h2[idx] = np.clip(h2[idx] + step * rng.choice([-1.0, 1.0]), 0, 1)
        h2 = projection(h2, target)
        c2 = compute_c5(h2, dx)
        if c2 < best_c5 - 1e-15:
            best, best_c5 = h2, c2
    return best, best_c5

def active_set_refine(h, dx, target, m_active=5, iters=50):
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h, dx)
    for _ in range(iters):
        corr = compute_corr(h)
        top_idx = np.argsort(corr)[-m_active:]
        top_vals = corr[top_idx]
        avg_val = np.mean(top_vals)
        # Push down the top correlations via simple gradient step
        grad = np.zeros(n)
        for k in top_idx:
            lag = k - (n - 1)
            if lag >= 0:
                s1, e1 = lag, n
            else:
                s1, e1 = 0, n + lag
            if s1 < e1:
                idx = np.arange(s1, e1)
                grad[idx] += (1.0 - h[idx - lag])
            if lag >= 0:
                s2, e2 = 0, n - lag
            else:
                s2, e2 = -lag, n
            if s2 < e2:
                idx2 = np.arange(s2, e2)
                grad[idx2] -= h[idx2 + lag]
        grad /= m_active
        lr = 0.02
        h_new = h - lr * grad
        h_new = projection(h_new, target)
        c_new = compute_c5(h_new, dx)
        if c_new < best_c5 - 1e-15:
            h, best_c5 = h_new, c_new
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    best_h = np.array(initial_h_values, dtype=float).copy()
    n = len(best_h)
    dx = 2.0 / n
    target = n / 2.0
    best_c5 = compute_c5(best_h, dx)

    def elapsed():
        return time.time() - t0

    def remaining():
        return budget_s - elapsed()

    def update(h_new):
        nonlocal best_h, best_c5
        if h_new is None:
            return
        h_new = projection(h_new, target)
        c5 = compute_c5(h_new, dx)
        if c5 < best_c5 - 1e-16:
            best_h = h_new.copy()
            best_c5 = c5

    # Phase 1: L-BFGS-B from initial with various temperatures
    temps = [5.0, 15.0, 40.0, 120.0, 300.0]
    pens = [50.0, 200.0, 1000.0]
    for temp in temps:
        if remaining() < 30:
            break
        for pen in pens:
            if remaining() < 10:
                break
            h_try = run_lbfgsb(best_h, dx, target, temp, pen, maxiter=1500)
            update(h_try)

    # Phase 2: Random restarts
    generators = [gen_block, gen_fourier, gen_sparse, gen_triang]
    n_restarts = 25
    for i in range(n_restarts):
        if remaining() < 40:
            break
        gen = generators[i % len(generators)]
        h0 = gen(rng, n)
        temp = [10.0, 50.0, 200.0][i % 3]
        h_try = run_lbfgsb(h0, dx, target, temp, 500.0, maxiter=800)
        update(h_try)
        # Also perturb best
        if i % 3 == 0 and remaining() > 60:
            h_pert = projection(best_h + rng.randn(n) * 0.08, target)
            h_try = run_lbfgsb(h_pert, dx, target, 30.0, 500.0, maxiter=600)
            update(h_try)

    # Phase 3: Active set refinement
    if remaining() > 30:
        h_as, c5_as = active_set_refine(best_h, dx, target, m_active=3, iters=100)
        update(h_as)
        h_as, c5_as = active_set_refine(best_h, dx, target, m_active=5, iters=100)
        update(h_as)
        h_as, c5_as = active_set_refine(best_h, dx, target, m_active=8, iters=100)
        update(h_as)

    # Phase 4: Another round of L-BFGS-B polish
    if remaining() > 50:
        for temp in [20.0, 80.0]:
            if remaining() < 20:
                break
            h_try = run_lbfgsb(best_h, dx, target, temp, 2000.0, maxiter=2000)
            update(h_try)

    # Phase 5: Coordinate descent
    if remaining() > 20:
        h_cd, c5_cd = coord_descend(best_h, dx, target, rng, n_steps=8000)
        update(h_cd)

    # Phase 6: Final active set + L-BFGS polish
    if remaining() > 30:
        h_as, _ = active_set_refine(best_h, dx, target, m_active=4, iters=50)
        update(h_as)
        if remaining() > 15:
            h_try = run_lbfgsb(best_h, dx, target, 50.0, 5000.0, maxiter=1000)
            update(h_try)

    best_h = projection(best_h, target)
    final_c5 = compute_c5(best_h, dx)
    return (best_h, final_c5, n)
