    return np.correlate(h, 1.0 - h, mode='full') * dx

def _fix(h, n):
    t = 0.5 * n
    h = h.copy()
    for _ in range(300):
        d = t - np.sum(h)
        if abs(d) < 1e-15:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        k = np.sum(mask)
        if k == 0:
            break
        h[mask] += d / k
        h = np.clip(h, 0.0, 1.0)
    return h

def _grad_c(h, c, dx, n, alpha):

    m = np.max(c)
    w = np.exp(np.clip(alpha * (c - m), -50, 50))

import numpy as np
import time

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _fix(h, n):
    t = 0.5 * n
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        d = t - np.sum(h)
        if abs(d) < 1e-15:
            break
        mask = (h < 1.0) if d > 0 else (h > 0.0)
        k = float(np.sum(mask))
        if k < 1e-12:
            break
        h[mask] += d / k
        h = np.clip(h, 0.0, 1.0)
    return h

def _grad_c(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(c)
    wc = np.exp(np.clip(alpha * (c - m), -60, 60))
    wc /= np.sum(wc)
    grad = np.zeros(n)
    shifts = np.arange(2 * n - 1) - (n - 1)
    for k in range(len(c)):
        wk = wc[k]
        if wk < 1e-14:
            continue
        s = shifts[k]
        wkd = wk * dx
        if s >= 0:
            j0, j1 = 0, n - s
        else:
            j0, j1 = -s, n
        if j1 > j0:
            grad[j0:j1] += wkd * (1.0 - h[j0+s:j1+s])
        if s >= 0:
            j0n, j1n = s, n
        else:
            j0n, j1n = 0, n + s
        if j1n > j0n:
            grad[j0n:j1n] -= wkd * h[j0n-s:j1n-s]
    return grad, m

def _interp_h(h, target_n):
    nc = len(h)
    if nc == target_n:
        return h.copy()
    xo = np.arange(nc) * (2.0 / nc)
    xn = np.arange(target_n) * (2.0 / target_n)
    return _fix(np.clip(np.interp(xn, xo, h), 0, 1), target_n)

def _pg_opt(h, alpha, max_it, eta0, t0, budget, rng):
    nn = len(h)
    best = _c5(h)
    hbest = h.copy()
    eta = eta0
    v = np.zeros(nn)
    mu = 0.82
    stalled = 0
    for it in range(max_it):
        if budget - (time.time() - t0) < 5:
            break
        grad, _ = _grad_c(h, nn, alpha)
        v = mu * v - eta * grad
        h_new = _fix(np.clip(h + v, 0, 1), nn)
        c5_new = _c5(h_new)
        if c5_new < best:
            h = h_new
            best = c5_new
            hbest = h.copy()
            eta = min(eta * 1.03, eta0 * 5)
            stalled = 0
        else:
            stalled += 1
            eta *= 0.6
            if stalled > 80:
                npert = max(5, nn // 50)
                idx = rng.choice(nn, size=npert, replace=False)
                h[idx] = np.clip(h[idx] + rng.randn(npert) * 0.5, 0, 1)
                h = _fix(h, nn)
                v *= 0.2
                eta = eta0
                stalled = 0
    return hbest, best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        bh = np.array(initial_h_values, dtype=float).copy()
    else:
        bh = np.full(400, 0.5)
    n = len(bh)
    bh = _fix(bh, n)
    bc = _c5(bh)

    def _rem():
        return budget_s - (time.time() - t0)

    # Phase 1: Optimize at current resolution with increasing alpha
    for alpha in [30, 100, 500, 2000, 8000]:
        if _rem() < 25:
            break
        htrial = bh.copy()
        hb, cb = _pg_opt(htrial, alpha, 6000, 0.08, t0, budget_s, rng)
        if cb < bc:
            bh = hb
            bc = cb

    # Phase 2: Try different resolutions
    for tn in [200, 256, 300, 500, 600]:
        if _rem() < 35:
            break
        hi = _interp_h(bh, tn)
        c5_i = _c5(hi)
        best_trial = c5_i
        best_h = hi.copy()
        for alpha in [200, 1000, 4000]:
            if _rem() < 10:
                break
            hm = best_h.copy()
            hb, cb = _pg_opt(hm, alpha, 4000, 0.06, t0, budget_s, rng)
            if cb < best_trial:
                best_trial = cb
                best_h = hb
        if best_trial < bc:
            bh = best_h
            bc = best_trial

    # Phase 3: Multi-start from perturbed versions of best
    nf = len(bh)
    for restart in range(4):
        if _rem() < 30:
            break
        hp = bh.copy()
        npert = max(10, nf // 10)
        idx = rng.choice(nf, size=npert, replace=False)
        hp[idx] = np.clip(hp[idx] + rng.randn(npert) * 0.6, 0, 1)
        hp = _fix(hp, nf)
        alpha = 3000 + restart * 2000
        hb, cb = _pg_opt(hp, alpha, 5000, 0.05, t0, budget_s, rng)
        if cb < bc:
            bh = hb
            bc = cb

    # Phase 4: Final fine-tuning with very small steps
    nf = len(bh)
    eta = 0.01
    v = np.zeros(nf)
    for it in range(4000):
        if _rem() < 3:
            break
        grad, _ = _grad_c(bh, nf, 10000)
        v = 0.7 * v - eta * grad
        bh_new = _fix(np.clip(bh + v, 0, 1), nf)
        c5_n = _c5(bh_new)
        if c5_n < bc:
            bh = bh_new
            bc = c5_n
            eta = min(eta * 1.01, 0.05)
        else:
            eta *= 0.85

    # Ensure exact integral
    bh = _fix(bh, len(bh))
    bc = _c5(bh)
    return bh, bc, len(bh)
