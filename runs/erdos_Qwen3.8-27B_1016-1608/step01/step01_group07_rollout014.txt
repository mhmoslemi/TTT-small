from scipy.optimize import minimize
import time

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def _fix(h, n):
    t = 0.5 * n
    for _ in range(300):
        d = t - np.sum(h)
        if abs(d) < 1e-15:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        k = int(np.sum(mask))
        if k == 0:
            break
        h = h + (d / k) * mask
        h = np.clip(h, 0.0, 1.0)
    return h

def _interp(h, n_new):
    n0 = len(h)
    xo = np.arange(n0) * (2.0 / n0)
    xn = np.arange(n_new) * (2.0 / n_new)
    return _fix(np.clip(np.interp(xn, xo, h), 0.0, 1.0), n_new)

def _obj_lse(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(c)
    e = np.exp(np.clip(c - m, -50.0, 50.0))
    return float(m + np.log(np.sum(e)) / alpha)

    m = np.max(c)

    return float(m + 0.15 * np.mean(srt[:k]))

                      bounds=bnds, constraints=cons,

def _subgrad_step(h, n, lr, top_k=5, tol=0.005):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(c)
    active = np.where(c >= m - tol)[0]
    active = active[np.argsort(c[active])[::-1][:top_k]]
    g = np.zeros(n)
    for k in active:
        shift = int(k) - (n - 1)
        if shift >= 0:
            g[0:n-shift] += (1.0 - h[shift:n])
            g[shift:n] -= h[0:n-shift]
        else:
            s = -shift
            g[s:n] += (1.0 - h[0:n-s])
            g[0:n-s] -= h[s:n]
    g *= dx
    gn = np.linalg.norm(g)
    if gn < 1e-14:
        return h
    return h - lr * g / gn
