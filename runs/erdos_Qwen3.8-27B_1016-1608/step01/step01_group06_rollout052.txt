def compute_c(h):

    return np.correlate(h, 1.0 - h, mode='full') * dx

def score(h):
    return np.max(compute_c(h))

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):

        if abs(d) < 1e-15: break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        if not m.any(): break
        h = np.clip(h + d / m.sum() * m, 0.0, 1.0)
    return h

def grad_max_lag(h, K=5):
    """Compute average gradient of top-K lag objectives."""
    n = len(h)
    c = compute_c(h)
    # Get top K lags
    top_idx = np.argsort(c)[-K:]
    grad = np.zeros(n)
    for idx in top_idx:
        lag = idx - (n - 1)
        if lag >= 0:
            grad[:n-lag] += (1.0 - h[lag:])
            grad[lag:] -= h[:n-lag]
        else:
            m = -lag
            grad[m:] += (1.0 - h[:n-m])
            grad[:n-m] -= h[m:]
    return grad / K

def projected_grad_opt(h, n, t0, budget, max_iter=5000):

    _global_n = n
    h = project(h.copy(), n)

    alpha = 0.01
