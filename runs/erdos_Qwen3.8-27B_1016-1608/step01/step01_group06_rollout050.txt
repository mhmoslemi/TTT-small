    return c

    return np.max(c)

    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):

        if abs(diff) < 1e-14:

        if diff > 0:

        else:

    c = compute_correlations(h)
    alpha = 0.015
    mom = np.zeros(n)
    beta = 0.75
    rng = np.random.RandomState(12345)
    for it in range(50000):
        if time.time() - t0 > tlim:
            break
        cm = np.max(c)
        tol = max(0.003, 0.015 * cm)
        active = np.where(c >= cm - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        g = compute_subgrad(h, active, n)
        g -= np.mean(g)
        mom = beta * mom + (1.0 - beta) * g
        h_new = project_feasible(h - alpha * mom, n)
        c_new = compute_correlations(h_new)
        s_new = np.max(c_new)
        if s_new < best_s - 1e-12:
            best_s = s_new
            best_h = h_new.copy()
            alpha = min(0.04, alpha * 1.01)
            mom *= beta
        else:
            alpha *= 0.93
            if alpha < 0.0005:
                h_new = project_feasible(h + rng.randn(n) * 0.08, n)
                c_new = compute_correlations(h_new)
                alpha = 0.015
                mom = np.zeros(n)
        h = h_new
        c = c_new
    return best_h, best_s

def coord_descent(h, n, t0, tlim, max_sweeps=60):
    best_h = h.copy()
    best_s = compute_score(h)
    c = compute_correlations(h)
    dx = 2.0 / n
    nm1 = n - 1
    for sweep in range(max_sweeps):
        if time.time() - t0 > tlim:
            break
        for j in range(n):
            A_raw = np.zeros(2 * n - 1)
            for idx in range(2 * n - 1):
                k = idx - nm1
                coeff = 0.0
                nj = j - k
                if 0 <= nj < n:
                    coeff += (1.0 - h[nj])
                nj2 = j + k
                if 0 <= nj2 < n:
                    coeff += -h[nj2]
                A_raw[idx] = coeff * dx
            B_raw = c - A_raw * h[j]
            lo, hi = 0.0, 1.0
            for _ in range(50):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                if np.max(A_raw * m1 + B_raw) < np.max(A_raw * m2 + B_raw):
                    hi = m2
                else:
                    lo = m1
            h[j] = (lo + hi) / 2.0
            h = np.clip(h, 0.0, 1.0)
            c = compute_correlations(h)
        h = project_feasible(h, n)
        c = compute_correlations(h)
        s = np.max(c)
        if s < best_s - 1e-12:
            best_s = s
            best_h = h.copy()
    return best_h, best_s

def thue_morse_h(n):
    tm = np.zeros(n, dtype=int)
    for i in range(1, n):
        tm[i] = tm[i >> 1] ^ (i & 1)
    h = 0.5 + 0.45 * (2.0 * tm - 1.0)
    return project_feasible(h, n)

def smooth_obj(h, tau=200.0):
    n = len(h)
    c = compute_correlations(h)
    cm = np.max(c)
    return cm + np.log(np.sum(np.exp(tau * (c - cm)))) / tau

def lagrangian_cd(h, n, t0, tlim, max_iter=40):
    best_h = h.copy()
    best_s = compute_score(h)
    c = compute_correlations(h)
    dx = 2.0 / n
    nm1 = n - 1
    lam = 0.0
    eta = 0.3
    for it in range(max_iter):
        if time.time() - t0 > tlim:
            break
        for j in range(n):
            A_raw = np.zeros(2 * n - 1)
            for idx in range(2 * n - 1):
                k = idx - nm1
                coeff = 0.0
                nj = j - k
                if 0 <= nj < n:
                    coeff += (1.0 - h[nj])
                nj2 = j + k
                if 0 <= nj2 < n:
                    coeff += -h[nj2]
                A_raw[idx] = coeff * dx
            B_raw = c - A_raw * h[j]
            C = A_raw + lam
            D = B_raw
            lo, hi = 0.0, 1.0
            for _ in range(50):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                if np.max(C * m1 + D) < np.max(C * m2 + D):
                    hi = m2
                else:
                    lo = m1
            h[j] = (lo + hi) / 2.0
            h = np.clip(h, 0.0, 1.0)
            c = compute_correlations(h)
        lam += eta * (np.sum(h) - n / 2.0)
        lam = np.clip(lam, -5.0, 5.0)
        h = project_feasible(h, n)
        c = compute_correlations(h)
        s = np.max(c)
        if s < best_s - 1e-12:
            best_s = s
            best_h = h.copy()
    return best_h, best_s

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    best_h = h_init.copy()
    best_s = compute_score(h_init)
    x_ref = np.linspace(0.0, 2.0, n_init, endpoint=False)

    n_list = sorted(set([n_init, 60, 80, 100]))
    time_per_n = budget_s * 0.75 / len(n_list)

    for n in n_list:
        if time.time() - t0 > budget_s * 0.9:
            break
        t_start = time.time()
        tlim_block = min(t0 + budget_s * 0.92, t_start + time_per_n)

        if n == n_init:
            h0 = h_init.copy()
        else:
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h0 = np.interp(x_new, x_ref, best_h)
            h0 = project_feasible(h0, n)

        h_tm = thue_morse_h(n)

        candidates = [h0, h_tm, project_feasible(rng.uniform(0, 1, n), n)]
        if n > 20:
            h_flip = project_feasible(h0 + rng.randn(n) * 0.2, n)
            candidates.append(h_flip)

        h_best_n = h0.copy()
        s_best_n = compute_score(h0)

        for h_cand in candidates:
            if time.time() > tlim_block * 0.3:
                break
            h1, s1 = subgrad_optimize(h_cand.copy(), n, t0, tlim_block * 0.3)
            if s1 < s_best_n:
                s_best_n = s1
                h_best_n = h1.copy()
            if time.time() < tlim_block * 0.5:
                h2, s2 = coord_descent(h1.copy(), n, t0, tlim_block * 0.5, max_sweeps=40)
                if s2 < s_best_n:
                    s_best_n = s2
                    h_best_n = h2.copy()

        if time.time() < tlim_block * 0.7:
            h3, s3 = lagrangian_cd(h_best_n.copy(), n, t0, tlim_block * 0.7, max_iter=30)
            if s3 < s_best_n:
                s_best_n = s3
                h_best_n = h3.copy()

        if time.time() < tlim_block * 0.9 and s_best_n < best_s + 0.005:
            h4, s4 = coord_descent(h_best_n.copy(), n, t0, tlim_block * 0.9, max_sweeps=30)
            if s4 < s_best_n:
                s_best_n = s4
                h_best_n = h4.copy()

        if s_best_n < best_s:
            best_s = s_best_n
            best_h = h_best_n.copy()

    # Final polish with SLSQP on smooth objective
    if time.time() < t0 + budget_s * 0.95:
        nf = len(best_h)
        bounds = [(0.0, 1.0)] * nf
        sum_target = nf / 2.0
        def _con(h):
            return np.sum(h) - sum_target
        constraints = [{'type': 'eq', 'fun': _con}]
        tlim_final = t0 + budget_s * 0.97
        try:
            res = minimize(smooth_obj, best_h.copy(), method='SLSQP',
                           bounds=bounds, constraints=constraints,
                           options={'maxiter': 300, 'ftol': 1e-14})
            hf = project_feasible(res['x'], nf)
            sf = compute_score(hf)
            if sf < best_s:
                best_s = sf
                best_h = hf
        except Exception:
            pass
        # One more coord descent pass
        if time.time() < tlim_final:
            hf2, sf2 = coord_descent(best_h.copy(), nf, t0, tlim_final, max_sweeps=15)
            if sf2 < best_s:
                best_s = sf2
                best_h = hf2

    best_h = project_feasible(best_h, len(best_h))
    best_s = compute_score(best_h)
    return best_h, best_s, len(best_h)
