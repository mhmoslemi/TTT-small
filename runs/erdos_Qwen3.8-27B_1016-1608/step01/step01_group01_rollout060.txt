def slsqp_run(h, n, t_limit, temp=0.005):
    t0 = time.time()
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        return m + temp * np.log(np.sum(np.exp((c - m) / temp)))
    def obj_grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        p = np.exp((c - m) / temp) / np.sum(np.exp((c - m) / temp))
        g = np.zeros_like(x)
        for k, pk in enumerate(p):
            shift = k - (n - 1)
            if shift >= 0:
                idx = np.arange(n - shift)
                g[idx] += pk * (1.0 - x[idx + shift])
                g[idx + shift] -= pk * x[idx]
            else:
                idx = np.arange(-shift)
                g[idx + (-shift)] += pk * (1.0 - x[idx])
                g[idx] -= pk * x[idx + (-shift)]
        return g * (2.0 / n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0, 'jac': lambda x: np.ones(n)}]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n))

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        d = n / 2.0 - h.sum()
        if abs(d) < 1e-13:
            break
        h += d / n
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_obj(x, temp):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    return (m + temp * np.log(np.sum(np.exp((c - m) / temp)))) * (2.0 / len(x))

def slsqp_opt(h, n, temp, maxiter=3000):
    def obj(x):
        return soft_max_obj(x, temp)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    res = minimize(obj, h.copy(), method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
    return project_feasible(res.x, n)

def anneal_2pt(h, n, budget, rng, dmax, T_init):
    t0 = time.time()
    h = h.copy()
    bh = h.copy()
    bc5 = compute_c5(h)
    cc5 = bc5
    T = T_init
    while time.time() - t0 < budget:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(-dmax, dmax)
        hi, hj = h[i] + d, h[j] - d
        if hi < 0 or hi > 1 or hj < 0 or hj > 1:
            continue
        h[i], h[j] = hi, hj
        nc5 = compute_c5(h)
        if nc5 < cc5 or rng.random() < np.exp(-(nc5 - cc5) / (T + 1e-15)):
            cc5 = nc5
            if nc5 < bc5:
                bc5 = nc5
                bh = h.copy()
        else:
            h[i], h[j] = hi - d, hj + d
        T *= 0.9997
        if T < 1e-8:
            T = T_init * 0.3
            h = bh + rng.normal(0, 0.02, n)
            h = project_feasible(h, n)
            cc5 = compute_c5(h)
    return bh, bc5

def coord_2pt(h, n, budget, rng):
    t0 = time.time()
    bh = h.copy()
    bc5 = compute_c5(h)
    h = h.copy()
    scales = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 5e-4, 2e-4, 1e-4, 5e-5, 2e-5]
    for scale in scales:
        if time.time() - t0 > budget:
            break
        imp = True
        while imp and time.time() - t0 < budget:
            imp = False
            for i in range(n):
                if imp or time.time() - t0 > budget:
                    break
                for j in range(n):
                    if i == j:
                        continue
                    for d in (scale, -scale):
                        hi, hj = h[i] + d, h[j] - d
                        if hi < 0 or hi > 1 or hj < 0 or hj > 1:
                            continue
                        h[i], h[j] = hi, hj
                        nc5 = compute_c5(h)
                        if nc5 < bc5 - 1e-14:
                            bc5 = nc5
                            bh = h.copy()
                            imp = True
                        else:
                            h[i], h[j] = hi - d, hj + d
                        if imp:
                            break
                    if imp:
                        break
            if imp:
                h = bh.copy()
    # 1-point moves
