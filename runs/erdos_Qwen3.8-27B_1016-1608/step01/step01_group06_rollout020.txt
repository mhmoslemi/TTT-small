        h_interp = project_feasible(h_interp, n_cand)
        s_interp = compute_score(h_interp)
        
        # Quick optimization
        h_opt, s_opt = slsqp_optimize(h_interp, n_cand, t0, budget_s)
        if s_opt < best_score:
            best_score = s_opt
            # Convert back to best_h representation
            best_h_n = h_opt.copy()
            # Keep track of n
            n_best = n_cand
            # Re-interpolate to n_init for comparison (or just update)
            best_h = h_opt
            best_score = s_opt
            print(f"n={n_cand}: C5={best_score:.8f}")
        
        # Gradient descent on this n
        if time.time() - t0 < budget_s * 0.85:
            h_gd, s_gd = custom_grad_descent(h_opt, n_cand, max_iter=1500,
                                             t_start=time.time(), t_end=budget_s)
            if s_gd < best_score:
                best_score = s_gd
                best_h = h_gd
                print(f"n={n_cand} GD: C5={best_score:.8f}")
    
    # Phase 3: Multi-restart from best
    if time.time() - t0 < budget_s * 0.88:
        n_best = len(best_h)
        for r in range(8):
            if time.time() - t0 > budget_s * 0.93:
                break
            h_pert = project_feasible(best_h + rng.randn(n_best) * 0.03, n_best)
            h_cand, s_cand = slsqp_optimize(h_pert, n_best, t0, budget_s)
            if s_cand > best_score - 0.005:
                h_cand, s_cand = custom_grad_descent(h_cand, n_best, max_iter=500,
                                                     t_start=time.time(), t_end=budget_s)
            if s_cand < best_score:
                best_score = s_cand
                best_h = h_cand
                print(f"Restart {r}: C5={best_score:.8f}")
    
    # Phase 4: Final polish - coordinate descent with tiny steps
    if time.time() - t0 < budget_s * 0.96:
        n_f = len(best_h)
        h_cand, s_cand = coordinate_descent(best_h, n_f, passes=50, eps=0.005)
        if s_cand < best_score:
            best_score = s_cand
            best_h = h_cand
    
    # Verify
    best_h = project_feasible(best_h, len(best_h))
    best_score = compute_score(best_h)
    print(f"Final: n={len(best_h)}, C5={best_score:.8f}")
    
    return best_h, best_score, len(best_h)
