def corr_full(h):

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_weighted(h, temp):
    c = corr_full(h)
    n = len(h)
    c_max = c.max()
    w = np.exp((c - c_max) / temp)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(n)
    w = w / ws
    grad = np.zeros(n)
    for k in range(len(c)):
        wk = w[k]
        if wk < 1e-10:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            m = n - lag
            if m > 0:
                grad[:m] += wk * (1.0 - h[lag:lag + m])
                grad[lag:lag + m] -= wk * h[:m]
        else:
            L = -lag
            m = n - L
            if m > 0:
                grad[L:L + m] += wk * (1.0 - h[:m])
                grad[:m] -= wk * h[L:L + m]
    return grad

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    h0 = project_feasible(h0, n)

    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]

    # Phase 1: Weighted subgradient with geometric annealing
    alpha = 0.15
    for it in range(12000):
        if time.time() - t_start > budget_s * 0.30:
            break
        frac = it / 12000.0
        temp = 0.08 * (0.01) ** frac
        grad = subgrad_weighted(best_h, temp)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        step = alpha / (1.0 + gnorm)
        h_trial = best_h - step * grad
        h_trial = project_feasible(h_trial, n)
        c5_t = compute_c5(h_trial)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_trial.copy()
        elif it % 400 == 399:
            alpha *= 0.88

    # Phase 2: SLSQP multi-temperature polishing
    for temp in [0.004, 0.002, 0.001, 0.0005]:
        if time.time() - t_start > budget_s * 0.50:
            break
        def obj_sm(hv, t=temp):
            c = corr_full(hv)
            m = c.max()
            return m + t * np.log(np.sum(np.exp((c - m) / t)))
        res = minimize(obj_sm, best_h.copy(), method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': 3000, 'ftol': 1e-14})
        h_new = project_feasible(res.x, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 3: Swap-based combinatorial search
    if time.time() - t_start < budget_s * 0.65:
        for sweep in range(500):
            if time.time() - t_start > budget_s * 0.65:
                break
            improved = False
            for i in range(n):
                for j in range(i + 1, n):
                    if best_h[i] == best_h[j]:
                        continue
                    h_trial = best_h.copy()
                    h_trial[i], h_trial[j] = h_trial[j], h_trial[i]
                    c5_t = compute_c5(h_trial)
                    if c5_t < best_c5 - 1e-14:
                        best_c5 = c5_t
                        best_h = h_trial.copy()
                        improved = True
            if not improved:
                break

    # Phase 4: Transfer-based local search (shift mass between pairs)
    if time.time() - t_start < budget_s * 0.78:
        for delta in [0.05, 0.03, 0.01, 0.005]:
            if time.time() - t_start > budget_s * 0.78:
                break
            for sweep in range(200):
                if time.time() - t_start > budget_s * 0.78:
                    break
                improved = False
                for i in range(n):
                    for j in range(n):
                        if i == j:
                            continue
                        if best_h[i] < delta or best_h[j] > 1.0 - delta:
                            continue
                        h_trial = best_h.copy()
                        h_trial[i] -= delta
                        h_trial[j] += delta
                        c5_t = compute_c5(h_trial)
                        if c5_t < best_c5 - 1e-14:
                            best_c5 = c5_t
                            best_h = h_trial.copy()
                            improved = True
                if not improved:
                    break

    # Phase 5: Basin hopping with SLSQP
    for i in range(40):
        if time.time() - t_start > budget_s * 0.90:
            break
        scale = 0.08 + 0.12 * rng.random()
        h_pert = best_h + rng.normal(0, scale, n)
        h_pert = project_feasible(h_pert, n)
        res = minimize(lambda x: compute_c5(x), h_pert, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': 800, 'ftol': 1e-12})
        h_new = project_feasible(res.x, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 6: Fine coordinate descent
    if time.time() - t_start < budget_s * 0.97:
        deltas = [0.01, -0.01, 0.005, -0.005, 0.002, -0.002, 0.001, -0.001]
        for sweep in range(20):
            if time.time() - t_start > budget_s * 0.97:
                break
            improved = True
            while improved and time.time() - t_start < budget_s * 0.97:
                improved = False
                for i in range(n):
                    for d in deltas:
                        h_trial = best_h.copy()
                        h_trial[i] = np.clip(h_trial[i] + d, 0.0, 1.0)
                        h_trial = project_feasible(h_trial, n)
                        c5_t = compute_c5(h_trial)
                        if c5_t < best_c5 - 1e-14:
                            best_c5 = c5_t
                            best_h = h_trial.copy()
                            improved = True
                            break
                    if improved:
                        break

    # Final feasibility
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, len(best_h))
