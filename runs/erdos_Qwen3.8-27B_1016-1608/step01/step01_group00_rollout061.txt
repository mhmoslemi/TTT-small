```python
    # contribution: -delta * h[i-k+(n-1)] for valid k
    
    eff = np.zeros(2*n-1)
    # First term: i as j in sum
    k_start1 = max(0, n-1-i)
    k_end1 = min(2*n-2, 2*n-2-i)
    if k_start1 <= k_end1:
        idx = np.arange(k_start1, k_end1+1)
        h_idx = i + idx - (n-1)  # = n-1-i+k_start1 to ... 
        # h_idx goes from i + k_start1 - (n-1) to i + k_end1 - (n-1)
        # = i + max(0,n-1-i) - (n-1) to i + min(2n-2,2n-2-i) - (n-1)
        eff[k_start1:k_end1+1] += delta * (1 - h[h_idx])
    
    # Second term: i = j+k-(n-1), so j = i-k+(n-1)
    k_start2 = max(0, i-(n-1))
    k_end2 = min(2*n-2, i+(n-1))
    if k_start2 <= k_end2:
        idx = np.arange(k_start2, k_end2+1)
        j_idx = i - idx + (n-1)
        eff[k_start2:k_end2+1] -= delta * h[j_idx]
    
    return eff
```

```python
    j1 = i_idx - k_idx + (n-1)  # (n, 2n-1)
    
    valid2 = (j2 >= 0) & (j2 < n)
    valid1 = (j1 >= 0) & (j1 < n)
    
    # Gather h values (with 0 padding for invalid)
    j2_clamped = np.clip(j2, 0, n-1)
    j1_clamped = np.clip(j1, 0, n-1)
    
    term1 = np.where(valid2, 1.0 - h[j2_clamped], 0.0)  # (n, 2n-1)
    term2 = np.where(valid1, h[j1_clamped], 0.0)  # (n, 2n-1)
    
    effects = deltas[:, None] * (term1 - term2)  # (n, 2n-1)
    return effects
```

This computes all n flip effects in one vectorized operation. Then for each flip i, the new C₅ is max(full_corr + effects[i]) * dx.

To find the best flip: new_c5[i] = max_k (full_corr[k] + effects[i,k]) * dx

This is a (n, 2n-1) matrix, and we take max along axis=1. Very fast.

```python
def repair(h, n):
```

```python
def flip_effects_matrix(h):
    """Compute (n, 2n-1) matrix of individual flip effects for all positions."""
    n = len(h)
    deltas = 1.0 - 2.0 * h  # +1 for 0->1, -1 for 1->0
    i_idx = np.arange(n)[:, None]  # (n,1)
    k_idx = np.arange(2 * n - 1)[None, :]  # (1, 2n-1)
    j2 = i_idx + k_idx - (n - 1)
    j1 = i_idx - k_idx + (n - 1)
    valid2 = (j2 >= 0) & (j2 < n)
    valid1 = (j1 >= 0) & (j1 < n)
    j2c = np.clip(j2, 0, n - 1)
    j1c = np.clip(j1, 0, n - 1)
    term1 = np.where(valid2, 1.0 - h[j2c], 0.0)
    term2 = np.where(valid1, h[j1c], 0.0)
    return deltas[:, None] * (term1 - term2)
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not np.any(mask):
            h[:] = target / n
            break
        h[mask] += diff / float(np.sum(mask))
        h = np.clip(h, 0.0, 1.0)
    return h

def binarize(h, n):

    idx = np.argsort(-h)

def flip_effects_matrix(h):
    n = len(h)
    deltas = 1.0 - 2.0 * h
    i_idx = np.arange(n)[:, None]
    k_idx = np.arange(2 * n - 1)[None, :]
    j2 = i_idx + k_idx - (n - 1)
    j1 = i_idx - k_idx + (n - 1)
    valid2 = (j2 >= 0) & (j2 < n)
    valid1 = (j1 >= 0) & (j1 < n)
    j2c = np.clip(j2, 0, n - 1)
    j1c = np.clip(j1, 0, n - 1)
    term1 = np.where(valid2, 1.0 - h[j2c], 0.0)
    term2 = np.where(valid1, h[j1c], 0.0)
    effects = deltas[:, None] * (term1 - term2)
    effects[:, n - 1] -= deltas ** 2
    return effects

    corr = np.correlate(x, 1.0 - x, mode='full')
    return float(np.max(corr) * (2.0 / n))

def _opt_con(x, n):
    return np.sum(x) - n / 2.0

def slsqp_opt(h0, n, maxiter=500, ftol=1e-14):
    bnds = [(0.0, 1.0)] * n
```
