    eq_con = ({'type': 'eq', 'fun': constraint_fun},)
    try:
        res = minimize(smooth_c5, h0, args=(dx, temp), method='SLSQP',
                       bounds=bounds, constraints=eq_con,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return projection(res.x, target)
    except Exception:
        return None

def subgradient_phase(h, dx, target, time_limit, rng):
    h = h.copy()
    n = len(h)
    t_start = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    schedules = [
        (0.02, 3, 150), (0.01, 5, 150), (0.005, 5, 200),
        (0.003, 8, 200), (0.001, 3, 300), (0.002, 5, 300),
        (0.001, 5, 300), (0.0005, 3, 400), (0.001, 8, 300),
    ]
    
    for ss, top_m, n_iters in schedules:
        for it in range(n_iters):
            if time.time() - t_start > time_limit:
                return best_h, best_c5
            corr_vals = compute_all_corrs(h, dx)
            c5_cur = float(np.max(corr_vals))
            if c5_cur < best_c5 - 1e-15:
                best_h = h.copy()
                best_c5 = c5_cur
            
            top_idx = np.argsort(corr_vals)[-top_m:]
            w = np.exp((corr_vals[top_idx] - np.max(corr_vals[top_idx])) * 200.0)
            wsum = w.sum()
            if wsum < 1e-15:
                w = np.ones(top_m) / top_m
            else:
                w /= wsum
            
            grad = np.zeros(n)
            for wi, ki in zip(w, top_idx):
                s = int(ki) - (n - 1)
                grad += wi * compute_grad_k(h, s, dx)
            
            h_try = projection(h - ss * grad, target)
            c5_try = compute_c5(h_try, dx)
            if c5_try < c5_cur:
                h = h_try
            else:
                h_try2 = projection(h - ss * 0.4 * grad, target)
                c5_try2 = compute_c5(h_try2, dx)
                if c5_try2 < c5_cur:
                    h = h_try2
                elif it % 30 == 29:
                    h = projection(h + rng.randn(n) * 0.02, target)
    
    return best_h, best_c5

def pair_search(h, dx, target, time_limit, rng):
    h = h.copy()
    n = len(h)
    c5_cur = compute_c5(h, dx)
    best_h = h.copy()
    best_c5 = c5_cur
    t_start = time.time()
    temp = 0.003
    n_pres = 139  # number of correlation values
    
   corr_all = compute_all_corrs(h, dx)
    
    while time.time() - t_start < time_limit:
        temp = max(temp * 0.9995, 1e-7)
        n_coords = int(rng.choice([1, 2, 3, 4], p=[0.3, 0.3, 0.25, 0.15]))
        idxs = rng.choice(n, size=n_coords, replace=False)
        h_new = h.copy()
        for idx in idxs:
            d = rng.uniform(-0.2, 0.2)
            h_new[idx] = np.clip(h_new[idx] + d, 0.0, 1.0)
        h_new = projection(h_new, target)
        corr_new = compute_all_corrs(h_new, dx)
        c5_new = float(np.max(corr_new))
        
        if c5_new < c5_cur or rng.random() < np.exp(-(c5_new - c5_cur) / temp):
            h = h_new
            c5_cur = c5_new
            if c5_new < best_c5:
                best_h = h.copy()
                best_c5 = c5_new
    
    return best_h, best_c5

def balance_top_shifts(h, dx, target, rng, n_iters=500):
    """Targeted reduction of top correlation shifts"""
    h = h.copy()
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    for it in range(n_iters):
        corr_vals = compute_all_corrs(h, dx)
        c5_cur = float(np.max(corr_vals))
        if c5_cur < best_c5 - 1e-15:
            best_h = h.copy()
            best_c5 = c5_cur
        
        top_idx = int(np.argmax(corr_vals))
        s = top_idx - (n - 1)
        g = compute_grad_k(h, s, dx)
        
        # Find coordinate where reducing helps most
        step = rng.uniform(0.005, 0.03)
        h_new = projection(h - step * g, target)
        c5_new = compute_c5(h_new, dx)
        if c5_new < c5_cur:
            h = h_new
    
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    global _GLOBAL_TARGET
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    best_h = np.array(initial_h_values, dtype=float).copy()
    n = len(best_h)
    dx = 2.0 / n
    target = n / 2.0
    _GLOBAL_TARGET = target
    best_c5 = compute_c5(best_h, dx)
    bounds = [(0.0, 1.0)] * n
    
    def tl():
        return budget_s - (time.time() - t0)
    
    def update(h_new):
        nonlocal best_h, best_c5
        if h_new is None:
            return
        c5 = compute_c5(h_new, dx)
        if c5 < best_c5 - 1e-15:
            best_h = h_new.copy()
            best_c5 = c5
    
    # Phase 1: Initial SLSQP polish
    if tl() > 40:
        h_new = try_slsqp(best_h, dx, target, bounds, maxiter=400, ftol=1e-14)
        update(h_new)
    
    # Phase 2: Smooth SLSQP annealing
    if tl() > 80:
        for temp in [3.0, 10.0, 30.0, 100.0, 300.0, 1000.0]:
            if tl() < 30:
                break
            h_s = try_smooth_slsqp(best_h, dx, target, bounds, temp, maxiter=150)
            if h_s is not None:
                c5_s = compute_c5(h_s, dx)
                if c5_s < best_c5 - 1e-15:
                    best_h, best_c5 = h_s.copy(), c5_s
                h_ref = try_slsqp(h_s, dx, target, bounds, maxiter=100)
                update(h_ref)
    
    # Phase 3: Subgradient descent (main global method)
    if tl() > 250:
        alloc = min(220, tl() * 0.35)
        h_sg, c5_sg = subgradient_phase(best_h, dx, target, alloc, rng)
        if c5_sg < best_c5 - 1e-15:
            best_h, best_c5 = h_sg, c5_sg
        h_new = try_slsqp(best_h, dx, target, bounds, maxiter=300, ftol=1e-15)
        update(h_new)
    
    # Phase 4: Pair/multi-coordinate perturbative search
    if tl() > 150:
        alloc = min(120, tl() * 0.3)
        h_mc, c5_mc = pair_search(best_h, dx, target, alloc, rng)
        if c5_mc < best_c5 - 1e-15:
            best_h, best_c5 = h_mc, c5_mc
        h_new = try_slsqp(best_h, dx, target, bounds, maxiter=200, ftol=1e-15)
        update(h_new)
    
    # Phase 5: Targeted top-shift balancing
    if tl() > 80:
        alloc = min(60, tl() * 0.4)
        h_bal, c5_bal = balance_top_shifts(best_h, dx, target, rng,
                                           n_iters=int(alloc * 30))
        if c5_bal < best_c5 - 1e-15:
            best_h, best_c5 = h_bal, c5_bal
        h_new = try_slsqp(best_h, dx, target, bounds, maxiter=200, ftol=1e-15)
        update(h_new)
    
    # Phase 6: Random restarts near best (diverse initializations)
    n_restarts = 25
    for i in range(n_restarts):
        if tl() < 12:
            break
        s_type = i % 6
        if s_type == 0:
            h0 = best_h + rng.randn(n) * 0.03
        elif s_type == 1:
            h0 = best_h + rng.randn(n) * 0.08
        elif s_type == 2:
            # Swap pairs randomly
            h0 = best_h.copy()
            for _ in range(n // 5):
                a, b = rng.choice(n, 2, replace=False)
                h0[a], h0[b] = h0[b], h0[a]
        elif s_type == 3:
            # Binary relaxation: round then re-project
            h0 = (rng.random(n) < best_h).astype(float)
        elif s_type == 4:
            # Sinusoidal modulation of best
            t = np.linspace(0, 4 * np.pi, n)
            h0 = best_h + 0.1 * np.sin(t + rng.uniform(0, 6.28))
        else:
            # Random blocks
            h0 = np.zeros(n)
            nblocks = rng.randint(4, 10)
            for _ in range(nblocks):
                blk = rng.randint(3, 12)
                st = rng.randint(0, max(1, n - blk))
                h0[st:st + blk] = 1.0
        h0 = projection(h0, target)
        
        # Quick optimization from this start
        h_new = try_slsqp(h0, dx, target, bounds, maxiter=150, ftol=1e-13)
        update(h_new)
        
        # If close to best, do smooth polish
        if h_new is not None:
            c5_check = compute_c5(h_new, dx)
            if c5_check < best_c5 + 0.005:
                h_sm = try_smooth_slsqp(h_new, dx, target, bounds, 20.0, maxiter=100)
                update(h_sm)
    
    # Phase 7: Second subgradient pass (finer)
    if tl() > 150:
        alloc = min(130, tl() * 0.5)
        h_sg2, c5_sg2 = subgradient_phase(best_h, dx, target, alloc, rng)
        if c5_sg2 < best_c5 - 1e-15:
            best_h, best_c5 = h_sg2, c5_sg2
        h_new = try_slsqp(best_h, dx, target, bounds, maxiter=400, ftol=1e-15)
        update(h_new)
    
    # Phase 8: Final pair search
    if tl() > 80:
        alloc = min(60, tl() * 0.5)
        h_ps, c5_ps = pair_search(best_h, dx, target, alloc, rng)
        if c5_ps < best_c5 - 1e-15:
            best_h, best_c5 = h_ps, c5_ps
        h_new = try_slsqp(best_h, dx, target, bounds, maxiter=500, ftol=1e-15)
        update(h_new)
    
    # Phase 9: Final SLSQP polish
    if tl() > 8:
        h_new = try_slsqp(best_h, dx, target, bounds, maxiter=500, ftol=1e-16)
        update(h_new)
    
    best_h = projection(best_h, target)
    final_c5 = compute_c5(best_h, dx)
    return (best_h, final_c5, n)
