import time

_g_n = 70

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:

        if not np.any(mask):
            h[:] = 0.5
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

    a, b = 0.0, 1.0

            c = d
            fc = fd
