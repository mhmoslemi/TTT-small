def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.array(h, dtype=float).copy()
    for _ in range(80):
        h = np.clip(h, 0.0, 1.0)
        d = n / 2.0 - np.sum(h)
        if abs(d) < 1e-14:
            break
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        if np.sum(mask) > 0:
            h[mask] += d / np.sum(mask)
        else:
            h += d / n
    return np.clip(h, 0.0, 1.0)

def interp_to(h, n_new):
    x_old = np.linspace(0, 2, len(h))
    x_new = np.linspace(0, 2, n_new)
    return repair(np.interp(x_new, x_old, h), n_new)

def pgd_grad(h, n, eps):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    s = np.clip((corr - m) / eps, -50, 50)
    w = np.exp(s) / np.sum(np.exp(s))
    grad = np.zeros(n)
    idx = np.arange(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-14:
            continue
        o = k - (n - 1)
        if o == 0:
            grad += wk * (1.0 - 2.0 * h)
        else:
            j1 = idx + o
            v1 = (j1 >= 0) & (j1 < n)
            j2 = idx - o
            v2 = (j2 >= 0) & (j2 < n)
            t1 = np.zeros(n)
            t1[v1] = 1.0 - h[j1[v1]]
            t2 = np.zeros(n)
            t2[v2] = -h[j2[v2]]
            grad += wk * (t1 + t2)
    return grad

    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    s = np.clip((corr - m) / eps, -50, 50)

def slqp_polish(h, n, eps, maxiter):

        return _smax(x, n, eps)
