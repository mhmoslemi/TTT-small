import numpy as np
import time
from scipy.optimize import minimize

def compute_score(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def compute_all_shifts(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return c

def grad_for_lag(h, m, n):
    """Gradient of dx*sum_j h[j]*(1-h[j-m]) w.r.t. h, where m is the shift (m=k-n+1)."""
    dx = 2.0 / n
    g = np.zeros(n)
    for i in range(n):
        im = i - m
        ip = i + m
        if 0 <= im < n:
            g[i] += (1.0 - h[im])
        if 0 <= ip < n:
            g[i] -= h[ip]
    return g * dx

def grad_subgrad(h):
    """Subgradient of max_k c[k]*dx using active set."""
    n = len(h)
    c = compute_all_shifts(h)
    mx = np.max(c)
    tol = 1e-8
    active = np.where(c >= mx - tol)[0]
    g = np.zeros(n)
    for k in active:
        m = k - (n - 1)
        g += grad_for_lag(h, m, n)
    g /= len(active)
    return g, mx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        s = np.sum(h)

        if abs(diff) < 1e-14:

    best_h = h.copy()

                stuck = 0

def slsqp_optimize(h0, n, maxiter=500):

def soft_max_obj(h, tau=50.0):

    variants.append(project_feasible(hs + rng.randn(n) * 0.03, n))
    return variants

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    best_h = h_init.copy()
    best_score = compute_score(h_init)
    
    # Phase 1: Optimize at initial n with subgradient
    h50, s50 = subgrad_optimize(best_h, n_init, max_iter=80000, time_limit=120)
    if s50 < best_score:
        best_score = s50
        best_h = h50
    t1 = time.time() - t0
    if t1 > budget_s * 0.5:
        best_h, best_score = slsqp_optimize(best_h, n_init, maxiter=300)
        return best_h, best_score, n_init
    
    # Phase 2: SLSQP polish
    h_pol, s_pol = slsqp_optimize(best_h, n_init, maxiter=800)
    if s_pol < best_score:
        best_score = s_pol
        best_h = h_pol
    
    # Phase 3: Nelder-Mead refinement
    h_nm, s_nm = nm_optimize(best_h, n_init, maxiter=3000)
    if s_nm < best_score:
        best_score = s_nm
        best_h = h_nm
    
    elapsed = time.time() - t0
    if elapsed > budget_s * 0.55:
        return best_h, best_score, n_init
    
    # Phase 4: Try other resolutions
    for n_try in [n_init + 10, n_init + 34, n_init + 50, n_init + 100]:
        if time.time() - t0 > budget_s * 0.75:
            break
        if n_try < 12:
            continue
        h_interp = parse_init(best_h, n_try, rng)
        h_sg, s_sg = subgrad_optimize(h_interp, n_try, max_iter=40000, time_limit=80)
        if s_sg > best_score:
            s_sg = compute_score(h_interp)
            if s_sg < best_score:
                h_sg = h_interp
                best_score = s_sg
                best_h = h_sg
        else:
            best_score = s_sg
            best_h = h_sg
        h_po, s_po = slsqp_optimize(h_sg, n_try, maxiter=500)
        if s_po < best_score:
            best_score = s_po
            best_h = h_po
        h_nm2, s_nm2 = nm_optimize(h_po, n_try, maxiter=1500)
        if s_nm2 < best_score:
            best_score = s_nm2
            best_h = h_nm2
    
    elapsed = time.time() - t0
    if elapsed > budget_s * 0.82:
        return best_h, best_score, len(best_h)
    
    # Phase 5: Diverse restarts at best n
    n_best = len(best_h)
    n_var = 6
    variants = make_structural_variants(best_h, n_best, rng, count=n_var)
    for vi, hv in enumerate(variants):
        if time.time() - t0 > budget_s * 0.88:
            break
        h_sg, s_sg = subgrad_optimize(hv, n_best, max_iter=30000, time_limit=40)
        if s_sg < best_score:
            best_score = s_sg
            best_h = h_sg
        h_po, s_po = slsqp_optimize(h_sg, n_best, maxiter=400)
        if s_po < best_score:
            best_score = s_po
            best_h = h_po
    
    # Phase 6: Final intensive polish
    if time.time() - t0 < budget_s * 0.95:
        h_fin, s_fin = subgrad_optimize(best_h, n_best, max_iter=60000, time_limit=30)
        if s_fin < best_score:
            best_score = s_fin
            best_h = h_fin
        h_fin2, s_fin2 = slsqp_optimize(best_h, n_best, maxiter=1000)
        if s_fin2 < best_score:
            best_score = s_fin2
            best_h = h_fin2
        h_fin3, s_fin3 = nm_optimize(best_h, n_best, maxiter=3000)
        if s_fin3 < best_score:
            best_score = s_fin3
            best_h = h_fin3
    
    # Final projection and validation
    best_h = project_feasible(best_h, len(best_h))
    best_score = compute_score(best_h)
    return best_h, best_score, len(best_h)
