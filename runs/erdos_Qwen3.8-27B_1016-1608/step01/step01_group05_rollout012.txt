def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx), corr * dx

def project_set(h, target):
    """Project h onto {0<=x<=1, sum(x)=target} using bisection."""
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def gradient_of_max_corr(h, n):
    """Compute gradient of max_k corr[k] * dx with respect to h."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    c_scaled = corr * dx
    k_star = np.argmax(c_scaled)
    
    # k_star is in [0, 2n-2]
    # Convert to shift
    if k_star >= n - 1:
        k = k_star - (n - 1)  # Right shift
        # R_k = sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])
        grad = np.zeros(n)
        # First term: (1-h[j+k]) for j in [0, n-1-k]
        if k == 0:
            grad[:n] = 1.0 - h  # (1-h[j+0]) for all j
        else:
            grad[:n-k] += (1.0 - h[k:])
        # Second term: -h[j-k] for j in [k, n-1]
        if k > 0:
            grad[k:] -= h[:n-k]
    else:
        k = (n - 1) - k_star  # Left shift
        # L_k = sum_{i=k}^{n-1} h[i]*(1-h[i-k])
        grad = np.zeros(n)
        # First term: (1-h[j-k]) for j in [k, n-1]
        if k == 0:

        else:
            grad[k:] += (1.0 - h[:n-k])
        # Second term: -h[j+k] for j in [0, n-1-k]
        if k > 0:
            grad[:n-k] -= h[k:]

```python
import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def c5_value(h, n):
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def project_set(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def max_corr_gradient(h, n):
    """Gradient of max_k corr[k]*dx. Returns (gradient, c5_value, argmax_k)."""
    dx = 2.0 / n
    corr = compute_corr(h)
    c = corr * dx
    p_star = int(np.argmax(c))
    
    grad = np.zeros(n)
    if p_star >= n - 1:
        k = p_star - (n - 1)
        # Right shift k
        if k == 0:
            grad[:] = 1.0 - 2.0 * h
        else:
            grad[:n-k] += (1.0 - h[k:])
            grad[k:] -= h[:n-k]
    else:
        k = (n - 1) - p_star
        # Left shift k
        if k == 0:
            grad[:] = 1.0 - 2.0 * h
        else:
            grad[k:] += (1.0 - h[:n-k])
            grad[:n-k] -= h[k:]
    
    return grad * dx, c[p_star], p_star

def smooth_max_gradient(h, n, beta=50.0):
    """Gradient of smooth max (log-sum-exp approximation)."""
    dx = 2.0 / n
    corr = compute_corr(h)
    c = corr * dx
    c_max = c.max()
    exp_c = np.exp(beta * (c - c_max))
    w = exp_c / exp_c.sum()
    
    # Weighted sum of gradients
    grad = np.zeros(n)
    nfull = 2 * n - 1
    for p in range(nfull):
        if w[p] < 1e-12:
            continue
        wp = w[p] * dx
        if p >= n - 1:
            k = p - (n - 1)
            if k == 0:
                grad[:] += wp * (1.0 - 2.0 * h)
            else:
                grad[:n-k] += wp * (1.0 - h[k:])
                grad[k:] -= wp * h[:n-k]
        else:
            k = (n - 1) - p
            if k == 0:
                grad[:] += wp * (1.0 - 2.0 * h)
            else:
                grad[k:] += wp * (1.0 - h[:n-k])
                grad[:n-k] -= wp * h[k:]
    
    return grad, c_max
```

def projected_gd(h, n, lr_init=0.01, n_iters=500, alpha_anneal=None):
    """Projected gradient descent with adaptive learning rate."""
    target = n / 2.0
    h = project_set(h.copy(), target)
    best_h = h.copy()
    best_c = c5_value(h, n)
    
    for it in range(n_iters):
        lr = lr_init / (1.0 + it * alpha_anneal) if alpha_anneal else lr_init
        grad, c_val, _ = max_corr_gradient(h, n)
        
        h_new = h - lr * grad
        h_new = project_set(h_new, target)
        
        c_new = c5_value(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        
        # Backtracking if it got worse
        if c_new > c_val + 1e-10:
            h = h_new  # still accept to avoid cycling
        else:
            h = h_new
    
    return best_h, best_c

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_set(h, target):

    for _ in range(60):
        mid = 0.5 * (lo + hi)
        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo+hi), 0.0, 1.0)

    """Returns (gradient, smooth_max_value, hard_max_value)."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    c = corr * dx
    c_max = c.max()

    exps = np.exp(beta * shifted)
    w = exps / exps.sum()

    grad = np.zeros(n)

                grad += wp * (1.0 - 2.0 * h)
            else:
                grad[k:] += wp * (1.0 - h[:n-k])
                grad[:n-k] -= wp * h[k:]

        h = h_new
    
    return best_h, best_c

def binary_local_search(h, n, n_sweeps=50, rng=None):
    target = int(round(n / 2.0))
    # Convert to binary: top n/2 values become 1
    h_bin = np.zeros(n)
    h_bin[np.argsort(h)[::-1][:target]] = 1.0
    
    best_c = compute_c5(h_bin, n)
    for sweep in range(n_sweeps):
        improved = False
        indices = np.random.permutation(n) if rng is not None else np.arange(n)
        for j in indices:
            h_bin[j] = 1 - h_bin[j]
            c = compute_c5(h_bin, n)
            if c < best_c:
                best_c = c
                improved = True
            else:
                h_bin[j] = 1 - h_bin[j]
        if not improved:
            break
    return h_bin, best_c

def upsample_h(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, h)

def make_frich_initial(n, rng):
    """Make a diverse random initialization."""
    x = np.linspace(0, 2, n)
    h = 0.5 * np.ones(n)
    nfreq = rng.randint(2, 8)
    for _ in range(nfreq):
        f = rng.randint(1, 15)
        phase = rng.uniform(0, 2*np.pi)
        amp = rng.uniform(0.1, 0.4)
        h += amp * np.sin(f * np.pi * x + phase)
    return project_set(h, n/2.0)
