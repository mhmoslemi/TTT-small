            grad[:] -= h
        else:
            grad[k:] += (1.0 - h[:n-k])
            grad[:n-k] -= h[k:]
    else:
        m = -k
        grad[:n-m] += (1.0 - h[m:])
        grad[m:] -= h[:n-m]
    return grad * dx

def subgradient_step(h, n, alpha, top_k=3):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    top_idx = np.argsort(c)[-top_k:]
    cm = c[top_idx[-1]]
    weights = np.exp(50.0 * (c[top_idx] - cm))
    weights /= np.sum(weights)
    grad = np.zeros(n)
    for w, idx in zip(weights, top_idx):
        grad += w * grad_corr(h, idx, n)
    h_new = h - alpha * grad
    return project_feasible(h_new, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    best_h = h_init.copy()
    best_score = compute_score(h_init)

    def elapsed():
        return time.time() - t0

    def update(h, s):
        nonlocal best_h, best_score
        if s < best_score:
            best_score = s
            best_h = h.copy()
        return best_score

    n_targets = [80, 100, 128, 160]

    for n_pts in n_targets:
        if elapsed() > budget_s * 0.82:
            break

        x_old = np.linspace(0.0, 2.0, len(best_h), endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_pts, endpoint=False)
        h0 = np.interp(x_new, x_old, best_h)
        h0 = project_feasible(h0, n_pts)

        bounds = [(0.0, 1.0)] * n_pts
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_pts / 2.0}]

        # Phase 1: SLSQP with soft max
        tau_seq = [400, 200, 100]
        for tau in tau_seq:
            if elapsed() > budget_s * 0.82:
                break
            def obj(h, t=tau):
                return soft_max_score(h, t)
            try:
                res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 400, 'ftol': 1e-12})
                h0 = project_feasible(res['x'], n_pts)
                update(h0, compute_score(h0))
            except Exception:
                pass

        # Phase 2: Subgradient iterations
        if elapsed() < budget_s * 0.78:
            h_sg = h0.copy()
            n_iters = 300
            for it in range(n_iters):
                if elapsed() > budget_s * 0.80:
                    break
                alpha = 0.04 / (1.0 + 0.01 * it)
                h_sg = subgradient_step(h_sg, n_pts, alpha, top_k=4)
                s_sg = compute_score(h_sg)
                update(h_sg, s_sg)
                if s_sg < compute_score(h0) - 1e-5:
                    h0 = h_sg.copy()

        # Phase 3: SLSQP on exact max
        if elapsed() < budget_s * 0.78:
            try:
                res = minimize(compute_score, h0, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 500, 'ftol': 1e-13})
                h0 = project_feasible(res['x'], n_pts)
                update(h0, compute_score(h0))
            except Exception:
                pass

        # Phase 4: Random restarts + subgradient
        for r in range(3):
            if elapsed() > budget_s * 0.80:
                break
            h_r = project_feasible(h0 + rng.randn(n_pts) * 0.08, n_pts)
            for it in range(200):
                if elapsed() > budget_s * 0.81:
                    break
                alpha = 0.03 / (1.0 + 0.01 * it)
                h_r = subgradient_step(h_r, n_pts, alpha, top_k=3)
            s_r = compute_score(h_r)
            if s_r < compute_score(h0):
                try:
                    res = minimize(soft_max_score, h_r, method='SLSQP', bounds=bounds,
                                   constraints=cons, options={'maxiter': 200, 'ftol': 1e-12})
                    h_r = project_feasible(res['x'], n_pts)
                except Exception:
                    pass
                update(h_r, compute_score(h_r))
                h0 = h_r.copy()

    # Final polish on best
    if elapsed() < budget_s * 0.93:
        n_best = len(best_h)
        bounds = [(0.0, 1.0)] * n_best
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_best / 2.0}]
        for attempt in range(3):
            if elapsed() > budget_s * 0.95:
                break
            try:
                res = minimize(compute_score, best_h, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 500, 'ftol': 1e-14})
                h_f = project_feasible(res['x'], n_best)
                update(h_f, compute_score(h_f))
            except Exception:
                pass
            try:
                res2 = minimize(soft_max_score, best_h, method='SLSQP', bounds=bounds,
                                constraints=cons, options={'maxiter': 300, 'ftol': 1e-14})
                h_f2 = project_feasible(res2['x'], n_best)
                update(h_f2, compute_score(h_f2))
            except Exception:
                pass

    return best_h, best_score, len(best_h)
