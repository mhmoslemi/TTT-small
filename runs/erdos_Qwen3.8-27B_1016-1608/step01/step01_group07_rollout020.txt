        return None
    return f

def _subgrad_projected(h, n, steps, t0, budget):
    dx = 2.0 / n
    best = h.copy()
    best_c = _c5(best)
    lr = 0.5
    for it in range(steps):
        if time.time() - t0 > budget - 5:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        cmax = np.max(c)
        active = np.where(c >= cmax - 1e-7)[0]
        if len(active) == 0:
            break
        grad = np.zeros(n)
        for k in active:
            lag = k - (n - 1)
            if lag >= 0:
                if n - lag > 0:
                    grad[:n - lag] += (1.0 - h[lag:n])
                if lag > 0:
                    grad[lag:] -= h[:n - lag]
            else:
                j2 = -lag
                if n - j2 > 0:
                    grad[j2:] += (1.0 - h[:n - j2])
                if j2 > 0:
                    grad[:n - j2] -= h[j2:]
        grad *= dx / len(active)
        # Projected gradient step
        h_new = h - lr * grad
        # Project onto box
        h_new = np.clip(h_new, 0.0, 1.0)
        # Project onto sum constraint
        h_new = _fix(h_new, n)
        c_new = _c5(h_new)
        if c_new < best_c:
            best = h_new.copy()
            best_c = c_new
        h = h_new
        # Adaptive step size
        if c_new > cmax + 0.01:
            lr *= 0.7
        elif c_new < cmax - 0.001:
            lr = min(lr * 1.05, 2.0)
    return best, best_c

def _lgbfsopt(h0, n, alpha, maxiter, t0, budget):
    if budget - (time.time() - t0) < 8:
        return None
    bnds = [(0.0, 1.0)] * n
    f = _make_smooth(alpha, n)
    try:
        res = minimize(f, h0.copy(), method='L-BFGS-B',
                       bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-11})
        if res.x is None:
            return None
        return _fix(np.clip(res.x, 0.0, 1.0), n)
    except Exception:
        return None

def _structure_perturb(h, rng, n, amp):
    x = np.arange(n)
    # Low-frequency structured perturbation
    nfreq = 4
    noise = np.zeros(n)
    for _ in range(nfreq):
        f = rng.randint(1, 8)
        phase = rng.uniform(0, 2 * np.pi)
        noise += amp * np.sin(2 * np.pi * f * x / n + phase)
    h_p = h + noise
    return _fix(np.clip(h_p, 0.0, 1.0), n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        h = np.full(100, 0.5)
    n = len(h)
    h = _fix(h, n)
    bh = h.copy()
    bc = _c5(bh)

    def remaining():
        return budget_s - (time.time() - t0)

    # Phase 1: L-BFGS-B at parent resolution, increasing alpha
    for alpha in [30, 80, 200, 500, 1500, 4000]:
        if remaining() < 12:
            break
        n_cur = len(bh)
        ht = _lgbfsopt(bh, n_cur, alpha, 5000, t0, budget_s)
        if ht is not None:
            ct = _c5(ht)
            if ct < bc:
                bh = ht
                bc = ct

    # Phase 2: Subgradient refinement
    if remaining() > 30:
        n_cur = len(bh)
        sg_steps = int(min(3000, remaining() * 8))
        bh, bc = _subgrad_projected(bh, n_cur, sg_steps, t0, budget_s)

    # Phase 3: Try smaller resolutions
    for target_n in [256, 200, 160, 128, 100]:
        if remaining() < 25:
            break
        if target_n >= len(bh):
            continue
        ht = _interp_to(bh, target_n)
        ct = _c5(ht)
        for alpha in [100, 400, 1200]:
            if remaining() < 12:
                break
            hres = _lgbfsopt(ht, target_n, alpha, 6000, t0, budget_s)
            if hres is not None:
                c_res = _c5(hres)
                if c_res < ct:
                    ht = hres
                    ct = c_res
                if c_res < bc:
                    bh = hres.copy()
                    bc = c_res
        # Subgradient at this resolution
        if remaining() > 20:
            hsg, csg = _subgrad_projected(ht, target_n, 2000, t0, budget_s)
            if csg < bc:
                bh = hsg.copy()
                bc = csg

    # Phase 4: Try larger resolution and polish
    if remaining() > 30:
        for target_n in [500, 600, 800]:
            if remaining() < 20:
                break
            ht = _interp_to(bh, target_n)
            ct = _c5(ht)
            for alpha in [300, 1000, 3000]:
                if remaining() < 12:
                    break
                hres = _lgbfsopt(ht, target_n, alpha, 4000, t0, budget_s)
                if hres is not None:
                    c_res = _c5(hres)
                    if c_res < ct:
                        ht = hres
                        ct = c_res
                    if c_res < bc:
                        bh = hres.copy()
                        bc = c_res
            if remaining() > 15:
                hsg, csg = _subgrad_projected(ht, target_n, 1500, t0, budget_s)
                if csg < bc:
                    bh = hsg.copy()
                    bc = csg
            if remaining() < 15:
                break

    # Phase 5: Multi-restart with structured perturbation at best resolution
    n_best = len(bh)
    for restart in range(12):
        if remaining() < 15:
            break
        h_pert = _structure_perturb(bh, rng, n_best, 0.04 + 0.02 * restart)
        # Small SLSQP-like via L-BFGS with medium alpha
        ht = _lgbfsopt(h_pert, n_best, 500, 3000, t0, budget_s)
        if ht is not None:
            ct = _c5(ht)
            if ct < bc:
                bh = ht
                bc = ct
        # Short subgradient from perturbed start
        if remaining() > 10:
            hsg, csg = _subgrad_projected(h_pert, n_best, 800, t0, budget_s)
            if csg < bc:
                bh = hsg
                bc = csg

    # Phase 6: Final tight polish
    if remaining() > 10:
        n_final = len(bh)
        for alpha in [5000, 10000]:
            if remaining() < 8:
                break
            ht = _lgbfsopt(bh, n_final, alpha, 8000, t0, budget_s)
            if ht is not None:
                ct = _c5(ht)
                if ct < bc:
                    bh = ht
                    bc = ct
        if remaining() > 8:
            bh, bc = _subgrad_projected(bh, len(bh), 3000, t0, budget_s)

    bh = _fix(bh, len(bh))
    bc = _c5(bh)
    return bh, bc, len(bh)
