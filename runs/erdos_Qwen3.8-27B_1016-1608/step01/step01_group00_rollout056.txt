    for k in range(2 * n - 1):
        wt = w[k] * dx
        if wt < 1e-18:
            continue
        s1 = k - (n - 1)
        s2 = (n - 1) - k
        jlo = max(0, -s1)
        jhi = min(n - 1, n - 1 - s1)
        if jlo <= jhi:
            result[jlo:jhi+1] += wt * (1.0 - h[s1+jlo:s1+jhi+1])
        jlo2 = max(0, -s2)
        jhi2 = min(n - 1, n - 1 - s2)
        if jlo2 <= jhi2:
            result[jlo2:jhi2+1] -= wt * h[s2+jlo2:s2+jhi2+1]
    return result / beta

def slsqp_smooth(h0, n, beta, maxiter=500):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con}]
    def obj(x):
        return smooth_c5(x, beta)
    def grad(x):
        return smooth_grad(x, beta)
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = repair(np.clip(res.x, 0.0, 1.0), n)
        if compute_c5(h_new) < compute_c5(h0) + 1e-14:
            return h_new
    except Exception:
        pass
    return h0

def slsqp_true(h0, n, maxiter=500):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con}]
    try:
        res = minimize(compute_c5, h0, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = repair(np.clip(res.x, 0.0, 1.0), n)
        if compute_c5(h_new) < compute_c5(h0) + 1e-14:
            return h_new
    except Exception:
        pass
    return h0

def flip_perturb(h, n, n_flips):
    h = h.copy()
    idx = np.random.choice(n, n_flips, replace=False)
    h[idx] = 1.0 - h[idx]
    return repair(h, n)

def shift_perturb(h, n):
    s = np.random.randint(1, n)
    h = np.roll(h, s)
    return repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_best = np.full(70, 0.5)
    n = len(h_best)
    c5_best = compute_c5(h_best)

    def elapsed():
        return time.time() - t0

    # Phase 1: Smooth SLSQP with increasing beta
    betas = [10, 30, 80, 200, 500, 1000]
    for beta in betas:
        if elapsed() > budget_s * 0.30:
            break
        h = slsqp_smooth(h_best, n, beta, maxiter=300)
        c = compute_c5(h)
        if c < c5_best:
            h_best, c5_best = h, c

    # Phase 2: True SLSQP polish
    if elapsed() < budget_s * 0.35:
        h = slsqp_true(h_best, n, maxiter=2000)
        c = compute_c5(h)
        if c < c5_best:
            h_best, c5_best = h, c

    # Phase 3: Main search loop with diverse perturbations + smooth SLSQP
    scales = [0.003, 0.008, 0.015, 0.03, 0.06, 0.12]
    betas_main = [50, 100, 200, 500]
    it = 0
    while elapsed() < budget_s * 0.68:
        scale = scales[it % len(scales)]
        beta = betas_main[it % len(betas_main)]
        r = np.random.random()
        if r < 0.4:
            h_p = repair(h_best + np.random.randn(n) * scale, n)
        elif r < 0.6:
            h_p = flip_perturb(h_best, n, int(np.random.randint(2, n // 4 + 1)))
            h_p = repair(h_p + np.random.randn(n) * scale * 0.5, n)
        elif r < 0.75:
            h_p = shift_perturb(h_best, n)
            h_p = repair(h_p + np.random.randn(n) * scale * 0.3, n)
        elif r < 0.9:
            # Block perturb
            k = np.random.randint(3, max(4, n // 3))
            start = np.random.randint(0, n - k + 1)
            h_p = h_best.copy()
            h_p[start:start+k] += np.random.randn(k) * scale * 2
            h_p = repair(h_p, n)
        else:
            # Complementary-like: nudge toward 1-h on a subset
            h_p = h_best.copy()
            idx = np.random.choice(n, n // 3, replace=False)
            h_p[idx] = 0.5 + (1.0 - h_best[idx] - 0.5) * 0.3
            h_p = repair(h_p, n)

        h = slsqp_smooth(h_p, n, beta, maxiter=200)
        c = compute_c5(h)
        if c < c5_best:
            h_best, c5_best = h, c
            # Immediate true-objective polish on improvement
            h = slsqp_true(h_best, n, maxiter=500)
            c = compute_c5(h)
            if c < c5_best:
                h_best, c5_best = h, c
        it += 1

    # Phase 4: Try different resolutions
    if elapsed() < budget_s * 0.85:
        for n2 in [60, 64, 68, 72, 76, 80, 90, 100, 120]:
            if elapsed() > budget_s * 0.84:
                break
            xp = np.linspace(0.0, 1.0, n)
            xn = np.linspace(0.0, 1.0, n2)
            h_interp = np.interp(xn, xp, h_best)
            h_interp = repair(h_interp, n2)
            for beta in [100, 500]:
                if elapsed() > budget_s * 0.84:
                    break
                h2 = slsqp_smooth(h_interp, n2, beta, maxiter=400)
                c2 = compute_c5(h2)
                if c2 < c5_best:
                    h_best, c5_best, n = h2, c2, n2
                    h2 = slsqp_true(h_best, n2, maxiter=500)
                    c2 = compute_c5(h2)
                    if c2 < c5_best:
                        h_best, c5_best, n = h2, c2, n2
            if c5_best <= 0.3806:
                break

    # Phase 5: Final polish rounds
    if elapsed() < budget_s * 0.95:
        h = slsqp_true(h_best, n, maxiter=3000)
        c = compute_c5(h)
        if c < c5_best:
            h_best, c5_best = h, c

        for _ in range(8):
            if elapsed() > budget_s * 0.95:
                break
            h_p = repair(h_best + np.random.randn(n) * 0.002, n)
            h = slsqp_smooth(h_p, n, 1000, maxiter=300)
            c = compute_c5(h)
            if c < c5_best:
                h_best = h
                c5_best = c
            h = slsqp_true(h_best, n, maxiter=800)
            c = compute_c5(h)
            if c < c5_best:
                h_best, c5_best = h, c

    # Final validation
    h_best = repair(h_best, n)
    c5_best = compute_c5(h_best)
    return (h_best, c5_best, n)
