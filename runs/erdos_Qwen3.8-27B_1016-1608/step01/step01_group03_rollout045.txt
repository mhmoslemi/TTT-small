        if abs(d) < 1e-13:

    s = np.sum(np.exp(beta * (corr - m)))

import numpy as np
from scipy.optimize import minimize
import time

_G_TARGET = 0.0
_G_N = 70

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def project_feasible(x, target):
    h = np.clip(np.asarray(x, dtype=float), 0.0, 1.0).copy()
    for _ in range(300):
        d = target - np.sum(h)
        if abs(d) < 1e-14:
            break
        mask = (h < 1.0 - 1e-15) if d > 0 else (h > 1e-15)
        c = mask.sum()
        if c == 0:
            break
        h[mask] += d / c
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_obj(x, beta, target, dx):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    s = np.sum(np.exp(beta * (corr - m)))
    return (m + np.log(s) / beta) * dx + 50.0 * (np.sum(h) - target) ** 2

    n = len(h)
    base = compute_c5(h, dx)
    g = np.zeros(n)
    for i in range(n):
        hp = h.copy()
        hp[i] = np.clip(hp[i] + eps, 0.0, 1.0)
        hp = project_feasible(hp, target)
        g[i] = (compute_c5(hp, dx) - base) / eps
