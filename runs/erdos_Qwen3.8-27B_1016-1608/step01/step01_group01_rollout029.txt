import numpy as np
from scipy.optimize import minimize, LinearConstraint
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return c.max() * dx

def corr_vec(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-15) & (h < 1.0 - 1e-15)
        if free.sum() > 0:
            h[free] += diff / free.sum()
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_obj(h, temp):
    c = corr_vec(h)
    m = c.max()
    return m + temp * np.log(np.sum(np.exp((c - m) / temp)))

def lse_grad(h, temp):
    c = corr_vec(h)
    m = c.max()
    w = np.exp((c - m) / temp)
    w /= w.sum()
    n = len(h)
    grad = np.zeros(n)
    shifts = np.arange(len(c)) - (n - 1)
    for si, wi in zip(shifts, w):
        if abs(wi) < 1e-14:
            continue
        si = int(si)
        for j in range(n):
            jp = j + si
            jm = j - si
            if 0 <= jp < n:
                grad[j] += wi * (1.0 - h[jp])
            if 0 <= jm < n:
                grad[j] -= wi * h[jm]
    return grad

def sum_eq_fun(h):
    return h.sum() - len(h) / 2.0

def sum_eq_jac(h):
    return np.ones(len(h))

def run_slsqp(x0, temp, maxiter, bounds, constraints, n):
    res = minimize(lse_obj, x0, args=(temp,), method='SLSQP',
                  bounds=bounds, constraints=constraints,
                  options={'maxiter': maxiter, 'ftol': 1e-13})
    return project_feasible(res.x, n)

def run_trust_constr(x0, temp, maxiter, bounds_list, n):
    lc = LinearConstraint(np.ones((1, n)), n/2, n/2)
    bl = [(0.0, 1.0) for _ in range(n)]
    res = minimize(lse_obj, x0, args=(temp,), method='trust-constr',
                  bounds=bl, constraints=[lc], jac=lse_grad,
                  options={'maxiter': maxiter, 'gtol': 1e-12, 'xtol': 1e-14})
    return project_feasible(res.x, n)

def coord_descent(h, n, max_rounds, time_budget):
    t0 = time.time()
    best_c5 = compute_c5(h)
    deltas_coarse = np.linspace(-0.1, 0.1, 21)
    deltas_fine = np.linspace(-0.02, 0.02, 21)
    for rnd in range(max_rounds):
        if time.time() - t0 > time_budget:
            break
        improved = False
        deltas = deltas_coarse if rnd < max_rounds // 3 else deltas_fine
        for i in range(n):
            for d in deltas:
                nv = h[i] + d
                if nv < -1e-10 or nv > 1.0 + 1e-10:
                    continue
                h_t = h.copy()
                h_t[i] = nv
                h_t = project_feasible(h_t, n)
                c_t = compute_c5(h_t)
                if c_t < best_c5 - 1e-12:
                    best_c5 = c_t
                    h = h_t.copy()
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5
