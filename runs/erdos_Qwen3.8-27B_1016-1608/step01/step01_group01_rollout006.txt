import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return c.max() * dx

def corr_vec(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            h += diff / n
        else:
            h[free] += diff / free.sum()
    return np.clip(h, 0.0, 1.0)

def soft_max_obj(h, temp):
    c = corr_vec(h)
    m = c.max()
    if temp < 1e-14:
        return m
    scaled = np.clip((c - m) / temp, -400, 400)
    return m + temp * np.logsumexp(scaled)

from scipy.special import logsumexp

def sum_fun(h):
    return np.sum(h) - len(h) / 2.0

def sum_jac(h):
    return np.ones(len(h))

def sa_step(sa_h, n, rng, move_type):
    h_new = sa_h.copy()
    if move_type == 0:
        i, j = rng.choice(n, 2, replace=False)
        d = rng.uniform(-0.25, 0.25)
        h_new[i] += d
        h_new[j] -= d
    elif move_type == 1:
        i = rng.randint(n)
        w = rng.randint(2, max(3, n // 4))
        d = rng.uniform(-0.1, 0.1)
        end = min(i + w, n)
        h_new[i:end] += d
        rest = np.concatenate([np.arange(0, i), np.arange(end, n)])
        if len(rest) > 0:
            h_new[rest] -= d * (end - i) / len(rest)
    else:
        i = rng.randint(n)
        d = rng.uniform(-0.4, 0.4)
        h_new[i] += d
        others = np.array([j for j in range(n) if j != i])
        h_new[others] -= d / len(others)
    return project_feasible(h_new, n)
