    return c.max() * dx

def corr_vec(h):
    return np.correlate(h, 1.0 - h, mode='full')

def sum_constraint(h):
    return np.sum(h) - len(h) / 2.0

def sum_constraint_jac(h):
    return np.ones(len(h))

def soft_max_obj(h, temp):
    c = corr_vec(h)
    m = c.max()
    c_shift = np.clip(c - m, -500, 0)
    return m + temp * np.log(np.sum(np.exp(c_shift / temp)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() == 0:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
        else:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_solve(h_start, n, temp, maxiter):
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(soft_max_obj, h_start, args=(temp,),
                  method='SLSQP', bounds=bnds, constraints=cons,
                  options={'maxiter': maxiter, 'ftol': 1e-12})
    return project_feasible(res.x, n)

def perturb_h(h, n, rng, mode, scale):
    h_new = h.copy()
    if mode == 0:
        i = rng.randint(n)
        h_new[i] = np.clip(h_new[i] + rng.uniform(-scale, scale), 0, 1)
    elif mode == 1:
        b = rng.randint(2, min(6, n))
        i = rng.randint(0, n - b)
        h_new[i:i+b] = np.clip(h_new[i:i+b] + rng.uniform(-0.5*scale, 0.5*scale), 0, 1)
    elif mode == 2:
        i, j = rng.choice(n, 2, replace=False)
        d = rng.uniform(-scale, scale)
        h_new[i] = np.clip(h_new[i] + d, 0, 1)
        h_new[j] = np.clip(h_new[j] - d, 0, 1)
    else:
        h_new = np.clip(h + rng.normal(0, scale, n), 0, 1)
    return project_feasible(h_new, n)

def coord_descent(h, n, best_c5, max_sweeps, t_limit):
    deltas = [0.05, -0.05, 0.02, -0.02, 0.01, -0.01,
              0.005, -0.005, 0.002, -0.002, 0.001, -0.001,
              0.0005, -0.0005]
    for _ in range(max_sweeps):
        if time.time() > t_limit:
            break
        improved = False
        for i in range(n):
            for d in deltas:
                val = h[i] + d
                if 0.0 <= val <= 1.0:
                    h_t = h.copy()
                    h_t[i] = val
                    h_t = project_feasible(h_t, n)
                    c5_t = compute_c5(h_t)
                    if c5_t < best_c5 - 1e-15:
                        best_c5 = c5_t
                        h = h_t
                        improved = True
                        break
        if not improved:
            break
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    h0 = project_feasible(h0, n)

    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    # Phase 1: Simulated Annealing (25% budget)
    t_end1 = t0 + budget_s * 0.25
    cur_h = h0.copy()
    cur_c5 = best_c5
    temp = 0.03
    while time.time() < t_end1 and temp > 0.00005:
        for _ in range(30):
            if time.time() >= t_end1:
                break
            mode = rng.randint(4)
            scale = 0.15 + 0.25 * min(1.0, temp / 0.03)
            h_new = perturb_h(cur_h, n, rng, mode, scale)
            c5_new = compute_c5(h_new)
            if c5_new < cur_c5:
                cur_h = h_new
                cur_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
            else:
                if rng.random() < np.exp(-(c5_new - cur_c5) / temp):
                    cur_h = h_new
                    cur_c5 = c5_new
        temp *= 0.993
        if rng.random() < 0.1:
            cur_h = best_h.copy()
            cur_c5 = best_c5

    # Phase 2: Iterated Local Search (30% budget)
    t_end2 = t0 + budget_s * 0.55
    for it in range(300):
        if time.time() > t_end2:
            break
        mode = rng.randint(4)
        scale = 0.04 + 0.08 * rng.random()
        h_pert = perturb_h(best_h, n, rng, mode, scale)
        h_opt = slsqp_solve(h_pert, n, 0.01, 800)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        # Quick coord polish after each ILS step
        best_h, best_c5 = coord_descent(best_h, n, best_c5, 3, t_end2)

    # Phase 3: SLSQP multi-temperature polishing (20% budget)
    t_end3 = t0 + budget_s * 0.75
    temps_p = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0001, 0.00005]
    for tp in temps_p:
        if time.time() > t_end3:
            break
        h_new = slsqp_solve(best_h.copy(), n, tp, 3000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        # Perturbed restart
        h_r = best_h + rng.normal(0, 0.03, n)
        h_r = project_feasible(h_r, n)
        h_new2 = slsqp_solve(h_r, n, tp, 1500)
        c5_new2 = compute_c5(h_new2)
        if c5_new2 < best_c5:
            best_c5 = c5_new2
            best_h = h_new2.copy()

    # Phase 4: Thorough coordinate descent (20% budget)
    t_end4 = t0 + budget_s * 0.95
    best_h, best_c5 = coord_descent(best_h, n, best_c5, 60, t_end4)

    # Phase 5: Final pair-swap search + SLSQP
    t_end5 = t0 + budget_s * 0.985
    for _ in range(30000):
        if time.time() > t_end5:
            break
        i, j = rng.choice(n, 2, replace=False)
        d = rng.uniform(-0.01, 0.01)
        h_t = best_h.copy()
        h_t[i] = np.clip(h_t[i] + d, 0, 1)
        h_t[j] = np.clip(h_t[j] - d, 0, 1)
        h_t = project_feasible(h_t, n)
        c5_t = compute_c5(h_t)
        if c5_t < best_c5 - 1e-15:
            best_c5 = c5_t
            best_h = h_t.copy()

    if time.time() < t0 + budget_s * 0.995:
        h_final = slsqp_solve(best_h.copy(), n, 1e-5, 2000)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()
        best_h, best_c5 = coord_descent(best_h, n, best_c5, 5, t0 + budget_s * 0.998)

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n)
