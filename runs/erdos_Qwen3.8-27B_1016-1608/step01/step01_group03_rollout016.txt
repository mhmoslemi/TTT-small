from scipy.optimize import minimize
import time

def project_constraint(x, target_sum):
    """Project onto {h: sum(h)=target, 0<=h<=1}"""
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        h = np.clip(x - mid, 0.0, 1.0)
        if np.sum(h) > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def get_corr_profile(h):
    return np.correlate(h, 1.0 - h, mode="full")

def get_c5(h, dx):
    return float(np.max(get_corr_profile(h))) * dx

def smoothed_obj(h, tau, dx):
    c = get_corr_profile(h) * dx
    m = np.max(c)
    e = np.exp(tau * (c - m))
    return m + np.log(e.sum()) / tau

def smoothed_grad(h, tau, dx):
    n = len(h)
    c = get_corr_profile(h) * dx
    m = np.max(c)
    log_e = tau * (c - m)
    log_e -= log_e.max()
    e = np.exp(log_e)
    p = e / e.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-14:
            continue
        s = k - (n - 1)
        pk_dx = p[k] * dx
        # contribution from j as "i": (1-h[j-s]) if 0<=j-s<n
        if s >= 0:
            js = np.arange(s, n)
            grad[js] += pk_dx * (1.0 - h[js - s])
        else:
            js = np.arange(0, n + s)  # j - s < n => j < n + s (s negative)
            # wait: 0 <= j-s < n => s <= j < n+s
            # s < 0: 0 <= j and j < n+s = n-|s|
            js = np.arange(0, n + s)
            grad[js] += pk_dx * (1.0 - h[js - s])
        # contribution from j as "i-s" => i = j+s: -h[j+s] if 0<=j+s<n
        if s >= 0:
            js2 = np.arange(0, n - s)
            grad[js2] += pk_dx * (-h[js2 + s])
        else:
            js2 = np.arange(-s, n)
            grad[js2] += pk_dx * (-h[js2 + s])
    return grad

def projected_lbfgs(h0, target_sum, dx, timeout):
    """L-BFGS projected gradient with annealed tau"""
    h = h0.copy()
    t0 = time.time()
    n = len(h)
    
    for phase in range(8):
        tau = 5.0 * (2 ** phase)  # 5, 10, 20, ..., 640
        lr = 0.5 / (tau * dx * n ** 0.5)
        for it in range(100):
            if time.time() - t0 > timeout:
                return h
            obj = smoothed_obj(h, tau, dx)
            g = smoothed_grad(h, tau, dx)
            # Adaptive step
            h_new = h - lr * g
            h = project_constraint(h_new, target_sum)
            # Check improvement
            if smoothed_obj(h, tau, dx) > obj + 1e-6:
                lr *= 0.5
                h_new = h - lr * g
                h = project_constraint(h_new, target_sum)
    return h

def binary_local_search(h, target_sum, n_swaps=5000, timeout=30):
    """Local search on binary arrangement"""
    n = len(h)
    t0 = time.time()
    # Round to binary
    h_bin = np.zeros(n)
    # Sort by value, top n/2 are 1s
    idx = np.argsort(h)[::-1]
    h_bin[idx[:int(target_sum)]] = 1.0
    
    def binary_c5(hb):
        c = get_corr_profile(hb) * (2.0 / n)
        return c.max()
    
    best_c = binary_c5(h_bin)
    ones = np.where(h_bin == 1)[0]
    zeros = np.where(h_bin == 0)[0]
    
    it = 0
    while it < n_swaps and time.time() - t0 < timeout:
        it += 1
        i1 = np.random.randint(len(ones))
        i0 = np.random.randint(len(zeros))
        p1, p0 = ones[i1], zeros[i0]
        
        h_bin[p1] = 0
        h_bin[p0] = 1
        new_c = binary_c5(h_bin)
        
        if new_c < best_c - 1e-12:
            best_c = new_c
            ones[i1] = p0
            zeros[i0] = p1
        else:
            # Revert
            h_bin[p1] = 1
            h_bin[p0] = 0
    
    return h_bin, best_c

import numpy as np
from scipy.optimize import minimize
import time

def project_constraint(x, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        h = np.clip(x - mid, 0.0, 1.0)
        if np.sum(h) > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        mask = (h > 1e-9) & (h < 1.0 - 1e-9)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def get_c5(h, dx):
    return float(np.max(get_corr(h))) * dx

def smoothed_grad(h, tau, dx):
    n = len(h)
    c = get_corr(h) * dx
    m = np.max(c)
    le = tau * (c - m)
    le -= le.max()
    e = np.exp(le)
    p = e / e.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-14:
            continue
        s = k - (n - 1)
        w = p[k] * dx
        if s >= 0:
            js = slice(s, n)
        else:
            js = slice(0, n + s)
        grad[js] += w * (1.0 - h[np.array(js) - s])
        if s >= 0:
            js2 = slice(0, n - s)
        else:
            js2 = slice(-s, n)
        grad[js2] += w * (-h[np.array(js2) + s])
    return grad

def projected_gradient(h0, target_sum, dx, timeout):
    h = h0.copy()
    t0 = time.time()
    n = len(h)
    for phase in range(10):
        tau = 4.0 * (2.2 ** phase)
        lr = 1.0 / (tau * dx * np.sqrt(n))
        for it in range(60):
            if time.time() - t0 > timeout:
                return h
            obj0 = get_c5(h, dx)
            g = smoothed_grad(h, tau, dx)
            step = lr
            for _bt in range(12):
                h_new = project_constraint(h - step * g, target_sum)
                if get_c5(h_new, dx) <= obj0 + 1e-10:
                    break
                step *= 0.5
            h = h_new
    return h

def binary_search(h0, target_sum, timeout):
    n = len(h0)
    dx = 2.0 / n
    t0 = time.time()
    h_bin = np.zeros(n)
    idx = np.argsort(h0)[::-1]
    h_bin[idx[:int(round(target_sum))]] = 1.0
    best_c = get_c5(h_bin, dx)
    ones = list(np.where(h_bin == 1)[0])
    zeros = list(np.where(h_bin == 0)[0])
    for it in range(50000):
        if time.time() - t0 > timeout:
            break
        i1 = np.random.randint(len(ones))
        i0 = np.random.randint(len(zeros))
        p1, p0 = ones[i1], zeros[i0]
        h_bin[p1] = 0
        h_bin[p0] = 1
        nc = get_c5(h_bin, dx)
        if nc < best_c - 1e-14:
            best_c = nc
            ones[i1] = p0
            zeros[i0] = p1
        else:
            h_bin[p1] = 1
            h_bin[p0] = 0
    return h_bin, best_c

def slsqp_polish(h0, target_sum, dx, n, timeout):
    def f(x):
        x = project_constraint(x, target_sum)
        c = get_corr(x) * dx
        m = np.max(c)
        return m + 0.01 * np.sum(np.log1p(np.exp(40*(c-m))))
    def jac(x):
        x = project_constraint(x, target_sum)
        return smoothed_grad(x, 40.0, dx)
    opts = {'maxiter': min(400, int(timeout*20)), 'ftol': 1e-14}
    try:
        res = minimize(f, h0, method='SLSQP', jac=jac,
                       bounds=[(0,1)]*n,
                       constraints=[{'type':'eq','fun':lambda x: np.sum(x)-target_sum}],
                       options=opts)
        h = project_constraint(res.x, target_sum)
        return h, get_c5(h, dx)
    except:
        return h0, get_c5(h0, dx)

def thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = bin(i).count('1') % 2
    return h
