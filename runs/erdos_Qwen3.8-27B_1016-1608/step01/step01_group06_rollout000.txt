                    for delta in [0.005, -0.005, 0.01, -0.01, 0.002, -0.002]:
                        nv = old + delta
                        if nv < 0 or nv > 1:
                            continue
                        test = best_h.copy()
                        test[i] = nv
                        test = project_feasible(test, n_fin)
                        s_test = compute_score(test)
                        if s_test < best_score - 1e-12:
                            best_score = s_test
                            best_h = test
                            old = best_h[i]
                            improved_any = True
                            break

    # Phase 7: Very tight SLSQP final pass
    if has_time(0.98) and best_score > 0.3805:
        n_fin = len(best_h)
        _g_n = n_fin
        bounds_fin = [(0.0, 1.0)] * n_fin
        slp_con_fin = [{'type': 'eq', 'fun': _sum_eq}]
        _g_tau = 20000
        try:
            res = minimize(_slp_obj_global, best_h, method='SLSQP',
                           bounds=bounds_fin, constraints=slp_con_fin,
                           options={'maxiter': 2000, 'ftol': 1e-16})
            h_cand = project_feasible(res.x, n_fin)
            s_cand = compute_score(h_cand)
            if s_cand < best_score:
                best_score = s_cand
                best_h = h_cand
        except Exception:
            pass

    return best_h, best_score, len(best_h)
