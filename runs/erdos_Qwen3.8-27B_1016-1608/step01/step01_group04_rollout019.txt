import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.array(h, dtype=float).copy()
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        d = n / 2.0 - np.sum(h)
        if abs(d) < 1e-12:
            break
        h += d / n
    return np.clip(h, 0.0, 1.0)

def to_balanced_binary(h, n):
    h = np.array(h, dtype=float)
    idx = np.argsort(h)[::-1]
    b = np.zeros(n, dtype=np.float64)
    b[idx[:n // 2]] = 1.0
    return b

def gen_starts(n, rng):
    """Generate diverse starting binary sequences"""
    starts = []
    # Random
    for _ in range(3):
        h = (rng.random(n) > 0.5).astype(float)
        starts.append(to_balanced_binary(h, n))
    # Periodic patterns
    for p in [3, 4, 5, 6, 7, 8, 10, 12, 13, 16, 20]:
        for duty in range(1, p):
            if duty == p - duty:  # Only unique duty values
                break
            pat = np.zeros(p)
            pat[:duty] = 1.0
            h = np.tile(pat, n // p + 1)[:n]
            starts.append(to_balanced_binary(h, n))
    # Van der Corput
    def vdc(k):
        v, d = 0.0, 0.5
        k += 1
        while k > 0:
            v += (k % 2) * d
            k //= 2
            d /= 2.0
        return v
    h = np.array([vdc(i) for i in range(n)])
    starts.append(to_balanced_binary(h, n))
    # Thue-Morse
    h = np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)
    starts.append(to_balanced_binary(h + 0.3, n))
    return starts

def sa_search(h0, n, rng, n_iter, T0=0.04, T_end=0.00005, kick_interval=5000):
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    cur_val = best_val
    
    zero_list = list(np.where(h == 0)[0])
    one_list = list(np.where(h == 1)[0])
    
    no_improve = 0
    for it in range(n_iter):
        T = T0 * (T_end / T0) ** (it / max(n_iter - 1, 1))
        
        i = rng.integers(len(zero_list))
        j = rng.integers(len(one_list))
        p, q = zero_list[i], one_list[j]
        
        h[p] = 1.0
        h[q] = 0.0
        new_val = compute_c5(h)
        
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / max(T, 1e-15)):
            cur_val = new_val
            zero_list[i] = q
            one_list[j] = p
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[p] = 0.0
            h[q] = 1.0
            no_improve += 1
        
        # Kick: random perturbation
        if no_improve > kick_interval:
            no_improve = 0
            for _ in range(5):
                i2 = rng.integers(len(zero_list))
                j2 = rng.integers(len(one_list))
                p2, q2 = zero_list[i2], one_list[j2]
                h[p2] = 1.0
                h[q2] = 0.0
                zero_list[i2] = q2
                one_list[j2] = p2
            cur_val = compute_c5(h)
    
    return best_h, best_val

def greedy_polish(h0, n, max_rounds=50):
    """Try all single swaps, take best improvement until convergence"""
    h = h0.copy()
    cur_val = compute_c5(h)
    
    for _ in range(max_rounds):
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_delta = 0
        best_pq = None
        
        for zi in range(len(zeros)):
            for oi in range(len(ones)):
                p, q = zeros[zi], ones[oi]
                h[p] = 1.0
                h[q] = 0.0
                v = compute_c5(h)
                delta = v - cur_val
                if delta < best_delta - 1e-15:
                    best_delta = delta
                    best_pq = (p, q)
                h[p] = 0.0
                h[q] = 1.0
        
        if best_pq is not None:
            p, q = best_pq
            h[p] = 1.0
            h[q] = 0.0
            cur_val += best_delta
            improved = True
        
        if not improved:
            break
    
    return h, cur_val

def slsqp_polish(h0, n, maxiter=300):
    """Continuous SLSQP polish"""
    eps = 0.001
    dx = 2.0 / n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        m = np.max(corr)
        s = np.clip((corr - m) / eps, -50, 50)
        return m + eps * np.log(np.sum(np.exp(s)))
    
    def eq_fun(x):
        return np.sum(x) - n / 2.0
    
    def eq_jac(x):
        return np.ones(n)
    
    bds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    
    result = minimize(obj, h0, method='SLSQP', bounds=bds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
    return repair(result.x, n), compute_c5(result.x)
