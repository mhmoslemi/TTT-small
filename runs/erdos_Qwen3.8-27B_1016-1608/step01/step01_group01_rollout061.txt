import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return c.max() * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float, copy=True)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def projected_c5(x, n):
    return compute_c5(project_feasible(x, n))

def coord_descent(h, n, max_passes, deadline):
    h = h.copy()
    best_c5 = compute_c5(h)
    gr = (5.0 ** 0.5 - 1.0) / 2.0
    omg = 1.0 - gr
    for p in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            old_val = h[i]
            a_pt, b_pt = 0.0, 1.0
            c_pt = a_pt + omg * (b_pt - a_pt)
            d_pt = a_pt + gr * (b_pt - a_pt)
            h[i] = c_pt
            fc = compute_c5(project_feasible(h, n))
            h[i] = d_pt
            fd = compute_c5(project_feasible(h, n))
            for _ in range(38):
                if b_pt - a_pt < 1e-13:
                    break
                if fc < fd:
                    b_pt = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = a_pt + omg * (b_pt - a_pt)
                    h[i] = c_pt
                    fc = compute_c5(project_feasible(h, n))
                else:
                    a_pt = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = a_pt + gr * (b_pt - a_pt)
                    h[i] = d_pt
                    fd = compute_c5(project_feasible(h, n))
            h[i] = (a_pt + b_pt) * 0.5
