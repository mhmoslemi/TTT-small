                trial = best_h.copy()
                s = int(rng.randint(0, max(1, n - 5)))
                l = int(rng.randint(2, min(6, n - s)))
                trial[s:s+l] += rng.uniform(-0.3, 0.3)
            else:
                trial = best_h.copy()
                idx = rng.choice(n, size=max(1, n // 10), replace=False)
                trial[idx] = rng.uniform(0, 1, size=len(idx))
            trial = repair(trial, n)
            c5 = compute_c5(trial)
            if c5 < best_c5 - 1e-10:
                best_c5 = c5
                best_h = trial
                improved = True
        if not improved:
            step *= 0.7
            if step < 0.001:
                break
    return best_h, best_c5

def coord_descent(h, n, time_limit):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t_start = time.time()
    step = 0.05
    while time.time() - t_start < time_limit:
        improved = False
        for _ in range(500):
            if time.time() - t_start > time_limit:
                break
            i = np.random.randint(n)
            for delta in [step, -step, step * 0.5, -step * 0.5]:
                trial = best_h.copy()
                trial[i] = np.clip(trial[i] + delta, 0.0, 1.0)
                trial = repair(trial, n)
                c5 = compute_c5(trial)
                if c5 < best_c5 - 1e-12:
                    best_c5 = c5
                    best_h = trial
                    improved = True
                    break
        if not improved:
            step *= 0.7
            if step < 0.003:
                break
    return best_h, best_c5

def interpolate_h(h_parent, n_new):
    n_old = len(h_parent)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_parent, period=2.0)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_parent = np.clip(0.5 + 0.35 * np.sin(np.pi * np.arange(80) / 40.0), 0, 1)
    n_par = len(h_parent)
    h_parent = repair(h_parent, n_par)
    c5_par = compute_c5(h_parent)

    best_h = h_parent.copy()
    best_c5 = c5_par
    best_n = n_par

    # Phase 1: DE at n=200 (higher resolution)
    n1 = 200
    h1_init = interpolate_h(h_parent, n1)
    h1_init = repair(h1_init, n1)
    t1_limit = min(450, budget_s * 0.4)
    h1, c5_1 = de_run(h1_init, n1, t1_limit, seed)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1
        best_n = n1

    # Phase 2: DE at parent n (different seed, more structured starts)
    elapsed = time.time() - t0
    if elapsed < budget_s * 0.65:
        t2_limit = min(200, budget_s * 0.2)
        h2, c5_2 = de_run(h_parent, n_par, t2_limit, seed + 7777, pop_size=50, F=0.7, CR=0.7)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2
            best_n = n_par

    # Phase 3: DE at n=400 (even higher resolution, shorter time)
    elapsed = time.time() - t0
    if elapsed < budget_s * 0.75:
        n3 = 400
        h3_init = interpolate_h(h_parent, n3)
        h3_init = repair(h3_init, n3)
        t3_limit = min(180, budget_s - elapsed - 100)
        if t3_limit > 30:
            h3, c5_3 = de_run(h3_init, n3, t3_limit, seed + 31415, pop_size=40, F=0.6, CR=0.7, max_gen=200)
            if c5_3 < best_c5:
                best_c5 = c5_3
                best_h = h3
                best_n = n3

    # Phase 4: Local polish on best solution
    elapsed = time.time() - t0
    if elapsed < budget_s - 50:
        t4_limit = min(200, budget_s - elapsed - 30)
        bh, bc = local_polish(best_h, best_n, t4_limit, seed + 999)
        if bc < best_c5:
            best_c5 = bc
            best_h = bh

    # Phase 5: Coordinate descent
    elapsed = time.time() - t0
    if elapsed < budget_s - 20:
        t5_limit = min(100, budget_s - elapsed - 10)
        bh, bc = coord_descent(best_h, best_n, t5_limit)
        if bc < best_c5:
            best_c5 = bc
            best_h = bh

    # Phase 6: Final tiny perturbations
    elapsed = time.time() - t0
    if elapsed < budget_s - 5:
        rng = np.random.RandomState(seed + 555)
        for _ in range(2000):
            if time.time() - t0 > budget_s - 2:
                break
            trial = best_h + rng.randn(best_n) * 0.005
            trial = repair(trial, best_n)
            c5 = compute_c5(trial)
            if c5 < best_c5:
                best_c5 = c5
                best_h = trial

    # Final repair and validation
    best_h = repair(best_h, best_n)
    best_c5 = compute_c5(best_h)
    assert 0.0 <= np.min(best_h) and np.max(best_h) <= 1.0
    assert abs(np.sum(best_h) - best_n / 2.0) < 1e-6

    return (best_h.tolist(), float(best_c5), best_n)
