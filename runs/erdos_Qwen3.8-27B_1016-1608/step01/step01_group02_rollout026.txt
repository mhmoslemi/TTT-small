        except Exception:
            pass
        if tl() > 120:
            try:
                de_res2 = differential_evolution(de_objective, de_bounds, args=(dx, target),
                                                seed=seed + 100, maxiter=70,
                                                popsize=20, tol=1e-6,
                                                mutation=(0.6, 1.2), recomb=0.9,
                                                polish=False)
                try_update(project(de_res2.x, target), n)
            except Exception:
                pass

    # Phase 4: Multi-start from diverse initializations
    n_starts = 25
    for i in range(n_starts):
        if tl() < 50:
            break
        s_mod = i % 5
        if s_mod == 0:
            h0 = best_h + rng.randn(n) * rng.uniform(0.05, 0.2)
        elif s_mod == 1:
            h0 = np.zeros(n)
            blk = rng.randint(4, 14)
            for st in range(0, n, 2 * blk):
                h0[st:st + blk] = 1.0
            h0 += rng.randn(n) * 0.05
        elif s_mod == 2:
            t_arr = np.linspace(0, 4 * np.pi * rng.randint(1, 4), n)
            h0 = 0.5 + 0.35 * np.sin(t_arr + rng.uniform(0, 6.28))
        elif s_mod == 3:
            h0 = 1.0 - best_h + rng.randn(n) * 0.08
        else:
            h0 = rng.random(n) * rng.uniform(0.4, 1.0) + 0.1
        h0 = project(h0, target)

        for T in [30.0, 100.0]:
            if tl() < 15:
                break
            try:
                res = minimize(lbgfs_wrapper, h0, args=(dx, T, target, 100.0),
                              jac=True, method='L-BFGS-B', bounds=bounds,
                              options={'maxiter': 300, 'ftol': 1e-16})
                try_update(project(res.x, target), n)
            except Exception:
                pass
        if tl() > 20:
            try:
                cons = ({'type': 'eq', 'fun': sum_con, 'args': (target,)},)
                res = minimize(compute_c5, best_h if i == 0 else project(h0, target),
                              args=(dx,), method='SLSQP', bounds=bounds,
                              constraints=cons, options={'maxiter': 400, 'ftol': 1e-15})
                try_update(project(res.x, target), n)
            except Exception:
                pass

    # Phase 5: Coordinate descent with adaptive steps
    if tl() > 50:
        h_c = best_h.copy()
        c5_c = best_c5
        step = 0.06
        for it in range(80000):
            if tl() < 10:
                break
            idx = rng.randint(n)
            improved = False
            for delta in [step, -step]:
                h_t = h_c.copy()
                h_t[idx] = np.clip(h_t[idx] + delta, 0.0, 1.0)
                h_t = project(h_t, target)
                c5_t = compute_c5(h_t, dx)
                if c5_t < c5_c - 1e-15:
                    h_c, c5_c = h_t.copy(), c5_t
                    if c5_t < best_c5:
                        best_h = h_c.copy()
                        best_c5 = c5_t
                    improved = True
                    break
            if it % 2000 == 1999:
                step = max(0.003, step * 0.75)

    # Phase 6: Try alternate n values
    for n_alt in [50, 80, 100, 140]:
        if tl() < 80:
            break
        dx_a = 2.0 / n_alt
        target_a = n_alt / 2.0
        bounds_a = [(0.0, 1.0)] * n_alt
        x_old = np.linspace(0.0, 2.0, n)
        x_new = np.linspace(0.0, 2.0, n_alt)
        h_init_a = np.interp(x_new, x_old, best_h)
        h_init_a = project(h_init_a, target_a)

        for T in [30.0, 150.0]:
            if tl() < 25:
                break
            try:
                res = minimize(lbgfs_wrapper, h_init_a, args=(dx_a, T, target_a, 200.0),
                              jac=True, method='L-BFGS-B', bounds=bounds_a,
                              options={'maxiter': 400, 'ftol': 1e-17})
                try_update(project(res.x, target_a), n_alt)
            except Exception:
                pass
        if tl() > 20:
            try:
                cons_a = ({'type': 'eq', 'fun': sum_con, 'args': (target_a,)},)
                res = minimize(compute_c5, h_init_a, args=(dx_a,), method='SLSQP',
                              bounds=bounds_a, constraints=cons_a,
                              options={'maxiter': 600, 'ftol': 1e-15})
                try_update(project(res.x, target_a), n_alt)
            except Exception:
                pass

    # Phase 7: Final SLSQP polish on best
    if tl() > 15:
        n_b = best_n
        dx_b = 2.0 / n_b
        target_b = n_b / 2.0
        bounds_b = [(0.0, 1.0)] * n_b
        try:
            cons_b = ({'type': 'eq', 'fun': sum_con, 'args': (target_b,)},)
            res = minimize(compute_c5, best_h, args=(dx_b,), method='SLSQP',
                          bounds=bounds_b, constraints=cons_b,
                          options={'maxiter': 2000, 'ftol': 1e-16})
            try_update(project(res.x, target_b), n_b)
        except Exception:
            pass

    best_h = project(best_h, best_n / 2.0)
    final_c5 = compute_c5(best_h, 2.0 / best_n)
    return (best_h, final_c5, best_n)
