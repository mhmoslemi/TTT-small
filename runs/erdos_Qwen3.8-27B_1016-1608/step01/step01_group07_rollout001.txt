def _c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def _fix(h, n, tol=1e-14):
    target = 0.5 * n
    for _ in range(200):
        d = target - np.sum(h)
        if abs(d) < tol:
            break
        if d > 0:
            mask = (h < 1.0).astype(float)
        else:
            mask = (h > 0.0).astype(float)

        if k == 0:
            break
        h = h + (d / k) * mask
        h = np.clip(h, 0.0, 1.0)
