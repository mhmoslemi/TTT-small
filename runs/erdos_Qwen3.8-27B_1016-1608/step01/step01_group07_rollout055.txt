import numpy as np
from scipy.optimize import minimize
import time

def _project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        d = target_sum - h.sum()
        if abs(d) < 1e-13:
            break
        mask = h < 1.0 if d > 0 else h > 0.0
        k = int(mask.sum())
        if k == 0:
            break
        h[mask] += d / k
        h = np.clip(h, 0.0, 1.0)
    return h

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def _c5_corrs(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c)), c

def _grad_at_shift(h, s):
    n = len(h)
    g = np.zeros(n)
    if s == 0:
        return 1.0 - 2.0 * h
    if s > 0:
        g[:n-s] += 1.0
        g[:n-s] -= h[s:]
        g[s:] -= h[:n-s]
    else:
        s2 = -s
        g[s2:] += 1.0
        g[s2:] -= h[:n-s2]
        g[:n-s2] -= h[s2:]
    return g

def _subgrad_optimize(h, n_iter, lr0, top_k, momentum, t0, budget):
    n = len(h)
    target = n / 2.0
    v = np.zeros(n)
    best_h = h.copy()
    best_c = _c5(h)
    lr = lr0
    
    for it in range(n_iter):
        if time.time() - t0 > budget - 10:
            break
        
        _, corrs = _c5_corrs(h)
        # Get top-k worst lags
        top_idx = np.argsort(corrs)[-top_k:]
        
        grad = np.zeros(n)
        for m in top_idx:
            s = int(m) - (n - 1)
            grad += _grad_at_shift(h, s)
        grad /= top_k
        
        v = momentum * v - lr * grad
        h_new = h + v
        h_new = _project(h_new, target)
        
        c_new = _c5(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        
        # Always step (even if not improving) for exploration
        h = h_new
        
        # Adaptive lr
        lr *= 0.999
        
    return best_h, best_c

def _binary_flip_search(h, n_iters, rng, t0, budget):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c = _c5(h)
    
    for _ in range(n_iters):
        if time.time() - t0 > budget - 5:
            break
        _, corrs = _c5_corrs(h)
        worst_m = int(np.argmax(corrs))
        s = worst_m - (n - 1)
        
        # Find problem positions
        if s > 0:
            # h[j] is 1 where h[j+s] is 0
            contrib = np.zeros(n)
            valid = slice(0, n-s)
            contrib[valid] = h[valid] * (1 - h[s:])
        elif s < 0:
            s2 = -s
            contrib = np.zeros(n)
            valid = slice(s2, n)
            contrib[valid] = h[valid] * (1 - h[valid - s2])
        else:
            contrib = h * (1 - h)
        
        # Pick a position with high bad contribution
        bad_mask = contrib > 0.2
        if not bad_mask.any():
            # Pick positions where we can try flipping
            if (h > 0.7).any() and (h < 0.3).any():
                candidates = np.where(h > 0.7)[0]
            else:
                break
            t = rng.choice(candidates)
        else:
            candidates = np.where(bad_mask)[0]
            t = rng.choice(candidates)
        
        # Try flipping
        h_orig = h.copy()
        h[t] = 1.0 - h[t]
        h = _project(h, target)
        c_new = _c5(h)
        
        if c_new < best_c:
            best_c = c_new
            best_h = h.copy()
        else:
            h = h_orig
    
    return best_h, best_c

def _slsqp_polish(h, n, alpha, maxiter, t0, budget):
    if budget - (time.time() - t0) < 15:
        return None
    target = n / 2.0
    
    def _obj(x):
        dx = 2.0 / n
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = np.max(c)
        e = np.exp(np.clip(c - m, -50.0, 50.0))
        return m + np.log(np.sum(e)) / alpha
    
    def _cons_fun(x):
        return np.sum(x) - target
    
    def _cons_jac(x):
        return np.ones(n)
    
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(_obj, h, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-10})
        if res.x is not None:
            ht = _project(np.clip(res.x, 0.0, 1.0), target)
            return ht
    except Exception:
        pass
    return None

def _interp(h, n_new):
    n_old = len(h)
    x_old = np.arange(n_old) * (2.0 / n_old)
    x_new = np.arange(n_new) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    # Get initial values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        h = np.full(45, 0.5)
    
    n = len(h)
    h = _project(h, n / 2.0)
    best_h = h.copy()
    best_c = _c5(h)
    
    # Phase 1: Subgradient with momentum
    for lr0 in [0.05, 0.02, 0.01, 0.005]:
        for top_k in [1, 3, 5, 10]:
            n_iter = 300
            hh, cc = _subgrad_optimize(h, n_iter, lr0, top_k, 0.9, t0, budget_s)
            if cc < best_c:
                best_c = cc
                best_h = hh
                h = hh
    
    # Phase 2: SLSQP polish
    for alpha in [100.0, 500.0, 2000.0]:
        ht = _slsqp_polish(h, n, alpha, 1000, t0, budget_s)
        if ht is not None:
            cc = _c5(ht)
            if cc < best_c:
                best_c = cc
                best_h = ht.copy()
                h = ht
    
    # Phase 3: Binary flip search
    hh, cc = _binary_flip_search(h, 500, rng, t0, budget_s)
