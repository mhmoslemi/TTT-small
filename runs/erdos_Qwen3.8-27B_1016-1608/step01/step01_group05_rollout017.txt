    for it in range(max_iter):
        c5 = compute_c5(h, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        g = topk_subgrad(h, n, k_top)
        v = momentum * v + (1.0 - momentum) * g
        h_new = project_to_set(h - lr * v, target)
        c5_new = compute_c5(h_new, n)
        if c5_new <= c5 + 1e-12:
            h = h_new
            if c5_new < c5 - 1e-10:
                lr = min(lr * 1.02, 0.15)
        else:
            lr *= 0.9
            h_new = project_to_set(h - lr * v, target)
            h = h_new
        if stall > 800:
            break
        if lr < 1e-9:
            break
    return best_h, best_c5

def coordinate_descent(h, n, rng, n_sweeps=30):
    target = n / 2.0
    h = h.copy()
    best = compute_c5(h, n)
    for sw in range(n_sweeps):
        order = rng.permutation(n)
        improved = False
        for j in order:
            old = h[j]
            step_sizes = [0.05, 0.1, 0.02, 0.2]
            best_c = old
            best_v = best
            for ss in step_sizes:
                for direction in (-1.0, 1.0):
                    c = old + direction * ss
                    c = max(0.0, min(1.0, c))
                    if abs(c - old) < 1e-8:
                        continue
                    h[j] = c
                    h = project_to_set(h, target)
                    v = compute_c5(h, n)
                    if v < best_v:
                        best_v = v
                        best_c = c
            if abs(best_c - old) > 1e-8:
                h[j] = best_c
                h = project_to_set(h, target)
                best = best_v
                improved = True
            else:
                h[j] = old
        if not improved:
            break
    return h, best

def upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)

def smooth_step(h, alpha=0.85):
    n = len(h)
    out = np.zeros(n)
    for i in range(n):
        out[i] = alpha * h[i] + (1.0 - alpha) * 0.5 * (h[(i - 1) % n] + h[(i + 1) % n])
    return out

def _time_left():
    return max(0.0, _budget - (time.time() - _start_t))

def run(seed=42, budget_s=1100, **kwargs):
    global _budget, _start_t
    _budget = float(budget_s)
    _start_t = time.time()
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = np.inf
    best_n = 0

    def _update(h, n):
        nonlocal best_h, best_c5, best_n
        h = project_to_set(np.asarray(h, dtype=float), n / 2.0)
        c = compute_c5(h, n)
        if c < best_c5:
            best_h = h.copy()
            best_c5 = c
            best_n = n

    # Phase 1: n=90 subgradient optimization
    n1 = 90
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        if len(h_init) != n1:
            h_init = upsample(h_init, n1)
    except Exception:
        x = np.linspace(0, 2, n1, endpoint=False)
        h_init = 0.5 + 0.4 * np.sin(np.pi * x) + 0.2 * np.sin(2 * np.pi * x)
    h_init = project_to_set(h_init, n1 / 2.0)
    _update(h_init, n1)

    # Subgradient with various k_top
    k_tops = [3, 5, 7]
    for kt in k_tops:
        if _time_left() < 100.0:
            break
        h_opt, c5 = subgrad_optimize(h_init, n1, lr_init=0.04, k_top=kt,
                                      momentum=0.7, max_iter=4000)
        _update(h_opt, n1)
        h_init = best_h.copy()

    # Coordinate descent polish
    h_cd, c5_cd = coordinate_descent(best_h, best_n, rng, n_sweeps=25)
    _update(h_cd, best_n)

    # Restart with perturbations
    n_restarts = 0
    while _time_left() > 250.0 and n_restarts < 6:
        n_restarts += 1
        noise = rng.normal(0, 0.08, best_n)
        h_r = project_to_set(best_h + noise, best_n / 2.0)
        h_opt, c5 = subgrad_optimize(h_r, best_n, lr_init=0.03, k_top=5,
                                      momentum=0.75, max_iter=2500)
        _update(h_opt, best_n)
        h_cd, c5_cd = coordinate_descent(best_h, best_n, rng, n_sweeps=10)
        _update(h_cd, best_n)

    # Phase 2: n=180
    if _time_left() > 180.0:
        n2 = 180
        h_up = project_to_set(upsample(best_h, n2), n2 / 2.0)
        _update(h_up, n2)
        h_opt, c5 = subgrad_optimize(h_up, n2, lr_init=0.03, k_top=5,
                                      momentum=0.7, max_iter=3000)
        _update(h_opt, n2)
        h_opt, c5 = subgrad_optimize(best_h, best_n, lr_init=0.02, k_top=3,
                                      momentum=0.75, max_iter=2000)
        _update(h_opt, best_n)
        h_cd, c5_cd = coordinate_descent(best_h, best_n, rng, n_sweeps=20)
        _update(h_cd, best_n)

        # More subgradient rounds at n=180
        while _time_left() > 100.0:
            h_soft = smooth_step(best_h, alpha=0.7)
            h_soft = project_to_set(h_soft, best_n / 2.0)
            h_opt, c5 = subgrad_optimize(h_soft, best_n, lr_init=0.02, k_top=5,
                                          momentum=0.7, max_iter=1500)
            _update(h_opt, best_n)
            if compute_c5(best_h, best_n) >= best_c5 + 1e-6:
                break

    # Phase 3: n=360 (if time allows)
    if _time_left() > 80.0 and best_n < 360:
        n3 = 360
        h_up = project_to_set(upsample(best_h, n3), n3 / 2.0)
        _update(h_up, n3)
        h_opt, c5 = subgrad_optimize(h_up, n3, lr_init=0.02, k_top=5,
                                      momentum=0.7, max_iter=2000)
        _update(h_opt, n3)
        h_cd, c5_cd = coordinate_descent(best_h, best_n, rng, n_sweeps=15)
        _update(h_cd, best_n)

    # Final coordinate descent polish
    if _time_left() > 15.0:
        h_cd, c5_cd = coordinate_descent(best_h, best_n, rng, n_sweeps=40)
        _update(h_cd, best_n)

    # Ensure valid output
    best_h = project_to_set(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h, best_n)
    return best_h, best_c5, best_n
