    for k_idx in lags:
        k = k_idx - (n - 1)
        ms = max(0, k)
        me = min(n, n + k)
        if me > ms:
            m = np.arange(ms, me)
            g[m] += (1.0 - h[m - k])
        ms2 = max(0, -k)
        me2 = min(n, n - k)
        if me2 > ms2:
            m2 = np.arange(ms2, me2)
            g[m2] += -h[m2 + k]
    g /= len(lags)
    g *= dx
    return g

def make_sinusoid(n, period, amp, phase):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = 0.5 + amp * np.sin(2.0 * np.pi * x / period + phase)
    h = np.clip(h, 0.0, 1.0)
    return h

def runactive_slsqp(h, n, bounds, cons, t0, budget_s, max_outer):
    local_best = compute_score(h)
    local_bh = h.copy()
    for outer in range(max_outer):
        if time.time() - t0 > budget_s * 0.86:
            break
        c = compute_c(local_bh)
        cm = np.max(c)
        n_act = min(4, len(c))
        idx = np.argsort(c)[::-1]
        lags = idx[:n_act]
        try:
            res = minimize(active_set_obj, local_bh.copy(), args=(lags,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 400, 'ftol': 1e-13})
            h_new = project_feasible(res['x'], n)
            s = compute_score(h_new)
            if s < local_best:
                local_best = s
                local_bh = h_new.copy()
        except Exception:
            pass
    return local_bh, local_best

def run_subgrad(h, n, t0, budget_s, n_iters):
    h_cur = h.copy()
    best_s = compute_score(h_cur)
    best_h = h_cur.copy()
    for it in range(n_iters):
        if time.time() - t0 > budget_s * 0.88:
            break
        c = compute_c(h_cur)
        cm = np.max(c)
        eps = 0.003
        act = np.where(c >= cm - eps)[0]
        if len(act) > 6:
            top_i = np.argsort(c[act])[::-1][:6]
            act = act[top_i]
        g = subgrad(h_cur, act, n)
        alpha = 0.08 / (1.0 + 0.003 * it)
        h_new = project_feasible(h_cur - alpha * g, n)
        s = compute_score(h_new)
        if s < best_s:
            best_s = s
            best_h = h_new.copy()
        h_cur = h_new
    return best_h, best_s

def run(seed=42, budget_s=1100, **kwargs):
    global _global_n
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    best_h = h_init.copy()
    best_score = compute_score(h_init)

    x_init = np.linspace(0.0, 2.0, n_init, endpoint=False)
    n_list = sorted(set([n_init, 40, 50, 64, 80, 100]))

    for n in n_list:
        if time.time() - t0 > budget_s * 0.80:
            break
        _global_n = n
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _con_fun}]

        if n == n_init:
            h_base = h_init.copy()
        else:
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h_base = np.interp(x_new, x_init, h_init)
            h_base = project_feasible(h_base, n)

        # Build diverse starting points
        starts = [h_base]
        # Sinusoidal starts
        for period in [1.0, 0.5, 2.0, 2.0/3.0]:
            for amp in [0.3, 0.4, 0.45]:
                for phase in [0.0, np.pi/4, np.pi/2]:
                    hs = project_feasible(make_sinusoid(n, period, amp, phase), n)
                    starts.append(hs)
        # Random starts
        for _ in range(4):
            hr = rng.uniform(0.0, 1.0, size=n)
            starts.append(project_feasible(hr, n))

        local_best_score = compute_score(h_base)
        local_best_h = h_base.copy()

        # Phase 1: Active-set SLSQP on multiple starts
        for si, h_start in enumerate(starts[:10]):
            if time.time() - t0 > budget_s * 0.82:
                break
            s_init = compute_score(h_start)
            if s_init > local_best_score + 0.05:
                continue
            max_outer = 25 if n <= 64 else 15
            h_out, s_out = runactive_slsqp(h_start, n, bounds, cons, t0, budget_s, max_outer)
            if s_out < local_best_score:
                local_best_score = s_out
                local_best_h = h_out.copy()

        # Phase 2: Subgradient refinement
        if time.time() - t0 < budget_s * 0.80:
            n_sg = 800 if n <= 64 else 400
            h_sg, s_sg = run_subgrad(local_best_h, n, t0, budget_s, n_sg)
            if s_sg < local_best_score:
                local_best_score = s_sg
                local_best_h = h_sg.copy()

        # Phase 3: Nelder-Mead with penalty for small n
        if n <= 80 and time.time() - t0 < budget_s * 0.75:
            def penal(h):
                return compute_score(h) + 500.0 * (np.sum(h) - n / 2.0) ** 2
            for nm_try in range(2):
                if time.time() - t0 > budget_s * 0.78:
                    break
                hs = local_best_h + rng.randn(n) * 0.03
                hs = project_feasible(hs, n)
                try:
                    res = minimize(penal, hs, method='Nelder-Mead',
                                   options={'maxiter': 3000, 'xatol': 1e-7, 'fatol': 1e-11})
                    h_cand = project_feasible(res['x'], n)
                    s_cand = compute_score(h_cand)
                    if s_cand < local_best_score:
                        local_best_score = s_cand
                        local_best_h = h_cand.copy()
                except Exception:
                    pass

        if local_best_score < best_score:
            best_score = local_best_score
            best_h = local_best_h.copy()

    # Final polish with exact objective
    if time.time() - t0 < budget_s * 0.92:
        n_fin = len(best_h)
        _global_n = n_fin
        bounds = [(0.0, 1.0)] * n_fin
        cons = [{'type': 'eq', 'fun': _con_fun}]
        c_codes = compute_c(best_h)
        lags_top = np.argsort(c_codes)[::-1][:4]
        try:
            res = minimize(active_set_obj, best_h.copy(), args=(lags_top,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 600, 'ftol': 1e-14})
            h_f = project_feasible(res['x'], n_fin)
            s_f = compute_score(h_f)
            if s_f < best_score:
                best_score = s_f
                best_h = h_f
        except Exception:
            pass
        if time.time() - t0 < budget_s * 0.96:
            try:
                res2 = minimize(compute_score, best_h, method='SLSQP',
                                bounds=bounds, constraints=cons,
                                options={'maxiter': 500, 'ftol': 1e-15})
                h_f2 = project_feasible(res2['x'], n_fin)
                s_f2 = compute_score(h_f2)
                if s_f2 < best_score:
                    best_score = s_f2
                    best_h = h_f2
            except Exception:
                pass

    return best_h, best_score, len(best_h)
