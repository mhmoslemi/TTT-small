from scipy.optimize import minimize
import time

def _c5_val(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def _fix_sum(h, n, iters=500):
    target = 0.5 * n
    for _ in range(iters):
        d = target - np.sum(h)
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h = np.clip(h + d / cnt * mask, 0.0, 1.0)
    return h

def _smooth_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(c)
    return m + np.log(np.sum(np.exp(np.clip(c - m, -60, 60)))) / alpha

def _smooth_obj_full(h, alpha):
    """Return (objective, gradient) for smooth max."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(c)
    cs = c - m
    e = np.exp(np.clip(cs, -60, 60))
    s = np.sum(e)
    w = e / s
    obj = m + np.log(s) / alpha

    grad_w = np.zeros(n)
    for p in range(n):

def _subgrad_step(h, lr, n):
    """One projected subgradient step for minimizing max c_k."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(c)
    # Active set: lags where c_k is close to max
    active = np.where(c >= m - 1e-6)[0]
    # Subgradient: sum of dc_k/dh for active k (normalized)
    # For simplicity, use the max lag
    k_star = np.argmax(c)
    L = k_star - (n - 1)
    aL = abs(L)
    
    g = np.zeros(n)
    for p in range(n):
        gp = 0.0
        if p + aL < n:
            gp += (1.0 - h[p + aL])
        if p >= aL:
            gp -= h[p - aL]
        g[p] = gp * dx  # scale by dx since c = correlate * dx
    
    # For aL = 0: gp = (1-h[p]) - 0 = 1-h[p]
    # Wait, if aL=0: (1-h[p+0])*1(p<n) - h[p-0]*1(p>=0) = (1-h[p]) - h[p] = 1-2h[p]
    # That's correct: dc_0/dh[p] = 1-2h[p] (since c_0 = sum h*(1-h) = sum(h - h²), d/dh = 1-2h)
    
    h_new = h - lr * g
    h_new = _fix_sum(np.clip(h_new, 0.0, 1.0), n)
    return h_new

def _slsqp_opt(h0, n, alpha, maxiter, t0, budget):
    if budget - (time.time() - t0) < 5:
        return None
    bnds = [(0.0, 1.0)] * n
    one_vec = np.ones(n)
    
    def objf(x):
        return _smooth_obj(x, alpha)
    
    def cons_eq(x):
        return np.sum(x) - 0.5 * n
    
    def cons_jac(x):
        return one_vec
    
    try:
        res = minimize(objf, h0, method='SLSQP',
                       bounds=bnds,
                       constraints=[{'type': 'eq', 'fun': cons_eq, 'jac': cons_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None:
            return _fix_sum(np.clip(res.x, 0.0, 1.0), n)
    except Exception:
        pass
    return None

def _interp(h, n_new):

    xo = np.linspace(0, 2, n_old, endpoint=False)
    xo_c = xo + 1.0/n_old  # cell centers
    xn_c = np.linspace(0, 2, n_new, endpoint=False) + 1.0/n_new
    h_new = np.interp(xn_c, xo_c, h)
    return _fix_sum(np.clip(h_new, 0.0, 1.0), n_new)
