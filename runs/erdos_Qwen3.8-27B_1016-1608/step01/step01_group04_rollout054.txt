        m = np.max(corr)
        s = np.clip((corr - m) / eps, -50, 50)
        return m + eps * np.log(np.sum(np.exp(s)))

        return np.ones(n)

    bounds = [(0, 1)] * n

                    c5_best = c5
                    h_best = h_ref

    # Phase 5: Final micro-refinement
    if time.time() - t0 < budget_s - 15:
        hm = n_best
        tm = target_best
        for _ in range(3000):
            if time.time() - t0 > budget_s - 5:
                break
            hp = h_best + np.random.randn(hm) * 0.002
            hp = np.clip(hp, 0.0, 1.0)
            hp = repair(hp, tm)
            c5 = compute_c5(hp)
            if c5 < c5_best:
                c5_best = c5
                h_best = hp

    if time.time() - t0 < budget_s - 8:
        h_ref = do_slsqp(h_best, n_best, target_best, 0.0005, 200)
        c5 = compute_c5(h_ref)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_ref
            n_best = len(h_best)
            target_best = target_best

    h_final = repair(h_best, target_best)
    c5_final = compute_c5(h_final)
    return (h_final.tolist(), float(c5_final), len(h_final))
