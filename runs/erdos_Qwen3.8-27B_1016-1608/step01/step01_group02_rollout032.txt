        x_n = np.linspace(0, 2, n_try)
        h_i = np.interp(x_n, x_o, best_h)
        h_i = project_box_sum(h_i, tgt_t)
        for temp in [50, 200]:
            if tl() < 30:
                break
            hr = slsqp_solve(h_i, dx_t, tgt_t, temp, 400)
            try_up(hr, n_try)
        hr = slsqp_solve(h_i, dx_t, tgt_t, 1e8, 350)
        try_up(hr, n_try)
        # Binary restart
        h_b = (rng.random(n_try) < 0.5).astype(float)
        h_b = project_box_sum(h_b, tgt_t)
        hr = slsqp_solve(h_b, dx_t, tgt_t, 50, 350)
        try_up(hr, n_try)
        hr = slsqp_solve(hr, dx_t, tgt_t, 1e8, 300)
        try_up(hr, n_try)
        # Block pattern restart
        h_blk = np.zeros(n_try)
        for i in range(0, n_try, 4):
            h_blk[i:i + 2] = 1.0
        h_blk = project_box_sum(h_blk, tgt_t)
        hr = slsqp_solve(h_blk, dx_t, tgt_t, 30, 350)
        try_up(hr, n_try)

    # Phase 5: More SA rounds at best_n
    if tl() > 80:
        for temp_e in [10, 30]:
            h_sa = sa_swaps(best_h, best_dx, best_tgt, temp_e, 25000, rng)
            try_up(h_sa, best_n)
        hr = slsqp_solve(best_h, best_dx, best_tgt, 5, 500)
        try_up(hr, best_n)
        hr = slsqp_solve(best_h, best_dx, best_tgt, 1e8, 600)
        try_up(hr, best_n)

    # Phase 6: Pairwise swap local search
    if tl() > 30:
        h_ls = best_h.copy()
        c5_ls = best_c5
        improved = True
        while improved and tl() > 10:
            improved = False
            order = rng.permutation(best_n)
            for i in order:
                if tl() < 5:
                    break
                for j in order:
                    if i == j or abs(h_ls[i] - h_ls[j]) < 1e-12:
                        continue
                    h_ls[i], h_ls[j] = h_ls[j], h_ls[i]
                    h_ls = project_box_sum(h_ls, best_tgt)
                    c = compute_c5(h_ls, best_dx)
                    if c < c5_ls - 1e-14:
                        c5_ls = c
                        improved = True
                        if c5_ls < best_c5:
                            best_h, best_c5 = h_ls.copy(), c5_ls
                        break
                    else:
                        h_ls[i], h_ls[j] = h_ls[j], h_ls[i]
                        h_ls = project_box_sum(h_ls, best_tgt)

    # Phase 7: Final polish
    if tl() > 15:
        hr = slsqp_solve(best_h, best_dx, best_tgt, 1e8, 800)
        try_up(hr, best_n)
    if tl() > 5:
        hr = slsqp_solve(best_h, best_dx, best_tgt, 1e8, 500)
        try_up(hr, best_n)

    best_h = project_box_sum(best_h, best_tgt)
    final_c5 = compute_c5(best_h, best_dx)
    return (best_h, final_c5, best_n)
