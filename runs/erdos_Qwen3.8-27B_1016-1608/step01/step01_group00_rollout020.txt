            h = slsqp_opt(h, n, maxiter=300)
        else:
            h_p = 1.0 - h_best
            h_p[:] *= 0.3
            h_p += 0.35
            h_p = repair(h_p, n)
            h = subgrad_opt(h_p, n, n_iter=60, lr0=0.5, mom=0.8)

        c = compute_c5(h)
        if c < c5_best - 1e-9:
            h_best, c5_best = h, c
            stall = 0
        else:
            stall += 1
        it += 1

    # Phase 4: Try higher resolution
    for n2 in [100, 140, 200]:
        if time.time() - t0 > budget_s * 0.88:
            break
        xp = np.linspace(0.0, 1.0, n)
        xn = np.linspace(0.0, 1.0, n2)
        h_i = np.interp(xn, xp, h_best)
        h_i = repair(h_i, n2)
        h2 = subgrad_opt(h_i, n2, n_iter=60, lr0=0.4, mom=0.8)
        h2 = slsqp_opt(h2, n2, maxiter=600)
        c2 = compute_c5(h2)
        if c2 < c5_best:
            h_best, c5_best, n = h2, c2, n2

    # Phase 5: Intensive final polish
    if time.time() - t0 < budget_s * 0.93:
        h = subgrad_opt(h_best, n, n_iter=200, lr0=0.3, mom=0.9)
        c = compute_c5(h)
        if c < c5_best:
            h_best, c5_best = h, c
        h = slsqp_opt(h_best, n, maxiter=3000)
        c = compute_c5(h)
        if c < c5_best:
            h_best, c5_best = h, c

    # Phase 6: Micro-tuning with very small steps
    if time.time() - t0 < budget_s * 0.98:
        for _ in range(50):
            h_p = repair(h_best + rng.randn(n) * 0.003, n)
            h = slsqp_opt(h_p, n, maxiter=1000)
            c = compute_c5(h)
            if c < c5_best:
                h_best, c5_best = h, c

    h_best = repair(h_best, n)
    c5_best = compute_c5(h_best)
    return (h_best, c5_best, n)
