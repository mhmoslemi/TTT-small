import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        d = n / 2.0 - np.sum(h)
        if abs(d) < 1e-13:
            break
        # distribute correction to elements not at boundary
        mask = (h > 0.001) & (h < 0.999)
        if np.sum(mask) == 0:
            h += d / n
        else:
            h[mask] += d / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    # Final exact adjustment
    h = np.clip(h, 0.0, 1.0)
    d = n / 2.0 - np.sum(h)
    if abs(d) > 1e-12:
        # spread over all
        h += d / n
        h = np.clip(h, 0.0, 1.0)
    return h

def smax(x, n, eps):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(c)
    s = np.clip((c - m) / eps, -50.0, 50.0)
    return float(m + eps * np.log(np.sum(np.exp(s))))

# For SLSQP
_st = {}
def _eq(x):
    return float(np.sum(x) - _st['n'] / 2.0)
def _eqj(x):
    return np.ones(_st['n'])
def _obj(x):
    return smax(x, _st['n'], _st['eps'])

def slsqp(h, n, eps, maxiter, ftol):
    _st['n'] = n
    _st['eps'] = eps
    bds = [(0.0, 1.0)] * n
    cs = [{'type': 'eq', 'fun': _eq, 'jac': _eqj}]
    r = minimize(_obj, h, method='SLSQP', bounds=bds, constraints=cs,
                 options={'maxiter': maxiter, 'ftol': ftol})
    return repair(r.x, n)

def coord_descent(h, n, max_rounds=50):
    """Coordinate descent with exact C5 evaluation."""
    h = h.copy()
    bc = compute_c5(h)
    improved = True
    rd = 0
    while improved and rd < max_rounds:
        improved = False
        rd += 1
        for i in range(n):
            best_val = h[i]
            best_c = bc
            # try a few values
            cands = [0.0, 1.0, 0.25, 0.5, 0.75, h[i]]
            for cv in cands:
                if cv == best_val:
                    continue
                h[i] = cv
                # repair sum
                d = n/2.0 - np.sum(h)
                # adjust a different coordinate
                if abs(d) > 1e-12:
                    # adjust the coordinate with most room
                    j = (i + 1) % n
                    if d > 0:
                        room = 1.0 - h[j]
                        h[j] = min(1.0, h[j] + min(d, room))
                    else:
                        room = h[j]
                        h[j] = max(0.0, h[j] - min(-d, room))
                    # if still off, spread
                    d2 = n/2.0 - np.sum(h)
                    if abs(d2) > 1e-12:
                        h += d2 / n
                        h = np.clip(h, 0.0, 1.0)
                c = compute_c5(h)
                if c < best_c:
                    best_c = c
                    best_val = cv
            h[i] = best_val
            # full repair
            h = repair(h, n)
            bc = compute_c5(h)
            if best_val != cands[0] or True:  # track improvement
                if compute_c5(h) < bc - 1e-10:
                    improved = True
                bc = compute_c5(h)
    return h, bc

    eps = _st['eps']
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(c)
    s = np.clip((c - m) / eps, -50.0, 50.0)
    return float(m + eps * np.log(np.sum(np.exp(s))))

def slsqp_opt(h, n, eps, maxiter, ftol):
    _st['n'] = n
    _st['eps'] = eps
    bds = [(0.0, 1.0)] * n
    cs = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    r = minimize(_obj_fun, h, method='SLSQP', bounds=bds,
                 constraints=cs, options={'maxiter': maxiter, 'ftol': ftol})
    return repair(r.x, n)

def coord_descent(h, n, max_rounds):
    h = h.copy()
    bc = compute_c5(h)
    for rd in range(max_rounds):
        improved = False
        for i in range(n):
            old = h[i]
            best_v = old
            best_c = bc
            candidates = [0.0, 1.0, 0.1, 0.25, 0.5, 0.75, 0.9]
            candidates.append(max(0.0, min(1.0, old + 0.15)))
            candidates.append(max(0.0, min(1.0, old - 0.15)))
            for cv in candidates:
                if abs(cv - old) < 1e-8:
                    continue
                h[i] = cv
                h = repair(h, n)
                c = compute_c5(h)
                if c < best_c - 1e-12:
                    best_c = c
                    best_v = cv
            if best_v != old:
                improved = True
                bc = best_c
            h[i] = best_v
            h = repair(h, n)
        if not improved:
            break
    return h, bc

def binary_swap_search(h, n, max_rounds):
    h_bin = (h >= 0.5).astype(float)
    ones = np.where(h_bin == 1.0)[0]
    zeros = np.where(h_bin == 0.0)[0]
    while len(ones) < n // 2:
        idx = zeros[len(ones) - n // 2] if len(ones) - n // 2 < len(zeros) else zeros[-1]
        h_bin[idx] = 1.0
        ones = np.append(ones, idx)
    h_bin = h_bin[:n]
    bc = compute_c5(h_bin)
    for _ in range(max_rounds):
        improved = False
        ones = np.where(h_bin == 1.0)[0]
        zeros = np.where(h_bin == 0.0)[0]
        for oi in ones[:n//2]:
            for zi in zeros[:n//2]:
                if oi == zi:
                    continue
                h_bin[oi] = 0.0
                h_bin[zi] = 1.0
                c = compute_c5(h_bin)
                if c < bc - 1e-12:
                    bc = c
                    improved = True
                else:
                    h_bin[oi] = 1.0
                    h_bin[zi] = 0.0
        if not improved:
            break
    return h_bin, bc

def sim_anneal_step(h, n, temp, bc):
    h_new = h + np.random.randn(n) * temp
    h_new = repair(h_new, n)
    c_new = compute_c5(h_new)
    if c_new < bc:
        return h_new, c_new, True
    delta = c_new - bc
    if delta < 0.01 and np.random.random() < np.exp(-delta / max(temp, 1e-8)):
        return h_new, c_new, False
    return h, bc, False

def make_periodic(n, p, fill_ratio):
    base = np.zeros(p)
    k = int(p * fill_ratio)
    base[:k] = 1.0
    h = np.tile(base, n // p + 1)[:n]
    return repair(h, n)

def make_sinusoid(n, k, phase, amp):
    x = np.arange(n) / n
    h = 0.5 + amp * np.sin(2.0 * np.pi * k * x + phase)
    return repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = 0.5 * np.ones(80)
    n = len(h0)
    bh = repair(h0, n)
    bc = compute_c5(bh)
    best_n = n

    # Phase 1: SLSQP multi-scale refinement from initial
    for eps in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
        if time.time() - t0 > budget_s * 0.2:
            break
        hn = slsqp_opt(bh, n, eps, 300, 1e-14)
        c5 = compute_c5(hn)
        if c5 < bc:
            bc, bh = c5, hn

    # Phase 2: Exact coordinate descent (fast, handles max exactly)
    if time.time() - t0 < budget_s * 0.45:
        bh, bc = coord_descent(bh, n, 60)

    # Phase 3: SLSQP polish after coord descent
    if time.time() - t0 < budget_s * 0.55:
        for eps in [0.0005, 0.0002, 0.0001]:
            hn = slsqp_opt(bh, n, eps, 200, 1e-14)
            c5 = compute_c5(hn)
            if c5 < bc:
                bc, bh = c5, hn

    # Phase 4: Binary projection + swap local search, then SLSQP refine
    if time.time() - t0 < budget_s * 0.7:
        hb, cb = binary_swap_search(bh, n, 30)
        # Refine binary result with SLSQP
        hn = slsqp_opt(hb, n, 0.001, 200, 1e-14)
        c5 = compute_c5(hn)
        if c5 < bc:
            bc, bh = c5, hn
        # Also try coord descent from binary
        hc, cc = coord_descent(hb, n, 40)
        if cc < bc:
            bc, bh = cc, hc
            hn = slsqp_opt(bh, n, 0.0005, 200, 1e-14)
            c5 = compute_c5(hn)
            if c5 < bc:
                bc, bh = c5, hn

    # Phase 5: Simulated annealing for exploration
    if time.time() - t0 < budget_s * 0.85:
        temp = 0.08
        for step in range(3000):
            if time.time() - t0 > budget_s - 60:
                break
            temp *= 0.9995
            hp, cp, _ = sim_anneal_step(bh, n, temp, bc)
            if cp < bc:
                bc, bh = cp, hp

    # Phase 6: Structured restarts and polish
    starts = []
    for p in [4, 5, 6, 7, 8, 10]:
        starts.append(make_periodic(n, p, 0.5))
    for k in [2, 3, 4, 5]:
        starts.append(make_sinusoid(n, k, 0.0, 0.4))
        starts.append(make_sinusoid(n, k, np.pi/3, 0.4))

    for s in starts:
        if time.time() - t0 > budget_s - 80:
            break
        hn = slsqp_opt(s, n, 0.005, 150, 1e-12)
        c5 = compute_c5(hn)
        if c5 < bc:
            bc, bh = c5, hn
            hn = slsqp_opt(bh, n, 0.001, 150, 1e-13)
            c5 = compute_c5(hn)
            if c5 < bc:
                bc, bh = c5, hn

    # Phase 7: Random restarts
    for i in range(200):
        if time.time() - t0 > budget_s - 50:
            break
        r = i % 5
        if r == 0:
            hp = bh + np.random.randn(n) * 0.03
        elif r == 1:
            hp = bh.copy()
            s = np.random.randint(0, max(1, n - 8))
            l = np.random.randint(2, min(9, n - s))
            hp[s:s+l] = 1.0 - hp[s:s+l]
        elif r == 2:
            hp = bh + np.random.randn(n) * 0.01
        elif r == 3:
            hp = bh.copy()
            idx = np.random.choice(n, size=n//4, replace=False)
            hp[idx] = 1.0 - hp[idx]
        else:
            hp = bh + np.random.randn(n) * 0.05
        hp = repair(hp, n)
        hn = slsqp_opt(hp, n, 0.003, 100, 1e-12)
        c5 = compute_c5(hn)
        if c5 < bc:
            bc, bh = c5, hn

    # Phase 8: Try different resolutions
    for n2 in [60, 100, 120]:
        if time.time() - t0 > budget_s - 80:
            break
        x_old = np.linspace(0, 2, best_n, endpoint=False) + 1.0/best_n
        x_new = np.linspace(0, 2, n2, endpoint=False) + 1.0/n2
        h_interp = np.interp(x_new, x_old, bh)
        h_interp = repair(h_interp, n2)
        hn = slsqp_opt(h_interp, n2, 0.005, 200, 1e-13)
        c5 = compute_c5(hn)
        if c5 < bc:
            bc, bh, best_n = c5, hn, n2
            hn = slsqp_opt(bh, n2, 0.001, 200, 1e-14)
            c5 = compute_c5(hn)
            if c5 < bc:
                bc, bh = c5, hn
        # Try periodic at new n
        for p in [5, 6, 7, 8, 10]:
            if n2 % p != 0:
                continue
            hp = make_periodic(n2, p, 0.5)
            hn = slsqp_opt(hp, n2, 0.005, 150, 1e-12)
            c5 = compute_c5(hn)
            if c5 < bc:
                bc, bh, best_n = c5, hn, n2

    # Phase 9: Final tight polish
    if time.time() - t0 < budget_s - 15:
        hn = slsqp_opt(bh, best_n, 0.00005, 400, 1e-15)
        c5 = compute_c5(hn)
        if c5 < bc:
            bc, bh = c5, hn
        hn, cc = coord_descent(bh, best_n, 30)
        if cc < bc:
            bc, bh = cc, hn
        hn = slsqp_opt(bh, best_n, 0.00005, 300, 1e-15)
        c5 = compute_c5(hn)
        if c5 < bc:
            bc, bh = c5, hn

    fh = repair(bh, best_n)
    fc = compute_c5(fh)
    return (fh.tolist(), float(fc), best_n)
