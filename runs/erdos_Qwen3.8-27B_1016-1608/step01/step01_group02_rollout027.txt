import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_c5_obj(h, dx, T, target_sum, lam):
    corr = np.correlate(h, 1.0 - h, mode="full")
    cmax = np.max(corr)
    lse = (cmax + np.log(np.sum(np.exp((corr - cmax) * T)) / len(corr))) / T
    return lse * dx + lam * (h.sum() - target_sum) ** 2

def eq_sum(x, target_sum):
    return x.sum() - target_sum

def projection(h, target_sum):
    h = np.array(h, float).copy()
    for _ in range(500):
        np.clip(h, 0.0, 1.0, out=h)
        excess = h.sum() - target_sum
        if abs(excess) < 1e-14:
            break
        if excess > 0:
            mask = h > 1e-12
            if not mask.any():
                break
            s = min(excess / mask.sum(), h[mask].min())
            h[mask] -= s
        else:
            mask = h < 1.0 - 1e-12
            if not mask.any():
                break
            s = min(-excess / mask.sum(), (1.0 - h[mask]).min())
            h[mask] += s
    return h

def anneal_lbfgs(h0, dx, T_list, target_sum, lam, maxiter, t_start, budget_s):

            h_new = projection(res.x, target_sum)

            if c5_new < c5_cur:
