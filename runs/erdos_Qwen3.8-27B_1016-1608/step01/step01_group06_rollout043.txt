from scipy.optimize import minimize

_GN = 0
_GAU = 30.0

    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def get_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def smax_top(c, tau, p):
    k = max(1, int(len(c) * p))
    idx = np.argpartition(c, -k)[-k:]
    m = float(c[idx].max())
    return m + float(np.log(np.sum(np.exp(tau * (c[idx] - m)))) / tau)

def project(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    tgt = n * 0.5
    for _ in range(2000):
        d = tgt - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            break
        h[mask] += d / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h

def _eq_con(h):
    return h.sum() - _GN * 0.5

def _smooth_obj(h):
    return smax_top(get_corr(h), _GAU, 0.3)

    off = ks - n + 1

        try:
            r = minimize(compute_score, best_h, method='SLSQP',
                         bounds=[(0.0, 1.0)] * nf,
                         constraints=[{'type': 'eq', 'fun': _eq_con}],
                         options={'maxiter': 500, 'ftol': 1e-14})
            hf = project(r['x'], nf)
            sf = compute_score(hf)
            if sf < best_s:
                best_s, best_h = sf, hf
        except Exception:
            pass

    best_h = project(best_h, len(best_h))
    return best_h, compute_score(best_h), len(best_h)
