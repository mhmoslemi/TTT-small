        if abs(diff) < 1e-15:

        if free.sum() == 0:

        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)

def coord_descent_step(h, i, n):
    """Find best delta for h[i] minimizing max correlation."""
    c = corr_vec(h)
    eps = 1e-8
    hp = h.copy(); hp[i] = min(1.0, h[i] + eps)
    hm = h.copy(); hm[i] = max(0.0, h[i] - eps)

        else:
            g = (c - corr_vec(hm)) / eps

        g = (corr_vec(hp) - corr_vec(hm)) / (2 * eps)

        return (c + delta * g).max()

    if can_dn_i and can_up_j:
        hm = h.copy()
        hm[i] -= eps
        hm[j] += eps
        g = (corr_vec(hp) - corr_vec(hm)) / ((hp[i] - hm[i]))
    else:
        g = (corr_vec(hp) - c) / (hp[i] - vi)
    lo = max(-vi, vj - 1.0)
    hi = min(1.0 - vi, vj)
    if hi - lo < 1e-14:
        return 0.0
    res = minimize_scalar(pair_line_obj, args=(c, g, vi, vj), bounds=(lo, hi),
                          method='bounded', options={'xatol': 1e-14, 'maxiter': 200})
    return res.x if res.success else 0.0

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    h = np.array(initial_h_values, dtype=float)
    n = len(h)
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    constraints = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_jac}]
    bounds = [(0.0, 1.0)] * n

    # Phase 1: Exact coordinate descent with random ordering
    for sweep in range(300):
        if time.time() - t_start > budget_s * 0.30:
            break
        any_improve = False
        order = rng.permutation(n)
        for i in order:
            delta = single_coord_delta(best_h, i, n)
            if abs(delta) > 1e-13:
                h_try = best_h.copy()
                h_try[i] += delta
                h_try = project_feasible(h_try, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-15:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    any_improve = True
        if not any_improve:
            break

    # Phase 2: SLSQP with soft-max, multiple temps and restarts
    temps = [0.05, 0.02, 0.01, 0.005, 0.001, 0.0005, 0.0001, 0.00005]
    for temp in temps:
        if time.time() - t_start > budget_s * 0.55:
            break
        res = minimize(soft_max_obj, best_h.copy(), args=(temp,),
                      method='SLSQP', bounds=bounds, constraints=constraints,
                      options={'maxiter': 10000, 'ftol': 1e-16, 'iprint': 0})
        h_new = project_feasible(res.x, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        for r in range(3):
            if time.time() - t_start > budget_s * 0.55:
                break
            scale = 0.03 + 0.02 * r
            h_pert = best_h + rng.normal(0, scale, n)
            h_pert = project_feasible(h_pert, n)
            res = minimize(soft_max_obj, h_pert, args=(temp,),
                          method='SLSQP', bounds=bounds, constraints=constraints,
                          options={'maxiter': 5000, 'ftol': 1e-16, 'iprint': 0})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 3: More coordinate descent
    for sweep in range(400):
        if time.time() - t_start > budget_s * 0.72:
            break
        any_improve = False
        for i in range(n):
            delta = single_coord_delta(best_h, i, n)
            if abs(delta) > 1e-14:
                h_try = best_h.copy()
                h_try[i] += delta
                h_try = project_feasible(h_try, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-15:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    any_improve = True
        if not any_improve:
            break

    # Phase 4: Pairwise swaps
    max_pair_iters = int((budget_s * 0.93 - (time.time() - t_start)) / 0.08)
    for it in range(max(0, max_pair_iters)):
        if time.time() - t_start > budget_s * 0.93:
            break
        i, j = rng.choice(n, 2, replace=False)
        delta = pair_delta(best_h, i, j, n)
        if abs(delta) > 1e-13:
            h_try = best_h.copy()
            h_try[i] += delta
            h_try[j] -= delta
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try.copy()

    # Phase 5: Final SLSQP polish
    if time.time() - t_start < budget_s * 0.98:
        for temp in [0.00001, 0.000005]:
            res = minimize(soft_max_obj, best_h.copy(), args=(temp,),
                          method='SLSQP', bounds=bounds, constraints=constraints,
                          options={'maxiter': 15000, 'ftol': 1e-18, 'iprint': 0})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            # One more coord descent pass
            for i in range(n):
                delta = single_coord_delta(best_h, i, n)
                if abs(delta) > 1e-14:
                    h_try = best_h.copy()
                    h_try[i] += delta
                    h_try = project_feasible(h_try, n)
                    c5_try = compute_c5(h_try)
                    if c5_try < best_c5 - 1e-15:
                        best_c5 = c5_try
                        best_h = h_try.copy()

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n)
