import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return c.max() * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        mask = (h < 0.9999) if diff > 0 else (h > 0.0001)
        if not mask.any():
            h += diff / n
            h = np.clip(h, 0, 1)
            continue
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_obj(h, temp):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    return m + temp * np.log(np.sum(np.exp((c - m) / temp)))

def sum_con(h):
    return np.sum(h) - len(h) / 2.0

def sum_con_jac(h):
    return np.ones(len(h))

def subgrad(h):
    c = np.correlate(h, 1.0 - h, mode='full')
    ks = int(np.argmax(c))
    n = len(h)
    sh = ks - (n - 1)
    m = np.arange(n)
    g = np.zeros(n)
    jp, jm = m + sh, m - sh
    vp = (jp >= 0) & (jp < n)
    vm_ = (jm >= 0) & (jm < n)
    g[vp] += 1.0 - h[jp[vp]]
    g[vm_] -= h[jm[vm_]]
    return g
