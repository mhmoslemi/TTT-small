                       jac=_grad_f,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        if res.x is not None:
            return _fixed(np.clip(res.x, 0.0, 1.0), n)
    except Exception:
        pass
    return None

def _proj_grad(h0, n, alpha, niter, t0, budget):
    h = _fixed(h0.copy(), n)
    f_best = _phi(h, alpha, n)
    lr = 0.005
    for i in range(niter):
        if budget - (time.time() - t0) < 5:
            break
        _, grad = _phi_grad(h, alpha, n)
        improved = False
        for _ls in range(40):
            h_new = _fixed(np.clip(h - lr * grad, 0.0, 1.0), n)
            f_new = _phi(h_new, alpha, n)
            if f_new < f_best - 1e-12:
                h = h_new
                f_best = f_new
                improved = True
                break
            lr *= 0.5
            if lr < 1e-10:
                break
        if improved:
            lr = min(lr * 1.08, 0.05)
        else:
            lr = max(lr * 1.5, 1e-4)
            if lr > 0.05:
                lr = 0.05
    return h

def _interp(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return _fixed(h.copy(), n_new)
    xo = (np.arange(n_old) + 0.5) * 2.0 / n_old
    xn = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(xn, xo, h)
    return _fixed(np.clip(h_new, 0.0, 1.0), n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        bh = np.array(initial_h_values, dtype=float).copy()
    else:
        bh = np.full(45, 0.5)
    n = len(bh)
    bh = _fixed(bh, n)
    bc = _c5(bh)

    # Phase 1: Projected gradient at current resolution
    for alpha in [50.0, 200.0, 800.0]:
        if budget_s - (time.time() - t0) < 20:
            break
        ht = _proj_grad(bh, n, alpha, 6000, t0, budget_s)
        ct = _c5(ht)
        if ct < bc:
            bh, bc = ht, ct

    # Phase 2: SLSQP polish at current resolution
    for alpha in [200.0, 1000.0, 5000.0]:
        if budget_s - (time.time() - t0) < 15:
            break
        ht = _slsqp(bh, n, alpha, 3000, t0, budget_s)
        if ht is not None:
            ct = _c5(ht)
            if ct < bc:
                bh, bc = ht, ct

    # Phase 3: Try multiple resolutions
    res_list = [200, 300, 500, 700, 1000]
    for nn in res_list:
        if budget_s - (time.time() - t0) < 50:
            break
        hi = _interp(bh, nn)
        ci = _c5(hi)
        if ci < bc:
            bh, bc = hi.copy(), ci
        for alpha in [100.0, 500.0, 2000.0]:
            if budget_s - (time.time() - t0) < 15:
                break
            ht = _proj_grad(hi, nn, alpha, 4000, t0, budget_s)
            ct = _c5(ht)
            if ct < bc:
                bh, bc = ht, ct
                hi = ht.copy()
            ht = _slsqp(hi, nn, alpha, 1500, t0, budget_s)
            if ht is not None:
                ct = _c5(ht)
                if ct < bc:
                    bh, bc = ht, ct
                    hi = ht.copy()

    # Phase 4: Multi-restart at best resolution
    nb = len(bh)
    for alpha in [500.0, 2000.0, 8000.0]:
        if budget_s - (time.time() - t0) < 10:
            break
        for r in range(8):
            if budget_s - (time.time() - t0) < 8:
                break
            if r == 0:
                h0 = bh.copy()
            else:
                scale = 0.02 + 0.02 * r
                h0 = _fixed(np.clip(bh + rng.randn(nb) * scale, 0, 1), nb)
            ht = _slsqp(h0, nb, alpha, 2000, t0, budget_s)
            if ht is not None:
                ct = _c5(ht)
                if ct < bc:
                    bh, bc = ht, ct
                    if ct < bc + 1e-6:
                        ht2 = _proj_grad(h0, nb, alpha, 2000, t0, budget_s)
                        ct2 = _c5(ht2)
                        if ct2 < bc:
                            bh, bc = ht2, ct2

    # Phase 5: Final SLSQP polish
    for alpha in [5000.0, 20000.0]:
        if budget_s - (time.time() - t0) < 8:
            break
        ht = _slsqp(bh, nb, alpha, 2000, t0, budget_s)
        if ht is not None:
            ct = _c5(ht)
            if ct < bc:
                bh, bc = ht, ct

    bh = _fixed(bh, len(bh))
    bc = _c5(bh)
    return bh, bc, len(bh)
