            if not still_going(0.92):
                break
            noise_scale = 0.08 + 0.04 * rep
            h_pert = best_h + rng.randn(n_best) * noise_scale
            h_pert = project_feasible(h_pert, n_best)
            if rep % 2 == 0:
                h_pert, s_pert = cobyLA_optimize(h_pert, n_best, maxiter=150)
            else:
                h_pert, s_pert = coordinate_descent(h_pert, n_best, max_sweeps=60)
            if s_pert < best_score:
                best_score, best_h = s_pert, h_pert.copy()

    # Phase 4: Final intensive coordinate descent
    if still_going(0.97):
        h_final = best_h.copy()
        h_final, s_final = coordinate_descent(h_final, n_best, max_sweeps=200)
        if s_final < best_score:
            best_score, best_h = s_final, h_final.copy()

    # Phase 5: Try SLSQP polish on best
    if still_going(0.99) and n_best <= 100:
        def smooth_obj(h):
            n = len(h)
            dx = 2.0 / n
            c = np.correlate(h, 1.0 - h, mode='full') * dx
            cm = np.max(c)
            return cm + np.log(np.sum(np.exp(200.0 * (c - cm)))) / 200.0

        def con_eq(h):
            return np.sum(h) - len(h) / 2.0

        try:
            res = minimize(smooth_obj, best_h, method='SLSQP',
                           bounds=[(0, 1)] * n_best,
                           constraints=[{'type': 'eq', 'fun': con_eq}],
                           options={'maxiter': 300, 'ftol': 1e-14})
            h_sl = project_feasible(res['x'], n_best)
            s_sl = compute_score(h_sl)
            if s_sl < best_score:
                best_score, best_h = s_sl, h_sl.copy()
                h_final, s_final = coordinate_descent(h_sl, n_best, max_sweeps=50)
                if s_final < best_score:
                    best_score, best_h = s_final, h_final.copy()
        except Exception:
            pass

    # Final verification
    best_h = project_feasible(best_h, n_best)
    best_score = compute_score(best_h)
    assert 0 <= np.min(best_h) and np.max(best_h) <= 1
    assert abs(np.sum(best_h) - n_best / 2.0) < 1e-8

    return best_h, best_score, n_best
