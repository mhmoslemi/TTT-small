def compute_c5_grad(h, dx, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = len(corr)  # 2n-1
    
    scaled = tau * corr - np.max(tau * corr)
    exp_scaled = np.exp(scaled)
    p = exp_scaled / np.sum(exp_scaled)  # length 2n-1
    
    # shifts: s[k] = k - (n-1) for k=0,...,2n-2
    shifts = np.arange(m) - (n - 1)  # -(n-1), ..., 0, ..., (n-1)
    
    # Build gradient: for each i, sum over k of p[k] * [(1-h[i+s_k])*I - h[i-s_k)*I]
    # Vectorize: create matrix of indices
    i_idx = np.arange(n)[:, None]  # n x 1
    s_idx = shifts[None, :]  # 1 x m
    
    # i + s: valid if 0 <= i+s < n
    is_plus = i_idx + s_idx
    mask_plus = (is_plus >= 0) & (is_plus < n)
    
    # i - s: valid if 0 <= i-s < n
    is_minus = i_idx - s_idx
    mask_minus = (is_minus >= 0) & (is_minus < n)
    
    # Term 1: sum_k p[k] * (1-h[i+s_k]) * mask_plus
    term1_mat = np.where(mask_plus, 1.0 - h[np.clip(is_plus, 0, n-1)], 0.0)
    term1 = term1_mat @ p  # n x 1
    
    # Term 2: sum_k p[k] * h[i-s_k] * mask_minus
    term2_mat = np.where(mask_minus, h[np.clip(is_minus, 0, n-1)], 0.0)
    term2 = term2_mat @ p  # n x 1
    
    g = (term1 - term2) * dx
    c5_val = float(np.max(corr) * dx)
    
    return c5_val, g

                h_t[idx] = np.clip(h_t[idx] + pert, 0, 1)
                h_t = projection(h_t, target)
                c5_t = compute_c5(h_t, dx)
                if c5_t < c5_c - 1e-14:
                    h_c, c5_c = h_t, c5_t
                    if c5_t < best_c5:
                        best_h, best_c5 = h_c.copy(), c5_t
                    break
            if it % 2000 == 1999:
                step_size = max(0.002, step_size * 0.8)

    # Phase 6: Final SLSQP polish with very tight tolerance
    if rem() > 8:
        for tau in [5.0, 2.0, 1.0]:
            if rem() < 5:
                break
            try:
                res = minimize(smooth_obj, best_h, args=(dx, tau),
                               jac=smooth_jac, method='SLSQP',
                               bounds=bounds, constraints=con,
                               options={'maxiter': 500, 'ftol': 1e-17})
                hn = projection(res.x, target)
                c5n = compute_c5(hn, dx)
                if c5n < best_c5:
                    best_h, best_c5 = hn, c5n
            except Exception:
                pass
        try:
            res = minimize(hard_obj, best_h, args=(dx,), method='SLSQP',
                           bounds=bounds, constraints=con,
                           options={'maxiter': 800, 'ftol': 1e-18})
            hn = projection(res.x, target)
            c5n = compute_c5(hn, dx)
            if c5n < best_c5:
                best_h, best_c5 = hn, c5n
        except Exception:
            pass

    best_h = projection(best_h, target)
    final_c5 = compute_c5(best_h, dx)
    return (best_h, final_c5, n)
