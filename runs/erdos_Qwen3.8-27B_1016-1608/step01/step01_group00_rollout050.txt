    n = len(h)

    return float(np.max(corr) * dx)

def compute_c5_lag(h):
    n = len(h)
    dx = 2.0 / n

    k_star = int(np.argmax(corr))

def repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:

        else:
