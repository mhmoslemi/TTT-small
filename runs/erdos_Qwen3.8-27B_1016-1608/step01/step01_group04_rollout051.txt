def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        d = target - np.sum(h)
        if abs(d) < 1e-13:
            break
        h += d / n
    return np.clip(h, 0.0, 1.0)

def _de_obj(x):
    n = _G['n']
    h = repair(x, n)
    return compute_c5(h)

def _sm_obj(x):
    n = _G['n']
    eps = _G['eps']
    h = np.clip(x, 0.0, 1.0)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    if eps < 1e-12:
        return float(m)
    s = np.clip((corr - m) / eps, -50.0, 50.0)
    return float(m + eps * np.log(np.sum(np.exp(s))))

def _eq_fun(x):
    return float(np.sum(x) - _G['n'] / 2.0)

def _eq_jac(x):
    return np.ones(_G['n'])

def interp_h(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    xs = np.linspace(0, 2, n_src, endpoint=False)
    xd = np.linspace(0, 2, n_dst, endpoint=False)
    return np.interp(xd, xs, h_src)

def slsqp_refine(h, n, eps, maxiter, ftol, t0, budget):
    if time.time() - t0 > budget - 30:
        return h
    _G['n'] = n
    _G['eps'] = eps
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    try:
        res = minimize(_sm_obj, h, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return repair(res.x, n)
    except Exception:
        return h

def local_search(h, n, t0, budget_s, stop_margin=20):
    best = h.copy()
    best_c5 = compute_c5(best)
    limit = budget_s - stop_margin
    for i in range(10**8):
        if time.time() - t0 > limit:
            break
        elapsed_frac = (time.time() - t0) / budget_s
        sigma = 0.003 + 0.04 * (1.0 - elapsed_frac)
        hp = best + np.random.randn(n) * sigma
        hp = repair(hp, n)
        c5 = compute_c5(hp)
        if c5 < best_c5:
            best_c5 = c5
            best = hp
    return best, best_c5

def perturb_archive_search(h, n, t0, budget_s, stop_margin=25):
    best = h.copy()
    best_c5 = compute_c5(best)
    limit = budget_s - stop_margin
    archive = [best.copy()]
    for i in range(10**8):
        if time.time() - t0 > limit:
            break
        r = np.random.rand()
        if r < 0.3:
            base = archive[np.random.randint(len(archive))]
            hp = base + np.random.randn(n) * 0.02
        elif r < 0.5:
            hp = best.copy()
            s = np.random.randint(0, max(1, n - 5))
            l = np.random.randint(2, min(10, n - s))
            hp[s:s+l] = 1.0 - hp[s:s+l]
        elif r < 0.7:
            k2 = np.random.randint(1, 8)
            ph = np.random.uniform(0, 2 * np.pi)
            hp = best + 0.04 * np.sin(2 * np.pi * k2 * np.arange(n) / n + ph)
        elif r < 0.85:
            idx = np.random.choice(n, size=max(1, n // 4), replace=False)
            hp = best.copy()
            hp[idx] = np.random.uniform(0, 1, size=len(idx))
        else:
            hp = best + np.random.randn(n) * 0.01
        hp = repair(hp, n)
        c5 = compute_c5(hp)
        if c5 < best_c5:
            best_c5 = c5
            best = hp.copy()
            archive.append(best.copy())
            if len(archive) > 20:
                archive.pop(0)
    return best, best_c5

        init_pop = np.zeros((npop, n))

        for i in range(15, 25):

            init_pop[i] = 0.5 + 0.4 * np.sin(2 * np.pi * kk * np.arange(n) / n +

        for i in range(npop):
            init_pop[i] = repair(init_pop[i], n)
        try:

                tol=1e-10, mutation=(0.5, 1.0),

            )
            h_de = repair(de_res.x, n)
            c5_de = compute_c5(h_de)
            if c5_de < best_c5:
                best_c5 = c5_de
                best_h = h_de
        except Exception:
            pass

    # Phase 3: SLSQP from potentially better DE result
    for eps in [0.005, 0.001, 0.0001]:
        if time.time() - t0 > budget_s - 200:

        h_new = slsqp_refine(best_h, n, eps, 500, 1e-14, t0, budget_s)

    # Phase 5: Archive-based diverse perturbation
    best_h, best_c5 = perturb_archive_search(best_h, n, t0, budget_s, stop_margin=50)

    # Phase 6: SLSQP polish
    if time.time() - t0 < budget_s - 30:
        h_new = slsqp_refine(best_h, n, 0.0001, 800, 1e-15, t0, budget_s)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new

    # Phase 7: Fine local search
    if time.time() - t0 < budget_s - 15:
        _G['n'] = n
        limit = budget_s - 8
        for i in range(10**8):
            if time.time() - t0 > limit:
                break
            hp = best_h + np.random.randn(n) * 0.004
            hp = repair(hp, n)
            c5 = compute_c5(hp)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hp

    # Final SLSQP
    if time.time() - t0 < budget_s - 10:
        h_new = slsqp_refine(best_h, n, 0.00005, 1000, 1e-15, t0, budget_s)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new

    final_h = repair(best_h, n)
    final_c5 = compute_c5(final_h)
    return (final_h.tolist(), float(final_c5), n)
