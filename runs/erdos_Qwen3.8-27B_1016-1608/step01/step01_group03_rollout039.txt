    return float(np.max(corr)) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def repair(h, target):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        d = target - np.sum(h)
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0 - 1e-12
        else:
            m = h > 1e-12
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(h, beta, target, lam):
    corr = get_corr(h)
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta
    penalty = lam * (np.sum(h) - target) ** 2
    return val + penalty

def smooth_max_grad(h, beta, target, lam):
    n = len(h)
    corr = get_corr(h)
    m = np.max(corr)
    w = np.exp(beta * (corr - m))
    w /= w.sum()
    
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-12:
            continue
        wi = w[idx]
        lag = idx - (n - 1)
        if lag >= 0:
            grad[lag:] += wi * (1.0 - h[:n-lag])
            grad[:n-lag] -= wi * h[lag:]
        else:
            l = -lag
            grad[:n-l] += wi * (1.0 - h[l:])
            grad[l:] -= wi * h[:n-l]
    
    grad += 2.0 * lam * (np.sum(h) - target)
    return grad

        for i in range(n):
            for d in deltas:
                if 0 <= h[i] + d <= 1:
                    h[i] += d
                    hn = repair(h, target)
                    cn = compute_c5(hn, dx)
                    if cn < c_cur - 1e-13:
                        h = hn.copy()
                        c_cur = cn
                        improved = True
                        break
                    h[i] -= d
        if not improved:
            break
    return h, c_cur

def run(seed=42, budget_s=1100, **kwargs):
    global _G_BETA, _G_TARGET, _G_LAM
    np.random.seed(seed)
    t0 = time.time()

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.ones(70) * 0.5

    n = len(h0)
    target = n / 2.0
    dx = 2.0 / n
    _G_TARGET = target
    _G_LAM = 100.0

    h0 = repair(h0, target)
    best_c5 = compute_c5(h0, dx)
    best_h = h0.copy()

    bnds = [(0.0, 1.0)] * n

    # Phase 1: Beta annealing from initial
    for beta in [5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 500.0, 1000.0]:
        if time.time() - t0 > budget_s - 200:
            break
        _G_BETA = beta
        try:
            r = minimize(obj_func, best_h, jac=obj_grad, method='L-BFGS-B',
                         bounds=bnds, options={'maxiter': 500, 'ftol': 1e-16,
                                               'gtol': 1e-12, 'maxfun': 5000})
            hn = repair(r.x, target)
            cn = compute_c5(hn, dx)
            if cn < best_c5:
                best_c5 = cn
                best_h = hn.copy()
        except Exception:
            pass

    # Phase 2: Multi-start with diverse perturbations
    n_restarts = 80
    for i in range(n_restarts):
        if time.time() - t0 > budget_s - 100:
            break
        strategy = i % 5
        hp = best_h.copy()
        if strategy == 0:
            sc = 0.02 + 0.1 * np.random.rand()
            hp += np.random.randn(n) * sc
        elif strategy == 1:
            c = np.random.randint(0, n)
            w = np.random.randint(3, max(4, n // 4))
            a = np.random.choice([-1, 1]) * np.random.uniform(0.05, 0.25)
            for j in range(n):
                dd = min(abs(j - c), n - abs(j - c))
                if dd < w:
                    hp[j] += a * np.cos(np.pi * dd / (2.0 * w))
        elif strategy == 2:
            f = np.random.randint(1, 4)
            a = np.random.uniform(0.03, 0.15)
            ph = np.random.uniform(0, 2 * np.pi)
            hp += a * np.sin(2 * np.pi * f * np.arange(n) / float(n) + ph)
        elif strategy == 3:
            hp = np.random.rand(n)
        else:
            hp = np.zeros(n)
            block = np.random.randint(3, 10)
            for j in range(0, n, 2 * block):
                hp[j:j + block] = 1.0
            hp = repair(hp + np.random.randn(n) * 0.05, target)

        hp = repair(hp, target)
        _G_BETA = 200.0
        try:
            r = minimize(obj_func, hp, jac=obj_grad, method='L-BFGS-B',
                         bounds=bnds, options={'maxiter': 200, 'ftol': 1e-15,
                                               'gtol': 1e-10})
            hn = repair(r.x, target)
            cn = compute_c5(hn, dx)
            if cn < best_c5:
                best_c5 = cn
                best_h = hn.copy()
        except Exception:
            pass

    # Phase 3: Re-anneal from best if improved
    if time.time() - t0 < budget_s - 60:
        for beta in [100.0, 500.0, 1000.0]:
            if time.time() - t0 > budget_s - 40:
                break
            _G_BETA = beta
            _G_LAM = 500.0
            try:
                r = minimize(obj_func, best_h, jac=obj_grad, method='L-BFGS-B',
                             bounds=bnds, options={'maxiter': 800, 'ftol': 1e-18,
                                                   'gtol': 1e-14, 'maxfun': 8000})
                hn = repair(r.x, target)
                cn = compute_c5(hn, dx)
                if cn < best_c5:
                    best_c5 = cn
                    best_h = hn.copy()
            except Exception:
                pass

    # Phase 4: Coordinate polish
    if time.time() - t0 < budget_s - 5:
        best_h, best_c5 = coord_polish(best_h, target, dx, max_rounds=30)

    best_h = repair(best_h, target)
    best_c5 = compute_c5(best_h, dx)
    return (best_h, best_c5, n)
