One more idea: **antagonistic optimization**. The problem has a natural structure where we want to minimize the max over lags. We can think of it as a game:
- Minimize over h: max_k c_k(h)
- This is equivalent to: find h that minimizes the "worst case"

A multiplicative weight approach:
1. Initialize weights w_k = 1 for all k
2. For h, find k* = argmax_k c_k
3. Update: w_{k*} *= η (increase weight on worst lag)
4. Now minimize Σ_k w_k * c_k (a smooth objective)
5. Repeat

This is like a mirror descent / multiplicative weight approach for the saddle point.

import numpy as np
from scipy.optimize import minimize
import time

_SUM_TGT = 0.0

def sum_constr(x):
    return np.sum(x) - _SUM_TGT

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def soft_max_obj(x, T):
    h = np.clip(x, 0.0, 1.0)
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    cm = corr.max()
    cs = corr - cm
    return T * np.log(np.exp(cs / T).sum()) + cm

def raw_obj(x):
    h = np.clip(x, 0.0, 1.0)
    return compute_c5(h, 2.0 / len(h))

def weighted_obj(x, weights):
    h = np.clip(x, 0.0, 1.0)
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.dot(weights, corr))

def repair(h, target):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        d = target - np.sum(h)
        if abs(d) < 1e-15:
            break
        if d > 0:
            m = h < 1.0 - 1e-12
        else:
            m = h > 1e-12
        c = int(m.sum())
        if c == 0:
            break
        h[m] += d / c
        h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    global _SUM_TGT
    np.random.seed(seed)
    t0 = time.time()
    best_c5 = 1.0
    best_h = None
    best_n = 70

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.ones(70) * 0.5

    n = len(h0)
    dx = 2.0 / n
    tgt = n / 2.0
    _SUM_TGT = tgt
    n_current = n
    dx_current = dx

    h_init = repair(h0, tgt)
    best_c5 = compute_c5(h_init, dx)
    best_h = h_init.copy()
    best_n = n

    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constr}]

    def elapsed():
        return time.time() - t0

    def remaining():
        return budget_s - elapsed()

    # Phase 1: Soft-max annealing
    corr_init = np.correlate(h_init, 1.0 - h_init, mode="full") * dx
    T_start = max(0.02, 0.5 * (corr_init.max() - corr_init.min()))
    n_T = 18
    T_levels = [T_start * (0.65 ** i) for i in range(n_T)]
    T_levels = [max(t, 0.0005) for t in T_levels]

    for T in T_levels:
        if remaining() < 150:
            break
        maxiter = 200 if T > 0.01 else 400
        try:
            r = minimize(soft_max_obj, best_h, args=(T,), method='SLSQP',
                        bounds=bnds, constraints=cons,
                        options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10})
            hn = repair(r.x, tgt)
            cn = compute_c5(hn, dx)
            if cn < best_c5:
                best_c5, best_h = cn, hn
        except Exception:
            pass

    # Phase 2: Multi-start SLSQP on raw max (more aggressive)
    n_restarts = min(100, int(remaining() / 12))
    for i in range(n_restarts):

                                bounds=bnds_a, constraints=cons_a,

                    h_a = repair(ra.x, tgt_a)
                    c_a = compute_c5(h_a, dx_a)
