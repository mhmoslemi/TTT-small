For the subgradient, I need the analytical gradient of each c(s) w.r.t. h. Let me derive it:

c(s) = sum_{i: valid} h[i] * (1-h[i+s])

For s ≥ 0, valid means 0≤i<n and 0≤i+s<n, i.e., 0≤i<n-s:
c(s) = sum_{i=0}^{n-1-s} h[i] * (1-h[i+s])

∂c(s)/∂h[j]:
- If 0≤j<n-s: h[j] appears as first factor → (1-h[j+s])
- If s≤j<n: h[j] appears as second factor (as h[i+s] where i=j-s) → -h[j-s]
- Both if s≤j<n-s

- If 0≤j<n-t: h[j] appears as second factor (h[i] where j=i) → -h[j+t]

import numpy as np
from scipy.optimize import minimize, linprog
import time

_TARGET = 35.0

def _eq_con(x):
    return x.sum() - _TARGET

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def projection(h, target):
    h = np.array(h, dtype=float, copy=True)
    for _ in range(2000):
        h = np.clip(h, 0.0, 1.0)
        excess = h.sum() - target
        if abs(excess) < 1e-15:
            return h
        if excess > 0:
            msk = h > 1e-14
            c = msk.sum()
            if c == 0:
                return h
            s = min(excess / c, h[msk].min() - 1e-15)
            if s <= 0:
                return h
            h[msk] -= s
        else:
            msk = h < 1.0 - 1e-14
            c = msk.sum()
            if c == 0:
                return h
            s = min(-excess / c, (1.0 - h[msk]).min() - 1e-15)
            if s <= 0:
                return h
            h[msk] += s
    return h

def _lp_solve_A(h_prev, n, target):
    b = 1.0 - h_prev
    ns = 2 * n - 1
    A_ub = np.zeros((ns, n + 1))
    b_ub = np.zeros(ns)
    for si in range(ns):
        s = si - (n - 1)
        lo = max(0, -s)
        hi = min(n, n - s)
        for i in range(lo, hi):
            A_ub[si, i] = b[i + s]
        A_ub[si, n] = -1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([float(target)])
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    bnds = [(0, 1)] * n + [(0, None)]
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bnds, method='highs', options={'presolve': True})
    if res.success:
        return np.clip(res.x[:n], 0, 1)
    return h_prev.copy()

def _lp_solve_B(h_prev, n, target):
    a = h_prev
    ns = 2 * n - 1
    A_ub = np.zeros((ns, n + 1))
    b_ub = np.zeros(ns)
    for si in range(ns):
        s = si - (n - 1)
        lo = max(0, -s)
        hi = min(n, n - s)
        cs = 0.0
        for i in range(lo, hi):
            j = i + s
            A_ub[si, j] = -a[i]
            cs += a[i]
        A_ub[si, n] = -1.0
        b_ub[si] = -cs
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([float(target)])
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    bnds = [(0, 1)] * n + [(0, None)]
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bnds, method='highs', options={'presolve': True})
    if res.success:
        return np.clip(res.x[:n], 0, 1)
    return h_prev.copy()

def alternating_lp(h0, n, target, max_iter=300):
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    prev_c5 = best_c5
    stagnation = 0
    for it in range(max_iter):
        h_a = _lp_solve_A(h, n, target)
        h_a = projection(h_a, target)
        c5_a = compute_c5(h_a, dx)
        if c5_a < best_c5 - 1e-15:
            best_h = h_a.copy()
            best_c5 = c5_a
        h = h_a
        h_b = _lp_solve_B(h, n, target)
        h_b = projection(h_b, target)
        c5_b = compute_c5(h_b, dx)
        if c5_b < best_c5 - 1e-15:
            best_h = h_b.copy()
            best_c5 = c5_b
        h = h_b
        if abs(prev_c5 - best_c5) < 1e-12:
            stagnation += 1
        else:
            stagnation = 0
        prev_c5 = best_c5
        if stagnation > 30:
            break
    return best_h, best_c5

def smooth_c5_obj(h, dx, temp):
    corr = np.correlate(h, 1.0 - h, mode="full")
    mx = np.max(corr)
    scaled = temp * (corr - mx)
    scaled -= np.max(scaled)
    return (mx + np.log(np.sum(np.exp(scaled)) / len(corr)) / temp) * dx

def slsqp_polish(h0, n, target, maxiter=500, use_smooth=False, temp=80.0):
    global _TARGET
    _TARGET = target
    dx = 2.0 / n
    bnds = [(0.0, 1.0)] * n
    con = ({'type': 'eq', 'fun': _eq_con},)
    obj = smooth_c5_obj if use_smooth else compute_c5
    args = (dx, temp) if use_smooth else (dx,)
    try:
        res = minimize(obj, h0, args=args, method='SLSQP',
                       bounds=bnds, constraints=con,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return projection(res.x, target)
    except Exception:
        return h0.copy()

def subgradient_step(h, n, target, alpha, n_active=5):
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode="full")
    order = np.argsort(corr_raw)[::-1]
    active = order[:n_active]
    g = np.zeros(n)
    for si in active:
        s = si - (n - 1)
        for i in range(n):
            j1 = i
            j2 = i + s
            if 0 <= j1 < n and 0 <= j2 < n:
                g[i] += (1.0 - h[j2]) - h[i] * 0  # first factor: d/dh[i] of h[i]*(1-h[i+s])
                if i + s != i:
                    pass
        # Redo properly:
        # c(s) = sum_{i valid} h[i]*(1-h[i+s])
        # d/dh[j]: if j is the "first" index i: (1-h[j+s]) for valid j
        #          if j is the "second" index i+s: -h[j-s] for valid j
        g[si - (n-1) * 0] = 0  # reset won't work, let me redo
    # Simpler: compute gradient analytically
    g = np.zeros(n)
    for si in active:
        s = si - (n - 1)
        for i in range(n):
            # h[i] as first factor: needs i+s in [0,n)
            if 0 <= i + s < n:
                g[i] += (1.0 - h[i + s])
            # h[i] as second factor: needs i-s in [0,n) and i != i-s
            if s != 0 and 0 <= i - s < n:
                g[i] += (-h[i - s])
    h_new = projection(h - alpha * g, target)
    return h_new
