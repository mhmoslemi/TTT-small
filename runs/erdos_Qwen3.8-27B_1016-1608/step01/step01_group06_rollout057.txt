                for fi in range(nf):
                    f = rng.randint(1, 8)
                    h_start += rng.uniform(-0.35, 0.35) * np.cos(2 * np.pi * f * x_grid + rng.uniform(0, 6.28))
                h_start = project_feasible(h_start, n_points)
            elif r == 5:
                h_start = np.zeros(n_points)
                nb = rng.randint(3, min(7, n_points // 6 + 1))
                for b in range(nb):
                    s = int(b * n_points / nb)
                    e = int((b + 1) * n_points / nb)
                    h_start[s:e] = rng.uniform(0.1, 0.95)
                h_start = project_feasible(h_start, n_points)
            elif r == 6:
                h_start = h0.copy()
                blen = max(3, n_points // 5)
                si = rng.randint(0, max(1, n_points - blen))
                h_start[si:si + blen] = 1.0 - h0[si:si + blen]
                h_start = project_feasible(h_start, n_points)
            elif r == 7:
                h_start = h0.copy()
                k = max(3, n_points // 10)
                idx = rng.choice(n_points, size=k, replace=False)
                h_start[idx] = 1.0 - h_start[idx]
                h_start = project_feasible(h_start, n_points)
            else:
                h_start = project_feasible(rng.uniform(0.0, 1.0, size=n_points), n_points)

            # Adaptive tau sequence: smooth to sharp
            for tau in [40.0, 120.0, 300.0, 800.0]:
                if time.time() - t0 > budget_s * 0.84:
                    break
                _global_tau = tau
                try:
                    res = minimize(obj_soft, h_start, method='SLSQP',
                                   bounds=bounds, constraints=cons,
                                   options={'maxiter': 350, 'ftol': 1e-13})
                    h_cand = project_feasible(res['x'], n_points)
                    s_cand = compute_score(h_cand)
                    if s_cand < best_score:
                        best_score = s_cand
                        best_h = h_cand.copy()
                    if s_cand < s0:
                        s0 = s_cand
                        h_start = h_cand.copy()
                    else:
                        h_start = project_feasible(h_cand + rng.randn(n_points) * 0.03, n_points)
                except Exception:
                    pass

        # Exact objective SLSQP on best at this n
        if time.time() - t0 < budget_s * 0.83:
            try:
                res = minimize(obj_exact, h0, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 500, 'ftol': 1e-14})
                h_cand = project_feasible(res['x'], n_points)
                s_cand = compute_score(h_cand)
                if s_cand < best_score:
                    best_score = s_cand
                    best_h = h_cand.copy()
                if s_cand < s0:
                    s0 = s_cand
                    h0 = h_cand.copy()
            except Exception:
                pass

        # Nelder-Mead adaptive
        if time.time() - t0 < budget_s * 0.83:
            try:
                res = minimize(obj_exact, h0, method='Nelder-Mead',
                               options={'maxiter': 4000, 'xatol': 1e-9, 'fatol': 1e-12,
                                        'adaptive': True})
                h_cand = project_feasible(res['x'], n_points)
                s_cand = compute_score(h_cand)
                if s_cand < best_score:
                    best_score = s_cand
                    best_h = h_cand.copy()
                if s_cand < s0:
                    s0 = s_cand
                    h0 = h_cand.copy()
            except Exception:
                pass

        # Coordinate descent polish at this n
        if time.time() - t0 < budget_s * 0.83:
            h_cd, s_cd = coordinate_descent(h0, n_points, max_iter=300)
            if s_cd < best_score:
                best_score = s_cd
                best_h = h_cd.copy()
            if s_cd < s0:
                s0 = s_cd
                h0 = h_cd.copy()

    # Final intensive polish on the global best
    n_final = len(best_h)
    _global_target = n_final / 2.0
    bounds = [(0.0, 1.0)] * n_final
    cons = [{'type': 'eq', 'fun': con_sum, 'jac': con_sum_jac}]

    if time.time() - t0 < budget_s * 0.93:
        # Multiple SLSQP passes with tightening
        for tau_seq in [200.0, 500.0, 2000.0]:
            if time.time() - t0 > budget_s * 0.94:
                break
            _global_tau = tau_seq
            try:
                res = minimize(obj_soft, best_h, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 800, 'ftol': 1e-15})
                h_c = project_feasible(res['x'], n_final)
                s_c = compute_score(h_c)
                if s_c < best_score:
                    best_score = s_c
                    best_h = h_c.copy()
            except Exception:
                pass

    if time.time() - t0 < budget_s * 0.95:
        try:
            res = minimize(obj_exact, best_h, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 1000, 'ftol': 1e-15})
            h_c = project_feasible(res['x'], n_final)
            s_c = compute_score(h_c)
            if s_c < best_score:
                best_score = s_c
                best_h = h_c.copy()
        except Exception:
            pass

    # Extended coordinate descent
    if time.time() - t0 < budget_s * 0.97:
        small_deltas = [0.002, 0.005, 0.01, 0.02, -0.002, -0.005, -0.01, -0.02]
        best_h, best_score = coordinate_descent(best_h, n_final, max_iter=500, deltas=small_deltas)

    # Final feasibility check
    best_h = project_feasible(best_h, len(best_h))
    best_score = compute_score(best_h)

    return best_h, best_score, len(best_h)
