        if abs(excess) < 1e-14:

            cnt = mask.sum()

    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr)
    k = idx - (n - 1)
    # gradient of f_k
    grad = np.zeros(n)
    for m in range(n):
        g = 0.0
        mk = m + k
        if 0 <= mk < n:
            g += 1.0 - h[mk]
        mk2 = m - k
        if 0 <= mk2 < n:
            g -= h[mk2]

    step = min(step, step_scale)

    # Phase 8: Structured diverse restarts
    for attempt in range(12):
        if tl() < 15:
            break
        s = attempt % 4
        if s == 0:
            period = rng.randint(7, 18)
            fill = max(2, period // 2 + rng.randint(-1, 1))
            h0 = make_periodic(n, period, fill)
        elif s == 1:
            freq = rng.randint(2, 7)
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.3, 0.6)
            h0 = make_sinusoidal(n, freq, phase, amp)
        elif s == 2:
            h0 = (rng.rand(n) > 0.5).astype(float)
        else:
            h0 = np.zeros(n)
            for blk in range(n // 4):
                st = rng.randint(0, n)
                ln = rng.randint(2, 9)
                h0[st:st + ln] = 1.0
        h0 = project_feasible(h0, target)
        h_opt = slsqp_opt(h0, dx, target, n, maxiter=250)
        if h_opt is not None:
            c5_v = compute_c5(h_opt, dx)
            if c5_v < best_c5:
                best_h, best_c5 = h_opt, c5_v

    # Phase 9: Final subgradient + coordinate + SLSQP
    if tl() > 20:
        h_sg, c5_sg = subgradient_refine(best_h, dx, target, n, iters=300, base_step=0.04)
        if c5_sg < best_c5:
            best_h, best_c5 = h_sg, c5_sg
    if tl() > 15:
        h_cd, c5_cd = cycle_coord_descent(best_h, dx, target, n, rounds=60, base_step=0.02)
        if c5_cd < best_c5:
            best_h, best_c5 = h_cd, c5_cd
    if tl() > 10:
        h_opt = slsqp_opt(best_h, dx, target, n, maxiter=600)
        if h_opt is not None:
            c5_v = compute_c5(h_opt, dx)
            if c5_v < best_c5:
                best_h, best_c5 = h_opt, c5_v

    # Final projection
    best_h = project_feasible(best_h, target)
    final_c5 = compute_c5(best_h, dx)
    return (best_h, final_c5, n)
