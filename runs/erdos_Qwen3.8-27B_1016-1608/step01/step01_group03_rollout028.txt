    """Coordinate descent with exact 1D optimization."""
    for sweep in range(n_sweeps):
        improved = False
        for p in range(n):
            # Build the 1D problem
            h_old = h[p]
            h[p] = 0.0
            corr_base = np.correlate(h, 1.0 - h, mode="full")
            h[p] = h_old
            
            # Build add arrays
            add_lin = np.zeros(2*n - 1)
            for m in range(2*n - 1):
                ell = m - (n - 1)

                    add_lin[m] = 1.0  # t - t^2: linear part is 1
                else:

                        add_lin[m] += (1.0 - h[p + ell])
                    if 0 <= p - ell < n:
                        add_lin[m] -= h[p - ell]
            
            # Evaluate at grid points
            ts = np.linspace(0.0, 1.0, 300)
            # For ell=0 (m = n-1): corr = corr_base[n-1] + t - t^2

            vals = corr_base + add_lin * ts
            vals[n-1] = corr_base[n-1] + ts - ts*ts  # quadratic term
            max_vals = vals.max(axis=0)

            if abs(t_new - h_old) > 1e-10:
                h[p] = t_new
                improved = True
        
        if not improved:

def repair(h, target):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(2000):

        if abs(d) < 1e-13:
            break
        if d > 0:
            m = h < 1.0 - 1e-12
        else:
            m = h > 1e-12

        if c == 0:
            break
        h[m] += d / c
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_sweep(h, n, n_sweeps=10, order='sequential'):
    best_c = compute_c5(h, 2.0 / n)
    for sweep in range(n_sweeps):
        if order == 'random':
            perp = list(range(n))
            np.random.shuffle(perp)
        else:
            perp = range(n)
        for p in perp:
            h[p] = 0.0
            corr_base = np.correlate(h, 1.0 - h, mode="full")
            h[p] = h[p]  # will be overwritten

            h[p] = 0.0

            h[p] = h_save

                ell = m - (n - 1)
                if ell == 0:
                    add_lin[m] = 1.0
                else:

                    if 0 <= p - ell < n:
                        add_lin[m] -= h[p - ell]

            vals[n - 1, :] = corr_base[n - 1] + ts - ts * ts
            max_vals = vals.max(axis=1)
            best_idx = int(np.argmin(max_vals))
            t_new = ts[best_idx]
