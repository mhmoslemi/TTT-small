from scipy.optimize import minimize, differential_evolution
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def projection(h, target):
    h = np.array(h, dtype=float, copy=True)
    for _ in range(1000):
        h = np.clip(h, 0.0, 1.0)
        excess = h.sum() - target
        if abs(excess) < 1e-12:
            return h
        if excess > 0:
            mask = h > 0
            if mask.sum() == 0: return h
            s = min(excess / mask.sum(), h[mask].min())
            h[mask] -= s
        else:
            mask = h < 1
            if mask.sum() == 0: return h
            s = min(-excess / mask.sum(), (1.0 - h[mask]).min())
            h[mask] += s
    return h

def smooth_c5_obj(h, dx, temp):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    s = np.sum(np.exp(temp * (corr - m)))
    return (m + np.log(s) / temp) * dx

    h_new = projection(h_new, target)

    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def projection(h, target):
    h = np.array(h, dtype=float, copy=True)
    for _ in range(5000):
        h = np.clip(h, 0.0, 1.0)
        excess = h.sum() - target
        if abs(excess) < 1e-14:
            return h
        if excess > 0:
            mask = h > 0
            cnt = mask.sum()
            if cnt == 0:
                return h
            s = min(excess / cnt, h[mask].min())
            h[mask] -= s
        else:
            mask = h < 1
            cnt = mask.sum()
            if cnt == 0:
                return h
            s = min(-excess / cnt, (1.0 - h[mask]).min())
            h[mask] += s
    return h

    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(temp * (corr - m)))) / temp) * dx
