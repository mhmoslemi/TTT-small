def subgradient_step(h, n, step_size):
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')  # length 2n-1
    c = corr * dx
    max_val = np.max(c)

    tol = 1e-6
    active = np.where(c > max_val - tol)[0]

    grad = np.zeros(n)
    for m in active:
        # m is the lag, 0 to 2n-2
        # corr[m] = sum_i h[i] * (1-h[i + (m - n + 1)])

                grad[i - lag] += (-h[i])

    corr = np.correlate(h, 1.0 - h, mode='full')
