import numpy as np
import time

def c5_value(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_obj_grad(h, dx, n, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    m = np.max(c)
    ev = np.exp((c - m) / T)
    sma = ev.sum()
    w = ev / sma
    f = m + T * np.log(sma)
    grad = np.zeros(n)
    for idx in range(len(c)):
        wi = w[idx]
        if wi < 1e-10:
            continue
        k = idx - (n - 1)
        lo1 = max(0, -k)
        hi1 = min(n, n - k)
        if hi1 > lo1:
            grad[lo1:hi1] += wi * (1.0 - h[lo1 + k:hi1 + k])
        lo2 = max(0, k)
        hi2 = min(n, n + k)
        if hi2 > lo2:
            grad[lo2:hi2] -= wi * h[lo2 - k:hi2 - k]
    return f, grad * dx

def optimize_inner(h0, dx, n, t_end, rng):
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = c5_value(h, dx)
    beta = 0.85
    alpha = 0.005
    T = 0.015
    no_imp = 0
    stall = 0
    while time.time() < t_end:
        f, grad = smooth_obj_grad(h, dx, n, T)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        v = beta * v + alpha * grad
        vn = np.linalg.norm(v)
        if vn > 0.3:
            v *= 0.3 / vn
        h_new = project_sum(h - v, n)
