    """Project v onto {0<=h<=1, sum(h)=target_sum}"""
    if v.min() >= 1.0 or v.max() <= 0.0:
        # Degenerate case
        h = np.clip(v, 0.0, 1.0)
        s = h.sum()
        if s > 0:
            h *= target_sum / s
            h = np.clip(h, 0.0, 1.0)
        return h
    lo, hi = v.min() - 2.0, v.max() + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx

def grad_at_lag(h, k):
    n = len(h)
    m = k - n + 1
    j = np.arange(n)
    grad = np.zeros(n)
    mask1 = (j + m >= 0) & (j + m < n)
    if np.any(mask1):
        grad[mask1] += (1.0 - h[j[mask1] + m])
    mask2 = (j - m >= 0) & (j - m < n)
    if np.any(mask2):
        grad[mask2] -= h[j[mask2] - m]
    return grad

def optimize_local(h, deadline, rng):

    target_sum = n / 2.0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    alpha = 0.05

    mom = np.zeros(n)

        mx = corr.max()

        active = np.where(corr >= mx * 0.98)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(corr))])
        
        grad = np.zeros(n)
        for k in active:
            grad += grad_at_lag(h, int(k))
        grad /= len(active)
        
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break

        h_new = project_feasible(h_new, target_sum)

import numpy as np
import time


def project_feasible(v, target_sum):
    """Project v onto {0<=h<=1, sum(h)=target_sum} via bisection on shift."""
    n = len(v)
    if target_sum <= 0:
        return np.zeros(n)
    if target_sum >= n:
        return np.ones(n)
    lo, hi = v.min() - 2.0, v.max() + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)


def compute_c5(h):
    """C5 = max(correlate(h, 1-h, full)) * dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx


def grad_at_lag(h, k):
    """Gradient of the k-th full-correlate value w.r.t. h."""
    n = len(h)
    m = k - n + 1
    j = np.arange(n)
    grad = np.zeros(n)
    mask1 = (j + m >= 0) & (j + m < n)
    if np.any(mask1):
        grad[mask1] += (1.0 - h[j[mask1] + m])
    mask2 = (j - m >= 0) & (j - m < n)
    if np.any(mask2):
        grad[mask2] -= h[j[mask2] - m]
    return grad


def optimize_local(h, deadline, rng):
    """Subgradient descent with momentum on the max-overlap objective."""
    n = len(h)
    target_sum = n / 2.0
    best_c5 = compute_c5(h)
    best_h = h.copy()

    no_improve = 0

        if len(active) == 0:
            active = np.array([int(np.argmax(corr))])

        grad = np.zeros(n)
        for k in active:
            grad += grad_at_lag(h, int(k))
        grad /= float(len(active))

        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad /= gn

        mom = 0.6 * mom + 0.4 * grad
        h_new = h - alpha * mom
        h_new = project_feasible(h_new, target_sum)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    return best_h, best_c5

    # Phase 1: Optimize from initial (60s)
    p1_end = t0 + min(60.0, budget_s * 0.05)
    h_opt, c5_opt = optimize_local(h0.copy(), p1_end, rng)
    if c5_opt < best_c5:
        best_h, best_c5 = h_opt.copy(), c5_opt

    # Phase 2: Iterative perturb + local opt (majority of budget)
    p2_end = t0 + min(budget_s * 0.65, 700.0)
    attempt = 0
    while time.time() < p2_end - 2.0:
        attempt += 1
        # Scale grows slowly to explore wider neighborhood
        scale = 0.04 + 0.12 * min(attempt / 30.0, 1.0)
        h_start = best_h + rng.randn(n) * scale
        h_start = project_feasible(h_start, target_sum)

        # Time slice for this local optimization
        remaining = p2_end - time.time()
        time_slice = min(25.0, max(3.0, remaining * 0.25))
        h_opt, c5_opt = optimize_local(h_start, time.time() + time_slice, rng)

        if c5_opt < best_c5:
            best_h, best_c5 = h_opt.copy(), c5_opt

        # Occasionally try a very different start
        if attempt % 20 == 0:
            h_rand = project_feasible(rng.rand(n), target_sum)
            time_slice2 = min(10.0, max(2.0, (p2_end - time.time()) * 0.15))
            h_opt2, c5_opt2 = optimize_local(h_rand, time.time() + time_slice2, rng)
            if c5_opt2 < best_c5:
                best_h, best_c5 = h_opt2.copy(), c5_opt2

    # Phase 3: Try higher resolution (if time remains)
    if time.time() < deadline - 40:
        for n_try in [100, 140, 200]:
            if time.time() > deadline - 20:
                break
            h_start = make_perturbed(best_h, n_try, rng, 0.03)
            h_opt, c5_opt = optimize_local(h_start, deadline - 5, rng)
            if c5_opt < best_c5:
                best_h, best_c5 = h_opt.copy(), c5_opt

    # Final: ensure exact feasibility and recompute
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final / 2.0)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n_final)
