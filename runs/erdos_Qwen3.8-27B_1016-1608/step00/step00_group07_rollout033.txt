import numpy as np
from scipy.optimize import minimize
import time

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5_val(h, dx):
    return float(np.max(_corr_full(h)) * dx)

def _smooth_obj(h, dx, tau, mu):
    h = np.clip(h, 0.0, 1.0)
    c = _corr_full(h) * dx
    m = c.max()
    sm = m + tau * np.log(np.sum(np.exp((c - m) / tau)))
    pen = mu * (h.sum() - len(h) / 2.0) ** 2
    return float(sm + pen)

def _project(h, n):
    target = n / 2.0
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        h = h + d / n
    return np.clip(h, 0.0, 1.0)

def _gaussian_smooth(h, sigma=1.0):
    n = len(h)
    if sigma < 0.5:
        return h.copy()
    r = int(3 * sigma)
    kernel = np.exp(-0.5 * (np.arange(-r, r + 1) / sigma) ** 2)
    kernel /= kernel.sum()
    hs = np.zeros(n)
    for i in range(n):
        lo = max(0, i - r)
        hi = min(n, i + r + 1)
        hs[i] = np.dot(h[lo:hi], kernel[r - (i - lo):r + (hi - i)])
    return hs

def run(seed=42, budget_s=1100, **kwargs):

    rng = np.random.RandomState(seed)

            best_c5 = c5_f
            best_h = hf.copy()

    return (best_h, best_c5, len(best_h))
