import time
from scipy.optimize import minimize as scipy_minimize

def compute_corr(h, dx):
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, dx):
    return np.max(compute_corr(h, dx))

def project_onto_set(h, S):
    """Project h onto {0<=x<=1, sum(x)=S}"""
    lo = h.min() - 2.0
    hi = h.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        val = np.sum(np.clip(h - mid, 0.0, 1.0))
        if val > S:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) * 0.5
    return np.clip(h - tau, 0.0, 1.0)

def lse_objective(x, dx, T):
    h = x
    corr = compute_corr(h, dx)
    max_c = np.max(corr)
    shifted = (corr - max_c) / T
    log_sum = np.log(np.sum(np.exp(shifted))) + max_c / T
    return T * log_sum

def lse_gradient(x, dx, T):
    n = len(x)
    corr = compute_corr(x, dx)
    max_c = np.max(corr)
    shifted = (corr - max_c) / T
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    
    # Gradient of weighted sum of f_k
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        if weights[k_idx] < 1e-12:
            continue
        lag = k_idx - (n - 1)
        if lag == 0:
            grad += weights[k_idx] * dx * (1.0 - 2.0 * x)
        elif lag > 0:
            l = lag
            grad[:n-l] += weights[k_idx] * dx * (1.0 - x[l:])
            grad[l:] += -weights[k_idx] * dx * x[:n-l]
        else:
            l = -lag
            grad[l:] += weights[k_idx] * dx * (1.0 - x[:n-l])
            grad[:n-l] += -weights[k_idx] * dx * x[l:]
    return grad

    best_c5 = compute_c5(h, dx)

        max_val = np.max(corr)
        # Active set: top lags

                grad[l:] += -dx * h[:n-l]
