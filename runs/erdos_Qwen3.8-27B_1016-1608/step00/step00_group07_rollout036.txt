Actually, a better approach for the "max of quadratics" is to use a soft-max approximation:
f(h) ≈ (1/τ) * log(sum_k exp(τ * g_k(h)))

This is smooth and its gradient is:
∇f = (1/τ) * sum_k exp(τ*g_k) * ∇g_k / sum_k exp(τ*g_k)
= weighted average of ∇g_k with weights proportional to exp(τ*g_k)

For large τ, this concentrates on the max. For moderate τ, it smooths out.

Let me implement this:

```python
def soft_max_obj(h, n, tau=50):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(corr)
    shifted = corr - max_c  # for numerical stability
    weights = np.exp(tau * shifted)
    f = max_c + np.log(np.sum(weights)) / tau
    return f

def soft_max_grad(h, n, tau=50):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(corr)
    shifted = corr - max_c
    weights = np.exp(tau * shifted)
    weights /= np.sum(weights)
    
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        if weights[k_idx] < 1e-10:
            continue
        lag = k_idx - (n - 1)
        for i in range(n):
            j = i + lag
            if 0 <= j < n:
                grad[i] += weights[k_idx] * (1.0 - h[j]) * dx
                grad[j] += weights[k_idx] * (-h[i]) * dx
    return grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_sum(h, n):
    """Project h onto {h: 0<=h<=1, sum(h)=n/2} via bisection on threshold."""
    target = 0.5 * n
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo+hi), 0.0, 1.0)

def grad_single(h, n, topk=1):
    """Gradient of max of top-k correlation values."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Get top-k indices
    if topk >= len(corr):
        idxs = np.argsort(corr)[::-1]
    else:
        idxs = np.argsort(corr)[::-1][:topk]
    
    grad = np.zeros(n)
    for k_idx in idxs:
        lag = k_idx - (n - 1)
        valid_i = range(max(0, -lag), min(n, n - lag))
        for i in valid_i:
            j = i + lag
            grad[i] += (1.0 - h[j])
            grad[j] -= h[i]
    grad *= dx / topk
    return grad, corr.max() * dx

def optimize_h(h, n, t0, budget, max_iter=3000, lr_init=0.05):
    """Projected gradient with momentum."""
    best = h.copy()
    best_c5 = compute_c5(h)
    momentum = np.zeros(n)
    beta = 0.9
    lr = lr_init
    
    for it in range(max_iter):
        if time.time() - t0 > budget:
            break
        grad, c5 = grad_single(h, n, topk=1)
        momentum = beta * momentum + (1.0 - beta) * grad
        h = h - lr * momentum
        h = project_sum(h, n)
        
        c5_new = compute_c5(h)
        if c5_new < best_c5:
            best_c5 = c5_new
            best = h.copy()
        
        if it % 300 == 299:
            lr *= 0.85
            beta = min(0.95, beta + 0.005)
    
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    # Initial solution
    if 'initial_h_values' in dir() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.full(80, 0.5)
    
    n = len(h0)
    h0 = project_sum(h0, n)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    
    # Stage 1: Optimize at original resolution
    time_limit = t0 + budget_s * 0.5
    best_h, best_c5 = optimize_h(h0, n, t0, time_limit, max_iter=3000)
    
    # Stage 2: Restart from perturbed best
    for r in range(3):
        if time.time() - t0 > budget_s * 0.7:
            break
        h_pert = best_h + rng.randn(n) * 0.03
        h_pert = project_sum(h_pert, n)
        time_limit_r = t0 + budget_s * (0.7 + 0.05 * (r + 1))
        h_r, c5_r = optimize_h(h_pert, n, t0, min(time_limit_r, t0 + budget_s * 0.75), max_iter=1000)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r
    
    # Stage 3: SLSQP polish
    if time.time() - t0 < budget_s * 0.85:
        try:
            def obj(h):
                return compute_c5(h)
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
                     'jac': lambda x: np.ones(n)}]
            bounds = [(0.0, 1.0)] * n
            res = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 100, 'ftol': 1e-12})
            if res.fun < best_c5:
                h_s = project_sum(res.x, n)
                c5_s = compute_c5(h_s)
                if c5_s < best_c5:
                    best_c5 = c5_s
                    best_h = h_s
        except:
            pass
    
    # Stage 4: Try higher resolution
    for n_try in [100, 120, 160]:
        if time.time() - t0 > budget_s * 0.93:
            break
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        h_try = np.interp(x_new, x_old, best_h)
        h_try = project_sum(h_try, n_try)
        
        time_limit_t = t0 + budget_s * (0.93 + 0.01 * n_try / 160)
        h_try, c5_try = optimize_h(h_try, n_try, t0, time_limit_t, max_iter=1500, lr_init=0.03)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try
    
    # Final check
    best_h = project_sum(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, len(best_h)
```
