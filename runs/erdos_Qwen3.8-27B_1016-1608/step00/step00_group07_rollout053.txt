            continue

        # contribution: wk * dx * [(1-h[i+s])*[0<=i+s<n] - h[i-s]*[0<=i-s<n]]
        # First term: (1-h[i+s]) for i where 0<=i+s<n
        #   i range: max(0,-s) to min(n-1, n-1-s)
        i_start = max(0, -s)

    return grad

import numpy as np
import time

def _c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def _corr(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode="full") * dx

def _fix(h, tgt):
    h = h.copy()
    h += (tgt - h.sum()) / len(h)
    np.clip(h, 0.0, 1.0, out=h)
    d = tgt - h.sum()
    if abs(d) > 1e-10:
        m = (h > 0.01) & (h < 0.99)
        if m.any():
            h[m] += d / m.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h

def _grad(h, T=50.0):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    mw = c.max()
    w = np.exp(T * (c - mw))
    s_w = w.sum()
    if s_w < 1e-30:
        return np.zeros(n)
    w /= s_w
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-7:
            continue
        s = k - (n - 1)
        a1 = max(0, -s)
        b1 = min(n, n - s)
        if a1 < b1:
            grad[a1:b1] += wk * dx * (1.0 - h[a1 + s:b1 + s])
        a2 = max(0, s)
        b2 = min(n, n + s)
        if a2 < b2:
            grad[a2:b2] -= wk * dx * h[a2 - s:b2 - s]
    return grad

def _grad_desc(h, steps=100, lr0=0.01, T0=30.0, T1=100.0):
    n = len(h)
    tg = n // 2
    h = h.copy()
    best, best_c = h.copy(), _c5(h)
    for s in range(steps):
        fr = s / max(1, steps - 1)
        T = T0 + (T1 - T0) * fr
        lr = lr0 * (1.0 - 0.6 * fr)
        g = _grad(h, T)
        h -= lr * g
        np.clip(h, 0.0, 1.0, out=h)
        h = _fix(h, tg)
        cv = _c5(h)
        if cv < best_c:
            best_c, best = cv, h.copy()
    return best, best_c

def _local_search(h, iters, seed, st0=0.05):
    r = np.random.RandomState(seed)
    n, tg = len(h), len(h) // 2
    bg, bc = h.copy(), _c5(h)
    st = st0
    no_imp = 0
    for it in range(iters):
        k = r.randint(1, 5)
        ix = r.choice(n, k, replace=False)
        tr = bg.copy()
        tr[ix] += r.uniform(-st, st, k)
        np.clip(tr, 0.0, 1.0, out=tr)
        tr = _fix(tr, tg)
        v = _c5(tr)
        if v < bc - 1e-12:
            bg, bc = tr.copy(), v
            no_imp = 0
            st *= 1.01
            if st > st0 * 1.5:
                st = st0 * 1.5
        else:
            no_imp += 1
            if no_imp > 300:
                st *= 0.9
                no_imp = 0
    return bg, bc

def _upscale(h, nn):
    on = len(h)
    xo = np.arange(on) / on
    xn = np.arange(nn) / nn
    hv = np.interp(xn, xo, h)
    hv = np.clip(hv, 0.0, 1.0)
    return _fix(hv, nn // 2)

def _make_binary(n, seed):
    r = np.random.RandomState(seed)
    nb = n // 2
    idx = r.choice(n, nb, replace=False)
    h = np.zeros(n)
    h[idx] = 1.0
    return h

    t0 = time.time()
    def elapsed():
        return time.time() - t0

    best_h, best_c, best_n = None, 1.0, 0

    if 'initial_h_values' in globals():

    # Phase 3: Try binary starting point
