def compute_c5_array(h):

import numpy as np
import time

def compute_c5_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h.astype(np.float64), 1.0 - h.astype(np.float64), mode='full') * dx

def get_c5(h):
    return compute_c5_array(h).max()

def to_binary(h, target_sum):
    n = len(h)
    k = int(target_sum)
    # Sort descending, take top k
    indices = np.argsort(-h[:])
    b = np.zeros(n, dtype=np.float64)
    b[indices[:k]] = 1.0
    return b

def exhaustive_ls(b, max_rounds=1000):
    n = len(b)
    b = b.astype(np.float64).copy()
    best = get_c5(b)
    for _ in range(max_rounds):
        improved = False
        ones = np.where(b == 1.0)[0]
        zeros = np.where(b == 0.0)[0]
        for i in ones:
            for j in zeros:
                b[i] = 0.0
                b[j] = 1.0
                c5 = get_c5(b)
                if c5 < best - 1e-15:
                    best = c5
                    improved = True
                    # After improvement, need to re-identify ones and zeros
                    # Actually, we already updated b, so the next iteration's one/j changes
                    # But we're in the middle of the loop. Let's just continue.
                else:
                    b[i] = 1.0
                    b[j] = 0.0
        if not improved:
            break
    return b, best
