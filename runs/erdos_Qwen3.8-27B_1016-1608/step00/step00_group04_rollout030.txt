import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-8:
        room = (1.0 - h) if diff > 0 else h.copy()
        tot = room.sum()
        if tot > 1e-10:
            h = h + diff * room / tot
    return np.clip(h, 0.0, 1.0)

def gradient_c5_step(h, n, dx, lr):
    corr = np.correlate(h, 1.0 - h, mode="full")
    scores = corr * dx
    k_best = int(np.argmax(scores))
    shift = k_best - (n - 1)
    grad = np.zeros(n)
    for m in range(n):
        g = 0.0
        i1 = m - shift
        if 0 <= i1 < n:
            g += (1.0 - h[i1])
        i2 = m + shift
        if 0 <= i2 < n:
            g -= h[i2]
        grad[m] = dx * g
    h_new = h - lr * grad
    h_new = h_new + (n / 2.0 - h_new.sum()) / n
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = repair(h_new, n)
    return h_new

def binary_local_search(h, n, dx, time_limit, rng):
    target = int(n / 2)
    h = (h >= 0.5).astype(float)
    while h.sum() > target:
        idx = np.where(h == 1)[0]
        if len(idx) == 0:
            break
        h[idx[rng.integers(len(idx))]] = 0
    while h.sum() < target:
        idx = np.where(h == 0)[0]
        if len(idx) == 0:
            break
        h[idx[rng.integers(len(idx))]] = 1
    best_score = compute_c5(h, dx)
    best_h = h.copy()
    start = time.time()
    stalls = 0
    while time.time() - start < time_limit:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_s = best_score
        best_pair = None
        for i in ones:
            for j in zeros:
                h[i], h[j] = 0.0, 1.0
                s = compute_c5(h, dx)
                h[i], h[j] = 1.0, 0.0
                if s < best_s - 1e-12:
                    best_s = s
                    best_pair = (i, j)
        if best_pair is None:
            stalls += 1
            if stalls > 5:
                for _ in range(int(rng.integers(2, 6))):
                    ii = int(rng.integers(len(ones)))
                    jj = int(rng.integers(len(zeros)))
                    h[ones[ii]], h[zeros[jj]] = 0.0, 1.0
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                stalls = 0
        else:
            h[best_pair[0]], h[best_pair[1]] = 0.0, 1.0
            best_score = best_s
            best_h = h.copy()
            stalls = 0
    return best_h, best_score

def continuous_search(h, n, dx, time_limit, rng):
    best_score = compute_c5(h, dx)
    best_h = h.copy()
    start = time.time()
    step = 0.08
    lr = 0.05
    stalls = 0

        improved = False
        # Random perturbations
        for _ in range(3000):

            j = int(rng.integers(n))
            if i == j:

            if best_h[i] + delta < -1e-10 or best_h[i] + delta > 1.0 + 1e-10:
                continue
            if best_h[j] - delta < -1e-10 or best_h[j] - delta > 1.0 + 1e-10:

            best_h[i] += delta
            best_h[j] -= delta
            s = compute_c5(best_h, dx)
            if s < best_score - 1e-12:
                best_score = s
                improved = True
            else:
                best_h[i] -= delta
                best_h[j] += delta
        # Gradient step
        h_grad = gradient_c5_step(best_h, n, dx, lr)

        if s_grad < best_score - 1e-12:

        if improved:
            stalls = 0
        else:
            stalls += 1
            step *= 0.85
            lr *= 0.9
            if step < 1e-7 or stalls > 60:

    return best_h, best_score

def upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    def remaining():
        return budget_s - elapsed()

    # Get initial values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = np.zeros(42)
        h_init[:21] = 1.0
        h_init = h_init[rng.permutation(42)]

    n = len(h_init)
    dx = 2.0 / n
    h = repair(h_init, n)

    best_h = h.copy()
    best_score = compute_c5(h, dx)

    # Phase 1: Binary local search
    if remaining() > 80:
        bt = min(180, remaining() * 0.25)
        bh, bs = binary_local_search(h.copy(), n, dx, bt, rng)
        if bs < best_score:
            best_score = bs
            best_h = bh.copy()

    # Phase 2: Continuous refinement
    if remaining() > 50:
        ct = min(150, remaining() * 0.25)
        ch, cs = continuous_search(best_h.copy(), n, dx, ct, rng)
        if cs < best_score:
            best_score = cs
            best_h = ch.copy()

    # Phase 3: Multi-start
    for attempt in range(40):
        if remaining() < 15:
            break
        h_try = best_h.copy()
        n_kicks = int(rng.integers(2, 8))
        for _ in range(n_kicks):
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j:
                h_try[i], h_try[j] = h_try[j], h_try[i]
        h_try = repair(h_try, n)
        ct = min(60, remaining() * 0.15)
        if ct < 5:
            break
        ch, cs = continuous_search(h_try, n, dx, ct, rng)
        if cs < best_score:
            best_score = cs
            best_h = ch.copy()

    # Phase 4: Higher resolution exploration
    for n2 in [84, 126, 210, 420]:
        if remaining() < 35:
            break
        h_up = upsample(best_h, n2)
        h_up = repair(h_up, n2)
        dx2 = 2.0 / n2
        ct = min(150, remaining() * 0.3)
        if ct < 10:
            break
        ch, cs = continuous_search(h_up, n2, dx2, ct, rng)
        if cs < best_score:
            best_score = cs
            best_h = ch.copy()
            n = n2
            dx = dx2

            best_h = ch.copy()

    final_n = len(best_h)
    final_score = compute_c5(best_h, 2.0 / final_n)
    return best_h, final_score, final_n
