                    h_trial = h_best.copy()
                    h_trial[j] += delta
                    h_trial[k] -= delta
                    if 0.0 <= h_trial[j] <= 1.0 and 0.0 <= h_trial[k] <= 1.0:
                        c5_trial = compute_c5(h_trial)
                        if c5_trial < best_c5 - 1e-12:
                            h_best = h_trial
                            best_c5 = c5_trial
                            improved = True
        if not improved and rnd > 40:
            break
    return h_best, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        h0 = None

    if h0 is not None and len(h0) > 0:
        n_points = len(h0)
    else:
        n_points = 70
        h0 = np.full(n_points, 0.5)

    target_sum = n_points / 2.0
    h0 = repair(h0.copy(), target_sum)
    bounds = [(0.0, 1.0)] * n_points

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: L-BFGS-B from initial
    h_opt, c5_opt = lbfgsb_optimize(h0.copy(), target_sum, bounds)
    if c5_opt < best_c5:
        best_c5 = c5_opt
        best_h = h_opt

    # Phase 2: Perturbed restarts
    for scale in [0.02, 0.05, 0.1, 0.15, 0.2, 0.03, 0.07, 0.12, 0.04, 0.08,
                  0.015, 0.06, 0.11, 0.18, 0.25]:
        if time.time() - t0 > budget_s - 60:
            break
        for trial in range(3):
            if time.time() - t0 > budget_s - 30:
                break
            h_pert = best_h + rng.randn(n_points) * scale
            h_pert = repair(h_pert, target_sum)
            h_opt, c5_opt = lbfgsb_optimize(h_pert, target_sum, bounds, maxiter=100)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

    # Phase 3: Structured random starts
    if time.time() - t0 < budget_s - 90:
        for f in [1, 2, 3, 4, 5, 7, 11]:
            if time.time() - t0 > budget_s - 30:
                break
            x = np.arange(n_points, dtype=float)
            phase = rng.rand() * 2.0 * np.pi
            h_struct = 0.5 + 0.35 * np.sin(2.0 * np.pi * f * x / n_points + phase)
            h_struct += 0.15 * np.sin(2.0 * np.pi * (f + 3) * x / n_points + phase * 1.7)
            h_struct = repair(h_struct, target_sum)
            h_opt, c5_opt = lbfgsb_optimize(h_struct, target_sum, bounds, maxiter=150)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

    # Phase 4: Coordinate descent polish
    best_h, best_c5 = coord_descent(best_h, best_c5, target_sum, n_points, rng, t0, budget_s)

    # Phase 5: Final L-BFGS-B from polished solution
    if time.time() - t0 < budget_s - 20:
        h_opt, c5_opt = lbfgsb_optimize(best_h.copy(), target_sum, bounds, maxiter=300)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 6: Another round of coordinate descent
    best_h, best_c5 = coord_descent(best_h, best_c5, target_sum, n_points, rng, t0, budget_s, max_rounds=100)

    # Final repair and verification
    best_h = repair(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n_points)
