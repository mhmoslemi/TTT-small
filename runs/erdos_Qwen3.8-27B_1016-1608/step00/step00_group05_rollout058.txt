    best_h = project_feasible(h0, n)
    best_c5 = c5_eval(best_h)

    def elapsed():
        return time.time() - t0

    def timeup(frac):
        return elapsed() > budget_s * frac

    # Phase 1: Nelder-Mead from initial (40% budget)
    h_start = best_h + rng.normal(0, 0.02, n)
    h_start = project_feasible(h_start, n)
    res = optimize.minimize(lambda x: obj_wrapped(x, n), h_start,
                            method='Nelder-Mead',
                            options={'maxiter': 15000, 'xatol': 1e-10,
                                     'fatol': 1e-13, 'adaptive': True})
    h_new = project_feasible(res.x, n)
    c5_new = c5_eval(h_new)
    if c5_new < best_c5:
        best_h, best_c5 = h_new, c5_new

    # Phase 2: Smooth-max gradient optimization (20% budget)
    if not timeup(0.6):
        t2_end = t0 + budget_s * 0.6
        x0 = best_h[:n - 1].copy()
        beta = 30.0
        bounds = [(0.0, 1.0)] * (n - 1)
        while time.time() < t2_end:
            beta = min(beta * 1.8, 200.0)

            def sg(x):
                return smooth_obj_grad(x, n, beta)

            res2 = optimize.minimize(sg, x0, method='L-BFGS-B',
                                     jac=True, bounds=bounds,
                                     options={'maxiter': 1000, 'gtol': 1e-10})
            h_cand = np.zeros(n)
            h_cand[:n - 1] = res2.x
            h_cand[n - 1] = n / 2.0 - np.sum(res2.x)
            h_cand = project_feasible(h_cand, n)
            c5_cand = c5_eval(h_cand)
            if c5_cand < best_c5:
                best_h, best_c5 = h_cand, c5_cand
                x0 = best_h[:n - 1].copy()
            else:
                x0 = res2.x.copy()

    # Phase 3: Coordinate descent polish (20% budget)
    if not timeup(0.8):
        t3_end = t0 + budget_s * 0.8
        steps = [0.1, 0.05, 0.02, 0.01, 0.005, -0.1, -0.05, -0.02, -0.01, -0.005]
        while time.time() < t3_end:
            improved = False
            for i in range(n):
                if time.time() > t3_end:
                    break
                for d in steps:
                    h_try = best_h.copy()
                    h_try[i] = best_h[i] + d
                    h_try = project_feasible(h_try, n)
                    c5_try = c5_eval(h_try)
                    if c5_try < best_c5 - 1e-14:
                        best_h, best_c5 = h_try, c5_try
                        improved = True
                        break
            if not improved:
                small = [0.002, 0.001, -0.002, -0.001]
                any_imp = False
                for i in range(n):
                    if time.time() > t3_end:
                        break
                    for d in small:
                        h_try = best_h.copy()
                        h_try[i] = best_h[i] + d
                        h_try = project_feasible(h_try, n)
                        c5_try = c5_eval(h_try)
                        if c5_try < best_c5 - 1e-14:
                            best_h, best_c5 = h_try, c5_try
                            any_imp = True
                            break
                if not any_imp:
                    break

    # Phase 4: Random restarts + Nelder-Mead (20% budget)
    t4_end = t0 + budget_s - 5
    for restart in range(50):
        if time.time() > t4_end:
            break
        scale = 0.01 + 0.04 * (restart / 50.0)
        h_r = best_h + rng.normal(0, scale, n)
        h_r = project_feasible(h_r, n)
        res4 = optimize.minimize(lambda x: obj_wrapped(x, n), h_r,
                                 method='Nelder-Mead',
                                 options={'maxiter': 5000, 'xatol': 1e-11,
                                          'fatol': 1e-13, 'adaptive': True})
        h_c = project_feasible(res4.x, n)
        c5_c = c5_eval(h_c)
        if c5_c < best_c5:
            best_h, best_c5 = h_c, c5_c

    # Final projection and evaluation
    best_h = project_feasible(best_h, n)
    final_c5 = c5_eval(best_h)
    return best_h, final_c5, n
