    return np.max(corr) * dx

def repair(h, n):
    """Repair h to satisfy 0<=h<=1 and sum(h)=n/2"""
    target = n / 2.0
    h = np.clip(h, 0, 1)
    for _ in range(20):
        s = np.sum(h)
        deficit = target - s
        if abs(deficit) < 1e-12:
            break
        if deficit > 0:
            # Need to add value
            mask = h < 1.0
            if not np.any(mask):
                break
            n_mask = np.sum(mask)
            add = deficit / n_mask
            h[mask] = np.min(h[mask] + add, 1.0)
        else:
            # Need to remove value
            mask = h > 0.0
            if not np.any(mask):
                break
            n_mask = np.sum(mask)
            rem = -deficit / n_mask
            h[mask] = np.max(h[mask] - rem, 0.0)
    # Final exact fix
    s = np.sum(h)
    deficit = target - s
    if abs(deficit) > 1e-15:
        # Spread over all non-bound elements
        mask = (h > 1e-15) & (h < 1 - 1e-15)
        if np.any(mask):
            h[mask] += deficit / np.sum(mask)
        else:
            # Fallback: adjust one element
            idx = np.argmax(h) if deficit < 0 else np.argmin(h)
            h[idx] += deficit
    return h
