import numpy as np
import time

def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _grad_smooth(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    mx = np.max(corr)
    e = np.exp((corr - mx) / T)
    w = e / e.sum()
    g = np.zeros(n)
    for j in range(2 * n - 1):
        wj = w[j]
        if wj < 1e-14:
            continue
        m = j - (n - 1)
        # (1-h[i-m]) for 0<=i-m<n
        lo = max(0, m)
        hi = min(n, m + n)
        if lo < hi:
            g[lo:hi] += wj * (1.0 - h[np.arange(lo, hi) - m])
        # -h[i+m] for 0<=i+m<n
        lo2 = max(0, -m)
        hi2 = min(n, n - m)
        if lo2 < hi2:
            g[lo2:hi2] -= wj * h[np.arange(lo2, hi2) + m]
