    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full")

def corr_gradient(h, n, alpha):
    """Return smooth objective value and gradient."""
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    m = np.max(corr)
    exps = np.exp(alpha * (corr - m))
    w = exps / exps.sum()
    obj_val = dx * (m + np.log(exps.sum()) / alpha)
    
    grad = np.zeros(n)
    for k in range(2*n-1):
        s = k - (n-1)
        # First term: (1-h[i+s]) for valid i
        i_start = max(0, -s)
        i_end = min(n, n - s)
        if i_start < i_end:
            grad[i_start:i_end] += w[k] * (1 - h[i_start+s:i_end+s])
        # Second term: -h[i-s] for valid i
        i_start2 = max(0, s)
        i_end2 = min(n, n + s)
        if i_start2 < i_end2:
            grad[i_start2:i_end2] -= w[k] * h[i_start2-s:i_end2-s]
    
    grad *= dx
    return obj_val, grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def smooth_c5(h, dx, alpha):
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    e = np.exp(alpha * (corr - m))
    return dx * (m + np.log(e.sum()) / alpha)

def project_feasible(x, n):
    x = np.clip(x.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - x.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - x
            tot = room.sum()
            if tot < 1e-14:
                break
            x += diff * room / tot
        else:
            space = x.copy()
            tot = space.sum()
            if tot < 1e-14:
                break
            x += diff * space / tot
        x = np.clip(x, 0.0, 1.0)
    return x

def local_search(h, dx, n, rng, n_iter=30000):
    best = h.copy()
    best_val = compute_c5(best, dx)
    steps = [0.01, 0.03, 0.05, 0.1, 0.2, 0.4]
    for it in range(n_iter):
        i, j = rng.choice(n, 2, replace=False)
        for s in steps:
            for sign in (1, -1):
                ni, nj = best[i] + sign * s, best[j] - sign * s
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                best[i], best[j] = ni, nj
                v = compute_c5(best, dx)
                if v < best_val - 1e-14:
                    best_val = v
                else:
                    best[i], best[j] = ni - sign * s, nj + sign * s
    return best, best_val

    cons = {'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}
    bnds = [(0.0, 1.0)] * n
    x_opt = x0.copy()
    for alpha in [10, 30, 100, 300, 1000]:
        if time.time() - t0 > budget_s * budget_frac:
            break
        def obj(a=alpha):
            def f(x):
                return smooth_c5(x, dx, a)
            return f
        o = obj()
        res = minimize(o, x_opt, method='SLSQP', bounds=bnds,
                       constraints=[cons], options={'maxiter': 400, 'ftol': 1e-12})
        if res.x is not None:
            x_opt = project_feasible(res.x, n)
            v = compute_c5(x_opt, dx)
            if v < best_c5:
                best_c5 = v
                best_h = x_opt.copy()
        # Random restart
        if time.time() - t0 < budget_s * (budget_frac - 0.1):
            x_r = project_feasible(np.clip(rng.random(n) * 1.3, 0, 1), n)
            res2 = minimize(o, x_r, method='SLSQP', bounds=bnds,
                            constraints=[cons], options={'maxiter': 300, 'ftol': 1e-12})
            if res2.x is not None:
                x_r = project_feasible(res2.x, n)
                v = compute_c5(x_r, dx)
                if v < best_c5:
                    best_c5 = v
                    best_h = x_r.copy()
                    x_opt = x_r.copy()
    return best_h, best_c5, x_opt
