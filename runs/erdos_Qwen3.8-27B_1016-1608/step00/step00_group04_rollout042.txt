            if n_try % 2 != 0:
                h_r[rng.integers(n_try)] = 0.5
            h_o, c5_o = fs_swap(h_r, n_try, rng, max_rounds=3000, time_limit=t_round)
            if c5_o < best_c5:
                best_h, best_c5, best_n = h_o, c5_o, n_try

    # Phase 4: Continuous refinement on best
    phase4_end = t0 + budget_s - 15
    if best_h is not None and elapsed() < phase4_end - 5:
        h_c = best_h.astype(float).copy()
        h_c, c5_c = continuous_refine(h_c, best_n, rng, max_iter=30000,
                                       time_limit=min(60, phase4_end - elapsed() - 5))
        if c5_c < best_c5:
            best_h, best_c5 = h_c, c5_c

    # Phase 5: Final squeeze - more restarts if time permits
    if elapsed() < budget_s - 10 and best_h is not None:
        n_final = best_n
        remaining = budget_s - elapsed() - 10
        n_more = int(remaining / 3)
        for _ in range(min(n_more, 200)):
            if elapsed() > budget_s - 10:
                break
            h_r = best_h.copy()
            nk = rng.integers(2, 6)
            for _ in range(nk):
                ones = np.where(h_r == np.round(h_r) + 0.5)[0]
                ones = np.where(h_r >= 0.5)[0] if len(ones) == 0 else ones
                zeros = np.where(h_r < 0.5)[0]
                if len(ones) == 0 or len(zeros) == 0:
                    break
                ii = ones[rng.integers(len(ones))]
                jj = zeros[rng.integers(len(zeros))]
                h_r[ii], h_r[jj] = h_r[jj], h_r[ii]
            h_r = make_binary(h_r, n_final)
            h_o, c5_o = fs_swap(h_r, n_final, rng, max_rounds=2000, time_limit=2.0)
            if c5_o < best_c5:
                best_h, best_c5, best_n = h_o, c5_o, n_final

    if best_h is None:
        best_h = np.zeros(n_init)
        best_h[::2] = 1.0
        best_c5 = compute_c5(best_h)
        best_n = n_init

    # Final validation
    best_h = np.clip(np.array(best_h, dtype=float), 0, 1)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, best_n
