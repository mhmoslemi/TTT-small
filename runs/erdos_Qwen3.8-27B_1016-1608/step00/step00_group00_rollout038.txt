def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    """Project h to [0,1]^n with sum = target."""
    h = np.clip(h, 0, 1)
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0, 1)

    t0 = time.time()
    lam = 30.0
    lr = 0.008
    dx = 2.0 / n
    for _ in range(150000):
        if time.time() - t0 > tlim:
            break
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        exps = np.exp(lam * (corr - corr.max()))
        w = exps / exps.sum()
        grad = np.zeros(n)
        for m in range(len(corr)):
            if w[m] < 1e-10:
                continue
            s = m - (n - 1)
            for i in range(n):
                js = i - s
                if 0 <= js < n:
                    grad[i] += w[m] * (1.0 - h[js])
                jp = i + s
                if 0 <= jp < n:
                    grad[i] -= w[m] * h[jp]
        h -= lr * grad
        h = project(h, tgt)
        c = eval_c5(h)
        if c < best:
            best = c
            best_h = h.copy()
        lam = min(200.0, lam * 1.002)
        lr = max(1e-6, lr * 0.999)
    return best_h, best

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    best_h, best_c5, best_n = None, np.inf, 0

    def rem():
        return max(0.0, budget_s - (time.time() - t_start))

    try:
        hi = np.array(initial_h_values, dtype=float)
        ni = len(hi)
        ci = eval_c5(hi)
        best_h, best_c5, best_n = hi.copy(), ci, ni
    except Exception:
        pass

    starts = []
    ns_to_try = [44, 46, 48, 50, 52, 54, 56, 60, 64, 72, 80, 90, 100]

    for n in ns_to_try:
        if rem() < 30:
            break
        alloc = min(40.0, rem() * 0.35)
        for r in range(3):
            if rem() < 12:
                break
            h = make_01(n, seed * 1000 + n * 37 + r * 13)
            h, c = ls_01(h, n, rng, alloc / (r + 1) + 5)
            if c < best_c5:
                best_h, best_c5, best_n = h.copy(), c, n
            if rem() > 10:
                hc, cc = sg_opt(h, n, rng, min(15, rem() * 0.15))
                if cc < best_c5:
                    best_h, best_c5, best_n = hc.copy(), cc, n

    if rem() > 20:
        for p in [43, 47, 53, 59]:
            if rem() < 15:
                break
            a = legendre_seq(p)
            n = p
            h = a.copy()
            tgt = n / 2.0
            h = project(h, tgt)
            h, c = ls_01(h, n, rng, min(20, rem() * 0.2))
            if c < best_c5:
                best_h, best_c5, best_n = h.copy(), c, n
            if rem() > 10:
                hc, cc = sumexp_opt(h, n, rng, min(15, rem() * 0.15))
                if cc < best_c5:
                    best_h, best_c5, best_n = hc.copy(), cc, n

    if rem() > 25 and best_n > 0:
        hr, cr = sumexp_opt(best_h, best_n, rng, min(80, rem() * 0.5))
        if cr < best_c5:
            best_h, best_c5 = hr.copy(), cr

    if rem() > 10 and best_n > 0:
        hr, cr = sg_opt(best_h, best_n, rng, min(40, rem() * 0.7))
        if cr < best_c5:
            best_h, best_c5 = hr.copy(), cr

    return best_h, best_c5, best_n
