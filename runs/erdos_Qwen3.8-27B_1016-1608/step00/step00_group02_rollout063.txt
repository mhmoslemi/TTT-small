import numpy as np
from scipy.optimize import minimize
from functools import partial
import time


def evaluate(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)


def project(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        mask = h < 1.0 if diff > 0 else h > 0.0
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        idx = int(np.argmax(1.0 - h)) if diff > 0 else int(np.argmin(h))
        h[idx] += diff
    return np.clip(h, 0.0, 1.0)


def penalized_obj(x, target):
    xc = np.clip(x, 0.0, 1.0)
    return evaluate(xc) + 200.0 * (xc.sum() - target) ** 2


def smooth_obj(x, target, beta=80.0):
    xc = np.clip(x, 0.0, 1.0)
    dx = 2.0 / len(xc)
    corr = np.correlate(xc, 1 - xc, mode='full') * dx
    m = np.max(corr)
    sm = m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta
    return sm + 200.0 * (xc.sum() - target) ** 2
