import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def make_feasible(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum}."""
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        s = room.sum()
        if s < 1e-12: return h
        h = h + diff * room / s
    else:
        s = h.sum()
        if s < 1e-12: return h
        h = h + diff * h / s
    return np.clip(h, 0.0, 1.0)

def initial_construction(n):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    # Multi-frequency pattern
    h = 0.5 + 0.25*np.sin(np.pi*x) + 0.15*np.sin(3*np.pi*x) + 0.1*np.sin(5*np.pi*x)
    h = np.clip(h, 0.0, 1.0)
    h = make_feasible(h, n/2.0)
    return h

def upscale(h, n_target):
    n_src = len(h)
    if n_src == n_target: return h.copy()
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_tgt = np.linspace(0.0, 2.0, n_target, endpoint=False)
    h_new = np.interp(x_tgt, x_src, h)
    h_new = make_feasible(h_new, n_target / 2.0)
    return h_new

    """Simulated annealing with sum-preserving moves."""
    target_sum = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    curr_h = h.copy()
    curr_c5 = best_c5
    
    T = 0.02
    T_min = 0.0005
    delta = 0.1
    accepted = 0
    total = 0

            curr_h[j] += amount

            blk = int(rng.randint(2, min(6, n // 4 + 1)))
            i1 = int(rng.randint(max(1, n - blk)))
            i2 = int(rng.randint(max(1, n - blk)))
            if abs(i1 - i2) < blk:
                i2 = (i2 + blk + 1) % max(1, n - blk)
            if i1 == i2:
                continue
            amt = d * rng.randn(blk)
            ni = cur[i1:i1+blk] + amt
            nj = cur[i2:i2+blk] - amt
            if np.any(ni < 0) or np.any(ni > 1) or np.any(nj < 0) or np.any(nj > 1):
                continue
            cur[i1:i1+blk] = ni
            cur[i2:i2+blk] = nj
            nc = compute_c5(cur)
            total += 1
            if nc <= cc or rng.random() < np.exp((cc - nc) / T):
                cc = nc
                accepted += 1
                if nc < bc:
                    bc = nc
                    best = cur.copy()
            else:
                cur[i1:i1+blk] = ni - amt
                cur[i2:i2+blk] = nj + amt
        else:
            ii = int(rng.randint(n))
            jj = int(rng.randint(n))
            if ii == jj:
                continue
            a = d * rng.randn()
            ni = cur[ii] + a
            nj = cur[jj] - a
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            cur[ii] = ni
            cur[jj] = nj
            nc = compute_c5(cur)
            total += 1
            if nc <= cc or rng.random() < np.exp((cc - nc) / T):
                cc = nc
                accepted += 1
                if nc < bc:
                    bc = nc
                    best = cur.copy()
            else:
                cur[ii] = ni - a
                cur[jj] = nj + a
        it += 1
        T = max(Tlo, T * 0.99996)
        if it % 15000 == 0:
            T = max(Tlo, 0.006)
            if total > 200:
                acc = accepted / total
                if acc < 0.15:
                    d = max(0.01, d * 0.7)
                elif acc > 0.85:
                    d = min(0.3, d * 1.3)
            accepted = 0
            total = 0
    return best, bc

def grad_refine(h, budget, n):
    ts = n / 2.0
    t0 = time.time()
    def obj(v):
        hh = make_feasible(v, ts)
        return compute_c5(hh)
    def penalty_obj(v):
        hh = make_feasible(v, ts)
        pen = 100.0 * (hh.sum() - ts) ** 2
        return compute_c5(hh) + pen
    best = h.copy()
    bc = compute_c5(h)
    for attempt in range(5):
        if time.time() - t0 > budget:
            break
        x0 = best + np.random.uniform(-0.05, 0.05, n)
        x0 = make_feasible(x0, ts)
        try:
            res = minimize(penalty_obj, x0, method='Nelder-Mead',
                           options={'maxiter': 3000, 'xatol': 0.001, 'fatol': 1e-7,
                                    'adaptive': True})
            hn = make_feasible(res.x, ts)
            cn = compute_c5(hn)
            if cn < bc:
                bc = cn
                best = hn.copy()
        except Exception:
            pass
    return best, bc

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t_start = time.time()

    best_c5 = 1.0
    best_h = None
    best_n = 80

    def rem():
        return max(0.0, budget_s - (time.time() - t_start))

    # Get starting point
    if globals().get('initial_h_values') is not None:
        h0 = np.array(initial_h_values, dtype=float).flatten()
    else:
        h0 = make_init(80, rng)
    n0 = len(h0)
    c5_0 = compute_c5(h0)
    best_c5, best_h, best_n = c5_0, h0.copy(), n0

    # Phase 1: SA on initial resolution
    b1 = min(280, rem() - 50)
    if b1 > 50:
        h1, c5_1 = sa_optimize(h0, b1, rng, n0, delta=0.07)
        if c5_1 < best_c5:
            best_c5, best_h, best_n = c5_1, h1, n0

    # Phase 2: Grad refine on best
    if rem() > 120 and best_h is not None:
        h2, c5_2 = grad_refine(best_h, min(80, rem() - 40), best_n)
        if c5_2 < best_c5:
            best_c5, best_h = c5_2, h2

    # Phase 3: Upscale and optimize
    for nt in [100, 128, 160, 200]:
        if rem() < 140:
            break
        h_up = upscale(best_h, nt)
        b3 = min(140, rem() - 30)
        h3, c5_3 = sa_optimize(h_up, b3, rng, nt, delta=0.05, T_init=0.01)
        if c5_3 < best_c5:
            best_c5, best_h, best_n = c5_3, h3, nt

    # Phase 4: Random restarts
    for _ in range(6):
        if rem() < 70:
            break
        nt = int(rng.choice([n0, 80, 100, 120]))
        h_r = make_init(nt, rng)
        h_r += rng.uniform(-0.15, 0.15, nt)
        h_r = make_feasible(h_r, nt / 2.0)
        b4 = min(60, rem() - 15)
        h4, c5_4 = sa_optimize(h_r, b4, rng, nt, delta=0.08, T_init=0.02)
        if c5_4 < best_c5:
            best_c5, best_h, best_n = c5_4, h4, nt

    # Phase 5: Coarse search then upscale
    if rem() > 200:
        nc = 32
        h_c = make_init(nc, rng)
        h_c += rng.uniform(-0.2, 0.2, nc)
        h_c = make_feasible(h_c, nc / 2.0)
        b5 = min(100, rem() - 100)
        h5c, c5_5c = sa_optimize(h_c, b5, rng, nc, delta=0.1, T_init=0.03)
        if rem() > 60:
            h5f = upscale(h5c, 120)
            b5f = min(60, rem() - 10)
            h5f, c5_5f = sa_optimize(h5f, b5f, rng, 120, delta=0.04, T_init=0.008)
            if c5_5f < best_c5:
                best_c5, best_h, best_n = c5_5f, h5f, 120

    # Phase 6: Final fine-tune
    if rem() > 20 and best_h is not None:
        b6 = min(50, rem() - 5)
        h6, c5_6 = sa_optimize(best_h, b6, rng, best_n, delta=0.02, T_init=0.003)
        if c5_6 < best_c5:
            best_c5, best_h = c5_6, h6

    # Final validation
    best_h = make_feasible(best_h, best_n / 2.0)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, best_n
