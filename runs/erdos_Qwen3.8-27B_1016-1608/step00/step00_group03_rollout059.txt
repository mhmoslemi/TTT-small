import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * 2.0 / n

def sum_constraint(x):
    return x.sum() - len(x) * 0.5

def sa_binary(n, rng, h_init=None, T_init=1.0, T_min=0.005, cooling=0.9998,
              max_iters=50000, deadline=None):
    if h_init is not None:
        h = (h_init >= 0.5).astype(np.float64)
    else:
        h = np.zeros(n, dtype=np.float64)
        h[rng.choice(n, n // 2, replace=False)] = 1.0
    tgt = n // 2
    ni = int(h.sum())
    if ni > tgt:
        idx = np.where(h == 1)[0]
        h[rng.choice(idx, ni - tgt, replace=False)] = 0
    elif ni < tgt:
        idx = np.where(h == 0)[0]
        h[rng.choice(idx, tgt - ni, replace=False)] = 1
    best_h = h.copy()
    best_cm = float(np.max(np.correlate(h, 1.0 - h, mode="full")))
    cur_cm = best_cm
    T = T_init
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    for _ in range(max_iters):
        if deadline is not None and time.time() > deadline:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h[i], h[j] = h[j], h[i]
        nm = float(np.max(np.correlate(h, 1.0 - h, mode="full")))
        delta = nm - cur_cm
        if delta < 0 or (T > T_min and rng.random() < np.exp(-delta / T)):
            cur_cm = nm
            if cur_cm < best_cm:
                best_cm = cur_cm
                best_h = h.copy()
                ones = np.where(h == 1)[0].tolist()
                zeros = np.where(h == 0)[0].tolist()
        else:
            h[i], h[j] = h[j], h[i]
        T *= cooling
        if T < T_min:
            break
    return best_h, best_cm
