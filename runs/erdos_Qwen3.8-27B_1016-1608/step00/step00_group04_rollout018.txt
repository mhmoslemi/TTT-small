import numpy as np
import math
import time
import random


def compute_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float((corr * dx).max())


def local_polish(h, max_passes=200):
    score = compute_score(h)
    for _ in range(max_passes):
        improved = False
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        for i in ones_idx:
            for j in zeros_idx:
                h[i], h[j] = 0.0, 1.0
                ns = compute_score(h)
                if ns < score - 1e-12:
                    score = ns
                    improved = True
                else:
                    h[i], h[j] = 1.0, 0.0
        if not improved:
            break
    return h, score


def sa_optimize(h, seed, time_limit):
    rng = random.Random(seed)
    t_start = time.time()
    h = (h > 0.5).astype(np.float64)
    n = len(h)
    score = compute_score(h)
    best_score = score
    best_h = h.copy()
    T0 = 0.025
    cool_time = max(time_limit * 0.9, 1.0)

    while time.time() - t_start < time_limit:
        elapsed = time.time() - t_start
        T = max(T0 * math.exp(-6.0 * elapsed / cool_time), 5e-7)
        if T <= 6e-7:
            # Greedy phase
            steps = 0
            while steps < 20000 and time.time() - t_start < time_limit:
                ones_idx = np.where(h == 1)[0]
                zeros_idx = np.where(h == 0)[0]
                i = int(ones_idx[rng.randrange(len(ones_idx))])
                j = int(zeros_idx[rng.randrange(len(zeros_idx))])
                if i == j:
                    steps += 1
                    continue
                h[i], h[j] = 0.0, 1.0
                ns = compute_score(h)
                if ns < score:
                    score = ns
                    if score < best_score:
                        best_score = score
                        best_h = h.copy()
                else:
                    h[i], h[j] = 1.0, 0.0
                steps += 1
            break

        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        i = int(ones_idx[rng.randrange(len(ones_idx))])
        j = int(zeros_idx[rng.randrange(len(zeros_idx))])
        if i == j:
            continue
        h[i], h[j] = 0.0, 1.0
        ns = compute_score(h)
        delta = ns - score
        if delta <= 0 or rng.random() < math.exp(-delta / T):
            score = ns
            if score < best_score:
                best_score = score
                best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
    return best_score, best_h


def make_random_h(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n, dtype=np.float64)
    pos = rng.choice(n, n // 2, replace=False)
    h[pos] = 1.0
    return h


def upscale_and_optimize(h_small, n_large, seed, time_limit):
    # Replace each bit with n_large/n_small copies
    ratio = n_large // len(h_small)
    h = np.repeat(h_small, ratio).astype(np.float64)
    # Ensure correct sum
    target = n_large // 2
    cur = int(np.sum(h))
    if cur > target:
        idx = np.where(h == 1)[0]
        rng_fix = np.random.RandomState(seed)
        rng_fix.shuffle(idx)
        h[idx[target:]] = 0
    elif cur < target:
        idx = np.where(h == 0)[0]
        rng_fix = np.random.RandomState(seed)
        rng_fix.shuffle(idx)
        h[idx[:(target - cur)]] = 1
    score, h = sa_optimize(h, seed, time_limit)
    h, score = local_polish(h)
    return score, h, n_large
