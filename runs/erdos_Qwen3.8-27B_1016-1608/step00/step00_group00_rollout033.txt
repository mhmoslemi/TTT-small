```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def smooth_objective(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    # Log-sum-exp for numerical stability
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    n = 45
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        if len(h0) != n:
            h0 = None
    except:
        h0 = None
    
    if h0 is None:
        h0 = np.full(n, 0.5)
    
    # Enforce constraints
    h0 = np.clip(h0, 0, 1)
    diff = n / 2.0 - np.sum(h0)
    h0 += diff / n
    h0 = np.clip(h0, 0, 1)
    
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, n)
    print(f"Initial C5: {best_c5:.6f}")
    
    # Optimize with increasing beta
    for beta in [5, 10, 20, 50, 100, 200, 500]:
        if time.time() - start_time > budget_s - 30:
            break
        
        try:
            res = minimize(
                lambda x: smooth_objective(x, n, beta),
                best_h,
                method='SLSQP',
                bounds=[(0, 1)] * n,
                constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0}],
                options={'maxiter': 500, 'ftol': 1e-10}
            )
            h_new = np.clip(res.x, 0, 1)
            # Fix sum constraint
            diff = n/2.0 - np.sum(h_new)
            h_new += diff / n
            h_new = np.clip(h_new, 0, 1)
            
            c5_new = compute_c5(h_new, n)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                print(f"beta={beta}: C5={best_c5:.6f}")
        except Exception as e:
            print(f"Error at beta={beta}: {e}")
    
    # Try multiple random restarts
    time_left = budget_s - (time.time() - start_time)
    n_restarts = max(1, int(time_left / 30))

        h_r += diff / n
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_g_n = 45
_g_beta = 50.0

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_constraints(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def sum_eq(x):
    return np.sum(x) - len(x) / 2.0

def sum_eq_jac(x):
    return np.ones(len(x))

def obj_func(x):
    x = np.clip(x, 0.0, 1.0)
    dx = 2.0 / _g_n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    mx = np.max(corr)
    scaled = _g_beta * (corr - mx)
    scaled = np.clip(scaled, -60.0, 60.0)
    return float(mx + np.log(np.sum(np.exp(scaled))) / _g_beta)

def linear_ramp(n):
    return 1.0 - (np.arange(n, dtype=float) + 0.5) / n

def optimize_candidate(h0, n, betas, max_iter, t0, budget):
    global _g_n, _g_beta
    _g_n = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac}]
    h_cur = h0.copy()
    c5_cur = compute_c5(h_cur, n)
    for beta in betas:
        if time.time() - t0 > budget - 15:
            break
        _g_beta = float(beta)
        best_local = h_cur
        best_local_c5 = c5_cur
        for attempt in range(2):
            if time.time() - t0 > budget - 15:
                break
            try:
                res = minimize(obj_func, h_cur, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-13})
                h_new = fix_constraints(res.x, n)
                c5_new = compute_c5(h_new, n)
                if c5_new < best_local_c5:
                    best_local_c5 = c5_new
                    best_local = h_new
                    h_cur = h_new
                    c5_cur = c5_new
            except Exception:
                pass
    return best_local, best_local_c5

def local_nudge(h, n, rng, steps, t0, budget):
    h = h.copy()
    c5_cur = compute_c5(h, n)
    for _ in range(steps):
        if time.time() - t0 > budget - 5:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        delta = 0.02 * rng.uniform(0.5, 1.0) * (1 if rng.rand() < 0.5 else -1)
        if h[i] + delta < 0 or h[i] + delta > 1 or h[j] - delta < 0 or h[j] - delta > 1:
            continue
        h[i] += delta
        h[j] -= delta
        c5_new = compute_c5(h, n)
        if c5_new < c5_cur:
            c5_cur = c5_new
        else:
            h[i] -= delta
            h[j] += delta
    return h, c5_cur

def structured_start(n, seed):
    rng = np.random.RandomState(seed)
    r = rng.rand(n)
    return fix_constraints(r, n)

def run(seed=42, budget_s=1100, **kwargs):

    rng = np.random.RandomState(seed)
    n = 45

    h_ramp = fix_constraints(linear_ramp(n), n)

    except Exception:
        pass

    for s in range(4):

    best_h = candidates[0].copy()
    best_c5 = compute_c5(best_h, n)
    for hc in candidates:
        c = compute_c5(fix_constraints(hc, n), n)
        if c < best_c5:
            best_c5 = c
            best_h = fix_constraints(hc, n)

    betas = [5, 15, 50, 150, 500, 1500]
    for hc in candidates:
        if time.time() - t0 > budget_s - 40:

        h_opt, c5_opt = optimize_candidate(hc, n, betas, 800, t0, budget_s)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Nudge search on best
    if time.time() - t0 < budget_s - 30:
        best_h, best_c5 = local_nudge(best_h, n, rng, 5000, t0, budget_s)

    # High-beta refinement
    if time.time() - t0 < budget_s - 20:
        h_opt, c5_opt = optimize_candidate(best_h, n, [3000, 8000], 1500, t0, budget_s)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    best_h = fix_constraints(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
```
