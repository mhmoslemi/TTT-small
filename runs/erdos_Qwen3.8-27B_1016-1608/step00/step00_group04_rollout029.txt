def _score(h):

    return float(np.max(np.correlate(h, 1.0 - h, "full")) * dx)

    for _ in range(100):
        d = ts - h.sum()

            break
        m = (h > 1e-15) & (h < 1.0 - 1e-15)
        cnt = int(m.sum())
        if cnt == 0:
            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _subgrad(h, n):
    """Compute subgradient of max overlap"""
    ooh = 1.0 - h
    corr = np.correlate(h, ooh, "full")
    maxv = np.max(corr)
    # Find all active lags (within tolerance)
    active = np.where(corr > maxv - 1e-6)[0]
    # Average gradient over active lags
    grad = np.zeros(n)
    for idx in active:
        k = idx - (n - 1)
        for i in range(n):
            v = 0.0
            if 0 <= i - k < n:
                v += 1.0 - h[i - k]
            if 0 <= i + k < n:
                v -= h[i + k]
            grad[i] += v
    if len(active) > 0:
        grad /= len(active)
    grad -= grad.mean()
    return grad, maxv

    h = h.copy()
    s = _score(h)
    temp = 0.05

            h[j] += d
