def _lse(arr, lam):
    m = np.max(arr)
    return m + (1.0 / lam) * math.log(float(np.sum(np.exp(lam * (arr - m)))))

def _proj(h, tgt):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = tgt - h.sum()
        if abs(d) < 1e-14:
            break
        mask = (h < 1.0) if d > 0 else (h > 0.0)
        if not mask.any():
            break
        h[mask] += d / mask.sum()
    return np.clip(h, 0.0, 1.0)

        if not mask.any():
            break
        h[mask] += d / mask.sum()
    return np.clip(h, 0.0, 1.0)

def _f_pen(x):
    h = np.clip(x, 0.0, 1.0)
    c = np.correlate(h, 1.0 - h, mode='full') * _G['dx']
    return _lse(c, _G['lam']) + _G['pen'] * (h.sum() - _G['tgt']) ** 2

def _f_con(x):
    c = np.correlate(x, 1.0 - x, mode='full') * _G['dx']
    return _lse(c, _G['lam'])

def _cf(x):
    return x.sum() - _G['tgt']

def _cj(x):
    return np.ones(_G['n'])

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    try:
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    except NameError:
        h0 = np.ones(45) * 0.5
    n = len(h0)
    tgt = float(n) / 2.0
    dx = 2.0 / n
    _G['n'], _G['dx'], _G['tgt'], _G['pen'] = n, dx, tgt, 200.0
    best = _proj(h0, tgt)
    bc5 = _c5(best)

    starts = [best.copy()]
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
    starts.append(_proj(alt, tgt))
    for per in [4, 6, 9]:
        w = 0.5 + 0.5 * np.cos(2.0 * np.pi * np.arange(n) * dx / (per * dx))
        starts.append(_proj(w, tgt))
    for _ in range(4):
        p = rng.permutation(n)
        r = np.zeros(n)
        k = int(tgt)
        r[p[:k]] = 1.0
        fr = tgt - k
        if fr > 0:
            r[p[k]] = fr
        starts.append(_proj(r, tgt))

    # Phase 1: L-BFGS-B with LSE annealing
    for lam in [8, 20, 50, 120, 300, 800, 2000]:
        _G['lam'] = lam
        ns = 14 if lam < 200 else 8
        for r in range(ns):
            if time.time() - t0 > budget_s - 120:
                break
            if r == 0:
                x0 = best.copy()
            elif r == 1:
                x0 = _proj(best + rng.randn(n) * 0.04, tgt)
            elif r == 2:
                x0 = _proj(best + rng.randn(n) * 0.1, tgt)
            elif r == 3:
                x0 = _proj(np.roll(best, rng.randint(1, max(2, n // 5))), tgt)
            else:
                x0 = _proj(starts[r % len(starts)] + rng.randn(n) * 0.03, tgt)
            try:
                res = minimize(_f_pen, x0, method='L-BFGS-B',
                              bounds=[(0.0, 1.0)] * n,
                              options={'maxiter': 600, 'ftol': 1e-15})
                ho = _proj(res.x, tgt)
                c5 = _c5(ho)
                if c5 < bc5:
                    bc5, best = c5, ho
                    starts[0] = best.copy()
            except Exception:
                pass

    # Phase 2: SLSQP with equality constraint
    for lam in [300, 1000, 3000]:
        _G['lam'] = lam
        for r in range(12):
            if time.time() - t0 > budget_s - 40:
                break
            x0 = _proj(best + rng.randn(n) * 0.01, tgt)
            try:
                res = minimize(_f_con, x0, method='SLSQP',
                              bounds=[(0.0, 1.0)] * n,
                              constraints=[{'type': 'eq', 'fun': _cf, 'jac': _cj}],
                              options={'maxiter': 2000, 'ftol': 1e-14})
                ho = _proj(res.x, tgt)
                c5 = _c5(ho)
                if c5 < bc5:
                    bc5, best = c5, ho
            except Exception:
                pass

    # Phase 3: Random pair-swap coordinate descent
    for it in range(300000):
        if time.time() - t0 > budget_s - 8:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(n - 1)) % n
        dt = rng.randn() * (0.003 + 0.04 * rng.rand())
        hv = best.copy()
        hv[i] += dt
        hv[j] -= dt
        if 0.0 <= hv[i] <= 1.0 and 0.0 <= hv[j] <= 1.0:
            c5t = _c5(hv)
            if c5t < bc5:
                best, bc5 = hv, c5t

    # Phase 4: Fine grid local search
    for i in range(n):
        if time.time() - t0 > budget_s - 3:
            break
        for j in range(n):
            if i == j:
                continue
            for dt in [0.005, -0.005, 0.01, -0.01, 0.02, -0.02]:
                hv = best.copy()
                hv[i] += dt
                hv[j] -= dt
                if 0.0 <= hv[i] <= 1.0 and 0.0 <= hv[j] <= 1.0:
                    c5t = _c5(hv)
                    if c5t < bc5:
                        bc5, best = c5t, hv

    best = _proj(best, tgt)
    bc5 = _c5(best)
    return best, bc5, n
