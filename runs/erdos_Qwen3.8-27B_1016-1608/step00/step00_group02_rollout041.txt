import numpy as np
import time

def project_feasible(h, n):
    """Project h onto {x in [0,1]^n : sum(x) = n/2} via bisection."""
    target = n / 2.0
    lo, hi = np.min(h) - 1.0, np.max(h) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_overlaps(h, n):
    """Compute overlap at each lag. Returns array of shape (2n-1,)."""
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_gradient(h, n, w):
    """
    Gradient of smooth max of overlaps.
    w: softmax weights for each of the 2n-1 lags, w[d+(n-1)] for displacement d.
    """
    grad = np.zeros(n)
    for d in range(-(n-1), n):
        wd = w[d + n - 1]
        if abs(wd) < 1e-15:
            continue
        if d >= 0:
            grad[:n-d] += wd * (1.0 - h[d:])
            grad[d:] -= wd * h[:n-d]
        else:
            dd = -d
            grad[dd:] += wd * (1.0 - h[:n-dd])
            grad[:n-dd] -= wd * h[dd:]
    return grad

def smooth_max(v, t):
    """Compute smooth max and softmax weights."""
    vmax = np.max(v)
    shifted = t * (v - vmax)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_v = np.exp(shifted)
    sum_exp = np.sum(exp_v)
    w = exp_v / sum_exp
    val = vmax + np.log(sum_exp) / t
    return val, w

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = np.min(h) - 1.0, np.max(h) + 1.0
    for _ in range(100):
