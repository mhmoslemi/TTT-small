                if budget_s - elapsed() < 15:
                    break

    # Phase 4: Final SLSQP polish
    if budget_s - elapsed() > 15:
        res = minimize(obj_fn, best_h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 3000, 'ftol': 1e-14})
        cand = project_feasible(res.x, target)
        c5_cand = compute_c5(cand, dx)
        if c5_cand < best_c5:
            best_h = cand
            best_c5 = c5_cand

    # Ensure exact feasibility
    best_h = project_feasible(best_h, target)
    best_c5 = compute_c5(best_h, dx)

    # Verify with pre-imported evaluator if available
    if 'evaluate_erdos_solution' in dir() and callable(evaluate_erdos_solution):
        try:
            verified = evaluate_erdos_solution(best_h)
            if verified < best_c5:
                best_c5 = verified
        except Exception:
            pass

    return (best_h, best_c5, n)
