    """Simulated annealing with pairwise swaps"""
    t0 = time.time()
    cur = compute_c5(h)
    best = cur
    best_h = h.copy()

            room = 1.0 - h
        else:
            room = h

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def fix_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            room = 1.0 - h
        else:
            room = h
        s = room.sum()
        if s > 1e-10:
            h = h + diff * room / s
    return np.clip(h, 0.0, 1.0)

def sa_optimize(h, n, rng, T_start, t_limit):
    t0 = time.time()
    T = T_start
    T_min = 1e-7
    cur = compute_c5(h)
    best = cur
    best_h = h.copy()
    while T > T_min and time.time() - t0 < t_limit:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            T *= 0.99999
            continue
        scale = 0.3 + 0.7 * (T / T_start)
        d = rng.uniform(-0.15 * scale, 0.15 * scale)
        nh = h[i] + d
        nv = h[j] - d
        if nh < 0.0 or nh > 1.0 or nv < 0.0 or nv > 1.0:
            T *= 0.99999
            continue
        h[i] = nh
        h[j] = nv
        nc = compute_c5(h)
        diff = nc - cur
        if diff < 0.0 or rng.rand() < np.exp(-diff / max(T, 1e-10)):
            cur = nc
            if nc < best:
                best = nc
                best_h = h.copy()
        else:
            h[i] -= d
            h[j] += d
        T *= 0.99999
    return best_h, best

def greedy_local(h, n, rng, t_limit):
    t0 = time.time()
    cur = compute_c5(h)
    steps = (0.02, 0.04, 0.07, 0.1, -0.02, -0.04, -0.07, -0.1)
    for _ in range(500):
        if time.time() - t0 > t_limit:
            break
        improved = False
        idx = np.arange(n)
        rng.shuffle(idx)
        for a in range(n - 1):
            if time.time() - t0 > t_limit:
                break
            i = idx[a]
            for b in range(a + 1, n):
                j = idx[b]
                for s in steps:
                    nh = h[i] + s
                    nv = h[j] - s
                    if 0.0 <= nh <= 1.0 and 0.0 <= nv <= 1.0:
                        h[i] = nh
                        h[j] = nv
                        c = compute_c5(h)
                        if c < cur:
                            cur = c
                            improved = True
                            break
                        h[i] -= s
                        h[j] += s
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return h.copy(), cur
