        h = h - alpha * grad
        h = project_bounded_simplex(h, n / 2.0)
        avg_h += (h - avg_h) / (avg_w + 1.0)
        avg_w += 1.0
        alpha *= 0.9998
        if it % 500 == 499:
            avg_c5 = compute_c5(avg_h)
            if avg_c5 < best_c5:
                best_c5 = avg_c5
                best_h = avg_h.copy()
                h = avg_h.copy()
    avg_h = project_bounded_simplex(avg_h, n / 2.0)
    avg_c5 = compute_c5(avg_h)
    if avg_c5 < best_c5:
        best_c5 = avg_c5
        best_h = avg_h
    return best_h, best_c5

def slsqp_optimize(h0, time_limit=30.0):
    n = len(h0)
    s = 25
    pen = 5000.0

    def obj(x):
        c = np.correlate(x, 1.0 - x, mode="full")
        val = smooth_max_val(c, s) * (2.0 / n)
        val += pen * (np.sum(x) - n / 2.0) ** 2
        return val

    def obj_grad(x):
        g = smooth_max_weighted_grad(x, s)
        g += 2.0 * pen * (np.sum(x) - n / 2.0) * np.ones(n)
        return g

    t_start = time.time()
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=obj_grad, method='SLSQP',
                       bounds=bounds,
                       options={'maxiter': 200, 'ftol': 1e-10})
        if time.time() - t_start < time_limit:
            res2 = minimize(obj, res.x, jac=obj_grad, method='SLSQP',
                            bounds=bounds,
                            options={'maxiter': 200, 'ftol': 1e-12})
            h1 = np.clip(res2.x, 0, 1)
        else:
            h1 = np.clip(res.x, 0, 1)
    except Exception:
        h1 = h0.copy()
    h1 = project_bounded_simplex(h1, n / 2.0)
    return h1

def polish_local(h, time_limit=30.0):
    n = len(h)
    h = h.astype(float).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    t_start = time.time()
    rng = np.random.RandomState(123)
    step = 0.05
    for _ in range(50000):
        if time.time() - t_start > time_limit:
            break
        i = rng.randint(n)
        delta = rng.choice([-1.0, 1.0]) * step
        h_new = h.copy()
        h_new[i] = np.clip(h_new[i] + delta, 0, 1)
        h_new = project_bounded_simplex(h_new, n / 2.0)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            h = h_new
            best_c5 = c5_new
            best_h = h.copy()
        else:
            step = max(0.005, step * 0.99)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    best_h = None
    best_c5 = 1.0
    best_n = 0

    def remaining_frac(used=0.09):
        return (budget_s - (time.time() - t0)) / (budget_s * (1.0 - used)) > 1.0

    starts = []

    try:
        h_init = np.array(initial_h_values, dtype=float)
        h_init = np.clip(h_init, 0.0, 1.0)
        h_init = project_bounded_simplex(h_init, len(h_init) / 2.0)
        starts.append((len(h_init), h_init, "initial"))
    except Exception:
        pass

    for n in [45, 90, 200]:
        x = np.arange(n)
        for freq, amp in [(1, 0.35), (1, 0.45), (2, 0.25), (2, 0.35)]:
            h_s = 0.5 + amp * np.sin(2.0 * np.pi * freq * x / n)
            h_s = project_bounded_simplex(h_s, n / 2.0)
            starts.append((n, h_s, "sin_f{}_a{}".format(freq, int(amp*100))))
        for trial in range(2):
            h_r = rng.rand(n)
            h_r = project_bounded_simplex(h_r, n / 2.0)
            starts.append((n, h_r, "rand_{}_{}".format(n, trial)))

    slsqp_time = min(20.0, budget_s * 0.04)
    sg_time = min(50.0, budget_s * 0.06)
    polish_time = min(20.0, budget_s * 0.03)

    seen_n = set()
    for n, h0, label in starts:
        if not remaining_frac():
            break
        if n in seen_n and n > 45:
            continue
        seen_n.add(n)

        c5_0 = compute_c5(h0)
        t_phase1 = time.time()
        h1 = slsqp_optimize(h0, time_limit=slsqp_time)
        c5_1 = compute_c5(h1)
        if c5_1 >= c5_0:
            h1, c5_1 = h0, c5_0

        h2, c5_2 = subgradient_optimize(h1, time_limit=sg_time)
        if c5_2 >= c5_1:
            h2, c5_2 = h1, c5_1

        h3, c5_3 = polish_local(h2, time_limit=polish_time)
        if c5_3 < c5_2:
            h2, c5_2 = h3, c5_3

        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
            best_n = n

    if best_h is not None and remaining_frac():
        t_left = budget_s * 0.85 - (time.time() - t0)
        if t_left > 15:
            h_final, c5_final = subgradient_optimize(best_h, time_limit=t_left * 0.6)
            if c5_final < best_c5:
                best_c5 = c5_final
                best_h = h_final.copy()
        t_left2 = budget_s * 0.88 - (time.time() - t0)
        if t_left2 > 10:
            h_pol, c5_pol = polish_local(best_h, time_limit=t_left2)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    if best_h is None:
        n = 45
        best_h = project_bounded_simplex(np.full(n, 0.5), n / 2.0)
        best_c5 = compute_c5(best_h)
        best_n = n

    return best_h, best_c5, best_n
