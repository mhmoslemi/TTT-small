    target = n // 2
    h_bin = (h > 0.5).astype(float)
    s = int(np.sum(h_bin))
    if s > target:
        idx = np.flatnonzero(h_bin)
        np.random.shuffle(idx)
        h_bin[idx[:s-target]] = 0
    elif s < target:
        idx = np.flatnonzero(1 - h_bin)
        np.random.shuffle(idx)
        h_bin[idx[:target-s]] = 1
    return h_bin

def steepest_descent(h, max_iter=2000, time_limit=None, start_time=None):
    n = len(h)
    h = h.copy()
    score = compute_c5(h)
    for _ in range(max_iter):
        if time_limit and start_time and time.time() - start_time > time_limit:
            break
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        best_improve = 1e-12
        best_i = best_j = -1
        for i in ones:
            h[i] = 0
            for j in zeros:
                h[j] = 1
                s = compute_c5(h)
                h[j] = 0
                if score - s > best_improve:
                    best_improve = score - s
                    best_i = i
                    best_j = j
            h[i] = 1
        if best_i < 0:
            break
        h[best_i] = 0
        h[best_j] = 1
        score = compute_c5(h)
    return h, score

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def makefeasible(h, n):
    target = n // 2
    hb = (h > 0.5).astype(float)
    s = int(np.sum(hb))
    if s > target:
        idx = np.flatnonzero(hb).copy()
        np.random.shuffle(idx)
        hb[idx[:s - target]] = 0
    elif s < target:
        idx = np.flatnonzero(1 - hb).copy()
        np.random.shuffle(idx)
        hb[idx[:target - s]] = 1
    return hb

def steepest(h, max_iter=3000, deadline=None):
    n = len(h)
    h = h.copy()
    sc = compute_c5(h)
    for _ in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_d = 1e-12
        best_i = best_j = -1
        for i in ones:
            h[i] = 0
            for j in zeros:
                h[j] = 1
                s = compute_c5(h)
                h[j] = 0
                d = sc - s
                if d > best_d:
                    best_d = d
                    best_i = i
                    best_j = j
            h[i] = 1
        if best_i < 0:
            break
        h[best_i] = 0
        h[best_j] = 1
        sc = compute_c5(h)
    return h, sc

def kick_and_optimize(best_h, n, nk_range=(2, 6), max_iter=2000, deadline=None):
    hp = best_h.copy()

        ones = np.flatnonzero(hp)
        zeros = np.flatnonzero(1 - hp)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        hp[i] = 0
        hp[j] = 1
    ho, so = steepest(hp, max_iter=max_iter, deadline=deadline)
    return ho, so

def simulated_anneal(h, n, T0=0.05, T_min=0.0005, cooling=0.9998, max_iter=100000, deadline=None):
    h = h.copy()
    sc = compute_c5(h)
    T = T0
    for _ in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        if T < T_min:
            break
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        h[i] = 0
        h[j] = 1
        ns = compute_c5(h)
        delta = ns - sc
        if delta < 0 or np.random.random() < np.exp(-delta / T):
            sc = ns
        else:
            h[j] = 0
            h[i] = 1
        T *= cooling
    return h, sc
