def compute_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def make_binary(arr, n, rng):
    h = np.zeros(n)
    if arr is not None:
        a = np.asarray(arr, dtype=float).ravel()
        m = min(len(a), n)
        h[:m] = (a[:m] > 0.5).astype(float)
    target = n // 2
    s = int(np.sum(h))
    if s > target:
        idx = np.where(h == 1)[0]
        rng.shuffle(idx)
        h[idx[:s - target]] = 0
    elif s < target:
        idx = np.where(h == 0)[0]
        rng.shuffle(idx)
        h[idx[:target - s]] = 1
    return h

def local_search(h, n, max_rounds=500):
    best = h.copy()
    best_s = compute_score(best)
    for _ in range(max_rounds):
        oi = np.where(best == 1)[0]
        zi = np.where(best == 0)[0]
        bi, bj = -1, -1
        bs = best_s - 1e-15  # need strict improvement
        for i in oi:
            best[i] = 0
            for j in zi:

                best[j] = 0
            best[i] = 1
        if bi < 0:
            break
        best[bi] = 0; best[bj] = 1
        best_s = bs

def perturb(h, n, k, rng):

def block_perturb(h, n, rng):
    h = h.copy()
    seg_len = rng.randint(2, max(3, n // 4))

    seg = h[start:start + seg_len].copy()
    shift = rng.randint(1, seg_len)
