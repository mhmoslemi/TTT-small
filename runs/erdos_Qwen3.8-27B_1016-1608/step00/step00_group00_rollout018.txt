                    if c5t < best_c5 - 1e-10:
                        best_c5 = c5t
                        h_cd = h_tmp.copy()
                        improved = True
                        break
            if not improved:
                break
        if np.sum(h_cd) != target_sum:
            h_cd = fix_sum(h_cd, target_sum)
        c5_final = compute_c5(h_cd, dx)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_cd.copy()

    # Ensure final feasibility
    best_h = fix_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h, dx)
    return best_h, best_c5, n_pts
