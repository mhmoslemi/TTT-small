            h_c[a] = 0.0
            h_c[b] = 1.0
            s_n = compute_score(h_c)
            h_c[a] = 1.0
            h_c[b] = 0.0
            if s_n < best_flip_s:
                best_flip_s = s_n
                best_a, best_b = a, b
        if best_a >= 0:
            h_c[best_a] = 0.0
            h_c[best_b] = 1.0
            s_c = best_flip_s
            if s_c < bs:
                bs = s_c
                bh = h_c.copy()
            improved = True
        if not improved:
            break
    return bh, bs

import numpy as np
import time


def compute_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def make_feasible(h, sv=0):
    n = len(h)
    target = n // 2
    h = np.array(h, dtype=float).copy()
    h = (h > 0.5).astype(float)
    cur = int(np.sum(h))
    if cur != target:
        rng = np.random.RandomState(sv)
        if cur > target:
            idx = rng.choice(np.flatnonzero(h), cur - target, replace=False)
            h[idx] = 0
        else:
            idx = rng.choice(np.flatnonzero(1 - h), target - cur, replace=False)
            h[idx] = 1
    return h


def greedy_local(h, deadline, rng, n_samples=40):
    h_c = h.copy()
    s_c = compute_score(h_c)
    bh = h_c.copy()
    bs = s_c
    while time.time() < deadline:
        ones = np.flatnonzero(h_c)
        zeros = np.flatnonzero(1.0 - h_c)
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_s = s_c
        best_a, best_b = -1, -1
        for _ in range(n_samples):
            a = ones[rng.randint(len(ones))]
            b = zeros[rng.randint(len(zeros))]
            if a == b:
                continue
            h_c[a] = 0.0
            h_c[b] = 1.0
            s_n = compute_score(h_c)
            h_c[a] = 1.0
            h_c[b] = 0.0
            if s_n < best_s:
                best_s = s_n
                best_a, best_b = a, b
        if best_a < 0:
            break
        h_c[best_a] = 0.0
        h_c[best_b] = 1.0
        s_c = best_s
        if s_c < bs:
            bs = s_c
            bh = h_c.copy()
    return bh, bs


def sa_run(h, deadline, rng, T0=0.004, Tmin=0.00005, cool=0.99998):
    h_c = h.copy()
    s_c = compute_score(h_c)
    T = T0
    bh = h_c.copy()
    bs = s_c
    while time.time() < deadline:
        ones = np.flatnonzero(h_c)
        zeros = np.flatnonzero(1.0 - h_c)
        if len(ones) == 0 or len(zeros) == 0:
            break
        a = ones[rng.randint(len(ones))]
        b = zeros[rng.randint(len(zeros))]
        if a == b:
            continue
        h_c[a] = 0.0
        h_c[b] = 1.0
        s_n = compute_score(h_c)
        d = s_n - s_c
        if d < 0 or rng.random() < np.exp(-d / T):
            s_c = s_n
            if s_n < bs:
                bs = s_n
                bh = h_c.copy()
        else:
            h_c[a] = 1.0
            h_c[b] = 0.0
        T = max(T * cool, Tmin)
    return bh, bs
