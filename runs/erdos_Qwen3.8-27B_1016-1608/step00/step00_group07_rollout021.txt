def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _repair_sum(h, target):
    d = target - h.sum()
    if abs(d) < 1e-13:
        return h
    if d > 0:
        room = 1.0 - h
        s = room.sum()
        if s > 1e-15:
            h = h + room * (d / s)
    else:
        room = h.copy()
        s = room.sum()
        if s > 1e-15:
            h = h - room * (-d / s)
    return np.clip(h, 0.0, 1.0)
