    diff = target - np.sum(h)
    if abs(diff) < 1e-12:

        for i in idx:
            if rem <= 1e-14:
                break

            h[i] += add
            rem -= add
    else:

        rem = -diff

            if rem <= 1e-14:
                break

            h[i] -= sub
            rem -= sub

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(v, alpha):
    a = alpha * v
    m = np.max(a)
    if m > 500:
        return float(np.max(v))
    return float((m + np.log(np.sum(np.exp(a - m)))) / alpha)

def fix_sum(h, target):
    h = np.array(h, dtype=np.float64, copy=True)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.argsort(1.0 - h)
        rem = diff
        for i in idx:
            if rem <= 1e-14: break
            a = min(1.0 - h[i], rem)
            h[i] += a; rem -= a
    else:
        idx = np.argsort(h)
        rem = -diff
        for i in idx:
            if rem <= 1e-14: break
            s = min(h[i], rem)
            h[i] -= s; rem -= s
    return h

        if len(ones) == 0 or len(zeros) == 0:
            break

                h[a] = 0.0; h[b] = 1.0
                c5_try = compute_c5(h, dx)

                    improved = True
                    break
