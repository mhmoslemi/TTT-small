    target = n_points / 2.0

    lo, hi = -2.0, 2.0

        s = 0.0
        for k in range(2 * n - 1):
            d = k - (n - 1)
            c = 0.0
            jd = j - d
            jd2 = j + d
            if 0 <= jd < n:
                c += (1.0 - h[jd])
            if 0 <= jd2 < n:
                c -= h[jd2]
            s += w[k] * c
        grad[j] = dx * s
    return grad

def gradient_descent(h, n, target, max_iters, deadline):
    c5_cur = c5_value(h, n)
    best_local = h.copy()
    best_c5 = c5_cur
    for beta in [5, 10, 20, 50, 100, 200]:
        lr = 1.0
        for it in range(max_iters):
            if time.time() > deadline:
                return best_local, best_c5
            grad = grad_smoothed_max(h, n, beta)
            gnorm = float(np.linalg.norm(grad))
            if gnorm < 1e-9:
                break
            obj_cur = c5_value(h, n)
            improved = False
            for trial in range(20):
                h_new = project_constraint(h - lr * grad, target)
                obj_new = c5_value(h_new, n)
                if obj_new < obj_cur:
                    h = h_new
                    c5_cur = obj_new
                    improved = True
                    if c5_cur < best_c5:
                        best_c5 = c5_cur
                        best_local = h.copy()
                    break
                lr *= 0.5
            if not improved:
                lr = 0.01
                h_try = project_constraint(h - lr * grad, target)
                if c5_value(h_try, n) < c5_cur:
                    h = h_try
                    c5_cur = c5_value(h, n)
                    if c5_cur < best_c5:
                        best_c5 = c5_cur
                        best_local = h.copy()
                else:
                    break
    return best_local, best_c5

def local_search_binary(h_bin, n, deadline, max_iters):
    c5_cur = c5_value(h_bin, n)
    best = h_bin.copy()
    best_c5 = c5_cur
    ones = np.where(h_bin == 1.0)[0]
    zeros = np.where(h_bin == 0.0)[0]
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        h_bin[i] = 0.0
        h_bin[j] = 1.0
        c5_new = c5_value(h_bin, n)
        if c5_new < c5_cur:
            c5_cur = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best = h_bin.copy()
        else:
            h_bin[i] = 1.0
            h_bin[j] = 0.0
    return best, best_c5

def make_binary(h, n):
    h_bin = np.where(h > 0.5, 1.0, 0.0)
    cnt = float(np.sum(h_bin))
    target = n // 2
    if cnt > target:
        idx = np.argsort(h)[::-1]
        for i in idx:
            if cnt > target and h_bin[i] == 1.0:
                h_bin[i] = 0.0
                cnt -= 1
    elif cnt < target:
        idx = np.argsort(h)
        for i in idx:
            if cnt < target and h_bin[i] == 0.0:
                h_bin[i] = 1.0
                cnt += 1
    return h_bin

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    deadline_gd = t0 + budget_s * 0.75
    deadline_ls = t0 + budget_s * 0.92
    deadline_end = t0 + budget_s * 0.97

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
        n_init = len(h_init)
    else:
        n_init = 80
        h_init = np.zeros(n_init)
        h_init[:n_init // 2] = 1.0

    best_h = h_init.copy()
    best_c5 = c5_value(h_init, n_init)
    best_n = n_init

    sizes = set()
    sizes.add(n_init)
    for s in [40, 60, 96, 120, 144, 160, 200]:
        if 16 <= s <= 300 and s % 2 == 0:
            sizes.add(s)
    size_list = sorted(sizes)

    for n in size_list:
        if time.time() > deadline_end:
            break
        target = n / 2.0

        if n == n_init:
            h = h_init.copy()
        else:
            x_old = np.linspace(0.0, 2.0, n_init)
            x_new = np.linspace(0.0, 2.0, n)
            h = project_constraint(np.interp(x_new, x_old, h_init), target)

        h, c5_gd = gradient_descent(h, n, target, 150, deadline_gd)
        if c5_gd < best_c5:
            best_c5 = c5_gd
            best_h = h.copy()
            best_n = n

        h_bin = make_binary(h, n)
        c5_bin = c5_value(h_bin, n)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
            best_n = n

        h_ls, c5_ls = local_search_binary(h_bin, n, deadline_ls, 4000)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()
            best_n = n

    # Try a second round with different seed on best size
    if time.time() < deadline_ls:
        h2 = best_h.copy()
        # Perturb
        mask = np.random.rand(best_n) < 0.1
        h2[mask] = np.clip(h2[mask] + np.random.randn(int(np.sum(mask))) * 0.3, 0.0, 1.0)
        h2 = project_constraint(h2, best_n / 2.0)
        h2, c5_r2 = gradient_descent(h2, best_n, best_n / 2.0, 100, deadline_ls)
        if c5_r2 < best_c5:
            best_c5 = c5_r2
            best_h = h2.copy()

        h2b = make_binary(h2, best_n)
        h2b, c5_r2b = local_search_binary(h2b, best_n, deadline_ls, 2000)
        if c5_r2b < best_c5:
            best_c5 = c5_r2b
            best_h = h2b.copy()

    final_c5 = c5_value(best_h, best_n)
    return best_h, final_c5, best_n
