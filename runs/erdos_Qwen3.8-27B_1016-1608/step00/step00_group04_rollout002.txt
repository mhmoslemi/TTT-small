For the coordinate descent:

  where a_k and b_k are constants (depending on other h[j] values)

Wait, the sum constraint couples all coordinates. So I can't optimize one coordinate independently without adjusting another.

Solution: optimize pairs (i, j). Fix h[j] = s - h[i] where s = h[i] + h[j] (constant). Then:

- 0 ≤ h[i] ≤ 1, 0 ≤ s - h[i] ≤ 1

The objective as a function of h[i]:

This is still complex. Let me just use the simpler approach:

For each pair (i, j), try swapping: h[i] += δ, h[j] -= δ for small δ.
Evaluate the new C₅. Accept if better.

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def binary_local_search(h_bin, max_passes=500, time_fn=None, budget_fn=None):
    n = len(h_bin)
    best = h_bin.copy()
    best_c5 = compute_c5(best)
    
    for _ in range(max_passes):
        if budget_fn and budget_fn() > 120:
            break
        improved = False
        for i in range(n):
            if best[i] != 1:
                continue
            for j in range(n):
                if best[j] != 0:
                    continue
                best[i], best[j] = 0, 1
                c5 = compute_c5(best)
                if c5 < best_c5 - 1e-15:
                    best_c5 = c5
                    improved = True
                else:
                    best[i], best[j] = 1, 0
        if not improved:
            break
    return best, best_c5

def continuous_refine(h, max_iters=1000, time_fn=None, budget_fn=None):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    deltas = [0.01, 0.05, 0.1, 0.2, 0.3]
    
    for it in range(max_iters):
        if budget_fn and budget_fn() > 60:
            break
        improved = False
        for i in range(n):
            for j in range(n):
                if i == j:
                    continue
                for d in deltas:
                    if best[i] + d > 1 or best[j] - d < 0:
                        continue
                    best[i] += d
                    best[j] -= d
                    c5 = compute_c5(best)
                    if c5 < best_c5 - 1e-15:
                        best_c5 = c5
                        improved = True
                    else:
                        best[i] -= d
                        best[j] += d
                        break
                # Reverse direction
                for d in deltas:
                    if best[j] + d > 1 or best[i] - d < 0:
                        continue
                    best[j] += d
                    best[i] -= d
                    c5 = compute_c5(best)
                    if c5 < best_c5 - 1e-15:
                        best_c5 = c5
                        improved = True
                    else:
                        best[j] -= d
                        best[i] += d
                        break
        if not improved:
            break
    return best, best_c5

    start = time.time()
    
    def elapsed():

    else:

        h0[:21] = 1.0

    best_h = h0.copy().clip(0, 1)
    best_c5 = compute_c5(best_h)
    
    # Phase 1: Binary local search
    h_bin = (h0 >= 0.5).astype(float)
    # Fix count
    n_ones = int(np.sum(h_bin))
    target_ones = n // 2
    if n_ones > target_ones:
        ones_idx = np.where(h_bin == 1)[0]
        for idx in ones_idx[target_ones:]:
            h_bin[idx] = 0
    elif n_ones < target_ones:
        zeros_idx = np.where(h_bin == 0)[0]
        for idx in zeros_idx[:target_ones - n_ones]:
            h_bin[idx] = 1
    
    h_bin, c5_bin = binary_local_search(h_bin, budget_fn=remaining)
    if c5_bin < best_c5:
        best_h, best_c5 = h_bin.copy(), c5_bin
    
    # Phase 2: Continuous refinement
    h_cont, c5_cont = continuous_refine(best_h, budget_fn=remaining)
    if c5_cont < best_c5:
        best_h, best_c5 = h_cont, c5_cont
    
    # Phase 3: More binary searches from perturbed starting points
    for retry in range(20):
        if remaining() < 60:
            break
        # Perturb the binary solution
        h_bin2 = (best_h >= 0.5).astype(float)
        # Random swap
        ones_idx = np.where(h_bin2 == 1)[0]
        zeros_idx = np.where(h_bin2 == 0)[0]
        i = ones_idx[rng.randint(len(ones_idx))]
        j = zeros_idx[rng.randint(len(zeros_idx))]
        h_bin2[i], h_bin2[j] = 0, 1
        h_bin2, c5_2 = binary_local_search(h_bin2, max_passes=100, budget_fn=remaining)
        if c5_2 < best_c5:
            best_h, best_c5 = h_bin2.copy(), c5_2
    
    # Phase 4: Final continuous refinement
    if remaining() > 30:
        h_cont, c5_cont = continuous_refine(best_h, max_iters=500, budget_fn=remaining)
        if c5_cont < best_c5:
            best_h, best_c5 = h_cont, c5_cont
    
    return best_h, best_c5, n

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def binary_search(h_bin, max_passes=2000, deadline=None):
    n = len(h_bin)
    best = h_bin.copy()
    best_c5 = compute_c5(best)
    for _ in range(max_passes):
        if deadline and time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if best[i] != 1.0:
                continue
            for j in range(n):
                if best[j] != 0.0:
                    continue
                best[i] = 0.0
                best[j] = 1.0
                c5 = compute_c5(best)
                if c5 < best_c5 - 1e-14:
                    best_c5 = c5
                    improved = True
                else:
                    best[i] = 1.0
                    best[j] = 0.0
        if not improved:
            break
    return best, best_c5

def binary_search_2phase(h_bin, max_passes=500, deadline=None):
    """Try swaps in two orderings for better escape."""
    n = len(h_bin)
    best = h_bin.copy()
    best_c5 = compute_c5(best)
    for p in range(max_passes):
        if deadline and time.time() > deadline:
            break
        improved = False
        order = range(n) if p % 2 == 0 else range(n - 1, -1, -1)
        for i in order:
            if best[i] != 1.0:
                continue
            for j in order:
                if best[j] != 0.0:
                    continue
                best[i] = 0.0
                best[j] = 1.0
                c5 = compute_c5(best)
                if c5 < best_c5 - 1e-14:
                    best_c5 = c5
                    improved = True
                else:
                    best[i] = 1.0
                    best[j] = 0.0
        if not improved:
            break

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
