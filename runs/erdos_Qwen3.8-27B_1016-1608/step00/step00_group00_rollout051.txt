import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1 - h, mode='full')
    g = corr_raw * dx
    m = np.max(g)
    exp_g = np.exp(g - m)
    s = exp_g.sum()
    softmax = exp_g / s
    val = (m + np.log(s)) / alpha
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        mk = softmax[k]
        if mk < 1e-14:
            continue
        mshift = k - (n - 1)
        j1s = max(0, -mshift)
        j1e = min(n, n - mshift)
        if j1e > j1s:
            grad[j1s:j1e] += mk * dx * (1.0 - h[j1s + mshift:j1e + mshift])
        j2s = max(0, mshift)
        j2e = min(n, n + mshift)
        if j2e > j2s:
            grad[j2s:j2e] -= mk * dx * h[j2s - mshift:j2e - mshift]
    return val, grad

def project_onto_constraint(h, target):
    for _ in range(50):
        diff = h.sum() - target
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() == 0:
            h += diff / len(h)
        else:
            h[mask] -= diff / mask.sum()
        h = np.clip(h, 0, 1)
    return h

def binary_local_search(h_bin, max_iters=2000, timeout=lambda: False):
    n = len(h_bin)
    target = n // 2
    ones = np.where(h_bin == 1)[0]
    zeros = np.where(h_bin == 0)[0]
    best_c5 = compute_c5(h_bin)
    for it in range(max_iters):
        if timeout():
            break
        best_delta = 0
        best_pair = None
        for i in range(min(len(ones), 12)):
            for j in range(min(len(zeros), 12)):
                oi, zi = ones[i], zeros[j]
                if oi == zi:
                    continue
                h_bin[oi] = 0
                h_bin[zi] = 1
                c5 = compute_c5(h_bin)
                delta = c5 - best_c5

                    best_delta = delta

                h_bin[oi] = 1
                h_bin[zi] = 0
        if best_pair is not None:
            oi, zi = best_pair
            h_bin[oi] = 0
            h_bin[zi] = 1
            ones = np.where(h_bin == 1)[0]
            zeros = np.where(h_bin == 0)[0]
            best_c5 = compute_c5(h_bin)
        else:
            break
    return h_bin, best_c5

def make_thue_morse(n):
    if n <= 0:
        return np.array([], dtype=float)
    t = np.array([0])
    while len(t) < n:
        t = np.concatenate([t, 1 - t])
    tm = t[:n].astype(float)
    # Adjust to have exactly n//2 ones
    num_ones = int(tm.sum())
    if num_ones > n // 2:
        idx = np.where(tm == 1)[0]
        rng = np.random.RandomState(0)

            tm[idx[rng.randint(len(idx))]] = 0
    elif num_ones < n // 2:
        idx = np.where(tm == 0)[0]
        rng = np.random.RandomState(0)
        for _ in range(n // 2 - num_ones):
            tm[idx[rng.randint(len(idx))]] = 1
    return tm

def slsqp_optimize(h_init, n, alpha_start=5, max_iter=300, timeout_check=None):
    target = n / 2.0
    constraints = [{'type': 'eq', 'fun': lambda h: h.sum() - target,
                    'jac': lambda h: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    h_cur = h_init.copy()
    best_c5 = compute_c5(h_cur)
    for alpha in [alpha_start, alpha_start * 4, alpha_start * 16, alpha_start * 64, alpha_start * 256]:
        if timeout_check is not None and timeout_check():
            break
        def obj(hv):
            return smooth_obj_grad(hv, alpha)[0]
        def grad(hv):
            return smooth_obj_grad(hv, alpha)[1]
        try:
            res = minimize(obj, h_cur, jac=grad, method='SLSQP',
                           bounds=bounds, constraints=constraints,
                           options={'maxiter': max(30, max_iter // 5), 'ftol': 1e-12})
            h_cur = np.clip(res.x, 0, 1)
            h_cur = project_onto_constraint(h_cur, target)
            c5 = compute_c5(h_cur)
            if c5 < best_c5:
                best_c5 = c5
        except Exception:
            pass
    return h_cur, best_c5

def projected_gradient(h_init, n, alpha=50, lr=0.01, n_iter=3000, timeout_check=None):
    target = n / 2.0
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)

    c5 = compute_c5(h)

    return best_h, best_c5
