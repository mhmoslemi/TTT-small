```python
    #   Delta: -h[j+s]
    # delta_corr_j[k] = +(1-h[j-s_k])*I[0<=j-s_k<n] - h[j+s_k]*I[0<=j+s_k<n]
    
    # Combined:
    # delta_corr[k] = -(1-h[i-s_k])*Ii1 + h[i+s_k]*Ii2 + (1-h[j-s_k])*Ij1 - h[j+s_k]*Ij2
    
    # Vectorize:
    # i-s ranges: for k=0..2n-2, s = k-(n-1), so i-s = i-k+(n-1) = (i+n-1)-k
    # i+s ranges: i+k-(n-1)
    
    k_arr = np.arange(2*n-1)
    i_minus_s = i + (n-1) - k_arr  # i - s_k
    i_plus_s = i + k_arr - (n-1)   # i + s_k
    j_minus_s = j + (n-1) - k_arr
    j_plus_s = j + k_arr - (n-1)
    
    valid_im = (i_minus_s >= 0) & (i_minus_s < n)
    valid_ip = (i_plus_s >= 0) & (i_plus_s < n)
    valid_jm = (j_minus_s >= 0) & (j_minus_s < n)
    valid_jp = (j_plus_s >= 0) & (j_plus_s < n)
    
    delta = np.zeros(2*n-1)
    # -(1-h[i-s]) for valid
    delta[valid_im] -= (1 - h[i_minus_s[valid_im]])
    # +h[i+s] for valid
    delta[valid_ip] += h[i_plus_s[valid_ip]]
    # +(1-h[j-s]) for valid
    delta[valid_jm] += (1 - h[j_minus_s[valid_jm]])
    # -h[j+s] for valid
    delta[valid_jp] -= h[j_plus_s[valid_jp]]
    
    return corr + delta
```

This is O(n) per swap evaluation (all vectorized numpy operations on arrays of size 2n-1). For n=70, that's 139 elements, very fast.

```python
def fast_swap_delta(h, i, j, n):
    """Return delta correlation vector for swapping h[i]1->0, h[j]0->1."""
    m = 2 * n - 1
    off = n - 1
    k_arr = np.arange(m)
    ims = i + off - k_arr
    ips = i + k_arr - off
    jms = j + off - k_arr
    jps = j + k_arr - off

    vip = (ips >= 0) & (ips < n)
    vjm = (jms >= 0) & (jms < n)

    delta = np.zeros(m)
    if np.any(vim):
        delta[vim] -= (1.0 - h[ims[vim]])
    if np.any(vip):
        delta[vip] += h[ips[vip]]
    if np.any(vjm):
        delta[vjm] += (1.0 - h[jms[vjm]])
    if np.any(vjp):
        delta[vjp] -= h[jps[vjp]]

    target = n // 2

    if s > target:

    elif s < target:
        zi = np.where(h == 0)[0]
        for x in zi[:target - s]:

def grad_step(h, n, lr=0.01):
    """One projected gradient step on continuous h."""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    max_val = np.max(corr)
    max_idx = np.where(corr >= max_val - 1e-12)[0]
    grad = np.zeros(n)
    s_range = np.arange(2 * n - 1) - (n - 1)
    for k in max_idx:
        s = s_range[k]
        j = np.arange(n)
        jm = j - s
        jp = j + s
        vm = (jm >= 0) & (jm < n)
        vp = (jp >= 0) & (jp < n)
        t1 = np.zeros(n)
        t1[vm] = 1.0 - h[jm[vm]]
        t2 = np.zeros(n)
        t2[vp] = h[jp[vp]]
        grad += t1 - t2
    grad *= dx / len(max_idx)
    h_new = h - lr * grad
    # Project: clip then adjust sum
    h_new = np.clip(h_new, 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - h_new.sum()
        if abs(diff) < 1e-12:
            break
        h_new += diff / n
        h_new = np.clip(h_new, 0.0, 1.0)
    return h_new

def continuous_refine(h_in, n, rng, max_time=60.0):
    dx = 2.0 / n
    h = np.array(h_in, dtype=np.float64)
    h = np.clip(h, 0.0, 1.0)
```
