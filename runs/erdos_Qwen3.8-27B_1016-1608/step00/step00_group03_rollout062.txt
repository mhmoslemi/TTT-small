            bh[i] = 1.0 if (i * 37) % n < n // 2 else 0.0
        bh = _fix_sum(bh, tgt)
    bc = _eval_c5(bh, dx)

    # Phase 1: SLSQP with increasing smoothness
    for eta in [50, 100, 200, 500, 1000, 2000]:
        if budget_s - (time.time() - t0) < 10:
            break
        x = _opt(bh, n, dx, tgt, eta, 600)
        x = _fix_sum(x, tgt)
        c = _eval_c5(x, dx)
        if c < bc:
            bh, bc = x.copy(), c

    # Phase 2: Line search polish
    if budget_s - (time.time() - t0) > 20:
        bh2, bc2 = _line_search(bh, n, dx, tgt)
        if bc2 < bc:
            bh, bc = bh2, bc2

    # Phase 3: Simulated annealing on binary
    if budget_s - (time.time() - t0) > 80:
        sa_dur = min(60.0, (budget_s - (time.time() - t0)) * 0.25)
        sh, sc = _sa(bh, n, dx, sa_dur)
        if sc < bc:
            bh, bc = sh.copy(), sc
        # SLSQP from SA result
        if budget_s - (time.time() - t0) > 30:
            x = _opt(bh, n, dx, tgt, 500, 500)
            x = _fix_sum(x, tgt)
            c = _eval_c5(x, dx)
            if c < bc:
                bh, bc = x.copy(), c
            bh2, bc2 = _line_search(bh, n, dx, tgt)
            if bc2 < bc:
                bh, bc = bh2, bc2

    # Phase 4: Multiple restarts
    for r in range(100):
        if budget_s - (time.time() - t0) < 8:
            break
        sc_ = 0.02 + 0.05 * (r % 5) * 0.2
        ht = _fix_sum(bh + np.random.randn(n) * sc_, tgt)
        x = _opt(ht, n, dx, tgt, 300, 300)
        x = _fix_sum(x, tgt)
        c = _eval_c5(x, dx)
        if c < bc:
            bh, bc = x.copy(), c
            bh2, bc2 = _line_search(bh, n, dx, tgt)
            if bc2 < bc:
                bh, bc = bh2, bc2

    best_h, best_c5 = bh.copy(), bc

    # Phase 5: Try n=100 if time allows
    if budget_s - (time.time() - t0) > 180:
        n2, dx2, t2 = 100, 0.02, 50.0
        h2 = np.interp(np.linspace(0, 2, 100, endpoint=False),
                       np.linspace(0, 2, n, endpoint=False), bh)
        h2 = _fix_sum(h2, t2)
        b2 = _eval_c5(h2, dx2)
        for eta in [100, 300, 800]:
            if budget_s - (time.time() - t0) < 30:
                break
            x = _opt(h2, n2, dx2, t2, eta, 500)
            x = _fix_sum(x, t2)
            c = _eval_c5(x, dx2)
            if c < b2:
                h2, b2 = x.copy(), c
        if budget_s - (time.time() - t0) > 20:
            h2, b2 = _line_search(h2, n2, dx2, t2)
        if b2 < best_c5:
            best_h, best_c5 = h2.copy(), b2

    return best_h, best_c5, len(best_h)
