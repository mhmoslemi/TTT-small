import numpy as np
import time
from scipy.optimize import minimize

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _fix_sum(h, n):
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-9:
        return h
    h = h.copy()
    diff = target - s
    mask = (h > 0.005) & (h < 0.995)
    if mask.sum() > 0:
        h[mask] += diff / mask.sum()
    else:
        h += diff / n
    return np.clip(h, 0.0, 1.0)

def _powell_opt(h0, n, dx, t_end, rng):
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return _c5(x, dx)
    best = h0.copy()
    best_val = obj(best)
    for attempt in range(4):
        if time.time() > t_end:
            break
        x0 = best + rng.randn(n) * (0.08 if attempt == 0 else 0.04)
        x0 = np.clip(x0, 0, 1)
        x0 = _fix_sum(x0, n)
        res = minimize(obj, x0, method='Powell', bounds=bounds,
                       options={'maxiter': 300, 'xtol': 1e-10, 'ftol': 1e-13, 'nit': 8})
        if res.x is not None and res.fun < best_val:
            h_new = _fix_sum(np.clip(res.x, 0, 1), n)
            v = _c5(h_new, dx)
            if v < best_val:
                best = h_new
                best_val = v
    return best, best_val

    # Annealing
    if time.time() < t_end:
        h_a, c5_a = _anneal_search(best_h, n_p, dx_p, t_end, rng, t0)
        if c5_a < best_c5:
            best_h, best_c5, best_n = h_a, c5_a, n_p
        # Local from anneal
        best_h, best_c5 = _local_search(best_h, n_p, dx_p, t_end, rng)

    # Try other n values
    for n_try in [33, 53, 37]:
        if time.time() > t_end:
            break
        dx_t = 2.0 / n_try
        h_try = _make_initial(n_try, rng)
        # Quick Powell
        h_try, c5_try = _powell_opt(h_try, n_try, dx_t, t_end, rng)
        # Local
        h_try, c5_try = _local_search(h_try, n_try, dx_t, t_end, rng)
        if c5_try < best_c5:
            best_h, best_c5, best_n = h_try, c5_try, n_try

    # Final local polish on best
    best_h, best_c5 = _local_search(best_h, best_n, 2.0 / best_n, t_end, rng)

    # Ensure feasibility
    best_h = _fix_sum(best_h, best_n)
    best_c5 = _c5(best_h, 2.0 / best_n)

    return best_h.tolist(), best_c5, best_n
