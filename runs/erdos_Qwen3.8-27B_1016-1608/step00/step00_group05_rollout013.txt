import numpy as np
import time

def _project(h, tsum):
    h = np.clip(h, 0.0, 1.0)
    d = tsum - h.sum()
    if d > 1e-10:
        room = np.maximum(1.0 - h, 0.0)
        if room.sum() > 1e-10:
            h = np.clip(h + d * room / room.sum(), 0.0, 1.0)
    elif d < -1e-10:
        space = np.maximum(h, 0.0)
        if space.sum() > 1e-10:
            h = np.clip(h + d * space / space.sum(), 0.0, 1.0)
    return h

def _corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _bin_search(h, dx, tsum, rng, t_end, margin):
    n = len(h)
    h = h.copy().astype(float)
    c5 = _c5(h, dx)
    best_c5 = c5
    best_h = h.copy()
    stall = 0
    while time.time() < t_end - margin:
        ones = list(np.where(h > 0.5)[0])
        zeros = list(np.where(h < 0.5)[0])
        if not ones or not zeros:
            break
        rng.shuffle(ones)
        rng.shuffle(zeros)
        improved = False
        for i in ones:
            if time.time() >= t_end - margin:
                break
            for j in zeros:
                if time.time() >= t_end - margin:
                    break
                h[i], h[j] = h[j], h[i]
                nc = _c5(h, dx)
                if nc < c5 - 1e-14:
                    c5 = nc
                    improved = True
                    if nc < best_c5:
                        best_c5 = nc
                        best_h = h.copy()
                else:
                    h[i], h[j] = h[j], h[i]
        if not improved:
            stall += 1
            if stall > 3:
                # Kick: flip 2-3 random pairs
                for _ in range(rng.integers(1, 4)):
                    o = list(np.where(h > 0.5)[0])
                    z = list(np.where(h < 0.5)[0])
                    if o and z:
                        a = int(rng.choice(o))
                        b = int(rng.choice(z))
                        h[a], h[b] = h[b], h[a]
                h_best_try = best_h.copy()
                c5 = _c5(h_best_try, dx)
                h = h.copy()
                # Re-evaluate current
                nc = _c5(h, dx)
                if nc < best_c5:
                    best_c5 = nc
                    best_h = h.copy()
                c5 = nc
                stall = 0
            # After kick, do random swaps for a while
            for _ in range(300):
                if time.time() >= t_end - margin:
                    break
                o = list(np.where(h > 0.5)[0])
                z = list(np.where(h < 0.5)[0])
                if not o or not z:
                    break
                a = int(rng.choice(o))
                b = int(rng.choice(z))
                h[a], h[b] = h[b], h[a]
                nc = _c5(h, dx)
                if nc <= c5:
                    c5 = nc
                    if nc < best_c5:
                        best_c5 = nc
                        best_h = h.copy()
                else:
                    h[a], h[b] = h[b], h[a]
        else:
            stall = 0
    return best_h, best_c5

def _cont_search(h, dx, tsum, rng, t_end, margin):
    n = len(h)
    h = h.copy()
    c5 = _c5(h, dx)
    best_c5 = c5
    best_h = h.copy()
    lrs = [0.01, 0.05, 0.1, 0.2]
    lr_i = 0
    stall = 0
    while time.time() < t_end - margin:
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        delta = lrs[lr_i] * (1.0 if rng.rand() < 0.5 else -1.0)
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        h_new = _project(h_new, tsum)
        nc = _c5(h_new, dx)
        if nc < c5 - 1e-14:
            h = h_new
            c5 = nc
            stall = 0
            lr_i = min(lr_i + 1, len(lrs) - 1)
            if nc < best_c5:
                best_c5 = nc
                best_h = h.copy()
        else:
            stall += 1
            if stall > 500:
                lr_i = max(0, lr_i - 1)
                stall = 0
    return best_h, best_c5

def _sa_search(h, dx, tsum, rng, t_end, margin):
    n = len(h)
    h = h.copy()
    c5 = _c5(h, dx)
    best_c5 = c5
    best_h = h.copy()

    T_min = 0.0005

                h = h.copy()
                h[a] -= d
                h[b] += d

        else:

            d = 0.15 * rng.randn()
            h = h.copy()
            h[a] -= d

            h = _project(h, tsum)
        nc = _c5(h, dx)
        dc = nc - c5

                best_c5 = nc
                best_h = h.copy()
        else:
            h = h_prev
            c5 = c5_prev

    return best_h, best_c5
