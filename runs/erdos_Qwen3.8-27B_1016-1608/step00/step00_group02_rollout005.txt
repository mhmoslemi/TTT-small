def project_h(h, n):
    """Project h to [0,1]^n with sum = n/2."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 0.005) & (h < 0.995)
        if free.sum() == 0:
            free = (h > 0.0) & (h < 1.0)
        if free.sum() == 0:
            free = np.ones(n, dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def eval_c5(h, n):
    """Compute C5 = max(correlate(h, 1-h, full)) * dx."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)

    return np.interp(x_new, x_old, h_src)
