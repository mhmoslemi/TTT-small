    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def subgrad_for_k(h, k):
    n = len(h)
    i_lo = max(0, k - n + 1)
    i_hi = min(n - 1, k - 1)
    if i_lo > i_hi:
        return np.zeros(n)
    grad = np.zeros(n)
    i_vals = np.arange(i_lo, i_hi + 1)
    j_vals = i_vals - k + n - 1
    grad[i_vals] += (1.0 - h[j_vals])
    # For the second part: for each j in j_vals, h[i_vals] contributes -h[i_vals] to grad[j]
    # grad[j_vals] -= h[i_vals]
    # But we need to be careful about overlaps
    np.add.at(grad, j_vals, -h[i_vals])
    return grad

def smooth_max_grad(h, beta=30.0):
    """Gradient of soft-max of corr * dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    weights = np.exp(beta * (corr - m))
    weights /= weights.sum()
    grad = np.zeros(n)
    for k in range(len(corr)):
        if weights[k] > 1e-10:
            grad += weights[k] * subgrad_for_k(h, k)
    return dx * grad

def optimize_subgrad(h0, n_iters=2000, lr_init=0.01, budget_s=60):

from scipy.optimize import minimize

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def subgrad_for_k(h, k):
    n = len(h)
    i_lo = max(0, k - n + 1)
    i_hi = min(n - 1, k)
    if i_lo > i_hi:
        return np.zeros(n)
    grad = np.zeros(n)
    i_vals = np.arange(i_lo, i_hi + 1)
    j_vals = i_vals - k + n - 1
    grad[i_vals] += (1.0 - h[j_vals])
    np.add.at(grad, j_vals, -h[i_vals])
    return grad

def smooth_max_obj(x, n, beta):
    dx = 2.0 / n
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    return dx * (m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta)

def smooth_max_grad(x, n, beta):

    h = np.clip(x, 0.0, 1.0)

    m = np.max(corr)

    return dx * grad
