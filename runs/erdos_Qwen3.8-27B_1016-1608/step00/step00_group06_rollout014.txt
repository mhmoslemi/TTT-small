def corr_gradient(h, m):
    n = len(h)
    # corr = sum_{i: valid} h[i] * (1 - h[i+m])
    # valid: 0 <= i < n and 0 <= i+m < n
    i_start = max(0, -m)
    i_end = min(n, n - m)  # exclusive
    # valid i range: [i_start, i_end)
    
    grad = np.zeros(n)
    h_valid = h[i_start:i_end]
    hm_valid = h[i_start+m:i_end+m]
    
    # First part: sum h[i] -> grad[i] += 1 for i in [i_start, i_end)
    grad[i_start:i_end] += 1.0
    
    # Second part: -sum h[i]*h[i+m]
    # d/dh[j] of -h[i]*h[i+m]:
    # if j = i (i in valid range): -h[j+m]
    # if j = i+m (i in valid range): -h[j-m]
    grad[i_start:i_end] -= hm_valid  # from j = i
    grad[i_start+m:i_end+m] -= h_valid  # from j = i+m
    
    return grad

            best_h, best_c5 = hr, cr
            no_improve = 0
            sigma = max(0.01, sigma * 0.8)
        else:
            no_improve += 1
            if no_improve == 15:
                sigma = min(0.15, sigma * 1.5)
                no_improve = 0

    # Try different n values (interpolate best solution)
    cur_n = len(best_h)
    for nv in [51, 101, 151, 201, 301, 401]:
        if time.time() - t0 > budget_s - 60:
            break
        xo = np.linspace(0.0, 2.0, cur_n)
        xn = np.linspace(0.0, 2.0, nv)
        h_init = np.interp(xn, xo, best_h)
        h_init = project_feasible(h_init, nv / 2.0)
        hr, cr = slsqp_optimize(h_init, nv, t0, budget_s)
        if hr is not None and cr < best_c5:
            best_h, best_c5 = hr, cr
            cur_n = nv
        # Also try a smoothed version at this n
        if time.time() - t0 > budget_s - 60:
            break
        h_init2 = np.interp(xn, xo, best_h)
        h_init2 = 0.5 * h_init2 + 0.5 * (1.0 - h_init2)
        h_init2 = project_feasible(h_init2, nv / 2.0)
        hr2, cr2 = slsqp_optimize(h_init2, nv, t0, budget_s)
        if hr2 is not None and cr2 < best_c5:
            best_h, best_c5 = hr2, cr2
            cur_n = nv

    # Final polish
    if time.time() - t0 < budget_s - 15:
        hr, cr = slsqp_optimize(best_h.copy(), len(best_h), t0, budget_s)
        if hr is not None and cr < best_c5:
            best_h, best_c5 = hr, cr

    # Verify
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
