def project_onto_feasible(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}."""
    target = n / 2.0
    # Simple projection: clip and adjust
    h = np.clip(h, 0, 1)
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    # Distribute the difference
    # If diff > 0, we need to increase some values
    # If diff < 0, we need to decrease some values
    if diff > 0:
        # Can only increase values < 1
        mask = h < 1.0
        n_avail = np.sum(mask)
        if n_avail == 0:
            return h
        h[mask] += diff / n_avail
    else:
        mask = h > 0.0
        n_avail = np.sum(mask)
        if n_avail == 0:
            return h
        h[mask] += diff / n_avail
    h = np.clip(h, 0, 1)
    # Exact fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argmin(h)
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = np.argmax(h)
            h[idx] = max(0.0, h[idx] + diff)

    """Project g onto {h: 0<=h<=1, sum(h)=n/2}."""
    target = n / 2.0
    
    def sum_at_lambda(lam):
        return np.sum(np.clip(g - lam, 0.0, 1.0))
    
    # Bisection for lambda
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = sum_at_lambda(mid)
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(g - lam, 0.0, 1.0)
    # Fix any tiny numerical errors
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # Adjust one element
        if diff > 0:
            idx = np.argmin(h)
            h[idx] = min(1.0, h[idx] + diff)
        else:

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(c))

def project_feasible(g, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(g - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] = float(np.clip(h[idx] + diff, 0.0, 1.0))
    return h

def make_ripple(n, freq, phase, amp):
    x = np.arange(n, dtype=float)
    return 0.5 + amp * np.sin(2.0 * np.pi * freq * x / n + phase)

def annealing_polish(h, n, deadline, rng, T0=0.004):
