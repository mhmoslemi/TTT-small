        lag = int(np.argmax(corr))

        def _obj1(h):
            gg = 1.0 - h
            cc = np.correlate(h, gg, mode="full")
            return cc[lag] * dx

        try:
            res = minimize(_obj1, best_h, method='SLSQP', bounds=bounds,
                           constraints=eqcon, options={'maxiter': 300, 'ftol': 1e-15})
            if res.fun < best_val:
                v = c5_val(res.x, n)
                if v < best_val:
                    best_val = v
                    best_h = project_sum(res.x, target, n)
                    if np.max(np.abs(res.x - best_h)) > 1e-6:
                        best_val = c5_val(best_h, n)
        except Exception:
            pass

    # Phase 2: Nelder-Mead from perturbed starts
    n_starts = 60
    scales = [0.05, 0.1, 0.15, 0.2, 0.3]
    for si in range(n_starts):
        if elapsed() > budget_s - 100:
            break
        sc = scales[si % len(scales)]
        h_r = best_h + np.random.normal(0, sc, n)
        h_r = project_sum(h_r, target, n)
        v_r = c5_val(h_r, n)

        def _obj_nm(h):
            return c5_val(h, n)

        try:
            res = minimize(_obj_nm, h_r, method='Nelder-Mead',
                           options={'maxiter': 8000, 'xatol': 1e-9, 'fatol': 1e-12,
                                    'adaptive': True})
            h_r = project_sum(res.x, target, n)
            v_r = c5_val(h_r, n)
        except Exception:
            pass

        if v_r < best_val:
            best_val = v_r
            best_h = h_r.copy()

        # Short active-lag polish on improved point
        for it in range(80):
            if elapsed() > budget_s - 100:
                break
            g = 1.0 - h_r
            corr = np.correlate(h_r, g, mode="full")
            lag = int(np.argmax(corr))
            def _obj2(h):
                gg = 1.0 - h
                cc = np.correlate(h, gg, mode="full")
                return cc[lag] * dx
            try:
                res2 = minimize(_obj2, h_r, method='SLSQP', bounds=bounds,
                               constraints=eqcon, options={'maxiter': 150, 'ftol': 1e-14})
                v2 = c5_val(res2.x, n)
                if v2 < v_r:
                    v_r = v2
                    h_r = project_sum(res2.x, target, n)
                    if v_r < best_val:
                        best_val = v_r
                        best_h = h_r.copy()
                else:
                    break
            except Exception:
                break

    # Phase 3: Pairwise coordinate descent (fine)
    remaining = budget_s - elapsed()
    if remaining > 3:
        step_sizes = [0.05, 0.02, 0.01, 0.005, 0.002]
        for ss in step_sizes:
            if elapsed() > budget_s - 2:
                break
            for _ in range(2000):
                if elapsed() > budget_s - 2:
                    break
                moved = False
                for i in range(n):
                    if elapsed() > budget_s - 2:
                        break
                    for j in ((i + 1) % n, (i + 2) % n, (i + n // 2) % n):
                        if j == i:
                            continue
                        for delta in (ss, -ss):
                            hi = best_h[i] + delta
                            hj = best_h[j] - delta
                            if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                                h_try = best_h.copy()
                                h_try[i] = hi
                                h_try[j] = hj
                                v = c5_val(h_try, n)
                                if v < best_val - 1e-14:
                                    best_val = v
                                    best_h = h_try
                                    moved = True
                                    break
                    if moved:
                        break
                if not moved:
                    break

    # Final check
    best_h = project_sum(best_h, target, n)
    best_val = c5_val(best_h, n)
    return list(best_h), best_val, n
