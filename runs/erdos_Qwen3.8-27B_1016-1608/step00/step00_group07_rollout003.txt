import time

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def compute_autocorr(h, n):
    dx = 2.0 / n
    # A_k = sum_i h[i]*h[i+k]*dx
    corr = np.correlate(h, h, mode="full") * dx
    # The middle element (lag 0) is sum h[i]^2 * dx

    h = h.copy()
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Fix sum
    for _ in range(20):
        diff = target - h.sum()
        if abs(diff) < 1e-12:

        if diff > 0:
            mask = h < 1.0
            if not mask.any():

            h = np.clip(h, 0.0, 1.0)
        else:

            if not mask.any():

        idx = np.argsort(np.abs(h - 0.5))[:50]
        h[idx] = np.clip(h[idx] + diff / len(idx), 0, 1)

            if not mask.any():
                break
            h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        idx = np.argsort(np.abs(h - 0.5))[:min(50, n)]
        h[idx] = np.clip(h[idx] + diff / len(idx), 0, 1)
    return h

def subgradient_ascent(h, n, n_steps, step_init, step_decay):
    dx = 2.0 / n
    step = step_init
    for _ in range(n_steps):
        ac = compute_autocorr(h, n)
        # Skip lag 0 and find min over lags 1..n-1
        k_star = int(np.argmin(ac[1:len(ac)-1])) + 1
        # Gradient of A_{k*} w.r.t. h
        g = np.zeros(n)
        g[k_star:] += h[:n-k_star]
        g[:n-k_star] += h[k_star:]
        h_new = h + step * g
        h_new = project_feasible(h_new, n)
        if compute_c5(h_new, n) < compute_c5(h, n) + 0.01:
            h = h_new
        step *= step_decay
    return h

def block_sa(h, n, n_iters, t_start_frac, t_end_frac, rng):
    dx = 2.0 / n
    best_c5 = compute_c5(h, n)
    best_h = h.copy()
    temp = 0.01
    for it in range(n_iters):
        blen = rng.randint(1, min(6, n // 5))
        start = rng.randint(0, n - blen)
        h_new = h.copy()
        h_new[start:start+blen] += rng.normal(0, 0.08, blen)
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new, n)
        if c5_new < best_c5 or rng.random() < np.exp(-(c5_new - best_c5) / max(temp, 1e-9)):
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        if it % 200 == 0:
            temp = t_start_frac + (t_end_frac - t_start_frac) * (it / max(n_iters, 1))
    return best_h, best_c5

def refine_with_mean(h, n, n_outer, rng):
    """Alternate between optimizing min autocorr and mean of bottom-5"""
    dx = 2.0 / n
    c5 = compute_c5(h, n)
    best_h = h.copy()
    best_c5 = c5
    for outer in range(n_outer):
        ac = compute_autocorr(h, n)
        # Use average of bottom 3 as surrogate
        sorted_ac = np.sort(ac[1:len(ac)-1])
        k_bottom = np.argsort(ac[1:len(ac)-1])[0] + 1
        g = np.zeros(n)
        g[k_bottom:] += h[:n-k_bottom]
        g[:n-k_bottom] += h[k_bottom:]
        h_new = project_feasible(h + 0.03 * g, n)
        c5_new = compute_c5(h_new, n)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        # Random perturbation to escape
        if outer % 5 == 4:
            h = project_feasible(h + rng.normal(0, 0.03, n), n)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    try:
        init = np.array(initial_h_values, dtype=float)
    except Exception:
        init = np.full(80, 0.5)

    n_init = len(init)
    target_sum = n_init / 2.0
    init = init * target_sum / max(init.sum(), 1e-10)
    init = np.clip(init, 0, 1)
    init = project_feasible(init, n_init)

    best_c5 = compute_c5(init, n_init)
    best_h = init.copy()
    best_n = n_init

    n_options = [n_init]
    for nn in [96, 120, 152, 192, 240, 300]:
        if nn not in n_options and nn < 400:
            n_options.append(nn)
    n_options = sorted(set(n_options))

    for idx, n_pts in enumerate(n_options):
        if time.time() - t0 > budget_s - 80:
            break

        if n_pts == n_init:
            h = init.copy()
        else:
            x_old = np.linspace(0, 2, n_init, endpoint=False)
            x_new = np.linspace(0, 2, n_pts, endpoint=False)
            h = np.interp(x_new, x_old, best_h[:n_init] if n_init == len(best_h) else np.interp(np.linspace(0,2,n_init,endpoint=False), np.linspace(0,2,len(best_h),endpoint=False), best_h))
            h = project_feasible(h, n_pts)

        c5 = compute_c5(h, n_pts)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n_pts

        # Time allocation
        remaining = budget_s - 80 - (time.time() - t0)
        time_left_options = len(n_options) - idx
        time_per = max(40, remaining / time_left_options)
        opt_t0 = time.time()

        # Phase 1: Subgradient ascent
        h = subgradient_ascent(h, n_pts, 3000, 0.05, 0.997)
        c5 = compute_c5(h, n_pts)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n_pts
        opt_t0 = time.time()

        # Phase 2: Block SA
        n_sa = int(min(20000, time_per * 8000))
        h, c5 = block_sa(h, n_pts, n_sa, 0.008, 0.001, rng)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n_pts
        opt_t0 = time.time()

        # Phase 3: Refine with alternating
        n_ref = int(min(2000, time_per * 3000))
        h, c5 = refine_with_mean(h, n_pts, n_ref, rng)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n_pts

        # Phase 4: Final local polish - gradient on active k
        if time.time() - t0 < budget_s - 40:
            step = 0.01
            for _ in range(5000):
                if time.time() - t0 > budget_s - 40:
                    break
                ac = compute_autocorr(h, n_pts)
                k_star = int(np.argmin(ac[1:len(ac)-1])) + 1
                g = np.zeros(n_pts)
                g[k_star:] += h[:n_pts-k_star]
                g[:n_pts-k_star] += h[k_star:]
                h_new = project_feasible(h + step * g, n_pts)
                c5_new = compute_c5(h_new, n_pts)
                if c5_new < c5:
                    h = h_new
                    c5 = c5_new
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h.copy()
                        best_n = n_pts
                step *= 0.998

    # Final check
    final_c5 = compute_c5(best_h, best_n)
    if final_c5 != best_c5:
        best_c5 = final_c5
    return best_h, best_c5, best_n
