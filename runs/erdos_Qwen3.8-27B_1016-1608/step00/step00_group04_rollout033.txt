def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    target_sum = n / 2.0

    h = np.clip(h0, 0.0, 1.0)
    h = fix_sum(h, target_sum, rng, n)

    cur = compute_C5(h)
    best_h = h.copy()
    best = cur
    no_imp = 0

    # Phase 1: SA with mixed moves
    T = 0.01
    T_min = 0.0002
    cool = 0.99998
    phase1_end = t0 + budget_s * 0.6

    while time.time() < phase1_end:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            j = (j + 1) % n
        vi, vj = h[i], h[j]
        r = rng.random()

        if r < 0.45:
            # Small transfer
            d = rng.normal(0, 0.12)
            ni = vi + d
            nj = vj - d
            if ni < 0.0 or ni > 1.0 or nj < 0.0 or nj > 1.0:
                continue
            h[i] = ni
            h[j] = nj
        elif r < 0.75:
            # Swap
            h[i] = vj
            h[j] = vi
        elif r < 0.9:
            # Directed transfer (towards more balanced pair)
            d = (vi - vj) * rng.uniform(0.3, 1.0)
            ni = vi - d
            nj = vj + d
            if ni < 0.0 or ni > 1.0 or nj < 0.0 or nj > 1.0:
                continue
            h[i] = ni
            h[j] = nj
        else:
            # Random reassignment of pair
            s = vi + vj
            lo = max(0.0, s - 1.0)
            hi = min(1.0, s)
            if lo >= hi:
                continue
            h[i] = rng.uniform(lo, hi)
            h[j] = s - h[i]

        new = compute_C5(h)
        dS = new - cur
        if dS < 0 or rng.random() < np.exp(-dS / T):
            cur = new
            if cur < best:
                best = cur
                best_h = h.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            h[i], h[j] = vi, vj
            no_imp += 1

        T = max(T * cool, T_min)
        if no_imp > 5000 and rng.random() < 0.005:
            no_imp = 0
            T = 0.015
            for _ in range(rng.randint(2, 6)):
                a, b = rng.choice(n, 2, replace=False)
                h[a], h[b] = h[b], h[a]
            cur = compute_C5(h)
            if cur > best * 1.03:
                h = best_h.copy()
                cur = best

    # Phase 2: Fine-tuning with very small moves
    T = 0.002
    T_min = 0.00005
    cool2 = 0.999995
    no_imp = 0

    while time.time() - t0 < budget_s - 3:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            j = (j + 1) % n
        vi, vj = h[i], h[j]

        # Very small perturbation
        d = rng.normal(0, 0.03)
        ni = vi + d
        nj = vj - d
        if ni < 0.0 or ni > 1.0 or nj < 0.0 or nj > 1.0:
            continue
        h[i] = ni
        h[j] = nj

        new = compute_C5(h)
        dS = new - cur
        if dS < 0 or rng.random() < np.exp(-dS / T):
            cur = new
            if cur < best:
                best = cur
                best_h = h.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            h[i], h[j] = vi, vj
            no_imp += 1

        T = max(T * cool2, T_min)
        if no_imp > 20000 and rng.random() < 0.003:
            no_imp = 0
            T = 0.004
            # Slightly perturb a few elements
            for _ in range(2):
                a = rng.randint(n)
                b = rng.randint(n)
                if a != b:
                    h[a], h[b] = h[b], h[a]
            h = fix_sum(h, target_sum, rng, n)
            cur = compute_C5(h)
            if cur > best * 1.01:
                h = best_h.copy()
                cur = best

    # Final verification
    h_final = np.clip(best_h, 0.0, 1.0)
    final_score = compute_C5(h_final)
    return h_final, final_score, n
