```python
from scipy.special import logsumexp
import time

    dx = 2.0 / len(h)
```

```python
    tau = args[0]
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")

    m = np.max(vals)
    exp_vals = np.exp(vals - m)
    weights = exp_vals / np.sum(exp_vals)
    
    grad = np.zeros(n)

        w = weights[k_idx]
```

```python
    c = np.correlate(h, 1.0 - h, mode="full")

    return logsumexp(tau * c * dx) / tau

def smooth_gradient(h, args):
    tau = args[0]
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    vals = tau * c * dx
    m = np.max(vals)
    exp_vals = np.exp(vals - m)
    weights = exp_vals / np.sum(exp_vals)

        w = weights[k_idx]

            grad[s:] += w * (1.0 - h[:n-s])
            grad[:n-s] -= w * h[s:]

            grad[:n+s] += w * (1.0 - h[-s:])
            grad[-s:] -= w * h[:n+s]
```

```python
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def compute_c5(h):
    c = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / len(h)
    return float(np.max(c * dx))

def smooth_obj(h, tau):
    c = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / len(h)
    return logsumexp(tau * c * dx) / tau

def smooth_obj_wrapper(h, args):
    return smooth_obj(h, args[0])

def smooth_obj_grad(h, args):
    return smooth_grad(h, args[0])

def smooth_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    vals = tau * c * dx
    m = vals.max()
    ev = np.exp(vals - m)
    w = ev / ev.sum()
    grad = np.zeros(n)
    for k in range(2*n-1):
        s = k - (n-1)
        wk = w[k]
        if wk < 1e-14:
            continue
        if s >= 0:
            grad[s:] += wk * (1.0 - h[:n-s])
            grad[:n-s] -= wk * h[s:]
        else:
            grad[:n+s] += wk * (1.0 - h[-s:])
            grad[-s:] -= wk * h[:n+s]
    grad *= dx
    return grad

def sum_eq(x):
    return np.sum(x) - len(x)/2.0

def sum_eq_jac(x):
    return np.ones(len(x))

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    h += (n/2.0 - h.sum()) / n
    h = np.clip(h, 0.0, 1.0)
    d = n/2.0 - h.sum()
    mask = (h > 1e-6) & (h < 1.0-1e-6)
    if mask.sum() > 0:
        h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_opt(h0, tau, maxiter=300, ftol=1e-14):
    n = len(h0)
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac}]
    bounds = [(0.0, 1.0)] * n
    res = minimize(smooth_obj_wrapper, h0, args=(tau,), jac=smooth_obj_grad,
                   method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project(res.x, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    # Initial point
    try:
        h = np.array(initial_h_values, dtype=float)
    except NameError:
        h = np.full(80, 0.5)
    n = len(h)
    h = project(h, n)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    def tl():
        return budget_s - 10 - (time.time() - t0)
    
    # Phase 1: SLSQP with increasing tau from initial
    cur = h.copy()
    for tau in [5, 10, 20, 40, 80, 150]:
        if tl() < 30: break
        cur = slsqp_opt(cur, tau)
        c5 = compute_c5(cur)
        if c5 < best_c5:
            best_h, best_c5 = cur.copy(), c5
    
    # Phase 2: Multi-start SLSQP
    for i in range(8):
        if tl() < 20: break
        h0 = best_h + rng.randn(n) * 0.05
        h0 = project(h0, n)
        for tau in [20, 50, 100]:
            if tl() < 15: break
            h1 = slsqp_opt(h0, tau, maxiter=200)
            c5 = compute_c5(h1)
            if c5 < best_c5:
                best_h, best_c5 = h1.copy(), c5
                h0 = h1.copy()
    
    # Phase 3: Coordinate descent refinement
    if tl() > 30:
        cur = best_h.copy()
        for _ in range(2000):
            if tl() < 5: break
            j = rng.randint(n)
            for delta in [0.01, -0.01, 0.02, -0.02, 0.05, -0.05]:
                h_try = cur.copy()
                h_try[j] = np.clip(h_try[j] + delta, 0, 1)
                h_try = project(h_try, n)
                c5 = compute_c5(h_try)
                if c5 < compute_c5(cur):
                    cur = h_try.copy()
                    if c5 < best_c5:
                        best_h, best_c5 = cur.copy(), c5
                    break
    
    return (best_h, best_c5, n)
```

```python
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def compute_c5(h):
    c = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / len(h)
    return float(np.max(c * dx))

def smooth_obj(h, tau):
    c = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / len(h)
    return logsumexp(tau * c * dx) / tau

def smooth_obj_wrap(h, args):
    return smooth_obj(h, args[0])

def smooth_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    vals = tau * c * dx
    m = vals.max()
    ev = np.exp(vals - m)
    w = ev / ev.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        s = k - (n - 1)
        wk = w[k]
        if wk < 1e-14:
            continue
        if s >= 0:
            grad[s:] += wk * (1.0 - h[:n - s])
            grad[:n - s] -= wk * h[s:]
        else:
            grad[:n + s] += wk * (1.0 - h[-s:])
            grad[-s:] -= wk * h[:n + s]
    grad *= dx
    return grad

def smooth_obj_grad_wrap(h, args):
    return smooth_grad(h, args[0])

def sum_eq(x):
    return np.sum(x) - len(x) / 2.0

def sum_eq_jac(x):
    return np.ones(len(x))

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    h += (n / 2.0 - h.sum()) / n
    h = np.clip(h, 0.0, 1.0)
    d = n / 2.0 - h.sum()
    mask = (h > 1e-6) & (h < 1.0 - 1e-6)
    if mask.sum() > 0:
        h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    d = n / 2.0 - h.sum()
    if abs(d) > 1e-12:
        h += d / n
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_opt(h0, tau, maxiter=400):
    n = len(h0)
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac}]
    bounds = [(0.0, 1.0)] * n
    res = minimize(smooth_obj_wrap, h0, args=(tau,), jac=smooth_obj_grad_wrap,
                   method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return project(res.x, n)

def coord_descend(cur, best_c5, max_iters, rng, time_check):
    n = len(cur)
    best_h = cur.copy()
    best_c = best_c5
    cur_c = compute_c5(cur)
    deltas = [0.02, -0.02, 0.01, -0.01, 0.05, -0.05, 0.005, -0.005]
    for it in range(max_iters):
        if time_check():
            break
        j = rng.randint(n)
        best_delta = None
        best_trial_c = cur_c
        for d in deltas:
            h_try = cur.copy()
            h_try[j] = np.clip(h_try[j] + d, 0.0, 1.0)
            h_try = project(h_try, n)
            c_try = compute_c5(h_try)
            if c_try < best_trial_c:
                best_trial_c = c_try
                best_delta = h_try.copy()
        if best_delta is not None:
            cur = best_delta
            cur_c = best_trial_c
            if cur_c < best_c:
                best_h = cur.copy()
                best_c = cur_c
        else:
            # Try a random next coordinate
            j2 = rng.randint(n)
            d2 = rng.choice(deltas)
            h_try = cur.copy()
            h_try[j2] = np.clip(h_try[j2] + d2, 0.0, 1.0)
            h_try = project(h_try, n)
            c_try = compute_c5(h_try)
            if c_try < cur_c:
                cur = h_try
                cur_c = c_try
                if cur_c < best_c:
                    best_h = cur.copy()
                    best_c = cur_c
    return best_h, best_c
```
