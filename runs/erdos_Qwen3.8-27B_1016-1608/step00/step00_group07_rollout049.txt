import time

def c5_of_h(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_to_constraint(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        diff = s - target_sum
        if abs(diff) < 1e-14:
            break
        # Water-filling
        h = h - diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    # Exact fix
    diff = h.sum() - target_sum
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.argsort(h)[::-1]
            for i in range(len(idx)):
                j = idx[i]
                rem = min(h[j], diff)
                h[j] -= rem
                diff -= rem
                if diff < 1e-14:
                    break
        else:
            idx = np.argsort(h)
            for i in range(len(idx)):
                j = idx[i]
                add = min(1.0 - h[j], -diff)
                h[j] += add
                diff += add
                if diff > -1e-14:
                    break
    return h

    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full') * dx
    max_val = np.max(corr)
    
    # Find max shifts
    eps = max_val * 1e-8 + 1e-12
    max_mask = corr >= max_val - eps
    max_indices = np.where(max_mask)[0]
    
    grad = np.zeros(n)
    n_max = len(max_indices)
    for idx in max_indices:
        m = idx - (n - 1)
        g = np.zeros(n)

        # grad_j += (1-h[j-m]) if 0<=j-m<=n-1
        #           + (-h[j+m]) if 0<=j+m<=n-1
