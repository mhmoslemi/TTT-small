import numpy as np

from scipy.optimize import minimize
from scipy.optimize import differential_evolution

def _compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _pool_project(h, target):
    h = h.copy()
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        if not np.any(m):
            break
        h[m] += diff / np.sum(m)
    return np.clip(h, 0.0, 1.0)

    return np.sum(x) - _g_target

    return np.ones(len(x))

    _g_target = n / 2.0
    bounds = _g_bounds(n)

                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        return res.x, res.fun

def _pair_swap_search(h, n, max_sweeps=100, step=0.02, time_limit=None, t_start=None):
    """Local search via pair swaps."""
    target = n / 2.0
    best = _compute_c5(h, n)
    h = h.copy()
    
    for sweep in range(max_sweeps):
        if time_limit and (time.time() - t_start) > time_limit:
            break
        improved = False
        perm = np.random.permutation(n)
        for i in perm:
            for j in perm:
                if i == j:
                    continue
                for d in [step, -step]:
                    if h[i] + d < -1e-12 or h[i] + d > 1 + 1e-12:
                        continue
                    if h[j] - d < -1e-12 or h[j] - d > 1 + 1e-12:
                        continue
                    h_new = h.copy()
                    h_new[i] += d
                    h_new[j] -= d
                    c5_new = _compute_c5(h_new, n)
                    if c5_new < best - 1e-12:
                        h = h_new
                        best = c5_new
                        improved = True
                        break
        if not improved:
            break
    return h, best
