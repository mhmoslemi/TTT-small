    for _ in range(300):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h = np.clip(h + diff / cnt * mask, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    global _global_n
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    best_h = h_init.copy()
    best_score = compute_score(h_init)

    n_list = []
    for n_cand in [n_init, 40, 50, 64, 80, 95, 100, 120, 150, 180]:
        if n_cand >= 12 and n_cand not in n_list:
            n_list.append(n_cand)
    n_list.sort()

    x_init = np.linspace(0.0, 2.0, n_init, endpoint=False)

    for n_points in n_list:
        if time.time() - t0 > budget_s * 0.85:
            break

        x_new = np.linspace(0.0, 2.0, n_points, endpoint=False)
        h0 = np.interp(x_new, x_init, h_init)
        h0 = project_feasible(h0, n_points)
        s0 = compute_score(h0)
        if s0 < best_score:
            best_score = s0
            best_h = h0.copy()

        _global_n = n_points
        bounds = [(0.0, 1.0)] * n_points
        constraints = [{'type': 'eq', 'fun': _constraint_fun}]
        n_restarts = 6 if n_points <= 100 else 4

        for r in range(n_restarts):
            if time.time() - t0 > budget_s * 0.9:
                break

            if r == 0:
                h_start = h0.copy()
            elif r == 1:
                h_start = project_feasible(h0 + rng.randn(n_points) * 0.06, n_points)
            elif r == 2:
                h_start = h0.copy()
                k = max(2, n_points // 12)
                idx = rng.choice(n_points, size=k, replace=False)
                h_start[idx] += rng.uniform(-0.5, 0.5, size=k)
                h_start = project_feasible(h_start, n_points)
            elif r == 3:
                h_start = np.zeros(n_points)
                nb = rng.randint(3, min(9, n_points // 4 + 1))
                for b in range(nb):
                    s = int(b * n_points / nb)
                    e = int((b + 1) * n_points / nb)
                    h_start[s:e] = rng.uniform(0.0, 1.0)
                h_start = project_feasible(h_start, n_points)
            elif r == 4:
                h_start = h0.copy()
                blen = max(2, n_points // 5)
                si = rng.randint(0, n_points - blen)
                h_start[si:si + blen] = 1.0 - h0[si:si + blen]
                h_start = project_feasible(h_start, n_points)
            else:
                h_start = project_feasible(h0 + rng.randn(n_points) * 0.15, n_points)

            # First pass: smooth objective for gradient quality
            try:
                res = minimize(soft_max_score, h_start, method='SLSQP',
                               bounds=bounds, constraints=constraints,
                               options={'maxiter': 300, 'ftol': 1e-12})
                h_cand = project_feasible(res['x'], n_points)
                s_cand = compute_score(h_cand)
            except Exception:
                s_cand = np.inf
                h_cand = h_start

            # Second pass: refine with exact max
            if s_cand < best_score + 0.01:
                try:
                    res2 = minimize(compute_score, h_cand, method='SLSQP',
                                    bounds=bounds, constraints=constraints,
                                    options={'maxiter': 300, 'ftol': 1e-12})
                    h_cand = project_feasible(res2['x'], n_points)
                    s_cand = compute_score(h_cand)
                except Exception:
                    pass

            if s_cand < best_score:
                best_score = s_cand
                best_h = h_cand.copy()
            if s_cand < s0:
                s0 = s_cand
                h0 = h_cand.copy()

        # Nelder-Mead with penalty (derivative-free, handles non-smooth max)
        if time.time() - t0 < budget_s * 0.88:
            for r in range(2):
                if time.time() - t0 > budget_s * 0.9:
                    break
                if r == 0:
                    h_start = h0.copy()
                else:
                    h_start = project_feasible(h0 + rng.randn(n_points) * 0.1, n_points)
                try:
                    res = minimize(_penalty_obj, h_start, method='Nelder-Mead',
                                   options={'maxiter': 3000, 'xatol': 1e-6, 'fatol': 1e-10})
                    h_cand = project_feasible(res['x'], n_points)
                    s_cand = compute_score(h_cand)
                    if s_cand < best_score:
                        best_score = s_cand
                        best_h = h_cand.copy()
                    if s_cand < s0:
                        s0 = s_cand
                        h0 = h_cand.copy()
                except Exception:
                    pass

    # Explore small n with multiple random starts
    if time.time() - t0 < budget_s * 0.8:
        for n_small in [16, 24, 32, 48]:
            if time.time() - t0 > budget_s * 0.88:
                break
            _global_n = n_small
            bounds = [(0.0, 1.0)] * n_small
            constraints = [{'type': 'eq', 'fun': _constraint_fun}]
            for trial in range(6):
                if time.time() - t0 > budget_s * 0.92:
                    break
                h_rand = rng.uniform(0.0, 1.0, size=n_small)
                h_rand = project_feasible(h_rand, n_small)
                try:
                    res = minimize(soft_max_score, h_rand, method='SLSQP',
                                   bounds=bounds, constraints=constraints,
                                   options={'maxiter': 200, 'ftol': 1e-12})
                    h_opt = project_feasible(res['x'], n_small)
                    s = compute_score(h_opt)
                    if s < best_score:
                        best_score = s
                        best_h = h_opt.copy()
                except Exception:
                    pass

    # Final polish on best solution if time remains
    if time.time() - t0 < budget_s * 0.95 and len(best_h) <= 200:
        n_final = len(best_h)
        _global_n = n_final
        bounds = [(0.0, 1.0)] * n_final
        constraints = [{'type': 'eq', 'fun': _constraint_fun}]
        try:
            res = minimize(compute_score, best_h, method='SLSQP',
                           bounds=bounds, constraints=constraints,
                           options={'maxiter': 500, 'ftol': 1e-14})
            h_final = project_feasible(res['x'], n_final)
            s_final = compute_score(h_final)
            if s_final < best_score:
                best_score = s_final
                best_h = h_final
        except Exception:
            pass

    return best_h, best_score, len(best_h)
