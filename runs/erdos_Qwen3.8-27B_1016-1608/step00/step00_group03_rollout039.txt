import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(params, n, alpha):
    h = np.clip(params, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Smooth max
    m = np.max(corr)
    shifted = alpha * (corr - m)
    sm = m + np.log(np.sum(np.exp(shifted))) / alpha
    dx = 2.0 / n
    # Add constraint penalty
    s = np.sum(h)
    target = n / 2.0
    penalty = 1000.0 * (s - target) ** 2
    return sm * dx + penalty

def repair_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-9)[0]
        else:
            idx = np.where(h > 1e-9)[0]
        if len(idx) == 0:
            break
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    best_c5 = np.inf
    best_h = None
    best_n = 70
    
    def elapsed():
        return time.time() - t0
    
    # Phase 1: Optimize from initial
    if 'initial_h_values' in dir() or 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, dtype=float)

        c5_init = compute_c5(h_init)
        best_c5 = min(best_c5, c5_init)
        best_h = h_init.copy()
        best_n = n_init
        
        # Optimize from this start
        for alpha in [20, 50, 100, 200]:
            if elapsed() > budget_s * 0.4:
                break
            h0 = best_h.copy()
            res = minimize(smooth_obj, h0, args=(n_init, alpha),
                          method='L-BFGS-B',
                          bounds=[(0,1)]*n_init,
                          options={'maxiter': 300, 'ftol': 1e-12})
            h_new = repair_h(res.x, n_init)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:

                best_h = h_new.copy()
                best_n = n_init

            res = minimize(smooth_obj, h0, args=(n, 50.0),

                          options={'maxiter': 200, 'ftol': 1e-12})
            h_new = repair_h(res.x, n)

                    else:

                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    print(f"  L-BFGS a={alpha:.0f}: {c5_new:.6f}")
            except Exception:
                pass

    # Nelder-Mead for local refinement
    if elapsed() < budget_s * 0.45:
        try:
            res = minimize(lambda x: smooth_obj(x, best_n, 200.0), best_h.copy(),
                           method='Nelder-Mead',
                           options={'maxiter': 2000, 'xatol': 1e-10, 'fatol': 1e-14})
            h_new = repair_h(res.x, best_n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                print(f"  NM: {c5_new:.6f}")
        except Exception:
            pass

    # Local search (pairwise swaps)
    deadline_ls = t0 + budget_s * 0.72
    if elapsed() < budget_s * 0.72:
        h_ls, c5_ls = local_search(best_h, best_n, max_passes=300, deadline=deadline_ls)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls
            print(f"  LocalSearch: {c5_ls:.6f}")

    # Try other resolutions
    for n_try in [70, 98, 56, 100, 140, 200, 300]:
        if elapsed() > budget_s * 0.93:
            break
        if n_try == best_n:
            continue
        for r in range(3):
            if elapsed() > budget_s * 0.95:
                break
            if r == 0:
                h0 = greedy_construct(n_try)
            else:
                h0 = (rng.rand(n_try) < 0.5).astype(float)
                h0 = repair_h(h0, n_try)
            for alpha in [50.0, 200.0]:
                if elapsed() > budget_s * 0.96:
                    break
                try:
                    res = minimize(smooth_obj, h0, args=(n_try, alpha),
                                   method='L-BFGS-B',
                                   bounds=[(0.0, 1.0)] * n_try,
                                   options={'maxiter': 250, 'ftol': 1e-15})
                    h_new = repair_h(res.x, n_try)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        best_n = n_try
                        print(f"  n={n_try} r={r} a={alpha:.0f}: {c5_new:.6f}")
                        # Quick local search on new best
                        h_ls2, c5_ls2 = local_search(h_new, n_try, max_passes=50,
                                                      deadline=t0 + budget_s * 0.97)
                        if c5_ls2 < best_c5:
                            best_c5 = c5_ls2
                            best_h = h_ls2
                            print(f"    LS: {c5_ls2:.6f}")
                except Exception:
                    pass

    # Final local search pass
    if elapsed() < budget_s * 0.99:
        h_ls, c5_ls = local_search(best_h, best_n, max_passes=200,
                                   deadline=t0 + budget_s * 0.995)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls
            print(f"  FinalLS: {c5_ls:.6f}")

    # Verify final solution
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = repair_h(best_h, best_n)
    best_c5 = compute_c5(best_h)

    print(f"BEST: C5={best_c5:.6f} n={best_n} time={elapsed():.1f}s")
    return best_h, best_c5, best_n
