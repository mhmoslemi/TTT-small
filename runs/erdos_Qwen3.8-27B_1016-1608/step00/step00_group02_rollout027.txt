                    zeros.append(i)
                    improved = True
                    break
                else:
                    h[i], h[j] = 1.0, 0.0
        if not improved:
            stalling += 1
            if stalling >= 3:
                break
        else:
            stalling = 0
    return h, c5

def simulated_annealing(h, n, rng, t0, time_limit):
    h = fix_binary_sum(h.copy(), n, rng)
    c5 = evaluate_c5(h, n)
    best_h = h.copy()
    best_c5 = c5
    T = 0.005
    T_min = 1e-7
    cooling = 0.995
    while T > T_min and time.time() - t0 < time_limit:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = rng.choice(ones)
        j = rng.choice(zeros)
        if i == j:
            continue
        h[i], h[j] = h[j], h[i]
        new_c5 = evaluate_c5(h, n)
        delta = new_c5 - c5
        if delta < 0 or rng.random() < math.exp(-delta / T):
            c5 = new_c5
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
        T *= cooling
    return best_h, best_c5

def multi_start(n, rng, t0, time_limit, init_h=None):
    best_h = None
    best_c5 = 999.0
    n_starts = max(3, int(60 * time_limit / n))
    per_start = time_limit / n_starts
    for s in range(n_starts):
        if time.time() - t0 > time_limit:
            break
        if s == 0 and init_h is not None:
            h = init_h.copy()
        else:
            h = np.zeros(n)
            idx = rng.choice(n, n // 2, replace=False)
            h[idx] = 1.0
        # Greedy phase
        t1 = time.time()
        h_g, c5_g = greedy_local_search(h, n, rng, t0, min(time_limit, t1 + per_start * 0.5))
        # SA phase
        t2 = time.time()
        h_s, c5_s = simulated_annealing(h_g, n, rng, t0, min(time_limit, t2 + per_start * 0.5))
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()
        elif c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g.copy()
    return best_h, best_c5

def interpolate_to(h_src, n_tgt, n_src):
    x_src = np.linspace(0, 1, n_src, endpoint=False)
    x_tgt = np.linspace(0, 1, n_tgt, endpoint=False)
    return np.interp(x_tgt, x_src, h_src)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    # Get initial values
    init_h = None
    try:
        init_h = np.array(initial_h_values, dtype=float)
    except:
        init_h = None
    
    best_h = None
    best_c5 = 999.0
    best_n = 0
    
    if init_h is not None:
        c5_init = evaluate_c5(init_h, len(init_h))
        best_c5 = c5_init
        best_h = init_h.copy()
        best_n = len(init_h)
    
    # Grid sizes to try (even n for binary)
    grids = [45, 90, 180]
    # Time allocation
    time_alloc = {45: 180, 90: 300, 180: 350}
    
    for n in grids:
        if time.time() - t0 > budget_s - 60:
            break
        t0_grid = time.time()
        time_limit = min(time_alloc.get(n, 100), budget_s - 60 - (time.time() - t0))
        if time_limit < 30:
            break
        
        # Initialize for this grid
        init_for_grid = None
        if init_h is not None:
            if n == len(init_h):
                init_for_grid = init_h.copy()
            else:
                init_for_grid = interpolate_to(init_h, n, len(init_h))
        
        h_grid, c5_grid = multi_start(n, rng, t0_grid, time_limit, init_for_grid)
        
        if h_grid is not None and c5_grid < best_c5:
            best_c5 = c5_grid
            best_h = h_grid.copy()
            best_n = n
    
    # Final: try one more round with best result re-optimized at its own size
    if time.time() - t0 < budget_s - 30:
        t1 = time.time()
        remaining = budget_s - 30 - (time.time() - t0)
        if remaining > 20:
            h_final, c5_final = greedy_local_search(best_h, best_n, rng, t1, remaining)
            if c5_final < best_c5:
                best_c5 = c5_final
                best_h = h_final.copy()
    
    # Validate
    assert best_h is not None
    assert best_c5 < 999.0
    final_c5 = evaluate_c5(best_h, best_n)
    return best_h, final_c5, best_n
