import numpy as np
from scipy.optimize import minimize, NonlinearConstraint
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_c5_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = float(np.max(corr)) * dx
    k_star = int(np.argmax(corr))
    lag = k_star - (n - 1)
    grad = np.zeros(n)
    for j in range(n):
        jp = j - lag
        jm = j + lag
        if 0 <= jp < n:
            grad[j] += (1.0 - h[jp])
        if 0 <= jm < n:
            grad[j] -= h[jm]
    return c5, grad * dx


def obj_wrapper(h):
    return compute_c5(h)


def grad_wrapper(h):
    return compute_c5_grad(h)[1]


def sum_wrapper(h):
    return float(np.sum(h))


def proj_sum(h, target):
    h = np.asarray(h, dtype=float).copy()
    np.clip(h, 0.0, 1.0, out=h)
    for _ in range(500):
        d = target - np.sum(h)
        if abs(d) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            mask = np.ones(len(h), dtype=bool)
            cnt = len(h)
        h[mask] += d / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h


def try_optimize_tc(h_start, bounds, sum_con, target, best_state, deadline):
    if time.time() > deadline:
        return
    try:
        res = minimize(
            obj_wrapper, h_start, method='trust-constr',
            jac=grad_wrapper, bounds=bounds,
            constraints=[sum_con],

        )
        hx = proj_sum(res.x, target)
        cx = compute_c5(hx)
        if abs(hx.sum() - target) < 0.01 and cx < best_state['c5']:
            best_state['h'] = hx
            best_state['c5'] = cx
    except Exception:
        pass
