import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_feasible(h, target_sum):
    """Project h onto {x: 0<=x<=1, sum(x)=target_sum}"""
    n = len(h)
    # Bisection on mu: find mu such that sum(clip(h - mu, 0, 1)) = target_sum
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2
    return np.clip(h - mu, 0.0, 1.0)

def subgradient_c5(h):
    """Return subgradient of C5 = max_k corr[k]*dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx_max = int(np.argmax(corr))
    k = idx_max - (n - 1)
    g = np.zeros(n)
    if k > 0:
        # c_k = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
        # d/dh[j] = (1-h[j-k]) for j in [k, n-1]  MINUS  h[j+k] for j in [0, n-1-k]
        g[k:] += (1.0 - h[:-k])
        g[:n-k] -= h[k:]
    elif k == 0:
        g[:] = 1.0 - 2.0 * h
    else:
        s = -k
        # c_{-s} = sum_{i=0}^{n-1-s} h[i]*(1-h[i+s])
        # d/dh[j] = (1-h[j+s]) for j in [0, n-1-s]  MINUS  h[j-s] for j in [s, n-1]
        g[:n-s] += (1.0 - h[s:])
        g[s:] -= h[:n-s]
    return g * dx

    n = len(h)

    zeros = np.where(h == 0.0)[0]

        if len(ones) == 0 or len(zeros) == 0:

                best_c5 = new_c5

            h[i] = 1.0
            h[j] = 0.0
