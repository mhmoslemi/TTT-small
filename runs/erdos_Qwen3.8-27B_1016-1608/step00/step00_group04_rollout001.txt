import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def lsq_bin(h, max_iters, rng):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best = float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)
    ones = [i for i in range(n) if h[i] > 0.5]
    zeros = [i for i in range(n) if h[i] < 0.5]
    stall = 0
    for _ in range(max_iters):
        if not ones or not zeros:
            break
        if stall > 3000:
            for _ in range(5):
                a = int(ones[rng.randint(len(ones))])
                b = int(zeros[rng.randint(len(zeros))])
                if a != b:
                    h[a], h[b] = h[b], h[a]
                    v = float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)
                    if v < best:
                        best = v
                        ones.remove(a); zeros.remove(b)
                        ones.append(b); zeros.append(a)
                    else:
                        h[a], h[b] = h[b], h[a]
            stall = 0
            continue
        i = int(ones[rng.randint(len(ones))])
        j = int(zeros[rng.randint(len(zeros))])
        if i == j:
            continue
        h[i], h[j] = h[j], h[i]
        v = float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)
        if v < best - 1e-15:
            best = v
            ones.remove(i); zeros.remove(j)
            ones.append(j); zeros.append(i)
            stall = 0
        else:
            h[i], h[j] = h[j], h[i]
            stall += 1
    return h, best

def lsq_bin_pairs(h, max_iters, rng):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best = float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)
    ones = [i for i in range(n) if h[i] > 0.5]
    zeros = [i for i in range(n) if h[i] < 0.5]
    for _ in range(max_iters):
        if len(ones) < 2 or len(zeros) < 2:
            break
        a1 = int(ones[rng.randint(len(ones))])
        a2 = int(ones[rng.randint(len(ones))])
        b1 = int(zeros[rng.randint(len(zeros))])
        b2 = int(zeros[rng.randint(len(zeros))])
        if a1 == a2 or b1 == b2 or a1 == b1 or a1 == b2 or a2 == b1 or a2 == b2:
            continue
        h[a1], h[b1] = h[b1], h[a1]
        h[a2], h[b2] = h[b2], h[a2]
        v = float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)
        if v < best - 1e-15:
            best = v
            ones.remove(a1); ones.remove(a2)
            zeros.remove(b1); zeros.remove(b2)
            ones.extend([b1, b2])
            zeros.extend([a1, a2])
        else:
            h[a1], h[b1] = h[b1], h[a1]
            h[a2], h[b2] = h[b2], h[a2]
    return h, best

def make_random_bin(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h

def make_structured_bin(n, rng):
    h = np.zeros(n)
    n1 = n // 2
    pattern = rng.permutation(n)
    h[pattern[:n1]] = 1.0
    return h
