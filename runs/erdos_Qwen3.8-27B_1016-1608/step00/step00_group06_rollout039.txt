def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def obj_and_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    max_c = np.max(corr)
    shifted = eps * (corr - max_c)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    Z = np.sum(exp_shifted)
    val = max_c + np.log(Z) / eps
    w = exp_shifted / Z
    
    grad = np.zeros(n)
    for ki in range(len(w)):
        if w[ki] < 1e-12:
            continue
        k = ki - (n - 1)
        # ∂g_k/∂h[j] = dx * [(1-h[j-k])*I(0≤j-k≤N-1) - h[j+k]*I(0≤j+k≤N-1)]
        # First term: (1-h[j-k]) for j-k in [0,N-1]
        # j ranges [0,N-1], j-k ranges [0,N-1] → j ranges [k, N-1+k] ∩ [0,N-1]
        for j in range(n):
            term1 = 0.0
            term2 = 0.0
            if 0 <= j - k <= n - 1:
                term1 = 1.0 - h[j - k]
            if 0 <= j + k <= n - 1:
                term2 = h[j + k]
            grad[j] += w[ki] * (term1 - term2)

    if np.sum(free2) > 0:

def grad_for_k(h, k):

    grad = np.zeros(n)
    if k >= 0:
        # First term: j in [k, n-1], grad[j] += 1-h[j-k]
        grad[k:] += 1 - h[:n-k]
        # Second term: j in [0, n-1-k], grad[j] -= h[j+k]
        grad[:n-k] -= h[k:]

        # First term: j in [0, n-1-kp], grad[j] += 1-h[j+kp]
        grad[:n-kp] += 1 - h[kp:]
        # Second term: j in [kp, n-1], grad[j] -= h[j-kp]
        grad[kp:] -= h[:n-kp]
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * 2.0 / n)

def _gf(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += 1.0 - h[:n - k]
        g[:n - k] -= h[k:]
    else:
        kp = -k
        g[:n - kp] += 1.0 - h[kp:]
        g[kp:] -= h[:n - kp]
    return g

def _sobj(x, eps, pen):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    mc = np.max(c)
    sh = np.clip(eps * (c - mc), -50, 50)
    e = np.exp(sh)
    v = x.sum() - n * 0.5
    return mc + np.log(e.sum()) / eps + pen * v * v

def _sgrad(x, eps, pen):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    mc = np.max(c)
    sh = np.clip(eps * (c - mc), -50, 50)
    w = np.exp(sh)
    w /= w.sum()
    g = np.zeros(n)
    for ki in range(2 * n - 1):
        if w[ki] < 1e-10:
            continue
        g += w[ki] * _gf(x, ki - (n - 1))
    g *= dx
    v = x.sum() - n * 0.5
    g += 2.0 * pen * v
    return g

def _nobj(x, pen):
    n = len(x)
    v = x.sum() - n * 0.5
    return _c5(x) + pen * v * v

def _proj(h, tgt):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)

        cn = m.sum()
        if cn == 0:
            break
        h[m] += d / cn
        np.clip(h, 0.0, 1.0, out=h)
