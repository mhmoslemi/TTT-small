    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_onto_feasible(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum}"""
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    # Water-filling style correction
    for _ in range(10):

            free = h < 1.0 - 1e-12
        else:

        h = np.clip(h, 0.0, 1.0)

        if abs(diff) < 1e-12:

    h_new[i] += delta
    h_new[j] -= delta
    np.clip(h_new, 0.0, 1.0, out=h_new)
    return h_new


def local_search(h, time_limit, start_time, n_fracs=3, frac_vals=None):
    n = len(h)
    target_sum = n / 2.0
    best_c5 = compute_c5(h)
    if frac_vals is None:
        frac_vals = np.array([0.2, 0.4, 0.6, 0.8])
    pair_list = [(i, j) for i in range(n) for j in range(i + 1, n)]
    
    for it in range(5000):
        if time.time() - start_time > time_limit - 5:
            break
        best_move = None
        best_new_c5 = best_c5
        
        for i, j in pair_list:
            # Move from j to i
            max_d = min(1.0 - h[i], h[j])
            if max_d > 1e-6:
                for f in frac_vals:
                    d = max_d * f
                    h_try = h.copy()
                    h_try[i] += d
                    h_try[j] -= d
                    c5_try = compute_c5(h_try)
                    if c5_try < best_new_c5 - 1e-12:
                        best_new_c5 = c5_try
                        best_move = h_try
            # Move from i to j
            max_d = min(1.0 - h[j], h[i])
            if max_d > 1e-6:
                for f in frac_vals:
                    d = max_d * f
                    h_try = h.copy()
                    h_try[j] += d
                    h_try[i] -= d
                    c5_try = compute_c5(h_try)
                    if c5_try < best_new_c5 - 1e-12:
                        best_new_c5 = c5_try
                        best_move = h_try
        
        if best_move is not None:
            h = best_move
            best_c5 = best_new_c5
        else:
            break
    
    return h, best_c5


def sa_phase(h, best_c5, rng, t_start, t_end, T_init, T_min, cool, accept_thresh):
    T = T_init
    while time.time() < t_end:
        h_new = make_perturbation(h, rng, scale=0.08)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            h = h_new
            best_c5 = c5_new
            T = min(T * 1.03, 0.01)
        else:
            d_c5 = c5_new - best_c5
            if d_c5 < accept_thresh and rng.random() < np.exp(-d_c5 / T):
                h = h_new
        T = max(T * cool, T_min)
    return h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    t0 = time.time()
    rng = np.random.default_rng(seed)
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except (NameError, UnboundLocalError):
        h0 = np.full(70, 0.5)
    
    n = len(h0)
    target_sum = n / 2.0
    h0 = project_feasible(h0, target_sum)
    
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    
    # Phase 1: SA (35% time)
    t1 = t0 + budget_s * 0.35
    best_h, best_c5 = sa_phase(best_h, best_c5, rng, time.time(), t1,
                               T_init=0.015, T_min=1e-5, cool=0.9998, accept_thresh=0.04)
    
    # Phase 2: Local search (20% time)
    t2 = t0 + budget_s * 0.55
    h_ls, c5_ls = local_search(best_h, budget_s * 0.52, t0)
    if c5_ls < best_c5:
        best_h, best_c5 = h_ls, c5_ls
    
    # Phase 3: SA from improved point (20% time)
    t3 = t0 + budget_s * 0.75
    best_h, best_c5 = sa_phase(best_h, best_c5, rng, time.time(), t3,
                               T_init=0.005, T_min=1e-6, cool=0.9997, accept_thresh=0.02)
    
    # Phase 4: Local search (15% time)
    t4 = t0 + budget_s * 0.90
    h_ls2, c5_ls2 = local_search(best_h, budget_s * 0.88, t0)
    if c5_ls2 < best_c5:
        best_h, best_c5 = h_ls2, c5_ls2
    
    # Phase 5: Final SA + local search (10% time)
    t5 = t0 + budget_s * 0.95
    best_h, best_c5 = sa_phase(best_h, best_c5, rng, time.time(), t5,
                               T_init=0.002, T_min=1e-6, cool=0.9995, accept_thresh=0.01)
    
    h_ls3, c5_ls3 = local_search(best_h, budget_s, t0, n_fracs=5,
                                  frac_vals=np.array([0.1, 0.25, 0.5, 0.75, 0.9]))
    if c5_ls3 < best_c5:
        best_h, best_c5 = h_ls3, c5_ls3
    
    # Final validation
    best_h = project_feasible(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, n
