import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def sum_constraint(x):
    return np.sum(x) - len(x) / 2.0

def repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not np.any(mask):
            h[:] = target / n
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp(h0, n, maxiter=500):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint}]
    try:
        res = minimize(compute_c5, h0, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.fun < compute_c5(h0) + 1e-14:
            return repair(np.clip(res.x, 0.0, 1.0), n)
    except Exception:
        pass
    return h0

def nm_local(h0, n):
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(compute_c5, h0, method='Nelder-Mead',
                       bounds=bnds, options={'maxiter': 20000, 'xatol': 1e-13, 'fatol': 1e-15})
        return repair(np.clip(res.x, 0.0, 1.0), n)
    except Exception:
        return h0

    out = h.copy()
    if k < n:
        out[:k] = 1.0 - h[:k]
        out = repair(out, n)
    return out
