    try:
        r1 = minimize(_smooth_max_fg, h0, jac=True, method='L-BFGS-B',
                     bounds=bounds,
                     options={'maxiter': 5000, 'ftol': 1e-15, 'gtol': 1e-12, 'maxfun': 20000})
        h1 = project_sum(np.clip(r1.x, 0.0, 1.0), n / 2.0)
        c1 = compute_c5(h1)
        if c1 < c_best:
            h_best, c_best = h1, c1
    except Exception:
        pass
    if n <= 120 and time.time() < deadline:
        try:
            r2 = minimize(_smooth_max_f, h_best, method='SLSQP',
                        constraints=[{'type': 'eq', 'fun': _con_eq, 'jac': _con_eq_jac}],
                        bounds=bounds,
                        options={'maxiter': 400, 'ftol': 1e-14})
            h2 = project_sum(np.clip(r2.x, 0.0, 1.0), n / 2.0)
            c2 = compute_c5(h2)
            if c2 < c_best:
                h_best, c_best = h2, c2
        except Exception:
            pass
    return h_best, c_best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = 10.0
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        ih = np.array(initial_h_values, dtype=float)
        ih = project_sum(ih, len(ih) / 2.0)
        ic = compute_c5(ih)
        best_h, best_c5, best_n = ih.copy(), ic, len(ih)

    # Phase 1: Bang-bang random + exhaustive
    bb_end = t0 + budget_s * 0.30
    for n in [80, 100, 120, 160]:
        if time.time() > bb_end:
            break
        hh, cc = bang_bang_random(n, rng, 50000, bb_end)
        hh, cc = bang_bang_exhaustive(hh, bb_end)
        if cc < best_c5:
            best_h, best_c5, best_n = hh.copy(), cc, n

    # Phase 2: Continuous refinement from best bang-bang and parent
    co_end = t0 + budget_s * 0.75
    tried_ns = set()
    for n in sorted(set([best_n, 80, 100, 120, 160])):
        if time.time() > co_end:
            break
        if n in tried_ns:
            continue
        tried_ns.add(n)
        if best_h is not None and len(best_h) == n:
            hs = best_h.copy()
        elif best_h is not None:
            hs = resample_to(best_h, n)
        else:
            hs = np.ones(n) * 0.5
        hc, cc = cont_opt(hs, n, co_end)
        if cc < best_c5:
            best_h, best_c5, best_n = hc.copy(), cc, n
        # Also try parent at this n
        if 'initial_h_values' in globals() and initial_h_values is not None and n != len(initial_h_values):
            hp = resample_to(np.array(initial_h_values, dtype=float), n)
            hc2, cc2 = cont_opt(hp, n, co_end)
            if cc2 < best_c5:
                best_h, best_c5, best_n = hc2.copy(), cc2, n
                tried_ns.add(n)

    # Phase 3: Random restarts
    rr_end = t0 + budget_s * 0.92
    for k in range(12):
        if time.time() > rr_end:
            break
        n = best_n
        hr = np.clip(0.5 + 0.4 * rng.randn(n), 0.0, 1.0)
        hr = project_sum(hr, n / 2.0)
        hc, cc = cont_opt(hr, n, rr_end)
        if cc < best_c5:
            best_h, best_c5, best_n = hc.copy(), cc, n

    # Phase 4: Final polish with more L-BFGS-B iterations
    if time.time() < t0 + budget_s * 0.95 and best_h is not None:
        hc, cc = cont_opt(best_h, best_n, t0 + budget_s * 0.95)
        if cc < best_c5:
            best_h, best_c5, best_n = hc.copy(), cc, best_n

    if best_h is None:
        best_h = np.ones(80) * 0.5
        best_c5 = compute_c5(best_h)
        best_n = 80

    return best_h, float(best_c5), int(best_n)
