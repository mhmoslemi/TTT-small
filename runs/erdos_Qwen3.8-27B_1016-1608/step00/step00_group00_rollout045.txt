from scipy.optimize import minimize

_ng = 45

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n * 0.5
    for _ in range(300):
        d = target - np.sum(h)
        if abs(d) < 1e-13:
            break
        m = (h < 1.0 - 1e-9) if d > 0 else (h > 1e-9)
        c = float(np.sum(m))
        if c == 0:
            break
        h[m] += d / c
        h = np.clip(h, 0.0, 1.0)

def _obj(h):
    return compute_c5(h, _ng)

def _con_eq(h):
    return np.sum(h) - _ng * 0.5

def _con_jac(h):
    return np.ones(_ng)

def _obj_pen(h):
    base = compute_c5(h, _ng)
    pen = 200.0 * (np.sum(h) - _ng * 0.5) ** 2
    return base + pen
