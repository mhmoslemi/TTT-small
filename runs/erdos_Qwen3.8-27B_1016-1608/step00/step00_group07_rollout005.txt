from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def compute_c5_all(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        s = np.sum(h)
        excess = s - target
        if abs(excess) < 1e-12:
            break
        free = (h > 0.0) & (h < 1.0)
        nf = int(np.sum(free))
        if nf > 0:
            h[free] -= excess / nf
            h = np.clip(h, 0.0, 1.0)
        else:
            break
    return h

def make_feasible_binary(n, rng):
    h = np.zeros(n)
    positions = rng.permutation(n)[:n//2]
    h[positions] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    
    # Get initial values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)

    best_c5 = compute_c5(best_h)
    n_init = len(h_init)
    
    def remaining():
        return budget_s - (time.time() - t0)
    
    # Stage 1: Local continuous optimization at initial resolution
    n = n_init
    target = n / 2.0
    
    for beta in [10.0, 30.0, 100.0, 300.0]:
        if remaining() < 30:
            break
        
        def obj(x):
            corr = np.correlate(x, 1.0 - x, mode="full") * (2.0/n)
            mx = np.max(corr)
            return mx + np.log(np.sum(np.exp(beta*(corr-mx))))/beta
        
        def obj_grad(x):
            # Numerical gradient (expensive but n is small)
            g = np.zeros(n)
            eps = 1e-7
            f0 = obj(x)
            for i in range(n):
                xp = x.copy()
                xp[i] += eps
                g[i] = (obj(xp) - f0) / eps
            return g

                       'jac': lambda x: np.ones(n)}]

                          bounds=bounds, constraints=constraints,
