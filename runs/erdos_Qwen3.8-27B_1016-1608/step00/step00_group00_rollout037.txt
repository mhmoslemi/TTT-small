import time

    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_c5_vec(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    return corr

def subgradient_c5(h):
    """Compute subgradient of C5 w.r.t. h at the active maximizer."""
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    k = int(np.argmax(corr))
    s = k - (n - 1)
    grad = np.zeros(n)
    for m in range(n):
        val = 0.0
        if 0 <= m - s < n:
            val += (1.0 - h[m - s])
        if 0 <= m + s < n:
            val -= h[m + s]
        grad[m] = val
    # This is the subgradient of corr[k] * dx, so scale by dx
    dx = 2.0 / n
    return dx * grad

def project_feasible(h, n):
    """Project h onto {x: 0<=x<=1, sum(x)=n/2}."""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Binary search on offset
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(h_proj)
        if s > target:
            hi = mid
        else:
            lo = mid
    h_proj = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    # Fine-tune: adjust the element closest to 0.5 by the residual
    residual = target - np.sum(h_proj)
    if abs(residual) > 1e-10:
        # Find closest to 0.5 that we can adjust
        idx = int(np.argmin(np.abs(h_proj - 0.5)))
        h_proj[idx] = h_proj[idx] + residual
        h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def subgradient_descent(h0, n, max_iter=5000, lr0=0.05, budget_start=None, total_budget=1100.0):
    """Run subgradient descent."""
    h = h0.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr0
    for it in range(max_iter):

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    h_p = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    # Fine correction
    res = target - np.sum(h_p)
    if abs(res) > 1e-9:
        # Adjust one element
        idx = int(np.argmax(h_p * (1.0 - h_p)))
        h_p[idx] += res
        h_p = np.clip(h_p, 0.0, 1.0)
    return h_p

def sg_descent(h0, n, max_iter, lr0, budget_check=None):
    h = h0.astype(float).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr0
    for it in range(max_iter):
        if budget_check and budget_check():
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr))
        s = idx - (n - 1)
        # Subgradient vectorized
        # d(corr[s])/d(h[m]) = (1-h[m-s])[m-s valid] - h[m+s][m+s valid]
        g1 = np.zeros(n)
        g2 = np.zeros(n)
        if s >= 0:
            # m-s valid: m >= s; m+s valid: m <= n-1-s
            g1[s:] = 1.0 - h[:n-s]  # (1-h[m-s]) for m=s,...,n-1
            valid2 = s <= n-1
            if valid2:
                g2[:n-s] = h[s:]  # h[m+s] for m=0,...,n-1-s
        else:
            t = -s
            # m-s = m+t valid: m <= n-1-t; m+s = m-t valid: m >= t
            g1[:n-t] = 1.0 - h[t:]  # (1-h[m-s]) = (1-h[m+t]) for m=0,...,n-1-t
            g2[t:] = h[:n-t]  # h[m+s] = h[m-t] for m=t,...,n-1
        grad = (g1 - g2) * (2.0 / n)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        if it % 1000 == 999:
            lr *= 0.85
    return best_h, best_c5

def upscale(h, n_new):
    """Upscale h from len(h) to n_new points, preserving integral."""
    n_old = len(h)
    # Simple linear interpolation
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0 / n_new
    h_new = np.interp(x_new, x_old, h, period=None)
    h_new = np.clip(h_new, 0, 1)
    # Fix integral
    target = n_new / 2.0
    current = np.sum(h_new)
    if abs(current - target) > 1e-6:
        h_new = project_feasible(h_new, n_new)
    return h_new

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    h_p = np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)
    res = target - np.sum(h_p)
    if abs(res) > 1e-12:
        idx = int(np.argmax(h_p * (1.0 - h_p)))
        h_p[idx] += res
        h_p = np.clip(h_p, 0.0, 1.0)
    return h_p

def subgrad_step(h, n, lr, deadline):
    """One subgradient step. Returns (new_h, c5, elapsed_ok)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    s = idx - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        if n - s > 0:
            g[s:] += 1.0 - h[:n-s]
            g[:n-s] -= h[s:]
    else:
        t = -s
        if n - t > 0:
            g[:n-t] += 1.0 - h[t:]
            g[t:] -= h[:n-t]
    g *= (2.0 / n)
    h_new = h - lr * g
    h_new = project_feasible(h_new, n)
    c5 = compute_c5(h_new)
    return h_new, c5

def sg_descent(h0, n, max_iter, lr0, lr_decay, dec_every, deadline):
    h = h0.astype(float).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        h, c5 = subgrad_step(h, n, lr, deadline)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        elif it > 0 and it % dec_every == 0:
            lr *= lr_decay
    return best_h, best_c5

def sg_descent_momentum(h0, n, max_iter, lr0, momentum, deadline):
    h = h0.astype(float).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    vel = np.zeros(n)
    lr = lr0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr))
        s = idx - (n - 1)
        g = np.zeros(n)
        if s >= 0:
            if n - s > 0:
                g[s:] += 1.0 - h[:n-s]
                g[:n-s] -= h[s:]
        else:
            t = -s
            if n - t > 0:
                g[:n-t] += 1.0 - h[t:]
                g[t:] -= h[:n-t]
        g *= (2.0 / n)
        vel = momentum * vel - lr * g
        h_new = project_feasible(h + vel, n)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new

    return best_h, best_c5

def smooth_max_obj(h, n, T):
    """Smoothed approximation to C5."""
    corr = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(T * (corr - m)))) / T

def upscale(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    h_new = project_feasible(h_new, n_new)
    return h_new

def try_bangbang(h, n, rng, deadline):

            best_h = h_trial.copy()
    return best_h, best_c5

    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    h_p = np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)
    res = target - np.sum(h_p)
    if abs(res) > 1e-12:
        idx = int(np.argmax(h_p * (1.0 - h_p)))
        h_p[idx] += res
        h_p = np.clip(h_p, 0.0, 1.0)
    return h_p

def get_subgrad(h, n, idx):
    s = idx - (n - 1)

            g[t:] -= h[:n-t]

        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr))
        g = get_subgrad(h, n, idx)
        h = project_feasible(h - lr * g, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        elif it > 0 and it % dec_every == 0:
            lr *= lr_decay
    return best_h, best_c5

def sg_mom_loop(h0, n, max_iter, lr0, mu, deadline):
    h = h0.astype(float).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    vel = np.zeros(n)
    lr = lr0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr))
        g = get_subgrad(h, n, idx)
        vel = mu * vel - lr * g
        h = project_feasible(h + vel, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if it % 3000 == 2999:
            lr *= 0.9
    return best_h, best_c5

def multi_sg_loop(h0, n, max_iter, lr0, deadline):
    """Adaptive multi-start subgradient with restart on stagnation."""
    h = h0.astype(float).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr0
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        top_k = np.argsort(corr)[-3:]
        g = np.zeros(n)
        for i in top_k:
            g += get_subgrad(h, n, int(i))
        g /= 3.0
        h = project_feasible(h - lr * g, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 5000:
            lr *= 0.7
            stall = 0
    return best_h, best_c5

def upscale(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old * 2.0
    x_new = (np.arange(n_new) + 0.5) / n_new * 2.0
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new
