def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(h, T, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    x = corr * dx
    mx = np.max(x)
    # Stable log-sum-exp
    shifted = T * (x - mx)
    # Clip to prevent overflow
    shifted = np.clip(shifted, -50, 50)
    c5_smooth = mx + np.log(np.sum(np.exp(shifted))) / T
    target = n / 2.0
    viol = np.sum(h) - target
    return c5_smooth + penalty * viol * viol

def repair_h(h, n):

    target = n / 2.0

    if s > 1e-12:
        h = h * (target / s)

        if abs(diff) < 1e-10:

        else:

            h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h
