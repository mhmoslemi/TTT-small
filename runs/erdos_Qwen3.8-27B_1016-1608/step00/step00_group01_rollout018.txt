import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(1000):
        diff = h.sum() - target_sum
        if abs(diff) < 1e-13:
            return h
        if diff > 0:
            mask = h > 0
            cnt = int(mask.sum())
            if cnt == 0:
                return h
            h[mask] -= diff / cnt
        else:
            mask = h < 1
            cnt = int(mask.sum())
            if cnt == 0:
                return h
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_minimize(h, j, corr):
    n = len(h)
    n2 = 2 * n - 1
    k_idx = np.arange(n2)
    lag = k_idx - (n - 1)
    i1 = j - lag
    i2 = j + lag
    valid1 = (i1 >= 0) & (i1 < n)
    valid2 = (i2 >= 0) & (i2 < n)
    deriv = np.zeros(n2)
    deriv[valid1] += 1.0 - h[i1[valid1]]
    deriv[valid2] += -h[i2[valid2]]
    lo = -h[j]
    hi = 1.0 - h[j]
    a, b = lo, hi
    for _ in range(80):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        f1 = np.max(corr + m1 * deriv)
        f2 = np.max(corr + m2 * deriv)
        if f1 < f2:
            b = m2
        else:
            a = m1
    delta = (a + b) / 2.0
    return np.clip(h[j] + delta, 0.0, 1.0)

def coordinate_descent(h, max_sweeps=200, time_limit=None, start_time=None):
    n = len(h)
    target_sum = n / 2.0
    h = project_feasible(h.copy(), target_sum)
    best_c5 = evaluate_c5(h)
    for sweep in range(max_sweeps):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        improved = False
        corr = np.correlate(h, 1.0 - h, mode='full')
        for j in range(n):
            new_hj = coord_minimize(h, j, corr)
            if abs(new_hj - h[j]) > 1e-12:
                h[j] = new_hj
                improved = True
                corr = np.correlate(h, 1.0 - h, mode='full')
        if not improved:
            break
        h = project_feasible(h, target_sum)
    return h

def random_perturb(h, rng, strength):
    n = len(h)
    target_sum = n / 2.0
    delta = rng.normal(0, strength, n)
    delta -= delta.mean()
    h_new = h + delta
    return project_feasible(h_new, target_sum)

def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0, 2.0, n_new, endpoint=False) + 1.0 / n_new
    f = np.interp(x_new, x_old, h_old, period=2.0)
    f = np.clip(f, 0.0, 1.0)
    return project_feasible(f, n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    best_h = None
    best_c5 = np.inf

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        h0 = None

    if h0 is not None:
        n0 = len(h0)
        h0 = project_feasible(h0.copy(), n0 / 2.0)
        c5_0 = evaluate_c5(h0)
        best_h = h0.copy()
        best_c5 = c5_0
    else:
        n0 = 47
        best_h = np.ones(n0) * 0.5
        best_h = project_feasible(best_h, n0 / 2.0)
        best_c5 = evaluate_c5(best_h)

    # Phase 1: Coordinate descent on initial
    remaining = budget_s - (time.time() - start_time)
    if remaining > 10:
        h_cd = coordinate_descent(best_h, max_sweeps=500, time_limit=remaining * 0.5, start_time=start_time)
        c5_cd = evaluate_c5(h_cd)
        if c5_cd < best_c5:
            best_h, best_c5 = h_cd, c5_cd

    # Phase 2: Random perturbations + coordinate descent
    remaining = budget_s - (time.time() - start_time)
    attempt = 0
    while time.time() - start_time < budget_s * 0.75:
        attempt += 1
        strength = 0.03 + 0.12 * np.exp(-attempt / 30.0)
        if attempt > 50:
            strength = 0.2 * np.exp(-(attempt - 50) / 50.0)
        h_p = random_perturb(best_h, rng, strength)
        max_sw = 30 if attempt < 20 else 10
        h_opt = coordinate_descent(h_p, max_sweeps=max_sw)
        c5_new = evaluate_c5(h_opt)
        if c5_new < best_c5 - 1e-12:
            best_h, best_c5 = h_opt, c5_new

    # Phase 3: Try different n by interpolation
    remaining = budget_s - (time.time() - start_time)
    if remaining > 30:
        n_base = len(best_h)
        for n_try in [n_base, n_base + 1, n_base - 1, 2 * n_base, n_base * 3 - 1]:
            if n_try < 10 or n_try > 500:
                continue
            if n_try == n_base:
                continue
            h_interp = interpolate_h(best_h, n_try)
            h_opt = coordinate_descent(h_interp, max_sweeps=100)
            c5_try = evaluate_c5(h_opt)
            if c5_try < best_c5 - 1e-12:
                best_h, best_c5 = h_opt, c5_try
            if time.time() - start_time > budget_s * 0.9:
                break

    # Phase 4: Final polish with more sweeps
    remaining = budget_s - (time.time() - start_time)
    if remaining > 5:
        h_final = coordinate_descent(best_h, max_sweeps=500, time_limit=remaining - 1, start_time=start_time)
        c5_final = evaluate_c5(h_final)
        if c5_final < best_c5:
            best_h, best_c5 = h_final, c5_final

    # Ensure feasibility
    n = len(best_h)
    best_h = project_feasible(best_h, n / 2.0)
    final_c5 = evaluate_c5(best_h)

    return (best_h, final_c5, n)
