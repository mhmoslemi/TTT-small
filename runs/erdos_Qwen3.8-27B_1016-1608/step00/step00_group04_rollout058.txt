def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1-h, mode='full'))) * dx

def repair_h(h, target_sum):
    """Ensure h is in [0,1] and sums to target_sum."""
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            # Need to increase some entries
            order = np.argsort(h)[::-1]  # most room for increase first (actually least h)
            # Better: increase entries with most room (1-h[i] is largest)
            order = np.argsort(1-h)  # entries closest to 0 first
            for idx in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
        else:
            order = np.argsort(h)  # entries closest to 1 first (most to decrease)
            for idx in order:
                if diff >= 0:
                    break
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
    return h

        zeros = np.where(h < 0.5)[0]

            break

        best_flip = None

            for j in ones:
                h[i] = 1.0
                h[j] = 0.0
                c5 = compute_c5(h)
                improvement = best_c5 - c5
                if improvement > best_improvement:
                    best_improvement = improvement
                    best_flip = (i, j, c5)

        h[i] = 1.0
        h[j] = 0.0

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def repair_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.argsort(h)
    else:
        idx = np.argsort(h)[::-1]
    for k in idx:
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            step = min(diff, 1.0 - h[k])
            h[k] += step
        else:
            step = min(-diff, h[k])
            h[k] -= step
        diff += step if diff > 0 else step
    return h

def greedy_binary(h, max_passes=3000):
    n = len(h)
    c5 = compute_c5(h)
    for _ in range(max_passes):
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_gain = 1e-12
        best_ij = None
        best_nc5 = c5

                h[i] = 1.0; h[j] = 0.0

            break
        h[best_ij[0]] = 1.0; h[best_ij[1]] = 0.0

    h = np.clip(h, 0.0, 1.0).copy()
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
