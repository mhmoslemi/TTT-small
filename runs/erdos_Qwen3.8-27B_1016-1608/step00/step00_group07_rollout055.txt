import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        if not np.any(mask):
            mask = np.ones(len(h), dtype=bool)
        h[mask] += diff / float(np.sum(mask))
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_lse(h, T):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = T * (corr - m)
    max_shift = np.max(shifted)
    return m + (m_shift + np.log(np.sum(np.exp(shifted - m_shift)))) / T if False else m + np.log(np.sum(np.exp(shifted))) / T

def obj_max(h):
    return np.max(np.correlate(h, 1.0 - h, mode="full"))

def eq_con(h, target):
    return np.sum(h) - target

def do_slsqp(x0, obj_func, obj_args, target_val, bounds_list, cons_target, maxiter, ftol):
    cons = [{'type': 'eq', 'fun': eq_con, 'args': (cons_target,)}]
    return minimize(obj_func, x0, args=obj_args, method='SLSQP',
                    bounds=bounds_list, constraints=cons,
                    options={'maxiter': maxiter, 'ftol': ftol})

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    target = n / 2.0
    bounds = list(zip(np.zeros(n), np.ones(n)))

    h0 = fix_sum(h0, target)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Multi-temperature log-sum-exp
    for T in [3.0, 10.0, 30.0, 100.0, 300.0, 1000.0, 3000.0]:
        if time.time() - t0 > budget_s - 80:
            break
        try:
            res = do_slsqp(best_h, obj_lse, (T,), target, bounds, target, 300, 1e-14)
            nh = fix_sum(res.x, target)
            c5 = compute_c5(nh)
            if c5 < best_c5:
                best_h = nh
                best_c5 = c5
        except Exception:
            pass

    # Phase 2: Direct max
    if time.time() - t0 < budget_s - 30:
        try:
            res = do_slsqp(best_h, obj_max, (), target, bounds, target, 300, 1e-14)
            nh = fix_sum(res.x, target)
            c5 = compute_c5(nh)
            if c5 < best_c5:
                best_h = nh
                best_c5 = c5
        except Exception:
            pass

    # Phase 3: Another LSE pass at high T to sharpen
    if time.time() - t0 < budget_s - 20:
        for T in [5000.0, 20000.0]:
            if time.time() - t0 > budget_s - 10:
                break
            try:
                res = do_slsqp(best_h, obj_lse, (T,), target, bounds, target, 200, 1e-14)
                nh = fix_sum(res.x, target)
                c5 = compute_c5(nh)
                if c5 < best_c5:
                    best_h = nh
                    best_c5 = c5
            except Exception:
                pass

    # Phase 4: Random restarts
    if time.time() - t0 < budget_s - 8:
        for trial in range(8):
            if time.time() - t0 > budget_s - 4:
                break
            scale = 0.05 + 0.02 * trial
            pert = best_h + scale * rng.randn(n)
            pert = fix_sum(pert, target)
            try:
                res = do_slsqp(pert, obj_max, (), target, bounds, target, 100, 1e-12)
                nh = fix_sum(res.x, target)
                c5 = compute_c5(nh)
                if c5 < best_c5:
                    best_h = nh
                    best_c5 = c5
            except Exception:
                pass

    # Phase 5: Try doubling resolution
    if time.time() - t0 < budget_s - 10:
        n2 = n * 2
        target2 = n2 / 2.0
        bounds2 = list(zip(np.zeros(n2), np.ones(n2)))
        x1 = np.linspace(0.0, 2.0, n, endpoint=False)
        x2 = np.linspace(0.0, 2.0, n2, endpoint=False)
        h_cat = np.concatenate([best_h, [best_h[0]]])
        h_interp = np.interp(x2, x1, h_cat)
        h_interp = fix_sum(h_interp, target2)
        c5_i = compute_c5(h_interp)

        if c5_i < best_c5 + 0.02:
            for T in [30.0, 100.0, 500.0, 2000.0]:
                if time.time() - t0 > budget_s - 5:
                    break
                try:
                    res = do_slsqp(h_interp if T == 30.0 else best_h, obj_lse, (T,), target2, bounds2, target2, 200, 1e-14)
                    nh = fix_sum(res.x, target2)
                    c5 = compute_c5(nh)
                    if c5 < best_c5:
                        best_h = nh
                        best_c5 = c5
                except Exception:
                    pass

            if time.time() - t0 < budget_s - 5:
                try:
                    res = do_slsqp(best_h, obj_max, (), target2, bounds2, target2, 200, 1e-14)
                    nh = fix_sum(res.x, target2)
                    c5 = compute_c5(nh)
                    if c5 < best_c5:
                        best_h = nh
                        best_c5 = c5
                except Exception:
                    pass

    # Final cleanup
    fin_n = len(best_h)
    fin_target = fin_n / 2.0
    best_h = fix_sum(best_h, fin_target)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, fin_n
