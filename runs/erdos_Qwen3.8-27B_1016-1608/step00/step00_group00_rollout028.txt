import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def make_even_binary(n):
    h = np.zeros(n, dtype=np.float64)
    idx = np.round(np.arange(n // 2) * n / (n // 2)).astype(int)
    h[idx] = 1.0
    return h

def make_rand_binary(n, rng):
    h = np.zeros(n, dtype=np.float64)
    h[rng.permutation(n)[:n // 2]] = 1.0
    return h

def make_block_binary(n, rng):
    half = n // 2
    h = np.zeros(n, dtype=np.float64)
    pos = 0
    count = 0
    while count < half and pos < n:
        blk = rng.randint(2, 5)
        for b in range(blk):
            if pos < n and count < half:
                h[pos] = 1.0
                count += 1
            pos += 1
        pos += rng.randint(1, 3)
    return h

def full_local_search(h, tlimit):
    t0 = time.time()
    c5 = compute_c5(h)
    while time.time() - t0 < tlimit:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_c5 = c5
        best_i, best_j = -1, -1
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                nc = compute_c5(h)
                if nc < best_c5 - 1e-12:
                    best_c5 = nc
                    best_i, best_j = int(i), int(j)
                h[i] = 1.0
                h[j] = 0.0
        if best_i >= 0:
            h[best_i] = 0.0
            h[best_j] = 1.0
            c5 = best_c5
        else:
            break
    return h, c5

def random_local_search(h, rng, tlimit, n_batch=200):
    t0 = time.time()
    c5 = compute_c5(h)
    while time.time() - t0 < tlimit:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_c5 = c5
        best_i, best_j = -1, -1
        for _ in range(n_batch):
            i = int(ones[rng.randint(len(ones))])
            j = int(zeros[rng.randint(len(zeros))])
            h[i] = 0.0
            h[j] = 1.0
            nc = compute_c5(h)
            if nc < best_c5 - 1e-12:
                best_c5 = nc
                best_i, best_j = i, j
            h[i] = 1.0
            h[j] = 0.0
        if best_i >= 0:
            h[best_i] = 0.0
            h[best_j] = 1.0
            c5 = best_c5
        else:
            break
    return h, c5

def simulate_anneal(h, rng, tlimit, T0=0.04, Tmult=0.9992):
    t0 = time.time()
    c5 = compute_c5(h)
    T = T0
    while time.time() - t0 < tlimit:
        T *= Tmult
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        i = int(ones[rng.randint(len(ones))])
        j = int(zeros[rng.randint(len(zeros))])
        h[i] = 0.0
        h[j] = 1.0
        nc = compute_c5(h)
        if nc <= c5 + 1e-12 or rng.random() < np.exp(-(nc - c5) / max(T, 1e-12)):
            c5 = nc
        else:
            h[i] = 1.0
            h[j] = 0.0
    return h, c5

    rng = np.random.RandomState(seed)
