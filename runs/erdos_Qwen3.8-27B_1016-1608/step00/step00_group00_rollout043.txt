    return max_Tc / T + np.log(np.sum(np.exp(Tc - max_Tc))) / T

    for _ in range(200):
        diff = h.sum() - target_sum
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h > 0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] -= diff / cnt
        else:
            mask = h < 1
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += (-diff) / cnt
        h = np.clip(h, 0.0, 1.0)
    # Final correction
    diff = h.sum() - target_sum
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = np.where(h > 0)[0]
            h[idx[0]] -= diff
        else:
            idx = np.where(h < 1)[0]
            h[idx[0]] += -diff
        h = np.clip(h, 0.0, 1.0)

    if max_c < -700 / T:

import numpy as np

def project_feasible(g, S):
    n = len(g)
    lo = g.min() - S
    hi = g.max() + S
    for _ in range(200):
        lam = 0.5 * (lo + hi)
        h = np.clip(g - lam, 0.0, 1.0)
        s = h.sum()
        if s > S:
            lo = lam
        else:
            hi = lam
    h = np.clip(g - 0.5 * (lo + hi), 0.0, 1.0)
    diff = h.sum() - S
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(h > 1e-15)[0]
            if len(idx) > 0:
                h[idx[0]] = max(0.0, h[idx[0]] - diff)
        else:
            idx = np.where(h < 1.0 - 1e-15)[0]
            if len(idx) > 0:
                h[idx[0]] = min(1.0, h[idx[0]] + (-diff))
    return h

def c5_val(h, dx):
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(c))

def sobj(h, dx, T):
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    m = c.max()
    if T * (c.max() - c.min()) > 500:
        return m
    sh = c - m
    return m + np.log(np.sum(np.exp(T * sh))) / T

def sgrad(h, dx, T):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    m = c.max()
    Tc = T * (c - m)
    p = np.exp(Tc)
    ps = p.sum()
    if ps > 0:
        p /= ps
    else:
        p = np.ones(2 * n - 1) / (2 * n - 1)
    grad = np.zeros(n)
    for mi in range(2 * n - 1):
        if p[mi] < 1e-16:
            continue
        lag = mi - (n - 1)
        j = np.arange(n)
        i1 = j - lag
        i2 = j + lag
        v1 = (i1 >= 0) & (i1 < n)
        v2 = (i2 >= 0) & (i2 < n)
        gc = np.zeros(n)
        gc[v1] = 1.0 - h[i1[v1]]
        if np.any(v2):
            gc[v2] -= h[i2[v2]]
        grad += p[mi] * gc * dx
    return grad
