        diff = target - s

            if cnt == 0: break
            h[mask] += diff / cnt

            mask = h > 0.0
            cnt = mask.sum()
            if cnt == 0: break

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    # Log-sum-exp
    m = c.max()
    obj = m + np.log(np.mean(np.exp(T*(c - m)))) / T
    # Weights
    w = np.exp(T * (c - m))
    w /= w.sum()
    # Gradient
    grad = np.zeros(n)
    for idx in range(2*n - 1):
        wi = w[idx]
        if wi < 1e-15: continue
        lag = idx - (n-1)
        # Vectorized: for valid m
        # term1: (1-h[m-lag]) for m where 0<=m-lag<n
        # term2: -h[m+lag] for m where 0<=m+lag<n
        m_range = np.arange(n)
        ml = m_range - lag
        valid1 = (ml >= 0) & (ml < n)
        grad[valid1] += wi * (1.0 - h[ml[valid1]])
        mp = m_range + lag
        valid2 = (mp >= 0) & (mp < n)
        grad[valid2] -= wi * h[mp[valid2]]
    return obj, grad

            obj_new, _ = compute_c5(h_new, n)
            if obj_new < obj or obj_new < best_obj + 0.01:
                h = h_new
                break
            lr *= 0.5
        else:
            h = project(h - lr * grad, n)

        if c5 < best_obj:
            best_obj = c5
            h_best = h.copy()

def swap_refine(h, n, max_iter=5000, time_limit=30):

    t0 = time.time()

        if d < 0 and h[j] - d > 1: continue
        h_new = h.copy()
        h_new[i] += d
        h_new[j] -= d
        h_new = np.clip(h_new, 0, 1)
        c5_new, _ = compute_c5(h_new, n)
        if c5_new < best_c5:
            best_c5 = c5_new
            h = h_new

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(c)), c

def project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_and_grad(h, n, T):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    m = c.max()
    obj = m + np.log(np.mean(np.exp(T * (c - m)))) / T
    w = np.exp(T * (c - m))
    w /= w.sum()
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-14:
            continue
        lag = idx - (n - 1)
        m_range = np.arange(n)
        ml = m_range - lag
        v1 = (ml >= 0) & (ml < n)
        grad[v1] += wi * (1.0 - h[ml[v1]])
        mp = m_range + lag
        v2 = (mp >= 0) & (mp < n)
        grad[v2] -= wi * h[mp[v2]]
    return obj, grad

def project_grad_descend(h, n, T=40.0, max_iter=3000, time_limit=300.0):

    best_c5, _ = compute_c5(h, n)
    h_best = h.copy()
    t0 = time.time()
    lr = 0.05
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        obj, grad = obj_and_grad(h, n, T)
        # Backtracking
        step = lr
        for _ in range(25):
            h_new = project(h - step * grad, n)
            c5_new, _ = compute_c5(h_new, n)
            if c5_new <= obj + 0.001:
                h = h_new
                break
            step *= 0.5
        else:
            h = project(h - step * grad, n)
        c5_now, _ = compute_c5(h, n)
        if c5_now < best_c5:
            best_c5 = c5_now
            h_best = h.copy()
        if it % 200 == 199:
            if c5_now > best_c5 * 1.02:
                lr *= 0.7
            else:
                lr = min(lr * 1.05, 0.3)
    return h_best, best_c5
