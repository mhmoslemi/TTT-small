    return float(np.max(corr) * dx)

def _repair_h(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)


def _fix(h, n):
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(50):
        d = target - h.sum()
        if abs(d) < 1e-10:
            break
        m = h < 1.0 if d > 0 else h > 0.0
        if not np.any(m):
            break
        h[m] += d / np.count_nonzero(m)
        h = np.clip(h, 0.0, 1.0)
    return h


def _soft_max(corr, beta):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp((corr - m) / beta)) / len(corr)) * beta
