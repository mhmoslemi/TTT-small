import numpy as np
from scipy import optimize

def compute_c5(h):
    """Compute C5 = max_k sum(h[n+k]*(1-h[n])) * dx"""
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return np.max(corr) * dx

def project_constraint(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum}"""
    n = len(h)
    # Bisection on mu: sum(clamp(h - mu, 0, 1)) = target_sum
    lo, hi = h.min() - 1, h.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        projected = np.clip(h - mid, 0, 1)
        s = projected.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def subgradient_step(h, lr):
    """One subgradient step for minimizing max_k O(k)"""
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    
    # Find the k that maximizes
    # corr has 2n-1 elements, indexed from -(n-1) to (n-1)
    k_max = np.argmax(corr) - (n - 1)
    
    # Compute gradient of O(k_max) w.r.t. h
    # O(k) = sum_n h[n+k](1-h[n]) * dx
    # For w.r.t. h[m]:
    #   If m = n+k: contribution is (1-h[m-k]) * dx (from first factor)
    #   If m = n: contribution is -h[m+k] * dx (from second factor)
    
    grad = np.zeros(n)
    k = k_max
    if k >= 0:
        # n ranges from 0 to n-1-k
        valid_n = range(n - k)
        for idx, i in enumerate(valid_n):
            n_idx = i
            nk_idx = i + k
            grad[nk_idx] += (1 - h[n_idx]) * dx
            grad[n_idx] += -h[nk_idx] * dx
    else:
        # k < 0, let k' = -k > 0
        # O(k) = sum_n h[n+k](1-h[n]) where n+k >= 0, n < n
        # n ranges from -k to n-1
        k_prime = -k
        valid_n = range(k_prime, n)
        for idx, i in enumerate(valid_n):
            n_idx = i
            nk_idx = i + k  # = i - k_prime
            grad[nk_idx] += (1 - h[n_idx]) * dx
            grad[n_idx] += -h[nk_idx] * dx
    
    return grad

    rng = np.random.RandomState(seed)

    else:
        n = 160
        h = np.ones(n) * 0.5

    target_sum = n / 2.0
    h = project_constraint(h, target_sum)
    
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    # Subgradient descent with decreasing learning rate
    max_iter = 50000
    for it in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.85:

        curr_c5 = compute_c5(h)
        if curr_c5 < best_c5:

            best_h = h.copy()
        
        # Subgradient step
        lr = 0.1 / (1 + 0.01 * it)
        grad = subgradient_step(h, lr)
        h_new = h - lr * grad
        h = project_constraint(h_new, target_sum)
    
    return best_h, best_c5, n

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, 'full')) * dx)

def all_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, 'full') * dx

def project(h, target):
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_k(h, k):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if k >= 0:
        m = n - k
        if m > 0:
            g[k:] += (1.0 - h[:m]) * dx
            g[:m] -= h[k:] * dx
    else:
        kp = -k
        m = n - kp
        if m > 0:
            g[:m] += (1.0 - h[kp:]) * dx
            g[kp:] -= h[:m] * dx
    return g

def top_grad(h, m=10, temp=8.0):
    n = len(h)
    ov = all_overlaps(h)
    idx = np.argsort(ov)[-m:]
    vals = ov[idx]
    w = np.exp((vals - vals.max()) * temp)
    w /= w.sum()
    g = np.zeros(n)
    for i in range(m):
        g += w[i] * grad_k(h, int(idx[i]) - (n - 1))
    return g

def optimize(h0, target, dur, m=10, lr=0.8, dec=0.995, freq=200):
    h = project(h0, target)
    best = h.copy()
    best_c = compute_c5(h)
    t0 = time.time()
    it = 0
    while time.time() - t0 < dur:
        h = project(h - lr * top_grad(h, m), target)
        it += 1
        if it % freq == 0:
            c = compute_c5(h)
            if c < best_c:
                best_c = c
                best = h.copy()
            lr *= dec
    return best, best_c

    t_start = time.time()

    try:
