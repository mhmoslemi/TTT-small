import numpy as np
from scipy.optimize import minimize
import time


def _c5_value(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _fix_sum(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def _obj_fun(x):
    return _c5_value(x)


def _con_fun(x):
    return np.sum(x) - len(x) / 2.0


def _slsqp(h0, maxiter=300, ftol=1e-12):
    n = len(h0)
    h0 = _fix_sum(h0.copy(), n)
    try:
        res = minimize(_obj_fun, h0, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': _con_fun}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_out = _fix_sum(res.x, n)
        return h_out, _c5_value(h_out)
    except Exception:
        return h0, _c5_value(h0)


def _nelder_perturb(h0, scale, rng, maxiter=500):
    n = len(h0)
    h0 = _fix_sum(h0 + rng.standard_normal(n) * scale, n)
    try:
        res = minimize(_obj_fun, h0, method='Nelder-Mead',
                       options={'maxiter': maxiter, 'xatol': 1e-9, 'fatol': 1e-12})
        h_out = _fix_sum(res.x, n)
        return h_out, _c5_value(h_out)
    except Exception:
        return h0, _c5_value(h0)


def _make_binary(n, rng):
    h = (rng.random(n) < 0.5).astype(float)
    target = n // 2
    ones_idx = np.where(h == 1.0)[0]
    zeros_idx = np.where(h == 0.0)[0]
    cnt_ones = len(ones_idx)
    if cnt_ones > target:
        perm = rng.permutation(cnt_ones)
        h[ones_idx[perm[:cnt_ones - target]]] = 0.0
    elif cnt_ones < target:
        perm = rng.permutation(len(zeros_idx))
        h[zeros_idx[perm[:target - cnt_ones]]] = 1.0
    return h


def _make_smooth(n, rng):
    freqs = rng.integers(1, 4, size=3)
    amps = rng.uniform(0.1, 0.3, size=3)
    phases = rng.uniform(0, 2 * np.pi, size=3)
    xs = np.linspace(0, 2, n)
    h = np.zeros(n)
    for f, a, p in zip(freqs, amps, phases):
        h += a * np.sin(f * np.pi * xs + p)
    h = 0.5 + h
    return np.clip(h, 0.0, 1.0)


def _make_block(n, rng):
    """Create a block-like structure: some 1s together, some 0s together."""
    h = np.zeros(n)
    half = n // 2
    offsets = rng.integers(0, n, size=half)
    h[offsets % n] = 1.0
    # Ensure exactly half are 1
    ones = int(np.sum(h))
    if ones > half:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:ones - half]] = 0
    elif ones < half:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:half - ones]] = 1
    return h
