    c = np.correlate(h, 1-h, mode='full')  # length 2n-1

    c = np.correlate(h, 1-h, mode='full')
    ac = alpha * c * dx
    m = ac.max()
    return (1.0/alpha) * (m + np.log(np.sum(np.exp(ac - m))))

def smooth_max_grad(h, alpha, dx):
    n = len(h)
    c = np.correlate(h, 1-h, mode='full')
    ac_orig = alpha * c * dx
    m = ac_orig.max()
    ac = ac_orig - m
    exp_ac = np.exp(ac)
    p = exp_ac / exp_ac.sum()
    
    lags = np.arange(2*n-1) - (n-1)
    grad = np.zeros(n)
    for idx in range(len(lags)):
        ell = lags[idx]
        w = p[idx]
        if w < 1e-15:
            continue
        if ell >= 0:
            if ell < n:
                grad[:n-ell] += w * (1.0 - h[ell:])
                if ell > 0:
                    grad[ell:] -= w * h[:n-ell]
        else:
            e = -ell
            if e < n:
                grad[e:] += w * (1.0 - h[:n-e])
                if e > 0:
                    grad[:n-e] -= w * h[e:]
    grad *= dx
    return grad

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)

            floor = h
            if floor.sum() < 1e-15:
                break
            h = h - (-diff) * floor / floor.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    c = np.correlate(h, g, 'full')
    ac = alpha * c * dx
    m = ac.max()
    exp_ac = np.exp(ac - m)
    s = exp_ac.sum()
    p = exp_ac / s
    fval = (1.0 / alpha) * (m + np.log(s))
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = p[k]
        if w < 1e-14:
            continue
        ell = k - (n - 1)
        if ell >= 0:
            grad[:n - ell] += w * (1.0 - h[ell:])
            if ell > 0:
                grad[ell:] -= w * h[:n - ell]
        else:
            e = -ell
            grad[e:] += w * (1.0 - h[:n - e])
            if e > 0:
                grad[:n - e] -= w * h[e:]
    grad *= dx
    return fval, grad

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            rs = room.sum()
            if rs < 1e-15:
                break
            h = h + diff * room / rs
        else:
            floor = h
            fs = floor.sum()
            if fs < 1e-15:
                break
            h = h + diff * floor / fs
        h = np.clip(h, 0.0, 1.0)
    return h

def projected_gradient(h0, n_iters, alpha, lr0, lr_decay):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    lr = lr0
    for i in range(n_iters):
        fval, grad = obj_and_grad(h, alpha)
        step = lr
        h_new = h
        for _ in range(12):
            cand = project_feasible(h - step * grad, n)
            c5 = compute_c5(cand)
            if c5 < cur_c5 - 1e-12:
                h_new = cand
                break
            step *= 0.5
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
        cur_c5 = c5
        lr *= lr_decay
        if lr < 1e-6:
            lr = 1e-6
    return best_h, best_c5

def coordinate_descent(h, n_rounds, step_size):
    n = len(h)
    best_c5 = compute_c5(h)
    h_work = h.copy()
    for r in range(n_rounds):
        sw = step_size
        improved = False
        for j in range(n):
            for d in (-sw, sw):
                h_work[j] += d
                if h_work[j] < 0.0 or h_work[j] > 1.0:
                    h_work[j] -= d
                    continue
                h_work[j] = np.clip(h_work[j], 0.0, 1.0)
                cand = project_feasible(h_work, n)
                c5 = compute_c5(cand)
                if c5 < best_c5 - 1e-14:
                    h_work = cand
                    best_c5 = c5
                    improved = True
                else:
                    h_work[j] -= d
                    h_work[j] = np.clip(h_work[j], 0.0, 1.0)
        if not improved:
            step_size *= 0.6
        if step_size < 1e-5:
            break
    return h_work, best_c5
