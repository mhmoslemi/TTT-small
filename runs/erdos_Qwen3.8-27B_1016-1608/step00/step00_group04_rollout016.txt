import numpy as np
import time

def compute_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def make_binary(h, n_ones, rng):
    n = len(h)
    b = (h > 0.5).astype(float)
    total = int(np.sum(b))
    if total > n_ones:
        idx = np.flatnonzero(b)
        rng.shuffle(idx)
        for i in range(total - n_ones):
            b[idx[i]] = 0
    elif total < n_ones:
        idx = np.flatnonzero(1 - b)
        rng.shuffle(idx)
        for i in range(n_ones - total):
            b[idx[i]] = 1
    return b

def greedy_local_search(h, rng, max_passes=200):
    n = len(h)
    h = h.copy()
    score = compute_score(h)
    for _ in range(max_passes):
        moved = False
        idx_ones = np.flatnonzero(h)
        idx_zeros = np.flatnonzero(1 - h)
        if len(idx_ones) == 0 or len(idx_zeros) == 0:
            break
        best_gain = 0
        best_move = None
        for i in idx_ones:
            for j in idx_zeros:
                h[i] = 0
                h[j] = 1
                s = compute_score(h)
                h[j] = 0
                h[i] = 1
                gain = score - s

                    best_gain = gain
                    best_move = (i, j)
        if best_move is not None:
            i, j = best_move
            h[i] = 0
            h[j] = 1
            score -= best_gain
            moved = True
        if not moved:
            break
    return h, score
