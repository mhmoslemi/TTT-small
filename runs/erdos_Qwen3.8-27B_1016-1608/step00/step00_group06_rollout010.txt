import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def make_feasible(h, target=None):
    """Adjust h to satisfy 0<=h<=1 and sum=target."""
    n = len(h)
    if target is None:
        target = n / 2.0
    h = np.clip(h, 0, 1)
    diff = target - h.sum()
    # Distribute adjustment
    if abs(diff) > 1e-10:
        # Adjust freely
        mask = (h > 0) & (h < 1)
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0, 1)
            # Redistribute remaining
            remaining = target - h.sum()
            if remaining > 0:
                mask2 = h < 1
            else:
                mask2 = h > 0
            if mask2.any():
                h[mask2] += remaining / mask2.sum()
                h = np.clip(h, 0, 1)
    return h

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_c5_vec(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def make_feasible(h, target=None):
    n = len(h)
    if target is None:
        target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # Proportional adjustment
        interior = (h > 1e-6) & (h < 1 - 1e-6)
        if interior.sum() > 0:
            h[interior] += diff / interior.sum()

        if remaining > 0:
            can_inc = h < 1.0
        else:
            can_inc = h > 0.0

def make_feasible(h, target):
    n = len(h)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def make_feasible(h, target):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(h, descending=True) if diff > 0 else np.argsort(h)
        for i in order:
            if abs(diff) < 1e-12:
                break
            if diff > 0:
                adj = min(1.0 - h[i], diff)
            else:
                adj = max(h[i] - 1.0, diff)
            h[i] += adj
            diff -= adj
    return h

def random_binary_h(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    if n % 2 == 1:
        rest = np.setdiff1d(np.arange(n), idx)
        h[rest[0]] = 0.5
    return h

def structured_h(n, rng):
    """Try a few structured patterns as alternatives."""
    h = np.zeros(n)
    pattern = rng.integers(0, 3)
    if pattern == 0:
        # Alternating
        h[::2] = 1.0
        if n % 2 == 1:
            h[-1] = 0.5
    elif pattern == 1:
        # Block
        h[:n // 2] = 1.0
        if n % 2 == 1:
            h[n // 2] = 0.5
    else:
        # Random
        h = random_binary_h(n, rng)
    return make_feasible(h, n / 2.0)

def local_search(h0, max_iters=30000, time_limit=None, t_ref=None):
    """Exhaustive swap local search on discrete structure."""
    n = len(h0)
    h = h0.copy()
    current = compute_c5(h)
    
    for it in range(max_iters):
        if time_limit is not None and t_ref is not None:
            if time.time() - t_ref > time_limit:
                break
        
        best_delta = -1e-14

        lows = np.where(h < 0.5)[0]
        mids = np.where(np.abs(h - 0.5) < 0.01)[0]
        
        # Swaps between high and low
        for i in highs:
            for j in lows:
                h[i], h[j] = h[j], h[i]
                c5 = compute_c5(h)
                h[i], h[j] = h[j], h[i]
                delta = c5 - current
                if delta < best_delta:
                    best_delta = delta
                    best_pair = (i, j)
        
        # Swaps involving mids (for odd n)
        if len(mids) > 0:
            for m in mids:
                for i in highs:
                    h[m], h[i] = h[i], h[m]
                    c5 = compute_c5(h)
                    h[m], h[i] = h[i], h[m]
                    delta = c5 - current
                    if delta < best_delta:
                        best_delta = delta
                        best_pair = (m, i)
                for j in lows:
                    h[m], h[j] = h[j], h[m]

                    h[m], h[j] = h[j], h[m]

                        best_pair = (m, j)

            h[best_pair[0]], h[best_pair[1]] = h[best_pair[1]], h[best_pair[0]]
            current += best_delta
        else:
            break
    
    return h, current

def perturbation_search(h0, rng, n_perturb=4, max_iters=10000, time_limit=None, t_ref=None):
    """Perturb then local search to escape local optima."""
    n = len(h0)
    h = h0.copy()
    current = compute_c5(h)
    
    # Perturb: randomly swap n_perturb pairs
    highs = np.where(h > 0.5)[0]
    lows = np.where(h < 0.5)[0]
    if len(highs) >= n_perturb and len(lows) >= n_perturb:
        swap_h = rng.choice(highs, size=n_perturb, replace=False)
        swap_l = rng.choice(lows, size=n_perturb, replace=False)
        for a, b in zip(swap_h, swap_l):
            h[a], h[b] = h[b], h[a]
    
    h, current = local_search(h, max_iters=max_iters, time_limit=time_limit, t_ref=t_ref)
    return h, current
