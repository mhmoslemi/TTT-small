    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def fix_sum(h, target):
    """Adjust h so that sum(h) = target, keeping 0<=h<=1."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    free = (h > 1e-8) & (h < 1.0 - 1e-8)

            for ht in [h_try, h_try2]:
                if time.time() - t0 > t_limit3 - 5:
                    break
                sub_lim = t_limit3 - 5
                h_opt, c5_opt = coord_descent(ht, max_sweeps=40, t0=t0, t_limit=sub_lim)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt

    best_h = fix_sum(best_h, len(best_h) / 2.0)
    best_c5 = c5_of(best_h)
    return (best_h, best_c5, len(best_h))
