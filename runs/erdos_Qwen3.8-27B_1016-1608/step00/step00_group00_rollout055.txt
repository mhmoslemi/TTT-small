            if c5n < best_c5:
                best_h, best_c5 = hnew, c5n
        except Exception:
            c5m = compute_c5(hm)
            if c5m < best_c5:
                best_h, best_c5 = hm.copy(), c5m

    # Phase 2: extensive random restarts
    attempt = 0
    while tl() > 15:
        if attempt % 3 == 0:
            hst = best_h + rng.standard_normal(n) * (0.04 + 0.015 * (attempt % 10))
        elif attempt % 3 == 1:
            hst = (rng.random(n) > 0.5).astype(float)
        else:
            hst = 0.5 + 0.5 * rng.standard_normal(n)
        hst = make_feasible(hst, n)

        try:
            r = minimize(obj, hst, method='SLSQP', bounds=bnds,
                        constraints=cons, options={'maxiter': 150, 'ftol': 1e-14})
            hnew = make_feasible(r.x, n)
            c5n = compute_c5(hnew)
            if c5n < best_c5:
                best_h, best_c5 = hnew, c5n
        except Exception:
            pass
        attempt += 1

    # Phase 3: local polishing from best
    if tl() > 10:
        for _ in range(5):
            hpol = best_h + rng.standard_normal(n) * 0.01
            hpol = make_feasible(hpol, n)
            try:
                r = minimize(obj, hpol, method='SLSQP', bounds=bnds,
                            constraints=cons, options={'maxiter': 300, 'ftol': 1e-16})
                hnew = make_feasible(r.x, n)
                c5n = compute_c5(hnew)
                if c5n < best_c5:
                    best_h, best_c5 = hnew, c5n
            except Exception:
                pass

    best_h = make_feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
