= δ(n+lag, j) * (1-h[n]) - h[n+lag] * δ(n, j)

= (1-h[j-lag]) * [0 ≤ j-lag < n AND j is the n+lag, so n = j-lag must be valid: 0 ≤ j-lag < n]

Actually, the projection onto the intersection of the box [0,1]^n and the hyperplane sum(x)=c can be done by binary search on a Lagrange multiplier:
project(h, c): find λ such that sum(clip(h_i + λ, 0, 1)) = c

This is because the projection is: min ||x - h||² s.t. 0≤x_i≤1, sum(x)=c
Lagrangian: 1/2 Σ(x_i - h_i)² + λ(sum(x)-c) + Σ(μ_i x_i + ν_i(1-x_i))
KKT: x_i = clip(h_i - λ, 0, 1) (with sign conventions)

So we binary search on λ. This is very efficient!

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(x, target_sum):
    """Project x onto {0<=x_i<=1, sum(x)=target_sum} via bisection on Lagrange multiplier."""
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        projected = np.clip(x + mid, 0, 1)
        s = np.sum(projected)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    projected = np.clip(x + (lo + hi) / 2, 0, 1)
    # Fine-tune
    diff = target_sum - np.sum(projected)
    if abs(diff) > 1e-12:
        free = (projected > 0) & (projected < 1)
        if np.any(free):
            projected[free] += diff / np.sum(free)
            projected = np.clip(projected, 0, 1)
    return projected

def smooth_objective(x, alpha=100.0):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full") * dx
    m = np.max(corr)
    if m < -100:
        return 0.0
    shifted = alpha * (corr - m)
    return m + np.log(np.sum(np.exp(shifted))) / alpha

def smooth_gradient(x, alpha=100.0):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full") * dx
    m = np.max(corr)
    if m < -100:
        return np.zeros_like(x)
    exp_vals = np.exp(alpha * (corr - m))
    weights = exp_vals / np.sum(exp_vals)
    
    # Gradient of corr[k] w.r.t. x[j]
    lags = np.arange(-(n-1), n)
    grad = np.zeros(n)
    for k in range(2*n-1):
        if weights[k] < 1e-10:
            continue
        lag = lags[k]
        # d corr[k]/d x[j] = dx * [(1-x[j-lag])*I[0<=j-lag<n] - x[j]*I[0<=j+lag<n]]
        j1_min, j1_max = max(0, lag), min(n-1, n-1+lag)
        j2_min, j2_max = max(0, -lag), min(n-1, n-1-lag)
        if j1_min <= j1_max:
            grad[j1_min:j1_max+1] += weights[k] * dx * (1 - x[j1_min-lag:j1_max+1-lag])
        if j2_min <= j2_max:
            grad[j2_min:j2_max+1] -= weights[k] * dx * x[j2_min:j2_max+1]
    return grad

def projected_gradient_descent(h_init, alpha=100.0, lr=None, max_iter=5000, tol=1e-8):
    n = len(h_init)
    target_sum = n / 2.0
    h = h_init.copy()
    h = project_box_sum(h, target_sum)
    
    hist = []
    prev_obj = smooth_objective(h, alpha)
    
    # Adaptive learning rate
    step = lr if lr else 0.01
    momentum = 0.9
    
    v = np.zeros(n)
    
    for it in range(max_iter):
        obj = smooth_objective(h, alpha)
        grad = smooth_gradient(h, alpha)
        
        hist.append(obj)
        
        if abs(prev_obj - obj) < tol and it > 10:
            break
        prev_obj = obj
        
        # Momentum
        v = momentum * v - step * grad
        h_new = h + v
        h_new = project_box_sum(h_new, target_sum)
        
        # Line search (simple backtracking)
        ls_obj = smooth_objective(h_new, alpha)
        if ls_obj > obj + 1e-4 * step * np.dot(grad, h_new - h):
            # Reduce step
            step *= 0.5
            h_new = h - 0.5 * step * grad
            h_new = project_box_sum(h_new, target_sum)
        
        h = h_new
    
    return h

    return float(np.max(corr) * dx)

def project_box_sum(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(x + mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid

        free = (proj > 1e-14) & (proj < 1.0 - 1e-14)
        if np.any(free):
            proj[free] += diff / np.sum(free)
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def smooth_obj_grad(x, alpha=100.0):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    m = np.max(corr)
    if m < -500.0:
        return 0.0, np.zeros(n)
    e = np.exp(alpha * (corr - m))
    se = np.sum(e)
    obj = m + np.log(se) / alpha
    w = e / se
    lags = np.arange(-(n - 1), n)

        if j2a <= j2b:

        i += 1

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_box_sum(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(x + mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - np.sum(proj)
    if abs(diff) > 1e-14:
        free = (proj > 1e-14) & (proj < 1.0 - 1e-14)
        if np.any(free):
            proj[free] += diff / np.sum(free)
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def smooth_obj_grad(x, alpha=100.0):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    m = np.max(corr)
    if m < -500.0:
        return 0.0, np.zeros(n)
    e = np.exp(alpha * (corr - m))
    se = np.sum(e)
    obj = m + np.log(se) / alpha
    w = e / se
    lags = np.arange(-(n - 1), n)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        lag = lags[k]
        j1a, j1b = max(0, lag), min(n - 1, n - 1 + lag)
        if j1a <= j1b:
            idx = j1a - lag
            grad[j1a:j1b + 1] += wk * dx * (1.0 - x[idx:idx + (j1b - j1a + 1)])
        j2a, j2b = max(0, -lag), min(n - 1, n - 1 - lag)
        if j2a <= j2b:
            idx2 = j2a + lag
            grad[j2a:j2b + 1] -= wk * dx * x[idx2:idx2 + (j2b - j2a + 1)]
    return obj, grad


def optimize_pgd(h_init, target_sum, alpha=100.0, max_iter=3000, time_limit=None, t_start=None):
    n = len(h_init)
    h = project_box_sum(h_init.copy(), target_sum)
    step = 0.015
    prev_obj = smooth_obj_grad(h, alpha)[0]
    momentum_v = np.zeros(n)
    mu = 0.7
    for it in range(max_iter):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        obj, grad = smooth_obj_grad(h, alpha)
        lr = step * (0.97 ** (it // 100))
        momentum_v = mu * momentum_v - lr * grad
        h_new = project_box_sum(h + momentum_v, target_sum)
        obj_new, _ = smooth_obj_grad(h_new, alpha)
        bt = 0
        while obj_new > obj + 1e-8 and bt < 8:
            lr *= 0.5
            momentum_v = mu * momentum_v - lr * grad
            h_new = project_box_sum(h + momentum_v, target_sum)
            obj_new, _ = smooth_obj_grad(h_new, alpha)
            bt += 1
        h = h_new
        if abs(prev_obj - obj_new) < 1e-12:
            break
        prev_obj = obj_new
    return h


def slsqp_refine(h_init, target_sum, alpha=200.0, max_iter=300):
    n = len(h_init)

    def objf(x):
        return smooth_obj_grad(x, alpha)[0]

    def gradf(x):
        return smooth_obj_grad(x, alpha)[1]

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(objf, project_box_sum(h_init.copy(), target_sum),
                       jac=gradf, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-14})
        return project_box_sum(res.x, target_sum)
    except Exception:
        return h_init


def make_initial_patterns(n, seed):
    rng = np.random.RandomState(seed)
    patterns = []
    # Block pattern
    h_block = np.zeros(n)
    h_block[:n // 2] = 1.0
    patterns.append(h_block)
    # Sine
    x = np.linspace(0, 2, n)
    h_sine = np.clip(0.5 + 0.4 * np.sin(2 * np.pi * x / 2.0), 0, 1)
    patterns.append(h_sine)
    # Cosine 2
    h_cos2 = np.clip(0.5 + 0.4 * np.cos(4 * np.pi * x / 2.0), 0, 1)
    patterns.append(h_cos2)
    # Random binary-ish
    h_rand = rng.randint(0, 2, n).astype(float) * 0.8 + 0.1
    patterns.append(h_rand)
    return patterns
