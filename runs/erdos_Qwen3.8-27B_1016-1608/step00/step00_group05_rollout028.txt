def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        else:
            mask = h > 1e-15
            if not np.any(mask):

            h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)

def compute_gradient_vec(h, n, t):
    c = np.correlate(h, 1.0 - h, mode="full")
    c_max = np.max(c)
    exp_c = np.exp(t * (c - c_max))
    w = exp_c / np.sum(exp_c)
    grad = np.zeros(n)
    for p in range(2 * n - 1):
        wp = w[p]
        if wp < 1e-15:
            continue
        d = p - n + 1

        else:

            idx = n - dp
            if idx > 0:
                grad[dp:] += wp * (1.0 - h[:idx])
            if dp < n:
                grad[:idx] -= wp * h[dp:]
    return grad
