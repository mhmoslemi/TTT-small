```python
    n = len(h)
    corr = compute_correlation(h, dx)

        threshold = max_val - 0.01
```

```python
def compute_gradient(h, dx, alpha=5.0):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_val = corr.max()
    # Softmax weights
    w = np.exp(alpha * (corr - max_val))
    w /= w.sum()
    
    grad = np.zeros(n)
    for j_idx in range(2*n - 1):
        if w[j_idx] < 1e-8:
            continue
        s = j_idx - (n - 1)
        # dO_s/dh[j] = (1-h[j+s])*I(valid) - h[j-s]*I(valid)
        # For all j:
        j = np.arange(n)
        # First term: (1-h[j+s]) where 0<=j+s<n
        valid1 = (j + s >= 0) & (j + s < n)
        grad[valid1] += w[j_idx] * (1.0 - h[j[valid1] + s])
        # Second term: -h[j-s] where 0<=j-s<n
        valid2 = (j - s >= 0) & (j - s < n)
        grad[valid2] += w[j_idx] * (-h[j[valid2] - s])
    
    return grad * dx
```

```python
import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

def get_correlation(h, dx):
    return np.correlate(h, 1.0 - h, mode='full').astype(np.float64) * dx

def compute_grad(h, dx, alpha=5.0):
    n = len(h)
    corr = get_correlation(h, dx)
    mx = float(corr.max())
    w = np.exp(alpha * (corr - mx))
    ws = float(w.sum())
    if ws < 1e-15:
        w = np.zeros_like(corr)
        w[int(corr.argmax())] = 1.0
    else:
        w /= ws
    grad = np.zeros(n)
    j = np.arange(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-10:
            continue
        s = k - (n - 1)
        n_min = max(0, -s)
        n_max = n - 1 - max(0, s)
        if n_min > n_max:
            continue
        # Term 2: -h[j+s] for j in [n_min, n_max]
        grad[n_min:n_max + 1] -= wk * h[n_min + s:n_max + 1 + s]
        # Term 1: (1-h[j-s]) for j where n_min <= j-s <= n_max
        j_lo = n_min + s
        j_hi = n_max + s
        if j_lo <= j_hi:
            grad[j_lo:j_hi + 1] += wk * (1.0 - h[n_min:n_max + 1])
    return grad * dx

def project(h, n):
    target = n / 2.0
    lo = float(h.min()) - 1.5
    hi = float(h.max()) + 1.5
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        hp = np.clip(h - mid, 0.0, 1.0)
        if float(hp.sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def gen_inits(n_pts, rng, init):
    dx = 2.0 / n_pts
    x = (np.arange(n_pts) + 0.5) * dx
    inits = []
    if init is not None:
        if len(init) == n_pts:
            inits.append(init.copy())
        else:
            x_old = (np.arange(len(init)) + 0.5) * 2.0 / len(init)
            inits.append(np.interp(x, x_old, init))
    inits.append(0.5 + 0.4 * np.sin(2 * np.pi * x))
    inits.append(0.5 + 0.4 * np.cos(4 * np.pi * x))
    inits.append(0.5 + 0.3 * np.sin(2 * np.pi * x) + 0.25 * np.sin(6 * np.pi * x))
    inits.append(0.5 + 0.35 * np.sin(3 * np.pi * x) + 0.15 * np.cos(9 * np.pi * x))
    inits.append(0.5 + 0.3 * np.sin(2 * np.pi * x * 2.5))
    for _ in range(3):
        inits.append(0.5 + 0.35 * rng.randn(n_pts))
    return inits

def optimize_single(h, n_pts, rng, t0, budget_s, alpha=5.0):
    dx = 2.0 / n_pts
    c5 = compute_c5(h, dx)
    best_c5 = c5
    best_h = h.copy()
    step = 0.1
    stall = 0
    for it in range(8000):
        if time.time() - t0 > budget_s - 12:
            break
        grad = compute_grad(h, dx, alpha)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        improved = False
        for _ls in range(20):
            h_new = project(h - step * grad, n_pts)
            c5_new = compute_c5(h_new, dx)
            if c5_new < c5 - 1e-14:
                h = h_new
                c5 = c5_new
                improved = True
                stall = 0
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                break
            step *= 0.5
            if step < 1e-9:
                break
        if not improved:
            stall += 1
            if stall > 50:
                h = h + 0.08 * rng.randn(n_pts)
                h = project(h, n_pts)
                c5 = compute_c5(h, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                step = 0.1
                stall = 0
            else:
                step = min(step * 1.5, 0.3)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    init = None
    try:
        if 'initial_h_values' in globals():
            init = np.array(initial_h_values, dtype=np.float64)
    except Exception:
        pass

    best_h = None
    best_c5 = np.inf
    best_n = 0

    resolutions = [80, 100, 120, 160, 200, 256]

    for n_pts in resolutions:
        if time.time() - t0 > budget_s - 50:
            break
        if n_pts > 320:
            break
        inits = gen_inits(n_pts, rng, init)
        # Budget per resolution
        rem = budget_s - 40 - (time.time() - t0)
        per_init = max(20, rem / max(len(inits), 1))

        for h0 in inits:
            if time.time() - t0 > budget_s - 15:
                break
            h0 = project(h0, n_pts)
            h_opt, c5_opt = optimize_single(h0, n_pts, rng, t0, budget_s)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_pts

        if best_h is not None:
            init = best_h

    if best_h is None:
        n_pts = 80
        best_h = np.full(n_pts, 0.5)
        best_c5 = compute_c5(best_h, 2.0 / n_pts)
        best_n = n_pts

    dx = 2.0 / best_n
    best_c5 = compute_c5(best_h, dx)

    return best_h, best_c5, best_n
```
