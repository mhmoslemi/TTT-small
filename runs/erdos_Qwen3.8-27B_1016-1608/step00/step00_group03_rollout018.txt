                c5_res = compute_c5(h_res, dx)
                if c5_res < best_c5:
                    best_h = h_res
                    best_c5 = c5_res
        except Exception:
            pass

    # Phase 2: Multiple scipy restarts from perturbed starts
    for attempt in range(8):
        if budget_s - elapsed() < 40:
            break
        try:
            noise_scale = rng.uniform(0.03, 0.12)
            h_start = best_h + rng.normal(0, noise_scale, n)
            h_start = make_feasible(h_start, n)
            res = scipy_optimize(h_start, n, dx, target_sum, 500)
            if res.x is not None:
                h_res = make_feasible(res.x, n)
                c5_res = compute_c5(h_res, dx)
                if c5_res < best_c5:
                    best_h = h_res
                    best_c5 = c5_res
        except Exception:
            pass

    # Phase 3: Simulated Annealing
    T = 0.003
    T_min = 1e-9
    cooling = 0.99998
    step_size = 0.08
    current_h = best_h.copy()
    current_c5 = best_c5
    no_improve = 0
    best_T_reset = 0.002
    iterations = 0

    while budget_s - elapsed() > 15:
        # Alternate perturbation types
        if iterations % 5 == 4:
            h_new = sa_perturb_block(current_h, n, rng)
        else:
            h_new = sa_perturb(current_h, n, step_size, rng)

        if h_new is not None:
            # Ensure sum constraint
            s = h_new.sum()
            diff = target_sum - s
            if abs(diff) > 1e-10:
                idx = np.where((h_new > 1e-8) & (h_new < 1.0 - 1e-8))[0]
                if len(idx) > 0:
                    h_new[idx] += diff / len(idx)
                    h_new = np.clip(h_new, 0.0, 1.0)
                else:
                    h_new = None

        if h_new is not None:
            c5_new = compute_c5(h_new, dx)
            if c5_new < current_c5:
                current_h = h_new
                current_c5 = c5_new
                no_improve = 0
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    best_T_reset = max(0.001, T * 0.7)
            else:
                d = c5_new - current_c5
                if d / T < 500.0:
                    prob = np.exp(-d / T)
                else:
                    prob = 0.0
                if rng.random() < prob:
                    current_h = h_new
                    current_c5 = c5_new
                no_improve += 1

        T *= cooling
        if T < T_min:
            T = T_min

        # Adaptive step size
        if no_improve > 4000:
            step_size *= 0.92
            no_improve = 0

        # Restart from best if stuck
        if no_improve > 15000:
            current_h = best_h + rng.normal(0, 0.04, n)
            current_h = make_feasible(current_h, n)
            current_c5 = compute_c5(current_h, dx)
            T = best_T_reset
            step_size = 0.07
            no_improve = 0

        iterations += 1

    # Phase 4: Final polish with scipy from best SA result
    if budget_s - elapsed() > 20:
        try:
            res = scipy_optimize(best_h, n, dx, target_sum, 800)
            if res.x is not None:
                h_res = make_feasible(res.x, n)
                c5_res = compute_c5(h_res, dx)
                if c5_res < best_c5:
                    best_h = h_res
                    best_c5 = c5_res
        except Exception:
            pass

    # Final validation
    best_h = make_feasible(best_h, n)
    best_c5 = compute_c5(best_h, dx)

    return best_h, best_c5, n
