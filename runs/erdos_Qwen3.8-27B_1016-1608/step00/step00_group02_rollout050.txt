    """Project h onto {0 ≤ h ≤ 1, sum(h) = target}."""
    n = len(h)
    h = np.clip(h, 0, 1)
    # Iterative water-filling
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # Add mass to elements below 1
            mask = h < 1.0 - 1e-12
            room = (1.0 - h)
            room[~mask] = 0
            if room.sum() > 0:

        else:

            avail = h.copy()
            avail[~mask] = 0
            if avail.sum() > 0:

    max_val = np.max(corr)
    # Active set: lags within some tolerance of max
    active = np.where(corr >= max_val - 1e-6)[0]
    
    # Subgradient: average over active lags

import numpy as np

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    n = len(h)
    h = h.copy()
    for _ in range(60):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = np.clip(1.0 - h, 0, None)
            if room.sum() < 1e-14:
                break
            h += room * (diff / room.sum())
        else:
            avail = np.clip(h, 0, None)
            if avail.sum() < 1e-14:
                break
            h -= avail * (-diff / avail.sum())
    return np.clip(h, 0, 1)

def compute_subgrad(h, active_lags):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    j_vals = np.arange(n)
    for k in active_lags:
        d = k - (n - 1)
        jp = j_vals + d
        jm = j_vals - d
        valid_p = (jp >= 0) & (jp < n)
        valid_m = (jm >= 0) & (jm < n)
        sg = np.zeros(n)
        sg[valid_p] += (1.0 - h[jp[valid_p]])
        sg[valid_m] -= h[jm[valid_m]]
        grad += sg
    grad /= max(len(active_lags), 1)
    grad *= dx
    return grad

        dx = 2.0 / n
