def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx

    h = (h > 0.5).astype(float)

    diff = target - int(np.sum(h))
    if diff > 0:

        h[rng.choice(zeros, diff, replace=False)] = 1

        h[rng.choice(ones, -diff, replace=False)] = 0

    for _ in range(max_rounds):

            if c < bc:
                bc, bh, bn = c, h.copy(), nv
            h, c = sa(h, nv, seed + nv * 31 + t, 60000)
            if c < bc:
                bc, bh, bn = c, h.copy(), nv

    # Phase 4: Intensive SA on best
    if time.time() - t0 < budget_s * 0.85:
        r4 = np.random.default_rng(seed + 999)
        for t in range(5):
            if time.time() - t0 > budget_s * 0.92:
                break
            hi = bh.copy()
            for _ in range(bn // 4):
                hi[int(r4.integers(0, bn))] = 1.0 - hi[int(r4.integers(0, bn))]
            h, c = sa(hi, bn, seed + 999 + t, 100000)
            if c < bc:
                bc, bh, bn = c, h.copy(), bn

    # Phase 5: Final local search polish
    if time.time() - t0 < budget_s * 0.95:
        h, c = local_search(bh, bn, 200, np.random.default_rng(seed + 777))
        if c < bc:
            bc, bh = c, h.copy()

    # Verify
    assert 0 <= np.min(bh) and np.max(bh) <= 1.0
    assert abs(np.sum(bh) - bn / 2.0) < 1e-6
    return bh, bc, bn
