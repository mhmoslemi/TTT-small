import numpy as np
from scipy.optimize import minimize
import time

def _c5_vec(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _c5_max(h):
    return float(np.max(_c5_vec(h)))

def _project(x, n):
    x = np.clip(x, 0.0, 1.0)
    x += (n / 2.0 - np.sum(x)) / n
    return np.clip(x, 0.0, 1.0)

def _smooth_obj_factory(n, dx, beta):
    def obj(x):
        cv = _c5_vec(x)
        m = cv.max()
        e = np.exp(cv - m)
        return m + np.log(np.sum(e)) / beta
    return obj

def _binary_polish(h, n, max_rounds=50):
    """Greedy rounding: fix positions one by one to 0 or 1, adjust the free one."""
    h = h.copy()
    improved = True
    rounds = 0
    while improved and rounds < max_rounds:
        improved = False
        rounds += 1
        order = np.argsort(h)[::-1]
        for j in order:
            trial_1 = h.copy()
            trial_1[j] = 1.0
            trial_1 = _project(trial_1, n)
            c5_1 = _c5_max(trial_1)
            if c5_1 < _c5_max(h) - 1e-12:
                h = trial_1
                improved = True
            trial_0 = h.copy()
            trial_0[j] = 0.0
            trial_0 = _project(trial_0, n)
            c5_0 = _c5_max(trial_0)
            if c5_0 < _c5_max(h) - 1e-12:
                h = trial_0
                improved = True
    return h
