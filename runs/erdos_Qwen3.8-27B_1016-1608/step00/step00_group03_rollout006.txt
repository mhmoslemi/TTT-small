                best_c5 = new_c5

                last_improve = n_evals
        else:
            h[i] -= delta
            h[j] += delta
        
        T *= cool
        if T < T_min:
            T = max(0.001, best_c5 * 0.02)
            delta_range = max(0.005, delta_range * 0.7)
        
        if n_evals - last_improve > 200000 and T > 0.0005:
            T *= 0.3
            delta_range *= 0.6
        
        if n_evals % 500000 == 0:
            print(f"  SA evals={n_evals}, best={best_c5:.8f}, T={T:.6f}")
    
    return best_h, best_c5

def local_refine(h, n_points, seed, time_limit):
    rng = np.random.RandomState(seed)
    target_sum = n_points / 2.0
    end_time = time.time() + time_limit
    
    current_c5 = compute_c5(h, n_points)
    best_c5 = current_c5
    best_h = h.copy()
    step = 0.008
    
    for it in range(500000):
        if time.time() > end_time:
            break
        i, j = rng.randint(0, n_points, 2)
        if i == j:
            continue
        improved = False
        for sign in [1, -1]:
            d = sign * step
            if h[i] + d < -1e-12 or h[i] + d > 1 + 1e-12:
                continue
            if h[j] - d < -1e-12 or h[j] - d > 1 + 1e-12:
                continue
            h[i] += d
            h[j] -= d
            new_c5 = compute_c5(h, n_points)
            if new_c5 < current_c5:
                current_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                improved = True
                step = min(step * 1.02, 0.015)
                break
            else:
                h[i] -= d
                h[j] += d
        if not improved:
            step *= 0.85
            if step < 1e-7:
                break
    
    return best_h, best_c5

def subgradient_refine(h, n_points, seed, time_limit):
    rng = np.random.RandomState(seed)
    target_sum = n_points / 2.0
    end_time = time.time() + time_limit
    eps = 1e-5
    step_size = 0.003
    
    current_c5 = compute_c5(h, n_points)
    best_c5 = current_c5
    best_h = h.copy()
    
    for it in range(50000):
        if time.time() > end_time:
            break
        # Finite-difference subgradient (sample some coordinates for speed)
        n_sample = min(n_points, 15)
        idxs = rng.choice(n_points, n_sample, replace=False)
        sg = np.zeros(n_points)
        for idx in idxs:
            j = (idx + 1) % n_points
            h_p = h.copy()
            h_m = h.copy()
            h_p[idx] += eps
            h_p[j] -= eps
            h_m[idx] -= eps
            h_m[j] += eps
            if h_p[idx] > 1 or h_p[j] < 0 or h_m[idx] < 0 or h_m[j] > 1:
                continue
            sg[idx] += (compute_c5(h_p, n_points) - compute_c5(h_m, n_points)) / (2 * eps)
        
        sg[np.arange(n_points)[np.isin(np.arange(n_points), idxs)==False]] = sg[idxs].mean()
        sg -= np.mean(sg)  # preserve sum
        
        h_new = h - step_size * sg
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = repair_h(h_new, target_sum)
        new_c5 = compute_c5(h_new, n_points)
        
        if new_c5 < current_c5:
            h = h_new
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
            step_size = min(step_size * 1.05, 0.01)
        else:
            step_size *= 0.7
            # Occasionally do a random kick
            if rng.random() < 0.1:
                i, j = rng.randint(0, n_points, 2)
                if i != j:
                    d = rng.uniform(-0.01, 0.01)
                    h[i] += d
                    h[j] -= d
                    h = repair_h(h, target_sum)
                    current_c5 = compute_c5(h, n_points)
        
        if step_size < 1e-8:
            break
    
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 10
    
    n_points = 70
    target_sum = n_points / 2.0
    
    h0 = repair_h(np.array(initial_h_values, dtype=np.float64), target_sum)
    c5_init = compute_c5(h0, n_points)
    print(f"Parent C5: {c5_init:.8f}")
    
    best_h = h0.copy()
    best_c5 = c5_init
    
    # Phase 1: SA (60% of budget)
    t1 = (deadline - start_time) * 0.55
    print(f"Phase 1 (SA): {t1:.0f}s")
    h1, c1 = sa_optimize(h0.copy(), n_points, seed, t1)
    if c1 < best_c5:
        best_c5 = c1
        best_h = h1
    print(f"  After SA: {best_c5:.8f}")
    
    # Phase 2: Local refinement (20%)
    t2 = (deadline - time.time()) * 0.35
    print(f"Phase 2 (Local): {t2:.0f}s")
    h2, c2 = local_refine(best_h.copy(), n_points, seed + 1, t2)
    if c2 < best_c5:
        best_c5 = c2
        best_h = h2
    print(f"  After local: {best_c5:.8f}")
    
    # Phase 3: Subgradient refinement (15%)
    t3 = (deadline - time.time()) * 0.5
    print(f"Phase 3 (Subgrad): {t3:.0f}s")
    h3, c3 = subgradient_refine(best_h.copy(), n_points, seed + 2, t3)
    if c3 < best_c5:
        best_c5 = c3
        best_h = h3
    print(f"  After subgrad: {best_c5:.8f}")
    
    # Phase 4: Final local polish
    if time.time() < deadline - 10:
        t4 = deadline - time.time() - 5
        h4, c4 = local_refine(best_h.copy(), n_points, seed + 3, t4)
        if c4 < best_c5:
            best_c5 = c4
            best_h = h4
        print(f"  Final polish: {best_c5:.8f}")
    
    # Ensure feasibility
    best_h = repair_h(best_h, target_sum)
    final_c5 = compute_c5(best_h, n_points)
    elapsed = time.time() - start_time
    print(f"DONE in {elapsed:.1f}s: C5 = {final_c5:.8f}, n = {n_points}")
    
    return (best_h, final_c5, n_points)
