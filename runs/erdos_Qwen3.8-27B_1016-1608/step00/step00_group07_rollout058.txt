    t = n // 2
    idx = np.argsort(h)[::-1]

    hb[idx[:t]] = 1.0
    return hb

def _proj_hypersimplex(h, target):
    """Project h onto {x: 0<=x_i<=1, sum x_i = target}"""
    n = len(h)
    h = h.copy()
    # Bisection on t: find t such that sum(clip(h-t,0,1)) = target
    lo, hi = h.min() - 1, h.max() + 1
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2
