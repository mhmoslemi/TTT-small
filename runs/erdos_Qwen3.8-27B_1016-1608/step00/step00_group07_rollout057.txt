import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_box_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(h + mid, 0.0, 1.0)) > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def grad_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag < n:
            g[lag:] += 1.0 - h[:n - lag]
        if lag < n:
            g[:n - lag] -= h[lag:]
    else:
        lp = -lag
        if lp < n:
            g[:n - lp] += 1.0 - h[lp:]
            g[lp:] -= h[:n - lp]
    return g

def grad_smooth(h, T):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cm = c.max()
    w = np.exp(T * (c - cm))
    ws = w.sum()
    if ws < 1e-30:
        j = int(np.argmax(c))
        return grad_lag(h, j - (n - 1))
    w /= ws
    g = np.zeros(n)
    for j in range(2 * n - 1):
        if w[j] > 1e-10:
            g += w[j] * grad_lag(h, j - (n - 1))
    return g

def pgd(h, iters, T, lr, t_limit):
    n = len(h)
    tgt = n / 2.0
    bh = h.copy()
    bf = compute_c5(h)
    f_cur = bf
    for i in range(iters):
        if time.time() > t_limit:
            break
        g = grad_smooth(h, T)
        gn = np.linalg.norm(g)
        if gn < 1e-10:
            break
        for trial in range(8):
            h_new = h - lr * g
            h_new = project_box_sum(h_new, tgt)
            f_new = compute_c5(h_new)
            if f_new < f_cur:
                h = h_new
                f_cur = f_new
                if f_new < bf:
                    bf = f_new
                    bh = h.copy()
                break
            lr = lr * 0.5
        else:
            lr = max(lr, 1e-6)
            break
        lr = min(lr * 1.03, 2.0)
    return bh, bf

def bin_search(hb, iters, rng, t_limit):
    n = len(hb)
    best = hb.astype(float).copy()

    for it in range(iters):
        if time.time() > t_limit:
            break
        imp = False
        ones = np.where(best > 0.5)[0]
        zeros = np.where(best < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        npoints = min(len(ones), len(zeros))
        for _ in range(40):
            i = int(rng.choice(ones))
            j = int(rng.choice(zeros))
            t = best.copy()
            t[i] = 0.0
            t[j] = 1.0
            f = compute_c5(t)
            if f < bf - 1e-14:
                bf = f
                best = t
                imp = True
                break
        if not imp:
            break
    return best, bf

def restart_pgd(h, rng, noise_scale, iters, T, lr, t_limit):

    h_noisy = project_box_sum(h_noisy, len(h) / 2.0)
    return pgd(h_noisy, iters, T, lr, t_limit)
