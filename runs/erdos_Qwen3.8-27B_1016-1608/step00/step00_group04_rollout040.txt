    """Compute the Erdos overlap score."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(sorted(corr * dx)[-1])

def compute_score_fast(h):
    """Faster score computation using manual loop for small n."""
    n = len(h)
    dx = 2.0 / n
    best = 0.0
    # For shift r = 0 to n-1, compute sum of h[m+r]*(1-h[m]) for valid m
    # This covers the "positive lag" part; the negative lag is symmetric
    for r in range(1, n):

        for m in range(n - r):
            val = h[m + r] * (1.0 - h[m])
            if val > 0:
                s += val
        if s * dx > best:
            best = s * dx
    return best
