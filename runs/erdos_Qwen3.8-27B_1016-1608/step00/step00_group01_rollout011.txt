    alpha = 0.99992
    base_noise = 0.04
    last_improve = time.time()
    stuck_epochs = 0

    while time.time() - t_start < time_limit:
        h_new = h.copy()
        r = rng.random()

        if r < 0.30:
            i, j = rng.choice(n, 2, replace=False)
            delta = rng.uniform(-1, 1) * base_noise
            h_new[i] -= delta
            h_new[j] += delta
        elif r < 0.50:
            i = rng.randint(n - 1)
            delta = rng.uniform(-1, 1) * base_noise
            h_new[i] -= delta
            h_new[i + 1] += delta
        elif r < 0.70:
            k = max(2, n // 10)
            idx = rng.choice(n, k, replace=False)
            noise = rng.normal(0, base_noise, k)
            noise -= noise.mean()
            h_new[idx] += noise
        elif r < 0.88:
            sz = rng.randint(2, min(7, n - 1))
            i = rng.randint(max(1, n - sz))
            pert = rng.normal(0, base_noise * 1.4, sz)
            pert -= pert.mean()
            h_new[i:i+sz] += pert
        else:
            i, j = rng.choice(n, 2, replace=False)
            delta = rng.uniform(-0.25, 0.25)
            h_new[i] -= delta
            h_new[j] += delta

        h_new = repair_h(h_new, n)
        new_c5 = eval_c5(h_new)

        if new_c5 <= best_c5 - 1e-13:
            h = h_new
            current_c5 = new_c5
            best_c5 = new_c5
            best_h = h_new.copy()
            last_improve = time.time()
            stuck_epochs = 0
            T = max(T, 0.004)
        elif new_c5 <= current_c5 + 1e-13:
            h = h_new
            current_c5 = new_c5
        else:
            if T > T_min:
                gap = new_c5 - current_c5
                if gap / T < 50.0:
                    if rng.random() < np.exp(-gap / T):
                        h = h_new
                        current_c5 = new_c5

        T *= alpha

        # Reheat if stuck for a long time
        if time.time() - last_improve > 30.0:
            if stuck_epochs < 5:
                T = max(T, 0.008)
                stuck_epochs += 1
                # Larger perturbation
                i, j = rng.choice(n, 2, replace=False)
                h_new = h.copy()
                h_new[i] -= rng.uniform(0.05, 0.3)
                h_new[j] += rng.uniform(0.05, 0.3)
                h_new = repair_h(h_new, n)
                nc = eval_c5(h_new)
                if nc < current_c5 + 0.02:
                    h = h_new
                    current_c5 = nc
            last_improve = time.time()

    return best_h, best_c5

def polish(h, n, rng, t_start, time_limit):
    h = repair_h(h.copy(), n)
    best_c5 = eval_c5(h)
    best_h = h.copy()
    step = 0.01

    for _ in range(200):
        if time.time() - t_start > time_limit:
            break
        improved = False
        for i in range(n):
            for delta in [step, -step, step*0.5, -step*0.5]:
                h_try = h.copy()
                h_try[i] += delta
                if h_try[i] < 0 or h_try[i] > 1:
                    continue
                h_try = repair_h(h_try, n)
                c = eval_c5(h_try)
                if c < best_c5 - 1e-14:
                    best_c5 = c
                    best_h = h_try.copy()
                    h = h_try
                    improved = True
        if not improved:
            step *= 0.7
            if step < 0.001:
                break

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')

    # Get parent
    init = None
    try:
        if 'initial_h_values' in dir() and initial_h_values is not None:
            init = np.asarray(initial_h_values, dtype=np.float64)
    except Exception:
        init = None

    # Build list of (start_h, n) pairs
    starters = []
    if init is not None and len(init) > 0:
        h0 = repair_h(init.copy(), len(init))
        starters.append(('parent', h0, len(h0)))

    for n in [33, 37, 41, 43, 45, 47, 49, 51, 53, 57]:
        starters.append(('bin'+str(n), make_binary(n, rng), n))
        if n % 2 == 1:
            starters.append(('alt'+str(n), make_alternating(n, rng), n))

    # Add Thue-Morse
    for n in [33, 47, 53]:
        starters.append(('tm'+str(n), make_thue_morse(n), n))

    # Random starts
    for n in [37, 47, 53]:
        h_r = repair_h(rng.rand(n), n)
        starters.append(('rnd'+str(n), h_r, n))

    # Evaluate all starters
    for name, h_s, n_s in starters:
        c = eval_c5(h_s)
        if c < best_c5:
            best_c5 = c
            best_h = h_s.copy()

    # Allocate time for SA runs
    time_per_sa = min(8.0, max(2.0, (budget_s * 0.75) / len(starters)))
    sa_budget_used = 0.0

    for name, h_s, n_s in starters:
        if time.time() - t0 > budget_s - 30.0:
            break
        limit = min(time_per_sa, budget_s - 35.0 - (time.time() - t0))
        if limit < 1.0:
            break
        t_s = time.time()
        h_res, c_res = sa_optimize(h_s, n_s, rng, t_s, limit, seed_offset=hash(name) % 10000)
        if c_res < best_c5:
            best_c5 = c_res
            best_h = h_res.copy()
        sa_budget_used += time.time() - t_s

    # Polish the best solution
    if best_h is not None and time.time() - t0 < budget_s - 10.0:
        n_best = len(best_h)
        t_p = time.time()
        polish_limit = min(60.0, budget_s - 12.0 - (time.time() - t0))
        best_h, best_c5 = polish(best_h, n_best, rng, t_p, max(1.0, polish_limit))

    # Final binary polish: try rounding near-0/1 values
    if best_h is not None and time.time() - t0 < budget_s - 5.0:
        n_b = len(best_h)
        h_bin = np.where(best_h > 0.5, 1.0, 0.0)
        if n_b % 2 == 1:
            h_bin = np.where(best_h > 0.5, 1.0, np.where(best_h < 0.5, 0.0, 0.5))
        h_bin = repair_h(h_bin, n_b)
        c_bin = eval_c5(h_bin)
        if c_bin < best_c5:
            best_c5 = c_bin
            best_h = h_bin

    assert best_h is not None, "No solution found"
    return best_h, best_c5, len(best_h)
