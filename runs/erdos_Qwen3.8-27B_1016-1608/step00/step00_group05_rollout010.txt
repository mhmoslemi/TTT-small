```python
def project_onto_constraint(g, target_sum):
    n = len(g)
    lo, hi = -2.0, 2.0
    for _ in range(200):
        mid = (lo + hi) / 2
        h = np.clip(g - mid, 0.0, 1.0)
        s = np.sum(h)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2, 0.0, 1.0)

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def compute_grad_smooth(h, dx, T):
    n = len(h)
    corrs = np.correlate(h, 1.0 - h, mode="full") * dx
    max_c = np.max(corrs)
    exp_vals = np.exp(T * (corrs - max_c))
    p = exp_vals / np.sum(exp_vals)
    grad = np.zeros(n)
```

```python
            if 0 <= i - k <= n - 1:
                term_i += (1 - h[i - k])  # was (1-h[i-k])*h[i]=...(1-h[i-k])*1, now *0
            # Actually: Δ from h[i] going to 0:
            # c_k = sum_l h[l+k]*(1-h[l])*dx
            # When h[i] changes by Δh_i:
            # - In (1-h[l]) when l=i: Δ = h[i+k]*(-Δh_i) (if i+k valid)
            # - In h[l+k] when l=i-k: Δ = (1-h[i-k])*Δh_i (if i-k valid)
            
            # Δh_i = -1 (going from 1 to 0)
            # Δh_j = +1 (going from 0 to 1)
            
            di = -1.0  # change in h[i]
            dj = +1.0  # change in h[j]
            
            dck = 0.0
            # h[i] in (1-h[l]) at l=i: change = h[i+k] * (-di) = ... 
            # (1-h[i]) changes from (1-1)=0 to (1-0)=1, change = +1 = -di
            # So c_k gets additional h[i+k] * (1 - 0) - h[i+k] * (1-1) = h[i+k] - 0
            # Wait I'm overcomplicating this.
            
            # Let's just compute: new_c_k = old_c_k + delta_k
            # delta_k from h[i]: 1→0
            #   - h[i] appears in (1-h[l]) when l=i: (1-h[i]) goes from 0 to 1, change +1
            #     contribution to c_k: h[i+k] * (+1) (if i+k valid)
            #   - h[i] appears in h[l+k] when l=i-k: h[i] goes from 1 to 0, change -1
            #     contribution to c_k: (1-h[i-k]) * (-1) (if i-k valid)
            
            if 0 <= i + k <= n - 1:
                dck += h[i + k]  # (1-h[i]) increases by 1
            if 0 <= i - k <= n - 1:
                dck -= (1 - h[i - k])  # h[i] decreases by 1
            
            # delta_k from h[j]: 0→1
            if 0 <= j + k <= n - 1:
                dck -= h[j + k]  # (1-h[j]) decreases by 1
            if 0 <= j - k <= n - 1:
                dck += (1 - h[j - k])  # h[j] increases by 1
            
            delta[n - 1 + k] = dck * dx
        
        new_corrs = corrs + delta
        new_max = np.max(new_corrs)
        
        if new_max < cur_max:
            h[i] = 0.0
            h[j] = 1.0
            corrs = new_corrs
            cur_max = new_max
            if cur_max < best_c5:
                best_c5 = cur_max
        # else: reject, keep h unchanged
    
    return h, best_c5
```

This is O(n) per iteration (computing delta is O(n) since we loop over 2n-1 lags). For n=70, that's 139 operations per swap. 200k iterations ≈ 28M operations. Should take a few seconds.
