from scipy.optimize import differential_evolution, minimize

_current_n = 45

def repair(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = np.maximum(1.0 - h, 0.0)
        tot = np.sum(room)
        if tot > 1e-14:
            h += diff * room / tot
        else:
            h += diff / n
    else:
        av = np.maximum(h, 0.0)
        tot = np.sum(av)
        if tot > 1e-14:
            h += diff * av / tot
        else:
            h += diff / n
    return np.clip(h, 0.0, 1.0)
