```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr), corr

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def smooth_max(corr, beta):
    """Log-sum-exp approximation of max"""
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta

def obj_and_grad(h, n, penalty, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    smax = smooth_max(corr, beta)
    
    # Gradient of smooth max
    weights = np.exp(beta * (corr - smax))  # softmax
    weights = weights / np.sum(weights)
    
    # Gradient of each corr[k] wrt h
    # corr[k] corresponds to lag = k - (n-1)
    # grad_k[m] = dx * [(1-h[m+lag]) if valid - h[m-lag] if valid]
    
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-10:
            continue
        lag = k - (n - 1)
        for m in range(n):
            g = 0.0
            if 0 <= m + lag < n:
                g += (1 - h[m + lag])
            if 0 <= m - lag < n:
                g -= h[m - lag]
            grad[m] += w * g * dx
    
    # Penalty gradient
    s = np.sum(h)
    penalty_val = penalty * (s - n/2.0)**2
    penalty_grad = 2 * penalty * (s - n/2.0) * np.ones(n)
    
    return smax + penalty_val, grad + penalty_grad
```

```python
def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n

def _project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def _grad_for_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag < n:
            g[:n-lag] += (1.0 - h[lag:])
            g[lag:] -= h[:n-lag]
    else:
        p = -lag
        if p < n:
            g[p:] += (1.0 - h[:n-p])
            g[:n-p] -= h[p:]
    return g

def _obj_grad(h, n, pen, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # Smooth max
    cmax = np.max(corr)
    if beta < 1e6:
        smax = cmax + np.log(np.sum(np.exp(beta * (corr - cmax)))) / beta
        w = np.exp(beta * (corr - smax))
        w /= np.sum(w)
    else:
        smax = cmax
        w = np.zeros(len(corr))
        w[np.argmax(corr)] = 1.0

        if w[k] < 1e-8:
            continue
        lag = k - (n - 1)
        if abs(lag) >= n:
            continue

    pen_val = pen * (s - n * 0.5) ** 2
    pen_grad = 2.0 * pen * (s - n * 0.5)
```
