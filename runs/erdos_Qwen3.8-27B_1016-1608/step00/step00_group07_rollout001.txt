    n_pat = min(6, max(2, int(remaining / 40)))
    for p in range(n_pat):
        if time.time() - t_start > budget_s - 200:
            break
        h_p = gen_pattern(n0, p, rng)
        t_lim = min(remaining * 0.035, 35.0)
        h_p, c_p = local_search(h_p, 3000000, rng, time_limit=t_lim)
        if c_p < best_c:
            best_c, best_h = c_p, h_p

    # Phase 3: SA from best (30%)
    elapsed = time.time() - t_start
    remaining = budget_s - elapsed
    if remaining > 120:
        h_sa, c_sa = sa_search(best_h, remaining * 0.30, rng)
        if c_sa < best_c:
            best_c, best_h = c_sa, h_sa

    # Phase 4: Refinement local search (15%)
    elapsed = time.time() - t_start
    remaining = budget_s - elapsed
    if remaining > 60:
        h_f, c_f = local_search(best_h, 8000000, rng, time_limit=remaining * 0.4)
        if c_f < best_c:
            best_c, best_h = c_f, h_f

    # Phase 5: Second SA round if time (5%)
    elapsed = time.time() - t_start
    remaining = budget_s - elapsed
    if remaining > 60:
        h_sa2, c_sa2 = sa_search(best_h, remaining * 0.5, rng)
        if c_sa2 < best_c:
            best_c, best_h = c_sa2, h_sa2

    # Phase 6: Try higher resolution (10%)
    elapsed = time.time() - t_start
    remaining = budget_s - elapsed
    if remaining > 80:
        for n_hi in [96, 120]:
            if time.time() - t_start > budget_s - 30:
                break
            x_lo = np.linspace(0, 2, n0, endpoint=False)
            x_hi = np.linspace(0, 2, n_hi, endpoint=False)
            h_hi = np.interp(x_hi, x_lo, np.concatenate([best_h, best_h[:1]]))
            h_hi = fix_sum(h_hi)
            c_hi = c5_value(h_hi)
            if c_hi < best_c + 0.02:
                t_lim = min(remaining * 0.3, 60.0)
                h_hi, c_hi2 = local_search(h_hi, 4000000, rng, time_limit=t_lim)
                if c_hi2 < best_c:
                    best_c, best_h = c_hi2, h_hi

    # Final polish
    elapsed = time.time() - t_start
    if budget_s - elapsed > 15:
        h_fin, c_fin = local_search(best_h, 2000000, rng, time_limit=(budget_s - elapsed) * 0.8)
        if c_fin < best_c:
            best_c, best_h = c_fin, h_fin

    best_h = np.clip(best_h, 0.0, 1.0)
    best_c = c5_value(best_h)
    return best_h, best_c, len(best_h)
