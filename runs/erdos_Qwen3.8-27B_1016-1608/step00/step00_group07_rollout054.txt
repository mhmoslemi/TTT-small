def compute_c5(h, dx):
    n = len(h)
    cor = np.correlate(h, 1-h, mode="full")
    return float(np.max(cor) * dx)

def c5_gradient_approx(h, dx):
    """Approximate gradient of C5 w.r.t. h using the argmax shift."""
    n = len(h)
    cor = np.correlate(h, 1-h, mode="full")
    k_max = np.argmax(cor)
    m = k_max - (n - 1)  # actual shift
    
    # Gradient of cor[k_max] w.r.t. h
    grad = np.zeros(n)
    if m >= 0:
        for i in range(n - m):
            # cor contribution: h[i+m] * (1-h[i])
            # d/dh[i] = -h[i+m]
            # d/dh[i+m] = (1-h[i])
            grad[i] += -h[i+m]
            grad[i+m] += (1-h[i])
    else:
        j = -m
        for i in range(j, n):
            grad[i] += (1-h[i-j])
            grad[i-j] += -h[i]
    
    # Note: cor also depends on 1-h, so there's a negative contribution
    # Actually, cor = sum h[i+m]*(1-h[i]) or similar
    # d/dh[i] of h[i+m]*(1-h[i]) = -h[i+m] (if i is the "1-h" position)
    # d/dh[i+m] of h[i+m]*(1-h[i]) = (1-h[i]) (if i+m is the "h" position)
    
    return grad * dx
