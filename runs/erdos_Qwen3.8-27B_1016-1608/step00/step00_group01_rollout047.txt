import numpy as np
from scipy.optimize import minimize
import time

_n_g = 47
_target_g = 23.5
_dx_g = 2.0 / 47.0

def _eq_constraint(h):
    return np.sum(h) - _target_g

def _smooth_obj(h, T):
    corr = np.correlate(h, 1.0 - h, mode='full') * _dx_g
    m = np.max(corr)
    if m > 1e-6:
        scaled = np.clip(T * (corr - m), -60.0, 0.0)
        return m + np.log(np.sum(np.exp(scaled))) / T
    return m

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _project_h(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s < 1e-12:
        h[:] = target / len(h)
        return h
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    # iterative correction
    for _ in range(20):
        d = target - np.sum(h)
        if abs(d) < 1e-12:
            break
        mask_up = h < 1.0 - 1e-14
        mask_dn = h > 1e-14
        if d > 0 and np.any(mask_up):
            add = d / np.sum(mask_up)
            h[mask_up] = np.minimum(h[mask_up] + add, 1.0)
        elif d < 0 and np.any(mask_dn):
            sub = (-d) / np.sum(mask_dn)
            h[mask_dn] = np.maximum(h[mask_dn] - sub, 0.0)
        else:
            break
    return h

def _try_slsqp(h_start, n, T_list, budget_s, t0, frac):
    global _n_g, _target_g, _dx_g
    _n_g = n
    _target_g = n / 2.0
    _dx_g = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_constraint}]
    best = h_start.copy()
    best_c5 = compute_c5(best)
    for T in T_list:
        if time.time() - t0 > budget_s * frac:
            break
        try:
            res = minimize(_smooth_obj, best, args=(T,), method='SLSQP',
                           bounds=bounds, constraints=cons,

        except Exception:
            continue
        if hasattr(res, 'x') and res.x is not None:
            h_t = _project_h(res.x, _target_g)
            c5_t = compute_c5(h_t)
            if c5_t < best_c5:
                best_c5 = c5_t
                best = h_t
    return best, best_c5

def _coord_descent(h, n, target, t0, budget_s, frac=0.95):
    best = h.copy()
    best_c5 = compute_c5(best)
    for outer in range(500):
        if time.time() - t0 > budget_s * frac:
            break
        improved = False
        for i in range(n):
            # Compute a_tau, b_tau for each lag
            a_arr = np.zeros(2 * n - 1)
            b_arr = np.zeros(2 * n - 1)
            for tau in range(-(n - 1), n):
                idx = tau + n - 1
                ip = i + tau
            im = i - tau
                hp = best[ip] if 0 <= ip < n else 0.0
                hm = best[im] if 0 <= im < n else 0.0
                b_arr[idx] = 1.0 - hp - hm
                a_arr[idx] = hm
            # f(t) = max_tau(a_arr[tau] + t * b_arr[tau])
            # Find t in [0,1] minimizing this (convex piecewise linear)
            f0 = np.max(a_arr)
            f1 = np.max(a_arr + b_arr)
            lo, hi = 0.0, 1.0
            for _ in range(40):
                mid = (lo + hi) / 2.0
                fm = np.max(a_arr + mid * b_arr)
                # Compare left and right slopes
                fl = np.max(a_arr + (lo + mid) / 2.0 * b_arr)
                fr = np.max(a_arr + (mid + hi) / 2.0 * b_arr)
                if fl < fr:
                    hi = mid
                else:
                    lo = mid
            t_opt = (lo + hi) / 2.0
            old_v = best[i]
            if abs(t_opt - old_v) > 1e-8:
                best[i] = t_opt
                best = _project_h(best, target)
                c5_now = compute_c5(best)

                    best_c5 = c5_now
                    improved = True
                else:
                    best[i] = old_v
                    best = _project_h(best, target)
        if not improved:
            break
    return best, best_c5

    idx = rng.choice(n, size=n_pert, replace=False)
