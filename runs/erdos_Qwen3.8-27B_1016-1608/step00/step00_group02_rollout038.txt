    if budget_s - elapsed() > 60:
        h_cd, c5_cd = coord_descent(best_h, target_sum, n, max_rounds=80, time_cap=80)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 3: Nelder-Meed refinement
    def objective(x):
        hp = np.clip(x, 0.0, 1.0)
        pen = 500.0 * (hp.sum() - target_sum) ** 2
        return eval_c5(hp) + pen

    if budget_s - elapsed() > 180:
        res = minimize(objective, best_h, method='Nelder-Meed',
                       options={'maxiter': 10000, 'xatol': 1e-10, 'fatol': 1e-12})
        h_nm = project_feasible(res.x, target_sum)
        c5_nm = eval_c5(h_nm)
        if c5_nm < best_c5:
            best_c5 = c5_nm
            best_h = h_nm.copy()

    # Phase 4: Another round of swap search from improved point
    if budget_s - elapsed() > 100:
        h_s2, c5_s2 = swap_local_search(best_h, target_sum, n, rng, max_iters=5000, time_cap=120)
        if c5_s2 < best_c5:
            best_c5 = c5_s2
            best_h = h_s2.copy()

    # Phase 5: Random restarts
    for restart in range(30):
        if budget_s - elapsed() < 80:
            break
        # Generate diverse initial points
        mode = restart % 4
        if mode == 0:
            h_r = rng.random(n)
        elif mode == 1:
            h_r = (rng.random(n) > 0.5).astype(float)
        elif mode == 2:
            # Modified block pattern
            h_r = np.zeros(n)
            block = int(rng.randint(2, 8))
            for i in range(n):
                h_r[i] = 1.0 if (i % (2 * block)) < block else 0.0
        else:
            # Perturb best
            h_r = best_h + 0.3 * rng.randn(n)

        h_r = project_feasible(h_r, target_sum)
        # Quick local search
        h_ls, c5_ls = swap_local_search(h_r, target_sum, n, rng, max_iters=1500, time_cap=40)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()
            # Refine the new best
            if budget_s - elapsed() > 60:
                h_cd2, c5_cd2 = coord_descent(best_h, target_sum, n, max_rounds=30, time_cap=30)
                if c5_cd2 < best_c5:
                    best_c5 = c5_cd2
                    best_h = h_cd2.copy()

    # Final refinement
    if budget_s - elapsed() > 120:
        res = minimize(objective, best_h, method='Nelder-Meed',
                       options={'maxiter': 10000, 'xatol': 1e-10, 'fatol': 1e-12})
        h_final = project_feasible(res.x, target_sum)
        c5_final = eval_c5(h_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Final swap polish
    if budget_s - elapsed() > 30:
        h_pol, c5_pol = swap_local_search(best_h, target_sum, n, rng, max_iters=3000, time_cap=25)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Ensure final solution is feasible
    best_h = project_feasible(best_h, target_sum)
    best_c5 = eval_c5(best_h)

    return best_h, best_c5, n
