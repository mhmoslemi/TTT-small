def project_feasible(h, target_sum):
    """Project h onto {x: 0<=x<=1, sum(x)=target_sum}"""
    n = len(h)
    # Bisection on t
    lo, hi = -1.0, 1.0  # t range
    for _ in range(100):
        mid = (lo + hi) / 2
        x = np.clip(h - mid, 0, 1)
        s = np.sum(x)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - (lo + hi) / 2, 0, 1)
    # Fix tiny numerical errors
    diff = target_sum - np.sum(x)
    if abs(diff) > 1e-12:
        # Distribute correction
        mask = (x > 1e-10) & (x < 1 - 1e-10)
        if np.sum(mask) > 0:
            x[mask] += diff / np.sum(mask)
            x = np.clip(x, 0, 1)
    return x

    m = np.max(corr)
    if m < -1000:
        return 0.0
    return m + np.log(np.sum(np.exp(corr - m))) / alpha

    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]

    alphas = [10, 30, 80, 200, 500]
    for alpha in alphas:
        if time.time() - t0 > budget_s - 50:
            break
        _G['alpha'] = alpha
        try:
            res = minimize(_obj_func, best_h, method='SLSQP', jac=_obj_grad_func,
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 600, 'ftol': 1e-13})
            if res.x is not None:
                new_h = _project_feasible(res.x, target_sum)
                new_c5 = _compute_c5(new_h, dx)
                if new_c5 < best_c5:
                    best_h = new_h.copy()
                    best_c5 = new_c5
        except Exception:
            pass

    # Perturbation restarts with high alpha
    _G['alpha'] = 500
    remaining = budget_s - (time.time() - t0)
    n_trials = max(5, int(min(40, remaining / 8)))
    for _ in range(n_trials):
        if time.time() - t0 > budget_s - 15:
            break
        for scale in [0.03, 0.015]:
            if time.time() - t0 > budget_s - 15:
                break
            pert = best_h + rng.randn(n) * scale
            pert = _project_feasible(pert, target_sum)
            try:
                res = minimize(_obj_func, pert, method='SLSQP', jac=_obj_grad_func,
                             bounds=bounds, constraints=cons,
                             options={'maxiter': 300, 'ftol': 1e-13})
                if res.x is not None:
                    new_h = _project_feasible(res.x, target_sum)
                    new_c5 = _compute_c5(new_h, dx)
                    if new_c5 < best_c5 - 1e-12:
                        best_h = new_h.copy()
                        best_c5 = new_c5
            except Exception:
                pass

    # Final coordinate-descent polish on hard max
    if time.time() - t0 < budget_s - 10:
        for iteration in range(200):
            if time.time() - t0 > budget_s - 10:
                break
            improved = False
            for i in range(n):
                if time.time() - t0 > budget_s - 10:
                    break
                old_val = best_h[i]
                for trial in [old_val - 0.005, old_val + 0.005, 0.0, 1.0]:
                    c = np.clip(trial, 0.0, 1.0)
                    if c == old_val:
                        continue
                    candidate = best_h.copy()
                    candidate[i] = c
                    candidate = _project_feasible(candidate, target_sum)
                    c5_val = _compute_c5(candidate, dx)
                    if c5_val < best_c5 - 1e-14:
                        best_h = candidate
                        best_c5 = c5_val
                        improved = True
                        break
            if not improved:
                break

    best_h = _project_feasible(best_h, target_sum)
    best_c5 = _compute_c5(best_h, dx)
    return (best_h, best_c5, n)
