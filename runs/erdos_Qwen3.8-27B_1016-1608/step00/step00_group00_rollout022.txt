def optimize_coordinate(h, i, n):
    """Optimize h[i] to minimize max correlate. Returns optimal h[i]."""
    # Compute slopes and intercepts for each lag
    # s_b = (1-h[(i+b) % n]) - h[(i-b) % n]  -- NO, it's not modulo
    # h[j] = 0 for j < 0 or j >= n
    
    h_i = h[i]
    a = np.zeros(n + n - 1)  # intercepts (value when h[i]=0)
    s = np.zeros(n + n - 1)  # slopes
    
    # Current correlate values
    c = np.correlate(h, 1.0 - h, mode='full')
    
    for lag_idx in range(len(c)):
        a = lag_idx - (n - 1)  # actual lag
        # slope contribution: (1 - h[i+a]) - h[i-a] (with 0 for out of range)
        ip = i + a
        im = i - a
        h_ip = h[ip] if 0 <= ip < n else 0.0
        h_im = h[im] if 0 <= im < n else 0.0
        s[lag_idx] = (1.0 - h_ip) - h_im
        a[lag_idx] = c[lag_idx] - h_i * s[lag_idx]
    
    # Now minimize max_b (a[b] + x * s[b]) for x in [0, 1]
    # This is a 1D convex optimization
    # Use the fact that the minimum is where the "envelope" minimum is
    # Simple approach: ternary search with ~60 iterations
    lo, hi = 0.0, 1.0
    for _ in range(60):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a + m1 * s)
        f2 = np.max(a + m2 * s)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_sum(h, target_sum):
    h = h.copy()
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if mask.sum() == 0:
            break
        adj = diff / mask.sum()
        h[mask] += adj
        h = np.clip(h, 0.0, 1.0)
    return h

def optimize_one_coord(h, c, i):
    n = len(h)
    m = len(c)
    lags = np.arange(m) - (n - 1)
    h_i = h[i]
    ip = i + lags
    im = i - lags
    h_ip = np.zeros(m)
    h_im = np.zeros(m)
    vp = (ip >= 0) & (ip < n)
    vm = (im >= 0) & (im < n)
    if np.any(vp):
        h_ip[vp] = h[ip[vp]]
    if np.any(vm):
        h_im[vm] = h[im[vm]]
    slopes = (1.0 - h_ip) - h_im
    intercepts = c - h_i * slopes
    lo, hi = 0.0, 1.0
    for _ in range(55):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(intercepts + m1 * slopes) < np.max(intercepts + m2 * slopes):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

                ip = i + (np.arange(len(c)) - (n - 1))

def smooth_polish(h, target_sum, alpha_start=20, alpha_end=200, n_iter=80):
    n = len(h)
    dx = 2.0 / n
    h = project_sum(h, target_sum)
    best_c5 = evaluate_c5(h)
    for it in range(n_iter):
        alpha = alpha_start + (alpha_end - alpha_start) * it / max(1, n_iter - 1)
        def obj(hv):
            c = np.correlate(hv, 1.0 - hv, mode='full')
            shift = np.max(c)
            return float(dx * (shift + np.log(np.sum(np.exp(alpha * (c - shift)))) / alpha))
        def obj_grad(hv):
            c = np.correlate(hv, 1.0 - hv, mode='full')
            shift = np.max(c)
            w = np.exp(alpha * (c - shift))
            w /= w.sum()
            m = len(c)
            lags = np.arange(m) - (n - 1)
            g = np.zeros(n)
            for i in range(n):
                ip = i + lags
                im = i - lags
                h_ip = np.zeros(m)
                h_im = np.zeros(m)
                vp = (ip >= 0) & (ip < n)
                vm = (im >= 0) & (im < n)
                if np.any(vp):
                    h_ip[vp] = hv[ip[vp]]
                if np.any(vm):
                    h_im[vm] = hv[im[vm]]
                slopes = (1.0 - h_ip) - h_im
                g[i] = np.sum(w * slopes) * dx
            return g
        res = minimize(obj, h, jac=obj_grad, method='L-BFGS-B',
                       bounds=Bounds(0, 1), options={'maxiter': 30, 'ftol': 1e-12, 'gtol': 1e-8})
        h2 = project_sum(res.x, target_sum)
        c5_2 = evaluate_c5(h2)
        if c5_2 < best_c5:
            best_c5 = c5_2
            h = h2
        else:
            break
    return h, best_c5

def make_alternating(n, nblocks):
    h = np.zeros(n)
    bs = n / nblocks
    for b in range(nblocks):
        s = int(b * bs)
        e = int((b + 1) * bs)
        if b % 2 == 0:
            h[s:e] = 1.0
    return h

    for i in range(n):
        t = (i + 0.5) * freq / n
        h[i] = 1.0 if (t % 1.0) < 0.5 else 0.0
    return h
