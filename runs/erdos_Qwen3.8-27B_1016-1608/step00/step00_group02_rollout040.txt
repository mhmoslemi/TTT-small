import numpy as np
import time

def project_simplex(h, target_sum):
    n = len(h)
    for _ in range(2000):
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        fm = (h > 1e-14) & (h < 1.0 - 1e-14)
        nf = int(fm.sum())
        if nf == 0:
            if diff > 0:
                idx = int(np.argmax(1.0 - h))
                h[idx] = min(1.0, h[idx] + diff)
            else:
                idx = int(np.argmin(h))
                h[idx] = max(0.0, h[idx] + diff)
        else:
            h[fm] += diff / nf
    return np.clip(h, 0.0, 1.0)

def eval_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def to_binary(h, target):
    n = len(h)
    h = np.sort(h)[::-1]
    h_bin = np.zeros(n)
    ip = int(target)
    fr = target - ip
    if ip >= 1:
        h_bin[:ip] = 1.0
    if fr > 0.01:
        h_bin[ip] = fr
    return h_bin

def local_search(h, n, deadline, max_passes=200):
    target = n / 2.0
    h = np.round(h)
    if abs(h.sum() - target) > 0.01:
        h = to_binary(h, target)
    c5 = eval_c5(h)
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h < 0.5)[0].tolist()
    no_, nz_ = len(ones), len(zeros)
    if no_ == 0 or nz_ == 0:
        return h, c5
    for p in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        rnd = np.random.RandomState(12345 + p * 7)
        n_tries = min(8000, no_ * nz_)
        for _ in range(n_tries):
            if time.time() > deadline:
                break
            i = int(rnd.randint(no_))
            j = int(rnd.randint(nz_))
            oi, oj = ones[i], zeros[j]
            h[oi], h[oj] = h[oj], h[oi]
            nc = eval_c5(h)
            if nc < c5 - 1e-15:
                c5 = nc
                ones[i], zeros[j] = oj, oi
                improved = True
            else:
                h[oi], h[oj] = h[oj], h[oi]
        if not improved:
            break
    return h, c5

def greedy_swap(h, n, deadline, max_passes=30):
    target = n / 2.0
    h = np.round(h)
    if abs(h.sum() - target) > 0.01:
        h = to_binary(h, target)
    c5 = eval_c5(h)
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h < 0.5)[0].tolist()
    for p in range(max_passes):
        if time.time() > deadline:
            break
        best_imp = 0.0
        best_oi, best_oj = -1, -1
        no_, nz_ = len(ones), len(zeros)
        for i in range(no_):
            for j in range(nz_):
                oi, oj = ones[i], zeros[j]
                h[oi], h[oj] = h[oj], h[oi]
                nc = eval_c5(h)
                imp = c5 - nc
                if imp > best_imp:
                    best_imp = imp
                    best_oi, best_oj = oi, oj
                h[oi], h[oj] = h[oj], h[oi]
                if time.time() > deadline:
                    break
            if time.time() > deadline:
                break
        if best_oi >= 0:
            oi, oj = best_oi, best_oj
            h[oi], h[oj] = h[oj], h[oi]
            c5 -= best_imp
            ones[ones.index(oi)] = oj
            zeros[zeros.index(oj)] = oi
        else:
            break
    return h, c5

def smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = corr.max()
    e = np.exp(beta * (corr - mc))
    w = e / e.sum()
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-12:
            continue
        s = i - (n - 1)
        lo1, hi1 = max(0, s), min(n, n + s)
        lo2, hi2 = max(0, -s), min(n, n - s)
        if lo1 < hi1:
            grad[lo1:hi1] += wi * (1.0 - h[lo1 - s:hi1 - s])
        if lo2 < hi2:
            grad[lo2:hi2] -= wi * h[lo2 + s:hi2 + s]
    return dx * grad

def gradient_descent(h, n, deadline):
    target = n / 2.0
    c5 = eval_c5(h)
    beta = 10.0
    lr = 0.05
    for it in range(30000):
        if time.time() > deadline:
            break
        g = smooth_grad(h, beta)
        h_new = project_simplex(h - lr * g, target)
        nc = eval_c5(h_new)
        if nc < c5 - 1e-15:
            h = h_new
            c5 = nc
            lr = min(lr * 1.03, 0.5)
        else:
            lr *= 0.7
            if lr < 1e-10:
                break
    return h, c5
