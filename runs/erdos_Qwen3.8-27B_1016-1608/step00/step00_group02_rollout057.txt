    m = np.max(corr)
    shifted = M * (corr - m)
    return m + np.log(np.sum(np.exp(shifted))) / M

def sum_eq(x, target_sum):
    return float(np.sum(x) - target_sum)

def make_starts(n, target_sum, base_h, rng):
    starts = [base_h.copy()]
    # Spike pattern
    h_sp = np.zeros(n)
    h_sp[:n // 2] = 1.0
    h_sp[n // 2] = target_sum - (n // 2)
    starts.append(repair_h(h_sp, target_sum))
    # Linear ramp
    h_li = np.linspace(0.0, 1.0, n)
    h_li = h_li / h_li.sum() * target_sum
    starts.append(repair_h(h_li, target_sum))
    # Zigzag
    h_zz = np.zeros(n)
    h_zz[::2] = 2.0
    h_zz[1::2] = 0.0
    h_zz = h_zz / h_zz.sum() * target_sum
    starts.append(repair_h(h_zz, target_sum))
    # Random
    h_rd = rng.rand(n)
    starts.append(repair_h(h_rd, target_sum))
    return starts

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    n = len(initial_h_values)
    target_sum = n / 2.0
    h0 = repair_h(np.array(initial_h_values, dtype=float), target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq, 'args': (target_sum,)}]
    starts = make_starts(n, target_sum, h0, rng)

    # Phase 1: SLSQP with increasing smoothness
    M_configs = [(10, 5, 1500), (30, 4, 2000), (80, 4, 2500), (200, 3, 3000), (500, 2, 4000)]

    for M, n_starts, maxiter in M_configs:
        if time.time() - t0 > budget_s - 30:
            break
        n_to_try = min(n_starts, len(starts))
        for s in range(n_to_try):
            if time.time() - t0 > budget_s - 15:
                break
            if s == 0:
                x0 = best_h.copy()
            elif s == 1:
                x0 = np.clip(best_h + rng.randn(n) * 0.08, 0.0, 1.0)
                x0 = repair_h(x0, target_sum)
            elif s < n_to_try and s < len(starts):
                x0 = starts[s].copy()
            else:
                x0 = np.clip(best_h + rng.randn(n) * 0.2, 0.0, 1.0)
                x0 = repair_h(x0, target_sum)

            res = minimize(obj_func, x0, args=(n, M), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-15})
            h_new = repair_h(res.x, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
        # Update starts for next M level
        if best_c5 < compute_c5(starts[0]):
            starts[0] = best_h.copy()

    # Phase 2: Coordinate descent with fine line search
    if time.time() - t0 < budget_s - 5:
        for sweep in range(80):
            if time.time() - t0 > budget_s - 5:
                break
            any_improved = False
            for i in range(n):
                best_trial_c5 = best_c5
                best_trial_h = best_h.copy()
                for trial in np.linspace(0.0, 1.0, 21):
                    h_try = best_h.copy()
                    h_try[i] = trial
                    diff = target_sum - np.sum(h_try)
                    if abs(diff) > 1e-13:
                        diff2 = target_sum - h_try[i] - np.sum(h_try[np.arange(n) != i])
                        h_try = repair_h(h_try, target_sum)
                        if abs(h_try[i] - trial) > 0.15:
                            continue
                    c5_try = compute_c5(h_try)
                    if c5_try < best_trial_c5 - 1e-13:
                        best_trial_c5 = c5_try
                        best_trial_h = h_try.copy()
                if best_trial_c5 < best_c5 - 1e-13:
                    best_c5 = best_trial_c5
                    best_h = best_trial_h.copy()
                    any_improved = True
            if not any_improved:
                break

    # Phase 3: Final SLSQP polish with very sharp max
    if time.time() - t0 < budget_s - 5:
        res = minimize(obj_func, best_h, args=(n, 1000.0), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 3000, 'ftol': 1e-16})
        h_new = repair_h(res.x, target_sum)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Final validation and repair
    best_h = repair_h(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
