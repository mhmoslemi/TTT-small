import numpy as np
from scipy.optimize import minimize
import time


def repair_h(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    s = h.sum()
    if s > 1e-15:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(1.0, h[idx] + d)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + d)
    return h


def c5_val(h, dx):
    return float(np.correlate(h, 1.0 - h, mode="full").max() * dx)


def nm_obj(x, n, dx):
    return c5_val(repair_h(x, n), dx)


def nm_run(h_start, n, dx, maxiter):
    res = minimize(nm_obj, h_start, args=(n, dx), method="Nelder-Mead",
                   options={"maxiter": maxiter, "xatol": 1e-9, "fatol": 1e-11, "adaptive": True})
    return repair_h(res.x, n)


def coord_descend(h, n, dx, iters):
    h = h.copy()
    f_cur = c5_val(h, dx)
    deltas = (-0.2, -0.1, -0.05, 0.05, 0.1, 0.2)
    for it in range(iters):
        imp = False
        for i in range(n):
            for dv in deltas:
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + dv, 0.0, 1.0)
                h_new = repair_h(h_new, n)
                f_new = c5_val(h_new, dx)
                if f_new < f_cur - 1e-12:
                    h, f_cur, imp = h_new, f_new, True
                    break
        if not imp:
            break
    return h, f_cur


def binary_sa(n, dx, n_steps, rng):
    h = np.zeros(n)
    h[::2] = 1.0
    cnt, tgt = int(h.sum()), n // 2
    if cnt > tgt:
        ones = np.where(h == 1)[0]
        for j in range(tgt, cnt):
            h[ones[j]] = 0.0
    elif cnt < tgt:
        zeros = np.where(h == 0)[0]
        for j in range(tgt - cnt):
            h[zeros[j]] = 1.0
    f_cur = c5_val(h, dx)
    best_h, best_f = h.copy(), f_cur
    for step in range(n_steps):
        T = 0.04 * (0.001 ** (step / max(1, n_steps)))
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i1 = ones[rng.randint(len(ones))]
        i0 = zeros[rng.randint(len(zeros))]
        h[i1], h[i0] = 0.0, 1.0
        f_new = c5_val(h, dx)
        if f_new < f_cur or rng.rand() < np.exp(-(f_new - f_cur) / max(T, 1e-12)):
            f_cur = f_new
            if f_cur < best_f:
                best_f, best_h = f_cur, h.copy()
        else:
            h[i1], h[i0] = 1.0, 0.0
    return best_h, best_f


def cont_sa(h_start, n, dx, n_steps, rng):
    h = h_start.copy()
    f_cur = c5_val(h, dx)
    best_h, best_f = h.copy(), f_cur
    for step in range(n_steps):
        T = 0.015 * (0.0003 ** (step / max(1, n_steps)))
        h_new = h + rng.randn(n) * 0.04
        h_new = repair_h(h_new, n)
        f_new = c5_val(h_new, dx)
        if f_new < f_cur or rng.rand() < np.exp(-(f_new - f_cur) / max(T, 1e-12)):
            h, f_cur = h_new, f_new
            if f_cur < best_f:
                best_f, best_h = f_cur, h.copy()
    return best_h, best_f

    h_t = nm_run(h0, n, dx, 20000)

        if i % 4 == 3 and time.time() < t3:
            h_t, f_t = coord_descend(best_h, n, dx, 40)
            if f_t < best_c5:
                best_h, best_c5 = h_t, f_t

    # Phase 4: Continuous SA
    t4 = t0 + budget_s * 0.88

    h_s, f_s = cont_sa(best_h, n, dx, n_steps, rng)
    if f_s < best_c5:
        best_c5, best_h = f_s, h_s

    # Phase 5: Final polish
    if time.time() < t0 + budget_s * 0.98:
        h_t = nm_run(best_h, n, dx, 15000)

        if f_t < best_c5:

        h_t, f_t = coord_descend(best_h, n, dx, 80)

            best_h, best_c5 = h_t, f_t

    best_h = repair_h(best_h, n)
    best_c5 = c5_val(best_h, dx)
    return best_h, best_c5, n
