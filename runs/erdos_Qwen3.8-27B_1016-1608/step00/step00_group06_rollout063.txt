import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n_points):
    target = 0.5 * n_points
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h

def smooth_obj(h, n_pts, beta):
    dx = 2.0 / n_pts
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    return m + np.log(np.sum(np.exp(beta * (c - m)))) / beta

def sum_con(h, n_pts):
    return h.sum() - 0.5 * n_pts

def local_search(h, n_pts, n_rounds=3):
    h = h.copy()
    for _ in range(n_rounds):
        improved = False
        idx = np.random.permutation(n_pts)
        for i in idx:
            for val in [0.0, 0.25, 0.5, 0.75, 1.0]:
                h_new = h.copy()
                h_new[i] = val
                h_new = project_feasible(h_new, n_pts)
                c_new = compute_c5(h_new, n_pts)
                c_old = compute_c5(h, n_pts)
                if c_new < c_old - 1e-12:
                    h = h_new
                    improved = True
                    break
        if not improved:
            break
    return h
