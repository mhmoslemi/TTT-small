                if 0 <= ht[i] and ht[j] <= 1:
                    s = compute_score(ht)
                    if s < best_score - 1e-10:
                        best_h = ht
                        best_score = s
                        improved = True
            if not improved:
                break
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_score = 10.0
    best_n = 0

    # Use parent if available
    h_parent = None
    if 'initial_h_values' in globals():
        h_parent = np.array(initial_h_values, dtype=float)
        h_parent = fix_sum(h_parent, len(h_parent))

    # n=42 from parent
    if h_parent is not None:
        n42 = len(h_parent)
        s0 = compute_score(h_parent)
        best_score = s0
        best_h = h_parent.copy()
        best_n = n42
        t_alloc_42 = min(400.0, budget_s * 0.4)
        h_r, s_r = swap_search(h_parent, t_alloc_42, time.time(), rng)
        if s_r < best_score:
            best_score = s_r
            best_h = h_r
            best_n = n42
        # More restarts on n=42
        for r in range(5):
            rem = budget_s - (time.time() - t0) - 200
            if rem < 10:
                break
            t_this = min(40.0, rem / (6 - r))
            h_try = random_binary(n42, seed + 7000 + r)
            h_r, s_r = swap_search(h_try, t_this, time.time(), rng)
            if s_r < best_score:
                best_score = s_r
                best_h = h_r
                best_n = n42

    # Try other n values
    n_list = [40, 44, 46, 48, 50, 52, 56, 60, 80]
    for n in n_list:
        rem = budget_s - (time.time() - t0) - 250
        if rem < 10:
            break
        t_this = min(30.0, rem / max(1, len(n_list)))
        for r in range(3):
            rem2 = budget_s - (time.time() - t0) - 200
            if rem2 < 10:
                break
            t_sub = min(15.0, rem2 / 4)
            h_try = random_binary(n, seed + n * 100 + r * 7)
            h_r, s_r = swap_search(h_try, t_sub, time.time(), rng)
            if s_r < best_score:
                best_score = s_r
                best_h = h_r
                best_n = n

    # Continuous refinement
    rem = budget_s - (time.time() - t0) - 20
    if rem > 10 and best_h is not None:
        t_ref = min(120.0, rem)
        h_ref, s_ref = continuous_refine(best_h, best_n, t_ref, time.time())
        if s_ref < best_score:
            best_score = s_ref
            best_h = h_ref

    # Final validation
    if best_h is not None:
        best_h = fix_sum(best_h, len(best_h))
        final_score = compute_score(best_h)
        return (best_h, final_score, len(best_h))

    h = random_binary(42, seed)
    return (h, compute_score(h), 42)
