from scipy.special import logsumexp

def _c5_from_h(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def _lse_objective(h, dx, tau):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return (1.0 / tau) * logsumexp(tau * corr)

def _lse_gradient(h, dx, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    # Softmax weights
    shifted = tau * corr - np.max(tau * corr)
    exp_vals = np.exp(shifted)
    w = exp_vals / exp_vals.sum()  # (2n-1,)

    # where m_k = k - (n-1)
    # dcorr_m/dh[j] = (1-h[j-m])*1[j in J_m] - h[j+m]*1[j+m in J_m]
    
    grad = np.zeros(n)

        wk = w[k]
        if wk < 1e-15:
            continue
        J_lo = max(0, m)

        if J_hi >= J_lo:

        j_hi = min(n - 1, J_hi - m)
        if j_hi >= j_lo:
            indices = np.arange(j_lo, j_hi + 1)
            grad[indices] -= wk * h[indices + m]
    
    grad *= dx
    return grad

    """Project h onto {0 <= x <= 1, sum(x) = target_sum}"""
    n = len(h)
    # Bisection on lambda
    lo, hi = h.min() - 1, h.max() + 1
    for _ in range(60):
        mid = (lo + hi) / 2
        projected = np.clip(h - mid, 0, 1)
        s = projected.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    projected = np.clip(h - (lo + hi) / 2, 0, 1)
    # Final adjustment for numerical precision
    diff = target_sum - projected.sum()
    if abs(diff) > 1e-12:
        # Distribute among elements not at 0 or 1
        mask = (projected > 0) & (projected < 1)
        if mask.sum() > 0:
            projected[mask] += diff / mask.sum()
            projected = np.clip(projected, 0, 1)
    return projected
