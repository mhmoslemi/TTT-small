import numpy as np
import time

def eval_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_binary(h, rng):
    n = len(h)
    h = np.where(h > 0.5, 1.0, 0.0).astype(float)
    target = n // 2
    s = int(np.sum(h))
    if s > target:
        ones = np.where(h == 1)[0]
        idx = rng.choice(ones, s - target, replace=False)
        h[idx] = 0.0
    elif s < target:
        zeros = np.where(h == 0)[0]
        idx = rng.choice(zeros, target - s, replace=False)
        h[idx] = 1.0
    return h

def local_search(h, rng, deadline, max_pairs=None):
    n = len(h)
    current_score = eval_score(h)
    best_h = h.copy()
    best_score = current_score
    stall = 0
    while time.time() < deadline:
        h = best_h.copy()
        ones = list(np.where(h == 1)[0])
        zeros = list(np.where(h == 0)[0])
        if not ones or not zeros:
            break
        if max_pairs is None or len(ones) * len(zeros) <= max_pairs:
            pairs = [(i, j) for i in ones for j in zeros]
        else:
            ntry = min(400, len(ones) * len(zeros))
            pairs = [(int(rng.choice(ones)), int(rng.choice(zeros))) for _ in range(ntry)]
        best_s = current_score
        best_pair = None
        for i, j in pairs:
            h[i] = 0.0
            h[j] = 1.0
            s = eval_score(h)
            if s < best_s - 1e-12:
                best_s = s
                best_pair = (i, j)
            h[i] = 1.0
            h[j] = 0.0
        if best_pair is not None:
            i, j = best_pair
            h[i] = 0.0
            h[j] = 1.0
            current_score = best_s
            best_h = h.copy()
            best_score = best_s
            stall = 0

            stall += 1

    return best_h, best_score

            break
        i = int(rng.choice(ones))
        j = int(rng.choice(zeros))

                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        t *= cool
