import numpy as np
from scipy.optimize import minimize
import time

_target_box = [0.0]

def _con_fun(h):
    return np.sum(h) - _target_box[0]

def _con_jac(h):
    return np.ones(len(h))

def compute_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)

def c5_value(h):
    return float(np.max(compute_corr(h)))

def _grad_weighted(h, w):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    idx = np.arange(n)
    for k in range(len(w)):
        wk = w[k]
        if abs(wk) < 1e-12: continue
        lag = k - (n - 1)
        v1 = (idx + lag >= 0) & (idx + lag < n)
        v2 = (idx - lag >= 0) & (idx - lag < n)
        g[v1] += wk * dx * (1.0 - h[idx[v1] + lag])
        g[v2] -= wk * dx * h[idx[v2] - lag]
    return g

def _smooth(h, eps):
    c = compute_corr(h)
    m = c.max()
    return m + eps * np.log(np.sum(np.exp((c - m) / eps)))

def _smooth_g(h, eps):
    c = compute_corr(h)
    m = c.max()
    e = np.exp((c - m) / eps)
    return _grad_weighted(h, e / e.sum())

def _proj(h, target):
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(h + mid, 0, 1)) > target: hi = mid
        else: lo = mid
    return np.clip(h + 0.5*(lo + hi), 0, 1)

def _sqsolve(h0, target, eps, iters=250):
    _target_box[0] = target
    n = len(h0)
    try:
        r = minimize(_smooth, h0, args=(eps,), jac=_smooth_g,
                     bounds=[(0,1)]*n,
                     constraints=[{'type':'eq','fun':_con_fun,'jac':_con_jac}],
                     method='SLSQP', options={'maxiter':iters,'ftol':1e-12})
        h = _proj(np.clip(r.x, 0, 1), target)
        return h if c5_value(h) <= c5_value(h0) else h0
    except: return h0

    h = h.copy()
    h[np.abs(h) < thresh] = 0.0
    h[np.abs(h - 1) < thresh] = 1.0

from scipy.optimize import minimize
import time

_tb = [0.0]

def _cf(h):
    return np.sum(h) - _tb[0]

def _cj(h):
    return np.ones(len(h))

def _comp(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)

def _c5(h):
    return float(_comp(h).max())

def _si(h, e):
    c = _comp(h)
    m = c.max()
    return m + e * np.log(np.sum(np.exp((c - m) / e)))

def _pj(h, t):
    lo, hi = -1.0, 1.0
    for _ in range(50):
        md = 0.5 * (lo + hi)
        if np.sum(np.clip(h + md, 0, 1)) > t: hi = md
        else: lo = md
    return np.clip(h + 0.5 * (lo + hi), 0, 1)

    n = len(h)

from scipy.optimize import minimize

_tb = [0.0]

def _comp(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)

def _c5(h):
    return float(_comp(h).max())

def _si(h, e):
    c = _comp(h)
    m = c.max()
    return m + e * np.log(np.sum(np.exp((c - m) / e)))

def _pj(h, t):
    lo, hi = -1.0, 1.0
    for _ in range(60):
        md = 0.5 * (lo + hi)
        if np.sum(np.clip(h + md, 0, 1)) > t:
            hi = md
        else:
            lo = md
    return np.clip(h + 0.5 * (lo + hi), 0, 1)

def _sq(h, t, e, mi=200):
    _tb[0] = t
    n = len(h)

                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _cf, 'jac': _cj}],
                     method='SLSQP', options={'maxiter': mi, 'ftol': 1e-12})
        hn = _pj(np.clip(r.x, 0, 1), t)
