def project_sum(h, target):
    """Project h to satisfy sum(h) ≈ target, 0<=h<=1."""
    h = h.copy()
    for _ in range(200):
        s = h.sum()
        if abs(s - target) < 1e-14:

        diff = (target - s) / len(h)
        if diff > 0:
            mask = h < 1
        else:

        if not mask.any():
            break
        h[mask] += diff * len(h) / mask.sum()
