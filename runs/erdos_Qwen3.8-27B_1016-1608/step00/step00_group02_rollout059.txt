    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient(h, k_star, dx):
    """Compute subgradient of max_k corr[k]*dx at h, where k_star achieves max."""
    n = len(h)
    grad = np.zeros(n)
    for j in range(n):
        g = 0.0
        # (1-h[j-k*]) if 0 <= j-k* < n
        jk = j - k_star
        if 0 <= jk < n:
            g += (1.0 - h[jk])
        # -h[j+k*] if 0 <= j+k* < n
        jk2 = j + k_star
        if 0 <= jk2 < n:
            g -= h[jk2]
        grad[j] = dx * g
    return grad

    """Projected subgradient descent with Polyak step size."""
    n = len(h)
    c5_cur, corr = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_cur
    
    # Polyak: need lower bound. Use c5_lb = 0 (trivially valid since corr >= 0)
    # Actually, the minimum possible value of max_k corr[k] is at least some positive number.
    # Let's use 0 as lower bound (safe).
    c5_lb = 0.0
    
    t = 0.01  # step size
    for it in range(max_iter):
        k_star_idx = np.argmax(corr)
        k_star = k_star_idx - (n - 1)  # convert to lag
        grad = subgradient(h, k_star, dx)
        
        # Polyak step
        t_polyak = (c5_cur - c5_lb) / (np.dot(grad, grad) + 1e-15)
        step = min(t_polyak, 0.5)  # cap step
        
        h_new = h - step * grad
        h_new = project(h_new, target_sum)
        
        c5_new, corr_new = compute_c5(h_new)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        
        h = h_new
        corr = corr_new
        c5_cur = c5_new
        
        if c5_cur - c5_lb < tol:
            break
    
    return best_h, best_c5

    """SLSQP optimization."""
    n = len(h)
    
    def obj(h_):
        return compute_c5(h_)[0]
    
    constraints = [{'type': 'eq', 'fun': lambda h_: h_.sum() - target_sum,
                    'jac': lambda h_: np.ones_like(h_)}]

    try:

        h_new = project(res.x, target_sum)
        c5_new, _ = compute_c5(h_new)
        return h_new, c5_new
    except Exception:
        return h, compute_c5(h)[0]

def get_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(vals, beta):
    m = np.max(vals)
    return float(m + np.log(np.sum(np.exp(beta * (vals - m))) / len(vals)) / beta)

    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return smooth_max(c, beta)

    return compute_c5(h)

    return h.sum() - target

    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
