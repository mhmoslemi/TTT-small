```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def repair(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s > 1e-12 and abs(s - target_sum) > 1e-10:
        h = h * (target_sum / s)
        h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
            if len(idx) > 0:
                h[idx] += diff / len(idx)
        else:
            idx = np.where(h > 1e-12)[0]
            if len(idx) > 0:
                h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h


def make_obj(dx, target_sum):
    def obj(h):
        return compute_c5(h, dx)
    return obj


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    bounds = [(0.0, 1.0)] * n_points
    obj = make_obj(dx, target_sum)
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum,
                   'jac': lambda h: np.ones(n_points)}]

    # Get initial point
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.full(n_points, 0.5)
    h_best = repair(h_best, target_sum)
    c5_best = compute_c5(h_best, dx)

    # Phase 1: Polish from initial
    try:
        res = minimize(obj, h_best, method='SLSQP', bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': 3000, 'ftol': 1e-15, 'disp': False})
        if res.x is not None:
            h_try = repair(res.x, target_sum)
            c_try = compute_c5(h_try, dx)
            if c_try < c5_best:
                h_best, c5_best = h_try, c_try
    except Exception:
        pass

    # Phase 2: Iterative restarts with adaptive perturbation
    attempt = 0
    stall = 0
    while time.time() - t0 < budget_s - 10:
        attempt += 1

        if stall > 40:
            # Near-random restart for diversity
            if rng.random() < 0.3:
                h_start = rng.rand(n_points)
            else:
                sigma = 0.25 + 0.15 * rng.random()
                h_start = h_best + rng.normal(0, sigma, n_points)
            stall = 0
        elif stall > 15:
            sigma = 0.08 + 0.05 * rng.random()
            h_start = h_best + rng.normal(0, sigma, n_points)
        else:
            sigma = 0.02 + 0.03 * rng.random()
            h_start = h_best + rng.normal(0, sigma, n_points)

        h_start = repair(h_start, target_sum)

        improved = False
        try:
            res = minimize(obj, h_start, method='SLSQP', bounds=bounds,
                           constraints=constraints,
                           options={'maxiter': 800, 'ftol': 1e-14, 'disp': False})
            if res.x is not None:
                h_try = repair(res.x, target_sum)
                c_try = compute_c5(h_try, dx)
                if c_try < c5_best - 1e-12:
                    h_best, c5_best = h_try, c_try
                    improved = True
        except Exception:
            pass

        # Extra: Nelder-Mead fine-tune on the best found so far
        if improved and time.time() - t0 < budget_s - 30:
            try:
                h_nm = h_best.copy()
                res_nm = minimize(obj, h_nm, method='Nelder-Mead',
                                  bounds=bounds,
                                  options={'maxiter': 2000, 'xatol': 1e-10,
                                           'fatol': 1e-12, 'disp': False})
                if res_nm.x is not None:
                    h_try = repair(res_nm.x, target_sum)
                    c_try = compute_c5(h_try, dx)
                    if c_try < c5_best - 1e-12:
                        h_best, c5_best = h_try, c_try
            except Exception:
                pass

        if not improved:
            stall += 1
        else:
            stall = max(0, stall - 2)

        # Periodic re-polish of current best
        if attempt % 50 == 0 and time.time() - t0 < budget_s - 20:
            try:
                res = minimize(obj, h_best, method='SLSQP', bounds=bounds,
                               constraints=constraints,
                               options={'maxiter': 2000, 'ftol': 1e-15, 'disp': False})
                if res.x is not None:
                    h_try = repair(res.x, target_sum)
                    c_try = compute_c5(h_try, dx)
                    if c_try < c5_best:
                        h_best, c5_best = h_try, c_try
            except Exception:
                pass

    # Final polish
    if time.time() - t0 < budget_s - 2:
        try:
            res = minimize(obj, h_best, method='SLSQP', bounds=bounds,
                           constraints=constraints,
                           options={'maxiter': 5000, 'ftol': 1e-16, 'disp': False})
            if res.x is not None:
                h_try = repair(res.x, target_sum)
                c_try = compute_c5(h_try, dx)
                if c_try < c5_best:
                    h_best, c5_best = h_try, c_try
        except Exception:
            pass

    return h_best, c5_best, n_points
```
