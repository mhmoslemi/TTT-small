    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    beta = 40.0
    mx = np.max(corr)
    lse = mx + np.log(np.sum(np.exp(corr - mx))) / beta
    return lse

def constraint_eq(x):
    return x.sum() - STATE['target']

def de_obj(x):
    h = np.clip(x, 0.0, 1.0)
    return evaluate_c5(h) + 20.0 * (h.sum() - STATE['target']) ** 2

def local_opt(h, n, maxiter=400, ftol=1e-12):
    STATE['target'] = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_eq}]
    try:
        res = optimize.minimize(obj_smooth, h, method='SLSQP',
                                bounds=bounds, constraints=cons,
                                options={'maxiter': maxiter, 'ftol': ftol})
        return fix_sum(res.x, n / 2.0)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h_init = None
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, dtype=float)

    best_h = None
    best_c5 = float('inf')

    if h_init is not None:
        best_h = fix_sum(h_init, len(h_init) / 2.0)
        best_c5 = evaluate_c5(best_h)

    n = len(best_h) if best_h is not None else 70

    # Phase 1: SLSQP from initial
    if best_h is not None and time.time() - t0 < budget_s * 0.2:
        h_loc = local_opt(best_h, n, maxiter=500)
        c5_loc = evaluate_c5(h_loc)
        if c5_loc < best_c5:
            best_c5 = c5_loc
            best_h = h_loc

    # Phase 2: Differential Evolution
    if time.time() - t0 < budget_s * 0.45:
        STATE['target'] = n / 2.0
        bounds = [(0.0, 1.0)] * n
        try:
            from scipy.optimize import differential_evolution
            res = differential_evolution(de_obj, bounds, seed=seed,
                                         maxiter=200, popsize=25,
                                         tol=1e-6, polish=False,
                                         mutation=(0.4, 1.0),
                                         crossover=0.9)
            h_de = fix_sum(res.x, n / 2.0)
            h_ref = local_opt(h_de, n, maxiter=300)
            c5_ref = evaluate_c5(h_ref)
            if c5_ref < best_c5:
                best_c5 = c5_ref
                best_h = h_ref
        except Exception:
            pass

    # Phase 3: Multi-restart perturbation + SLSQP
    n = len(best_h)
    target = n / 2.0
    max_iter3 = 40
    for it in range(max_iter3):
        if time.time() - t0 > budget_s * 0.82:
            break
        frac = it / max_iter3
        scale = 0.12 * (1.0 - frac) + 0.01
        if it % 4 == 2:
            noise = rng.randn(n) * scale
            h_pert = fix_sum(best_h + noise, target)
        elif it % 4 == 3:
            h_pert = best_h.copy()
            k = rng.randint(1, 6)
            for _ in range(k):
                idx = rng.randint(n)
                h_pert[idx] = rng.uniform(0, 1)
            h_pert = fix_sum(h_pert, target)
        else:
            noise = rng.randn(n) * scale
            h_pert = fix_sum(best_h + noise, target)
        h_new = local_opt(h_pert, n, maxiter=150)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Phase 4: Binary swap local search
    if time.time() - t0 < budget_s * 0.93:
        h_bin = (best_h > 0.5).astype(float)
        n_half = n // 2
        while h_bin.sum() > n_half:
            idx = rng.choice(np.where(h_bin == 1.0)[0])
            h_bin[idx] = 0.0
        while h_bin.sum() < n_half:
            idx = rng.choice(np.where(h_bin == 0.0)[0])
            h_bin[idx] = 1.0
        c5_bin = evaluate_c5(h_bin)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()

        for _ in range(8000):
            if time.time() - t0 > budget_s * 0.93:
                break
            i, j = rng.choice(n, 2, replace=False)
            if (best_h[i] > 0.5) == (best_h[j] > 0.5):
                continue
            best_h[i], best_h[j] = best_h[j], best_h[i]
            c5_tmp = evaluate_c5(best_h)
            if c5_tmp > best_c5:
                best_h[i], best_h[j] = best_h[j], best_h[i]
        best_c5 = min(best_c5, evaluate_c5(best_h))

    # Phase 5: Continuous refinement if best is binary
    if time.time() - t0 < budget_s * 0.97:
        n = len(best_h)
        if np.all((best_h < 0.1) | (best_h > 0.9)):
            h_start = best_h.astype(float) + 0.15 * rng.standard_t(3, n)
        else:
            h_start = best_h + 0.02 * rng.randn(n)
        h_start = fix_sum(h_start, n / 2.0)
        h_fin = local_opt(h_start, n, maxiter=1500, ftol=1e-13)
        c5_fin = evaluate_c5(h_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin

    # Final cleanup
    best_h = fix_sum(best_h, len(best_h) / 2.0)
    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, len(best_h)
