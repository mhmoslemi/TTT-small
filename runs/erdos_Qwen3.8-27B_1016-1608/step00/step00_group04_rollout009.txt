        to_flip = rng.choice(ones, s - target, replace=False)
        h[to_flip] = 0
    elif s < target:
        zeros = np.where(h == 0)[0]
        to_flip = rng.choice(zeros, target - s, replace=False)
        h[to_flip] = 1
    return h

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def fix_binary_sum(h, rng):
    n = len(h)
    target = n // 2
    h = (h > 0.5).astype(float)
    s = int(np.sum(h))
    if s > target:
        ones = np.where(h == 1)[0]
        for idx in rng.choice(ones, s - target, replace=False):
            h[idx] = 0
    elif s < target:
        zeros = np.where(h == 0)[0]
        for idx in rng.choice(zeros, target - s, replace=False):
            h[idx] = 1
    return h

def binary_local_search(h, max_time=30, deadline=None):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    t_start = time.time()
    while time.time() - t_start < max_time:
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        for iv in ones:

                h[iv] = 0
                h[jv] = 1
                nc = compute_c5(h)
                if nc < c5:
                    c5 = nc
                    improved = True
                else:
                    h[iv] = 1
                    h[jv] = 0
        if not improved:
            break
    return h, c5
