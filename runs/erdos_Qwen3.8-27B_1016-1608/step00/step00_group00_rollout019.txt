                h4 = fix_sum(np.clip(r.x, 0, 1), tgt2)
                c4 = c5_of(h4, dx2)
                if c4 < best_c5:
                    best_c5, best_h, n = c4, h4, n2
                    dx, tgt = dx2, tgt2
            except Exception:
                pass
        # Coordinate descent at higher res
        if rem() > 30 and c5_of(h_interp, dx2) < best_c5 + 0.01:
            hc2 = h_interp.copy()
            corr2 = np.correlate(hc2, 1.0 - hc2, mode="full")
            for it2 in range(int(rem() * 50)):
                if rem() < 15:
                    break
                i2 = it2 % n2
                a2 = _coord_effect(hc2, i2)
                lo2, hi2 = -hc2[i2], 1.0 - hc2[i2]
                for _ in range(35):
                    mm1 = lo2 + (hi2 - lo2) / 3.0
                    mm2 = hi2 - (hi2 - lo2) / 3.0
                    if (corr2 + mm1 * a2).max() <= (corr2 + mm2 * a2).max():
                        hi2 = mm2
                    else:
                        lo2 = mm1
                dv2 = (lo2 + hi2) / 2.0
                if abs(dv2) < 1e-12:
                    continue
                hc2[i2] = np.clip(hc2[i2] + dv2, 0, 1)
                hc2 = fix_sum(hc2, tgt2)
                corr2 = np.correlate(hc2, 1.0 - hc2, mode="full")
                cc2 = corr2.max() * dx2
                if cc2 < best_c5:
                    best_c5, best_h, n = cc2, hc2.copy(), n2
                    dx, tgt = dx2, tgt2

    return best_h, best_c5, n
