import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair_sum(h, target_sum):
    h = h.copy()
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = np.maximum(1.0 - h, 1e-12)
        tot = np.sum(room)
        if tot < 1e-12:
            return np.clip(h + diff / len(h), 0, 1)
        h += diff * room / tot
    else:
        room = np.maximum(h, 1e-12)
        tot = np.sum(room)
        if tot < 1e-12:
            return np.clip(h + diff / len(h), 0, 1)
        h += diff * room / tot
    return np.clip(h, 0, 1)

def local_search_binary(h, target_sum, max_passes=80, t0=None, deadline=None, rng=None):

    h = (h >= 0.5).astype(float)
    h = repair_sum(h, target_sum)

            break

        ones = list(np.where(h == 1)[0])

            done = False
            for j in ones:
                h[i], h[j] = 1.0, 0.0
