import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def repair_binary(h, n):
    h = (h > 0.5).astype(float)
    target = n // 2
    diff = target - int(h.sum())
    if diff > 0:
        zeros = np.where(h == 0)[0]
        h[zeros[:diff]] = 1.0
    elif diff < 0:
        ones = np.where(h == 1)[0]
        h[ones[(-diff):]] = 0.0
    return h

def binary_sa(h, n, dx, time_limit, rng):
    h = repair_binary(h, n).copy()
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    T = 0.003
    min_T = 1e-6
    no_improve = 0
    end_time = time.time() + time_limit
    batch = 200
    while time.time() < end_time:
        for _ in range(batch):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j or h[i] == h[j]:
                continue
            h[i], h[j] = 1.0 - h[i], 1.0 - h[j]
            c5 = compute_c5(h, dx)
            delta = c5 - best_c5
            if delta < -1e-12:
                best_c5 = c5
                best_h = h.copy()
                no_improve = 0
            elif delta <= 0 or (delta / T < -np.log(max(rng.random(), 1e-15))):
                no_improve += 1
            else:
                h[i], h[j] = 1.0 - h[i], 1.0 - h[j]
                no_improve += 1
        if no_improve > 20000:
            T = max(T * 0.5, min_T)
            no_improve = 0
            if T <= min_T * 2:
                h = best_h.copy()
    return best_h, best_c5

def block_move(h, n, rng):
    """Larger perturbation: shift a block or swap 3-4 elements"""
    h = h.copy()
    move_type = rng.randint(3)
    if move_type == 0:
        # Swap 3 elements cyclically
        idx = rng.choice(n, 4, replace=False)
        a, b, c, d = idx
        va, vb = h[a], h[b]
        h[a], h[b] = 1.0 - vb, 1.0 - va
    elif move_type == 1:
        # Swap two blocks
        p = rng.randint(0, n - 4)
        length = rng.randint(1, min(4, n - p))
        ones_in_block = np.where(h[p:p+length] == 1)[0]
        if len(ones_in_block) == 0 or len(ones_in_block) == length:
            return h
        q = rng.randint(0, n - length)
        if q >= p and q < p + length:
            return h
        seg_a = h[p:p+length]
        seg_b = h[q:q+length]
        ones_a = seg_a.sum()
        ones_b = seg_b.sum()
        if ones_a != ones_b:
            return h
        h[p:p+length] = seg_b
        h[q:q+length] = seg_a
    else:
        # Random 4-swap
        idx = rng.choice(n, 8, replace=False)
        for k in range(4):
            i, j = idx[2*k], idx[2*k+1]
            if h[i] != h[j]:
                h[i], h[j] = 1.0 - h[i], 1.0 - h[j]
    return h

def binary_sa_perturb(h, n, dx, time_limit, rng):
    h = repair_binary(h, n).copy()
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    T = 0.003
    min_T = 1e-6
    no_improve = 0
    end_time = time.time() + time_limit
    while time.time() < end_time:
        for _ in range(150):
            if rng.random() < 0.15:
                new_h = block_move(h, n, rng)
                if np.sum(np.abs(new_h - h)) < 0.5:
                    continue
                h = new_h
            else:
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j or h[i] == h[j]:
                    continue
                h[i], h[j] = 1.0 - h[i], 1.0 - h[j]
            c5 = compute_c5(h, dx)
            delta = c5 - best_c5
            if delta < -1e-12:
                best_c5 = c5
                best_h = h.copy()
                no_improve = 0
            elif delta <= 0 or (delta / T < -np.log(max(rng.random(), 1e-15))):
                no_improve += 1
            else:
                # revert last move (approximate: just keep going)
                no_improve += 1
        if no_improve > 15000:
            T = max(T * 0.5, min_T)
            no_improve = 0
            if T <= min_T * 3:
                h = best_h.copy()
    return best_h, best_c5
