            temp = max(temp * 0.97, 1e-6)
        else:
            no_improve += 1
            temp *= 0.88
            if no_improve >= 10:
                h = h_best.copy()
                for _ in range(rng.randint(3, 10)):
                    a, b = rng.randint(n), rng.randint(n)
                    h[a], h[b] = h[b], h[a]
                c5 = _eval_c5(h, dx)
                temp = 0.004
                no_improve = 0
    return h_best, c5_best

def _local_refine(h_best, c5_best, n, dx, target, rng, t_end):
    h = h_best.copy()
    c5 = c5_best
    steps = [0.005, 0.01, 0.02]
    while time.time() < t_end:
        improved = False
        for i in range(n):
            for s in steps:
                for dd in (s, -s):
                    ht = h.copy()
                    ht[i] = np.clip(ht[i] + dd, 0.0, 1.0)
                    ht = _repair(ht, target)
                    ct = _eval_c5(ht, dx)
                    if ct < c5 - 1e-11:
                        h, c5, improved = ht, ct, True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    final = (h, c5) if c5 < c5_best else (h_best, c5_best)
    return final

def run(seed=42, budget_s=1100, **kwargs):
    T0 = time.time()
    TEND = T0 + budget_s - 5.0
    rng = np.random.RandomState(seed)
    n = 45
    dx = 2.0 / n
    target = n / 2.0

    candidates = []
    try:
        init = np.array(initial_h_values, dtype=np.float64)
        if len(init) == n:
            candidates.append(_repair(init.copy(), target))
    except Exception:
        pass

    for _ in range(10):
        hr = np.zeros(n)
        idx = rng.choice(n, size=22, replace=False)
        hr[idx] = 1.0
        hr[rng.randint(n)] = 0.5
        candidates.append(_repair(hr, target))

    h_best, c5_best = None, 1.0
    for c in candidates:
        cc = _eval_c5(c, dx)
        if h_best is None or cc < c5_best:
            c5_best, h_best = cc, c.copy()

    h_best, c5_best = _try_sa(h_best, c5_best, n, dx, target, rng, TEND - 100.0)

    if time.time() < TEND - 15.0:
        h_best, c5_best = _local_refine(h_best, c5_best, n, dx, target, rng, TEND - 15.0)
        h_r = _round_binary(h_best, n, target)
        cr = _eval_c5(h_r, dx)
        if cr < c5_best:
            c5_best, h_best = cr, h_r

    if time.time() < TEND - 5.0 and n != 44:
        n2, dx2, targ2 = 44, 2.0 / 44, 22.0
        h2 = np.zeros(n2)
        h2[:22] = 1.0
        t2 = time.time()
        t_end2 = min(t2 + 40.0, TEND - 5.0)
        h2_best, c5_2 = _try_sa(h2, _eval_c5(h2, dx2), n2, dx2, targ2, np.random.RandomState(seed + 1), t_end2)
        if c5_2 < c5_best:
            h_best, c5_best, n = h2_best, c5_2, n2

    h_best = _repair(h_best, n / 2.0)
    c5_final = _eval_c5(h_best, 2.0 / n)
    return h_best, c5_final, n
