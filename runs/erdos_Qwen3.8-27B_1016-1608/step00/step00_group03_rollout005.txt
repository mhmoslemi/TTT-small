    return np.correlate(h, 1.0 - h, mode='full')

    n = len(h)
    lag = k_star - (n - 1)
    grad = np.zeros(n)
    # corr[k_star] = sum_i h[i+lag] * (1-h[i]) for valid i

import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5_val(h):
    n = len(h)
    return float(np.max(compute_corr(h)) * (2.0 / n))

def subgradient_one(h, k_idx):
    n = len(h)
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        if lag < n:
            m = n - lag
            grad[lag:] += (1.0 - h[:m])
            grad[:m] -= h[lag:]
    else:
        l = -lag
        if l < n:
            m = n - l
            grad[:m] += (1.0 - h[l:])
            grad[l:] -= h[:m]
    return grad

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0

        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:

        else:

def do_subgradient_phase(h, best_h, best_c5, rng, t_end, n, target_sum):
