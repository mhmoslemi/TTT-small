import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def lag_gradient(h, d):
    """Gradient of the overlap at lag d with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    for i in range(n):
        g = 0.0
        # h[i] appears as first factor if i is valid start
        if 0 <= i + d < n:  # i + d valid
            g += 1.0 - h[i + d]
        # h[i] appears as second factor if i-d is valid start
        if 0 <= i - d < n:  # i - d valid, meaning the pair is (i-d, i)
            g -= h[i - d]
        grad[i] = g
    return grad

        new_diff = target_sum - np.sum(h)

import numpy as np
import time

def proj_sum_clip(h, target):
    """Project onto {x: 0<=x<=1, sum(x)=target}"""
    h = h.copy()
    for _ in range(10):
        h = np.clip(h, 0, 1)
        diff = target - h.sum()
        if abs(diff) < 1e-10:
            break
        # Distribute to non-saturated elements
        free = (h > 0) & (h < 1)
        if free.sum() == 0:
            h = np.clip(h + diff / len(h), 0, 1)
        else:
            h[free] += diff / free.sum()
    return h

def full_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def c5_of(h):
    return full_corr(h).max() * 2.0 / len(h)

def grad_at_lag(h, d):
    n = len(h)
    g = np.zeros(n)
    for i in range(n):
        val = 0.0
        j1 = i + d
        if 0 <= j1 < n:
            val += 1.0 - h[j1]
        j2 = i - d
        if 0 <= j2 < n:
            val -= h[j2]
        g[i] = val
    return g

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    h = np.array(initial_h_values, dtype=float)
    n = len(h)
    target = n / 2.0
    # Ensure constraint
    h = proj_sum_clip(h, target)
    
    best = h.copy()
    best_c5 = c5_of(best)
    
    # Subgradient with momentum
    lr = 0.1
    momentum = 0.9
    v = np.zeros(n)
    it = 0
    patience = 0
    
    while time.time() - t0 < budget_s - 5:
        corr = full_corr(best)
        c5_curr = corr.max() * 2.0 / n
        # Top K lags
        top_k = np.argsort(corr)[-3:][::-1]
        lags = top_k - (n - 1)
        
        # Average gradient over top K lags
        grad = np.zeros(n)
        for d in lags:
            grad += grad_at_lag(best, d)
        grad /= len(lags)
        
        # Momentum update
        v = momentum * v + grad
        h_new = best - lr * v
        h_new = proj_sum_clip(h_new, target)
        
        c5_new = c5_of(h_new)
        if c5_new < best_c5:
            best = h_new
            best_c5 = c5_new
            it += 1
            patience = 0
            if it % 500 == 0:
                lr *= 0.98
        else:
            patience += 1
            if patience % 100 == 0:
                lr *= 0.95
                if patience > 500:
                    # Random restart perturbation
                    h_p = best + 0.05 * rng.randn(n)
                    h_p = proj_sum_clip(h_p, target)
                    if c5_of(h_p) < best_c5:
                        best = h_p
                        best_c5 = c5_of(best)
                    patience = 0
    
    return best, best_c5, n

import numpy as np

def proj_sum(h, target):
    """Project h onto {x: 0<=x<=1, sum(x)=target}"""
    h = np.clip(h, 0, 1)
    for _ in range(20):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-12) & (h < 1 - 1e-12)
        if free.sum() == 0:
            h = np.clip(h + diff / len(h), 0, 1)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0, 1)
    return h

def c5_score(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * 2.0 / n)

    base = h.copy()
    base[i] = 0.0
    corr_base = np.correlate(base, 1.0 - base, mode="full")

    delta = np.zeros(2 * n - 1)
    k_range = np.arange(2 * n - 1)

    idx2 = i - k_range + (n - 1)

    # Fix the middle index
    vals[:, k_mid] = corr_base[k_mid] + t_grid - t_grid**2

def proj_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(30):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)

            h = np.clip(h + diff / len(h), 0, 1)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0, 1)
    return h

    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * 2.0 / n)

    n = len(h)
    k_mid = n - 1
    k_range = np.arange(2 * n - 1)
    
    # Randomize coordinate order
    order = np.arange(n)
    rng.shuffle(order)
    
    for i in order:
        base = h.copy()
        base[i] = 0.0
        corr_base = np.correlate(base, 1.0 - base, mode="full")
        
        idx1 = i + k_range - k_mid
        idx2 = i - k_range + k_mid
        valid1 = (idx1 >= 0) & (idx1 < n)
        valid2 = (idx2 >= 0) & (idx2 < n) & (idx2 != i)
        
        delta = np.where(valid1, 1.0 - h[np.clip(idx1, 0, n-1)], 0.0)
        delta -= np.where(valid2, h[np.clip(idx2, 0, n-1)], 0.0)
        
        t_grid = np.linspace(0, 1, grid_n)

        vals[:, k_mid] = corr_base[k_mid] + t_grid - t_grid * t_grid
        f_vals = vals.max(axis=1)
        
        best_idx = f_vals.argmin()
        best_t = t_grid[best_idx]
        
        # Refine

        t_hi = min(1.0, best_t + 2.0 / grid_n)
