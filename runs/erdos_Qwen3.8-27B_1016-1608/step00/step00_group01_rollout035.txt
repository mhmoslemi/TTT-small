    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _repair(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    return h

def _obj(x, n):
    return _c5_eval(x, n)

def _grad(x, n):
    # Numerical gradient
    h = x.copy()
    f0 = _c5_eval(h, n)
    g = np.zeros(n)
    eps = 1e-7
    for i in range(n):
        h[i] += eps
        g[i] = (_c5_eval(h, n) - f0) / eps
        h[i] -= eps
    return g

def _slsqp_run(h0, n, maxiter=500):
    target = n / 2.0
    def csum(x):
        return x.sum() - target
    constraints = [{'type': 'eq', 'fun': csum}]
    bounds = [(0.0, 1.0)] * n
    res = minimize(_obj, h0, args=(n,), method='SLSQP', bounds=bounds,
                   constraints=constraints, options={'maxiter': maxiter, 'ftol': 1e-14})
    return res.x, res.fun

def _subgrad_step(h, n, lr, iters=50):
    for it in range(iters):
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_star = np.argmax(corr)
        lag = k_star - (n - 1)
        # Compute gradient of f_{lag}
        g = np.zeros(n)

import numpy as np
from scipy.optimize import minimize
import time

_gn = [47]

def _c5_eval(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _repair(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _objective(x):
    return _c5_eval(x, _gn[0])

def _con_fun(x):
    return float(x.sum() - _gn[0] / 2.0)

def _slsqp(h0, n, maxiter=500, ftol=1e-14):
    _gn[0] = n
    cons = {'type': 'eq', 'fun': _con_fun}
    bnds = [(0.0, 1.0)] * n
    res = minimize(_objective, h0, method='SLSQP', bounds=bnds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    return res.x, float(res.fun)

def _subgrad(h, n, lr, iters):
    for _ in range(iters):
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_star = int(np.argmax(corr))
        lag = k_star - (n - 1)
        g = np.zeros(n)
        if lag >= 0:
            for j in range(n - lag):
                g[j] += (1.0 - h[j + lag])
            for j in range(lag, n):
                g[j] -= h[j - lag]
        else:
            m = -lag
            for j in range(m, n):
                g[j] += (1.0 - h[j - m])
            for j in range(n - m):
                g[j] -= h[j + m]
        h -= lr * (2.0 / n) * g
        h = _repair(h, n)
    return h

def _perturb_init(n, rng, base_h):
    h_r = base_h + rng.randn(n) * 0.04
    return _repair(h_r, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except (NameError, UnboundLocalError, TypeError, ValueError):
        n = 47
        h_init = np.full(n, 0.5)

    h_init = _repair(h_init, n)
    best_h = h_init.copy()
    best_c5 = _c5_eval(best_h, n)

    # Primary SLSQP
    x, f = _slsqp(h_init, n, 1500, 1e-15)
    x = _repair(x, n)
    f = _c5_eval(x, n)
    if f < best_c5:
        best_h, best_c5 = x.copy(), f

    # Subgradient descent then SLSQP polish
    if time.time() - t0 < budget_s * 0.3:
        h_sg = _subgrad(best_h.copy(), n, 0.003, 150)
        f_sg = _c5_eval(h_sg, n)
        if f_sg < best_c5:
            best_h, best_c5 = h_sg, f_sg
        x, f = _slsqp(best_h, n, 800, 1e-15)
        x = _repair(x, n)
        f = _c5_eval(x, n)
        if f < best_c5:
            best_h, best_c5 = x.copy(), f

    # Multiple restarts with increasing perturbation
    for r in range(50):
        if time.time() - t0 > budget_s * 0.70:
            break
        scale = 0.02 + 0.003 * r
        h_r = _repair(best_h + rng.randn(n) * scale, n)
        x, f = _slsqp(h_r, n, 250, 1e-13)
        x = _repair(x, n)
        f = _c5_eval(x, n)
        if f < best_c5:
            best_h, best_c5 = x.copy(), f

    # Try structured initializations
    for n2 in [n, 59, 79, 101]:
        if time.time() - t0 > budget_s * 0.82:

        for init_type in range(3):

    best_h = _repair(best_h, len(best_h))
    best_c5 = _c5_eval(best_h, len(best_h))
    return best_h, best_c5, len(best_h)
