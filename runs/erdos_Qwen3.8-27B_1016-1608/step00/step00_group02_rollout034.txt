```python
def project_onto_feasible(v, S):
    """Project v onto {h : 0<=h<=1, sum(h)=S}"""
    n = len(v)
    lo, hi = -1.0, 1.0
    for _ in range(100):

        h = np.clip(v + mid, 0, 1)
        if np.sum(h) > S:

        else:

    if len(idx) > 0:

    n = len(h)
    corr_vals = np.correlate(h, 1 - h, mode='full')

    # lag = k - (n-1)

                grad[j1] += (1 - h[j2])
```

```python
    n = len(h0)
    S = n / 2.0
    dx = 2.0 / n
    
    def objective(x):
        return x[-1]  # t
    
    def constraints(x):
        h = x[:-1]
        t = x[-1]
        corr = np.correlate(h, 1-h, mode='full') * dx
        # Inequality constraints: corr_k - t <= 0
        ineq = t - corr  # >= 0
        # Equality: sum(h) - S = 0
        eq = np.sum(h) - S
        return list(ineq) + [eq]

    t0 = np.max(corr) + 0.1
```

```python
    n = len(h)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr)

def project_feasible(v, S):
    n = len(v)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        h = np.clip(v + mid, 0.0, 1.0)
        if np.sum(h) > S:
            hi = mid
        else:
            lo = mid
    h = np.clip(v + 0.5 * (lo + hi), 0.0, 1.0)
    diff = S - np.sum(h)
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h

def grad_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        m = n - lag
        g[:m] += (1.0 - h[lag:])
        g[lag:] += (-h[:m])
    else:
        la = -lag
        m = n - la
        g[la:] += (1.0 - h[:m])
        g[:m] += (-h[la:])
    return g

def subgrad_step(h, dx, soft_tol=1e-4):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    active = corr >= cmax - soft_tol * max(cmax + 1e-10, 1.0)
    if not np.any(active):
        active = np.zeros(len(corr), dtype=bool)
        active[np.argmax(corr)] = True
    grad = np.zeros(n)
    for k in np.where(active)[0]:
        lag = k - (n - 1)
        grad += grad_lag(h, lag)
    n_active = float(np.sum(active))
    grad /= n_active
    val = cmax * dx
    return val, grad * dx

def subgradient_opt(h0, max_iter=20000, time_s=300, quiet=True):
    n = len(h0)
    S = n / 2.0
    dx = 2.0 / n
    h = project_feasible(h0.astype(float).copy(), S)
    best_h = h.copy()
    best_val = compute_c5(h, dx)
    t0 = time.time()
```

```python
            hi = mid

def grad_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:

        g[:m] += (1.0 - h[lag:])
        g[lag:] += (-h[:m])
    else:

        m = n - la
        g[la:] += (1.0 - h[:m])
        g[:m] += (-h[la:])
    return g
```

```python
    for lag in range(1, n):
        w_pos = weights[n-1+lag]  # position for positive lag
        w_neg = weights[n-1-lag]  # position for negative lag
        m = n - lag
        w = w_pos + w_neg
        if w > 1e-10:
            # Both positive and negative lag contribute the same pattern
            grad[:m] += w * (1.0 - h[lag:])
            grad[lag:] += w * (-h[:m])
    
    return cmax * dx, grad * dx
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_feasible(v, S):
    n = len(v)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        h = np.clip(v + mid, 0.0, 1.0)
        if np.sum(h) > S:
            hi = mid
        else:
            lo = mid
    h = np.clip(v + 0.5 * (lo + hi), 0.0, 1.0)
    diff = S - np.sum(h)
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_fast(h, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    spread = 0.02 * (cmax + 1e-10)
    weights = np.maximum(0.0, cmax - corr + spread) ** 2
    wsum = np.sum(weights)
    if wsum < 1e-15:
        weights = np.zeros(len(corr))
        weights[np.argmax(corr)] = 1.0
    else:
        weights /= wsum
    grad = np.zeros(n)
    w0 = weights[n - 1]
    if w0 > 1e-12:
        grad += w0 * (1.0 - 2.0 * h)
    for lag in range(1, n):
        wp = weights[n - 1 + lag]
        wn = weights[n - 1 - lag]
        m = n - lag
        if wp > 1e-12:
            grad[:m] += wp * (1.0 - h[lag:])
            grad[lag:] += wp * (-h[:m])
        if wn > 1e-12:
            grad[lag:] += wn * (1.0 - h[:m])
            grad[:m] += wn * (-h[lag:])
    return cmax * dx, grad * dx

def subgradient_opt(h0, max_iter=30000, time_s=300):
    n = len(h0)
    S = n / 2.0
    dx = 2.0 / n
    h = project_feasible(h0.astype(float).copy(), S)
    best_h = h.copy()
    best_val = compute_c5(h, dx)
    t0 = time.time()
    v = np.zeros(n)
    mu = 0.6
    for it in range(max_iter):
        if time.time() - t0 > time_s:
            break
        val, grad = subgrad_fast(h, dx)
        if val < best_val - 1e-13:
            best_val = val
            best_h = h.copy()
        alpha = 1.5 / (1.0 + 0.02 * it)
        v = mu * v - alpha * grad
        vnorm = np.linalg.norm(v)
        if vnorm > 0.5:
            v *= 0.5 / vnorm
        h = project_feasible(h + v, S)
        if it % 800 == 0 and it > 0:
            h_alt = project_feasible(h - 0.05 * grad, S)
            va = compute_c5(h_alt, dx)
            if va < val:
                h = h_alt
                v = np.zeros(n)
                if va < best_val:
                    best_val = va
                    best_h = h.copy()
    return best_h, best_val

def refine_local(h0, time_s=60):
    n = len(h0)
    S = n / 2.0
    dx = 2.0 / n
    h = project_feasible(h0.astype(float).copy(), S)
    best_h = h.copy()
    best_val = compute_c5(h, dx)
    t0 = time.time()
    it = 0
    while time.time() - t0 < time_s:
        val, grad = subgrad_fast(h, dx)
        a = 0.005 / (1.0 + 0.002 * it)
        h = project_feasible(h - a * grad, S)
        v = compute_c5(h, dx)
        if v < best_val:
            best_val = v
            best_h = h.copy()
        it += 1
    return best_h, best_val

def slsqp_obj(x):
    return x[-1]

def slsqp_ineq(x):
    n = len(x) - 1
    h = x[:-1]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return x[-1] - corr

def slsqp_eq(x):
    n = len(x) - 1
    return np.sum(x[:-1]) - n / 2.0

def slsqp_solve(h0, maxiter=500):
    n = len(h0)
    dx = 2.0 / n
    c0 = compute_c5(h0, dx)
    x0 = np.append(h0.copy(), c0 + 0.05)
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {'type': 'ineq', 'fun': slsqp_ineq},
        {'type': 'eq', 'fun': slsqp_eq},
    ]
    res = minimize(slsqp_obj, x0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-10})
    if res.success or res.x is not None:
        h_opt = project_feasible(res.x[:-1], n / 2.0)
        c_opt = compute_c5(h_opt, dx)
        return h_opt, c_opt
    return h0.copy(), c0

def make_ripple(n, n_riple=5, amp=0.35, seed=0):
    x = np.linspace(0.0, 2.0, n)
    h = np.full(n, 0.5)
    for j in range(n_riple):
        freq = (j + 1) * np.pi
        h += (amp / (j + 1.0)) * np.sin(freq * x + j * 1.3)
    return project_feasible(h, n / 2.0)

def upsample(h_old, n_new):
    x_old = np.linspace(0.0, 2.0, len(h_old))
    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    pv = globals().get('initial_h_values', None)
    if pv is not None:
        h0 = np.asarray(pv, dtype=float).flatten()
    else:
        h0 = make_ripple(45, n_riple=5, seed=seed)
    n0 = len(h0)
    h0 = project_feasible(h0, n0 / 2.0)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h, 2.0 / n0)
    best_n = n0

    def _elapsed():
        return time.time() - t_start

    # Phase 1: SLSQP at native resolution
    rem = budget_s - _elapsed() - 10
    if rem > 30:
        h_sl, c5_sl = slsqp_solve(h0, maxiter=400)
        if c5_sl < best_c5:
            best_c5 = c5_sl
            best_h = h_sl.copy()
            best_n = n0
        # Refine after SLSQP
        rem = budget_s - _elapsed() - 10
        if rem > 20:
            h_rf, c5_rf = refine_local(h_sl, time_s=min(50, int(rem * 0.6)))
            if c5_rf < best_c5:
                best_c5 = c5_rf
                best_h = h_rf.copy()
                best_n = n0

    # Phase 2: Subgradient at native resolution
    rem = budget_s - _elapsed() - 10
    if rem > 60:
        h_sg, c5_sg = subgradient_opt(h0, max_iter=30000, time_s=min(250, int(rem * 0.7)))
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            best_n = n0
        # SLSQP polish on subgradient result
        rem = budget_s - _elapsed() - 10
        if rem > 20:
            h_sl2, c5_sl2 = slsqp_solve(h_sg, maxiter=300)
            if c5_sl2 < best_c5:
                best_c5 = c5_sl2
                best_h = h_sl2.copy()
                best_n = n0
            rem = budget_s - _elapsed() - 10
            if rem > 15:
                h_rf2, c5_rf2 = refine_local(best_h, time_s=min(40, int(rem * 0.7)))
                if c5_rf2 < best_c5:
                    best_c5 = c5_rf2
                    best_h = h_rf2.copy()
                    best_n = n0

    # Phase 3: Higher resolution
    for n_hi in [90, 180]:
        rem = budget_s - _elapsed() - 15
        if rem < 80:
            break
        h_up = upsample(best_h, n_hi)
        h_up = project_feasible(h_up, n_hi / 2.0)
        t_alloc = min(int(rem * 0.5), 220)
        h_hi, c5_hi = subgradient_opt(h_up, max_iter=20000, time_s=t_alloc)
        if c5_hi < best_c5:
            best_c5 = c5_hi
            best_h = h_hi.copy()
            best_n = n_hi
        rem = budget_s - _elapsed() - 15
        if rem > 25 and best_n == n_hi:
            h_sl3, c5_sl3 = slsqp_solve(h_hi, maxiter=200)
            if c5_sl3 < best_c5:
                best_c5 = c5_sl3
                best_h = h_sl3.copy()
                best_n = n_hi

    # Phase 4: Random restarts at n0
    rem = budget_s - _elapsed() - 10
    for att in range(10):
        if rem < 35:
            break
        h_rand = 0.5 * h0 + 0.5 * make_ripple(n0, n_riple=att + 3, seed=seed + 500 + att)
        h_rand = project_feasible(h_rand, n0 / 2.0)
        t_att = min(60, int(rem * 0.35))
        h_r, c5_r = subgradient_opt(h_rand, max_iter=15000, time_s=t_att)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n0
        rem = budget_s - _elapsed() - 10

    # Final SLSQP + refine on best
    rem = budget_s - _elapsed() - 5
    if rem > 15 and best_n < 200:
        h_final, c5_final = slsqp_solve(best_h, maxiter=200)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()
    rem = budget_s - _elapsed() - 3
    if rem > 10:
        h_final2, c5_final2 = refine_local(best_h, time_s=min(20, int(rem * 0.8)))
        if c5_final2 < best_c5:
            best_c5 = c5_final2
            best_h = h_final2.copy()

    # Ensure exact constraint
    S = best_n / 2.0
    best_h = project_feasible(best_h, S)
    best_c5 = compute_c5(best_h, 2.0 / best_n)
    return best_h, best_c5, best_n
```
