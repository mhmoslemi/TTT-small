import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def c5_obj_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    c5 = float(np.max(corr) * dx)
    k = int(np.argmax(corr))
    shift = k - (n - 1)
    grad = np.zeros(n)
    for m in range(n):
        g = 0.0
        i1 = m + shift
        if 0 <= i1 < n:
            g += (1.0 - h[i1])
        i2 = m - shift
        if 0 <= i2 < n:
            g -= h[i2]
        grad[m] = g
    return c5, grad * dx

def repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)
    for _ in range(200):
        diff = (target - s) / n
        h += diff
        h = np.clip(h, 0.0, 1.0)
        s = np.sum(h)
        if abs(s - target) < 1e-12:
            break
    return h

def slsqp_optimize(h0, n, maxiter=300):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq',
             'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]

    def obj(x):
        return c5_obj_grad(x)[0]
    def objg(x):
        return c5_obj_grad(x)[1]

    try:
        res = minimize(obj, h0, jac=objg, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        hx = repair(res.x, n)
        if compute_c5(hx) < compute_c5(h0):
            return hx
    except Exception:
        pass
    return h0

def subgradient_refine(h, n, t_end, max_iter=80000):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h = h.copy()
    lr = 0.003
    for it in range(max_iter):
        if time.time() > t_end:
            break
        val, grad = c5_obj_grad(h)
        h_new = h - lr * grad
        h_new = repair(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            lr = min(lr * 1.03, 0.08)
        else:
            lr *= 0.97
            h = h_new
    return best_h, best_c5

def make_binary_start(n, rng):
    target = n // 2
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:target]] = 1.0
    if n % 2 == 1:
        h[idx[target]] = 0.5
    return repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    t_end_slsqp = t0 + budget_s * 0.50
    t_end_sub = t0 + budget_s * 0.75

    n = len(h0)

    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP from multiple starts
    starts = [best_h.copy()]
    for _ in range(8):
        s = best_h + rng.normal(0, 0.12, n)
        starts.append(repair(s, n))
    starts.append(make_binary_start(n, rng))
    starts.append(np.full(n, 0.5))

    for hs in starts:
        if time.time() > t_end_slsqp:

        ho = slsqp_optimize(hs, n, maxiter=250)
        c5o = compute_c5(ho)
        if c5o < best_c5:

            best_h = ho.copy()

    # Second round SLSQP (finer)

        hs = best_h + rng.normal(0, 0.05, n)

            best_c5 = c5_o
            best_h = ho.copy()

    if time.time() < t0 + budget_s - 10:

        ho = slsqp_optimize(best_h, n_fin, maxiter=300)

            best_h = ho

    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, len(best_h))
