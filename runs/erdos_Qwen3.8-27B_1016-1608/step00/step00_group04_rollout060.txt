import numpy as np
import time
from scipy.optimize import minimize

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h.astype(float), 1.0 - h.astype(float), mode='full')
    return float(np.max(corr) * dx)

def repair_sum(h, n):
    h = np.clip(h.astype(float), 0, 1)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-10:
        return h
    # Adjust
    if diff > 0:
        order = np.argsort(-(1 - h))
    else:
        diff = -diff
        order = np.argsort(h)
    for idx in order:
        if diff > 0:
            add = min(diff, 1 - h[idx])
            h[idx] += add
            diff -= add
        else:
            sub = min(diff, h[idx])
            h[idx] -= sub
            diff -= sub
        if diff < 1e-12:
            break
    return h

def swap_local_search(h, n, time_limit, rng):
    h = h.copy()
    h[h >= 0.5] = 1.0
    h[h < 0.5] = 0.0
    cur = evaluate_c5(h)
    best_h = h.copy()
    best = cur
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    start = time.time()
    stagnation = 0
    max_stagnation = 3000
    
    while time.time() - start < time_limit:
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        new = evaluate_c5(h)
        if new < cur - 1e-15:
            cur = new
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            stagnation = 0
            if new < best:
                best = new
                best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
            stagnation += 1
            if stagnation > max_stagnation:
                # random restart perturbation
                n_swap = 3 + rng.integers(0, 4)
                for _ in range(n_swap):
                    i2 = ones[rng.integers(len(ones))]
                    j2 = zeros[rng.integers(len(zeros))]
                    h[i2], h[j2] = 0.0, 1.0
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                cur = evaluate_c5(h)
                stagnation = 0
    
    return best_h, best

def continuous_optimize(h, n, time_limit, rng):
    """Optimize non-binary h using projected subgradient."""
    h = h.astype(float).copy()
    cur = evaluate_c5(h)
    best_h = h.copy()
    best = cur
    dx = 2.0 / n
    lr = 0.02
    start = time.time()
    
    while time.time() - start < time_limit:
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        shift = max_idx - (n - 1)
        
        # Subgradient of corr[max_idx] w.r.t. h
        grad = np.zeros(n)
        if shift >= 0:
            for i in range(n - shift):
                grad[i + shift] += (1 - h[i]) * dx
                grad[i] -= h[i + shift] * dx
        else:
            s = -shift
            for i in range(s, n):
                grad[i] += (1 - h[i - s]) * dx
                grad[i - s] -= h[i] * dx
        
        h_new = h - lr * grad
        h_new = repair_sum(h_new, n)
        new = evaluate_c5(h_new)
        
        if new < cur:
            h = h_new
            cur = new
            if new < best:
                best = new
                best_h = h_new.copy()
            lr = min(lr * 1.05, 0.1)
        else:
            lr *= 0.85
    
    return best_h, best

import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h.astype(np.float64), 1.0 - h.astype(np.float64), mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        idx = np.argsort(1.0 - h)
    else:
        diff = -diff
        idx = np.argsort(h)
    for i in idx:
        if diff > 0:
            a = min(diff, 1.0 - h[i])
            h[i] += a
            diff -= a
        else:
            a = min(diff, h[i])
            h[i] -= a
            diff -= a
        if diff < 1e-12:
            break
    return h

def binarize(h):
    h2 = (h >= 0.5).astype(np.float64)
    return repair(h2, len(h2))

def local_search(h, n, t_limit, rng):

        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h[i], h[j] = 0.0, 1.0

            ones = np.where(h == 1)[0]

        else:

import numpy as np
import time


def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    hf = h.astype(np.float64)
    corr = np.correlate(hf, 1.0 - hf, mode='full')
    return float(np.max(corr) * dx)


def repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        idx = np.argsort(1.0 - h)
    else:
        diff = -diff
        idx = np.argsort(h)
    for i in idx:
        if diff > 0:
            a = min(diff, 1.0 - h[i])
            h[i] += a
            diff -= a
        else:
            a = min(diff, h[i])
            h[i] -= a
            diff -= a
        if diff < 1e-12:
            break
    return h


def binarize(h):
    h2 = (h >= 0.5).astype(np.float64)
    n = len(h2)
    target = n // 2
    ss = int(h2.sum())
    if ss > target:
        oi = np.where(h2 == 1)[0]
        for _ in range(ss - target):
            h2[oi[0]] = 0.0
            oi = np.where(h2 == 1)[0]
    elif ss < target:
        zi = np.where(h2 == 0)[0]
        for _ in range(target - ss):
            h2[zi[0]] = 1.0
            zi = np.where(h2 == 0)[0]
    return h2


def local_search(h, n, t_limit, rng):
    h = binarize(h)
    cur = evaluate_c5(h)
    best_h = h.copy()
    best = cur
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    t0 = time.time()
    stall = 0
    MAX_STALL = 5000

    while time.time() - t0 < t_limit:
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        nv = evaluate_c5(h)
        if nv < cur - 1e-15:
            cur = nv
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            stall = 0
            if nv < best:
                best = nv
                best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
            stall += 1
            if stall >= MAX_STALL:
                n_pert = rng.integers(3, 8)
                for _ in range(n_pert):
                    k = ones[rng.integers(len(ones))]
                    m = zeros[rng.integers(len(zeros))]
                    h[k], h[m] = 0.0, 1.0
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                cur = evaluate_c5(h)
                stall = 0
    return best_h, best


def cont_opt(h, n, t_limit, rng):
    h = h.astype(np.float64).copy()
    cur = evaluate_c5(h)
    best_h = h.copy()
    best = cur
    dx = 2.0 / n
    lr = 0.012
    t0 = time.time()

    while time.time() - t0 < t_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        mi = np.argmax(corr)
        s = mi - (n - 1)
        g = np.zeros(n)
        if s >= 0:
            g[s:n] += (1.0 - h[0:n - s]) * dx
            g[0:n - s] -= h[s:n] * dx
        else:
            ss = -s
            g[0:n - ss] += (1.0 - h[ss:n]) * dx
            g[ss:n] -= h[0:n - ss] * dx
        h2 = np.clip(h - lr * g, 0.0, 1.0)
        h2 = repair(h2, n)
        nv = evaluate_c5(h2)
        if nv < cur - 1e-15:
            h = h2
            cur = nv
            lr = min(lr * 1.04, 0.06)
            if nv < best:
                best = nv
                best_h = h2.copy()
        else:
            lr *= 0.92
    return best_h, best


def interp_to(h, n, n2):
    x1 = (np.arange(n) + 0.5) * (2.0 / n)
    x2 = (np.arange(n2) + 0.5) * (2.0 / n2)
    return np.interp(x2, x1, h)


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.time()

    def elapsed():
        return time.time() - t_start

    def remaining():
        return budget_s - elapsed()

    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, dtype=np.float64)
    else:
        h_init = np.zeros(42)
        h_init[rng.choice(42, 21, replace=False)] = 1.0

    n0 = len(h_init)
    best_h = h_init.copy()
    best_s = evaluate_c5(best_h)

    # Phase 1: Binary local search from initial
    rem = remaining()
    if rem > 30:
        bh, bs = local_search(h_init, n0, min(rem * 0.22, 200), rng)
        if bs < best_s:
            best_h, best_s = bh, bs

    # Phase 2: Continuous refinement
    rem = remaining()
    if rem > 15:
        bh, bs = cont_opt(best_h, n0, min(rem * 0.08, 50), rng)
        if bs < best_s:
            best_h, best_s = bh, bs

    # Phase 3: Re-binarize and search again
    rem = remaining()
    if rem > 20:
        bh, bs = local_search(best_h, n0, min(rem * 0.15, 120), rng)
        if bs < best_s:
            best_h, best_s = bh, bs

    # Phase 4: Multi-resolution search (even n for binary)
    n_try_list = [28, 56, 64, 70, 80, 90, 100]
    for n2 in n_try_list:
        if n2 % 2 != 0:
            continue
        rem = remaining()
        if rem < 25:
            break
        al = interp_to(best_h, n0, n2)
        al_b = binarize(al)
        tl = min(rem * 0.10, 50)
        bh, bs = local_search(al_b, n2, tl, rng)
        if bs < best_s:
            best_h, best_s = bh, bs
            # quick continuous polish on the new best
            rem = remaining()
            if rem > 15:
                bh2, bs2 = cont_opt(best_h, n2, min(rem * 0.05, 25), rng)
                if bs2 < best_s:
                    best_h, best_s = bh2, bs2

    # Phase 5: Random restarts at best resolution
    nb = len(best_h)
    rem = remaining()
    if rem > 40 and nb % 2 == 0:
        for _ in range(4):
            rem = remaining()
            if rem < 15:
                break
            # Perturb best solution
            hp = best_h.astype(np.float64).copy()
            ones = np.where(hp == 1)[0]
            zeros = np.where(hp == 0)[0]
            npert = rng.integers(3, 6)
            for _ in range(npert):
                k = ones[rng.integers(len(ones))]
                m = zeros[rng.integers(len(zeros))]
                hp[k], hp[m] = 0.0, 1.0
            bh, bs = local_search(hp, nb, min(rem * 0.15, 60), rng)
            if bs < best_s:
                best_h, best_s = bh, bs

    # Phase 6: Final continuous + binary polish
    rem = remaining()
    if rem > 10:
        bh, bs = cont_opt(best_h, len(best_h), min(rem * 0.25, 100), rng)
        if bs < best_s:
            best_h, best_s = bh, bs

    rem = remaining()
    if rem > 10:
        bh, bs = local_search(best_h, len(best_h), min(rem * 0.35, 150), rng)
        if bs < best_s:
            best_h, best_s = bh, bs

    # Final repair and evaluation
    best_h = repair(best_h, len(best_h))
    best_s = evaluate_c5(best_h)
    return (best_h, best_s, len(best_h))
