import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        w = 1.0 - h
    else:
        w = h
    tot = w.sum()
    if tot < 1e-14:
        return h
    s = min(1.0, abs(diff) / tot)
    h = h + (s * w if diff > 0 else -s * w)
    return np.clip(h, 0.0, 1.0)

def make_initial(n, rng):
    h = np.where(rng.rand(n) < 0.5, 1.0, 0.0)
    return h

def slsqp_optimize(h, dx, target_sum, max_iter=300):
    n = len(h)
    def obj(x):
        return compute_c5(x, dx)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bnds = tuple((0.0, 1.0) for _ in range(n))
    try:
        res = minimize(obj, h.copy(), method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-12})
        if res.fun < obj(h):
            return np.clip(res.x, 0, 1)
    except Exception:
        pass
    return h
