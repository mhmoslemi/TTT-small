import numpy as np
import time

def c5_value(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def fix_sum(h, target):
    h = np.array(h, dtype=float)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-8
    else:
        mask = h > 1e-8
    if not mask.any():
        return np.clip(h, 0, 1)
    h = h.copy()
    h[mask] += diff / mask.sum()
    return np.clip(h, 0, 1)

def sa_optimize(h, n, target, rng, t_end):
    cur_c5 = c5_value(h)
    best_c5 = cur_c5
    best_h = h.copy()
    T = 0.0015
    stag = 0
    no_imp = 0

    while time.time() < t_end:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        if h[i] >= h[j]:
            src, dst = i, j
        else:
            src, dst = j, i
        mx = min(h[src], 1.0 - h[dst])
        if mx < 1e-8:
            continue
        amt = rng.uniform(0.005, min(0.25, mx))
        h[src] -= amt
        h[dst] += amt
        new_c5 = c5_value(h)

        if new_c5 <= cur_c5:
            cur_c5 = new_c5
            no_imp = 0
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        elif np.log(max(rng.random(), 1e-300)) > -(new_c5 - cur_c5) / T:
            cur_c5 = new_c5
        else:
            h[src] += amt
            h[dst] -= amt

        stag += 1
        if stag > 150:
            T *= 0.93
            stag = 0
        if T < 1e-10:
            T = 1e-10

        no_imp += 1
        if no_imp > 5000:
            for _ in range(15):
                a = rng.randint(n)
                b = rng.randint(n)
                if a != b and h[a] > h[b]:
                    t_ = min(0.3 * h[a], 0.3 * (1.0 - h[b]))
                    if t_ > 1e-6:
                        h[a] -= t_
                        h[b] += t_
            cur_c5 = c5_value(h)
            T = 0.001
            no_imp = 0

    return best_h, best_c5

def local_descent(h, n, target, rng, t_end):
    cur = c5_value(h)
    while time.time() < t_end:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:

        if h[i] >= h[j]:
            src, dst = i, j
        else:

    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).flatten()
    else:

    # Start 1: parent

    h1 = fix_sum(h1, target)
    candidates.append(h1)

    # Start 2: random 0/1
    h2 = np.zeros(n)
    positions = rng.choice(n, n // 2, replace=False)
    h2[positions] = 1.0
    candidates.append(h2)

    # Start 3: structured (block pattern)
    h3 = np.zeros(n)
    h3[::2] = 1.0
    s = h3.sum()
    if abs(s - target) > 0.5:
        extra = int(target) - int(s)
        if extra > 0:
            for k in range(int(s), n, 3):
                if h3[k] < 0.5 and extra > 0:
                    h3[k] = 1.0
                    extra -= 1
                elif extra <= 0:
                    break
            s = h3.sum()
            if s > target:
                for k in range(n - 1, int(target), -1):
                    if h3[k] > 0.5:
                        h3[k] = 0.0
                        break

    if best_h is not None:
        best_h = np.clip(best_h, 0, 1)
        diff = target - best_h.sum()
        if abs(diff) > 1e-9:
            mask = (best_h > 1e-8) & (best_h < 1.0 - 1e-8)
            if mask.any():
                best_h[mask] += diff / mask.sum()
                best_h = np.clip(best_h, 0, 1)

    if best_h is None:
        best_h = np.ones(n) * 0.5
        best_c5 = c5_value(best_h)

    return best_h, best_c5, n
