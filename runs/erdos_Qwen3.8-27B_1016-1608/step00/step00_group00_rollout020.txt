        x_new = np.linspace(0, 2, n2)
        h2 = np.interp(x_new, x_old, best_h)
        target2 = n2 / 2.0
        h2 = fix_sum(h2, target2)
        c5_2 = compute_c5(h2)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
        for alpha in [100.0, 300.0, 700.0]:
            if tl() < 80:
                break
            new_h = do_slsqp(h2, n2, target2, alpha, maxiter=500)
            new_c5 = compute_c5(new_h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()
                h2 = new_h.copy()

    # Phase 5: If time remains, try another upscale
    if tl() > 200 and len(best_h) < 500:
        n3 = 2 * len(best_h)
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n3)
        h3 = np.interp(x_new, x_old, best_h)
        target3 = n3 / 2.0
        h3 = fix_sum(h3, target3)
        for alpha in [200.0, 600.0]:
            if tl() < 80:
                break
            new_h = do_slsqp(h3, n3, target3, alpha, maxiter=500)
            new_c5 = compute_c5(new_h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()
                h3 = new_h.copy()

    # Final: ensure exact constraint satisfaction
    best_h = fix_sum(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
