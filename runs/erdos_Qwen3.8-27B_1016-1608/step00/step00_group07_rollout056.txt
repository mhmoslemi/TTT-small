    """Compute change in correlation vector for swap i:1->0, j:0->1"""
    d_range = np.arange(-(n-1), n)  # lags
    delta = np.zeros(2*n - 1)

    delta[valid4] -= h[j - d_range[valid4]]
    
    return delta
