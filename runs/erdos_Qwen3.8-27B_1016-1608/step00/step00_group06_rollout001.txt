        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-15:
            best_c5 = c5_try
            best_h = h_try.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
        n_init = len(h_init)
    else:
        n_init = 100
        h_init = np.zeros(n_init)
        h_init[:n_init // 2] = 1.0

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Main optimization at multiple resolutions
    n_options = sorted(set([n_init, n_init + 1 if n_init % 2 else n_init - 1, 100, 128, 160, 200]))
    n_options = [n for n in n_options if 50 <= n <= 300]

    for n in n_options:
        if get_elapsed(t0) > budget_s * 0.55:
            break
        if n == n_init:
            h_start = h_init.copy()
        else:
            x_old = np.linspace(0.0, 2.0, n_init)
            x_new = np.linspace(0.0, 2.0, n)
            h_start = np.interp(x_new, x_old, h_init)
        h_start = project(h_start, n)

        h_opt, c5_opt = optimize_block(h_start, n, t0, budget_s)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Local search and swap search on best solution
    n = len(best_h)
    if get_elapsed(t0) < budget_s * 0.7:
        best_h, best_c5 = local_search(best_h, n, t0, budget_s, n_iter=300)

    if get_elapsed(t0) < budget_s * 0.75:
        best_h, best_c5 = swap_search(best_h, n, t0, budget_s, n_iter=100)

    # Final polish with high-alpha (near-exact max)
    if get_elapsed(t0) < budget_s * 0.88:
        for alpha in [200.0, 500.0]:
            if get_elapsed(t0) > budget_s * 0.94:
                break
            res = minimize(obj_func, best_h, args=(n, alpha, 5.0),
                         method='L-BFGS-B',
                         bounds=[(0.0, 1.0)] * n,
                         options={'maxiter': 400, 'ftol': 1e-15})
            h_pol = project(res.x, n)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Another round of local search after polish
    if get_elapsed(t0) < budget_s * 0.95:
        best_h, best_c5 = local_search(best_h, n, t0, budget_s, n_iter=100)

    return best_h, best_c5, len(best_h)
