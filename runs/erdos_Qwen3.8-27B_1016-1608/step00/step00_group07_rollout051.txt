def compute_c5(h):

def project_feasible(h, target_sum):
    h = np.clip(h, 0, 1)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0, 1)
    return h

def smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full") * dx
    amax = corr.max()
    exponents = alpha * (corr - amax)
    w = np.exp(exponents)
    wsum = w.sum()
    if wsum < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= wsum
    grad = np.zeros(n)
    shifts = np.arange(len(corr)) - (n - 1)
    for idx in range(len(corr)):
        wk = w[idx]
        if wk < 1e-13:
            continue
        sa = abs(shifts[idx])
        if sa == 0:
            grad += wk * (1.0 - 2.0 * h)
        else:
            grad += wk
            grad[:n-sa] -= wk * h[sa:]
            grad[sa:] -= wk * h[:n-sa]
    grad *= dx
    return amax, grad

    n = len(h)
    target = n / 2.0
    h = project_feasible(h, target)

    best_h = h.copy()
    for it in range(n_iters):

This is related to the "autocorrelation" of the binary sequence. Minimizing the maximum "off-peak" autocorrelation is a well-studied problem (related to Barker sequences, Golay codes, etc.).
