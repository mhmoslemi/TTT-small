import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
        else:
            mask = h > 1e-13
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_descend(h, n, max_sweeps=400, grid_size=21, time_limit_cb=None):
    h = h.copy()
    for sweep in range(max_sweeps):
        if time_limit_cb is not None and time_limit_cb() < 8:
            break
        any_improve = False
        for j in range(n):
            best_val = h[j]
            best_obj = compute_c5(h)
            for frac in np.linspace(0.0, 1.0, grid_size):
                if abs(frac - best_val) < 1e-10:
                    continue
                h_try = h.copy()
                h_try[j] = frac
                h_try = project_feasible(h_try, n)
                obj = compute_c5(h_try)
                if obj < best_obj - 1e-13:
                    best_obj = obj
                    best_val = frac
                    h = h_try
                    any_improve = True
        if not any_improve:
            break
    return h
