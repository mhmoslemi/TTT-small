def evaluate_c5(h):
    """Compute C5 = max of correlation * dx"""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def smooth_objective(h, eps, target_sum):
    """Smoothed max with penalty for sum constraint"""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    vals = c * dx
    m = np.max(vals)
    sm = m + eps * np.log(np.sum(np.exp((vals - m) / eps)))
    penalty = 100.0 * (np.sum(h) - target_sum) ** 2
    return sm + penalty

def smooth_grad(h, eps, target_sum):

    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    vals = c * dx
    m = np.max(vals)
    p = np.exp((vals - m) / eps)
    p /= np.sum(p)

    # f_s = sum_i h[i]*(1-h[i+s]) for valid i

```python
def project_to_feasible(h, n, target_sum):
    h = np.clip(h, 0, 1)
    # Find λ such that sum(clip(h - λ, 0, 1)) = target_sum
    lo, hi = -1, 1  # range for λ
    for _ in range(60):
        mid = (lo + hi) / 2
        h_proj = np.clip(h - mid, 0, 1)
        s = np.sum(h_proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2, 0, 1)
    # Fine-tune: adjust one element to get exact sum
    diff = target_sum - np.sum(h_proj)
    if abs(diff) > 1e-15:
        # Add to the element with most room
        if diff > 0:
            idx = np.argmax(1 - h_proj)
            h_proj[idx] = min(1, h_proj[idx] + diff)
        else:
            idx = np.argmin(h_proj)
            h_proj[idx] = max(0, h_proj[idx] + diff)
    return h_proj
```

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    h = np.clip(h - lam, 0.0, 1.0)
    # Exact fix
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argmax(1.0 - h)
            room = 1.0 - h[idx]
            h[idx] += min(diff, room)
        else:
            idx = np.argmin(h)
            h[idx] += max(diff, -h[idx])
    return h

def smooth_obj_and_grad(h, eps, target_sum):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    vals = c * dx
    m = np.max(vals)
    sh = (vals - m) / eps
    exp_sh = np.exp(sh - np.max(sh))
    p = exp_sh / np.sum(exp_sh)
    sm_val = m + eps * (np.log(np.sum(exp_sh)) + np.max(sh))
    
    # Gradient
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            # valid: j in [0, n-1-s]
            grad[:n-s] += pk
            grad[:n-s] -= pk * h[s:]
            # h[j-s] for j in [s, n-1]
            if s > 0:
                grad[s:] -= pk * h[:n-s]
        else:
            ss = -s
            grad[ss:] += pk
            grad[ss:] -= pk * h[:n-ss]
            if ss > 0:
                grad[:n-ss] -= pk * h[ss:]
    
    # Penalty for sum
    penalty = 50.0 * (np.sum(h) - target_sum)
    sm_val += 50.0 * (np.sum(h) - target_sum) ** 2
    grad += 2.0 * penalty
    
    return sm_val, grad

def optimize_lbfgsb(h0, n, target_sum, max_time=60.0):
    h = h0.copy()
    h = project_feasible(h, target_sum)
    eps = 0.05
    t0 = time.time()
    for round_i in range(8):
        if time.time() - t0 > max_time:
            break
        res = minimize(
            fun=lambda x: smooth_obj_and_grad(x, eps, target_sum)[0],
            x0=h,
            jac=lambda x: smooth_obj_and_grad(x, eps, target_sum)[1],
            method='L-BFGS-B',
            bounds=[(0.0, 1.0)] * n,
            options={'maxiter': 100, 'ftol': 1e-12}
        )
        h = project_feasible(res.x, target_sum)
        eps *= 0.5
    return h

def coordinate_descent(h, n, target_sum, max_iter=30, n_grid=8):
    for it in range(max_iter):
        improved = False
        c5_cur = evaluate_c5(h)
        for i in range(n):
            old = h[i]
            best_val = old
            best_c5 = c5_cur
            grid = np.linspace(0.0, 1.0, n_grid)
            for gv in grid:
                h[i] = gv
                h = project_feasible(h, target_sum)
                c5_new = evaluate_c5(h)
                if c5_new < best_c5 - 1e-10:
                    best_c5 = c5_new
                    best_val = h[i]
                    # store best h
            # restore best
            h[i] = best_val
            h = project_feasible(h, target_sum)
            if best_c5 < c5_cur - 1e-10:
                improved = True
                c5_cur = best_c5
        if not improved:
            break
    return h
