import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_max_c5(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    v = corr * dx
    m = np.max(v)
    e = np.exp(alpha * (v - m))
    return m + np.log(np.sum(e)) / alpha

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    for _ in range(2000):
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def obj_lbm(x, alpha, lam, n):
    return smooth_max_c5(x, alpha) + lam * (np.sum(x) - n / 2.0) ** 2

def obj_sm(x, alpha):
    return smooth_max_c5(x, alpha)

def integral_eq(x, n):
    return np.sum(x) - n / 2.0

def resample_h(h, n_new):

        return h.copy()
    x_old = np.arange(n_old, dtype=float)
    x_new = np.linspace(0, n_old - 1, n_new)
    return np.interp(x_new, x_old, h)
