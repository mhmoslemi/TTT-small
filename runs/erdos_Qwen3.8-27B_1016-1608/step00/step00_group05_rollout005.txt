                                 constraints=[con_try],
                                 options={'maxiter': 300, 'ftol': 1e-12})
                    c_r = compute_c5(r.x, dx_try)
                    if c_r < c_new:
                        h_new = project_sum(r.x, target_try)
                        c_new = compute_c5(h_new, dx_try)
                except Exception:
                    pass
            # Pair search on new n
            if remaining() > 40:
                h_new, c_new = pair_search(h_new, n_try, target_try, dx_try,
                                           min(remaining() - 30, 60))
            if c_new < best_c5:
                best_h = h_new
                best_c5 = c_new
                n = n_try

    # Final verification
    best_h = np.clip(best_h, 0.0, 1.0)
    s_val = np.sum(best_h)
    if abs(s_val - n / 2.0) > 1e-8:
        best_h = best_h * (n / 2.0) / s_val
        best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h, 2.0 / len(best_h))

    if 'evaluate_erdos_solution' in globals():
        try:
            verified = evaluate_erdos_solution(best_h)
            if verified is not None and verified < best_c5:
                pass  # keep our computed value
        except Exception:
            pass

    return best_h, best_c5, len(best_h)
