    """Compute gradient of weighted sum of correlations."""
    n = len(h)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        w = weights[m]
        if abs(w) < 1e-14:
            continue
        L = m - (n - 1)
        if L >= 0:
            # C_L = sum_{j=L}^{n-1} h[j]*(1-h[j-L])
            # d/dh[t]: (1-h[t-L]) for t in [L, n-1]
            if L < n:
                grad[L:n] += w * (1 - h[:n-L])
            # -h[t+L] for t in [0, n-1-L]
            if n-1-L >= 0:
                grad[0:n-L] += w * (-h[L:n])
        else:
            p = -L
            # C_{-p} = sum_{j=0}^{n-1-p} h[j]*(1-h[j+p])
            # d/dh[t]: (1-h[t+p]) for t in [0, n-1-p]
            if n-1-p >= 0:
                grad[0:n-p] += w * (1 - h[p:n])
            # -h[t-p] for t in [p, n-1]
            if p < n:
                grad[p:n] += w * (-h[0:n-p])
    return grad

import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h, 0, 1)
    for _ in range(100):
        diff = h.sum() - target
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-10) & (h < 1 - 1e-10)
        if mask.sum() == 0:
            break
        h[mask] -= diff / mask.sum()
        np.clip(h, 0, 1, out=h)
    return h

def compute_gradient(h, weights, dx):
    n = len(h)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        w = weights[m] * dx
        if abs(w) < 1e-14:
            continue
        L = m - (n - 1)
        if L >= 0:
            k = min(L, n - 1)
            if k < n:
                grad[k:n] += w * (1.0 - h[:n - k])
            n2 = n - k
            if n2 > 0:
                grad[0:n2] += w * (-h[k:n])
        else:
            p = -L
            np_ = n - p
            if np_ > 0:
                grad[0:np_] += w * (1.0 - h[p:n])
            if p < n:
                grad[p:n] += w * (-h[0:np_])
    return grad

def optimize_from_start(h_init, target_n, rng, deadline, max_iter=2000):
    n = target_n
    x_old = np.linspace(0.0, 2.0, len(h_init))
    x_new = np.linspace(0.0, 2.0, n)
    h = np.interp(x_new, x_old, h_init)
    h = project_sum(h, n / 2.0)
    dx = 2.0 / n
    best_local = evaluate_c5(h)
    best_h = h.copy()
    beta = 30.0
    lr = 0.005
    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        cmax = corr.max()
        shifted = beta * (corr - cmax)
        shifted -= shifted.max()
        weights = np.exp(shifted)
        sw = weights.sum()
        if sw < 1e-300:
            weights = np.full_like(corr, 1.0 / len(corr))
        else:
            weights /= sw
        grad = compute_gradient(h, weights, dx)
        grad -= grad.mean()
        h_new = h - lr * grad
        h_new = project_sum(h_new, n / 2.0)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_local:
            best_local = c5_new
            best_h = h_new.copy()
            if best_local < 0.37:
                beta *= 1.5
        else:
            if it % 100 == 0 and it > 0:
                lr *= 0.85
                beta = max(10.0, beta * 1.3)
            h = h_new
        if it % 200 == 199 and it < max_iter - 100:
            perturb = rng.standard_normal(n) * 0.03
            h_p = h + perturb
            h_p = project_sum(h_p, n / 2.0)
            if evaluate_c5(h_p) < best_local + 0.005:
                h = h_p
    return best_h, best_local

    rng = np.random.default_rng(seed)
    t0 = time.time()
    deadline = t0 + budget_s * 0.92

    for sp_idx, sp in enumerate(start_points):

    best_h = project_sum(best_h, len(best_h) / 2.0)
    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, len(best_h)
