def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-14
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_for_shift(s, h, n):
    m = np.arange(n)
    grad = np.zeros(n)
    mask1 = (m + s >= 0) & (m + s <= n - 1)
    grad[mask1] += (1.0 - h[m[mask1] + s])
    mask2 = (m - s >= 0) & (m - s <= n-1)
    grad[mask2] -= h[m[mask2] - s]
    return grad

def project(h, n):
    """Project onto {0<=h<=1, sum(h)=n/2}"""
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        d = target - h.sum()
        if abs(d) < 1e-12:
            break
        if d > 0:
            m = h < 1.0
            c = m.sum()
            if c == 0: break
            h[m] += d / c
        else:
            m = h > 0.0
            c = m.sum()
            if c == 0: break
            h[m] += d / c
        np.clip(h, 0.0, 1.0, out=h)
    return h

def grad_shift(s, h, n):
    """Subgradient contribution for shift s"""
    m = np.arange(n)
    g = np.zeros(n)
    p1 = (m + s >= 0) & (m + s <= n-1)
    g[p1] += 1.0 - h[m[p1] + s]
    p2 = (m - s >= 0) & (m - s <= n-1)
    g[p2] -= h[m[p2] - s]
    return g

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def project_feasible(h, n):
    target = 0.5 * n
    h = np.clip(h, 0.0, 1.0)
    for _ in range(30):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            m = h < 1.0
            c = int(m.sum())
            if c == 0:
                break
            h[m] += d / c
        else:
            m = h > 0.0
            c = int(m.sum())
            if c == 0:
                break
            h[m] += d / c
        np.clip(h, 0.0, 1.0, out=h)
    return h

def subgrad_optimize(h0, n, iters, lr0, topk, seed):
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    mom = np.zeros(n)
    beta = 0.8
    
    corr = np.correlate(h, 1.0-h, mode="full")
