import time

def c5_val(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))

def repair(h):
    n = len(h)
    target = 0.5 * n
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s > 1e-15:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-13:
        if d > 0:
            idx = np.argsort(-h)
        else:
            idx = np.argsort(h)
        for i in idx:
            if d > 0:
                add = min(d, 1.0 - h[i])
                h[i] += add
                d -= add
            else:
                sub = min(-d, h[i])
                h[i] -= sub
                d += sub
            if abs(d) < 1e-15:
                break
    return h

def pen_obj(x, n):
    val = c5_val(x)
    p1 = (x.sum() - 0.5 * n) ** 2
    p2 = np.maximum(0.0, x).sum() ** 2 + np.maximum(0.0, -x).sum() ** 2
    # Better penalties
    p2 = np.sum(np.clip(x, None, 0.0)**2) + np.sum(np.clip(1.0-x, None, 0.0)**2)
    return val + 10.0 * p1 + 10.0 * p2

            for j in zeros:
                h[i] = 0; h[j] = 1
                c = c5_val(h)
                if c < best:
                    best = c
                    found = True
                    break
                h[i] = 1; h[j] = 0
            if found:
                break
        if not found:
