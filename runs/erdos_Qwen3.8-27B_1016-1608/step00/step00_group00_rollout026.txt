def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def fix_sum(h, target):
    """Fix the sum of h to target, maintaining [0,1] bounds."""
    h = h.copy()
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-6) & (h < 1.0 - 1e-6)
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
    h = np.clip(h, 0, 1)
    return h

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def fix_sum(h, target):
    h = h.copy()
    for _ in range(20):
        d = target - h.sum()
        if abs(d) < 1e-12:
            break
        m = (h > 1e-5) & (h < 1.0 - 1e-5)
        if not m.any():
            break
        h[m] += d / m.sum()
    return np.clip(h, 0.0, 1.0)

def sa_search(h_init, rng, time_limit):
    n = len(h_init)
    t0 = time.time()
    tgt = n / 2.0
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    cur_h = h_init.copy()
    cur_c5 = best_c5
    stuck = 0
    worst_lag = 0

    while time.time() - t0 < time_limit:
        elapsed = time.time() - t0
        temp = 0.01 * np.exp(-4.0 * elapsed / max(time_limit, 1e-6))
        if temp < 1e-6:
            temp = 1e-6

        h_t = cur_h.copy()
        mv = rng.integers(0, 5)

        if mv == 0:
            i, j = rng.integers(0, n, 2)
            if i == j:
                continue
            d = rng.uniform(-0.5, 0.5)
            h_t[i] += d
            h_t[j] -= d
        elif mv == 1:
            i, j = rng.integers(0, n, 2)
            if i == j:
                continue
            h_t[i], h_t[j] = h_t[j], h_t[i]
        elif mv == 2:
            k = int(rng.integers(1, 6))
            idx = rng.integers(0, n, k)
            h_t[idx] += rng.normal(0, 0.15, k)
            h_t = fix_sum(h_t, tgt)
        elif mv == 3:
            # Targeted move on worst lag
            c = int(worst_lag)
            if 0 < c < n:
                idxs = np.arange(0, n - c)
                prods = h_t[idxs] * (1.0 - h_t[idxs + c])
                pmax = int(np.argmax(prods))
                h_t[pmax] -= 0.15
                h_t[pmax + c] += 0.15
            elif -n < c < 0:
                m = -c
                idxs = np.arange(m, n)
                prods = h_t[idxs] * (1.0 - h_t[idxs - m])
                pmax = int(np.argmax(prods))
                h_t[pmax + m] -= 0.15

                h_t[i], h_t[j] = h_t[j], h_t[i]

            h_t[idx] = rng.uniform(0, 1, k)
            h_t = fix_sum(h_t, tgt)

        h_t = np.clip(h_t, 0.0, 1.0)

def upscale(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    h_new = fix_sum(h_new, n_new / 2.0)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    rng = np.random.default_rng(seed)
    t_start = time.time()

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        h0 = np.full(45, 0.5)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: SA on original grid
    alloc1 = budget_s * 0.40
    t_remain = alloc1 - (time.time() - t_start)
    if t_remain > 5:
        h_r, c5_r = sa_search(best_h, rng, t_remain)
        if c5_r < best_c5:
            best_h, best_c5 = h_r, c5_r

    # Phase 2: Try higher resolutions
    alloc2_end = t_start + budget_s * 0.78
    for n_new in [60, 80, 100, 120]:
        if time.time() > alloc2_end:
            break
        t_rem = alloc2_end - time.time() - 10
        if t_rem < 15:
            break
        h_s = upscale(best_h, n_new)
        h_s, c5_s = sa_search(h_s, rng, t_rem)
        if c5_s < best_c5:
            best_h, best_c5 = h_s, c5_s

    # Phase 3: Final polish
    t_rem = t_start + budget_s - 8 - time.time()
    if t_rem > 10:
        best_h, best_c5 = sa_search(best_h, rng, t_rem)

    # Validation
    n_out = len(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)
    tgt = n_out / 2.0
    d = tgt - best_h.sum()
    if abs(d) > 1e-9:
        m = (best_h > 0.001) & (best_h < 0.999)
        if m.any():
            best_h[m] += d / m.sum()
        best_h = np.clip(best_h, 0.0, 1.0)

    c5_final = compute_c5(best_h)
    return best_h, c5_final, n_out
