def coordinate_descent(h, n, rng, start_time, time_limit):
    h = h.copy()
    current = compute_c5(h)
    target = n / 2.0
    improved = True
    while improved and time.time() < time_limit:
        improved = False
        for i in range(n):
            if time.time() > time_limit:
                break
            for j in range(n):
                if i == j:
                    continue
                # Move delta from j to i
                for delta in [0.01, 0.05, 0.1]:

                    h_new[j] -= delta

                        c5_new = compute_c5(h_new)
                        if c5_new < current - 1e-10:
                            h = h_new
                            current = c5_new
                            improved = True
                            break
    return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_penalized(h, n):
    hc = np.clip(h, 0, 1)
    corr = np.correlate(hc, 1.0 - hc, mode='full')
    c5 = np.max(corr) * (2.0 / n)
    sv = abs(np.sum(h) - n * 0.5)
    bv = np.sum(np.maximum(0, h - 1)**2) + np.sum(np.maximum(0, -h)**2)
    return c5 + 50.0 * sv + 50.0 * bv

def two_exchange_search(h, n, rng, start_time, time_limit):
    h = h.copy()
    cur = compute_c5(h)
    deltas = np.array([0.02, 0.05, 0.1, 0.15])
    sweeps = 0
    while time.time() < time_limit and sweeps < 50:
        improved = False
        sweep_start = time.time()
        idx_pairs = list(zip(*np.triu_indices(n, k=1)))
        rng.shuffle(idx_pairs)
        for i, j in idx_pairs:
            if time.time() > time_limit:
                break
            for d in deltas:
                for sign in (1, -1):
                    h_try = h.copy()
                    h_try[i] += sign * d
                    h_try[j] -= sign * d
                    if 0.0 <= h_try[i] <= 1.0 and 0.0 <= h_try[j] <= 1.0:
                        c5_try = compute_c5(h_try)
                        if c5_try < cur - 1e-12:
                            h = h_try
                            cur = c5_try
                            improved = True
                            break
                if improved:
                    break
        sweeps += 1
        if not improved and sweeps > 5:
            break
    return h, cur

def sa_search(h_init, n, rng, start_time, time_limit, temp0=0.04):
    h = project_feasible(h_init, n)
    cur = compute_c5(h)
    best = h.copy()
    best_c5 = cur
    temp = temp0
    while time.time() < time_limit:
        npert = rng.randint(1, min(4, n // 6 + 1))
        idx = rng.choice(n, size=npert, replace=False)
        h_new = h.copy()
        h_new[idx] += rng.uniform(-0.25, 0.25, size=npert)
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        delta = c5_new - cur
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-6)):
            h, cur = h_new, c5_new
            if cur < best_c5:
                best_c5, best = cur, h.copy()
        temp = max(0.0005, temp * 0.998)
    return best, best_c5
