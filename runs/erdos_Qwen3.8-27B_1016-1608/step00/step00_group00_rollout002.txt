import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float((corr * dx).max())

def smooth_obj(h, alpha=100.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = corr.max()
    return cmax + np.log(np.sum(np.exp(alpha * (corr - cmax)))) / alpha

def smooth_obj_grad(h, alpha=100.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = corr.max()
    exp_v = np.exp(alpha * (corr - cmax))
    w = exp_v / exp_v.sum()
    obj = cmax + np.log(exp_v.sum()) / alpha
    
    grad = np.zeros(n)
    for j in range(n):
        s = 0.0
        for m in range(2 * n - 1):
            idx1 = j + m - (n - 1)
            idx2 = j - m + (n - 1)
            if 0 <= idx1 < n:
                s += w[m] * (1.0 - h[idx1])
            if 0 <= idx2 < n:
                s -= w[m] * h[idx2]
        grad[j] = alpha * dx * s
    return obj, grad

def project_to_simplex_box(h, target_sum):
    """Project onto {0<=h<=1, sum(h)=target_sum}"""
    n = len(h)
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float((corr * dx).max())

def smooth_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = corr.max()
    s = np.sum(np.exp(alpha * (corr - cmax)))
    return cmax + np.log(s) / alpha

def smooth_obj_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = corr.max()
    exp_v = np.exp(alpha * (corr - cmax))
    w = exp_v / exp_v.sum()
    obj = cmax + np.log(exp_v.sum()) / alpha
    grad = np.zeros(n)
    for j in range(n):
        s = 0.0
        for m in range(2 * n - 1):
            i1 = j + m - (n - 1)
            i2 = j - m + (n - 1)
            if 0 <= i1 < n:
                s += w[m] * (1.0 - h[i1])
            if 0 <= i2 < n:
                s -= w[m] * h[i2]
        grad[j] = alpha * dx * s
    return obj, grad

def project_box_sum(h, target):
    n = len(h)
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        sv = np.sum(np.clip(h - mid, 0.0, 1.0))
        if sv > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(h - lam, 0.0, 1.0)

    t0 = time.time()
    global initial_h_values

        best_h[:22] = 1.0

    n = len(best_h)
    target = n / 2.0
    best_h = project_box_sum(best_h, target)
    best_c5 = compute_c5(best_h)

    bounds = [(0.0, 1.0)] * n
    con = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: np.ones(n)}]

    # Phase 1: SLSQP with increasing smoothness
    for alpha in [30, 60, 120, 250, 500]:
        if time.time() - t0 > budget_s * 0.45:
            break
        try:
            res = minimize(smooth_obj, best_h.copy(), args=(alpha,),
                           jac=lambda x: smooth_obj_grad(x, alpha)[1],
                           method='SLSQP', bounds=bounds, constraints=con,
                           options={'maxiter': 300, 'ftol': 1e-14})
            h_new = project_box_sum(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new
        except Exception:
            pass

    # Phase 2: Projected gradient with adaptive steps
    alpha_pg = 200.0
    for lr in [0.05, 0.02, 0.01, 0.005, 0.002]:
        if time.time() - t0 > budget_s * 0.65:
