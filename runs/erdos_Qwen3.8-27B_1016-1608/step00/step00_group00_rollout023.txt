```python
    n = len(h)

    m = np.max(overlaps)
```

```python
import numpy as np
import time
from scipy.optimize import minimize

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def compute_c5(h):
    return np.max(compute_overlaps(h))

def lse_abs_max(x, alpha):
    """Smooth max using log-sum-exp."""
    m = np.max(x)
    return m + alpha * np.log(np.sum(np.exp((x - m) / alpha)))

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            free = np.where(h < 1.0)[0]
            if len(free) == 0:
                break
            h[free] = np.minimum(1.0, h[free] + diff / len(free))
        else:
            free = np.where(h > 0.0)[0]
            if len(free) == 0:
                break
            h[free] = np.maximum(0.0, h[free] + diff / len(free))
    # Final exact adjustment
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        # Adjust the element closest to the needed direction
        if diff > 0:
            idx = np.argmin(h)
        else:
            idx = np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0, 1)
    return h

def grad_lse(h, alpha):
    """Numerical gradient of LSE of overlaps."""
    n = len(h)
    dx = 2.0 / n
    eps = 1e-7
    overlaps = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(overlaps)
    exps = np.exp((overlaps - m) / alpha)
    base_lse = m + alpha * np.log(np.sum(exps))
    
    grad = np.zeros(n)
    for j in range(n):
        h_p = h.copy()
        h_p[j] = min(1.0, h[j] + eps)
        h_m = h.copy()
        h_m[j] = max(0.0, h[j] - eps)
        gap = h_p[j] - h_m[j]
        if gap < 1e-14:
            continue
        ov_p = np.correlate(h_p, 1.0 - h_p, mode="full") * dx
        ov_m = np.correlate(h_m, 1.0 - h_m, mode="full") * dx
        mp = np.max(ov_p)
        mm = np.max(ov_m)
        lse_p = mp + alpha * np.log(np.sum(np.exp((ov_p - mp) / alpha)))
        lse_m = mm + alpha * np.log(np.sum(np.exp((ov_m - mm) / alpha)))
        grad[j] = (lse_p - lse_m) / gap
    return grad

def optimize_continuous(h0, target_sum, max_iter=500, alpha_init=10, alpha_final=100):

    h = h0.copy()
    n = len(h)
    best_h = h.copy()
    best_val = compute_c5(h)
    
    for it in range(max_iter):
        alpha = alpha_init + (alpha_final - alpha_init) * it / max(1, max_iter-1)
        lr = 1.0 / (it + 10)
        g = grad_lse(h, alpha)
        h_new = h - lr * g
        h_new = project_feasible(h_new, target_sum)

        # Accept/reject (simple descent)
        if h_new is not h:
            # Check if LSE decreased
            h = h_new
    
    return best_h, best_val

def local_search_binary(h, target_sum, max_iter=10000):

                    h_step[idx] = 0.0
                    diff += 1
```

```python
def apply_swap(h, i, j, overlaps, dx):
    """Apply swap of h[i] and h[j], update overlaps. Returns new overlaps array."""
    n = len(h)
    old_i, old_j = h[i], h[j]
    if old_i == old_j:
        return overlaps
    h[i], h[j] = old_j, old_i
    
    # Overlaps: overlaps[k] = dx * sum_valid a h[a]*(1-h[a + d]) where d = k-(n-1)
    # For a given shift d, the terms involving index i:
    #   a=i: term = h[i]*(1-h[i+d]) if 0<=i+d<n
    #   a+d=i, i.e., a=i-d: term = h[i-d]*(1-h[i]) if 0<=i-d<n
    # Same for j.
    
    # Let's use the convention: overlaps[k] for k=0,...,2n-2
    # d = k - (n-1)
    # overlaps[k] = dx * sum_{a: 0<=a<n, 0<=a+d<n} h[a]*(1-h[a+d])
    
    for idx_i in [i]:
```

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    overlaps = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(overlaps))

def lse_val(x, alpha):
    m = np.max(x)
    return m + alpha * np.log(np.sum(np.exp((x - m) / alpha)))

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-11:
            break
        if diff > 0:
            free = np.where(h < 1.0)[0]
            if len(free) == 0:
                break
            h[free] = np.minimum(1.0, h[free] + diff / len(free))
        else:
            free = np.where(h > 0.0)[0]
            if len(free) == 0:
                break
            h[free] = np.maximum(0.0, h[free] + diff / len(free))
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argmin(h)
        else:
            idx = np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0, 1)
    return h

def grad_lse_numerical(h, alpha):
    n = len(h)
    dx = 2.0 / n
    eps = 1e-7
    grad = np.zeros(n)
    for j in range(n):
        hp = h.copy(); hp[j] = min(1.0, h[j] + eps)
        hm = h.copy(); hm[j] = max(0.0, h[j] - eps)
        gap = hp[j] - hm[j]
        if gap < 1e-14:
            grad[j] = 0.0
            continue
        ov_p = np.correlate(hp, 1.0 - hp, mode="full") * dx
        ov_m = np.correlate(hm, 1.0 - hm, mode="full") * dx
        grad[j] = (lse_val(ov_p, alpha) - lse_val(ov_m, alpha)) / gap
    return grad

def optimize_continuous(h0, target_sum, max_iter=400):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iter):
        alpha = 5.0 + 95.0 * it / max(1, max_iter - 1)
        lr = 4.0 / (it + 20.0)
        g = grad_lse_numerical(h, alpha)
        h_new = project_feasible(h - lr * g, target_sum)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

def round_to_step(h, target_sum):
    n = len(h)
    if n % 2 == 0:
        h_bin = (h > 0.5).astype(float)
        curr = np.sum(h_bin)
        target = int(target_sum)
        order = np.argsort(h)
        if curr > target:
            for idx in order:
                if curr <= target:
                    break
                if h_bin[idx] == 1:
                    h_bin[idx] = 0; curr -= 1
        else:
            for idx in order[::-1]:
                if curr >= target:
                    break
                if h_bin[idx] == 0:
                    h_bin[idx] = 1; curr += 1
        return h_bin
    else:
        h_step = np.zeros(n)
        mid = int(np.argmin(np.abs(h - 0.5)))
        h_step[mid] = 0.5
        others = [i for i in range(n) if i != mid]
        h_step[others] = (h[others] > 0.5).astype(float)
        diff = target_sum - np.sum(h_step)
        if abs(diff) > 1e-10:
            order = np.argsort(h)
            if diff > 0:
                for idx in order:
                    if abs(diff) < 1e-10:
                        break
                    if h_step[idx] < 1.0:
                        h_step[idx] = 1.0; diff -= (1.0 - 0.5) if idx == mid else 1.0
            else:
                for idx in order[::-1]:
                    if abs(diff) < 1e-10:
                        break
                    if h_step[idx] > 0.0:
                        h_step[idx] = 0.0; diff += (0.5) if idx == mid else 1.0
        return h_step

def local_search(h, target_sum, max_swaps=8000):
    n = len(h)
    h = h.copy()
    best_val = compute_c5(h)
    best_h = h.copy()
    for it in range(max_swaps):
        improved = False
        rng_local = np.random.RandomState(it)
        pairs = [(i, j) for i in range(n) for j in range(i + 1, n) if h[i] != h[j]]
        if not pairs:
            break
        rng_local.shuffle(pairs)
        for i, j in pairs:
            h[i], h[j] = h[j], h[i]
            if abs(np.sum(h) - target_sum) < 1e-10:
                val = compute_c5(h)
                if val < best_val - 1e-12:
                    best_val = val
                    best_h = h.copy()
                    improved = True
                    break
            else:
                h[i], h[j] = h[j], h[i]
            h[i], h[j] = h[j], h[i]
        if not improved and it % 200 == 199:
            # Perturb: swap a random pair
            i, j = rng_local.randint(0, n), rng_local.randint(0, n)
            if h[i] != h[j]:
                h[i], h[j] = h[j], h[i]
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = np.inf

    def remaining():
        return budget_s - (time.time() - t0)

    init_h = None
    try:
        if 'initial_h_values' in globals() and initial_h_values is not None:
            init_h = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        pass

    n_list = [44, 45, 46, 50, 55, 60, 80]

    for n in n_list:
        if remaining() < 80:
            break
        target = n / 2.0
        n_restarts = max(2, int(min(15, remaining() / 40)))

        for rs in range(n_restarts):
            if remaining() < 35:
                break
            # Starting point
            if rs == 0 and init_h is not None and len(init_h) == n:
                h0 = init_h.copy()
            elif rs == 0 and n % 2 == 0:
                h0 = np.zeros(n); h0[:n // 2] = 1.0
            elif rs == 0:
                h0 = np.zeros(n)
                h0[:n // 2] = 1.0
                h0[n // 2] = 0.5
            else:
                h0 = rng.binomial(1, 0.5, n).astype(float)
            h0 = project_feasible(h0, target)

            # Continuous optimization
            h_opt, c5_opt = optimize_continuous(h0, target, max_iter=300)

            # Rounding + local search
            h_round = round_to_step(h_opt, target)
            h_ls, c5_ls = local_search(h_round, target, max_swaps=5000)

            for cand, val in [(h_opt, c5_opt), (h_ls, c5_ls)]:
                if val < best_c5:
                    best_c5 = val
                    best_h = cand.copy()

            # Also try starting local search directly from h0 (binary)
            if n % 2 == 0:
                h_bin = (h0 > 0.5).astype(float)
                h_bin2, c5_bin2 = local_search(h_bin, target, max_swaps=3000)
                if c5_bin2 < best_c5:
                    best_c5 = c5_bin2
                    best_h = h_bin2.copy()

    if best_h is None:
        n = 44
        best_h = np.full(n, 0.5)
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
```
