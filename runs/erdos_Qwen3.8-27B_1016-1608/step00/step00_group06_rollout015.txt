    h = np.clip(h, 0.0, 1.0)

    if time.time() > t0_deadline:
        return h_start
    return res.x if res.x is not None else h_start

def interpolate_h(h_src, n_src, n_dst):
    return np.interp(np.linspace(0.0, 2.0, n_dst),
                     np.linspace(0.0, 2.0, n_src), h_src)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    deadline = t_start + budget_s * 0.92

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.asarray(initial_h_values, dtype=np.float64).copy()
    else:
        h_init = np.full(95, 0.5)

    n_base = len(h_init)
    best_h = project_feasible(h_init, n_base)
    best_c5 = compute_c5(best_h, n_base)
    best_n = n_base

    # Stage 1: Optimize at base resolution
    h_opt = do_slsqp(best_h.copy(), best_n, 30.0, 300, deadline)
    h_opt = project_feasible(h_opt, best_n)
    c5_opt = compute_c5(h_opt, best_n)
    if c5_opt < best_c5:
        best_c5, best_h, best_n = c5_opt, h_opt.copy(), best_n

    # Stage 2: Upsample and optimize at 2x
    n_up = n_base * 2
    if n_up <= 400 and time.time() < deadline:
        h_up = interpolate_h(best_h, best_n, n_up)
        h_up = project_feasible(h_up, n_up)
        h_up = do_slsqp(h_up, n_up, 25.0, 250, deadline)
        h_up = project_feasible(h_up, n_up)
        c5_up = compute_c5(h_up, n_up)
        if c5_up < best_c5:
            best_c5, best_h, best_n = c5_up, h_up.copy(), n_up
        else:
            # Downsample back and re-optimize at base
            h_back = interpolate_h(h_up, n_up, n_base)
            h_back = project_feasible(h_back, n_base)
            h_back = do_slsqp(h_back, n_base, 50.0, 300, deadline)
            h_back = project_feasible(h_back, n_base)
            c5_back = compute_c5(h_back, n_base)
            if c5_back < best_c5:
                best_c5, best_h, best_n = c5_back, h_back.copy(), n_base

    # Stage 3: Higher beta refinement at best_n
    if time.time() < deadline:
        h_ref = do_slsqp(best_h.copy(), best_n, 80.0, 300, deadline)
        h_ref = project_feasible(h_ref, best_n)
        c5_ref = compute_c5(h_ref, best_n)
        if c5_ref < best_c5:
            best_c5, best_h, best_n = c5_ref, h_ref.copy(), best_n

    # Stage 4: Nelder-Mead with true max (derivative-free, handles non-smoothness)
    if best_n <= 60 and time.time() < deadline:
        h_nm = best_h.copy()
        res_nm = minimize(obj_true_max, h_nm, args=(best_n, 0),
                          method='Nelder-Mead',
                          options={'maxiter': 800, 'xatol': 1e-10, 'fatol': 1e-14})
        if res_nm.x is not None:
            h_nm = project_feasible(res_nm.x, best_n)
            c5_nm = compute_c5(h_nm, best_n)
            if c5_nm < best_c5:
                best_c5, best_h = c5_nm, h_nm.copy()

    # Stage 5: Random restarts
    n_restarts = 8
    for ri in range(n_restarts):
        if time.time() > deadline:
            break
        sigma = 0.08 / (1.0 + 0.5 * ri)
        h_r = best_h + np.random.randn(best_n) * sigma
        h_r = project_feasible(h_r, best_n)
        h_r = do_slsqp(h_r, best_n, 40.0, 150, deadline)
        h_r = project_feasible(h_r, best_n)
        c5_r = compute_c5(h_r, best_n)
        if c5_r < best_c5:
            best_c5, best_h = c5_r, h_r.copy()

    # Stage 6: Try a completely different structure - two-block pattern
    if time.time() < deadline:
        h_blk = np.zeros(best_n)
        h_blk[:best_n // 4] = 1.0
        h_blk[best_n // 2:best_n // 2 + best_n // 4] = 1.0
        h_blk = project_feasible(h_blk, best_n)
        h_blk = do_slsqp(h_blk, best_n, 30.0, 200, deadline)
        h_blk = project_feasible(h_blk, best_n)
        c5_blk = compute_c5(h_blk, best_n)
        if c5_blk < best_c5:
            best_c5, best_h = c5_blk, h_blk.copy()

    # Final: ensure constraints
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h, best_n)
    return (best_h, final_c5, best_n)
