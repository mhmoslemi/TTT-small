import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def repair(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        if s < target:
            mask = h < 1.0 - 1e-12
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] = np.clip(h[mask] + (target - s) / cnt, 0.0, 1.0)
        else:
            mask = h > 1e-12
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] = np.clip(h[mask] - (s - target) / cnt, 0.0, 1.0)
    return h

def smooth_max_obj(x, n, beta):
    dx = 2.0 / n
    x = np.clip(x, 0.0, 1.0)
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    if m < 1e-15:
        return 0.0
    shifted = beta * (corr - m)
    lse = m + np.log(np.sum(np.exp(shifted))) / beta
    return lse * dx

def slsqp_solve(h_start, n, beta, maxiter):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: np.ones(n)}]
    res = minimize(
        lambda x: smooth_max_obj(x, n, beta), h_start,
        method='SLSQP', bounds=bounds, constraints=cons,
        options={'maxiter': maxiter, 'ftol': 1e-14}
    )
    return repair(res.x, n)

def coordinate_descent(h0, n, step, n_iters, rng, budget_frac, t0, budget_s):

    best_c5 = evaluate_c5(best_h)
    idx_arr = np.arange(n)

        if time.time() - t0 > budget_s * budget_frac:

        i = rng.choice(idx_arr)
        j = rng.choice(idx_arr)
        if i == j:
            continue
        for d in (step, -step):
            h_try = best_h.copy()
            h_try[i] -= d
            h_try[j] += d
            if 0.0 <= h_try[i] and h_try[i] <= 1.0 and 0.0 <= h_try[j] and h_try[j] <= 1.0:
                c5_try = evaluate_c5(h_try)
                if c5_try < best_c5 - 1e-15:
                    best_h = h_try
                    best_c5 = c5_try
                    break
    return best_h, best_c5

    """Push values toward 0 or 1 while preserving sum."""
