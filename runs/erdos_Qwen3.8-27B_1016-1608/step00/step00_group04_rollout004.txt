import numpy as np
import random
import math
import time

def compute_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float((corr * dx).max())

def make_binary(n, k, rng):
    h = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, size=k, replace=False)
    h[idx] = 1.0
    return h

def _do_swap(h, ones, zeros, i, j):
    h[i] = 0.0
    h[j] = 1.0
    pi = ones.index(i)
    ones[pi] = j
    pj = zeros.index(j)
    zeros[pj] = i

def _undo_swap(h, ones, zeros, i, j):
    h[j] = 0.0
    h[i] = 1.0
    p1 = ones.index(j)
    ones[p1] = i
    p2 = zeros.index(i)
    zeros[p2] = j

def sa_search(h_init, time_budget, seed, T_init=0.003, cool=0.99999, T_floor=1e-10, restart_limit=300000):
    rng = random.Random(seed)
    n = len(h_init)
    h = h_init.astype(np.float64).copy()
    current = compute_score(h)
    best = current
    best_h = h.copy()

    ones = [i for i in range(n) if h[i] == 1.0]
    zeros = [i for i in range(n) if h[i] == 0.0]

    t0 = time.time()
    T = T_init
    no_improve = 0

    while True:
        if time.time() - t0 >= time_budget:
            break
        if not ones or not zeros:
            break

        i = ones[rng.randrange(len(ones))]
        j = zeros[rng.randrange(len(zeros))]

        _do_swap(h, ones, zeros, i, j)
        new_score = compute_score(h)

        if new_score < best:
            best = new_score
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        delta = new_score - current
        if delta <= 0:
            current = new_score
        elif T > T_floor:
            ratio = -delta / T
            if ratio > 500.0:
                accept = False
            else:
                accept = rng.random() < math.exp(ratio)
            if accept:
                current = new_score
            else:
                _undo_swap(h, ones, zeros, i, j)
        else:
            _undo_swap(h, ones, zeros, i, j)

        T = max(T * cool, T_floor)
