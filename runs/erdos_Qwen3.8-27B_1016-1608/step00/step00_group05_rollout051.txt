    if remaining() > 20:
        h_r, c5_r = sa_binary(best_h, n, 0.0005, 20000, rng)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 6: One more NM pass on final binary
    if remaining() > 15:
        h_opt, c5_opt = nm_opt(best_h, n, 2000)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    best_h = make_feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
