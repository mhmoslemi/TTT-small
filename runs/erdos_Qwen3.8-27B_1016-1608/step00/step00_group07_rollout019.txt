    n = len(h)
    h = np.clip(h, 0, 1)
    # Water filling
    for _ in range(100):
        diff = np.sum(h) - S
        if abs(diff) < 1e-12:
            break
        mask_upper = h >= 1.0
        mask_lower = h <= 0.0
        mask_mid = ~mask_upper & ~mask_lower
