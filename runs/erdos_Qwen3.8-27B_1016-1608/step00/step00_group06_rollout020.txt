                order = np.argsort(-h)  # elements closest to 1 first
                for i in order:
                    add = min(diff, 1 - h[i])
                    h[i] += add
                    diff -= add

                order = np.argsort(h)  # elements closest to 0 first
                for i in order:

                    h[i] -= sub
                    diff += sub
                    if diff > -1e-10:
                        break
        
        # Adaptive learning rate
        if step % 20 == 19:
            lr *= 0.9
    
    return h, compute_c5(h, dx)

import numpy as np
import time
import math

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def c5_all(h, dx):
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_constrain(y, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(y - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def subgradient(h, dx, k_act):
    n = len(h)
    vals = c5_all(h, dx)
    top = np.argsort(vals)[-k_act:]
    grad = np.zeros(n)
    for k in top:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo > hi:
            continue
        m = np.arange(lo, hi + 1)
        km = k - m
        grad[m] += 1.0 - 2.0 * h[km]
    grad *= dx / float(k_act)
    return grad

def continuous_optimize(h, dx, time_limit=15.0):
    n = len(h)
    h = project_constrain(h.astype(float), n)
    t_start = time.time()
    lr = 0.12
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    mom = np.zeros(n)
    mu = 0.85
    stall_count = 0

    for step in range(20000):
        if time.time() - t_start > time_limit:
            break
        k_act = max(1, min(n, max(3, (2 * n - 1) // 10)))
        grad = subgradient(h, dx, k_act)
        mom = mu * mom + (1.0 - mu) * grad
        h_new = project_constrain(h - lr * mom, n)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5 - 1e-10:
            if c5_new < best_c5 - 1e-6:
                stall_count = 0
            else:

            best_c5 = c5_new

        else:
            stall_count += 1
            if stall_count > 200:
                lr *= 0.85
                stall_count = 0
        if step % 300 == 299:
            lr *= 0.95
    return best_h, best_c5

            h[i] = 1.0
    t_start = time.time()
