import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_score(h):
    n = len(h)
    return float(np.max(compute_corr(h)) * (2.0 / n))

def project_feasible(x, target):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        if np.clip(x - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def grad_offset(h, s):
    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        g[s:] += (1.0 - h[:n - s])
        g[:n - s] -= h[s:]
    else:
        a = -s
        g[:n - a] += (1.0 - h[a:])
        g[a:] -= h[:n - a]
    return g

def continuous_opt(h, t0, deadline):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0, 1)
    h = project_feasible(h, target)
    score = compute_score(h)
    
    alpha = 2.0 / n
    plateau = 0
    
    while time.time() < deadline:
        corr = compute_corr(h)
        max_c = corr.max()
        active = np.where(corr >= 0.98 * max_c)[0]
        
        g = np.zeros(n)
        for k in active:
            s = k - (n - 1)
            g += grad_offset(h, s)
        g /= len(active)
        
        h_new = project_feasible(h - alpha * g, target)
        new_score = compute_score(h_new)
        
        if new_score < score - 1e-12:
            h = h_new
            score = new_score
            plateau = 0
            alpha = min(alpha * 1.05, 5.0 / n)
        else:
            plateau += 1
            alpha *= 0.95

def binary_local_search(h, t0, deadline, n_max_exhaustive=5000):

            pairs = [(i, j) for i in ones for j in zeros]
            np.random.shuffle(pairs)
        else:
            pairs = [(ones[np.random.randint(len(ones))], zeros[np.random.randint(len(zeros))])
                     for _ in range(min(5000, nc))]

            if time.time() > deadline:
                break
            h[i] = 0.0
            h[j] = 1.0
            ns = compute_score(h)

                score = ns
                improved = True
            else:
                h[i] = 1.0
                h[j] = 0.0
        
        if not improved:
            no_improve += 1
            if no_improve > 3:
                break
        else:
            no_improve = 0
    
    return h, score
