def compute_corr_fast(h):
    n = len(h)
    # corr[k] for k = -(n-1) to (n-1)
    # corr[k] = sum_i h[i+k] * (1-h[i]) for valid i
    result = np.zeros(2*n-1)

    for k in range(1, n):
        # shift by -k (h[i-k] * (1-h[i]))
        result[n-1-k] = np.dot(h[:n-k], 1-h[k:])
    return result

    current_score = compute_score(current)
    ones = list(np.where(current > 0.5)[0])
    zeros = list(np.where(current < 0.5)[0])
    T = T0
    best = current.copy()
    best_score = current_score
    for _ in range(n_steps):
        if T < 1e-9:
            break
        ii = rng.randint(len(zeros))
        jj = rng.randint(len(ones))
        i = zeros[ii]
        j = ones[jj]
        current[i] = 1.0
        current[j] = 0.0
        new_score = compute_score(current)
        delta = new_score - current_score
        if delta < 0.0 or np.log(rng.random()) < -delta / T:
            current_score = new_score
            ones[jj] = i
            zeros[ii] = j
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        else:
            current[i] = 0.0
            current[j] = 1.0
        T *= T_factor
    return best, best_score

def perturb_binary(h, rng, n_flips):
    h_p = h.copy().astype(float)
    for _ in range(n_flips):
        z = list(np.where(h_p < 0.5)[0])
        o = list(np.where(h_p > 0.5)[0])
        if not z or not o:
            break
        h_p[z[rng.randint(len(z))]] = 1.0
        h_p[o[rng.randint(len(o))]] = 0.0
    return h_p

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    best_h = None
    best_score = np.inf

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        h0 = None

    if h0 is not None:
        n = len(h0)
        h_bin = np.round(h0)
        target = n // 2
        cs = int(h_bin.sum())
        if cs > target:
            idx = np.where(h_bin == 1)[0]
            h_bin[idx[cs - target:]] = 0
        elif cs < target:
            idx = np.where(h_bin == 0)[0]
            h_bin[idx[:target - cs]] = 1
        s = compute_score(h_bin)
        best_score = s
        best_h = h_bin.astype(float).copy()
    else:
        best_h = np.zeros(42)
        best_h[rng.permutation(42)[:21]] = 1.0
        best_score = compute_score(best_h)

    # Phase 1: Exhaustive best-improvement on initial
    limit1 = budget_s * 0.15
    h1, s1 = best_improve_search(best_h, max_passes=2000, time_limit=limit1, t0=t0)
    if s1 < best_score:
        best_score, best_h = s1, h1

    # Phase 2: SA from best (multiple restarts with perturbations)
    limit2 = budget_s * 0.
    limit2 = budget_s * 0.45
    n_sa1 = 30000
    while time.time() - t0 < limit2:
        h_pert = perturb_binary(best_h, rng, 3 + rng.randint(4))
        hs, ss = sa_search(h_pert, rng, n_sa1, 0.05, 0.99996)
        if ss < best_score:
            best_score, best_h = ss, hs
        n_sa1 = max(10000, int(n_sa1 * 0.95))

    # Phase 3: Try other n values
    limit3 = budget_s * 0.80
    n_list = [44, 46, 48, 50, 52, 54, 56, 60, 64, 72, 80]
    for n_try in n_list:
        if time.time() - t0 > limit3:
            break
        for rs in range(2):
            if time.time() - t0 > limit3:
                break
            h_t = np.zeros(n_try)
            h_t[rng.permutation(n_try)[:n_try // 2]] = 1.0
            sub_limit = min(limit3, time.time() - t0 + budget_s * 0.05)
            h_t, s_t = best_improve_search(h_t, max_passes=300, time_limit=sub_limit, t0=t0)
            hs, ss = sa_search(h_t, rng, 15000, 0.05, 0.99997)
            if ss < s_t:
                h_t, s_t = hs, ss
            if s_t < best_score:
                best_score, best_h = s_t, h_t

    # Phase 4: Extended SA from best found
    limit4 = budget_s * 0.95
    while time.time() - t0 < limit4:
        h_pert = perturb_binary(best_h, rng, 1 + rng.randint(3))
        hs, ss = sa_search(h_pert, rng, 40000, 0.03, 0.99998)
        if ss < best_score:
            best_score, best_h = ss, hs
            h_pert = best_h.copy()
        # Also try from unperturbed
        hs2, ss2 = sa_search(best_h, rng, 20000, 0.02, 0.99999)
        if ss2 < best_score:
            best_score, best_h = ss2, hs2

    # Final: one more best-improve pass
    if time.time() - t0 < budget_s * 0.97:
        h_f, s_f = best_improve_search(best_h, max_passes=200, time_limit=budget_s * 0.98, t0=t0)
        if s_f < best_score:
            best_score, best_h = s_f, h_f

    best_h = np.clip(best_h, 0.0, 1.0)
    final_score = compute_score(best_h)
    return (best_h, final_score, len(best_h))
