    # Normalize curve to integral=1
    total_area = np.trapz(base_curve, grid)
    weights = base_curve / total_area * 1.0  # Preserve shape
    
    # Noise injection
    weights = weights * (1-0.05) + gaussian_noise(size=len(weights)) * 0.05
    
    # Clip to [0,1]
    weights = np.clip(weights, 0, 1)
    
    # Reconstruct discrete grid
    discrete = np.zeros(n)
    for i, x_i in enumerate(grid):
        discrete[i] = np.sum(basis(x_i) * weights) + small_noise
    discrete = np.clip(discrete, 0, 1)
    
    return discrete
```

def optimize_erdos_problem(base_config, budget=1100, max_n=150):
    config = base_config.copy()
    best_cost = float('inf')
    converging = False
    best_theta = config['params'].copy()
    time_elapsed = 0

    # 1. Initialize using trust-regions
    trust_size = 0.1
    merit_function = compute_euler_merit(best_theta, config, trust_size)

    while time_elapsed < budget and not converged:
        # 2. Create candidate using quasi-Newton step
        grad_theta = compute_gradient(best_theta)
        step = -LBFGS_step(grad_theta) * trust_size
        
        # 3. Update candidate with projection
        candidate = best_theta + step
        candidate_grid = project_to_feasible(candidate, config['interp'], config['grid_orig'])
        
        # 4. Evaluate and accept/reject
        cost_candidate = compute_overlap(candidate_grid)
        cost_current = compute_overlap(best_theta_grid)
        
        # 5. Update trust-region based on improvement
        if cost_candidate < cost_current - 0.5*trust_size:
            trust_size *= 1.5
            accept(candidate)
            best_cost = cost_candidate
        elif cost_candidate < cost_current + 0.5*trust_size:
            trust_size *= 0.9
            accept(candidate)
        else:
            trust_size *= 0.8
        
        # Check convergence
        if abs(cost_candidate - best_cost) < 1e-4:
            converging = True
            time.sleep(2)  # Cooldown to prevent oscillations
        
        time_elapsed += 1.5  # Per-iteration time estimate
        
    return {'h_grid': best_theta_grid, 'cost': best_cost}
```

    fourier_coeffs = theta[:10]
    spline_weights = theta[10:15]

    sample_points = x_series.reshape(-1,1)
    curve_vals = base_curve(sample_points)
