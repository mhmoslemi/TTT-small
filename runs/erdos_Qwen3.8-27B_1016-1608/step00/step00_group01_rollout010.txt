import numpy as np
import time
import random

def compute_c5(h, n):
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    return float(np.max(corr) * dx)

def repair(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s == 0:
        h[:] = target_sum / len(h)
        return h
    h = h * (target_sum / s)
    h = np.clip(h, 0.0, 1.0)
    return h

def local_search(h, n, target_sum, max_iter, deadline, best_ref):
    best_h, best_c5 = best_ref
    current_c5 = compute_c5(h, n)
    h = h.copy()
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        i, j = random.sample(range(n), 2)
        delta = random.uniform(-0.05, 0.05)
        hi_new = h[i] + delta
        hj_new = h[j] - delta
        if hi_new < 0 or hi_new > 1 or hj_new < 0 or hj_new > 1:
            continue
        h[i] = hi_new
        h[j] = hj_new
        c5 = compute_c5(h, n)
        if c5 < current_c5:
            current_c5 = c5
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[i] -= delta
            h[j] += delta
    return best_h, best_c5

    current_c5 = compute_c5(h, n)

            h[j] += delta
        T = max(T * cool, T_min)
    return h, best_h, best_c5

def binary_init(n, target_sum, rng):
    h = np.zeros(n)
    k = int(target_sum)
    rest = target_sum - k
    ones = rng.choice(n, k, replace=False)
    for idx in ones:
        h[int(idx)] = 1.0
    filled = set(ones.astype(int).tolist())
    remaining = [x for x in range(n) if x not in filled]
    if remaining:
        h[remaining[0]] = rest
    return h

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    t0 = time.time()
    total_budget = budget_s - 5.0
    deadline = t0 + total_budget
    n = 47
    target_sum = n / 2.0

    try:
        h = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h = np.full(n, 0.5)
    h = np.clip(h, 0.0, 1.0)
    sm = h.sum()
    if sm > 1e-10:
        h *= target_sum / sm
    h = np.clip(h, 0.0, 1.0)

    best_h = h.copy()
    best_c5 = compute_c5(best_h, n)

    # Phase 1: Pairwise coordinate descent (35% budget)
    dl1 = t0 + total_budget * 0.35
    h, best_h, best_c5 = pairwise_cd(h, n, 999999, dl1, best_h, best_c5)

    # Phase 2: Simulated annealing (25% budget)
    dl2 = t0 + total_budget * 0.60
    h, best_h, best_c5 = sa_optimize(h, n, dl2, best_h, best_c5)

    # Phase 3: CD refinement (25% budget)
    dl3 = t0 + total_budget * 0.85
    h, best_h, best_c5 = pairwise_cd(h, n, 999999, dl3, best_h, best_c5)

    # Phase 4: Multi-restart exploration (15% budget)
    for _ in range(10000):
        if time.time() > deadline:
            break
        if random.random() < 0.3:
            h_pert = binary_init(n, target_sum, np.random)
        else:
            h_pert = best_h + np.random.normal(0, 0.02, n)
            h_pert = np.clip(h_pert, 0.0, 1.0)
            h_pert += (target_sum - h_pert.sum()) / n
            h_pert = np.clip(h_pert, 0.0, 1.0)
        sub_dl = min(deadline, time.time() + 1.5)
        _, best_h, best_c5 = pairwise_cd(h_pert, n, 99999, sub_dl, best_h, best_c5)

    best_c5 = compute_c5(best_h, n)
    return list(best_h), best_c5, n
