from scipy.optimize import minimize
import time

# Global parameters for constraint functions
_G = {'target': 0.0, 'dx': 0.0}

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def _project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        d = target - np.sum(h)
        if abs(d) < 1e-15:

        if d > 0:
            m = h < 1.0 - 1e-13
        else:
            m = h > 1e-13
        c = np.sum(m)
        if c == 0:
            break
        h[m] += d / c
        h = np.clip(h, 0.0, 1.0)
    return h

    return x[-1]

def _sum_c(x):
    return np.sum(x[:-1]) - _G['target']

def _ovl_c(x):
    h = x[:-1]
    corr = np.correlate(h, 1.0 - h, mode='full') * _G['dx']
    return x[-1] - np.max(corr)

def _slsqp(h0, n, maxiter=3000):
    _G['target'] = n / 2.0
    _G['dx'] = 2.0 / n
    t0 = _c5(h0) + 0.05
    x0 = np.concatenate([h0, [t0]])

    dx = 2.0 / n
    cur = _c5(h)
    for it in range(iterations):

        i, j = idx[0], idx[1]
        if i == j:

        # Scan alpha

def _c5(h):

    return float(np.max(corr))

def _project(h, n):
