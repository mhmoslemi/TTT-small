    return float(np.max(c)) * dx

    return np.array([h.sum() - len(h) / 2.0])

    return np.ones((1, len(h)))

    val = mc + np.log(np.sum(np.exp(t * (c - mc)))) / t

def project_sum(h, target):
    h = np.asarray(h, dtype=float).copy()

    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

    # Phase 2: aggressive final polish on best solution
    if best_h is not None and remaining() > 15:
        n = best_n
        for func in (obj_func, smooth_obj2, obj_func):
            if remaining() < 10:
                break
            ho, c5o = run_tc(best_h.copy(), n, func, 800)
            if ho is not None and c5o < best_c5:
                best_c5, best_h = c5o, ho
            if remaining() < 10:
                break
            ho, c5o = run_slsqp(best_h.copy(), n, func, 500)
            if ho is not None and c5o < best_c5:
                best_c5, best_h = c5o, ho
        # Local search
        if remaining() > 10:
            best_h, best_c5 = local_search(best_h, n, 60)
            best_n = n

    # Phase 3: cross-check with a different n by interpolating
    if best_h is not None and remaining() > 30:
        n_src = len(best_h)
        n_dst = 140
        x_src = np.linspace(0.0, 2.0, n_src)
        x_dst = np.linspace(0.0, 2.0, n_dst)
        h_interp = np.interp(x_dst, x_src, best_h, left=best_h[0], right=best_h[-1])
        h_interp = project_sum(h_interp, n_dst / 2.0)
        ho, c5o = run_tc(h_interp, n_dst, obj_func, 400)
        if ho is not None and c5o < best_c5:
            best_c5, best_h, best_n = c5o, ho, n_dst
        ho, c5o = run_tc(h_interp, n_dst, smooth_obj, 300)
        if ho is not None and c5o < best_c5:
            best_c5, best_h, best_n = c5o, ho, n_dst

    # Ensure exact sum
    if best_h is not None:
        ts = best_n / 2.0
        diff = ts - best_h.sum()
        if abs(diff) > 1e-14:
            best_h += diff / best_n
            best_h = np.clip(best_h, 0.0, 1.0)

    # Final validation
    try:
        ev = evaluate_erdos_solution(best_h)
        if isinstance(ev, (int, float)) and float(ev) < best_c5:
            best_c5 = float(ev)
    except Exception:
        pass

    return (best_h, float(best_c5), best_n)
