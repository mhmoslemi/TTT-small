        # Choose move type
        r = rng.random()
        if r < 0.7:
            h2 = try_move(h, n, rng, step)
        elif r < 0.85:
            h2 = block_move(h, n, rng, step * 0.5)
        else:
            corr = np.correlate(h, 1.0 - h, mode='full')
            worst_k = int(np.argmax(corr))
            h2 = targeted_move(h, n, rng, worst_k, step)
        
        if h2 is None:
            continue
        
        new_c5 = compute_c5(h2, n)
        
        if new_c5 <= cur_c5:
            h = h2
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            if rng.random() < math.exp(-(new_c5 - cur_c5) / T):
                h = h2
                cur_c5 = new_c5
                no_improve += 1
            else:
                no_improve += 1
        
        it += 1
        T = max(T * cool, T_floor)
        
        # Adaptive step: shrink if too many rejections
        if it % 5000 == 0:
            if no_improve > 4000:
                step *= 0.9
            elif no_improve < 1000:
                step = min(step * 1.05, step0)
            no_improve = 0
        
        # Periodic restart from best with small perturbation
        if it % 50000 == 0 and T < T_floor * 100:
            h = best_h.copy()
            d = rng.normal(0, step * 0.5, n)
            h = h + d
            h = repair_sum(h, n)
            cur_c5 = compute_c5(h, n)
            T = T0 * 0.3
    
    return best_h, best_c5

def local_descent(h, n, rng, t0, budget_s, margin=3.0):
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    t_limit = t0 + budget_s - margin
    step = 0.03
    
    while time.time() < t_limit:
        improved = False
        for _ in range(500):
            if time.time() >= t_limit:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            d = rng.uniform(-step, step)
            h2 = h.copy()
            h2[i] += d
            h2[j] -= d
            h2 = np.clip(h2, 0.0, 1.0)
            h2 = repair_sum(h2, n)
            c = compute_c5(h2, n)
            if c < best_c5:
                best_c5 = c
                best_h = h2.copy()
                h = h2
                improved = True
        if not improved:
            step *= 0.9
            if step < 1e-5:
                break
    
    return best_h, best_c5

def make_structured_initial(n, rng):
    # Create a structured starting point (clustered blocks)
    h = np.zeros(n)
    # Random partition into blocks of 1s and 0s
    blocks = rng.integers(3, 8)
    seg_sizes = rng.integers(1, n // blocks + 1, blocks)
    seg_sizes = (seg_sizes * (n // sum(seg_sizes) + 1)).astype(int)
    # Adjust to fill n
    while len(seg_sizes) < n:
        idx = int(rng.integers(len(seg_sizes)))
        seg_sizes = np.insert(seg_sizes, idx + 1, seg_sizes[idx])
        seg_sizes = seg_sizes[:n]
    pos = 0
    for s in seg_sizes:
        val = 1.0 if rng.random() < 0.5 else 0.0
        end = min(pos + s, n)
        h[pos:end] = val
        pos = end
    h = repair_sum(h, n)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    
    # Get initial solution
    n = len(initial_h_values)
    h_init = np.array(initial_h_values, dtype=np.float64).copy()
    h_init = repair_sum(h_init, n)
    
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init, n)
    
    # Phase 1: SA from parent (60% of budget)
    h_pa, c5_pa = simulated_anneal(
        h_init, n, rng,
        T0=0.008, T_floor=1e-8, cool=0.99996,
        step0=0.12, budget_frac=0.60, t0=t0, budget_s=budget_s
    )
    if c5_pa < best_c5:
        best_c5 = c5_pa
        best_h = h_pa.copy()
    
    # Phase 2: SA from different structured starts (20%)
    for trial in range(3):
        if time.time() - t0 > budget_s * 0.78:
            break
        h_s = make_structured_initial(n, rng)
        h_pa2, c5_pa2 = simulated_anneal(
            h_s, n, rng,
            T0=0.012, T_floor=1e-8, cool=0.99995,
            step0=0.15, budget_frac=0.06, t0=time.time(), budget_s=budget_s
        )
        if c5_pa2 < best_c5:
            best_c5 = c5_pa2
            best_h = h_pa2.copy()
    
    # Phase 3: SA from random binary (10%)
    if time.time() - t0 < budget_s * 0.85:
        h_bin = np.zeros(n)
        idx = rng.choice(n, size=n // 2, replace=False)
        h_bin[idx] = 1.0
        h_pa3, c5_pa3 = simulated_anneal(
            h_bin, n, rng,
            T0=0.015, T_floor=1e-8, cool=0.99994,
            step0=0.2, budget_frac=0.10, t0=time.time(), budget_s=budget_s
        )
        if c5_pa3 < best_c5:
            best_c5 = c5_pa3
            best_h = h_pa3.copy()
    
    # Phase 4: Local descent polish (remaining time)
    best_h, best_c5 = local_descent(best_h, n, rng, t0, budget_s)
    
    # Final verification
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = repair_sum(best_h, n)
    best_c5 = compute_c5(best_h, n)
    
    return best_h, best_c5, n
