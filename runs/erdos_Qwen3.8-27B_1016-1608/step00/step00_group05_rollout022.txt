import time

def compute_corr(h):
    return np.correlate(h, 1 - h, mode="full")

def compute_c5(h):
    n = len(h)
    return float(np.max(compute_corr(h)) * 2.0 / n)

def project_feasible(h, target_sum):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def binary_sa(h_bin, rng, n_steps=20000, t_start=1.0, t_end=0.01):
    """Simulated annealing on binary string."""
    n = len(h_bin)
    h = h_bin.copy()
    best_h = h.copy()
    corr = compute_corr(h)
    best_energy = corr.max()
    energy = best_energy
    
    t = t_start
    cooling = (t_end / t_start) ** (1.0 / max(1, n_steps))
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for step in range(n_steps):
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h[i], h[j] = h[j], h[i]
        corr_new = compute_corr(h)
        new_energy = corr_new.max()
        delta = new_energy - energy
        
        if delta < 0 or rng.rand() < np.exp(-delta / max(t, 1e-10)):
            energy = new_energy
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if energy < best_energy:
                best_energy = energy
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
        
        t *= cooling
    
    return best_h, best_energy

        top_frac = min(5, len(corr) // 20)
        top_idx = np.argsort(corr)[-top_frac:]
        n_top = len(top_idx)
        
        sg = np.zeros(n)
        j = np.arange(n)
        for idx in top_idx:
            lag = int(idx - (n - 1))
            i1 = j - lag
            m1 = (i1 >= 0) & (i1 < n)
            sg[m1] += 1.0 - h[i1[m1]]
            i2 = j + lag
            m2 = (i2 >= 0) & (i2 < n)
            sg[m2] -= h[i2[m2]]

        sg -= np.mean(sg)

            best_c5 = c5

            h_bin[one_idx[order]] = 0
        else:

            to_add = target_ones - n_ones
            order = np.argsort(-best_h[zero_idx])[:to_add]
            h_bin[zero_idx[order]] = 1

import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1 - h, mode="full")

def compute_c5(h):
    n = len(h)
    return float(np.max(compute_corr(h)) * 2.0 / n)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_at_lag(h, lag):
    n = len(h)
    sg = np.zeros(n)
    j = np.arange(n)
    i1 = j - lag
    m1 = (i1 >= 0) & (i1 < n)
    sg[m1] += 1.0 - h[i1[m1]]
    i2 = j + lag
    m2 = (i2 >= 0) & (i2 < n)
    sg[m2] -= h[i2[m2]]
    return sg

def binary_sa(h_bin, rng, n_steps, t_start=2.0, t_end=0.02):
    n = len(h_bin)
    h = h_bin.copy()
    best_h = h.copy()
    best_e = compute_corr(h).max()
    e = best_e
    t = t_start
    cool = (t_end / t_start) ** (1.0 / max(1, n_steps))
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for _ in range(n_steps):
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = h[j], h[i]
        ne = compute_corr(h).max()
        d = ne - e
        if d < 0 or rng.rand() < np.exp(-d / max(t, 1e-10)):
            e = ne
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if e < best_e:
                best_e = e
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]

                best_h = h_res

    # Final: ensure feasibility
    if np.allclose(best_h, np.round(best_h), atol=0.01):
        best_h = np.round(best_h).astype(float)
    else:
        best_h = project_feasible(best_h, target_sum)

    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
