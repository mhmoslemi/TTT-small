import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr), corr

def repair_h(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        # Distribute adjustment to interior points
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        nm = np.sum(mask)
        if nm > 0:
            step = diff / nm
            # Ensure we don't go out of bounds
            max_step_up = (1.0 - h[mask])
            max_step_down = h[mask]
            if step > 0:
                adj = np.minimum(max_step_up, step)
                h[mask] += adj
            else:
                adj = np.maximum(max_step_down, step)
                h[mask] += adj

        # If still off, adjust one element

        if abs(diff) > 1e-12:

                    h[i] = min(1.0, h[i] + diff)
                    break
                elif diff < 0 and h[i] > 0.0:
                    h[i] = max(0.0, h[i] + diff)
                    break
    return h
