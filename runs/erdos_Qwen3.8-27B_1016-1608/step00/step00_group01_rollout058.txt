    h = np.clip(h, 0, 1)
    target = n / 2.0
    current = np.sum(h)
    if current == 0:
        h = np.ones(n) * target / n
        return h
    # Scale to meet sum constraint, but respect bounds

        if np.sum(mask) > 0:

from scipy.optimize import minimize_scalar

def repair_h(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}"""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    # Distribute the difference
    for _ in range(10):
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        if not np.any(mask):
            break
        step = diff / np.sum(mask)
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
    # Final enforcement
    if abs(np.sum(h) - target) > 1e-10:
        # Force exact sum by adjusting one element
        idx = n // 2
        h[idx] = np.clip(h[idx] + (target - np.sum(h)), 0, 1)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_c5_and_argmax(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    return float(corr[idx]) * dx, idx, corr * dx

def project_onto_constraint(v, n):
    """Project v onto {0<=x<=1, sum(x)=n/2} using bisection"""
    target = n / 2.0
    def sum_after_bisect(tau):
        return np.sum(np.clip(v - tau, 0, 1))
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        if sum_after_bisect(mid) > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2
    return np.clip(v - tau, 0, 1)

def gradient_subgradient(h, n):
    """Compute subgradient of max_k f_k(h)"""
    # Find the max shift
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = int(np.argmax(corr))
    # Offset k = max_idx - (n-1)
    k = max_idx - (n - 1)
    # f_k(h) = sum_n h[n+k](1-h[n]) for valid n
    # df_k/df h[m] = (1-h[m-k]) * 1_{valid} - h[m+k] * 1_{valid}
    grad = np.zeros(n)
    for m in range(n):
        val = 0.0
        # m = n+k => n = m-k, valid if 0<=m-k<n
        if 0 <= m - k < n:
            val += (1.0 - h[m - k])
        # m = n => n+k = m+k, valid if 0<=m+k<n
        if 0 <= m + k < n:
            val -= h[m + k]
        grad[m] = val
    return grad * dx

            it1 = max(30, min(200, int(tl() * 1.5)))
            ho, co = local_opt(hp, n_best, it1, lr0=0.1)
            if co < best_c5:
                best_h, best_c5, best_n = ho.copy(), co, n_best
                it2 = max(100, min(2000, int(tl() * 2)))
                hs2, cs2 = sa_opt(ho, n_best, t0=0.03, iters=it2, alpha=0.9997)
                if cs2 < best_c5:
                    best_h, best_c5, best_n = hs2.copy(), cs2, n_best

    # Final coordinate polish
    if best_h is not None and tl() > 40:
        hm, cm = coord_polish(best_h, best_n, max_sweeps=200)
        if cm < best_c5:
            best_h, best_c5, best_n = hm, cm, best_n

    return best_h, best_c5, best_n
