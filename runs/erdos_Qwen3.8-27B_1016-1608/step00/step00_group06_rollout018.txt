import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def nm_optimize(h0, n, max_iter, seed):
    rng = np.random.RandomState(seed)
    def obj(x):
        return compute_c5(project_feasible(x, n))
    res = minimize(obj, h0, method='Nelder-Mead',
                  options={'maxiter': max_iter, 'xatol': 1e-12, 'fatol': 1e-12,
                          'adaptive': True})
    return project_feasible(res['x'], n)

def coord_descent(h, n, max_sweeps, seed):
    rng = np.random.RandomState(seed)
    best = h.copy()
    best_c5 = compute_c5(best)
    for sweep in range(max_sweeps):
        improved = False
        order = list(range(n))
        rng.shuffle(order)
        for i in order:
            for dv in [0.01, -0.01, 0.05, -0.05, 0.1, -0.1, 0.001, -0.001]:
                test = best.copy()
                test[i] = np.clip(best[i] + dv, 0, 1)

                if v < best_c5 - 1e-15:
                    best = test
                    best_c5 = v
                    improved = True
                    break
        if not improved:
            break
    return best, best_c5

                best = h.copy()
