import time
from scipy.optimize import minimize

def compute_overlaps(h):
    """Compute all overlap values (before multiplying by dx)"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(compute_overlaps(h)) * dx

def smooth_max(vals, temp=0.05):
    m = np.max(vals)
    return m + temp * np.log(np.sum(np.exp((vals - m) / temp)))

def smooth_max_grad(vals, temp=0.05):
    m = np.max(vals)
    shifted = (vals - m) / temp
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    return weights  # softmax weights

def project_onto_simplex_box(h, target_sum):
    """Project h onto {x: 0<=x_i<=1, sum(x)=target_sum}"""
    # Use bisection on lambda
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        projected = np.clip(h - mid, 0, 1)
        s = np.sum(projected)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    projected = np.clip(h - (lo + hi) / 2, 0, 1)
    # Final adjustment for numerical precision
    diff = target_sum - np.sum(projected)
    if abs(diff) > 1e-10:
        # Adjust the element closest to the boundary
        if diff > 0:
            idx = np.argmin(projected[projected < 1])
        else:
            idx = np.argmax(projected[projected > 0])
        projected[idx] += diff
    return projected

def compute_gradient(h, w):
    """
    h: array of length n
    w: weights for each correlation index (length 2n-1)
    Returns gradient of sum_k w[k] * corr[k] wrt h
    """
    n = len(h)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-6:
            continue
        s = k - (n - 1)  # actual shift
        if s >= 0:
            # corr = sum_{i=0}^{n-1-s} h[i]*(1-h[i+s])
            # d/dh[j]: for j in [0, n-1-s]: (1-h[j+s])
            #           for j in [s, n-1]: -h[j-s]
            valid_i = np.arange(n - s)
            grad[valid_i] += w[k] * (1 - h[valid_i + s])
            valid_j = np.arange(s, n)
            grad[valid_j] += w[k] * (-h[valid_j - s])
        else:
            t = -s
            # corr = sum_{i=t}^{n-1} h[i]*(1-h[i-t])
            valid_i = np.arange(t, n)
            grad[valid_i] += w[k] * (1 - h[valid_i - t])
            valid_j = np.arange(n - t)
            grad[valid_j] += w[k] * (-h[valid_j + t])
    return grad

    return float(np.max(compute_overlaps(h)) * (2.0 / n))

def smooth_max(w, t):
    m = w.max()
    return m + t * np.log(np.exp((w - m) / t).sum())

def smooth_weights(w, t):
    m = w.max()
    e = np.exp((w - m) / t)
    return e / e.sum()

        if d > 0:
            idx = np.argmin(p)
        else:
            idx = np.argmax(p)
        p[idx] = np.clip(p[idx] + d, 0, 1)

import numpy as np
import time
from scipy.optimize import minimize

def compute_overlaps(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return float(np.max(compute_overlaps(h)) * (2.0 / n))

def smooth_max(w, t):
    m = w.max()
    return m + t * np.log(np.exp((w - m) / t).sum())

def smooth_weights(w, t):
    m = w.max()
    e = np.exp((w - m) / t)
    return e / e.sum()

def project(h, ts):
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > ts:
            lo = mid
        else:
            hi = mid
    p = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    d = ts - p.sum()
    if abs(d) > 1e-12:
        if d > 0:
            idx = np.argmin(p)
        else:
            idx = np.argmax(p)
        p[idx] = np.clip(p[idx] + d, 0.0, 1.0)
    return p

def grad_corr(h, w):
    n = len(h)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-8:
            continue
        s = k - (n - 1)
        if s >= 0:
            g[:n - s] += wk * (1.0 - h[s:])
            g[s:] -= wk * h[:n - s]
        else:
            t = -s
            g[t:] += wk * (1.0 - h[:-t])
            g[:n - t] -= wk * h[t:]
    return g

def optimize_h(h_init, n, ts, budget_s, start_time, seed):
    rng = np.random.RandomState(seed)
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    temp = 0.12
    lr = 0.25
    patience = 0
    while time.time() - start_time < budget_s:
        ov = compute_overlaps(h)
        w = smooth_weights(ov, temp)
        g = grad_corr(h, w)
        h_new = project(h - lr * g, ts)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-11:
            best_c5 = c5_new
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1
        h = h_new
        temp = max(0.008, temp * 0.997)
        lr = max(0.015, lr * 0.999)
        if patience > 4000:
            h = best_h + 0.08 * rng.randn(n)
            h = project(h, ts)
            temp = 0.06
            lr = 0.12
            patience = 0
    return best_h, best_c5

def local_search(h_init, ts, budget_s, start_time, seed):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = 0.12
    stuck = 0
    while time.time() - start_time < budget_s:
        i, j = rng.choice(n, 2, replace=False)
        d = rng.uniform(-step, step)
        h2 = h.copy()
        h2[i] = np.clip(h2[i] + d, 0.0, 1.0)
        h2[j] = np.clip(h2[j] - d, 0.0, 1.0)

        c52 = compute_c5(h2)
