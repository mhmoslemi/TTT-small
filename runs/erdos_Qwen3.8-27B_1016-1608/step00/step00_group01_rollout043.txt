```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def grad_max_lag(h, n, k):
    dx = 2.0 / n
    g = np.zeros(n)
    ak = abs(k)
    m = n - ak
    if m <= 0:
        return g
    if k >= 0:
        g[:m] += (1.0 - h[ak:]) * dx
        g[ak:] += (-h[:m]) * dx
    else:
        g[ak:] += (1.0 - h[:m]) * dx
        g[:m] += (-h[ak:]) * dx
    return g

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        if mask.sum() == 0:
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def optimize_subgrad(h, n, budget_s, start_time, max_iter=200000):
    target_sum = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    
    lr = 0.05 / n  # scaled learning rate
    min_lr = 1e-6
    no_improve = 0
    patience = 500
    
    for it in range(max_iter):
        if time.time() - start_time > budget_s * 0.98:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_val = np.max(corr)
        c5 = max_val * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        
        # Find top-3 lags for a smoothed subgradient
        top3 = np.argsort(corr)[-3:]
        g = np.zeros(n)
        for idx in top3:
            k = idx - (n - 1)
            g += grad_max_lag(h, n, k)
        g /= len(top3)
        
        h = h - lr * g
        h = project_feasible(h, target_sum)
        
        no_improve += 1
        if no_improve > patience:
            lr *= 0.8
            no_improve = 0
            if lr < min_lr:
                break
    
    return best_h, best_c5

def optimize_slsqp(h0, n, max_iter=200):
    target_sum = n / 2.0
    dx = 2.0 / n
    
    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * dx
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0, 1)] * n
    
    res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                   constraints=constraints,
                   options={'maxiter': max_iter, 'ftol': 1e-12})
    if res.success or res.fun < np.inf:
        h = np.clip(res.x, 0, 1)
        h = project_feasible(h, target_sum)
        return h, compute_c5(h, n)
    return h0, compute_c5(h0, n)

        h = np.clip(h, 0, 1)

            best_c5 = c5
            best_n = n_local

        update(h_s, n0)
        # Subgradient from SLSQP result
        h_sg, c5_sg = optimize_subgrad(h_s, n0, budget_s, start_time, max_iter=50000)
        update(h_sg, n0)
```

```python
    best_h = h_init.copy()
    best_c5 = _compute_c5(h_init)
    best_n = n0

    # SLSQP from initial
    h_s, c5_s = _slsqp_opt(h_init, n0, 800)
    if c5_s < best_c5:
        best_h, best_c5 = h_s, c5_s

    # Subgradient polish
    t_mid = t0 + budget_s * 0.4
    h_sg, c5_sg = _subgrad_opt(best_h, best_n, min(t_mid, t_end - 5), 300000)
    if c5_sg < best_c5:
        best_h, best_c5 = h_sg, c5_sg

    # SLSQP from polished
    t_q3 = t0 + budget_s * 0.55
    h_s2, c5_s2 = _slsqp_opt(best_h, best_n, 500)
    if c5_s2 < best_c5:
        best_h, best_c5 = h_s2, c5_s2

    # More subgradient
    h_sg2, c5_sg2 = _subgrad_opt(best_h, best_n, min(t_q3, t_end - 10), 200000)
    if c5_sg2 < best_c5:
        best_h, best_c5 = h_sg2, c5_sg2

    # Random restarts from n0
    for tr in range(80):
        if time.time() > t_end - 20:
            break
        h0 = _project(0.5 + rng.randn(n0) * 0.3, n0 / 2.0)
        h_r, c5_r = _slsqp_opt(h0, n0, 300)
        if c5_r < best_c5:
            best_h, best_c5 = h_r, c5_r
            h_p, c5_p = _subgrad_opt(h_r, n0, min(time.time() + 3, t_end - 10), 50000)
            if c5_p < best_c5:
                best_h, best_c5 = h_p, c5_p

    # Nelder-Mead from best
    if time.time() < t_end - 30:
        h_nm, c5_nm = _nelder_opt(best_h.copy(), best_n, t_end - 20, 5000)
        if c5_nm < best_c5:
            best_h, best_c5 = h_nm, c5_nm

    # Try other n values
    for nt in [30, 40, 48, 50, 60, 80]:
        if time.time() > t_end - 20:
            break
        h0 = _project(0.5 + rng.randn(nt) * 0.3, nt / 2.0)
        h_r, c5_r = _slsqp_opt(h0, nt, 200)
        if c5_r < best_c5:
            best_h, best_c5, best_n = h_r, c5_r, nt

    # Final SLSQP polish on best
    if time.time() < t_end - 10:
        h_f, c5_f = _slsqp_opt(best_h.copy(), best_n, 300)
        if c5_f < best_c5:
            best_h, best_c5 = h_f, c5_f

    best_h = _project(best_h, best_n / 2.0)
    best_c5 = _compute_c5(best_h)
    return best_h, best_c5, best_n
```
