        k = k_max - (n - 1)
        for i in range(n):
            # Term where h[i] appears as first factor: h[i]*(1-h[i+k]) if 0<=i+k<n
            if 0 <= i + k < n:
                g[i] += (1.0 - h[i + k])
            # Term where h[i] appears as (1-h[i-k]) factor: h[i-k]*(1-h[i]) if 0<=i-k<n
            if 0 <= i - k < n:
                g[i] += -h[i - k]
        return g * (2.0 / n)

    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=constraints, options={'maxiter': 2000, 'ftol': 1e-10})
        h_opt = np.clip(res.x, 0.0, 1.0)
        # Fix sum
        diff = n / 2.0 - h_opt.sum()
        mask = (h_opt > 0.01) & (h_opt < 0.99)
        if mask.sum() > 0:
            h_opt[mask] += diff / mask.sum()
        h_opt = np.clip(h_opt, 0.0, 1.0)
        return h_opt
    except Exception:
        return h0

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    best_h = None
    best_c5 = float('inf')

    def try_update(h):
        nonlocal best_h, best_c5
        c = c5_val(h)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()

    # Phase 1: Hill climb from initial (45% time)
    h_init = to_binary(np.array(initial_h_values, dtype=float))
    try_update(h_init)
    t1 = t0 + budget_s * 0.45
    h1, _ = hill_climb(h_init, t1, samples=50)
    try_update(h1)

    # Phase 2: Simulated annealing (20% time)
    t2 = t0 + budget_s * 0.65
    h2, _ = sa_binary(h1, t1, t2)
    try_update(h2)

    # Phase 3: Continuous optimization (15% time)
    if best_h is not None:
        t3 = t0 + budget_s * 0.80
        h_cont = cont_optimize(best_h, len(best_h), t3)
        try_update(h_cont)

    # Phase 4: Different starting points (10% time)
    t4 = t0 + budget_s * 0.90
    n_base = len(h1)
    for s in range(10, 25):
        if time.time() > t4:
            break
        t_sub = min(t4, time.time() + (t4 - time.time()) / max(1, 25 - s))
        h_r = random_binary(n_base, seed + s)
        h_r, _ = hill_climb(h_r, t_sub, samples=30)
        try_update(h_r)

    # Phase 5: Resolution expansion (5% time)
    if best_h is not None:
        t5 = t0 + budget_s * 0.95
        for ne in [len(best_h) * 2, len(best_h) + 14]:
            if time.time() > t5:
                break
            h_e = expand_res(best_h, ne)
            h_e, _ = hill_climb(h_e, t5, samples=25)
            try_update(h_e)

    # Phase 6: Final polish (last 5%)
    if best_h is not None:
        t6 = t0 + budget_s - 2
        best_h, _ = hill_climb(best_h, t6, samples=50)
        try_update(best_h)

    if best_h is None:
        best_h = h_init
    best_c5 = c5_val(best_h)
    return best_h, best_c5, len(best_h)
