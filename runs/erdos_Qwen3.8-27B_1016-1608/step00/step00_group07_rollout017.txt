Now, the constraint sum(h) = n/2 is the issue. Coordinate descent doesn't naturally preserve this. I need to handle it.

Option 3 is cleanest but more complex. Let me go with option 1: after each full sweep, rescale to satisfy the sum constraint.

Actually, rescaling might push values outside [0,1]. Let me use a different projection:

- If sum < n/2, increase the smallest values

Or better: solve for the projection onto {h: 0≤h≤1, sum(h)=n/2}. This is a water-filling problem.

Actually, the simplest approach: after coordinate descent, if the sum is wrong, do a linear correction:
h_new = h + (n/2 - sum(h))/n

Then clip to [0,1] and repeat until the sum is correct. This converges quickly.

    corr = np.correlate(h, comp, mode='full')
    return np.max(corr) * dx

def project_sum(h, target_sum):
    """Project h onto {x: 0<=x<=1, sum(x)=target_sum}"""
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            avail = mask.sum()
            if avail == 0:
                break
            h[mask] += diff / avail
            h = np.minimum(h, 1.0)
        else:
            mask = h > 0.0
            avail = mask.sum()
            if avail == 0:
                break
            h[mask] -= (-diff) / avail
            h = np.maximum(h, 0.0)
    return h

    n = len(h)
    dx = 2.0 / n
    best_c5 = compute_c5(h)

            best_t = h[i]

                h[i] = t
                c5_val = compute_c5(h)

        h = project_sum(h, n * 0.5)

```python
def coord_step(h, i):
    """Find optimal h[i] given all other coordinates fixed"""
    n = len(h)
    dx = 2.0 / n
    old_val = h[i]
    
    # For each shift d (from -(n-1) to (n-1)):
    # c[d+(n-1)] contains terms h[j]*(1-h[j+d]) where j+d is valid
    # The terms involving h[i] are:
    #   j = i: h[i]*(1-h[i+d]) if i+d in [0,n)
    #   j+d = i: h[j]*(1-h[i]) if j = i-d in [0,n)
    
    # So c[d+(n-1)] = (contributions not involving h[i]) + h[i]*(1-h[i+d])_{if valid} + h[i-d]*(1-h[i])_{if valid}
    
    # Linear in h[i]: coefficient = (1-h[i+d])_{if valid} - (h[i-d])_{if valid}
    #                        constant = h[i-d]*(1-h[i])... wait no
    #                          h[i-d]*(1-h[i]) = h[i-d] - h[i-d]*h[i]
    # So coefficient of h[i] from this term: -h[i-d]_{if valid}
    # And from h[i]*(1-h[i+d]): coefficient is (1-h[i+d])_{if valid}
    
    # Total coefficient of h[i] in c[d+(n-1)]:
    #   a_d = (1-h[i+d])_{if i+d valid} - h[i-d]_{if i-d valid}
    # Constant (not involving h[i]):
    #   Everything else in c[d+(n-1)] except the two terms above
    #   b_d = c[d+(n-1)] - h[i]*a_d (computed with old h[i])
    
    # Actually, simpler: compute c for old h, then for new h[i]=t:
    # c_new[d+(n-1)] = c_old[d+(n-1)] + (t - old_val) * a_d
    
    # So f(t) = max_d (c_old[d+(n-1)] + (t - old_val) * a_d) * dx
    #        = max_d ((c_old[d+(n-1)] - old_val * a_d) + t * a_d) * dx
    #        = max_d (B_d + t * a_d) * dx
    
    # where B_d = c_old[d+(n-1)] - old_val * a_d
    
    # Compute correlations for current h
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full')  # length 2n-1
    
    # For index i, a_d for each d:
    # d ranges from -(n-1) to (n-1), index in corr is d + (n-1)
    a = np.zeros(2*n - 1)
    for d in range(-(n-1), n):
        idx = d + (n - 1)
        coeff = 0.0
        const = 0.0
        # Term h[i]*(1-h[i+d]): valid if 0 <= i+d < n
        if 0 <= i + d < n:
            coeff += (1.0 - h[i + d])
        # Term h[i-d]*(1-h[i]): valid if 0 <= i-d < n
        if 0 <= i - d < n:
            coeff -= h[i - d]
        a[idx] = coeff
    
    # B_d = corr[idx] - old_val * a[idx]
    B = corr - old_val * a
    
    # f(t) = max(B + t * a) * dx
    # Minimize over t in [0, 1]
    
    # Evaluate on a grid
    t_grid = np.linspace(0, 1, 500)
    # f(t) for all t: max over d of (B_d + t_d * a_d)
    # This is (B + outer(t_grid, a)) max along axis=1
    vals = (B[np.newaxis, :] + t_grid[:, np.newaxis] * a[np.newaxis, :]).max(axis=1) * dx
    
    best_idx = np.argmin(vals)
    best_t = t_grid[best_idx]
    best_val = vals[best_idx]
    
    return best_t, best_val
```

This is much more efficient! Per coordinate: O(n) to compute a, O(n) to compute B, O(500*n) to evaluate on grid. Total per coordinate: O(n) with small constant. Per sweep: O(n²) ≈ 6400 for n=80. Very fast!

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_sum(h, target):
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-12:
            break
        if d > 0:
            m = h < 1.0
            if m.sum() == 0: break
            h[m] = np.minimum(h[m] + d / m.sum(), 1.0)
        else:
            m = h > 0.0
            if m.sum() == 0: break
            h[m] = np.maximum(h[m] - (-d) / m.sum(), 0.0)
    return h

def cd_step(h, i):
    n = len(h)
    dx = 2.0 / n
    old = h[i]
    corr = np.correlate(h, 1.0 - h, mode='full')
    a = np.zeros(2 * n - 1)
    for d in range(-(n - 1), n):
        idx = d + n - 1
        c = 0.0
        if 0 <= i + d < n:
            c += 1.0 - h[i + d]
        if 0 <= i - d < n:
            c -= h[i - d]

    t = np.linspace(0.0, 1.0, 401)

            h[m] = np.maximum(h[m] + d / cnt, 0.0)

    dx = 2.0 / n
    old = h[i]
    corr = np.correlate(h, 1.0 - h, mode='full')
    a = np.zeros(2 * n - 1)
    for d in range(-(n - 1), n):
        idx = d + n - 1
        c = 0.0

        if 0 <= i - d < n:
