                nc = float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)
                h[i], h[j] = 1.0, 0.0
                if nc < best_nc:
                    best_nc, best_ij = nc, (i, j)
        if best_ij is None:
            break
        h[best_ij[0]], h[best_ij[1]] = 0.0, 1.0
        c5 = best_nc
    return h, c5

def _anneal(h, dx, rng, tlimit):
    c5 = _c5(h)
    bc5, bh = c5, h.copy()
    n_iter_est = int(tlimit * 5000)
    cool = (1e-4 / 0.08) ** (1.0 / max(1, n_iter_est))
    temp = 0.08
    t0 = time.time()
    while time.time() - t0 < tlimit:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        h[i], h[j] = 0.0, 1.0
        nc = _c5(h)
        d = nc - c5
        if d < 0 or (temp > 1e-8 and rng.random() < np.exp(-d / temp)):
            c5 = nc
            if nc < bc5:
                bc5, bh = nc, h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
        temp = max(1e-4, temp * cool)
    return bh, bc5

def _kick(h, rng, k):
    h = h.copy()
    for _ in range(k):
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i != j:
            h[i], h[j] = 0.0, 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    dl = t_start + budget_s
    rng = np.random.RandomState(seed)
    gb_h = None
    gb_c5 = 1e9
    
    n_list = [42, 44, 46, 48, 50, 52, 56, 60, 64, 72, 80, 96]
    
    for n in n_list:
        if time.time() > dl - 12:
            break
        dx = 2.0 / n
        hf = n // 2
        inits = []
        
        ghv = globals().get('initial_h_values')
        if ghv is not None and len(ghv) == n:
            v = np.array(ghv, dtype=float)
            v = (v > 0.5).astype(float)
            while v.sum() > hf:
                idx = np.where(v == 1)[0]
                v[idx[rng.randint(len(idx))]] = 0.0
            while v.sum() < hf:
                idx = np.where(v == 0)[0]
                v[idx[rng.randint(len(idx))]] = 1.0
            inits.append(v)
        
        # Spread
        v = np.zeros(n)
        for k in range(hf):
            v[(k * n + hf // 2) // hf] = 1.0
        inits.append(v)
        
        # Random
        n_rand = max(2, min(5, int((dl - time.time()) / 30)))
        for _ in range(n_rand):
            v = np.zeros(n)
            v[rng.choice(n, hf, replace=False)] = 1.0
            inits.append(v)
        
        for v0 in inits:
            if time.time() > dl - 8:
                break
            rem = dl - time.time()
            if rem < 6:
                break
            h = v0.copy()
            
            # Phase 1: Hill climb
            hc_t = min(30.0, rem * 0.3)
            h, c5 = _hclimb(h, dx, rng, hc_t)
            if c5 < gb_c5:
                gb_c5, gb_h = c5, h.copy()
            
            # Phase 2: Simulated annealing
            sa_t = min(30.0, rem * 0.35)
            if sa_t > 3 and time.time() < dl - 6:
                h2, c5s = _anneal(h.copy(), dx, rng, sa_t)
                if c5s < gb_c5:
                    gb_c5, gb_h = c5s, h2.copy()
                # Phase 3: Re-hill-climb from SA
                if time.time() < dl - 5:
                    rt = min(15.0, dl - time.time() - 3)
                    h3, c5h = _hclimb(h2.copy(), dx, rng, rt)
                    if c5h < gb_c5:
                        gb_c5, gb_h = c5h, h3.copy()
            
            # Phase 4: Iterated local search (kick + re-climb)
            if time.time() < dl - 5:
                for _ in range(3):
                    if time.time() > dl - 4:
                        break
                    h4 = _kick(h, rng, 2)
                    rt = min(15.0, dl - time.time() - 3)
                    if rt < 2:
                        break
                    h5, c5k = _hclimb(h4, dx, rng, rt)
                    if c5k < gb_c5:
                        gb_c5, gb_h = c5k, h5.copy()
