                a = min(h[idx], diff)
                h[idx] -= a
                diff -= a
                if diff < 1e-10:
                    break

    h = np.where(h > 0.5, 1.0, 0.0).astype(float)
    best_h = h.copy()
    best_score = compute_c5(h)
    t0 = time.time()
    stall = 0

        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        bi, bj = -1, -1
        for i in ones:
            hi = h[i]
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                s = compute_c5(h)
                h[i] = hi
                h[j] = 0.0
                if s < best_s - 1e-12:
                    best_s = s
                    bi, bj = i, j
        if bi >= 0:
            h[bi] = 0.0
            h[bj] = 1.0
            best_score = best_s
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall >= 3 and stall % 3 == 0:
                # Random kick
                for _ in range(min(3, len(ones))):

                    jj = zeros[rng.randint(len(zeros))]

                s = compute_c5(h)
                if s > best_score + 0.05:
                    h = best_h.copy()
                else:
                    best_h = h.copy()
                    best_score = s
            if stall > 200:
                break
    return best_h, best_score

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def repair_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                a = min(1.0 - h[idx], diff)
                h[idx] += a
                diff -= a
                if diff < 1e-10:
                    break
        else:
            diff = -diff
            order = np.argsort(-h)
            for idx in order:
                a = min(h[idx], diff)
                h[idx] -= a
                diff -= a
                if diff < 1e-10:
                    break
    return np.clip(h, 0.0, 1.0)
