```python
import numpy as np
from scipy import optimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def c5_all_lags(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def grad_single_lag(h, m):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if m >= 0:
        for j in range(n):
            if m <= j <= n-1:
                grad[j] += (1.0 - h[j-m])
            if 0 <= j <= n-1-m:
                grad[j] -= h[j+m]
    else:
        p = -m
        for j in range(n):
            if 0 <= j <= n-1-p:
                grad[j] += (1.0 - h[j+p])
            if p <= j <= n-1:
                grad[j] -= h[j-p]
    return dx * grad

def grad_c5(h):
    n = len(h)
    all_lags = c5_all_lags(h)
    max_val = np.max(all_lags)
    active = list(np.where(np.isclose(all_lags, max_val, atol=1e-10))[0])
    if len(active) == 0:
        active = [int(np.argmax(all_lags))]
    grad = np.zeros(n)
    for idx in active:
        m = idx - (n - 1)
        grad += grad_single_lag(h, m)
    grad /= len(active)
    return grad

def project(h, n):
    """Project h onto {0 ≤ h ≤ 1, sum(h) = n/2}."""
    h = np.clip(h, 0, 1)
    # Adjust to satisfy sum constraint
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        # Distribute the difference
        if diff > 0:
            # Need to increase sum; add to elements < 1
            room = 1.0 - h
            total_room = np.sum(room)
            if total_room > 0:
                h += diff * room / total_room
        else:
            # Need to decrease sum; subtract from elements > 0
            room = h.copy()
            total_room = np.sum(room)
            if total_room > 0:
                h += diff * room / total_room
    h = np.clip(h, 0, 1)
    # Final correction (might not be exact due to clipping)
    # Use a more careful approach: water-filling
    h = np.clip(h, 0, 1)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        # Add diff uniformly, then clip
        h += diff / n
        h = np.clip(h, 0, 1)
    return h
```

The feasible set is the intersection of the box [0,1]^n and the hyperplane sum(h) = n/2. This is a (n-1)-dimensional polytope.

A proper projection would be:
1. Find λ such that sum(clip(h + λ, 0, 1)) = n/2
2. The projection is clip(h + λ, 0, 1)

This is because we're projecting onto {sum(h) = n/2} ∩ [0,1]^n, and the projection onto the affine set {sum(h) = n/2} is h + λ*1, then we intersect with [0,1]^n via clipping. But this isn't quite right in general...

Actually, for the intersection of a box and a hyperplane, the Euclidean projection can be found by bisection on λ:
- min_λ ||h + λ*1 - p||² s.t. 0 ≤ p ≤ 1, sum(p) = n/2
- By KKT, p_i = clip(h_i + λ, 0, 1) for some λ
- Find λ by bisection such that sum(clip(h + λ, 0, 1)) = n/2

```python
def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -h.max() - 1, -h.min() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h + mid, 0, 1))
        if s < target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h + (lo + hi) / 2, 0, 1)
    # Tiny correction for floating point
    diff = target - np.sum(h_proj)
    if abs(diff) > 1e-14:
        h_proj += diff / n
        h_proj = np.clip(h_proj, 0, 1)
    return h_proj
```

```python
def optimize_from(h0, n, max_iters=5000, lr_init=0.01):
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    lr = lr_init
    for it in range(max_iters):
        c5_val = evaluate_c5(h)
        if c5_val < best_c5:
            best_c5 = c5_val
        g = grad_c5(h)
        h_new = h - lr * g
        h_new = project_feasible(h_new, n)
        # Line search: reduce lr if objective increases
        c5_new = evaluate_c5(h_new)
        if c5_new > c5_val * 1.001 + 1e-10:
            lr *= 0.5
            continue
        h = h_new
        lr = min(lr * 1.05, 0.1)
    return h, best_c5
```

```python
    return best_h, best

def coord_polish(h, n, time_limit=20.0, rng=None):
    t0 = time.time()
    best = evaluate_c5(h)
    best_h = h.copy()
    n_cands = 5
    for sweep in range(200):
        if time.time() - t0 > time_limit:
            break
        improved = False
        for i in range(n):
            if time.time() - t0 > time_limit:
                break
            old = h[i]
            cands = np.linspace(0.0, 1.0, n_cands)
            best_val = evaluate_c5(h)
            best_c = old
            for c in cands:
                if abs(c - old) < 1e-6:
                    continue
                h[i] = c
                h_tmp = project_feasible(h, n)
                v = evaluate_c5(h_tmp)
                if v < best_val:
                    best_val = v
                    best_c = c
            h[i] = best_c
            h = project_feasible(h, n)
            if best_val < best - 1e-12:
                best = best_val
                best_h = h.copy()
                improved = True
        if not improved:
            break
    return best_h, best

def make_block_init(n, num_blocks, rng):
    h = np.zeros(n)
    block_size = n // num_blocks
    for b in range(0, num_blocks, 2):
        start = b * block_size
        end = min(start + block_size, n)
        h[start:end] = 1.0
    h = project_feasible(h, n)
    if rng is not None:
        h = h + rng.randn(n) * 0.1
        h = project_feasible(h, n)
    return h

def make_random_init(n, rng):
    h = rng.rand(n)
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    try:
        n = len(initial_h_values)
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        n = 95
        h_init = np.full(n, 0.5)

    best_h = h_init.copy()
    best_c5 = evaluate_c5(h_init)

    def elapsed():
        return time.time() - t_start

    def remaining():
        return budget_s - 5 - elapsed()

    # Phase 1: Optimize from initial
    t1 = min(remaining() * 0.3, 250.0)
    h, c5 = projected_gradient(h_init, n, max_iters=20000, lr0=0.004, time_limit=t1, rng=rng)
    if c5 < best_c5:
        best_c5, best_h = c5, h

    if remaining() < 10:
        return best_h, best_c5, n

    # Phase 2: Coordinate polish on current best
    t2 = min(remaining() * 0.15, 60.0)
    h, c5 = coord_polish(best_h, n, time_limit=t2, rng=rng)
    if c5 < best_c5:
        best_c5, best_h = c5, h

    if remaining() < 10:
        return best_h, best_c5, n

    # Phase 3: Perturbations of best
    n_pert = 5
    t_per = max(3.0, remaining() * 0.35 / n_pert)
    for i in range(n_pert):
        if remaining() < 15:
            break
        scale = 0.03 + 0.04 * (i % 3)
        pert = best_h + rng.randn(n) * scale
        pert = project_feasible(pert, n)
        h, c5 = projected_gradient(pert, n, max_iters=5000, lr0=0.002, time_limit=t_per, rng=rng)
        if c5 < best_c5:
            best_c5, best_h = c5, h
            h, c5 = coord_polish(best_h, n, time_limit=min(15.0, remaining() * 0.1), rng=rng)
            if c5 < best_c5:
                best_c5, best_h = c5, h

    if remaining() < 10:
        return best_h, best_c5, n

    # Phase 4: Different structured inits
    t_per = max(3.0, remaining() * 0.2 / 4)
    inits = [
        make_block_init(n, 2, rng),
        make_block_init(n, 4, rng),
        make_random_init(n, rng),
        make_random_init(n, rng),
    ]
    for h0 in inits:
        if remaining() < 15:
            break
        h, c5 = projected_gradient(h0, n, max_iters=8000, lr0=0.003, time_limit=t_per, rng=rng)
        if c5 < best_c5:
            best_c5, best_h = c5, h

    # Phase 5: Final polish
    if remaining() > 10:
        t5 = remaining() * 0.5
        h, c5 = projected_gradient(best_h, n, max_iters=20000, lr0=0.001, time_limit=t5, rng=rng)
        if c5 < best_c5:
            best_c5, best_h = c5, h
        if remaining() > 5:
            h, c5 = coord_polish(best_h, n, time_limit=min(20.0, remaining()), rng=rng)
            if c5 < best_c5:
                best_c5, best_h = c5, h

    # Final: try a tiny bit more with very small steps
    if remaining() > 5:
        h, c5 = projected_gradient(best_h, n, max_iters=5000, lr0=0.0005, time_limit=min(10.0, remaining() - 2), rng=rng)
        if c5 < best_c5:
            best_c5, best_h = c5, h

    return best_h, best_c5, n
```
