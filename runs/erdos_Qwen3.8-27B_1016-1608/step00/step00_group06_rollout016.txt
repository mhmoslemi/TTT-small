        hi1 = min(n, n + m)
        if lo1 < hi1:
            grad[lo1:hi1] += wk * (1.0 - h[lo1 - m:hi1 - m])
        lo2 = max(0, -m)
        hi2 = min(n, n - m)
        if lo2 < hi2:
            grad[lo2:hi2] -= wk * h[lo2 + m:hi2 + m]
    grad *= dx
    return grad, cmax + temperature * np.log(np.sum(exp_c) / len(corr))

def optimize_h(h_init, n, deadline, seed=0):
    """Projected gradient descent with smooth max, annealing."""
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    target_sum = n / 2.0
    h = project_box_sum(h_init.astype(float).copy(), target_sum)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = 0.08
    temp = 0.04
    no_improve = 0
    stall_count = 0
    for it in range(20000):
        if time.time() > deadline - 15:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-10:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
            stall_count = 0
        else:
            no_improve += 1
        if no_improve > 150:
            temp = min(temp * 1.4, 0.5)
            lr *= 0.65
            no_improve = 0
            stall_count += 1
            if stall_count > 8:
                # Perturbation to escape local min
                idx = rng.choice(n, size=max(3, n // 15), replace=False)
                h[idx] = np.clip(h[idx] + rng.uniform(-0.3, 0.3, len(idx)), 0, 1)
                h = project_box_sum(h, target_sum)
                stall_count = 0
                no_improve = 0
        if lr < 1e-6:
            break
        grad, _ = smooth_gradient(h, temp)
        h = h - lr * grad
        h = project_box_sum(h, target_sum)
    return best_h, best_c5

def interpolate_to(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0, 2.0, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s
    rng = np.random.RandomState(seed)
    
    # Get initial construction
    try:
        h_parent = np.array(initial_h_values, dtype=float)
    except:
        h_parent = None
    
    if h_parent is None:
        n_parent = 95
        h_parent = np.ones(n_parent)
        h_parent[::2] = 0.0
        h_parent = project_box_sum(h_parent, n_parent / 2.0)
    
    n_parent = len(h_parent)
    
    # Evaluate parent
    h_best = h_parent.copy()
    h_best = project_box_sum(h_best, n_parent / 2.0)
    c5_best = compute_c5(h_best)
    
    # Try a range of sample sizes
    n_candidates = sorted(set([n_parent, 
                                max(30, n_parent - 20),
                                max(30, n_parent - 10),
                                max(40, n_parent + 10),
                                max(50, n_parent + 20),
                                60, 80, 100, 120, 150]))
    
    for n_try in n_candidates:
        if time.time() > deadline - 60:
            break
        if n_try < 20 or n_try > 200:
            continue
        # Interpolate parent to this size
        if n_try == n_parent:
            h_try = h_parent.astype(float).copy()
        else:
            h_try = interpolate_to(h_parent.astype(float), n_parent, n_try)
        h_try = project_box_sum(h_try, n_try / 2.0)
        
        # Optimize
        h_opt, c5_opt = optimize_h(h_try, n_try, deadline, seed=seed + n_try)
        
        if c5_opt < c5_best:
            c5_best = c5_opt
            h_best = h_opt.copy()
        
        # Also try a random binary start at this size
        if time.time() > deadline - 60:
            break
        h_bin = (rng.rand(n_try) < 0.5).astype(float)
        # Fix to exact count
        target = n_try // 2
        ones = np.where(h_bin > 0.5)[0]
        zeros = np.where(h_bin == 0)[0]
        if len(ones) > target:
            to_flip = rng.choice(ones, len(ones) - target, replace=False)
            h_bin[to_flip] = 0.0
        elif len(ones) < target:
            to_flip = rng.choice(zeros, target - len(ones), replace=False)
            h_bin[to_flip] = 1.0
        h_try2 = h_bin.copy()
        h_opt2, c5_opt2 = optimize_h(h_try2, n_try, deadline, seed=seed + n_try + 1000)
        
        if c5_opt2 < c5_best:
            c5_best = c5_opt2
            h_best = h_opt2.copy()
    
    # Final polish: one more optimization round on the best
    if time.time() < deadline - 30:
        n_final = len(h_best)
        h_polish, c5_polish = optimize_h(h_best, n_final, deadline, seed=seed + 9999)
        if c5_polish < c5_best:
            c5_best = c5_polish
            h_best = h_polish
    
    # Ensure feasibility
    h_best = project_box_sum(h_best, len(h_best) / 2.0)
    c5_final = compute_c5(h_best)
    
    return (h_best, c5_final, len(h_best))
