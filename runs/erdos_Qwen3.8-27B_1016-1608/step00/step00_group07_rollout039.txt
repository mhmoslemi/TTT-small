def gradient_single_shift(h, k, n):
    """Gradient of raw_corr[k] w.r.t. h."""
    s = k - n + 1
    grad = np.zeros(n)

    p_lo1 = max(0, s)

    if p_hi2 > p_lo2:

            hi = mid

        if total_room > 0:

            proj[idx] += adj

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float((corr * dx).max())

def gradient_at_max_shift(h, n):
    """Gradient of c5 at the current argmax shift."""
    raw_corr = np.correlate(h, 1.0 - h, mode="full")
    k = np.argmax(raw_corr)
    s = k - n + 1
    dx = 2.0 / n
    grad = np.zeros(n)
    p1lo, p1hi = max(0, s), min(n, n + s)
    if p1hi > p1lo:
        grad[p1lo:p1hi] += (1.0 - h[p1lo-s:p1hi-s])
    p2lo, p2hi = max(0, -s), min(n, n - s)
    if p2hi > p2lo:
        grad[p2lo:p2hi] += -h[p2lo+s:p2hi+s]
    return grad * dx, k

def project_bounded_simplex(v, target):
    n = len(v)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        p = np.clip(v - mid, 0.0, 1.0)
        if p.sum() > target:
            lo = mid
        else:
            hi = mid
    p = np.clip(v - (lo+hi)*0.5, 0.0, 1.0)
    diff = target - p.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.where(p < 1.0 - 1e-12)[0]
        else:
            idx = np.where(p > 1e-12)[0]
        if len(idx) > 0:
            p[idx] += diff / len(idx)
            p = np.clip(p, 0.0, 1.0)
    return p

    h = h_init.copy().astype(float)

    # Phase 1: LSE subgradient at initial resolution
    e1 = t0 + budget_s * 0.33
    h1, c1 = _lse_opt(h0, e1)
    if c1 < bc:
        bh, bc = h1, c1

    # Phase 2: Local search
    e2 = t0 + budget_s * 0.50
    h2, c2 = _local(bh, e2, rng)
    if c2 < bc:
        bh, bc = h2, c2

    # Phase 3: Perturb and local search (escape local minima)
    if time.time() < t0 + budget_s * 0.60:
        for trial in range(3):
            e3 = min(t0 + budget_s * 0.60, time.time() + 25.0)
            hp = _proj(bh + rng.randn(bn) * 0.03, bn * 0.5)
            h3, c3 = _local(hp, e3, rng)
            if c3 < bc:
                bh, bc = h3, c3

    # Phase 4: Upscale and re-optimize
    for mult in (2, 3):
        n2 = bn * mult
        if n2 > 600 or time.time() > t0 + budget_s * 0.80:
            break
        e4 = min(t0 + budget_s * 0.75, time.time() + 80.0)
        hu = _interp(bh, n2)
        h4, c4 = _lse_opt(hu, e4)
        e5 = min(t0 + budget_s * 0.80, time.time() + 40.0)
        h4b, c4b = _local(h4, e5, rng)
        if c4b < bc:
            bh, bc, bn = h4b, c4b, n2

    # Phase 5: Final local search
    e6 = t0 + budget_s * 0.97
    h6, c6 = _local(bh, e6, rng)
    if c6 < bc:
        bh, bc = h6, c6

    bh = _proj(bh, bn * 0.5)
    return bh, evaluate_c5(bh), bn
