def c5_score(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def c5_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    """Ensure h is binary with exactly n/2 ones"""
    h = np.array(h, dtype=float)
    h = (h > 0.5).astype(float)
    target = n // 2
    ones = np.flatnonzero(h)
    zeros = np.flatnonzero(1 - h)
    while len(ones) > target:
        idx = ones[np.random.randint(len(ones))]
        h[idx] = 0
        ones = np.flatnonzero(h)
    while len(ones) < target:
        idx = zeros[np.random.randint(len(zeros))]
        h[idx] = 1
        ones = np.flatnonzero(h)
    return h

def local_search_binary(h, n, time_limit):
    """Exhaustive local search for binary h"""
    best = h.copy()
    best_score = c5_score(best)
    deadline = time.time() + time_limit
    no_improve_streak = 0
    
    while time.time() < deadline:
        ones = np.where(best == 1)[0]
        zeros = np.where(best == 0)[0]
        
        best_delta = best_score
        best_move = None
        
        for i in ones:

            best[i] = 0
            best[j] = 1
            best_score = best_delta

Let me define helper arrays:
For each position p and each lag k (or index idx):
- A[p, idx] = h[p + k] if 0 ≤ p+k < n else 0 (this is the "forward" contribution)
- B[p, idx] = (1 - h[p - k]) if 0 ≤ p-k < n else 0 (this is the "backward" contribution)

Then Δc[idx] for swap (i→0, j→1):
= A[i, idx] - B[i, idx] - A[j, idx] + B[j, idx]

I can precompute A and B as n×(2n-1) matrices. Then for each swap, Δc = A[i] - B[i] - A[j] + B[j], and the new max is max(c + Δc).

import numpy as np
import time

def c5_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n, rng):
    h = np.array(h, dtype=float).copy()
    if len(h) != n:
        h = np.zeros(n)
    h = (h > 0.5).astype(float)
    target = n // 2
    s = h.sum()
    if s > target:
        ones = np.flatnonzero(h)
        order = rng.permutation(len(ones))
        for idx in order[:int(s - target)]:
            h[ones[idx]] = 0
    elif s < target:
        zeros = np.flatnonzero(1 - h)
        order = rng.permutation(len(zeros))
        for idx in order[:int(target - s)]:
            h[zeros[idx]] = 1
    return h

def fast_local_search(h, n, deadline, rng):
    h = h.copy()
    best_s = c5_score(h)
    ncorr = 2 * n - 1
    dx = 2.0 / n
    k_range = np.arange(-n + 1, n)
    
    iter_count = 0
    stall = 0
    
    while time.time() < deadline:
        # Build A and B matrices
        A = np.zeros((n, ncorr))
        B = np.zeros((n, ncorr))
        for p in range(n):
            ip = p + k_range
            jp = p - k_range
            mask_a = (ip >= 0) & (ip < n)
            mask_b = (jp >= 0) & (jp < n)
            A[p, mask_a] = h[ip[mask_a]]
            B[p, mask_b] = 1.0 - h[jp[mask_b]]
        
        curr = np.correlate(h, 1.0 - h, mode='full')
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Vectorized neighbor evaluation
        delta = (A[ones][:, None, :] - B[ones][:, None, :] 
                 - A[zeros][None, :, :] + B[zeros][None, :, :])
        new_max = (curr[None, None, :] + delta).max(axis=2)
        
        best_idx = new_max.argmin()
        best_row = best_idx // len(zeros)
        best_col = best_idx % len(zeros)
        
        if new_max[best_row, best_col] * dx < best_s - 1e-12:
            i = ones[best_row]
            j = zeros[best_col]
            h[i] = 0.0
            h[j] = 1.0
            best_s = new_max[best_row, best_col] * dx
            stall = 0
        else:
            stall += 1
            # Perturbation
            n_perturb = min(stall, 5)
            for _ in range(n_perturb):
                ones2 = np.where(h == 1)[0]
                zeros2 = np.where(h == 0)[0]
                if len(ones2) == 0 or len(zeros2) == 0:
                    break
                i = ones2[rng.integers(len(ones2))]
                j = zeros2[rng.integers(len(zeros2))]
                h[i] = 0.0
                h[j] = 1.0
            best_s = c5_score(h)
        
        iter_count += 1
    
    return h, best_s

def frac_refine(h, n, t_start, deadline, rng):
    """Project subgradient for non-binary refinement"""
    h = h.astype(float).copy()
    best_s = c5_score(h)
    dx = 2.0 / n
    ncorr = 2 * n - 1
    
    lr = 0.01
    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx_max = np.argmax(corr)
        # Gradient of corr[idx] w.r.t. h
        # corr[idx] = sum_i h[i+k]*(1-h[i]) where k = idx - (n-1)
        k = idx_max - (n - 1)
        grad = np.zeros(n)
        # d/dh[p] of corr[idx] = -h[p-k] + (1-h[p+k]) (for valid indices)
        for p in range(n):
            pk = p + k
            pmk = p - k
            val = 0.0
            if 0 <= pk < n:
                val += (1.0 - h[pk])
            if 0 <= pmk < n:
                val -= h[pmk]
            grad[p] = val
        
        # Subgradient step: h -= lr * grad (to reduce the max component)
        h_new = h - lr * grad
        # Project to [0, 1] and fix sum
        h_new = np.clip(h_new, 0, 1)
        # Fix sum to n/2
        target = n / 2.0
        diff = target - h_new.sum()
        if abs(diff) > 1e-10:
            h_new += diff / n
            h_new = np.clip(h_new, 0, 1)
            # If still off, adjust individual elements
            while abs(h_new.sum() - target) > 1e-6:
                diff = target - h_new.sum()
                if diff > 0:
                    idx_add = np.where(h_new < 1.0)[0]
                else:
                    idx_add = np.where(h_new > 0.0)[0]
                if len(idx_add) == 0:
                    break
                p = idx_add[rng.integers(len(idx_add))]
                adj = min(abs(diff), abs(1.0 - h_new[p]) if diff < 0 else (1.0 - h_new[p]) if diff > 0 else 0)
                # Simpler: just add a small amount
                step = np.sign(diff) * min(abs(diff) / len(idx_add), 0.01)
                h_new[p] += step
                h_new[p] = np.clip(h_new[p], 0, 1)
        
        s_new = c5_score(h_new)
        if s_new < best_s:
            h = h_new
            best_s = s_new
            lr = max(lr * 0.95, 0.001)
        else:
            lr = max(lr * 0.7, 0.001)
    
    return h, c5_score(h)

            # Try 2-swap if time allows and n is small
            if n <= 60 and time.time() < t_end - 5:
                h2 = h.copy()
                h2, s2 = local_search_2swap(h2, n, min(t_end - 5, time.time() + time_each * 0.3), rng)
                if s2 < best_score:
                    best_score = s2
                    best_h = h2.copy()

        # Reflected init
        if best_h is not None and len(best_h) == n and time.time() < t_end:
            h_ref = 1.0 - best_h
            h_ref, s_ref = local_search(h_ref, n, t_end, rng)
            if s_ref < best_score:
                best_score = s_ref
                best_h = h_ref.copy()

    # Fractional refinement
    remaining = budget_s - (time.time() - t0)
    if remaining > 25 and best_h is not None:
        n = len(best_h)
        t_frac_end = time.time() + min(remaining - 10, budget_s * 0.12)
        h_frac, s_frac = frac_refine(best_h, n, t_frac_end, rng)
        if s_frac < best_score:
            best_score = s_frac
            best_h = h_frac

        # Also try from binary version
        h_bin = init_random(n, rng)
        h_bin, s_bin_init = local_search(h_bin, n, time.time() + 5, rng)
        h_frac2, s_frac2 = frac_refine(h_bin, n, t_frac_end, rng)
        if s_frac2 < best_score:
            best_score = s_frac2
            best_h = h_frac2

    # Final sanity check
    n = len(best_h)
    target = n / 2.0
    diff = target - best_h.sum()
    best_h = np.clip(best_h + diff / n, 0, 1)
    final_score = c5_score(best_h)
    return best_h, final_score, n
