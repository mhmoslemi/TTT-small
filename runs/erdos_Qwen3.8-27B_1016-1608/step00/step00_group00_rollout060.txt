def compute_c5(h):

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def smooth_max(corr, tau):
    # Returns smooth max of corr values
    scaled = tau * corr
    max_val = np.max(scaled)
    return (1.0/tau) * (max_val + np.log(np.sum(np.exp(scaled - max_val))))

def project_feasible(h, target_sum):
    # Project h onto {0<=x<=1, sum(x)=target_sum}
    # Binary search on lambda

        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

    corr = compute_corr(h)
    # Smooth max gradient
    scaled = tau * corr
    max_val = np.max(scaled)
    weights = np.exp(scaled - max_val)
    weights /= np.sum(weights)
    
    # Gradient of corr[k] with respect to h[j]
    # corr[k] = sum_i h[i] * (1-h[i+k-(n-1)]) for valid i

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_grad(corr, tau):
    """Compute smooth max value and gradient weights."""
    scaled = tau * corr
    max_val = np.max(scaled)
    exps = np.exp(scaled - max_val)
    total = np.sum(exps)
    smax = (max_val + np.log(total)) / tau
    weights = exps / total
    return smax, weights

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def grad_step(h, tau, lr):
    n = len(h)
    target = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode="full")
    smax, weights = smooth_max_grad(corr, tau)

        return float(np.max(corr))

    scaled = tau * corr
    mx = np.max(scaled)
    shifted = scaled - mx
    exps = np.exp(shifted)
    s = np.sum(exps)
    smax = (mx + np.log(s)) / tau
    weights = exps / s
    return smax, weights

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_grad(h, tau, weights):

                t2 = np.zeros(n)

            t1[-lag:] = one_m_h[:n+lag]

            t2[:n+lag] = h[-lag:]
