            curr_c5 = c5_new
            if curr_c5 < best_c5:
                best_h = curr_h.copy()
                best_c5 = curr_c5

    h_new = np.interp(x_new, x_old, h)

    for i in range(min(target_int, n)):
        result[order[i]] = 1.0
    if abs(frac) > 1e-10:
        result[order[target_int]] = frac
    return result

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def repair_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s < 1e-15:
        return np.full_like(h, target_sum / len(h))
    h = h * (target_sum / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)

            idx = np.where(h > 0.0)[0]
