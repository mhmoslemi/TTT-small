import numpy as np
import math
import time
import random

def compute_score(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c)) * dx

def hill_climb_step(h, n, dx):
    """Try all single swaps, return best."""
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])
    best_score = compute_score(h, dx)
    best_h = h.copy()
    for i in ones:
        for j in zeros:
            h[i] = 0.0
            h[j] = 1.0
            s = compute_score(h, dx)
            if s < best_score:
                best_score = s
                best_h = h.copy()
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    h0 = np.asarray(initial_h_values, dtype=float)
    n = len(h0)
    dx = 2.0 / n
    h = (h0 >= 0.5).astype(float)
    
    best_score = compute_score(h, dx)
    best_h = h.copy()
    
    # Initial hill climbing
    for _ in range(10):
        h, best_score = hill_climb_step(best_h, n, dx)
        best_h = h
    
    t0 = time.time()
    cur = best_h.copy()
    cur_score = best_score
    
    # Phase 1: SA
    T = 0.05
    alpha = 0.9999995
    T_min = 1e-10
    
    no_improve = 0
    stagnation_limit = 50000
    
    while time.time() - t0 < budget_s - 30:
        ones = np.where(cur == 1.0)[0]
        zeros = np.where(cur == 0.0)[0]
        i = ones[random.randrange(len(ones))]
        j = zeros[random.randrange(len(zeros))]
        
        cur[i] = 0.0
        cur[j] = 1.0
        new_score = compute_score(cur, dx)
        delta = new_score - cur_score
        
        if delta <= 0 or (T > T_min and random.random() < math.exp(-delta / T)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            cur[i] = 1.0
            cur[j] = 0.0
            no_improve += 1
        
        T = max(T * alpha, T_min)
        
        # Restart if stuck
        if no_improve > stagnation_limit:
            cur = best_h.copy()
            cur_score = best_score
            T = 0.05
            no_improve = 0
    
    # Phase 2: Final hill climbing
    time_left = budget_s - (time.time() - t0)
    if time_left > 10:
        for _ in range(20):
            best_h, best_score = hill_climb_step(best_h, n, dx)
