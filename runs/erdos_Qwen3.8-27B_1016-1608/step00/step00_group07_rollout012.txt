import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target):
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    h = h + diff / len(h)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    for _ in range(len(h) * 3):
        if abs(diff) < 1e-12:
            break
        idx = int(np.random.randint(len(h)))
        adj = np.clip(diff, -h[idx], 1.0 - h[idx])
        h[idx] += adj
        diff -= adj
    return h

def sa_search(h, n, target_sum, T_init, T_min, cooling, t_end, rng, track_best=True):
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5
    T = T_init
    while time.time() < t_end:
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            T = max(T * cooling, T_min)
            continue
        scale = 0.01 + 0.6 * T
        delta = rng.uniform(0.005, scale)
        if cur[i] >= cur[j]:
            d = min(delta, cur[i], 1.0 - cur[j])
        else:
            d = min(delta, cur[j], 1.0 - cur[i])
            i, j = j, i
        if d < 1e-9:
            T = max(T * cooling, T_min)
            continue
        new_h = cur.copy()
        new_h[i] -= d
        new_h[j] += d
        nc = compute_c5(new_h)
        dc = nc - cur_c5
        if dc <= 0 or rng.random() < np.exp(-dc / max(T, 1e-15)):
            cur = new_h
            cur_c5 = nc
            if track_best and cur_c5 < best_c5:
                best = cur.copy()
                best_c5 = cur_c5
        T = max(T * cooling, T_min)
    return best, best_c5

def hill_climb(h, target_sum, t_end, rng):
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    n = len(h)
    deltas = [0.005, 0.01, 0.02, 0.05, 0.1]
    while time.time() < t_end:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > t_end:
                break
            for d in deltas:
                for sign in [1, -1]:
                    dd = sign * d
                    if cur[i] + dd < 0 or cur[i] + dd > 1:
                        continue
                    # Find j to compensate
                    j_found = -1
                    for _ in range(20):
                        j = int(rng.integers(n))
                        if j == i:
                            continue
                        if sign > 0 and cur[j] >= d:
                            j_found = j
                            break
                        elif sign < 0 and 1.0 - cur[j] >= d:
                            j_found = j
                            break
                    if j_found < 0:
                        continue
                    new_h = cur.copy()
                    new_h[i] += dd
                    new_h[j_found] -= dd
                    nc = compute_c5(new_h)
                    if nc < cur_c5 - 1e-12:
                        cur = new_h
                        cur_c5 = nc
                        improved = True
                        break
                if improved:
                    break
    return cur, cur_c5
