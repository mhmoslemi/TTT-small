def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def smooth_max_obj(h, dx, n_points, alpha, penalty):
    corr = np.correlate(h, 1.0 - h, mode="full")
    c = corr * dx
    m = np.max(c)
    if alpha * m > 700:
        sm = m
    else:
        sm = m + np.log(np.sum(np.exp(alpha * (c - m)))) / alpha
    s = np.sum(h) - n_points / 2.0
    return sm + penalty * s * s

    h = np.clip(h, 0.0, 1.0)
    diff = n_points / 2.0 - np.sum(h)
    if abs(diff) > 1e-12:
        h += diff / n_points
        h = np.clip(h, 0.0, 1.0)
    return h

    for it in range(max_iter):
        improved = False
        for i in range(n):

        sm = m
    else:
        sm = m + float(np.log(np.sum(np.exp(alpha * (c - m))))) / alpha
    s = float(np.sum(h)) - n_points / 2.0
    return sm + penalty * s * s

def fix_sum(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    diff = n_points / 2.0 - float(np.sum(h))
    if abs(diff) > 1e-14:
        h += diff / n_points
        h = np.clip(h, 0.0, 1.0)
    return h

def hard_max_obj(h, dx, n_points, penalty):
    corr = np.correlate(h, 1.0 - h, mode="full")
    c = corr * dx
    s = float(np.sum(h)) - n_points / 2.0
    return float(np.max(c)) + penalty * s * s

def lbb_optimize(h, dx, n_points, alpha, penalty, maxiter=500):
    try:
        res = minimize(
            smooth_max_obj, h.copy(),
            args=(dx, n_points, alpha, penalty),
            method='L-BFGS-B',
            bounds=[(0.0, 1.0)] * n_points,
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10}
        )
        return fix_sum(res.x, n_points)
    except:
        return h.copy()

def coord_descent(h, dx, n_points, penalty, max_sweeps=20):
    n = n_points
    h = h.copy()
    prev_obj = hard_max_obj(h, dx, n, penalty)
    for sweep in range(max_sweeps):
        improved = False
        for i in range(n):
            old = h[i]
            def sub(v):
                h[i] = v
                return hard_max_obj(h, dx, n, penalty)
            h[i] = old
            try:
                r = minimize_scalar(sub, bounds=(0.0, 1.0), method='bounded',
                                    options={'maxiter': 40, 'xatol': 1e-12})
                h[i] = float(np.clip(r.x, 0.0, 1.0))
            except:
                pass
            if abs(h[i] - old) > 1e-12:
                improved = True
        h = fix_sum(h, n)
        cur_obj = hard_max_obj(h, dx, n, penalty)
        if cur_obj < prev_obj - 1e-14:
            prev_obj = cur_obj
        else:
            if not improved:
                break
    return h

def try_solution(h, dx, n_points, t0, t_limit, max_iter_lbb=400):
    best = h.copy()
    best_c5 = compute_c5(best)
    cur = h.copy()
    for alpha in [5, 10, 20, 40, 80, 160, 300]:
        if time.time() - t0 > t_limit:
            break
        cur = lbb_optimize(cur, dx, n_points, alpha, 200.0, max_iter_lbb)
        c5 = compute_c5(cur)
        if c5 < best_c5:
            best_c5 = c5
            best = cur.copy()
    if time.time() - t0 < t_limit:
        cur = coord_descent(best, dx, n_points, 200.0, max_sweeps=15)
        c5 = compute_c5(cur)
        if c5 < best_c5:
            best_c5 = c5
            best = cur.copy()
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_limit = budget_s - 5.0

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except:
        h0 = np.full(45, 0.5)

    n0 = len(h0)
    dx0 = 2.0 / n0
    h0 = fix_sum(h0, n0)

    # Phase 1: Optimize at original resolution
    best_h, best_c5, best_n = try_solution(h0, dx0, n0, t0, t_limit, max_iter_lbb=500)

    # Phase 2: Upscale and optimize
    if time.time() - t0 < t_limit * 0.55:
        n1 = n0 * 2
        dx1 = 2.0 / n1
        x_old = (np.arange(n0) + 0.5) * dx0
        x_new = (np.arange(n1) + 0.5) * dx1
        h_up = np.interp(x_new, x_old, best_h)
        h_up = fix_sum(h_up, n1)
        h_up2, c5_up = try_solution(h_up, dx1, n1, t0, t_limit, max_iter_lbb=300)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up2
            best_n = n1

    # Phase 3: Try a few random perturbations from best
    if time.time() - t0 < t_limit * 0.4:
        for trial in range(4):
            if time.time() - t0 > t_limit:
                break
            h_pert = best_h.copy()
            n_use = len(h_pert)
            dx_use = 2.0 / n_use
            idx = np.random.choice(n_use, size=max(3, n_use // 8), replace=False)
            h_pert[idx] = np.clip(h_pert[idx] + np.random.uniform(-0.15, 0.15, len(idx)), 0, 1)
            h_pert = fix_sum(h_pert, n_use)
            h_p, c5_p = try_solution(h_pert, dx_use, n_use, t0, t_limit, max_iter_lbb=200)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_p
                best_n = n_use

    # Phase 4: Final polish
    if time.time() - t0 < t_limit:
        n_f = len(best_h)
        dx_f = 2.0 / n_f
        final = coord_descent(best_h, dx_f, n_f, 500.0, max_sweeps=25)
        c5_f = compute_c5(final)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = final
            best_n = n_f

    # Final validation
    best_h = fix_sum(best_h, len(best_h))
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
