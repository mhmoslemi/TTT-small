        if r < 0.30:
            # Small random perturbation
            k = int(np.random.randint(1, 4))
            for _ in range(k):
                i = int(np.random.randint(n))
                h_trial[i] += np.random.randn() * 0.03
        elif r < 0.50:
            # Pair mass transfer
            i, j = np.random.choice(n, size=2, replace=False)
            d = (h_trial[i] - h_trial[j]) * 0.5 + np.random.randn() * 0.02
            h_trial[i] -= d
            h_trial[j] += d
        elif r < 0.70:
            # Gradient-aware: target max-lag contributing pairs
            corrs, _ = compute_c5_values(best_h)
            k_idx = int(np.argmax(corrs))
            lag = k_idx - (n - 1)
            if lag > 0 and lag < n:
                i = int(np.random.randint(n - lag))
                # Move from high to low at the lagged pair
                if best_h[i + lag] > best_h[i]:
                    transfer = min(best_h[i + lag] - best_h[i], 0.05) * np.random.uniform(0.5, 1.0)
                    h_trial[i + lag] -= transfer
                    h_trial[i] += transfer
                else:
                    transfer = min(best_h[i] - best_h[i + lag], 0.05) * np.random.uniform(0.5, 1.0)
                    h_trial[i] -= transfer
                    h_trial[i + lag] += transfer
            else:
                i = int(np.random.randint(n))
                h_trial[i] += np.random.randn() * 0.03
        elif r < 0.85:
            # Neighbor smoothing
            i = int(np.random.randint(1, n - 1))
            delta = (h_trial[i+1] - 2*h_trial[i] + h_trial[i-1]) * 0.2
            h_trial[i] += delta
        else:
            # Binary nudge
            i = int(np.random.randint(n))
            h_trial[i] = 1.0 - h_trial[i]

        h_trial = repair_h(h_trial, n)
        c5_trial = compute_c5_values(h_trial)[1]

        if c5_trial < best_c5:
            best_h = h_trial.copy()
            best_c5 = c5_trial

    # Phase 3: Final SLSQP polish
    if budget_s - (time.time() - t_start) > 30:
        _state['alpha'] = 2000.0
        cons = [{'type': 'eq', 'fun': sum_constraint}]
        bnds = [(0.0, 1.0)] * n
        try:
            res = minimize(smooth_objective, best_h.copy(), method='SLSQP',
                          bounds=bnds, constraints=cons,
                          options={'maxiter': 5000, 'ftol': 1e-15})
            if res.x is not None:
                h_new = repair_h(res.x, n)
                c5_new = compute_c5_values(h_new)[1]
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
        except Exception:
            pass

    # Phase 4: Very short local search polish
    for it2 in range(50000):
        if budget_s - (time.time() - t_start) < 3:
            break
        h_trial = best_h.copy()
        k = int(np.random.randint(1, 3))
        for _ in range(k):
            i = int(np.random.randint(n))
            h_trial[i] += np.random.randn() * 0.01
        h_trial = repair_h(h_trial, n)
        c5_trial = compute_c5_values(h_trial)[1]
        if c5_trial < best_c5:
            best_h = h_trial.copy()
            best_c5 = c5_trial

    best_h = repair_h(best_h, n)
    best_c5 = compute_c5_values(best_h)[1]
    return best_h, best_c5, n
