    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    corr = full_correlation(h)
    score = (corr * dx).max()
    
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h < 0.5)[0])

        best_gain = 0
        best_i = -1
        best_j = -1
