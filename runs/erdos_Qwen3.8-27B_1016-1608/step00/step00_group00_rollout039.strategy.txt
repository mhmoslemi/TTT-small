def grid_upscale(points, n_new):
    # points has current grid size 'n_old'
    n_old = points.shape[0]
    scale = n_new / n_old
    new_data = []
    for i in range(n_old):
        for j in range(int(scale)):
            new_data.append(points[i])
    new_data = np.array(new_data)
    # Normalize to preserve integral since grid spacing changes: dx_new = 2/n_new
    # The interpolation already preserves the integral
    return new_data
