            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            break
        step = diff / mask.sum()
        if diff > 0:
            step = min(step, (1.0 - h[mask]).max())
        else:
            step = max(step, -h[mask].max())
        h[mask] += step
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = int(np.argmin(1.0 - h))
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + diff)

            for i in range(n):

                h_minus[i] = max(0.0, h_curr[i] - eps)
                c5_minus = compute_c5(h_minus, n)
                grad[i] = (c5_plus - c5_minus) / (2 * eps)

        grad /= gnorm
        improved = False
        for trial in [lr, lr * 0.5, lr * 0.2, lr * 0.05]:
            h_new = h_curr - trial * grad
            h_new = repair_h(h_new, n)
            c5_new = compute_c5(h_new, n)
            if c5_new < c5_curr - 1e-12:
                h_curr = h_new
                c5_curr = c5_new
                lr = trial
                improved = True
                if c5_curr < c5_best:
                    c5_best = c5_curr
                    h_best = h_curr.copy()
                break
        if not improved:
            lr *= 0.7
            if lr < 1e-7:
                break
    return h_best, c5_best

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.time()
    time_budget = budget_s - 12
    try:
        n_init = len(initial_h_values)
        h_initial = np.array([float(x) for x in initial_h_values])
    except Exception:
        n_init = 47
        h_initial = np.full(n_init, 0.5)
    h_best = h_initial.copy()
    c5_best = compute_c5(h_best, n_init)
    n_best = n_init
    # Phase 1: SA on initial n
    t1 = time_budget * 0.50
    t_elapsed = time.time() - t_start
    h1, c5_1 = sa_optimize(h_initial, n_init, max(0, t1 - t_elapsed), rng)
    # Phase 2: Gradient polish
    if time.time() - t_start < time_budget:
        h1, c5_1 = gradient_polish(h1, n_init, n_steps=120, lr=0.004)
    if c5_1 < c5_best:
        c5_best, h_best, n_best = c5_1, h1.copy(), n_init
    # Phase 3: Try other n
    other_ns = [48, 50, 56, 64, 80]
    t3_total = time_budget * 0.35
    t_per = t3_total / len(other_ns)
    for n_try in other_ns:
        if time.time() - t_start > time_budget * 0.85:
            break
        t_left = min(t_per, time_budget - (time.time() - t_start) - 5)
        if t_left < 2:
            break
        x_old = np.linspace(0, 2, n_init, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_try = np.interp(x_new, x_old, h_initial)
        h_try = repair_h(h_try, n_try)
        ht, c5t = sa_optimize(h_try, n_try, t_left, rng)
        ht, c5t = gradient_polish(ht, n_try, n_steps=60, lr=0.003)
        if c5t < c5_best:
            c5_best, h_best, n_best = c5t, ht.copy(), n_try
    # Phase 4: Final SA burst on best if time remains
    t_rem = time_budget - (time.time() - t_start)
    if t_rem > 10:
        hb, cb = sa_optimize(h_best, n_best, t_rem * 0.6, rng)
        hb, cb = gradient_polish(hb, n_best, n_steps=50, lr=0.002)
        if cb < c5_best:
            c5_best, h_best, n_best = cb, hb.copy(), n_best
    return h_best, c5_best, n_best
