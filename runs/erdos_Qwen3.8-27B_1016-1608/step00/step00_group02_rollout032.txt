        t_g3 = min(budget_left() * 0.3, 180)
        h3, c5_3 = greedy_local(h3, dx3, t_g3, rng)
        h3 = to_01(h3, n3, rng)
        c5_3 = compute_c5(h3, dx3)
        if c5_3 < best_c5:
            best_c5, best_h, best_n = c5_3, h3.copy(), n3

        if budget_left() > 60:
            h_sa, c5_sa = sa_local(h3, dx3, min(budget_left() * 0.3, 200), seed + 5)
            h_sa = to_01(h_sa, n3, rng)
            c5_sa = compute_c5(h_sa, dx3)
            if c5_sa < best_c5:
                best_c5, best_h, best_n = c5_sa, h_sa.copy(), n3

    # Final polish: SA at best grid
    if budget_left() > 30 and best_h is not None:
        dn = 2.0 / best_n
        t_fin = min(budget_left() * 0.5, 120)
        h_fin, c5_fin = sa_local(best_h, dn, t_fin, seed + 100)
        h_fin = to_01(h_fin, best_n, rng)
        c5_fin = compute_c5(h_fin, dn)
        if c5_fin < best_c5:
            best_c5, best_h, best_n = c5_fin, h_fin.copy(), best_n

    # Ensure final validity
    if best_h is not None:
        dn = 2.0 / best_n
        best_h = to_01(best_h, best_n, rng)
        best_c5 = compute_c5(best_h, dn)

    return best_h, best_c5, best_n
