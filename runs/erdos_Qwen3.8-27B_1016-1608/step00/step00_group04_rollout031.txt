import numpy as np

import math
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_random_binary(n, rng):
    perm = rng.permutation(n)
    h = np.zeros(n, dtype=np.float64)
    h[perm[:n // 2]] = 1.0
    return h
