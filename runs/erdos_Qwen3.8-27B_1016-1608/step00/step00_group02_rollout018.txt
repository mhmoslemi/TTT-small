import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def repair(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    for iteration in range(20):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if not mask.any():
                break
            adj = diff / mask.sum()
            h[mask] = np.minimum(h[mask] + adj, 1.0)
        else:
            mask = h > 0.0
            if not mask.any():
                break
            adj = diff / mask.sum()
            h[mask] = np.maximum(h[mask] + adj, 0.0)
    h = np.clip(h, 0.0, 1.0)
    final_diff = target - h.sum()
    if abs(final_diff) > 1e-13:
        for i in range(n):
            if final_diff > 0 and h[i] < 1.0:
                a = min(final_diff, 1.0 - h[i])
                h[i] += a; final_diff -= a
            elif final_diff < 0 and h[i] > 0.0:
                a = max(final_diff, -h[i])
                h[i] += a; final_diff -= a
            if abs(final_diff) < 1e-14:
                break
    return h

def interp_to(h, new_n):
    old_n = len(h)
    ox = (np.arange(old_n) + 0.5) * 2.0 / old_n
    nx = (np.arange(new_n) + 0.5) * 2.0 / new_n
    return repair(np.interp(nx, ox, h), new_n)
