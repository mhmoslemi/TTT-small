import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = h.sum() - target_sum
    if abs(diff) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if np.clip(h + mid, 0.0, 1.0).sum() > target_sum:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    residual = target_sum - result.sum()
    if abs(residual) > 1e-10:
        mask = (result > 1e-8) & (result < 1.0 - 1e-8)
        if mask.sum() > 0:
            result[mask] += residual / mask.sum()
            result = np.clip(result, 0.0, 1.0)
    return result

def penalized_obj(h, n, target_sum, tau, penalty):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    m = np.max(vals)
    sh = tau * (vals - m)
    sh -= np.max(sh)
    obj = m + np.log(np.sum(np.exp(sh))) / tau
    sv = h.sum() - target_sum
    obj += penalty * sv * sv
    return obj

def corr_gradient(m_idx, h):
    n = len(h)
    k = m_idx - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        nk = n - 1 - k
        for j in range(n):
            if j <= nk:
                grad[j] += 1.0 - h[j + k]
            if j >= k:
                grad[j] -= h[j - k]
    else:
        p = -k
        for j in range(n):
            if j >= p:
                grad[j] += 1.0 - h[j - p]
            if j <= n - 1 - p:
                grad[j] -= h[j + p]
    return grad

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()

    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    target_sum = n / 2.0
    h0 = project_feasible(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    bounds = [(0.0, 1.0)] * n

    def elapsed():
        return time.time() - t_start

    # Phase 1: SLSQP with soft-LSE and penalty, multiple tau
    for tau in [200.0, 100.0, 50.0, 25.0]:
        if elapsed() > budget_s * 0.20:
            break
        for penalty in [200.0, 1000.0]:
            try:
                res = minimize(penalized_obj, best_h,
                              args=(n, target_sum, tau, penalty),
                              method='SLSQP', bounds=bounds,
                              options={'maxiter': 500, 'ftol': 1e-14})
                if res.x is not None:
                    h_new = project_feasible(res.x, target_sum)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5:
                        best_h = h_new
                        best_c5 = c5_new
            except Exception:
                pass

    # Phase 2: Projected subgradient on exact max
    if elapsed() < budget_s * 0.50:
        h = best_h.copy()
        dx = 2.0 / n
        alpha = 0.5 / n
        best_in_phase = best_c5
        for it in range(20000):
            if elapsed() > budget_s * 0.50:
                break
            corr = np.correlate(h, 1.0 - h, mode='full')
            m_idx = int(np.argmax(corr))
            grad = corr_gradient(m_idx, h)
            h_new = project_feasible(h - alpha * dx * grad, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < best_in_phase:
                best_in_phase = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
            h = h_new
            if it % 2000 == 1999:
                alpha *= 0.5
                if alpha < 1e-6:
                    break

    # Phase 3: Multi-start perturbations + SLSQP refinement
    if elapsed() < budget_s * 0.85:
        max_pert = 40
        for trial in range(max_pert):
            if elapsed() > budget_s * 0.85:
                break
            scale = 0.01 + 0.09 * (trial / max_pert)
            h_pert = best_h + scale * np.random.randn(n)
            h_pert = project_feasible(h_pert, target_sum)
            try:
                res = minimize(penalized_obj, h_pert,
                              args=(n, target_sum, 100.0, 1000.0),
                              method='SLSQP', bounds=bounds,
                              options={'maxiter': 150, 'ftol': 1e-11})
                if res.x is not None:
                    h_new = project_feasible(res.x, target_sum)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5:
                        best_h = h_new
                        best_c5 = c5_new
            except Exception:
                pass

    # Phase 4: Try nearby n values via interpolation
    if elapsed() < budget_s * 0.95:
        for n_try in [n + 1, n + 2, n - 1, n + 4, n + 5]:
            if elapsed() > budget_s * 0.95:
                break
            if n_try < 10 or n_try > 500:
                continue
            ts = float(n_try) / 2.0
            x_orig = np.linspace(0.0, 2.0, n)
            x_new = np.linspace(0.0, 2.0, n_try)
            h_interp = np.interp(x_new, x_orig, best_h)
            h_interp = project_feasible(h_interp, ts)
            c5_try = compute_c5(h_interp)
            if c5_try < best_c5 + 0.005:
                bounds_try = [(0.0, 1.0)] * n_try
                try:
                    res = minimize(penalized_obj, h_interp,
                                  args=(n_try, ts, 100.0, 1000.0),
                                  method='SLSQP', bounds=bounds_try,
                                  options={'maxiter': 400, 'ftol': 1e-13})
                    if res.x is not None:
                        h_new = project_feasible(res.x, ts)
                        c5_new = compute_c5(h_new)
                        if c5_new < best_c5:
                            best_h = h_new
                            best_c5 = c5_new
                            n = n_try
                            n_save = n
                except Exception:
                    pass

    # Phase 5: Final SLSQP polish
    if elapsed() < budget_s * 0.97:
        n_final = len(best_h)
        ts_final = n_final / 2.0
        bounds_final = [(0.0, 1.0)] * n_final
        try:
            res = minimize(penalized_obj, best_h,
                          args=(n_final, ts_final, 300.0, 5000.0),
                          method='SLSQP', bounds=bounds_final,
                          options={'maxiter': 300, 'ftol': 1e-14})
            if res.x is not None:
                h_new = project_feasible(res.x, ts_final)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
        except Exception:
            pass

    # Ensure final feasibility
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final / 2.0)
    best_c5 = compute_c5(best_h)

    # Use official evaluator if available
    try:
        official = evaluate_erdos_solution(best_h)
        if official is not None and np.isfinite(official):
            best_c5 = float(official)
    except Exception:
        pass

    return best_h, best_c5, n_final
