    return softmax_c5(x)

def slsqp_optimize_smooth(h0, target, maxiter=300, beta=80.0):
    global _target
    _target = target
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_eq, 'jac': _con_eq_jac}]
    try:
        res = minimize(_sm_obj, h0, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        h_res = project_onto_set(res.x, target)
        return h_res, compute_c5(h_res)
    except Exception:
        return project_onto_set(h0, target), compute_c5(h0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except Exception:
        h_init = np.ones(48) * 0.5

    n = len(h_init)
    target = n * 0.5
    best_h = project_onto_set(h_init, target)
    best_c5 = compute_c5(best_h)

    def _tl():
        return budget_s - (time.time() - t0)

    # Phase 1: SLSQP on initial
    if _tl() > 60:
        hs, c5s = slsqp_optimize(best_h, target, maxiter=500, ftol=1e-13)
        if c5s < best_c5:
            best_h, best_c5 = hs, c5s

    # Phase 2: PGD polish
    if _tl() > 30:
        hp, c5p = pgd_optimize(best_h, target, max_iter=8000, timeout=20.0)
        if c5p < best_c5:
            best_h, best_c5 = hp, c5p

    # Phase 3: Multi-start with perturbations
    n_restarts = 20
    for r in range(n_restarts):
        if _tl() < 80:
            break
        scale = 0.2 + 0.5 * (r / float(n_restarts))
        h_pert = best_h + rng.uniform(-scale, scale, size=n)
        h_pert = project_onto_set(h_pert, target)
        hs, c5s = slsqp_optimize(h_pert, target, maxiter=250, ftol=1e-12)
        if c5s < best_c5:
            best_h, best_c5 = hs, c5s
            if _tl() > 40:
                hp, c5p = pgd_optimize(best_h, target, max_iter=4000, timeout=12.0)
                if c5p < best_c5:
                    best_h, best_c5 = hp, c5p

    # Phase 4: Try even n (allows 0-1 solutions) and other sizes
    for n_try in [48, 64, 80, 96, 128, 160]:
        if _tl() < 100:
            break
        target_t = n_try * 0.5
        h_interp = np.interp(np.linspace(0.0, 1.0, n_try),
                             np.linspace(0.0, 1.0, len(best_h)), best_h)
        h_interp = project_onto_set(h_interp, target_t)
        hs, c5s = slsqp_optimize(h_interp, target_t, maxiter=300, ftol=1e-12)
        if c5s < best_c5:
            best_h, best_c5 = hs, c5s
            if _tl() > 40:
                hp, c5p = pgd_optimize(best_h, target_t, max_iter=4000, timeout=15.0)
                if c5p < best_c5:
                    best_h, best_c5 = hp, c5p

    # Phase 5: Smooth-aided SLSQP from current best
    if _tl() > 50:
        n_b = len(best_h)
        t_b = n_b * 0.5
        hs, c5s = slsqp_optimize_smooth(best_h, t_b, maxiter=300, beta=60.0)
        if c5s < best_c5:
            best_h, best_c5 = hs, c5s
        if _tl() > 20:
            hp, c5p = pgd_optimize(best_h, t_b, max_iter=3000, timeout=15.0)
            if c5p < best_c5:
                best_h, best_c5 = hp, c5p

    # Phase 6: Final PGD polish with small steps
    if _tl() > 15:
        n_b = len(best_h)
        t_b = n_b * 0.5
        hp, c5p = pgd_optimize(best_h, t_b, max_iter=10000, timeout=min(30.0, _tl() - 5.0))
        if c5p < best_c5:
            best_h, best_c5 = hp, c5p

    # Final projection and evaluation
    n_final = len(best_h)
    best_h = project_onto_set(best_h, n_final * 0.5)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n_final)
