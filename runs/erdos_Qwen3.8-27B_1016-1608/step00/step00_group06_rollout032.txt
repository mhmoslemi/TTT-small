def subgradient_step(h, n, lr, top_k=5):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # Find top lags
    idx = np.argpartition(corr, -top_k)[-top_k:]
    weights = corr[idx]
    weights = weights / weights.sum()
    
    # Gradient of max (avg of top gradients)
    grad = np.zeros(n)
    for w, m in zip(weights, idx):
        # m is the index in the full correlate output
        # full correlate: result[m] = sum_i h[i] * (1-h[i + m - (n-1)])
        # So lag = m - (n-1)
        k = m - (n - 1)
        for j in range(n):
            g = 0.0
            if j + k < n and j + k >= 0:
                g += 1.0 - h[j + k]
            if j - k < n and j - k >= 0:
                g -= h[j - k]
            grad[j] += w * g
    
    h_new = h - lr * grad
    h_new = project_feasible(h_new, n)
    return h_new

            old = h[i]
            best_c5 = compute_c5(h)
            cands = [0.0, 1.0, 0.5, old + 0.08, old - 0.08]
            for v in cands:
                v = float(np.clip(v, 0, 1))
                if abs(v - old) < 1e-7:
                    continue
                h[i] = v
                h = project_feasible(h, n)
                c5 = compute_c5(h)
                if c5 < best_c5 - 1e-12:
                    best_c5 = c5
                    improved = True
                else:
                    h[i] = old
            h[i] = old if not improved else h[i]
        h = project_feasible(h, n)
        if not improved:
            break
    return h, compute_c5(h)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    def time_left():
        return budget_s - (time.time() - t0)

    best_h = initial_h_values.astype(float).copy()
    best_n = len(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP from initial
    h_o, c5_o = slsqp_solve(best_h, best_n, max_iter=300)
    if c5_o < best_c5:
        best_h, best_c5 = h_o, c5_o

    # Phase 2: Subgradient refinement
    h_s, c5_s = subgradient_opt(best_h, best_n, n_steps=30000, time_limit=40)
    if c5_s < best_c5:
        best_h, best_c5 = h_s, c5_s

    # Phase 3: Alternating local optimization + randomization
    for it in range(30):
        if time_left() < 120:
            break
        # SLSQP
        h_new, c5_new = slsqp_solve(best_h, best_n, max_iter=200)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new
        # Subgradient
        h_new, c5_new = subgradient_opt(best_h, best_n, n_steps=15000, time_limit=20)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new
        # Nelder-Mead (small perturbation)
        h_pert = project_feasible(best_h + rng.standard_normal(best_n) * 0.05, best_n)
        h_new, c5_new = nm_solve(h_pert, best_n, maxiter=300)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new
        # Larger perturbation every 5 iterations
        if it % 5 == 4 and time_left() > 90:
            sigma = rng.uniform(0.15, 0.4)
            h_pert = project_feasible(best_h + rng.standard_normal(best_n) * sigma, best_n)
            h_new, c5_new = slsqp_solve(h_pert, best_n, max_iter=150)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new
                h_new, c5_new = subgradient_opt(h_new, best_n, n_steps=10000, time_limit=15)
                if c5_new < best_c5:
                    best_h, best_c5 = h_new, c5_new

    # Phase 4: Coordinate descent polish
    if time_left() > 30:
        best_h, best_c5 = coord_descent(best_h, best_n, rounds=5)

    # Phase 5: Try different resolutions
    for n_try in [48, 64, 80, 128, 160, 200]:
        if time_left() < 80:
            break
        h_i = np.interp(np.linspace(0, 2, n_try), np.linspace(0, 2, best_n), best_h)
        h_i = project_feasible(h_i, n_try)
        h_new, c5_new = slsqp_solve(h_i, n_try, max_iter=200)
        if c5_new < best_c5 - 1e-10:
            h_new, c5_new = subgradient_opt(h_new, n_try, n_steps=8000, time_limit=20)
            if c5_new < best_c5:
                best_h, best_c5, best_n = h_new, c5_new, n_try

    # Final polish
    if time_left() > 20:
        best_h, best_c5 = subgradient_opt(best_h, best_n, n_steps=10000, time_limit=15)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
