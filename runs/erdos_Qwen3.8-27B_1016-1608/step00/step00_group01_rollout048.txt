    """Project h onto {0<=h<=1, sum(h)=target_sum}"""
    h = np.clip(h, 0, 1)

        order = np.argsort(h)

        order = np.argsort(-h)

import numpy as np
from scipy.optimize import minimize
import time

def c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(h)
        for i in order:
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
            if diff <= 1e-12:
                break
    else:
        order = np.argsort(-h)
        for i in order:
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
            if diff >= -1e-12:
                break
    return h

def smooth_objective(x, t, dx):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    if m <= 0:
        return 0.0
    return (m + np.log(np.sum(np.exp(corr - m))) / t) * dx

def run_slsqp(h_init, n, target_sum, dx, t, max_iter=300):
    def obj(x):
        return smooth_objective(x, t, dx)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h_init.copy(), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12})
    return res.x
