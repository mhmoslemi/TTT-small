from scipy.optimize import minimize, linear_sum_assignment
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    """Project h onto {0 <= x <= 1, sum(x) = target_sum}"""
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    # Add/subtract uniformly, then re-clip
    for _ in range(20):
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-10:
            break
    return h

def subgradient(h, n):
    """Compute subgradient of max overlap w.r.t. h"""
    ones_minus_h = 1.0 - h
    corr = np.correlate(h, ones_minus_h, mode="full")
    max_idx = np.argmax(corr)
    lag = max_idx - (n - 1)
    
    grad = np.zeros(n)
    if lag > 0:
        # c = sum_{i=0}^{n-1-lag} h[i]*(1-h[i+lag])
        # d/dh[j] = (1-h[j+lag]) for j < n-lag
        #          -h[j-lag] for j >= lag
        for j in range(n):
            g = 0.0
            if j < n - lag:
                g += (1.0 - h[j + lag])
            if j >= lag:
                g -= h[j - lag]
            grad[j] = g
    elif lag < 0:
        lag_abs = -lag
        # c = sum_{i=0}^{n-1-lag_abs} h[i+lag_abs]*(1-h[i])

        #          -h[j+lag_abs] for j < n-lag_abs  [from -(h[...]*h[j])]

            g = 0.0
            if j >= lag_abs:
                g += (1.0 - h[j - lag_abs])
            if j < n - lag_abs:
                g -= h[j + lag_abs]
            grad[j] = g

        # c = sum h[i]*(1-h[i])
        # d/dh[j] = (1-h[j]) - h[j] = 1 - 2*h[j]

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            free = (h < 1.0 - 1e-15)
        else:
            free = (h > 1e-15)
        nf = float(np.sum(free))
        if nf == 0:
            break
        h[free] += diff / nf
        h = np.clip(h, 0.0, 1.0)
    return h

def subgrad(h, n, top_k=1):
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode="full")
    top_idx = np.argsort(corr)[-top_k:][::-1]
    grad = np.zeros(n)
    for idx in top_idx:
        m = int(idx - (n - 1))
        if m > 0:
            grad[:n - m] += (1.0 - h[m:n])
            grad[m:] -= h[:n - m]
        elif m < 0:
            mm = -m
            grad[mm:] += (1.0 - h[:n - mm])
            grad[:n - mm] -= h[mm:]
        else:
            grad += 1.0 - 2.0 * h
    grad /= float(len(top_idx))
    return grad, corr

def opt_subgrad(h, n_iter=5000, step0=0.04, top_k=1):
    n = len(h)
    ts = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = step0
    for it in range(n_iter):
        g, corr = subgrad(h, n, top_k=top_k)
        c5 = float(np.max(corr) * (2.0 / n))
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        h_new = h - step * g
        h_new = project_feasible(h_new, ts)
        c5_new = compute_c5(h_new)
        if c5_new < c5:
            h = h_new
            step = min(step * 1.03, 0.4)
        else:
            step *= 0.88
        if step < 1e-10:
            break
    return best_h, best_c5

def local_search(h, n_iter=40000, delta=0.005):
    n = len(h)
    h = h.copy()
    bc5 = compute_c5(h)
    stuck = 0
    for it in range(n_iter):
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        d = min(delta, h[i], 1.0 - h[j])
        if d > 1e-9:
            h[i] -= d
            h[j] += d
            c5 = compute_c5(h)
            if c5 < bc5:
                bc5 = c5
                stuck = 0
            else:
                h[i] += d
                h[j] -= d
        d = min(delta, 1.0 - h[i], h[j])
        if d > 1e-9:
            h[i] += d
            h[j] -= d
            c5 = compute_c5(h)
            if c5 < bc5:
                bc5 = c5
                stuck = 0
            else:
                h[i] -= d
                h[j] += d
        stuck += 1
        if stuck > 800:
            break
    return h, bc5

def anneal(h, n_iter=30000, T0=0.005, T_end=1e-7):
    n = len(h)
    h = h.copy()
    bh = h.copy()
    bc5 = compute_c5(h)
    ratio = T_end / T0
    for it in range(n_iter):
        T = T0 * (ratio ** (it / max(n_iter, 1)))
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        d = 0.01
        d = min(d, h[i], 1.0 - h[j])
        if d <= 0:
            continue
        h[i] -= d
        h[j] += d
        nc5 = compute_c5(h)
        if nc5 < bc5:
            bc5 = nc5
            bh = h.copy()
        elif np.random.random() < np.exp(-(nc5 - bc5) / max(T, 1e-12)):
            pass
        else:
            h[i] += d
            h[j] -= d
    return bh, bc5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=np.float64)
    else:
        h0 = np.full(80, 0.5)

    n = len(h0)
    ts = n / 2.0
    h0 = project_feasible(h0, ts)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    def check_time():
        return time.time() - t0

    # Phase 1: coarse subgradient
    if check_time() < budget_s * 0.2:
        h1, c5_1 = opt_subgrad(h0, n_iter=8000, step0=0.05, top_k=1)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()

    # Phase 2: local search
    if check_time() < budget_s * 0.4:
        h2, c5_2 = local_search(best_h, n_iter=80000, delta=0.004)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 3: finer subgradient with top_k=3
    if check_time() < budget_s * 0.55:
        h3, c5_3 = opt_subgrad(best_h, n_iter=8000, step0=0.015, top_k=3)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()

    # Phase 4: tight local search
    if check_time() < budget_s * 0.7:
        h4, c5_4 = local_search(best_h, n_iter=100000, delta=0.002)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    # Phase 5: simulated annealing for escape
    if check_time() < budget_s * 0.85:
        h5, c5_5 = anneal(best_h, n_iter=40000, T0=0.003, T_end=1e-7)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()

    # Phase 6: final refinement

        if c5_7 < best_c5:
            best_c5 = c5_7

    final_c5 = compute_c5(best_h)
    return (best_h, final_c5, n)
