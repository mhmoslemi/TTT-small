import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14 * n:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        idx = int(np.argmin(np.abs(h - 0.5)))
        h[idx] += diff
    return h

def smooth_objective(x, beta, dx, target_sum):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    shifted = (corr - m) * beta
    val = m + np.log(np.sum(np.exp(shifted))) / beta
    return val

def local_optimize(h0, n, dx, target_sum, beta, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    res = minimize(smooth_objective, h0, args=(beta, dx, target_sum),
                   method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    h_out = fix_sum(res.x, target_sum)
    return h_out, compute_c5(h_out, dx)
