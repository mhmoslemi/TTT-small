def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
            # Final correction
            diff2 = target - np.sum(h)
            if abs(diff2) > 1e-15 and mask.sum() > 0:
                # Distribute remaining
                idx = np.where(mask)[0]
                # Add to a single element
                i = idx[0]
                h[i] += diff2
                h = np.clip(h, 0.0, 1.0)
    return h

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    for _ in range(3):
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
    return h

def sa_opt(h, n, rng, t_start, time_limit, iters=200000, T0=0.3, T1=0.001):
    current = h.copy()
    best = h.copy()
    best_c5 = compute_c5(best)
    current_c5 = best_c5
    alpha = (max(T1, 1e-8) / T0) ** (1.0 / iters)
    for it in range(iters):
        if time.time() > time_limit:
            break
        T = T0 * (alpha ** it)
        n_change = max(1, int(n * 0.12 * max(T / T0, 0.1)))
        n_change = min(n_change, n)
        indices = rng.choice(n, size=n_change, replace=False)
        step = 0.15 * max(T / T0, 0.05)
        current[indices] += (rng.random(n_change) - 0.5) * 2.0 * step
        current = project_feasible(current, n)
        c5 = compute_c5(current)
        delta = c5 - current_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            current_c5 = c5
            if c5 < best_c5:
                best_c5 = c5
                best = current.copy()
    return best, best_c5

def binary_hill(h, n, rng, t_start, time_limit, max_iters=500000):

    n_ones = int(np.sum(current))

        current[tf] = 0.0
    elif n_ones < target_ones:
        zi = np.where(current == 0)[0]
        tf = rng.choice(zi, size=target_ones - n_ones, replace=False)
        current[tf] = 1.0
    best = current.copy()
    best_c5 = compute_c5(best)
    current_c5 = best_c5
    stall = 0
    for it in range(max_iters):
        if time.time() > time_limit:

        c5 = compute_c5(current)
        if c5 <= current_c5:
            current_c5 = c5
            stall = 0
