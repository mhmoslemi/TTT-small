    return float(np.max(corr) * dx)

def project_feasible(h, n):

        if cnt == 0:

        h = np.clip(h, 0.0, 1.0)

def corr_grad_lag(h, lag):

    grad = np.zeros(n)
    idx1 = j - lag
    v1 = (idx1 >= 0) & (idx1 < n)
    grad[v1] += (1.0 - h[idx1[v1]])
    idx2 = j + lag
    v2 = (idx2 >= 0) & (idx2 < n)
    grad[v2] -= h[idx2[v2]]
