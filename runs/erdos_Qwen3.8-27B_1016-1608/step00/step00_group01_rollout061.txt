import numpy as np
import time


def _smooth_val(hh, dx, beta):
    cc = np.correlate(hh, 1.0 - hh, mode='full') * dx
    mc = cc.max()
    return (1.0 / beta) * (mc + np.log(np.exp(beta * (cc - mc)).sum()))


def _smooth_grad(h, dx, beta):
    n = len(h)
    eps = 1e-4
    grad = np.zeros(n)
    for i in range(n):
        h[i] += eps
        fp = _smooth_val(h, dx, beta)
        h[i] -= 2.0 * eps
        fm = _smooth_val(h, dx, beta)
        h[i] += eps
        grad[i] = (fp - fm) / (2.0 * eps)
    return grad


def _sumsq_val(hh, dx):
    cc = np.correlate(hh, 1.0 - hh, mode='full') * dx
    return np.sum(cc * cc)


def _sumsq_grad(h, dx):
    n = len(h)
    eps = 1e-5
    grad = np.zeros(n)
    for i in range(n):
        h[i] += eps
        fp = _sumsq_val(h, dx)
        h[i] -= 2.0 * eps
        fm = _sumsq_val(h, dx)
        h[i] += eps
        grad[i] = (fp - fm) / (2.0 * eps)
    return grad


def _project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(30):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h


def _binary_round(h):
    n = len(h)
    k = n // 2
    h_bin = np.zeros(n)
    idx = np.argsort(h)[-k:]
    h_bin[idx] = 1.0
    return h_bin
