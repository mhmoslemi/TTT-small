        grad[idx2] -= h[idx2 + a]
    h_new = h - lr * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = adjust_sum(h_new, n)
    return h_new, k_star

def coordinate_descent(h, n, max_iters=50, delta_steps=(0.02, 0.05, 0.1)):
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                for delta in delta_steps:
                    if h[i] < delta or h[j] < delta:
                        continue
                    h_test = h.copy()
                    h_test[i] -= delta
                    h_test[j] += delta
                    c5_t = compute_c5(h_test)
                    if c5_t < best_c5 - 1e-12:
                        h = h_test
                        best_c5 = c5_t
                        improved = True
                        break
                    h_test = h.copy()
                    h_test[i] += delta
                    h_test[j] -= delta
                    c5_t = compute_c5(h_test)
                    if c5_t < best_c5 - 1e-12:
                        h = h_test
                        best_c5 = c5_t
                        improved = True
                        break
        if not improved:
            break
    return h, best_c5

def nm_objective(x, n):
    h = np.clip(x, 0.0, 1.0)
    penalty = 50.0 * (np.sum(h) - n / 2.0) ** 2
    return compute_c5(h) + penalty

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)

    def elapsed():
        return time.time() - t0

    # Load initial values
    try:
        h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        h_init = np.full(45, 1.0 / 45.0 * 2.0)
        h_init = np.clip(h_init, 0, 1)

    n = len(h_init)
    h_init = np.clip(h_init, 0.0, 1.0)
    h_init = adjust_sum(h_init, n)

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Subgradient descent
    lr = 0.3
    for it in range(2000):
        if elapsed() > budget_s * 0.35:
            break
        h_new, _ = subgrad_step(best_h, lr)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
            lr = min(lr * 1.02, 1.0)
        else:
            lr = max(lr * 0.95, 0.01)

    # Phase 2: Nelder-Mead local optimization
    if elapsed() < budget_s * 0.55:
        starts = [best_h.copy()]
        for _ in range(3):
            pert = best_h + np.random.randn(n) * 0.03
            pert = adjust_sum(np.clip(pert, 0, 1), n)
            starts.append(pert)
        for hs in starts:
            if elapsed() > budget_s * 0.55:
                break
            try:
                res = minimize(nm_objective, hs, method='Nelder-Mead',
                              options={'maxiter': 3000, 'xatol': 1e-10, 'fatol': 1e-12})
                h_cand = adjust_sum(np.clip(res.x, 0, 1), n)
                c5_cand = compute_c5(h_cand)
                if c5_cand < best_c5:
                    best_h = h_cand
                    best_c5 = c5_cand
            except Exception:
                pass

    # Phase 3: Subgradient again with smaller lr
    if elapsed() < budget_s * 0.75:
        lr = 0.05
        for it in range(3000):
            if elapsed() > budget_s * 0.75:
                break
            h_new, _ = subgrad_step(best_h, lr)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
                lr = min(lr * 1.01, 0.2)
            else:
                lr = max(lr * 0.97, 0.005)

    # Phase 4: Coordinate descent (limited)
    if elapsed() < budget_s * 0.90:
        best_h, best_c5 = coordinate_descent(best_h, n, max_iters=20,
                                             delta_steps=(0.02, 0.05))

    # Phase 5: Try n=90 via interpolation
    if elapsed() < budget_s * 0.88:
        n2 = 2 * n
        x_old = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
        x_new = np.linspace(0, 2, n2, endpoint=False) + 1.0 / n2
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = adjust_sum(h_interp, n2)
        c5_90 = compute_c5(h_interp)
        if c5_90 < best_c5:
            # Optimize at n2
            h2 = h_interp.copy()
            lr = 0.3
            for it in range(3000):
                if elapsed() > budget_s * 0.95:
                    break
                h_new, _ = subgrad_step(h2, lr)
                c5_new = compute_c5(h_new)
                if c5_new < c5_90:
                    h2 = h_new
                    c5_90 = c5_new
                    lr = min(lr * 1.02, 1.0)
                else:
                    lr = max(lr * 0.95, 0.02)
            if c5_90 < best_c5:
                best_h = h2
                best_c5 = c5_90
                n = n2

    # Final: ensure exact sum
    n = len(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)
    # Exact sum fix
    target = n / 2.0
    diff = target - np.sum(best_h)
    if abs(diff) > 1e-14:
        mask = (best_h > 0) & (best_h < 1)
        if mask.any():
            best_h[mask] += diff / mask.sum()
            best_h = np.clip(best_h, 0, 1)
    best_c5 = compute_c5(best_h)

    # If time remains, one more NM pass
    if elapsed() < budget_s * 0.92:
        try:
            res = minimize(nm_objective, best_h, method='Nelder-Mead',
                          options={'maxiter': 2000, 'xatol': 1e-11, 'fatol': 1e-13})
            h_cand = adjust_sum(np.clip(res.x, 0, 1), n)
            c5_cand = compute_c5(h_cand)
            if c5_cand < best_c5:
                best_h = h_cand
                best_c5 = c5_cand
        except Exception:
            pass

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = adjust_sum(best_h, len(best_h))
    best_c5 = compute_c5(best_h)

    return best_h, float(best_c5), len(best_h)
