def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return np.max(corr) * dx

def smooth_max_c5(h, T=50.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # Smooth max
    m = np.max(corr)
    return m + np.log(np.mean(np.exp((corr - m) * T))) / T

def project_to_constraint(h, n):

    target = n / 2.0

        h = np.clip(h, 0, 1)

            if mask:
                free = h < 1.0
            else:
                free = h > 0.0
            h[free] += diff / free.sum()
            h = np.clip(h, 0, 1)
    return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_val(corr, T):
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -200, 200)
    return float(m + np.log(np.sum(np.exp(shifted))) / (T * len(corr)))

def obj_func(x, n, T, lam):
    h = np.clip(x, 0.0, 1.0)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    f = smooth_max_val(corr, T)
    f += lam * (float(np.sum(h)) - 0.5 * n) ** 2
    return f

def project_sum(h, n):
    h = np.array(h, dtype=float)
    for _ in range(500):
        diff = 0.5 * n - float(np.sum(h))
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def single_optimize(h_start, n, T, lam, maxiter):
    res = minimize(
        obj_func, h_start, args=(n, T, lam),
        method='L-BFGS-B',
        bounds=[(0.0, 1.0)] * n,
        options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-11}
    )
    h_new = np.clip(res.x, 0.0, 1.0)
    h_new = project_sum(h_new, n)
    return h_new

def optimize_loop(h_start, n, T, lam, rounds, time_limit):
    h_cur = h_start.copy()
    for r in range(rounds):
        if time.time() > time_limit:
            break
        T_cur = T * (1.0 + 0.5 * r)
        h_new = single_optimize(h_cur, n, T_cur, lam, 500)
        if np.max(np.abs(h_new - h_cur)) < 5e-12:
            break
        h_cur = h_new
    return h_cur

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    time_limit = t0 + budget_s * 0.90
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.full(80, 0.5)

    n = len(h_best)
    h_best = project_sum(h_best, n)
    best_c5 = compute_c5(h_best)

    T0 = 30.0
    lam0 = 200.0

    # Phase 1: Optimize at native resolution
    h_best = optimize_loop(h_best, n, T0, lam0, 4, time_limit)
    best_c5 = compute_c5(h_best)

    # Phase 2: Multi-start perturbations at native n
    for trial in range(12):
        if time.time() > time_limit * 0.50:
            break
        scale = 0.005 + 0.008 * trial
        h_pert = h_best + rng.normal(0, scale, n)
        h_pert = project_sum(h_pert, n)
        h_opt = optimize_loop(h_pert, n, T0, lam0, 3, time_limit)
        c5_t = compute_c5(h_opt)
        if c5_t < best_c5:
            best_c5 = c5_t
            h_best = h_opt

    # Phase 3: Try different resolutions by interpolating
    for target_n in [120, 160, 200, 240]:
        if time.time() > time_limit * 0.75:
            break
        old_n = len(h_best)
        x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
        x_new = np.linspace(0.0, 1.0, target_n, endpoint=False)
        h_interp = np.interp(x_new, x_old, h_best)
        h_interp = project_sum(h_interp, target_n)
        h_opt = optimize_loop(h_interp, target_n, T0, lam0, 3, time_limit)
        c5_t = compute_c5(h_opt)
        if c5_t < best_c5:
            best_c5 = c5_t
            h_best = h_opt

    # Phase 4: Final polish with higher T
    n_final = len(h_best)
    h_best = optimize_loop(h_best, n_final, T0 * 2.5, lam0 * 2, 3, t0 + budget_s * 0.96)
    best_c5 = compute_c5(h_best)

    # Ensure feasibility
    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = project_sum(h_best, len(h_best))
    best_c5 = compute_c5(h_best)

    return h_best, best_c5, len(h_best)
