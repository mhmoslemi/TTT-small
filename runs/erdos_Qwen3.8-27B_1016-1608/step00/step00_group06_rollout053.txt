import time

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n * 0.5
    s = np.sum(h)
    if s > 1e-12:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        idx = np.argsort(h) if diff > 0 else np.argsort(-h)
        for i in idx:
            if abs(diff) < 1e-12:
                break
            if diff > 0 and h[i] < 1.0:
                a = min(diff, 1.0 - h[i])
                h[i] += a
                diff -= a
            elif diff < 0 and h[i] > 0.0:
                a = min(-diff, h[i])
                h[i] -= a
                diff += a
    return h

def soft_max_obj(v, n, beta):
    dx = 2.0 / n
    corr = np.correlate(v, 1.0 - v, mode='full') * dx
    cm = np.max(corr)
    return cm + np.log(np.sum(np.exp(beta * (corr - cm)))) / beta
