    def __init__(self, M, n):
        self.M = M
        self.n = n

    def __call__(self, h):
        dx = 2.0 / self.n
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        return float(np.sum(corr ** self.M) ** (1.0 / self.M))


class LogSumExpObj:
    def __init__(self, tau, n):
        self.tau = tau
        self.n = n

    def __call__(self, h):
        dx = 2.0 / self.n
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        cmax = np.max(corr)
        return float(cmax + np.log(np.sum(np.exp(corr - cmax))) / self.tau)

        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
    h = np.clip(h, 0.0, 1.0)
    return h
