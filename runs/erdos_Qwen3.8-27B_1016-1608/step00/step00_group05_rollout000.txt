import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    om = 1.0 - h
    cor = np.correlate(h, om, mode="full") * dx
    return float(np.max(cor))

def logsumexp(x):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(x - m)))

def smooth_obj(h, temp, n):
    dx = 2.0 / n
    om = 1.0 - h
    cor = np.correlate(h, om, mode="full") * dx
    pen = 50.0 * (np.sum(h) - 0.5 * n) ** 2
    return logsumexp(cor / temp) * temp + pen

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = 0.5 * n
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h[mask] = np.clip(h[mask], 0.0, 1.0)
    return h

def coord_swap(h, n, iters=5000, rng=None):
    h = h.copy()
    best = compute_c5(h)
    if rng is None:
        rng = np.random.default_rng(0)
    for _ in range(iters):
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        d = rng.uniform(0.02, 0.15)
        if h[i] < d or h[j] > d:
            continue
        h[i] -= d
        h[j] += d
        c = compute_c5(h)
        if c < best - 1e-12:
            best = c
        else:
            h[i] += d
            h[j] -= d
    return h, best

    np.random.seed(seed)

    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    def elapsed():
        return time.time() - t0

    # Phase 1: Multi-temperature L-BFGS
    temps = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
    for temp in temps:
        if elapsed() > budget_s * 0.35:
            break
        try:
            res = minimize(
                lambda h: smooth_obj(h, temp, n),
                best_h.copy(),
                method='L-BFGS-B',
                bounds=[(0.0, 1.0)] * n,
                options={'maxiter': 200, 'maxfun': 500, 'ftol': 1e-14, 'gtol': 1e-8}
            )
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        except Exception:
            break
