def eval_c5(h):

    h_bin = (h > 0.5).astype(float)
    if h_bin.sum() != n // 2:
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        if len(ones) > n // 2:
            idx = rng.choice(len(ones), len(ones) - n // 2, replace=False)
            h_bin[ones[idx]] = 0.0
        elif len(zeros) > n // 2:
            idx = rng.choice(len(zeros), n // 2 - len(ones), replace=False)
            h_bin[zeros[idx]] = 1.0
        else:
            return h, eval_c5(h)
    best_h = h_bin.copy()
    best_c = float(np.max(np.correlate(h_bin, 1.0 - h_bin, mode='full')))

    stall = 0

        j = zeros[rng.randint(len(zeros))]

def multi_point_search(h, n, rng, deadline):
    """Adjust 3-4 points simultaneously."""
    best_h = h.copy()
    best_c = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
    scale = 2.0 / n
    stall = 0
    while time.time() < deadline:
        k = rng.randint(2, 5)
        idx = rng.choice(n, size=k, replace=False)
        deltas = rng.uniform(-0.1, 0.1, size=k)
        deltas -= deltas.mean()  # preserve sum
        h_try = h.copy()
        h_try[idx] += deltas
        if h_try.min() < -1e-10 or h_try.max() > 1.0 + 1e-10:

        c = float(np.max(np.correlate(h_try, 1.0 - h_try, mode='full')))

            best_h = h_try
            stall = 0
        else:
            stall += 1
        if stall > 3000:
            h = best_h.copy()
            stall = 0

def fix_sum(h, n, rng):
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-10:
        return h
    h = h.copy()
    for _ in range(10000):
        if diff > 1e-10:
            cand = np.where(h < 1.0)[0]
        else:
            cand = np.where(h > 0.0)[0]
        if len(cand) == 0:
            break
        j = cand[rng.randint(len(cand))]
        step = min(abs(diff), 0.01)
        if diff > 0:
            h[j] += min(step, 1.0 - h[j])
        else:
            h[j] -= min(step, h[j])
        diff = target - h.sum()
        if abs(diff) < 1e-10:
            break
    return h
