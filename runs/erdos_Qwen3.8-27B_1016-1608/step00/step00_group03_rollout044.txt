import numpy as np
from scipy.optimize import minimize

def c5_value(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def smooth_obj(x, beta):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = beta * (corr - m)
    a_max = np.max(shifted)
    return m + (a_max + np.log(np.sum(np.exp(shifted - a_max)))) / beta


def constraint_sum(x):
    return np.sum(x) - len(x) / 2.0


def repair(h, n):
    h = np.clip(h, 0.0, 1.0)

    if abs(diff) > 1e-14:
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
