def project_onto_constraint(v, target_sum):
    """Project v onto {h : 0<=h_i<=1, sum(h)=target_sum}"""
    n = len(v)
    # Bisection for lambda
    lo = -v.max() - 1
    hi = 1 - v.min() + 1
    
    for _ in range(80):
        mid = (lo + hi) / 2
        proj = np.clip(v + mid, 0, 1)
        s = proj.sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    
    return np.clip(v + (lo + hi) / 2, 0, 1)

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

import numpy as np
import time

def project_onto_constraint(v, target_sum):
    """Project v onto {h : 0<=h_i<=1, sum(h)=target_sum} via bisection on shift."""
    lo = -v.max() - 0.5
    hi = 1.0 - v.min() + 0.5
    for _ in range(70):
        mid = (lo + hi) * 0.5
        s = np.clip(v + mid, 0.0, 1.0).sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    lam = (lo + hi) * 0.5
    return np.clip(v + lam, 0.0, 1.0)

def c5_value(h):
    """Compute C5 = max_k correlate(h,1-h,'full')[k] * dx."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_grad(h, beta):
    """Gradient of (1/beta)*logsumexp(beta * corr * dx) w.r.t. h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    s = beta * corr
    s -= s.max()
    e = np.exp(s)
    w = e / e.sum()
    
    grad = np.zeros(n)
    ncorr = len(w)
    for idx in range(ncorr):
        wi = w[idx]
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        if k >= 0:
            if k < n:
                grad[k:] += wi * (1.0 - h[:n-k])
                grad[:n-k] -= wi * h[k:]
        else:
            m = -k
            if m < n:
                grad[:n-m] += wi * (1.0 - h[m:])
                grad[m:] -= wi * h[:n-m]
    return grad * dx

def pgd_optimize(h, target_sum, max_time, beta, lr_init, rng):
    """Projected gradient descent with momentum for C5 minimization."""
    n = len(h)
    t0 = time.time()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = c5_value(h)
    lr = lr_init
    stall = 0
    mom = 0.85
    
    while time.time() - t0 < max_time:
        g = smooth_grad(h, beta)
        v = mom * v - lr * g
        h_new = project_onto_constraint(h + v, target_sum)
        c5_new = c5_value(h_new)
        
        if c5_new < best_c5 - 1e-12:
            best_h = h_new.copy()
            best_c5 = c5_new
            stall = 0
        else:
            stall += 1
        
        h = h_new
        lr *= 0.9995
        
        if stall > 300:

            h = best_h + 0.005 * rng.randn(n)
            h = project_onto_constraint(h, target_sum)
            v = np.zeros(n)
            lr = lr_init * 0.6
            stall = 0
            mom = 0.8
    
    return best_h, best_c5

def interpolate_to_size(h_old, n_new):
    """Interpolate h from its grid to a new grid of size n_new."""
    n_old = len(h_old)
    x_old = np.arange(n_old) / n_old * 2.0
    x_new = np.arange(n_new) / n_new * 2.0
    # Handle periodicity by extending

    x_ext = np.concatenate([x_old[:1] - 2.0/n_old, x_old, x_old[-1:] + 2.0/n_old])
    h_new = np.interp(x_new, x_ext, h_ext)
    return np.clip(h_new, 0.0, 1.0)
