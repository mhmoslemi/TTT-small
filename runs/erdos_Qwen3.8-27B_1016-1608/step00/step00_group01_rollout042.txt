For h in [0,1]: h*(1-h) ≤ 1/4, with equality at h=1/2.

But sum(h) = n/2, so sum(h^2) is minimized when all h = 1/2 (by Cauchy-Schwarz), giving sum(h^2) ≥ (n/2)^2/n = n/4.
So sum(h) - sum(h^2) ≤ n/2 - n/4 = n/4.
C5 ≥ 0 (always non-negative since h(i) and (1-h(i+k)) are both in [0,1]).

The zero-lag contribution: (sum(h) - sum(h^2)) * dx = (n/2 - sum(h^2)) * 2/n = 1 - 2*sum(h^2)/n.

To minimize this, we want to MAXIMIZE sum(h^2). This means making h as "extreme" as possible (values near 0 or 1). With sum(h) = n/2:

import numpy as np
import time


def eval_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def ternary_min(h, a, b, dx):
    ha, hb = h[a], h[b]
    D = min(ha, 1.0 - hb)
    if D < 1e-12:
        return 0.0, eval_c5(h, dx)
    lo, hi = 0.0, D
    for _ in range(8):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        h[a] = ha - m1
        h[b] = hb + m1
        f1 = eval_c5(h, dx)
        h[a] = ha - m2
        h[b] = hb + m2
        f2 = eval_c5(h, dx)
        h[a] = ha
        h[b] = hb
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    d = (lo + hi) / 2.0
    h[a] = ha - d
    h[b] = hb + d
    fd = eval_c5(h, dx)
    h[a] = ha
    h[b] = hb
    h[a] = ha - D
    h[b] = hb + D
    fD = eval_c5(h, dx)
    h[a] = ha
    h[b] = hb
    f0 = eval_c5(h, dx)
    if fD < fd and fD < f0:
        return D, fD
    if f0 < fd and f0 < fD:
        return 0.0, f0
    return d, fd


def cont_optimize(h, n, dx, target_sum, deadline):
    h = np.array(h, dtype=float)
    cur = eval_c5(h, dx)
    stall = 0
    while time.time() < deadline and stall < 100:
        imp = False
        for _ in range(600):
            a = np.random.randint(n)
            b = np.random.randint(n)
            if a == b or h[a] < 1e-9 or h[b] > 1.0 - 1e-9:
                continue
            d, fv = ternary_min(h, a, b, dx)
            if d > 1e-10 and fv < cur - 1e-13:
                h[a] -= d
                h[b] += d
                cur = fv
                imp = True
        stall = 0 if imp else stall + 1
    return h, cur


def bin_optimize(h, n, dx, target_sum, deadline):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    h[h > 0.45] = np.minimum(1.0, h[h > 0.45] + 0.35)
    h[h <= 0.45] = np.maximum(0.0, h[h <= 0.45] - 0.35)
    s = h.sum()
    if s > 1e-9:
        h *= target_sum / s
        h = np.clip(h, 0.0, 1.0)
    cur = eval_c5(h, dx)
    T = 0.004
    stall = 0
    while time.time() < deadline and stall < 400:
        imp = False
        for _ in range(1500):
            a = np.random.randint(n)
            b = np.random.randint(n)
            if a == b:
                continue
            h[a], h[b] = h[b], h[a]
            v = eval_c5(h, dx)
            if v < cur or np.random.rand() < np.exp(-(v - cur) / max(T, 1e-12)):
                cur = v
                imp = True
            else:
                h[a], h[b] = h[b], h[a]
            T = max(T * 0.9997, 1e-6)
        stall = 0 if imp else stall + 1
