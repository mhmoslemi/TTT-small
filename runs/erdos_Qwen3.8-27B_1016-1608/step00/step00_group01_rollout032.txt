        h_opt, c5_opt = optimize_from_start(h0, target_sum, n, 25.0, time_left)
        if c5_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_opt

    # Second pass: optimize from best with higher alpha and more care
    if time_left() > 80:
        h_opt, c5_opt = optimize_from_start(best_h, target_sum, n, 80.0, time_left)
        if c5_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_opt

    # Random restarts from best (shake and re-optimize)
    for rs in range(20):
        if time_left() < 40:
            break
        h_shake = best_h + rng.randn(n) * 0.04
        h_shake = project_feasible(h_shake, target_sum)
        h_opt, c5_opt = optimize_from_start(h_shake, target_sum, n, 50.0, time_left)
        if c5_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_opt

    # Fine coordinate descent polish
    if time_left() > 50:
        h = best_h.copy()
        for cd_pass in range(200):
            if time_left() < 25:
                break
            improved = False
            for j in range(n):
                orig = h[j]
                found = False
                for delta in (0.004, -0.004, 0.002, -0.002, 0.001, -0.001):
                    h[j] = np.clip(orig + delta, 0.0, 1.0)
                    h_t = project_feasible(h, target_sum)
                    c5_t = compute_c5(h_t)
                    if c5_t < best_c5 - 1e-15:
                        h = h_t
                        best_h = h.copy()
                        best_c5 = c5_t
                        improved = True
                        found = True
                        break
                if not found:
                    h[j] = orig
            if not improved:
                break

    best_h = project_feasible(best_h, target_sum)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n
