    bounds = [(0.0, 1.0)] * n

    def de_obj(x):
        hr = repair(x, n)
        return compute_c5(hr, n)

    def init_pop(bounds, rng_local):
        pop = np.zeros((pop_size, n))
        for i in range(pop_size):
            if i == 0:
                pop[i] = h0
            else:
                pop[i] = h0 + rng_local.normal(0, 0.2, size=n)
            pop[i] = repair(pop[i], n)
        return pop

    try:
        ml = int(40 * time_left() / (budget_s / 100))
        ml = max(30, min(ml, 400))
        res = differential_evolution(
            de_obj, bounds, seed=seed, maxiter=ml,
            popsize=pop_size, tolerance=1e-7, polish=False,
            init=init_pop, mutation=(0.4, 1.0), recombination=0.8, workers=1
        )
        h_new = repair(res.x, n)
        c5_new = compute_c5(h_new, n)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    except Exception:
        pass

    # Phase 2: Nelder-Mead
    if time_left() > budget_s * 0.1:
        try:
            def nm_obj(x):
                return compute_c5(repair(x, n), n)
            res2 = minimize(nm_obj, best_h, method='Nelder-Mead',
                           options={'maxiter': 8000, 'fatol': 1e-11, 'xatol': 1e-11})
            h_r = repair(res2.x, n)
            c_r = compute_c5(h_r, n)
            if c_r < best_c5:
                best_c5 = c_r
                best_h = h_r
        except Exception:
            pass

    # Phase 3: Coordinate hill-climbing / local refine
    if time_left() > budget_s * 0.05:
        h_cl, c_cl = local_refine(best_h, n, rng, max_iters=int(8000 * time_left() / budget_s))
        if c_cl < best_c5:
            best_c5 = c_cl
            best_h = h_cl

        h_ms, c_ms = multi_start_local(best_h, n, rng, n_starts=6, max_iters=2000)
        if c_ms < best_c5:
            best_c5 = c_ms
            best_h = h_ms

    # Phase 4: Try 2*n resolution if time allows
    if time_left() > budget_s * 0.25:
        n2 = n0 * 2
        x_old = np.linspace(0, 2.0, n0, endpoint=False)
        x_new = np.linspace(0, 2.0, n2, endpoint=False)
        h2 = np.interp(x_new, x_old, best_h, period=2.0)
        h2 = repair(h2, n2)
        c5_2 = compute_c5(h2, n2)

        def de_obj2(x):
            hr = repair(x, n2)
            return compute_c5(hr, n2)

        bounds2 = [(0.0, 1.0)] * n2
        pop_size2 = 30

        def init_pop2(bounds, rng_local):
            pop = np.zeros((pop_size2, n2))
            for i in range(pop_size2):
                if i == 0:
                    pop[i] = h2
                else:
                    pop[i] = h2 + rng_local.normal(0, 0.1, size=n2)
                pop[i] = repair(pop[i], n2)
            return pop

        try:
            ml2 = int(8 * time_left() / (budget_s / 100))
            ml2 = max(15, min(ml2, 150))
            res3 = differential_evolution(
                de_obj2, bounds2, seed=seed + 100, maxiter=ml2,
                popsize=pop_size2, tolerance=1e-7, polish=False,
                init=init_pop2, mutation=(0.4, 1.0), recombination=0.8, workers=1
            )
            h_new2 = repair(res3.x, n2)
            c5_new2 = compute_c5(h_new2, n2)
            if c5_new2 < best_c5:
                best_c5 = c5_new2
                best_h = h_new2
        except Exception:
            pass

        if time_left() > budget_s * 0.08 and c5_2 < best_c5 + 0.005:
            h_cl2, c_cl2 = local_refine(best_h, n2, rng, max_iters=5000)
            if c_cl2 < best_c5:
                best_c5 = c_cl2
                best_h = h_cl2

    # Final evaluation
    try:
        fc5 = evaluate_erdos_solution(best_h)
        if fc5 is not None and fc5 < best_c5:
            best_c5 = float(fc5)
    except Exception:
        pass

    return (best_h, best_c5, len(best_h))
