from scipy.optimize import minimize
import time

_g_n = 45

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    comp = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(comp) * dx)

def project_feasible(h):
    n = len(h)
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _obj(x):
    return compute_c5(x)

def _cons(x):
    return np.sum(x) - _g_n / 2.0

def _cons_jac(x):
    return np.ones(_g_n)

def _nm_obj(x):
    h = np.clip(x, 0.0, 1.0)
    pen = 100.0 * (np.sum(h) - _g_n / 2.0) ** 2
    return compute_c5(h) + pen

    global _g_n
    _g_n = len(h0)
    cons = [{'type': 'eq', 'fun': _cons, 'jac': _cons_jac}]
    bounds = [(0.0, 1.0)] * _g_n
    try:
        res = minimize(_obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 400, 'ftol': 1e-15})
        return project_feasible(res.x)
    except Exception:
        return project_feasible(h0)
