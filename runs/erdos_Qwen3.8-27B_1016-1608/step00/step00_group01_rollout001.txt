import numpy as np
import time
import random
from scipy.optimize import minimize

def compute_c5(h, dx):
    h = np.asarray(h, dtype=np.float64)
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    return float(np.max(corr)) * dx

def smooth_max_obj(h_flat, dx, alpha):
    h = np.asarray(h_flat, dtype=np.float64)
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full') * dx
    c = alpha * corr
    c_max = np.max(c)
    return (c_max + np.log(np.sum(np.exp(c - c_max)))) / alpha

def project_feasible(h, target_sum):
    h = np.clip(np.asarray(h, dtype=np.float64).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        mask = h < 1.0 if diff > 0 else h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h
