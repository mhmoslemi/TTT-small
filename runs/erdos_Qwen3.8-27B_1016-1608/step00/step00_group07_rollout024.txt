import numpy as np
import time

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def compute_c5(h):
    return float(np.max(compute_corr(h)))

def grad_corr_at_k(h, k):
    n = len(h)
    dx = 2.0 / n
    m = k - (n - 1)
    g = np.zeros(n)
    for j in range(n):
        val = 0.0
        if 0 <= j + m < n:
            val += 1.0 - h[j + m]
        if 0 <= j - m < n:
            val -= h[j - m]
        g[j] = dx * val
    return g

def project_onto_feasible(h, target_sum):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
        else:
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_optimize(h0, max_iter=5000, lr0=0.05, time_limit=60):
    n = len(h0)
    target_sum = n / 2.0
    h = h0.copy()
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    lr = lr0
    start = time.time()
    
    stagnant = 0
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        corr = compute_corr(h)
        c5 = float(np.max(corr))
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 200:
                lr *= 0.7
                stagnant = 0
        
        # Get subgradient from top few max k values
        top_k = np.argsort(corr)[-3:]
        g = np.zeros(n)
        for k in top_k:
            g += grad_corr_at_k(h, k)
        g /= len(top_k)
        
        # Subtract mean to preserve sum
        g -= np.mean(g)
        
        h = project_onto_feasible(h - lr * g, target_sum)
        
        # Occasionally restart lr
        if it % 500 == 0 and it > 0:
            lr = max(lr * 0.8, 0.001)
    
    return best_h, best_c5

def coordinate_descent(h0, max_iter=50000, time_limit=60, seed=42):
    n = len(h0)
    target_sum = n / 2.0
    h = h0.copy()
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    rng = np.random.RandomState(seed)
    start = time.time()
    delta = 0.05
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        # Pick a random pair
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        sign = 1.0 if rng.rand() < 0.5 else -1.0
        d = sign * delta
        
        if h[i] + d < 0 or h[i] + d > 1 or h[j] - d < 0 or h[j] - d > 1:
            continue
        
        h[i] += d
        h[j] -= d
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        else:
            h[i] -= d
            h[j] += d
    
    return best_h, best_c5

def random_perturb(h0, n_perturbs=100, seed=42):

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def grad_corr_at_k(h, k):
    n = len(h)
    dx = 2.0 / n
    m = k - (n - 1)
    g = np.zeros(n)
    # corr[k] = dx * (sum_{i in S} h[i] - sum_{i in S} h[i]*h[i+m])
    # where S = {i: 0<=i<n, 0<=i+m<n}

        jp = j + m
        if 0 <= jp < n:  # j in S
            val += 1.0 - h[jp]
        if 0 <= jm < n:  # j-m in S (and j is already valid)

import numpy as np
import time

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def compute_c5(h):
    return float(np.max(compute_corr(h)))

def grad_corr_at_k(h, k):
    n = len(h)
    dx = 2.0 / n
    m = k - (n - 1)
    g = np.zeros(n)
    for j in range(n):
        val = 0.0
        jp = j + m
        jm = j - m
        if 0 <= jp < n:
            val += 1.0 - h[jp]
        if 0 <= jm <= n - 1:
            val -= h[jm]
        g[j] = dx * val
    return g

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h
