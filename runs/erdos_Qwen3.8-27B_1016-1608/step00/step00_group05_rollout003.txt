import numpy as np
import time

def project_onto_feasible(h, target_sum):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-14:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr * dx))

def subgradient_vec(h, top_k=5):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    corr_scaled = corr * dx
    top_idx = np.argsort(corr_scaled)[-top_k:][::-1]
    
    grad = np.zeros(n)
    omh = 1.0 - h
    
    for ki in top_idx:
        s = ki - (n - 1)
        # Term 1: +omh[m-s] for valid m
        a = max(0, s)
        b = min(n, s + n)
        if b > a:
            grad[a:b] += omh[a - s : b - s]
        # Term 2: -h[m+s] for valid m
        a2 = max(0, -s)
        b2 = min(n, n - s)
        if b2 > a2:
            grad[a2:b2] -= h[a2 + s : b2 + s]
    
    return -grad  # descent direction

def optimize(h0, budget_s, rng, t0):
    n = len(h0)
    target_sum = n / 2.0
    h = project_onto_feasible(h0.copy(), target_sum)
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    
    alpha = 0.05
    no_improve = 0
    
    while time.time() - t0 < budget_s:
        if no_improve > 200:
            break
        
        direction = subgradient_vec(h, top_k=5)
        norm = np.linalg.norm(direction)
        if norm < 1e-14:
            no_improve += 1
            continue

        improved = False
        for shrink in range(20):
            h_trial = project_onto_feasible(h + alpha * direction, target_sum)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5 - 1e-12:
                h = h_trial
                best_h = h_trial.copy()
                best_c5 = c5_trial
                no_improve = 0
                improved = True
                break
            alpha *= 0.7
        
        if not improved:
            alpha *= 0.7
            no_improve += 1
            if alpha < 1e-8:

import numpy as np
import time

def project_onto_feasible(h, target_sum):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-13:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(90):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr * dx))

def subgradient_vec(h, top_k=8):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    corr_scaled = corr * dx
    top_idx = np.argsort(corr_scaled)[-top_k:][::-1]
    grad = np.zeros(n)
    omh = 1.0 - h
    for ki in top_idx:
        s = ki - (n - 1)
        a = max(0, s)
        b = min(n, s + n)
        if b > a:
            grad[a:b] += omh[a - s:b - s]
        a2 = max(0, -s)
        b2 = min(n, n - s)
        if b2 > a2:
            grad[a2:b2] -= h[a2 + s:b2 + s]
    return -grad

def projected_gradient_descent(h0, max_time, rng, t_start):

    alpha = 0.04
    stall = 0

        direction = subgradient_vec(h, top_k=8)
        norm = np.linalg.norm(direction)
        if norm < 1e-15:
            stall += 10
            break
        direction /= norm

            if c5_try < best_c5 - 1e-13:
