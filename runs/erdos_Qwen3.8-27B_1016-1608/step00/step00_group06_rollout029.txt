import numpy as np
import time
from scipy.optimize import minimize

_g_n = 0

def _obj(x):
    return float(np.max(np.correlate(x, 1.0 - x, mode="full"))) * (2.0 / _g_n)

def _ct(f, x, n):
    return np.sum(x) - n / 2.0

def _ct_fun(x):
    return _ct(None, x, _g_n)

def _proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = np.sum(h)
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        s = np.sum(h)
        if abs(s - target) < 1e-12:
            break
        mask = (h > 1e-9) & (h < 1.0 - 1e-9)
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            mask = (h > 0.0) & (h < 1.0)
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
        h[mask] += (target - s) / cnt
    return np.clip(h, 0.0, 1.0)

def _c5_val(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full"))) * (2.0 / n)

    xs = np.linspace(0.0, 2.0, n_src)
    xd = np.linspace(0.0, 2.0, n_dst)
    return np.interp(xd, xs, h_src)
