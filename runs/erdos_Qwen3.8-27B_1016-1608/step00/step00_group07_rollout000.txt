import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            room[room < 1e-15] = 0
            s = room.sum()
            if s < 1e-15:
                break
            h += diff * room / s
        else:
            avail = h.copy()
            avail[avail < 1e-15] = 0
            s = avail.sum()
            if s < 1e-15:
                break
            h += diff * avail / s
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] = np.clip(h[idx] + diff, 0, 1)
    return h

def project_feasible(x, n):
    """Project x onto {0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    # Bisection on lambda
    lo, hi = -1.0, 1.0
    for _ in range(100):
        lam = (lo + hi) / 2.0
        h = np.clip(x - lam, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = lam
        else:
            hi = lam
    h = np.clip(x - lam, 0.0, 1.0)
    # Fix any numerical issues
    h = make_feasible(h, n)
    return h

def lse(val, temp):
    m = np.max(val)
    return m + temp * np.log(np.sum(np.exp((val - m) / temp)))

def smooth_obj(h_flat, n, temp):
    h = h_flat.reshape(n)
    dx = 2.0 / n
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full') * dx
    return lse(corr, temp)

def smooth_obj_grad(h_flat, n, temp):
    h = h_flat.reshape(n)
    dx = 2.0 / n
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full') * dx
    obj = lse(corr, temp)
    # Gradient of log-sum-exp
    probs = np.exp((corr - np.max(corr)) / temp)
    probs /= probs.sum()
    # Gradient of corr[k] w.r.t. h[j]
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if probs[k] < 1e-8:
            continue
        shift = k - (n - 1)
        if shift >= 0:
            for j in range(n - shift):
                grad[j] += probs[k] * (1.0 - h[j + shift])
                grad[j + shift] -= probs[k] * h[j]
        else:
            s = -shift
            for j in range(s, n):
                grad[j] += probs[k] * (1.0 - h[j - s])
                grad[j - s] -= probs[k] * h[j]
    grad *= dx
    return obj, grad

def sa_optimize(h, n, iterations, t0, t1, rng, time_limit=None, start_time=None):
    """Simulated annealing."""
    dx = 2.0 / n
    current = h.copy()
    current_c5 = compute_c5(current, dx)
    best = current.copy()
    best_c5 = current_c5
    
    for it in range(iterations):
        if time_limit is not None:
            if time.time() - start_time > time_limit:
                break
        T = t0 * (t1 / t0) ** (it / max(iterations, 1))
        if T < 1e-10:
            T = 1e-10
        
        # Choose move type
        move_type = rng.random()
        if move_type < 0.5:
            # Block move

            new_h = current.copy()
            i = int(rng.integers(0, n - bs + 1))
            idx = np.arange(i, i + bs)
            target_sum = current[idx].sum()
            new_h[idx] = current[idx] + rng.normal(0, 0.15, bs)
            new_h = make_feasible(new_h, n)
        elif move_type < 0.7:
            # Swap
            i, j = rng.integers(0, n, 2)
            new_h = current.copy()
            new_h[i], new_h[j] = current[j], current[i]
        elif move_type < 0.85:
            # Perturb single point
            i = int(rng.integers(0, n))
            new_h = current.copy()
            new_h[i] += rng.normal(0, 0.2)
            new_h = make_feasible(new_h, n)
        else:
            # Large block

            i = int(rng.integers(0, n - bs + 1))

            new_h = make_feasible(new_h, n)
        
        new_c5 = compute_c5(new_h, dx)

            current_c5 = new_c5
            if new_c5 < best_c5:

                best_c5 = new_c5
    
    return best, best_c5

def upsample(h_in, n_in, n_out):
    """Upsample h_in from n_in to n_out points, preserving integral."""
    x_in = np.linspace(0, 2, n_in, endpoint=False) + 1.0 / n_in
    x_out = np.linspace(0, 2, n_out, endpoint=False) + 1.0 / n_out
    # Interpolate (treating h as step function values at cell centers)
    h_interp = np.interp(x_out, x_in, h_in)
    return make_feasible(h_interp, n_out)

def downsample(h_in, n_in, n_out):
    """Downsample h_in from n_in to n_out points, preserving integral."""
    # Average groups
    ratio = n_in / n_out
    groups = np.zeros(n_out)
    for i in range(n_out):
        start = int(i * ratio)
        end = int((i + 1) * ratio)
        if end > n_in:
            end = n_in
        if end > start:
            groups[i] = h_in[start:end].sum() / (end - start)
    return make_feasible(groups, n_out)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            room[room < 1e-15] = 0.0
            s = room.sum()
            if s < 1e-15:
                break
            h = h + diff * room / s
        else:
            avail = h.copy()
            avail[avail < 1e-15] = 0.0
            s = avail.sum()
            if s < 1e-15:
                break
            h = h + diff * avail / s
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        idx = int(np.argmax(1.0 - h)) if diff > 0 else int(np.argmin(h))
        h[idx] = float(np.clip(h[idx] + diff, 0.0, 1.0))
    return h

def lse_val(val, temp):
    m = np.max(val)
    v = (val - m) / temp
    v = np.clip(v, -500, 500)
    return m + temp * np.log(np.sum(np.exp(v)))

def slsqp_obj_and_grad(x, n, temp):
    h = x
    dx = 2.0 / n
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full') * dx
    obj = lse_val(corr, temp)
    probs = np.exp(np.clip((corr - np.max(corr)) / temp, -500, 0))
    probs /= probs.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = probs[k]
        if pk < 1e-10:
            continue
        shift = k - (n - 1)
        if shift >= 0:
            for j in range(n - shift):
                grad[j] += pk * (1.0 - h[j + shift])
                grad[j + shift] -= pk * h[j]
        else:
            s = -shift
            for j in range(s, n):
                grad[j] += pk * (1.0 - h[j - s])
                grad[j - s] -= pk * h[j]
    grad *= dx
    return obj, grad

def upsample(h_in, n_in, n_out):
    x_in = (np.arange(n_in) + 0.5) * (2.0 / n_in)
    x_out = (np.arange(n_out) + 0.5) * (2.0 / n_out)
    h_interp = np.interp(x_out, x_in, h_in)
    return make_feasible(h_interp, n_out)

def slsqp_refine(h0, n, temps, max_iter=100, time_check=None, start_time=None, time_limit=None):
    best = h0.copy()
    dx = 2.0 / n
    best_c5 = compute_c5(best, dx)
    eq_con = {'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
              'jac': lambda x: np.ones(n)}
    bounds = [(0.0, 1.0)] * n
    for temp in temps:
        if time_check is not None and time.time() - start_time > time_limit:
            break
        def obj(x):
            return slsqp_obj_and_grad(x, n, temp)[0]
        def grad(x):
            return slsqp_obj_and_grad(x, n, temp)[1]
        try:
            res = minimize(obj, best, jac=grad, method='SLSQP',
                           bounds=bounds, constraints=[eq_con],
                           options={'maxiter': max_iter, 'ftol': 1e-12})
            h_new = np.clip(res.x, 0, 1)
            h_new = make_feasible(h_new, n)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best = h_new
                best_c5 = c5_new
        except Exception:
            pass
    return best, best_c5

    x = (np.arange(n) + 0.5) * (2.0 / n)
    freq = int(rng.integers(2, 5))
    phase = rng.uniform(0, 2 * np.pi)
    amp = rng.uniform(0.3, 0.5)
    h = 0.5 + amp * np.sin(2 * np.pi * freq * x / 2.0 + phase)
    return make_feasible(h, n)

def construct_two_hump(n, rng):
    x = (np.arange(n) + 0.5) * (2.0 / n)

    h = np.zeros(n)
    for c, w in zip(centers, widths):
        h += np.exp(-0.5 * ((x - c) / w) ** 2)
    h = h / max(h.max(), 1e-10)
    h *= 1.0
    return make_feasible(h, n)
