    diff = target - np.sum(h)

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:

        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12

            break

    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    m = np.max(corr)
    s = (corr - m) * beta
    s = np.clip(s, -50, 50)

        res = minimize(smooth_obj, h_init, args=(n, beta),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-12})

        pass

One more thought: the SLSQP optimization with n=45 and a bilinear objective might struggle. Let me also try a "subgradient" approach:
- At each iteration, find the lag k* that gives the maximum correlation
- Compute the subgradient of c[k*] with respect to h
- Take a projected gradient step: h_new = h - alpha * grad + projection to satisfy constraints

The subgradient of c[k] = sum_i h[i+k]*(1-h[i]) with respect to h[j] is:
- Terms where i+k = j: contributes (1-h[i]) for each i = j-k
- Terms where i = j: contributes -h[i+k] for each valid k' (but we only fix k)

Wait, for a FIXED lag k:
c[k] = sum_{i: valid} h[i+k] * (1-h[i])

∂c[k]/∂h[j] = sum over terms where either i+k=j or i=j:
- If i+k=j, i.e., i=j-k: term is (1-h[j-k]) if j-k is valid, and 0<=j-k<n
- If i=j: term is -h[j+k] if j+k is valid, 0<=j+k<n

So ∂c[k]/∂h[j] = (1-h[j-k]) * [0<=j-k<n] - h[j+k] * [0<=j+k<n]

where [condition] is 1 if condition is true, 0 otherwise.

This gives us a way to do subgradient descent on the active (maximum) constraint.

    best = h.copy()
    best_val = evaluate_c5(h)

        # k_max ranges from 0 to 2n-2
        # correlation index k_max corresponds to lag = k_max - (n-1)
        lag = k_max - (n - 1)

        # or equivalently, from the correlation definition:
        # result[k] = sum_n a[n+k-(n-1)] * v[n]
        # where a=h, v=1-h
        # So result[k_max] = sum_n h[n+k_max-(n-1)] * (1-h[n])
        # = sum_n h[n+lag] * (1-h[n]) where lag = k_max-(n-1)

        # h[n+lag] = h[j] when j = n+lag, i.e., n = j-lag

        # Actually let me vectorize:
        # For each j in [0,n-1], check if j-lag is in [0,n-1]
        idx1 = np.arange(n) - lag
        valid1 = (idx1 >= 0) & (idx1 < n)
        grad[valid1] += (1.0 - h[idx1[valid1]])

        idx2 = np.arange(n) + lag

    h = np.clip(h.astype(float), 0.0, 1.0).copy()

        diff = target - np.sum(h)
        if abs(diff) < 1e-13:

        else:

    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    m = np.max(corr)
    s = np.clip((corr - m) * beta, -50, 50)
    return m + np.log(np.sum(np.exp(s))) / beta
