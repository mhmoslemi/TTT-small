                h_p = fix_sum(np.clip(h_p, 0, 1), n)
                res2 = minimize(obj_smoothed, h_p, args=(n, 80.0),
                              method='L-BFGS-B', bounds=bounds,
                              options={'maxiter': 300, 'ftol': 1e-14})
                h_p = fix_sum(np.clip(res2.x, 0, 1), n)
                c5p = compute_c5(h_p, n)
                if c5p < best_c5:
                    best_c5, best_h = c5p, h_p.copy()
    
    # Phase 4: Final high-precision polish
    if time.time() < t0 + budget_s - 3:
        for tau in [200.0, 100.0, 500.0]:
            if time.time() > t0 + budget_s - 3:
                break
            res = minimize(obj_smoothed, best_h, args=(n, tau),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 2000, 'ftol': 1e-16, 'gtol': 1e-12})
            h_o = fix_sum(np.clip(res.x, 0, 1), n)
            c5 = compute_c5(h_o, n)
            if c5 < best_c5:
                best_c5, best_h = c5, h_o.copy()
    
    # Ensure constraints are met
    best_h = fix_sum(best_h, n)
    best_c5 = compute_c5(best_h, n)
    
    return best_h, best_c5, n
