                            best_c5 = c5_test

    s = best_h.sum()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def smooth_obj(h, dx, tau):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    return m + np.log(np.mean(np.exp((corr - m) * tau))) / tau

def smooth_grad(h, dx, tau):
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx
    m = np.max(vals)
    w = np.exp((vals - m) * tau)
    w /= w.sum()
    n = len(h)
    grad = np.zeros(n)
    for p in range(2 * n - 1):
        wp = w[p]
        if wp < 1e-12:
            continue
        lag = p - (n - 1)
        lo = max(0, -lag)
        hi = min(n, n - lag)
        for i in range(lo, hi):
            j = i + lag
            grad[i] += wp * (1.0 - h[j])
            grad[j] -= wp * h[i]
    grad *= dx
    obj = m + np.log(w.sum() / len(vals)) / tau
    return obj, grad

def subgrad_obj(h, dx):
    return compute_c5(h, dx)

def subgrad_fun(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    active = np.where(corr >= m - 1e-10)[0]
    n = len(h)
    grad = np.zeros(n)
    na = len(active)
    for p in active:
        lag = p - (n - 1)
        lo = max(0, -lag)
        hi = min(n, n - lag)
        for i in range(lo, hi):
            j = i + lag
            grad[i] += (1.0 - h[j]) / na
            grad[j] -= h[i] / na
    grad *= dx
    return grad

def make_start(rng, n, target_sum, mode):
    if mode == 0:
        r = rng.rand(n)
    elif mode == 1:
        x = np.linspace(0, 2, n)
        c = rng.uniform(0.5, 1.5)
        s = rng.uniform(0.3, 0.8)
        r = np.exp(-((x - c) ** 2) / (2 * s ** 2))
    elif mode == 2:
        x = np.linspace(0, 2, n)
        c1, c2 = sorted(rng.uniform(0.3, 1.7, 2))
        s = rng.uniform(0.2, 0.6)
        r = np.exp(-((x - c1) ** 2) / (2 * s ** 2)) + np.exp(-((x - c2) ** 2) / (2 * s ** 2))
    elif mode == 3:
        x = np.linspace(0, 2, n)
        f = rng.uniform(1, 4)
        r = 0.5 + 0.3 * np.cos(np.pi * f * x)
    else:
        r = rng.rand(n)
    r = np.clip(r, 0.01, 0.99)
    r *= target_sum / r.sum()
    r = np.clip(r, 0, 1)
    diff = target_sum - r.sum()
    mask = (r > 0.01) & (r < 0.99)
    if mask.sum() > 0:
        r[mask] += diff / mask.sum()
        r = np.clip(r, 0, 1)
    return r

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    n = len(initial_h_values)

        return time.time() - t0

        if sn < 1e-12:

                    ht[j] = nj

    best_c5 = compute_c5(best_h, dx)
    return (best_h, best_c5, n)
