def repair_h(h, n, target_sum):
    """Repair h to satisfy 0<=h<=1 and sum(h)=target_sum"""
    h = np.clip(h, 0, 1)
    for _ in range(20):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            n_mask = np.sum(mask)
            if n_mask == 0:
                break
            adj = diff / n_mask
            h[mask] = np.minimum(h[mask] + adj, 1.0)
        else:
            mask = h > 0.0
            n_mask = np.sum(mask)
            if n_mask == 0:
                break
            adj = diff / n_mask
            h[mask] = np.maximum(h[mask] + adj, 0.0)

    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-15:
