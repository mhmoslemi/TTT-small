                best_h = h_cd.copy()

    # Phase 4: Random restarts
    if elapsed_frac() < 0.92:
        for trial in range(8):
            if time.time() - t_start > budget_s * 0.93:
                break
            scale = 0.03 + 0.03 * trial
            h_pert = h + np.random.randn(n) * scale
            h_pert = fix_constraint(h_pert, n)
            try:
                res = minimize(obj_fn, h_pert.copy(), args=(400,), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 300, 'ftol': 1e-11})
                hx = fix_constraint(res.x, n)
                c5 = compute_c5(hx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = hx.copy()
                    h = hx.copy()
            except Exception:
                pass

    # Phase 5: Binary-rounding attempt (allow one 0.5)
    if elapsed_frac() < 0.95:
        h_bin = (best_h > 0.5).astype(float)
        # ensure sum = n/2; n is odd so need one 0.5
        s = np.sum(h_bin)
        if n % 2 == 1:
            target = n / 2.0
            if s < target:
                idx = np.argmin(best_h)
                h_bin[idx] = 0.5
            elif s > target:
                idx = np.argmax(best_h)
                h_bin[idx] = 0.5
            h_bin = fix_constraint(h_bin, n)
            c5 = compute_c5(h_bin)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_bin.copy()

    # Final validation
    best_h = fix_constraint(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
