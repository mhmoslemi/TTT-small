def get_smooth_subgrad(h, temperature=0.01):

            g[k:] -= w * h[:n-k]
        else:
            ks = -k
            g[ks:] += w * omh[:n-ks]
            g[:n-ks] -= w * h[ks:]
    
    return dx * g
