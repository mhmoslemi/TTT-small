import time

def compute_score(h):

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(c))

    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        np.clip(h, 0.0, 1.0, out=h)

def soft_max_obj(h, dx, alpha):
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    cmax = c.max()
    if cmax < 1e-15:
        return 0.0
    s = alpha * (c - cmax)
    s -= s.max()
    return cmax + np.log(np.sum(np.exp(s))) / alpha

import numpy as np
from scipy.optimize import minimize
import time

def compute_score(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(c))

def project_feasible(h, n):
    h = np.clip(h.astype(np.float64), 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h

def soft_max_obj(h, dx, alpha):
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    cmax = c.max()
    if cmax < 1e-15:
        return 0.0
    s = alpha * (c - cmax)
    s -= s.max()
    return float(cmax + np.log(np.sum(np.exp(s))) / alpha)

def lp_obj(h, dx, p):
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.sum(c ** p) ** (1.0 / p))

def sum_con(h, n):
    return h.sum() - n / 2.0

def penalty_obj(h, dx, alpha, pen, n):
    base = soft_max_obj(h, dx, alpha)
    v = h.sum() - n / 2.0
    return base + pen * v * v

def opt_slsqp_sm(h0, n, dx, alpha, miter):
    bds = [(0.0, 1.0)] * n
    cns = [{'type': 'eq', 'fun': sum_con, 'args': (n,)}]
    try:
        r = minimize(soft_max_obj, h0.copy(), args=(dx, alpha),
                     method='SLSQP', bounds=bds, constraints=cns,
                     options={'maxiter': miter, 'ftol': 1e-12})
        return project_feasible(np.clip(r.x, 0.0, 1.0), n)
    except Exception:
        return h0.copy()

def opt_slsqp_lp(h0, n, dx, p, miter):
    bds = [(0.0, 1.0)] * n
    cns = [{'type': 'eq', 'fun': sum_con, 'args': (n,)}]
    try:
        r = minimize(lp_obj, h0.copy(), args=(dx, p),
                     method='SLSQP', bounds=bds, constraints=cns,
                     options={'maxiter': miter, 'ftol': 1e-12})
        return project_feasible(np.clip(r.x, 0.0, 1.0), n)
    except Exception:
        return h0.copy()

def opt_lbfgs(h0, n, dx, alpha, pen, miter):
    bds = [(0.0, 1.0)] * n
    try:
        r = minimize(penalty_obj, h0.copy(), args=(dx, alpha, pen, n),
                     method='L-BFGS-B', bounds=bds,
                     options={'maxiter': miter, 'ftol': 1e-12})
        return project_feasible(np.clip(r.x, 0.0, 1.0), n)
    except Exception:
        return h0.copy()
