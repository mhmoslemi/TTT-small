import numpy as np
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return float(corr.max()) * dx


def fix_sum(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    return h


def local_search(h, time_limit, step_init, rng_seed):
    n = len(h)
    rng = np.random.RandomState(rng_seed)
    h = h.copy()
    c5 = compute_c5(h)
    step = step_init
    stall = 0
    t_start = time.time()
    while time.time() - t_start < time_limit:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        sign = 1.0 if rng.random() < 0.5 else -1.0
        hi, hj = h[i], h[j]
        h[i] += sign * step
        h[j] -= sign * step
        if 0.0 <= h[i] <= 1.0 and 0.0 <= h[j] <= 1.0:
            c5_new = compute_c5(h)
            if c5_new < c5 - 1e-15:
                c5 = c5_new
                stall = 0
            else:
                h[i], h[j] = hi, hj
                stall += 1
        else:
            h[i], h[j] = hi, hj
            stall += 1
        if stall > 400:
            step *= 0.95
            stall = 0
            if step < 0.0005:
                break
    return h, c5


def sim_anneal(h, time_limit, T_init, rng_seed):
    n = len(h)
    rng = np.random.RandomState(rng_seed)
    h = h.copy()
    c5 = compute_c5(h)
    T = T_init
    t_start = time.time()
    while time.time() - t_start < time_limit and T > 1e-12:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        delta = (rng.random() - 0.5) * 0.06
        hi, hj = h[i], h[j]
        h[i] += delta
        h[j] -= delta
        if 0.0 <= h[i] <= 1.0 and 0.0 <= h[j] <= 1.0:
            c5_new = compute_c5(h)
            dc = c5_new - c5
            if dc < 0.0 or rng.random() < np.exp(-dc / T):
                c5 = c5_new
            else:
                h[i], h[j] = hi, hj
        else:
            h[i], h[j] = hi, hj
        T *= 0.99997
    return h, c5


def block_search(h, time_limit, rng_seed):
    n = len(h)
    rng = np.random.RandomState(rng_seed)
    h = h.copy()
    c5 = compute_c5(h)
    t_start = time.time()
    while time.time() - t_start < time_limit:
        a = rng.randint(0, n - 1)
        b = a + rng.randint(1, min(6, n - a))
        c = rng.randint(0, n - 1)
        d = min(c + (b - a), n)

        seg_src = h[a:b].copy()
