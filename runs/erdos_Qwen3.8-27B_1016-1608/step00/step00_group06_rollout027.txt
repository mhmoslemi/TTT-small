        return h

        # Need to increase sum; prioritize variables closest to 1 but not at 1

            diff -= add

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
    else:
        order = np.argsort(h)
    for idx in order:
        if diff > 0:
            add = min(1.0 - h[idx], diff)
            h[idx] += add
            diff -= add
        else:
            sub = min(h[idx], -diff)
            h[idx] -= sub
            diff += sub
        if abs(diff) < 1e-12:
            break
    return h

def smooth_max(corr, alpha):
    m = np.max(corr)
    return m + (1.0 / alpha) * np.log(np.sum(np.exp(alpha * (corr - m))))

        h0 = np.interp(x_new, x_old, h_init)

        c5_cur = compute_c5(h0)

        def sum_con(x):
            return np.sum(x) - n / 2.0
