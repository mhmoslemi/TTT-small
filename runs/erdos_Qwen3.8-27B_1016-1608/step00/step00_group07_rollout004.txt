import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def project_feasible(h, target_sum):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tr = room.sum()
            if tr < 1e-14:
                break
            h += room * (diff / tr)
        else:
            av = h.copy()
            ta = av.sum()
            if ta < 1e-14:
                break
            h -= av * (-diff / ta)
        h = np.clip(h, 0.0, 1.0)
    return h

def resample_to(h, n_target):
    n = len(h)
    if n == n_target:
        return project_feasible(h, n_target / 2.0)
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_target / 2.0)

def optimize_local(h, n, t_list=(150.0, 70.0, 30.0, 15.0, 7.0, 3.0), maxiter_base=300):
    target_sum = n / 2.0
    dx = 2.0 / n
    h_best = project_feasible(h, target_sum)
    c5_best = compute_c5(h_best)
    n_corr = 2 * n - 1
    for T in t_list:
        mi = int(maxiter_base * min(1.0, 50.0 / max(T, 3.0)))
        def obj(x):
            vals = np.correlate(x, 1.0 - x, mode="full") * dx
            m = vals.max()
            e = np.exp(np.clip(T * (vals - m), -700, 700))
            return m + np.log(e.sum()) / T
        res = minimize(obj, h_best, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': mi, 'ftol': 1e-12, 'gtol': 1e-9})
        h_cand = project_feasible(res.x, target_sum)
        c5_cand = compute_c5(h_cand)
        if c5_cand < c5_best:
            c5_best = c5_cand
            h_best = h_cand.copy()
    return h_best, c5_best

def hill_climb_binary(h_bin, n, max_iters=5000):
    dx = 2.0 / n
    target_sum = n / 2.0
    h = h_bin.copy()
    c5 = compute_c5(h)
    for _ in range(max_iters):
        n_ones = int(h.sum())
        candidates = np.where(rng_hc < 0.5)[0]
        flip_set = np.where(h > 0.5)[0] if n_ones > target_sum else np.where(h < 0.5)[0]
        if len(candidates) == 0 or len(flip_set) == 0:
            break
        i = int(rng_hc.randint(len(flip_set)))
        j = int(rng_hc.randint(len(candidates)))
        if i == j:
            continue
        old_i, old_j = h[i], h[j]
        h[i], h[j] = h[j], h[i]
        c5_new = compute_c5(h)
        if c5_new >= c5:
            h[i], h[j] = old_i, old_j
        else:
            c5 = c5_new
    return h, c5
