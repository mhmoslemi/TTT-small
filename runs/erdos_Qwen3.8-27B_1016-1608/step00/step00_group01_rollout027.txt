    hc = h_start.copy()
    cc = compute_c5(hc)
    bc = cc
    best_local = hc.copy()
    stall = 0
    STALL_MAX = 4000
    while stall < STALL_MAX:
        if time_limit_fn():
            break
        hi = np.where(hc >= 0.5)[0]
        lo = np.where(hc < 0.5)[0]
        if len(hi) == 0 or len(lo) == 0:
            break
        a = hi[rng.integers(len(hi))]
        b = lo[rng.integers(len(lo))]
        va, vb = hc[a], hc[b]
        hc[a], hc[b] = vb, va
        cn = compute_c5(hc)
        if cn < cc - 1e-15:
            cc = cn
            stall = 0
            if cn < bc:
                bc = cn
                best_local = hc.copy()
        else:
            hc[a], hc[b] = va, vb
            stall += 1
        if stall >= STALL_MAX:
            hc = best_local.copy()
            cc = bc
            for _ in range(rng.integers(3, 10)):
                hi2 = np.where(hc >= 0.5)[0]
                lo2 = np.where(hc < 0.5)[0]
                if len(hi2) > 0 and len(lo2) > 0:
                    a2 = hi2[rng.integers(len(hi2))]
                    b2 = lo2[rng.integers(len(lo2))]
                    hc[a2], hc[b2] = hc[b2], hc[a2]
            stall = 0
    return best_local, bc

def slsqp_polish(h_start, n, ts, dx, rng, time_limit_fn, max_calls=20):
    best_h = h_start.copy()
    best_c5 = compute_c5(best_h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - ts,
             'jac': lambda h: np.ones(n)}]
    def obj(h):
        c = correlate(h, 1.0 - h, mode="full")
        return float(np.max(c)) * dx
    for att in range(max_calls):
        if time_limit_fn():
            break
        ht = best_h.copy()
        if att > 0:
            m = rng.choice(n, max(2, n // 5), replace=False)
            ht[m] += rng.normal(0, 0.1, len(m))
            ht = make_feasible(ht, ts)
        try:
            r = minimize(obj, ht, method='SLSQP', bounds=bnds,
                         constraints=cons, options={'maxiter': 200, 'ftol': 1e-14})
            if r.fun < best_c5:
                hn = make_feasible(r.x, ts)
                c5n = compute_c5(hn)
                if c5n < best_c5:
                    best_h = hn.copy()
                    best_c5 = c5n
        except Exception:
            pass
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except (NameError, UnboundLocalError):
        h0 = np.ones(47) * 0.5

    n0 = len(h0)
    best_overall_h = h0.copy()
    best_overall_c5 = compute_c5(h0)

    n_list = [n0]
    if n0 < 200:
        n_list.append(2 * n0)
    if n0 < 500:
        n_list.append(4 * n0)

    for n in n_list:
        if time.time() - t0 > budget_s - 120:
            break
        ts = n / 2.0
        dx = 2.0 / n

        if n == n0:
            h_start = h0.copy()
        else:
            dx0 = 2.0 / n0
            x0 = (np.arange(n0) + 0.5) * dx0
            x1 = (np.arange(n) + 0.5) * dx
            h_start = np.interp(x1, x0, h0)
        h_start = make_feasible(h_start, ts)

        elapsed_fn = (lambda: time.time() - t0 > budget_s - 60)
        h_ls, c5_ls = local_search(h_start, n, ts, dx, rng, elapsed_fn)

        if c5_ls < best_overall_c5:
            best_overall_h = h_ls.copy()
            best_overall_c5 = c5_ls

        h_sp, c5_sp = slsqp_polish(h_ls, n, ts, dx, rng, elapsed_fn, max_calls=15)
        if c5_sp < best_overall_c5:
            best_overall_h = h_sp.copy()
            best_overall_c5 = c5_sp

        # Second round of local search from SLSQP result
        elapsed_fn2 = (lambda: time.time() - t0 > budget_s - 30)
        h_ls2, c5_ls2 = local_search(h_sp, n, ts, dx, rng, elapsed_fn2)
        if c5_ls2 < best_overall_c5:
            best_overall_h = h_ls2.copy()
            best_overall_c5 = c5_ls2

    return best_overall_h, best_overall_c5, len(best_overall_h)
