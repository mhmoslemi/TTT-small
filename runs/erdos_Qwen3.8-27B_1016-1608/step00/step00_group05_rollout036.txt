def fix_sum(h, target):
    """Adjust h in-place to have sum = target while keeping h in [0,1]."""
    h = np.clip(h, 0, 1)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        # Need to increase some entries
        order = np.argsort(h)  # most room to grow first
        for i in order:
            if h[i] < 1.0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
                if abs(diff) < 1e-12:
                    break
    else:
        # Need to decrease some entries
        order = np.argsort(-h)  # entries closest to 1 first
        for i in order:
            if h[i] > 0.0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
                if abs(diff) < 1e-12:

import numpy as np

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx), corr

def fix_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        idx = np.argsort(-room)
        for i in idx:
            if diff <= 0:
                break
            add = min(diff, room[i])
            h[i] += add
            diff -= add
    else:
        idx = np.argsort(h)  # smallest first
        for i in idx:
            if diff >= 0:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

        idx = np.argmax(corr)
