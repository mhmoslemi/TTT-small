    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return np.max(corr) * dx

def local_search_binary(h, rng, patience=50):
    """Hill climbing with single-swap moves."""
    n = len(h)
    h = h.copy()
    score = compute_c5(h)
    no_improve = 0
    
    while no_improve < patience:
        improved = False
        ones = list(np.where(h == 1)[0])
        zeros = list(np.where(h == 0)[0])
        rng.shuffle(ones)
        rng.shuffle(zeros)
        
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                new_score = compute_c5(h)
                if new_score < score - 1e-12:
                    score = new_score
                    improved = True
                    break
                h[i] = 1.0
                h[j] = 0.0
            if improved:
                break
        
        if not improved:
            no_improve += 1
        else:
            no_improve = 0
    
    return h, score

def simulated_annealing(h, rng, n_steps=50000, T0=0.5):
    """Simulated annealing for binary optimization."""
    n = len(h)
    h = h.copy()
    score = compute_c5(h)
    T = T0
    cooling = 0.99997
    
    for step in range(n_steps):
        T *= cooling
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5(h)
        
        if new_score <= score or rng.rand() < np.exp(-(new_score - score) / max(T, 1e-10)):
            score = new_score
        else:
            h[i] = 1.0
            h[j] = 0.0

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def hill_climb(h, rng, max_rounds=500):
    n = len(h)
    h = h.astype(float).copy()
    score = compute_c5(h)
    for r in range(max_rounds):
        ones = np.where(h >= 0.999)[0]
        zeros = np.where(h <= 0.001)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_delta = 0.0
        best_pair = None
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                ns = compute_c5(h)
                d = score - ns
                h[i] = 1.0
                h[j] = 0.0
                if d > best_delta + 1e-14:
                    best_delta = d
                    best_pair = (i, j)
        if best_pair is not None:
            i, j = best_pair
            h[i] = 0.0
            h[j] = 1.0
            score -= best_delta
        else:
            break
    return h, score

def pair_hill_climb(h, rng, max_rounds=50):
    n = len(h)
    h = h.astype(float).copy()
    score = compute_c5(h)
    for r in range(max_rounds):
        ones = np.where(h >= 0.999)[0]
        zeros = np.where(h <= 0.001)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        best_delta = 0.0
        best_move = None
        rng.shuffle(ones)
        rng.shuffle(zeros)
        for ii in range(len(ones)):
            for jj in range(ii + 1, len(ones)):
                h[ones[ii]] = 0.0
                h[ones[jj]] = 0.0
                for kk in range(len(zeros)):
                    for ll in range(kk + 1, len(zeros)):
                        h[zeros[kk]] = 1.0
                        h[zeros[ll]] = 1.0
                        ns = compute_c5(h)
                        d = score - ns
                        h[zeros[kk]] = 0.0
                        h[zeros[ll]] = 0.0
                        if d > best_delta + 1e-14:
                            best_delta = d
                            best_move = (ones[ii], ones[jj], zeros[kk], zeros[ll])
                        h[zeros[kk]] = 0.0
                h[ones[ii]] = 1.0
                h[ones[jj]] = 1.0
        if best_move is not None:
            i1, i2, j1, j2 = best_move
            h[i1] = 0.0; h[i2] = 0.0; h[j1] = 1.0; h[j2] = 1.0
            score -= best_delta
        else:
            break
    return h, score

def simulated_annealing(h, rng, n_steps=200000, T0=0.3):
    n = len(h)
    h = h.astype(float).copy()
    score = compute_c5(h)
    T = T0
    cool = (0.01 / T0) ** (1.0 / n_steps)
    best_h = h.copy()
    best_score = score
    for step in range(n_steps):
        T *= cool
        ones = np.where(h >= 0.999)[0]
        zeros = np.where(h <= 0.001)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        ns = compute_c5(h)
        if ns <= score or rng.rand() < np.exp(-(ns - score) / max(T, 1e-12)):
            score = ns
            if score < best_score:
                best_score = score
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_score

def continuous_refine(h0, rng, max_iters=5000):
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h = h0.astype(float).copy()
    score = compute_c5(h)
    lr = 0.02
    for it in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        grad = np.zeros(n)
        if k >= 0:
            valid_i = np.arange(0, n - k)
            for idx in valid_i:
                j = idx + k
                grad[j] += (1.0 - h[idx])
                grad[idx] -= h[j]
        else:
            valid_i = np.arange(-k, n)
            for idx in valid_i:
                j = idx + k
                if 0 <= j < n:
                    grad[j] += (1.0 - h[idx])
                    grad[idx] -= h[j]
        h -= lr * grad * dx
        h = np.clip(h, 0.0, 1.0)
        s = np.sum(h)
        if abs(s - target_sum) > 1e-8:
            h = h * (target_sum / max(s, 1e-12))
            h = np.clip(h, 0.0, 1.0)
            diff = target_sum - np.sum(h)
            if abs(diff) > 1e-8:
                if diff > 0:
                    mask = h < 0.999
                else:
                    mask = h > 0.001
                if mask.sum() > 0:
                    h[mask] += diff / mask.sum()
                    h = np.clip(h, 0.0, 1.0)
        ns = compute_c5(h)
        if ns < score:
            score = ns
        if it % 500 == 0 and lr > 0.0005:
            lr *= 0.85
    return h, score

def fix_sum(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-10:
        return h
    h = h * (target / max(s, 1e-12))
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 0.999
        else:
            mask = h > 0.001
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)
