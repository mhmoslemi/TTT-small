    best_c5 = c5
    best_h = h.copy()
    temp = 0.008
    t_start = time.time()
    t_span = max(t_end - t_start, 1.0)
    while time.time() < t_end:
        frac = (time.time() - t_start) / t_span
        temp = 0.008 * (1.0 - frac) + 1e-9
        i = int(lrng.integers(0, n))
        j = int(lrng.integers(0, n))
        if i == j:
            continue
        oi, oj = h[i], h[j]
        if lrng.random() < 0.5:
            md = min(oi, 1.0 - oj)
            if md < 1e-12:
                continue
            d = lrng.uniform(1e-6, md)
            h[i] = oi - d
            h[j] = oj + d
        else:
            md = min(oj, 1.0 - oi)
            if md < 1e-12:
                continue
            d = lrng.uniform(1e-6, md)
            h[j] = oj - d
            h[i] = oi + d
        nc5 = evaluate_c5(h)
        accept = nc5 < c5 or lrng.random() < np.exp(-(nc5 - c5) / max(temp, 1e-10))
        if accept:
            c5 = nc5
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[i] = oi
            h[j] = oj
    return best_c5, best_h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 6.0

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        h_init = np.full(45, 0.5)
    n0 = len(h_init)
    h_init = make_feasible(h_init, n0)

    best_h = h_init.copy()
    best_c5 = evaluate_c5(best_h)

    # Phase 1: Greedy local search at n0
    t1 = t0 + 0.35 * (budget_s - 15)
    if time.time() < t1:
        c5, hs = local_search(h_init.copy(), n0, seed, min(t1, deadline))
        if c5 < best_c5:
            best_c5, best_h = c5, hs

    # Phase 2: Simulated annealing at n0
    t2 = t0 + 0.55 * (budget_s - 15)
    if time.time() < t2:
        c5, hs = anneal_search(best_h.copy(), n0, seed + 7, min(t2, deadline))
        if c5 < best_c5:
            best_c5, best_h = c5, hs

    # Phase 3: Iterated local search with kicks
    t3 = t0 + 0.80 * (budget_s - 15)
    if time.time() < t3:
        irl = np.random.default_rng(seed + 200)
        it = 0
        while time.time() < t3 and it < 15:
            hs = best_h.copy()
            nk = int(irl.integers(2, 8))
            for _ in range(nk):
                a = int(irl.integers(0, n0))
                b = int(irl.integers(0, n0))
                if a != b:
                    oa, ob = hs[a], hs[b]
                    if irl.random() < 0.5:
                        md = min(oa, 1.0 - ob)
                    else:
                        a, b, oa, ob, = b, a, ob, oa
                        md = min(oa, 1.0 - ob)
                    if md > 0.02:
                        d = irl.uniform(0.04, min(md, 0.45))
                        hs[a] -= d
                        hs[b] += d
            hs = make_feasible(hs, n0)
            t_sub = min(t3, time.time() + 0.25 * (t3 - time.time() + 1.0))
            c5, hs = local_search(hs, n0, seed + 800 + it, t_sub)
            if c5 < best_c5:
                best_c5, best_h = c5, hs
            it += 1

    # Phase 4: Other resolutions
    t4 = deadline
    if time.time() < t4 - 15:
        ns_try = []
        for nt in [n0 - 2, n0 + 2, n0 + 5, n0 + 10, 60, 72, 90, 108]:
            if 4 <= nt <= 200 and nt != n0 and nt not in ns_try:
                ns_try.append(nt)
        tl = max(0.0, t4 - 15 - time.time())
        tper = max(25.0, tl / max(1, len(ns_try)))
        for nt in ns_try:
            if time.time() >= t4 - 8:
                break
            ht = interp_n(best_h, nt)
            ct = evaluate_c5(ht)
            if ct < best_c5:
                best_c5, best_h = ct, ht
            tsub = min(time.time() + tper, t4 - 8)
            if tsub > time.time() + 5:
                ct, ht = local_search(ht, nt, seed + 5000 + nt, tsub)
                if ct < best_c5:
                    best_c5, best_h = ct, ht

    # Final polish at best resolution
    if time.time() < t4 - 3:
        fn = len(best_h)
        c5, hs = local_search(best_h.copy(), fn, seed + 9999, min(time.time() + 15, t4))
        if c5 < best_c5:
            best_c5, best_h = c5, hs

    fn = len(best_h)
    best_h = make_feasible(best_h, fn)
    fc5 = evaluate_c5(best_h)
    if fc5 < best_c5:
        best_c5 = fc5

    return best_h, best_c5, fn
