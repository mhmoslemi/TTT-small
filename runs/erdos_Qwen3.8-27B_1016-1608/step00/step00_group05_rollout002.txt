import numpy as np
import time

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if diff > 1e-14:
        idx = np.argsort(1.0 - h)
        for ii in idx:
            room = 1.0 - h[ii]
            add = min(room, diff)
            h[ii] += add
            diff -= add
            if diff <= 1e-14:
                break
    elif diff < -1e-14:
        idx = np.argsort(h)
        for ii in idx:
            sub = min(h[ii], -diff)
            h[ii] -= sub
            diff += sub
            if diff >= -1e-14:
                break
    return h

def eval_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_subgrad(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(corr)
    tol = 1e-10 + 1e-8 * max_val
    active = np.where(corr >= max_val - tol)[0]
    grad = np.zeros(n)
    for k_idx in active:
        lag = k_idx - (n - 1)
        for j in range(n):
            if 0 <= j - lag < n:
                grad[j] += (1.0 - h[j - lag])
            if 0 <= j + lag < n:
                grad[j] += (-h[j + lag])
    grad *= dx
    return grad, float(max_val * dx)

def subgrad_optimize(h, n, max_iters, time_limit_fn):

    best_local_c5 = eval_c5(h, n)
    best_local_h = h.copy()

        if time_limit_fn():
            break
        grad, c5_cur = compute_subgrad(h, n)
        momentum = beta * momentum + (1 - beta) * grad
        h_new = project_h(h - lr * momentum, n)
        new_c5 = eval_c5(h_new, n)
        if new_c5 < c5_cur - 1e-15:

                stall = 0
                momentum *= 0.1
    return best_local_h, best_local_c5

def coordinate_descent(h, n, time_limit_fn, max_rounds=500):

    for rnd in range(max_rounds):

            break
        improved = False

            if time_limit_fn():

            for trial_val in [max(0, orig - 0.05), max(0, orig - 0.02),

                h[j] = trial_val
                h = project_h(h, n)

                    best_h = h.copy()
                    improved = True
                    break
