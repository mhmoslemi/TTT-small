import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def repair(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-9
            if not mask.any():
                break
            step = diff / mask.sum()
            h[mask] = np.minimum(1.0, h[mask] + step)
        else:
            mask = h > 1e-9
            if not mask.any():
                break
            step = -diff / mask.sum()
            h[mask] = np.maximum(0.0, h[mask] - step)
    diff = target - h.sum()
    if abs(diff) > 1e-9:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def sa_continuous(h, n, deadline, rng):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    T = 0.04
    cool = 0.9997
    target = n / 2.0

        T = max(1e-6, T * cool)
        k = rng.integers(1, 4)
        idxs = rng.choice(n, k, replace=False)
        h_new = h.copy()
        for j in idxs:
            h_new[j] = np.clip(h_new[j] + rng.normal(0, 0.12), 0, 1)
        h_new = repair(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < cur_c5 or rng.random() < np.exp(-max(0, c5_new - cur_c5) / T):
            h = h_new
            cur_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
    return best_h, best_c5

def binary_ga(n, pop_size, deadline, rng):
    target = n // 2
    pop = np.zeros((pop_size, n))
    for i in range(pop_size):
        pop[i, rng.choice(n, target, replace=False)] = 1.0

    bi = np.argmin(fit)
    best_h = pop[bi].copy()
    best_c5 = fit[bi]
    stagnant = 0
    while time.time() < deadline:
        new_pop = [best_h.copy()]
        while len(new_pop) < pop_size:
            r = rng.random()
            if r < 0.6:
                child = best_h.copy()

                ones = np.where(child == 1)[0]

                for _ in range(nf):
                    if len(ones) == 0 or len(zeros) == 0:
                        break
                    o = rng.choice(ones)
                    z = rng.choice(zeros)
                    child[o] = 0.0
                    child[z] = 1.0
                    ones = np.where(child == 1)[0]
                    zeros = np.where(child == 0)[0]
                new_pop.append(child)
            elif r < 0.85:
                i1 = rng.integers(0, pop_size)
                i2 = rng.integers(0, pop_size)
