        h0 = np.array(hv, dtype=float).copy()
    else:
        h0 = 0.5 * np.ones(70)
    n = len(h0)
    h0 = repair(h0, n)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    def tl():
        return budget_s - (time.time() - t0)

    # Phase 1: SLSQP with smooth max, increasing sharpness
    for alpha in [15, 40, 100, 250]:
        if tl() < 25:
            break
        def obj1(h, a=alpha, nn=n):
            hc = np.clip(h, 0.0, 1.0)
            corr = np.correlate(hc, 1.0 - hc, mode="full") * (2.0 / nn)
            return smooth_max(corr, a)
        def eqc(h):
            return np.sum(h) - n / 2.0
        res = optimize.minimize(obj1, best_h.copy(), method='SLSQP',
                                bounds=[(0.0, 1.0)] * n,
                                constraints=[{'type': 'eq', 'fun': eqc}],
                                options={'maxiter': 600, 'ftol': 1e-13, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            hc = repair(res.x, n)
            c5c = compute_c5(hc)
            if c5c < best_c5:
                best_h = hc
                best_c5 = c5c

    # Phase 2: Nelder-Mead with penalty
    if tl() > 90:
        def obj2(h):
            hc = np.clip(h, 0.0, 1.0)
            pen = 300.0 * abs(np.sum(hc) - n / 2.0)
            corr = np.correlate(hc, 1.0 - hc, mode="full") * (2.0 / n)
            return float(np.max(corr)) + pen
        res2 = optimize.minimize(obj2, best_h.copy(), method='Nelder-Mead',
                                 options={'maxiter': 6000, 'xatol': 1e-13,
                                          'fatol': 1e-14, 'adaptive': True})
        if res2.x is not None and np.all(np.isfinite(res2.x)):
            hc = repair(res2.x, n)
            c5c = compute_c5(hc)
            if c5c < best_c5:
                best_h = hc
                best_c5 = c5c

    # Phase 3: Binary local search
    if tl() > 50:
        h_bin = (best_h >= 0.5).astype(float)
        target = n // 2
        n_ones = int(np.sum(h_bin))
        if n_ones > target:
            order = np.argsort(best_h[h_bin == 1])
            h_bin[np.where(h_bin == 1)[0][len(order) - (n_ones - target):]] = 0
        elif n_ones < target:
            order = np.argsort(-best_h[h_bin == 0])
            h_bin[np.where(h_bin == 0)[0][:target - n_ones]] = 1
        h_b, c5_b = binary_search(h_bin, n, max_iter=30000, deadline=t0 + budget_s)
        if c5_b < best_c5:
            best_h = h_b
            best_c5 = c5_b

    # Phase 4: Continuous local search (perturbation + repair)
    if tl() > 30:
        rng = np.random.RandomState(seed + 3)
        step = 0.012
        stall = 0
        while tl() > 12:
            h_try = best_h.copy()
            kp = max(1, min(n, int(rng.poisson(2)) + 1))
            idx = rng.choice(n, kp, replace=False)
            h_try[idx] += rng.uniform(-step, step, kp)
            h_try = repair(h_try, n)
            c5t = compute_c5(h_try)
            if c5t < best_c5 - 1e-14:
                best_h = h_try
                best_c5 = c5t
                stall = 0
                step = min(step * 1.04, 0.05)
            else:
                stall += 1
                if stall % 80 == 0:
                    step *= 0.7
                    if step < 0.0008:
                        h_try = best_h + rng.normal(0, 0.06, n)
                        h_try = repair(h_try, n)
                        c5t = compute_c5(h_try)
                        if c5t < best_c5:
                            best_h = h_try
                            best_c5 = c5t
                        step = 0.012
                        stall = 0

    # Phase 5: Try neighboring n
    if tl() > 45:
        for dn in [1, -1, 2, -2, 3, -3, 4, -4]:
            tn = n + dn
            if tn < 30 or tn > 250 or tl() < 15:
                continue
            xo = np.linspace(0, 2, n)
            xn = np.linspace(0, 2, tn)
            hi = np.interp(xn, xo, best_h)
            hi = repair(hi, tn)
            def obj_t(h, tt=tn):
                hc = np.clip(h, 0, 1)
                corr = np.correlate(hc, 1.0 - hc, mode="full") * (2.0 / tt)
                return smooth_max(corr, 100.0)
            def eq_t(h, tt=tn):
                return np.sum(h) - tt / 2.0
            res_t = optimize.minimize(obj_t, hi, method='SLSQP',
                                      bounds=[(0, 1)] * tn,
                                      constraints=[{'type': 'eq', 'fun': eq_t}],
                                      options={'maxiter': 400, 'ftol': 1e-12})
            if res_t.x is not None and np.all(np.isfinite(res_t.x)):
                hc = repair(res_t.x, tn)
                c5c = compute_c5(hc)
                if c5c < best_c5:
                    best_h = hc
                    best_c5 = c5c
                    n = tn

    # Final polish SLSQP
    if tl() > 20:
        def obj_final(h, nn=n):
            hc = np.clip(h, 0.0, 1.0)
            corr = np.correlate(hc, 1.0 - hc, mode="full") * (2.0 / nn)
            return smooth_max(corr, 300.0)
        def eq_f(h, nn=n):
            return np.sum(h) - nn / 2.0
        res_f = optimize.minimize(obj_final, best_h.copy(), method='SLSQP',
                                  bounds=[(0.0, 1.0)] * n,
                                  constraints=[{'type': 'eq', 'fun': eq_f}],
                                  options={'maxiter': 300, 'ftol': 1e-14, 'iprint': 0})
        if res_f.x is not None and np.all(np.isfinite(res_f.x)):
            hc = repair(res_f.x, n)
            c5c = compute_c5(hc)
            if c5c < best_c5:
                best_h = hc
                best_c5 = c5c

    best_h = repair(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, len(best_h))
