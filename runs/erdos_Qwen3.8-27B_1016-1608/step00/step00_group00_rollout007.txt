import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_onto_set(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return interp1d(x_old, h, kind='linear', bounds_error=False, fill_value=0)(x_new)

def make_binary_random(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h

def make_smooth_random(n, rng):
    # Random sum of sines, projected
    x = np.linspace(0, 2, n)
    h = np.zeros(n)
    for _ in range(rng.randint(2, 6)):
        freq = rng.randint(1, 8)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.3, 0.7)
        h += amp * np.sin(freq * np.pi * x + phase)
    h = 0.5 + h * 0.4
    return project_onto_set(h, n / 2.0)

        return compute_c5(h, n)

                      constraints=constraints,

            h_opt = project_onto_set(res.x, n / 2.0)
            return h_opt, compute_c5(h_opt, n)
    except Exception:
        pass

    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_set(h, target):
    """Project h to {x: 0<=x<=1, sum(x)=target} via bisection on offset."""
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def upsample(h, n_new):

    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h)

def make_random_smooth(n, rng):

    n_terms = rng.randint(2, 6)
    for _ in range(n_terms):
        f = rng.randint(1, 8)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.2, 0.6)
        h += a * np.sin(f * np.pi * x + p)
    h = 0.5 + 0.4 * h
    return project_to_set(h, n / 2.0)

def slsqp_optimize(h0, n, maxiter=500):
    target = n / 2.0
    h0 = project_to_set(h0.copy(), target)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target}]
    
    def obj(x):
        return compute_c5(x, n)
    
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        h_out = project_to_set(res.x.copy(), target)
        return h_out, compute_c5(h_out, n)
    except Exception:
        return h0, compute_c5(h0, n)

def coordinate_search(h, n, rng, n_sweeps=50):
    """Simple coordinate descent local search."""
    target = n / 2.0
    h = h.copy()
    best = compute_c5(h, n)
    
    for sweep in range(n_sweeps):
        order = rng.permutation(n)
        for j in order:
            old = h[j]
            # Try a few candidate values
            cands = np.array([0.0, 1.0, max(0.0, old - 0.1), min(1.0, old + 0.1),
                            max(0.0, old - 0.05), min(1.0, old + 0.05),
                            max(0.0, old - 0.02), min(1.0, old + 0.02)])
            best_cand = old
            best_val = best
            h[j] = old
            for c in cands:
                if abs(c - old) < 1e-6:
                    continue
                h[j] = c
                h = project_to_set(h, target)
                val = compute_c5(h, n)
                if val < best_val:
                    best_val = val
                    best_cand = c
            if best_cand != old:
                h[j] = best_cand
                h = project_to_set(h, target)
                best = best_val
            else:
                h[j] = old
                h = project_to_set(h, target)
    return h, best
