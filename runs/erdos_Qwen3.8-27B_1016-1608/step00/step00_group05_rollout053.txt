import numpy as np
from scipy.optimize import minimize
import time

def project_onto_box_sum(h, target_sum):
    """Project h onto {x : sum(x) = target_sum, 0 <= x[i] <= 1}"""
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if mask.sum() == 0:
            h[:] = target_sum / len(h)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_values(h):
    """Compute all correlation values (not yet scaled by dx)"""
    return np.correlate(h, 1.0 - h, mode='full')

def c5_value(h, dx):
    return np.max(corr_values(h)) * dx

def smooth_max(corr, temp):
    """Smooth max via log-sum-exp"""
    c = np.max(corr)
    if c == 0:
        return 0.0
    return c + temp * np.log(np.sum(np.exp((corr - c) / temp)))

import numpy as np
from scipy.optimize import minimize
import time


def project_onto_box_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if mask.sum() == 0:
            h[:] = target_sum / len(h)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def corr_values(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_value(h, dx):
    return float(np.max(corr_values(h)) * dx)


def corr_gradient(h, weights):
    """Gradient of weighted sum of correlation values w.r.t. h.
    weights has length 2n-1, indexed by k. lag = k - (n-1)."""
    n = len(h)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if abs(w) < 1e-12:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            # d corr[k]/dh[j] = (1-h[j-lag]) for j in [lag, n) minus h[j+lag] for j in [0, n-lag)
            if lag < n:
                grad[lag:] += w * (1.0 - h[:n - lag])
            if lag < n:
                grad[:n - lag] -= w * h[lag:]
        else:
            L = -lag
            if L < n:
                grad[:n - L] += w * (1.0 - h[L:])
                grad[L:] -= w * h[:n - L]
    return grad

        return c5_value(h, dx)

        weights[k_max] = 1.0
        return objective(h), dx * corr_gradient(h, weights)

    bounds = [(0.0, 1.0)] * n

        return h_res, c5_value(h_res, dx)
    except Exception:
        return h_init.copy(), c5_value(h_init, dx)

        grad = corr_gradient(h, weights) * dx
