        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)


def interp_to_n(h_parent, n_new):
    if len(h_parent) == n_new:
        return h_parent.copy()
    x_old = np.linspace(0.0, 2.0, len(h_parent), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.clip(np.interp(x_new, x_old, h_parent), 0.0, 1.0)


def obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    exp_c = np.exp(beta * (corr - max_c))
    s = np.sum(exp_c)
    w = exp_c / s
    obj = (max_c + np.log(s)) / beta * dx
    grad = np.zeros(n)
    for k in range(len(corr)):
        m = k - n + 1
        wk = w[k]
        if wk < 5e-8:
            continue
        lo1 = max(0, m)
        hi1 = min(n, m + n)
        if hi1 > lo1:
            grad[lo1:hi1] += wk * (1.0 - h[lo1 - m:hi1 - m])
        lo2 = max(0, -m)
        hi2 = min(n, n - m)
        if hi2 > lo2:
            grad[lo2:hi2] -= wk * h[lo2 + m:hi2 + m]
    grad *= dx
    return obj, grad


def pgd_solve(h_init, n, beta, lr0, n_iter, time_lim):
    target = n / 2.0
    h = project_feasible(h_init.copy(), target)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    t0 = time.time()
    lr = lr0
    prev_c5 = best_c5
    stall = 0
    for _ in range(n_iter):
        if time.time() - t0 > time_lim:
            break
        _, grad = obj_grad(h, beta)
        h_new = project_feasible(h - lr * grad, target)
        c5_new = compute_c5(h_new)
        if c5_new < prev_c5 - 1e-12:
            prev_c5 = c5_new
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            lr = min(lr * 1.04, 8.0)
            stall = 0
        else:
            h_half = project_feasible(h - lr * 0.5 * grad, target)
            c5_half = compute_c5(h_half)
            if c5_half < prev_c5 - 1e-12:
                prev_c5 = c5_half
                h = h_half
                lr = lr * 0.7
                stall = 0
                if c5_half < best_c5:
                    best_c5 = c5_half
                    best_h = h.copy()
            else:
                lr *= 0.5
                stall += 1
                if stall > 60:
                    break
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float)
    else:
        h_parent = np.full(95, 0.5)
    n_parent = len(h_parent)

    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    best_n = n_parent

    def time_left():
        return budget_s - (time.time() - t_start)

    n_list = sorted(set([n_parent, 50, 60, 70, 80, 90, 100, 110, 120, 130, 150]))
    n_list = [n for n in n_list if 30 <= n <= 250]

    for n in n_list:
        if time_left() < 40:
            break
        h_init = project_feasible(interp_to_n(best_h, n), n / 2.0)
        for beta in [20.0, 50.0, 100.0]:
            if time_left() < 15:
                break
            t_lim = min(time_left() * 0.12, 35.0)
            h_opt, c5_opt = pgd_solve(h_init, n, beta, 0.8, 4000, t_lim)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = n
                h_init = h_opt.copy()
        # Random perturbation restart
        if time_left() > 30:
            h_pert = h_init + np.random.normal(0, 0.1, n)
            h_pert = project_feasible(h_pert, n / 2.0)
            t_lim = min(time_left() * 0.08, 25.0)
            h_opt, c5_opt = pgd_solve(h_pert, n, 60.0, 0.5, 2500, t_lim)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = n

    # Final high-beta polish
    if time_left() > 25:
        t_lim = min(time_left() * 0.35, 50.0)
        h_fin, c5_fin = pgd_solve(best_h, best_n, 300.0, 0.3, 8000, t_lim)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin

    # Final feasibility check
    best_h = project_feasible(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
