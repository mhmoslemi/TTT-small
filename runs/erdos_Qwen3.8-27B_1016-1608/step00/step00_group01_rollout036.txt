import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def smooth_max(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    mx = np.max(corr)
    return float(mx + np.log(np.sum(np.exp(alpha * (corr - mx)))) / alpha)

def fix_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-10
    else:
        mask = h > 1e-10
    if np.sum(mask) == 0:
        return h
    h = h.copy()
    h[mask] += diff / np.sum(mask)
    return np.clip(h, 0.0, 1.0)

    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(47, 0.5)

    n = len(h0)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint, 'args': (target,)}]

    best_h = fix_sum(h0, target)
    best_c5 = evaluate_c5(best_h)

    elapsed = lambda: time.time() - t0

    # Build diverse starting points
    starts = [best_h.copy()]
    for i in range(6):
        starts.append(make_perturbed_start(rng, n, target, best_h, 0.05 + 0.03 * i))
    for i in range(4):
        starts.append(make_random_start(rng, n, target))
    for freq in [1, 2, 3]:
        starts.append(make_sinusoidal_start(rng, n, target, freq, 0.25))

    # Main optimization: continuation in alpha
    alphas = [10, 30, 100, 300, 1000]
    for s_i in range(len(starts)):
        if elapsed() > budget_s - 150:
            break
        h_curr = starts[s_i].copy()
        for alpha in alphas:
            if elapsed() > budget_s - 80:
                break
            try:
                res = minimize(smooth_max_c5, h_curr, args=(alpha,),
                               method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 120, 'ftol': 1e-10})
                h_new = fix_sum(res.x, target)
                c5 = evaluate_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()
                h_curr = res.x.copy()
            except Exception:
                pass

    # Polish with high alpha
    for alpha in [3000, 10000, 30000]:
        if elapsed() > budget_s - 60:
            break
        try:
            res = minimize(smooth_max_c5, best_h, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 400, 'ftol': 1e-12})
            h_new = fix_sum(res.x, target)
            c5 = evaluate_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
        except Exception:
            pass

    # Try n=48 (allows binary solutions)
    n48 = 48
    target48 = 24.0
    if elapsed() < budget_s - 200:
        bounds48 = [(0.0, 1.0)] * n48
        cons48 = [{'type': 'eq', 'fun': sum_constraint, 'args': (target48,)}]
        # Interpolate best to n=48
        xp = np.linspace(0, 2, n, endpoint=False)
        x48 = np.linspace(0, 2, n48, endpoint=False)
        h48 = np.interp(x48, xp, np.concatenate([best_h, [best_h[0]]]))
        h48 = fix_sum(h48, target48)
        # Also try a "spiky" start
        h_spiky = np.zeros(n48)
        h_spiky[::2] = 1.0
        h_spiky[:24] = 1.0
        h_spiky = fix_sum(h_spiky, target48)

        for h_s in [h48, h_spiky]:
            h_curr48 = h_s.copy()
            for alpha in [30, 100, 300, 1000]:
                if elapsed() > budget_s - 100:
                    break
                try:
                    res = minimize(smooth_max_c5, h_curr48, args=(alpha,),
                                   method='SLSQP', bounds=bounds48, constraints=cons48,
                                   options={'maxiter': 100, 'ftol': 1e-10})
                    h_new = fix_sum(res.x, target48)
                    c5 = evaluate_c5(h_new)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_new.copy()
                        n = n48
                        target = target48
                        bounds = bounds48
                        cons = cons48
                    h_curr48 = res.x.copy()
                except Exception:
                    pass

    # Try n=93 for finer resolution
    n93 = 93
    target93 = 46.5
    if elapsed() < budget_s - 120:
        bounds93 = [(0.0, 1.0)] * n93
        cons93 = [{'type': 'eq', 'fun': sum_constraint, 'args': (target93,)}]
        xp = np.linspace(0, 2, len(best_h), endpoint=False)
        x93 = np.linspace(0, 2, n93, endpoint=False)
        h93 = np.interp(x93, xp, np.concatenate([best_h, [best_h[0]]]))
        h93 = fix_sum(h93, target93)
        h_curr93 = h93.copy()
        for alpha in [30, 100, 300, 1000]:
            if elapsed() > budget_s - 50:
                break
            try:
                res = minimize(smooth_max_c5, h_curr93, args=(alpha,),
                               method='SLSQP', bounds=bounds93, constraints=cons93,
                               options={'maxiter': 80, 'ftol': 1e-10})
                h_new = fix_sum(res.x, target93)
                c5 = evaluate_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()
                    n = n93
                h_curr93 = res.x.copy()
            except Exception:
                pass

    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, n
