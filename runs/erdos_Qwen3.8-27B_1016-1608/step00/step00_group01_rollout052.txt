import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    return float(np.max(corr) * dx), corr * dx

def repair_h(h, n_points):
    target = n_points / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        mask = (h > 1e-6) & (h < 1.0 - 1e-6)
        if np.sum(mask) > 0:
            adj = diff / np.sum(mask)
            h[mask] = np.clip(h[mask] + adj, 0.0, 1.0)
            diff2 = target - np.sum(h)
            if abs(diff2) > 1e-12:
                mask2 = (h > 0) & (h < 1)
                if np.sum(mask2) > 0:
                    h[mask2] += diff2 / np.sum(mask2)
                    h = np.clip(h, 0.0, 1.0)
        else:
            h[:] += diff / n_points
            h = np.clip(h, 0.0, 1.0)
    return h

def corr_jacobian(h, n):
    n_corr = 2 * n - 1
    J = np.zeros((n_corr, n))
    for k in range(n_corr):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            jb = i + k - (n - 1)
            if jb == i:
                J[k, i] += (1.0 - 2.0 * h[i])
            else:
                J[k, i] += (1.0 - h[jb])
                J[k, jb] += (-h[i])
    return J

def slsqp_solve(h_init, n_points, tau=40.0, maxiter=300):
    n_p = n_points
    dx = 2.0 / n_p
    target_sum = n_p / 2.0
    n_corr = 2 * n_p - 1

    def objective(x):
        g = 1.0 - x
        corr = np.correlate(x, g, mode='full') * dx
        m = np.max(corr)
        ev = np.exp(tau * (corr - m))
        return float(m + np.log(np.sum(ev)) / tau)

    def jac(x):
        g = 1.0 - x
        corr = np.correlate(x, g, mode='full') * dx
        m = np.max(corr)
        ev = np.exp(tau * (corr - m))
        w = ev / np.sum(ev)
        J = corr_jacobian(x, n_p)
        return (J.T @ w) * dx

    constraints = [
        {'type': 'eq',
         'fun': lambda x: np.sum(x) - target_sum,
         'jac': lambda x: np.ones(n_p)}
    ]
    bounds = [(0.0, 1.0)] * n_p

    try:
        res = minimize(objective, h_init.copy(), method='SLSQP', jac=jac,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        h_opt = np.clip(res.x, 0.0, 1.0)
        h_opt = repair_h(h_opt, n_p)
        return h_opt
    except Exception:
        return h_init.copy()
