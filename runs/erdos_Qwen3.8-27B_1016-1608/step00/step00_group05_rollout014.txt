import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = h.sum() - target_sum
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h > 1e-14
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] -= diff / cnt
        else:
            mask = h < 1.0 - 1e-14
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += (-diff) / cnt
        h = np.clip(h, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def subgrad_for_corr(h, offset, n):
    g = np.zeros(n)
    j_lo = max(0, -offset)
    j_hi = min(n, n - offset)
    if j_hi > j_lo:
        g[j_lo:j_hi] += (1.0 - h[j_lo + offset : j_hi + offset])
    j_lo2 = max(0, offset)
    j_hi2 = min(n, n + offset)
    if j_hi2 > j_lo2:
        g[j_lo2:j_hi2] -= h[j_lo2 - offset : j_hi2 - offset]
    return g

def subgrad_max(h, n, beta=8.0):
    corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    shift = beta * corr - np.max(beta * corr)
    weights = np.exp(shift)
    weights /= weights.sum()
    g = np.zeros(n)
    for m in range(2 * n - 1):
        if weights[m] < 1e-8:
            continue
        offset = m - (n - 1)
        g += weights[m] * subgrad_for_corr(h, offset, n)
    return g, np.max(corr), weights
