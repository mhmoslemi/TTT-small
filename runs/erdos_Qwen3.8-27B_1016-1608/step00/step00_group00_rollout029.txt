def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c5_grad(h):
    """Return (c5, subgradient)"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(corr))
    lag = k_star - (n - 1)
    c5 = float(corr[k_star] * dx)
    
    # Subgradient
    grad = np.zeros(n)
    # For each j: grad[j] = (1-h[j-lag]) if valid - h[j+lag] if valid
    # Vectorized:
    if lag >= 0:
        # j-lag valid: j >= lag, j-lag < n => j in [lag, n-1]
        grad[lag:] += (1.0 - h[:n-lag])
        # j+lag valid: j+lag < n => j in [0, n-1-lag]
        grad[:n-lag] -= h[lag:]
    else:
        # lag < 0, let l = -lag
        l = -lag
        # j-lag = j+l valid: j+l < n => j in [0, n-1-l]
        grad[:n-l] += (1.0 - h[l:])

        grad[l:] -= h[:n-l]

import numpy as np

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))


def _project(h, n):
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        f = (h > 1e-14) & (h < 1.0 - 1e-14)
        if not f.any():
            h += d / n
            h = np.clip(h, 0, 1)
            break
        h[f] += d / f.sum()
    return np.clip(h, 0.0, 1.0)


def _soft_grad(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    shifted = corr - corr.max()
    w = np.exp(beta * shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.full(2 * n - 1, 1.0 / (2 * n - 1))
    else:
        w /= ws

            grad[l:] -= h[:n - l] * wk

    h = h.copy()
    best_h = h.copy()
    best_c5 = _c5(h)
    m = 2 * n - 1
    dx = 2.0 / n
    for _ in range(n_passes):
        for i in range(n):
            old = h[i]
            h[i] = 0.0
            corr = np.correlate(h, 1.0 - h, mode='full')
            a = np.zeros(m)
            for k in range(m):
                lag = k - (n - 1)
                ak = 0.0
                if 0 <= i + lag < n:
                    ak += (1.0 - h[i + lag])
                if 0 <= i - lag < n:
                    ak -= h[i - lag]
                a[k] = ak

            # Candidates: 0, 1, and breakpoints in (0,1)
