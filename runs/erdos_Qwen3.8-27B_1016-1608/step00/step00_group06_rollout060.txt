```python
    # lags: d_k = k - (n-1) for k = 0..2n-2
    # weights[k] corresponds to lag d_k
    
    grad = np.zeros(n)
    
    # For each j, grad[j] = dx * sum_k weights[k] * [(1-h[j+d_k])*I(0<=j+d_k<n) - h[j-d_k]*I(0<=j-d_k<n)]
    
    # Vectorize: create a 2D array of lags
    # For position j and lag d_k, the first term is valid if 0 <= j+d_k < n
    # The second term is valid if 0 <= j-d_k < n
    
    # Approach: for each valid (j, k) pair, add the contribution
    # This is a convolution-like operation
    
    # Term 1: sum_k weights[k] * (1-h[j+d_k]) for valid pairs
    # Let's think of it as: for each lag d (from -(n-1) to n-1), 
    #   contribute weights[d+n-1] * (1-h) shifted by d
    
    lags = np.arange(-(n-1), n)  # length 2n-1
    # For lag d, the contribution to grad[j] from term 1 is:
    #   weights[d+(n-1)] * (1-h[j+d]) if 0<=j+d<n
    #   This is a shifted version of (1-h)
    
    for d in lags:
        w = weights[d + (n-1)]
        if w < 1e-15:
            continue
        if d >= 0:
            # j+d < n means j < n-d
            # term 1: (1-h[j+d]) for j in [0, n-d)
            grad[:n-d] += w * (1.0 - h[d:])
            # term 2: -h[j-d] for j in [d, n)  (j-d >= 0 means j >= d)
            grad[d:] -= w * h[:n-d]
        else:
            dd = -d
            # j+d < n: always (since j<n and d<0)
            # j+d >= 0: j >= -d = dd
            # term 1: (1-h[j+d]) for j in [dd, n)
            grad[dd:] += w * (1.0 - h[:n-dd])
            # term 2: -h[j-d] = -h[j+dd] for j in [0, n) where j+dd < n, i.e., j < n-dd
            grad[:n-dd] -= w * h[dd:]
    
    return grad * dx
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx
    return np.max(vals), vals

def project_feasible(y, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def grad_smooth(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx
    m = np.max(vals)
    ev = np.exp((vals - m) * beta)
    w = ev / (np.sum(ev) + 1e-300)
    grad = np.zeros(n)
    for ki in range(2 * n - 1):
        k = ki - (n - 1)
        wi = w[ki]
        if wi < 1e-15:
            continue
        if k >= 0:
            grad[k:] += wi * (1.0 - h[:n - k])
            grad[:n - k] -= wi * h[k:]
        else:
            kk = -k
            grad[:n - kk] += wi * (1.0 - h[kk:])
            grad[kk:] -= wi * h[:n - kk]
    return grad * dx

def smooth_max_val(vals, beta):
    m = np.max(vals)
    if beta < 1:
        return m
    return m + np.log(np.mean(np.exp((vals - m) * beta))) / beta

    bounds = [(0.0, 1.0)] * n
```
