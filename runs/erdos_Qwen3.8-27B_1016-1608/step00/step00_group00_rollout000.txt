def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt > 0:
                h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt > 0:
                h[mask] += diff / cnt
    h = np.clip(h, 0.0, 1.0)

    if abs(resid) > 1e-10:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] += resid
        h = np.clip(h, 0.0, 1.0)

    h = np.clip(x, 0.0, 1.0)
    pen = 200.0 * (np.sum(h) - target_sum) ** 2
