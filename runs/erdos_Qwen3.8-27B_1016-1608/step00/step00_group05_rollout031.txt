    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # Adjust: add diff/n to all, then clip
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        # If still off, adjust individual elements
        diff2 = target - h.sum()
        if abs(diff2) > 1e-12:
            step = 1.0 / n
            idx = 0
            if diff2 > 0:
                while h.sum() < target - 1e-12 and idx < n * n:
                    if h[idx % n] < 1.0:
                        h[idx % n] = min(1.0, h[idx % n] + step)
                    idx += 1
            else:
                while h.sum() > target + 1e-12 and idx < n * n:
                    if h[idx % n] > 0.0:
                        h[idx % n] = max(0.0, h[idx % n] - step)
                    idx += 1
    return h

def thue_morse(n):
    """Generate Thue-Morse sequence of length n (0-indexed)"""
    tm = np.zeros(n)
    for i in range(1, n):
        tm[i] = tm[i >> 1] ^ (i & 1)
    return tm

def local_search_binary(h, n, max_iters, rng):
    """Local search on binary sequence by swapping 1s with 0s"""
    h = (h >= 0.5).astype(float)
    target = n // 2
    # Fix count
    while h.sum() != target:
        if h.sum() > target:
            idx = np.where(h == 1)[0]
            h[idx[rng.integers(len(idx))]] = 0
        else:
            idx = np.where(h == 0)[0]
            h[idx[rng.integers(len(idx))]] = 1
    
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        c5_new = compute_c5(h)
        if c5_new < best_c5:
            best_c5 = c5_new
        else:
            h[i], h[j] = 1.0, 0.0
    return h, best_c5

def gradient_step(h, n, lr):
    """One gradient descent step using finite differences"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    corr_dx = corr * dx
    c5 = np.max(corr_dx)
    
    # Subgradient approach: focus on the maximizing lag
    k_star = np.argmax(corr_dx)
    
    # Compute gradient at k_star
    # corr[k_star] = sum_i h[i] * (1-h[i + k_star - (n-1)]) ... 
    # Let's just do finite differences
    grad = np.zeros(n)
    eps = 0.01
    for j in range(n):
        h_trial = h.copy()
        h_trial[j] = min(1.0, h[j] + eps)
        c = compute_c5(h_trial)
        h_trial2 = h.copy()
        h_trial2[j] = max(0.0, h[j] - eps)
        c2 = compute_c5(h_trial2)
        grad[j] = (c - c2) / (2 * eps)
    
    h_new = h - lr * grad
    h_new = project_feasible(h_new, n)
    return h_new

Let me think about a more efficient approach. Actually, I can compute the gradient analytically.

For the full correlation:
Let's denote a = h (length n), b = 1-h (length n).
corr = np.correlate(a, b, 'full') has length 2n-1.

From numpy docs: for correlate(a, v, 'full'):

Wait, I think the convention is:
c[k] = sum_m a[m+k-(N-1)] * v[m] for k = 0, ..., M+N-2

where M = len(a), N = len(v).

So corr[k] = sum_m h[m + k - (n-1)] * (1-h[m])

where h is extended by 0 outside [0, n-1].

d(corr[k])/d(h[j]) = sum_m [d(h[m+k-(n-1)])/d(h[j]) * (1-h[m]) + h[m+k-(n-1)] * d(1-h[m])/d(h[j])]
= (1-h[j+k-(n-1)]) * 1_{j = m+k-(n-1) for some valid m} - h[j+(n-1)-k] * 1_{j = m for some valid m}

Let me substitute: for the first term, j = m + k - (n-1), so m = j - k + (n-1). This is valid when 0 ≤ m ≤ n-1, i.e., 0 ≤ j - k + n - 1 ≤ n-1, i.e., k - n + 1 ≤ j ≤ k.

For the second term, j = m, valid when 0 ≤ j ≤ n-1 (always true), and we need m + k - (n-1) valid, i.e., 0 ≤ j + k - n + 1 ≤ n-1, i.e., n-1-k ≤ j ≤ n-1-k+n-1 = 2n-2-k.

So:
d(corr[k])/d(h[j]) = (1-h[j-k+n-1]) * [0 ≤ j-k+n-1 ≤ n-1] - h[j+k-n+1] * [0 ≤ j+k-n+1 ≤ n-1]

Let me verify: let i = j - k + n - 1. Then the first term says: at position j in h, if we increase h[j] by eps, it increases corr[k] by (1-h[i]) where i = j - k + n - 1, provided i is in [0, n-1].

And the second term: let j' = j + k - n + 1. If we increase h[j] by eps, it decreases corr[k] by h[j'] where j' = j + k - n + 1, provided j' is in [0, n-1].

This makes sense! So the gradient of the max is the gradient at the maximizing k*.

Let me code this up properly:

```python
def compute_grad_at_max(h, n):
    """Compute subgradient of max correlation"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = np.argmax(corr)
    # k_star ranges from 0 to 2n-2
    grad = np.zeros(n)
    for j in range(n):
        # First term: i = j - k_star + n - 1
        i = j - k_star + n - 1
        if 0 <= i < n:
            grad[j] += (1.0 - h[i])
        # Second term: j2 = j + k_star - n + 1
        j2 = j + k_star - n + 1
        if 0 <= j2 < n:
            grad[j] -= h[j2]
    return grad, k_star
```

This is O(n) per evaluation (for the max) + O(n) for the gradient. Much better than O(n²) for finite differences.

But wait, this only gives the subgradient at the single maximizing lag. In practice, the max is achieved at multiple lags, and we might want to weight them. Let me use a soft-max approach:

For a temperature parameter T:
f_T(h) = (1/T) * log(sum_k exp(T * corr[k] * dx))

∂f_T/∂h[j] = sum_k exp(T*corr[k]*dx) / sum_k' exp(T*corr[k']*dx) * (∂corr[k]/∂h[j] * dx)

This gives a weighted combination of gradients. As T → ∞, it approaches the subgradient at the max.

For practical purposes, I'll use a moderate T (like 10-50) to get a smooth approximation.

        step = 1.0 / n
        order = np.argsort(h) if diff2 > 0 else np.argsort(-h)
        for idx in order:
            if abs(diff2) < step:
                h[idx] += diff2
                break
            if diff2 > 0 and h[idx] < 1.0:

                diff2 -= step
            elif diff2 < 0 and h[idx] > 0.0:
                h[idx] -= step
                diff2 += step

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
    # Iterative fix
    for _ in range(10):
        diff = target - h.sum()
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            mask = h < 1.0
            if mask.any():
                h[mask] = np.minimum(1.0, h[mask] + diff / mask.sum())
        else:
            mask = h > 0.0
            if mask.any():
                h[mask] = np.maximum(0.0, h[mask] + diff / mask.sum())
    return h

def soft_max_grad(h, n, T=30.0):
    """Gradient of soft-max approximation to max correlation"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_shift = corr - corr.max()
    w = np.exp(T * c_shift)
    w_sum = w.sum()
    w /= w_sum
    
    grad = np.zeros(n)
    # Vectorized: for each lag m, compute contribution
    # corr[m] = sum_i h[i + m - (n-1)] * (1-h[i]) where indices must be in [0,n-1]
    # d(corr[m])/d(h[j]) = (1-h[j-m+n-1])*[valid] - h[j+m-n+1]*[valid]

        wm = w[m]
        if wm < 1e-8:

        # First term: valid j where 0 <= j-m+n-1 <= n-1
        j_lo1 = max(0, m - n + 1)
        j_hi1 = min(n - 1, m)

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            if mask.sum() == 0:
                break
            h[mask] = np.minimum(1.0, h[mask] + diff / mask.sum())
        else:
            mask = h > 0.0
            if mask.sum() == 0:
                break
            h[mask] = np.maximum(0.0, h[mask] + diff / mask.sum())
    return h

def soft_max_grad(h, n, T=30.0):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_shift = corr - corr.max()
    w = np.exp(T * c_shift)
    w /= w.sum()
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-10:
            continue
        j_lo = max(0, n - 1 - m)
        j_hi = min(n - 1, 2 * n - 2 - m)
        if j_lo <= j_hi:
            i_lo = j_lo - m + n - 1
            grad[j_lo:j_hi + 1] += wm * (1.0 - h[i_lo:i_lo + (j_hi - j_lo + 1)])
        j_lo2 = max(0, m - n + 1)
        j_hi2 = min(n - 1, m - 1)
        if j_lo2 <= j_hi2:
            i_lo2 = j_lo2 - m + n - 1
            grad[j_lo2:j_hi2 + 1] -= wm * h[i_lo2:i_lo2 + (j_hi2 - j_lo2 + 1)]
    return grad

def gradient_optimize(h, n, max_iters=200, T_init=20.0, lr_init=0.04):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    T = T_init
    for it in range(max_iters):
        grad = soft_max_grad(h, n, T)
        h = h - lr * grad
        h = project_feasible(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        lr *= 0.992
        T = T_init * (1 + 0.3 * it / max_iters)
    return best_h, best_c5

def binary_local_search(h, n, max_iters=20000, rng=None):
    if rng is None:
        rng = np.random.default_rng(0)
    h_bin = (h >= 0.5).astype(float)
    target = n // 2
    while h_bin.sum() != target:
        if h_bin.sum() > target:
            idx = np.where(h_bin == 1)[0]
            h_bin[idx[rng.integers(len(idx))]] = 0
        else:
            idx = np.where(h_bin == 0)[0]
            h_bin[idx[rng.integers(len(idx))]] = 1
    best_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    for it in range(max_iters):
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h_bin[i], h_bin[j] = 0.0, 1.0
        c5_new = compute_c5(h_bin)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_bin.copy()
        else:
            h_bin[i], h_bin[j] = 1.0, 0.0
    return best_h, best_c5

def two_swap_search(h, n, max_iters=5000, rng=None):
    if rng is None:
        rng = np.random.default_rng(99)
    h_bin = (h >= 0.5).astype(float)
    target = n // 2
    while h_bin.sum() != target:
        if h_bin.sum() > target:
            idx = np.where(h_bin == 1)[0]
            h_bin[idx[rng.integers(len(idx))]] = 0
        else:
            idx = np.where(h_bin == 0)[0]
            h_bin[idx[rng.integers(len(idx))]] = 1
    best_c5 = compute_c5(h_bin)
    for _ in range(max_iters):
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        i1, i2 = ones[rng.integers(len(ones), 2)]
        j1, j2 = zeros[rng.integers(len(zeros), 2)]
        h_bin[i1] = h_bin[i2] = 0.0
        h_bin[j1] = h_bin[j2] = 1.0
        c5_new = compute_c5(h_bin)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
        else:
            h_bin[i1] = h_bin[i2] = 1.0
            h_bin[j1] = h_bin[j2] = 0.0
    return h_bin, best_c5

def thue_morse_init(n):
    tm = np.zeros(n)
    for i in range(1, n):
        tm[i] = tm[i >> 1] ^ (i & 1)
    return tm

def golden_init(n):
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    xs = np.arange(n)
    h = ((xs * phi) % 1.0) < 0.5
    return h.astype(float)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start = time.time()

    try:
        h_init = np.array(initial_h_values, dtype=float)
        n_init = len(h_init)
    except (NameError, UnboundLocalError):
        n_init = 70
        h_init = thue_morse_init(n_init)

    h_init = project_feasible(h_init.copy(), n_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    best_n = n_init

    # Phase 1: Gradient on initial
    if budget_s - (time.time() - start) > 40:
        h_g, c5_g = gradient_optimize(h_init, n_init, max_iters=150, T_init=20.0, lr_init=0.03)
        if c5_g < best_c5:
            best_c5, best_h, best_n = c5_g, h_g, n_init

    # Phase 2: Binary local search on best
    elapsed = time.time() - start
    if budget_s - elapsed > 20:
        n_iter = min(25000, int((budget_s - elapsed - 15) * 100))
        h_b, c5_b = binary_local_search(best_h, best_n, n_iter, rng)
        if c5_b < best_c5:
            best_c5, best_h = c5_b, h_b

    # Phase 3: Two-swap search
    elapsed = time.time() - start
    if budget_s - elapsed > 15:
        n_iter = min(8000, int((budget_s - elapsed - 10) * 60))
        h_2s, c5_2s = two_swap_search(best_h, best_n, n_iter, rng)
        if c5_2s < best_c5:
            best_c5, best_h = c5_2s, h_2s

    # Phase 4: Try different starting patterns and resolutions
    test_ns = [60, 70, 80, 100, 120]
    for tn in test_ns:
        elapsed = time.time() - start
        if budget_s - elapsed < 25:
            break
        # Interpolate best to new resolution
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, tn, endpoint=False)
        h_t = np.interp(x_new, x_old, best_h)
        h_t = project_feasible(h_t, tn)
        # Quick gradient
        h_g2, c5_g2 = gradient_optimize(h_t, tn, max_iters=80, T_init=25.0, lr_init=0.03)
        # Binary search
        elapsed2 = time.time() - start
        if budget_s - elapsed2 > 15:
            n_iter2 = min(12000, int((budget_s - elapsed2 - 10) * 70))
            h_b2, c5_b2 = binary_local_search(h_g2, tn, n_iter2, rng)
            if c5_b2 < c5_g2:
                h_g2, c5_g2 = h_b2, c5_b2
        if c5_g2 < best_c5:
            best_c5, best_h, best_n = c5_g2, h_g2, tn

    # Phase 5: Also try Thue-Morse and golden at a few sizes
    for tn in [70, 84, 100]:
        elapsed = time.time() - start
        if budget_s - elapsed < 20:
            break
        for init_fn in [thue_morse_init, golden_init]:
            h_try = init_fn(tn)
            h_try = project_feasible(h_try, tn)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 + 0.05:
                h_g3, c5_g3 = gradient_optimize(h_try, tn, max_iters=60, T_init=25.0, lr_init=0.02)
                c5_g4 = c5_g3
                h_g4 = h_g3
                elapsed3 = time.time() - start
                if budget_s - elapsed3 > 10:
                    h_b3, c5_b3 = binary_local_search(h_g3, tn, min(8000, int((budget_s - elapsed3 - 5) * 60)), rng)
                    if c5_b3 < c5_g4:
                        c5_g4, h_g4 = c5_b3, h_b3
                if c5_g4 < best_c5:
                    best_c5, best_h, best_n = c5_g4, h_g4, tn

    # Phase 6: Final refinement
    elapsed = time.time() - start
    if budget_s - elapsed > 15:
        n_final = min(20000, int((budget_s - elapsed - 5) * 80))
        h_f, c5_f = binary_local_search(best_h, best_n, n_final, rng)
        if c5_f < best_c5:
            best_c5, best_h = c5_f, h_f

    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
