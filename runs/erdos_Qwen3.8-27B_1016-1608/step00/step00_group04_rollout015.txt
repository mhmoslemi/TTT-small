def continuous_refine(h01, rng, duration):
    h = h01.copy().astype(float)
    n = len(h)
    target = int(h01.sum())
    dx = 2.0 / n
    t_start = time.time()
    deadline = t_start + duration
    step = 0.05
    while time.time() < deadline:
        # Compute all correlations
        corr = np.correlate(h, 1.0 - h, mode='full')
        worst_k = np.argmax(corr)
        k = worst_k - (n - 1)
        # Gradient of g_k wrt h
        grad = np.zeros(n)
        if k >= 0:
            # g_k = sum_{i=0}^{n-k-1} h[i]*(1-h[i+k])
            # d/dh[j]: (1-h[j+k]) for j in 0..n-k-1, plus -h[j-k] for j in k..n-1
            grad[:n-k] += (1.0 - h[k:])
            if k > 0:
                grad[k:] += -h[:n-k]
        else:
            kk = -k
            # g_k = sum_{i=kk}^{n-1} h[i]*(1-h[i-kk])
            grad[kk:] += (1.0 - h[:n-kk])
            if kk > 0:
                grad[:n-kk] += -h[kk:]
        grad *= dx
        # Subgradient step (minimize)
        # Project: keep sum(h) = target, 0 <= h <= 1
        h_new = h - step * grad
        # Project back
        h_new = np.clip(h_new, 0, 1)
        # Adjust sum
        diff = target - h_new.sum()
        # Distribute adjustment
        if abs(diff) < 1e-6:
            h = h_new
        else:
            # Adjust the least "committed" values
            # Simple: add diff/n to all, then clip
            h_new += diff / n
            h_new = np.clip(h_new, 0, 1)
            # Iterate a few times
            for _ in range(10):
                h_new = np.clip(h_new, 0, 1)
                diff = target - h_new.sum()
                if abs(diff) < 1e-8:
                    break
                # Adjust non-boundary values
                mask = (h_new > 0.01) & (h_new < 0.99)
                if mask.any():
                    h_new[mask] += diff / mask.sum()
                    h_new = np.clip(h_new, 0, 1)
            h = h_new
        step *= 0.999
    return h, eval_overlap(h)
