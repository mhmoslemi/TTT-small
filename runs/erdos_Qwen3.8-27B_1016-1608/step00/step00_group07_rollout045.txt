import numpy as np
from scipy.optimize import minimize
import time


def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    diff = target - h.sum()
    if diff > 1e-12:
        order = np.argsort(1.0 - h)
        for idx in order:
            if diff <= 0:
                break
            add = min(1.0 - h[idx], diff)
            h[idx] += add
            diff -= add
    elif diff < -1e-12:
        diff = -diff
        order = np.argsort(h)
        for idx in order:
            if diff <= 0:
                break
            sub = min(h[idx], diff)
            h[idx] -= sub
            diff -= sub
    return h


def lse_obj(h, eps):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = corr.max()
    return m + eps * np.log(np.exp((corr - m) / eps).sum())
