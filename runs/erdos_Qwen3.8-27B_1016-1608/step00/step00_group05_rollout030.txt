import numpy as np
from scipy import optimize
import time

def project_onto_feasible(h, total):
    """Project h onto {x: 0 <= x <= 1, sum(x) = total}"""
    # Bisection on threshold
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > total:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2, 0, 1)
    # Fix small numerical issues
    diff = total - np.sum(proj)
    if abs(diff) < 1e-10:
        return proj
    # Distribute the difference
    mask = (proj > 1e-10) & (proj < 1 - 1e-10)
    if np.sum(mask) > 0:
        proj[mask] += diff / np.sum(mask)
        proj = np.clip(proj, 0, 1)
    return proj

def compute_c5(h, dx):

    corr = np.correlate(h, one_minus_h, mode='full')
    return np.max(corr) * dx

def compute_subgradient(h, dx):

    n = len(h)
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    max_val = np.max(corr)
    
    # Find all lags achieving the max (within tolerance)
    tol = 1e-10
    active_mask = corr >= max_val - tol
    n_active = np.sum(active_mask)
    
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        if not active_mask[k_idx]:
            continue
        k = k_idx - (n - 1)  # actual lag
        # c_k = sum_n h[n+k]*(1-h[n]) for valid n
        # d(c_k)/d(h[i]) = (1-h[i-k]) - h[i+k] (with boundary checks)
        # where the terms are included only if the corresponding index is valid
        for i in range(n):
            # h[i] appears as h[n+k] when n = i-k
            # Valid if 0 <= i-k < n and 0 <= i < n and 0 <= i < n (n=i-k)
            # Contribution to c_k from term h[n+k]*(1-h[n]) where i = n+k:
            #   (1-h[n]) = (1-h[i-k])
            # Valid if 0 <= i-k < n
            # h[i] appears as h[n] when n = i
            # Valid if 0 <= i < n and 0 <= i+k < n
            # Contribution: -h[i+k]
            term = 0.0
            if 0 <= i - k < n:
                term += (1.0 - h[i - k])
            if 0 <= i + k < n:
                term -= h[i + k]
            grad[i] += term / n_active
    
    return grad * dx, max_val * dx

def project_onto_feasible(h, total):
    """Project h onto {x: 0<=x<=1, sum(x)=total} via bisection."""
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > total:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    proj = np.clip(h - t, 0.0, 1.0)
    # Correct for small numerical error
    diff = total - np.sum(proj)
    if abs(diff) > 1e-12:
        mask = (proj > 1e-12) & (proj < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            proj[mask] += diff / cnt
            proj = np.clip(proj, 0.0, 1.0)

    max_val = np.max(corr)
    tol = 1e-8
    active = corr >= max_val - tol
    n_act = np.sum(active)
    if n_act == 0:
        n_act = 1
        active = np.zeros(len(corr), dtype=bool)
        active[np.argmax(corr)] = True

        k = k_idx - (n - 1)

        # First term: i from max(0,k) to min(n, n+k)-1, add (1-h[i-k])
        s1 = max(0, k)
        e1 = min(n, n + k)
        if s1 < e1:

            grad[idx] += (1.0 - h[idx - k]) / n_act
        # Second term: i from max(0,-k) to min(n, n-k)-1, subtract h[i+k]

        e2 = min(n, n - k)
        if s2 < e2:
            idx = np.arange(s2, e2)
            grad[idx] -= h[idx + k] / n_act
    return max_val * dx, grad * dx

import time

def project_feasible(h, total):
    """Project onto {x: 0<=x<=1, sum(x)=total}."""
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > total:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    proj = np.clip(h - t, 0.0, 1.0)
    diff = total - np.sum(proj)
    if abs(diff) > 1e-12:
        mask = (proj > 1e-8) & (proj < 1.0 - 1e-8)
        cnt = int(np.sum(mask))
        if cnt > 0:
            proj[mask] += diff / cnt
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def corr_and_max(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr, float(np.max(corr))

def subgradient_step(h, dx, step_size, active_tol=1e-6):
    """One subgradient step for min C5."""
    n = len(h)
    corr, max_c = corr_and_max(h)
    active = corr >= max_c - active_tol
    n_act = int(np.sum(active))
    if n_act == 0:
        n_act = 1
        active[np.argmax(corr)] = True
    
    grad = np.zeros(n)
    for k_idx in np.where(active)[0]:
        k = k_idx - (n - 1)
        s1 = max(0, k)
        e1 = min(n, n + k)
        if s1 < e1:
            idx = np.arange(s1, e1)
            grad[idx] += (1.0 - h[idx - k])
        s2 = max(0, -k)
        e2 = min(n, n - k)
        if s2 < e2:
            idx = np.arange(s2, e2)
            grad[idx] -= h[idx + k]
    grad /= n_act
    return grad * dx

    """Kronecker sequence with golden ratio."""
    alpha = (1.0 + np.sqrt(5.0)) / 2.0

def smooth_obj(x, dx, t, penalty):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    # log-sum-exp for numerical stability
    shifted = corr / t
    m = np.max(shifted)
    smax = t * (m + np.log(np.sum(np.exp(shifted - m))))
    obj = smax * dx
    obj += penalty * (np.sum(x) - n / 2.0) ** 2
    return obj

def smooth_obj_grad(x, dx, t, penalty):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    shifted = corr / t
    m = np.max(shifted)
    exp_shifted = np.exp(shifted - m)
    weights = exp_shifted / np.sum(exp_shifted)
    
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        if weights[k_idx] < 1e-15:
            continue
        k = k_idx - (n - 1)
        s1 = max(0, k)
        e1 = min(n, n + k)
        if s1 < e1:
            idx = np.arange(s1, e1)
            grad[idx] += weights[k_idx] * (1.0 - x[idx - k])
        s2 = max(0, -k)
        e2 = min(n, n - k)
        if s2 < e2:
            idx = np.arange(s2, e2)
            grad[idx] -= weights[k_idx] * x[idx + k]
    grad *= dx
    
    # Gradient of penalty term
    grad += 2 * penalty * (np.sum(x) - n / 2.0) * np.ones(n)
    
    return grad
