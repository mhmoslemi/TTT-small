            c5_r = compute_c5(h_r)
            best_h, best_c5 = sa_run(h_r, c5_r, rng, n, 0.008, 150000, d5, best_h, best_c5)
            if time.time() > d5 - 5:
                break
            best_h, best_c5 = fine_tune(best_h, compute_c5(best_h), rng, n, d5, best_h, best_c5)

    # Final validation
    best_h = make_feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n)
