import numpy as np

def compute_c5_full(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            break
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_c5_pen(x, dx, n, target):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    tm = 0.01
    mx = vals.max()
    c5 = mx + tm * np.log(np.exp((vals - mx) / tm).sum())
    pen = 1000.0 * (h.sum() - target) * (h.sum() - target)
    return c5 + pen
