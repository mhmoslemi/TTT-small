import numpy as np
import time
from scipy.optimize import minimize

def corr_max(h):
    n = len(h)
    dx = 2.0/n
    return (np.correlate(h, 1.0-h, mode="full") * dx).max()

def project_feasible(h, target_sum):
    # Project onto {x: 0<=x<=1, sum(x)=target_sum}
    # Water-filling: find lambda s.t. sum(clip(h-lambda, 0, 1)) = target_sum
    lo, hi = h.max() - 1, h.min()
    for _ in range(200):
        mid = (lo+hi)/2
        s = np.clip(h - mid, 0, 1).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2, 0, 1)

def smooth_obj_grad(h, dx, alpha):
    n = len(h)
    c = np.correlate(h, 1.0-h, mode="full") * dx
    c_max = c.max()
    e = np.exp(alpha*(c - c_max))
    obj = c_max + np.log(e.sum())/alpha
    p = e / e.sum()
    
    grad = np.zeros(n)
    for k in range(2*n-1):
        if p[k] < 1e-14:
            continue
        lo = max(0, k-n+1)
        hi = min(n-1, k)
        pk = p[k] * dx
        # First term
        for j in range(lo, hi+1):
            grad[j] += pk * (1.0 - h[j+n-1-k])
        # Second term
        jl = max(0, lo-k+n-1)
        jh = min(n-1, hi-k+n-1)
        for j in range(jl, jh+1):
            grad[j] -= pk * h[j+k-(n-1)]
    
    return obj, grad

        lr = 1.0

            obj, grad = smooth_obj_grad(h, dx, alpha)
            # gradient step
            h_new = h - lr * grad
            h_new = project_feasible(h_new, target_sum)
            # check improvement
            obj_new = (np.correlate(h_new, 1.0-h_new, mode="full")*dx).max()

                h_new = project_feasible(h_new, target_sum)
