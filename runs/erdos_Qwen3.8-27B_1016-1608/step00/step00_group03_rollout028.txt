from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def smooth_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    return float(m + alpha * np.log(np.sum(np.exp((corr - m) / alpha))))

def smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    weights = np.exp((corr - m) / alpha)
    wsum = np.sum(weights)
    p = weights / wsum
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-8:
            continue
        j = k - (n - 1)
        c = p[k] * dx
        if j >= 0:
            if j == 0:
                grad += c * (1.0 - 2.0 * h)
            else:
                grad[j:] += c * (1.0 - h[:n-j])
                grad[:n-j] -= c * h[:n-j]
        else:
            jj = -j
            grad[jj:] += c * (1.0 - h[:n-jj])
            grad[:n-jj] -= c * h[jj:]
    return grad

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    # Fine-tune
    diff = target - h.sum()

        n_fix = min(n, max(1, int(np.ceil(abs(diff) / 0.5))))
        if diff > 0:
            idx = np.argsort(h)[:n_fix]
            h[idx] += diff / n_fix
        else:
            idx = np.argsort(h)[-n_fix:]
            h[idx] -= (-diff) / n_fix

    alphas = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.0008, 0.0004]
    for alpha in alphas:
        if time.time() - t0 > budget_s - 80:
            break
        _ALPHA = alpha
        try:
            res = minimize(obj_func, best_h, jac=grad_func, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 200, 'ftol': 1e-12})
            h_test = fix_sum(res.x, n)
            c5_test = compute_c5(h_test)
            if c5_test < best_c5:
                best_h = h_test.copy()
                best_c5 = c5_test
        except Exception:
            pass

    # Stage 2: Direct SLSQP on actual max (multiple attempts)
    for it in range(4):
        if time.time() - t0 > budget_s - 60:
            break
        try:
            res = minimize(compute_c5, best_h, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 400, 'ftol': 1e-14})
            h_test = fix_sum(res.x, n)
            c5_test = compute_c5(h_test)
            if c5_test < best_c5:
                best_h = h_test.copy()
                best_c5 = c5_test
        except Exception:
            pass

    # Stage 3: Multi-start with perturbations and local refine
    n_starts = 150
    for i in range(n_starts):
        if time.time() - t0 > budget_s - 10:
            break
        # Varying scale perturbations
        scale = 0.03 + 0.12 * np.random.rand()
        h_pert = best_h + scale * np.random.randn(n)
        h_pert = fix_sum(h_pert, n)
        c5_pert = compute_c5(h_pert)
        # Also try structured starts every 10th
        if i % 10 == 5:
            h_pert = np.sin(np.linspace(0, 4 * np.pi, n)) * 0.3 + 0.5
            h_pert = fix_sum(h_pert, n)
            c5_pert = compute_c5(h_pert)
        if c5_pert < best_c5:
            best_h = h_pert.copy()
            best_c5 = c5_pert
            if time.time() - t0 < budget_s - 20:
                _ALPHA = 0.001
                try:
                    r1 = minimize(obj_func, best_h, jac=grad_func, method='SLSQP',
                                  bounds=bounds, constraints=cons,
                                  options={'maxiter': 150, 'ftol': 1e-12})
                    h_test = fix_sum(r1.x, n)
                    c5_test = compute_c5(h_test)
                    if c5_test < best_c5:
                        best_h = h_test.copy()
                        best_c5 = c5_test
                except Exception:
                    pass
                if time.time() - t0 < budget_s - 15:
                    try:
                        r2 = minimize(compute_c5, best_h, method='SLSQP',
                                      bounds=bounds, constraints=cons,
                                      options={'maxiter': 200, 'ftol': 1e-13})
                        h_test = fix_sum(r2.x, n)
                        c5_test = compute_c5(h_test)
                        if c5_test < best_c5:
                            best_h = h_test.copy()
                            best_c5 = c5_test
                    except Exception:
                        pass

    # Stage 4: Fine-tune with Nelder-Mead on a smoothed version
    if time.time() - t0 < budget_s - 10:
        _ALPHA = 0.0001
        try:
            res = minimize(obj_func, best_h, method='Nelder-Mead',
                           options={'maxiter': 500, 'xatol': 1e-9, 'fatol': 1e-12,
                                    'adaptive': True})
            h_test = fix_sum(res.x, n)
            c5_test = compute_c5(h_test)
            if c5_test < best_c5:
                best_h = h_test.copy()
                best_c5 = c5_test
        except Exception:
            pass

    # Final SLSQP polish
    if time.time() - t0 < budget_s - 3:
        try:
            res = minimize(compute_c5, best_h, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 300, 'ftol': 1e-14})
            h_test = fix_sum(res.x, n)
            c5_test = compute_c5(h_test)
            if c5_test < best_c5:
                best_h = h_test.copy()
                best_c5 = c5_test
        except Exception:
            pass

    best_h = fix_sum(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
