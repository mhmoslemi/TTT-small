from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def repair_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    n = len(initial_h_values)
    h0 = np.array(initial_h_values, dtype=float).copy()
    h0 = repair_sum(h0, n)
    
    best_h = h0.copy()
    best_c5 = evaluate_c5(best_h)
    
    def elapsed():
        return time.time() - t0
    
    # Phase 1: SLSQP optimization
    def obj_and_grad(x):
        h = x[:n]
        dx = 2.0 / n
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full') * dx
        c_val = np.max(corr)
        return c_val
    
    # SLSQP with equality constraint
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(obj_and_grad, h0, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': 500, 'ftol': 1e-10})

            h_new = repair_sum(res.x, n)
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

        pass
    
    if elapsed() > budget_s - 100:
        return best_h, best_c5, n

                z = int(np.random.randint(n))
                ht[z] = 0.5
                ht[ones[0]] = 0.5
                ht[ones[0]] = 1.0
        ht = repair_sum(ht, n)
        try:
            r = minimize(objf, ht, jac=objg, method='SLSQP', bounds=bounds,
                        constraints=cons, options={'maxiter': 250, 'ftol': 1e-14})
            hn = repair_sum(r.x, n)
            c5n = evaluate_c5(hn)
            if c5n < best_c5:
                best_h, best_c5 = hn, c5n
        except Exception:
            pass

    # Phase 3: Smoothing-based refinement (log-sum-exp)
    if tleft() > 100:
        for alpha in [10, 30, 80, 200, 500]:
            if tleft() < 60:
                break
            def sobj(x):
                return make_smooth_obj(x, n, alpha)
            def sgrad(x):
                corr = np.correlate(x, 1.0 - x, mode='full') * dx
                mx = np.max(corr)
                w = np.exp(alpha * (corr - mx))
                w /= np.sum(w)
                # grad of (1/alpha)log(sum exp(alpha corr_k)) = sum w_k grad corr_k
                g = np.zeros(n)
                for kk in range(2 * n - 1):
                    if w[kk] < 1e-12:
                        continue
                    m = kk - (n - 1)
                    if m == 0:
                        g[:] += w[kk] * (1.0 - 2.0 * x) * dx
                    else:
                        for i in range(n):
                            if 0 <= i - m < n:
                                g[i] += w[kk] * (1.0 - x[i - m]) * dx
                            if 0 <= i + m < n:
                                g[i] -= w[kk] * x[i + m] * dx
                return g
            try:
                r = minimize(sobj, best_h, jac=sgrad, method='SLSQP',
                            bounds=bounds, constraints=cons,
                            options={'maxiter': 150, 'ftol': 1e-14})
                hn = repair_sum(r.x, n)
                c5n = evaluate_c5(hn)
                if c5n < best_c5:
                    best_h, best_c5 = hn, c5n
            except Exception:
                pass

    # Phase 4: Fine coordinate descent
    if tleft() > 40:
        step = 0.04
        stagnant = 0
        iters = 0
        while tleft() > 15 and iters < 300000:
            iters += 1
            i = int(np.random.randint(n))
            improved = False
            for d in (step, -step, step * 0.3, -step * 0.3, step * 0.1, -step * 0.1):
                ht = best_h.copy()
                ht[i] += d
                ht = repair_sum(ht, n)
                c_val = evaluate_c5(ht)
                if c_val < best_c5 - 1e-15:
                    best_h, best_c5 = ht, c_val
                    improved = True
                    break
            if improved:
                stagnant = 0
            else:
                stagnant += 1
            if stagnant > 5000:
                step *= 0.75
                stagnant = 0
                if step < 1e-5:
                    break

    # Final verification
    best_h = repair_sum(best_h, n)
    best_c5 = evaluate_c5(best_h)

    return best_h, best_c5, n
