def compute_overlap(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def make_valid_binary(h, n):
    h = np.array(h, dtype=float)
    h = (h > 0.5).astype(float)
    target = n // 2
    s = int(np.sum(h))
    if s > target:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:s - target]] = 0.0
    elif s < target:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:target - s]] = 1.0
    return h

def make_valid_continuous(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    diff = n / 2.0 - np.sum(h)
    if abs(diff) > 1e-12:
        # Distribute correction
        for i in range(n):
            if h[i] < 1.0 and diff > 0:
                amt = min(1.0 - h[i], diff)
                h[i] += amt
                diff -= amt
            elif h[i] > 0.0 and diff < 0:
                amt = min(h[i], -diff)
                h[i] -= amt
                diff += amt
            if abs(diff) < 1e-12:
                break
    return h

                h[i] = 1.0

            current -= best_gain
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= max_no_improve:
                break

                g -= h[j + p]
            grad[j] = g * dx
        grad -= np.mean(grad)
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        diff = n / 2.0 - np.sum(h_new)
        if abs(diff) > 1e-10:
            mask = (h_new > 0.01) & (h_new < 0.99)
            cnt = np.sum(mask)
            if cnt > 0:
                h_new[mask] += diff / cnt
                h_new = np.clip(h_new, 0.0, 1.0)
        s_new = compute_overlap(h_new)
        if s_new < cur - 1e-15:
            h = h_new; cur = s_new; stall = 0
        else:
            stall += 1
            lr *= 0.9
            if lr < 0.001 or stall > 500:
                break
    return h, cur

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h0 = make_valid_bin(h0, n0)

    best_h = h0.copy()
    best_sc = compute_overlap(h0)
    best_n = n0

    # Phase 1: Intensive binary local search (12%)
    tl = t0 + 0.12 * budget_s
    h, sc = ls_binary(h0, tl)
    if sc < best_sc:
        best_sc, best_h = sc, h.copy()

    # Phase 2: Perturbation + local search (28%)
    tl = t0 + 0.40 * budget_s
    while time.time() < tl:
        h_p = best_h.copy()
        k = np.random.randint(1, 8)
        for _ in range(k):
            oi = np.where(h_p == 1)[0]
            zj = np.where(h_p == 0)[0]
            if len(oi) == 0 or len(zj) == 0:
                break
            h_p[np.random.choice(oi)] = 0
            h_p[np.random.choice(zj)] = 1
        rem = tl - time.time()
        if rem > 0.5:
            h_p, sc = ls_binary(h_p, min(time.time() + 3, tl))
        if sc < best_sc:
            best_sc, best_h = sc, h_p.copy()

    # Phase 3: Continuous refinement (18%)
    tl = t0 + 0.58 * budget_s
    if time.time() < tl:
        h_c = best_h.copy()
        h_c, sc = ls_continuous(h_c, tl, lr=0.05)
        if sc < best_sc:
            best_sc, best_h = sc, h_c.copy()
        # Re-binarize and local search
        h_b = make_valid_bin(h_c, len(h_c))
        sc = compute_overlap(h_b)
        if time.time() < tl:
            h_b, sc = ls_binary(h_b, tl)
        if sc < best_sc:
            best_sc, best_h = sc, h_b.copy()

    # Phase 4: Multi-resolution (24%)
    tl = t0 + budget_s - 15
    for d in [0, -2, 2, -14, 14, -28, 28, 42, -42]:
        n = n0 + d
        if n < 14 or n % 2 != 0 or time.time() > tl:
            continue
        if n == best_n and abs(d) <= 2:
            continue
        xo = np.linspace(0, 2, len(best_h), endpoint=False)
        xn = np.linspace(0, 2, n, endpoint=False)
        hi = np.interp(xn, xo, best_h)
        hn = make_valid_bin(hi, n)
        sc = compute_overlap(hn)
        rem = tl - time.time()
        if rem > 5:
            hn, sc = ls_binary(hn, min(time.time() + rem * 0.35, tl))
            if time.time() < tl - 2:
                hn, sc = ls_continuous(hn, tl, lr=0.03)
        if sc < best_sc:
            best_sc, best_h, best_n = sc, hn.copy(), n

    # Phase 5: Random restarts with local search (remaining)
    tl = t0 + budget_s - 5
    while time.time() < tl:
        hr = np.zeros(n0)
        pos = np.random.choice(n0, n0 // 2, replace=False)
        hr[pos] = 1
        tl_inner = min(time.time() + 2, tl)
        hr, sc = ls_binary(hr, tl_inner)
        if sc < best_sc:
            best_sc, best_h = sc, hr.copy()

    return list(best_h), best_sc, best_n
