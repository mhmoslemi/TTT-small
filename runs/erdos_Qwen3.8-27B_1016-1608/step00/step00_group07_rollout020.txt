def compute_c5(h):

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(c))

def project_feasible(h, n):

    for _ in range(2000):

        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
    np.clip(h, 0.0, 1.0, out=h)
    return h

def lag_gradient(h, n, k):
    g = np.zeros(n)
    if k >= 0:
        if n - k > 0:
            g[:n-k] += (1.0 - h[k:])
            g[k:] -= h[:n-k]
    else:
        ka = -k
        if n - ka > 0:
            g[ka:] += (1.0 - h[:n-ka])
            g[:n-ka] -= h[ka:]
    return g

def gradient_top_lags(h, n, top_k=5, tau=0.04):
    c = np.correlate(h, 1.0 - h, mode="full")
    c_max = c.max()
    exp_c = np.exp((c - c_max) / tau)
    w = exp_c / exp_c.sum()
    # Only keep top lag indices for efficiency
    top_idx = np.argsort(w)[::-1][:top_k]
    w_sum = w[top_idx].sum()
    if w_sum < 1e-15:
        w_top = np.ones(top_k) / top_k
    else:
        w_top = w[top_idx] / w_sum
    grad = np.zeros(n)
    for li in range(top_k):
        j = int(top_idx[li])
        k = j - (n - 1)
        grad += w_top[li] * lag_gradient(h, n, k)
    return grad

    t0 = time.time()

    def budget_fn():
        return time.time() - t0 > budget_s - 15

    global initial_h_values
    if initial_h_values is not None and len(initial_h_values) > 0:
        h_start = np.array(initial_h_values, dtype=float)
    else:
        h_start = np.tile(np.array([1.0, 0.5, 0.5, 0.0]), 20)
        h_start = project_feasible(h_start, 80)

    best_h = h_start.copy()
    best_c5 = compute_c5(best_h)

    # Try base size and a few others
    base_n = len(h_start)
    sizes = sorted(set([base_n, 100, 120, 160]))
    sizes = [s for s in sizes if 40 <= s <= 500]

    for n in sizes:
        if budget_fn():
            break
        x_old = np.linspace(0, 2, base_n, endpoint=False)
        x_new = np.linspace(0, 2, n, endpoint=False)
        h_interp = np.interp(x_new, x_old, h_start)
        h_interp = project_feasible(h_interp, n)

        # Multiple starting points
        starts = [h_interp.copy()]
        # Alternating pattern
        h_alt = np.zeros(n)
        h_alt[::2] = 1.0
        h_alt = project_feasible(h_alt, n)
        starts.append(h_alt)
        # Smooth sinusoidal
        h_sin = 0.5 + 0.3 * np.cos(2 * np.pi * x_new)
        h_sin = project_feasible(h_sin, n)
        starts.append(h_sin)
        # Random
        h_rand = rng.rand(n)
        h_rand = project_feasible(h_rand, n)
        starts.append(h_rand)

        for h_init in starts:
            if budget_fn():
                break
            c5_init = compute_c5(h_init)
            if c5_init < best_c5:
                best_c5 = c5_init
                best_h = h_init.copy()
            # Optimize
            h_out, c5_out = optimize_h(h_init, n, budget_fn, rng)
            if c5_out < best_c5:
                best_c5 = c5_out
                best_h = h_out.copy()

    # Final local refinement at best size
    n_final = len(best_h)
    if not budget_fn():
        best_h, best_c5 = optimize_h(best_h, n_final, budget_fn, rng, lr_init=0.01)

    best_h = project_feasible(best_h, n_final)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_final
