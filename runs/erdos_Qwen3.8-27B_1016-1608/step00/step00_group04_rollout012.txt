        npert = max(3, n // 5)
        h_start = _perturb(best_h, npert, rng)
        rem = min(100, budget_s - 250 - (time.time() - t0))
        if rem < 5:
            break
        h_new, s_new = _local_search(h_start, rem, seed + rnd * 1000, use_best=(rnd % 2 == 0))
        if s_new < best_s - 1e-12:
            best_s = s_new
            best_h = h_new
        rnd += 1

    # Phase 3: Simulated annealing on best
    while time.time() - t0 < budget_s - 100:
        rem = min(60, budget_s - 100 - (time.time() - t0))
        if rem < 5:
            break
        h_a, s_a = _anneal(best_h, rem, seed + rnd * 2000)
        if s_a < best_s - 1e-12:
            best_s = s_a
            best_h = h_a
        rnd += 1

    # Phase 4: Try other resolutions
    res_list = [30, 36, 48, 54, 60, 50, 56, 64, 40, 44, 38, 52, 66, 72, 46, 34, 78, 84]
    for n2 in res_list:
        if time.time() - t0 > budget_s - 10:
            break
        if n2 % 2 != 0:
            continue
        h2 = zoom(best_h, n2 / float(len(best_h)), order=0)
        if len(h2) != n2:
            h2 = h2[:n2] if len(h2) > n2 else np.concatenate([h2, np.zeros(n2 - len(h2))])
        h2 = _fix_binary(h2, n2 // 2)
        rem = min(20, budget_s - 10 - (time.time() - t0))
        if rem < 3:
            continue
        h2, s2 = _local_search(h2, rem, seed + 5000 + n2, use_best=True)
        if s2 < best_s - 1e-12:
            best_s = s2
            best_h = h2

    # Final check
    best_h = np.clip(best_h, 0.0, 1.0)
    final_s = _score(best_h)

    return best_h, final_s, len(best_h)
