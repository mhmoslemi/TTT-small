    """Gradient of correlate(h, 1-h, 'full')[k_idx] w.r.t. h."""

    # Term 1: (1-h[j-s]) for j where 0<=j-s<=n-1
    lo1 = max(0, s)
    hi1 = min(n-1, s + n - 1)
    if lo1 <= hi1:
        grad[lo1:hi1+1] += 1.0 - h[lo1-s:hi1-s+1]
    # Term 2: -h[j+s] for j where 0<=j+s<=n-1
    lo2 = max(0, -s)
    hi2 = min(n-1, n-1-s)
    if lo2 <= hi2:
        grad[lo2:hi2+1] -= h[lo2+s:hi2+s+1]
    return grad

def _project_feasible(h, target):
    """Project h onto {h in [0,1]^n, sum(h)=target}."""
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            break
        step = diff / free.sum()
        h[free] = np.clip(h[free] + step, 0.0, 1.0)
    return h

def _eval_c5(h, dx):
    h = np.asarray(h, dtype=float)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _grad_gk(h, n, s):
    grad = np.zeros(n, dtype=float)
    lo1 = max(0, s)
    hi1 = min(n - 1, s + n - 1)
    if lo1 <= hi1:
        grad[lo1:hi1 + 1] += 1.0 - h[lo1 - s:hi1 - s + 1]
    lo2 = max(0, -s)
    hi2 = min(n - 1, n - 1 - s)
    if lo2 <= hi2:
        grad[lo2:hi2 + 1] -= h[lo2 + s:hi2 + s + 1]
    return grad

def _project_feasible(h, target):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            break
        h[free] += diff / free.sum()
        np.clip(h, 0.0, 1.0, out=h)
