import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1 - h, mode='full')

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(compute_corr(h)) * dx

def smooth_objective(h, s=100, n=70):
    dx = 2.0 / n
    corr = compute_corr(h) * dx
    # Smooth max
    cmax = np.max(corr)
    return np.max(corr) / s * np.log(np.sum(np.exp(s * (corr - cmax)))) + cmax
    # This is the log-sum-exp trick for stability

def smooth_grad(h, s=100, n=70):
    # Numerical gradient
    eps = 1e-6
    grad = np.zeros_like(h)
    f0 = smooth_objective(h, s, n)
    for i in range(n):
        h[i] += eps
        f1 = smooth_objective(h, s, n)
        h[i] -= eps
        grad[i] = (f1 - f0) / eps
    return grad

from scipy.optimize import minimize

N_PTS = 70
T_SUM = 35.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_obj(x, s):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    gm = np.max(corr)
    return gm + (1.0 / s) * np.log(np.sum(np.exp(s * (corr - gm))))

def smooth_max_grad(x, s):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    gm = np.max(corr)
    shifted = s * (corr - gm)
    shifted -= np.max(shifted)
    w = np.exp(shifted)
    ws = np.sum(w)
    if ws == 0:
        return np.zeros(n)
    w /= ws
    grad = np.zeros(n)
    nm1 = n - 1
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-13:
            continue
        for j in range(n):
            p1 = j + k - nm1
            p2 = j - k + nm1
            v = 0.0
            if 0 <= p1 < n:
                v += 1.0 - x[p1]
            if 0 <= p2 < n:
                v -= x[p2]
            if v != 0.0:
                grad[j] += wk * v
    return grad * dx
