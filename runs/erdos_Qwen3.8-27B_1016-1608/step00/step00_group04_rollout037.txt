def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

    h_new = h.copy()
    h_new[i] = 0.0
    h_new[j] = 1.0
    return h_new, compute_c5(h_new)

def local_search_binary(h, max_rounds=200, time_limit=None, start_t=None):
    """Exhaustive 1-swap local search for binary vectors."""
    n = len(h)
    h = h.copy()
    current_score = compute_c5(h)
    
    for _ in range(max_rounds):
        if time_limit is not None:
            if time.time() - start_t > time_limit:

        improved = False
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        best_delta = 0.0
        best_i, best_j = -1, -1
        
        for i in ones:

                s = compute_c5_swap(h, i, j)
                delta = s - current_score
                if delta < best_delta:
                    best_delta = delta
                    best_i, best_j = i, j

            h[best_i] = 0.0
            h[best_j] = 1.0
            current_score += best_delta
            improved = True

    h[i] = 0.0

import numpy as np
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def local_search(h, max_rounds=2000, time_limit_s=None, t_start=None):
    if t_start is None:
        t_start = time.time()
    h = h.copy()
    cur = compute_c5(h)
    for _ in range(max_rounds):
        if time_limit_s is not None and time.time() - t_start > time_limit_s:
            break
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        best_s = cur
        best_ij = None
        for i in ones:
            h[i] = 0.0
            for j in zeros:
                h[j] = 1.0
                s = compute_c5(h)
                if s < best_s:
                    best_s = s
                    best_ij = (i, j)
                h[j] = 0.0
            h[i] = 1.0
        if best_ij is not None:
            h[best_ij[0]] = 0.0
            h[best_ij[1]] = 1.0
            cur = best_s
        else:
            break
    return h, cur


def random_binary(n, k):
    h = np.zeros(n)
    h[np.random.choice(n, k, replace=False)] = 1.0
    return h


def kick(h, n_swaps):
    h_new = h.copy()
    for _ in range(n_swaps):
        ones = np.where(h_new > 0.5)[0]
        zeros = np.where(h_new < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        h_new[i] = 0.0
        h_new[j] = 1.0
    return h_new
