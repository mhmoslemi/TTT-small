import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def smooth_obj(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    return m + (1.0/tau) * np.log(np.mean(np.exp(tau * (corr - m))))

def smooth_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    tc = tau * dx * corr
    tc -= np.max(tc)
    w = np.exp(tc)
    w /= w.sum()
    
    grad = np.zeros(n)
    for k in range(-(n-1), n):
        j = k + n - 1
        wk = w[j] * dx
        if abs(wk) < 1e-18:
            continue
        if k >= 0:
            grad[k:] += wk * (1.0 - h[:n-k])
            grad[:n-k] -= wk * h[k:]
        else:
            m = -k
            grad[:n-m] += wk * (1.0 - h[m:])
            grad[m:] -= wk * h[:n-m]
    return grad

def project_feasible(h, target):
    h = h.copy()
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
    return h

    if n == new_n:
        return h.copy()
    old_x = np.arange(n) * (n - 1) / max(n - 1, 1)
    new_x = np.arange(new_n) * (new_n - 1) / max(new_n - 1, 1)
    return np.interp(new_x, old_x, h)
