import numpy as np
import time
import random
from scipy.optimize import minimize


def c5_val(h_arr, dx):
    return float(np.max(np.correlate(h_arr, 1.0 - h_arr, mode='full')) * dx)


def smooth_obj(x, alpha, dx):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.clip(corr - m, -50.0, 50.0)
    return (m + np.log(np.sum(np.exp(shifted))) / alpha) * dx


def smooth_obj_pen(x, alpha,
                 dx, target_sum, pen_w):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.clip(corr - m, -50.0, 50.0)
    base = (m + np.log(np.sum(np.exp(shifted))) / alpha) * dx
    sv = np.sum(x) - target_sum
    return base + pen_w * sv * sv


def sum_con_fun(x, target):
    return np.sum(x) - target


def repair_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    for _ in range(300):
        d = target_sum - np.sum(h)
        if abs(d) < 1e-13:

        mask = (h > 1e-9) & (h < 1.0 - 1e-9)
        if not np.any(mask):
            mask = np.ones(n, dtype=bool)
        h[mask] += d / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h


def do_slsqp(h_init, alpha, dx, target_sum, n_points, maxiter):
    bounds = [(0.0, 1.0)] * n_points
    cons = [{'type': 'eq', 'fun': sum_con_fun, 'args': (target_sum,)}]

        res = minimize(smooth_obj, h_init, args=(alpha, dx),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})

    except Exception:

                        method='L-BFGS-B', bounds=bounds,
                        options={'maxiter': maxiter, 'ftol': 1e-15})
        return res2.x
    except Exception:
        return h_init.copy()
