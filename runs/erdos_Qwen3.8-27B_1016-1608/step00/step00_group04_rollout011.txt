def compute_c5(h):

    dx = 2.0 / n

    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return None
    
    best_delta = 0

            h[i], h[j] = h[j], h[i]

                best_pair = (i, j)
