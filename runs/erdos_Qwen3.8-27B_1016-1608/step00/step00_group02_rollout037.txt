import numpy as np
import time

def project_h(h, target):
    """Project h onto {0 <= h <= 1, sum(h) = target} via bisection on lambda."""
    # h_proj = clip(h - lambda, 0, 1), find lambda s.t. sum = target
    lo, hi = -1.0, 2.0  # lambda range
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    return np.clip(h - lam, 0, 1)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1-h, mode='full')) * dx

def gradient_at_max_lag(h, max_idx):
    """Compute gradient of correlate(h,1-h,'full')[max_idx] w.r.t. h."""
    n = len(h)
    lag = max_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:

        idx = np.arange(m)
        grad[idx] += (1.0 - h[idx + lag])

        m = n - lag_abs

        grad[idx + lag_abs] += (1.0 - h[idx])
        grad[idx] -= h[idx + lag_abs]

def project_bounded(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5*(lo+hi)
        if np.sum(np.clip(h-mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo+hi), 0.0, 1.0)

def c5_of(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * 2.0/n)

def grad_lag(h, p):
    n = len(h)
    l = p - (n-1)
    g = np.zeros(n)
    if l >= 0:
        g[l:] += 1.0 - h[:n-l]
        g[:n-l] -= h[l:]
    else:
        la = -l
        g[:n-la] += 1.0 - h[la:]
        g[la:] -= h[:n-la]
    return g
