            hd, cd = gd_optimize(h_init.copy(), n, deadline_main, use_smooth=True)
            if cd < bc:
                bh, bc, bn = hd.copy(), cd, n
            if time.time() < deadline_main:
                hd2, cd2 = gd_optimize(h_init.copy(), n, deadline_main, use_smooth=False)
                if cd2 < bc:
                    bh, bc, bn = hd2.copy(), cd2, n

    deadline_ls = t0 + budget_s * 0.85
    if bh is not None and time.time() < deadline_ls:
        bh, bc, _ = local_search(bh.copy(), bn, deadline_ls, seed=77)

    if bh is not None and time.time() < t0 + budget_s * 0.92:
        bh, bc, _ = pair_swap_search(bh.copy(), bn, t0 + budget_s * 0.92, seed=55)

    if bh is not None and time.time() < t0 + budget_s * 0.95:
        bh, bc, _ = local_search(bh.copy(), bn, t0 + budget_s * 0.95, seed=123)

    if bh is None:
        n = 45
        h = np.random.uniform(0, 1, n)
        bh = project_sum_box(h, n / 2.0)
        bn = n

    bh = project_sum_box(bh, bn / 2.0)
    bc = c5_value(bh)
    return (bh, bc, bn)
