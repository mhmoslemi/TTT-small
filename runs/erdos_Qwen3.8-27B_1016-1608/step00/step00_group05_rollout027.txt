    if d == 0:

    else:

            s = 0.0
            if 0 <= j - d < n:
                s += (1.0 - h[j - d])
            if 0 <= j + d < n:
                s -= h[j + d]
            grad[j] = s

        h_pert = project(best_h + rng.normal(0, 0.04, n), n)
        bh2, bc52 = optimize(h_pert, n, min(t2, t_end), rng, alpha0=0.04, soft_k=12, soft_T=15.0)
        if bc52 < best_c5:
            best_c5, best_h, best_n = bc52, bh2, n

    # Phase 3: random binary restarts (remaining time)
    attempt = 0
    while time.time() < t_end - 20:
        remaining = t_end - 20 - time.time()
        if remaining < 30:
            break
        h_r = np.zeros(n)
        h_r[rng.choice(n, n // 2, replace=False)] = 1.0
        h_r = project(h_r + rng.normal(0, 0.25, n), n)
        sub_end = min(t_end - 20, time.time() + min(remaining, 80))
        bh3, bc53 = optimize(h_r, n, sub_end, rng, alpha0=0.06, soft_k=6, soft_T=5.0)
        if bc53 < best_c5:
            best_c5, best_h, best_n = bc53, bh3, n
        attempt += 1

    # Final: one more polish pass if time remains
    if time.time() < t_end - 5:
        bh_f, bc5_f = optimize(best_h, n, t_end, rng, alpha0=0.02, soft_k=10, soft_T=20.0)
        if bc5_f < best_c5:
            best_c5, best_h = bc5_f, bh_f

    best_h = project(best_h, best_n)
    best_c5, _ = compute_c5(best_h)
    return best_h, best_c5, best_n
