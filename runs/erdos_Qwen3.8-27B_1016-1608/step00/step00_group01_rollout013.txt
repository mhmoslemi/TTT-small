import numpy as np
from scipy.optimize import minimize
import time

_G_TARGET = 0
_G_DX = 0

def _c_sum(h):
    return np.sum(h) - _G_TARGET

def _cj_sum(h):
    return np.ones(len(h))

def _hard_obj(h):
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    return float(np.max(corr)) * _G_DX

def _soft_obj(h, beta):
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    m = np.max(corr)
    shifted = corr - m
    s = np.sum(np.exp(beta * shifted))
    return (m + np.log(s) / len(corr) / beta) * _G_DX

def _feasible(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
    else:
        order = np.argsort(h)
    for idx in order:
        if diff > 0:
            add = min(diff, 1.0 - h[idx])
            h[idx] += add
            diff -= add
        else:
            rem = min(-diff, h[idx])
            h[idx] -= rem
            diff += rem
        if abs(diff) < 1e-14:
            break
    return h

def _slsqp_phase(h, target, dx, n, t0, budget):
    global _G_TARGET, _G_DX
    _G_TARGET = target
    _G_DX = dx
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': _c_sum, 'jac': _cj_sum}]
    best_h, best_c5 = h.copy(), _hard_obj(h)

            hc = _feasible(res.x.copy(), target)
            c5c = _hard_obj(hc)
            if c5c < best_c5:
                best_h, best_c5 = hc, c5c
        except Exception:
            pass
    return best_h, best_c5

    _G_DX = dx

        if c5c < _hard_obj(h):
