import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
from scipy.interpolate import pchip_interpolate
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max(vals, temp=0.01):
    m = np.max(vals)
    return m + temp * np.log(np.sum(np.exp((vals - m) / temp)))

def smooth_c5(h, dx, temp=0.01):
    corr = correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr * dx, temp)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not np.any(mask):
            break
        adj = diff / np.sum(mask)
        h[mask] = np.clip(h[mask] + adj, 0.0, 1.0)
    return h

def binary_push(h, target, strength=5.0):
    hp = 0.5 + 0.5 * np.tanh(strength * (h - 0.5))
    return project_sum(hp, target)

def make_obj(target, dx, temp, pen_w):
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        sc = smooth_c5(h, dx, temp)
        pen = pen_w * (np.sum(h) - target) ** 2
        return sc + pen
    return obj

def optimize_h(x0, n, target, dx, temp=0.01, pen_w=500.0, max_iter=2000):
    x0 = np.clip(x0, 0.0, 1.0)
    obj = make_obj(target, dx, temp, pen_w)
    res = minimize(obj, x0, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-10})
    h_opt = project_sum(res.x, target)
    return h_opt, compute_c5(h_opt)

def generate_thue_morse(n):
    return np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.time()
    best_h = None
    best_c5 = np.inf

    n_orig = len(h_init)

    dx_orig = 2.0 / n_orig
    best_h = project_sum(h_init, target_orig)
    best_c5 = compute_c5(best_h)

    # Phase 1: Optimize from initial
    h_opt, c5_opt = optimize_h(best_h, n_orig, target_orig, dx_orig, 0.01, 500, 2000)
    if c5_opt < best_c5:

    # Binary push and re-optimize
    h_pushed = binary_push(best_h, target_orig, 5.0)
    h_opt, c5_opt = optimize_h(h_pushed, n_orig, target_orig, dx_orig, 0.005, 1000, 1500)
    if c5_opt < best_c5:
        best_c5, best_h = c5_opt, h_opt

    # Phase 2: Random restarts
    tm_base = generate_thue_morse(256)
    for i in range(60):
        if time.time() - t_start > budget_s - 180:
            break
        r = rng.random(n_orig)
        mtype = i % 6
        if mtype == 0:
            freq = rng.integers(1, 8)
            phase = rng.random() * 2 * np.pi
            r = 0.5 + 0.35 * np.sin(2 * np.pi * freq * np.linspace(0, 1, n_orig) + phase)
        elif mtype == 1:

    return best_h, best_c5, len(best_h)
