import numpy as np
from scipy.optimize import minimize
from functools import partial
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr)), corr

def smooth_max(v, t):
    m = np.max(v)
    return m + t * np.log(np.sum(np.exp((v - m) / t)))

def project_to_sum(h, target):
    h = np.asarray(h, dtype=float)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)

def obj_fn(x, temp, n):
    h_t = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h_t, 1.0 - h_t, mode='full')
    return smooth_max(corr, temp) * (2.0 / n)

def const_fn(x, target):
    return float(np.sum(x) - target)

def binary_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

    n = n0
    target = n / 2.0
    h = project_to_sum(h0, target)
    c5, _ = evaluate_c5(h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': partial(const_fn, target=target)}]

    return best_h, final_c5, len(best_h)
