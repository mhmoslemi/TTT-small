    return x_max + temp * np.log(np.sum(np.exp((x - x_max) / temp)))

    h_init = np.clip(h_init, 0, 1)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

    c = corr * dx
    m = np.max(c)
    return m + temp * np.log(np.sum(np.exp((c - m) / temp)))

    c = corr * dx
    m = np.max(c)
    p = np.exp((c - m) / temp)
    p /= p.sum()
    n = len(h)
    g = np.zeros(n)
    for offset in range(-(n-1), n):
        k = n - 1 + offset
        w = p[k] * dx
        if w < 1e-14:
            continue
        # Term 1: (1-h[j+offset]) for j where j in [0,n-1] and j+offset in [0,n-1]
        j_lo = max(0, -offset)
        j_hi = min(n-1, n-1-offset)
        if j_lo <= j_hi:

        j_lo2 = max(0, offset)
