            best_c5 = c5
            best_h = h_bin.copy()

    # Add a few random starts for diversity
    for n_try in [80, 96, 100]:
        if remaining() < 90:
            break
        if h0 is not None and n_try == len(h0):
            continue
        h_rand = np.zeros(n_try)
        h_rand[rng.choice(n_try, n_try // 2, replace=False)] = 1.0
        c5 = evaluate(h_rand)
        candidates.append((h_rand, c5))

    # Optimize each candidate with steepest descent + SA
    for idx, (h_cand, _) in enumerate(candidates):
        if remaining() < 50:
            break
        n = len(h_cand)

        # Phase A: Steepest descent
        t_alloc = min(remaining() * 0.35, 180)
        h_opt, c5_opt = steepest_descent(h_cand, time_limit=t_alloc)

        # Phase B: Simulated annealing to escape local optima
        if remaining() > 50:
            t_sa = min(remaining() * 0.15, 80)
            h_sa, c5_sa = simulated_annealing(h_opt, time_limit=t_sa, rng=rng)
            if c5_sa < c5_opt:
                h_opt, c5_opt = h_sa, c5_sa

        # Phase C: Steepest descent again after SA
        if remaining() > 40:
            t2 = min(remaining() * 0.15, 90)
            h_opt, c5_opt = steepest_descent(h_opt, time_limit=t2)

        # Phase D: Random kick + re-optimize
        if remaining() > 50:
            h_kick = h_opt.copy()
            for _ in range(max(3, n // 6)):
                ones = np.where(h_kick == 1)[0]
                zeros = np.where(h_kick == 0)[0]
                if len(ones) == 0 or len(zeros) == 0:
                    break
                i = ones[rng.integers(len(ones))]
                j = zeros[rng.integers(len(zeros))]
                h_kick[i] = 0.0
                h_kick[j] = 1.0
            t3 = min(remaining() * 0.1, 60)
            h_kick_opt, c5_kick = steepest_descent(h_kick, time_limit=t3)
            if c5_kick < c5_opt:
                h_opt, c5_opt = h_kick_opt, c5_kick

        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Final polish: long steepest descent on best
    if remaining() > 20:
        t_final = min(remaining() * 0.5, 300)
        best_h, best_c5 = steepest_descent(best_h, time_limit=t_final)

    # Ensure valid output
    best_h = np.clip(best_h, 0.0, 1.0)
    n = len(best_h)
    # Verify sum constraint
    s = np.sum(best_h)
    target = n / 2.0
    if abs(s - target) > 1e-9:
        # Fix: adjust one element
        if s > target:
            idx = np.where(best_h > 0)[0]
            if len(idx) > 0:
                k = idx[np.argmax(best_h[idx])]
                best_h[k] = max(0.0, best_h[k] - (s - target))
        else:
            idx = np.where(best_h < 1)[0]
            if len(idx) > 0:
                k = idx[np.argmin(best_h[idx])]
                best_h[k] = min(1.0, best_h[k] + (target - s))

    best_c5 = evaluate(best_h)
    return best_h, best_c5, n
