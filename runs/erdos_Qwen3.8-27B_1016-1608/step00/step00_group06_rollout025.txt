from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to_n(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return project_feasible(h.copy(), n_new)
    x_old = np.arange(n_old, dtype=float)
    x_new = np.linspace(0.0, n_old - 1.0, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def soft_max_val(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    m = vals.max()
    return float(m + np.log(np.sum(np.exp(beta * (vals - m)))) / beta)

def soft_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    m = vals.max()
    exps = np.exp(beta * (vals - m))
    total = exps.sum()
    if total < 1e-300:
        return np.zeros(n)
    p = exps / total
    grad = np.zeros(n)
    for k in range(len(corr)):
        if p[k] < 1e-8:
            continue
        shift = k - (n - 1)
        w = p[k] * dx
        if shift >= 0:
            i_max = n - shift
            for i in range(i_max):
                grad[i] += w * (1.0 - h[i + shift])
                grad[i + shift] += w * (-h[i])
        else:
            s = -shift
            for i in range(s, n):
                grad[i] += w * (1.0 - h[i - s])
                grad[i - s] += w * (-h[i])
    return grad

def pgd_optimize(h, n, time_limit=30.0, max_iter=5000):
    t0 = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    beta = 15.0
    lr = 0.05
    prev_obj = soft_max_val(h, beta)
    
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        if it % 200 == 200 and beta < 80:
            beta *= 1.5
            lr *= 0.6
        g = soft_max_grad(h, beta)
        h_new = project_feasible(h - lr * g, n)
        obj_new = soft_max_val(h_new, beta)
        if obj_new < prev_obj:
            h = h_new
            prev_obj = obj_new
            lr = min(lr * 1.03, 0.3)
        else:
            lr *= 0.7
            if lr < 1e-6:

                beta = max(beta * 1.3, 10)
                prev_obj = soft_max_val(h, beta)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return project_feasible(best_h, n), best_c5

def slsqp_optimize(h, n, time_limit=20.0):
    t0 = time.time()
    n_constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
                      'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n

        return soft_max_val(x, 40.0)

            c5_out = compute_c5(h_out)
            if c5_out < compute_c5(h):
                return h_out, c5_out
    except Exception:
        pass
    return h, compute_c5(h)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = h < 1.0 - 1e-14 if diff > 0 else h > 1e-14
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to_n(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return project_feasible(h.copy(), n_new)
    x_old = np.arange(n_old, dtype=float)
    x_new = np.linspace(0.0, n_old - 1.0, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def soft_max_val(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    m = vals.max()
    return float(m + np.log(np.sum(np.exp(beta * (vals - m)))) / beta)

def soft_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    m = vals.max()
    exps = np.exp(beta * (vals - m))
    p = exps / exps.sum()
    grad = np.zeros(n)
    for k in range(len(corr)):
        if p[k] < 1e-8:
            continue
        w = p[k] * dx
        shift = k - (n - 1)
        if shift >= 0:
            grad[:n - shift] += w * (1.0 - h[shift:])
            grad[shift:] += w * (-h[:n - shift])
        else:
            s = -shift
            grad[s:] += w * (1.0 - h[:n - s])
            grad[:n - s] += w * (-h[s:])
    return grad

def pgd_optimize(h, n, time_limit=30.0, max_iter=5000):
    t0 = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    beta = 12.0
    lr = 0.05
    prev_obj = soft_max_val(h, beta)
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        if it % 300 == 300 and beta < 100:
            beta *= 1.6
            lr *= 0.55
        g = soft_max_grad(h, beta)
        norm_g = np.linalg.norm(g)
        if norm_g < 1e-12:
            break
        h_new = project_feasible(h - lr * g, n)
        obj_new = soft_max_val(h_new, beta)
        if obj_new < prev_obj:
            h = h_new
            prev_obj = obj_new
            lr = min(lr * 1.04, 0.4)
        else:
            lr *= 0.65
            if lr < 1e-6:
                lr = 0.02
                beta = max(beta * 1.4, 12.0)
                prev_obj = soft_max_val(h, beta)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return project_feasible(best_h, n), best_c5

def slsqp_optimize(h, n, time_limit=20.0):
    t0 = time.time()
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n

    def obj_fn(x):
        if time.time() - t0 > time_limit:
            raise TimeoutError
        return soft_max_val(x, 50.0)

    try:
        res = minimize(obj_fn, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 150, 'ftol': 1e-8})
        if hasattr(res, 'fun') and res.fun < soft_max_val(h, 50.0):
            h_out = project_feasible(res.x, n)
            c5_out = compute_c5(h_out)
            if c5_out < compute_c5(h):
                return h_out, c5_out
    except Exception:
        pass
    return h, compute_c5(h)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()

    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = np.full(95, 0.5)

    n_init = len(h_init)
    h_init = project_feasible(h_init, n_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)

    n_candidates = sorted(set([n_init, 50, 60, 70, 80, 90, 95, 100, 110, 120, 135, 150]))
    n_candidates = [n for n in n_candidates if 40 <= n <= 200]

    for n in n_candidates:
        if time.time() - t_start > budget_s - 50:
            break
        h_curr = interp_to_n(h_init, n)
        c5_curr = compute_c5(h_curr)
        if c5_curr < best_c5:
            best_c5 = c5_curr
            best_h = h_curr.copy()

        time_left = budget_s - (time.time() - t_start) - 40
        pg_time = min(35.0, time_left * 0.35)
        if pg_time > 5:
            h_opt, c5_opt = pgd_optimize(h_curr, n, time_limit=pg_time)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
            h_curr = h_opt

        time_left = budget_s - (time.time() - t_start) - 20
        if time_left > 15:
            h_sq, c5_sq = slsqp_optimize(h_curr, n, time_limit=min(15.0, time_left))
            if c5_sq < best_c5:
                best_c5 = c5_sq
                best_h = h_sq.copy()
                h_curr = h_sq

        time_left = budget_s - (time.time() - t_start) - 10
        for trial in range(4):
            if time_left < 8:
                break
            h_base = best_h if len(best_h) == n else interp_to_n(best_h, n)
            scale = 0.06 + 0.04 * trial
            h_pert = h_base + scale * rng.randn(n)
            h_pert = project_feasible(h_pert, n)
            h_r, c5_r = pgd_optimize(h_pert, n, time_limit=min(12.0, time_left * 0.4))
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
            time_left = budget_s - (time.time() - t_start) - 10

    time_left = budget_s - (time.time() - t_start)
    if time_left > 10:
        n_best = len(best_h)
        h_pol, c5_pol = pgd_optimize(best_h, n_best, time_limit=min(25.0, time_left - 5))
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
