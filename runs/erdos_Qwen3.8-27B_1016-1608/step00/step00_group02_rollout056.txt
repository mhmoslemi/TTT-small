    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-10:
        return h
    n = len(h)
    h_shift = h + (target - s) / n
    if np.all(h_shift >= 0) and np.all(h_shift <= 1):
        return h_shift
    if s > 1e-10:
        h_scale = h * (target / s)
        if np.all(h_scale <= 1.0 + 1e-10):
            return np.clip(h_scale, 0.0, 1.0)
    return np.clip(h_shift, 0.0, 1.0)

    new_x = np.linspace(0, 2, new_n, endpoint=False)
    return np.interp(new_x, old_x, h)

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    n = len(h)
    h_shift = h + (target - s) / n
    if np.all(h_shift >= -1e-12) and np.all(h_shift <= 1.0 + 1e-12):
        return np.clip(h_shift, 0.0, 1.0)
    if s > 1e-12:
        h_scale = h * (target / s)
        if np.all(h_scale <= 1.0 + 1e-12):
            return np.clip(h_scale, 0.0, 1.0)
    return np.clip(h_shift, 0.0, 1.0)

def nm_obj(x, target):
    return evaluate_c5(repair(x, target))

def interp_to_n(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0, 2, old_n, endpoint=False)
    new_x = np.linspace(0, 2, new_n, endpoint=False)
    return np.interp(new_x, old_x, h)

def local_search_cont(h, target, n_iters, step_init, seed):
    rng = np.random.RandomState(seed)
    h = h.copy()
    n = len(h)
    best = evaluate_c5(h)
    step = step_init
    for it in range(n_iters):
        i = rng.randint(n)
        d = rng.choice([-1.0, 1.0]) * step * rng.uniform(0.5, 1.5)
        h_try = h.copy()
        h_try[i] += d
        h_try = repair(h_try, target)
        c5 = evaluate_c5(h_try)
        if c5 < best:
            h = h_try
            best = c5
        else:
            step *= 0.995
            if step < 1e-7:
                break
    return h, best

def sa_binary_search(h_init, target, n_iters, seed, t_start, t_end):

    n = len(h_init)
    h = np.zeros(n)
    sorted_idx = np.argsort(h_init)
    n_ones = int(round(target))
    if n_ones * 2 != n:
        return local_search_cont(h_init, target, n_iters, 0.05, seed)
    h[sorted_idx[n - n_ones:]] = 1.0
    best = evaluate_c5(h)
    best_h = h.copy()
    ones_idx = np.where(h == 1.0)[0]
    zeros_idx = np.where(h == 0.0)[0]
    t_current = t_start
    t_min = t_end
    for it in range(n_iters):
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        h[i] = 0.0
        h[j] = 1.0
        c5 = evaluate_c5(h)
        if c5 < best or rng.rand() < np.exp(-(c5 - best) / (0.001 * t_current)):
            best = min(best, c5)
            if c5 < best:
                best_h = h.copy()
                best = c5
            ones_idx = np.where(h == 1.0)[0]
            zeros_idx = np.where(h == 0.0)[0]
        else:
            h[i] = 1.0
            h[j] = 0.0
        t_current = max(t_min, t_current * 0.999)
    return best_h, best

        h_try = repair(res.x, target45)
