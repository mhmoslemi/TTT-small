def try_binary(h, ts, n):
    """Round to binary and try local search"""
    # Round: h[i] > 0.5 → 1, else 0
    h_bin = (h > 0.5).astype(float)
    # Fix sum
    cur_sum = np.sum(h_bin)
    if cur_sum > ts:
        # Remove (cur_sum - ts) ones (those with lowest h values)
        ones = np.where(h_bin == 1)[0]
        order = np.argsort(h[ones])
        for i in order:
            if cur_sum == ts:
                break
            h_bin[ones[i]] = 0
            cur_sum -= 1
    elif cur_sum < ts:
        zeros = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros])
        for i in order:
            if cur_sum == ts:

            cur_sum += 1

    improved = True
    while improved:

                res = minimize(objective_fn, h_start, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 350, 'ftol': 1e-14})
            else:
                res = minimize(objective_fn, h_start, method='COBYLA',
                               options={'maxiter': 500, 'rhobeg': 0.1})

            h_new = project_feasible(res.x, ts)
            c5_new = compute_c5(h_new)

            h_new, c5_new = coord_desc(h_new, ts, 1500, steps_small, rng)

            if c5_new < best_c5 - 1e-13:
                best_c5 = c5_new
                best_h = h_new
                pilosity = 0
            else:
                pilosity += 1
        except Exception:
            pilosity += 1

        attempt += 1

        if pilosity > 15 and attempt > 20:
            best_h = project_feasible(best_h + rng.randn(n) * 0.08, ts)
            pilosity = 0

    # Final polish
    if time.time() - t0 < budget_s - 1:
        best_h, best_c5 = coord_desc(best_h, ts, 20000, steps_small, rng)

    best_h = project_feasible(best_h, ts)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
