import numpy as np
from scipy import optimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_gradient(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # Smooth max gradient
    m = np.max(corr)
    exp_vals = np.exp(beta * (corr - m))
    weights = exp_vals / np.sum(exp_vals)
    
    # lags
    lags = np.arange(-(n-1), n)  # length 2n-1
    grad = np.zeros(n)
    
    for idx, k in enumerate(lags):
        w = weights[idx] * beta * dx
        if w < 1e-15:
            continue
        # d(corr_k)/dh[j] = (1-h[j-k]) - h[j+k] for valid indices
        # term1: (1-h[j-k]) for valid j-k in [0,n-1]
        # j-k in [0,n-1] => j in [k, k+n-1], intersected with [0,n-1]
        js = max(0, k)
        je = min(n, k + n)
        if js < je:
            grad[js:je] += w * (1.0 - h[js-k:je-k])
        # term2: -h[j+k] for valid j+k in [0,n-1]
        # j+k in [0,n-1] => j in [-k, n-k-1], intersected with [0,n-1]
        js = max(0, -k)
        je = min(n, n - k)
        if js < je:
            grad[js:je] += -w * h[js+k:je+k]
    
    return grad

def project_feasible(h, target_sum):
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2.0, 0.0, 1.0)

    h = project_feasible(h_init.copy(), target_sum)
    best_local = h.copy()
    best_c5_local = compute_c5(h)

        if time.time() - start_time > budget_s - 5:
            break
        
        # Anneal beta and learning rate
        frac = it / max(1, n_iters - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac
        
        grad = compute_gradient(h, beta)
        h_new = h - lr * grad
        h = project_feasible(h_new, target_sum)
        
        c5 = compute_c5(h)
        if c5 < best_c5_local:

        h_start = np.clip(h_start, 0.0, 1.0)

        n_iters = 5000 if n <= 120 else 3000
        h_opt, c5_opt = run_opt(h_start, n, n_iters, deadline)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

        if time.time() < deadline - 80:
            h_pert = h_start + rng.randn(n) * 0.12
            h_pert = np.clip(h_pert, 0.0, 1.0)
            h_opt2, c5_opt2 = run_opt(h_pert, n, 2500, deadline)
            if c5_opt2 < best_c5:
                best_c5 = c5_opt2
                best_h = h_opt2.copy()

    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n_final
