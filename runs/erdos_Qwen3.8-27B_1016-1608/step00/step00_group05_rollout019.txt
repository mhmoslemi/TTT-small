                                     method='SLSQP', bounds=bounds, constraints=cons,
                                     options={'maxiter': 1000, 'ftol': 1e-14})
                        if res.x is not None:
                            h_new = repair(res.x, n)
                            c5_new = compute_c5(h_new, dx)
                            if c5_new < best_c5:
                                best_c5 = c5_new
                                best_h = h_new
                    except Exception:
                        pass

    # Phase 5: Fine-tune with very low temperature
    if elapsed() < budget_s - 30:
        for temp in [0.0005, 0.0001, 0.00005]:
            if time.time() - t_start > budget_s - 30:
                break
            try:
                res = minimize(smooth_objective, best_h, args=(dx, temp),
                            method='SLSQP', bounds=bounds, constraints=cons,
                            options={'maxiter': 3000, 'ftol': 1e-16})
                if res.x is not None:
                    h_new = repair(res.x, n)
                    c5_new = compute_c5(h_new, dx)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new
            except Exception:
                pass

    # Final repair and evaluation
    best_h = repair(best_h, n)
    best_c5 = compute_c5(best_h, dx)
    return best_h, best_c5, n
