            c5_b = _eval_c5(h_bin, n)
            if c5_b < best_c5:
                best_c5, best_h, best_n = c5_b, h_bin.copy(), n
                # Quick greedy on this binary
                h_bin, c5_b2 = _sa_search(h_bin.copy(), n, rng, t0, budget_s, 0.99)
                if c5_b2 < best_c5:
                    best_c5, best_h = c5_b2, h_bin.copy()

    c5_final = _eval_c5(best_h, best_n)
    if c5_final < best_c5:
        best_c5 = c5_final
    return best_h, best_c5, best_n
