def try_transfer(h, i, j, delta):
    """Try transferring delta from j to i. Returns (new_h, new_score) or None."""
    if h[j] - delta < -1e-10 or h[i] + delta > 1.0 + 1e-10:
        return None
    new_h = h.copy()
    new_h[i] += delta
    new_h[j] -= delta
    score = compute_score(new_h)
    return new_h, score

    n = len(h)

        if time_limit and time.time() - start_time > time_limit:

        np.random.shuffle(range(n))  # randomize order

    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def compute_corr(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode="full")

def project(h, target_sum):
    """Project h onto {sum(h)=target_sum, 0<=h<=1} approximately."""
    h = h.copy()
    h = np.clip(h, 0, 1)
    # Adjust sum
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        # Distribute adjustment
        if diff > 0:
            # Need to add: increase elements that are < 1
            mask = h < 1.0
            if np.sum(mask) > 0:
                h[mask] += diff / np.sum(mask)
                h = np.clip(h, 0, 1)
        else:
            # Need to subtract: decrease elements that are > 0
            mask = h > 0.0
            if np.sum(mask) > 0:
                h[mask] += diff / np.sum(mask)
                h = np.clip(h, 0, 1)
    return h

def local_search(h, candidate_deltas, max_sweeps=500, time_check=None, t_start=None):
