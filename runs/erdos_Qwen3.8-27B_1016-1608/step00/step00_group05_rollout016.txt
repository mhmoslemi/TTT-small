```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum}"""
    h = np.clip(h, 0, 1)
    # Adjust for sum constraint
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-15:
        # Distribute adjustment
        h += diff / len(h)
        h = np.clip(h, 0, 1)
        # Iterate if needed
        for _ in range(50):
            diff = target_sum - np.sum(h)
            if abs(diff) < 1e-12:
                break
            mask = (h > 0) & (h < 1) if diff > 0 else (h < 1) & (h > 0)
            if diff > 0:
                mask = h < 1
            else:
                mask = h > 0
            if np.any(mask):
                h[mask] += diff / np.sum(mask)
            else:
                break
            h = np.clip(h, 0, 1)
    return h

def subgradient(h, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(corr)
    active = corr >= max_val - 1e-10
    grad = np.zeros(n)
    # Vectorized computation
    for k in np.where(active)[0]:
        offset = k - (n - 1)
        # For each j: contribution is (1-h[j-offset]) - h[j+offset] (when valid)
        j_arr = np.arange(n)
        idx_down = j_arr - offset
        idx_up = j_arr + offset
        valid_down = (idx_down >= 0) & (idx_down < n)
        valid_up = (idx_up >= 0) & (idx_up < n)
        grad[valid_down] += 1.0 - h[idx_down[valid_down]]
        grad[valid_up] += -h[idx_up[valid_up]]
    return grad, float(max_val * dx)
```
