def smooth_obj_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    shifted = np.clip((corr - m) / eps, -50, 50)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    w = exp_s / sum_exp

    grad = np.zeros(n)

    for k in range(2 * n - 1):
        wi = w[k]
        if wi < 1e-10:
