    n = len(h)

    cur_score = compute_score(cur)

                cur_score = new_score
                if new_score < best_score:

import numpy as np
import math
import time

def compute_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float((corr * dx).max())

def make_random_binary(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h

def greedy_hill_climb(h, deadline):
    n = len(h)
    cur = h.astype(np.float64).copy()
    cur_score = compute_score(cur)
    best = cur.copy()
    best_score = cur_score
    for _ in range(500):
        if time.time() > deadline:
            break
        ones = np.where(cur > 0.5)[0]
        zeros = np.where(cur < 0.5)[0]
        best_gain = 1e-30
        best_i = -1
        best_j = -1
        for i in ones:
            for j in zeros:
                cur[i] = 0.0
                cur[j] = 1.0
                s = compute_score(cur)
                cur[i] = 1.0
                cur[j] = 0.0
                gain = cur_score - s
                if gain > best_gain:
                    best_gain = gain
                    best_i = i
                    best_j = j
        if best_i < 0:
            break
        cur[best_i] = 0.0
        cur[best_j] = 1.0
        cur_score -= best_gain
        if cur_score < best_score:
            best_score = cur_score
            best = cur.copy()
    return best, best_score
