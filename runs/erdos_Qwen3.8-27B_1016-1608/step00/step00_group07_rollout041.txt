def compute_c5_array(h):

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = h.sum() - target_sum
    if abs(diff) < 1e-15:
        return h
    # Water-filling projection
    h = h.copy()
    if diff > 0:
        # Reduce: take from largest
        order = np.argsort(-h)
        remaining = diff
        per = diff / target_sum  # uniform reduction amount (approx)
        for idx in order:
            if remaining <= 0:
                break
            take = min(h[idx], remaining)
            h[idx] -= take
            remaining -= take
    else:
        # Increase: give to smallest
        order = np.argsort(h)
        remaining = -diff
        for idx in order:
            if remaining <= 0:
                break
            take = min(1.0 - h[idx], remaining)
            h[idx] += take
            remaining -= take
    return h

def compute_c5_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def c5_max(h):
    return float(np.max(compute_c5_array(h)))

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = h.sum() - target_sum
    if abs(diff) < 1e-12:
        return h
    h = h.copy()
    if diff > 0:
        order = np.argsort(-h)
        remaining = diff
        for i in order:
            if remaining <= 0:
                break
            take = min(h[i], remaining)
            h[i] -= take
            remaining -= take
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in order:
            if remaining <= 0:
                break
            take = min(1.0 - h[i], remaining)
            h[i] += take
            remaining -= take
    return h

    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(c)
    exp_c = np.exp(beta * (c - max_c))

    part2 = np.correlate(W, h, mode='full')[:n][::-1]

import numpy as np
import time

def c5_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def c5_val(h):
    return float(np.max(c5_array(h)))

def project(h, ts):
    h = np.clip(h, 0.0, 1.0)
    d = h.sum() - ts
    if abs(d) < 1e-14:
        return h
    h = h.copy()
    if d > 0:
        for i in np.argsort(-h):
            if d <= 0: break
            t = min(h[i], d)
            h[i] -= t; d -= t
    else:
        for i in np.argsort(h):
            if d >= 0: break
            t = min(1.0 - h[i], -d)
            h[i] += t; d += t
    return h

def grad_sm(h, beta):
    n = len(h); dx = 2.0/n
    c = np.correlate(h, 1.0-h, mode='full') * dx
    mc = c.max()
    e = np.exp(beta*(c - mc))
    W = e / e.sum()
    p1 = np.convolve(W, 1.0-h, mode='full')[n-1:2*n-1]
    p2 = np.correlate(W, h, mode='full')[:n][::-1]
    return (p1 - p2) * dx, float(mc)

def optimize(h0, beta0, lr0, maxit, tlim):
    t0 = time.time()
    h = h0.astype(float).copy()
    n = len(h); ts = n/2.0
    h = project(h, ts)
    bh = h.copy(); bv = c5_val(bh)
    beta = beta0; lr = lr0
    for _ in range(maxit):
        if time.time() - t0 > tlim: break
        g, _ = grad_sm(h, beta)
        hn = h - lr*g
        hn = project(hn, ts)
        v = c5_val(hn)
        if v < bv:
            bh = hn.copy(); bv = v
            beta = min(beta*1.03, 500.0)
            lr = max(lr*0.9995, 0.0001)

            if lr < 0.0002:
                lr = lr0*0.3
                h = project(h + 0.02*np.random.randn(n), ts)
        h = hn
    return bh, bv
