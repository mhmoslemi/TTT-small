import numpy as np
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def fix_sum(h, target):
    """Ensure sum(h) = target while keeping 0<=h<=1."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    h += diff / len(h)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(h) if diff > 0 else np.argsort(-h)
        for j in order:
            if abs(diff) < 1e-12:
                break
            h[j] = np.clip(h[j] + diff, 0.0, 1.0)
            diff = target - h.sum()
    return h

def subgrad_step(h, alpha):
    """One subgradient step: minimize max_k corr[k]."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_max = int(np.argmax(corr))
    offset = k_max - (n - 1)
    
    idx = np.arange(n)
    # A: j where 0<=j<n and 0<=j-offset<n => max(0,offset)<=j<min(n,n+offset)
    a_lo = max(0, offset)
    a_hi = min(n, n + offset)
    # B: j where 0<=j<n and 0<=j+offset<n => max(0,-offset)<=j<min(n,n-offset)
    b_lo = max(0, -offset)
    b_hi = min(n, n - offset)
    
    grad = np.zeros(n)
    in_a = (idx >= a_lo) & (idx < a_hi)
    in_b = (idx >= b_lo) & (idx < b_hi)
    
    grad[in_a] += (1.0 - h[idx[in_a] - offset])
    grad[in_b] -= h[idx[in_b] + offset]
    
    return h - alpha * grad

def local_search(h, time_budget):
    """Pair-swap local search with occasional random restarts."""
    n = len(h)
    target = n / 2.0
    h = fix_sum(h, target).copy()
    best_c5 = eval_c5(h)
    best_h = h.copy()
    start = time.time()
    
    no_improve_rounds = 0
    while time.time() - start < time_budget:
        improved = False
        # Shuffle order for diversity
        pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
        np.random.shuffle(pairs)
        
        for i, j in pairs:
            if time.time() - start >= time_budget:
                break
            if abs(h[i] - h[j]) < 1e-10:
                continue
            h[i], h[j] = h[j], h[i]
            c5 = eval_c5(h)
            if c5 < best_c5 - 1e-12:
                best_c5 = c5
                best_h = h.copy()
                improved = True
            else:
                h[i], h[j] = h[j], h[i]
        
        h = best_h.copy()
        if not improved:
            no_improve_rounds += 1
            if no_improve_rounds > 3:
                break
        else:
            no_improve_rounds = 0
    
    return best_h, best_c5

def sg_descent(h, time_budget):
    """Subgradient descent for continuous h."""
    n = len(h)
    target = n / 2.0
    h = fix_sum(h, target).copy()
    best = h.copy()
    best_c5 = eval_c5(h)
    start = time.time()

        if time.time() - start >= time_budget:

        alpha = 4.0 / (1.0 + 0.1 * it)
        h = subgrad_step(h, alpha)
        h = fix_sum(h, target)
        c5 = eval_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
    
    return best, best_c5
