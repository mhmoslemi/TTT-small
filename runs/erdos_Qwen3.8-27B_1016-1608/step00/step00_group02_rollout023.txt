        if c5_alt < best_c5 * 0.98:
            h_alt, c5_alt = local_search(h_alt, rng, 100000, min(dl4, start_time + budget_s * 0.9))
            update_best(h_alt, c5_alt)

    # Phase 5: Try even-length alternatives (n=44, n=46)
    if time.time() < start_time + budget_s * 0.88:
        for n_alt in [44, 46, 50]:
            if time.time() > start_time + budget_s * 0.92:
                break
            h_alt = make_binary_start(n_alt, seed + n_alt)
            c5_alt, _ = local_search(h_alt, rng, 80000, start_time + budget_s * 0.92)
            update_best(h_alt, c5_alt)

    # Phase 6: Upsample to 90 and search
    if time.time() < start_time + budget_s * 0.85 and len(best_h) < 90:
        n2 = 90
        h_up = upsample(best_h, n2)
        h_up = fix_sum(h_up, n2 / 2.0)
        c5_up = evaluate_c5(h_up)
        h_up, c5_up = local_search(h_up, rng, 150000, start_time + budget_s - 5)
        update_best(h_up, c5_up)

    # Final time: if we still have time, do one more pass
    if time.time() < start_time + budget_s - 10:
        h_work = best_h.copy()
        h_work, c5 = local_search(h_work, rng, 200000, start_time + budget_s - 3)
        update_best(h_work, c5)

    # Ensure final validity
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = fix_sum(best_h, len(best_h) / 2.0)
    final_c5 = evaluate_c5(best_h)

    return best_h, final_c5, len(best_h)
