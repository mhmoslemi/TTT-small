    n0 = len(h0)
    best_h = h0.copy()
    best_n = n0
    best_c5 = float(np.max(compute_corr(h0, n0)))
    
    # Phase 1: Main optimization from initial (50%)
    alloc1 = 0.50 * budget_s
    h_opt, c5_opt = optimize_from(h0, n0, alloc1, rng, top_k=5, lr_init=0.4)
    if c5_opt < best_c5:
        best_c5, best_h, best_n = c5_opt, h_opt, n0
    
    # Phase 2: Continue with more smoothing / different top_k (20%)
    alloc2 = 0.20 * budget_s
    h_opt2, c5_opt2 = optimize_from(best_h, best_n, alloc2, rng, top_k=3, lr_init=0.2)
    if c5_opt2 < best_c5:
        best_c5, best_h, best_n = c5_opt2, h_opt2, best_n
    
    # Phase 3: Local perturbations (15%)
    alloc3_total = 0.15 * budget_s
    for pkt in range(5):
        rem = budget_s - (time.time() - t0) - 20
        if rem < 30:
            break
        alloc3 = min(alloc3_total / 5.0, rem)
        pert = rng.randn(best_n) * 0.008
        h_pert = project_feasible(best_h + pert, best_n)
        h_opt3, c5_opt3 = optimize_from(h_pert, best_n, alloc3, rng, top_k=4, lr_init=0.3)
        if c5_opt3 < best_c5:
            best_c5, best_h, best_n = c5_opt3, h_opt3, best_n
    
    # Phase 4: Try different n (15%)
    alloc4_total = 0.15 * budget_s
    for n_try in [56, 84, 112]:
        rem = budget_s - (time.time() - t0) - 15
        if rem < 30:
            break
        alloc4 = min(alloc4_total / 3.0, rem)
        x_old = np.arange(n0) * (2.0 / n0)
        x_new = np.arange(n_try) * (2.0 / n_try)
        h_interp = np.interp(x_new, x_old, h0)
        h_interp = project_feasible(h_interp, n_try)
        h_opt4, c5_opt4 = optimize_from(h_interp, n_try, alloc4, rng, top_k=5, lr_init=0.3)
        if c5_opt4 < best_c5:
            best_c5, best_h, best_n = c5_opt4, h_opt4, n_try
    
    # Final check
    best_c5 = float(np.max(compute_corr(best_h, best_n)))
    return best_h, best_c5, best_n
