import numpy as np
from scipy import optimize
import time

_DX = 0.0
_TARGET = 0.0


def c5_value(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))


def project_to_sum(h, target):
    h = np.array(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        adj = diff / cnt
        h[mask] += adj
        h = np.clip(h, 0.0, 1.0)
    return h


def _obj(h):
    return c5_value(h, _DX)


def _con_eq(h):
    return np.sum(h) - _TARGET
