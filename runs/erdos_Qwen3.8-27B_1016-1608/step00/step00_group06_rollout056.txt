        if s < 1e-15:
            h[:] = target / n
            break
        ratio = target / s
        h_new = np.clip(h * ratio, 0.0, 1.0)
        if np.allclose(h_new, h, atol=1e-14):
            break
        h = h_new

    if n_ones > target_ones:
        idx = np.where(h == 1)[0]
        h[idx[:n_ones - target_ones]] = 0
    elif n_ones < target_ones:
        idx = np.where(h == 0)[0]
        h[idx[:target_ones - n_ones]] = 1

    zeros = list(np.where(h == 0)[0])

        improved = False
        max_iter -= 1
        best_delta = best_score
        best_pair = None

                h[i] = 0

            h[i] = 0
            h[j] = 1
            ones.remove(i)
            zeros.remove(j)
            zeros.append(i)
            ones.append(j)
            best_score = best_delta
            improved = True
    return h, best_score
