```python
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def project_to_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = 0.5 * n
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() == 0:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
            break
        step = diff / mask.sum()
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    return h

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_score(h, n):
    return np.max(get_corr(h)) * (2.0 / n)

def obj_smooth(h, n, beta, lam):
    dx = 2.0 / n
    corr = get_corr(h) * dx
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta
    penalty = lam * (np.sum(h) - 0.5 * n) ** 2
    return val + penalty

def grad_smooth(h, n, beta, lam):
    dx = 2.0 / n
    corr = get_corr(h) * dx
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    p = exps / exps.sum()
    grad = np.zeros(n)
    for i in range(n):
        g = 0.0
        for s in range(-(n-1), n):
            w = p[s + n - 1] * dx
            if w < 1e-18:
                continue
            c = 0.0
            j1 = i - s
            if 0 <= j1 < n:
                c += 1.0 - h[j1]
            j2 = i + s
            if 0 <= j2 < n:
                c -= h[j2]
            g += w * c
        grad[i] = g + 2.0 * lam * (np.sum(h) - 0.5 * n)
```

```python
def optimize_one_coord(h, i, n):
    # f(v) = max_k c_k(h with h[i]=v) * dx
    # c_k(v) = c_k_base + (v - h[i]) * slope_k
    # where slope_k = (1-h[i-k])*I[valid] - h[i+k]*I[valid]
    dx = 2.0 / n
    h_orig = h[i]
    # Compute slopes and base values
    corr_base = get_corr(h) * dx  # current correlation values
    
    # For each lag s, c_s(v) = c_s_base + (v - h_orig) * slope_s
    slopes = np.zeros(2*n - 1)
    for s in range(-(n-1), n):
        sl = 0.0
        j1 = i - s
        if 0 <= j1 < n:
            sl += 1.0 - h[j1]
        j2 = i + s
        if 0 <= j2 < n:
            sl -= h[j2]
        slopes[s + n - 1] = sl
    
    def f(v):
        return np.max(corr_base + (v - h_orig) * slopes)
    
    # Ternary search on [0, 1]
    a, b = 0.0, 1.0
    for _ in range(50):
        m1 = a + (b-a)/3
        m2 = b - (b-a)/3
        if f(m1) < f(m2):
            b = m2
        else:
            a = m1
    v_opt = (a+b)/2
    return v_opt
```

```python
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def project_to_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = 0.5 * n
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() == 0:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_score(h, n):
    return float(np.max(get_corr(h)) * (2.0 / n))

def smooth_max_obj(h, n, beta, lam):
    dx = 2.0 / n
    c = get_corr(h) * dx
    m = c.max()
    val = m + np.log(np.sum(np.exp(beta * (c - m)))) / beta
    return val + lam * (h.sum() - 0.5 * n) ** 2

def smooth_max_grad(h, n, beta, lam):
    dx = 2.0 / n
    c = get_corr(h) * dx
    m = c.max()
    exps = np.exp(beta * (c - m))
    p = exps / exps.sum()
    grad = np.zeros(n)
    for i in range(n):
        g = 0.0
        for s in range(-(n-1), n):
            w = p[s + n - 1] * dx
            if w < 1e-18:
                continue
            ct = 0.0
            j1 = i - s
            if 0 <= j1 < n:
                ct += 1.0 - h[j1]
            j2 = i + s
            if 0 <= j2 < n:
                ct -= h[j2]
            g += w * ct
        grad[i] = g + 2.0 * lam * (h.sum() - 0.5 * n)
    return grad

def coord_optimize_coord(h, i, n):
    dx = 2.0 / n
    h_orig = h[i]
    corr_base = get_corr(h) * dx
    slopes = np.zeros(2 * n - 1)
    for s in range(-(n-1), n):

        j2 = i + s

            sl -= h[j2]
        slopes[s + n - 1] = sl

        m2 = b - (b - a) / 3.0
        if np.max(corr_base + (m1 - h_orig) * slopes) < np.max(corr_base + (m2 - h_orig) * slopes):
            b = m2
        else:
            a = m1
    return (a + b) / 2.0

            if abs(v_new - h[i]) > 1e-10:
                h[i] = v_new
                h = project_to_feasible(h, n)
                c = c5_score(h, n)
                if c < best - 1e-12:
                    best = c
                    improved += 1
        if improved == 0:
            break
    return h, best
```
