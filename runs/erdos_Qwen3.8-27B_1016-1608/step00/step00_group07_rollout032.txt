    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

    return np.correlate(h, 1.0 - h, mode="full") * dx

    corr = compute_corr_array(h, 2.0/n)
    m = corr.max()
    return m + np.log(np.exp(alpha*(corr-m)).sum()) / alpha
