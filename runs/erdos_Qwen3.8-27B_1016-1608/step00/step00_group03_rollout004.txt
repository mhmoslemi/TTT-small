import time

def projection(h, target):

    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        excess = h.sum() - target
        if abs(excess) < 1e-13:
            return h
        if excess > 0:
            mask = h > 0
            cnt = mask.sum()
            if cnt == 0:

            cnt = mask.sum()
            if cnt == 0:

            s = min(-excess / cnt, (1.0 - h[mask]).min())
            h[mask] += s

def try_slsqp(h0, dx, target, bounds, eq_con, maxiter=300, ftol=1e-14):
    try:
        res = minimize(compute_c5, h0, args=(dx,), method='SLSQP',
                      bounds=bounds, constraints=eq_con,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return projection(res.x, target)
    except Exception:
        return None

def try_smooth_slsqp(h0, dx, target, bounds, eq_con, temp, maxiter=300):

        res = minimize(smooth_c5, h0, args=(dx, temp), method='SLSQP',

        return projection(res.x, target)

        return None
