                    h_new3 = project_sum(np.clip(res3.x, 0, 1), target_sum)
                    c5_new3 = compute_c5(h_new3)
                    if c5_new3 < best_c5:
                        best_c5 = c5_new3
                        best_h = h_new3
                except Exception:
                    pass

    # Final: extra gradient refinement
    for _ in range(5):
        h_final = _grad_refine(best_h)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final

    best_h = project_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n)
