def _project_sum(x, target):
    """Adjust x to have sum=target while keeping in [0,1]"""
    x = np.clip(x, 0, 1)
    diff = target - x.sum()
    # Distribute diff proportionally to available room
    mask_up = x < 1.0
    mask_dn = x > 0.0
    if diff > 0 and mask_up.sum() > 0:
        room = (1.0 - x[mask_up]).sum()
        if room > 0:
            scale = min(1.0, diff / room)
            x[mask_up] += (1.0 - x[mask_up]) * scale
    elif diff < 0 and mask_dn.sum() > 0:
        room = x[mask_dn].sum()
        if room > 0:
            scale = min(1.0, -diff / room)
            x[mask_dn] -= x[mask_dn] * scale
    return x

    for r in range(30):
        if time.time() - t0 > budget_s - 50:
            break
        h_r = best_h + rng.normal(0, 0.05, n)
        h_r = _project(h_r, _TARGET_SUM)
        x = _lbfgsb_optimize(h_r, bounds, 800, 1e-16)
        if x is not None:
            x = _project(x, _TARGET_SUM)
            c5 = _c5_val(x)
            if c5 < best_c5:
                best_h = x.copy()
                best_c5 = c5

    # Phase 4: Final high-precision SLSQP polish
    if time.time() - t0 < budget_s - 5:
        x = _slsqp_optimize(best_h, bounds, cons, 5000, 1e-15)
        if x is not None:
            c5 = _c5_val(x)
            if c5 < best_c5:
                best_h = np.clip(x, 0.0, 1.0)
                best_c5 = c5

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project(best_h, _TARGET_SUM)
    best_c5 = _c5_val(best_h)
    return best_h, best_c5, n
