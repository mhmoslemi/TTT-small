import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def project_feasible(g, target):
    lo, hi = -2.0, 3.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        h = np.clip(g - mid, 0.0, 1.0)
        if h.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - 0.5*(lo+hi), 0.0, 1.0)

def smooth_obj(h, p, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    S = np.sum(corr**p)  # corr should be non-negative since h in [0,1]
    return S**(1.0/p) * dx

def smooth_grad(h, p, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    corr = np.maximum(corr, 0.0)  # should be non-negative
    S = np.sum(corr**p)
    if S < 1e-30:
        return np.zeros(n)
    factor = S**(1.0/p - 1.0) * dx
    w = corr**(p-1) * np.sign(corr)
    wf = w[::-1]
    oh = 1.0 - h
    c1 = np.convolve(w, oh, 'full')  # length n + 2n-1 - 1 = 3n-2
    c2 = np.convolve(h, wf, 'full')
    grad = c1[n-1:n-1+n] - c2[n-1:n-1+n]
    grad *= factor
    return grad

def pgd(h0, p, dx, target, lr=0.01, max_iter=500, tol=1e-8):
    h = h0.copy().astype(float)
    h = project_feasible(h, target)
    best_h = h.copy()
    best_obj = compute_c5(h, dx)
    obj = smooth_obj(h, p, dx)
    for it in range(max_iter):
        g = smooth_grad(h, p, dx)
        h_new = project_feasible(h - lr * g, target)
        obj_new = smooth_obj(h_new, p, dx)
        if obj_new < obj:
            h = h_new
            obj = obj_new
            c5 = compute_c5(h, dx)
            if c5 < best_obj:
                best_obj = c5
                best_h = h.copy()
        else:
            lr *= 0.5
            if lr < 1e-10:
                break
    return best_h, best_obj

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h.astype(float), 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def project_feasible(g, target):
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        h = np.clip(g - mid, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - 0.5*(lo+hi), 0.0, 1.0)

def smooth_obj_grad(h, p, dx):
    n = len(h)
    oh = 1.0 - h
    corr = np.correlate(h, oh, mode="full")
    corr = np.maximum(corr, 1e-15)
    corr_p = corr ** p
    S = np.sum(corr_p)
    obj = S ** (1.0/p) * dx
    factor = S ** (1.0/p - 1.0) * dx
    w_arr = corr ** (p-1)
    c1 = np.convolve(oh, w_arr, 'full')
    c2 = np.correlate(h, w_arr, 'full')
    grad = (c1[n-1:2*n-1] - c2[2*n-2:3*n-2][::-1]) * factor
    return obj, grad

def pgd_optimize(h0, target, p_list, total_iters=2000, dx=None):
    n = len(h0)
    if dx is None:
        dx = 2.0 / n
    h = h0.copy().astype(float)
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = 0.1
    for p in p_list:
        for it in range(total_iters // len(p_list)):
            obj, g = smooth_obj_grad(h, p, dx)
            h_new = project_feasible(h - lr * g, target)
            obj_new, _ = smooth_obj_grad(h_new, p, dx)
            if obj_new < obj:
                h = h_new
                obj = obj_new
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                lr = min(lr * 1.05, 0.5)
            else:
                lr *= 0.5
                if lr < 1e-8:
                    break
    return best_h, best_c5

def binary_local_search(h0, target, max_iters=5000, time_limit=60):
    n = len(h0)
    dx = 2.0 / n
    h = np.zeros(n)
    # Make binary: threshold at 0.5, then adjust to get exact target
    h = (h0 > 0.5).astype(float)
    # Adjust: if sum > target, flip some 1s to 0s; if sum < target, flip some 0s to 1s
    while h.sum() > target:
        idx = np.where(h == 1)[0]
        # Flip the one that most improves
        best_i = idx[0]
        best_c5 = compute_c5(h)
        for i in idx[:20]:  # Limit for speed
            h[i] = 0
            c5 = compute_c5(h)
            h[i] = 1
            if c5 < best_c5:
                best_c5 = c5
                best_i = i
        h[best_i] = 0
    while h.sum() < target:
        idx = np.where(h == 0)[0]
        best_i = idx[0]
        best_c5 = compute_c5(h)
        for i in idx[:20]:
            h[i] = 1
            c5 = compute_c5(h)
            h[i] = 0
            if c5 < best_c5:
                best_c5 = c5
                best_i = i
        h[best_i] = 1
    
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    start = time.time()
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        # Pick random 0 and 1 to swap
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        i = zeros[np.random.randint(len(zeros))]
        j = ones[np.random.randint(len(ones))]

            best_c5 = c5
            best_h = h.copy()
        else:
            h[i] = 0
            h[j] = 1
    return best_h, best_c5

import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h.astype(float), 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def project_feasible(g, target):
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        h = np.clip(g - mid, 0.0, 1.0)
        if h.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - 0.5 * (lo + hi), 0.0, 1.0)

def smooth_obj_grad(h, p, dx):
    n = len(h)
    oh = 1.0 - h
    corr = np.correlate(h, oh, mode="full")
    corr = np.maximum(corr, 1e-15)
    S = np.sum(corr ** p)
    obj = S ** (1.0 / p) * dx
    factor = S ** (1.0 / p - 1.0) * dx
    w_arr = corr ** (p - 1)
    c1 = np.convolve(oh, w_arr, 'full')
    c2 = np.correlate(h, w_arr, 'full')
