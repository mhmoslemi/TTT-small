        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h


def _make_perturbation(h_base, magnitude, rng):
    h = h_base + rng.normal(0, magnitude, len(h_base))
    h = np.clip(h, 0.0, 1.0)
    h = _repair_sum(h, 23.5)
    return h


def _slsqp_optimize(h0, t, maxiter):
    bounds = [(0.0, 1.0)] * len(h0)
    cons = [{'type': 'eq', 'fun': _sum_constraint}]
    try:
        res = minimize(
            _soft_obj, h0, args=(t,),
            jac=_soft_grad, method='SLSQP',
            bounds=bounds, constraints=cons,
            options={'maxiter': maxiter, 'ftol': 1e-16}
        )
        return res.x
    except Exception:
        return h0


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.default_rng(seed)

    n_pts = 47
    target_sum = n_pts / 2.0

    # Initialize from parent construction if available
    try:
        h_base = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_base = np.ones(n_pts) * 0.5

    best_h = h_base.copy()
    best_c5 = _compute_c5(best_h)

    # Phase 1: Multi-start SLSQP with moderate smoothing
    temperatures = [20.0, 40.0, 80.0, 150.0, 300.0]
    perturb_mags = [0.0, 0.05, 0.1, 0.2, 0.3, 0.5]

    attempts = 0
    for t in temperatures:
        for mag in perturb_mags:
            if time.time() - t_start > budget_s - 60:
                break
            h0 = h_base if mag == 0.0 else _make_perturbation(h_base, mag, rng)
            attempts += 1
            h_new = _slsqp_optimize(h0, t, 150)
            h_new = _repair_sum(h_new, target_sum)
            c5_new = _compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

        # Also try random starts
        for _ in range(3):
            if time.time() - t_start > budget_s - 60:
                break
            h0 = rng.uniform(0, 1, n_pts)
            h0 = _repair_sum(h0, target_sum)
            h_new = _slsqp_optimize(h0, t, 100)
            h_new = _repair_sum(h_new, target_sum)
            c5_new = _compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 2: Polish with high temperature from best solution
    if time.time() - t_start < budget_s - 30:
        for t_polish in [200.0, 400.0, 800.0]:
            if time.time() - t_start > budget_s - 15:
                break
            h_new = _slsqp_optimize(best_h, t_polish, 200)
            h_new = _repair_sum(h_new, target_sum)
            c5_new = _compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 3: Try slightly different n_points
    if time.time() - t_start < budget_s - 20:
        for n_alt in [49, 51, 45]:
            if time.time() - t_start > budget_s - 10:
                break
            target_alt = n_alt / 2.0
            # Interpolate best_h to n_alt points
            x_old = np.linspace(0, 2, n_pts, endpoint=False)
            x_new = np.linspace(0, 2, n_alt, endpoint=False)
            h_alt = np.interp(x_new, x_old, best_h, period=2.0)
            h_alt = _repair_sum(h_alt, target_alt)
            c5_alt = _compute_c5(h_alt)
            for t_alt in [50.0, 150.0]:
                if time.time() - t_start > budget_s - 5:
                    break
                h_new = _slsqp_optimize(h_alt, t_alt, 120)
                h_new = _repair_sum(h_new, target_alt)
                c5_new = _compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    n_pts = n_alt

    # Final verification
    best_h = _repair_sum(best_h, target_sum)
    best_c5 = _compute_c5(best_h)

    return best_h, best_c5, n_pts
