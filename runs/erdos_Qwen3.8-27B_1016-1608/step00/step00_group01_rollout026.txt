    lo, hi = -1.0, 2.0

        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(corr)
    # Subgradient: gradient of the max component
    # For simplicity, use the gradient of the argmax component
    k_star = np.argmax(corr)
    
    # Or use weighted average of all max components
    mask = corr >= max_val - 1e-10
    # For each max component, compute its gradient
    
    grad = np.zeros(n)
    for k in np.where(mask)[0]:
        L = max(0, k - n + 1)
        R = min(n, k)
        # indices i from L to R-1
        for i in range(L, R):
            m = i - k + n - 1  # the other index
            grad[i] += (1.0 - h[m])
            grad[m] += -h[i]
    
    # Average over max components
    n_max = np.sum(mask)
    grad /= n_max
    grad *= dx
    return grad

One more thought: the `grad_c5` function with the Python loop over k and i might be slow for large n. Let me vectorize it:

For each k, the valid i range is [max(0, k-n+1), min(n-1, k-1)].
The contribution to grad[i] is (1-h[i-k+n-1]) for i in that range.
The contribution to grad[i-k+n-1] is -h[i] for i in that range.

I can vectorize by creating the index arrays:
```python
def grad_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(corr)
    mask = corr >= max_val - 1e-8
    n_max = np.sum(mask)
    
    grad = np.zeros(n)
    for k in np.where(mask)[0]:
        L = max(0, k - n + 1)
        R = min(n, k)
        if R <= L:
            continue
        i = np.arange(L, R)
        m = i - k + n - 1
        grad[i] += (1.0 - h[m])
        grad[m] -= h[i]
    
    grad /= n_max
    grad *= dx
    return grad
```

This is better - the inner loop is vectorized with NumPy. The outer loop over k (which could be up to 2n-1 iterations) is still Python, but each iteration is fast.

Actually, for the max component, typically only 1 or a few k values are near the max. So the outer loop usually runs just 1-3 times.

import numpy as np
import time

def project_capped(h, target_sum):
    h = h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(64):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def grad_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(corr)
    mask = corr >= max_val - 1e-8
    n_max = max(np.sum(mask), 1)
    grad = np.zeros(n)
    for k in np.where(mask)[0]:
        L = max(0, k - n + 1)
        R = min(n, k)
        if R <= L:
            continue
        i_idx = np.arange(L, R)
        m_idx = i_idx - k + n - 1
        if 0 <= m_idx.min() and m_idx.max() < n:
            grad[i_idx] += (1.0 - h[m_idx])
            np.add.at(grad, m_idx, -h[i_idx])
    grad /= n_max
    grad *= dx
    return grad

def pgd_optimize(h_init, target_sum, n_iter, deadline):
    n = len(h_init)
    h = project_capped(h_init, target_sum)
    best = h.copy()
    best_val = eval_c5(h)
    lr = 0.008
    for it in range(n_iter):
        if time.time() > deadline:
            break
        g = grad_c5(h)
        h_try = project_capped(h - lr * g, target_sum)
        val = eval_c5(h_try)
        if val < best_val:
            best = h_try.copy()
            best_val = val
            h = h_try
            lr = min(lr * 1.03, 0.05)
        else:
            lr *= 0.85
            if lr < 1e-8:
                lr = 0.005
                h = project_capped(h + np.random.randn(n) * 0.08, target_sum)
    return best, best_val

def local_search(h_init, n_perturb, sigma_init, target_sum, deadline):
    n = len(h_init)
    h = h_init.copy()
    best = h.copy()
    best_val = eval_c5(h)
    sigma = sigma_init
    for i in range(n_perturb):
        if time.time() > deadline:
            break
        h_try = project_capped(h + np.random.randn(n) * sigma, target_sum)
        val = eval_c5(h_try)
        if val < best_val:
            best = h_try.copy()
            best_val = val
            h = h_try
            sigma = min(sigma * 1.01, 0.4)
        else:
            sigma = max(sigma * 0.985, 0.002)
    return best, best_val

def coord_descent(h_init, target_sum, n_rounds, deadline):
    n = len(h_init)
    h = h_init.copy()
    f = eval_c5(h)
    deltas = np.array([-0.5, -0.3, -0.15, -0.08, -0.03, 0.03, 0.08, 0.15, 0.3, 0.5])
    for r in range(n_rounds):
        if time.time() > deadline:
            break
        for i in range(n):
            h0 = h[i]
            best_hi = h0
            best_f = f
            for dv in deltas:
                h[i] = np.clip(h0 + dv, 0, 1)
                fv = eval_c5(h)
                if fv < best_f:
                    best_f = fv
                    best_hi = h[i]
            h[i] = best_hi
            f = best_f
        h = project_capped(h, target_sum)
        f = eval_c5(h)
    return h, f

def make_sine_init(n, freq):
    x = np.arange(n)
    h = 0.5 + 0.45 * np.cos(2.0 * np.pi * freq * x / n)
    return h
