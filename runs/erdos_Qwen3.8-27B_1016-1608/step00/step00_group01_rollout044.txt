def _grad_step(h, n, alpha, steps=2000):
    h = h.copy()
    c5 = _c5_eval(h)
    for s in range(steps):
        eps = 1e-7
        base = c5
        g = np.zeros(n)
        for i in range(n):
            h2 = h.copy()
            h2[i] += eps
            g[i] = (_c5_eval(h2) - base) / eps
        g -= np.mean(g)
        h_new = _project(h - alpha * g, n)
        c5_new = _c5_eval(h_new)
        if c5_new < c5 - 1e-15:
            h, c5 = h_new, c5_new
            alpha = min(alpha * 1.15, 2.0)
        else:
            alpha *= 0.5
            if alpha < 1e-12:
                break
    return h, c5

def _random_feasible(n, rng):
    h = rng.beta(2.5, 2.5, size=n)
    return _project(h, n)

def _perturb(h, n, scale, rng):
    h2 = h + scale * rng.randn(n)
    return _project(h2, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    def tleft():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.ones(47) * 0.5
    best_n = len(best_h)
    best_c5 = _c5_eval(best_h)

    # Phase 1: NM + gradient from initial
    if tleft() > 5:
        best_h, best_c5 = _nm_local(best_h, best_n, 8000)
    if tleft() > 5:
        h, c5 = _grad_step(best_h, best_n, 0.05, 2000)
        if c5 < best_c5:
            best_h, best_c5 = h, c5
    if tleft() > 5:
        best_h, best_c5 = _nm_local(best_h, best_n, 8000)

    # Phase 2: Perturbation restarts
    for trial in range(40):
        if tleft() < 3:
            break
        scale = 0.02 * (1.0 + 0.5 * rng.rand())
        h_init = _perturb(best_h, best_n, scale, rng)
        h, c5 = _nm_local(h_init, best_n, 3000)
        if c5 < best_c5:
            best_h, best_c5 = h, c5
            if trial % 10 == 5:
                h, c5 = _grad_step(h, best_n, 0.03, 500)
                if c5 < best_c5:
                    best_h, best_c5 = h, c5

    # Phase 3: Random restarts (different basins)
    for trial in range(15):
        if tleft() < 5:
            break
        h_rand = _random_feasible(best_n, rng)
        c5r = _c5_eval(h_rand)
        # Quick NM to see if promising
        h, c5 = _nm_local(h_rand, best_n, 2000)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 4: Try other n_points
    for n_try in [55, 64, 80, 96, 120, 150]:
        if tleft() < 8:
            break
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = _project(h_interp, n_try)
        c5i = _c5_eval(h_interp)
        h, c5 = _nm_local(h_interp, n_try, 5000)
        if c5 < best_c5:
            best_h, best_c5, best_n = h, c5, n_try

    # Phase 5: Final polish
    if tleft() > 5:
        best_h, best_c5 = _nm_local(best_h, best_n, 10000)
    if tleft() > 5:
        h, c5 = _grad_step(best_h, best_n, 0.02, 1000)
        if c5 < best_c5:
            best_h, best_c5 = h, c5
        h, c5 = _nm_local(best_h, best_n, 5000)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Ensure feasibility
    best_h = _project(best_h, best_n)
    final_c5 = _c5_eval(best_h)
    return best_h, final_c5, best_n
