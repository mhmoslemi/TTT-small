        n = 70
        best_h = repair(np.full(n, 0.5) + rng.randn(n) * 0.2, n)
        best_c5 = compute_c5(best_h)

    n_par = len(best_h)
    dl1 = deadline - 40.0

    # Phase 1: exploration at parent resolution
    h1, c5_1 = sa_optimize(best_h, n_par, rng, dl1, 0.008, 0.0008, 0.04, 40000)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1

    # Phase 2: refinement
    dl2 = deadline - 40.0
    h2, c5_2 = sa_optimize(best_h, n_par, rng, dl2, 0.0008, 0.00005, 0.006, 40000)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2

    # Phase 3: ultra-fine
    h3, c5_3 = sa_optimize(best_h, n_par, rng, deadline - 30.0, 0.0001, 0.00001, 0.001, 20000)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3

    # Try other resolutions
    for n_try in [70, 100, 140, 200]:
        if n_try == n_par or time.time() > deadline - 50:
            continue
        t_old = np.linspace(0, 1, len(best_h))
        t_new = np.linspace(0, 1, n_try)
        h_try = np.interp(t_new, t_old, best_h)
        h_try = repair(h_try, n_try)
        h_opt, c5_opt = sa_optimize(h_try, n_try, rng, deadline - 20.0, 0.003, 0.0002, 0.015, 30000)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Random restart (different starting point)
    if time.time() < deadline - 40:
        n = n_par
        for restart in range(3):
            if time.time() > deadline - 30:
                break
            h_rand = np.full(n, 0.5) + rng.randn(n) * 0.3
            h_rand = repair(h_rand, n)
            h_rs, c5_rs = sa_optimize(h_rand, n, rng, deadline - 10.0, 0.006, 0.0004, 0.03, 20000)
            if c5_rs < best_c5:
                best_c5 = c5_rs
                best_h = h_rs

    # Nelder-Mead final polish
    if time.time() < deadline - 3:
        n = len(best_h)
        try:
            from scipy.optimize import minimize
            res = minimize(_obj_wrapper, best_h, args=(n,), method='Nelder-Mead',
                          options={'maxiter': 4000, 'xatol': 1e-13, 'fatol': 1e-15, 'adaptive': True})
            h_opt = repair(res.x, n)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
        except Exception:
            pass

    best_h = repair(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
