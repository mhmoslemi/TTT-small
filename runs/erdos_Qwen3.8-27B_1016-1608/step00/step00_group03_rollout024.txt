from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    return np.max(corr) * dx

def make_binary(h):
    """Round to binary and fix sum constraint."""
    n = len(h)
    target = n // 2
    h_bin = (h > 0.5).astype(float)
    # Fix sum
    while h_bin.sum() > target:
        # Find the one with highest value to turn off
        idx = np.where(h_bin == 1)[0]
        h_bin[idx[np.argmax(h[idx])]] = 0
    while h_bin.sum() < target:
        idx = np.where(h_bin == 0)[0]
        h_bin[idx[np.argmax(1-h[idx])]] = 1
    return h_bin

    n = len(h)
    target = n // 2
    h = h.copy()
    # Ensure binary
    h = (h >= 0.5).astype(float)
    if h.sum() != target:
        # Fix
        sorted_idx = np.argsort(h)[::-1]
        for i in range(len(sorted_idx)):
            if h.sum() == target:
                break
            if h.sum() > target:
                h[sorted_idx[i]] = 0
            else:
                h[sorted_idx[i]] = 1
    
    current_c5 = compute_c5(h)
    start = time.time()
    
    while time.time() - start < time_limit:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]

            break

        for i in zeros:

            for j in ones:
                h[j] = 0
                new_c5 = compute_c5(h)
                delta = new_c5 - current_c5
                if delta < best_delta:
                    best_delta = delta
                    best_pair = (i, j)
                h[j] = 1
            h[i] = 0
        
        if best_pair is not None:

        else:

        elapsed = time.time() - start
        T = T_init * (T_min / T_init) ** (elapsed / time_limit)

    n = len(h)

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def make_binary(h):
    n = len(h)
    target = n // 2
    sorted_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:target]] = 1.0
    return h_bin

def local_search(h_bin, time_limit=200):
    n = len(h_bin)
    h = h_bin.astype(float).copy()
    current_c5 = compute_c5(h)
    start = time.time()
    while time.time() - start < time_limit:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_delta = -1e-15
        best_pair = None
        for i in zeros:
            h[i] = 1.0
            for j in ones:
                h[j] = 0.0
                new_c5 = compute_c5(h)
                delta = new_c5 - current_c5
                if delta < best_delta:
                    best_delta = delta
                    best_pair = (i, j)
                h[j] = 1.0
            h[i] = 0.0
        if best_pair is not None:
            i, j = best_pair
            h[i] = 1.0
            h[j] = 0.0
            current_c5 += best_delta
        else:
            break
    return h, current_c5

def sa_search(h_bin, time_limit=200, rng=None):
    if rng is None:
        rng = np.random.default_rng()
    n = len(h_bin)
    h = h_bin.astype(float).copy()
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    start = time.time()
    T_init = 0.08
    T_min = 5e-6
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        T = T_init * (T_min / T_init) ** (elapsed / time_limit)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = zeros[rng.integers(len(zeros))]
        j = ones[rng.integers(len(ones))]
        h[i] = 1.0
        h[j] = 0.0
        new_c5 = compute_c5(h)
        delta = new_c5 - current_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            h[i] = 0.0
            h[j] = 1.0
    return best_h, best_c5

def multi_start_local(h0, n_iter=5, time_per=30, rng=None):
    if rng is None:
        rng = np.random.default_rng()
    n = len(h0)
    target = n // 2
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    for it in range(n_iter):
        # Random perturbation: flip k random pairs
        h_pert = h0.copy()
        k = rng.integers(3, max(4, n // 10))
        ones = np.where(h_pert == 1)[0]
        zeros = np.where(h_pert == 0)[0]
        for _ in range(k):
            if len(ones) == 0 or len(zeros) == 0:
                break
            a = zeros[rng.integers(len(zeros))]
            b = ones[rng.integers(len(ones))]
            h_pert[a] = 1.0
            h_pert[b] = 0.0
            ones = np.where(h_pert == 1)[0]
            zeros = np.where(h_pert == 0)[0]
        h_res, c5_res = local_search(h_pert, time_limit=time_per)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res
    return best_h, best_c5

def continuous_refine(h_bin, max_iter=80):
    n = len(h_bin)
    target = n / 2.0
    x0 = h_bin.astype(float).copy()
    
    def obj(x):
        hc = np.clip(x, 0, 1)
        return compute_c5(hc)
    
    def con_sum(x):
        return x.sum() - target
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum, 'jac': lambda x: np.ones(n)}]
    result = minimize(obj, x0, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': max_iter, 'ftol': 1e-12})
    if result.fun < compute_c5(h_bin):
        return np.clip(result.x, 0, 1), float(result.fun)
    return h_bin, compute_c5(h_bin)

    if c5_4 < best_c5:

    # Phase 5: Try different n values
    for n_try in [50, 60, 80, 90, 100]:
        t_rem = budget_s - (time.time() - t0)

    t_rem = budget_s - (time.time() - t0)

    final_c5 = compute_c5(best_h)
    
    return best_h, final_c5, len(best_h)
