    s = np.arange(-(n-1), n)  # 2n-1 values
    grad = np.zeros(n)
    comp = 1.0 - h
    
    for j in range(n):

        valid1 = (j + s >= 0) & (j + s < n)
        if np.any(valid1):

        valid2 = (j - s >= 0) & (j - s < n)
        if np.any(valid2):
            term2 = np.sum(weights[valid2] * h[j - s[valid2]])
        else:
            term2 = 0.0
        grad[j] = dx * (term1 - term2)
    
    return grad

    t0 = time.time()
    h = h_init.copy()
    n = len(h)
    dx = 2.0 / n
    target_sum = n / 2.0
    
    # Adaptive smoothing parameter
    alpha = 10.0  # starts moderate
    alpha_decay = 0.999  # decrease over time for sharper max approximation
    step_size = 0.05  # initial step
    
    best_c5 = _c5_value(h)
    best_h = h.copy()
    
    for it in range(n_iterations):

            break

        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        max_corr = np.max(corr)
        
        # Log-sum-exp weights
        shifted = alpha * (corr - max_corr)
        exp_shifted = np.exp(shifted)
        weights = exp_shifted / np.sum(exp_shifted)
        
        # Gradient

        # Gradient step
        h_new = h - step_size * grad
        
        # Project
        h_new = _project(h_new, target_sum)
        
        # Check improvement
        c5 = _c5_value(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        
        h = h_new
        
        # Adaptive step size / smoothing
        alpha *= alpha_decay
        step_size *= 0.999
    
    return best_h, best_c5

```python
def _grad_smoothed(h, weights, dx):
    """
    h: array of length n
    weights: array of length 2n-1 (same order as full correlation)
    dx: 2/n
    Returns gradient of smoothed objective w.r.t. h
    """
    n = len(h)
    s_vals = np.arange(-(n-1), n)  # 2n-1 values
    comp = 1.0 - h
    grad = np.zeros(n)
    
    for j in range(n):
        # valid indices for term1: 0 <= j+s < n
        t1_mask = (j + s_vals >= 0) & (j + s_vals < n)
        if np.any(t1_mask):
            idx = j + s_vals[t1_mask]
            w = weights[t1_mask]
            term1 = np.dot(w, comp[idx])
        else:
            term1 = 0.0
        
        # valid indices for term2: 0 <= j-s < n
        t2_mask = (j - s_vals >= 0) & (j - s_vals < n)
        if np.any(t2_mask):
            idx = j - s_vals[t2_mask]
            w = weights[t2_mask]
            term2 = np.dot(w, h[idx])
        else:
            term2 = 0.0
        
        grad[j] = dx * (term1 - term2)
    
    return grad
```

    diff = target_sum - np.sum(g)
    if abs(diff) > 1e-10:

        unsaturated = (g > 1e-10) & (g < 1 - 1e-10)
        if np.any(unsaturated):
            g[unsaturated] += diff / np.sum(unsaturated)
            g = np.clip(g, 0.0, 1.0)
    return g

        candidates.append((h_p, n_init))
    
    # Time allocation
    n_cand = len(candidates)
    time_per = (budget_s - 60) / max(n_cand, 1)
    time_per = max(time_per, 10)  # minimum 10s per candidate
    
    for idx, (h_c, n_c) in enumerate(candidates):
        if time.time() - t_global > budget_s - 30:
            break
        alloc = min(time_per, budget_s - 30 - (time.time() - t_global))

        h_opt, c5_opt = _optimize(h_c, alloc, rng)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_c
    
    # Final verification
    best_c5 = _c5(best_h)
    return best_h, best_c5, best_n
