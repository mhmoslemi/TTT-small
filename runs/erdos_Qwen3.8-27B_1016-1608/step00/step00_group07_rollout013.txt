def _eval(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")))

def _fix(h, tgt):
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    d = tgt - h.sum()
    if abs(d) > 1e-9:
        h += d / n
    return np.clip(h, 0.0, 1.0)

def _sa_loop(h, n, tgt, T, n_iter, rng, bh, bv):

        j = rng.integers(n)

            continue
