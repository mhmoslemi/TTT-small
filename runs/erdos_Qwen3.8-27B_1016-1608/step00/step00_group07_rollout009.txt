import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def grad_for_lag(h, k, n):
    g = np.zeros(n)
    if k >= 0:
        g[k:] += 1.0 - h[:n - k]
        g[:n - k] -= h[k:]
    else:
        kk = -k
        g[:n - kk] += 1.0 - h[kk:]
        g[kk:] -= h[:n - kk]
    return g


def obj_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    mv = np.max(corr)
    active = np.where(corr >= mv - 1e-10)[0]
    grad = np.zeros(n)
    for idx in active:
        k = idx - (n - 1)
        grad += grad_for_lag(h, k, n)
    return mv, grad * dx / len(active)


def eq_constraint(h):
    return np.sum(h) - len(h) / 2.0


def eq_jac(h):
    return np.ones(len(h))


def project_feasible(h, s_target):
    h = np.clip(h, 0.0, 1.0)
    diff = s_target - np.sum(h)
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h


def sgd_optimize(h_init, n_steps, alpha0, alpha_min, deadline):
    h = h_init.copy()
    s_t = len(h) / 2.0
    alpha = alpha0
    best_c = compute_c5(h)
    best_h = h.copy()
    for s in range(n_steps):
        if time.time() > deadline:
            break
        obj, grad = obj_grad(h)
        h = h - alpha * grad
        h = project_feasible(h, s_t)
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        alpha = max(alpha_min, alpha * 0.995)
    return best_h, best_c


def local_search(h_init, deadline, rng):
    h = h_init.copy()
    n = len(h)

    best_h = h.copy()
    best_c = c5_curr
    sigma = 0.02

        if time.time() > deadline:
            break
        i, j = rng.integers(0, n, 2)
        if i == j:
            continue
        d = rng.normal(0, sigma)
        h[i] += d
        h[j] -= d
        h = np.clip(h, 0.0, 1.0)
        diff = s_t - np.sum(h)
        if abs(diff) > 1e-12:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
        c = compute_c5(h)
        if c < c5_curr:
            c5_curr = c
            if c < best_c:
                best_c = c
                best_h = h.copy()
        else:
            h[i] -= d
            h[j] += d
        if it % 3000 == 2999:
            sigma = max(0.003, sigma * 0.88)
    return best_h, best_c
