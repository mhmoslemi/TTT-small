Actually, for a convex PL function, I can use binary search on the subgradient:
- f(t) = max_m (a_m + b_m * t) is convex in t
- f'(t) = b_m* where m* is the argmax
- At the minimum, 0 should be in the subgradient, i.e., there exist m1, m2 with b_m1 < 0 < b_m2 and both achieving the max

So I can find the minimum by:
1. Evaluate f at t=0 and t=1, check subgradients
2. If f'(0+) > 0, minimum is at t=0
3. If f'(1-) < 0, minimum is at t=1
4. Otherwise, binary search for where the active constraint's slope crosses 0

This is O(n log n) per coordinate (compute f and argmax in O(n), do O(log(1/ε)) bisection steps).

Let me implement this:

```python
def find_min_of_max(a, b, tol=1e-10):
    """Find t in [0,1] minimizing max(a + b*t)"""
    n = len(a)
    
    def f_and_argmax(t):
        vals = a + b * t
        idx = np.argmax(vals)
        return vals[idx], idx
    
    # Check endpoints
    f0, _ = f_and_argmax(0.0)
    f1, _ = f_and_argmax(1.0)
    
    # Subgradient at 0
    vals_0 = a  # a + b*0
    max_0 = np.max(vals_0)
    active_0 = vals_0 >= max_0 - tol
    if np.min(b[active_0]) > tol:
        return 0.0  # all active slopes positive, f increasing at 0
    if np.max(b[active_0]) < -tol:
        pass  # f decreasing at 0, minimum is interior or at 1
    
    # Subgradient at 1
    vals_1 = a + b
    max_1 = np.max(vals_1)
    active_1 = vals_1 >= max_1 - tol
    if np.max(b[active_1]) < -tol:
        return 1.0  # all active slopes negative, f decreasing at 1
    if np.min(b[active_1]) > tol:
        pass  # f increasing at 1, minimum is interior or at 0
    
    # Binary search
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        vals = a + b * mid
        max_v = np.max(vals)
        active = vals >= max_v - tol
        min_slope = np.min(b[active])
        max_slope = np.max(b[active])
        if min_slope > -tol and max_slope < tol:
            break
        if min_slope > tol:
            hi = mid  # f is increasing at mid, min is to the left
        elif max_slope < -tol:
            lo = mid  # f is decreasing at mid, min is to the right
        else:
            break  # 0 is in subgradient, we found it
    
    t = (lo + hi) / 2
    # Evaluate both t and endpoints
    best_t = 0.0
    best_val = np.inf
    for candidate in [0.0, 1.0, t, lo, hi]:
        candidate = np.clip(candidate, 0.0, 1.0)
        v = np.max(a + b * candidate)
        if v < best_val:
            best_val = v
            best_t = candidate
    return best_t
```

This is O(n × log(1/ε)) per coordinate, so O(n² × log(1/ε)) per sweep. Much faster!

    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def find_min_of_max_feval(a, b):
    """Find t in [0,1] minimizing max_m(a[m] + b[m]*t)"""
    n = len(a)
    tol = 1e-12

    v0 = float(np.max(a))
    v1 = float(np.max(a + b))

def cd_sweep(h, n_sweeps=50):
    """Cyclic coordinate descent to minimize max overlap"""
    n = len(h)
    lags = np.arange(-(n-1), n)  # length 2n-1
    h = h.copy()
    
    for sweep in range(n_sweeps):
        for j in range(n):
            # Compute b (coefficient of h[j] in each corr value)
            b = (1.0 - 0.0)  # start with 1
            idx_p = j + lags  # positions j+lag
            b = np.ones(2*n-1)
            valid_p = (idx_p >= 0) & (idx_p < n)
            b[valid_p] -= h[idx_p[valid_p]]
            idx_m = j - lags
            valid_m = (idx_m >= 0) & (idx_m < n)
            b[valid_m] -= h[idx_m[valid_m]]
            
            # Current corr values
            corr = np.correlate(h, 1.0 - h, mode='full')
            a = corr - h[j] * b
            
            # Find optimal t
            t_opt = find_min_of_max_feval(a, b)
            h[j] = t_opt
        
        # Project onto sum constraint
        target = n / 2.0
        diff = target - np.sum(h)
        if abs(diff) > 1e-8:
            # Distribute to interior points
            mask = (h > 0.01) & (h < 0.99)
            if np.sum(mask) > 0:
                h[mask] += diff / np.sum(mask)
            else:
                h += diff / n
            h = np.clip(h, 0.0, 1.0)
    
    return h

    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    return m + np.log(np.sum(np.exp((corr - m) * T))) / T
