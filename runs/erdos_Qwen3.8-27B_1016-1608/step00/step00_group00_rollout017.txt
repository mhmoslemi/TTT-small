def subgradient_c5(h):

def subgradient_c5(h, top_k=5):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(corr)

    if top_k <= len(corr):

        top_idx = np.arange(len(corr))
    # Weight by closeness to max

        for _ in range(2):
            if time_up(0.65):
                break
            ho, v = subgradient_descent(best_h, 5000, 0.01, 0.9, rng)
            if v < best_c5:
                best_c5, best_h = v, ho.copy()
            ho, v = local_search(best_h, 15000, rng)
            if v < best_c5:
                best_c5, best_h = v, ho.copy()

    # Phase 5: Try slightly higher resolution
    n_best = len(best_h)
    for dn in [5, 10, 15]:
        if time_up(0.85):
            break
        n_try = n_best + dn
        x_old = np.arange(n_best) * (2.0 / n_best)
        x_new = np.arange(n_try) * (2.0 / n_try)
        h_try = np.interp(x_new, x_old, best_h)
        h_try = project_feasible(h_try, n_try / 2.0)
        ho, v = subgradient_descent(h_try, 2000, 0.02, 0.85, rng)
        if v < best_c5:
            best_c5, best_h = v, ho.copy()
        if not time_up(0.85):
            ho, v = local_search(best_h, 10000, rng)
            if v < best_c5:
                best_c5, best_h = v, ho.copy()

    # Phase 6: Final polish
    if not time_up(0.97):
        ho, v = local_search(best_h, 20000, rng)
        if v < best_c5:
            best_c5, best_h = v, ho.copy()

    # Ensure constraint satisfaction
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n_final
