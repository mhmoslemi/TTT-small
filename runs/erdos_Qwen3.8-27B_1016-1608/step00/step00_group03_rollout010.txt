import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        s = np.sum(room)
        if s > 1e-15:
            h = h + diff * room / s
    else:
        room = h.copy()
        s = np.sum(room)
        if s > 1e-15:
            h = h + diff * room / s
    return np.clip(h, 0.0, 1.0)

def corr_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_obj(h, target_sum, alpha):
    hv = make_feasible(h, target_sum)
    c = corr_values(hv)
    m = np.max(c)
    a = alpha
    if a * m > 500:
        return m
    s = np.sum(np.exp(a * (c - m)))
    return m + np.log(s) / (a * len(c))

                    val = h_try[j] + delta
                    val = np.clip(val, 0.0, 1.0)
                    actual_delta = val - h_try[j]
                    h_try[j] = val
                    h_try[j2] = np.clip(h_try[j2] - actual_delta, 0.0, 1.0)
                    h_try = make_feasible(h_try, target_sum)
                    f_try = compute_c5(h_try)
                    if f_try < f_cd - 1e-12:
                        h_cd = h_try
                        f_cd = f_try
                        improved = True
                        break
            if not improved:
                break
        if f_cd < best_c5:
            best_c5 = f_cd
            best_h = h_cd.copy()

    best_h = make_feasible(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
