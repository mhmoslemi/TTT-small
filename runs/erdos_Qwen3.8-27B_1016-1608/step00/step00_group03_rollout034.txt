    lo, hi = -3.0, 3.0

    """Return h shifted: out[i] = h[i+s] if valid, else 0"""
    n = len(h)
    out = np.zeros(n)
    if s >= 0:
        out[:n-s] = h[s:]
    else:

    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")

    p = np.exp(beta * shifted_corr)

        grad += w * contrib

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_polytope(h, target_sum):
    h = np.array(h, dtype=float)
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def binarize(h):
    n = len(h)
    order = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[order[:n // 2]] = 1.0
    return h_bin

def local_search_binary(h_bin, rng, t_end, T_init=0.01):
    n = len(h_bin)
    h_curr = h_bin.copy()
    c5_curr = compute_c5(h_curr)
    best_local = h_curr.copy()
    best_c5 = c5_curr
    T = T_init
    no_improve = 0
    while time.time() < t_end:
        ones_idx = np.where(h_curr == 1.0)[0]
        zeros_idx = np.where(h_curr == 0.0)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i = ones_idx[rng.randint(len(ones_idx))]
        j = zeros_idx[rng.randint(len(zeros_idx))]
        h_new = h_curr.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        c5_new = compute_c5(h_new)
        if c5_new <= c5_curr or (c5_new - c5_curr) / T < np.log(rng.random()):
            h_curr = h_new
            c5_curr = c5_new
            if c5_curr < best_c5:
                best_local = h_curr.copy()
                best_c5 = c5_curr
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        T *= 0.9998
        if no_improve > 2000:
            T = max(T, 0.008)
            no_improve = 0
    return best_local, best_c5

def lse_projected_gradient(h, target_sum, beta, t_end):
    n = len(h)
    h_curr = h.copy().astype(float)
    alpha = 0.05
    stall = 0
    while time.time() < t_end and stall < 200:
        corr = np.correlate(h_curr, 1.0 - h_curr, mode="full")
        max_c = corr.max()
        p = np.exp(beta * (corr - max_c))
        p_sum = p.sum()
        if p_sum < 1e-300:
            break
        p /= p_sum
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            s = k - (n - 1)
            w = p[k]
            if w < 1e-13:
                continue
            contrib = np.ones(n)
            if s > 0:
                contrib[:n - s] -= h_curr[s:]
                contrib[s:] -= h_curr[:n - s]
            elif s < 0:
                s2 = -s
                contrib[s2:] -= h_curr[:n - s2]
                contrib[:n - s2] -= h_curr[s2:]
            else:
                contrib -= 2.0 * h_curr
            grad += w * contrib
        grad *= 2.0 / n
        h_new = project_polytope(h_curr - alpha * grad, target_sum)
        c5_old = compute_c5(h_curr)
        c5_new = compute_c5(h_new)
        if c5_new < c5_old * (1 - 1e-10):
            h_curr = h_new
            stall = 0
            alpha = min(alpha * 1.05, 0.5)
        else:
            alpha *= 0.7
            stall += 1
    return h_curr
