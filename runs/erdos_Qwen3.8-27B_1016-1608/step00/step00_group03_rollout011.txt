import numpy as np
from scipy.optimize import minimize
import time

_TARGET_SUM = 0.0
_SMOOTH_BETA = 50.0

def c5_objective(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    return float(np.max(corr))

def c5_smooth(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = float(np.max(corr))
    return m + np.log1p(np.sum(np.exp(_SMOOTH_BETA * (corr - m)))) / _SMOOTH_BETA

def constraint_fun(x):
    return float(np.sum(x) - _TARGET_SUM)

def repair(h, n_points):
    h = np.array(h, dtype=float).flatten()[:n_points]
    if len(h) < n_points:
        h = np.pad(h, (0, n_points - len(h)))
    target = n_points / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = float(np.sum(h))
    if s < 1e-15:
        return np.full(n_points, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - float(np.sum(h))
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
            if len(idx) == 0:
                break
            h[idx] += diff / len(idx)
            h = np.minimum(h, 1.0)
        else:
            idx = np.where(h > 1e-12)[0]
            if len(idx) == 0:
                break
            h[idx] += diff / len(idx)
            h = np.maximum(h, 0.0)
    return h

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx
