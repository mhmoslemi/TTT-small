import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_onto_constraint(h, n, tol=1e-10):
    """Project h onto {0 ≤ h ≤ 1, sum(h) = n/2}"""
    for _ in range(100):
        s = np.sum(h)
        diff = s - n/2
        if abs(diff) < tol:
            break
        h = h - diff / n
        h = np.clip(h, 0, 1)
    # Final bisection for exact sum
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h_trial = np.clip(h + mid, 0, 1)
        if np.sum(h_trial) > n/2:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo+hi)/2, 0, 1)

def subgradient_step(h, n, step_size):
    """One subgradient step for the max correlation problem"""
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full') * dx
    # Find the maximizing shift
    idx = np.argmax(corr)
    s = idx - (n - 1)  # shift in grid points
    
    # Compute gradient of corr[idx] with respect to h
    # corr[idx] = dx * sum_j h[j] * (1-h[j+s])
    # d/dh[j] = dx * (1-h[j+s]) - dx*h[j-s]
    grad = dx * np.ones(n)
    # h[j+s] contribution for j in [0, n-1-s] if s >= 0
    if s >= 0:
        grad[:n-s] -= dx * h[s:]
        grad[s:] -= dx * h[:n-s]
    else:
        s_pos = -s
        grad[s_pos:] -= dx * h[:n-s_pos]
        grad[:n-s_pos] -= dx * h[s_pos:]
    
    return h - step_size * grad

def optimize_subgradient(h0, n, max_iters=10000, time_limit=60):
    h = h0.copy()
    h = project_onto_constraint(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    t0 = time.time()
    
    step_size = 0.5
    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        
        h = subgradient_step(h, n, step_size)
        h = project_onto_constraint(h, n)
        
        c5 = compute_c5(h, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Decay step size
        if it % 100 == 0:
            step_size *= 0.95
    
    return best_h, best_c5

    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)

def _project(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n * 0.5
    # Bisection on additive shift
    lo, hi = -h.max() - 0.01, 1.0 - h.min() + 0.01

        ht = np.clip(h + mid, 0.0, 1.0)
        if ht.sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)

def _grad_weighted(h, n, weights, lags):
    """Compute weighted gradient: sum_k w_k * grad(g_k)"""
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, s in zip(weights, lags):
        if wi < 1e-12:
            continue
        if s == 0:
            grad += wi * dx * (1.0 - 2.0 * h)
        elif s > 0:
            grad[:n-s] += wi * dx * (1.0 - h[s:])
            grad[s:] -= wi * dx * h[:n-s]
        else:
            sp = -s
            grad[sp:] += wi * dx * (1.0 - h[:n-sp])
            grad[:n-sp] -= wi * dx * h[sp:]
    return grad

def _smooth_single(h, n, temp):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    mc = corr.max()
    return mc + temp * np.log(np.exp((corr - mc) / temp).sum())

def _smooth_single_grad(h, n, temp):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    mc = corr.max()
    w = np.exp((corr - mc) / temp)
    w /= w.sum()
    lags = np.arange(len(corr)) - (n - 1)
    return _grad_weighted(h, n, w, lags)

            return _smooth_single_grad(x, n, temp)

            return np.ones(n)

        v = compute_c5(h, n)
        if v < bv:

    return bh, bv

        elif s > 0:
            grad[:n - s] += wi * dx * (1.0 - h[s:])
            grad[s:] -= wi * dx * h[:n - s]
        else:
            sp = -s
            grad[sp:] += wi * dx * (1.0 - h[:n - sp])
            grad[:n - sp] -= wi * dx * h[sp:]
    return grad

def _smooth_val(h, n, temp):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    mc = float(corr.max())
    return mc + temp * float(np.log(np.exp((corr - mc) / temp).sum()))

def _smooth_grad(h, n, temp):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    mc = float(corr.max())
    w = np.exp((corr - mc) / temp)
    w /= w.sum()
    lags = np.arange(len(corr)) - (n - 1)
    return _grad_weighted(h, n, w, lags)

def _obj_slsqp(x):
    return _smooth_val(x, len(x), _T)

def _obj_grad_slsqp(x):
    return _smooth_grad(x, len(x), _T)

def _constr_eq(x):
    return x.sum() - len(x) * 0.5

def _constr_jac(x):
    return np.ones(len(x))

def _opt_slsqp(h0, n, tlim):
    h = _project(h0.copy(), n)
    t0 = time.time()
    bh, bv = h.copy(), compute_c5(h, n)
    for temp in (0.05, 0.02, 0.008, 0.003):
        if time.time() - t0 > tlim:
            break
        _T = temp
        res = minimize(_obj_slsqp, h, jac=_obj_grad_slsqp, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': _constr_eq, 'jac': _constr_jac}],
                       options={'maxiter': 400, 'ftol': 1e-14})
        h = _project(res.x, n)
        v = compute_c5(h, n)
        if v < bv:
            bh, bv = h.copy(), v
    return bh, bv

def _opt_subgrad(h0, n, tlim):
    h = _project(h0.copy(), n)
    t0 = time.time()
    bh, bv = h.copy(), compute_c5(h, n)
    dx = 2.0 / n
    nc = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    w = np.full(nc, 1.0 / nc)
    lags = np.arange(nc) - (n - 1)
    lr0 = 0.4
    for it in range(600000):
        if time.time() - t0 > tlim:
            break
        lr = lr0 / (1.0 + 0.005 * it)
        if nc > 16:
            top = np.argpartition(corr, -8)[-8:]
            we = np.zeros(nc)
            we[top] = w[top]
            we /= we.sum()
        else:
            we = w
        grad = _grad_weighted(h, n, we, lags)
        h = _project(h - lr * grad, n)
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        w *= np.exp(0.15 * (corr - corr.max()))
        w /= w.sum()
        if it % 200 == 0:
            v = compute_c5(h, n)
            if v < bv:
                bh, bv = h.copy(), v
    v = compute_c5(h, n)
    if v < bv:
        bh, bv = h.copy(), v
    return bh, bv

def _upsample(h, n0, n1):
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
    x1 = np.linspace(0.0, 2.0, n1, endpoint=False)
    return np.interp(x1, x0, h)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    try:
        h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        h_init = np.full(45, 0.5)
    n_init = len(h_init)
    h_init = _project(h_init, n_init)
    bh, bc = h_init.copy(), compute_c5(h_init, n_init)

    def rem():
        return max(0.0, budget_s - (time.time() - t_start))

    # Phase 1: SLSQP at initial resolution
    if rem() > 40:
        oh, oc = _opt_slsqp(bh, n_init, min(rem() - 25, 100))
        if oc < bc:
            bh, bc = oh, oc

    # Phase 2: Progressive upsample + optimize
    cn = n_init
    for mult in (2, 4, 8, 16, 32):
        nn = n_init * mult
        if nn > 2000 or rem() < 25:
            break
        hu = _project(_upsample(bh, cn, nn), nn)
        cn = nn
        if nn <= 250:
            oh, oc = _opt_slsqp(hu, nn, min(rem() - 15, 140))
        else:
            oh, oc = _opt_subgrad(hu, nn, min(rem() - 15, 140))
        if oc < bc:
            bh, bc = oh, oc

    # Phase 3: Extra SLSQP on best (if small enough)
    rem2 = rem()
    if rem2 > 40 and len(bh) <= 300:
        oh, oc = _opt_slsqp(bh, len(bh), min(rem2 - 15, 160))
        if oc < bc:
            bh, bc = oh, oc

    # Phase 4: Subgradient refinement
    rem3 = rem()
    if rem3 > 40:
        oh, oc = _opt_subgrad(bh, len(bh), min(rem3 - 15, 200))
        if oc < bc:
            bh, bc = oh, oc

    # Phase 5: Binarize then re-optimize
    rem4 = rem()
    if rem4 > 40:
        nb = (bh > 0.5).astype(float)
        nb = _project(nb, len(bh))
        cb = compute_c5(nb, len(bh))
        best_pair = (cb, nb)
        if len(bh) <= 300 and rem4 > 60:
            oh, oc = _opt_slsqp(nb, len(bh), min(rem4 - 30, 120))
            if oc < best_pair[0]:
                best_pair = (oc, oh)
        if rem4 > 50:
            oh, oc = _opt_subgrad(nb, len(bh), min(rem4 - 40, 80))
            if oc < best_pair[0]:
                best_pair = (oc, oh)
        if best_pair[0] < bc:
            bc, bh = best_pair

    # Phase 6: Random restarts if time allows
    rem5 = rem()
    if rem5 > 60 and len(bh) <= 200:
        for _ in range(3):
            if rem5 < 25:
                break
            hr = np.random.rand(len(bh))
            hr = _project(hr, len(bh))
            if len(bh) <= 200:
                oh, oc = _opt_slsqp(hr, len(bh), min(rem5 - 15, 100))
            else:
                oh, oc = _opt_subgrad(hr, len(bh), min(rem5 - 15, 100))
            rem5 = rem()
            if oc < bc:
                bh, bc = oh, oc

    return bh, bc, len(bh)
