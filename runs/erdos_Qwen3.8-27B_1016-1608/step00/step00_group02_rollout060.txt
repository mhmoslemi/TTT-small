                    best_h = h_res.copy()
                    best_n = n
                h_start = h_res
            except Exception:
                pass

        if time.time() - t0 < budget_s - 45:
            h_cd, c_cd = coordinate_descent(
                best_h.copy() if best_n == n else h0, n, max_iter=200)
            if c_cd < best_c5:
                best_c5 = c_cd
                best_h = h_cd.copy()
                best_n = n

        n_restarts = 3 if n <= 81 else 2
        for r in range(n_restarts):
            if time.time() - t0 > budget_s - 30:
                break
            h_rand = fix_sum(rng.rand(n), n)
            try:
                res_r = minimize(
                    smooth_obj, h_rand, args=(n, 100.0),
                    method='SLSQP', bounds=bounds,
                    constraints=[{'type': 'eq', 'fun': eq_constraint, 'args': (n,)}],
                    options={'maxiter': 1000, 'ftol': 1e-13}
                )
                h_r = fix_sum(res_r.x, n)
                c_r = compute_c5(h_r, n)
                if c_r < best_c5:
                    best_c5 = c_r
                    best_h = h_r.copy()
                    best_n = n
            except Exception:
                pass

    best_c5 = compute_c5(best_h, best_n)
    return best_h, best_c5, best_n
