import time

def compute_c5(h, n):
    dx = 2.0 / n
    cor = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(cor) * dx)

def compute_c5_grad(h, n):
    """Compute C5 and subgradient"""
    dx = 2.0 / n
    cor = np.correlate(h, 1.0 - h, mode="full")
    c5 = np.max(cor) * dx
    
    # Find active constraint(s)
    max_val = np.max(cor)
    active = (cor == max_val)
    
    # Subgradient: average gradient over active constraints
    # cor_k = sum_i h[i+k] * (1-h[i]) for shift k
    # d cor_k / d h[j] involves boundary effects
    
    # Actually, let me use finite differences for safety
    grad = np.zeros(n)
    eps = 1e-6
    for j in range(n):
        h_plus = h.copy()
        h_plus[j] = min(1.0, h[j] + eps)
        h_minus = h.copy()
        h_minus[j] = max(0.0, h[j] - eps)
        grad[j] = (compute_c5(h_plus, n) - compute_c5(h_minus, n)) / (h_plus[j] - h_minus[j])

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)
    if abs(s - target) < 1e-10:
        return h
    # Water-filling projection
    if s > target:
        diff = s - target
        idx = np.argsort(-h)
        for i in idx:
            if diff <= 1e-12:
                break
            dec = min(diff, h[i])
            h[i] -= dec
            diff -= dec
    else:
        diff = target - s
        idx = np.argsort(h)
        for i in idx:
            if diff <= 1e-12:
                break
            inc = min(diff, 1.0 - h[i])
            h[i] += inc
            diff -= inc
    # Fix any floating point issues
    h = np.clip(h, 0.0, 1.0)
    # Final adjustment
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        h[np.argmax(h if residual > 0 else 1-h)] += residual if residual > 0 else residual
    return h

        obj = cor[mi]
        grad = np.zeros(n)
        for j in range(n):

            neg = min(eps, h[j])
            gp = 0.0
            gm = 0.0

def binary_local_search(h, n, t_end, rng):
    h = h.copy()
    c5 = compute_c5(h, n)
    while time.time() < t_end:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i1 = int(ones[rng.integers(len(ones))])
        i0 = int(zeros[rng.integers(len(zeros))])
        h[i1] = 0.0
        h[i0] = 1.0
        c5_new = compute_c5(h, n)
        if c5_new <= c5:
            c5 = c5_new
        else:
            h[i1] = 1.0
            h[i0] = 0.0
    return h, c5

def round_to_binary(h, n, rng):
    h = h.copy()
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    target = n // 2
    s = int(np.sum(h))
    if s > target:
        ones = np.where(h == 1.0)[0]
        perm = rng.permutation(len(ones))
        h[ones[perm[:s - target]]] = 0.0
    elif s < target:
        zeros = np.where(h == 0.0)[0]
        perm = rng.permutation(len(zeros))
        h[zeros[perm[:target - s]]] = 1.0
    return h
