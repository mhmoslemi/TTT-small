                    break
            if not improved:
                break

    # Phase 3: Multiple random restarts with local optimization
    restart_count = 0
    while time_left() > 25:
        restart_count += 1
        scale = rng.uniform(0.05, 0.25) * max(1.0, 30.0 / restart_count)
        h_p = best_h + rng.randn(n) * scale
        h_p = repair_h(h_p, target_sum)

        local_alpha = rng.choice([40, 25, 15, 10])

        def obj_r(h_flat, a=local_alpha):
            h_v = np.clip(h_flat, 0, 1)
            c = np.correlate(h_v, 1.0 - h_v, mode="full") * dx
            sm = smooth_max(c, a)
            s = np.sum(h_v)
            return sm + 1000.0 * (s - target_sum) ** 2

        try:
            res = minimize(obj_r, h_p, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': 120, 'ftol': 1e-12})
            h_new = repair_h(res.x, target_sum)
            c5_new, _ = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        except Exception:
            pass

        # Quick coordinate polish after each restart
        if c5_new < best_c5 + 0.001:
            for _cd in range(20):
                if time_left() < 5:
                    break
                i = rng.randint(n)
                j = rng.randint(n)
                if j == i:
                    j = (j + 1) % n
                delta = rng.choice([0.03, -0.03, 0.015, -0.015])
                h_t = best_h.copy()
                h_t[i] += delta
                h_t[j] -= delta
                if 0 <= h_t[i] <= 1 and 0 <= h_t[j] <= 1:
                    c5_t, _ = compute_c5(h_t)
                    if c5_t < best_c5:
                        best_c5 = c5_t
                        best_h = h_t.copy()

    # Final verification
    best_c5, _ = compute_c5(best_h)
    return (best_h, best_c5, n)
