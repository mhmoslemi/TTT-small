import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_sum(v, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def local_search(h, time_limit, rng, step0=0.08):
    n = len(h)
    h = h.copy()
    current = compute_c5(h)
    start = time.time()
    step = step0
    fails = 0
    while time.time() - start < time_limit:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        found = False
        oi, oj = h[i], h[j]
        for d in (step, step * 0.4, 0.02, 0.007):
            v, w = oi + d, oj - d
            if 0.0 <= v <= 1.0 and 0.0 <= w <= 1.0:
                h[i], h[j] = v, w
                c5n = compute_c5(h)
                if c5n < current:
                    current = c5n
                    found = True
                    break
            v, w = oi - d, oj + d
            if 0.0 <= v <= 1.0 and 0.0 <= w <= 1.0:
                h[i], h[j] = v, w
                c5n = compute_c5(h)
                if c5n < current:
                    current = c5n
                    found = True
                    break
        if not found:
            fails += 1
            if fails > 400:
                step = max(0.003, step * 0.7)
                fails = 0
        else:
            step = min(0.1, max(step, abs(d) * 1.3))
    return h, current

    lr = 0.01
    while time.time() - start < time_limit:

        dx = 2.0 / n
        idx = np.argmax(c)
        p = idx - (n - 1)
        grad = np.zeros(n)

            ip, im = i - p, i + p
