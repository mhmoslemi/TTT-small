def c5_value(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    """Project h onto {x : sum(x) = target_sum, 0 <= x_i <= 1}"""
    # Bisection on tau: find tau such that sum(clip(h_i - tau, 0, 1)) = target_sum
    lo, hi = h.min() - 1, h.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def subgradient(h, n):
    """Compute subgradient of max_k c_k with respect to h"""
    corr = np.correlate(h, 1.0 - h, mode='full')

    # Element k in full mode corresponds to lag k - (n-1)
    max_idx = np.argmax(corr)
    k = max_idx - (n - 1)  # the actual lag
    # c_k = sum_n h[n+k] * (1-h[n])

            g += (1.0 - h[j1])
        j2 = i + k
        if 0 <= j2 < n:

        grad[i] = g
    return grad

        if abs_k < n:

import numpy as np
import time

def c5_value(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    lo = float(h.min()) - 1.0
    hi = float(h.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)

def to_binary(h, target_sum):
    n = len(h)
    h_bin = (h >= 0.5).astype(float)
    cs = int(round(h_bin.sum()))
    ts = int(round(target_sum))
    while cs > ts:
        oi = np.where(h_bin == 1.0)[0]
        idx = oi[np.argmin(h[oi])]
        h_bin[idx] = 0.0
        cs -= 1
    while cs < ts:
        zi = np.where(h_bin == 0.0)[0]
        idx = zi[np.argmax(h[zi])]
        h_bin[idx] = 1.0
        cs += 1
    return h_bin

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.zeros(70)
        h0[:35] = 1.0

    n = len(h0)
    ts = n // 2

    h = to_binary(h0, float(ts))
    best_h = h.copy()
    best_c5 = c5_value(h)

    cur_c5 = best_c5
    no_improve = 0
    kick_count = 0

    while time.time() - t0 < budget_s - 5:
        oi = np.where(h == 1.0)[0]
        zi = np.where(h == 0.0)[0]
        i = int(oi[rng.randint(len(oi))])
        j = int(zi[rng.randint(len(zi))])

        h[i] = 0.0
        h[j] = 1.0
        new_c5 = c5_value(h)

        if new_c5 < cur_c5:
            cur_c5 = new_c5
            no_improve = 0
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1

        if no_improve >= 2000 + 500 * kick_count:
            h = best_h.copy()
            cur_c5 = best_c5
            nk = rng.randint(3, 6 + kick_count)
            for _ in range(nk):
                oi2 = np.where(h == 1.0)[0]
                zi2 = np.where(h == 0.0)[0]
                a = int(oi2[rng.randint(len(oi2))])
                b = int(zi2[rng.randint(len(zi2))])
                h[a] = 0.0
                h[b] = 1.0
            cur_c5 = c5_value(h)
            no_improve = 0
            kick_count += 1

    # Continuous refinement on best binary solution
    if time.time() - t0 < budget_s - 10:
        hc = best_h.copy()
        hc5 = best_c5
        lr = 0.02
        for it in range(20000):
            if time.time() - t0 > budget_s - 8:
                break
            corr = np.correlate(hc, 1.0 - hc, mode='full')
            ki = int(np.argmax(corr))
            k = ki - (n - 1)
            grad = np.zeros(n)
            for ii in range(n):
                v = 0.0
                p1 = ii - k
                if 0 <= p1 < n:
                    v += 1.0 - hc[p1]

                    v -= hc[p2]

            else:
                lr *= 0.95
            if hc5 < best_c5:
                best_h = hc.copy()
                best_c5 = hc5

    return best_h, best_c5, n
