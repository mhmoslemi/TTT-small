    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def make_neighbor(h, n, rng, step_size):
    idx = rng.integers(0, n, size=2)
    if idx[0] == idx[1]:
        return None
    i, j = int(idx[0]), int(idx[1])
    delta = rng.uniform(-step_size, step_size)
    hi = h[i] + delta
    hj = h[j] - delta
    if hi < 0 or hi > 1 or hj < 0 or hj > 1:
        max_d_min = min(h[i], 1.0 - h[i], h[j], 1.0 - h[j])
        if max_d_min < 1e-8:
            return None
        sign = 1 if delta > 0 else -1
        delta = sign * rng.uniform(0.05, 1.0) * max_d_min
        hi = h[i] + delta
        hj = h[j] - delta
        if hi < -1e-12 or hi > 1 + 1e-12 or hj < -1e-12 or hj > 1 + 1e-12:
            return None
    return i, j, delta, hi, hj

def sa_loop(h, n, rng, t_end):
    best_h = h.copy()
    best_c5 = eval_c5(h)
    h = h.copy()
    c5 = best_c5
    target = n / 2.0

    T = best_c5 * 0.8
    T_min = 1e-10
    alpha = 0.99985
    step_size = 0.15
    stuck = 0
    no_improve = 0
    evals = 0

    while time.time() < t_end - 1:
        nb = make_neighbor(h, n, rng, step_size)
        if nb is None:
            continue
        i, j, delta, hi, hj = nb
        h[i] = hi
        h[j] = hj
        c5_new = eval_c5(h)
        evals += 1

        if c5_new <= c5:
            c5 = c5_new
            no_improve = 0
            stuck = 0
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            stuck += 1
            no_improve += 1
            if rng.random() < np.exp(-(c5_new - c5) / max(T, T_min)):
                c5 = c5_new
            else:
                h[i] = hi - delta
                h[j] = hj + delta

        T = max(T * alpha, T_min)

        if stuck > 200:
            step_size = min(step_size * 1.3, 0.5)
            T = max(T * 2.0, best_c5 * 0.1)
            stuck = 0
        elif no_improve > 5000:
            step_size = max(step_size * 0.85, 0.03)
            no_improve = 0
        elif stuck == 0:
            step_size = max(step_size * 0.997, 0.02)

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    h_init = None
    n_init = None
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        pass

    if h_init is None:
        n_init = 48
        h_init = np.full(n_init, 0.5)
        for _ in range(100):
            a, b = int(rng.integers(n_init)), int(rng.integers(n_init))
            if a != b:
                d = rng.uniform(-0.4, 0.4)
                if 0 <= h_init[a] + d <= 1 and 0 <= h_init[b] - d <= 1:
                    h_init[a] += d
                    h_init[b] -= d
        h_init = repair_h(h_init, n_init)

    # Evaluate initial
    c5_init = eval_c5(h_init)
    best_c5 = c5_init
    best_h = h_init.copy()
    best_n = n_init

    # Build n list: parent first, then nearby
    n_list = [n_init]
    for n in [n_init - 1, n_init + 1, n_init + 2, n_init - 2,
              40, 44, 48, 50, 52, 54, 56, 60, 64, 72, 80, 96]:
        if n >= 16 and n not in n_list and n != n_init:
            n_list.append(n)

    # Prioritize: parent, then nearby, then others
    n_list.sort(key=lambda x: (x != n_init, abs(x - n_init)))

    t_remaining = budget_s - 15
    # Allocate time: parent gets 50%, next 2 get 20% each, rest share 10%
    n_to_try = min(len(n_list), max(4, int(t_remaining / 60)))
    n_list = n_list[:n_to_try]

    for idx, n in enumerate(n_list):
        elapsed = time.time() - start_time
        t_left = budget_s - 15 - elapsed
        if t_left < 20:
            break

        # Time allocation
        frac = [0.45, 0.2, 0.15, 0.1, 0.05, 0.05][:n_to_try]
        total_frac = sum(frac)
        alloc = t_left * frac[idx] / total_frac
        alloc = max(alloc, 15)
        alloc = min(alloc, t_left - 5)
        t_end = time.time() + alloc

        # Build initial h for this n
        if n == n_init:
            h0 = h_init.copy()
        else:
            x_old = np.linspace(0, 2, n_init, endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            h_ext = np.concatenate([h_init, [h_init[0]]])
            h_interp = np.interp(x_new, x_old, h_ext)
            h0 = repair_h(h_interp, n)
            # Also try random start
            h_rand = np.full(n, 0.5)
            for _ in range(80):
                a, b = int(rng.integers(n)), int(rng.integers(n))
                if a != b:
                    d = rng.uniform(-0.35, 0.35)
                    if 0 <= h_rand[a] + d <= 1 and 0 <= h_rand[b] - d <= 1:
                        h_rand[a] += d
                        h_rand[b] -= d
            h_rand = repair_h(h_rand, n)

        # Run SA on h0
        h_opt, c5_opt = sa_loop(h0, n, rng, t_end * 0.55)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n

        # If time remains and n != n_init, try random start
        if n != n_init and time.time() < t_end - 5:
            t_end2 = t_end - 1
            h_opt2, c5_opt2 = sa_loop(h_rand, n, rng, t_end2)
            if c5_opt2 < best_c5:
                best_c5 = c5_opt2
                best_h = h_opt2.copy()
                best_n = n
            elif c5_opt2 < c5_opt and time.time() < t_end2 - 3:
                # Restart from this
                h_r = h_opt2 + rng.normal(0, 0.05, n)
                h_r = repair_h(h_r, n)
                h_opt3, c5_opt3 = sa_loop(h_r, n, rng, t_end2)
                if c5_opt3 < best_c5:
                    best_c5 = c5_opt3
                    best_h = h_opt3.copy()
                    best_n = n

        # For parent n, do additional SA round from best
        if n == n_init and time.time() < t_end - 10:
            h_r = best_h + rng.normal(0, 0.08, n_init)
            h_r = repair_h(h_r, n_init)
            h_opt_r, c5_opt_r = sa_loop(h_r, n_init, rng, t_end)
            if c5_opt_r < best_c5:
                best_c5 = c5_opt_r
                best_h = h_opt_r.copy()
                best_n = n_init

    # Final validation
    best_h = repair_h(best_h, best_n)
    final_c5 = eval_c5(best_h)
    best_c5 = min(best_c5, final_c5)

    return best_h, best_c5, best_n
