            if fsnap < fr:
                hr, fr = hsnap, fsnap

            if fr < best_c5:
                best_c5 = fr
                best_h = hr.copy()
                best_n = n

    if best_h is None:
        best_n = 47
        best_h = block_init(best_n)
        best_c5 = eval_c5(best_h)

    best_h = repair(best_h, best_n)
    best_c5 = eval_c5(best_h)
    return best_h, float(best_c5), int(best_n)
