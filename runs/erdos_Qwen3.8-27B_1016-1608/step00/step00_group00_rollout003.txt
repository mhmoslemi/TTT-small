import numpy as np
from scipy.optimize import minimize
import time

_G = {}

def _corr_max(h, dx):

    return corr * dx, np.max(corr) * dx

    m = np.max(corr_vals)
    return m + np.log(np.sum(np.exp(corr_vals - m))) / _G['alpha']

def _eq_fun(h):
    return h.sum() - _G['ts']

def _eq_jac(h):
    return np.ones(_G['n'])

def _project(h, n, ts):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = ts - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            break
        step = diff / mask.sum()
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    return h

def _optimize(h_init, n, dx, ts, maxiter, alpha):
    _G['n'] = n
    _G['dx'] = dx
    _G['ts'] = ts
    _G['alpha'] = float(alpha)
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    bnds = [(0.0, 1.0)] * n
    try:

                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = _project(res.x, n, ts)
        _, c5 = _corr_max(h_new, dx)
        return h_new, c5
    except Exception:
        return None, None
