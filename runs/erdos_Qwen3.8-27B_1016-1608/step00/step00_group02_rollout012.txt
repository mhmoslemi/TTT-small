from scipy.optimize import minimize

def compute_c5(h_values):
    n = len(h_values)
    dx = 2.0 / n
    comp = 1.0 - h_values
    corr = np.correlate(h_values, comp, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-8) & (h < 1.0 - 1e-8)
        nf = int(np.sum(free))
        if nf > 0:
            h[free] += diff / nf
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_func(params, alpha, penalty):
    h = params
    n = len(h)
    dx = 2.0 / n
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode="full")
    c = corr * dx
    m = np.max(c)
    shifted = alpha * (c - m)
    sm = m + np.log(np.sum(np.exp(shifted))) / alpha
    sv = np.sum(h) - n / 2.0
    return sm + penalty * sv * sv
