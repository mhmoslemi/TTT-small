import numpy as np
import time

def compute_overlap(h):
    """Return all overlap values (not multiplied by dx)."""
    return np.correlate(h, 1 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(compute_overlap(h)) * dx

def project(h, target_sum):
    """Project h onto {0 <= h[i] <= 1, sum(h) = target_sum}."""
    h = np.clip(h, 0, 1)
    # Bisection on Lagrange multiplier
    lo, hi = -10, 10
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0, 1)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    # Get initial
    h = None
    if 'initial_h_values' in dir():
        try:
            h = np.array(initial_h_values, dtype=float).copy()
        except:
            pass
    if h is None or len(h) == 0:
        n = 42
        h = np.full(n, 0.5)
    n = len(h)
    target_sum = n / 2.0
    
    h = project(h, target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Subgradient descent
    lr = 0.1
    for it in range(100000):
        if time.time() - t0 > budget_s - 30:
            break
        
        ov = compute_overlap(h)
        max_val = np.max(ov)
        # Active set: shifts within 5% of max
        active = np.where(ov > max_val * 0.95)[0]
        
        # Compute gradient of sum of active overlaps
        grad = np.zeros(n)
        for k_idx in active:
            shift = k_idx - (n - 1)
            # c_k = sum_n h[n+shift] * (1-h[n])
            # ∂c_k/∂h[j] = (1-h[j-shift]) if 0<=n-shift<n AND 0<=j<n
            #               - h[j+shift] if 0<j+shift<n
            for j in range(n):
                # h[j] appears as h[n+shift] when n = j-shift
                if 0 <= j - shift < n:
                    grad[j] += (1 - h[j - shift])
                # h[j] appears in (1-h[n]) when n = j
                if 0 <= j + shift < n:
                    grad[j] += -h[j + shift]
        
        # Negative gradient step
        h_new = h - lr * grad
        h_new = project(h_new, target_sum)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            lr *= 0.99  # slow decay
        else:
            lr *= 0.9

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full"))) * dx

def project_sum(h, target):
    lo, hi = -10.0, 10.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def grad_active(h, active_idx, n):
    center = n - 1
    grad = np.zeros(n)
    for idx in active_idx:
        k = idx - center
        if k >= 0:
            if n - k > 0:
                grad[k:] += (1.0 - h[:n-k])
                grad[:n-k] -= h[k:]
        else:
            k2 = -k
            if n - k2 > 0:
                grad[:n-k2] += (1.0 - h[k2:])
                grad[k2:] -= h[:n-k2]
    return grad

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 15
    
    # Initialize
    h = None
    try:
        h = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h = None
    if h is None or len(h) == 0:
        n = 42
        h = np.full(n, 0.5)
    n = len(h)
    target = n / 2.0
    h = project_sum(h, target)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)

    lr = 0.02
    momentum = 0.7
    vel = np.zeros(n)
    patience = 0
    max_patience = 5000
    
    while time.time() < deadline - 60:
        ov = np.correlate(h, 1.0 - h, mode="full")
        mx = np.max(ov)
        # Active set: top 30% of overlaps or within 10% of max
        threshold = max(mx * 0.9, mx - 2.0)
        active = np.where(ov >= threshold)[0]
        
        g = grad_active(h, active, n)
        vel = momentum * vel - lr * g
        h_new = project_sum(h + vel, target)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new

                lr = 0.02

    h = best_h.copy()

    surplus = int(bb.sum()) - int(tgt)
    if surplus > 0:
        idx = np.argsort(h)[:surplus]
        bb[idx] = 0.0
    elif surplus < 0:
        idx = np.argsort(h)[-surplus:]
        bb[idx] = 1.0
    best_h, best_c5 = bb.copy(), compute_c5(bb)
    for _ in range(n_iter):
        ones = np.where(bb == 1)[0]
        zeros = np.where(bb == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        bb2 = bb.copy()
        bb2[i], bb2[j] = 0.0, 1.0
        c5 = compute_c5(bb2)
        if c5 < best_c5:
            best_c5, best_h = c5, bb2.copy()
            bb = bb2
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    dl = t0 + budget_s - 8

    # Initialize from parent if available
    h = None
    try:
        h = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        pass
    if h is None or len(h) == 0:
        h = np.full(42, 0.5)
    n = len(h)
    tgt = n / 2.0
    h = project(h, tgt)

    # Phase 1: Subgradient (60% of time)
    t1 = min(t0 + budget_s * 0.6, dl - 120)
    h, c5 = subgrad_optimize(h, tgt, t1, lr0=0.03, seed_offset=seed)

    # Phase 2: Pair descent (20% of time)
    t2 = min(t0 + budget_s * 0.8, dl - 40)
    n_iter2 = max(50000, int((t2 - time.time()) * 3000))
    h, c5 = pair_descent(h, tgt, n_iter2, step=0.012, seed_offset=seed + 1)

    # Phase 3: Binary search (10% of time)
    t3 = min(t0 + budget_s * 0.92, dl - 15)
    n_iter3 = max(20000, int((t3 - time.time()) * 2000))
    hb, cb = binary_swap_search(h, tgt, n_iter3, seed_offset=seed + 2)
    if cb < c5:
        h, c5 = hb, cb

    # Phase 4: Final fine-tuning if time remains
    if time.time() < dl - 5:
        h, c5 = pair_descent(h, tgt, 100000, step=0.005, seed_offset=seed + 3)

    # Phase 5: Try higher resolution if time permits
    if time.time() < dl - 60 and n <= 42:
        n2 = 2 * n
        # Upsample
        h_up = np.zeros(n2)
        for i in range(n2):
            x = i / 2.0
            i0 = int(x)
            frac = x - i0
            h_up[i] = h[i0] * (1 - frac) + h[(i0 + 1) % n] * frac
        h_up = project(h_up, n2 / 2.0)
        h_up, c5_up = subgrad_optimize(h_up, n2 / 2.0, min(time.time() + 50, dl - 10), lr0=0.02, seed_offset=seed + 10)
        if c5_up < c5:
            h, c5, n = h_up, c5_up, n2

    # Final verify
    h = np.clip(h, 0, 1)
    h = project(h, len(h) / 2.0)
    c5 = compute_c5(h)
    return h, c5, len(h)
