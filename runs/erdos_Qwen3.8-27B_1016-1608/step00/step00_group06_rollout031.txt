from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def smooth_max_c5(h, dx, T=50.0):
    corr = np.correlate(h, 1 - h, mode="full")
    vals = corr * dx
    m = np.max(vals)
    # softmax approximation of max
    return m + (1.0/T) * np.log(np.sum(np.exp(T * (vals - m))))

def project_feasible(y, target_sum):
    n = len(y)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def subgradient(c5_corr, h, n):
    """Compute subgradient of max(corr) w.r.t. h."""
    kmax = np.argmax(c5_corr)
    tau = kmax - (n - 1)
    grad = np.zeros(n)
    # For each j, grad[j] = (1-h[j-tau])*[valid] - h[j+tau]*[valid]

        jp = j + tau
        if 0 <= jt < n:
            grad[j] += (1.0 - h[jt])
        if 0 <= jp < n:
            grad[j] -= h[jp]
    return grad
