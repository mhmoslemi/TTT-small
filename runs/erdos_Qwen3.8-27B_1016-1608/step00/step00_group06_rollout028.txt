import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_sum(h, target):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - float(np.sum(h))
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(free):
            h[free] += diff / float(np.sum(free))
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h


_sum_target = 0.0
_n_vars = 0


def _cons_fun(h):
    return float(np.sum(h)) - _sum_target


def _cons_jac(h):
    return np.ones(_n_vars)


def slsqp_refine(h, n, dx, target, maxiter=200):
    global _sum_target, _n_vars
    _sum_target = target
    _n_vars = n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    bounds = [(0.0, 1.0)] * n
    c5_old = compute_c5(h, dx)
    try:
        res = minimize(compute_c5, h, args=(dx,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = np.clip(res.x, 0.0, 1.0)
        h_new = project_sum(h_new, target)
        c5_new = compute_c5(h_new, dx)
        if c5_new < c5_old:
            return h_new, c5_new
    except Exception:
        pass
    return h, c5_old

    best_c5 = compute_c5(h, dx)
    h_curr = h.copy()
    deltas = [0.02, 0.05, 0.1, 0.15, 0.3, -0.02, -0.05, -0.1, -0.15, -0.3]
    no_improve = 0

                    h_curr[k] = val_k
                    c5_try = compute_c5(h_curr, dx)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try

                        break
                    h_curr[j] -= d

def simulated_anneal(h, n, dx, target, rng, deadline):
