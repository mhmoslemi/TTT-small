            corr = np.correlate(x, 1.0 - x, mode="full")
            return float(np.max(corr)) * dx_val

        def obj_grad(x):
            g = np.zeros(n)
            eps = 1e-7
            for k in range(n):
                x[k] += eps
                fp = float(np.max(np.correlate(x, 1.0 - x, mode="full"))) * dx_val
                x[k] -= 2.0 * eps
                fm = float(np.max(np.correlate(x, 1.0 - x, mode="full"))) * dx_val
                x[k] += eps
                g[k] = (fp - fm) / (2.0 * eps)
            return g

        constr = ({'type': 'eq', 'fun': lambda x: x.sum() - target,
                   'jac': lambda x: np.ones(n)},)
        bnds = [(0, 1)] * n
        res_start = time.time()
        try:
            res = minimize(obj_f, h_start, jac=obj_grad, method='SLSQP',
                           bounds=bnds, constraints=constr,
                           options={'maxiter': 3000, 'ftol': 1e-12,
                                    'disp': False, 'eps': 1e-7})
            if res.fun < best_c5 - 1e-10:
                h_slsq = np.clip(res.x, 0, 1)
                h_slsq = fix_sum(h_slsq, target)
                c5_slsq = compute_c5(h_slsq)
                if c5_slsq < best_c5:
                    best_h = h_slsq
                    best_c5 = c5_slsq
        except Exception:
            pass

    # Final: ensure strict validity
    best_h = np.clip(best_h, 0.0, 1.0)
    h_fixed = fix_sum(best_h.copy(), target)
    # Use the version with exact sum if it's better or equal
    c5_check = compute_c5(h_fixed)
    if c5_check <= best_c5 + 1e-12:
        best_h = h_fixed
        best_c5 = c5_check

    return (best_h, best_c5, n)
