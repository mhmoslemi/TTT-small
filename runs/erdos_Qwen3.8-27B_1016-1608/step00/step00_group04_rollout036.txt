import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_binary_fixed(h, n):
    target = n // 2
    h = (h > 0.5).astype(float)
    s = int(np.sum(h))
    if s > target:
        idx = np.random.permutation(np.flatnonzero(h))
        h[idx[:s-target]] = 0
    elif s < target:
        idx = np.random.permutation(np.flatnonzero(1-h))
        h[idx[:target-s]] = 1
    return h

def best_improvement_search(h, n, time_limit=30.0):
    h = h.copy()
    start = time.time()
    best = compute_c5(h)
    while time.time() - start < time_limit:
        ones = np.flatnonzero(h).tolist()
        zeros = np.flatnonzero(1-h).tolist()
        improvement = 0
        best_pair = None
        for i in ones:
            for j in zeros:
                h[i], h[j] = 0.0, 1.0
                s = compute_c5(h)
                h[i], h[j] = 1.0, 0.0
                if best - s > improvement:
                    improvement = best - s
                    best_pair = (i, j)
        if best_pair is None:
            break
        h[best_pair[0]], h[best_pair[1]] = 0.0, 1.0
        best -= improvement
    return h, best

def sa_binary(h, n, time_limit=30.0):
    h = h.copy()
    start = time.time()
    score = compute_c5(h)
    best_h = h.copy()
    best_score = score
    T = 0.01
    while time.time() - start < time_limit:
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1-h)
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        new_score = compute_c5(h)
        d = new_score - score
        if d < 0 or np.random.random() < np.exp(-d / max(T, 1e-10)):
            score = new_score
            if score < best_score:
                best_score = score
                best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
        T *= 0.99995
