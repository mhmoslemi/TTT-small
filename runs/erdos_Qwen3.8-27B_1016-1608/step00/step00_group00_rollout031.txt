```python
def compute_grad(h, w, n, dx):
    grad = np.zeros(n)
    for k in range(len(w)):
        wk = w[k]
        if wk < 1e-8:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += wk * dx * (1.0 - 2.0 * h)
        elif lag > 0:
            # (1-h[i-lag]) valid for i >= lag
            # -h[i+lag] valid for i <= n-1-lag
            n_valid_first = n - lag  # i from lag to n-1
            grad[lag:] += wk * dx * (1.0 - h[:n_valid_first])
            n_valid_second = n - lag  # i from 0 to n-1-lag
            grad[:n_valid_second] += -wk * dx * h[lag:lag+n_valid_second]
        else:
            abs_lag = -lag
            # (1-h[i-lag]) = (1-h[i+abs_lag]) valid for i <= n-1-abs_lag
            # -h[i+lag] = -h[i-abs_lag] valid for i >= abs_lag
            n_valid_first = n - abs_lag
            grad[:n_valid_first] += wk * dx * (1.0 - h[abs_lag:abs_lag+n_valid_first])
            n_valid_second = n - abs_lag
            grad[abs_lag:] += -wk * dx * h[:n_valid_second]
    return grad
```

```python
import numpy as np
```

```python
def project_poly(h, target):
    """Project h onto {x: 0<=x_i<=1, sum(x)=target} via bisection."""
    lo, hi = -2.0, 2.0

        if np.sum(np.clip(h + mid, 0.0, 1.0)) > target:
            hi = mid

            lo = mid
```
