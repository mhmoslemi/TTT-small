import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def repair(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s > 1e-15:
        h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        k = mask.sum()
        if k > 0:
            h[mask] += diff / k
            h = np.clip(h, 0.0, 1.0)
    return h

def obj_repaired(h_raw, n, target):
    h = np.clip(h_raw, 0.0, 1.0)
    s = h.sum()
    if s > 1e-15:
        h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        k = mask.sum()
        if k > 0:
            h[mask] += diff / k
            h = np.clip(h, 0.0, 1.0)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def subgradient_step(h, step_size, n, target):
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full")
    max_val = np.max(corr)
   thresh = max_val - 1e-8
    active = np.where(corr >= thresh)[0]
    grad = np.zeros(n)
    for m_idx in active:
        m = m_idx - (n - 1)
        for j in range(n):
            if 0 <= j - m < n:
                grad[j] += (1.0 - h[j - m])
            if 0 <= j + m < n:
                grad[j] -= h[j + m]
    grad /= len(active)
    h_new = h - step_size * grad
    h_new = repair(h_new, target)
    return h_new

    t0 = time.time()

        return time.time() - t0

    # Phase 1: Nelder-Mead from many restarts (70% budget)

        x0 = repair(x0, target)
        n_restart += 1

    return (best_h, best_c5, n)
