def simulated_annealing(h_init, n, dx, time_limit, seed):
    """Simulated annealing with pair flips for binary sequences."""
    rng = np.random.RandomState(seed)
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    T = 0.01  # Initial temperature
    T_min = 1e-6
    cooling = 0.9995
    start = time.time()
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    while time.time() - start < time_limit:
        # Pick random pair: one from ones, one from zeros
        if len(ones) == 0 or len(zeros) == 0:
            break
        i1 = ones[rng.randint(len(ones))]
        i2 = zeros[rng.randint(len(zeros))]
        
        # Flip: h[i1] 1→0, h[i2] 0→1
        h[i1] = 0
        h[i2] = 1
        c5_new = compute_c5(h, dx)
        
        if c5_new < best_c5:
            best_h = h.copy()
            best_c5 = c5_new
            # Update sets
            ones = ones[ones != i1]
            ones = np.append(ones, i2)
            zeros = zeros[zeros != i2]
            zeros = np.append(zeros, i1)
        else:
            # Accept with probability exp(-(c5_new - best_c5)/T)
            delta = c5_new - best_c5
            if delta > 0 and rng.random() < np.exp(-delta / T):
                # Accept worse solution
                ones = ones[ones != i1]
                ones = np.append(ones, i2)
                zeros = zeros[zeros != i2]
                zeros = np.append(zeros, i1)
            else:
                # Reject
                h[i1] = 1
                h[i2] = 0
        
        T = max(T * cooling, T_min)
    
    return best_h, best_c5

    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))
