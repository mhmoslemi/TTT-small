import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    hc = np.clip(h, 0.0, 1.0)
    corr = np.correlate(hc, 1.0 - hc, mode="full")
    mx = np.max(corr)
    # log-sum-exp for numerical stability
    if alpha > 500:
        return mx * dx
    shifted = corr - mx
    val = mx + np.log(np.sum(np.exp(alpha * shifted))) / alpha
    return val * dx

    h0 = h_init.copy()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n / 2.0}]

    best_val = compute_c5(h0)

        if abs(np.sum(h) - n / 2.0) > 1e-10:
            h = h + (n / 2.0 - np.sum(h)) / n
            h = np.clip(h, 0.0, 1.0)
    return h

def optimize_h(h_init, n, alpha_schedule, max_iter=150, time_limit=200):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n / 2.0}]
    best_h = h_init.copy()
    best_val = compute_c5(h_init)
    t0 = time.time()
    h_curr = h_init.copy()
    for alpha in alpha_schedule:
        if time.time() - t0 > time_limit - 5:
            break
        try:
            res = minimize(smooth_obj, h_curr, args=(alpha,),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': max_iter, 'ftol': 1e-14})
            h_new = enforce_sum(res.x, n)
        except Exception:
            h_new = h_curr.copy()
        val = compute_c5(h_new)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
        h_curr = h_new
    return best_h, best_val

def upsample(h, n_new):
    n_old = len(h)
    x_old = np.arange(n_old, dtype=float)
    x_new = np.linspace(0.0, n_old - 1.0, n_new)
    h_new = np.interp(x_new, x_old, h)
    h_new = enforce_sum(h_new, n_new)
    return h_new

def downsample(h, n_new):
    n_old = len(h)
    x_old = np.arange(n_old, dtype=float)
    x_new = np.linspace(0.0, n_old - 1.0, n_new)
    h_new = np.interp(x_new, x_old, h)
    h_new = enforce_sum(h_new, n_new)
    return h_new

def local_improve(h, n, n_steps=2000):
    best = h.copy()
    best_val = compute_c5(best)
    rng = np.random.default_rng(123)
    for _ in range(n_steps):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        delta = rng.uniform(-0.1, 0.1)
        new_h = best.copy()
        new_h[i] += delta
        new_h[j] -= delta
        if new_h[i] < 0 or new_h[i] > 1 or new_h[j] < 0 or new_h[j] > 1:
            continue
        new_val = compute_c5(new_h)
        if new_val < best_val - 1e-10:
            best = new_h
            best_val = new_val
    return best, best_val

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(45, 0.5)

    n0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Optimize at parent resolution
    alphas1 = [5, 10, 20, 50, 100, 200, 400]
    h_opt, c5_opt = optimize_h(h0.copy(), n0, alphas1, max_iter=200, time_limit=180)
    if c5_opt < best_c5:
        best_c5 = c5_opt
        best_h = h_opt

    # Local improvement at parent resolution
    elapsed = time.time() - t_start
    if budget_s - elapsed > 60:
        h_opt, c5_opt = local_improve(best_h, n0, n_steps=3000)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 2: Upsample and optimize at higher resolutions
    for n_new in [n0 * 2, n0 * 4]:
        elapsed = time.time() - t_start
        remaining = budget_s - elapsed - 40
        if remaining < 80:
            break
        h_start = upsample(best_h, n_new)
        time_alloc = int(min(remaining * 0.55, 280))
        alphas2 = [10, 30, 80, 150, 300]
        h_try, c5_try = optimize_h(h_start, n_new, alphas2, max_iter=100, time_limit=time_alloc)

        # Local improvement at new resolution
        elapsed2 = time.time() - t_start
        if budget_s - elapsed2 > 50:
            h_try, c5_try = local_improve(h_try, n_new, n_steps=2000)

        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    # Phase 3: Try alternate initializations at parent resolution
    elapsed = time.time() - t_start
    if budget_s - elapsed > 120:
        rng = np.random.default_rng(seed + 1)
        for trial in range(3):
            elapsed = time.time() - t_start
            if budget_s - elapsed < 40:
                break
            # Create perturbed version of best solution
            pert = best_h.copy()
            mask = rng.random(n0) < 0.3
            pert[mask] += rng.uniform(-0.3, 0.3, mask.sum())
            pert = enforce_sum(pert, n0)
            alphas3 = [50, 150, 300]
            h_try, c5_try = optimize_h(pert, n0, alphas3, max_iter=100, time_limit=60)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try

    # Phase 4: Final polish
    elapsed = time.time() - t_start
    remaining = budget_s - elapsed - 5
    if remaining > 20:
        n_final = len(best_h)
        alphas4 = [300, 500, 800]
        h_try, c5_try = optimize_h(best_h.copy(), n_final, alphas4, max_iter=100, time_limit=int(remaining * 0.7))
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    # Final constraint enforcement
    n_final = len(best_h)
    best_h = enforce_sum(best_h, n_final)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n_final)
