import numpy as np
import time

def project_to_polytope(h, target_sum):
    """Project h onto {x: 0<=x[i]<=1, sum(x)=target_sum}"""
    n = len(h)
    # Bisection on lambda: x_i = clip(h_i - lambda, 0, 1), find lambda so sum = target_sum
    lo, hi = h.min() - target_sum, h.max()  # rough bounds
    for _ in range(100):
        mid = (lo + hi) / 2
        x = np.clip(h - mid, 0, 1)
        if x.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - (lo+hi)/2, 0, 1)
    return x
