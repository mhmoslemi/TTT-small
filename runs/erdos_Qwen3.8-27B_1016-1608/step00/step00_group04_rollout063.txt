    dx = 2.0 / n
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    """Project onto {0<=x<=1, sum(x)=target_sum} via bisection on Lagrange mult"""
    lo, hi = -10.0, 10.0
    for _ in range(80):
        mid = (lo + hi) / 2
        trial = np.clip(h + mid, 0.0, 1.0)
        if np.sum(trial) < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2, 0.0, 1.0)

def subgradient(h, k_star):
    """Compute subgradient of f_{k_star} at h"""
    n = len(h)
    k = k_star - (n - 1)  # actual shift
    grad = np.zeros(n)
    for j in range(n):
        g = 0.0
        if 0 <= j + k < n:
            g = 1.0 - h[j + k]
        if 0 <= j - k < n:
            g -= h[j - k]
        grad[j] = g
    return grad
