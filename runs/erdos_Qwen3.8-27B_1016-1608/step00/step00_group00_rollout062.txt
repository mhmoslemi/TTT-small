```python
import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def smooth_c5(h, T=0.01):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    max_c = np.max(c)
    if max_c < 1e-15:
        return max_c  # or 0
    shifted = (c - max_c) / T
    log_sum = np.log(np.sum(np.exp(shifted)))
    return max_c + T * log_sum

def smooth_c5_grad(h, T=0.01, eps=1e-7):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    max_c = np.max(c)
    shifted = (c - max_c) / T
    p = np.exp(shifted - np.max(shifted))
    p /= p.sum()
    
    # Numerical gradient
    grad = np.zeros(n)
    for j in range(n):
        h_plus = h.copy()
        h_minus = h.copy()
        h_plus[j] = min(1.0, h[j] + eps)
        h_minus[j] = max(0.0, h[j] - eps)
        actual_eps = h_plus[j] - h_minus[j]
        fp = smooth_c5(h_plus, T)
        fm = smooth_c5(h_minus, T)
        grad[j] = (fp - fm) / actual_eps
    return grad
```

```python
def projected_gradient_descent(h, n, T, lr_schedule, max_iter, time_limit):
    start = time.time()
    best = h.copy()
    best_val = compute_c5(h)
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        # Compute smoothed gradient
        grad = smooth_c5_grad(h, T)
        
        # Projected gradient step
        lr = lr_schedule(it)
        h_new = h - lr * grad
        h_new = project_onto_feasible(h_new, n)
        
        # Line search (backtracking)
        val = smooth_c5(h_new, T)
        if val > smooth_c5(h, T) + 1e-4 * np.dot(grad, h_new - h):
            # Reduce step
            for alpha in [0.5, 0.25, 0.1, 0.05, 0.01]:
                h_try = h - lr * alpha * grad
                h_try = project_onto_feasible(h_try, n)
                val_try = smooth_c5(h_try, T)
                if val_try < smooth_c5(h, T):
                    h_new = h_try
                    break
            else:
                h_new = h
        h = h_new
        
        c5 = compute_c5(h)
        if c5 < best_val:
            best_val = c5
            best = h.copy()
    
    return best, best_val
```

```python
import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = np.max(c)
    if m < 1e-12:
        return m
    e = np.exp((c - m) / T)
    return m + T * np.log(np.sum(e))

def project_feasible(h, target_sum):
    """Project h onto {x: 0<=x<=1, sum(x)=target_sum}"""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0, 1))
        if s < target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def optimize_at_resolution(h0, n, T_list, max_time):
    """Optimize h at a given resolution using projected gradient with line search."""
    start = time.time()
    best_h = h0.copy()
    best_val = compute_c5(h0)
    target_sum = n / 2.0
    
    for T in T_list:
        h = best_h.copy()
        f = smooth_obj(h, T)
        
        # Gradient via central differences
        eps = 1e-6
        grad = np.zeros(n)
        for j in range(n):
            hp = h.copy(); hp[j] = min(1.0, h[j] + eps)
            hm = h.copy(); hm[j] = max(0.0, h[j] - eps)
            grad[j] = (smooth_obj(hp, T) - smooth_obj(hm, T)) / (hp[j] - hm[j])
        
        max_iter = 200
        for it in range(max_iter):
            if time.time() - start > max_time:
                break
            
            # Gradient step
            lr = 1.0 / (it + 10) * 0.1
            direction = -grad
            h_new = project_feasible(h + lr * direction, target_sum)
            
            # Backtracking line search
            f_new = smooth_obj(h_new, T)
            if f_new > f:
                for alpha in [0.5, 0.25, 0.1, 0.05, 0.01, 0.005]:
                    h_try = project_feasible(h + lr * alpha * direction, target_sum)
                    f_try = smooth_obj(h_try, T)
                    if f_try < f:
                        h_new = h_try
                        f_new = f_try
                        break
                else:
                    break
            
            h = h_new
            f = f_new
            
            # Update gradient
            grad = np.zeros(n)
            for j in range(n):
                hp = h.copy(); hp[j] = min(1.0, h[j] + eps)
                hm = h.copy(); hm[j] = max(0.0, h[j] - eps)
                grad[j] = (smooth_obj(hp, T) - smooth_obj(hm, T)) / (hp[j] - hm[j])
            
            c5 = compute_c5(h)
            if c5 < best_val:
                best_val = c5
                best_h = h.copy()
            
            if np.max(np.abs(grad)) < 1e-8:
                break
    
    return best_h, best_val
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    if m < 1e-15:
        return m
```
```python
    return m + T * np.log(np.sum(np.exp((c - m) / T)))

def grad_smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    c_full = np.correlate(h, 1 - h, mode='full')
    c_scaled = c_full * dx
    max_c = c_scaled.max()
    p = np.exp((c_scaled - max_c) / T)
    p /= p.sum()
    
    grad = np.zeros(n)
    for j in range(n):
        A_j = np.dot(p[j:n], (1-h)[n-1-j::-1])
        B_j = np.dot(p[n-1-j:n], h[0:j+1])
        grad[j] = dx * (A_j - B_j)
    return grad
```
