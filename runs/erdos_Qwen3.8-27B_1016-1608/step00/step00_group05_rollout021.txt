    lo2 = max(0, -lag)
    hi2 = min(n, n - lag)
    if lo2 < hi2:
        grad[lo2:hi2] -= h[lo2+lag:hi2+lag]
    
    return grad

            h0 = best_h + rng.randn(n) * 0.01
            h0 = project_sum(h0, n)
            stuck = 0
            lr = 0.003

    # Phase 3: Multi-restart perturbation
    for rs in range(60):
        if elapsed() > budget_s * 0.88:
            break
        noise = max(0.003, 0.06 * (1.0 - rs / 60.0))
        h_t = best_h + rng.randn(n) * noise
        h_t = project_sum(h_t, n)
        lr3 = 0.002
        for it in range(40000):
            if elapsed() > budget_s * 0.92:
                break
            corr = np.correlate(h_t, 1.0 - h_t, mode="full")
            k = int(np.argmax(corr))
            L = k - (n - 1)
            g = grad_lag(h_t, n, L)
            h_t = project_sum(h_t - lr3 * g, n)
            c5_t = compute_c5(h_t)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()

    # Phase 4: Fine coordinate refinement
    if elapsed() < budget_s * 0.97:
        hw = best_h.copy()
        for _ in range(500000):
            if elapsed() > budget_s * 0.97:
                break
            i = rng.randint(n)
            delta = rng.uniform(-0.008, 0.008)
            hw[i] = np.clip(hw[i] + delta, 0.0, 1.0)
            diff = n / 2.0 - hw.sum()
            if abs(diff) > 1e-12:
                j = rng.randint(n)
                while j == i:
                    j = rng.randint(n)
                hw[j] = np.clip(hw[j] + diff, 0.0, 1.0)
                hw = project_sum(hw, n)
            c5_t = compute_c5(hw)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = hw.copy()

    best_h = project_sum(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
