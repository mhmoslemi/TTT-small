import numpy as np
from scipy.optimize import minimize
from scipy import correlate as sp_corr
import time

def compute_c5(h, dx):
    corr = sp_corr.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_sum(h, target):
    lo = -float(np.max(h)) - 0.01
    hi = 1.0 - float(np.min(h)) + 0.01
    for _ in range(35):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def nm_opt(h0, n, dx, ts, maxiter=3000):
    def obj(p):
        return compute_c5(project_sum(np.clip(p, 0.0, 1.0), ts), dx)

    except Exception:
