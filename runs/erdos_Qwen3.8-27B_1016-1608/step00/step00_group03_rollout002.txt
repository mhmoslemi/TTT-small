    c = np.correlate(h, 1.0 - h, mode='full')

        # (1-h[j+l]) for j where 0<=j+l<n, i.e., j in [max(0,-l), min(n-1, n-1-l)]
        lo1 = max(0, -l)
        hi1 = min(n, n - l)  # exclusive
        if lo1 < hi1:

        hi2 = min(n, n + l)  # exclusive
        if lo2 < hi2:
            grad[lo2:hi2] -= h[lo2-l:hi2-l]
    
    grad /= len(active) if active else 1
    return max_val * dx, grad * dx

import numpy as np
from scipy.optimize import minimize
import time

def proj_onto_box_hyp(h, target_sum):
    """Project h onto {x: 0<=x<=1, sum(x)=target_sum} via bisection on lambda."""
    lo, hi = -1.0, 2.0  # range for lambda
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def c8_value(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return np.max(c) * dx

def c5_grad(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(c)
    tol = 1e-10
    active = np.where(np.abs(c - max_val) < tol)[0]
    
    grad = np.zeros(n)
    for m in active:
        l = int(m - (n - 1))
        # (1-h[j+l]) for valid j
        lo1 = max(0, -l)
        hi1 = min(n, n - l)
        if lo1 < hi1:
            grad[lo1:hi1] += (1.0 - h[lo1+l:hi1+l])
        # -h[j-l] for valid j
        lo2 = max(0, l)
        hi2 = min(n, n + l)
        if lo2 < hi2:
            grad[lo2:hi2] -= h[lo2-l:hi2-l]
    
    if len(active) > 0:
        grad /= len(active)
    return max_val * dx, grad * dx

    best_c = c5_value(h)

            if alpha < 1e-12:

        if np.max(np.abs(g)) < 1e-10:

    return best_h, best_c

def slsqp_optimize(h0, max_iter=500):

    target_sum = n / 2.0
    
    def obj_and_grad(x):
        f, g = c5_grad(x)
        return f, g

                    'jac': lambda x: np.ones(n)}]

import numpy as np
from scipy.optimize import minimize
import time


def _proj_box_hyp(h, target):
    """Project h onto {x: 0<=x<=1, sum(x)=target} via bisection."""
    lo, hi = -2.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)


def _c5_val(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx


def _c5_grad(h):
    """Return (objective, gradient) for C5 = max_k cor(h, 1-h)[k]*dx."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    mx = float(np.max(c))
    active = np.where(np.abs(c - mx) < 1e-12)[0]
    grad = np.zeros(n)
    for m in active:
        l = int(m - (n - 1))
        # (1-h[j+l]) for 0<=j+l<n
        a, b = max(0, -l), min(n, n - l)
        if a < b:
            grad[a:b] += (1.0 - h[a + l:b + l])
        # -h[j-l] for 0<=j-l<n
        a2, b2 = max(0, l), min(n, n + l)
        if a2 < b2:
            grad[a2:b2] -= h[a2 - l:b2 - l]
    if len(active) > 0:
        grad /= len(active)
    return mx * dx, grad * dx


def _proj_grad(h0, max_iter, time_fn):
    """Projected gradient with backtracking line search."""
    n = len(h0)
    tgt = n / 2.0
    h = _proj_box_hyp(h0.copy(), tgt)
    best_c = _c5_val(h)
    best_h = h.copy()
    alpha = 0.5
    for _ in range(max_iter):
        if time_fn():
            break
        f, g = _c5_grad(h)
        if np.max(np.abs(g)) < 1e-12:
            break
        for _ in range(40):
            h_new = _proj_box_hyp(h - alpha * g, tgt)
            f_new = _c5_val(h_new)
            if f_new <= f - 1e-4 * alpha * np.dot(g, h_new - h):
                h = h_new
                alpha = min(alpha * 1.3, 2.0)
                if h_new is not h:
                    pass
                f_cur = _c5_val(h)
                if f_cur < best_c:
                    best_c = f_cur
                    best_h = h.copy()
                break
            alpha *= 0.5
        else:
            alpha = max(alpha, 1e-8)
            if alpha < 1e-12:
                break
    return best_h, best_c


def _slsqp(h0, max_iter=300):
    n = len(h0)
    tgt = n / 2.0

    def _obj(x):
        return _c5_grad(x)[0]

    def _jac(x):
        return _c5_grad(x)[1]

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - tgt,
             'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_obj, h0, jac=_jac, method='SLSQP',
                       constraints=cons, bounds=bnds,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        hx = _proj_box_hyp(np.clip(res.x, 0, 1), tgt)
        return hx, _c5_val(hx)
    except Exception:
        return h0, _c5_val(h0)


def _binary_local_search(h_init, max_iter, rng, time_fn):
    """Local search on binary vectors with 2-flip moves."""
    n = len(h_init)
    tgt = n // 2
    # Round to binary
    h = (h_init >= 0.5).astype(float)
    # Ensure exactly tgt ones
    ones = np.where(h == 1)[0]
    if len(ones) > tgt:
        for idx in ones[tgt:]:
            h[idx] = 0
    elif len(ones) < tgt:
        zeros = np.where(h == 0)[0]
        for idx in zeros[:tgt - len(ones)]:
            h[idx] = 1
    best_c = _c5_val(h)
    best_h = h.copy()
    temp = 0.05
    for _ in range(max_iter):
        if time_fn():
            break
        i, j = rng.integers(0, n, 2)
        if h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        c_new = _c5_val(h)
        delta = c_new - best_c
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
            best_h, best_c = h.copy(), c_new
        else:
            h[i], h[j] = h[j], h[i]
        temp *= 0.999
    return _proj_box_hyp(best_h, n / 2.0), _c5_val(best_h)


def _random_feasible(n, rng):
    h = rng.uniform(0, 1, n)
    return _proj_box_hyp(h, n / 2.0)


def _structured_start(n, rng):
    """Create diverse structured starting points."""
    style = rng.integers(0, 4)
    h = np.zeros(n)
    tgt = n / 2.0
    if style == 0:
        # Block pattern with random boundary
        split = rng.integers(n // 4, 3 * n // 4)
        h[:split] = 1.0
        h[split:] = 0.0
    elif style == 1:
        # Alternating blocks
        block = rng.integers(2, max(3, n // 10) + 1)
        h = np.tile(np.array([1.0] * block + [0.0] * block), n // (2 * block) + 1)[:n]
    elif style == 2:
        # Sine-like
        t = np.linspace(0, 2 * np.pi * rng.integers(2, 5), n)
        h = 0.5 + 0.4 * np.sin(t + rng.uniform(0, 6.28))
    else:
        h = rng.uniform(0, 1, n)
    return _proj_box_hyp(h, tgt)

    t0 = time.time()
    rng = np.random.default_rng(seed)

    # Get initial values
    if 'initial_h_values' in globals() and initial_h_values is not None:

    else:
        h_init = np.ones(70) / 2.0
    n = len(h_init)

    h_init = _proj_box_hyp(h_init, n / 2.0)

    best_c = _c5_val(best_h)

    # Phase 1: Projected gradient from initial
    h_p, c_p = _proj_grad(h_init, 3000, exceeded)

    # Phase 2: SLSQP polish
    if not exceeded():

        if c_s < best_c:

    # Phase 3: More projected gradient

        if c_p2 < best_c:

        if not exceeded():
            h_s2, c_s2 = _slsqp(best_h, 400)
            if c_s2 < best_c:

    # Phase 4: Binary local search from current best
    if not exceeded():
        h_b, c_b = _binary_local_search(best_h, 20000, rng, exceeded)
        # Smooth the binary result back

        # Polish continuous

        h_r = _random_feasible(n, rng)
        h_r, c_r = _proj_grad(h_r, 1500, exceeded)
