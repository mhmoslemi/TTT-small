from scipy.optimize import minimize

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    target = n / 2.0
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        s = np.sum(h)
        if abs(s - target) < 1e-13:
            break
        diff = target - s
        free = (h > 1e-9) & (h < 1.0 - 1e-9)
        nf = int(np.sum(free))
        if nf > 0:
            h[free] += diff / nf
        else:
            h += diff / n
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) > 1e-9:
        diff = target - s
        idx = int(np.argmin(np.abs(h - 0.5)))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h


def make_obj(n):
    dx = 2.0 / n
    def obj(h):
        h = np.clip(h, 0.0, 1.0)
        c = np.correlate(h, 1.0 - h, mode='full')
        return float(np.max(c) * dx)
    return obj


def make_eq(n):
    t = n / 2.0

        return np.sum(h) - t
    return eq

                h_try[j] += delta
                # spread correction across others

                h_try[others] -= delta / (n - 1)

                if val < best - 1e-10:
                    best = val
                    h = h_try
                    improved = True
                    break
        if not improved:
            break

                    0.05 + 0.2 * rng.random(nblocks))
    h = np.zeros(n)
    for b in range(nblocks):
        mask = (x >= edges[b]) & (x < edges[b + 1])
        h[mask] = vals[b]
    return project_feasible(h, n)
