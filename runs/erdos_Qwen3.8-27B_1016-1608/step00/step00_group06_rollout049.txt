import numpy as np
from scipy.optimize import minimize, minimize_neldermead
import time

def compute_score(h):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode="full")
    return np.max(corr) * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    current = np.sum(h)
    diff = target - current
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h < 1.0
            cnt = mask.sum()
            if cnt > 0:
                h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = mask.sum()
            if cnt > 0:
                h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def smooth_max_corr(h, n, beta):
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode="full") * dx
    m = np.max(corr)
    shifted = corr - m
    return m + np.log(np.sum(np.exp(beta * shifted))) / beta

def raw_max_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def constraint_eq(h):
    return np.sum(h) - 0.5 * len(h)

def constraint_grad(h):
    return np.ones(len(h)) / len(h)

def make_bounds(n):
    return [(0.0, 1.0)] * n

def make_cons():
    return [{'type': 'eq', 'fun': constraint_eq, 'jac': constraint_grad}]

def interpolate_parent(h_parent, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, h_parent)

def try_optimize(h0, n_pts, t0, budget_s, max_frac):
    bounds = make_bounds(n_pts)
    cons = make_cons()
    best_local = h0.copy()
    best_s = compute_score(h0)
    for beta in [60, 120, 250]:
        if time.time() - t0 > budget_s * max_frac:
            break
        try:
            res = minimize(smooth_max_corr, h0, args=(n_pts, beta),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 350, 'ftol': 1e-12})
            h_opt = project_feasible(res.x, n_pts)
            s = compute_score(h_opt)
            if s < best_s:
                best_s = s
                best_local = h_opt.copy()
        except Exception:
            pass
    if time.time() - t0 < budget_s * max_frac:
        try:
            res = minimize(raw_max_corr, best_local,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 200, 'ftol': 1e-12})
            h_opt = project_feasible(res.x, n_pts)
            s = compute_score(h_opt)
            if s < best_s:
                best_s = s
                best_local = h_opt.copy()
        except Exception:
            pass
    return best_local, best_s
