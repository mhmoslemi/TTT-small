    """Project h to satisfy sum(h) = n/2."""
    h = np.clip(h, 0.0, 1.0)

import numpy as np
from scipy import optimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    """Project h to {0<=h<=1, sum(h)=n/2} via bisection on shift."""
    h = np.array(h, dtype=float)
    lo, hi = -1.0, 1.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < n / 2.0:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-14:
            break
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_c5(h, n, beta=30.0):
    """Smooth approximation of C5 using log-sum-exp."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    return float((m + np.log(np.sum(np.exp((corr - m) * beta))) / beta) * dx)

def generate_initial(n, rng):
    """Generate a valid initial h with structure."""
    h = np.zeros(n)
    half = n // 2
    # Create near-binary: alternate 1s and 0s, adjust for exact sum
    h[np.arange(0, n, 2)] = 1.0
    s = np.sum(h)
    target = n / 2.0
    if s > target:
        # Reduce some 1s
        idx = np.where(h == 1.0)[0]
        excess = s - target
        for j in idx:
            red = min(1.0, excess)
            h[j] -= red
            excess -= red
            if excess < 1e-12:
                break
    elif s < target:
        idx = np.where(h == 0.0)[0]
        deficit = target - s
        for j in idx:
            add = min(1.0, deficit)
            h[j] += add
            deficit -= add
            if deficit < 1e-12:
                break
    return project_feasible(h, n)
