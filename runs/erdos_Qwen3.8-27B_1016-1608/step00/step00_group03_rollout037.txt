import time

    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def c5_of(h):
    return float(np.max(corr_vals(h)))

def smooth_max(vals, beta):
    m = float(np.max(vals))
    e = np.exp(beta * (vals - m))
    return m + float(np.log(np.sum(e))) / beta

def make_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        d = float(np.sum(h)) - target
        if abs(d) < 1e-14:
            break
        mask = h > 1e-12 if d > 0 else h < 1.0 - 1e-12
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            break
        h[mask] -= d / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h

def mcmc_phase(cur, cur_c5, best, best_c5, n, t0, t_end, T_start, cool, rng, step_frac=1.0):

    while time.time() < t_end:
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        scale = 0.05 * step_frac
        eps_lo = max(-cur[i], -(1.0 - cur[j]))
        eps_hi = min(1.0 - cur[i], cur[j])
        lo = max(eps_lo, -scale)
        hi = min(eps_hi, scale)

            continue

        if abs(eps) < 1e-12:

        cur[j] -= eps
        new_c5 = c5_of(cur)
        if new_c5 <= cur_c5 or (T > 1e-15 and rng.random() < np.exp(-(new_c5 - cur_c5) / T)):

            if new_c5 < best_c5:

                best_c5 = new_c5
        else:
            cur[i] -= eps
            cur[j] += eps
        T = max(T * cool, 1e-14)
        if rng.random() < 5e-5:
            T = max(T, 2 * T_start * 0.1)
    return cur, cur_c5, best, best_c5
