Let me also think about a smarter local search. Instead of random pair moves, I can:
1. Compute all correlation values
2. Identify the worst lag k*
3. For that lag, compute the "gradient" - which coordinates contribute most to the worst term
4. Propose moves that specifically reduce the worst term

For lag k, the correlation is:
c_k = sum_i h[i] * (1 - h[i - k + n - 1])

The contribution of h[i] to c_k (for valid i) is: (1 - h[i-k+n-1]) if we increase h[i], and -h[i] if we increase h[i-k+n-1].

So to reduce c_k, we want to:
- Decrease h[i] where (1-h[i-k+n-1]) is large (i.e., h[i-k+n-1] is small)
- Decrease h[i-k+n-1] where h[i] is large

This gives a directional local search!

```python
def directional_search(h, k_worst):
    """Propose moves to reduce correlation at lag k_worst."""
    n = len(h)
    # For lag k_worst, valid i are in [max(0, k_worst-n+1), min(n-1, k_worst)]
    i_lo = max(0, k_worst - n + 1)
    i_hi = min(n - 1, k_worst)
    
    moves = []
    for i in range(i_lo, i_hi + 1):
        j = i - k_worst + n - 1  # the paired index
        # Increasing h[i] increases c_k by (1-h[j])
        # Increasing h[j] increases c_k by (-h[i])
        # So to decrease c_k: decrease h[i] (if h[i]>0) or decrease h[j] (if h[j]>0)
        if h[i] > 0.01:
            moves.append((i, j, -0.05, 0.05))  # decrease h[i], increase h[j]
        if h[j] > 0.01:
            moves.append((j, i, -0.05, 0.05))  # decrease h[j], increase h[i]
    return moves
```

import numpy as np
import time

def eval_c5(h, dx):
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def repair(h, ts):
    h = np.clip(h, 0.0, 1.0).copy()
    diff = ts - h.sum()
    if abs(diff) > 1e-10:
        m = (h < 1.0) if diff > 0 else (h > 0.0)
        c = m.sum()
        if c > 0:
            h[m] += diff / c
    h = np.clip(h, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) > 1e-10:
        for i in range(len(h)):
            if diff > 0 and h[i] < 1.0:
                h[i] = min(h[i] + diff, 1.0)
                break
            elif diff < 0 and h[i] > 0.0:
                h[i] = max(h[i] + diff, 0.0)
                break
    return h

def full_sweep(h, n, dx, deadline, passes=20):
    h = h.copy()
    cur = eval_c5(h, dx)
    deltas_all = [0.2, -0.2, 0.1, -0.1, 0.05, -0.05, 0.02, -0.02, 0.01, -0.01]
    for p in range(passes):
        if time.time() > deadline:
            break
        if p < 4:
            deltas = deltas_all
        elif p < 8:
            deltas = deltas_all[2:]
        else:
            deltas = deltas_all[4:]
        imp = False
        for i in range(n):
            if time.time() > deadline:
                return h, cur
            for j in range(i + 1, n):
                for d in deltas:
                    hi = h[i] + d
                    hj = h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i] = hi
                        h[j] = hj
                        v = eval_c5(h, dx)
                        if v < cur - 1e-13:
                            cur = v
                            imp = True
                        else:
                            h[i] -= d
                            h[j] += d
        if not imp:
            break
    return h, cur

def random_sweep(h, n, dx, ts, rng, deadline, n_trials=5000):
    h = h.copy()
    cur = eval_c5(h, dx)
    step = 0.12
    for _ in range(n_trials):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n - 1)
        if j >= i:
            j += 1
        d = rng.uniform(-step, step)
        hi = h[i] + d
        hj = h[j] - d
        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
            h[i] = hi
            h[j] = hj
            v = eval_c5(h, dx)
            if v < cur - 1e-12:
                cur = v
            else:
                h[i] -= d
                h[j] += d
    return h, cur

def make_binary(n, ts, rng):
    h = np.zeros(n)
    o = int(ts)
    f = ts - o
    p = rng.permutation(n)
    h[p[:o]] = 1.0
    if f > 1e-9:
        h[p[o]] = f
    return h

    rng = np.random.RandomState(seed)
    t0 = time.time()
    dl = t0 + budget_s - 8.0
    bC = float('inf')
    bH = None

    try:

    # Phase 2: Try random binary-like starts at various sizes
