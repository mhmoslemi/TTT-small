import numpy as np
from scipy.optimize import minimize

_BETA = 50.0
_N_PTS = 45

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_obj(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    if m < 1e-300:
        return 0.0
    return m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta

def _wrapped_obj(x):
    return smooth_obj(x, _BETA)

def _con_fun(x):
    return np.sum(x) - _N_PTS / 2.0

def _con_jac(x):
    return np.ones(_N_PTS)

def fix_feasibility(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    global _BETA, _N_PTS
    import time
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    # Get initial values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.full(45, 0.5)

    n = len(h0)
    h0 = fix_feasibility(h0, n)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    # Helper to run SLSQP with current globals
    def slsqp_opt(h_init, n_pts, beta, maxiter, ftol=1e-10):
        global _BETA, _N_PTS
        _BETA = beta
        _N_PTS = n_pts
        bounds = [(0.0, 1.0)] * n_pts
        cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
        res = minimize(_wrapped_obj, h_init, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return res

    # Phase 1: Progressive beta annealing from initial
    for beta in [20, 40, 80, 160, 320]:
        if elapsed() > budget_s * 0.45:
            break
        res = slsqp_opt(best_h, n, beta, 500)
        if res.success:
            h_new = fix_feasibility(res.x, n)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h = h_new
                best_c5 = c5

    # Phase 2: Perturbation-based restarts
    for i in range(15):
        if elapsed() > budget_s * 0.65:
            break
        scale = 0.03 + 0.02 * rng.rand()
        noise = rng.randn(n) * scale
        h_pert = fix_feasibility(best_h + noise, n)
        beta_r = 60 + 40 * rng.rand() * 8
        res = slsqp_opt(h_pert, n, beta_r, 350)
        if res.success:
            h_new = fix_feasibility(res.x, n)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h = h_new
                best_c5 = c5

    # Phase 3: Try random structured initializations at current n
    for i in range(6):
        if elapsed() > budget_s * 0.78:
            break
        # Construct random-ish binary-like pattern
        h_rand = (rng.rand(n) < 0.5).astype(float)
        # Add small continuous component
        h_rand += rng.rand(n) * 0.1
        h_rand = fix_feasibility(h_rand, n)
        res = slsqp_opt(h_rand, n, 80, 400)
        if res.success:
            h_new = fix_feasibility(res.x, n)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h = h_new
                best_c5 = c5

    # Phase 4: Try higher resolutions
    for n_new in [60, 90, 120, 180]:
        if elapsed() > budget_s * 0.92:
            break
        h_up = np.interp(
            np.linspace(0.0, 2.0, n_new, endpoint=False),
            np.linspace(0.0, 2.0, len(best_h), endpoint=False),
            best_h
        )
        h_up = fix_feasibility(h_up, n_new)
        for beta in [60, 120]:
            if elapsed() > budget_s * 0.92:
                break
            res = slsqp_opt(h_up, n_new, beta, 350)
            if res.success:
                h_new = fix_feasibility(res.x, n_new)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new
                    best_c5 = c5

    # Phase 5: Final polish with high-beta smooth objective
    if elapsed() < budget_s * 0.96:
        n_f = len(best_h)
        _BETA = 500.0
        _N_PTS = n_f
        bounds = [(0.0, 1.0)] * n_f
        cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
        res = minimize(_wrapped_obj, best_h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': 400, 'ftol': 1e-12})
        if res.fun < best_c5:
            h_final = fix_feasibility(res.x, n_f)
            best_h = h_final
            best_c5 = compute_c5(best_h)

    # Ensure final feasibility
    best_h = fix_feasibility(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
