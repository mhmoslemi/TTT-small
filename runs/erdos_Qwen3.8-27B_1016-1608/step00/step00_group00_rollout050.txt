def compute_gradient_at_lag(h, L):
    n = len(h)
    grad = np.zeros(n)
    # (1-h[j-L]) for valid j: j-L in [0,n) means j in [L, n+L) ∩ [0,n)
    if L >= 0:
        # j-L in [0, n-1] means j in [L, n-1] + L... 
        # j ranges from L to n-1 (so j-L ranges from 0 to n-1-L)
        grad[L:2*L] += 0  # placeholder
        # Actually: for j in [max(0,L), min(n, n+L)):
        # if L >= 0: j in [L, n-1] (j-L in [0, n-1-L])
        # contribution: 1 - h[j-L]
        if L < n:
            grad[L:] += (1.0 - h[:n-L])
        # -h[j+L] for valid j: j+L in [0,n) means j in [-L, n-1-L]
        # if L >= 0: j in [0, n-1-L]
        if L < n:
            grad[:n-L] -= h[L:]
    else:
        M = -L  # positive
        # j-L = j+M in [0,n-1] means j in [0, n-1-M]
        grad[:n-M] += (1.0 - h[M:])
        # j+L = j-M in [0,n-1] means j in [M, n-1]
        grad[M:] -= h[:n-M]
    return grad

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr)

def compute_all_corr(h):

def grad_at_lag(h, L):
    n = len(h)

    if L >= 0:
        if L < n:
            grad[L:] += (1.0 - h[:n - L])
            grad[:n - L] -= h[L:]
    else:
