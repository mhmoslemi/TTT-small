    best_h = np.clip(best_h, 0.0, 1.0)
    n_final = len(best_h)
    target = n_final / 2.0
    diff = np.sum(best_h) - target
    if abs(diff) > 1e-10:
        # Binary repair
        best_h = (best_h > 0.5).astype(float)
        cnt = int(np.sum(best_h))
        if cnt > target:
            ones = np.where(best_h == 1)[0]
            rng.shuffle(ones)
            for k in range(cnt - target):
                best_h[ones[k]] = 0.0
        elif cnt < target:
            zeros = np.where(best_h == 0)[0]
            rng.shuffle(zeros)
            for k in range(target - cnt):
                best_h[zeros[k]] = 1.0

    final_score = compute_score(best_h)
    return (best_h, final_score, len(best_h))
