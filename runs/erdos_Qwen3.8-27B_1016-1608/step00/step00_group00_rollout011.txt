```python
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def upsample(h, new_n):
    """Upsample h from its current grid to new_n points using linear interpolation."""
    old_n = len(h)
    x_old = np.linspace(0, 2, old_n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    # Renormalize
    target = new_n / 2.0
    h_new = np.clip(h_new, 0, 1)
    # Adjust sum
    diff = target - np.sum(h_new)
    interior = (h_new > 0.01) & (h_new < 0.99)
    if np.sum(interior) > 0:
        h_new[interior] += diff / np.sum(interior)
        h_new = np.clip(h_new, 0, 1)
    return h_new

def project_feasible(h, n):
    """Project onto {0<=h<=1, sum(h)=n/2}."""
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2.0, 0.0, 1.0)

def subgradient_step(h, lr):
    """One subgradient step on the max-overlap objective."""
    n = len(h)
    dx = 2.0 / n
    overlaps = np.correlate(h, 1.0 - h, mode='full') * dx
    k_max = int(np.argmax(overlaps))
    s = k_max - (n - 1)  # shift
    
    # Gradient of f_s(h) = dx * sum_i h[i]*(1-h[i-s])
    grad = np.zeros(n)
    # Vectorized computation
    # For j where 0 <= j-s < n: grad[j] += (1-h[j-s])
    if s >= 0:
        # j-s in [0,n) => j in [s, s+n), intersect [0,n) = [s, n)
        grad[s:] += 1.0 - h[:n-s]
    else:
        # j-s in [0,n) => j in [0, |s|+n), intersect [0,n) = [0, n+|s|) ∩ [0,n) = [0, n)

        grad[:n+s] += 1.0 - h[-s:]

        grad[:n-s] -= h[s:]
    else:

        grad[-s:] -= h[:n-s]  # h[j+s] for j in [-s, n) is h[0:n-s]

    m = np.max(x)
```

```python
def compute_subgradient(h):
    n = len(h)
    dx = 2.0 / n
    overlaps = np.correlate(h, 1.0 - h, mode='full') * dx
    k_max = int(np.argmax(overlaps))
    s = k_max - (n - 1)
    
    grad = np.zeros(n)
    
    # First term: (1-h[j-s]) for j where 0<=j<n and 0<=j-s<n
    if s >= 0:
        # j in [s, n), h[j-s] = h[0:n-s]
        grad[s:] += 1.0 - h[:n-s]
    else:
        # j in [0, n+|s|) ∩ [0,n) = [0, n), but j-s >= 0 means j >= |s|
        # Wait: j-s in [0,n) and j in [0,n). s < 0, so j-s = j+|s|.
        # j+|s| in [0,n) means j in [0, n-|s|)
        # j in [0,n) and j in [0, n-|s|) → j in [0, n-|s|)
        abs_s = -s
        grad[:n - abs_s] += 1.0 - h[abs_s:]
    
    # Second term: -h[j+s] for j where 0<=j<n and 0<=j+s<n
    if s >= 0:
        # j+s in [0,n) means j in [0, n-s)
        grad[:n-s] -= h[s:]
    else:
        # j+s in [0,n), s<0, j+s = j-|s|, j-|s| in [0,n) means j in [|s|, n)
        abs_s = -s
        grad[abs_s:] -= h[:n- abs_s]
    
    return grad * dx, k_max
```

```python
        grad[:n-a] += 1.0 - h[a:]
        grad[a:] -= h[:n-a]

        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2.0, 0.0, 1.0)

    best_c5 = compute_c5(h)
    lr = lr_init
    for it in range(n_iters):
        grad, _ = compute_subgrad(h)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
        lr = max(0.0005, lr * lr_decay)
```

Actually, for this type of problem (minimizing a max of smooth functions with box + affine constraints), a very effective method is:

**Proximal Gradient / FISTA with smooth max approximation**

The smoothed objective: F_β(h) = (1/β) * log(Σ_k exp(β * f_k(h)))

This is convex (since it's a max of convex... wait, are the f_k convex? We showed they're not. So F_β is not necessarily convex either.)

But F_β is still smooth, and its gradient is:
∇F_β(h) = Σ_k p_k * ∇f_k(h) where p_k = exp(β*f_k) / Σ_j exp(β*f_j)

This weighted average gradient is always available and can be computed efficiently.

For non-convex smooth optimization with constraints, I can use:
- L-BFGS-B with the smooth objective (ignoring the equality constraint, using only box constraints)
- But I need to enforce sum(h) = n/2

Alternative: Reparameterize. Let h_i = σ(z_i) where σ is a sigmoid, and optimize over z. But this makes the sum constraint harder.

Better: Use the penalty method. Minimize F_β(h) + μ * (sum(h) - n/2)² with box constraints.

Or: Use scipy.optimize.minimize with method='SLSQP' with the equality constraint.

For n=45, SLSQP can handle 45 variables with one equality constraint fine.
For n=180, it might be slower but still doable.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_subgrad(h):
    n = len(h)
    dx = 2.0 / n
    overlaps = np.correlate(h, 1.0 - h, mode='full') * dx
    k_max = int(np.argmax(overlaps))
    s = k_max - (n - 1)
    grad = np.zeros(n)
    if s >= 0:
        grad[s:] += 1.0 - h[:n - s]
        grad[:n - s] -= h[s:]
    else:
        a = -s
        grad[:n - a] += 1.0 - h[a:]
        grad[a:] -= h[:n - a]
    return grad * dx, float(overlaps[k_max]), k_max

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_grad_hess(h, beta):
    n = len(h)
    dx = 2.0 / n
    overlaps = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(overlaps)
    exps = np.exp(overlaps - m)
    weights = exps / np.sum(exps)
    val = m + np.log(np.sum(exps)) / beta
    grad = np.zeros(n)
    for k in range(len(overlaps)):
        w = weights[k]
        if w < 1e-12:
            continue
        s = k - (n - 1)
        g = np.zeros(n)
        if s >= 0:
            g[s:] += 1.0 - h[:n - s]
            g[:n - s] -= h[s:]
        else:
            a = -s
            g[:n - a] += 1.0 - h[a:]
            g[a:] -= h[:n - a]
        grad += w * g * dx
    return val, grad

def subgrad_descent(h, n, n_iters, lr_init, lr_decay, use_bb=True):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h_prev = h.copy()
    lr = lr_init
    for it in range(n_iters):
        grad, fval, _ = compute_subgrad(h)
        if use_bb and it > 0:
            s_vec = h - h_prev
            y_vec = np.zeros_like(h)
            g_prev, _, _ = compute_subgrad(h_prev)
            y_vec = grad - g_prev
            sy = np.dot(s_vec, y_vec)
            yy = np.dot(y_vec, y_vec)
            if sy > 1e-10 and yy > 1e-10:
                lr_bb = np.dot(s_vec, s_vec) / sy
                lr = np.clip(lr_bb, 0.001, 0.1)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h_prev = h.copy()
        h = h_new
        lr = max(0.001, lr * lr_decay)
    return best_h, best_c5

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.arange(old_n) * 2.0 / old_n
    x_new = np.arange(new_n) * 2.0 / new_n
    h_new = np.interp(x_new, x_old, h, period=2.0)
    return project_feasible(h_new, new_n)

def random_perturb(h, n, sigma):
    return project_feasible(h + np.random.randn(n) * sigma, n)

def scipy_polish(h, n, budget_left):
    if budget_left < 10:
        return h, compute_c5(h)
    beta = 30.0
    target = n / 2.0

    def obj(z):
        hh = project_feasible(z, n)
        val, _ = smooth_grad_hess(hh, beta)
        return val

    def obj_grad(z):
        hh = project_feasible(z, n)
        val, g = smooth_grad_hess(hh, beta)
        return val, g

    res = minimize(obj_grad, h, method='L-BFGS-B',
                   bounds=[(0, 1)] * n,
                   options={'maxiter': 200, 'ftol': 1e-10})
    h_pol = project_feasible(res.x, n)
    c5_pol = compute_c5(h_pol)
    c5_orig = compute_c5(h)
    if c5_pol < c5_orig:
        return h_pol, c5_pol
    return h, c5_orig

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        h = np.full(90, 0.5)
    n = len(h)
    h = project_feasible(h, n)

    best_h = h.copy()
    best_c5 = compute_c5(h)

    # Phase 1: Subgradient on initial grid with multiple restarts
    for restart in range(4):
        if time.time() - t0 > budget_s * 0.25:
            break
        start = best_h.copy() if restart > 0 else h
        if restart > 0:
            start = random_perturb(best_h, n, 0.03)
        h_cur, c5_cur = subgrad_descent(start, n, 8000, 0.015, 0.9998)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur

    # Phase 2: Multi-resolution refinement
    for factor in [2, 4]:
        if time.time() - t0 > budget_s * 0.55:
            break
        new_n = min(best_h.shape[0] * factor, 400)
        if new_n <= best_h.shape[0]:
            continue
        h_hi = upsample(best_h, new_n)
        h_cur, c5_cur = subgrad_descent(h_hi, new_n, 10000, 0.008, 0.99985)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur

    # Phase 3: More subgradient with random restarts at working resolution
    nw = len(best_h)
    while time.time() - t0 < budget_s * 0.78:
        sigma = 0.02 + 0.03 * np.random.rand()
        h_rand = random_perturb(best_h, nw, sigma)
        h_cur, c5_cur = subgrad_descent(h_rand, nw, 4000, 0.01, 0.9997)
        if c5_cur < best_c5 - 1e-6:
            best_c5 = c5_cur
            best_h = h_cur

    # Phase 4: Scipy polish
    if time.time() - t0 < budget_s * 0.90:
        remaining = budget_s * 0.92 - (time.time() - t0)
        h_pol, c5_pol = scipy_polish(best_h, len(best_h), remaining)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol

    # Final subgradient polish
    if time.time() - t0 < budget_s * 0.97:
        h_final, c5_final = subgrad_descent(best_h, len(best_h), 3000, 0.005, 0.9999)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```
