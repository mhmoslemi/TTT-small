    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:

            if room < 1e-14:

        else:

            if avail < 1e-14:

            h = h + diff * h / avail
        h = np.clip(h, 0.0, 1.0)
    return h

        m = n - s
        val = float(np.dot(h[:m], 1.0 - h[s:]))

import numpy as np
from scipy.optimize import minimize
from functools import partial
import time


def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx


def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = (1.0 - h).sum()
            if room < 1e-15:
                break
            h = h + diff * (1.0 - h) / room
        else:
            avail = h.sum()
            if avail < 1e-15:
                break
            h = h + diff * h / avail
        h = np.clip(h, 0.0, 1.0)
    return h


def corr_grad_single(h, k, n):
    s = k - (n - 1)
    if s >= 0:
        m = n - s
        val = float(np.dot(h[:m], 1.0 - h[s:]))
        g = np.zeros(n)
        g[:m] = 1.0 - h[s:]
        g[s:] -= h[:m]
    else:
        s2 = -s
        m = n - s2
        val = float(np.dot(h[s2:], 1.0 - h[:m]))
        g = np.zeros(n)
        g[s2:] = 1.0 - h[:m]
        g[:m] -= h[s2:]
    return val, g


def f_max(x, n, dx):
    h = np.clip(x, 0, 1)
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(c.max())


def g_max(x, n, dx):
    h = np.clip(x, 0, 1)
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    k = int(np.argmax(c))
    _, g = corr_grad_single(h, k, n)
    return g * dx


def f_eq(x, target):
    return float(x.sum() - target)


def g_eq(x, n):
    return np.ones(n)


def f_smooth(x, n, dx, tau, target, pen):
    h = np.clip(x, 0, 1)
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    mx = c.max()
    s = c - mx
    logsum = float(np.log(np.sum(np.exp(tau * s))))
    f = (mx + logsum / tau) + pen * (h.sum() - target) ** 2
    return f


def g_smooth(x, n, dx, tau, target, pen):
    h = np.clip(x, 0, 1)
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    mx = c.max()
    s = c - mx
    w = np.exp(tau * s)
    w = w / w.sum()
    g = np.zeros(n)

            _, gk = corr_grad_single(h, k, n)
            g += w[k] * gk
    g = g * dx
    g += 2.0 * pen * (h.sum() - target)
    return g
