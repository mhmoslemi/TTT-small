    corr = np.correlate(h, 1.0 - h, mode="full")
    shifted = corr - np.max(corr)

    weights /= weights.sum()

import numpy as np
import time

def _project(h, target):
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        val = np.sum(np.clip(h - mid, 0.0, 1.0))
        if val > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _grad_hard(h, dx):
    nl = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    k = int(np.argmax(corr))
    j = np.arange(nl)
    idx1 = nl - 1 - k + j
    idx2 = j + k - (nl - 1)
    g = np.zeros(nl)
    m1 = (idx1 >= 0) & (idx1 < nl)
    m2 = (idx2 >= 0) & (idx2 < nl)
    g[m1] += (1.0 - h[idx1[m1]])
    g[m2] -= h[idx2[m2]]
    return g * dx

def _grad_soft(h, dx, eps=4.0):
    nl = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    shifted = corr - np.max(corr)
    weights = np.exp(shifted / eps)
    weights /= weights.sum()
    g = np.zeros(nl)
    for k in range(2 * nl - 1):
        wk = weights[k]
        if wk < 1e-8:
            continue
        j = np.arange(nl)
        idx1 = nl - 1 - k + j
        idx2 = j + k - (nl - 1)
        m1 = (idx1 >= 0) & (idx1 < nl)
        m2 = (idx2 >= 0) & (idx2 < nl)
        g[m1] += wk * (1.0 - h[idx1[m1]])
        g[m2] -= wk * h[idx2[m2]]
    return g * dx

def _optimize(h, dx, target, grad_fn, max_iters, deadline):
    nl = len(h)
    result = _project(h, target)
    v = np.zeros(nl)
    best_obj = _c5(result, dx)
    best_h = result.copy()
    lr = 0.02
    mu = 0.85
    patience = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        g = grad_fn(result, dx)
        v = mu * v + lr * g
        cand = _project(result + v, target)
        obj = _c5(cand, dx)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_h = cand
            result = cand
            v = 0.3 * v
            patience = 0
            lr = min(lr * 1.01, 0.15)
        else:
            result = _project(result + 0.1 * v, target)
            patience += 1
            lr *= 0.98
            mu *= 0.999
            if patience > 400:
                break
    return best_h, best_obj

def _pair_local_search(h, dx, target, deadline):
    nl = len(h)
    result = h.copy()
    best_obj = _c5(result, dx)
    improved = True
    while improved and time.time() < deadline:
        improved = False
        for i in range(0, nl, 3):
            for j in range(i + 2, nl, 3):
                for delta in (0.02, 0.05, 0.1, -0.02, -0.05, -0.1):
                    if result[i] + delta < -0.01 or result[i] + delta > 1.01:
                        continue
                    if result[j] - delta < -0.01 or result[j] - delta > 1.01:
                        continue
                    cand = result.copy()
                    cand[i] += delta
                    cand[j] -= delta
                    cand = _project(cand, target)
                    obj = _c5(cand, dx)
                    if obj < best_obj - 1e-10:
                        best_obj = obj
                        result = cand
                        improved = True
                        break
    return result, best_obj

            h_try[idx] += 0.1 * np.random.randn(len(idx))
        h_try = _project(h_try, target)
        h_try, c5_try = _optimize(h_try, dx, target, _grad_hard, 1500, deadline)
        if c5_try < best_c5:
