def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def repair_h(x, target_sum):
    x = np.clip(x.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(x)
    if abs(diff) > 1e-12:
        mask = (x > 1e-6) & (x < 1.0 - 1e-6)
        if np.sum(mask) > 0:
            x[mask] += diff / np.sum(mask)
            x = np.clip(x, 0.0, 1.0)
    return x

def obj_max_corr(x, dx):
    corr = np.correlate(x, 1.0 - x, mode="full")
    return np.max(corr) * dx

def obj_smooth_max(x, dx, alpha):
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    if m < -1e10:
        return 0.0
    s = np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha

def sum_constraint(x, target_sum):
    return np.sum(x) - target_sum
