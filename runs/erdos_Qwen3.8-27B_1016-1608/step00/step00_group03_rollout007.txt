import numpy as np
from scipy.optimize import minimize
import time

def _corr_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return np.max(c) * dx

def _make_valid(h, n, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    n = 70
    dx = 2.0 / n
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    
    best_h = np.array(initial_h_values, dtype=float)
    best_h = _make_valid(best_h, n, target_sum)
    best_c5 = _corr_c5(best_h, dx)
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    
    def obj(x):
        return _corr_c5(x, dx)
    
    rng = np.random.default_rng(seed)
    n_restarts = 500
    max_iter = 500
    
    for r in range(n_restarts):
        if time.time() - t0 > budget_s - 20:
            break
        
        if r == 0:
            x0 = best_h.copy()
        elif r == 1:
            x0 = _make_valid(rng.random(n), n, target_sum)
        elif r % 3 == 2:
            # Perturb best solution
            x0 = best_h + 0.1 * rng.standard_normal(n)
            x0 = _make_valid(x0, n, target_sum)
        else:
            x0 = _make_valid(rng.random(n), n, target_sum)
        
        try:
            res = minimize(obj, x0, method='SLSQP', bounds=bounds,
                          constraints=cons,
                          options={'maxiter': max_iter, 'ftol': 1e-12})
            h_new = _make_valid(res.x, n, target_sum)
            c5_new = _corr_c5(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        except Exception:
            pass
    
    return (best_h, best_c5, n)

    best_c5 = _c5(best_h, dx)

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]

    T_sm = 25.0

    def obj_smooth(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = np.max(c)
        e = T_sm * (c - m)
        return m + (1.0 / T_sm) * np.log(np.sum(np.exp(e)))

    def obj_raw(x):
        return _c5(x, dx)

    rng = np.random.default_rng(seed)
    time_limit = budget_s - 15.0
    phase1_end = t0 + (budget_s - 15.0) * 0.6

    def try_optimize(x0, f, maxiter):
        try:
            res = minimize(f, x0, method='SLSQP', bounds=bounds,
                          constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-12})
            h_new = _make_valid(res.x, n, target_sum)
            return h_new
        except Exception:
            return None

    for r in range(50000):
        if time.time() - t0 > time_limit:
            break

        if time.time() < phase1_end:
            # Phase 1: exploration
            if r == 0:
                x0 = best_h.copy()
                f = obj_raw
            elif r == 1:
                x0 = _make_valid(rng.random(n), n, target_sum)
                f = obj_smooth
            elif r % 7 == 2:
                x0 = _make_valid(best_h + 0.05 * rng.standard_normal(n), n, target_sum)
                f = obj_raw
            elif r % 7 == 3:
                x0 = _make_valid(best_h + 0.2 * rng.standard_normal(n), n, target_sum)
                f = obj_smooth
            elif r % 7 == 4:
                x0 = _make_valid(rng.random(n), n, target_sum)
                f = obj_smooth
            elif r % 7 == 5:
                x0 = (rng.random(n) > 0.5).astype(float)
                x0 = _make_valid(x0, n, target_sum)
                f = obj_smooth
            elif r % 7 == 6:
                x0 = 0.5 + 0.5 * np.sign(np.sin(np.linspace(0, rng.uniform(1, 4) * np.pi, n)))
                x0 = _make_valid(x0, n, target_sum)
                f = obj_smooth
            else:
                x0 = _make_valid(rng.random(n), n, target_sum)
                f = obj_smooth
            mi = 150
        else:
            # Phase 2: exploitation / polish
            if r % 3 == 0:
                x0 = _make_valid(best_h + 0.01 * rng.standard_normal(n), n, target_sum)
            elif r % 3 == 1:
                x0 = _make_valid(best_h + 0.05 * rng.standard_normal(n), n, target_sum)
            else:
                x0 = best_h.copy()
            f = obj_raw
            mi = 300

        h_new = try_optimize(x0, f, mi)
        if h_new is not None:
            c5_new = _c5(h_new, dx)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Final polish
    if time.time() - t0 < time_limit - 5:
        h_new = try_optimize(best_h.copy(), obj_raw, 500)
        if h_new is not None:
            c5_new = _c5(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
