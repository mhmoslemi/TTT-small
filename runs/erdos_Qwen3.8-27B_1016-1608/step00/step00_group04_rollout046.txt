```python
import numpy as np
import time

def compute_score(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return c.max() * dx

def local_search_binary(h, max_iters=100000):
    """Local search on binary vectors maintaining sum constraint."""
    n = len(h)
    h = h.copy()
    current_score = compute_score(h)
    ones = set(np.where(h > 0.5)[0])
    zeros = set(np.where(h < 0.5)[0])
    
    for _ in range(max_iters):
        improved = False
        # Try all single swaps: move a 1 from position a to position b
        for a in list(ones):
            for b in list(zeros):
                h[a], h[b] = 0, 1
                s = compute_score(h)
                if s < current_score - 1e-12:
                    current_score = s
                    ones.remove(a)
                    ones.add(b)
                    zeros.add(a)
                    zeros.remove(b)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return h, current_score

def local_search_continuous(h, max_iters=50000, step=0.1):
    """Local search on continuous [0,1] vectors maintaining sum constraint."""
    n = len(h)
    h = h.copy()
    current_score = compute_score(h)
    
    for _ in range(max_iters):
        improved = False
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        delta = step * (1 if np.random.rand() < 0.5 else -1)
        new_i = h[i] + delta
        new_j = h[j] - delta
        if 0 <= new_i <= 1 and 0 <= new_j <= 1:
            h[i], h[j] = new_i, new_j
            s = compute_score(h)
            if s < current_score:
                current_score = s
                improved = True
            else:
                h[i], h[j] = new_i - delta, new_j + delta
        if not improved and _ % 1000 == 0:
            step *= 0.95
    return h, current_score
```

```python
def project_binary(h, n_target_sum):
    """Project h to a binary vector with exactly n_target_sum ones."""
    n = len(h)
    h_bin = (h >= 0.5).astype(float)
    s = int(h_bin.sum())
    if s == n_target_sum:
        return h_bin
    if s > n_target_sum:
        # Need to turn (s - n_target_sum) ones to zeros
        ones_idx = np.where(h_bin == 1)[0]
        # Turn off those with smallest h value (closest to 0.5 from below)
        order = np.argsort(h[ones_idx])
        for i in range(s - n_target_sum):
            h_bin[ones_idx[order[i]]] = 0
    else:

        order = np.argsort(-h[zeros_idx])
        for i in range(n_target_sum - s):
```

```python
        if m >= 0:
            # f = sum_{i=0}^{n-1-m} h[i+m]*(1-h[i])
            # To decrease: increase h[i] where h[i+m]=1, or decrease h[i+m] where h[i]=1
            # Or increase h[i+m] where h[i]=0 (reduces 1-h[i+m]... no)
            # Actually d(f)/dh[i] for i in 0..n-1-m: (1-h[i+m]) if i is the first index
            #                            + h[0] if i >= m (as h[i-m] * (1-h[i]))
            # Let's just pick random valid pairs weighted toward active region
            lo = max(0, 0)
            hi = n - abs(m)
        else:
            lo = 0
            hi = n + m  # n - |m|
        
        i = int(rng.integers(max(hi - lo, 1))) + lo
        j = int(rng.integers(n))
        if i == j:
            continue
        d = step * (1.0 if rng.random() < 0.5 else -1.0)
        si, sj = h[i], h[j]
        h[i] = np.clip(si + d, 0.0, 1.0)
        h[j] = np.clip(sj - d, 0.0, 1.0)
        s = score(h)
        if s < cur - 1e-15:
            cur = s
            stall = 0
        else:
            h[i], h[j] = si, sj
            stall += 1
        if stall > 5000:
            step *= 0.8
            stall = 0
            if step < 5e-5:
                break
    return h, cur

def random_binary(n, target, rng):
    idx = rng.permutation(n)
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    return h

def fix_sum(h, target):
    """Adjust h so sum equals target, keeping in [0,1]."""
    h = h.copy()
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # Distribute among interior points
    interior = np.where((h > 1e-6) & (h < 1.0 - 1e-6))[0]
    if len(interior) == 0:
        # Use all points
        interior = np.arange(len(h))
    per = diff / len(interior)
    h[interior] = np.clip(h[interior] + per, 0.0, 1.0)
    # Final correction on one point
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        # Find a point with room
        for i in range(len(h)):
            if residual > 0 and h[i] < 1.0:
                add = min(residual, 1.0 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 0.0:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8.0
    rng = np.random.default_rng(seed)

    h0 = initial_h_values.copy().astype(float)
    n0 = len(h0)
    target0 = int(round(h0.sum()))
    if target0 * 2 != n0:
        target0 = n0 // 2

    best_h = h0.copy()
    best_s = score(best_h)

    # Phase 1: Binary local search from initial
    h_b = to_binary(h0, target0)
    h_b, s_b = binary_climb(h_b, rng, max_sweeps=8000, deadline=deadline)
    if s_b < best_s:
        best_s, best_h = s_b, h_b.copy()

    # Phase 2: Continuous + targeted refinement
    if time.time() < deadline - 60:
        h_c, s_c = continuous_climb(best_h, rng, max_iters=200000, deadline=deadline)
        if s_c < best_s:
            best_s, best_h = s_c, h_c.copy()
    if time.time() < deadline - 30:
        h_t, s_t = targeted_climb(best_h, rng, max_iters=100000, deadline=deadline)
        if s_t < best_s:
            best_s, best_h = s_t, h_t.copy()

    # Phase 3: Random restarts (binary + refine)
    n_restarts = 40
    for r in range(n_restarts):
        if time.time() > deadline - 10:
            break
        h_r = random_binary(n0, target0, rng)
        h_r, s_r = binary_climb(h_r, rng, max_sweeps=3000, deadline=deadline)
        if s_r < best_s:
            best_s, best_h = s_r, h_r.copy()
            if time.time() < deadline - 20:
                h_c, s_c = continuous_climb(best_h, rng, max_iters=80000, deadline=deadline)
                if s_c < best_s:
                    best_s, best_h = s_c, h_c.copy()

    # Phase 4: Try structured starting points
    for pattern in range(3):
        if time.time() > deadline - 15:
            break
        if pattern == 0:
            # Block pattern: groups of 3 alternating
            h_p = np.zeros(n0)
            for i in range(n0):
                h_p[i] = 1.0 if (i // 3) % 2 == 0 else 0.0
        elif pattern == 1:
            # Golden ratio sprinkling
            phi = (1 + 5**0.5) / 2
            h_p = np.zeros(n0)
            for i in range(target0):
                pos = int(i * phi) % n0
                h_p[pos] = 1.0
        else:
            # Low-discrepancy
            h_p = np.zeros(n0)
            for i in range(target0):
                pos = int(round(i * n0 / target0 + 0.5 * (i % 2))) % n0
                h_p[pos] = 1.0
        
        target_p = int(h_p.sum())
        if target_p != target0:
            h_p = to_binary(h_p, target0)
        h_p, s_p = binary_climb(h_p, rng, max_sweeps=4000, deadline=deadline)
        if s_p < best_s:
            best_s, best_h = s_p, h_p.copy()
            if time.time() < deadline - 20:
                h_c, s_c = continuous_climb(best_h, rng, max_iters=60000, deadline=deadline)
                if s_c < best_s:
                    best_s, best_h = s_c, h_c.copy()

    # Phase 5: Higher resolution attempt
    for n_try in [n0 * 2, n0 * 3]:
        if n_try > 500 or time.time() > deadline - 30:
            break
        target_t = n_try // 2
        # Upsample best
        x_old = np.linspace(0, 2.0, n0)
        x_new = np.linspace(0, 2.0, n_try)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = np.clip(h_up, 0.0, 1.0)
        h_up = fix_sum(h_up, target_t)
        h_u, s_u = continuous_climb(h_up, rng, max_iters=60000, deadline=deadline)
        if s_u < best_s:
            best_s, best_h = s_u, h_u.copy()

    # Final: fix sum exactly and validate
    n_f = len(best_h)
    target_f = n_f // 2
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = fix_sum(best_h, target_f)
    final_s = score(best_h)
    if final_s < best_s:
        best_s = final_s

    return best_h, best_s, n_f
```
