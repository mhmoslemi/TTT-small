    lo, hi = z.min() - 2.0, z.max() + 2.0

    """Compute gradient of max_k correlate(h, 1-h, 'full')[k] * dx"""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    k_star = np.argmax(c)
    offset = k_star - (n - 1)  # range: -(n-1) to (n-1)

    lo2 = max(0, -offset)
    hi2 = min(n, n - offset)
    if lo2 < hi2:

    return grad * dx, k_star, c

        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    w = np.exp(alpha * (c - c.max()))
    w /= w.sum()
    n_half = n - 1
    grad = np.zeros(n)
    for ki in range(len(c)):
        wi = w[ki]
        if wi < 1e-10:
            continue
        off = ki - n_half
        lo1 = max(0, off)
        hi1 = min(n, n + off)
        if lo1 < hi1:
            grad[lo1:hi1] += wi * (1.0 - h[lo1 - off:hi1 - off])
        lo2 = max(0, -off)
        hi2 = min(n, n - off)
        if lo2 < hi2:
            grad[lo2:hi2] -= wi * h[lo2 + off:hi2 + off]
    return grad * dx

def sharp_grad(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    k = int(np.argmax(c))
    off = k - (n - 1)
    grad = np.zeros(n)
    lo1 = max(0, off)
    hi1 = min(n, n + off)
    if lo1 < hi1:
        grad[lo1:hi1] += (1.0 - h[lo1 - off:hi1 - off])
    lo2 = max(0, -off)
    hi2 = min(n, n - off)
    if lo2 < hi2:
        grad[lo2:hi2] -= h[lo2 + off:hi2 + off]
    return grad * dx, k

def optimize(h0, t_limit, start_time):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mu = 0.85
    v = np.zeros(n)
    lr = 0.015
    alpha = 15.0
    it_count = 0
    no_improve = 0
    while time.time() - start_time < t_limit:
        if no_improve > 500:
            break
        a = alpha * (1.0 + 0.05 * min(it_count, 3000) / 10.0)
        if it_count > 400:
            g = smooth_grad(h, a)
        else:
            g = smooth_grad(h, a)
        v = mu * v + (1.0 - mu) * g
        h_new = h - lr * v
        h_proj = project_feasible(h_new, n)
        c5_new = compute_c5(h_proj)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_proj.copy()
            no_improve = 0
            lr = min(lr * 1.03, 0.06)
        else:
            no_improve += 1
            lr *= 0.85
        h = h_proj
        it_count += 1
        lr *= 0.999
    return best_h, best_c5

def interpolate_to(src, new_n):
    n_old = len(src)
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, src)

def balanced_init(n, seed_val):
    m = n // 2
    h = np.zeros(n)
    for i in range(m):
        pos = int((i + 0.5) * n / m)
        pos = min(pos, n - 1)
        h[pos] = 1.0
    if n % 2 == 1:
        h[np.sum(h == 0.0) // 2 if np.sum(h == 0) > 0 else 0] = 0.5
        h = project_feasible(h, n)
    return project_feasible(h, n)

def round_binary(h, n):
    h_r = np.zeros(n)
    target = n / 2.0
    if n % 2 == 0:
        idx = np.argsort(h)[::-1]
        h_r[idx[:n // 2]] = 1.0
    else:
        idx = np.argsort(h)[::-1]
        h_r[idx[:(n - 1) // 2]] = 1.0
        h_r[idx[(n - 1) // 2]] = 0.5
    return h_r

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        parent = np.array(initial_h_values, dtype=float).copy()
    except:
        parent = balanced_init(95, seed)

    parent = project_feasible(parent, len(parent))
    best_h = parent.copy()
    best_c5 = compute_c5(parent)

    # Try multiple sizes
    sizes = [len(parent)]
    for s in [64, 80, 100, 128, 150, 192, 256]:
        if s != len(parent) and s >= 40:
            sizes.append(s)
    sizes = sorted(set(sizes))

    # Optimize at each size
    for n_pts in sizes:
        elapsed = time.time() - t0
        remaining = budget_s - elapsed
        if remaining < 20:
            break
        alloc = min(remaining * 0.35, 180.0)

        h0 = interpolate_to(parent, n_pts)
        h0 = project_feasible(h0, n_pts)

        h_opt, c5_opt = optimize(h0, alloc, time.time())
        c5_check = compute_c5(h_opt)
        if c5_check < best_c5:
            best_c5 = c5_check
            best_h = h_opt.copy()

    # Final polish on best
    elapsed = time.time() - t0
    remaining = budget_s - elapsed
    if remaining > 20:
        n_best = len(best_h)
        h_final, c5_final = optimize(best_h, remaining - 5, time.time())
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final

    # Try sharp gradient polish
    elapsed = time.time() - t0
    if time.time() - t0 < budget_s - 10:
        n_best = len(best_h)
        h = best_h.copy()
        for _ in range(200):
            if time.time() - t0 > budget_s - 3:
                break
            g, k = sharp_grad(h)
            h_new = h - 0.005 * g
            h_proj = project_feasible(h_new, n_best)
            c5_new = compute_c5(h_proj)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_proj.copy()
            h = h_proj

    # Try binary rounding
    n_best = len(best_h)
    h_bin = round_binary(best_h, n_best)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_h = h_bin
        best_c5 = c5_bin
        # Polish the binary solution
        if time.time() - t0 < budget_s - 5:
            h_pol, c5_pol = optimize(best_h, min(30.0, budget_s - time.time() - t0 - 3), time.time())
            if c5_pol < best_c5:
                best_h = h_pol
                best_c5 = c5_pol

    # Final feasibility check
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
