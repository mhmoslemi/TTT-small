import time
import math
from scipy.optimize import minimize

def c5_value(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def repair(x, n):
    target = n / 2.0
    x = np.clip(x, 0.0, 1.0)
    s = np.sum(x)
    if s < 1e-10:
        return np.full(n, 0.5)
    x = x * (target / s)
    x = np.clip(x, 0.0, 1.0)
    diff = target - np.sum(x)
    if abs(diff) > 1e-8:
        idx = np.argsort(x)
        if diff > 0:
            for j in idx:
                add = min(diff, 1.0 - x[j])
                x[j] += add
                diff -= add
                if diff < 1e-8:
                    break
        else:
            for j in reversed(idx):
                sub = min(-diff, x[j])
                x[j] -= sub
                diff += sub
                if diff > -1e-8:
                    break
    return x

result[n-1+k] = sum_i a[i+k] * v[i] for k>0 (positive lag, a shifted right)
result[n-1-k] = sum_i a[i] * v[i+k] for k>0 (a lag on v)

So for correlate(h, 1-h, 'full'):

- Right of center by k: sum h[i+k]*(1-h[i]) = sum h[j]*(1-h[j-k]) for j=i+k
- Left of center by k: sum h[i]*(1-h[i+k])

The max of all these values, times dx, is C5.

import numpy as np
import time
import math

def c5_value(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def repair(x, n):
    target = n / 2.0
    x = np.clip(x, 0.0, 1.0)
    s = np.sum(x)
    if s < 1e-10:
        return np.full(n, 0.5)
    x = x * (target / s)
    x = np.clip(x, 0.0, 1.0)
    diff = target - np.sum(x)
    if abs(diff) > 1e-10:
        idx = np.argsort(x)
        if diff > 0:
            for j in idx:
                add = min(diff, 1.0 - x[j])
                x[j] += add
                diff -= add
                if abs(diff) < 1e-10:
                    break
        else:
            for j in reversed(idx):
                sub = min(-diff, x[j])
                x[j] -= sub
                diff += sub
                if abs(diff) < 1e-10:
                    break
    return x

def golden_init(n, seed=0):
    best_h = None
    best_c5 = 1.0
    alphas = [0.61803398875, 0.38196601125, 0.70710678118, 0.45342923972, 0.44504186791]
    for alpha in alphas:
        h = np.zeros(n)
        fracs = np.array([(i * alpha) % 1.0 for i in range(n)])
        # Take the n/2 smallest fractions as 1s
        thresh = np.sort(fracs)[n // 2 - 1]
        h[fracs <= thresh] = 1.0
        # Ensure exactly n/2 ones
        cnt = int(np.sum(h == 1))
        if cnt > n // 2:
            idx1 = np.where(h == 1)[0]
            fr1 = fracs[idx1]
            order = np.argsort(-fr1)
            for i in range(cnt - n // 2):
                h[idx1[order[i]]] = 0
        elif cnt < n // 2:
            idx0 = np.where(h == 0)[0]
            fr0 = fracs[idx0]
            order = np.argsort(fr0)
            for i in range(n // 2 - cnt):
                h[idx0[order[i]]] = 1
        h_rep = repair(h, n)
        c = c5_value(h_rep)
        if c < best_c5:
            best_c5 = c
            best_h = h_rep.copy()
    return best_h, best_c5

        new_h[i] += delta
        new_h[j] -= delta
