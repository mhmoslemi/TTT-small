            # Wait, m+k = m-d. m-d in [0,n-1] => m in [d, n-1+d], intersect [0,n-1] => m in [d, n-1]
            m_lo2, m_hi2 = d, n - 1

def compute_c5(h, n):

def project_h(h, n):

        return h

        if s_mid > target:
            lo = mid
        else:
            hi = mid

def compute_gradient(h, n, T):

    g = 1.0 - h
