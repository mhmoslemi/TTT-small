import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full")) * dx)

def repair_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
            for j in order:
                if diff <= 1e-12: break
                add = min(1.0 - h[j], diff)
                h[j] += add
                diff -= add
        else:
            order = np.argsort(h)
            for j in order:
                if diff >= -1e-12: break
                rem = min(h[j], -diff)
                h[j] -= rem
                diff += rem
    return h

def subgrad_max_overlap(h, dx):
    n = len(h)
