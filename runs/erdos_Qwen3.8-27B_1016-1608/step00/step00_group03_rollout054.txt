                    best_c5 = c5_new
            except Exception:
                pass

    # Phase 3: Differential Evolution for global exploration
    if time.time() - t0 < budget_s - 100:
        try:
            de_bounds = [(-0.3, 1.3)] * n
            de_res = differential_evolution(de_objective, de_bounds, seed=seed,
                                            maxiter=35, popsize=25,
                                            tol=1e-5, mutation=(0.3, 1.0),
                                            recombination=0.9, polish=False)
            h_de = repair(de_res.x, n)
            c5_de = c5_eval(h_de)
            if c5_de < best_c5:
                best_h = h_de
                best_c5 = c5_de
        except Exception:
            pass

    # Phase 4: Random restarts + local refinement
    for restart in range(15):
        if time.time() - t0 > budget_s - 30:
            break
        if restart == 0:
            h_start = best_h.copy()
        else:
            noise = np.random.normal(0, 0.04, n)
            h_start = repair(best_h + noise, n)
        for alpha in [200, 800]:
            if time.time() - t0 > budget_s - 30:
                break
            try:
                res = minimize(smooth_c5, h_start.copy(), args=(alpha,),
                              method='SLSQP', bounds=bounds, constraints=cons,
                              options={'maxiter': 500, 'ftol': 1e-14})
                h_new = repair(res.x, n)
                c5_new = c5_eval(h_new)
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
                    h_start = best_h.copy()
            except Exception:
                pass

    # Phase 5: Final high-precision polish
    for alpha in [500, 2000]:
        if time.time() - t0 > budget_s - 3:
            break
        try:
            res = minimize(smooth_c5, best_h.copy(), args=(alpha,),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': 5000, 'ftol': 1e-15})
            h_new = repair(res.x, n)
            c5_new = c5_eval(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
        except Exception:
            pass

    # Ensure validity of final output
    best_h = repair(best_h, n)
    best_c5 = c5_eval(best_h)

    return (best_h, best_c5, n)
