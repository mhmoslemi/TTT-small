import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = h.copy()
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - 0.5 * n
        if abs(diff) < 1e-12:
            break
        h -= diff / n
    return h

def obj_grad(params, n, beta):
    h = np.clip(0.5 + params, 0.0, 1.0)
    active = np.ones(n)
    active[params <= -0.5 + 1e-12] = 0.0
    active[params >= 0.5 - 1e-12] = 0.0
    v = 1.0 - h
    dx = 2.0 / n
    corr = np.correlate(h, v, mode="full") * dx
    m = np.max(corr)
    nc = 2 * n - 1
    if beta >= 1e6:
        idx = int(np.argmax(corr))
        l = idx - (n - 1)
        gh = np.zeros(n)
        s1, e1 = max(0, l), min(n, n + l)
        s2, e2 = max(0, -l), min(n, n - l)
        if e1 > s1:
            gh[s1:e1] = (1.0 - h[s1 - l:e1 - l])
        if e2 > s2:
            gh[s2:e2] -= h[s2 + l:e2 + l]
        return m, (gh * dx * active)
    shifted = beta * (corr - m)
    e = np.exp(np.clip(shifted, -500, 500))
    w = e / np.sum(e)
    val = m + np.log(np.sum(e)) / beta
    gh = np.zeros(n)
    for l_idx in range(nc):
        wl = w[l_idx]
        if wl < 1e-10:
            continue
        l = l_idx - (n - 1)
        s1, e1 = max(0, l), min(n, n + l)
        s2, e2 = max(0, -l), min(n, n - l)
        if e1 > s1:
            gh[s1:e1] += wl * (1.0 - h[s1 - l:e1 - l])
        if e2 > s2:
            gh[s2:e2] -= wl * h[s2 + l:e2 + l]

def slsqp_optimize(h_init, n, beta_list, maxiter=300):
    params0 = h_init - 0.5
    bounds = [(-0.5, 0.5)] * n
    cons = [{'type': 'eq', 'fun': lambda p: np.sum(p)}]
    best_h = h_init.copy()
    best_f = compute_c5(h_init)
    for beta in beta_list:
        p = best_h - 0.5
        for _ in range(2):
            res = minimize(
                lambda pp: obj_grad(pp, n, beta)[0],
                p, method='SLSQP',

                best_h = h_new.copy()
                p = h_new - 0.5

    x_old = np.arange(n_old, dtype=float)
    x_new = np.linspace(0.0, n_old - 1, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def coordinate_descent(h, n, n_sweeps=50):
    h = h.copy()
    dx = 2.0 / n
    for _ in range(n_sweeps):
        improved = False
        order = np.random.permutation(n)
        for j in order:
            old = h[j]
            for trial in [old - 0.05, old - 0.02, old + 0.02, old + 0.05]:
                if trial < 0 or trial > 1:
                    continue
                h[j] = trial
                h_adj = h.copy()
                # Renormalize
                diff = np.sum(h_adj) - 0.5 * n
                h_adj -= diff / n
                h_adj = np.clip(h_adj, 0, 1)
                c = compute_c5(h_adj)
                if c < compute_c5(h) - 1e-9:
                    improved = True
                    break
            h[j] = old
            if not improved:
                continue
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_c = compute_c5(h0)

    # Resolution ladder
    ladder = []
    for f in [1, 2, 4, 8, 16]:
        nn = n0 * f
        if nn <= 720 and nn not in ladder:
            ladder.append(nn)

    for idx, n_curr in enumerate(ladder):
        if time.time() - t0 > budget_s * 0.85:
            break
        if idx == 0:
            h_cur = h0.copy()
        else:
            n_prev = ladder[idx - 1]
            h_cur = interp_to(best_h, n_curr)

        c_cur = compute_c5(h_cur)
        if c_cur < best_c:
            best_c = c_cur
            best_h = h_cur.copy()

        # Number of optimization attempts
        time_left = budget_s * 0.85 - (time.time() - t0)
        est_time_call = 0.02 * n_curr + 1.0
        n_attempts = max(1, min(int(time_left / (est_time_call * 5)), 8 if n_curr <= 90 else 4 if n_curr <= 180 else 2))

        inits = [h_cur.copy()]
        inits.append(np.full(n_curr, 0.5))
        x_arr = np.linspace(0, 2, n_curr)
        inits.append(project_feasible(0.5 + 0.3 * np.sin(np.pi * x_arr), n_curr))
        inits.append(h_cur + rng.randn(n_curr) * 0.08)
        inits[len(inits)-1] = project_feasible(inits[len(inits)-1], n_curr)

        betas = [5, 20, 100, 500, 5000, 1e8]
        bi = 0
        for attempt in range(min(n_attempts, len(inits))):
            if time.time() - t0 > budget_s * 0.88:
                break
            h_start = inits[attempt] if attempt < len(inits) else h_cur + rng.randn(n_curr) * 0.05
            h_start = project_feasible(h_start, n_curr)
            h_res, c_res = slsqp_optimize(h_start, n_curr, betas, maxiter=200 if n_curr <= 90 else 100)
            if c_res < best_c:
                best_c = c_res
                best_h = h_res.copy()
            if c_res < c_cur:
                h_cur = h_res

        # Quick coordinate descent polish
        if time.time() - t0 < budget_s * 0.85:
            h_cd = coordinate_descent(best_h, len(best_h), n_sweeps=20)
            c_cd = compute_c5(h_cd)
            if c_cd < best_c:
                best_c = c_cd
                best_h = h_cd

    # Interpolate best to moderate resolution for final answer
    final_n = len(best_h)
    if final_n < 90:
        final_h = interp_to(best_h, 90)
        c_final = compute_c5(final_h)
        if c_final < best_c:
            best_c = c_final
            best_h = final_h
            final_n = 90

    return (best_h, best_c, final_n)
