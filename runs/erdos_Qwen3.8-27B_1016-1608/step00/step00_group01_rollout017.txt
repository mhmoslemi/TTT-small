def evaluate_c5(h_values):
    """Compute C5 for a given h_values array."""
    n = len(h_values)
    dx = 2.0 / n
    corr = np.correlate(h_values, 1 - h_values, mode="full")

def repair(h, target_sum):
    """Repair h to satisfy 0<=h[i]<=1 and sum(h)=target_sum."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):

        if abs(diff) < 1e-10:
            break
        if diff > 0:
            # Need to increase sum: add to elements < 1
            mask = h < 1.0 - 1e-12
            if not np.any(mask):
                break
            n_mask = np.sum(mask)
            step = min(diff, n_mask * 1.0 / n_mask)  # can add up to (1-h[i])

            if max_add < 1e-12:

            h[mask] = np.clip(h[mask], 0, 1)

                break
