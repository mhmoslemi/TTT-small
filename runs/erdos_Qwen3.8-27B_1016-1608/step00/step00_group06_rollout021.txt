If h_j - λ < 0 → x_j = 0 → h_j < λ
If h_j - λ > 1 → x_j = 1 → h_j > λ + 1

- If λ is very positive, all x_j = 0, sum = 0

- If sum(clip(h-λ, 0, 1)) > target: need larger λ → lo = mid
- If sum < target: need smaller λ → hi = mid

def round_and_local_search(h, max_swaps=10000, seed=42):
    n = len(h)
    target = n // 2
    # Round: pick the top n/2 values
    sorted_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:target]] = 1.0
    val = compute_c5(h_bin)
    
    # Local search: swap 0s and 1s
    ones = np.where(h_bin == 1.0)[0]
    zeros = np.where(h_bin == 0.0)[0]
    rng = np.random.default_rng(seed)
    
    for _ in range(max_swaps):
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]

        h_bin[j] = 1.0
        new_val = compute_c5(h_bin)

            zeros = np.where(h_bin == 0.0)[0]
        else:
            h_bin[i] = 1.0
            h_bin[j] = 0.0

from scipy.interpolate import interp1d

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    overlap = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(overlap) * dx)

def project_sum_box(h, target_sum):
    lo = h.min() - 2.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def overlap_grad_weighted(h, weights):
    n = len(h)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-14:
            continue
        shift = m - n + 1
        if shift >= 0:
            if n > shift:
                grad[shift:] += w * (1.0 - h[:n - shift])
                grad[:n - shift] -= w * h[shift:]
        else:
            s = -shift
            if n > s:
                grad[:n - s] += w * (1.0 - h[s:])
                grad[s:] -= w * h[:n - s]
    return grad

def soft_max_and_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    overlap = np.correlate(h, 1.0 - h, mode='full')
    max_ov = np.max(overlap)
    shifted = overlap - max_ov
    exp_sh = np.exp(tau * shifted)
    zsum = np.sum(exp_sh)
    weights = exp_sh / zsum
    value = (max_ov + np.log(zsum) / tau) * dx
    grad = overlap_grad_weighted(h, weights) * dx
    return value, grad

def optimize_with_grad(h0, n_iters, tau_min, tau_max, seed):

    target = n / 2.0

            best_h = h.copy()
        # Adaptive lr

    return best_h, best_val

    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0 / n_new

    h_new = f(x_new)
    h_new = np.clip(h_new, 0, 1)
    return h_new

def binary_and_refine(h, max_sweeps=50, seed=42):

    y_old = np.concatenate([h, [h[0]]])
    for i in range(n_old - 1, -1, -1):
        x_old[i] = i * 2.0 / n_old
    x_new = np.linspace(0, 2, n_new + 1)[:-1]
    f = interp1d(x_old, y_old, kind='cubic', fill_value='extrapolate')
    h_new = f(x_new)
    return np.clip(h_new, 0, 1)

def binary_local_search(h, max_sweeps=30, seed=42):
    n = len(h)
    target = n // 2
    if n % 2 != 0:
        return h.copy(), compute_c5(h)
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    val = compute_c5(h_bin)
    rng = np.random.RandomState(seed)
    ones = list(np.where(h_bin == 1.0)[0])
    zeros = list(np.where(h_bin == 0.0)[0])
    for sweep in range(max_sweeps):
        improved = False
        rng.shuffle(ones)
        rng.shuffle(zeros)
        for i in ones:
            for j in zeros:
                h_bin[i] = 0.0
                h_bin[j] = 1.0
                nv = compute_c5(h_bin)
                if nv < val - 1e-12:
                    val = nv
                    improved = True
                    ones.remove(i)
                    zeros.remove(j)
                    ones.append(j)
                    zeros.append(i)
                else:
                    h_bin[i] = 1.0
                    h_bin[j] = 0.0
        if not improved:
            break
    return h_bin, val

def random_init(n, seed):
    rng = np.random.RandomState(seed)
    h = rng.rand(n)
    return project_sum_box(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    best_h = None
    best_val = float('inf')

    global initial_h_values
    if initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = np.zeros(96)
        h_init[::2] = 1.0

    n_init = len(h_init)
    h_init = project_sum_box(h_init.copy(), n_init / 2.0)

    val0 = compute_c5(h_init)
    if val0 < best_val:
        best_val = val0
        best_h = h_init.copy()

    n_candidates = sorted(set([n_init, n_init + 1, n_init - 1,
                                96, 98, 100, 112, 120, 128, 144, 160, 200]))
    n_candidates = [n for n in n_candidates if 48 <= n <= 256]

    for n in n_candidates:
        elapsed = time.time() - t_start
        if elapsed > budget_s * 0.8:
            break

        h_interp = interp_to_n(h_init, n)
        h_proj = project_sum_box(h_interp, n / 2.0)

        h_opt, val_opt = optimize_grad(h_proj, n_iters=3000,
                                        tau_min=40.0, tau_max=400.0, seed=seed)

        if n % 2 == 0 and time.time() - t_start < budget_s * 0.85:
            h_bin, val_bin = binary_local_search(h_opt, max_sweeps=40, seed=seed)
            if val_bin < val_opt:
                h_opt = h_bin
                val_opt = val_bin

        if val_opt < best_val:
            best_val = val_opt
            best_h = h_opt.copy()

    # Multi-start with random inits for a few n values
    for n_rs in [n_init, n_init + 1, 128]:
        if time.time() - t_start > budget_s * 0.9:
            break
        for trial in range(3):
            if time.time() - t_start > budget_s * 0.92:
                break
            h_rand = random_init(n_rs, seed + trial * 100)
            h_rs, val_rs = optimize_grad(h_rand, n_iters=2000,
                                          tau_min=60.0, tau_max=500.0,
                                          seed=seed + trial)
            if n_rs % 2 == 0:
                h_b2, val_b2 = binary_local_search(h_rs, max_sweeps=20,
                                                    seed=seed + trial + 50)
                if val_b2 < val_rs:
                    h_rs, val_rs = h_b2, val_b2
            if val_rs < best_val:
                best_val = val_rs
                best_h = h_rs.copy()

    # Final polish
    if best_h is not None and time.time() - t_start < budget_s * 0.6:
        h_pol, val_pol = optimize_grad(best_h, n_iters=2000,
                                        tau_min=300.0, tau_max=800.0, seed=seed + 999)
        if val_pol < best_val:
            best_val = val_pol
            best_h = h_pol

    return best_h, best_val, len(best_h)
