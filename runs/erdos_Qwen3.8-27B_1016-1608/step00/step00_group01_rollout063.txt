import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max()), corr

    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(h_raw + mid, 0.0, 1.0)) > ts:
            lo = mid
        else:
            hi = mid

def grad_f(h, k):
    n = len(h)
    g = np.zeros(n)
    ak = abs(k)
    if k >= 0:
        g[k:] += 1.0 - h[:n - k]
        g[:n - k] -= h[k:]
    else:
        g[:n - ak] += 1.0 - h[ak:]
        g[ak:] -= h[:n - ak]
