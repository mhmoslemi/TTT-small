import numpy as np
import time
import random
from scipy.optimize import minimize


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def smooth_obj(h, alpha, target_sum, mu):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    sh = np.clip(alpha * (corr - m), -300, 300)
    f = (m + np.log(np.sum(np.exp(sh))) / alpha) * dx
    if mu > 0:
        d = np.sum(h) - target_sum
        f += mu * d * d
    return f


def project_sum(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(h - mid, 0, 1)) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)


def coord_descent(h, target_sum, deadline):
    n = len(h)
    h = h.copy()
    grid = np.linspace(0, 1, 21)
    for sweep in range(80000):
        if time.time() > deadline:
            break
        improved = False
        order = list(range(n))
        random.shuffle(order)
        for j in order:
            if time.time() > deadline:
                return h
            old = h[j]
            best_v, best_o = old, compute_c5(h)
            for t in grid:
                h[j] = t
                o = compute_c5(h)
                if o < best_o:
                    best_o, best_v = o, t
            if 0.05 < best_v < 0.95:
                lo_r = max(0.0, best_v - 0.05)
                hi_r = min(1.0, best_v + 0.05)
                fine = np.linspace(lo_r, hi_r, 9)
                for t in fine:
                    h[j] = t
                    o = compute_c5(h)
                    if o < best_o:
                        best_o, best_v = o, t
            h[j] = best_v
            if abs(best_v - old) > 1e-10:
                improved = True
        h = project_sum(h, target_sum)
        if not improved:
            break
    return h


def try_scipy(h, n, target_sum, alpha, mu, maxiter):
    res = minimize(
        smooth_obj, h.copy(), args=(alpha, target_sum, mu),
        method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
        options={'maxiter': maxiter, 'ftol': 1e-15, 'maxls': 100}
    )
    h_new = project_sum(np.clip(res.x, 0, 1), target_sum)
    return h_new
