import time

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair_h(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt == 0:
            h = np.full(n, 0.5)
            break
        h[mask] += diff / cnt
        np.clip(h, 0.0, 1.0, out=h)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        mask = (h > 0) & (h < 1)
        if np.any(mask):
            idxs = np.where(mask)[0]

                    break
                step = min(abs(rem), 1.0 - h[j]) if rem > 0 else min(abs(rem), h[j])
                h[j] += (1 if rem > 0 else -1) * step
                rem -= (1 if rem > 0 else -1) * step
    return h

    """Smooth approximation of max using log-sum-exp"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx

    m = np.max(corr)
    if m < -1000:

    return float(m + np.log(np.sum(np.exp(corr - m))) / tau)
