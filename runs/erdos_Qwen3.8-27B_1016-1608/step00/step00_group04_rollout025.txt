import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    target = 0.5 * n
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = float(mask.sum())
        if cnt == 0:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
        else:
            h[mask] += diff / cnt
            h[mask] = np.clip(h[mask], 0.0, 1.0)
    # Final fixup
    diff = target - h.sum()
    if abs(diff) > 0:
        mid = int(np.argmin(np.abs(h - 0.5)))
        h[mid] += diff
        h[mid] = np.clip(h[mid], 0.0, 1.0)
    return h
