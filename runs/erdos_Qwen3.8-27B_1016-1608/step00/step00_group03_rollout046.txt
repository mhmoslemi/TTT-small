import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def repair(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    if total < 1e-15:
        h = np.ones(len(h)) * 0.5
        return h
    h = h * (target_sum / total)
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = np.argsort(1.0 - h)
            for i in idx:
                add = min(1.0 - h[i], diff)
                if add <= 0:
                    break
                h[i] += add
                diff -= add
                if diff < 1e-12:
                    break
        else:
            idx = np.argsort(h)
            for i in idx:
                sub = min(h[i], -diff)
                if sub <= 0:
                    break
                h[i] -= sub
                diff += sub
                if diff > -1e-12:
                    break
    return h


def slsqp_solve(h0, n, target_sum, maxiter=800, ftol=1e-11):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
            'jac': lambda x: np.ones(n)}]
    res = minimize(lambda x: compute_c5(x), h0, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    if res.success or res.fun < compute_c5(h0):
        return repair(res.x, target_sum)
    return h0


def nm_solve(h0, n, target_sum, maxiter=3000):
