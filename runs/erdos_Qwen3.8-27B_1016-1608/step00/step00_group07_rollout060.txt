def project(h, S):
    """Project h onto {0 <= h <= 1, sum(h) = S}"""
    # h is already in [0,1] mostly, just need sum constraint
    # Use bisection
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h_proj = np.clip(h - mid, 0, 1)
        s = np.sum(h_proj)
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

import numpy as np
from scipy.optimize import minimize

def compute_c5(h, dx):
    out = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(out))

def compute_grad(h, dx, top_k=5):
    n = len(h)
    out = np.correlate(h, 1.0 - h, mode="full")
    # Use top_k for smoothing
    threshold = np.sort(out)[-top_k][0]
    # Soft weighting: use top_k values with equal weight
    idx = np.where(out >= threshold)[0]
    if len(idx) == 0:
        idx = np.array([np.argmax(out)])
    grad = np.zeros(n)
    for k in idx:
        s = int(k) - (n - 1)
        if s >= 0:
            valid = n - s
            if valid > 0:
                grad[:valid] += (1.0 - h[s:s+valid])
                grad[s:] += -h[:valid]
        else:
            ss = -s
            valid = n - ss
            if valid > 0:
                grad[ss:] += (1.0 - h[:valid])
                grad[:valid] += -h[ss:]
    grad /= len(idx)
    return grad * dx

def project_box_sum(h, S):

        s = h_p.sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(h - lam, 0.0, 1.0)

def compute_softgrad(h, dx, T=0.01):
    n = len(h)
    out = np.correlate(h, 1.0 - h, mode="full")
    # Soft max weights
    out_shifted = out - out.max()
    weights = np.exp(out_shifted / T)
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for k in range(len(out)):
        if weights[k] < 1e-10:
            continue
        s = k - (n - 1)
        if s >= 0:
            valid = n - s
            if valid > 0:
                grad[:valid] += weights[k] * (1.0 - h[s:s+valid])
                grad[s:] += -weights[k] * h[:valid]
        else:
            ss = -s

import numpy as np
import time


def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))


def _grad(h, dx, T):
    n = len(h)
    out = np.correlate(h, 1.0 - h, mode="full") * dx
    w = np.exp((out - out.max()) / T)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - n + 1
        if s >= 0:
            v = n - s
            g[:v] += wk * (1.0 - h[s:s + v])
            g[s:] -= wk * h[:v]
        else:
            v = n + s
            g[-s:] += wk * (1.0 - h[:v])
            g[:v] -= wk * h[-s:]
    return g * dx


def _proj(h, S):
    lo, hi = -1.5, 1.5
    for _ in range(80):
        m = 0.5 * (lo + hi)
        if np.clip(h - m, 0.0, 1.0).sum() > S:
            lo = m
        else:
            hi = m
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)


def _optimize(h, dx, S, rng, max_time, lr_init=1.0, T_init=0.02, mu_init=0.85):
    n = len(h)
    h = _proj(h, S)
    best_h = h.copy()
    best_c5 = _c5(h, dx)
    lr = lr_init
    mu = mu_init
    v = np.zeros(n)
    stall = 0
    t0 = time.time()

    while time.time() - t0 < max_time:
        frac = min(1.0, (time.time() - t0) / max_time)
        T = max(0.002, T_init * (1.0 - 0.85 * frac))

        g = _grad(h, dx, T)
        v = mu * v - lr * g
        h_new = _proj(h + v, S)
        c5_new = _c5(h_new, dx)

        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0 / n_new
    return np.interp(x_new, x_old, h, period=2.0)
