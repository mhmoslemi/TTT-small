        x_new = np.linspace(0.0, 2.0, n_pts, endpoint=False)
        h_work = np.interp(x_new, x_par, initial_h_values)
        h_work = project_feasible(h_work, n_pts)
        if best_n != n_pts:
            x_b = np.linspace(0.0, 2.0, best_n, endpoint=False)
            h_alt = np.interp(x_new, x_b, best_h)
            h_alt = project_feasible(h_alt, n_pts)
            if compute_score(h_alt) < compute_score(h_work):
                h_work = h_alt
        _g_target = float(n_pts) / 2.0
        bounds = [(0.0, 1.0)] * n_pts
        for alpha in [20, 50, 120, 300, 800, 2000]:
            if tl() < 20:
                break
            _g_alpha = float(alpha)
            _g_penalty = 500.0
            try:
                res = minimize(_obj, h_work.copy(), method='SLSQP',
                               bounds=bounds,
                               options={'maxiter': 300, 'ftol': 1e-12})
                hc = project_feasible(res.x, n_pts)
                sc = compute_score(hc)
                if sc < best_score:
                    best_score = sc
                    best_h = hc.copy()
                    best_n = n_pts
                h_work = hc
            except Exception:
                pass
        if tl() > 25:
            h_work, s_ls = local_search(h_work, n_pts, rng, 300)
            if s_ls < best_score:
                best_score = s_ls
                best_h = h_work.copy()
                best_n = n_pts
        if tl() > 30:
            h_bin = try_binary_round(project_feasible(h_work, n_pts), n_pts)
            s_bin = compute_score(h_bin)
            if s_bin < best_score:
                best_score = s_bin
                best_h = h_bin.copy()
                best_n = n_pts
        if tl() > 40:
            h_rand = 0.5 * np.interp(x_new, x_par, initial_h_values) + 0.5 * rng.rand(n_pts)
            h_rand = project_feasible(h_rand, n_pts)
            _g_alpha = 200.0
            _g_penalty = 500.0
            try:
                res = minimize(_obj, h_rand, method='SLSQP',
                               bounds=bounds,
                               options={'maxiter': 200, 'ftol': 1e-10})
                hc = project_feasible(res.x, n_pts)
                sc = compute_score(hc)
                if sc < best_score:
                    best_score = sc
                    best_h = hc.copy()
                    best_n = n_pts
            except Exception:
                pass

    if tl() > 20:
        n_pts = best_n
        _g_alpha = 5000.0
        _g_target = float(n_pts) / 2.0
        _g_penalty = 5000.0
        try:
            res = minimize(_obj, best_h.copy(), method='Nelder-Mead',
                           options={'maxiter': 800, 'xatol': 1e-8, 'fatol': 1e-12})
            hc = project_feasible(res.x, n_pts)
            sc = compute_score(hc)
            if sc < best_score:
                best_score = sc
                best_h = hc
                best_n = n_pts
        except Exception:
            pass

    best_h = project_feasible(best_h, best_n)
    best_score = compute_score(best_h)
    return (best_h, best_score, best_n)
