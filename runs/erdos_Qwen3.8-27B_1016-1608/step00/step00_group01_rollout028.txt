    dx = 2.0 / n

import numpy as np
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def smooth_c5(h, tau=0.01):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(c)
    return m + tau * float(np.log(np.sum(np.exp((c - m) / tau))))

def grad_smooth(h, tau=0.01):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(c)
    p = np.exp((c - m) / tau)
    p /= p.sum()
    g = np.zeros(n)
    for d in range(-(n - 1), n):
        j = d + n - 1
        w = p[j] * dx
        lo, hi = max(0, -d), min(n, n - d)
        if lo < hi:
            g[lo:hi] += w * (1.0 - h[lo + d:hi + d])
        lo2, hi2 = max(0, d), min(n, n + d)
        if lo2 < hi2:
            g[lo2:hi2] += w * (-h[lo2 - d:hi2 - d])
    return g

def project(h, tgt):
    h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) < 1e-12:
        return h
    if d > 0:
        m = h < 1.0
        if m.any():
            cap = (1.0 - h[m]).sum()
            if cap > 0:
                h[m] += (1.0 - h[m]) * min(d / cap, 1.0)
    else:
        m = h > 0.0
        if m.any():
            cap = h[m].sum()
            if cap > 0:
                h[m] -= h[m] * min(-d / cap, 1.0)
    return h

def sa_binary(h0, deadline, rng):
    n = len(h0)
    tgt = n / 2.0
    h = h0.copy()
    si = np.argsort(h)
    nz = n // 2
    h[si[:nz]] = 0.0
    h[si[nz:]] = 1.0
    if n % 2:
        h[si[nz]] = 0.5
    best = h.copy()
    bc = eval_c5(h)
    cc = bc
    T = 0.005
    T_floor = 1e-9
    reheat_T = 0.003
    while time.time() < deadline:
        i, j = rng.randint(0, n, 2)
        if h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        nc = eval_c5(h)
        if nc < cc or (T > T_floor and rng.rand() < np.exp(-(nc - cc) / T)):
            cc = nc
            if nc < bc:
                bc = nc
                best = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
        T *= 0.999995
        if T < T_floor:
            T = reheat_T * rng.uniform(0.5, 1.5)
            h = best.copy()
            cc = bc
    return best, bc

def pgd(h0, deadline, tau=0.008):
    n = len(h0)
    tgt = n / 2.0
    h = project(h0.copy(), tgt)
    best = h.copy()
    bc = eval_c5(h)
    lr = 0.04
    for _ in range(500000):
        if time.time() > deadline:
            break
        g = grad_smooth(h, tau)
        h2 = project(h - lr * g, tgt)
        nc = eval_c5(h2)
        if nc < bc:
            bc = nc
            best = h2.copy()
            lr = min(lr * 1.03, 0.15)
        else:
            lr *= 0.92
        h = h2
    return best, bc

def rand_bin(n, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n // 2]] = 1.0
    if n % 2:
        h[idx[n // 2]] = 0.5
    return h

def multi_swap_neighborhood(h, rng, n):
    """Try a few different swap strategies."""
    i, j = rng.randint(0, n, 2)
    if h[i] == h[j]:
        return i, j
    return i, j
