```python
def compute_gradient(h, n, t_temp):
    dx = 2.0 / n
    cor = np.correlate(h, 1.0 - h, mode='full')
    c = cor * dx
    
    # Softmax weights
    c_shifted = c - np.max(c)
    w = np.exp(c_shifted / t_temp)
    w /= np.sum(w)
    
    grad = np.zeros(n)
    for k in range(2*n - 1):
        if w[k] < 1e-15:
            continue
        s = k - (n - 1)
        i_lo = max(0, -s)
        i_hi = min(n - 1, n - 1 - s)
        
        # Term 1: (1-h[m-s]) for m in [i_lo+s, i_hi+s]
        m_lo1 = i_lo + s  # = max(0, -s) + s = max(s, 0)
        m_hi1 = i_hi + s  # = min(n-1, n-1-s) + s = min(n-1+s, n-1)
        # But also need 0 <= m-s < n, which is 0 <= i_lo <= i_hi (already satisfied)
        # m-s = i, so i in [i_lo, i_hi], m in [i_lo+s, i_hi+s]
        # Need 0 <= m < n: max(0, i_lo+s) <= m <= min(n-1, i_hi+s)
        m_lo1 = max(0, i_lo + s)
        m_hi1 = min(n - 1, i_hi + s)
        if m_lo1 <= m_hi1:
            # grad[m] += w[k]*dx*(1-h[m-s])
            idx = m_lo1 - s  # = i_lo (should be >= 0)
            grad[m_lo1:m_hi1+1] += w[k] * dx * (1 - h[idx:idx + (m_hi1 - m_lo1 + 1)])
        
        # Term 2: -h[m+s] for m in [i_lo, i_hi]
        # Need 0 <= m < n: max(0, i_lo) <= m <= min(n-1, i_hi)
        # And need 0 <= m+s < n
        m_lo2 = max(0, i_lo)
        m_hi2 = min(n - 1, i_hi)
        if m_lo2 <= m_hi2:
            # Also need 0 <= m+s < n
            m_lo2 = max(m_lo2, -s)
            m_hi2 = min(m_hi2, n - 1 - s)
            if m_lo2 <= m_hi2:
                idx = m_lo2 + s
                grad[m_lo2:m_hi2+1] -= w[k] * dx * h[idx:idx + (m_hi2 - m_lo2 + 1)]
    
    return grad
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_score(h):
    n = len(h)
    dx = 2.0 / n
    cor = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(cor) * dx)

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h + mid, 0.0, 1.0)
        if np.sum(proj) > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def obj_func(h_v, n, t_temp):
    dx = 2.0 / n
    cor = np.correlate(h_v, 1.0 - h_v, mode='full')
    c = cor * dx
    m = np.max(c)
    return m + t_temp * np.log(np.sum(np.exp((c - m) / t_temp)))

def con_eq_func(h_v, n):
    return np.sum(h_v) - n / 2.0

def compute_gradient(h, n, t_temp):
    dx = 2.0 / n
    cor = np.correlate(h, 1.0 - h, mode='full')
    c = cor * dx
    c_shifted = c - np.max(c)
    w = np.exp(c_shifted / t_temp)
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        return np.zeros(n)
    w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        i_lo = max(0, -s)
        i_hi = min(n - 1, n - 1 - s)
        m1_lo = max(0, i_lo + s)
        m1_hi = min(n - 1, i_hi + s)
        if m1_lo <= m1_hi:
            i_start = m1_lo - s
            cnt = m1_hi - m1_lo + 1
            grad[m1_lo:m1_hi + 1] += wk * dx * (1.0 - h[i_start:i_start + cnt])
        m2_lo = i_lo
        m2_hi = i_hi
        if m2_lo <= m2_hi:
            i_start = m2_lo + s
            cnt = m2_hi - m2_lo + 1
            grad[m2_lo:m2_hi + 1] -= wk * dx * h[i_start:i_start + cnt]
    return grad

def pg_refine(h, n, t_temp, max_iters, budget_deadline):
    h = h.copy()
    lr = 0.5
    for it in range(max_iters):
        if time.time() > budget_deadline:
            break
        g = compute_gradient(h, n, t_temp)
        h_new = h - lr * g
        h_proj = project_feasible(h_new, n)
        v_old = obj_func(h, n, t_temp)
        v_new = obj_func(h_proj, n, t_temp)
        if v_new <= v_old:
            h = h_proj
            lr = min(lr * 1.05, 5.0)
        else:
            lr *= 0.5
            if lr < 1e-6:
                break
    return h

def slsqp_optimize(h0, n, budget_deadline):
    t_temps = [0.05, 0.02, 0.01]
    h = h0.copy()
    for t_temp in t_temps:
        if time.time() > budget_deadline:
            break
        bounds = [(0.0, 1.0)] * n
        constraints = [{'type': 'eq', 'fun': con_eq_func, 'args': (n,)}]
        try:
            res = minimize(obj_func, h, method='SLSQP', bounds=bounds,
                          constraints=constraints, args=(n, t_temp),
                          options={'maxiter': 400, 'ftol': 1e-10})
            if res.x is not None:
                h = project_feasible(np.clip(res.x, 0, 1), n)
        except Exception:
            pass
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
        n_best = len(h_best)
    else:
        n_best = 95
        h_best = np.full(n_best, 0.5)

    h_best = project_feasible(h_best, n_best)
    v_best = compute_score(h_best)

    n_list = sorted(set([n_best, max(20, n_best - 25), min(200, n_best + 25),
                         max(20, n_best - 50), min(300, n_best + 50),
                         40, 60, 80, 100, 120, 150]))
    n_list = [n for n in n_list if 16 <= n <= 300]
    n_list.sort(key=lambda x: abs(x - n_best))

    for n in n_list:
        if time.time() > deadline * 0.82:
            break
        # Interpolate from best
        x_old = np.linspace(0, 2, n_best)
        x_new = np.linspace(0, 2, n)
        h_interp = project_feasible(np.interp(x_new, x_old, h_best), n)

        # Also try perturbed version
        h_perturb = project_feasible(h_interp + 0.3 * np.random.randn(n), n)

        for h_init in [h_interp, h_perturb]:
            if time.time() > deadline * 0.82:
                break
            h_opt = slsqp_optimize(h_init.copy(), n, deadline * 0.82)
            v_opt = compute_score(h_opt)
            if v_opt < v_best:
                h_best, v_best, n_best = h_opt.copy(), v_opt, n

        # Projected gradient refinement
        if time.time() > deadline * 0.88:
            break
        h_cur = h_best.copy()
        if len(h_cur) != n:
            x_old = np.linspace(0, 2, len(h_cur))
            h_cur = project_feasible(np.interp(np.linspace(0, 2, n), x_old, h_cur), n)
            n = n
        for t_temp in [0.05, 0.02, 0.01, 0.005]:
            if time.time() > deadline * 0.88:
                break
            h_cur = pg_refine(h_cur, n, t_temp, 300, deadline * 0.88)
            v_cur = compute_score(h_cur)
            if v_cur < v_best:
                h_best, v_best, n_best = h_cur.copy(), v_cur, n

    # Final rounding heuristic: push near-binary values
    if time.time() < deadline * 0.95:
        h_round = h_best.copy()
        mask0 = h_round < 0.15
        mask1 = h_round > 0.85
        h_round[mask0] = 0.0
        h_round[mask1] = 1.0
        h_round = project_feasible(h_round, n_best)
        v_round = compute_score(h_round)
        if v_round < v_best:
            h_best, v_best = h_round, v_round

    # Final refinement
    if time.time() < deadline * 0.97:
        h_final = slsqp_optimize(h_best.copy(), n_best, deadline * 0.97)
        v_final = compute_score(h_final)
        if v_final < v_best:
            h_best, v_best = h_final, v_final

    h_best = project_feasible(h_best, n_best)
    v_best = compute_score(h_best)
    return h_best, v_best, n_best
