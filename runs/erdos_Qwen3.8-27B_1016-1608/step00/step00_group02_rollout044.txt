import numpy as np
from scipy.optimize import minimize, nelder_mead
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def repair(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nf = float(np.sum(free))
        if nf < 1e-12:
            if diff > 0:
                h = np.minimum(h + diff / n, 1.0)
            else:
                h = np.maximum(h + diff / n, 0.0)
            break
        h[free] += diff / nf
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_obj(x, n, beta, lam):
    h = np.clip(x, 0.0, 1.0)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    cmax = np.max(corr)
    if cmax < 1e-14:
        val = 0.0
    else:
        shifted = beta * (corr - cmax)
        val = cmax + np.log(np.sum(np.exp(shifted))) / beta
    pen = lam * (np.sum(h) - n / 2.0) ** 2
    return val + pen

def subgradient_step(h, n, step):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    k_max = int(np.argmax(corr))
    s = k_max - (n - 1)
    grad = np.zeros(n)
    for j in range(n):
        j1 = j - s
        if 0 <= j1 < n:
            grad[j] += (1.0 - h[j1]) * dx
        j2 = j + s
        if 0 <= j2 < n:
            grad[j] -= h[j2] * dx
    h_new = h - step * grad
    return repair(h_new, n)

def coordinate_descent(h, n, max_iter=2000):
    h = h.copy()
    best_c5 = evaluate_c5(h)
    for _ in range(max_iter):
        improved = False
        for i in range(n):
            for delta in [0.01, -0.01, 0.02, -0.02, 0.05, -0.05]:
                h_try = h.copy()
                h_try[i] += delta
                h_try = repair(h_try, n)
                c5_try = evaluate_c5(h_try)
                if c5_try < best_c5:
                    h = h_try
                    best_c5 = c5_try
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5

    best_h = h0.copy()
    best_c5 = evaluate_c5(best_h)

    # Phase 1: L-BFGS-B with log-sum-exp, varying beta and lambda
    betas = [1.0, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0, 200.0]
    for beta in betas:
        if elapsed() > budget_s - 250:

        for lam in [10.0, 100.0, 500.0]:

            )
            h_new = repair(res.x, n)
