import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0, 1)
    target = n / 2.0
    s = np.sum(h)
    if s < 1e-10:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0, 1)
    s = np.sum(h)
    if abs(s - target) > 1e-6:
        diff = target - s
        if diff > 0:
            room = 1.0 - h
            total_room = room.sum()
            if total_room > 0:
                h += diff * room / total_room
            h = np.clip(h, 0, 1)
        else:
            total_avail = h.sum()
            if total_avail > 0:
                h += diff * (h / total_avail)
            h = np.clip(h, 0, 1)
    return h

def subgrad_lag(h, shift):
    n = len(h)
    g = np.zeros(n)
    if shift >= 0:
        g[:n-shift] += (1 - h[shift:])
        g[shift:] -= h[:n-shift]
    else:
        s = -shift
        g[s:] += (1 - h[:n-s])
        g[:n-s] -= h[s:]
    return g

def optimize(h0, n, alpha0=0.01, n_iters=5000, time_limit=60):
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    alpha = alpha0
    start = time.time()
    
    for it in range(n_iters):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1 - h, mode='full')
        max_val = np.max(corr)
        c5 = max_val * dx
        
        # Find active lags (top 10% or those within 1% of max)
        threshold = max_val * 0.99
        active = np.where(corr >= threshold)[0]
        
        grad = np.zeros(n)
        for k in active:
            shift = k - (n - 1)
            grad += subgrad_lag(h, shift)
        grad /= len(active)

        h_new = repair(h_new, n)

                c5_new = compute_c5(h_new, dx)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()

    h = np.clip(h, 0, 1)
    target = n / 2.0
    s = np.sum(h)
    if s < 1e-10:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0, 1)
    s = np.sum(h)

        if diff > 0:
            room = 1.0 - h
            tr = room.sum()
            if tr > 1e-10:
                h = h + diff * room / tr
                h = np.clip(h, 0, 1)
        else:
            ta = h.sum()
            if ta > 1e-10:
                h = h + diff * h / ta
                h = np.clip(h, 0, 1)
    return h

def subgrad_lag(h, shift):
    n = len(h)

            g[:n-shift] += (1.0 - h[shift:])

    return g
