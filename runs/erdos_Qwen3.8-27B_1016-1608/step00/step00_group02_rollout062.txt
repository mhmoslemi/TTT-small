import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))


def smooth_max_val(c, gamma):
    m = np.max(c)
    e = np.exp(gamma * (c - m))
    s = np.sum(e)
    if s < 1e-300:
        return m
    return m + np.log(s / len(c)) / gamma


def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(400):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
        else:
            mask = h > 1e-13
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    # Final exact fix
    diff = target - np.sum(h)
    if abs(diff) > 0:
        idx = np.argsort(h) if diff > 0 else np.argsort(-h)
        i = 0
        while abs(diff) > 1e-12 and i < n:
            j = idx[i]
            room = (1.0 - h[j]) if diff > 0 else h[j]
            adj = min(room, abs(diff))
            h[j] += adj if diff > 0 else -adj
            diff -= adj if diff > 0 else -adj
            i += 1
    return h


def grad_smooth(h, n, dx, gamma):

    m = np.max(c)
