            h = np.clip(h, 0.0, 1.0)
        # Fine adjustment on remaining freedom
        diff2 = target - h.sum()
        if abs(diff2) > 1e-12:
            if diff2 > 0:
                mask2 = h < 1.0
            else:
                mask2 = h > 0.0
            cnt2 = mask2.sum()
            if cnt2 > 0:
                h[mask2] += diff2 / cnt2
    return h

def topk_smooth_grad(h, n, k_top=8):
    corr = compute_correlations(h, n)
    top_idx = np.argsort(corr)[::-1][:k_top]
    top_vals = corr[top_idx]
    weights = np.exp((top_vals - top_vals.max()) * 20.0)
    weights /= weights.sum()
    grad = np.zeros(n)
    for i, m in enumerate(top_idx):
        if weights[i] < 1e-6:
            continue
        lag = m - (n - 1)
        grad += weights[i] * compute_grad_lag(h, n, lag)
    return grad, corr.max()

def optimize_phase(h, n, max_iter, time_frac, budget_s):
    start = time.time()
    c5 = compute_correlations(h, n).max()
    best_local_c5 = c5
    best_local_h = h.copy()
    lr = 0.05
    stall_count = 0
    for it in range(max_iter):
        if time.time() - start > budget_s * time_frac:
            break
        if (time.time() - start) > max_iter * 0.05:
            break
        grad, cur_max = topk_smooth_grad(h, n, k_top=max(4, min(12, n // 4)))
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        new_c5 = compute_correlations(h_new, n).max()
        if new_c5 < c5 - 1e-9:
            h = h_new
            c5 = new_c5
            lr = min(lr * 1.02, 0.3)
            stall_count = 0
            if c5 < best_local_c5:
                best_local_c5 = c5
                best_local_h = h.copy()
        else:
            lr *= 0.65
            stall_count += 1
            if stall_count > 30:
                h = best_local_h + np.random.default_rng(it).normal(0, 0.03, n)
                h = project_feasible(h, n)
                c5 = compute_correlations(h, n).max()
                lr = 0.03
                stall_count = 0
    return best_local_h, best_local_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    best_h = h_init.copy()
    best_c5 = compute_correlations(best_h, n_init).max()
    best_n = n_init

    def elapsed():
        return time.time() - t0

    # Phase 1: Optimize at initial resolution
    h1, c5_1 = optimize_phase(h_init.copy(), n_init, 8000, 0.15, budget_s)
    if c5_1 < best_c5:
        best_c5, best_h, best_n = c5_1, h1.copy(), n_init

    # Phase 2: Try multiple resolutions
    res_list = []
    for n_try in [n_init + 40, n_init * 2, 240, 320, 480, 640]:
        if n_try > 800 or n_try <= n_init:
            continue
        res_list.append(n_try)
    # Remove duplicates and sort
    res_list = sorted(set(res_list))

    for n_try in res_list:
        if elapsed() / budget_s > 0.85:
            break
        x_best = np.linspace(0, 2, best_n, endpoint=False)
        x_try = np.linspace(0, 2, n_try, endpoint=False)
        h_try = np.interp(x_try, x_best, best_h)
        h_try = project_feasible(h_try, n_try)
        remain_frac = min(0.08, max(0.02, (budget_s - elapsed()) / budget_s / max(len(res_list), 1)))
        h_opt, c5_opt = optimize_phase(h_try, n_try, 4000, remain_frac, budget_s)
        if c5_opt < best_c5:
            best_c5, best_h, best_n = c5_opt, h_opt.copy(), n_try

    # Phase 3: Random restarts at best resolution
    if elapsed() / budget_s < 0.92:
        for r in range(8):
            if elapsed() / budget_s > 0.96:
                break
            h_r = best_h + rng.normal(0, 0.04, best_n)
            h_r = project_feasible(h_r, best_n)
            h_opt, c5_opt = optimize_phase(h_r, best_n, 3000, 0.02, budget_s)
            if c5_opt < best_c5:
                best_c5, best_h = c5_opt, h_opt.copy()

    # Phase 4: Final polish
    if elapsed() / budget_s < 0.98:
        h_opt, c5_opt = optimize_phase(best_h.copy(), best_n, 2000, 0.02, budget_s)
        if c5_opt < best_c5:
            best_c5, best_h = c5_opt, h_opt.copy()

    # Ensure exact sum constraint
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_correlations(best_h, best_n).max()

    return best_h, best_c5, best_n
