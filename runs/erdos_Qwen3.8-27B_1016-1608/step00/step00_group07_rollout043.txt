def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def compute_corr_all(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def grad_for_lag(h, k):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if k > 0:
        g[:n-k] += (1.0 - h[k:]) * dx
        g[k:] -= h[:n-k] * dx
    elif k < 0:
        k2 = -k
        g[k2:] += (1.0 - h[:n-k2]) * dx
        g[:n-k2] -= h[k2:] * dx
    else:
        g[:] += (1.0 - h) * dx
        g[:] += (1.0 - h) * dx
        g[:] /= 1.0
        g[:] = -h * dx + (1.0 - h) * dx
    return g
