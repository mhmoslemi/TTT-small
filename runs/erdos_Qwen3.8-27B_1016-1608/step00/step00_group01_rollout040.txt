If sum(h) > target, we need to decrease h. We subtract λ: h_new = clip(h-λ, 0, 1). As λ increases, sum(h_new) decreases. We want sum(h_new) = target.
- If λ=0: sum = sum(h) > target
- If λ is large: sum → 0 < target
So we want to find λ where sum(clip(h-λ,0,1)) = target. This is a monotonically decreasing function of λ, so binary search works.

If sum(h) < target, we need to increase h. We add λ: h_new = clip(h+λ, 0, 1). As λ increases, sum increases.
- If λ=0: sum = sum(h) < target
- If λ is large: sum → n > target

So in general, we want to find λ such that sum(clip(h+λ, 0, 1)) = target. If target > sum(h), λ > 0. If target < sum(h), λ < 0.

import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def eval_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_objective(h, dx, T):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(logsumexp(T * corr * dx) / T)


def smooth_gradient(h, dx, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx
    m = np.max(vals)
    exp_v = np.exp(T * (vals - m))
    s = np.sum(exp_v)
    w = exp_v / s
    g = np.zeros(n)
    for p in range(2 * n - 1):
        if w[p] < 1e-14:
            continue
        k = p - (n - 1)
        c = w[p] * dx
        for i in range(n):
            gi = 0.0
            a = i + k
            if 0 <= a < n:
                gi += 1.0 - h[a]
            b = i - k
            if 0 <= b < n:
                gi -= h[b]
            g[i] += c * gi
    return g


def sum_eq(h, target):
    return np.sum(h) - target


def sum_eq_jac(h, target):
    return np.ones(len(h))


def project_sum(h, target):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)


def local_search(h, dx, target, rng, max_iters, eps):
    n = len(h)
    best_c5 = eval_c5(h, dx)
    h = h.copy()
    stall = 0
    for _ in range(max_iters):
        improved = False
        for _ in range(60):
            i, j = rng.randint(0, n, 2)
            if i == j:
                continue
            delta = min(eps, h[i], 1.0 - h[j])
            if delta < 1e-10:
                continue
            h[i] -= delta
            h[j] += delta
            c_new = eval_c5(h, dx)
            if c_new < best_c5 - 1e-15:
                best_c5 = c_new
                improved = True
            else:
                h[i] += delta
                h[j] -= delta
        if improved:
            stall = 0
        else:
            stall += 1
            if stall >= 4:
                eps *= 0.65
                if eps < 0.0005:
                    break
                stall = 0
    return h, best_c5


def lag_objective(h, dx, k, target):
    n = len(h)
    s = 0.0
    for i in range(n):
        j = i + k
        if 0 <= j < n:
            s += h[i] * (1.0 - h[j])

def active_set_opt(h_init, dx, n, target, bounds, cons, rng):
    h = h_init.copy()
    for iteration in range(30):
        corr = np.correlate(h, 1.0 - h, mode="full")
