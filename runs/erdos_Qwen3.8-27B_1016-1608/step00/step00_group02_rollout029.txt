    best_c5 = evaluate_c5(h)

    improved = True
    while improved and (time.time() - t0) < timeout:
        improved = False
        for i in range(n):
            if (time.time() - t0) > timeout:
                break
            for direction in [-1, 1]:
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + direction * step, 0.0, 1.0)

                s = np.sum(h_new)
