def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx
    return float(np.max(vals)), vals

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()

    for _ in range(1000):
        s = np.sum(h)
        diff = target - s

            room = 1.0 - h
            tr = np.sum(room)
            if tr < 1e-15:
                break
            t = min(diff, tr)
