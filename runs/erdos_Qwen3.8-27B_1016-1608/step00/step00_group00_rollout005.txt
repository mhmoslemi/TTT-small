    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    return float(m + np.log(np.sum(np.exp(corr - m))) / beta)

def projected(h, n):
    """Project h onto [0,1]^n with sum = n/2."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if diff > 0:
        mask = h < 1.0
        idx = np.where(mask)[0]
        if len(idx) == 0:
            return h
        per = diff / len(idx)
        h[idx] = np.min(h[idx] + per, 1.0)
    elif diff < 0:
        mask = h > 0.0
        idx = np.where(mask)[0]
        if len(idx) == 0:
            return h
        per = diff / len(idx)
        h[idx] = np.max(h[idx] + per, 0.0)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_c5(h, beta=50.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    return float(m + np.log(np.sum(np.exp(corr - m))) / beta)

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        idx = np.where(mask)[0]
        if len(idx) > 0:
            per = diff / len(idx)
            h[idx] = np.clip(h[idx] + per, 0.0, 1.0)
        h += (target - np.sum(h)) / n
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_at_max_lag(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(corr))
    lag = k_star - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        grad[:n - lag] -= h[lag:]
        grad[lag:] += (1.0 - h[:n - lag])
    else:
        m = -lag
        grad[m:] -= h[:n - m]
        grad[:n - m] += (1.0 - h[m:])
    return grad * dx, float(corr[k_star] * dx)

def projected_gradient_descent(h, n_steps, lr, n):
    h = h.copy()
    best_local = compute_c5(h)
    for _ in range(n_steps):
        grad, _ = grad_at_max_lag(h)
        h_new = project_feasible(h - lr * grad, n)
        val = compute_c5(h_new)
        if val < best_local:
            h = h_new
            best_local = val
        else:
            lr *= 0.8
            if lr < 1e-8:
                break
    return h, best_local

def make_random_init(n, seed):
    rng = np.random.RandomState(seed)
    h = (rng.rand(n) > 0.5).astype(float)
    target = n / 2.0
    s = np.sum(h)
    diff = target - s
    if diff > 0:
        idx = np.where(h == 0)[0]
        rng.shuffle(idx)
        for i in range(min(int(diff), len(idx))):
            h[idx[i]] = 1.0
        if int(diff) < diff:
            h[idx[int(diff)]] = diff - int(diff) if len(idx) > int(diff) else 0.5
    elif diff < 0:
        idx = np.where(h == 1)[0]
        rng.shuffle(idx)
        for i in range(min(int(-diff), len(idx))):
            h[idx[i]] = 0.0
        if int(-diff) < -diff:
            h[idx[int(-diff)]] = diff + int(-diff)
    return project_feasible(h, n)
