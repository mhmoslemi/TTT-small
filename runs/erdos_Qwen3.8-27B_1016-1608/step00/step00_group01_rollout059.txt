import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            slack = 1.0 - h
            s = np.sum(slack)
            if s < 1e-14:
                break
            h += (diff / s) * slack
        else:
            s = np.sum(h)
            if s < 1e-14:
                break
            h += (diff / s) * h
        h = np.clip(h, 0.0, 1.0)
    return h

def active_grad(h, max_idx):
    n = len(h)
    k = max_idx - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        m1 = np.arange(k, n)
        grad[m1] += (1.0 - h[m1 - k])
        m2 = np.arange(0, n - k)
        grad[m2] += -h[m2 + k]
    else:
        l = -k
        m1 = np.arange(l, n)
        grad[m1] += (1.0 - h[m1 - l])
        m2 = np.arange(0, n - l)
        grad[m2] += -h[m2 + l]
    return grad

def obj_grad(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    max_idx = int(np.argmax(corr))
    val = float(corr[max_idx] * dx)
    grads = active_grad(h, max_idx)
    grad = grads * dx
    return val, grad

def optimize_nlp(h0, n, max_iter=500, ftol=1e-14):
    target = n / 2.0
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target,
             'jac': lambda h: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_grad, h0, method='SLSQP', jac=True,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': ftol})
        if res.fun < np.inf:
            return repair(res.x, n), float(res.fun)
    except Exception:
        pass
    return repair(h0, n), compute_c5(h0)

def make_random_h(n, rng):
    h = rng.uniform(0.0, 1.0, n)
    return repair(h, n)

    n_best = len(h_best)
    h_best = repair(h_best, n_best)
    c5_best = compute_c5(h_best)

    # Phase 1: Polished optimization from initial
    h_opt, c5_opt = optimize_nlp(h_best, n_best, max_iter=800)
    if c5_opt < c5_best:
        c5_best, h_best = c5_opt, h_opt
    elapsed = time.time() - t_start
    print(f"After phase 1: C5={c5_best:.8f}, n={n_best}, t={elapsed:.1f}s")

    # Phase 2: Local perturbations + re-optimize
    n_attempts = 0
    while time.time() - t_start < budget_s * 0.75:
        scale = max(0.01, 0.15 * (1 - (time.time() - t_start) / budget_s))
        h_try = h_best + rng.normal(0, scale, n_best)
        h_try = repair(h_try, n_best)
        h_opt, c5_opt = optimize_nlp(h_try, n_best, max_iter=400, ftol=1e-13)
        if c5_opt < c5_best:
            c5_best, h_best = c5_opt, h_opt
            print(f"  Improved: C5={c5_best:.8f} (attempt {n_attempts})")
        n_attempts += 1

    # Phase 3: Try slightly different n
    for n_try in [50, 56, 64, 80, 96]:
        if time.time() - t_start > budget_s * 0.90:
            break
        h_try = make_random_h(n_try, rng)
        # Also try interpolated version of best
        if n_best != n_try:
            x_old = np.linspace(0, 2, n_best, endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_interp = np.interp(x_new, x_old, h_best)
            h_interp = repair(h_interp, n_try)
            if compute_c5(h_interp) < compute_c5(h_try):
                h_try = h_interp
        h_opt, c5_opt = optimize_nlp(h_try, n_try, max_iter=600)
        if c5_opt < c5_best:
            c5_best, h_best, n_best = c5_opt, h_opt, n_try
            print(f"  Better with n={n_try}: C5={c5_best:.8f}")
        # Second try with different random init
        if time.time() - t_start < budget_s * 0.90:
            h_try2 = make_random_h(n_try, rng)
            h_opt2, c5_opt2 = optimize_nlp(h_try2, n_try, max_iter=400)
            if c5_opt2 < c5_best:
                c5_best, h_best, n_best = c5_opt2, h_opt2, n_try
                print(f"  Better (2nd) with n={n_try}: C5={c5_best:.8f}")

    # Phase 4: Final local polish
    if time.time() - t_start < budget_s * 0.95:
        h_opt, c5_opt = optimize_nlp(h_best, n_best, max_iter=1000, ftol=1e-15)
        if c5_opt < c5_best:
            c5_best, h_best = c5_opt, h_opt

    h_best = repair(h_best, n_best)
    c5_final = compute_c5(h_best)
    try:
        c5_eval = evaluate_erdos_solution(h_best)
        if c5_eval < c5_final:
            c5_final = float(c5_eval)
    except Exception:
        pass

    print(f"Final: C5={c5_final:.8f}, n={n_best}, total_time={time.time()-t_start:.1f}s")
    return (h_best, c5_final, n_best)
