import numpy as np
import time
import scipy.optimize as opt


def _c5_eval(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def _fix_sum(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        free = (h > 1e-8) & (h < 1.0 - 1e-8)
        if np.any(free):
            h[free] += diff / np.sum(free)
            h = np.clip(h, 0.0, 1.0)
            diff2 = target - np.sum(h)
            if abs(diff2) > 1e-12:
                h += diff2 / n
                h = np.clip(h, 0.0, 1.0)
        else:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
    return h


def _slsqp(h_init, n, maxiter=400):
    dx = 2.0 / n
    target = n / 2.0
    bounds = list(zip(np.zeros(n), np.ones(n)))
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]

    def obj(x):
        return float(np.max(np.correlate(x, 1.0 - x, mode='full')) * dx)

    try:
        res = opt.minimize(obj, h_init, method='SLSQP', bounds=bounds,
                           constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
        x = _fix_sum(res.x, n)
        return x, _c5_eval(x, n)
    except Exception:
        return _fix_sum(h_init, n), _c5_eval(h_init, n)


def _coord_descent(h, n, max_rounds=30):
    h = h.copy()
    target = n / 2.0
    dx = 2.0 / n
    for _ in range(max_rounds):
        improved = False
        for i in range(n):
            old = h[i]
            best_t = old
            best_v = _c5_eval(h, n)
            # Try a grid of values
            for t in np.linspace(0.0, 1.0, 21):
                h[i] = t
                # Adjust one other element to maintain sum
                others = np.arange(n) != i
                need = target - t
                cur_others_sum = np.sum(h[others])
                if cur_others_sum > 1e-12:
                    scale = need / cur_others_sum
                    h[others] = np.clip(h[others] * scale, 0.0, 1.0)
                v = _c5_eval(h, n)
                if v < best_v:
                    best_v = v
                    best_t = t
                    best_h = h.copy()
            if best_t != old:
                h = best_h
                improved = True
            else:
                h[i] = old
        if not improved:
            break
    return _fix_sum(h, n), _c5_eval(h, n)


def _sa(h_init, n, n_iter=80000, rng=None):
    target = n / 2.0
    h = h_init.copy()
    cur_v = _c5_eval(h, n)
    best = h.copy()
    best_v = cur_v
    t0, t1 = 0.04, 0.0005
    for i in range(n_iter):
        temp = t0 * (t1 / t0) ** (i / n_iter)
        h_new = h.copy()
        j = rng.randint(n)
        h_new[j] = np.clip(h_new[j] + rng.randn() * 0.12, 0.0, 1.0)
        h_new = _fix_sum(h_new, n)
        v_new = _c5_eval(h_new, n)
        d = v_new - cur_v
        if d < 0 or rng.rand() < np.exp(-d / max(temp, 1e-15)):
            h = h_new
            cur_v = v_new
            if v_new < best_v:
                best = h.copy()
                best_v = v_new
    return best, best_v
