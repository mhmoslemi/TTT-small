    """Compute C5 for a given h array."""
    n = len(h)
    dx = 2.0 / n
    overlap = np.correlate(h, 1-h, mode="full") * dx

    h = np.round(h)
    h = np.clip(h, 0, 1).astype(int)
    s = h.sum()

        idx = np.where(h == 1)[0]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx, corr

def make_valid_binary(h, n, rng):
    h = np.round(np.clip(h, 0, 1)).astype(float)
    target = n // 2
    s = int(h.sum())
    if s > target:
        idx = np.where(h == 1)[0]
        rng.shuffle(idx)
        h[idx[:s - target]] = 0
    elif s < target:
        idx = np.where(h == 0)[0]
        rng.shuffle(idx)
        h[idx[:target - s]] = 1
    return h

def local_search(h0, n, time_limit, start_time, rng, verbose=False):
    h = h0.copy()
    _, corr = compute_c5(h)
    dx = 2.0 / n
    best_score = np.max(corr) * dx
    best_h = h.copy()
    stagnant = 0
    best_stagnant = 0
    
    while time.time() - start_time < time_limit:

        best_move = None
        best_move_score = best_score
        best_move_corr = None
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Try all single swaps (i: 1->0, j: 0->1)
        for i in ones:

            for j in zeros:

                h[i] = 0
                h[j] = 1

                if s < best_move_score - 1e-12:

                h[i] = hi_val

    dx = 2.0 / n
    best_score = np.max(corr) * dx
