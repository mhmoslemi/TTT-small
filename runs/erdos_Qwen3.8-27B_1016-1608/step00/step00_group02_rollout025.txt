import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(dx * np.max(corr))

def make_valid(h, n):
    h = np.asarray(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # Spread correction among non-saturated entries
        mask = (h > 1e-9) & (h < 1.0 - 1e-9)
        if mask.any():
            per = diff / mask.sum()
            h[mask] = np.clip(h[mask] + per, 0.0, 1.0)
        # Final fix
        residual = target - h.sum()
        if abs(residual) > 1e-12:
            if residual > 0:
                idxs = np.where(h < 1.0)[0]
            else:
                idxs = np.where(h > 0.0)[0]
            if len(idxs) > 0:
                h[idxs[0]] = np.clip(h[idxs[0]] + residual, 0.0, 1.0)
    return h

    x_old = np.arange(n_old) * 2.0 / n_old
    x_new = np.arange(n_new) * 2.0 / n_new
    return np.interp(x_new, x_old, h)

                cur_h = best_h.copy()
                patience = 0
                continue
        
        cur_h = cands[best_idx].copy()
        rounds += 1
        if patience >= max_patience:
            cur_h = best_h.copy()

def sequential_polish(h0, time_budget, rng):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    t0 = time.time()
    
    while time.time() - t0 < time_budget:
        i1 = rng.integers(0, n)
        i2 = rng.integers(0, n)
        if i1 == i2:
            i2 = (i2 + 1) % n
        delta = rng.uniform(-0.02, 0.02)
        
        new_h = h.copy()
        new_h[i1] += delta
        new_h[i2] -= delta
        new_h = np.clip(new_h, 0.0, 1.0)
        diff = n / 2.0 - new_h.sum()
        if abs(diff) > 1e-12:
            new_h += diff / n
            new_h = np.clip(new_h, 0.0, 1.0)
        
        new_val = compute_c5(new_h)
        if new_val < best_val:
            best_val = new_val
            best_h = new_h.copy()
            h = new_h
    
    return best_h, best_val
