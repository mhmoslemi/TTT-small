def _corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def _c5(h):
    return float(np.max(_corr(h)) * 2.0 / len(h))

def _proj(h, ts):
    h = h.astype(float).copy()
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        d = ts - h.sum()
        if abs(d) < 1e-14:
            break
        fr = (h > 1e-14) & (h < 1.0 - 1e-14)
        nf = fr.sum()
        if nf == 0:
            break
        h[fr] += d / nf
    return np.clip(h, 0.0, 1.0)

def _sobj(h, beta):
    c = _corr(h)
    m = c.max()
    if m <= 0:
        return 0.0
    return (m + np.log(np.sum(np.exp(beta * (c - m)))) / beta) * (2.0 / len(h))
