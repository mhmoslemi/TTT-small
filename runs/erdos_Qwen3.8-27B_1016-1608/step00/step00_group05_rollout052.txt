def compute_c5(h):

    dx = 2.0 / n

    return float(np.max(corr) * dx)

def smooth_obj(h, params):
    beta, n = params
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    scaled = np.clip(beta * (corr - m), -500.0, 0.0)
    return (m + np.log(np.sum(np.exp(scaled))) / beta) * (2.0 / n)

    return np.sum(h) - target

    return np.ones(len(h))

        h = np.clip(h, 0.0, 1.0)

= (1/beta) * (beta*max + log(1 + sum_{k≠max} exp(beta*(c_k - max)))) * dx
= max*dx + (1/beta)*log(1 + sum_{k≠max} exp(beta*(c_k - max)))*dx

So it's close to max*dx but smoothed.

The gradient: df/dh[i] = dx * sum_k (beta * c_k') / sum_j exp(beta * (c_j - m)) ...

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(h, params):
    beta, n = params
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    scaled = np.clip(beta * (corr - m), -500.0, 0.0)
    return (m + np.log(np.sum(np.exp(scaled))) / beta) * (2.0 / n)

def eq_fun(h, target):
    return np.sum(h) - target

def eq_jac(h, target):
    return np.ones(len(h))

def fix_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def coordinate_polish(h, target, n_iter=30):
    """Polish via coordinate descent with random restarts on each coordinate."""
    best = h.copy()
    best_c5 = compute_c5(best)
    n = len(best)
    rng = np.random.RandomState(12345)
    for _ in range(n_iter):
        improved = False
        for i in range(n):
            if abs(best[i]) < 1e-12 or abs(best[i] - 1.0) < 1e-12:
                continue
            old = best[i]
            for t in [0.0, 1.0, old - 0.1, old - 0.05, old + 0.05, old + 0.1, 0.5]:
                t = max(0.0, min(1.0, t))
                trial = best.copy()
                trial[i] = t
                trial = fix_sum(trial, target)
                c5 = compute_c5(trial)
                if c5 < best_c5 - 1e-12:
                    best = trial
                    best_c5 = c5
                    improved = True
            if not improved:
                break
        if not improved:
            break
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(70, 0.5)

    n = len(h0)
    target = n / 2.0
    best_h = fix_sum(h0, target)
    best_c5 = compute_c5(best_h)
    bounds = [(0.0, 1.0)] * n
    constraint = {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac, 'args': target}

    attempt = 0
    while time.time() - t0 < budget_s - 5:
        attempt += 1
        remaining = budget_s - (time.time() - t0)
        if remaining < 3:
            break

        # Choose starting point
        if attempt == 1:
            h_start = best_h.copy()
        elif attempt <= 8:
            h_start = best_h.copy()
            k = np.random.randint(2, max(3, n // 4))
            idx = np.random.choice(n, k, replace=False)
            h_start[idx] += np.random.normal(0, 0.03, k)
        elif attempt <= 25:
            h_start = best_h.copy()
            k = np.random.randint(n // 5, max(n // 3, n // 2))
            idx = np.random.choice(n, k, replace=False)
            h_start[idx] += np.random.normal(0, 0.08, k)
        elif attempt <= 60:
            h_start = best_h.copy()
            k = int(n * np.random.uniform(0.3, 0.7))
            idx = np.random.choice(n, k, replace=False)
            h_start[idx] += np.random.normal(0, 0.15, k)
        else:
            r = np.random.random()
            if r < 0.25:
                h_start = np.random.uniform(0, 1, n)
            elif r < 0.6:
                h_start = best_h.copy()
                k = int(n * np.random.uniform(0.4, 0.9))
                idx = np.random.choice(n, k, replace=False)
                h_start[idx] += np.random.normal(0, 0.25, k)
            else:
                # Structured random: alternating blocks
                block = np.random.randint(2, 6)
                h_start = np.tile(np.random.uniform(0.2, 0.8, block * (n // block + 1))[:n], 1)
        
        h_start = fix_sum(h_start, target)
        beta = min(25.0 + attempt * 4.0, 250.0)
        maxiter = max(50, int(min(600, remaining * 30)))

        try:
            res = minimize(smooth_obj, h_start, args=(beta, n),
                           method='SLSQP', bounds=bounds,
                           constraints=[constraint],
                           options={'maxiter': maxiter, 'ftol': 1e-13})
            if res.x is not None and np.all(np.isfinite(res.x)):
                h_new = fix_sum(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-10:
                    best_c5 = c5_new
                    best_h = h_new
        except Exception:
            pass

        # Occasional coordinate polish
        if attempt % 7 == 0 and remaining > 15:
            try:
                h_pol, c5_pol = coordinate_polish(best_h.copy(), target, n_iter=15)
                if c5_pol < best_c5 - 1e-10:
                    best_c5 = c5_pol
                    best_h = h_pol
            except Exception:
                pass

        # Try a few different resolutions periodically
        if attempt in (15, 35, 55) and remaining > 60:
            for n_try in [50, 100, 140]:
                if time.time() - t0 > budget_s - 15:
                    break
                h_try = np.interp(np.linspace(0, 2, n_try), np.linspace(0, 2, n), best_h)
                h_try = fix_sum(h_try, n_try / 2.0)
                bnds_try = [(0.0, 1.0)] * n_try
                con_try = {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac, 'args': n_try / 2.0}
                try:
                    res_t = minimize(smooth_obj, h_try, args=(60.0, n_try),
                                     method='SLSQP', bounds=bnds_try,
                                     constraints=[con_try],
                                     options={'maxiter': 300, 'ftol': 1e-12})
                    h_opt = fix_sum(res_t.x, n_try / 2.0)
                    c5_opt = compute_c5(h_opt)
                    if c5_opt < best_c5 - 1e-10:
                        best_c5 = c5_opt
                        best_h = h_opt
                        n = n_try
                        target = n / 2.0
                        bounds = [(0.0, 1.0)] * n
                        constraint = {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac, 'args': target}
                except Exception:
                    pass

    # Final polish
    if time.time() - t0 < budget_s - 1:
        try:
            best_h, best_c5 = coordinate_polish(best_h.copy(), target, n_iter=20)
        except Exception:
            pass

    best_h = fix_sum(best_h, target)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, len(best_h))
