    if remaining > 10:
        n_iter = min(5000, int(remaining * 200))
        h_lp, f_lp = local_polish(h_best, n, dx, max_iter=n_iter)
        if f_lp < f_best:
            f_best, h_best = f_lp, h_lp

    # Phase 4: Try a few more random restarts + polish if time allows
    remaining = (t0 + budget_s - 10) - time.time()
    if remaining > 20:
        rng = np.random.RandomState(seed + 9999)
        for _ in range(5):
            if time.time() > t0 + budget_s - 10:
                break
            h_try = h_best + rng.randn(n) * 0.01
            h_try = repair_h(h_try, n)
            f_try = compute_c5(h_try, dx)
            if f_try < f_best:
                f_best, h_best = f_try, h_try
            h_lp2, f_lp2 = local_polish(h_best, n, dx, max_iter=1000)
            if f_lp2 < f_best:
                f_best, h_best = f_lp2, h_lp2

    # Final verification
    f_best = compute_c5(h_best, dx)
    if 'evaluate_erdos_solution' in globals():
        f_verify = evaluate_erdos_solution(h_best)
        f_best = min(f_best, f_verify)

    return (h_best, f_best, n)
