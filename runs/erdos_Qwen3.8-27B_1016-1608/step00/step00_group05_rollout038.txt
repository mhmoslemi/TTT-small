import numpy as np
from scipy.optimize import minimize
import time

def compute_coral(h):
    return np.correlate(h, 1.0 - h, mode='full')

def fix_sum(h, target):
    for _ in range(20):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_descent_step(h, n, target):
    changed_total = False
    for i in range(n):
        old = h[i]
        h[i] = 0.0
        base = compute_coral(h).copy()
        h[i] = old
        coeffs = np.zeros(2 * n - 1)
        for k in range(-(n - 1), n):
            m = n - 1 + k
            c = 0.0
            j1 = i - k
            if 0 <= j1 < n:
                c += (1.0 - h[j1])
            j2 = i + k
            if 0 <= j2 < n:
                c -= h[j2]
            coeffs[m] = c
        xs = np.linspace(0.0, 1.0, 200)
        vals = np.max(base[None, :] + coeffs[None, :] * xs[:, None], axis=1)
        x_opt = xs[np.argmin(vals)]
        lo = max(0.0, x_opt - 0.015)
        hi = min(1.0, x_opt + 0.015)
        xs2 = np.linspace(lo, hi, 80)
        vals2 = np.max(base[None, :] + coeffs[None, :] * xs2[:, None], axis=1)
        x_opt = xs2[np.argmin(vals2)]
        if abs(x_opt - old) > 1e-12:
            h[i] = x_opt
            changed_total = True
    return h, changed_total

def slsqp_optimize(h, n, target):
    def obj(xv):
        return np.max(compute_coral(xv))
    constr = ({'type': 'eq', 'fun': lambda xv: xv.sum() - target,
               'jac': lambda xv: np.ones(n)},)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=constr, options={'maxiter': 300, 'ftol': 1e-14})
        if res.fun < 1e6:
            return fix_sum(np.clip(res.x, 0.0, 1.0), target)
    except Exception:
        pass
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = 70
    dx = 2.0 / n
    target = n / 2.0

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        h = np.full(n, 0.5)
    h = np.clip(h, 0.0, 1.0)
    h = fix_sum(h, target)

    best_h = h.copy()
    best_val = np.max(compute_coral(h))

    def elapsed():
        return time.time() - t0

    # Main coordinate descent loop with restarts
    stagnation = 0
    for outer in range(2000):
        if elapsed() > budget_s - 45:
            break

        # Coordinate descent (multiple inner rounds)
        for _ in range(8):
            h, ch = coord_descent_step(h, n, target)
            if not ch:
                break
        h = fix_sum(h, target)
        val = np.max(compute_coral(h))

        if val < best_val - 1e-12:
            best_val = val
            best_h = h.copy()
            stagnation = 0
        else:
            stagnation += 1

        # Restart with perturbation after stagnation
        if stagnation >= 6 and elapsed() < budget_s * 0.65:
            sigma = 0.015 + 0.01 * (stagnation // 6)
            h = best_h + rng.normal(0, sigma, n)
            h = np.clip(h, 0.0, 1.0)
            h = fix_sum(h, target)
            stagnation = 0

        # Occasionally try SLSQP from current best
        if outer % 40 == 20 and elapsed() < budget_s * 0.5:
            h_try = slsqp_optimize(best_h.copy(), n, target)
            v_try = np.max(compute_coral(h_try))
            if v_try < best_val:
                best_val = v_try
                best_h = h_try.copy()
                h = h_try.copy()

    best_c5 = best_val * dx
    best_h = fix_sum(best_h, target)
    return best_h, best_c5, n
