    t0 = time.time()

    def _elapsed():
        return time.time() - t0

    def _c5(h):
        n = len(h)
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        return float(np.max(corr))

    def _corr_vec(h):
        n = len(h)
        dx = 2.0 / n
        return np.correlate(h, 1.0 - h, mode="full") * dx

    def _smooth_obj_grad(h, temp, pen_w=50.0):
        n = len(h)
        target = n / 2.0
        corr = _corr_vec(h)
        cmax = np.max(corr)
        exp_c = np.exp(temp * (corr - cmax))
        s_exp = np.sum(exp_c)
        p = exp_c / s_exp
        f = cmax + np.log(s_exp) / temp
        grad = np.zeros(n)
        idxs = np.nonzero(p > 1e-14)[0]
        for idx in idxs:
            lag = idx - (n - 1)
            ii = np.arange(n)
            il = ii - lag
            ih = ii + lag
            vl = (il >= 0) & (il < n)
            vh = (ih >= 0) & (ih < n)
            contrib = np.zeros(n)
            if np.any(vl):
                contrib[vl] += (1.0 - h[il[vl]])
            if np.any(vh):
                contrib[vh] -= h[ih[vh]]
            grad += p[idx] * contrib
        f += pen_w * (np.sum(h) - target) ** 2
        grad += 2.0 * pen_w * (np.sum(h) - target)
        return f, grad

    def _fix_sum(h, n_pts):
        target = n_pts / 2.0
        h = np.clip(h, 0.0, 1.0)
        s = np.sum(h)
        if s > 1e-12:
            h *= target / s
        h = np.clip(h, 0.0, 1.0)
        return h

    def _optimize(h_start, temps, max_iter_each=600):
        n_pts = len(h_start)
        h_cur = h_start.copy()
        for temp in temps:
            if budget_s - _elapsed() < 30:
                break
            def obj(h):
                return _smooth_obj_grad(h, temp)[0]
            def grad(h):
                return _smooth_obj_grad(h, temp)[1]
            bounds = [(0.0, 1.0)] * n_pts
            try:
                res = minimize(obj, h_cur, jac=grad, method='L-BFGS-B',
                               bounds=bounds,
                               options={'maxiter': max_iter_each,
                                        'ftol': 1e-15, 'gtol': 1e-12,
                                        'maxls': 40})
                h_cur = np.clip(res.x, 0.0, 1.0)
            except Exception:
                pass
            h_cur = _fix_sum(h_cur, n_pts)
        return h_cur

    def _local_search(h_init, max_rounds=2000):
        n_pts = len(h_init)
        h = h_init.copy()
        c5_cur = _c5(h)
        rng = np.random.RandomState(seed + 7)
        for _ in range(max_rounds):
            if budget_s - _elapsed() < 15:
                break
            improved = False
            for _ in range(60):
                i = rng.randint(n_pts)
                j = rng.randint(n_pts)
                if i == j:
                    continue
                if h[i] > 0.02 and h[j] < 0.98:
                    for delta in (0.15, 0.3, 0.5):
                        if h[i] - delta < 0 or h[j] + delta > 1:
                            continue
                        h[i] -= delta
                        h[j] += delta
                        c5_new = _c5(h)
                        if c5_new < c5_cur - 1e-10:
                            c5_cur = c5_new
                            improved = True
                            break
                        else:
                            h[i] += delta
                            h[j] -= delta
                    else:
                        continue
                    break
            if not improved:
                break
        return h, c5_cur

    h_best = np.array(initial_h_values, dtype=float).copy()
    h_best = _fix_sum(h_best, n)
    c5_best = _c5(h_best)

    if budget_s - _elapsed() > 200:
        h_opt = _optimize(h_best.copy(), [25.0, 10.0, 5.0, 2.5], max_iter_each=700)
        c5_opt = _c5(h_opt)
        if c5_opt < c5_best:
            h_best = h_opt
            c5_best = c5_opt

    # Phase 2: Local search
    if budget_s - _elapsed() > 80:
        h_ls, c5_ls = _local_search(h_best.copy(), max_rounds=1500)
        if c5_ls < c5_best:
            h_best = h_ls
            c5_best = c5_ls

    # Phase 3: Try structured initial guesses with different n
    for n_try in [50, 80, 100, 120]:
        if budget_s - _elapsed() < 120:
            break
        # Alternating pattern
        h_alt = np.zeros(n_try)
        h_alt[::2] = 1.0
        h_alt = _fix_sum(h_alt, n_try)
        h_opt = _optimize(h_alt, [30.0, 10.0, 4.0], max_iter_each=500)
        c5_opt = _c5(h_opt)
        if c5_opt < c5_best:
            h_best = h_opt
            c5_best = c5_opt

        # Random initialization
        h_rnd = np.random.RandomState(seed + n_try).rand(n_try)
        h_rnd = _fix_sum(h_rnd, n_try)
        h_opt2 = _optimize(h_rnd, [20.0, 8.0, 3.0], max_iter_each=400)
        c5_opt2 = _c5(h_opt2)
        if c5_opt2 < c5_best:
            h_best = h_opt2
            c5_best = c5_opt2

    # Phase 4: Final polish on best solution
    if budget_s - _elapsed() > 40:
        n_fin = len(h_best)
        h_final = _optimize(h_best.copy(), [8.0, 4.0, 2.0], max_iter_each=400)
        c5_fin = _c5(h_final)
        if c5_fin < c5_best:
            h_best = h_final
            c5_best = c5_fin

    # Ensure feasibility
    n_fin = len(h_best)
    h_best = _fix_sum(h_best, n_fin)
    c5_final = _c5(h_best)

    return (h_best, c5_final, n_fin)
