```python
import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

_g_n = 45
_g_w = 1e6

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_valid(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n * 0.5
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-9
        else:
            mask = h > 1e-9
        cnt = int(mask.sum())
        if cnt == 0:
            h += diff / n
        else:
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _pen_obj(x):
    h = np.clip(x, 0.0, 1.0)
    s = h.sum()
    return compute_c5(h) + _g_w * (s - _g_n * 0.5) ** 2

def _remain(t0, b):
    return b - (time.time() - t0)

def _lbfgs(x0, bounds, maxiter, ftol):
    return minimize(_pen_obj, x0, method='L-BFGS-B',
                    bounds=bounds, options={'maxiter': maxiter, 'ftol': ftol})

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values, _g_n, _g_w
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = 0.5 * np.ones(45)
        h_init[:22] = 1.0
        h_init[22] = 0.5
        h_init[23:] = 0.0

    n0 = len(h_init)
    h_init = make_valid(h_init, n0)

    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    best_n = n0

    bounds0 = [(0.0, 1.0)] * n0

    # Phase 1: DE global search
    _g_n = n0
    _g_w = 1e5
    if _remain(t0, budget_s) > 15:
        try:
            de_r = differential_evolution(_pen_obj, bounds0, seed=seed,
                        maxiter=800, popsize=10, tol=1e-12,
                        mutation=(0.5, 1.0), recombination=0.85,
                        polish=False, init='latinhypercube')
            hc = make_valid(de_r.x, n0)
            c = compute_c5(hc)
            if c < best_c5:
                best_h, best_c5 = hc, c
        except Exception:
            pass

    # Phase 2: L-BFGS-B polish from best and symmetric
    _g_w = 1e8
    for st in [best_h, 1.0 - best_h, h_init]:
        if _remain(t0, budget_s) < 2:
            break
        x0 = make_valid(st, n0)
        r = _lbfgs(x0, bounds0, 3000, 1e-16)
        hc = make_valid(r.x, n0)
        c = compute_c5(hc)
        if c < best_c5:
            best_h, best_c5 = hc, c

    # Phase 3: Many L-BFGS-B restarts with decreasing noise
    for i in range(3000):
        if _remain(t0, budget_s) < 2:
            break
        sc = 0.12 * (0.9992 ** i) + 0.005
        x0 = best_h + rng.randn(n0) * sc
        x0 = make_valid(x0, n0)
        if i % 30 == 15:
            idx = rng.randint(n0)
            x0[idx] = 1.0 - x0[idx]
            x0 = make_valid(x0, n0)
        r = _lbfgs(x0, bounds0, 400, 1e-13)
        hc = make_valid(r.x, n0)
        c = compute_c5(hc)
        if c < best_c5:
            best_h, best_c5 = hc, c

    # Phase 4: Higher resolutions
    for mult in [2, 3]:
        n2 = n0 * mult
        if n2 > 300 or _remain(t0, budget_s) < 50:
            break
        xq = np.linspace(0, 1, n2)
        xq_old = np.linspace(0, 1, best_n)
        h_up = np.interp(xq, xq_old, best_h)
        h_up = make_valid(h_up, n2)
        _g_n = n2
        _g_w = 1e5
        b2 = [(0.0, 1.0)] * n2

        if _remain(t0, budget_s) > 25:
            try:
                de2 = differential_evolution(_pen_obj, b2, seed=seed + n2,
                            maxiter=400, popsize=5, tol=1e-11, polish=False)
                hc = make_valid(de2.x, n2)
                c = compute_c5(hc)
                if c < best_c5:
                    best_h, best_c5, best_n = hc, c, n2
            except Exception:
                pass

        _g_w = 1e8
        h_start = make_valid(de2.x, n2) if _remain(t0, budget_s) > 0 else h_up
        for i in range(300):
            if _remain(t0, budget_s) < 2:
                break
            sc2 = 0.08 * (0.998 ** i) + 0.01
            x0 = best_h + rng.randn(n2) * sc2
            x0 = make_valid(x0, n2)
            r = _lbfgs(x0, b2, 300, 1e-13)
            hc = make_valid(r.x, n2)
            c = compute_c5(hc)
            if c < best_c5:
                best_h, best_c5, best_n = hc, c, n2

    best_h = make_valid(best_h, best_n)
    best_c5 = compute_c5(best_h)
    return list(best_h), best_c5, best_n
```
