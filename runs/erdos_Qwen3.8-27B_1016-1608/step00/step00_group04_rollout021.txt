        if rng is not None:
            rng.shuffle(idx)
        else:
            np.random.shuffle(idx)
        for ix in idx[:target - s]:
            h[ix] = 1.0
    return h

def greedy_optimize(h, max_rounds=300, deadline=None):
    n = len(h)
    h = to_binary(h)
    h = fix_sum_binary(h, n // 2)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_max = float(corr.max())
    for _ in range(max_rounds):
        if deadline is not None and time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_val = cur_max
        best_pi = -1
        best_pj = -1
        best_corr = None
        for pi in ones:
            for pj in zeros:
                h[pi] = 0.0
                h[pj] = 1.0
                nc = np.correlate(h, 1.0 - h, mode='full')
                nv = float(nc.max())
                h[pi] = 1.0
                h[pj] = 0.0
                if nv < best_val - 1e-10:
                    best_val = nv
                    best_pi = pi
                    best_pj = pj
                    best_corr = nc
        if best_pi == -1:
            break
        h[best_pi] = 0.0
        h[best_pj] = 1.0
        corr = best_corr
        cur_max = best_val
    return h, cur_max * dx

def sa_optimize(h, deadline, seed_val):
    n = len(h)
    rng = random.Random(seed_val)
    h = to_binary(h)
    h = fix_sum_binary(h, n // 2, rng)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_s = float(corr.max()) * dx
    best_s = cur_s
    best_h = h.copy()
    while time.time() < deadline:
        T = max(1e-7, cur_s * 0.04)
        ones = list(np.where(h == 1)[0])
        zeros = list(np.where(h == 0)[0])
        if not ones or not zeros:
            break
        pi = ones[rng.randrange(len(ones))]
        pj = zeros[rng.randrange(len(zeros))]
        h[pi] = 0.0
        h[pj] = 1.0
        nc = np.correlate(h, 1.0 - h, mode='full')
        ns = float(nc.max()) * dx
        h[pi] = 1.0
        h[pj] = 0.0
        d = ns - cur_s
        if d <= 0 or rng.random() < math.exp(-d / T):
            h[pi] = 0.0
            h[pj] = 1.0
            cur_s = ns
            if ns < best_s - 1e-12:
                best_s = ns
                best_h = h.copy()
    return best_h, best_s

def golden_init(n, seed_val):
    h = np.zeros(n, dtype=np.float64)
    phi = (1.0 + 5.0**0.5) / 2.0
    used = set()
    for i in range(n // 2):
        p = int((i * phi * n) % n)
        while p in used:
            p = (p + 1) % n
        used.add(p)
        h[p] = 1.0
    return h

def random_init(n, rng):
    h = np.zeros(n, dtype=np.float64)
    ones = rng.sample(range(n), n // 2)
    for ox in ones:
        h[ox] = 1.0
    return h

def kick_greedy(h, n_kicks, max_rounds, deadline, rng):
    n = len(h)
    h = to_binary(h)
    best_h = h.copy()
    best_s = eval_score(h)
    for _ in range(n_kicks):
        if time.time() > deadline:
            break
        for _ in range(3):
            ones = list(np.where(h == 1)[0])
            zeros = list(np.where(h == 0)[0])
            if not ones or not zeros:
                break
            pi = ones[rng.randrange(len(ones))]
            pj = zeros[rng.randrange(len(zeros))]
            h[pi] = 0.0
            h[pj] = 1.0
        h_k, s_k = greedy_optimize(h, max_rounds=max_rounds, deadline=deadline)
        if s_k < best_s:
            best_s = s_k
            best_h = h_k.copy()
            h = h_k.copy()
    return best_h, best_s

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    np.random.seed(seed)
    master_rng = random.Random(seed)
    best_h = None
    best_c5 = float('inf')

    candidates = []

    try:
        h0 = np.array(initial_h_values, dtype=np.float64)
        n0 = len(h0)
        h0b = to_binary(h0)
        h0b = fix_sum_binary(h0b, n0 // 2, master_rng)
        candidates.append(h0b)
    except Exception:
        pass

    for nn in [42, 44, 46, 50, 52, 56, 60, 66, 70, 84]:
        candidates.append(golden_init(nn, seed + nn * 3))

    for nn in [42, 56, 70, 100]:
        candidates.append(random_init(nn, master_rng))

    n_cands = len(candidates)
    phase_time = (budget_s * 0.7) / n_cands

    for ci in range(n_cands):
        phase_deadline = min(deadline - 120, t0 + (ci + 1) * phase_time)
        if time.time() > phase_deadline:
            break
        cand = candidates[ci]
        h_g, s_g = greedy_optimize(cand, max_rounds=250, deadline=phase_deadline)
        if s_g < best_c5:
            best_c5 = s_g
            best_h = h_g.copy()
        if time.time() < phase_deadline - 3:
            h_k, s_k = kick_greedy(h_g, 3, 150, phase_deadline, random.Random(seed + ci * 997))
            if s_k < best_c5:
                best_c5 = s_k
                best_h = h_k.copy()

    # Final SA on best solution
    if best_h is not None and time.time() < deadline - 30:
        sa_deadline = deadline - 30
        h_sa, s_sa = sa_optimize(best_h, sa_deadline, seed + 777)
        if s_sa < best_c5:
            best_c5 = s_sa
            best_h = h_sa.copy()
        h_g2, s_g2 = greedy_optimize(best_h, max_rounds=400, deadline=deadline - 5)
        if s_g2 < best_c5:
            best_c5 = s_g2
            best_h = h_g2.copy()

    if best_h is not None:
        vs = eval_score(best_h)
        best_c5 = min(best_c5, vs)

    return best_h, best_c5, len(best_h)
