from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    h = np.clip(h, 0, 1)
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        # Adjust elements not at bounds
        mask = (h > 1e-10) & (h < 1 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0, 1)
        else:
            break
    return h

def objective_penalty(h, n, penalty=100.0):
    target = n / 2.0
    base = compute_c5(h, n)
    sum_viol = abs(np.sum(h) - target) * penalty
    bound_viol = (np.sum(np.maximum(0, -h)) + np.sum(np.maximum(0, h - 1))) * penalty
    return base + sum_viol + bound_viol

    t0 = time.time()

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)

    h0 = project_feasible(h0, n)
    
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, n)
    
    def elapsed():
        return time.time() - t0
    
    # Phase 1: Nelder-Mead

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = h.astype(float).copy()
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt == 0:
            mask = np.ones(n, dtype=bool)
            cnt = n
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_with_penalty(h, n, pen=50.0):
    base = compute_c5(h, n)
    target = n / 2.0
    sv = abs(np.sum(h) - target)
    bv = np.sum(np.maximum(0.0, -h)) + np.sum(np.maximum(0.0, h - 1.0))
    return base + pen * (sv + bv)

def coordinate_refine(h, n, max_rounds=300, rng=None):

            for d in dets:

                h[j] += d

                else:
