        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_new, endpoint=False)
        h135 = np.interp(x_new, x_old, best_h)
        h135 = project_feasible(h135, n_new * 0.5)
        h135, c135 = optimize_pg(h135, t_end - 10, rng, T0=100, T1=800, max_iter=20000)
        if c135 < best_c5:
            best_c5 = c135
            best_h = h135

    # Final polish: coordinate refine at best resolution
    if time.time() < t_end - 2:
        h, c = coordinate_refine(best_h, t_end - 1, rng)
        if c < best_c5:
            best_c5 = c
            best_h = h

    # Ensure feasibility
    best_h = np.clip(best_h, 0.0, 1.0)
    s = np.sum(best_h)
    target = len(best_h) * 0.5
    if abs(s - target) > 1e-10:
        best_h = project_feasible(best_h, target)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
