import numpy as np
from scipy.optimize import minimize
import time


def compute_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        h += diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if diff > 1e-10:
        room = 1.0 - h
        idx = np.argsort(-room)
        for i in idx:
            add = min(diff, room[i])
            h[i] += add
            diff -= add
            if diff <= 1e-13:
                break
    elif diff < -1e-10:
        diff = -diff
        idx = np.argsort(h)
        for i in idx:
            sub = min(diff, h[i])
            h[i] -= sub
            diff -= sub
            if diff <= 1e-13:
                break
    return h


def integral_constraint(h, n):
    return np.sum(h) - n / 2.0
