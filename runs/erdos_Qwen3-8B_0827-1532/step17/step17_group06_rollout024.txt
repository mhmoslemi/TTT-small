from scipy.optimize import minimize
import time

    # Generate diverse initial guesses

    # Step function: first half 1, second half 0

    # Alternating 1 and 0

    initial_guesses.append(np.random.uniform(0, 1, n_points))

    initial_guesses.append(h_binary)
    
    # Periodic patterns with different spacing
    # 0.4 spacing
    h_periodic_04 = np.zeros(n_points)
    for i in range(n_points):
        pos = i * dx
        if (pos // 0.4) % 2 == 0:
            h_periodic_04[i] = 1.0
        else:
            h_periodic_04[i] = 0.0
    sum_h = np.sum(h_periodic_04)
    scale = target_sum / sum_h
    initial_guesses.append(h_periodic_04 * scale)
    
    # 0.2 spacing
    h_periodic_02 = np.zeros(n_points)
    for i in range(n_points):
        pos = i * dx
        if (pos // 0.2) % 2 == 0:
            h_periodic_02[i] = 1.0
        else:
            h_periodic_02[i] = 0.0
    sum_h = np.sum(h_periodic_02)
    scale = target_sum / sum_h
    initial_guesses.append(h_periodic_02 * scale)
    
    # Optimized initial guess with COBYLA
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    
    for guess in initial_guesses:
        # Use COBYLA for better constraint handling
        result = minimize(
            objective,
            guess,
            method='COBYLA',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Perturbation phase with adaptive noise
    num_perturbations = 500
    phase_shift = 0.25
    phase_decay = 0.99
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Use SLSQP for refinement
        result = minimize(
