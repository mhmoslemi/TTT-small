from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    start_time = time.time()
    
    # Generate diverse initial guesses
    initial_guesses = []
    initial_guesses.append(initial_guess.copy())  # Original initial guess
    
    # Step function: first half 1, second half 0
    h_step = np.zeros(n_points)
    h_step[:n_points//2] = 1.0
    sum_h = np.sum(h_step)
    scale = target_sum / sum_h
    initial_guesses.append(h_step * scale)
    
    # Alternating 1 and 0
    h_alt = np.zeros(n_points)
    h_alt[::2] = 1.0
    sum_h = np.sum(h_alt)
    scale = target_sum / sum_h

    # Random binary sequence

    initial_guesses.append(h_binary)

    periods = [0.2, 0.25, 0.3, 0.4, 0.5]
    for period in periods:
        h_periodic = np.zeros(n_points)
        for i in range(n_points):
            pos = i * dx
            if (pos // period) % 2 == 0:
                h_periodic[i] = 1.0
            else:
                h_periodic[i] = 0.0
        sum_h = np.sum(h_periodic)
        scale = target_sum / sum_h
        initial_guesses.append(h_periodic * scale)

    initial_guesses.append(np.random.uniform(0, 1, n_points))

    for guess in initial_guesses:
        # Objective function
        def objective(h):

        bounds = [(0.0, 1.0) for _ in range(n_points)]
