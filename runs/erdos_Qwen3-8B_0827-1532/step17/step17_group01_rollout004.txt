    initial_guesses.append(h_step * scale)
    
    # Alternating 1 and 0
    h_alt = np.zeros(n_points)
    h_alt[::2] = 1.0
    sum_h = np.sum(h_alt)
    scale = target_sum / sum_h
    initial_guesses.append(h_alt * scale)
    
    # Central 50% step function
    h_center = np.zeros(n_points)
    center_start = n_points // 2 - n_points // 4
    center_end = n_points // 2 + n_points // 4
    h_center[center_start:center_end] = 1.0
    sum_h = np.sum(h_center)
    scale = target_sum / sum_h
    initial_guesses.append(h_center * scale)
    
    # Two-step function (25% each)
    h_two_steps = np.zeros(n_points)
    h_two_steps[:n_points//4] = 1.0
    h_two_steps[n_points//2:n_points//2 + n_points//4] = 1.0
    sum_h = np.sum(h_two_steps)
    scale = target_sum / sum_h
    initial_guesses.append(h_two_steps * scale)
    
    # Four-step function (25% each)
    h_four_steps = np.zeros(n_points)
    h_four_steps[:n_points//4] = 1.0
    h_four_steps[n_points//2:n_points//2 + n_points//4] = 1.0
    h_four_steps[3*n_points//4: ] = 1.0
    sum_h = np.sum(h_four_steps)
    scale = target_sum / sum_h
    initial_guesses.append(h_four_steps * scale)
    
    # Random step function with 3 segments
    h_random_steps = np.zeros(n_points)
    num_segments = 3
    total_length = 0
    for _ in range(num_segments):
        length = np.random.randint(1, n_points // 2)
        start = np.random.randint(0, n_points - length)
        h_random_steps[start:start+length] = 1.0
        total_length += length
    sum_h = np.sum(h_random_steps)
    scale = target_sum / sum_h
    initial_guesses.append(h_random_steps * scale)
    
    # Random initial guess
    initial_guesses.append(np.random.uniform(0, 1, n_points))
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 1000}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Perturbation phase on the best_h found so far
    num_perturbations = 300  # Increased from 200
    phase_shift = 0.25
    phase_decay = 0.98
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 500}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
