def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate diverse initial guesses
    initial_guesses = []
    initial_guesses.append(initial_guess.copy())  # Original initial guess
    
    # Step function: first half 1, second half 0
    h_step = np.zeros(n_points)
    h_step[:n_points//2] = 1.0
    sum_h = np.sum(h_step)
    scale = target_sum / sum_h
    initial_guesses.append(h_step * scale)
    
    # Alternating 1 and 0
    h_alt = np.zeros(n_points)
    h_alt[::2] = 1.0
    sum_h = np.sum(h_alt)
    scale = target_sum / sum_h
    initial_guesses.append(h_alt * scale)
    
    # Random initial guess
    initial_guesses.append(np.random.uniform(0, 1, n_points))
    
    # Sinusoidal initial guess
    h_sinusoidal = 0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / n_points)
    sum_h = np.sum(h_sinusoidal)
    scale = target_sum / sum_h
    initial_guesses.append(h_sinusoidal * scale)
    
    # Optimize each initial guess with differential evolution
    for guess in initial_guesses:
        # Define bounds
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Objective function with penalty
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            c5 = np.max(corr) * dx
            penalty = 1e2 * (np.sum(h) - target_sum)**2
            return c5 + penalty
        
        # Run differential evolution
        result = differential_evolution(
            objective,
            bounds,
            popsize=10,
            maxiter=100,
            tol=1e-10,
            mutation=(0.5, 1),
            recombination=0.9,
            strategy='best1bin',
            polish=True,
            disp=False
        )
        
        # Check if result is successful
        if result.success:
            current_c5 = result.fun - 1e2 * (np.sum(result.x) - target_sum)**2
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # Perturbation phase on the best_h found so far
    num_perturbations = 100  # Reduced from 1000 to save time
    phase_shift = 0.2
    phase_decay = 0.99
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function with penalty
        def objective_perturb(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            c5 = np.max(corr) * dx
            penalty = 1e2 * (np.sum(h) - target_sum)**2
            return c5 + penalty
        
        # Run differential evolution on perturbed solution
        result_perturb = differential_evolution(
            objective_perturb,
            bounds,
            popsize=10,
            maxiter=50,
            tol=1e-10,
            mutation=(0.5, 1),
            recombination=0.9,
            strategy='best1bin',
            polish=True,
            disp=False
        )
        
        if result_perturb.success:
            current_c5_perturb = result_perturb.fun - 1e2 * (np.sum(result_perturb.x) - target_sum)**2
            if current_c5_perturb < best_c5:
                best_c5 = current_c5_perturb
                best_h = result_perturb.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
