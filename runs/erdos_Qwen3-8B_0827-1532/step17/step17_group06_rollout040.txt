        bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]
        cons_slsqp = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result_slsqp = minimize(
            objective_slsqp,
            best_h_de,
            method='SLSQP',
            bounds=bounds_slsqp,
            constraints=cons_slsqp,
            options={'ftol': 1e-10, 'maxiter': 500}
        )
        
        if result_slsqp.success and result_slsqp.fun < best_c5:
            best_c5 = result_slsqp.fun
            best_h = result_slsqp.x
    except:
        pass  # Proceed with other methods if DE fails
    
    # Try initial_h_values
    result_initial = minimize(
        lambda h: np.max(np.correlate(h, 1 - h, mode='full')) * dx,
        initial_h_values,
        method='SLSQP',
        bounds=bounds_slsqp,
        constraints=cons_slsqp,
        options={'ftol': 1e-10, 'maxiter': 500}
    )
    
    if result_initial.success and result_initial.fun < best_c5:
        best_c5 = result_initial.fun
        best_h = result_initial.x
    
    # Try step function
    h_step = np.zeros(n_points)
    h_step[:n_points//2] = 1.0
    sum_h = np.sum(h_step)
    scale = target_sum / sum_h
    h_step_scaled = h_step * scale
    result_step = minimize(
        lambda h: np.max(np.correlate(h, 1 - h, mode='full')) * dx,
        h_step_scaled,
        method='SLSQP',
        bounds=bounds_slsqp,
        constraints=cons_slsqp,
        options={'ftol': 1e-10, 'maxiter': 500}
    )
    
    if result_step.success and result_step.fun < best_c5:
        best_c5 = result_step.fun
        best_h = result_step.x
    
    # Perturbation phase
    num_perturbations = 500
    phase_shift = 0.25
    phase_decay = 0.99
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        result = minimize(
            lambda h: np.max(np.correlate(h, 1 - h, mode='full')) * dx,
            perturbed,
            method='SLSQP',
            bounds=bounds_slsqp,
            constraints=cons_slsqp,
            options={'ftol': 1e-10, 'maxiter': 500}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
