from scipy.optimize import minimize

            x = i * dx
            if (x >= shift and x < shift + 0.5) or (x >= shift + 1.0 and x < shift + 1.5):
                h[i] = 1.0
        initial_guesses.append(h)
    # Add random guesses
    for _ in range(3):
        h = np.random.uniform(0, 1, n_points)

        initial_guesses.append(h)

    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx

        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
