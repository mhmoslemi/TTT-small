from scipy.optimize import minimize

    sum_h = np.sum(h_step)
    scale = target_sum / sum_h
    initial_guesses.append(h_step * scale)
    
    # Alternating 1 and 0
    h_alt = np.zeros(n_points)
    h_alt[::2] = 1.0

    scale = target_sum / sum_h
