from scipy.optimize import minimize
import time

    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate diverse initial guesses based on initial_h_values
    initial_guesses = [initial_guess.copy()]
    
    # Step function: first half 1, second half 0

    h_step[:n_points//2] = 1.0
    sum_h = np.sum(h_step)
    scale = target_sum / sum_h
    initial_guesses.append(h_step * scale)
    
    # Alternating 1 and 0
    h_alt = np.zeros(n_points)
    h_alt[::2] = 1.0
    sum_h = np.sum(h_alt)
    scale = target_sum / sum_h
    initial_guesses.append(h_alt * scale)
    
    # Random initial guess
    initial_guesses.append(np.random.uniform(0, 1, n_points))
    
    # Binary shuffled pattern
    h_binary = np.random.permutation([1]*50 + [0]*50)
    initial_guesses.append(h_binary)
    
    # Periodic patterns with varying spacing
    for spacing in [0.2, 0.3, 0.4]:
        h_periodic = np.zeros(n_points)
        for i in range(n_points):
            pos = i * dx
            if (pos // spacing) % 2 == 0:
                h_periodic[i] = 1.0
            else:
                h_periodic[i] = 0.0
        sum_h = np.sum(h_periodic)
        scale = target_sum / sum_h
        initial_guesses.append(h_periodic * scale)
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 300}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Perturbation phase with targeted adjustments

    phase_shift = 0.25

        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale

        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx

        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
