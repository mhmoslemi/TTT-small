        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Compute overlaps for the perturbed h
        corr = np.correlate(perturbed, 1.0 - perturbed, mode='full')
        max_corr_idx = np.argmax(corr)
        worst_overlap = corr[max_corr_idx] * dx
        
        # Define objective to minimize the worst overlap
        def objective_perturb(h):
            one_minus_h = 1.0 - h
            corr_perturb = np.correlate(h, one_minus_h, mode='full')
            return corr_perturb[max_corr_idx] * dx
        
        # Optimization with COBYLA
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective_perturb,
            perturbed,
            method='COBYLA',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 200}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
