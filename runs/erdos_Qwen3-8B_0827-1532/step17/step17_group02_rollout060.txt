            x = i * dx
            if (x >= shift and x < shift + 0.5) or (x >= shift + 1.0 and x < shift + 1.5):
                h[i] = 1.0
        initial_guesses.append(h)
    # Add structured patterns with different numbers of intervals
    for num_intervals in [2, 3, 4]:

            h = np.zeros(n_points)

                start = shift + i * interval_length
                end = start + interval_length
                for j in range(n_points):
                    x = j * dx
                    if start <= x < end:
                        h[j] = 1.0
            initial_guesses.append(h)

        bounds = [(0.0, 1.0) for _ in range(n_points)]
