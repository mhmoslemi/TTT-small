import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate diverse initial guesses
    initial_guesses = [initial_guess.copy()]
    
    # Symmetric step function: 1 in central 50%
    h_sym = np.zeros(n_points)
    center_start = n_points // 2 - n_points // 4
    center_end = n_points // 2 + n_points // 4
    h_sym[center_start:center_end] = 1.0
    sum_h = np.sum(h_sym)
    scale = target_sum / sum_h
    initial_guesses.append(h_sym * scale)
    
    # Alternating 1 and 0 with offset
    h_alt = np.zeros(n_points)
    h_alt[::2] = 1.0
    sum_h = np.sum(h_alt)
    scale = target_sum / sum_h
    initial_guesses.append(h_alt * scale)
    
    # Random initial guess with uniform distribution
    initial_guesses.append(np.random.uniform(0, 1, n_points))
    
    # Optimized initial guess with COBYLA
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    
    for guess in initial_guesses:
        result = minimize(
            objective,
            guess,
            method='COBYLA',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 1000, 'ftol': 1e-9}
        )
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Perturbation phase with adaptive scaling
    num_perturbations = 300
    phase_shift = 0.2
    phase_decay = 0.98
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        result = minimize(
            objective,
            perturbed,
            method='COBYLA',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 300, 'ftol': 1e-9}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
