        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    def de_constraint(h):
        return np.sum(h) - target_sum
    
    # Bounds and constraints for DE
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': de_constraint})
    
    # Run differential evolution
    result_de = differential_evolution(
        de_objective,
        bounds,
        constraints=cons,
        popsize=15,
        mutation=(0.5, 1.0),
        recombination=0.9,
        tol=1e-6,
        maxiter=100,
        disp=False
    )
    
    if result_de.success and result_de.fun < best_c5:
        best_c5 = result_de.fun
        best_h = result_de.x
    
    # Perturbation phase on the best_h found so far
    num_perturbations = 300  # Reduced from 500
    phase_shift = 0.25
    phase_decay = 0.98
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 300}
        )
        
        # Fallback on failure
        if not result.success:
            result = minimize(
                objective,
                best_h,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'ftol': 1e-9, 'maxiter': 300}
            )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
