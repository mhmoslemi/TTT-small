        maxiter=100,
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        polish=True,
        seed=seed
    )
    
    # Update best solution
    if result.success:
        best_h = result.x
        best_c5 = result.fun
    
    # Perturbation phase on the best_h found so far
    num_perturbations = 200
    phase_shift = 0.15
    phase_decay = 0.96
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective_perturbed(h):
            sum_h = np.sum(h)
            if sum_h != 0:
                h_scaled = h * (target_sum / sum_h)
            else:
                h_scaled = h
            one_minus_h = 1.0 - h_scaled
            corr = np.correlate(h_scaled, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Run local optimization with SLSQP
        bounds_perturb = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result_perturb = minimize(
            objective_perturbed,
            perturbed,
            method='SLSQP',
            bounds=bounds_perturb,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 200}
        )
        
        if result_perturb.success and result_perturb.fun < best_c5:
            best_c5 = result_perturb.fun
            best_h = result_perturb.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
