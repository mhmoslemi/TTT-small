            h_sinusoidal[i] = 1.0
        else:
            h_sinusoidal[i] = 0.0
    sum_h = np.sum(h_sinusoidal)

    for guess in initial_guesses:

        def objective(h):
            one_minus_h = 1.0 - h

            return np.max(corr) * dx

        bounds = [(0.0, 1.0) for _ in range(n_points)]
