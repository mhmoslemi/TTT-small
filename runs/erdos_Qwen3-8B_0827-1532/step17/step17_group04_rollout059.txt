            return np.max(corr) * dx
        
        # Optimization with Nelder-Mead
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='Nelder-Mead',
            bounds=bounds,
            constraints=cons,
            options={'xatol': 1e-8, 'maxiter': 150}
        )
        
        # Fallback on failure
        if not result.success:
            result = minimize(
                objective,
                best_h,
                method='Nelder-Mead',
                bounds=bounds,
                constraints=cons,
                options={'xatol': 1e-8, 'maxiter': 150}
            )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
