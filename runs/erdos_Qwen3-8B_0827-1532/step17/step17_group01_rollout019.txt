                mutation=(0.5, 1),
                recombination=0.8,
                seed=seed
            )
            
            if result.success:
                current_h = result.x
                current_c5 = objective(current_h) - 1e5 * abs(np.sum(current_h) - target_sum)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h
    
    return best_h, best_c5, n_points
