def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate diverse initial guesses
    initial_guesses = []
    initial_guesses.append(initial_guess.copy())  # Original initial guess
    
    # Uniform distribution
    uniform = np.full(n_points, 0.5)
    uniform = uniform * (target_sum / np.sum(uniform))
    initial_guesses.append(uniform)
    
    # Blocky pattern (alternating 1 and 0)
    blocky = np.zeros(n_points)
    blocky[::2] = 1.0
    blocky = blocky * (target_sum / np.sum(blocky))
    initial_guesses.append(blocky)
    
    # Random pattern
    random = np.random.uniform(0, 1, n_points)
    random = random * (target_sum / np.sum(random))
    initial_guesses.append(random)
    
    # Half pattern (first half is 1)
    half = np.zeros(n_points)
    half[:n_points//2] = 1.0
    half = half * (target_sum / np.sum(half))
    initial_guesses.append(half)
    
    # Two blocks pattern (first and last quarter are 1)
    two_blocks = np.zeros(n_points)
    two_blocks[:n_points//4] = 1.0
    two_blocks[-n_points//4:] = 1.0
    two_blocks = two_blocks * (target_sum / np.sum(two_blocks))
    initial_guesses.append(two_blocks)
    
    # Sinusoidal pattern
    def sinusoidal_pattern(n_points):
        x = np.linspace(0, 2, n_points)
        h = 0.5 + 0.5 * np.sin(np.pi * x)
        target_sum = n_points / 2
        current_sum = np.sum(h)
        scale = target_sum / current_sum
        return h * scale
    initial_guesses.append(sinusoidal_pattern(n_points))
    
    # Optimize each initial guess with differential evolution and SLSQP
    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Differential evolution for global search
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result_de = differential_evolution(objective, bounds, strategy='best1bin', maxiter=100, popsize=15, tol=1e-6, mutation=(0.5, 1), recombination=0.8, disp=False)
        
        # Refine with SLSQP
        result_slsqp = minimize(
            objective,
            result_de.x,
            method='SLSQP',
            bounds=bounds,
            constraints=({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}),
            options={'ftol': 1e-9, 'maxiter': 500}
        )
        
        if result_slsqp.success and result_slsqp.fun < best_c5:
            best_c5 = result_slsqp.fun
            best_h = result_slsqp.x
    
    # Perturbation phase on the best_h found so far
    num_perturbations = 500  # Increased from 300
    phase_shift = 0.25
    phase_decay = 0.98
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 300}
        )
        
        # Fallback on failure
        if not result.success:
            result = minimize(
                objective,
                best_h,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'ftol': 1e-9, 'maxiter': 300}
            )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
