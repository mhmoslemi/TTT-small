    x = np.linspace(0, 2, n_points)
    sine_wave = 0.5 + 0.5 * np.sin(np.pi * x)
    sine_wave = sine_wave * (target_sum / np.sum(sine_wave))
    initial_guesses.append(sine_wave)
    
    # Optimize each initial guess
    for guess in initial_guesses:
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': constraint})
        
        # Use differential evolution for global search
        result = differential_evolution(
            objective,
            bounds,
            constraints=cons,
            popsize=15,
            mutation=(0.5, 1.0),
            recombination=0.9,
            tol=1e-8,
            maxiter=100
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Perturbation phase
    num_perturbations = 300
    phase_shift = 0.15
    phase_decay = 0.98
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': constraint})
        
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 500}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
