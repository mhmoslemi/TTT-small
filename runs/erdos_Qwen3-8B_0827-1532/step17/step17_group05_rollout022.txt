    binary_h = np.zeros(n_points)
    for i in range(n_points):
        if (x[i] >= 0 and x[i] <= 1/3) or (x[i] >= 1 and x[i] <= 4/3) or (x[i] >= 5/3 and x[i] <= 2):
            binary_h[i] = 1.0
    binary_h = binary_h * (target_sum / np.sum(binary_h))
    initial_guesses.append(binary_h)
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Objective function with penalty for sum constraint
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr) * dx
            penalty = 1e4 * (np.sum(h) - target_sum)**2
            return max_corr + penalty
        
        # Bounds and constraints
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Use differential evolution for global optimization
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=100,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            polish=True
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Perturbation phase on the best_h found so far
    num_perturbations = 500  # Increased from 300
    phase_shift = 0.15
    phase_decay = 0.98
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 500}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
