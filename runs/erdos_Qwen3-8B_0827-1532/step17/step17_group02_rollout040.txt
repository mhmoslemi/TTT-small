                if (x >= shift and x < shift + width) or (x >= shift + 1.0 and x < shift + 1.0 + width):
                    h[i] = 1.0

    # Add random guesses

        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')

        result = minimize(
            objective,
            guess,
            method='SLSQP',
