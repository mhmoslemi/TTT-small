    start_time = time.time()
    
    # Initial guess: provided initial_h_values
    initial_guess = initial_h_values
    initial_guess = np.clip(initial_guess, 0, 1)
    scale = target_sum / np.sum(initial_guess)
    initial_guess = initial_guess * scale
    best_h = initial_guess.copy()
    best_c5 = compute_c5(best_h, dx)
    
    # Generate diverse initial guesses
    initial_guesses = [initial_guess]
    
    # Step function: first half 1, second half 0
    h_step = np.zeros(n_points)
    h_step[:n_points//2] = 1.0
    sum_h = np.sum(h_step)
    scale = target_sum / sum_h
    initial_guesses.append(h_step * scale)
    
    # Alternating 1 and 0
    h_alt = np.zeros(n_points)
    h_alt[::2] = 1.0
    sum_h = np.sum(h_alt)
    scale = target_sum / sum_h
    initial_guesses.append(h_alt * scale)
    
    # Central 50% step function
    h_center = np.zeros(n_points)
    center_start = n_points // 2 - n_points // 4
    center_end = n_points // 2 + n_points // 4
    h_center[center_start:center_end] = 1.0
    sum_h = np.sum(h_center)
    scale = target_sum / sum_h
    initial_guesses.append(h_center * scale)
    
    # 25% step function
    h_25 = np.zeros(n_points)
    h_25[:n_points//4] = 1.0
    h_25[n_points//2:n_points//2 + n_points//4] = 1.0
    sum_h = np.sum(h_25)
    scale = target_sum / sum_h
    initial_guesses.append(h_25 * scale)
    
    # Random initial guess
    initial_guesses.append(np.random.uniform(0, 1, n_points))
    
    # Generate more diverse initial guesses with different patterns
    for _ in range(5):
        h_rand = np.random.uniform(0, 1, n_points)
        sum_h = np.sum(h_rand)
        scale = target_sum / sum_h
        initial_guesses.append(h_rand * scale)
    
    # Three-segment pattern
    h_three = np.zeros(n_points)
    segment_length = n_points // 3
    h_three[:segment_length] = 1.0
    h_three[segment_length:2*segment_length] = 0.5
    h_three[2*segment_length:] = 1.0
    sum_h = np.sum(h_three)
    scale = target_sum / sum_h
    initial_guesses.append(h_three * scale)
    
    # Three intervals spaced apart
    h_three_intervals = np.zeros(n_points)
    interval_length = 0.3333
    start = 0.0
    for i in range(3):
        end = start + interval_length
        start_idx = int(start / dx)
        end_idx = int(end / dx)
        h_three_intervals[start_idx:end_idx] = 1.0
        start = end
    sum_h = np.sum(h_three_intervals)
    scale = target_sum / sum_h
    initial_guesses.append(h_three_intervals * scale)
    
    # Random segments with varying heights
    h_rand_seg = np.zeros(n_points)
    for i in range(n_points):
        if np.random.rand() < 0.3:
            h_rand_seg[i] = 1.0
        elif np.random.rand() < 0.5:
            h_rand_seg[i] = 0.5
    sum_h = np.sum(h_rand_seg)
    scale = target_sum / sum_h
    initial_guesses.append(h_rand_seg * scale)
    
    # Four-segment pattern
    h_four = np.zeros(n_points)
    segment_length = n_points // 4
    h_four[:segment_length] = 1.0
    h_four[segment_length:2*segment_length] = 0.5
    h_four[2*segment_length:3*segment_length] = 1.0
    h_four[3*segment_length:] = 0.5
    sum_h = np.sum(h_four)
    scale = target_sum / sum_h
    initial_guesses.append(h_four * scale)
    
    # Three-segment with alternating values
    h_three_alt = np.zeros(n_points)
    segment_length = n_points // 3
    h_three_alt[:segment_length] = 1.0
    h_three_alt[segment_length:2*segment_length] = 0.0
    h_three_alt[2*segment_length:] = 1.0
    sum_h = np.sum(h_three_alt)
    scale = target_sum / sum_h
    initial_guesses.append(h_three_alt * scale)
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 2000}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Perturbation phase on the best_h found so far
    num_perturbations = 300
    phase_shift = 0.15
    phase_decay = 0.96
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 700}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points

def compute_c5(h, dx):
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return np.max(corr) * dx
