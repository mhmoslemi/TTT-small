from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate diverse initial guesses with varying segment counts and random placements
    initial_guesses = [initial_guess.copy()]
    
    # Step function with different segment counts
    for num_segments in [5, 10, 15]:
        h_step = np.zeros(n_points)
        segment_length = 2.0 / num_segments
        for i in range(num_segments):
            start = int(i * segment_length / dx)
            end = int((i + 1) * segment_length / dx)
            h_step[start:end] = 1.0
        sum_h = np.sum(h_step)
        scale = target_sum / sum_h
        initial_guesses.append(h_step * scale)
    
    # Random initial guess with more structure
    h_random = np.random.uniform(0, 1, n_points)
    sum_h = np.sum(h_random)
    scale = target_sum / sum_h
    initial_guesses.append(h_random * scale)
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-12, 'maxiter': 1000}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Perturbation phase with more diverse strategies
    num_perturbations = 300  # Increased to explore more
    phase_shift = 0.15

    for perturbation_idx in range(num_perturbations):

        # Apply structured perturbations: shift, scale, add noise
        perturbed = best_h.copy()
        shift = np.random.uniform(-0.1, 0.1)
        shifted_h = np.roll(perturbed, int(shift / dx))
        shifted_h = np.clip(shifted_h, 0, 1)
        current_sum = np.sum(shifted_h)
        scale = target_sum / current_sum
        perturbed = shifted_h * scale
        
        # Add random noise
        noise = np.random.normal(0, noise_scale, n_points)
        perturbed = np.clip(perturbed + noise, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
