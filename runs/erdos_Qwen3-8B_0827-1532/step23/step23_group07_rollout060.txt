            start = int(i * segment_length / dx)
            end = int((i + 1) * segment_length / dx)
            h_step[start:end] = 1.0
        sum_h = np.sum(h_step)
        scale = target_sum / sum_h
        initial_guesses.append(h_step * scale)
    
    # Random initial guess with some structure

    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx

        bounds = [(0.0, 1.0) for _ in range(n_points)]
