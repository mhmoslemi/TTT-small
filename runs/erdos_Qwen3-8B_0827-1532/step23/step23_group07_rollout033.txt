    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 2000}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Perturbation phase with diverse strategies
    num_perturbations = 300  # Increased to explore more
    phase_shift = 0.15
    phase_decay = 0.98
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        # Apply structured perturbations: shift, scale, add noise
        perturbed = best_h.copy()
        shift = np.random.uniform(-0.1, 0.1)
        shifted_h = np.roll(perturbed, int(shift / dx))
        shifted_h = np.clip(shifted_h, 0, 1)
        current_sum = np.sum(shifted_h)
        scale = target_sum / current_sum
        perturbed = shifted_h * scale
        
        # Add random noise
        noise = np.random.normal(0, noise_scale, n_points)
        perturbed = np.clip(perturbed + noise, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 500}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
