    population.append(h_three * scale)
    
    # Triangular function
    h_tri = np.zeros(n_points)
    for i in range(n_points):
        x = i * dx
        if x < 1.0:
            h_tri[i] = x / 1.0
        elif x < 2.0:
            h_tri[i] = (2.0 - x) / 1.0
        else:
            h_tri[i] = 0.0
    sum_h = np.sum(h_tri)
    scale = target_sum / sum_h
    population.append(h_tri * scale)
    
    # Random initial guess
    h_rand = np.random.uniform(0, 1, n_points)
    sum_h = np.sum(h_rand)
    scale = target_sum / sum_h
    population.append(h_rand * scale)
    
    # Generate more diverse initial guesses with different patterns
    for _ in range(3):
        h_rand = np.random.uniform(0, 1, n_points)
        sum_h = np.sum(h_rand)
        scale = target_sum / sum_h
        population.append(h_rand * scale)
    
    # Add random step functions with varying numbers of segments
    for m in [3, 4, 5]:
        for _ in range(2):
            h_step = np.zeros(n_points)
            lengths = np.random.rand(m)
            lengths /= np.sum(lengths)
            lengths *= (n_points / 2)
            current = 0.0
            for l in lengths:
                start = current
                end = current + l
                start_idx = int(start / dx)
                end_idx = int(end / dx)
                start_idx = max(0, start_idx)
                end_idx = min(n_points, end_idx)
                h_step[start_idx:end_idx] = 1.0
                current = end
            sum_h = np.sum(h_step)
            scale = target_sum / sum_h
            population.append(h_step * scale)
    
    # Now, perform genetic algorithm
    generations = 0
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        # Evaluate fitness for all individuals
        fitness = []
        for h in population:
            corr = np.correlate(h, 1.0 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
        
        # Find the best individual
        best_idx = np.argmin(fitness)
        best_c5_current = fitness[best_idx]
        best_h_current = population[best_idx].copy()
        
        if best_c5_current < best_c5:
            best_c5 = best_c5_current
            best_h = best_h_current
        
        # Select parents for next generation
        # Select top 20% individuals
        top_indices = np.argsort(fitness)[:int(len(population)*0.2)]
        selected_parents = [population[i] for i in top_indices]
        
        # Generate new population
        new_population = []
        while len(new_population) < population_size:
            # Select two parents
            parent1 = np.random.choice(selected_parents)
            parent2 = np.random.choice(selected_parents)
            
            # Crossover
            child = (parent1 + parent2) / 2
            # Normalize
            current_sum = np.sum(child)
            scale = target_sum / current_sum
            child *= scale
            new_population.append(child)
            
            # Mutation
            if np.random.rand() < 0.1:
                # Mutate
                mutated = child.copy()
                indices = np.random.choice(n_points, size=int(n_points*0.1), replace=False)
                for i in indices:
                    mutated[i] += np.random.uniform(-0.1, 0.1)
                    mutated[i] = np.clip(mutated[i], 0, 1)
                current_sum = np.sum(mutated)
                scale = target_sum / current_sum
                mutated *= scale
                new_population.append(mutated)
        
        population = new_population
        generations += 1
    
    return best_h, best_c5, n_points

def compute_c5(h, dx):
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return np.max(corr) * dx
