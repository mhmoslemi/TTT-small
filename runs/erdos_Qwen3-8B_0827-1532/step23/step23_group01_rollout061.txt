    h_three_seg = np.zeros(n_points)
    h_three_seg[seg1_start:seg1_end] = 1.0
    h_three_seg[seg2_start:seg2_end] = 1.0
    h_three_seg[seg3_start:seg3_end] = 1.0
    sum_h = np.sum(h_three_seg)
    scale = target_sum / sum_h
    initial_guesses.append(h_three_seg * scale)
    
    # Sine-like function
    h_sine = np.zeros(n_points)
    for i in range(n_points):
        h_sine[i] = 0.5 * (1 + np.sin(np.pi * (i - n_points/2) / (n_points/2)))
    sum_h = np.sum(h_sine)
    scale = target_sum / sum_h
    initial_guesses.append(h_sine * scale)
    
    # Random binary initial guess
    h_random_binary = np.random.choice([0, 1], size=n_points)
    sum_h = np.sum(h_random_binary)
    scale = target_sum / sum_h
    initial_guesses.append(h_random_binary * scale)
    
    # Random initial guess
    initial_guesses.append(np.random.uniform(0, 1, n_points))
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-12, 'maxiter': 2000}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Perturbation phase with varied perturbations
    num_perturbations = 200  # Increased to explore more
    phase_shift = 0.15
    phase_decay = 0.98
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        # Apply random noise and scaling
        perturbed = best_h.copy()
        noise = np.random.normal(0, noise_scale, n_points)
        perturbed += noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-12, 'maxiter': 300}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
