            return np.max(corr) * dx
        
        # Bounds for differential evolution
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Use differential evolution with penalty for sum constraint
        def objective_with_penalty(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr) * dx
            sum_h = np.sum(h)
            penalty = 1e6 * abs(sum_h - target_sum)
            return max_corr + penalty
        
        # Run differential evolution
        result = differential_evolution(
            objective_with_penalty,
            bounds,
            strategy='best1bin',
            popsize=15,
            mutation=(0.5, 1),
            recombination=0.9,
            tol=1e-8,
            maxiter=100,
            disp=False
        )
        
        if result.success:
            current_h = result.x
            current_c5 = objective(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
    
    # Perturbation phase with structured perturbations
    num_perturbations = 200  # Increased to explore more
    phase_shift = 0.15
    phase_decay = 0.98
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        # Apply structured perturbations: shift and scale
        perturbed = best_h.copy()
        shift = np.random.uniform(-0.1, 0.1)
        shifted_h = np.roll(perturbed, int(shift / dx))
        shifted_h = np.clip(shifted_h, 0, 1)
        current_sum = np.sum(shifted_h)
        scale = target_sum / current_sum
        perturbed = shifted_h * scale
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Run differential evolution on perturbed guess
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=15,
            mutation=(0.5, 1),
            recombination=0.9,
            tol=1e-8,
            maxiter=100,
            disp=False
        )
        
        if result.success:
            current_h = result.x
            current_c5 = objective(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
