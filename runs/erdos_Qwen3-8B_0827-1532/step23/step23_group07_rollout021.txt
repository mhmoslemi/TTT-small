            end = int((i + 1) * segment_length / dx)
            h_step[start:end] = 1.0
        sum_h = np.sum(h_step)
        scale = target_sum / sum_h
        initial_guesses.append(h_step * scale)

    # Optimize each initial guess with SLSQP
    for guess in initial_guesses:
        # Objective function
        def objective(h):

        bounds = [(0.0, 1.0) for _ in range(n_points)]
