    sum_h = np.sum(h_4seg)
    scale = target_sum / sum_h
    initial_guesses.append(h_4seg * scale)
    
    # 5-segment step function
    h_5seg = np.zeros(n_points)
    h_5seg[:n_points//5] = 1.0
    h_5seg[2*n_points//5:3*n_points//5] = 1.0
    h_5seg[4*n_points//5:] = 1.0
    sum_h = np.sum(h_5seg)
    scale = target_sum / sum_h
    initial_guesses.append(h_5seg * scale)
    
    # Random initial guess
    initial_guesses.append(np.random.uniform(0, 1, n_points))
    
    # Generate more diverse initial guesses with different patterns
    for _ in range(3):
        h_rand = np.random.uniform(0, 1, n_points)
        sum_h = np.sum(h_rand)
        scale = target_sum / sum_h
        initial_guesses.append(h_rand * scale)
    
    # Optimize each initial guess with differential evolution
    for guess in initial_guesses:
        # Objective function with penalty for constraint violation
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr) * dx
            penalty = 1e6 * abs(np.sum(h) - target_sum)
            return max_corr + penalty
        
        # Bounds
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Run differential evolution
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=15,
            tol=1e-8,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed,
            maxiter=100
        )
        
        if result.success:
            current_h = result.x
            current_c5 = compute_c5(current_h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
    
    # Perturbation phase on the best_h found so far
    num_perturbations = 300
    phase_shift = 0.25
    phase_decay = 0.98
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale

            return max_corr + penalty
        
        # Bounds
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Run differential evolution for perturbed initial guess
        result = differential_evolution(
            objective,

        elapsed = time.time() - start_time
        if elapsed >= budget_s:

    return best_h, best_c5, n_points

def compute_c5(h, dx):
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return np.max(corr) * dx
