from scipy.optimize import minimize
import time

    uniform = np.full(n_points, 0.5)
    initial_guesses.append(uniform)

    blocky[::2] = 1.0
    initial_guesses.append(blocky)
    # Random pattern
    random = np.random.uniform(0, 1, n_points)
    initial_guesses.append(random)
    # Binary pattern (alternating 1 and 0, scaled)
    binary = np.zeros(n_points)
    binary[::2] = 1.0
    initial_guesses.append(binary)
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx

        bounds = [(0.0, 1.0) for _ in range(n_points)]

            method='COBYLA',

            constraints=cons,

        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h

            return np.max(corr) * dx
