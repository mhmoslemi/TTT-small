import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Define the objective function
    def objective(h):
        # Normalize h to have the correct sum
        total = np.sum(h)
        if total == 0:
            return float('inf')

        one_minus_h = 1.0 - h_norm
        corr = np.correlate(h_norm, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    # Bounds for each parameter
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Run differential evolution
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=20,
        tol=1e-5,
        mutation=(0.5, 1),
        recombination=0.5,
        seed=seed,
        maxiter=100
    )
    
    # Update best solution
    best_h = result.x
    best_c5 = result.fun
    
    elapsed = time.time() - start_time

        # Additional iterations if time permits

            popsize=20,

        )
        best_h = result.x
        best_c5 = result.fun
