    initial_guesses.append(h_sym)
    
    # Sinusoidal pattern
    x = np.linspace(0, 2, n_points)
    h_sine = 0.5 * (1 + np.sin(np.pi * x))
    sum_h = np.sum(h_sine)
    scale = target_sum / sum_h
    h_sine = h_sine * scale
    initial_guesses.append(h_sine)
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Use DE for sinusoidal and symmetric patterns
        if np.any(np.isclose(guess, h_sine)) or np.any(np.isclose(guess, h_sym)):
            # Differential Evolution
            def objective_de(h):
                one_minus_h = 1.0 - h
                corr = np.correlate(h, one_minus_h, mode='full')
                c5 = np.max(corr) * dx
                penalty = 1e6 * np.abs(np.sum(h) - target_sum)
                return c5 + penalty
            
            result = differential_evolution(
                objective_de,
                bounds,
                strategy='best1bin',
                popsize=15,
                mutation=(0.5, 1),
                recombination=0.8,
                tol=1e-6,
                maxiter=100,
                disp=False
            )
            if result.success:
                current_h = result.x
                current_c5 = objective(current_h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h
        else:
            # SLSQP
            result = minimize(
                objective,
                guess,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'ftol': 1e-9, 'maxiter': 300}
            )
            if result.success and result.fun < best_c5:
                best_c5 = result.fun
                best_h = result.x
    
    # Perturbation phase on the best_h found so far
    num_perturbations = 600  # Increased from 400
    phase_shift = 0.15
    phase_decay = 0.98
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 300}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
