def generate_interval_guess(n_points, intervals):
    h = np.zeros(n_points)
    dx = 2.0 / n_points
    for start, end in intervals:
        for i in range(n_points):
            if start <= i*dx <= end:
                h[i] = 1.0
    total = np.sum(h)
    if total == 0:
        return h
    scale = (n_points / 2) / total
    return h * scale

def compute_overlap_integral(h, n_points, dx):
    max_integral = 0.0
    for m in range(-n_points + 1, n_points):
        start_i = max(0, -m)
        end_i = min(n_points - 1, n_points - 1 - m)
        if start_i > end_i:
            continue
        integral = 0.0
        for i in range(start_i, end_i + 1):
            j = i + m
            integral += h[i] * (1 - h[j])
        integral *= dx
        if integral > max_integral:
            max_integral = integral
    return max_integral
