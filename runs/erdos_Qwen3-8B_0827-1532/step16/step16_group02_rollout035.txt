    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Define the objective function
    def objective(h):
        h_scaled = h.copy()
        sum_h = np.sum(h_scaled)
        if sum_h != 0:
            scale = target_sum / sum_h
            h_scaled *= scale
        h_scaled = np.clip(h_scaled, 0, 1)
        one_minus_h = 1.0 - h_scaled
        corr = np.correlate(h_scaled, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    # Bounds for each parameter
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial population: use initial_guess and some random guesses
    initial_population = [initial_guess.copy()]
    for _ in range(5):  # Add a few random guesses
        initial_population.append(np.random.uniform(0, 1, n_points))
    
    # Run differential evolution
    result = differential_evolution(
        objective,
        bounds,
        popsize=15,
        strategy='best1bin',
        mutation=(0.5, 1),
        recombination=0.9,
        tol=1e-9,
        maxiter=100,
        polish=True,
        seed=seed
    )
    
    # Update best solution
    best_h = result.x
    best_c5 = result.fun
    
    # Check elapsed time
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return best_h, best_c5, n_points
    
    # Perturbation phase
    num_perturbations = 300
    phase_shift = 0.15
    phase_decay = 0.98
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective_perturbed(h):
            h_scaled = h.copy()
            sum_h = np.sum(h_scaled)
            if sum_h != 0:
                scale = target_sum / sum_h
                h_scaled *= scale
            h_scaled = np.clip(h_scaled, 0, 1)
            one_minus_h = 1.0 - h_scaled
            corr = np.correlate(h_scaled, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Run local optimization
        result_perturbed = differential_evolution(
            objective_perturbed,
            bounds,
            popsize=10,
            strategy='best1bin',
            mutation=(0.5, 1),
            recombination=0.9,
            tol=1e-9,
            maxiter=50,
            polish=True,
            seed=seed
        )
        
        if result_perturbed.fun < best_c5:
            best_c5 = result_perturbed.fun
            best_h = result_perturbed.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
