            corr = np.correlate(h, one_minus_h, mode='full')
            c5 = np.max(corr) * dx
            # Penalty for constraint violation
            penalty = 1e6 * abs(np.sum(h) - target_sum)
            return c5 + penalty
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Run differential evolution
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=15,
            maxiter=100,
            tol=1e-9,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed + perturb_idx
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
