    intervals_3_spaced = [(0.0, 0.6667), (0.6667, 1.3333), (1.3333, 2.0)]
    initial_guesses.append(generate_interval_guess(n_points, intervals_3_spaced))
    intervals_4_spaced = [(0.0, 0.5), (0.5, 1.0), (1.0, 1.5), (1.5, 2.0)]
    initial_guesses.append(generate_interval_guess(n_points, intervals_4_spaced))
    intervals_1_centered = [(0.5, 1.5)]
    initial_guesses.append(generate_interval_guess(n_points, intervals_1_centered))
    
    # Add symmetric pattern
    symmetric = np.zeros(n_points)
    for i in range(n_points):
        symmetric[i] = 1.0 if i < n_points//2 else 0.0
    initial_guesses.append(symmetric)
    # Add periodic pattern
    periodic = np.zeros(n_points)
    for i in range(n_points):
        periodic[i] = 0.5 + 0.5 * np.sin(2 * np.pi * i / n_points)
    initial_guesses.append(periodic)
    # Add random guesses
    for _ in range(5):
        guess = np.random.uniform(0, 1, n_points)
        guess = guess * (target_sum / np.sum(guess))
        initial_guesses.append(guess)
    
    # Optimization parameters
    max_generations = 50  # Adjust based on time budget
    
    # Define objective function for DE
    def objective_de(h):
        h_sum = np.sum(h)
        if h_sum == 0:
            return float('inf')
        h_scaled = h * (target_sum / h_sum)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx
    
    # Run DE on the initial guesses
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result_de = differential_evolution(objective_de, bounds, popsize=10, maxiter=max_generations, tol=1e-6)
    best_h_de = result_de.x
    best_c5_de = objective_de(best_h_de)
    
    # Refine with SLSQP
    def objective_slsqp(h):
        h_sum = np.sum(h)
        if h_sum == 0:
            return float('inf')
        h_scaled = h * (target_sum / h_sum)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx
    
    bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    
    result_slsqp = minimize(
        objective_slsqp,
        best_h_de,
        method='SLSQP',
        bounds=bounds_slsqp,
        constraints=cons,
        options={'ftol': 1e-9, 'maxiter': 100}
    )
    
    best_h_slsqp = result_slsqp.x
    best_c5_slsqp = result_slsqp.fun
    
    # Compare and choose the best
    best_c5 = min(best_c5_de, best_c5_slsqp)
    best_h = best_h_slsqp if best_c5_slsqp < best_c5_de else best_h_de
    
    # Check initial guesses
    for guess in initial_guesses:
        h_scaled = guess * (target_sum / np.sum(guess)) if np.sum(guess) != 0 else guess
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = guess
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
