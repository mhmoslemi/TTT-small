from scipy.optimize import minimize
import time

    random = np.random.uniform(0, 1, n_points)

    for guess in initial_guesses:

        def objective(h):

        bounds = [(0.0, 1.0) for _ in range(n_points)]

        def objective(h):

            corr = np.correlate(h, one_minus_h, mode='full')
