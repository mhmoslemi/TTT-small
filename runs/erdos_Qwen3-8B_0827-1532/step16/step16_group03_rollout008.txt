        h = np.zeros(n_points)

                h[i] = 1.0
        initial_guesses.append(h)
    # Add structured guesses with two intervals
    h = np.zeros(n_points)
    for i in range(n_points):

    h = np.zeros(n_points)

            h[i] = 1.0

    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
