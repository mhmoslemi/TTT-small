import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate diverse initial guesses
    initial_guesses = []
    initial_guesses.append(initial_guess.copy())  # Original initial guess
    
    # Step function: first half 1, second half 0
    h_step = np.zeros(n_points)
    h_step[:n_points//2] = 1.0
    sum_h = np.sum(h_step)
    scale = target_sum / sum_h
    initial_guesses.append(h_step * scale)
    
    # Alternating 1 and 0
    h_alt = np.zeros(n_points)
    h_alt[::2] = 1.0
    sum_h = np.sum(h_alt)
    scale = target_sum / sum_h
    initial_guesses.append(h_alt * scale)
    
    # Random initial guess
    initial_guesses.append(np.random.uniform(0, 1, n_points))
    
    # Binary shuffled pattern
    h_binary = np.random.permutation([1]*50 + [0]*50)
    initial_guesses.append(h_binary)
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with L-BFGS-B
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 100}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Perturbation phase on the best_h found so far
    num_perturbations = 400  # Reduced from 600 for time efficiency
    phase_shift = 0.15
    phase_decay = 0.98
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx

        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(

            perturbed,

        )
