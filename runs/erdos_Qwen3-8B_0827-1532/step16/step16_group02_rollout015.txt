import numpy as np
from scipy.optimize import differential_evolution, minimize

def run(seed=42, budget_s=100, **kwargs):

    n_points = 100  # Using 100 points for efficiency
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Define the objective function with penalty for sum constraint
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr) * dx
        # Penalize if the sum is not correct
        penalty = 1e4 * abs(np.sum(h) - target_sum)
        return max_corr + penalty
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Run differential evolution for global search
    result_de = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=15,
        maxiter=100,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed
    )
    
    best_h = result_de.x
    best_c5 = result_de.fun
    
    # Refine with SLSQP for local optimization
    def objective_slsqp(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]

    )
