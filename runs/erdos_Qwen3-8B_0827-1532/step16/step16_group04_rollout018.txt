    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Scale the initial guess to meet the sum constraint
    initial_scaled = initial_guess * (target_sum / np.sum(initial_guess)) if np.sum(initial_guess) != 0 else initial_guess
    
    # Define the objective function for differential evolution
    def objective(h):
        h_scaled = h * (target_sum / np.sum(h)) if np.sum(h) != 0 else h
        one_minus_h = 1.0 - h_scaled
        corr = np.correlate(h_scaled, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    # Bounds for differential evolution
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Run differential evolution
    result_de = differential_evolution(objective, bounds, strategy='best1bin', 
                                      popsize=15, mutation=(0.5, 1), recombination=0.9, 
                                      tol=1e-6, maxiter=100, seed=seed)
    
    # Check if the result is successful
    if result_de.success:
        current_c5 = result_de.fun
        best_c5 = current_c5
        best_h = result_de.x
    
    # Refine the best solution with SLSQP
    if best_h is not None:
        def objective_slsqp(h):
            h_scaled = h * (target_sum / np.sum(h)) if np.sum(h) != 0 else h
            one_minus_h = 1.0 - h_scaled
            corr = np.correlate(h_scaled, one_minus_h, mode='full')
            return np.max(corr) * dx
        bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        result_slsqp = minimize(
            objective_slsqp,
            best_h,
            method='SLSQP',
            bounds=bounds_slsqp,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 100}
        )
        if result_slsqp.success:
            best_c5 = result_slsqp.fun
            best_h = result_slsqp.x
    
    # Check elapsed time
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
