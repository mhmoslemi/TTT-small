import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    def constraint(h):
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint}]
    
    def evaluate_guess(guess):
        nonlocal best_c5, best_h
        result = differential_evolution(
            objective,
            bounds,
            constraints=constraints,
            popsize=15,
            maxiter=100,
            tol=1e-8,
            mutation=(0.5, 1),
            recombination=0.8,
            x0=guess,
            seed=seed
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    evaluate_guess(initial_guess)
    
    # Generate and evaluate diverse initial guesses
    guesses = [initial_guess]
    
    # Symmetric pattern
    symmetric = np.zeros(n_points)
    for i in range(n_points):
        symmetric[i] = 1.0 if i < n_points//2 else 0.0
    total = np.sum(symmetric)
    scale = target_sum / total
    symmetric = symmetric * scale
    guesses.append(symmetric)
    
    # Interval-based patterns
    def generate_intervals(intervals):
        h = np.zeros(n_points)
        dx = 2.0 / n_points
        for start, end in intervals:
            for i in range(n_points):
                if start <= i*dx <= end:
                    h[i] = 1.0
        total = np.sum(h)
        if total == 0:
            return h
        scale = target_sum / total
        return h * scale
    
    guesses.append(generate_intervals([(0.0, 0.5), (1.5, 2.0)]))
    guesses.append(generate_intervals([(0.0, 0.6667), (0.6667, 1.3333), (1.3333, 2.0)]))
    guesses.append(generate_intervals([(0.0, 0.5), (0.5, 1.0), (1.0, 1.5), (1.5, 2.0)]))
    guesses.append(generate_intervals([(0.5, 1.5)]))
    
    # Periodic pattern
    periodic = np.zeros(n_points)
    for i in range(n_points):
        periodic[i] = 0.5 + 0.5 * np.sin(2 * np.pi * i / n_points)
    total = np.sum(periodic)
    scale = target_sum / total
    periodic = periodic * scale
    guesses.append(periodic)
    
    # Random guesses
    for _ in range(3):
        guess = np.random.uniform(0, 1, n_points)
        guess = guess * (target_sum / np.sum(guess))

    for guess in guesses:

        def local_minimize(h):
