    scale = (n_points / 2) / total
    return h * scale

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate diverse initial guesses
    initial_guesses = [initial_guess.copy()]

    symmetric = np.zeros(n_points)
    for i in range(n_points):
        symmetric[i] = 1.0 if i < n_points//2 else 0.0
    initial_guesses.append(symmetric)
    # Add periodic pattern
    periodic = np.zeros(n_points)
    for i in range(n_points):
        periodic[i] = 0.5 + 0.5 * np.sin(2 * np.pi * i / n_points)
    initial_guesses.append(periodic)
    # Add more structured patterns
    intervals_3_spaced_2 = [(0.0, 0.6), (0.8, 1.4), (1.6, 2.0)]
    initial_guesses.append(generate_interval_guess(n_points, intervals_3_spaced_2))
    intervals_3_spaced_3 = [(0.0, 0.5), (1.0, 1.5), (2.0, 2.5)]
    initial_guesses.append(generate_interval_guess(n_points, intervals_3_spaced_3))
    intervals_3_spaced_4 = [(0.0, 0.4), (0.6, 1.2), (1.4, 2.0)]
    initial_guesses.append(generate_interval_guess(n_points, intervals_3_spaced_4))
    intervals_3_spaced_5 = [(0.0, 0.3), (0.7, 1.3), (1.7, 2.0)]
    initial_guesses.append(generate_interval_guess(n_points, intervals_3_spaced_5))
    intervals_3_spaced_6 = [(0.0, 0.7), (0.9, 1.6), (1.8, 2.0)]
    initial_guesses.append(generate_interval_guess(n_points, intervals_3_spaced_6))
    intervals_3_spaced_7 = [(0.0, 0.55), (0.85, 1.45), (1.65, 2.0)]
    initial_guesses.append(generate_interval_guess(n_points, intervals_3_spaced_7))
    intervals_3_spaced_8 = [(0.0, 0.45), (0.75, 1.35), (1.55, 2.0)]
    initial_guesses.append(generate_interval_guess(n_points, intervals_3_spaced_8))
    intervals_3_spaced_9 = [(0.0, 0.5), (0.7, 1.3), (1.5, 2.0)]
    initial_guesses.append(generate_interval_guess(n_points, intervals_3_spaced_9))
    intervals_3_spaced_10 = [(0.0, 0.65), (0.8, 1.4), (1.6, 2.0)]
    initial_guesses.append(generate_interval_guess(n_points, intervals_3_spaced_10))
    
    # Add three-interval symmetric pattern
    three_intervals = np.zeros(n_points)
    for i in range(n_points):
        if 0 <= i*dx <= 1/3 or 2/3 <= i*dx <= 2:
            three_intervals[i] = 1.0
    total = np.sum(three_intervals)
    scale = (n_points / 2) / total
    initial_guesses.append(three_intervals * scale)
    
    # Add random guesses
    for _ in range(3):
        guess = np.random.uniform(0, 1, n_points)
        guess = guess * (target_sum / np.sum(guess))
        initial_guesses.append(guess)
    
    # Optimization parameters
    max_iterations = 300
    perturbation_count = 800
    noise_scales = [0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
    
    for perturb_idx in range(perturbation_count):
        # Select a guess
        if perturb_idx < len(initial_guesses):
            perturbed = initial_guesses[perturb_idx].copy()
        else:
            # Generate a new perturbation based on best_h
            if perturb_idx % 6 == 0:
                # Random noise with larger scale
                noise = np.random.uniform(-noise_scales[0], noise_scales[0], n_points)
            elif perturb_idx % 6 == 1:
                # Random noise with medium scale
                noise = np.random.uniform(-noise_scales[1], noise_scales[1], n_points)
            elif perturb_idx % 6 == 2:
                # Random noise with small scale
                noise = np.random.uniform(-noise_scales[2], noise_scales[2], n_points)
            elif perturb_idx % 6 == 3:
                # Swap segments
                noise = np.zeros(n_points)
                for _ in range(2):
                    start = np.random.randint(0, n_points-5)
                    end = start + 5
                    noise[start:end] = np.random.uniform(-noise_scales[3], noise_scales[3], end - start)
            elif perturb_idx % 6 == 4:
                # Flip values
                noise = np.random.choice([0, 1], size=n_points)
                noise = noise * 0.5 - 0.25
            else:
                # Random noise with tiny scale
                noise = np.random.uniform(-noise_scales[5], noise_scales[5], n_points)
            perturbed = best_h + noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
        
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-12, 'maxiter': max_iterations}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
