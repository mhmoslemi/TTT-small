        h = h * target_sum / h_sum
        population.append(h)
    # Add initial_h_values
    population.append(initial_h_values)
    # Remove duplicates if any
    population = [np.array(p) for p in population]
    population = list(set(tuple(p) for p in population))[:pop_size]
    population = [np.array(p) for p in population]

    best_h = population[0]
    best_c5 = evaluate_c5(best_h, dx)

    start_time = time.time()

    # Parameters for DE
    generations = 50
    mutation_factor = 0.8
    crossover_rate = 0.9

    for gen in range(generations):
        new_population = []
        for i in range(pop_size):
            # Select three distinct indices
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            # Generate mutant vector
            mutant = population[a] + mutation_factor * (population[b] - population[c])
            # Crossover with target individual i
            trial = population[i].copy()
            for j in range(n_points):
                if np.random.rand() < crossover_rate:
                    trial[j] = mutant[j]
            # Clip to [0, 1]
            trial = np.clip(trial, 0, 1)
            # Normalize to meet sum constraint
            trial_sum = np.sum(trial)
            if trial_sum == 0:
                trial = np.zeros(n_points)
            else:
                scaling_factor = target_sum / trial_sum
                trial = trial * scaling_factor
            # Evaluate
            current_c5 = evaluate_c5(trial, dx)
            if current_c5 < evaluate_c5(population[i], dx):
                new_population.append(trial)
            else:
                new_population.append(population[i])
        population = new_population
        # Update best
        for h in population:
            current_c5 = evaluate_c5(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h
        # Check time
        if time.time() - start_time > budget_s:
            break

    return best_h, best_c5, n_points
