    intervals_2_sep = [(0.0, 0.5), (1.5, 2.0)]
    initial_guesses.append(generate_interval_guess(n_points, intervals_2_sep))
    intervals_3_spaced = [(0.0, 0.6667), (0.6667, 1.3333), (1.3333, 2.0)]

    intervals_4_spaced = [(0.0, 0.5), (0.5, 1.0), (1.0, 1.5), (1.5, 2.0)]

    # Add symmetric pattern
    symmetric = np.zeros(n_points)
    for i in range(n_points):
        symmetric[i] = 1.0 if i < n_points//2 else 0.0
    initial_guesses.append(symmetric)
    # Add periodic pattern
    periodic = np.zeros(n_points)
    for i in range(n_points):
        periodic[i] = 0.5 + 0.5 * np.sin(2 * np.pi * i / n_points)
    initial_guesses.append(periodic)
    # Add random guesses
    for _ in range(5):
        guess = np.random.uniform(0, 1, n_points)
        guess = guess * (target_sum / np.sum(guess))
        initial_guesses.append(guess)
    
    # Prepare initial population for differential evolution
    initial_population = []
    for guess in initial_guesses:
        initial_population.append(guess)
    # Add some random guesses to fill the population
    for _ in range(5):
        guess = np.random.uniform(0, 1, n_points)
        guess = guess * (target_sum / np.sum(guess))
        initial_population.append(guess)
    
    # Convert initial_population to a 2D array
    initial_population = np.array(initial_population)
    
    # Define objective function
    def objective(h):
        h_scaled = h * (target_sum / np.sum(h))
        one_minus_h = 1.0 - h_scaled
        corr = np.correlate(h_scaled, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    start_time = time.time()
    
    # Run differential evolution
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result_de = differential_evolution(
        objective,

    best_h = result_de.x
