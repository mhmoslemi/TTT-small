    intervals_4_spaced = [(0.0, 0.5), (0.5, 1.0), (1.0, 1.5), (1.5, 2.0)]

    intervals_1_centered = [(0.5, 1.5)]
    initial_guesses.append(generate_interval_guess(n_points, intervals_1_centered))
    
    # Add symmetric pattern
    symmetric = np.zeros(n_points)
    for i in range(n_points):
        symmetric[i] = 1.0 if i < n_points//2 else 0.0
    initial_guesses.append(symmetric)
    # Add periodic pattern
    periodic = np.zeros(n_points)
    for i in range(n_points):
        periodic[i] = 0.5 + 0.5 * np.sin(2 * np.pi * i / n_points)
    initial_guesses.append(periodic)
    # Add random guesses
    for _ in range(5):
        guess = np.random.uniform(0, 1, n_points)
        guess = guess * (target_sum / np.sum(guess))
        initial_guesses.append(guess)
    
    # Optimization parameters
    max_generations = 100
    population_size = 20
    
    for guess in initial_guesses:
        # Scale the guess to meet the sum constraint
        current_sum = np.sum(guess)
        if current_sum == 0:
            scaled_guess = guess
        else:
            scaled_guess = guess * (target_sum / current_sum)
        
        # Define the objective function with penalty
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr) * dx
            # Penalty for sum constraint
            penalty = 1e4 * (np.sum(h) - target_sum)**2
            return max_corr + penalty
        
        # Bounds for each parameter
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Run differential evolution
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=population_size,
            maxiter=max_generations,
            tol=1e-6,

        # Check if the result is successful

            if current_c5 < best_c5:

                best_h = result.x
        
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
