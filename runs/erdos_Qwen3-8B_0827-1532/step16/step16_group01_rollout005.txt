from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate diverse initial guesses
    initial_guesses = []

    # Uniform distribution
    uniform = np.full(n_points, 0.5)
    uniform = uniform * (target_sum / np.sum(uniform))
    initial_guesses.append(uniform)
    # Blocky pattern (alternating 1 and 0)
    blocky = np.zeros(n_points)
    blocky[::2] = 1.0
    blocky = blocky * (target_sum / np.sum(blocky))
    initial_guesses.append(blocky)
    # Random pattern
    random = np.random.uniform(0, 1, n_points)
    random = random * (target_sum / np.sum(random))

    # Sinusoidal pattern
    x = np.linspace(0, 2, n_points)
    sinusoidal = 0.5 + 0.5 * np.sin(np.pi * x)

    initial_guesses.append(sinusoidal)

    triangular = np.zeros(n_points)

    # Optimize each initial guess
    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with COBYLA
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(

            guess,
            method='COBYLA',

            constraints=cons,

        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x

    num_perturbations = 400  # Increased from 300
    phase_shift = 0.15

        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
