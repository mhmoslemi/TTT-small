    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    
    result_slsqp = minimize(
        objective_slsqp,
        best_h_de,
        method='SLSQP',
        bounds=bounds_slsqp,
        constraints=cons,
        options={'ftol': 1e-9, 'maxiter': 200}
    )
    
    best_h = result_slsqp.x
    best_c5 = result_slsqp.fun
    
    # Check if DE's solution is better
    if best_c5_de < best_c5:
        best_h = best_h_de
        best_c5 = best_c5_de
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
