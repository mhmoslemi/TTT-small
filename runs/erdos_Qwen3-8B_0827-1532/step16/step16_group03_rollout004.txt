        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result_local = minimize(
            objective_local,
            best_h,
            method='SLSQP',
            bounds=bounds_local,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 300}
        )
        
        if result_local.success and result_local.fun < best_c5:
            best_c5 = result_local.fun
            best_h = result_local.x
    
    return best_h, best_c5, n_points
