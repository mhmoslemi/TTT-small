import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        h_full = np.concatenate([h, [target_sum - np.sum(h)]])
        h_full[-1] = np.clip(h_full[-1], 0, 1)
        one_minus_h = 1.0 - h_full
        corr = np.correlate(h_full, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points - 1)]
    initial_guess_de = initial_guess[:-1]
    
    start_time = time.time()
    result = differential_evolution(
        objective,
        bounds,
        popsize=15,
        maxiter=100,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed
    )
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        best_h = initial_guess
        best_c5 = np.max(np.correlate(initial_guess, 1 - initial_guess, mode='full')) * dx
    else:
        best_h = np.concatenate([result.x, [target_sum - np.sum(result.x)]])
        best_h[-1] = np.clip(best_h[-1], 0, 1)
        best_c5 = result.fun
    
    return best_h, best_c5, n_points
