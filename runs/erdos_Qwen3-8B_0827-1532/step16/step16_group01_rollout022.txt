from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate diverse initial guesses
    initial_guesses = []
    initial_guesses.append(initial_guess.copy())  # Original initial guess
    # Uniform distribution
    uniform = np.full(n_points, 0.5)
    initial_guesses.append(uniform)
    # Blocky pattern (alternating 1 and 0)
    blocky = np.zeros(n_points)
    blocky[::2] = 1.0
    initial_guesses.append(blocky)
    # Random pattern
    random = np.random.uniform(0, 1, n_points)
    initial_guesses.append(random)
    # Sinusoidal pattern
    x = np.linspace(0, 2, n_points)
    sinusoidal = 0.5 + 0.5 * np.sin(np.pi * x)
    initial_guesses.append(sinusoidal)
    # Triangular pattern
    triangular = np.abs(x - 1) / 1.0
    triangular = triangular / np.sum(triangular) * target_sum
    initial_guesses.append(triangular)
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 500}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Perturbation phase on the best_h found so far
    num_perturbations = 500  # Increased from 300
    phase_shift = 0.25

    for perturbation_idx in range(num_perturbations):

        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx

        bounds = [(0.0, 1.0) for _ in range(n_points)]
