    initial_population.append(generate_interval_guess(n_points, intervals_4_spaced))
    intervals_1_centered = [(0.5, 1.5)]
    initial_population.append(generate_interval_guess(n_points, intervals_1_centered))
    
    # Add symmetric and periodic patterns
    symmetric = np.zeros(n_points)
    for i in range(n_points):
        symmetric[i] = 1.0 if i < n_points//2 else 0.0
    initial_population.append(symmetric)
    periodic = np.zeros(n_points)
    for i in range(n_points):
        periodic[i] = 0.5 + 0.5 * np.sin(2 * np.pi * i / n_points)
    initial_population.append(periodic)
    
    # Add random guesses
    for _ in range(5):
        guess = np.random.uniform(0, 1, n_points)
        guess = guess * (target_sum / np.sum(guess))
        initial_population.append(guess)
    
    # Define objective function with penalty for sum constraint
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        c5 = np.max(corr) * dx
        sum_h = np.sum(h)
        penalty = 1e4 * np.abs(sum_h - target_sum)
        return c5 + penalty
    
    # Run differential evolution
    start_time = time.time()
    result = differential_evolution(
        objective,
        bounds=[(0.0, 1.0) for _ in range(n_points)],
        popsize=min(20, len(initial_population)),
        strategy='best1bin',
        mutation=(0.5, 1),
        recombination=0.9,
        tol=1e-6,
        maxiter=100,
        seed=seed
    )
    
    # Extract best solution
    best_h = result.x
    best_c5 = objective(best_h) - 1e4 * np.abs(np.sum(best_h) - target_sum)
    
    # Ensure the solution meets the sum constraint
    sum_h = np.sum(best_h)
    if np.abs(sum_h - target_sum) > 1e-6:
        # Scale to meet the constraint
        scale = target_sum / sum_h
        best_h = best_h * scale
    
    elapsed = time.time() - start_time
    return best_h, best_c5, n_points
