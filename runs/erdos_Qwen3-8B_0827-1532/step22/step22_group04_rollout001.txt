    sum_h = np.sum(h_25)
    scale = target_sum / sum_h
    initial_guesses.append(h_25 * scale)
    
    # Random initial guess
    initial_guesses.append(np.random.uniform(0, 1, n_points))
    
    # Generate more diverse initial guesses with different patterns
    for _ in range(3):
        h_rand = np.random.uniform(0, 1, n_points)
        sum_h = np.sum(h_rand)
        scale = target_sum / sum_h
        initial_guesses.append(h_rand * scale)
    
    # Three-segment pattern
    h_three = np.zeros(n_points)
    segment_length = n_points // 3
    h_three[:segment_length] = 1.0
    h_three[segment_length:2*segment_length] = 0.5
    h_three[2*segment_length:] = 1.0
    sum_h = np.sum(h_three)
    scale = target_sum / sum_h
    initial_guesses.append(h_three * scale)
    
    # Three intervals spaced apart
    h_three_intervals = np.zeros(n_points)
    interval_length = 0.3333
    start = 0.0
    for i in range(3):
        end = start + interval_length
        start_idx = int(start / dx)
        end_idx = int(end / dx)
        h_three_intervals[start_idx:end_idx] = 1.0
        start = end
    sum_h = np.sum(h_three_intervals)
    scale = target_sum / sum_h
    initial_guesses.append(h_three_intervals * scale)
    
    # Random segments with varying heights
    h_rand_seg = np.zeros(n_points)
    for i in range(n_points):
        if np.random.rand() < 0.3:
            h_rand_seg[i] = 1.0
        elif np.random.rand() < 0.5:
            h_rand_seg[i] = 0.5
    sum_h = np.sum(h_rand_seg)
    scale = target_sum / sum_h
    initial_guesses.append(h_rand_seg * scale)
    
    # Define objective function with penalty
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr) * dx
        penalty = 1e6 * (np.sum(h) - target_sum)**2
        return max_corr + penalty
    
    # Bounds for differential evolution
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimize each initial guess with differential evolution
    for guess in initial_guesses:
        # Run differential evolution
        result = differential_evolution(
            objective,
            bounds,
            popsize=15,
            maxiter=100,
            tol=1e-10,
            mutation=(0.5, 1),
            recombination=0.9,
            strategy='best1bin',
            polish=True,
            disp=False
        )
        
        # Check if the result is successful
        if result.success:
            current_h = result.x
            # Compute C5 without penalty
            one_minus_h = 1.0 - current_h
            corr = np.correlate(current_h, one_minus_h, mode='full')
            current_c5 = np.max(corr) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
    
    # Perturbation phase on the best_h found so far
    num_perturbations = 250
    phase_shift = 0.15
    phase_decay = 0.96
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Run differential evolution on the perturbed guess
        result = differential_evolution(
            objective,
            bounds,
            popsize=15,
            maxiter=50,
            tol=1e-10,
            mutation=(0.5, 1),
            recombination=0.9,
            strategy='best1bin',
            polish=True,
            disp=False
        )
        
        if result.success:
            current_h = result.x
            # Compute C5 without penalty
            one_minus_h = 1.0 - current_h
            corr = np.correlate(current_h, one_minus_h, mode='full')
            current_c5 = np.max(corr) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
