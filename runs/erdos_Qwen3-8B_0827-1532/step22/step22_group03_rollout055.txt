            best_c5 = result.fun
            best_h = result.x
    
    # Perturbation phase with structured perturbations
    num_perturbations = 100  # Increased to explore more
    phase_shift = 0.15
    phase_decay = 0.98
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        # Apply structured perturbations: shift and scale
        perturbed = best_h.copy()
        shift = np.random.uniform(-0.2, 0.2, n_points)
        perturbed += noise_scale * shift
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            penalty = 1e6 * abs(np.sum(h) - target_sum)
            return np.max(corr) * dx + penalty
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Use differential evolution with adjusted parameters
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=15,
            mutation=(0.5, 1),
            recombination=0.9,
            tol=1e-6,
            maxiter=50,
            polish=True,
            seed=seed
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
