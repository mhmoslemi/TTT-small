    # Four-segment step function
    h_four_seg = np.zeros(n_points)
    seg1_start = n_points // 4
    seg1_end = n_points // 2
    seg3_start = 3 * n_points // 4
    seg3_end = n_points
    h_four_seg[seg1_start:seg1_end] = 1.0
    h_four_seg[seg3_start:seg3_end] = 1.0
    sum_h = np.sum(h_four_seg)
    scale = target_sum / sum_h
    initial_guesses.append(h_four_seg * scale)
    
    # Triangular wave
    h_tri = np.zeros(n_points)
    for i in range(n_points):
        h_tri[i] = 1.0 - abs((i - n_points/2) / (n_points/2) - 1.0)
    sum_h = np.sum(h_tri)
    scale = target_sum / sum_h
    initial_guesses.append(h_tri * scale)
    
    # Sine-like function
    h_sine = np.zeros(n_points)
    for i in range(n_points):
        h_sine[i] = 0.5 * (1 + np.sin(np.pi * (i - n_points/2) / (n_points/2)))
    sum_h = np.sum(h_sine)
    scale = target_sum / sum_h
    initial_guesses.append(h_sine * scale)
    
    # Gaussian-like function
    h_gauss = np.zeros(n_points)
    for i in range(n_points):
        h_gauss[i] = np.exp(-((i - n_points/2) / (n_points/4))**2)
    sum_h = np.sum(h_gauss)
    scale = target_sum / sum_h
    initial_guesses.append(h_gauss * scale)
    
    # Random initial guess
    initial_guesses.append(np.random.uniform(0, 1, n_points))
    
    # Optimization using differential evolution
    for guess in initial_guesses:
        # Define objective function with penalty for sum constraint
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            c5 = np.max(corr) * dx
            penalty = 1e4 * abs(np.sum(h) - target_sum)
            return c5 + penalty
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Run differential evolution
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=50,
            popsize=10,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed
        )
        
        if result.success:
            current_c5 = result.fun - 1e4 * abs(np.sum(result.x) - target_sum)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # Perturbation phase with structured perturbations
    num_perturbations = 200
    phase_shift = 0.15
    phase_decay = 0.98
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        # Apply structured perturbations: shift and scale
        perturbed = best_h.copy()
        shift = np.random.uniform(-0.2, 0.2, n_points)
        perturbed += noise_scale * shift
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Define objective function with penalty for sum constraint
        def objective_perturbed(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            c5 = np.max(corr) * dx
            penalty = 1e4 * abs(np.sum(h) - target_sum)
            return c5 + penalty
        
        bounds_perturbed = [(0.0, 1.0) for _ in range(n_points)]
        
        # Run differential evolution on perturbed solution
        result_perturbed = differential_evolution(
            objective_perturbed,
            bounds_perturbed,
            strategy='best1bin',
            maxiter=50,
            popsize=10,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed
        )
        
        if result_perturbed.success:
            current_c5_perturbed = result_perturbed.fun - 1e4 * abs(np.sum(result_perturbed.x) - target_sum)
            if current_c5_perturbed < best_c5:
                best_c5 = current_c5_perturbed
                best_h = result_perturbed.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
