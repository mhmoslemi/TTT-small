    # Central 50% step function
    h_center = np.zeros(n_points)
    center_start = n_points // 2 - n_points // 4
    center_end = n_points // 2 + n_points // 4
    h_center[center_start:center_end] = 1.0
    sum_h = np.sum(h_center)
    scale = target_sum / sum_h
    initial_guesses.append(h_center * scale)
    
    # 25% step function
    h_25 = np.zeros(n_points)
    h_25[:n_points//4] = 1.0
    h_25[n_points//2:n_points//2 + n_points//4] = 1.0
    sum_h = np.sum(h_25)
    scale = target_sum / sum_h
    initial_guesses.append(h_25 * scale)
    
    # 4-segment step function
    h_4seg = np.zeros(n_points)
    h_4seg[:n_points//4] = 1.0
    h_4seg[n_points//2:n_points//2 + n_points//4] = 1.0
    h_4seg[3*n_points//4:] = 1.0
    sum_h = np.sum(h_4seg)
    scale = target_sum / sum_h
    initial_guesses.append(h_4seg * scale)
    
    # 5-segment step function
    h_5seg = np.zeros(n_points)
    h_5seg[:n_points//5] = 1.0
    h_5seg[2*n_points//5:3*n_points//5] = 1.0
    h_5seg[4*n_points//5:] = 1.0
    sum_h = np.sum(h_5seg)
    scale = target_sum / sum_h
    initial_guesses.append(h_5seg * scale)
    
    # Random initial guess
    initial_guesses.append(np.random.uniform(0, 1, n_points))
    
    # Generate more diverse initial guesses with different patterns
    for _ in range(3):
        h_rand = np.random.uniform(0, 1, n_points)
        sum_h = np.sum(h_rand)
        scale = target_sum / sum_h
        initial_guesses.append(h_rand * scale)
    
    # Generate segmented initial guesses
    def generate_segmented_h(n_points, m):
        dx = 2.0 / n_points
        h = np.zeros(n_points)
        total_length = 1.0
        segment_length = total_length / m
        start = 0.0
        for i in range(m):
            end = start + segment_length
            start_idx = int(start / dx)
            end_idx = int(end / dx)
            end_idx = min(end_idx, n_points - 1)
            h[start_idx:end_idx+1] = 1.0
            start = end
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h
        return h * scale
    
    for m in [3, 4, 5, 6]:
        h_seg = generate_segmented_h(n_points, m)
        initial_guesses.append(h_seg)
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 1000}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Perturbation phase on the best_h found so far
    num_perturbations = 500
    phase_shift = 0.25
    phase_decay = 0.98
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 300}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points

def compute_c5(h, dx):
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return np.max(corr) * dx
