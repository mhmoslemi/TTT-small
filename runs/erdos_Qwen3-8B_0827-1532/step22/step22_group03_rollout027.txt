        h_scaled = h * (target_sum / np.sum(h)) if np.sum(h) != 0 else h
        one_minus_h = 1.0 - h_scaled
        corr = np.correlate(h_scaled, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    # Bounds for each variable
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Run differential evolution
    result = differential_evolution(
        objective,
        bounds,
        popsize=len(initial_population),
        strategy='best1bin',
        maxiter=100,
        tol=1e-10,
        mutation=(0.5, 1),
        recombination=0.9,
        seed=seed,
        polish=True,
        disp=False
    )
    
    # Update best solution
    best_h = result.x
    best_c5 = result.fun
    
    # Check elapsed time and return
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
