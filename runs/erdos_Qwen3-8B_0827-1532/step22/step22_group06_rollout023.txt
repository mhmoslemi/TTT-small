        one_minus_h = 1.0 - h_scaled
        corr = np.correlate(h_scaled, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    # Bounds for differential evolution
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Run differential evolution
    result_de = differential_evolution(objective, bounds, strategy='best1bin', maxiter=50, popsize=15, tol=1e-6, mutation=(0.5, 1), recombination=0.8, seed=seed)
    best_h_de = result_de.x
    best_c5_de = result_de.fun
    
    # Now refine with SLSQP
    def objective_slsqp(h):
        h_scaled = h * (target_sum / np.sum(h))
        one_minus_h = 1.0 - h_scaled
        corr = np.correlate(h_scaled, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    
    result_slsqp = minimize(
        objective_slsqp,
        best_h_de,
        method='SLSQP',
        bounds=bounds_slsqp,
        constraints=cons,
        options={'ftol': 1e-9, 'maxiter': 200}
    )
    
    best_h = result_slsqp.x
    best_c5 = result_slsqp.fun
    
    # Perturbation phase
    num_perturbations = 50
    phase_shift = 0.15
    phase_decay = 0.96
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective_perturb(h):
            h_scaled = h * (target_sum / np.sum(h))
            one_minus_h = 1.0 - h_scaled
            corr = np.correlate(h_scaled, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        result = minimize(
            objective_perturb,
            perturbed,
            method='SLSQP',
            bounds=bounds_slsqp,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 300}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
