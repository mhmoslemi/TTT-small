    initial_guess = initial_guess * scale
    best_h = initial_guess.copy()
    best_c5 = compute_c5(best_h, dx)
    
    # Generate diverse initial guesses
    initial_guesses = [initial_guess]

    for m in [2, 3, 4, 5]:
        interval_length = 1.0 / m
        gap = 1.0 / (m - 1)
        h = np.zeros(n_points)
        current_pos = 0.0
        for i in range(m):
            start = current_pos
            end = start + interval_length
            # Find the indices in h corresponding to [start, end]
            start_idx = int(start / dx)
            end_idx = int(end / dx)
            h[start_idx:end_idx] = 1.0
            current_pos = end + gap
        # Scale to meet the target_sum
        sum_h = np.sum(h)
        scale = target_sum / sum_h
        initial_guesses.append(h * scale)

    h_step = np.zeros(n_points)

    sum_h = np.sum(h_step)
    scale = target_sum / sum_h

    h_alt = np.zeros(n_points)

    initial_guesses.append(h_alt * scale)

    h_center = np.zeros(n_points)

    center_end = n_points // 2 + n_points // 4
    h_center[center_start:center_end] = 1.0
    sum_h = np.sum(h_center)
    scale = target_sum / sum_h
    initial_guesses.append(h_center * scale)
    
    # 25% step function
    h_25 = np.zeros(n_points)
    h_25[:n_points//4] = 1.0
    h_25[n_points//2:n_points//2 + n_points//4] = 1.0

    initial_guesses.append(h_25 * scale)

    h_sym_bin = np.random.choice([0, 1], size=n_points)
    h_sym_bin = (h_sym_bin + h_sym_bin[::-1]) / 2
    sum_h = np.sum(h_sym_bin)

    initial_guesses.append(h_sym_bin * scale)

    initial_guesses.append(np.random.uniform(0, 1, n_points))

    h_three_intervals = np.zeros(n_points)
    interval_length = 1.0 / 3.0
    for i in range(3):
        start = i * interval_length * n_points
        end = (i + 1) * interval_length * n_points
        h_three_intervals[int(start):int(end)] = 1.0
    sum_h = np.sum(h_three_intervals)
    scale = target_sum / sum_h
    initial_guesses.append(h_three_intervals * scale)

    h_four_intervals = np.zeros(n_points)
    interval_length = 0.25
    for i in range(4):
        start = i * interval_length * n_points / 2.0

    sum_h = np.sum(h_four_intervals)
    scale = target_sum / sum_h
    initial_guesses.append(h_four_intervals * scale)

    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 2000}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Perturbation phase on the best_h found so far
    num_perturbations = 500
    phase_shift = 0.25
    phase_decay = 0.98
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h

            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:

def compute_c5(h, dx):
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return np.max(corr) * dx
