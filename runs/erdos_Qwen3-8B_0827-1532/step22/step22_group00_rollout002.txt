    h_three[:n_points//3] = 1.0
    h_three[n_points//2:n_points//2 + n_points//3] = 1.0
    h_three[2*n_points//3:] = 1.0
    sum_h = np.sum(h_three)
    scale = target_sum / sum_h
    initial_guesses.append(h_three * scale)
    
    # Triangular function
    h_tri = np.zeros(n_points)
    for i in range(n_points):
        x = i * dx
        if x < 1.0:
            h_tri[i] = x / 1.0
        elif x < 2.0:
            h_tri[i] = (2.0 - x) / 1.0
        else:
            h_tri[i] = 0.0
    sum_h = np.sum(h_tri)
    scale = target_sum / sum_h
    initial_guesses.append(h_tri * scale)
    
    # Random initial guess
    initial_guesses.append(np.random.uniform(0, 1, n_points))
    
    # Generate more diverse initial guesses with different patterns
    for _ in range(3):
        h_rand = np.random.uniform(0, 1, n_points)
        sum_h = np.sum(h_rand)
        scale = target_sum / sum_h
        initial_guesses.append(h_rand * scale)
    
    # Optimize each initial guess with differential evolution
    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Bounds
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Use differential evolution
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=100,
            popsize=20,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.9,
            disp=False
        )
        
        # Normalize the solution to meet the sum constraint
        if result.success:
            h_normalized = result.x * (target_sum / np.sum(result.x))
            current_c5 = compute_c5(h_normalized, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_normalized
    
    # Perturbation phase on the best_h found so far
    num_perturbations = 500
    phase_shift = 0.25
    phase_decay = 0.98
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Use differential evolution for perturbation
        result = differential_evolution(
            objective,
            [(0.0, 1.0) for _ in range(n_points)],
            strategy='best1bin',
            maxiter=50,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.9,
            disp=False
        )
        
        if result.success:
            h_normalized = result.x * (target_sum / np.sum(result.x))
            current_c5 = compute_c5(h_normalized, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_normalized
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points

def compute_c5(h, dx):
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return np.max(corr) * dx
