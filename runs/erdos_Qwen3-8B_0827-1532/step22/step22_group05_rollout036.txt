    h_step = np.zeros(n_points)
    h_step[:n_points//2] = 1.0
    initial_guesses.append(h_step)
    # Alternating 1 and 0
    h_alt = np.zeros(n_points)
    h_alt[::2] = 1.0
    initial_guesses.append(h_alt)
    # Central 50% step
    h_center = np.zeros(n_points)
    center_start = n_points // 2 - n_points // 4
    center_end = n_points // 2 + n_points // 4
    h_center[center_start:center_end] = 1.0
    initial_guesses.append(h_center)
    # 25% step
    h_25 = np.zeros(n_points)
    h_25[:n_points//4] = 1.0
    h_25[n_points//2:n_points//2 + n_points//4] = 1.0
    initial_guesses.append(h_25)
    # Random initial guess
    initial_guesses.append(np.random.uniform(0, 1, n_points))
    
    # Optimize each initial guess with differential evolution
    for guess in initial_guesses:
        # Scale to meet the target sum
        guess = np.clip(guess, 0, 1)
        sum_guess = np.sum(guess)
        scale = target_sum / sum_guess if sum_guess != 0 else 1.0
        guess = guess * scale
        # Bounds for differential evolution
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        # Run differential evolution
        result = differential_evolution(
            objective,
            bounds,
            popsize=15,
            maxiter=100,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.9,
            strategy='best1bin',
            polish=True,
            seed=seed
        )
        # Check if the result is better
        if result.success:
            current_c5 = result.fun - 1e4 * abs(np.sum(result.x) - target_sum)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # Perturbation phase on the best_h found so far
    num_perturbations = 250
    phase_shift = 0.15
    phase_decay = 0.96
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum if current_sum != 0 else 1.0
        perturbed = perturbed * scale
        
        # Objective function without penalty for refinement
        def objective_refine(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Run SLSQP on the perturbed solution
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective_refine,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 300}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
