        h = h / np.sum(h) * target_sum
        initial_guesses.append(np.clip(h, 0, 1))
    
    start_time = time.time()
    
    for guess in initial_guesses:
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 300}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    if best_h is None:
        best_h = initial_guesses[0]
        best_c5 = objective(best_h)
    
    elapsed = time.time() - start_time
    return best_h, best_c5, n_points
