    binary_guess_3 = binary_guess_3 * (target_sum / np.sum(binary_guess_3))
    initial_guesses.append(binary_guess_3)
    
    # Binary pattern with four intervals
    binary_guess_4 = np.zeros(n_points)
    for i in range(n_points):
        if (0 <= i*dx <= 0.25 or 0.5 <= i*dx <= 0.75 or 1.0 <= i*dx <= 1.25 or 1.5 <= i*dx <= 1.75):
            binary_guess_4[i] = 1.0
    binary_guess_4 = binary_guess_4 * (target_sum / np.sum(binary_guess_4))
    initial_guesses.append(binary_guess_4)
    
    # Binary pattern with five intervals
    binary_guess_5 = np.zeros(n_points)
    for i in range(n_points):
        if (0 <= i*dx <= 0.2 or 0.4 <= i*dx <= 0.6 or 1.0 <= i*dx <= 1.2 or 1.4 <= i*dx <= 1.6 or 1.8 <= i*dx <= 2.0):
            binary_guess_5[i] = 1.0
    binary_guess_5 = binary_guess_5 * (target_sum / np.sum(binary_guess_5))
    initial_guesses.append(binary_guess_5)
    
    # Evaluate each initial guess
    for guess in initial_guesses:
        # Ensure the guess is valid
        guess = np.clip(guess, 0, 1)
        guess = guess * (target_sum / np.sum(guess))
        # Run optimization
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-12, 'maxiter': 2000}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Refinement with targeted perturbations
    for _ in range(50):
        if time.time() - start_time >= budget_s:
            break
        # Compute current maximum overlap
        one_minus_h = 1.0 - best_h
        corr = np.correlate(best_h, one_minus_h, mode='full')
        max_overlap_idx = np.argmax(corr)
        max_overlap_val = corr[max_overlap_idx]
        
        # Identify regions contributing to max_overlap
        shift_idx = max_overlap_idx
        k = (shift_idx - n_points + 1) * dx
        contributing_indices = []
        for i in range(n_points):
            x = i * dx
            x_shifted = x + k
            if x_shifted > 2.0:
                continue
            j = int(round(x_shifted / dx))
            if j >= n_points:
                j = n_points - 1
            if j < 0:
                j = 0
            if best_h[i] > 0.5 and best_h[j] < 0.5:
                contributing_indices.append(i)
        
        # Perturb the contributing regions
        perturb = np.zeros(n_points)
        for i in contributing_indices:
            perturb[i] = np.random.uniform(-0.1, 0.1)
        perturbed = best_h + perturb
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Run optimization on perturbed guess
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx

        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,

            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-12, 'maxiter': 1000}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    return best_h, best_c5, n_points
