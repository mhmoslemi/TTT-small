import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate diverse initial guesses: original, random, and structured
    initial_guesses = [initial_guess]
    
    # Add structured initial guesses
    structured_guess = np.zeros(n_points)
    # Example: place 1s in the first half and last half
    for i in range(n_points // 2):
        structured_guess[i] = 1.0
        structured_guess[-i-1] = 1.0
    structured_guess = structured_guess * (target_sum / np.sum(structured_guess))
    initial_guesses.append(structured_guess)
    
    # Add more structured guesses with different patterns
    for _ in range(2):
        # Randomly place 1s in non-overlapping regions
        num_ones = int(np.round(target_sum))
        indices = np.random.choice(n_points, num_ones, replace=False)
        structured_guess = np.zeros(n_points)
        structured_guess[indices] = 1.0
        structured_guess = structured_guess * (target_sum / np.sum(structured_guess))
        initial_guesses.append(structured_guess)
    
    # Generate random initial guesses
    for _ in range(5):  # 5 random initial guesses
        guess = np.random.uniform(0, 1, n_points)
        guess = guess * (target_sum / np.sum(guess))
        initial_guesses.append(guess)
    
    # Perturbation parameters
    num_perturbations = 600  # Increased for better exploration
    noise_scales = [0.2, 0.1, 0.05, 0.02, 0.01, 0.005]  # Different noise scales for perturbations
    
    for perturbation_idx in range(num_perturbations):
        # Randomly select an initial guess or perturb the best_h
        if perturbation_idx < len(initial_guesses):
            perturbed = initial_guesses[perturbation_idx].copy()
        else:
            # Perturb best_h with varying noise
            noise_scale = noise_scales[perturbation_idx % len(noise_scales)]
            noise = np.random.uniform(-noise_scale, noise_scale, n_points)
            perturbed = best_h + noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
        
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-12, 'maxiter': 300}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
