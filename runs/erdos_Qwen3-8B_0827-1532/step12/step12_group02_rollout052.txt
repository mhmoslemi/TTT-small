            popsize=15,
            strategy='best1bin',
            maxiter=100,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.9,
            seed=seed
        )
        if result.success:
            current_c5 = result.fun - 1e4 * np.abs(np.sum(result.x) - target_sum)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
