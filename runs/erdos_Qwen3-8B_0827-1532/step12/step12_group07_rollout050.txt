import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Use the provided initial guess and generate diverse initial guesses
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate initial guesses
    initial_guesses = []
    # Add initial_h_values
    initial_guesses.append(initial_guess)
    
    # Add structured guess with alternating 1s and 0s
    structured_guess = np.zeros(n_points)
    for i in range(n_points):
        if i % 2 == 0:
            structured_guess[i] = 1.0
    initial_guesses.append(structured_guess)
    
    # Add Thue-Morse sequence as an initial guess
    def generate_thue_morse(n):
        return np.array([bin(i).count('1') % 2 for i in range(n)])
    thue_morse = generate_thue_morse(n_points)
    initial_guesses.append(thue_morse)
    
    # Add random binary guesses
    for _ in range(5):
        np.random.seed(seed + len(initial_guesses))
        h = np.zeros(n_points)
        indices = np.random.choice(n_points, size=int(n_points//2), replace=False)
        h[indices] = 1.0
        initial_guesses.append(h)
    
    # Add sine wave-like initial guess
    def generate_sine_guess(n):
        h = np.zeros(n)
        for i in range(n):
            h[i] = 0.5 + 0.5 * np.sin(2 * np.pi * i / n)
        # Normalize to meet the sum constraint
        h = h * (target_sum / np.sum(h))
        return h
    initial_guesses.append(generate_sine_guess(n_points))
    
    best_c5 = float('inf')
    best_h = initial_guesses[0].copy()
    
    start_time = time.time()
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds and constraints
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization
        result = minimize(
            objective,
            guess,
            method='COBYLA',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 500, 'disp': False}
        )
        
        # Check if successful
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
