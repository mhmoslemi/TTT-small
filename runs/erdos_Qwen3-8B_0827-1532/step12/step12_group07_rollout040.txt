import numpy as np
from scipy.optimize import minimize
import time
from multiprocessing import Pool, cpu_count

def run(seed=42, budget_s=100, **kwargs):
    # Predefined initial guess
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate diverse initial guesses
    initial_guesses = [initial_guess]
    
    # Structured guess: alternating 1s and 0s
    structured_guess = np.zeros(n_points)
    for i in range(n_points):
        if i % 2 == 0:
            structured_guess[i] = 1.0
    initial_guesses.append(structured_guess)
    
    # Thue-Morse sequence
    def generate_thue_morse(n):
        return np.array([bin(i).count('1') % 2 for i in range(n)])
    thue_morse = generate_thue_morse(n_points)
    initial_guesses.append(thue_morse)
    
    # Random binary guesses
    for _ in range(10):
        np.random.seed(seed + len(initial_guesses))
        h = np.zeros(n_points)
        indices = np.random.choice(n_points, size=int(n_points//2), replace=False)
        h[indices] = 1.0
        initial_guesses.append(h)
    
    # Add perturbed versions of initial guess
    for perturb in [0.1, 0.2, 0.3]:
        perturbed = initial_guess.copy()
        np.random.seed(seed + len(initial_guesses))
        perturbed += np.random.normal(0, perturb, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        initial_guesses.append(perturbed)
    
    best_c5 = float('inf')
    best_h = initial_guesses[0].copy()
    
    start_time = time.time()
    
    # Function to evaluate and optimize a single guess
    def optimize_guess(guess):
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds and constraints
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 500}
        )
        
        # Return result if successful
        if result.success:
            return result.fun, result.x
        return float('inf'), guess
    
    # Parallel optimization
    with Pool(min(cpu_count(), 10)) as pool:
        results = pool.map(optimize_guess, initial_guesses)
    
    # Update best solution
    for c5, h in results:
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    
    # Check elapsed time
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
