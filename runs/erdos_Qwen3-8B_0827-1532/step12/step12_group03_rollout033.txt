from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    
    start_time = time.time()
    
    # Generate multiple initial guesses
    initial_guesses = [initial_h_values]
    
    # Two-interval binary guess
    binary_guess_two_intervals = np.zeros(n_points)
    for i in range(n_points):
        if i < 25 or i >= 50 and i < 75:
            binary_guess_two_intervals[i] = 1.0
    binary_guess_two_intervals = binary_guess_two_intervals * (target_sum / np.sum(binary_guess_two_intervals))
    initial_guesses.append(binary_guess_two_intervals)
    
    # Random binary guess
    binary_guess = np.random.permutation(np.concatenate([np.ones(50), np.zeros(50)]))
    initial_guesses.append(binary_guess)
    
    # Random uniform guess
    random_guess = np.random.uniform(0, 1, n_points)
    random_guess = random_guess * (target_sum / np.sum(random_guess))
    initial_guesses.append(random_guess)
    
    # Periodic guess
    periodic_guess = np.zeros(n_points)
    for i in range(n_points):
        if i % 2 == 0:
            periodic_guess[i] = 1.0
    periodic_guess = periodic_guess * (target_sum / np.sum(periodic_guess))
    initial_guesses.append(periodic_guess)
    
    # Evaluate each initial guess with differential evolution
    for guess in initial_guesses:
        # Ensure the guess is valid
        guess = np.clip(guess, 0, 1)
        guess = guess * (target_sum / np.sum(guess))
        # Run optimization
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr) * dx
            # Penalty for constraint violation
            penalty = 1e6 * (np.sum(h) - target_sum)**2
            return max_corr + penalty
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Run differential evolution
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=100,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Additional perturbations to refine the solution
    for _ in range(50):
        if time.time() - start_time >= budget_s:
            break

        noise = np.random.uniform(-0.01, 0.01, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale

            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')

            # Penalty for constraint violation
            penalty = 1e6 * (np.sum(h) - target_sum)**2
            return max_corr + penalty
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Run differential evolution
        result = differential_evolution(

            maxiter=100,

            tol=1e-6,
            mutation=(0.5, 1),

            seed=seed
        )

            best_c5 = result.fun
