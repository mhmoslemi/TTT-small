        cons = ({'type': 'eq', 'fun': lambda h: target_sum - np.sum(h)})
        
        result_slsqp = minimize(
            objective_slsqp,
            result.x,
            method='SLSQP',
            bounds=bounds_slsqp,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 100}
        )
        
        best_h_slsqp = np.clip(result_slsqp.x, 0, 1)
        best_h_slsqp = np.concatenate([best_h_slsqp, [target_sum - np.sum(best_h_slsqp)]])
        best_h_slsqp = np.clip(best_h_slsqp, 0, 1)
        best_c5_slsqp = objective_slsqp(result_slsqp.x)
        
        if best_c5_slsqp < best_c5:
            best_c5 = best_c5_slsqp
            best_h = best_h_slsqp
    
    return best_h, best_c5, n_points
