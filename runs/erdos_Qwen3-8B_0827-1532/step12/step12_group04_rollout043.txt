import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')

        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Run differential evolution with multiple restarts
    for _ in range(3):
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),

            maxiter=100,

            vectorized=True
        )

        elapsed = time.time() - start_time
