    # Use multiprocessing to parallelize the optimizations
    with Pool(processes=10) as pool:
        results = []
        for guess in initial_guesses:
            # Submit the optimization task
            results.append(pool.apply_async(optimize_guess, (guess,)))
        
        # Collect results as they complete
        while results:
            # Check elapsed time
            elapsed = time.time() - start_time
            if elapsed >= budget_s:
                break
            
            # Get the next result
            result = results.pop(0)
            c5, h = result.get()
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
    
    return best_h, best_c5, n_points
