from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Multi-start optimization with adaptive perturbation and diverse initial guesses
    num_starts = 200  # Increased for better exploration
    phase_shift = 0.15  # Larger initial noise magnitude
    phase_decay = 0.98  # Faster decay for noise magnitude
    
    for start_idx in range(num_starts):
        # Generate diverse initial guess
        if start_idx < 50:
            # Random initial guess with uniform distribution
            h_rand = np.random.uniform(0, 1, n_points)
        else:
            # Perturb best_h with noise
            noise = np.random.uniform(-phase_shift, phase_shift, n_points)
            h_rand = best_h + noise
            h_rand = np.clip(h_rand, 0, 1)
        
        # Scale to meet the sum constraint
        current_sum = np.sum(h_rand)
        scale = target_sum / current_sum
        h_rand = h_rand * scale
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with L-BFGS-B for faster convergence
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            h_rand,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 200}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
