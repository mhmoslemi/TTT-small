import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    
    start_time = time.time()
    
    # Generate multiple initial guesses
    num_initial_guesses = 5  # Adjust as needed
    initial_guesses = []
    # Add the initial_h_values as one guess
    initial_guesses.append(initial_h_values.copy())
    # Generate random guesses
    for _ in range(num_initial_guesses - 1):
        h = np.random.uniform(0, 1, n_points)
        h = h * target_sum / np.sum(h)
        initial_guesses.append(h)
    
    # For each initial guess, run optimization
    for guess in initial_guesses:
        # Optimize this guess
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 300}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
