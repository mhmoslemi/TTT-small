    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    
    start_time = time.time()
    
    # Generate diverse initial guesses
    initial_guesses = [initial_h_values]
    
    # Random binary pattern with exactly 50 ones (for n_points=100)
    if n_points == 100:
        random_binary = np.zeros(n_points)
        np.random.shuffle(random_binary)
        random_binary[:50] = 1.0
        initial_guesses.append(random_binary)
    
    # Periodic pattern with period 0.5
    periodic_guess = np.zeros(n_points)
    for i in range(n_points):
        if i % 4 == 0 or i % 4 == 1:
            periodic_guess[i] = 1.0
    periodic_guess = periodic_guess * (target_sum / np.sum(periodic_guess))
    initial_guesses.append(periodic_guess)
    
    # Binary pattern with three intervals
    binary_guess_3 = np.zeros(n_points)
    for i in range(n_points):
        if 0 <= i*dx <= 1/3 or 1 <= i*dx <= 2/3 or 2 <= i*dx <= 3/3:
            binary_guess_3[i] = 1.0
    binary_guess_3 = binary_guess_3 * (target_sum / np.sum(binary_guess_3))
    initial_guesses.append(binary_guess_3)
    
    # Binary pattern with five intervals
    binary_guess_5 = np.zeros(n_points)
    for i in range(n_points):
        if (0 <= i*dx <= 0.2 or 0.4 <= i*dx <= 0.6 or 1.0 <= i*dx <= 1.2 or 1.4 <= i*dx <= 1.6 or 1.8 <= i*dx <= 2.0):
            binary_guess_5[i] = 1.0
    binary_guess_5 = binary_guess_5 * (target_sum / np.sum(binary_guess_5))
    initial_guesses.append(binary_guess_5)
    
    # Staggered pattern with 5 intervals of 0.2 length each
    staggered_guess = np.zeros(n_points)
    for i in range(5):
        start = i * 0.4
        end = start + 0.2
        for j in range(n_points):
            if start <= j*dx <= end:
                staggered_guess[j] = 1.0
    staggered_guess = staggered_guess * (target_sum / np.sum(staggered_guess))
    initial_guesses.append(staggered_guess)
    
    # Binary pattern with 10 intervals of 0.1 length each
    binary_guess_10 = np.zeros(n_points)
    for i in range(10):
        start = i * 0.2
        end = start + 0.1
        for j in range(n_points):
            if start <= j*dx <= end:
                binary_guess_10[j] = 1.0
    binary_guess_10 = binary_guess_10 * (target_sum / np.sum(binary_guess_10))
    initial_guesses.append(binary_guess_10)
    
    # Evaluate each initial guess
    for guess in initial_guesses:
        # Ensure the guess is valid
        guess = np.clip(guess, 0, 1)
        guess = guess * (target_sum / np.sum(guess))
        # Run optimization
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 2000}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Refinement with targeted perturbations
    for _ in range(50):
        if time.time() - start_time >= budget_s:
            break
        # Compute current maximum overlap
        one_minus_h = 1.0 - best_h
        corr = np.correlate(best_h, one_minus_h, mode='full')
        max_overlap_idx = np.argmax(corr)
        max_overlap_val = corr[max_overlap_idx]
        
        # Identify regions contributing to max_overlap
        # For simplicity, perturb the region around the shift
        shift = max_overlap_idx - n_points + 1  # Adjust for full mode
        if shift < 0:
            shift += n_points
        # Perturb the region around shift
        perturb = np.zeros(n_points)
        for i in range(n_points):
            if abs(i - shift) < 5:  # Adjust perturbation range
                perturb[i] = np.random.uniform(-0.1, 0.1)
        perturbed = best_h + perturb
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Run optimization on perturbed guess
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 1000}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    return best_h, best_c5, n_points
