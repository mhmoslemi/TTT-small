    for i in range(n_points):
        if (0 <= i*dx <= 0.25 or 0.5 <= i*dx <= 0.75 or 
            1.0 <= i*dx <= 1.25 or 1.5 <= i*dx <= 1.75):
            binary_guess_2[i] = 1.0
    initial_guesses.append(binary_guess_2)
    
    # Binary step function with eight intervals
    binary_guess_3 = np.zeros(n_points)
    for i in range(n_points):
        if (0 <= i*dx <= 0.25 or 0.5 <= i*dx <= 0.75 or 
            1.0 <= i*dx <= 1.25 or 1.5 <= i*dx <= 1.75 or 
            0.25 <= i*dx <= 0.5 or 0.75 <= i*dx <= 1.0 or 
            1.25 <= i*dx <= 1.5 or 1.75 <= i*dx <= 2.0):
            binary_guess_3[i] = 1.0
    initial_guesses.append(binary_guess_3)
    
    # Evaluate each initial guess
    for guess in initial_guesses:
        # Ensure the guess is valid
        guess = np.clip(guess, 0, 1)
        guess = guess * (target_sum / np.sum(guess))
        # Run optimization
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            # Extract relevant part of cross-correlation for shifts in [0, 2]
            start_idx = n_points - 1
            end_idx = 2 * n_points - 2
            relevant_corr = corr[start_idx:end_idx+1]
            return np.max(relevant_corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-12, 'maxiter': 1000}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Additional perturbations to refine the solution
    for _ in range(50):
        if time.time() - start_time >= budget_s:
            break
        # Perturb the best_h with targeted noise
        indices = np.random.choice(n_points, size=10, replace=False)
        noise = np.zeros(n_points)
        noise[indices] = np.random.uniform(-0.1, 0.1, size=10)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            # Extract relevant part of cross-correlation for shifts in [0, 2]
            start_idx = n_points - 1
            end_idx = 2 * n_points - 2
            relevant_corr = corr[start_idx:end_idx+1]
            return np.max(relevant_corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-12, 'maxiter': 500}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    return best_h, best_c5, n_points
