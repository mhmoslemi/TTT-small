        max_overlap_val = corr[max_overlap_idx]
        
        # Identify regions contributing to max_overlap
        # For simplicity, perturb the region around the shift
        shift = max_overlap_idx - n_points + 1  # Adjust for full mode
        if shift < 0:
            shift += n_points
        # Perturb the region around shift
        perturb = np.zeros(n_points)
        for i in range(n_points):
            if abs(i - shift) < 5:  # Adjust perturbation range
                perturb[i] = np.random.uniform(-0.1, 0.1)
        perturbed = best_h + perturb
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Run optimization on perturbed guess
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='COBYLA',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 1000}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    return best_h, best_c5, n_points
