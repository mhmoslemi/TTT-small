from scipy.optimize import minimize
import time

    alt_pattern[::2] = 1.0
    initial_guesses.append(alt_pattern)
    
    # Add Thue-Morse sequence
    def generate_thue_morse(n):
        return np.array([bin(i).count('1') % 2 for i in range(n)])
    thue_morse = generate_thue_morse(n_points)
    initial_guesses.append(thue_morse)
    
    # Add random binary guesses
    for _ in range(5):
        np.random.seed(seed + len(initial_guesses))
        h = np.zeros(n_points)

        h[indices] = 1.0
        initial_guesses.append(h)

    best_h = initial_guesses[0].copy()

    # Optimize each initial guess
    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds and constraints
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization with tighter tolerances
        result = minimize(
