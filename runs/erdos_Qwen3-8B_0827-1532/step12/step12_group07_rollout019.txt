                if pos in used:
                    continue
                # Calculate overlap with existing 1s
                overlap = 0
                for existing in used:
                    dist = abs(pos - existing)
                    if dist < n:
                        overlap += 1
                if overlap < min_overlap:
                    min_overlap = overlap
                    best_pos = pos
            used.add(best_pos)
