def run(seed=42, budget_s=100, **kwargs):

    n_points = len(initial_guess)
    dx = 2.0 / n_points

    best_c5 = float('inf')
    best_h = initial_guess.copy()

    # Define the objective function
    def objective(h):

        return max_overlap + penalty

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    result = differential_evolution(

        seed=seed
    )
    
    # Check if the result is successful
    if result.success:
        best_h = result.x
