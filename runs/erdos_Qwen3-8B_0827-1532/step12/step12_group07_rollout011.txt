from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

            structured_guess[i] = 1.0
    initial_guesses.append(structured_guess)
    
    # Add Thue-Morse sequence as an initial guess
    def generate_thue_morse(n):
        return np.array([bin(i).count('1') % 2 for i in range(n)])
    thue_morse = generate_thue_morse(n_points)
    initial_guesses.append(thue_morse)
    
    # Add more random binary guesses
    for _ in range(10):  # Increase number of random guesses
        np.random.seed(seed + len(initial_guesses))
        h = np.zeros(n_points)

        h[indices] = 1.0
        initial_guesses.append(h)
    
    best_c5 = float('inf')
    best_h = initial_guesses[0].copy()
    
    start_time = time.time()
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds and constraints
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization with tighter tolerances and more iterations
        result = minimize(

            method='SLSQP',
            bounds=bounds,
            constraints=cons,

        )
        
        # Check if successful
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check elapsed time
