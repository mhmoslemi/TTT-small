    # Add Thue-Morse sequence as an initial guess

        return np.array([bin(i).count('1') % 2 for i in range(n)])

    initial_guesses.append(thue_morse)
    
    # Add structured guess with 1s spaced as evenly as possible

    initial_guesses.append(structured_guess)
    
    # Add a guess with 1s placed at positions that minimize overlap
    def generate_min_overlap_guess(n):

    initial_guesses.append(min_overlap_guess)

        np.random.seed(seed + len(initial_guesses))

    best_h = initial_guesses[0].copy()
    
    start_time = time.time()
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds and constraints
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization with tighter tolerances
        result = minimize(
