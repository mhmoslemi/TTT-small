    phase_shift = 0.1  # Initial noise magnitude
    phase_decay = 0.95  # Decay factor for noise magnitude
    
    for perturbation_idx in range(num_perturbations):
        # Adaptive noise magnitude based on phase
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        
        # Select a random initial guess to start from
        current_guess = initial_guesses[np.random.randint(len(initial_guesses))]

        noise = np.random.uniform(-noise_scale, noise_scale, n_points)

        scale = target_sum / current_sum
        perturbed = perturbed * scale

        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with COBYLA (derivative-free, handles bounds)
