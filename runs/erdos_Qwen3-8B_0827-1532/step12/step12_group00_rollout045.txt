import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    # Start with a structured initial guess: alternating blocks of 1 and 0
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate initial guess with alternating blocks
    initial_guess = np.zeros(n_points)
    block_size = n_points // 4
    for i in range(0, n_points, 2 * block_size):
        initial_guess[i:i+block_size] = 1.0
    initial_guess = initial_guess * (target_sum / np.sum(initial_guess))
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Adaptive perturbation with multiple phases and diverse exploration
    num_perturbations = 300  # Reduced for faster convergence
    phase_shift = 0.2  # Larger initial noise
    phase_decay = 0.98  # Slower decay for more exploration
    exploration_ratio = 0.5  # Fraction of perturbations in exploration phase
    
    for perturbation_idx in range(num_perturbations):
        # Adaptive noise magnitude based on phase
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        
        # Exploration vs refinement phase
        if perturbation_idx < num_perturbations * exploration_ratio:
            # Initial exploration with larger noise
            noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        else:
            # Refinement with smaller noise
            noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        
        # Generate perturbed guess
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 200}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
