            min_corr = np.min(corr)
            return -min_corr
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 2000}
        )
        
        if result.success:
            current_c5 = 1.0 - (np.min(np.correlate(result.x, result.x, mode='full')) / dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # Final C5 computation
    corr = np.correlate(best_h, best_h, mode='full')
    min_corr = np.min(corr)
    c5_bound = 1.0 - (min_corr / dx)
    
    return best_h, c5_bound, n_points
