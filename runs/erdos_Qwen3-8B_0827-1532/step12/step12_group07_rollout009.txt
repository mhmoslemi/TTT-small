import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    # Use the provided initial guess and generate diverse initial guesses
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate initial guesses
    initial_guesses = []
    # Add initial_h_values
    initial_guesses.append(initial_guess)
    
    # Add structured guess with alternating 1s and 0s
    structured_guess = np.zeros(n_points)
    for i in range(n_points):
        if i % 2 == 0:
            structured_guess[i] = 1.0
    initial_guesses.append(structured_guess)
    
    # Add Thue-Morse sequence as an initial guess
    def generate_thue_morse(n):
        return np.array([bin(i).count('1') % 2 for i in range(n)])
    thue_morse = generate_thue_morse(n_points)
    initial_guesses.append(thue_morse)
    
    # Add more structured guesses
    # Binary sequence with 1s spaced evenly
    spaced_guess = np.zeros(n_points)
    num_ones = int(n_points / 2)
    spaced_indices = np.linspace(0, n_points - 1, num_ones, dtype=int)
    spaced_guess[spaced_indices] = 1.0
    initial_guesses.append(spaced_guess)
    
    # Add more random binary guesses
    for _ in range(10):  # Increase number of random guesses
        np.random.seed(seed + len(initial_guesses))
        h = np.zeros(n_points)
        indices = np.random.choice(n_points, size=int(n_points//2), replace=False)
        h[indices] = 1.0
        initial_guesses.append(h)
    
    best_c5 = float('inf')
    best_h = initial_guesses[0].copy()
    
    start_time = time.time()
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds and constraints
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization
        # Use differential evolution for some guesses to escape local minima
        if np.random.rand() < 0.5:  # 50% chance to use differential evolution
            result = differential_evolution(
                objective,
                bounds,
                constraints=cons,
                strategy='best1bin',
                popsize=15,
                maxiter=100,
                tol=1e-6,
                mutation=(0.5, 1),
                recombination=0.8,
                seed=seed
            )
            h_opt = result.x
        else:
            result = minimize(
                objective,
                guess,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'ftol': 1e-10, 'maxiter': 500}
            )
            h_opt = result.x
        
        # Check if successful
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = h_opt
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
