    best_h = result.x
    best_c5 = result.fun
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return best_h, best_c5, n_points
    else:
        # If time is not up, maybe do more iterations or other optimizations
        # But for simplicity, return the result
        return best_h, best_c5, n_points
