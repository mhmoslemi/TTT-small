        # Check if the result is successful and better
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Check if we have time left and can perform more optimizations
    elapsed = time.time() - start_time
    if elapsed < budget_s:
        # Perform additional optimizations with perturbations
        num_perturbations = 100  # Reduced from previous
        phase_shift = 0.1
        phase_decay = 0.95
        
        for perturbation_idx in range(num_perturbations):
            noise_scale = phase_shift * (phase_decay ** perturbation_idx)
            noise = np.random.uniform(-noise_scale, noise_scale, n_points)
            perturbed = best_h + noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
            
            # Objective function
            def objective_perturbed(h):
                one_minus_h = 1.0 - h
                corr = np.correlate(h, one_minus_h, mode='full')
                return np.max(corr) * dx
            
            bounds_perturbed = [(0.0, 1.0) for _ in range(n_points)]
            cons_perturbed = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
            
            result_perturbed = minimize(
                objective_perturbed,
                perturbed,
                method='SLSQP',
                bounds=bounds_perturbed,
                constraints=cons_perturbed,
                options={'ftol': 1e-9, 'maxiter': 30}
            )
            
            if result_perturbed.success and result_perturbed.fun < best_c5:
                best_c5 = result_perturbed.fun
                best_h = result_perturbed.x
            
            # Check time
            if time.time() - start_time >= budget_s:
                break
    
    return best_h, best_c5, n_points
