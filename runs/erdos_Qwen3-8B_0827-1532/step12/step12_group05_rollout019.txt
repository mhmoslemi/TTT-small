        indices = np.random.choice(n_points, size=int(target_sum), replace=False)
        guess[indices] = 1.0
        guess = guess * (target_sum / np.sum(guess))
        initial_guesses.append(guess)
    
    # Periodic pattern with period 0.5
    periodic_guess = np.zeros(n_points)
    for i in range(n_points):
        if i % 4 == 0 or i % 4 == 1:
            periodic_guess[i] = 1.0
    periodic_guess = periodic_guess * (target_sum / np.sum(periodic_guess))
    initial_guesses.append(periodic_guess)
    
    # Binary pattern with three intervals
    binary_guess_3 = np.zeros(n_points)
    for i in range(n_points):
        if 0 <= i*dx <= 1/3 or 1 <= i*dx <= 2/3 or 2 <= i*dx <= 3/3:
            binary_guess_3[i] = 1.0
    binary_guess_3 = binary_guess_3 * (target_sum / np.sum(binary_guess_3))
    initial_guesses.append(binary_guess_3)
    
    # Binary pattern with five intervals
    binary_guess_5 = np.zeros(n_points)
    for i in range(n_points):
        if (0 <= i*dx <= 0.2 or 0.4 <= i*dx <= 0.6 or 1.0 <= i*dx <= 1.2 or 1.4 <= i*dx <= 1.6 or 1.8 <= i*dx <= 2.0):
            binary_guess_5[i] = 1.0
    binary_guess_5 = binary_guess_5 * (target_sum / np.sum(binary_guess_5))
    initial_guesses.append(binary_guess_5)
    
    # Evaluate each initial guess
    for guess in initial_guesses:
        # Ensure the guess is valid
        guess = np.clip(guess, 0, 1)
        guess = guess * (target_sum / np.sum(guess))
        # Run optimization
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 1000}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Refinement with targeted perturbations
    for _ in range(50):
        if time.time() - start_time >= budget_s:
