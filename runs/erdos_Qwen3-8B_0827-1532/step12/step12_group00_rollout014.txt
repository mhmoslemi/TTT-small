import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate a few random initial guesses
    initial_guesses = [initial_guess]
    for _ in range(3):
        h = np.random.uniform(0, 1, n_points)
        h = h * (target_sum / np.sum(h))
        initial_guesses.append(h)
    
    # Adaptive perturbation strategy with varying noise and exploration phases
    num_perturbations = 100  # Reduced for more diverse initial guesses
    phase_shift = 0.2  # Larger initial noise
    phase_decay = 0.98  # Slower decay
    
    for guess in initial_guesses:
        # Optimization with COBYLA
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        result = minimize(
            objective,
            guess,
            method='COBYLA',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 500}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    return best_h, best_c5, n_points
