        guess = guess * (target_sum / np.sum(guess))
        
        # Define objective function with penalty
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr) * dx
            penalty = 1e4 * (np.sum(h) - target_sum)**2
            return max_corr + penalty
        
        # Bounds for each parameter
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Run differential evolution
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=100,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            polish=True,
            disp=False
        )
        
        # Check if successful
        if result.success:
            current_h = result.x
            # Compute actual C5
            one_minus_h = 1.0 - current_h
            corr = np.correlate(current_h, one_minus_h, mode='full')
            current_c5 = np.max(corr) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
    
    # Additional perturbations
    for _ in range(50):
        if time.time() - start_time >= budget_s:
            break
        # Perturb the best_h
        indices = np.random.choice(n_points, size=10, replace=False)
        noise = np.zeros(n_points)
        noise[indices] = np.random.uniform(-0.1, 0.1, size=10)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Define objective function
        def objective_perturbed(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr) * dx
            penalty = 1e4 * (np.sum(h) - target_sum)**2
            return max_corr + penalty
        
        # Run differential evolution on perturbed guess
        result_perturbed = differential_evolution(
            objective_perturbed,
            bounds,
            strategy='best1bin',
            maxiter=50,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            polish=True,
            disp=False
        )
        
        if result_perturbed.success:
            current_h_perturbed = result_perturbed.x
            # Compute actual C5
            one_minus_h_perturbed = 1.0 - current_h_perturbed
            corr_perturbed = np.correlate(current_h_perturbed, one_minus_h_perturbed, mode='full')
            current_c5_perturbed = np.max(corr_perturbed) * dx
            if current_c5_perturbed < best_c5:
                best_c5 = current_c5_perturbed
                best_h = current_h_perturbed
    
    return best_h, best_c5, n_points
