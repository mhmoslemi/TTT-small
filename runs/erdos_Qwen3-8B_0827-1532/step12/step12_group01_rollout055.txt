    # Evaluate each initial guess
    for guess in initial_guesses:
        guess = np.clip(guess, 0, 1)
        guess = guess * (target_sum / np.sum(guess))
        
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 5000}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Perturbation phase with targeted adjustments
    for _ in range(100):
        if time.time() - start_time >= budget_s:
            break
        
        # Compute current overlap
        one_minus_h = 1.0 - best_h
        corr = np.correlate(best_h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        max_index = np.argmax(corr)
        
        # Determine the shift k (mode 'full' gives shifts from -n_points+1 to n_points-1)
        # For simplicity, assume the shift is max_index - n_points + 1
        shift = max_index - n_points + 1
        
        # Adjust regions contributing to the maximum overlap
        # For shift k, the overlap is over x in [0, 2 -k]
        # We adjust the first half of the array to reduce overlap
        perturbed = best_h.copy()
        delta = 0.05
        perturbed[:n_points//2] -= delta
        perturbed[n_points//2:] += delta
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 3000}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    return best_h, best_c5, n_points
