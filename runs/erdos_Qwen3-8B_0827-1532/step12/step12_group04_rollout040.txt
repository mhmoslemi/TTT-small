from scipy.optimize import minimize
import time

    # Generate diverse initial guesses
    initial_guesses = [initial_guess.copy()]

    symmetric = np.zeros(n_points)

        symmetric[i] = 1.0 if i < n_points//2 else 0.0
    initial_guesses.append(symmetric)
    # Add periodic pattern
    periodic = np.zeros(n_points)
    for i in range(n_points):
        periodic[i] = 0.5 + 0.5 * np.sin(2 * np.pi * i / n_points)
    initial_guesses.append(periodic)
    # Add alternating pattern
    alternating = np.zeros(n_points)
    for i in range(n_points):
        alternating[i] = 1.0 if (i % 4) < 2 else 0.0
    initial_guesses.append(alternating)
    # Add random guesses
    for _ in range(3):
        guess = np.random.uniform(0, 1, n_points)
        guess = guess * (target_sum / np.sum(guess))
        initial_guesses.append(guess)
    
    # Optimization parameters
    max_iterations = 300
    perturbation_count = 250
    noise_scales = [0.2, 0.1, 0.05, 0.02]
    
    for perturb_idx in range(perturbation_count):
        # Select a guess
        if perturb_idx < len(initial_guesses):
            perturbed = initial_guesses[perturb_idx].copy()
        else:
            # Generate a new perturbation based on best_h
            if perturb_idx % 4 == 0:
                # Random noise with larger scale
                noise = np.random.uniform(-0.2, 0.2, n_points)
            elif perturb_idx % 4 == 1:
                # Random noise with medium scale
                noise = np.random.uniform(-0.1, 0.1, n_points)
            elif perturb_idx % 4 == 2:
                # Random noise with small scale
                noise = np.random.uniform(-0.05, 0.05, n_points)
            else:
                # Swap values in random segments
                noise = np.zeros(n_points)
                for _ in range(2):
                    start = np.random.randint(0, n_points-5)

                    noise[start:end] = np.random.uniform(-0.1, 0.1, end - start)
            perturbed = best_h + noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
        
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

        )
        
        if result.success and result.fun < best_c5:

            best_h = result.x
