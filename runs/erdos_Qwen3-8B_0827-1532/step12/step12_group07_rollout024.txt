    # Add structured guess with three intervals
    structured_guess3 = np.zeros(n_points)
    interval_length = 2.0 / 3
    for i in range(n_points):

    initial_guesses.append(structured_guess3)
    
    # Add Thue-Morse sequence as an initial guess

        np.random.seed(seed + len(initial_guesses))

    start_time = time.time()
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds and constraints
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
