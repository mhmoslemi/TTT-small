from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate diverse initial guesses: original, random, and binary step
    initial_guesses = [initial_guess]
    for _ in range(5):  # 5 random initial guesses
        guess = np.random.uniform(0, 1, n_points)
        guess = guess * (target_sum / np.sum(guess))
        initial_guesses.append(guess)
    # Add binary step initial guesses
    for _ in range(3):
        perm = np.random.permutation(n_points)
        binary_guess = np.zeros(n_points)
        binary_guess[perm[:int(n_points/2)]] = 1
        binary_guess = binary_guess * (target_sum / np.sum(binary_guess))
        initial_guesses.append(binary_guess)
    
    # Perturbation parameters
    num_perturbations = 600  # Increased for better exploration
    noise_scales = [0.2, 0.1, 0.05, 0.02, 0.01]  # Different noise scales for perturbations
    
    for perturbation_idx in range(num_perturbations):
        # Randomly select an initial guess or perturb the best_h
        if perturbation_idx < len(initial_guesses):
            perturbed = initial_guesses[perturbation_idx].copy()
        else:
            # Perturb best_h with varying noise
            noise_scale = noise_scales[perturbation_idx % len(noise_scales)]
            noise = np.random.uniform(-noise_scale, noise_scale, n_points)
            perturbed = best_h + noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
        
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Use differential evolution for global search
        if perturbation_idx % 10 == 0:
            result = differential_evolution(
                objective,
                bounds,
                constraints=cons,
                strategy='best1bin',
                popsize=15,
                mutation=(0.5, 1),
                recombination=0.9,
                tol=1e-6,
                maxiter=50,
                disp=False
            )
            if result.success:
                best_h = result.x
                best_c5 = result.fun
        else:
            # Use SLSQP for local refinement
            result = minimize(
                objective,
                perturbed,

                bounds=bounds,
                constraints=cons,

            )
            if result.success and result.fun < best_c5:
                best_c5 = result.fun
                best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
