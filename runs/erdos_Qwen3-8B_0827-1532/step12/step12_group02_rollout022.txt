from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate diverse initial guesses: original, random, and structured
    initial_guesses = [initial_guess]
    # Add a structured initial guess: binary pattern with 1s in non-overlapping intervals
    structured_guess = np.zeros(n_points)
    interval_length = 2.0 / 5  # Divide into 5 intervals of 0.4 each
    for i in range(5):
        start_idx = int(i * interval_length / dx)
        end_idx = int((i + 1) * interval_length / dx)
        structured_guess[start_idx:end_idx] = 1.0
    # Adjust to meet the sum constraint
    current_sum = np.sum(structured_guess)
    scale = target_sum / current_sum
    structured_guess = structured_guess * scale
    initial_guesses.append(structured_guess)
    
    # Perturbation parameters
    num_perturbations = 300  # Reduced for better exploration
    noise_scales = [0.2, 0.1, 0.05, 0.02, 0.01]  # Different noise scales for perturbations
    
    for perturbation_idx in range(num_perturbations):
        # Randomly select an initial guess or perturb the best_h
        if perturbation_idx < len(initial_guesses):
            perturbed = initial_guesses[perturbation_idx].copy()
        else:
            # Perturb best_h with varying noise
            noise_scale = noise_scales[perturbation_idx % len(noise_scales)]
            noise = np.random.uniform(-noise_scale, noise_scale, n_points)
            perturbed = best_h + noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
        
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Use L-BFGS-B for better handling of bounds
        result = minimize(
