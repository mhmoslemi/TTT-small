import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_guesses = [initial_h_values]
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    
    start_time = time.time()
    
    # Generate additional random initial guesses
    for _ in range(2):
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        h = np.clip(h, 0, 1)
        initial_guesses.append(h)
    
    # Adaptive perturbation strategy with varying noise and exploration phases
    num_perturbations = 500  # Increased for better exploration
    phase_shift = 0.1  # Initial noise magnitude
    phase_decay = 0.95  # Decay factor for noise magnitude
    
    for initial_guess in initial_guesses:
        current_h = initial_guess.copy()
        current_sum = np.sum(current_h)
        scale = target_sum / current_sum
        current_h = current_h * scale
        
        best_c5_current = float('inf')
        best_h_current = current_h.copy()
        
        for perturbation_idx in range(num_perturbations):
            # Adaptive noise magnitude based on phase
            noise_scale = phase_shift * (phase_decay ** perturbation_idx)
            
            # Generate perturbed guess
            noise = np.random.uniform(-noise_scale, noise_scale, n_points)
            perturbed = current_h + noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum_perturbed = np.sum(perturbed)
            scale_perturbed = target_sum / current_sum_perturbed
            perturbed = perturbed * scale_perturbed
            
            # Objective function
            def objective(h):
                one_minus_h = 1.0 - h
                corr = np.correlate(h, one_minus_h, mode='full')
                return np.max(corr) * dx
            
            # Optimization with SLSQP
            bounds = [(0.0, 1.0) for _ in range(n_points)]
            cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
            
            result = minimize(
                objective,
                perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'ftol': 1e-9, 'maxiter': 300}
            )
            
            if result.success and result.fun < best_c5_current:
                best_c5_current = result.fun
                best_h_current = result.x
        
        if best_c5_current < best_c5:
            best_c5 = best_c5_current
            best_h = best_h_current
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
