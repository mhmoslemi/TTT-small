import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate multiple initial guesses
    initial_guesses = [initial_guess]
    # Add a random guess
    random_guess = np.random.rand(n_points)

    initial_guesses.append(random_guess)
    # Add a structured guess with alternating 0.5 values
    structured_guess = np.zeros(n_points)
    for i in range(n_points):
        structured_guess[i] = 0.5 if i % 2 == 0 else 0.5
    structured_guess = structured_guess * (target_sum / np.sum(structured_guess))
    initial_guesses.append(structured_guess)
    
    # For each initial guess, perform optimization
    for guess in initial_guesses:
        # Clip to [0,1]
        guess = np.clip(guess, 0, 1)
        # Ensure sum is correct (due to rounding errors)
        guess = guess * (target_sum / np.sum(guess))

        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')

        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 300}
        )
        
        if result.success and result.fun < best_c5:
