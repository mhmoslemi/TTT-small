import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate a more informed initial guess with periodic structure
    initial_guess = np.zeros(n_points)
    for i in range(n_points):
        x = i * dx
        if 0 <= x < 0.5:
            initial_guess[i] = 1.0
        elif 0.5 <= x < 1.5:
            initial_guess[i] = 0.0
        else:
            initial_guess[i] = 1.0
    initial_guess = initial_guess * target_sum / np.sum(initial_guess)
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Adaptive perturbation strategy with varying noise and exploration phases
    num_perturbations = 800  # Increased for better exploration
    phase_shift = 0.15  # Initial noise magnitude
    phase_decay = 0.92  # Slower decay for more exploration
    
    for perturbation_idx in range(num_perturbations):
        # Adaptive noise magnitude based on phase
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        
        # Generate perturbed guess
        perturbed = best_h + noise_scale * np.random.randn(n_points)
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 150}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
