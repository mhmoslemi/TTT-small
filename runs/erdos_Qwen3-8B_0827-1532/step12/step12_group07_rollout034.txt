import numpy as np
from scipy.optimize import minimize
import time

    initial_guesses.append(initial_guess)

    for i in range(n_points):
        if i % 2 == 0:
            structured_guess[i] = 1.0
    initial_guesses.append(structured_guess)
    
    # Add Thue-Morse sequence as an initial guess
    def generate_thue_morse(n):
        return np.array([bin(i).count('1') % 2 for i in range(n)])
    thue_morse = generate_thue_morse(n_points)
    initial_guesses.append(thue_morse)
    
    # Add more structured guesses
    # Add a binary sequence with 1s every 3rd point
    structured_guess_3 = np.zeros(n_points)
    for i in range(n_points):
        if (i % 3) == 0:
            structured_guess_3[i] = 1.0
    initial_guesses.append(structured_guess_3)
    
    # Add a binary sequence with 1s every 4th point
    structured_guess_4 = np.zeros(n_points)
    for i in range(n_points):
        if (i % 4) == 0:
            structured_guess_4[i] = 1.0
    initial_guesses.append(structured_guess_4)
    
    # Add more random binary guesses
    for _ in range(10):
        np.random.seed(seed + len(initial_guesses))
        h = np.zeros(n_points)
        indices = np.random.choice(n_points, size=int(n_points//2), replace=False)
        h[indices] = 1.0
        initial_guesses.append(h)
    
    best_c5 = float('inf')
    best_h = initial_guesses[0].copy()
    
    start_time = time.time()
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds and constraints
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 5000}
        )
        
        # Check if successful
        if result.success and result.fun < best_c5:

            best_h = result.x
