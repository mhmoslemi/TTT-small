import numpy as np
from scipy.optimize import minimize
import time
import multiprocessing as mp

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Adaptive perturbation strategy with varying noise and exploration phases
    num_perturbations = 1000  # Increased for better exploration
    phase_shift = 0.1  # Initial noise magnitude
    phase_decay = 0.99  # Faster decay for noise magnitude
    
    # Function to evaluate the overlap integral
    def evaluate_overlap(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    # Function to optimize a single perturbation
    def optimize_perturbation(perturbed):
        # Ensure the perturbed guess is within bounds and scaled
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Optimization with COBYLA
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            evaluate_overlap,
            perturbed,
            method='COBYLA',
            bounds=bounds,
            constraints=cons,
            options={'disp': False, 'maxiter': 200}
        )
        
        return result.x, result.fun
    
    # Parallel processing
    pool = mp.Pool(processes=10)
    results = []
    
    for perturbation_idx in range(num_perturbations):
        # Adaptive noise magnitude based on phase
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        
        # Submit to pool
        results.append(pool.apply_async(optimize_perturbation, (perturbed,)))
        
        # Check time budget
        if time.time() - start_time >= budget_s:
            break
    
    # Collect results
    for res in results:
        h_opt, c5_val = res.get()
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt
    
    pool.close()
    pool.join()
    
    return best_h, best_c5, n_points
