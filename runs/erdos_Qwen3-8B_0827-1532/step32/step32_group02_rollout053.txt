            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points

def compute_c5(h, dx):
    n_points = len(h)
    max_c5 = 0.0
    for m in range(n_points):
        k = m * dx
        integral = 0.0
        for i in range(n_points):
            x = i * dx
            x_plus_k = x + k
            if 0 <= x_plus_k <= 2:
                j = int(x_plus_k / dx)
                if j < n_points:
                    integral += h[i] * (1 - h[j]) * dx
                else:
                    integral += 0.0
            else:
                integral += 0.0
        max_c5 = max(max_c5, integral)
    return max_c5
