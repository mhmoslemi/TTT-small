def compute_c5(h, dx):
    n_points = len(h)
    max_overlap = 0.0
    # Iterate over all possible shifts k
    # The maximum k that can have any overlap is 2.0
    # So, k ranges from -2.0 to 2.0 in steps of dx
    for k in np.arange(-2.0, 2.0 + dx, dx):
        integral = 0.0
        for x in range(n_points):
            x_val = x * dx
            x_plus_k = x_val + k
            if 0 <= x_plus_k <= 2.0:
                # Find the index corresponding to x_plus_k
                idx = int(x_plus_k / dx)
                if idx >= n_points:
                    idx = n_points - 1
                elif idx < 0:
                    idx = 0
                h_x_plus_k = h[idx]
            else:
                h_x_plus_k = 0.0
            integral += h[x] * (1.0 - h_x_plus_k) * dx
        if integral > max_overlap:
            max_overlap = integral
    return max_overlap
