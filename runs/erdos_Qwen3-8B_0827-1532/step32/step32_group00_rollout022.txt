    phase_decay = 0.98
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function with penalty
        def objective_perturb(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr) * dx
            penalty = 1e-3 * (np.sum(h) - target_sum)**2
            return max_corr + penalty
        
        # Use differential evolution for perturbations
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result_perturb = differential_evolution(
            objective_perturb,
            bounds,
            strategy='best1bin',
            popsize=15,
            maxiter=50,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.9,
            x0=perturbed
        )
        
        if result_perturb.success:
            current_c5 = compute_c5(result_perturb.x, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_perturb.x.copy()
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points

def compute_c5(h, dx):
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return np.max(corr) * dx
