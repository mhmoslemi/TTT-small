    h_five_intervals = np.zeros(n_points)
    interval_length = 0.3333
    start = 0.0
    for i in range(5):
        end = start + interval_length
        start_idx = int(start / dx)
        end_idx = int(end / dx)
        h_five_intervals[start_idx:end_idx] = 1.0
        start = end
    sum_h = np.sum(h_five_intervals)
    scale = target_sum / sum_h
    initial_guesses.append(h_five_intervals * scale)
    
    # Generate random step functions with varying numbers of segments
    for m_segments in [2, 3, 4]:
        h_rand_seg = np.zeros(n_points)
        segment_lengths = np.random.rand(m_segments)
        segment_lengths /= np.sum(segment_lengths)
        start_positions = np.random.rand(m_segments) * (2 - 1)
        for i in range(m_segments):
            start = start_positions[i]
            end = start + segment_lengths[i]
            start_idx = int(start / dx)
            end_idx = int(end / dx)
            h_rand_seg[start_idx:end_idx] = 1.0
        sum_h = np.sum(h_rand_seg)
        scale = target_sum / sum_h
        initial_guesses.append(h_rand_seg * scale)
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 5000}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Perturbation phase on the best_h found so far
    num_perturbations = 500
    phase_shift = 0.15
    phase_decay = 0.96
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 1000}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points

def compute_c5(h, dx):
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return np.max(corr) * dx
