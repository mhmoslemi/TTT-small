    sum_h = np.sum(h_three)
    scale = target_sum / sum_h
    h_three = h_three * scale
    initial_guesses.append(h_three)
    
    # Four-segment pattern
    h_four = np.zeros(n_points)
    segment_length = n_points // 4
    h_four[:segment_length] = 1.0
    h_four[segment_length:2*segment_length] = 0.5
    h_four[2*segment_length:3*segment_length] = 1.0
    h_four[3*segment_length:] = 0.5
    sum_h = np.sum(h_four)
    scale = target_sum / sum_h
    h_four = h_four * scale
    initial_guesses.append(h_four)
    
    # Five-segment pattern
    h_five = np.zeros(n_points)
    segment_length = n_points // 5
    h_five[:segment_length] = 1.0
    h_five[segment_length:2*segment_length] = 0.5
    h_five[2*segment_length:3*segment_length] = 1.0
    h_five[3*segment_length:4*segment_length] = 0.5
    h_five[4*segment_length:] = 1.0
    sum_h = np.sum(h_five)
    scale = target_sum / sum_h
    h_five = h_five * scale
    initial_guesses.append(h_five)
    
    # Optimize each initial guess with differential evolution
    for guess in initial_guesses:
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=15,
            maxiter=100,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.9,
            seed=seed
        )
        
        if result.success:
            current_c5 = result.fun - 1e3 * abs(np.sum(result.x) - target_sum)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # Perturbation phase
    num_perturbations = 100
    phase_shift = 0.15
    phase_decay = 0.96
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Optimize the perturbed guess
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=15,
            maxiter=50,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.9,
            seed=seed
        )
        
        if result.success:
            current_c5 = result.fun - 1e3 * abs(np.sum(result.x) - target_sum)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points

def compute_c5(h, dx):
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return np.max(corr) * dx
