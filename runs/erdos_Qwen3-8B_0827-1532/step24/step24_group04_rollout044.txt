        strategy='best1bin',
        popsize=15,
        mutation=0.5,
        recombination=0.8,
        tol=1e-8,
        maxiter=100,
        x0=initial_guess
    )
    
    if result.success:
        best_h = result.x
        best_c5 = result.fun
    
    # Generate other initial guesses
    # Step function
    h_step = np.zeros(n_points)
    h_step[:n_points//2] = 1.0
    sum_h = np.sum(h_step)
    scale = target_sum / sum_h
    h_step = h_step * scale
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=15,
        mutation=0.5,
        recombination=0.8,
        tol=1e-8,
        maxiter=100,
        x0=h_step
    )
    if result.success and result.fun < best_c5:
        best_h = result.x
        best_c5 = result.fun
    
    # Alternating pattern
    h_alt = np.zeros(n_points)
    h_alt[::2] = 1.0
    sum_h = np.sum(h_alt)
    scale = target_sum / sum_h
    h_alt = h_alt * scale
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=15,
        mutation=0.5,
        recombination=0.8,
        tol=1e-8,
        maxiter=100,
        x0=h_alt
    )
    if result.success and result.fun < best_c5:
        best_h = result.x
        best_c5 = result.fun
    
    # Central 50% step function
    h_center = np.zeros(n_points)
    center_start = n_points // 2 - n_points // 4
    center_end = n_points // 2 + n_points // 4
    h_center[center_start:center_end] = 1.0
    sum_h = np.sum(h_center)
    scale = target_sum / sum_h
    h_center = h_center * scale
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=15,
        mutation=0.5,
        recombination=0.8,
        tol=1e-8,
        maxiter=100,
        x0=h_center
    )
    if result.success and result.fun < best_c5:
        best_h = result.x
        best_c5 = result.fun
    
    # Perturbation phase
    num_perturbations = 200
    phase_shift = 0.7
    phase_decay = 0.98
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Run DE on perturbed solution
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=15,
            mutation=0.5,
            recombination=0.8,
            tol=1e-8,
            maxiter=100,
            x0=perturbed
        )
        
        if result.success and result.fun < best_c5:
            best_h = result.x
            best_c5 = result.fun
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points

def compute_c5(h, dx):
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return np.max(corr) * dx
