    initial_guesses.append(h_three * scale)
    
    # Triangular function
    h_tri = np.zeros(n_points)
    for i in range(n_points):
        x = i * dx
        if x < 1.0:
            h_tri[i] = x / 1.0
        elif x < 2.0:
            h_tri[i] = (2.0 - x) / 1.0
        else:
            h_tri[i] = 0.0
    sum_h = np.sum(h_tri)
    scale = target_sum / sum_h
    initial_guesses.append(h_tri * scale)
    
    # Random initial guess
    initial_guesses.append(np.random.uniform(0, 1, n_points))
    
    # Generate more diverse initial guesses with different patterns
    for _ in range(3):
        h_rand = np.random.uniform(0, 1, n_points)
        sum_h = np.sum(h_rand)
        scale = target_sum / sum_h
        initial_guesses.append(h_rand * scale)
    
    # Add more structured initial guesses
    # Two intervals with different spacing
    h_two_diff = np.zeros(n_points)
    h_two_diff[:n_points//3] = 1.0  # first interval
    h_two_diff[-n_points//3:] = 1.0  # second interval
    sum_h = np.sum(h_two_diff)
    scale = target_sum / sum_h
    initial_guesses.append(h_two_diff * scale)
    
    # Three intervals with different spacing
    h_three_diff = np.zeros(n_points)
    h_three_diff[:n_points//4] = 1.0  # first interval
    h_three_diff[n_points//2:n_points//2 + n_points//4] = 1.0  # second interval
    h_three_diff[-n_points//4:] = 1.0  # third interval
    sum_h = np.sum(h_three_diff)
    scale = target_sum / sum_h
    initial_guesses.append(h_three_diff * scale)
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-12, 'maxiter': 2000}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Perturbation phase on the best_h found so far
    num_perturbations = 1000
    phase_shift = 0.5
    phase_decay = 0.95
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-12, 'maxiter': 1000}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points

def compute_c5(h, dx):
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return np.max(corr) * dx
