            current_c5 = compute_c5(current_h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
    
    # Refine the best solution with SLSQP
    def objective_slsqp(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]
    cons_slsqp = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    
    result_slsqp = minimize(
        objective_slsqp,
        best_h,
        method='SLSQP',
        bounds=bounds_slsqp,
        constraints=cons_slsqp,
        options={'ftol': 1e-10, 'maxiter': 2000}
    )
    
    if result_slsqp.success:
        best_h = result_slsqp.x
        best_c5 = result_slsqp.fun
    
    # Perturbation phase
    num_perturbations = 1500
    phase_shift = 0.5
    phase_decay = 0.99
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective_perturb(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        result_perturb = minimize(
            objective_perturb,
            perturbed,
            method='SLSQP',
            bounds=bounds_slsqp,
            constraints=cons_slsqp,
            options={'ftol': 1e-10, 'maxiter': 300}
        )
        
        if result_perturb.success and result_perturb.fun < best_c5:
            best_c5 = result_perturb.fun
            best_h = result_perturb.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points

def compute_c5(h, dx):
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return np.max(corr) * dx
