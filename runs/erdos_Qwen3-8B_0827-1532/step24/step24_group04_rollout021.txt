    start_time = time.time()
    
    # Run differential evolution
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.9,
        tol=1e-12,
        maxiter=100,
        seed=seed,
        polish=True,
        disp=False
    )
    
    best_h = result.x
    best_c5 = result.fun
    
    # Perturbation phase to refine the solution
    num_perturbations = 500
    phase_shift = 0.7
    phase_decay = 0.98
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective_perturb(h):
            h_perturb = h.copy()
            h_perturb = h_perturb / np.sum(h_perturb) * target_sum
            one_minus_h = 1.0 - h_perturb
            corr = np.correlate(h_perturb, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Run local optimization
        result_perturb = differential_evolution(
            objective_perturb,
            bounds,
            strategy='best1bin',
            popsize=10,
            mutation=(0.5, 1),
            recombination=0.9,
            tol=1e-12,
            maxiter=50,
            seed=seed,
            polish=True,
            disp=False
        )
        
        if result_perturb.fun < best_c5:
            best_c5 = result_perturb.fun
            best_h = result_perturb.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
