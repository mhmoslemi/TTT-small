    initial_guess = np.clip(initial_guess, 0, 1)
    scale = target_sum / np.sum(initial_guess)
    initial_guess = initial_guess * scale
    best_h = initial_guess.copy()
    best_c5 = compute_c5(best_h, dx)
    
    # Generate more initial guesses with different patterns
    initial_guesses = [initial_guess]
    
    # Step function: first half 1, second half 0
    h_step = np.zeros(n_points)
    h_step[:n_points//2] = 1.0
    sum_h = np.sum(h_step)
    scale = target_sum / sum_h
    initial_guesses.append(h_step * scale)
    
    # Alternating 1 and 0
    h_alt = np.zeros(n_points)
    h_alt[::2] = 1.0
    sum_h = np.sum(h_alt)
    scale = target_sum / sum_h
    initial_guesses.append(h_alt * scale)
    
    # Central 50% step function
    h_center = np.zeros(n_points)
    center_start = n_points // 2 - n_points // 4
    center_end = n_points // 2 + n_points // 4
    h_center[center_start:center_end] = 1.0
    sum_h = np.sum(h_center)
    scale = target_sum / sum_h
    initial_guesses.append(h_center * scale)
    
    # 25% step function
    h_25 = np.zeros(n_points)
    h_25[:n_points//4] = 1.0
    h_25[n_points//2:n_points//2 + n_points//4] = 1.0
    sum_h = np.sum(h_25)
    scale = target_sum / sum_h
    initial_guesses.append(h_25 * scale)
    
    # 4-segment step function
    h_4seg = np.zeros(n_points)
    h_4seg[:n_points//4] = 1.0
    h_4seg[n_points//2:n_points//2 + n_points//4] = 1.0
    h_4seg[3*n_points//4:] = 1.0
    sum_h = np.sum(h_4seg)
    scale = target_sum / sum_h
    initial_guesses.append(h_4seg * scale)
    
    # 5-segment step function
    h_5seg = np.zeros(n_points)
    h_5seg[:n_points//5] = 1.0
    h_5seg[2*n_points//5:3*n_points//5] = 1.0
    h_5seg[4*n_points//5:] = 1.0
    sum_h = np.sum(h_5seg)
    scale = target_sum / sum_h
    initial_guesses.append(h_5seg * scale)
    
    # 3-segment step function: first 1/3, middle 1/3, last 1/3
    h_3seg = np.zeros(n_points)
    h_3seg[:n_points//3] = 1.0
    h_3seg[n_points//3:2*n_points//3] = 0.0
    h_3seg[2*n_points//3:] = 1.0
    sum_h = np.sum(h_3seg)
    scale = target_sum / sum_h
    initial_guesses.append(h_3seg * scale)
    
    # 3-segment step function: first 1/4, middle 1/2, last 1/4
    h_3seg2 = np.zeros(n_points)
    h_3seg2[:n_points//4] = 1.0
    h_3seg2[n_points//4:3*n_points//4] = 0.0
    h_3seg2[3*n_points//4:] = 1.0
    sum_h = np.sum(h_3seg2)
    scale = target_sum / sum_h
    initial_guesses.append(h_3seg2 * scale)
    
    # 4-segment step function: first 1/4, second 1/4, third 1/4, fourth 1/4
    h_4seg2 = np.zeros(n_points)
    h_4seg2[:n_points//4] = 1.0
    h_4seg2[n_points//4:2*n_points//4] = 1.0
    h_4seg2[2*n_points//4:3*n_points//4] = 1.0
    h_4seg2[3*n_points//4:] = 1.0
    sum_h = np.sum(h_4seg2)
    scale = target_sum / sum_h
    initial_guesses.append(h_4seg2 * scale)
    
    # Random initial guesses
    for _ in range(5):
        h_rand = np.random.uniform(0, 1, n_points)
        sum_h = np.sum(h_rand)
        scale = target_sum / sum_h
        initial_guesses.append(h_rand * scale)
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-12, 'maxiter': 2000}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Perturbation phase on the best_h found so far
    num_perturbations = 1000
    phase_shift = 0.25
    phase_decay = 0.99
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-12, 'maxiter': 300}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points

def compute_c5(h, dx):
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return np.max(corr) * dx
