from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate multiple initial guesses
    initial_guesses = [initial_guess.copy()]
    for _ in range(4):  # Generate 4 more diverse initial guesses
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        initial_guesses.append(h)
    
    # For each initial guess, run the optimization
    for guess in initial_guesses:
        # Initialize best for this guess
        current_best_c5 = float('inf')
        current_best_h = guess.copy()
        
        # Adaptive perturbation strategy with varying noise and exploration phases
        num_perturbations = 300  # Reduced for efficiency
        phase_shift = 0.15
        phase_decay = 0.96
        
        for perturbation_idx in range(num_perturbations):
            noise_scale = phase_shift * (phase_decay ** perturbation_idx)
            noise = np.random.uniform(-noise_scale, noise_scale, n_points)
            perturbed = guess + noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
            
            # Objective function
            def objective(h):
                one_minus_h = 1.0 - h
                corr = np.correlate(h, one_minus_h, mode='full')
                return np.max(corr) * dx
            
            # Optimization with SLSQP
            bounds = [(0.0, 1.0) for _ in range(n_points)]
            cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
            
            result = minimize(
                objective,
                perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'ftol': 1e-9, 'maxiter': 100}
            )
            
            if result.success and result.fun < current_best_c5:
                current_best_c5 = result.fun
                current_best_h = result.x
        
        # Update global best if this guess improves it
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = current_best_h
    
    return best_h, best_c5, n_points
