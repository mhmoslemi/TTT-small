    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    
    start_time = time.time()
    
    # Generate multiple initial guesses
    initial_guesses = [initial_h_values]
    
    # Structured guess with three intervals
    structured_guess = np.zeros(n_points)
    start1 = 0.0
    end1 = 1/3.0
    start2 = 5/6.0
    end2 = 7/6.0
    start3 = 5/3.0
    end3 = 2.0
    for i in range(n_points):
        x = i * dx
        if (start1 <= x <= end1) or (start2 <= x <= end2) or (start3 <= x <= end3):
            structured_guess[i] = 1.0
    initial_guesses.append(structured_guess)
    
    # Random binary guess
    binary_guess = np.random.permutation(np.concatenate([np.ones(50), np.zeros(50)]))
    initial_guesses.append(binary_guess)
    
    # Random uniform guess
    random_guess = np.random.uniform(0, 1, n_points)
    random_guess = random_guess * (target_sum / np.sum(random_guess))
    initial_guesses.append(random_guess)
    
    # Evaluate each initial guess
    for guess in initial_guesses:
        # Ensure the guess is valid
        guess = np.clip(guess, 0, 1)
        # Run optimization
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 2000}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Additional perturbations to refine the solution
    for _ in range(20):  # Reduced number of perturbations to save time
        if time.time() - start_time >= budget_s:
            break
        # Perturb the best_h with varying noise
        noise = np.random.uniform(-0.01, 0.01, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 1500}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    return best_h, best_c5, n_points
