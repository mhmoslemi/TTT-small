import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    
    start_time = time.time()
    
    # Generate diverse initial guesses
    initial_guesses = [initial_h_values]

    structured_guess = np.zeros(n_points)
    structured_guess[::3] = 1.0  # Every third point

    initial_guesses.append(structured_guess)
    
    structured_guess_2 = np.zeros(n_points)
    structured_guess_2[::2] = 1.0  # Every other point
    initial_guesses.append(structured_guess_2)
    
    structured_guess_3 = np.zeros(n_points)
    structured_guess_3[::4] = 1.0  # Every fourth point
    initial_guesses.append(structured_guess_3)
    
    # Random binary guess with spread-out 1s

    # Random uniform guess
    random_guess = np.random.uniform(0, 1, n_points)
    random_guess = random_guess * (target_sum / np.sum(random_guess))

    # Periodic guess with varying phase
    periodic_guess = np.zeros(n_points)

    # Evaluate each initial guess
    for guess in initial_guesses:
        # Ensure the guess is valid
        guess = np.clip(guess, 0, 1)

        # Run optimization
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]

        )
