            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points - 1)]
        result = differential_evolution(
            objective, bounds, 
            strategy='best1bin', 
            maxiter=100, 
            popsize=15, 
            tol=1e-6, 
            mutation=(0.5, 1), 
            recombination=0.8, 
            seed=seed
        )
        
        if result.success:
            h_opt = np.zeros(n_points)
            h_opt[:-1] = result.x
            h_opt[-1] = target_sum - np.sum(result.x)
            h_opt[-1] = np.clip(h_opt[-1], 0, 1)
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_opt
    
    # Additional perturbations with varied strategies
    for _ in range(100):  # More perturbations for thorough exploration
        if time.time() - start_time >= budget_s:
            break

        noise = np.random.uniform(-0.01, 0.01, n_points - 1)
        perturbed = best_h[:-1] + noise

        # Ensure sum is target_sum
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale

        last = target_sum - np.sum(perturbed)

        result = differential_evolution(
