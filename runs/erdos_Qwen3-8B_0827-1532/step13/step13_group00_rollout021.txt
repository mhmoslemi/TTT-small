            bounds,
            strategy=strategy,
            popsize=popsize,
            maxiter=max_generations,
            tol=1e-10,
            mutation=mutation,
            recombination=recombination,
            seed=seed
        )
        
        # Check if the result is successful
        if result.success:
            # Extract the optimal h and compute the actual C5
            current_h = result.x
            one_minus_h = 1.0 - current_h
            corr = np.correlate(current_h, one_minus_h, mode='full')
            current_c5 = np.max(corr) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
