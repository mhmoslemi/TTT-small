        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed
    )
    
    # Update best_h and best_c5
    best_h = result.x
    best_c5 = result.fun
    
    elapsed = time.time() - start_time
    if elapsed < budget_s:
        # If time allows, perform additional optimizations
        # For example, use SLSQP to refine the solution
        def objective_slsqp(h):
            h = h.copy()
            current_sum = np.sum(h)
            if current_sum == 0:
                return float('inf')
            h_scaled = h * (target_sum / current_sum)
            one_minus_h = 1.0 - h_scaled
            corr = np.correlate(h_scaled, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result_slsqp = minimize(
            objective_slsqp,
            best_h,
            method='SLSQP',
            bounds=bounds_slsqp,
            constraints=cons,
            options={'ftol': 1e-12, 'maxiter': 300}
        )
        
        if result_slsqp.success and result_slsqp.fun < best_c5:
            best_c5 = result_slsqp.fun
            best_h = result_slsqp.x
    
    return best_h, best_c5, n_points
