            periodic_guess[i] = 1.0
    periodic_guess = periodic_guess * (target_sum / np.sum(periodic_guess))
    initial_guesses.append(periodic_guess)
    
    # Binary step functions with different interval counts
    # Binary guess with 2 intervals
    binary_guess_2 = np.zeros(n_points)
    for i in range(n_points):
        x = i * dx
        if (0 <= x <= 0.5 or 1.0 <= x <= 1.5):
            binary_guess_2[i] = 1.0
    binary_guess_2 = binary_guess_2 * (target_sum / np.sum(binary_guess_2))
    initial_guesses.append(binary_guess_2)
    
    # Binary guess with 3 intervals
    binary_guess_3 = np.zeros(n_points)
    for i in range(n_points):
        x = i * dx
        if (0 <= x <= 1/3 or 0.8333 <= x <= 1.1666 or 1.6666 <= x <= 2.0):
            binary_guess_3[i] = 1.0
    binary_guess_3 = binary_guess_3 * (target_sum / np.sum(binary_guess_3))
    initial_guesses.append(binary_guess_3)
    
    # Binary guess with 4 intervals
    binary_guess_4 = np.zeros(n_points)
    for i in range(n_points):
        if (0 <= i*dx <= 0.25 or 0.5 <= i*dx <= 0.75 or 1.0 <= i*dx <= 1.25 or 1.5 <= i*dx <= 1.75):
            binary_guess_4[i] = 1.0
    binary_guess_4 = binary_guess_4 * (target_sum / np.sum(binary_guess_4))
    initial_guesses.append(binary_guess_4)
    
    # Binary guess with 8 intervals
    binary_guess_8 = np.zeros(n_points)
    for i in range(n_points):
        if (0 <= i*dx <= 0.25 or 0.5 <= i*dx <= 0.75 or 1.0 <= i*dx <= 1.25 or 1.5 <= i*dx <= 1.75 or 
            0.25 <= i*dx <= 0.5 or 0.75 <= i*dx <= 1.0 or 1.25 <= i*dx <= 1.5 or 1.75 <= i*dx <= 2.0):
            binary_guess_8[i] = 1.0
    binary_guess_8 = binary_guess_8 * (target_sum / np.sum(binary_guess_8))
    initial_guesses.append(binary_guess_8)
    
    # Binary guess with 16 intervals
    binary_guess_16 = np.zeros(n_points)
    for i in range(n_points):
        if (0 <= i*dx <= 0.125 or 0.25 <= i*dx <= 0.375 or 0.5 <= i*dx <= 0.625 or 0.75 <= i*dx <= 0.875 or 
            1.0 <= i*dx <= 1.125 or 1.25 <= i*dx <= 1.375 or 1.5 <= i*dx <= 1.625 or 1.75 <= i*dx <= 1.875 or 
            0.125 <= i*dx <= 0.25 or 0.375 <= i*dx <= 0.5 or 0.625 <= i*dx <= 0.75 or 0.875 <= i*dx <= 1.0 or 
            1.125 <= i*dx <= 1.25 or 1.375 <= i*dx <= 1.5 or 1.625 <= i*dx <= 1.75 or 1.875 <= i*dx <= 2.0):
            binary_guess_16[i] = 1.0
    binary_guess_16 = binary_guess_16 * (target_sum / np.sum(binary_guess_16))
    initial_guesses.append(binary_guess_16)
    
    # Evaluate each initial guess
    for guess in initial_guesses:
        guess = np.clip(guess, 0, 1)
        guess = guess * (target_sum / np.sum(guess))
        # Run optimization
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-12, 'maxiter': 2000}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Additional refinement with targeted perturbations
    for _ in range(30):
        if time.time() - start_time >= budget_s:
            break
        # Identify regions contributing to maximum overlap
        one_minus_h = 1.0 - best_h
        corr = np.correlate(best_h, one_minus_h, mode='full')
        max_corr_idx = np.argmax(corr)
        shift = max_corr_idx - (n_points - 1)
        
        # Adjust the function to reduce overlap at this shift
        # Perturb regions where the overlap is highest
        overlap_regions = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            x_shifted = x + shift
            if 0 <= x_shifted <= 2:
                overlap_regions[i] = best_h[i] * one_minus_h[np.clip(int(x_shifted / dx), 0, n_points - 1)]
        
        # Perturb the regions with highest overlap
        indices = np.argsort(overlap_regions)[::-1][:15]
        noise = np.zeros(n_points)
        noise[indices] = np.random.uniform(-0.1, 0.1, size=15)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-12, 'maxiter': 500}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    return best_h, best_c5, n_points
