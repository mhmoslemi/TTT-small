import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate diverse initial guesses: original, random, and structured
    initial_guesses = [initial_guess]
    for _ in range(3):  # 3 random initial guesses
        guess = np.random.uniform(0, 1, n_points)
        guess = guess * (target_sum / np.sum(guess))
        initial_guesses.append(guess)
    
    # Add a structured guess: alternating 0.5 and 0.5 with some variation
    structured_guess = np.zeros(n_points)
    for i in range(n_points):
        structured_guess[i] = 0.5 + 0.2 * np.sin(np.pi * i / n_points)
    structured_guess = structured_guess * (target_sum / np.sum(structured_guess))
    initial_guesses.append(structured_guess)
    
    # Perturbation parameters
    num_perturbations = 200  # Reduced for better exploration
    noise_scales = [0.3, 0.15, 0.075, 0.0375, 0.01875]  # Different noise scales for perturbations
    
    for perturbation_idx in range(num_perturbations):
        # Randomly select an initial guess or perturb the best_h
        if perturbation_idx < len(initial_guesses):
            perturbed = initial_guesses[perturbation_idx].copy()
        else:
            # Perturb best_h with varying noise
            noise_scale = noise_scales[perturbation_idx % len(noise_scales)]
            noise = np.random.uniform(-noise_scale, noise_scale, n_points)
            perturbed = best_h + noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
        
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='trust-constr',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 200}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
