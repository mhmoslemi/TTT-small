    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate diverse initial guesses: original, random, and structured
    initial_guesses = [initial_guess]
    for _ in range(2):  # 2 random initial guesses
        guess = np.random.uniform(0, 1, n_points)
        guess = guess * (target_sum / np.sum(guess))
        initial_guesses.append(guess)
    
    # Structured initial guess: alternating 1 and 0 with smooth transitions
    structured_guess = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 2:
            structured_guess[i] = 1.0
        else:
            structured_guess[i] = 0.0
    structured_guess = structured_guess * (target_sum / np.sum(structured_guess))
    initial_guesses.append(structured_guess)
    
    # Perturbation parameters
    num_perturbations = 150  # Reduced for better exploration
    noise_scales = [0.2, 0.1, 0.05, 0.02, 0.01]  # Different noise scales for perturbations
    
    for perturbation_idx in range(num_perturbations):
        # Randomly select an initial guess or perturb the best_h
        if perturbation_idx < len(initial_guesses):
            perturbed = initial_guesses[perturbation_idx].copy()
        else:
            # Perturb best_h with varying noise
            noise_scale = noise_scales[perturbation_idx % len(noise_scales)]
            noise = np.random.uniform(-noise_scale, noise_scale, n_points)
            perturbed = best_h + noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
        
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr) * dx
            # Penalty for sum constraint
            penalty = 1e4 * (np.sum(h) - target_sum)**2
            return max_corr + penalty
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Use differential evolution with adjusted parameters
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=15,
            mutation=0.5,
            recombination=0.8,
            tol=1e-12,
            maxiter=100,
            polish=True,
            vectorized=True
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
