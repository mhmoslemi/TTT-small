from scipy.optimize import minimize
import time

    candidates = [initial_guess, random_guess]
    
    # Evaluate initial candidates
    def evaluate_candidate(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    for h in candidates:
        current_c5 = evaluate_candidate(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    # Adaptive perturbation strategy with varying noise and exploration phases
    num_perturbations = 800  # Increased for better exploration
    phase_shift = 0.25  # Initial noise magnitude
    phase_decay = 0.98  # Decay factor for noise magnitude

        noise_scale = phase_shift * (phase_decay ** perturbation_idx)

        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h

            return np.max(corr) * dx

        bounds = [(0.0, 1.0) for _ in range(n_points)]
