import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Define the objective function with sum constraint enforcement
    def objective(h):
        h_scaled = np.clip(h, 0, 1)
        current_sum = np.sum(h_scaled)
        if current_sum == 0:
            return float('inf')
        scale = target_sum / current_sum
        h_scaled = h_scaled * scale
        one_minus_h = 1.0 - h_scaled
        corr = np.correlate(h_scaled, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    # Bounds for differential evolution
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Run differential evolution
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=20,
        mutation=(0.5, 1),
        recombination=0.8,
        tol=1e-8,
        maxiter=100,
        polish=True,
        seed=seed
    )
    
    # Extract the best solution from differential evolution
    best_h = result.x
    best_c5 = result.fun
    
    # Check remaining time for local optimization
    elapsed = time.time() - start_time
    if elapsed < budget_s:
        # Perform local optimization with SLSQP
        def objective_slsqp(h):
            h_scaled = np.clip(h, 0, 1)
            current_sum = np.sum(h_scaled)
            if current_sum == 0:
                return float('inf')
            scale = target_sum / current_sum
            h_scaled = h_scaled * scale
            one_minus_h = 1.0 - h_scaled
            corr = np.correlate(h_scaled, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result_slsqp = minimize(
            objective_slsqp,
            best_h,
            method='SLSQP',
            bounds=bounds_slsqp,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 300}
        )
        
        if result_slsqp.success and result_slsqp.fun < best_c5:
            best_c5 = result_slsqp.fun
            best_h = result_slsqp.x
    
    return best_h, best_c5, n_points
