import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):

    symmetric = np.zeros(n_points)
    for i in range(n_points):
        symmetric[i] = 1.0 if i < n_points//2 else 0.0
    initial_guesses.append(symmetric)
    
    # Periodic pattern
    periodic = np.zeros(n_points)
    for i in range(n_points):
        periodic[i] = 0.5 + 0.5 * np.sin(2 * np.pi * i / n_points)
    initial_guesses.append(periodic)
    
    # Binary patterns with varying intervals
    for interval_length in [0.25, 0.5, 0.75]:
        binary = np.zeros(n_points)
        for i in range(n_points):
            if i * dx < interval_length:
                binary[i] = 1.0
        initial_guesses.append(binary)
    
    # Three-interval pattern
    three_intervals = np.zeros(n_points)
    interval_length = 1.0 / 3.0
    spacing = interval_length
    start = 0.0
    for i in range(3):
        start_i = start + i * (interval_length + spacing)
        end_i = start_i + interval_length
        start_idx = int(start_i / dx)
        end_idx = int(end_i / dx)
        end_idx = min(end_idx, n_points - 1)
        three_intervals[start_idx:end_idx+1] = 1.0
    sum_three = np.sum(three_intervals)
    scale_factor = (n_points / 2) / sum_three
    three_intervals *= scale_factor
    initial_guesses.append(three_intervals)
    
    # Four-interval pattern
    four_intervals = np.zeros(n_points)
    interval_length = 0.5
    spacing = interval_length
    start = 0.0
    for i in range(4):
        start_i = start + i * (interval_length + spacing)
        end_i = start_i + interval_length
        start_idx = int(start_i / dx)
        end_idx = int(end_i / dx)
        end_idx = min(end_idx, n_points - 1)
        four_intervals[start_idx:end_idx+1] = 1.0
    sum_four = np.sum(four_intervals)
    scale_factor = (n_points / 2) / sum_four
    four_intervals *= scale_factor
    initial_guesses.append(four_intervals)
    
    # Random guesses
    for _ in range(3):
        guess = np.random.uniform(0, 1, n_points)
        guess = guess * (target_sum / np.sum(guess))
        initial_guesses.append(guess)
    
    # Optimization parameters
    max_iterations = 1000
    perturbation_count = 300
    noise_scales = [0.2, 0.1, 0.05, 0.02]
    
    for perturb_idx in range(perturbation_count):
        # Select a guess
        if perturb_idx < len(initial_guesses):
            perturbed = initial_guesses[perturb_idx].copy()
        else:
            # Generate a new perturbation based on best_h
            if perturb_idx % 4 == 0:
                # Random noise with larger scale
                noise = np.random.uniform(-0.2, 0.2, n_points)
            elif perturb_idx % 4 == 1:
                # Random noise with medium scale
                noise = np.random.uniform(-0.1, 0.1, n_points)
            elif perturb_idx % 4 == 2:
                # Random noise with small scale
                noise = np.random.uniform(-0.05, 0.05, n_points)
            else:

            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
