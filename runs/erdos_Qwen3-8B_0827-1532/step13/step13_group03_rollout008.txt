        rand_guess = rand_guess * scale
        guesses.append(rand_guess)
        # Periodic pattern
        periodic = np.zeros(n_points)
        for i in range(n_points):
            periodic[i] = 1 if (i % 3 == 0) else 0
        periodic = periodic * (target_sum / np.sum(periodic))
        guesses.append(periodic)
        # Spaced intervals
        spaced = np.zeros(n_points)
        for i in range(n_points):
            if (i % 5 == 0) or (i % 5 == 1):
                spaced[i] = 1
        spaced = spaced * (target_sum / np.sum(spaced))
        guesses.append(spaced)
        # Random binary pattern
        binary = np.random.choice([0, 1], size=n_points)
        binary_sum = np.sum(binary)
        binary = binary * (target_sum / binary_sum)
        guesses.append(binary)
        # Symmetric pattern
        symmetric = np.zeros(n_points)
        for i in range(n_points):
            if i < n_points // 2:
                symmetric[i] = 1
            elif i > n_points // 2:
                symmetric[i] = 1
        symmetric = symmetric * (target_sum / np.sum(symmetric))
        guesses.append(symmetric)
        return guesses
    
    initial_guesses = generate_initial_guesses()
    
    # Adaptive perturbation strategy with varying noise and exploration phases
    num_perturbations = 600  # Reduced for better exploration

    for guess in initial_guesses:
        # Clone and normalize
        current_h = guess.copy()
        current_sum = np.sum(current_h)
        current_h = current_h * (target_sum / current_sum)
        
        for perturbation_idx in range(num_perturbations):
            # Adaptive noise magnitude based on phase
            noise_scale = phase_shift * (phase_decay ** perturbation_idx)

            perturbed = current_h + noise

            scale = target_sum / current_sum
            perturbed = perturbed * scale
            
            # Objective function
            def objective(h):
                one_minus_h = 1.0 - h
                corr = np.correlate(h, one_minus_h, mode='full')
                return np.max(corr) * dx
