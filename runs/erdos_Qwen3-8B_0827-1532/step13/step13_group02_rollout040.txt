    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Constraints: sum(h) = target_sum
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    
    # Run optimization for each initial guess
    for guess in initial_guesses:
        # Use SLSQP optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 300}
        )
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    elapsed = time.time() - start_time
    if elapsed < budget_s:
        # If time allows, perform additional refinement
        # For example, use SLSQP on the best_h found
        # But this is optional
        pass
    
    return best_h, best_c5, n_points
