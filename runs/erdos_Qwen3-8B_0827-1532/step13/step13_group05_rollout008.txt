from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    initial_guesses = [initial_h_values]

    structured_guess = structured_guess * (target_sum / np.sum(structured_guess))
    initial_guesses.append(structured_guess)
    
    # Binary pattern with period 3

    initial_guesses.append(binary_guess)

    random_guess = np.random.uniform(0, 1, n_points)

        def objective(h):
