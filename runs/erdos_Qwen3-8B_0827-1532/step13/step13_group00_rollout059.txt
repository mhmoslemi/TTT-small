        scale = target_sum / current_sum
        binary = binary * scale
        binary_patterns.append(binary)
    initial_guesses.extend(binary_patterns)
    # Add two-interval binary pattern
    two_intervals = np.zeros(n_points)
    for i in range(n_points):
        if i * dx < 0.5:
            two_intervals[i] = 1.0
        elif i * dx < 1.0:
            two_intervals[i] = 0.0
        else:
            two_intervals[i] = 1.0
    current_sum = np.sum(two_intervals)
    scale = target_sum / current_sum
    two_intervals = two_intervals * scale
    initial_guesses.append(two_intervals)
    # Add random guesses
    for _ in range(3):
        guess = np.random.uniform(0, 1, n_points)
        guess = guess * (target_sum / np.sum(guess))
        initial_guesses.append(guess)
    
    # Optimization parameters
    max_iterations = 1000
    noise_scales = [0.2, 0.1, 0.05, 0.02]
    
    # Initial optimization on initial guesses
    for guess in initial_guesses:
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': max_iterations}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Perturbation loop
    while time.time() - start_time < budget_s:
        # Generate a new perturbed guess based on best_h
        # Use different perturbation strategies
        noise = np.zeros(n_points)
        if np.random.rand() < 0.5:
            # Random noise with varying scale
            scale = np.random.choice(noise_scales)
            noise = np.random.uniform(-scale, scale, n_points)
        else:
            # Swap values in random segments
            for _ in range(2):
                start = np.random.randint(0, n_points-5)
                end = start + 5
                noise[start:end] = np.random.uniform(-0.1, 0.1, end - start)
        
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': max_iterations}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    return best_h, best_c5, n_points
