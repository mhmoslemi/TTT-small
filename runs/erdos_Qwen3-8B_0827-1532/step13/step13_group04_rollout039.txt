        corr = np.correlate(best_h, one_minus_h, mode='full')
        max_corr_idx = np.argmax(corr)
        shift = max_corr_idx - (n_points - 1)
        
        # Compute overlap contributions for this shift
        overlap_contributions = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            x_shifted = x + shift
            if 0 <= x_shifted <= 2:
                j = int(x_shifted / dx)
                if 0 <= j < n_points:
                    overlap_contributions[i] = best_h[i] * (1.0 - best_h[j])
        
        # Find indices with highest contributions
        indices = np.argsort(overlap_contributions)[::-1][:10]
        
        # Decrease h at these indices
        perturbation = np.zeros(n_points)
        perturbation[indices] = -0.05
        perturbed = best_h + perturbation
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='COBYLA',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 500}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    return best_h, best_c5, n_points
