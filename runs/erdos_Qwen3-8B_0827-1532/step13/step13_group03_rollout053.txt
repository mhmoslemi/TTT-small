from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    # Generate diverse initial guesses
    def generate_initial_guesses():
        guesses = [initial_guess.copy()]

        rand_bin = np.random.choice([0, 1], size=n_points)
        rand_bin_sum = np.sum(rand_bin)
        rand_bin = rand_bin * (target_sum / rand_bin_sum)
        guesses.append(rand_bin)
        # More spaced intervals
        spaced2 = np.zeros(n_points)
        for i in range(n_points):
            if (i % 4 == 0) or (i % 4 == 1):
                spaced2[i] = 1
        spaced2 = spaced2 * (target_sum / np.sum(spaced2))
        guesses.append(spaced2)
        # Another periodic pattern
        periodic2 = np.zeros(n_points)
        for i in range(n_points):
            periodic2[i] = 1 if (i % 5 == 0) else 0
        periodic2 = periodic2 * (target_sum / np.sum(periodic2))
        guesses.append(periodic2)
        # Random perturbation
        rand_guess = initial_guess + np.random.uniform(-0.15, 0.15, n_points)
        rand_guess = np.clip(rand_guess, 0, 1)
        scale = target_sum / np.sum(rand_guess)
        rand_guess = rand_guess * scale
        guesses.append(rand_guess)
        return guesses
    
    initial_guesses = generate_initial_guesses()
    
    # Adaptive perturbation strategy with larger initial noise
    num_perturbations = 400  # Reduced for time efficiency
    phase_shift = 0.3  # Larger initial noise magnitude
    phase_decay = 0.95  # Slightly slower decay for more exploration
    
    for guess in initial_guesses:
        # Clone and normalize
        current_h = guess.copy()
        current_sum = np.sum(current_h)
        current_h = current_h * (target_sum / current_sum)
        
        for perturbation_idx in range(num_perturbations):
            # Adaptive noise magnitude based on phase
            noise_scale = phase_shift * (phase_decay ** perturbation_idx)
            
            # Generate perturbed guess
            noise = np.random.uniform(-noise_scale, noise_scale, n_points)
            perturbed = current_h + noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
            
            # Objective function
            def objective(h):
                one_minus_h = 1.0 - h
                corr = np.correlate(h, one_minus_h, mode='full')
                return np.max(corr) * dx
            
            # Optimization with SLSQP
            bounds = [(0.0, 1.0) for _ in range(n_points)]
            cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
            
            result = minimize(
                objective,
                perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'ftol': 1e-10, 'maxiter': 300}
            )
            
            if result.success and result.fun < best_c5:
                best_c5 = result.fun
                best_h = result.x
            
            elapsed = time.time() - start_time
            if elapsed >= budget_s:
                break
    
    return best_h, best_c5, n_points
