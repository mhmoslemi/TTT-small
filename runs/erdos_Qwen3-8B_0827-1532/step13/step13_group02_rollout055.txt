from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    # Adaptive perturbation with diverse exploration and refinement
    num_perturbations = 400  # Reduced for efficiency
    phase_shift = 0.15
    phase_decay = 0.98
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        
        # Generate diverse perturbations: random and structured
        if perturbation_idx % 2 == 0:
            # Random noise
            noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        else:
            # Structured noise: alternate between high and low
            noise = np.zeros(n_points)
            noise[::2] = noise_scale  # Even indices
            noise[1::2] = -noise_scale  # Odd indices
        
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with COBYLA for constraint handling
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='COBYLA',
            bounds=bounds,
            constraints=cons,
            options={'disp': False, 'maxiter': 200}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
