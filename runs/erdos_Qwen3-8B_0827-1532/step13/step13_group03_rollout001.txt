        # Periodic pattern
        periodic = np.zeros(n_points)

        periodic = periodic * (target_sum / np.sum(periodic))
        guesses.append(periodic)
        # Spaced intervals
        spaced = np.zeros(n_points)
        for i in range(n_points):
            if (i % 5 == 0) or (i % 5 == 1):

        guesses.append(spaced)
        # Random initial guess
        random_guess = np.random.uniform(0, 1, n_points)

        guesses.append(random_guess)
        return guesses

    def objective(h):

        penalty = 1e4 * abs(np.sum(h) - target_sum)
        return c5 + penalty
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Adaptive optimization with differential evolution
    for guess in initial_guesses:

        current_sum = np.sum(current_h)

        # Differential evolution optimization
        result = minimize(
