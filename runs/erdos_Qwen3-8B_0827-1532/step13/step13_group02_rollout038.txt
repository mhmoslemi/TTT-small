from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate diverse initial guesses with structured perturbations
    num_initial_guesses = 5
    initial_guesses = []
    for _ in range(num_initial_guesses):
        perturbed = initial_guess + np.random.uniform(-0.1, 0.1, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        initial_guesses.append(perturbed)
    
    # Optimization parameters
    max_iterations = 1000
    phase_shift = 0.2
    phase_decay = 0.98
    noise_scale = phase_shift
    
    for iter in range(max_iterations):
        # Generate a new perturbed guess
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with L-BFGS-B
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,

        )

        # Decay noise scale
