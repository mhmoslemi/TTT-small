        corr = np.correlate(de_solution, one_minus_h, mode='full')
        c5 = np.max(corr) * dx
        
        # Update the best solution if this is better
        if c5 < best_c5:
            best_c5 = c5
            best_h = de_solution
    
    # After all initial guesses, perform SLSQP on the best solution
    if best_h is not None:
        # Define the objective function for SLSQP
        def objective_slsqp(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Bounds and constraints
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Run SLSQP optimization
        result_slsqp = minimize(
            objective_slsqp,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 500}
        )
        
        # Update best_h and best_c5 if successful
        if result_slsqp.success:
            best_h = result_slsqp.x
            one_minus_h = 1.0 - best_h
            corr = np.correlate(best_h, one_minus_h, mode='full')
            best_c5 = np.max(corr) * dx
    
    # Check elapsed time and return the best solution
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
