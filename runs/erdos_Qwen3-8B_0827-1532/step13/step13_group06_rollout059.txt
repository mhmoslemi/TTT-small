import numpy as np
from scipy.optimize import minimize
import time
from multiprocessing import Pool, cpu_count

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate diverse initial guesses: original, random, and structured
    initial_guesses = [initial_guess]
    for _ in range(3):  # 3 random initial guesses
        guess = np.random.uniform(0, 1, n_points)
        guess = guess * (target_sum / np.sum(guess))
        initial_guesses.append(guess)
    
    # Add structured initial guess: two blocks of 1.0
    structured_guess = np.zeros(n_points)
    block_size = 0.5
    structured_guess[0:int(block_size / dx)] = 1.0
    structured_guess[int((1.0 - block_size) / dx):int((1.0 - block_size) / dx + block_size / dx)] = 1.0
    initial_guesses.append(structured_guess)
    
    # Perturbation parameters
    num_perturbations = 500  # Increased for better exploration
    noise_scales = [0.3, 0.2, 0.1, 0.05, 0.02, 0.01]  # Different noise scales for perturbations
    
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    
    # Function to run optimization for a single guess
    def optimize_guess(guess):
        perturbed = guess.copy()
        noise_scale = noise_scales[np.random.randint(len(noise_scales))]
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = perturbed + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-12, 'maxiter': 500}
        )
        
        if result.success:
            return result.fun, result.x
        else:
            return best_c5, best_h  # Return current best if optimization fails
    
    # Use multiprocessing to run optimizations in parallel
    pool = Pool(processes=min(cpu_count(), 10))  # Limit to 10 CPUs
    results = []
    
    # Submit initial guesses to the pool
    for guess in initial_guesses:
        results.append(pool.apply_async(optimize_guess, (guess,)))
    
    # Submit perturbations
    for _ in range(num_perturbations):
        if time.time() - start_time >= budget_s:
            break
        results.append(pool.apply_async(optimize_guess, (best_h,)))
    
    # Collect results
    for res in results:
        c5, h = res.get()
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    
    pool.close()
    pool.join()
    
    return best_h, best_c5, n_points
