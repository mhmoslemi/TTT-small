                    noise[start:end] = np.random.uniform(-0.1, 0.1, end - start)
            perturbed = best_h + noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
        
        # Define objective function with penalty
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            penalty = 1e4 * (np.sum(h) - target_sum)**2
            return np.max(corr) * dx + penalty
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Use differential evolution
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=15,
            mutation=(0.5, 1),
            recombination=0.9,
            tol=1e-6,
            maxiter=100,
            seed=seed
        )
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
