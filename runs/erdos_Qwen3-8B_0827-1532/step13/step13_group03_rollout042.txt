            bounds = [(0.0, 1.0) for _ in range(n_points)]
            result = minimize(
                objective,
                perturbed,
                method='COBYLA',
                bounds=bounds,
                options={'ftol': 1e-10, 'maxiter': 300}
            )
            
            if result.success and result.fun < best_c5:
                best_c5 = result.fun
                best_h = result.x
            
            elapsed = time.time() - start_time
            if elapsed >= budget_s:
                break
    
    return best_h, best_c5, n_points
