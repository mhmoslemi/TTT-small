            popsize=15,
            maxiter=100,
            strategy='best1bin',
            tol=1e-10,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed
        )
        # Check if the result is better
        if result.success:
            current_h = result.x
            current_c5 = objective(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
    
    # Refine the best solution with SLSQP
    if best_c5 != float('inf'):
        # Refine with SLSQP
        def objective_slsqp(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result_slsqp = minimize(
            objective_slsqp,
            best_h,
            method='SLSQP',
            bounds=bounds_slsqp,
            constraints=cons,
            options={'ftol': 1e-12, 'maxiter': 1500}
        )
        
        if result_slsqp.success:
            best_h = result_slsqp.x
            best_c5 = result_slsqp.fun
    
    # Additional refinement with targeted perturbations
    for _ in range(30):
        if time.time() - start_time >= budget_s:
            break
        # Identify regions contributing to maximum overlap
        one_minus_h = 1.0 - best_h
        corr = np.correlate(best_h, one_minus_h, mode='full')
        max_corr_idx = np.argmax(corr)
        shift = max_corr_idx - (n_points - 1)
        
        # Adjust the function to reduce overlap at this shift
        # Perturb regions where the overlap is highest
        overlap_regions = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            x_shifted = x + shift
            if 0 <= x_shifted <= 2:
                overlap_regions[i] = best_h[i] * one_minus_h[np.clip(int(x_shifted / dx), 0, n_points - 1)]
        
        # Perturb the regions with highest overlap
        indices = np.argsort(overlap_regions)[::-1][:10]
        noise = np.zeros(n_points)
        noise[indices] = np.random.uniform(-0.1, 0.1, size=10)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        def objective_slsqp_perturb(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds_perturb = [(0.0, 1.0) for _ in range(n_points)]
        cons_perturb = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result_perturb = minimize(
            objective_slsqp_perturb,
            perturbed,
            method='SLSQP',
            bounds=bounds_perturb,
            constraints=cons_perturb,
            options={'ftol': 1e-12, 'maxiter': 500}
        )
        
        if result_perturb.success and result_perturb.fun < best_c5:
            best_c5 = result_perturb.fun
            best_h = result_perturb.x
    
    return best_h, best_c5, n_points
