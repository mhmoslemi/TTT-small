from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    target_sum = n_points / 2

    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate a few diverse initial guesses
    def generate_initial_guesses():
        # Original initial guess
        guesses = [initial_guess.copy()]
        # Random initial guess
        rand_guess = np.random.uniform(0, 1, n_points)
        rand_guess = rand_guess * (target_sum / np.sum(rand_guess))
        guesses.append(rand_guess)
        # Structured guess: alternating 0.5 and 0.5 with some variation
        structured_guess = np.zeros(n_points)
        for i in range(n_points):
            structured_guess[i] = 0.5 + 0.2 * np.sin(np.pi * i / n_points)
        structured_guess = structured_guess * (target_sum / np.sum(structured_guess))
        guesses.append(structured_guess)
        return guesses
    
    initial_guesses = generate_initial_guesses()
    
    # Adaptive perturbation strategy with varying noise and exploration phases
    num_perturbations = 400  # Reduced for better exploration
    phase_shift = 0.25  # Larger initial noise magnitude
    phase_decay = 0.98  # Faster decay for noise magnitude
    
    for perturbation_idx in range(num_perturbations):
        # Adaptive noise magnitude based on phase
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        
        # Use different initial guess for each perturbation
        current_guess = initial_guesses[perturbation_idx % len(initial_guesses)].copy()
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = current_guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with COBYLA
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='COBYLA',
            bounds=bounds,
            constraints=cons,

        )
        
        if result.success and result.fun < best_c5:

            best_h = result.x
