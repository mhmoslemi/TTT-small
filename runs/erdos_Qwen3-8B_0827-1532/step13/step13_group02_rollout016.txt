import time

    dx = 2.0 / n_points

    random_guess = random_guess / np.sum(random_guess) * target_sum
    
    # List of initial guesses

    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Adaptive perturbation strategy with varying noise and exploration phases
    num_perturbations = 800  # Increased for better exploration
    phase_shift = 0.15  # Initial noise magnitude
    phase_decay = 0.95  # Slower decay for more exploration
    
    for initial_guess in initial_guesses:
        current_h = initial_guess.copy()
        current_h = np.clip(current_h, 0, 1)
        current_sum = np.sum(current_h)
        scale = target_sum / current_sum
        current_h = current_h * scale
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Perform optimization
        result = minimize(
            objective,
            current_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 300}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
