    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    
    start_time = time.time()
    
    # Generate diverse initial guesses
    initial_guesses = [initial_h_values]
    
    # Structured binary guess with varying spacing
    structured_guess = np.zeros(n_points)
    structured_guess[::3] = 1.0  # Every third point
    structured_guess[1::4] = 1.0  # Additional spaced points
    initial_guesses.append(structured_guess)
    
    # Random binary guess with spread-out 1s
    binary_guess = np.random.permutation(np.concatenate([np.ones(50), np.zeros(50)]))
    initial_guesses.append(binary_guess)
    
    # Random uniform guess
    random_guess = np.random.uniform(0, 1, n_points)
    random_guess = random_guess * (target_sum / np.sum(random_guess))
    initial_guesses.append(random_guess)
    
    # Periodic guess with varying phase
    periodic_guess = np.zeros(n_points)
    for i in range(n_points):
        if i % 2 == 0:
            periodic_guess[i] = 1.0
    periodic_guess = periodic_guess * (target_sum / np.sum(periodic_guess))
    initial_guesses.append(periodic_guess)
    
    # Greedy placement initial guess
    greedy_positions = greedy_placement(n_points, int(target_sum))
    greedy_guess = np.zeros(n_points)
    for pos in greedy_positions:
        greedy_guess[pos] = 1.0
    greedy_guess = greedy_guess * (target_sum / np.sum(greedy_guess))
    initial_guesses.append(greedy_guess)
    
    # Evaluate each initial guess
    for guess in initial_guesses:
        # Ensure the guess is valid
        guess = np.clip(guess, 0, 1)
        guess = guess * (target_sum / np.sum(guess))
        # Run optimization
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 2000}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Additional perturbations with varied strategies
    for _ in range(150):  # More perturbations for thorough exploration
        if time.time() - start_time >= budget_s:
            break
        # Perturb the best_h with systematic noise
        noise = np.random.uniform(-0.01, 0.01, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 1500}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    return best_h, best_c5, n_points
