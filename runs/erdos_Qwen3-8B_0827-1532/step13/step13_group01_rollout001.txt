        initial_guesses.append(guess)
    
    # Optimization parameters
    max_iterations = 100  # Adjust based on time budget
    
    for guess in initial_guesses:
        # Define objective function with penalty for sum constraint
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            c5 = np.max(corr) * dx
            penalty = 1e4 * (np.sum(h) - target_sum)**2
            return c5 + penalty
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Run differential evolution
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=15,
            maxiter=max_iterations,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.9,
            disp=False
        )
        
        # Check if the result is successful
        if result.success:
            current_c5 = result.fun - 1e4 * (np.sum(result.x) - target_sum)**2  # Subtract penalty
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
