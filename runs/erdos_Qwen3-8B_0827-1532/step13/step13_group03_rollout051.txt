import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate diverse initial guesses
    def generate_initial_guesses():
        guesses = [initial_guess.copy()]
        # Random uniform
        rand_guess = np.random.uniform(0, 1, n_points)
        rand_guess = rand_guess * (target_sum / np.sum(rand_guess))
        guesses.append(rand_guess)
        # Periodic
        periodic = np.zeros(n_points)
        for i in range(n_points):
            periodic[i] = 1 if (i % 3 == 0) else 0
        periodic = periodic * (target_sum / np.sum(periodic))
        guesses.append(periodic)
        # Spaced
        spaced = np.zeros(n_points)
        for i in range(n_points):
            if (i % 5 == 0) or (i % 5 == 1):
                spaced[i] = 1
        spaced = spaced * (target_sum / np.sum(spaced))
        guesses.append(spaced)
        # Sine wave
        sine = np.sin(np.linspace(0, np.pi, n_points))
        sine = (sine + 1) / 2
        sine = sine * (target_sum / np.sum(sine))
        guesses.append(sine)
        # Alternating
        alt = np.zeros(n_points)
        for i in range(n_points):
            alt[i] = 1 if (i % 2 == 0) else 0
        alt = alt * (target_sum / np.sum(alt))
        guesses.append(alt)
        return guesses
    
    initial_guesses = generate_initial_guesses()
    
    # Adaptive optimization with multiple methods
    num_perturbations = 400
    phase_shift = 0.2
    phase_decay = 0.95
    
    for guess in initial_guesses:
        current_h = guess.copy()
        current_sum = np.sum(current_h)
        current_h = current_h * (target_sum / current_sum)
        
        for perturbation_idx in range(num_perturbations):
            noise_scale = phase_shift * (phase_decay ** perturbation_idx)
            noise = np.random.uniform(-noise_scale, noise_scale, n_points)
            perturbed = current_h + noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
            
            # Objective function
            def objective(h):
                one_minus_h = 1.0 - h
                corr = np.correlate(h, one_minus_h, mode='full')
                return np.max(corr) * dx
            
            # Optimization with COBYLA
            bounds = [(0.0, 1.0) for _ in range(n_points)]
            cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
            
            result = minimize(
                objective,
                perturbed,
                method='COBYLA',
                bounds=bounds,
                constraints=cons,
                options={'ftol': 1e-10, 'maxiter': 300}
            )
            
            if result.success and result.fun < best_c5:
                best_c5 = result.fun
                best_h = result.x
            
            elapsed = time.time() - start_time
            if elapsed >= budget_s:
                break
    
    return best_h, best_c5, n_points
