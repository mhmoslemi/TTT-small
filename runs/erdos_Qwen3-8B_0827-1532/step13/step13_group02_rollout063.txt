import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Adaptive perturbation strategy with varying noise and exploration phases
    num_perturbations = 400  # Increased for better exploration
    phase_shift = 0.25  # Initial noise magnitude
    phase_decay = 0.98  # Decay factor for noise magnitude
    exploration_phase = 200  # Number of perturbations in exploration phase
    
    for perturbation_idx in range(num_perturbations):
        # Adaptive noise magnitude based on phase
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        
        # Generate diverse initial guesses by adding different types of noise
        if perturbation_idx < exploration_phase:
            # Initial exploration with larger noise and random shifts
            noise = np.random.uniform(-noise_scale, noise_scale, n_points)
            # Add a random shift to the initial guess
            shifted_guess = np.roll(initial_guess, np.random.randint(0, n_points))
            perturbed = shifted_guess + noise
        else:
            # Refinement with smaller noise
            noise = np.random.uniform(-noise_scale, noise_scale, n_points)
            perturbed = best_h + noise
        
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP with tighter tolerances
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 300, 'disp': False}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
