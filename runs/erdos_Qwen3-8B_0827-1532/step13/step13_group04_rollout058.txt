            0.125 <= i*dx <= 0.25 or 0.375 <= i*dx <= 0.5 or 0.625 <= i*dx <= 0.75 or 0.875 <= i*dx <= 1.0 or 
            1.125 <= i*dx <= 1.25 or 1.375 <= i*dx <= 1.5 or 1.625 <= i*dx <= 1.75 or 1.875 <= i*dx <= 2.0):
            binary_guess_16[i] = 1.0
    initial_guesses.append(binary_guess_16)
    
    # Random binary initial guess
    binary_guess = np.zeros(n_points)
    indices = np.random.choice(n_points, size=int(target_sum), replace=False)
    binary_guess[indices] = 1.0
    initial_guesses.append(binary_guess)
    
    # Evaluate each initial guess
    for guess in initial_guesses:
        guess = np.clip(guess, 0, 1)
        guess = guess * (target_sum / np.sum(guess))
        # Run optimization
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-12, 'maxiter': 1500}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Additional refinement with targeted perturbations
    for _ in range(30):
        if time.time() - start_time >= budget_s:
            break
        # Identify regions contributing to maximum overlap
        one_minus_h = 1.0 - best_h
        corr = np.correlate(best_h, one_minus_h, mode='full')
        max_corr_idx = np.argmax(corr)
        shift = max_corr_idx - (n_points - 1)
        
        # Compute the overlap contribution for each sample point at this shift
        overlap_contributions = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            x_shifted = x + shift
            if 0 <= x_shifted <= 2:
                j = np.clip(int(x_shifted / dx), 0, n_points - 1)
                overlap_contributions[i] = best_h[i] * one_minus_h[j]
            else:
                overlap_contributions[i] = 0
        
        # Adjust the function to reduce overlap at this shift
        perturbation = np.zeros(n_points)
        for i in range(n_points):
            if overlap_contributions[i] > 0:
                if best_h[i] > 0.5:
                    perturbation[i] = -0.05
                elif best_h[i] < 0.5:
                    perturbation[i] = 0.05
        perturbed = best_h + perturbation
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-12, 'maxiter': 500}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    return best_h, best_c5, n_points
