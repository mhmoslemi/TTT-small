import numpy as np
from scipy.optimize import minimize
import time
from concurrent.futures import ProcessPoolExecutor, as_completed

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate diverse initial guesses: original, random, and structured
    initial_guesses = [initial_guess]
    for _ in range(4):  # 4 random initial guesses
        guess = np.random.uniform(0, 1, n_points)
        guess = guess * (target_sum / np.sum(guess))
        initial_guesses.append(guess)
    
    # Add a structured guess: alternating high and low values
    structured_guess = np.zeros(n_points)
    structured_guess[::2] = 1.0
    structured_guess = structured_guess * (target_sum / np.sum(structured_guess))
    initial_guesses.append(structured_guess)
    
    # Perturbation parameters
    num_perturbations = 500  # Increased for better exploration
    noise_scales = [0.3, 0.2, 0.1, 0.05, 0.02, 0.01]  # More varied noise scales
    
    # Function to evaluate a single perturbation
    def evaluate_perturbation(perturbed):
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-12, 'maxiter': 500}
        )
        
        if result.success:
            return result.fun, result.x
        return float('inf'), perturbed
    
    # Run perturbations in parallel
    with ProcessPoolExecutor(max_workers=10) as executor:
        futures = []
        for perturbation_idx in range(num_perturbations):
            if perturbation_idx < len(initial_guesses):
                perturbed = initial_guesses[perturbation_idx].copy()
            else:
                # Perturb best_h with varying noise
                noise_scale = noise_scales[perturbation_idx % len(noise_scales)]
                noise = np.random.uniform(-noise_scale, noise_scale, n_points)
                perturbed = best_h + noise
                perturbed = np.clip(perturbed, 0, 1)
                current_sum = np.sum(perturbed)
                scale = target_sum / current_sum
                perturbed = perturbed * scale
            
            futures.append(executor.submit(evaluate_perturbation, perturbed))
        
        for future in as_completed(futures):
            c5, h = future.result()
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
        
        # Check elapsed time and exit if budget is exceeded
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
