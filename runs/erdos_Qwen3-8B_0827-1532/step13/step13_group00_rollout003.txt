import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Define the objective function
    def objective(h):
        # Normalize to ensure the sum constraint
        h_normalized = h * (target_sum / np.sum(h))
        # Compute cross-correlation
        corr = np.correlate(h_normalized, 1 - h_normalized, mode='full')
        return np.max(corr) * dx
    
    # Bounds for each variable
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Run differential evolution with a population-based approach
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        polish=True,
        disp=False
    )
    
    # Update best solution
    best_h = result.x
    best_c5 = result.fun
    
    # Check elapsed time
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
