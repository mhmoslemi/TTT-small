    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Reduce n_points for faster computation
    n_points = 50
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate diverse initial guesses
    initial_guesses = []
    for _ in range(5):  # 5 initial guesses
        guess = np.random.uniform(0, 1, n_points)
        guess = guess * (target_sum / np.sum(guess))
        initial_guesses.append(guess)
    
    # Define objective with parameterization to enforce sum constraint
    def objective(h_params):
        h = np.array(h_params)
        h_full = np.append(h, target_sum - np.sum(h))
        h_full = np.clip(h_full, 0, 1)
        one_minus_h = 1.0 - h_full
        corr = np.correlate(h_full, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    # Bounds for parameters (n_points-1 variables)
    bounds = [(0.0, 1.0) for _ in range(n_points - 1)]
    
    # Differential evolution for global search
    result_de = differential_evolution(objective, bounds, strategy='best1bin', maxiter=100, popsize=15, tol=1e-6, mutation=(0.5, 1), recombination=0.8, seed=seed)
    
    # Refine with SLSQP
    best_h_de = result_de.x
    h_full_de = np.append(best_h_de, target_sum - np.sum(best_h_de))
    h_full_de = np.clip(h_full_de, 0, 1)
    
    def objective_slsqp(h_full):
        one_minus_h = 1.0 - h_full
        corr = np.correlate(h_full, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]
    cons_slsqp = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    
    result_slsqp = minimize(objective_slsqp, h_full_de, method='SLSQP', bounds=bounds_slsqp, constraints=cons_slsqp, options={'ftol': 1e-12, 'maxiter': 300})
    
    best_c5 = result_slsqp.fun
    best_h = result_slsqp.x
    
    # Check time and proceed with perturbations
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return best_h, best_c5, n_points
    
    # Perturb and optimize
    noise_scale = 0.1
    noise = np.random.uniform(-noise_scale, noise_scale, n_points)
    perturbed = best_h + noise
    perturbed = np.clip(perturbed, 0, 1)
    current_sum = np.sum(perturbed)
    scale = target_sum / current_sum
    perturbed = perturbed * scale
    
    result_perturb = minimize(objective_slsqp, perturbed, method='SLSQP', bounds=bounds_slsqp, constraints=cons_slsqp, options={'ftol': 1e-12, 'maxiter': 300})
    
    if result_perturb.fun < best_c5:
        best_c5 = result_perturb.fun
        best_h = result_perturb.x
    
    return best_h, best_c5, n_points
