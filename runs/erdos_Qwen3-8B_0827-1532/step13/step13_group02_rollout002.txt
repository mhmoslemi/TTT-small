            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 300}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Now perform perturbations from the best_h
    # Adaptive perturbation strategy with varying noise and exploration phases
    num_perturbations = 300  # Reduced for time
    phase_shift = 0.15  # Initial noise magnitude
    phase_decay = 0.96  # Decay factor for noise magnitude
    
    for perturbation_idx in range(num_perturbations):
        # Adaptive noise magnitude based on phase
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        
        # Generate perturbed guess
        perturbed = best_h + np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 300}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
