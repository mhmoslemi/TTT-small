        guess = guess * (target_sum / np.sum(guess))
        initial_guesses.append(guess)
    
    # Evaluate each initial guess
    for guess in initial_guesses:
        guess = np.clip(guess, 0, 1)
        guess = guess * (target_sum / np.sum(guess))
        # Run optimization
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-12, 'maxiter': 1500}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Additional refinement with targeted perturbations
    for _ in range(30):
        if time.time() - start_time >= budget_s:
            break
        # Identify regions contributing to maximum overlap
        one_minus_h = 1.0 - best_h
        corr = np.correlate(best_h, one_minus_h, mode='full')
        max_corr_idx = np.argmax(corr)
        shift = max_corr_idx - (n_points - 1)
        
        # Adjust the function to reduce overlap at this shift
        # Perturb regions where the overlap is highest
        overlap_regions = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            x_shifted = x + shift
            if 0 <= x_shifted <= 2:
                overlap_regions[i] = best_h[i] * one_minus_h[np.clip(int(x_shifted / dx), 0, n_points - 1)]
        
        # Perturb the regions with highest overlap
        indices = np.argsort(overlap_regions)[::-1][:10]
        noise = np.zeros(n_points)
        noise[indices] = np.random.uniform(-0.1, 0.1, size=10)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-12, 'maxiter': 500}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    return best_h, best_c5, n_points
