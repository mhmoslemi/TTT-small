import numpy as np
from scipy.optimize import basinhopping, minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Define objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    # Define constraints
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Setup minimizer_kwargs for basinhopping
    minimizer_kwargs = {
        'method': 'SLSQP',
        'bounds': bounds,
        'constraints': cons,
        'options': {'ftol': 1e-9, 'maxiter': 300}
    }
    
    # Perform basinhopping optimization
    result = basinhopping(
        objective,
        initial_guess,
        niter=100,  # Number of basin-hopping iterations
        T=1.0,      # Initial temperature
        stepsize=0.5,  # Step size for perturbations
        minimizer_kwargs=minimizer_kwargs,
        # Aggressive settings to explore more
        niter_success=20,  # Number of successful iterations to reduce temperature
        disp=False
    )
    
    # Update best solution
    best_h = result.x
    best_c5 = result.fun
    
    # Ensure we respect the time budget
    elapsed = time.time() - start_time
    if elapsed < budget_s:
        # If time remains, perform a final local optimization
        result_local = minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 300}
        )
        if result_local.success and result_local.fun < best_c5:
            best_c5 = result_local.fun
            best_h = result_local.x
    
    return best_h, best_c5, n_points
