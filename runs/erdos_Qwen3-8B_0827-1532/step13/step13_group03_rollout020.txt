from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2

        # Original initial guess
        guesses = [initial_guess.copy()]

        rand_guess = np.clip(rand_guess, 0, 1)
        scale = target_sum / np.sum(rand_guess)
        rand_guess = rand_guess * scale
        guesses.append(rand_guess)
        # Periodic pattern
        periodic = np.zeros(n_points)
        for i in range(n_points):
            periodic[i] = 1 if (i % 3 == 0) else 0
        periodic = periodic * (target_sum / np.sum(periodic))
        guesses.append(periodic)
        # Spaced intervals
        spaced = np.zeros(n_points)
        for i in range(n_points):
            if (i % 5 == 0) or (i % 5 == 1):
                spaced[i] = 1
        spaced = spaced * (target_sum / np.sum(spaced))
        guesses.append(spaced)

        two_intervals = np.zeros(n_points)

            if 0 <= i < n_points//4 or (n_points//2) <= i < (3*n_points)//4:

        two_intervals = two_intervals * (target_sum / np.sum(two_intervals))
        guesses.append(two_intervals)
        return guesses

    for guess in initial_guesses:
        # Clone and normalize
        current_h = guess.copy()

        current_h = current_h * (target_sum / current_sum)
        
        for perturbation_idx in range(num_perturbations):
            # Adaptive noise magnitude based on phase

            def objective(h):
