        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    # Define constraints
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)

    # Perform optimization
    result = minimize(
        objective,
        h_initial,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        tol=1e-6,
        options={'maxiter': 1000}
    )

    # Check if optimization was successful
    if result.success:
        best_h = result.x
        best_c5 = result.fun
    else:
        # Fallback to initial solution if optimization fails
        best_h = h_initial
        best_c5 = objective(h_initial)

    # Return the best solution found
    return (best_h, best_c5, n_points)
