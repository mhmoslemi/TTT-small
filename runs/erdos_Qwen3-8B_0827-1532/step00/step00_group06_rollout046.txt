    n_points = 95
    h_initial = initial_h_values.copy()
    dx = 2.0 / n_points
    
    # Define the objective function to minimize the maximum overlap
    def objective(h):
        conv = correlate(h, 1 - h, mode='full')
        max_val = np.max(conv) * dx
        return max_val
    
    # Constraints: sum(h) = n_points / 2
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
    
    # Bounds for each h[i] in [0, 1]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial guess
    initial_guess = h_initial.copy()
    
    # Optimization settings
    result = minimize(
        objective,
        initial_guess,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    
    if result.success:
        h_values = result.x
        c5_bound = result.fun
    else:
        h_values = initial_guess
        c5_bound = objective(h_values)
    
    return h_values, c5_bound, n_points
