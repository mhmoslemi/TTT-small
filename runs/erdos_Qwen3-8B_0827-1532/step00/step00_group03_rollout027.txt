import numpy as np
import random
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Parameters
    n_points = 50
    dx = 2.0 / n_points
    required_sum = n_points / 2
    
    # Initial population
    population_size = 50
    population = []
    for _ in range(population_size):
        h = np.random.uniform(0, 1, n_points)
        h = h * required_sum / np.sum(h)
        population.append(h)
    
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    
    # Genetic algorithm
    generations = 100
    for gen in range(generations):
        # Evaluate population
        evaluations = []
        for h in population:
            conv = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(conv) * dx
            evaluations.append((c5, h))
        
        # Sort by c5
        evaluations.sort()
        best_c5_current = evaluations[0][0]
        if best_c5_current < best_c5:
            best_c5 = best_c5_current
            best_h = evaluations[0][1].copy()
        
        # Check time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        
        # Select parents
        parents = [evals[1][1] for evals in evaluations[:int(len(evaluations)*0.2)]]
        
        # Generate new population
        new_population = parents[:]
        while len(new_population) < population_size:
            # Crossover
            p1, p2 = random.sample(parents, 2)
            child = np.zeros(n_points)
            for i in range(n_points):
                if random.random() < 0.5:
                    child[i] = p1[i]
                else:
                    child[i] = p2[i]
            
            # Mutation
            for i in range(n_points):
                if random.random() < 0.1:
                    delta = np.random.uniform(-0.1, 0.1)
                    child[i] += delta
                    child[i] = np.clip(child[i], 0, 1)
            
            # Normalize
            child_sum = np.sum(child)
            if child_sum == 0:
                child = np.random.uniform(0, 1, n_points)
                child = child * required_sum / np.sum(child)
            else:
                child = child * required_sum / child_sum
            
            new_population.append(child)
        
        population = new_population
    
    return (best_h, best_c5, n_points)
