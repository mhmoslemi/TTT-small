import time

    h = initial_h_values.copy()

    def objective(x):
        # x is the h_values array
        corr = np.correlate(x, 1 - x, mode='full')
        max_corr = np.max(corr) * (2.0 / n_points)
        return max_corr
    
    # Define constraints
    bounds = Bounds(0.0, 1.0)
    constraint = LinearConstraint(np.ones(n_points), lb=n_points / 2, ub=n_points / 2)
    
    # Perform optimization
    start_time = time.time()
    result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
    
    # Check if we have enough time to run more iterations
    elapsed = time.time() - start_time
    if elapsed < budget_s:
        # Try with more iterations
        result = minimize(objective, result.x, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 200})
    
    # Return the best solution
    best_h = result.x
    best_c5 = result.fun
    return (best_h, best_c5, n_points)
