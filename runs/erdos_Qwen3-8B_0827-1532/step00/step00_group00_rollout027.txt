import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    # Load initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    # Define the objective function
    def objective(h):
        # Compute the convolution of h with (1 - h)
        conv = correlate(h, 1 - h, mode='full')
        # The maximum overlap is the maximum value of the convolution scaled by dx
        return np.max(conv) * dx

    # Define constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    cons = [{'type': 'eq', 'fun': constraint_sum}]

    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Initial guess
    h0 = initial_h.copy()

    # Perform optimization with Nelder-Mead method
    result = minimize(objective, h0, method='Nelder-Mead', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 100})

    # Extract the best solution
    best_h = result.x
    best_c5 = objective(best_h)

    return best_h, best_c5, n_points
