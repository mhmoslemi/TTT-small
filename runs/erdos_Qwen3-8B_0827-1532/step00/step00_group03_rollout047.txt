        return max_corr * dx
    
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
    bounds = [(0, 1) for _ in range(n_points)]
    
    result = minimize(
        objective,
        h_initial,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        tol=1e-6,
        options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8}
    )
    
    if result.success:
        h_values = result.x
        c5_bound = result.fun
    else:
        h_values = h_initial
        c5_bound = objective(h_initial)
    
    return h_values, c5_bound, n_points
