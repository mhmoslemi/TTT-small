import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    # Set the number of points (smaller is better for speed)
    n_points = 100
    dx = 2.0 / n_points

    # Initial guess: scale the provided initial_h_values to match the required sum
    # Assume initial_h_values is a global variable with length matching n_points
    h_initial = initial_h_values.copy()
    # Rescale to ensure sum(h) = n_points / 2
    sum_initial = np.sum(h_initial)
    scaling_factor = (n_points / 2) / sum_initial

    def objective(h):

        corr = np.correlate(h_np, 1 - h_np, mode='full')
        corr *= dx
        return np.max(corr)

    def constraint_sum(h):
