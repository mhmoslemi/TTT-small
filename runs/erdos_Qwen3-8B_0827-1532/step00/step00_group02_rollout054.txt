from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    # Set n_points
    n_points = 100
    dx = 2.0 / n_points
    
    # Initial guess: uniform distribution
    initial_guess = np.full(n_points, 0.5)
    
    # Define the objective function
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        sum_h = np.sum(h)
        penalty = 1e6 * abs(sum_h - n_points / 2)
        return max_corr * dx + penalty
    
    # Bounds for each variable
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Run differential evolution
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed,
        maxiter=100
    )
    
    # Extract the best h
    best_h = result.x
    
    # Compute the C5 value
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    c5_bound = np.max(corr) * dx
    
    return (best_h, c5_bound, n_points)
