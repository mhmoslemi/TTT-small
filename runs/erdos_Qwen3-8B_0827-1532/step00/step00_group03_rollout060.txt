import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    # Set n_points
    n_points = 100  # Increased for better precision
    dx = 2.0 / n_points

    # Initial guess
    h_initial = initial_h_values  # Assuming initial_h_values is a global numpy array

    # Define the objective function
    def objective(h):
        sum_h = np.sum(h)
        penalty = 1000 * abs(sum_h - n_points / 2)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx + penalty

    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Run differential evolution with early stopping
    start_time = time.time()
    result = differential_evolution(
        objective,
        bounds,
        popsize=15,
        maxiter=100,
        tol=1e-5,
        seed=seed,
        strategy='best1bin',
        mutation=(0.5, 1),
        recombination=0.9
    )
    elapsed = time.time() - start_time

    # Check if we are within budget
    if elapsed > budget_s:
        # Return initial guess if time is exceeded
        h_values = h_initial
        c5_bound = np.max(np.correlate(h_initial, 1 - h_initial, mode='full')) * dx
        return (h_values, c5_bound, n_points)

    # Get the best solution
    h_values = result.x
    c5_bound = np.max(np.correlate(h_values, 1 - h_values, mode='full')) * dx

    return (h_values, c5_bound, n_points)
