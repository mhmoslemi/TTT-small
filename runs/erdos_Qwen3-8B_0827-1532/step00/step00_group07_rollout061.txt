import numpy as np

    def objective(h_flat):

        max_corr = np.max(corr)
        return max_corr * dx

    def constraint_func(h_flat):
        return np.sum(h_flat) - n_points / 2

    cons = {'type': 'eq', 'fun': constraint_func}

    result = minimize(

        method='L-BFGS-B',
