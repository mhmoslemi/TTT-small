import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 200  # Smaller n_points for faster computation
    # Generate initial guess with sum h = n_points / 2
    h = np.random.uniform(0, 1, n_points)
    h = h * (n_points / 2) / np.sum(h)
    
    def objective(h_vec):
        # Compute cross-correlation with (1 - h)
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        max_corr = np.max(corr)
        return max_corr * (2.0 / n_points)
    
    # Constraints
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization settings
    start_time = time.time()
    result = minimize(
        objective,
        h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 100, 'ftol': 1e-6}
    )
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        # If time is up, return the best found so far
        return h, objective(h), n_points
    
    best_h = result.x
    c5_bound = result.fun
    return best_h, c5_bound, n_points
