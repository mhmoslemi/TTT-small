import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    # Initial guess from provided function
    initial_h_values = evaluate_erdos_solution()
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    # Objective function: max overlap integral
    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Constraint: sum(h) = n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial guess
    initial_guess = initial_h_values.copy()
    
    # Start optimization
    start_time = time.time()
    result = minimize(
        objective,
        x0=initial_guess,
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        method='L-BFGS-B',
        options={'maxiter': 100, 'ftol': 1e-6}
    )
    
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s:
        # If time runs out, return the initial guess
        return (initial_h_values, objective(initial_h_values), n_points)
    
    best_h = result.x
    c5_bound = objective(best_h)
    return (best_h, c5_bound, n_points)
