import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    # Initial setup
    initial_h = initial_h_values  # Assume this is a global variable
    n_points = 45
    dx = 2.0 / n_points
    initial_h = np.array(initial_h)
    
    def compute_max_corr_and_k(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        max_index = np.argmax(corr)
        n = len(h)
        k = max_index - (n - 1)
        return max_corr * dx, k
    
    def compute_gradient(h, k, dx, n):
        grad = np.zeros(n)
        for j in range(n):
            j_plus_k = j + k
            if 0 <= j_plus_k < n:
                term1 = (1 - h[j_plus_k]) * dx
            else:
                term1 = 1 * dx
            j_minus_k = j - k
            if 0 <= j_minus_k < n:
                term2 = h[j_minus_k] * dx
            else:
                term2 = 0
            grad[j] = term1 - term2
        return grad
    
    def optimize_h(initial_h, n_points, dx, max_iter=1000, learning_rate=0.01):
        h = initial_h.copy()
        for _ in range(max_iter):
            max_corr, k = compute_max_corr_and_k(h)
            grad = compute_gradient(h, k, dx, n_points)
            h -= learning_rate * grad
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            if sum_h != 0:
                scaling_factor = (n_points / 2) / sum_h
                h *= scaling_factor
        return h, max_corr
    
    start_time = time.time()
    h_opt, c5_bound = optimize_h(initial_h, n_points, dx, max_iter=1000, learning_rate=0.01)
    elapsed = time.time() - start_time
