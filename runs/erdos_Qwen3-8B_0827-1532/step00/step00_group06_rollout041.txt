    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed,
        polish=True,
        disp=False
    )
    
    h_opt = result.x
    c5_bound = result.fun
    return h_opt, c5_bound, n_points
