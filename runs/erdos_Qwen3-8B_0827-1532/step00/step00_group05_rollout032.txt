    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Define the objective function
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        c5 = max_corr * dx
        # Penalty for sum constraint
        sum_h = np.sum(h)
        penalty = 1e4 * (sum_h - n_points / 2)**2
        return c5 + penalty
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Start the timer
    start_time = time.time()
    
    # Run differential evolution with time-based stopping
    result = None
    max_iter = 100  # Adjust based on expected time per iteration
    for _ in range(max_iter):
        if time.time() - start_time >= budget_s:
            break
        result = differential_evolution(
            objective,
            bounds,
            popsize=15,
            mutation=(0.5, 1.0),
            recombination=0.8,
            strategy='best1bin',
            tol=1e-6,
            maxiter=1,
            polish=True,
            disp=False,
            seed=seed
        )
    
    # If no result found, use initial_h
    if result is None or result.success:
        best_h = result.x
    else:
        best_h = initial_h
    
    # Compute the actual C5 value
    def compute_c5(h, n_points):
        dx = 2.0 / n_points
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    best_c5 = compute_c5(best_h, n_points)
    
    return (best_h, best_c5, n_points)
