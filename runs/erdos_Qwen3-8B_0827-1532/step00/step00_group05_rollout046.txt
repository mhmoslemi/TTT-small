from scipy.optimize import minimize
from scipy.signal import correlate
import time

        # Find maximum value
        max_corr = np.max(corr)
        return max_corr * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = h_initial

    num_restarts = 5

        h_perturbed = h_initial + perturbation
