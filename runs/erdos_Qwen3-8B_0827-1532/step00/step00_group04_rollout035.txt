                else:
                    child[i] = parent2[i]
            
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[i] += np.random.uniform(-0.1, 0.1)
                    child[i] = np.clip(child[i], 0, 1)
            
            # Ensure sum constraint
            child_sum = np.sum(child)
            if child_sum != 0:
                child = child / child_sum * (n_points / 2)
            
            new_population.append(child)
        
        population = new_population
        fitness = [evaluate(h) for h in population]
        
        # Check time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    # Find the best solution
    best_idx = np.argmin(fitness)
    best_h = population[best_idx]
    c5_bound = fitness[best_idx]
    
    return (best_h, c5_bound, n_points)
