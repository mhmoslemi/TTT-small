def compute_c5(h, n_points):

        # Ensure each value is between 0 and 1
        h = np.clip(h, 0, 1)
        population.append(h)
    return population

def crossover(parent1, parent2, n_points):
    # Single-point crossover
    point = np.random.randint(1, n_points)
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    # Normalize the children to have the correct sum
    child1_sum = np.sum(child1)
    child1 = child1 * (n_points / 2) / child1_sum
    child1 = np.clip(child1, 0, 1)
    child2_sum = np.sum(child2)
    child2 = child2 * (n_points / 2) / child2_sum
    child2 = np.clip(child2, 0, 1)
    return child1, child2

def mutate(individual, mutation_rate, n_points):
    for i in range(n_points):
        if np.random.rand() < mutation_rate:
            # Randomly change the value
            delta = np.random.uniform(-0.1, 0.1)
            new_val = individual[i] + delta
            new_val = np.clip(new_val, 0, 1)
            individual[i] = new_val
    # Normalize the sum
    individual_sum = np.sum(individual)
    individual = individual * (n_points / 2) / individual_sum
    individual = np.clip(individual, 0, 1)
    return individual

        # Create new population

            # Tournament selection
            a = np.random.randint(len(top_parents))
