def objective(h, n_points):
    return compute_c5(h, n_points)

def constraint_sum(h, n_points):
    return np.sum(h) - n_points / 2

def run(seed=42, budget_s=100, **kwargs):
    # Initial guess from provided initial_h_values
    initial_h_values = evaluate_erdos_solution()
    n_points = len(initial_h_values)
    initial_guess = initial_h_values.copy()
    
    # Define bounds for optimization
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    
    # Optimization using differential evolution for global search
    result = differential_evolution(
        objective,
        bounds,
        args=(n_points,),
        constraints=[constraint_sum],
        strategy='best1bin',
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.8,
        tol=1e-6,
        maxiter=100,
        disp=False
    )
    
    # Extract the best solution
    best_h = result.x
    best_c5 = compute_c5(best_h, n_points)
    
    # Ensure the solution meets the constraint
    if np.abs(np.sum(best_h) - n_points / 2) > 1e-6:
        # Fallback to initial guess if constraint is violated
        best_h = initial_guess
        best_c5 = compute_c5(best_h, n_points)
    
    return best_h, best_c5, n_points
