import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    h_array = np.array(h)
    shifted = 1 - h_array
    corr = np.correlate(h_array, shifted, mode='full')
    c5 = np.max(corr) * dx
    return c5

def run(seed=42, budget_s=100, **kwargs):
    # Initial setup
    n_points = 80
    initial_h = initial_h_values  # Assuming initial_h_values is defined globally
    
    start_time = time.time()
    
    def objective(h):
        return compute_c5(h, n_points)
    
    # Constraints: sum(h) = n_points / 2
    constraints = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
    
    # Bounds for each h[i]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization settings
    result = minimize(
        fun=objective,
        x0=initial_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 100, 'ftol': 1e-6, 'gtol': 1e-6}
    )
    
    elapsed_time = time.time() - start_time
    
    # Early exit if budget is exceeded
    if elapsed_time > budget_s:

    best_h = result.x
    c5_bound = compute_c5(best_h, n_points)
