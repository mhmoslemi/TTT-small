import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        corr *= dx
        return np.max(corr)

    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    temperature = 100.0
    cooling_rate = 0.995
    start_time = time.time()

    while time.time() - start_time < budget_s:
        new_h = h.copy()
        i = np.random.randint(n_points)
        j = np.random.randint(n_points)
        while i == j:
            j = np.random.randint(n_points)

        if new_h[i] == 0 and new_h[j] == 1:
            new_h[i], new_h[j] = 1, 0
