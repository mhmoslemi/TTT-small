    result = scipy.optimize.minimize(
        fun=objective,
        x0=initial_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
    )
    
    # Check elapsed time and stop if budget is exceeded
    elapsed = time.time() - start_time
    if elapsed >= budget_s - 1:  # Leave a buffer
        # If time is up, return the current best
        return result.x, result.fun, n_points
    
    # Further refinement with COBYLA
    result_cobyla = scipy.optimize.minimize(
        fun=objective,
        x0=result.x,
        method='COBYLA',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 50, 'ftol': 1e-8}
    )
    
    # Final check on time
    elapsed = time.time() - start_time
    if elapsed >= budget_s - 1:
        return result_cobyla.x, result_cobyla.fun, n_points
    
    # Return the best solution found
    return result_cobyla.x, result_cobyla.fun, n_points
