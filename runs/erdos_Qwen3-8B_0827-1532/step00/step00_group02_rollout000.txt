    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded")
        return initial_h, objective(initial_h), n_points
    
    if result.success:
        h_values = result.x
        c5_bound = result.fun
    else:
        h_values = initial_h
        c5_bound = objective(initial_h)
    
    return h_values, c5_bound, n_points
