import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=100, **kwargs):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    # Generate starting points
    starting_points = [initial_h_values.copy()]
    np.random.seed(seed)
    for _ in range(3):  # Generate 3 perturbed starting points
        perturb = np.random.normal(0, 0.05, size=n_points)

        h_perturbed = np.clip(h_perturbed, 0, 1)
        current_sum = np.sum(h_perturbed)
        target_sum = n_points / 2
        scale_factor = target_sum / current_sum
        h_perturbed *= scale_factor
        starting_points.append(h_perturbed)
