    # Constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Start the timer
    start_time = time.time()
    
    # Perform optimization
    result = minimize(objective, h_initial, method='L-BFGS-B', bounds=bounds, constraints=cons)
    
    # Calculate elapsed time
    elapsed_time = time.time() - start_time
    
    # If within budget, return the best solution
    if elapsed_time < budget_s:
        best_h = result.x
        c5_bound = evaluate_erdos_solution(best_h)
        return (best_h, c5_bound, n_points)
    else:
        # If time is up, return the initial guess
        return (h_initial, evaluate_erdos_solution(h_initial), n_points)
