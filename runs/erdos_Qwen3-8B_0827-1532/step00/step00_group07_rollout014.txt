def run(seed=42, budget_s=100, **kwargs):
    import numpy as np
    from scipy.optimize import minimize

    # Assume initial_h_values is a global variable provided
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points

    # Objective function to minimize the maximum overlap
    def objective(x):
        h_new = x.reshape(n_points)
        sum_h = np.sum(h_new)
        shifts = np.arange(n_points)
        shifted_h = np.roll(h_new, shifts).T
        dot_products = np.dot(h_new, shifted_h)
        overlaps = (sum_h - dot_products) * dx
        return np.max(overlaps)

    # Constraints: sum(h) == n_points / 2 and 0 <= h[i] <= 1
    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    cons = [{'type': 'eq', 'fun': constraint_sum}]

    result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
