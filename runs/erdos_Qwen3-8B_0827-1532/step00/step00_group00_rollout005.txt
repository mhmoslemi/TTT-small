        h_array = np.array(h)
        one_minus_h = 1.0 - h_array
        corr = correlate(h_array, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraints: sum(h) = n_points / 2, 0 <= h[i] <= 1
    cons = (
        {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2},
        {'type': 'ineq', 'fun': lambda h: h},  # h >= 0
        {'type': 'ineq', 'fun': lambda h: 1 - h}  # h <= 1
    )
    
    # Initial guess
    h0 = initial_h.copy()
    
    # Optimization settings
    result = minimize(
        compute_c5,
        h0,
        method='L-BFGS-B',
        bounds=[(0.0, 1.0) for _ in range(n_points)],
        constraints=cons,
        options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
    )
    
    best_h = result.x
    best_c5 = result.fun
    
    return best_h, best_c5, n_points
