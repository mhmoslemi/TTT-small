        sum_h = np.sum(h)
        constraint_violation = (sum_h - n_points / 2) ** 2
        return max_corr + 1e6 * constraint_violation
    
    # Bounds and constraints
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial guess
    h_initial = initial_h_values.copy()
    
    # Use differential evolution for global search
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=50,
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed
    )
    
    # Extract the best solution
    h_values = result.x
    c5_bound = result.fun
    
    return h_values, c5_bound, n_points
