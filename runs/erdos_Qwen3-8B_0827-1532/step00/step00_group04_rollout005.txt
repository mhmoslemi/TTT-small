import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Assume initial_h_values is provided as a global variable
    global initial_h_values
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    max_iterations = 100  # Adjust based on time constraints
    start_time = time.time()
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    for _ in range(max_iterations):
        # Define objective function
        def objective(h_flat):
            h = h_flat.reshape(n_points)
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx

        # Define constraints
        cons = ({'type': 'eq', 'fun': lambda h_flat: np.sum(h_flat) - n_points / 2})

        # Define bounds
        bounds = [(0, 1) for _ in range(n_points)]

        # Initial guess
        h_initial = h.copy()

        # Run optimization
        result = minimize(objective, h_initial, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)

        # Update best solution
        if result.success:
            current_c5 = np.max(np.correlate(result.x, 1 - result.x, mode='full')) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
            h = result.x
        else:
            # If optimization fails, do not update h
            pass

        # Check time
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break

    # Compute final C5
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    c5_bound = np.max(corr) * dx

    return best_h, c5_bound, n_points
