    result = differential_evolution(
        objective, 
        bounds, 
        strategy='best1bin', 
        popsize=15, 
        mutation=(0.5, 1), 
        recombination=0.8, 
        tol=1e-6, 
        maxiter=100, 
        seed=seed
    )
    
    best_h = result.x
    best_c5 = evaluate_erdos_solution(best_h)
    return (best_h, best_c5, n_points)
