        x0=h_initial,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=constraint,
        options=options
    )
    
    # Extract the best solution
    h_values = result.x
    c5_bound = compute_c5(h_values, dx)
    return h_values, c5_bound, n_points
