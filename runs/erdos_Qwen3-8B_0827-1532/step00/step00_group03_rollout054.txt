    # Run the optimization with COBYLA method for derivative-free optimization
    result = minimize(
        objective, 
        h_initial, 
        method='COBYLA', 
        bounds=bounds, 
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 1000}
    )
    
    # Compute the resulting C5
    cross_corr = np.correlate(result.x, result.x, mode='full')
    min_cross_corr = np.min(cross_corr)
    c5_bound = 1 - min_cross_corr
    
    return (result.x, c5_bound, n_points)
