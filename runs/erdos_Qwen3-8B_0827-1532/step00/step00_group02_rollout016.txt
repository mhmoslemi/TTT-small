```python
        bounds=bounds,
        constraints=constraint,
        options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
    )
    elapsed_time = time.time() - start_time
    
    # Early stopping if time is up
    if elapsed_time >= budget_s:
        return h_initial, objective(h_initial), n_points
    
    # Extract the best solution
    best_h = result.x
    c5_bound = objective(best_h)
    
    return best_h, c5_bound, n_points
```
