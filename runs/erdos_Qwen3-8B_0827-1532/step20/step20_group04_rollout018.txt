    initial_h_values = evaluate_erdos_solution()
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Define the optimization variable
    h = cp.Variable(n_points)
    h.value = initial_h_values.copy()  # Set initial guess
    
    # Compute cross-correlation for all shifts
    cross_corrs = []
    for k in range(2 * n_points - 1):
        terms = []
        for i in range(n_points):
            j = (i + k) % n_points
            terms.append(h[i] * (1 - h[j]))
        cross_corr = cp.sum(terms) * dx
        cross_corrs.append(cross_corr)
    
    # Objective: minimize the maximum cross-correlation
    objective = cp.Minimize(cp.Max(cross_corrs))
    
    # Constraint: sum of h is target_sum
    constraints = [cp.sum(h) == target_sum]
    
    # Solve the problem
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.SCS, verbose=False)
    
    # Extract the solution
    best_h = h.value
    best_c5 = prob.value
    
    return best_h, best_c5, n_points
