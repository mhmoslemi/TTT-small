from scipy.optimize import minimize, differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):

    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate diverse initial guesses
    initial_guesses = [initial_guess.copy()]
    
    # Random binary sequence with exactly half 1s
    h_binary = np.zeros(n_points)
    h_binary[:n_points//2] = 1.0
    np.random.shuffle(h_binary)
    initial_guesses.append(h_binary)
    
    # Alternating 1 and 0
    h_alt = np.zeros(n_points)
    h_alt[::2] = 1.0
    sum_h = np.sum(h_alt)
    scale = target_sum / sum_h
    initial_guesses.append(h_alt * scale)
    
    # Central 50% step function
    h_center = np.zeros(n_points)
    center_start = n_points // 2 - n_points // 4
    center_end = n_points // 2 + n_points // 4
    h_center[center_start:center_end] = 1.0
    sum_h = np.sum(h_center)
    scale = target_sum / sum_h
    initial_guesses.append(h_center * scale)
    
    # Two-segment step function
    h_two_seg = np.zeros(n_points)
    h_two_seg[:n_points//4] = 1.0
    h_two_seg[n_points//2:n_points//2 + n_points//4] = 1.0
    sum_h = np.sum(h_two_seg)
    scale = target_sum / sum_h
    initial_guesses.append(h_two_seg * scale)

        end = int((seg_positions[i] + seg_lengths[i]) * n_points / 2)

    sum_h = np.sum(h_three_seg)

    sum_h = np.sum(h_four_seg)
    scale = target_sum / sum_h
    initial_guesses.append(h_four_seg * scale)

    initial_guesses.append(np.random.uniform(0, 1, n_points))
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,

        perturbed = np.clip(perturbed, 0, 1)

            one_minus_h = 1.0 - h

            return np.max(corr) * dx
