    best_c5 = float('inf')
    best_h = h_initial
    
    start_time = time.time()
    
    # Optimize each initial guess with differential evolution
    for guess in initial_guesses:
        # Bounds and constraints
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': constraint})
        
        # Use differential evolution for global optimization
        result = differential_evolution(
            objective,
            bounds,
            constraints=cons,
            popsize=15,
            maxiter=50,
            tol=1e-9,
            mutation=(0.5, 1.0),
            recombination=0.9,
            seed=seed
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Perturbation phase on the best_h found so far
    num_perturbations = 300
    phase_shift = 0.2
    phase_decay = 0.95
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 200}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
