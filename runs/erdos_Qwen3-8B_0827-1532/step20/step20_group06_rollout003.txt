    # Random binary sequence with exactly half 1s
    h_binary = np.zeros(n_points)
    h_binary[:n_points//2] = 1.0
    np.random.shuffle(h_binary)
    initial_guesses.append(h_binary)
    
    # Alternating 1 and 0
    h_alt = np.zeros(n_points)
    h_alt[::2] = 1.0
    sum_h = np.sum(h_alt)
    scale = target_sum / sum_h
    initial_guesses.append(h_alt * scale)
    
    # Central 50% step function
    h_center = np.zeros(n_points)
    center_start = n_points // 2 - n_points // 4
    center_end = n_points // 2 + n_points // 4
    h_center[center_start:center_end] = 1.0
    sum_h = np.sum(h_center)
    scale = target_sum / sum_h
    initial_guesses.append(h_center * scale)
    
    # Two-segment step function
    h_two_seg = np.zeros(n_points)
    h_two_seg[:n_points//4] = 1.0
    h_two_seg[n_points//2:n_points//2 + n_points//4] = 1.0
    sum_h = np.sum(h_two_seg)
    scale = target_sum / sum_h
    initial_guesses.append(h_two_seg * scale)
    
    # Random initial guess
    initial_guesses.append(np.random.uniform(0, 1, n_points))
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            # Compute lags from -2 to 2
            lag_values = np.arange(-2, 3)  # includes -2, -1, 0, 1, 2
            indices = [n_points - 1 + k for k in lag_values]
            valid_indices = [i for i in indices if 0 <= i < len(corr)]
            max_corr = np.max(corr[valid_indices]) * dx
            return max_corr
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 1500}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Perturbation phase with structured variations
    num_perturbations = 400  # Increased from 300
    phase_shift = 0.2
    phase_decay = 0.95
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            # Compute lags from -2 to 2
            lag_values = np.arange(-2, 3)  # includes -2, -1, 0, 1, 2
            indices = [n_points - 1 + k for k in lag_values]
            valid_indices = [i for i in indices if 0 <= i < len(corr)]
            max_corr = np.max(corr[valid_indices]) * dx
            return max_corr
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 500}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
