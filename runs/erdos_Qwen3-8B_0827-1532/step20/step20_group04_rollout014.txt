    h_four_seg[seg3_start:seg3_end] = 1.0
    sum_h = np.sum(h_four_seg)
    scale = target_sum / sum_h
    initial_guesses.append(h_four_seg * scale)
    
    # Random initial guess
    initial_guesses.append(np.random.uniform(0, 1, n_points))
    
    # Optimize each initial guess with differential evolution
    for guess in initial_guesses:
        # Objective function with penalty for sum constraint
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr) * dx
            penalty = 1e6 * np.abs(np.sum(h) - target_sum)
            return max_corr + penalty
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Run differential evolution
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=10,
            maxiter=100,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed
        )
        
        if result.success:
            current_h = result.x
            one_minus_h = 1.0 - current_h
            corr = np.correlate(current_h, one_minus_h, mode='full')
            current_c5 = np.max(corr) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
    
    # Perturbation phase with random noise
    num_perturbations = 50  # Reduced from 100 to save time
    phase_shift = 0.15
    phase_decay = 0.96
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        # Apply random noise perturbation
        perturbed = best_h + noise_scale * np.random.uniform(-1, 1, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale

        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Run differential evolution
        result = differential_evolution(

            strategy='best1bin',
            popsize=10,
            maxiter=50,
            tol=1e-6,
            mutation=(0.5, 1),
