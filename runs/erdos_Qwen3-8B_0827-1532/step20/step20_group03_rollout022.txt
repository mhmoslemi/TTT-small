    scale = target_sum / sum_h
    initial_guesses.append(h_center * scale)
    
    # Three regions step function
    h_three_regions = np.zeros(n_points)
    h_three_regions[:n_points//3] = 1.0
    h_three_regions[n_points//3:2*n_points//3] = 0.0
    h_three_regions[2*n_points//3:] = 1.0
    sum_h = np.sum(h_three_regions)
    scale = target_sum / sum_h
    initial_guesses.append(h_three_regions * scale)
    
    # Varying widths step function
    h_varying_widths = np.zeros(n_points)
    h_varying_widths[:n_points//4] = 1.0
    h_varying_widths[n_points//2:n_points//2 + n_points//4] = 1.0
    h_varying_widths[3*n_points//4:] = 1.0
    sum_h = np.sum(h_varying_widths)
    scale = target_sum / sum_h
    initial_guesses.append(h_varying_widths * scale)
    
    # Four regions step function
    h_four_regions = np.zeros(n_points)
    h_four_regions[:n_points//4] = 1.0
    h_four_regions[n_points//4:2*n_points//4] = 0.5
    h_four_regions[2*n_points//4:3*n_points//4] = 0.0
    h_four_regions[3*n_points//4:] = 0.5
    sum_h = np.sum(h_four_regions)
    scale = target_sum / sum_h
    initial_guesses.append(h_four_regions * scale)
    
    # Random step function with more steps
    h_random_more_steps = np.zeros(n_points)
    num_steps = np.random.randint(5, 8)
    positions = np.random.choice(n_points, num_steps, replace=False)
    h_random_more_steps[positions] = 1.0
    sum_h = np.sum(h_random_more_steps)
    scale = target_sum / sum_h
    initial_guesses.append(h_random_more_steps * scale)
    
    # Random initial guess
    initial_guesses.append(np.random.uniform(0, 1, n_points))
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 1000}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Perturbation phase on the best_h found so far
    num_perturbations = 700  # Increased from 500
    phase_shift = 0.3
    phase_decay = 0.95
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 400}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
