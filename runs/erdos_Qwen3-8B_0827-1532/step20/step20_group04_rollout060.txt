    # Alternating 1 and 0
    h_alt = np.zeros(n_points)
    h_alt[::2] = 1.0
    sum_h = np.sum(h_alt)
    scale = target_sum / sum_h
    initial_guesses.append(h_alt * scale)
    
    # Four-segment step function
    h_four_seg = np.zeros(n_points)
    seg1_start = n_points // 4
    seg1_end = n_points // 2
    seg3_start = 3 * n_points // 4
    seg3_end = n_points
    h_four_seg[seg1_start:seg1_end] = 1.0
    h_four_seg[seg3_start:seg3_end] = 1.0
    sum_h = np.sum(h_four_seg)
    scale = target_sum / sum_h
    initial_guesses.append(h_four_seg * scale)
    
    # Sinusoidal pattern
    h_sinusoidal = 0.5 + 0.5 * np.sin(np.pi * np.arange(n_points) / n_points)
    sum_h = np.sum(h_sinusoidal)
    scale = target_sum / sum_h
    initial_guesses.append(h_sinusoidal * scale)
    
    # Random initial guess
    initial_guesses.append(np.random.uniform(0, 1, n_points))
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Objective function for DE
        def objective_de(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Bounds for DE
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Run DE
        result_de = differential_evolution(
            objective_de, bounds, strategy='best1bin', 
            popsize=15, mutation=(0.5, 1.0), recombination=0.8, 
            tol=1e-6, maxiter=100
        )
        
        # Scale to meet sum constraint
        h_de = result_de.x
        sum_de = np.sum(h_de)
        scale = target_sum / sum_de
        h_scaled = h_de * scale
        
        # Objective function for SLSQP
        def objective_slsqp(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Run SLSQP
        bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result_slsqp = minimize(
            objective_slsqp,
            h_scaled,
            method='SLSQP',
            bounds=bounds_slsqp,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 800}
        )
        
        if result_slsqp.success and result_slsqp.fun < best_c5:
            best_c5 = result_slsqp.fun
            best_h = result_slsqp.x
    
    # Perturbation phase with random noise
    num_perturbations = 100  # Reduced from 200 to save time
    phase_shift = 0.15
    phase_decay = 0.96
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        # Apply random noise perturbation
        perturbed = best_h + noise_scale * np.random.uniform(-1, 1, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function for SLSQP
        def objective_slsqp(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Run SLSQP
        bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result_slsqp = minimize(
            objective_slsqp,
            perturbed,
            method='SLSQP',
            bounds=bounds_slsqp,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 300}
        )
        
        if result_slsqp.success and result_slsqp.fun < best_c5:
            best_c5 = result_slsqp.fun
            best_h = result_slsqp.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
