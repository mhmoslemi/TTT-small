from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Set the number of points
    n_points = 100
    dx = 2.0 / n_points

    num_guesses = 5
    initial_guesses = []

        perturbed = h_initial + np.random.normal(0, 0.1, n_points)

        sum_perturbed = np.sum(perturbed)

        initial_guesses.append(perturbed)
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    for guess in initial_guesses:
        # Define the objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')

        result = minimize(
