from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = 80
    dx = 2.0 / n_points
    np.random.seed(seed)

    initial_guesses = []
    
    # 1. Use provided initial_h_values
    initial_guesses.append(initial_h_values)

    random_initial[indices] = 1
    initial_guesses.append(random_initial)
    
    # 3. Structured every other interval
    structured_initial = np.zeros(n_points)
    structured_initial[::2] = 1
    initial_guesses.append(structured_initial)
    
    # 4. Evenly spaced initial guess
    evenly_spaced_initial = np.zeros(n_points)
    for i in range(n_points//2):
        pos = int(i * n_points / (n_points//2))
        evenly_spaced_initial[pos] = 1
    initial_guesses.append(evenly_spaced_initial)
    
    # 5. Structured every 3rd interval
    structured_initial_3 = np.zeros(n_points)
    for i in range(n_points//3):
        pos = int(i * n_points / (n_points//3))
        structured_initial_3[pos] = 1
    initial_guesses.append(structured_initial_3)
    
    # 6. Clustered pattern
    cluster_initial = np.zeros(n_points)
    for i in range(0, n_points, 5):
        cluster_initial[i:i+2] = 1
    initial_guesses.append(cluster_initial)
    
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    
    def optimize(initial_h):
        def objective(h):
            corr = correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)

        constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        result = minimize(

            initial_h,
            method='L-BFGS-B',

        )

    for guess in initial_guesses:
        h, c5 = optimize(guess)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
        # Check if time is exceeded
        if time.time() - start_time > budget_s:
            break
    
    return (best_h, best_c5, n_points)
