            max_overlap = max(max_overlap, integral)
    return max_overlap

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    # Generate diverse initial guesses
    initial_guesses = []
    initial_guesses.append(initial_h_values)
    # Perturb initial guess
    for _ in range(3):
        perturbed = initial_h_values + np.random.uniform(-0.05, 0.05, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        target_sum = n_points / 2
        scale = target_sum / current_sum
        perturbed *= scale
        initial_guesses.append(perturbed)
    # Random guess
    random_guess = np.random.rand(n_points)
    current_sum = np.sum(random_guess)
    target_sum = n_points / 2
    scale = target_sum / current_sum
    random_guess *= scale
    initial_guesses.append(random_guess)
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    for guess in initial_guesses:
        # Define objective function
        def objective(h):
            return evaluate_erdos_solution(h, n_points)
        
        # Optimization
        h_initial = guess.copy()
        constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
        bounds = Bounds(0, 1)
        
        # Run SLSQP with tighter tolerance
        result = minimize(objective, h_initial, method='SLSQP', bounds=bounds, constraints=constraint, tol=1e-6)
        
        # Check if we have time left to refine
        elapsed = time.time() - start_time
        if elapsed < budget_s:
            # Try a few more iterations with a smaller tolerance
            result = minimize(objective, result.x, method='SLSQP', bounds=bounds, constraints=constraint, tol=1e-8)
        
        current_c5 = evaluate_erdos_solution(result.x, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x
    
    return best_h, best_c5, n_points
