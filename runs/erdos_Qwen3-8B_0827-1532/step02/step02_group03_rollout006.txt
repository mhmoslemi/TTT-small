        return np.max(conv) * dx
    
    best_h = None
    best_c5 = float('inf')
    elapsed_time = 0
    
    # Generate multiple initial guesses
    num_initial_guesses = 5
    initial_guesses = []
    # Add the provided initial_h_values
    initial_guesses.append(initial_h_values)
    # Generate random guesses
    for _ in range(num_initial_guesses - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        initial_guesses.append(h)
    
    # For each initial guess, run optimization
    for initial_guess in initial_guesses:
        # Perturb the initial guess slightly
        perturbed = initial_guess + np.random.uniform(-0.05, 0.05, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scaling_factor = (n_points / 2) / current_sum
        perturbed *= scaling_factor
        
        # Run optimization
        start_time = time.time()
        result = minimize(objective, perturbed, method='SLSQP', bounds=bounds, constraints=constraint, tol=1e-6)
        elapsed = time.time() - start_time
        elapsed_time += elapsed
        
        # Refine with tighter tolerance if time permits
        if elapsed < budget_s:
            result = minimize(objective, result.x, method='SLSQP', bounds=bounds, constraints=constraint, tol=1e-8)
        
        current_c5 = evaluate_erdos_solution(result.x, n_points)
        if current_c5 < best_c5:
            best_h = result.x
            best_c5 = current_c5
        
        # Early exit if budget is exhausted
        if elapsed_time >= budget_s:
            break
    
    return best_h, best_c5, n_points
