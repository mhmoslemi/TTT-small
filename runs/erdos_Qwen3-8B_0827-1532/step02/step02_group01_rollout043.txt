    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    sum_h = np.sum(h)
    penalty = 1e4 * np.abs(sum_h - target_sum)
    return max_corr * dx + penalty

def run(seed=42, budget_s=100, **kwargs):
    n_points = 80
    dx = 2.0 / n_points
    target_sum = n_points / 2
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    
    num_starts = 5
    for i in range(num_starts):
        np.random.seed(seed + i)
        h_initial = initial_h_values.copy()
        h_perturbed = h_initial + np.random.normal(0, 0.05, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        current_sum = np.sum(h_perturbed)
        scale = target_sum / current_sum
        h_perturbed = h_perturbed * scale
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        result = differential_evolution(
            lambda h: objective(h, dx, target_sum),
            bounds,
            popsize=15,
            strategy='best1bin',
            maxiter=100,
            tol=1e-8,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed + i
        )
        
        current_c5 = objective(result.x, dx, target_sum)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x
    
    return best_h, best_c5, n_points
