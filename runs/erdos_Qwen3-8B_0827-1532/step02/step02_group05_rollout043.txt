    def compute_c5(h):
        n = len(h)
        max_val = 0.0
        for m in range(n):
            # Vectorized computation for each shift
            sum_val = np.sum(h[:n - m] * (1 - h[m:]))
            max_val = max(max_val, sum_val * dx)
        return max_val

    def objective(h):
        return compute_c5(h)

    constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    for guess in initial_guesses:
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=constraint,
            options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8}
        )
        h_opt = result.x
        c5 = result.fun
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt
        if time.time() - start_time > budget_s:
            break

    return (best_h, best_c5, n_points)
