    initial_guesses.append(uniform_guess)
    # Structured pattern guess (5 intervals of 0.2 length)
    pattern_guess = np.zeros(n_points)
    for i in range(5):
        start_idx = i * 14
        end_idx = start_idx + 7
        if end_idx < n_points:
            pattern_guess[start_idx:end_idx] = 1.0
    initial_guesses.append(pattern_guess)
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Perturb and optimize
        for _ in range(5):  # 5 perturbations per guess
            perturbed = guess + np.random.uniform(-0.05, 0.05, size=n_points)
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scaling_factor = (n_points / 2) / current_sum
            perturbed *= scaling_factor
            h_initial = perturbed.copy()
            start_time = time.time()
            result = minimize(objective, h_initial, method='SLSQP', bounds=bounds, constraints=constraint, tol=1e-6)
            elapsed = time.time() - start_time
            elapsed_time += elapsed
            # Refine with tighter tolerance if time permits
            if elapsed < budget_s:
                result = minimize(objective, result.x, method='SLSQP', bounds=bounds, constraints=constraint, tol=1e-8)
            current_c5 = evaluate_erdos_solution(result.x, n_points)
            if current_c5 < best_c5:
                best_h = result.x
                best_c5 = current_c5
            # Early exit if budget is exhausted
            if elapsed_time >= budget_s:
                break
    
    return best_h, best_c5, n_points
