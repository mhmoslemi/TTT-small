    n_points = 80
    dx = 2.0 / n_points
    initial_guesses = []

    # Generate initial guesses
    np.random.seed(seed)
    # 1. Use the provided initial_h_values
    initial_guesses.append(initial_h_values)

    structured_initial = np.zeros(n_points)
    structured_initial[::2] = 1
    initial_guesses.append(structured_initial)
    # 4. Evenly spaced initial guess
    evenly_spaced_initial = np.zeros(n_points)
    for i in range(n_points//2):

        evenly_spaced_initial[pos] = 1
    initial_guesses.append(evenly_spaced_initial)

    alternating_initial[::2] = 1
    initial_guesses.append(alternating_initial)
    # 6. Greedy initial guess (spread out)
    greedy_initial = np.zeros(n_points)
    used = set()
    for i in range(n_points//2):
        # Choose position farthest from existing ones
        if not used:
            pos = n_points // 2
        else:
            best_dist = -1
            best_pos = 0
            for p in range(n_points):
                if p in used:
                    continue
                dist = min(abs(p - u) for u in used)
                if dist > best_dist:
                    best_dist = dist
                    best_pos = p
            pos = best_pos
        used.add(pos)
        greedy_initial[pos] = 1
    initial_guesses.append(greedy_initial)
