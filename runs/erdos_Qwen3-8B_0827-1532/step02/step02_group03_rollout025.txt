import numpy as np
from scipy.optimize import minimize, Bounds, LinearConstraint
import time

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h_values, 1 - h_values, mode='full')
    max_overlap = np.max(conv) * dx
    return max_overlap

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
    bounds = Bounds(0, 1)
    
    def objective(h):
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv) * dx
    
    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h, n_points)
    elapsed_time = 0
    
    # Generate multiple initial guesses
    initial_guesses = [initial_h]
    # Symmetric guess
    symmetric_h = np.zeros_like(initial_h)
    for i in range(n_points):
        symmetric_h[i] = initial_h[n_points - 1 - i]
    initial_guesses.append(symmetric_h)
    # Binary guess
    binary_h = np.zeros(n_points)
    indices = np.random.choice(n_points, size=n_points // 2, replace=False)
    binary_h[indices] = 1
    initial_guesses.append(binary_h)
    
    # Generate multiple perturbations for each initial guess
    num_perturbations = 5
    for guess in initial_guesses:
        # Perturb the guess
        perturbed = guess + np.random.uniform(-0.1, 0.1, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scaling_factor = (n_points / 2) / current_sum
        perturbed *= scaling_factor
        
        # Run optimization
        h_initial = perturbed.copy()
        start_time = time.time()
        result = minimize(objective, h_initial, method='COBYLA', bounds=bounds, constraints=constraint, tol=1e-6)
        elapsed = time.time() - start_time
        elapsed_time += elapsed
        
        # If time permits, refine with tighter tolerance
        if elapsed < budget_s:
            result = minimize(objective, result.x, method='COBYLA', bounds=bounds, constraints=constraint, tol=1e-8)
        
        current_c5 = evaluate_erdos_solution(result.x, n_points)
        if current_c5 < best_c5:
            best_h = result.x
            best_c5 = current_c5
        
        # Early exit if budget is exhausted
        if elapsed_time >= budget_s:
            break
    
    return best_h, best_c5, n_points
