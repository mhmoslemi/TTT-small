        initial_guesses.append(perturbed)
    
    # Generate random initial guess
    random_guess = np.random.rand(n_points)
    current_sum = np.sum(random_guess)
    target_sum = n_points / 2
    scale = target_sum / current_sum
    random_guess *= scale
    initial_guesses.append(random_guess)
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    for guess in initial_guesses:
        def objective(h):
            conv = np.correlate(h, 1 - h, mode='full')
            return np.max(conv) * dx
        
        h_initial = guess.copy()
        constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
        bounds = Bounds(0, 1)
        
        # Use COBYLA for robust optimization
        result = minimize(objective, h_initial, method='COBYLA', bounds=bounds, constraints=constraint, tol=1e-6)
        
        # Refine if time permits
        elapsed = time.time() - start_time
        if elapsed < budget_s:
            result = minimize(objective, result.x, method='COBYLA', bounds=bounds, constraints=constraint, tol=1e-8)
        
        current_c5 = evaluate_erdos_solution(result.x, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x
    
    return best_h, best_c5, n_points
