import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    n_points = 80
    dx = 2.0 / n_points
    initial_h = initial_h_values  # Assume this is a global variable
    
    # Generate multiple initial guesses
    def generate_binary_initial_guess(n_points):
        indices = np.random.choice(n_points, size=n_points//2, replace=False)
        h = np.zeros(n_points)
        h[indices] = 1.0
        return h
    
    def generate_random_initial_guess(n_points):
        h = np.random.rand(n_points)
        scale = (n_points / 2) / np.sum(h)
        h = np.clip(h * scale, 0, 1)
        return h
    
    initial_guesses = []
    initial_guesses.append(initial_h)
    initial_guesses.append(generate_binary_initial_guess(n_points))
    initial_guesses.append(generate_random_initial_guess(n_points))
    
    # Define the objective function
    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the constraint: sum(h) must equal n_points / 2
    constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
    
    # Define bounds for each h[i] to be in [0, 1]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = None
    best_c5 = float('inf')
    elapsed_time = 0
    
    for guess in initial_guesses:
        # Run optimization from this guess
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=constraint,
            options={'maxiter': 500, 'ftol': 1e-8, 'eps': 1e-8}
        )
        # Check if result is successful
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        # Check elapsed time and break if budget is exceeded
        if elapsed_time + result.message.split()[0].split(' ')[-1] > budget_s:
            break
        elapsed_time += result.message.split()[0].split(' ')[-1]
    
    return (best_h, best_c5, n_points)
