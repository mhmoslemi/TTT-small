    initial_guesses = []

    new_x = np.linspace(0, 2, n_points)
    h_initial = np.interp(new_x, original_x, initial_h_values)
    sum_initial = np.sum(h_initial)
    scale_factor = (n_points / 2) / sum_initial
    h_initial *= scale_factor
    initial_guesses.append(h_initial)
    
    # Generate random initial guesses
    np.random.seed(seed)
    for _ in range(3):  # Generate 3 random initial guesses
        h_rand = np.random.rand(n_points)
        sum_rand = np.sum(h_rand)
        scale_factor = (n_points / 2) / sum_rand
        h_rand *= scale_factor
        initial_guesses.append(h_rand)
    
    # Define the objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraints: sum(h) == n_points / 2
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Function to run optimization for a single initial guess
    def optimize(initial_guess):
        start_time = time.time()
        result = minimize(
            objective,
            initial_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-8, 'maxiter': 500}
        )
        elapsed = time.time() - start_time
        return result.x, result.fun
    
    # Run optimizations in parallel
    with Pool() as pool:
        results = pool.map(optimize, initial_guesses)
    
    # Find the best result
    best_h, best_c5 = min(results, key=lambda x: x[1])
