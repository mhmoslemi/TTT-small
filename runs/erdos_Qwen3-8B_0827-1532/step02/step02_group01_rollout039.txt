        scale = (n_points / 2) / current_sum
        h_perturbed = h_perturbed * scale
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
        
        # Optimize with COBYLA method
        result = minimize(
            objective, 
            h_perturbed, 
            args=(dx,), 
            method='COBYLA', 
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'disp': False}
        )
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            # Fallback to perturbed guess if optimization fails
            current_c5 = objective(h_perturbed, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_perturbed
    
    return best_h, best_c5, n_points
