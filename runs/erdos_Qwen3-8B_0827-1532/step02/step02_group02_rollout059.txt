from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Set the number of points
    n_points = 100
    dx = 2.0 / n_points
    
    # Generate initial guesses
    initial_guesses = []
    
    # Resample initial_h_values
    original_x = np.linspace(0, 2, len(initial_h_values))
    new_x = np.linspace(0, 2, n_points)
    h_initial = np.interp(new_x, original_x, initial_h_values)
    sum_initial = np.sum(h_initial)
    scale_factor = (n_points / 2) / sum_initial
    h_initial *= scale_factor
    initial_guesses.append(h_initial)
    
    # Random initial guess
    h_random = np.random.rand(n_points)
    sum_random = np.sum(h_random)
    scale_factor_random = (n_points / 2) / sum_random
    h_random *= scale_factor_random
    initial_guesses.append(h_random)
    
    # Piecewise initial guess (1 on [0, 0.5], 0 elsewhere)
    h_piecewise = np.zeros(n_points)
    for i in range(n_points):
        if new_x[i] < 0.5:
            h_piecewise[i] = 1.0
        else:
            h_piecewise[i] = 0.0
    sum_piecewise = np.sum(h_piecewise)
    scale_factor_piecewise = (n_points / 2) / sum_piecewise
    h_piecewise *= scale_factor_piecewise
    initial_guesses.append(h_piecewise)
    
    # Now, run optimization for each initial guess
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Constraints: sum(h) == n_points / 2
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Start optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-8, 'maxiter': 200}
        )
        
        # Update best result
        current_c5 = result.fun
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x
    
    # Check if time is exceeded
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points

import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Set the number of points
    n_points = 100
    dx = 2.0 / n_points
    
    # Generate initial guesses
    initial_guesses = []
    
    # Resample initial_h_values
    original_x = np.linspace(0, 2, len(initial_h_values))
    new_x = np.linspace(0, 2, n_points)
    h_initial = np.interp(new_x, original_x, initial_h_values)
    sum_initial = np.sum(h_initial)
    scale_factor = (n_points / 2) / sum_initial
    h_initial *= scale_factor
    initial_guesses.append(h_initial)
    
    # Random initial guess
    h_random = np.random.rand(n_points)
    sum_random = np.sum(h_random)
    scale_factor_random = (n_points / 2) / sum_random
    h_random *= scale_factor_random
    initial_guesses.append(h_random)
    
    # Piecewise initial guess (1 on [0, 0.5], 0 elsewhere)
    h_piecewise = np.zeros(n_points)
    for i in range(n_points):
        if new_x[i] < 0.5:
            h_piecewise[i] = 1.0
        else:
            h_piecewise[i] = 0.0
    sum_piecewise = np.sum(h_piecewise)
    scale_factor_piecewise = (n_points / 2) / sum_piecewise
    h_piecewise *= scale_factor_piecewise
    initial_guesses.append(h_piecewise)
    
    # Now, run optimization for each initial guess
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Constraints: sum(h) == n_points / 2
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Start optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-8, 'maxiter': 200}
        )
        
        # Update best result
        current_c5 = result.fun
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x
    
    # Check if time is exceeded
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
