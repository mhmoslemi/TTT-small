        initial_guesses.append(h)
    
    constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
    bounds = Bounds(0, 1)
    
    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h, n_points)
    elapsed_time = 0
    
    # Optimize from each initial guess
    for h_initial in initial_guesses:
        start_time = time.time()
        # First optimization with moderate tolerance
        result = minimize(objective, h_initial, method='SLSQP', bounds=bounds, constraints=constraint, tol=1e-6)
        elapsed = time.time() - start_time
        elapsed_time += elapsed
        
        # Refine with tighter tolerance if time permits
        if elapsed < budget_s:
            result = minimize(objective, result.x, method='SLSQP', bounds=bounds, constraints=constraint, tol=1e-8)
        
        current_c5 = evaluate_erdos_solution(result.x, n_points)
        if current_c5 < best_c5:
            best_h = result.x
            best_c5 = current_c5
        
        # Early exit if budget is exhausted
        if elapsed_time >= budget_s:
            break
    
    return best_h, best_c5, n_points
