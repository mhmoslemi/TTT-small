from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = 100
    dx = 2.0 / n_points
    initial_guesses = []

    # Seed for reproducibility
    np.random.seed(seed)
    
    # Use the provided initial_h_values as the primary guess
    initial_guesses.append(initial_h_values)
    
    # Generate a few more diverse initial guesses
    # 1. Random initial guess
    random_initial = np.zeros(n_points)
    indices = np.random.choice(n_points, size=n_points//2, replace=False)
    random_initial[indices] = 1
    initial_guesses.append(random_initial)
    
    # 2. Structured initial guess (every other interval)
    structured_initial = np.zeros(n_points)
    structured_initial[::2] = 1
    initial_guesses.append(structured_initial)
    
    # 3. Evenly spaced initial guess
    evenly_spaced_initial = np.zeros(n_points)
    for i in range(n_points//2):
        pos = int(i * n_points / (n_points//2))
        evenly_spaced_initial[pos] = 1
    initial_guesses.append(evenly_spaced_initial)
    
    # 4. A more refined initial guess based on known optimal patterns
    # For example, alternating between high and low values with spacing
    pattern_initial = np.zeros(n_points)
    for i in range(n_points):
        if i % 4 == 0 or i % 4 == 3:
            pattern_initial[i] = 1
    initial_guesses.append(pattern_initial)
    
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()

    def objective(h):
        # Compute the cross-correlation between h and (1 - h)
        corr = correlate(h, 1 - h, mode='full')
        # The overlap integral is the maximum of the cross-correlation multiplied by dx
        max_corr = np.max(corr)
        return max_corr * dx

    # Equality constraint: sum(h) == n_points / 2
    constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    for guess in initial_guesses:
        # Use SLSQP with more iterations and tighter tolerances
        result = minimize(

            method='SLSQP',

        if c5 < best_c5:
