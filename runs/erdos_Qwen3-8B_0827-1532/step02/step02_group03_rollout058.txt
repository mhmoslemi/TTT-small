import numpy as np
from scipy.optimize import minimize, Bounds, LinearConstraint
import time

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h_values, 1 - h_values, mode='full')
    max_overlap = np.max(conv) * dx
    return max_overlap

def create_alternating_initial(n_points):
    h = np.zeros(n_points)
    h[::2] = 1.0
    return h

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 70  # assuming initial_h_values is for 70 points
    initial_guesses = [initial_h_values, create_alternating_initial(n_points)]
    
    best_h = None
    best_c5 = float('inf')
    elapsed_time = 0
    
    for initial_guess in initial_guesses:
        # Perturb the initial guess
        perturbed = initial_guess + np.random.uniform(-0.05, 0.05, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scaling_factor = (n_points / 2) / current_sum
        perturbed *= scaling_factor
        
        # Run optimization
        h_initial = perturbed.copy()
        start_time = time.time()
        result = minimize(evaluate_erdos_solution, h_initial, method='SLSQP', 
                          bounds=Bounds(0, 1), constraints=LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2]), 
                          tol=1e-6)
        elapsed = time.time() - start_time
        elapsed_time += elapsed
        
        # Refine with tighter tolerance if time permits
        if elapsed < budget_s:
            result = minimize(evaluate_erdos_solution, result.x, method='SLSQP', 
                              bounds=Bounds(0, 1), constraints=LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2]), 
                              tol=1e-8)
        
        current_c5 = evaluate_erdos_solution(result.x, n_points)
        if current_c5 < best_c5:
            best_h = result.x
            best_c5 = current_c5
        
        # Early exit if budget is exhausted
        if elapsed_time >= budget_s:
            break
    
    return best_h, best_c5, n_points
