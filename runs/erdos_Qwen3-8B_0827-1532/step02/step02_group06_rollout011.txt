import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    n_points = 100  # Increased resolution for better precision
    dx = 2.0 / n_points
    initial_h = initial_h_values  # Assume this is a global variable
    
    # Generate binary initial guess
    def generate_binary_initial_guess(n_points):
        indices = np.random.choice(n_points, size=n_points//2, replace=False)
        h = np.zeros(n_points)
        h[indices] = 1.0
        return h
    
    # Generate random initial guess
    def generate_random_initial_guess(n_points):
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
        return h
    
    # Define the objective function
    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the constraint: sum(h) must equal n_points / 2
    constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
    
    # Define bounds for each h[i] to be in [0, 1]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Generate initial guesses
    initial_guesses = [
        initial_h.copy(),
        generate_binary_initial_guess(n_points),
        generate_random_initial_guess(n_points)
    ]
    
    # Run optimizations
    best_h = None
    best_c5 = float('inf')
    elapsed_time = 0
    
    for guess in initial_guesses:
        # Perturb the guess slightly to avoid local optima
        perturbed_guess = guess + np.random.normal(0, 0.01, n_points)
        perturbed_guess = np.clip(perturbed_guess, 0, 1)
        perturbed_guess = perturbed_guess * (n_points / 2) / np.sum(perturbed_guess)
        
        # Run optimization
        result = minimize(
            objective,
            perturbed_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=constraint,
            options={'maxiter': 500, 'ftol': 1e-9, 'eps': 1e-9}
        )
        
        # Check if optimization succeeded and update best solution
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            # Fallback: try without perturbation
            result = minimize(
                objective,
                guess,
                method='SLSQP',
                bounds=bounds,
                constraints=constraint,
                options={'maxiter': 500, 'ftol': 1e-9, 'eps': 1e-9}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        
        # Check elapsed time to avoid exceeding budget
        elapsed_time += result.nfev * 0.001  # Approximate time per function evaluation
        if elapsed_time >= budget_s:
