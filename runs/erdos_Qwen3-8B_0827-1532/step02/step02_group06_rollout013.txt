from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    initial_h = initial_h_values  # Assume this is a global variable
    
    # Generate multiple initial guesses
    initial_guesses = [initial_h]
    # Binary initial guess
    def generate_binary_initial_guess(n_points):
        indices = np.random.choice(n_points, size=n_points//2, replace=False)
        h = np.zeros(n_points)
        h[indices] = 1.0
        return h
    initial_guesses.append(generate_binary_initial_guess(n_points))
    # Random perturbations of initial_h
    for _ in range(3):
        perturb = np.random.normal(0, 0.05, n_points)
        h_perturb = initial_h + perturb
        h_perturb = np.clip(h_perturb, 0, 1)
        # Adjust sum to meet constraint
        sum_h = np.sum(h_perturb)
        if sum_h != 0:
            scale = (n_points / 2) / sum_h
            h_perturb *= scale
        initial_guesses.append(h_perturb)
    
    # Define the objective function
    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the constraint: sum(h) must equal n_points / 2
    constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = None
    best_c5 = float('inf')
    elapsed_time = 0
    
    # Run optimization for each initial guess
    for h0 in initial_guesses:
        # Ensure sum constraint is met
        sum_h = np.sum(h0)
        if sum_h != 0:
            scale = (n_points / 2) / sum_h
            h0 = h0 * scale
        # Run optimization
        result = minimize(
            objective,
            h0,
