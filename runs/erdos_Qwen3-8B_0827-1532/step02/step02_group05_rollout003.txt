        pos = int(i * n_points / (n_points//2))
        evenly_spaced_initial[pos] = 1
    initial_guesses.append(evenly_spaced_initial)
    # 5. Perturbed initial guess
    perturbed_initial = initial_h_values.copy()
    np.random.seed(seed + 1)
    perturbed_initial += np.random.normal(0, 0.05, size=n_points)
    perturbed_initial = np.clip(perturbed_initial, 0, 1)
    initial_guesses.append(perturbed_initial)

    best_c5 = float('inf')
    best_h = None
    start_time = time.time()

    def optimize(initial_h):
        def objective(h):
            corr = correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx

        constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
        bounds = [(0.0, 1.0) for _ in range(n_points)]

        result = minimize(
            objective,
            initial_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraint,
            options={'maxiter': 500, 'ftol': 1e-8, 'eps': 1e-8}
        )
        return result.x, result.fun

    for guess in initial_guesses:
        h, c5 = optimize(guess)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
        # Check if time is exceeded
        if time.time() - start_time > budget_s:
            break

    # Perform additional optimization on the best_h
    if best_h is not None:
        # Run another optimization with more iterations
        def optimize_final(h):
            def objective(h):
                corr = correlate(h, 1 - h, mode='full')
                max_corr = np.max(corr)
                return max_corr * dx

            constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
            bounds = [(0.0, 1.0) for _ in range(n_points)]

            result = minimize(
                objective,
                h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=constraint,
                options={'maxiter': 500, 'ftol': 1e-8, 'eps': 1e-8}
            )
            return result.x, result.fun

        final_h, final_c5 = optimize_final(best_h)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = final_h

    return (best_h, best_c5, n_points)
