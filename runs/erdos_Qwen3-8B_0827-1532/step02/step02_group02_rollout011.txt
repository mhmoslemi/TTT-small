        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Run optimization
        result = minimize(
            objective,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-8, 'maxiter': 200}
        )
        
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        
        if result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    return best_h, best_c5, n_points
