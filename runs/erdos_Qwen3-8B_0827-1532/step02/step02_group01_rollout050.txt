import numpy as np
from scipy.optimize import minimize

def objective(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

    dx = 2.0 / n_points
    num_starts = 20  # Increased number of starts for better exploration

        scale = (n_points / 2) / current_sum

        # Optimize with tighter tolerances and more iterations
        result = minimize(
            objective, 
            h_perturbed, 
            args=(dx,), 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 500, 'ftol': 1e-8, 'xtol': 1e-8}
        )
        
        if result.success:

            if current_c5 < best_c5:
