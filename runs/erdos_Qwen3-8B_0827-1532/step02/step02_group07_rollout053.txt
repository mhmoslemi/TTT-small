from scipy.optimize import minimize

def objective(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

    n_points = 80  # Lower n_points for faster computation
    dx = 2.0 / n_points

    best_c5 = float('inf')
    best_h = np.zeros(n_points)
