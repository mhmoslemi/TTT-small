from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Set the number of points
    n_points = 100
    dx = 2.0 / n_points
    
    # Resample initial_h_values to n_points
    original_x = np.linspace(0, 2, len(initial_h_values))
    new_x = np.linspace(0, 2, n_points)
    h_initial = np.interp(new_x, original_x, initial_h_values)
    
    # Ensure the sum is correct (approximate)
    sum_initial = np.sum(h_initial)
    scale_factor = (n_points / 2) / sum_initial
    h_initial *= scale_factor
    
    # Generate multiple initial guesses
    num_perturbations = 5
    best_c5 = float('inf')
    best_h = h_initial.copy()
    
    # Set the seed for reproducibility
    np.random.seed(seed)
    
    start_time = time.time()
    
    for i in range(num_perturbations):
        # Perturb the initial guess
        perturbation = np.random.normal(0, 0.05, size=n_points)
        perturbed_h = h_initial + perturbation
        # Clip to [0, 1]
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        # Scale to meet the sum constraint
        sum_perturbed = np.sum(perturbed_h)
        scale_factor = (n_points / 2) / sum_perturbed
        perturbed_h *= scale_factor
        
        # Define objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Constraints: sum(h) == n_points / 2
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Start optimization
        result = minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-8, 'maxiter': 200}
        )
        
        # Check if we ran out of time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        
        if result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
