import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Increased for better precision
    dx = 2.0 / n_points
    initial_guesses = []
    
    # Add initial_h_values as a guess
    initial_guesses.append(initial_h_values.copy())
    
    # Generate binary initial guess
    def generate_binary_initial_guess(n_points):
        indices = np.random.choice(n_points, size=n_points//2, replace=False)
        h = np.zeros(n_points)
        h[indices] = 1.0
        return h
    initial_guesses.append(generate_binary_initial_guess(n_points))
    
    # Generate perturbed versions of initial_h_values
    perturbed_initial = initial_h_values.copy()
    noise = np.random.normal(0, 0.05, n_points)
    perturbed_initial += noise
    perturbed_initial = np.clip(perturbed_initial, 0.0, 1.0)
    initial_guesses.append(perturbed_initial)
    
    noise2 = np.random.normal(0, 0.03, n_points)
    perturbed_initial2 = initial_h_values.copy()
    perturbed_initial2 += noise2
    perturbed_initial2 = np.clip(perturbed_initial2, 0.0, 1.0)
    initial_guesses.append(perturbed_initial2)
    
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    
    for h0 in initial_guesses:
        # Try different optimization methods
        for method in ['SLSQP', 'L-BFGS-B']:
            # Define constraints and bounds
            constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
            bounds = [(0.0, 1.0) for _ in range(n_points)]
            
            # Define objective function
            def objective(h):
                corr = correlate(h, 1 - h, mode='full')
                max_corr = np.max(corr)
                return max_corr * dx
            
            # Run optimization
            result = minimize(
                objective,
                h0,
                method=method,
                bounds=bounds,
                constraints=constraint,
                options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8}
            )
            
            # Check if optimization was successful
            if result.success:
                if result.fun < best_c5:
                    best_c5 = result.fun
                    best_h = result.x
                    print(f"New best C5: {best_c5}")
    
    # Check elapsed time and return
    elapsed = time.time() - start_time
    if elapsed > budget_s - 10:  # leave 10 seconds buffer
        return (best_h, best_c5, n_points)
    return (best_h, best_c5, n_points)
