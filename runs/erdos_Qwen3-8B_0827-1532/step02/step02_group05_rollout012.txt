from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = 80
    dx = 2.0 / n_points
    np.random.seed(seed)
    
    # Generate initial guesses
    initial_guesses = [initial_h_values]
    # Perturb initial_h_values with small noise

    noise = np.random.normal(0, 0.05, n_points)
    perturbed_initial += noise
    perturbed_initial = np.clip(perturbed_initial, 0, 1)
    initial_guesses.append(perturbed_initial)
    # Add a few more structured guesses
    structured_initial = np.zeros(n_points)
    structured_initial[::3] = 1  # Every third point
    initial_guesses.append(structured_initial)
    structured_initial = np.zeros(n_points)
    structured_initial[::4] = 1  # Every fourth point
    initial_guesses.append(structured_initial)
    # Evenly spaced guess
    evenly_spaced_initial = np.zeros(n_points)
    for i in range(n_points//2):
        pos = int(i * n_points / (n_points//2))
        evenly_spaced_initial[pos] = 1
    initial_guesses.append(evenly_spaced_initial)
    
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()

    def optimize(initial_h):
        def objective(h):
            corr = correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx

        constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
        bounds = [(0.0, 1.0) for _ in range(n_points)]

        result = minimize(
            objective,
            initial_h,
            method='COBYLA',
            bounds=bounds,
            constraints=constraint,
            options={'maxiter': 500, 'disp': False}
        )
        return result.x, result.fun

    for guess in initial_guesses:
        h, c5 = optimize(guess)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
        # Check if time is exceeded
        if time.time() - start_time > budget_s:
