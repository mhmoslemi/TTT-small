import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 200  # Increased for better resolution
    dx = 2.0 / n_points
    
    # Generate initial guesses
    initial_guesses = []
    
    # Resample initial_h_values
    original_x = np.linspace(0, 2, len(initial_h_values))
    new_x = np.linspace(0, 2, n_points)
    h_initial = np.interp(new_x, original_x, initial_h_values)
    sum_initial = np.sum(h_initial)
    scale_factor = (n_points / 2) / sum_initial
    h_initial *= scale_factor
    initial_guesses.append(h_initial)
    
    # Generate perturbed initial guesses
    for _ in range(4):
        perturb = np.random.normal(0, 0.05, n_points)
        h_perturb = h_initial + perturb
        h_perturb = np.clip(h_perturb, 0, 1)
        sum_perturb = np.sum(h_perturb)
        scale = (n_points / 2) / sum_perturb
        h_perturb *= scale
        initial_guesses.append(h_perturb)
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    for guess in initial_guesses:

        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Constraints
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Run optimization

        if elapsed >= budget_s:
