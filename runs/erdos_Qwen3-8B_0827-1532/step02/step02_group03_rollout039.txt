        recombination=0.8,
        polish=True,
        disp=False
    )
    elapsed = time.time() - start_time
    elapsed_time += elapsed
    
    # Refine with SLSQP if time permits
    if elapsed < budget_s:
        from scipy.optimize import minimize, LinearConstraint
        constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
        bounds = Bounds(0, 1)
        
        def objective_slsqp(h):
            conv = np.correlate(h, 1 - h, mode='full')
            return np.max(conv) * dx
        
        result_slsqp = minimize(
            objective_slsqp,
            result.x,
            method='SLSQP',
            bounds=bounds,
            constraints=constraint,
            tol=1e-8
        )
        best_h = result_slsqp.x
        best_c5 = evaluate_erdos_solution(best_h, n_points)
    
    return best_h, best_c5, n_points
