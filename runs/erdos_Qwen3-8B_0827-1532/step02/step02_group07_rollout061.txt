import numpy as np
from scipy.optimize import minimize
from multiprocessing import Pool, cpu_count

def objective(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def evaluate_h(h, dx):
    return objective(h, dx)

def optimize_h(args):
    seed, n_points, dx, initial_h, bounds, cons = args
    np.random.seed(seed)
    if np.random.rand() < 0.5:
        h_perturbed = initial_h + np.random.normal(0, 0.01, n_points)
    else:
        h_perturbed = np.random.rand(n_points)
        h_perturbed = h_perturbed / np.sum(h_perturbed) * (n_points / 2)
    h_perturbed = np.clip(h_perturbed, 0, 1)
    result = minimize(evaluate_h, h_perturbed, args=(dx,), method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
    if result.success:
        return result.fun, result.x
    else:
        return evaluate_h(h_perturbed, dx), h_perturbed

def run(seed=42, budget_s=100, **kwargs):
    n_points = 150
    dx = 2.0 / n_points
    initial_h = initial_h_values.copy()
    initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h)
    initial_sum = np.sum(initial_h)
    initial_h = initial_h * (n_points / 2) / initial_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
    
    num_starts = min(20, cpu_count())
    args_list = []
    for i in range(num_starts):
        args_list.append((seed + i, n_points, dx, initial_h.copy(), bounds, cons))
    
    with Pool(processes=cpu_count()) as pool:
        results = pool.map(optimize_h, args_list)
    
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    for c5, h in results:
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    
    return best_h, best_c5, n_points
