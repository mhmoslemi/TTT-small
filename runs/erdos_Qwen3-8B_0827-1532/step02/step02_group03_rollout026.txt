import numpy as np
from scipy.optimize import minimize, Bounds, LinearConstraint
import time

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h_values, 1 - h_values, mode='full')
    max_overlap = np.max(conv) * dx
    return max_overlap

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
    bounds = Bounds(0, 1)
    
    def objective(h):
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv) * dx
    
    initial_guesses = []
    initial_guesses.append(initial_h_values)
    for _ in range(5):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        initial_guesses.append(h)
    
    best_h = initial_h_values.copy()
    best_c5 = evaluate_erdos_solution(best_h, n_points)
    elapsed_time = 0
    
    for h_initial in initial_guesses:
        start_time = time.time()
        result = minimize(objective, h_initial, method='trust-constr', bounds=bounds, constraints=constraint, tol=1e-8)
        elapsed = time.time() - start_time
        elapsed_time += elapsed
        
        if elapsed_time >= budget_s:

        if current_c5 < best_c5:

    return best_h, best_c5, n_points
