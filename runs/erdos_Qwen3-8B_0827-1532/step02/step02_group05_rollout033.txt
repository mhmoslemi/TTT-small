from scipy.optimize import minimize
from scipy.signal import correlate
import time

    n_points = 100  # Increased resolution for better approximation
    dx = 2.0 / n_points
    initial_guesses = []

    # Seed for reproducibility
    np.random.seed(seed)

    # Add the provided initial guess
    initial_guesses.append(initial_h_values)

    # Generate perturbed versions of the initial guess
    for _ in range(3):
        perturbed = initial_h_values.copy()
        noise = np.random.normal(0, 0.1, n_points)
        perturbed += noise
        perturbed = np.clip(perturbed, 0, 1)
        initial_guesses.append(perturbed)

    # Add random initial guess
    random_initial = np.zeros(n_points)
    indices = np.random.choice(n_points, size=n_points//2, replace=False)
    random_initial[indices] = 1
    initial_guesses.append(random_initial)

    # Add structured initial guess (every other interval)
    structured_initial = np.zeros(n_points)
    structured_initial[::2] = 1
    initial_guesses.append(structured_initial)

    # Add evenly spaced initial guess
    evenly_spaced_initial = np.zeros(n_points)
    for i in range(n_points//2):
        pos = int(i * n_points / (n_points//2))
        evenly_spaced_initial[pos] = 1
    initial_guesses.append(evenly_spaced_initial)

    best_c5 = float('inf')
    best_h = None
    start_time = time.time()

    def optimize(initial_h):
        def objective(h):
            corr = correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx

        constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
        bounds = [(0.0, 1.0) for _ in range(n_points)]

        result = minimize(
            objective,
            initial_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraint,
            options={
                'maxiter': 300,  # Increased iterations for better convergence
                'ftol': 1e-9,    # Tighter tolerance
                'eps': 1e-8

        )
        return result.x, result.fun

        h, c5 = optimize(guess)
