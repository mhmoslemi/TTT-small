    start_time = time.time()
    result = minimize(
        objective,
        h_initial,
        method='L-BFGS-B',  # More efficient for bounded optimization
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-8, 'maxiter': 200}
    )
    
    # Check if we ran out of time
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return h_initial, evaluate_erdos_solution(h_initial), n_points
    
    h_values = result.x
    c5_bound = result.fun
    return h_values, c5_bound, n_points
