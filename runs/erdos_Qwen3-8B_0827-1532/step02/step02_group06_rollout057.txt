import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    n_points = 100  # Increased resolution for better approximation
    dx = 2.0 / n_points
    initial_h = initial_h_values  # Assume this is a global variable
    
    # Generate multiple initial guesses
    initial_guesses = []
    # Add initial_h_values
    initial_guesses.append(initial_h.copy())
    # Add a binary initial guess
    def generate_binary_initial_guess(n_points):
        np.random.seed(seed)
        indices = np.random.choice(n_points, size=n_points//2, replace=False)
        h = np.zeros(n_points)
        h[indices] = 1.0
        return h
    initial_guesses.append(generate_binary_initial_guess(n_points))
    # Add perturbed versions of initial_h
    for i in range(3):
        np.random.seed(seed + i)
        perturbation = np.random.normal(0, 0.1, size=n_points)
        h_perturbed = np.clip(initial_h + perturbation, 0, 1)
        initial_guesses.append(h_perturbed)
    
    # Define the objective function
    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the constraint: sum(h) must equal n_points / 2
    constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
    
    # Define bounds for each h[i] to be in [0, 1]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = None
    best_c5 = float('inf')
    elapsed_time = 0
    
    # Run optimization for each initial guess
    for h0 in initial_guesses:
        # Use L-BFGS-B method for better convergence
        result = minimize(
            objective,
            h0,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraint,

        )
