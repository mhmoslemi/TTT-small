import numpy as np
from scipy.optimize import minimize
import scipy.stats as stats

def objective(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=100, **kwargs):
    n_points = 128
    dx = 2.0 / n_points
    num_starts = 20
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    
    np.random.seed(seed)
    
    for i in range(num_starts):

        dirichlet_sample = stats.dirichlet.rvs(alpha=[1.0]*n_points, size=1)
        h_perturbed = dirichlet_sample * (n_points / 2)

        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
        
        # Optimize with tighter tolerances and more iterations
        result = minimize(
            objective, 
            h_perturbed, 
            args=(dx,), 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-8, 'xtol': 1e-8}
        )
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            # Fallback to perturbed guess if optimization fails
            current_c5 = objective(h_perturbed, dx)
