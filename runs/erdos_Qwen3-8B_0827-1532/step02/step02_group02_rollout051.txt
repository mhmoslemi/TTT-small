import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = 100
    dx = 2.0 / n_points
    initial_guesses = []

    # Generate multiple initial guesses with perturbations
    for i in range(5):
        np.random.seed(seed + i)
        perturbed = initial_h_values + np.random.normal(0, 0.05, len(initial_h_values))
        original_x = np.linspace(0, 2, len(initial_h_values))
        new_x = np.linspace(0, 2, n_points)
        h_initial = np.interp(new_x, original_x, perturbed)
        sum_initial = np.sum(h_initial)
        scale_factor = (n_points / 2) / sum_initial
        h_initial *= scale_factor
        h_initial = np.clip(h_initial, 0.0, 1.0)
        initial_guesses.append(h_initial)

    best_c5 = float('inf')
    best_h = None
    start_time = time.time()

    for guess in initial_guesses:
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx

        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
        bounds = [(0.0, 1.0) for _ in range(n_points)]

        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-8, 'maxiter': 500}
        )

        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

        if result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x

    return best_h, best_c5, n_points
