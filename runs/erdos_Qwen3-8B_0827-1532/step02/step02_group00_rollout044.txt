import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Use the provided initial guess
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Generate multiple perturbations
    num_perturbations = 5
    best_c5 = float('inf')
    best_h = initial_h.copy()
    
    start_time = time.time()
    
    for i in range(num_perturbations):
        # Generate a small random perturbation
        perturb = np.random.normal(0, 0.01, n_points)
        h_perturbed = initial_h + perturb
        # Adjust to meet sum constraint
        sum_perturbed = np.sum(h_perturbed)
        adjustment = (n_points / 2) - sum_perturbed
        h_perturbed[-1] += adjustment
        # Clip to [0, 1] and ensure sum constraint
        h_perturbed = np.clip(h_perturbed, 0, 1)
        # Adjust last value again to maintain sum
        h_perturbed[-1] = (n_points / 2 - np.sum(h_perturbed[:-1])) / dx
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Constraints
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Run optimization
        result = minimize(
            objective,
            h_perturbed,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 200}
        )
        
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        
        if result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Final check with initial guess
    def objective_initial(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    result_initial = minimize(
        objective_initial,
        initial_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-9, 'maxiter': 200}
    )
