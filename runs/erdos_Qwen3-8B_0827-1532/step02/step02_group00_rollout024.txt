import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Predefined initial guess
    initial_h_values = np.load('initial_h_values.npy')  # Assuming this is provided
    n_points = 100
    dx = 2.0 / n_points
    
    # Generate perturbations of the initial guess
    def generate_perturbations(base, num_perturbations=5):
        perturbations = []
        for _ in range(num_perturbations):
            noise = np.random.normal(0, 0.05, base.shape)
            perturbed = base + noise
            scale_factor = (n_points / 2) / np.sum(perturbed)
            perturbed = np.clip(perturbed * scale_factor, 0, 1)
            perturbations.append(perturbed)
        return perturbations
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraints: sum(h) == n_points / 2
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Generate initial guesses
    initial_guesses = generate_perturbations(initial_h_values, num_perturbations=5)
    
    best_c5 = float('inf')
    best_h = initial_h_values
    
    start_time = time.time()
    
    for guess in initial_guesses:
        # Run optimization
        result = minimize(
            objective,
            guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 300}
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Final check with original initial guess
    result = minimize(
        objective,
        initial_h_values,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-9, 'maxiter': 300}
    )
    if result.fun < best_c5:
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points
