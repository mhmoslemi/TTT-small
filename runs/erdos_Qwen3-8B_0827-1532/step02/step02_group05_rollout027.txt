import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = 80
    dx = 2.0 / n_points
    initial_guesses = []

    np.random.seed(seed)
    # Start with provided initial guess
    initial_guesses.append(initial_h_values)

    for _ in range(3):
        noise = np.random.normal(0, 0.05, n_points)
        perturbed = initial_h_values + noise
        perturbed = np.clip(perturbed, 0, 1)
        total = np.sum(perturbed)
        scaling = (n_points / 2) / total
        perturbed *= scaling
        initial_guesses.append(perturbed)

    random_initial = np.zeros(n_points)
    indices = np.random.choice(n_points, size=n_points//2, replace=False)
    random_initial[indices] = 1
    initial_guesses.append(random_initial)

    structured_initial = np.zeros(n_points)

    initial_guesses.append(structured_initial)

        evenly_spaced_initial[pos] = 1
    initial_guesses.append(evenly_spaced_initial)

    best_c5 = float('inf')
    best_h = None
    start_time = time.time()

    def optimize(initial_h):

        constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
        bounds = [(0.0, 1.0) for _ in range(n_points)]

        result = minimize(
            objective,
            initial_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraint,
            options={'maxiter': 500, 'ftol': 1e-8, 'eps': 1e-8}
        )
        return result.x, result.fun

    for guess in initial_guesses:
        h, c5 = optimize(guess)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
        if time.time() - start_time > budget_s:
            break
