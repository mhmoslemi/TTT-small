import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
import multiprocessing

def run(seed=42, budget_s=100, **kwargs):
    n_points = 100
    dx = 2.0 / n_points
    initial_guesses = []

    # Generate initial guesses
    np.random.seed(seed)
    # 1. Use the provided initial_h_values
    initial_guesses.append(initial_h_values)
    # 2. Perturb initial_h_values slightly
    perturbed_initial = np.copy(initial_h_values)
    perturbed_initial += np.random.normal(0, 0.1, n_points)
    perturbed_initial = np.clip(perturbed_initial, 0, 1)
    initial_guesses.append(perturbed_initial)
    # 3. Structured initial guess (every other interval)
    structured_initial = np.zeros(n_points)
    structured_initial[::2] = 1
    initial_guesses.append(structured_initial)
    # 4. Evenly spaced initial guess
    evenly_spaced_initial = np.zeros(n_points)
    for i in range(n_points//2):
        pos = int(i * n_points / (n_points//2))
        evenly_spaced_initial[pos] = 1
    initial_guesses.append(evenly_spaced_initial)

    best_c5 = float('inf')
    best_h = None
    start_time = time.time()

    def optimize(initial_h):
        def objective(h):
            corr = correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx

        constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
        bounds = [(0.0, 1.0) for _ in range(n_points)]

        result = minimize(
            objective,
            initial_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraint,
            options={'maxiter': 500, 'ftol': 1e-9, 'eps': 1e-9}
        )
        return result.x, result.fun

    for guess in initial_guesses:
        h, c5 = optimize(guess)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
        # Check if time is exceeded
        if time.time() - start_time > budget_s:
            break

    return (best_h, best_c5, n_points)
