        perturbed_guesses.append(h_perturbed)
    
    # Define the objective function
    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the constraint: sum(h) must equal n_points / 2
    constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
    
    # Define bounds for each h[i] to be in [0, 1]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Run optimization from each perturbed guess
    best_h = None
    best_c5 = float('inf')
    for h0 in perturbed_guesses:
        # Run optimization
        result = minimize(
            objective,
            h0,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraint,
            options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            # If optimization fails, try another guess
            pass
    
    return (best_h, best_c5, n_points)
