        evenly_spaced_initial[pos] = 1
    initial_guesses.append(evenly_spaced_initial)
    # 5. Periodic initial guess (e.g., 1, 0, 0, 1, 0, 0, ...)
    periodic_initial = np.zeros(n_points)
    for i in range(n_points):
        if i % 3 == 0:
            periodic_initial[i] = 1
    initial_guesses.append(periodic_initial)
    # 6. Randomly perturbed initial guess
    perturbed_initial = initial_h_values.copy()
    perturbed_initial[np.random.choice(n_points, size=n_points//4, replace=False)] = 1 - perturbed_initial[np.random.choice(n_points, size=n_points//4, replace=False)]
    initial_guesses.append(perturbed_initial)

    best_c5 = float('inf')
    best_h = None
    start_time = time.time()

    def optimize(initial_h):
        def objective(h):
            corr = correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx

        constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
        bounds = [(0.0, 1.0) for _ in range(n_points)]

        result = minimize(
            objective,
            initial_h,
            method='COBYLA',
            bounds=bounds,
            constraints=constraint,
            options={'maxiter': 500, 'disp': False}
        )
        return result.x, result.fun

    for guess in initial_guesses:
        h, c5 = optimize(guess)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
        # Check if time is exceeded
        if time.time() - start_time > budget_s:
            break

    return (best_h, best_c5, n_points)
