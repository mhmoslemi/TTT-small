import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    n_points = 100  # Increased for better resolution
    dx = 2.0 / n_points
    initial_h = initial_h_values  # Assume this is a global variable
    
    # Generate diverse initial guesses
    def generate_binary_initial_guess(n_points):
        indices = np.random.choice(n_points, size=n_points//2, replace=False)
        h = np.zeros(n_points)
        h[indices] = 1.0
        return h
    
    def generate_random_initial_guess(n_points):
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
        return h
    
    def generate_perturbed_guess(h, perturb=0.1):
        h = h.copy()
        for i in range(n_points):
            if np.random.rand() < perturb:
                h[i] += np.random.uniform(-0.2, 0.2)
                h[i] = np.clip(h[i], 0.0, 1.0)
        return h
    
    # Define the objective function
    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the constraint: sum(h) must equal n_points / 2
    constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
    
    # Define bounds for each h[i] to be in [0, 1]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Generate initial guesses
    initial_guesses = []
    initial_guesses.append(initial_h.copy())
    initial_guesses.append(generate_binary_initial_guess(n_points))
    initial_guesses.append(generate_random_initial_guess(n_points))
    initial_guesses.append(generate_perturbed_guess(initial_h))
    
    best_h = None
    best_c5 = float('inf')
    elapsed_time = 0
    
    for h0 in initial_guesses:
        # Run optimization with timeout
        start_time = np.datetime64('now')
        try:
            result = minimize(
                objective,
                h0,
                method='SLSQP',
                bounds=bounds,
                constraints=constraint,
                options={'maxiter': 500, 'ftol': 1e-8, 'eps': 1e-8}
            )
            elapsed_time = np.datetime64('now') - start_time
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass  # Fallback to other guesses
    
    return (best_h, best_c5, n_points)
