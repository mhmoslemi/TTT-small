        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Start optimization
        iter_start = time.time()
        result = minimize(
            objective,
            h_perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-8, 'maxiter': 200}
        )
        iter_duration = time.time() - iter_start
        
        # Check if time budget is exceeded
        if time.time() - start_time >= budget_s:
            break
        
        # Update best solution
        current_c5 = result.fun
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x
    
    return best_h, best_c5, n_points
