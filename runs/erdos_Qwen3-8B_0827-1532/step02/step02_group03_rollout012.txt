import numpy as np
from scipy.optimize import minimize, Bounds, LinearConstraint
import time

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h_values, 1 - h_values, mode='full')
    max_overlap = np.max(conv) * dx
    return max_overlap

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 70  # Use 70 points as per initial_h_values
    dx = 2.0 / n_points
    constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
    bounds = Bounds(0, 1)
    
    def objective(h):
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv) * dx
    
    best_h = None
    best_c5 = float('inf')
    elapsed_time = 0
    
    # Generate multiple initial guesses: include initial_h_values and random ones
    num_initial_guesses = 10

        h_initial = np.random.uniform(0, 1, n_points)

        scaling_factor = (n_points / 2) / current_sum
        h_initial *= scaling_factor
        initial_guesses.append(h_initial)
