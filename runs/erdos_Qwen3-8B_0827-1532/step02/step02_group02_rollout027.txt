import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points

    original_x = np.linspace(0, 2, len(initial_h_values))
    new_x = np.linspace(0, 2, n_points)
    h_initial = np.interp(new_x, original_x, initial_h_values)
    sum_initial = np.sum(h_initial)
    scale_factor = (n_points / 2) / sum_initial
    h_initial *= scale_factor

    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h_initial.copy()
    start_time = time.time()

    num_perturbations = 5
    for _ in range(num_perturbations):
        noise = np.random.normal(0, 0.01, n_points)
        h_perturbed = h_initial + noise
        h_perturbed = np.clip(h_perturbed, 0, 1)
        sum_perturbed = np.sum(h_perturbed)
        scale_factor = (n_points / 2) / sum_perturbed
        h_perturbed *= scale_factor

        result = minimize(
            objective,
            h_perturbed,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-8, 'maxiter': 300}
        )

        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
