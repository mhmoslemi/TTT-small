        result = minimize(objective, h_initial, method='L-BFGS-B', bounds=bounds, constraints=constraint, tol=1e-6)
        
        # Refine with SLSQP if time permits
        elapsed = time.time() - start_time
        if elapsed < budget_s:
            result = minimize(objective, result.x, method='SLSQP', bounds=bounds, constraints=constraint, tol=1e-8)
        
        current_c5 = evaluate_erdos_solution(result.x, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x
    
    return best_h, best_c5, n_points
