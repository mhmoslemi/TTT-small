    def objective(h):
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv) * dx
    
    best_h = None
    best_c5 = float('inf')
    elapsed_time = 0
    
    # Generate multiple random initial guesses
    num_initial_guesses = 15  # Increased for better exploration
    initial_guesses = []
    
    # Add initial_h_values as a guess if available and compatible
    try:
        initial_h = initial_h_values
        # Resample to n_points by interpolation
        if len(initial_h) != n_points:
            initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h)), initial_h)
        initial_guesses.append(initial_h)
    except NameError:
        pass  # initial_h_values not available
    
    for _ in range(num_initial_guesses):
        # Generate random initial guess
        h_initial = np.random.uniform(0, 1, n_points)
        total = np.sum(h_initial)
        scaling = (n_points / 2) / total
        h_initial *= scaling
        initial_guesses.append(h_initial)
    
    # Optimize each initial guess
    for h_initial in initial_guesses:
        # Run optimization
        start_time = time.time()
        result = minimize(objective, h_initial, method='L-BFGS-B', bounds=bounds, constraints=constraint, tol=1e-6)
        elapsed = time.time() - start_time
        elapsed_time += elapsed
        
        # If time permits, refine with tighter tolerance
        if elapsed < budget_s:
            result = minimize(objective, result.x, method='L-BFGS-B', bounds=bounds, constraints=constraint, tol=1e-8)
        
        current_c5 = evaluate_erdos_solution(result.x, n_points)
        if current_c5 < best_c5:
            best_h = result.x
            best_c5 = current_c5
        
        # Early exit if budget is exhausted
        if elapsed_time >= budget_s:
            break
    
    return best_h, best_c5, n_points
