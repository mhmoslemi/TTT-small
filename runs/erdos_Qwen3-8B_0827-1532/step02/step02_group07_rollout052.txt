def create_structured_initial_guess(n_points):
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * dx
        if (x >= 0 and x <= 1/3) or (x >= 2/3 and x <= 1.0) or (x >= 4/3 and x <= 5/3):

    return h

def objective(h, dx):

    max_corr = np.max(corr)
    return max_corr * dx
