from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Load initial guess and set parameters
    n_points = 100
    dx = 2.0 / n_points
    initial_h = initial_h_values  # Assume this is a global variable provided
    
    np.random.seed(seed)

    for _ in range(5):  # Try 5 different perturbations
        # Perturb the initial guess with small noise
        h_perturbed = initial_h + np.random.normal(0, 0.05, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        
        # Adjust to meet the integral constraint
        sum_h = np.sum(h_perturbed)
        adjustment = (n_points / 2) - sum_h
        h_perturbed[-1] += adjustment
        h_perturbed = np.clip(h_perturbed, 0, 1)

        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)

        # Constraints
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})

            objective,

            options={'ftol': 1e-9, 'maxiter': 500}
