import numpy as np
from scipy.optimize import minimize
from multiprocessing import Pool, cpu_count

def objective(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def evaluate_candidate(h, dx):
    return objective(h, dx)

def run(seed=42, budget_s=100, **kwargs):
    n_points = 50  # Reduced for faster evaluation
    dx = 2.0 / n_points
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    
    # Generate diverse initial guesses
    initial_guesses = []
    
    # Use provided initial_h_values scaled to 50 points
    initial_h = initial_h_values.copy()
    initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h)
    initial_sum = np.sum(initial_h)
    initial_h = initial_h * (n_points / 2) / initial_sum
    initial_guesses.append(initial_h)
    
    # Add a structured initial guess (two intervals of 0.5 length)
    structured_h = np.zeros(n_points)
    for i in range(n_points):
        x = i * dx
        if x < 0.5 or (x >= 1 and x < 1.5):
            structured_h[i] = 1.0
    structured_h = structured_h * (n_points / 2) / np.sum(structured_h)
    initial_guesses.append(structured_h)

        h_perturbed = h_perturbed / np.sum(h_perturbed) * (n_points / 2)
        initial_guesses.append(h_perturbed)
    
    # Use parallel processing
    num_cpus = min(cpu_count(), len(initial_guesses))
