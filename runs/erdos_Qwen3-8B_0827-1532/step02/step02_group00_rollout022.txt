def run(seed=42, budget_s=100, **kwargs):

    n_points = 200
    dx = 2.0 / n_points

    initial_guesses = []

    noise = np.random.normal(0, 0.1, n_points)
    h_perturb2 += noise

    adjustment = (n_points / 2 - sum_h) / dx

    # Random guess
