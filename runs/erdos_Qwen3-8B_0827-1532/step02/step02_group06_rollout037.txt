from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Increased for better resolution
    dx = 2.0 / n_points
    initial_guesses = []

    # Add initial_h_values
    initial_guesses.append(initial_h_values.copy())
    
    # Binary initial guess
    indices = np.random.choice(n_points, size=n_points // 2, replace=False)
    h_binary = np.zeros(n_points)
    h_binary[indices] = 1.0
    initial_guesses.append(h_binary)
    
    # Random initial guess
    h_random = np.random.uniform(0, 1, n_points)
    h_random = h_random * (n_points / 2) / np.sum(h_random)
    initial_guesses.append(h_random)
    
    # Perturbed initial guess
    h_perturbed = initial_h_values.copy()
    noise = np.random.normal(0, 0.05, n_points)
    h_perturbed += noise
    h_perturbed = np.clip(h_perturbed, 0, 1)
    initial_guesses.append(h_perturbed)
    
    # Remove duplicates
    unique_guesses = []
    seen = set()
    for guess in initial_guesses:
        if tuple(guess.round(6)) not in seen:
            seen.add(tuple(guess.round(6)))
            unique_guesses.append(guess)
    
    # Define objective function
    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define constraints
    constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = None
    best_c5 = float('inf')
    
    for guess in unique_guesses:
        try:
            result = minimize(
                objective,
                guess,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=constraint,
                options={'maxiter': 300, 'ftol': 1e-8, 'eps': 1e-8}
            )
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        except:
            pass  # Fallback to other guesses
