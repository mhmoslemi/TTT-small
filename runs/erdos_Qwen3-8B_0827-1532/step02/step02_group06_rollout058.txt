import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    n_points = 100
    dx = 2.0 / n_points
    np.random.seed(seed)
    
    # Generate diverse initial guesses
    initial_guesses = []

    indices = np.random.choice(n_points, size=n_points//2, replace=False)
    h_binary = np.zeros(n_points)
    h_binary[indices] = 1.0
    initial_guesses.append(h_binary)

    h_random = np.random.rand(n_points)

    h_perturbed = h_binary.copy()

    h_perturbed += perturb
    h_perturbed = np.clip(h_perturbed, 0, 1)
    initial_guesses.append(h_perturbed)
    
    # Add a smoothed version of binary guess
    h_smooth = np.zeros(n_points)
    h_smooth[indices] = 1.0
    h_smooth = np.convolve(h_smooth, np.ones(3)/3, mode='same')
    h_smooth = h_smooth / np.sum(h_smooth) * (n_points / 2)
    initial_guesses.append(h_smooth)
    
    best_h = None

    for h0 in initial_guesses:

            max_corr = np.max(corr)
