import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Resample initial_h_values to n_points
    original_x = np.linspace(0, 2, len(initial_h_values))
    new_x = np.linspace(0, 2, n_points)
    h_initial = np.interp(new_x, original_x, initial_h_values)
    sum_initial = np.sum(h_initial)
    scale_factor = (n_points / 2) / sum_initial
    h_initial *= scale_factor
    
    best_c5 = float('inf')
    best_h = h_initial.copy()
    start_time = time.time()
    
    # Generate multiple initial guesses with perturbations
    num_perturbations = 5
    for _ in range(num_perturbations):
        # Add noise and scale
        noise = np.random.normal(0, 0.05, n_points)
        h_perturbed = h_initial + noise
        h_perturbed = np.clip(h_perturbed, 0, 1)
        sum_perturbed = np.sum(h_perturbed)
        scale_factor = (n_points / 2) / sum_perturbed
        h_perturbed *= scale_factor
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Constraints
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Optimization
        result = minimize(
            objective,
            h_perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-8, 'maxiter': 300}
        )
        
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        
        if result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
