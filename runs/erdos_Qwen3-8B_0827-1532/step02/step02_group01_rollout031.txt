import numpy as np
from scipy.optimize import minimize

def objective(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

        h = h * (n_points / 2) / np.sum(h)
