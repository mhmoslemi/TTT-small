import numpy as np
from scipy.optimize import minimize, Bounds, LinearConstraint
import time

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h_values, 1 - h_values, mode='full')
    max_overlap = np.max(conv) * dx
    return max_overlap

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
    bounds = Bounds(0, 1)
    
    def objective(h):
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv) * dx
    
    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h, n_points)
    elapsed_time = 0
    
    # Generate multiple initial guesses
    num_initial_guesses = 10
    initial_guesses = [initial_h]
    for _ in range(num_initial_guesses - 1):
        random_guess = np.random.uniform(0, 1, n_points)
        random_guess = random_guess * (n_points / 2) / np.sum(random_guess)
        initial_guesses.append(random_guess)
    
    # Run optimization for each initial guess
    for initial_guess in initial_guesses:
        start_time = time.time()
        result = minimize(objective, initial_guess, method='SLSQP', bounds=bounds, constraints=constraint, tol=1e-6)
        elapsed = time.time() - start_time
        elapsed_time += elapsed
        
        # Refine with tighter tolerance if time permits
        if elapsed < budget_s:
            result = minimize(objective, result.x, method='SLSQP', bounds=bounds, constraints=constraint, tol=1e-8)
        
        current_c5 = evaluate_erdos_solution(result.x, n_points)
        if current_c5 < best_c5:
            best_h = result.x
            best_c5 = current_c5
        
        # Early exit if budget is exhausted
        if elapsed_time >= budget_s:
            break
    
    return best_h, best_c5, n_points
