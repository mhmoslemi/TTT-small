        h_scaled = h * scaling
        initial_guesses.append(h_scaled)
    
    # Define constraints and bounds
    constraint = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
    bounds = Bounds(0, 1)
    
    best_h = None
    best_c5 = float('inf')
    elapsed_time = 0
    
    for guess in initial_guesses:
        start_time = time.time()
        # Run optimization with tighter tolerances
        result = minimize(evaluate_erdos_solution, guess, method='SLSQP', bounds=bounds, constraints=constraint, tol=1e-6)
        elapsed = time.time() - start_time
        elapsed_time += elapsed
        
        # Refine with tighter tolerance if time allows
        if elapsed < budget_s:
            result = minimize(evaluate_erdos_solution, result.x, method='SLSQP', bounds=bounds, constraints=constraint, tol=1e-8)
        
        current_c5 = evaluate_erdos_solution(result.x, n_points)
        if current_c5 < best_c5:
            best_h = result.x.copy()
            best_c5 = current_c5
        
        # Early exit if budget is exhausted
        if elapsed_time >= budget_s:
            break
    
    return best_h, best_c5, n_points
