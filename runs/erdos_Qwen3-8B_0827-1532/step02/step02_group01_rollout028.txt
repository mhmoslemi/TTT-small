import numpy as np
from scipy.optimize import minimize

def create_initial_guess(n_points):
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    for i in range(5):
        start = i * 0.4
        start_idx = int(start / dx)
        end_idx = start_idx + int(0.2 / dx)
        h[start_idx:end_idx] = 1.0
    return h

def objective(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=100, **kwargs):

    dx = 2.0 / n_points
    h_initial = create_initial_guess(n_points)

    best_h = h_initial
    
    for i in range(num_starts):

        h_perturbed = np.clip(h_perturbed, 0, 1)
        # Adjust to meet the sum constraint
        current_sum = np.sum(h_perturbed)
        scale = (n_points / 2) / current_sum
        h_perturbed = h_perturbed * scale
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
