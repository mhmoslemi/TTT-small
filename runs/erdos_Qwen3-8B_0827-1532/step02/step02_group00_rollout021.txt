from scipy.optimize import minimize
import time

    num_perturbations = 5

        # Perturb the initial guess

        current_sum = np.sum(perturbed)
        desired_sum = n_points / 2

        perturbed[-1] += delta

        perturbed[-1] = np.clip(perturbed[-1], 0, 1)

        current_sum = np.sum(perturbed)
        delta = desired_sum - current_sum
        perturbed[-1] += delta
