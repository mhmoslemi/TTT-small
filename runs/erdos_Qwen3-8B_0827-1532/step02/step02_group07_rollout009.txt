def objective(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

    initial_sum = np.sum(initial_h)

    for _ in range(20):
        h_perturbed = np.random.rand(n_points)
        h_perturbed = h_perturbed / np.sum(h_perturbed) * (n_points / 2)
        initial_guesses.append(h_perturbed)
    # Now perform optimization for each initial guess
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    for guess in initial_guesses:
        # Perturb the guess slightly to avoid getting stuck in the same local minimum
        # Maybe add some noise
        perturbed_guess = guess + np.random.normal(0, 0.01, n_points)
        perturbed_guess = np.clip(perturbed_guess, 0, 1)
        # Define bounds and constraints
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
        # Run optimization
        result = minimize(objective, perturbed_guess, args=(dx,), method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        # Check if optimization was successful
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            # Fallback to perturbed guess
            current_c5 = objective(perturbed_guess, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = perturbed_guess
    return best_h, best_c5, n_points
