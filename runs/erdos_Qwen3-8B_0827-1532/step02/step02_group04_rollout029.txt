import numpy as np
from scipy.optimize import minimize, Bounds, LinearConstraint
import time

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h_values, 1 - h_values, mode='full')
    max_overlap = np.max(conv) * dx
    return max_overlap

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    initial_guesses = []
    initial_guesses.append(initial_h_values)
    
    # Generate 4 perturbed versions of initial guess
    for _ in range(4):
        perturbed = initial_h_values + np.random.uniform(-0.05, 0.05, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        target_sum = n_points / 2
        scale = target_sum / current_sum
        perturbed *= scale
        initial_guesses.append(perturbed)
    
    # Generate 5 random initial guesses
    for _ in range(5):
        random_guess = np.random.rand(n_points)
        current_sum = np.sum(random_guess)
        target_sum = n_points / 2
        scale = target_sum / current_sum
        random_guess *= scale
        initial_guesses.append(random_guess)
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    for guess in initial_guesses:
        def objective(h):
            conv = np.correlate(h, 1 - h, mode='full')
            return np.max(conv) * dx
        
        h_initial = guess.copy()
        constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
        bounds = Bounds(0, 1)
        
        # First optimization with SLSQP
        result = minimize(objective, h_initial, method='SLSQP', bounds=bounds, constraints=constraint, tol=1e-6)
        
        # Refinement with smaller tolerance
        elapsed = time.time() - start_time
        if elapsed < budget_s:
            result = minimize(objective, result.x, method='SLSQP', bounds=bounds, constraints=constraint, tol=1e-8)
        
        current_c5 = evaluate_erdos_solution(result.x, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x
    
    return best_h, best_c5, n_points
