import numpy as np
from scipy.optimize import minimize, Bounds, LinearConstraint
import time

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h_values, 1 - h_values, mode='full')
    max_overlap = np.max(conv) * dx
    return max_overlap

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
    bounds = Bounds(0, 1)
    
    def objective(h):
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv) * dx
    
    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h, n_points)
    elapsed_time = 0
    
    # Generate multiple initial guesses
    initial_guesses = []
    initial_guesses.append(initial_h)
    # Symmetric guess from initial_h
    symmetric_guess = initial_h.copy()
    for i in range(n_points // 2):
        symmetric_guess[i] = symmetric_guess[n_points - 1 - i]
    initial_guesses.append(symmetric_guess)
    # Random guess
    random_guess = np.random.uniform(0, 1, size=n_points)
    total = np.sum(random_guess)
    scaling_factor = (n_points / 2) / total
    random_guess *= scaling_factor
    initial_guesses.append(random_guess)
    # Another random guess
    another_random_guess = np.random.uniform(0, 1, size=n_points)
    total = np.sum(another_random_guess)
    scaling_factor = (n_points / 2) / total
    another_random_guess *= scaling_factor
    initial_guesses.append(another_random_guess)
    
    # Optimize each initial guess
    for guess in initial_guesses:
        start_time = time.time()
        # Use L-BFGS-B for faster convergence
        result = minimize(objective, guess, method='L-BFGS-B', bounds=bounds, constraints=constraint, tol=1e-6)
        elapsed = time.time() - start_time
        elapsed_time += elapsed
        
        # Refine with tighter tolerance if time permits
        if elapsed < budget_s:
            result = minimize(objective, result.x, method='L-BFGS-B', bounds=bounds, constraints=constraint, tol=1e-8)

        if current_c5 < best_c5:
            best_h = result.x

        if elapsed_time >= budget_s:
            break
    
    return best_h, best_c5, n_points
