import numpy as np
from scipy.optimize import minimize, LinearConstraint, Bounds
from concurrent.futures import ThreadPoolExecutor
import time

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h_values, 1 - h_values, mode='full')
    max_overlap = np.max(conv) * dx
    return max_overlap

def generate_initial_guess(n_points):
    num_ones = n_points // 2
    h = np.zeros(n_points)
    h[:num_ones] = 1
    np.random.shuffle(h)
    return h

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 50  # Using a smaller n_points for faster computation
    dx = 2.0 / n_points
    num_initial_guesses = 5  # Number of initial guesses to try
    constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
    bounds = Bounds(0, 1)
    
    def objective(h):
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv) * dx
    
    # Generate initial guesses
    initial_guesses = [generate_initial_guess(n_points) for _ in range(num_initial_guesses)]
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    # Use ThreadPoolExecutor to run optimizations in parallel
    with ThreadPoolExecutor(max_workers=10) as executor:
        futures = []
        for guess in initial_guesses:
            # Submit the optimization task
            future = executor.submit(minimize, objective, guess, method='SLSQP', bounds=bounds, constraints=constraint, tol=1e-6)
            futures.append(future)
        
        # Track elapsed time and collect results
        while futures:
            # Check if time is exceeded
            elapsed_time = time.time() - start_time
            if elapsed_time >= budget_s:
                break
            # Wait for the next result
            done, not_done = concurrent.futures.wait(futures, return_when=concurrent.futures.FIRST_COMPLETED)
            for future in done:
                result = future.result()
                current_h = result.x
                current_c5 = evaluate_erdos_solution(current_h, n_points)
                if current_c5 < best_c5:
                    best_h = current_h
                    best_c5 = current_c5
            # Update the list of futures
            futures = not_done
    
    return best_h, best_c5, n_points
