import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = 128  # Increased resolution for better optimization
    dx = 2.0 / n_points
    np.random.seed(seed)
    
    # Generate initial guesses
    initial_guesses = []
    initial_guesses.append(initial_h_values)  # Use provided initial guess
    # Random initial guess
    random_initial = np.zeros(n_points)
    indices = np.random.choice(n_points, size=n_points//2, replace=False)
    random_initial[indices] = 1
    initial_guesses.append(random_initial)
    # Structured initial guess (every other interval)
    structured_initial = np.zeros(n_points)
    structured_initial[::2] = 1
    initial_guesses.append(structured_initial)
    # Evenly spaced initial guess
    evenly_spaced_initial = np.zeros(n_points)
    for i in range(n_points//2):
        pos = int(i * n_points / (n_points//2))
        evenly_spaced_initial[pos] = 1
    initial_guesses.append(evenly_spaced_initial)
    # Perturbed initial guess based on initial_h_values
    perturbed_initial = initial_h_values.copy()
    perturbed_initial += np.random.normal(0, 0.05, n_points)
    perturbed_initial = np.clip(perturbed_initial, 0, 1)
    initial_guesses.append(perturbed_initial)
    
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    
    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    for guess in initial_guesses:
        # Use L-BFGS-B for faster convergence with bounds
        result = minimize(
            objective,
            guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraint,
            options={'maxiter': 500, 'ftol': 1e-8, 'eps': 1e-8}
        )
        h_opt, c5 = result.x, result.fun
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt
        # Early exit if time is exceeded
        if time.time() - start_time > budget_s:
            break
    
    return (best_h, best_c5, n_points)
