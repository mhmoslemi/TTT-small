import numpy as np
from scipy.optimize import minimize, Bounds, LinearConstraint
import time

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h_values, 1 - h_values, mode='full')
    max_overlap = np.max(conv) * dx
    return max_overlap

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values  # Assume this is available
    n_points = len(initial_h)
    dx = 2.0 / n_points
    constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
    bounds = Bounds(0, 1)
    
    def objective(h):
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv) * dx
    
    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h, n_points)
    elapsed_time = 0
    
    # Generate diverse initial guesses
    initial_guesses = []
    initial_guesses.append(initial_h)
    # Random initial guess
    h_rand = np.random.rand(n_points)
    h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
    initial_guesses.append(h_rand)
    # Perturbed versions of initial_h
    for _ in range(5):
        perturbed = initial_h + np.random.uniform(-0.1, 0.1, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scaling_factor = (n_points / 2) / current_sum
        perturbed *= scaling_factor
        initial_guesses.append(perturbed)
    
    # Optimize each initial guess
    for guess in initial_guesses:
        start_time = time.time()
        result = minimize(objective, guess, method='SLSQP', bounds=bounds, constraints=constraint, tol=1e-6)
        elapsed = time.time() - start_time
        elapsed_time += elapsed
        if elapsed_time >= budget_s:
            break
        current_c5 = evaluate_erdos_solution(result.x, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x
    
    # Final refinement with tighter tolerance
    if elapsed_time < budget_s:
        result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=constraint, tol=1e-8)
        current_c5 = evaluate_erdos_solution(result.x, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x
    
    return best_h, best_c5, n_points
