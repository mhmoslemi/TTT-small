from scipy.optimize import minimize, Bounds, LinearConstraint
import time

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h_values, 1 - h_values, mode='full')
    max_overlap = np.max(conv) * dx
    return max_overlap

def generate_structured_initial(n_points):
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    h[:int(0.2 * n_points)] = 1.0
    h[int(0.2 * n_points):int(0.2 * n_points + 0.5 * n_points)] = 0.0
    h[int(0.2 * n_points + 0.5 * n_points):] = 1.0
    return h

def run(seed=42, budget_s=100, **kwargs):

    n_points = 50

    constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
