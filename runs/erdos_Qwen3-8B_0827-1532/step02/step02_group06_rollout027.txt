from scipy.signal import correlate

        random_guess = np.random.uniform(0, 1, n_points)
        guesses.append(random_guess)

    # Define the objective function
    def objective(h):

        max_corr = np.max(corr)
        return max_corr * dx

    constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}

    elapsed_time = 0
