    initial_guesses.append(binary_guess_1)
    
    # Binary step function with four intervals
    binary_guess_2 = np.zeros(n_points)
    for i in range(n_points):
        if 0 <= i*dx <= 0.25 or 0.5 <= i*dx <= 0.75 or 1.0 <= i*dx <= 1.25 or 1.5 <= i*dx <= 1.75:
            binary_guess_2[i] = 1.0
    initial_guesses.append(binary_guess_2)
    
    # Evaluate each initial guess
    for guess in initial_guesses:
        guess = np.clip(guess, 0, 1)
        guess = guess * (target_sum / np.sum(guess))
        # Run optimization
        def objective(h):
            h_full = np.concatenate([h, [target_sum - np.sum(h)]])
            h_full[-1] = np.clip(h_full[-1], 0, 1)
            one_minus_h = 1.0 - h_full
            corr = np.correlate(h_full, one_minus_h, mode='full')
            max_corr = np.max(corr) * dx
            penalty = 0.0
            if h_full[-1] < 0 or h_full[-1] > 1:
                penalty = 1e6 * (abs(h_full[-1]) + 1 - h_full[-1])
            return max_corr + penalty
        
        bounds = [(0.0, 1.0) for _ in range(n_points - 1)]
        
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=100,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed
        )
        
        h_opt = result.x
        h_full = np.concatenate([h_opt, [target_sum - np.sum(h_opt)]])
        h_full = np.clip(h_full, 0, 1)
        current_sum = np.sum(h_full)
        h_full = h_full * (target_sum / current_sum)
        
        one_minus_h = 1.0 - h_full
        corr = np.correlate(h_full, one_minus_h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_full
    
    # Additional perturbations to refine the solution
    for _ in range(100):
        if time.time() - start_time >= budget_s:
            break
        noise = np.random.uniform(-0.05, 0.05, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        def objective_perturbed(h):
            h_full = np.concatenate([h, [target_sum - np.sum(h)]])
            h_full[-1] = np.clip(h_full[-1], 0, 1)
            one_minus_h = 1.0 - h_full
            corr = np.correlate(h_full, one_minus_h, mode='full')
            max_corr = np.max(corr) * dx
            penalty = 0.0
            if h_full[-1] < 0 or h_full[-1] > 1:
                penalty = 1e6 * (abs(h_full[-1]) + 1 - h_full[-1])
            return max_corr + penalty
        
        bounds_perturbed = [(0.0, 1.0) for _ in range(n_points - 1)]
        
        result_perturbed = differential_evolution(
            objective_perturbed,
            bounds_perturbed,
            strategy='best1bin',
            maxiter=50,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed
        )
        
        h_opt_perturbed = result_perturbed.x
        h_full_perturbed = np.concatenate([h_opt_perturbed, [target_sum - np.sum(h_opt_perturbed)]])
        h_full_perturbed = np.clip(h_full_perturbed, 0, 1)
        current_sum_perturbed = np.sum(h_full_perturbed)
        h_full_perturbed = h_full_perturbed * (target_sum / current_sum_perturbed)
        
        one_minus_h_perturbed = 1.0 - h_full_perturbed
        corr_perturbed = np.correlate(h_full_perturbed, one_minus_h_perturbed, mode='full')
        c5_perturbed = np.max(corr_perturbed) * dx
        
        if c5_perturbed < best_c5:
            best_c5 = c5_perturbed
            best_h = h_full_perturbed
    
    return best_h, best_c5, n_points
