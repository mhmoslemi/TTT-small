from scipy.optimize import minimize
import time

    n_points = len(initial_guess)

    initial_guesses = [initial_guess.copy()]
    
    # Binary patterns with different periodicities
    for m in [3, 4, 5, 6, 7]:
        binary = np.zeros(n_points)
        for i in range(n_points):
            if i % m == 0:
                binary[i] = 1.0
        binary_sum = np.sum(binary)
        scale = target_sum / binary_sum
        binary = binary * scale
        initial_guesses.append(binary)
    
    # Sinusoidal pattern
    sinusoidal = 0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / n_points)
    sinusoidal_sum = np.sum(sinusoidal)
    scale = target_sum / sinusoidal_sum
    sinusoidal = sinusoidal * scale
    initial_guesses.append(sinusoidal)
    
    # Random binary patterns
    for _ in range(3):
        random_binary = np.random.rand(n_points)
        random_binary = (random_binary < 0.5).astype(float)
        random_binary_sum = np.sum(random_binary)
        scale = target_sum / random_binary_sum
        random_binary = random_binary * scale
        initial_guesses.append(random_binary)
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Objective function to minimize C5
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization with tighter tolerances and more iterations
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 1000}
        )
        
        # Check if this result is better
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check elapsed time
        elapsed = time.time() - start_time
