from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):

    dx = 2.0 / n_points
    target_sum = n_points / 2

    start_time = time.time()

    def objective(h):

        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        penalty = 1e4 * (np.sum(h) - target_sum)**2
        return max_corr * dx + penalty
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Run differential evolution with initial guess
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=15,
        maxiter=100,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed
    )
    
    if result.success:
        best_c5 = result.fun
        best_h = result.x
    
    # Generate perturbations based on best_h
    num_perturbations = 50
    for perturbation_idx in range(num_perturbations):
        # Generate perturbations based on best_h
        noise = np.random.uniform(-0.1, 0.1, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function with penalty
        def objective_perturbed(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            penalty = 1e4 * (np.sum(h) - target_sum)**2
            return max_corr * dx + penalty
        
        # Run differential evolution on perturbed guess
        result_perturbed = differential_evolution(
