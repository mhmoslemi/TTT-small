import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    
    start_time = time.time()
    
    # Generate structured initial guesses
    structured_guesses = []
    # Alternating 1 and 0
    alt_guess = np.zeros(n_points)
    alt_guess[::2] = 1.0
    structured_guesses.append(alt_guess)
    # First half 1, second half 0
    first_half_guess = np.zeros(n_points)
    first_half_guess[:n_points//2] = 1.0
    structured_guesses.append(first_half_guess)
    # Second half 1, first half 0
    second_half_guess = np.zeros(n_points)
    second_half_guess[n_points//2:] = 1.0
    structured_guesses.append(second_half_guess)
    # Random structured guesses
    for _ in range(3):
        guess = np.random.choice([0, 1], size=n_points)
        guess = guess * (target_sum / np.sum(guess))
        structured_guesses.append(guess)
    
    # Initial perturbations: structured and random
    perturbations = []
    for guess in structured_guesses:
        perturbations.append(guess)
    for _ in range(5):
        guess = np.random.uniform(0, 1, n_points)
        guess = guess * (target_sum / np.sum(guess))
        perturbations.append(guess)
    perturbations.append(initial_h_values)
    
    num_perturbations = len(perturbations)
    
    for perturbation_idx in range(num_perturbations):
        if perturbation_idx < len(perturbations):
            perturbed = perturbations[perturbation_idx].copy()
        else:
            # Perturb best_h with varying noise
            if perturbation_idx < 30:
                noise = np.random.uniform(-0.2, 0.2, n_points)
            elif perturbation_idx < 60:
                noise = np.random.uniform(-0.1, 0.1, n_points)
            else:
                noise = np.random.uniform(-0.05, 0.05, n_points)
            perturbed = best_h + noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
        
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-12, 'maxiter': 500}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
