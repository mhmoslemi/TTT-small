import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    initial_guess[indices] = 1
    best_h = initial_guess.copy()
    
    # Function to evaluate the overlap integral
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    # Constraints: sum(h) == target_sum
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization parameters
    num_perturbations = 200
    for perturbation_idx in range(num_perturbations):
        # Perturb the current best solution
        noise = np.random.uniform(-0.1, 0.1, n_points)
