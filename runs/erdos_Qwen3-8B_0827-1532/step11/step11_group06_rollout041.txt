    initial_guesses = [initial_guess, periodic_h, random_binary]
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Define objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define bounds and constraints
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    
    # Optimize each initial guess
    for h_initial in initial_guesses:
        result = minimize(
            objective,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 2000}
        )
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    # Further refine with perturbations of the best solution
    num_perturbations = 5
    for _ in range(num_perturbations):
        perturbation = np.random.uniform(-0.1, 0.1, n_points)
        perturbed = best_h + perturbation
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 2000}
        )
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
