    structured_h3 = np.zeros(n_points)
    for i in range(n_points):
        if i % 4 == 0:
            structured_h3[i] = 1.0
        else:
            structured_h3[i] = 0.0
    
    structured_h4 = np.zeros(n_points)
    for i in range(n_points):
        if i % 5 == 0:
            structured_h4[i] = 1.0
        else:
            structured_h4[i] = 0.0

            best_h = result.x
