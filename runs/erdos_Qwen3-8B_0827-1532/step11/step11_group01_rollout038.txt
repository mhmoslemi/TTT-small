    initial_guesses.append(initial_h_values.copy())
    
    # Add structured patterns
    structured_h = np.zeros(n_points)
    for i in range(n_points):
        if i % 2 == 0:
            structured_h[i] = 1.0
        else:
            structured_h[i] = 0.0
    initial_guesses.append(structured_h)
    
    structured_h2 = np.zeros(n_points)
    for i in range(n_points):
        if i % 3 == 0:
            structured_h2[i] = 1.0
        else:
            structured_h2[i] = 0.0
    initial_guesses.append(structured_h2)
    
    structured_h3 = np.zeros(n_points)
    for i in range(n_points):
        if i % 4 == 0:
            structured_h3[i] = 1.0
        else:
            structured_h3[i] = 0.0
    initial_guesses.append(structured_h3)
    
    structured_h4 = np.zeros(n_points)
    for i in range(n_points):
        if i % 5 == 0:
            structured_h4[i] = 1.0
        else:
            structured_h4[i] = 0.0
    initial_guesses.append(structured_h4)
    
    # Add random binary sequences
    np.random.seed(seed)
    for _ in range(5):
        indices = np.random.choice(n_points, size=int(target_sum), replace=False)
        h_rand = np.zeros(n_points)
        h_rand[indices] = 1.0
        initial_guesses.append(h_rand)
    
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    
    start_time = time.time()
    
    for guess in initial_guesses:
        # Objective function to minimize C5
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization with tighter tolerances and more iterations
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 1500}
        )
        
        # Check if this result is better
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
