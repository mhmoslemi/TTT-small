    # Generate a few random initial guesses
    num_random_guesses = 5

        current_sum = np.sum(random_guess)

        scaled_guess = random_guess * scale

        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(

            scaled_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
