    # Generate perturbations of the initial guess
    np.random.seed(seed)
    perturbations = []
    for _ in range(5):  # generate 5 perturbations
        noise = np.random.uniform(-0.1, 0.1, n_points)
        perturbed = initial_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        perturbations.append(perturbed)
    
    # Combine all initial guesses
    initial_guesses = [initial_h] + structured_guesses + random_guesses + perturbations
    
    best_c5 = float('inf')
    best_h = initial_h.copy()

    for guess in initial_guesses:
        # Objective function to minimize C5
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization with tighter tolerances and more iterations
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,

        )
