import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate diverse initial guesses: original, random, and binary
    initial_guesses = [initial_guess.copy()]
    # Add random guesses
    for _ in range(2):
        guess = np.random.uniform(0, 1, n_points)
        guess = guess * (target_sum / np.sum(guess))
        initial_guesses.append(guess)
    # Add binary guesses (alternating 0 and 1)
    binary_guess = np.zeros(n_points)
    binary_guess[::2] = 1  # alternate 1 and 0
    binary_guess = binary_guess * (target_sum / np.sum(binary_guess))
    initial_guesses.append(binary_guess)
    
    # Optimization parameters
    max_iter_per_run = 200
    for guess in initial_guesses:
        # Scale to meet the sum constraint
        guess = guess * (target_sum / np.sum(guess))
        # Define objective
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Use COBYLA method for better constraint handling
        result = minimize(
            objective,
            guess,
            method='COBYLA',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': max_iter_per_run, 'disp': False}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
