from scipy.optimize import differential_evolution, minimize

def run(seed=42, budget_s=100, **kwargs):

        # Penalty for sum constraint
        penalty = 1e4 * (np.sum(h) - target_sum)**2
        return max_corr + penalty
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial differential evolution run
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=15,
        maxiter=100,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed
    )
    
    if result.success:
        best_h = result.x
        best_c5 = result.fun
    
    # Refinement with local optimization
    def refine_objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * dx
