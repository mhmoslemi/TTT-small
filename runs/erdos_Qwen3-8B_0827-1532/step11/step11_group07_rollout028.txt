import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    start_time = time.time()
    
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        penalty = 1e4 * (np.sum(h) - target_sum)**2
        return max_corr * dx + penalty
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    initial_guesses = [
        initial_guess,
        np.full(n_points, 0.5),
        np.concatenate([np.ones(n_points//2), np.zeros(n_points//2)])[:(n_points)],
        np.zeros(n_points),
    ]
    
    for guess in initial_guesses:
        try:
            result_de = differential_evolution(
                objective,
                bounds,
                strategy='best1bin',
                popsize=15,
                maxiter=100,
                tol=1e-6,
                mutation=(0.5, 1),
                recombination=0.8,
                seed=seed
            )
            de_h = result_de.x
            def objective_slsqp(h):
                one_minus_h = 1.0 - h
                corr = np.correlate(h, one_minus_h, mode='full')
                max_corr = np.max(corr)
                return max_corr * dx
            
            cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
            bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]
            result_slsqp = minimize(
                objective_slsqp,
                de_h,
                method='SLSQP',
                bounds=bounds_slsqp,
                constraints=cons,
                options={'ftol': 1e-10, 'maxiter': 150}
            )
            if result_slsqp.success and result_slsqp.fun < best_c5:
                best_c5 = result_slsqp.fun
                best_h = result_slsqp.x
        except:
            pass
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
