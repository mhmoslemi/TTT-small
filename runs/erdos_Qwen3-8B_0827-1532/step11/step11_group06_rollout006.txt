    for i in range(n_points):

        else:
            structured_h[i] = 0.0
    
    # Generate random binary initial guess

    random_binary = (random_binary < 0.5).astype(float)
    random_binary *= target_sum / np.sum(random_binary)
    
    # Combine initial guesses
    initial_guesses = [initial_guess, structured_h, random_binary]
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Ensure the guess is valid
        guess = np.clip(guess, 0, 1)
        current_sum = np.sum(guess)
        guess = guess * (target_sum / current_sum)
        
        # Objective function to minimize C5
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization with tighter tolerances and more iterations
        result = minimize(
