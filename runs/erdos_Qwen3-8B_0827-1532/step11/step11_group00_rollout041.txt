        initial_guesses.append(block_pattern)
    
    # Add random binary guesses
    for _ in range(5):
        np.random.seed(seed + len(initial_guesses))

    for guess in initial_guesses:

        def objective(h):
            one_minus_h = 1.0 - h

            max_corr = np.max(corr)

        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
