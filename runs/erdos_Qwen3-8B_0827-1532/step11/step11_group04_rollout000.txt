import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate diverse initial guesses
    initial_guesses = []
    initial_guesses.append(initial_guess)  # Original guess
    # Uniform distribution
    uniform_guess = np.full(n_points, 0.5)
    initial_guesses.append(uniform_guess)
    # Step function
    step_guess = np.concatenate([np.ones(n_points//2), np.zeros(n_points//2)])
    initial_guesses.append(step_guess)
    # Sinusoidal pattern
    sinusoidal_guess = 0.5 + 0.5 * np.sin(np.pi * np.arange(n_points) / n_points)
    initial_guesses.append(sinusoidal_guess)
    # Random guesses
    for _ in range(3):
        rand_guess = np.random.uniform(0, 1, n_points)
        rand_guess = rand_guess * (target_sum / np.sum(rand_guess))
        initial_guesses.append(rand_guess)
    
    # Perturbations: start with initial guesses and add noise
    perturbations = initial_guesses
    
    for perturbation_idx in range(len(perturbations)):
        perturbed = perturbations[perturbation_idx].copy()
        # Add noise with decreasing magnitude
        noise_scale = 0.1 if perturbation_idx < len(perturbations)//2 else 0.05
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed += noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 300}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
