import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Predefined initial guess
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate diverse initial guesses
    initial_guesses = []
    # Add initial_h_values
    initial_guesses.append(initial_guess)
    
    # Add structured guesses
    # 1. Alternating 1s and 0s
    structured_guess = np.zeros(n_points)
    for i in range(n_points):
        if i % 2 == 0:
            structured_guess[i] = 1.0
    initial_guesses.append(structured_guess)
    
    # 2. Evenly spaced 1s (every other sample)
    even_guess = np.zeros(n_points)
    for i in range(n_points):
        if i % 2 == 0:
            even_guess[i] = 1.0
    initial_guesses.append(even_guess)
    
    # 3. Random binary guesses
    for _ in range(5):
        np.random.seed(seed + len(initial_guesses))
        h = np.zeros(n_points)
        indices = np.random.choice(n_points, size=int(n_points//2), replace=False)
        h[indices] = 1.0
        initial_guesses.append(h)
    
    # 4. Additional structured guesses with different patterns
    # 4a. Block pattern with 2 samples of 1 followed by 2 of 0
    block_guess = np.zeros(n_points)
    for i in range(n_points):
        if i % 4 == 0 or i % 4 == 1:
            block_guess[i] = 1.0
    initial_guesses.append(block_guess)
    
    # 4b. Block pattern with 3 samples of 1 followed by 1 of 0
    block_guess2 = np.zeros(n_points)
    for i in range(n_points):
        if i % 4 == 0 or i % 4 == 1 or i % 4 == 2:
            block_guess2[i] = 1.0
    initial_guesses.append(block_guess2)
    
    best_c5 = float('inf')
    best_h = initial_guesses[0].copy()
    
    start_time = time.time()
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds and constraints
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization with L-BFGS-B
        result = minimize(
            objective,
            guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-12, 'maxiter': 1000}
        )
        
        # Check if successful
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
