import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate structured initial guesses
    structured_guesses = []
    
    # Guess 1: First half 1s, second half 0s
    guess1 = np.zeros(n_points)
    guess1[:n_points//2] = 1.0
    guess1 = guess1 * (target_sum / np.sum(guess1))
    structured_guesses.append(guess1)
    
    # Guess 2: Three equal parts
    guess2 = np.zeros(n_points)
    guess2[:n_points//3] = 1.0
    guess2[n_points//3:2*n_points//3] = 0.0
    guess2[2*n_points//3:] = 1.0
    guess2 = guess2 * (target_sum / np.sum(guess2))
    structured_guesses.append(guess2)
    
    # Guess 3: Alternating 1s and 0s
    guess3 = np.zeros(n_points)
    guess3[::2] = 1.0
    guess3 = guess3 * (target_sum / np.sum(guess3))
    structured_guesses.append(guess3)
    
    # Add random guesses
    random_guesses = [np.random.uniform(0, 1, n_points) for _ in range(3)]
    for guess in random_guesses:
        guess = guess * (target_sum / np.sum(guess))
        structured_guesses.append(guess)
    
    # Combine structured guesses and initial guess
    perturbations = structured_guesses + [initial_guess]
    
    num_perturbations = min(100, len(perturbations))  # Reduce number of perturbations for speed
    
    for perturbation_idx in range(num_perturbations):
        if perturbation_idx < len(perturbations):
            perturbed = perturbations[perturbation_idx].copy()
        else:
            # Perturb best_h with varying noise
            noise = np.random.uniform(-0.05, 0.05, n_points)
            perturbed = best_h + noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
        
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-12, 'maxiter': 200}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
