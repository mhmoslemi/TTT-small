import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    
    start_time = time.time()
    
    # Generate multiple initial guesses
    initial_guesses = []
    # Add the provided initial guess
    initial_guesses.append(initial_h_values.copy())
    # Add a random initial guess
    random_initial = np.random.uniform(0, 1, n_points)
    current_sum = np.sum(random_initial)
    scale = target_sum / current_sum
    random_initial = random_initial * scale
    initial_guesses.append(random_initial)
    # Add a structured initial guess (first half 1, second half 0)
    structured_initial = np.zeros(n_points)
    structured_initial[:n_points//2] = 1.0
    initial_guesses.append(structured_initial)
    
    # Optimize each initial guess
    for guess in initial_guesses:
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 500}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Perform perturbations on the best_h found so far
    elapsed = time.time() - start_time
    remaining_time = budget_s - elapsed
    if remaining_time > 0:
        num_perturbations = int(remaining_time / 0.1)  # 100ms per perturbation
        for _ in range(num_perturbations):
            # Random perturbation
            noise = np.random.uniform(-0.05, 0.05, n_points)
            perturbed = best_h + noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
            
            def objective(h):
                one_minus_h = 1.0 - h
                corr = np.correlate(h, one_minus_h, mode='full')
                return np.max(corr) * dx
            
            bounds = [(0.0, 1.0) for _ in range(n_points)]
            cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
            
            result = minimize(
                objective,
                perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'ftol': 1e-10, 'maxiter': 200}
            )
            
            if result.success and result.fun < best_c5:
                best_c5 = result.fun
                best_h = result.x
    
    return best_h, best_c5, n_points
