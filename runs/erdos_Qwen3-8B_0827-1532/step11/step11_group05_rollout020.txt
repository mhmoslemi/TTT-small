import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    # Start with initial guess if available, else generate random
    if 'initial_h_values' in globals():
        initial_guess = initial_h_values
    else:
        initial_guess = np.random.uniform(0, 1, 100)
        target_sum = 100 / 2
        initial_guess = initial_guess * target_sum / np.sum(initial_guess)
    
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Use a more aggressive exploration strategy
    num_perturbations = 300  # Increased for better exploration
    perturbation_types = [
        ('random', 0.2, 0.2),  # Random noise
        ('local', 0.05, 0.05),  # Small local perturbations
        ('global', 0.1, 0.1),   # Larger global perturbations
    ]
    
    for perturbation_idx in range(num_perturbations):
        # Select perturbation type based on iteration
        pert_type, noise_low, noise_high = perturbation_types[perturbation_idx % 3]
        
        if pert_type == 'random':
            # Random initial guess
            random_noise = np.random.uniform(-noise_low, noise_high, n_points)
            perturbed = initial_guess + random_noise
        elif pert_type == 'local':
            # Perturb best_h with small noise
            noise = np.random.uniform(-noise_low, noise_high, n_points)
            perturbed = best_h + noise
        elif pert_type == 'global':
            # Perturb best_h with larger noise
            noise = np.random.uniform(-noise_low, noise_high, n_points)
            perturbed = best_h + noise
        
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 300}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
