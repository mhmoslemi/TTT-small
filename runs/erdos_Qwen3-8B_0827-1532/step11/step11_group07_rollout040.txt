import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        penalty = 1e4 * np.abs(np.sum(h) - target_sum)
        return max_corr * dx + penalty
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def optimize_with_de(initial_pop):
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=15,
            mutation=(0.5, 1),
            recombination=0.8,
            tol=1e-10,
            maxiter=100,
            polish=True,
            vectorized=True,
            init=initial_pop
        )
        if result.success:
            return result.x, objective(result.x)
        return best_h, best_c5
    
    # Initial optimization with provided guess
    best_h, best_c5 = optimize_with_de(initial_guess)
    
    # Generate random initial populations
    for _ in range(3):
        random_guess = np.random.uniform(0, 1, n_points)
        random_guess = random_guess * (target_sum / np.sum(random_guess))
        new_h, new_c5 = optimize_with_de(random_guess)
        if new_c5 < best_c5:
            best_h, best_c5 = new_h, new_c5
    
    # Check elapsed time
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
