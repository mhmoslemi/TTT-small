from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    
    start_time = time.time()
    
    def generate_initial_guesses():
        guesses = [initial_h_values.copy()]
        # Random initial guess
        random_guess = np.random.uniform(0, 1, n_points)
        random_guess = random_guess * (target_sum / np.sum(random_guess))
        guesses.append(random_guess)
        # Single block
        block_length = n_points // 2
        block_start = n_points // 4
        block_end = block_start + block_length
        single_block = np.zeros(n_points)
        single_block[block_start:block_end] = 1.0
        guesses.append(single_block)
        # Two blocks
        block_length = n_points // 4
        block1_start = n_points // 4
        block1_end = block1_start + block_length
        block2_start = 3 * n_points // 4
        block2_end = block2_start + block_length
        two_blocks = np.zeros(n_points)
        two_blocks[block1_start:block1_end] = 1.0
        two_blocks[block2_start:block2_end] = 1.0
        guesses.append(two_blocks)
        return guesses
    
    initial_guesses = generate_initial_guesses()
    
    for initial_guess in initial_guesses:
        current_h = initial_guess.copy()
        current_sum = np.sum(current_h)
        scale = target_sum / current_sum
        current_h = current_h * scale
        
        for perturbation_idx in range(500):
            if perturbation_idx < 100:
                noise = np.random.uniform(-0.2, 0.2, n_points)
            elif perturbation_idx < 300:
                noise = np.random.uniform(-0.1, 0.1, n_points)
            else:
                noise = np.random.uniform(-0.05, 0.05, n_points)
            perturbed = current_h + noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
            
            def objective(h):
                one_minus_h = 1.0 - h
                corr = np.correlate(h, one_minus_h, mode='full')
                return np.max(corr) * dx
            
            bounds = [(0.0, 1.0) for _ in range(n_points)]
            cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
            
            result = minimize(
                objective,
                perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'ftol': 1e-10, 'maxiter': 200}
            )
            
            if result.success and result.fun < best_c5:
                best_c5 = result.fun
                best_h = result.x
            
            elapsed = time.time() - start_time
            if elapsed >= budget_s:
                break
    
    return best_h, best_c5, n_points
