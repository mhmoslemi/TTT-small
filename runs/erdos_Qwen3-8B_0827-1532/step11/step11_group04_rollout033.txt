from scipy.optimize import minimize
import time

    initial_guesses = [initial_guess]

    structured_guesses = []

    block_guess = np.zeros(n_points)
    block_size = n_points // 4
    block_guess[:block_size] = 1.0
    structured_guesses.append(block_guess)
    # 2. Alternating blocks
    alt_guess = np.zeros(n_points)
    for i in range(n_points):
        if i % 2 == 0:
            alt_guess[i] = 1.0
    structured_guesses.append(alt_guess)
    # 3. Randomized structured
    for _ in range(2):
        rand_struct = np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])
        rand_struct = rand_struct * (target_sum / np.sum(rand_struct))
        structured_guesses.append(rand_struct)
    initial_guesses.extend(structured_guesses)
    # Add random guesses
    for _ in range(5):
        rand_guess = np.random.uniform(0, 1, n_points)
        rand_guess = rand_guess * (target_sum / np.sum(rand_guess))
        initial_guesses.append(rand_guess)
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Scale to meet the sum constraint
        guess = guess * (target_sum / np.sum(guess))
        # Optimization
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 300}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
