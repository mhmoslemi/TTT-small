            scale = target_sum / current_sum
            perturbed = perturbed * scale
        
        # Run differential evolution on the perturbed guess
        result_perturb = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=15,
            tol=1e-10,
            mutation=(0.5, 1),
            recombination=0.8,
            maxiter=200,
            seed=seed + perturbation_idx
        )
        
        # Check if this result is better
        if result_perturb.success and result_perturb.fun < best_c5:
            best_c5 = result_perturb.fun
            best_h = result_perturb.x
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
