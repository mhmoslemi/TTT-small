import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    
    start_time = time.time()
    
    # Use initial_h_values as the starting guess
    initial_guess = initial_h_values.copy()
    initial_guess = np.clip(initial_guess, 0, 1)
    initial_guess = initial_guess * (target_sum / np.sum(initial_guess))
    
    # Define the objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    
    # First optimization with COBYLA
    result = minimize(
        objective,
        initial_guess,
        method='COBYLA',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 5000, 'ftol': 1e-10}
    )
    
    if result.success and result.fun < best_c5:
        best_c5 = result.fun
        best_h = result.x
    
    # Perturbations
    for _ in range(100):  # More perturbations
        if time.time() - start_time >= budget_s:

        indices = np.random.choice(n_points, size=10, replace=False)
        perturbed = best_h.copy()
        for i in indices:
            perturbed[i] += np.random.uniform(-0.05, 0.05)
            perturbed[i] = np.clip(perturbed[i], 0, 1)
        current_sum = np.sum(perturbed)

        perturbed = perturbed * scale
        
        # Optimize the perturbed guess
        result = minimize(
            objective,
            perturbed,
            method='COBYLA',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 2000, 'ftol': 1e-10}
        )
