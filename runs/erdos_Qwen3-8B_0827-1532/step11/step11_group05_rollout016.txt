import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate a few random initial guesses
    random_initials = []
    for _ in range(3):
        random_h = np.random.uniform(0, 1, n_points)
        random_h = random_h * (target_sum / np.sum(random_h))
        random_initials.append(random_h)
    
    # Combine initial guesses
    initial_guesses = [initial_guess] + random_initials
    
    # Optimization parameters
    max_perturbations = 200
    for perturbation_idx in range(max_perturbations):
        # Select a random initial guess
        current_guess = initial_guesses[np.random.randint(len(initial_guesses))]
        
        # Perturb the guess
        noise_scale = 0.1 if perturbation_idx < 50 else 0.05
        perturbed = current_guess + np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * (target_sum / np.sum(perturbed))
        
        # Define objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization constraints
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Use trust-constr for better convergence
        result = minimize(
            objective,
            perturbed,
            method='trust-constr',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 300, 'ftol': 1e-9, 'gtol': 1e-9}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Early stopping if time is up
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
