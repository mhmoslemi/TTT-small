import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate diverse initial guesses
    initial_guesses = [
        initial_guess.copy(),
        np.random.uniform(0, 1, n_points),
        np.random.choice([0, 1], size=n_points, p=[0.5, 0.5]),
        np.random.normal(0.5, 0.1, n_points)
    ]
    
    # Scale initial guesses to meet the sum constraint
    scaled_guesses = []
    for guess in initial_guesses:
        current_sum = np.sum(guess)
        scale = target_sum / current_sum
        scaled = np.clip(guess * scale, 0, 1)
        scaled_guesses.append(scaled)
    
    # Optimization parameters
    max_perturbations = 200
    for perturbation_idx in range(max_perturbations):
        # Select a random initial guess
        if perturbation_idx < len(scaled_guesses):
            current_guess = scaled_guesses[perturbation_idx]
        else:
            # Random perturbation of best_h
            noise = np.random.uniform(-0.1, 0.1, n_points)
            current_guess = best_h + noise
            current_guess = np.clip(current_guess, 0, 1)
            current_sum = np.sum(current_guess)
            scale = target_sum / current_sum
            current_guess = current_guess * scale
        
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            current_guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 300}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
