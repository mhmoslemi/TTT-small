import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

            scale = 0.0

        scaled_h = h * scale

    result_de = differential_evolution(
        objective,
        bounds,
        popsize=15,
        mutation=(0.5, 1.0),
        recombination=0.9,
        strategy='best1bin',
        tol=1e-10,
        maxiter=500,
        polish=True,
        seed=seed
    )
    
    if result_de.success:
        best_h = result_de.x
        best_c5 = result_de.fun
    
    # Refine with SLSQP if time remains

            if current_sum == 0:
                scale = 0.0
            else:
                scale = target_sum / current_sum
