    structured_guess[0:int(n_points/4)] = 1.0  # 0.5 in domain
    structured_guess[int(n_points/2):int(3*n_points/4)] = 1.0  # another 0.5
    initial_guesses.append(structured_guess)
    
    # Add structured guess with three intervals of 0.333 each
    structured_guess_3 = np.zeros(n_points)
    structured_guess_3[0:int(n_points/3)] = 1.0  # 0.333 in domain
    structured_guess_3[int(2*n_points/3):n_points] = 1.0  # another 0.333
    structured_guess_3[int(n_points/6):int(n_points/2)] = 1.0  # third 0.333
    initial_guesses.append(structured_guess_3)
    
    # Add random binary guesses
    for _ in range(5):
        np.random.seed(seed + len(initial_guesses))
        h = np.zeros(n_points)
        indices = np.random.choice(n_points, size=int(n_points//2), replace=False)
        h[indices] = 1.0
        initial_guesses.append(h)
    
    best_c5 = float('inf')
    best_h = initial_guesses[0].copy()
    
    start_time = time.time()
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds and constraints
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 5000}
        )
        
        # Check if successful
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
