    structured_h4 = np.zeros(n_points)

            structured_h4[i] = 0.0
    
    structured_h5 = np.zeros(n_points)
    for i in range(n_points):

            structured_h5[i] = 1.0

            structured_h5[i] = 0.0

    for noise_scale in [0.05, 0.1, 0.15]:

            perturbed = initial_guess + noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
            random_guesses.append(perturbed)
    
    # Combine all initial guesses
    initial_guesses = [initial_guess, structured_h, structured_h2, structured_h3, structured_h4, structured_h5] + random_guesses

    start_time = time.time()
    
    for guess in initial_guesses:
        # Objective function to minimize C5
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
