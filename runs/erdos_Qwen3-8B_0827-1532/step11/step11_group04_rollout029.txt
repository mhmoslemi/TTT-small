import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate structured initial guesses
    structured_guesses = []
    # Half-ones and half-zeros
    half_ones = np.zeros(n_points)
    half_ones[:n_points//2] = 1.0
    structured_guesses.append(half_ones)
    # Alternating 1 and 0
    alt_guess = np.zeros(n_points)
    alt_guess[::2] = 1.0
    structured_guesses.append(alt_guess)
    # Randomly shuffled ones and zeros
    shuffled_guess = np.random.permutation([1.0]*int(n_points/2) + [0.0]*int(n_points/2))
    structured_guesses.append(shuffled_guess)
    # Add initial guess
    structured_guesses.append(initial_guess)
    
    # Generate random guesses
    random_guesses = [np.random.uniform(0, 1, n_points) for _ in range(2)]
    for guess in random_guesses:
        guess = guess * (target_sum / np.sum(guess))
        structured_guesses.append(guess)
    
    # Perturbations
    perturbations = structured_guesses
    
    for perturbation_idx in range(len(perturbations)):
        perturbed = perturbations[perturbation_idx].copy()
        # Scale to meet the sum constraint
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        # Clip to [0,1]
        perturbed = np.clip(perturbed, 0, 1)
        
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Use COBYLA for derivative-free optimization
        result = minimize(
            objective,
            perturbed,
            method='COBYLA',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'disp': False}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
