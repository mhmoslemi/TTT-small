    initial_guesses.append(initial_guess)
    
    # Add a structured binary initial guess
    structured_guess = np.zeros(n_points)
    structured_guess[:int(n_points/4)] = 1.0
    structured_guess[int(n_points/2)-int(n_points/4):int(n_points/2)] = 1.0
    structured_guess[int(3*n_points/4):] = 1.0
    initial_guesses.append(structured_guess)
    
    # Add some random guesses
    for _ in range(3):
        random_guess = np.random.uniform(0, 1, n_points)
        random_guess = random_guess * (target_sum / np.sum(random_guess))
        initial_guesses.append(random_guess)
    
    # Parameters for differential evolution
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    for guess in initial_guesses:
        # Define objective function with penalty for sum constraint
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr) * dx
            penalty = 1e4 * abs(np.sum(h) - target_sum)
            return max_corr + penalty
        
        # Run differential evolution
        result = differential_evolution(
            objective,
            bounds,
            popsize=15,
            mutation=0.5,
            recombination=0.7,
            strategy='best1bin',
            maxiter=100,
            tol=1e-6,
            x0=guess,
            polish=True
        )
        
        if result.success:
            current_c5 = result.fun - 1e4 * abs(np.sum(result.x) - target_sum)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
