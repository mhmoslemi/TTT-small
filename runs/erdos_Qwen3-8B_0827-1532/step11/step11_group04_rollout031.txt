import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate diverse initial guesses
    random_guesses = [np.random.uniform(0, 1, n_points) for _ in range(3)]
    for guess in random_guesses:
        guess = guess * (target_sum / np.sum(guess))
        # Add structured perturbations
        guess = np.roll(guess, np.random.randint(n_points))
        perturbations.append(guess)
    perturbations.append(initial_guess)
    
    for perturbation_idx in range(200):  # Reduced number of perturbations for efficiency
        if perturbation_idx < len(perturbations):
            perturbed = perturbations[perturbation_idx].copy()
        else:
            # Perturb best_h with varying noise
            if perturbation_idx < 30:
                noise = np.random.uniform(-0.2, 0.2, n_points)
            elif perturbation_idx < 60:
                noise = np.random.uniform(-0.1, 0.1, n_points)
            else:
                noise = np.random.uniform(-0.05, 0.05, n_points)
            perturbed = best_h + noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
        
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Use L-BFGS-B for faster convergence
        result = minimize(
            objective,
            perturbed,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 200}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
