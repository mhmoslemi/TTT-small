from scipy.optimize import minimize
import time

    num_perturbations = 500  # Increased for better exploration

        elif perturbation_idx < 200:
            # Perturb best_h with medium noise
            noise = np.random.uniform(-0.15, 0.15, n_points)

            perturbed = np.clip(perturbed, 0, 1)

            scale = target_sum / current_sum
            perturbed = perturbed * scale

            # Perturb best_h with small noise

            scale = target_sum / current_sum

        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
