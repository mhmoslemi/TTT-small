import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    def objective(h):
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.inf
        scale = target_sum / current_sum
        h_normalized = h * scale
        h_normalized = np.clip(h_normalized, 0, 1)
        one_minus_h = 1.0 - h_normalized
        corr = np.correlate(h_normalized, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Run differential evolution with multiple strategies
    strategies = ['best1bin', 'rand1bin', 'best2bin', 'rand2bin']
    for strategy in strategies:
        result = differential_evolution(
            objective,
            bounds,
            strategy=strategy,
            popsize=15,
            mutation=(0.5, 1),
            recombination=0.8,
            tol=1e-10,
            maxiter=100,
            polish=True,
            seed=seed
        )
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Check elapsed time
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
