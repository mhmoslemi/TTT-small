from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    dx = 2.0 / n_points

    best_c5 = float('inf')

    # Generate multiple initial guesses
    initial_guesses = [initial_h_values]

    binary_guess = np.zeros(n_points)
    binary_guess[::2] = 1.0
    binary_guess = binary_guess * (target_sum / np.sum(binary_guess))
    initial_guesses.append(binary_guess)
    
    # Binary guess with 1s every third point
    binary_guess_3 = np.zeros(n_points)
    binary_guess_3[::3] = 1.0

    initial_guesses.append(binary_guess_3)
    
    # Random uniform guess
    random_guess = np.random.uniform(0, 1, n_points)
    random_guess = random_guess * (target_sum / np.sum(random_guess))
    initial_guesses.append(random_guess)
    
    # Periodic guess
    periodic_guess = np.zeros(n_points)
    for i in range(n_points):
        if i % 2 == 0:

    periodic_guess = periodic_guess * (target_sum / np.sum(periodic_guess))
    initial_guesses.append(periodic_guess)
    
    # Evaluate each initial guess
    for guess in initial_guesses:
        # Ensure the guess is valid
        guess = np.clip(guess, 0, 1)
        guess = guess * (target_sum / np.sum(guess))
        # Run optimization
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx

        result = minimize(

            constraints=cons,

        )
