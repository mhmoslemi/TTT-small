from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Use the provided initial guess and generate diverse initial guesses
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate initial guesses
    initial_guesses = []
    # Add initial_h_values
    initial_guesses.append(initial_guess)
    
    # Add structured guesses with alternating 1s and 0s
    structured_guess = np.zeros(n_points)
    for i in range(n_points):
        if i % 2 == 0:
            structured_guess[i] = 1.0
    initial_guesses.append(structured_guess)
    
    # Add structured guesses with blocks
    block_guess = np.zeros(n_points)
    block_size = n_points // 4
    for i in range(block_size):
        block_guess[i] = 1.0
    initial_guesses.append(block_guess)
    for i in range(block_size, 2 * block_size):
        block_guess[i] = 0.0
    initial_guesses.append(block_guess)
    for i in range(2 * block_size, 3 * block_size):
        block_guess[i] = 1.0
    initial_guesses.append(block_guess)
    for i in range(3 * block_size, 4 * block_size):
        block_guess[i] = 0.0
    initial_guesses.append(block_guess)
    
    # Add random binary guesses
    for _ in range(10):
        np.random.seed(seed + len(initial_guesses))
        h = np.zeros(n_points)
        indices = np.random.choice(n_points, size=int(n_points//2), replace=False)
        h[indices] = 1.0
        initial_guesses.append(h)
    
    best_c5 = float('inf')
    best_h = initial_guesses[0].copy()
    
    start_time = time.time()
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds and constraints
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,

            best_h = result.x
