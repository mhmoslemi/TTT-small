import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Construct a binary initial guess: 1 on [0, 0.5], 0 on [0.5, 1.5], 1 on [1.5, 2]
    h_initial = np.zeros(n_points)
    h_initial[:25] = 1.0  # 0.5 length
    h_initial[25:75] = 0.0  # 1.0 length
    h_initial[75:] = 1.0  # 0.5 length
    best_h = h_initial.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    num_perturbations = 150  # Reduced for efficiency
    
    for perturbation_idx in range(num_perturbations):
        if perturbation_idx < 20:
            # Larger noise for initial exploration
            noise = np.random.uniform(-0.3, 0.3, n_points)
        else:
            # Smaller noise for fine-tuning
            noise = np.random.uniform(-0.1, 0.1, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='COBYLA',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'disp': False}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
