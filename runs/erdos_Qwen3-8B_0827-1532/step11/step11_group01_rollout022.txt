import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Access initial_h_values from global scope
    initial_h_values = initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate diverse initial guesses
    initial_guesses = []
    
    # Add initial guess
    initial_guesses.append(initial_h_values)
    
    # Generate structured guesses with different periods
    for p in [2, 3, 4, 5, 6, 7, 8, 9, 10]:
        structured = np.zeros(n_points)
        for i in range(n_points):
            if i % p == 0:
                structured[i] = 1.0
            else:
                structured[i] = 0.0
        initial_guesses.append(structured)
    
    # Generate random binary sequence
    random_binary = np.zeros(n_points)
    indices = np.random.choice(n_points, size=n_points // 2, replace=False)
    random_binary[indices] = 1.0
    initial_guesses.append(random_binary)
    
    # Generate perturbations of initial guesses
    np.random.seed(seed)
    perturbations = []
    for guess in initial_guesses:
        noise = np.random.uniform(-0.1, 0.1, n_points)
        perturbed = guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        perturbations.append(perturbed)
    initial_guesses += perturbations
    
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    
    start_time = time.time()
    
    for guess in initial_guesses:
        # Objective function to minimize C5
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization with tighter tolerances and more iterations
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 1500}
        )
        
        # Check if this result is better
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
