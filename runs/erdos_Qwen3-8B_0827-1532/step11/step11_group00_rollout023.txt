import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Generate diverse initial guesses
    initial_guesses = [initial_guess]
    
    # Add structured guess with alternating 1s and 0s
    structured_guess = np.zeros(n_points)
    for i in range(n_points):
        if i % 2 == 0:
            structured_guess[i] = 1.0
    initial_guesses.append(structured_guess)
    
    # Add random binary guesses
    for _ in range(5):
        np.random.seed(seed + len(initial_guesses))
        h = np.zeros(n_points)
        indices = np.random.choice(n_points, size=int(n_points//2), replace=False)
        h[indices] = 1.0
        initial_guesses.append(h)
    
    best_c5 = float('inf')
    best_h = initial_guesses[0].copy()
    
    start_time = time.time()
    
    for guess in initial_guesses:
        # Adjust guess to satisfy sum constraint
        adjusted_guess = guess.copy()
        adjusted_guess[-1] = target_sum - np.sum(adjusted_guess[:-1])
        
        # Define bounds for first n-1 elements
        bounds = [(0.0, 1.0) for _ in range(n_points - 1)]
        
        # Objective function
        def objective(x):
            h = np.zeros(n_points)
            h[:-1] = x
            h[-1] = adjusted_guess[-1]
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Differential evolution optimization
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=100,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed
        )
        
        if result.success:

            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = np.zeros(n_points)
                best_h[:-1] = result.x
                best_h[-1] = adjusted_guess[-1]
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
