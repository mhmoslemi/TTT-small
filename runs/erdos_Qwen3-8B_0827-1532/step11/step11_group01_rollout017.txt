        h = h / np.sum(h) * target_sum
        return h
    
    # Create initial guesses: structured, random, and perturbations
    initial_guesses = [initial_guess]
    for _ in range(5):
        initial_guesses.append(generate_random_guess())
    
    # Structured patterns
    structured_patterns = []
    for period in [2, 3, 4, 5, 6, 7, 8, 9, 10]:
        pattern = np.zeros(n_points)
        for i in range(n_points):
            if i % period == 0:
                pattern[i] = 1.0
        structured_patterns.append(pattern)
    initial_guesses.extend(structured_patterns)
    
    # Function to compute C5 for a given h
    def compute_c5(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Objective function for optimization
    def objective(h):
        return compute_c5(h)
    
    # Constraints: sum(h) == target_sum
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization settings
    opt_options = {
