        sum_structured = np.sum(structured)

        structured *= scale

            perturbed = initial_guess + sinusoid
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)

        def objective(h):

        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
