from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate structured initial guesses with different periodicities
    structured_guesses = []
    for m in range(2, 8):  # m=2 to 7
        structured = np.zeros(n_points)
        for i in range(n_points):
            if i % m == 0:
                structured[i] = 1.0
            else:
                structured[i] = 0.0
        structured_guesses.append(structured)
    
    # Generate random initial guesses
    random_guesses = []
    for _ in range(5):  # 5 random guesses
        random_guess = np.random.uniform(0, 1, n_points)
        random_guess = random_guess * (target_sum / np.sum(random_guess))
        random_guesses.append(random_guess)
    
    # Generate perturbations of initial guess
    np.random.seed(seed)
    perturbations = []
    for _ in range(3):
        noise = np.random.uniform(-0.1, 0.1, n_points)
        perturbed = initial_guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        perturbations.append(perturbed)
    
    # Combine all initial guesses
    initial_guesses = [initial_guess] + structured_guesses + random_guesses + perturbations
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    for guess in initial_guesses:
        # Objective function to minimize C5
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

        result = minimize(
