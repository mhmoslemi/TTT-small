    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Run differential evolution to find a good candidate
    result_de = differential_evolution(
        objective_de,
        bounds,
        strategy='best1bin',
        popsize=15,
        mutation=(0.5, 1.0),
        recombination=0.8,
        tol=1e-8,
        maxiter=100,
        polish=True,
        vectorized=True
    )
    
    if result_de.success:
        best_h = result_de.x
        best_c5 = result_de.fun
    
    # Define the objective function for SLSQP
    def objective_slsqp(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraints: sum(h) == target_sum
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    
    # Bounds for each h[i]
    bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]
    
    # Perform SLSQP optimization on the best_h from DE
    result_slsqp = minimize(
        objective_slsqp,
        best_h,
        method='SLSQP',
        bounds=bounds_slsqp,
        constraints=cons,
        options={'ftol': 1e-10, 'maxiter': 1500}
    )
    
    if result_slsqp.success and result_slsqp.fun < best_c5:
        best_c5 = result_slsqp.fun
        best_h = result_slsqp.x
    
    # Check elapsed time and return the best solution
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
