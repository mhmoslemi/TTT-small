import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)

    start_time = time.time()
    
    num_perturbations = 500  # Increased number of perturbations
    
    for perturbation_idx in range(num_perturbations):
        # Select a random initial guess
        initial_guess = np.random.choice(initial_guesses)
        # Perturb the initial guess
        if perturbation_idx < 20:
            # Large random noise for exploration
            noise = np.random.uniform(-0.2, 0.2, n_points)
        elif perturbation_idx < 50:
            # Medium noise
            noise = np.random.uniform(-0.1, 0.1, n_points)
        else:
            # Small noise for refinement
            noise = np.random.uniform(-0.05, 0.05, n_points)
        perturbed = initial_guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = (n_points / 2) / current_sum
