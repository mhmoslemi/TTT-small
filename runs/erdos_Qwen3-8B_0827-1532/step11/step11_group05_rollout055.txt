    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    
    start_time = time.time()
    
    # Adaptive noise scaling parameters
    initial_noise = 0.2
    noise_decay = 0.95
    max_iterations = 1000
    
    for iteration in range(max_iterations):
        if time.time() - start_time >= budget_s:
            break
        
        # Select a random initial guess
        current_guess = np.random.choice(initial_guesses)
        
        # Generate perturbation with adaptive noise
        noise = np.random.uniform(-initial_noise, initial_noise, n_points)
        perturbed = current_guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Define objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Optimize
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 500}
        )
        
        # Update best solution
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Decay noise for next iteration
        initial_noise *= noise_decay
    
    return best_h, best_c5, n_points
