import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    # Use the provided initial guess
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Define objective function with penalty for sum constraint
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        # Penalty for sum constraint
        penalty = 1e6 * (np.sum(h) - target_sum)**2
        return max_corr * dx + penalty
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Run differential evolution with tuned parameters
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=15,
        mutation=(0.5, 1.0),
        recombination=0.9,
        tol=1e-6,
        maxiter=1000,
        seed=seed
    )
    
    # Evaluate the best solution found
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Check elapsed time and return the best solution
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
