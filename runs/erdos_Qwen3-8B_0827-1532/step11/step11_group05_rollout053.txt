from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    dx = 2.0 / n_points

    h = np.zeros(n_points)

    h[75:] = 0.0
    sum_h = np.sum(h)
    scale = target_sum / sum_h
    h *= scale
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    start_time = time.time()
    
    num_perturbations = 500  # Increased for better exploration

        if perturbation_idx < 50:
            # Random initial guess with larger noise

            perturbed = h + noise
            perturbed = np.clip(perturbed, 0, 1)

            perturbed = perturbed * scale
        else:
            # Perturb best_h with varying noise
            if perturbation_idx < 200:

            perturbed = best_h + noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
        
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')

        bounds = [(0.0, 1.0) for _ in range(n_points)]
