    )
    
    # Refine with local optimization
    def local_objective(h):
        scaled_h = h * (target_sum / np.sum(h)) if np.sum(h) != 0 else h
        one_minus_h = 1.0 - scaled_h
        corr = np.correlate(scaled_h, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    from scipy.optimize import minimize
    result_local = minimize(
        local_objective,
        result.x,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}],
        options={'ftol': 1e-10, 'maxiter': 500}
    )
    
    best_h = result_local.x
    best_c5 = result_local.fun
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
