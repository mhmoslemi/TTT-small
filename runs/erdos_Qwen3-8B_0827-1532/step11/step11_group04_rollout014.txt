import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate diverse initial guesses: original, random, and structured
    random_guesses = [np.random.uniform(0, 1, n_points) for _ in range(3)]
    structured_guesses = []
    for guess in random_guesses:
        guess = guess * (target_sum / np.sum(guess))
        structured_guesses.append(guess)
    structured_guesses.append(initial_guess)
    
    # Optimization parameters
    max_iter = 100
    perturbation_steps = 150
    noise_scales = [0.2, 0.1, 0.05]
    
    for perturbation_idx in range(perturbation_steps):
        # Generate a new perturbation
        if perturbation_idx < len(structured_guesses):
            perturbed = structured_guesses[perturbation_idx].copy()
        else:
            # Perturb best_h with varying noise
            noise_scale = noise_scales[perturbation_idx % len(noise_scales)]
            noise = np.random.uniform(-noise_scale, noise_scale, n_points)
            perturbed = best_h + noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
        
        # Define objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization settings
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Use L-BFGS-B for better convergence
        result = minimize(
            objective,
            perturbed,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': max_iter}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
