import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    start_time = time.time()
    
    # Objective function with penalty for sum constraint
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr) * dx
        penalty = 1e4 * np.abs(np.sum(h) - target_sum)
        return max_corr + penalty
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Run differential evolution for global exploration
    result_de = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=20,
        maxiter=100,
        tol=1e-12,
        mutation=(0.5, 1),
        recombination=0.8,
        polish=True,
        seed=seed
    )
    
    # Refine with SLSQP
    def objective_slsqp(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    
    result_slsqp = minimize(
        objective_slsqp,
        result_de.x,
        method='SLSQP',
        bounds=bounds_slsqp,
        constraints=cons,
        options={'ftol': 1e-12, 'maxiter': 500}
    )
    
    best_h = result_slsqp.x
    best_c5 = result_slsqp.fun
    
    # Early exit if time is up
    elapsed = time.time() - start_time
    if elapsed < budget_s:
        # Additional perturbations and local optimization

            result = minimize(
                objective_slsqp,

                constraints=cons,

            )
            if result.success and result.fun < best_c5:
