import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate structured initial guess with two intervals of 0.5 length
    structured_guess = np.zeros(n_points)
    interval_length = 0.5
    interval1_start = 0.0
    interval1_end = interval_length
    interval2_start = 1.0
    interval2_end = 1.5
    for i in range(n_points):
        x = i * dx
        if interval1_start <= x <= interval1_end or interval2_start <= x <= interval2_end:
            structured_guess[i] = 1.0
        else:
            structured_guess[i] = 0.0
    
    best_c5 = float('inf')
    best_h = structured_guess.copy()
    
    start_time = time.time()
    
    num_perturbations = 200  # Reduced for efficiency
    perturbations = [structured_guess]
    
    # Add a few random initial guesses
    for _ in range(3):
        guess = np.random.uniform(0, 1, n_points)
        guess = guess * (target_sum / np.sum(guess))
        perturbations.append(guess)
    
    for perturbation_idx in range(num_perturbations):
        if perturbation_idx < len(perturbations):
            perturbed = perturbations[perturbation_idx].copy()
        else:
            # Perturb best_h with varying noise
            if perturbation_idx < 50:
                noise_scale = 0.2
            elif perturbation_idx < 100:
                noise_scale = 0.1
            else:
                noise_scale = 0.05
            noise = np.random.uniform(-noise_scale, noise_scale, n_points)
            perturbed = best_h + noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
        
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 200}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
