    if result_de is not None and result_de.success:
        def objective_slsqp(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]
        
        result_slsqp = minimize(objective_slsqp, result_de.x, 
                               method='SLSQP', bounds=bounds_slsqp, 
                               constraints=cons, 
                               options={'ftol': 1e-10, 'maxiter': 1500})
        
        if result_slsqp.success and result_slsqp.fun < best_c5:
            best_c5 = result_slsqp.fun
            best_h = result_slsqp.x
    
    # Try initial guess with SLSQP
    def objective_slsqp(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]
    
    result_initial = minimize(objective_slsqp, initial_guess,

    if result_initial.success and result_initial.fun < best_c5:

    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return best_h, best_c5, n_points
    
    # Generate perturbations based on best_h
    num_perturbations = 50
    for perturbation_idx in range(num_perturbations):
        if perturbation_idx < 10:
            # Random initial guess based on best_h
            random_noise = np.random.uniform(-0.2, 0.2, n_points)
            perturbed = best_h + random_noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
        else:

            if perturbation_idx < 30:
