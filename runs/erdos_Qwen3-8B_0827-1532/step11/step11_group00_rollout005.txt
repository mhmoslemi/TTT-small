import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Use the provided initial guess and generate diverse initial guesses
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate initial guesses
    initial_guesses = []
    # Add initial_h_values
    initial_guesses.append(initial_guess)
    
    # Add structured guesses
    # 1. Alternating 1s and 0s
    structured_guess = np.zeros(n_points)
    for i in range(n_points):
        if i % 2 == 0:
            structured_guess[i] = 1.0
    initial_guesses.append(structured_guess)
    
    # 2. 1s on first 25% and last 25%
    structured_guess2 = np.zeros(n_points)
    quarter = n_points // 4
    structured_guess2[:quarter] = 1.0
    structured_guess2[-quarter:] = 1.0
    initial_guesses.append(structured_guess2)
    
    # 3. 1s on every third point
    structured_guess3 = np.zeros(n_points)
    for i in range(n_points):
        if i % 3 == 0:
            structured_guess3[i] = 1.0
    initial_guesses.append(structured_guess3)
    
    # Add random binary guesses
    for _ in range(5):
        np.random.seed(seed + len(initial_guesses))
        h = np.zeros(n_points)
        indices = np.random.choice(n_points, size=int(n_points//2), replace=False)
        h[indices] = 1.0
        initial_guesses.append(h)
    
    best_c5 = float('inf')
    best_h = initial_guesses[0].copy()
    
    start_time = time.time()
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds and constraints
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization
        result = minimize(
            objective,
            guess,
            method='trust-constr',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 500, 'disp': False}
        )
        
        # Check if successful
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
