import numpy as np
from scipy.optimize import minimize
import time
from multiprocessing import Pool, cpu_count

def run(seed=42, budget_s=100, **kwargs):

    initial_h_values = globals()['initial_h_values']

    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate initial guesses
    initial_guesses = [initial_h_values.copy()]
    
    # Add structured patterns
    for mod in [2, 3, 4, 5, 6, 7, 8, 9, 10]:
        structured_h = np.zeros(n_points)
        for i in range(n_points):
            if i % mod == 0:
                structured_h[i] = 1.0
            else:
                structured_h[i] = 0.0
        structured_h = structured_h * (target_sum / np.sum(structured_h))
        initial_guesses.append(structured_h)
    
    # Add random initial guesses
    for _ in range(5):
        random_h = np.random.rand(n_points)
        random_h = random_h / np.sum(random_h) * target_sum
        initial_guesses.append(random_h)
    
    # Define objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define constraints
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    
    # Helper function for multiprocessing
    def optimize_guess(guess):
        result = minimize(
            objective,
            guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 1500}
        )
        if result.success:
            return result.x, result.fun
        else:
            return guess, float('inf')
    
    # Use multiprocessing
    with Pool(processes=cpu_count()) as pool:
        start_time = time.time()
        results = []
        for guess in initial_guesses:
            results.append(pool.apply_async(optimize_guess, (guess,)))
        
        best_c5 = float('inf')
        best_h = initial_guesses[0].copy()
        
        while results:
            elapsed = time.time() - start_time
