import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate structured initial guesses
    structured_guesses = []
    # Two-interval pattern: [0, 0.5] and [1.0, 1.5]
    two_interval = np.zeros(n_points)
    interval_length = 0.5
    interval_points = int(interval_length / dx)
    two_interval[:interval_points] = 1.0
    two_interval[interval_points + interval_points:] = 1.0
    structured_guesses.append(two_interval)
    
    # Three-interval pattern: [0, 0.33], [0.66, 1.0], [1.32, 1.66]
    three_interval = np.zeros(n_points)
    interval_length = 0.33
    interval_points = int(interval_length / dx)
    three_interval[:interval_points] = 1.0
    three_interval[interval_points + interval_points + interval_points:] = 1.0
    three_interval[interval_points + interval_points: interval_points + interval_points + interval_points] = 1.0
    structured_guesses.append(three_interval)
    
    # Add initial guess and random guesses
    random_guesses = [np.random.uniform(0, 1, n_points) for _ in range(2)]
    for guess in random_guesses:
        guess = guess * (target_sum / np.sum(guess))
        structured_guesses.append(guess)
    
    # Perturbations
    perturbations = structured_guesses
    
    # Optimization parameters
    max_iter = 100
    ftol = 1e-8
    
    for perturbation in perturbations:
        # Scale to meet the target sum
        perturbed = perturbation.copy()
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        perturbed = np.clip(perturbed, 0, 1)
        
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': ftol, 'maxiter': max_iter}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
