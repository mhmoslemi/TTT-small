    # Periodic guess: alternate 1 and 0

            periodic_guess[i] = 1.0
    initial_guesses.append(periodic_guess)
    # Random binary guess
    random_binary = np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])
    random_binary = random_binary[:n_points]

    initial_guesses.append(random_binary * scale)

    random_perturb = np.random.uniform(0.4, 0.6, n_points)

    initial_guesses.append(random_perturb * scale)
    
    # Optimize each initial guess with differential evolution
    for guess in initial_guesses:
        # Objective function with penalty for sum constraint
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            penalty = 1e3 * (np.sum(h) - target_sum)**2
            return max_corr * dx + penalty
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Perform differential evolution
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=15,
            mutation=(0.5, 1.0),
            recombination=0.9,
            tol=1e-6,
            maxiter=100,
            polish=True,
            disp=False
        )
        
        # Check if this result is better
        if result.success:
            candidate_h = result.x
            # Refine with SLSQP for better precision
            def slsqp_objective(h):
                one_minus_h = 1.0 - h
                corr = np.correlate(h, one_minus_h, mode='full')
                max_corr = np.max(corr)
                return max_corr * dx
            
            bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]
            cons_slsqp = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
            
            result_slsqp = minimize(
