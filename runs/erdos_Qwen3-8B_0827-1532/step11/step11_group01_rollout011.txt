import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate diverse initial guesses
    np.random.seed(seed)
    initial_guesses = [initial_guess]
    
    # Add random initial guesses
    for _ in range(5):
        h = np.random.uniform(0, 1, n_points)
        h = h * (target_sum / np.sum(h))
        initial_guesses.append(h)
    
    # Add structured initial guesses with different periodicities
    for m in [2, 3, 4, 5, 6]:
        structured = np.zeros(n_points)
        for i in range(n_points):
            if i % m == 0:
                structured[i] = 1.0
        # Scale to meet the target sum
        structured_sum = np.sum(structured)
        if structured_sum > 0:
            structured = structured * (target_sum / structured_sum)
        initial_guesses.append(structured)
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    for guess in initial_guesses:
        # Objective function to minimize C5
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization with tighter tolerances and more iterations
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-8, 'maxiter': 500}
        )
        
        # Check if this result is better
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
