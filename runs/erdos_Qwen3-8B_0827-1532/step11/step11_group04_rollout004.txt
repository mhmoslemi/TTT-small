    # Generate diverse initial guesses
    def generate_initial_guesses(base_h):
        guesses = []
        # Original
        guesses.append(base_h.copy())

        shifted = np.roll(base_h, 1)
        guesses.append(shifted)
        # Inverted
        inverted = 1.0 - base_h
        guesses.append(inverted)
        # Random variations
        for _ in range(2):
            noise = np.random.uniform(-0.2, 0.2, n_points)
            perturbed = base_h + noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
            guesses.append(perturbed)
        return guesses
    
    initial_guesses = generate_initial_guesses(initial_guess)
    
    # Adaptive noise scaling
    noise_scale = 0.2
    noise_decay = 0.995
    max_iterations = 1000
    
    for iteration in range(max_iterations):
        if time.time() - start_time >= budget_s:
            break
        # Select current guess
        current_h = best_h.copy()
        # Add noise
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = current_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 200}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Decay noise scale
