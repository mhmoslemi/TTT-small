from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    initial_guesses.append(initial_guess)
    
    # Add structured guesses with single interval
    for shift in [0.0, 0.5, 1.0]:
        h = np.zeros(n_points)
        start = int(shift * n_points)
        end = int((shift + 0.5) * n_points)
        h[start:end] = 1.0
        initial_guesses.append(h)
    
    # Add structured guesses with two intervals
    for shift in [0.0, 0.5, 1.0]:
        h = np.zeros(n_points)
        start1 = int(shift * n_points)
        end1 = int((shift + 0.5) * n_points)
        start2 = int((2.0 - 0.5 - shift) * n_points)
        end2 = int((2.0 - shift) * n_points)
        h[start1:end1] = 1.0
        h[start2:end2] = 1.0
        initial_guesses.append(h)
    
    # Add random binary guesses
    for _ in range(3):
        np.random.seed(seed + len(initial_guesses))
        h = np.zeros(n_points)
        indices = np.random.choice(n_points, size=int(n_points//2), replace=False)
        h[indices] = 1.0
        initial_guesses.append(h)
    
    best_c5 = float('inf')
    best_h = initial_guesses[0].copy()
    
    start_time = time.time()
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds and constraints
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
