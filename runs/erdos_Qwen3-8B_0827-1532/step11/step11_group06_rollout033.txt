import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate diverse initial guesses
    initial_guesses = []
    
    # Random binary patterns
    for _ in range(5):
        random_binary = np.random.rand(n_points)
        random_binary = (random_binary < 0.5).astype(float)
        random_binary *= target_sum / np.sum(random_binary)
        initial_guesses.append(random_binary)
    
    # Structured patterns
    for period in [3, 4, 5]:
        structured = np.zeros(n_points)
        for i in range(n_points):
            if i % period == 0:
                structured[i] = 1.0
        structured *= target_sum / np.sum(structured)
        initial_guesses.append(structured)
    
    # Perturbations of initial guess
    perturbations = [np.random.uniform(-0.1, 0.1, n_points) for _ in range(5)]
    for p in perturbations:
        perturbed = initial_guess + p
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        initial_guesses.append(perturbed)
    
    # Add initial guess itself
    initial_guesses.append(initial_guess)
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Objective function to minimize C5
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization with tighter tolerances and more iterations
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-8, 'maxiter': 200}
        )
        
        # Check if this result is better
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
