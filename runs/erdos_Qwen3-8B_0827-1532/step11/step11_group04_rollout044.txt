    # Add sinusoidal guess
    sinusoidal_guess = generate_sinusoidal(n_points)
    initial_guesses.append(sinusoidal_guess)
    # Add symmetric binary guess
    binary_guess = generate_symmetric_binary(n_points)
    initial_guesses.append(binary_guess)
    # Add some random guesses
    random_guesses = [np.random.uniform(0, 1, n_points) for _ in range(3)]
    for guess in random_guesses:
        guess = guess * (target_sum / np.sum(guess))
        initial_guesses.append(guess)
    
    perturbations = initial_guesses
    
    num_perturbations = 500  # Increased number of perturbations
    for perturbation_idx in range(num_perturbations):
        if perturbation_idx < len(perturbations):
            perturbed = perturbations[perturbation_idx].copy()
        else:
            # Perturb best_h with varying noise
            if perturbation_idx < 30:
                noise = np.random.uniform(-0.2, 0.2, n_points)
            elif perturbation_idx < 60:
                noise = np.random.uniform(-0.1, 0.1, n_points)
            else:
                noise = np.random.uniform(-0.05, 0.05, n_points)
            perturbed = best_h + noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
        
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-12, 'maxiter': 500}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
