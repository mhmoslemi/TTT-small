import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Use the provided initial guess
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Define objective with penalty for constraint violation
    def objective_with_penalty(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        penalty = (np.sum(h) - target_sum)**2 * 1e4  # Penalize constraint violation
        return max_corr * dx + penalty
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Differential evolution to find a good initial guess
    result_de = differential_evolution(
        objective_with_penalty,
        bounds,
        strategy='best1bin',
        popsize=15,
        maxiter=100,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed
    )
    
    # Refine with SLSQP
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraints: sum(h) == target_sum
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    
    # Initial guess from DE
    result_slsqp = minimize(
        objective,
        result_de.x,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-10, 'maxiter': 500}
    )
    
    # Check if this result is better
    if result_slsqp.success and result_slsqp.fun < best_c5:
        best_c5 = result_slsqp.fun
        best_h = result_slsqp.x
    
    # Generate a few perturbations based on best_h
    num_perturbations = 10  # Reduced number for time efficiency
    for perturbation_idx in range(num_perturbations):
        # Perturb best_h with varying noise
        noise = np.random.uniform(-0.05, 0.05, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Perform optimization with tighter tolerances and more iterations
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 300}
        )
        
        # Check if this result is better
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
