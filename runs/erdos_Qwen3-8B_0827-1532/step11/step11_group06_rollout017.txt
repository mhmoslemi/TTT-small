    perturbed_initials = [np.clip(pi, 0, 1) * (target_sum / np.sum(np.clip(pi, 0, 1))) for pi in perturbed_initials]
    initial_guesses += perturbed_initials
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Parameters for differential evolution
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Objective function for DE with penalty
    def objective_de(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        penalty = 1e4 * (np.sum(h) - target_sum)**2
        return max_corr * dx + penalty
    
    # Number of DE runs
    num_de_runs = 20  # Adjust based on time budget
    for i in range(num_de_runs):
        # Select an initial guess
        if i < len(initial_guesses):
            h_initial = initial_guesses[i]
        else:
            # Generate a random initial guess
            h_initial = np.random.rand(n_points)
            h_initial = np.clip(h_initial, 0, 1)
            h_initial *= target_sum / np.sum(h_initial)
        
        # Run differential evolution
        result = differential_evolution(
            objective_de,
            bounds,
            strategy='best1bin',
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            maxiter=100,
            seed=seed + i
        )
        
        # Check if result is successful
        if result.success:
            current_h = result.x
            one_minus_h = 1.0 - current_h
            corr = np.correlate(current_h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            current_c5 = max_corr * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
