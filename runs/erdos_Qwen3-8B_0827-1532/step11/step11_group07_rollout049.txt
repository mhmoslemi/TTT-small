        sum_h = np.sum(h)
        if sum_h == 0:
            return np.inf
        scale = target_sum / sum_h
        scaled_h = h * scale
        one_minus_h = 1.0 - scaled_h
        corr = np.correlate(scaled_h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function for SLSQP
    def objective_slsqp(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Generate initial guesses
    initial_guesses = [initial_guess]
    # Uniform guess
    uniform_guess = np.full(n_points, 0.5)
    initial_guesses.append(uniform_guess)
    # Binary guess
    binary_guess = np.concatenate([np.ones(n_points//2), np.zeros(n_points//2)])
    binary_guess = binary_guess[:n_points]
    initial_guesses.append(binary_guess)
    # Periodic guess: alternate 1 and 0 with spacing
    periodic_guess = np.zeros(n_points)
    for i in range(n_points):
        if i % 2 == 0:
            periodic_guess[i] = 1.0
    periodic_guess = periodic_guess[:n_points]
    initial_guesses.append(periodic_guess)
    
    # Optimize each initial guess with differential evolution and SLSQP
    for guess in initial_guesses:
        # Run differential evolution
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result_de = differential_evolution(objective_de, bounds, strategy='best1bin', popsize=15, maxiter=100, tol=1e-6, mutation=(0.5, 1), recombination=0.8, polish=True)
        # Refine with SLSQP
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        result_slsqp = minimize(objective_slsqp, result_de.x, method='SLSQP', bounds=bounds, constraints=cons, options={'ftol': 1e-10, 'maxiter': 1500})
        # Check if this result is better
        if result_slsqp.success and result_slsqp.fun < best_c5:
            best_c5 = result_slsqp.fun
            best_h = result_slsqp.x
    
    # Generate perturbations based on best_h
    num_perturbations = 50  # Reduced number for time efficiency
    for perturbation_idx in range(num_perturbations):
        # Generate perturbations based on best_h
        if perturbation_idx < 10:
            # Random initial guess based on best_h
            random_noise = np.random.uniform(-0.2, 0.2, n_points)
            perturbed = best_h + random_noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
        else:
            # Perturb best_h with varying noise
            if perturbation_idx < 30:
                # Larger noise for early perturbations
                noise = np.random.uniform(-0.2, 0.2, n_points)
            else:
                # Smaller noise for later perturbations
                noise = np.random.uniform(-0.05, 0.05, n_points)
            perturbed = best_h + noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
        
        # Run differential evolution on the perturbed guess
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result_de = differential_evolution(objective_de, bounds, strategy='best1bin', popsize=15, maxiter=50, tol=1e-6, mutation=(0.5, 1), recombination=0.8, polish=True)
        # Refine with SLSQP
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        result_slsqp = minimize(objective_slsqp, result_de.x, method='SLSQP', bounds=bounds, constraints=cons, options={'ftol': 1e-10, 'maxiter': 1500})
        # Check if this result is better
        if result_slsqp.success and result_slsqp.fun < best_c5:
            best_c5 = result_slsqp.fun
            best_h = result_slsqp.x
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
