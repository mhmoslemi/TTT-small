        sum_h = np.sum(h_clipped)
        if sum_h == 0:
            return float('inf')
        scale = target_sum / sum_h
        scaled_h = h_clipped * scale
        one_minus_h = 1.0 - scaled_h
        corr = np.correlate(scaled_h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    result = differential_evolution(

    )

    if result.success:
        best_h = result.x

    # Check elapsed time

    if elapsed >= budget_s:
        return best_h, best_c5, n_points

    num_perturbations = 10  # Reduced number for time efficiency

        if perturbation_idx < 5:
            # Random initial guess based on best_h
            random_noise = np.random.uniform(-0.2, 0.2, n_points)
            perturbed = best_h + random_noise

            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum

            if perturbation_idx < 10:
