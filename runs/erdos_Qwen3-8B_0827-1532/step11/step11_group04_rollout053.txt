    a = 0.5
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * dx
        if a <= x <= a + 1:
            h[i] = 1.0
    structured_guesses.append(h)
    
    # Two intervals
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * dx
        if (0 <= x <= 0.5) or (1.5 <= x <= 2.0):
            h[i] = 1.0
    structured_guesses.append(h)
    
    # Another two intervals
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * dx
        if (0.25 <= x <= 0.75) or (1.25 <= x <= 1.75):
            h[i] = 1.0
    structured_guesses.append(h)
    
    # Add structured guesses to perturbations
    perturbations = structured_guesses.copy()
    
    # Add some random guesses
    random_guesses = [np.random.uniform(0, 1, n_points) for _ in range(3)]
    for guess in random_guesses:
        guess = guess * (target_sum / np.sum(guess))
        perturbations.append(guess)
    
    perturbations.append(initial_guess)
    
    num_perturbations = len(perturbations)
    
    for perturbation_idx in range(num_perturbations):
        perturbed = perturbations[perturbation_idx].copy()
        
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 200}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
