import numpy as np
from scipy.optimize import minimize
import time
from concurrent.futures import ThreadPoolExecutor

def run(seed=42, budget_s=100, **kwargs):
    # Pre-imported initial guess
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate diverse initial guesses
    initial_guesses = []
    # Add initial_h_values
    initial_guesses.append(initial_guess)
    
    # Add structured guesses
    structured_guesses = []
    # Alternating 1s and 0s
    structured_guesses.append(np.where(np.arange(n_points) % 2 == 0, 1.0, 0.0))
    # Checkerboard pattern (every third sample)
    structured_guesses.append(np.where(np.arange(n_points) % 3 == 0, 1.0, 0.0))
    # Random binary guesses
    for _ in range(10):  # More random guesses
        np.random.seed(seed + len(initial_guesses))
        h = np.zeros(n_points)
        indices = np.random.choice(n_points, size=int(n_points//2), replace=False)
        h[indices] = 1.0
        structured_guesses.append(h)
    # Add structured guesses to initial_guesses
    initial_guesses.extend(structured_guesses)
    
    best_c5 = float('inf')
    best_h = initial_guesses[0].copy()
    
    start_time = time.time()
    
    # Function to evaluate a single initial guess
    def evaluate_guess(guess):
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds and constraints
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-12, 'maxiter': 3000}
        )
        
        # Return result if successful
        if result.success:
            return result.fun, result.x
        else:
            return float('inf'), guess  # fallback to original guess
    
    # Use ThreadPoolExecutor for parallel evaluation
    with ThreadPoolExecutor(max_workers=10) as executor:
        futures = [executor.submit(evaluate_guess, guess) for guess in initial_guesses]
        for future in futures:
            c5, h = future.result()
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
            # Check elapsed time
            elapsed = time.time() - start_time
            if elapsed >= budget_s:
                break
    
    return best_h, best_c5, n_points
