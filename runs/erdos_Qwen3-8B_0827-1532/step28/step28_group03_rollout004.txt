    segment_length = n_points // 5
    h_five[:segment_length] = 1.0
    h_five[segment_length:2*segment_length] = 0.5
    h_five[2*segment_length:3*segment_length] = 0.0
    h_five[3*segment_length:4*segment_length] = 0.5
    h_five[4*segment_length:] = 1.0
    sum_h = np.sum(h_five)
    scale = target_sum / sum_h
    initial_guesses.append(h_five * scale)
    
    # Random segments with varying heights
    h_rand_seg = np.zeros(n_points)
    for i in range(n_points):
        if np.random.rand() < 0.3:
            h_rand_seg[i] = 1.0
        elif np.random.rand() < 0.5:
            h_rand_seg[i] = 0.5
    sum_h = np.sum(h_rand_seg)
    scale = target_sum / sum_h
    initial_guesses.append(h_rand_seg * scale)
    
    # Random initial guess
    initial_guesses.append(np.random.uniform(0, 1, n_points))
    
    # Generate more diverse initial guesses with different patterns
    for _ in range(5):
        h_rand = np.random.uniform(0, 1, n_points)
        sum_h = np.sum(h_rand)
        scale = target_sum / sum_h
        initial_guesses.append(h_rand * scale)
    
    # Define objective function with penalty
    def objective(h, dx, target_sum):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr) * dx
        penalty = 1e6 * np.abs(np.sum(h) - target_sum)
        return max_corr + penalty
    
    # Optimize each initial guess with differential evolution
    for guess in initial_guesses:
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = differential_evolution(
            objective,
            bounds,
            args=(dx, target_sum),
            strategy='best1bin',
            maxiter=100,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            polish=True,
        )
        if result.success:
            current_h = result.x
            current_c5 = compute_c5(current_h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
    
    # Perturbation phase on the best_h found so far
    num_perturbations = 500
    phase_shift = 0.2
    phase_decay = 0.98
    
    for perturbation_idx in range(num_perturbations):
        # Select a random segment to perturb
        segment_idx = np.random.randint(n_points)
        perturbed = best_h.copy()
        perturbed[segment_idx] += np.random.uniform(-0.2, 0.2)
        perturbed[segment_idx] = np.clip(perturbed[segment_idx], 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Optimize with differential evolution
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = differential_evolution(
            objective,

            current_c5 = compute_c5(current_h, dx)

                best_c5 = current_c5
                best_h = current_h
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points

def compute_c5(h, dx):
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return np.max(corr) * dx
