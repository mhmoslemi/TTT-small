    h_two = np.zeros(n_points)
    segment_length = n_points // 2
    h_two[:segment_length] = 1.0
    h_two[segment_length:] = 0.5
    sum_h = np.sum(h_two)

    h_six[2*segment_length:3*segment_length] = 1.0
    h_six[3*segment_length:4*segment_length] = 0.5
    h_six[4*segment_length:5*segment_length] = 1.0

    sum_h = np.sum(h_six)

    # Seven-segment pattern
    h_seven = np.zeros(n_points)
    segment_length = n_points // 7
    h_seven[:segment_length] = 1.0
    h_seven[segment_length:2*segment_length] = 0.5
    h_seven[2*segment_length:3*segment_length] = 1.0
    h_seven[3*segment_length:4*segment_length] = 0.5
    h_seven[4*segment_length:5*segment_length] = 1.0

    sum_h = np.sum(h_seven)

    # Optimizing each initial guess
    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 1500}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Perturbation phase on the best_h found so far
    num_perturbations = 500
    phase_shift = 0.15
    phase_decay = 0.96
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h

        elapsed = time.time() - start_time

def compute_c5(h, dx):

    corr = np.correlate(h, one_minus_h, mode='full')
    return np.max(corr) * dx
