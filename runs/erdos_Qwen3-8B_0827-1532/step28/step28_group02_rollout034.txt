            h[np.random.randint(n_points)] -= 0.01
        initial_guesses.append(h)
    
    # Add structured patterns
    # Pattern 1: 1 on one segment of length 1, 0.5 on one segment of length 0.5
    h = np.zeros(n_points)
    segment_length_1 = int(1.0 * n_points / 2.0)
    segment_length_0_5 = int(0.5 * n_points / 2.0)
    h[0:segment_length_1] = 1.0
    h[segment_length_1:segment_length_1 + segment_length_0_5] = 0.5
    current_sum = np.sum(h)
    scale = target_sum / current_sum
    h = h * scale
    initial_guesses.append(h)
    
    # Pattern 2: 1 on two segments of length 0.5 each, 0.5 on two segments of length 0.5 each
    h = np.zeros(n_points)
    h[0:50] = 1.0
    h[50:100] = 0.5
    h[100:150] = 0.5
    h[150:200] = 0.0
    current_sum = np.sum(h)
    scale = target_sum / current_sum
    h = h * scale
    initial_guesses.append(h)
    
    # Optimizing each initial guess with a more efficient method
    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP with tighter tolerances
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-12, 'maxiter': 3000, 'eps': 1e-8}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Perturbation phase with more systematic exploration
    num_perturbations = 500
    phase_shift = 0.25
    phase_decay = 0.98
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP with tighter tolerances
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

            objective,

            method='SLSQP',
            bounds=bounds,

            options={'ftol': 1e-12, 'maxiter': 500, 'eps': 1e-8}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points

def compute_c5(h, dx):
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return np.max(corr) * dx
