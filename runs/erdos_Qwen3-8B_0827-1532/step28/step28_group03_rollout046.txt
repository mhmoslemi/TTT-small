        perturbed = best_h.copy()
        perturbed[segment_idx] += np.random.uniform(-0.2, 0.2)
        perturbed[segment_idx] = np.clip(perturbed[segment_idx], 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective_perturb(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Run differential evolution on the perturbed guess
        bounds_perturb = [(0.0, 1.0) for _ in range(n_points)]
        constraints_perturb = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result_perturb = differential_evolution(
            objective_perturb,
            bounds_perturb,
            constraints=constraints_perturb,
            popsize=10,
            maxiter=50,
            tol=1e-12,
            mutation=(0.5, 1),
            recombination=0.9,
            seed=seed
        )
        
        if result_perturb.success and result_perturb.fun < best_c5:
            best_c5 = result_perturb.fun
            best_h = result_perturb.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points

def compute_c5(h, dx):
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return np.max(corr) * dx
