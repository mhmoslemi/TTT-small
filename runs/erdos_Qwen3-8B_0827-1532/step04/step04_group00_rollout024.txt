            return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points

def optimize_single(guess, dx, n_points):
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Run optimization
    result = minimize(
        objective,
        guess,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint},
        options={'ftol': 1e-9, 'maxiter': 1000}
    )
    
    if result.success:
        return result.x, result.fun
    else:
        return None, float('inf')
