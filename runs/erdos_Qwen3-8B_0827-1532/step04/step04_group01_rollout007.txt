import numpy as np
from scipy.optimize import minimize
import time
from multiprocessing import Pool, cpu_count

def run(seed=42, budget_s=100, **kwargs):
    # Predefined initial guess
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate multiple diverse initial guesses
    num_initial_guesses = 20  # More diverse starting points
    initial_guesses = [initial_guess]
    
    # Generate random initial guesses
    for _ in range(num_initial_guesses - 1):
        h = np.random.uniform(0, 1, n_points)
        current_sum = np.sum(h)
        scale = target_sum / current_sum
        h = h * scale
        initial_guesses.append(h)
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    start_time = time.time()
    
    # Function to optimize a single initial guess
    def optimize_single(guess):
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-8, 'maxiter': 500}
        )
        return result.x, result.fun if result.success else float('inf')
    
    # Use multiprocessing to parallelize optimizations
    with Pool(processes=min(cpu_count(), len(initial_guesses))) as pool:
        results = pool.map(optimize_single, initial_guesses)
    
    # Evaluate all results and find the best
    for h, c5 in results:
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    
    # Check elapsed time and return best solution
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
