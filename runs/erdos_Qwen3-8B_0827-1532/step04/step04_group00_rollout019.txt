import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 200  # Increased resolution for better optimization
    dx = 2.0 / n_points
    
    # Generate initial guesses
    num_initial_guesses = 20
    initial_guesses = []
    
    # Use initial_h_values if available for a structured starting point
    if 'initial_h_values' in globals():
        x_initial = np.linspace(0, 2, len(initial_h_values))
        x_new = np.linspace(0, 2, n_points)
        initial_guess = np.interp(x_new, x_initial, initial_h_values)
        initial_guess = initial_guess / np.sum(initial_guess) * (n_points / 2)
        initial_guesses.append(initial_guess)
    
    # Generate diverse random initial guesses
    for _ in range(num_initial_guesses - 1):
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
        initial_guesses.append(h)
    
    # Objective function: maximize overlap integral
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint: ensure integral of h is 1
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Run optimizations for each initial guess
    for guess in initial_guesses:
        # Run optimization with L-BFGS-B for faster convergence
        result = minimize(
            objective,
            guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-8, 'maxiter': 1000}
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # Fallback to first guess if no improvement found
    if best_h is None:
        result = minimize(
            objective,
            initial_guesses[0],
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-8, 'maxiter': 1000}
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points
