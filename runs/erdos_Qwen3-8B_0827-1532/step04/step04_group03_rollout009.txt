import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = 100
    dx = 2.0 / n_points
    np.random.seed(seed)
    
    # Generate diverse initial guesses
    initial_guesses = []
    # 1. Use the provided initial_h_values
    initial_guesses.append(initial_h_values)
    # 2. Symmetric random initial guess
    random_initial = np.zeros(n_points)
    selected_pairs = set()
    while len(selected_pairs) < n_points // 4:
        i = np.random.randint(0, n_points)
        j = n_points - 1 - i
        if i != j and i not in selected_pairs and j not in selected_pairs:
            selected_pairs.add(i)
            selected_pairs.add(j)
    for i in selected_pairs:
        random_initial[i] = 1
    initial_guesses.append(random_initial)
    # 3. Structured initial guess (every other interval, symmetric)
    structured_initial = np.zeros(n_points)
    structured_initial[::2] = 1
    initial_guesses.append(structured_initial)
    # 4. Evenly spaced initial guess (symmetric)
    evenly_spaced_initial = np.zeros(n_points)
    for i in range(n_points // 2):
        pos = int(i * n_points / (n_points // 2))
        evenly_spaced_initial[pos] = 1
        evenly_spaced_initial[n_points - 1 - pos] = 1
    initial_guesses.append(evenly_spaced_initial)
    # 5. Perturbations of initial_h_values
    for _ in range(15):  # More perturbations
        perturb = np.random.normal(0, 0.05, n_points)
        perturbed = initial_h_values + perturb
        perturbed = np.clip(perturbed, 0, 1)
        sum_perturbed = np.sum(perturbed)
        if sum_perturbed != 0:
            perturbed = perturbed * (n_points / 2) / sum_perturbed
        initial_guesses.append(perturbed)
    # 6. Additional structured guesses
    for m in [3, 4, 5]:
        structured = np.zeros(n_points)
        for i in range(n_points):
            if i % m == 0:
                structured[i] = 1
        structured = np.concatenate([structured, structured[::-1]])[:n_points]
        initial_guesses.append(structured)
    # 7. Random cluster initial guess
    cluster_initial = np.zeros(n_points)
    for _ in range(5):
        center = np.random.randint(0, n_points)
        for i in range(max(0, center - 5), min(n_points, center + 6)):
            cluster_initial[i] = 1
    cluster_initial = np.clip(cluster_initial, 0, 1)
    sum_cluster = np.sum(cluster_initial)
    if sum_cluster != 0:
        cluster_initial = cluster_initial * (n_points / 2) / sum_cluster
    initial_guesses.append(cluster_initial)
    
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    
    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    for guess in initial_guesses:
        # Perturb the guess slightly to avoid local optima
        perturb = np.random.normal(0, 0.01, n_points)
        perturbed_guess = np.clip(guess + perturb, 0, 1)
        sum_perturbed = np.sum(perturbed_guess)
        if sum_perturbed != 0:
            perturbed_guess = perturbed_guess * (n_points / 2) / sum_perturbed
        # Optimize
        result = minimize(
            objective,
            perturbed_guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraint,
            options={'maxiter': 500, 'ftol': 1e-9, 'eps': 1e-9}
        )
        current_h = result.x
        current_c5 = result.fun
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h
        # Check time
        if time.time() - start_time > budget_s:
            break
    
    return (best_h, best_c5, n_points)
