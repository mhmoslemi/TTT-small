from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Use the provided initial guess
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate multiple diverse initial guesses
    num_perturbations = 20  # Increased number of perturbations for better exploration
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Function to generate a random initial guess
    def generate_random_guess(n_points, target_sum):
        h = np.random.uniform(0, 1, n_points)
        h = h * (target_sum / np.sum(h))
        return h
    
    # Function to generate a structured initial guess (alternating pattern)
    def generate_structured_guess(n_points, target_sum):
        h = np.zeros(n_points)
        for i in range(n_points):
            if i % 3 == 0:
                h[i] = 1.0
            else:
                h[i] = 0.0
        h = h * (target_sum / np.sum(h))
        return h
    
    # Generate initial guesses: original, random, and structured
    initial_guesses = [initial_guess.copy(), generate_random_guess(n_points, target_sum), generate_structured_guess(n_points, target_sum)]
    
    for perturbation_idx in range(num_perturbations):
        # Select a different initial guess for each perturbation
        if perturbation_idx < len(initial_guesses):
            perturbed = initial_guesses[perturbation_idx]
        else:
            # Generate a new random guess for additional diversity
            perturbed = generate_random_guess(n_points, target_sum)
        
        # Objective function to minimize C5
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization with tighter tolerances and more iterations
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-8, 'maxiter': 1000}
        )
        
        # Check if this result is better
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
