            sum_perturbed = np.sum(perturbed_guess)
            if sum_perturbed != 0:
                perturbed_guess = perturbed_guess * (n_points / 2) / sum_perturbed
            # Submit to executor
            futures.append(executor.submit(minimize, 
                                          objective, 
                                          perturbed_guess, 
                                          method='L-BFGS-B', 
                                          bounds=bounds, 
                                          constraints=constraint, 
                                          options={'maxiter': 500, 'ftol': 1e-9, 'eps': 1e-9}))
        
        # Process results
        for future in as_completed(futures):
            try:
                result = future.result()
                current_h = result.x
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h
            except Exception as e:
                # Handle any exceptions, but for simplicity, ignore
                pass
            
            # Check time
            if time.time() - start_time > budget_s:
                break
    
    return (best_h, best_c5, n_points)
