import numpy as np
from scipy.optimize import minimize

def objective(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=100, **kwargs):
    n_points = 80
    dx = 2.0 / n_points

    # Generate random initial guesses

        current_sum = np.sum(h_initial)
        scale = (n_points / 2) / current_sum
        h_initial = h_initial * scale
        initial_guesses.append(h_initial)
    # Add a simple piecewise initial guess
    h_initial = np.zeros(n_points)
    h_initial[:n_points//2] = 1.0
    initial_guesses.append(h_initial)
    # Perturb the initial guesses
    num_starts = 50
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    
    for i in range(num_starts):
        np.random.seed(seed + i)
        # Select a random initial guess
        h_perturbed = initial_guesses[np.random.randint(len(initial_guesses))]

        h_perturbed = h_perturbed + np.random.uniform(-0.15, 0.15, n_points)

        current_sum = np.sum(h_perturbed)
        scale = (n_points / 2) / current_sum
        h_perturbed = h_perturbed * scale
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
