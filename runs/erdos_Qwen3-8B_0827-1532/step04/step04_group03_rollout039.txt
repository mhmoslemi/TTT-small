            selected_pairs.add(j)
    for i in selected_pairs:
        random_initial[i] = 1
    initial_guesses.append(random_initial)
    # 3. Structured initial guess (every other interval, symmetric)
    structured_initial = np.zeros(n_points)
    structured_initial[::2] = 1
    initial_guesses.append(structured_initial)
    # 4. Evenly spaced initial guess (symmetric)
    evenly_spaced_initial = np.zeros(n_points)
    for i in range(n_points // 2):
        pos = int(i * n_points / (n_points // 2))
        evenly_spaced_initial[pos] = 1
        evenly_spaced_initial[n_points - 1 - pos] = 1
    initial_guesses.append(evenly_spaced_initial)
    # 5. Perturbations of initial_h_values
    for _ in range(5):  # More perturbations
        perturb = np.random.normal(0, 0.05, n_points)
        perturbed = initial_h_values + perturb
        perturbed = np.clip(perturbed, 0, 1)
        sum_perturbed = np.sum(perturbed)
        if sum_perturbed != 0:
            perturbed = perturbed * (n_points / 2) / sum_perturbed
        initial_guesses.append(perturbed)
    
    # Normalize initial guesses
    normalized_guesses = []
    for guess in initial_guesses:
        sum_guess = np.sum(guess)
        if sum_guess == 0:
            continue
        normalized = guess * (n_points / 2) / sum_guess
        normalized_guesses.append(normalized)
    
    # Define objective function with normalization
    def objective(h):
        sum_h = np.sum(h)
        if sum_h == 0:
            return float('inf')
        h_normalized = h * (n_points / 2) / sum_h
        corr = correlate(h_normalized, 1 - h_normalized, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    
    # Run differential evolution with multiple initial guesses
    for guess in normalized_guesses[:5]:  # Limit to 5 initial guesses for efficiency
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=15,
            mutation=(0.5, 1),
            recombination=0.9,
            tol=1e-9,
            maxiter=100,
            polish=True,
            disp=False,
            x0=guess
        )
        current_h = result.x
        current_c5 = result.fun
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h
        # Check time
        if time.time() - start_time > budget_s:
            break
    
    return (best_h, best_c5, n_points)
