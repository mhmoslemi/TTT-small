from scipy.signal import correlate
import time

def run(seed=42, budget_s=100, **kwargs):

    for i in range(n_points // 2):
        pos = int(i * n_points / (n_points // 2))
        evenly_spaced_symmetric[pos] = 1
        evenly_spaced_symmetric[n_points - 1 - pos] = 1
    initial_guesses.append(evenly_spaced_symmetric)

            symmetric_perturbed[i] = perturbed[i]
            symmetric_perturbed[n_points - 1 - i] = perturbed[i]
        initial_guesses.append(symmetric_perturbed)
    
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    
    def objective(h_half):
        h = np.zeros(n_points)

        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    constraint = {'type': 'eq', 'fun': lambda h_half: np.sum(h_half) - (n_points // 2) / 2}
    bounds = [(0.0, 1.0) for _ in range(n_points // 2)]
