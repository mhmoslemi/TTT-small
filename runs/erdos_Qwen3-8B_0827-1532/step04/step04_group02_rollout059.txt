import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate multiple initial guesses: original, random, and symmetric
    num_initial_guesses = 10
    initial_guesses = [initial_guess.copy()]
    
    # Random initial guesses
    for _ in range(num_initial_guesses - 1):
        random_guess = np.random.rand(n_points)
        random_guess /= np.sum(random_guess)
        random_guess *= target_sum
        initial_guesses.append(random_guess)
    
    # Symmetric initial guess
    symmetric_guess = np.zeros(n_points)
    for i in range(n_points):
        symmetric_guess[i] = 0.5
    symmetric_guess = np.random.rand(n_points)
    symmetric_guess /= np.sum(symmetric_guess)
    symmetric_guess *= target_sum
    initial_guesses.append(symmetric_guess)
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    start_time = time.time()
    
    for guess in initial_guesses:
        # Perturb the guess with larger noise to explore more
        noise = np.random.normal(0, 0.05, n_points)
        perturbed = guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function to minimize C5
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

        result = minimize(

            perturbed,
            method='COBYLA',
            bounds=bounds,
            constraints=cons,

        )
        
        # Check if this result is better
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check elapsed time

    return best_h, best_c5, n_points
