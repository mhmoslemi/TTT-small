import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Smaller n_points for faster computation
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate initial guess based on provided initial_h_values
    x_initial = np.linspace(0, 2, len(initial_h_values))
    x_new = np.linspace(0, 2, n_points)

    initial_guess = initial_guess / np.sum(initial_guess) * target_sum
    
    # Define objective function with penalty for constraint violation
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        # Penalize deviation from the required integral
        penalty = 1e4 * (np.sum(h) - target_sum)**2
        return max_corr * dx + penalty
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    
    # Run differential evolution to find a good initial guess
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        maxiter=100,
        disp=False
    )
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        # If time is up, return the initial guess
        return initial_guess, objective(initial_guess), n_points
    
    best_h = result.x
    best_c5 = result.fun
