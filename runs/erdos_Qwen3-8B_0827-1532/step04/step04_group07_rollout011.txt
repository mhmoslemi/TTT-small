from scipy.optimize import minimize

    for i in range(n_points//2):

        initial_guesses.append(pattern)
    
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    
    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
