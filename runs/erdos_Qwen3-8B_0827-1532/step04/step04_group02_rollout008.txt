import numpy as np
from scipy.optimize import minimize
import time
import multiprocessing

def run(seed=42, budget_s=100, **kwargs):
    # Use the provided initial guess
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate multiple initial guesses: original and random
    num_initial_guesses = 20  # Increased number of initial guesses
    initial_guesses = [initial_guess.copy()]
    for _ in range(num_initial_guesses - 1):
        # Generate random initial guess with sum constraint
        random_guess = np.random.rand(n_points)
        random_guess /= np.sum(random_guess)
        random_guess *= target_sum
        initial_guesses.append(random_guess)
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    start_time = time.time()
    
    # Define the objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define bounds and constraints
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    
    # Function to perform optimization for a single guess
    def optimize_guess(guess):
        # Perturb the guess with small noise
        noise = np.random.normal(0, 0.01, n_points)  # Smaller noise for finer perturbation
        perturbed = guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Perform optimization
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-8, 'maxiter': 1000}  # Tighter tolerance and more iterations
        )
        return result
    
    # Use multiprocessing to parallelize the optimization
    pool = multiprocessing.Pool(processes=10)  # Use all available CPUs
    results = []
    
    for guess in initial_guesses:
        results.append(pool.apply_async(optimize_guess, (guess,)))
    
    # Collect results
    for result in results:
        res = result.get()
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x
    
    pool.close()
    pool.join()
    
    # Check elapsed time
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
