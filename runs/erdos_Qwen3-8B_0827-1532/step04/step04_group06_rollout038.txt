from scipy.optimize import minimize

def objective(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=100, **kwargs):
    n_points = 80

    # Generate multiple initial guesses

    # Add a simple piecewise initial guess
    h_initial = np.zeros(n_points)
    h_initial[:n_points//2] = 1.0
    initial_guesses.append(h_initial)
    # Add a sine-based initial guess
    x = np.linspace(0, 2, n_points)
    sine_guess = 0.5 * (1 + np.sin(np.pi * x))
    initial_guesses.append(sine_guess)
    # Add a random initial guess
    random_guess = np.random.rand(n_points)
    random_guess = (random_guess - 0.5) * 2  # Scale to [0, 1]
    initial_guesses.append(random_guess)
    # Perturb the initial guesses
    num_starts = 100
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    
    for i in range(num_starts):
        np.random.seed(seed + i)
        # Select a random initial guess
        h_perturbed = initial_guesses[np.random.randint(len(initial_guesses))]
        # Perturb with uniform noise
        h_perturbed = h_perturbed + np.random.uniform(-0.1, 0.1, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        # Adjust to meet the sum constraint
        current_sum = np.sum(h_perturbed)
        scale = (n_points / 2) / current_sum
        h_perturbed = h_perturbed * scale
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
        
        # Optimize with more iterations and tighter tolerances
        result = minimize(
            objective, 
            h_perturbed, 
            args=(dx,), 
            method='SLSQP',
