import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        scaled_h = h * (target_sum / np.sum(h))
        one_minus_h = 1.0 - scaled_h
        corr = np.correlate(scaled_h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    for i in range(5):  # Run 5 DE optimizations
        np.random.seed(seed + i)
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=15,
            mutation=(0.5, 1),
            recombination=0.8,
            tol=1e-8,
            maxiter=100,
            seed=seed + i
        )
        if result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
