import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = 100  # Increased resolution for better optimization
    dx = 2.0 / n_points
    np.random.seed(seed)
    
    # Use the provided initial_h_values as the main starting point
    initial_guesses = [initial_h_values]
    
    # Perturb the initial guess slightly to avoid local optima
    for _ in range(5):
        perturb = np.random.normal(0, 0.02, n_points)
        perturbed = initial_h_values + perturb
        perturbed = np.clip(perturbed, 0, 1)
        sum_perturbed = np.sum(perturbed)
        if sum_perturbed != 0:
            perturbed = perturbed * (n_points / 2) / sum_perturbed
        initial_guesses.append(perturbed)
    
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    
    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    for guess in initial_guesses:
        # Optimize with COBYLA for better constraint handling
        result = minimize(
            objective,
            guess,
            method='COBYLA',
            bounds=bounds,
            constraints=constraint,
            options={'maxiter': 500, 'disp': False}
        )
        current_h = result.x
        current_c5 = result.fun
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h
        # Check time
        if time.time() - start_time > budget_s:
            break
    
    return (best_h, best_c5, n_points)
