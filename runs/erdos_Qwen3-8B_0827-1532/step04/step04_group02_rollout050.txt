            if (0 <= x <= 0.5) or (1.5 <= x <= 2.0):
                h[i] = 1.0

        current_sum = np.sum(h)
        scale = target_sum / current_sum
        h = h * scale
        return h

    initial_guesses = [initial_guess.copy()]

        def objective(h):

            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
