from scipy.optimize import minimize
from scipy.signal import correlate
import time
import multiprocessing

def run(seed=42, budget_s=100, **kwargs):

    dx = 2.0 / n_points

        if i != j and i not in selected_pairs and j not in selected_pairs:
            selected_pairs.add(i)
            selected_pairs.add(j)
    for i in selected_pairs:
        random_initial[i] = 1
    initial_guesses.append(random_initial)
    # 3. Structured initial guess (alternating 1s and 0s)
    structured_initial = np.zeros(n_points)

        pos = int(i * n_points / 3)

            if 0 <= j < n_points:
                clustered_initial[j] = 1
    initial_guesses.append(clustered_initial)
    # 5. Perturbations of initial_h_values with varying magnitudes
    for perturb_mag in [0.05, 0.1]:

        sum_perturbed = np.sum(perturbed)
        if sum_perturbed != 0:

        initial_guesses.append(perturbed)
    
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    
    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Use multiprocessing to parallelize optimizations
    pool = multiprocessing.Pool(processes=10)
    results = []
    
    for guess in initial_guesses:

        if sum_perturbed != 0:
            perturbed_guess = perturbed_guess * (n_points / 2) / sum_perturbed
        # Submit to pool
        results.append(pool.apply_async(minimize,
