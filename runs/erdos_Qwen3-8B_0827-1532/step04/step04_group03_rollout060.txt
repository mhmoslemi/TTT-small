            selected_pairs.add(i)
            selected_pairs.add(j)
    for i in selected_pairs:
        random_initial[i] = 1
    initial_guesses.append(random_initial)

        evenly_spaced_initial[pos] = 1
        evenly_spaced_initial[n_points - 1 - pos] = 1
    initial_guesses.append(evenly_spaced_initial)
    # 5. Perturbations of initial_h_values
    for _ in range(20):  # More perturbations

        perturbed = initial_h_values + perturb

        if sum_perturbed != 0:
            perturbed = perturbed * (n_points / 2) / sum_perturbed
        initial_guesses.append(perturbed)
    # 6. Additional structured patterns
    # 6a. Alternating 1s and 0s with offset
    alt_initial = np.zeros(n_points)
    for i in range(n_points):
        if i % 3 == 0:
            alt_initial[i] = 1
    initial_guesses.append(alt_initial)
    # 6b. Two clusters
    cluster_initial = np.zeros(n_points)
    cluster_initial[:n_points//3] = 1
    cluster_initial[-n_points//3:] = 1

    # 6c. Random sparse
    sparse_initial = np.zeros(n_points)
    for _ in range(n_points//5):

        sparse_initial[i] = 1
    initial_guesses.append(sparse_initial)
    
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    
    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    for guess in initial_guesses:
        # Perturb the guess slightly to avoid local optima
        perturb = np.random.normal(0, 0.01, n_points)
        perturbed_guess = np.clip(guess + perturb, 0, 1)
        sum_perturbed = np.sum(perturbed_guess)
        if sum_perturbed != 0:
