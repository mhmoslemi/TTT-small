        max_corr = np.max(corr)
        return max_corr * dx
    
    constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    for guess in initial_guesses:
        # Perturb the guess slightly to avoid local optima
        perturb = np.random.normal(0, 0.1, n_points)
        perturbed_guess = np.clip(guess + perturb, 0, 1)
        sum_perturbed = np.sum(perturbed_guess)
        if sum_perturbed != 0:
            perturbed_guess = perturbed_guess * (n_points / 2) / sum_perturbed
        # Optimize
        result = minimize(
            objective,
            perturbed_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=constraint,
            options={'maxiter': 1000, 'ftol': 1e-9, 'eps': 1e-9}
        )
        current_h = result.x
        current_c5 = result.fun
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h
        # Check time
        if time.time() - start_time > budget_s:
            break
    
    return (best_h, best_c5, n_points)
