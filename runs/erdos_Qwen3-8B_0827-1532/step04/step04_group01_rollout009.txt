from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate multiple perturbations with different noise strategies
    num_perturbations = 20
    for perturbation_idx in range(num_perturbations):
        if perturbation_idx < 5:
            noise = np.random.uniform(-0.2, 0.2, n_points)
        elif perturbation_idx < 10:
            noise = np.random.uniform(-0.1, 0.1, n_points)
        else:
            noise = np.random.uniform(-0.05, 0.05, n_points)
        
        perturbed = initial_guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 1000}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
