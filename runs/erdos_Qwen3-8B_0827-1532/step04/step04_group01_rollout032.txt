import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Precompute dx and target sum
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    initial_guesses = []
    initial_guesses.append(initial_h_values.copy())  # Provided initial guess

    half = n_points // 2

    initial_guesses.append(binary)

    alt = np.zeros(n_points)
    for i in range(n_points):
        alt[i] = 1 if i % 2 == 0 else 0

    scale = target_sum / current_sum
    alt = alt * scale
    initial_guesses.append(alt)

    rand = np.random.uniform(0, 1, n_points)

    scale = target_sum / current_sum
    rand = rand * scale
    initial_guesses.append(rand)
    
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    start_time = time.time()
    
    for guess in initial_guesses:

        guess = np.clip(guess, 0, 1)
        current_sum = np.sum(guess)
        scale = target_sum / current_sum
        guess = guess * scale

        def objective(h):

            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Optimization bounds and constraints
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-8, 'maxiter': 500}
        )
        
        # Update best solution
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check time budget
        if time.time() - start_time >= budget_s:
            break
    
    return best_h, best_c5, n_points
