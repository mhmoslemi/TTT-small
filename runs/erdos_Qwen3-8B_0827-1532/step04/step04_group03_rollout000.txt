            perturbed = perturbed * (n_points / 2) / sum_perturbed
        initial_guesses.append(perturbed)
    
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    
    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    for guess in initial_guesses:
        # Perturb the guess slightly to avoid local optima
        perturb = np.random.normal(0, 0.01, n_points)
        perturbed_guess = np.clip(guess + perturb, 0, 1)
        sum_perturbed = np.sum(perturbed_guess)
        if sum_perturbed != 0:
            perturbed_guess = perturbed_guess * (n_points / 2) / sum_perturbed
        # Optimize
        result = differential_evolution(
            objective,
            bounds,
            constraints=constraint,
            popsize=20,
            maxiter=100,
            tol=1e-5,
            mutation=(0.5, 1),
            recombination=0.8,
            strategy='best1bin'
        )
        current_h = result.x
        current_c5 = result.fun
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h
        # Check time
        if time.time() - start_time > budget_s:
            break
    
    return (best_h, best_c5, n_points)
