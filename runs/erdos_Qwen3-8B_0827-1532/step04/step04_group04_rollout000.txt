import numpy as np
from scipy.optimize import differential_evolution

def objective(h, dx, n_points):
    current_sum = np.sum(h)
    if current_sum == 0:
        return float('inf')
    scale_factor = (n_points / 2) / current_sum
    adjusted_h = h * scale_factor
    adjusted_h = np.clip(adjusted_h, 0, 1)
    corr = np.correlate(adjusted_h, 1 - adjusted_h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=100, **kwargs):
    n_points = 80
    dx = 2.0 / n_points
    
    # Bounds for each variable
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Run differential evolution with multiple restarts
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    elapsed_time = 0.0
    num_restarts = 10  # Number of independent DE runs
    
    for i in range(num_restarts):
        np.random.seed(seed + i)
        result = differential_evolution(
            objective,
            bounds,
            args=(dx, n_points),
            strategy='best1bin',
            maxiter=100,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed + i
        )
        current_c5 = objective(result.x, dx, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x
        # Check elapsed time and break if budget is exceeded
        if elapsed_time >= budget_s:
            break
        elapsed_time += 0.1  # Approximate time per iteration
    
    return best_h, best_c5, n_points
