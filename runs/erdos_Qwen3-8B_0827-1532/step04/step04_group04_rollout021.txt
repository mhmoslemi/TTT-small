import numpy as np
from scipy.optimize import minimize

def objective(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=100, **kwargs):
    n_points = 80
    dx = 2.0 / n_points
    
    # Define diverse initial guesses
    initial_guesses = [
        initial_h_values.copy(),  # Provided initial guess
        np.concatenate([np.ones(n_points//2), np.zeros(n_points//2)]),  # 1 in first half
        np.full(n_points, 0.5),  # Flat function
        np.zeros(n_points),  # All zeros (will be adjusted)
        np.ones(n_points),  # All ones (will be adjusted)
        np.random.rand(n_points),  # Random initial guess
        # Structured guess: alternating 1 and 0
        np.array([1.0 if i % 2 == 0 else 0.0 for i in range(n_points)]),
    ]
    
    num_starts = 50  # Increased number of starts
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    elapsed_time = 0.0
    
    for i in range(num_starts):
        np.random.seed(seed + i)
        # Randomly select an initial guess
        initial_idx = np.random.randint(len(initial_guesses))
        h_perturbed = initial_guesses[initial_idx].copy()
        
        # Add random perturbation
        perturbation = np.random.normal(0, 0.05, n_points)  # Larger perturbation
        h_perturbed += perturbation
        h_perturbed = np.clip(h_perturbed, 0, 1)
        
        # Adjust to meet the sum constraint
        current_sum = np.sum(h_perturbed)
        scale = (n_points / 2) / current_sum
        h_perturbed = h_perturbed * scale
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
        
        # Optimize with SLSQP method
        result = minimize(
            objective, 
            h_perturbed, 
            args=(dx,), 
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 300, 'ftol': 1e-9, 'xtol': 1e-9}
        )
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            # Fallback to perturbed guess if optimization fails
            current_c5 = objective(h_perturbed, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_perturbed
        
        # Check elapsed time and break if budget is exceeded
        if elapsed_time >= budget_s:
            break
        elapsed_time += 0.1  # Approximate time per iteration
    
    return best_h, best_c5, n_points
