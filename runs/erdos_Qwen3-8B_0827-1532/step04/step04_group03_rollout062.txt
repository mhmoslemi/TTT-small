    # 2. Random symmetric initial guess
    random_initial = np.zeros(half_n)
    selected_pairs = set()
    while len(selected_pairs) < half_n // 2:
        i = np.random.randint(0, half_n)
        j = half_n - 1 - i
        if i != j and i not in selected_pairs and j not in selected_pairs:
            selected_pairs.add(i)
            selected_pairs.add(j)
    for i in selected_pairs:
        random_initial[i] = 1
    initial_guesses.append(random_initial)
    # 3. Structured symmetric initial guess (every other interval)
    structured_initial = np.zeros(half_n)
    structured_initial[::2] = 1
    initial_guesses.append(structured_initial)
    # 4. Evenly spaced symmetric initial guess
    evenly_spaced_initial = np.zeros(half_n)
    for i in range(half_n // 2):
        pos = int(i * half_n / (half_n // 2))
        evenly_spaced_initial[pos] = 1
        evenly_spaced_initial[half_n - 1 - pos] = 1
    initial_guesses.append(evenly_spaced_initial)
    # 5. Perturbations of symmetric initial_h_values
    for _ in range(10):
        perturb = np.random.normal(0, 0.05, half_n)
        perturbed = symmetric_initial + perturb
        perturbed = np.clip(perturbed, 0, 1)
        sum_perturbed = np.sum(perturbed)
        if sum_perturbed != 0:
            perturbed = perturbed * (n_points / 4) / sum_perturbed
        initial_guesses.append(perturbed)
    
    best_c5 = float('inf')
    best_h_values = None
    start_time = time.time()
    
    def objective(h_values):
        # Construct full h
        full_h = np.zeros(n_points)
        for i in range(half_n):
            full_h[i] = h_values[i]
            full_h[n_points - 1 - i] = h_values[i]
        # Compute cross-correlation
        corr = correlate(full_h, 1 - full_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint: sum(h_values) = n_points / 4
    constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 4}
    bounds = [(0.0, 1.0) for _ in range(half_n)]
    
    for guess in initial_guesses:
        # Perturb the guess slightly to avoid local optima
        perturb = np.random.normal(0, 0.01, half_n)
        perturbed_guess = np.clip(guess + perturb, 0, 1)
        sum_perturbed = np.sum(perturbed_guess)
        if sum_perturbed != 0:
            perturbed_guess = perturbed_guess * (n_points / 4) / sum_perturbed
        # Optimize
        result = minimize(
            objective,
            perturbed_guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraint,
            options={'maxiter': 1000, 'ftol': 1e-9, 'eps': 1e-9}
        )
        current_h_values = result.x
        current_c5 = result.fun
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h_values = current_h_values
        # Check time
        if time.time() - start_time > budget_s:
            break
    
    # Construct the best full h
    best_h = np.zeros(n_points)
    for i in range(half_n):
        best_h[i] = best_h_values[i]
        best_h[n_points - 1 - i] = best_h_values[i]
    
    return (best_h, best_c5, n_points)
