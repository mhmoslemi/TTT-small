from scipy.optimize import minimize
import time

    base_guess = np.interp(x_new, x_initial, initial_h_values)
    base_guess = base_guess / np.sum(base_guess) * (n_points / 2)
    for _ in range(5):
        noise = np.random.normal(0, 0.02, n_points)
        perturbed = base_guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        scale_factor = (n_points / 2) / np.sum(perturbed)
        perturbed *= scale_factor
        initial_guesses.append(perturbed)
    
    # Add random guesses
    for _ in range(15):
        h = np.random.uniform(0, 1, n_points)
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h
        h_scaled = h * scale
        initial_guesses.append(h_scaled)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Run optimizations for each initial guess
    for guess in initial_guesses:
        # Run optimization
        result = minimize(
            objective,
            guess,
            method='COBYLA',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 500, 'ftol': 1e-9}
        )
        elapsed = time.time() - start_time
