        alternating[::2] = 1
        initial_guesses.append(alternating)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Use parallel processing for optimization
    with ProcessPoolExecutor(max_workers=10) as executor:
        futures = []
        for guess in initial_guesses:
            future = executor.submit(
                optimize_guess,
                guess,
                objective,
                constraint,
                bounds
            )
            futures.append(future)
        
        # Process results as they complete
        for future in as_completed(futures):
            try:
                result = future.result()
                if result is not None:
                    current_c5, current_h = result
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = current_h
            except Exception as e:
                print(f"Optimization failed: {e}")
    
    # Fallback to single-threaded optimization if needed
    if best_h is None:
        result = minimize(
            objective,
            initial_guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-9, 'maxiter': 1000}
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points

def optimize_guess(guess, objective, constraint, bounds):
    result = minimize(
        objective,
        guess,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint},
        options={'ftol': 1e-9, 'maxiter': 1000}
    )
    if result.success:
        return (result.fun, result.x)
    else:
        return None
