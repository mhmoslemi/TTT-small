    initial_guess = np.clip(initial_guess * scale_factor, 0, 1)
    
    # Define bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Objective function
    def objective(h):
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.inf
        scale = (n_points / 2) / sum_h
        scaled_h = h * scale
        scaled_h = np.clip(scaled_h, 0, 1)
        one_minus_h = 1.0 - scaled_h
        corr = np.correlate(scaled_h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Run differential evolution
    start_time = time.time()
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=20,
        mutation=(0.5, 1),
        recombination=0.9,
        maxiter=100,
        tol=1e-9,
        polish=True,
        vectorized=True
    )
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        # If time is up, return the initial guess or the best found so far
        best_h = result.x
        best_c5 = result.fun
    else:
        # If time is not up, perhaps run more iterations or other steps
        best_h = result.x
        best_c5 = result.fun
    
    return best_h, best_c5, n_points
