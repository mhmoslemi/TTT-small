import numpy as np
from scipy.optimize import minimize

def objective(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=100, **kwargs):
    n_points = 80
    dx = 2.0 / n_points
    # Generate multiple initial guesses
    initial_guesses = []
    # Add the provided initial_h_values
    initial_guesses.append(initial_h_values.copy())
    # Add alternating pattern
    h_alternating = np.zeros(n_points)
    h_alternating[::2] = 1.0
    initial_guesses.append(h_alternating)
    # Add two-interval pattern
    h_two_intervals = np.zeros(n_points)
    h_two_intervals[:20] = 1.0
    h_two_intervals[40:60] = 1.0
    initial_guesses.append(h_two_intervals)
    # Add simple half-ones pattern
    h_initial = np.zeros(n_points)
    h_initial[:n_points//2] = 1.0
    initial_guesses.append(h_initial)
    # Perturb the initial guesses
    num_starts = 50
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    
    for i in range(num_starts):

        h_perturbed = initial_guesses[np.random.randint(len(initial_guesses))]
