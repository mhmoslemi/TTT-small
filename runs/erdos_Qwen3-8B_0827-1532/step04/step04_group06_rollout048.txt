from scipy.optimize import minimize
from multiprocessing import Pool, cpu_count

def objective(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def evaluate_and_scale(h, dx, target_sum):
    current_sum = np.sum(h)
    scale = target_sum / current_sum
    return h * scale

def optimize_single(h_initial, dx, target_sum, method='L-BFGS-B'):
    bounds = [(0.0, 1.0) for _ in range(len(h_initial))]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    result = minimize(

        options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8}
    )
    return result.x if result.success else h_initial
