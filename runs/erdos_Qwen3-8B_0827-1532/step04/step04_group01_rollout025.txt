import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Predefined initial guess
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate multiple diverse initial guesses
    def generate_initial_guesses():
        guesses = []
        # Add provided initial guess
        guesses.append(initial_guess)
        # Uniform distribution
        guesses.append(np.full(n_points, 0.5))
        # Binary-like (half 1s, half 0s)
        binary = np.zeros(n_points)
        binary[:n_points//2] = 1.0
        guesses.append(binary)
        # Random guess with correct sum
        random = np.random.uniform(0, 1, n_points)
        random = random * (target_sum / np.sum(random))
        guesses.append(random)
        return guesses
    
    initial_guesses = generate_initial_guesses()
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    for idx, guess in enumerate(initial_guesses):
        # Perturb the guess with different noise levels
        if idx == 0:
            # Original guess, no perturbation
            perturbed = guess
        else:
            # Add noise for diversity
            noise = np.random.uniform(-0.1, 0.1, n_points)
            perturbed = guess + noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
        
        # Objective function to minimize C5
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization with tighter tolerances and more iterations
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-8, 'maxiter': 500}
        )
        
        # Check if this result is better
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
