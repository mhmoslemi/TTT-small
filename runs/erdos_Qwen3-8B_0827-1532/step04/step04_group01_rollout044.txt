from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Use the provided initial guess
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate multiple perturbations with varying noise scales
    num_perturbations = 20  # Increased number of perturbations for better exploration
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    for perturbation_idx in range(num_perturbations):
        # Generate diverse initial guesses with varying noise scales
        if perturbation_idx < 5:
            # Larger noise for early perturbations to explore more
            noise = np.random.uniform(-0.25, 0.25, n_points)
        elif perturbation_idx < 10:
            # Medium noise for mid perturbations
            noise = np.random.uniform(-0.15, 0.15, n_points)
        else:
            # Smaller noise for later perturbations to refine
            noise = np.random.uniform(-0.05, 0.05, n_points)
        
        perturbed = initial_guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function to minimize C5
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization with tighter tolerances and more iterations
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 1000}
        )
        
        # Check if this result is better
        if result.success and result.fun < best_c5:

            best_h = result.x
