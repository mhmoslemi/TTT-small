import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Smaller n_points for faster evaluation
    dx = 2.0 / n_points
    
    # Generate diverse initial guesses
    num_initial_guesses = 20
    initial_guesses = []

        # Random initial guess with sum constraint
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
        initial_guesses.append(h)

    def objective(h):

        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint: sum(h) = n_points / 2
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    
    # Run optimizations with COBYLA for better exploration
    for guess in initial_guesses:
        result = minimize(
            objective,
            guess,
            method='COBYLA',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 500}
        )
        elapsed = time.time() - start_time

            break
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # Fallback to first guess if no improvement
    if best_h is None:
        result = minimize(
            objective,
            initial_guesses[0],
            method='COBYLA',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 500}
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points
