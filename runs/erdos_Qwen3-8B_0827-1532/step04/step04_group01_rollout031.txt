import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    # Set n_points
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate multiple initial guesses
    num_initial_guesses = 20
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    
    start_time = time.time()
    
    for i in range(num_initial_guesses):
        # Generate a random initial guess
        h_initial = np.random.uniform(0, 1, n_points)
        # Scale to meet the sum constraint
        h_initial = h_initial * (target_sum / np.sum(h_initial))
        
        # Define the objective function
        def objective(h_candidate):
            scaled_h = h_candidate * (target_sum / np.sum(h_candidate))
            one_minus_h = 1.0 - scaled_h
            corr = np.correlate(scaled_h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each parameter
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Perform differential evolution optimization
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=15,
            mutation=(0.5, 1.0),
            recombination=0.8,
            tol=1e-6,
            maxiter=100,
            polish=True,
            disp=False
        )
        
        # Check if this result is better
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
