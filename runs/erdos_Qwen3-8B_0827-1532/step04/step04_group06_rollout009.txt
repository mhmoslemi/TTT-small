    scale = (n_points / 2) / current_sum
    h_random = h_random * scale
    initial_guesses.append(h_random)
    # Add structured initial guesses
    for m in [2, 3, 4]:
        structured_h = generate_structured_initial(m, n_points, dx)
        initial_guesses.append(structured_h)
    # Add more initial guesses if needed
    
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    num_starts = 100  # Increased number of starts
    
    for i in range(num_starts):
        np.random.seed(seed + i)
        # Select a random initial guess
        h_perturbed = initial_guesses[np.random.randint(len(initial_guesses))]
        # Perturb with uniform noise
        h_perturbed = h_perturbed + np.random.uniform(-0.1, 0.1, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        # Adjust to meet the sum constraint
        current_sum = np.sum(h_perturbed)
        scale = (n_points / 2) / current_sum
        h_perturbed = h_perturbed * scale
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
        
        # Optimize with more iterations and tighter tolerances
        result = minimize(
            objective, 
            h_perturbed, 
            args=(dx,), 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 1000, 'ftol': 1e-9, 'xtol': 1e-9}
        )
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            # Fallback to perturbed guess if optimization fails
            current_c5 = objective(h_perturbed, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_perturbed
    
    return best_h, best_c5, n_points
