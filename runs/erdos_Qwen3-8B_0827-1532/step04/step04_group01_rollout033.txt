from scipy.optimize import minimize
import time

    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    for perturbation_idx in range(num_perturbations):
        # Generate diverse perturbations: some with larger noise, some with structured patterns
        if perturbation_idx < 5:

        elif perturbation_idx < 10:
            # Medium noise for moderate exploration
            noise = np.random.uniform(-0.15, 0.15, n_points)
        else:
            # Small noise for fine-tuning
            noise = np.random.uniform(-0.05, 0.05, n_points)
        
        # Apply noise and scale to maintain integral constraint
        perturbed = initial_guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function to minimize C5
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization with tighter tolerances and more iterations
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 1000}
        )
        
        # Check if this result is better
        if result.success and result.fun < best_c5:
