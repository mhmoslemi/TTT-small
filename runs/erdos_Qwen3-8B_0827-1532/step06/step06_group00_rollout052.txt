            periodic_h[i] = 1.0
        else:
            periodic_h[i] = 0.0
    periodic_h = periodic_h / np.sum(periodic_h) * (n_points / 2)
    initial_guesses.append(periodic_h)
    
    # Random initial guess

    def objective(h):

        return max_corr * dx
    
    # Constraint function
    def constraint(h):

    # Bounds for each h[i]

    for guess in initial_guesses:

        result = minimize(
