from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    # Use the provided initial guess
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate diverse initial guesses
    num_initial_guesses = 30
    initial_guesses = [initial_guess.copy()]
    
    # Add random noise perturbations
    for _ in range(num_initial_guesses - 1):
        noise = np.random.normal(0, 0.05, n_points)
        perturbed = initial_guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        initial_guesses.append(perturbed)
    
    # Add sinusoidal variations
    for _ in range(num_initial_guesses - len(initial_guesses)):
        freq = np.random.uniform(0.5, 2.0)
        phase = np.random.uniform(0, 2 * np.pi)
        x = np.linspace(0, 2, n_points)
        sinusoidal = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x + phase)
        sinusoidal = np.clip(sinusoidal, 0, 1)
        current_sum = np.sum(sinusoidal)
        scale = target_sum / current_sum
        sinusoidal = sinusoidal * scale
        initial_guesses.append(sinusoidal)
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    start_time = time.time()
    
    for guess in initial_guesses:
        # Ensure the guess is within bounds and scaled
        guess = np.clip(guess, 0, 1)
        current_sum = np.sum(guess)
        scale = target_sum / current_sum
        guess = guess * scale
        
        # Define objective function with penalty for sum constraint
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            penalty = 1e4 * (np.sum(h) - target_sum)**2
            return max_corr * dx + penalty
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Run differential evolution
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=15,
            mutation=(0.5, 1),

            tol=1e-6,
            maxiter=500,
            polish=True,
            disp=False
        )
        
        # Check if this result is better
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
