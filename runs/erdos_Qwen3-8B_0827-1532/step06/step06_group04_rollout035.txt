import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Use the provided initial guess
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate multiple diverse initial guesses
    num_perturbations = 20  # Reduced number of perturbations for efficiency
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Function to create structured initial guesses
    def create_two_interval_guess(n_points):
        dx = 2.0 / n_points
        h = np.zeros(n_points)
        # First interval: [0, 0.5]
        start = 0
        end = int(0.5 / dx)
        h[start:end+1] = 1.0
        # Second interval: [1.0, 1.5]
        start = int(1.0 / dx)
        end = int(1.5 / dx)
        h[start:end+1] = 1.0
        # Normalize the sum to target_sum
        current_sum = np.sum(h)
        scale = target_sum / current_sum
        h = h * scale
        return h
    
    # Generate structured initial guesses
    structured_guesses = [create_two_interval_guess(n_points)]
    
    # Add random initial guesses
    random_guesses = [np.random.uniform(0, 1, n_points) * (target_sum / np.sum(np.random.uniform(0, 1, n_points))) for _ in range(5)]
    
    # Combine all initial guesses
    all_initial_guesses = [initial_guess] + structured_guesses + random_guesses
    
    for perturbation_idx, guess in enumerate(all_initial_guesses):
        # Objective function to minimize C5
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization with tighter tolerances and more iterations
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 5000}
        )
        
        # Check if this result is better
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
