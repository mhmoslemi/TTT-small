from scipy.optimize import minimize
import time

    # Use the provided initial guess
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    num_initial_guesses = 20  # More diverse initial guesses
    initial_guesses = [initial_guess]
    
    # Generate random binary initial guesses
    for _ in range(num_initial_guesses - 1):
        h = np.random.choice([0, 1], size=n_points)
        while np.sum(h) != target_sum:
            h = np.random.choice([0, 1], size=n_points)
        initial_guesses.append(h)
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Define objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define constraints
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    
    # Function to perform optimization on a single initial guess
    def optimize_guess(guess):
        # Scale to maintain sum constraint (though already satisfied)
        result = minimize(
            objective,
            guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 200}
        )
        return result
    
    # Use ThreadPoolExecutor to parallelize optimizations
    with ThreadPoolExecutor(max_workers=10) as executor:

            if result.success and result.fun < best_c5:

                best_h = result.x

            if elapsed >= budget_s:
