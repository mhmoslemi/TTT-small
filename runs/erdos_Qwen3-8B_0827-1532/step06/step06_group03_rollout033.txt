            one_minus_h = 1.0 - result.x
            corr = np.correlate(result.x, one_minus_h, mode='full')
            max_corr = np.max(corr)
            best_c5 = max_corr * dx
            best_h = result.x
        else:
            # Fallback to initial guess
            best_h = initial_guess
            # Compute the objective for the initial guess
            one_minus_h = 1.0 - best_h
            corr = np.correlate(best_h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            best_c5 = max_corr * dx
    
    return best_h, best_c5, n_points
