import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Use smaller n_points for faster optimization
    n_points = 50
    dx = 2.0 / n_points
    target_sum = n_points / 2  # 25

    # Generate diverse initial guesses
    initial_guesses = []
    
    # Structured initial guess: sine wave
    x = np.linspace(0, 2, n_points)
    h_sine = 0.5 + 0.5 * np.sin(np.pi * x)
    h_sine = np.clip(h_sine, 0, 1)
    h_sine = h_sine * (target_sum / np.sum(h_sine))
    initial_guesses.append(h_sine)
    
    # Random initial guesses
    for _ in range(10):
        random_guess = np.random.uniform(0, 1, n_points)
        random_guess = random_guess * (target_sum / np.sum(random_guess))
        initial_guesses.append(random_guess)
    
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    start_time = time.time()
    
    for guess in initial_guesses:
        # Objective function to minimize max correlation
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Optimize with tighter tolerances and fewer iterations
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 1000}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
