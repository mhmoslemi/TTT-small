from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    target_sum = n_points / 2

    num_initial_guesses = 50
    initial_guesses = [initial_guess.copy()]
    
    # Generate sinusoidal initial guess
    x = np.linspace(0, 2, n_points)
    sinusoidal = 0.5 + 0.5 * np.sin(np.pi * x)
    integral = np.trapz(sinusoidal, x)
    scale = 1.0 / integral
    sinusoidal_scaled = sinusoidal * scale
    initial_guesses.append(sinusoidal_scaled)
    
    # Generate piecewise alternating initial guess
    piecewise = np.zeros(n_points)
    piecewise[::2] = 1.0
    sum_piecewise = np.sum(piecewise)
    if n_points % 2 == 1:
        # Adjust last element to meet target sum
        piecewise[-1] = (target_sum - sum_piecewise + 1) / 1
    initial_guesses.append(piecewise)
    
    # Generate random noise perturbations
    for _ in range(num_initial_guesses - len(initial_guesses)):
        # Add random noise
        noise = np.random.normal(0, 0.05, n_points)
        perturbed = initial_guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        initial_guesses.append(perturbed)
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    start_time = time.time()
    
    for guess in initial_guesses:
        # Perturb the guess with small noise
        noise = np.random.normal(0, 0.02, n_points)
        perturbed = guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function to minimize C5
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
