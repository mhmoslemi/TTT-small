    # Generate perturbations of initial guesses
    perturbations = []
    for guess in initial_guesses:
        noise = np.random.uniform(-0.1, 0.1, n_points)
        perturbed = guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        perturbations.append(perturbed)
    initial_guesses.extend(perturbations)
    
    # Define objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define constraints
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    
    # Function to perform optimization on a single initial guess
    def optimize_initial_guess(initial_guess):
        result = minimize(
            objective,
            initial_guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 1000}
        )
        if not result.success:
            # Evaluate the initial guess's C5 if optimization fails
            initial_c5 = objective(initial_guess)
            return {'success': False, 'fun': initial_c5, 'x': initial_guess}
        return result
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Use ThreadPoolExecutor to parallelize optimizations
    with ThreadPoolExecutor(max_workers=10) as executor:
        futures = [executor.submit(optimize_initial_guess, guess) for guess in initial_guesses]
        for future in futures:
            result = future.result()
            if result['success']:
                if result['fun'] < best_c5:
                    best_c5 = result['fun']
                    best_h = result['x']
            else:
                initial_c5 = result['fun']
                if initial_c5 < best_c5:
                    best_c5 = initial_c5
                    best_h = result['x']
            # Check elapsed time
            elapsed = time.time() - start_time
            if elapsed >= budget_s:
                break
    
    return best_h, best_c5, n_points
