    with ThreadPoolExecutor(max_workers=10) as executor:
        futures = [executor.submit(optimize_initial_guess, h) for h in initial_guesses]
        for future in futures:
            result = future.result()
            if result.success and result.fun < best_c5:
                best_c5 = result.fun
                best_h = result.x
            elapsed = time.time() - start_time
            if elapsed >= budget_s:
                break
    
    return best_h, best_c5, n_points
