    # Generate diverse initial guesses
    num_initial_guesses = 100  # Increased from 50
    initial_guesses = []
    
    # Generate perturbed initial guesses
    for _ in range(num_initial_guesses // 2):
        noise = np.random.normal(0, 0.005, n_points)
        perturbed = initial_guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        scale_factor = (n_points / 2) / np.sum(perturbed)
        perturbed *= scale_factor
        initial_guesses.append(perturbed)
    
    # Generate random initial guesses
    for _ in range(num_initial_guesses // 2):
        h_rand = np.random.rand(n_points)
        h_rand = h_rand / np.sum(h_rand) * (n_points / 2)
        initial_guesses.append(h_rand)
    
    # Objective function using FFT
    def objective(h):
        one_minus_h = 1.0 - h
        n = len(h)
        # Compute FFT of h and 1-h
        h_fft = np.fft.fft(h, 2*n - 1)
        one_minus_h_fft = np.fft.fft(one_minus_h, 2*n - 1)
        # Compute correlation via FFT
        corr_fft = h_fft * np.conj(one_minus_h_fft)
        corr = np.fft.ifft(corr_fft).real
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Run optimizations for each initial guess
    # Use multiprocessing to parallelize
    def optimize_single(guess):
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-9, 'maxiter': 1000}
        )
        if result.success:
            return (result.fun, result.x)
        else:
            return (float('inf'), guess)  # Fallback to initial guess
    
    with multiprocessing.Pool(processes=10) as pool:  # Use all 10 CPUs
        results = pool.map(optimize_single, initial_guesses)
    
    # Extract the best result
    for c5, h in results:
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    
    # Fallback to initial guess if no optimization was successful
    if best_h is None:
        result = minimize(
            objective,
            initial_guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-9, 'maxiter': 1000}
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points
