        # Clip to [0, 1]

    def objective(h):

        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):

    # Bounds for each h[i]

        result = minimize(
