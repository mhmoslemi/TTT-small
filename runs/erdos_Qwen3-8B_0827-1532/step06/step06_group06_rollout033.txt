    random_guess *= target_sum / np.sum(random_guess)
    initial_guesses.append(random_guess)
    freq = 2 * np.pi / n_points
    sinusoidal_guess = 0.5 + 0.5 * np.sin(freq * np.arange(n_points))
    sinusoidal_guess *= target_sum / np.sum(sinusoidal_guess)
    initial_guesses.append(sinusoidal_guess)
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    start_time = time.time()
    
    for guess in initial_guesses:
        result = differential_evolution(
            objective,
            bounds,
            popsize=15,
            mutation=(0.5, 1.0),
            recombination=0.8,
            tol=1e-8,
            maxiter=100,
            seed=seed
        )
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
