import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 50  # Reduced points for faster evaluation
    dx = 2.0 / n_points
    
    # Generate initial guess via random perturbation of uniform distribution
    initial_guess = np.random.rand(n_points)
    # Adjust sum to meet the constraint
    initial_guess = initial_guess / np.sum(initial_guess) * (n_points / 2)
    
    # Generate diverse initial guesses with random noise
    num_initial_guesses = 30  # Increased diversity
    initial_guesses = []
    for _ in range(num_initial_guesses):
        # Add random noise with varying amplitude
        noise_amp = np.random.uniform(0.01, 0.1)
        noise = noise_amp * np.random.randn(n_points)
        perturbed = initial_guess + noise
        # Clip to [0, 1]
        perturbed = np.clip(perturbed, 0, 1)
        # Adjust sum to meet the constraint
        scale_factor = (n_points / 2) / np.sum(perturbed)
        perturbed *= scale_factor
        initial_guesses.append(perturbed)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Run optimizations for each initial guess with tighter tolerances
    for guess in initial_guesses:
        # Run optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-9, 'maxiter': 2000}  # Increased maxiter
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # If no optimization was done, use the initial guess

        result = minimize(
            objective,
            initial_guess,
            method='SLSQP',
            bounds=bounds,

            options={'ftol': 1e-9, 'maxiter': 2000}
        )
        best_c5 = result.fun
        best_h = result.x
