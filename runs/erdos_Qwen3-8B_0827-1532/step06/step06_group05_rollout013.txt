    start_time = time.time()
    
    # Define objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define constraints
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    
    # Function to perform optimization on a single guess
    def optimize_guess(guess):
        # Scale to maintain sum constraint
        current_sum = np.sum(guess)
        scale = target_sum / current_sum
        guess = guess * scale
        guess = np.clip(guess, 0, 1)
        
        # Perform optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 1000}
        )
        return result
    
    # Use ThreadPoolExecutor to parallelize optimizations
    with ThreadPoolExecutor(max_workers=10) as executor:
        futures = [executor.submit(optimize_guess, g) for g in all_guesses]
        for future in futures:
            result = future.result()
            if result.success and result.fun < best_c5:
                best_c5 = result.fun
                best_h = result.x
            
            # Check elapsed time
            elapsed = time.time() - start_time
            if elapsed >= budget_s:
                break
    
    return best_h, best_c5, n_points
