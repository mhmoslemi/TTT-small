    scale = (n_points / 2) / sum_sinusoidal
    sinusoidal = sinusoidal * scale
    initial_guesses.append(sinusoidal)
    
    # Add random binary guess
    random_binary = np.random.choice([0, 1], size=n_points)
    sum_binary = np.sum(random_binary)
    if sum_binary == 0:
        random_binary = np.random.choice([0, 1], size=n_points)
        sum_binary = np.sum(random_binary)
    scale = (n_points / 2) / sum_binary
    random_binary = random_binary * scale
    initial_guesses.append(random_binary)
    
    # Add a random guess
    random_guess = np.random.uniform(0, 1, n_points)
    sum_random = np.sum(random_guess)
    scale = (n_points / 2) / sum_random
    random_guess = random_guess * scale
    initial_guesses.append(random_guess)
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    start_time = time.time()
    
    for guess in initial_guesses:
        # Define objective function with penalty for sum constraint
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            penalty = 1e4 * (np.sum(h) - target_sum)**2
            return max_corr * dx + penalty
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Run differential evolution
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=100,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed
        )
        
        # Check if this result is better
        if result.success:
            h_opt = result.x
            one_minus_h = 1.0 - h_opt
            corr = np.correlate(h_opt, one_minus_h, mode='full')
            max_corr = np.max(corr)
            current_c5 = max_corr * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_opt
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
