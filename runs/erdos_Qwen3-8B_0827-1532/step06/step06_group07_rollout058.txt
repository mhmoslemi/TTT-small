from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Use the provided initial guess
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate diverse initial guesses: binary, sinusoidal, and noise-perturbed
    initial_guesses = [initial_guess.copy()]
    
    # Generate binary patterns
    for _ in range(5):
        indices = np.random.choice(n_points, 50, replace=False)
        binary = np.zeros(n_points)
        binary[indices] = 1.0
        initial_guesses.append(binary)
    
    # Generate sinusoidal pattern
    x = np.linspace(0, 2, n_points)
    sinusoidal = 0.5 + 0.5 * np.sin(np.pi * x)
    sinusoidal = np.clip(sinusoidal, 0, 1)
    initial_guesses.append(sinusoidal)
    
    # Generate noise-perturbed versions of initial guess
    for _ in range(5):
        noise = np.random.normal(0, 0.05, n_points)
        perturbed = initial_guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        initial_guesses.append(perturbed)
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    start_time = time.time()
    
    for guess in initial_guesses:
        # Perturb the guess with small noise
        noise = np.random.normal(0, 0.02, n_points)
        perturbed = guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function to minimize C5
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 1500}
        )

        if result.success and result.fun < best_c5:

            best_h = result.x

        if elapsed >= budget_s:

    return best_h, best_c5, n_points
