        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        penalty = 1e4 * (np.sum(h) - n_points / 2)**2
        return max_corr * dx + penalty
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Run differential evolution for each initial guess
    for guess in initial_guesses:
        # Run differential evolution
        result = differential_evolution(
            objective,
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            strategy='best1bin',
            popsize=15,
            maxiter=100,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        current_c5 = result.fun
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x
    
    # Fallback to initial guess if no optimization was successful
    if best_h is None:
        # Use the first initial guess as a fallback
        best_h = initial_guesses[0]
        # Compute the objective for the fallback
        one_minus_h = 1.0 - best_h
        corr = np.correlate(best_h, one_minus_h, mode='full')
        best_c5 = np.max(corr) * dx
    
    return best_h, best_c5, n_points
