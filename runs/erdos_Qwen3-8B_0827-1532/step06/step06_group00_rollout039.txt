            initial_guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-9, 'maxiter': 1500}
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points
