        max_corr = np.max(corr)
        return max_corr * dx + penalty
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Run differential evolution for each initial guess
    for guess in initial_guesses:
        result = differential_evolution(
            objective,
            bounds,
            popsize=15,
            mutation=(0.5, 1.0),
            recombination=0.8,
            tol=1e-10,
            maxiter=100,
            seed=seed
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
