from scipy.optimize import minimize
import time

    # Generate diverse initial guesses with multiple perturbation types
    num_initial_guesses = 100  # More diverse initial guesses
    initial_guesses = []
    for _ in range(num_initial_guesses):
        # Random noise perturbation
        noise_amp = np.random.uniform(0.001, 0.05)
        noise = noise_amp * np.random.randn(n_points)
        perturbed = initial_guess + noise
        # Sine wave perturbation
        noise_freq = np.random.uniform(1, 10)
        noise_sine = noise_amp * np.sin(2 * np.pi * noise_freq * x_new)
        perturbed_sine = initial_guess + noise_sine
        # Combine both perturbations
        perturbed_combined = (perturbed + perturbed_sine) / 2
        # Clip and scale
        perturbed_combined = np.clip(perturbed_combined, 0, 1)
        scale_factor = (n_points / 2) / np.sum(perturbed_combined)
        perturbed_combined *= scale_factor
        initial_guesses.append(perturbed_combined)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')

    start_time = time.time()
    
    # Run optimizations for each initial guess with tighter tolerances
    for guess in initial_guesses:
        # Run optimization with SLSQP
        result = minimize(

            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
