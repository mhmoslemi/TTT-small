import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 200  # Increased for better resolution
    dx = 2.0 / n_points
    
    # Interpolate initial_h_values to 200 points
    x_initial = np.linspace(0, 2, len(initial_h_values))
    x_new = np.linspace(0, 2, n_points)
    initial_guess = np.interp(x_new, x_initial, initial_h_values)
    initial_guess = initial_guess / np.sum(initial_guess) * (n_points / 2)
    
    # Generate diverse initial guesses with random noise
    num_initial_guesses = 5
    initial_guesses = []
    for _ in range(num_initial_guesses):
        noise = np.random.normal(0, 0.05, n_points)
        perturbed = initial_guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        scale_factor = (n_points / 2) / np.sum(perturbed)
        perturbed *= scale_factor
        initial_guesses.append(perturbed)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    
    # Run optimizations for each initial guess
    for guess in initial_guesses:
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-6, 'maxiter': 1000}
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # Fallback to initial guess if no improvement
    if best_h is None:
        result = minimize(
            objective,
            initial_guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-6, 'maxiter': 1000}
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points
