        perturbed *= scale_factor
        initial_guesses.append(perturbed)
    
    # Generate random initial guesses
    for _ in range(num_initial_guesses // 2):
        h_rand = np.random.rand(n_points)
        h_rand = h_rand / np.sum(h_rand) * (n_points / 2)
        initial_guesses.append(h_rand)
    
    # Objective function with penalty for constraint violation
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        # Penalize constraint violation
        penalty = 1e3 * np.abs(np.sum(h) - n_points / 2)
        return max_corr * dx + penalty
    
    # Constraint function (equality constraint)
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')

    for guess in initial_guesses:
        # Run optimization with SLSQP
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},

        )
