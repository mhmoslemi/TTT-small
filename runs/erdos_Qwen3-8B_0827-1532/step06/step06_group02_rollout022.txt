from scipy.optimize import minimize, differential_evolution
import time

    initial_guess = np.interp(x_new, x_initial, initial_guess)
    initial_guess = initial_guess / np.sum(initial_guess) * (n_points / 2)

    initial_guesses = []

    initial_guesses.append(initial_guess.copy())

    noise_amp = 0.05
    noise = noise_amp * np.random.randn(n_points)
    perturbed = initial_guess + noise
    perturbed = np.clip(perturbed, 0, 1)
    scale_factor = (n_points / 2) / np.sum(perturbed)
    perturbed *= scale_factor
    initial_guesses.append(perturbed)
    # 3. Sinusoidal variation
    freq = 5
    noise = 0.05 * np.sin(2 * np.pi * freq * x_new)
    perturbed = initial_guess + noise
    perturbed = np.clip(perturbed, 0, 1)
    scale_factor = (n_points / 2) / np.sum(perturbed)
    perturbed *= scale_factor
    initial_guesses.append(perturbed)
    # 4. Alternating pattern
    alt_pattern = np.zeros(n_points)
    for i in range(n_points):
        if i % 2 == 0:
            alt_pattern[i] = 1.0
        else:
            alt_pattern[i] = 0.0
    alt_pattern = alt_pattern / np.sum(alt_pattern) * (n_points / 2)
    initial_guesses.append(alt_pattern)
    # 5. Random binary pattern
    binary = np.random.rand(n_points) > 0.5
    binary = binary.astype(float)
    binary = binary / np.sum(binary) * (n_points / 2)
    initial_guesses.append(binary)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    for guess in initial_guesses:

        result_de = differential_evolution(
