                options={'ftol': 1e-9, 'maxiter': 100}
            )
            futures.append(future)
        
        # Process results within budget
        for future in futures:
            if time.time() - start_time >= budget_s:
                break
            result = future.result()
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
    
    # Fallback to initial guess if no optimization was successful
    if best_h is None:
        result = minimize(
            objective,
            initial_guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-9, 'maxiter': 100}
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points
