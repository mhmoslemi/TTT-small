import numpy as np
from scipy.optimize import minimize
import time
from concurrent.futures import ThreadPoolExecutor

def run(seed=42, budget_s=100, **kwargs):
    # Use the provided initial guess
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate multiple initial guesses with varied strategies
    num_initial_guesses = 100  # Increased number of initial guesses for better exploration
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Define objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define constraints
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    
    # Function to generate a random initial guess
    def generate_random_guess():
        random_guess = np.random.rand(n_points)
        random_guess = random_guess * (target_sum / np.sum(random_guess))
        return random_guess
    
    # Generate initial guesses: mix of perturbations and random guesses
    initial_guesses = []
    for i in range(num_initial_guesses):
        if i < 50:
            # Perturbations of the initial guess with varying noise
            noise = np.random.normal(0, 0.1 if i < 25 else 0.05, n_points)
            perturbed = initial_guess + noise
        else:
            # Random guesses
            perturbed = generate_random_guess()
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        initial_guesses.append(perturbed)
    
    # Function to perform optimization on a single initial guess
    def optimize_guess(guess):
        # Perform optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 500}
        )
        return result
    
    # Use ThreadPoolExecutor to parallelize optimizations
    with ThreadPoolExecutor(max_workers=10) as executor:
        futures = [executor.submit(optimize_guess, guess) for guess in initial_guesses]
        for future in futures:
            result = future.result()
            if result.success and result.fun < best_c5:
                best_c5 = result.fun
                best_h = result.x
            
            # Check elapsed time
            elapsed = time.time() - start_time
            if elapsed >= budget_s:
                break
    
    return best_h, best_c5, n_points
