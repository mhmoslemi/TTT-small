                best_c5 = current_c5
                best_h = result.x
    
    # Fallback to initial guess if no optimization was successful
    if best_h is None:
        result = minimize(
            objective,
            initial_h_values,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-6, 'maxiter': 500}
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points
