    sine_wave = sine_wave * scale

    square_wave = np.zeros(n_points)
    square_wave[:n_points//2] = 1.0
    current_sum = np.sum(square_wave)
    scale = target_sum / current_sum
    square_wave = square_wave * scale
    initial_guesses.append(square_wave)

        noise = np.random.normal(0, 0.05, n_points)

        scale = target_sum / current_sum
        perturbed = perturbed * scale
        initial_guesses.append(perturbed)
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()

    for guess in initial_guesses:
        # Objective function to minimize C5
        def objective(h):

        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Differential evolution for global optimization
        try:
            result = differential_evolution(
                objective,
                bounds,

                popsize=15,
