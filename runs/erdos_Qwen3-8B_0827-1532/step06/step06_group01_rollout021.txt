    for _ in range(num_initial_guesses // 2):
        # Random initial guess
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
        initial_guesses.append(h)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Use ThreadPoolExecutor for parallel optimization
    with ThreadPoolExecutor(max_workers=10) as executor:
        futures = []
        for guess in initial_guesses:
            future = executor.submit(
                minimize,
                objective,
                guess,
                method='L-BFGS-B',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint},
                options={'ftol': 1e-10, 'maxiter': 100}
            )
            futures.append(future)
        
        for future in as_completed(futures):
            elapsed = time.time() - start_time
            if elapsed >= budget_s:
                break
            try:
                result = future.result()
                if result.success:
                    current_c5 = result.fun
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = result.x
            except Exception as e:
                # Handle any exceptions, e.g., optimization failure
                pass
    
    # If no optimization was done, use the initial guess
    if best_h is None:
        result = minimize(
            objective,
            initial_guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-10, 'maxiter': 100}
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points
