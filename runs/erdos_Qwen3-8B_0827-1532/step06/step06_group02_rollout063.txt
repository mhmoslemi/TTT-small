        initial_guesses.append(h)
    
    # Objective function
    def objective(h):

        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')

    # Run optimizations for each initial guess
    for guess in initial_guesses:
        # Run optimization with COBYLA
        result = minimize(

            constraints={'type': 'eq', 'fun': constraint},
