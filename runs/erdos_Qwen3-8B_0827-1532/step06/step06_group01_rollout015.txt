from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Smaller n_points for faster evaluation and more iterations
    dx = 2.0 / n_points
    
    # Generate initial guess based on initial_h_values (interpolated to n_points)
    if len(initial_h_values) == 200:
        initial_guess = initial_h_values[::2]  # Take every other point
    else:
        initial_guess = np.random.rand(n_points)
    initial_guess = initial_guess / np.sum(initial_guess) * (n_points / 2)
    
    # Generate diverse initial guesses
    num_initial_guesses = 100
    initial_guesses = []
    
    # Perturb initial guess with noise
    for _ in range(num_initial_guesses // 2):
        noise = np.random.normal(0, 0.001, n_points)
        perturbed = initial_guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        scale_factor = (n_points / 2) / np.sum(perturbed)
        perturbed *= scale_factor
        initial_guesses.append(perturbed)
    
    # Random initial guesses
    for _ in range(num_initial_guesses // 2):
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
        initial_guesses.append(h)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')

    # Run optimizations for each initial guess
    for guess in initial_guesses:
        # Run optimization with tighter tolerances
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-9, 'maxiter': 1000}
        )
        elapsed = time.time() - start_time

        if result.success:

    # Fallback to initial guess if no improvement found
    if best_h is None:
        result = minimize(
            objective,
            initial_guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-9, 'maxiter': 1000}
        )
