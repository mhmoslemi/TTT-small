from scipy.optimize import minimize
import time

    num_perturbations = 100  # Increased number of perturbations for better exploration
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    for perturbation_idx in range(num_perturbations):
        # Generate diverse initial guesses: some random, some perturbed
        if perturbation_idx < 10:
            # Random initial guess
            random_guess = np.random.uniform(0, 1, n_points)
            random_guess = random_guess * (target_sum / np.sum(random_guess))
            perturbed = random_guess
        elif perturbation_idx < 30:
            # Perturb the initial guess with larger noise
            noise = np.random.uniform(-0.2, 0.2, n_points)
            perturbed = initial_guess + noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
        else:
            # Perturb the initial guess with smaller noise
            noise = np.random.uniform(-0.05, 0.05, n_points)
            perturbed = initial_guess + noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
        
        # Objective function to minimize C5
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization with tighter tolerances and more iterations
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,

        )
