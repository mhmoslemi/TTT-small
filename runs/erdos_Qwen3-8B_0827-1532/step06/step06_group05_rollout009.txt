from scipy.optimize import minimize
import time
from concurrent.futures import ThreadPoolExecutor

def run(seed=42, budget_s=100, **kwargs):
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    best_c5 = float('inf')
    best_h = initial_guess.copy()
    start_time = time.time()

    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    def optimize_perturbation(perturbed):

        scale = target_sum / current_sum
        perturbed = perturbed * scale

        result = minimize(
            objective,
            perturbed,
            method='trust-constr',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-6, 'maxiter': 500}
        )
        return result

    perturbations = []
    for i in range(50):
        noise = np.random.uniform(-0.2, 0.2, n_points)
