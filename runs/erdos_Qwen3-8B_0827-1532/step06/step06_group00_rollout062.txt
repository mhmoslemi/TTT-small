from scipy.optimize import minimize
import time

    initial_guess = np.interp(x_new, x_initial, initial_guess)
    # Adjust sum to meet the constraint
    initial_guess = initial_guess / np.sum(initial_guess) * (n_points / 2)
    
    # Generate a few structured perturbations
    perturbations = []
    # Add sinusoidal perturbation
    noise_amp = 0.05
    noise_freq = 5.0
    noise = noise_amp * np.sin(2 * np.pi * noise_freq * x_new)
    perturbed = initial_guess + noise
    perturbed = np.clip(perturbed, 0, 1)
    scale_factor = (n_points / 2) / np.sum(perturbed)
    perturbed *= scale_factor
    perturbations.append(perturbed)
    
    # Add random perturbation
    random_h = initial_guess + np.random.uniform(-0.05, 0.05, n_points)
    random_h = np.clip(random_h, 0, 1)
    scale_factor = (n_points / 2) / np.sum(random_h)
    random_h *= scale_factor
    perturbations.append(random_h)

    shifted *= scale_factor
    perturbations.append(shifted)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Run optimizations for each initial guess with tighter tolerances
    for guess in perturbations:
        # Run optimization
        result = minimize(
