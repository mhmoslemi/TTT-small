    def objective(h):

        return max_corr * dx
    
    # Define constraints
    bounds = [(0.0, 1.0) for _ in range(n_points)]

        perturbed = np.clip(perturbed, 0, 1)
        
        # Perform optimization
        result = minimize(

            perturbed,
            method='L-BFGS-B',
