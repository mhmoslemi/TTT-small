from scipy.optimize import minimize
import time

    initial_guesses = [initial_guess.copy()]
    
    # Sinusoidal initial guess
    x = np.linspace(0, 2, n_points)
    h_sinusoidal = 0.5 + 0.5 * np.sin(np.pi * x)
    h_sinusoidal = np.clip(h_sinusoidal, 0, 1)
    current_sum = np.sum(h_sinusoidal)
    scale = target_sum / current_sum
    h_sinusoidal *= scale
    initial_guesses.append(h_sinusoidal)
    
    # Piecewise constant initial guess: 1 on [0, 0.5] and [1.5, 2.0]
    h_piecewise = np.zeros(n_points)
    for i in range(n_points):
        x_i = i * dx
        if 0 <= x_i < 0.5 or 1.5 <= x_i <= 2.0:
            h_piecewise[i] = 1.0
    current_sum = np.sum(h_piecewise)
    scale = target_sum / current_sum
    h_piecewise *= scale
    initial_guesses.append(h_piecewise)
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    start_time = time.time()
    
    for guess in initial_guesses:
        # Perturb the guess with small noise
        noise = np.random.normal(0, 0.02, n_points)
        perturbed = guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function to minimize C5
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization
        result = minimize(
