import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 50  # Reduced for speed
    dx = 2.0 / n_points
    
    # Generate initial guesses
    initial_guesses = []
    
    # Initial guess from initial_h_values (interpolated)
    x_initial = np.linspace(0, 2, 100)
    x_new = np.linspace(0, 2, n_points)
    initial_guess = np.interp(x_new, x_initial, initial_h_values)
    initial_guess = initial_guess / np.sum(initial_guess) * (n_points / 2)
    initial_guesses.append(initial_guess)
    
    # Pattern 1: 1 on [0, 0.5], 0 on [0.5, 1.5], 1 on [1.5, 2]
    def pattern1(x):
        return np.where((x >= 0) & (x < 0.5), 1.0,
                        np.where((x >= 1.5) & (x <= 2), 1.0, 0.0))
    pattern1_vals = pattern1(x_new)
    pattern1_vals = pattern1_vals / np.sum(pattern1_vals) * (n_points / 2)
    initial_guesses.append(pattern1_vals)
    
    # Pattern 2: sine wave h(x) = 0.5 + 0.5*sin(πx)
    pattern2_vals = 0.5 + 0.5 * np.sin(np.pi * x_new)
    initial_guesses.append(pattern2_vals)
    
    # Pattern 3: random steps with adjusted sum
    def pattern3(x):
        return np.random.uniform(0.2, 0.8, size=len(x))
    pattern3_vals = pattern3(x_new)
    pattern3_vals = pattern3_vals / np.sum(pattern3_vals) * (n_points / 2)
    initial_guesses.append(pattern3_vals)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    
    for guess in initial_guesses:
        # Run optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-9, 'maxiter': 2000}
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # If no optimization was done, use the first initial guess
    if best_h is None:
        result = minimize(
            objective,
            initial_guesses[0],
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-9, 'maxiter': 2000}
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points
