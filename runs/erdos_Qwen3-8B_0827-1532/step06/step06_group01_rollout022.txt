import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Reduced for faster computation
    dx = 2.0 / n_points
    
    # Define bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Objective function for DE: normalize h to meet the integral constraint
    def objective_de(h):
        h_normalized = h / np.sum(h) * (n_points / 2)
        one_minus_h = 1.0 - h_normalized
        corr = np.correlate(h_normalized, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Run differential evolution
    start_time = time.time()
    result_de = differential_evolution(
        objective_de,
        bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=20,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        polish=True,
        disp=False
    )
    elapsed = time.time() - start_time
    
    # Optionally refine with SLSQP
    h_refined = result_de.x
    # Define objective for SLSQP without normalization
    def objective_slsqp(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]
    
    # Run SLSQP optimization
    result_slsqp = minimize(
        objective_slsqp,
        h_refined,
        method='SLSQP',
        bounds=bounds_slsqp,
        constraints={'type': 'eq', 'fun': constraint},
        options={'ftol': 1e-10, 'maxiter': 1000}
    )
    elapsed_slsqp = time.time() - start_time
    
    # Determine best result
    if elapsed_slsqp >= budget_s:
        best_h = result_de.x
        best_c5 = result_de.fun
    else:
        if result_slsqp.success and result_slsqp.fun < result_de.fun:
            best_h = result_slsqp.x
            best_c5 = result_slsqp.fun
        else:
            best_h = result_de.x
            best_c5 = result_de.fun
    
    return best_h, best_c5, n_points
