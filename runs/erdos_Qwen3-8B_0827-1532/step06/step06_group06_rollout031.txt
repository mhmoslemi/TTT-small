        scale = (n_points / 2) / sum_sinusoidal

    def generate_random(n_points):

        return random_pattern
    
    initial_guesses = [initial_guess.copy()]
    initial_guesses.append(generate_square_wave(n_points))

    best_c5 = float('inf')

    start_time = time.time()
    
    for guess in initial_guesses:
        # Apply sinusoidal perturbation
        freq = 2 * np.pi * (np.random.rand() + 1) / n_points
        amplitude = 0.05

        perturbed = guess + perturbation
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function to minimize C5
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
