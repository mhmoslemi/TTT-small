from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()

    initial_guesses = []

    sine_guess = 0.5 + 0.5 * np.sin(np.pi * x)
    initial_guesses.append(sine_guess)

    binary_guess = np.zeros(n_points)
    binary_guess[::2] = 1.0
    initial_guesses.append(binary_guess)
    
    # Perturbations of initial guess
    for _ in range(5):
        noise = np.random.uniform(-0.1, 0.1, n_points)
        perturbed = initial_guess + noise

        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        initial_guesses.append(perturbed)
    
    # Optimize each initial guess
    for guess in initial_guesses:
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-8, 'maxiter': 500}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:

    return best_h, best_c5, n_points
