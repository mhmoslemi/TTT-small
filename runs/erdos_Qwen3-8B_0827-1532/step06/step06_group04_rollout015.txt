    best_c5 = float('inf')

    start_time = time.time()

    scale = (n_points / 2) / sum_h_sine
    h_sine = h_sine * scale
    initial_guesses = [initial_guess, h_sine]
    
    # Add random initial guesses
    for _ in range(20):
        random_guess = np.random.uniform(0, 1, n_points)
        random_guess = random_guess * (target_sum / np.sum(random_guess))
        initial_guesses.append(random_guess)
    
    # Add perturbations of initial guess
    for _ in range(num_perturbations - 22):  # 22 already added
        if np.random.rand() < 0.5:
            # Larger noise
            noise = np.random.uniform(-0.2, 0.2, n_points)
        else:
            # Smaller noise

        perturbed = initial_guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        initial_guesses.append(perturbed)
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Objective function to minimize C5
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

        result = minimize(

            method='SLSQP',
