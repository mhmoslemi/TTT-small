    initial_guesses.append(binary_alternating)
    
    # Random binary initial guess
    random_binary = np.random.permutation([0, 1] * (n_points // 2))
    initial_guesses.append(random_binary)
    
    # Random continuous initial guesses
    random_continuous = np.random.uniform(0, 1, n_points)
    random_continuous = random_continuous * (target_sum / np.sum(random_continuous))
    initial_guesses.append(random_continuous)
    
    random_continuous2 = np.random.uniform(0, 1, n_points)
    random_continuous2 = random_continuous2 * (target_sum / np.sum(random_continuous2))
    initial_guesses.append(random_continuous2)
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    
    def optimize_initial_guess(initial):
        current_sum = np.sum(initial)
        scale = target_sum / current_sum
        initial = initial * scale
        initial = np.clip(initial, 0, 1)
        result = minimize(
            objective,
            initial,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 1000}
        )
        return result
    
    with ThreadPoolExecutor(max_workers=10) as executor:
        futures = [executor.submit(optimize_initial_guess, guess) for guess in initial_guesses]
        for future in futures:
            result = future.result()
            if result.success and result.fun < best_c5:
                best_c5 = result.fun
                best_h = result.x
            elapsed = time.time() - start_time
            if elapsed >= budget_s:
                break
    
    return best_h, best_c5, n_points
