    for perturbation_idx in range(num_perturbations):
        # Generate diverse initial guesses: some random, some perturbed
        if perturbation_idx < 10:
            # Random initial guess
            random_guess = np.random.uniform(0, 1, n_points)
            random_guess = random_guess * (target_sum / np.sum(random_guess))
            perturbed = random_guess
        elif perturbation_idx < 30:
            # Perturb the initial guess with varying noise
            noise = np.random.uniform(-0.1, 0.1, n_points)
            perturbed = initial_guess + noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
        else:
            # Generate structured initial guess: alternating blocks
            block_size = n_points // 4
            structured = np.zeros(n_points)
            structured[:block_size] = 1.0
            structured[block_size:2*block_size] = 0.0
            structured[2*block_size:3*block_size] = 1.0
            structured[3*block_size:] = 0.0
            structured = structured * (target_sum / np.sum(structured))
            perturbed = structured
        
        # Objective function to minimize C5
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization with different methods and tighter tolerances
        try:
            # Try different optimization methods
            result = minimize(
                objective,
                perturbed,
                method='L-BFGS-B',

                constraints=cons,
