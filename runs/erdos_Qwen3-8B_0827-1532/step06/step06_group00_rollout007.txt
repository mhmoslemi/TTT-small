import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Balanced between speed and precision
    dx = 2.0 / n_points
    
    # Generate initial guess based on initial_h_values
    x_initial = np.linspace(0, 2, len(initial_h_values))
    x_new = np.linspace(0, 2, n_points)
    initial_guess = np.interp(x_new, x_initial, initial_h_values)
    # Adjust sum to meet the constraint
    initial_guess = initial_guess / np.sum(initial_guess) * (n_points / 2)
    
    # Define bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Objective function with penalty
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        penalty = 1e3 * abs(np.sum(h) - n_points / 2)
        return max_corr * dx + penalty
    
    # Run differential evolution
    start_time = time.time()
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=20,
        maxiter=100,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed
    )
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        # If time is up, return the initial guess
        return initial_guess, objective(initial_guess), n_points
    
    best_h = result.x
    # Recalculate C5 without penalty
    one_minus_h = 1.0 - best_h
    corr = np.correlate(best_h, one_minus_h, mode='full')
    best_c5 = np.max(corr) * dx
    
    return best_h, best_c5, n_points
