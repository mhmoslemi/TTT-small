from scipy.optimize import minimize

from concurrent.futures import ThreadPoolExecutor

    # Use the provided initial guess
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate multiple initial guesses
    initial_guesses = [initial_guess.copy()]
    # Binary initial guess
    binary_initial = np.zeros(n_points)
    binary_initial[:n_points//2] = 1
    np.random.shuffle(binary_initial)
    initial_guesses.append(binary_initial)
    # Sine wave initial guess
    sine_initial = np.sin(np.linspace(0, 2*np.pi, n_points)) * 0.5 + 0.5
    initial_guesses.append(sine_initial)
    
    # Generate perturbations for each initial guess
    perturbations = []
    for guess in initial_guesses:
        for _ in range(5):  # 5 perturbations per initial guess
            noise = np.random.normal(0, 0.1, n_points)
            perturbed = guess + noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
            perturbations.append(perturbed)
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    start_time = time.time()
    
    # Define objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define constraints
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    def optimize_perturbation(perturbed):
        # Scale to maintain sum constraint
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        perturbed = np.clip(perturbed, 0, 1)
        
        # Perform optimization with COBYLA
        result = minimize(
            objective,
            perturbed,
            method='COBYLA',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 500, 'disp': False}
        )
        return result
    
    # Use ThreadPoolExecutor to parallelize optimizations
    with ThreadPoolExecutor(max_workers=10) as executor:
