import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Use a smaller n_points for faster optimization
    n_points = 50
    dx = 2.0 / n_points
    target_sum = n_points / 2  # Ensure integral is 1

    # Generate multiple structured initial guesses
    initial_guesses = []

    # Sinusoidal initial guess
    x_values = np.linspace(0, 2, n_points)
    h_sinusoidal = 0.5 + 0.5 * np.sin(np.pi * x_values)
    initial_guesses.append(h_sinusoidal)

    # Symmetric sinusoidal initial guess
    h_symmetric = 0.5 + 0.5 * np.sin(np.pi * (x_values - 1))
    initial_guesses.append(h_symmetric)

    # Random initial guess
    random_guess = np.random.rand(n_points)
    random_guess = random_guess * (target_sum / np.sum(random_guess))
    initial_guesses.append(random_guess)

    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    start_time = time.time()

    for guess in initial_guesses:
        # Add small noise and scale to meet the sum constraint
        noise = np.random.normal(0, 0.05, n_points)
        perturbed = guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale

        # Objective function to minimize C5
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx

        # Bounds and constraints
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

        # Perform optimization
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-12, 'maxiter': 2000}
        )

        # Update best solution
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x

        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

    return best_h, best_c5, n_points
