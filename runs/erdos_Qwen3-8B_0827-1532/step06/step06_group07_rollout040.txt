import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Use the provided initial guess
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate diverse initial guesses
    initial_guesses = [initial_guess.copy()]
    
    # Sinusoidal function
    x = np.linspace(0, 2, n_points)
    h_sinusoidal = 0.5 + 0.5 * np.sin(np.pi * x)
    initial_guesses.append(h_sinusoidal)
    
    # Random binary function
    random_binary = np.random.choice([0, 1], size=n_points)
    sum_binary = np.sum(random_binary)
    scale = (n_points / 2) / sum_binary if sum_binary != 0 else 1.0
    random_binary_scaled = np.round(random_binary * scale).astype(float)
    initial_guesses.append(random_binary_scaled)
    
    # Alternating pattern
    alternating = np.zeros(n_points)
    for i in range(n_points):
        if i % 2 == 0:
            alternating[i] = 1
        else:
            alternating[i] = 0
    sum_alternating = np.sum(alternating)
    scale = (n_points / 2) / sum_alternating if sum_alternating != 0 else 1.0
    alternating_scaled = np.round(alternating * scale).astype(float)
    initial_guesses.append(alternating_scaled)
    
    # Add some random perturbations to initial guesses
    for guess in initial_guesses:
        noise = np.random.normal(0, 0.05, n_points)
        perturbed = guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        initial_guesses.append(perturbed)
    
    # Trim to avoid excessive initial guesses
    initial_guesses = initial_guesses[:20]
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    start_time = time.time()
    
    for guess in initial_guesses:
        # Ensure guess is within bounds and scaled
        guess = np.clip(guess, 0, 1)
        current_sum = np.sum(guess)
        scale = target_sum / current_sum if current_sum != 0 else 1.0
        guess = guess * scale
        
        # Objective function to minimize C5
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization
        result = minimize(
            objective,
            guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 1000}
        )
        
        # Check if this result is better
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
