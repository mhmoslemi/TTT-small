import numpy as np
from scipy.optimize import minimize
import time
from concurrent.futures import ThreadPoolExecutor

def run(seed=42, budget_s=100, **kwargs):
    # Use the provided initial guess
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate multiple diverse initial guesses
    num_perturbations = 80  # Increased number for better exploration
    perturbations = []
    
    # Add perturbations of initial guess
    for i in range(num_perturbations // 2):
        if i < 10:
            noise = np.random.uniform(-0.3, 0.3, n_points)
        elif i < 30:
            noise = np.random.uniform(-0.15, 0.15, n_points)
        else:
            noise = np.random.uniform(-0.05, 0.05, n_points)
        perturbed = initial_guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        perturbations.append(perturbed)
    
    # Add random initial guesses
    for i in range(num_perturbations // 2):
        random_guess = np.random.uniform(0, 1, n_points)
        random_guess = random_guess * (target_sum / np.sum(random_guess))
        perturbations.append(random_guess)
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    start_time = time.time()
    
    # Define objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define constraints
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    
    # Function to perform optimization on a single perturbation
    def optimize_perturbation(perturbed):
        # Scale to maintain sum constraint
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        perturbed = np.clip(perturbed, 0, 1)
        
        # Perform optimization with L-BFGS-B
        result = minimize(
            objective,
            perturbed,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 1500}
        )
        return result
    
    # Use ThreadPoolExecutor to parallelize optimizations
    with ThreadPoolExecutor(max_workers=10) as executor:
        futures = [executor.submit(optimize_perturbation, p) for p in perturbations]
        for future in futures:
            result = future.result()
            if result.success and result.fun < best_c5:
                best_c5 = result.fun
                best_h = result.x
            
            # Check elapsed time
            elapsed = time.time() - start_time
            if elapsed >= budget_s:
                break
    
    return best_h, best_c5, n_points
