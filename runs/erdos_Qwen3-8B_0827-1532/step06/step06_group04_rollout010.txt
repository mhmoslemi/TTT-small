import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate diverse initial guesses
    num_perturbations = 10
    for perturbation_idx in range(num_perturbations):
        if perturbation_idx < 5:
            # Random initial guess
            random_guess = np.random.uniform(0, 1, n_points)
            random_guess = random_guess * (target_sum / np.sum(random_guess))
            perturbed = random_guess
        else:
            # Perturb the initial guess
            noise = np.random.uniform(-0.1, 0.1, n_points)
            perturbed = initial_guess + noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
        
        # Objective function with penalty for sum constraint
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            penalty = 1e3 * (np.sum(h) - target_sum)**2
            return max_corr * dx + penalty
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Run differential evolution
        result_de = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=100,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            disp=False
        )
        
        if result_de.success:
            # Refine with SLSQP
            def objective_slsqp(h):
                one_minus_h = 1.0 - h
                corr = np.correlate(h, one_minus_h, mode='full')
                max_corr = np.max(corr)
                return max_corr * dx
            
            bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]
            cons_slsqp = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
            
            result_slsqp = minimize(
                objective_slsqp,
                result_de.x,
                method='SLSQP',
                bounds=bounds_slsqp,
                constraints=cons_slsqp,
                options={'ftol': 1e-10, 'maxiter': 200}
            )
            
            if result_slsqp.success and result_slsqp.fun < best_c5:
                best_c5 = result_slsqp.fun
                best_h = result_slsqp.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
