import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(initial_guess)
    dx = 2.0 / n_points

    initial_guesses = [initial_guess.copy()]
    
    # Generate sinusoidal initial guess

    h_sinusoidal = 0.5 + 0.5 * np.sin(np.pi * x_values)
    sum_h = np.sum(h_sinusoidal)
    scale = (n_points / 2) / sum_h
    h_sinusoidal = h_sinusoidal * scale
    initial_guesses.append(h_sinusoidal)
    
    # Generate binary initial guess
    h_binary = np.zeros(n_points)
    h_binary[::2] = 1.0
    sum_h = np.sum(h_binary)
    scale = (n_points / 2) / sum_h
    h_binary = h_binary * scale
    initial_guesses.append(h_binary)
    
    # Generate random initial guesses
    for _ in range(num_initial_guesses - 3):  # 10 total, subtract 3 already added
        # Add random noise
        noise = np.random.normal(0, 0.05, n_points)
        perturbed = initial_guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        initial_guesses.append(perturbed)
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    start_time = time.time()
    
    for guess in initial_guesses:
        # Perturb the guess with small noise
        noise = np.random.normal(0, 0.02, n_points)
        perturbed = guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function to minimize C5
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization
        try:
            result = minimize(
                objective,
                perturbed,
                method='trust-constr',
                bounds=bounds,
                constraints=cons,
                options={'ftol': 1e-12, 'maxiter': 500}
            )
        except:
            # Fallback to the original guess
            result = {'x': guess, 'fun': objective(guess)}
        
        # Check if this result is better
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points

                method='trust-constr',
                bounds=bounds,
                constraints=cons,
                options={'ftol': 1e-12, 'maxiter': 500}
            )
        except:
            # Fallback to the original guess
            result = {'x': guess, 'fun': objective(guess)}
        
        # Check if this result is better
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
