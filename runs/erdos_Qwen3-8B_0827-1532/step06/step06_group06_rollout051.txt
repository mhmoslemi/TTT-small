import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    num_initial_guesses = 10
    initial_guesses = []

    h_base = initial_guess.copy()

    initial_guesses.append(h_base[:-1])
    
    # Sinusoidal perturbations
    for i in range(num_initial_guesses - 1):
        freq = 2 * np.pi * (i + 1) / n_points
        amplitude = 0.1
        perturbation = amplitude * np.sin(freq * np.arange(n_points))
        perturbed = initial_guess + perturbation
        perturbed[-1] = target_sum - np.sum(perturbed[:-1])
        initial_guesses.append(perturbed[:-1])

    for i in range(num_initial_guesses - 1):

        perturbed = initial_guess + noise

        initial_guesses.append(perturbed[:-1])
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    start_time = time.time()
    
    for guess in initial_guesses:
        # Objective function
        def objective(x):

        bounds = [(0.0, 1.0) for _ in range(n_points - 1)]
        
        # Differential evolution optimization
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=100,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed
        )
        
        # Update best solution
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = np.zeros(n_points)
            best_h[:-1] = result.x
            best_h[-1] = target_sum - np.sum(result.x)
        
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
