from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    
    # Generate initial guess based on provided initial_h_values
    initial_guess = initial_h_values
    
    # Generate diverse initial guesses
    num_initial_guesses = 100
    initial_guesses = []
    
    # Generate perturbed initial guesses with larger noise
    for _ in range(num_initial_guesses // 2):
        noise = np.random.normal(0, 0.01, n_points)
        perturbed = initial_guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        scale_factor = (n_points / 2) / np.sum(perturbed)
        perturbed *= scale_factor
        initial_guesses.append(perturbed)
    
    # Generate random initial guesses
    for _ in range(num_initial_guesses // 2):
        h_rand = np.random.rand(n_points)
        h_rand = h_rand / np.sum(h_rand) * (n_points / 2)
        initial_guesses.append(h_rand)
    
    # Add structured initial guesses
    # 1. Block of 1s
    structured_guess = np.zeros(n_points)
    structured_guess[:50] = 1.0
    structured_guess = structured_guess * (50.0 / np.sum(structured_guess))
    initial_guesses.append(structured_guess)
    
    # 2. Alternating 1s and 0s scaled to sum 50
    structured_guess2 = np.zeros(n_points)
    structured_guess2[::2] = 1.0
    structured_guess2 = structured_guess2 * (50.0 / np.sum(structured_guess2))
    initial_guesses.append(structured_guess2)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Run optimizations for each initial guess
    for guess in initial_guesses:
        # Run optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-9, 'maxiter': 1000}
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:

                best_h = result.x

        )

        best_h = result.x
