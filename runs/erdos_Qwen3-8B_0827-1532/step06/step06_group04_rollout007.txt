            perturbed = random_guess

            # Perturb the initial guess with varying noise
            if perturbation_idx < 20:

                # Smaller noise for later perturbations

            perturbed = initial_guess + noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
        
        # Objective function to minimize C5
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx

        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        # Using penalty method for differential evolution
        def penalty_objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            penalty = 1e6 * abs(np.sum(h) - target_sum)
            return max_corr * dx + penalty
        
        # Use differential evolution for some perturbations
        if perturbation_idx % 2 == 0:
            # Differential evolution for global search
            result = differential_evolution(
