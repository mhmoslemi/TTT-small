import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    def objective(h):
        # Scale h to meet the sum constraint
        if np.sum(h) == 0:
            return float('inf')
        h_scaled = h * (target_sum / np.sum(h))
        h_scaled = np.clip(h_scaled, 0, 1)
        one_minus_h = 1.0 - h_scaled
        corr = np.correlate(h_scaled, one_minus_h, mode='full')
        max_corr = np.max(corr) * dx
        return max_corr
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Run differential evolution with multiple restarts
    for _ in range(3):
        result = differential_evolution(
            objective, bounds, strategy='best1bin',
            popsize=15, mutation=(0.5, 1.0), recombination=0.8,
            tol=1e-10, maxiter=100, disp=False
        )
        if result.success:
            current_h = result.x
            current_c5 = objective(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
    
    # Evaluate initial guess
    initial_c5 = objective(initial_guess)
    if initial_c5 < best_c5:
        best_c5 = initial_c5
        best_h = initial_guess
    
    # Check elapsed time
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
