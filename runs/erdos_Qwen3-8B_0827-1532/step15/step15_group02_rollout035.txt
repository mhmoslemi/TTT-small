import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    # Use a more structured initial guess: alternating 0.5 and 0.5 with some randomness
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate a structured initial guess with alternating values
    initial_guess = np.zeros(n_points)
    for i in range(n_points):
        if i % 2 == 0:
            initial_guess[i] = 0.5
        else:
            initial_guess[i] = 0.5 + np.random.uniform(-0.2, 0.2)
    initial_guess = np.clip(initial_guess, 0, 1)
    initial_guess = initial_guess * (target_sum / np.sum(initial_guess))
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Parameters for perturbation
    num_perturbations = 300
    phase_shift = 0.3
    phase_decay = 0.97
    
    current_best_h = initial_guess.copy()
    
    for perturbation_idx in range(num_perturbations):
        # Adaptive noise magnitude based on phase
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        
        # Generate perturbed guess
        perturbed = current_best_h + np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 300}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
