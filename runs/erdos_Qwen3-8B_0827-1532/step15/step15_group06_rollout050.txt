    dx = 2.0 / n_points
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    intervals_4 = [(0.0, 0.5), (0.5, 1.0), (1.0, 1.5), (1.5, 2.0)]

    intervals_1 = [(0.5, 1.5)]
    initial_guesses.append(generate_interval_guess(n_points, intervals_1))
    
    # Add symmetric pattern
    symmetric = np.zeros(n_points)
    for i in range(n_points):
        symmetric[i] = 1.0 if i < n_points//2 else 0.0
    initial_guesses.append(symmetric)
    # Add periodic pattern
    periodic = np.zeros(n_points)
    for i in range(n_points):
        periodic[i] = 0.5 + 0.5 * np.sin(2 * np.pi * i / n_points)
    initial_guesses.append(periodic)
    # Add random guesses
    for _ in range(3):
        guess = np.random.uniform(0, 1, n_points)
        guess = guess * (target_sum / np.sum(guess))
        initial_guesses.append(guess)
    
    # Prepare initial population for DE
    initial_population = np.array([guess for guess in initial_guesses])
    
    # Bounds for DE
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Objective function
    def objective(h):
        scaled_h = h * (target_sum / np.sum(h))
        return evaluate_c5(scaled_h, n_points)
    
    # Run differential evolution
    start_time = time.time()
    result = differential_evolution(
        objective,

        popsize=20,
