import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate diverse initial guesses
    initial_guesses = [initial_guess.copy()]
    for _ in range(4):
        # Perturb the initial guess
        perturbed = initial_guess + np.random.normal(0, 0.1, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * (target_sum / np.sum(perturbed))
        initial_guesses.append(perturbed)
        
        # Generate a random guess
        random_guess = np.random.uniform(0, 1, n_points)
        random_guess = random_guess * (target_sum / np.sum(random_guess))
        initial_guesses.append(random_guess)
    
    # For each initial guess, run the optimization
    for guess in initial_guesses:
        # Optimization with SLSQP with tighter tolerances and more iterations
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 1000, 'ftol': 1e-10, 'eps': 1e-8}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Fallback to initial guess if all optimizations fail
    if best_c5 == float('inf'):
        best_c5 = objective(initial_guess)
        best_h = initial_guess
    
    return best_h, best_c5, n_points
