import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate diverse initial guesses by perturbing the initial guess
    initial_guesses = [initial_guess]
    for _ in range(5):
        perturbed = initial_guess + np.random.normal(0, 0.05, n_points)
        perturbed = perturbed * (target_sum / np.sum(perturbed))
        initial_guesses.append(perturbed)
    
    # For each initial guess, run the optimization
    for guess in initial_guesses:
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Use Nelder-Mead for global exploration and SLSQP for refinement
        result = minimize(
            objective,
            guess,
            method='Nelder-Mead',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'xatol': 1e-8}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Final check for time budget
    elapsed = time.time() - start_time
    if elapsed < budget_s:
        # Try SLSQP with tighter tolerances on the best guess
        def objective_final(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        result_final = minimize(
            objective_final,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-8}
        )
        
        if result_final.success and result_final.fun < best_c5:
            best_c5 = result_final.fun
            best_h = result_final.x
    
    return best_h, best_c5, n_points
