        perturbed = initial_guess + np.random.uniform(-0.15, 0.15, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        initial_guesses.append(perturbed)
    
    # Optimize each initial guess with differential evolution
    for guess in initial_guesses:
        # Objective function with penalty
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_overlap = np.max(corr) * dx
            penalty = 1e4 * (np.sum(h) - target_sum)**2
            return max_overlap + penalty
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Use differential evolution with adjusted parameters
        result = minimize(
            objective,
            guess,
            method='differential_evolution',
            bounds=bounds,
            options={
                'popsize': 15,
                'maxiter': 100,
                'tol': 1e-6,
                'mutation': (0.5, 1),
                'recombination': 0.8
            }
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Perturbation phase on the best_h found so far
    num_perturbations = 200
    phase_shift = 0.15
    phase_decay = 0.96
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function with penalty
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_overlap = np.max(corr) * dx
            penalty = 1e4 * (np.sum(h) - target_sum)**2
            return max_overlap + penalty
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Use differential evolution with adjusted parameters
        result = minimize(
            objective,
            perturbed,
            method='differential_evolution',
            bounds=bounds,
            options={
                'popsize': 15,
                'maxiter': 100,
                'tol': 1e-6,
                'mutation': (0.5, 1),
                'recombination': 0.8
            }
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
