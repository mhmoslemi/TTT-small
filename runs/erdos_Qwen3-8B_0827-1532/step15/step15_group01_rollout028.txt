from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    target_sum = n_points / 2
    
    # Generate a structured initial guess with alternating 1s and 0s every 0.5 units
    initial_guess = np.zeros(n_points)
    initial_guess[0:25] = 1.0
    initial_guess[50:75] = 1.0
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate diverse initial guesses with structured patterns
    num_initial_guesses = 5
    initial_guesses = []
    for _ in range(num_initial_guesses):
        # Create a shifted version of the initial guess
        shifted = np.roll(initial_guess, np.random.randint(0, n_points))
        # Add small random noise and scale to maintain the sum constraint
        perturbed = shifted + np.random.uniform(-0.1, 0.1, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        initial_guesses.append(perturbed)
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with L-BFGS-B
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 300}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Perturbation phase on the best_h found so far
    num_perturbations = 200
    phase_shift = 0.2
