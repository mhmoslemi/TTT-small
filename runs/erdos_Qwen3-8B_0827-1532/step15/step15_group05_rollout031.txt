    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate multiple initial guesses
    initial_guesses = [initial_guess]
    
    # Structured initial guesses
    # First half
    first_half = np.zeros(n_points)
    first_half[:n_points//2] = 1.0
    initial_guesses.append(first_half)
    
    # Second half
    second_half = np.zeros(n_points)
    second_half[n_points//2:] = 1.0
    initial_guesses.append(second_half)
    
    # Alternating pattern
    alternating = np.array([1.0 if i % 2 == 0 else 0.0 for i in range(n_points)])
    initial_guesses.append(alternating)
    
    # Single interval (first half)
    single_interval = np.array([1.0 if i < n_points//2 else 0.0 for i in range(n_points)])
    initial_guesses.append(single_interval)
    
    # Add some random guesses
    for _ in range(3):
        random_h = np.random.uniform(0, 1, n_points)
        random_h = random_h * (target_sum / np.sum(random_h))
        initial_guesses.append(random_h)
    
    # For each initial guess, run the optimization
    for guess in initial_guesses:
        # Optimization with COBYLA
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='COBYLA',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 1000, 'disp': False}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    return best_h, best_c5, n_points
