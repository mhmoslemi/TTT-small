            penalty = 1e4 * (np.sum(h) - target_sum)**2
            return max_corr + penalty
        
        result = differential_evolution(
            objective, bounds, strategy='best1bin',
            popsize=population_size, maxiter=max_generations,
            tol=1e-10, mutation=(0.5, 1), recombination=0.8,
            seed=seed
        )
        
        if result.success:
            current_c5 = result.fun - 1e4 * (np.sum(result.x) - target_sum)**2
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
