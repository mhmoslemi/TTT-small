import numpy as np
from scipy.optimize import differential_evolution
import time

def generate_interval_guess(n_points, intervals):
    h = np.zeros(n_points)
    dx = 2.0 / n_points
    for start, end in intervals:
        for i in range(n_points):
            if start <= i*dx <= end:
                h[i] = 1.0
    # Adjust to meet the sum constraint
    total = np.sum(h)
    if total == 0:
        return h
    scale = (n_points / 2) / total
    return h * scale

def evaluate_c5(h, n_points):
    dx = 2.0 / n_points
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return np.max(corr) * dx

        h_scaled = h * scale
        return evaluate_c5(h_scaled, n_points)
    
    # Bounds for DE
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Run differential evolution
    start_time = time.time()
    result = differential_evolution(
        objective,
        bounds,
        popsize=10,
        strategy='best1bin',
        maxiter=100,
        tol=1e-10,
        mutation=(0.5, 1),
        recombination=0.9,
        seed=seed,
        polish=True,
        init=initial_guesses
    )
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return result.x, result.fun, n_points
    
    best_h = result.x
    best_c5 = result.fun
    return best_h, best_c5, n_points
