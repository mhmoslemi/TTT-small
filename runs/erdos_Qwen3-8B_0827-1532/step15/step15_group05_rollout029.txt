import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Define the objective function
    def objective(h):
        # Normalize h to meet the sum constraint
        h_normalized = h * (target_sum / np.sum(h))
        one_minus_h = 1.0 - h_normalized
        corr = np.correlate(h_normalized, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    # Bounds for each parameter
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Generate multiple initial populations based on initial guess and random perturbations
    initial_populations = []
    for _ in range(3):
        perturbed = initial_guess + np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * (target_sum / np.sum(perturbed))
        initial_populations.append(perturbed)
    
    # Run differential evolution with multiple initial populations
    for pop in initial_populations:
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=15,
            mutation=(0.5, 1.0),
            recombination=0.9,
            tol=1e-8,
            maxiter=100,
            polish=True,
            disp=False,
            x0=pop
        )
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Ensure the best_h is within [0, 1] and sum constraint is satisfied
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (target_sum / np.sum(best_h))
    
    # Check elapsed time and return early if needed
    elapsed = time.time() - start_time
    if elapsed > budget_s - 10:
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
