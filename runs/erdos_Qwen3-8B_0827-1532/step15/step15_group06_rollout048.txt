    # Periodic pattern
    periodic = np.zeros(n_points)
    for i in range(n_points):
        periodic[i] = 0.5 + 0.5 * np.sin(2 * np.pi * i / n_points)
    initial_guesses.append(periodic)
    
    # Random guesses
    for _ in range(5):
        guess = np.random.uniform(0, 1, n_points)
        guess = guess * (target_sum / np.sum(guess))
        initial_guesses.append(guess)
    
    # Define objective with penalty for sum constraint
    def objective(h):
        c5 = evaluate_c5(h, n_points)
        sum_h = np.sum(h)
        penalty = 1e6 * abs(sum_h - target_sum)
        return c5 + penalty
    
    # Use differential evolution with initial guesses
    result = differential_evolution(
        objective, 
        bounds=[(0, 1) for _ in range(n_points)], 
        popsize=15, 
        mutation=(0.5, 1), 
        recombination=0.8, 
        strategy='best1bin', 
        tol=1e-8, 
        maxiter=100, 
        callback=None, 
        disp=False
    )
    
    best_h = result.x
    best_c5 = evaluate_c5(best_h, n_points)
    
    # Refine with SLSQP
    def objective_slsqp(h):
        return evaluate_c5(h, n_points)
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    
    result_slsqp = minimize(
        objective_slsqp,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-12, 'maxiter': 100}
    )
    
    best_h = result_slsqp.x
    best_c5 = result_slsqp.fun
    
    return best_h, best_c5, n_points
