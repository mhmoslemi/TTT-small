    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate diverse initial guesses
    initial_guesses = [initial_guess.copy()]
    intervals_2 = [(0.0, 0.5), (1.0, 1.5)]
    initial_guesses.append(generate_interval_guess(n_points, intervals_2))
    intervals_3 = [(0.0, 0.6667), (0.6667, 1.3333), (1.3333, 2.0)]
    initial_guesses.append(generate_interval_guess(n_points, intervals_3))
    intervals_4 = [(0.0, 0.5), (0.5, 1.0), (1.0, 1.5), (1.5, 2.0)]
    initial_guesses.append(generate_interval_guess(n_points, intervals_4))
    intervals_1 = [(0.5, 1.5)]
    initial_guesses.append(generate_interval_guess(n_points, intervals_1))
    symmetric = np.zeros(n_points)
    for i in range(n_points):
        symmetric[i] = 1.0 if i < n_points//2 else 0.0
    initial_guesses.append(symmetric)
    periodic = np.zeros(n_points)
    for i in range(n_points):
        periodic[i] = 0.5 + 0.5 * np.sin(2 * np.pi * i / n_points)
    initial_guesses.append(periodic)
    for _ in range(3):
        guess = np.random.uniform(0, 1, n_points)
        guess = guess * (target_sum / np.sum(guess))
        initial_guesses.append(guess)
    
    # Define objective function with penalty
    def objective(h):
        c5 = evaluate_c5(h, n_points)
        penalty = 1e6 * abs(np.sum(h) - target_sum)
        return c5 + penalty
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Run differential evolution
    try:
        result_de = differential_evolution(
            objective,
            bounds,
            popsize=15,
            strategy='best1bin',
            maxiter=50,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8
        )
        best_h_de = result_de.x
        best_c5_de = evaluate_c5(best_h_de, n_points)
    except:
        best_h_de = initial_guess.copy()
        best_c5_de = evaluate_c5(best_h_de, n_points)
    
    # Refine with SLSQP
    def objective_slsqp(h):
        return evaluate_c5(h, n_points)
    
    bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    
    try:
        result_slsqp = minimize(
            objective_slsqp,
            best_h_de,
            method='SLSQP',
            bounds=bounds_slsqp,
            constraints=cons,
            options={'ftol': 1e-12, 'maxiter': 100}
        )
        best_h = result_slsqp.x
        best_c5 = result_slsqp.fun
    except:
        best_h = best_h_de
        best_c5 = best_c5_de
    
    # Compare with initial guesses
    for guess in initial_guesses:
        current_c5 = evaluate_c5(guess, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = guess
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
