    initial_guesses.append(generate_interval_guess(n_points, intervals_2))
    intervals_3 = [(0.0, 0.6667), (0.6667, 1.3333), (1.3333, 2.0)]
    initial_guesses.append(generate_interval_guess(n_points, intervals_3))
    intervals_4 = [(0.0, 0.5), (0.5, 1.0), (1.0, 1.5), (1.5, 2.0)]
    initial_guesses.append(generate_interval_guess(n_points, intervals_4))
    intervals_1 = [(0.5, 1.5)]
    initial_guesses.append(generate_interval_guess(n_points, intervals_1))
    
    # Add symmetric pattern
    symmetric = np.zeros(n_points)
    for i in range(n_points):
        symmetric[i] = 1.0 if i < n_points//2 else 0.0
    initial_guesses.append(symmetric)
    
    # Add periodic pattern
    periodic = np.zeros(n_points)
    for i in range(n_points):
        periodic[i] = 0.5 + 0.5 * np.sin(2 * np.pi * i / n_points)
    initial_guesses.append(periodic)
    
    # Add random guesses
    for _ in range(3):
        guess = np.random.uniform(0, 1, n_points)
        guess = guess * (target_sum / np.sum(guess))
        initial_guesses.append(guess)
    
    # Prepare initial population for differential evolution
    initial_population = [guess for guess in initial_guesses[:5]]
    initial_population = np.array(initial_population)
    
    # Define the objective function
    def objective(h):
        # Normalize h to meet the sum constraint
        h_normalized = h * (target_sum / np.sum(h))
        return evaluate_c5(h_normalized, n_points)
    
    # Run differential evolution
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    start_time = time.time()
    result = differential_evolution(
        objective,
        bounds,
        popsize=20,
        mutation=(0.5, 1.0),
        recombination=0.9,
        strategy='best1bin',
        maxiter=100,
        tol=1e-12,
        init=initial_population,
        polish=True,
        vectorized=True,
        seed=seed
    )
    
    # Check if the result is successful

        best_h = result.x
        best_c5 = result.fun
    else:
        # Fallback to the initial guesses
        best_h = initial_guesses[0]
        best_c5 = evaluate_c5(best_h, n_points)

    # Check elapsed time
