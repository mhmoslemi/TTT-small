            mutation=(0.5, 1),
            recombination=0.9,
            tol=1e-8,
            maxiter=100,
            polish=True,
            disp=False,
            seed=seed
        )
        # Check if the result is successful
        if result.success:
            current_sum = np.sum(result.x)
            penalty = 1e6 * np.abs(current_sum - target_sum)
            current_c5 = result.fun - penalty
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
