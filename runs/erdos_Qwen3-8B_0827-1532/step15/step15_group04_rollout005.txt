                h[i] = 1.0
    # Adjust to meet the sum constraint
    total = np.sum(h)
    if total == 0:
        return h
    scale = (n_points / 2) / total
    return h * scale

def run(seed=42, budget_s=100, **kwargs):

    initial_population.append(generate_interval_guess(n_points, [(0.0, 0.5), (1.5, 2.0)]))
    initial_population.append(generate_interval_guess(n_points, [(0.0, 0.6667), (0.6667, 1.3333), (1.3333, 2.0)]))
    initial_population.append(generate_interval_guess(n_points, [(0.5, 1.5)]))
    
    # Add symmetric pattern
    symmetric = np.zeros(n_points)
    for i in range(n_points):
        symmetric[i] = 1.0 if i < n_points//2 else 0.0
    initial_population.append(symmetric)
    
    # Add periodic pattern
    periodic = np.zeros(n_points)
    for i in range(n_points):
        periodic[i] = 0.5 + 0.5 * np.sin(2 * np.pi * i / n_points)
    initial_population.append(periodic)
    
    # Add random guesses
    for _ in range(3):
        guess = np.random.uniform(0, 1, n_points)
        guess = guess * (target_sum / np.sum(guess))
        initial_population.append(guess)

        total = np.sum(h)
        if total == 0:
            return float('inf')
        scale = (target_sum) / total
        scaled_h = h * scale
        # Clip to [0, 1] to avoid exceeding bounds
        scaled_h = np.clip(scaled_h, 0, 1)
        # Compute overlap
        one_minus_h = 1.0 - scaled_h
        corr = np.correlate(scaled_h, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    # Set bounds for each parameter (0 to 1)
    bounds = [(0.0, 1.0) for _ in range(n_points)]
