import time

        perturbed = initial_guess.copy()
        noise = np.random.normal(0, 0.01, n_points)

        perturbed = perturbed * (target_sum / np.sum(perturbed))
        initial_guesses.append(perturbed)
    # Add 3 random initial guesses
    for _ in range(3):
        random_h = np.random.uniform(0, 1, n_points)
        random_h = random_h * (target_sum / np.sum(random_h))

    # For each initial guess, run the optimization
    for guess in initial_guesses:
        # Optimization with SLSQP with tighter tolerances and more iterations
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
