            disp=False
        )
        
        # Check if the result is successful
        if result.success:
            current_h = result.x
            current_c5 = objective(current_h) - 1e6 * (np.sum(current_h) - target_sum)**2  # Subtract penalty
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
