    # Add periodic pattern
    periodic = np.zeros(n_points)
    for i in range(n_points):
        periodic[i] = 0.5 + 0.5 * np.sin(2 * np.pi * i / n_points)
    initial_guesses.append(periodic)
    # Add random guesses
    for _ in range(3):
        guess = np.random.uniform(0, 1, n_points)
        guess = guess * (target_sum / np.sum(guess))
        initial_guesses.append(guess)
    
    # Optimization parameters
    max_iterations = 100
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Run differential evolution for each initial guess
    for guess in initial_guesses:
        # Define objective function with penalty
        def objective(h):
            c5 = evaluate_c5(h, n_points)
            penalty = 1e4 * (np.sum(h) - target_sum)**2
            return c5 + penalty
        
        # Run differential evolution
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=max_iterations,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8
        )
        
        if result.success:
            current_c5 = result.fun - 1e4 * (np.sum(result.x) - target_sum)**2
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
