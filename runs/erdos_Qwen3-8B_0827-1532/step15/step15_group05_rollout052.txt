from scipy.optimize import minimize
import time

    dx = 2.0 / n_points
    target_sum = n_points / 2

    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate multiple initial guesses
    initial_guesses = [initial_guess]

    for _ in range(3):
        random_h = np.random.uniform(0, 1, n_points)
        random_h = random_h * (target_sum / np.sum(random_h))
        initial_guesses.append(random_h)
    
    # Generate a structured initial guess (alternating 0 and 1)
    structured_h = np.zeros(n_points)
    structured_h[::2] = 1.0
    structured_h = structured_h * (target_sum / np.sum(structured_h))
    initial_guesses.append(structured_h)
    
    # For each initial guess, run the optimization
    for guess in initial_guesses:
        # Optimization with SLSQP with tighter tolerances
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Try optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 1000, 'ftol': 1e-10, 'eps': 1e-8}
        )
        
        # Check if optimization was successful
        if result.success and result.fun < best_c5:
            best_c5 = result.fun

        else:
            # If optimization failed, try a perturbed guess
            perturbed_guess = guess + 0.01 * np.random.randn(n_points)
            perturbed_guess = np.clip(perturbed_guess, 0.0, 1.0)
