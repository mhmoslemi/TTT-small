from scipy.optimize import minimize

    h_step3 = np.zeros(n_points)
    h_step3[:n_points//3] = 1.0

    h_step3[2*n_points//3:] = 1.0
    sum_h = np.sum(h_step3)
    scale = target_sum / sum_h
    initial_guesses.append(h_step3 * scale)
    
    # Step function: first quarter 1, middle 0, last quarter 1
    h_step4 = np.zeros(n_points)

    scale = target_sum / sum_h
    initial_guesses.append(h_step4 * scale)

        if i % 3 == 0:

            h_alt_var[i] = 0.0

        bounds = [(0.0, 1.0) for _ in range(n_points)]
