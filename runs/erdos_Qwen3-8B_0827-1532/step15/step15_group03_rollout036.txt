    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate diverse initial guesses
    initial_guesses = [initial_guess.copy()]
    
    # Add interval-based patterns
    intervals_2_sep = [(0.0, 0.5), (1.5, 2.0)]
    initial_guesses.append(generate_interval_guess(n_points, intervals_2_sep))
    intervals_3_spaced = [(0.0, 0.6667), (0.6667, 1.3333), (1.3333, 2.0)]
    initial_guesses.append(generate_interval_guess(n_points, intervals_3_spaced))
    intervals_1_centered = [(0.5, 1.5)]
    initial_guesses.append(generate_interval_guess(n_points, intervals_1_centered))
    intervals_4_spaced = [(0.0, 0.5), (0.5, 1.0), (1.0, 1.5), (1.5, 2.0)]
    initial_guesses.append(generate_interval_guess(n_points, intervals_4_spaced))
    intervals_3_varied = [(0.0, 0.6), (0.6, 1.2), (1.2, 2.0)]
    initial_guesses.append(generate_interval_guess(n_points, intervals_3_varied))
    intervals_2_overlapping = [(0.0, 1.0), (0.5, 1.5)]
    initial_guesses.append(generate_interval_guess(n_points, intervals_2_overlapping))
    
    # Add symmetric pattern
    symmetric = np.zeros(n_points)
    for i in range(n_points):
        symmetric[i] = 1.0 if i < n_points//2 else 0.0
    initial_guesses.append(symmetric)
    # Add periodic pattern
    periodic = np.zeros(n_points)
    for i in range(n_points):
        periodic[i] = 0.5 + 0.5 * np.sin(2 * np.pi * i / n_points)
    initial_guesses.append(periodic)
    # Add random guesses
    for _ in range(3):
        guess = np.random.uniform(0, 1, n_points)
        guess = guess * (target_sum / np.sum(guess))
        initial_guesses.append(guess)
    
    # Define objective function
    def objective(h):
        h_normalized = h * (target_sum / np.sum(h)) if np.sum(h) != 0 else h
        one_minus_h = 1.0 - h_normalized
        corr = np.correlate(h_normalized, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    # Bounds for each parameter
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Run differential evolution
    result = differential_evolution(
        objective,
        bounds,
        popsize=15,
        strategy='best1bin',
        maxiter=100,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.9,
        seed=seed
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        # If time is up, return the best solution found so far
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
