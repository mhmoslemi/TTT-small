        popsize=len(initial_guesses),
        maxiter=100,
        strategy='best1bin',
        mutation=(0.5, 1),
        recombination=0.9,
        tol=1e-12,
        disp=False,
        polish=True,
        vectorized=True,
        x0=initial_guesses
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        # If time is up, return the best found so far
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
