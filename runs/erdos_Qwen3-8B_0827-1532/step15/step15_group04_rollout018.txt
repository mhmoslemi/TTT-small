    symmetric = np.zeros(n_points)
    for i in range(n_points):
        symmetric[i] = 1.0 if i < n_points//2 else 0.0
    initial_guesses.append(symmetric)
    # Add periodic pattern
    periodic = np.zeros(n_points)
    for i in range(n_points):
        periodic[i] = 0.5 + 0.5 * np.sin(2 * np.pi * i / n_points)
    initial_guesses.append(periodic)
    # Add random guesses
    for _ in range(3):
        guess = np.random.uniform(0, 1, n_points)
        guess = guess * (target_sum / np.sum(guess))
        initial_guesses.append(guess)
    
    # Optimization parameters
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Define the objective function
    def objective(h):
        sum_h = np.sum(h)
        penalty = 1e4 * (sum_h - target_sum)**2
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr) * dx
        return max_corr + penalty
    
    # Run differential evolution
    start_time = time.time()
    result = differential_evolution(
        objective,
        bounds,
        popsize=10,
        maxiter=100,
        tol=1e-10,
        mutation=(0.5, 1),
        recombination=0.8,
        strategy='best1bin',
        seed=seed,
        disp=False
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        # If time is up, return the best found so far
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
