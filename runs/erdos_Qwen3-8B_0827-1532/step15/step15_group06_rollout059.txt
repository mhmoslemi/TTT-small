import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_c5(h, n_points):
    dx = 2.0 / n_points
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    target_sum = n_points / 2
    # Adjust initial guess to meet the sum constraint
    current_sum = np.sum(initial_guess)
    scale = target_sum / current_sum
    initial_guess = initial_guess * scale
    
    # Define the objective function
    def objective(h):
        # Adjust sum
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.inf
        scale = target_sum / current_sum
        adjusted_h = h * scale
        return evaluate_c5(adjusted_h, n_points)
    
    # Define bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Run differential evolution with early stopping
    start_time = time.time()
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=50,
        maxiter=100,
        tol=1e-12,
        mutation=(0.5, 1),
        recombination=0.9,
        polish=True
    )
    elapsed = time.time() - start_time
    
    # Early stopping if time is exhausted
    if elapsed >= budget_s:
        return initial_guess, evaluate_c5(initial_guess, n_points), n_points
    
    best_h = result.x
    best_c5 = result.fun
    
    return best_h, best_c5, n_points
