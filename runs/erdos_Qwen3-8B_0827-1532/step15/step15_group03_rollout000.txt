import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Define objective function with penalty for sum constraint
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        penalty = 1e6 * np.abs(np.sum(h) - target_sum)
        return max_corr * dx + penalty
    
    # Bounds for differential evolution
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Run differential evolution to find a good candidate
    result_de = differential_evolution(
        objective, bounds, strategy='best1bin',
        popsize=15, maxiter=50, tol=1e-6,
        mutation=(0.5, 1), recombination=0.8,
        seed=seed
    )
    
    # Refine with SLSQP to meet the sum constraint exactly
    def objective_slsqp(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    result_slsqp = minimize(
        objective_slsqp, result_de.x,
        method='SLSQP', bounds=bounds, constraints=cons,
        options={'ftol': 1e-12, 'maxiter': 100}
    )
    
    # Update best solution
    if result_slsqp.fun < best_c5:
        best_c5 = result_slsqp.fun

    # Check elapsed time and allow additional refinement if needed

        )

            best_c5 = result_slsqp.fun
            best_h = result_slsqp.x
    
    return best_h, best_c5, n_points
