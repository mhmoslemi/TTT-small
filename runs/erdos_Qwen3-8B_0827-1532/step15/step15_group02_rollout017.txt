import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate initial guesses
    initial_guesses = [initial_h_values]
    # Alternating pattern
    h_alternating = np.zeros(n_points)
    h_alternating[::2] = 1.0
    initial_guesses.append(h_alternating)
    # Random initial guess
    h_random = np.random.uniform(0, 1, n_points)
    h_random = h_random * (target_sum / np.sum(h_random))
    initial_guesses.append(h_random)
    
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    
    start_time = time.time()
    
    # Parameters for optimization
    max_iterations_per_optimize = 100
    
    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': max_iterations_per_optimize, 'ftol': 1e-9}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Perturb and optimize the best solution
    num_perturbations = 50
    for _ in range(num_perturbations):
        # Generate a perturbed guess
        perturbed = best_h + np.random.uniform(-0.05, 0.05, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
