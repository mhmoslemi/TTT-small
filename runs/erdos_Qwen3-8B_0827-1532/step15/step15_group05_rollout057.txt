import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate diverse initial guesses
    initial_guesses = [initial_guess]
    for _ in range(3):
        # Perturb initial guess
        perturbed = initial_guess * (1 + 0.1 * np.random.randn(n_points))
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * (target_sum / np.sum(perturbed))
        initial_guesses.append(perturbed)
    for _ in range(3):
        # Random guess
        random_h = np.random.uniform(0, 1, n_points)
        random_h = random_h * (target_sum / np.sum(random_h))
        initial_guesses.append(random_h)
    
    # Optimize each initial guess with differential evolution
    for guess in initial_guesses:
        def objective(h):
            h_normalized = h * (target_sum / np.sum(h))
            one_minus_h = 1.0 - h_normalized
            corr = np.correlate(h_normalized, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Run differential evolution with adaptive parameters
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=100,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Fallback to initial guess if no improvement
    if best_c5 == float('inf'):
        best_h = initial_guess
        best_c5 = np.max(np.correlate(initial_guess, 1 - initial_guess, mode='full')) * dx
    
    return best_h, best_c5, n_points
