        scaled_guess = guess * (target_sum / np.sum(guess))
        current_c5 = evaluate_c5(scaled_guess, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = scaled_guess
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
