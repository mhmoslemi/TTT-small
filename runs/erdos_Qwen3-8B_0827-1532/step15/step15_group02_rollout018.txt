    # Ensure the sum constraint is met
    best_h = best_h * (target_sum / np.sum(best_h)) if np.sum(best_h) != 0 else best_h
    
    elapsed = time.time() - start_time
    # Return the best solution found
    return best_h, best_c5, n_points
