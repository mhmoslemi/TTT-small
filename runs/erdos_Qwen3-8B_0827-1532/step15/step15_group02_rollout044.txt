import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    
    start_time = time.time()
    
    # Generate diverse initial guesses
    initial_guesses = []
    initial_guesses.append(initial_h_values.copy())
    
    # Random binary pattern
    h_random = np.zeros(n_points)
    h_random[:n_points//2] = 1
    np.random.shuffle(h_random)
    initial_guesses.append(h_random)
    
    # Uniform distribution
    h_uniform = np.full(n_points, 0.5)
    initial_guesses.append(h_uniform)
    
    # Alternating pattern
    h_alternate = np.zeros(n_points)
    for i in range(n_points):
        h_alternate[i] = 1 if (i % 3 == 0 or i % 3 == 1) else 0
    h_alternate = h_alternate * (target_sum / np.sum(h_alternate))
    initial_guesses.append(h_alternate)
    
    # Optimization loop
    for guess in initial_guesses:
        # Normalize to meet sum constraint
        guess = guess * (target_sum / np.sum(guess))
        guess = np.clip(guess, 0, 1)
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Constraints and bounds
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Optimization with SLSQP
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 300}
        )
        
        # Update best solution
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    return best_h, best_c5, n_points
