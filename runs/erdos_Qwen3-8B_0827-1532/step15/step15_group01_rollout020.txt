    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate multiple initial guesses including symmetric pattern
    def create_symmetric_guess(n_points):

            h[i] = 1.0
        for i in range(n_points - n_points // 2, n_points):
            h[i] = 1.0
        return h
    
    initial_guesses = [initial_guess, create_symmetric_guess(n_points)]
    
    # Function to optimize a single guess
    def optimize_guess(guess):
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 300}
        )
        return result.fun, result.x
    
    # Use multiprocessing to run optimizations in parallel
    with Pool() as pool:
        results = pool.map(optimize_guess, initial_guesses)
    
    # Update best solution
    for c5, h in results:

            best_h = h

    num_perturbations = 150  # Reduced from 200

        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
