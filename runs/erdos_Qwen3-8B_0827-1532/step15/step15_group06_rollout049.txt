    for start, end in intervals:
        for i in range(n_points):
            if start <= i*dx <= end:
                h[i] = 1.0
    # Adjust to meet the sum constraint
    total = np.sum(h)
    if total == 0:

    scale = (n_points / 2) / total
    return h * scale
