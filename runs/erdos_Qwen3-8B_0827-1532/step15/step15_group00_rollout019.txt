    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate diverse initial guesses
    initial_guesses = []
    initial_guesses.append(initial_guess.copy())  # Original initial guess
    
    # Step function: first half 1, second half 0
    h_step = np.zeros(n_points)
    h_step[:n_points//2] = 1.0
    sum_h = np.sum(h_step)
    scale = target_sum / sum_h
    initial_guesses.append(h_step * scale)
    
    # Alternating 1 and 0
    h_alt = np.zeros(n_points)
    h_alt[::2] = 1.0
    sum_h = np.sum(h_alt)
    scale = target_sum / sum_h
    initial_guesses.append(h_alt * scale)
    
    # Random initial guess
    initial_guesses.append(np.random.uniform(0, 1, n_points))
    
    # Add more structured initial guesses
    # Step function with different widths
    h_step2 = np.zeros(n_points)
    h_step2[:n_points//3] = 1.0
    sum_h = np.sum(h_step2)
    scale = target_sum / sum_h
    initial_guesses.append(h_step2 * scale)
    
    # Pattern: 1, 1, 0, 0, 1, 1, 0, 0,...
    h_pattern = np.zeros(n_points)
    for i in range(n_points):
        if i % 4 < 2:
            h_pattern[i] = 1.0
    sum_h = np.sum(h_pattern)
    scale = target_sum / sum_h
    initial_guesses.append(h_pattern * scale)
    
    # Perturb each initial guess
    num_initial_guesses = len(initial_guesses)
    for i in range(num_initial_guesses):
        perturbed = initial_guesses[i] + np.random.uniform(-0.1, 0.1, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        initial_guesses[i] = perturbed
    
    # Optimize each initial guess with differential evolution
    for guess in initial_guesses:
        # Objective function with penalty
        def objective(h):
            sum_h = np.sum(h)
            penalty = 1e6 * np.abs(sum_h - target_sum)
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx + penalty
        
        # Bounds
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Run differential evolution
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=100,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Perturbation phase on the best_h found so far
    num_perturbations = 400  # Increased from 300
    phase_shift = 0.15
    phase_decay = 0.96
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function with penalty
        def objective(h):
            sum_h = np.sum(h)
            penalty = 1e6 * np.abs(sum_h - target_sum)
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx + penalty
        
        # Run differential evolution on the perturbed guess
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=100,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
