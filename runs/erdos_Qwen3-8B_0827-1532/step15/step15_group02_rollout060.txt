from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Define the objective function
    def objective(h):
        # Scale h to meet the sum constraint
        sum_h = np.sum(h)
        if sum_h == 0:
            return float('inf')  # Avoid division by zero
        scale = target_sum / sum_h
        h_scaled = h * scale
        # Compute the C5 value
        one_minus_h = 1.0 - h_scaled
        corr = np.correlate(h_scaled, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    # Bounds for each h[i] are [0, 1]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Run differential evolution
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=15,
        maxiter=100,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.9,
        seed=seed,
        disp=False
    )
    
    # Check if the result is successful
    if result.success:
        best_h = result.x
        best_c5 = result.fun
    
    # Check elapsed time and return the best solution
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
