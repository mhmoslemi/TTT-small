import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate multiple initial guesses
    initial_guesses = [initial_guess]
    # Generate 5 random initial guesses
    for _ in range(5):
        random_h = np.random.uniform(0, 1, n_points)
        random_h = random_h * (target_sum / np.sum(random_h))
        initial_guesses.append(random_h)
    
    # For each initial guess, run the optimization
    for guess in initial_guesses:
        # Optimization with COBYLA method
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='COBYLA',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 1000, 'disp': False}
        )
        
        # Fallback to perturbed guess if optimization fails
        if not result.success:
            perturbed = guess + np.random.normal(0, 0.05, n_points)
            perturbed = np.clip(perturbed, 0, 1)
            perturbed = perturbed * (target_sum / np.sum(perturbed))
            result = minimize(
                objective,
                perturbed,
                method='COBYLA',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 500, 'disp': False}
            )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Early exit if time is exhausted
    if time.time() - start_time > budget_s - 1:
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
