    np.random.seed(seed)
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate diverse initial guesses
    initial_guesses = [initial_guess.copy()]
    
    # Add interval-based patterns
    intervals_2_sep = [(0.0, 0.5), (1.5, 2.0)]
    initial_guesses.append(generate_interval_guess(n_points, intervals_2_sep))
    intervals_3_spaced = [(0.0, 0.6667), (0.6667, 1.3333), (1.3333, 2.0)]
    initial_guesses.append(generate_interval_guess(n_points, intervals_3_spaced))
    intervals_1_centered = [(0.5, 1.5)]
    initial_guesses.append(generate_interval_guess(n_points, intervals_1_centered))
    
    # Add symmetric pattern
    symmetric = np.zeros(n_points)
    for i in range(n_points):
        symmetric[i] = 1.0 if i < n_points//2 else 0.0
    initial_guesses.append(symmetric)
    # Add periodic pattern
    periodic = np.zeros(n_points)
    for i in range(n_points):
        periodic[i] = 0.5 + 0.5 * np.sin(2 * np.pi * i / n_points)
    initial_guesses.append(periodic)
    # Add random guesses
    for _ in range(3):
        guess = np.random.uniform(0, 1, n_points)
        guess = guess * (target_sum / np.sum(guess))
        initial_guesses.append(guess)
    
    # Use differential evolution on the initial guesses
    for guess in initial_guesses:
        # Define objective function with penalty
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr) * dx
            penalty = 1e4 * abs(np.sum(h) - target_sum)
            return max_corr + penalty
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = differential_evolution(objective, bounds, strategy='best1bin', maxiter=50, popsize=15, tol=1e-6, mutation=(0.5, 1), recombination=0.8)
        
        if result.success:
            current_c5 = result.fun - 1e4 * abs(np.sum(result.x) - target_sum)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # Refine with SLSQP
    def objective_slsqp(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    
    result_slsqp = minimize(
        objective_slsqp,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-10, 'maxiter': 100}
    )
    
    if result_slsqp.success and result_slsqp.fun < best_c5:
        best_c5 = result_slsqp.fun
        best_h = result_slsqp.x
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
