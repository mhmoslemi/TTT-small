import numpy as np
from scipy.optimize import differential_evolution
import time

def generate_interval_guess(n_points, intervals):
    h = np.zeros(n_points)
    dx = 2.0 / n_points
    for start, end in intervals:
        for i in range(n_points):
            if start <= i*dx <= end:
                h[i] = 1.0
    # Adjust to meet the sum constraint
    total = np.sum(h)
    if total == 0:
        return h
    scale = (n_points / 2) / total
    return h * scale

def evaluate_c5(h, n_points):
    dx = 2.0 / n_points
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return np.max(corr) * dx
