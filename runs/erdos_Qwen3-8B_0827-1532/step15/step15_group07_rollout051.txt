    initial_guesses.append(generate_interval_guess(n_points, intervals_2))
    # Three intervals of length 1/3
    intervals_3 = [(0.0, 0.6667), (0.6667, 1.3333), (1.3333, 2.0)]
    initial_guesses.append(generate_interval_guess(n_points, intervals_3))
    # Four intervals of length 0.5
    intervals_4 = [(0.0, 0.5), (0.5, 1.0), (1.0, 1.5), (1.5, 2.0)]
    initial_guesses.append(generate_interval_guess(n_points, intervals_4))
    # Single interval of length 1
    intervals_1 = [(0.5, 1.5)]
    initial_guesses.append(generate_interval_guess(n_points, intervals_1))
    
    # Add symmetric pattern
    symmetric = np.zeros(n_points)
    for i in range(n_points):
        symmetric[i] = 1.0 if i < n_points//2 else 0.0
    initial_guesses.append(symmetric)
    # Add periodic pattern
    periodic = np.zeros(n_points)
    for i in range(n_points):
        periodic[i] = 0.5 + 0.5 * np.sin(2 * np.pi * i / n_points)
    initial_guesses.append(periodic)
    # Add random guesses
    for _ in range(5):
        guess = np.random.uniform(0, 1, n_points)
        guess = guess * (target_sum / np.sum(guess))
        initial_guesses.append(guess)
    
    # Optimization parameters
    max_iterations = 1000
    perturbation_count = 800
    noise_scales = [0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
    
    for perturb_idx in range(perturbation_count):
        # Select a guess
        if perturb_idx < len(initial_guesses):
            perturbed = initial_guesses[perturb_idx].copy()
        else:
            # Generate a new perturbation based on best_h
            scale_idx = perturb_idx // (perturbation_count // len(noise_scales))
            noise_scale = noise_scales[scale_idx % len(noise_scales)]
            noise = np.random.uniform(-noise_scale, noise_scale, n_points)
            perturbed = best_h + noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
        
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='COBYLA',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': max_iterations, 'disp': False}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
