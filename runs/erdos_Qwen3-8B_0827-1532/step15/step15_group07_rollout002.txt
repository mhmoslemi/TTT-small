    h = np.zeros(n_points)
    dx = 2.0 / n_points
    for start, end in intervals:
        for i in range(n_points):
            if start <= i*dx <= end:
                h[i] = 1.0
    total = np.sum(h)
    if total == 0:
        return h
    scale = (n_points / 2) / total
    return h * scale

def objective(h, dx, target_sum):
    sum_h = np.sum(h)
    if sum_h == 0:
        return float('inf')
    h_scaled = h * (target_sum / sum_h)
    one_minus_h = 1.0 - h_scaled
    corr = np.correlate(h_scaled, one_minus_h, mode='full')
    return np.max(corr) * dx
