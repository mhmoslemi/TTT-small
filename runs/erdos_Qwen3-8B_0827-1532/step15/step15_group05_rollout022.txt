    for _ in range(3):
        random_h = np.random.uniform(0, 1, n_points)
        random_h = random_h * (target_sum / np.sum(random_h))
        initial_guesses.append(random_h)
    
    # For each initial guess, run the optimization
    for guess in initial_guesses:
        # Define the objective function
        def objective(h):
            max_overlap = 0.0
            for m in range(-n_points, n_points + 1):
                integral = 0.0
                for i in range(n_points):
                    j = i + m
                    if 0 <= j < n_points:
                        integral += h[i] * (1 - h[j]) * dx
                max_overlap = max(max_overlap, integral)
            return max_overlap
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-8}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    return best_h, best_c5, n_points
