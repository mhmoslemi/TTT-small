from scipy.optimize import differential_evolution, minimize
import time

    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    start_time = time.time()

    h_step = np.zeros(n_points)

    sum_h = np.sum(h_step)
    scale = target_sum / sum_h
    initial_guesses.append(h_step * scale)
    
    # Alternating 1 and 0
    h_alt = np.zeros(n_points)
    h_alt[::2] = 1.0
    sum_h = np.sum(h_alt)
    scale = target_sum / sum_h

    # Thirds function: first third 1, second third 0, last third 1
    h_thirds = np.zeros(n_points)
    h_thirds[:n_points//3] = 1.0
    h_thirds[n_points//3:2*n_points//3] = 0.0
    h_thirds[2*n_points//3:] = 1.0
    sum_h = np.sum(h_thirds)

    initial_guesses.append(h_thirds * scale)

    initial_guesses.append(np.random.uniform(0, 1, n_points))
    
    # Generate more initial guesses by perturbing existing ones
    for guess in initial_guesses:
        perturbed = guess + np.random.uniform(-0.1, 0.1, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        initial_guesses.append(perturbed)
    
    # Optimize each initial guess with differential evolution
    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Bounds and constraints
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Use differential evolution for global search
        result = differential_evolution(
            objective,
            bounds,

            tol=1e-9,
            maxiter=100,
            disp=False
        )

    phase_shift = 0.15

    for perturbation_idx in range(num_perturbations):

        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP for local refinement
        bounds = [(0.0, 1.0) for _ in range(n_points)]
