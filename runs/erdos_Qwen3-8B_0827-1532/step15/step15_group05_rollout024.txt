import numpy as np
from scipy.optimize import minimize
import time
from multiprocessing import Pool, cpu_count

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate diverse initial guesses: original + perturbed
    initial_guesses = [initial_guess]
    for _ in range(5):
        noise = np.random.normal(0, 0.05, n_points)
        perturbed = initial_guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * (target_sum / np.sum(perturbed))
        initial_guesses.append(perturbed)
    
    # Define objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    # Define constraints
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    
    # Optimization function for parallel processing
    def optimize(guess):
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 1000, 'ftol': 1e-10, 'eps': 1e-8}
        )
        return result
    
    # Run optimizations in parallel
    with Pool(processes=cpu_count()) as pool:
        results = pool.map(optimize, initial_guesses)
    
    # Track best result
    for result in results:
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Check remaining time for additional optimizations
    elapsed = time.time() - start_time
    if elapsed < budget_s:
        # Generate more guesses if time remains
        additional_guesses = []
        for _ in range(3):
            noise = np.random.normal(0, 0.03, n_points)
            perturbed = initial_guess + noise
            perturbed = np.clip(perturbed, 0, 1)
            perturbed = perturbed * (target_sum / np.sum(perturbed))
            additional_guesses.append(perturbed)
        results.extend(pool.map(optimize, additional_guesses))
        for result in results:
            if result.success and result.fun < best_c5:
                best_c5 = result.fun
                best_h = result.x
    
    return best_h, best_c5, n_points
