            return max_overlap
        
        # Constraints
        constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]
        
        # Bounds
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Optimization with SLSQP
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'ftol': 1e-9, 'maxiter': 200}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Perturbation phase on the best_h found so far
    num_perturbations = 500
    phase_shift = 0.15
    phase_decay = 0.96
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective_perturbed(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_overlap = np.max(corr) * dx
            return max_overlap
        
        # Constraints
        constraints_perturbed = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]
        
        # Bounds
        bounds_perturbed = [(0.0, 1.0) for _ in range(n_points)]
        
        # Optimization with SLSQP
        result = minimize(
            objective_perturbed,
            perturbed,
            method='SLSQP',
            bounds=bounds_perturbed,
            constraints=constraints_perturbed,
            options={'ftol': 1e-9, 'maxiter': 200}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points

def compute_c5(h, dx):
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return np.max(corr) * dx
