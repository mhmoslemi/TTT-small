from scipy.optimize import minimize
import time

    initial_guess = initial_guess * scale
    best_h = initial_guess.copy()

    # Define objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    # Define equality constraint for sum of h
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    # Optimization with SLSQP
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Perform optimization
    result = minimize(
        objective,
        initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-9, 'maxiter': 200}
    )
    
    if result.success and result.fun < best_c5:
        best_c5 = result.fun
