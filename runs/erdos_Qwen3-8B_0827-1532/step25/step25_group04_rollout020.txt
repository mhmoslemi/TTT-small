        elif np.random.rand() < 0.5:
            h_rand_seg[i] = 0.5
    sum_h = np.sum(h_rand_seg)
    scale = target_sum / sum_h
    initial_guesses.append(h_rand_seg * scale)
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Objective function with penalty for sum constraint
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_overlap = np.max(corr) * dx
            penalty = 1e3 * (np.sum(h) - target_sum)**2  # Reduced penalty
            return max_overlap + penalty
        
        # Optimization with L-BFGS-B
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = minimize(
            objective,
            guess,
            method='L-BFGS-B',
            bounds=bounds,
            options={'ftol': 1e-9, 'maxiter': 200}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Perturbation phase on the best_h found so far
    num_perturbations = 500
    phase_shift = 0.15
    phase_decay = 0.96
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function with penalty for sum constraint
        def objective_perturbed(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_overlap = np.max(corr) * dx
            penalty = 1e3 * (np.sum(h) - target_sum)**2
            return max_overlap + penalty
        
        # Optimization with L-BFGS-B
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = minimize(
            objective_perturbed,
            perturbed,
            method='L-BFGS-B',
            bounds=bounds,
            options={'ftol': 1e-9, 'maxiter': 200}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points

def compute_c5(h, dx):
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return np.max(corr) * dx
