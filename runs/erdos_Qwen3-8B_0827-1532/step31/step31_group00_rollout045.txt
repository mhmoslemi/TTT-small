import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def compute_c5(h):
        h_sum = np.sum(h)
        if h_sum == 0:
            return float('inf')
        h_scaled = h * (target_sum / h_sum)
        one_minus_h = 1.0 - h_scaled
        corr = np.correlate(h_scaled, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    # Initial guess from provided initial_h_values
    initial_guess = initial_h_values.copy()
    initial_guess = np.clip(initial_guess, 0, 1)
    initial_guess = initial_guess * (target_sum / np.sum(initial_guess))
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Bounds for each parameter
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Run differential evolution
    start_time = time.time()
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.8,
        tol=1e-8,
        maxiter=100,
        seed=seed
    )
    
    best_h = result.x
    best_c5 = result.fun
    
    # Refinement with SLSQP
    def objective_slsqp(h):
        return compute_c5(h)
    
    bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]
    cons_slsqp = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    
    result_slsqp = minimize(
        objective_slsqp,
        best_h,
        method='SLSQP',
        bounds=bounds_slsqp,
        constraints=cons_slsqp,
        options={'ftol': 1e-8, 'maxiter': 200}
    )
    
    if result_slsqp.success and result_slsqp.fun < best_c5:
