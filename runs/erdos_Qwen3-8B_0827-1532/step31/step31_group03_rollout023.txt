    h_seven[segment_length:2*segment_length] = 0.5
    h_seven[2*segment_length:3*segment_length] = 1.0

    h_seven[4*segment_length:5*segment_length] = 1.0
    h_seven[5*segment_length:6*segment_length] = 0.5

    sum_h = np.sum(h_seven)
    scale = target_sum / sum_h
    initial_guesses.append(h_seven * scale)
    
    # Eight-segment pattern
    h_eight = np.zeros(n_points)
    segment_length = n_points // 8
    h_eight[:segment_length] = 1.0
    h_eight[segment_length:2*segment_length] = 0.5
    h_eight[2*segment_length:3*segment_length] = 1.0
    h_eight[3*segment_length:4*segment_length] = 0.5
    h_eight[4*segment_length:5*segment_length] = 1.0

    h_eight[6*segment_length:7*segment_length] = 1.0

    initial_guesses.append(h_eight * scale)

    for guess in initial_guesses:
        # Objective function with penalty for sum constraint
        def objective(h):

            corr = np.correlate(h, one_minus_h, mode='full')

            penalty = 1e4 * (np.sum(h) - target_sum)**2
            return max_corr + penalty
        
        # Bounds for each parameter
        bounds = [(0.0, 1.0) for _ in range(n_points)]

        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=15,
            mutation=(0.5, 1),
            recombination=0.8,
            tol=1e-10,
            maxiter=100,
            disp=False
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x

    num_perturbations = 300
    phase_shift = 0.15
    phase_decay = 0.97
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.normal(0, noise_scale, n_points)
        perturbed = best_h + noise

def compute_c5(h, dx):
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return np.max(corr) * dx
