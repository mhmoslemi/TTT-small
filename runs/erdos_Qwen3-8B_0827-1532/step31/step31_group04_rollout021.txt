    segment_length = n_points // 6
    h_six[:segment_length] = 1.0
    h_six[segment_length:2*segment_length] = 0.75
    h_six[2*segment_length:3*segment_length] = 0.5
    h_six[3*segment_length:4*segment_length] = 0.25
    h_six[4*segment_length:5*segment_length] = 0.0
    h_six[5*segment_length:] = 0.5
    sum_h = np.sum(h_six)
    scale = target_sum / sum_h
    initial_guesses.append(h_six * scale)
    
    # Optimizing each initial guess with differential evolution
    for guess in initial_guesses:
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr) * dx
            penalty = 1e6 * np.abs(np.sum(h) - target_sum)
            return max_corr + penalty
        
        # Run differential evolution
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=15,
            tol=1e-8,
            maxiter=100,
            mutation=(0.5, 1),
            recombination=0.9,
            seed=seed
        )
        
        if result.success:
            current_h = result.x
            current_c5 = compute_c5(current_h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    # Perturbation phase on the best_h found so far
    num_perturbations = 500
    phase_shift = 0.25
    phase_decay = 0.98
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-12, 'maxiter': 1500}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points

def compute_c5(h, dx):
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return np.max(corr) * dx
