    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    
    start_time = time.time()
    
    # Initial guess: provided initial_h_values
    initial_guess = initial_h_values
    initial_guess = np.clip(initial_guess, 0, 1)
    scale = target_sum / np.sum(initial_guess)
    initial_guess = initial_guess * scale
    
    # Define the objective function for differential evolution
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr) * dx
        # Penalty for sum not equal to target_sum
        penalty = 1e4 * abs(np.sum(h) - target_sum)
        return max_corr + penalty
    
    # Use differential evolution to optimize the initial guess
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result_de = differential_evolution(objective, bounds, strategy='best1bin', maxiter=50, popsize=15, tol=1e-6, mutation=(0.5, 1), recombination=0.8)
    best_h = result_de.x
    best_c5 = result_de.fun
    
    # Refine with SLSQP
    def objective_slsqp(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    
    result_slsqp = minimize(
        objective_slsqp,
        best_h,
        method='SLSQP',
        bounds=bounds_slsqp,
        constraints=cons,
        options={'ftol': 1e-12, 'maxiter': 2500}
    )
    
    if result_slsqp.success and result_slsqp.fun < best_c5:
        best_c5 = result_slsqp.fun
        best_h = result_slsqp.x
    
    # Generate diverse initial guesses
    initial_guesses = [initial_guess]
    for _ in range(15):
        h_rand = np.random.uniform(0, 1, n_points)
        sum_h = np.sum(h_rand)
        scale = target_sum / sum_h
        h_rand = h_rand * scale
        initial_guesses.append(h_rand)
    
    # Add predefined segment patterns
    # Two-segment pattern
    h_two = np.zeros(n_points)
    segment_length = n_points // 2
    h_two[:segment_length] = 1.0
    h_two[segment_length:] = 0.5
    sum_h = np.sum(h_two)
    scale = target_sum / sum_h
    h_two = h_two * scale
    initial_guesses.append(h_two)
    
    # Three-segment pattern
    h_three = np.zeros(n_points)
    segment_length = n_points // 3
    h_three[:segment_length] = 1.0
    h_three[segment_length:2*segment_length] = 0.5
    h_three[2*segment_length:] = 1.0
    sum_h = np.sum(h_three)
    scale = target_sum / sum_h
    h_three = h_three * scale
    initial_guesses.append(h_three)
    
    # Four-segment pattern
    h_four = np.zeros(n_points)
    segment_length = n_points // 4
    h_four[:segment_length] = 1.0
    h_four[segment_length:2*segment_length] = 0.5
    h_four[2*segment_length:3*segment_length] = 1.0
    h_four[3*segment_length:] = 0.5
    sum_h = np.sum(h_four)
    scale = target_sum / sum_h
    h_four = h_four * scale
    initial_guesses.append(h_four)
    
    # Five-segment pattern
    h_five = np.zeros(n_points)
    segment_length = n_points // 5
    h_five[:segment_length] = 1.0
    h_five[segment_length:2*segment_length] = 0.75
    h_five[2*segment_length:3*segment_length] = 0.5
    h_five[3*segment_length:4*segment_length] = 0.25
    h_five[4*segment_length:] = 0.0
    sum_h = np.sum(h_five)
    scale = target_sum / sum_h
    h_five = h_five * scale
    initial_guesses.append(h_five)
    
    # Six-segment pattern
    h_six = np.zeros(n_points)
    segment_length = n_points // 6
    h_six[:segment_length] = 1.0
    h_six[segment_length:2*segment_length] = 0.75
    h_six[2*segment_length:3*segment_length] = 0.5
    h_six[3*segment_length:4*segment_length] = 0.25
    h_six[4*segment_length:5*segment_length] = 0.0
    h_six[5*segment_length:] = 0.5
    sum_h = np.sum(h_six)
    scale = target_sum / sum_h
    h_six = h_six * scale
    initial_guesses.append(h_six)
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Differential evolution optimization
        result_de = differential_evolution(objective, bounds, strategy='best1bin', maxiter=50, popsize=15, tol=1e-6, mutation=(0.5, 1), recombination=0.8)
        current_h = result_de.x
        current_c5 = result_de.fun
        
        # SLSQP refinement
        result_slsqp = minimize(
            objective_slsqp,
            current_h,
            method='SLSQP',
            bounds=bounds_slsqp,
            constraints=cons,
            options={'ftol': 1e-12, 'maxiter': 2500}
        )
        
        if result_slsqp.success and result_slsqp.fun < best_c5:
            best_c5 = result_slsqp.fun
            best_h = result_slsqp.x
    
    # Perturbation phase
    num_perturbations = 500
    phase_shift = 0.25
    phase_decay = 0.98
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # SLSQP refinement
        result = minimize(
            objective_slsqp,
            perturbed,
            method='SLSQP',
            bounds=bounds_slsqp,
            constraints=cons,
            options={'ftol': 1e-12, 'maxiter': 1500}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
