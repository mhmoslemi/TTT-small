    h_six[3*segment_length:4*segment_length] = 0.25
    h_six[4*segment_length:5*segment_length] = 0.0
    h_six[5*segment_length:] = 0.5
    sum_h = np.sum(h_six)
    scale = target_sum / sum_h
    initial_guesses.append(h_six * scale)
    
    # Optimizing each initial guess with differential evolution
    for guess in initial_guesses:
        # Objective function
        def objective(h):
            # Ensure sum is correct
            current_sum = np.sum(h)
            if current_sum == 0:
                return float('inf')
            scale = target_sum / current_sum
            h_normalized = h * scale
            h_normalized = np.clip(h_normalized, 0, 1)
            one_minus_h = 1.0 - h_normalized
            corr = np.correlate(h_normalized, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Bounds
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Run differential evolution
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=15,
            maxiter=100,
            tol=1e-12,
            mutation=(0.5, 1),
            recombination=0.9,
            polish=True,
            disp=False
        )
        
        # Check if result is successful
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    # After DE, use SLSQP to refine the best_h
    # Objective function
    def objective_slsqp(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    # Constraints
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]
    
    # Run SLSQP on the best_h found
    result_slsqp = minimize(
        objective_slsqp,
        best_h,
        method='SLSQP',
        bounds=bounds_slsqp,
        constraints=cons,
        options={'ftol': 1e-12, 'maxiter': 2500}
    )
    
    if result_slsqp.success and result_slsqp.fun < best_c5:
        best_c5 = result_slsqp.fun
        best_h = result_slsqp.x
    
    return best_h, best_c5, n_points

def compute_c5(h, dx):
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return np.max(corr) * dx
