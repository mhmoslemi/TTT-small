    scale = target_sum / sum_h
    initial_guesses.append(h_four * scale)
    
    # Five-segment pattern
    h_five = np.zeros(n_points)
    segment_length = n_points // 5
    h_five[:segment_length] = 1.0
    h_five[segment_length:2*segment_length] = 0.75
    h_five[2*segment_length:3*segment_length] = 0.5
    h_five[3*segment_length:4*segment_length] = 0.25
    h_five[4*segment_length:] = 0.0
    sum_h = np.sum(h_five)
    scale = target_sum / sum_h
    initial_guesses.append(h_five * scale)
    
    # Six-segment pattern
    h_six = np.zeros(n_points)
    segment_length = n_points // 6
    h_six[:segment_length] = 1.0
    h_six[segment_length:2*segment_length] = 0.75
    h_six[2*segment_length:3*segment_length] = 0.5
    h_six[3*segment_length:4*segment_length] = 0.25
    h_six[4*segment_length:5*segment_length] = 0.0
    h_six[5*segment_length:] = 0.5
    sum_h = np.sum(h_six)
    scale = target_sum / sum_h
    initial_guesses.append(h_six * scale)
    
    # Optimizing each initial guess using differential evolution
    for guess in initial_guesses:
        # Define objective function with penalty for sum constraint
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            c5 = np.max(corr) * dx
            # Penalty for sum not being correct
            penalty = 1e4 * abs(np.sum(h) - target_sum)
            return c5 + penalty
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Run differential evolution
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=15,
            tol=1e-12,
            mutation=(0.5, 1),
            recombination=0.9,
            seed=seed,
            maxiter=100
        )
        
        # Check if result is successful and update best solution
        if result.success:
            current_c5 = result.fun - 1e4 * abs(np.sum(result.x) - target_sum)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x.copy()
    
    # Perturbation phase using differential evolution on the best_h
    num_perturbations = 50
    phase_shift = 0.25
    phase_decay = 0.98
    
    for perturbation_idx in range(num_perturbations):
        # Generate a perturbed guess based on best_h
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Define objective function with penalty
        def objective_perturb(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            c5 = np.max(corr) * dx
            penalty = 1e4 * abs(np.sum(h) - target_sum)
            return c5 + penalty
        
        bounds_perturb = [(0.0, 1.0) for _ in range(n_points)]
        
        # Run differential evolution on the perturbed guess
        result_perturb = differential_evolution(
            objective_perturb,
            bounds_perturb,
            strategy='best1bin',
            popsize=15,
            tol=1e-12,

            seed=seed,

        )
        
        if result_perturb.success:
            current_c5_perturb = result_perturb.fun - 1e4 * abs(np.sum(result_perturb.x) - target_sum)
            if current_c5_perturb < best_c5:
                best_c5 = current_c5_perturb
                best_h = result_perturb.x.copy()
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points

def compute_c5(h, dx):
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return np.max(corr) * dx
