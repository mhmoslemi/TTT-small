    h_eight = np.zeros(n_points)
    segment_length = n_points // 8
    h_eight[:segment_length] = 1.0
    h_eight[segment_length:2*segment_length] = 0.75
    h_eight[2*segment_length:3*segment_length] = 0.5
    h_eight[3*segment_length:4*segment_length] = 0.25
    h_eight[4*segment_length:5*segment_length] = 0.0
    h_eight[5*segment_length:6*segment_length] = 0.5
    h_eight[6*segment_length:7*segment_length] = 0.75

    for guess in initial_guesses:
        # Objective function with penalty for sum constraint
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr) * dx
            penalty = 1e6 * np.abs(np.sum(h) - target_sum)
            return max_corr + penalty
        
        # Bounds
        bounds = [(0.0, 1.0) for _ in range(n_points)]

            bounds,

    num_perturbations = 500
    phase_shift = 0.25

    for perturbation_idx in range(num_perturbations):

        # Run SLSQP optimization
        result = minimize(

        if result.success and result.fun < best_c5:

            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:

    return best_h, best_c5, n_points

def compute_c5(h, dx):
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return np.max(corr) * dx
