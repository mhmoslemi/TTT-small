        # Define the objective function with penalty for constraint
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr) * dx
            sum_h = np.sum(h)
            penalty = 1e4 * abs(sum_h - target_sum)
            return max_corr + penalty
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Run differential evolution
        result = differential_evolution(objective, bounds, args=(), strategy='best1bin', 
                                       popsize=15, mutation=(0.5, 1), recombination=0.9, 
                                       tol=1e-6, maxiter=100, disp=False)
        
        # Extract the best solution from differential evolution
        best_h_de = result.x
        # Scale to meet the target_sum
        sum_h_de = np.sum(best_h_de)
        scale_de = target_sum / sum_h_de
        best_h_de = best_h_de * scale_de
        
        # Evaluate the C5 value
        current_c5 = compute_c5(best_h_de, dx)
        
        # Update best solution if better
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = best_h_de.copy()
    
    # Perform local optimization on the best solution found
    # Define the objective function without penalty
    def objective_local(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    
    result_local = minimize(
        objective_local,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-12, 'maxiter': 2500}
    )
    
    if result_local.success and result_local.fun < best_c5:
        best_c5 = result_local.fun
        best_h = result_local.x
    
    # Perturbation phase
    num_perturbations = 500
    phase_shift = 0.25
    phase_decay = 0.98
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Local optimization on the perturbed solution
        result_perturb = minimize(
            objective_local,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-12, 'maxiter': 1500}
        )
        
        if result_perturb.success and result_perturb.fun < best_c5:
            best_c5 = result_perturb.fun
            best_h = result_perturb.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points

def compute_c5(h, dx):
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return np.max(corr) * dx
