    h_four[2*segment_length:3*segment_length] = 1.0
    h_four[3*segment_length:] = 0.5
    sum_h = np.sum(h_four)
    scale = target_sum / sum_h
    initial_guesses.append(h_four * scale)
    
    # Five-segment pattern
    h_five = np.zeros(n_points)
    segment_length = n_points // 5
    h_five[:segment_length] = 1.0
    h_five[segment_length:2*segment_length] = 0.75
    h_five[2*segment_length:3*segment_length] = 0.5
    h_five[3*segment_length:4*segment_length] = 0.25
    h_five[4*segment_length:] = 0.0
    sum_h = np.sum(h_five)
    scale = target_sum / sum_h
    initial_guesses.append(h_five * scale)
    
    # Six-segment pattern
    h_six = np.zeros(n_points)
    segment_length = n_points // 6
    h_six[:segment_length] = 1.0
    h_six[segment_length:2*segment_length] = 0.75
    h_six[2*segment_length:3*segment_length] = 0.5
    h_six[3*segment_length:4*segment_length] = 0.25
    h_six[4*segment_length:5*segment_length] = 0.0
    h_six[5*segment_length:] = 0.5
    sum_h = np.sum(h_six)
    scale = target_sum / sum_h
    initial_guesses.append(h_six * scale)
    
    # Define objective function
    def objective(h, dx, target_sum):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        c5 = np.max(corr) * dx
        penalty = 100 * (np.sum(h) - target_sum)**2
        return c5 + penalty
    
    # Optimize each initial guess with differential evolution
    for guess in initial_guesses:
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = differential_evolution(
            objective,
            bounds,
            args=(dx, target_sum),
            strategy='best1bin',
            popsize=15,
            maxiter=100,
            tol=1e-12,
            mutation=(0.5, 1),
            recombination=0.8
        )
        if result.success:
            if result.fun < best_c5:
                best_c5 = result.fun
                best_h = result.x
    
    # Perturbation phase
    num_perturbations = 500
    phase_shift = 0.25
    phase_decay = 0.98
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Optimize with differential evolution
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = differential_evolution(
            objective,
            bounds,
            args=(dx, target_sum),
            strategy='best1bin',
            popsize=15,
            maxiter=100,
            tol=1e-12,
            mutation=(0.5, 1),
            recombination=0.8
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
