    h_eight = np.zeros(n_points)

    for guess in initial_guesses:
        # Objective function
        def objective(h):

            corr = compute_c5(h, dx)
            return corr

        bounds = [(0.0, 1.0) for _ in range(n_points)]

        result = minimize(

    num_perturbations = 300

        elapsed = time.time() - start_time

    return best_h, best_c5, n_points

def compute_c5(h, dx):
    one_minus_h = 1.0 - h
    N = len(h)
    pad_length = 2 * N - 1
    fft_h = np.fft.fft(h, pad_length)
    fft_1_minus_h = np.fft.fft(one_minus_h, pad_length)
    corr_fft = fft_h * np.conj(fft_1_minus_h)
    corr = np.fft.ifft(corr_fft).real
    return np.max(corr) * dx
