            return c5 + penalty
        
        # Bounds for differential evolution
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Run differential evolution
        result = differential_evolution(
            objective_differential_evolution,
            bounds,
            strategy='best1bin',
            maxiter=100,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            polish=True,
            disp=False
        )
        
        # Check if the result is successful
        if result.success:
            current_h = result.x
            current_c5 = compute_c5(current_h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
    
    # Perturbation phase on the best_h found so far
    num_perturbations = 500
    phase_shift = 0.25
    phase_decay = 0.98
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function for SLSQP refinement
        def objective_slsqp(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective_slsqp,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-12, 'maxiter': 1500}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points

def compute_c5(h, dx):
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return np.max(corr) * dx
