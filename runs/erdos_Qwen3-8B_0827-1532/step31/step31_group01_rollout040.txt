            return np.max(corr) * dx
        
        # Optimization with SLSQP with tighter tolerances
        bounds_perturb = [(0.0, 1.0) for _ in range(n_points)]
        cons_perturb = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result_perturb = minimize(
            objective_perturb,
            perturbed,
            method='SLSQP',
            bounds=bounds_perturb,
            constraints=cons_perturb,
            options={'ftol': 1e-10, 'maxiter': 800}
        )
        
        if result_perturb.success and result_perturb.fun < best_c5:
            best_c5 = result_perturb.fun
            best_h = result_perturb.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
