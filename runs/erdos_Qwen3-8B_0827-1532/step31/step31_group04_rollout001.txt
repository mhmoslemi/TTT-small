from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=100, **kwargs):

            else:
                h_pattern[i*segment_length:(i+1)*segment_length] = 0.5
        sum_h = np.sum(h_pattern)
        scale = target_sum / sum_h
        initial_guesses.append(h_pattern * scale)

            h_pattern[i*segment_length:(i+1)*segment_length] = values[i % len(values)]
        sum_h = np.sum(h_pattern)
        scale = target_sum / sum_h
        initial_guesses.append(h_pattern * scale)
    
    # Optimize each initial guess with differential evolution
    for guess in initial_guesses:

            sum_penalty = 1e6 * np.abs(np.sum(h) - target_sum)
            return max_corr + sum_penalty
        
        # Bounds for differential evolution
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Run differential evolution
        result = differential_evolution(

            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
    
    # Local refinement with SLSQP

def compute_c5(h, dx):
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return np.max(corr) * dx
