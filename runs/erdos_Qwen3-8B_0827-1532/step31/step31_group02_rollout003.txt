def compute_c5(h, dx):
    n_points = len(h)
    max_shift = n_points - 1
    max_c5 = 0.0
    for m in range(-max_shift, max_shift + 1):
        i_start = max(0, -m)
        i_end = min(n_points - 1, n_points - 1 - m)
        if i_start > i_end:
            continue
        integral = 0.0
        for i in range(i_start, i_end + 1):
            j = i + m
            integral += h[i] * (1.0 - h[j]) * dx
        max_c5 = max(max_c5, integral)
    return max_c5
