    initial_guesses.append(h_three_seg * scale)
    
    # Four-segment step function
    h_four_seg = np.zeros(n_points)
    seg_lengths = [1/4, 1/4, 1/4, 1/4]
    seg_positions = [0.1, 0.3, 0.5, 0.7]
    for i in range(4):
        start = int(seg_positions[i] * n_points / 2)
        end = int((seg_positions[i] + seg_lengths[i]) * n_points / 2)
        h_four_seg[start:end] = 1.0
    sum_h = np.sum(h_four_seg)
    scale = target_sum / sum_h
    initial_guesses.append(h_four_seg * scale)
    
    # Five-segment step function
    h_five_seg = np.zeros(n_points)
    seg_lengths = [1/5, 1/5, 1/5, 1/5, 1/5]

        start = int(seg_positions[i] * n_points / 2)

    sum_h = np.sum(h_five_seg)
    scale = target_sum / sum_h
    initial_guesses.append(h_five_seg * scale)

    initial_guesses.append(np.random.uniform(0, 1, n_points))

    for guess in initial_guesses:
        # Objective function

            h_scaled = h * (target_sum / np.sum(h))
            one_minus_h = 1.0 - h_scaled
            corr = np.correlate(h_scaled, one_minus_h, mode='full')
            return np.max(corr) * dx

        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',

            maxiter=100,
            tol=1e-10,

            recombination=0.9
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x

    num_perturbations = 200  # Reduced from 300
    phase_shift = 0.25
    phase_decay = 0.95

        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
