        end = (i + 1) * interval_length * n_points
        h_three_intervals[int(start):int(end)] = 1.0
    sum_h = np.sum(h_three_intervals)
    scale = target_sum / sum_h
    initial_guesses.append(h_three_intervals * scale)
    
    # Random binary initial guess
    h_rand_bin = np.random.choice([0, 1], size=n_points)
    sum_h = np.sum(h_rand_bin)
    scale = target_sum / sum_h
    initial_guesses.append(h_rand_bin * scale)
    
    # Symmetric random binary initial guess
    h_sym_bin = np.random.choice([0, 1], size=n_points)
    h_sym_bin = (h_sym_bin + h_sym_bin[::-1]) / 2
    sum_h = np.sum(h_sym_bin)
    scale = target_sum / sum_h
    initial_guesses.append(h_sym_bin * scale)
    
    # Random initial guess
    initial_guesses.append(np.random.uniform(0, 1, n_points))
    
    # Optimize each initial guess with differential evolution
    for guess in initial_guesses:
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        # Run differential evolution
        result = differential_evolution(
            objective_de,
            bounds,
            strategy='best1bin',
            maxiter=100,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed
        )
        # Refine with SLSQP
        def objective_slsqp(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result_slsqp = minimize(
            objective_slsqp,
            result.x,
            method='SLSQP',
            bounds=bounds_slsqp,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 1500}
        )
        
        if result_slsqp.success and result_slsqp.fun < best_c5:
            best_c5 = result_slsqp.fun
            best_h = result_slsqp.x
    
    # Perturbation phase on the best_h found so far
    num_perturbations = 300
    phase_shift = 0.15
    phase_decay = 0.96
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function for SLSQP
        def objective_slsqp(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result_slsqp = minimize(
            objective_slsqp,
            perturbed,
            method='SLSQP',
            bounds=bounds_slsqp,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 400}
        )
        
        if result_slsqp.success and result_slsqp.fun < best_c5:
            best_c5 = result_slsqp.fun
            best_h = result_slsqp.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points

def compute_c5(h, dx):
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return np.max(corr) * dx
