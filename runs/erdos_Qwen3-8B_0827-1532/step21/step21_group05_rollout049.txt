from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    initial_guess = initial_h_values

    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate diverse initial guesses
    initial_guesses = [initial_guess.copy()]
    
    # Step function: first half 1, second half 0
    h_step = np.zeros(n_points)
    h_step[:n_points//2] = 1.0
    sum_h = np.sum(h_step)
    scale = target_sum / sum_h
    initial_guesses.append(h_step * scale)
    
    # Alternating 1 and 0
    h_alt = np.zeros(n_points)
    h_alt[::2] = 1.0
    sum_h = np.sum(h_alt)
    scale = target_sum / sum_h
    initial_guesses.append(h_alt * scale)
    
    # Four-segment step function
    h_four_seg = np.zeros(n_points)
    seg1_start = n_points // 4
    seg1_end = n_points // 2
    seg3_start = 3 * n_points // 4
    seg3_end = n_points
    h_four_seg[seg1_start:seg1_end] = 1.0
    h_four_seg[seg3_start:seg3_end] = 1.0
    sum_h = np.sum(h_four_seg)
    scale = target_sum / sum_h
    initial_guesses.append(h_four_seg * scale)
    
    # Three-segment function
    h_three_seg = np.zeros(n_points)
    seg1_start = n_points // 4
    seg1_end = seg1_start + n_points // 4

    h_three_seg[seg2_start:seg2_end] = 1.0

    for _ in range(3):

    for guess in initial_guesses:

        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')

        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
