import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    
    start_time = time.time()
    
    # Initial guess: provided initial_h_values
    initial_guess = initial_h_values
    initial_guess = np.clip(initial_guess, 0, 1)
    scale = target_sum / np.sum(initial_guess)
    initial_guess = initial_guess * scale
    best_h = initial_guess.copy()
    best_c5 = compute_c5(best_h, dx)
    
    # Optimize from the initial guess with SLSQP
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    
    result = minimize(
        objective,
        initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-9, 'maxiter': 1000}
    )
    
    if result.success and result.fun < best_c5:
        best_c5 = result.fun
        best_h = result.x
    
    # Perturbation phase with L-BFGS-B for better convergence
    num_perturbations = 300
    phase_shift = 0.20
    phase_decay = 0.95
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective_perturb(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with L-BFGS-B
        result = minimize(
            objective_perturb,
            perturbed,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 300}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points

def compute_c5(h, dx):
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return np.max(corr) * dx
