import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Define objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    # Define constraints
    def constraint(h):
        return np.sum(h) - target_sum
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial optimization with SLSQP
    result = minimize(
        objective,
        initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint},
        options={'ftol': 1e-10, 'maxiter': 1000}
    )
    
    if result.success and result.fun < best_c5:
        best_c5 = result.fun
        best_h = result.x
    
    # Perturbation phase with structured variations
    num_perturbations = 200  # Reduced from 300 to save time
    phase_shift = 0.25
    phase_decay = 0.95
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        # Find segments
        segments = np.where(best_h > 0.5)[0]
        if len(segments) > 0:
            # Randomly shift one segment
            seg_idx = np.random.choice(len(segments))
            seg_start = segments[seg_idx]
            seg_end = segments[seg_idx] + 1
            while seg_end < n_points:
                if best_h[seg_end] > 0.5:
                    break
                seg_end += 1
            # Shift the segment
            shift = np.random.uniform(-noise_scale, noise_scale)
            new_start = max(0, int(seg_start + shift))
            new_end = min(n_points, int(seg_end + shift))
            # Adjust the segment
            perturbed = best_h.copy()
            perturbed[new_start:new_end] = 1.0
            # Adjust other segments to maintain sum
            sum_h = np.sum(perturbed)
            scale = target_sum / sum_h
            perturbed = perturbed * scale
        
        # Optimization with SLSQP
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,

        )
