        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective_perturbed(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr) * dx
            penalty = 1e4 * abs(np.sum(h) - target_sum)
            return max_corr + penalty
        
        # Run differential evolution on the perturbed guess
        result_perturbed = differential_evolution(
            objective_perturbed,
            bounds,
            strategy='best1bin',
            maxiter=50,
            popsize=10,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            polish=True
        )
        
        # Update best_h and best_c5
        if result_perturbed.fun < best_c5:
            best_h = result_perturbed.x
            best_c5 = result_perturbed.fun
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points

def compute_c5(h, dx):
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return np.max(corr) * dx
