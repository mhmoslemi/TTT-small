    seg1_end = n_points // 2
    seg3_start = 3 * n_points // 4
    seg3_end = n_points
    h_four_seg[seg1_start:seg1_end] = 1.0
    h_four_seg[seg3_start:seg3_end] = 1.0
    sum_h = np.sum(h_four_seg)
    scale = target_sum / sum_h
    initial_guesses.append(h_four_seg * scale)
    
    # Triangular wave
    h_tri = np.zeros(n_points)
    for i in range(n_points):
        h_tri[i] = 1.0 - abs((i - n_points/2) / (n_points/2) - 1.0)
    sum_h = np.sum(h_tri)
    scale = target_sum / sum_h
    initial_guesses.append(h_tri * scale)
    
    # Sine-like function
    h_sine = np.zeros(n_points)
    for i in range(n_points):
        h_sine[i] = 0.5 * (1 + np.sin(np.pi * (i - n_points/2) / (n_points/2)))
    sum_h = np.sum(h_sine)
    scale = target_sum / sum_h
    initial_guesses.append(h_sine * scale)
    
    # Random initial guess
    initial_guesses.append(np.random.uniform(0, 1, n_points))
    
    # Add perturbations of initial_h_values
    for perturb in [0.05, 0.1, 0.15]:
        perturbed = initial_guess.copy()
        noise = np.random.normal(0, perturb, n_points)
        perturbed += noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        initial_guesses.append(perturbed)
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with COBYLA
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='COBYLA',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 500, 'disp': False}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Perturbation phase with structured perturbations
    num_perturbations = 150  # Increased to explore more
    phase_shift = 0.15
    phase_decay = 0.98
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        # Apply structured perturbations: shift and scale
        perturbed = best_h.copy()
        shift = np.random.uniform(-0.2, 0.2, n_points)
        perturbed += noise_scale * shift
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with COBYLA
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='COBYLA',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 300, 'disp': False}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
