from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Initial optimization with initial_guess
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    
    result = minimize(
        objective,
        initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-9, 'maxiter': 1000}
    )
    
    if result.success and result.fun < best_c5:
        best_c5 = result.fun
        best_h = result.x
    
    # Perturbation phase with random noise
    num_perturbations = 300  # Reduced from 500 to save time
    noise_decay = 0.95
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = 0.1 * (noise_decay ** perturbation_idx)
        # Add random noise and clip
        perturbed = best_h.copy()

        perturbed += noise

        # Scale to maintain sum
        sum_h = np.sum(perturbed)
        scale = target_sum / sum_h
        perturbed = perturbed * scale
        
        # Objective function
        def objective_perturb(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        result_perturb = minimize(
            objective_perturb,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 400}
        )
        
        if result_perturb.success and result_perturb.fun < best_c5:
