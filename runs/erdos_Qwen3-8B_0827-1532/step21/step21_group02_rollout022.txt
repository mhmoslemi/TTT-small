from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    initial_guesses = []

    initial_guesses.append(h_step * scale)
    
    # Alternating 1 and 0
    h_alt = np.zeros(n_points)
    h_alt[::2] = 1.0
    sum_h = np.sum(h_alt)
    scale = target_sum / sum_h

    sum_h = np.sum(h_rand_step)
