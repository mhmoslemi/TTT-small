            penalty = 1e6 * (np.sum(h) - target_sum)**2
            return max_corr + penalty
        
        # Run differential evolution on perturbed guess
        result = differential_evolution(
            objective_perturbed,
            bounds,
            strategy='best1bin',
            popsize=15,
            mutation=(0.5, 1.0),
            recombination=0.8,
            tol=1e-8,
            maxiter=100,
            polish=True,
            seed=seed
        )
        
        # Check if result is successful
        if result.success:
            current_c5 = result.fun - 1e6 * (np.sum(result.x) - target_sum)**2
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
