    # Optimize initial guess with tighter tolerances
    result = minimize(
        objective,
        initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-10, 'maxiter': 1000}
    )
    
    if result.success and result.fun < best_c5:
        best_c5 = result.fun
        best_h = result.x
    
    # Generate additional initial guesses
    initial_guesses = [best_h.copy()]
    
    # Random binary sequence with exactly half 1s
    h_binary = np.zeros(n_points)
    h_binary[:n_points//2] = 1.0
    np.random.shuffle(h_binary)
    initial_guesses.append(h_binary)
    
    # Alternating 1 and 0
    h_alt = np.zeros(n_points)
    h_alt[::2] = 1.0
    sum_h = np.sum(h_alt)
    scale = target_sum / sum_h
    initial_guesses.append(h_alt * scale)
    
    # Central 50% step function
    h_center = np.zeros(n_points)
    center_start = n_points // 2 - n_points // 4
    center_end = n_points // 2 + n_points // 4
    h_center[center_start:center_end] = 1.0
    sum_h = np.sum(h_center)
    scale = target_sum / sum_h
    initial_guesses.append(h_center * scale)
    
    # Two-segment step function
    h_two_seg = np.zeros(n_points)
    h_two_seg[:n_points//4] = 1.0
    h_two_seg[n_points//2:n_points//2 + n_points//4] = 1.0
    sum_h = np.sum(h_two_seg)
    scale = target_sum / sum_h
    initial_guesses.append(h_two_seg * scale)
    
    # Three-segment step function
    h_three_seg = np.zeros(n_points)
    seg_lengths = [1/3, 1/3, 1/3]
    seg_positions = [0.3, 0.9, 1.5]
    for i in range(3):
        start = int(seg_positions[i] * n_points / 2)
        end = int((seg_positions[i] + seg_lengths[i]) * n_points / 2)
        h_three_seg[start:end] = 1.0
    sum_h = np.sum(h_three_seg)
    scale = target_sum / sum_h
    initial_guesses.append(h_three_seg * scale)
    
    # Four-segment step function
    h_four_seg = np.zeros(n_points)
    seg_positions = [0.2, 0.5, 0.8, 1.1]
    seg_lengths = [0.2, 0.2, 0.2, 0.2]
    for i in range(4):
        start = int(seg_positions[i] * n_points / 2)
        end = int((seg_positions[i] + seg_lengths[i]) * n_points / 2)
        h_four_seg[start:end] = 1.0
    sum_h = np.sum(h_four_seg)
    scale = target_sum / sum_h
    initial_guesses.append(h_four_seg * scale)
    
    # Random initial guess
    initial_guesses.append(np.random.uniform(0, 1, n_points))
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 1000}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Perturbation phase with structured variations
    num_perturbations = 200  # Reduced from 300 to save time
    phase_shift = 0.25
    phase_decay = 0.95
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        # Structured perturbation: add small random noise and scale
        perturbed = best_h.copy()
        noise = np.random.normal(0, noise_scale, n_points)
        perturbed += noise
        perturbed = np.clip(perturbed, 0, 1)
        sum_h = np.sum(perturbed)
        scale = target_sum / sum_h
        perturbed = perturbed * scale
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 400}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
