from scipy.optimize import differential_evolution, minimize

    # Use the provided initial guess
    initial_guess = initial_h_values

    dx = 2.0 / n_points

    best_c5 = float('inf')

    start_time = time.time()
    
    # Generate diverse initial guesses based on best_h
    num_perturbations = 150  # Reduced number for efficiency
    for perturbation_idx in range(num_perturbations):
        # Generate perturbations based on best_h
        if perturbation_idx < 50:
            # Random initial guess based on best_h with larger noise
            noise = np.random.uniform(-0.2, 0.2, n_points)
        else:
            # Smaller noise for later perturbations
            noise = np.random.uniform(-0.05, 0.05, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function to minimize C5
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Use differential evolution for global search
        result = differential_evolution(

        )

        if result.success and result.fun < best_c5:

            best_h = result.x
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
