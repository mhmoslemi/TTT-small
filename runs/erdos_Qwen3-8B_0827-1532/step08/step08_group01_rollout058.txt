    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate diverse initial guesses
    num_initial_guesses = 20  # More initial guesses for better exploration
    initial_guesses = []
    
    # Add initial guess
    initial_guesses.append(initial_guess.copy())
    
    # Add random binary patterns
    if n_points % 2 == 0:
        for _ in range(5):
            indices = np.random.choice(n_points, size=int(n_points/2), replace=False)
            binary_pattern = np.zeros(n_points)
            binary_pattern[indices] = 1
            initial_guesses.append(binary_pattern)
    
    # Add random perturbations of initial guess
    for _ in range(15):
        noise = np.random.uniform(-0.1, 0.1, n_points)
        perturbed = initial_guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        initial_guesses.append(perturbed)
    
    # Evaluate each initial guess and optimize
    for guess in initial_guesses:
        # Ensure the guess satisfies the sum constraint
        current_sum = np.sum(guess)
        scale = target_sum / current_sum
        guess_scaled = guess * scale
        guess_scaled = np.clip(guess_scaled, 0, 1)
        
        # Objective function to minimize C5
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization with tighter tolerances and more iterations
        result = minimize(
            objective,
            guess_scaled,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 1000}
        )
        
        # Check if this result is better
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
