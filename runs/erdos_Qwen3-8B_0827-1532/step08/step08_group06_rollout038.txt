    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Run differential evolution for each initial guess
    for guess in initial_guesses:
        # Run differential evolution
        result = differential_evolution(
            objective,
            bounds,
            popsize=15,
            maxiter=100,
            strategy='best1bin',
            tol=1e-10,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.success:
            current_c5 = result.fun - 1e6 * np.abs(np.sum(result.x) - n_points / 2)  # Subtract penalty
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # If no optimization was done, use the initial guess
    if best_h is None:
        result = differential_evolution(
            objective,
            bounds,
            popsize=15,
            maxiter=100,
            strategy='best1bin',
            tol=1e-10,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed
        )
        best_c5 = result.fun - 1e6 * np.abs(np.sum(result.x) - n_points / 2)
        best_h = result.x
    
    return best_h, best_c5, n_points
