    for _ in range(2):
        # Random symmetric perturbation
        noise_amp = np.random.uniform(0.001, 0.05)
        noise = noise_amp * np.random.rand(n_points)
        noise = (noise - np.min(noise)) / (np.max(noise) - np.min(noise))
        # Ensure symmetry
        noise_sym = np.zeros(n_points)
        for i in range(n_points):
            noise_sym[i] = noise[i]
        for i in range(n_points // 2):
            noise_sym[n_points - 1 - i] = noise_sym[i]
        # Scale to meet integral constraint
        scale_factor = (n_points / 2) / np.sum(noise_sym)
        noise_sym *= scale_factor
        initial_guesses.append(noise_sym)
    
    # Objective function
    def objective(h_half):
        h = np.zeros(n_points)
        for i in range(n_points // 2):
            h[i] = h_half[i]
            h[n_points - 1 - i] = h_half[i]
        if n_points % 2 == 1:
            h[n_points // 2] = h_half[-1]
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h_half):
        h = np.zeros(n_points)
        for i in range(n_points // 2):
            h[i] = h_half[i]
            h[n_points - 1 - i] = h_half[i]
        if n_points % 2 == 1:
            h[n_points // 2] = h_half[-1]
        return np.sum(h) - n_points / 2
    
    # Bounds for each h_half[i]
    bounds = [(0.0, 1.0) for _ in range(n_points // 2 + (1 if n_points % 2 else 0))]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Run optimizations for each initial guess
    for guess in initial_guesses:
        # Extract the first half (and middle if odd)
        h_half_initial = np.zeros(n_points // 2 + (1 if n_points % 2 else 0))
        for i in range(n_points // 2):
            h_half_initial[i] = guess[i]
        if n_points % 2:
            h_half_initial[-1] = guess[n_points // 2]
        # Run optimization
        result = minimize(
            objective,
            h_half_initial,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-9, 'maxiter': 2000}
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                # Reconstruct full h from h_half
                best_h_half = result.x
                best_h = np.zeros(n_points)
                for i in range(n_points // 2):
                    best_h[i] = best_h_half[i]
                    best_h[n_points - 1 - i] = best_h_half[i]
                if n_points % 2:
                    best_h[n_points // 2] = best_h_half[-1]
    
    # If no optimization was done, use the first initial guess
    if best_h is None:
        # Extract the first half (and middle if odd)
        h_half_initial = np.zeros(n_points // 2 + (1 if n_points % 2 else 0))
        for i in range(n_points // 2):
            h_half_initial[i] = initial_guesses[0][i]
        if n_points % 2:
            h_half_initial[-1] = initial_guesses[0][n_points // 2]
        # Run optimization
        result = minimize(
            objective,
            h_half_initial,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-9, 'maxiter': 2000}
        )
        best_c5 = result.fun
        # Reconstruct full h from h_half
        best_h_half = result.x
        best_h = np.zeros(n_points)
        for i in range(n_points // 2):
            best_h[i] = best_h_half[i]
            best_h[n_points - 1 - i] = best_h_half[i]
        if n_points % 2:
            best_h[n_points // 2] = best_h_half[-1]
    
    return best_h, best_c5, n_points
