        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # Fallback to initial guess if no optimization was successful
    if best_h is None:
        best_h = initial_h_values.copy()
        best_h = best_h / np.sum(best_h) * (n_points / 2)
        best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
