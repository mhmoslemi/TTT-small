import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate diverse initial guesses
    num_initial_guesses = 300  # Increased number of initial guesses
    initial_guesses = []
    
    # Add random initial guess
    random_h = np.random.uniform(0, 1, n_points)
    random_h = random_h * target_sum / np.sum(random_h)
    initial_guesses.append(random_h)
    
    # Add perturbations of initial guess
    for _ in range(num_initial_guesses - 1):
        if np.random.rand() < 0.5:
            # Random noise with varying magnitude
            noise = np.random.uniform(-0.2, 0.2, n_points)
        else:
            # Small noise for fine-tuning
            noise = np.random.uniform(-0.05, 0.05, n_points)
        perturbed = initial_guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        initial_guesses.append(perturbed)
    
    # Objective function to minimize C5
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Constraints: sum(h) == target_sum
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Use COBYLA for derivative-free optimization
        result = minimize(
            objective,
            guess,
            method='COBYLA',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'disp': False}
        )
        
        # Check if this result is better
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
