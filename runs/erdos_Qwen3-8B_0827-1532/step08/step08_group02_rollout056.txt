    initial_guesses = []
    # Generate a few step functions with different positions
    for shift in np.linspace(0, 1, 5):
        # Step function from [shift, shift + 1]
        h = np.zeros(n_points)
        start_idx = int(np.floor(shift * n_points))
        end_idx = int(np.ceil((shift + 1) * n_points))
        h[start_idx:end_idx] = 1.0
        # Adjust sum to meet the constraint
        scale_factor = (n_points / 2) / np.sum(h)
        h *= scale_factor
        initial_guesses.append(h)
    # Add some perturbations to initial guesses
    for h in initial_guesses:
        # Add small random noise
        noise = np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(h + noise, 0, 1)
        # Adjust sum to meet the constraint
        scale_factor = (n_points / 2) / np.sum(perturbed)
        perturbed *= scale_factor
        initial_guesses.append(perturbed)
    # Add some random initial guesses
    for _ in range(10):
        h = np.random.uniform(0, 1, n_points)
        # Adjust sum to meet the constraint
        scale_factor = (n_points / 2) / np.sum(h)
        h *= scale_factor
        initial_guesses.append(h)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Run optimizations for each initial guess with tighter tolerances
    for guess in initial_guesses:
        # Run optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-10, 'maxiter': 3000}  # Tighter tolerances and more iterations
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # If no optimization was done, use the initial guess
    if best_h is None:
        result = minimize(
            objective,
            initial_guesses[0],
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-10, 'maxiter': 3000}
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points
