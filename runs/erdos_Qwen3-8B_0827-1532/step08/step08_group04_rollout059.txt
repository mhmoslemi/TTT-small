    # If no optimization was done, use the initial guess
    if best_h is None:
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=15,
            tol=1e-10,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed,
            maxiter=100,
            polish=True,
            disp=False
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points
