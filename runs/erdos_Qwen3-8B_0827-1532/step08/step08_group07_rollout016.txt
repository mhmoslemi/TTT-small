from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    dx = 2.0 / n_points
    
    # Generate initial guess based on provided initial_h_values
    initial_guess = np.array(initial_h_values)

    initial_guesses = []

    sine_wave = 0.5 + 0.5 * np.sin(np.pi * np.linspace(0, 2, n_points))
    initial_guesses.append(sine_wave)
    # Triangular wave
    triangular_wave = np.zeros(n_points)
    for i in range(n_points):
        x = np.linspace(0, 2, n_points)[i]
        if x < 1.0:
            triangular_wave[i] = x / 1.0
        else:
            triangular_wave[i] = (2.0 - x) / 1.0
    initial_guesses.append(triangular_wave)
    # Square wave
    square_wave = np.zeros(n_points)
    for i in range(n_points):
        if np.linspace(0, 2, n_points)[i] < 1.0:
            square_wave[i] = 1.0
        else:
            square_wave[i] = 0.0
    initial_guesses.append(square_wave)
    # Random binary function
    binary_h = np.zeros(n_points)
    indices = np.random.choice(n_points, size=int(n_points/2), replace=False)
    binary_h[indices] = 1.0
    initial_guesses.append(binary_h)
    # Sawtooth wave
    sawtooth_wave = np.linspace(0, 1, n_points)
    initial_guesses.append(sawtooth_wave)
    # Inverse sawtooth wave
    inverse_sawtooth_wave = 1.0 - np.linspace(0, 1, n_points)
    initial_guesses.append(inverse_sawtooth_wave)
    
    # Add perturbed and random guesses based on initial guess
    for _ in range(num_initial_guesses):
        # Perturb with sine wave
        noise_amp = np.random.uniform(0.001, 0.05)

        perturbed = initial_guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        scale_factor = (n_points / 2) / np.sum(perturbed)
        perturbed *= scale_factor
        initial_guesses.append(perturbed)
        # Add a random guess
        random_h = np.random.uniform(0, 1, n_points)
        random_h = random_h / np.sum(random_h) * (n_points / 2)
        initial_guesses.append(random_h)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    for guess in initial_guesses:
        # Run optimization
        result = minimize(
