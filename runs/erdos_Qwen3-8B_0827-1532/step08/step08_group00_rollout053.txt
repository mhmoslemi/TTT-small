from scipy.optimize import minimize
import time
from concurrent.futures import ThreadPoolExecutor

def run(seed=42, budget_s=100, **kwargs):
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Generate structured initial guesses
    structured_h = np.zeros(n_points)
    for i in range(n_points):
        if i % 2 == 0:
            structured_h[i] = 1.0
        else:
            structured_h[i] = 0.0

    # Additional structured initial guess with 3 intervals
    structured_h_3 = np.zeros(n_points)
    for i in range(n_points):
        if i % 3 == 0 or i % 3 == 1:
            structured_h_3[i] = 1.0
        else:
            structured_h_3[i] = 0.0

    initial_guesses = [initial_guess, structured_h, structured_h_3]

    best_c5 = float('inf')
    best_h = initial_guess.copy()

    start_time = time.time()

    def optimize_perturbation(perturbed):
        # Objective function to minimize C5
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx

        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]

        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

        # Perform optimization with tighter tolerances and more iterations
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 1000}
        )
        return result

    with ThreadPoolExecutor(max_workers=10) as executor:
        futures = []
        for guess in initial_guesses:
            # Generate multiple perturbations from each initial guess
            for _ in range(50):
                # Random noise perturbation
                noise = np.random.uniform(-0.1, 0.1, n_points)
                perturbed = guess + noise
                perturbed = np.clip(perturbed, 0, 1)
                current_sum = np.sum(perturbed)
                scale = target_sum / current_sum
                perturbed = perturbed * scale
                futures.append(executor.submit(optimize_perturbation, perturbed))

        for future in futures:
            result = future.result()
            if result.success and result.fun < best_c5:
                best_c5 = result.fun
                best_h = result.x
