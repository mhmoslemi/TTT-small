import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Predefined initial guess
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate diverse initial guesses
    initial_guesses = []
    # Add the provided initial guess
    initial_guesses.append(initial_guess)
    
    # Generate random initial guess
    np.random.seed(seed)
    random_guess = np.random.uniform(0, 1, n_points)
    current_sum = np.sum(random_guess)
    scale = target_sum / current_sum
    random_guess = np.clip(random_guess * scale, 0, 1)
    initial_guesses.append(random_guess)
    
    # Generate alternating pattern
    alternating = np.zeros(n_points)
    for i in range(n_points):
        alternating[i] = 1.0 if i % 2 == 0 else 0.0
    current_sum = np.sum(alternating)
    scale = target_sum / current_sum
    alternating = np.clip(alternating * scale, 0, 1)
    initial_guesses.append(alternating)
    
    # Generate blocky pattern (alternating blocks of 0 and 1)
    block_size = 5
    blocky = np.zeros(n_points)
    for i in range(n_points):
        blocky[i] = 1.0 if (i // block_size) % 2 == 0 else 0.0
    current_sum = np.sum(blocky)
    scale = target_sum / current_sum
    blocky = np.clip(blocky * scale, 0, 1)
    initial_guesses.append(blocky)
    
    # Optimization parameters
    max_iterations = 1000
    tolerance = 1e-8
    
    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds and constraints
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Optimization with COBYLA
        result = minimize(
            objective,
            guess,
            method='COBYLA',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': max_iterations, 'ftol': tolerance}
        )
        
        # Update best solution
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check elapsed time

        if elapsed >= budget_s:
            break
