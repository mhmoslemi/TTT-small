import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate multiple diverse initial guesses
    num_perturbations = 150  # Reduced number of perturbations for efficiency
    for perturbation_idx in range(num_perturbations):
        # Generate different initial guesses
        if perturbation_idx == 0:
            # Use initial_h_values
            perturbed = initial_guess.copy()
        elif perturbation_idx == 1:
            # Uniform distribution
            perturbed = np.full(n_points, 0.5)
        else:
            # Random perturbation
            noise = np.random.uniform(-0.15, 0.15, n_points)
            perturbed = initial_guess + noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
        
        # Objective function with penalty for sum constraint
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            penalty = 1e3 * np.abs(np.sum(h) - target_sum)
            return max_corr * dx + penalty
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Run differential evolution for global search
        result_de = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=15,
            maxiter=50,
            tol=1e-10,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed + perturbation_idx
        )
        
        # Refine with SLSQP for local optimization
        def objective_slsqp(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result_slsqp = minimize(
            objective_slsqp,
            result_de.x,
            method='SLSQP',
            bounds=bounds_slsqp,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 200}
        )
        
        # Check if this result is better
        if result_slsqp.success and result_slsqp.fun < best_c5:
            best_c5 = result_slsqp.fun
            best_h = result_slsqp.x
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
