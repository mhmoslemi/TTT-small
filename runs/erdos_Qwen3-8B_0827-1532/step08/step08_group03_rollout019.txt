            strategy='best1bin',  # Strategy for differential evolution
            tol=1e-6,  # Tolerance for convergence
            maxiter=200,  # Max iterations
            polish=True,  # Refine with local optimization
            seed=seed
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.success:
            # Subtract penalty to get actual overlap integral
            current_c5 = result.fun - 1e6 * abs(np.sum(result.x) - n_points / 2)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # If no optimization was done, use the initial guess
    if best_h is None:
        result = differential_evolution(
            objective,
            bounds,
            popsize=15,
            mutation=(0.5, 1.0),
            recombination=0.8,
            strategy='best1bin',
            tol=1e-6,
            maxiter=200,
            polish=True,
            seed=seed
        )
        best_c5 = result.fun - 1e6 * abs(np.sum(result.x) - n_points / 2)
        best_h = result.x
    
    return best_h, best_c5, n_points
