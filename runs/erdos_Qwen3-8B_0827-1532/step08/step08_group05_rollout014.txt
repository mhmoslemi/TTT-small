import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Keep it at 100 for faster evaluation
    dx = 2.0 / n_points
    
    # Generate initial guesses
    initial_guesses = []
    
    # Add initial_h_values
    initial_guesses.append(initial_h_values)
    
    # Add binary guess
    binary_guess = np.zeros(n_points)
    binary_guess[::2] = 1.0
    sum_binary = np.sum(binary_guess)
    scale_factor = (n_points / 2) / sum_binary
    binary_guess *= scale_factor
    initial_guesses.append(binary_guess)
    
    # Add sinusoidal guess
    x = np.linspace(0, 2, n_points)
    sinusoidal_guess = 0.5 + 0.5 * np.sin(2 * np.pi * x / 2)
    sum_sinusoidal = np.sum(sinusoidal_guess)
    scale_factor = (n_points / 2) / sum_sinusoidal
    sinusoidal_guess *= scale_factor
    initial_guesses.append(sinusoidal_guess)
    
    # Add some random initial guesses
    num_random_guesses = 5
    for _ in range(num_random_guesses):
        random_guess = np.random.uniform(0, 1, n_points)
        sum_random = np.sum(random_guess)
        scale_factor = (n_points / 2) / sum_random
        random_guess *= scale_factor
        initial_guesses.append(random_guess)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Run optimizations for each initial guess
    for guess in initial_guesses:
        # Run optimization
        result = minimize(
            objective,
            guess,
            method='trust-constr',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 1000, 'ftol': 1e-8, 'gtol': 1e-8}
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # If no optimization was done, use the initial guess
    if best_h is None:
        result = minimize(
            objective,
            initial_guesses[0],
            method='trust-constr',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 1000, 'ftol': 1e-8, 'gtol': 1e-8}
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points
