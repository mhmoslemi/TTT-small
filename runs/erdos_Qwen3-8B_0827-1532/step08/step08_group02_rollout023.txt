    x_initial = np.linspace(0, 2, len(initial_h_values))

    initial_guess = initial_guess / np.sum(initial_guess) * (n_points / 2)
    
    # Generate diverse initial guesses with different noise types

            noise_amp = np.random.uniform(0.001, 0.05)
            noise_freq = np.random.uniform(1, 10)
            noise = noise_amp * np.sin(2 * np.pi * noise_freq * x_new)
        elif noise_type == 'random_steps':
            num_steps = np.random.randint(1, 10)
            indices = np.random.choice(n_points, num_steps, replace=False)
            noise = np.zeros(n_points)
            noise[indices] = np.random.uniform(-0.1, 0.1, size=num_steps)

        perturbed = initial_guess + noise

    def objective(h):

    # Constraint function
    def constraint(h):
