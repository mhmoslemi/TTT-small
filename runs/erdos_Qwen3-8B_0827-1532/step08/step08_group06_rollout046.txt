from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Balanced between speed and precision
    dx = 2.0 / n_points
    
    # Generate initial guesses
    initial_guesses = []
    
    # Use initial_h_values as the primary guess
    initial_guesses.append(initial_h_values)
    
    # Generate structured patterns
    x_new = np.linspace(0, 2, n_points)
    
    # Square wave
    square_wave = np.zeros(n_points)
    square_wave[x_new < 1.0] = 1.0
    initial_guesses.append(square_wave)
    
    # Triangular wave
    triangular_wave = np.zeros(n_points)
    triangular_wave = np.clip(x_new / 1.0, 0, 1)
    triangular_wave[x_new > 1.0] = np.clip((2.0 - x_new) / 1.0, 0, 1)
    initial_guesses.append(triangular_wave)
    
    # Sine wave
    sine_wave = 0.5 + 0.5 * np.sin(np.pi * x_new)
    initial_guesses.append(sine_wave)
    
    # Sawtooth wave
    sawtooth_wave = np.zeros(n_points)
    sawtooth_wave = np.clip(x_new / 1.0, 0, 1)
    sawtooth_wave[x_new > 1.0] = np.clip((x_new - 1.0) / 1.0, 0, 1)
    initial_guesses.append(sawtooth_wave)
    
    # Add perturbations of initial_h_values
    for _ in range(3):
        # Sinusoidal perturbation
        noise_amp = np.random.uniform(0.005, 0.05)
        noise_freq = np.random.uniform(0.5, 10)
        noise = noise_amp * np.sin(2 * np.pi * noise_freq * x_new)
        perturbed = initial_h_values + noise
        perturbed = np.clip(perturbed, 0, 1)
        scale_factor = (n_points / 2) / np.sum(perturbed)
        perturbed *= scale_factor
        initial_guesses.append(perturbed)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Run optimizations for each initial guess with tighter tolerances
    for guess in initial_guesses:
        # Run optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-10, 'maxiter': 2000}
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.success:
            current_c5 = result.fun

                best_h = result.x
