                step = np.random.uniform(-0.01, 0.01)
                current += step
                noise[j] = current
        elif noise_type == 'sinusoidal':

        perturbed = initial_guess + noise

        perturbed = np.clip(perturbed, 0, 1)

    def objective(h):

    def constraint(h):
