from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Keeping it at 100 for speed
    dx = 2.0 / n_points
    
    # Generate initial guesses
    initial_guesses = []
    
    # Add initial_h_values as an initial guess
    # Assuming initial_h_values is a global variable with n=100 samples
    initial_guesses.append(initial_h_values)
    
    # Add a binary guess
    binary_guess = np.zeros(n_points)
    binary_guess[::2] = 1.0  # Alternate 1 and 0
    binary_guess = binary_guess / np.sum(binary_guess) * (n_points / 2)
    initial_guesses.append(binary_guess)
    
    # Add a sine-like guess
    sine_guess = 0.5 + 0.5 * np.sin(2 * np.pi * np.linspace(0, 2, n_points))
    sine_guess = sine_guess / np.sum(sine_guess) * (n_points / 2)
    initial_guesses.append(sine_guess)
    
    # Add a random guess
    random_guess = np.random.uniform(0, 1, n_points)
    random_guess = random_guess / np.sum(random_guess) * (n_points / 2)
    initial_guesses.append(random_guess)
    
    # Add a few more structured guesses
    # For example, a triangular wave
    triangular_guess = np.linspace(0, 1, n_points)
    triangular_guess = np.concatenate([triangular_guess, np.flip(triangular_guess[1:-1])])
    triangular_guess = triangular_guess / np.sum(triangular_guess) * (n_points / 2)
    initial_guesses.append(triangular_guess)
    
    # Now, run optimization for each initial guess using differential evolution
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    
    for guess in initial_guesses:
        # Define the objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Constraint function
        def constraint(h):
            return np.sum(h) - n_points / 2
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Use differential evolution
        result = minimize(
            objective,
            guess,
            method='L-BFGS-B',  # Or 'DE' if available
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-10, 'maxiter': 1000}
        )
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # If no optimization was done, use the initial guess

        one_minus_h = 1.0 - best_h
        corr = np.correlate(best_h, one_minus_h, mode='full')
        best_c5 = np.max(corr) * dx
    
    return best_h, best_c5, n_points
