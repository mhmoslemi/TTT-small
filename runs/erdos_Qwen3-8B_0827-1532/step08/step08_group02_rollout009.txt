import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Optimal balance between speed and precision
    dx = 2.0 / n_points
    
    # Generate initial guesses based on structured patterns
    x_new = np.linspace(0, 2, n_points)
    
    # Initial guess from provided initial_h_values
    initial_guesses = [initial_h_values]
    
    # Sine-based initial guess
    h_sine = 0.5 + 0.5 * np.sin(np.pi * x_new)
    initial_guesses.append(h_sine)
    
    # Symmetric interval-based initial guess
    h_symmetric = np.zeros(n_points)
    for i in range(n_points):
        if x_new[i] <= 0.5 or x_new[i] >= 1.5:
            h_symmetric[i] = 1.0
        else:
            h_symmetric[i] = 0.0
    sum_h = np.sum(h_symmetric) * dx
    scale_factor = (n_points / 2) / sum_h
    h_symmetric *= scale_factor
    initial_guesses.append(h_symmetric)
    
    # Add a few random noise-based guesses
    for _ in range(5):
        noise = np.random.uniform(-0.1, 0.1, size=n_points)
        perturbed = initial_guesses[0] + noise
        perturbed = np.clip(perturbed, 0, 1)
        sum_perturbed = np.sum(perturbed) * dx
        scale_factor = (n_points / 2) / sum_perturbed
        perturbed *= scale_factor
        initial_guesses.append(perturbed)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Run optimizations for each initial guess with tighter tolerances
    for guess in initial_guesses:
        # Run optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-10, 'maxiter': 3000}  # Tighter tolerances and more iterations
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # If no optimization was done, use the initial guess
    if best_h is None:
        result = minimize(
            objective,
            initial_guesses[0],
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-10, 'maxiter': 3000}
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points
