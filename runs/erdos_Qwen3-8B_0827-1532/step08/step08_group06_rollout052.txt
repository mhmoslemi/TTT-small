from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    dx = 2.0 / n_points

    x_initial = np.linspace(0, 2, len(initial_h_values))

    # Adjust sum to meet the constraint
    initial_guess = initial_guess / np.sum(initial_guess) * (n_points / 2)
    
    # Generate diverse initial guesses
    num_initial_guesses = 100  # Increased for better exploration
    initial_guesses = []
    
    # Add structured patterns
    # Square wave
    square_wave = np.zeros(n_points)
    for i in range(n_points):
        if x_new[i] < 1.0:
            square_wave[i] = 1.0
        else:
            square_wave[i] = 0.0
    initial_guesses.append(square_wave)
    # Triangular wave
    triangular_wave = np.zeros(n_points)
    for i in range(n_points):
        x = x_new[i]
        if x < 1.0:
            triangular_wave[i] = x / 1.0
        else:
            triangular_wave[i] = (2.0 - x) / 1.0
    initial_guesses.append(triangular_wave)
    # Sine wave
    sine_wave = 0.5 + 0.5 * np.sin(np.pi * x_new)
    initial_guesses.append(sine_wave)
    # Sawtooth wave
    sawtooth_wave = np.zeros(n_points)
    for i in range(n_points):
        x = x_new[i]
        if x < 1.0:
            sawtooth_wave[i] = x / 1.0
        else:
            sawtooth_wave[i] = (x - 1.0) / 1.0
    initial_guesses.append(sawtooth_wave)
    # Pulse train
    pulse_train = np.zeros(n_points)
    for i in range(n_points):
        x = x_new[i]
        if 0.0 <= x < 0.5 or 1.0 <= x < 1.5:
            pulse_train[i] = 1.0
        else:
            pulse_train[i] = 0.0
    initial_guesses.append(pulse_train)
    # Add initial_h_values as a strong starting point
    initial_guesses.append(initial_guess)
    
    # Add perturbed and random guesses
    for _ in range(num_initial_guesses - len(initial_guesses)):
        # Perturb with sine wave
        noise_amp = np.random.uniform(0.001, 0.05)

        noise = noise_amp * np.sin(2 * np.pi * noise_freq * x_new)
        perturbed = initial_guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        scale_factor = (n_points / 2) / np.sum(perturbed)
        perturbed *= scale_factor
        initial_guesses.append(perturbed)
        # Add a random guess
        random_h = np.random.uniform(0, 1, n_points)
        random_h = random_h / np.sum(random_h) * (n_points / 2)
        initial_guesses.append(random_h)
    
    # Objective function
    def objective(h):

        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
