from scipy.optimize import minimize

    initial_guesses = [initial_guess, structured_h, random_h]
    
    best_c5 = float('inf')

            perturbed = best_h + random_noise

        perturbed = best_h + noise

        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function to minimize C5
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
