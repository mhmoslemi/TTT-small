    dx = 2.0 / n_points
    
    # Generate diverse initial guesses
    def generate_initial_guesses():
        guesses = []

        binary_guess = np.zeros(n_points)

        guesses.append(binary_guess)

        guesses.append(random_binary)

        sine_wave = 0.5 * (1 + np.sin(2 * np.pi * x_new))

        sine_wave = sine_wave / np.sum(sine_wave) * (n_points / 2)
        guesses.append(sine_wave)
        
        # Perturb each base guess
        for guess in guesses:
            noise_amp = np.random.uniform(0.001, 0.05)
            noise_freq = np.random.uniform(1, 5)
            noise = noise_amp * np.sin(2 * np.pi * noise_freq * x_new)
            random_noise = np.random.normal(0, 0.01, n_points)
            perturbed = guess + noise + random_noise
            perturbed = np.clip(perturbed, 0, 1)
            perturbed = perturbed / np.sum(perturbed) * (n_points / 2)
            guesses.append(perturbed)
        
        # Trim to 50 guesses to stay within time budget
        return guesses[:50]
    
    initial_guesses = generate_initial_guesses()
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Run optimizations for each initial guess
    for guess in initial_guesses:

        result = minimize(
