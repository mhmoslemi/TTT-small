from scipy.optimize import minimize
import time

    initial_guesses.append(base_guess)
    
    # Binary pattern
    binary_guess = np.zeros(n_points)
    binary_guess[::2] = 1.0  # Alternate 1 and 0
    binary_guess = binary_guess / np.sum(binary_guess) * (n_points / 2)
    initial_guesses.append(binary_guess)
    
    # Sinusoidal pattern
    sin_guess = 0.5 + 0.5 * np.sin(2 * np.pi * np.linspace(0, 1, n_points))
    sin_guess = sin_guess / np.sum(sin_guess) * (n_points / 2)
    initial_guesses.append(sin_guess)
    
    # Random initial guesses with structured noise
    for _ in range(15):  # More random guesses
        noise_amp = np.random.uniform(0.001, 0.05)
        noise_freq = np.random.uniform(1, 5)
        noise = noise_amp * np.sin(2 * np.pi * noise_freq * np.linspace(0, 1, n_points))
        random_noise = np.random.normal(0, 0.01, n_points)
        perturbed = base_guess + noise + random_noise
        perturbed = np.clip(perturbed, 0, 1)
        scale_factor = (n_points / 2) / np.sum(perturbed)

    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]

    # Run optimizations for each initial guess with tighter tolerances
    for guess in initial_guesses:
        # Run optimization
        result = minimize(

            constraints={'type': 'eq', 'fun': constraint},
