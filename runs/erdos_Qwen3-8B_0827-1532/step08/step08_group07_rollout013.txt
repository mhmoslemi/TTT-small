        strategy='best1bin',
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1.0),
        recombination=0.8,
        maxiter=200,
        polish=True,
        disp=False
    )
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        # If time is up, return the best found so far
        best_h = result.x
        best_c5 = result.fun
    else:
        # Optionally refine with SLSQP if time allows
        # For simplicity, return the result
        best_h = result.x
        best_c5 = result.fun
    
    return best_h, best_c5, n_points
