import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 200  # Increased to 200 for better resolution
    dx = 2.0 / n_points
    
    # Generate initial guesses
    initial_guesses = []
    
    # 1. Interpolate initial_h_values to 200 points
    x_initial = np.linspace(0, 2, len(initial_h_values))
    x_new = np.linspace(0, 2, n_points)
    initial_guess = np.interp(x_new, x_initial, initial_h_values)
    # Adjust sum to meet the constraint
    initial_guess = initial_guess / np.sum(initial_guess) * (n_points / 2)
    initial_guesses.append(initial_guess)
    
    # 2. Binary pattern: alternate 1 and 0
    binary_guess = np.zeros(n_points)
    binary_guess[::2] = 1.0  # Alternate 1 and 0
    binary_guess = binary_guess / np.sum(binary_guess) * (n_points / 2)
    initial_guesses.append(binary_guess)
    
    # 3. Random patterns
    for _ in range(5):  # Generate 5 random initial guesses
        random_guess = np.random.uniform(0, 1, n_points)
        random_guess = random_guess / np.sum(random_guess) * (n_points / 2)
        initial_guesses.append(random_guess)
    
    # 4. Perturb existing guesses with noise
    for guess in initial_guesses[:3]:  # Perturb first 3 guesses
        # Add sinusoidal noise
        noise_amp = np.random.uniform(0.001, 0.05)
        noise_freq = np.random.uniform(1, 5)
        noise = noise_amp * np.sin(2 * np.pi * noise_freq * np.linspace(0, 2, n_points))
        # Add random noise
        random_noise = np.random.normal(0, 0.01, n_points)
        perturbed = guess + noise + random_noise
        # Clip to [0, 1]
        perturbed = np.clip(perturbed, 0, 1)
        # Adjust sum to meet the constraint
        scale_factor = (n_points / 2) / np.sum(perturbed)
        perturbed *= scale_factor
        initial_guesses.append(perturbed)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Run optimizations for each initial guess with tighter tolerances
    for guess in initial_guesses:
        # Run optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-10, 'maxiter': 3000}  # Increased maxiter and tighter tolerance
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # If no optimization was done, use the initial guess
    if best_h is None:
        result = minimize(
            objective,
            initial_guesses[0],
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-10, 'maxiter': 3000}
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points
