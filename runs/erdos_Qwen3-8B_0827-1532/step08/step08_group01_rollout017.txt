import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Assume initial_h_values is given for n_points=100
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_h_values.copy()

    # Generate diverse initial guesses
    initial_guesses = []

    initial_guesses.append(initial_h_values.copy())

    for _ in range(10):
        random_guess = np.random.uniform(0, 1, n_points)
        random_guess = random_guess * (target_sum / np.sum(random_guess))
        initial_guesses.append(random_guess)
    # Add structured guesses
    structured_guess = np.zeros(n_points)
    structured_guess[:n_points//2] = 1.0
    initial_guesses.append(structured_guess)
    # Add alternating pattern
    alt_guess = np.zeros(n_points)
    for i in range(n_points):
        alt_guess[i] = 1.0 if i % 2 == 0 else 0.0
    alt_guess = alt_guess * (target_sum / np.sum(alt_guess))
    initial_guesses.append(alt_guess)
    
    # Optimize each initial guess
    for guess in initial_guesses:
        guess = np.clip(guess, 0, 1)
        guess = guess * (target_sum / np.sum(guess))

        def objective(h):

        # Use COBYLA for better constraint handling
        result = minimize(
