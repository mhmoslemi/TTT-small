    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Generate initial population including the initial guess
    initial_population = [initial_guess]
    for _ in range(14):  # 15 total, including initial guess
        initial_population.append(np.random.uniform(0, 1, n_points))
    initial_population = np.array(initial_population)
    
    start_time = time.time()
    
    # Run differential evolution with time constraint
    result = differential_evolution(
        objective,
        bounds,
        popsize=15,
        mutation=(0.5, 1.0),
        recombination=0.9,
        strategy='best1bin',
        tol=1e-10,
        maxiter=500,
        seed=seed,
        initial_population=initial_population
    )
    
    # Extract the best solution
    best_h = result.x
    best_c5 = result.fun
    
    # Check elapsed time
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        # If time is up, return the best found so far
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
