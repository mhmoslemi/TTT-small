from scipy.optimize import minimize
import time

    x_initial = np.linspace(0, 2, len(initial_h_values))

    initial_guess = initial_guess / np.sum(initial_guess) * (n_points / 2)

        noise = noise_amp * np.sin(2 * np.pi * noise_freq * x_new)
        # Add random noise
        random_noise = np.random.normal(0, 0.02, n_points)
        perturbed = initial_guess + noise + random_noise
        # Clip to [0, 1]
        perturbed = np.clip(perturbed, 0, 1)
        # Adjust sum to meet the constraint
        scale_factor = (n_points / 2) / np.sum(perturbed)
        perturbed *= scale_factor
        initial_guesses.append(perturbed)
    
    # Generate additional structured initial guesses

    binary_guess = np.zeros(n_points)
    binary_guess[::2] = 1.0  # Alternate 1 and 0

    random_step_guess = np.random.uniform(0, 1, n_points)

    # Objective function
    def objective(h):

        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    for guess in initial_guesses:
        # Run optimization
        result = minimize(

            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-10, 'maxiter': 5000}  # Increased maxiter and tighter tolerance
        )
