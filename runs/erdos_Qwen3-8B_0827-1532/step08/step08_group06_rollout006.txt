            objective,
            initial_guesses[0],
            method='COBYLA',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 500, 'disp': False}
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points
