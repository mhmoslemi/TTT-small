import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Balanced between speed and precision
    dx = 2.0 / n_points
    
    # Generate initial guesses
    initial_guesses = []
    initial_guesses.append(initial_h_values)  # Use the provided initial guess
    
    # Add structured patterns
    x_new = np.linspace(0, 2, n_points)
    # Square wave
    square_wave = np.zeros(n_points)
    square_wave[:n_points//2] = 1.0
    initial_guesses.append(square_wave)
    # Triangular wave
    triangular_wave = np.zeros(n_points)
    for i in range(n_points):
        x = x_new[i]
        if x < 1.0:
            triangular_wave[i] = x / 1.0
        else:
            triangular_wave[i] = (2.0 - x) / 1.0
    initial_guesses.append(triangular_wave)
    # Sine wave
    sine_wave = 0.5 + 0.5 * np.sin(np.pi * x_new)
    initial_guesses.append(sine_wave)
    # Sawtooth wave
    sawtooth_wave = np.zeros(n_points)
    for i in range(n_points):
        x = x_new[i]
        if x < 1.0:
            sawtooth_wave[i] = x / 1.0
        else:
            sawtooth_wave[i] = (x - 1.0) / 1.0
    initial_guesses.append(sawtooth_wave)
    
    # Add perturbed and random guesses
    num_initial_guesses = 60  # Increased to 60
    for _ in range(num_initial_guesses - len(initial_guesses)):
        # Perturb with sine wave
        noise_amp = np.random.uniform(0.001, 0.05)
        noise_freq = np.random.uniform(0.5, 10)
        noise = noise_amp * np.sin(2 * np.pi * noise_freq * x_new)
        perturbed = initial_h_values + noise
        perturbed = np.clip(perturbed, 0, 1)
        scale_factor = (n_points / 2) / np.sum(perturbed)
        perturbed *= scale_factor
        initial_guesses.append(perturbed)
        # Add a random guess
        random_h = np.random.uniform(0, 1, n_points)
        random_h = random_h / np.sum(random_h) * (n_points / 2)
        initial_guesses.append(random_h)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Run optimizations for each initial guess with tighter tolerances
    for guess in initial_guesses:
        # Run optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-9, 'maxiter': 1500}
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # If no optimization was done, use the initial guess

            objective,
            initial_h_values,
            method='SLSQP',

        )

        best_h = result.x
    
    return best_h, best_c5, n_points
