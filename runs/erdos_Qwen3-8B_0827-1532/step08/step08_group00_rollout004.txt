import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Use the provided initial guess and generate a structured initial guess
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate multiple initial guesses
    initial_guesses = [initial_guess]
    
    # Add structured initial guess (alternating 1s and 0s)
    structured_h = np.zeros(n_points)
    for i in range(n_points):
        if i % 2 == 0:
            structured_h[i] = 1.0
        else:
            structured_h[i] = 0.0
    initial_guesses.append(structured_h)
    
    # Add random initial guesses
    for _ in range(5):
        random_initial = np.random.rand(n_points)
        random_initial = random_initial / np.sum(random_initial) * target_sum
        initial_guesses.append(random_initial)
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Objective function to minimize C5
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization with tighter tolerances and more iterations
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 1000}
        )
        
        # Check if this result is better
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
