    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Run optimizations for each initial guess
    for guess in initial_guesses:
        # Run optimization with tighter tolerances and more iterations
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-10, 'maxiter': 3000}
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # Fallback to initial guess if no optimization succeeded
    if best_h is None:
        result = minimize(
            objective,
            initial_guesses[0],
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-10, 'maxiter': 3000}
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points
