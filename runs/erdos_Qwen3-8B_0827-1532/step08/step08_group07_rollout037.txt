            triangular_wave[i] = (2.0 - x) / 1.0

            x = np.linspace(0, 2, n_points)[i]

                square_wave[i] = 1.0

                square_wave[i] = 0.0

    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2

        result = minimize(
