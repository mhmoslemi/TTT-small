import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = 200  # Increased for better precision
    dx = 2.0 / n_points

    x_initial = np.linspace(0, 2, len(initial_h_values))

    # Adjust sum to meet the constraint
    initial_guess = initial_guess / np.sum(initial_guess) * (n_points / 2)
    
    # Generate diverse initial guesses
    num_initial_guesses = 40  # Increased for better exploration
    initial_guesses = []
    
    # Add structured patterns
    # Binary pattern with alternating 1 and 0
    binary_pattern = np.zeros(n_points)
    binary_pattern[::2] = 1.0
    initial_guesses.append(binary_pattern)
    # Binary pattern with spaced 1s
    spaced_binary = np.zeros(n_points)
    for i in range(n_points):
        if i % 4 == 0:
            spaced_binary[i] = 1.0
    initial_guesses.append(spaced_binary)
    # Add initial_h_values as a strong starting point
    initial_guesses.append(initial_guess)
    
    # Add perturbed and random guesses
    for _ in range(num_initial_guesses - len(initial_guesses)):
        # Perturb with sine wave
        noise_amp = np.random.uniform(0.001, 0.05)
        noise_freq = np.random.uniform(0.5, 10)
        noise = noise_amp * np.sin(2 * np.pi * noise_freq * x_new)
        perturbed = initial_guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        scale_factor = (n_points / 2) / np.sum(perturbed)
        perturbed *= scale_factor
        initial_guesses.append(perturbed)
        # Add a random guess
        random_h = np.random.uniform(0, 1, n_points)
        random_h = random_h / np.sum(random_h) * (n_points / 2)
        initial_guesses.append(random_h)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Run optimizations for each initial guess with tighter tolerances
    for guess in initial_guesses:
        # Run optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
