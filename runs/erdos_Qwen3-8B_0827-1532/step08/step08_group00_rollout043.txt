        x = i * dx
        if 0 <= x < 0.5 or 1.0 <= x < 1.5:
            initial_guess[i] = 1.0
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Objective function to minimize C5
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    
    # Use COBYLA solver for better constraint handling
    result = minimize(
        objective,
        initial_guess,
        method='COBYLA',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 1000}
    )
    
    if result.success and result.fun < best_c5:
        best_c5 = result.fun
        best_h = result.x
    
    # Check elapsed time
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return best_h, best_c5, n_points
    
    # Additional perturbation with a different initial guess
    # Perturb the best_h with small noise
    perturbed = best_h + np.random.uniform(-0.05, 0.05, n_points)
    perturbed = np.clip(perturbed, 0, 1)
    current_sum = np.sum(perturbed)
    scale = target_sum / current_sum
    perturbed = perturbed * scale
    
    result = minimize(
        objective,
        perturbed,
        method='COBYLA',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 500}
    )
    
    if result.success and result.fun < best_c5:
        best_c5 = result.fun
        best_h = result.x
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
