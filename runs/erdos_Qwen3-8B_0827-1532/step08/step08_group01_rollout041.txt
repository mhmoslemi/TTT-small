from scipy.optimize import minimize
import time
from multiprocessing import Pool, cpu_count

def run(seed=42, budget_s=100, **kwargs):

    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate diverse initial guesses based on best_h
    num_perturbations = 150  # Reduced number for efficiency
    perturbation_strategies = [
        ('random', -0.15, 0.15),
        ('random', -0.05, 0.05),
        ('shift', 0.1, 0.1),
        ('flip', 0.5, 0.5),
        ('sparse', -0.2, 0.2)
    ]
    
    def generate_perturbation(h, strategy, noise_low, noise_high):
        if strategy == 'random':
            noise = np.random.uniform(noise_low, noise_high, n_points)
        elif strategy == 'shift':
            noise = np.random.uniform(-noise_high, noise_high, n_points)
        elif strategy == 'flip':
            noise = np.random.uniform(-noise_high, noise_high, n_points)
        elif strategy == 'sparse':
            noise = np.zeros(n_points)
            indices = np.random.choice(n_points, size=int(n_points * 0.1), replace=False)
            noise[indices] = np.random.uniform(-noise_high, noise_high, len(indices))
        else:
            noise = np.zeros(n_points)
        perturbed = h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        return perturbed
    
    # Objective function to minimize C5
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Constraints: sum(h) == target_sum
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    
    # Function to run optimization for a given perturbation
    def optimize_perturbation(perturbed):
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 1000}
        )
        if result.success:
            return result.fun, result.x
        return float('inf'), perturbed
    
    # Parallel processing
    with Pool(processes=min(cpu_count(), num_perturbations)) as pool:
