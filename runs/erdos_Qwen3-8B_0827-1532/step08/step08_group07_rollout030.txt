            triangular_wave[i] = (2.0 - x) / 1.0
    triangular_wave = triangular_wave / np.sum(triangular_wave) * (n_points / 2)
    initial_guesses.append(triangular_wave)
    
    # Square wave
    square_wave = np.zeros(n_points)
    for i in range(n_points):

            square_wave[i] = 0.0

    def objective(h):

    # Constraint function
    def constraint(h):

    # Bounds for each h[i]

        result = minimize(
