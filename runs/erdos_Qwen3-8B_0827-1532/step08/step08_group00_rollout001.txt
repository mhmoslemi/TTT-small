import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    np.random.seed(seed)
    initial_guesses = []

    initial_guesses.append(initial_guess.copy())

    initial_guesses.append(random_binary)

    structured_h = np.zeros(n_points)

    initial_guesses.append(structured_h)

    best_h = initial_guess.copy()

    for guess in initial_guesses:

        scale = target_sum / current_sum
        scaled_guess = np.clip(guess * scale, 0, 1)

        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Constraints and bounds
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]
        
        # Optimization with COBYLA
        result = minimize(
            objective,
            scaled_guess,
            method='COBYLA',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 1000}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
