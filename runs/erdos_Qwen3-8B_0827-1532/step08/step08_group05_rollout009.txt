    initial_guesses.append(structured_guess)
    
    # Generate more initial guesses with noise
    for _ in range(num_initial_guesses - 2):
        # Add structured noise with varying amplitude and frequency
        noise_amp = np.random.uniform(0.001, 0.05)
        noise_freq = np.random.uniform(1, 5)
        noise = noise_amp * np.sin(2 * np.pi * noise_freq * x_new)
        # Add random noise
        random_noise = np.random.normal(0, 0.01, n_points)
        perturbed = initial_guess + noise + random_noise
        # Clip to [0, 1]
        perturbed = np.clip(perturbed, 0, 1)
        # Adjust sum to meet the constraint
        scale_factor = (n_points / 2) / np.sum(perturbed)
        perturbed *= scale_factor
        initial_guesses.append(perturbed)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Run optimizations for each initial guess with tighter tolerances
    for guess in initial_guesses:
        # Run optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-10, 'maxiter': 3000}  # Increased maxiter and tighter tolerance
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # If no optimization was done, use the initial guess
    if best_h is None:
        result = minimize(
            objective,
            initial_guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-10, 'maxiter': 3000}
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points
