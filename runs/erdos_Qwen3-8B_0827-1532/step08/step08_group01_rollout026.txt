from scipy.optimize import minimize
import time

    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate multiple diverse initial guesses
    num_perturbations = 300  # Increased number of perturbations
    for perturbation_idx in range(num_perturbations):
        # Generate diverse initial guesses
        if perturbation_idx < 50:
            # Random initial guess with uniform distribution
            random_h = np.random.uniform(0, 1, n_points)
            random_h = random_h * (target_sum / np.sum(random_h))
        else:
            # Perturb best_h with varying noise
            if perturbation_idx < 100:
                # Larger noise for early perturbations
                noise = np.random.uniform(-0.2, 0.2, n_points)
            else:
                # Smaller noise for later perturbations
                noise = np.random.uniform(-0.05, 0.05, n_points)
            perturbed = best_h + noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
            random_h = perturbed
        
        # Objective function to minimize C5
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization with adjusted tolerances
        result = minimize(
            objective,
            random_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-8, 'maxiter': 1500}
        )

        if result.success and result.fun < best_c5:
