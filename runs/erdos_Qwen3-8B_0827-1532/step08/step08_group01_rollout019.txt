                h[-1] = np.clip(h[-1], 0.0, 1.0)
                one_minus_h = 1.0 - h
                corr = np.correlate(h, one_minus_h, mode='full')
                max_corr = np.max(corr)
                return max_corr * dx
            
            bounds_de = [(0.0, 1.0) for _ in range(n_points - 1)]
            result = differential_evolution(objective_de, bounds_de, strategy='best1bin', 
                                          maxiter=100, popsize=15, tol=1e-10, mutation=(0.5, 1), recombination=0.8)
            h_opt = np.concatenate([result.x, [target_sum - np.sum(result.x)]])
            h_opt[-1] = np.clip(h_opt[-1], 0.0, 1.0)
            current_c5 = objective(h_opt)
        else:
            # Use SLSQP for refinement
            result = minimize(
                objective,
                perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'ftol': 1e-10, 'maxiter': 1500}
            )
            h_opt = result.x
            current_c5 = result.fun
        
        # Check if this result is better
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_opt
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
