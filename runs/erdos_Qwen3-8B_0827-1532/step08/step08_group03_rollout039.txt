import time

    initial_guess = initial_guess / np.sum(initial_guess) * (n_points / 2)

    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h

        return max_corr * dx
    
    # Constraint function
    def constraint(h):

    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
