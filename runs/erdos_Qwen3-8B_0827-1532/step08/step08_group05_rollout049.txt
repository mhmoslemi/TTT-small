from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):

    n_points = 100  # Balance between speed and resolution
    dx = 2.0 / n_points

        return max_corr * dx
    
    # Constraint: integral of h over [0,2] equals 1

        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
