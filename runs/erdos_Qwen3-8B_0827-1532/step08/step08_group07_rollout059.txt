from scipy.optimize import differential_evolution, minimize

    initial_guesses = []

    initial_guesses.append(sine_wave)
    # Triangular wave
    triangular_wave = np.zeros(n_points)
    for i in range(n_points):
        x = np.linspace(0, 2, n_points)[i]
        if x < 1.0:
            triangular_wave[i] = x / 1.0
        else:
            triangular_wave[i] = (2.0 - x) / 1.0
    initial_guesses.append(triangular_wave)
    # Square wave
    square_wave = np.zeros(n_points)
    for i in range(n_points):
        if np.linspace(0, 2, n_points)[i] < 1.0:
            square_wave[i] = 1.0
        else:
            square_wave[i] = 0.0
    initial_guesses.append(square_wave)
    
    # Add perturbed and random guesses based on initial guess
    for _ in range(num_initial_guesses):
        # Perturb with sine wave
        noise_amp = np.random.uniform(0.001, 0.05)
        noise_freq = np.random.uniform(0.5, 10)
        noise = noise_amp * np.sin(2 * np.pi * noise_freq * np.linspace(0, 2, n_points))
        perturbed = initial_guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        scale_factor = (n_points / 2) / np.sum(perturbed)
        perturbed *= scale_factor
        initial_guesses.append(perturbed)
        # Add a random guess
        random_h = np.random.uniform(0, 1, n_points)
        random_h = random_h / np.sum(random_h) * (n_points / 2)
        initial_guesses.append(random_h)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Run optimizations for each initial guess with tighter tolerances
    for guess in initial_guesses:
        # Run differential evolution for global search
        result_de = differential_evolution(
            objective,
            bounds,
            constraints={'type': 'eq', 'fun': constraint},
            popsize=15,
            maxiter=50,
            tol=1e-5,
            seed=seed
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result_de.success:
            current_c5 = result_de.fun
            if current_c5 < best_c5:

                best_h = result_de.x

        result = minimize(
