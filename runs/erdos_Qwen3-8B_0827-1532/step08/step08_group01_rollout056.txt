import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Precompute constants
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate diverse initial guesses
    initial_guesses = []
    initial_guesses.append(initial_guess)
    
    # Uniform guess
    uniform_guess = np.full(n_points, 0.5)
    initial_guesses.append(uniform_guess)
    
    # Periodic guess (alternating 0 and 1)
    periodic_guess = np.zeros(n_points)
    for i in range(n_points):
        periodic_guess[i] = 0.5 * (1 - (-1)**i)
    initial_guesses.append(periodic_guess)
    
    # Random guesses
    for _ in range(2):
        random_guess = np.random.uniform(0, 1, n_points)
        random_guess = random_guess * (target_sum / np.sum(random_guess))
        initial_guesses.append(random_guess)
    
    # Optimization parameters
    options = {
        'ftol': 1e-8,
        'maxiter': 1000,
        'disp': False
    }
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Bounds and constraints
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    
    # Process each initial guess
    for guess in initial_guesses:
        # Use trust-constr for better convergence
        result = minimize(
            objective,
            guess,
            method='trust-constr',
            bounds=bounds,
            constraints=cons,
            options=options
        )
        
        # Update best solution
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
