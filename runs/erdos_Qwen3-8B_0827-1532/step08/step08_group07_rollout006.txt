from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Balanced between speed and precision
    dx = 2.0 / n_points
    
    # Generate initial guess based on provided initial_h_values
    initial_guess = np.array(initial_h_values)
    # Rescale to meet the constraint
    initial_guess = initial_guess / np.sum(initial_guess) * (n_points / 2)
    
    # Generate diverse initial guesses
    num_initial_guesses = 20  # Increased for better exploration
    initial_guesses = []
    
    # Add structured patterns
    # Sine wave
    sine_wave = 0.5 + 0.5 * np.sin(np.pi * np.linspace(0, 2, n_points))
    initial_guesses.append(sine_wave)
    # Triangular wave
    triangular_wave = np.zeros(n_points)
    for i in range(n_points):
        x = np.linspace(0, 2, n_points)[i]
        if x < 1.0:
            triangular_wave[i] = x / 1.0
        else:
            triangular_wave[i] = (2.0 - x) / 1.0
    initial_guesses.append(triangular_wave)
    # Square wave
    square_wave = np.zeros(n_points)
    for i in range(n_points):
        if np.linspace(0, 2, n_points)[i] < 1.0:
            square_wave[i] = 1.0
        else:
            square_wave[i] = 0.0
    initial_guesses.append(square_wave)
    # Pulse train
    pulse_train = np.zeros(n_points)
    for i in range(n_points):
        x = np.linspace(0, 2, n_points)[i]
        if (x >= 0.0 and x < 0.2) or (x >= 0.6 and x < 0.8) or (x >= 1.2 and x < 1.4) or (x >= 1.8 and x < 2.0):
            pulse_train[i] = 1.0
        else:
            pulse_train[i] = 0.0
    initial_guesses.append(pulse_train)
    # Add initial_h_values as a base guess
    initial_guesses.append(initial_guess)
    
    # Add perturbed and random guesses based on initial guess
    for _ in range(num_initial_guesses):
        # Perturb with sine wave
        noise_amp = np.random.uniform(0.001, 0.05)
        noise_freq = np.random.uniform(0.5, 10)
        noise = noise_amp * np.sin(2 * np.pi * noise_freq * np.linspace(0, 2, n_points))
        perturbed = initial_guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        scale_factor = (n_points / 2) / np.sum(perturbed)
        perturbed *= scale_factor
        initial_guesses.append(perturbed)
        # Add a random guess
        random_h = np.random.uniform(0, 1, n_points)
        random_h = random_h / np.sum(random_h) * (n_points / 2)
        initial_guesses.append(random_h)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Run optimizations for each initial guess with tighter tolerances
    for guess in initial_guesses:
        # Run optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-9, 'maxiter': 3000}
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:

        if result.success:

        result = minimize(

            options={'ftol': 1e-9, 'maxiter': 3000}
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points
