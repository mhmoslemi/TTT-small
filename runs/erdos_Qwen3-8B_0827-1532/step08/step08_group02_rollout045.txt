    dx = 2.0 / n_points
    
    # Generate initial guesses
    initial_guesses = []
    
    # Original initial guess
    x_new = np.linspace(0, 2, n_points)
    initial_guess = np.interp(x_new, initial_h_values, initial_h_values)
    # Adjust sum to meet the constraint
    initial_guess = initial_guess / np.sum(initial_guess) * (n_points / 2)
    initial_guesses.append(initial_guess)
    
    # Symmetric initial guess
    symmetric_guess = np.zeros(n_points)
    for i in range(n_points):
        j = n_points - 1 - i
        symmetric_guess[i] = initial_guess[j]
    # Adjust sum to meet the constraint (should already be correct)
    symmetric_guess = symmetric_guess / np.sum(symmetric_guess) * (n_points / 2)
    initial_guesses.append(symmetric_guess)
    
    # Step function initial guess
    step_guess = np.zeros(n_points)
    for i in range(n_points):
        if x_new[i] < 1.0:
            step_guess[i] = 1.0
        else:
            step_guess[i] = 0.0
    # Adjust sum to meet the constraint (already meets)
    step_guess = step_guess / np.sum(step_guess) * (n_points / 2)
    initial_guesses.append(step_guess)
    
    # Sinusoidal initial guess
    sinusoidal_guess = 0.5 + 0.5 * np.sin(np.pi * x_new)
    # Clip to [0, 1]
    sinusoidal_guess = np.clip(sinusoidal_guess, 0, 1)
    # Adjust sum to meet the constraint
    sinusoidal_guess = sinusoidal_guess / np.sum(sinusoidal_guess) * (n_points / 2)
    initial_guesses.append(sinusoidal_guess)
    
    # Add a few random initial guesses
    num_random_guesses = 5
    for _ in range(num_random_guesses):
        # Random perturbation
        noise = np.random.uniform(-0.05, 0.05, size=n_points)
        perturbed = initial_guess + noise
        # Clip to [0, 1]
        perturbed = np.clip(perturbed, 0, 1)
        # Adjust sum to meet the constraint
        scale_factor = (n_points / 2) / np.sum(perturbed)
        perturbed *= scale_factor
        initial_guesses.append(perturbed)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Run optimizations for each initial guess
    for guess in initial_guesses:
        # Run optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-10, 'maxiter': 3000}  # Tighter tolerances and more iterations
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # If no optimization was done, use the initial guess
    if best_h is None:
        result = minimize(
            objective,
            initial_guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-10, 'maxiter': 3000}
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points
