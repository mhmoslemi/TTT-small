    # Add a periodic pattern (alternating 1 and 0)
    periodic_guess = np.zeros(n_points)
    for i in range(n_points):

    periodic_guess = periodic_guess * (target_sum / np.sum(periodic_guess))

        def objective(h):

        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
