from scipy.optimize import minimize
import time

            noise = noise_amp * np.sin(2 * np.pi * noise_freq * x_new)
        elif noise_type == 'random_steps':
            num_steps = np.random.randint(1, 10)
            indices = np.random.choice(n_points, num_steps, replace=False)
            noise = np.zeros(n_points)
            noise[indices] = np.random.uniform(-0.1, 0.1, size=num_steps)
        elif noise_type == 'random_walk':
            noise = np.zeros(n_points)
            current = 0
            for i in range(n_points):
                step = np.random.uniform(-0.01, 0.01)
                current += step
                noise[i] = current
        # Perturb the initial guess
        perturbed = initial_guess + noise
        # Clip to [0, 1]
        perturbed = np.clip(perturbed, 0, 1)
        # Adjust sum to meet the constraint

    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    for guess in initial_guesses:
        # Run optimization with L-BFGS-B method
        result = minimize(
