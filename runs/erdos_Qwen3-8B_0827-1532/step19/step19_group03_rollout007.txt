    sum_h = np.sum(h_pattern2)
    scale = target_sum / sum_h
    initial_guesses.append(h_pattern2 * scale)
    
    h_pattern3 = np.zeros(n_points)
    h_pattern3[::10] = 1.0
    sum_h = np.sum(h_pattern3)
    scale = target_sum / sum_h
    initial_guesses.append(h_pattern3 * scale)
    
    h_pattern4 = np.zeros(n_points)
    h_pattern4[:n_points//2] = 1.0
    sum_h = np.sum(h_pattern4)
    scale = target_sum / sum_h
    initial_guesses.append(h_pattern4 * scale)
    
    h_pattern5 = np.zeros(n_points)
    h_pattern5[n_points//2:] = 1.0
    sum_h = np.sum(h_pattern5)
    scale = target_sum / sum_h
    initial_guesses.append(h_pattern5 * scale)
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 2000}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Perturbation phase on the best_h found so far
    num_perturbations = 800  # Increased perturbations
    phase_shift = 0.15
    phase_decay = 0.95
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 600}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
