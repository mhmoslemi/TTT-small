            current_sum_perturb = np.sum(current_h_perturb)
            if current_sum_perturb != 0:
                current_h_perturb = current_h_perturb * (target_sum / current_sum_perturb)
            current_c5_perturb = objective_perturb(current_h_perturb)
            if current_c5_perturb < best_c5:
                best_c5 = current_c5_perturb
                best_h = current_h_perturb
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
