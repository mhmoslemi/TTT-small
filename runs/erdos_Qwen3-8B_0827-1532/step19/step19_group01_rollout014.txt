from scipy.optimize import minimize
import time

    # Alternating 1 and 0
    h_alt = np.zeros(n_points)
    h_alt[::2] = 1.0

    scale = target_sum / sum_h

    # Central 50% step function

    initial_guesses.append(h_center * scale)

    for guess in initial_guesses:

        def objective(h):
            one_minus_h = 1.0 - h

            return np.max(corr) * dx

        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
