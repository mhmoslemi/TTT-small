    scale = target_sum / sum_h
    initial_guesses.append(h_three_intervals * scale)
    
    # Four intervals spaced with gaps
    h_four_intervals = np.zeros(n_points)
    interval_length = 1.0 / 4.0
    start1 = 0.0
    end1 = start1 + interval_length
    start2 = end1 + interval_length / 2
    end2 = start2 + interval_length
    start3 = end2 + interval_length / 2
    end3 = start3 + interval_length
    start4 = end3 + interval_length / 2
    end4 = start4 + interval_length
    
    start1_idx = int(start1 / dx)
    end1_idx = int(end1 / dx)
    start2_idx = int(start2 / dx)
    end2_idx = int(end2 / dx)
    start3_idx = int(start3 / dx)
    end3_idx = int(end3 / dx)
    start4_idx = int(start4 / dx)
    end4_idx = int(end4 / dx)
    
    h_four_intervals[start1_idx:end1_idx] = 1.0
    h_four_intervals[start2_idx:end2_idx] = 1.0
    h_four_intervals[start3_idx:end3_idx] = 1.0
    h_four_intervals[start4_idx:end4_idx] = 1.0
    
    sum_h = np.sum(h_four_intervals)
    scale = target_sum / sum_h
    initial_guesses.append(h_four_intervals * scale)
    
    # Random step functions with varying intervals
    for _ in range(10):  # Increased from 5
        h_rand = np.zeros(n_points)
        num_intervals = np.random.randint(3, 6)
        for i in range(num_intervals):
            start = np.random.randint(0, n_points - 1)
            end = start + np.random.randint(1, n_points - start)
            h_rand[start:end] = 1.0
        sum_h = np.sum(h_rand)
        scale = target_sum / sum_h
        initial_guesses.append(h_rand * scale)
    
    # Optimize each initial guess with SLSQP
    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 1000}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Perturbation phase on the best_h found so far
    num_perturbations = 800  # Increased from 600
    phase_shift = 0.15
    phase_decay = 0.95  # Reduced from 0.96 to allow more variation
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 500}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
