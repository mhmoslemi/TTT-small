from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Define objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    # Define constraint
    def constraint(h):
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint})
    
    # Initial optimization from initial guess
    result = minimize(
        objective,
        initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-9, 'maxiter': 1000}
    )
    
    if result.success and result.fun < best_c5:
        best_c5 = result.fun
        best_h = result.x
    
    # Perturbation phase with structured variations
    num_perturbations = 200  # Reduced from 300 for time efficiency
    phase_shift = 0.15
    phase_decay = 0.96
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        # Structured perturbation: shift segments
        perturbed = best_h.copy()
        # Randomly select a segment to adjust
        segment_start = np.random.randint(0, n_points - 1)
        segment_length = np.random.randint(1, n_points // 4)
        segment_end = segment_start + segment_length
        perturbed[segment_start:segment_end] += np.random.uniform(-noise_scale, noise_scale, segment_length)
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Optimization with SLSQP
        result = minimize(
            objective,
            perturbed,

        )
        
        if result.success and result.fun < best_c5:
