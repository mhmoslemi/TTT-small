from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate diverse initial guesses
    initial_guesses = [initial_guess.copy()]
    
    # Step function: first half 1, second half 0
    h_step = np.zeros(n_points)
    h_step[:n_points//2] = 1.0
    sum_h = np.sum(h_step)
    scale = target_sum / sum_h
    initial_guesses.append(h_step * scale)
    
    # Alternating 1 and 0
    h_alt = np.zeros(n_points)
    h_alt[::2] = 1.0
    sum_h = np.sum(h_alt)
    scale = target_sum / sum_h
    initial_guesses.append(h_alt * scale)
    
    # Three-segment step function
    h_three_seg = np.zeros(n_points)
    seg1_start = n_points // 4
    seg1_end = n_points // 2
    seg2_start = n_points // 2
    seg2_end = 3 * n_points // 4
    seg3_start = 3 * n_points // 4
    seg3_end = n_points
    h_three_seg[seg1_start:seg1_end] = 1.0
    h_three_seg[seg2_start:seg2_end] = 1.0
    h_three_seg[seg3_start:seg3_end] = 1.0
    sum_h = np.sum(h_three_seg)
    scale = target_sum / sum_h
    initial_guesses.append(h_three_seg * scale)
    
    # Central step function
    h_center = np.zeros(n_points)
    center_start = n_points // 4
    center_end = 3 * n_points // 4
    h_center[center_start:center_end] = 1.0
    sum_h = np.sum(h_center)
    scale = target_sum / sum_h
    initial_guesses.append(h_center * scale)
    
    # Two-segment step function
    h_two_seg = np.zeros(n_points)
    seg1_start = n_points // 4
    seg1_end = 3 * n_points // 4
    seg2_start = 3 * n_points // 4
    seg2_end = n_points
    h_two_seg[seg1_start:seg1_end] = 1.0
    h_two_seg[seg2_start:seg2_end] = 1.0
    sum_h = np.sum(h_two_seg)
    scale = target_sum / sum_h
    initial_guesses.append(h_two_seg * scale)
    
    # Random initial guess
    initial_guesses.append(np.random.uniform(0, 1, n_points))
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 800}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Perturbation phase with random variations
    num_perturbations = 100  # Reduced from 150 to save time
    phase_shift = 0.15
    phase_decay = 0.96
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        # Apply random perturbation
        perturbed = best_h + noise_scale * np.random.normal(0, 0.1, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(

        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
