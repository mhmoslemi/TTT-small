    h_four_seg = np.zeros(n_points)

    h_four_seg[:seg1_end] = 1.0
    h_four_seg[seg1_end:seg2_end] = 1.0
    h_four_seg[seg2_end:seg3_end] = 0.0
    h_four_seg[seg3_end:] = 1.0
    sum_h = np.sum(h_four_seg)
    scale = target_sum / sum_h
    initial_guesses.append(h_four_seg * scale)

    initial_guesses.append(np.random.uniform(0, 1, n_points))
    
    # Optimize each initial guess
    for guess in initial_guesses:

        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(

            constraints=cons,
