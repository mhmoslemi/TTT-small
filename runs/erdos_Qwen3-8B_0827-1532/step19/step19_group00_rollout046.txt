    h_three_seg[seg1_start:seg1_end] = 1.0
    h_three_seg[seg2_start:seg2_end] = 1.0
    h_three_seg[seg3_start:seg3_end] = 1.0
    sum_h = np.sum(h_three_seg)
    scale = target_sum / sum_h
    initial_guesses.append(h_three_seg * scale)
    
    # Single segment in the middle
    h_single_seg = np.zeros(n_points)
    seg_start = n_points // 2 - 10
    seg_end = n_points // 2 + 10
    h_single_seg[seg_start:seg_end] = 1.0
    sum_h = np.sum(h_single_seg)
    scale = target_sum / sum_h
    initial_guesses.append(h_single_seg * scale)
    
    # Two segments, symmetric
    h_two_seg = np.zeros(n_points)
    seg1_start = n_points // 4
    seg1_end = n_points // 2
    seg2_start = 3 * n_points // 4
    seg2_end = n_points
    h_two_seg[seg1_start:seg1_end] = 1.0
    h_two_seg[seg2_start:seg2_end] = 1.0
    sum_h = np.sum(h_two_seg)
    scale = target_sum / sum_h
    initial_guesses.append(h_two_seg * scale)
    
    # Random initial guess
    initial_guesses.append(np.random.uniform(0, 1, n_points))
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 800}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Perturbation phase with structured variations
    num_perturbations = 200
    phase_shift = 0.15
    phase_decay = 0.96
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        # Apply random noise and sinusoidal noise
        perturbed = best_h + noise_scale * (np.random.normal(0, 0.1, n_points) + np.sin(2 * np.pi * np.linspace(0, 1, n_points)))
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 300}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
