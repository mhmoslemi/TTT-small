        num_intervals = np.random.randint(3, 6)
        for i in range(num_intervals):
            start = np.random.randint(0, n_points - 1)
            end = start + np.random.randint(1, n_points - start)
            h_rand[start:end] = 1.0
        sum_h = np.sum(h_rand)
        scale = target_sum / sum_h
        initial_guesses.append(h_rand * scale)
    
    # Evenly spaced intervals initial guesses
    for m in [2, 3, 4, 5]:
        h_even = np.zeros(n_points)
        l = 1.0 / m  # length of each interval
        s = 1.0 / (m - 1) if m > 1 else 0.0  # spacing between intervals
        start = 0.0
        for i in range(m):
            end = start + l
            start_idx = int(start / dx)
            end_idx = int(end / dx)
            # Ensure end_idx doesn't exceed n_points
            end_idx = min(end_idx, n_points)
            h_even[start_idx:end_idx] = 1.0
            start = end + s
        sum_h = np.sum(h_even)
        scale = target_sum / sum_h
        initial_guesses.append(h_even * scale)
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 1500}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Perturbation phase on the best_h found so far
    num_perturbations = 1000  # Increased from 800
    phase_shift = 0.15
    phase_decay = 0.95  # Slightly slower decay
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 600}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
