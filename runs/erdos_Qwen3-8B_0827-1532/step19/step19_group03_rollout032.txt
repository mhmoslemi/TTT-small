from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate diverse initial guesses
    initial_guesses = []
    initial_guesses.append(initial_guess.copy())  # Original initial guess
    
    # Step function: first half 1, second half 0
    h_step = np.zeros(n_points)
    h_step[:n_points//2] = 1.0
    sum_h = np.sum(h_step)
    scale = target_sum / sum_h
    initial_guesses.append(h_step * scale)
    
    # Alternating 1 and 0
    h_alt = np.zeros(n_points)
    h_alt[::2] = 1.0
    sum_h = np.sum(h_alt)
    scale = target_sum / sum_h
    initial_guesses.append(h_alt * scale)
    
    # Central 50% step function
    h_center = np.zeros(n_points)
    center_start = n_points // 2 - n_points // 4
    center_end = n_points // 2 + n_points // 4
    h_center[center_start:center_end] = 1.0
    sum_h = np.sum(h_center)
    scale = target_sum / sum_h
    initial_guesses.append(h_center * scale)
    
    # 25% step function
    h_25 = np.zeros(n_points)
    h_25[:n_points//4] = 1.0
    h_25[n_points//2:n_points//2 + n_points//4] = 1.0
    sum_h = np.sum(h_25)
    scale = target_sum / sum_h
    initial_guesses.append(h_25 * scale)
    
    # Symmetric step function
    h_sym = np.zeros(n_points)
    center = n_points // 2
    width = n_points // 4
    h_sym[center - width:center + width] = 1.0
    sum_h = np.sum(h_sym)
    scale = target_sum / sum_h
    initial_guesses.append(h_sym * scale)
    
    # Ends step function
    h_ends = np.zeros(n_points)
    h_ends[:n_points//4] = 1.0
    h_ends[-n_points//4:] = 1.0
    sum_h = np.sum(h_ends)
    scale = target_sum / sum_h
    initial_guesses.append(h_ends * scale)
    
    # Random initial guess
    initial_guesses.append(np.random.uniform(0, 1, n_points))
    
    # Generate random step functions with varying intervals
    for _ in range(10):  # More varied initial guesses

        num_intervals = np.random.randint(3, 6)
        for i in range(num_intervals):
            start = np.random.randint(0, n_points - 1)
            end = start + np.random.randint(1, n_points - start)
            h_rand[start:end] = 1.0
        sum_h = np.sum(h_rand)
        scale = target_sum / sum_h
        initial_guesses.append(h_rand * scale)
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 1500}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun

        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function

            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
