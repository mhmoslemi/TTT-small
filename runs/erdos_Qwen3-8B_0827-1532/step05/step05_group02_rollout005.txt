from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Use the provided initial guess
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate diverse initial guesses
    num_perturbations = 30  # More perturbations for better exploration
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    for perturbation_idx in range(num_perturbations):
        # Generate diverse initial guesses: some random, some perturbed
        if perturbation_idx < 5:
            # Random initial guess
            random_guess = np.random.uniform(0, 1, n_points)
            random_guess = random_guess * (target_sum / np.sum(random_guess))
            perturbed = random_guess
        else:
            # Perturb the initial guess with varying noise
            if perturbation_idx < 10:
                # Larger noise for early perturbations
                noise = np.random.uniform(-0.2, 0.2, n_points)
            else:
                # Smaller noise for later perturbations
                noise = np.random.uniform(-0.05, 0.05, n_points)
            perturbed = initial_guess + noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
        
        # Objective function to minimize C5
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization with tighter tolerances and more iterations
        result = minimize(
            objective,
            perturbed,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-8, 'maxiter': 500}
        )

            best_h = result.x

        if elapsed >= budget_s:
