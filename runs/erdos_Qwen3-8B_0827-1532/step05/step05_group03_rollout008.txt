import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Generate multiple initial guesses
    n_points = len(initial_h_values)
    initial_guesses = [initial_h_values]
    
    # Add alternating pattern
    h_alternating = np.zeros(n_points)
    for i in range(n_points):
        if i % 2 == 0:
            h_alternating[i] = 0.0
        else:
            h_alternating[i] = 1.0
    sum_h = np.sum(h_alternating)
    scale = (n_points / 2) / sum_h
    h_alternating *= scale
    initial_guesses.append(h_alternating)
    
    # Add random initial guess
    np.random.seed(seed)
    h_random = np.random.uniform(0, 1, n_points)
    sum_h = np.sum(h_random)
    scale = (n_points / 2) / sum_h
    h_random *= scale
    initial_guesses.append(h_random)
    
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    start_time = time.time()
    
    for guess in initial_guesses:
        # Objective function to minimize C5
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * (2.0 / n_points)
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        target_sum = n_points / 2
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization with tighter tolerances and more iterations
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 1000}
        )
        
        # Check if this result is better
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
