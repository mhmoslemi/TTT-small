def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    
    # Generate initial guess
    h_initial = np.zeros(n_points)
    h_initial[0:50] = 1.0
    h_initial[100:150] = 1.0
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization
    result = minimize(
        objective,
        h_initial,
        method='trust-constr',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint},
        options={'gtol': 1e-8, 'maxiter': 1000}
    )
    
    # Check if optimization was successful
    if result.success:
        best_c5 = result.fun
        best_h = result.x
    else:
        # Fallback to initial guess
        best_c5 = objective(h_initial)
        best_h = h_initial
    
    return best_h, best_c5, n_points
