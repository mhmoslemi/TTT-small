            # Random perturbation with varying scales
            noise = np.random.uniform(-0.2, 0.2, n_points)
            perturbed = base_guess + noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
            guesses.append(perturbed)
            # Add structured variations: shift and mirror
            shifted = np.roll(perturbed, i+1)
            guesses.append(shifted)
            mirrored = np.flip(perturbed)
            guesses.append(mirrored)
        return guesses
    
    initial_guesses = generate_initial_guesses(initial_guess, num_variants=5)

    start_time = time.time()
    
    # Define objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define constraints
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    
    # Function to run optimization for a single guess
    def optimize_guess(h_initial):
        result = minimize(
            objective,
            h_initial,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 200}
        )
        if result.success:
            return result.x, result.fun

    # Use multiprocessing to parallelize optimization
