        # First interval: [0, 0.5]
        start = 0.0
        end = 0.5
        for i in range(n_points):
            if start <= i*dx <= end:
                h[i] = 1.0
        # Second interval: [1, 1.5]
        start = 1.0
        end = 1.5
        for i in range(n_points):
            if start <= i*dx <= end:
                h[i] = 1.0
        return h
    
    # Generate three-interval pattern
    def generate_three_interval_guess(n_points):
        dx = 2.0 / n_points
        h = np.zeros(n_points)
        # First interval: [0, 1/3]
        start = 0.0
        end = 1.0/3.0
        for i in range(n_points):
            if start <= i*dx <= end:
                h[i] = 1.0
        # Second interval: [1/3 + 1/3, 1/3 + 2/3]
        start = 1.0/3.0 + 1.0/3.0
        end = start + 1.0/3.0
        for i in range(n_points):
            if start <= i*dx <= end:
                h[i] = 1.0
        # Third interval: [1/3 + 2/3, 1/3 + 3/3]
        start = 1.0/3.0 + 2.0/3.0
        end = start + 1.0/3.0
        for i in range(n_points):
            if start <= i*dx <= end:
                h[i] = 1.0
        return h
    
    # Generate four-interval pattern
    def generate_four_interval_guess(n_points):
        dx = 2.0 / n_points
        h = np.zeros(n_points)
        # First interval: [0, 0.25]
        start = 0.0
        end = 0.25
        for i in range(n_points):
            if start <= i*dx <= end:
                h[i] = 1.0
        # Second interval: [0.5, 0.75]
        start = 0.5
        end = 0.75
        for i in range(n_points):
            if start <= i*dx <= end:
                h[i] = 1.0
        # Third interval: [1.0, 1.25]
        start = 1.0
        end = 1.25
        for i in range(n_points):
            if start <= i*dx <= end:
                h[i] = 1.0
        # Fourth interval: [1.5, 1.75]
        start = 1.5
        end = 1.75
        for i in range(n_points):
            if start <= i*dx <= end:
                h[i] = 1.0
        return h
    
    # Add structured guesses
    initial_guesses.append(generate_two_interval_guess(n_points))
    initial_guesses.append(generate_three_interval_guess(n_points))
    initial_guesses.append(generate_four_interval_guess(n_points))
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    start_time = time.time()
    
    for guess in initial_guesses:
        # Add small perturbation
        noise = np.random.normal(0, 0.05, n_points)
        perturbed = guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function to minimize C5
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 1000}
        )
        
        # Check if this result is better
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
