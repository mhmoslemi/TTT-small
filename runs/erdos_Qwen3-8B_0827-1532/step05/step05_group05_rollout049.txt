import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    structured_guesses = []
    for period in [2, 3, 4, 5, 6]:

        for i in range(n_points):
            if i % period == 0:
                structured[i] = 1.0
        structured_guesses.append(structured.copy())
    initial_guesses.extend(structured_guesses)
    
    # Add random guesses
    for _ in range(num_initial_guesses - len(initial_guesses)):
        random_guess = np.random.rand(n_points)
        random_guess /= np.sum(random_guess)
        random_guess *= target_sum
        initial_guesses.append(random_guess)
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    start_time = time.time()
    
    for guess in initial_guesses:
        # Perturb with noise and scale
        noise = np.random.normal(0, 0.05, n_points)
        perturbed = guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Bounds and constraints
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Optimization with Nelder-Mead
        result = minimize(
            objective,
            perturbed,
            method='Nelder-Mead',
            bounds=bounds,
            constraints=cons,
            options={'xatol': 1e-6, 'maxiter': 500}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
