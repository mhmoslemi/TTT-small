import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 200  # Increased resolution for better optimization
    dx = 2.0 / n_points
    
    # Generate initial guesses: interpolated and random
    x_initial = np.linspace(0, 2, len(initial_h_values))
    x_new = np.linspace(0, 2, n_points)
    initial_guess_interpolated = np.interp(x_new, x_initial, initial_h_values)
    initial_guess_interpolated = initial_guess_interpolated / np.sum(initial_guess_interpolated) * (n_points / 2)
    
    # Generate random initial guesses
    num_random_guesses = 20
    random_initial_guesses = []
    for _ in range(num_random_guesses):
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
        random_initial_guesses.append(h)
    
    initial_guesses = [initial_guess_interpolated] + random_initial_guesses
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    
    # Run optimizations for each initial guess
    for guess in initial_guesses:
        # Run optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-9, 'maxiter': 2000}
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # If no optimization was done, use the initial guess
    if best_h is None:
        result = minimize(
            objective,
            initial_guess_interpolated,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-9, 'maxiter': 2000}
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points
