import numpy as np
from scipy.optimize import minimize
import time
from joblib import Parallel, delayed

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate multiple perturbations with varying noise scales
    num_perturbations = 50  # Increased number of perturbations for better exploration
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    def optimize_perturbation(perturbed):
        # Objective function to minimize C5
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization with tighter tolerances and more iterations
        result = minimize(
            objective,
            perturbed,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 200}
        )
        
        return result.x, result.fun if result.success else None
    
    # Generate perturbations
    perturbations = []
    for i in range(num_perturbations):
        if i < 10:
            # Larger noise for early perturbations to explore more
            noise = np.random.uniform(-0.3, 0.3, n_points)
        else:
            # Smaller noise for later perturbations to refine
            noise = np.random.uniform(-0.1, 0.1, n_points)
        perturbed = initial_guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        perturbations.append(perturbed)
    
    # Parallel optimization
    results = Parallel(n_jobs=10, backend='threading')(

    )
    
    for h, c5 in results:
