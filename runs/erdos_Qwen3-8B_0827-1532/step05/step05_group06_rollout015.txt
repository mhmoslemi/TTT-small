import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = 80
    dx = 2.0 / n_points
    np.random.seed(seed)
    
    # Generate diverse initial guesses
    initial_guesses = []
    # 1. Use the provided initial_h_values
    initial_guesses.append(initial_h_values)
    # 2. Symmetric random initial guess
    random_initial = np.zeros(n_points)
    selected_pairs = set()
    while len(selected_pairs) < n_points // 4:
        i = np.random.randint(0, n_points)
        j = n_points - 1 - i
        if i != j and i not in selected_pairs and j not in selected_pairs:
            selected_pairs.add(i)
            selected_pairs.add(j)
    for i in selected_pairs:
        random_initial[i] = 1
    initial_guesses.append(random_initial)
    # 3. Structured initial guess (every other interval, symmetric)
    structured_initial = np.zeros(n_points)
    structured_initial[::2] = 1
    initial_guesses.append(structured_initial)
    # 4. Evenly spaced initial guess (symmetric)
    evenly_spaced_initial = np.zeros(n_points)
    for i in range(n_points // 2):
        pos = int(i * n_points / (n_points // 2))
        evenly_spaced_initial[pos] = 1
        evenly_spaced_initial[n_points - 1 - pos] = 1
    initial_guesses.append(evenly_spaced_initial)
    # 5. Perturbations of initial_h_values
    for _ in range(30):  # More perturbations
        perturb = np.random.normal(0, 0.05, n_points)
        perturbed = initial_h_values + perturb
        perturbed = np.clip(perturbed, 0, 1)
        sum_perturbed = np.sum(perturbed)
        if sum_perturbed != 0:
            perturbed = perturbed * (n_points / 2) / sum_perturbed
        initial_guesses.append(perturbed)
    # 6. Random block initial guess
    block_initial = np.zeros(n_points)
    block_size = n_points // 4
    for i in range(2):
        start = np.random.randint(0, n_points - block_size)
        block_initial[start:start+block_size] = 1
    block_initial = np.clip(block_initial, 0, 1)
    sum_block = np.sum(block_initial)
    if sum_block != 0:
        block_initial = block_initial * (n_points / 2) / sum_block
    initial_guesses.append(block_initial)
    # 7. Random sparse initial guess
    sparse_initial = np.zeros(n_points)
    num_ones = np.random.randint(10, 15)
    indices = np.random.choice(n_points, num_ones, replace=False)
    sparse_initial[indices] = 1
    sum_sparse = np.sum(sparse_initial)
    if sum_sparse != 0:
        sparse_initial = sparse_initial * (n_points / 2) / sum_sparse
    initial_guesses.append(sparse_initial)
    # 8. Additional structured guess: staggered intervals
    staggered_initial = np.zeros(n_points)
    for i in range(0, n_points, 4):
        staggered_initial[i] = 1
        staggered_initial[i+1] = 1
        staggered_initial[i+2] = 1
        staggered_initial[i+3] = 1
    staggered_initial = np.clip(staggered_initial, 0, 1)
    sum_staggered = np.sum(staggered_initial)
    if sum_staggered != 0:
        staggered_initial = staggered_initial * (n_points / 2) / sum_staggered
    initial_guesses.append(staggered_initial)
    
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    
    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
    bounds = [(0.0, 1.0) for _ in range(n_points)]
