        start = 0.0
        end = 1.0 / 3.0
        for i in range(n_points):
            x = i * dx
            if start <= x <= end:
                h[i] = 1.0
        # interval2: [0.5 + 1/3, 0.5 + 2/3]
        start = 0.5 + 1.0 / 3.0
        end = 0.5 + 2.0 / 3.0
        for i in range(n_points):
            x = i * dx
            if start <= x <= end:
                h[i] = 1.0
        # interval3: [1.666..., 2.0]
        start = 2.0 / 3.0 + 1.0
        end = 2.0
        for i in range(n_points):
            x = i * dx
            if start <= x <= end:
                h[i] = 1.0
        # Ensure the sum is correct
        total = np.sum(h) * dx
        scale_factor = (n_points / 2) / total
        h *= scale_factor
        return h
    
    # Generate initial guesses
    initial_guesses = []
    # Add structured initial guess
    structured_guess = create_structured_initial_guess(n_points)
    initial_guesses.append(structured_guess)
    # Add perturbations of structured guess
    for _ in range(10):
        # Add noise
        noise_amp = np.random.uniform(0.005, 0.02)
        noise_freq = np.random.uniform(1, 5)
        noise = noise_amp * np.sin(2 * np.pi * noise_freq * np.linspace(0, 2, n_points))
        perturbed = structured_guess + noise
        # Clip to [0, 1]
        perturbed = np.clip(perturbed, 0, 1)
        # Adjust sum to meet the constraint
        scale_factor = (n_points / 2) / np.sum(perturbed)
        perturbed *= scale_factor
        initial_guesses.append(perturbed)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Run optimizations for each initial guess with tighter tolerances
    for guess in initial_guesses:
        # Run optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-9, 'maxiter': 1500}
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # If no optimization was done, use the initial guess
    if best_h is None:
        result = minimize(
            objective,
            structured_guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-9, 'maxiter': 1500}
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points
