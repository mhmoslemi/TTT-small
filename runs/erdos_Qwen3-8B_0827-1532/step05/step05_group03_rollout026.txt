from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Use the provided initial guess and generate random ones
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Generate multiple initial guesses: some random, some perturbed
    num_initial_guesses = 10  # Adjust based on time

    # Add the original initial guess
    initial_guesses.append(initial_guess.copy())
    # Generate random initial guesses
    for _ in range(num_initial_guesses - 1):
        h = np.random.uniform(0, 1, n_points)
        h = h * target_sum / np.sum(h)
        initial_guesses.append(h)
    # Generate perturbed versions of the initial guess
    for _ in range(5):  # Add some perturbations
        noise = np.random.uniform(-0.1, 0.1, n_points)
        perturbed = initial_guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        initial_guesses.append(perturbed)
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()

            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx

        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization with tighter tolerances and more iterations

            objective,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 1000}
        )
        
        # Check if this result is better
