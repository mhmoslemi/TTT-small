import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Generate multiple initial guesses
    initial_guesses = [initial_h_values]
    # Generate some random initial guesses
    num_random_guesses = 5
    for _ in range(num_random_guesses):
        h = np.random.uniform(0, 1, len(initial_h_values))
        h = h * (len(initial_h_values) / 2) / np.sum(h)
        initial_guesses.append(h)
    
    best_c5 = float('inf')
    best_h = initial_guesses[0].copy()
    n_points = len(initial_guesses[0])
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    start_time = time.time()
    
    for guess in initial_guesses:
        # Perturb the guess with different noise scales
        perturbed = guess.copy()
        # Apply noise with varying scales
        noise_scale = 0.3
        perturbed += np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed *= scale
        
        # Objective function to minimize C5
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization with tighter tolerances and more iterations
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 1000}
        )
        
        # Check if this result is better
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
