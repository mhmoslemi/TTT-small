import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 200  # Increased for better precision
    dx = 2.0 / n_points
    
    # Generate initial guesses based on initial_h_values and random samples
    initial_guesses = []
    # Use initial_h_values as a base
    x_initial = np.linspace(0, 2, len(initial_h_values))
    x_new = np.linspace(0, 2, n_points)
    base_guess = np.interp(x_new, x_initial, initial_h_values)
    base_guess = base_guess / np.sum(base_guess) * (n_points / 2)
    
    # Add noise to base guess
    noise_scale = 0.01
    for _ in range(15):
        noise = np.random.normal(0, noise_scale, n_points)
        perturbed = base_guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        scale_factor = (n_points / 2) / np.sum(perturbed)
        perturbed *= scale_factor
        initial_guesses.append(perturbed)
    
    # Add random initial guesses
    for _ in range(10):
        random_guess = np.random.rand(n_points)
        random_guess = random_guess / np.sum(random_guess) * (n_points / 2)
        initial_guesses.append(random_guess)
    
    # Objective function with penalty for constraint violation
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        penalty = 1e4 * (np.sum(h) - n_points / 2)**2
        return max_corr * dx + penalty
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    
    # Run differential evolution for each initial guess
    for guess in initial_guesses:
        # Run optimization
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=100,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            disp=False,
            x0=guess
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        current_c5 = result.fun
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x
    
    # If no optimization was done, use the base guess
    if best_h is None:
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=100,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            disp=False,
            x0=base_guess
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points
