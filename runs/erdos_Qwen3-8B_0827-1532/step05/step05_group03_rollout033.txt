import numpy as np
from scipy.optimize import minimize
import time
from concurrent.futures import ProcessPoolExecutor, as_completed

def optimize_h(initial_guess, target_sum, n_points, max_time):
    dx = 2.0 / n_points
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    start_opt = time.time()
    result = minimize(
        objective,
        initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-9, 'maxiter': 1000}
    )
    elapsed_opt = time.time() - start_opt
    if elapsed_opt > max_time:
        return None
    return result.x, result.fun

def run(seed=42, budget_s=100, **kwargs):
    n_points = 100
    target_sum = n_points / 2
    initial_guesses = [initial_h_values]
    for _ in range(4):
        h = np.random.uniform(0, 1, n_points)
        h = h * (target_sum / np.sum(h))
        initial_guesses.append(h)
    
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    start_time = time.time()
    
    with ProcessPoolExecutor(max_workers=10) as executor:
        futures = []
        for guess in initial_guesses:
            remaining_time = budget_s - (time.time() - start_time)
            if remaining_time <= 0:

            future = executor.submit(optimize_h, guess, target_sum, n_points, remaining_time)
            futures.append(future)

            if result is not None:

                if c5 < best_c5:

                    best_h = h
    
    return best_h, best_c5, n_points
