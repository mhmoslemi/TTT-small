    # Generate multiple initial guesses
    def generate_initial_guesses(n_points):
        guesses = [initial_guess.copy()]
        
        # Random guess
        random_guess = np.random.uniform(0, 1, n_points)
        random_guess = random_guess * (target_sum / np.sum(random_guess))
        guesses.append(random_guess)
        
        # Structured guess: first half 1, second half 0
        structured_guess = np.zeros(n_points)
        structured_guess[:n_points//2] = 1.0
        guesses.append(structured_guess)
        
        # Alternating 1 and 0
        alt_guess = np.zeros(n_points)
        for i in range(n_points):
            if i % 2 == 0:
                alt_guess[i] = 1.0
            else:
                alt_guess[i] = 0.0
        # Ensure sum is correct
        guesses.append(alt_guess)
        
        # Block guess: 1, 0, 1, 0, etc. with block size
        block_guess = np.zeros(n_points)
        block_size = n_points // 4
        block_guess[:block_size] = 1.0
        block_guess[block_size:2*block_size] = 0.0
        block_guess[2*block_size:3*block_size] = 1.0
        block_guess[3*block_size:] = 0.0
        guesses.append(block_guess)
        
        return guesses
    
    initial_guesses = generate_initial_guesses(n_points)
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    for guess in initial_guesses:
        # Define objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 1000}
        )
        
        # Check if this result is better
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
