import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Use the provided initial guess
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate diverse initial guesses: original, random, and perturbed
    initial_guesses = [initial_guess.copy()]
    np.random.seed(seed)
    # Add a random initial guess
    random_guess = np.random.uniform(0, 1, n_points)
    random_guess = random_guess * (target_sum / np.sum(random_guess))
    initial_guesses.append(random_guess)
    # Add a few perturbed versions of the initial guess
    for _ in range(2):
        noise = np.random.uniform(-0.2, 0.2, n_points)
        perturbed = initial_guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        initial_guesses.append(perturbed)
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    start_time = time.time()
    
    for guess in initial_guesses:
        # Objective function to minimize C5
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization with COBYLA method for better constraint handling
        result = minimize(

            method='COBYLA',

            constraints=cons,
            options={'maxiter': 200, 'disp': False}
        )
        
        # Check if this result is better
        if result.success and result.fun < best_c5:

            best_h = result.x
