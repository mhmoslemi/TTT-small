from scipy.optimize import minimize
import time

        initial_guesses.append(random_h)
    
    best_c5 = float('inf')
    best_h = initial_guesses[0].copy()
    
    start_time = time.time()
    
    for guess in initial_guesses:
        # Perturb the guess with varying noise scales
        perturbed = guess.copy()
        noise_scale = np.random.uniform(0.1, 0.3)
        noise = np.random.normal(0, noise_scale, n_points)
        perturbed += noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function to minimize C5
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]

        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
