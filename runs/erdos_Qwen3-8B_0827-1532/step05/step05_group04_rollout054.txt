import time

        initial_guesses.append(random_guess)
    
    # Generate sinusoidal perturbations of the initial guess
    for i in range(num_initial_guesses - len(initial_guesses)):
        freq = 2 * np.pi * (i + 1) / n_points

        perturbed = initial_guess + perturbation
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        initial_guesses.append(perturbed)
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    start_time = time.time()
    
    for guess in initial_guesses:
        # Directly use the guess without adding noise
        h = guess.copy()
        
        # Objective function to minimize C5
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization
        result = minimize(

            h,
            method='SLSQP',
            bounds=bounds,

            best_h = result.x
