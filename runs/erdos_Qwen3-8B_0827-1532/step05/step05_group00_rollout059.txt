import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 150  # Use 150 points for balance between speed and precision
    dx = 2.0 / n_points
    
    # Generate symmetric initial guesses
    def generate_symmetric_initial_guess():
        half = n_points // 2
        # Generate first half with random values
        first_half = np.random.uniform(0, 1, half)
        # Mirror them
        symmetric_guess = np.zeros(n_points)
        symmetric_guess[:half] = first_half
        symmetric_guess[half:] = 1 - first_half[::-1]
        # Adjust sum to meet the constraint
        scale_factor = (n_points / 2) / np.sum(symmetric_guess)
        symmetric_guess *= scale_factor
        return symmetric_guess
    
    # Generate multiple initial guesses
    num_initial_guesses = 50
    initial_guesses = []
    for _ in range(num_initial_guesses):
        guess = generate_symmetric_initial_guess()
        initial_guesses.append(guess)
    
    # Objective function
    def objective(h_half):
        # h_half is the first half of the symmetric h_values
        h = np.zeros(n_points)
        h[:n_points//2] = h_half
        h[n_points//2:] = 1 - h_half[::-1]
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function: sum of h must be n_points / 2
    def constraint(h_half):
        h = np.zeros(n_points)
        h[:n_points//2] = h_half
        h[n_points//2:] = 1 - h_half[::-1]
        return np.sum(h) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points // 2)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()

    for guess in initial_guesses:
        # Run optimization
        result = minimize(
            objective,

            bounds=bounds,
