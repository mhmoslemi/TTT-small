    # List of initial guesses to perturb
    initial_guesses = [initial_guess, alternating_guess]
    
    start_time = time.time()
    
    for perturbation_idx in range(num_perturbations):
        # Generate diverse initial guesses: some random, some perturbed
        if perturbation_idx < 5:
            # Random initial guess
            random_guess = np.random.uniform(0, 1, n_points)
            random_guess = random_guess * (target_sum / np.sum(random_guess))
            perturbed = random_guess
        else:
            # Perturb one of the initial guesses
            if perturbation_idx % 2 == 0:
                # Use the alternating guess
                perturbed = alternating_guess.copy()
            else:
                # Use the initial guess
                perturbed = initial_guess.copy()
            # Add noise
            noise = np.random.uniform(-0.1, 0.1, n_points)
            perturbed += noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
        
        # Objective function to minimize C5
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

        result = minimize(
