import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Use the provided initial guess
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate diverse initial guesses
    initial_guesses = [initial_guess.copy()]
    
    # Random guess
    random_guess = np.random.uniform(0, 1, n_points)
    random_guess = random_guess * (target_sum / np.sum(random_guess))
    initial_guesses.append(random_guess)
    
    # Blocky guess with 5 segments
    segments = 5
    block_size = 2 / segments
    h_block = np.zeros(n_points)
    for i in range(segments):
        start = int(i * n_points / segments)
        end = int((i + 1) * n_points / segments)
        h_block[start:end] = 1.0
    current_sum = np.sum(h_block)
    scale = target_sum / current_sum
    h_block = h_block * scale
    h_block = np.clip(h_block, 0, 1)
    initial_guesses.append(h_block)
    
    # Sinusoidal guess
    freq = 2 * np.pi * (1) / n_points
    amplitude = 0.5
    sinusoidal_guess = initial_guess + amplitude * np.sin(freq * np.arange(n_points))
    sinusoidal_guess = np.clip(sinusoidal_guess, 0, 1)
    current_sum = np.sum(sinusoidal_guess)
    scale = target_sum / current_sum
    sinusoidal_guess = sinusoidal_guess * scale
    initial_guesses.append(sinusoidal_guess)
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    start_time = time.time()
    
    for guess in initial_guesses:
        # Perform optimization
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-8, 'maxiter': 500}
        )
        
        # Check if this result is better
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
