        return np.sum(h) - n_points / 2
    
    # Define bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Generate initial guesses based on structured functions
    initial_guesses = []
    
    # Structured initial guess functions
    def square_wave(x):
        return np.where(x < 1, 1, 0)
    
    def triangular_wave(x):
        t = x - 1
        return 1 - np.abs(t)
    
    def two_intervals(x):
        h = np.zeros_like(x)
        idx = np.where(x < 0.5)
        h[idx] = 1
        idx = np.where(x >= 1.5)
        h[idx] = 1
        return h
    
    # Add structured initial guesses
    for func in [square_wave, triangular_wave, two_intervals]:
        x = np.linspace(0, 2, n_points)
        h = func(x)
        initial_guesses.append(h)
    
    # Also add the initial_h_values interpolated to n_points=200
    x_initial = np.linspace(0, 2, len(initial_h_values))
    x_new = np.linspace(0, 2, n_points)
    initial_guess = np.interp(x_new, x_initial, initial_h_values)
    scale_factor = (n_points / 2) / np.sum(initial_guess)
    initial_guess *= scale_factor
    initial_guesses.append(initial_guess)
    
    # Generate more initial guesses by adding noise to the structured ones
    num_initial_guesses = 10
    for _ in range(num_initial_guesses):
        base_guess = initial_guesses[np.random.choice(len(initial_guesses))]
        noise_amp = np.random.uniform(0.005, 0.02)
        noise_freq = np.random.uniform(1, 5)
        noise = noise_amp * np.sin(2 * np.pi * noise_freq * np.linspace(0, 2, n_points))
        perturbed = base_guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        scale_factor = (n_points / 2) / np.sum(perturbed)
        perturbed *= scale_factor
        initial_guesses.append(perturbed)
    
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    
    for guess in initial_guesses:
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-9, 'maxiter': 1500}
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    if best_h is None:
        result = minimize(
            objective,
            initial_guesses[0],
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-9, 'maxiter': 1500}
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points
