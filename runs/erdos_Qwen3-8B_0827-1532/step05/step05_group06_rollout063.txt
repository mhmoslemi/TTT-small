    for i in range(2):
        start = np.random.randint(0, n_points - block_size)
        block_initial[start:start+block_size] = 1
    block_initial = np.clip(block_initial, 0, 1)
    sum_block = np.sum(block_initial)
    if sum_block != 0:
        block_initial = block_initial * (n_points / 2) / sum_block
    initial_guesses.append(block_initial)
    # 7. Random sparse initial guess
    sparse_initial = np.zeros(n_points)
    num_ones = np.random.randint(10, 15)
    indices = np.random.choice(n_points, num_ones, replace=False)
    sparse_initial[indices] = 1
    sum_sparse = np.sum(sparse_initial)
    if sum_sparse != 0:
        sparse_initial = sparse_initial * (n_points / 2) / sum_sparse
    initial_guesses.append(sparse_initial)
    
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    
    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    for guess in initial_guesses:
        # Perturb the guess slightly to avoid local optima
        perturb = np.random.normal(0, 0.01, n_points)
        perturbed_guess = np.clip(guess + perturb, 0, 1)
        sum_perturbed = np.sum(perturbed_guess)
        if sum_perturbed != 0:
            perturbed_guess = perturbed_guess * (n_points / 2) / sum_perturbed
        # Optimize
        result = minimize(
            objective,
            perturbed_guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraint,
            options={'maxiter': 2000, 'ftol': 1e-9, 'eps': 1e-9}
        )
        current_h = result.x
        current_c5 = result.fun
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h
        # Check time
        if time.time() - start_time > budget_s:
            break
    
    return (best_h, best_c5, n_points)
