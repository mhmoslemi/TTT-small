        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        result_slsqp = minimize(
            objective_refine,
            result_de.x,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 200}
        )
        
        # Update best solution
        if result_slsqp.fun < best_c5:
            best_c5 = result_slsqp.fun
            best_h = result_slsqp.x
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
