import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate diverse initial guesses with random noise
    num_initial_guesses = 50
    initial_guesses = [initial_guess.copy()]
    
    for i in range(num_initial_guesses - 1):
        noise = np.random.normal(0, 0.05, n_points)
        perturbed = initial_guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        initial_guesses.append(perturbed)
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    start_time = time.time()
    
    for guess in initial_guesses:
        # Add small noise for exploration
        noise = np.random.normal(0, 0.02, n_points)
        perturbed = guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function to minimize C5
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Use L-BFGS-B for faster convergence with bounds
        result = minimize(
            objective,
            perturbed,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 500}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
