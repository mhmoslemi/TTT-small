from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 150  # Increased for better precision
    dx = 2.0 / n_points
    
    # Generate initial guess via interpolation
    x_initial = np.linspace(0, 2, len(initial_h_values))
    x_new = np.linspace(0, 2, n_points)
    initial_guess = np.interp(x_new, x_initial, initial_h_values)
    # Adjust sum to meet the constraint
    initial_guess = initial_guess / np.sum(initial_guess) * (n_points / 2)
    
    # Generate diverse initial guesses with multiple perturbation types
    num_initial_guesses = 30
    initial_guesses = []
    for _ in range(num_initial_guesses):
        # Random noise perturbation
        noise_amp = np.random.uniform(0.005, 0.02)
        noise = noise_amp * np.random.rand(n_points)
        perturbed = initial_guess + noise
        # Clip and scale
        perturbed = np.clip(perturbed, 0, 1)
        scale_factor = (n_points / 2) / np.sum(perturbed)
        perturbed *= scale_factor
        initial_guesses.append(perturbed)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Run optimizations for each initial guess with tighter tolerances
    for guess in initial_guesses:
        # Run optimization with trust-constr for better handling of constraints
        result = minimize(
            objective,
            guess,
            method='trust-constr',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 200, 'gtol': 1e-8, 'eps': 1e-8}
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.success:
            current_c5 = result.fun

                best_c5 = current_c5

    # If no optimization was done, use the initial guess

        result = minimize(

            initial_guess,

        )
        best_c5 = result.fun
        best_h = result.x
