import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Determine n_points from initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate initial guesses
    initial_guesses = [initial_h_values.copy()]
    np.random.seed(seed)
    for _ in range(15):  # Generate 15 random initial guesses
        h = np.random.uniform(0, 1, n_points)
        h_sum = np.sum(h)
        scale = target_sum / h_sum
        h = h * scale
        h = np.clip(h, 0, 1)
        initial_guesses.append(h)
    
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    start_time = time.time()
    
    for guess in initial_guesses:
        # Optimization setup
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 1000}
        )
        
        # Check if successful and better
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
