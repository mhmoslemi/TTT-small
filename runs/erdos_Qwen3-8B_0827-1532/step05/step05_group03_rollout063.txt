import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Use the provided initial guess
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate multiple random initial guesses
    num_initial_guesses = 15  # Reduced to balance exploration and time
    initial_guesses = [initial_guess]
    for _ in range(num_initial_guesses):
        h = np.random.uniform(0, 1, n_points)
        h = h * (target_sum / np.sum(h))
        initial_guesses.append(h)
    
    best_c5 = float('inf')
    best_h = initial_guesses[0]
    
    start_time = time.time()
    
    for guess in initial_guesses:
        # Optimize this guess
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='L-BFGS-B',  # Faster for bound-constrained problems
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 500}  # Reduced iterations for speed
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
