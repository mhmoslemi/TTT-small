import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    # Use the provided initial guess
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate diverse initial guesses
    num_perturbations = 30  # Increased number of perturbations for better exploration
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Define objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define constraints: sum(h) == target_sum
    def constraint(h):
        return np.sum(h) - target_sum
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Function to generate diverse initial guesses
    def generate_initial_guesses(num_guesses):
        guesses = []
        # Random initial guess
        for _ in range(num_guesses // 2):
            guess = np.random.uniform(0, 1, n_points)
            guess = guess * (target_sum / np.sum(guess))
            guesses.append(guess)
        # Perturbed initial guess
        for _ in range(num_guesses // 2):
            perturbed = initial_guess + np.random.uniform(-0.1, 0.1, n_points)
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
            guesses.append(perturbed)
        return guesses
    
    # Generate initial guesses
    initial_guesses = generate_initial_guesses(num_perturbations)
    
    # Optimize each initial guess
    for i, guess in enumerate(initial_guesses):
        # Use differential evolution for global search
        result_de = differential_evolution(
            objective,
            bounds,
            constraints={"type": "eq", "fun": constraint},
            strategy='best1bin',
            popsize=15,
            maxiter=50,
            tol=1e-8,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed + i
        )
        if result_de.success:
            # Refine with SLSQP
            result_slsqp = minimize(
                objective,
                result_de.x,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint},
                options={'ftol': 1e-9, 'maxiter': 200}
            )
            if result_slsqp.success and result_slsqp.fun < best_c5:
                best_c5 = result_slsqp.fun
                best_h = result_slsqp.x
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
