            options={'maxiter': 1500, 'ftol': 1e-9, 'eps': 1e-9}
        )
        return result.x, result.fun
    
    # Track best result
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    # Use multiprocessing to parallelize the optimization
    with Pool() as pool:
        # Limit the number of initial guesses to avoid exceeding time
        # For example, take the first 20 guesses
        limited_guesses = initial_guesses[:20]
        results = pool.map(optimize_guess, limited_guesses)
        for h_candidate, c5_candidate in results:
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_candidate
            # Check time
            if time.time() - start_time > budget_s:
                break
    
    return (best_h, best_c5, n_points)
