from scipy.optimize import minimize
from scipy.signal import correlate
import time
from multiprocessing import Pool, cpu_count

def run(seed=42, budget_s=100, **kwargs):
    n_points = 100  # Increased for better resolution
    dx = 2.0 / n_points
    np.random.seed(seed)
    
    # Generate diverse initial guesses
    initial_guesses = []
    # 1. Use the provided initial_h_values
    initial_guesses.append(initial_h_values)
    # 2. Evenly spaced binary initial guess
    evenly_spaced_initial = np.zeros(n_points)
    for i in range(n_points // 2):
        pos = int(i * n_points / (n_points // 2))
        evenly_spaced_initial[pos] = 1
        evenly_spaced_initial[n_points - 1 - pos] = 1
    initial_guesses.append(evenly_spaced_initial)
    # 3. Random binary initial guess

    random_binary[indices] = 1
    initial_guesses.append(random_binary)
    # 4. Symmetric binary initial guess
    symmetric_binary = np.zeros(n_points)
    for i in range(n_points // 2):
        symmetric_binary[i] = 1
        symmetric_binary[n_points - 1 - i] = 1
    initial_guesses.append(symmetric_binary)
    # 5. Perturbations of initial_h_values

        sum_perturbed = np.sum(perturbed)
        if sum_perturbed != 0:
            perturbed = perturbed * (n_points / 2) / sum_perturbed
        initial_guesses.append(perturbed)
    # 6. Additional initial guesses: ternary pattern
    ternary_initial = np.zeros(n_points)
    for i in range(n_points):
        if i % 3 == 0:
            ternary_initial[i] = 1
        elif i % 3 == 1:
            ternary_initial[i] = 0.5
        else:
            ternary_initial[i] = 0
    initial_guesses.append(ternary_initial)
    # 7. Additional initial guesses: alternating pattern
    alternating_initial = np.zeros(n_points)
    for i in range(n_points):
        if i % 2 == 0:
            alternating_initial[i] = 1
        else:
            alternating_initial[i] = 0
    initial_guesses.append(alternating_initial)
    
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    
    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def optimize_perturbed_guess(perturbed_guess):
        # Optimize with L-BFGS-B method for better convergence
        result = minimize(

        return result.x, result.fun
