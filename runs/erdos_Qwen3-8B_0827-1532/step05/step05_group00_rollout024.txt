import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = 150
    dx = 2.0 / n_points
    
    # Generate symmetric initial guess
    half = n_points // 2
    first_half = np.random.uniform(0, 1, half)
    mirrored = first_half[::-1]
    symmetric_guess = np.concatenate([first_half, mirrored])
    current_sum = np.sum(symmetric_guess)
    scale_factor = (n_points / 2) / current_sum
    symmetric_guess *= scale_factor
    
    # Generate multiple initial guesses by adding noise
    num_initial_guesses = 50
    initial_guesses = []
    for _ in range(num_initial_guesses):
        # Add random noise
        noise = np.random.normal(0, 0.01, n_points)
        perturbed = symmetric_guess + noise
        # Clip to [0, 1]
        perturbed = np.clip(perturbed, 0, 1)
        # Adjust sum to meet the constraint
        scale_factor = (n_points / 2) / np.sum(perturbed)
        perturbed *= scale_factor
        initial_guesses.append(perturbed)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()

    for guess in initial_guesses:
        # Run optimization
        result = minimize(

            method='trust-constr',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},

        )
