import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Balance between speed and resolution
    dx = 2.0 / n_points
    
    # Generate initial guesses
    initial_guesses = []
    
    # Uniform initial guess
    uniform_guess = 0.5 * np.ones(n_points)
    initial_guesses.append(uniform_guess)
    
    # Interpolated initial_h_values
    x_initial = np.linspace(0, 2, len(initial_h_values))
    x_new = np.linspace(0, 2, n_points)
    interpolated_guess = np.interp(x_new, x_initial, initial_h_values)
    initial_guesses.append(interpolated_guess)
    
    # Generate perturbations
    for base_guess in [uniform_guess, interpolated_guess]:
        for _ in range(20):  # Random noise perturbations
            noise_amp = np.random.uniform(0.005, 0.02)
            noise = noise_amp * np.random.randn(n_points)
            perturbed = base_guess + noise
            perturbed = np.clip(perturbed, 0, 1)
            scale_factor = (n_points / 2) / np.sum(perturbed)
            perturbed *= scale_factor
            initial_guesses.append(perturbed)
    
    for base_guess in [uniform_guess, interpolated_guess]:
        for _ in range(10):  # Sine wave perturbations
            noise_amp = np.random.uniform(0.005, 0.02)
            noise_freq = np.random.uniform(1, 5)
            noise = noise_amp * np.sin(2 * np.pi * noise_freq * x_new)
            perturbed = base_guess + noise
            perturbed = np.clip(perturbed, 0, 1)
            scale_factor = (n_points / 2) / np.sum(perturbed)
            perturbed *= scale_factor
            initial_guesses.append(perturbed)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    
    # Run optimizations for each initial guess
    for guess in initial_guesses:
        # Run optimization with trust-constr
        result = minimize(
            objective,
            guess,
            method='trust-constr',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-9, 'maxiter': 1000}
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # If no optimization was done, use the uniform guess
    if best_h is None:
        result = minimize(
            objective,
            uniform_guess,
            method='trust-constr',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-9, 'maxiter': 1000}
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points
