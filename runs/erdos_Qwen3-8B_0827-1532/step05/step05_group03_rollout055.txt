from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate diverse initial guesses
    def generate_initial_guesses():
        # Uniform guess
        uniform = np.full(n_points, 0.5)
        # Half 1s and half 0s
        half_ones = np.concatenate([np.ones(n_points // 2), np.zeros(n_points - n_points // 2)])
        # Random guess
        random = np.random.uniform(0, 1, n_points)
        random_sum = np.sum(random)
        random = random * (target_sum / random_sum)
        # Include provided initial guess
        return [uniform, half_ones, random, initial_guess]
    
    initial_guesses = generate_initial_guesses()
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    for guess in initial_guesses:
        # Ensure guess is within [0,1] and sum constraint
        guess = np.clip(guess, 0, 1)
        current_sum = np.sum(guess)
        scale = target_sum / current_sum
        guess = guess * scale
        
        # Objective function to minimize C5
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization with tighter tolerances and more iterations
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,

            options={'ftol': 1e-9, 'maxiter': 1000}
        )
