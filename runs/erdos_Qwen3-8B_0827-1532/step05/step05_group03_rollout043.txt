from scipy.optimize import minimize, differential_evolution

    # Function to evaluate C5 for a given h
    def evaluate_c5(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Constraints: sum(h) == target_sum
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    
    # Function to optimize with SLSQP
    def optimize_with_slsqp(perturbed):
        result = minimize(
            evaluate_c5,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 1500}
        )
        if result.success:
            return result.x, result.fun
        return perturbed, float('inf')

        perturbed = initial_guess + noise

        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum

        # Optimize

        if c5_opt < best_c5:
