    sinusoidal = np.sin(np.linspace(0, np.pi, n_points)) * 0.5 + 0.5

    # Example: piecewise constant with varying intervals
    piecewise = np.zeros(n_points)

    initial_guesses.append(piecewise)
    
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    
    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Function to run optimization for a single guess
    def optimize_single(guess):
        # Perturb the guess slightly to avoid local optima
        perturb = np.random.normal(0, 0.01, n_points)

        sum_perturbed = np.sum(perturbed_guess)

        # Optimize
        result = minimize(
