from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    dx = 2.0 / n_points

    initial_guesses = []

    interpolated = interpolated / np.sum(interpolated) * (n_points / 2)
    initial_guesses.append(interpolated)
    
    # Random guesses
    num_random_guesses = 10
    for _ in range(num_random_guesses):
        random_guess = np.random.rand(n_points)
        random_guess = random_guess / np.sum(random_guess) * (n_points / 2)
        initial_guesses.append(random_guess)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
