    # Fill remaining with random individuals
    while len(population) < population_size:
        random_individual = np.random.rand(n_points)
        sum_rand = np.sum(random_individual)
        if sum_rand != 0:
            random_individual = random_individual * (n_points / 2) / sum_rand
        population.append(random_individual)
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        # Evaluate all individuals
        evaluations = []
        for h in population:
            corr = correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            evaluations.append(max_corr * dx)
        # Update best solution
        best_idx = np.argmin(evaluations)
        best_c5_current = evaluations[best_idx]
        if best_c5_current < best_c5:
            best_c5 = best_c5_current
            best_h = population[best_idx].copy()
        # Select parents (top 20%)
        num_parents = int(0.2 * population_size)
        parent_indices = np.argsort(evaluations)[:num_parents]
        parents = [population[i] for i in parent_indices]
        # Generate offspring
        offspring = []
        for _ in range(population_size - num_parents):
            # Crossover
            parent1 = np.random.choice(parents)
            parent2 = np.random.choice(parents)
            child = (parent1 + parent2) / 2
            sum_child = np.sum(child)
            if sum_child == 0:
                child = np.zeros_like(child)
            else:
                scaling = (n_points / 2) / sum_child
                child = child * scaling
                child = np.clip(child, 0, 1)
            # Mutation
            noise = np.random.normal(0, 0.05, child.shape)
            mutated = child + noise
            mutated = np.clip(mutated, 0, 1)
            sum_mutated = np.sum(mutated)
            if sum_mutated == 0:
                mutated = np.zeros_like(mutated)
            else:
                scaling = (n_points / 2) / sum_mutated
                mutated = mutated * scaling
            offspring.append(mutated)
        # Replace population
        population = parents + offspring
    
    return (best_h, best_c5, n_points)
