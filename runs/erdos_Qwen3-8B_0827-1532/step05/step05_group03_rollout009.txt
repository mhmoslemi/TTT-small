import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Use the provided initial guess
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate multiple perturbations
    num_perturbations = 30  # Increased number of perturbations for better exploration
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    for perturbation_idx in range(num_perturbations):
        # Generate more diverse perturbations with varying noise scales
        if perturbation_idx < 10:
            # Larger noise for early perturbations to explore more
            noise = np.random.uniform(-0.3, 0.3, n_points)
        else:
            # Smaller noise for later perturbations to refine
            noise = np.random.uniform(-0.1, 0.1, n_points)
        
        perturbed = initial_guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function to minimize C5 with penalty for sum constraint
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            # Penalty term for sum constraint
            penalty = 1e4 * (np.sum(h) - target_sum)**2
            return max_corr * dx + penalty
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Perform optimization with L-BFGS-B method
        result = minimize(
            objective,
            perturbed,
            method='L-BFGS-B',
            bounds=bounds,
            options={'ftol': 1e-9, 'maxiter': 200}
        )
        
        # Check if this result is better
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
