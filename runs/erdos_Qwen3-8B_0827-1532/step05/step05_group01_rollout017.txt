        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Bounds for the first n_points-1 variables
    bounds = [(0.0, 1.0) for _ in range(n_points - 1)]
    
    start_time = time.time()
    
    # Run differential evolution with optimized parameters
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed,
        disp=False
    )
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        # Extract best solution found within time
        best_params = result.x
        last = target_sum - np.sum(best_params[:-1])
        last = np.clip(last, 0, 1)
        best_h = np.append(best_params[:-1], last)
        return best_h, result.fun, n_points
    
    # Final solution after optimization
    best_params = result.x
    last = target_sum - np.sum(best_params[:-1])
    last = np.clip(last, 0, 1)
    best_h = np.append(best_params[:-1], last)
    return best_h, result.fun, n_points
