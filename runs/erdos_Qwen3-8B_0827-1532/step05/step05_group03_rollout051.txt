import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Use the provided initial guess
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate multiple diverse initial guesses
    num_initial_guesses = 30  # Increased number of initial guesses
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Precompute random seed for reproducibility
    np.random.seed(seed)
    
    for guess_idx in range(num_initial_guesses):
        # Generate diverse initial guesses: mix of perturbations and random
        if guess_idx < 10:
            # Perturbations of initial guess with varying noise scales
            noise = np.random.uniform(-0.2, 0.2, n_points)
            perturbed = initial_guess + noise
        else:
            # Random initial guess with sum constraint
            random_vals = np.random.uniform(0, 1, n_points)
            random_vals = random_vals * (target_sum / np.sum(random_vals))
            perturbed = random_vals
        
        perturbed = np.clip(perturbed, 0, 1)
        
        # Objective function to minimize C5
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization with tighter tolerances and more iterations
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 1000}
        )
        
        # Check if this result is better
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
