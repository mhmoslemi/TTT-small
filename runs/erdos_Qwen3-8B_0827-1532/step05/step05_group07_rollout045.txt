import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = 80
    dx = 2.0 / n_points
    np.random.seed(seed)
    
    # Generate diverse initial guesses
    initial_guesses = []
    # 1. Use the provided initial_h_values
    initial_guesses.append(initial_h_values)
    # 2. Evenly spaced binary initial guess
    evenly_spaced_initial = np.zeros(n_points)
    for i in range(n_points // 2):
        pos = int(i * n_points / (n_points // 2))
        evenly_spaced_initial[pos] = 1
        evenly_spaced_initial[n_points - 1 - pos] = 1
    initial_guesses.append(evenly_spaced_initial)
    # 3. Random binary initial guess
    random_binary = np.zeros(n_points)
    indices = np.random.choice(n_points, size=n_points//2, replace=False)
    random_binary[indices] = 1
    initial_guesses.append(random_binary)
    # 4. Symmetric binary initial guess
    symmetric_binary = np.zeros(n_points)
    for i in range(n_points // 2):
        symmetric_binary[i] = 1
        symmetric_binary[n_points - 1 - i] = 1
    initial_guesses.append(symmetric_binary)
    # 5. Random non-binary initial guess
    random_nonbinary = np.random.uniform(0, 1, n_points)
    sum_rand = np.sum(random_nonbinary)
    random_nonbinary = random_nonbinary * (n_points / 2) / sum_rand
    initial_guesses.append(random_nonbinary)
    # 6. Binary clusters initial guess
    binary_clusters = np.zeros(n_points)
    for i in range(3):
        pos = int(i * n_points / 3)
        binary_clusters[pos] = 1
        binary_clusters[n_points - 1 - pos] = 1
    initial_guesses.append(binary_clusters)
    # 7. Perturbations of initial_h_values
    for _ in range(5):  # More perturbations
        perturb = np.random.normal(0, 0.02, n_points)
        perturbed = initial_h_values + perturb
        perturbed = np.clip(perturbed, 0, 1)
        sum_perturbed = np.sum(perturbed)
        if sum_perturbed != 0:
            perturbed = perturbed * (n_points / 2) / sum_perturbed
        initial_guesses.append(perturbed)
    
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    
    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    for guess in initial_guesses:
        # Perturb the guess slightly to avoid local optima
        perturb = np.random.normal(0, 0.01, n_points)
        perturbed_guess = np.clip(guess + perturb, 0, 1)
        sum_perturbed = np.sum(perturbed_guess)
        if sum_perturbed != 0:
            perturbed_guess = perturbed_guess * (n_points / 2) / sum_perturbed
        # Optimize
        result = minimize(
            objective,
            perturbed_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=constraint,
            options={'maxiter': 2000, 'ftol': 1e-10, 'eps': 1e-9}
        )
        current_h = result.x
        current_c5 = result.fun
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h
        # Check time

            break
    
    return (best_h, best_c5, n_points)
