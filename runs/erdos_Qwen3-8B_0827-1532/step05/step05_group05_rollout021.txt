import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate multiple initial guesses: original, structured, and random
    num_initial_guesses = 10  # Reduced to balance exploration and time
    initial_guesses = [initial_guess.copy()]
    
    # Generate structured guess: evenly spaced binary sequence
    structured_guess = np.zeros(n_points)
    for i in range(n_points):
        if i % 2 == 0:  # For n_points=100, this gives 50 ones
            structured_guess[i] = 1.0
    initial_guesses.append(structured_guess.copy())
    
    # Generate random initial guesses
    for _ in range(num_initial_guesses - 2):  # Subtract the two already added
        # Generate random initial guess with sum constraint
        random_guess = np.random.rand(n_points)
        random_guess /= np.sum(random_guess)
        random_guess *= target_sum
        initial_guesses.append(random_guess)
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    start_time = time.time()
    
    for guess in initial_guesses:
        # Use this guess as the initial population for differential evolution
        bounds = [(0.0, 1.0) for _ in range(n_points)]

            max_corr = np.max(corr)
            return max_corr * dx
        
        # Run differential evolution with adjusted parameters for speed
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=15,
            mutation=(0.5, 1),
