    random_continuous = np.random.uniform(0, 1, n_points)
    sum_random = np.sum(random_continuous)
    random_continuous = random_continuous * (n_points / 2) / sum_random
    initial_guesses.append(random_continuous)
    # 6. Another random continuous initial guess
    random_continuous2 = np.random.uniform(0, 1, n_points)
    sum_random2 = np.sum(random_continuous2)
    random_continuous2 = random_continuous2 * (n_points / 2) / sum_random2
    initial_guesses.append(random_continuous2)
    # 7. Perturbations of initial_h_values
    for _ in range(5):  # More perturbations
        perturb = np.random.normal(0, 0.02, n_points)
        perturbed = initial_h_values + perturb
        perturbed = np.clip(perturbed, 0, 1)
        sum_perturbed = np.sum(perturbed)
        if sum_perturbed != 0:
            perturbed = perturbed * (n_points / 2) / sum_perturbed
        initial_guesses.append(perturbed)
    
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    
    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Function to run optimization for a single guess
    def optimize_single(guess):
        # Perturb the guess slightly to avoid local optima
        perturb = np.random.normal(0, 0.01, n_points)
        perturbed_guess = np.clip(guess + perturb, 0, 1)
        sum_perturbed = np.sum(perturbed_guess)
        if sum_perturbed != 0:
            perturbed_guess = perturbed_guess * (n_points / 2) / sum_perturbed
        # Optimize
        result = minimize(
            objective,
            perturbed_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=constraint,
            options={'maxiter': 1000, 'ftol': 1e-8, 'eps': 1e-8}
        )
        current_h = result.x
        current_c5 = result.fun
        return (current_c5, current_h)
    
    # Use multiprocessing to run optimizations in parallel
    num_cpus = min(cpu_count(), len(initial_guesses))
    with Pool(num_cpus) as pool:
        results = pool.map(optimize_single, initial_guesses)
    
    # Find the best result
    for c5, h in results:
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    
    # Check time and return the best solution
    if time.time() - start_time > budget_s:
        # If time is up, return the best found so far
        return (best_h, best_c5, n_points)
    else:
        # If time is not up, maybe do more? But for simplicity, return the best
        return (best_h, best_c5, n_points)
