    structured_guess = structured_guess * (target_sum / np.sum(structured_guess))
    
    # Add structured guess to the list of initial guesses
    initial_guesses = [initial_guess, structured_guess]
    
    # Generate random initial guesses
    for _ in range(num_perturbations - 2):
        random_guess = np.random.uniform(0, 1, n_points)

        initial_guesses.append(random_guess)
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]

        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

        def objective(h):

            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx

        result = minimize(
