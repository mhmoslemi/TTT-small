    # Use multiprocessing to run optimizations in parallel
    with Pool(processes=10) as pool:  # Using 10 CPUs
        results = []
        for guess in initial_guesses:
            # Submit the task to the pool
            results.append(pool.apply_async(optimize_guess, (guess,)))
        
        # Collect results as they come in
        while results:
            # Check if time is up
            elapsed = time.time() - start_time
            if elapsed >= budget_s:
                break
            # Get the next result
            result = results.pop(0)
            try:
                current_c5, current_h = result.get(timeout=0.1)  # Timeout to avoid blocking
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h
            except:
                # Handle any exceptions, maybe the task is still running
                pass
    
    # If no optimization was done, use the initial guess
    if best_h is None:
        # Use the first initial guess
        result = minimize(
            objective,
            initial_guesses[0],
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-9, 'maxiter': 1000}
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points
