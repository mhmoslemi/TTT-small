    structured_guess = np.zeros(n_points)
    for i in range(n_points):

    initial_guesses.append(structured_guess.copy())
    
    # Generate structured guess: blocks of 1s and 0s
    block_guess = np.zeros(n_points)
    block_size = n_points // 4  # Block size of 25 for n_points=100

    initial_guesses.append(block_guess.copy())
    
    # Generate random initial guesses
    for _ in range(num_initial_guesses - 2):  # Subtract the two already added
        # Generate random initial guess with sum constraint
        random_guess = np.random.rand(n_points)
        random_guess /= np.sum(random_guess)
        random_guess *= target_sum
        initial_guesses.append(random_guess)
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    start_time = time.time()
    
    for guess in initial_guesses:
        # Perturb the guess with small noise
        noise = np.random.normal(0, 0.02, n_points)
        perturbed = guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function to minimize C5
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization
        result = minimize(
            objective,
            perturbed,
            method='COBYLA',  # Changed to COBYLA
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 1000}  # Adjusted parameters
        )
        
        # Check if this result is better
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
