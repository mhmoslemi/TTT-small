from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    for guess in initial_guesses:
        # Run optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
