import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Generate initial guesses with random sampling and scaling
    num_initial_guesses = 30

    for _ in range(num_initial_guesses):

        penalty = 1e6 * (np.sum(h) - n_points / 2)**2
        return max_corr * dx + penalty
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Run differential evolution with multiple initial populations
    for i in range(3):
        # Use a subset of initial guesses as starting points
        if i == 0:
            initial_population = initial_guesses[:10]
        elif i == 1:
            initial_population = initial_guesses[10:20]
        else:
            initial_population = initial_guesses[20:]
        
        # Run differential evolution with the current initial population
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=100,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed + i,
            polish=True,
            disp=False
        )
        
        elapsed = time.time() - start_time

        # Update best solution
        current_c5 = objective(result.x)
        if current_c5 < best_c5:

    # If no solution found, use the first initial guess
    if best_h is None:
        best_h = initial_guesses[0]
        best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
