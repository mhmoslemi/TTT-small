from scipy.optimize import minimize
import time

    n_points = len(initial_guess)

    initial_guesses = [initial_guess.copy()]
    
    # Generate structured guess: evenly spaced binary sequence
    structured_guess = np.zeros(n_points)
    for i in range(n_points):
        if i % 3 == 0:  # For n_points=100, this gives ~33 ones
            structured_guess[i] = 1.0
    initial_guesses.append(structured_guess.copy())
    
    # Generate random initial guesses
    for _ in range(num_initial_guesses - 2):  # Subtract the two already added
        # Generate random initial guess with sum constraint
        random_guess = np.random.rand(n_points)
        random_guess /= np.sum(random_guess)
        random_guess *= target_sum
        initial_guesses.append(random_guess)
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    start_time = time.time()
    
    for guess in initial_guesses:
        # Perturb the guess with small noise
        noise = np.random.normal(0, 0.015, n_points)
        perturbed = guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function to minimize C5
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization with trust-constr method
        result = minimize(
            objective,
            perturbed,
            method='trust-constr',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 200, 'verbose': 0}  # Adjusted parameters
        )

        if result.success and result.fun < best_c5:
