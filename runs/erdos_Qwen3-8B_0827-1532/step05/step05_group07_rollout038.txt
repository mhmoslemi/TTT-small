import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
from multiprocessing import Pool, cpu_count

def run(seed=42, budget_s=100, **kwargs):
    n_points = 100  # Increased resolution for better precision
    dx = 2.0 / n_points
    np.random.seed(seed)
    
    # Generate diverse initial guesses
    initial_guesses = []
    # 1. Use the provided initial_h_values
    initial_guesses.append(initial_h_values)
    # 2. Symmetric binary initial guess
    symmetric_binary = np.zeros(n_points)
    for i in range(n_points // 2):
        symmetric_binary[i] = 1
        symmetric_binary[n_points - 1 - i] = 1
    initial_guesses.append(symmetric_binary)
    # 3. Evenly spaced binary initial guess
    evenly_spaced_initial = np.zeros(n_points)
    for i in range(n_points // 2):
        pos = int(i * n_points / (n_points // 2))
        evenly_spaced_initial[pos] = 1
        evenly_spaced_initial[n_points - 1 - pos] = 1
    initial_guesses.append(evenly_spaced_initial)
    # 4. Perturbations of initial_h_values
    for _ in range(3):  # Fewer perturbations to save time
        perturb = np.random.normal(0, 0.01, n_points)
        perturbed = initial_h_values + perturb
        perturbed = np.clip(perturbed, 0, 1)
        sum_perturbed = np.sum(perturbed)
        if sum_perturbed != 0:
            perturbed = perturbed * (n_points / 2) / sum_perturbed
        initial_guesses.append(perturbed)
    
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    
    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def optimize_single(guess):
        # Perturb the guess slightly to avoid local optima
        perturb = np.random.normal(0, 0.01, n_points)
        perturbed_guess = np.clip(guess + perturb, 0, 1)
        sum_perturbed = np.sum(perturbed_guess)
        if sum_perturbed != 0:
            perturbed_guess = perturbed_guess * (n_points / 2) / sum_perturbed
        # Optimize
        result = minimize(
            objective,
            perturbed_guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraint,
            options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-9}
        )
        current_h = result.x
        current_c5 = result.fun
        return (current_c5, current_h)
    
    # Use multiprocessing to parallelize optimization
    with Pool(processes=min(cpu_count(), len(initial_guesses))) as pool:
        results = pool.map(optimize_single, initial_guesses)

            best_c5 = c5
            best_h = h
