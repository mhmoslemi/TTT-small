        perturb = np.random.normal(0, 0.02, n_points)
        perturbed = initial_h_values + perturb
        perturbed = np.clip(perturbed, 0, 1)
        sum_perturbed = np.sum(perturbed)
        if sum_perturbed != 0:
            perturbed = perturbed * (n_points / 2) / sum_perturbed
        initial_guesses.append(perturbed)
    # 6. Random initial guess with some structure (e.g., sine wave)
    random_sine = np.sin(np.linspace(0, np.pi, n_points)) * 0.5 + 0.5
    random_sine = np.clip(random_sine, 0, 1)
    sum_sine = np.sum(random_sine)
    if sum_sine != 0:
        random_sine = random_sine * (n_points / 2) / sum_sine
    initial_guesses.append(random_sine)
    # 7. Another random initial guess
    random_initial = np.random.rand(n_points)
    sum_rand = np.sum(random_initial)
    if sum_rand != 0:
        random_initial = random_initial * (n_points / 2) / sum_rand
    initial_guesses.append(random_initial)
    
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    
    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def optimize_guess(guess):
        # Perturb the guess slightly to avoid local optima
        perturb = np.random.normal(0, 0.01, n_points)
        perturbed_guess = np.clip(guess + perturb, 0, 1)
        sum_perturbed = np.sum(perturbed_guess)
        if sum_perturbed != 0:
            perturbed_guess = perturbed_guess * (n_points / 2) / sum_perturbed
        # Optimize
        result = minimize(
            objective,
            perturbed_guess,
            method='Nelder-Mead',
            bounds=bounds,
            constraints=constraint,
            options={'maxiter': 200, 'xatol': 1e-6, 'fatol': 1e-6}
        )
        current_h = result.x
        current_c5 = result.fun
        return (current_c5, current_h)
    
    # Create a pool of workers
    with Pool(processes=cpu_count()) as pool:
        results = pool.map(optimize_guess, initial_guesses)
    
    # Find the best result
    for c5, h in results:
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    
    # Check time
    if time.time() - start_time > budget_s:
        # If time is up, return the best found so far
        return (best_h, best_c5, n_points)
    else:
        # If time is not up, maybe do more optimizations? But the budget is tight.
        return (best_h, best_c5, n_points)
