from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(initial_guess)

    target_sum = n_points / 2
    
    # Generate multiple initial guesses: original, structured, and random
    num_initial_guesses = 20  # Increased number of initial guesses
    initial_guesses = [initial_guess.copy()]
    
    # Generate structured guess: alternating 0s and 1s
    structured_guess = np.zeros(n_points)
    for i in range(n_points):
        if i % 2 == 0:
            structured_guess[i] = 1.0
    initial_guesses.append(structured_guess.copy())
    
    # Generate block pattern: first half 1s, second half 0s
    block_guess = np.zeros(n_points)
    block_guess[:n_points//2] = 1.0
    initial_guesses.append(block_guess.copy())
    
    # Generate random initial guesses
    for _ in range(num_initial_guesses - 3):  # Subtract the three already added
        # Generate random initial guess with sum constraint
        random_guess = np.random.rand(n_points)
        random_guess /= np.sum(random_guess)
        random_guess *= target_sum
        initial_guesses.append(random_guess)
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    start_time = time.time()
    
    for guess in initial_guesses:
        # Perturb the guess with larger noise
        noise = np.random.normal(0, 0.05, n_points)
        perturbed = guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function to minimize C5
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization with COBYLA method
        result = minimize(
            objective,
            perturbed,
            method='COBYLA',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 1000}  # Adjusted parameters
        )

        if elapsed >= budget_s:
