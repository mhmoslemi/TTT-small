        perturbed[segment_idx] = np.clip(perturbed[segment_idx], 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        def objective_local(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds_local = [(0.0, 1.0) for _ in range(n_points)]
        cons_local = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result_local = minimize(
            objective_local,
            perturbed,
            method='SLSQP',
            bounds=bounds_local,
            constraints=cons_local,
            options={'ftol': 1e-12, 'maxiter': 500}
        )
        
        if result_local.success and result_local.fun < best_c5:
            best_c5 = result_local.fun
            best_h = result_local.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points

def compute_c5(h, dx):
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return np.max(corr) * dx
