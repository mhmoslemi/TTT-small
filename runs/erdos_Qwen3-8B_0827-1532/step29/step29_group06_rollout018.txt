    # Optimize using differential evolution with adjusted parameters
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr) * dx
        penalty = 1e4 * (np.sum(h) - target_sum)**2
        return max_corr + penalty
    
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=20,
        maxiter=200,
        tol=1e-8,
        mutation=(0.8, 1),
        recombination=0.8,
        seed=seed
    )
    
    if result.success:
        current_c5 = result.fun - 1e4 * (np.sum(result.x) - target_sum)**2
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x
    
    # Perturbation and local refinement
    num_perturbations = 1000
    for perturbation_idx in range(num_perturbations):
        # Select a random segment to perturb
        segment_idx = np.random.randint(n_points)
        perturbed = best_h.copy()
        perturbed[segment_idx] += np.random.uniform(-0.2, 0.2)
        perturbed[segment_idx] = np.clip(perturbed[segment_idx], 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        def objective_local(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds_local = [(0.0, 1.0) for _ in range(n_points)]
        cons_local = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result_local = minimize(
            objective_local,
            perturbed,
            method='SLSQP',
            bounds=bounds_local,
            constraints=cons_local,
            options={'ftol': 1e-12, 'maxiter': 1000}
        )
        
        if result_local.success and result_local.fun < best_c5:
            best_c5 = result_local.fun
            best_h = result_local.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points

def compute_c5(h, dx):
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return np.max(corr) * dx
