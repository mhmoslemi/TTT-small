import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    
    start_time = time.time()
    
    # Generate diverse initial guesses
    initial_guesses = []
    # Use provided initial_h_values
    initial_guesses.append(initial_h_values)
    # Add random patterns with 0, 0.5, 1 values
    for _ in range(5):
        h = np.random.choice([0, 0.5, 1], size=n_points)
        sum_h = np.sum(h)
        scale = target_sum / sum_h
        initial_guesses.append(h * scale)
    # Add symmetric patterns
    for _ in range(3):
        h = np.zeros(n_points)
        length = n_points // 4
        h[0:length] = 1.0
        h[-length:] = 1.0
        sum_h = np.sum(h)
        scale = target_sum / sum_h
        initial_guesses.append(h * scale)
    
    # Optimize each initial guess with differential evolution
    for guess in initial_guesses:
        # Define objective function with automatic scaling
        def objective(h):
            h_scaled = h * (target_sum / np.sum(h))
            one_minus_h = 1.0 - h_scaled
            corr = np.correlate(h_scaled, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Bounds for each variable
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Run differential evolution
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=15,
            maxiter=100,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed
        )
        
        # Update best solution if successful
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # Final check for early termination
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
