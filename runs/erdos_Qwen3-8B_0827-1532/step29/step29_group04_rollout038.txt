        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective_perturb(h):
            h_normalized = h.copy()
            sum_h = np.sum(h_normalized)
            if sum_h != 0:
                scale = target_sum / sum_h
                h_normalized = h_normalized * scale
            h_normalized = np.clip(h_normalized, 0, 1)
            one_minus_h = 1.0 - h_normalized
            corr = np.correlate(h_normalized, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Run differential evolution on the perturbed solution
        result = differential_evolution(
            objective_perturb,
            bounds,
            strategy='best1bin',
            maxiter=50,
            popsize=10,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed
        )
        
        if result.success:
            current_h = result.x
            current_c5 = objective_perturb(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points

def compute_c5(h, dx):
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return np.max(corr) * dx
