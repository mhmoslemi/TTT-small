    initial_guesses = [g for i, g in enumerate(initial_guesses) if i == 0 or np.any(g != initial_guesses[i-1])]
    
    # Differential Evolution
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]
    
    result = differential_evolution(
        fun=lambda h: compute_c5(h, dx),
        bounds=bounds,
        constraints=cons,
        x0=initial_guesses,
        popsize=10,
        mutation=(0.5, 1),
        recombination=0.9,
        tol=1e-12,
        maxiter=100,
        polish=True,
        disp=False
    )
    
    best_h = result.x
    best_c5 = result.fun
    
    # Local refinement with SLSQP
    def objective(h):
        return compute_c5(h, dx)
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    
    result_slsqp = minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-12, 'maxiter': 3000}
    )
    
    best_h = result_slsqp.x
    best_c5 = result_slsqp.fun
    
    # Perturbation phase
    num_perturbations = 500
    phase_shift = 0.2
    phase_decay = 0.98
    
    for perturbation_idx in range(num_perturbations):
        # Select a random segment to perturb
        segment_idx = np.random.randint(n_points)
        perturbed = best_h.copy()
        perturbed[segment_idx] += np.random.uniform(-0.2, 0.2)
        perturbed[segment_idx] = np.clip(perturbed[segment_idx], 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective_perturb(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        result = minimize(
            objective_perturb,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-12, 'maxiter': 1000}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points

def compute_c5(h, dx):
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return np.max(corr) * dx
