        h_rand = generate_random_segment_pattern(n_points, num_segments)
        initial_guesses.append(h_rand)
    
    # Add predefined segment patterns
    def generate_segment_pattern(n_points, segments):
        dx = 2.0 / n_points
        h = np.zeros(n_points)
        for i in range(len(segments)):
            val = segments[i]
            start = i * (n_points // len(segments))
            end = (i + 1) * (n_points // len(segments))
            h[start:end] = val
        sum_h = np.sum(h)
        scale = target_sum / sum_h
        return h * scale
    
    # Two-segment pattern
    initial_guesses.append(generate_segment_pattern(n_points, [1.0, 0.5]))
    # Three-segment pattern
    initial_guesses.append(generate_segment_pattern(n_points, [1.0, 0.5, 1.0]))
    # Four-segment pattern
    initial_guesses.append(generate_segment_pattern(n_points, [1.0, 0.5, 1.0, 0.5]))
    # Five-segment pattern
    initial_guesses.append(generate_segment_pattern(n_points, [1.0, 0.5, 1.0, 0.5, 1.0]))
    # Six-segment pattern
    initial_guesses.append(generate_segment_pattern(n_points, [1.0, 0.5, 1.0, 0.5, 1.0, 0.5]))
    # Seven-segment pattern
    initial_guesses.append(generate_segment_pattern(n_points, [1.0, 0.5, 1.0, 0.5, 1.0, 0.5, 1.0]))
    # Eight-segment pattern
    initial_guesses.append(generate_segment_pattern(n_points, [1.0, 0.5, 1.0, 0.5, 1.0, 0.5, 1.0, 0.5]))
    
    # Optimizing each initial guess
    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-12, 'maxiter': 2000}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Perturbation phase on the best_h found so far
    num_perturbations = 300
    phase_shift = 0.15
    phase_decay = 0.96
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,

        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points

def generate_random_segment_pattern(n_points, num_segments):
    dx = 2.0 / n_points
    segment_length = 2.0 / num_segments
    h = np.zeros(n_points)
    for i in range(n_points):
        x_i = i * dx
        segment_number = int(np.floor(x_i / segment_length))
        if segment_number >= num_segments:
            segment_number = num_segments - 1
        value = np.random.choice([0.0, 0.5, 1.0])
        h[i] = value
    sum_h = np.sum(h)
    scale = target_sum / sum_h
    h = h * scale
    return h

def compute_c5(h, dx):
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return np.max(corr) * dx
