    best_c5 = compute_c5(best_h, dx)
    
    # Generate diverse initial guesses
    initial_guesses = [initial_guess]
    
    # Random segment patterns
    for _ in range(5):
        h = np.zeros(n_points)
        for i in range(n_points):
            if np.random.rand() < 0.3:
                h[i] = 1.0
            elif np.random.rand() < 0.5:
                h[i] = 0.5
        sum_h = np.sum(h)
        scale = target_sum / sum_h
        initial_guesses.append(h * scale)
    
    # Structured symmetric patterns
    for _ in range(3):
        h = np.zeros(n_points)
        length = n_points // 4
        h[0:length] = 1.0
        h[-length:] = 1.0
        sum_h = np.sum(h)
        scale = target_sum / sum_h
        initial_guesses.append(h * scale)
    
    # Run differential evolution on each initial guess
    start_time = time.time()
    for guess in initial_guesses:
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        def objective(h):
            one_minus_h = 1.0 - h
            corr = correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr) * dx
            penalty = 1e4 * abs(np.sum(h) - target_sum)
            return max_corr + penalty
        
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=20,
            maxiter=100,
            tol=1e-6,
            polish=True
        )
        
        if result.success:
            current_h = result.x
            current_c5 = compute_c5(current_h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    # Refine with SLSQP
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    
    def objective_slsqp(h):
        one_minus_h = 1.0 - h
        corr = correlate(h, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    result_slsqp = minimize(
        objective_slsqp,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-10, 'maxiter': 500, 'eps': 1e-8}
    )
    
    if result_slsqp.success:
        best_h = result_slsqp.x
        best_c5 = result_slsqp.fun
    
    # Perturbation phase
    num_perturbations = 300
    phase_shift = 0.25
    phase_decay = 0.98
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        def objective_perturb(h):
            one_minus_h = 1.0 - h
            corr = correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        result_perturb = minimize(
            objective_perturb,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 300, 'eps': 1e-8}
        )
        
        if result_perturb.success and result_perturb.fun < best_c5:
            best_c5 = result_perturb.fun
            best_h = result_perturb.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points

def compute_c5(h, dx):
    one_minus_h = 1.0 - h
    corr = correlate(h, one_minus_h, mode='full')
    return np.max(corr) * dx
