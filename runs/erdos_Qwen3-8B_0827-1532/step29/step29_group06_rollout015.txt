    h_25[:n_points//4] = 1.0
    h_25[n_points//2:n_points//2 + n_points//4] = 1.0
    sum_h = np.sum(h_25)
    scale = target_sum / sum_h
    initial_guesses.append(h_25 * scale)
    
    # Three-segment pattern
    h_three = np.zeros(n_points)
    segment_length = n_points // 3
    h_three[:segment_length] = 1.0
    h_three[segment_length:2*segment_length] = 0.5
    h_three[2*segment_length:] = 1.0
    sum_h = np.sum(h_three)
    scale = target_sum / sum_h
    initial_guesses.append(h_three * scale)
    
    # Four-segment pattern
    h_four = np.zeros(n_points)
    segment_length = n_points // 4
    h_four[:segment_length] = 1.0
    h_four[segment_length:2*segment_length] = 0.5
    h_four[2*segment_length:3*segment_length] = 1.0
    h_four[3*segment_length:] = 0.5
    sum_h = np.sum(h_four)
    scale = target_sum / sum_h
    initial_guesses.append(h_four * scale)
    
    # Five-segment pattern
    h_five = np.zeros(n_points)
    segment_length = n_points // 5
    h_five[:segment_length] = 1.0
    h_five[segment_length:2*segment_length] = 0.5
    h_five[2*segment_length:3*segment_length] = 0.0
    h_five[3*segment_length:4*segment_length] = 0.5
    h_five[4*segment_length:] = 1.0
    sum_h = np.sum(h_five)
    scale = target_sum / sum_h
    initial_guesses.append(h_five * scale)
    
    # Random segments with varying heights
    h_rand_seg = np.zeros(n_points)
    for i in range(n_points):
        if np.random.rand() < 0.3:
            h_rand_seg[i] = 1.0
        elif np.random.rand() < 0.5:
            h_rand_seg[i] = 0.5
    sum_h = np.sum(h_rand_seg)
    scale = target_sum / sum_h
    initial_guesses.append(h_rand_seg * scale)
    
    # Random initial guess
    initial_guesses.append(np.random.uniform(0, 1, n_points))
    
    # Generate more diverse initial guesses with different patterns
    for _ in range(5):
        h_rand = np.random.uniform(0, 1, n_points)
        sum_h = np.sum(h_rand)
        scale = target_sum / sum_h
        initial_guesses.append(h_rand * scale)
    
    # Optimize using differential evolution with penalty
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr) * dx
        penalty = 1e4 * (np.sum(h) - target_sum)**2
        return max_corr + penalty
    
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=15,
        maxiter=100,
        tol=1e-8,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed
    )
    
    if result.success:
        current_c5 = result.fun - 1e4 * (np.sum(result.x) - target_sum)**2
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x
    
    # Perturbation and local refinement
    num_perturbations = 500
    for perturbation_idx in range(num_perturbations):
        # Select a random segment to perturb
        segment_idx = np.random.randint(n_points)
        perturbed = best_h.copy()
        perturbed[segment_idx] += np.random.uniform(-0.2, 0.2)
        perturbed[segment_idx] = np.clip(perturbed[segment_idx], 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        def objective_local(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds_local = [(0.0, 1.0) for _ in range(n_points)]
        cons_local = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result_local = minimize(
            objective_local,
            perturbed,
            method='SLSQP',
            bounds=bounds_local,
            constraints=cons_local,
            options={'ftol': 1e-12, 'maxiter': 1000}
        )
        
        if result_local.success and result_local.fun < best_c5:
            best_c5 = result_local.fun
            best_h = result_local.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points

def compute_c5(h, dx):
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return np.max(corr) * dx
