            bounds,
            popsize=5,
            strategy='best1bin',
            maxiter=200,
            tol=1e-12,
            args=(),
            mutation=(0.5, 1),
            recombination=0.9,
            seed=seed,
            polish=True,
            init=[best_h]
        )
        best_h = result.x
        best_c5 = result.fun
    
    return best_h, best_c5, n_points

def compute_c5(h, dx):
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return np.max(corr) * dx
