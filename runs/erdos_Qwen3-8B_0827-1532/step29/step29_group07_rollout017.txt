    scale = target_sum / sum_h
    initial_guesses.append(h_five * scale)
    
    # Random segments with varying heights
    h_rand_seg = np.zeros(n_points)
    for i in range(n_points):
        if np.random.rand() < 0.3:
            h_rand_seg[i] = 1.0
        elif np.random.rand() < 0.5:
            h_rand_seg[i] = 0.5
    sum_h = np.sum(h_rand_seg)
    scale = target_sum / sum_h
    initial_guesses.append(h_rand_seg * scale)
    
    # Random initial guess
    initial_guesses.append(np.random.uniform(0, 1, n_points))
    
    # Generate more diverse initial guesses with different patterns
    for _ in range(5):
        h_rand = np.random.uniform(0, 1, n_points)
        sum_h = np.sum(h_rand)
        scale = target_sum / sum_h
        initial_guesses.append(h_rand * scale)
    
    # Optimize each initial guess with DE
    for guess in initial_guesses:
        # Objective function with penalty for sum constraint
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr) * dx
            sum_penalty = (np.sum(h) - target_sum)**2 * 1e4
            return max_corr + sum_penalty
        
        # Bounds
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Run differential evolution
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=15,
            mutation=(0.5, 1),
            recombination=0.9,
            tol=1e-6,
            maxiter=100,
            polish=True
        )
        
        # Check if the result is valid and update best solution
        if result.success:
            # Compute the actual max_corr without penalty
            one_minus_h = 1.0 - result.x
            corr = np.correlate(result.x, one_minus_h, mode='full')
            current_c5 = np.max(corr) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # Refine the best solution with SLSQP
    if best_c5 != float('inf'):
        # Objective function for SLSQP
        def objective_slsqp(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Constraints for SLSQP
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Bounds
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Run SLSQP
        result_slsqp = minimize(
            objective_slsqp,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-12, 'maxiter': 300}
        )
        
        if result_slsqp.success:
            best_c5 = result_slsqp.fun
            best_h = result_slsqp.x
    
    # Perturbation phase
    num_perturbations = 500
    phase_shift = 0.2
    phase_decay = 0.98
    
    for perturbation_idx in range(num_perturbations):
        # Select a random segment to perturb
        segment_idx = np.random.randint(n_points)
        perturbed = best_h.copy()
        perturbed[segment_idx] += np.random.uniform(-0.2, 0.2)
        perturbed[segment_idx] = np.clip(perturbed[segment_idx], 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function for SLSQP
        def objective_slsqp_perturb(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Constraints for SLSQP
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Bounds
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Run SLSQP
        result_perturb = minimize(
            objective_slsqp_perturb,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-12, 'maxiter': 100}
        )
        
        if result_perturb.success and result_perturb.fun < best_c5:
            best_c5 = result_perturb.fun
            best_h = result_perturb.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:

    return best_h, best_c5, n_points

def compute_c5(h, dx):
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return np.max(corr) * dx
