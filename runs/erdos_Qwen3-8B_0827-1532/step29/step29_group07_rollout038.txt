    # Four-segment pattern
    h_four = np.zeros(n_points)
    segment_length = n_points // 4
    h_four[:segment_length] = 1.0
    h_four[segment_length:2*segment_length] = 0.5
    h_four[2*segment_length:3*segment_length] = 1.0
    h_four[3*segment_length:] = 0.5
    sum_h = np.sum(h_four)

    # Five-segment pattern
    h_five = np.zeros(n_points)

    h_five[2*segment_length:3*segment_length] = 0.0
    h_five[3*segment_length:4*segment_length] = 0.5
    h_five[4*segment_length:] = 1.0
    sum_h = np.sum(h_five)
    scale = target_sum / sum_h
    initial_guesses.append(h_five * scale)
    
    # Random segments with varying heights
    h_rand_seg = np.zeros(n_points)
    for i in range(n_points):
        if np.random.rand() < 0.3:
            h_rand_seg[i] = 1.0
        elif np.random.rand() < 0.5:
            h_rand_seg[i] = 0.5
    sum_h = np.sum(h_rand_seg)
    scale = target_sum / sum_h
    initial_guesses.append(h_rand_seg * scale)
    
    # Random initial guess
    initial_guesses.append(np.random.uniform(0, 1, n_points))
    
    # Generate more diverse initial guesses with different patterns
    for _ in range(5):
        h_rand = np.random.uniform(0, 1, n_points)
        sum_h = np.sum(h_rand)
        scale = target_sum / sum_h
        initial_guesses.append(h_rand * scale)
    
    # Define objective function for differential evolution
    def objective(x, n_points, dx, target_sum):
        h = np.zeros(n_points)
        h[:-1] = x
        last_var = target_sum - np.sum(x)
        last_var = np.clip(last_var, 0, 1)
        h[-1] = last_var
        h = np.clip(h, 0, 1)
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Clip and scale the guess
        guess = np.clip(guess, 0, 1)
        sum_guess = np.sum(guess)
        if sum_guess == 0:
            guess = np.zeros(n_points)
        else:
            scale = target_sum / sum_guess
            guess = guess * scale
        # Extract first n_points-1 variables

        # Run differential evolution
        bounds = [(0.0, 1.0) for _ in range(n_points-1)]
        result = differential_evolution(
            lambda x: objective(x, n_points, dx, target_sum),
            bounds,
            strategy='best1bin',
            popsize=15,
            mutation=(0.5, 1),
            recombination=0.9,
            tol=1e-8,
            maxiter=100,
            polish=True,
            seed=seed
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                # Reconstruct the full h array
                h = np.zeros(n_points)
                h[:-1] = result.x
                last_var = target_sum - np.sum(result.x)
                last_var = np.clip(last_var, 0, 1)
                h[-1] = last_var
                h = np.clip(h, 0, 1)
                sum_h = np.sum(h)
                scale = target_sum / sum_h
                h = h * scale
                best_h = h
    
    # Early stopping based on time budget
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return best_h, best_c5, n_points
    
    # Perturbation phase on the best_h found so far
    num_perturbations = 500

        bounds_perturb = [(0.0, 1.0) for _ in range(n_points-1)]
        result = differential_evolution(

        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
