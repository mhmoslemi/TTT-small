    h_six[:segment_length] = 1.0
    h_six[segment_length:2*segment_length] = 0.5
    h_six[2*segment_length:3*segment_length] = 1.0
    h_six[3*segment_length:4*segment_length] = 0.5
    h_six[4*segment_length:5*segment_length] = 1.0
    h_six[5*segment_length:] = 0.5
    sum_h = np.sum(h_six)
    scale = target_sum / sum_h
    initial_guesses.append(h_six * scale)
    
    # Seven-segment pattern
    h_seven = np.zeros(n_points)
    segment_length = n_points // 7
    h_seven[:segment_length] = 1.0
    h_seven[segment_length:2*segment_length] = 0.5
    h_seven[2*segment_length:3*segment_length] = 1.0
    h_seven[3*segment_length:4*segment_length] = 0.5

    h_seven[5*segment_length:6*segment_length] = 0.5
    h_seven[6*segment_length:] = 1.0
    sum_h = np.sum(h_seven)
    scale = target_sum / sum_h
    initial_guesses.append(h_seven * scale)
    
    # Eight-segment pattern
    h_eight = np.zeros(n_points)
    segment_length = n_points // 8
    h_eight[:segment_length] = 1.0
    h_eight[segment_length:2*segment_length] = 0.5
    h_eight[2*segment_length:3*segment_length] = 1.0
    h_eight[3*segment_length:4*segment_length] = 0.5
    h_eight[4*segment_length:5*segment_length] = 1.0
    h_eight[5*segment_length:6*segment_length] = 0.5
    h_eight[6*segment_length:7*segment_length] = 1.0
    h_eight[7*segment_length:] = 0.5
    sum_h = np.sum(h_eight)
    scale = target_sum / sum_h
    initial_guesses.append(h_eight * scale)
    
    # Ten-segment pattern
    h_ten = np.zeros(n_points)
    segment_length = n_points // 10
    for i in range(5):
        h_ten[i*segment_length : (i+1)*segment_length] = 1.0
        h_ten[(i+5)*segment_length : (i+6)*segment_length] = 0.5
    sum_h = np.sum(h_ten)
    scale = target_sum / sum_h
    initial_guesses.append(h_ten * scale)
    
    # Optimizing each initial guess
    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-12, 'maxiter': 2000}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Perturbation phase on the best_h found so far
    num_perturbations = 500
    phase_shift = 0.15
    phase_decay = 0.98
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,

            method='SLSQP',
            bounds=bounds,

            options={'ftol': 1e-12, 'maxiter': 600}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points

def compute_c5(h, dx):
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return np.max(corr) * dx
