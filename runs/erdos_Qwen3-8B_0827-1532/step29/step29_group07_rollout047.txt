                best_c5 = current_c5
                best_h = result.x
    
    # Local refinement with SLSQP
    def objective_slsqp(h):
        h_scaled = h * (target_sum / np.sum(h))
        one_minus_h = 1.0 - h_scaled
        corr = np.correlate(h_scaled, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]
    
    result_slsqp = minimize(
        objective_slsqp,
        best_h,
        method='SLSQP',
        bounds=bounds_slsqp,
        constraints=cons,
        options={'ftol': 1e-12, 'maxiter': 3000}
    )
    
    if result_slsqp.success:
        best_c5 = result_slsqp.fun
        best_h = result_slsqp.x
    
    # Perturbation phase
    num_perturbations = 500
    phase_shift = 0.2
    phase_decay = 0.98
    
    for perturbation_idx in range(num_perturbations):
        # Select a random segment to perturb
        segment_idx = np.random.randint(n_points)
        perturbed = best_h.copy()
        perturbed[segment_idx] += np.random.uniform(-0.2, 0.2)
        perturbed[segment_idx] = np.clip(perturbed[segment_idx], 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective_perturb(h):
            h_scaled = h * (target_sum / np.sum(h))
            one_minus_h = 1.0 - h_scaled
            corr = np.correlate(h_scaled, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        result = minimize(
            objective_perturb,
            perturbed,
            method='SLSQP',
            bounds=bounds_slsqp,
            constraints=cons,
            options={'ftol': 1e-12, 'maxiter': 1000}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
