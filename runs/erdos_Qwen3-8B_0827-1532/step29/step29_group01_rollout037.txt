    h_seven[4*segment_length:5*segment_length] = 1.0
    h_seven[5*segment_length:6*segment_length] = 0.5
    h_seven[6*segment_length:] = 1.0
    sum_h = np.sum(h_seven)
    scale = target_sum / sum_h
    initial_guesses.append(h_seven * scale)
    
    # Eight-segment pattern
    h_eight = np.zeros(n_points)
    segment_length = n_points // 8
    h_eight[:segment_length] = 1.0
    h_eight[segment_length:2*segment_length] = 0.5
    h_eight[2*segment_length:3*segment_length] = 1.0
    h_eight[3*segment_length:4*segment_length] = 0.5
    h_eight[4*segment_length:5*segment_length] = 1.0
    h_eight[5*segment_length:6*segment_length] = 0.5
    h_eight[6*segment_length:7*segment_length] = 1.0
    h_eight[7*segment_length:] = 0.5
    sum_h = np.sum(h_eight)
    scale = target_sum / sum_h
    initial_guesses.append(h_eight * scale)
    
    # Optimizing each initial guess
    for guess in initial_guesses:
        # Objective function
        def objective(x):
            h = np.zeros(n_points)
            h[:-1] = x
            h[-1] = target_sum - np.sum(x)
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Bounds for the first n-1 variables
        bounds = [(0.0, 1.0) for _ in range(n_points - 1)]
        
        # Use differential evolution for initial optimization
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=15,
            tol=1e-12,
            mutation=(0.5, 1),
            recombination=0.9,
            maxiter=100,
            disp=False
        )
        
        # Convert the result to full h
        x_opt = result.x
        h_opt = np.zeros(n_points)
        h_opt[:-1] = x_opt
        h_opt[-1] = target_sum - np.sum(x_opt)
        h_opt = np.clip(h_opt, 0, 1)
        
        # Refine with SLSQP
        def objective_slsqp(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]
        cons_slsqp = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result_slsqp = minimize(
            objective_slsqp,
            h_opt,
            method='SLSQP',
            bounds=bounds_slsqp,
            constraints=cons_slsqp,
            options={'ftol': 1e-12, 'maxiter': 2000}
        )
        
        if result_slsqp.success and result_slsqp.fun < best_c5:
            best_c5 = result_slsqp.fun
            best_h = result_slsqp.x
    
    # Perturbation phase on the best_h found so far
    num_perturbations = 300
    phase_shift = 0.15
    phase_decay = 0.96
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective_perturb(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds_perturb = [(0.0, 1.0) for _ in range(n_points)]
        cons_perturb = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result_perturb = minimize(
            objective_perturb,
            perturbed,
            method='SLSQP',
            bounds=bounds_perturb,
            constraints=cons_perturb,
            options={'ftol': 1e-12, 'maxiter': 600}
        )
        
        if result_perturb.success and result_perturb.fun < best_c5:
            best_c5 = result_perturb.fun
            best_h = result_perturb.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points

def compute_c5(h, dx):
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return np.max(corr) * dx
