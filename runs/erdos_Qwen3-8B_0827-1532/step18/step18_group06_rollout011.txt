from scipy.optimize import minimize

    scale = target_sum / sum_h

    # Step function (asymmetric)

    # Alternating pattern (asymmetric)

    initial_guesses.append(h_sinusoidal)
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Define the objective function
        def objective(h_full):
            one_minus_h = 1.0 - h_full

            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
