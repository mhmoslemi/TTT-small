import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    
    start_time = time.time()
    
    # Generate diverse initial guesses
    initial_guesses = []

    h_random = np.random.uniform(0, 1, n_points)
    sum_h = np.sum(h_random)
    scale = target_sum / sum_h
    initial_guesses.append(h_random * scale)
    
    # Step function
    h_step = np.zeros(n_points)
    h_step[:n_points//2] = 1.0
    sum_h = np.sum(h_step)
    scale = target_sum / sum_h
    initial_guesses.append(h_step * scale)
    
    # Alternating pattern
    h_alt = np.zeros(n_points)
    h_alt[::2] = 1.0
    sum_h = np.sum(h_alt)
    scale = target_sum / sum_h
    initial_guesses.append(h_alt * scale)
    
    # Sinusoidal
    h_sinusoidal = 0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / n_points)
    sum_h = np.sum(h_sinusoidal)
    scale = target_sum / sum_h
    initial_guesses.append(h_sinusoidal)
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Define the objective function
        def objective(h):
            h = np.clip(h, 0, 1)
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 500}
        )
        
        if result.success and result.fun < best_c5:

            best_h = result.x
    
    # Perturbation phase on the best_h found so far
    num_perturbations = 300
    phase_shift = 0.2
    phase_decay = 0.99
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        # Perturb the best_h
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed_h = best_h + noise
        perturbed_h = np.clip(perturbed_h, 0, 1)
        # Scale to maintain the sum
        current_sum = np.sum(perturbed_h)
        scale = target_sum / current_sum
        perturbed_h = perturbed_h * scale
        
        # Define the objective function
        def objective(h):
            h = np.clip(h, 0, 1)
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
