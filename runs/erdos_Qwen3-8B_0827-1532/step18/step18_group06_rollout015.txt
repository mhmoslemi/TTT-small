    initial_guesses.append(h_random)
    
    # Sinusoidal symmetric initial guess
    h_sinusoidal = 0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points//2) / (n_points//2))
    initial_guesses.append(h_sinusoidal)
    
    # Optimize each initial guess with differential evolution
    for guess in initial_guesses:
        # Define the objective function
        def objective(h_half):
            h_full = np.concatenate([h_half, h_half[::-1]])
            sum_h_full = np.sum(h_full)
            scale = target_sum / sum_h_full
            h = h_full * scale
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Bounds for each variable in h_half
        bounds = [(0.0, 1.0) for _ in range(n_points//2)]
        
        # Perform differential evolution
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=100,
            popsize=15,
            tol=1e-10,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed
        )
        
        if result.success:
            best_c5 = result.fun
            # Reconstruct the best h from the optimized h_half
            best_h_half = result.x
            best_h = np.concatenate([best_h_half, best_h_half[::-1]]) * (target_sum / np.sum(np.concatenate([best_h_half, best_h_half[::-1]])))
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    # Perturbation phase on the best_h found so far
    num_perturbations = 500  # Reduced from 1000
    phase_shift = 0.2
    phase_decay = 0.99
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        # Perturb the first half
        noise = np.random.uniform(-noise_scale, noise_scale, n_points//2)
        perturbed_half = best_h[:n_points//2] + noise
        perturbed_half = np.clip(perturbed_half, 0, 1)
        # Scale to maintain the sum
        current_sum = np.sum(np.concatenate([perturbed_half, perturbed_half[::-1]]))
        scale = target_sum / current_sum
        perturbed_half = perturbed_half * scale
        
        # Define the objective function
        def objective(h_half):
            h_full = np.concatenate([h_half, h_half[::-1]])
            sum_h_full = np.sum(h_full)
            scale = target_sum / sum_h_full
            h = h_full * scale
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Bounds for each variable in h_half
        bounds = [(0.0, 1.0) for _ in range(n_points//2)]
        
        # Perform local optimization with SLSQP
        def objective_slsqp(h_half):
            h_full = np.concatenate([h_half, h_half[::-1]])
            sum_h_full = np.sum(h_full)
            scale = target_sum / sum_h_full
            h = h_full * scale
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds_slsqp = [(0.0, 1.0) for _ in range(n_points//2)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(np.concatenate([h, h[::-1]])) - target_sum})
        
        result = minimize(
            objective_slsqp,
            perturbed_half,
            method='SLSQP',
            bounds=bounds_slsqp,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 500}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            # Reconstruct the best h from the optimized h_half
            best_h_half = result.x
            best_h = np.concatenate([best_h_half, best_h_half[::-1]]) * (target_sum / np.sum(np.concatenate([best_h_half, best_h_half[::-1]])))
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
