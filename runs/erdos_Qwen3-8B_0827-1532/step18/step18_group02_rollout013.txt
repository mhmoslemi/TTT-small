from scipy.optimize import minimize

    initial_guesses.append(h_sin * scale)

            h_tri[i] = (2.0 - x) / 1.0

    sum_h = np.sum(h_two_intervals)

    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
