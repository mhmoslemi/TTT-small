    initial_guesses = []
    # Add initial_h_values
    initial_guesses.append(initial_guess.copy())
    # Add structured guesses with multiple intervals
    for num_intervals in [2, 3, 4]:
        w = n_points // (2 * num_intervals)
        gap = n_points // 4
        for shift in [0, n_points//4, n_points//2, 3*n_points//4]:
            h = np.zeros(n_points)
            for i in range(n_points):
                x = i * dx
                for k in range(num_intervals):
                    start = shift + k * (w + gap)
                    end = start + w
                    if start <= x < end:
                        h[i] = 1.0
            # Scale to meet the sum constraint
            current_sum = np.sum(h)
            scale = (target_sum) / current_sum
            h = h * scale
            initial_guesses.append(h)
    # Add random guesses with different scaling
    for _ in range(5):
        h = np.random.uniform(0, 1, n_points)
        current_sum = np.sum(h)
        scale = target_sum / current_sum
        h = h * scale
        initial_guesses.append(h)
    
    # Optimize each initial guess with tighter tolerances
    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 400}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Enhanced perturbation phase with adaptive noise
    num_perturbations = 300
    phase_shift = 0.3
    phase_decay = 0.96
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 250}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
