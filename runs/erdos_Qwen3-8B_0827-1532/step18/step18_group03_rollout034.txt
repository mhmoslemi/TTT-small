import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Define the objective function with normalization
    def objective(h):
        if np.sum(h) == 0:
            return float('inf')
        h_normalized = h * (target_sum / np.sum(h))
        one_minus_h = 1.0 - h_normalized
        corr = np.correlate(h_normalized, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    # Bounds for each parameter (each h[i] must be between 0 and 1)
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Run differential evolution for global optimization
    result_de = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=50,
        popsize=20,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed,
        polish=True,
        disp=False
    )
    
    best_h = result_de.x
    best_c5 = result_de.fun
    
    # Refine with SLSQP for local optimization
    def objective_slsqp(h):
        if np.sum(h) == 0:
            return float('inf')
        h_normalized = h * (target_sum / np.sum(h))
        one_minus_h = 1.0 - h_normalized
        corr = np.correlate(h_normalized, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    
    result_slsqp = minimize(
        objective_slsqp,
        best_h,
        method='SLSQP',
        bounds=bounds_slsqp,
        constraints=cons,
        options={'ftol': 1e-9, 'maxiter': 200}
    )
    
    if result_slsqp.success and result_slsqp.fun < best_c5:
        best_c5 = result_slsqp.fun
        best_h = result_slsqp.x
    
    # Check remaining time for additional refinement
    elapsed = time.time() - start_time
    if elapsed < budget_s:
        # Additional perturbation and optimization
        num_perturbations = 50
        phase_shift = 0.15
        phase_decay = 0.96
        
        for perturbation_idx in range(num_perturbations):
            noise_scale = phase_shift * (phase_decay ** perturbation_idx)
            noise = np.random.uniform(-noise_scale, noise_scale, n_points)
            perturbed = best_h + noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
            
            result = minimize(
                objective_slsqp,
                perturbed,
                method='SLSQP',
                bounds=bounds_slsqp,
                constraints=cons,
                options={'ftol': 1e-9, 'maxiter': 100}
            )
            
            if result.success and result.fun < best_c5:
                best_c5 = result.fun
                best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed < budget_s:
            # Final refinement with differential evolution
            result_de_final = differential_evolution(
                objective,
                bounds,
                strategy='best1bin',
                maxiter=20,
                popsize=15,
                tol=1e-6,
                mutation=(0.5, 1),
                recombination=0.8,
                seed=seed,
                polish=True,
                disp=False
            )
            
            if result_de_final.fun < best_c5:
                best_c5 = result_de_final.fun
                best_h = result_de_final.x
    
    return best_h, best_c5, n_points
