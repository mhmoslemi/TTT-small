from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    dx = 2.0 / n_points
    target_sum = n_points / 2

    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate symmetric initial guesses
    initial_guesses = []
    # Original initial guess (symmetric)
    initial_guesses.append(initial_guess.copy())
    
    # Step function (symmetric)
    h_step = np.zeros(n_points)
    h_step[:n_points//2] = 1.0
    sum_h = np.sum(h_step)
    scale = target_sum / sum_h
    initial_guesses.append(h_step * scale)
    
    # Alternating pattern (symmetric)
    h_alt = np.zeros(n_points)
    h_alt[::2] = 1.0
    sum_h = np.sum(h_alt)
    scale = target_sum / sum_h
    initial_guesses.append(h_alt * scale)
    
    # Random symmetric initial guess
    h_random = np.random.uniform(0, 1, n_points//2)
    h_random = np.concatenate([h_random, h_random[::-1]])
    sum_h = np.sum(h_random)
    scale = target_sum / sum_h
    initial_guesses.append(h_random * scale)
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Extract the first half of the guess
        h_half = guess[:n_points//2]
        # Define the objective function
        def objective(h_half):
            h = np.concatenate([h_half, h_half[::-1]])
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points//2)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(np.concatenate([h, h[::-1]])) - target_sum})
        
        result = minimize(
            objective,

            options={'ftol': 1e-10, 'maxiter': 500}

        if result.success and result.fun < best_c5:
