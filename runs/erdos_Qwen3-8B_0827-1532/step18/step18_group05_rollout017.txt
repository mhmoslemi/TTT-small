        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr) * dx
            penalty = 1e4 * (np.sum(h) - target_sum)**2
            return max_corr + penalty
        
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=50,
            popsize=20,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed
        )
        
        def objective_slsqp(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]
        cons_slsqp = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result_slsqp = minimize(
            objective_slsqp,
            result.x,
            method='SLSQP',
            bounds=bounds_slsqp,
            constraints=cons_slsqp,
            options={'ftol': 1e-9, 'maxiter': 300}
        )
        
        if result_slsqp.success and result_slsqp.fun < best_c5:
            best_c5 = result_slsqp.fun
            best_h = result_slsqp.x
    
    # Perturbation phase
    num_perturbations = 150
    phase_shift = 0.15
    phase_decay = 0.96
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        def objective_perturb(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds_perturb = [(0.0, 1.0) for _ in range(n_points)]
        cons_perturb = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result_perturb = minimize(
            objective_perturb,
            perturbed,
            method='SLSQP',
            bounds=bounds_perturb,
            constraints=cons_perturb,
            options={'ftol': 1e-9, 'maxiter': 200}
        )
        
        if result_perturb.success and result_perturb.fun < best_c5:
            best_c5 = result_perturb.fun
            best_h = result_perturb.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
