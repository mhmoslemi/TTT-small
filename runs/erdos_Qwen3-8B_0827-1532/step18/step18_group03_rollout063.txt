from scipy.optimize import minimize
import time

    start_time = time.time()
    
    # Define objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    # Define constraint function
    def constraint(h):
        return np.sum(h) - target_sum
    
    # Optimization with Nelder-Mead for global exploration
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = minimize(
        objective,
        initial_guess,
        method='Nelder-Mead',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint},
        options={'xatol': 1e-6, 'maxiter': 300}
    )
    
    if result.success and result.fun < best_c5:
        best_c5 = result.fun
        best_h = result.x
    
    # Refinement with SLSQP for local optimization
    result = minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint},
        options={'ftol': 1e-9, 'maxiter': 500}
    )
    
    if result.success and result.fun < best_c5:
        best_c5 = result.fun

    # Perturbation phase with structured variations

    phase_shift = 0.15

    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Optimization with SLSQP
