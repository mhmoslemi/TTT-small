    h_sinusoidal = 0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / n_points)
    h_sinusoidal = np.concatenate([h_sinusoidal[:n_points//2], h_sinusoidal[:n_points//2][::-1]])
    sum_h = np.sum(h_sinusoidal)
    scale = target_sum / sum_h
    initial_guesses.append(h_sinusoidal * scale)
    
    # Three-step function
    h_three_step = np.zeros(n_points)
    h_three_step[:n_points//3] = 1.0
    h_three_step[n_points//3:2*n_points//3] = 0.5
    h_three_step[2*n_points//3:] = 0.0
    sum_h = np.sum(h_three_step)
    scale = target_sum / sum_h
    initial_guesses.append(h_three_step * scale)
    
    # Triangular function
    h_triangular = np.zeros(n_points)
    h_triangular[:n_points//2] = np.linspace(0, 1, n_points//2)
    h_triangular[n_points//2:] = np.linspace(1, 0, n_points//2)
    sum_h = np.sum(h_triangular)
    scale = target_sum / sum_h
    initial_guesses.append(h_triangular * scale)
    
    # Optimize each initial guess
    for guess in initial_guesses:
        h_half = guess[:n_points//2]
        def objective(h_half):
            h = np.concatenate([h_half, h_half[::-1]])
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points//2)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(np.concatenate([h, h[::-1]])) - target_sum})
        
        result = minimize(
            objective,
            h_half,
            method='COBYLA',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 500}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h_half = result.x
            best_h = np.concatenate([best_h_half, best_h_half[::-1]])
    
    # Perturbation phase with adaptive strategies
    num_perturbations = 500
    phase_shift = 0.2
    phase_decay = 0.99
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points//2)
        perturbed_half = best_h[:n_points//2] + noise
        perturbed_half = np.clip(perturbed_half, 0, 1)
        current_sum = np.sum(np.concatenate([perturbed_half, perturbed_half[::-1]]))
        scale = target_sum / current_sum
        perturbed_half = perturbed_half * scale
        
        def objective(h_half):
            h = np.concatenate([h_half, h_half[::-1]])
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points//2)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(np.concatenate([h, h[::-1]])) - target_sum})
        
        result = minimize(
            objective,
            perturbed_half,
            method='COBYLA',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 500}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h_half = result.x
            best_h = np.concatenate([best_h_half, best_h_half[::-1]])
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
