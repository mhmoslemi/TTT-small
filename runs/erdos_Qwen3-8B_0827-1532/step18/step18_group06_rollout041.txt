from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    target_sum = n_points / 2

    initial_guesses = []

    initial_guesses.append(initial_guess.copy())

    h_step = np.zeros(n_points)
    h_step[:n_points//2] = 1.0
    sum_h = np.sum(h_step)
    scale = target_sum / sum_h
    initial_guesses.append(h_step * scale)
    
    # Alternating pattern (symmetric)
    h_alt = np.zeros(n_points)
    h_alt[::2] = 1.0
    sum_h = np.sum(h_alt)
    scale = target_sum / sum_h
    initial_guesses.append(h_alt * scale)
    
    # Random asymmetric initial guess
    h_random = np.random.uniform(0, 1, n_points)

    # Sinusoidal asymmetric initial guess
    h_sinusoidal = 0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / n_points)

    initial_guesses.append(h_sinusoidal)

    for guess in initial_guesses:

        # Define the objective function
        def objective(h_half):
            h = np.concatenate([h_half, h_half[::-1]])
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with COBYLA
        bounds = [(0.0, 1.0) for _ in range(n_points//2)]
