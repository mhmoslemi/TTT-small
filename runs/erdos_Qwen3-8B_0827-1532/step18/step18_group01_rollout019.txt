from scipy.optimize import minimize, differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):

    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')

    start_time = time.time()
    
    # Initial guess: uniform distribution
    initial_guess = np.full(n_points, 0.5)

    sum_h = np.sum(h_step)
    scale = target_sum / sum_h
    initial_guesses = [initial_guess, h_step * scale]
    
    # Generate more diverse initial guesses
    for _ in range(5):
        h_rand = np.zeros(n_points)

            start = np.random.randint(0, n_points - 1)
            end = start + np.random.randint(1, n_points - start)
            h_rand[start:end] = 1.0
        sum_h = np.sum(h_rand)
        scale = target_sum / sum_h
        initial_guesses.append(h_rand * scale)

    for guess in initial_guesses:
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 1000}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Use differential evolution for global search
    def de_objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result_de = differential_evolution(

    )

    num_perturbations = 300
    phase_shift = 0.15

        noise = np.random.uniform(-noise_scale, noise_scale, n_points)

        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        def objective_perturb(h):
            one_minus_h = 1.0 - h
