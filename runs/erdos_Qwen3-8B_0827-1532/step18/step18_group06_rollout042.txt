    sum_h = np.sum(h_random)
    scale = target_sum / sum_h
    h_random = h_random * scale
    initial_guesses.append(h_random)
    
    # Step function
    h_step = np.zeros(n_points)
    h_step[:n_points//2] = 1.0
    sum_h = np.sum(h_step)
    scale = target_sum / sum_h
    h_step = h_step * scale
    initial_guesses.append(h_step)
    
    # Sinusoidal function
    h_sinusoidal = 0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / n_points)
    sum_h = np.sum(h_sinusoidal)
    scale = target_sum / sum_h
    h_sinusoidal = h_sinusoidal * scale
    initial_guesses.append(h_sinusoidal)
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Define the objective function
        def objective(h):
            h_full = h
            one_minus_h = 1.0 - h_full
            corr = np.correlate(h_full, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Define the constraint function
        def constraint(h):
            return np.sum(h) - target_sum
        
        # Optimization with L-BFGS-B
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': constraint})
        
        result = minimize(
            objective,
            guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 500}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Perturbation phase
    num_perturbations = 200  # Reduced from 500
    phase_shift = 0.2
    phase_decay = 0.99
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        # Perturb the best_h
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed_h = best_h + noise
        perturbed_h = np.clip(perturbed_h, 0, 1)
        # Scale to maintain the sum
        current_sum = np.sum(perturbed_h)
        scale = target_sum / current_sum
        perturbed_h = perturbed_h * scale
        
        # Define the objective function
        def objective(h):
            h_full = h
            one_minus_h = 1.0 - h_full
            corr = np.correlate(h_full, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Define the constraint function
        def constraint(h):
            return np.sum(h) - target_sum
        
        # Optimization with L-BFGS-B
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': constraint})
        
        result = minimize(
            objective,
            perturbed_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 500}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
