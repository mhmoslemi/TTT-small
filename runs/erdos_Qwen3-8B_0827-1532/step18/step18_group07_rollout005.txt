        for i in range(m):
            start = i * (interval_length + spacing) if i > 0 else 0.0
            end = start + interval_length
            for j in range(n_points):
                x = j * dx
                if start <= x < end:
                    h[j] = 1.0
        initial_guesses.append(h)
