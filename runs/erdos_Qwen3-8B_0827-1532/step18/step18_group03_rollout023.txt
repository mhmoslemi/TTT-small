    sum_h = np.sum(h_center)
    scale = target_sum / sum_h
    initial_guesses.append(make_symmetric(h_center * scale))
    # Symmetric two-segment step function
    h_two_seg = np.zeros(n_points)
    h_two_seg[:n_points//4] = 1.0
    h_two_seg[n_points//2:n_points//2 + n_points//4] = 1.0
    sum_h = np.sum(h_two_seg)
    scale = target_sum / sum_h
    initial_guesses.append(make_symmetric(h_two_seg * scale))
    # Random symmetric initial guess
    h_rand = np.random.uniform(0, 1, n_points)
    h_rand = make_symmetric(h_rand)
    sum_h = np.sum(h_rand)
    scale = target_sum / sum_h
    initial_guesses.append(h_rand * scale)
    
    # Optimize each initial guess
    for guess in initial_guesses:
        h_vars = guess[:n_points//2]
        full_h = construct_full_array(h_vars)
        def objective(h_vars):
            full_h = construct_full_array(h_vars)
            one_minus_h = 1.0 - full_h
            corr = np.correlate(full_h, one_minus_h, mode='full')
            return np.max(corr) * dx
        bounds = [(0.0, 1.0) for _ in range(n_points//2)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(construct_full_array(h)) - target_sum})
        result = minimize(
            objective,
            h_vars,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 800}
        )
        if result.success:
            full_h = construct_full_array(result.x)
            current_c5 = np.max(np.correlate(full_h, 1.0 - full_h, mode='full')) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = full_h
    
    # Perturbation phase with structured variations
    num_perturbations = 200
    phase_shift = 0.15
    phase_decay = 0.96
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        h_vars = best_h[:n_points//2]
        noise = np.random.uniform(-noise_scale, noise_scale, n_points//2)
        perturbed_vars = h_vars + noise
        perturbed_vars = np.clip(perturbed_vars, 0, 1)
        current_sum = np.sum(construct_full_array(perturbed_vars))
        scale = target_sum / current_sum
        perturbed_vars = perturbed_vars * scale
        
        def objective(h_vars):
            full_h = construct_full_array(h_vars)
            one_minus_h = 1.0 - full_h
            corr = np.correlate(full_h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points//2)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(construct_full_array(h)) - target_sum})
        result = minimize(
            objective,
            perturbed_vars,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 300}
        )
        if result.success:
            full_h = construct_full_array(result.x)
            current_c5 = np.max(np.correlate(full_h, 1.0 - full_h, mode='full')) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = full_h
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
