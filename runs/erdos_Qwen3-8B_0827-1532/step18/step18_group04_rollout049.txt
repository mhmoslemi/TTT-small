from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(initial_guess)

    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate diverse initial guesses
    initial_guesses = []

    initial_guesses.append(initial_guess.copy())
    
    # Add structured binary guesses with different interval patterns
    for shift in [0.0, 0.25, 0.5, 0.75, 1.0, 1.25, 1.5, 1.75]:
        h = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            if (x >= shift and x < shift + 0.5) or (x >= shift + 1.0 and x < shift + 1.5):
                h[i] = 1.0
        initial_guesses.append(h)
    
    # Add more structured patterns
    for pattern in [
        [0.0, 0.5, 1.0, 1.5],  # 0.5 spacing

        [0.0, 0.25, 0.5, 0.75, 1.0, 1.25, 1.5, 1.75],  # dense intervals
    ]:
        h = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            for start, end in zip(pattern[:-1], pattern[1:]):
                if start <= x < end:
                    h[i] = 1.0
                    break
        initial_guesses.append(h)
    
    # Add random binary patterns with exactly target_sum ones
    for _ in range(5):  # increased from 5
        h = np.zeros(n_points)

        h[indices] = 1.0
        initial_guesses.append(h)
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with COBYLA
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='COBYLA',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'disp': False}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x

        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx

        bounds = [(0.0, 1.0) for _ in range(n_points)]
