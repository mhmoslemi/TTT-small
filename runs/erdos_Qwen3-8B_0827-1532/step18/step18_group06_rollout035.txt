        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            # Reconstruct the full h from the optimized half
            best_h_half = result.x
            best_h = np.concatenate([best_h_half, best_h_half[::-1]])
    
    # Perturbation phase on the best_h found so far
    num_perturbations = 300  # Reduced from 500
    phase_shift = 0.2
    phase_decay = 0.99
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        # Perturb the first half
        noise = np.random.uniform(-noise_scale, noise_scale, n_points//2)
        perturbed_half = best_h[:n_points//2] + noise
        perturbed_half = np.clip(perturbed_half, 0, 1)
        # Scale to maintain the sum
        current_sum = np.sum(np.concatenate([perturbed_half, perturbed_half[::-1]]))
        scale = target_sum / current_sum
        perturbed_half = perturbed_half * scale
        
        # Define the objective function
        def objective(h_half):
            h = np.concatenate([h_half, h_half[::-1]])
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points//2)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(np.concatenate([h, h[::-1]])) - target_sum})
        
        result = minimize(
            objective,
            perturbed_half,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 300}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            # Reconstruct the full h from the optimized half
            best_h_half = result.x
            best_h = np.concatenate([best_h_half, best_h_half[::-1]])
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
