        cons = ({'type': 'eq', 'fun': lambda h: np.sum(np.concatenate([h, h[::-1]])) - target_sum})
        
        result = minimize(
            objective,
            perturbed_half,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 200}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = np.concatenate([result.x, result.x[::-1]])
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
