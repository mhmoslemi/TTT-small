            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points//2)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(np.concatenate([h, h[::-1]])) - target_sum})
        
        result = minimize(
            objective,
            perturbed_half,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 500}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            # Reconstruct the full h from the optimized half
            best_h_half = result.x
            best_h = np.concatenate([best_h_half, best_h_half[::-1]])
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
