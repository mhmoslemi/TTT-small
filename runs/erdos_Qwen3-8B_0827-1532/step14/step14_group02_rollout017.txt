from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate multiple initial guesses
    initial_guesses = [initial_guess]
    # Generate 4 random initial guesses based on initial_guess
    for _ in range(4):
        random_h = initial_guess + np.random.uniform(-0.1, 0.1, n_points)
        random_h = np.clip(random_h, 0, 1)
        random_h = random_h * (target_sum / np.sum(random_h))
        initial_guesses.append(random_h)
    # Add 2 completely random initial guesses
    for _ in range(2):
        random_h = np.random.uniform(0, 1, n_points)
        random_h = random_h * (target_sum / np.sum(random_h))
        initial_guesses.append(random_h)
    
    # For each initial guess, run the optimization
    for guess in initial_guesses:
        # Perturbation strategy with larger initial noise and slower decay
        num_perturbations = 100  # Reduced to save time
        phase_shift = 0.25  # Larger initial noise
        phase_decay = 0.98  # Slower decay
        
        current_best_c5 = float('inf')
        current_best_h = guess.copy()
        
        for perturbation_idx in range(num_perturbations):
            # Adaptive noise magnitude based on phase
            noise_scale = phase_shift * (phase_decay ** perturbation_idx)
            
            # Generate perturbed guess
            perturbed = current_best_h + np.random.uniform(-noise_scale, noise_scale, n_points)
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
            
            # Objective function
            def objective(h):
                one_minus_h = 1.0 - h
                corr = np.correlate(h, one_minus_h, mode='full')
                return np.max(corr) * dx
            
            # Optimization with SLSQP
            bounds = [(0.0, 1.0) for _ in range(n_points)]
            cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
            
            result = minimize(
                objective,
                perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'ftol': 1e-10, 'maxiter': 500}
            )
            
            if result.success and result.fun < current_best_c5:
                current_best_c5 = result.fun
                current_best_h = result.x
            
            elapsed = time.time() - start_time
            if elapsed >= budget_s:
                break
        
        # Update the global best
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = current_best_h
    
    return best_h, best_c5, n_points
