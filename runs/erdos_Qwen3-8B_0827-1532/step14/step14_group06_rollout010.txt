from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points

    def generate_initial_guesses():
        guesses = [initial_guess.copy()]

        rand_guess = np.clip(rand_guess, 0, 1)
        scale = target_sum / np.sum(rand_guess)
        rand_guess = rand_guess * scale
        guesses.append(rand_guess)
        # Periodic pattern every 3 points
        periodic3 = np.zeros(n_points)
        for i in range(n_points):
            if (i % 3 == 0):
                periodic3[i] = 1
        periodic3 = periodic3 * (target_sum / np.sum(periodic3))
        guesses.append(periodic3)
        # Periodic pattern every 4 points
        periodic4 = np.zeros(n_points)
        for i in range(n_points):
            if (i % 4 == 0):
                periodic4[i] = 1
        periodic4 = periodic4 * (target_sum / np.sum(periodic4))
        guesses.append(periodic4)
        # Alternating pattern
        alt_pattern = np.zeros(n_points)
        for i in range(n_points):
            if i % 2 == 0:
                alt_pattern[i] = 1
        alt_pattern = alt_pattern * (target_sum / np.sum(alt_pattern))
        guesses.append(alt_pattern)
        # Random binary pattern with higher density
        binary_high = np.random.choice([0, 1], size=n_points, p=[0.25, 0.75])
        binary_high_sum = np.sum(binary_high)
        binary_high = binary_high * (target_sum / binary_high_sum)
        guesses.append(binary_high)
        return guesses
    
    initial_guesses = generate_initial_guesses()
    
    # Adaptive perturbation strategy with varying noise and exploration phases
    num_perturbations = 500  # Increased for better exploration
    phase_shift = 0.5  # Larger initial noise magnitude
    phase_decay = 0.95  # Faster decay for more exploration
    
    for guess in initial_guesses:
        # Clone and normalize
        current_h = guess.copy()
        current_sum = np.sum(current_h)
        current_h = current_h * (target_sum / current_sum)
        
        for perturbation_idx in range(num_perturbations):
            # Adaptive noise magnitude based on phase
            noise_scale = phase_shift * (phase_decay ** perturbation_idx)
            
            # Generate perturbed guess
            noise = np.random.uniform(-noise_scale, noise_scale, n_points)
            perturbed = current_h + noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
            
            # Objective function
            def objective(h):
                one_minus_h = 1.0 - h
                corr = np.correlate(h, one_minus_h, mode='full')
                return np.max(corr) * dx
            
            # Optimization with SLSQP
            bounds = [(0.0, 1.0) for _ in range(n_points)]
            cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

            )

                best_h = result.x

    return best_h, best_c5, n_points
