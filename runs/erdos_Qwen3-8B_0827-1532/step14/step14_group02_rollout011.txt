import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate multiple initial guesses
    initial_guesses = [initial_guess]
    
    # Add structured initial guesses
    structured_guess1 = np.zeros(n_points)
    structured_guess1[:n_points//2] = 1.0
    initial_guesses.append(structured_guess1)
    
    structured_guess2 = np.zeros(n_points)
    structured_guess2[n_points//4 : 3*n_points//4] = 1.0
    initial_guesses.append(structured_guess2)
    
    structured_guess3 = np.zeros(n_points)
    structured_guess3[3*n_points//4 : 5*n_points//4] = 1.0
    initial_guesses.append(structured_guess3)
    
    # Generate 2 random initial guesses
    for _ in range(2):
        random_h = np.random.uniform(0, 1, n_points)
        random_h = random_h * (target_sum / np.sum(random_h))
        initial_guesses.append(random_h)
    
    # For each initial guess, run the optimization
    for guess in initial_guesses:
        # Perturbation strategy
        num_perturbations = 500  # Increased to allow more exploration
        phase_shift = 0.2  # Larger initial noise magnitude
        phase_decay = 0.95  # Slightly slower decay for more exploration
        
        current_best_c5 = float('inf')
        current_best_h = guess.copy()
        
        for perturbation_idx in range(num_perturbations):
            # Adaptive noise magnitude based on phase
            noise_scale = phase_shift * (phase_decay ** perturbation_idx)
            
            # Generate perturbed guess
            perturbed = current_best_h + np.random.uniform(-noise_scale, noise_scale, n_points)
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
            
            # Objective function
            def objective(h):
                one_minus_h = 1.0 - h
                corr = np.correlate(h, one_minus_h, mode='full')
                return np.max(corr) * dx
            
            # Optimization with SLSQP
            bounds = [(0.0, 1.0) for _ in range(n_points)]
            cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
            
            result = minimize(
                objective,
                perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'ftol': 1e-9, 'maxiter': 300}
            )
            
            if result.success and result.fun < current_best_c5:
                current_best_c5 = result.fun
                current_best_h = result.x
            
            elapsed = time.time() - start_time
            if elapsed >= budget_s:
                break
        
        # Update the global best
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = current_best_h
    
    return best_h, best_c5, n_points
