            if i * dx < interval_length:
                binary[i] = 1.0
        initial_guesses.append(binary)
    # Add checkerboard pattern
    checkerboard = np.zeros(n_points)
    for i in range(n_points):
        checkerboard[i] = 1.0 if i % 2 == 0 else 0.0
    initial_guesses.append(checkerboard)
    # Add random guesses
    for _ in range(3):
        guess = np.random.uniform(0, 1, n_points)
        guess = guess * (target_sum / np.sum(guess))
        initial_guesses.append(guess)
    
    # Optimization parameters
    max_iterations = 100  # Adjust based on time budget
    
    for guess in initial_guesses:
        # Ensure the guess meets the sum constraint
        guess = guess * (target_sum / np.sum(guess))
        # Define bounds
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        # Define objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_overlap = np.max(corr) * dx
            penalty = 1e4 * (np.sum(h) - target_sum)**2
            return max_overlap + penalty
        # Run differential evolution
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=max_iterations,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed
        )
        # Check if the result is successful
        if result.success:
            current_c5 = result.fun - 1e4 * (np.sum(result.x) - target_sum)**2  # Subtract penalty
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        # Check time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
