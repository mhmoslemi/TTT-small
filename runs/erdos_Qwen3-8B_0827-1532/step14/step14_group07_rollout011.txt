            if i < n_points // 2:

            elif i > n_points // 2:

        symmetric = symmetric * (target_sum / np.sum(symmetric))
        guesses.append(symmetric)
        # Quarters pattern
        quarters = np.zeros(n_points)

        quarters = quarters * (target_sum / np.sum(quarters))
        guesses.append(quarters)
        # Alternating pattern
        alternating = np.zeros(n_points)
