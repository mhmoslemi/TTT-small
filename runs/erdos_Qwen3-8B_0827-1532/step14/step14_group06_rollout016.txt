from scipy.optimize import minimize
import time

            guesses.append(binary)
        # Periodic patterns with different periods
        for period in [3, 4, 5]:
            periodic = np.zeros(n_points)
            for i in range(n_points):
                periodic[i] = 1 if (i % period == 0) else 0
            periodic = periodic * (target_sum / np.sum(periodic))
            guesses.append(periodic)
        # Spaced intervals with different spacing
        for spacing in [3, 4, 5]:
            spaced = np.zeros(n_points)
            for i in range(n_points):
                if (i % spacing == 0) or (i % spacing == 1):
                    spaced[i] = 1
            spaced = spaced * (target_sum / np.sum(spaced))
            guesses.append(spaced)
        # Random perturbations of initial guess
        rand_guess = initial_guess + np.random.uniform(-0.2, 0.2, n_points)
        rand_guess = np.clip(rand_guess, 0, 1)
        scale = target_sum / np.sum(rand_guess)
        rand_guess = rand_guess * scale
        guesses.append(rand_guess)
        return guesses
    
    initial_guesses = generate_initial_guesses()
    
    # Adaptive perturbation strategy with varied exploration
    num_perturbations = 500  # Increased for better exploration
    phase_shift = 0.5  # Larger initial noise magnitude
    phase_decay = 0.98  # Slower decay for more exploration
    
    for guess in initial_guesses:
        # Clone and normalize
        current_h = guess.copy()
        current_sum = np.sum(current_h)
        current_h = current_h * (target_sum / current_sum)
        
        for perturbation_idx in range(num_perturbations):
            # Adaptive noise magnitude based on phase
            noise_scale = phase_shift * (phase_decay ** perturbation_idx)
            
            # Generate perturbed guess
            noise = np.random.uniform(-noise_scale, noise_scale, n_points)
            perturbed = current_h + noise

            scale = target_sum / current_sum
            perturbed = perturbed * scale
            
            # Objective function
            def objective(h):
                one_minus_h = 1.0 - h
                corr = np.correlate(h, one_minus_h, mode='full')
                return np.max(corr) * dx
