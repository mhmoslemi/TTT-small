        objective,
        bounds,
        strategy='best1bin',
        popsize=30,
        mutation=(0.5, 1.0),
        crossover=0.9,
        maxiter=100,
        tol=1e-10,
        disp=False,
        vectorized=True
    )
    
    # Check if the result is successful
    if result.success:
        best_h = result.x
        best_c5 = result.fun
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        # If time is up, return the best found so far
        return best_h, best_c5, n_points
    else:
        # If time is not up, perform a final refinement with SLSQP
        def objective_slsqp(h):
            h = h.copy()
            h_sum = np.sum(h)
            if h_sum == 0:
                return float('inf')
            h = h * (target_sum / h_sum)
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result_slsqp = minimize(
            objective_slsqp,
            best_h,
            method='SLSQP',
            bounds=bounds_slsqp,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 100}
        )
        
        if result_slsqp.success and result_slsqp.fun < best_c5:
            best_c5 = result_slsqp.fun
            best_h = result_slsqp.x
        
        return best_h, best_c5, n_points
