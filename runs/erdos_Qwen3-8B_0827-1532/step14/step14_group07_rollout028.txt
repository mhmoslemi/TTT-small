        rand_guess = rand_guess * scale
        guesses.append(rand_guess)

        # Random binary pattern with varying density

        binary_var_sum = np.sum(binary_var)
        binary_var = binary_var * (target_sum / binary_var_sum)
        guesses.append(binary_var)
        # Random positions
        random_positions = np.random.choice(n_points, size=int(n_points // 2), replace=False)
        random_h = np.zeros(n_points)
        random_h[random_positions] = 1
        random_h_sum = np.sum(random_h)
        random_h = random_h * (target_sum / random_h_sum)
        guesses.append(random_h)
        # Symmetric pattern
        symmetric = np.zeros(n_points)
        for i in range(n_points):
            if i < n_points // 2:
                symmetric[i] = 1
            elif i > n_points // 2:
                symmetric[i] = 1
        symmetric = symmetric * (target_sum / np.sum(symmetric))
        guesses.append(symmetric)
        return guesses
    
    initial_guesses = generate_initial_guesses()

    num_perturbations = 800  # Increased for more exploration

    for guess in initial_guesses:
        # Clone and normalize
        current_h = guess.copy()
        current_sum = np.sum(current_h)
        current_h = current_h * (target_sum / current_sum)
        
        for perturbation_idx in range(num_perturbations):
            # Adaptive noise magnitude based on phase
            noise_scale = phase_shift * (phase_decay ** perturbation_idx)
            
            # Generate perturbed guess
            noise = np.random.uniform(-noise_scale, noise_scale, n_points)
            perturbed = current_h + noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
            
            # Objective function
            def objective(h):
                one_minus_h = 1.0 - h
                corr = np.correlate(h, one_minus_h, mode='full')
                return np.max(corr) * dx
            
            # Optimization with SLSQP
            bounds = [(0.0, 1.0) for _ in range(n_points)]
            cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
