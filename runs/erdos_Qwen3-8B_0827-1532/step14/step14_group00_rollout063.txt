from scipy.optimize import minimize
import time

    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate multiple initial guesses
    num_initial_guesses = 10
    initial_guesses = []
    for _ in range(num_initial_guesses):
        perturbed = initial_guess + np.random.uniform(-0.15, 0.15, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        initial_guesses.append(perturbed)
    
    # Optimize each initial guess with L-BFGS-B
    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with L-BFGS-B
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 150}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Perturbation phase on the best_h found so far

            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')

        # Optimization with L-BFGS-B
