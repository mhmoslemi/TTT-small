                best_c5 = result.fun
        else:
            # Use SLSQP for local refinement
            result = minimize(
                objective,
                perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'ftol': 1e-10, 'maxiter': max_iterations}
            )
            if result.success and result.fun < best_c5:
                best_c5 = result.fun
                best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
