        guesses.append(rand_guess)
        # Periodic pattern
        periodic = np.zeros(n_points)
        for i in range(n_points):
            periodic[i] = 1 if (i % 5 == 0) else 0
        periodic = periodic * (target_sum / np.sum(periodic))
        guesses.append(periodic)
        # Spaced intervals
        spaced = np.zeros(n_points)
        for i in range(n_points):
            if (i % 7 == 0) or (i % 7 == 1) or (i % 7 == 2):
                spaced[i] = 1
        spaced = spaced * (target_sum / np.sum(spaced))
        guesses.append(spaced)
        # Random binary pattern
        binary = np.random.choice([0, 1], size=n_points)
        binary_sum = np.sum(binary)
        binary = binary * (target_sum / binary_sum)
        guesses.append(binary)
        # Half-interval pattern
        half_interval = np.zeros(n_points)
        for i in range(n_points):
            if i < n_points / 2:
                half_interval[i] = 1
        half_interval = half_interval * (target_sum / np.sum(half_interval))
        guesses.append(half_interval)
        # Uniform pattern
        uniform = np.ones(n_points) * (target_sum / n_points)
        guesses.append(uniform)
        return guesses
    
    initial_guesses = generate_initial_guesses()
    
    # Adaptive perturbation strategy with varying noise and exploration phases
    num_perturbations = 200  # Reduced for better exploration efficiency
    phase_shift = 0.3  # Larger initial noise magnitude
    phase_decay = 0.9  # Slower decay for more exploration
    
    for guess in initial_guesses:
        # Clone and normalize
        current_h = guess.copy()
        current_sum = np.sum(current_h)
        current_h = current_h * (target_sum / current_sum)
        
        for perturbation_idx in range(num_perturbations):
            # Adaptive noise magnitude based on phase
            noise_scale = phase_shift * (phase_decay ** perturbation_idx)
            
            # Generate perturbed guess
            noise = np.random.uniform(-noise_scale, noise_scale, n_points)
            perturbed = current_h + noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
            
            # Objective function
            def objective(h):
                one_minus_h = 1.0 - h
                corr = np.correlate(h, one_minus_h, mode='full')
                return np.max(corr) * dx
            
            # Optimization with SLSQP
            bounds = [(0.0, 1.0) for _ in range(n_points)]
            cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
            
            result = minimize(
                objective,
                perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'ftol': 1e-9, 'maxiter': 200}
            )
            
            if result.success and result.fun < best_c5:
                best_c5 = result.fun
                best_h = result.x
            
            elapsed = time.time() - start_time
            if elapsed >= budget_s:
                break
    
    return best_h, best_c5, n_points
