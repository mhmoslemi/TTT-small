        # Clone and normalize
        current_h = guess.copy()
        current_sum = np.sum(current_h)
        current_h = current_h * (target_sum / current_sum)
        
        # Define objective function with normalization
        def objective(h):
            h_normalized = h * (target_sum / np.sum(h))
            one_minus_h = 1.0 - h_normalized
            corr = np.correlate(h_normalized, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Perform differential evolution
        result = differential_evolution(objective, bounds, strategy='best1bin', 
                                       maxiter=100, popsize=15, tol=1e-6, mutation=(0.5, 1), recombination=0.8)
        
        # Check if the result is successful
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
