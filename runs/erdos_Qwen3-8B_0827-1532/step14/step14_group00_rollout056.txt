from scipy.optimize import minimize
import time
from concurrent.futures import ProcessPoolExecutor, as_completed

def run(seed=42, budget_s=100, **kwargs):

    start_time = time.time()
    
    # Generate initial guesses with structured perturbations
    def generate_initial_guesses(base, num_guesses=10):
        guesses = []
        for i in range(num_guesses):
            # Add noise with varying amplitude
            noise_scale = 0.15 * (0.95 ** i)
            noise = np.random.uniform(-noise_scale, noise_scale, n_points)
            perturbed = base + noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
            guesses.append(perturbed)
        return guesses
    
    initial_guesses = generate_initial_guesses(initial_guess, num_guesses=20)
    
    # Function to optimize a single guess
    def optimize_guess(guess):
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 300}
        )
        return result.x, result.fun
    
    # Parallel optimization of initial guesses
    with ProcessPoolExecutor(max_workers=10) as executor:
        futures = [executor.submit(optimize_guess, guess) for guess in initial_guesses]
        for future in as_completed(futures):
            h_candidate, c5_candidate = future.result()
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_candidate
    
    # Perturbation phase with parallel processing
    num_perturbations = 300
    phase_shift = 0.15
    phase_decay = 0.96
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Optimize the perturbed guess
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(

            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 300}
        )
