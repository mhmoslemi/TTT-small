from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    start_time = time.time()
    
    # Generate diverse initial guesses based on initial_h_values
    def generate_initial_guesses():
        guesses = [initial_guess.copy()]
        # Add noise
        noise = np.random.normal(0, 0.1, n_points)
        noisy = initial_guess + noise
        noisy = np.clip(noisy, 0, 1)
        scale = target_sum / np.sum(noisy)
        noisy = noisy * scale
        guesses.append(noisy)
        # Shift
        shifted = np.roll(initial_guess, 1)
        shifted = shifted * (target_sum / np.sum(shifted))
        guesses.append(shifted)
        # Scale up/down
        scaled_up = initial_guess * 1.1
        scaled_up = np.clip(scaled_up, 0, 1)
        scaled_up = scaled_up * (target_sum / np.sum(scaled_up))
        guesses.append(scaled_up)
        scaled_down = initial_guess * 0.9
        scaled_down = np.clip(scaled_down, 0, 1)
        scaled_down = scaled_down * (target_sum / np.sum(scaled_down))
        guesses.append(scaled_down)
        return guesses
    
    initial_guesses = generate_initial_guesses()
    
    # Adaptive optimization with L-BFGS-B for better convergence
    for guess in initial_guesses:
        # Clone and normalize
        current_h = guess.copy()
        current_sum = np.sum(current_h)
        current_h = current_h * (target_sum / current_sum)
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with L-BFGS-B
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(

        if result.success and result.fun < best_c5:
