                bounds,
                constraints=cons,
                popsize=15,
                strategy='best1bin',
                tol=1e-6,
                maxiter=50,
                mutation=(0.5, 1),
                recombination=0.8,
                polish=True,
                disp=False
            )
            if result.success and result.fun < best_c5:
                best_c5 = result.fun
                best_h = result.x
        else:
            result = minimize(
                objective,
                perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'ftol': 1e-10, 'maxiter': max_iterations}
            )
            if result.success and result.fun < best_c5:
                best_c5 = result.fun
                best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
