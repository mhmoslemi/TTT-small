def run(seed=42, budget_s=100, **kwargs):

        rand_guess = np.clip(rand_guess, 0, 1)

        rand_guess = rand_guess * scale
        guesses.append(rand_guess)
        # Periodic pattern
        periodic = np.zeros(n_points)
        for i in range(n_points):

        guesses.append(periodic)

        for i in range(n_points):
            if (i % 5 == 0) or (i % 5 == 1):

        spaced = spaced * (target_sum / np.sum(spaced))
        guesses.append(spaced)
        # Random binary pattern
        binary = np.random.choice([0, 1], size=n_points)
        binary_sum = np.sum(binary)
        binary = binary * (target_sum / binary_sum)
        guesses.append(binary)
        # Symmetric pattern
        symmetric = np.zeros(n_points)
        for i in range(n_points):
            if i < n_points // 2:
                symmetric[i] = 1
            elif i > n_points // 2:
                symmetric[i] = 1
        symmetric = symmetric * (target_sum / np.sum(symmetric))
        guesses.append(symmetric)
        # Random binary with higher density
        binary_high = np.random.choice([0, 1], size=n_points, p=[0.2, 0.8])
        binary_high_sum = np.sum(binary_high)
        binary_high = binary_high * (target_sum / binary_high_sum)
        guesses.append(binary_high)
        # Random binary with lower density
        binary_low = np.random.choice([0, 1], size=n_points, p=[0.8, 0.2])
        binary_low_sum = np.sum(binary_low)
        binary_low = binary_low * (target_sum / binary_low_sum)
        guesses.append(binary_low)
        return guesses
    
    initial_guesses = generate_initial_guesses()
    
    # Adaptive perturbation strategy with varying noise and exploration phases
    num_perturbations = 300  # Reduced for better exploration
    phase_shift = 0.25  # Larger initial noise magnitude
    phase_decay = 0.95  # Slower decay for more exploration
    
    for guess in initial_guesses:
        # Clone and normalize
        current_h = guess.copy()
        current_sum = np.sum(current_h)
        current_h = current_h * (target_sum / current_sum)
        
        for perturbation_idx in range(num_perturbations):
            # Adaptive noise magnitude based on phase
            noise_scale = phase_shift * (phase_decay ** perturbation_idx)
            
            # Generate perturbed guess
            noise = np.random.uniform(-noise_scale, noise_scale, n_points)
            perturbed = current_h + noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
            
            # Objective function
            def objective(h):
                one_minus_h = 1.0 - h
                corr = np.correlate(h, one_minus_h, mode='full')
                return np.max(corr) * dx
            
            # Optimization with COBYLA
            bounds = [(0.0, 1.0) for _ in range(n_points)]
            cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    return best_h, best_c5, n_points
