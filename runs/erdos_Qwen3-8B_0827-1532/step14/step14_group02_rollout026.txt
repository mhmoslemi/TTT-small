            current_best_h = result_cobyla.x
        
        # Update the global best
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = current_best_h
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
