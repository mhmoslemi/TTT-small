import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')

    def objective(h):

        corr = np.correlate(h_normalized, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Run differential evolution
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=15,
        maxiter=100,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.9,
        seed=seed
    )
    
    # Check if the result is successful
    if result.success:
        best_h = result.x
        best_c5 = result.fun
    
    # Try a few random initial guesses to escape local minima
    for _ in range(3):
