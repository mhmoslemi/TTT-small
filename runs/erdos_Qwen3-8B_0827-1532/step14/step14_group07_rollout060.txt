import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate diverse initial guesses
    def generate_initial_guesses():
        guesses = []
        # Original initial guess
        guesses.append(initial_guess.copy())
        # Random binary pattern
        binary = np.random.choice([0, 1], size=n_points)
        binary_sum = np.sum(binary)
        binary = binary * (target_sum / binary_sum)
        guesses.append(binary)
        # Periodic pattern
        periodic = np.zeros(n_points)
        for i in range(n_points):
            periodic[i] = 1 if (i % 3 == 0) else 0
        periodic = periodic * (target_sum / np.sum(periodic))
        guesses.append(periodic)
        # Spaced intervals
        spaced = np.zeros(n_points)
        for i in range(n_points):
            if (i % 5 == 0) or (i % 5 == 1):
                spaced[i] = 1
        spaced = spaced * (target_sum / np.sum(spaced))
        guesses.append(spaced)
        # Symmetric pattern
        symmetric = np.zeros(n_points)
        for i in range(n_points):
            if i < n_points // 2:
                symmetric[i] = 1
            elif i > n_points // 2:
                symmetric[i] = 1
        symmetric = symmetric * (target_sum / np.sum(symmetric))
        guesses.append(symmetric)
        # Sine wave pattern
        sine = np.sin(np.linspace(0, 2 * np.pi, n_points))
        sine = (sine + 1) / 2  # Scale to [0, 1]
        sine = sine * (target_sum / np.sum(sine))
        guesses.append(sine)
        # Random perturbation of initial guess
        rand_guess = initial_guess + np.random.uniform(-0.15, 0.15, n_points)
        rand_guess = np.clip(rand_guess, 0, 1)
        scale = target_sum / np.sum(rand_guess)
        rand_guess = rand_guess * scale
        guesses.append(rand_guess)
        return guesses
    
    initial_guesses = generate_initial_guesses()
    
    # Adaptive optimization with multiple methods and exploration
    num_optimizations = 100  # Reduced for better exploration
    for guess in initial_guesses:
        # Clone and normalize
        current_h = guess.copy()
        current_sum = np.sum(current_h)
        current_h = current_h * (target_sum / current_sum)
        
        for opt_idx in range(num_optimizations):
            # Objective function
            def objective(h):
                one_minus_h = 1.0 - h
                corr = np.correlate(h, one_minus_h, mode='full')
                return np.max(corr) * dx
            
            # Optimization with COBYLA
            bounds = [(0.0, 1.0) for _ in range(n_points)]
            cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
            
            result = minimize(
                objective,
                current_h,
                method='COBYLA',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 100, 'disp': False}
            )
            
            if result.success and result.fun < best_c5:
                best_c5 = result.fun
                best_h = result.x
            
            elapsed = time.time() - start_time
            if elapsed >= budget_s:
                break
    
    return best_h, best_c5, n_points
