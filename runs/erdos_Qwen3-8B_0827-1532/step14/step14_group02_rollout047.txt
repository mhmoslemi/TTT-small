        strategy='best1bin',
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.8,
        tol=1e-9,
        maxiter=100,
        polish=True,
        disp=False
    )
    if result.success and result.fun < best_c5:
        best_c5 = result.fun
        best_h = result.x
    
    # Check elapsed time
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
