from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate a few perturbations of the initial guess
    perturbations = [initial_guess.copy()]
    for _ in range(2):
        noise = np.random.normal(0, 0.1, n_points)
        perturbed = initial_guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * (target_sum / np.sum(perturbed))
        perturbations.append(perturbed)
    
    # Define objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    # Define constraint
    def constraint(h):
        return np.sum(h) - target_sum
    
    # Optimization settings
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint})
    
    # Run optimization for each perturbation
    for guess in perturbations:
        # Use SLSQP with more iterations and tighter tolerances
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 500}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
