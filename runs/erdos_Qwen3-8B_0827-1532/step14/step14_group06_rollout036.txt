        guesses.append(periodic)
        # Spaced intervals
        spaced = np.zeros(n_points)
        for i in range(n_points):
            if (i % 7 == 0) or (i % 7 == 1) or (i % 7 == 2):

        spaced = spaced * (target_sum / np.sum(spaced))
        guesses.append(spaced)

        binary = binary * (target_sum / binary_sum)
        guesses.append(binary)
        # Random step pattern
        step = np.zeros(n_points)
        for i in range(n_points):
            if i % 3 == 0:
                step[i] = 1
        step = step * (target_sum / np.sum(step))
        guesses.append(step)
        return guesses
    
    initial_guesses = generate_initial_guesses()
    
    # Adaptive perturbation strategy with varying noise and exploration phases
    num_perturbations = 200  # Reduced for better exploration efficiency

    phase_decay = 0.98  # Slower decay for more exploration
    
    for guess in initial_guesses:
        # Clone and normalize
        current_h = guess.copy()
        current_sum = np.sum(current_h)
        current_h = current_h * (target_sum / current_sum)
        
        for perturbation_idx in range(num_perturbations):
            # Adaptive noise magnitude based on phase
            noise_scale = phase_shift * (phase_decay ** perturbation_idx)
            
            # Generate perturbed guess
            noise = np.random.uniform(-noise_scale, noise_scale, n_points)
            perturbed = current_h + noise

            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
            
            # Objective function
            def objective(h):
                one_minus_h = 1.0 - h
                corr = np.correlate(h, one_minus_h, mode='full')
                return np.max(corr) * dx
