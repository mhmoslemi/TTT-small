import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Prepare initial guess for differential evolution
    initial_x = initial_guess[:-1]
    last = target_sum - np.sum(initial_x)
    initial_x = np.clip(initial_x, 0, 1)
    
    # Define objective function
    def objective(x):
        sum_x = np.sum(x)
        last = target_sum - sum_x
        h = np.concatenate([x, [last]])
        h = np.clip(h, 0, 1)
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    # Bounds for the first n_points-1 variables
    bounds = [(0.0, 1.0) for _ in range(n_points - 1)]
    
    start_time = time.time()
    
    # Run differential evolution with time constraint
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed,
        polish=True,
        disp=False
    )
    
    # Extract the best solution
    best_x = result.x
    last = target_sum - np.sum(best_x)
    best_h = np.concatenate([best_x, [last]])
    best_h = np.clip(best_h, 0, 1)
    
    # Check elapsed time and return early if needed
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return best_h, result.fun, n_points
    
    return best_h, result.fun, n_points
