    structured_guesses.append(structured_guess)

    for _ in range(2):

        random_h = random_h * (target_sum / np.sum(random_h))
        structured_guesses.append(random_h)
    
    initial_guesses.extend(structured_guesses)
    
    # For each initial guess, run the optimization
    for guess in initial_guesses:
        # Perturbation strategy
        num_perturbations = 200  # Reduced to save time
        phase_shift = 0.15  # Initial noise magnitude
        phase_decay = 0.98  # Decay factor for noise magnitude
        
        current_best_c5 = float('inf')
        current_best_h = guess.copy()
        
        for perturbation_idx in range(num_perturbations):
            # Adaptive noise magnitude based on phase
            noise_scale = phase_shift * (phase_decay ** perturbation_idx)
            
            # Generate perturbed guess
            perturbed = current_best_h + np.random.uniform(-noise_scale, noise_scale, n_points)
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
            
            # Objective function
            def objective(h):
                one_minus_h = 1.0 - h
                corr = np.correlate(h, one_minus_h, mode='full')
                return np.max(corr) * dx
            
            # Optimization with SLSQP
            bounds = [(0.0, 1.0) for _ in range(n_points)]
            cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
            
            result = minimize(
                objective,
                perturbed,
                method='SLSQP',

                constraints=cons,
