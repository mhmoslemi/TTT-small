from scipy.optimize import minimize
import time

    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate diverse initial guesses
    initial_guesses = []

    # Generate random binary guesses
    for _ in range(5):
        binary_guess = np.zeros(n_points, dtype=np.float64)
        indices = np.random.choice(n_points, size=int(target_sum), replace=False)
        binary_guess[indices] = 1.0
        initial_guesses.append(binary_guess)
    # Generate structured guesses (alternating 0.5)
    structured_guess = np.zeros(n_points)
    for i in range(n_points):
        structured_guess[i] = 0.5 if (i % 2 == 0) else 0.5
    initial_guesses.append(structured_guess)
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 100}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Perturbation phase on the best_h found so far
