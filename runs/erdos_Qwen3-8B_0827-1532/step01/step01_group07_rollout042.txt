import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    # Access initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    # Define objective function
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    # Define constraints
    cons = ({'type': 'eq', 'fun': lambda h_flat: np.sum(h_flat) - n_points / 2})
    bounds = [(0, 1) for _ in range(n_points)]

    # Generate initial guesses
    initial_guesses = [initial_h.copy()]

    # Add structured initial guess (every other sample)
    structured_h = np.zeros(n_points)
    structured_h[::2] = 1.0
    initial_guesses.append(structured_h)

    # Add random initial guesses
    for _ in range(3):
        h_rand = np.random.rand(n_points)
        h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
        initial_guesses.append(h_rand)

    best_c5 = float('inf')
    best_h = initial_h.copy()

    # Track elapsed time
    import time
    start_time = time.time()

    for guess in initial_guesses:
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
            # Early exit if time is up
            if time.time() - start_time > budget_s - 1:
                break

    return (best_h, best_c5, n_points)
