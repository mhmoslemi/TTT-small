            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8}
        )
        
        # Check if the optimization completed
        elapsed_time = time.time() - start_time
        if elapsed_time >= budget_s:
            break
        
        # Update best result
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # If time is up, return the best result found
    return best_h, best_c5, n_points
