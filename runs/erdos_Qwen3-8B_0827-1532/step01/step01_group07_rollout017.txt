import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    # Access initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Define objective function
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define constraints
    cons = ({'type': 'eq', 'fun': lambda h_flat: np.sum(h_flat) - n_points / 2})
    bounds = [(0, 1) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = initial_h.copy()
    
    # Initial guess
    result = minimize(objective, initial_h, method='L-BFGS-B', bounds=bounds, constraints=cons, options={'maxiter': 200})
    if result.success:
        best_c5 = result.fun
        best_h = result.x
    
    # Generate diverse initial guesses
    for _ in range(5):
        # Perturb initial guess
        h_rand = initial_h.copy()

        h_rand += noise
        h_rand = np.clip(h_rand, 0, 1)
        h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
        
        # Run optimization
        result = minimize(objective, h_rand, method='L-BFGS-B', bounds=bounds, constraints=cons, options={'maxiter': 200})
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
