import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h_initial = initial_h_values.copy()

    def objective(h):

        max_value = np.max(correlate_result) * dx
        return max_value

    bounds = [(0.0, 1.0) for _ in range(n_points)]

        if result.success:

            return starting_h, objective(starting_h)

        current_sum = np.sum(perturbed_h)

        scale = target_sum / current_sum
