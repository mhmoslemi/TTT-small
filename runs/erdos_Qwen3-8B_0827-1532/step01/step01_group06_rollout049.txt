        return max_value + penalty
    
    result_de = differential_evolution(objective_with_penalty, bounds, strategy='best1bin', maxiter=100, popsize=15, tol=1e-6, mutation=(0.5, 1), recombination=0.8, disp=False)
    best_de_h = result_de.x
    best_de_c5 = objective(best_de_h)
    
    if best_de_c5 < best_c5:
        best_h, best_c5 = best_de_h, best_de_c5
    
    return (best_h, best_c5, n_points)
