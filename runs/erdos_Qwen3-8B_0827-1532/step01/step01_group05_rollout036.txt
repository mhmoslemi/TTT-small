        start_idx = n_points - 3
        end_idx = n_points + 1
        relevant_corr = corr[start_idx:end_idx+1]
        max_corr = np.max(relevant_corr)
        return max_corr * dx  # Scale by dx as per the problem statement
    
    # Define the constraint: sum(h) must equal n_points / 2
    constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
    
    # Define bounds for each h[i] to be in [0, 1]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial guess
    h0 = initial_h.copy()
    
    # Perform optimization using a different method, like L-BFGS-B
    result = minimize(
        objective,
        h0,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=constraint,
        options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8}
    )
    
    # Extract the best solution found
    best_h = result.x
    c5_bound = result.fun
    
    return (best_h, c5_bound, n_points)
