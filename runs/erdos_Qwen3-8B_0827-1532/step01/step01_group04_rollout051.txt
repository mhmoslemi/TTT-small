    n_points = 100
    dx = 2.0 / n_points
    
    # Generate initial guess by interpolating initial_h_values to n_points
    original_h = initial_h_values
    x_original = np.linspace(0, 2, len(original_h))
    x_new = np.linspace(0, 2, n_points)
    h_initial = np.interp(x_new, x_original, original_h)
    
    # Ensure the sum is correct (due to interpolation, it might not be)
    # Adjust h_initial to have sum n_points / 2
    scale_factor = (n_points / 2) / np.sum(h_initial)
    h_initial = h_initial * scale_factor
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraints: sum(h) == n_points / 2
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Start optimization
    start_time = time.time()
    result = minimize(
        objective,
        h_initial,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-8, 'maxiter': 200}
    )
    
    # Check if we ran out of time
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return h_initial, evaluate_erdos_solution(h_initial), n_points
    
    h_values = result.x
    c5_bound = result.fun
    return h_values, c5_bound, n_points
