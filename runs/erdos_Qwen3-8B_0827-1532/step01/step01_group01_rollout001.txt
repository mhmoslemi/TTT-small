        h = h / np.sum(h) * (n_points / 2)
        initial_guesses.append(h)
    
    # Perturb initial guess
    perturbed = initial_h_values.copy()
    perturbed += np.random.normal(0, 0.05, n_points)
    perturbed = np.clip(perturbed, 0, 1)
    perturbed = perturbed / np.sum(perturbed) * (n_points / 2)
    initial_guesses.append(perturbed)
    
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    dx = 2.0 / n_points
    
    def objective(h):
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv) * dx
    
    for guess in initial_guesses:
        h_initial = guess.copy()
        constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
        bounds = Bounds(0, 1)
        
        result = minimize(objective, h_initial, method='SLSQP', bounds=bounds, constraints=constraint, tol=1e-6)
        elapsed = time.time() - start_time
        if elapsed < budget_s:
            result = minimize(objective, result.x, method='SLSQP', bounds=bounds, constraints=constraint, tol=1e-8)
        
        current_c5 = evaluate_erdos_solution(result.x, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x
    
    return best_h, best_c5, n_points
