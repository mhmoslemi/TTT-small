import numpy as np
from scipy.optimize import minimize, Bounds, LinearConstraint
from scipy.optimize import COBYLA
import time

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h_values, 1 - h_values, mode='full')
    max_overlap = np.max(conv) * dx
    return max_overlap

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Constraint: sum(h) = n_points / 2
    constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
    bounds = Bounds(0, 1)
    
    # Objective function
    def objective(h):
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv) * dx
    
    # Generate multiple initial guesses
    initial_guesses = [initial_h.copy()]
    for _ in range(3):
        perturb = np.random.normal(0, 0.05, n_points)
        guess = initial_h + perturb
        guess = np.clip(guess, 0, 1)
        initial_guesses.append(guess)
    
    best_h = initial_h
    best_c5 = evaluate_erdos_solution(best_h, n_points)
    start_time = time.time()
    
    for guess in initial_guesses:
        # Clip to [0,1]
        guess = np.clip(guess, 0, 1)
        # Run optimization
        result = minimize(objective, guess, method=COBYLA, bounds=bounds, constraints=constraint, tol=1e-6)
        if result.success:
            current_h = result.x
            current_c5 = evaluate_erdos_solution(current_h, n_points)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
        # Check time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
