def run(seed=42, budget_s=100, **kwargs):
    n_points = 80
    dx = 2.0 / n_points

    # Function to generate a perturbed initial guess
    def generate_perturbed_guess(h, seed=None):
        np.random.seed(seed)
        perturbation = np.random.normal(0, 0.05, size=len(h))
        perturbed = h + perturbation
        perturbed = np.clip(perturbed, 0.0, 1.0)
        # Ensure the sum is approximately n_points / 2
        sum_perturbed = np.sum(perturbed)
        if sum_perturbed != 0:
            scale = (n_points / 2) / sum_perturbed
            perturbed *= scale
        return perturbed

    # Define the objective function
    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    # Define the constraint: sum(h) = n_points / 2
    constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}

    # Bounds for h
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Generate multiple initial guesses
    initial_guesses = []
    for i in range(5):  # Try 5 different initial guesses
        seed_i = seed + i
        guess = generate_perturbed_guess(initial_h, seed_i)
        initial_guesses.append(guess)

    best_result = None
    best_c5 = float('inf')

    # Run optimization for each initial guess
    for guess in initial_guesses:
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=constraint,
            options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8}

        if result.success:
            current_c5 = result.fun

            # Fallback: use the initial guess if optimization fails
            current_c5 = objective(guess)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_result = guess

    return (best_result, best_c5, n_points)
