    for _ in range(num_restarts):
        initial_guesses.append(generate_binary_initial())

    for initial_guess in initial_guesses:
        current_sum = np.sum(initial_guess)
        scale_factor = target_sum / current_sum
        h_perturbed = initial_guess * scale_factor
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)

        start_time = time.time()
        result = minimize(objective, h_perturbed, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        elapsed += time.time() - start_time

        if elapsed >= budget_s:
            break

        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            current_c5 = objective(h_perturbed)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_perturbed

    return (best_h, best_c5, n_points)
