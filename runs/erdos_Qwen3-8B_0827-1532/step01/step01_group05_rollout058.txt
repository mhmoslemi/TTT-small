import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    n_points = 80
    dx = 2.0 / n_points
    
    # Generate a random initial guess using Dirichlet distribution
    np.random.seed(seed)
    h_initial = np.random.dirichlet(np.ones(n_points)) * (n_points / 2)
    
    # Define the objective function
    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint: sum(h) = n_points / 2
    constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization with L-BFGS-B
    result = minimize(
        objective,
        h_initial,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=constraint,
        options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8}
    )
    
    best_h = result.x
    c5_bound = result.fun
    
    return (best_h, c5_bound, n_points)
