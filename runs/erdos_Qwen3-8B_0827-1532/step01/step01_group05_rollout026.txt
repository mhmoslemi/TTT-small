    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Perform optimization using differential evolution with adjusted parameters
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=15,
        mutation=0.5,
        recombination=0.8,
        tol=1e-8,
        maxiter=200,
        seed=seed
    )
    
    # Extract the best solution found
    best_h = result.x
    # Compute the actual max overlap
    one_minus_h = 1 - best_h
    corr = correlate(best_h, one_minus_h, mode='full')
    max_corr = np.max(corr)
    c5_bound = max_corr * dx
    
    return (best_h, c5_bound, n_points)
