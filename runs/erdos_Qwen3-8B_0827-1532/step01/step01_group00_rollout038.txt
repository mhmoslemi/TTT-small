import numpy as np
from scipy.optimize import minimize, Bounds, LinearConstraint
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    
    # Define the objective function
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define constraints: sum(h) == n_points / 2 and 0 <= h[i] <= 1
    constraints = [LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])]
    bounds = Bounds(np.zeros(n_points), np.ones(n_points))
    
    # Generate initial guesses
    initial_guesses = []
    # Uniform distribution
    initial_guesses.append(np.full(n_points, 0.5))
    # Random binary pattern
    random_binary = np.zeros(n_points)
    indices = np.random.choice(n_points, n_points // 2, replace=False)
    random_binary[indices] = 1
    initial_guesses.append(random_binary)
    # Perturbed initial guess
    perturbed = np.random.uniform(0.4, 0.6, size=n_points)
    initial_guesses.append(perturbed)
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    for guess in initial_guesses:
        # Ensure the guess meets the integral constraint (already satisfied by construction)
        result = minimize(
            objective,
            guess,
            method='Nelder-Mead',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 50, 'xatol': 1e-8}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        elapsed_time = time.time() - start_time
        if elapsed_time >= budget_s:
            break
    
    return best_h, best_c5, n_points
