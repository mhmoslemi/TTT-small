    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    # Define objective function with penalty
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        # Penalty for constraint violation
        penalty = 1e4 * (np.sum(h) - n_points / 2)**2
        return max_corr * dx + penalty
    # Bounds
    bounds = [(0, 1) for _ in range(n_points)]
    # Start timer
    start_time = time.time()
    # Run differential evolution with adaptive parameters
    result = differential_evolution(
        objective, bounds, 
        strategy='best1bin', 
        popsize=15, 
        mutation=(0.5, 1), 
        recombination=0.9, 
        tol=1e-6, 
        maxiter=100, 
        seed=seed
    )
    # Calculate elapsed time
    elapsed = time.time() - start_time
    # Check if we need to stop early (if time is up)
    # But the differential_evolution function may not be interruptible
    # So, we have to rely on the maxiter parameter
    best_h = result.x
    best_c5 = objective(best_h)
    return (best_h, best_c5, n_points)
