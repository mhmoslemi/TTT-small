import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    # Use a higher number of points for better approximation
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate a reasonable initial guess: a step function with alternating blocks
    h_initial = np.zeros(n_points)
    h_initial[:n_points//2] = 1.0
    
    # Objective function to minimize the max overlap integral
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        # Penalty for constraint violation
        penalty = 1e4 * (np.sum(h) - target_sum)**2
        return max_corr * dx + penalty
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=15,
        tol=1e-8,
        mutation=(0.5, 1),
        recombination=0.8,
        maxiter=100,
        seed=seed
    )
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        # Fallback to initial guess if time runs out
        h_values = h_initial
        c5_bound = objective(h_initial)
        return h_values, c5_bound, n_points
    
    h_values = result.x
    c5_bound = result.fun
    return h_values, c5_bound, n_points
