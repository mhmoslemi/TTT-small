import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    # Use a moderate number of points for balance between resolution and speed
    n_points = 100
    dx = 2.0 / n_points
    initial_h = initial_h_values  # Assume this is a global variable

    # Define the objective function to minimize the maximum overlap integral
    def objective(h):
        # Compute cross-correlation between h and (1 - h)
        corr = correlate(h, 1 - h, mode='full')
        # The maximum value of the cross-correlation corresponds to the maximum overlap integral
        max_corr = np.max(corr)
        return max_corr * dx  # Scale by dx as per the problem statement

    # Define the constraint: sum(h) must equal n_points / 2
    constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}

    # Define bounds for each h[i] to be in [0, 1]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Initial guess
    h0 = initial_h.copy()

    # Use differential evolution for global optimization
    result = minimize(
        objective,
        h0,
        method='differential_evolution',
        bounds=bounds,
        constraints=constraint,
        options={
            'popsize': 15,
            'mutation': 0.5,
            'recombination': 0.8,
            'maxiter': 50,
            'tol': 1e-6,
            'mutation_scale': 0.1
        }
    )

    # Extract the best solution found
    best_h = result.x
    c5_bound = result.fun

    return (best_h, c5_bound, n_points)
