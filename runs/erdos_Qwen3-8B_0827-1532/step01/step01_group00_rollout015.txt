        new_h = np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])
        # Ensure sum is correct (should be 30)
        # But since we have exactly 30 ones and 30 zeros, it's already correct
        result = minimize(
            objective,
            new_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
        )
        elapsed_time = time.time() - start_time
        if elapsed_time >= budget_s:
            break

    return result.x, result.fun, n_points
