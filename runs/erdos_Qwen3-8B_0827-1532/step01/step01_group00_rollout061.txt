from scipy.optimize import minimize

import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60  # Increased for better resolution
    dx = 2.0 / n_points
    # Generate multiple initial guesses
    initial_guesses = []

        max_corr = np.max(corr)

    constraints = [LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])]
    bounds = Bounds(np.zeros(n_points), np.ones(n_points))

        if best_result is None or result.fun < best_result.fun:

    # Return the best solution found
