import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Objective function with penalty for sum constraint
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        penalty = 1e4 * abs(np.sum(h) - n_points / 2)
        return max_corr * dx + penalty
    
    # Bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    start_time = time.time()
    
    # Initial differential evolution run
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        disp=False
    )
    
    elapsed_time = time.time() - start_time
    if elapsed_time >= budget_s:
        return result.x, result.fun, n_points
    
    # Try multiple runs with different initializations
    for _ in range(3):
        # Generate random initial guess
        initial_guess = np.random.uniform(0, 1, n_points)
        # Run differential evolution with this initial guess
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=50,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            disp=False,
            x0=initial_guess
        )
        elapsed_time = time.time() - start_time
        if elapsed_time >= budget_s:
            break
    
    # Final check to ensure sum constraint is satisfied
    final_h = result.x
    final_sum = np.sum(final_h)
    if abs(final_sum - n_points / 2) > 1e-5:
        # Adjust last element to satisfy the constraint
        final_h[-1] = (n_points / 2) - (final_sum - final_h[-1])
        final_h = np.clip(final_h, 0, 1)
    
    return final_h, result.fun, n_points
