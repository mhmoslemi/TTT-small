        recombination=0.8,
        tol=1e-6,
        maxiter=100,
        seed=seed
    )
    
    best_h = result.x
    best_c5 = result.fun - 1e6 * abs(np.sum(best_h) - n_points / 2)
    
    return best_h, best_c5, n_points
