import numpy as np
from scipy.optimize import minimize, Bounds, LinearConstraint
from scipy.optimize import differential_evolution
import time

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h_values, 1 - h_values, mode='full')
    max_overlap = np.max(conv) * dx
    return max_overlap

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    # Start with initial_h_values (n=70)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
    bounds = Bounds(0, 1)
    
    # Objective function
    def objective(h):
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv) * dx
    
    # Generate multiple initial guesses
    num_perturbations = 5
    initial_guesses = []
    for _ in range(num_perturbations):
        perturbation = np.random.normal(0, 0.05, n_points)
        perturbed = initial_h + perturbation
        perturbed = np.clip(perturbed, 0, 1)
        initial_guesses.append(perturbed)
    
    best_c5 = float('inf')
    best_h = initial_h.copy()
    start_time = time.time()
    
    # Try each initial guess
    for guess in initial_guesses:
        result = minimize(objective, guess, method='SLSQP', bounds=bounds, constraints=constraint, tol=1e-6)
        if result.success:
            current_c5 = evaluate_erdos_solution(result.x, n_points)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        # Check time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    # Refine the best solution if time allows
    if elapsed < budget_s:
        result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=constraint, tol=1e-8)
        if result.success:
            best_c5 = evaluate_erdos_solution(result.x, n_points)
            best_h = result.x
    
    return best_h, best_c5, n_points
