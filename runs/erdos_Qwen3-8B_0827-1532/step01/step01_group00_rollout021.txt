    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=15,
        maxiter=100,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed
    )
    
    elapsed_time = time.time() - start_time
    if elapsed_time >= budget_s:
        return result.x, result.fun, n_points
    
    # Try a few more iterations with different starting points
    for _ in range(3):
        # Generate a random initial guess
        h_rand = np.random.uniform(0, 1, n_points)
        h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
        h_rand = np.clip(h_rand, 0, 1)
        
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=15,
            maxiter=50,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            x0=h_rand
        )
        elapsed_time = time.time() - start_time
        if elapsed_time >= budget_s:
            break
    
    # Return the best solution found
    return result.x, result.fun, n_points
