def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h_values, 1 - h_values, mode='full')
    max_overlap = np.max(conv) * dx
    return max_overlap

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    # Generate a better initial guess with more spread-out 1s
    n_points = 70
    h_initial = np.zeros(n_points)
    indices = np.random.permutation(n_points)
    h_initial[indices[:n_points//2]] = 1.0
    # Constraint: sum(h) = n_points / 2
    constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
    bounds = Bounds(0, 1)
    
    # Objective function
    def objective(h):
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv) * (2.0 / n_points)
    
    # Initial guess
    h_initial = h_initial.copy()
    
    # Optimization with COBYLA for better exploration
    start_time = time.time()
    result = minimize(objective, h_initial, method='COBYLA', bounds=bounds, constraints=constraint, tol=1e-6)
    
    # Refine with tighter tolerance if time allows
    elapsed = time.time() - start_time
    if elapsed < budget_s:
        result = minimize(objective, result.x, method='COBYLA', bounds=bounds, constraints=constraint, tol=1e-8)
    
    best_h = result.x
    c5_bound = evaluate_erdos_solution(best_h, n_points)
    return best_h, c5_bound, n_points
