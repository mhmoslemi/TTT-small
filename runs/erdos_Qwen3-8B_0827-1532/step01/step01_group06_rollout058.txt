import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 64  # Increased resolution for better approximation
    dx = 2.0 / n_points
    h_initial = np.random.uniform(0, 1, n_points)
    # Enforce sum constraint
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    
    def objective(h):
        correlate_result = np.correlate(h, 1 - h, mode='full')
        max_value = np.max(correlate_result) * dx
        return max_value
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    bounds = Bounds(0.0, 1.0)
    
    def optimize(starting_h):
        result = minimize(objective, starting_h, method='COBYLA', bounds=bounds, constraints={'type': 'eq', 'fun': constraint}, options={'maxiter': 200})
        if result.success:
            return result.x, result.fun
        else:
            return starting_h, objective(starting_h)
    
    best_h, best_c5 = optimize(h_initial)
    
    for _ in range(10):
        # Generate diverse perturbations
        perturbation = np.random.uniform(-0.2, 0.2, size=n_points)
        perturbed_h = h_initial + perturbation
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        current_sum = np.sum(perturbed_h)
        target_sum = n_points / 2
        scale = target_sum / current_sum
        perturbed_h *= scale
        h_perturbed, c5_perturbed = optimize(perturbed_h)
        if c5_perturbed < best_c5:
            best_h, best_c5 = h_perturbed, c5_perturbed
    
    return (best_h, best_c5, n_points)
