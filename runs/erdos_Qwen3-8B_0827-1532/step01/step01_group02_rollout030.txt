import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    # Initial guess from provided initial_h_values (n_points=70)
    h_initial = initial_h_values.copy()
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        # Compute cross-correlation
        corr = correlate(h, 1 - h, mode='full')
        # Find maximum value
        max_corr = np.max(corr)
        return max_corr * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    best_h = h_initial
    best_c5 = objective(best_h)
    elapsed = 0.0

    # Run multiple restarts with perturbations
    num_restarts = 10  # Increased number of restarts
    for _ in range(num_restarts):
        # Generate a random initial guess
        h_perturbed = np.random.uniform(0.0, 1.0, size=n_points)
        # Scale to meet sum constraint
        current_sum = np.sum(h_perturbed)
        scale_factor = target_sum / current_sum
        h_perturbed *= scale_factor

        # Optimize
        start_time = time.time()
        result = minimize(objective, h_perturbed, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        elapsed += time.time() - start_time

        if elapsed >= budget_s:

        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            # Fallback to perturbed solution
            current_c5 = objective(h_perturbed)
            if current_c5 < best_c5:
                best_c5 = current_c5

    return (best_h, best_c5, n_points)
