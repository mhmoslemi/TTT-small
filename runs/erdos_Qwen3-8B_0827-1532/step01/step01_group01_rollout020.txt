import numpy as np
from scipy.optimize import minimize, Bounds, LinearConstraint
from scipy.optimize import COBYLA
import time

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    # Correctly compute the overlap integral for each k in [0, 2]
    max_overlap = 0.0
    for k in np.linspace(0, 2, 1000):
        a = max(0.0, -k)
        b = min(2.0, 2.0 - k)
        if a >= b:
            continue
        # Find indices where x_i is in [a, b]
        indices = np.where((h_values >= a) & (h_values <= b))[0]
        overlap = 0.0
        for i in indices:
            x = h_values[i]
            x_plus_k = x + k
            if 0.0 <= x_plus_k <= 2.0:
                j = np.argmin(np.abs(h_values - x_plus_k))
                overlap += h_values[i] * (1.0 - h_values[j])
            else:
                overlap += h_values[i] * 1.0
        overlap *= dx
        if overlap > max_overlap:
            max_overlap = overlap
    return max_overlap

def generate_structured_initial_guess(n_points):
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    # Set h to 1 in [0, 0.5] and [1.0, 1.5]
    for i in range(int(0.5 / dx)):
        h[i] = 1.0
    for i in range(int(1.0 / dx), int(1.5 / dx) + 1):
        h[i] = 1.0
    return h

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    # Try multiple initial guesses
    initial_guesses = []
    # Original initial_h_values (n=70)
    initial_h_values = initial_h_values
    n_points = len(initial_h_values)
    initial_guesses.append(initial_h_values)
    # Structured initial guess for n_points=100
    n_points = 100
    structured_guess = generate_structured_initial_guess(n_points)
    initial_guesses.append(structured_guess)
    # Random initial guess
    random_guess = np.random.rand(n_points)
    random_guess = (random_guess * 2.0 - 1.0) * 0.5 + 0.5
    initial_guesses.append(random_guess)
    
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    
    for guess in initial_guesses:
        # Ensure the guess is of length n_points
        if len(guess) != n_points:
            guess = np.resize(guess, n_points)
        # Constraint: sum(h) = n_points / 2
        constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
        bounds = Bounds(0, 1)
        
        # Objective function
        def objective(h):
            return evaluate_erdos_solution(h, n_points)
        
        # Initial guess
        h_initial = guess.copy()
        
        # Optimization
        result = minimize(objective, h_initial, method='COBYLA', bounds=bounds, constraints=constraint, tol=1e-6)
        
        # Check if we have time left to refine
        elapsed = time.time() - start_time
        if elapsed < budget_s:
            # Try a few more iterations with a smaller tolerance
            result = minimize(objective, result.x, method='COBYLA', bounds=bounds, constraints=constraint, tol=1e-8)
        
        current_c5 = result.fun
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x
    
    return best_h, best_c5, n_points
