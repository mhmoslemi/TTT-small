import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60  # Increased number of points for better resolution
    dx = 2.0 / n_points
    initial_h = initial_h_values.copy()  # Assume initial_h_values is provided
    
    # Define the objective function with penalty for constraint
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        value = max_corr * dx
        # Constraint penalty to ensure sum(h) * dx = 1
        constraint_penalty = 1e6 * np.abs(np.sum(h) * dx - 1)
        return value + constraint_penalty
    
    # Bounds for each h[i]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Run differential evolution with adjusted parameters for efficiency
    start_time = time.time()
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed
    )
    
    # Check elapsed time and return best solution
    elapsed_time = time.time() - start_time
    if elapsed_time >= budget_s:
        return result.x, result.fun, n_points
    
    # Additional refinement with SLSQP if time permits
    from scipy.optimize import minimize, Bounds, LinearConstraint
    constraints = [LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])]
    bounds_slsqp = Bounds(np.zeros(n_points), np.ones(n_points))
    result_slsqp = minimize(
        objective,
        result.x,
        method='SLSQP',
        bounds=bounds_slsqp,
        constraints=constraints,
        options={'maxiter': 50, 'ftol': 1e-8, 'eps': 1e-8}
    )
    
    # Return the best result between DE and SLSQP
    final_result = result_slsqp if result_slsqp.fun < result.fun else result
    return final_result.x, final_result.fun, n_points
