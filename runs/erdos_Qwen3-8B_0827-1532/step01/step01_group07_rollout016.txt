        return max_corr * dx
    
    cons = ({'type': 'eq', 'fun': lambda h_flat: np.sum(h_flat) - n_points / 2})
    bounds = [(0, 1) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = initial_h.copy()
    
    # Initial guess
    result = minimize(objective, initial_h, method='L-BFGS-B', bounds=bounds, constraints=cons, options={'maxiter': 200})
    if result.success:
        best_c5 = result.fun
        best_h = result.x
    
    # Generate diverse initial guesses
