    constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Run optimization for each initial guess
    best_h = None
    best_c5 = float('inf')
    for h_initial in generate_initial_guesses():
        # Ensure h_initial is a float array
        h_initial = h_initial.astype(float)
        # Run SLSQP optimization
        result = minimize(
            objective,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints=constraint,
            options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-9}
        )
        # Update best result
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    return (best_h, best_c5, n_points)
