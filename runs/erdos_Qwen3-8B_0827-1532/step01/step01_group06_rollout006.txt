import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    # Set n_points to 100 for better resolution
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Generate diverse initial guesses
    def generate_initial_guesses():
        guesses = []
        # Uniform distribution
        guesses.append(np.full(n_points, 0.5))
        # Alternating pattern
        alt = np.zeros(n_points)
        alt[::2] = 1.0
        alt[1::2] = 0.0
        alt *= target_sum / np.sum(alt)
        guesses.append(alt)
        # Two blocks of 1s
        blocks = np.zeros(n_points)
        blocks[:n_points//3] = 1.0
        blocks[n_points//2:] = 1.0
        blocks *= target_sum / np.sum(blocks)
        guesses.append(blocks)
        # Random initial guess
        random_guess = np.random.rand(n_points)
        random_guess *= target_sum / np.sum(random_guess)
        guesses.append(random_guess)
        # Random perturbation of uniform
        perturbed_uniform = np.full(n_points, 0.5) + np.random.normal(0, 0.1, n_points)
        perturbed_uniform = np.clip(perturbed_uniform, 0, 1)
        perturbed_uniform *= target_sum / np.sum(perturbed_uniform)
        guesses.append(perturbed_uniform)
        return guesses

    initial_guesses = generate_initial_guesses()

    best_h = None
    best_c5 = float('inf')

    # Run optimization for each initial guess
    for guess in initial_guesses:
        def objective(h):
            correlate_result = np.correlate(h, 1 - h, mode='full')
            max_value = np.max(correlate_result) * dx
            return max_value

        def constraint(h):
            return np.sum(h) - target_sum

        bounds = [(0.0, 1.0) for _ in range(n_points)]

        result = minimize(objective, guess, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint}, options={'maxiter': 100})
        if result.success:
            current_h = result.x
            current_c5 = result.fun
        else:
            current_h = guess
            current_c5 = objective(guess)

        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5

    return (best_h, best_c5, n_points)
