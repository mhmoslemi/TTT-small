from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=100, **kwargs):

    # Initial guess from provided initial_h_values (n_points=70)
    h_initial = initial_h_values.copy()
    n_points = 50  # Reduced for faster evaluation
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        # Compute cross-correlation
        corr = correlate(h, 1 - h, mode='full')
        # Find maximum value
        max_corr = np.max(corr)
        return max_corr * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    best_h = h_initial
    best_c5 = objective(best_h)
    elapsed = 0.0

    # Generate diverse initial guesses
    initial_guesses = [h_initial]
    # Add a random guess
    random_guess = np.random.rand(n_points)
    random_guess = (random_guess * target_sum) / np.sum(random_guess)
    initial_guesses.append(random_guess)
    # Add a structured guess (alternating 0.5 and 0.5 with small variations)
    structured_guess = np.zeros(n_points)
    for i in range(n_points):
        structured_guess[i] = 0.5 + 0.1 * np.sin(2 * np.pi * i / n_points)
    structured_guess = (structured_guess * target_sum) / np.sum(structured_guess)
    initial_guesses.append(structured_guess)

    # Run multiple restarts with diverse initializations
    num_restarts = 10
    for i, guess in enumerate(initial_guesses):
        # Perturb and scale to meet sum constraint
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        h_perturbed = guess + perturbation
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        current_sum = np.sum(h_perturbed)
        scale_factor = target_sum / current_sum
        h_perturbed *= scale_factor

        # Optimize
        start_time = time.time()
        result = minimize(objective, h_perturbed, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5)
        elapsed += time.time() - start_time

        if elapsed >= budget_s:

        if result.success:
            current_c5 = result.fun
