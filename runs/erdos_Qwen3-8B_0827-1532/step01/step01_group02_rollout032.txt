    # Create initial guess with two intervals of 0.5 each
    h_initial = np.zeros(n_points)
    # First interval [0, 0.5] corresponds to indices 0 to 24 (since 0.5 / dx = 25)
    h_initial[0:25] = 1.0
    # Second interval [1.5, 2.0] corresponds to indices 75 to 99 (since 1.5/dx = 75)
    h_initial[75:100] = 1.0
    # Check sum
    assert np.isclose(np.sum(h_initial), target_sum), "Initial guess sum is incorrect"
    
    def objective(h):
        g = 1 - h
        corr = fftconvolve(h, g[::-1], mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    
    best_h = h_initial
    best_c5 = objective(best_h)
    elapsed = 0.0
    
    # Run multiple restarts with perturbations
    num_restarts = 10  # Increase the number of restarts
    for _ in range(num_restarts):
        # Perturb initial solution
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        h_perturbed = h_initial + perturbation
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        # Scale to meet sum constraint
        current_sum = np.sum(h_perturbed)
        scale_factor = target_sum / current_sum
        h_perturbed *= scale_factor
        
        # Optimize
        start_time = time.time()
        result = minimize(objective, h_perturbed, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        elapsed += time.time() - start_time
        
        if elapsed >= budget_s:
            break
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            # Fallback to perturbed solution
            current_c5 = objective(h_perturbed)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_perturbed
    
    return (best_h, best_c5, n_points)
