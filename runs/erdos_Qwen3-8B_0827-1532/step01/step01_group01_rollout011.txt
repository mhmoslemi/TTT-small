import numpy as np
from scipy.optimize import differential_evolution
from scipy.optimize import Bounds
import time

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h_values, 1 - h_values, mode='full')
    max_overlap = np.max(conv) * dx
    return max_overlap

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    # Start with initial_h_values (n=70)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Define the objective function with sum constraint handling
    def objective(h):
        # Adjust the sum to meet the constraint
        current_sum = np.sum(h)
        target_sum = n_points / 2
        delta = target_sum - current_sum
        h[-1] += delta
        # Clip to [0, 1]
        h = np.clip(h, 0, 1)
        # Compute the overlap
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv) * dx
    
    # Bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Set up differential evolution with optimized parameters
    start_time = time.time()
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        maxiter=100,
        seed=seed
    )
    
    # Ensure we respect the time budget
    elapsed = time.time() - start_time
    if elapsed < budget_s:
        # Perform a final refinement with tighter tolerance
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=15,
            tol=1e-8,
            mutation=(0.5, 1),
            recombination=0.8,
            maxiter=50,
            seed=seed
        )
    
    best_h = result.x
    c5_bound = evaluate_erdos_solution(best_h, n_points)
    return best_h, c5_bound, n_points
