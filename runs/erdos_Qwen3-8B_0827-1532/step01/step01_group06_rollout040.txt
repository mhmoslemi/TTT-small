from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h_initial = initial_h_values.copy()

    def objective(h):
        correlate_result = np.correlate(h, 1 - h, mode='full')
        max_value = np.max(correlate_result) * dx
        penalty = 1e4 * np.abs(np.sum(h) - n_points / 2)
        return max_value + penalty

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Initial optimization with differential evolution
    result_de = differential_evolution(

        popsize=15, mutation=(0.5, 1), recombination=0.8, 
        tol=1e-6, maxiter=100, seed=seed
    )
    best_h = result_de.x
    best_c5 = objective(best_h)

    # Refinement with SLSQP
    def constraint(h):
        return np.sum(h) - n_points / 2

    bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]
    result_slsqp = minimize(
        objective, best_h, method='SLSQP', bounds=bounds_slsqp, 
        constraints={'type': 'eq', 'fun': constraint},

    )
