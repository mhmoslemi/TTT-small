import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    n_points = 64
    dx = 2.0 / n_points
    np.random.seed(seed)
    
    def objective(h):
        correlate_result = np.correlate(h, 1 - h, mode='full')
        max_value = np.max(correlate_result) * dx
        return max_value
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint}
    
    # Generate multiple initial guesses
    initial_guesses = []
    initial_guesses.append(initial_h_values.copy())
    
    # Random initial guess
    random_h = np.random.uniform(0, 1, n_points)
    random_h = random_h * (n_points / 2) / np.sum(random_h)
    initial_guesses.append(random_h)
    
    # Alternating pattern
    pattern_h = np.zeros(n_points)
    pattern_h[::2] = 1.0
    pattern_h = pattern_h * (n_points / 2) / np.sum(pattern_h)
    initial_guesses.append(pattern_h)
    
    best_h = None
    best_c5 = float('inf')
    
    for guess in initial_guesses:
        result = minimize(objective, guess, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 200})
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_h = result.x
                best_c5 = current_c5
    
    return (best_h, best_c5, n_points)
