from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
import time

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h_values, 1 - h_values, mode='full')
    max_overlap = np.max(conv) * dx
    return max_overlap

def create_initial_guesses(n_points):
    guesses = []
    # Add initial_h_values
    guesses.append(initial_h_values)
    # Single interval guess
    h_single = np.zeros(n_points)
    num_points = n_points // 2
    h_single[:num_points] = 1.0
    guesses.append(h_single)
    # Two intervals guess
    h_two = np.zeros(n_points)
    num_points_per_interval = n_points // 4
    h_two[:num_points_per_interval] = 1.0
    h_two[-num_points_per_interval:] = 1.0
    guesses.append(h_two)
    # Random guess
    h_random = np.random.rand(n_points)
    scale = (n_points / 2) / np.sum(h_random)
    h_random *= scale
    guesses.append(h_random)
    return guesses

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    # Determine n_points based on initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    # Constraint: sum(h) = n_points / 2
    constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
    bounds = Bounds(0, 1)
    
    # Objective function
    def objective(h):
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv) * dx
    
    # Generate initial guesses
    initial_guesses = create_initial_guesses(n_points)
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    for guess in initial_guesses:
        # Ensure guess is a numpy array
        h_initial = np.array(guess, dtype=np.float64)
        # Run optimization
        result = minimize(objective, h_initial, method='SLSQP', bounds=bounds, constraints=constraint, tol=1e-6)
        # Check if we have time left to refine
        elapsed = time.time() - start_time
        if elapsed < budget_s:
            # Refine with smaller tolerance
            result = minimize(objective, result.x, method='SLSQP', bounds=bounds, constraints=constraint, tol=1e-8)
        current_c5 = evaluate_erdos_solution(result.x, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x
    
    # Final refinement if time allows
    elapsed = time.time() - start_time
    if elapsed < budget_s:
        # Run one more refinement with the best guess
        result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=constraint, tol=1e-8)
        best_h = result.x
        best_c5 = evaluate_erdos_solution(best_h, n_points)
