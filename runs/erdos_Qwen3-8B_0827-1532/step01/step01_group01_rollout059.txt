    dx = 2.0 / n_points
    # Generate a structured initial guess with alternating high/low values
    h_initial = np.zeros(n_points)
    for i in range(n_points):
        if i % 4 == 0 or i % 4 == 3:
            h_initial[i] = 1.0
        else:
            h_initial[i] = 0.0
    # Adjust to meet the sum constraint
    total = np.sum(h_initial)
    h_initial = h_initial * (n_points / 2) / total
    
    # Constraint: sum(h) = n_points / 2
    constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
    bounds = Bounds(0, 1)
    
    # Objective function
    def objective(h):
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv) * dx
    
    # Initial guess
    h_initial = h_initial
    
    # Optimization
    start_time = time.time()
    result = minimize(objective, h_initial, method='L-BFGS-B', bounds=bounds, constraints=constraint, tol=1e-6)
    
    # Check if we have time left to refine
    elapsed = time.time() - start_time
    if elapsed < budget_s:
        # Try a few more iterations with a smaller tolerance
        result = minimize(objective, result.x, method='L-BFGS-B', bounds=bounds, constraints=constraint, tol=1e-8)
    
    best_h = result.x
    c5_bound = evaluate_erdos_solution(best_h, n_points)
    return best_h, c5_bound, n_points
