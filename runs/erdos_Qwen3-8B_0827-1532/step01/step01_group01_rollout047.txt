import numpy as np
from scipy.optimize import minimize, Bounds, LinearConstraint
from scipy.optimize import L_BFGS_B, COBYLA
import time

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h_values, 1 - h_values, mode='full')
    max_overlap = np.max(conv) * dx
    return max_overlap

def generate_initial_guesses(n_points, initial_h_values):
    guesses = [initial_h_values.copy()]
    # Perturb initial guess
    perturbed = initial_h_values + np.random.normal(0, 0.05, n_points)
    perturbed = np.clip(perturbed, 0, 1)
    perturbed_sum = np.sum(perturbed)
    perturbed = perturbed * (n_points / 2) / perturbed_sum
    guesses.append(perturbed)
    # Random guess
    random_guess = np.random.rand(n_points)
    random_guess_sum = np.sum(random_guess)
    random_guess = random_guess * (n_points / 2) / random_guess_sum
    guesses.append(random_guess)
    return guesses

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    initial_guesses = generate_initial_guesses(n_points, initial_h_values)
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    for guess in initial_guesses:
        try:
            # Define objective function
            def objective(h):
                conv = np.correlate(h, 1 - h, mode='full')
                return np.max(conv) * dx
            
            # Constraints
            constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
            bounds = Bounds(0, 1)
            
            # Optimization
            result = minimize(objective, guess, method='L-BFGS-B', bounds=bounds, constraints=constraint, tol=1e-6)
            current_h = result.x
            current_c5 = evaluate_erdos_solution(current_h, n_points)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
        except:
            # Fallback to initial guess
            pass
        
        # Check remaining time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
