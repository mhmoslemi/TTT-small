import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def objective(h):
        correlate_result = np.correlate(h, 1 - h, mode='full')
        max_value = np.max(correlate_result) * dx
        return max_value
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    best_h = initial_h_values.copy()
    best_c5 = objective(best_h)
    
    # Generate multiple starting points
    num_starts = 10
    for _ in range(num_starts):
        h_initial = np.random.uniform(0.0, 1.0, size=n_points)
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        result = minimize(objective, h_initial, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint}, options={'maxiter': 100})
        if result.success:
            current_h = result.x
            current_c5 = result.fun
        else:
            current_h = h_initial
            current_c5 = objective(current_h)
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return (best_h, best_c5, n_points)
