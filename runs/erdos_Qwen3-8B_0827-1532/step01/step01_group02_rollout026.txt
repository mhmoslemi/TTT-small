import numpy as np
from scipy.optimize import minimize, differential_evolution
from scipy.signal import correlate
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Initial guess from provided initial_h_values (n_points=70)
    h_initial = initial_h_values.copy()
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        # Compute cross-correlation
        corr = correlate(h, 1 - h, mode='full')
        # Find maximum value
        max_corr = np.max(corr)
        return max_corr * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    
    best_h = h_initial
    best_c5 = objective(best_h)
    elapsed = 0.0
    
    # Run multiple restarts with perturbations and differential evolution
    num_restarts = 10
    for _ in range(num_restarts):
        # Generate random initial guess
        h_perturbed = np.random.uniform(0.0, 1.0, size=n_points)
        current_sum = np.sum(h_perturbed)
        scale_factor = target_sum / current_sum
        h_perturbed *= scale_factor
        
        # Optimize using differential evolution
        start_time = time.time()
        result = differential_evolution(
            objective,
            bounds,
            constraints=cons,
            strategy='best1bin',
            popsize=15,
            maxiter=50,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.9,
            seed=seed + _
        )
        elapsed += time.time() - start_time
        
        if elapsed >= budget_s:
            break
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            # Fallback to perturbed solution
            current_c5 = objective(h_perturbed)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_perturbed
    
    return (best_h, best_c5, n_points)
