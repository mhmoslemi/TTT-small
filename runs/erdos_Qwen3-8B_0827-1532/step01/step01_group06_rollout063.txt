import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Increased resolution for better approximation
    dx = 2.0 / n_points
    best_h = None
    best_c5 = float('inf')
    
    # Generate multiple initial guesses
    initial_guesses = []
    for _ in range(5):  # Try 5 different starting points
        h = np.random.uniform(0.0, 1.0, n_points)
        h_sum = np.sum(h)
        scale = (n_points / 2) / h_sum
        h *= scale
        initial_guesses.append(h)
    
    # Define objective function
    def objective(h):
        correlate_result = np.correlate(h, 1 - h, mode='full')
        max_value = np.max(correlate_result) * dx
        return max_value
    
    # Define constraint
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    
    # Run optimization for each initial guess
    for h_initial in initial_guesses:
        result = minimize(
            objective,
            h_initial,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 200, 'ftol': 1e-8}
        )
        if result.success:
            current_c5 = result.fun
            current_h = result.x
        else:
            current_c5 = objective(h_initial)
            current_h = h_initial
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h
    
    return (best_h, best_c5, n_points)
