        penalty = 1e6 * (np.sum(h) - target_sum)**2
        return c5 + penalty

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Scale initial guess to meet sum constraint
    initial_scaled = h_initial.copy()
    current_sum = np.sum(initial_scaled)
    scale_factor = target_sum / current_sum
    initial_scaled *= scale_factor

    # Run differential evolution with optimized parameters
    start_time = time.time()
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=20,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed
    )
    elapsed = time.time() - start_time

    # Check if time budget is exceeded
    if elapsed >= budget_s:
        # Fallback to scaled initial guess
        corr = correlate(initial_scaled, 1 - initial_scaled, mode='full')
        best_c5 = np.max(corr) * dx
        return (initial_scaled, best_c5, n_points)

    # Extract best solution
    best_h = result.x
    # Compute actual C5 by recomputing cross-correlation
    corr = correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr) * dx

    return (best_h, best_c5, n_points)
