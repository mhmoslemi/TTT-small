from scipy.optimize import minimize

    bounds = Bounds(0.0, 1.0)

    def optimize(starting_h):
        result = minimize(objective, starting_h, method='L-BFGS-B', bounds=bounds, constraints={'type': 'eq', 'fun': constraint}, options={'maxiter': 100})
        if result.success:

        else:
            return starting_h, objective(starting_h)
