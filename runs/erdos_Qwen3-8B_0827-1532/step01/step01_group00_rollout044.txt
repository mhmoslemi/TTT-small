import numpy as np
from scipy.optimize import minimize, Bounds, LinearConstraint
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Start with the initial_h_values provided
    n_points = 45
    dx = 2.0 / n_points
    initial_h = initial_h_values.copy()
    
    # Define the objective function to minimize the maximum overlap
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define constraints: sum(h) == n_points / 2 and 0 <= h[i] <= 1
    constraints = [LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])]
    bounds = Bounds(np.zeros(n_points), np.ones(n_points))
    
    start_time = time.time()
    result = minimize(
        objective,
        initial_h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-9}
    )
    
    elapsed_time = time.time() - start_time
    if elapsed_time >= budget_s:
        return result.x, result.fun, n_points
    
    # Perturb and optimize multiple times to escape local minima
    for _ in range(5):
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        new_h = result.x + perturbation
        new_h = np.clip(new_h, 0, 1)
        result = minimize(
            objective,
            new_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-9}
        )
        elapsed_time = time.time() - start_time
        if elapsed_time >= budget_s:
            break
    
    return result.x, result.fun, n_points
