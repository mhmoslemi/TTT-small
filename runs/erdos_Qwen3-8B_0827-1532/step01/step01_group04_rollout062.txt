        bounds,
        strategy='best1bin',
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.8,
        tol=1e-8,
        maxiter=100,
        disp=False
    )
    
    # Check if we ran out of time
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        # Use initial guess if time is up
        h_values = np.zeros(n_points)
        h_values[:-1] = initial_guess
        h_values[-1] = (n_points / 2) - np.sum(initial_guess)
        h_values = np.clip(h_values, 0, 1)
        c5_bound = evaluate_erdos_solution(h_values)
        return h_values, c5_bound, n_points
    
    h_values = np.zeros(n_points)
    h_values[:-1] = result.x
    h_values[-1] = (n_points / 2) - np.sum(result.x)
    h_values = np.clip(h_values, 0, 1)
    c5_bound = result.fun
    return h_values, c5_bound, n_points
