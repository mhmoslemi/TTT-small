            guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8}
        )
        # Check if the optimization was successful
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        # Check time
        elapsed_time = time.time() - start_time
        if elapsed_time >= budget_s:
            break
    
    # If no result found, return the best so far
    return best_h, best_c5, n_points
