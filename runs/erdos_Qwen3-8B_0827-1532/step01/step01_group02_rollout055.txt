from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):

    # Initial guess from provided initial_h_values (n_points=70)

    n_points = 70

    target_sum = n_points / 2

    def objective(h):
        # Compute cross-correlation
        corr = np.correlate(h, 1 - h, mode='full')
        # Find maximum value
        max_corr = np.max(corr)
        # Compute penalty for constraint violation
        penalty = 1e6 * abs(np.sum(h) - target_sum)
        return max_corr * dx + penalty

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Run differential evolution
    start_time = time.time()
    result = differential_evolution(

    )
