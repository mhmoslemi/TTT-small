import numpy as np
from scipy.optimize import minimize, differential_evolution
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    def constraint(h_flat):
        return np.sum(h_flat) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint})

    # Initial guesses
    initial_guesses = [initial_h.copy()]
    for _ in range(5):
        perturbation = np.random.normal(0, 0.05, n_points)
        perturbed = initial_h + perturbation
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        initial_guesses.append(perturbed)

    for _ in range(3):
        h_rand = np.random.rand(n_points)
        h_rand = h_rand * target_sum / np.sum(h_rand)
        initial_guesses.append(h_rand)

    best_c5 = float('inf')
    best_h = initial_h.copy()

    for guess in initial_guesses:
        result = minimize(objective, guess, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 500})
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x

    return (best_h, best_c5, n_points)
