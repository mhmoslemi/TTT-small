import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h_values, 1 - h_values, mode='full')
    max_overlap = np.max(conv) * dx
    return max_overlap

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Objective function with penalty for sum constraint
    def objective(h):
        conv = np.correlate(h, 1 - h, mode='full')
        max_overlap = np.max(conv) * dx
        penalty = 1e4 * abs(np.sum(h) - target_sum)
        return max_overlap + penalty
    
    # Bounds for each h[i]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Run differential evolution
    start_time = time.time()
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed,
        maxiter=100,
        disp=False
    )
    
    # Check if we have time left to refine
    elapsed = time.time() - start_time
    if elapsed < budget_s:
        # Refine with SLSQP
        from scipy.optimize import minimize
        from scipy.optimize import Bounds
        from scipy.optimize import LinearConstraint
        
        constraint = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
        bounds = Bounds(0, 1)
        
        def objective_slsqp(h):
            conv = np.correlate(h, 1 - h, mode='full')
            return np.max(conv) * dx
        
        result_slsqp = minimize(
            objective_slsqp,
            result.x,
            method='SLSQP',
            bounds=bounds,
            constraints=constraint,
            tol=1e-8
        )
        best_h = result_slsqp.x
    else:
        best_h = result.x
    
    c5_bound = evaluate_erdos_solution(best_h, n_points)
    return best_h, c5_bound, n_points
