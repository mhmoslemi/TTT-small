import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h_initial = initial_h_values.copy()
    
    def symmetric_h():
        symmetric = np.zeros(n_points)
        for i in range(n_points):
            symmetric[i] = h_initial[i]
            symmetric[n_points - 1 - i] = h_initial[i]
        return symmetric
    
    def objective(h):
        correlate_result = np.correlate(h, 1 - h, mode='full')
        max_value = np.max(correlate_result) * dx
        return max_value
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    bounds = Bounds(0.0, 1.0)
    
    def optimize(starting_h):
        result = minimize(
            objective,
            starting_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 200, 'ftol': 1e-8, 'gtol': 1e-8}
        )
        if result.success:
            return result.x, result.fun
        else:
            return starting_h, objective(starting_h)
    
    best_h, best_c5 = optimize(h_initial)
    
    # Try symmetric initial guess
    symmetric_guess = symmetric_h()
    h_sym, c5_sym = optimize(symmetric_guess)
    if c5_sym < best_c5:
        best_h, best_c5 = h_sym, c5_sym
    
    # Perturbation and optimization
    for _ in range(5):
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        perturbed_h = h_initial + perturbation
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        current_sum = np.sum(perturbed_h)

        scale = target_sum / current_sum
        perturbed_h *= scale

        if c5_perturbed < best_c5:
            best_h, best_c5 = h_perturbed, c5_perturbed
    
    return (best_h, best_c5, n_points)
