import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    # Use n_points=50 for faster optimization
    n_points = 50
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        # Compute cross-correlation
        corr = correlate(h, 1 - h, mode='full')
        # Find maximum value
        max_corr = np.max(corr)
        return max_corr * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    best_h = np.zeros(n_points)
    best_c5 = 1.0  # Initialize with a high value

    elapsed = 0.0

    # Run multiple restarts with random initial guesses
    num_restarts = 10
    for _ in range(num_restarts):
        # Generate random initial guess
        h_initial = np.random.rand(n_points)
        h_initial = h_initial / np.sum(h_initial) * target_sum
        # Optimize
        start_time = time.time()
        result = minimize(objective, h_initial, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        elapsed += time.time() - start_time

        if elapsed >= budget_s:
            break

        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            # Fallback to random initial guess
            current_c5 = objective(h_initial)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_initial

    return (best_h, best_c5, n_points)
