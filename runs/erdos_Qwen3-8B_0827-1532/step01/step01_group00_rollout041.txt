    bounds = [(0.0, 1.0) for _ in range(n_points)]
    start_time = time.time()
    
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=10,
        maxiter=100,
        tol=1e-8,
        mutation=(0.5, 1),
        recombination=0.8,
        polish=True,
        init=initial_h,
        seed=seed
    )
    
    elapsed_time = time.time() - start_time
    if elapsed_time >= budget_s:
        return result.x, result.fun, n_points
    
    # Run a second optimization with a different initial guess
    initial_h2 = np.random.uniform(0, 1, n_points)
    initial_h2 = initial_h2 / np.sum(initial_h2) * (n_points / 2)
    result2 = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=10,
        maxiter=100,
        tol=1e-8,
        mutation=(0.5, 1),
        recombination=0.8,
        polish=True,
        init=initial_h2,
        seed=seed
    )
    
    elapsed_time2 = time.time() - start_time
    if elapsed_time2 >= budget_s:
        return result.x, result.fun, n_points
    
    # Return the better of the two results
    if result.fun < result2.fun:
        return result.x, result.fun, n_points
    else:
        return result2.x, result2.fun, n_points
