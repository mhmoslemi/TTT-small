        np.random.seed(seed + i)
        result = differential_evolution(
            objective_with_penalty, bounds, 
            strategy='best1bin', popsize=15, maxiter=100, 
            tol=1e-6, seed=seed + i
        )
        current_c5 = result.fun
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x
    
    return best_h, best_c5, n_points
