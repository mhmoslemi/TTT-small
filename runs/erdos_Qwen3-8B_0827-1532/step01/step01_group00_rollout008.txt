    # This is a heuristic to escape local minima
    for _ in range(3):
        # Perturb the current solution slightly
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        new_h = result.x + perturbation
        new_h = np.clip(new_h, 0, 1)
        # Re-run optimization with the perturbed solution
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=15,
            tol=1e-8,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed,
            maxiter=50
        )
        elapsed_time = time.time() - start_time
        if elapsed_time >= budget_s:
            break
    
    # Return the best solution found
    return result.x, result.fun, n_points
