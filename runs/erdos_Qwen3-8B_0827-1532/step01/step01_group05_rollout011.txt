    random_initial = random_initial / np.sum(random_initial) * (n_points / 2)
    initial_guesses.append(random_initial)
    
    # Perturb the initial_h_values
    perturbed_initial = initial_h_values.copy()
    perturbed_initial += np.random.normal(0, 0.05, n_points)
    perturbed_initial = np.clip(perturbed_initial, 0, 1)
    initial_guesses.append(perturbed_initial)
    
    # Now, run the optimization for each initial guess
    best_c5 = float('inf')
    best_h = None
    
    for guess in initial_guesses:
        # Ensure the guess is within [0,1] and sum is correct
        guess = np.clip(guess, 0, 1)
        guess = guess * (n_points / 2) / np.sum(guess) if np.sum(guess) != 0 else guess
        
        # Define the objective function
        def objective(h):
            corr = correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Define the constraint
        constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
        
        # Bounds
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Run the optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=constraint,
            options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
        )
        
        # Check if the optimization was successful
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return (best_h, best_c5, n_points)
