    bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]
    result_slsqp = minimize(
        objective, best_h_de_normalized,
        method='SLSQP', bounds=bounds_slsqp,
        constraints={'type': 'eq', 'fun': constraint},
        options={'maxiter': 100}
    )
    best_h_slsqp = result_slsqp.x
    best_c5_slsqp = result_slsqp.fun

    best_h, best_c5 = (best_h_slsqp, best_c5_slsqp) if best_c5_slsqp < best_c5_de else (best_h_de_normalized, best_c5_de)
    return (best_h, best_c5, n_points)
