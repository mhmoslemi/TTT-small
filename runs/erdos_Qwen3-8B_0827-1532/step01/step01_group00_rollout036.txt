        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraints: sum(h) == n_points / 2
    constraints = [LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])]
    bounds = Bounds(np.zeros(n_points), np.ones(n_points))
    
    best_h = initial_h
    best_c5 = objective(initial_h)
    start_time = time.time()
    
    # Try different optimization methods
    for guess in initial_guesses:
        # Use L-BFGS-B for bounded optimization
        result = minimize(
            objective,
            guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8}
        )
        elapsed_time = time.time() - start_time
        if elapsed_time >= budget_s:
            break
        if result.success and result.fun < best_c5:
            best_h = result.x
            best_c5 = result.fun
    
    return best_h, best_c5, n_points
