import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
import time

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h_values, 1 - h_values, mode='full')
    max_overlap = np.max(conv) * dx
    return max_overlap

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    # Start with initial_h_values (n=70)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    # Constraint: sum(h) = n_points / 2
    constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
    bounds = Bounds(0, 1)
    
    # Objective function
    def objective(h):
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv) * dx
    
    # Generate multiple initial guesses
    initial_guesses = [initial_h.copy()]
    # Add 3 random binary initial guesses
    for _ in range(3):
        h = np.zeros(n_points)
        ones = n_points // 2
        h[:ones] = 1.0
        np.random.shuffle(h)
        initial_guesses.append(h.copy())
    
    best_h = None
