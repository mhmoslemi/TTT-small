import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h_initial = initial_h_values.copy()

    def objective(h):
        correlate_result = np.correlate(h, 1 - h, mode='full')
        max_value = np.max(correlate_result) * dx
        return max_value

    def constraint(h):
        return np.sum(h) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def optimize(starting_h):
        result = minimize(objective, starting_h, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint}, options={'maxiter': 500})
        if result.success:
            return result.x, result.fun
        else:
            return starting_h, objective(starting_h)

    best_h, best_c5 = optimize(h_initial)

    # Generate multiple initial guesses
    initial_guesses = [h_initial.copy()]
    for _ in range(5):
        random_guess = np.random.uniform(0, 1, size=n_points)
        random_guess = random_guess * (n_points / 2) / np.sum(random_guess)
        initial_guesses.append(random_guess)

    for guess in initial_guesses:
        h_perturbed, c5_perturbed = optimize(guess)
        if c5_perturbed < best_c5:
            best_h, best_c5 = h_perturbed, c5_perturbed

    return (best_h, best_c5, n_points)
