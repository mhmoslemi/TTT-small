from scipy.optimize import minimize
from scipy.signal import correlate

    # Use a higher number of points for better precision
    n_points = 100
    dx = 2.0 / n_points
    initial_h = initial_h_values  # Assume this is a global variable

        max_corr = np.max(corr)

    constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
    
    # Define bounds for each h[i] to be in [0, 1]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
