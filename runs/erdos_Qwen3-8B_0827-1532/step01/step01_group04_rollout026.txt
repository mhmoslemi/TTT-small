            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-8, 'maxiter': 200}
        )
        
        # Check if optimization was successful
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    # If no optimization was done, return initial guess
    if best_h is None:
        best_h = initial_guess_upsampled
    
    # Evaluate the best solution
    c5_bound = evaluate_erdos_solution(best_h)
    
    return best_h, c5_bound, n_points
