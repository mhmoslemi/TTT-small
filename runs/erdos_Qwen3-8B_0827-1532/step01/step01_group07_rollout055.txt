            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed + i
        )
        current_h = result.x
        current_c5 = compute_c5(current_h)
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5

    return (best_h, best_c5, n_points)
