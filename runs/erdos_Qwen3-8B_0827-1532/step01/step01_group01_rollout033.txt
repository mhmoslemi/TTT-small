    for _ in range(num_initial_guesses):
        perturb = np.random.normal(0, 0.05, n_points)
        perturbed = h_initial + perturb
        perturbed = np.clip(perturbed, 0, 1)
        sum_perturbed = np.sum(perturbed)
        scaling_perturbed = (n_points / 2) / sum_perturbed
        perturbed = perturbed * scaling_perturbed
        initial_guesses.append(perturbed)
    
    # Define objective function
    def objective(h):
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv) * dx
    
    # Define constraints and bounds
    constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
    bounds = Bounds(0, 1)
    
    # Optimization
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    
    for guess in initial_guesses:
        result = minimize(objective, guess, method='L-BFGS-B', bounds=bounds, constraints=constraint, tol=1e-6)
        if result.success:
            current_c5 = evaluate_erdos_solution(result.x, n_points)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        # Check time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    # If no result found, use the initial guess
    if best_h is None:
        result = minimize(objective, h_initial, method='L-BFGS-B', bounds=bounds, constraints=constraint, tol=1e-6)
        best_h = result.x
        best_c5 = evaluate_erdos_solution(best_h, n_points)
    
    return best_h, best_c5, n_points
