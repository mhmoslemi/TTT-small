import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    initial_guesses = [initial_h_values.copy()]
    
    # Generate 5 random initial guesses
    for _ in range(5):
        h = np.random.uniform(0.0, 1.0, n_points)
        h_sum = np.sum(h)
        scale = (n_points / 2) / h_sum
        h = h * scale
        initial_guesses.append(h)
    
    # Shuffle to avoid bias
    np.random.shuffle(initial_guesses)
    
    def objective(h):
        correlate_result = np.correlate(h, 1 - h, mode='full')
        max_value = np.max(correlate_result) * dx
        return max_value
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    bounds = Bounds(0.0, 1.0)
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    for h_initial in initial_guesses:
        result = minimize(
            objective,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8}
        )
        if result.success:
            current_c5 = result.fun
            current_h = result.x
        else:
            current_c5 = objective(h_initial)
            current_h = h_initial
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h
        
        # Early stopping if time is exhausted
        if time.time() - start_time > budget_s - 1:
            break
    
    return (best_h, best_c5, n_points)
