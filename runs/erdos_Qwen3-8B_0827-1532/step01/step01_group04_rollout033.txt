import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Increased number of points for better precision
    dx = 2.0 / n_points
    
    # Generate a symmetric initial guess with a triangular shape
    h_initial = np.zeros(n_points)
    for i in range(n_points):
        x = i * dx
        if x < 1.0:
            h_initial[i] = x / 1.0
        else:
            h_initial[i] = (2.0 - x) / 1.0
    # Adjust to satisfy the integral constraint
    current_sum = np.sum(h_initial)
    target_sum = n_points / 2
    scale_factor = target_sum / current_sum
    h_initial *= scale_factor
    
    # Objective function to minimize C5
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraints: sum(h) == n_points / 2
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Start optimization
    start_time = time.time()
    result = minimize(
        objective,
        h_initial,
        method='COBYLA',  # More robust for constrained problems
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 150, 'disp': False}  # Increased maxiter for better convergence
    )
    
    # Check if we ran out of time
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        # Fallback: return the initial guess
        return h_initial, evaluate_erdos_solution(h_initial), n_points
    
    h_values = result.x
    c5_bound = result.fun
    return h_values, c5_bound, n_points
