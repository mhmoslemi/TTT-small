        corr = correlate(scaled_h, 1 - scaled_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()
    result = differential_evolution(
        objective, bounds, 
        strategy='best1bin', 
        popsize=15, 
        mutation=(0.5, 1.0), 
        recombination=0.8, 
        tol=1e-6, 
        maxiter=100, 
        seed=seed
    )
    elapsed = time.time() - start_time

    # Early exit if time is exceeded
    if elapsed >= budget_s:
        return (np.zeros(n_points), 1.0, n_points)

    best_h = result.x
    best_c5 = result.fun
    return (best_h, best_c5, n_points)
