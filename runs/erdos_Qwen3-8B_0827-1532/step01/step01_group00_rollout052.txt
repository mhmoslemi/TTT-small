import numpy as np
from scipy.optimize import minimize, Bounds, LinearConstraint
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 45  # Use the same n_points as initial_h_values

    # Initial guess from provided initial_h_values

        initial_guesses.append(h)
