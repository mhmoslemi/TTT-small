from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Increase resolution to 100 points for better precision
    n_points = 100
    np.random.seed(seed)

    h_initial = np.random.rand(n_points)

    def objective(h):
        one_minus_h = 1.0 - h

        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint: sum(h) == 50
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - 50})

    )
