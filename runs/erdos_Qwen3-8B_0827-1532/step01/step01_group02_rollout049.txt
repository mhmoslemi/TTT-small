        if np.random.rand() < 0.5:
            # Use initial guess with perturbation
            h_perturbed = h_initial + np.random.uniform(-0.1, 0.1, size=n_points)
            h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
            current_sum = np.sum(h_perturbed)
            scale_factor = target_sum / current_sum
            h_perturbed *= scale_factor
        else:
            # Generate random initial guess
            h_perturbed = np.random.uniform(0, 1, size=n_points)
            h_perturbed = h_perturbed * (target_sum / np.sum(h_perturbed))

        # Optimize
        start_time = time.time()
        result = minimize(objective, h_perturbed, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        elapsed += time.time() - start_time

        if elapsed >= budget_s:
            break

        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            # Fallback to perturbed solution
            current_c5 = objective(h_perturbed)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_perturbed

    return (best_h, best_c5, n_points)
