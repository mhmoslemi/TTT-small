import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    initial_guesses = [initial_h_values.copy()]
    
    # Generate additional initial guesses
    for _ in range(5):
        random_h = np.random.rand(n_points)
        random_h = random_h / np.sum(random_h) * (n_points / 2)
        initial_guesses.append(random_h)
    
    bounds = Bounds(0.0, 1.0)
    
    def objective(h):
        correlate_result = np.correlate(h, 1 - h, mode='full')
        max_value = np.max(correlate_result) * dx
        return max_value
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    best_h = initial_h_values.copy()
    best_c5 = float('inf')
    
    for guess in initial_guesses:
        # Use COBYLA for constraint handling
        result = minimize(objective, guess, method='COBYLA', bounds=bounds, constraints={'type': 'eq', 'fun': constraint}, options={'maxiter': 200})
        if result.success:
            current_h = result.x
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
        else:
            # Fallback to the guess
            current_c5 = objective(guess)
            if current_c5 < best_c5:
                best_h = guess
                best_c5 = current_c5
    
    return (best_h, best_c5, n_points)
