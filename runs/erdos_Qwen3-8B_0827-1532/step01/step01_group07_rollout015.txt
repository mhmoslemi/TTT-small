import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    # Define objective function
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    # Define constraints
    cons = ({'type': 'eq', 'fun': lambda h_flat: np.sum(h_flat) - n_points / 2})
    bounds = [(0, 1) for _ in range(n_points)]
    # Initial guess
    best_c5 = float('inf')
    best_h = initial_h.copy()
    # Try initial guess
    result = minimize(objective, initial_h, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 500})
    if result.success:
        best_c5 = result.fun
        best_h = result.x
    # Generate more random initial guesses
    for _ in range(10):  # Increase from 3 to 10
        # Generate random initial guess
        h_rand = np.random.rand(n_points)
        h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
        # Run optimization
        result = minimize(objective, h_rand, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 500})
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    return (best_h, best_c5, n_points)
