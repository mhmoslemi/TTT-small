import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
import time

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h_values, 1 - h_values, mode='full')
    max_overlap = np.max(conv) * dx
    return max_overlap

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
    bounds = Bounds(0, 1)
    
    def objective(h):
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv) * dx
    
    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h, n_points)
    start_time = time.time()
    
    # Generate multiple initial guesses
    num_guesses = 5
    for _ in range(num_guesses):
        perturb = np.random.uniform(-0.1, 0.1, size=n_points)
        h_initial = np.clip(initial_h + perturb, 0, 1)
        result = minimize(objective, h_initial, method='trust-constr', bounds=bounds, constraints=constraint, tol=1e-6)
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.success:
            current_h = result.x
            current_c5 = evaluate_erdos_solution(current_h, n_points)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
        else:
            current_h = h_initial
            current_c5 = evaluate_erdos_solution(current_h, n_points)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
    
    # Final refinement
    if time.time() - start_time < budget_s:
        result = minimize(objective, best_h, method='trust-constr', bounds=bounds, constraints=constraint, tol=1e-8)
        if result.success:
            current_h = result.x
            current_c5 = evaluate_erdos_solution(current_h, n_points)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
    
    return best_h, best_c5, n_points
