import numpy as np
from scipy.optimize import minimize, Bounds, LinearConstraint
from scipy.optimize import COBYLA, L_BFGS_B
import time

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h_values, 1 - h_values, mode='full')
    max_overlap = np.max(conv) * dx
    return max_overlap

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    # Start with initial_h_values (n=70)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    # Constraint: sum(h) = n_points / 2
    constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
    bounds = Bounds(0, 1)
    
    # Objective function
    def objective(h):
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv) * dx
    
    # Generate multiple initial guesses
    num_initial_guesses = 5
    best_c5 = float('inf')
    best_h = initial_h.copy()
    elapsed_time = 0
    
    for i in range(num_initial_guesses):
        # Perturb the initial guess
        perturbation = np.random.normal(0, 0.05, n_points)
        h_initial = initial_h + perturbation
        h_initial = np.clip(h_initial, 0, 1)  # Ensure values are within [0,1]
        
        # Run optimization
        start_time = time.time()
        result = minimize(objective, h_initial, method='COBYLA', bounds=bounds, constraints=constraint, tol=1e-6)
        elapsed = time.time() - start_time
        elapsed_time += elapsed
        
        # Check if we have time left to refine
        if elapsed < budget_s and elapsed_time < budget_s:
            # Try a few more iterations with a smaller tolerance
            result = minimize(objective, result.x, method='COBYLA', bounds=bounds, constraints=constraint, tol=1e-8)
        
        # Evaluate the result
        current_c5 = evaluate_erdos_solution(result.x, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x
    
    # Fallback to initial guess if all optimizations fail
    if best_c5 == float('inf'):
        best_c5 = evaluate_erdos_solution(initial_h, n_points)
        best_h = initial_h.copy()
    
    return best_h, best_c5, n_points
