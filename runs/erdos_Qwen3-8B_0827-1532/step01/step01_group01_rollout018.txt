import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
import time

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h_values, 1 - h_values, mode='full')
    max_overlap = np.max(conv) * dx
    return max_overlap

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    # Use n_points = 100 for better approximation
    n_points = 100
    dx = 2.0 / n_points
    
    # Generate a random initial guess with sum(h) = n_points / 2
    h_initial = np.random.rand(n_points)
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    
    # Constraint: sum(h) = n_points / 2
    constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
    bounds = Bounds(0, 1)
    
    # Objective function
    def objective(h):
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv) * dx
    
    # Optimization
    start_time = time.time()
    result = minimize(objective, h_initial, method='L-BFGS-B', bounds=bounds, constraints=constraint, tol=1e-6)
    
    # Check if we have time left to refine
    elapsed = time.time() - start_time
    if elapsed < budget_s:
        # Try a few more iterations with a smaller tolerance
        result = minimize(objective, result.x, method='L-BFGS-B', bounds=bounds, constraints=constraint, tol=1e-8)
    
    best_h = result.x
    c5_bound = evaluate_erdos_solution(best_h, n_points)
    return best_h, c5_bound, n_points
