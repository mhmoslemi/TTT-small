import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def objective(h):
        correlate_result = np.correlate(h, 1 - h, mode='full')
        max_value = np.max(correlate_result) * dx
        return max_value
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Generate multiple initial guesses
    initial_guesses = [initial_h_values.copy()]
    # Add a random initial guess
    random_guess = np.random.uniform(0, 1, n_points)
    random_guess = random_guess * (n_points / 2) / np.sum(random_guess)
    initial_guesses.append(random_guess)
    # Add a uniform initial guess
    uniform_guess = np.full(n_points, 0.5)
    initial_guesses.append(uniform_guess)
    
    best_h, best_c5 = None, float('inf')
    
    # Optimize each initial guess
    for guess in initial_guesses:
        result = minimize(objective, guess, method='SLSQP', bounds=bounds, 
                          constraints={'type': 'eq', 'fun': constraint}, 
                          options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8})
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_h, best_c5 = result.x, current_c5
    
    # Perturb and optimize further
    for _ in range(15):
        if best_h is None:
            break
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        perturbed_h = best_h + perturbation
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        current_sum = np.sum(perturbed_h)
        target_sum = n_points / 2
        scale = target_sum / current_sum
        perturbed_h *= scale
        result = minimize(objective, perturbed_h, method='SLSQP', bounds=bounds, 
                          constraints={'type': 'eq', 'fun': constraint}, 
                          options={'maxiter': 150, 'ftol': 1e-8, 'eps': 1e-8})
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_h, best_c5 = result.x, current_c5
    
    return (best_h, best_c5, n_points)
