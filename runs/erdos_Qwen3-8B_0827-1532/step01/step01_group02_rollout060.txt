from scipy.optimize import minimize
from scipy.signal import correlate
import time

        # Find maximum value
        max_corr = np.max(corr)
        return max_corr * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    best_h = h_initial

        h_perturbed = h_initial + perturbation

        # Scale to meet sum constraint
        current_sum = np.sum(h_perturbed)
