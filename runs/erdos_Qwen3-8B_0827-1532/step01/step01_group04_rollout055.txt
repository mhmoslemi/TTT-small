import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = 100

        h_initial,

        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-8, 'maxiter': 500}
    )
    
    # Check if we ran out of time
    elapsed = time.time() - start_time
    if elapsed >= budget_s:

    c5_bound = result.fun
