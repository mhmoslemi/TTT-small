import numpy as np
from scipy.optimize import differential_evolution, minimize

def run(seed=42, budget_s=100, **kwargs):
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def objective(h_flat):
        h = h_flat.copy()
        penalty = 1e6 * (np.sum(h) - n_points / 2)**2
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx + penalty

    result = differential_evolution(

    )
    best_h = result.x
    best_c5 = result.fun

    # Refine with SLSQP
    def objective_slsqp(h_flat):
        h = h_flat.copy()
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    cons = ({'type': 'eq', 'fun': lambda h_flat: np.sum(h_flat) - n_points / 2})
    bounds_slsqp = [(0, 1) for _ in range(n_points)]
    result_slsqp = minimize(
        objective_slsqp, best_h, method='SLSQP', bounds=bounds_slsqp,
        constraints=cons, options={'maxiter': 100}
    )
    if result_slsqp.success:
        best_c5 = result_slsqp.fun
        best_h = result_slsqp.x

    return (best_h, best_c5, n_points)
