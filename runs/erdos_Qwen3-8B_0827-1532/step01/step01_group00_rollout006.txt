import numpy as np
from scipy.optimize import minimize, Bounds, LinearConstraint
from scipy.optimize import SLSQP
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    n_points = 45
    dx = 2.0 / n_points
    
    # Generate initial guesses
    initial_guesses = []
    # Add provided initial guess
    initial_guesses.append(initial_h_values.copy())
    # Generate 4 random binary initial guesses
    for _ in range(4):
        sum_h = n_points / 2
        sum_h_rounded = int(round(sum_h))
        indices = np.random.choice(n_points, size=sum_h_rounded, replace=False)
        h = np.zeros(n_points)
        h[indices] = 1
        initial_guesses.append(h)
    
    # Define the objective function to minimize the maximum overlap
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define constraints: sum(h) == n_points / 2 and 0 <= h[i] <= 1
    constraints = [LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])]
    bounds = Bounds(np.zeros(n_points), np.ones(n_points))
    
    best_h = None
    best_c5 = np.inf
    start_time = time.time()
    
    for h_initial in initial_guesses:
        # Run optimization
        result = minimize(
            objective,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
        )
        # Update best result
        if result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        # Check time
        elapsed_time = time.time() - start_time
        if elapsed_time >= budget_s:
            break
    
    return best_h, best_c5, n_points
