    for _ in range(num_initial_points - 1):
        # Generate random initial point
        random_h = np.random.uniform(0, 1, n_points)
        random_h = random_h * (n_points / 2) / np.sum(random_h)
        initial_points.append(random_h)
        # Generate perturbed initial point
        perturbed_h = initial_h + np.random.uniform(-0.1, 0.1, n_points)
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        perturbed_h = perturbed_h * (n_points / 2) / np.sum(perturbed_h)
        initial_points.append(perturbed_h)

    best_h = initial_h.copy()
    best_c5 = objective(best_h)

    def optimize(starting_h):
        result = minimize(objective, starting_h, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint}, options={'maxiter': 200})
        if result.success:
            return result.x, result.fun
        else:
            return starting_h, objective(starting_h)

    for starting_h in initial_points:
        current_time = time.time()
        if current_time - start_time > budget_s - 10:  # leave some buffer
            break
        h_opt, c5_opt = optimize(starting_h)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt

    return (best_h, best_c5, n_points)
