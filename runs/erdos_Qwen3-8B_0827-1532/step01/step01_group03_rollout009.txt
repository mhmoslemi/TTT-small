import numpy as np
from scipy.optimize import minimize

def objective(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=100, **kwargs):
    n_points = 80
    dx = 2.0 / n_points
    np.random.seed(seed)
    
    # Generate initial guesses
    initial_guesses = []
    # Add the initial_h_values as one guess
    initial_guesses.append(initial_h_values.copy())
    # Generate 5 random guesses
    for _ in range(5):
        h_rand = np.random.rand(n_points)
        h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
        initial_guesses.append(h_rand)
    
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    
    for guess in initial_guesses:
        # Use L-BFGS-B for bounded optimization
        result = minimize(objective, guess, args=(dx,), method='L-BFGS-B', bounds=[(0, 1)]*n_points, 
                          constraints=[{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}])
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            # Fallback to the guess if optimization fails
            current_c5 = objective(guess, dx)
