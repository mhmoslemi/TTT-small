    constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
    
    # Define bounds for each h[i] to be in [0, 1]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Run optimization from initial_h_values
    h0_initial = initial_h.copy()
    result_initial = minimize(
        objective,
        h0_initial,
        method='SLSQP',
        bounds=bounds,
        constraints=constraint,
        options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8}
    )
    
    # Run optimization from binary initial guess
    h0_binary = generate_binary_initial_guess(n_points)
    result_binary = minimize(
        objective,
        h0_binary,
        method='SLSQP',
        bounds=bounds,
        constraints=constraint,
        options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8}
    )
    
    # Compare the two results
    if result_initial.fun < result_binary.fun:
        best_h = result_initial.x
        c5_bound = result_initial.fun
    else:
        best_h = result_binary.x
        c5_bound = result_binary.fun
    
    return (best_h, c5_bound, n_points)
