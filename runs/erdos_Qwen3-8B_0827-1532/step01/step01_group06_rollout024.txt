import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    # Use the initial_h_values as the base configuration
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h_initial = initial_h_values.copy()

    # Objective function to minimize the maximum overlap
    def objective(h):
        correlate_result = np.correlate(h, 1 - h, mode='full')
        max_value = np.max(correlate_result) * dx
        return max_value

    # Constraint to maintain the integral of h over [0,2] equal to 1
    def constraint(h):
        return np.sum(h) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Optimization function using L-BFGS-B with increased iterations
    def optimize(starting_h):
        result = minimize(
            objective,
            starting_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 200, 'ftol': 1e-8, 'gtol': 1e-8}
        )
        if result.success:
            return result.x, result.fun
        else:
            return starting_h, objective(starting_h)

    # Initial optimization with the initial configuration
    best_h, best_c5 = optimize(h_initial)

    # Perform multiple restarts with perturbations to escape local optima
    for _ in range(15):  # Increased number of restarts

        perturbation = np.random.uniform(-0.15, 0.15, size=n_points)
        perturbed_h = h_initial + perturbation
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        # Scale to maintain the integral constraint
        current_sum = np.sum(perturbed_h)
        target_sum = n_points / 2
        scale = target_sum / current_sum
        perturbed_h *= scale
        # Optimize the perturbed configuration
