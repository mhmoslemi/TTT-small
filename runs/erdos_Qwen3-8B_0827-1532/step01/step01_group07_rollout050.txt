            h[i] = h[n_points - 1 - i]
        h[-1] = (n_points / 2) - np.sum(h[:n_vars])
        return h[:n_vars]
    initial_guesses.append(generate_symmetric_guess())

    # 3. Random guess
    def generate_random_guess():
        h_flat = np.random.rand(n_vars)
        last_var = (n_points / 2) - np.sum(h_flat)
        if last_var < 0:
            last_var = 0
        elif last_var > 1:
            last_var = 1
        return h_flat
    initial_guesses.append(generate_random_guess())

    best_c5 = float('inf')
    best_h = np.zeros(n_points)

    for guess in initial_guesses:
        try:
            result = differential_evolution(
                objective,
                bounds,
                strategy='best1bin',
                maxiter=50,
                popsize=15,
                tol=1e-6,
                mutation=(0.5, 1),
                recombination=0.8,
                seed=seed
            )
            current_h = np.zeros(n_points)
            current_h[:n_vars] = result.x
            current_h[-1] = (n_points / 2) - np.sum(result.x)
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
        except:
            pass

    return (best_h, best_c5, n_points)
