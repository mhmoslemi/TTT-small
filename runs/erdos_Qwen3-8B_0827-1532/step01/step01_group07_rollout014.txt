from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def objective(h_flat):

        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    bounds = [(0, 1) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = initial_h.copy()

    result = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, options={'maxiter': 150})

        best_c5 = result.fun
        best_h = result.x
