from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=100, **kwargs):

    # Initial guess from provided initial_h_values (n_points=70)
    initial_h = initial_h_values.copy()
    n_points = 100  # Increased resolution for better accuracy
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        # Compute cross-correlation
        corr = correlate(h, 1 - h, mode='full')
        # Find maximum value
        max_corr = np.max(corr)
        return max_corr * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    best_h = np.zeros(n_points)
    best_c5 = 1.0  # Initialize with worst possible value
    elapsed = 0.0

    # Generate multiple initial guesses
    initial_guesses = [initial_h.copy()]
    for _ in range(4):  # Generate 4 random initial guesses
        h_rand = np.random.uniform(0, 1, n_points)
        h_rand = h_rand * (target_sum / np.sum(h_rand))
        initial_guesses.append(h_rand)

    # Run multiple restarts with different methods
    num_restarts = 10
    for i, h_initial in enumerate(initial_guesses):
        # Perturb initial solution
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        h_perturbed = h_initial + perturbation
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        current_sum = np.sum(h_perturbed)
        scale_factor = target_sum / current_sum
        h_perturbed *= scale_factor

        # Optimize with COBYLA method
        start_time = time.time()
        result = minimize(objective, h_perturbed, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-6)
