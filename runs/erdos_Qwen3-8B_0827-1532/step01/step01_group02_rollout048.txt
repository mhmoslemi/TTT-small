    # Generate initial guesses

    dx = 2.0 / n_points
    target_sum = n_points / 2

        guesses = []
        # Initial guess from provided initial_h_values
        guesses.append(initial_h_values.copy())
        # Sinusoidal initial guess
        x = np.linspace(0, 2, n_points)
        h_sinusoidal = 0.5 + 0.5 * np.sin(np.pi * x)
        h_sinusoidal = np.clip(h_sinusoidal, 0.0, 1.0)
        guesses.append(h_sinusoidal)
        return guesses

    initial_guesses = generate_initial_guesses()

    def objective(h):
        # Compute cross-correlation
        corr = correlate(h, 1 - h, mode='full')
        # Find maximum value
        max_corr = np.max(corr)
        return max_corr * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    best_h = initial_guesses[0]
    best_c5 = objective(best_h)
    elapsed = 0.0

    # Run multiple restarts with different initial guesses
    num_restarts = len(initial_guesses)
    for i, guess in enumerate(initial_guesses):
        # Perturb the guess slightly
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        h_perturbed = guess + perturbation
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        # Scale to meet sum constraint
        current_sum = np.sum(h_perturbed)
        scale_factor = target_sum / current_sum
        h_perturbed *= scale_factor

        # Optimize
        start_time = time.time()
        result = minimize(objective, h_perturbed, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-5)
        elapsed += time.time() - start_time

        if elapsed >= budget_s:
            break

        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

            # Fallback to perturbed solution
            current_c5 = objective(h_perturbed)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_perturbed

    return (best_h, best_c5, n_points)
