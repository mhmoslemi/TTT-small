import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run(seed=42, budget_s=100, **kwargs):
    # Access initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    # Define dx
    dx = 2.0 / n_points
    # Define objective function
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    # Define constraints
    cons = ({'type': 'eq', 'fun': lambda h_flat: np.sum(h_flat) - n_points / 2})
    bounds = Bounds(0, 1)
    # Initial guess
    h_initial = initial_h.copy()
    # Track best solution
    best_c5 = float('inf')
    best_h = h_initial
    # Try initial guess with more iterations
    result = minimize(objective, h_initial, method='L-BFGS-B', bounds=bounds, constraints=cons, options={'maxiter': 200})
    if result.success:
        best_c5 = result.fun
        best_h = result.x
    # Generate more random initial guesses
    for _ in range(5):
        # Generate random initial guess
        h_rand = np.random.rand(n_points)
        h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
        # Run optimization
        result = minimize(objective, h_rand, method='L-BFGS-B', bounds=bounds, constraints=cons, options={'maxiter': 200})
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    return (best_h, best_c5, n_points)
