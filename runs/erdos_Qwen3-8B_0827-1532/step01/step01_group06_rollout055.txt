import numpy as np
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h_initial = initial_h_values.copy()
    
    def objective(h):

        max_value = np.max(correlate_result) * dx
        constraint_violation = np.abs(np.sum(h) - n_points / 2)
        return max_value + 1e6 * constraint_violation

    # Initial differential evolution with penalty
    result = differential_evolution(objective, bounds, strategy='best1bin', maxiter=50, popsize=15, tol=1e-6, mutation=(0.5, 1), recombination=0.8, seed=seed)
    best_h = result.x
    best_c5 = result.fun
    
    # Perturb and optimize further
    for _ in range(5):
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        perturbed_h = h_initial + perturbation
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        current_sum = np.sum(perturbed_h)
        target_sum = n_points / 2
        scale = target_sum / current_sum
        perturbed_h *= scale

        result_perturbed = differential_evolution(objective, bounds, strategy='best1bin', maxiter=30, popsize=15, tol=1e-6, mutation=(0.5, 1), recombination=0.8, seed=seed)

            best_c5 = result_perturbed.fun
    
    return (best_h, best_c5, n_points)
