from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Choose n_points
    n_points = 100
    dx = 2.0 / n_points
    
    # Generate initial guess
    np.random.seed(seed)
    h_initial = np.random.rand(n_points)
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    h_initial = np.clip(h_initial, 0.0, 1.0)

    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraints: sum(h) == n_points / 2
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Start optimization
    start_time = time.time()
    result = minimize(
        objective,
        h_initial,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-8, 'maxiter': 200}
    )
    
    # Check if we ran out of time
    elapsed = time.time() - start_time
