            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-9}
        )
        elapsed_opt = time.time() - start_opt
        elapsed_time += elapsed_opt
        
        # Update best result
        if result.fun < best_c5:
            best_c5 = result.fun
            best_result = result.x
        
        # Check time budget
        if elapsed_time >= budget_s:
            break
    
    # Also try the initial_h_values provided (assuming it's available)
    if hasattr(globals(), 'initial_h_values'):
        initial_guess = initial_h_values.copy()
        # Ensure the sum is correct (should already be)
        # Run optimization
        start_opt = time.time()
        result = minimize(
            objective,
            initial_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-9}
        )
        elapsed_opt = time.time() - start_opt
        elapsed_time += elapsed_opt
        
        # Update best result
        if result.fun < best_c5:
            best_c5 = result.fun
            best_result = result.x
    
    # Fallback to a random guess if no result found
    if best_result is None:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))  # Adjust to meet the sum constraint
        # Run optimization
        start_opt = time.time()
        result = minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-9}
        )
        elapsed_opt = time.time() - start_opt
        elapsed_time += elapsed_opt
        best_result = result.x
        best_c5 = result.fun
    
    return best_result, best_c5, n_points
