import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    cons = ({'type': 'eq', 'fun': lambda h_flat: np.sum(h_flat) - n_points / 2})
    bounds = [(0, 1) for _ in range(n_points)]

    initial_guesses = [initial_h.copy()]

    # Structured guess: first half 1, second half 0
    structured_guess = np.zeros(n_points)
    structured_guess[:n_points//2] = 1.0
    structured_guess = structured_guess * (n_points / 2) / np.sum(structured_guess)
    initial_guesses.append(structured_guess)

    # Structured guess: alternating 1 and 0
    structured_guess = np.zeros(n_points)
    for i in range(n_points):
        if i % 2 == 0:
            structured_guess[i] = 1.0
    structured_guess = structured_guess * (n_points / 2) / np.sum(structured_guess)
    initial_guesses.append(structured_guess)

    # Random guesses
    for _ in range(5):
        h_rand = np.random.rand(n_points)
        h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
        initial_guesses.append(h_rand)

    best_c5 = float('inf')
    best_h = initial_h.copy()

    for guess in initial_guesses:
        result = minimize(objective, guess, method='L-BFGS-B', bounds=bounds, constraints=cons, options={'maxiter': 200})
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            # Fallback: perturb and retry
            perturbed_guess = guess + np.random.normal(0, 0.01, size=n_points)
            perturbed_guess = np.clip(perturbed_guess, 0, 1)
            perturbed_guess = perturbed_guess * (n_points / 2) / np.sum(perturbed_guess)
            result = minimize(objective, perturbed_guess, method='L-BFGS-B', bounds=bounds, constraints=cons, options={'maxiter': 200})
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x

    return (best_h, best_c5, n_points)
