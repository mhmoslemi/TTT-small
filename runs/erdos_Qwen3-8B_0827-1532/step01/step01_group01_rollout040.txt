import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h_values, 1 - h_values, mode='full')
    max_overlap = np.max(conv) * dx
    return max_overlap

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    # Start with initial_h_values (n=70)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Objective function with penalty for sum constraint
    def objective(h):
        conv = np.correlate(h, 1 - h, mode='full')
        max_overlap = np.max(conv) * dx
        penalty = 1e6 * (np.sum(h) - target_sum)**2
        return max_overlap + penalty
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial guess: perturb initial_h to form initial population
    def generate_initial_population(popsize):
        pop = np.zeros((popsize, n_points))
        for i in range(popsize):
            perturb = np.random.normal(0, 0.05, n_points)
            pop[i] = np.clip(initial_h + perturb, 0, 1)
        return pop
    
    # Run differential evolution
    start_time = time.time()
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed,
        polish=True,
        disp=False
    )
    
    # Check if we have time left to refine
    elapsed = time.time() - start_time
    if elapsed < budget_s:
        # Try with more iterations
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=200,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed,
            polish=True,
            disp=False
        )
    
    best_h = result.x
    c5_bound = evaluate_erdos_solution(best_h, n_points)
    return best_h, c5_bound, n_points
