    for guess in initial_guesses:
        result = minimize(
            objective,
            guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_h = result.x
                best_c5 = current_c5
        else:
            current_c5 = objective(guess)
            if current_c5 < best_c5:
                best_h = guess
                best_c5 = current_c5
    
    # Perturb and optimize the best solution
    for _ in range(5):
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        perturbed_h = best_h + perturbation
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        current_sum = np.sum(perturbed_h)
        target_sum = n_points / 2
        scale = target_sum / current_sum
        perturbed_h *= scale
        
        result = minimize(
            objective,
            perturbed_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_h = result.x
                best_c5 = current_c5
        else:
            current_c5 = objective(perturbed_h)
            if current_c5 < best_c5:
                best_h = perturbed_h
                best_c5 = current_c5
    
    return (best_h, best_c5, n_points)
