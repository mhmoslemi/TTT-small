import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
from multiprocessing import Pool, cpu_count

def objective(h, dx):
    # Compute the full cross-correlation of h and (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # The maximum value of the correlation corresponds to the maximum overlap
    return np.max(corr) * dx

def generate_initial_guess(n_points):
    # Generate a random initial guess with sum(h) = n_points / 2
    h = np.random.uniform(0, 1, n_points)
    sum_h = np.sum(h)
    scale = (n_points / 2) / sum_h
    h = h * scale
    h = np.clip(h, 0, 1)
    return h

def evaluate_and_optimize(args):
    seed, n_points, dx = args
    np.random.seed(seed)
    h_initial = generate_initial_guess(n_points)
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
    result = minimize(objective, h_initial, args=(dx,), method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
    if result.success:
        return result.fun, result.x
    else:
        # Fallback to the initial guess if optimization fails
        return objective(h_initial, dx), h_initial

def run(seed=42, budget_s=100, **kwargs):
    n_points = 100  # Increased to 100 for better resolution
    dx = 2.0 / n_points
    num_starts = 20  # More starting points for diversity
    seeds = [seed + i for i in range(num_starts)]
    args_list = [(s, n_points, dx) for s in seeds]
    
    # Use multiprocessing to parallelize the optimization
    with Pool(processes=min(cpu_count(), len(args_list))) as pool:
        results = pool.map(evaluate_and_optimize, args_list)
    
    # Find the best solution
    best_c5, best_h = min(results, key=lambda x: x[0])
    
    return best_h, best_c5, n_points
