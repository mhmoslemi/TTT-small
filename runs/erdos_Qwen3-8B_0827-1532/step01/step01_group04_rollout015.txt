    best_c5 = float('inf')
    elapsed_time = 0
    
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    def constraint(h):
        return np.sum(h) - target_sum
    
    for guess in initial_guesses:
        start_time = time.time()
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': constraint}],
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        elapsed_time = time.time() - start_time
        if elapsed_time >= budget_s:
            break
        if result.success:
            current_h = result.x
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
        else:
            # Fallback to the guess
            current_h = guess
            current_c5 = objective(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
    
    if best_h is None:
        best_h = initial_h_values
        best_c5 = evaluate_erdos_solution(best_h)
    
    return best_h, best_c5, n_points
