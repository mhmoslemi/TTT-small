import numpy as np
from scipy.optimize import minimize, Bounds, LinearConstraint
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    conv = correlate(h_values, 1 - h_values, mode='full')
    max_overlap = np.max(conv) * dx
    return max_overlap

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = 50  # Reduced for faster optimization
    dx = 2.0 / n_points
    sum_h = n_points / 2
    constraint = LinearConstraint(np.ones(n_points), [sum_h], [sum_h])
    bounds = Bounds(0, 1)
    
    def objective(h):
        conv = correlate(h, 1 - h, mode='full')
        return np.max(conv) * dx
    
    # Generate multiple initial guesses with perturbations
    initial_guesses = []
    for _ in range(5):
        perturbed = initial_h.copy()
        noise = np.random.normal(0, 0.05, size=len(perturbed))
        perturbed += noise
        perturbed = np.clip(perturbed, 0, 1)
        initial_guesses.append(perturbed)
    
    best_c5 = float('inf')
    best_h = initial_h.copy()
    
    start_time = time.time()
    
    for guess in initial_guesses:
        # Run optimization with COBYLA for better constraint handling
        result = minimize(
            objective, guess, method='COBYLA', bounds=bounds, constraints=constraint, tol=1e-6
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.success:
            current_c5 = evaluate_erdos_solution(result.x, n_points)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # Fallback to initial guess if no improvement
    if best_c5 == float('inf'):
        best_c5 = evaluate_erdos_solution(best_h, n_points)
    
    return best_h, best_c5, n_points
