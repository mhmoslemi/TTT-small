import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = 60
    dx = 2.0 / n_points
    
    # Generate a symmetric initial guess with binary values
    half = n_points // 2
    ones_in_half = half // 2

    # Mirror the first half to the second half

    def objective(h):

        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Start optimization with increased iterations for better convergence
    start_time = time.time()

        objective,
        h_initial,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-8, 'maxiter': 200}
    )

    if elapsed >= budget_s:

    h_values = result.x
