import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    n_points = 80
    dx = 2.0 / n_points
    initial_h = initial_h_values  # Assume this is a global variable
    
    # Define the objective function to minimize the maximum overlap integral
    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx  # Scale by dx as per the problem statement
    
    # Define the constraint: sum(h) must equal n_points / 2
    constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
    
    # Define bounds for each h[i] to be in [0, 1]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Generate a population that includes the initial guess and some perturbations
    def generate_initial_population():
        pop = []
        # Add the initial guess
        pop.append(initial_h.copy())
        # Add perturbed versions of the initial guess
        for _ in range(5):
            perturbation = np.random.normal(0, 0.05, n_points)
            perturbed = initial_h + perturbation
            perturbed = np.clip(perturbed, 0, 1)
            pop.append(perturbed)
        # Add random guesses
