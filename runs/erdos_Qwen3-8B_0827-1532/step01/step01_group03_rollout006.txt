import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint

def objective(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=100, **kwargs):

    n_points = 100
    dx = 2.0 / n_points
    
    # Generate multiple diverse initial guesses

    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    
    # Generate random initial guesses with different structures
    for start_idx in range(num_starts):
        # Random binary-like initial guess with sum constraint

        scale = (n_points / 2) / sum_h
        h_initial = h_initial * scale
        h_initial = np.clip(h_initial, 0, 1)  # Ensure values are within [0,1]
        
        # Define bounds and constraints

        cons = LinearConstraint(np.ones(n_points), lb=n_points/2, ub=n_points/2)
        
        # Run optimization
