import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h_initial = initial_h_values.copy()

    def objective(h):
        correlate_result = np.correlate(h, 1 - h, mode='full')
        max_value = np.max(correlate_result) * dx
        return max_value

    def constraint(h):
        return np.sum(h) - n_points / 2

    bounds = Bounds(0.0, 1.0)

    def optimize(starting_h):
        result = minimize(objective, starting_h, method='L-BFGS-B', bounds=bounds, constraints={'type': 'eq', 'fun': constraint}, options={'maxiter': 200})
        if result.success:
            return result.x, result.fun
        else:
            return starting_h, objective(starting_h)

    initial_guesses = [h_initial.copy()]
    for _ in range(5):
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        perturbed_h = h_initial + perturbation
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        current_sum = np.sum(perturbed_h)
        target_sum = n_points / 2
        scale = target_sum / current_sum
        perturbed_h *= scale
        initial_guesses.append(perturbed_h)

    best_h, best_c5 = optimize(h_initial)
    best_h = best_h.copy()

    for guess in initial_guesses[1:]:
        h_perturbed, c5_perturbed = optimize(guess)
        if c5_perturbed < best_c5:
            best_h, best_c5 = h_perturbed, c5_perturbed

    for _ in range(5):
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)

        perturbed_h *= scale
        h_perturbed, c5_perturbed = optimize(perturbed_h)
        if c5_perturbed < best_c5:
            best_h, best_c5 = h_perturbed, c5_perturbed
