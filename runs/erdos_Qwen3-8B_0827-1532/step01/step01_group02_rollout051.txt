from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=100, **kwargs):

    # Initial guess from provided initial_h_values (n_points=70)
    h_initial = initial_h_values.copy()
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        # Compute cross-correlation
        corr = correlate(h, 1 - h, mode='full')
        # Find maximum value
        max_corr = np.max(corr)
        return max_corr * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    best_h = h_initial
    best_c5 = objective(best_h)
    elapsed = 0.0

        h = np.random.uniform(0.0, 1.0, n_points)
        scale_factor = target_sum / np.sum(h)
        return h * scale_factor

    # Run multiple restarts with different initial guesses
    num_restarts = 10
    for _ in range(num_restarts):

            h_perturbed += perturbation
            h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
            # Scale to meet sum constraint
            current_sum = np.sum(h_perturbed)
            scale_factor = target_sum / current_sum
            h_perturbed *= scale_factor

        # Optimize using L-BFGS-B method
        start_time = time.time()
        result = minimize(objective, h_perturbed, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
