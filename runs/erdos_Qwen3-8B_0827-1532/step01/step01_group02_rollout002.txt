from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    # Initial guess from provided initial_h_values (n_points=70)
    h_initial = initial_h_values.copy()
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        # Compute cross-correlation
        corr = correlate(h, 1 - h, mode='full')
        # Find maximum value
        max_corr = np.max(corr)
        return max_corr * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    best_h = h_initial
    best_c5 = objective(best_h)
    elapsed = 0.0

    # Generate a random initial guess
    def generate_random_h():
        h_rand = np.random.uniform(0.0, 1.0, size=n_points)
        h_rand = h_rand * (target_sum / np.sum(h_rand))
        return h_rand

    # Run multiple restarts with different initial guesses
    num_restarts = 10
    initial_guesses = [h_initial, generate_random_h()]
    
    for i, initial_guess in enumerate(initial_guesses):
        # Perturb initial solution
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        h_perturbed = initial_guess + perturbation
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        # Scale to meet sum constraint
        current_sum = np.sum(h_perturbed)
        scale_factor = target_sum / current_sum
        h_perturbed *= scale_factor

        # Optimize
        start_time = time.time()
        result = minimize(objective, h_perturbed, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5)
        elapsed += time.time() - start_time
