import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
import time

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h_values, 1 - h_values, mode='full')
    max_overlap = np.max(conv) * dx
    return max_overlap

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    # Use a larger number of points for better resolution
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate a better initial guess with a structured pattern

    # Create a periodic pattern with alternating 0.5 and 0.5 to balance the sum
    h_initial[::2] = 0.5
    h_initial[1::2] = 0.5
    # Adjust to meet the sum constraint
    current_sum = np.sum(h_initial)
    scaling_factor = target_sum / current_sum

    constraint = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
    bounds = Bounds(0, 1)
    
    # Objective function
    def objective(h):
        conv = np.correlate(h, 1 - h, mode='full')
