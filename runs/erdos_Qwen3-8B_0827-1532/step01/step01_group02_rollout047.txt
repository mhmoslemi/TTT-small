from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    # Use n_points=100 for better resolution without excessive computation
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Generate multiple initial guesses
    initial_guesses = []
    initial_guesses.append(initial_h_values.copy())
    for _ in range(4):
        h_rand = np.random.uniform(0.0, 1.0, size=n_points)
        current_sum = np.sum(h_rand)
        scale_factor = target_sum / current_sum
        h_rand *= scale_factor
        initial_guesses.append(h_rand)

    best_h = initial_guesses[0]
    best_c5 = float('inf')
    elapsed = 0.0

    # Define objective function
    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    # Run multiple restarts with different methods and initial guesses
    num_restarts = 10
    for i, h_initial in enumerate(initial_guesses):
        # Perturb initial solution
        perturbation = np.random.normal(0, 0.05, size=n_points)
        h_perturbed = h_initial + perturbation
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        current_sum = np.sum(h_perturbed)
        scale_factor = target_sum / current_sum
        h_perturbed *= scale_factor

        # Try different optimization methods
        for method in ['L-BFGS-B', 'SLSQP']:
            start_time = time.time()
            result = minimize(objective, h_perturbed, method=method, bounds=bounds, constraints=cons, tol=1e-6)
            elapsed += time.time() - start_time

            if elapsed >= budget_s:
                break

            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
            else:
                # Fallback to perturbed solution
                current_c5 = objective(h_perturbed)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h_perturbed
