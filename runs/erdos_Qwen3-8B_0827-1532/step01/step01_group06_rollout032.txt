import numpy as np
from scipy.optimize import differential_evolution, minimize

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def objective(h):
        correlate_result = np.correlate(h, 1 - h, mode='full')
        max_value = np.max(correlate_result) * dx
        return max_value

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def optimize_with_de():
        def objective_with_penalty(h):
            obj = objective(h)
            penalty = 1e3 * np.abs(constraint_sum(h))
            return obj + penalty
        result = differential_evolution(
            objective_with_penalty, bounds, 
            strategy='best1bin', maxiter=100, popsize=15, 
            tol=1e-6, mutation=(0.5, 1), recombination=0.8, 
            seed=seed
        )
        return result.x, result.fun

    def optimize_with_slsqp(starting_h):
        result = minimize(
            objective, starting_h, method='SLSQP', 
            bounds=bounds, constraints={'type': 'eq', 'fun': constraint_sum}, 
            options={'maxiter': 100}
        )
        if result.success:
            return result.x, result.fun
        else:
            return starting_h, objective(starting_h)

    np.random.seed(seed)
    h_random = np.random.rand(n_points)
    h_random = h_random * (n_points / 2) / np.sum(h_random)

    best_h, best_c5 = optimize_with_de()
    refined_h, refined_c5 = optimize_with_slsqp(best_h)
    if refined_c5 < best_c5:
        best_h, best_c5 = refined_h, refined_c5

    for guess in [h_random, initial_h_values.copy()]:
        h_perturbed, c5_perturbed = optimize_with_slsqp(guess)
        if c5_perturbed < best_c5:
            best_h, best_c5 = h_perturbed, c5_perturbed

    for _ in range(5):
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        perturbed_h = best_h + perturbation
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        current_sum = np.sum(perturbed_h)
        target_sum = n_points / 2
        scale = target_sum / current_sum
        perturbed_h *= scale
        h_perturbed, c5_perturbed = optimize_with_slsqp(perturbed_h)
        if c5_perturbed < best_c5:
            best_h, best_c5 = h_perturbed, c5_perturbed

    return (best_h, best_c5, n_points)
