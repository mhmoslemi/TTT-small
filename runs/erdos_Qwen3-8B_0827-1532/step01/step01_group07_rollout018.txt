import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def objective(h_flat):

    bounds = [(0, 1) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = initial_h.copy()

    method = 'trust-constr'
    maxiter = 200
    
    # Initial guess
    result = minimize(objective, best_h, method=method, bounds=bounds, constraints=cons, options={'maxiter': maxiter})

        best_h = result.x
