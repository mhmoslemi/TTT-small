import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Generate initial guesses
    def generate_initial_guess():
        h = np.zeros(n_points)
        indices = np.random.choice(n_points, size=n_points//2, replace=False)
        h[indices] = 1.0
        return h
    
    h_initial1 = generate_initial_guess()
    h_initial2 = np.zeros(n_points)
    h_initial2[::2] = 1.0  # Alternating pattern
    h_initial3 = np.zeros(n_points)
    h_initial3[::3] = 1.0  # Spaced pattern
    
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h_initial1
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Run optimization for each initial guess
    for h_initial in [h_initial1, h_initial2, h_initial3]:
        result = minimize(
            objective,
            h_initial,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-8, 'maxiter': 200}
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Fallback to initial guess if optimization fails
    if best_c5 == float('inf'):
        best_h = h_initial1
        best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
