from scipy.optimize import differential_evolution

def objective(h, dx, n_points, penalty_coeff):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    sum_penalty = penalty_coeff * (np.sum(h) - n_points / 2)**2
    return max_corr * dx + sum_penalty

    n_points = 80
    dx = 2.0 / n_points
    h_initial = initial_h_values.copy()
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    penalty_coeff = 1e6
    
    best_c5 = float('inf')
    best_h = h_initial
    
    # Generate multiple initial guesses with different perturbations
    num_starts = 5
    for i in range(num_starts):
        np.random.seed(seed + i)
        h_perturbed = h_initial + np.random.normal(0, 0.05, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        current_sum = np.sum(h_perturbed)
        scale = (n_points / 2) / current_sum
        h_perturbed = h_perturbed * scale
        
        # Run differential evolution
        result = differential_evolution(
            objective,
            bounds,
            args=(dx, n_points, penalty_coeff),
            strategy='best1bin',
            maxiter=100,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed + i
        )
        
        if result.success:
            current_c5 = result.fun - penalty_coeff * (np.sum(result.x) - n_points / 2)**2
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            # Fallback to perturbed guess
            current_c5 = objective(h_perturbed, dx, n_points, penalty_coeff) - penalty_coeff * (np.sum(h_perturbed) - n_points / 2)**2
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_perturbed
    
    return best_h, best_c5, n_points
