from scipy.optimize import differential_evolution, minimize

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h_initial = initial_h_values.copy()

        mutation=(0.5, 1), recombination=0.8, seed=seed

    best_h_de = result_de.x
    best_c5_de = objective_de(best_h_de)

    # Refine with SLSQP
    def objective_slsqp(h):
        correlate_result = np.correlate(h, 1 - h, mode='full')
        max_value = np.max(correlate_result) * dx

    bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]
    result_slsqp = minimize(

    best_h, best_c5 = result_slsqp.x, result_slsqp.fun
