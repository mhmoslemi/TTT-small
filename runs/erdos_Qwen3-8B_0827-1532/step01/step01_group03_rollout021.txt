    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        sum_h = np.sum(h)
        penalty = 1e4 * abs(sum_h - n_points / 2)
        return max_corr * dx + penalty
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Run differential evolution
    result = differential_evolution(objective, bounds, strategy='best1bin',

                                   tol=1e-6, maxiter=100, seed=seed)
    
    # Adjust the best_h to meet the sum constraint
    best_h = result.x
    sum_h = np.sum(best_h)
    scale = (n_points / 2) / sum_h

    # Compute the actual C5 value
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    max_corr = np.max(corr)
    best_c5 = max_corr * dx
    
    return best_h, best_c5, n_points
