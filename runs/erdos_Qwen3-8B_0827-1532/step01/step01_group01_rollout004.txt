    num_starts = 5
    initial_guesses = []
    for _ in range(num_starts):
        perturbed = initial_h.copy()
        # Add random noise
        noise = np.random.normal(0, 0.05, n_points)
        perturbed += noise
        # Clip to [0,1] and adjust to meet the sum constraint
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        adjustment = (target_sum - current_sum) / n_points
        perturbed += adjustment
        perturbed = np.clip(perturbed, 0, 1)
        initial_guesses.append(perturbed)
    
    for guess in initial_guesses:
        # Run SLSQP
        result = minimize(objective, guess, method='SLSQP', bounds=bounds, constraints=constraint, tol=1e-6)
        if result.success:
            c5 = evaluate_erdos_solution(result.x, n_points)
            if c5 < best_c5:
                best_c5 = c5
                best_h = result.x
    
    # Check remaining time
    elapsed = time.time() - start_time
    if elapsed < budget_s:
        # Run one more optimization with a random initial guess
        random_guess = np.random.rand(n_points)
        random_guess = np.clip(random_guess, 0, 1)
        current_sum = np.sum(random_guess)
        adjustment = (target_sum - current_sum) / n_points
        random_guess += adjustment
        random_guess = np.clip(random_guess, 0, 1)
        result = minimize(objective, random_guess, method='SLSQP', bounds=bounds, constraints=constraint, tol=1e-6)
        if result.success:
            c5 = evaluate_erdos_solution(result.x, n_points)
            if c5 < best_c5:
                best_c5 = c5
                best_h = result.x
    
    return best_h, best_c5, n_points
