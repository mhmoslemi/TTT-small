from scipy.signal import correlate
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
