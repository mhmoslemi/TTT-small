import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate

def objective(h, dx):
    # Compute the full cross-correlation between h and 1-h
    corr = correlate(h, 1 - h, mode='full')
    # The maximum value of the correlation (scaled by dx)
    return np.max(corr) * dx

def run(seed=42, budget_s=100, **kwargs):

    best_h = np.zeros(n_points)
    
    # Generate multiple starting points

    np.random.seed(seed)
    
    for start_idx in range(num_starts):
        # Random initial guess with sum constraint

        # Perturb with small noise

        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        
        # Define bounds and constraints
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
        
        # Run optimization with COBYLA
        result = minimize(
