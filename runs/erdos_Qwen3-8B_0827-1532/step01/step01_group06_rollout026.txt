        correlate_result = np.correlate(h, 1 - h, mode='full')

        return max_value

        return np.sum(h) - n_points / 2

            method='trust-constr',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 200, 'disp': False}
