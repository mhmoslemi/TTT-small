import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Define n_points
    n_points = 45
    dx = 2.0 / n_points
    
    # Generate initial guesses
    initial_guesses = []
    # Add the initial_h_values provided
    initial_guesses.append(initial_h_values.copy())
    # Generate a few random initial guesses
    for _ in range(3):
        h = np.random.rand(n_points)
        total = np.sum(h)
        scaling = (n_points / 2) / total
        h *= scaling
        initial_guesses.append(h)
    
    # Define the objective function
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define constraints
    constraints = [LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])]
    bounds = Bounds(np.zeros(n_points), np.ones(n_points))
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    for guess in initial_guesses:
        # Run optimization with Nelder-Mead
        result = minimize(
            objective,
            guess,
            method='Nelder-Mead',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 100, 'xatol': 1e-8, 'fatol': 1e-8}
        )
        elapsed_time = time.time() - start_time
        if elapsed_time >= budget_s:
            break
        # Update best solution
        if result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # If time is up, return the best solution found
    return best_h, best_c5, n_points
