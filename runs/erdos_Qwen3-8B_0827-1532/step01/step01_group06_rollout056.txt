            a, b = random.sample(range(pop_size), 2)
            if fitness[a] < fitness[b]:
                parents.append(population[a])
            else:
                parents.append(population[b])
        
        offspring = []
        for i in range(0, len(parents), 2):
            p1, p2 = parents[i], parents[i+1]
            child = np.zeros(n_points)
            for j in range(n_points):
                if random.random() < 0.5:
                    child[j] = p1[j]
                else:
                    child[j] = p2[j]
            for j in range(n_points):
                if random.random() < mutation_rate:
                    child[j] += np.random.uniform(-mutation_delta, mutation_delta)
            sum_child = np.sum(child)
            if sum_child != 0:
                child = child * (n_points / 2) / sum_child
            offspring.append(child)
        
        population = parents + offspring
        population.sort(key=lambda x: evaluate_erdos_solution(x))
        population = population[:pop_size]
    
    return (best_h, best_c5, n_points)
