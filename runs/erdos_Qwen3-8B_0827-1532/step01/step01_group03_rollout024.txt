    num_starts = 10
    for i in range(num_starts):
        np.random.seed(seed + i)
        # Perturb initial guess with larger variation
        h_perturbed = h_initial + np.random.normal(0, 0.15, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        current_sum = np.sum(h_perturbed)
        scale = (n_points / 2) / current_sum
        h_perturbed = h_perturbed * scale
        
        # Use differential evolution for global search
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = differential_evolution(
            objective,
            bounds,
            args=(dx, n_points),
            strategy='best1bin',
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.9,
            polish=True,
            seed=seed + i,
            maxiter=100
        )
        
        if result.success:
            # Extract the actual C5 value by subtracting penalty
            c5 = result.fun - 1e6 * abs(np.sum(result.x) - n_points / 2)
            if c5 < best_c5:
                best_c5 = c5
                best_h = result.x
        else:
            # Fallback to perturbed guess if optimization fails
            c5 = objective(h_perturbed, dx, n_points) - 1e6 * abs(np.sum(h_perturbed) - n_points / 2)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_perturbed
    
    # Final refinement with SLSQP
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
    result = minimize(
        lambda h: objective(h, dx, n_points) - 1e6 * abs(np.sum(h) - n_points / 2),
        best_h,
        bounds=bounds,
        constraints=cons,
        method='SLSQP',
        options={'ftol': 1e-8, 'maxiter': 100}
    )
    
    if result.success:
        best_c5 = result.fun - 1e6 * abs(np.sum(result.x) - n_points / 2)
        best_h = result.x
    
    return best_h, best_c5, n_points
