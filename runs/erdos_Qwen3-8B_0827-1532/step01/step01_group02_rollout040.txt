    num_restarts = 10  # Increased number of restarts
    for _ in range(num_restarts):
        # Perturb initial solution
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        h_perturbed = h_initial + perturbation
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        # Scale to meet sum constraint
        current_sum = np.sum(h_perturbed)
        scale_factor = target_sum / current_sum
        h_perturbed *= scale_factor

        # Optimize with COBYLA method
        start_time = time.time()
        result = minimize(objective, h_perturbed, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-6)
        elapsed += time.time() - start_time

        if elapsed >= budget_s:
            break

        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            # Fallback to perturbed solution
            current_c5 = objective(h_perturbed)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_perturbed

    return (best_h, best_c5, n_points)
