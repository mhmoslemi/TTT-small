            options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8}
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if best_result is None or result.fun < best_result.fun:
            best_result = result
    
    # If no result found, return the initial guess
    if best_result is None:
        return initial_h_values, objective(initial_h_values), n_points
    
    return best_result.x, best_result.fun, n_points
