        result_de = differential_evolution(objective_de, bounds_de, strategy='best1bin', maxiter=50, popsize=15, tol=1e-6, mutation=(0.5, 1), recombination=0.8, seed=seed)
        # Evaluate
        current_c5_de = evaluate_erdos_solution(result_de.x, n_points)
        if current_c5_de < best_c5:
            best_c5 = current_c5_de
            best_h = result_de.x
    return best_h, best_c5, n_points
