def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Try different n_points values for better resolution
    n_points = 60  # Increased from 45 for better resolution
    dx = 2.0 / n_points
    initial_guesses = []
    
    # Add initial_h_values as a starting point
    initial_guesses.append(initial_h_values.copy())
    
    # Generate a few diverse initial guesses
    # 1. Random guess with constraint enforcement
    random_guess = np.random.rand(n_points)
    # Adjust to meet the sum constraint
    random_guess -= np.mean(random_guess)
    random_guess = np.clip(random_guess, 0, 1)
    initial_guesses.append(random_guess)
    
    # 2. Binary step function with half 1s and half 0s
    binary_guess = np.zeros(n_points)
    binary_guess[:n_points//2] = 1
    np.random.shuffle(binary_guess)
    initial_guesses.append(binary_guess)
    
    # 3. Random perturbation of initial_h_values
    perturbed_guess = initial_h_values.copy()
    perturbed_guess += np.random.uniform(-0.1, 0.1, size=n_points)
    perturbed_guess = np.clip(perturbed_guess, 0, 1)
    initial_guesses.append(perturbed_guess)
    
    # Define objective function
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define constraints: sum(h) == n_points / 2 and 0 <= h[i] <= 1
    constraints = [LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])]
    bounds = Bounds(np.zeros(n_points), np.ones(n_points))

    start_time = time.time()
    
    # Run optimization for each initial guess
    for guess in initial_guesses:
        # Clip and ensure feasibility
        guess = np.clip(guess, 0, 1)
        # Run optimization
        result = minimize(
            objective,
            guess,
            method='trust-constr',
            bounds=bounds,
            constraints=constraints,
            options={

                'eps': 1e-8,

        )
