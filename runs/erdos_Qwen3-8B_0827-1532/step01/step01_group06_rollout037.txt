from scipy.optimize import minimize, differential_evolution
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Increased for better resolution
    dx = 2.0 / n_points
    initial_h = initial_h_values.copy()
    initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h)
    initial_h = np.clip(initial_h, 0.0, 1.0)
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)

    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr) * dx
        return max_corr

    def constraint(h):
        return np.sum(h) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def optimize(starting_h):
        result = minimize(objective, starting_h, method='L-BFGS-B', bounds=bounds, constraints={'type': 'eq', 'fun': constraint}, options={'maxiter': 200, 'ftol': 1e-8})
        if result.success:
            return result.x, result.fun
        else:
            return starting_h, objective(starting_h)

    best_h, best_c5 = optimize(initial_h)

    for _ in range(5):
        perturbation = np.random.uniform(-0.2, 0.2, size=n_points)
        perturbed_h = initial_h + perturbation
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        current_sum = np.sum(perturbed_h)
        target_sum = n_points / 2
        scale = target_sum / current_sum
        perturbed_h *= scale
        h_perturbed, c5_perturbed = optimize(perturbed_h)
        if c5_perturbed < best_c5:
            best_h, best_c5 = h_perturbed, c5_perturbed
