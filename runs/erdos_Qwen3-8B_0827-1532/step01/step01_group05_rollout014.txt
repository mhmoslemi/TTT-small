import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 160  # Increased points for finer resolution
    dx = 2.0 / n_points
    
    # Generate a random initial guess with correct sum
    h = np.random.rand(n_points)
    h = h / np.sum(h) * (n_points / 2)
    h = np.clip(h, 0, 1)
    
    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    result = minimize(
        objective,
        h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=constraint,
        options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8}
    )
    
    best_h = result.x
    c5_bound = result.fun
    
    return (best_h, c5_bound, n_points)
