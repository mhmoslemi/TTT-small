import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    # Use a lower n_points for faster evaluation
    n_points = 50
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        # Normalize to ensure sum constraint
        h_norm = h * (target_sum / np.sum(h)) if np.sum(h) != 0 else h
        # Compute cross-correlation
        corr = np.correlate(h_norm, 1 - h_norm, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Use initial guess if available
    if 'initial_h_values' in globals():
        initial_guess = initial_h_values.copy()
    else:
        initial_guess = np.random.rand(n_points)

    # Scale initial guess to meet sum constraint
    initial_guess = initial_guess * (target_sum / np.sum(initial_guess))

    best_h = initial_guess
    best_c5 = objective(best_h)
    elapsed = 0.0

    # Run differential evolution with multiple restarts
    num_restarts = 3
    for i in range(num_restarts):
        start_time = time.time()
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=15,
            mutation=(0.5, 1),
            recombination=0.8,
            tol=1e-6,
            maxiter=50,
            seed=seed + i
        )
        elapsed += time.time() - start_time

        if elapsed >= budget_s:
            break

        if result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x

        # Fallback to perturbed initial guess if needed
        if np.isnan(best_c5):
            perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
            perturbed = initial_guess + perturbation
            perturbed = np.clip(perturbed, 0.0, 1.0)
            perturbed = perturbed * (target_sum / np.sum(perturbed))
            best_c5 = objective(perturbed)
            best_h = perturbed

    return (best_h, best_c5, n_points)
