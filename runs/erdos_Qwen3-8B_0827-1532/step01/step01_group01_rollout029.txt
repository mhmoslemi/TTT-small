import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
import time

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h_values, 1 - h_values, mode='full')
    max_overlap = np.max(conv) * dx
    return max_overlap

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    # Choose n_points
    n_points = 50
    dx = 2.0 / n_points
    # Generate initial guesses
    initial_guesses = []
    # First guess: even-odd pattern
    h_initial = np.zeros(n_points)
    h_initial[::2] = 1
    initial_guesses.append(h_initial)
    # Generate a few random binary initial guesses
    for _ in range(3):
        indices = np.random.choice(n_points, size=25, replace=False)
        h = np.zeros(n_points)
        h[indices] = 1
        initial_guesses.append(h)
    # Also, add a perturbed version of the even-odd guess
    perturbed = h_initial.copy()
    perturbed += np.random.uniform(-0.1, 0.1, size=n_points)
    perturbed = np.clip(perturbed, 0, 1)
    initial_guesses.append(perturbed)
    # Define objective
    def objective(h):
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv) * dx
    # Constraints: sum(h) = n_points / 2
    constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
    bounds = Bounds(0, 1)
    # Track best result
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    # Run optimization for each initial guess
    for guess in initial_guesses:
        result = minimize(objective, guess, method='SLSQP', bounds=bounds, constraints=constraint, tol=1e-6)
        if result.success:
            current_c5 = evaluate_erdos_solution(result.x, n_points)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            # Fallback to initial guess
            current_c5 = evaluate_erdos_solution(guess, n_points)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = guess
        # Check time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    return best_h, best_c5, n_points
