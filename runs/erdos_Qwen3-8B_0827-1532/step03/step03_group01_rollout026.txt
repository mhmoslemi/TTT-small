        if result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # If no optimization was done, use the first initial guess
    if best_h is None:
        result = minimize(
            objective,
            initial_guesses[0],
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 200}
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points
