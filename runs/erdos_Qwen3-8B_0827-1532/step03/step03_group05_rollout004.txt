import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = 128
    dx = 2.0 / n_points
    np.random.seed(seed)
    
    # Generate diverse initial guesses
    initial_guesses = []
    
    # 1. Resampled initial_h_values
    x_initial = np.linspace(0, 2, 80)
    f_initial = np.interp(x_initial, x_initial, initial_h_values)
    x_new = np.linspace(0, 2, n_points)
    resampled_initial = np.interp(x_new, x_initial, f_initial)
    initial_guesses.append(resampled_initial)
    
    # 2. Random initial guess
    random_initial = np.zeros(n_points)
    indices = np.random.choice(n_points, size=n_points//2, replace=False)
    random_initial[indices] = 1
    initial_guesses.append(random_initial)
    
    # 3. Structured initial guess (every other interval)
    structured_initial = np.zeros(n_points)
    structured_initial[::2] = 1
    initial_guesses.append(structured_initial)
    
    # 4. Evenly spaced initial guess
    evenly_spaced_initial = np.zeros(n_points)
    for i in range(n_points//2):
        pos = int(i * n_points / (n_points//2))
        evenly_spaced_initial[pos] = 1
    initial_guesses.append(evenly_spaced_initial)
    
    # 5. Perturbed versions of initial_h_values
    for _ in range(3):
        perturbed_initial = resampled_initial.copy()
        perturbed_initial += np.random.normal(0, 0.05, size=n_points)
        perturbed_initial = np.clip(perturbed_initial, 0, 1)
        perturbed_initial = perturbed_initial * (n_points / 2) / np.sum(perturbed_initial)
        initial_guesses.append(perturbed_initial)
    
    # 6. Additional structured patterns
    for pattern in [np.zeros(n_points), np.ones(n_points)]:
        for shift in range(3):
            shifted = np.roll(pattern, shift)
            shifted = shifted * (n_points / 2) / np.sum(shifted)
            initial_guesses.append(shifted)
    
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    
    def objective(h):
        one_minus_h = 1 - h
        corr = correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    for guess in initial_guesses:
        # Use different optimization methods for different guesses
        if np.random.rand() < 0.5:
            method = 'SLSQP'
        else:
            method = 'L-BFGS-B'
        
        result = minimize(
            objective,
            guess,
            method=method,
            bounds=bounds,
            constraints=constraint,
            options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8}
        )
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        # Early stopping if time is exceeded
        if time.time() - start_time > budget_s:
            break
    
    return (best_h, best_c5, n_points)
