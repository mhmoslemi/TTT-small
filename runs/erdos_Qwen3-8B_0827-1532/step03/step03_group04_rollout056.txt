import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import COBYLA

def objective(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=100, **kwargs):
    n_points = 64  # Smaller for faster evaluation
    dx = 2.0 / n_points
    np.random.seed(seed)
    
    # Generate multiple random initial guesses
    num_starts = 30
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    
    for i in range(num_starts):
        # Generate random initial guess
        h_initial = np.random.uniform(0, 1, n_points)
        current_sum = np.sum(h_initial)
        h_initial = h_initial * (n_points / 2) / current_sum
        
        bounds = Bounds(0.0, 1.0)
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
        
        # Use COBYLA for constrained optimization
        result = minimize(
            objective,

            args=(dx,), 
            method='COBYLA', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 200, 'disp': False}
        )
        
        if result.success:
