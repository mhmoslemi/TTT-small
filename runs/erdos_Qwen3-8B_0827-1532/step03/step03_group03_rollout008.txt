    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    
    def objective(h):
        corr = correlate(h, 1 - h, mode='full')

        return max_corr * dx
    
    constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}

    # Use multiprocessing to parallelize optimization
    pool = mp.Pool(processes=mp.cpu_count())

    for guess in initial_guesses:

        perturb = np.random.normal(0, 0.01, n_points)

            perturbed_guess = perturbed_guess * (n_points / 2) / sum_perturbed
        # Submit to pool
        result = pool.apply_async(minimize,
