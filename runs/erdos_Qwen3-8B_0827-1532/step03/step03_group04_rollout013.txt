import numpy as np
from scipy.optimize import minimize

def objective(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=100, **kwargs):
    n_points = 80
    dx = 2.0 / n_points
    best_c5 = float('inf')
    best_h = None
    
    # Use initial guess and random starts
    num_starts = 30
    for i in range(num_starts):
        np.random.seed(seed + i)
        # Generate random initial guess
        h_initial = np.random.uniform(0, 1, n_points)
        current_sum = np.sum(h_initial)
        scale = (n_points / 2) / current_sum
        h_initial = h_initial * scale
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
        
        # Use L-BFGS-B for faster convergence
        result = minimize(
            objective, 
            h_initial, 
            args=(dx,), 
            method='L-BFGS-B', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 500, 'ftol': 1e-9, 'gtol': 1e-9}
        )
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            # Fallback to perturbed guess if optimization fails
            current_c5 = objective(h_initial, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_initial
    
    return best_h, best_c5, n_points
