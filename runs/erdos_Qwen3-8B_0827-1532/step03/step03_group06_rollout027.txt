        def objective_de(h):
            corr = correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            penalty = 1e4 * abs(np.sum(h) - n_points / 2)
            return max_corr * dx + penalty
        
        result = differential_evolution(
            objective_de,
            bounds,
            strategy='best1bin',
            popsize=15,
            maxiter=50,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            disp=False
        )
        
        # Refine with SLSQP
        h_refine = result.x
        sum_h = np.sum(h_refine)
        scaling = (n_points / 2) / sum_h
        h_refine *= scaling
        h_refine = np.clip(h_refine, 0, 1)
        
        result_slsqp = minimize(
            objective,
            h_refine,
            method='SLSQP',
            bounds=bounds,
            constraints=constraint,
            options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8}
        )
        
        if result_slsqp.success and result_slsqp.fun < best_c5:
            best_c5 = result_slsqp.fun
            best_h = result_slsqp.x
    
    return (best_h, best_c5, n_points)
