    # 5. Perturbed versions of initial_h_values
    perturbed_initial = resampled_initial.copy()
    perturbed_initial += np.random.normal(0, 0.05, size=n_points)
    perturbed_initial = np.clip(perturbed_initial, 0, 1)
    perturbed_initial = perturbed_initial * (n_points / 2) / np.sum(perturbed_initial)
    initial_guesses.append(perturbed_initial)

    best_c5 = float('inf')
    best_h = None
    start_time = time.time()

    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        penalty = 1e4 * (np.sum(h) - n_points / 2)**2
        return max_corr * dx + penalty

    for guess in initial_guesses:
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        # Run differential evolution
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=100,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed
        )
        h_opt = result.x
        # Compute the actual C5 value
        corr = correlate(h_opt, 1 - h_opt, mode='full')
        max_corr = np.max(corr)
        c5 = max_corr * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt
        # Check time
        if time.time() - start_time > budget_s:
            break

    return (best_h, best_c5, n_points)
