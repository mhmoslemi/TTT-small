        pos = int(i * n_points / (n_points//2))
        evenly_spaced_initial[pos] = 1
    initial_guesses.append(evenly_spaced_initial)
    # 5. Perturbed versions of initial_h_values
    perturbed_initial = resampled_initial.copy()
    perturbed_initial += np.random.normal(0, 0.1, size=n_points)
    perturbed_initial = np.clip(perturbed_initial, 0, 1)
    perturbed_initial = perturbed_initial * (n_points / 2) / np.sum(perturbed_initial)
    initial_guesses.append(perturbed_initial)
    # 6. Another perturbed version with larger noise
    perturbed_initial2 = resampled_initial.copy()
    perturbed_initial2 += np.random.normal(0, 0.2, size=n_points)
    perturbed_initial2 = np.clip(perturbed_initial2, 0, 1)
    perturbed_initial2 = perturbed_initial2 * (n_points / 2) / np.sum(perturbed_initial2)
    initial_guesses.append(perturbed_initial2)
    # 7. Binary initial guess with certain spacing
    binary_initial = np.zeros(n_points)
    for i in range(0, n_points, 4):
        binary_initial[i] = 1
    binary_initial = binary_initial * (n_points / 2) / np.sum(binary_initial)
    initial_guesses.append(binary_initial)

    best_c5 = float('inf')
    best_h = None
    start_time = time.time()

    def optimize(initial_h):
        def objective(h):
            corr = correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx

        constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
        bounds = [(0.0, 1.0) for _ in range(n_points)]

        result = minimize(
            objective,
            initial_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraint,
            options={'maxiter': 1000, 'ftol': 1e-9, 'eps': 1e-9}
        )
        return result.x, result.fun

    for guess in initial_guesses:
        h, c5 = optimize(guess)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
        # Check if time is exceeded
        if time.time() - start_time > budget_s:
            break

    return (best_h, best_c5, n_points)
