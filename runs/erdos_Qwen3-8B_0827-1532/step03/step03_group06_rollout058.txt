    def generate_perturbed_initial_guess(h_base):
        perturbation = np.random.normal(0, 0.05, size=n_points)
        h_perturbed = h_base + perturbation
        h_perturbed = np.clip(h_perturbed, 0, 1)
        sum_h = np.sum(h_perturbed)
        scaling = (n_points / 2) / sum_h
        h_perturbed *= scaling
        return h_perturbed
    
    initial_guesses = []
    initial_guesses.append(initial_h.copy())
    initial_guesses.append(generate_binary_initial_guess(n_points))
    initial_guesses.append(generate_single_interval_initial_guess(n_points))
    for _ in range(3):
        initial_guesses.append(generate_perturbed_initial_guess(initial_h))
    
    # Objective function
    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint: sum(h) = n_points / 2
    constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = None
    best_c5 = float('inf')
    
    for h0 in initial_guesses:
        result = minimize(
            objective,
            h0,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraint,
            options={'maxiter': 500, 'ftol': 1e-8, 'eps': 1e-8}
        )
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    return (best_h, best_c5, n_points)
