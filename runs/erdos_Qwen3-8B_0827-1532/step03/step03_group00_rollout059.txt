import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    start_time = time.time()
    
    # Generate multiple initial guesses
    num_initial_guesses = 10
    initial_guesses = [initial_guess]
    
    for _ in range(num_initial_guesses - 1):
        h = np.random.uniform(0, 1, n_points)
        sum_h = np.sum(h)
        scale = target_sum / sum_h
        h = h * scale
        h = np.clip(h, 0, 1)
        initial_guesses.append(h)
    
    # Optimization parameters
    max_iter = 300
    ftol = 1e-9
    
    for h in initial_guesses:
        def objective(h_vec):
            one_minus_h = 1.0 - h_vec
            corr = np.correlate(h_vec, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h_vec: np.sum(h_vec) - target_sum})
        
        result = minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': ftol, 'maxiter': max_iter}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
