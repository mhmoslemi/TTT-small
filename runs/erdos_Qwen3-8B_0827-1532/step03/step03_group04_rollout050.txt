import numpy as np
from scipy.optimize import minimize

def objective(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=100, **kwargs):
    n_points = 100  # Increased resolution for better approximation
    dx = 2.0 / n_points
    np.random.seed(seed)
    
    num_starts = 50  # More starts for better exploration
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    
    for i in range(num_starts):
        # Generate random initial guess with sum constraint
        h_perturbed = np.random.rand(n_points)
        h_perturbed = h_perturbed / np.sum(h_perturbed) * (n_points / 2)
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
        
        # Use COBYLA for better constraint handling
        result = minimize(
            objective, 
            h_perturbed, 
            args=(dx,), 
            method='COBYLA',
            bounds=bounds,
            constraints=cons,

        )
        
        if result.success:
