import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Predefined initial guess with n_points=100
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    # Generate multiple initial guesses: 1 provided + 4 random
    initial_guesses = [initial_guess]
    for _ in range(4):
        guess = np.random.rand(n_points)
        guess = guess * (target_sum / np.sum(guess))
        initial_guesses.append(guess)
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    start_time = time.time()
    
    for guess in initial_guesses:
        # Objective function to minimize max overlap
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds and constraints
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Optimization with SLSQP
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 300}
        )
        
        # Update best result
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
