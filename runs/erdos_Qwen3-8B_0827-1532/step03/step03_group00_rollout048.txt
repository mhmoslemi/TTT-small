import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    # Use the provided initial guess
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Objective function with penalty for sum constraint
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        # Penalty for sum constraint
        sum_h = np.sum(h)
        penalty = 1e6 * np.abs(sum_h - target_sum)
        return max_corr * dx + penalty
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Start optimization with initial guess
    start_time = time.time()
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=15,
        maxiter=100,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed,
        disp=False
    )
    
    # Extract best solution
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Check elapsed time
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return initial_guess, objective(initial_guess), n_points
    
    return best_h, best_c5, n_points
