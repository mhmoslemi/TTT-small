        results = pool.map(optimize, initial_guesses)
        for h, c5 in results:
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
            # Check if time is exceeded
            if time.time() - start_time > budget_s:
                break

    return (best_h, best_c5, n_points)
