from scipy.optimize import minimize
from scipy.signal import correlate
import time
from scipy.interpolate import interp1d

def run(seed=42, budget_s=100, **kwargs):
    n_points = 64
    dx = 2.0 / n_points
    initial_guesses = []

    np.random.seed(seed)
    # 1. Use the provided initial_h_values (resampled to n_points=64)
    x_initial = np.linspace(0, 2, 128)
    f_initial = interp1d(x_initial, initial_h_values, kind='linear', fill_value='extrapolate')
    x_new = np.linspace(0, 2, n_points)
    resampled_initial = f_initial(x_new)
    initial_guesses.append(resampled_initial)
    # 2. Structured initial guess (four intervals of length 0.25)
    structured_initial = np.zeros(n_points)
    for i in range(4):
        start = int(i * 0.5 / dx)
        end = start + 8
        structured_initial[start:end] = 1
    initial_guesses.append(structured_initial)
    # 3. Random initial guess
    random_initial = np.zeros(n_points)
    indices = np.random.choice(n_points, size=n_points//2, replace=False)
    random_initial[indices] = 1
    initial_guesses.append(random_initial)
    # 4. Evenly spaced initial guess
    evenly_spaced_initial = np.zeros(n_points)
    for i in range(n_points//2):
        pos = int(i * n_points / (n_points//2))
        evenly_spaced_initial[pos] = 1
    initial_guesses.append(evenly_spaced_initial)
    # 5. Perturbed versions of initial_h_values
    perturbed_initial = resampled_initial.copy()
    perturbed_initial += np.random.normal(0, 0.05, size=n_points)
    perturbed_initial = np.clip(perturbed_initial, 0, 1)
    perturbed_initial = perturbed_initial * (n_points / 2) / np.sum(perturbed_initial)
    initial_guesses.append(perturbed_initial)

    best_c5 = float('inf')
    best_h = None
    start_time = time.time()

    def optimize(initial_h):
        def objective(h):
            corr = correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx

        constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
        bounds = [(0.0, 1.0) for _ in range(n_points)]

        result = minimize(
            objective,
            initial_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraint,
            options={'maxiter': 1000, 'ftol': 1e-9, 'eps': 1e-9}
        )
        return result.x, result.fun

    for guess in initial_guesses:
        h, c5 = optimize(guess)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
        # Check if time is exceeded
        if time.time() - start_time > budget_s:
            break

    return (best_h, best_c5, n_points)
