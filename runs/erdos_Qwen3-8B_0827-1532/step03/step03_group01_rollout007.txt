    initial_guess = np.concatenate((initial_guess, [last_var]))
    
    # Define bounds for the first n_points - 1 variables
    bounds = [(0.0, 1.0) for _ in range(n_points - 1)]
    
    # Objective function
    def objective(h):
        h_full = np.zeros(n_points)
        h_full[:-1] = h
        h_full[-1] = (n_points / 2) - np.sum(h)
        h_full[-1] = np.clip(h_full[-1], 0, 1)
        one_minus_h = 1.0 - h_full
        corr = np.correlate(h_full, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    start_time = time.time()
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=20,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed
    )
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        # If time is up, return the initial guess
        h_full = np.zeros(n_points)
        h_full[:-1] = initial_guess
        h_full[-1] = (n_points / 2) - np.sum(initial_guess)
        h_full[-1] = np.clip(h_full[-1], 0, 1)
        return h_full, objective(initial_guess), n_points
    
    best_h = result.x
    # Compute the full h_full for the best_h
    h_full = np.zeros(n_points)
    h_full[:-1] = best_h
    h_full[-1] = (n_points / 2) - np.sum(best_h)
    h_full[-1] = np.clip(h_full[-1], 0, 1)
    best_c5 = objective(best_h)
    
    return h_full, best_c5, n_points
