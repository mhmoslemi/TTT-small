        current_sum = np.sum(h_perturbed)
        scale = (n_points / 2) / current_sum
        h_perturbed = h_perturbed * scale
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
        
        # Use COBYLA for better constraint handling
        result = minimize(
            objective, 
            h_perturbed, 
            args=(dx,), 
            method='COBYLA', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-9, 'xatol': 1e-9}
        )
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            current_c5 = objective(h_perturbed, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_perturbed
    
    return best_h, best_c5, n_points
