    structured_initial[::2] = 1
    initial_guesses.append(structured_initial)
    # 4. Evenly spaced initial guess
    evenly_spaced_initial = np.zeros(n_points)
    for i in range(n_points//2):
        pos = int(i * n_points / (n_points//2))
        evenly_spaced_initial[pos] = 1
    initial_guesses.append(evenly_spaced_initial)
    # 5. Binary initial guess with spread-out 1s
    spread_out_initial = np.zeros(n_points)
    for i in range(n_points//2):
        pos = int(i * n_points / (n_points//2))
        spread_out_initial[pos] = 1
    initial_guesses.append(spread_out_initial)
    # 6. Perturbations of initial_h_values
    for _ in range(5):
        perturb = np.random.normal(0, 0.05, n_points)
        perturbed = initial_h_values + perturb
        perturbed = np.clip(perturbed, 0, 1)
        sum_perturbed = np.sum(perturbed)
        if sum_perturbed != 0:
            perturbed = perturbed * (n_points / 2) / sum_perturbed
        initial_guesses.append(perturbed)
    # 7. Additional structured initial guess with staggered 1s
    staggered_initial = np.zeros(n_points)
    for i in range(n_points//2):
        pos = int((i * 2 + 1) * n_points / (n_points//2))
        staggered_initial[pos] = 1
    initial_guesses.append(staggered_initial)
    
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    
    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    for guess in initial_guesses:
        # Perturb the guess slightly to avoid local optima
        perturb = np.random.normal(0, 0.01, n_points)
        perturbed_guess = np.clip(guess + perturb, 0, 1)
        sum_perturbed = np.sum(perturbed_guess)
        if sum_perturbed != 0:
            perturbed_guess = perturbed_guess * (n_points / 2) / sum_perturbed
        # Optimize
        result = minimize(
            objective,
            perturbed_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=constraint,
            options={'maxiter': 1000, 'ftol': 1e-9, 'eps': 1e-9}
        )
        current_h = result.x
        current_c5 = result.fun
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h
        # Check time
        if time.time() - start_time > budget_s:
            break
    
    return (best_h, best_c5, n_points)
