import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    start_time = time.time()
    
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        penalty = 1e4 * (np.sum(h) - target_sum)**2
        return max_corr * dx + penalty
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    for _ in range(5):
        h_initial = np.random.rand(n_points)
        h_initial = h_initial * (target_sum / np.sum(h_initial))
        
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=100,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed
        )
        
        if result.success:
            current_h = result.x
            current_c5 = np.max(np.correlate(current_h, 1.0 - current_h, mode='full')) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
