    # Define optimization function
    def optimize_single(guess):
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
        result = minimize(
            objective, 
            guess, 
            args=(dx,), 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons, 
            tol=1e-8, 
            options={'maxiter': 200}
        )
        if result.success:
            return (result.fun, result.x)
        else:
            # Fallback to the guess
            return (objective(guess, dx), guess)
    
    # Run optimizations in parallel
    with Pool(processes=min(cpu_count(), num_starts)) as pool:
        results = pool.map(optimize_single, initial_guesses)
    
    # Select the best result
    for c5, h in results:
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    
    return best_h, best_c5, n_points
