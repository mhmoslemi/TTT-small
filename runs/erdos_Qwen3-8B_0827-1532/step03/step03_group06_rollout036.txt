import numpy as np
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    n_points = 100  # Increased resolution for better approximation
    dx = 2.0 / n_points
    initial_h = initial_h_values  # Assume this is a global variable

    # Objective function with normalization to satisfy the integral constraint
    def objective(h):
        # Normalize h to ensure ∫h(x)dx = 1
        h_normalized = h * (n_points / 2) / np.sum(h)
        # Compute cross-correlation
        corr = np.correlate(h_normalized, 1 - h_normalized, mode='full')
        # Return the maximum overlap integral
        return np.max(corr) * dx

    # Bounds for each parameter
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Use a hybrid approach: start with a population-based search and refine
    # Initial guess based on provided initial_h_values
    h0 = initial_h.copy()
    # Run differential evolution with adaptive parameters
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=20,
        mutation=(0.5, 1.0),
        recombination=0.8,
        tol=1e-8,
        maxiter=150,
        seed=seed
    )

    best_h = result.x
    best_c5 = result.fun

    return (best_h, best_c5, n_points)
