    bounds = [(0.0, 1.0) for _ in range(n_points)]
    def objective_de(h, dx):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        penalty = 1e3 * abs(np.sum(h) - n_points / 2)
        return max_corr * dx + penalty
    
    result_de = differential_evolution(objective_de, bounds, args=(dx,), strategy='best1bin', maxiter=100, popsize=15, tol=1e-6, mutation=(0.5, 1), recombination=0.8, disp=False)
    if result_de.success:
        current_c5 = result_de.fun
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_de.x
    
    return best_h, best_c5, n_points
