import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = 128
    dx = 2.0 / n_points
    initial_guesses = []

    np.random.seed(seed)
    # Resample initial_h_values to n_points
    x_initial = np.linspace(0, 2, len(initial_h_values))
    f_initial = np.interp(np.linspace(0, 2, n_points), x_initial, initial_h_values)
    initial_guesses.append(f_initial)
    
    # Add perturbations of the initial guess
    for perturb in [0.01, 0.02, 0.05]:
        perturbed = f_initial.copy()
        perturbed += np.random.normal(0, perturb, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * (n_points / 2) / np.sum(perturbed)
        initial_guesses.append(perturbed)
    
    # Add structured initial guesses
    structured = np.zeros(n_points)
    structured[::2] = 1
    initial_guesses.append(structured)
    structured = np.zeros(n_points)
    structured[::3] = 1
    initial_guesses.append(structured)
    structured = np.zeros(n_points)
    structured[::4] = 1
    initial_guesses.append(structured)
    
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()

    def optimize(initial_h):
        def objective(h):
            corr = correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
        bounds = [(0.0, 1.0) for _ in range(n_points)]

        result = minimize(
            objective,
            initial_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraint,
            options={'maxiter': 1000, 'ftol': 1e-9, 'eps': 1e-9}
        )
        return result.x, result.fun

    for guess in initial_guesses:
        h, c5 = optimize(guess)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
        if time.time() - start_time > budget_s:
            break

    return (best_h, best_c5, n_points)
