import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_c5 = float('inf')

        def objective(h):

            max_corr = np.max(corr)
            penalty = 1e4 * abs(np.sum(h) - target_sum)
            return max_corr * dx + penalty
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Run differential evolution
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=15,
            maxiter=100,
            tol=1e-6,
            mutation=(0.5, 1),

            seed=seed
        )
        
        # Update best solution
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check elapsed time

        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
