    max_corr = np.max(corr)
    return max_corr * dx

    initial_h = initial_h_values.copy()

    def objective_with_penalty(h):
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        penalty = 1e6 * (np.sum(h) - n_points / 2)**2
        return max_corr * dx + penalty
    
    # Differential evolution with multiple runs
    best_c5 = float('inf')
    best_h = initial_h

    for strategy in ['best1bin', 'best2bin', 'current-to-pbest']:
