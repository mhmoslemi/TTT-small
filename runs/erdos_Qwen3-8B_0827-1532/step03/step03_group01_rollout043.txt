import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Reduced for faster computation
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_and_scale(h):
        h_clipped = np.clip(h, 0, 1)
        current_sum = np.sum(h_clipped)
        if current_sum == 0:
            return np.zeros_like(h_clipped)
        scale_factor = target_sum / current_sum
