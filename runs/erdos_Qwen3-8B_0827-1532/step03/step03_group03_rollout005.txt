import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = 100  # Increased for better resolution
    dx = 2.0 / n_points
    np.random.seed(seed)
    
    # Generate diverse initial guesses
    initial_guesses = []
    # 1. Use the provided initial_h_values
    initial_guesses.append(initial_h_values)
    # 2. Random initial guess
    random_initial = np.zeros(n_points)
    indices = np.random.choice(n_points, size=n_points//2, replace=False)
    random_initial[indices] = 1
    initial_guesses.append(random_initial)
    # 3. Structured initial guess (every other interval)
    structured_initial = np.zeros(n_points)
    structured_initial[::2] = 1
    initial_guesses.append(structured_initial)
    # 4. Evenly spaced initial guess
    evenly_spaced_initial = np.zeros(n_points)
    positions = np.linspace(0, 2, n_points//2, endpoint=False)
    for pos in positions:
        idx = int(pos / dx)
        evenly_spaced_initial[idx] = 1
    initial_guesses.append(evenly_spaced_initial)
    # 5. Sine-like initial guess
    theta = np.linspace(0, 2*np.pi, n_points)
    sine_initial = np.zeros(n_points)
    threshold = 0.5
    for i in range(n_points):
        if np.sin(theta[i]) > threshold:
            sine_initial[i] = 1
    initial_guesses.append(sine_initial)
    # 6. Perturbations of initial_h_values
    for _ in range(5):
        perturb = np.random.normal(0, 0.05, n_points)
        perturbed = initial_h_values + perturb
        perturbed = np.clip(perturbed, 0, 1)
        sum_perturbed = np.sum(perturbed)
        if sum_perturbed != 0:
            perturbed = perturbed * (n_points / 2) / sum_perturbed
        initial_guesses.append(perturbed)
    
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    
    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    for guess in initial_guesses:
        # Perturb the guess slightly to avoid local optima
        perturb = np.random.normal(0, 0.01, n_points)
        perturbed_guess = np.clip(guess + perturb, 0, 1)
        sum_perturbed = np.sum(perturbed_guess)
        if sum_perturbed != 0:
            perturbed_guess = perturbed_guess * (n_points / 2) / sum_perturbed
        # Optimize with COBYLA for better exploration
        result = minimize(
            objective,
            perturbed_guess,
            method='COBYLA',
            bounds=bounds,
            constraints=constraint,
            options={'maxiter': 1000, 'disp': False}
        )
        current_h = result.x
        current_c5 = result.fun
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h
        # Check time
        if time.time() - start_time > budget_s:
            break
    
    return (best_h, best_c5, n_points)
