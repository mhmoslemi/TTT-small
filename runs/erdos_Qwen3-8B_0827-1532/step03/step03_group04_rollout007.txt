    max_corr = np.max(corr)
    sum_penalty = (np.sum(h) - target_sum) ** 2 * penalty_coeff

    target_sum = n_points / 2

    result = differential_evolution(
        fun=lambda h: objective(h, dx, target_sum, penalty_coeff),

        tol=1e-6,
