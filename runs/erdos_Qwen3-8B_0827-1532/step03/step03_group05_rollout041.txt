    h_de = result_de.x
    
    # Now refine with SLSQP
    def objective_slsqp(h):
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
    bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]
    
    result_slsqp = minimize(
        objective_slsqp,
        h_de,
        method='SLSQP',
        bounds=bounds_slsqp,
        constraints=constraint,
        options={'maxiter': 500, 'ftol': 1e-9, 'eps': 1e-9}
    )
    
    best_h = result_slsqp.x
    best_c5 = result_slsqp.fun
    
    # Check if time is exceeded
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return (best_h, best_c5, n_points)
    
    return (best_h, best_c5, n_points)
