from scipy.optimize import minimize, differential_evolution
from scipy.signal import correlate

def objective(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=100, **kwargs):

    dx = 2.0 / n_points

    # Generate symmetric initial guess
    half = n_points // 2
    h_initial = np.random.uniform(0, 1, half)
    h_initial = np.concatenate([h_initial, h_initial[::-1]])
    current_sum = np.sum(h_initial) * dx
    scale = (n_points / 2) / current_sum
    h_initial = h_initial * scale
    
    # Function to evaluate C5
    def evaluate(h):
        return objective(h, dx)
    
    # Define bounds and constraints
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
    
    # Optimization parameters
    num_starts = 30  # Increased number of starts
    best_c5 = float('inf')

        result = minimize(
