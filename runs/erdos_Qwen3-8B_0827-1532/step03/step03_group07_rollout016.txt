from scipy.optimize import minimize

def objective(first_half, dx, n_points):
    h = np.zeros(n_points)
    h[:n_points//2] = first_half
    h[n_points//2:] = np.flipud(h[:n_points//2])
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=100, **kwargs):
