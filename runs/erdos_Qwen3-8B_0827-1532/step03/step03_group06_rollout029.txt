import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    n_points = 100
    dx = 2.0 / n_points
    initial_h = initial_h_values  # Assume this is a global variable
    
    # Generate diverse initial guesses
    def generate_perturbed_initial_guess(h_base):
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        h_perturbed = h_base + perturbation
        h_perturbed = np.clip(h_perturbed, 0, 1)
        sum_h = np.sum(h_perturbed)
        scaling = (n_points / 2) / sum_h
        h_perturbed *= scaling
        return h_perturbed
    
    def generate_binary_initial_guess(n_points):
        indices = np.random.choice(n_points, size=n_points//2, replace=False)
        h = np.zeros(n_points)
        h[indices] = 1.0
        return h
    
    def generate_random_initial_guess(n_points):
        h = np.random.uniform(0, 1, size=n_points)
        sum_h = np.sum(h)
        scaling = (n_points / 2) / sum_h
        h *= scaling
        return h
    
    initial_guesses = []
    initial_guesses.append(initial_h.copy())
    initial_guesses.append(generate_binary_initial_guess(n_points))
    initial_guesses.append(generate_random_initial_guess(n_points))
    for _ in range(3):
        initial_guesses.append(generate_perturbed_initial_guess(initial_h))
    
    # Objective function
    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint: sum(h) = n_points / 2
    constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = None
    best_c5 = float('inf')
    
    for h0 in initial_guesses:
        # Run optimization with more iterations
        result = minimize(
            objective,
            h0,
            method='SLSQP',
            bounds=bounds,
            constraints=constraint,
            options={'maxiter': 500, 'ftol': 1e-8, 'eps': 1e-8}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return (best_h, best_c5, n_points)
