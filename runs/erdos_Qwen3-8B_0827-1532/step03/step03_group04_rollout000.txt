            h_perturbed, 
            args=(dx,), 
            method='L-BFGS-B',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints=cons,
            options={'maxiter': 500, 'ftol': 1e-9, 'gtol': 1e-9}
        )
        # Check result
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            # Fallback to perturbed guess
            current_c5 = objective(h_perturbed, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_perturbed
    return best_h, best_c5, n_points
