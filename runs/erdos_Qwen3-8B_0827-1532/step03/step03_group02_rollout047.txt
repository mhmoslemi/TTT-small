        np.concatenate([np.ones(n_points//2), np.zeros(n_points//2)]),  # first half 1
        np.concatenate([np.ones(n_points//3), np.zeros(n_points//3), np.ones(n_points//3)]),  # first and last third
        np.concatenate([np.ones(n_points//4), np.zeros(n_points//4), np.ones(n_points//4), np.zeros(n_points//4)]),  # first and third quarter
    ]
    
    # Add random binary guess with exactly n_points/2 ones
    np.random.seed(seed)
    indices = np.random.choice(n_points, size=n_points//2, replace=False)
    initial_guesses.append(np.zeros(n_points))
    initial_guesses[-1][indices] = 1
    
    num_starts = 50  # Increased number of starts
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    
    for i in range(num_starts):
        np.random.seed(seed + i)
        # Randomly select an initial guess
        initial_idx = np.random.randint(len(initial_guesses))
        h_perturbed = initial_guesses[initial_idx].copy()
        
        # Add small random perturbation
        perturbation = np.random.normal(0, 0.05, n_points)
        h_perturbed += perturbation
        h_perturbed = np.clip(h_perturbed, 0, 1)
        
        # Adjust to meet the sum constraint
        current_sum = np.sum(h_perturbed)
        scale = (n_points / 2) / current_sum
        h_perturbed = h_perturbed * scale
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
        
        # Optimize with tighter tolerances and more iterations
        result = minimize(
            objective, 
            h_perturbed, 
            args=(dx,), 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 500, 'ftol': 1e-9, 'xtol': 1e-9}
        )
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            # Fallback to perturbed guess if optimization fails
            current_c5 = objective(h_perturbed, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_perturbed
    
    return best_h, best_c5, n_points
