                method='SLSQP',
                bounds=bounds,
                constraints=constraint,
                options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8}
            )
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    return (best_h, best_c5, n_points)
