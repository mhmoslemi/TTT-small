def run(seed=42, budget_s=100, **kwargs):

    dx = 2.0 / n_points

    def generate_structured_initial_guess(n_points, m=2):
        l = 1.0 / m

        for i in range(m):
            start_idx = int(start / dx)
            end_idx = int((start + l) / dx)
            end_idx = min(end_idx, n_points)
            h[start_idx:end_idx] = 1.0
            start += l + spacing
        sum_h = np.sum(h)
        scaling = (n_points / 2) / sum_h
        h *= scaling
        return h
    
    def generate_binary_initial_guess(n_points):
        indices = np.random.choice(n_points, size=n_points//2, replace=False)
        h = np.zeros(n_points)
        h[indices] = 1.0
        return h
    
    initial_guesses = []
    initial_guesses.append(initial_h.copy())
    initial_guesses.append(generate_binary_initial_guess(n_points))

    initial_guesses.append(generate_structured_initial_guess(n_points, m=2))
    initial_guesses.append(generate_structured_initial_guess(n_points, m=3))
    
    # Objective function
    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint: sum(h) = n_points / 2
    constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = None
    best_c5 = float('inf')
    elapsed_time = 0.0
    
    for h0 in initial_guesses:
        # Early exit if time is up
