                best_h = result.x
        else:
            # Fallback to SLSQP
            result = minimize(
                objective,
                h0,
                method='SLSQP',
                bounds=[(0.0, 1.0) for _ in range(n_points)],
                constraints={'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2},
                options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
    
    return (best_h, best_c5, n_points)
