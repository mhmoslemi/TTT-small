import numpy as np
from scipy.optimize import differential_evolution

def objective(h, dx, n_points):
    sum_h = np.sum(h)
    if sum_h == 0:
        return float('inf')
    h_normalized = h * (n_points / 2) / sum_h
    corr = np.correlate(h_normalized, 1 - h_normalized, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=100, **kwargs):
    n_points = 100  # Increased for better resolution
    dx = 2.0 / n_points
    
    # Interpolate initial_h_values to n_points
    initial_h = initial_h_values.copy()
    initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h)
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)  # Ensure sum is correct
    
    # Define bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Run differential evolution with multiple strategies and seeds
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    
    # Run multiple differential evolution instances with different seeds
    for i in range(3):  # 3 independent runs
        np.random.seed(seed + i)
        result = differential_evolution(
            objective,
            bounds,
            args=(dx, n_points),
            strategy='best1bin',
            popsize=15,
            mutation=(0.5, 1),
            recombination=0.9,
            tol=1e-8,
            maxiter=100,
            x0=initial_h,
            seed=seed + i
        )
        current_c5 = result.fun
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x
    
    return best_h, best_c5, n_points
