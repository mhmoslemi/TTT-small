import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds

def objective(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=100, **kwargs):
    n_points = 80
    dx = 2.0 / n_points
    num_starts = 50  # Increased number of starts for better exploration
    best_c5 = float('inf')
    best_h = None
    
    for i in range(num_starts):
        np.random.seed(seed + i)
        # Generate random initial guess with sum constraint
        h_perturbed = np.random.uniform(0, 1, n_points)
        current_sum = np.sum(h_perturbed)
        scale = (n_points / 2) / current_sum
        h_perturbed = h_perturbed * scale
        
        bounds = Bounds([0.0]*n_points, [1.0]*n_points)
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
        
        # Try different solvers for better convergence
        if i % 2 == 0:
            method = 'L-BFGS-B'
        else:
            method = 'SLSQP'
        
        result = minimize(
            objective, 
            h_perturbed, 
            args=(dx,), 
            method=method, 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 1000, 'ftol': 1e-9, 'xtol': 1e-9}
        )
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            # Fallback to perturbed guess if optimization fails
            current_c5 = objective(h_perturbed, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_perturbed
    
    return best_h, best_c5, n_points
