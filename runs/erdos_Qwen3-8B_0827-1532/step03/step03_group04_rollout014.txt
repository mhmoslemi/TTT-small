    current_sum = np.sum(h_initial)
    scale = (n_points / 2) / current_sum
    h_initial = h_initial * scale
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
    
    result = minimize(
        objective, 
        h_initial, 
        args=(dx,), 
        method='SLSQP', 
        bounds=bounds, 
        constraints=cons,
        options={'maxiter': 500, 'ftol': 1e-9, 'xtol': 1e-9}
    )
    if result.success:
        current_c5 = result.fun
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x
    else:
        current_c5 = objective(h_initial, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_initial
    
    # Generate random starts
    for i in range(num_starts - 1):
        h_initial = np.random.rand(n_points)
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        result = minimize(
            objective, 
            h_initial, 
            args=(dx,), 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 500, 'ftol': 1e-9, 'xtol': 1e-9}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            current_c5 = objective(h_initial, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_initial
    
    return best_h, best_c5, n_points
