from scipy.signal import correlate

    n_points = 80
    dx = 2.0 / n_points
    initial_h = initial_h_values  # Assume this is a global variable
    
    # Generate diverse initial guesses
    def generate_sine_initial_guess(n_points):
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(np.pi * x))
        return h
    
    def generate_binary_initial_guess(n_points):
        indices = np.random.choice(n_points, size=n_points//2, replace=False)
        h = np.zeros(n_points)
        h[indices] = 1.0
        return h
    
    initial_guesses = []
    initial_guesses.append(initial_h.copy())
    initial_guesses.append(generate_binary_initial_guess(n_points))
    initial_guesses.append(generate_sine_initial_guess(n_points))
    
    # Add some perturbations of the initial_h
    for _ in range(3):
        perturbation = np.random.normal(0, 0.05, size=n_points)
        h_perturbed = initial_h + perturbation
        h_perturbed = np.clip(h_perturbed, 0, 1)
        sum_h = np.sum(h_perturbed)
        scaling = (n_points / 2) / sum_h
        h_perturbed *= scaling
        initial_guesses.append(h_perturbed)
    
    # Objective function
    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint: sum(h) = n_points / 2
    constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = None
    best_c5 = float('inf')
    
    for h0 in initial_guesses:
        # Use L-BFGS-B method with more iterations
        result = minimize(
            objective,
            h0,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraint,
            options={'maxiter': 500, 'ftol': 1e-8, 'eps': 1e-8}
        )
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    return (best_h, best_c5, n_points)
