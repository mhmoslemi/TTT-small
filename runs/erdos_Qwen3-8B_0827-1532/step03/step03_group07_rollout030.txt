    h = cp.Variable(n_points)
    t = cp.Variable()

    # Constraints
    constraints = [

        cp.Minimize(-t)  # Maximize t
    ]

    # Add constraints for each k
    for m in range(n_points):
        shifted_indices = (np.arange(n_points) + m) % n_points
        integral = dx * cp.sum(h * h[shifted_indices])
        constraints.append(integral >= t)

    # Add bounds for h
    for i in range(n_points):
        constraints.append(h[i] >= 0)
        constraints.append(h[i] <= 1)

    # Set initial guess
    h.value = initial_h

    # Solve the problem
    try:
        prob = cp.Problem(cp.Maximize(t), constraints)
        prob.solve(solver=cp.ECOS, verbose=False)
    except:
        # Fallback to initial guess if solver fails
        prob = cp.Problem(cp.Maximize(t), constraints)

    # Extract the solution
    h_values = h.value
    c5_bound = 1 - prob.objective.value
