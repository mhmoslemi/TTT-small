def objective(h, dx, n_points):
    max_integral = 0.0
    for m in range(n_points):
        valid_i = np.arange(n_points - m)
        integral = np.sum(h[valid_i] * (1 - h[valid_i + m])) * dx
        if integral > max_integral:
            max_integral = integral
    return max_integral
