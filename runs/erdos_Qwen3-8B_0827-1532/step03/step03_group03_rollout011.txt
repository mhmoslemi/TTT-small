from scipy.optimize import minimize
from scipy.signal import correlate

    initial_guesses.append(random_initial)

    structured_initial[::2] = 1
    initial_guesses.append(structured_initial)

    evenly_spaced_initial = np.zeros(n_points)
    for i in range(n_points//2):
        pos = int(i * n_points / (n_points//2))
        evenly_spaced_initial[pos] = 1
    initial_guesses.append(evenly_spaced_initial)
    # 5. Perturbations of initial_h_values
    for _ in range(5):
        perturb = np.random.normal(0, 0.05, n_points)
        perturbed = initial_h_values + perturb

        sum_perturbed = np.sum(perturbed)
        if sum_perturbed != 0:
            perturbed = perturbed * (n_points / 2) / sum_perturbed
        initial_guesses.append(perturbed)
    # 6. Spread-out initial guess (alternating 1s and 0s)
    spread_initial = np.zeros(n_points)
    spread_initial[::3] = 1  # Spacing of 3 samples
    sum_spread = np.sum(spread_initial)
    if sum_spread != 0:
        spread_initial = spread_initial * (n_points / 2) / sum_spread
    initial_guesses.append(spread_initial)
    # 7. Randomly spaced initial guess with lower density
    low_density_initial = np.zeros(n_points)
    indices = np.random.choice(n_points, size=n_points//4, replace=False)
    low_density_initial[indices] = 1
    sum_low = np.sum(low_density_initial)
    if sum_low != 0:
        low_density_initial = low_density_initial * (n_points / 2) / sum_low
    initial_guesses.append(low_density_initial)
    
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    
    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    for guess in initial_guesses:
        # Perturb the guess slightly to avoid local optima
        perturb = np.random.normal(0, 0.01, n_points)
