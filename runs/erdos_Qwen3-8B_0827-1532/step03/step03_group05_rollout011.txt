from scipy.optimize import minimize
from scipy.signal import correlate
import time

    dx = 2.0 / n_points
    initial_guesses = []

    np.random.seed(seed)

    x_initial = np.linspace(0, 2, 128)
    resampled_initial = np.interp(x_initial, np.linspace(0, 2, len(initial_h_values)), initial_h_values)
    initial_guesses.append(resampled_initial)
    
    # Random initial guess
    random_initial = np.zeros(n_points)
    indices = np.random.choice(n_points, size=n_points//2, replace=False)
    random_initial[indices] = 1
    initial_guesses.append(random_initial)
    
    # Structured initial guess (every other interval)
    structured_initial = np.zeros(n_points)
    structured_initial[::2] = 1
    initial_guesses.append(structured_initial)
    
    # Evenly spaced initial guess
    evenly_spaced_initial = np.zeros(n_points)
    for i in range(n_points//2):
        pos = int(i * n_points / (n_points//2))
        evenly_spaced_initial[pos] = 1
    initial_guesses.append(evenly_spaced_initial)
    
    # Perturbed versions of initial_h_values
    for noise_level in [0.05, 0.1, 0.15]:
        perturbed_initial = resampled_initial.copy()
        perturbed_initial += np.random.normal(0, noise_level, size=n_points)
        perturbed_initial = np.clip(perturbed_initial, 0, 1)
        perturbed_initial = perturbed_initial * (n_points / 2) / np.sum(perturbed_initial)
        initial_guesses.append(perturbed_initial)
    
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()

    def optimize(initial_h):
        def objective(h):
            corr = correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx

        constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
        bounds = [(0.0, 1.0) for _ in range(n_points)]

        result = minimize(
            objective,
            initial_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraint,
            options={'maxiter': 1000, 'ftol': 1e-9, 'eps': 1e-9}
