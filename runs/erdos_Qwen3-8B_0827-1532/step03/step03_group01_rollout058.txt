import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Smaller n_points for faster evaluation
    dx = 2.0 / n_points
    
    # Generate diverse initial guesses
    initial_guesses = []
    # Use provided initial_h_values as base
    initial_guesses.append(initial_h_values)
    # Add random guesses
    for _ in range(5):
        h_rand = np.random.uniform(0, 1, n_points)
        h_rand = h_rand / np.sum(h_rand) * (n_points / 2)
        initial_guesses.append(h_rand)
    # Add perturbations of initial_h_values
    for _ in range(5):
        noise = np.random.normal(0, 0.01, n_points)
        perturbed = initial_h_values + noise
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed / np.sum(perturbed) * (n_points / 2)
        initial_guesses.append(perturbed)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraints: sum(h) == n_points / 2
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    for guess in initial_guesses:
        # Run optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 300}
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Fallback to initial guess if no improvement
    if best_h is None:
        result = minimize(
            objective,
            initial_guesses[0],
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 300}
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points
