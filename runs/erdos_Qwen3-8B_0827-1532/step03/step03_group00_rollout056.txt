        # Constraints
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Optimization
        result = minimize(
            objective,
            scaled_guess,
            method='COBYLA',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 300}
        )
        
        # Update best solution
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
