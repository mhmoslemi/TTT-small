import numpy as np
from scipy.optimize import minimize

def objective(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=100, **kwargs):
    n_points = 120
    dx = 2.0 / n_points
    # Construct initial guess with three intervals of length 1/3
    initial_h = np.zeros(n_points)
    # First interval [0, 1/3]
    start = int(0.0 / dx)
    end = int((1/3) / dx)
    initial_h[start:end+1] = 1.0
    # Second interval [2/3, 1]
    start = int((2/3) / dx)
    end = int(1.0 / dx)
    initial_h[start:end+1] = 1.0
    # Third interval [4/3, 2]
    start = int((4/3) / dx)
    end = int(2.0 / dx)
    initial_h[start:end+1] = 1.0
    
    num_starts = 20
    best_c5 = float('inf')
    best_h = initial_h.copy()
    
    for i in range(num_starts):
        np.random.seed(seed + i)
        # Perturb the initial guess with uniform noise
        h_perturbed = initial_h + np.random.uniform(-0.1, 0.1, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        # Adjust to meet the sum constraint
        current_sum = np.sum(h_perturbed)
        scale = (n_points / 2) / current_sum
        h_perturbed = h_perturbed * scale
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
        
        # Optimize with more iterations and tighter tolerances
        result = minimize(
            objective, 
            h_perturbed, 
            args=(dx,), 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 500, 'ftol': 1e-9, 'xtol': 1e-9}
        )
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            # Fallback to perturbed guess if optimization fails
            current_c5 = objective(h_perturbed, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_perturbed
    
    return best_h, best_c5, n_points
