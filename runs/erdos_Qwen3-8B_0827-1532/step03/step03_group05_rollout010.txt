    for i in range(n_points // 3):
        pos = int(i * 3)
        every_third_initial[pos] = 1
    scale = (n_points / 2) / np.sum(every_third_initial)
    every_third_initial = every_third_initial * scale
    initial_guesses.append(every_third_initial)
    
    # 8. Random perturbation of sinusoidal
    perturbed_sinusoidal = sinusoidal_initial.copy()
    perturbed_sinusoidal += np.random.normal(0, 0.05, size=n_points)
    perturbed_sinusoidal = np.clip(perturbed_sinusoidal, 0, 1)
    perturbed_sinusoidal = perturbed_sinusoidal * (n_points / 2) / np.sum(perturbed_sinusoidal)
    initial_guesses.append(perturbed_sinusoidal)
    
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    
    def optimize(initial_h):
        def objective(h):
            corr = correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        result = minimize(
            objective,
            initial_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraint,
            options={'maxiter': 1000, 'ftol': 1e-8, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Parallel processing
    num_cpus = min(cpu_count(), len(initial_guesses))
    with Pool(num_cpus) as pool:
        results = pool.map(lambda guess: optimize(guess), initial_guesses)
    
    for h, c5 in results:
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
        if time.time() - start_time > budget_s:
            break
    
    return (best_h, best_c5, n_points)
