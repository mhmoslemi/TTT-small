from scipy.optimize import minimize
import time

    n_points = 200
    dx = 2.0 / n_points

    initial_guess = initial_guess / np.sum(initial_guess) * (n_points / 2)
    
    # Generate multiple initial guesses by perturbing the initial_guess
    num_initial_guesses = 10
    initial_guesses = []
    for _ in range(num_initial_guesses):
        # Add random noise
        noise = np.random.uniform(-0.05, 0.05, n_points)
        perturbed = initial_guess + noise
        # Clip to [0, 1]
        perturbed = np.clip(perturbed, 0, 1)
        # Adjust sum to meet the constraint
        scale_factor = (n_points / 2) / np.sum(perturbed)
        perturbed *= scale_factor
        initial_guesses.append(perturbed)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraints: sum(h) == n_points / 2
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    for guess in initial_guesses:
        # Run optimization
        result = minimize(
            objective,
            guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 300}
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
