import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Smaller for faster evaluation
    dx = 2.0 / n_points
    sum_h = n_points / 2  # Integral constraint
    
    # Generate initial guesses: 5 perturbed and 5 random
    initial_guesses = []
    # Perturb initial_h_values (assuming it's length 100)
    initial_guess = initial_h_values.copy()
    for _ in range(5):
        noise = np.random.normal(0, 0.02, n_points)
        perturbed = initial_guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        scale_factor = sum_h / np.sum(perturbed)
        perturbed *= scale_factor
        initial_guesses.append(perturbed)
    # Generate random initial guesses
    for _ in range(5):
        h = np.random.uniform(0, 1, n_points)
        scale_factor = sum_h / np.sum(h)
        h *= scale_factor
        initial_guesses.append(h)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraints: sum(h) == sum_h
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - sum_h})
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    
    for guess in initial_guesses:
        # Run optimization with COBYLA for bounds handling
        result = minimize(
            objective,
            guess,
            method='COBYLA',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'disp': False}
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Fallback to initial guess if no improvement
    if best_h is None:
        result = minimize(
            objective,
            initial_guess,
            method='COBYLA',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'disp': False}
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points
