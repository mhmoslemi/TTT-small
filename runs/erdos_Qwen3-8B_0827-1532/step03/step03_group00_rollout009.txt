import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate multiple initial guesses
    num_initial_guesses = 5
    for i in range(num_initial_guesses):
        # Random initial guess with sum constraint
        h = np.random.rand(n_points)
        h = h * (target_sum / np.sum(h))  # Scale to meet sum constraint
        h = np.clip(h, 0, 1)
        
        # Objective function with penalty for sum constraint
        def objective(h_flat):
            h = h_flat.reshape(n_points)
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            penalty = 1e3 * (np.sum(h) - target_sum)**2
            return max_corr * dx + penalty
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Differential evolution for global optimization
        result_de = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=50,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed + i
        )
        
        # Refine with SLSQP
        def objective_slsqp(h_flat):
            h = h_flat.reshape(n_points)
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        result_slsqp = minimize(
            objective_slsqp,
            result_de.x,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 100}
        )
        
        # Update best solution
        if result_slsqp.success and result_slsqp.fun < best_c5:
            best_c5 = result_slsqp.fun
            best_h = result_slsqp.x
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
