        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
        
        # Optimize with more iterations and tighter tolerances
        result = minimize(
            objective, 
            h, 
            args=(dx,), 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 300, 'ftol': 1e-9, 'xtol': 1e-9}
        )
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            # Fallback to the initial guess
            current_c5 = objective(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h
    
    return best_h, best_c5, n_points
