    random_pattern = random_pattern * (n_points / 2) / np.sum(random_pattern)
    initial_guesses.append(random_pattern)
    
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    start_time = time.time()
    
    for guess in initial_guesses:
        # Scale the initial guess to meet the sum constraint
        scaled_guess = guess * (target_sum / np.sum(guess))
        # Define the objective function
        def objective(h):
            # Avoid division by zero
            sum_h = np.sum(h)
            if sum_h == 0:
                return float('inf')
            scaled_h = h * (target_sum / (sum_h + 1e-12))
            one_minus_h = 1.0 - scaled_h
            corr = np.correlate(scaled_h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Perform differential evolution optimization
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=100,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed
        )
        
        # Check if this result is better
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
