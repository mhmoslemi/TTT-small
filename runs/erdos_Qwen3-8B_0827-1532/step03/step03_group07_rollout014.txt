    for i in range(num_starts):
        np.random.seed(seed + i)
        if i == 0:
            # Start from the initial guess with larger perturbation
            h_perturbed = initial_h + np.random.normal(0, 0.05, n_points)
        else:
            # Random initialization with constraint
            h_perturbed = np.random.rand(n_points)
            h_perturbed = h_perturbed / np.sum(h_perturbed) * (n_points / 2)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        
        # Define bounds and constraints
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
        
        # Run optimization with tighter tolerances and more iterations
        result = minimize(
            objective, 
            h_perturbed, 
            args=(dx,), 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons, 
            tol=1e-6, 
            options={'maxiter': 200}
        )
        
        # Check if optimization was successful
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            # Fallback to perturbed guess if optimization fails
            current_c5 = objective(h_perturbed, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_perturbed
    
    return best_h, best_c5, n_points
