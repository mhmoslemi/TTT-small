import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Use n_points=100 for faster computation and better exploration
    n_points = 100
    dx = 2.0 / n_points
    
    # Generate initial guess based on initial_h_values (assumed to be for n=100)
    initial_guess = initial_h_values.copy()
    # Adjust sum to meet the constraint
    initial_guess = initial_guess / np.sum(initial_guess) * (n_points / 2)
    
    # Generate diverse initial guesses with different perturbations
    num_initial_guesses = 15
    initial_guesses = []
    for _ in range(num_initial_guesses):
        # Add random noise with varying magnitudes
        noise = np.random.normal(0, 0.05, n_points)
        perturbed = initial_guess + noise
        # Clip to [0, 1]
        perturbed = np.clip(perturbed, 0, 1)
        # Adjust sum to meet the constraint
        scale_factor = (n_points / 2) / np.sum(perturbed)
        perturbed *= scale_factor
        initial_guesses.append(perturbed)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraints: sum(h) == n_points / 2
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    for guess in initial_guesses:
        # Run optimization with tighter tolerances and more iterations
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 1000}
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Fallback to initial guess if no improvement found
    if best_h is None:
        result = minimize(
            objective,
            initial_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 1000}
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points
