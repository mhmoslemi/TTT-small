        h_initial = np.random.uniform(0, 1, n_points)
        current_sum = np.sum(h_initial)
        scale = (n_points / 2) / current_sum
        h_initial = h_initial * scale
        
        # Run differential evolution
        result = differential_evolution(
            fun=lambda h: objective(h, dx, n_points),
            bounds=bounds,
            strategy='best1bin',
            popsize=15,
            maxiter=100,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed + run_idx
        )
        
        # Check if the result is successful
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            # Fallback to the initial guess
            current_c5 = objective(h_initial, dx, n_points)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_initial
    
    return best_h, best_c5, n_points
