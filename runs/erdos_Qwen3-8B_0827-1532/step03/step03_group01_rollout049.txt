import time

    n_points = 100

    initial_guesses = []

    random_guess = np.random.rand(n_points)

    random_guess2 = random_guess2 / np.sum(random_guess2) * (n_points / 2)
    initial_guesses.append(random_guess2)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)

    bounds = [(0.0, 1.0) for _ in range(n_points)]

        )
