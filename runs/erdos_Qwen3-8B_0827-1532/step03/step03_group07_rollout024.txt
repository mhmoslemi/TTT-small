    initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h)
    initial_sum = np.sum(initial_h)
    initial_h = initial_h * (n_points / 2) / initial_sum
    
    # Set penalty coefficient
    penalty_coeff = 1e6
    
    # Bounds for differential evolution
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Run differential evolution
    result = differential_evolution(
        fun=lambda h: penalty_objective(h, dx, n_points, penalty_coeff),
        bounds=bounds,
        strategy='best1bin',
        popsize=15,
        maxiter=100,
        tol=1e-8,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed
    )
    
    best_h = result.x
    best_c5 = objective(best_h, dx, n_points)
    
    return best_h, best_c5, n_points
