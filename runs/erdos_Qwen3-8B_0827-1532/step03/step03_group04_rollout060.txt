import numpy as np
from scipy.optimize import minimize

def objective(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=100, **kwargs):
    n_points = 80
