import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = 64  # Smaller n_points for faster evaluation
    dx = 2.0 / n_points
    initial_guesses = []

    np.random.seed(seed)
    # Resample initial_h_values to n_points=64
    x_initial = np.linspace(0, 2, 128)
    f_initial = np.interp(np.linspace(0, 2, n_points), x_initial, initial_h_values)
    initial_guesses.append(f_initial)
    
    # Random initial guess
    random_initial = np.zeros(n_points)
    indices = np.random.choice(n_points, size=n_points//2, replace=False)
    random_initial[indices] = 1
    initial_guesses.append(random_initial)
    
    # Structured initial guess (every other interval)
    structured_initial = np.zeros(n_points)
    structured_initial[::2] = 1
    initial_guesses.append(structured_initial)
    
    # Evenly spaced initial guess
    evenly_spaced_initial = np.zeros(n_points)
    for i in range(n_points//2):
        pos = int(i * n_points / (n_points//2))
        evenly_spaced_initial[pos] = 1
    initial_guesses.append(evenly_spaced_initial)
    
    # Perturbed versions of initial_h_values
    perturbed_initial = f_initial.copy()
    perturbed_initial += np.random.normal(0, 0.05, size=n_points)
    perturbed_initial = np.clip(perturbed_initial, 0, 1)
    perturbed_initial = perturbed_initial * (n_points / 2) / np.sum(perturbed_initial)
    initial_guesses.append(perturbed_initial)

    best_c5 = float('inf')
    best_h = None
    start_time = time.time()

    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        penalty = 1e4 * abs(np.sum(h) - n_points / 2)
        return max_corr * dx + penalty

    for guess in initial_guesses:
        # Use differential evolution with the guess as initial population
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = differential_evolution(
            objective,
            bounds,
            popsize=15,
            strategy='best1bin',
            maxiter=100,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed,
            args=(),
            polish=True,
            disp=False
        )
        h_opt = result.x
        c5 = objective(h_opt) - 1e4 * abs(np.sum(h_opt) - n_points / 2)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt
        # Check if time is exceeded
        if time.time() - start_time > budget_s:
            break

    return (best_h, best_c5, n_points)
