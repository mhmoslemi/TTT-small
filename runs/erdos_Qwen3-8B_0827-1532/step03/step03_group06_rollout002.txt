import numpy as np
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    n_points = 60  # Reduced points for faster optimization
    dx = 2.0 / n_points
    initial_h = initial_h_values  # Assume this is a global variable
    
    # Define the objective function
    def objective(h_candidate):
        h = h_candidate.copy()
        sum_h = np.sum(h)
        # Avoid division by zero by adding a small epsilon
        scaling = (n_points / 2) / (sum_h + 1e-12)
        h *= scaling

    # Bounds for each h[i] are [0, 1]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Use differential evolution with adaptive parameters
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=20,
        tol=1e-8,
        mutation=(0.5, 1),
        recombination=0.8,
        maxiter=150,
        seed=seed
    )

    best_c5 = result.fun
    
    return (best_h, best_c5, n_points)
