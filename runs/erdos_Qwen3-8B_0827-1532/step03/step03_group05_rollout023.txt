from scipy.interpolate import interp1d

def run(seed=42, budget_s=100, **kwargs):
    n_points = 256  # Increased from 128
    dx = 2.0 / n_points
    initial_guesses = []

    np.random.seed(seed)
    # 1. Use the provided initial_h_values (resampled to n_points=256)
    x_initial = np.linspace(0, 2, 80)
    f_initial = interp1d(x_initial, initial_h_values, kind='linear', fill_value='extrapolate')
    x_new = np.linspace(0, 2, n_points)
    resampled_initial = f_initial(x_new)
    initial_guesses.append(resampled_initial)
    # 2. Random initial guess
    random_initial = np.zeros(n_points)
    indices = np.random.choice(n_points, size=n_points//2, replace=False)
    random_initial[indices] = 1
    initial_guesses.append(random_initial)
    # 3. Structured initial guess (every other interval)
    structured_initial = np.zeros(n_points)
    structured_initial[::2] = 1
    initial_guesses.append(structured_initial)
    # 4. Evenly spaced initial guess
    evenly_spaced_initial = np.zeros(n_points)
    for i in range(n_points//2):
        pos = int(i * n_points / (n_points//2))
        evenly_spaced_initial[pos] = 1
    initial_guesses.append(evenly_spaced_initial)

    best_c5 = float('inf')
    best_h = None
    start_time = time.time()

    def optimize(initial_h):
        def objective(h):
            corr = correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx

        constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
        bounds = [(0.0, 1.0) for _ in range(n_points)]

        result = minimize(
            objective,
            initial_h,
            method='trust-constr',
            bounds=bounds,
            constraints=constraint,
            options={'maxiter': 1000, 'ftol': 1e-9, 'gtol': 1e-9, 'eps': 1e-9}
        )
        return result.x, result.fun

    for guess in initial_guesses:
        h, c5 = optimize(guess)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
        # Check if time is exceeded

            break

    return (best_h, best_c5, n_points)
