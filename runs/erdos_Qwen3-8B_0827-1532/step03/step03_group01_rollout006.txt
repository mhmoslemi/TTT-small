    h = np.zeros(n_points)

    end = int( (1/3) / dx )
    h[start:end+1] = 1.0
    # Second interval: [2/3, 1]
    start = int( (2/3) / dx )

    start = int( (4/3) / dx )

    # Adjust sum to be exactly n_points / 2
    total = np.sum(h)
    scale = (n_points / 2) / total

import numpy as np

def run(seed=42, budget_s=100, **kwargs):

    def create_structured_initial_guess(n_points):
        dx = 2.0 / n_points
        h = np.zeros(n_points)
        # First interval: [0, 1/3]
        start = 0
        end = int((1/3) / dx)
        h[start:end+1] = 1.0
        # Second interval: [2/3, 1]
        start = int((2/3) / dx)
        end = int(1.0 / dx)
        h[start:end+1] = 1.0
        # Third interval: [4/3, 5/3]
        start = int((4/3) / dx)
        end = int((5/3) / dx)
        h[start:end+1] = 1.0
        # Adjust sum to be exactly n_points / 2
        total = np.sum(h)
        scale = (n_points / 2) / total
        h *= scale
        return h
    
    structured_guess = create_structured_initial_guess(n_points)
    
    # Generate multiple initial guesses: structured and random
    num_initial_guesses = 5
    initial_guesses = [structured_guess]
    for _ in range(num_initial_guesses - 1):
        # Add random noise
        noise = np.random.normal(0, 0.01, n_points)
        perturbed = structured_guess + noise

        perturbed *= scale_factor

        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
