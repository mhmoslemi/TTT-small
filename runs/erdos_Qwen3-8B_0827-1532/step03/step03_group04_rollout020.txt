from scipy.optimize import minimize, differential_evolution
from scipy.signal import correlate

def objective(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=100, **kwargs):
    n_points = 80
    dx = 2.0 / n_points
    np.random.seed(seed)
    
    num_starts = 50
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    
    for _ in range(num_starts):
        # Generate random initial guess
        h_rand = np.random.uniform(0, 1, n_points)
        current_sum = np.sum(h_rand)
        scale = (n_points / 2) / current_sum
        h_perturbed = h_rand * scale
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
        
        result = minimize(
            objective,

            args=(dx,), 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,

        )
