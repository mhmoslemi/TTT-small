import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    # Use the provided initial guess
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate multiple random initial guesses
    num_initial_guesses = 5
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    for i in range(num_initial_guesses):
        # Generate random initial guess
        h = np.random.uniform(0, 1, n_points)
        # Scale to have correct sum
        current_sum = np.sum(h)
        scale = target_sum / current_sum
        h = h * scale
        # Clip to [0,1] in case of numerical issues
        h = np.clip(h, 0, 1)
        
        # Objective function with penalty for sum constraint
        def objective(h_vec):
            one_minus_h = 1.0 - h_vec
            corr = np.correlate(h_vec, one_minus_h, mode='full')
            max_corr = np.max(corr)
            penalty = 1e6 * (np.sum(h_vec) - target_sum)**2
            return max_corr * dx + penalty
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Perform differential evolution optimization
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=100,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed + i
        )
        
        # Check if this result is better
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
