    indices = np.random.choice(n_points, size=n_points//2, replace=False)
    random_initial[indices] = 1
    initial_guesses.append(random_initial)
    
    # 3. Structured initial guess (alternating)
    structured_initial = np.zeros(n_points)
    structured_initial[::2] = 1
    initial_guesses.append(structured_initial)
    
    # 4. Evenly spaced initial guess
    evenly_spaced_initial = np.zeros(n_points)
    for i in range(n_points//2):
        pos = int(i * n_points / (n_points//2))
        evenly_spaced_initial[pos] = 1
    initial_guesses.append(evenly_spaced_initial)
    
    # 5. Perturbed initial_h_values
    perturbed_initial = resampled_initial.copy()
    perturbed_initial += np.random.normal(0, 0.05, size=n_points)
    perturbed_initial = np.clip(perturbed_initial, 0, 1)
    perturbed_initial = perturbed_initial * (n_points / 2) / np.sum(perturbed_initial)
    initial_guesses.append(perturbed_initial)
    
    # 6. Randomly perturbed with different noise
    perturbed_initial_2 = resampled_initial.copy()
    perturbed_initial_2 += np.random.normal(0, 0.1, size=n_points)
    perturbed_initial_2 = np.clip(perturbed_initial_2, 0, 1)
    perturbed_initial_2 = perturbed_initial_2 * (n_points / 2) / np.sum(perturbed_initial_2)
    initial_guesses.append(perturbed_initial_2)
    
    # 7. Random initial guess with different distribution
    random_initial_2 = np.zeros(n_points)
    indices = np.random.choice(n_points, size=n_points//4, replace=False)
    random_initial_2[indices] = 1
    random_initial_2 = random_initial_2 * (n_points / 2) / np.sum(random_initial_2)
    initial_guesses.append(random_initial_2)
    
    # 8. Structured initial guess with different pattern
    structured_initial_2 = np.zeros(n_points)
    structured_initial_2[::3] = 1
    structured_initial_2 = structured_initial_2 * (n_points / 2) / np.sum(structured_initial_2)
    initial_guesses.append(structured_initial_2)
    
    # 9. Randomly generated with some clustering
    random_initial_3 = np.zeros(n_points)
    cluster_centers = np.random.choice(n_points, size=4, replace=False)
    for center in cluster_centers:
        for i in range(max(0, center-5), min(n_points, center+5)):
            random_initial_3[i] += 1
    random_initial_3 = random_initial_3 * (n_points / 2) / np.sum(random_initial_3)
    initial_guesses.append(random_initial_3)
    
    # 10. Initial guess based on initial_h_values with different resampling
    x_initial_2 = np.linspace(0, 2, 160)
    f_initial_2 = interp1d(x_initial_2, initial_h_values, kind='cubic', fill_value='extrapolate')
    x_new_2 = np.linspace(0, 2, n_points)
    resampled_initial_2 = f_initial_2(x_new_2)
    resampled_initial_2 = resampled_initial_2 * (n_points / 2) / np.sum(resampled_initial_2)
    initial_guesses.append(resampled_initial_2)
    
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    
    def optimize(initial_h):
        def objective(h):
            # Compute the cross-correlation
            corr = correlate(h, 1 - h, mode='full')
            # Find the maximum value
            max_corr = np.max(corr)
            return max_corr * dx
        
        constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        result = minimize(
            objective,
            initial_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraint,
            options={'maxiter': 1000, 'ftol': 1e-10, 'eps': 1e-10}
        )
        return result.x, result.fun
    
    for guess in initial_guesses:
        h, c5 = optimize(guess)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
        # Check if time is exceeded
        if time.time() - start_time > budget_s:
            break
    
    return (best_h, best_c5, n_points)
