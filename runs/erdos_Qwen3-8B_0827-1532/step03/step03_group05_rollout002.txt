        constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
        result_slsqp = minimize(
            objective_slsqp,
            result_de.x,
            method='SLSQP',
            bounds=bounds,
            constraints=constraint,
            options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-9}
        )
        current_c5 = result_slsqp.fun
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_slsqp.x
        # Check time
        if time.time() - start_time > budget_s:
            break

    return (best_h, best_c5, n_points)
