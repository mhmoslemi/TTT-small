import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Use n_points=100 for faster optimization and better exploration
    n_points = 100
    dx = 2.0 / n_points
    
    # Generate diverse initial guesses
    initial_guesses = []
    
    # First guess: upsampled initial_h_values (if available)
    if 'initial_h_values' in globals():

        # Scale to meet the sum constraint
        initial_guess = initial_guess / np.sum(initial_guess) * (n_points / 2)
        initial_guesses.append(initial_guess)
    
    # Second guess: random initialization
    random_guess = np.random.rand(n_points)
    random_guess = random_guess / np.sum(random_guess) * (n_points / 2)
    initial_guesses.append(random_guess)
    
    # Third guess: alternating pattern
    structured_guess = np.zeros(n_points)
    structured_guess[::2] = 0.5
    structured_guess[1::2] = 0.5
    structured_guess = structured_guess / np.sum(structured_guess) * (n_points / 2)
    initial_guesses.append(structured_guess)
    
    # Fourth guess: perturbed initial_h_values
    if 'initial_h_values' in globals():
        perturbed = initial_h_values.copy()
        noise = np.random.normal(0, 0.05, n_points)
        perturbed += noise
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed / np.sum(perturbed) * (n_points / 2)
        initial_guesses.append(perturbed)
    
    # Fifth guess: block pattern
    block_guess = np.zeros(n_points)

    block_guess[n_points//2:] = 0.0
    block_guess = block_guess / np.sum(block_guess) * (n_points / 2)
    initial_guesses.append(block_guess)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraints: sum(h) == n_points / 2
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    for guess in initial_guesses:
        # Run optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 500}
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Fallback to initial guess if no improvement
    if best_h is None:

            objective,
            initial_guesses[0],
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 500}
        )
        best_c5 = result.fun
        best_h = result.x
