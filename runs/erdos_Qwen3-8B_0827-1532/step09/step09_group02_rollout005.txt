import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Predefined initial guess
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate structured initial guess with 5 non-overlapping intervals of 1s
    structured_initial = np.zeros(n_points)
    for i in range(5):
        start = i * 20
        end = start + 10
        structured_initial[start:end] = 1.0

    initial_guesses = [initial_h_values, structured_initial]
    
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    
    start_time = time.time()
    
    num_perturbations = 200  # Increased for better exploration
    
    for perturbation_idx in range(num_perturbations):
        # Select initial guess
        if perturbation_idx < 100:
            current_guess = initial_guesses[perturbation_idx % 2]
        else:
            # Random initial guess
            random_noise = np.random.uniform(-0.2, 0.2, n_points)
            current_guess = initial_h_values + random_noise
            current_guess = np.clip(current_guess, 0, 1)
            current_sum = np.sum(current_guess)
            scale = target_sum / current_sum
            current_guess = current_guess * scale
        
        # Apply perturbation
        if perturbation_idx < 50:
            noise = np.random.uniform(-0.2, 0.2, n_points)
        else:
            noise = np.random.uniform(-0.05, 0.05, n_points)
        perturbed = current_guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')

        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Optimization
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-12, 'maxiter': 2000}
        )
        
        # Update best solution
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
