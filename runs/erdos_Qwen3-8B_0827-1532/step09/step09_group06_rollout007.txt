import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Optimal balance between precision and speed
    dx = 2.0 / n_points
    x_new = np.linspace(0, 2, n_points)
    
    # Generate structured initial guesses
    initial_guesses = []
    
    # Step function guess: 1 on [0, 0.5] and [1.0, 1.5]
    step_guess = np.zeros(n_points)
    step_guess[:25] = 1.0
    step_guess[50:75] = 1.0
    initial_guesses.append(step_guess)
    
    # Binary guess: alternate 1 and 0
    binary_guess = np.zeros(n_points)
    binary_guess[::2] = 1.0
    initial_guesses.append(binary_guess)
    
    # Sine wave guess: (sin(πx) + 1)/2
    sine_guess = (np.sin(np.pi * x_new) + 1.0) / 2.0
    initial_guesses.append(sine_guess)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Run optimizations for each initial guess with tighter tolerances
    for guess in initial_guesses:
        # Run optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-8, 'maxiter': 1500}  # Tighter tolerance and more iterations
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # If no optimization was done, use the first initial guess
    if best_h is None:
        result = minimize(
            objective,
            initial_guesses[0],
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-8, 'maxiter': 1500}
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points
