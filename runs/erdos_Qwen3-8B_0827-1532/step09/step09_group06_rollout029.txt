from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Optimal balance between precision and speed
    dx = 2.0 / n_points
    x_new = np.linspace(0, 2, n_points)
    
    # Generate initial guesses with diverse structures
    initial_guesses = []
    
    # Use provided initial_h_values as base

    initial_guesses.append(initial_guess)
    
    # Generate sinusoidal initial guess
    freq = np.random.uniform(0.1, 1.0)
    sin_guess = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x_new)
    sin_guess = sin_guess / np.sum(sin_guess) * (n_points / 2)
    initial_guesses.append(sin_guess)
    
    # Generate triangular initial guess
    tri_guess = 0.5 * (1 + np.abs(2 * x_new - 1))
    tri_guess = tri_guess / np.sum(tri_guess) * (n_points / 2)
    initial_guesses.append(tri_guess)
    
    # Generate binary initial guess
    binary_guess = np.zeros(n_points)
    binary_guess[::2] = 1.0
    binary_guess = binary_guess / np.sum(binary_guess) * (n_points / 2)
    initial_guesses.append(binary_guess)
    
    # Generate noisy perturbations from initial_h_values
    for _ in range(5):
        noise_amp = np.random.uniform(0.001, 0.05)

        random_noise = np.random.normal(0, 0.01, n_points)

        perturbed = np.clip(perturbed, 0, 1)
        scale_factor = (n_points / 2) / np.sum(perturbed)
        perturbed *= scale_factor
        initial_guesses.append(perturbed)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    
    # Run optimizations for each initial guess with tighter tolerances
    for guess in initial_guesses:
        # Run optimization
        result = minimize(

            guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-8, 'maxiter': 200}  # Reduced maxiter and tighter tolerance
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.success:
            current_c5 = result.fun

        )
