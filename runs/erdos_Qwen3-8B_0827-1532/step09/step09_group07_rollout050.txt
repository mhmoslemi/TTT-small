from scipy.optimize import minimize
import time

    x_new = np.linspace(0, 2, n_points)
    initial_guess = np.interp(x_new, x_initial, initial_guess)
    # Adjust sum to meet the constraint
    initial_guess = initial_guess / np.sum(initial_guess) * (n_points / 2)
    
    # Generate a few perturbations of the initial guess
    perturbations = []
    for _ in range(5):
        # Add noise with varying amplitude and frequency
        noise_amp = np.random.uniform(0.001, 0.05)

        noise = noise_amp * np.sin(2 * np.pi * noise_freq * x_new)
        perturbed = initial_guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        scale_factor = (n_points / 2) / np.sum(perturbed)
        perturbed *= scale_factor
        perturbations.append(perturbed)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Run optimizations for each initial guess with tighter tolerances
    for guess in [initial_guess] + perturbations:
        # Run optimization
        result = minimize(
            objective,
            guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
