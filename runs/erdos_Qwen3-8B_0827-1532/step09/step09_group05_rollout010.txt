    initial_guesses = []
    
    # Step function initial guess
    x = np.linspace(0, 2, n_points)

    h_step[(x >= 0.5) & (x < 1.5)] = 0.5
    h_step[x >= 1.5] = 0.0
    initial_guesses.append(h_step)
    
    # Sinusoidal initial guess
    h_sinusoidal = 0.5 + 0.5 * np.sin(np.pi * x)
    initial_guesses.append(h_sinusoidal)
    
    # Random initial guess
    h_random = np.random.uniform(0, 1, n_points)
    h_random = h_random / np.sum(h_random) * (n_points / 2)
    initial_guesses.append(h_random)
    
    # Add noise to each initial guess
    for guess in initial_guesses:
        noise_amp = np.random.uniform(0.01, 0.05)
        noise = noise_amp * np.random.normal(0, 1, n_points)
        perturbed = guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        scale_factor = (n_points / 2) / np.sum(perturbed)
        perturbed *= scale_factor
        initial_guesses.append(perturbed)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Run optimizations for each initial guess
    for guess in initial_guesses:
        # Run optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-10, 'maxiter': 1000}  # Reduced iterations for efficiency
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # If no optimization was done, use the initial guess
    if best_h is None:
        result = minimize(
            objective,
            initial_guesses[0],
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-10, 'maxiter': 1000}
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points

import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 200  # Increased for better resolution
    dx = 2.0 / n_points
    
    # Generate initial guesses
    initial_guesses = []
    
    # Step function initial guess
    x = np.linspace(0, 2, n_points)
    h_step = np.zeros_like(x)
    h_step[x < 0.5] = 1.0
    h_step[(x >= 0.5) & (x < 1.5)] = 0.5
    h_step[x >= 1.5] = 0.0
    initial_guesses.append(h_step)
    
    # Sinusoidal initial guess
    h_sinusoidal = 0.5 + 0.5 * np.sin(np.pi * x)
    initial_guesses.append(h_sinusoidal)
    
    # Random initial guess
    h_random = np.random.uniform(0, 1, n_points)
    h_random = h_random / np.sum(h_random) * (n_points / 2)
    initial_guesses.append(h_random)
    
    # Add noise to each initial guess
    for guess in initial_guesses:
        noise_amp = np.random.uniform(0.01, 0.05)
        noise = noise_amp * np.random.normal(0, 1, n_points)
        perturbed = guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        scale_factor = (n_points / 2) / np.sum(perturbed)
        perturbed *= scale_factor
        initial_guesses.append(perturbed)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Run optimizations for each initial guess
    for guess in initial_guesses:
        # Run optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-10, 'maxiter': 1000}  # Reduced iterations for efficiency
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # If no optimization was done, use the initial guess

        result = minimize(
            objective,
            initial_guesses[0],
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-10, 'maxiter': 1000}
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points
