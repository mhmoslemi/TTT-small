from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)

    dx = 2.0 / n_points
    
    # Use initial_h_values as base and scale to meet the constraint
    initial_guess = np.array(initial_h_values)

    num_initial_guesses = 50  # Increased number for better exploration
    initial_guesses = []
    for _ in range(num_initial_guesses):

        if np.random.rand() < 0.5:
            # Binary function with shuffled positions
            binary = np.zeros(n_points)
            binary[:n_points//2] = 1
            np.random.shuffle(binary)
            initial_guesses.append(binary)
        else:
            # Perturb the initial guess with sinusoidal noise
            noise_amp = np.random.uniform(0.001, 0.1)  # Increased amplitude range
            noise_freq = np.random.uniform(1, 10)

            perturbed = initial_guess + noise

            perturbed = np.clip(perturbed, 0, 1)

            scale_factor = (n_points / 2) / np.sum(perturbed)

    def objective(h):

        return max_corr * dx
    
    # Constraint function
    def constraint(h):

    # Run optimizations for each initial guess with tighter tolerances
    for guess in initial_guesses:
        # Run optimization
        result = minimize(

            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-8, 'maxiter': 1500}  # Adjusted tolerances and iterations
        )
        elapsed = time.time() - start_time
