    x_new = np.linspace(0, 2, n_points)
    initial_guess = np.interp(x_new, x_initial, initial_h_values)
    # Adjust sum to meet the constraint
    initial_guess = initial_guess / np.sum(initial_guess) * (n_points / 2)
    
    # Generate diverse initial guesses with structured and random noise
    num_initial_guesses = 50  # Reduced from 200
    initial_guesses = []
    for _ in range(num_initial_guesses):
        # Add structured noise with varying amplitude and frequency
        noise_amp = np.random.uniform(0.001, 0.1)
        noise_freq = np.random.uniform(1, 10)
        noise = noise_amp * np.sin(2 * np.pi * noise_freq * x_new)
        # Add random noise
        random_noise = np.random.normal(0, 0.01, n_points)
        perturbed = initial_guess + noise + random_noise
        # Clip to [0, 1]
        perturbed = np.clip(perturbed, 0, 1)
        # Adjust sum to meet the constraint
        scale_factor = (n_points / 2) / np.sum(perturbed)
        perturbed *= scale_factor
        initial_guesses.append(perturbed)
    
    # Generate structured initial guesses
    def generate_sine_guess(x):
        h = 0.5 + 0.5 * np.sin(2 * np.pi * x)
        h = np.clip(h, 0, 1)
        scale_factor = (n_points / 2) / np.sum(h)
        return h * scale_factor
    
    def generate_triangular_guess(x):
        h = np.abs(x - 1) / 1.0
        h = np.clip(h, 0, 1)
        scale_factor = (n_points / 2) / np.sum(h)
        return h * scale_factor
    
    def generate_binary_guess_2(x):
        binary = np.zeros_like(x)
        binary[:int(n_points/4)] = 1.0
        binary[-int(n_points/4):] = 1.0
        binary = binary / np.sum(binary) * (n_points / 2)
        return binary
    
    initial_guesses.append(generate_sine_guess(x_new))
    initial_guesses.append(generate_triangular_guess(x_new))
    initial_guesses.append(generate_binary_guess_2(x_new))
    
    # Generate a binary initial guess
    binary_guess = np.zeros(n_points)
    binary_guess[::2] = 1.0  # Alternate 1 and 0
    binary_guess = binary_guess / np.sum(binary_guess) * (n_points / 2)
    initial_guesses.append(binary_guess)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Run optimizations for each initial guess with tighter tolerances
    for guess in initial_guesses:
        # Run optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-9, 'maxiter': 1000}  # Reduced maxiter and tighter tolerance
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # If no optimization was done, use the initial guess
    if best_h is None:
        result = minimize(
            objective,
            initial_guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-9, 'maxiter': 1000}
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points
