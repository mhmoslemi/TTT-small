        block[::2] = 1.0  # alternate 1 and 0
        sum_block = np.sum(block)
        scale = (n_points / 2) / sum_block
        perturbed = block * scale
        initial_guesses.append(perturbed)
    
    # Define objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Define the optimization function
    def optimize_guess(guess):
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-9, 'maxiter': 1000}
        )
        return (result.fun, result.x)
    
    # Use multiprocessing to evaluate initial guesses
    with Pool(processes=10) as pool:
        results = pool.map(optimize_guess, initial_guesses)
    
    # Collect the best result
    for c5, h in results:
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    
    # If no result was found (unlikely), use the initial guess
    if best_h is None:
        result = minimize(
            objective,
            initial_guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-9, 'maxiter': 1000}
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points
