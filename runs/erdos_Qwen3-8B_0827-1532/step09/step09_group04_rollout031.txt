import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Optimal balance between speed and precision
    dx = 2.0 / n_points
    
    # Generate diverse initial guesses with different patterns
    initial_guesses = []
    
    # Base initial guess from provided values
    initial_guess = np.array(initial_h_values)
    initial_guess = initial_guess / np.sum(initial_guess) * (n_points / 2)
    initial_guesses.append(initial_guess)
    
    # Sine wave patterns
    for freq in [1, 2, 3, 4]:
        x = np.linspace(0, 2, n_points)
        h_sine = 0.5 * (1 + np.sin(2 * np.pi * freq * x))
        scale_factor = (n_points / 2) / np.sum(h_sine)
        h_sine *= scale_factor
        initial_guesses.append(h_sine)
    
    # Square wave patterns
    for duty_cycle in [0.3, 0.5, 0.7]:
        h_square = np.zeros(n_points)
        for i in range(n_points):
            if i / n_points < duty_cycle:
                h_square[i] = 1
            else:
                h_square[i] = 0
        scale_factor = (n_points / 2) / np.sum(h_square)
        h_square *= scale_factor
        initial_guesses.append(h_square)
    
    # Binary random pattern
    binary_guess = np.zeros(n_points)
    indices = np.random.choice(n_points, size=50, replace=False)
    binary_guess[indices] = 1
    scale_factor = (n_points / 2) / np.sum(binary_guess)
    binary_guess *= scale_factor
    initial_guesses.append(binary_guess)
    
    # Triangular wave
    x = np.linspace(0, 2, n_points)
    h_tri = np.abs(x - 1) / 1  # Triangular wave from 0 to 2
    h_tri = np.clip(h_tri, 0, 1)
    scale_factor = (n_points / 2) / np.sum(h_tri)
    h_tri *= scale_factor
    initial_guesses.append(h_tri)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Run optimizations for each initial guess with tighter tolerances
    for guess in initial_guesses:
        # Run optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-10, 'maxiter': 3000}  # Tighter tolerances and more iterations
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # If no optimization was done, use the initial guess
    if best_h is None:
        result = minimize(
            objective,
            initial_guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-10, 'maxiter': 3000}
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points
