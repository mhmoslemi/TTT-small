    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Function to run optimization for a single guess
    def optimize_single(guess):
        result = minimize(
            objective,
            guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-6, 'maxiter': 100}
        )
        if result.success:
            return (result.fun, result.x)
        else:
            return (float('inf'), guess)  # Fallback to original guess
    
    # Use multiprocessing to run optimizations in parallel
    with Pool(processes=10) as pool:
        results = pool.map(optimize_single, initial_guesses)
    
    # Find the best result from all optimizations
    for c5, h in results:
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    
    # If no optimization was done, use the initial guess
    if best_h is None:
        result = minimize(
            objective,
            initial_guesses[0],
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-6, 'maxiter': 100}
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points
