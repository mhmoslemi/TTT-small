    structured_h2 = np.zeros(n_points)
    for i in range(n_points):
        if i % 3 == 0:
            structured_h2[i] = 1.0
        else:
            structured_h2[i] = 0.0
    
    structured_h3 = np.zeros(n_points)
    for i in range(n_points):
        if i % 4 == 0:
            structured_h3[i] = 1.0
        else:
            structured_h3[i] = 0.0
    
    # Generate a random initial guess with sum constraint
    np.random.seed(seed)

    random_guess = random_guess / np.sum(random_guess) * target_sum
    
    # Combine initial guesses
    initial_guesses = [initial_guess, structured_h, structured_h2, structured_h3, random_guess]
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Ensure the guess is within [0, 1] and scaled correctly
        guess = np.clip(guess, 0, 1)
        current_sum = np.sum(guess)
        scale = target_sum / current_sum
        guess = guess * scale
        
        # Objective function to minimize C5
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

        result = minimize(
