        # Random step function with 50% ones and 50% zeros

        # Ensure sum is correct
        current_sum = np.sum(random_h)
        scale = target_sum / current_sum
        random_h = random_h * scale
        # Add small noise to diversify
        noise = np.random.uniform(-0.05, 0.05, n_points)
        random_h = np.clip(random_h + noise, 0, 1)

        if perturbation_idx < 10:
            # Large noise for early exploration

            # Small noise for refinement
            noise = np.random.uniform(-0.05, 0.05, n_points)
        perturbed = initial_guess + noise

        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum

    # Optimize each perturbation
    for perturbed in perturbations:
        # Objective function to minimize C5
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
