import numpy as np
from scipy.optimize import minimize
import time
from concurrent.futures import ThreadPoolExecutor

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate diverse initial guesses
    def generate_initial_guesses(num_guesses):
        guesses = []
        # Add initial guess
        guesses.append(initial_guess.copy())
        # Add random guesses
        for _ in range(num_guesses):
            h = np.random.uniform(0, 1, n_points)
            h = h * (target_sum / np.sum(h))
            guesses.append(h)
        # Add structured guesses (alternating 0s and 1s)
        for pattern in [np.array([0, 1]*n_points), np.array([1, 0]*n_points)]:
            h = pattern[:n_points]
            h = h * (target_sum / np.sum(h))
            guesses.append(h)
        return guesses
    
    initial_guesses = generate_initial_guesses(100)
    
    # Objective function to minimize C5
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Constraints: sum(h) == target_sum
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    
    # Optimization parameters
    opt_params = {
        'method': 'SLSQP',
        'bounds': bounds,
        'constraints': cons,
        'options': {'ftol': 1e-10, 'maxiter': 1500}
    }
    
    # Function to optimize a single guess
    def optimize_guess(h0):
        result = minimize(objective, h0, **opt_params)
        if result.success:
            return result.x, result.fun
        return None, float('inf')
    
    # Use ThreadPoolExecutor for parallel optimization
    with ThreadPoolExecutor(max_workers=10) as executor:
        futures = [executor.submit(optimize_guess, h) for h in initial_guesses]
        for future in futures:
            h_opt, c5_val = future.result()
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt
