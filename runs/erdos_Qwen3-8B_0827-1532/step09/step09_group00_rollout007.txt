    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Generate structured initial guesses
    structured_h = np.zeros(n_points)
    for i in range(n_points):
        if i % 2 == 0:
            structured_h[i] = 1.0
        else:
            structured_h[i] = 0.0
    
    structured_h2 = np.zeros(n_points)
    for i in range(n_points):
        if i % 3 == 0:
            structured_h2[i] = 1.0
        else:
            structured_h2[i] = 0.0
    
    structured_h3 = np.zeros(n_points)
    for i in range(n_points):
        if i % 4 == 0:
            structured_h3[i] = 1.0
        else:
            structured_h3[i] = 0.0
    
    initial_guesses = [initial_guess, structured_h, structured_h2, structured_h3]
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    start_time = time.time()
    
    for guess in initial_guesses:
        # Define objective with penalty for constraint violation
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            penalty = 1e6 * abs(np.sum(h) - target_sum)
            return max_corr * dx + penalty
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Run differential evolution
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=15,
            mutation=(0.5, 1.0),
            recombination=0.9,
            tol=1e-10,
            maxiter=100,
            seed=seed
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
