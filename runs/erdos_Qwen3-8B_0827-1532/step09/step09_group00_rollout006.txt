import time

    n_points = len(initial_guess)

    for period in [2, 3, 4, 5, 6]:
        structured = np.zeros(n_points)
        for i in range(n_points):
            if i % period == 0:
                structured[i] = 1.0
            else:
                structured[i] = 0.0
        structured_guesses.append(structured)

    np.random.seed(seed)
    random_guesses = []
    for _ in range(10):
        h = np.random.uniform(0, 1, n_points)
        h_sum = np.sum(h)
        scale = target_sum / h_sum
        h = h * scale
        random_guesses.append(h)
    
    # Combine all initial guesses
    initial_guesses = [initial_guess] + structured_guesses + random_guesses
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Objective function to minimize C5
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization with tighter tolerances and more iterations
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 1000}
        )
        
        # Check if this result is better
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
