        x = x_new[i]
        if 0.0 <= x < 0.5 or 1.0 <= x < 1.5:
            two_interval[i] = 1.0
        else:
            two_interval[i] = 0.0
    initial_guesses.append(two_interval)
    # Add the initial guess
    initial_guesses.append(initial_guess)
    
    # Now, run optimizations in parallel
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    
    # Define the objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Use a pool of workers
    with ProcessPoolExecutor(max_workers=10) as executor:
        futures = []
        for guess in initial_guesses:
            # Submit the optimization task
            future = executor.submit(minimize, 
                                    objective, 
                                    guess, 
                                    method='SLSQP', 
                                    bounds=bounds, 
                                    constraints={'type': 'eq', 'fun': constraint}, 
                                    options={'ftol': 1e-12, 'maxiter': 3000})
            futures.append(future)
        
        # Wait for results and track the best
        for future in as_completed(futures):
            result = future.result()
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
            # Check elapsed time
            elapsed = time.time() - start_time
            if elapsed >= budget_s:
                break
    
    # If no optimization was done, use the initial guess
    if best_h is None:
        result = minimize(
            objective,
            initial_guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-12, 'maxiter': 3000}
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points
