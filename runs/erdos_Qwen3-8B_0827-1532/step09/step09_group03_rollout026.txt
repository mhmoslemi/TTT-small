    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate structured initial guesses
    def create_two_intervals_guess(n_points):
        dx = 2.0 / n_points
        h = np.zeros(n_points)
        # First interval [0, 0.5]
        start = 0
        end = int(0.5 / dx)
        h[start:end+1] = 1.0
        # Second interval [1.0, 1.5]
        start = int(1.0 / dx)
        end = int(1.5 / dx)
        h[start:end+1] = 1.0
        # Scale to meet integral constraint
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h
        h *= scale
        return h
    
    def create_three_intervals_guess(n_points):
        dx = 2.0 / n_points
        h = np.zeros(n_points)
        # First interval [0, 0.4]
        start = 0
        end = int(0.4 / dx)
        h[start:end+1] = 1.0
        # Second interval [0.6, 1.0]
        start = int(0.6 / dx)
        end = int(1.0 / dx)
        h[start:end+1] = 1.0
        # Third interval [1.2, 1.6]
        start = int(1.2 / dx)
        end = int(1.6 / dx)
        h[start:end+1] = 1.0
        # Scale to meet integral constraint
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h
        h *= scale
        return h
    
    initial_guesses = [initial_guess, create_two_intervals_guess(n_points), create_three_intervals_guess(n_points)]
    
    for guess in initial_guesses:
        # Objective function to minimize C5
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 1500}
        )
        
        # Check if this result is better
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
