            constraints=[{'type': 'eq', 'fun': constraint}],
            popsize=15,
            maxiter=100,
            tol=1e-6,
            mutation=(0.5, 1.0),
            recombination=0.8,
            seed=seed
        )
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Additional refinement with local optimization
    if best_c5 < float('inf'):
        result = minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            tol=1e-8,
            options={'maxiter': 100}
        )
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Check elapsed time
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
