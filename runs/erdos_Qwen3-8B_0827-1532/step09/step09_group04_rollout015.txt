from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    dx = 2.0 / n_points

    initial_guesses = []
    
    # Base initial guess with noise

    noise_freq = np.random.uniform(1, 10)
    noise = noise_amp * np.sin(2 * np.pi * noise_freq * np.linspace(0, 2, n_points))
    perturbed = initial_guess + noise
    perturbed = np.clip(perturbed, 0, 1)
    scale_factor = (n_points / 2) / np.sum(perturbed)
    perturbed *= scale_factor
    initial_guesses.append(perturbed)
    
    # Random binary patterns
    for _ in range(5):
        binary = np.zeros(n_points)
        binary[:n_points//2] = 1
        np.random.shuffle(binary)
        initial_guesses.append(binary)
    
    # Random uniform distributions
    for _ in range(5):
        uniform = np.random.uniform(0, 1, n_points)
        scale_factor = (n_points / 2) / np.sum(uniform)
        initial_guesses.append(uniform * scale_factor)
    
    # Periodic functions
    for _ in range(5):
        freq = np.random.uniform(1, 5)
        amp = np.random.uniform(0.1, 0.5)
        periodic = amp * np.sin(2 * np.pi * freq * np.linspace(0, 2, n_points))
        periodic = (periodic + 1) / 2  # Shift to [0.5, 1.5], then scale to [0, 1]
        periodic = np.clip(periodic, 0, 1)

        initial_guesses.append(periodic * scale_factor)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Run optimizations for each initial guess with tighter tolerances
    for guess in initial_guesses:
        # Run optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-10, 'maxiter': 3000}  # Tighter tolerances and more iterations
        )
        elapsed = time.time() - start_time
