import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 50  # Smaller n_points for faster computation
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    
    start_time = time.time()
    
    # Generate multiple random initial guesses
    num_initial_guesses = 100  # Adjust based on time
    for _ in range(num_initial_guesses):
        # Generate random initial guess
        h = np.random.uniform(0, 1, n_points)
        current_sum = np.sum(h)
        scale = target_sum / current_sum
        h = h * scale
        
        # Objective function to minimize C5
        def objective(h_vec):
            one_minus_h = 1.0 - h_vec
            corr = np.correlate(h_vec, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization with tighter tolerances and more iterations
        result = minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-8, 'maxiter': 500}
        )
        
        # Check if this result is better
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
