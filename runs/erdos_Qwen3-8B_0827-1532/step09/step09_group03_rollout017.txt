import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    # Use the provided initial guess
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Define the objective function with penalty
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        c5 = max_corr * dx
        # Penalty for sum deviation
        penalty = 1e4 * (np.sum(h) - target_sum)**2
        return c5 + penalty
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Run differential evolution with multiple restarts
    for _ in range(3):  # Limited restarts to stay within budget
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=15,
            maxiter=100,
            tol=1e-10,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed
        )
        
        # Check if the result is better
        if result.success:
            current_h = result.x
            current_c5 = result.fun
            # Normalize to ensure sum constraint
            current_sum = np.sum(current_h)
            if current_sum != 0:
                current_h = current_h * (target_sum / current_sum)
            # Evaluate C5 again with normalized h
            one_minus_h = 1.0 - current_h
            corr = np.correlate(current_h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            current_c5 = max_corr * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
