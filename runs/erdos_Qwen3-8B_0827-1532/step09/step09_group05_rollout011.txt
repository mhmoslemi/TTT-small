from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Optimal balance between speed and precision
    dx = 2.0 / n_points
    
    # Generate initial guess based on provided initial_h_values
    initial_guess = np.array(initial_h_values)
    # Adjust sum to meet the constraint
    initial_guess = initial_guess / np.sum(initial_guess) * (n_points / 2)
    
    # Generate diverse initial guesses with mixed noise types
    num_initial_guesses = 100  # Increased diversity
    initial_guesses = []
    for _ in range(num_initial_guesses):
        # Randomly choose noise type
        if np.random.rand() < 0.5:
            # Sinusoidal noise with varying frequency and amplitude
            noise_amp = np.random.uniform(0.001, 0.1)
            noise_freq = np.random.uniform(1, 20)
            noise = noise_amp * np.sin(2 * np.pi * noise_freq * np.linspace(0, 2, n_points))
        else:
            # Random noise with uniform distribution
            noise = np.random.uniform(-0.05, 0.05, size=n_points)
        # Perturb the initial guess
        perturbed = initial_guess + noise
        # Clip to [0, 1]
        perturbed = np.clip(perturbed, 0, 1)
        # Adjust sum to meet the constraint
        scale_factor = (n_points / 2) / np.sum(perturbed)
        perturbed *= scale_factor
        initial_guesses.append(perturbed)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Run optimizations for each initial guess with tighter tolerances
    for guess in initial_guesses:
        # Run optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-12, 'maxiter': 2000}  # Tighter tolerance and more iterations
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:

        if result.success:

        )

        best_h = result.x
