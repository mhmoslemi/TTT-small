import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Predefined initial guess
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Generate diverse initial guesses
    initial_guesses = []
    initial_guesses.append(initial_h_values)  # Provided initial guess

    initial_guesses.append(random_guess)
    
    # Structured guess: alternating 1s and 0s
    structured_guess = np.zeros(n_points)
    for i in range(n_points):
        structured_guess[i] = 1.0 if i % 2 == 0 else 0.0
    initial_guesses.append(structured_guess)
    
    # Block-based guess: 1s in first half, 0s in second
    block_guess = np.zeros(n_points)
    block_guess[:n_points//2] = 1.0
    initial_guesses.append(block_guess)
    
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    
    start_time = time.time()
    
    for guess in initial_guesses:

        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx

        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

        result = minimize(

            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 200}
        )
        
        # Update best solution
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
