import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate multiple initial guesses
    initial_guesses = [initial_guess]
    for _ in range(4):
        h = np.random.uniform(0, 1, n_points)
        scale = target_sum / np.sum(h)
        initial_guesses.append(h * scale)
    
    # Define objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    # Define constraints
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    
    # Bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization parameters
    max_iter = 2000
    ftol = 1e-10
    
    for guess in initial_guesses:
        # Perturb and scale to meet sum constraint
        perturbed = guess.copy()
        noise = np.random.uniform(-0.2, 0.2, n_points)
        perturbed += noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed *= scale
        
        # Run optimization
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': ftol, 'maxiter': max_iter}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
