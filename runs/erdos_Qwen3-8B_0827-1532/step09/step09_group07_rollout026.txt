    scale_factor = (n_points / 2) / np.sum(perturbed_guess)
    perturbed_guess *= scale_factor
    
    # Generate structured guess with three intervals
    structured_guess = np.zeros(n_points)
    for i in range(n_points):
        x = x_new[i]
        if 0.0 <= x < 0.333 or 0.666 <= x < 1.0 or 1.333 <= x < 1.666:
            structured_guess[i] = 1.0
        else:
            structured_guess[i] = 0.0
    # Adjust sum to meet the constraint
    structured_guess = structured_guess / np.sum(structured_guess) * (n_points / 2)
    
    # List of initial guesses
    initial_guesses = [initial_guess, perturbed_guess, structured_guess]
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Define a function to optimize a single guess
    def optimize_guess(guess):
        result = minimize(
            objective,
            guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-12, 'maxiter': 1000}
        )
        if result.success:
            return result.fun, result.x
        else:
            return float('inf'), guess  # fallback to initial guess
    
    # Use multiprocessing to run optimizations in parallel
    with multiprocessing.Pool() as pool:
        results = pool.map(optimize_guess, initial_guesses)
    
    # Find the best result
    for c5, h in results:
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    
    # If no result was found, use the initial guess
    if best_h is None:
        result = minimize(
            objective,
            initial_guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-12, 'maxiter': 1000}
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points
