            periodic_h[i] = 0.0

    initial_guesses = [initial_guess, structured_h, periodic_h]
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()

        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)

        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
