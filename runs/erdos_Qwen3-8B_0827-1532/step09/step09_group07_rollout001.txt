            square_wave[i] = 1.0
        else:
            square_wave[i] = 0.0
    initial_guesses.append(square_wave)
    # Triangular wave
    triangular_wave = np.zeros(n_points)
    for i in range(n_points):
        x = x_new[i]
        if x < 1.0:
            triangular_wave[i] = x / 1.0
        else:
            triangular_wave[i] = (2.0 - x) / 1.0
    initial_guesses.append(triangular_wave)
    # Sine wave
    sine_wave = 0.5 + 0.5 * np.sin(np.pi * x_new)
    initial_guesses.append(sine_wave)
    # Sawtooth wave
    sawtooth_wave = np.zeros(n_points)
    for i in range(n_points):
        x = x_new[i]
        if x < 1.0:
            sawtooth_wave[i] = x / 1.0
        else:
            sawtooth_wave[i] = (x - 1.0) / 1.0
    initial_guesses.append(sawtooth_wave)
    # Two-interval wave
    two_interval = np.zeros(n_points)
    for i in range(n_points):
        x = x_new[i]
        if 0.0 <= x < 0.5 or 1.0 <= x < 1.5:
            two_interval[i] = 1.0
        else:
            two_interval[i] = 0.0
    initial_guesses.append(two_interval)
    # Three-interval wave
    three_interval = np.zeros(n_points)
    for i in range(n_points):
        x = x_new[i]
        if 0.0 <= x < 0.333 or 0.666 <= x < 1.0 or 1.333 <= x < 1.666:
            three_interval[i] = 1.0
        else:
            three_interval[i] = 0.0
    initial_guesses.append(three_interval)
    # Add initial guess
    initial_guesses.append(initial_guess)
    
    # Add a few perturbations of the initial guess
    for _ in range(5):
        # Perturb with sine wave
        noise_amp = np.random.uniform(0.001, 0.05)
        noise_freq = np.random.uniform(0.5, 10)
        noise = noise_amp * np.sin(2 * np.pi * noise_freq * x_new)
        perturbed = initial_guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        scale_factor = (n_points / 2) / np.sum(perturbed)
        perturbed *= scale_factor
        initial_guesses.append(perturbed)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Run optimizations for each initial guess with tighter tolerances
    # Using L-BFGS-B method
    with ProcessPoolExecutor(max_workers=10) as executor:
        futures = []
        for guess in initial_guesses:
            future = executor.submit(
                minimize,
                objective,
                guess,
                method='L-BFGS-B',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint},
                options={'ftol': 1e-12, 'maxiter': 1000}
            )
            futures.append(future)
        
        for future in futures:
            if time.time() - start_time >= budget_s:
                break
            result = future.result()
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
    
    # If no optimization was done, use the initial guess
    if best_h is None:
        result = minimize(
            objective,
            initial_guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-12, 'maxiter': 1000}
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points
