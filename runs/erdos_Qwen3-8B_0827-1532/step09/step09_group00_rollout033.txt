from scipy.optimize import minimize
import time

    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Constraints: sum(h) == target_sum
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    best_c5 = float('inf')
    best_h = initial_guess.copy()

    # Generate a few diverse initial guesses based on best_h

        perturbed = best_h + random_noise
