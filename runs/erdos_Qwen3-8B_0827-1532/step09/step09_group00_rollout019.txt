from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Assume initial_h_values is defined globally
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Generate structured initial guesses
    structured_h = np.zeros(n_points)

            structured_h[i] = 0.0

    for i in range(n_points):
        if i % 3 == 0:
            structured_h2[i] = 1.0
        else:
            structured_h2[i] = 0.0

    structured_h3 = np.zeros(n_points)

        if i % 4 == 0:
            structured_h3[i] = 1.0
        else:
            structured_h3[i] = 0.0

    # Combine initial guesses
    initial_guesses = [initial_guess, structured_h, structured_h2, structured_h3]

    best_c5 = float('inf')
    best_h = initial_guess.copy()

    start_time = time.time()

    # Define objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    for guess in initial_guesses:

            objective,
            guess_scaled,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 1000}
        )

        # Check if this result is better
        if result.success and result.fun < best_c5:
