import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Optimal balance between precision and speed
    dx = 2.0 / n_points
    x_new = np.linspace(0, 2, n_points)

    initial_guesses = []
    
    # Initial guess from provided initial_h_values
    initial_guess = initial_h_values.copy()
    scale_factor = (n_points / 2) / np.sum(initial_guess)
    initial_guess *= scale_factor
    initial_guesses.append(initial_guess)
    
    # Binary pattern
    binary_guess = np.zeros(n_points)
    binary_guess[::2] = 1.0
    scale_factor = (n_points / 2) / np.sum(binary_guess)
    binary_guess *= scale_factor
    initial_guesses.append(binary_guess)
    
    # Sinusoidal pattern
    sinusoidal_guess = 0.5 + 0.5 * np.sin(2 * np.pi * x_new / 2)
    scale_factor = (n_points / 2) / np.sum(sinusoidal_guess)

    initial_guesses.append(sinusoidal_guess)
    
    # Random pattern
    random_guess = np.random.uniform(0, 1, n_points)
    scale_factor = (n_points / 2) / np.sum(random_guess)
    random_guess *= scale_factor
    initial_guesses.append(random_guess)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')

    # Run optimizations for each initial guess
    for guess in initial_guesses:
        # Run optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-9, 'maxiter': 1000}  # Reduced maxiter and tighter tolerance
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.success:

            initial_guesses[0],
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-9, 'maxiter': 1000}
        )

        best_h = result.x
