import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Use the provided initial guess
    initial_guesses = [initial_h_values]
    n_points = len(initial_guesses[0])
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guesses[0].copy()
    
    start_time = time.time()
    
    # Generate additional initial guesses
    # Random initial guess
    random_h = np.random.uniform(0, 1, n_points)
    random_h = random_h * (target_sum / np.sum(random_h))
    initial_guesses.append(random_h)
    
    # Block pattern: alternate blocks of 1 and 0
    block_size = n_points // 4
    block_h = np.zeros(n_points)
    for i in range(0, n_points, block_size * 2):
        block_h[i:i+block_size] = 1.0
    block_h = block_h * (target_sum / np.sum(block_h))
    initial_guesses.append(block_h)
    
    # Alternating pattern
    alt_h = np.zeros(n_points)
    for i in range(n_points):
        if i % 2 == 0:
            alt_h[i] = 1.0
    alt_h = alt_h * (target_sum / np.sum(alt_h))
    initial_guesses.append(alt_h)
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Objective function to minimize C5
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization with tighter tolerances and more iterations
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 1500}
        )
        
        # Check if this result is better
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
