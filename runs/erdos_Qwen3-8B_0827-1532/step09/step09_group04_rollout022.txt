from scipy.optimize import minimize
import time

    n_points = 100  # Optimal balance between speed and precision
    dx = 2.0 / n_points
    
    # Use initial_h_values as base and scale to meet the constraint
    initial_guess = np.array(initial_h_values)
    # Adjust sum to meet the constraint
    initial_guess = initial_guess / np.sum(initial_guess) * (n_points / 2)

    num_initial_guesses = 30  # Increased for better exploration
    initial_guesses = []
    for _ in range(num_initial_guesses // 2):
        # Add random noise to initial guess
        noise = np.random.uniform(-0.05, 0.05, size=n_points)
        perturbed = initial_guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        scale_factor = (n_points / 2) / np.sum(perturbed)
        perturbed *= scale_factor
        initial_guesses.append(perturbed)
    for _ in range(num_initial_guesses // 2):
        # Generate completely random guess
        h = np.random.uniform(0, 1, n_points)
        scale_factor = (n_points / 2) / np.sum(h)
        h *= scale_factor
        initial_guesses.append(h)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Run optimizations for each initial guess with tighter tolerances
    for guess in initial_guesses:
        # Run optimization
        result = minimize(
            objective,
            guess,
            method='trust-constr',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-8, 'gtol': 1e-8, 'maxiter': 1000}  # Tighter tolerances and more iterations
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # If no optimization was done, use the initial guess

        result = minimize(

            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-8, 'gtol': 1e-8, 'maxiter': 1000}
        )
        best_c5 = result.fun
        best_h = result.x
