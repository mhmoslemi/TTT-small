import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate diverse initial guesses
    num_guesses = 300
    for _ in range(num_guesses):
        # Random or perturbed initial guess
        if np.random.rand() < 0.5:
            # Random guess
            h = np.random.uniform(0, 1, n_points)
            h = h * (target_sum / np.sum(h))
        else:
            # Perturb best_h
            noise = np.random.uniform(-0.1, 0.1, n_points)
            h = np.clip(best_h + noise, 0, 1)
            h = h * (target_sum / np.sum(h))
        
        # Objective function
        def objective(x):
            one_minus_h = 1.0 - x
            corr = np.correlate(x, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Constraints and bounds
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        
        # Optimization
        result = minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 1200}
        )
        
        # Update best solution
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check time budget
        if time.time() - start_time >= budget_s:
            break
    
    return best_h, best_c5, n_points
