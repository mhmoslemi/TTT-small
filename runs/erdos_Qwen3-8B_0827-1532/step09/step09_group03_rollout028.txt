import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Precompute constants
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Generate diverse initial guesses
    initial_guesses = []
    
    # 1. Original initial guess
    initial_guesses.append(initial_h_values.copy())
    
    # 2. Random binary pattern
    num_ones = n_points // 2
    random_indices = np.random.choice(n_points, size=num_ones, replace=False)
    h_random = np.zeros(n_points)
    h_random[random_indices] = 1.0
    initial_guesses.append(h_random)
    
    # 3. Periodic alternating pattern
    h_periodic = np.zeros(n_points)
    h_periodic[::2] = 1.0  # Alternating 1s and 0s
    initial_guesses.append(h_periodic)
    
    best_c5 = float('inf')
    best_h = initial_guesses[0].copy()
    
    start_time = time.time()
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Scale to ensure correct sum
        current_sum = np.sum(guess)
        scale = target_sum / current_sum
        guess_scaled = guess * scale
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds and constraints
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization
        result = minimize(
            objective,
            guess_scaled,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 1500}
        )
        
        # Update best result
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
