        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # If no optimization was done, use the initial guess
    if best_h is None:
        # Use one of the initial guesses
        best_h = initial_guesses[0]
        best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
