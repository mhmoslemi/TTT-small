from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    dx = 2.0 / n_points

    x_initial = np.linspace(0, 2, len(initial_h_values))

    # Adjust sum to meet the constraint
    initial_guess = initial_guess / np.sum(initial_guess) * (n_points / 2)
    
    # Generate a binary initial guess
    binary_guess = np.zeros(n_points)
    binary_guess[::2] = 1.0  # Alternate 1 and 0
    binary_guess = binary_guess / np.sum(binary_guess) * (n_points / 2)
    
    # Generate perturbed initial guesses
    perturbed_guesses = []
    for _ in range(3):  # Fewer perturbations for efficiency
        noise_amp = np.random.uniform(0.001, 0.05)
        noise_freq = np.random.uniform(1, 10)
        noise = noise_amp * np.sin(2 * np.pi * noise_freq * x_new)
        random_noise = np.random.normal(0, 0.01, n_points)
        perturbed = initial_guess + noise + random_noise
        perturbed = np.clip(perturbed, 0, 1)
        scale_factor = (n_points / 2) / np.sum(perturbed)
        perturbed *= scale_factor
        perturbed_guesses.append(perturbed)
    
    # Combine all initial guesses
    initial_guesses = [initial_guess, binary_guess] + perturbed_guesses
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Run optimizations for each initial guess with tighter tolerances
    for guess in initial_guesses:
        # Run optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-9, 'maxiter': 1500}  # Increased maxiter for better convergence
        )
        elapsed = time.time() - start_time
