import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Optimal balance between speed and precision
    dx = 2.0 / n_points
    
    # Use initial_h_values as base and scale to meet the constraint
    initial_guess = np.array(initial_h_values)
    # Adjust sum to meet the constraint
    initial_guess = initial_guess / np.sum(initial_guess) * (n_points / 2)
    
    # Objective function
    def objective(h):
        # Scale h to meet the constraint
        h_scaled = h * (n_points / 2) / np.sum(h)
        # Compute cross-correlation
        one_minus_h = 1.0 - h_scaled
        corr = np.correlate(h_scaled, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    
    # Run differential evolution with initial guess
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.8,
        tol=1e-6,
        maxiter=100,
        polish=True,
        disp=False,
        x0=initial_guess
    )
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        # Fallback to initial guess if time runs out
        best_h = initial_guess
        best_c5 = objective(initial_guess)
    else:
        best_h = result.x
        best_c5 = result.fun
    
    return best_h, best_c5, n_points
