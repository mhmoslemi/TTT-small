import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Create a structured initial guess with three intervals
    n_points = 100
    dx = 2.0 / n_points
    initial_h = np.zeros(n_points)
    interval_length = 1.0 / 3.0
    interval1_start = 0.0
    interval1_end = interval_length
    interval2_start = 0.5
    interval2_end = interval2_start + interval_length
    interval3_start = 1.0
    interval3_end = interval3_start + interval_length

    for i in range(n_points):
        x = i * dx
        if interval1_start <= x <= interval1_end:
            initial_h[i] = 1.0
        elif interval2_start <= x <= interval2_end:
            initial_h[i] = 1.0
        elif interval3_start <= x <= interval3_end:
            initial_h[i] = 1.0

    best_c5 = float('inf')
    best_h = initial_h.copy()
    start_time = time.time()

    # Perturbation strategy: explore with varying noise scales
    num_perturbations = 150
    for perturbation_idx in range(num_perturbations):
        if perturbation_idx < 20:
            # Large random perturbations
            noise = np.random.uniform(-0.3, 0.3, n_points)
        elif perturbation_idx < 50:
            # Medium random perturbations
            noise = np.random.uniform(-0.15, 0.15, n_points)
        else:
            # Small random perturbations
            noise = np.random.uniform(-0.05, 0.05, n_points)
        
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = (n_points / 2) / current_sum
        perturbed = perturbed * scale

        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx

        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - (n_points / 2)})

        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 1500}
        )

        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x

        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

    return best_h, best_c5, n_points
