            else:
                h[i] = 0.0

        scale = (n_points / 2) / current_sum
        return h * scale
    
    def generate_three_segment_pattern(n_points):
        h = np.zeros(n_points)
        for i in range(n_points // 3):
            h[i] = 1.0
        for i in range(2 * n_points // 3, n_points):

        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(

            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 1500}
        )

    # Use parallel processing
    with ProcessPoolExecutor(max_workers=10) as executor:
