import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    np.random.seed(seed)
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate diverse initial guesses
    initial_guesses = [initial_guess]
    for _ in range(5):
        h = np.random.uniform(0, 1, n_points)
        h_sum = np.sum(h)
        scale = target_sum / h_sum
        h = h * scale
        h = np.clip(h, 0, 1)
        initial_guesses.append(h)
    
    # Function to compute C5
    def compute_c5(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    # Optimization parameters
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    
    # Evaluate initial guesses
    for guess in initial_guesses:
        result = minimize(
            compute_c5,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 1000}
        )
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Perturb best_h and refine
    elapsed = time.time() - start_time
    if elapsed < budget_s:
        for _ in range(50):
            noise = np.random.uniform(-0.05, 0.05, n_points)
            perturbed = best_h + noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
            
            result = minimize(
                compute_c5,
                perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'ftol': 1e-10, 'maxiter': 100}
            )
            if result.success and result.fun < best_c5:
                best_c5 = result.fun
                best_h = result.x
            
            elapsed = time.time() - start_time
            if elapsed >= budget_s:
                break
    
    return best_h, best_c5, n_points
