            initial_guess,
            method='Nelder-Mead',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'xatol': 1e-8, 'maxiter': 1000}
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points
