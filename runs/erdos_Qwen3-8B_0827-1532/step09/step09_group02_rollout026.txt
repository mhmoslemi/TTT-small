from scipy.optimize import minimize
import time

def make_symmetric(h):
    n = len(h)
    symmetric = np.zeros(n)
    for i in range(n // 2):
        symmetric[i] = h[i]
        symmetric[n - 1 - i] = h[i]
    return symmetric

def run(seed=42, budget_s=100, **kwargs):
    # Generate multiple initial guesses

    initial_guesses.append(random_initial)
    
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    
    start_time = time.time()
    
    for initial in initial_guesses:
        # Ensure initial guess is within [0,1] and has correct sum
        initial = np.clip(initial, 0, 1)
        current_sum = np.sum(initial)

        initial = initial * scale
        
        # Perturb and optimize
        best_h_current = initial.copy()
        best_c5_current = float('inf')
        
        # Perturbations
        num_perturbations = 100
        for perturbation_idx in range(num_perturbations):
            if perturbation_idx < 20:

            else:
                # Perturb best_h_current with varying noise

                perturbed = best_h_current + noise

                scale = (n_points / 2) / current_sum

            def objective(h):

            bounds = [(0.0, 1.0) for _ in range(n_points)]
