import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Use the provided initial guess and generate other initial guesses
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate other initial guesses
    # Alternating pattern
    structured_h = np.zeros(n_points)
    for i in range(n_points):
        if i % 2 == 0:
            structured_h[i] = 1.0
        else:
            structured_h[i] = 0.0
    
    # Symmetric initial guess
    symmetric_initial = np.zeros(n_points)
    num_ones = n_points // 4  # For even n_points, this gives half of half
    for i in range(num_ones):
        symmetric_initial[i] = 1.0
        symmetric_initial[n_points - 1 - i] = 1.0
    
    # Random binary initial guess
    random_binary = np.zeros(n_points)
    random_binary[:n_points//2] = 1.0
    np.random.shuffle(random_binary)
    
    initial_guesses = [initial_guess, structured_h, symmetric_initial, random_binary]
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Ensure the guess has the correct sum
        current_sum = np.sum(guess)
        scale = target_sum / current_sum
        perturbed = guess * scale
        perturbed = np.clip(perturbed, 0, 1)
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 200}
        )
        
        # Check if this result is better
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
