import time

    n_points = 100  # Balance between speed and precision
    dx = 2.0 / n_points

        max_corr = np.max(corr)
        return max_corr * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = None

    for guess in initial_guesses:
