from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 80  # Smaller for faster optimization
    dx = 2.0 / n_points
    
    # Generate initial guesses with structured patterns
    initial_guesses = []

    x_initial = np.linspace(0, 2, len(initial_h_values))

    initial_guess = np.interp(x_new, x_initial, initial_h_values)
    initial_guess = initial_guess / np.sum(initial_guess) * (n_points / 2)
    initial_guesses.append(initial_guess)
    
    # Binary pattern: alternate 1 and 0
    binary_guess = np.zeros(n_points)
    binary_guess[::2] = 1.0
    binary_guess = binary_guess / np.sum(binary_guess) * (n_points / 2)
    initial_guesses.append(binary_guess)
    
    # Every third point pattern
    every_third_guess = np.zeros(n_points)
    every_third_guess[::3] = 1.0
    every_third_guess = every_third_guess / np.sum(every_third_guess) * (n_points / 2)
    initial_guesses.append(every_third_guess)
    
    # Sine wave pattern
    sine_guess = 0.5 + 0.5 * np.sin(2 * np.pi * np.linspace(0, 1, n_points))

    initial_guesses.append(sine_guess)
    
    # Add some random noise variations

        random_noise = np.random.normal(0, 0.01, n_points)
        perturbed = initial_guess + noise + random_noise
        perturbed = np.clip(perturbed, 0, 1)
        scale_factor = (n_points / 2) / np.sum(perturbed)
        perturbed *= scale_factor
        initial_guesses.append(perturbed)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Run optimizations for each initial guess with tighter tolerances
    for guess in initial_guesses:
        # Run optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-9, 'maxiter': 1000}  # Tighter tolerance and more iterations
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # If no optimization was done, use the initial guess
    if best_h is None:
        result = minimize(
            objective,
            initial_guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-9, 'maxiter': 1000}
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points
