from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Create symmetric initial guess
    symmetric_initial = np.zeros(n_points)
    for i in range(n_points):
        symmetric_initial[i] = (initial_guess[i] + initial_guess[n_points - 1 - i]) / 2
    initial_guess = symmetric_initial
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    num_perturbations = 200  # Increased for better exploration
    
    for perturbation_idx in range(num_perturbations):
        if perturbation_idx < 20:
            # Random initial guess with larger noise
            random_noise = np.random.uniform(-0.3, 0.3, n_points)
            perturbed = initial_guess + random_noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
        else:
            # Perturb best_h with varying noise
            if perturbation_idx < 30:
                noise = np.random.uniform(-0.2, 0.2, n_points)
            else:
                noise = np.random.uniform(-0.05, 0.05, n_points)
            perturbed = best_h + noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
        
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Add symmetry constraints
        symmetry_constraints = []

        )
        
        if result.success and result.fun < best_c5:

            best_h = result.x
