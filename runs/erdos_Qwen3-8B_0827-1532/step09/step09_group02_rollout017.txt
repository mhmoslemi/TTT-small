import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    # Assume initial_h_values is provided globally
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h):
        sum_h = np.sum(h)
        if sum_h == 0:
            return float('inf')
        h_scaled = h * (target_sum / sum_h)
        one_minus_h = 1.0 - h_scaled
        corr = np.correlate(h_scaled, one_minus_h, mode='full')

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=15,
        tol=1e-12,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed
    )
    
    elapsed = time.time() - start_time
    best_h = result.x
    best_c5 = result.fun
    
    return best_h, best_c5, n_points
