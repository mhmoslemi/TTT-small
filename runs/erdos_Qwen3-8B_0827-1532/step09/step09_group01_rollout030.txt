import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Use the provided initial guess and generate a structured initial guess
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate structured initial guesses
    structured_guesses = []
    # Alternating 1s and 0s
    alt_h = np.zeros(n_points)
    for i in range(n_points):
        if i % 2 == 0:
            alt_h[i] = 1.0
        else:
            alt_h[i] = 0.0
    structured_guesses.append(alt_h)
    
    # Block pattern: 2 ones followed by 2 zeros
    block_h = np.zeros(n_points)
    for i in range(n_points):
        if i % 4 < 2:
            block_h[i] = 1.0
        else:
            block_h[i] = 0.0
    structured_guesses.append(block_h)
    
    # Initial guesses: combine initial guess and structured guesses
    initial_guesses = [initial_guess] + structured_guesses
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Optimization parameters
    max_iterations = 100
    num_initial_guesses = len(initial_guesses)
    
    for guess_idx in range(num_initial_guesses):
        # Start with the current initial guess
        h = initial_guesses[guess_idx].copy()
        
        # Ensure sum constraint
        current_sum = np.sum(h)
        scale = target_sum / current_sum
        h = h * scale
        
        # Objective function to minimize C5
        def objective(h_flat):
            h = h_flat.reshape(n_points)
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization with COBYLA method for better constraint handling
        result = minimize(
            objective,
            h.flatten(),
            method='COBYLA',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': max_iterations, 'disp': False}
        )
        
        # Check if this result is better
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x.reshape(n_points)
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
