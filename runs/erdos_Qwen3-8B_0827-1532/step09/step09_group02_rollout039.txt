import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Use the provided initial_h_values
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Parameters for perturbation and optimization
    num_perturbations = 200  # Increased for more exploration
    noise_scales = [0.2, 0.15, 0.1, 0.05, 0.02]  # Decreasing noise scales over time
    
    for perturbation_idx in range(num_perturbations):
        # Generate perturbed guess
        if perturbation_idx < len(noise_scales):
            noise_scale = noise_scales[perturbation_idx]
        else:
            noise_scale = 0.01  # Final fine-tuning
        
        if perturbation_idx < 20:
            # Random initial guess with larger noise
            random_noise = np.random.uniform(-noise_scale, noise_scale, n_points)
            perturbed = initial_guess + random_noise
        else:
            # Perturb best_h with varying noise
            noise = np.random.uniform(-noise_scale, noise_scale, n_points)
            perturbed = best_h + noise
        
        perturbed = np.clip(perturbed, 0.0, 1.0)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization settings
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Use L-BFGS-B for faster convergence
        result = minimize(
            objective,
            perturbed,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 1000}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Early stopping if time is up
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
