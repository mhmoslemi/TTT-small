import numpy as np
from scipy.optimize import minimize
from multiprocessing import Pool, cpu_count
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Balance between speed and precision
    dx = 2.0 / n_points
    
    # Generate initial guesses based on provided initial_h_values
    initial_guess = np.array(initial_h_values)
    initial_guess = initial_guess / np.sum(initial_guess) * (n_points / 2)

    num_initial_guesses = 50  # Increased for better exploration
    initial_guesses = []
    for _ in range(num_initial_guesses):
        # Add random noise with varying amplitude
        noise_amp = np.random.uniform(0.005, 0.05)
        noise = noise_amp * np.random.randn(n_points)
        perturbed = initial_guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        scale_factor = (n_points / 2) / np.sum(perturbed)
        perturbed *= scale_factor
        initial_guesses.append(perturbed)
    
    # Objective function to minimize maximum overlap
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint to ensure integral equals 1
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Function to run optimization for a single initial guess
    def optimize(guess):
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-10, 'maxiter': 3000}
        )
        return result.fun, result.x
    
    # Track best solution
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    
    # Use multiprocessing to parallelize optimizations
    with Pool(processes=min(cpu_count(), len(initial_guesses))) as pool:
        results = pool.map(optimize, initial_guesses)
    
    # Extract best result from all optimizations
    for c5, h in results:
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    
    return best_h, best_c5, n_points
