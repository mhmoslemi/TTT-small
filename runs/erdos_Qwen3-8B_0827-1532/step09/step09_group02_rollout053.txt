from scipy.optimize import minimize
import time

    dx = 2.0 / n_points
    target_sum = n_points / 2

    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    num_perturbations = 200  # Increased for better exploration
    
    for perturbation_idx in range(num_perturbations):
        if perturbation_idx < 50:
            # Large random perturbations for initial exploration
            noise = np.random.uniform(-0.3, 0.3, n_points)
        elif perturbation_idx < 150:
            # Moderate perturbations for refinement
            noise = np.random.uniform(-0.15, 0.15, n_points)
        else:
            # Small perturbations for fine-tuning
            noise = np.random.uniform(-0.05, 0.05, n_points)
        
        # Generate perturbed guess
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Optimization parameters
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 3000}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time

            break
    
    return best_h, best_c5, n_points
