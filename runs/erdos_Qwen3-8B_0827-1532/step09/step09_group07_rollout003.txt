    for i in range(n_points):
        if x_new[i] < 1.0:
            square_wave[i] = 1.0
        else:
            square_wave[i] = 0.0
    square_wave = square_wave / np.sum(square_wave) * (n_points / 2)
    initial_guesses.append(square_wave)
    # Triangular wave
    triangular_wave = np.zeros(n_points)
    for i in range(n_points):
        x = x_new[i]
        if x < 1.0:
            triangular_wave[i] = x / 1.0
        else:
            triangular_wave[i] = (2.0 - x) / 1.0
    triangular_wave = triangular_wave / np.sum(triangular_wave) * (n_points / 2)
    initial_guesses.append(triangular_wave)
    # Sine wave
    sine_wave = 0.5 + 0.5 * np.sin(np.pi * x_new)
    sine_wave = sine_wave / np.sum(sine_wave) * (n_points / 2)
    initial_guesses.append(sine_wave)
    # Sawtooth wave
    sawtooth_wave = np.zeros(n_points)
    for i in range(n_points):
        x = x_new[i]
        if x < 1.0:
            sawtooth_wave[i] = x / 1.0
        else:
            sawtooth_wave[i] = (x - 1.0) / 1.0
    sawtooth_wave = sawtooth_wave / np.sum(sawtooth_wave) * (n_points / 2)
    initial_guesses.append(sawtooth_wave)
    # Two-interval wave
    two_interval = np.zeros(n_points)
    for i in range(n_points):
        x = x_new[i]
        if 0.0 <= x < 0.5 or 1.0 <= x < 1.5:
            two_interval[i] = 1.0
        else:
            two_interval[i] = 0.0
    two_interval = two_interval / np.sum(two_interval) * (n_points / 2)
    initial_guesses.append(two_interval)
    # Three-interval wave
    three_interval = np.zeros(n_points)
    for i in range(n_points):
        x = x_new[i]
        if 0.0 <= x < 0.333 or 0.666 <= x < 1.0 or 1.333 <= x < 1.666:
            three_interval[i] = 1.0
        else:
            three_interval[i] = 0.0
    three_interval = three_interval / np.sum(three_interval) * (n_points / 2)
    initial_guesses.append(three_interval)
    
    # Add some random guesses
    for _ in range(5):  # Fewer random guesses
        random_h = np.random.uniform(0, 1, n_points)
        random_h = random_h / np.sum(random_h) * (n_points / 2)
        initial_guesses.append(random_h)
    
    # Objective function with penalty
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        penalty = 1e6 * (np.sum(h) - n_points / 2)**2
        return max_corr * dx + penalty
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Run differential evolution for each initial guess
    for guess in initial_guesses:
        # Run differential evolution
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=15,
            maxiter=100,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        current_c5 = result.fun
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x
    
    # If no optimization was done, use the initial guess
    if best_h is None:
        # Use the initial guess and optimize with DE
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=15,
            maxiter=100,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points
