from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = 200  # Increased for better precision
    dx = 2.0 / n_points
    
    # Interpolate initial_h_values to n_points
    x_initial = np.linspace(0, 2, len(initial_h_values))

    initial_guess = np.interp(x_new, x_initial, initial_h_values)
    # Adjust sum to meet the constraint
    initial_guess = initial_guess / np.sum(initial_guess) * (n_points / 2)
    
    # Generate initial guesses
    initial_guesses = [initial_guess]
    
    # Add perturbed versions
    for _ in range(5):
        noise = np.random.normal(0, 0.05, n_points)
        perturbed = initial_guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        scale_factor = (n_points / 2) / np.sum(perturbed)
        perturbed *= scale_factor
        initial_guesses.append(perturbed)
    
    # Add shifted and reversed versions
    shifted = np.roll(initial_guess, 10)
    shifted = shifted / np.sum(shifted) * (n_points / 2)
    initial_guesses.append(shifted)
    reversed_guess = initial_guess[::-1]
    reversed_guess = reversed_guess / np.sum(reversed_guess) * (n_points / 2)
    initial_guesses.append(reversed_guess)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Run optimizations for each initial guess
    for guess in initial_guesses:
        # Run optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-12, 'maxiter': 1000}
        )
        elapsed = time.time() - start_time
