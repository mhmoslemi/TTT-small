    binary_h = np.zeros(n_points)
    indices = np.random.choice(n_points, size=int(target_sum), replace=False)
    binary_h[indices] = 1.0
    
    # Combine initial guesses
    initial_guesses = [initial_guess, structured_h, binary_h]
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate multiple diverse initial guesses based on best_h
    num_perturbations = 200  # Reduced to save time
    for perturbation_idx in range(num_perturbations):
        # Generate random initial guess
        random_h = np.random.uniform(0, 1, n_points)
        sum_random = np.sum(random_h)
        scaled_random = random_h * (target_sum / sum_random)
        
        # Define objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization
        result = minimize(
            objective,
            scaled_random,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 1000}
        )
        
        # Check if this result is better
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
