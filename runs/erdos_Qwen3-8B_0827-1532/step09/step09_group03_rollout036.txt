import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Use the provided initial guess
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate multiple diverse initial guesses
    num_initial_guesses = 50  # Adjust as needed
    for _ in range(num_initial_guesses):

        random_h = np.random.uniform(0, 1, n_points)
        random_h = random_h / np.sum(random_h) * target_sum
        # Perturb initial guess slightly for diversity

        perturbed = random_h + perturbation

        def objective(h):

            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
