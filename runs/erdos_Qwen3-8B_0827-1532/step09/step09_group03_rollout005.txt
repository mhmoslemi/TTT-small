from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Use the provided initial guess
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate multiple diverse initial guesses based on best_h and random
    num_perturbations = 200  # Reduced number for more iterations per optimization
    for perturbation_idx in range(num_perturbations):
        # 30% chance to generate a completely random initial guess
        if np.random.rand() < 0.3:
            random_guess = np.random.uniform(0, 1, n_points)
            current_sum = np.sum(random_guess)
            scale = target_sum / current_sum
            perturbed = random_guess * scale
        else:
            # Perturb best_h with varying noise
            if perturbation_idx < 10:
                # Random initial guess based on best_h with larger noise
                random_noise = np.random.uniform(-0.2, 0.2, n_points)
                perturbed = best_h + random_noise
                perturbed = np.clip(perturbed, 0, 1)
                current_sum = np.sum(perturbed)
                scale = target_sum / current_sum
                perturbed = perturbed * scale
            else:
                # Perturb best_h with smaller noise for later iterations
                noise = np.random.uniform(-0.05, 0.05, n_points)
                perturbed = best_h + noise
                perturbed = np.clip(perturbed, 0, 1)
                current_sum = np.sum(perturbed)
                scale = target_sum / current_sum
                perturbed = perturbed * scale
        
        # Objective function to minimize C5
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization with tighter tolerances and more iterations
        result = minimize(
            objective,
            perturbed,
            method='trust-constr',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 1000}
        )
        
        # Check if this result is better
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
