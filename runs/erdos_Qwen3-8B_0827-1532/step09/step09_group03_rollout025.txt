from scipy.optimize import minimize
import time

            noise = np.random.uniform(-0.2, 0.2, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function to minimize C5
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]

        # Perform optimization with trust-constr for better convergence
        result = minimize(
            objective,
            perturbed,
            method='trust-constr',
            bounds=bounds,
            constraints=cons,
            options={
