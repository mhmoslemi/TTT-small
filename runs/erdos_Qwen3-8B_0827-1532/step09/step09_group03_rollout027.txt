    initial_guesses.append(symmetric_initial)
    
    # Uniform initial guess
    uniform_initial = np.full(n_points, 0.5)
    initial_guesses.append(uniform_initial)
    
    # Alternating initial guess
    alternating_initial = np.zeros(n_points)
    for i in range(n_points):
        if i % 2 == 0:
            alternating_initial[i] = 1.0
        else:
            alternating_initial[i] = 0.0
    initial_guesses.append(alternating_initial)
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Clip and scale to meet constraints
        guess = np.clip(guess, 0.0, 1.0)
        current_sum = np.sum(guess)
        scale = target_sum / current_sum
        guess = guess * scale
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds and constraints
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 2000}
        )
        
        # Update best solution
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Check elapsed time
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
