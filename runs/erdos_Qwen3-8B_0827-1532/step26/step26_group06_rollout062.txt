        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Differential Evolution for global search
    def de_objective(h):
        return compute_c5(h)
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result_de = differential_evolution(
        de_objective,
        bounds,
        strategy='best1bin',
        popsize=15,
        maxiter=50,
        tol=1e-10,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed
    )
    
    if result_de.success and result_de.fun < best_c5:
        best_c5 = result_de.fun
        best_h = result_de.x
    
    # Perturbation phase
    num_perturbations = 100
    phase_shift = 0.15
    phase_decay = 0.96
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 300}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
