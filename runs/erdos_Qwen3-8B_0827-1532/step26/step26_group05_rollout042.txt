    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')

    initial_guess = initial_h_values

    best_c5 = compute_c5(best_h, dx)
    
    # Generate diverse initial guesses
    initial_guesses = [initial_guess]
    
    # Step function: first half 1, second half 0
    h_step = np.zeros(n_points)
    h_step[:n_points//2] = 1.0
    sum_h = np.sum(h_step)
    scale = target_sum / sum_h
    initial_guesses.append(h_step * scale)
    
    # Alternating 1 and 0
    h_alt = np.zeros(n_points)

    sum_h = np.sum(h_alt)

    h_center = np.zeros(n_points)

    h_two = np.zeros(n_points)

    h_three[2*segment_length:] = 1.0

    h_three_intervals = np.zeros(n_points)

    initial_guesses.append(np.random.uniform(0, 1, n_points))

    for _ in range(5):
        h_rand = np.random.uniform(0, 1, n_points)
        sum_h = np.sum(h_rand)
        scale = target_sum / sum_h
        initial_guesses.append(h_rand * scale)
    
    # Four-segment pattern
    h_four = np.zeros(n_points)
    segment_length = n_points // 4

    h_four[2*segment_length:3*segment_length] = 1.0

    h_five[2*segment_length:3*segment_length] = 1.0

    h_five[4*segment_length:] = 1.0

    scale = target_sum / sum_h
    initial_guesses.append(h_five * scale)
    
    # Two-segment spaced apart
    h_two_spaced = np.zeros(n_points)
    segment_length = n_points // 4
    h_two_spaced[:segment_length] = 1.0
    h_two_spaced[n_points//2:n_points//2 + segment_length] = 1.0
    sum_h = np.sum(h_two_spaced)
    scale = target_sum / sum_h
    initial_guesses.append(h_two_spaced * scale)
    
    # Three-segment spaced apart
    h_three_spaced = np.zeros(n_points)
    segment_length = n_points // 4

    h_three_spaced[n_points//2:n_points//2 + segment_length] = 1.0
    h_three_spaced[2*n_points//2:2*n_points//2 + segment_length] = 1.0
    sum_h = np.sum(h_three_spaced)
    scale = target_sum / sum_h
    initial_guesses.append(h_three_spaced * scale)
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with COBYLA
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='COBYLA',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 1000, 'disp': False}
        )
        
        if result.success and result.fun < best_c5:

            best_h = result.x
    
    # Perturbation phase on the best_h found so far
    num_perturbations = 300
    phase_shift = 0.15
    phase_decay = 0.96
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with COBYLA
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,

            method='COBYLA',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 800, 'disp': False}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points

def compute_c5(h, dx):
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return np.max(corr) * dx
