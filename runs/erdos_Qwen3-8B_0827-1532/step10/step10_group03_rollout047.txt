from scipy.optimize import minimize
import time

    n_points = len(initial_guess)

    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Define objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define constraint function
    def constraint(h):
        return np.sum(h) - target_sum
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial guesses
    initial_guesses = [initial_guess]
    # Add a few structured initial guesses
    initial_guesses.append(np.full(n_points, 0.5))  # Uniform
    initial_guesses.append(np.concatenate([np.ones(n_points//2), np.zeros(n_points//2)]))
    initial_guesses[-1] = initial_guesses[-1][:n_points]
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Use SLSQP with tighter tolerances
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-10, 'maxiter': 500}
        )
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Generate perturbations based on best_h
    num_perturbations = 50  # Reduced for efficiency
    for perturbation_idx in range(num_perturbations):
        if perturbation_idx < 10:

            perturbed = best_h + noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
        else:
            # Small noise
            noise = np.random.uniform(-0.05, 0.05, n_points)
            perturbed = best_h + noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum

        result = minimize(
