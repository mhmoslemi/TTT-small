from scipy.optimize import minimize
import time

    target_sum = n_points / 2
    
    # Generate diverse initial guesses

    initial_guesses.append(initial_h_values.copy())
    # Binary pattern: first half is 1, second half is 0
    binary_initial = np.zeros(n_points)
    binary_initial[:n_points//2] = 1.0
    initial_guesses.append(binary_initial)
    # Sine wave pattern
    x = np.linspace(0, 2, n_points)
    sine_initial = 0.5 * (1 + np.sin(np.pi * x))
    initial_guesses.append(sine_initial)
    # Random initial guess
    random_initial = np.random.rand(n_points)
    random_initial = random_initial / np.sum(random_initial) * target_sum
    initial_guesses.append(random_initial)
    # Another random pattern
    another_random_initial = np.random.rand(n_points)
    another_random_initial = another_random_initial / np.sum(another_random_initial) * target_sum
    initial_guesses.append(another_random_initial)
    
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    
    start_time = time.time()
    
    # Parameters for optimization
    for guess in initial_guesses:
        # Ensure the guess is within [0,1] and scaled correctly
        guess = np.clip(guess, 0, 1)
        current_sum = np.sum(guess)
        scale = target_sum / current_sum if current_sum != 0 else 1.0
        perturbed = guess * scale
        
        # Objective function to minimize C5
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization with tighter tolerances
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-8, 'maxiter': 200}
        )
        
        # Check if this result is better
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
