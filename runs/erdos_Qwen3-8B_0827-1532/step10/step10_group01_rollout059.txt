    for i in range(n_points):
        if i % 2 == 0:
            periodic_guess[i] = 1.0
    periodic_guess = periodic_guess * (target_sum / np.sum(periodic_guess))
    initial_guesses.append(periodic_guess)
    
    # Symmetric guess: 1 on first half, 0 on second half
    symmetric_guess = np.zeros(n_points)
    symmetric_guess[:n_points//2] = 1.0
    symmetric_guess = symmetric_guess * (target_sum / np.sum(symmetric_guess))
    initial_guesses.append(symmetric_guess)
    
    # Evaluate each initial guess
    for guess in initial_guesses:
        # Ensure the guess is valid
        guess = np.clip(guess, 0, 1)
        guess = guess * (target_sum / np.sum(guess))
        # Run optimization
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 1000}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Additional perturbations to refine the solution
    for _ in range(50):
        if time.time() - start_time >= budget_s:
            break
        # Perturb the best_h with varying noise
        noise = np.random.uniform(-0.05, 0.05, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 1000}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    return best_h, best_c5, n_points
