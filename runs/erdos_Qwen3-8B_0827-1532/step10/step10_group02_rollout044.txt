from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    start_time = time.time()
    
    num_perturbations = 300  # Increased for better exploration
    
    for perturbation_idx in range(num_perturbations):
        if perturbation_idx < 50:
            # Large noise for broad exploration
            noise_scale = 0.3
        elif perturbation_idx < 150:
            # Medium noise for local refinement
            noise_scale = 0.15
        else:
            # Small noise for fine-tuning
            noise_scale = 0.05
        
        # Generate perturbation
        if perturbation_idx < 20:
            # Random initial guess
            random_noise = np.random.uniform(-noise_scale, noise_scale, n_points)
            perturbed = initial_guess + random_noise
        else:
            # Perturb best_h with varying noise
            noise = np.random.uniform(-noise_scale, noise_scale, n_points)
            perturbed = best_h + noise
        
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(

            perturbed,
            method='SLSQP',

        )
        
        if result.success and result.fun < best_c5:

            best_h = result.x
