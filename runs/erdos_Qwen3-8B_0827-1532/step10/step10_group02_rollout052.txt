import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Use initial_h_values as the starting point
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate multiple initial guesses with varying noise
    num_perturbations = 300  # Increased for better exploration
    noise_magnitudes = [0.2, 0.1, 0.05]
    perturbation_phases = np.linspace(0, 1, num_perturbations)
    
    for perturbation_idx in range(num_perturbations):
        # Vary noise magnitude based on phase
        noise_mag = noise_magnitudes[int(perturbation_idx / 100)]
        # Add noise with phase to diversify perturbations
        noise = np.random.uniform(-noise_mag, noise_mag, n_points) * np.sin(2 * np.pi * perturbation_phases[perturbation_idx])
        perturbed = initial_guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Define objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization constraints
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Use L-BFGS-B for faster convergence with bounds
        result = minimize(
            objective,
            perturbed,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 200}
        )
        
        # Update best solution if improvement is found
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
