import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    # Precompute initial guesses
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate initial guesses
    initial_guesses = [initial_h_values.copy()]
    
    # Evenly spaced binary pattern
    num_ones = n_points // 2
    evenly_spaced = np.zeros(n_points)
    for i in range(num_ones):
        pos = i * (2.0 / (n_points - 1))  # spacing between consecutive ones
        idx = int(pos / dx)
        evenly_spaced[idx] = 1.0
    initial_guesses.append(evenly_spaced)
    
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    
    start_time = time.time()
    
    # Parameters for optimization
    for guess in initial_guesses:
        # Scale initial guess to meet the sum constraint
        current_sum = np.sum(guess)
        scale = target_sum / current_sum
        scaled_guess = guess * scale
        scaled_guess = np.clip(scaled_guess, 0, 1)
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Gradient function
        def gradient(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr_index = np.argmax(corr)
            n = len(h)
            k_index = max_corr_index - n + 1
            k = k_index * dx
            
            grad = np.zeros_like(h)
            for i in range(n):
                x = i * dx
                x_plus_k = x + k
                if 0 <= x_plus_k <= 2:
                    j = int(x_plus_k / dx)
                    if 0 <= j < n:
                        grad[i] = (1 - h[j]) * dx
            return grad
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization
        result = minimize(
            objective,
            scaled_guess,
            method='SLSQP',
            jac=gradient,
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 1000}
        )
        
        # Check if this result is better
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
