import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    
    start_time = time.time()
    
    # Generate multiple initial guesses
    initial_guesses = [initial_h_values]
    
    # Add a random initial guess
    random_guess = np.random.uniform(0, 1, n_points)
    random_guess = random_guess * (target_sum / np.sum(random_guess))
    initial_guesses.append(random_guess)
    
    # Add a symmetric initial guess (mirror the initial_h_values)
    symmetric_guess = np.zeros(n_points)
    for i in range(n_points):
        symmetric_guess[i] = initial_h_values[i] if i < n_points//2 else initial_h_values[n_points - 1 - i]
    symmetric_guess = symmetric_guess * (target_sum / np.sum(symmetric_guess))
    initial_guesses.append(symmetric_guess)
    
    # Add a binary initial guess with 1s and 0s
    binary_guess = np.zeros(n_points)
    num_ones = int(np.round(target_sum))
    indices = np.random.choice(n_points, num_ones, replace=False)
    binary_guess[indices] = 1.0
    binary_guess = binary_guess * (target_sum / np.sum(binary_guess))
    initial_guesses.append(binary_guess)
    
    # Add a periodic initial guess with different period
    periodic_guess = np.zeros(n_points)
    for i in range(n_points):
        if i % 3 == 0:
            periodic_guess[i] = 1.0
    periodic_guess = periodic_guess * (target_sum / np.sum(periodic_guess))
    initial_guesses.append(periodic_guess)
    
    # Evaluate each initial guess
    for guess in initial_guesses:
        # Ensure the guess is valid
        guess = np.clip(guess, 0, 1)
        guess = guess * (target_sum / np.sum(guess))
        # Run optimization
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 1500}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Additional perturbations to refine the solution
    for _ in range(50):
        if time.time() - start_time >= budget_s:
            break
        # Perturb the best_h with varying noise
        noise = np.random.uniform(-0.05, 0.05, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 1500}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    return best_h, best_c5, n_points
