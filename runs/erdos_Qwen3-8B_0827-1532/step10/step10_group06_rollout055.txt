from scipy.optimize import minimize
import time

    initial_guesses.append(binary_initial)
    
    # Add a structured variation: triangular wave
    x = np.linspace(0, 2, n_points)
    triangle_initial = 0.5 * (1 + np.abs(2 * x - 1) - 1)
    triangle_initial = np.clip(triangle_initial, 0, 1)
    initial_guesses.append(triangle_initial)
    
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    
    start_time = time.time()
    
    # Parameters for optimization
    num_perturbations = 50  # Reduced to save time
    for guess in initial_guesses:
        # Perturb the guess with structured noise
        for perturbation_idx in range(num_perturbations):
            # Vary noise magnitude based on perturbation index
            if perturbation_idx < 10:
                noise = np.random.uniform(-0.15, 0.15, n_points)
            elif perturbation_idx < 30:
                noise = np.random.uniform(-0.05, 0.05, n_points)
            else:
                noise = np.random.uniform(-0.02, 0.02, n_points)
            
            perturbed = guess + noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
            
            # Objective function to minimize C5
            def objective(h):
                one_minus_h = 1.0 - h
                corr = np.correlate(h, one_minus_h, mode='full')
                max_corr = np.max(corr)
                return max_corr * dx
            
            # Bounds for each h[i]
            bounds = [(0.0, 1.0) for _ in range(n_points)]
            
            # Constraints: sum(h) == target_sum
            cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
            
            # Perform optimization with tighter tolerances
            result = minimize(
                objective,

                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'ftol': 1e-10, 'maxiter': 1000}
