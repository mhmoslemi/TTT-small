import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Create symmetric initial guess
    symmetric_initial = np.zeros(n_points)
    half = n_points // 2
    for i in range(half):
        symmetric_initial[i] = initial_guess[i]
        symmetric_initial[n_points - 1 - i] = initial_guess[i]
    if n_points % 2 == 1:
        symmetric_initial[half] = initial_guess[half]
    best_h = symmetric_initial.copy()
    best_c5 = compute_c5(best_h)
    
    def compute_c5(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    start_time = time.time()
    
    # Perform multiple optimizations starting from the best solution
    num_iterations = 30  # Reduced for time efficiency
    for _ in range(num_iterations):
        # Define objective function
        def objective(h):
            return compute_c5(h)
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization
        result = minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-8, 'maxiter': 200}
        )
        
        # Check if optimization was successful
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
