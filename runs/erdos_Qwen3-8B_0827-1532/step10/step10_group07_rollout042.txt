import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Use the provided initial guess
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Function to compute C5
    def compute_c5(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Function to generate random initial guess with 50 ones and 50 zeros
    def generate_random_initial():
        indices = np.random.choice(n_points, size=n_points//2, replace=False)
        h = np.zeros(n_points)
        h[indices] = 1.0
        return h
    
    # Initial guesses: include the provided one and 5 random ones
    initial_guesses = [initial_guess.copy()]
    for _ in range(5):
        initial_guesses.append(generate_random_initial())
    
    # Shuffle to avoid bias
    np.random.shuffle(initial_guesses)
    
    best_h = initial_guesses[0].copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Optimization parameters
    max_iterations = 10  # Reduce to save time
    for h_initial in initial_guesses:
        # Define objective function
        def objective(h):
            return compute_c5(h)
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization
        result = minimize(
            objective,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': max_iterations}
        )
        
        # Check if optimization was successful
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
