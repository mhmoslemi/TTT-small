from scipy.optimize import minimize

            if i % period == 0:
                h[i] = 1.0
        current_sum = np.sum(h)
        scale = target_sum / current_sum
        h = h * scale
        return h
    
    # Function to generate random initial guess

        return h

    ]

    # Run optimization for each initial guess
    for guess in initial_guesses:

        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization with tighter tolerances and more iterations
