            perturbed,
            method='COBYLA',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 500, 'disp': False}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    return best_h, best_c5, n_points
