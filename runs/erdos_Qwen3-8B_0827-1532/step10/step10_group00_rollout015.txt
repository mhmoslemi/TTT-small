from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    initial_guess = initial_h_values

        if i % 3 == 0:
            structured_h2[i] = 1.0
        else:
            structured_h2[i] = 0.0

    for i in range(n_points):
        if i % 4 == 0:
            structured_h3[i] = 1.0
