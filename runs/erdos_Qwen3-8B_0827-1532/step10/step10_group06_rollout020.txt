from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    random_initial = random_initial / np.sum(random_initial) * target_sum
    initial_guesses.append(random_initial)

    start_time = time.time()

        def objective(h):

        result = minimize(

            method='SLSQP',

            constraints=cons,

        )
