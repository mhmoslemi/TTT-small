from scipy.optimize import differential_evolution, minimize

def run(seed=42, budget_s=100, **kwargs):
    initial_guess = initial_h_values

    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    best_c5 = float('inf')

    start_time = time.time()

    def objective(h):

        max_corr = np.max(corr)
        penalty = 1e4 * (np.sum(h) - target_sum)**2
        return max_corr * dx + penalty
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Run differential evolution with initial guess
    result_de = differential_evolution(
        objective,
        bounds,

        maxiter=100,
        tol=1e-10,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed
    )
    
    if result_de.success:
        best_h = result_de.x
        best_c5 = objective(best_h)
