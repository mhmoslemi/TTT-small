        if result.success:
            current_h = result.x
            current_c5 = compute_c5(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
    
    # Local refinement with SLSQP
    def objective_slsqp(h):
        return compute_c5(h)
    
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]
    result_slsqp = minimize(
        objective_slsqp,
        best_h,
        method='SLSQP',
        bounds=bounds_slsqp,
        constraints=cons,
        options={'ftol': 1e-10, 'maxiter': 1000}
    )
    if result_slsqp.success:
        best_h = result_slsqp.x
        best_c5 = result_slsqp.fun
    
    elapsed = time.time() - start_time
    return best_h, best_c5, n_points
