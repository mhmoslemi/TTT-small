    triangular_initial = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points//2:

        else:
            triangular_initial[i] = 1 - (i - n_points//2) / (n_points//2)
    initial_guesses.append(triangular_initial)
    # Randomized piecewise constant pattern
    piecewise_initial = np.zeros(n_points)
    for i in range(n_points):
        piecewise_initial[i] = 0.5 if np.random.rand() < 0.5 else 1.0
    initial_guesses.append(piecewise_initial)
    
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    
    start_time = time.time()
    
    # Parameters for perturbations
    num_perturbations_per_guess = 50  # Reduced to save time

        # Perturb the guess with varying noise
        for perturbation_idx in range(num_perturbations_per_guess):
            # Vary noise magnitude based on perturbation index
            if perturbation_idx < 10:
                noise = np.random.uniform(-0.3, 0.3, n_points)
            elif perturbation_idx < 30:
                noise = np.random.uniform(-0.15, 0.15, n_points)
            else:
                noise = np.random.uniform(-0.05, 0.05, n_points)
            
            perturbed = guess + noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
            
            # Objective function to minimize C5
            def objective(h):
                one_minus_h = 1.0 - h
                corr = np.correlate(h, one_minus_h, mode='full')
                max_corr = np.max(corr)
                return max_corr * dx
            
            # Bounds for each h[i]
            bounds = [(0.0, 1.0) for _ in range(n_points)]
            
            # Constraints: sum(h) == target_sum
            cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

            result = minimize(
