import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Access initial_h_values from global scope
    initial_h_values = evaluate_erdos_solution()  # Assuming this is defined globally
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    
    start_time = time.time()
    
    # Generate multiple initial guesses
    initial_guesses = [
        initial_h_values,
        np.full(n_points, 0.5),
        np.concatenate([np.ones(n_points//2), np.zeros(n_points//2)]),
        np.random.rand(n_points) * 0.5 + 0.25  # Randomized initial guess
    ]
    
    # Function to evaluate the objective
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    
    # Perturbation parameters
    num_perturbations = 250
    noise_magnitude = [0.2, 0.1, 0.05, 0.02]
    
    for perturbation_idx in range(num_perturbations):
        # Select initial guess
        if perturbation_idx % 4 == 0:
            h_initial = initial_guesses[0]  # Original initial guess
        elif perturbation_idx % 4 == 1:
            h_initial = initial_guesses[1]  # Uniform guess
        elif perturbation_idx % 4 == 2:
            h_initial = initial_guesses[2]  # Binary guess
        else:
            h_initial = initial_guesses[3]  # Randomized guess
        
        # Apply noise with varying magnitude
        noise = np.random.uniform(-noise_magnitude[perturbation_idx % 4], noise_magnitude[perturbation_idx % 4], n_points)
        perturbed = h_initial + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Optimize
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 1000}
        )
        
        # Update best solution
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check time budget
