import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate diverse initial guesses
    np.random.seed(seed)
    initial_guesses = [initial_guess]
    for _ in range(5):
        random_guess = np.random.uniform(0, 1, n_points)
        random_guess = random_guess * (target_sum / np.sum(random_guess))
        initial_guesses.append(random_guess)
    
    # Perturbation parameters
    num_perturbations = 200
    noise_range = [0.2, 0.05]
    
    for perturbation_idx in range(num_perturbations):
        # Generate diverse perturbations
        if perturbation_idx < 20:
            noise = np.random.uniform(-noise_range[0], noise_range[0], n_points)
        else:
            noise = np.random.uniform(-noise_range[1], noise_range[1], n_points)
        
        # Create perturbed guess
        perturbed = np.clip(initial_guess + noise, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Define objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Use COBYLA for better exploration
        result = minimize(
            objective,
            perturbed,
            method='COBYLA',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 1000, 'disp': False}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
