from scipy.optimize import minimize

    # Periodic guess (alternating 1 and 0)

    for i in range(n_points):
        if i % 2 == 0:

    periodic_guess = periodic_guess * (target_sum / np.sum(periodic_guess))
    initial_guesses.append(periodic_guess)

        def objective(h):
