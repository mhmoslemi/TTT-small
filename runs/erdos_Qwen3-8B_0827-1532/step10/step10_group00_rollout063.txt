import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Use the provided initial guess and generate diverse initial guesses
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate a few diverse initial guesses
    initial_guesses = [initial_guess]
    
    # Random binary initial guess
    np.random.seed(seed)
    random_binary = np.zeros(n_points)
    indices = np.random.choice(n_points, size=int(n_points // 2), replace=False)
    random_binary[indices] = 1.0
    initial_guesses.append(random_binary)
    
    # Periodic patterns
    def generate_periodic_pattern(period):
        pattern = np.zeros(n_points)
        for i in range(n_points):
            if (i % period) == 0:
                pattern[i] = 1.0
        # Adjust to meet the sum constraint
        current_sum = np.sum(pattern)
        scale = target_sum / current_sum
        return pattern * scale
    
    initial_guesses.append(generate_periodic_pattern(3))
    initial_guesses.append(generate_periodic_pattern(4))
    initial_guesses.append(generate_periodic_pattern(5))
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Objective function to minimize C5
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization with tighter tolerances and more iterations
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 1500}
        )
        
        # Check if this result is better
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
