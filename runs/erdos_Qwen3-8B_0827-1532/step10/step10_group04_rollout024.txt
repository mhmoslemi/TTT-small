        generate_periodic_guess(2),
        generate_periodic_guess(3),
        generate_periodic_guess(4),
        generate_periodic_guess(5),
        generate_random_guess(),
    ]
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Run optimization for each initial guess
    for guess in initial_guesses:
        # Objective function to minimize C5
        def objective(h):
            max_val = 0.0
            for m in range(n_points):
                # Compute the integral for shift m
                # The valid indices for i are 0 to n_points - m - 1
                # So the slice is h[0 : n_points - m] and h[m : n_points]
                integral = np.dot(h[:n_points - m], 1 - h[m:]) * dx
                if integral > max_val:
                    max_val = integral
            return max_val
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization with tighter tolerances and more iterations
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-12, 'maxiter': 2000}
        )
        
        # Check if this result is better
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
