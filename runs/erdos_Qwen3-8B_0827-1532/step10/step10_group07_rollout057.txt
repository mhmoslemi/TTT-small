from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Predefined initial guess
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Function to compute C5
    def compute_c5(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Generate multiple initial guesses
    np.random.seed(seed)
    initial_guesses = [initial_guess]
    for _ in range(4):  # Add 4 random initial guesses
        h = np.random.rand(n_points)
        h = h * (target_sum / np.sum(h))
        initial_guesses.append(h)
    
    best_h = initial_guess.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Optimization parameters
    max_iter_per_run = 200
    for guess in initial_guesses:
        # Define objective function
        def objective(h):
            return compute_c5(h)
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': max_iter_per_run}
        )
        
        # Check if optimization was successful
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check elapsed time
