import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Use the provided initial guess
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Function to compute C5
    def compute_c5(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Generate multiple initial guesses
    initial_guesses = [initial_guess]
    # Generate 4 random binary initial guesses
    for _ in range(4):
        binary = np.random.permutation([1]*(n_points//2) + [0]*(n_points//2))
        initial_guesses.append(binary)
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    # Run optimization for each initial guess
    for guess in initial_guesses:
        # Ensure the guess is a numpy array
        guess = np.array(guess, dtype=float)
        # Define objective function
        def objective(h):
            return compute_c5(h)
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]

            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 1000}
        )
        
        # Check if optimization was successful

        for _ in range(num_iterations):
