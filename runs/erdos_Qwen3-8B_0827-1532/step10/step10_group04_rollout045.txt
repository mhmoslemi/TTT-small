    target_sum = n_points / 2
    
    def generate_periodic_guess(period):
        h = np.zeros(n_points)
        for i in range(n_points):
            if i % period == 0:
                h[i] = 1.0
        current_sum = np.sum(h)
        scale = target_sum / current_sum
        h = h * scale
        return h
    
    def generate_random_guess():
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        return h
    
    initial_guesses = [
        initial_guess,
        generate_periodic_guess(2),
        generate_periodic_guess(3),
        generate_periodic_guess(4),
        generate_periodic_guess(5),
        generate_random_guess(),
    ]
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    start_time = time.time()
    
    for guess in initial_guesses:
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            penalty = 1e6 * (np.sum(h) - target_sum)**2
            return max_corr * dx + penalty
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=100,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            disp=False
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
