import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Define objective function with penalty for sum constraint
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        sum_h = np.sum(h)
        penalty = 1e3 * (sum_h - target_sum) ** 2
        return max_corr * dx + penalty
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Run multiple differential evolution instances with different initial populations
    for _ in range(3):  # Run 3 independent differential evolution trials
        # Generate a random initial population based on the current best_h
        if _ == 0:
            # Use initial guess as starting point
            initial_pop = np.array([initial_guess])
        else:
            # Add noise to the current best_h for diversity
            noise = np.random.uniform(-0.1, 0.1, n_points)
            initial_pop = np.array([best_h + noise])
        
        # Ensure initial population is within bounds
        initial_pop = np.clip(initial_pop, 0, 1)
        
        # Run differential evolution
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=15,
            mutation=(0.5, 1),
            recombination=0.8,
            tol=1e-6,
            maxiter=100,
            disp=False,
            x0=initial_pop
        )
        
        # Update best solution
        current_c5 = objective(result.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
