from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    # Binary pattern: first half is 1, second half is 0

    initial_guesses.append(binary_initial)
    # Sine wave pattern
    x = np.linspace(0, 2, n_points)
    sine_initial = 0.5 * (1 + np.sin(np.pi * x))
    initial_guesses.append(sine_initial)
    # Triangular wave pattern
    triangular_initial = 0.5 * (1 + np.abs(2 * x - 1) - 1)
    triangular_initial = np.clip(triangular_initial, 0, 1)
    initial_guesses.append(triangular_initial)
    
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    
    start_time = time.time()
    
    # Parameters for perturbations
    num_perturbations_per_guess = 50  # Reduce number to save time
    for guess_idx, guess in enumerate(initial_guesses):
        # Perturb the guess with varying noise
        for perturbation_idx in range(num_perturbations_per_guess):
            # Vary noise magnitude based on perturbation index and guess index
            if guess_idx == 0:  # Focus more on initial guess
                if perturbation_idx < 30:
                    noise = np.random.uniform(-0.2, 0.2, n_points)
                elif perturbation_idx < 60:
                    noise = np.random.uniform(-0.1, 0.1, n_points)
                else:
                    noise = np.random.uniform(-0.05, 0.05, n_points)
            else:
                if perturbation_idx < 10:
                    noise = np.random.uniform(-0.2, 0.2, n_points)
                elif perturbation_idx < 30:
                    noise = np.random.uniform(-0.1, 0.1, n_points)
                else:
                    noise = np.random.uniform(-0.05, 0.05, n_points)
            
            perturbed = guess + noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
            
            # Objective function to minimize C5
            def objective(h):
                one_minus_h = 1.0 - h
                corr = np.correlate(h, one_minus_h, mode='full')
                max_corr = np.max(corr)
                return max_corr * dx
            
            # Bounds for each h[i]
            bounds = [(0.0, 1.0) for _ in range(n_points)]
            
            # Constraints: sum(h) == target_sum
            cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
            
            # Perform optimization with tighter tolerances
            result = minimize(
                objective,
                perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'ftol': 1e-10, 'maxiter': 1500}
            )

            if result.success and result.fun < best_c5:
                best_c5 = result.fun
                best_h = result.x
            
            # Check elapsed time
            elapsed = time.time() - start_time
            if elapsed >= budget_s:
                break
