from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    target_sum = n_points / 2
    
    # Structured initial guess: every other interval is 1
    structured_initial = np.zeros(n_points)
    structured_initial[::2] = 1.0
    initial_guess = structured_initial.copy()
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    num_perturbations = 300  # Increased number of perturbations
    
    for perturbation_idx in range(num_perturbations):
        # Varying noise magnitude for different perturbations
        if perturbation_idx < 50:
            # Large noise for initial exploration
            noise = np.random.uniform(-0.3, 0.3, n_points)
        elif perturbation_idx < 150:
            # Medium noise for exploration
            noise = np.random.uniform(-0.15, 0.15, n_points)

            # Small noise for fine-tuning
            noise = np.random.uniform(-0.05, 0.05, n_points)
        
        perturbed = initial_guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
