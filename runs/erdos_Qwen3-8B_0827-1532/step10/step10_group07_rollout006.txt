from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Generate a better initial guess with random binary values

    n_points = 200  # Increased points for better precision
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate initial guess with half 1s and half 0s
    initial_guess = np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])
    initial_guess = initial_guess.astype(float)
    # Ensure the sum is exactly target_sum (for even n_points)
    if np.sum(initial_guess) != target_sum:
        # Adjust one value if necessary
        diff = target_sum - np.sum(initial_guess)

            initial_guess[np.argmin(initial_guess)] = 1
        else:
            initial_guess[np.argmax(initial_guess)] = 0
    
    # Function to compute C5
    def compute_c5(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Initial best solution
    best_h = initial_guess.copy()
    best_c5 = compute_c5(best_h)
    
    start_time = time.time()
    
    # Perform multiple optimizations starting from the best solution
    num_iterations = 30  # Reduced iterations for speed
    for _ in range(num_iterations):
        # Define objective function
        def objective(h):
            return compute_c5(h)
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization
        result = minimize(
