        h = np.zeros(n_points)
        for i in range(n_points):
            if i % period == 0:
                h[i] = 1.0
        current_sum = np.sum(h)
        scale = target_sum / current_sum
        h = h * scale
        return h
    
    # Function to generate random initial guess
    def generate_random_guess():
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        return h
    
    # Generate initial guesses
    initial_guesses = [
        initial_guess,
        generate_periodic_guess(2),
        generate_periodic_guess(3),
        generate_periodic_guess(4),
        generate_periodic_guess(5),
        generate_random_guess(),
    ]
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Run optimization for each initial guess
    for guess in initial_guesses:
        # Objective function to minimize C5
        def objective(h):
            nonlocal dx, n_points, target_sum
            # Compute C5 for shifts in [0, 1]
            max_c5 = 0.0
            for m in range(int(n_points / 2) + 1):
                shifted_h = np.zeros_like(h)
                for i in range(n_points):
                    if i + m < n_points:
                        shifted_h[i] = h[i + m]
                    else:
                        shifted_h[i] = 0.0
                integral = np.dot(h, 1.0 - shifted_h) * dx
                if integral > max_c5:
                    max_c5 = integral
            return max_c5
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization with tighter tolerances and more iterations
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-12, 'maxiter': 2000}
        )
        
        # Check if this result is better
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
