from scipy.optimize import minimize

        current_sum = np.sum(h)
        scale = target_sum / current_sum
        h = h * scale
        return h

    ]

        def objective(h):

            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
