import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Use the provided initial guess
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Function to compute C5
    def compute_c5(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Initial best solution
    best_h = initial_guess.copy()
    best_c5 = compute_c5(best_h)
    
    start_time = time.time()
    
    # Perform multiple optimizations with perturbations
    num_iterations = 100  # Increased number of iterations
    for _ in range(num_iterations):
        # Define objective function
        def objective(h):
            return compute_c5(h)
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization from current best
        result = minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 500}
        )
        
        # Check if optimization was successful
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Perturb the current best solution and optimize again
        perturb = np.random.normal(0, 0.05, n_points)
        perturbed_h = np.clip(best_h + perturb, 0.0, 1.0)
        result_perturb = minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 500}
        )
        
        # Update best solution if perturbed optimization is better
        if result_perturb.success and result_perturb.fun < best_c5:
            best_c5 = result_perturb.fun
            best_h = result_perturb.x
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
