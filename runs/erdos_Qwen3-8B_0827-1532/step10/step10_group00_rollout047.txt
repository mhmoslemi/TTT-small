from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    # Generate structured initial guesses with different periodicities
    structured_h = []
    for period in [2, 3, 4, 5, 6]:
        h = np.zeros(n_points)
        for i in range(n_points):
            if i % period == 0:
                h[i] = 1.0
            else:
                h[i] = 0.0
        structured_h.append(h)
    
    # Generate random binary initial guesses
    np.random.seed(seed)
    random_guesses = []
    for _ in range(5):
        h = np.zeros(n_points)
        indices = np.random.choice(n_points, size=int(n_points // 2), replace=False)
        h[indices] = 1.0
        random_guesses.append(h)
    
    # Combine all initial guesses
    initial_guesses = [initial_guess] + structured_h + random_guesses
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Evaluate each initial guess
    for guess in initial_guesses:
        # Ensure the sum is correct
        current_sum = np.sum(guess)
        scale = target_sum / current_sum
        scaled_guess = guess * scale
        
        # Objective function to minimize C5
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization with tighter tolerances and more iterations
        result = minimize(
            objective,

            method='SLSQP',
            bounds=bounds,
            constraints=cons,

        )

        if result.success and result.fun < best_c5:

            best_h = result.x
