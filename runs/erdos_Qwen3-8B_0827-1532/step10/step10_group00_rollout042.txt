def run(seed=42, budget_s=100, **kwargs):
    # Use the provided initial guess and generate diverse initial guesses
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate structured initial guesses
    structured_h = np.zeros(n_points)
    for i in range(n_points):
        if i % 2 == 0:
            structured_h[i] = 1.0
        else:
            structured_h[i] = 0.0
    
    structured_h2 = np.zeros(n_points)
    for i in range(n_points):
        if i % 3 == 0:
            structured_h2[i] = 1.0
        else:
            structured_h2[i] = 0.0
    
    structured_h3 = np.zeros(n_points)
    for i in range(n_points):
        if i % 4 == 0:
            structured_h3[i] = 1.0
        else:
            structured_h3[i] = 0.0
    
    structured_h4 = np.zeros(n_points)
    for i in range(n_points):
        if i % 5 == 0:
            structured_h4[i] = 1.0
        else:
            structured_h4[i] = 0.0
    
    # Generate random binary initial guess
    np.random.seed(seed)
    random_binary = np.zeros(n_points)
    indices = np.random.choice(n_points, size=int(n_points // 2), replace=False)
    random_binary[indices] = 1.0
    
    # Combine initial guesses
    initial_guesses = [initial_guess, structured_h, structured_h2, structured_h3, structured_h4, random_binary]
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-8, 'maxiter': 1000}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Now perform perturbations based on best_h
    start_time = time.time()
    num_perturbations = 150
    for perturbation_idx in range(num_perturbations):
        # Generate perturbations based on best_h
        if perturbation_idx < 10:
            # Random initial guess based on best_h
            random_noise = np.random.uniform(-0.2, 0.2, n_points)
            perturbed = best_h + random_noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
        else:
            # Perturb best_h with varying noise
            if perturbation_idx < 30:
                # Larger noise for early perturbations
                noise = np.random.uniform(-0.2, 0.2, n_points)
            else:
                # Smaller noise for later perturbations
                noise = np.random.uniform(-0.05, 0.05, n_points)
            perturbed = best_h + noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-8, 'maxiter': 1000}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
