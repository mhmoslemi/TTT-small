from scipy.optimize import minimize
import time

    # Binary pattern: first half is 1, second half is 0

    binary_initial[:n_points//2] = 1.0
    initial_guesses.append(binary_initial)
    # Sine wave pattern
    x = np.linspace(0, 2, n_points)
    sine_initial = 0.5 * (1 + np.sin(np.pi * x))
    initial_guesses.append(sine_initial)
    # Random initial guess
    random_initial = np.random.uniform(0, 1, n_points)
    random_initial = random_initial / np.sum(random_initial) * target_sum
    random_initial = np.clip(random_initial, 0, 1)
    initial_guesses.append(random_initial)
    
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    
    start_time = time.time()
    
    # Parameters for perturbations
    num_perturbations_per_guess = 50  # Reduced to save time
    
    for guess in initial_guesses:
        # Perturb the guess with varying noise
        for perturbation_idx in range(num_perturbations_per_guess):
            # Vary noise magnitude based on perturbation index
            if perturbation_idx < 10:
                noise = np.random.uniform(-0.2, 0.2, n_points)
            elif perturbation_idx < 30:
                noise = np.random.uniform(-0.1, 0.1, n_points)
            else:
                noise = np.random.uniform(-0.05, 0.05, n_points)
            
            perturbed = guess + noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
            
            # Objective function to minimize C5
            def objective(h):
                one_minus_h = 1.0 - h
                corr = np.correlate(h, one_minus_h, mode='full')
                max_corr = np.max(corr)
                return max_corr * dx
            
            # Bounds for each h[i]
            bounds = [(0.0, 1.0) for _ in range(n_points)]
            
            # Constraints: sum(h) == target_sum
            cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
            
            # Perform optimization with tighter tolerances
            result = minimize(
                objective,
                perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'ftol': 1e-10, 'maxiter': 1000}
            )
            
            # Check if this result is better
            if result.success and result.fun < best_c5:
                best_c5 = result.fun
                best_h = result.x

            if elapsed >= budget_s:
                break
