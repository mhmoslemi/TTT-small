import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate initial guesses
    def generate_periodic_guess(period):
        h = np.zeros(n_points)
        for i in range(n_points):
            if i % period == 0:
                h[i] = 1.0
        current_sum = np.sum(h)
        scale = target_sum / current_sum if current_sum != 0 else 1.0
        return h * scale
    
    def generate_random_guess():
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        return h
    
    initial_guesses = [
        initial_guess,
        generate_periodic_guess(2),
        generate_periodic_guess(3),
        generate_periodic_guess(4),
        generate_periodic_guess(5),
        generate_random_guess()
    ]
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    for guess in initial_guesses:
        # Objective function with penalty for sum constraint
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            penalty = 1e4 * np.abs(np.sum(h) - target_sum)
            return max_corr * dx + penalty
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Run differential evolution
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=15,
            mutation=(0.5, 1.0),
            recombination=0.8,
            tol=1e-12,
            maxiter=200,
            seed=seed
        )
        
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
