from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Use the provided initial guess and generate a structured initial guess
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Function to generate periodic initial guess
    def generate_periodic_guess(period):
        h = np.zeros(n_points)
        for i in range(n_points):
            if i % period == 0:
                h[i] = 1.0
        current_sum = np.sum(h)
        scale = target_sum / current_sum
        h = h * scale
        return h
    
    # Function to generate binary initial guess
    def generate_binary_guess():
        indices = np.random.choice(n_points, size=n_points//2, replace=False)
        h = np.zeros(n_points)
        h[indices] = 1.0
        return h
    
    # Generate initial guesses
    initial_guesses = [
        initial_guess,
        generate_periodic_guess(2),
        generate_periodic_guess(3),
        generate_periodic_guess(4),
        generate_periodic_guess(5),
        generate_binary_guess(),
    ]
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Run optimization for each initial guess
    for guess in initial_guesses:
        # Objective function to minimize C5
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization with tighter tolerances and more iterations
        result = minimize(
            objective,
            guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-8, 'maxiter': 1000}
        )
        
        # Check if this result is better
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check elapsed time

        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
