import numpy as np
from scipy.optimize import minimize
import time
from multiprocessing import Pool, cpu_count

def run(seed=42, budget_s=100, **kwargs):
    # Predefined initial guess
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate diverse initial guesses
    initial_guesses = []
    initial_guesses.append(initial_h_values)
    
    # Structured patterns with varying periods
    for period in [2, 3, 4, 5, 6, 7, 8]:
        structured = np.zeros(n_points)
        for i in range(n_points):
            if i % period == 0:
                structured[i] = 1.0
            else:
                structured[i] = 0.0
        initial_guesses.append(structured)
    
    # Random binary initial guesses
    np.random.seed(seed)
    for _ in range(5):
        random_binary = np.zeros(n_points)
        indices = np.random.choice(n_points, size=int(n_points // 2), replace=False)
        random_binary[indices] = 1.0
        initial_guesses.append(random_binary)
    
    # Function to optimize a single initial guess
    def optimize(initial_h):
        # Scale to meet the sum constraint
        current_sum = np.sum(initial_h)
        scale = target_sum / current_sum
        h = initial_h * scale
        h = np.clip(h, 0, 1)
        
        # Objective function to minimize C5
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization with tighter tolerances
        result = minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 1000}
        )
        
        # Return the result if successful
        if result.success:
            return (result.x, result.fun)
        else:
            return (h, np.inf)
    
    # Use multiprocessing to run optimizations in parallel
    start_time = time.time()
    with Pool(processes=cpu_count()) as pool:
        results = pool.map(optimize, initial_guesses)
    
    # Find the best result
    best_h, best_c5 = min(results, key=lambda x: x[1])
    
    # Early exit if time is exhausted
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
