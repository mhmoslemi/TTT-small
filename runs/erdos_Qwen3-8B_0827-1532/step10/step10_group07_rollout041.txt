import numpy as np
from scipy.optimize import minimize
import time

    # Use the provided initial guess
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Function to compute C5
    def compute_c5(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Initial best solution
    best_h = initial_guess.copy()
    best_c5 = compute_c5(best_h)
    
    start_time = time.time()
    
    # Generate multiple initial guesses
    np.random.seed(seed)
    initial_guesses = [initial_guess]
    for _ in range(3):  # Add 3 random initial guesses
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        initial_guesses.append(h)
    
    # Define objective function
    def objective(h):
        return compute_c5(h)
    
    # Define bounds and constraints
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    for guess in initial_guesses:
