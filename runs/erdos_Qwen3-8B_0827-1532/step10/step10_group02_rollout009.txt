from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    target_sum = n_points / 2

    initial_binary = np.sort(initial_binary)  # Sort to spread out 1s
    initial_binary = np.concatenate([initial_binary[:int(n_points/2)], np.ones(int(n_points/2))])
    initial_binary = np.random.permutation(initial_binary)  # Shuffle for randomness
    initial_binary = initial_binary.astype(float)
    initial_guess = initial_binary.copy()
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    num_perturbations = 300  # Increased for better exploration
    
    for perturbation_idx in range(num_perturbations):
        if perturbation_idx < 50:
            # Large noise for initial exploration
            noise = np.random.uniform(-0.5, 0.5, n_points)
        elif perturbation_idx < 150:
            # Moderate noise for refinement
            noise = np.random.uniform(-0.2, 0.2, n_points)
        else:
            # Small noise for fine-tuning
            noise = np.random.uniform(-0.05, 0.05, n_points)
        
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 500}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
