    def generate_periodic_guess(period):
        h = np.zeros(n_points)

            if i % period == 0:
                h[i] = 1.0
        current_sum = np.sum(h)
        scale = target_sum / current_sum
        h = h * scale
        return h

    def objective(h):

        max_corr = np.max(corr)
        return max_corr * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
