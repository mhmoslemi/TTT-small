from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):

    initial_guess = initial_h_values

    dx = 2.0 / n_points
    target_sum = n_points / 2

    def generate_initial_guesses():

        guesses.append(initial_guess.copy())

            guesses.append(structured)

        np.random.seed(seed)

            random_binary = np.zeros(n_points)

            random_binary[indices] = 1.0
            guesses.append(random_binary)

        scaled_guess = guess * (target_sum / np.sum(guess))

        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)

        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        # Optimization
        result = minimize(
            objective,
            scaled_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 1500}
        )
        if result.success:

        return (float('inf'), np.zeros(n_points))
