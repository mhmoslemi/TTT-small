from scipy.optimize import minimize
import time

    n_points = len(initial_guess)
    dx = 2.0 / n_points

    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Increase the number of perturbations for better exploration
    num_perturbations = 300  # More iterations for thorough search
    
    for perturbation_idx in range(num_perturbations):
        if perturbation_idx < 20:
            # Generate diverse initial guesses with larger noise
            random_noise = np.random.uniform(-0.3, 0.3, n_points)
            perturbed = initial_guess + random_noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
        else:
            # Perturb the best solution with decreasing noise
            if perturbation_idx < 30:
                noise = np.random.uniform(-0.2, 0.2, n_points)
            else:
                noise = np.random.uniform(-0.05, 0.05, n_points)
            perturbed = best_h + noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
        
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Use trust-constr for better convergence and handling of constraints
        result = minimize(
            objective,
            perturbed,
            method='trust-constr',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 200}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
