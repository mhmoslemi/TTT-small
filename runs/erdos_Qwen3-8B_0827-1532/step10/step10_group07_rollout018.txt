    if elapsed >= budget_s:
        # If time is up, return the best solution found so far
        return best_h, best_c5, n_points
    else:
        # Perform a final refinement with SLSQP
        def objective_slsqp(h):
            return compute_c5(h)
        
        bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result_slsqp = minimize(
            objective_slsqp,
            best_h,
            method='SLSQP',
            bounds=bounds_slsqp,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 100}
        )
        
        if result_slsqp.success and result_slsqp.fun < best_c5:
            best_c5 = result_slsqp.fun
            best_h = result_slsqp.x
        
        return best_h, best_c5, n_points
