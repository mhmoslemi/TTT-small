        return initial_h_values, np.max(np.correlate(initial_h_values, 1 - initial_h_values, mode='full')) * dx, n_points

    initial_guess = result_de.x
    
    # Define the objective function without penalty for SLSQP
    def objective_slsqp(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraints: sum(h) == target_sum
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    
    # Bounds
    bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]
    
    # Run SLSQP optimization
    result_slsqp = minimize(
        objective_slsqp,
        initial_guess,
        method='SLSQP',
        bounds=bounds_slsqp,
        constraints=cons,
        options={'ftol': 1e-10, 'maxiter': 1000}
    )
    
    best_h = result_slsqp.x
    best_c5 = result_slsqp.fun
    
    return best_h, best_c5, n_points
