import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Use the provided initial guess
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Function to compute C5
    def compute_c5(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Initial best solution
    best_h = initial_guess.copy()
    best_c5 = compute_c5(best_h)
    
    start_time = time.time()
    
    # First, run differential evolution to find a good initial solution
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def objective_de(h):
        c5 = compute_c5(h)
        penalty = 1e4 * abs(np.sum(h) - target_sum)
        return c5 + penalty
    
    # Run differential evolution with limited time
    try:
        result_de = differential_evolution(
            objective_de, 
            bounds, 
            strategy='best1bin', 
            maxiter=50, 
            popsize=15, 
            tol=1e-6, 
            mutation=(0.5, 1), 
            recombination=0.8,
            polish=True,
            disp=False
        )
        de_h = result_de.x
        de_c5 = compute_c5(de_h)
        
        # Update best solution
        if de_c5 < best_c5:
            best_h = de_h
            best_c5 = de_c5
    except:
        pass  # Fall back to initial guess if DE fails
    
    # Then, perform SLSQP optimization starting from the differential evolution solution
    # Define objective function
    def objective_slsqp(h):
        return compute_c5(h)
    
    # Constraints: sum(h) == target_sum
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    
    # Perform optimization
    try:
        result_slsqp = minimize(
            objective_slsqp,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 1000}
        )
        
        # Check if optimization was successful
        if result_slsqp.success and result_slsqp.fun < best_c5:
            best_c5 = result_slsqp.fun
            best_h = result_slsqp.x
    except:
        pass  # Fall back to previous best
    
    # Check elapsed time
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
