from scipy.optimize import minimize
import time

    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    num_perturbations = 300  # Increased for better exploration
    
    for perturbation_idx in range(num_perturbations):
        if perturbation_idx < 50:
            # Large noise for early exploration
            noise = np.random.uniform(-0.2, 0.2, n_points)
        elif perturbation_idx < 150:
            # Medium noise for moderate exploration
            noise = np.random.uniform(-0.1, 0.1, n_points)
        else:
            # Small noise for fine-tuning
            noise = np.random.uniform(-0.05, 0.05, n_points)
        
        perturbed = initial_guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

        )

            best_h = result.x
