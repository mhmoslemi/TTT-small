        max_corr = np.max(corr)
        return max_corr * dx
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Set up the optimization
    start_time = time.time()
    
    # Run differential evolution with adjusted parameters for better convergence
    result = differential_evolution(
        objective,
        bounds,
        popsize=20,
        mutation=(0.6, 1.0),
        recombination=0.8,
        tol=1e-6,
        polish=True,
        maxiter=150,
        seed=seed
    )
    
    # Check elapsed time and return the best solution found
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return result.x, result.fun, n_points
    
    return result.x, result.fun, n_points
