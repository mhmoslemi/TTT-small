            options={'ftol': 1e-10, 'maxiter': 1000}
        )
        
        # Check if optimization was successful
        if result.success:
            return result.x, result.fun
        else:
            return guess, compute_c5(guess)
    
    # Run optimization from each initial guess
    best_h = initial_guess
    best_c5 = compute_c5(initial_guess)
    
    start_time = time.time()
    
    for guess in initial_guesses:
        # Run optimization
        h_opt, c5_opt = optimize_from_guess(guess)
        # Update best solution
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
