from scipy.optimize import minimize

    # Sine wave pattern

    initial_guesses.append(sine_initial)
    # Structured pattern: 1 on [0, 0.5], [1, 1.5], 0 elsewhere
    structured_initial = np.zeros(n_points)
    for i in range(n_points):
        x_i = i * dx
        if (0 <= x_i < 0.5) or (1 <= x_i < 1.5):
            structured_initial[i] = 1.0
    initial_guesses.append(structured_initial)
    
    best_c5 = float('inf')
    best_h = initial_h_values.copy()

                noise = np.random.uniform(-0.1, 0.1, n_points)
            else:
                noise = np.random.uniform(-0.05, 0.05, n_points)
            
            perturbed = guess + noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum

            # Objective function to minimize C5
            def objective(h):
                one_minus_h = 1.0 - h
                corr = np.correlate(h, one_minus_h, mode='full')
                max_corr = np.max(corr)
                return max_corr * dx
            
            # Bounds for each h[i]
            bounds = [(0.0, 1.0) for _ in range(n_points)]
