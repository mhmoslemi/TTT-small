import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Define initial guesses
    initial_guesses = [
        initial_h_values.copy(),
        np.zeros(n_points),  # Binary pattern: first half is 1, second half is 0
        0.5 * (1 + np.sin(np.pi * np.linspace(0, 2, n_points)))
    ]
    binary_initial = np.zeros(n_points)
    binary_initial[:n_points//2] = 1.0
    initial_guesses.append(binary_initial)
    
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    
    start_time = time.time()
    
    for guess in initial_guesses:
        # Scale guess to meet the sum constraint
        current_sum = np.sum(guess)
        if current_sum == 0:
            scaled_guess = np.zeros(n_points)
        else:
            scale = target_sum / current_sum
            scaled_guess = np.clip(guess * scale, 0, 1)
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds and constraints
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization
        result = minimize(
            objective,
            scaled_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 500}
        )
        
        # Update best solution
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
