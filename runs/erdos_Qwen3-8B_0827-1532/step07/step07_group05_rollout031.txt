import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Generate initial guesses
    initial_guesses = []
    
    # 1. Initial guess from initial_h_values
    initial_guesses.append(initial_h_values)
    
    # 2. Random guess
    random_h = np.random.uniform(0, 1, n_points)
    random_h = random_h / np.sum(random_h) * (n_points / 2)
    initial_guesses.append(random_h)
    
    # 3. Sinusoidal guess
    x = np.linspace(0, 2, n_points)
    sin_h = 0.5 + 0.5 * np.sin(np.pi * x)
    sin_h = sin_h / np.sum(sin_h) * (n_points / 2)
    initial_guesses.append(sin_h)
    
    # 4. Two-interval guess
    two_intervals = np.zeros(n_points)
    idx1 = np.where((x >= 0) & (x <= 0.5))[0]
    two_intervals[idx1] = 1.0
    idx2 = np.where((x >= 1.5) & (x <= 2.0))[0]
    two_intervals[idx2] = 1.0
    two_intervals = two_intervals / np.sum(two_intervals) * (n_points / 2)
    initial_guesses.append(two_intervals)
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    for guess in initial_guesses:
        # Ensure the guess meets the sum constraint
        guess = guess / np.sum(guess) * (n_points / 2)
        # Perform optimization
        result = minimize(
            objective,
            guess,
            method='L-BFGS-B',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints={'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2},
            options={'ftol': 1e-9, 'maxiter': 200}
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # If no optimization was done, use the initial guess
    if best_h is None:
        best_h = initial_h_values
        one_minus_h = 1.0 - best_h
        corr = np.correlate(best_h, one_minus_h, mode='full')
        best_c5 = np.max(corr) * dx
    
    return best_h, best_c5, n_points
