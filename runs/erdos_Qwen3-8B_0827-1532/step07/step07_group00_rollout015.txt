from scipy.optimize import minimize
import time

    start_time = time.time()
    
    # Generate diverse initial guesses

    for perturbation_idx in range(num_perturbations):
        # Generate diverse perturbations
        if perturbation_idx < 50:

            noise = np.random.uniform(-0.3, 0.3, n_points)
        elif perturbation_idx < 100:
            # Medium noise for refinement
            noise = np.random.uniform(-0.15, 0.15, n_points)
        else:
            # Small noise for fine-tuning
            noise = np.random.uniform(-0.05, 0.05, n_points)
        
        # Apply noise and clip
        perturbed = initial_guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function to minimize C5
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
