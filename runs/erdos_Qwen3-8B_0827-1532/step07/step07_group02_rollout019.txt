from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    # Adjust sum to meet the constraint
    initial_guess = initial_guess / np.sum(initial_guess) * (n_points / 2)
    
    # Generate diverse initial guesses
    num_initial_guesses = 100  # Increased diversity
    initial_guesses = []

    for _ in range(20):

        perturbed = initial_guess + noise
        perturbed = np.clip(perturbed, 0, 1)

    def objective(h):

        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Run optimizations for each initial guess
    for guess in initial_guesses:
        # Run optimization
        result = minimize(

            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-9, 'maxiter': 1500}  # Reduced maxiter for speed
