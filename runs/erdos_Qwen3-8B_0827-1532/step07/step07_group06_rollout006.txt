                corr = np.correlate(best_h, one_minus_h, mode='full')
                max_corr = np.max(corr)
                best_c5 = max_corr * dx
            else:
                best_h = theta_opt * (n_points / 2) / sum_theta
        else:
            # Fallback to initial guess
            h_initial = initial_guess
            one_minus_h = 1.0 - h_initial
            corr = np.correlate(h_initial, one_minus_h, mode='full')
            max_corr = np.max(corr)
            best_c5 = max_corr * dx
            best_h = h_initial
    
    return best_h, best_c5, n_points
