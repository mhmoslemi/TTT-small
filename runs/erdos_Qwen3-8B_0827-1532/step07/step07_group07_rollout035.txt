import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 200  # Increased for better resolution
    dx = 2.0 / n_points
    
    # Generate structured initial guesses
    def generate_structured_guesses():
        guesses = []
        # Step function on [0, 1]
        step1 = np.zeros(n_points)
        step1[:n_points//2] = 1.0
        step1 = step1 / np.sum(step1) * (n_points / 2)
        guesses.append(step1)
        # Two intervals [0, 0.5] and [1, 1.5]
        step2 = np.zeros(n_points)
        step2[:n_points//4] = 1.0
        step2[n_points//2 : 3*n_points//4] = 1.0
        step2 = step2 / np.sum(step2) * (n_points / 2)
        guesses.append(step2)
        # Sinusoidal function
        sin_guess = 0.5 + 0.5 * np.sin(np.pi * np.linspace(0, 2, n_points))
        sin_guess = sin_guess / np.sum(sin_guess) * (n_points / 2)
        guesses.append(sin_guess)
        return guesses
    
    structured_guesses = generate_structured_guesses()
    
    # Generate random initial guesses
    random_guesses = []
    for _ in range(10):
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
        random_guesses.append(h)
    
    # Combine structured and random guesses
    initial_guesses = structured_guesses + random_guesses
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Run optimizations for each initial guess
    for guess in initial_guesses:
        # Run optimization with tighter tolerances
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-10, 'maxiter': 1000}
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # If no optimization was done, use the initial guess
    if best_h is None:
        result = minimize(
            objective,
            initial_guesses[0],
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-10, 'maxiter': 1000}
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points
