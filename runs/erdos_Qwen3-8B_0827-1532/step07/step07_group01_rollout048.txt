    # Adjust sum to meet the constraint
    initial_guess = initial_guess / np.sum(initial_guess) * (n_points / 2)
    initial_guesses.append(initial_guess)

    binary_guess[::2] = 1.0

    initial_guesses.append(sin_guess)
    
    # Triangular wave pattern
    triangular_guess = np.clip(2 * np.abs(x - 1) - 1, 0, 1)

    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]

    for guess in initial_guesses:

        result = minimize(
