from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Optimal balance between precision and speed
    dx = 2.0 / n_points
    x_new = np.linspace(0, 2, n_points)
    
    # Start with the provided initial construction
    initial_guess = np.array(initial_h_values)
    # Ensure sum constraint is met
    initial_guess = initial_guess / np.sum(initial_guess) * (n_points / 2)
    
    # Generate a few structured variations of the initial guess
    variations = [initial_guess]
    
    # Add sinusoidal perturbations
    for _ in range(3):
        noise_amp = np.random.uniform(0.01, 0.05)
        noise_freq = np.random.uniform(0.5, 10)
        noise = noise_amp * np.sin(2 * np.pi * noise_freq * x_new)
        perturbed = initial_guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        scale_factor = (n_points / 2) / np.sum(perturbed)
        perturbed *= scale_factor
        variations.append(perturbed)
    
    # Add random perturbations
    for _ in range(2):
        random_noise = np.random.uniform(-0.05, 0.05, n_points)
        perturbed = initial_guess + random_noise
        perturbed = np.clip(perturbed, 0, 1)
        scale_factor = (n_points / 2) / np.sum(perturbed)
        perturbed *= scale_factor
        variations.append(perturbed)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = initial_guess
    
    start_time = time.time()
    
    # Run optimizations for each initial guess with tighter tolerances
    for guess in variations:
        # Run optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-9, 'maxiter': 1500}
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # If no optimization was done, use the initial guess

        best_c5 = objective(initial_guess)
        best_h = initial_guess
