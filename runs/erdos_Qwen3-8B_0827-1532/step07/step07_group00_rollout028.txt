import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate structured initial guesses
    structured_guesses = []
    
    # Block pattern: first half 1, second half 0
    block_h = np.zeros(n_points)
    block_h[:n_points//2] = 1.0
    block_h = block_h * (target_sum / np.sum(block_h))
    structured_guesses.append(block_h)
    
    # Random initial guess with some structure
    random_h = np.random.uniform(0.4, 0.6, n_points)
    random_h = random_h * (target_sum / np.sum(random_h))
    structured_guesses.append(random_h)
    
    # Add initial guess to structured guesses
    structured_guesses.append(initial_guess)
    
    # Total number of initial guesses
    num_initial_guesses = len(structured_guesses)
    
    # Perturbation parameters
    num_perturbations = 200  # Increased number of perturbations
    
    for perturbation_idx in range(num_perturbations):
        # Use structured guesses first
        if perturbation_idx < num_initial_guesses:
            perturbed = structured_guesses[perturbation_idx].copy()
        else:
            # Random perturbation based on best_h
            if perturbation_idx < num_initial_guesses + 10:
                # Larger noise for early perturbations
                noise = np.random.uniform(-0.2, 0.2, n_points)
            else:
                # Smaller noise for later perturbations
                noise = np.random.uniform(-0.05, 0.05, n_points)
            perturbed = best_h + noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
        
        # Objective function to minimize C5
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization with tighter tolerances and fewer iterations
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 500}
        )
        
        # Check if this result is better
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
