from scipy.optimize import minimize

    symmetric_guess = np.clip(symmetric_guess, 0, 1)

    initial_guesses.append(symmetric_guess)

    for _ in range(num_initial_guesses // 2):

        perturbed *= scale_factor
        initial_guesses.append(perturbed)

    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
