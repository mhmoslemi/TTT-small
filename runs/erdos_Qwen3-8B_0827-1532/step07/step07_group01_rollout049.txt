from scipy.optimize import minimize

    initial_guesses = [initial_guess]

        if (i % 3) == 0:

    periodic_guess = periodic_guess / np.sum(periodic_guess) * (n_points / 2)
    initial_guesses.append(periodic_guess)

    # Objective function
    def objective(h):

        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    for guess in initial_guesses:
        # Run optimization
        result = minimize(

            method='SLSQP',

            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-10, 'maxiter': 3000}  # Increased maxiter and tighter tolerance
