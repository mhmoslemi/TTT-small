    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    num_initial_guesses = 12
    initial_guesses = [initial_guess.copy()]
    
    # Generate perturbations
    for _ in range(num_initial_guesses - 4):
        noise = np.random.normal(0, 0.05, n_points)
        perturbed = initial_guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        initial_guesses.append(perturbed)
    
    # Generate sinusoidal
    x = np.linspace(0, 2, n_points)
    sinusoidal = 0.5 + 0.5 * np.sin(np.pi * x)
    sinusoidal = np.clip(sinusoidal, 0, 1)
    current_sum = np.sum(sinusoidal)
    scale = target_sum / current_sum
    sinusoidal = sinusoidal * scale
    initial_guesses.append(sinusoidal)
    
    # Generate block function
    block_guess = np.zeros(n_points)
    n_block = n_points // 2
    block_guess[:n_block] = 1.0
    initial_guesses.append(block_guess)
    
    # Generate two-block function
    two_block_guess = np.zeros(n_points)
    for i in range(n_points):
        x = i * dx
        if 0.25 <= x <= 0.75:
            two_block_guess[i] = 1.0
        if 1.25 <= x <= 1.75:
            two_block_guess[i] = 1.0
    initial_guesses.append(two_block_guess)
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    start_time = time.time()
    
    for guess in initial_guesses:
        noise = np.random.normal(0, 0.02, n_points)
        perturbed = guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 1500}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
