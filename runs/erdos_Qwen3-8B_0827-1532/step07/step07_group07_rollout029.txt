from scipy.optimize import minimize, differential_evolution

        scale_factor = (n_points / 2) / np.sum(perturbed)

    for _ in range(num_initial_guesses // 2):

    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2

    def differential_evolution_objective(h):
