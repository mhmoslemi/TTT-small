import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Balanced between speed and precision
    dx = 2.0 / n_points
    
    # Generate initial guess based on provided initial_h_values
    initial_guess = initial_h_values.copy()
    # Rescale to meet the constraint
    initial_guess = initial_guess / np.sum(initial_guess) * (n_points / 2)

    def objective(h):

        corr = np.correlate(h, one_minus_h, mode='full')

        # Penalty term for constraint violation
        penalty = 1e4 * (np.sum(h) - n_points / 2)**2
        return max_corr * dx + penalty
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    
    # Run differential evolution with adjusted parameters
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.8,

        maxiter=1000,

        # Note: scipy's differential_evolution does not support x0 parameter
    )
    
    elapsed = time.time() - start_time

        # Assuming that the result is valid

        best_c5 = result.fun
    else:
        best_h = result.x
