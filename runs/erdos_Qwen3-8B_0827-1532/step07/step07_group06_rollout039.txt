import numpy as np
from scipy.optimize import minimize
import time
from concurrent.futures import ProcessPoolExecutor, as_completed

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    
    # Generate initial guesses
    initial_guesses = []
    
    # Add initial_h_values as a base guess
    initial_guesses.append(initial_h_values)
    
    # Generate perturbed initial guesses
    for _ in range(50):
        noise = np.random.normal(0, 0.01, n_points)
        perturbed = initial_h_values + noise
        perturbed = np.clip(perturbed, 0, 1)
        scale_factor = (n_points / 2) / np.sum(perturbed)
        perturbed *= scale_factor
        initial_guesses.append(perturbed)
    
    # Generate random initial guesses
    for _ in range(50):
        h_rand = np.random.rand(n_points)
        h_rand = h_rand / np.sum(h_rand) * (n_points / 2)
        initial_guesses.append(h_rand)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    
    # Use ProcessPoolExecutor for parallel processing
    with ProcessPoolExecutor(max_workers=10) as executor:
        futures = []
        for guess in initial_guesses:
            # Submit the optimization task
            future = executor.submit(
                optimize_single_guess, guess, objective, constraint, bounds, n_points
            )
            futures.append(future)
        
        # Process results as they come in
        for future in as_completed(futures):
            try:
                result = future.result()
                if result is not None:
                    current_c5 = result[0]
                    current_h = result[1]
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = current_h
            except Exception as e:
                print(f"Error in optimization: {e}")
            # Check if time budget is exceeded
            elapsed = time.time() - start_time
            if elapsed >= budget_s:
                break
    
    return best_h, best_c5, n_points

def optimize_single_guess(guess, objective, constraint, bounds, n_points):
    # Perform the optimization
    result = minimize(
        objective,
        guess,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint},
        options={'ftol': 1e-10, 'maxiter': 2000}
    )
    if result.success:
        return (result.fun, result.x)
    else:
        return (float('inf'), None)
