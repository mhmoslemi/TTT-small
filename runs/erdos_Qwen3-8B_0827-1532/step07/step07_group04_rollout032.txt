    current_sum = np.sum(sinusoidal)
    scale = target_sum / current_sum
    sinusoidal = sinusoidal * scale
    initial_guesses.append(sinusoidal)
    
    # Triangular function
    triangular = np.minimum(x, 2 - x)
    triangular = np.clip(triangular, 0, 1)
    initial_guesses.append(triangular)
    
    # Square wave
    square_wave = np.zeros(n_points)
    square_wave[:n_points//2] = 1.0
    square_wave = np.clip(square_wave, 0, 1)
    current_sum = np.sum(square_wave)
    scale = target_sum / current_sum
    square_wave = square_wave * scale
    initial_guesses.append(square_wave)
    
    # Piecewise constant function
    piecewise = np.zeros(n_points)

    piecewise[n_points//3: 2*n_points//3] = 0.0

    piecewise = np.clip(piecewise, 0, 1)
    current_sum = np.sum(piecewise)
    scale = target_sum / current_sum
    piecewise = piecewise * scale
    initial_guesses.append(piecewise)
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    start_time = time.time()
    
    for guess in initial_guesses:
        # Add noise and scale
        noise = np.random.normal(0, 0.05, n_points)
        perturbed = guess + noise

        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
