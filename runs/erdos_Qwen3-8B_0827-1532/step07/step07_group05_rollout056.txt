import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Balanced between speed and precision
    dx = 2.0 / n_points
    
    # Generate initial guesses based on structured patterns and initial_h_values
    initial_guesses = []
    
    # Use the provided initial_h_values as a starting point
    initial_guesses.append(initial_h_values)
    
    # Step function on [0, 1]
    step1 = np.zeros(n_points)
    step1[:50] = 1.0
    initial_guesses.append(step1)
    
    # Step function on [0, 0.5] and [1, 1.5]
    step2 = np.zeros(n_points)
    step2[:25] = 1.0
    step2[25:75] = 1.0
    initial_guesses.append(step2)
    
    # Random guess with proper scaling
    random_h = np.random.uniform(0, 1, n_points)
    random_h = random_h / np.sum(random_h) * (n_points / 2)
    initial_guesses.append(random_h)
    
    # Additional structured guess with three intervals
    step3 = np.zeros(n_points)
    step3[:17] = 1.0
    step3[17:34] = 0.0
    step3[34:51] = 1.0
    step3[51:68] = 0.0
    step3[68:] = 1.0
    # Scale to meet the integral constraint
    step3 = step3 / np.sum(step3) * (n_points / 2)
    initial_guesses.append(step3)
    
    # Objective function to minimize the maximum overlap
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function to ensure integral is 1
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    
    # Optimize each initial guess with COBYLA for better constraint handling
    for guess in initial_guesses:
        # Ensure the guess meets the constraint
        guess = guess / np.sum(guess) * (n_points / 2)
        # Run optimization
        result = minimize(
            objective,
            guess,
            method='COBYLA',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 1000, 'disp': False}
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # Fallback to initial guess if no optimization was successful
    if best_h is None:
        result = minimize(
            objective,
            initial_guesses[0],
            method='COBYLA',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 1000, 'disp': False}
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points
