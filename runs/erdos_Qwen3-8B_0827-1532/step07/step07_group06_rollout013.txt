import numpy as np
from scipy.optimize import minimize
import time
from multiprocessing import Pool, cpu_count

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Reduced number of points for faster evaluation
    dx = 2.0 / n_points
    
    # Generate initial guess based on structured patterns
    initial_guesses = []
    
    # Structured pattern: alternating 1s and 0s
    structured = np.zeros(n_points)
    structured[::2] = 1.0
    structured = structured / np.sum(structured) * (n_points / 2)
    initial_guesses.append(structured)
    
    # Perturb structured guess
    for _ in range(20):
        noise = np.random.normal(0, 0.01, n_points)
        perturbed = structured + noise
        perturbed = np.clip(perturbed, 0, 1)
        scale_factor = (n_points / 2) / np.sum(perturbed)
        perturbed *= scale_factor
        initial_guesses.append(perturbed)
    
    # Random initial guesses
    for _ in range(20):
        h_rand = np.random.rand(n_points)
        h_rand = h_rand / np.sum(h_rand) * (n_points / 2)
        initial_guesses.append(h_rand)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Use multiprocessing to parallelize optimization
    def optimize_guess(guess):
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-10, 'maxiter': 3000}
        )
        return result
    
    # Run optimizations in parallel
    with Pool(processes=cpu_count()) as pool:
        results = pool.map(optimize_guess, initial_guesses)
        for result in results:
            elapsed = time.time() - start_time
            if elapsed >= budget_s:
                break
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
