        h_clipped = np.clip(h, 0, 1)
        h_scaled = h_clipped * (target_sum / np.sum(h_clipped))
        one_minus_h = 1.0 - h_scaled
        # Compute cross-correlation using FFT
        n = len(h_scaled)
        m = len(one_minus_h)
        len_output = n + m - 1
        # Pad both arrays to len_output
        h_padded = np.zeros(len_output)
        h_padded[:n] = h_scaled
        one_minus_h_padded = np.zeros(len_output)
        one_minus_h_padded[:m] = one_minus_h
        # Compute FFT
        fft_h = fft(h_padded)
        fft_1mh = fft(one_minus_h_padded)
        # Multiply
        corr_fft = fft_h * fft_1mh
        # Inverse FFT
        corr = ifft(corr_fft).real
        # The maximum value in the cross-correlation
        max_corr = np.max(corr)
        return max_corr * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=15,
        mutation=(0.5, 1.0),
        recombination=0.8,
        tol=1e-10,
        maxiter=100,
        seed=seed
    )
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        # Return the best found so far
        return result.x, result.fun, n_points
    
    # If time allows, maybe do more iterations or refine
    # But for simplicity, return the result
    return result.x, result.fun, n_points
