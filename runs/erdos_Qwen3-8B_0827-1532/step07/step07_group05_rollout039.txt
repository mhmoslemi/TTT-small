import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Balanced between speed and precision
    dx = 2.0 / n_points
    
    # Generate initial guess based on initial_h_values
    x_initial = np.linspace(0, 2, len(initial_h_values))
    x_new = np.linspace(0, 2, n_points)
    initial_guess = np.interp(x_new, x_initial, initial_h_values)
    # Scale to meet the constraint
    initial_guess = initial_guess / np.sum(initial_guess) * (n_points / 2)
    
    # Define bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Objective function
    def objective(h):
        # Normalize h to meet the sum constraint
        h_normalized = h / np.sum(h) * (n_points / 2)
        one_minus_h = 1.0 - h_normalized
        corr = np.correlate(h_normalized, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Run differential evolution
    start_time = time.time()
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=15,
        maxiter=100,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed
    )
    elapsed = time.time() - start_time
    
    # Return the best solution found
    best_h = result.x
    best_c5 = result.fun
    return best_h, best_c5, n_points
