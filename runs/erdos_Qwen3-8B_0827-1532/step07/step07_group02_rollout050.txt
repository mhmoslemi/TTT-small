    # Step function
    step_guess = np.zeros_like(x_new)
    step_guess[:n_points//2] = 1.0
    structured_guesses.append(step_guess)
    
    # Generate diverse initial guesses with random perturbations
    num_initial_guesses = 50  # Increased diversity
    initial_guesses = []
    # Add original initial guess with noise
    for _ in range(num_initial_guesses):
        # Add random noise with varying amplitude and frequency
        noise_amp = np.random.uniform(0.001, 0.05)
        noise_freq = np.random.uniform(1, 10)
        noise = noise_amp * np.sin(2 * np.pi * noise_freq * x_new)
        perturbed = initial_guess + noise
        # Clip to [0, 1]
        perturbed = np.clip(perturbed, 0, 1)
        # Adjust sum to meet the constraint
        scale_factor = (n_points / 2) / np.sum(perturbed)
        perturbed *= scale_factor
        initial_guesses.append(perturbed)
    
    # Add structured guesses with noise
    for guess in structured_guesses:
        for _ in range(20):  # Perturb each structured guess 20 times
            noise_amp = np.random.uniform(0.001, 0.05)
            noise_freq = np.random.uniform(1, 10)
            noise = noise_amp * np.sin(2 * np.pi * noise_freq * x_new)
            perturbed = guess + noise
            # Clip to [0, 1]
            perturbed = np.clip(perturbed, 0, 1)
            # Adjust sum to meet the constraint
            scale_factor = (n_points / 2) / np.sum(perturbed)
            perturbed *= scale_factor
            initial_guesses.append(perturbed)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Run optimizations for each initial guess with tighter tolerances
    for guess in initial_guesses:
        # Run optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-9, 'maxiter': 2000}  # Increased maxiter
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # If no optimization was done, use the initial guess
    if best_h is None:
        result = minimize(
            objective,
            initial_guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-9, 'maxiter': 2000}
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points
