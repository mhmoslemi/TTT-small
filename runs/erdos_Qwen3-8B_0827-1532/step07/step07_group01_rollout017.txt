    sinusoidal_guess = sinusoidal_guess / np.sum(sinusoidal_guess) * (n_points / 2)
    initial_guesses.append(sinusoidal_guess)
    
    # Add perturbations of the initial guess
    for _ in range(3):
        noise = np.random.normal(0, 0.05, n_points)
        perturbed = initial_h_values + noise
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed / np.sum(perturbed) * (n_points / 2)
        initial_guesses.append(perturbed)
    
    # Objective function with constraint penalty
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        penalty = 1e6 * np.abs(np.sum(h) - n_points / 2)
        return max_corr * dx + penalty
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    
    # Run differential evolution for each initial guess
    for guess in initial_guesses:
        result = differential_evolution(
            objective,
            bounds,
            popsize=15,
            mutation=(0.5, 1.0),
            recombination=0.8,
            strategy='best1bin',
            tol=1e-6,
            maxiter=100,
            polish=True,
            disp=False
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.success:
            # Recompute C5 without penalty
            one_minus_h = 1.0 - result.x
            corr = np.correlate(result.x, one_minus_h, mode='full')
            current_c5 = np.max(corr) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # Fallback to initial guess if no improvement
    if best_h is None:
        best_h = initial_h_values
        one_minus_h = 1.0 - best_h
        corr = np.correlate(best_h, one_minus_h, mode='full')
        best_c5 = np.max(corr) * dx
    
    return best_h, best_c5, n_points
