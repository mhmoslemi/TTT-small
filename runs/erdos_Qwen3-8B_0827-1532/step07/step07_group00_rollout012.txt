from scipy.optimize import minimize
import time

    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate multiple diverse initial guesses based on best_h
    num_perturbations = 200  # Increased number of perturbations
    for perturbation_idx in range(num_perturbations):
        # Generate structured initial guesses for some perturbations
        if perturbation_idx < 20:
            # Structured initial guess: two 1 regions and one 0 region
            a = 0.3
            b = a + 1.0
            h_structured = np.zeros(n_points)
            for i in range(n_points):
                x = i * dx
                if x < a:
                    h_structured[i] = 1.0
                elif x < b:
                    h_structured[i] = 0.0
                else:
                    h_structured[i] = 1.0
            # Normalize
            current_sum = np.sum(h_structured)
            scale = target_sum / current_sum
            perturbed = h_structured * scale
        else:
            # Random initial guess based on best_h
            if perturbation_idx < 30:
                # Larger noise for early perturbations
                noise = np.random.uniform(-0.2, 0.2, n_points)
            else:
                # Smaller noise for later perturbations
                noise = np.random.uniform(-0.05, 0.05, n_points)
            perturbed = best_h + noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
        
        # Objective function to minimize C5
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

        result = minimize(
