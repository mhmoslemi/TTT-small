import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')

            noise = np.random.uniform(-0.2, 0.2, n_points)

            scale = target_sum / current_sum
            perturbed = perturbed * scale
        else:
            # Smaller noise for later perturbations
