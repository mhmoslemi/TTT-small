import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 150  # Optimal balance between speed and resolution
    dx = 2.0 / n_points
    
    # Resample initial guess to match n_points
    x_initial = np.linspace(0, 2, len(initial_h_values))
    x_new = np.linspace(0, 2, n_points)
    initial_guess = np.interp(x_new, x_initial, initial_h_values)
    # Normalize to meet the sum constraint
    initial_guess = initial_guess * (n_points / 2) / np.sum(initial_guess)
    
    # Objective function with normalization
    def objective(h):
        h_normalized = h * (n_points / 2) / np.sum(h)
        one_minus_h = 1.0 - h_normalized
        corr = np.correlate(h_normalized, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=20,
        maxiter=100,
        tol=1e-10,
        mutation=(0.5, 1),
        recombination=0.8,
        polish=True,
        disp=False,
        x0=initial_guess
    )
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        # Fallback to initial guess if time is up
        return initial_guess, objective(initial_guess), n_points
    
    best_h = result.x
    best_c5 = result.fun
    
    return best_h, best_c5, n_points
