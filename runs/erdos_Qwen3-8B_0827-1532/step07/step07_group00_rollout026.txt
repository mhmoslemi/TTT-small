    # Function to optimize a single guess
    def optimize_guess(guess):

        def objective(h):

            constraints=cons,

                best_h = result.x
        
            # Check elapsed time
            elapsed = time.time() - start_time
            if elapsed >= budget_s:
                break
    
    return best_h, best_c5, n_points
