        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        penalty = 1e4 * np.abs(np.sum(h) - n_points / 2)
        return max_corr * dx + penalty
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    
    # Run differential evolution for global search
    result_de = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=15,
        maxiter=100,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed
    )
    
    # Fallback to initial guess if time is exhausted
    if time.time() - start_time >= budget_s:
        best_h = initial_guess
        best_c5 = objective(best_h)
    else:
        # Refine with SLSQP for local optimization
        result_slsqp = minimize(
            objective,
            result_de.x,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2},
            options={'ftol': 1e-9, 'maxiter': 200}
        )
        best_h = result_slsqp.x
        best_c5 = result_slsqp.fun
    
    return best_h, best_c5, n_points
