    binary_guess[::3] = 1.0
    binary_guess = binary_guess / np.sum(binary_guess) * (n_points / 2)
    initial_guesses.append(binary_guess)
    
    # Add sine wave pattern
    x_new = np.linspace(0, 2, n_points)
    sine_guess = 0.5 + 0.5 * np.sin(np.pi * x_new)
    sine_guess = sine_guess / np.sum(sine_guess) * (n_points / 2)
    initial_guesses.append(sine_guess)
    
    # Add random pattern
    random_guess = np.random.rand(n_points)
    random_guess = random_guess / np.sum(random_guess) * (n_points / 2)
    initial_guesses.append(random_guess)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Run optimizations for each initial guess with tighter tolerances
    for guess in initial_guesses:
        # Run optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-10, 'maxiter': 3000}  # Increased maxiter and tighter tolerance
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # If no optimization was done, use the initial guess
    if best_h is None:
        result = minimize(
            objective,
            initial_guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-10, 'maxiter': 3000}
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points
