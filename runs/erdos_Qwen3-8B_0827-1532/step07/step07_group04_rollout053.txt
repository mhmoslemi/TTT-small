import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate diverse initial guesses
    initial_guesses = []
    initial_guesses.append(initial_h_values.copy())

    h_binary = np.zeros(n_points)
    h_binary[:n_points//2] = 1.0
    initial_guesses.append(h_binary)
    
    # Random binary pattern with correct sum
    h_random = np.random.choice([0, 1], size=n_points)
    h_random = h_random.astype(float)
    current_sum = np.sum(h_random)
    h_random = h_random * (target_sum / current_sum)
    initial_guesses.append(h_random)
    
    # Triangular wave
    x = np.linspace(0, 2, n_points)
    h_triangular = 0.5 * (np.sin(np.pi * x) + 1)
    h_triangular = np.clip(h_triangular, 0, 1)
    current_sum = np.sum(h_triangular)
    h_triangular = h_triangular * (target_sum / current_sum)
    initial_guesses.append(h_triangular)
    
    # Square wave: 1 in [0,1], 0 elsewhere
    h_square = np.zeros(n_points)
    h_square[:n_points//2] = 1.0
    initial_guesses.append(h_square)
    
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    start_time = time.time()
    
    for guess in initial_guesses:
        # Add small perturbation to avoid local minima
        noise = np.random.normal(0, 0.05, n_points)
        perturbed = guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)

        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-12, 'maxiter': 2000}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
