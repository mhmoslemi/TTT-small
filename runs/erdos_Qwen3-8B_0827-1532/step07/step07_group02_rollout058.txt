import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Optimal balance between speed and precision
    dx = 2.0 / n_points
    
    # Generate initial guess based on initial_h_values
    x_initial = np.linspace(0, 2, len(initial_h_values))
    x_new = np.linspace(0, 2, n_points)
    initial_guess = np.interp(x_new, x_initial, initial_h_values)
    # Adjust sum to meet the constraint
    initial_guess = initial_guess / np.sum(initial_guess) * (n_points / 2)
    
    # Generate diverse initial guesses
    initial_guesses = []
    # Add the initial guess
    initial_guesses.append(initial_guess)
    # Generate binary initial guesses
    num_binary_guesses = 10
    for _ in range(num_binary_guesses):
        binary = np.zeros(n_points)
        binary[:n_points//2] = 1
        np.random.shuffle(binary)
        initial_guesses.append(binary)
    # Generate sinusoidal initial guesses
    num_sinusoidal_guesses = 10
    for _ in range(num_sinusoidal_guesses):
        x = np.linspace(0, 2, n_points)
        amp = 0.5
        freq = np.random.uniform(1, 5)
        phase = np.random.uniform(0, 2*np.pi)
        sinusoidal = amp * np.sin(2 * np.pi * freq * x + phase)
        sinusoidal = (sinusoidal - sinusoidal.min()) / (sinusoidal.max() - sinusoidal.min())
        scale_factor = (n_points / 2) / np.sum(sinusoidal)
        sinusoidal *= scale_factor
        initial_guesses.append(sinusoidal)
    # Generate perturbed initial guesses
    num_perturbed_guesses = 30
    for _ in range(num_perturbed_guesses):
        noise_amp = np.random.uniform(0.001, 0.05)
        noise_freq = np.random.uniform(1, 10)
        noise = noise_amp * np.sin(2 * np.pi * noise_freq * x_new)
        perturbed = initial_guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        scale_factor = (n_points / 2) / np.sum(perturbed)
        perturbed *= scale_factor
        initial_guesses.append(perturbed)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Run optimizations for each initial guess with tighter tolerances
    for guess in initial_guesses:
        # Run optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-9, 'maxiter': 2000}  # Increased maxiter
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.success:

            if current_c5 < best_c5:

                best_h = result.x

            initial_guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-9, 'maxiter': 2000}
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points
