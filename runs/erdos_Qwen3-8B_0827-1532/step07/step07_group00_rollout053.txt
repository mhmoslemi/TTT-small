            scale = target_sum / current_sum
            perturbed = perturbed * scale
        else:
            # Perturb best_h with varying noise while maintaining symmetry
            if perturbation_idx < 30:

            perturbed = best_h.copy()
            perturbed[:n_points // 2] += noise
            perturbed[n_points // 2:] = perturbed[:n_points // 2][::-1]

        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
