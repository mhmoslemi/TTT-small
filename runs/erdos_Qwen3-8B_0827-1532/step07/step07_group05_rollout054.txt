    start_time = time.time()
    
    # Run differential evolution for each initial guess
    for guess in initial_guesses:
        # Run differential evolution
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=100,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            args=(),
            seed=seed
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.success:
            current_c5 = result.fun - 1e4 * (np.sum(result.x) - n_points/2)**2  # Subtract penalty
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # If no optimization was done, use the initial guess
    if best_h is None:
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=100,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            args=(),
            seed=seed
        )
        best_c5 = result.fun - 1e4 * (np.sum(result.x) - n_points/2)**2
        best_h = result.x
    
    return best_h, best_c5, n_points
