            one_minus_h = 1.0 - h_vec
            corr = np.correlate(h_vec, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Constraints
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Optimization
        result = minimize(
            objective,
            h,
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints=cons,
            options={'ftol': 1e-8, 'maxiter': 500}
        )
        
        # Update best solution
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check time budget
        if time.time() - start_time >= budget_s:
            break
    
    return best_h, best_c5, n_points
