from scipy.optimize import minimize

    target_sum = n_points / 2

    binary_step = np.zeros(n_points)
    for i in range(n_points):
        if i % 2 == 0:
            binary_step[i] = 1.0
        else:
            binary_step[i] = 0.0
    initial_guesses.append(binary_step)
    
    # Add random binary step function as initial guess
    random_binary = np.zeros(n_points)
    random_binary[:n_points//2] = 1.0
    np.random.shuffle(random_binary)
    initial_guesses.append(random_binary)
    
    # Generate more initial guesses by perturbing
    for _ in range(5):
        noise = np.random.normal(0, 0.1, n_points)
        perturbed = initial_guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        initial_guesses.append(perturbed)
    
    x = np.linspace(0, 2, n_points)
    sinusoidal = 0.5 + 0.5 * np.sin(np.pi * x)
    sinusoidal = np.clip(sinusoidal, 0, 1)
    current_sum = np.sum(sinusoidal)
    scale = target_sum / current_sum
    sinusoidal = sinusoidal * scale
    initial_guesses.append(sinusoidal)
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    start_time = time.time()
    
    for guess in initial_guesses:
        noise = np.random.normal(0, 0.05, n_points)
        perturbed = guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(

            perturbed,

        if result.success and result.fun < best_c5:

            best_h = result.x
