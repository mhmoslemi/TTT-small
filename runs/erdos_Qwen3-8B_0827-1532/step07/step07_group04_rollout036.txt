from scipy.optimize import minimize
import time

    target_sum = n_points / 2
    
    # Generate diverse initial guesses
    initial_guesses = []
    x = np.linspace(0, 2, n_points)
    
    # Add initial guess
    initial_guesses.append(initial_guess.copy())
    
    # Sinusoidal
    sinusoidal = 0.5 + 0.5 * np.sin(np.pi * x)
    sinusoidal = np.clip(sinusoidal, 0, 1)
    current_sum = np.sum(sinusoidal)
    scale = target_sum / current_sum
    sinusoidal = sinusoidal * scale
    initial_guesses.append(sinusoidal)
    
    # Triangular
    triangular = np.minimum(x / 1.0, (2 - x) / 1.0)
    triangular = np.clip(triangular, 0, 1)
    current_sum = np.sum(triangular)
    scale = target_sum / current_sum
    triangular = triangular * scale
    initial_guesses.append(triangular)
    
    # Square wave
    square_wave = np.where(x < 1, 0, 1)
    square_wave = np.clip(square_wave, 0, 1)
    current_sum = np.sum(square_wave)
    scale = target_sum / current_sum
    square_wave = square_wave * scale
    initial_guesses.append(square_wave)
    
    # Random
    random_guess = np.random.uniform(0, 1, n_points)
    current_sum = np.sum(random_guess)
    scale = target_sum / current_sum
    random_guess = random_guess * scale
    initial_guesses.append(random_guess)
    
    # Add more random variations
    for _ in range(4):
        noise = np.random.normal(0, 0.1, n_points)
        perturbed = initial_guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        initial_guesses.append(perturbed)
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    start_time = time.time()
    
    for guess in initial_guesses:
        # Add small noise to diversify
        noise = np.random.normal(0, 0.05, n_points)
        perturbed = guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
