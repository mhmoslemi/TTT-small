import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Optimal balance between speed and precision
    dx = 2.0 / n_points
    
    # Generate initial guesses with structured perturbations
    initial_guesses = []
    
    # Base guess: interpolated initial_h_values with sinusoidal noise
    x_initial = np.linspace(0, 2, len(initial_h_values))
    x_new = np.linspace(0, 2, n_points)
    base_guess = np.interp(x_new, x_initial, initial_h_values)
    noise_amp = 0.05
    noise_freq = 5
    noise = noise_amp * np.sin(2 * np.pi * noise_freq * x_new)
    base_guess += noise
    base_guess = np.clip(base_guess, 0, 1)
    scale_factor = (n_points / 2) / np.sum(base_guess)
    base_guess *= scale_factor
    initial_guesses.append(base_guess)
    
    # Binary pattern guess
    binary_guess = np.zeros(n_points)
    binary_guess[::2] = 1.0
    binary_guess = binary_guess / np.sum(binary_guess) * (n_points / 2)
    initial_guesses.append(binary_guess)
    
    # Sinusoidal guess with different frequency
    noise_freq = 10
    noise = noise_amp * np.sin(2 * np.pi * noise_freq * x_new)
    sin_guess = np.clip(base_guess + noise, 0, 1)
    scale_factor = (n_points / 2) / np.sum(sin_guess)
    sin_guess *= scale_factor
    initial_guesses.append(sin_guess)
    
    # Triangular wave guess
    triangular_guess = np.clip(0.5 + 0.5 * np.sin(np.pi * x_new), 0, 1)
    triangular_guess = triangular_guess / np.sum(triangular_guess) * (n_points / 2)
    initial_guesses.append(triangular_guess)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    
    # Run optimizations for each initial guess
    for guess in initial_guesses:
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-10, 'maxiter': 3000}
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # Fallback to base guess if no optimization was successful
    if best_h is None:
        result = minimize(
            objective,
            base_guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-10, 'maxiter': 3000}
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points
