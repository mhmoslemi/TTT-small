import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 50  # Smaller n_points for faster evaluation
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate diverse initial guesses
    initial_guesses = []
    # Uniform distribution
    uniform = np.full(n_points, 0.5)
    initial_guesses.append(uniform)
    # Random binary patterns
    for _ in range(5):
        binary = np.zeros(n_points)
        indices = np.random.choice(n_points, size=int(target_sum), replace=False)
        binary[indices] = 1
        initial_guesses.append(binary)
    # Structured alternating pattern
    structured = np.zeros(n_points)
    for i in range(n_points):
        if i % 2 == 0:
            structured[i] = 1
    structured_sum = np.sum(structured)
    structured = structured * (target_sum / structured_sum)
    initial_guesses.append(structured)
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    for guess in initial_guesses:
        h_initial = guess.copy()
        
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 500}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
