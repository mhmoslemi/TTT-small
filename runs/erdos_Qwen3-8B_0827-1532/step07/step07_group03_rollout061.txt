    initial_guess = initial_guess / np.sum(initial_guess) * (n_points / 2)
    
    # Generate diverse initial guesses
    initial_guesses = []
    initial_guesses.append(initial_guess)  # Use provided initial guess
    
    # Add structured patterns
    # Square wave
    square_wave = np.zeros(n_points)
    for i in range(n_points):
        if x_new[i] < 1.0:
            square_wave[i] = 1.0
        else:
            square_wave[i] = 0.0
    initial_guesses.append(square_wave)
    # Triangular wave
    triangular_wave = np.zeros(n_points)
    for i in range(n_points):
        x = x_new[i]
        if x < 1.0:
            triangular_wave[i] = x / 1.0
        else:
            triangular_wave[i] = (2.0 - x) / 1.0
    initial_guesses.append(triangular_wave)
    # Sine wave
    sine_wave = np.zeros(n_points)
    for i in range(n_points):
        x = x_new[i]
        sine_wave[i] = 0.5 * (1 + np.sin(np.pi * x))
    initial_guesses.append(sine_wave)
    
    # Add a single random guess
    random_h = np.random.uniform(0, 1, n_points)
    random_h = random_h / np.sum(random_h) * (n_points / 2)
    initial_guesses.append(random_h)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Run optimizations for each initial guess with tighter tolerances
    for guess in initial_guesses:
        # Run optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-9, 'maxiter': 2000}
        )
        elapsed = time.time() - start_time
