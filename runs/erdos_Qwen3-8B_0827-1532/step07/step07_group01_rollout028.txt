    x_new = np.linspace(0, 2, n_points)
    initial_guess_interpolated = np.interp(x_new, x_initial, initial_h_values)
    # Adjust sum to meet the constraint
    initial_guess_interpolated = initial_guess_interpolated / np.sum(initial_guess_interpolated) * (n_points / 2)
    
    # Generate structured initial guess with 5 intervals of 0.2
    structured_guess = np.zeros(n_points)
    interval_length = 0.2
    gap_length = 0.2

        end = start + interval_length
        # Find indices
        start_idx = int(start / dx)
        end_idx = int(end / dx)
        # Ensure we don't go out of bounds
        start_idx = max(0, start_idx)
        end_idx = min(n_points, end_idx)
        structured_guess[start_idx:end_idx] = 1.0
    # Check sum
    structured_guess = structured_guess / np.sum(structured_guess) * (n_points / 2)
    
    # Generate binary initial guess
    binary_guess = np.zeros(n_points)

        random_noise = np.random.normal(0, 0.01, n_points)
        perturbed = initial_guess_interpolated + noise + random_noise
        # Clip to [0, 1]
        perturbed = np.clip(perturbed, 0, 1)
        # Adjust sum to meet the constraint
        scale_factor = (n_points / 2) / np.sum(perturbed)
        perturbed *= scale_factor
        initial_guesses.append(perturbed)
    
    # Objective function
    def objective(h):
