            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 1000, 'ftol': 1e-10}
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # If no optimization was done, use the initial guess
    if best_h is None:
        result = minimize(
            objective,
            initial_guesses[0],
            method='COBYLA',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 1000, 'ftol': 1e-10}
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points
