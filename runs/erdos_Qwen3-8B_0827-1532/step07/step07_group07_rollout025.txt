def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 150  # Balanced resolution for speed and accuracy
    dx = 2.0 / n_points

    x_new = np.linspace(0, 2, n_points)

    # Adjust sum to meet the constraint
    initial_guess = initial_guess / np.sum(initial_guess) * (n_points / 2)

    num_initial_guesses = 50  # Increased to explore more of the solution space
    initial_guesses = []
    
    # Perturb initial guess with structured noise
    for _ in range(num_initial_guesses // 2):
        noise = np.random.normal(0, 0.001, n_points)
        perturbed = initial_guess + noise

        scale_factor = (n_points / 2) / np.sum(perturbed)
        perturbed *= scale_factor
