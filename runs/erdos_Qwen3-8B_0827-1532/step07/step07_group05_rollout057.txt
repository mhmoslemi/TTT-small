        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        
        current_c5 = result.fun
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x
    
    # Fallback to SLSQP if no improvement found
    if best_h is None:
        def objective_slsqp(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        def constraint(h):
            return np.sum(h) - n_points / 2
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = minimize(
            objective_slsqp,
            initial_guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-9, 'maxiter': 2000}
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points
