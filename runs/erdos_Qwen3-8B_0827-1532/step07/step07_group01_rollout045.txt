    h_sinusoidal = 0.5 + 0.5 * np.sin(2 * np.pi * x / T)
    sum_h = np.sum(h_sinusoidal)
    scale_factor = (n_points / 2) / sum_h
    h_sinusoidal *= scale_factor
    initial_guesses.append(h_sinusoidal)
    
    # Block pattern with block_size=4
    block_size = 4
    h_block = np.zeros(n_points)
    for i in range(n_points // block_size):
        start = i * block_size
        end = start + block_size
        h_block[start:end] = 1.0
    sum_h = np.sum(h_block)
    scale_factor = (n_points / 2) / sum_h
    h_block *= scale_factor
    initial_guesses.append(h_block)
    
    # Add a few random perturbations to initial guesses
    for guess in initial_guesses[:2]:  # Perturb first two guesses
        noise_amp = np.random.uniform(0.001, 0.1)
        noise_freq = np.random.uniform(1, 10)
        noise = noise_amp * np.sin(2 * np.pi * noise_freq * x_new)
        random_noise = np.random.normal(0, 0.01, n_points)
        perturbed = guess + noise + random_noise
        perturbed = np.clip(perturbed, 0, 1)
        scale_factor = (n_points / 2) / np.sum(perturbed)
        perturbed *= scale_factor
        initial_guesses.append(perturbed)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Run optimizations for each initial guess with tighter tolerances
    for guess in initial_guesses:
        # Run optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-10, 'maxiter': 3000}  # Increased maxiter and tighter tolerance
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # If no optimization was done, use the initial guess
    if best_h is None:
        result = minimize(
            objective,
            initial_guesses[0],
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-10, 'maxiter': 3000}
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points
