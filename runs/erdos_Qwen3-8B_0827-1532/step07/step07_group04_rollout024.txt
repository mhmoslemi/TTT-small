    uniform = np.full(n_points, 0.5)
    initial_guesses.append(uniform)
    # Random binary
    binary = np.random.choice([0, 1], size=n_points)
    binary_sum = np.sum(binary)
    scale = target_sum / binary_sum
    binary = binary * scale
    initial_guesses.append(binary)
    # Sinusoidal
    x = np.linspace(0, 2, n_points)
    sinusoidal = 0.5 + 0.5 * np.sin(np.pi * x)
    sinusoidal = np.clip(sinusoidal, 0, 1)
    current_sum = np.sum(sinusoidal)
    scale = target_sum / current_sum
    sinusoidal = sinusoidal * scale
    initial_guesses.append(sinusoidal)
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    start_time = time.time()
    
    for guess in initial_guesses:
        # Perturb with small noise
        noise = np.random.normal(0, 0.05, n_points)
        perturbed = guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        def objective(h):
            one_minus_h = 1.0 - h
            # Pad h and one_minus_h with zeros to handle all shifts
            h_padded = np.pad(h, (0, n_points), 'constant')
            one_minus_h_padded = np.pad(one_minus_h, (0, n_points), 'constant')
            corr = np.correlate(h_padded, one_minus_h_padded, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Use COBYLA for better constraint handling
        result = minimize(
            objective,
            perturbed,
            method='COBYLA',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'disp': False}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
