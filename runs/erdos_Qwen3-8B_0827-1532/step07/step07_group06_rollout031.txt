import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    
    # Generate initial guess based on provided initial_h_values
    initial_guess = initial_h_values
    
    # Generate diverse initial guesses
    num_initial_guesses = 200
    initial_guesses = []
    
    # Generate perturbed initial guesses
    for _ in range(num_initial_guesses // 3):
        noise = np.random.normal(0, 0.01, n_points)
        perturbed = initial_guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        scale_factor = (n_points / 2) / np.sum(perturbed)
        perturbed *= scale_factor
        initial_guesses.append(perturbed)
    
    # Generate random initial guesses
    for _ in range(num_initial_guesses // 3):
        h_rand = np.random.rand(n_points)
        h_rand = h_rand / np.sum(h_rand) * (n_points / 2)
        initial_guesses.append(h_rand)
    
    # Generate structured initial guesses (alternating high/low)
    for _ in range(num_initial_guesses // 3):
        h_structured = np.zeros(n_points)
        for i in range(n_points):
            if i % 3 == 0:
                h_structured[i] = 1.0
            elif i % 3 == 1:
                h_structured[i] = 0.5
            else:
                h_structured[i] = 0.0
        h_structured = h_structured / np.sum(h_structured) * (n_points / 2)
        initial_guesses.append(h_structured)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Run optimizations for each initial guess
    for guess in initial_guesses:
        # Run optimization with SLSQP
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-9, 'maxiter': 1000}
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # Fallback to differential evolution if no optimization was successful
    if best_h is None:
        result_de = differential_evolution(
            objective,
            bounds,
            constraints=constraint,
            strategy='best1bin',
            popsize=15,
            maxiter=100,
            tol=1e-9,
            seed=seed
        )
        best_c5 = result_de.fun
        best_h = result_de.x
    
    return best_h, best_c5, n_points
