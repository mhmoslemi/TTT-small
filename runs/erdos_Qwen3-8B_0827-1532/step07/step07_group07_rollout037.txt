    # Adjust sum to meet the constraint
    initial_guess = initial_guess / np.sum(initial_guess) * (n_points / 2)
    
    # Generate structured initial guesses
    structured_guesses = []
    # Example 1: 1 on [0, 0.5] and [1.0, 1.5]
    structured_guess1 = np.zeros(n_points)
    for i in range(n_points):
        x = i * dx
        if (0 <= x < 0.5) or (1.0 <= x < 1.5):
            structured_guess1[i] = 1.0
    structured_guesses.append(structured_guess1)
    
    # Example 2: 1 on [0.5, 1.5]
    structured_guess2 = np.zeros(n_points)
    for i in range(n_points):
        x = i * dx
        if 0.5 <= x < 1.5:
            structured_guess2[i] = 1.0
    structured_guesses.append(structured_guess2)
    
    # Generate multiple initial guesses: some based on initial guess with noise, some random, and structured
    num_initial_guesses = 30  # Increased to 30
    initial_guesses = []
    # Add structured guesses
    for guess in structured_guesses:
        initial_guesses.append(guess)
    # Add initial guess with noise
    for _ in range(num_initial_guesses // 2):
        noise = np.random.normal(0, 0.001, n_points)
        perturbed = initial_guess + noise
        # Clip to [0, 1]
        perturbed = np.clip(perturbed, 0, 1)
        # Adjust sum to meet the constraint
        scale_factor = (n_points / 2) / np.sum(perturbed)
        perturbed *= scale_factor
        initial_guesses.append(perturbed)
    # Add random initial guesses
    for _ in range(num_initial_guesses // 2):
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
        initial_guesses.append(h)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Run optimizations for each initial guess
    for guess in initial_guesses:
        # Run optimization with tighter tolerances
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-12, 'maxiter': 2000}
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # If no optimization was done, use the initial guess
    if best_h is None:
        result = minimize(
            objective,
            initial_guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-12, 'maxiter': 2000}
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points
