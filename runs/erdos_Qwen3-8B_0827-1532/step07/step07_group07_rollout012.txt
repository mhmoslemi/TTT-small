def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 150  # Reduced for faster computation
    dx = 2.0 / n_points
    
    # Generate initial guess via interpolation
    x_initial = np.linspace(0, 2, len(initial_h_values))
    x_new = np.linspace(0, 2, n_points)
    initial_guess = np.interp(x_new, x_initial, initial_h_values)
    # Adjust sum to meet the constraint
    initial_guess = initial_guess / np.sum(initial_guess) * (n_points / 2)
    
    # Generate structured initial guesses
    structured_guesses = []
    
    def generate_structured_guess(n_points):
        x = np.linspace(0, 2, n_points)
        h = np.zeros(n_points)
        # Define intervals where h is 1
        interval1 = (x >= 0) & (x <= 0.5)
        interval2 = (x >= 1.0) & (x <= 1.5)
        h[interval1 | interval2] = 1.0
        return h
    
    structured_guesses.append(generate_structured_guess(n_points))
    
    def generate_structured_guess2(n_points):
        x = np.linspace(0, 2, n_points)
        h = np.zeros(n_points)
        interval1 = (x >= 0.25) & (x <= 0.75)
        interval2 = (x >= 1.25) & (x <= 1.75)
        h[interval1 | interval2] = 1.0
        return h
    
    structured_guesses.append(generate_structured_guess2(n_points))
    
    def generate_structured_guess3(n_points):
        x = np.linspace(0, 2, n_points)
        h = np.zeros(n_points)
        interval1 = (x >= 0) & (x <= 0.33)
        interval2 = (x >= 0.66) & (x <= 1.0)
        interval3 = (x >= 1.33) & (x <= 1.66)
        h[interval1 | interval2 | interval3] = 1.0
        return h
    
    structured_guesses.append(generate_structured_guess3(n_points))
    
    # Generate initial guesses
    initial_guesses = []
    
    # Add interpolated initial guess with noise
    for _ in range(10):
        noise = np.random.normal(0, 0.001, n_points)
        perturbed = initial_guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        scale_factor = (n_points / 2) / np.sum(perturbed)
        perturbed *= scale_factor
        initial_guesses.append(perturbed)
    
    # Add structured guesses
    initial_guesses.extend(structured_guesses)
    
    # Generate random initial guesses
    for _ in range(10):
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
        initial_guesses.append(h)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Run optimizations for each initial guess
    for guess in initial_guesses:
        # Run optimization with tighter tolerances
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-10, 'maxiter': 1000}
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # If no optimization was done, use the initial guess
    if best_h is None:
        result = minimize(
            objective,
            initial_guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-10, 'maxiter': 1000}
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points
