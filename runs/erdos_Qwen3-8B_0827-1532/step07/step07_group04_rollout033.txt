from scipy.optimize import minimize
import time

    sinusoidal = np.clip(sinusoidal, 0, 1)
    current_sum = np.sum(sinusoidal)
    scale = target_sum / current_sum
    sinusoidal = sinusoidal * scale
    initial_guesses.append(sinusoidal)
    
    # Add random guess
    random_guess = np.random.uniform(0, 1, n_points)
    current_sum = np.sum(random_guess)
    scale = target_sum / current_sum
    random_guess = random_guess * scale
    initial_guesses.append(random_guess)
    
    # Add piecewise constant guess (alternating 0 and 1)
    piecewise = np.zeros(n_points)
    for i in range(n_points):
        if i % 2 == 0:
            piecewise[i] = 1.0
    current_sum = np.sum(piecewise)
    scale = target_sum / current_sum
    piecewise = piecewise * scale
    initial_guesses.append(piecewise)
    
    # Generate more perturbed guesses
    for _ in range(5):
        noise = np.random.normal(0, 0.1, n_points)
        perturbed = initial_guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        initial_guesses.append(perturbed)
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    start_time = time.time()
    
    for guess in initial_guesses:
        # Add small noise to avoid getting stuck in same local minima
        noise = np.random.normal(0, 0.05, n_points)
        perturbed = guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
