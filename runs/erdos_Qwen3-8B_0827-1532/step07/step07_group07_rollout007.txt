from scipy.optimize import minimize

    initial_guess = initial_guess / np.sum(initial_guess) * (n_points / 2)
    
    # Generate diverse initial guesses: perturbations, random, and structured

        scale_factor = (n_points / 2) / np.sum(perturbed)
        perturbed *= scale_factor
        initial_guesses.append(perturbed)
    # Generate random initial guesses
    for _ in range(num_initial_guesses // 2):

    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Run optimizations for each initial guess
    for guess in initial_guesses:
        # Run optimization with tighter tolerances
        result = minimize(

            method='SLSQP',
