    for i in range(0, n_points, 2 * cluster_size):
        cluster_guess[i:i+cluster_size] = 1.0
    cluster_guess = cluster_guess * (target_sum / np.sum(cluster_guess))
    initial_guesses.append(cluster_guess)
    
    # Optimize each initial guess
    for guess in initial_guesses:
        # Objective function with variance penalty
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            var_corr = np.var(corr)
            return max_corr * dx + 1e-3 * var_corr
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 500}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
