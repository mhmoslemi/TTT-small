            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-9, 'maxiter': 2000}
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points
