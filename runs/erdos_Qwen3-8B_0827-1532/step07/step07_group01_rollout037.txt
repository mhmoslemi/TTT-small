from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Optimal balance between speed and precision
    dx = 2.0 / n_points
    
    # Generate initial guess from provided initial_h_values
    initial_guess = np.array(initial_h_values)
    # Rescale to meet the integral constraint
    initial_guess = initial_guess / np.sum(initial_guess) * (n_points / 2)
    
    # Generate a binary initial guess with alternating 1s and 0s
    binary_guess = np.zeros(n_points)
    binary_guess[::2] = 1.0  # Alternate 1 and 0
    binary_guess = binary_guess / np.sum(binary_guess) * (n_points / 2)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Run optimizations for each initial guess with tighter tolerances
    initial_guesses = [initial_guess, binary_guess]
    
    for guess in initial_guesses:
        # Run optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-10, 'maxiter': 3000}  # Increased maxiter and tighter tolerance
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # If no optimization was done, use the initial guess
    if best_h is None:
        result = minimize(

            initial_guess,

        )
