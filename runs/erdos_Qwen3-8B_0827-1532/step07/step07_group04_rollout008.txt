import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = len(initial_guess)

    target_sum = n_points / 2

    initial_guesses = [initial_guess.copy()]

    sinusoidal = 0.5 + 0.5 * np.sin(np.pi * x)
    sinusoidal = np.clip(sinusoidal, 0, 1)
    current_sum = np.sum(sinusoidal)
    scale = target_sum / current_sum
    sinusoidal = sinusoidal * scale
    initial_guesses.append(sinusoidal)
    
    # Triangular
    triangular = np.zeros(n_points)
    for i in range(n_points):
        if x[i] <= 1:
            triangular[i] = x[i] / 1.0
        else:
            triangular[i] = (2 - x[i]) / 1.0
    current_sum = np.sum(triangular)
    scale = target_sum / current_sum
    triangular = triangular * scale
    initial_guesses.append(triangular)
    
    # Square wave
    square_wave = np.zeros(n_points)
    square_wave[:n_points//2] = 1.0
    square_wave[n_points//2:] = 0.0
    initial_guesses.append(square_wave)
    
    # Random binary
    random_binary = np.random.choice([0, 1], size=n_points)
    current_sum = np.sum(random_binary)
    scale = target_sum / current_sum
    random_binary = random_binary * scale
    initial_guesses.append(random_binary)
    
    # Add more perturbed versions
    for guess in initial_guesses:
        noise = np.random.normal(0, 0.1, n_points)
        perturbed = guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        initial_guesses.append(perturbed)
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    start_time = time.time()
    
    for guess in initial_guesses:
        # Add small noise for diversity
        noise = np.random.normal(0, 0.05, n_points)
        perturbed = guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(

            perturbed,

            best_h = result.x
