import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Optimal balance between speed and precision
    dx = 2.0 / n_points
    
    # Generate diverse initial guesses with structured patterns
    initial_guesses = []
    
    # Pattern 1: Single interval
    x_single = np.linspace(0, 2, n_points)
    h_single = np.zeros(n_points)
    h_single[(x_single >= 0.5) & (x_single <= 1.5)] = 1.0
    initial_guesses.append(h_single)
    
    # Pattern 2: Two intervals
    h_two = np.zeros(n_points)
    h_two[(x_single >= 0.0) & (x_single <= 0.5)] = 1.0
    h_two[(x_single >= 1.5) & (x_single <= 2.0)] = 1.0
    initial_guesses.append(h_two)
    
    # Pattern 3: Triangular
    h_tri = np.zeros(n_points)
    for i in range(n_points):
        x = x_single[i]
        if x <= 1.0:
            h_tri[i] = x
        else:
            h_tri[i] = 2.0 - x
    h_tri = h_tri * 2.0  # Scale to meet integral constraint
    initial_guesses.append(h_tri)
    
    # Pattern 4: Sinusoidal
    h_sine = 0.5 * (1.0 + np.sin(np.pi * x_single))
    h_sine = h_sine * 2.0  # Scale to meet integral constraint
    initial_guesses.append(h_sine)
    
    # Add random perturbations to each pattern
    num_initial_guesses = 50  # Increased diversity
    for _ in range(num_initial_guesses):
        # Select a base pattern
        base = np.random.choice(initial_guesses)
        # Add random noise with varying amplitude and frequency
        noise_amp = np.random.uniform(0.001, 0.05)
        noise_freq = np.random.uniform(1, 10)
        noise = noise_amp * np.sin(2 * np.pi * noise_freq * x_single)
        perturbed = base + noise
        # Clip to [0, 1]
        perturbed = np.clip(perturbed, 0, 1)
        # Adjust sum to meet the constraint
        scale_factor = (n_points / 2) / np.sum(perturbed)
        perturbed *= scale_factor
        initial_guesses.append(perturbed)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Run optimizations for each initial guess with tighter tolerances
    for guess in initial_guesses:
        # Run optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-9, 'maxiter': 2000}  # Increased maxiter
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.success:
            current_c5 = result.fun

                best_h = result.x

            options={'ftol': 1e-9, 'maxiter': 2000}
        )
        best_c5 = result.fun
        best_h = result.x
