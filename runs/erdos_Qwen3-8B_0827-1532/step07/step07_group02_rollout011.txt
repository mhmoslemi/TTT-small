def run(seed=42, budget_s=100, **kwargs):

    # Generate initial guesses
    num_initial_guesses = 100  # More guesses for diversity
    initial_guesses = []
    for _ in range(num_initial_guesses):
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
        initial_guesses.append(h)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Run optimizations for each initial guess
    for guess in initial_guesses:
        # Run optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-9, 'maxiter': 500}  # Reduced maxiter
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:

        if result.success:

            options={'ftol': 1e-9, 'maxiter': 500}
