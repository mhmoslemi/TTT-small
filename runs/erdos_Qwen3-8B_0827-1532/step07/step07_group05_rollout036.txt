    # Run optimizations for each initial guess with tighter tolerances
    for guess in initial_guesses:
        # Run optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-9, 'maxiter': 2000}
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # If no optimization was done, use the initial guess
    if best_h is None:
        result = minimize(
            objective,
            initial_guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-9, 'maxiter': 2000}
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points
