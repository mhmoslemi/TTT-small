from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 200  # Increased for better resolution
    dx = 2.0 / n_points
    
    # Generate initial guess based on provided initial_h_values
    x_initial = np.linspace(0, 2, len(initial_h_values))
    x_new = np.linspace(0, 2, n_points)
    initial_guess = np.interp(x_new, x_initial, initial_h_values)
    scale_factor = (n_points / 2) / np.sum(initial_guess)
    initial_guess *= scale_factor
    
    # Generate diverse initial guesses with structured perturbations
    initial_guesses = []
    
    # Sine wave perturbations
    for freq in [0.5, 1.0, 2.0, 3.0]:
        for amp in [0.01, 0.02, 0.03]:
            noise = amp * np.sin(2 * np.pi * freq * x_new)
            perturbed = initial_guess + noise
            perturbed = np.clip(perturbed, 0, 1)
            scale_factor = (n_points / 2) / np.sum(perturbed)
            perturbed *= scale_factor
            initial_guesses.append(perturbed)
    
    # Mirrored version
    mirrored = initial_guess[::-1]
    mirrored = np.clip(mirrored, 0, 1)
    scale_factor = (n_points / 2) / np.sum(mirrored)
    mirrored *= scale_factor
    initial_guesses.append(mirrored)
    
    # Shifted version
    shifted = np.roll(initial_guess, int(0.5 * n_points))
    shifted = np.clip(shifted, 0, 1)
    scale_factor = (n_points / 2) / np.sum(shifted)
    shifted *= scale_factor
    initial_guesses.append(shifted)
    
    # Random perturbations
    for _ in range(2):
        noise = np.random.uniform(-0.01, 0.01, n_points)
        perturbed = initial_guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        scale_factor = (n_points / 2) / np.sum(perturbed)
        perturbed *= scale_factor
        initial_guesses.append(perturbed)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Run optimizations for each initial guess with tighter tolerances
    for guess in initial_guesses:
        # Run optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-9, 'maxiter': 2000}
        )
        elapsed = time.time() - start_time

        if result.success:
            current_c5 = result.fun

        )

        best_h = result.x
