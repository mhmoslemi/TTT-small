import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Balance between resolution and speed
    dx = 2.0 / n_points
    
    # Precompute x grid
    x_new = np.linspace(0, 2, n_points)
    
    # Generate initial guesses
    initial_guesses = []
    
    # 1. Initial construction
    x_initial = np.linspace(0, 2, len(initial_h_values))
    initial_guess = np.interp(x_new, x_initial, initial_h_values)
    scale_factor = (n_points / 2) / np.sum(initial_guess)
    initial_guess *= scale_factor
    initial_guesses.append(initial_guess)
    
    # 2. Binary pattern
    binary_guess = np.zeros(n_points)
    binary_guess[::2] = 1.0
    scale_factor = (n_points / 2) / np.sum(binary_guess)
    binary_guess *= scale_factor
    initial_guesses.append(binary_guess)
    
    # 3. Random pattern
    random_guess = np.random.uniform(0, 1, n_points)
    scale_factor = (n_points / 2) / np.sum(random_guess)
    random_guess *= scale_factor
    initial_guesses.append(random_guess)
    
    # 4. Periodic pattern (sine wave)
    periodic_guess = 0.5 * (1 + np.sin(2 * np.pi * x_new))
    periodic_guess = np.clip(periodic_guess, 0, 1)
    scale_factor = (n_points / 2) / np.sum(periodic_guess)
    periodic_guess *= scale_factor
    initial_guesses.append(periodic_guess)
    
    # 5. Clustered pattern
    cluster_guess = np.zeros(n_points)
    for i in range(5):
        start = i * 0.4
        end = start + 0.2
        cluster_guess[int(start/dx) : int(end/dx)] = 1.0
    scale_factor = (n_points / 2) / np.sum(cluster_guess)
    cluster_guess *= scale_factor
    initial_guesses.append(cluster_guess)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    
    # Run optimizations for each initial guess
    for guess in initial_guesses:
        # Run optimization with L-BFGS-B
        result = minimize(
            objective,
            guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-10, 'maxiter': 1000}
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # Fallback to initial guess if no improvement
    if best_h is None:
        result = minimize(
            objective,
            initial_guesses[0],
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-10, 'maxiter': 1000}
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points
