import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Balanced between speed and precision
    dx = 2.0 / n_points
    x_new = np.linspace(0, 2, n_points)
    
    # Generate initial guesses
    initial_guesses = []
    
    # Add initial_h_values as the first guess
    initial_guesses.append(initial_h_values)
    
    # Add square wave
    square_wave = np.zeros(n_points)
    square_wave[x_new < 1.0] = 1.0
    initial_guesses.append(square_wave)
    
    # Add triangular wave
    triangular_wave = np.zeros(n_points)
    triangular_wave = np.clip(x_new / 1.0, 0, 1)
    triangular_wave[x_new > 1.0] = np.clip((2.0 - x_new) / 1.0, 0, 1)
    initial_guesses.append(triangular_wave)
    
    # Add random guesses
    for _ in range(5):  # Fewer random guesses
        random_h = np.random.uniform(0, 1, n_points)
        random_h = random_h / np.sum(random_h) * (n_points / 2)
        initial_guesses.append(random_h)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Run optimizations for each initial guess
    for guess in initial_guesses:
        # Run optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-8, 'maxiter': 1000}  # Adjusted parameters
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # If no optimization was done, use the initial guess
    if best_h is None:
        result = minimize(
            objective,
            initial_guesses[0],
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-8, 'maxiter': 1000}
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points
