import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Balance between detail and speed
    dx = 2.0 / n_points
    
    # Generate diverse initial guesses
    initial_guesses = []
    
    # 1. Initial_h_values (predefined)
    x_initial = np.linspace(0, 2, len(initial_h_values))

    initial_guess = np.interp(x_new, x_initial, initial_h_values)
    initial_guess = initial_guess / np.sum(initial_guess) * (n_points / 2)
    initial_guesses.append(initial_guess)
    
    # 2. Sine wave pattern
    sine_guess = 0.5 * (1 + np.sin(np.pi * x_new))
    sine_guess = sine_guess / np.sum(sine_guess) * (n_points / 2)
    initial_guesses.append(sine_guess)
    
    # 3. Triangular wave pattern
    triangular_guess = np.zeros(n_points)
    for i, x in enumerate(x_new):
        if x <= 1:
            triangular_guess[i] = x
        else:
            triangular_guess[i] = 2 - x
    triangular_guess = triangular_guess / np.sum(triangular_guess) * (n_points / 2)
    initial_guesses.append(triangular_guess)
    
    # 4. Random binary pattern
    random_binary = np.random.rand(n_points) > 0.5
    random_binary = random_binary.astype(float)
    random_binary = random_binary / np.sum(random_binary) * (n_points / 2)
    initial_guesses.append(random_binary)
    
    # 5. Alternating binary pattern
    binary_guess = np.zeros(n_points)
    binary_guess[::2] = 1.0
    binary_guess = binary_guess / np.sum(binary_guess) * (n_points / 2)
    initial_guesses.append(binary_guess)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Run optimizations for each initial guess with tighter tolerances
    for guess in initial_guesses:
        # Run optimization
        result = minimize(
            objective,
            guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-8, 'maxiter': 1000}  # Balanced speed and accuracy
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.success:
            current_c5 = result.fun

        )
