import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 50  # Reduced for faster evaluation
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    x = np.linspace(0, 2, n_points)
    
    # Generate diverse initial guesses
    initial_guesses = []
    
    # Sinusoidal initial guess
    sinusoidal = 0.5 + 0.5 * np.sin(np.pi * x)
    sinusoidal = np.clip(sinusoidal, 0, 1)
    current_sum = np.sum(sinusoidal)
    scale = target_sum / current_sum
    sinusoidal = sinusoidal * scale
    initial_guesses.append(sinusoidal)
    
    # Binary step function
    binary_step = np.zeros(n_points)
    binary_step[:n_points//2] = 1.0
    current_sum = np.sum(binary_step)
    scale = target_sum / current_sum
    binary_step = binary_step * scale
    initial_guesses.append(binary_step)
    
    # Random binary initial guess
    random_binary = np.random.choice([0, 1], size=n_points)
    current_sum = np.sum(random_binary)
    scale = target_sum / current_sum
    random_binary = random_binary * scale
    initial_guesses.append(random_binary)
    
    # Random initial guess with smooth variation
    random_smooth = np.random.uniform(0, 1, n_points)
    current_sum = np.sum(random_smooth)
    scale = target_sum / current_sum
    random_smooth = random_smooth * scale
    initial_guesses.append(random_smooth)
    
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    start_time = time.time()
    
    for guess in initial_guesses:
        # Add small noise to avoid local minima

        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum

        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 500}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
