from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 50  # Smaller grid for faster evaluation
    dx = 2.0 / n_points
    target_sum = n_points / 2

    initial_guesses = []
