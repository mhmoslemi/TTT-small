import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Balanced between speed and precision
    dx = 2.0 / n_points
    
    # Generate structured initial guesses
    x = np.linspace(0, 2, n_points)
    # Original initial guess
    initial_guess = initial_h_values
    # Symmetric version
    symmetric_initial = np.zeros(n_points)
    for i in range(n_points):
        symmetric_initial[i] = initial_h_values[n_points - 1 - i]
    # Binary step function
    binary_step = np.zeros(n_points)
    binary_step[:50] = 1.0
    # Sinusoidal function
    sinusoidal = 0.5 + 0.5 * np.sin(np.pi * x)
    sinusoidal = sinusoidal / np.sum(sinusoidal) * (n_points / 2)
    
    initial_guesses = [initial_guess, symmetric_initial, binary_step, sinusoidal]
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Run optimizations for each initial guess
    for guess in initial_guesses:
        # Run optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-6, 'maxiter': 200}
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # If no optimization was done, use the initial guess
    if best_h is None:
        result = minimize(
            objective,
            initial_guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-6, 'maxiter': 200}
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points
