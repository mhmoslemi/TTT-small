import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate diverse initial guesses
    x = np.linspace(0, 2, n_points)
    initial_guesses = [initial_guess.copy()]
    
    # Triangular wave
    triangular = np.minimum(x, 2 - x) / 1.0
    initial_guesses.append(triangular)
    
    # Square wave
    square = np.where(x < 1, 1, 0)
    initial_guesses.append(square)
    
    # Sinusoidal wave
    sinusoidal = 0.5 + 0.5 * np.sin(np.pi * x)
    sinusoidal = np.clip(sinusoidal, 0, 1)
    current_sum = np.sum(sinusoidal)
    scale = target_sum / current_sum
    sinusoidal = sinusoidal * scale
    initial_guesses.append(sinusoidal)
    
    # Additional initial guess: random binary pattern
    binary = np.random.choice([0, 1], size=n_points)
    current_sum = np.sum(binary)
    scale = target_sum / current_sum
    binary = binary * scale
    initial_guesses.append(binary)
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    start_time = time.time()
    
    for guess in initial_guesses:
        # Add noise and scale
        noise = np.random.normal(0, 0.05, n_points)
        perturbed = guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 1500}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
