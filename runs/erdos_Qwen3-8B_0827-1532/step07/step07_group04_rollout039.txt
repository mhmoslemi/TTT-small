import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    # Assume initial_h_values is provided as a global variable
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate diverse initial guesses
    initial_guesses = []
    initial_guesses.append(initial_h_values.copy())
    
    # Uniform distribution
    uniform = 0.5 * np.ones(n_points)
    initial_guesses.append(uniform)
    
    # Random binary
    if n_points % 2 == 0:
        indices = np.random.choice(n_points, size=n_points//2, replace=False)
        random_binary = np.zeros(n_points)
        random_binary[indices] = 1.0
        initial_guesses.append(random_binary)
    
    # Structured initial guess with 3 intervals
    def generate_structured(n_points):
        l = 1.0 / 3
        g = (2.0 - 3*l) / (3-1)
        intervals = []
        start = 0.0
        for _ in range(3):
            end = start + l
            intervals.append((start, end))
            start = end + g
        x = np.linspace(0, 2, n_points)
        h = np.zeros(n_points)
        for i in range(n_points):
            for (s, e) in intervals:
                if s <= x[i] < e:
                    h[i] = 1.0
                    break
        return h
    structured = generate_structured(n_points)
    initial_guesses.append(structured)
    
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    start_time = time.time()
    
    for guess in initial_guesses:
        # Add small noise to perturb the guess
        noise = np.random.normal(0, 0.05, n_points)
        perturbed = guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Use SLSQP with tighter tolerances
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-12, 'maxiter': 2000}
        )
        
        if result.success and result.fun < best_c5:

            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
