import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Reduced for faster evaluation
    dx = 2.0 / n_points
    
    # Generate initial guesses
    initial_guesses = []
    
    # Add initial_h_values as a base guess
    initial_guesses.append(initial_h_values)
    
    # Add structured guess with two intervals of 0.5 length
    structured_guess = np.zeros(n_points)
    structured_guess[0:25] = 1.0  # [0, 0.5]
    structured_guess[50:75] = 1.0  # [1.0, 1.5]
    initial_guesses.append(structured_guess)
    
    # Add random guesses
    for _ in range(20):
        h_rand = np.random.rand(n_points)
        h_rand = h_rand / np.sum(h_rand) * (n_points / 2)
        initial_guesses.append(h_rand)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    
    # Run optimizations for each initial guess
    for guess in initial_guesses:
        # Run optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-9, 'maxiter': 500}
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # Fallback to initial guess if no optimization was successful
    if best_h is None:
        result = minimize(
            objective,
            initial_guesses[0],
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-9, 'maxiter': 500}
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points
