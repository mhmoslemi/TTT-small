from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Reduced points for faster evaluation
    dx = 2.0 / n_points
    
    # Generate initial guesses
    initial_guesses = []
    
    # Interpolate initial_h_values
    x_initial = np.linspace(0, 2, len(initial_h_values))
    x_new = np.linspace(0, 2, n_points)
    initial_guess = np.interp(x_new, x_initial, initial_h_values)
    initial_guess = initial_guess / np.sum(initial_guess) * (n_points / 2)
    initial_guesses.append(initial_guess)
    
    # Binary guess
    binary_guess = np.zeros(n_points)
    binary_guess[::2] = 1.0
    binary_guess = binary_guess / np.sum(binary_guess) * (n_points / 2)
    initial_guesses.append(binary_guess)
    
    # Sinusoidal guess
    sinusoidal_guess = 0.5 + 0.5 * np.sin(np.pi * x_new)
    sum_sinusoidal = np.sum(sinusoidal_guess) * dx

    sinusoidal_guess *= scale_factor
    initial_guesses.append(sinusoidal_guess)
    
    # Random noise guesses
    for _ in range(5):
        random_noise = np.random.normal(0, 0.05, n_points)
        perturbed = initial_guess + random_noise

        scale_factor = (n_points / 2) / np.sum(perturbed)
        perturbed *= scale_factor
        initial_guesses.append(perturbed)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Run optimizations for each initial guess
    for guess in initial_guesses:
        # Run optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-10, 'maxiter': 3000}  # Increased maxiter and tighter tolerance
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:

        )

        best_h = result.x
