import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 200  # Increased for better precision while keeping it manageable
    dx = 2.0 / n_points
    
    # Generate initial guesses
    initial_guesses = []
    
    # 1. Interpolated initial guess from provided initial_h_values
    x_initial = np.linspace(0, 2, len(initial_h_values))
    x_new = np.linspace(0, 2, n_points)
    initial_guess = np.interp(x_new, x_initial, initial_h_values)
    # Ensure sum constraint
    sum_initial = np.sum(initial_guess) * dx
    initial_guess = initial_guess / sum_initial * (n_points / 2)
    initial_guesses.append(initial_guess)
    
    # 2. Periodic guess: 1 on [0, 0.5] and [1, 1.5], 0 elsewhere
    periodic_h = np.zeros(n_points)
    for i in range(n_points):
        x = i * dx
        if (x >= 0 and x < 0.5) or (x >= 1 and x < 1.5):
            periodic_h[i] = 1.0
        else:
            periodic_h[i] = 0.0
    initial_guesses.append(periodic_h)
    
    # 3. Random guess
    random_h = np.random.uniform(0, 1, n_points)
    random_h = random_h / np.sum(random_h) * (n_points / 2)
    initial_guesses.append(random_h)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Run optimizations for each initial guess with tighter tolerances
    for guess in initial_guesses:
        # Run optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-9, 'maxiter': 2000}
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # If no optimization was done, use the initial guess
    if best_h is None:
        result = minimize(
            objective,
            initial_guesses[0],
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-9, 'maxiter': 2000}
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points
