import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Reduced for faster evaluation
    dx = 2.0 / n_points
    
    # Generate initial guess based on provided initial_h_values
    initial_guess = initial_h_values
    
    # Generate diverse initial guesses
    num_initial_guesses = 50  # Reduced to fit within time budget
    initial_guesses = []
    
    # Add initial guess and its symmetric variant
    initial_guesses.append(initial_guess)
    symmetric_guess = np.zeros(n_points)
    for i in range(n_points):
        symmetric_guess[i] = initial_guess[n_points - 1 - i]
    initial_guesses.append(symmetric_guess)
    
    # Generate perturbed initial guesses
    for _ in range(num_initial_guesses // 2 - 1):
        noise = np.random.normal(0, 0.01, n_points)
        perturbed = initial_guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        scale_factor = (n_points / 2) / np.sum(perturbed)
        perturbed *= scale_factor
        initial_guesses.append(perturbed)
    
    # Generate random binary initial guesses
    for _ in range(num_initial_guesses // 2):
        indices = np.random.choice(n_points, size=int(n_points / 2), replace=False)
        h_rand = np.zeros(n_points)
        h_rand[indices] = 1.0
        h_rand = h_rand / np.sum(h_rand) * (n_points / 2)
        initial_guesses.append(h_rand)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Run optimizations for each initial guess
    for guess in initial_guesses:
        # Run optimization with tighter tolerances
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-10, 'maxiter': 500}
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # Fallback to initial guess if no optimization was successful
    if best_h is None:
        result = minimize(
            objective,
            initial_guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-10, 'maxiter': 500}
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points
