        perturbed = initial_guess + noise

        perturbed *= scale_factor

    for _ in range(num_initial_guesses // 2):

        initial_guesses.append(random_h)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2

    for guess in initial_guesses:
        # Run optimization
        result = minimize(
