        square_wave = square_wave / np.sum(square_wave) * (n_points / 2)
        initial_guesses.append(square_wave)
    
    # Generate triangular waves with varying slopes
    for slope in np.linspace(0.5, 2.0, 5):
        triangular_wave = np.zeros(n_points)
        for i in range(n_points):
            x = x_new[i]
            if x < 1.0:
                triangular_wave[i] = x * slope
            else:
                triangular_wave[i] = (2.0 - x) * slope
        triangular_wave = triangular_wave / np.sum(triangular_wave) * (n_points / 2)
        initial_guesses.append(triangular_wave)
    
    # Generate sine waves with varying frequencies and amplitudes
    for freq in np.linspace(0.5, 5.0, 5):
        for amp in np.linspace(0.1, 0.5, 5):
            sine_wave = np.zeros(n_points)
            for i in range(n_points):
                sine_wave[i] = amp * np.sin(2 * np.pi * freq * x_new[i])
            sine_wave = (sine_wave - np.min(sine_wave)) / (np.max(sine_wave) - np.min(sine_wave))
            sine_wave = sine_wave / np.sum(sine_wave) * (n_points / 2)
            initial_guesses.append(sine_wave)
    
    # Generate random perturbations
    for _ in range(20):
        # Perturb with sine wave
        noise_amp = np.random.uniform(0.001, 0.05)
        noise_freq = np.random.uniform(0.5, 10)
        noise = noise_amp * np.sin(2 * np.pi * noise_freq * x_new)
        perturbed = initial_h_values + noise
        perturbed = np.clip(perturbed, 0, 1)
        scale_factor = (n_points / 2) / np.sum(perturbed)
        perturbed *= scale_factor
        initial_guesses.append(perturbed)
        # Add a random guess
        random_h = np.random.uniform(0, 1, n_points)
        random_h = random_h / np.sum(random_h) * (n_points / 2)
        initial_guesses.append(random_h)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Run optimizations for each initial guess
    for guess in initial_guesses:
        # Run optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-9, 'maxiter': 1500}
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # If no optimization was done, use the initial guess
    if best_h is None:
        result = minimize(
            objective,
            initial_guesses[0],
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-9, 'maxiter': 1500}
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points
