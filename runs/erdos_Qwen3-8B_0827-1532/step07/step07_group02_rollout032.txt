from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    dx = 2.0 / n_points

    x_initial = np.linspace(0, 2, len(initial_h_values))

    initial_guess = np.interp(x_new, x_initial, initial_h_values)
    # Adjust sum to meet the constraint
    initial_guess = initial_guess / np.sum(initial_guess) * (n_points / 2)
    
    # Generate diverse initial guesses with sinusoidal perturbations
    num_initial_guesses = 30  # Reduced diversity for speed
    initial_guesses = []
    for _ in range(num_initial_guesses):
        # Add sinusoidal perturbation with varying parameters
        freq = np.random.uniform(1, 10)
        phase = np.random.uniform(0, 2 * np.pi)
        amp = np.random.uniform(0.01, 0.1)
        noise = amp * np.sin(2 * np.pi * freq * x_new + phase)
        perturbed = initial_guess + noise
        # Clip to [0, 1]
        perturbed = np.clip(perturbed, 0, 1)
        # Adjust sum to meet the constraint
        scale_factor = (n_points / 2) / np.sum(perturbed)
        perturbed *= scale_factor
        initial_guesses.append(perturbed)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Run optimizations for each initial guess with tighter tolerances
    for guess in initial_guesses:
        # Run optimization
        result = minimize(
            objective,
            guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-9, 'maxiter': 1000}  # Adjusted method and parameters
        )
        elapsed = time.time() - start_time
