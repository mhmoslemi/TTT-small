import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    # Use the provided initial guess
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = initial_guess.copy()
    
    start_time = time.time()
    
    # Generate multiple diverse initial guesses based on best_h and random
    num_perturbations = 100  # Increased number of perturbations
    for perturbation_idx in range(num_perturbations):
        # Generate perturbations based on best_h
        if perturbation_idx < 10:
            # Random initial guess based on best_h
            random_noise = np.random.uniform(-0.2, 0.2, n_points)
            perturbed = best_h + random_noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
        else:
            # Perturb best_h with varying noise
            if perturbation_idx < 30:
                # Larger noise for early perturbations
                noise = np.random.uniform(-0.2, 0.2, n_points)
            else:
                # Smaller noise for later perturbations
                noise = np.random.uniform(-0.05, 0.05, n_points)
            perturbed = best_h + noise
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            scale = target_sum / current_sum
            perturbed = perturbed * scale
        
        # Generate a few random initial guesses
        if perturbation_idx % 5 == 0:
            # Generate a random initial guess
            random_guess = np.random.uniform(0, 1, n_points)
            current_sum = np.sum(random_guess)
            scale = target_sum / current_sum
            random_guess = random_guess * scale
            perturbed = random_guess
        
        # Objective function to minimize C5
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints: sum(h) == target_sum
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Perform optimization with tighter tolerances and more iterations
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 500}
        )

            best_h = result.x

        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
