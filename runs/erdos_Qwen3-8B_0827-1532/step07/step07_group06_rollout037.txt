from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    
    # Generate initial guess based on provided initial_h_values
    initial_guess = initial_h_values
    
    # Generate diverse initial guesses
    num_initial_guesses = 50
    initial_guesses = []

    initial_guesses.append(initial_guess)
    
    # Generate perturbed initial guesses
    for _ in range(num_initial_guesses // 2 - 1):
        noise = np.random.normal(0, 0.01, n_points)
        perturbed = initial_guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        scale_factor = (n_points / 2) / np.sum(perturbed)
        perturbed *= scale_factor
        initial_guesses.append(perturbed)
    
    # Generate structured initial guesses
    for _ in range(num_initial_guesses // 2):
        # Create a structured guess with two intervals
        structured = np.zeros(n_points)
        # First interval: [0, 0.5]
        start = 0
        end = 0.5
        idx_start = int(start / dx)
        idx_end = int(end / dx)
        structured[idx_start:idx_end] = 1.0
        # Second interval: [1.0, 1.5]
        start = 1.0
        end = 1.5
        idx_start = int(start / dx)
        idx_end = int(end / dx)
        structured[idx_start:idx_end] = 1.0
        # Normalize
        total = np.sum(structured) * dx
        scale = (n_points / 2) / total
        structured *= scale
        initial_guesses.append(structured)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Run optimizations for each initial guess
    for guess in initial_guesses:
        # Run optimization
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-10, 'maxiter': 1000}
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.success:
            current_c5 = result.fun

                best_h = result.x
