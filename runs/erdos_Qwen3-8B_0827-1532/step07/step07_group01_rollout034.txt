    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Function to run optimization for a single guess
    def optimize_guess(guess):
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-10, 'maxiter': 3000}
        )
        if result.success:
            return result.fun, result.x
        else:
            return float('inf'), None
    
    # Use multiprocessing to run optimizations in parallel
    with Pool(processes=cpu_count()) as pool:
        results = pool.map(optimize_guess, initial_guesses)
    
    # Collect the best result
    for c5, h in results:
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    
    # If no result was found, use the base guess
    if best_h is None:
        result = minimize(
            objective,
            base_guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-10, 'maxiter': 3000}
        )
        best_c5 = result.fun
        best_h = result.x
    
    # Check if we exceeded the budget and return the best found so far
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
