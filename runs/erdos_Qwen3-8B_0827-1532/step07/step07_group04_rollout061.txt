import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def scale_h(h):
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scale = target_sum / current_sum
        scaled = h * scale
        scaled = np.clip(scaled, 0, 1)
        return scaled
    
    def objective(h):
        h_scaled = scale_h(h)
        one_minus_h = 1.0 - h_scaled
        corr = np.correlate(h_scaled, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial guess based on provided initial_h_values
    initial_guess = initial_h_values.copy()
    initial_guess = scale_h(initial_guess)
    
    start_time = time.time()
    
    # Run differential evolution with time constraint
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.8,
        tol=1e-12,
        maxiter=100,
        seed=seed,
        polish=True,
        disp=False
    )
    
    # Check elapsed time and possibly refine if budget allows
    elapsed = time.time() - start_time
    if elapsed < budget_s:
        # Additional refinement with local optimization
        from scipy.optimize import minimize
        def local_objective(h):
            h_scaled = scale_h(h)
            one_minus_h = 1.0 - h_scaled
            corr = np.correlate(h_scaled, one_minus_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        bounds_local = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result_local = minimize(
            local_objective,
            result.x,
            method='SLSQP',
            bounds=bounds_local,
            constraints=cons,
            options={'ftol': 1e-12, 'maxiter': 200}
        )
        
        if result_local.success:
            best_h = result_local.x
            best_c5 = result_local.fun
        else:
            best_h = result.x
            best_c5 = result.fun
    else:
        best_h = result.x
        best_c5 = result.fun
    
    return best_h, best_c5, n_points
