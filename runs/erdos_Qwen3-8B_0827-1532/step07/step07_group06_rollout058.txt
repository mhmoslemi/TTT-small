        h_rand = h_rand / np.sum(h_rand) * (n_points / 2)
        initial_guesses.append(h_rand)
    
    # Objective function
    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Run optimizations for each initial guess
    for guess in initial_guesses:
        # Run optimization
        result = minimize(
            objective,
            guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-10, 'maxiter': 500}
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # Fallback to initial guess if no optimization was successful
    if best_h is None:
        result = minimize(
            objective,
            initial_h_values,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-10, 'maxiter': 500}
        )
        best_c5 = result.fun
        best_h = result.x
    
    return best_h, best_c5, n_points
