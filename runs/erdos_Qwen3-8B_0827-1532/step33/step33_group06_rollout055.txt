    sum_h = np.sum(h_two)

    h_six[2*segment_length:3*segment_length] = 1.0

    h_six[4*segment_length:5*segment_length] = 1.0

    # Optimizing each initial guess with differential evolution
    for guess in initial_guesses:
        # Define objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            c5 = np.max(corr) * dx
            penalty = 1e5 * np.abs(np.sum(h) - target_sum)
            return c5 + penalty
        
        # Bounds for each h[i]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Run differential evolution
        result = differential_evolution(

        elapsed = time.time() - start_time

def compute_c5(h, dx):
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return np.max(corr) * dx
