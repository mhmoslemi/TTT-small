        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        # Use differential evolution for some initial guesses
        if np.random.rand() < 0.5:
            result = differential_evolution(
                objective,
                bounds,
                constraints=cons,
                strategy='best1bin',
                popsize=15,
                mutation=(0.5, 1),
                recombination=0.9,
                tol=1e-10,
                maxiter=100
            )
            if result.success and result.fun < best_c5:
                best_c5 = result.fun
                best_h = result.x
        else:
            result = minimize(
                objective,
                guess,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'ftol': 1e-12, 'maxiter': 5000}
            )
            if result.success and result.fun < best_c5:
                best_c5 = result.fun
                best_h = result.x
    
    # Perturbation phase on the best_h found so far
    num_perturbations = 500
    phase_shift = 0.25
    phase_decay = 0.99
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-12, 'maxiter': 1000}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points

def compute_c5(h, dx):
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return np.max(corr) * dx
