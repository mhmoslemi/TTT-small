    # Initial guess: provided initial_h_values
    initial_guess = initial_h_values
    initial_guess = np.clip(initial_guess, 0, 1)
    scale = target_sum / np.sum(initial_guess)
    initial_guess = initial_guess * scale
    best_h = initial_guess.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    
    # Generate diverse initial guesses
    initial_guesses = [initial_guess]
    
    # Add random initial guesses with more diversity
    for _ in range(10):
        h_rand = np.random.rand(n_points)
        sum_h = np.sum(h_rand)
        scale = target_sum / sum_h
        h_rand = h_rand * scale
        initial_guesses.append(h_rand)
    
    # Add predefined segment patterns
    # Two-segment pattern
    h_two = np.zeros(n_points)
    segment_length = n_points // 2
    h_two[:segment_length] = 1.0
    h_two[segment_length:] = 0.5
    sum_h = np.sum(h_two)
    scale = target_sum / sum_h
    initial_guesses.append(h_two * scale)
    
    # Three-segment pattern
    h_three = np.zeros(n_points)
    segment_length = n_points // 3
    h_three[:segment_length] = 1.0
    h_three[segment_length:2*segment_length] = 0.75
    h_three[2*segment_length:] = 0.5
    sum_h = np.sum(h_three)
    scale = target_sum / sum_h
    initial_guesses.append(h_three * scale)
    
    # Four-segment pattern
    h_four = np.zeros(n_points)
    segment_length = n_points // 4
    h_four[:segment_length] = 1.0
    h_four[segment_length:2*segment_length] = 0.5
    h_four[2*segment_length:3*segment_length] = 1.0
    h_four[3*segment_length:] = 0.5
    sum_h = np.sum(h_four)
    scale = target_sum / sum_h
    initial_guesses.append(h_four * scale)
    
    # Five-segment pattern
    h_five = np.zeros(n_points)
    segment_length = n_points // 5
    h_five[:segment_length] = 1.0
    h_five[segment_length:2*segment_length] = 0.5
    h_five[2*segment_length:3*segment_length] = 1.0
    h_five[3*segment_length:4*segment_length] = 0.5
    h_five[4*segment_length:] = 1.0
    sum_h = np.sum(h_five)
    scale = target_sum / sum_h
    initial_guesses.append(h_five * scale)
    
    # Six-segment pattern
    h_six = np.zeros(n_points)
    segment_length = n_points // 6
    h_six[:segment_length] = 1.0
    h_six[segment_length:2*segment_length] = 0.5
    h_six[2*segment_length:3*segment_length] = 1.0
    h_six[3*segment_length:4*segment_length] = 0.5
    h_six[4*segment_length:5*segment_length] = 1.0
    h_six[5*segment_length:] = 0.5
    sum_h = np.sum(h_six)
    scale = target_sum / sum_h
    initial_guesses.append(h_six * scale)
    
    # Optimizing each initial guess
    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            relevant_corr = corr[n_points-1 : 2*n_points-1]
            return np.max(relevant_corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-12, 'maxiter': 3000}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Perturbation phase on the best_h found so far
    num_perturbations = 300
    phase_shift = 0.15
    phase_decay = 0.96
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        noise = np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            relevant_corr = corr[n_points-1 : 2*n_points-1]
            return np.max(relevant_corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,

            options={'ftol': 1e-12, 'maxiter': 600}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points

def compute_c5(h, dx, n_points):
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    relevant_corr = corr[n_points-1 : 2*n_points-1]
    return np.max(relevant_corr) * dx
