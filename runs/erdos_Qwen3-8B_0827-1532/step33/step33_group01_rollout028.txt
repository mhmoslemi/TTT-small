    sum_h = np.sum(h_seven)
    scale = target_sum / sum_h
    initial_guesses.append(h_seven * scale)
    
    # Eight-segment pattern
    h_eight = np.zeros(n_points)
    segment_length = n_points // 8
    h_eight[:segment_length] = 1.0
    h_eight[segment_length:2*segment_length] = 0.5
    h_eight[2*segment_length:3*segment_length] = 1.0
    h_eight[3*segment_length:4*segment_length] = 0.5
    h_eight[4*segment_length:5*segment_length] = 1.0
    h_eight[5*segment_length:6*segment_length] = 0.5
    h_eight[6*segment_length:7*segment_length] = 1.0
    h_eight[7*segment_length:] = 0.5
    sum_h = np.sum(h_eight)
    scale = target_sum / sum_h
    initial_guesses.append(h_eight * scale)
    
    # Optimizing each initial guess
    for guess in initial_guesses:
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-12, 'maxiter': 2000}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # Perturbation phase on the best_h found so far
    num_perturbations = 500
    phase_shift = 0.25
    phase_decay = 0.99
    
    for perturbation_idx in range(num_perturbations):
        noise_scale = phase_shift * (phase_decay ** perturbation_idx)
        # Create a perturbed version by modifying a subset of points
        indices = np.random.choice(n_points, size=10, replace=False)
        noise = np.zeros(n_points)
        noise[indices] = np.random.uniform(-noise_scale, noise_scale, size=10)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale = target_sum / current_sum
        perturbed = perturbed * scale
        
        # Objective function
        def objective(h):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Optimization with SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-12, 'maxiter': 1000}
        )
        
        # Handle failure
        if not result.success:
            current_c5 = compute_c5(perturbed, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = perturbed.copy()
            continue
        
        if result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points

def compute_c5(h, dx):
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return np.max(corr) * dx
