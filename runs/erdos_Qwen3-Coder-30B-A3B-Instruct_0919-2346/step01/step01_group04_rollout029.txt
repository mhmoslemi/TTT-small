import numpy as np
import time
import random
from scipy.optimize import differential_evolution, minimize


def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    overlap = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(overlap) * dx
    return c5


def project_to_feasible(h):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) * dx = 1"""
    # Clip to bounds
    h = np.clip(h, 0, 1)
    # Normalize
    dx = 2.0 / len(h)
    h = h * (1.0 / np.sum(h)) / dx
    return h
