import numpy as np
import time

def softmax_weights(fs, gamma):
    """Compute soft-max weights from a list of function values"""
    exp_fs = np.exp(gamma * fs)
    return exp_fs / np.sum(exp_fs)

def compute_shifted_indices(n, s):
    """Return indices of h_i and h_{i+s} for a given shift s"""
    i_indices = np.arange(n)
    is_indices = i_indices + s
    valid = (is_indices >= 0) & (is_indices < n)
    return i_indices[valid], is_indices[valid]

def compute_gradient(h, dx):
    """Compute ∇ f_s for all shifts s based on current h"""
    n = len(h)
    grad = np.zeros(n)
    for s in range(-n+1, n):
        i_indices, is_indices = compute_shifted_indices(n, s)
        if len(i_indices) == 0:
            continue
        # g_i = 1_{i+s<n} (1 - h_{i+s}) - 1_{i-s>=0} h_{i-s}
        # But h_{i+s} (if i+s < n) and h_{i-s} (if i-s >= 0)
        g = np.zeros(n)
        g[i_indices] = 1 - h[is_indices]
        if s < 0:
            i_minus_s = i_indices - s
            valid_minus_s = i_minus_s < n
            g[i_indices[valid_minus_s]] -= h[i_minus_s[valid_minus_s]]
        # Add weighted gradient
        grad += g
    return dx * grad

def project_to_simplex(h, target_sum):
    """Project vector h onto the simplex sum(h_i)=target_sum with 0 ≤ h_i ≤ 1"""
    # Use bisection on the Lagrange multiplier
    # Define φ(λ) = sum_i clip(h_i - λ, 0, 1)
    h = np.asarray(h)
    a, b = np.min(h), np.max(h) + 1.0  # upper bound on λ
    tol = 1e-12
    while b - a >= tol:
        mid = (a + b) / 2
        proj = np.clip(h - mid, 0, 1)
        sum_proj = np.sum(proj)
        if sum_proj > target_sum:
            a = mid
        else:
            b = mid
    proj = np.clip(h - a, 0, 1)
    return proj

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value scaled by dx
    max_overlap = np.max(correlation) * dx
    return max_overlap

def upsample_initial(h0, n):
    """Linearly upsample h0 (length nc) to vector h (length n)"""
    nc = len(h0)
    x0 = np.linspace(0, 1, nc)
    x = np.linspace(0, 1, n)
    h_upsampled = np.interp(x, x0, h0)
    return h_upsampled

    dx = 2.0 / n
