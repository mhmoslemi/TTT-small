import numpy as np
import time
import cvxpy as cp
from cvxpy import ECOS, GLPK_MI, CBC

def project_sum(h, target_sum):
    """Project h onto the sum = target_sum using waterfilling."""
    h = np.array(h)
    if np.abs(np.sum(h) - target_sum) < 1e-10:
        return h
    # Water-filling to adjust h to sum to target_sum
    # Sort in descending order
    idx = np.argsort(h)[::-1]
    h_sorted = h[idx]
    remaining = target_sum - np.sum(h_sorted)
    # Find the water level
    n = len(h_sorted)
    w = 0.0
    for i in range(n):
        w = (remaining + np.sum(h_sorted[:i])) / (i + 1)
        if w <= h_sorted[i]:
            break
        remaining -= h_sorted[i]
    else:
        # All elements should be set to w
        w = (target_sum - np.sum(h_sorted)) / n
    h_new = np.copy(h)
    # Apply water level
    for i in range(n):
        if h_sorted[i] > w:
            h_new[idx[i]] = w
    h_new = np.clip(h_new, 0, 1)
    return h_new

def compute_c5(h):
    """Compute the maximum overlap integral."""
    dx = 2.0 / len(h)
    c = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(c) * dx
    return c5

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    N = 140
    dx = 2.0 / N

    # Phase 0: Set up data structures
    # Build the set of all valid (i,s) pairs for the McCormick constraints
    pairs = []
    for s in range(N):
        for i in range(N - s):
            pairs.append((i, s))
    
    if len(pairs) < 100:
        raise ValueError("Too few pairs.")

    # Phase 1: Solve LP relaxation
    # Variables
    h = cp.Variable(N, name='h')
    y = cp.Variable(len(pairs), name='y')
    alpha = cp.Variable(name='alpha')
    
    # Constraints
    constraints = []
    # Box constraints
    constraints.extend([0 <= h[i] for i in range(N)])
    constraints.extend([h[i] <= 1 for i in range(N)])
    
    # Sum constraint
    constraints.append(cp.sum(h) == N/2)
    
    # McCormick constraints for each pair
    # i corresponds to the index in pairs
    for i, (idx, s) in enumerate(pairs):
        # y[i] is h[idx] * h[idx+s]
        constraints.append(y[i] <= h[idx])                # (1)
        constraints.append(y[i] <= h[idx + s])            # (2)
        constraints.append(y[i] >= h[idx] + h[idx + s] - 1) # (3)
        constraints.append(y[i] >= 0)                     # (4)

    # Maximum overlap constraints
    # Compute indices for each shift
    shifts = list(range(N))
    # Precompute shifted indices for each shift
    shift_indices = []
    for s in shifts:
        shift_indices.append((s, []))
        for i in range(N - s):
            # index in pairs[] for this term
            idx_pair = i + (N - (s+1)) + s * (N - (s+1))
            shift_indices[-1][1].append(idx_pair)
    
    # Create new indices mapping
    shift_pair_map = {}
    for s in shifts:
        # Find the index of the first (i,s) in pairs list
        pair_off = 0
        for k, (i, ss) in enumerate(pairs):
            if ss == s and i == 0:
                pair_off = k
                break
        shift_pair_map[s] = [k for k in range(pair_off, pair_off + (N - s))]
    
    # For each shift s, add the constraint
    for s in shifts:
        if s not in shift_pair_map:
            continue
        y_idx_list = shift_pair_map[s]
        sum_y_s = cp.sum(y[y_idx_list])
        # Constraint is alpha >= dx*(N/2 - sum_y_s)
        constraints.append(alpha >= dx * (N/2 - sum_y_s))

    # Objective
    objective = cp.Minimize(alpha)
    prob = cp.Problem(objective, constraints)
    
    # Solve LP
    try:
        prob.solve(solver=ECOS, max_iters=20000)
        h_lp = h.value
        alpha_lp = alpha.value
        c5_lp = compute_c5(h_lp)
        if c5_lp <= 0.3807:
            # Project to sum exactly
            h_proj = project_sum(h_lp, N/2)
            return (h_proj, c5_lp, N)
    except Exception:
        # Fall back to a known starting point
        h_lp = np.ones(N) * 0.5
        c5_lp = 0.5

    # Phase 2: If LP not sufficient, solve binary MILP with CBC and limited time
    # Convert h to binary variables
    h_bin = cp.Variable(N, boolean=True)
    y_bin = cp.Variable(len(pairs))
    alpha_bin = cp.Variable()

    constraints_bin = []
    constraints_bin.extend([0 <= h_bin[i] for i in range(N)])
    constraints_bin.extend([h_bin[i] <= 1 for i in range(N)])
    constraints_bin.append(cp.sum(h_bin) == N/2)

    for i, (idx, s) in enumerate(pairs):
        constraints_bin.append(y_bin[i] <= h_bin[idx])
        constraints_bin.append(y_bin[i] <= h_bin[idx + s])
        constraints_bin.append(y_bin[i] >= h_bin[idx] + h_bin[idx + s] - 1)
        constraints_bin.append(y_bin[i] >= 0)

    # Treat each shift
    for s in shifts:
        if s not in shift_pair_map:
            continue
        y_idx_list = shift_pair_map[s]
        sum_y_s = cp.sum(y_bin[y_idx_list])
        constraints_bin.append(alpha_bin >= dx * (N/2 - sum_y_s))

    objective_bin = cp.Minimize(alpha_bin)
    prob_bin = cp.Problem(objective_bin, constraints_bin)
    
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 20:
        # Use a time limit for the MILP solve
        try:
            prob_bin.solve(solver=CBC, verbose=False, maxTime=remaining_time - 10)
            h_bin_sol = h_bin.value
            c5_bin = compute_c5(h_bin_sol)
            if c5_bin <= 0.3807:
                return (h_bin_sol, c5_bin, N)
        except Exception:
            pass

    # Phase 3: Greedy rounding, restart from LP solution.
    h_rounded = np.copy(h_lp)
    # Create sorted indices of h_rounded  
    sorted_idx = np.argsort(h_rounded)[::-1]
    # Set top N/2 to 1, rest to 0
    h_rounded.fill(0)
    for i in range(int(N/2)):
        h_rounded[sorted_idx[i]] = 1
    h_rounded = project_sum(h_rounded, N/2)
    c5_rounded = compute_c5(h_rounded)
    if c5_rounded <= 0.3807:
        return (h_rounded, c5_rounded, N)

    # Phase 4: Local refinement
    h_refined = h_rounded.copy()
    c5_refined = c5_rounded

    # Rewind to before local polishing (e.g. as a reproducer)
    best_h = h_rounded.copy()
    best_c5 = c5_rounded

    # Simple gradient descent step with fixed steps and projection
    for _ in range(500):
        # Compute overlaps and find worst overlap
        c_values = np.correlate(h_refined, 1 - h_refined, mode='full') * dx
        worst_s = np.argmax(c_values)
        # Compute gradient
        # For shift Worse_s = shift s in band of -N+1 to N-1
        s_val = worst_s - (N - 1)
        # Normalize to get a subgradient in later iteration of gradient update
        # z = h_refined copy for shifting...
        if s_val >= 0:
            dz = np.zeros_like(h_refined)
            dz[:-s_val] = 1
            dz = dz[s_val:]
        else:
            dz = np.zeros_like(h_refined)
            dz[s_val:] = 1
            dz = dz[:s_val]
        # Vertical impact of y subscript is   h_refined[i] * h_refined[i + s]
        # We have the shift this gives us, and
        # we use subgradient of overlap for shift s_val, do update direction
        # dy = dz - dz * h_refined  <-- wrong, need to explain properly ---
        # Since we compute more generally s out of [0,N-1], consider shiftbackward relative to end:
        # shift = 2 - s where s is in 0,1,... N - 1, or just v = 1 to N not 0 to N - 1
        # Which makes sense of the spread of h values to form a filter (convolution)'s response

        # Focused on shift s_val, the subgradient w.r.t. h[i] is using values (i, i+s_val)
        # The change in output to change h[k] is -(h[k] raise that to matching index in output)

        # This is more complete from earlier Style:
        # For fixed j fix mapping to indicating upward / downward marginal relation:
        # via series measurement `c = sum h(i) * h(i+j)` (j is shift index)
        # we can compute gradient of c with respect to each h(i)
        # it's the h(i+j) (when i+j below max index)

        # j = 1 = one positive correlation that affects current point i via its output link
        # here j corresponds to shift which cancels the integral form when you reverse
        # so i mov -= grad[i] * eps, but for fixed shift j:
        # consider:{c = sum_i h[i] * h[i+j] }, then one parameter is h[i], so
        # partial c / partial h[i] = h[i+j], performance should be faster for computing gradients
        # Slope Del(c)/Del(h[i]) = h[i+j] — i.e., theاستقلال（就 JB） of i+j in h_refined
        # This is the simplest sum of pair in each place where overlaps occur.

        # However, the gradient of overlap integral related to one shift, would be (1-h[i+s]) over all terms:
        # Each element contributes marginally to the term.
        # one extra manual code that effectively calculates batch gradient from one shift:
        grad = np.zeros_like(h_refined)
        if s_val >= 0:
            grad[s_val:] = -h_refined[:-s_val]
        else:
            grad[:s_val] = -h_refined[-s_val:]

        # Normalization by scaling for safer step size
        # Take step on gradient direction (scaled)
        step_size = 0.02
        h_new = h_refined - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = project_sum(h_new, N/2)
        c5_new = compute_c5(h_new)
        if c5_new < c5_refined:
            h_refined = h_new
            c5_refined = c5_new
            # Potentially update best solution
            if c5_refined < best_c5:
                best_h = h_refined.copy()
                best_c5 = c5_refined

            # Early exit if we're close to target
            if best_c5 <= 0.3807:
                break
    
    return (best_h, best_c5, N)
