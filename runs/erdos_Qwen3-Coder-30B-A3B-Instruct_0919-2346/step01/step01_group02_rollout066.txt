import numpy as np
import time
from scipy.optimize import differential_evolution, minimize

def c5_of(h):
    """Compute the maximum overlap integral for the given step function h."""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    c = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(c) * dx
    return c5

def evaluate_hill_climb(h, steps=500):
    """Refine solution with local hill climbing."""
    dx = 2.0 / len(h)
    current_h = h.copy()
    current_c5 = c5_of(current_h)
    
    for _ in range(steps):
        # Try small perturbations
        idx = np.random.randint(len(h))
        direction = np.random.choice([-1, 1])
        step = 0.001
        
        # Ensure feasibility
        if direction > 0:
            max_step = min(1.0 - current_h[idx], step)
        else:
            max_step = min(current_h[idx], step)
            
        if max_step <= 0:
            continue
            
        new_h = current_h.copy()
        new_h[idx] += direction * max_step
        
        new_c5 = c5_of(new_h)
        if new_c5 < current_c5:
            current_h = new_h
            current_c5 = new_c5
    
    return current_h, current_c5

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    N = 120
    dx = 2.0 / N
    
    # Try multiple starting configurations
    configs = []
    
    # 1. Initial from seed if available
    if 'initial_h_values' in globals():
        h0 = np.interp(np.linspace(0, 2, N), 
                      np.linspace(0, 2, len(initial_h_values)), 
                      initial_h_values)
        h0 = h0 * (N/2) / np.sum(h0)  # Renormalize exactly
        configs.append(h0)
    
    # 2. Two-peak configuration
    h1 = np.ones(N)
    x_vals = np.linspace(0, 2, N)
    a = 0.3
    u = 0.9
    v = 0.1
    mask = (x_vals < a) | (x_vals >= 2 - a)
    h1[mask] = u
    h1[~mask] = v
    h1 = h1 * (N/2) / np.sum(h1)  # Renormalize
    configs.append(h1)
    
    # 3. Uniform configuration
    h2 = np.ones(N) * 0.5
    configs.append(h2)
    
    # 4. Sinusoidal configuration
    x_vals = np.linspace(0, 2, N)
    h3 = 0.5 + 0.3 * np.sin(2 * np.pi * x_vals)
    h3 = np.clip(h3, 0, 1)
    h3 = h3 * (N/2) / np.sum(h3)
    configs.append(h3)
    
    # 5. Block configuration 
    h4 = np.zeros(N)
    h4[N//4:3*N//4] = 1.0
    h4 = h4 * (N/2) / np.sum(h4)
    configs.append(h4)
    
    # Use differential evolution to find better solution
    start_time = time.time()
    
    # Initialize with best of configurations
    best_h = None
    best_c5 = float('inf')
    
    for i, h in enumerate(configs):
        if time.time() - start_time > 0.7 * budget_s:
            break
            
        # Local refinement
        refined_h, refined_c5 = evaluate_hill_climb(h, 500)
        if refined_c5 < best_c5:
            best_h = refined_h
            best_c5 = refined_c5
    
    # Now use differential evolution for more thorough search
    def objective(h):
        return c5_of(h) 
    
    # Bounds for each variable (0 to 1)
    bounds = [(0, 1) for _ in range(N)]
    
    # Use both existing best and random init
    for _ in range(10):
        if time.time() - start_time > 0.9 * budget_s:
            break
        
        # Initialize with random within bounds
        h0 = np.random.rand(N)
        h0 = h0 * (N/2) / np.sum(h0)
        
        # Run differential evolution
        try:
            res = differential_evolution(objective, bounds, 
                                       x0=h0, maxiter=200, 
                                       popsize=10, seed=seed+i)
            if res.success:
                h_final = res.x
                h_final = h_final * (N/2) / np.sum(h_final)
                c5_final = c5_of(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    # Final polishing with hill climbing
    if best_h is not None:
        final_h, final_c5 = evaluate_hill_climb(best_h, 1000)
        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5
    
    if best_h is None:
        # Fallback to uniform initialization
        best_h = np.ones(N) * 0.5
        best_c5 = c5_of(best_h)
    
    return (best_h, best_c5, N)
