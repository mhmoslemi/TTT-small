import numpy as np
import time
import random
from scipy.fft import rfft, irfft

def compute_C_vector(h, dx):
    """Compute all overlaps C_tau for tau = 0, ..., 2n-2"""
    n = len(h)
    # Compute auto-correlation of h with zero padding
    S = np.correlate(h, h, mode="full")
    c = 1.0 - dx * S
    return c

def compute_softmax_weights(c, alpha):
    """Compute softmax weights from overlap vector c"""
    c_max = np.max(c)
    exp_c = np.exp(alpha * (c - c_max))
    return exp_c / np.sum(exp_c)

def compute_gradient(h, w, dx):
    """Compute gradient of smooth objective using convolutions"""
    n = len(h)
    # For the second term, we reverse weights for indexing
    w_rev = w[::-1]  
  
    # Convolve with h to get the contributions
    # These are full correlations so we keep only the valid parts
    conv_w_h = np.convolve(w, h, mode="same")
    conv_wrev_h = np.convolve(w_rev, h, mode="same")  

    # Note: transforms are used differently in this to get what we want.
    # Actually, we can still use these as right convolution with zero padding.
    
    # Here: use rfft approach for more control
    N = 2 * n - 1
    h_pad = np.pad(h, (0, N - n), mode='constant')
    w_pad = np.pad(w, (0, N - len(w)), mode='constant')
    
    H = rfft(h_pad)
    W = rfft(w_pad)
    W_rev = rfft(w_rev)
    
    # Convolution: (W * conjugate(H)) + (W_rev * H)  
    # Or simply: (w * h) + (w_rev * h) in frequency domain
    # Efficiently compute using FFTs (normalize appropriately)
    grad = -dx * irfft( W * np.conj(H) + W_rev * H, n=N)[:n]
    return grad

def project_to_feasible(x, target_sum, bound_lo=0.0, bound_hi=1.0):
    """Project vector x onto [bound_lo, bound_hi] with target_sum constraint
    Uses water-filling (bisection).
    """
    x = np.array(x, dtype=float)
    n = len(x)
    # Ensure mass constraint
    if abs(np.sum(x) - target_sum) < 1e-12:
        x = np.clip(x, bound_lo, bound_hi)
        return x
    
    def sum_at_lambda(lam):
        # return sum of clip(x_i - lam, 0, 1)
        tmp = np.clip(x - lam, 0.0, 1.0)
        return np.sum(tmp) - target_sum
    
    # Find range to bracket root
    lb = np.min(x) - 1.0
    ub = np.max(x) + 1.0
    
    # Bisection on lambda
    for _ in range(100):  # safe iteration limit
        mid = (lb + ub) / 2.0
        val = sum_at_lambda(mid)
        if abs(val) < 1e-12:
            break
        if val > 0:
            lb = mid
        else:
            ub = mid
    
    # Apply final projection
    x_proj = np.clip(x - (lb + ub) / 2.0, bound_lo, bound_hi)
    return x_proj

    v = 0.1
    x = np.linspace(0, 2, n)
    mask = (x < a) | (x > (2 - a))
    h_double[mask] = u
    h_double[~mask] = v
    h_double = h_double * target_sum / np.sum(h_double)

    seeds = [h_uniform, h_initial, h_double]

    best_h = None
    best_c5 = float('inf')

    # Reserve 5% for initialization
    init_budget = 0.05 * budget_s
    start = time.time()
    init_end = start + init_budget
    for h_seed in seeds:
        if time.time() >= init_end:
            break
        h = project_to_feasible(h_seed, target_sum)
        c = compute_C_vector(h, dx)
        c5 = np.max(c)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    if best_h is None:
        best_h = h_uniform.copy()
        best_c5 = 1.0

    # Core gradient descent loop
    max_iter = 100000
    t_start = time.time()
    t_budget = 0.80 * budget_s
    t_end = t_start + t_budget
    
    # Optimizer parameters
    alpha_start = 5.0
    alpha_end = 80.0
    eta0 = 3.0
    beta = 0.7  # momentum
    k_every = 300   # restart every 300 iterations
    
    # Initialize state
    h = best_h.copy()
    h = project_to_feasible(h, target_sum)
    c = compute_C_vector(h, dx)
    
    # decay rules
    def alpha_schedule(t):
        return alpha_start + (alpha_end - alpha_start) * min(t / (t_budget/3.0), 1.0)
    
    def eta_schedule(t):
        return eta0 / np.sqrt(1.0 + t)
    
    # Grad momentum
    m = np.zeros_like(h)
    iter_count = 0
    
    last_improvement = 0
    c5_current = np.max(c)
    
    while time.time() < t_end and iter_count < max_iter:
        t = time.time() - t_start
        iter_count += 1

        # Smoothing
        alpha = alpha_schedule(t)
        w = compute_softmax_weights(c, alpha)
        
        # Gradient direction
        g = compute_gradient(h, w, dx)

        # Momentum
        m = beta * m + (1.0 - beta) * g
    
        # Step-size
        eta = eta_schedule(iter_count)
        
        # Update 
        h_new = h - eta * m
        
        # Project
        h_new = project_to_feasible(h_new, target_sum)

        # Update h
        h = h_new
        
        # Recompute so c reflects updated h
        c = compute_C_vector(h, dx)
        c5_current = np.max(c)
        
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
            last_improvement = iter_count

        # Periodic perturbation
        if iter_count % k_every == 0:
            # Add noise and reproject
            noise = np.random.normal(0, 0.01, n)
            h_perturbed = h + noise
            h_perturbed = project_to_feasible(h_perturbed, target_sum)
            c_pert = compute_C_vector(h_perturbed, dx)
            c5_pert = np.max(c_pert)
            if c5_pert < best_c5:
                best_c5 = c5_pert
                best_h = h_perturbed.copy()
                last_improvement = iter_count

        if iter_count - last_improvement > 2000:
            # reduce step size if no progress
            eta0 /= 2.0     
    
    # Final polishing stage
    polish_budget = 0.15 * budget_s
    polish_end = time.time() + polish_budget
    cur = best_h.copy()
    c_cur = best_c5
    
    while time.time() < polish_end:
        # Use pairwise move like in reference
        h_new = cur.copy()
        i, j = random.sample(range(n), 2)
        
        delta_low = max(-h_new[i], h_new[j] - 1.0)
        delta_high = min(1.0 - h_new[i], h_new[j])
        if delta_low < delta_high:
            delta = random.uniform(delta_low, delta_high)
            h_new[i] += delta
            h_new[j] -= delta
            
            # Enforce exact mass
            h_new = project_to_feasible(h_new, target_sum)
            c_new = compute_C_vector(h_new, dx)
            c5_new = np.max(c_new)
            if c5_new <= c_cur:
                cur = h_new
                c_cur = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = cur
    
    return (best_h, best_c5, n)
