import numpy as np
import time
import random
from scipy.optimize import differential_evolution, minimize

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    overlap = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(overlap) * dx
    return c5

def evaluate_constraints(h):
    """Check if h satisfies constraints"""
    dx = 2.0 / len(h)
    integral = np.sum(h) * dx
    return abs(integral - 1.0) < 1e-10

def propose_move(h, max_step):
    """Propose a new feasible h vector by adjusting two elements"""
    h_new = h.copy()
    N = len(h)
    
    max_tries = 100
    for _ in range(max_tries):
        i = random.randint(0, N - 1)
        j = random.randint(0, N - 1)
        if i == j:
            continue
            
        # Determine feasible interval for δ
        delta_low = max(-h_new[i], h_new[j] - 1)
        delta_high = min(1 - h_new[i], h_new[j])
        
        if delta_low >= delta_high:
            continue
            
        # Sample δ from uniform distribution over feasible range
        step = min(max_step, delta_high - delta_low)
        if step <= 0:
            continue
        delta = random.uniform(delta_low, delta_high)
        
        # Apply move
        h_new[i] += delta
        h_new[j] -= delta
        
        return h_new
    
    # If no valid move found, return original
    return h.copy()

def shake_solution(h, shake_magnitude=0.05):
    """Create a slightly perturbed version of h"""
    h_new = h.copy()
    N = len(h_new)
    
    # Randomly select a few points to perturb
    num_perturb = max(1, N // 20)
    indices = random.sample(range(N), num_perturb)
    
    for i in indices:
        delta = random.uniform(-shake_magnitude, shake_magnitude)
        h_new[i] = np.clip(h_new[i] + delta, 0, 1)
    
    # Renormalize
    dx = 2.0 / N
    h_new = h_new * (N / 2) / np.sum(h_new)
    
    return h_new

    else:

    # Attempt several optimization strategies
    strategies = []
    
    # Strategy 1: Differential Evolution
    def objective_de(x):
        # Reshape to proper dimensions
        h = x.copy()
        # Renormalize to sum = N/2
        h = h * (N / 2) / np.sum(h)
        return c5_of(h)
    
    bounds = [(0, 1) for _ in range(N)]
    
    de_result = None
    try:
        de_result = differential_evolution(
            objective_de,
            bounds,
            seed=seed,
            maxiter=200,
            popsize=15,
            mutation=(0.5, 1),
            recombination=0.7,
            atol=1e-6,
            disp=False
        )
        
        if de_result.success:
            h_de = de_result.x.copy()
            h_de = h_de * (N / 2) / np.sum(h_de)
            strategies.append((h_de, c5_of(h_de)))
    except:
        pass

    # Strategy 2: Simulated Annealing
    def anneal(initial_h, time_budget_s):
        T_start = 0.1
        T_end = 1e-6
        step_start = 0.1
        
        cur = initial_h.copy()
        c_cur = c5_of(cur)
        
        best = cur.copy()
        c_best = c_cur
        
        start = time.time()
        while time.time() - start < time_budget_s:
            t = time.time() - start
            elapsed_fraction = t / time_budget_s
            
            T = T_start * (T_end / T_start) ** elapsed_fraction
            step = step_start * (1 - elapsed_fraction)
            
            h_proposed = propose_move(cur, step)
            c_proposed = c5_of(h_proposed)
            
            if c_proposed <= c_cur:
                cur = h_proposed
                c_cur = c_proposed
                if c_proposed < c_best:
                    best = h_proposed.copy()
                    c_best = c_proposed
            else:
                # Metropolis acceptance rule
                if random.random() < np.exp(-(c_proposed - c_cur) / T):
                    cur = h_proposed
                    c_cur = c_proposed

        )

            h = h * (N / 2) / np.sum(h)

    for i in range(5):

        h_random = np.random.rand(N)
        h_random = h_random * (N / 2) / np.sum(h_random)
        strategies.append((h_random, c5_of(h_random)))

    # Final polishing with local search
    best_strategy = None
    best_c5 = float('inf')
    
    for i, (h_strat, c5_strat) in enumerate(strategies):
        if c5_strat < best_c5:
            best_c5 = c5_strat
            best_strategy = h_strat.copy()
    
    # Fine-tune best with local moves
    if best_strategy is not None:
        fine_tune_time = min(100, budget_s*0.2)
        h_final = best_strategy.copy()
        c_final = c5_of(h_final)
        
        # Perform additional local search
        start_fine = time.time()
        while time.time() - start_fine < fine_tune_time:
            h_proposed = propose_move(h_final, 0.02)
            c_proposed = c5_of(h_proposed)
            
            if c_proposed < c_final:
                h_final = h_proposed
                c_final = c_proposed
        
        best_strategy = h_final
        best_c5 = c_final
    
    if best_strategy is None:
        # Fallback to initial solution
        best_strategy = h_provided
        best_c5 = c5_of(h_provided)
    
    return (best_strategy, best_c5, N)
