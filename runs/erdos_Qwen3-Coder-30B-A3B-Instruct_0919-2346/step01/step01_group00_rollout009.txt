import numpy as np
import time
import random
from scipy.optimize import differential_evolution, minimize

def c5_of(h):
    """Compute C5 = max_k integral h(x)(1-h(x+k)) dx."""
    dx = 2.0 / len(h)
    # Compute correlation: h * (1 - h) with full mode
    c = np.correlate(h, 1 - h, mode="full")
    # Multiply by dx to get continuous integral
    c *= dx
    # Return maximum value among all shifts
    return np.max(c)

def enforce_constraints(h, n_points):
    """Ensure the constraint sum(h) * dx = 1"""
    dx = 2.0 / n_points
    current_sum = np.sum(h)
    target_sum = n_points / 2
    scale_factor = target_sum / current_sum
    return h * scale_factor

def propose_move(h, max_step):
    """Generate a new feasible h by adjusting two entries."""
    h = h.copy()
    N = len(h)
    i, j = random.sample(range(N), 2)
    
    # Determine feasible interval for delta
    delta_low = max(-h[i], h[j] - 1.0)
    delta_high = min(1.0 - h[i], h[j])
    
    if delta_low >= delta_high:
        return h  # No valid move for this pair
    
    # Sample delta within allowed range
    step = min(max_step, delta_high - delta_low)
    delta = random.uniform(-step, step)
    delta = max(delta_low, min(delta, delta_high))
    
    h[i] += delta
    h[j] -= delta
    
    return h

def local_search(initial_h, time_budget):
    """Local search with adaptive step size"""
    start = time.time()
    cur = initial_h.copy()
    best = initial_h.copy()
    c_cur = c5_of(cur)
    c_best = c_cur
    
    # Large initial step, decreasing over time
    step_size = 0.1
    step_decay = 0.999
    
    while time.time() - start < time_budget:
        h_new = propose_move(cur, step_size)
        c_new = c5_of(h_new)
        
        if c_new <= c_cur:
            cur = h_new
            c_cur = c_new
            if c_new < c_best:
                best = h_new
                c_best = c_new
                step_size = min(0.1, step_size * 1.05)  # speed up if improvement
        else:
            step_size *= step_decay  # slow down if no improvement
            
    return best, c_best

def optimize_with_differential_evolution(n_points, time_budget):
    """Use differential evolution for global search"""
    start = time.time()
    
    # Bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        # Ensure constraint satisfaction
        h = enforce_constraints(h, n_points)
        return c5_of(h)
    
    # Set up DE optimization
    DE_kwargs = {
        'bounds': bounds,
        'maxiter': 1000,
        'popsize': min(50, n_points * 2),
        'tol': 1e-6,
        'mutation': (0.5, 1),
        'recombination': 0.7,
        'seed': 42,
        'callback': None
    }
    
    # Run just the first part to get a good starting point
    result = differential_evolution(objective, bounds, **DE_kwargs)
    h_start = result.x
    
    # Refine with local search from this point
    h_start = enforce_constraints(h_start, n_points)
    h_local, c5_local = local_search(h_start, time_budget * 0.8)
    
    return h_local, c5_local
