import numpy as np

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value scaled by dx
    max_overlap = np.max(correlation) * dx
    return max_overlap

def upsample_initial(h0, n):
    """Upsample a given h0 (length k) to a new length n using linear interpolation"""
    if len(h0) == n:
        return h0.copy()
    x_old = np.linspace(0, 2, len(h0))
    x_new = np.linspace(0, 2, n)
    h_new = np.interp(x_new, x_old, h0)
    return h_new

def gradient(h, gamma=1.0):
    """Compute the gradient of the soft max approximation to F(h)"""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    
    # For each lag (shift s) compute contribution to gradient
    for s in range(1-n, n):  # from -n+1 to n-1 inclusive
        f_s_val = 0.0
        for i in range(n):
            i_s = i+s
            if 0 <= i_s < n:
                f_s_val += h[i] * (1 - h[i_s])
        
        # Build local `g_i^{(s)}`
        # g_{i+s} = 1 - h_{i} but only when 0 <= i+s < n
        # g_{i-s} = h_{i} but only when 0 <= i-s < n  
        g_s = np.zeros(n)
        
        for i in range(n):
            i_s = i + s
            i_neg_s = i - s
            if 0 <= i_s < n:
                g_s[i] += 1 - h[i_s]
            if 0 <= i_neg_s < n:
                g_s[i] -= h[i_neg_s]
        grad += np.exp(gamma * f_s_val) * g_s
    
    # Normalize by the sum of exp terms (softmax weights)
    # Add fictitious small number to avoid exploding float
    # Also return a vector, not a scalar
    w_sum = 0.0
    for s in range(1-n, n):
        f_s_val = 0.0
        for i in range(n):
            i_s = i+s
            if 0 <= i_s < n:
                f_s_val += h[i] * (1 - h[i_s])
        w_sum += np.exp(gamma * f_s_val)
    if w_sum > 0:
        # Alternatively, just use normalizer from soft max weights
    
    # Scale the accumulated gradient
    return dx * grad / w_sum if w_sum > 0 else grad

def softmax_weights(h, gamma):
    """Compute the softmax weights used for gradient"""
    n = len(h)
    dx = 2.0 / n
    weights = []
    exp_terms = []
    total = 0.0
    
    for s in range(1-n, n):
        f_s_val = 0.0
        for i in range(n):
            i_s = i+s
            if 0 <= i_s < n:
                f_s_val += h[i] * (1 - h[i_s])
        exp_term = np.exp(gamma * f_s_val)
        exp_terms.append(exp_term)
        total += exp_term
        
    # Normalize, to get weights
    if total == 0.0:  # Degenerate case
        weights = [1.0 / len(exp_terms)] * len(exp_terms)
    else:
        weights = [t/total for t in exp_terms]
    return weights
