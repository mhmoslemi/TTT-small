import numpy as np
import time
import random
from scipy.fft import rfft, irfft

def compute_C_vector(h, dx):
    """Compute all C_tau values for shift indices tau."""
    # Autocorrelation S_tau = sum h_i * h_{i+tau}
    # With zero padding, this is correlate(h,h) with mode='full'
    S = np.correlate(h, h, mode='full')
    # Restrict to relevant indices: tau from -n+1 to n-1, so full is 2n-1
    # But C_tau is defined for tau=0..2n-2 (as full has that many elements)
    # The corresponding shift indices go from -n+1 to n-1
    S = S[n-1:]  # take from 0 up to n-1 (corresponds to tau=0 to n-1)
    # Actually, improvement: extract only full range needed
    # S_0 to S_{2n-2}, but via full mode, we have 2*n-1 elements indexed 0..2n-2.
    # Actually let's recompute properly.
    n = len(h)
    S_full = np.correlate(h, h, mode='full')
    C = 1 - dx * S_full[n-1 : 2*n-1]
    return C

def compute_softmax_weights(c, alpha):
    """Compute soft-max weights w such that sum(w) = 1."""
    # Apply log-sum-exp trick for numerical stability
    c_max = np.max(c)
    exp_c = np.exp(alpha * (c - c_max))
    w = exp_c / np.sum(exp_c)
    return w

    #   where tau ranges from -n+1 to n-1
    #   i goes from 0 to 2n-2
    #   So we need (2n-1) elements for w in general to define this.
    # Better: keep vector indices aligned.

    # This is equivalent to computing:
    #  sum_i=0^{n-1} w[t] * h[i-t]  for i fixed.

    # but when computing only one row i=... of cross-correlation, it’s

    # This is the pattern: the two parts of convolution.
    # Now correctly generate g:
    # There are two parts: We compute two convolutions
    # Conv w and h with full mode or use fast FFTs
    # Use rfft per definitions?:
    # In numpy FFT with real inputs

    # Target for real inputs with length n, normal NFFT gives n//2+1 output bins
    # Reconstruct same real trigran from rfft input.
    # But mathematically better:
    # First, observe that we are computing:
    #   g_i = -dx * sum_{k} w_k * (h_{i+k} + h_{i-k})
    # This is:
    #   g = -dx * (w * h) + (w_rev * h)
    # with inversions of indices in counts -> prefix notation. However importantly when authors write w*...
    #     they mean an actual FFT or conv operation (no shift inversion generally) -- we are aligning here the two
    # What we'll ensure is that g has right correctly defined indices.
    # Define proper shifted convolution space:
    # Polarity of index stays same in (w) & (h) due to i-k, i+k
    # Use real FFT-based FFT trick provided by scipy
    h_pad = np.pad(h, (0, n - 1), 'constant', constant_values=0)
    w_pad = np.pad(w, (0, n - 1), 'constant', constant_values=0)

    # Here, let's actually think clearly:
    # w has length 2n-1 (two extra due to cross corr);  h has length n
    # But for correct interpretation the same. The direct sum
    # g_i = -dx * sum_k w_k * (h_{i+k} + h{i-k}) approach.

    # Let's simplify using full convolution with proper truncation
    w_full = np.concatenate([w, np.zeros(n-1)])
    # corr1 with pad -> pad to size 2(n-1) + 1 = 2n-1
    winv_full = w_full[::-1]

    # Due to indexing issues below, manually do each term.
    # y = conv(w_full, h_pad, mode='full')[:2n-1]
    # The final array for i from 0 to n-1 contains discrete convolution terms that

    # y_i = sum_{k} w_k h_{i-k}

    cf2 = rfft(h_pad, 2 * n - 1)
    cf_g = cf1 * cf2
    g1 = irfft(cf_g, 2 * n - 1)

    cf3 = rfft(winv_full, 2 * n - 1)
    cf4 = rfft(h_pad, 2 * n - 1)
    cf_g2 = cf3 * cf4
    g2 = irfft(cf_g2, 2 * n - 1)

    g = -(dx * (g1[:n] + g2[:n]))
    return g

def project(h, target_sum):
    """Project h onto [0,1] with exact sum = target_sum."""
    # water-filling using bisection
    h = np.asarray(h)
    low = np.min(h) - 1.0
    high = np.max(h) + 1.0
    e = 1e-12
    N = len(h)
    target = target_sum
    while (high - low > e):
        mid = (low + high) / 2.0
        # laplace approximation
        x = np.clip(h - mid, 0, 1)
        s = np.sum(x)
        if s > target:
            low = mid
        else:
            high = mid
    x = np.clip(h - low, 0, 1)
    x *= target / np.sum(x)
    return x
