import numpy as np
import time

def c5_of(h):
    """Compute the maximum overlap integral for the given step function h."""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    c = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(c) * dx
    return c5

def compute_gradient(h, shift):
    """Compute gradient of C5 with respect to h for the given shift."""
    N = len(h)
    dx = 2.0 / N
    g = np.full(N, 1.0) * dx
    if shift >= 0:
        if shift < N:
            # subtract right shift
            g[:N-shift] -= h[shift:] * dx
            # subtract left shift
            g[shift:] -= h[:N-shift] * dx
    else:
        shift = -shift
        if shift < N:
            g[shift:] -= h[:N-shift] * dx
            g[:N-shift] -= h[shift:] * dx
    return g

def project_onto_bounds_and_sum(v, target):
    """Project v onto the set of vectors with sum = target and bounds [0,1]."""
    # Binary search for lambda such that sum of clipped values equals target
    lower = -np.max(v) - 1
    upper = 1 - np.min(v) + 1
    for _ in range(60):
        mid = (lower + upper) / 2.0
        w = np.clip(v + mid, 0, 1)
        s = np.sum(w)
        if s < target:
            lower = mid
        else:
            upper = mid
    w = np.clip(v + (lower + upper) / 2.0, 0, 1)
    
    # Ensure exact sum by distributing error
    diff = target - np.sum(w)
    if abs(diff) > 1e-12:
        free_idx = (w > 0) & (w < 1)
        if np.any(free_idx):
            w[free_idx] += diff / np.sum(free_idx)
            w = np.clip(w, 0, 1)
        else:
            w += diff / len(w)
            w = np.clip(w, 0, 1)
    return w

    np.random.seed(seed)

    best_h = h0.copy()
    best_score = c5_of(best_h)
    
    # Two-peak seed 
    h1 = np.ones(N)
    x_vals = np.linspace(0, 2, N)
    a = 0.3
    u = 0.9
    v = 0.1
    mask = (x_vals < a) | (x_vals >= 2 - a)
    h1[mask] = u
    h1[~mask] = v
    h1 = h1 * (N/2) / np.sum(h1)  # Renormalize
    
    # Test this shape
    score1 = c5_of(h1)
    if score1 < best_score:
        best_h = h1.copy()
        best_score = score1

    # Binary seeds
    for _ in range(5):
        h_bin = np.zeros(N)
        idx = np.random.choice(N, N//2, replace=False)
        h_bin[idx] = 1.0
        score_bin = c5_of(h_bin)
        if score_bin < best_score:
            best_h = h_bin.copy()
            best_score = score_bin

    # Gradient descent with restarts
    start = time.time()
    
    # Inner loop with time management
    while time.time() - start < budget_s * 0.9:
        cur_h = best_h.copy()
        cur_score = best_score
        
        # Inner gradient loop for up to 300 iterations or time limit
        inner_start = time.time()
        iterations = 0
        while time.time() - start < budget_s * 0.9 and iterations < 300:
            # Compute correlation and find worst shift
            corr = np.correlate(cur_h, 1 - cur_h, mode='full')
            max_idx = np.argmax(corr)
            s = max_idx - (N-1)
            
            # Compute gradient for worst shift
            grad = compute_gradient(cur_h, s)
            
            # Adaptive step size
            t_frac = (time.time() - start) / budget_s
            step = 15.0 * (1 - t_frac) + 0.1
            
            # Update h
            v = cur_h - step * grad
            cur_h = project_onto_bounds_and_sum(v, N/2)
            
            # Evaluate score
            new_score = c5_of(cur_h)
            if new_score < best_score:
                best_h = cur_h.copy()
                best_score = new_score
                
            iterations += 1
            
        # Random restart occasionally
        if np.random.rand() < 0.3:
            h_tmp = np.random.rand(N)
            h_tmp = h_tmp * (N/2) / np.sum(h_tmp)
            score_tmp = c5_of(h_tmp)
            if score_tmp < best_score:
                best_h = h_tmp.copy()
                best_score = score_tmp

    # Final local refinement
    for _ in range(1000):
        if time.time() - start >= budget_s * 0.99:
            break
        i, j = np.random.choice(N, size=2, replace=False)
        delta_low = max(-best_h[i], best_h[j] - 1)
        delta_high = min(1 - best_h[i], best_h[j])
        if delta_low < delta_high:
            delta = np.random.uniform(-0.03, 0.03)
            delta = np.clip(delta, delta_low, delta_high)
            best_h[i] += delta
            best_h[j] -= delta
            score = c5_of(best_h)
            if score < best_score:
                best_score = score
            else:
                best_h[i] -= delta
                best_h[j] += delta
    
    return (best_h, best_score, N)
