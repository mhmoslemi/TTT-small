import numpy as np
import time
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import pdist, squareform

def c5_of(h):
    """Compute the maximum overlap integral for the given step function h."""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    c = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(c) * dx
    return c5

def evaluate_hill_climb(h, max_iters=1000, step_size=0.01):
    """Hill climb optimization with bounded adjustments."""
    h = h.copy()
    c_current = c5_of(h)
    best_h = h.copy()
    best_c = c_current
    
    for _ in range(max_iters):
        # Choose random indices to adjust
        i, j = np.random.choice(len(h), size=2, replace=False)
        # Determine feasible adjustments
        delta_low = max(-h[i], h[j] - 1)
        delta_high = min(1 - h[i], h[j])
        if delta_low >= delta_high:
            continue

        # Propose adjustment
        step = np.random.uniform(max(-step_size, delta_low),
                                min(step_size, delta_high))
        
        h_new = h.copy()
        h_new[i] += step
        h_new[j] -= step
        
        # Ensure constraints
        if np.any(h_new < 0) or np.any(h_new > 1):
            continue
            
        c_new = c5_of(h_new)
        if c_new < c_current:
            h = h_new
            c_current = c_new
            if c_new < best_c:
                best_h = h.copy()
                best_c = c_new
                
    return best_h, best_c

    best_c5 = c5_of(h0)

    def anneal_step():

        step = 0.05
        cur_h = h0.copy()
        cur_c = c5_of(cur_h)
        best_h = cur_h.copy()
        best_c = cur_c

            T *= 0.999
            step *= 0.999

            i, j = np.random.choice(N, size=2, replace=False)
            delta_low = max(-cur_h[i], cur_h[j] - 1)
            delta_high = min(1 - cur_h[i], cur_h[j])
            if delta_low >= delta_high:
                continue
                
            step_size = np.random.uniform(max(-step, delta_low),
                                        min(step, delta_high))
            
            h_new = cur_h.copy()
            h_new[i] += step_size
            h_new[j] -= step_size
            
            if np.any(h_new < 0) or np.any(h_new > 1):

            c_new = c5_of(h_new)
            if c_new < cur_c or np.random.rand() < np.exp(-(c_new - cur_c) / T):
