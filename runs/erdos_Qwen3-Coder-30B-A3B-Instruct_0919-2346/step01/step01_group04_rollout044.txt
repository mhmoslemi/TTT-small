import numpy as np
import time
import random
from scipy.optimize import differential_evolution, minimize


def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    overlap = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(overlap) * dx
    return c5


def propose_move(h, max_step):
    """Propose a new feasible h vector by adjusting two elements"""
    h_new = h.copy()
    N = len(h)
    
    # Try to find a valid move
    max_tries = 100
    for _ in range(max_tries):
        i = random.randint(0, N - 1)
        j = random.randint(0, N - 1)
        if i == j:
            continue
            
        # Determine feasible interval for δ
        delta_low = max(-h_new[i], h_new[j] - 1)
        delta_high = min(1 - h_new[i], h_new[j])
        
        if delta_low >= delta_high:
            continue
            
        # Sample δ from uniform distribution over feasible range
        step = min(max_step, delta_high - delta_low)
        if step <= 0:
            continue
        delta = random.uniform(delta_low, delta_high)
        
        # Apply move
        h_new[i] += delta
        h_new[j] -= delta
        
        return h_new
    
    # If no valid move found, return original
    return h.copy()


def refine_with_local_search(h_init, time_budget_s):
    """Refine using local search with adaptive step size"""
    start = time.time()
    
    cur = h_init.copy()
    c_cur = c5_of(cur)
    best = cur.copy()
    c_best = c_cur
    
    # Adaptive step size based on progress
    step_size = 0.02
    min_step = 1e-6
    max_step = 0.1
    
    # Number of consecutive unsuccessful moves
    no_improvement_count = 0
    max_no_improvement = 50
    
    while time.time() - start < time_budget_s:
        # Reduce step size if no improvement for a while
        if no_improvement_count > max_no_improvement:
            step_size = max(min_step, step_size * 0.9)
            no_improvement_count = 0
            
        # Make a proposal
        h_proposed = propose_move(cur, step_size)
        c_proposed = c5_of(h_proposed)
        
        if c_proposed <= c_cur:
            cur = h_proposed
            c_cur = c_proposed
            if c_proposed < c_best:
                best = h_proposed.copy()
                c_best = c_proposed
                no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Occasionally increase step size if we're making progress
        if no_improvement_count < 10 and step_size < max_step:
            step_size = min(max_step, step_size * 1.05)
    
    return best, c_best

    initial_configs = [
        h_provided,                  # Provided solution
        np.ones(N) * 0.5,            # Uniform
        np.concatenate([np.ones(N//2) * 0.8, np.ones(N//2) * 0.2]),  # Step function
    ]
    
    # Add some random configurations
    for _ in range(3):
        h_rand = np.random.rand(N)
        h_rand = h_rand * (N / 2) / np.sum(h_rand)
        initial_configs.append(h_rand)
    
    best_h = None
    best_c5 = float('inf')
    
    # Try multiple configurations with local search
    remaining_time = budget_s * 0.9
    time_per_config = remaining_time / len(initial_configs)

        if time.time() - start >= budget_s * 0.95:
            break

        try:
            h_opt, c5_opt = refine_with_local_search(h_seed, time_per_config)
