import numpy as np
from scipy.optimize import minimize
from math import pi
import time

def build_h_from_coeffs(a, n):
    """Build h vector from Fourier coefficients"""
    h = 0.5 * np.ones(n)
    for m in range(1, len(a)+1):
        cos_term = np.cos(2 * pi * m * np.arange(n) / n)
        h += a[m-1] * cos_term
    return h

def compute_overlap_and_gradient(a, n, dx, M, gamma=1.0):
    """Compute the soft-max overlap and its gradient w.r.t. a"""
    h = build_h_from_coeffs(a, n)
    h = np.clip(h, 0, 1)
    
    # Compute overlaps for all shifts
    correlation = np.correlate(h, 1 - h, mode="full")
    shifts = np.arange(-n + 1, n)
    overlaps = correlation * dx
    
    # Soft max weight
    max_overlap = np.max(overlaps)
    exp_overlaps = np.exp(gamma * (overlaps - max_overlap))
    weights = exp_overlaps / np.sum(exp_overlaps)
    
    # Soft-max value (ignoring penalty for gradient calculation)
    soft_max_val = max_overlap + np.log(np.sum(exp_overlaps)) / gamma
    
    # Gradient computation
    grad = np.zeros(M)
    for m in range(1, M+1):
        # gradient of h_i w.r.t. a_m
        cos_term = np.cos(2 * pi * m * np.arange(n) / n)
        # derivative of overlap w.r.t. a_m
        grad_overlap = 0.0
        for i in range(n):
            if 0 <= i < n and 0 <= i + m <= n - 1:
                grad_overlap += cos_term[i] * (1 - h[i + m]) \
                             - h[i] * cos_term[i + m]
        grad[m-1] = grad_overlap * dx
    
    # Weighted gradient
    weighted_grad = np.sum(weights * grad)
    
    # Penalty gradient
    penalty_grad = np.zeros(M)
    for m in range(1, M+1):
        cos_term = np.cos(2 * pi * m * np.arange(n) / n)
        # h_i - clip(h_i, 0, 1)
        clipped = np.clip(h, 0, 1)
        diff = h - clipped
        penalty_grad[m-1] = 2 * 10**4 * np.sum(diff * cos_term)
    
    total_gradient = weighted_grad + penalty_grad
    
    return soft_max_val, total_gradient

def project_to_simplex_box(h, n, dx):
    """Project h onto the intersection of box constraints and the simplex"""
    # First clip to [0,1]
    h = np.clip(h, 0, 1)
    # Then project to sum = n/2 using waterfilling
    target = n / 2
    current_sum = np.sum(h) * dx
    epsilon = 1e-10
    
    if abs(current_sum - target) < epsilon:
        return h
    
    # Simple bisection for waterfilling if needed
    # For very small lists, do direct projection
    # In this case, we compute the difference and adjust
    if current_sum > target:
        # Reduce values by amount
        reduction = (current_sum - target) / n
        h = h - reduction
        # Re-clip
        h = np.clip(h, 0, 1)
    elif current_sum < target:
        # Add an amount
        increment = (target - current_sum) / n
        h = h + increment
        # Re-clip
        h = np.clip(h, 0, 1)
    
    return h

def spectral_optim(n_points, M, max_iter, gamma_start=0.5, gamma_end=40, t0=None):
    """Run spectral optimization with homotopy"""
    # Temperature schedule
    gamma_schedule = []
    gamma = gamma_start
    while gamma <= gamma_end:
        gamma_schedule.append(gamma)
        gamma *= 1.5
    
    # Build Hessian approximations
    # We only do L-BFGS-based computation
    coeffs = np.random.normal(0, 0.05, M)
    bounds = [(-0.6, 0.6) for _ in range(M)]
    dx = 2.0 / n_points
    last_iter = 0
    
    # Try different initializations
    best_sol = None
    best_val = float('inf')
    
    for s in range(6):  # Multiple restarts
        
        if t0 is not None and time.time() - t0 > 0.95 * 600:
            break
            
        np.random.seed(s)
        coeffs = np.random.normal(0, 0.05, M)
        
        # Optimize
        iter_count = 0
        while iter_count < max_iter:
            
            if t0 is not None and time.time() - t0 > 0.95 * 600:
                break
                
            if iter_count % 500 == 0 and len(gamma_schedule) > 0:
                gamma = gamma_schedule.pop(0)
            
            try:
                # Use minimizer with bounds, and custom gradient
                def obj_and_grad(a):
                    val, grad = compute_overlap_and_gradient(a, n_points, dx, M, gamma)
                    return val, grad
                
                result = minimize(obj_and_grad, coeffs, method='L-BFGS-B', 
                                 bounds=bounds, jac=True, options={'maxiter': 100})
                if result.success:
                    coeffs = result.x
                else:
                    pass # just keep current coeffs, might restart as we're in a loop
            except Exception as e:
                pass
            iter_count += 1
        
        # Final projection
        h = build_h_from_coeffs(coeffs, n_points)
        h = project_to_simplex_box(h, n_points, dx)
        h = np.clip(h, 0, 1)
        
        # Evaluate its performance
        from scipy.signal import correlate
        corr = correlate(h, 1 - h, mode='full')
        max_overlap_val = np.max(corr) * dx
        if max_overlap_val < best_val:
            best_val = max_overlap_val
            best_sol = h.copy()
    
    return best_sol, best_val

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 240
    max_iterations = 10000
    
    # Stages with increasing M
    stages = [8, 15, 24]
    total_iterations = 0
    
    best_h = None
    best_c5 = float('inf')
    
    # Begin optimization loop
    for stage, M in enumerate(stages):
        
        if (time.time() - start_time) > 0.95 * budget_s:
            break
        
        # Start the process for this stage
        h_stage, c5_stage = spectral_optim(n_points, M, 
                                          max_iter=max_iterations, 
                                          t0=start_time)
        
        if c5_stage < best_c5:
            best_c5 = c5_stage
            best_h = h_stage.copy()
    
    return (best_h.tolist(), best_c5, n_points)
