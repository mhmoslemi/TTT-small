import numpy as np
import time
import random
from scipy import optimize
from scipy.signal import correlate

def compute_c5(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    overlap = correlate(h, 1 - h, mode="full")
    c5 = np.max(overlap) * dx
    return c5

def project_h(h, dx):
    """Project h onto feasible set: sum(h) = 1, 0 <= h <= 1"""
    # Enforce bounds
    h = np.clip(h, 0, 1)
    # Enforce integral constraint
    h = h * (1.0 / np.sum(h)) / dx
    return h

def estimate_gradient(h, eps=1e-4):
    """Estimate gradient of C5 using finite differences"""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    
    # Estimate gradient w.r.t. each point
    for i in range(n):
        h_plus = h.copy()
        h_minus = h.copy()
        h_plus[i] += eps
        h_minus[i] -= eps
        h_plus = project_h(h_plus, dx)
        h_minus = project_h(h_minus, dx)
        grad[i] = (compute_c5(h_plus) - compute_c5(h_minus)) / (2 * eps)
    
    return grad
