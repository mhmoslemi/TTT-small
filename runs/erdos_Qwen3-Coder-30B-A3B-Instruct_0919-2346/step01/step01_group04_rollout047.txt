import numpy as np
import time
import random
from scipy.optimize import differential_evolution, minimize

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    overlap = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(overlap) * dx
    return c5

def propose_move(h, max_step):
    """Propose a new feasible h vector by adjusting two elements"""
    h_new = h.copy()
    N = len(h)
    
    # Try to find a valid move
    max_tries = 100
    for _ in range(max_tries):
        i = random.randint(0, N - 1)
        j = random.randint(0, N - 1)
        if i == j:
            continue
            
        # Determine feasible interval for δ
        delta_low = max(-h_new[i], h_new[j] - 1)
        delta_high = min(1 - h_new[i], h_new[j])
        
        if delta_low >= delta_high:
            continue
            
        # Sample δ from uniform distribution over feasible range
        step = min(max_step, delta_high - delta_low)
        if step <= 0:
            continue
        delta = random.uniform(delta_low, delta_high)
        
        # Apply move
        h_new[i] += delta
        h_new[j] -= delta
        
        return h_new
    
    # If no valid move found, return original
    return h.copy()

def optimize_with_scipy(h_initial, budget_s):
    """Use scipy's optimization methods to refine solution"""
    start = time.time()
    N = len(h_initial)
    dx = 2.0 / N
    
    # Define objective
    def obj(h):
        # Ensure h is within bounds
        h = np.clip(h, 0, 1)
        # Constraint: integral equals 1
        integral_constraint = np.sum(h) * dx - 1.0
        # Set penalty for constraint violation
        penalty = 1000 * (integral_constraint ** 2)
        return c5_of(h) + penalty
    
    # Bounds: [0, 1] for each h[i]
    bounds = [(0, 1) for _ in range(N)]
    
    # Use differential evolution with custom reinitialization
    max_iter = 1000
    for i in range(max_iter):
        if time.time() - start > budget_s * 0.9:
            break
            
        # Run DE
        result = differential_evolution(
            obj, bounds, seed=42, maxiter=200, popsize=15,
            mutation=(0.5, 1), recombination=0.7, atol=1e-6, 
            callback=lambda x, convergence: None
        )
        
        # Refine with local solver
        try:
            refined = minimize(obj, result.x, method='L-BFGS-B', bounds=bounds)
            if not refined.success:
                continue
            h_initial = refined.x
        except:
            pass
            
        # Check if we're making progress
        if i % 50 == 0:
            current_c5 = c5_of(h_initial)
            if current_c5 < 0.3805:  # Early termination if good enough
                break
    
    return h_initial, c5_of(h_initial)
