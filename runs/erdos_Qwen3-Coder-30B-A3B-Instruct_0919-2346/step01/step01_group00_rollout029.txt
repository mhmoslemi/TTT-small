import numpy as np
import time
import random
from scipy.optimize import differential_evolution, minimize

def c5_of(h):
    """Compute C5 = max_k integral h(x)(1-h(x+k)) dx."""
    dx = 2.0 / len(h)
    # Compute correlation: h * (1 - h) with full mode
    c = np.correlate(h, 1 - h, mode="full")
    # Multiply by dx to get continuous integral
    c *= dx
    # Return maximum value among all shifts
    return np.max(c)

def project_to_feasible(h):
    """Project h to feasible region: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    dx = 2.0 / len(h)
    h = h * (1.0 / np.sum(h)) * (2.0 / dx)
    return h

def propose_move(h, max_step):
    """Generate a new feasible h by adjusting two entries."""
    h = h.copy()
    N = len(h)
    i, j = random.sample(range(N), 2)
    
    # Determine feasible interval for delta
    delta_low = max(-h[i], h[j] - 1.0)
    delta_high = min(1.0 - h[i], h[j])
    
    if delta_low >= delta_high:
        return h  # No valid move for this pair
    
    # Sample delta within allowed range
    step = min(max_step, delta_high - delta_low)
    delta = random.uniform(-step, step)
    delta = max(delta_low, min(delta, delta_high))
    
    h[i] += delta
    h[j] -= delta
    
    return project_to_feasible(h)

def local_search(h, max_iter=1000, max_step=0.05):
    """Local search using simulated annealing with small steps."""
    cur = h.copy()
    c_cur = c5_of(cur)
    best = cur.copy()
    c_best = c_cur
    
    for _ in range(max_iter):
        h_new = propose_move(cur, max_step)
        c_new = c5_of(h_new)
        
        if c_new < c_cur:
            cur = h_new
            c_cur = c_new
            if c_new < c_best:
                best = h_new
                c_best = c_new
        elif random.random() < 0.1:  # Occasionally accept worse moves
            cur = h_new
            c_cur = c_new
    
    return best, c_best

        return c5_of(h)

    # Use differential evolution on the whole space
    res = differential_evolution(objective, bnds, maxiter=100, popsize=10, seed=42)
    return res.x, res.fun
