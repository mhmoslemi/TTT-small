import numpy as np
import time
import random
from scipy import signal


def compute_c5(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    overlap = signal.correlate(h, 1 - h, mode="full")
    c5 = np.max(overlap) * dx
    return c5


def project_h(h, dx):
    """Project h to satisfy integral constraint and bounds"""
    # Ensure bounds
    h = np.clip(h, 0, 1)
    # Fix sum constraint
    sum_h = np.sum(h)
    if abs(sum_h - 1.0/dx) < 1e-10:
        return h
    factor = (1.0/dx) / sum_h
    h = h * factor
    h = np.clip(h, 0, 1)
    return h


def gradient_descent_step(h, step_size, beta=10.0):
    """Perform one gradient descent step"""
    dx = 2.0 / len(h)
    
    # Compute correlations
    corr_1 = signal.correlate(h, 1 - h, mode="full")
    corr_2 = signal.correlate(1 - h, h, mode="full")
    
    # Approximate worst-case shift index
    max_idx = np.argmax(corr_1)
    
    # Construct pseudo-gradient (simplified)
    # Gradient of the integral w.r.t. h(x) is the sum of convolved terms
    # We have int h(x)(1 - h(x + k)) dx, so partial w.r.t. h_i is:
    # sum_{k : x_i + k within bounds} (1 - h(x_i + k))
    
    # Simplified: gradient = (1 - h) * delta(0) + h * delta(0)
    # This is only an approximation for the extreme case, as gradient is not
    # available in closed form. For practical use, evaluate changes directly or
    # use numerical approximations. For this code, we'll use current local
    # structure to create directional moves.
    
    # For approximate gradient, we sample shifts around the worst index of correlation
    # and pick worst direction
    
    # Generate a new h by moving mass
    h_new = h.copy()
    
    # Sample random movement from worst correlation
    # These directions are also heuristic, based on local score change
    
    # Find indices near the peak of correlation (effectively, worst shift)
    peak_pos = max_idx - (len(h) - 1)
    peak_pos = max(0, min(len(h) - 1, peak_pos))
    
    # Make a directional move
    i = peak_pos
    if i < len(h) - 1:
        # Move some mass from i to i+1
        move_amount = min(h[i], step_size, 1 - h[i+1])
        # Ensure we don't ruin the integral
        if move_amount > 0:
            h_new[i] -= move_amount
            h_new[i+1] += move_amount
            
    # Apply projection to constraint sets
    h_new = project_h(h_new, dx)
    
    return h_new
