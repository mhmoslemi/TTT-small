import numpy as np
import time
from scipy.interpolate import interp1d

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value scaled by dx
    max_overlap = np.max(correlation) * dx
    return max_overlap

def upsample_initial(initial_h, n_target):
    """Upsample initial solution to target size using linear interpolation"""
    n_orig = len(initial_h)
    x_orig = np.linspace(0, 2, n_orig)
    x_target = np.linspace(0, 2, n_target)
    f = interp1d(x_orig, initial_h, kind='linear', fill_value='extrapolate')
    h_upsampled = f(x_target)
    return np.clip(h_upsampled, 0, 1)

def project_to_simplex(v):
    """Project vector v onto the simplex {x : sum x = n/2, x in [0,1]^n}"""
    n = len(v)
    # Values for bisection on lambda
    def phi(lam):
        x = np.clip(v - lam, 0, 1)
        return np.sum(x) - n/2
    # Bisection search
    lam_lo, lam_hi = np.min(v), np.max(v) + 1.0
    for _ in range(40):
        lam = (lam_lo + lam_hi) / 2
        if phi(lam) > 0:
            lam_lo = lam
        else:
            lam_hi = lam
    lam = (lam_lo + lam_hi) / 2
    x = np.clip(v - lam, 0, 1)
    return x

def gradient_single_shift(h, shift, dx):
    """Evaluate gradient contribution for a single shift"""
    n = len(h)
    grad = np.zeros(n)
    # For indices where both h[i] and h[i+shift] exist:
    if shift >= 0:
        # i ranges from 0 to n-1-shift
        i_valid = slice(0, n-shift)
        grad[i_valid] += 1 - h[i_valid + shift]
        # Subtract terms where i-shift exists:
        i_valid2 = slice(shift, n)
        grad[i_valid2] -= h[i_valid2 - shift]
    else:
        # i ranges from -shift to n-1
        i_valid = slice(-shift, n)
        grad[i_valid] += 1 - h[i_valid + shift]
        # Subtract terms where i-shift exists:
        i_valid2 = slice(0, n+shift)
        grad[i_valid2] -= h[i_valid2 - shift]
    return dx * grad

def softmax_weights(f_vals, gamma):
    """Compute softmax weights with temperature gamma"""
    f_max = np.max(f_vals)
    exp_f = np.exp(gamma * (f_vals - f_max))
    return exp_f / np.sum(exp_f)
