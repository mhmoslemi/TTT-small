import numpy as np
import time
from scipy.optimize import minimize
import math

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral via full convolution of h with (1-h)
    overlaps = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum of the full correlation vector
    # scaled by dx
    c5 = np.max(overlaps) * dx
    return c5

def propose_move(h, max_step):
    """Propose a feasible move by adjusting two entries."""
    h = h.copy()
    N = len(h)
    # Try to find a valid pair of indices
    for _ in range(100):  # avoid infinite loop
        i, j = np.random.randint(0, N, 2)
        if i == j:
            continue
        # Determine feasible δ range
        delta_low = max(-h[i], h[j] - 1)
        delta_high = min(1 - h[i], h[j])
        if delta_low >= delta_high:
            continue
        # Sample a move with limited step size
        step = min(max_step, delta_high - delta_low)
        delta = np.random.uniform(
            max(-step, delta_low),
            min(step, delta_high)
        )
        # Apply move
        h[i] += delta
        h[j] -= delta
        return h
    # If no valid move found, return copy
    return h

def local_search(initial_h, time_budget, seed):
    """Run local search with hill climbing."""
    np.random.seed(seed)
    start = time.time()
    cur = initial_h.copy()
    best = initial_h.copy()
    c_cur = c5_of(cur)
    c_best = c_cur
    
    while time.time() - start < time_budget:
        # Try a number of neighbor moves
        h_new = propose_move(cur, 0.05)
        c_new = c5_of(h_new)
        
        if c_new < c_cur:
            cur = h_new
            c_cur = c_new
            if c_new < c_best:
                best = h_new.copy()
                c_best = c_new
        else:
            # Small chance to accept uphill steps to escape local optima
            if np.random.rand() < 0.05:
                cur = h_new
                c_cur = c_new
    
    return best, c_best

def optimize_with_scipy(initial_h, time_budget, seed):
    """Use scipy optimization to refine solution."""
    start = time.time()
    
    def objective(h):
        return c5_of(h)
    
    def constraint(h):
        return np.sum(h) - len(h) / 2.0  # sum(h) should equal n/2
    
    N = len(initial_h)
    dx = 2.0 / N
    
    # Scale constraint to avoid numerical issues
    cons = {'type': 'eq', 'fun': constraint}
    
    # Bounds for each variable
    bounds = [(0, 1) for _ in range(N)]
    
    # Optimize
    try:
        result = minimize(objective, initial_h, method='SLSQP', 
                         bounds=bounds, constraints=cons,
                         options={'maxiter': 1000, 'ftol': 1e-8})
        if result.success:
            h_opt = result.x.copy()
            # Make sure constraint is satisfied
            h_opt *= (N / 2) / np.sum(h_opt)
            c_opt = c5_of(h_opt)
            return h_opt, c_opt
    except:
        pass
    
    return initial_h, c5_of(initial_h)

    N = 140

    # Generate multiple seeds

    h_uniform = 0.5 * np.ones(N)
    h_uniform *= (N / 2) / np.sum(h_uniform)

    # 2. Initial_h_values seed (if available)
    if 'initial_h_values' in globals():

        h_initial = h0 * (N / 2) / np.sum(h0)
    else:
        h_initial = np.random.rand(N) * 0.5
        h_initial *= (N / 2) / np.sum(h_initial)

    # 3. Double-peak seed
    h_double_peak = np.full(N, 0.1)
    a = 0.3
    idxs = np.where((np.linspace(0, 2, N) < a) | (np.linspace(0, 2, N) > (2 - a)))[0]
    h_double_peak[idxs] = 0.9
    h_double_peak *= (N / 2) / np.sum(h_double_peak)

    # 4. Triple-peak seed
    h_triple_peak = np.full(N, 0.2)
    a = 0.2
    idxs = np.where((np.linspace(0, 2, N) < a) |

    h_triple_peak[idxs] = 0.8
    h_triple_peak *= (N / 2) / np.sum(h_triple_peak)

    # Run local search on seeds
    best_h = None
    best_c5 = float('inf')
    seeds_to_try = [h_uniform, h_initial, h_double_peak, h_triple_peak]
    budget_per_run = 0.2 * budget_s
    
    for i, h_seed in enumerate(seeds_to_try):

            break

    # Further refinement with scipy optimization

        if c_refined < best_c5:
            best_h = h_refined
            best_c5 = c_refined

    # Final local search to polish
    if best_h is not None and time.time() - start < budget_s * 0.95:
        polish_budget = budget_s - (time.time() - start)
        h_final, c_final = local_search(best_h, polish_budget, seed + 100)
