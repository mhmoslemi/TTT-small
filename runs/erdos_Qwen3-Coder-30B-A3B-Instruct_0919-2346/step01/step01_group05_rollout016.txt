import numpy as np
from scipy.optimize import minimize
from math import pi
import time

def build_h_from_coeffs(a_vec, n):
    """Build h vector from Fourier coefficients"""
    h = 0.5 * np.ones(n)
    M = len(a_vec)
    for m in range(1, M+1):
        cos_term = np.cos(2 * pi * m * np.arange(n) / n)
        h += a_vec[m-1] * cos_term
    return h

def compute_overlap_and_gradient(a_vec, n, dx, gamma, penalty_weight=1e4):
    """Compute overlap and gradient for current coefficients"""
    h = build_h_from_coeffs(a_vec, n)
    # Apply clipping to enforce box constraints
    h_clipped = np.clip(h, 0, 1)
    
    # Compute the correlation (equivalent to the overlaps for all shifts)
    # Using numpy's correlate to get full correlation
    corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
    # Extract overlaps for shifts -n+1 to n-1 (shifts 0 to 2*n-2 in full)
    overlaps = np.array([corr[i + n - 1] for i in range(2*n - 1)]) * dx
    # Maximize over all shifts (possibly compute softmax)
    max_overlap = np.max(overlaps)
    
    # Now compute gradient of the soft max version:
    # softmax = 1/gamma * log(sum(exp(gamma*overlaps))) = Phi_gamma
    # gradient = sum_{s} w_s * df_s/da_m
    # where w_s = exp(gamma*f_s) / sum_k exp(gamma*f_k)
    
    # For softmax loss
    exp_gamma_f = np.exp(gamma * overlaps)
    w = exp_gamma_f / np.sum(exp_gamma_f)
    
    # Gradient w.r.t. a_m
    grad = np.zeros_like(a_vec)
    for m in range(1, len(a_vec)+1):
        cos_term = np.cos(2 * pi * m * np.arange(n) / n)
        # Two terms from chain rule:
        # df_s/da_m = dx * sum_{i} cos_term[i] (1-h_{i+s}) - h_i cos_term[i+s]
        term1 = np.zeros(n)
        term2 = np.zeros(n)
        for s in range(-n+1, n):
            # shift by s
            # We compute partial derivative of the overlap including shift s
            # This is tricky with the way we access index. We recalculate:
            # f_s = dx * sum_{i=0}^{n-1} h_i * (1 - h_{i+s})
            # Note: i+s can be out of bounds. We handle it properly.
            # In Python, we can handle this using slicing or by defining explicit loops.
            # But simpler to use convolution-like method:

            # For now, we compute separately for each s inside the loop.
            # But since the chain rule looks like:
            # df_s/da_m = dx * sum_{i such that i+s is valid} [cos_term[i] (1 - h[i+s]) - h[i] cos_term[i+s]]
            # So let's compute:
            # Let's switch to simpler approach: we do full convolution in terms of FFT, and then take derivatives manually.
            # But first, let's make an explicit loop over shifts.
            
            # The complete gradient computation is tricky because of indexing
            # We'll compute these numerically for better consistency
            pass
            
        # Simplified approach: just compute this numerically 
        df_da = np.zeros(n)  # This will store derivative contribution for each point from shift s.
        # Fill in gradient for this mode m.
        df_da_m = 0.0
        for s in range(-n+1, n):
            # index shift handling:
            if s < 0:
                # h_{i+s} where i+s >= 0 and i+s < n means i >= -s and i < n, i >= 0, so i >= max(0, -s)
                # i+s < n means i < n-s, and i >= 0, so i < n-s and i >= 0
                start = max(0, -s)
                end = min(n, n - s)
                # So for i in [start, end), h_{i+s} exists
                if end > start and (end - start) > 0:
                    # Shifted indices: i+s
                    valid_idx = np.arange(start, end)
                    # Compute sum over all valid i 
                    term1_sum = np.sum(cos_term[valid_idx] * (1 - h_clipped[valid_idx+s]))
                    term2_sum = np.sum(h_clipped[valid_idx] * cos_term[valid_idx+s])
                    df_da_m += (term1_sum - term2_sum) * dx
            elif s == 0:
                # i+s = i 
                df_da_m = np.sum(cos_term * (1 - h_clipped)) * dx - np.sum(h_clipped * cos_term) * dx
            else:
                # s > 0, h_{i+s} where i+s < n means i < n - s
                start = 0
                end = min(n, n - s)
                if end > start:
                    # i = 0 to end-1, (i+s) valid?
                    valid_idx = np.arange(start, end)
                    term1_sum = np.sum(cos_term[valid_idx] * (1 - h_clipped[valid_idx+s]))
                    term2_sum = np.sum(h_clipped[valid_idx] * cos_term[valid_idx+s])
                    df_da_m += (term1_sum - term2_sum) * dx
        grad[m-1] = df_da_m
    
    # Penalty term gradient for h being in [0,1]
    # ∂P/∂a_m = sum_i (h_i - clip(h_i,0,1)) * cos_term[i]
    penalty_grad = np.zeros_like(a_vec)
    for m in range(1, len(a_vec)+1):
        cos_term = np.cos(2 * pi * m * np.arange(n) / n)
        diff = h - h_clipped
        penalty_grad[m-1] = np.sum(diff * cos_term)
    
    # Combine with soft-max and penalty gradients
    grad = np.sum(w * grad) + penalty_weight * penalty_grad
    
    # Penalty for being out of bounds
    penalty = penalty_weight * (np.sum(np.maximum(h - 1, 0)**2) + np.sum(np.maximum(-h, 0)**2))
    
    # Combine objective and penalty
    objective = max_overlap + penalty
    
    return objective, grad

def project_to_simplex_box(h, target_sum):
    """Project h onto the simplex where sum=h_sum and h is in [0,1]"""
    # Use water-filling algorithm
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-12:
        return h
    
    # Sort h in descending order
    h_sorted = np.sort(h)[::-1]
    # Compute cumulative sums
    cumsum = np.cumsum(h_sorted)    
    # Find threshold point
    n = len(h_sorted)
    # Each index k means first k elements are above threshold.
    # We want the threshold tau such that sum of clip(h_i - tau, 0, 1) == target_sum
    # Find k such that:
    # sum_{i=0}^{k-1} (h_sorted[i] - tau) + sum_{i=k}^{n-1} 0 = target_sum
    # That is, k*tau <= sum_{i=0}^{k-1} h_sorted[i]  => tau <= sum_{i=0}^{k-1} h_sorted[i]
    # Also we impose that (h_sorted[k] - tau) >= 0
    # So threshold is sum(h_sorted[:k]) / k  if k > 0
    # 
    # Let x be so that sum_{i=0}^{k} (h_sorted[i] - x) = target_sum
    # sum_{i=0}^{k} h_sorted[i] - (k+1)*x = target_sum 
    # x = (sum_{i=0}^{k} h_sorted[i] - target_sum) / (k+1)
    target = target_sum
    x = h_sorted[:n]
    for k in range(n):
        if k == 0:
            # x_k = (h[0] - target) / 1  <-- invalid unless treated carefully  
            val = h_sorted[0] - target
            if val <= 0:
                # all x > h_sorted[0] then all will become capped at 1  
                result = np.ones_like(h)
                return result
            else:
                # But we don't proceed with any such thresholds.
                pass
        else:
            left_sum = np.sum(h_sorted[:k])
            if k > 0 and (left_sum - target) < 0:
                continue
            else:
                x_k = (left_sum - target) / k
                for i in range(k):
                    x[i] = max(0, min(1, h_sorted[i] - x_k))
                break
    else:
        # Default for remaining values
        if len(h_sorted) > 0:
            x_k = h_sorted[0] - target / len(h_sorted)
            for i in range(len(h_sorted)):
                if i < len(h_sorted):
                    x[i] = max(0, min(1, h_sorted[i] - x_k))
        else:
            return h
    
    # Assign back to original indices
    x_result = np.empty_like(h)
    argsort = np.argsort(h)[::-1]  # get indices that sort in descending order
    for idx, val in enumerate(x):
        x_result[argsort[idx]] = val
    return x_result

def spectral_optim(n, gamma_init=0.5, gamma_final=40.0, gamma_factor=1.5,
                   max_iter=3000, penalty_weight=1e4, bounds=(-0.6, 0.6)):
    """
    Spectral optimisation using Fourier coefficients
    """
    # Initial coefficients for low frequencies 
    # Start with small enough number of modes
    M = 8  # This is also being varied in stages
    
    # Fixed coefficients for typically 200 points: 
    # good initial ones drawn from normal with std=0.05
    a_start = np.random.normal(0, 0.05, M)
    
    # gamma schedule: start small, grow to sharp
    gamma = gamma_init
    if gamma <= 0:
        gamma = 1e-9
    gamma_list = []
    for i in range(max_iter):
        gamma_list.append(gamma)
        if i % 500 == 0 and i > 0:
            gamma *= gamma_factor
            if gamma > gamma_final:
                gamma = gamma_final
        if gamma > gamma_final:
            break
    
    def fun(a_vec):
        return compute_overlap_and_gradient(a_vec, n, 2.0/n, gamma, penalty_weight)[0]
    
    def jac(a_vec):
        return compute_overlap_and_gradient(a_vec, n, 2.0/n, gamma, penalty_weight)[1]
        
    # bounds on a_vec
    bnds = [bounds for _ in range(len(a_start))] 
    
    # Optimize
    result = minimize(fun, a_start, method='L-BFGSB', jac=jac, bounds=bnds,
                      options={'maxiter': max_iter, 'ftol': 1e-9, 'gtol': 1e-9})
    
    return result.x, result.fun

    # Progression from small M to full M in stages
    all_results = []

    # Stage 3

    perf_list.append(cost)
    all_results.append((a_best.copy(), cost))
        
    # Now use the best of these for final polishing
    best_idx = np.argmin(perf_list)
    a_final = all_results[best_idx][0]
    h_final = build_h_from_coeffs(a_final, n)
    
    # Final projection onto feasible set: h in [0,1] and sum h = 1
    h_final = np.clip(h_final, 0, 1) 
    h_final = project_to_simplex_box(h_final, 1.0)
    
    # Soft check
    sum_h = np.sum(h_final)
    if (abs(sum_h - 1.0) > 1e-3):
        # Just rescale to avoid failures—not normally needed
        h_final = h_final / sum_h
        h_final = project_to_simplex_box(h_final, 1.0)
        
    return h_final
