import numpy as np
import time
from scipy.optimize import minimize
from scipy.special import logsumexp

def evaluate_c5(h):
    """Evaluate the true C5 value using the original formulation."""
    dx = 2.0 / len(h)
    c = np.correlate(h, 1 - h, mode='full')
    return np.max(c) * dx

def project_to_feasible(h, N_half):
    """Project h onto the feasible set: 0 <= h[i] <= 1 and sum(h) = N_half."""
    # Binary search for the Lagrange multiplier
    v = h.copy()
    low, high = -1000.0, 1000.0
    for _ in range(50):
        lam = (low + high) / 2.0
        projected = np.clip(v + lam, 0, 1)
        total = np.sum(projected)
        if total > N_half:
            low = lam
        else:
            high = lam
    lam = (low + high) / 2.0
    h_proj = np.clip(v + lam, 0, 1)
    # Adjust for precision errors
    diff = N_half - np.sum(h_proj)
    if abs(diff) > 1e-10:
        # Redistribute the difference on free entries
        free_mask = (h_proj > 0) & (h_proj < 1)
        if np.sum(free_mask) > 0:
            h_proj[free_mask] += diff / np.sum(free_mask)
    return np.clip(h_proj, 0, 1)

def compute_smooth_surrogate(h, eps):
    """Compute F_eps(h) = eps * log(sum_s exp(C_s/eps))"""
    dx = 2.0 / len(h)
    c = np.correlate(h, 1 - h, mode='full')
    c_scaled = c * dx
    # subtract max for numerical stability
    max_c = np.max(c_scaled)
    if max_c == -np.inf:
        max_c = 0
    c_scaled_stable = c_scaled - max_c
    F_eps = eps * logsumexp(c_scaled_stable)
    return F_eps, c_scaled

def compute_gradient(h, eps):
    """Return ∇F_eps(h)"""
    dx = 2.0 / len(h)
    c = np.correlate(h, 1 - h, mode='full')
    c_scaled = c * dx
    
    # numerical stability
    max_c = np.max(c_scaled)
    if max_c == -np.inf:
        max_c = 0
    c_scaled_stable = c_scaled - max_c
    w = np.exp(c_scaled_stable)
    w = w / np.sum(w)
    
    # Gradient = sum_s w_s * grad(C_s)
    grad = np.zeros_like(h)
    N = len(h)
    
    # For each shift s, compute gradient of C_s
    for s in range(len(c)):
        # for shift s, the gradient wrt h[i] is 1 - h[i+s] - h[i-s] (zero padding)
        shift_left = np.roll(h, s)
        shift_right = np.roll(h, -s)
        grad_term = 1 - shift_left - shift_right
        # Handle zero padding
        if s < N:
            # s is a valid shift that pairs the two arrays properly
            # We just set corresponding elements to 1 if the index is valid,
            # and 0 where it's invalid (i.e., padding).
            # Instead, do a reverse engineering as valid part in lhs offset someone else.
            valid_mask = np.arange(N) < (N - s)
            grad_term[:s] = 0
            #for i in range(N):
            #    if i - s >= 0 and i + s < N:
            #        grad_term[i] += 1 - h[i+s] - h[i-s] 
            #    else:
            #        grad_term[i] = 0
        else:
            # This is a far shift: most probably give 0
            pass
        # i-th component of ∇C_s = dx*(1 - h[i+s] - h[i-s])
        # So ∇F_eps = sum_s w_s * dx * (1 - h[i+s] - h[i-s]) 
        # We need gradient components for each i with proper addressing
        # So the easier algorithm: compute shift_left and shift_right correctly
    
        # Shifting logic for vectors:
        shift_left = np.zeros_like(h)
        shift_right = np.zeros_like(h)
        
        # Much simpler:
        # shift_left[i] = h[(i-s)%N] nếu s>=0, else 0 (i.e., need to handle zero-padding)
        if s < N:
            shift_left[s:] = h[:-s]
            shift_left[:s] = 0
            shift_right[:-s] = h[s:]
            shift_right[-s:] = 0
            # But need exact consistency with np.correlate which uses zero padding on both ends, same as np correlate?
            # For shift s = 0 to N-1:
            # grad_C_s = dx * (1 - h[:N] - h[s:s+N])
            # This is wrong indexing probably. Need to ensure that entries at the end are 0 from paddings.
            # Actually we are going to use numpy.correlate semantics to ensure standards.
            # next let's calculate element by element accordingly to np correlation.
            shift_left = np.zeros_like(h)
            shift_right = np.zeros_like(h)
            shift_left[s:] = h[:-s]  
            shift_left[:s] = 0        
            shift_right[:-s] = h[s:]
            shift_right[-s:] = 0
        else:
            # If s >= N, then nothing is valid
            shift_left = np.zeros_like(h)
            shift_right = np.zeros_like(h)
        # Subtract 1 to align properly even though the expressions differ slightly:
        # But let's match what works. Use what we know works based on .
        # deriving very directly for loop and matching np.correlate behavior
        
        # The shape with mode='full' gives len + len -1 = 2*len - 1 entries.
        # For array1[A] and array2[B], correlate returns i from -(B-1) to (A-1) = -len+1 to len-1
        # And shift i means shift by i, so for i=0, no shift, for i=+s, shift_right by s
        # for i=-s, shift_left by s
        # The gradients are the same.
        # Let’s recompute carefully:
        K = len(c)
        grad += w[s] * dx * (1 - shift_left - shift_right)
    return grad

def continuation_optimization(h0, N_half, eps_list, max_iter=150, tol=1e-6):
    h = h0.copy()
    h = project_to_feasible(h, N_half)
    for eps in eps_list:
        # Inner optimization using L-BFGS-B
        def obj(h):
            val, _ = compute_smooth_surrogate(h, eps)
            return val
            
        def grad(h):
            return compute_gradient(h, eps)
        
        def fun_and_grad(h):
            val, _ = compute_smooth_surrogate(h, eps)
            g = compute_gradient(h, eps)
            return val, g

        # Optimize with L-BFGS-B
        res = minimize(
            fun_and_grad,
            h,
            jac=True,
            bounds=[(0, 1)] * len(h),
            method='L-BFGS-B',
            options={'maxiter': max_iter, 'gtol': tol}
        )
        h = res.x
        h = project_to_feasible(h, N_half)
    return h

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    N = 140
    N_half = N / 2
    dx = 2.0 / N
    
    # Initialize seeds
    seeds = []
    
    # Uniform seed
    h_uniform = np.ones(N) * 0.5
    h_uniform = project_to_feasible(h_uniform, N_half)
    seeds.append(("uniform", h_uniform.copy()))

    # Two-peak seed
    a_vals = [0.12, 0.18, 0.24, 0.30]
    for a in a_vals:
        h = np.ones(N) * 0.1
        x_vals = np.linspace(0, 2, N)
        mask = (x_vals < a) | (x_vals >= 2 - a)
        h[mask] = 0.9
        h = project_to_feasible(h, N_half)
        seeds.append((f"two_peak_a_{a}", h.copy()))
    
    # Binary seeds
    for i in range(5):
        h = np.zeros(N)
        indices = np.random.choice(N, N // 2, replace=False)
        h[indices] = 1.0
        h = project_to_feasible(h, N_half)
        seeds.append((f"binary_seed_{i}", h.copy()))
        
    # Symmetric two-block
    h = np.zeros(N)
    h[:N//2] = 1
    h = project_to_feasible(h, N_half)
    seeds.append(("sym_two_blocks", h.copy()))
    
    # Random continuous seed
    h = np.random.rand(N)
    h = project_to_feasible(h, N_half)
    seeds.append(("random_continuous", h.copy()))

    # Continuation sequence for ε
    eps_list = [0.10, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]  

    # Process each seed
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    
    for name, h_start in seeds:
        if time.time() - start_time > 0.9 * budget_s:
            break
        try:
            # Run continuation optimization
            h_opt = continuation_optimization(h_start, N_half, eps_list, 
                                             max_iter=150, tol=1e-6)
            c5 = evaluate_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()
        except Exception:
            continue
            
    # Final polishing with sub-gradient descent on the max version
    if best_h is not None and time.time() - start_time < 0.95 * budget_s:
        h = best_h.copy()
        # Optimistic estimate: refine for the remaining 30 seconds
        last_c5 = best_c5
        final_time = time.time() + 0.1 * budget_s
        while time.time() < final_time:
            # Find worst shift
            dx = 2.0 / N
            c = np.correlate(h, 1 - h, mode='full')
            max_idx = np.argmax(c)
            max_shift = max_idx - (N - 1)
            # compute gradient for that shift
            shift_left = np.roll(h, max_shift)
            shift_right = np.roll(h, -max_shift)
            # Use zero-padding as done by np.correlate using standard normalization
            # For shift s that is valid, grad[i] = dx*(1 - h[i+s] - h[i-s])
            # Actually we do a shift by zeros on both sides equal to the  
            # length of the output (so 0-padding on both ends 0 to N-1)  
            # Each. the entries get corrected.
            # Simply define shifts:
            # Need to define a modification for s
            # Because for shift s = k, the contribution to index i in output indexed by i-k,
            # we see that for shift -s, which is compatible with Python indexing, 
            # looks like shift_right = np.roll(h, -s) and  shift_left = np.roll(h, s) 
            # but since numpy does zero-padding, so its shift left and right of PO 
            # in position i correspond to original values at i+s and i-s respectively 
            # with proper attention to truncation.

            # Construct shift for the exact shift that caused the max
            shift_left = np.zeros_like(h) 
            shift_right = np.zeros_like(h)
            if max_shift <= 0:
                # positive shift, so shift_right of max_shift
                shift_left[N + max_shift:] = h[:N - max_shift]
                shift_right[:N + max_shift] = h[max_shift:]
            else:
                # negative shift, so shift_right = offset -max_shift,  make it inside
                shift_left[max_shift:] = h[:-max_shift]
                shift_right[:-max_shift] = h[max_shift:]
            # Gradient of C_s = dx*(1 - h[i+s] - h[i-s]) 
            grad = dx * (1 - shift_left - shift_right)
            h_new = h - 0.03 * grad
            h_new = project_to_feasible(h_new, N_half)
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h = h_new.copy()
            if abs(c5_new - last_c5) < 1e-7:
                break
            last_c5 = c5_new
            
    return (best_h, best_c5, N)
