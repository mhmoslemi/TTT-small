import numpy as np
import cvxpy as cp
import time
import math

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    overlap = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(overlap) * dx
    return c5

    best_sol = None
    best_C = C_high + 1

    # Define bounds for time allowed per feasibility search
    max_seconds_per_search = 60
    
    # Timing logic (subloop) for better speed control
    def solve_feasibility_with_solver(C_bound):
        # Build MILP model from scratch each time
        x = cp.Variable(N, boolean=True)
        y_vars = {}
        # All y[i, ℓ] for ℓ=1..N-1, i=0..N-ℓ-1
        for ℓ in range(1, N):
            for i in range(N - ℓ):
                name = f"y_{ℓ}_{i}"
                y_vars[(ℓ, i)] = cp.Variable(boolean=True)
        
        # Constraints
        # Measure constraint
        constraints = [cp.sum(x) == N // 2]
        
        # Product constraints
        for ℓ in range(1, N):
            for i in range(N - ℓ):
                y = y_vars[(ℓ, i)]
                constraints.append(y <= x[i])
                constraints.append(y <= x[i + ℓ])
                constraints.append(y >= x[i] + x[i + ℓ] - 1)
        
        # Overlap constraints (bound C)
        Cvar = cp.Variable(1, integer=True)
        constraints.append(Cvar == C_bound)
        for ℓ in range(1, N):
            H_ℓ = cp.sum(x[:N - ℓ])
            O_ℓ = cp.sum([y_vars[(ℓ, i)] for i in range(N - ℓ)])
            constraints.append(H_ℓ - O_ℓ <= Cvar)
        
        # Objective: dummy
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        # Solve
        try:
            prob.solve(solver=cp.CBC, verbose=False, max_seconds=max_seconds_per_search)
            if prob.status in [cp.OPTIMAL, cp.FEASIBLE]:
                return True, np.array([int(v.value) for v in x])
        except:
            pass
        return False, None
    
    # Binary search
    while C_low < C_high:
        mid = (C_low + C_high) // 2
        
        # Check feasibility
        feasible, sol = solve_feasibility_with_solver(mid)
        elapsed = time.time() - start_time
        if elapsed >= budget_s - 10:
            break
            
        if feasible:
            best_C = mid
            best_sol = sol
            C_high = mid
        else:
            C_low = mid + 1

    # Final try to evaluate best solution found and return
    if best_sol is None:

                # Temporarily swap
                fixed_h[i], fixed_h[j] = fixed_h[j], fixed_h[i]
                
                # Recompute C5
                c_new = c5_of(fixed_h)
                
                # Accept only if better

                    c5 = c_new
                else:
                    # Swap back
                    fixed_h[i], fixed_h[j] = fixed_h[j], fixed_h[i]

    return (fixed_h, c5, N)
