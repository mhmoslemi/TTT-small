import numpy as np
import time
import random
from scipy.optimize import differential_evolution, minimize


def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    overlap = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(overlap) * dx
    return c5


def propose_move(h, max_step):
    """Propose a new feasible h vector by adjusting two elements"""
    h_new = h.copy()
    N = len(h)
    
    # Try to find a valid move
    max_tries = 100
    for _ in range(max_tries):
        i = random.randint(0, N - 1)
        j = random.randint(0, N - 1)
        if i == j:
            continue
            
        # Determine feasible interval for δ
        delta_low = max(-h_new[i], h_new[j] - 1)
        delta_high = min(1 - h_new[i], h_new[j])
        
        if delta_low >= delta_high:
            continue
            
        # Sample δ from uniform distribution over feasible range
        step = min(max_step, delta_high - delta_low)
        if step <= 0:
            continue
        delta = random.uniform(delta_low, delta_high)
        
        # Apply move
        h_new[i] += delta
        h_new[j] -= delta
        
        return h_new
    
    # If no valid move found, return original
    return h.copy()


def optimize_with_de(h_initial, budget_s):
    """Use differential evolution for global optimization"""
    start = time.time()
    N = len(h_initial)
    dx = 2.0 / N
    
    def objective(h):
        # Normalize
        h = h.copy()
        h = h * (N / 2) / np.sum(h)
        return c5_of(h)
    
    # Bounds for each variable
    bounds = [(0, 1) for _ in range(N)]
    
    # Run differential evolution
    try:
        result = differential_evolution(
            objective, 
            bounds, 
            maxiter=int(budget_s/2), 
            popsize=15, 
            seed=42,
            mutation=(0.5, 1.0),
            recombination=0.7
        )
        h_opt = result.x
        h_opt = h_opt * (N / 2) / np.sum(h_opt)
        c5_opt = result.fun
        return h_opt, c5_opt
    except:
        return h_initial.copy(), c5_of(h_initial)


def refine_solution(h_initial, time_budget_s):
    """Refine solution using local search and simulated annealing"""
    start = time.time()
    best_h = h_initial.copy()
    best_c5 = c5_of(best_h)
    
    # Local refinement using scipy minimize
    N = len(h_initial)
    dx = 2.0 / N
    
    def obj(h):
        h_normalized = h * (N / 2) / np.sum(h) 
        return c5_of(h_normalized)
    
    # Constraints
    def constraint_sum(h):
        return np.sum(h) - N / 2
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Bounds
    bounds = [(0, 1) for _ in range(N)]
    
    # Try different initial points from the current best
    for _ in range(5):
        if time.time() - start > time_budget_s - 5:
            break
        try:
            # Add small random noise to current solution
            h_perturbed = best_h + np.random.normal(0, 0.01, N)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            
            # Optimize with constraints
            result = minimize(obj, h_perturbed, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
            if result.success:
                h_opt = result.x
                h_opt = h_opt * (N / 2) / np.sum(h_opt)
                c5_opt = obj(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            continue
    
    return best_h, best_c5

    else:
        h_provided = np.full(N, 0.5)

    # Try a few different approaches
    best_h = h_provided.copy()
    best_c5 = c5_of(best_h)
    
    # Approach 1: Differential evolution on the initial solution
    if time.time() - start < budget_s * 0.3:
        h_de, c5_de = optimize_with_de(h_provided, budget_s * 0.3)
        if c5_de < best_c5:
            best_c5 = c5_de
            best_h = h_de.copy()
    
    # Approach 2: Refinement with local optimization
    if time.time() - start < budget_s * 0.7:
        h_refined, c5_refined = refine_solution(best_h, budget_s * 0.4)
        if c5_refined < best_c5:
            best_c5 = c5_refined
            best_h = h_refined.copy()
    
    # Approach 3: Hill climbing with simulated annealing
    if time.time() - start < budget_s - 10:
        # Run a few iterations of simulated annealing to polish
        cur = best_h.copy()
        c_cur = c5_of(cur)
        
        T_start = 0.1
        T_end = 1e-6
        step_start = 0.1
        
        # Cooling schedule
        for i in range(20000):
            if time.time() - start > budget_s - 5:
                break
                
            t = time.time() - start
            elapsed_fraction = t / budget_s
            
            T = T_start * (T_end / T_start) ** elapsed_fraction
            step = step_start * (1 - elapsed_fraction)
            if step <= 0:
                step = 1e-5
                
            h_proposed = propose_move(cur, step)
            c_proposed = c5_of(h_proposed)
            
            if c_proposed <= c_cur:
                cur = h_proposed
                c_cur = c_proposed
                if c_proposed < best_c5:
                    best_c5 = c_proposed
                    best_h = h_proposed.copy()

                    cur = h_proposed
                    c_cur = c_proposed
    
    return (best_h, best_c5, N)
