def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral via full convolution of h with (1-h)
    overlaps = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum of the full correlation vector
    # scaled by dx
    c5 = np.max(overlaps) * dx
    return c5

def evaluate_constraint(h, n_points):
    """Check if the integral constraint is satisfied"""
    return np.sum(h) * (2.0 / n_points) - 1.0

def project_to_feasible(h, n_points):
    """Project h to satisfy the integral constraint"""
    current_sum = np.sum(h)
    target_sum = n_points / 2
    scale = target_sum / current_sum
    h_proj = h * scale
    return np.clip(h_proj, 0, 1)

def propose_move(h, max_step):
    """Propose a feasible move by adjusting two entries."""
    h = h.copy()
    N = len(h)
    # Try to find a valid pair of indices
    for _ in range(100):  # avoid infinite loop
        i, j = np.random.randint(0, N, 2)
        if i == j:
            continue
        # Determine feasible δ range
        delta_low = max(-h[i], h[j] - 1)
        delta_high = min(1 - h[i], h[j])
        if delta_low >= delta_high:
            continue
        # Sample a move with limited step size
        step = min(max_step, delta_high - delta_low)
        delta = np.random.uniform(
            max(-step, delta_low),
            min(step, delta_high)
        )
        # Apply move
        h[i] += delta
        h[j] -= delta
        return h
    # If no valid move found, return copy
    return h

def gradient_descent_step(h, learning_rate=0.01):
    """Refine h using gradient-based approach"""

    n_points = len(h)
    dx = 2.0 / n_points
    
    # Sample a few nearby values to estimate gradient
    grad = np.zeros_like(h)

    c5_base = c5_of(h)

        c5_plus = c5_of(h_plus)

    # Update with gradient descent
    h_new = h - learning_rate * grad
    h_new = project_to_feasible(h_new, n_points)
    
    return h_new
