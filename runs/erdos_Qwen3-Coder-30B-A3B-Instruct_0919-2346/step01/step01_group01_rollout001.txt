import numpy as np
import time
import cvxpy as cp

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using full convolution
    overlap = np.correlate(h, 1 - h, mode='full')
    # The maximum value scaled by dx
    c5 = np.max(overlap) * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with smaller N to reduce time
    N = 139 # odd so that we can construct difference set
    dx = 2.0 / N
    
    # Try to construct binary solution with difference set properties
    # For prime p = 139 ≡ 1 (mod 4), there exists Paley difference set of size (p-1)/2 = 69
    # This gives each non-zero shift intersection of exactly λ = (p-3)/4 = 34
    # So we'll use N=139 with 69 ones; we'll add a small extra element to make it exactly sum = N/2 = 69.5
    
    # Initialize h with 69 ones, 70 zeros
    h = np.zeros(N)
    h[:69] = 1
    
    # Try to assign fractional part to make sum exactly N/2 = 69.5
    # We'll make one element 0.5 to balance the sum
    h[69] = 0.5  # Now total sum = 69 + 0.5 = 69.5
    
    # Use the branch and bound solution from MILP for more precise result
    # Try solving it using cvxpy semi-integer constraint 
    
    # Construct MILP to find binary configuration with high minimum correlation
    # Try N=140 for highest fidelity to previous work but make bounded process
    N = 140
    dx = 2.0 / N
    
    # Now solve optimization problem with cvxpy
    problem_start = time.time()
    
    try:
        # Create variables
        x = cp.Variable(N, boolean=True)
        C = cp.Variable()
        
        # Constraints
        constraints = [cp.sum(x) == N // 2]
        
        # For each lag compute H and O
        # Optimized way: iterate once
        for ell in range(1, N):
            length = N - ell
            if length <= 0:
                continue
                
            # Compute H_ℓ = sum_{i=0}^{N-ℓ-1} x_i
            H_l = cp.sum(x[:length])
            
            # Introduce y variables for products x_i * x_{i+ℓ} 
            y = cp.Variable(length, boolean=True)
            
            # Constrain y to equal x[i] * x[i+ℓ]
            # By using big M constraints
            for i in range(length):
                constraints.append(y[i] <= x[i])
                constraints.append(y[i] <= x[i+ell])
                constraints.append(y[i] >= x[i] + x[i+ell] - 1)
                
            O_l = cp.sum(y)
            c_l = H_l - O_l
            
            # Add to constraints
            constraints.append(c_l <= C)
            constraints.append(c_l >= 0)  # Ensure c_l >= 0
            
        # Define objective 
        prob = cp.Problem(cp.Minimize(C), constraints)
        
        # Solve
        time_left = max(1, budget_s - (time.time() - start_time) - 2)
        prob.solve(solver=cp.CBC, max_seconds=time_left, verbose=False) 
        
        # If solved optimally, then retrieve solution
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            x_val = x.value
            h = np.round(x_val).astype(float)
            # Check if we get sum = 70 (half of 140)
            if abs(np.sum(h) - 70) < 1e-5:
                pass
            else:
                # Resort to binary until we have 70 ones
                h = np.zeros(N)
                h[:70] = 1
                h = h.astype(float)  # Ensure float array
            c5_bound = c5_of(h)
        else:
            # Fallback to local optimization approach
            # Use a rough binary starting point
            h = np.zeros(N)
            h[:70] = 1
            h = h.astype(float)
            c5_bound = c5_of(h)
            
    except Exception as e:
        # Fall back if LP fails
        h = np.zeros(N)
        h[:70] = 1
        h = h.astype(float)  
        c5_bound = c5_of(h)
        
    # Ensure feasibility and cleanup if needed
    if abs(np.sum(h) - 70) > 1e-5:
        h[:70] = 1
        h[70:] = 0
        h = h.astype(float)
    c5_bound = c5_of(h)
    
    # Possibly even better strategy: solve with binary matrix / exact algorithm
    # But with reasonable options in place
    
    return (h, c5_bound, N)
