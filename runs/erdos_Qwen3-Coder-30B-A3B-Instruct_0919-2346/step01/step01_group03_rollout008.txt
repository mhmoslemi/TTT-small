import numpy as np
import time

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral via full convolution of h with (1-h)
    overlaps = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum of the full correlation vector
    # scaled by dx
    c5 = np.max(overlaps) * dx
    return c5

def full_overlap(h):
    """Compute the raw correlation vector (unscaled)"""
    return np.correlate(h, 1 - h, mode="full")

def project_onto_P(v, total):
    """Project vector v onto the constraint set {x: 0 ≤ x_i ≤ 1, sum x_i = total}"""
    # Bisection to find lambda
    def phi(lam):
        # x = clip(v - lam, 0, 1)
        x = np.clip(v - lam, 0, 1)
        return np.sum(x) - total
    
    a, b = np.min(v), np.max(v) + (np.sum(v) - total)
    while b - a > 1e-12:
        mid = (a + b) / 2
        if phi(mid) > 0:
            a = mid
        else:
            b = mid
    lam = (a + b) / 2
    return np.clip(v - lam, 0, 1)

def smooth_gradient(h, beta):
    """Compute gradient of smooth surrogate Φ_β(h)"""
    N = len(h)
    # Compute the overlaps F_k(h) = N/2 - sum_i h_i h_{i+k}
    F = full_overlap(h)
    F = F - np.max(F)  # for numerical stability
    exp_F = np.exp(beta * F)
    w = exp_F / np.sum(exp_F)  # Softmax weights

    # Initialize gradient
    g = np.ones(N)
    
    # Accumulate gradient from all shifts
    for k in range(-N+1, N):
        wk = w[k + N - 1]  # Index into the full overlap array
        if k >= 0:
            # Shift h forward by k, without padding, but we'll use INDEXING approach
            g[:N-k] -= wk * h[k:]
            g[k:] -= wk * h[:N-k]
        else:
            # k < 0, shift back
            d = -k
            g[d:] -= wk * h[:N-d]
            g[:N-d] -= wk * h[d:]
    
    return g

def projected_gradient_descent(h0, time_budget, seed):
    """Run projected gradient descent on the smooth surrogate"""
    np.random.seed(seed)
    start = time.time()
    
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of(h)
    
    N = len(h)
    total = N / 2.0
    beta = 2.0
    beta_max = 30.0
    beta_update_interval = 100
    gamma_beta = 1.5
    gamma_step = 0.9995
    step = 0.5
    max_iter = int(1e6)
    
    step_size_decrease_counter = 0
    
    for iteration in range(max_iter):
        if time.time() - start > time_budget:
            break
            
        # Compute all overlaps and weights once per iteration
        F = full_overlap(h)
        F = F - np.max(F)
        exp_F = np.exp(beta * F)
        w = exp_F / np.sum(exp_F)

        # Compute gradient of the smooth surrogate
        g = smooth_gradient(h, beta)
        
        # Tentative update
        v = h - step * g
        
        # Project onto the feasible set P
        h_new = project_onto_P(v, total)
        
        # Step size adjustment for stagnation
        if iteration % 10 == 0:
            # Check for stagnation using C5 change
            c5_new = c5_of(h_new)
            if abs(c5_new - best_c5) / (1 + best_c5) < 1e-8:
                step_size_decrease_counter += 1
                if step_size_decrease_counter > 5:
                    step = max(step * 0.5, 1e-4)
                    step_size_decrease_counter = 0
            else:
                best_c5 = c5_new
                best_h = h_new.copy()
                step_size_decrease_counter = 0
        
        h = h_new
        
        # Update β
        if iteration % beta_update_interval == 0 and beta < beta_max:
            beta = min(beta * gamma_beta, beta_max)
        
        # Update step size
        step *= gamma_step
    
    return best_h, best_c5

def targeted_refinement(h, time_budget):
    """Perform deterministic pairwise refinement to reduce the worst overlap"""
    start = time.time()
    h = h.copy()
    best_c5 = c5_of(h)
    alpha = 1.0
    no_improve_count = 0
    max_no_improve = 200
    
    while time.time() - start < time_budget:
        C = full_overlap(h)
        d = np.argmax(C) - (len(h) - 1)
        N = len(h)
        
        # Extract the overlapping slices
        if d >= 0:
            hi = h[:N - d]
            hj = h[d:]
        else:
            d = -d
            hi = h[-d:]
            hj = h[:N - d]
            
        # Compute the contribution of the pairs
        c = hi * (1 - hj)
        i, j = np.unravel_index(np.argmax(c), c.shape)
        
        # Compute feasible transfer amount
        delta = min(alpha * c[i, j], h[i], 1 - h[j])
        
        # Transfer mass
        h[i] -= delta
        h[j] += delta
        
        # Evaluate new c5
        new_c5 = c5_of(h)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            no_improve_count = 0
            alpha = 1.0
        else:
            # Revert the change
            h[i] += delta
            h[j] -= delta
            no_improve_count += 1
            # Halve alpha after stagnation
            if no_improve_count >= 5:
                alpha *= 0.5
                no_improve_count = 0
                if alpha < 1e-6:
                    break
    
    return h, best_c5

    best_h = None
    best_c5 = float('inf')
