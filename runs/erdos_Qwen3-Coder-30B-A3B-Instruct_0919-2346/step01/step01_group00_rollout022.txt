import numpy as np
import time
import random
from scipy.optimize import differential_evolution

def c5_of(h):
    """Compute C5 = max_k integral h(x)(1-h(x+k)) dx."""
    dx = 2.0 / len(h)
    # Compute correlation: h * (1 - h) with full mode
    c = np.correlate(h, 1 - h, mode="full")
    # Multiply by dx to get continuous integral
    c *= dx
    # Return maximum value among all shifts
    return np.max(c)

def evaluate_construction(h):
    """Evaluate a construction with the c5_of function."""
    return c5_of(h)

def propose_move(h, max_step):
    """Generate a new feasible h by adjusting two entries."""
    h = h.copy()
    N = len(h)
    i, j = random.sample(range(N), 2)
    
    # Determine feasible interval for delta
    delta_low = max(-h[i], h[j] - 1.0)
    delta_high = min(1.0 - h[i], h[j])
    
    if delta_low >= delta_high:
        return h  # No valid move for this pair
    
    # Sample delta within allowed range
    step = min(max_step, delta_high - delta_low)
    delta = random.uniform(-step, step)
    delta = max(delta_low, min(delta, delta_high))
    
    h[i] += delta
    h[j] -= delta
    
    return h

def sample_construction(N):
    """Sample a new construction."""
    h = np.random.uniform(0, 1, N)
    h = h * (N / 2) / np.sum(h)
    return h

def local_search(current_h, callback, max_iter=1000):
    """Simple local search around current solution."""
    h = current_h.copy()
    c = c5_of(h)
    
    for _ in range(max_iter):
        h_new = propose_move(h, 0.05)
        c_new = c5_of(h_new)
        if c_new < c:
            h = h_new
            c = c_new
            callback(h, c)
            
    return h, c

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    # Start time
    start_time = time.time()
    
    # Define problem parameters
    N = 140
    dx = 2.0 / N
    
    # Initial solution from imported values with slight adjustment
    h_initial = np.interp(np.linspace(0, 2, N), 
                          np.linspace(0, 2, len(initial_h_values)), 
                          initial_h_values)
    h_initial = h_initial * (N / 2) / np.sum(h_initial)
    
    # Try different starting points
    candidates = []
    candidates.append(h_initial)
    
    # Add a few random constructions
    for _ in range(3):
        h = sample_construction(N)
        candidates.append(h)
    
    # Add a piecewise constant construction
    h_piecewise = np.ones(N) * 0.5
    h_piecewise[:N//3] = 0.3
    h_piecewise[2*N//3:] = 0.7
    h_piecewise = h_piecewise * (N / 2) / np.sum(h_piecewise)
    candidates.append(h_piecewise)
    
    # Best solution tracking
    best_c5 = float('inf')
    best_h = None
    
    # Differential evolution optimization
    def objective(h):
        # Re-normalize
        h = h * (N / 2) / np.sum(h)
        return c5_of(h)
        
    # Ensure bounds are correct for optimization
    bounds = [(0, 1) for _ in range(N)]
    
    # Run differential evolution for a while
    de_budget = budget_s * 0.5
    try:
        result = differential_evolution(
            objective, 
            bounds,
            maxiter=500,
            popsize=15,
            seed=seed
        )
        h_de = result.x
        h_de = h_de * (N / 2) / np.sum(h_de)
        c5_de = c5_of(h_de)
        if c5_de < best_c5:
            best_c5 = c5_de
            best_h = h_de
    except:
        pass
        
    # Now local search around all candidates
    for i, h in enumerate(candidates):
        h_normalized = h * (N / 2) / np.sum(h)
        # Local search
        h_local, c5_local = local_search(h_normalized, lambda h, c: None, max_iter=500)
        if c5_local < best_c5:
            best_c5 = c5_local
            best_h = h_local

    # Final polishing with simulated annealing
    if best_h is not None:
        # Annealing parameters
        max_iter = 5000
        T = 0.1
        T_min = 1e-5
        alpha = 0.995
        
        cur = best_h.copy()
        c_cur = c5_of(cur)
        if c_cur < best_c5:
            best_c5 = c_cur
            best_h = cur
            
        for i in range(max_iter):
            if time.time() - start_time > budget_s:
                break
                
            h_new = propose_move(cur, T)
            c_new = c5_of(h_new)
            
            if c_new <= c_cur or random.random() < np.exp(-(c_new - c_cur) / T):
                cur = h_new
                c_cur = c_new
                if c_cur < best_c5:
                    best_c5 = c_cur
                    best_h = cur
            
            T = max(T_min, T * alpha)
    
    if best_h is None:
        # Fallback to initial construction
        best_h = h_initial
        best_c5 = c5_of(best_h)
        
    return (best_h, best_c5, N)
