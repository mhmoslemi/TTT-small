import time
from scipy.optimize import differential_evolution

def c5_of(h):
    """Compute the maximum overlap integral for the given step function h."""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    c = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(c) * dx
    return c5

def project_to_feasible(h):
    """Project h onto feasible region: [0,1] and sum(h) = n/2"""

    h = np.clip(h, 0, 1)
    # Renormalize
    dx = 2.0 / len(h)
    scale = (len(h) / 2) / np.sum(h)
    h = h * scale
    return h
