import numpy as np
import time
import random
from scipy.signal import correlate


def compute_c5(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    overlap = correlate(h, 1 - h, mode="full")
    c5 = np.max(overlap) * dx
    return c5


def project_h(h, dx):
    """Project h to feasible set: sum(h) = 1"""
    h_proj = h.copy()
    h_proj = np.clip(h_proj, 0, 1)
    sum_h = np.sum(h_proj)
    if sum_h > 0:
        h_proj = h_proj * (1.0 / sum_h) * (2.0 / dx)
    else:
        h_proj = np.full_like(h_proj, 1.0 / len(h_proj))
    return h_proj


def compute_grad_c5(h, beta=10.0):
    """Compute approximate gradient using softmax weighted correlation"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlations for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # Apply softmax on shifts
    shifts = np.arange(-n + 1, n)
    # Use softmax to weight the shifts
    weights = np.exp(beta * shifts)
    weights /= np.sum(weights)
    # Weighted gradient
    grad = np.zeros_like(h)
    # compute weighted gradient term by term
    for k in range(-n + 1, n):
        # this is approx the derivative of correlation with respect to h_i
        # we get contribution from locs where at shift k, both terms (h_i and 1-h_{i+k}) matter
        # it's convolution of h with vectors of 1 and -1-shifted
        if k > 0:
            left = 0
            right = -k
        else:
            left = -k
            right = None
        grad_slice = np.zeros_like(h)
        if right is None:
            grad_slice[left:] = 1
        else:
            grad_slice[left:right] = 1
        if k < 0:
            grad_slice = grad_slice[-k:]
        else:
            grad_slice = grad_slice[:-k] if k > 0 else grad_slice
        if len(grad_slice) < len(h):
            grad_slice = np.pad(grad_slice, (0, len(h) - len(grad_slice)), 'constant')
        else:
            grad_slice = grad_slice[:len(h)]
        grad += weights[k + n - 1] * grad_slice
    # adjust sign  
    grad = -grad
    return grad


def project_step(h, grad, step_size, dx):
    """Apply projected gradient step"""
    h_new = h - step_size * grad
    h_new = project_h(h_new, dx)
    return h_new


def phase1_optimize(h_start, time_budget_s, dx):
    """Smooth gradient ascent with softmax weighting"""
    start = time.time()
    h = h_start.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    # Schedule on step size and beta
    t0 = time.time()
    for epoch in range(100000):
        if time.time() - start > time_budget_s:
            break
        t = time.time() - t0
        beta = 50.0 * (0.999 ** epoch)
        step_size = 0.1 * (1.0 / (1.0 + 0.01 * epoch))
        grad = compute_grad_c5(h, beta)
        h_new = project_step(h, grad, step_size, dx)
        c5_new = compute_c5(h_new)
        if c5_new < c5:
            h = h_new
            c5 = c5_new
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
        elif random.random() < 0.05:
            h = h_new
            c5 = c5_new
    return best_h, best_c5


def phase2_local_refinement(h_start, time_budget_s, dx):
    """Targeted pairwise push/pull near dominant shift"""
    start = time.time()
    h = h_start.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5

    # Compute correlation once
    corr_vals = np.correlate(h, 1 - h, mode="full")
    max_shift = np.argmax(corr_vals) - len(h) + 1
    max_shift = min(max_shift, len(h) - 1)
    max_shift = max(max_shift, -len(h) + 1)

    # Find the indices most responsible for max shift
    shifts = np.arange(-len(h) + 1, len(h))
    # Look for indices that contribute significantly to this shift
    indices_affected = []
    for i in range(len(h)):
        j = i + max_shift
        if 0 <= j < len(h):
            indices_affected.append((i, j))

    # Random pairwise push/pull algorithm with selective targets
    max_tries = 10000
    for i in range(max_tries):
        if time.time() - start > time_budget_s:
            break
        # Pick a random pair, sub-sample from affected indices if desired
        if np.random.random() < 0.5:
            i1, i2 = np.random.randint(0, len(h), 2)
            i1, i2 = min(i1, i2), max(i1, i2)
        else:
            if indices_affected:
                i1, i2 = indices_affected[np.random.randint(len(indices_affected))]
            else:
                i1, i2 = np.random.randint(0, len(h), 2)
                i1, i2 = min(i1, i2), max(i1, i2)
        if i1 == i2:
            continue
        d = random.uniform(-0.1, 0.1)
        # Ensure feasibility
        dmax = min(0.99 - h[i1], h[i2])
        dmin = max(-h[i1], 0.99 - h[i2])
        if dmax <= dmin + 1e-8: continue
        d = np.clip(d, dmin, dmax)
        # Apply
        h_copy = h.copy()
        h_copy[i1] += d
        h_copy[i2] -= d
        h_copy = project_h(h_copy, dx)
        c5_new = compute_c5(h_copy)
        if c5_new < c5:
            h = h_copy
            c5 = c5_new
            if c5_new < best_c5:
                best_h = h_copy.copy()
                best_c5 = c5_new
        elif random.random() < 0.05:
            h = h_copy
            c5 = c5_new

    return best_h, best_c5
