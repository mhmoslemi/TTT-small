import numpy as np
import time

def c5_of(h):
    """Compute the maximum overlap integral for the given step function h."""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    c = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(c) * dx
    return c5

def evaluate(h):
    """Evaluate the objective function (C5) for a given step function."""
    return c5_of(h)

def propose_move(h, max_step):
    """Propose a new feasible step function by adjusting two entries."""
    N = len(h)
    while True:
        i, j = np.random.choice(N, size=2, replace=False)
        # Determine feasible step
        delta_low = max(-h[i], h[j] - 1)
        delta_high = min(1 - h[i], h[j])
        if delta_low >= delta_high:
            continue  # Try another pair
        # Sample step size
        step_size = np.random.uniform(max(-max_step, delta_low),
                                      min(max_step, delta_high))
        # Create new h
        h_new = h.copy()
        h_new[i] += step_size
        h_new[j] -= step_size
        return h_new

def perturb_noise(h, noise_level):
    """Add small random noise to the step function while maintaining constraints."""
    N = len(h)
    h_new = h.copy()
    # Add small random perturbations
    for i in range(N):
        delta = np.random.uniform(-noise_level, noise_level)
        h_new[i] += delta
        # Project back to [0,1]
        h_new[i] = max(0, min(1, h_new[i]))
    return h_new

                best = cur.copy()

        temperature = 0.1 * (1e-4 / 0.1) ** frac
        step_size = 0.5 * (1.0 - frac)  # Decrease step size over time
        
    return best, c_best
