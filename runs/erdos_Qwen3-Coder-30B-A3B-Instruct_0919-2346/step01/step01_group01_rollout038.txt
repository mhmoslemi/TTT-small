import numpy as np
import time
import random
from scipy.optimize import differential_evolution, minimize

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using full convolution
    overlap = np.correlate(h, 1 - h, mode='full')
    # The maximum value scaled by dx
    c5 = np.max(overlap) * dx
    return c5

def project_to_constraint(h):
    """Project h to satisfy the integral constraint exactly"""
    dx = 2.0 / len(h)
    current_sum = np.sum(h) * dx
    target_sum = 1.0
    if abs(current_sum - target_sum) > 1e-10:
        scale = target_sum / current_sum
        h = h * scale
    return h

def propose_move(h, max_step):
    """Propose a move that maintains sum constraint exactly"""
    h_new = h.copy()
    N = len(h)
    while True:
        i, j = random.sample(range(N), 2)
        # Compute feasible range for delta
        delta_low = max(-h_new[i], h_new[j] - 1.0)
        delta_high = min(1.0 - h_new[i], h_new[j])
        if delta_low >= delta_high:
            continue  # No valid move, try again
        break
    
    # Sample delta from feasible interval
    step = min(max_step, delta_high - delta_low)
    if step <= 0:
        delta = 0.0
    else:
        delta = random.uniform(max(-step, delta_low), min(step, delta_high))
        
    h_new[i] += delta
    h_new[j] -= delta
    return h_new

def optimize_with_differential_evolution(h_initial, bounds, maxiter=300):
    """Use differential evolution for global optimization"""
    def objective(h):
        # Project to constraint to satisfy integral constraint exactly
        h_projected = project_to_constraint(h)
        return c5_of(h_projected)
    
    # Use bounds to constrain h[i] to [0,1]
    result = differential_evolution(
        objective, 
        bounds, 
        maxiter=maxiter,
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.7,
        seed=42,
        tol=1e-6,
        disp=False
    )
    return result.x
