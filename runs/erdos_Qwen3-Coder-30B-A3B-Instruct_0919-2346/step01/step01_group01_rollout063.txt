import numpy as np
import time
import cvxpy as cp

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using full convolution
    overlap = np.correlate(h, 1 - h, mode='full')
    # The maximum value scaled by dx
    c5 = np.max(overlap) * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Configuration
    N = 140
    dx = 2.0 / N
    best_c5_bound = float('inf')
    best_h = None
    best_n_points = N

    # Initial upper bound from previous best
    initial_upper_bound = int(np.ceil(0.381 * N / 2))  # approx 27
    low = 0
    high = initial_upper_bound

    # To store best feasible solution found during binary search
    best_feasible_solution = None
    best_feasible_c = None

    # Solver settings
    solver = 'CBC'
    try:
        cp.installed_solvers()
    except:
        solver = 'GLPK_MI'

    # Timing
    solver_time_limit = 50  # seconds per solve (leaving a bigger margin)

    while low < high and (time.time() - start_time) < budget_s - 10:
        mid = (low + high) // 2
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time <= 20:
            break
        solver_time_limit = min(solver_time_limit, remaining_time - 5)

        # --- Build variable and constraint structure ---
        x = cp.Variable(N, boolean=True)
        y = {}  # y[ℓ][i] should be x[i]*x[i+ℓ]
        constraints = []

        # Sum constraint
        constraints.append(cp.sum(x) == N // 2)

        # For binary search feasibility check, we'll create vector instead of variable
        # but constraint is only with fixed RHS `mid`
        Cvar = mid  # RHS of inequality, will be fixed during feasibility check

        # Create y variables for each lag and overlapping index
        for ell in range(1, N):
            y[ell] = cp.Variable((N - ell), boolean=True)
            for i in range(N - ell):
                # McCormick constraints
                constraints.extend([
                    y[ell][i] <= x[i],
                    y[ell][i] <= x[i + ell],
                    y[ell][i] >= x[i] + x[i + ell] - 1
                ])

            # Overlap constraint
            H_ell = cp.sum(x[:N - ell])  # sum_i x[i] for i=0..N-ell-1
            O_ell = cp.sum(y[ell])
            constraints.append(H_ell - O_ell <= Cvar)

        # The propagation of x into the constraints happens in the actual solve

        # --- Define and solve the mixed integer program (as feasibility check) ---
        # Objective is fake (to avoid solver error); only constraints matter
        prob = cp.Problem(cp.Minimize(0), constraints)

        try:
            # Solve with specific settings
            prob.solve(solver=solver, verbose=False, time_limit=solver_time_limit)
            # Check if solution is feasible and optimal
            if prob.status in ["optimal", "feasible"]:
                # Extract solution
                x_sol = np.round(x.value).astype(int)
                # Validate feasibility
                if np.sum(x_sol) == N // 2:
                    # Compute C5 and check if better
                    h = x_sol
                    c5 = c5_of(h)
                    if c5 < best_c5_bound:
                        best_c5_bound = c5
                        best_h = h
                        best_feasible_solution = h.copy()
                        best_feasible_c = mid
                else:
                    # Not valid; try again
                    pass
                high = mid  # Found a bound below or equal to current mid; narrow down
            else:
                low = mid + 1   # Try higher
        except Exception as e:
            # Failed to solve; fallback
            low = mid + 1

    # Fallback if no MILP solution was found: run local search from best known.
    if best_h is None:
        # Fall back to previous sampling method as last resort
        # Created to ensure we return something reasonable
        try:
            h_initial = initial_h_values
            if len(h_initial) != N:
                x_orig = np.linspace(0, 2, len(h_initial))
                x_new = np.linspace(0, 2, N)
                h_initial = np.interp(x_new, x_orig, h_initial)
            h_initial = h_initial * (N / 2) / np.sum(h_initial)
        except:
            h_initial = np.ones(N) * 0.5
        best_h = h_initial

    # Use local search refinement if better than known or if no good MILP found
    if best_h is not None:
        # Now we have best_h, possibly from MILP; refine further via local search
        cur = best_h.copy()
        c_cur = c5_of(cur)

        # Simple hill climbing to polish
        last_improvement = time.time()
        max_refine_time = max(30, budget_s * 0.1)
        while (time.time() - start_time) < budget_s - 5 and (time.time() - last_improvement) < max_refine_time:
            # Evaluate current C5
            c_cur = c5_of(cur)
            if c_cur < best_c5_bound:
                best_c5_bound = c_cur
                best_h = cur.copy()
                last_improvement = time.time()
            # Try a random swap between 1 and 0
            ones = np.where(cur > 0.5)[0]
            zeros = np.where(cur < 0.5)[0]
            if len(ones) > 0 and len(zeros) > 0:
                i = np.random.choice(ones)
                j = np.random.choice(zeros)
                old_c = c_cur
                # Test a swap of h[i] and h[j]
                cur_new = cur.copy()
                cur_new[i] = 0 if cur_new[i] > 0.5 else 1
                cur_new[j] = 1 if cur_new[j] < 0.5 else 0
                new_c = c5_of(cur_new)
                if new_c < old_c:
                    cur = cur_new
                    c_cur = new_c
                    last_improvement = time.time()
            else:
                break
    else:
        # If even fallback didn't work
        h_uniform = np.ones(N) * 0.5
        best_c5_bound = c5_of(h_uniform)
        best_h = h_uniform

    return (best_h, best_c5_bound, best_n_points)
