import numpy as np
import time

def c5_of(h):
    """Compute the maximum overlap integral for the given step function h."""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    c = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(c) * dx
    return c5

def compute_gradient(h, shift):
    """Compute gradient of C5 for a given shift."""
    N = len(h)
    dx = 2.0 / N
    g = np.full(N, 1.0) * dx
    if shift >= 0:
        if shift < N:
            g[:N-shift] -= h[shift:] * dx
            g[shift:] -= h[:N-shift] * dx
    else:
        shift = -shift
        if shift < N:
            g[shift:] -= h[:N-shift] * dx
            g[:N-shift] -= h[shift:] * dx
    return g

def project_onto_bounds_and_sum(v, target):
    """Project vector onto box [0,1] and hyperplane sum=target."""
    # Binary search for lambda
    lower = -np.max(v)
    upper = 1 - np.min(v)
    for _ in range(100):
        mid = (lower + upper) / 2.0
        w = np.clip(v + mid, 0, 1)
        s = np.sum(w)
        if s < target:
            lower = mid
        else:
            upper = mid
    w = np.clip(v + (lower + upper) / 2.0, 0, 1)
    diff = target - np.sum(w)
    if abs(diff) > 1e-12:
        free_idx = (w > 0) & (w < 1)
        if np.any(free_idx):
            w[free_idx] += diff / np.sum(free_idx)
            w = np.clip(w, 0, 1)
        else:
            w += diff / len(w)
            w = np.clip(w, 0, 1)
    return w

def run(seed=42, budget_s=600):
    np.random.seed(seed)

        h0 = h0 * (N/2) / np.sum(h0)  # Renormalize exactly
    else:
        h0 = np.ones(N) * 0.5  # fallback
        
    # Two-peak seed 
    h1 = np.ones(N)
    x_vals = np.linspace(0, 2, N)
    a = 0.3
    u = 0.9
    v = 0.1
    mask = (x_vals < a) | (x_vals >= 2 - a)
    h1[mask] = u
    h1[~mask] = v
    h1 = h1 * (N/2) / np.sum(h1)  # Renormalize

    # Get best from multiple initializations
    best_h = h0.copy()
    best_c5 = c5_of(best_h)
    
    h0_c5 = c5_of(h0)
    if h0_c5 < best_c5:
        best_h = h0.copy()
        best_c5 = h0_c5
    
    h1_c5 = c5_of(h1)
    if h1_c5 < best_c5:
        best_h = h1.copy()
        best_c5 = h1_c5
        
    # Binary seeds
    for _ in range(5):
        h_bin = np.zeros(N)
        idx = np.random.choice(N, N//2, replace=False)
        h_bin[idx] = 1.0
        c5_bin = c5_of(h_bin)
        if c5_bin < best_c5:
            best_h = h_bin.copy()
            best_c5 = c5_bin

    # Gradient descent with restarts
    start = time.time()
    iteration = 0
    
    while time.time() - start < 0.9 * budget_s:
        # Restart with best solution so far
        cur_h = best_h.copy()
        cur_score = best_c5 
        
        # Inner gradient descent loop
        inner_start = time.time()
        iter_count = 0
        max_inner_iter = 1000
        
        while time.time() - start < 0.9 * budget_s and iter_count < max_inner_iter:

            grad = compute_gradient(cur_h, s)

            elapsed = time.time() - start

            v = cur_h - step * grad

    cur_score = best_c5

            cur_h[i] += delta
            cur_h[j] -= delta
            # Project
            cur_h = project_onto_bounds_and_sum(cur_h, N/2)
            new_score = c5_of(cur_h)
            if new_score < cur_score:
                cur_score = new_score
                best_h = cur_h.copy()
                best_c5 = new_score

    return (best_h, best_c5, N)
