import numpy as np

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    overlap = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(overlap) * dx
    return c5


def propose_move(h, max_step, idxes=None):
    """Propose a new feasible h vector by adjusting two elements"""
    h_new = h.copy()
    N = len(h)
    
    # Try to find a valid move
    max_tries = 100
    for _ in range(max_tries):
        if idxes is None:
            i = random.randint(0, N - 1)
            j = random.randint(0, N - 1)
            if i == j:
                continue
        else:
            i, j = random.sample(idxes, 2)
            
        # Determine feasible interval for δ
        delta_low = max(-h_new[i], h_new[j] - 1)
        delta_high = min(1 - h_new[i], h_new[j])
        
        if delta_low >= delta_high:
            continue
            
        # Sample δ from uniform distribution over feasible range
        step = min(max_step, delta_high - delta_low)
        if step <= 0:
            continue
        delta = random.uniform(delta_low, delta_high)
        
        # Apply move
        h_new[i] += delta
        h_new[j] -= delta
        
        return h_new
    
    # If no valid move found, return original
    return h.copy()


def smooth_h(h, smooth_factor=0.1):
    """Apply smoothing to h to make it more continuous"""
    h_smooth = h.copy()
    for _ in range(5):
        # Simple averaging filter
        h_smooth = np.convolve(h_smooth, np.ones(5)/5, mode='same')
    # Clip and renormalize
    h_smooth = np.clip(h_smooth, 0, 1)
    h_smooth = h_smooth * (len(h_smooth) / 2) / np.sum(h_smooth)
    return h_smooth

def optimize_seed(h_seed, time_budget_s, N):
    """Use scipy minimize with bounds for further local optimization"""
    start = time.time()
    
    # Define bounds (0 <= h[i] <= 1)
    bounds = [(0, 1) for _ in range(N)]
    
    def objective(h_vec):
        # Ensure component-wise constraints
        h_vec = np.clip(h_vec, 0, 1)
        h = h_vec.copy()
        return c5_of(h)
    
    # Add constraint to enforce sum(h) * dx = 1
    def constraint_sum(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        return np.sum(h_vec) * (2.0 / N) - 1
        
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Start optimization with the provided seed
    result = minimize(objective, h_seed.copy(), method='SLSQP', bounds=bounds, constraints=cons, 
                      options={'maxiter': 1000}, callback=lambda x: None)
    
    # Post processing and returning results
    if result.success:
        h_opt = np.clip(result.x, 0, 1)
        h_opt = h_opt * (N / 2) / np.sum(h_opt)  # Re-normalize
        c5_opt = c5_of(h_opt)
    else:
        # Use another method if optimization fails
        h_opt = h_seed.copy()
        c5_opt = c5_of(h_opt)
        
    return h_opt, c5_opt


def anneal_refine(h0, time_budget_s, N):
    """Refined simulated annealing with different cooling schedule and strategy"""
    start = time.time()
    
    # Parameters
    T_start = 0.03
    T_end = 1e-6
    step_start = 0.15
    
    # Set up the current state
    cur = h0.copy()
    c_cur = c5_of(cur)
    
    best = cur.copy()
    c_best = c_cur
    
    # Main loop
    while time.time() - start < time_budget_s:
        t = time.time() - start
        elapsed_fraction = t / time_budget_s
        
        # Cooling schedule
        T = T_start * (T_end / T_start) ** (elapsed_fraction**1.5)
        
        # Step size decays slightly less aggressively
        step = step_start * (1 - elapsed_fraction**0.7)
        
        # Make a proposal
        h_proposed = propose_move(cur, step)
        c_proposed = c5_of(h_proposed)
        
        # Accept the proposal if it's a better solution
        if c_proposed <= c_cur:
            cur = h_proposed
            c_cur = c_proposed
            if c_proposed < c_best:
                best = h_proposed.copy()
                c_best = c_proposed
        else:
            # Metropolis acceptance rule
            if random.random() < np.exp(-(c_proposed - c_cur) / max(T, 1e-10)):
                cur = h_proposed
                c_cur = c_proposed
                
    return best, c_best
