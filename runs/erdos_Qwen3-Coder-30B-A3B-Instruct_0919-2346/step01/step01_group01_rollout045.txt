import numpy as np
import time
import random
from scipy.optimize import dual_annealing

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using full convolution
    overlap = np.correlate(h, 1 - h, mode='full')
    # The maximum value scaled by dx
    c5 = np.max(overlap) * dx
    return c5

def project_to_constraint(h, target_sum=1.0):
    """Project h to satisfy sum constraint with exact target"""
    current_sum = np.sum(h)
    if current_sum == 0:
        return h
    scale = target_sum / current_sum
    return h * scale

def initialize_h(n_points):
    """Initialize h with multiple strategies"""

    # Try to use initial_h_values if available

            x_new = np.linspace(0, 2, n_points)
            h_initial = np.interp(x_new, x_orig, h_initial)
        h_initial = project_to_constraint(h_initial, 1.0)
    except:
        h_initial = h_uniform.copy()
    
    # Two-peak initialization
    h_two_peak = np.full(n_points, 0.1)
    a = 0.3  # width of peaks
    u = 0.9  # peak height
    v = 0.1  # valley height
    peak_indices = np.where((np.linspace(0, 2, n_points) < a) | (np.linspace(0, 2, n_points) > 2-a))[0]
    h_two_peak[peak_indices] = u
    valley_idx = np.setdiff1d(range(n_points), peak_indices)
    if len(valley_idx) > 0:
        h_two_peak[valley_idx] = v
    h_two_peak = project_to_constraint(h_two_peak, 1.0)
    
    # Gaussian-like initialization
    x = np.linspace(0, 2, n_points)
    h_gaussian = np.exp(-((x - 1)**2) / 0.2)
    h_gaussian = project_to_constraint(h_gaussian, 1.0)
    
    return [h_uniform, h_initial, h_two_peak, h_gaussian]

def bound_constraints(n_points):
    """Return bounds for each parameter"""
    return [(-1e-6, 1.0 + 1e-6)] * n_points

def adjust_solution(h):
    """Adjust solution to ensure all constraints are met"""
    # Ensure all values in [0,1]
    h = np.clip(h, 0, 1)
    # Project onto constraint sum = 1.0
    h = project_to_constraint(h, 1.0)
    return h

def run(seed=42, budget_s=600, **kwargs):

    np.random.seed(seed)
    
    # Start time tracking
    start_time = time.time()

    initial_strategies = initialize_h(n_points)

    best_c5 = float('inf')

    for i, h0 in enumerate(initial_strategies):

        current_h = best_h.copy()

            # Ensure sum is exactly 1
            h_candidate = project_to_constraint(h_candidate, 1.0)
            
            c5_candidate = c5_of(h_candidate)

                current_c5 = c5_candidate
                
        best_h = current_h
        best_c5 = current_c5
    
    return (best_h, best_c5, n_points)
