import numpy as np
import time
import random
from scipy.optimize import differential_evolution, minimize


def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    overlap = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(overlap) * dx
    return c5


def propose_move(h, max_step):
    """Propose a new feasible h vector by adjusting two elements"""
    h_new = h.copy()
    N = len(h)
    
    # Try to find a valid move
    max_tries = 100
    for _ in range(max_tries):
        i = random.randint(0, N - 1)
        j = random.randint(0, N - 1)
        if i == j:
            continue
            
        # Determine feasible interval for δ
        delta_low = max(-h_new[i], h_new[j] - 1)
        delta_high = min(1 - h_new[i], h_new[j])
        
        if delta_low >= delta_high:
            continue
            
        # Sample δ from uniform distribution over feasible range
        step = min(max_step, delta_high - delta_low)
        if step <= 0:
            continue
        delta = random.uniform(delta_low, delta_high)
        
        # Apply move
        h_new[i] += delta
        h_new[j] -= delta
        
        return h_new
    
    # If no valid move found, return original
    return h.copy()


def optimize_with_de(h_initial, maxiter=1000):
    """Use differential evolution for global optimization"""
    def objective(h):
        return c5_of(h)
    
    bounds = [(0, 1) for _ in range(len(h_initial))]
    
    result = differential_evolution(
        objective, 
        bounds, 
        maxiter=maxiter,
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.7,
        seed=42,
        disp=False
    )
    
    return result.x, result.fun


def refine_solution(h_initial, maxiter=2000):
    """Use scipy minimize with L-BFGS-B for local refinement"""
    def objective(h):
        return c5_of(h)
    
    def constraint(h):
        return np.sum(h) - len(h) / 2  # sum(h) * dx = 1, so sum(h) = n_points/2
    
    # Create bounds and initial guess
    bounds = [(0, 1) for _ in range(len(h_initial))]
    constraints = {'type': 'eq', 'fun': constraint}
    
    # Use L-BFGS-B or trust-constr for local refinement
    result = minimize(
        objective,
        h_initial,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': maxiter},
        tol=1e-8
    )
    
    return result.x, result.fun
