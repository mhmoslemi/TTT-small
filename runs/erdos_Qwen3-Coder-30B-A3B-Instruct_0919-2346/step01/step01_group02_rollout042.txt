import numpy as np
import time

def c5_of(h):
    """Compute the maximum overlap integral for the given step function h."""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    c = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(c) * dx
    return c5

def worst_shift(h):
    """Determine the shift that yields the maximum overlap."""
    corr = np.correlate(h, 1-h, mode='full')
    idx = np.argmax(corr)
    return idx - (len(h)-1)               # signed shift

def subgradient(h, s):
    """Compute the gradient of C5 w.r.t. h for the worst shift s."""
    N = len(h)
    dx = 2.0 / N
    m = abs(s)
    g = np.full(N, dx)
    if m < N:
        g[:N-m] -= dx * h[m:]
        g[m:]   -= dx * h[:N-m]
    return g

def project(v, target):
    """Project v onto the feasible set: 0 <= h <= 1 and sum(h) = target."""
    lo = -np.max(v)
    hi = 1.0 - np.min(v)
    for _ in range(50):
        lam = (lo + hi) / 2.0
        w = np.clip(v + lam, 0.0, 1.0)
        if w.sum() < target:
            lo = lam
        else:
            hi = lam
    w = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    # Ensure target sum exactly
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        free = (w > 0) & (w < 1)
        if free.any():
            w[free] += diff / free.sum()
            w = np.clip(w, 0.0, 1.0)
    return w

def seed_initials(N):
    """Generate various starting points."""
    initials = []
    # Pre-defined if available
    if 'initial_h_values' in globals():
        h0 = np.interp(np.linspace(0, 2, N),
                       np.linspace(0, 2, len(initial_h_values)),
                       initial_h_values)
        h0 = h0 * (N/2) / np.sum(h0)
        initials.append(h0)
    
    # Two-peak template with varying parameters
    a_range = np.linspace(0.15, 0.35, 10)
    for a in a_range:
        h = np.ones(N)
        x_vals = np.linspace(0, 2, N)
        u = 0.9
        v = 0.1
        mask = (x_vals < a) | (x_vals >= 2 - a)
        h[mask] = u
        h[~mask] = v
        h = h * (N/2) / np.sum(h)  # Renormalize
        initials.append(h)
    
    # Uniform 0.5 vector
    initials.append(np.ones(N) * 0.5)
    
    # Random binary vectors (exactly sum to N/2)
    for _ in range(5):
        indices = np.random.permutation(N)[:N//2]
        h = np.zeros(N)
        h[indices] = 1.
        initials.append(h)
    
    # Random vectors (projected)
    for _ in range(5):
        h = np.random.rand(N)
        h = project(h, N/2)
        initials.append(h)
    
    # Clean the list of initializations
    clean_initials = []
    for h in initials:
        # Ensure sum is exact:
        h = project(h, N/2)
        clean_initials.append(h)
    return clean_initials

def optimize_spd(initial_h, t_end, best_h, best_c5, eta0=30.0, gamma=0.9995):
    """Subgradient projected descent."""
    h = initial_h.copy()
    t_start = time.time()
    t = 0
    while time.time() < t_end:
        # Compute C5 and worst shift
        c5 = c5_of(h)
        if c5 < best_c5:
            best_h[:] = h
            best_c5 = c5
        s = worst_shift(h)
        g = subgradient(h, s)
        # Adaptive step-size
        eta = eta0 * (gamma ** t)
        # Cap step size to prevent oscillation
        eta = min(eta, 50.0 * (2.0/N))
        # Update h
        v = h - eta * g
        h = project(v, N/2)
        # Occasionally jitter to escape plateaux
        if t % 30 == 0:
            j = np.random.rand(N)
            delta = 0.02 * (2 * j - 1)
            v = h + delta
            h = project(v, N/2)
        t += 1
    return best_h, best_c5

def optimize_anneal(initial_h, t_end, best_h, best_c5, T0=0.05, Tf=1e-6, Δ0=0.2):
    """Simulated annealing."""
    def propose_move(h, Δ_max):
        N = len(h)
        while True:
            i, j = np.random.choice(N, size=2, replace=False)
            delta_low = max(-h[i], h[j] - 1)
            delta_high = min(1 - h[i], h[j])
            if delta_low >= delta_high:
                continue
            step_size = np.random.uniform(max(-Δ_max, delta_low),
                                          min(Δ_max, delta_high))
            h_new = h.copy()
            h_new[i] += step_size
            h_new[j] -= step_size
            return h_new

    h = initial_h.copy()
    c_cur = c5_of(h)
    if c_cur < best_c5:
        best_h[:] = h
        best_c5 = c_cur
    T = T0
    while time.time() < t_end:
        Δmax = Δ0 * (1.0 - (time.time() - t_start) / (t_end - t_start))
        T = T0 * (Tf / T0) ** ((time.time() - t_start) / (t_end - t_start))
        h_new = propose_move(h, Δmax)
        c_new = c5_of(h_new)
        if c_new <= c_cur or np.random.rand() < np.exp(-(c_new - c_cur) / T):
            h = h_new
            c_cur = c_new
            if c_cur < best_c5:
                best_h[:] = h
                best_c5 = c_cur
    return best_h, best_c5

def hill_climb(initial_h, t_end, best_h, best_c5, Δmax=0.02):
    """Fine-tune with local deterministic moves."""
    h = initial_h.copy()
    while time.time() < t_end:
        # Also accept only strictly better moves
        h_new = propose_move(h, Δmax)
        c_new = c5_of(h_new)
        if c_new < best_c5:
            h = h_new
            best_h[:] = h
            best_c5 = c_new
    return best_h, best_c5

def propose_move(h, max_step):
    """Propose a new feasible step function by adjusting two entries."""
    N = len(h)
    while True:
        i, j = np.random.choice(N, size=2, replace=False)
        # Determine feasible step
        delta_low = max(-h[i], h[j] - 1)
        delta_high = min(1 - h[i], h[j])
        if delta_low >= delta_high:
            continue  # Try another pair
        # Sample step size
        step_size = np.random.uniform(max(-max_step, delta_low),
                                      min(max_step, delta_high))
        # Create new h
        h_new = h.copy()
        h_new[i] += step_size
        h_new[j] -= step_size
        return h_new

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    N = 140
    t_start = time.time()
    best_h = None
    best_c5 = float('inf')
    # Phase 0: Seeding
    initials = seed_initials(N)
    for h in initials:
        c5 = c5_of(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
    # Allocation of time
    # Phase 1: SPD (55%)
    t_phase1_end = t_start + 0.55 * budget_s
    best_h, best_c5 = optimize_spd(best_h, t_phase1_end, best_h, best_c5)
    # Phase 2: Annealing (30%)
    t_phase2_end = t_start + 0.85 * budget_s
    best_h, best_c5 = optimize_anneal(best_h, t_phase2_end, best_h, best_c5)
    # Phase 3: Final hill climb (10%)
    t_phase3_end = t_start + 0.95 * budget_s
    best_h, best_c5 = hill_climb(best_h, t_phase3_end, best_h, best_c5)
    # Final check for best value
    final_c5 = c5_of(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return (best_h, best_c5, N)
