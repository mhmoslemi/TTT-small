import numpy as np
import time
import random
from scipy.optimize import differential_evolution

def c5_of(h):
    """Compute C5 = max_k integral h(x)(1-h(x+k)) dx."""
    dx = 2.0 / len(h)
    # Compute correlation: h * (1 - h) with full mode
    c = np.correlate(h, 1 - h, mode="full")
    # Multiply by dx to get continuous integral
    c *= dx
    # Return maximum value among all shifts
    return np.max(c)

def evaluate_constraint(h):
    """Check if the constraint sum(h) * dx == 1 is satisfied."""
    dx = 2.0 / len(h)
    return abs(np.sum(h) * dx - 1.0)

def constraint_func(h):
    """Constraint function for optimization."""
    dx = 2.0 / len(h)
    return np.sum(h) * dx - 1.0

def propose_move(h, max_step):
    """Generate a new feasible h by adjusting two entries."""
    h = h.copy()
    N = len(h)
    i, j = random.sample(range(N), 2)
    
    # Determine feasible interval for delta
    delta_low = max(-h[i], h[j] - 1.0)
    delta_high = min(1.0 - h[i], h[j])
    
    if delta_low >= delta_high:
        return h  # No valid move for this pair
    
    # Sample delta within allowed range
    step = min(max_step, delta_high - delta_low)
    delta = random.uniform(-step, step)
    delta = max(delta_low, min(delta, delta_high))
    
    h[i] += delta
    h[j] -= delta
    
    return h

def local_search(initial_h, time_budget):
    """Local search using simulated annealing."""
    start = time.time()
    cur = initial_h.copy()
    best = initial_h.copy()
    c_cur = c5_of(cur)
    c_best = c_cur
    
    # Temperature schedule
    T_start = 0.05
    T_end = 1e-6
    step_start = 0.2
    
    while time.time() - start < time_budget:
        t = time.time() - start
        T = T_start * (T_end / T_start) ** (t / time_budget)
        step = step_start * (1 - t / time_budget)
        
        h_new = propose_move(cur, step)
        c_new = c5_of(h_new)
        
        if c_new <= c_cur or random.random() < np.exp(-(c_new - c_cur) / T):
            cur = h_new
            c_cur = c_new
            if c_new < c_best:
                best = h_new
                c_best = c_new
    
    return best, c_best

def adaptive_search(h_start, time_budget):
    """Run multiple search strategies adaptively."""
    start = time.time()
    best_h = h_start.copy()
    best_c5 = c5_of(best_h)
    
    # Strategy 1: Simulated Annealing
    sa_budget = time_budget * 0.6
    if sa_budget > 0:
        h_sa, c5_sa = local_search(h_start, sa_budget)
        if c5_sa < best_c5:
            best_h = h_sa
            best_c5 = c5_sa
    
    # Strategy 2: Random local search with varied steps
    if time.time() - start < time_budget:
        ls_budget = time_budget * 0.2
        cur_h = best_h.copy()
        cur_c5 = best_c5
        start_ls = time.time()
        
        while time.time() - start_ls < ls_budget and time.time() - start < time_budget:
            # Vary step size dynamically
            step_size = 0.2 * (1 - (time.time() - start_ls) / ls_budget)
            h_new = propose_move(cur_h, step_size)
            c5_new = c5_of(h_new)
            if c5_new < cur_c5:
                cur_h = h_new
                cur_c5 = c5_new
                if c5_new < best_c5:
                    best_h = cur_h.copy()
                    best_c5 = c5_new
    
    # Strategy 3: Pure random with constraint checking
    if time.time() - start < time_budget:
        random_budget = time_budget * 0.2
        start_random = time.time()
        
        while time.time() - start_random < random_budget and time.time() - start < time_budget:
            # Generate a random feasible point
            N = len(best_h)
            h_random = np.random.rand(N)
            # Adjust to satisfy constraint sum(h) * dx = 1
            dx = 2.0 / N
            h_random = h_random * (N / 2) / np.sum(h_random)
            # Clamp values to [0, 1]
            h_random = np.clip(h_random, 0, 1)
            c5_random = c5_of(h_random)
            if c5_random < best_c5:
                best_h = h_random
                best_c5 = c5_random
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    N = 140
    dx = 2.0 / N
    
    # Build seed using initial_h_values if available
    if 'initial_h_values' in globals():
        h_initial = np.interp(np.linspace(0, 2, N), 
                              np.linspace(0, 2, len(initial_h_values)), 
                              initial_h_values)
        # Renormalize so sum == N/2
        h_initial = h_initial * (N / 2) / np.sum(h_initial)
    else:
        # Fallback to uniform if initial_h_values is not available
        h_initial = np.ones(N) * 0.5
    
    # Also try a simple ramp as a secondary start
    h_ramp = np.linspace(0, 1, N)
    h_ramp = h_ramp * (N / 2) / np.sum(h_ramp)
    
    # Initialize with better-known configuration
    seeds = [h_initial, h_ramp]
    
    best_h = None
    best_c5 = float('inf')
    
    # Try different seeds with adaptive search
    for h_seed in seeds:
        # If we're running low on time, use only one seed to make sure we can
        # finish the search in time
        remaining_time = budget_s - (time.time() - time.time() if 'start_time' in locals() else 0)
        if remaining_time < 10:  # Not enough time for full exploration
            break
        
        h_result, c5_result = adaptive_search(h_seed, budget_s * 0.5)
        if c5_result < best_c5:
            best_c5 = c5_result
            best_h = h_result
    
    # Try a more refined approach with fewer points
    if best_c5 > 0.380:
        # Try reducing points but not to less than 100
        N_fine = max(100, N - 20)
        dx_fine = 2.0 / N_fine
        
        h_fine_initial = np.interp(np.linspace(0, 2, N_fine), 
                                   np.linspace(0, 2, N), 
                                   best_h)
        h_fine_initial = h_fine_initial * (N_fine / 2) / np.sum(h_fine_initial)
        
        h_fine, c5_fine = adaptive_search(h_fine_initial, budget_s * 0.5)
        if c5_fine < best_c5:
            best_c5 = c5_fine
            best_h = h_fine
            N = N_fine
    
    return (best_h, best_c5, N)
