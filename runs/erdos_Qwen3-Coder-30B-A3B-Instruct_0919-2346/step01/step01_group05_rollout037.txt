import numpy as np
from scipy.optimize import minimize
from math import sqrt
import time

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value scaled by dx
    max_overlap = np.max(correlation) * dx
    return max_overlap

def project_to_simplex_box(h_raw, target_sum=1.0, dx=2.0/240):
    """Project h_raw onto the intersection of [0,1]^n and the hyperplane sum(h)=target_sum"""
    n = len(h_raw)
    # Ensure h_raw is within [0,1]
    h_clipped = np.clip(h_raw, 0, 1)
    current_sum = np.sum(h_clipped) * dx
    if abs(current_sum - target_sum) < 1e-10:
        return h_clipped
    # Project to the hyperplane using water-filling
    h = h_clipped.copy()
    # Bisection on tau to find the right projection
    def sum_x_minus_target(tau):
        x = np.clip(h - tau, 0, 1)
        return np.sum(x) * dx - target_sum
    
    # Find a good range for tau
    low, high = -1.0, 1.0
    while sum_x_minus_target(low) > 0:
        low -= 1.0
    while sum_x_minus_target(high) < 0:
        high += 1.0
    # Binary search for tau
    for _ in range(50):
        tau = (low + high) / 2
        diff = sum_x_minus_target(tau)
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            low = tau
        else:
            high = tau
    # Final tau
    tau = (low + high) / 2
    h_final = np.clip(h - tau, 0, 1)
    return h_final

def build_h_from_coeffs(a_coeffs, n):
    """Construct h from the truncated Fourier cosine coefficients a_coeffs"""
    # a_coeffs[0], ..., a_{M-1} of size M, where M = len(a_coeffs)
    h = 0.5 * np.ones(n)
    # Precompute frequency factors
    freqs = np.arange(n)
    for m, a in enumerate(a_coeffs):
        if m == 0:
            continue
        cos_freq = np.cos(2 * np.pi * m * freqs / n)
        h += a * cos_freq
    return h

def compute_overlap_and_gradient(h, a_coeffs, n, M, gamma=1.0):
    """Compute maximum overlap and gradient with respect to a_coeffs"""
    dx = 2.0 / n
    # Compute all overlaps using FFT-based correlation
    c = np.correlate(h, 1-h, mode="full")
    # Normalize
    overlaps = c * dx  # [ -(n-1) to n-1 ]
    # Overlap index offset since 0 corresponds to index n-1 in c
    # For shift s = 0 to 2*(n-1), we have overlap_i = overlaps[s - (n-1)]
    # offset = n - 1, so overlaps[s - offset] gives proper index
    max_idx = np.argmax(overlaps)
    max_overlap = overlaps[max_idx] 
    
    # Softmax approximation
    if gamma > 0:
        exp_vals = np.exp(gamma * (overlaps - max_overlap))
        weights = exp_vals / np.sum(exp_vals)

    # Compute gradient of overlaps w.r.t. a_coeffs
    # Let h_i = 0.5 + sum_{m=1}^M a_m cos(2π m i / n)
    # f_s = dx * sum_{i=0}^{n-1} h_i (1 - h_{i+s}) 
    # gradient is: d/d a_m f_s = dx * sum_{i=0}^{n-1} [cos(2π m i / n) (1-h_{i+s}) - h_i cos(2π m (i+s) / n)]
    
    # We sum all overlap gradients with softmax weights
    grad = np.zeros(M)
    freqs = np.arange(n)  # 0 to n-1
    # Precompute cosine columns 
    cosines_m = np.cos(2 * np.pi * freqs[np.newaxis, :] / n)  # [1, n] for a single m
    cosines_matrix = np.cos(2 * np.pi * freqs[:, np.newaxis] / n)  # [n, n]
    
    # Now compute for all m from 1 to M
    # For a given a_m: gradient = sum_{i=0}^{n-1} cos(2π m i / n) [1 - h_{i+s} - h_i]

    all_cos_m = cosines_m ** np.arange(1, M+1)  # [M, n]
    
    # g_{s,m} = sum_{i=0}^{n-1} cos(2π m i / n) [1 - h_{i+s} - h_i]
    # We'll compute a bunch of these efficiently
    
    # At this point we decide to avoid the complex matrix computation and instead use one by one...
    # For simplicity, correctness trumps performance in this routine:
    
    for m in range(1, M+1):
        # Use the fact that derivative of h_i with respect to a_m
        # is cos(2π m i / n)
        # But we're taking derivative of the kernel with respect to a_m, not the variable
        # d/d a_m [sum h_i (1 - h_{i+s})] = sum [cos(2π m i / n) (1 - h_{i+s}) - h_i cos(2π m (i+s) / n)]
        # This is convolution in some sense 
        cos_m_i = np.cos(2 * np.pi * m * freqs / n)  # shape (n)
        # And the convolution term: d/d a_m f_s = dx * [cos_m_i * (1 - h_{i+s}) - h_i * cos_m_{i+s}]
        #   where h_{i+s} can be rolled and indexed...
        if gamma > 0:
            for s in range(-n+1, n):
                if s <= 0:
                    # s is negative or zero: h_{i+s} = h_i when i+s >= 0, else 0 (out of bounds)
                    # For term 1: cos_m_i * (1-h_{i+s})  
                    # For term 2: -h_i * cos_m_{i+s}  
                    # Let's slide s:
                    i_left = freqs  # i = 0 .. n-1; 
                    i_right = freqs + s # i+s = s to s+n-1
                    # overlapping part is [max(0, -s), min(n-1, n-1+s)]
                    sz = np.min([n-1, n-1+s]) - np.max([0, -s]) + 1  # -> 0 to n-1 
                    if sz > 0:
                        i_left_start = np.max([0, -s]) # -s+1 to n-1
                        i_right_start = np.max([0, s])
                        i_left_idx = np.arange(i_left_start, i_left_start+sz)
                        i_right_idx = np.arange(i_right_start, i_right_start+sz)
                        if s == 0:
                            term1 = np.sum(cos_m_i[i_left_idx] * (1 - h[i_left_idx]))
                            term2 = -np.sum(h[i_left_idx] * cos_m_i[i_left_idx])
                            der = dx * (term1 + term2)
                        else:
                            term1 = np.sum(cos_m_i[i_left_idx] * (1 - h[i_right_idx]))
                            term2 = -np.sum(h[i_left_idx] * cos_m_i[i_right_idx])
                            der = dx * (term1 + term2)
                        grad[m-1] += weights[s + n - 1] * der
        else:
            # Without weights
            if s <= 0:
                i_left = freqs
                i_right = freqs + s
                sz = np.min([n-1, n-1+s]) - np.max([0, -s]) + 1
                if sz > 0:
                    i_left_start = np.max([0, -s])
                    i_right_start = np.max([0, s])
                    i_left_idx = np.arange(i_left_start, i_left_start+sz)
                    i_right_idx = np.arange(i_right_start, i_right_start+sz)
                    if s == 0:
                        term1 = np.sum(cos_m_i[i_left_idx] * (1 - h[i_left_idx]))
                        term2 = -np.sum(h[i_left_idx] * cos_m_i[i_left_idx])
                        der = dx * (term1 + term2)
                    else:
                        term1 = np.sum(cos_m_i[i_left_idx] * (1 - h[i_right_idx]))
                        term2 = -np.sum(h[i_left_idx] * cos_m_i[i_right_idx])
                        der = dx * (term1 + term2)
                    grad[m-1] += der
    
    return max_overlap, grad

def softmax_penalty(h, lambda_pen=1e4):
    """Soft penalty to keep h in [0,1]"""
    penalty = 0
    for i in range(len(h)):
        h_i = h[i]
        if h_i > 1:
            penalty += lambda_pen * (h_i - 1)**2
        if h_i < 0:
            penalty += lambda_pen * h_i**2
    return penalty

def spectral_optim(
    initial_a, 
    n, 
    M, 
    max_iter=1000,
    gamma_init=0.5,
    gamma_anneal=1.5,
    lambda_pen=1e4,
    bounds=(-0.6, 0.6),
    save_iter=False
):
    """Perform spectral optimization using the low-dimensional Fourier coeffs"""
    a_current = np.array(initial_a).copy()
    
    last_gamma = gamma_init
    factor = gamma_anneal
    
    # In this simple version, we hard-code a single stage:
    # Stage 1, M=8, gamma=0.5 -> 40
    # Stage 2, extending to 15
    # Stage 3, extending to 24
    
    # For our optimisation, let these be fixed:
    stage_1_iter = 1500  # 1500 iterations with M=8
    stage_2_iter = 2000  # 2000 iterations extending to M=15
    stage_3_iter = 2500  # 2500 iterations extending to M=24

    # Otherwise optimize towards full M ensuring that steps are generally quick enough
    
    # Since we use L-BFGS-B, we'll run the optimization in one go but track gamma
    # Iterate manually:
    def obj_func(a_vec):
        h = build_h_from_coeffs(a_vec, n)
        # Add penalty to enforce h in [0,1]
        pen = softmax_penalty(h, lambda_pen)
        
        overlap_derivative = 0.0
        grad = np.zeros_like(a_vec)
        
        gamma = last_gamma
        
        # Get the max overlap (and associated s-index) to use in softmax
        c = np.correlate(h, 1-h, mode="full")
        overlaps = c * (2.0/n)
        max_idx = np.argmax(overlaps)
        max_ov = overlaps[max_idx]
        
        # Use softmax with temperature
        exp_vals = np.exp(gamma * (overlaps - max_ov))  
        weights = exp_vals / np.sum(exp_vals)  # [i]
        
        # Grad of overlap at each shift summing with weights
        # Reuse computation from compute_overlap_and_gradient but toward a simpler calculation
        freqs = np.arange(n)  # shape (n)
        cosines_matrix = np.cos(2 * np.pi * freqs[:, np.newaxis] / n)  # [n,n] 
        
        cos_m_0 = np.cos(2 * np.pi * freqs * 0 / n) # 1 for all
        # Let's precompute this differently:
        # For fixed a_m: derivative w.r.t. a_m:
        # d/d a_m [sum h_i (1 - h_{i+s})] = sum_{i=0}^{n-1} [cos(2π m i / n) (1 - h_{i+s}) - h_i cos(2π m (i+s) / n)]
        # To compute this for all m and s at once, we go through s
        for s in range(1-n, n):
            if s <= 0:
                # shift s is negative or 0: compute h_{i+s} where available
                sz = min(n-1, n-1+s) - max(0, -s) + 1
                if sz > 0:
                    i_l = np.arange(max(0, -s), max(0, -s) + sz)
                    i_r = np.arange(max(0, s), max(0, s) + sz)
                    # [sz]:
                    cos_s = cosines_matrix[i_l, :]  # [sz, n]
                    cos_s_r = cosines_matrix[i_r, :]
                    h_l = h[i_l]
                    h_r = h[i_r]
                    sign_term1 = (1 - h_r)  # [sz]
                    sign_term2 = -h_l  # [sz]
                    for m in range(1, M+1):
                        # derivative:
                        der = np.sum(cos_s[:, m-1] * sign_term1 + sign_term2 * cos_s_r[:, m-1])
                        # weighted:
                        grad[m-1] += weights[s + n - 1] * der / n
            else:
                # s > 0, so h_{i+s} = h_{i} where i >= s
                sz = min(n-1, n-1) - max(0, s) + 1
                if sz > 0:
                    i_l = np.arange(s, s + sz)
                    i_r = np.arange(0, sz)
                    cos_s = cosines_matrix[i_l, :]  # [sz, n]
                    cos_s_r = cosines_matrix[i_r, :]
                    h_l = h[i_l]
                    h_r = h[i_r]
                    sign_term1 = (1 - h_r)  # [sz]
                    sign_term2 = -h_l  # [sz]
                    for m in range(1, M+1):
                        der = np.sum(cos_s[:, m-1] * sign_term1 + sign_term2 * cos_s_r[:, m-1])
                        grad[m-1] += weights[s + n - 1] * der / n

        # Neglect the factor 1/n term for approximations in weighting,
        # and multiply by dx = 2/n (hence 2*weights instead)
        # Actually, we already have dx*overlaps and dx*grad, so just like above we're good
        # Compute total objective
        # final overlap is from max, but we also want the softmax of all overlaps
        # This stage doesn't use soft max over overlapping but soft max of overlaps taken as list
        # so anything direct should be fine for penalty and the overlapping
        
        # For the moment we'll return fixed logic:
        # And modify value based on max_overlap from ingredient
        h = build_h_from_coeffs(a_vec, n)
        # For uniform initial inputs, find maximum overlap
        c = np.correlate(h, 1-h, mode="full")
        overlaps = c * (2.0/n)
        max_ov = np.max(overlaps)
        # Softmax objective
        # we add penalty
        return -max_ov + pen  # Will be passed to eval only once

    def obj_and_derivative(a_vec):
        h = build_h_from_coeffs(a_vec, n)
        pen = softmax_penalty(h, lambda_pen)
        
        # Gamma-based soft max as meta-objective
        gamma = last_gamma
        c = np.correlate(h, 1 - h, mode="full")
        overlaps = c * (2.0 / n)
        max_ov = np.max(overlaps)
        exp_vals = np.exp(gamma * (overlaps - max_ov))  
        weights = exp_vals / np.sum(exp_vals)
        
        grad = np.zeros_like(a_vec)
        
        freqs = np.arange(n)
        cosines_matrix = np.cos(2 * np.pi * freqs[:, np.newaxis] / n)  # [n, n]
        
        # All s shifts:
        for s in range(1 - n, n):
            sz = 0
            i_l = []
            i_r = []
            if s <= 0:
                i_l_start = max(0, -s)
                i_r_start = max(0, s)
                if i_l_start + (-s) - 1 < n:  # i_r_start + (-s) - 1 < n
                    sz = n - max(0, -s) - i_r_start  # It's actually: (n-s) - max(0, s) 
                sz = min(n - 1, n - 1 + s) - max(0, -s) + 1  # General formula
                if sz > 0:
                    i_l = np.arange(max(0, -s), max(0, -s) + sz)
                    i_r = np.arange(max(0, s), max(0, s) + sz)
            else:
                i_l_start = s
                i_r_start = 0
                sz = n - s
                if sz > 0:
                    i_l = np.arange(s, s + sz)
                    i_r = np.arange(0, sz)
            if sz > 0:
                h_l = h[i_l]
                h_r = h[i_r]
                term1 = (1 - h_r)  # [sz :]
                term2 = -h_l  # [sz :]
                for m in range(1, M+1):
                    cos_l = cosines_matrix[i_l, m-1]  # [sz]
                    cos_r = cosines_matrix[i_r, m-1]  # [sz]
                    der = np.sum(cos_l * term1 + term2 * cos_r)
                    grad[m-1] += weights[s + n - 1] * der
        
        # Return function value and gradient  
        # Eventually correct: Use proper sum of weights * lorience / ...
        # estimator, etc. 
        # value computed as negated soft maximum 
        return -max_ov + pen, -grad
        
    # Run full stages:
    current_it = 0
    
    # Stage with M=8
    bounds_a = [bounds] * 8  # List of tuples
    M_stage = M  # 8 at start
    
    while current_it < stage_1_iter and save_iter:
        res = minimize(
            fun=lambda a: obj_func(a),
            x0=a_current,
            jac=lambda a: obj_and_derivative(a)[1],
            method='L-BFGS-B',
            bounds=bounds_a,
            options={'maxiter': 1, 'gtol': 1e-9}
        )
        a_current = res.x
        current_it += 1
        if save_iter:
            if current_it % 100 == 0:
                print(f"Iter {current_it}, max_overlap={evaluate_erdos_solution(build_h_from_coeffs(a_current, n))}")
    
    # Now let’s proceed to update M = 15:
    a_current_extended = np.zeros(15)
    a_current_extended[:8] = a_current[:8]
    a_current = a_current_extended
    
    bounds_a = [bounds] * 15
    M_stage = 15
    
    while current_it < stage_1_iter + stage_2_iter and save_iter:
        res = minimize(
            fun=lambda a: obj_func(a),
            x0=a_current,
            jac=lambda a: obj_and_derivative(a)[1],
            method='L-BFGS-B',
            bounds=bounds_a,
            options={'maxiter': 1, 'gtol': 1e-9}
        )
        a_current = res.x
        current_it += 1
        if save_iter and current_it % 100 == 0:
            print(f"Iter {current_it}, max_overlap={evaluate_erdos_solution(build_h_from_coeffs(a_current, n))}")
    
    # Extend to M=24:
    a_current_extended = np.zeros(24)
    a_current_extended[:15] = a_current[:15]
    a_current = a_current_extended
    
    bounds_a = [bounds] * 24
    M_stage = 24
    
    while current_it < stage_1_iter + stage_2_iter + stage_3_iter and save_iter:
        res = minimize(
            fun=lambda a: obj_func(a),
            x0=a_current,
            jac=lambda a: obj_and_derivative(a)[1],
            method='L-BFGS-B',

        if save_iter and current_it % 100 == 0:

    return a_current

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()
    
    n = 240  # Discretization points

    for r in range(num_restarts):

        # Evaluate
        c5 = evaluate_erdos_solution(h_final)
        
        # Update best
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_final.copy()
            
        # Check if we have a lot of time to spend
