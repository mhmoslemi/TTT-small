import numpy as np
import time
import cvxpy as cp
from cvxpy import ECOS, GLPK_MI

def solve_lp_relaxation(N, dx, symmetry=False, monotonic=False):
    # Build index pairs (i, s) for valid overlaps
    pairs = []
    for s in range(N):
        for i in range(N - s):
            pairs.append((i, s))
    
    # Create variables
    h = cp.Variable(N, name="h")
    y = cp.Variable(len(pairs), name="y")
    alpha = cp.Variable(name="alpha")
    
    # Constraints
    constraints = [
        0 <= h,
        h <= 1,
        cp.sum(h) == N / 2,
        alpha >= 0
    ]
    
    # Add McCormick constraints for all pairs
    for idx, (i, s) in enumerate(pairs):
        constraints.extend([
            y[idx] <= h[i],
            y[idx] <= h[i + s],
            y[idx] >= h[i] + h[i + s] - 1,
            y[idx] >= 0
        ])
    
    # Add upper bound constraints for alpha
    # For each shift s, we compute sum_{i=0}^{N-1-s} y_{i,s}
    # which is the sum of y entries indexed by pairs with shift s
    shift_groups = {}
    for idx, (i, s) in enumerate(pairs):
        if s not in shift_groups:
            shift_groups[s] = []
        shift_groups[s].append(idx)
    
    for s, idxs in shift_groups.items():
        # sum_y_s = sum_{i: i+s < N} y[i,s]
        sum_y_s = cp.sum(y[idxs])
        constraints.append(alpha >= dx * (N / 2 - sum_y_s))
    
    # Optional structural constraints
    if symmetry:
        for i in range(N // 2):
            constraints.append(h[i] == h[N - 1 - i])
    
    if monotonic:
        for i in range(N // 2):
            constraints.append(h[i] >= h[i + 1])
    
    # Objective
    objective = cp.Minimize(alpha)
    prob = cp.Problem(objective, constraints)
    
    # Solve
    try:
        prob.solve(solver=ECOS, max_iters=5000)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return h.value, alpha.value
        else:
            return None, None
    except Exception as e:
        return None, None

def project_to_sum_constraint(h, target_sum):
    """Project h to have the exact sum."""
    h = np.clip(h, 0, 1)
    total = np.sum(h)
    if abs(total - target_sum) < 1e-10:
        return h
    # Waterfilling
    if total > target_sum:
        # Reduce h to target_sum by sorting non-increasing and clamping
        h_sorted = np.sort(h)[::-1]
        keep = 0
        cumsum = 0
        for i, val in enumerate(h_sorted):
            if cumsum + val <= target_sum:
                cumsum += val
                keep = i + 1
            else:
                break
        # Set all values that make us exceed target_sum to 0
        h_copy = h.copy()
        sort_ix = np.argsort(h)[::-1]
        for i in range(keep, len(h)):
            h_copy[sort_ix[i]] = 0
        h_copy = np.clip(h_copy, 0, 1)
        # Scale down to exactly target_sum
        curr_sum = np.sum(h_copy)
        if curr_sum > 1e-10:
            h_copy *= target_sum / curr_sum
        return h_copy
    else:
        # Increase sum to exactly target
        h = np.clip(h, 0, 1)
        h /= np.sum(h)
        h *= target_sum
        return h

def propose_move_binary(h, dx):
    """Propose a move between two points choosing pairs that increase overlap."""
    # This helper mimics a simpler version; in practice, could be more sophisticated to handle
    # binary change but here for efficient greedy tuning only
    N = len(h)
    assert(len(h) == N)
    assert(abs(np.sum(h) - N/2) < 1e-10)
    h_new = h.copy()
    # Try to get some neighbourhood by swapping
    while True:
        i = np.random.randint(N)
        j = np.random.randint(N)
        if i != j:
            break
    
    # Transfer one unit fraction from i to j if possible
    # Define feasible change for i
    if h_new[i] <= 0:
        return h_new
    elif h_new[j] >= 1:
        return h_new
    
    # Amount to transfer
    amount = min(1.0 - h_new[j], h_new[i])
    if amount <= 0:
        return h_new
        
    h_new[i] -= amount
    h_new[j] += amount
    
    # Ensure correct sum
    h_new = project_to_sum_constraint(h_new, N / 2)
    return h_new

def c5_of(h):
    """Compute the maximum overlap integral for the given step function h."""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    if len(h) <= 1000:
        c = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(c) * dx
    else:
        # For larger arrays we still compute in batches or optimized way
        # For now, rely on numpy cor function
        c = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(c) * dx
    return c5
