import numpy as np
import time

def c5_of(h):
    """Compute the maximum overlap integral for the given step function h."""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    c = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(c) * dx
    return c5

def worst_shift(h):
    """Find the shift that gives the maximum overlap integral."""
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    return idx - (len(h) - 1)  # signed shift

def subgradient(h, s):
    """Compute subgradient of C5 at shift s."""
    N = len(h)
    dx = 2.0 / N
    m = abs(s)
    g = np.full(N, dx)  # 1*dx everywhere
    if m < N:
        g[:N-m] -= dx * h[m:]  # term with h[i+m]
        g[m:] -= dx * h[:N-m]  # term with h[i-m]
    return g

def project(v, target):
    """Project v onto the feasible set: 0 ≤ h[i] ≤ 1 and sum(h) = target."""
    lo = -np.max(v)
    hi = 1.0 - np.min(v)
    for _ in range(50):
        lam = (lo + hi) / 2.0
        w = np.clip(v + lam, 0.0, 1.0)
        if w.sum() < target:
            lo = lam
        else:
            hi = lam
    w = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        free = (w > 0) & (w < 1)
        if free.any():
            w[free] += diff / free.sum()
            w = np.clip(w, 0.0, 1.0)
    return w

def spd_phase(initial_h, time_limit, N, best_val):
    """Subgradient projected descent phase."""
    h = initial_h.copy()
    start = time.time()
    t = 0
    eta = 30 * 2.0 / N  # Initial step size
    gamma = 0.9995  # Geometric decay
    while time.time() - start < time_limit:
        s_star = worst_shift(h)
        g = subgradient(h, s_star)
        v = h - eta * g
        h = project(v, N / 2)
        c_val = c5_of(h)
        if c_val < best_val:
            best_val = c_val
        t += 1
        if t % 30 == 0:
            # Small random jitter to escape plateaux
            jitter = np.random.uniform(-0.02, 0.02, size=N)
            v = h + jitter
            h = project(v, N / 2)
            c_val = c5_of(h)
            if c_val < best_val:
                best_val = c_val
        eta *= gamma
    return h, best_val

def anneal_phase(initial_h, time_limit, N, best_val):
    """Annealed stochastic moves phase."""
    h = initial_h.copy()
    start = time.time()
    T_0 = 0.05
    T_f = 1e-6
    max_step_0 = 0.2
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        frac = elapsed / time_limit
        T = T_0 * (T_f / T_0) ** frac
        max_step = max_step_0 * (1.0 - frac)
        # Propose a new feasible step function
        i, j = np.random.choice(N, size=2, replace=False)
        delta_low = max(-h[i], h[j] - 1)
        delta_high = min(1 - h[i], h[j])
        if delta_low >= delta_high:
            continue  # Try another pair
        step_size = np.random.uniform(max(-max_step, delta_low),
                                      min(max_step, delta_high))
        h_new = h.copy()
        h_new[i] += step_size
        h_new[j] -= step_size
        c_new = c5_of(h_new)
        if c_new <= c5_of(h) or np.random.rand() < np.exp(-(c_new - c5_of(h)) / T):
            h = h_new
        if c_new < best_val:
            best_val = c_new
    return h, best_val

def hill_climb_phase(initial_h, time_limit, N, best_val):
    """Deterministic hill climbing phase."""
    h = initial_h.copy()
    start = time.time()
    while time.time() - start < time_limit:
        # For rapid convergence: small perturbations around pairs
        i, j = np.random.choice(N, size=2, replace=False)
        delta_low = max(-h[i], h[j] - 1)
        delta_high = min(1 - h[i], h[j])
        if delta_low >= delta_high:
            continue  # Try another pair
        for sign in [-1, 1]:
            step_size = sign * 0.02
            h_new = h.copy()
            h_new[i] += step_size
            h_new[j] -= step_size
            if np.all((0 <= h_new) & (h_new <= 1)):
                c_new = c5_of(h_new)
                if c_new < best_val:
                    h = h_new
                    best_val = c_new
    return h, best_val

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    N = 140
    dx = 2.0 / N

    # Initialize first best solution
    best_h = np.ones(N) * 0.5  # fallback to uniform
    best_val = c5_of(best_h)

    # Phase 0: seeds
    seeds = []

    # From `initial_h_values` if exists
    if 'initial_h_values' in globals():
        h0 = np.interp(np.linspace(0, 2, N),
                       np.linspace(0, 2, len(initial_h_values)),
                       initial_h_values)
        h0 = h0 * (N / 2) / np.sum(h0)  # Renormalize exactly
        seeds.append(h0)
    else:
        seeds.append(np.ones(N) * 0.5)

    # Two-peak seed: 3 intervals (symmetric left/right middle)
    h1 = np.ones(N)
    x_vals = np.linspace(0, 2, N)
    a = 0.3
    u = 0.9
    v = 0.1
    mask = (x_vals < a) | (x_vals >= 2 - a)
    h1[mask] = u
    h1[~mask] = v
    h1 = h1 * (N/2) / np.sum(h1)  # Renormalize
    seeds.append(h1)

    # Random binary seeds
    for _ in range(3):
        h_bin = np.random.choice([0, 1], size=N, p=[0.5, 0.5])
        h_bin = h_bin * (N / 2) / np.sum(h_bin)  # Make sum = N/2
        seeds.append(h_bin)

    # Refine seeds with projection
    for i, s in enumerate(seeds):
        if np.all((0 <= s) & (s <= 1)):
            s = project(s, N / 2)
        c_s = c5_of(s)
        if c_s < best_val:
            best_val = c_s
            best_h = s.copy()

    # Time allocation
    start = time.time()
    phase_length = [0.05, 0.55, 0.30, 0.10]  # Fractions per phase
    phase_time = [length * budget_s for length in phase_length]

    # Phase 1: Subgradient projected descent
    h1, best_val = spd_phase(best_h.copy(), phase_time[0], N, best_val)
    best_h = h1.copy()

    # Phase 2: Annealed moves
    h2, best_val = anneal_phase(best_h.copy(), phase_time[1], N, best_val)
    best_h = h2.copy()

    # Phase 3: Hill climbing
    h3, best_val = hill_climb_phase(best_h.copy(), phase_time[2], N, best_val)
    best_h = h3.copy()

    # Phase 4: Final polishing
    final_time = time.time() - start
    remaining = budget_s - final_time
    if remaining > 0.1:
        final_h, final_val = hill_climb_phase(best_h.copy(), remaining, N, best_val)
        if final_val < best_val:
            best_h = final_h.copy()
            best_val = final_val

    return (best_h, best_val, N)
