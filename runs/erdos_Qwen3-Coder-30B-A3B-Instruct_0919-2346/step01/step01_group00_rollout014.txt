import numpy as np
import time
import random

def project_h(h, target_sum):
    """Project h onto [0,1] and sum=target_sum using water-filling."""
    h = h.copy()
    current_sum = np.sum(h)
    if np.isclose(current_sum, target_sum):
        return h
    
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    
    # Binary search for lambda
    low, high = -1.0, 1.0
    tol = 1e-10
    
    while high - low > tol:
        mid = (low + high) / 2
        h_test = np.clip(h - mid, 0, 1)
        test_sum = np.sum(h_test)
        if test_sum > target_sum:
            low = mid
        else:
            high = mid
    
    lambda_val = (low + high) / 2
    h = np.clip(h - lambda_val, 0, 1)
    
    return h

def compute_C_and_grad(h, alpha):
    """Compute max C_k and gradient with smoothing."""
    N = len(h)
    dx = 2.0 / N
    
    # Compute correlations via FFT
    fft_h = np.fft.fft(h, n=2*N-1)
    fft_comp = np.fft.fft(1-h, n=2*N-1)
    S = np.real(np.fft.ifft(fft_h * np.conj(fft_comp)))
    C = 1 - dx * S[:2*N-1]
    
    # Compute softmax weights
    w = np.exp(alpha * C)
    w = w / np.sum(w)
    
    # Compute gradient
    conv1 = np.convolve(h, w[::-1], mode='same')
    conv2 = np.convolve(h, w, mode='same')
    grad = dx * (1 - (conv1 + conv2))
    
    # Compute max C
    c_max = np.max(C)
    
    return c_max, grad

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    N = 140
    dx = 2.0 / N
    target_sum = N / 2
    
    # Initial shapes
    initial_h = np.interp(np.linspace(0, 2, N), 
                          np.linspace(0, 2, len(initial_h_values)), 
                          initial_h_values)
    initial_h = initial_h * (N / 2) / np.sum(initial_h)
    
    best_h = initial_h.copy()
    best_c5 = 1e10
    
    start_time = time.time()
    
    # Use different starting points
    seeds = [
        initial_h,
        np.ones(N) * 0.5,
        np.array([0.9 if abs(x - 1) < 0.3 else 0.1 for x in np.linspace(0, 2, N)])
    ]
    
    random.shuffle(seeds)
    
    for seed_idx, h_start in enumerate(seeds):
        h = h_start.copy()
        h = project_h(h, target_sum)
        
        # Temporal schedule parameters
        max_iter = 10000
        eta0 = 0.5
        alpha_start = 5.0
        alpha_end = 100.0
        
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 1.0:
                break
                
            # Time-dependent scheduling
            alpha = alpha_start + (alpha_end - alpha_start) * i / max_iter
            eta = eta0 / (1 + i / 1000)
            
            c_max, grad = compute_C_and_grad(h, alpha)
            
            # Update with gradient
            h_new = h - eta * grad
            
            # Project to feasible region
            h_new = project_h(h_new, target_sum)
            
            # Maybe accept with some probability based on better c_max
            if c_max < best_c5 or i % 500 == 0:
                # Evaluate exact c5
                # Correlation of h with 1-h
                h_extended = np.pad(h, (0, N-1), 'constant', constant_values=0)
                comp = 1 - h
                comp_extended = np.pad(comp, (0, N-1), 'constant', constant_values=0)
                corr = np.correlate(comp_extended, h_extended, mode='full')
                c_exact = np.max(corr[:2*N-1]) * dx
                
                if c_exact < best_c5:
                    best_c5 = c_exact
                    best_h = h.copy()
            
            h = h_new
    
    # Final polishing with smaller step size
    if time.time() - start_time < budget_s - 1.0:
        max_iter_final = 5000
        eta_final = 0.1
        alpha_final = 150.0
        
        for i in range(max_iter_final):
            if time.time() - start_time > budget_s - 0.5:
                break
                
            c_max, grad = compute_C_and_grad(best_h, alpha_final)
            best_h_new = best_h - eta_final * grad
            best_h_new = project_h(best_h_new, target_sum)
            
            # Compute actual max
            h_extended = np.pad(best_h_new, (0, N-1), 'constant', constant_values=0)
            comp = 1 - best_h_new
            comp_extended = np.pad(comp, (0, N-1), 'constant', constant_values=0)
            corr = np.correlate(comp_extended, h_extended, mode='full')
            c_exact = np.max(corr[:2*N-1]) * dx
            
            if c_exact < best_c5:
                best_c5 = c_exact
                best_h = best_h_new.copy()
            
            best_h = best_h_new
    
    return (best_h, best_c5, N)
