import numpy as np
import time
import random
from scipy.signal import correlate
import scipy.optimize as opt

def compute_c5(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    overlap = correlate(h, 1 - h, mode="full")
    c5 = np.max(overlap) * dx
    return c5

def project_h(h, dx):
    """Project h to satisfy the constraint sum(h) * dx = 1"""
    total = np.sum(h) * dx
    if abs(total - 1.0) < 1e-10:
        return h
    factor = 1.0 / total
    h_proj = h * factor
    # Ensure bounds
    h_proj = np.clip(h_proj, 0, 1)
    return h_proj

def project_all(h, dx):
    """Project h onto feasible set [0,1] with exact integral constraint"""
    h = np.clip(h, 0, 1)
    h = project_h(h, dx)
    return h

def eval_grad_c5(h, eps=1e-6):
    """Approximate gradient of C5 w.r.t h using finite differences"""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros_like(h)
    
    # Use finite differences with central differences
    for i in range(n):
        h_plus = h.copy()
        h_minus = h.copy()
        h_plus[i] += eps
        h_minus[i] -= eps
        h_plus = project_all(h_plus, dx)
        h_minus = project_all(h_minus, dx)
        
        c5_plus = compute_c5(h_plus)
        c5_minus = compute_c5(h_minus)
        grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
    return grad

def proj_grad_descent(h0, max_iter=1000, step_size=1e-3, tol=1e-6, verbose=False):
    """Projected gradient descent on h to minimize C5"""
    h = h0.copy()
    dx = 2.0 / len(h)
    prev_c5 = compute_c5(h)
    for i in range(max_iter):
        grad = eval_grad_c5(h)
        h = h - step_size * grad
        h = project_all(h, dx)
        c5 = compute_c5(h)
        if verbose and i % 100 == 0:
            print(f"  Iter {i}: C5 = {c5}")
        if abs(c5 - prev_c5) < tol:
            break
        prev_c5 = c5
    return h, c5

        )
        h_interp = project_h(h_interp, 2.0/N)
        seeds.append(h_interp)
    else:
        # Fallback uniform
        h2 = np.ones(N) * 0.5
        seeds.append(h2)
        
    # 3. Binary-like with two peaks
    h3 = np.full(N, 0.1)
    a = 0.3
    u = 0.9
    v = 0.1
    left = int(a * N / 2)
    right = int((2 - a) * N / 2)
    h3[:left] = u
    h3[right:] = u
    h3 = project_h(h3, 2.0/N)
    seeds.append(h3)
    
    # 4. Random with hard constraint on sum
    h4 = np.random.rand(N)
    h4 = project_h(h4, 2.0/N)
    seeds.append(h4)
    
    return seeds

def refine_with_perturbations(h_start, budget_s):
    """Refine by perturbing the current best solution"""
    start = time.time()
    h = h_start.copy()
    c5 = compute_c5(h)
    dx = 2.0 / len(h)
    
    # Find the worst shift
    overlap = correlate(h, 1-h, mode='full')
    k_worst = np.argmax(overlap)
    # k_worst implies shift of k_worst - (2*N-1) // 2
    
    # Do local one-move iterations
    max_iter = int(budget_s * 200)  # 200 moves per second estimate
    
    for i in range(max_iter):
        if time.time() - start > budget_s:
            break
            
        # Try random adjacent pairs and move mass
        idx = random.randint(0, len(h)-1)
        idx2 = (idx + random.choice([-1, 1])) % len(h)
        if idx == idx2:
            continue
            
        # Determine allowable movement
        delta_min = max(-h[idx], h[idx2] - 1)
        delta_max = min(1 - h[idx], h[idx2])
        if delta_min >= delta_max:
            continue
            
        # Move mass
        delta = random.uniform(delta_min, delta_max)
        h[idx] += delta
        h[idx2] -= delta
        
        # Project
        h = project_all(h, dx)
        
        new_c5 = compute_c5(h)
        if new_c5 <= c5:
            c5 = new_c5
        else:
            # Or simply do not accept worse solutions (but accept equal ones)
            pass
    
    return h, c5

    best_h = None
    best_c5 = float('inf')
    
    # Allocate time: 50% for main optimization, 25% for local refinement, 25% for polish

    alloc2 = budget_s * 0.25
    alloc3 = budget_s * 0.25
    
    for i, h_seed in enumerate(seeds):

        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            
    # Step 2: Do further local refinement using targeted moves
    best_h, best_c5 = refine_with_perturbations(best_h, alloc2)
    
    # Step 3: Final short polishing
    final_h, final_c5 = refine_with_perturbations(best_h, alloc3) 
    if final_c5 < best_c5:
        best_h = final_h
        best_c5 = final_c5
    
    return (best_h, best_c5, N)
