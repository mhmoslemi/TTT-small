import numpy as np
import time
import random
from scipy.optimize import differential_evolution

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using full convolution
    overlap = np.correlate(h, 1 - h, mode='full')
    # The maximum value scaled by dx
    c5 = np.max(overlap) * dx
    return c5

def propose_move(h, max_step):
    """Propose a move that maintains sum constraint exactly"""
    h_new = h.copy()
    N = len(h)
    while True:
        i, j = random.sample(range(N), 2)
        # Compute feasible range for delta
        delta_low = max(-h_new[i], h_new[j] - 1.0)
        delta_high = min(1.0 - h_new[i], h_new[j])
        if delta_low >= delta_high:
            continue  # No valid move, try again
        break
    
    # Sample delta from feasible interval
    step = min(max_step, delta_high - delta_low)
    if step <= 0:
        delta = 0.0
    else:
        delta = random.uniform(max(-step, delta_low), min(step, delta_high))
        
    h_new[i] += delta
    h_new[j] -= delta
    return h_new

def optimize_with_differential_evolution(h_initial):
    """Use differential evolution for global optimization"""
    N = len(h_initial)
    bounds = [(0, 1) for _ in range(N)]
    
    def objective(h):
        # Ensure sum constraint
        h = h * (N / 2) / np.sum(h)
        return c5_of(h)
    
    # Differential evolution with bounds and constraints
    result = differential_evolution(
        objective, 
        bounds, 
        maxiter=100, 
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.7,
        seed=42,
        disp=False,
        polish=True
    )
    
    # Apply sum constraint manually
    h_opt = result.x * (N / 2) / np.sum(result.x)
    return h_opt, objective(h_opt)

    smoothed = np.clip(smoothed, 0, 1)

    smoothed = smoothed * (len(h) / 2) / np.sum(smoothed)
    return smoothed
