import numpy as np

def compute_C_vector(h):
    """Compute vector of overlaps C_tau for all shifts."""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation S_tau = sum h_i h_{i+tau}
    # Zero pad h to length 2n-1 for full correlation
    h_pad = np.pad(h, (0, n-1), mode='constant', constant_values=0)
    S = np.correlate(h_pad, h, mode='valid')
    # C_tau = 1 - dx * S_tau
    c = 1.0 - dx * S
    return c

def compute_softmax_weights(c, alpha):
    """Compute soft-max weights w_tau for smooth max."""
    # Stability trick: subtract max to avoid overflow
    c_max = np.max(c)
    exp_c = np.exp(alpha * (c - c_max))
    w = exp_c / np.sum(exp_c)
    return w

def compute_gradient(h, w):
    """Compute gradient of smooth objective with weights w."""
    n = len(h)
    dx = 2.0 / n
    # Two convolutions: w * h and w_rev * h
    # w_rev is w reversed
    w_rev = w[::-1]
    # Use same mode as in numpy's convolve
    conv1 = np.convolve(w, h, mode='same')
    conv2 = np.convolve(w_rev, h, mode='same')
    g = -dx * (conv1 + conv2)
    return g

def project_box_and_mass(h, target_sum):
    """Project h onto box [0,1] and sum = target_sum using water filling."""
    h_clipped = np.clip(h, 0, 1)
    current_sum = np.sum(h_clipped)
    if abs(current_sum - target_sum) < 1e-12:
        return h_clipped.copy()
    
    # Bisection on lambda
    def phi(lam):
        # Compute sum of clipped values
        x = np.clip(h - lam, 0, 1)
        return np.sum(x) - target_sum
    
    # Find a bracketing interval
    lower, upper = np.min(h) - 1.0, np.max(h) + 1.0
    while phi(lower) > 0:
        lower -= 1.0
    while phi(upper) < 0:
        upper += 1.0
    
    # Binary search for lambda
    tol = 1e-12
    while upper - lower > tol:
        mid = (lower + upper) / 2
        if phi(mid) > 0:
            upper = mid
        else:
            lower = mid
    
    lambda_opt = (lower + upper) / 2
    h_proj = np.clip(h - lambda_opt, 0, 1)
    
    # Double-check sum and enforce with small adjustment if needed
    current_sum = np.sum(h_proj)
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        # Adjust by distributing diff uniformly among allowed (non-boundary) entries
        mask = (h_proj > 0) & (h_proj < 1)
        if np.sum(mask) > 0:
            adjust = diff / np.sum(mask)
            h_proj[mask] += adjust
    
    # Final clip is necessary to prevent numerical errors
    h_proj = np.clip(h_proj, 0, 1)
    return h_proj
