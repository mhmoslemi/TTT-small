def c5_of(h):
    """Compute C5 = max_k integral h(x)(1-h(x+k)) dx."""
    dx = 2.0 / len(h)
    # Compute correlation: h * (1 - h) with full mode
    c = np.correlate(h, 1 - h, mode="full")
    # Multiply by dx to get continuous integral
    c *= dx
    # Return maximum value among all shifts
    return np.max(c)

def smooth_h(h, alpha=0.5):
    """Apply smoothing to make h more continuous"""
    h_new = h.copy()
    for i in range(1, len(h)-1):
        h_new[i] = alpha * h[i] + (1-alpha) * (h[i-1] + h[i+1]) / 2

def constrain_h(h):
    """Ensure h is in [0,1] and sum == n_points/2"""
    h = np.clip(h, 0, 1)
