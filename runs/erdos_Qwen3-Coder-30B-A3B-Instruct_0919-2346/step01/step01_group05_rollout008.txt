import numpy as np
import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value scaled by dx
    max_overlap = np.max(correlation) * dx
    return max_overlap

def generate_sturmian_binary(n):
    """Generate a balanced binary sequence using irrational rotation"""
    alpha = (np.sqrt(5) - 1) / 2
    h = np.zeros(n)
    positions = np.mod(np.arange(n) * alpha, 1)
    h[positions < 0.5] = 1
    ones_count = int(np.sum(h))
    if ones_count > n // 2:
        # Change some 1s to 0s
        zeros = np.where(h == 1)[0][:ones_count - n // 2]
        h[zeros] = 0
    elif ones_count < n // 2:
        # Change some 0s to 1s
        ones = np.where(h == 0)[0][:n // 2 - ones_count]
        h[ones] = 1
    return h

def compute_overlap_vector(h, dx):
    """Compute all overlap values for given h"""
    correlation = np.correlate(h, 1 - h, mode="full")
    return correlation * dx

def compute_delta_s(i, j, h, overlap_vec, s_max, dx):
    """Compute delta delta_s for swap of positions i and j"""
    n = len(h)
    delta = np.zeros(2 * n - 1)
    
    # For each shift, compute the change in overlap
    for s in range(2 * n - 1):
        # Calculate where i and j contribute to overlap
        if i < n - s:
            contribution_i = h[i] * (1 - h[i + s])
            if h[i] == 1 and h[i + s] == 0:
                delta[s] -= (1 - h[i + s])
        if (i + s) < n and i >= s:
            contribution_i = h[i - s] * (1 - h[i])
            if h[i - s] == 1 and h[i] == 0:
                delta[s] += h[i - s]

        if j < n - s:
            contribution_j = h[j] * (1 - h[j + s])
            if h[j] == 0 and h[j + s] == 1:
                delta[s] += (1 - h[j + s])
        if (j + s) < n and j >= s:
            contribution_j = h[j - s] * (1 - h[j])
            if h[j - s] == 1 and h[j] == 0:
                delta[s] -= h[j - s]

    return delta

def local_swap_search(h, max_time):
    """Perform local swaps to improve C5"""
    start_time = time.time()
    n = len(h)
    dx = 2.0 / n
    worst_shift = np.argmax(compute_overlap_vector(h, dx))
    current_c5 = evaluate_erdos_solution(h)
    overlap_vec = compute_overlap_vector(h, dx)
    best_h = h.copy()
    best_c5 = current_c5
    
    # Tabu list to avoid cycling
    tabu_list = []
    tabu_size = 50
    improvement_count = 0 
    stagnation_count = 0
    total_swaps = 0
    
    while time.time() - start_time < max_time:
        # Find current bottleneck
        max_index = np.argmax(overlap_vec)
        max_val = overlap_vec[max_index]
        if max_val < best_c5:
            best_c5 = max_val
            best_h = h.copy()
            
        # Early termination if improvement not seen recently
        if (stagnation_count > 500):
            break
            
        # Choose a random 1 and 0 to swap, but with preferences
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
            
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Check if swap is not tabu
        if tuple(sorted([i,j])) in tabu_list:
            if np.random.rand() < 0.1:  # 10% chance to ignore tabu
                pass
            else:
                continue
                
        # Compute delta's
        delta = compute_delta_s(i, j, h, overlap_vec, max_index, dx)
        new_overlap = overlap_vec + delta
        
        new_max = np.max(new_overlap)
        if new_max < current_c5 or np.random.rand() < 0.01:
            # Accept the swap
            h[i] = 0
            h[j] = 1
            
            overlap_vec = new_overlap
            current_c5 = new_max
            
            # Update tabu list
            tabu_list.append(tuple(sorted([i, j])))
            if len(tabu_list) > tabu_size:
                tabu_list.pop(0)
            
            improvement_count += 1
            stagnation_count = 0
            total_swaps += 1
            
        else:
            stagnation_count += 1
            
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Multi-scale optimization
    resolutions = [120, 240]
    best_h = None
    best_c5 = float('inf')
    
    for n in resolutions:
        if time.time() - start_time > budget_s * 0.9:
            break
            
        h = generate_sturmian_binary(n)
        
        # Local swap search with 80% of remaining time
        remaining_time = budget_s * 0.8 - (time.time() - start_time)
        h, c5 = local_swap_search(h, min(remaining_time, 500))
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Try to refine if time allows
        if time.time() - start_time < budget_s * 0.95:
            # Small perturbation to improve
            # Use least significant change in correlate
            try:
                correlation = np.correlate(h, 1 - h, mode="full") * (2.0 / n)
                max_shift_idx = np.argmax(correlation)
                if max_shift_idx > 0 and max_shift_idx < len(correlation) - 1:
                    # Perturb positions near max shift by slightly changing values
                    max_shift = max_shift_idx - len(h) + 1
                    idxs = np.array([max_shift, -max_shift])
                    for idx in idxs:
                        if 0 <= idx < len(h):
                            h[idx] = max(0, min(1, h[idx] + 0.01 * np.random.choice([-1, 1])))
                    
                    h_t = h.copy()
                    # Re-optimize
                    h_t = np.clip(h_t, 0, 1)
                    h_t = h_t * (1.0 / np.sum(h_t) * n / 2)
                    
                    c5_refined = evaluate_erdos_solution(h_t)
                    if c5_refined < best_c5:
                        best_c5 = c5_refined
                        best_h = h_t.copy()
            except:
                pass
                
    # Final refinement using SLSQP
    if best_h is not None and (time.time() - start_time) < budget_s * 0.98:
        h = best_h.copy()
        n = len(h)
        dx = 2.0 / n
        
        def objective(h_vec):
            h = np.array(h_vec)
            return evaluate_erdos_solution(h)
            
        bounds = [(0, 1) for _ in range(n)]
        
        def constraint_fun(h_vec):
            h = np.array(h_vec)
            return np.sum(h) - n / 2.0
            
        constraint = {'type': 'eq', 'fun': constraint_fun}
        
        try:
            result = minimize(objective, h, method='SLSQP', 
                              bounds=bounds, constraints=[constraint], 
                              tol=1e-7, options={'maxiter': 200})
            if result.success:
                h = result.x
                c5 = evaluate_erdos_solution(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
        except:
            pass
    
    if best_h is None:
        # Default solution if nothing worked
        best_h = np.ones(45) * 0.5
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), best_c5, len(best_h))
