import numpy as np
import time

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral via full convolution of h with (1-h)
    overlaps = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum of the full correlation vector
    # scaled by dx
    c5 = np.max(overlaps) * dx
    return c5

def full_overlap(h):
    """Return full correlation vector of h with (1-h)"""
    return np.correlate(h, 1 - h, mode="full")

def project_onto_P(v, total):
    """Project v onto the set {x : 0 ≤ x_i ≤ 1, sum x_i = total}"""
    # Use bisection to find the correct λ
    def project_with_lambda(lam):
        x = np.clip(v - lam, 0, 1)
        return x, np.sum(x) - total

    # Initial bounds for lambda
    v_min = np.min(v)
    v_max = np.max(v)
    lam_min = v_min - 1
    lam_max = v_max + 1
    
    # Safeguard bound on lambda from the projection formula
    while True:
        x, error = project_with_lambda(lam_min)
        if error < 0:
            break
        lam_min -= 1
    while True:
        x, error = project_with_lambda(lam_max)
        if error > 0:
            break
        lam_max += 1

    # Tolerance is small enough for machine precision
    while lam_max - lam_min > 1e-12:
        lam = (lam_min + lam_max) / 2
        _, error = project_with_lambda(lam)
        if error > 0:
            lam_min = lam
        else:
            lam_max = lam

    # Final projection with rounded lambda to avoid numerical issues
    x = np.clip(v - lam_max, 0, 1)
    scale = total / np.sum(x)
    x *= scale
    return x

def smooth_gradient(h, beta):
    """Compute gradient ∇Φ_β(h) = sum_k w_k [1 - (h^{+k} + h^{-k})]"""
    N = len(h)
    # Compute overlap vector F_k(h) = sum h_i * (1 - h_{i+k}) = N/2 - sum h_i * h_{i+k}
    F = full_overlap(h)
    F = F - np.sum(h) / 2  # So F_k = sum h_i h_{i+k} if properly recentered
    
    # Apply softmax on beta * F with proper centering for numerical stability
    F_centered = F - np.max(F)
    w = np.exp(beta * F_centered)
    w /= np.sum(w)
    
    # Compute the weighted gradient
    g = np.ones(N)
    for k in range(-N+1, N):
        wk = w[k + N - 1]
        if k >= 0:
            g[:N-k] -= wk * h[k:]
        else:
            d = -k
            g[d:] -= wk * h[:N-d]
    return g

def projected_gradient_descent(h0, max_iter, beta_start, beta_max, gamma_beta, gamma_step, time_budget=None):
    start = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of(h)
    
    beta = beta_start
    step = 0.5
    
    for iteration in range(max_iter):
        if time_budget and time.time() - start > time_budget:
            break
            
        # gradient computation for current β
        g = smooth_gradient(h, beta)
        
        # tentative update
        v = h - step * g
        
        # projection
        h = project_onto_P(v, np.sum(h0))
        
        # evaluation
        cur_c5 = c5_of(h)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
        
        # adaptation
        if iteration % 50 == 0:
            beta = min(beta * gamma_beta, beta_max)
        step *= gamma_step
    
    return best_h, best_c5

def targeted_refinement(h, time_budget):
    """Perform greedy, pairwise local search on the worst overlap pair."""
    start = time.time()
    h = h.copy()
    c5 = c5_of(h)
    alpha = 1.0
    patience = 0
    update_threshold = 0.1  # Min improvement threshold
    
    while time.time() - start < time_budget and patience < 200:
        C_full = full_overlap(h)
        d = np.argmax(C_full) - (len(h) - 1)
        N = len(h)
        delta = 0
        
        if d >= 0:
            hi = h[:N - d]
            hj = h[d:]
        else:
            hi = h[-d:]
            hj = h[:N + d]
        
        # Find worst contributing pair (maximizes h_i * (1 - h_j))
        max_val = -1
        i_star = j_star = -1
        for i in range(len(hi)):
            val = hi[i] * (1 - hj[i])
            if val > max_val:
                max_val = val
                i_star = i
                j_star = i + d if d >= 0 else i - d
                
        # Deterministic transfer
        if i_star == -1:
            patience += 1
            continue
            
        delta = min(alpha * max_val, h[i_star], 1 - h[j_star])
        if delta < 1e-12:
            patience += 1
            continue
            
        h[i_star] -= delta
        h[j_star] += delta
        
        new_c5 = c5_of(h)
        if new_c5 <= c5 - update_threshold:
            c5 = new_c5
            patience = 0
            alpha = 1.0
        else:
            h[i_star] += delta
            h[j_star] -= delta
            alpha *= 0.5
            patience += 1  # increase patience since no improvement
            
    return h, c5

    # Run on all seeds with 30% of total budget
    best_h = None
