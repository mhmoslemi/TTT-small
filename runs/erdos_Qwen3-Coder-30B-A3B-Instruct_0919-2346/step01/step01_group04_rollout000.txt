from scipy.signal import correlate

def compute_c5(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    overlap = correlate(h, 1 - h, mode="full")
    c5 = np.max(overlap) * dx
    return c5

def project_h(h, dx):
    """Project h onto feasible set: 0 <= h[i] <= 1 and sum(h)*dx = 1"""
    h = np.clip(h, 0, 1)

    if h_sum > 0:

    else:

        h = np.full_like(h, 1.0 / len(h))
    return h
