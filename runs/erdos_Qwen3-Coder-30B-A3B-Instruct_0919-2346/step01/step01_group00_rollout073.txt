import numpy as np
import time
import random
from scipy.optimize import differential_evolution, minimize

def c5_of(h):
    """Compute C5 = max_k integral h(x)(1-h(x+k)) dx."""
    dx = 2.0 / len(h)
    # Compute correlation: h * (1 - h) with full mode
    c = np.correlate(h, 1 - h, mode="full")
    # Multiply by dx to get continuous integral
    c *= dx
    # Return maximum value among all shifts
    return np.max(c)

def project_to_feasible(h):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n/2."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    n = len(h)
    # Scale to meet sum constraint
    h = h * (n / 2) / h_sum
    return h

def propose_move(h, max_step):
    """Generate a new feasible h by adjusting two entries."""
    h = h.copy()
    N = len(h)
    i, j = random.sample(range(N), 2)
    
    # Determine feasible interval for delta
    delta_low = max(-h[i], h[j] - 1.0)
    delta_high = min(1.0 - h[i], h[j])
    
    if delta_low >= delta_high:
        return h  # No valid move for this pair
    
    # Sample delta within allowed range
    step = min(max_step, delta_high - delta_low)
    delta = random.uniform(-step, step)
    delta = max(delta_low, min(delta, delta_high))
    
    h[i] += delta
    h[j] -= delta
    
    return project_to_feasible(h)

def get_random_feasible_h(N):
    """Generate a random feasible h."""
    h = np.random.random(N)
    h = project_to_feasible(h)
    return h

                c_cur = c_initial
                best = better_initial
                c_best = c_initial
    except:
        pass
    
    # Phase 1: High temperature exploration
    T_start = 0.2
    T_end = 1e-6
    step_start = 0.1
    phase1_time = 0.3 * time_budget
    
    while time.time() - start < phase1_time:
        t = time.time() - start
        T = T_start * (T_end / T_start) ** (t / phase1_time)
        step = step_start * (1 - t / phase1_time)
        
        h_new = propose_move(cur, step)
        c_new = c5_of(h_new)
        
        if c_new <= c_cur or random.random() < np.exp(-(c_new - c_cur) / T):
            cur = h_new
            c_cur = c_new
            if c_new < c_best:
                best = h_new
                c_best = c_new
    
    # Phase 2: Medium temperature exploitation
    phase2_time = 0.5 * time_budget
    T_start = 0.05
    T_end = 1e-5
    step_start = 0.05
    
    while time.time() - start < phase2_time + start:
        t = time.time() - start
        T = T_start * (T_end / T_start) ** ((t - phase1_time) / (phase2_time))
        step = step_start * (1 - (t - phase1_time) / phase2_time)
        
        h_new = propose_move(cur, step)
        c_new = c5_of(h_new)
        
        if c_new <= c_cur or random.random() < np.exp(-(c_new - c_cur) / T):
            cur = h_new
            c_cur = c_new
            if c_new < c_best:
                best = h_new
                c_best = c_new
    
    # Phase 3: Low temperature fine-tuning
    phase3_time = time_budget - (phase1_time + phase2_time)
    T_start = 0.005
    T_end = 1e-7
    step_start = 0.01
    
    while time.time() - start < time_budget and time.time() - start < phase3_time + start:
        t = time.time() - start
        T = T_start * (T_end / T_start) ** ((t - phase1_time - phase2_time) / phase3_time)
        step = step_start * (1 - (t - phase1_time - phase2_time) / phase3_time)
        
        h_new = propose_move(cur, step)
        c_new = c5_of(h_new)
        
        if c_new <= c_cur or random.random() < np.exp(-(c_new - c_cur) / T):
            cur = h_new
            c_cur = c_new
            if c_new < c_best:
                best = h_new
                c_best = c_new
    
    return best, c_best
