import numpy as np
import time

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral via full convolution of h with (1-h)
    overlaps = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum of the full correlation vector
    # scaled by dx
    c5 = np.max(overlaps) * dx
    return c5

def full_overlap(h):
    """Compute the raw correlation vector"""
    return np.correlate(h, 1 - h, mode="full")

def project_onto_P(v, total):
    """Project v onto the set {x : 0 <= x_i <= 1, sum x_i = total}"""
    # Bisection to find lambda
    lb, ub = np.min(v) - 1, np.max(v) + 1
    while ub - lb > 1e-12:
        mid = (lb + ub) / 2
        x = np.clip(v - mid, 0, 1)
        if np.sum(x) > total:
            lb = mid
        else:
            ub = mid
    x = np.clip(v - lb, 0, 1)
    return x

def smooth_gradient(h, beta, F):
    """Compute gradient of log-sum-exp of F scaled by beta"""
    # F_k = sum_i h_i*(1-h_{i+k})
    # We use that dF_k/dh_i = 1 - (h_{i+k} + h_{i-k})
    # For numerical stability, subtract F.max()
    F_shifted = F - np.max(F)
    w = np.exp(beta * F_shifted)
    w = w / np.sum(w)
    
    g = np.ones_like(h)
    for k in range(-len(h)+1, len(h)):
        wk = w[k + len(h) - 1]
        if k >= 0:
            g[:len(h)-k] -= wk * h[k:]
        else:
            g[-k:] -= wk * h[:len(h)+k]
    return g

def projected_gradient_descent(h0, time_budget, beta_start=2, beta_max=30, gamma_beta=1.5, gamma_step=0.9995, max_iter=100000):
    start = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_C5 = c5_of(h)
    
    beta = beta_start
    step = 0.5
    
    for i in range(max_iter):
        if time.time() - start > time_budget:
            break
            
        F = full_overlap(h)
        g = smooth_gradient(h, beta, F)
        v = h - step * g
        h = project_onto_P(v, len(h) / 2)
        
        cur_C5 = c5_of(h)
        if cur_C5 < best_C5:
            best_C5 = cur_C5
            best_h = h.copy()
        
        # Adaptive step and beta scheduling
        step *= gamma_step
        if i % 100 == 0:
            beta = min(beta * gamma_beta, beta_max)
    
    return best_h, best_C5

def targeted_refinement(h0, time_budget):
    start = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_C5 = c5_of(h)
    
    alpha = 1.0
    stale_count = 0
    
    while time.time() - start < time_budget:
        C = full_overlap(h)
        d = np.argmax(C) - (len(h) - 1)
        C_max = np.max(C)
        
        if stale_count >= 200:
            break
            
        # Extract the overlapping slices
        if d >= 0:
            hi = h[:len(h)-d]
            hj = h[d:]
        else:
            d_abs = -d
            hi = h[d_abs:]
            hj = h[:len(h)-d_abs]

        # Compute the contribution of each pair
        contribs = hi * (1 - hj)
        # Find the one with max contribution
        idx = np.argmax(contribs)
        c = contribs[idx]
        
        # Compute feasible delta
        h_i = hi[idx]
        h_j = hj[idx]
        delta = min(alpha * c, h_i, 1 - h_j)
        
        # Update
        if delta > 1e-10:
            hi[idx] -= delta
            hj[idx] += delta
            
            # Restore to h array
            if d >= 0:
                h[:len(h)-d] = hi
                h[d:] = hj
            else:
                h[d_abs:] = hi
                h[:len(h)-d_abs] = hj
            
            new_C5 = c5_of(h)
            if new_C5 < best_C5:
                best_C5 = new_C5
                best_h = h.copy()
                stale_count = 0
                alpha = 1.0  # reset
            else:
                stale_count += 1
                alpha *= 0.5  # halve it
                # revert changes
                if d >= 0:
                    hi[idx] += delta
                    hj[idx] -= delta
                    # Restore to h array
                    h[:len(h)-d] = hi
                    h[d:] = hj
                else:
                    hi[idx] += delta
                    hj[idx] -= delta
                    # Restore to h array
                    h[d_abs:] = hi
                    h[:len(h)-d_abs] = hj
                if alpha < 1e-6:
                    break
        else:
            stale_count += 1
            if stale_count >= 200:
                break
    
    return best_h, best_C5
