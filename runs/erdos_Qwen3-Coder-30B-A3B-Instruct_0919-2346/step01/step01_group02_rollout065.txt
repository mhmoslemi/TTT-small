def c5_of(h):
    """Compute the maximum overlap integral for the given step function h."""
    dx = 2.0 / len(h)
    c = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(c) * dx
    return c5

def project_to_feasible(h, N):
    """Project h onto the feasible set: [0,1]^N and sum = N/2"""
    # Use water-filling to project onto sum constraint
    v = h.copy()
    target_sum = N / 2.0
    
    def F(lambda_val):
        return np.sum(np.clip(v + lambda_val, 0, 1))
    
    # Binary search for lambda
    low, high = -2.0, 2.0
    for _ in range(50):
        mid = (low + high) / 2.0
        if F(mid) < target_sum:
            low = mid
        else:
            high = mid
    
    lambda_star = (low + high) / 2.0
    h_proj = np.clip(v + lambda_star, 0, 1)
    
    # Adjust for numerical precision
    current_sum = np.sum(h_proj)
    diff = target_sum - current_sum
    
    # Distribute difference among non-bound entries
    free_indices = np.where((h_proj > 0) & (h_proj < 1))[0]
    if len(free_indices) > 0:
        adjust = diff / len(free_indices)
        for i in free_indices:
            h_proj[i] += adjust
    
    return h_proj

def solve_continuation(h0, N, epsilon_sequence, max_iter=150):
    """Perform continuation scheme with smooth surrogate minimization"""
    h = h0.copy()
    h = project_to_feasible(h, N)
    
    # Project at start before inner loop to ensure exact sum
    # for each soft parameter:
    # run outer loop, with bounds
    for epsilon in epsilon_sequence:
        # Solve smooth surrogate F_epsilon using L-BFGS-B
        bounds = Bounds(np.zeros(N), np.ones(N))
        
        # subfunction to evaluate surrogate value and gradient
        def func_and_grad(x):
            return smooth_surrogate_and_grad(x, epsilon, 2.0/N)
            
        result = minimize(func_and_grad, h, method='L-BFGS-B', bounds=bounds, jac=True, 
                          options={'maxiter': max_iter, 'gtol': 1e-6})
        
        h = result.x
        h = project_to_feasible(h, N)  # Ensure feasible after iteration

        # Do final projection  after inner loop
        h = project_to_feasible(h, N)
    
    # Final projection
    h = project_to_feasible(h, N)
    return h

def run(seed=42, budget_s=600):
    start = time.time()
    
    np.random.seed(seed)
    N = 140   # Same as the benchmark
    dx = 2.0 / N
    
    # Seeding strategies
    # 1. Uniform start
    h_uniform = np.ones(N) * 0.5
    
    # 2. Two-peak seed 
    h_two_peak = np.ones(N)
    a = 0.30
    u = 0.90
    v = 0.10
    x_vals = np.linspace(0, 2, N)
    mask = (x_vals < a) | (x_vals >= 2 - a)
    h_two_peak[mask] = u
    h_two_peak[~mask] = v
    h_two_peak = h_two_peak * (N/2) / np.sum(h_two_peak)
    
    # 3. Binary seeds (random)
    h_binary1 = np.zeros(N)
    h_binary1[:N//2] = 1
    np.random.shuffle(h_binary1)
    h_binary1 = h_binary1 * (N/2) / np.sum(h_binary1)
    
    h_binary2 = np.zeros(N)
    h_binary2[:N//2] = 1
    np.random.shuffle(h_binary2)
    h_binary2 = h_binary2 * (N/2) / np.sum(h_binary2)
    
    # 4. Basic binary skew
    h_skew = np.ones(N) * 0.1
    h_skew[:N//2] = 1.0
    h_skew = h_skew * (N/2) / np.sum(h_skew)
    
    # 5. A random continuous seed
    h_random = np.random.rand(N)
    h_random = h_random * (N/2) / np.sum(h_random)
    
    # Broadcasting across seeds:
    seeds = [h_uniform, h_two_peak, h_binary1, h_binary2, h_skew, h_random]
    gamma = [0.10, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    best_h = None
    best_c5 = float('inf')
    
    t_remaining = budget_s * 0.95
    for h_seed in seeds:
        if time.time() - start > budget_s * 0.95:
            break
            
        try:
            h_final = solve_continuation(h_seed, N, gamma)
            c5 = c5_of(h_final)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_final
        except:
            print("Continuation sol failed for seed")

    # Do a final pure subgradient polishing:
    if best_h is not None:
        th = 0.02
        t_remaining = budget_s * 0.05
        t_start = time.time()
        while time.time() - t_start < t_remaining:
            c = np.correlate(best_h, 1 - best_h, mode='full')
            max_shift = np.argmax(c)
            dx = 2.0 / N
            g = dx * (np.roll(best_h, -max_shift) - np.array([1]*max_shift + [0]*max_shift))
            g[0] = 0
            new_h = best_h - th * g
            new_h = project_to_feasible(new_h, N)
            new_c5 = c5_of(new_h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h
            else:
                th *= 0.95
    return (best_h, best_c5, N)
