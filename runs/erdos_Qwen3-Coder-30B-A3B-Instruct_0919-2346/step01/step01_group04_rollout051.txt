import numpy as np
import time
import random
from scipy import signal

def compute_c5(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    overlap = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(overlap) * dx
    return c5

def project_h(h, dx):
    """Project h to feasible set: 0 <= h[i] <= 1 and sum(h)*dx = 1"""
    # Ensure bounds
    h = np.clip(h, 0, 1)
    # Rescale to correct integral
    current_sum = np.sum(h) * dx
    if abs(current_sum) > 1e-10:
        h = h / current_sum * 0.5
    return h

def compute_grad(h, k_worst):
    """Compute gradient of C5 w.r.t. h for a specific k"""
    dx = 2.0 / len(h)
    # Build kernel for convolution: 1 - h(k) for shift k
    # Since we want to compute derivative of int h(x)(1-h(x+k)) dx
    # This is the derivative of h(x) * (1 - h(x+k))
    # With respect to h(x), we get (1 - h(x+k)) 
    # Varying h(x) affects h(x) and h(x+k) which are shifted versions
    # So for h'(x), we should compute 1 - h(x-k) for k shifts
    # For k=0, we have h'(x) = 1 - h(x) 
    # But that's for a fixed shift k
    # The derivative with respect to h(x) is a term (1 - h(x+k)) where x+k is such that we differentiate w.r.t. h(x)
    # More carefully:
    # For shift k, the integrand is h(x)(1-h(x+k))
    # So for given x, d/dh(x) of this term = (1 - h(x+k))
    # Weighted contribution is integrated over x, so gradient is just (1 - h(x-k)) where x-k is shifted properly
    # Let's take a FDM-like approximation for derivative. Using convolution with kernel 1-h[-k] (from right)
    # Actually, let's just finsitely difference a bit more carefully:
    # d/dh(x) integral h(x)(1 - h(x+k)) dx = 1 - h(x+k) (at the point where derivative is taken)
    # So gradient at h(x) = 1 - h(x+k) for shift k_{worst}
    # This is because we're taking derivative of int h(x)*(1-h(x+k)) dx w.r.t. h(x)
    
    # Compute kernel effect for k_worst
    # kernel = (1-h) shifted by k
    # for k=0, k is forward shifted 0: we use 1-h
    # for k=1, we need the kernel shifted 1 places backwards to 'match' pattern
    
    # Let's compute what this actually is more precisely:
    # We want d/dh(x) integral h(x) * (1 - h(x+k)) dx 
    # We think of this as let f(x) = h(x) * (1 - h(x+k)) 
    # But we know how to represent this through convolution?
    # Let g(x) = 1 - h(x) = h_kernel(x)  
    # Then f(x) = h(x) * g(x-k) = scipy.signal.convolve(h, np.flip(g), mode='full') but that's not quite right
    
    # The right thing is: for integrand h(x) * (1-h(x+k))
    # we differentiate w.r.t. h(x) to get 1 - h(x+k)  
    # so derivative dC5/dh(x) = 1 - h(x+k) 
    # But must be computed for the worst shift k_worst
    
    # So if k=0, we compute 1 - h(0) at positions where we take derivative
    # and generate gradient components consistent with this
    
    # Let's make the correction simply:
    # Gradient of value at position x is 1 - h(x-k_worst)  
    # Since d/dh(x) integral h(x)(1 - h(x+k)) dx = (1 - h(x+k))
    # So for k_worst we get: d/dh(x) of int h(x) (1-h(x+k_worst)) = (1 - h(x+k_worst))  
    # grad_x = 1 - h_shifted 
    # h_shifted is pre-computed h(x - k_worst) - so it's shifted appropriately
    
    # But signal.convolution reversal needs to be understood.
    # In convolution: integral f(x)g(x') dx' where g is shifted
    # Integrand is: h(x) * (1-h(x+k_worst)) is: 
    # By convolution of h with reflected (1-h) we must be careful
    # or compute point-wise gradient as: grad_x = 1 - h(x + k_worst) where I
    # I.e., we went from any position x to that same x with some k refinement?
    
    # Actually, let's write:
    # For shift k: kernel is 1-h[-k] (reversed) 
    #  
    # And git the derivatives correctly:
    # For integral from x to (x+k)  
    # The variation d/dh(x)  = (1 - h(x+k)) 
    # So gradient should be set to 1 - h(x+k) for this x 
    # Use linear indexing, say:
    # h = [h0 h1 h2 ...] 
    # 
    # if k=1, we want: [(...), 1-h2, 1-h1, 1-h0]  
    # 1-h[1] -> 1-h[0] 
    # 1-h[2] -> 1-h[1]  
    # Assume h is extended by padding:
    # Pad the array such that dimension is enough for a symmetry.
    # More directly:
    # let k = k_worst in output gradients.
    # The j-th component of gradient is where the derivative w.r.t. h[j] is computed.  
    #  
    # Thus, gradient_kj = 1 - h[j+k] != 1-h[j-k] = 1 - h_j+k 
    # (if k=0: gradient[0..n-1] = 1-h[0..n-1])  
    # Wait let's recheck: 
    # integral_{x} h(x) (1 - h(x + k_worst)) dx  
    # d/dh_j  integral h(x)(1-h(x+k)) dx = 1-h[j+k_worst] (only the place in x where the integrand probes h_ij is relevant)  
    # That's a good idea.
    
    # Therefore gradient component is: for j = 0..n-1, gradient[j] = 1 - h[(j + k_worst) % n] 
    
    # But also, stretch it back to length n:
    
    if k_worst == 0:
        grad = 1 - h  # No shift
    else:
        # The gradient at position j is 1 - h[j + k_worst]
        # To get new array of gradients: use correct indices  
        # Instead of moving by k, we need to get backward dependence
    
        grad = np.zeros_like(h)
        for i in range(len(h)-k_worst):
            grad[i] = 1 - h[i+k_worst]
        for i in range(len(h)-k_worst, len(h)):
            grad[i] = 1 - h[i-k_worst]
    return grad

def project_gradient(h, grad):
    """Project gradient to feasible set: 0 <= h <= 1"""
    # Clip h to [0,1], but then we want gradient to respect limits
    # Let's deform gradient
    h_clipped = np.clip(h, 0, 1)
    mask = (h_clipped == 0) & (grad < 0)  # Cannot decrease below 0
    grad[mask] = 0
    mask = (h_clipped == 1) & (grad > 0)  # Cannot increase above 1
    grad[mask] = 0
    return grad

def gradient_descent_optimize(h0, num_iter, step_size, beta_schedule, verbose=False):
    """Optimize h using projected gradient descent"""
    h = h0.copy()
    dx = 2.0 / len(h)
    best_c5 = float('inf')
    best_h = h.copy()
    for i in range(num_iter):
        c5 = compute_c5(h) 
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if verbose and i % 1000 == 0:
            print(f"Iter {i}: C5 = {c5}")
            
        # 1. Find the worst shift
        overlap = np.correlate(h, 1 - h, mode="full")
        k_worst_idx = np.argmax(overlap)
        full_len = len(h) * 2 - 1
        # Convert index back to shift k
        k_worst = k_worst_idx - len(h) + 1
        # Readjust range to -n+1 to n-1 
        if k_worst < 0:
            k_worst = k_worst + len(h)
        # Compute gradient
        grad = compute_grad(h, k_worst)
        # Project gradient 
        grad = project_gradient(h, grad)
        
        # Update with step size
        h = h - step_size * grad
        h = project_h(h, dx)
        
        # Schedule for step size
        beta = beta_schedule(i, num_iter)
        step_size = step_size * beta
    return best_h, best_c5

def proposal_move(h, max_step):
    """Propose a new feasible h vector by adjusting two elements"""
    h_new = h.copy()
    N = len(h)
    
    # Try to find a valid move
    max_tries = 100
    for _ in range(max_tries):
        i = random.randint(0, N - 1)
        j = random.randint(0, N - 1)
        if i == j:
            continue
            
        # Determine feasible interval for δ
        delta_low = max(-h_new[i], h_new[j] - 1)
        delta_high = min(1 - h_new[i], h_new[j])
        
        if delta_low >= delta_high:
            continue
            
        # Sample δ from uniform distribution over feasible range
        step = min(max_step, delta_high - delta_low)
        if step <= 0:
            continue
        delta = random.uniform(delta_low, delta_high)
        
        # Apply move
        h_new[i] += delta
        h_new[j] -= delta
        
        return h_new
    
    # If no valid move found, return original
    return h.copy()

def local_search(h_start, time_budget_s):
    """Perform local refinement with simulated annealing-style approach"""
    start = time.time()
    
    # Parameters
    T_start = 0.02
    T_end = 1e-8
    step_start = 0.1
    
    cur = h_start.copy()
    c_cur = compute_c5(cur)
    
    best = cur.copy()
    c_best = c_cur
    
    # Main loop
    while time.time() - start < time_budget_s:
        t = time.time() - start
        elapsed_fraction = t / time_budget_s
        
        # Cooling schedule
        T = T_start * (T_end / T_start) ** elapsed_fraction
        step = step_start * (1 - elapsed_fraction) * 0.5
        
        # Make a proposal
        h_proposed = proposal_move(cur, step)
        c_proposed = compute_c5(h_proposed)
        
        # Accept the proposal
        if c_proposed <= c_cur:
            cur = h_proposed
            c_cur = c_proposed
            if c_proposed < c_best:
                best = h_proposed.copy()
                c_best = c_proposed
        else:
            # Metropolis acceptance rule
            if random.random() < np.exp(-(c_proposed - c_cur) / T):
                cur = h_proposed
                c_cur = c_proposed
                
    return best, c_best
