import numpy as np
import time
from scipy.optimize import differential_evolution

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    overlaps = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(overlaps) * dx
    return c5

def project_to_feasible(h):
    """Project h to feasible region: 0 <= h[i] <= 1 and integral = 1"""
    h = np.clip(h, 0, 1)
    dx = 2.0 / len(h)
    scale = (len(h) / 2) / np.sum(h)
    return h * scale

def optimize_with_de(initial_h, budget_s):
    """Use differential evolution for global optimization"""
    start = time.time()
    
    # Wrapper for scipy's differential evolution
    def objective(h):
        h = project_to_feasible(h)
        return c5_of(h)
    
    # Bounds for each variable
    bounds = [(0, 1) for _ in range(len(initial_h))]
    
    # Run differential evolution
    result = differential_evolution(
        objective, 
        bounds, 
        maxiter=int(budget_s * 0.7), 
        popsize=15,
        seed=42,
        disp=False,
        atol=1e-6,
        x0=initial_h
    )
    
    best_h = project_to_feasible(result.x)
    return best_h, c5_of(best_h)
