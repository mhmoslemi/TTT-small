import numpy as np
import time
from scipy.optimize import differential_evolution

def c5_of(h):
    """Compute the maximum overlap integral for the given step function h."""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    c = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(c) * dx
    return c5

def feasibility_check(h):
    """Check if h satisfies the constraints."""
    dx = 2.0 / len(h)
    integral = np.sum(h) * dx
    return abs(integral - 1.0) < 1e-10

def objective(h):
    """Objective function to minimize."""
    return -c5_of(h)  # Negative because we want to minimize C5

def optimize_with_differential_evolution(h0, budget_s):
    """Optimize using differential evolution."""
    
    # Set bounds for each variable (0 to 1)
    bounds = [(0, 1) for _ in range(len(h0))]
    
    start = time.time()
    
    # Use differential evolution for global optimization
    result = differential_evolution(
        lambda h: -c5_of(h), 
        bounds,
        maxiter=int(1000 * budget_s / 600),
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.7,
        atol=1e-6,
        rtol=1e-6,
        seed=42,
        callback=lambda x, convergence: None,
        disp=False,
    )
    
    return result.x, -result.fun

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        N = len(initial_h_values)
        h0 = np.interp(np.linspace(0, 2, 140), 
                      np.linspace(0, 2, N), 
                      initial_h_values)
        h0 = h0 * (140/2) / np.sum(h0)  # Renormalize exactly
    else:
        N = 140
        h0 = np.ones(N) * 0.5  # fallback
    
    # Optimize using differential evolution
    h_opt, c5_bound = optimize_with_differential_evolution(h0, budget_s)
    
    # Final polishing with simulated annealing
    start = time.time()
    best_h = h_opt.copy()
    best_c5 = c5_bound
    
    T_start = 0.05
    T_end = 1e-6
    step_start = 0.1
    
    T = T_start
    step = step_start
    while time.time() - start < 0.2 * budget_s:
        # Update parameters
        elapsed = time.time() - start
        frac = elapsed / (0.2 * budget_s)
        T = T_start * (T_end / T_start) ** frac
        step = step_start * (1.0 - frac)
        
        # Propose move
        i, j = np.random.choice(N, size=2, replace=False)
        delta_low = max(-best_h[i], best_h[j] - 1)
        delta_high = min(1 - best_h[i], best_h[j])
        if delta_low >= delta_high:
            continue
        
        step_size = np.random.uniform(max(-step, delta_low),
                                      min(step, delta_high))
        
        h_new = best_h.copy()
        h_new[i] += step_size
        h_new[j] -= step_size
        
        # Ensure bounds
        h_new = np.clip(h_new, 0, 1)
        # Renormalize
        h_new = h_new * (N/2) / np.sum(h_new)
        
        c_new = c5_of(h_new)
        if c_new < best_c5:
            best_h = h_new
            best_c5 = c_new
        
        # Accept with some probability
        if c_new <= best_c5 or np.random.rand() < np.exp(-(c_new - best_c5) / T):
            best_h = h_new
            best_c5 = c_new
    
    # Final check of constraint
    dx = 2.0 / len(best_h)
    integral_check = np.sum(best_h) * dx
    if abs(integral_check - 1.0) > 1e-8:
        best_h = best_h * (1.0/integral_check)
    
    return (best_h, best_c5, len(best_h))
