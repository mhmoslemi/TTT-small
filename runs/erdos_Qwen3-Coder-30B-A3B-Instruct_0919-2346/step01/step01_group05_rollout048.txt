import numpy as np
import time
from math import sqrt

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value scaled by dx
    max_overlap = np.max(correlation) * dx
    return max_overlap

def generate_sturmian_binary(n):
    """Generate a balanced binary vector using irrational rotation with golden ratio"""
    phi = (sqrt(5) - 1) / 2  # Golden ratio conjugate
    h = np.zeros(n, dtype=int)
    count_ones = 0
    
    for i in range(n):
        t = (i * phi) % 1
        if t < 0.5:
            h[i] = 1
            count_ones += 1
    
    # Adjust to get exactly n/2 ones
    target = n // 2
    if count_ones > target:
        # Need to flip some 1s to 0s
        ones_indices = np.where(h == 1)[0]
        flip_indices = ones_indices[:count_ones - target]
        h[flip_indices] = 0
    elif count_ones < target:
        # Need to flip some 0s to 1s
        zeros_indices = np.where(h == 0)[0]
        flip_indices = zeros_indices[:target - count_ones]
        h[flip_indices] = 1
        
    return h

def compute_overlaps(h):
    """Compute overlaps for all shifts, returning array with dx already included"""
    n = len(h)
    dx = 2.0 / n
    correlation = np.correlate(h, 1 - h, mode="full")
    return correlation * dx

def get_worst_shift(overlaps):
    return np.argmax(overlaps)

def compute_delta_shifts(h, i, j, overlaps):
    """
    Compute the change in each overlap value due to swapping h[i] with h[j].
    Returns the updated overlaps after swapping.
    """
    n = len(h)
    new_overlaps = overlaps.copy()
    # The contribution of position i to overlap s is h[i] * (1 - h[i+s])
    # For swapping the value h[i] we compute changes for all s:
    for s in range(1 - n, n):
        # For shift s, positions with contribution: i, i-s, or -i+s
        # The change depends on how h[s+i] or h[i-s] may be affected...
        
        # Position i as first factor in h[i] * (1 - h[i+s])
        # Before swap: change contribution by - (1 - h[i+s])
        # After swap: change contribution by + (1 - h[i+s])  (since h[i] now 0)
        delta = -(1 - h[i+s]) if i+s < n and i+s >= 0 else 0
        # Same factor (but as second factor), if applicable:
        # h[k] * (1 - h[i])  where k = i-s, contributes:
        if i-s >= 0 and i-s < n:
            delta += h[i-s]
            
        # Similarly for j:
        # Position j as first factor: h[j] * (1 - h[j+s]) 
        delta += (1 - h[j+s]) if j+s < n and j+s >= 0 else 0
        # Second factor: if k = j-s in range, adds h[j-s]
        if j-s >= 0 and j-s < n:
            delta -= h[j-s]
        
        if i+s < n and i+s >= 0:
            new_overlaps[s+n-1] += delta
        # This covers all relevant overlaps.

    return new_overlaps

# Optimized version of previous computation:
def compute_delta_shifts_fast(h, i, j, current_overlaps):
    """
    Fast version to compute change in all shifts when swapping h[i] with h[j]
    """
    n = len(h)
    deltas = np.zeros(2*n - 1)
    
    # Argument positions for fast lookup
    for s in range(-n+1, n):  # range of all shifts - same as full mode
        idx_i = i + s
        idx_j = j + s
        
        if 0 <= idx_i < n:
            deltas[s+n-1] -= (1 - h[idx_i])  # From h[i] term
            
        if 0 <= idx_j < n:
            deltas[s+n-1] += (1 - h[idx_j])  # From h[j] term
            
        if 0 <= i - s < n:  # This is the second factor for i due to h[i]* (1-h[i+s])
            deltas[s+n-1] += h[i - s]  # From h[i-s] term
            
        if 0 <= j - s < n:  # Same reverse for j  
            deltas[s+n-1] -= h[j - s]  # From h[j-s] term

    return current_overlaps + deltas

def local_swap_search(h, max_time):
    """Improve h via targeted swap moves"""
    start = time.time()
    n = len(h)
    dx = 2.0 / n
    
    # Compute initial overlaps and worst shift
    overlaps = compute_overlaps(h)
    max_shift = get_worst_shift(overlaps)  # index in -n+1 to n-1 range 
    current_c5 = overlaps[max_shift]
    
    # Tabu list (keep track of swaps recently done)
    tabu = set() 
    tabu_size = int(0.1 * n)  # 10% of n for small tabu list
    
    iters = 0
    stuck_iters = 0
    max_stuck = 5000
    
    # Find all 1 positions and 0 positions for fast random sampling
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    # Begin swaps
    while time.time() - start < max_time:
        iters += 1
        if stuck_iters > max_stuck:
            # Perturb a bit to get out of local minima
            # Flip ~5% of bits, trying to maintain balance
            k = int(0.05 * n)
            indices = np.random.choice(n, k, replace=False)
            new_h = h.copy()
            for idx in indices:
                new_h[idx] = 1 - new_h[idx]
            # Adjust to get correct count
            count_ones = np.sum(new_h)
            diff = int(count_ones - n/2)
            if diff > 0:
                # Too many 1s, flip some 1s to 0s
                flip_idx = np.where(new_h == 1)[0][:diff]
                new_h[flip_idx] = 0
            elif diff < 0: 
                flip_idx = np.where(new_h == 0)[0][:abs(diff)]
                new_h[flip_idx] = 1
            h = new_h
            overlaps = compute_overlaps(h)
            max_shift = get_worst_shift(overlaps)
            current_c5 = overlaps[max_shift]
            stuck_iters = 0
            
        # Pick random 1 position and random 0 position 
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
            
        # Skip if in tabu
        if (i, j) in tabu or (j, i) in tabu:
            if np.random.rand() < 0.1:  # 10% chance to ignore tabu
                pass
            else:
                continue
        
        # Compute change in overlaps if we swap
        new_overlaps = compute_delta_shifts_fast(h, i, j, overlaps)
        new_max_shift = get_worst_shift(new_overlaps)
        new_c5 = new_overlaps[new_max_shift]
        
        improvement = current_c5 - new_c5
        
        # Accept if improving, or sometimes accept poor moves
        accept = improvement >= 0.0 or np.random.rand() < 0.005
        
        if accept and new_c5 < current_c5:
            # Do actual swap
            h[i], h[j] = h[j], h[i]
            
            # Update
            overlaps = new_overlaps
            current_c5 = new_c5
            max_shift = new_max_shift
            
            # Update one/zero indices
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            
            # Update tabu list
            if len(tabu) >= tabu_size:
                tabu.pop()
            tabu.add((i,j))
            
            stuck_iters = 0
        else:
            # Small chance to proceed even with no improvement
            if np.random.rand() < 0.0001:
                # Still might want to accept even if not very good
                # This helps with escape from local minima
                h[i], h[j] = h[j], h[i]
                overlaps = new_overlaps
                current_c5 = new_c5
                max_shift = new_max_shift
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                if len(tabu) >= tabu_size:
                    tabu.pop()
                tabu.add((i,j))
                stuck_iters = 0
            else:
                stuck_iters += 1
            
        # Occasionally, now and then, check improvements for
        # better energetic criteria
        
        # For verbose tracking, uncomment:
        # if iters % 1000 == 0: 
        #     print(f"{iters}: Max C5 = {current_c5:.5f}")
        
    return h, current_c5
    
def continuous_polish(h, num_iters=100):
    """Apply PgD to refine fractional values"""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    
    for iter in range(num_iters): 
        overlaps = compute_overlaps(h)
        max_shift = get_worst_shift(overlaps)
        
        # Take gradient w.r.t. to h components
        # g_s(h) = sum_i h_i * (1 - h_{i+s})
        # For a fixed s, d/dh_i = (1 - h_{i+s})
        # but there are many such terms, so we're interested
        # in overall gradient of max_s g_s(h), using prob hit s
        
        # However for continuous variables, we can directly use
        # an improved rough gradient estimate.
        # Compute partial derivatives: we focus on the current max shift.
        # Let us compute gradient g_s for a fixed s. 
        # Each contribution i to g_s is h_i (1 - h_{i+s})
        # So we care about d/dh_i (h_i * (1 - h_{i+s}))  
        # Which is 1 - h_{i+s}, for i in support of s overlap.
        # This tends to be biased toward pushing h toward balance.
        
        # We will solve a local optimization step:
        # dm = Σ_i [1 - h_{i+s}] dx_i
        # where dx_i is push towards reducing max_s g_s
        
        # MS: to quickly find good update:
        grad = np.zeros_like(h)
        s = max_shift - n + 1 # map to actual shift index
        
        # Now compute for shift s:
        #   fx = 1 - h_{i+s} if 0 <= i+s < n else 0  
        for i in range(n):
            pos = i + s
            if 0 <= pos < n: 
                grad[i] += 1 - h[pos]
                
        # Now update step:
        # To reduce max slide, we move toward reducing max shift gap
        step = 0.03 * grad
        h -= step
        h = np.clip(h, 0, 1)
    
    # Project back to sum constraint
    actual_sum = np.sum(h) * dx
    h = h * (1.0 / actual_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    min_time = 0.1  # minimum time per phase
    
    # Start with smaller resolution to explore fast
    n0 = 120
    n1 = 240  # upsample for final polishing  
    n2 = 480
    smallest_n = n0
    
    # Create start binary vector
    h = generate_sturmian_binary(n0)
    c5_best = evaluate_erdos_solution(h)
    
    # Phase 1: local search on low res
    max_search_time = budget_s * 0.6
    h1, c51 = local_swap_search(h, max_search_time) 
    
    # Track best solution so far
    if c51 < c5_best:
        h_best = h1.copy()
        c5_best = c51
        smallest_n = n0
        
    # Phase 2: upscale and refine 
    if budget_s > 0.9 * min_time:
        h_upscaled = np.repeat(h1, 2)  # duplicate each element to get n1
        if len(h_upscaled) == n1 + 1:
            h_upscaled = h_upscaled[:-1]
        # Final refinement on up-scaled pattern  
        h2, c52 = local_swap_search(h_upscaled, max_search_time * 0.7)
        if c52 < c5_best:
            h_best = h2.copy() 
            c5_best = c52
            smallest_n = n1
            
    # Phase 3: continuous polish if needed
    try:
        h_final = continuous_polish(h_best)
        c5_final = evaluate_erdos_solution(h_final)
        if c5_final < c5_best:
            h_best = h_final
            c5_best = c5_final
    except:
        pass
    
    return (h_best.tolist(), c5_best, len(h_best))
