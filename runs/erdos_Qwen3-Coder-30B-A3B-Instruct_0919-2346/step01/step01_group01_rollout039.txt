def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using full convolution
    overlap = np.correlate(h, 1 - h, mode='full')
    # The maximum value scaled by dx
    c5 = np.max(overlap) * dx
    return c5

def propose_move(h, max_step):
    """Propose a move that maintains sum constraint exactly"""
    h_new = h.copy()
    N = len(h)
    while True:
        i, j = random.sample(range(N), 2)
        # Compute feasible range for delta
        delta_low = max(-h_new[i], h_new[j] - 1.0)
        delta_high = min(1.0 - h_new[i], h_new[j])
        if delta_low >= delta_high:
            continue  # No valid move, try again
        break
    
    # Sample delta from feasible interval
    step = min(max_step, delta_high - delta_low)
    if step <= 0:
        delta = 0.0
    else:
        delta = random.uniform(max(-step, delta_low), min(step, delta_high))
        
    h_new[i] += delta
    h_new[j] -= delta
    return h_new

def project_to_constraint(h):
    """Project h to satisfy sum constraint exactly"""
    N = len(h)
    dx = 2.0 / N
    current_sum = np.sum(h) * dx
    target_sum = 1.0
    h = h * (target_sum / current_sum)
    # Ensure bounds
    h = np.clip(h, 0, 1)
    return h

    T_end = 1e-8

    decay_rate = 0.95
    
    iteration = 0

        elapsed = time.time() - start

            c_cur = c_prop
            
        iteration += 1
            
    return best, c_best

        if x < peak_width or x > 2 - peak_width:
            h[i] = peak_height
    # Normalize to exact sum
    h = h * (n_points / 2) / np.sum(h)
    return h
