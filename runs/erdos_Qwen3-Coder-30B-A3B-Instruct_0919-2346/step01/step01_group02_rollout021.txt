import numpy as np
import time

def c5_of(h):
    """Compute the maximum overlap integral for the given step function h."""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    c = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(c) * dx
    return c5

def compute_gradient(h, shift):
    """Compute gradient of C5 for the given shift."""
    N = len(h)
    dx = 2.0 / N
    g = np.full(N, 1.0) * dx
    if shift >= 0:
        if shift < N:
            # subtract right shift
            g[:N-shift] -= h[shift:] * dx
            # subtract left shift
            g[shift:] -= h[:N-shift] * dx
    else:
        shift = -shift
        if shift < N:
            g[shift:] -= h[:N-shift] * dx
            g[:N-shift] -= h[shift:] * dx
    return g

def project_onto_bounds_and_sum(v, target):
    """Project vector v onto the feasible region: 0 <= h <= 1, sum = target."""
    # Binary search for lambda
    lower = -np.max(v)
    upper = 1 - np.min(v)
    for _ in range(60):
        mid = (lower + upper) / 2.0
        w = np.clip(v + mid, 0, 1)
        s = np.sum(w)
        if s < target:
            lower = mid
        else:
            upper = mid
    w = np.clip(v + (lower + upper) / 2.0, 0, 1)
    
    # Correct any small numerical errors
    diff = target - np.sum(w)
    if abs(diff) > 1e-12:
        free_idx = (w > 0) & (w < 1)
        if np.any(free_idx):
            w[free_idx] += diff / np.sum(free_idx)
            w = np.clip(w, 0, 1)
        else:
            w += diff / len(w)
            w = np.clip(w, 0, 1)
    return w

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    N = 140
    dx = 2.0 / N
    
    # Create initial seed
    if 'initial_h_values' in globals():
        h0 = np.interp(np.linspace(0, 2, N), 
                      np.linspace(0, 2, len(initial_h_values)), 
                      initial_h_values)
        h0 = h0 * (N/2) / np.sum(h0)  # Renormalize exactly
    else:
        h0 = np.ones(N) * 0.5  # fallback

    # Two-peak seed 
    h1 = np.ones(N)
    x_vals = np.linspace(0, 2, N)
    a = 0.3
    u = 0.9
    v = 0.1
    mask = (x_vals < a) | (x_vals >= 2 - a)
    h1[mask] = u
    h1[~mask] = v
    h1 = h1 * (N/2) / np.sum(h1)  # Renormalize

    # Uniform
    h2 = np.ones(N) * 0.5

    # Set of initial solutions 
    initial_solutions = [h0, h1, h2]
    
    # Try few random binary seeds
    for _ in range(2):
        h_bin = np.zeros(N)
        idx = np.random.choice(N, N//2, replace=False)
        h_bin[idx] = 1.0
        initial_solutions.append(h_bin)
    
    best_h = None
    best_c5 = float('inf')
    
    # Find best among initial solutions
    for h in initial_solutions:
        c5 = c5_of(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Main optimization loop
    start = time.time()
    time_left = budget_s
    
    while time_left > 0.1:
        # Run gradient descent for a fixed number of steps
        cur_h = best_h.copy()
        cur_c5 = best_c5
        
        iterations = 0
        max_iter = 5000
        
        while iterations < max_iter and time.time() - start < budget_s - 0.1:
            # Compute correlation and worst shift
            corr = np.correlate(cur_h, 1 - cur_h, mode='full')
            max_idx = np.argmax(corr)
            s = max_idx - (N-1)
            
            # Compute gradient
            grad = compute_gradient(cur_h, s)
            
            # Adaptive step size
            t = (time.time() - start) / budget_s
            step = 20.0 * (1 - t) + 0.1
            
            # Update
            v = cur_h - step * grad
            cur_h = project_onto_bounds_and_sum(v, N/2)
            
            # Evaluate
            cur_c5 = c5_of(cur_h)
            
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()
            
            iterations += 1
            
        # Occasionally jump to a new random solution
        if np.random.rand() < 0.3:
            h_new = np.random.rand(N)
            h_new = h_new * (N/2) / np.sum(h_new)
            best_h = h_new        
            best_c5 = c5_of(best_h)
            
        time_left = budget_s - (time.time() - start)
    
    # Final polishing
    start = time.time()
    while time.time() - start < 0.1 * budget_s:
        i, j = np.random.choice(N, size=2, replace=False)
        delta_low = max(-best_h[i], best_h[j] - 1)
        delta_high = min(1 - best_h[i], best_h[j])
        if delta_low >= delta_high:
            continue
        step_size = np.random.uniform(0.01, 0.1)
        delta = np.random.uniform(-step_size, step_size)
        delta = np.clip(delta, delta_low, delta_high)
        best_h[i] += delta
        best_h[j] -= delta
        new_c5 = c5_of(best_h)
        if new_c5 < best_c5:
            best_c5 = new_c5
        else:
            # Revert if worse
            best_h[i] -= delta
            best_h[j] += delta
    
    return (best_h, best_c5, N)
