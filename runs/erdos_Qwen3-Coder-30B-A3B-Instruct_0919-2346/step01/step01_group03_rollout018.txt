import numpy as np
import time

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral via full convolution of h with (1-h)
    overlaps = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum of the full correlation vector
    # scaled by dx
    c5 = np.max(overlaps) * dx
    return c5

def full_overlap(h):
    """Compute the raw correlation vector (unscaled)"""
    return np.correlate(h, 1 - h, mode="full")

def project_onto_P(v, total):
    """Project vector v onto the set {x : 0 <= x_i <= 1, sum x_i = total}"""
    # Using bisection to find lambda
    def phi(lam):
        # Clamp and sum
        x = np.clip(v - lam, 0, 1)
        return np.sum(x) - total
    
    # Initial bounds for lambda
    lb = np.min(v)
    ub = np.max(v) + 1
    
    # Bisection
    for _ in range(50):
        lam = (lb + ub) / 2
        val = phi(lam)
        if val > 0:
            ub = lam
        else:
            lb = lam
            
    # Get final projection
    x = np.clip(v - lam, 0, 1)
    return x

def smooth_gradient(h, beta):
    """Compute the gradient of Phi_beta(h)"""
    # Get the full correlation
    F = full_overlap(h)
    # Stability: subtract max before exponentiating
    F_shifted = F - np.max(F)
    expF = np.exp(beta * F_shifted)
    w = expF / np.sum(expF)  # Softmax weights
    
    # Compute gradient
    g = np.ones_like(h)
    for k in range(len(F)):
        wk = w[k]
        shift = k - (len(h) - 1)
        if shift >= 0:
            # h[:N-k] -= wk * h[k:]
            g[:len(h)-shift] -= wk * h[shift:]
            # h[k:] -= wk * h[:N-k]
            g[shift:] -= wk * h[:len(h)-shift]
        else:
            # h[shift:] -= wk * h[:N-shift]
            g[-shift:] -= wk * h[:-shift]
            # h[:N-shift] -= wk * h[shift:]
            g[:-shift] -= wk * h[-shift:]
    
    return g

def projected_gradient_descent(h0, time_budget, seed):
    """Run projected gradient descent with β-schedule from h0"""
    np.random.seed(seed)
    start = time.time()
    
    # Parameters
    β_start = 2.0
    β_max = 30.0
    γ_step = 0.9995
    γ_beta = 1.5
    β_update_interval = 50
    step_start = 0.5
    max_iter = int(1e6)
    
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of(h)
    
    β = β_start
    step = step_start
    last_improvement = 0
    
    for iteration in range(max_iter):
        if time.time() - start > time_budget:
            break
            
        # Evaluate and compute weights
        F = full_overlap(h)
        F_shifted = F - np.max(F)
        expF = np.exp(β * F_shifted)
        w = expF / np.sum(expF)
        
        # Compute gradient
        g = smooth_gradient(h, β)
        
        # Update
        v = h - step * g
        h = project_onto_P(v, len(h) / 2)
        
        # Evaluate true objective
        cur_c5 = c5_of(h)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
            
        # Schedule updates
        step *= γ_step
        if iteration % β_update_interval == 0 and β < β_max:
            β = min(β * γ_beta, β_max)
            
        last_improvement += 1
        if last_improvement > 1000:
            step = max(step * 0.5, 1e-4)
            last_improvement = 0
            
    return best_h, best_c5

def targeted_refinement(h, time_budget):
    """Refine h using greedy pairwise transfers to reduce max overlap"""
    np.random.seed(42)
    start = time.time()
    
    h = h.copy()
    best_h = h.copy()
    best_c5 = c5_of(h)
    
    alpha = 1.0
    attempts = 0
    max_attempts = 200
    
    while time.time() - start < time_budget:
        # Get correlation and find worst shift
        C = full_overlap(h)
        d = np.argmax(C) - (len(h) - 1)
        # Extract overlapping slices
        if d >= 0:
            hi = h[:len(h)-d]
            hj = h[d:]
        else:
            dd = -d
            hi = h[dd:]
            hj = h[:len(h)-dd]
            
        # Compute overlaps
        c = hi * (1 - hj)
        idx = np.argmax(c)
        c_max = c[idx]
        
        # Choose transfer amount
        Δ = min(alpha * c_max, h[idx], 1 - hj[idx % len(hj)])
        
        # Apply move
        h[idx] -= Δ
        h[idx % len(hj)] += Δ
        
        # Evaluate
        new_c5 = c5_of(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            attempts = 0
            alpha = 1.0  # reset
        else:
            # Revert change
            h[idx] += Δ
            h[idx % len(hj)] -= Δ
            attempts += 1
            if attempts >= max_attempts:
                alpha *= 0.5
                if alpha < 1e-8:
                    break
                attempts = 0  # Reset
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start = time.time()
    N = 200  # Slightly higher resolution for better improvement
    dx = 2.0 / N
    
    # Generate four diverse seeds
    # 1. Uniform seed
    h_uniform = 0.5 * np.ones(N)
    h_uniform *= (N / 2) / np.sum(h_uniform)
    
    # 2. Interpolated from initial_h_values (if exists)
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        if len(h0) != N:
            x0 = np.linspace(0, 2, len(h0))
            x_new = np.linspace(0, 2, N)
            h0 = np.interp(x_new, x0, h0)
        h_initial = h0 * (N / 2) / np.sum(h0)
    else:
        h_initial = np.random.rand(N) * 0.5
        h_initial *= (N / 2) / np.sum(h_initial)
        
    # 3. Double-peak seed
    h_double_peak = np.full(N, 0.1)
    a = 0.3
    idxs = np.where((np.linspace(0, 2, N) < a) | (np.linspace(0, 2, N) > (2 - a)))[0]
    h_double_peak[idxs] = 0.9
    h_double_peak *= (N / 2) / np.sum(h_double_peak)
        
    # 4. Random Dirichlet (Beta(0.5, 0.5) sample, then normalize)
    h_dirichlet = np.random.beta(0.5, 0.5, N)
    h_dirichlet *= (N / 2) / np.sum(h_dirichlet)
    
    # Run optimization on seeds
    best_h = None
    best_c5 = float('inf')
    
    seeds = [h_uniform, h_initial, h_double_peak, h_dirichlet]
    budget_per_seed = 0.3 * budget_s / len(seeds)
    
    for i, h_seed in enumerate(seeds):
        if time.time() - start > 0.9 * budget_s:
            break
        h_opt, c5_opt = projected_gradient_descent(h_seed, budget_per_seed, seed+i)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt

    # Polishing refinement
    if best_h is not None and time.time() - start < 0.95 * budget_s:
        refine_budget = budget_s - (time.time() - start)
        polished_h, polished_c5 = targeted_refinement(best_h, refine_budget)
        if polished_c5 < best_c5:
            best_h = polished_h
            best_c5 = polished_c5

    return (best_h, best_c5, N)
