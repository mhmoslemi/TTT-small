```python
import numpy as np
import time
import cvxpy as cp
from cvxpy import ECOS, GLPK, GLPK_MI
import math

def water_fill(h, target_sum):
    """Project h onto the simplex with sum = target_sum using binary search for lambda."""
    h = np.array(h)
    low, high = -1.0, 2.0
    while high - low > 1e-10:
        lam = (low + high) / 2
        s = np.sum(np.clip(h + lam, 0, 1))
        if s >= target_sum:
            low = lam
        else:
            high = lam
    lam = (low + high) / 2
    return np.clip(h + lam, 0, 1)

def c5_of(h):
    """Compute the maximum overlap integral for the given step function h."""
    dx = 2.0 / len(h)
    c = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(c) * dx
    return c5

def build_mccormick_constraints(h, y, pairs, constraints_list):
    """Add McCormick constraints for each pair."""
    for i, s in pairs:
        # y[i,s] <= h[i]
        constraints_list.append(y[i] <= h[i])
        # y[i,s] <= h[i+s]  
        constraints_list.append(y[i] <= h[i+s])
        # y[i,s] >= h[i] + h[i+s] - 1
        constraints_list.append(y[i] >= h[i] + h[i+s] - 1)
        # y[i,s] >= 0
        constraints_list.append(y[i] >= 0)

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    N = 140
    dx = 2.0 / N
    
    # Build index pairs for all valid (i,s) such that i+s < N
    pairs = []
    shift_to_pair_idx = {}
    pair_idx_to_shift = []
    
    idx = 0
    for s in range(N):  # shifts from 0 to N-1
        for i in range(N - s):
            pairs.append((i, s))
            shift_to_pair_idx[(i, s)] = idx
            pair_idx_to_shift.append(s)
            idx += 1
    
    # LP (Relaxation)
    print("Solving LP relaxation...")
    h = cp.Variable(N, name='h')
    y = cp.Variable(len(pairs), name='y')
    alpha = cp.Variable(name='alpha')
    
    # Create constraints list
    constraints = []
    
    # Constraints: 0 <= h[i] <= 1
    for i in range(N):
        constraints.append(h[i] >= 0)
        constraints.append(h[i] <= 1)
        
    # Constraint: sum h = N/2
    constraints.append(cp.sum(h) == N/2)
    
    # McCormick constraints
    build_mccormick_constraints(h, y, pairs, constraints)
    
    # Alpha constraints for each shift
    # We compute the sum of y[i,s] over all valid i for each shift
    # and enforce alpha >= dx*(N/2 - sum_y_s) = dx*(N/2 - Σ_{i,s} y[i,s])
    # That's same as sum of sums over i for each shift s
    
    # Precompute for each shift s, the list of pairs (i,s)
    shift_pair_indices = {}
    for s in range(N):
        shift_pair_indices[s] = []
    for i, s in pairs:
        shift_pair_indices[s].append(i)
    
    # Add alpha constraints per shift
    for s in range(N):
        idx_list = shift_pair_indices[s]
        if len(idx_list) > 0:
            # Compute sum_y = sum_{i : (i,s) valid} y[i,s]
            # Each triplet (i,s) is at index shift_to_pair_idx[(i,s)]
            # So we sum over all index_list
            sum_y = cp.sum([y[shift_to_pair_idx[(i,s)]] for i in idx_list])
            # Enforce: alpha >= dx * (N/2 - sum_y)
            constraints.append(alpha >= dx * (N/2 - sum_y))
    
    # Objective
    objective = cp.Minimize(alpha)
    
    # Create and solve the problem
    prob = cp.Problem(objective, constraints)
    time_spent = time.time() - start_time
    solve_time = min(0.1 * budget_s, budget_s - time_spent - 5)
    if solve_time <= 0:
        return (np.ones(N)*0.5, 1.0, N)
        
    try:
        prob.solve(solver=ECOS, max_iters=10000)
    except Exception as e:
        print(f"LP solve failed: {e}")
        hp = np.ones(N) * 0.5
        hp = water_fill(hp, N/2)
        return (hp, c5_of(hp), N)

    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        h_lp_val = h.value
        alpha_lp = alpha.value
        print(f"LP solved, alpha = {alpha_lp:.6f}")
        # Compute true C5 from LP solution
        c5_lp = c5_of(h_lp_val)
        print(f"True C5 from LP = {c5_lp:.6f}")
        if c5_lp <= 0.3807:
            return (h_lp_val, c5_lp, N)
        
        # Continue to MILP if needed
        print("Attempting MILP...")
        # Convert to binary MILP
        h_bin = cp.Variable(N, boolean=True, name='h_bin')
        y_bin = cp.Variable(len(pairs), name='y_bin')
        alpha_milp = cp.Variable(name='alpha_milp')
        
        constraints_milp = []
        # Bounds
        for i in range(N):
            constraints_milp.append(h_bin[i] >= 0)
            constraints_milp.append(h_bin[i] <= 1)
        # Integer
        # Sum
        constraints_milp.append(cp.sum(h_bin) == N/2)
        
        # McCormick
        build_mccormick_constraints(h_bin, y_bin, pairs, constraints_milp)
        
        # Alpha constraints
        for s in range(N):
            idx_list = shift_pair_indices[s]
            if len(idx_list) > 0:
                sum_y = cp.sum([y_bin[shift_to_pair_idx[(i,s)]] for i in idx_list])
                constraints_milp.append(alpha_milp >= dx * (N/2 - sum_y))
        
        # Objective
        objective_milp = cp.Minimize(alpha_milp)
        prob_milp = cp.Problem(objective_milp, constraints_milp)
        
        # Solving MILP with warm start and limited time
        time_spent = time.time() - start_time
        milp_time = min(0.45 * budget_s, budget_s - time_spent - 5)
        if milp_time <= 0:
            h_bin_val = h_lp_val.copy()
            h_bin_val = np.round(h_bin_val)
            h_bin_val = water_fill(h_bin_val, N/2)
            return (h_bin_val, c5_of(h_bin_val), N)
            
        # Set solver parameters with max time
        try:
            prob_milp.solve(solver=GLPK_MI, maxTime=milp_time, msg=0)
        except Exception as e:
            print(f"MILP solve failed: {e}")
            
        if prob_milp.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            print("MILP solved successfully")
            h_milp_val = h_bin.value
            if h_milp_val is not None:
                c5_milp = c5_of(h_milp_val)
                if c5_milp <= 0.3807:
                    return (h_milp_val, c5_milp, N)
        
        # Fallback: Greedy rounding + hill-climb
        print("Falling back to rounding and refinement")
        # Round to binary
        h_round = np.round(h_lp_val)
        h_round = water_fill(h_round, N/2)  # renormalize with water fill
        best_h = h_round
        best_c5 = c5_of(best_h)
        
        # Local refinement with hill-climb
        max_refinement_time = min(0.2 * budget_s, budget_s - (time.time() - start_time) - 5)
        if max_refinement_time > 0:
            refine_start = time.time()
            while time.time() - refine_start < max_refinement_time:
                # Find worst shift
                c_vals = np.correlate(best_h, 1 - best_h, mode='full')
                worst_shift_idx = np.argmax(c_vals)
                worst_shift = worst_shift_idx - N + 1
                
                # Compute gradient via forward differences
                grad = np.zeros(N)
                for i in range(N):
                    if i + worst_shift >= 0 and i + worst_shift < N:
                        grad[i] = best_h[i + worst_shift] - best_h[i]
                    else:
                        grad[i] = 0
                        
                # Update in negative gradient direction with small step
                step_size = 0.02
                new_h = best_h - step_size * grad
                new_h = np.clip(new_h, 0, 1)
                new_h = water_fill(new_h, N/2)
                
                new_c5 = c5_of(new_h)
                if new_c5 < best_c5:
                    best_h = new_h
                    best_c5 = new_c5
                    if best_c5 <= 0.3807:
                        break
                        
        return (best_h, best_c5, N)
        
    else:
        print(f"LP status: {prob.status}")
        hp = np.ones(N) * 0.5  
        hp = water_fill(hp, N/2)
        return (hp, c5_of(hp), N)
        
    # One last fallback to original method if needed  
    hp = np.ones(N) * 0.5  
    hp = water_fill(hp, N/2)
    return (hp, c5_of(hp), N)
```
