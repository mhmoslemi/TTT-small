import numpy as np
import time

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral via full convolution of h with (1-h)
    overlaps = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum of the full correlation vector
    # scaled by dx
    c5 = np.max(overlaps) * dx
    return c5

def full_overlap(h):
    """Compute raw correlation vector"""
    return np.correlate(h, 1 - h, mode="full")

def project_onto_P(v, total):
    """Project onto box + sum constraint using bisection"""
    # Clip to [0,1] then adjust
    v = np.clip(v, 0, 1)
    s = np.sum(v)
    if abs(s - total) < 1e-12:
        return v
    # Bisection for lambda
    if s > total:
        a, b = 0, 1
    else:
        a, b = -1, 0
    for _ in range(50):  # bisection iterations
        lam = (a + b) / 2
        x = np.clip(v - lam, 0, 1)
        s_x = np.sum(x)
        if s_x > total:
            a = lam
        else:
            b = lam
    lam = (a + b) / 2
    x = np.clip(v - lam, 0, 1)
    return x

def smooth_gradient(h, beta):
    """Compute gradient of smooth surrogate Φ_β(h)"""
    # Compute overlaps F_k = N/2 - S_k, where S_k = sum h_i * h_{i+k}
    N = len(h)
    F = full_overlap(h)
    
    # Softmax weights w_k
    F_centered = F - np.max(F)
    exp_F = np.exp(beta * F_centered)
    w = exp_F / np.sum(exp_F)
    
    # Compute gradient ∇Φ_β = sum_k w_k (1 - h_{i+k} - h_{i-k})
    g = np.ones(N)
    for k in range(len(w)):
        wk = w[k]
        if k < N:  # k >= 0
            g[:N-k] -= wk * h[k:]
        else:  # k < 0
            d = k - N + 1
            g[d:] -= wk * h[:N+d]
    return g

def projected_gradient_descent(h0, time_budget, seed):
    """Main smoothed gradient descent optimizer"""
    np.random.seed(seed)
    start = time.time()
    
    h = h0.copy()
    best_h = h.copy()
    c_best = c5_of(h)
    
    # Parameters
    beta = 2.0
    beta_max = 30.0
    beta_update_interval = 200
    gamma_beta = 1.5
    gamma_step = 0.9995
    step = 0.5
    max_iter = int(1e6)
    N = len(h)
    total = N / 2
    
    for iter in range(max_iter):
        if time.time() - start > time_budget:
            break
            
        # Evaluate smooth surrogate
        F = full_overlap(h)
        F_centered = F - np.max(F)
        exp_F = np.exp(beta * F_centered)
        w = exp_F / np.sum(exp_F)
        
        # Compute gradient
        g = np.ones(N)
        for k in range(len(w)):
            wk = w[k]
            if k <= N - 1:  # k >= 0
                if k == 0:
                    g[:N] -= wk * h[:N]
                else:
                    g[:N-k] -= wk * h[k:]
                    g[k:] -= wk * h[:N-k]
            else:  # k < 0
                d = k - N + 1
                g[d:] -= wk * h[:N+d]
                g[:N+d] -= wk * h[d:]
        
        # Tentative update and projection
        v = h - step * g
        h = project_onto_P(v, total)
        
        # Evaluate true objective
        c = c5_of(h)
        if c < c_best:
            c_best = c
            best_h = h.copy()
            
        # Schedule updates
        step *= gamma_step
        if iter % beta_update_interval == 0 and beta < beta_max:
            beta = min(beta * gamma_beta, beta_max)
            
        # Check for stagnation
        if iter % 1000 == 0 and iter > 0:
            if abs(c - c_best) < 1e-8:
                step = max(step * 0.5, 1e-4)
                beta = min(beta * 1.2, beta_max)
    
    return best_h, c_best

def targeted_refinement(h0, time_budget):
    """Deterministic pairwise refinement"""
    start = time.time()
    h = h0.copy()
    c = c5_of(h)
    alpha = 1.0
    max_iters = 10000
    patience = 200
    no_improvement_count = 0
    
    for i in range(max_iters):
        if time.time() - start > time_budget:
            break
        
        # Evaluate all correlations
        C = full_overlap(h)
        d = np.argmax(C) - (len(h) - 1)
        N = len(h)
        
        # Extract overlapping parts
        if d >= 0:
            hi = h[:N-d]
            hj = h[d:]
        else:
            hi = h[-d:]
            hj = h[:N-(-d)]
        
        # Compute contributions
        c_contrib = hi * (1 - hj)
        
        # Find worst pair
        idx = np.argmax(c_contrib)
        if d >= 0:
            i_idx, j_idx = idx, idx + d
        else:
            i_idx, j_idx = idx - d, idx
        
        # Compute move amount
        Delta = min(alpha * c_contrib[idx], h[i_idx], 1 - h[j_idx])
        
        if Delta > 1e-10:
            # Apply transfer
            h[i_idx] -= Delta
            h[j_idx] += Delta
            
            # Evaluate new C5
            c_new = c5_of(h)
            if c_new < c:
                c = c_new
                no_improvement_count = 0
            else:
                # Revert
                h[i_idx] += Delta
                h[j_idx] -= Delta
                alpha *= 0.5
                no_improvement_count += 1
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= patience:
            break
            
    return h, c

    h_double_peak[idxs] = 0.9

    h_dirichlet *= (N / 2) / np.sum(h_dirichlet)

    # Run projected gradient and refine on seeds
    best_h = None
    best_c5 = float('inf')
    
    seeds = [h_uniform, h_initial, h_double_peak, h_dirichlet]
    budget_per_seed = 0.3 * budget_s / len(seeds)
    
    for i, h_seed in enumerate(seeds):

        h, c5 = projected_gradient_descent(h_seed, budget_per_seed, seed + i)

            best_c5 = c5
