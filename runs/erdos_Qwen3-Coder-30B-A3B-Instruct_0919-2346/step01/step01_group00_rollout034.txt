def project_box_sum(h, target_sum, lower=0.0, upper=1.0):
    """Project h onto box constraints [lower, upper] and sum constraint."""
    h = np.array(h, dtype=float)
    h_clipped = np.clip(h, lower, upper)
    current_sum = np.sum(h_clipped)
    
    if np.isclose(current_sum, target_sum):
        return h_clipped
    
    # Binary search for optimal lambda
    def sum_with_lambda(lam):
        proj = np.clip(h - lam, lower, upper)
        return np.sum(proj)
    
    if current_sum < target_sum:
        # Need to increase sum
        lam_lo, lam_hi = 0.0, 1.0
        while sum_with_lambda(lam_hi) < target_sum:
            lam_hi *= 2.0
        # Binary search
        for _ in range(50):
            lam = (lam_lo + lam_hi) / 2.0
            s = sum_with_lambda(lam)
            if s < target_sum:
                lam_lo = lam
            else:
                lam_hi = lam
        lam = (lam_lo + lam_hi) / 2.0
        h = np.clip(h - lam, lower, upper)
    else:
        # Need to decrease sum
        lam_lo, lam_hi = 0.0, 1.0
        while sum_with_lambda(lam_hi) > target_sum:
            lam_hi *= 2.0
        # Binary search
        for _ in range(50):
            lam = (lam_lo + lam_hi) / 2.0
            s = sum_with_lambda(lam)
            if s > target_sum:
                lam_lo = lam
            else:
                lam_hi = lam
        lam = (lam_lo + lam_hi) / 2.0
        h = np.clip(h + lam, lower, upper)
    
    return h

def compute_C5(h):
    """Compute C5 = max_k integral h(x)(1 - h(x+k)) dx."""
    dx = 2.0 / len(h)
    # Compute correlation: h * (1 - h) with full mode
    c = np.correlate(h, 1 - h, mode="full")
    # Multiply by dx to get continuous integral
    c *= dx
    # Return maximum value among all shifts
    return np.max(c)

def compute_C5_smoothed(h, alpha=50.0):
    """Compute a smoothed approximation to C5."""
    dx = 2.0 / len(h)
    # Compute correlation: h * (1 - h) with full mode
    c = np.correlate(h, 1 - h, mode="full")
    c *= dx
    # Smoothed max using log-sum-exp
    max_c = np.log(np.sum(np.exp(alpha * c))) / alpha
    return max_c

def compute_grad(h, alpha=50.0):
    """Compute gradient of the smoothed C5."""
    dx = 2.0 / len(h)
    N = len(h)
    # Compute correlation of h with 1-h
    c = np.correlate(h, 1 - h, mode="full")
    c *= dx
    
    # Compute weights for smoothing (softmax)
    w = np.exp(alpha * c)
    w = w / np.sum(w)
    
    # Compute gradients for each point using convolution
    # w_rev is w with reversed indices
    w_rev = w[::-1]
    
    # Compute convolution: sum w_k * h_{i+k}
    v_plus = np.convolve(h, w_rev, mode='same')
    # Compute convolution: sum w_k * h_{i-k}
    v_minus = np.convolve(h, w, mode='same')
    
    # Gradient: g[i] = dx * (1 - (v_plus[i] + v_minus[i]))
    g = dx * (1 - (v_plus + v_minus))
    
    return g
