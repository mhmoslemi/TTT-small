import numpy as np
import time

def c5_of(h):
    """Compute the maximum overlap integral for the given step function h."""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    c = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(c) * dx
    return c5

def evaluate_constraint(h):
    """Check normalization constraint"""
    dx = 2.0 / len(h)
    return abs(np.sum(h) * dx - 1.0)

def propose_move(h, max_step):
    """Propose a new feasible step function by adjusting two entries."""
    N = len(h)
    while True:
        i, j = np.random.choice(N, size=2, replace=False)
        # Determine feasible step
        delta_low = max(-h[i], h[j] - 1)
        delta_high = min(1 - h[i], h[j])
        if delta_low >= delta_high:
            continue  # Try another pair
        # Sample step size
        step_size = np.random.uniform(max(-max_step, delta_low),
                                      min(max_step, delta_high))
        # Create new h
        h_new = h.copy()
        h_new[i] += step_size
        h_new[j] -= step_size
        return h_new

def local_search(initial_h, time_budget):
    """Use local search to refine solution."""
    start = time.time()
    cur = initial_h.copy()
    c_cur = c5_of(cur)
    best = cur.copy()
    c_best = c_cur
    step = 0.05
    while time.time() - start < time_budget:
        h_new = propose_move(cur, step)
        c_new = c5_of(h_new)
        if c_new < c_best:
            best = h_new
            c_best = c_new
        elif c_new < c_cur:
            cur = h_new
            c_cur = c_new
    return best, c_best
