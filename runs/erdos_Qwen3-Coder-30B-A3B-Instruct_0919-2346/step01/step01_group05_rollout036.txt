import numpy as np
from scipy.optimize import minimize
from math import sqrt
import time

def build_h_from_coeffs(a, n):
    """Build h vector from Fourier coefficients."""
    h = 0.5 * np.ones(n)
    for m in range(1, len(a) + 1):
        h += a[m-1] * np.cos(2 * np.pi * m * np.arange(n) / n)
    return np.clip(h, 0, 1)

def compute_overlap_and_gradient(a, n, dx, gamma, penalty_weight=1e4):
    """Compute the soft-max overlap objective and its gradient."""
    h = build_h_from_coeffs(a, n)
    # Ensure h sum is n/2 by clipping and then reweighting the mean
    h_sum = np.sum(h)
    if abs(h_sum - n/2) > 1e-6:
        h = h * (n/2) / h_sum
    h = np.clip(h, 0, 1)
    
    # Penalty for box constraints
    penalty = penalty_weight * (np.sum(np.maximum(0, h-1)**2) + np.sum(np.maximum(0, -h)**2))
    
    # Compute correlation
    c = np.correlate(h, 1-h, mode="full")
    # Extract shifts s from -(n-1) to n-1
    shifts_s = np.arange(-(n-1), n)
    overlaps = c * dx  # Renormalise by dx
    
    # Soft max: log(sum(exp(gamma * overlaps))) / gamma
    exp_g = np.exp(gamma * overlaps)
    soft_sum = np.sum(exp_g)
    objective = (1.0 / gamma) * np.log(soft_sum) + penalty
    
    # Compute gradients
    grad = np.zeros_like(a)
    # Gradients of each shift
    # Compute the FFT here for efficiency
    cos_vals = np.array([np.cos(2 * np.pi * m * np.arange(n) / n) for m in range(1, len(a)+1)])
    for m in range(len(a)):
        # d/d a_m of each f_s
        grad_component = 0
        for i, shift in enumerate(shifts_s):
            s = shift
            # Indices where both h_i and h_{i+s} are valid
            if s >= 0:
                valid_i = np.arange(n - s) if s < n else np.array([])
                i_s = valid_i + s
            else:
                valid_i = np.arange(-s, n) if -s < n else np.array([])
                i_s = valid_i - s
            if len(valid_i) > 0:
                # Compute cos(2π m i/n) and cos(2π m (i+s)/n) for valid indices
                cos_i = cos_vals[m, valid_i]
                cos_is = cos_vals[m, i_s]
                # f_s = dx * sum_{i: valid} h_i (1 - h_{i+s})
                delta_h = 1 - h[i_s]
                term1 = cos_i * delta_h
                term2 = -h[valid_i] * cos_is
                grad_component += np.sum(term1 + term2)
        grad[m] = grad_component * dx
    
    # Add penalty gradient
    for m in range(len(a)):
        d_penalty = 2 * penalty_weight * np.sum((h - np.clip(h, 0, 1)) * cos_vals[m])
        grad[m] += d_penalty

    return objective, grad

def project_to_simplex_box(h, desired_sum):
    """Project vector to a box and a simplex"""
    # Ensure h is in [0, 1]
    h = np.clip(h, 0, 1)
    delta = desired_sum - np.sum(h)
    if abs(delta) < 1e-9:
        return h
    # Use bisection to find threshold τ
    # but here simpler: delayed adjustment until projection
    if np.sum(h) == 0:
        h = np.full_like(h, desired_sum/len(h))
        return h
    # Median search modified for earth mover type adjustment
    scale = 1.0
    # Compute adjustment iteratively
    tolerance = 1e-9
    max_iter = 1000
    for _ in range(max_iter):
        h_new = h + delta * scale / len(h)
        if np.min(h_new) >= 0 and np.max(h_new) <= 1:
            h = h_new.copy()
            break
        scale *= 0.5
    # Final adjustment to ensure exact sum
    h_with_sum = h.copy()
    h_with_sum = h_with_sum * desired_sum / np.sum(h_with_sum)
    return h_with_sum
    
def spectral_optimise(n, M, gamma_schedule, penalty_weight=1e4, max_iter=3000, initial_guess=None):
    """Perform optimisation in Fourier coefficient space"""
    dx = 2.0 / n
    init = np.random.normal(0, 0.05, M) if initial_guess is None else initial_guess
    bounds = [(-0.6, 0.6)] * M
    current_gamma = gamma_schedule[0]
    gamma_idx = 0
    
    obj_func_and_grad = lambda a: compute_overlap_and_gradient(a, n, dx, current_gamma, penalty_weight)
    
    options = {'maxiter': max_iter}
    
    res = minimize(obj_func_and_grad, init, method='L-BFGSB', bounds=bounds, jac=True, options=options)
    
    # Update to next gamma if needed
    if len(gamma_schedule) > 1:
        gamma_idx += 1
        if gamma_idx < len(gamma_schedule):
            current_gamma = gamma_schedule[gamma_idx]

    return res.x
