```python
import numpy as np
import time
import cvxpy as cp
from cvxpy import ECOS, GLPK

def water_fill(h, target_sum):
    """Project h onto the space of vectors with sum = target_sum using water filling."""
    h = np.array(h)
    h_sorted = np.sort(h)[::-1]
    cumsum = np.cumsum(h_sorted)
    n = len(h_sorted)
    # Find largest k such that (h_sorted[k] - (cumsum[k] - target_sum) / (k+1)) > 0
    # Use binary search or direct check
    for k in range(n):
        if (h_sorted[k] - (cumsum[k] - target_sum) / (k + 1)) <= 0:
            k = k - 1
            break
    else:
        k = n - 1
    if k == -1:
        return np.zeros_like(h)
    threshold = (cumsum[k] - target_sum) / (k + 1)
    h_proj = np.maximum(h - threshold, 0)
    return h_proj

def compute_c5(h, dx):
    """Compute the actual C5 value from the function h."""
    # Correlation with 1-h
    c = np.correlate(h, 1-h, mode='full')
    return np.max(c) * dx

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start = time.time()
    N = 140
    dx = 2.0 / N
    total_time = budget_s
    remaining_budget = total_time

    # Build pair indices
    pairs = []  # list of (i,s)
    shift_to_pairs_idx = {}  # maps shift to list of indices in pairs
    
    # For shift s from -(N-1) to N-1
    pair_idx = 0
    for s in range(-(N-1), N):
        indices = []
        for i in range(N):
            j = i + s
            if 0 <= i < N and 0 <= j < N:
                pairs.append((i, s))
                indices.append(pair_idx)
                pair_idx += 1
        shift_to_pairs_idx[s] = indices

    # Build variables
    h = cp.Variable(N)
    y = cp.Variable(len(pairs))
    alpha = cp.Variable()

    # Constraints
    constraints = []
    
    # Bounds on h
    constraints += [0 <= h, h <= 1]
    
    # Total mass constraint
    constraints.append(cp.sum(h) == N/2)
    
    # McCormick constraints for each pair
    for idx, (i, s) in enumerate(pairs):
        constraints.append(y[idx] <= h[i])
        constraints.append(y[idx] <= h[i + s])
        constraints.append(y[idx] >= h[i] + h[i + s] - 1)
        constraints.append(y[idx] >= 0)
        
    # α bounds
    for s in range(-(N-1), N):
        sidx = shift_to_pairs_idx[s]
        if len(sidx) > 0:
            # Sum over all pairs for this shift
            sum_y_s = cp.sum(y[sidx])
            constraints.append(alpha >= dx * (N/2 - sum_y_s))
    
    # Objective
    objective = cp.Minimize(alpha)
    prob = cp.Problem(objective, constraints)

    # Solve the LP relaxation
    remaining_budget = total_time - (time.time() - start)
    prob.solve(solver=ECOS, max_iters=20000, verbose=False)
    
    # Check against budget
    if time.time() - start >= total_time - 1e-3:
        # Final fallback: keep the best h found
        h_val = np.array(h.value).reshape(-1)
        h_val = water_fill(h_val, N/2)
        c5_val = compute_c5(h_val, dx)
        return (h_val, c5_val, N)
        
    h_LP = np.array(h.value).reshape(-1)
    h_LP = water_fill(h_LP, N/2)
    c5_LP = compute_c5(h_LP, dx)
    alpha_LP = float(alpha.value)
    
    # If LP solution is already good enough, return it
    if c5_LP <= 0.3807:
        return (h_LP, c5_LP, N)
    
    # Try to solve MILP - mixed integer program for binary h
    h_bin = cp.Variable(N, boolean=True)
    y_bin = cp.Variable(len(pairs))

    # Again add same constraints (McCormick), and sum constraint
    bin_constraints = []
    bin_constraints += [0 <= h_bin, h_bin <= 1]
    bin_constraints.append(cp.sum(h_bin) == N/2)

    for idx, (i, s) in enumerate(pairs):
        bin_constraints.append(y_bin[idx] <= h_bin[i])
        bin_constraints.append(y_bin[idx] <= h_bin[i + s])
        bin_constraints.append(y_bin[idx] >= h_bin[i] + h_bin[i + s] - 1)
        bin_constraints.append(y_bin[idx] >= 0)
        
    bin_alpha = cp.Variable()
    for s in range(-(N-1), N):
        sidx = shift_to_pairs_idx[s]
        if len(sidx) > 0:
            sum_y_s = cp.sum(y_bin[sidx])
            bin_constraints.append(bin_alpha >= dx * (N/2 - sum_y_s))
            
    bin_objective = cp.Minimize(bin_alpha)
    bin_prob = cp.Problem(bin_objective, bin_constraints)
    
    # Give remaining budget to MILP solver
    remaining_budget = total_time - (time.time() - start)
    if remaining_budget <= 1e-3:
        # Fallback:
        return (h_LP, c5_LP, N)
    # Use time limit
    try:
        bin_prob.solve(solver=cp.CBC, verbose=False, maximumSeconds=remaining_budget)
        
        if bin_prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_bin_val = np.array(h_bin.value).reshape(-1)
            c5_bin = compute_c5(h_bin_val, dx)
            return (h_bin_val, c5_bin, N)
        else:
            # If no feasible solution (or timeout), fall back to rounding LP solution
            pass
    except Exception as e:
        pass
    
    # Greedy rounding + local refinement
    h_rounded = h_LP.copy()
    idx_order = np.argsort(-h_rounded)  # descending order
    h_rounded.fill(0)
    h_rounded[idx_order[:int(N/2)]] = 1

    # Refinement
    h_current = h_rounded.copy()
    c_current = compute_c5(h_current, dx)
    remaining_budget = total_time - (time.time() - start)
    refine_start = time.time()
    
    if remaining_budget > 1e-3:
        while time.time() - refine_start < remaining_budget and c_current > 0.3807:
            # We know all entries are 0 or 1 in h_current
            # Compute all overlaps
            c_vals = []
            correlations = np.correlate(h_current, 1 - h_current, mode='full')
            for i in range(len(correlations)):
                c_vals.append(correlations[i] * dx)
            worst_idx = np.argmax(c_vals) - (len(correlations) - 1) // 2  # revert shift index
            
            # Compute gradient (here we just do point-wise descent – approximate)
            # Have to use very small step to avoid going out of [0,1] bounds because h is not continuous here
            # Let's do actual gradient-like step: if diff < 0 in original shift, make changes
            # We use approximate gradient computation: for shift i, the derivative with respect to h[j] is -1 if i+j is in shift, and vice versa.
            
            # Approximate subgradient direction
            if abs(worst_idx) < N:
                idx = worst_idx + (len(correlations) - 1) // 2  # full correlation index
                grad_h = np.zeros(N)
                for i in range(N):
                    j = i + worst_idx
                    if 0 <= j < N:
                        grad_h[i] -= 1
                
                # Now, update h with small step
                step = 0.02
                h_new = h_current - step * grad_h
                h_new = np.clip(h_new, 0, 1)
                h_new = water_fill(h_new, N/2)
                c_new = compute_c5(h_new, dx)
                if c_new < c_current:
                    h_current = h_new
                    c_current = c_new
            else:
                break  # stop if we are out of bounds or we go below threshold
            
    return (h_current, c_current, N)
```
