import numpy as np

from scipy.optimize import differential_evolution, minimize

def c5_of(h):
    """Compute C5 = max_k integral h(x)(1-h(x+k)) dx."""
    dx = 2.0 / len(h)
    # Compute correlation: h * (1 - h) with full mode
    c = np.correlate(h, 1 - h, mode="full")
    # Multiply by dx to get continuous integral
    c *= dx
    # Return maximum value among all shifts
    return np.max(c)

def project_to_feasible(h):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = n/2."""
    h = np.clip(h, 0, 1)
    dx = 2.0 / len(h)
    current_sum = np.sum(h) * dx
    h = h * (1.0 / current_sum)  # Scale to sum to 1
    h = h * (len(h) / 2.0)  # Scale to sum to n/2
    return h

def local_search(h, step_size=0.05):
    """Perform a simple local search to improve h."""
    h = h.copy()

    dx = 2.0 / N
    best_h = h.copy()

        i = random.randint(0, N - 1)
        old_val = h[i]
        delta = random.uniform(-step_size, step_size)
        h[i] += delta
        
        # Project back to feasible region
        h = project_to_feasible(h)
        
        c5 = c5_of(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        else:
            h[i] = old_val  # Revert if not better
    
    return best_h, best_c5

        return np.sum(h) * dx - 1.0

        recombination=0.7,

        constraints=constraints,
        disp=False
    )
    
    # Normalize and return the best solution
    if result.success:
        h_final = result.x * (n_points / 2) / np.sum(result.x)
        return h_final, c5_of(h_final)
    else:
        return None, float('inf')
