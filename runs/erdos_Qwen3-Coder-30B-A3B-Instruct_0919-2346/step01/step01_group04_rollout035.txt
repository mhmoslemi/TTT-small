import numpy as np
import time
import random
from scipy.optimize import differential_evolution, minimize


def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    overlap = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(overlap) * dx
    return c5


def project_to_feasible(h):
    """Project h to feasible region: 0 <= h[i] <= 1 and sum(h)*dx = 1"""
    # Ensure bounds
    h = np.clip(h, 0, 1)
    # Rescale to exact integral
    dx = 2.0 / len(h)
    scale = (2.0 / len(h)) / np.sum(h)
    h = h * scale
    return h


def propose_move(h, max_step):
    """Propose a new feasible h vector by adjusting two elements"""
    h_new = h.copy()
    N = len(h)
    
    # Try to find a valid move
    max_tries = 50
    for _ in range(max_tries):
        i = random.randint(0, N - 1)
        j = random.randint(0, N - 1)
        if i == j:
            continue
            
        # Determine feasible interval for δ
        delta_low = max(-h_new[i], h_new[j] - 1)
        delta_high = min(1 - h_new[i], h_new[j])
        
        if delta_low >= delta_high:
            continue
            
        # Sample δ from uniform distribution over feasible range
        step = min(max_step, delta_high - delta_low)
        if step <= 0:
            continue
        delta = random.uniform(delta_low, delta_high)
        
        # Apply move
        h_new[i] += delta
        h_new[j] -= delta
        
        # Ensure within bounds again
        h_new = project_to_feasible(h_new)
        
        return h_new
    
    # If no valid move found, return original
    return h.copy()


def local_search(initial_h, time_budget_s):
    """Use local search with bounds"""
    start = time.time()
    
    # Parameters
    T_start = 0.2
    T_end = 1e-6
    
    # Set up the current state
    cur = project_to_feasible(initial_h.copy())
    c_cur = c5_of(cur)
    
    best = cur.copy()
    c_best = c_cur
    
    # Main loop
    while time.time() - start < time_budget_s:
        t = time.time() - start
        elapsed_fraction = t / time_budget_s
        
        # Cooling schedule
        T = T_start * (T_end / T_start) ** elapsed_fraction
        step = 0.1 * (1 - elapsed_fraction)
        
        # Make a proposal
        h_proposed = propose_move(cur, step)
        c_proposed = c5_of(h_proposed)
        
        # Accept the proposal
        if c_proposed <= c_cur:
            cur = h_proposed
            c_cur = c_proposed
            if c_proposed < c_best:
                best = h_proposed.copy()
                c_best = c_proposed
        else:
            # Metropolis acceptance rule
            if random.random() < np.exp(-(c_proposed - c_cur) / T):
                cur = h_proposed
                c_cur = c_proposed
                
    return best, c_best

    # Some structured seeds

    mid = N // 2
    width = N // 4
    h_step[mid-width:mid+width] = 1.0
    h_step = project_to_feasible(h_step)
    
    # Try optimization using scipy
    def objective(h):
        return c5_of(h)

    remaining_budget = budget_s * 0.95
