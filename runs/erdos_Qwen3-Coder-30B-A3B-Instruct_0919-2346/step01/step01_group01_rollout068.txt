def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using full convolution
    overlap = np.correlate(h, 1 - h, mode='full')
    # The maximum value scaled by dx
    c5 = np.max(overlap) * dx
    return c5

def propose_move(h, max_step):
    """Propose a move that maintains sum constraint exactly"""
    h_new = h.copy()
    N = len(h)
    while True:
        i, j = random.sample(range(N), 2)
        # Compute feasible range for delta
        delta_low = max(-h_new[i], h_new[j] - 1.0)
        delta_high = min(1.0 - h_new[i], h_new[j])
        if delta_low >= delta_high:
            continue  # No valid move, try again
        break
    
    # Sample delta from feasible interval
    step = min(max_step, delta_high - delta_low)
    if step <= 0:
        delta = 0.0
    else:
        delta = random.uniform(max(-step, delta_low), min(step, delta_high))
        
    h_new[i] += delta
    h_new[j] -= delta
    return h_new

def particle_swarm(h0, max_iter, n_particles=10, w=0.7, c1=1.5, c2=1.5, bounds=None):
    """Swarm optimization for C5 minimization"""
    n = len(h0)
    if bounds is None:
        bounds = [(0, 1) for _ in range(n)]
    
    # Initialize particles
    particles = [h0.copy() for _ in range(n_particles)]
    velocities = [np.zeros(n) for _ in range(n_particles)]
    pbest = [h.copy() for h in particles]
    pbest_scores = [c5_of(h) for h in particles]
    
    # Global best
    gbest_idx = np.argmin(pbest_scores)
    gbest = pbest[gbest_idx].copy()
    gbest_score = pbest_scores[gbest_idx]
    
    for iteration in range(max_iter):
        for i in range(n_particles):
            # Update velocity
            r1, r2 = random.random(), random.random()
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply bounds and sum constraint
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Normalize to sum = n/2
            particles[i] *= (n / 2) / np.sum(particles[i])
            
            # Evaluate
            score = c5_of(particles[i])
            
            # Update personal best
            if score < pbest_scores[i]:
                pbest[i] = particles[i].copy()
                pbest_scores[i] = score
                if score < gbest_score:
                    gbest = particles[i].copy()
                    gbest_score = score
    
    return gbest, gbest_score
