import numpy as np
import time
import random
from scipy.optimize import differential_evolution

def c5_of(h):
    """Compute C5 = max_k integral h(x)(1-h(x+k)) dx."""
    dx = 2.0 / len(h)
    # Compute correlation: h * (1 - h) with full mode
    c = np.correlate(h, 1 - h, mode="full")
    # Multiply by dx to get continuous integral
    c *= dx
    # Return maximum value among all shifts
    return np.max(c)

def project_to_feasible(h):
    """Project h to feasible region: [0,1] and sum==n_points/2."""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to sum = n_points/2
    n_points = len(h)
    h = h * (n_points / 2) / np.sum(h)
    return h

def objective(h):
    """Objective function to minimize."""
    return c5_of(h)

    """Use differential evolution for global optimization."""

    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use differential evolution with bounds
    result = differential_evolution(
        objective,

        maxiter=max(100, int(budget_s / 10)),
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.7,
        seed=42,
        disp=False
    )
    
    best_h = project_to_feasible(result.x)
    best_c5 = result.fun
