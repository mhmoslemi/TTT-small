```python
import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def water_fill(h, target_sum):
    """Project h onto the set {x: sum(x) = target_sum, 0 <= x <= 1}"""
    h = np.clip(h, 0, 1)
    if np.sum(h) == target_sum:
        return h
    # Use binary search on the Lagrange multiplier
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) / 2
        proposed = np.clip(h + lam, 0, 1)
        s = np.sum(proposed)
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) / 2
    result = np.clip(h + lam, 0, 1)
    return result

def c5_of(h):
    """Compute the maximum overlap integral for the given step function h."""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    c = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(c) * dx
    return c5

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start = time.time()
    if 'initial_h_values' in globals():
        N = len(initial_h_values)
    else:
        N = 140
    dx = 2.0 / N
    # Build pairs for McCormick constraints: (i,s) where both h[i] and h[i+s] exist
    pairs = []
    s_indices = {}
    s_len = 2*N - 1
    for s in range(-(N-1), N):  # Shifts from -(N-1) to N-1
        for i in range(N):
            if 0 <= i+s and i+s < N:
                pairs.append((i, s))
        s_indices[s] = len(pairs) - len([x for x in pairs if x[1] == s])  # start index for this shift
    
    model_start = time.time()
    # Variables
    h = cp.Variable(N, name='h')
    y = cp.Variable(len(pairs), name='y')
    alpha = cp.Variable(name='alpha')
    
    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == N/2,
        alpha >= 0
    ]
    
    # McCormick envelope constraints for each pair (i,s)
    for idx, (i, s) in enumerate(pairs):
        # y[i,s] <= h[i]
        constraints.append(y[idx] <= h[i])
        # y[i,s] <= h[i+s]
        constraints.append(y[idx] <= h[i+s])
        # y[i,s] >= h[i] + h[i+s] - 1
        constraints.append(y[idx] >= h[i] + h[i+s] - 1)
        # y[i,s] >= 0
        constraints.append(y[idx] >= 0)
    
    # Alpha constraints for each shift (can reuse s_indices)
    for s in range(-(N-1), N):
        if s >= 0:
            s_len = N - s
            range_start = s_indices[s]
            y_sum = cp.sum(y[range_start:range_start+s_len])
        else:
            s_len = N + s
            range_start = s_indices[s]
            y_sum = cp.sum(y[range_start:range_start+s_len])
        # For shift s, C_s = dx * (sum(h) - sum_y) = dx * (N/2 - sum_y)
        # So require: alpha >= dx * (N/2 - sum_y)
        # This becomes: alpha >= (dx * N/2) - (dx * sum_y)
        # So we have: alpha + dx * sum_y >= dx * N/2
        constraints.append(alpha >= dx * (N/2 - y_sum))
    
    # Objective
    prob = cp.Problem(cp.Minimize(alpha), constraints)
    
    # Solve the LP relaxation
    try:
        # Use ECOS as it works well for large scale LPs
        prob.solve(solver=cp.ECOS, max_iters=10000)
        h_val = h.value
        alpha_val = alpha.value
    except:
        # Fallback to basic initialization
        h_val = np.ones(N) * 0.5
        h_val = water_fill(h_val, N/2)
        alpha_val = c5_of(h_val)
    print(f"LP bound: {alpha_val}, time: {time.time() - model_start:.2f}s")
    
    # Now we do a short hill-climb refinement from the LP solution
    h_ref = h_val.copy()
    c5_ref = c5_of(h_ref)
    local_start = time.time()
    
    # Two-stage refinement
    for _ in range(200):
        if time.time() - start > 0.95 * budget_s:
            break
        # Find worst shift
        c_vals = np.correlate(h_ref, 1 - h_ref, mode='full')
        worst_shift_idx = np.argmax(c_vals)
        worst_c = c_vals[worst_shift_idx]
        # Compute subgradient approximately
        # For shift s_worst, we compute gradient of sum w.r.t h
        # Let's compute it manually by finite difference for debugging or as a helper
        if worst_shift_idx >= N:
            shift = worst_shift_idx - N + 1  # shift key
        else:
            shift = worst_shift_idx - N + 1  # negative shift
        
        # Shift key for indexing
        if shift >= 0 and shift < N:
            # This is correct for shift s, i goes from 0 to N-1-shift
            s = shift
            grad = np.zeros_like(h_ref)
            for i in range(N - s):
                grad[i] += 1  # h[i] contributes to both 1-h[i+s] and h[i]
                grad[i+s] -= 1  # h[i+s] contributes to 1-h[i+s] and potentially h[i]
            grad /= dx  # Scale by dx
            # We want to move in the opposite direction of gradient
            step_size = 0.01
            new_h = h_ref - step_size * grad
            new_h = water_fill(new_h, N/2)
            new_c5 = c5_of(new_h)
            if new_c5 < c5_ref:
                h_ref = new_h
                c5_ref = new_c5
    
    # If MILP is called, prefer scaling down N so it can finish
    c5_final = c5_ref
    
    # Now consider MILP if LP did not finish the desired bound
    if time.time() - start < 0.8 * budget_s and c5_final > 0.381:
        try:
            # Binary MILP formulation here
            h_bin = cp.Variable(N, boolean=True)
            y_bin = cp.Variable(len(pairs), name='y_bin')
            alpha_milp = cp.Variable(name='alpha_milp')
            # McCormick constraints for binary variables
            constraints_milp = [
                h_bin >= 0,
                h_bin <= 1,
                cp.sum(h_bin) == N/2,
                alpha_milp >= 0
            ]
            for idx, (i, s) in enumerate(pairs):
                constraints_milp.append(y_bin[idx] <= h_bin[i])
                constraints_milp.append(y_bin[idx] <= h_bin[i+s])
                constraints_milp.append(y_bin[idx] >= h_bin[i] + h_bin[i+s] - 1)
                constraints_milp.append(y_bin[idx] >= 0)
            # Alpha constraints
            for s in range(-(N-1), N):
                if s >= 0:
                    s_len = N - s
                    range_start = s_indices[s]
                    y_sum = cp.sum(y_bin[range_start:range_start+s_len])
                else:
                    s_len = N + s
                    range_start = s_indices[s]
                    y_sum = cp.sum(y_bin[range_start:range_start+s_len])
                constraints_milp.append(alpha_milp >= dx * (N/2 - y_sum))
            # Try to minimize alpha
            prob_milp = cp.Problem(cp.Minimize(alpha_milp), constraints_milp)
            # Solve with limited computational time
            solver_options = {'maxTime': min(100, budget_s - (time.time() - start) - 10)}
            prob_milp.solve(solver=cp.CBC, verbose=False, solver_options=solver_options)
            if prob_milp.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_milp = h_bin.value
                c5_milp = c5_of(h_milp)
                if c5_milp < c5_final:
                    c5_final = c5_milp
                    h_ref = h_milp
        except:
            pass
    
    # Final global check and return
    h_final = water_fill(h_ref.copy(), N/2)
    c5_best = c5_of(h_final)
    
    return (h_final, c5_best, N)
```
