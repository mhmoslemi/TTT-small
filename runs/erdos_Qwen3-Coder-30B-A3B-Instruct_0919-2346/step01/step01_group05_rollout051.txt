def project_onto_bounds_sum(v, target_sum, lower=0, upper=1):
    """Project vector v onto the set of vectors with sum = target_sum and bounds [lower, upper]"""
    # Binary search for the correct lambda
    left = v.min() - upper
    right = v.max() - lower
    
    def compute_sum(lambda_val):
        return np.clip(v - lambda_val, lower, upper).sum()
    
    for _ in range(60):
        mid = (left + right) / 2
        s = compute_sum(mid)
        if s > target_sum:
            left = mid
        else:
            right = mid
    
    return np.clip(v - (left + right) / 2, lower, upper)

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value scaled by dx
    max_overlap = np.max(correlation) * dx
    return max_overlap

def compute_gradient(h, correlation, weights, dx):
    """Compute gradient of the weighted sum of overlaps"""
    n = len(h)
    grad = np.zeros_like(h)
    
    # weights are for all l in range -n+1 to n-1
    # correlation is already in full mode, let's match the indexing
    # correlation index l + (n-1) corresponds to shift l
    
    # For each index i, compute sum over shifts of weights * grad contribution
    for i in range(n):
        for shift_l in range(-n+1, n):
            # indices valid for this shift
            # i and i+shift_l must be within 0..n-1
            if 0 <= i < n and 0 <= i + shift_l < n:
                # The weight for this shift
                weight = weights[shift_l + (n-1)]
                # Gradient contribution from (1-h[i+shift_l])
                grad[i] += weight * (1 - h[i + shift_l])
                # Gradient contribution from -h[i-shift_l] 
                if 0 <= i - shift_l < n:
                    grad[i] -= weight * h[i - shift_l]
    
    return grad * dx
