import numpy as np
import time

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value scaled by dx
    max_overlap = np.max(correlation) * dx
    return max_overlap

def softmax_weights(gam, fs):
    """Compute softmax weights for a given γ and list of f_s values"""
    exp_gam_fs = np.exp(gam * fs)
    return exp_gam_fs / np.sum(exp_gam_fs)

def compute_gradient(h, dx, shifts):
    """Compute ∇J_γ(h) for given h, dx and shifts"""
    grad = np.zeros_like(h)
    n = len(h)
    for s in shifts:
        # Compute g_i^(s) = (1-h_{i+s}) - h_{i-s}
        g = np.zeros_like(h)
        if s >= 0:
            # For s >= 0, h_{i+s} exists for i from 0 to n-1-s
            g[:-s] = 1 - h[s:]
        else:
            # For s < 0, h_{i+s} exists for i from -s to n-1
            g[-s:] = 1 - h[:n+s]
        if s <= 0:
            # For s <= 0, h_{i-s} exists for i from -s to n-1
            g[-s:] -= h[:n+s]
        else:
            # For s > 0, h_{i-s} exists for i from s to n-1
            g[:-s] -= h[s:]
        grad += g
    return dx * grad

def project_to_simplex(v, target_sum):
    """Project vector v onto the simplex {x: Σx_i = target_sum, 0 ≤ x_i ≤ 1}"""
    # This uses the waterfilling method with binary search on λ
    # Compute φ(λ) = Σ_i clip(v_i - λ, 0, 1)
    v = np.asarray(v)
    low, high = np.min(v), np.max(v) + 1.0
    target = target_sum
    
    def phi(λ):
        return np.sum(np.clip(v - λ, 0, 1))
    
    # Binary search to find λ such that φ(λ) = target
    for _ in range(50):  # should converge in about 40 steps
        mid = (low + high) / 2.0
        val = phi(mid)
        if val > target:
            low = mid
        else:
            high = mid
            
    lambda_val = (low + high) / 2.0
    return np.clip(v - lambda_val, 0, 1)

def upsample_initial(initial_h, n):
    """Upsample initial_h to length n using linear interpolation"""
    old_n = len(initial_h)
    old_x = np.linspace(0, 2, old_n, endpoint=False)
    new_x = np.linspace(0, 2, n, endpoint=False)
    new_h = np.interp(new_x, old_x, initial_h)
    return new_h
