import numpy as np
import time
import random
from scipy.optimize import differential_evolution

def c5_of(h):
    """Compute C5 = max_k integral h(x)(1-h(x+k)) dx."""
    dx = 2.0 / len(h)
    # Compute correlation: h * (1 - h) with full mode
    c = np.correlate(h, 1 - h, mode="full")
    # Multiply by dx to get continuous integral
    c *= dx
    # Return maximum value among all shifts
    return np.max(c)

def propose_move(h, max_step):
    """Generate a new feasible h by adjusting two entries."""
    h = h.copy()
    N = len(h)
    i, j = random.sample(range(N), 2)
    
    # Determine feasible interval for delta
    delta_low = max(-h[i], h[j] - 1.0)
    delta_high = min(1.0 - h[i], h[j])
    
    if delta_low >= delta_high:
        return h  # No valid move for this pair
    
    # Sample delta within allowed range
    step = min(max_step, delta_high - delta_low)
    delta = random.uniform(-step, step)
    delta = max(delta_low, min(delta, delta_high))
    
    h[i] += delta
    h[j] -= delta
    
    return h

def adaptive_anneal(initial_h, time_budget):
    """Adaptive simulated annealing with multiple cooling schedules."""
    start = time.time()
    cur = initial_h.copy()
    best = initial_h.copy()
    c_cur = c5_of(cur)
    c_best = c_cur
    
    # Try several different cooling schedules
    schedules = [
        # Fast cooling
        (0.1, 1e-6, 0.2),
        # Moderate cooling
        (0.05, 1e-5, 0.1),
        # Slow cooling  
        (0.02, 1e-7, 0.05)
    ]
    
    best_schedule = schedules[0]
    best_c = float('inf')
    
    for T_start, T_end, step_start in schedules:
        if time.time() - start > time_budget * 0.8:  # Leave some buffer
            break
            
        cur_trial = initial_h.copy()
        c_trial = c5_of(cur_trial)
        
        t = 0
        while time.time() - start < time_budget * 0.8 and t < 10000:
            T = T_start * (T_end / T_start) ** (t / 10000)
            step = step_start * (1 - t / 10000)
            if T < 1e-8:
                break
                
            h_new = propose_move(cur_trial, step)
            c_new = c5_of(h_new)
            
            if c_new <= c_trial or random.random() < np.exp(-(c_new - c_trial) / T):
                cur_trial = h_new
                c_trial = c_new
                if c_new < c_best:
                    best = cur_trial
                    c_best = c_new
                    
            t += 1
            
        if c_best < best_c:
            best_c = c_best
            best_schedule = (T_start, T_end, step_start)
    
    return best, c_best

def genetic_search(initial_h, time_budget):
    """Genetic algorithm approach."""
    start = time.time()
    N = len(initial_h)
    pop_size = 20
    generations = 1000
    
    # Initialize population
    population = [initial_h.copy()]
    for _ in range(pop_size - 1):
        h = initial_h.copy()
        # Add some perturbation
        for i in range(N):
            if random.random() < 0.2:
                h[i] = max(0, min(1, h[i] + random.gauss(0, 0.05)))
        population.append(h)
    
    # Normalize
    for h in population:
        h *= (N / 2) / np.sum(h)
        
    best_h = initial_h.copy()
    best_c5 = c5_of(best_h)
    
    for gen in range(generations):
        if time.time() - start > time_budget * 0.7:
            break
            
        # Evaluate fitness
        fitness = [1.0 / (1e-10 + c5_of(h)) for h in population]
        
        # Update best
        for h in population:
            c5 = c5_of(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Selection (tournament)
        new_population = []
        for _ in range(pop_size):
            # Tournament selection
            candidates = random.sample(list(zip(population, fitness)), 3)
            winner = max(candidates, key=lambda x: x[1])[0]
            new_population.append(winner.copy())
            
        # Crossover and mutation
        for i in range(0, pop_size, 2):
            if i + 1 < pop_size:
                # Uniform crossover
                child1 = new_population[i].copy()
                child2 = new_population[i+1].copy()
                for j in range(N):
                    if random.random() < 0.5:
                        child1[j], child2[j] = child2[j], child1[j]
                        
                # Mutation  
                for j in range(N):
                    if random.random() < 0.1:
                        child1[j] = max(0, min(1, child1[j] + random.gauss(0, 0.02)))
                    if random.random() < 0.1:
                        child2[j] = max(0, min(1, child2[j] + random.gauss(0, 0.02)))
                        
                # Renormalize
                child1 *= (N / 2) / np.sum(child1)
                child2 *= (N / 2) / np.sum(child2)
                
                new_population[i] = child1
                new_population[i+1] = child2
        
        population = new_population
        
    return best_h, best_c5
