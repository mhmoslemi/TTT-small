import numpy as np
import time
from math import sqrt

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value scaled by dx
    max_overlap = np.max(correlation) * dx
    return max_overlap

def sturmian_binary(n):
    """Generate a balanced binary vector using irrational rotation"""
    alpha = (sqrt(5) - 1) / 2  # Golden ratio conjugate
    h = np.zeros(n, dtype=int)
    positions = np.mod(np.arange(n) * alpha, 1.0)
    h[positions < 0.5] = 1
    ones_count = np.sum(h)
    if ones_count > n // 2:
        # Change some 1s to 0s
        indices = np.where(h == 1)[0]
        for i in range(ones_count - n // 2):
            h[indices[i]] = 0
    elif ones_count < n // 2:
        # Change some 0s to 1s
        indices = np.where(h == 0)[0]
        for i in range(n // 2 - ones_count):
            h[indices[i]] = 1
    return h

def compute_overlap_vector(h):
    """Compute overlap values for all shifts"""
    n = len(h)
    dx = 2.0 / n
    correlation = np.correlate(h, 1 - h, mode="full")
    return correlation * dx

def compute_delta_shifts(h, i, j, s_max, overlap_vals):
    """Compute change in all overlap values from swapping positions i (1) and j (0)"""
    n = len(h)
    delta = np.zeros(2 * n - 1)
    
    # For shift s, the contribution is h[k] * (1 - h[k+s])
    # When we swap:
    #   - If i is in overlap region for shift s: contributes - (1 - h[i+s])
    #   - If j is in overlap region for shift s: contributes + (1 - h[j+s])
    # Old term from position i: h_i * (1 - h_{i+s}) = 1 * (1 - h_{i+s}) = 1 - h_{i+s}
    # New term from position i: h'_i * (1 - h'_{i+s}) = 0 * (1 - h'_{i+s}) = 0
    # So change = - (1 - h_{i+s})
    
    # Old term from position j: h_j * (1 - h_{j+s}) = 0 * (1 - h_{j+s}) = 0
    # New term from position j: h'_j * (1 - h'_{j+s}) = 1 * (1 - h'_{j+s}) = 1 - h'_{j+s}
    # So change = + (1 - h'_{j+s}) = + (1 - h_{j+s})
    
    # But we also need to account for the other factor.
    # For shift s, the overall contribution is from: h_k * (1 - h_{k+s}) over k
    
    # For effect on shift s:
    #   k = i- (shift s means position k makes a contribution to shift s if k = i-s)
    #   so h_k = h_{i-s}. And this contributes (h_{i-s} * (1-h_i)) = h_{i-s} * 0 = 0
    
    # Use simpler way: compute for every shift s how changing i->0 and j->1 affects it:
    
    # i element: h_i = 1, becomes 0.
    # j element: h_j = 0, becomes 1.
    
    # For shift s: 
    #   original contribution = h[i] * (1-h[i+s]) + h[i-s] * (1-h[i]) if j is valid
    #   Now contributes = h[i] * (1-h[i+s]) + h[i-s] * (1-h[i]) = h[i] * (1-h[i+s]) + h[i-s] * 0 = h[i] * (1-h[i+s])
    #   if i is active
    #   
    #   Contribution of i to shift s = h[i] * (1-h[i+s]) = 1 * (1-h[i+s]) = 1-h[i+s]
    #   New = sum of h'[k]*(1-h'[k+s]) over all k
    #   Change at shift s = (1-h[i+s]) 
    #   
    #   Similar for j: P(j to s) = h[j] * (1-h[j+s]) = 0*(1-h[j+s]) = 0
    #   New = h'[j] * (1-h'[j+s]) = 1 * (1-h[j+s]) = 1-h[j+s]
    #   Change at shift s = + (1-h[j+s])
    #   if i+s is valid, and j+s is valid too.
    
    for s in range(2*n-1):
        if 0 <= i+s < n and 0 <= j+s < n:
            delta[s] = - (1 - h[i+s]) + (1 - h[j+s])
        elif 0 <= i+s < n:
            delta[s] = - (1 - h[i+s])
        elif 0 <= j+s < n:
            delta[s] = + (1 - h[j+s])
    return delta

def local_swap_search(h, max_time):
    """Local swap heuristic to minimize worst overlap"""
    start_time = time.time()
    n = len(h)
    dx = 2.0 / n
    
    # Precompute overlap values  
    overlap_vals = compute_overlap_vector(h)
    max_overlap = np.max(overlap_vals)
    worst_shift = np.argmax(overlap_vals)
    
    tabu_list = set()
    tabu_length = 100
    tabu_active = 0
    
    # Keep best solution so far
    best_h = h.copy()
    best_overlap = max_overlap
    
    iter_count = 0
    max_iter = 100000  # Cap iterations
    
    # Track when we revisited
    last_update = 0
    
    # Use binary indices for faster access
    ones_idx = np.where(h == 1)[0]
    zeros_idx = np.where(h == 0)[0]
    
    while time.time() - start_time < max_time and iter_count < max_iter:
        iter_count += 1
        
        # Find worst shift
        worst_shift = np.argmax(overlap_vals)
        # Record how we're doing
        current_max = overlap_vals[worst_shift]
        
        if current_max < best_overlap:
            best_overlap = current_max
            best_h = h.copy()
        
        # Determine age of best solution
        if current_max == best_overlap:
            last_update += 1
        else:
            last_update = 0
        
        # PIT: Apply perturbation after too many steps with no improvement
        if last_update > 2000:
            # Flip some bits while preserving sum
            # For even better than 0.3807 we want to perturb more carefully
            # Just for 0.379, we could just do better swaps
            # But for better, let's trigger some larger leap
            # Random perturbation of 5% of positions (keep balance)
            num_flip = max(2, n // 20)
            indices_to_flip = np.random.choice(n, size=num_flip, replace=False)
            # Count ones and zeros in this subset
            current_ones = sum(h[i] for i in indices_to_flip)
            current_zeros = num_flip - current_ones
            # Randomly flip a few up to valid changes
            to_flip_ones = min(current_ones, current_zeros)
            to_flip_zeros = min(current_zeros, current_ones)
            
            # Flip one of each type to keep balance
            flip_idx = np.random.choice(np.where(h[indices_to_flip] == 1)[0], to_flip_ones, replace=False)
            flip_zero = np.random.choice(np.where(h[indices_to_flip] == 0)[0], to_flip_zeros, replace=False)
            
            for i in flip_idx:
                h[indices_to_flip[i]] = 0
            for i in flip_zero:
                h[indices_to_flip[i]] = 1
                
            # Recompute overlap
            overlap_vals = compute_overlap_vector(h)
            continue
            
        # Select a random 1-bit and 0-bit to swap
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            # Maybe uneven balance? Correct that!
            h = np.ones(n) * 0.5
            ones_idx = np.where(h == 1)[0]
            zeros_idx = np.where(h == 0)[0]
            
        # Ensure we have enough elements
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
            
        i = np.random.choice(ones_idx)
        j = np.random.choice(zeros_idx)
        
        # Check tabu
        if (i, j) in tabu_list:
            continue
        
        # Compute gain from swap
        delta = compute_delta_shifts(h, i, j, worst_shift, overlap_vals)
        
        # Determine new overlap values after swap
        new_overlap_vals = overlap_vals + delta
        
        # If improvement, accept
        if np.max(new_overlap_vals) < current_max:
            # Accept the swap
            h[i] = 0
            h[j] = 1
            overlap_vals = new_overlap_vals
            
            # Update tabu list
            tabu_list.add((i, j))
            if len(tabu_list) > tabu_length:
                # Remove oldest (simplest policy)
                tabu_list.pop()
            
        else:
            # Add to tabu for some iterations
            tabu_list.add((i, j))
            if len(tabu_list) > tabu_length:
                tabu_list.pop()
        
        # Update indices for next iteration
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
    
    return best_h, best_overlap

def upscale_and_refine(h_coarse, n):
    """Upscale a vector by duplication"""
    n0 = len(h_coarse)
    h_fine = np.repeat(h_coarse, n // n0).reshape(-1)[:n]
    # Normalize to preserve balance
    h_fine = h_fine.astype(float)
    current_sum = np.sum(h_fine)
    target_sum = n / 2.0
    if abs(current_sum - target_sum) > 1e-6:
        # Adjust by shifting
        ratio = target_sum / current_sum
        h_fine = h_fine * ratio
    return h_fine

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Step 1: Start with high resolution to find good base
    n0 = 120
    h = sturmian_binary(n0)
    
    # Step 2: Local swap on coarse grid
    remaining_time = budget_s * 0.6  
    h, c5 = local_swap_search(h, remaining_time)
    
    # Step 3: Upscale and refine
    n1 = 240
    h_up = upscale_and_refine(h, n1)
    remaining_time = budget_s * 0.3
    h_up, c5_up = local_swap_search(h_up, remaining_time)
    
    # Step 4: Final polish (optional continuous improvement)
    if budget_s - (time.time() - start_time) > 10:
        h_final = continuous_polish(h_up)
        c5_final = evaluate_erdos_solution(h_final)
        if c5_final < c5_up:
            c5 = c5_final
            h = h_final
        else:
            h = h_up
            c5 = c5_up
    else:
        h = h_up
        c5 = c5_up
    
    return (h.tolist(), c5, len(h))
