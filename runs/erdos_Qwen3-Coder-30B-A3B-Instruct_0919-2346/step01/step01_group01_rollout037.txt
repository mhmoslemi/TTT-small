import numpy as np
import time
import random
from scipy.optimize import differential_evolution

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using full convolution
    overlap = np.correlate(h, 1 - h, mode='full')
    # The maximum value scaled by dx
    c5 = np.max(overlap) * dx
    return c5

def evaluate_solution(h):
    """Evaluate a solution for C5"""
    return c5_of(h)

def propose_move(h, step_size):
    """Propose a move that maintains sum constraint exactly"""
    h_new = h.copy()
    N = len(h)
    
    # Try multiple times to find valid move
    for _ in range(100):
        i, j = random.sample(range(N), 2)
        # Compute feasible range for delta
        delta_low = max(-h_new[i], h_new[j] - 1.0)
        delta_high = min(1.0 - h_new[i], h_new[j])
        
        if delta_low >= delta_high:
            continue  # No valid move, try again
            
        delta = random.uniform(max(-step_size, delta_low), min(step_size, delta_high))
        
        h_new[i] += delta
        h_new[j] -= delta
        
        # Ensure constraints
        if np.all(h_new >= 0) and np.all(h_new <= 1):
            return h_new
    
    return h.copy()  # Return unchanged if no valid move 

def local_search(h_start, budget_s):
    """Perform local search using simulated annealing"""
    start = time.time()
    cur = h_start.copy()
    best = h_start.copy()
    c_cur = c5_of(cur)
    c_best = c_cur
    
    # Annealing parameters
    T_start = 0.1
    T_end = 1e-6
    step_start = 0.1
    
    while time.time() - start < budget_s:
        elapsed = time.time() - start
        T = T_start * (T_end / T_start) ** (elapsed / budget_s)
        step = step_start * (1.0 - elapsed / budget_s)
        
        h_prop = propose_move(cur, step)
        c_prop = c5_of(h_prop)
        
        if c_prop < c_best:
            best = h_prop.copy()
            c_best = c_prop
            
        if c_prop <= c_cur or random.random() < np.exp(-(c_prop - c_cur) / T):
            cur = h_prop
            c_cur = c_prop
    
    return best, c_best
