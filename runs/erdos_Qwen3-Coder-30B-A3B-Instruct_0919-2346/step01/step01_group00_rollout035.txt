def project_h(h, target_sum):
    """Project h onto [0,1] and sum=target_sum using water-filling."""
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if np.isclose(current_sum, target_sum):
        return h
    # Use binary search to find lambda
    h_sorted = np.sort(h)[::-1]  # Descending
    # Compute cumulative sums
    cumsum = np.cumsum(h_sorted)
    # Find where sum exceeds target
    diff = cumsum - target_sum
    n = len(h)
    # Find largest k such that cumsum[k] <= target_sum
    k = -1
    for i in range(n):
        if diff[i] <= 0:
            k = i
    # Lambda = (cumsum[k] - target_sum) / (k+1) if k >= 0
    if k >= 0:
        lambda_val = diff[k] / (k + 1)
        # Apply lambda to h_sorted
        h_new = np.clip(h_sorted - lambda_val, 0, 1)
        # Reassign back to h
        h_sorted = h_new
        # Map back to original indices
        h = h.copy()
        idx = np.argsort(h)[::-1][:k+1]
        h[idx] = h_sorted[:k+1]
    else:
        # Must reduce every element to meet sum
        lambda_val = current_sum - target_sum
        h = np.clip(h - lambda_val, 0, 1)
    return h
