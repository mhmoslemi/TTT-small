import numpy as np
import time

def softmax_weights(f_values, gamma):
    """Compute softmax weights for given values and temperature gamma."""
    f_array = np.array(f_values)
    exp_f = np.exp(gamma * f_array)
    return exp_f / np.sum(exp_f)

def compute_gradient(h, dx, shifts):
    """Compute gradient of the smooth max surrogate."""
    grad = np.zeros_like(h)
    for s in shifts:
        g = np.zeros_like(h)
        if s >= 0:
            g[s:] = 1 - h[:-s] if s > 0 else 1 - h
        else:
            g[:s] = 1 - h[-s:] if s < 0 else 1 - h
        if s < 0:
            g[s:] -= h[:-s] if s < 0 else h
        grad += g
    return grad * dx

def project_to_simplex(v, target_sum):
    """Project vector v onto the simplex with sum = target_sum."""
    v = np.array(v)
    n = len(v)
    # Bisection on lambda to find the projection
    def phi(lam):
        return np.sum(np.clip(v - lam, 0, 1)) - target_sum

    a, b = np.min(v), np.max(v) + 1.0
    for _ in range(40):
        lam = (a + b) / 2
        if phi(lam) > 0:
            a = lam
        else:
            b = lam
    lam = (a + b) / 2
    return np.clip(v - lam, 0, 1)

def upsample_initial(h0, n):
    """Upsample initial_h_values to length n using linear interpolation."""
    old_x = np.linspace(0, 2, len(h0))
    new_x = np.linspace(0, 2, n)
    h_upsampled = np.interp(new_x, old_x, h0)
    return h_upsampled

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    correlation = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(correlation) * dx
    return max_overlap
