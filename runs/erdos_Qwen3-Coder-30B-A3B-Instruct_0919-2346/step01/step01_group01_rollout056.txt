import numpy as np
import time
import random
from scipy.optimize import differential_evolution, minimize

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using full convolution
    overlap = np.correlate(h, 1 - h, mode='full')
    # The maximum value scaled by dx
    c5 = np.max(overlap) * dx
    return c5

def evaluate_h(h):
    """Evaluate the C5 value for a given h"""
    return c5_of(h)

def constrain_sum(h, target_sum):
    """Adjust h to maintain the sum constraint"""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    scale_factor = target_sum / current_sum
    return h * scale_factor

def project_to_bounds(h):
    """Project h to [0,1] bounds"""
    return np.clip(h, 0, 1)

def random_perturbation(h, step_size):
    """Generate a random perturbation maintaining sum constraint"""
    h_new = h.copy()
    n = len(h)
    
    # Sample two random indices
    i, j = random.sample(range(n), 2)
    
    # Determine feasible range for delta
    delta_low = max(-h_new[i], h_new[j] - 1.0)
    delta_high = min(1.0 - h_new[i], h_new[j])
    
    if delta_low >= delta_high:
        return h_new
    
    # Sample delta
    delta = random.uniform(max(-step_size, delta_low), min(step_size, delta_high))
    
    h_new[i] += delta
    h_new[j] -= delta
    
    return h_new

    h_final = project_to_bounds(result.x)

        else:
            step_size = max(0.001, step_size * 0.99)  # Decrease step size when not improving
    
    return current_h, current_c5
