import numpy as np
import time

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral via full convolution of h with (1-h)
    overlaps = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum of the full correlation vector
    # scaled by dx
    c5 = np.max(overlaps) * dx
    return c5

def compute_c5_and_worst_info(h):
    """Return (c5, offset, max_val, i, j, hi_val, hj_val, contrib)"""
    N = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    max_idx = np.argmax(corr)
    max_val = corr[max_idx]
    offset = max_idx - (N - 1)
    # contributions vector:
    if offset >= 0:
        hi = h[:N - offset]
        hj = h[offset:]
        contrib = hi * (1 - hj)
        i_local = np.argmax(contrib)
        i_index = i_local
        j_index = i_local + offset
    else:
        off = -offset
        hi = h[off:]
        hj = h[:N - off]
        contrib = hi * (1 - hj)
        i_local = np.argmax(contrib)
        i_index = i_local + off
        j_index = i_local
    return max_val * (2.0/N), offset, max_val * (2.0/N), i_index, j_index, hi[i_local], hj[i_local], contrib[i_local]

def propose_direct_move(h, i, j, step):
    """Propose a move by transferring mass from i to j"""
    delta = min(step, h[i], 1 - h[j])
    if delta <= 0:
        return None
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new

def target_descent(h, time_budget, start_time, max_iters=None):
    """Run targeted descent to improve h"""
    if max_iters is None:
        max_iters = 10000
    N = len(h)
    best_h = h.copy()
    best_c5 = c5_of(h)
    step = 0.5
    fails = 0
    it = 0
    while time.time() - start_time < time_budget and it < max_iters:
        it += 1
        _, _, cur_c5, i, j, hi_val, hj_val, _ = compute_c5_and_worst_info(h)
        # Compute delta
        delta = min(step, h[i], 1 - h[j])
        if delta > 0:
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            c5_new = c5_of(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
                step = min(step * 1.1, 0.5)
                fails = 0
                continue
        # Try moving in opposite direction
        delta2 = min(step, h[j], 1 - h[i])
        if delta2 > 0:
            h_new2 = h.copy()
            h_new2[j] -= delta2
            h_new2[i] += delta2
            c5_new2 = c5_of(h_new2)
            if c5_new2 < best_c5:
                best_c5 = c5_new2
                best_h = h_new2.copy()
                h = h_new2
                step = min(step * 1.1, 0.5)
                fails = 0
                continue
        # No improvement, reduce step
        step *= 0.9
        fails += 1
        # Occasionally random perturbation
        if fails > 500 and np.random.rand() < 0.1:
            idxs = np.random.choice(N, size=2, replace=False)
            delta_r = step * min(h[idxs[0]], 1 - h[idxs[1]])
            if delta_r > 0:
                h = propose_direct_move(h, idxs[0], idxs[1], step)
                fails = 0
    return best_h, best_c5

def project_onto_simplex_box(v, total):
    """Project v onto set {x: 0 <= x_i <= 1, sum x_i = total}"""
    # Bisection on lambda
    lo = v.min() - 1
    hi = v.max() + 1
    for _ in range(60):
        lam = (lo + hi) / 2
        x = np.clip(v - lam, 0, 1)
        s = x.sum()
        if s > total:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2, 0, 1)

def subgradient_descent(h, max_iters, start_step=0.5):
    """Run subgradient descent with projection"""
    N = len(h)
    total = N / 2
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = c5_of(h_cur)
    step = start_step
    for it in range(max_iters):
        offset,_,_,_,_,_,_,_ = compute_c5_and_worst_info(h_cur)
        # compute gradient for worst shift only
        g = np.ones(N)
        d = offset
        if d >= 0:
            g[:N-d] -= h_cur[d:]
            g[d:] -= h_cur[:N-d]
        else:
            off = -d
            g[off:] -= h_cur[:N-off]
            g[:N-off] -= h_cur[off:]
        # step size
        step = start_step * (1 - it / max_iters) + 0.01
        v = h_cur - step * g
        h_cur = project_onto_simplex_box(v, total)
        c5 = c5_of(h_cur)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_cur.copy()
    return best_h, best_c5
