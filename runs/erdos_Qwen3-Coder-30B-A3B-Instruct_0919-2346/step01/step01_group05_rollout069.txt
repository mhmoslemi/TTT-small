import numpy as np
import time
from math import sqrt

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value scaled by dx
    max_overlap = np.max(correlation) * dx
    return max_overlap

def sturmian_binary(n, alpha=(sqrt(5)-1)/2):
    """Generate a balanced binary sequence using irrational rotation."""
    h = np.zeros(n, dtype=int)
    ones_count = 0
    for i in range(n):
        t = (i * alpha) % 1
        if t < 0.5:
            h[i] = 1
            ones_count += 1
    
    # Adjust to get exactly n/2 ones
    if ones_count > n // 2:
        # Flip some 1s to 0s
        flip_indices = np.where(h == 1)[0][:ones_count - n // 2]
        h[flip_indices] = 0
    elif ones_count < n // 2:
        # Flip some 0s to 1s
        flip_indices = np.where(h == 0)[0][:n // 2 - ones_count]
        h[flip_indices] = 1
    
    return h

def compute_overlaps(h):
    """Compute overlap values for all shifts."""
    n = len(h)
    dx = 2.0 / n
    # Compute the correlation as np.correlate(h, 1-h, mode='full')
    correlation = np.correlate(h, 1 - h, mode="full")
    # Convert to actual overlap
    overlaps = correlation * dx
    return overlaps

def compute_delta_shifts(i, j, h, shifts_to_compute):
    """Compute the change in overlap for each shift if we swap h[i] and h[j]"""
    n = len(h)
    delta_shifts = np.zeros(len(shifts_to_compute))
    h_inv = np.array([1 - x for x in h])

    # Compute for all relevant shifts: for shift s, i contributes if i in overlapping region
    for s_idx, s in enumerate(shifts_to_compute):
        delta = 0
        # Original contribution from i as first factor: 1 * (1 - h[i+s])
        # After swap: 0 * (1 - h[i+s]) = 0, so loss = -(1 - h[i+s])
        if 0 <= i + s < n:
            delta -= (1 - h[i + s])
        # Contribution from i as second factor: h[i-s] * (1 - 1) = 0
        # After swap: h[i-s] * (1 - 0) = h[i-s], so gain = + h[i-s] if valid
        if 0 <= i - s < n:
            delta += h[i - s]
        # Similar for j (0 becomes 1)
        # j as first factor: h[j] * (1 - h[j+s]) = 0 * (1 - h[j+s]) = 0
        # After swap: 1 * (1 - h[j+s]) = (1 - h[j+s]), so gain = (1 - h[j+s])
        if 0 <= j + s < n:
            delta += (1 - h[j + s])
        # j as second factor: h[j-s] * (1 - 0) = h[j-s]
        # After swap: h[j-s] * (1 - 1) = 0, so loss = - h[j-s]
        if 0 <= j - s < n:
            delta -= h[j - s]

        delta_shifts[s_idx] = delta

    return delta_shifts

def local_swap_search(h, n_points, time_limit):
    """Perform targeted local swap search."""
    start_time = time.time()
    dx = 2.0 / n_points
    h = h.copy()
    
    # Pre-compute overlaps for all shifts and get max
    overlaps = compute_overlaps(h)
    
    # Find worst shift
    best_max = np.max(overlaps)
    worst_shift_idx = np.argmax(overlaps)
    best_h = h.copy()
    
    # List of all shifts for fast indexing
    shifts = np.arange(-n_points + 1, n_points)
    
    # Tabu list: avoid repeated swaps for a few iterations
    tabu_list = set()
    tabu_duration = 50
    
    iteration = 0
    no_improvement_count = 0
    max_no_improvement = 3000
    
    # Up to 10000 iterations or time allowance
    max_iterations = 10000
    time_limit_seconds = time_limit * 0.95  # leave some room
    
    while iteration < max_iterations and (time.time() - start_time) < time_limit_seconds:
        if no_improvement_count > max_no_improvement:
            # Perturb the solution slightly
            num_flip = max(1, n_points // 40)
            flip_positions = np.random.choice(n_points, num_flip, replace=False)
            for pos in flip_positions:
                h[pos] = 1 - h[pos]
            # Renormalize to maintain balance
            ones_count = np.sum(h)
            if ones_count != n_points / 2:
                # Adjust balance
                if ones_count > n_points / 2:
                    flipped_positions = np.where(h == 1)[0]
                    flip_back = flipped_positions[:int(ones_count - n_points / 2)]
                    h[flip_back] = 0
                else:
                    flipped_positions = np.where(h == 0)[0]
                    flip_back = flipped_positions[:int(n_points / 2 - ones_count)]
                    h[flip_back] = 1
            no_improvement_count = 0
            iteration = 0
            continue
        
        # Evalulate all overlaps again to update the max
        overlaps = compute_overlaps(h)
        current_max = np.max(overlaps)
        if current_max < best_max:
            best_max = current_max
            best_h = h.copy()
        
        # Choose worst shift
        worst_shift_idx = np.argmax(overlaps)
        worst_shift = shifts[worst_shift_idx]
        
        # Find good 1's and 0's to swap
        one_positions = np.where(h == 1)[0]
        zero_positions = np.where(h == 0)[0]
        
        if len(one_positions) == 0 or len(zero_positions) == 0:
            break
            
        # Samplings per iteration for efficiency:
        # Use ~20 swaps per iteration, remember 10 best successful ones
        tries = 0
        best_swaps = []
        while len(best_swaps) < 10 and tries < 20:
            i = np.random.choice(one_positions)
            j = np.random.choice(zero_positions)
            
            # Skip if previously recently swapped
            if (i, j) in tabu_list or (j, i) in tabu_list:
                tries += 1
                continue
            
            delta = compute_delta_shifts(i, j, h, [worst_shift])
            if delta[0] < 0 or np.random.random() < 0.01:  # accept or with some probability
                # Update overlapping values
                # Get all relevant shifts to update
                relevant_shifts = []
                for s in range(max(0, -n_points + 1 + 1), min(n_points, n_points - 1)):
                    # If i or j contributes to shift s, add it to list
                    c1 = (0 <= (i + s) < n_points)
                    c2 = (0 <= (i - s) < n_points)
                    c3 = (0 <= (j + s) < n_points)
                    c4 = (0 <= (j - s) < n_points)
                    if c1 or c2 or c3 or c4:
                        relevant_shifts.append(np.searchsorted(shifts, s))
                # Actually inclusive range of shifts we care about:
                min_s = -n_points + 1
                max_s = n_points - 1
                rel_shifts = shifts[(shifts >= min_s) & (shifts <= max_s)]
                
                # Swap h[i] and h[j]
                h[i], h[j] = h[j], h[i]
                
                # Update overlaps 
                new_overlaps = compute_overlaps(h)
                new_max = np.max(new_overlaps)
                
                # If improvement, accept new h and new overlaps
                if new_max < current_max or np.random.random() < 0.1:
                    overlaps = new_overlaps
                    current_max = new_max
                else:
                    # Revert the swap
                    h[i], h[j] = h[j], h[i]
                
                # Add to tabu list
                tabu_list.add((i, j))
                if len(tabu_list) > 1000:
                    # Rotate tabu list
                    tabu_list = set(list(tabu_list)[1:])
            
            tries += 1
         
        iteration += 1
        if current_max >= best_max:
            no_improvement_count += 1
        else:
            no_improvement_count = 0
    
    overlaps = compute_overlaps(best_h)
    final_max = np.max(overlaps)
    return best_h, final_max

def continuous_polish(h, time_limit):
    """Polish final solution with projected gradient descent."""
    start_time = time.time()
    
    # Gradient: derivative of financial loss when we vary h
    # This is more complex but we approximate by local first differences
    dx = 2.0 / len(h)
    
    iteration = 0
    while iteration < 2000 and (time.time() - start_time) < time_limit * 0.5:
        # Re-evaluate overlaps
        overlaps = compute_overlaps(h)
        max_overlap = np.max(overlaps)
        max_shift_idx = np.argmax(overlaps) 
        worst_shift = np.arange(-len(h) + 1, len(h))[max_shift_idx] 

        # Compute hypothetical change in overlap if we perturb value h[i] 
        # Approximate gradient by finite difference with epsilon = 1e-3
        epsilon = 1e-3
        grad_h = np.zeros(len(h))
        # Only for good candidates of significant impact
        for i in range(len(h)):
            h_plus = h.copy() 
            h_plus[i] += epsilon
            h_plus = np.clip(h_plus, 0, 1)
            h_plus_sum = np.sum(h_plus) * dx
            h_plus /= h_plus_sum / (2.0 / len(h))  # Normalize

            overlaps_plus = compute_overlaps(h_plus) 
            max_plus = np.max(overlaps_plus)
            if max_plus < max_overlap + 1e-10:   # Not much worse
                grad_h[i] = (max_plus - max_overlap) * 1000000  # Boost gradient
        
        # Gradient descent step
        lr = 1e-2
        h_new = h - lr * grad_h
        h_new = np.clip(h_new, 0, 1) 
        h_new_sum = np.sum(h_new) * dx
        h_new = h_new * (2.0 / len(h)) / h_new_sum  # keep constraint

        # Keep update if it does decrease the max overlap
        if np.sum(np.abs(h_new - h)) > 1e-10:
            h = h_new.copy()
        else:
            # No change
            break
        iteration += 1
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Phase 1: Coarse grid - use n = 120 to find good structure
    n_points = 120
    h = sturmian_binary(n_points)
    h, c5_1 = local_swap_search(h, n_points, budget_s * 0.4)
    
    # Phase 2: Refine on upscaled grid - n = 240
    n_points = 240
    h_up = np.zeros(n_points)
    for i in range(0, n_points, 2):
        h_up[i:i+2] = h[i//2]
    h_up = np.clip(h_up, 0, 1)
    ones_count = np.sum(h_up)
    h_up = h_up * (n_points/2) / ones_count  # scale to maintain sum
    h_up, c5_2 = local_swap_search(h_up, n_points, budget_s * 0.45)

    # Phase 3: Continuous polishing
    h_final = h_up.copy()
    h_final, c5_final = local_swap_search(h_final, n_points, budget_s * 0.05)
    h_final = continuous_polish(h_final, budget_s * 0.05)
    final_c5 = evaluate_erdos_solution(h_final)
    
    return (h_final.tolist(), final_c5, len(h_final))
