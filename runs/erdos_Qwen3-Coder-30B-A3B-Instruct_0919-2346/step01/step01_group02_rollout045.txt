import numpy as np
import time
from scipy.optimize import minimize
from scipy.special import logsumexp

def c5_of(h):
    """Compute the maximum overlap integral for the given step function h."""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    c = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(c) * dx
    return c5

def project_to_feasible(h, N):
    """Project h onto the feasible set: 0 <= h[i] <= 1 and sum(h) = N/2."""
    # Initial project to box constraints
    h_proj = np.clip(h, 0, 1)
    # Water-filling to enforce sum constraint
    sum_h = np.sum(h_proj)
    if np.abs(sum_h - N/2) < 1e-10:
        return h_proj
    
    # Create auxiliary array
    v = h_proj.copy()
    # Binary search on lambda
    lambda_low, lambda_high = -10.0, 10.0
    for _ in range(50):
        lambda_c = (lambda_low + lambda_high) / 2.0
        sum_v = np.sum(np.clip(v + lambda_c, 0, 1))
        if sum_v < N/2:
            lambda_low = lambda_c
        else:
            lambda_high = lambda_c
    lambda_c = (lambda_low + lambda_high) / 2.0
    # Reconstruct h
    h_proj = np.clip(v + lambda_c, 0, 1)
    # Small adjustment to exactly fulfill the sum constraint
    diff = np.sum(h_proj) - N/2
    # Adjust any free variables to reduce error
    free_indices = np.where((h_proj > 0) & (h_proj < 1))[0]
    # Simple redistribution
    if len(free_indices) > 0:
        criterion = np.diff(h_proj[free_indices])
        if len(criterion) > 0:
            # Try adding a small amount to avoid boundary
            for i in range(min(int(abs(diff)*100), len(free_indices))):
                idx = free_indices[i % len(free_indices)]
                if h_proj[idx] < 1.0:
                    delta = min(1.0 - h_proj[idx], abs(diff) / len(free_indices))
                    h_proj[idx] += delta
            # Now subtract from other entries if needed
            diff = np.sum(h_proj) - N/2
            if diff > 0:
                for i in range(min(int(diff*100), len(free_indices))):
                    idx = free_indices[i % len(free_indices)]
                    if h_proj[idx] > 0:
                        delta = min(h_proj[idx], abs(diff) / len(free_indices))
                        h_proj[idx] -= delta
    # Final check
    return h_proj

def smooth_surrogate(h, eps, dx=2.0/140):
    """Compute approximate smooth surrogate value using log-sum-exp."""
    # Compute correlations for all shifts
    c = np.correlate(h, 1 - h, mode='full')
    c = c * dx  # Scale by dx
    # apply log-sum-exp
    log_sums = logsumexp(c / eps) + np.log(len(c))
    # This gives log of the sum of exponentials; we want its argument scaled by dx
    value = eps * log_sums
    return value

def compute_grad(h, eps, dx=2.0/140):
    """Compute gradient of smooth surrogate using correlation."""
    # Full correlation
    c = np.correlate(h, 1 - h, mode='full')  # shape (2N-1,)
    c = c * dx  # Scale by dx
    
    # Softmax weights
    c_max = np.max(c)
    exp_c = np.exp(c - c_max)
    weights = exp_c / np.sum(exp_c)
    
    # Gradient, compute using correlation
    grad = np.zeros_like(h)
    for s in range(len(c)):
        # We have c_s = dx * sum_i h[i] * (1 - h[i+s]) 
        #              = dx * ( sum_i h[i] - sum_i h[i]*h[i+s] )
        #              where h[end:k] etc. zero-padded
        if s < len(h):
            grad += weights[s] * (dx * (1 - h[s] - h[0]))  # [i+s] in h or 0,
        if s < len(h) and 0 <= s:
            grad += weights[s] * (dx * (1 - h[0] - h[-s]))  # [i-s] in h or 0,
    return grad

def lbfgsb_smooth_optimize(h0, N, eps_sequence, max_iter=200):
    """Optimize using L-BFGS-B with smooth surrogate"""
    h = h0.copy()
    h = project_to_feasible(h, N)
    for eps in eps_sequence:
        def obj(h):
            return smooth_surrogate(h, eps)
        def grad(h):
            return compute_grad(h, eps)
        
        # Run L-BFGS-B
        res = minimize(obj, h, method='L-BFGS-B', jac=grad, 
                       bounds=[(0,1)]*N, options={'maxiter':max_iter, 'ftol':1e-9, 'gtol':1e-6})
        h = res.x
        if not res.success:
            h = project_to_feasible(h, N)
        h = project_to_feasible(h, N)
    return h

def seed_generators(N):
    seeds = []
    # Uniform seed
    h_uniform = np.ones(N) * 0.5
    h_uniform = project_to_feasible(h_uniform, N)
    seeds.append(('uniform', h_uniform))
    
    # Two-peak seed
    a = 0.24
    u = 0.9 
    v = 0.1
    h_two_peak = np.ones(N)
    x_vals = np.linspace(0, 2, N)
    mask = (x_vals < a) | (x_vals >= 2 - a)
    h_two_peak[mask] = u
    h_two_peak[~mask] = v
    h_two_peak = project_to_feasible(h_two_peak, N)
    seeds.append(('two_peak', h_two_peak))
    
    # One-block binary seed
    h_binary = np.zeros(N)
    h_binary[:N//2] = 1
    h_binary = project_to_feasible(h_binary, N)
    seeds.append(('binary', h_binary))
    
    # Two-block binary seed
    h_binary2 = np.zeros(N)
    h_binary2[:N//4] = 1
    h_binary2[3*N//4:] = 1
    h_binary2 = project_to_feasible(h_binary2, N)
    seeds.append(('binary2', h_binary2))
    
    # Random continuous seed
    np.random.seed(42)
    h_rand = np.random.rand(N) * 0.8 + 0.1
    h_rand = project_to_feasible(h_rand, N)
    seeds.append(('random', h_rand))
    
    return seeds

def refine_solution(h_final, N, dx):
    """Refine using sub-gradient direction of max shift."""
    # Refinement using exact max
    c = np.correlate(h_final, 1 - h_final, mode='full')
    s_max = np.argmax(c)
    # The subgradient at that shift
    g = np.zeros(N)
    for i in range(N):
        if i-s_max >= 0 and i-s_max < N:
            # Left-shift (negative shift)
            g[i] = dx * (1 - h_final[i-s_max] - h_final[i])
        if i+s_max < N and i+s_max >= 0:
            # Right-shift (positive shift)
            g[i] = dx * (1 - h_final[i+s_max] - h_final[i])
            
    # Take steps in negative gradient direction (i.e., decrease objective)
    h_new = h_final - 0.03 * g
    h_new = project_to_feasible(h_new, N)
    return h_new

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    N = 140
    dx = 2.0 / N
    start_time = time.time()
    
    # Set time limits
    seed_generation_time = 0.05 * budget_s
    continuation_time = 0.70 * budget_s
    multi_start_time = 0.10 * budget_s
    polish_time = 0.10 * budget_s
    safety_time = 0.05 * budget_s
    
    # Build seeds
    seed_list = seed_generators(N)
    best_c5 = float('inf')
    best_h = None
    
    max_iter_per_seed = 150
    
    # Continuation scheme
    eps_sequence = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    
    print("Starting multi-start optimization...")
    
    # Process each seed
    for name, h_seed in seed_list:
        if time.time() - start_time > continuation_time:
            break
        h_opt = lbfgsb_smooth_optimize(h_seed.copy(), N, eps_sequence, max_iter_per_seed)
        
        # Evaluate final C5
        c5 = c5_of(h_opt)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()
    
    print(f"Initial best: {best_c5}")
    
    # Final polishing phase
    refine_start = time.time()
    # Run sub-gradient descent
    h_refined = best_h.copy()
    max_iter_refine = 200
    iter_counter = 0
    
    while (time.time() - refine_start < polish_time) and iter_counter < max_iter_refine:
        last_c5 = c5_of(h_refined)
        # Compute overlap integrals
        c = np.correlate(h_refined, 1 - h_refined, mode='full')
        s_best = np.argmax(c)
        # Sub-gradient
        N_half = len(h_refined)
        g = np.zeros(N_half)
        for i in range(N_half):
            if i+s_best < N_half:
                g[i] += dx * (1 - h_refined[i+s_best] - h_refined[i])
            if i-s_best >= 0:
                g[i] += dx * (1 - h_refined[i-s_best] - h_refined[i])

        # Take step
        eta = 0.03
        h_new = h_refined - eta * g
        h_new = project_to_feasible(h_new, N)
        new_c5 = c5_of(h_new)
        if new_c5 < last_c5:
            h_refined = h_new
        iter_counter += 1
    
    # Final evaluation
    best_c5_final = c5_of(h_refined)
    print(f"Final polished C5: {best_c5_final}")
    
    return (h_refined, best_c5_final, N)
