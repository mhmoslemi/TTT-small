import numpy as np
import time
import random
from scipy.optimize import minimize


def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    overlap = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(overlap) * dx
    return c5


def project_feasible(h):
    """Project h to feasible region with exact integral constraint"""
    dx = 2.0 / len(h)
    current_sum = np.sum(h) * dx
    h_proj = h * (1.0 / current_sum)
    return h_proj


def propose_move(h, max_step, step_factor=1.0):
    """Propose a new feasible h vector by adjusting two elements"""
    h_new = h.copy()
    N = len(h)
    
    # Try to find a valid move
    max_tries = 100
    for _ in range(max_tries):
        i = random.randint(0, N - 1)
        j = random.randint(0, N - 1)
        if i == j:
            continue
            
        # Determine feasible interval for δ
        delta_low = max(-h_new[i], h_new[j] - 1)
        delta_high = min(1 - h_new[i], h_new[j])
        
        if delta_low >= delta_high:
            continue
            
        # Sample δ from uniform distribution over feasible range
        step = min(max_step * step_factor, delta_high - delta_low)
        if step <= 0:
            continue
        delta = random.uniform(delta_low, delta_high)
        
        # Apply move
        h_new[i] += delta
        h_new[j] -= delta
        
        # Normalize to maintain constraint
        h_new = project_feasible(h_new)
        
        return h_new
    
    # If no valid move found, return original
    return h.copy()


def gradient_based_local_search(initial_h, time_budget_s):
    """Use gradient-based optimization for local refinement"""
    start = time.time()
    
    # Since h is discrete, use gradient info heuristically
    h_current = initial_h.copy()
    c_current = c5_of(h_current)
    
    best_h = h_current.copy()
    best_c = c_current
    
    # Try to improve with a local search variant
    max_evals = 1000
    eval_count = 0
    
    while eval_count < max_evals and time.time() - start < time_budget_s:
        # Create a small perturbation
        h_perturbed = h_current.copy()
        if len(h_perturbed) > 10:
            # Perturb a few points
            N = len(h_perturbed)
            for _ in range(min(5, N // 10)):
                idx = random.randint(0, N-1)
                h_perturbed[idx] = max(0, min(1, h_perturbed[idx] + random.gauss(0, 0.01)))
            
            # Ensure constraints are met
            h_perturbed = project_feasible(h_perturbed)
            c_perturbed = c5_of(h_perturbed)
            
            if c_perturbed < c_current:
                h_current = h_perturbed
                c_current = c_perturbed
                if c_perturbed < best_c:
                    best_c = c_perturbed
                    best_h = h_perturbed.copy()
        
        eval_count += 1
    
    return best_h, best_c

        )

    # Coarse optimization: use simulated annealing for global search
    # Modified - use a much more focused search
    
    # Parameters
    T_start = 0.02
    T_end = 1e-8
    
    # Eligible points to adjust (to avoid too sparse sampling)
    cur = h0.copy()
    c_cur = c5_of(cur)
    best = cur.copy()
    c_best = c_cur
    
    # Main loop with phase-wise cooling
    total_time = 0
    phase = 0
    phase_time = budget_s / 3.0
    
    while phase < 3 and total_time < budget_s:
        # Phase-specific parameters
        phase_start = time.time()
        if phase == 0:
            T = T_start * 0.5
            step = 0.1
        elif phase == 1:
            T = T_start * 0.2
            step = 0.05
        else:
            T = T_start * 0.05
            step = 0.02

        while total_time < budget_s and time.time() - phase_start < phase_time:

            h_proposed = propose_move(cur, step, 1.0)
