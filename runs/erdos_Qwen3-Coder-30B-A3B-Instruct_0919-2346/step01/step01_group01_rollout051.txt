import numpy as np
import time
import cvxpy as cp

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using full convolution
    overlap = np.correlate(h, 1 - h, mode='full')
    # The maximum value scaled by dx
    c5 = np.max(overlap) * dx
    return c5

    # Variables: x[i] binary indicating h[i] = 1
    x = cp.Variable(N, boolean=True)
    
    # Constraint: sum of x equals N/2 (half the interval's length)
    constraints = [cp.sum(x) == N // 2]
    
    # Objective variable
    C = cp.Variable()
    
    # For each shift ℓ = 1 to N-1 (we'll treat ℓ = 0 separately since it's always 0)
    for ell in range(1, N):
        length = N - ell  # Length of overlapping region
        if length <= 0:
            continue
            
        # H_ell = number of ones in first length entries (overlapping region)
        H = cp.sum(x[:length])
        
        # Introduce variables y[i] = x[i] * x[i+ell] for i=0..length-1
        # These shifts make sense only within the overlapping window
        y = cp.Variable(length, boolean=True)
        
        # Y_ell constraints for product variables
        constraints.append(y <= x[:length])  # y[i] <= x[i]
        constraints.append(y <= x[ell:])     # y[i] <= x[i+ell]
        constraints.append(y >= x[:length] + x[ell:] - 1)  # y[i] >= x[i] + x[i+ell] - 1
        
        # O_ell = number of overlapping pairs (ones at distance ell)
        O = cp.sum(y)
        
        # c_ell = H_ell - O_ell (overlap integral part)
        c = H - O
        
        # Constraint: c <= C to enforce that C is an upper bound on all c_ell
        constraints.append(c <= C)
        
        # Constraint: c >= 0 to ensure non-negativity (valid for binaries)
        # Although c will always be non-negative given set constraints, add for completeness
        # Actually this constraint is implied.
    
    # Define the problem
    prob = cp.Problem(cp.Minimize(C), constraints)
    
    # Solve with CBC solver; set time limit to leave some time for fallback
    remaining_time = budget_s - (time.time() - start_time)
