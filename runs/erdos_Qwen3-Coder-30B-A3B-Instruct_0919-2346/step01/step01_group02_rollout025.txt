import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def evaluate_c5(h):
    """Evaluate the C5 constant using the original non-smooth objective."""
    dx = 2.0 / len(h)
    c = np.correlate(h, 1 - h, mode='full')
    return np.max(c) * dx

def project_to_box_and_sum(h, target_sum):
    """Project h onto the feasible set: [0,1]^n and sum(h) = target_sum."""
    # Water-filling for sum constraint
    v = h.copy()
    # Ensure all entries are in [0,1]
    v = np.clip(v, 0, 1)
    
    # Binary search for lambda such that sum(clamp(v + lambda)) = target_sum
    def F(lam):
        return np.sum(np.clip(v + lam, 0, 1))
    
    low, high = -10.0, 10.0
    for _ in range(50):
        mid = (low + high) / 2
        val = F(mid)
        if val < target_sum:
            low = mid
        else:
            high = mid
    lambda_opt = (low + high) / 2
    
    # Apply projection
    h_proj = np.clip(v + lambda_opt, 0, 1)
    
    # Adjust final values to make sum exactly target_sum
    diff = target_sum - np.sum(h_proj)
    if abs(diff) > 1e-10:
        # Redistribute the difference among free variables
        free = np.where((h_proj > 0) & (h_proj < 1))[0]
        if len(free) > 0:
            inc = diff / len(free)
            h_proj[free[:min(int(np.ceil(abs(diff)/inc)), len(free))]] += inc * np.sign(diff)
        else:
            # Just take the first few to adjust
            h_proj[:min(int(np.ceil(abs(diff))), len(h_proj))] += inc * np.sign(diff)
    
    return np.clip(h_proj, 0, 1)

def smooth_objective_and_grad(h, eps, dx):
    """Compute smooth objective F_eps and its gradient."""
    # Compute c_s = dx * sum(h[i] * (1 - h[i+s]))
    # c_s corresponds to correlation of h with 1-h (forward shift) 
    c = correlate(h, 1 - h, mode='full') 
    c *= dx
    
    # Soft max weights
    max_c = np.max(c)
    # Protect against overflow
    exp_c = np.exp((c - max_c) / eps)
    Z = np.sum(exp_c)
    weights = exp_c / Z
    
    # Compute gradient 
    # ∇C_s = dx·(1 − h[i+s] − h[i−s]) 
    # ∇F_eps = Σ_s weights_s · ∇C_s
    grad = np.zeros_like(h)
    n = len(h)
    # For s=0 to 2n-2:
    for s in range(2 * n - 1):
        c_s = c[s]
        w_s = weights[s]
        # Need to compute derivative of c_s = dx mult sum h[i] (1 - h[i+s])
        # Recall forward correlation:
        # [h[0],h[1],...,h[n-1]] with [1-h[0],1-h[1],...,1-h[n-1]]
        # so shift amount s contributes to index k in the result
        # diff = dx * (1 - h[i+s] - h[i-s])
        # we don't consider negative indices as part of the sum
        # But this backwards view is not trivial; best: use shift vectors
        
        # extract indices for current shift s
        left = max(0, s - n + 1)  
        right = min(s, n - 1)
        # valid indices in h corresponding to shift s: i and i-s
        shift = s - (n - 1)  # actual shift
        
        # only sum over valid indices where both h[i] and h[i-shift] exist
        # using second pass with the correlation
        # Non-optimized but precise: reuse computationally part of correlate 
        if s < n:
            # This corresponds to corr[h, 1-h] from index s
            # indices of h are (i = 0 to n-1)
            # h[i] * (1 - h[i+s]) for all s such that 0 <= i <= n-1, 0 <= i+s <= n-1
            # i ranges from max(0, -s) to min(n-1, n-1-s) = min(n-1, n-1-s)
            for i in range(max(0, -shift), min(n, n - shift)): 
                j = i + shift
                if 0 <= j < n: 
                    # ∂c_s/∂h[i] = dx * (1 - h[j] - h[i])
                    grad[i] += w_s * dx * (1 - h[j] - h[i])
                    grad[j] += w_s * dx * (-h[i])
        else:
            # Reverse shift to avoid confusion
            # This shift is negative: h[i] * (1 - h[i + shift]) where shift is negative
            shift = s - (n - 1)
            for i in range(max(0, -shift), min(n, n - shift)):
                j = i + shift
                if 0 <= j < n: 
                    grad[i] += w_s * dx * (1 - h[j] - h[i])
                    grad[j] += w_s * dx * (-h[i])

    # Actual smooth value
    F_eps = eps * np.log(Z) + max_c  

    return F_eps, grad

def continuation_optimize(h_start, eps_list, dx, max_iter=150, tol=1e-6):
    """Continuation-based smooth optimization with bound constraints."""
    h = h_start.copy()
    
    for eps in eps_list:
        # L-BFGS-B optimizer
        # Only objective and gradient
        def fun_and_grad(h_vec):
            val, grad = smooth_objective_and_grad(h_vec, eps, dx)
            return val, grad

        # Every iteration
        converged = False
        opt_result = minimize(fun_and_grad, h, method='L-BFGS-B', jac=True,
                              bounds=[(0, 1)]*len(h), 
                              options={'maxiter': max_iter,
                                       'gtol': tol})
        if opt_result.success:
            h = opt_result.x
        else:
            h = opt_result.x

        # Project onto feasible set
        h = project_to_box_and_sum(h, len(h)/2) 
        
    return h

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    N = 140
    dx = 2.0 / N
    start = time.time()
    
    # Seed generation strategies
    seeds = []

    # Uniform seed
    h_uniform = np.ones(N) * 0.5
    seeds.append(("uniform", h_uniform.copy()))
    
    # Two-peak seeds
    peaks = [0.12, 0.18, 0.24, 0.30]  # different a values
    for a in peaks:
        h_two_peak = np.ones(N)
        x_vals = np.linspace(0, 2, N)
        u = 0.9
        v = 0.1
        mask = (x_vals < a) | (x_vals >= 2 - a)
        h_two_peak[mask] = u
        h_two_peak[~mask] = v
        h_two_peak = h_two_peak * (N/2) / np.sum(h_two_peak)  # Renormalize
        seeds.append((f"two_peak_{a}", h_two_peak.copy()))
    
    # Binary seeds
    for i in range(5):
        h_binary = np.zeros(N)
        idx = np.random.choice(N, size=N//2, replace=False)
        h_binary[idx] = 1.0
        seeds.append((f"binary_{i}", h_binary.copy()))
    
    # Symmetric two-blocks binary seed
    h_sym = np.zeros(N)
    h_sym[:N//2] = 1.0
    seeds.append(("symmetric_binary", h_sym.copy()))
    
    # Random continuous seed with renormalization
    h_random = np.random.rand(N)
    h_random = h_random * (N/2) / np.sum(h_random)
    seeds.append(("random", h_random.copy()))

    # Continuation parameters
    eps_sequence = [0.10, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    
    # Best solution so far 
    best_h = None
    best_c5 = float('inf')
    
    # Multi-start: process each seed
    time_per_seed = 0.65 * budget_s / len(seeds)
    main_start = time.time()
    
    for name, h_seed in seeds:
        if time.time() - main_start > 0.9 * budget_s:
            break
        h_seed_feasible = project_to_box_and_sum(h_seed, N/2)
        h_opt = continuation_optimize(h_seed_feasible, eps_sequence, dx, max_iter=100)
        h_opt = project_to_box_and_sum(h_opt, N/2)  # Final correction
        c5 = evaluate_c5(h_opt)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()
    
    # Final polishing: sub-gradient descent
    time_left = budget_s - (time.time() - start)
    polishing_time = 0.1 * budget_s
    while time.time() - start < budget_s - 0.05:
        # Find s* that maximizes correlation for current h
        c = correlate(best_h, 1 - best_h, mode='full')
        s_star = np.argmax(c)
        # Compute sub-gradient of C_s_star
        # derivative of h[i] term in C_{s*} = dx*(1 - h[i+s*] - h[i-s*])
        g = np.zeros_like(best_h)
        n = len(best_h)
        # s* affects terms h[i] and h[i+s*] (and maybe h[i-s*])
        # Note the shape of corr
        #      h[0]  h[1] ...
        # h[0] c[0]  c[1] ...
        # h[1] c[1]  c[2] ...  
        # ...
        # Element s* of c is correlation with shift s*
        shift = s_star - (n - 1)  
        # Apply as: 
        #  h[i] * (1 - h[i+s])
        if shift >= 0:
            # from index 0 to n-shift-1, i=some_start to (n-shift-1)
            for i in range(n - shift):
                idx = i
                jdx = i + shift
                if jdx < n:
                    g[idx] += -dx # coefficient from 1-h[j]
                    g[jdx] += -dx # coefficient from -h[i] 
        else:            
            # shift < 0; h[i] * (1-h[i-shift])
            for i in range(-shift, n):
                idx = i
                jdx = i + shift # jdx < i therefore shift < 0 means jdx > idx
                if jdx >= 0:
                    g[idx] += -dx # coefficient from 1-h[j]
                    g[jdx] += -dx # coefficient from -h[i]
        g = g * dx # account for the dx handling in correlation

        # Update 
        eta = 0.02
        h_new = best_h - eta * g
        h_new = project_to_box_and_sum(h_new, N/2)  # Projection
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
        else:
            # Convergence check?
            break

    return (best_h, best_c5, N)
