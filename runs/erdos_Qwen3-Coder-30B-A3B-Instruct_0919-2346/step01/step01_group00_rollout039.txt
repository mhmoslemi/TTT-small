import numpy as np
import time
import random
from scipy.optimize import differential_evolution

def c5_of(h):
    """Compute C5 = max_k integral h(x)(1-h(x+k)) dx."""
    dx = 2.0 / len(h)
    # Compute correlation: h * (1 - h) with full mode
    c = np.correlate(h, 1 - h, mode="full")
    # Multiply by dx to get continuous integral
    c *= dx
    # Return maximum value among all shifts
    return np.max(c)

def propose_move(h, max_step):
    """Generate a new feasible h by adjusting two entries."""
    h = h.copy()
    N = len(h)
    i, j = random.sample(range(N), 2)
    
    # Determine feasible interval for delta
    delta_low = max(-h[i], h[j] - 1.0)
    delta_high = min(1.0 - h[i], h[j])
    
    if delta_low >= delta_high:
        return h  # No valid move for this pair
    
    # Sample delta within allowed range
    step = min(max_step, delta_high - delta_low)
    delta = random.uniform(-step, step)
    delta = max(delta_low, min(delta, delta_high))
    
    h[i] += delta
    h[j] -= delta
    
    return h

def optimize_with_differential_evolution(h_start, budget_s):

        # Rescale h to satisfy constraint
        h_scaled = h * (len(h) / 2) / np.sum(h)
        return c5_of(h_scaled)

    bounds = [(0, 1) for _ in range(len(h_start))]
    
    # Store the best solution so far
    best_h = h_start
    best_c5 = c5_of(h_start)
    
    # Run differential evolution with limited time budget
    start = time.time()
    
    try:
        result = differential_evolution(
            objective,
            bounds,
            maxiter=int(1000 * budget_s / 600),
            popsize=15,
            mutation=(0.5, 1),
            recombination=0.7,
            atol=1e-6,
            rtol=1e-6,
            seed=42,
            callback=lambda x, convergence: time.time() - start > budget_s * 0.9,
            workers=1
        )
        
        if not result.success:
            return best_h, best_c5
            
        # Rescale and evaluate
        h_final = result.x * (len(h_start) / 2) / np.sum(result.x)
        c5_final = c5_of(h_final)
        
        if c5_final < best_c5:
            return h_final, c5_final
            
    except Exception:
        pass
    
    return best_h, best_c5
