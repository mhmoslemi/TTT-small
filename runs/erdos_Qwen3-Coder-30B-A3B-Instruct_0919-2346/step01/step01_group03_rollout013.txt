import numpy as np
import time
from scipy.optimize import differential_evolution

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral via full convolution of h with (1-h)
    overlaps = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum of the full correlation vector
    # scaled by dx
    c5 = np.max(overlaps) * dx
    return c5

def evaluate_constraint(h, n_points):
    """Check that the integral constraint is satisfied."""
    dx = 2.0 / n_points
    integral = np.sum(h) * dx
    return abs(integral - 1.0)

def objective(h, n_points):
    """Objective function to minimize."""
    return c5_of(h)

def bound_constraint(h, n_points):
    """Constraint function for bounds [0, 1]."""
    return np.min(h), 1.0 - np.max(h)

def integral_constraint(h, n_points):
    """Constraint function for integral = 1."""
    dx = 2.0 / n_points
    integral = np.sum(h) * dx
    return integral - 1.0
