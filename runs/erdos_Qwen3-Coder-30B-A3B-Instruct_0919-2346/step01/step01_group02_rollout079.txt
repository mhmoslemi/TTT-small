```python
import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize_scalar

def compute_C5(h):
    """Compute the maximum overlap integral for the given step function h."""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    c = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(c) * dx
    return c5

def build_index_map(N):
    """Build mapping from (i,s) pairs to indices, for y variables."""
    P = []
    for s in range(N):
        for i in range(N - s):
            P.append((i, s))
    pair_to_idx = {(i, s): j for j, (i, s) in enumerate(P)}
    return P, pair_to_idx

def waterfilling(h, target_sum):
    """Project h onto sum == target_sum and clip to [0,1]."""
    h = np.array(h)
    if np.sum(h) == target_sum:
        return h.copy()
    # Sort in descending order to do waterfilling
    idxs = np.argsort(h)[::-1]
    h_sorted = h[idxs]
    # Compute cumulative sums
    cumsum = np.cumsum(h_sorted)
    # Find where sum exceeds target
    # We can set first k entries to 1, let remaining be fractional
    # i_index: largest i such that cumsum[i] <= target
    # We'll set values [0..i] to 1, and fraction of [i+1]
    h_proj = h_sorted.copy()
    # Find threshold
    k = 0
    while k < len(h_sorted) and cumsum[k] < target_sum:
        k += 1
    # If we have not enough for floor 1s, all values can be fractional
    if k == 0:
        # All must be 1. Allow values beyond 1 to be capped at 1; 
        # but sum should not exceed target.
        # We cap at 1, but we can only set h[i] = min(1, target_sum)
        # Only when target_sum <= N. If target_sum > N we are overconstrained.
        if target_sum <= len(h):
            h_proj[:] = 1.0 * (target_sum / len(h)) 
        else:
            # Feasibility issue?
            # This is not typical since we should have already checked
            # we're in the box where total can be N/2
            pass
    elif k >= len(h_sorted):
        h_proj[:] = 1.0  # all should be 1, but we have to double-
        # One possibility: set to min(1, something)
        scaling = target_sum / np.sum(h) if np.sum(h) > 0 else 1.0
        h_proj = np.clip(h_proj * scaling, 0, 1)
    else:
        # k-1 entries are set to 1, 
        # then the k-th entry to min(1, target_sum - (k-1))
        # but make sure we still respect sum
        remaining = target_sum - (k - 1)
        if remaining <= 0:
            # case we set first k entries to 1 or more, but then overflow
            # so reset others absurdly to 0 
            h_proj[k:] = 0
        else:
            h_proj[k:] = np.clip(remaining, 0, 1)
    # restore shape
    h_proj_full = h.copy()
    h_proj_full[idxs] = h_proj
    return h_proj_full

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    start_time = time.time()
    
    N = 140
    dx = 2.0 / N
    
    # Phase 0: Build index mapping
    P, pair_to_idx = build_index_map(N)
    # Make sure we handle all valid pairs
    num_pairs = len(P)
    
    # Phase 1: Solve LP relaxation
    print("Solving LP relaxation...")
    
    # Variables
    h = cp.Variable(N, name="h")
    y = cp.Variable(num_pairs, name="y")
    alpha = cp.Variable(name="alpha")
    
    # Constraints
    constraints = [
        0 <= h, h <= 1,
        cp.sum(h) == N / 2,
        alpha >= 0  # We'll later add constraints for all shifts s
    ]
    
    # Add McCormick constraints for each (i,s) pair
    for i, s in P:
        idx = pair_to_idx[(i, s)]
        # y[i,s] <= h[i]
        constraints.append(y[idx] <= h[i])
        # y[i,s] <= h[i+s]
        constraints.append(y[idx] <= h[i+s])
        # y[i,s] >= h[i] + h[i+s] - 1
        constraints.append(y[idx] >= h[i] + h[i+s] - 1)
        # y[i,s] >= 0
        constraints.append(y[idx] >= 0)
    
    # Add α constraints for all shifts s
    for s in range(N):
        # Sum of y over shift s
        y_sum = 0
        for i in range(N - s):
            y_sum -= y[pair_to_idx[(i, s)]]
        # The full constraint, we compute sum_y, sum of y for particular shift s,
        # y_sum = total of y values for shift s.
        # Current: y_sum is negated
        constraints.append(alpha >= dx * (N/2 - y_sum))
    
    # Set objective
    objective = cp.Minimize(alpha)
    
    prob = cp.Problem(objective, constraints)
    
    # Optional: Solve with a time limit or maximum iterations 
    # cvxpy doesn't support wall time directly in ECOS. 
    try:
        prob.solve(solver=cp.ECOS, max_iters=20000, verbose=False)
        # Optional expensive IPM could be used with SCS or OSQP if needed.
    except:
        # Fallback if solver fails
        prob.solve(solver=cp.GLPK, verbose=False)
    
    print(f"LP solution took {time.time()-start_time:.2f}s")
    
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        h_val = h.value
        alpha_val = alpha.value
        print(f"LP lower bound: {alpha_val:.6f}")
        c5_lp = compute_C5(h_val)
        print(f"LP Estimated C5: {c5_lp:.6f}")
        if c5_lp <= 0.3807:
            # Already good, return
            # Ensure the sum is exactly correct
            h_final = waterfilling(h_val, N/2)
            return (h_final, c5_lp, N)
    else:
        print(f"LP solver status: {prob.status}")
        # If we can't solve LP properly, exit early
        return (np.ones(N)*0.5, 0.5, N)
    
    # Phase 2: Try MILP (Use binary h, but continuous y and α)
    print("Solving MILP (binary h)...")
    h_bin = cp.Variable(N, boolean=True)
    y_bin = cp.Variable(num_pairs)
    alpha_bin = cp.Variable()
    
    # Constraints
    bin_constraints = [
        0 <= h_bin, h_bin <= 1,
        cp.sum(h_bin) == N / 2
    ]
    # Keep all other constraints
    for i, s in P:
        idx = pair_to_idx[(i, s)]
        bin_constraints.append(y_bin[idx] <= h_bin[i])
        bin_constraints.append(y_bin[idx] <= h_bin[i+s])
        bin_constraints.append(y_bin[idx] >= h_bin[i] + h_bin[i+s] - 1)
        bin_constraints.append(y_bin[idx] >= 0)
        
    # Add α constraints from MILP perspective
    for s in range(N):
        y_sum = 0
        for i in range(N - s):
            y_sum -= y_bin[pair_to_idx[(i, s)]]
        bin_constraints.append(alpha_bin >= dx * (N/2 - y_sum))

    bin_objective = cp.Minimize(alpha_bin)
    bin_prob = cp.Problem(bin_objective, bin_constraints)
    
    # Approximately use half the remaining time for MILP solve
    remaining_milp_time = budget_s - (time.time() - start_time) 
    if remaining_milp_time < 30:
        # Not enough time to solve MILP effectively
        # Use greedy rounding from previous result or default
        h_final = waterfilling(h_val, N/2)
        c5 = compute_C5(h_final)
        return (h_final, c5, N)
    
    try:
        if remaining_milp_time > 50:
            bin_prob.solve(solver=cp.CBC, verbose=False, maxTime=remaining_milp_time - 20)
            # Even with shorter times, CBC is robust; let's at least attempt to finish
        else:
            # Use a faster solver with suboptimal time limits or do greedy rounding first
            # Since this is network/python then take what we can get
            bin_prob.solve(solver=cp.CBC, verbose=False, maxTime=30)
    except Exception as e:
        print("MILP solver error:", e)
        # Fall back to rounding
        pass

    if bin_prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        h_bin_val = h_bin.value
        print("MILP feasible solution found")
        c5_milp = compute_C5(h_bin_val)
        if c5_milp <= 0.3807:
            return (h_bin_val, c5_milp, N)
        # else keep trying rounding, local search

    # Phase 3: Greedy rounding of LP
    print("Performing local refinement on rounded solution...")
    h_rounded = waterfilling(h_val, N/2)
    # Optionally symmetrize if needed for optimal solvers, but constraint is already linear
    # here, so symmetric solution will itself be optimal if symmetry constraint holds
    
    best_h = h_rounded
    best_c5 = compute_C5(best_h)
    
    # Simple gradient step with "steepest descent" heuristic; for small steps only 
    # to stay in feasible region
    def move_towards_improvement(curr_h, step_size=0.02):
        # Find worst shift s
        dx = 2.0 / len(curr_h)
        # Compute correlation for all shifts
        corr = np.correlate(curr_h, 1 - curr_h, mode='full')
        worst_shift_idx = np.argmax(corr)
        worst_shift = worst_shift_idx - (len(curr_h) - 1)
        # Derivative to update (approximate)
        # ∂C_s / ∂h_i = d/dh_i [ sum h_j (1 - h_{j+s}) ] 
        # = 1 - h_{i+s} for i = 0,...N-1-s; 0 otherwise
        # So to decrease C_s, we want to increase h[i] i.f.f. h[i+s] = 0
        # But we keep it bounded and sum fixed by projection
        grad = np.zeros_like(curr_h)
        # Apply gradient for the worst shift only
        for i in range(len(curr_h)-abs(worst_shift)):
            j = i + worst_shift
            if j >= 0 and j < len(curr_h): 
                if curr_h[i+s] == 0:
                    grad[i] += 1
                else:
                    if curr_h[j] == 0:
                        grad[i] -= 1
        # Normalize or scale appropriately
        # Gradient ascent: update h - step_size*grad
        h_new = curr_h - step_size * grad
        # Project back to [0,1] 
        h_new = np.clip(h_new, 0, 1)
        # Waterfill to maintain exact sum
        h_new = waterfilling(h_new, N/2)
        return h_new
    
    # Do a few iterations of local hill climb
    num_iters = 100
    for _ in range(num_iters):
        if time.time() - start_time > budget_s * 0.95:
            break
        try:
            h_new = move_towards_improvement(best_h, step_size=0.02)
            c5_new = compute_C5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
        except Exception as e:
            print("Local search error", e)
            break
    
    return (best_h, best_c5, N)
```
