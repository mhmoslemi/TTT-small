import numpy as np
import time
import cvxpy as cp

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using full convolution
    overlap = np.correlate(h, 1 - h, mode='full')
    # The maximum value scaled by dx
    c5 = np.max(overlap) * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use N=139 for better chance of finding integer solution
    N = 139
    dx = 2.0 / N
    
    # Build MILP model
    x = cp.Variable(N, boolean=True)
    constraints = [cp.sum(x) == N // 2]  # Sum of ones = N/2  (i.e., 69.5; but binary means we have integer count; using half)
    # Actually recheck constraint precision. The sum == N//2
    # N=139, so N//2 = 69. 
    # but if using binary x, sum must be integer 69 or 70?
    # Need exactly half of 139?
    # 139 is odd. Arrays points {0,1,..,N-1)=139 elements.
    # Each element h[i] 0 or 1. If h=69 ones, sum=69.
    # dx = 2/N = 2/139, so sum h_i * dx = sum h_i * 2/139. Want this = 1.
    # So sum h_i = 139/2 = 69.5. Not integer. Odd constraint. 
    # Need some elements to be 0.5? But then h not integer. 
    # But the framing says "∫₀² h(x) dx = 1" 
    # But sampling with N points: discrete coarse version h[i] * dx for i = 0,...,N-1  
    # Want sum h[i] * dx = 1 => sum h[i] = N/2 
    # Since h[i] ∈ [0,1] and we are optimizing, could h[i] be fractional? 
    # No, you said: "h(x) ∈ [0, 1] for all x"  
    # But for continuous h with integral = 1, can fractional values. 
    # The constraint on integrals in discrete sense (like sampling): 
    # sum h[i]*dx = 1  => sum h[i] = 1/dx = 139/2  = 69.5    (not exact integer)
    # So sum of h must be 69.5? 
    # We can't force integer sum without using floating point (h here discrete of N values).
    # Euclidean space as real functions inferred. But discretizing the solution, constraints enforce:
    # A priori: h[i] in [0,1], sum h[i] * 2/N = 1 => sum h[i] = N/2 = 69.5.
    # We can enforce but require floating?

    # Let's redefine: we work with h[i] in [0,1] and sum h[i] = 69.5. 
    # Construct a mixed integer linear problem where we force h[i] being continuous in [0, 1], 
    # but in discrete model we blame for sums being fractional?
    # Hmm investment.
    # Use a lower N = 140.  
    # N=140 (even) gives sum h[i] = 70
    # Using integer constraint on h[i] = binary, sum = 70 is naturally ok.
    # Indeed using 139 vs 140 may give full advantage with even N and integer constraint.
    # Better to just try N=140 (even)  
    N = 140
    dx = 2.0 / N
    x = cp.Variable(N, boolean=True)
    constraints = [cp.sum(x) == N // 2]  # sum = 70
    
    C = cp.Variable()
    
    for ell in range(1, N):
        length = N - ell
        if length == 0:
            continue
        H = cp.sum(x[:length])  # sum of first (length) elements
        # Introduce y variables to represent x[i] * x[i+ell] 
        y = cp.Variable(length, boolean=True)
        # Constraints to compute product. This is correct for linearization of x * y for binary.
        for i in range(length):
            constraints += [y[i] <= x[i],
                            y[i] <= x[i + ell],
                            y[i] >= x[i] + x[i + ell] - 1]
        O = cp.sum(y)
        c = H - O
        constraints += [c <= C, c >= 0]  # Both constraints to ensure feasibility
    
    # Minimize C (it's the maximum integral, not certainty, but true goal)
    prob = cp.Problem(cp.Minimize(C), constraints)
    
    # Solve
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time <= 5:
        # Not enough time left, fallback
        # Return simple solution
        h = np.ones(N) * 0.5
        h[10:20] = 0.0
        h[40:50] = 0.0
        return (h, c5_of(h), N)
        
    # Solve MILP  
    prob.solve(solver=cp.CBC, max_seconds=remaining_time-5, verbose=False)
    
    # Check solution
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        h_array = np.array(x.value).astype(float)  
        # Prevent any numerical problems (should already be integer due to domain)
        h_array = np.clip(h_array, 0, 1)
        c5_bound = c5_of(h_array)
        return (h_array, c5_bound, N)
    else:
        # Fall back to old method
        h = np.ones(N) * 0.5
        h[20:40] = 1.0  
        h[100:120] = 0.0
        return (h, c5_of(h), N)
