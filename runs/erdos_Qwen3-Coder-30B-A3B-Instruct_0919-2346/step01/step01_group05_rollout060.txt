def project_onto_bounds_sum(v, target_sum, lower=0, upper=1):
    """Project vector v onto the simplex with sum = target_sum and bounds [lower, upper]"""
    def f_lambda(lambda_val):
        projected = np.clip(v - lambda_val, lower, upper)
        return np.sum(projected) - target_sum
    
    # Binary search for lambda
    left = v.min() - upper
    right = v.max() - lower
    
    for _ in range(60):
        mid = (left + right) / 2
        if f_lambda(mid) > 0:
            left = mid
        else:
            right = mid
    
    lambda_val = (left + right) / 2
    return np.clip(v - lambda_val, lower, upper)

def evaluate_erdos_solution(h):

    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value scaled by dx
    max_overlap = np.max(correlation) * dx
    return max_overlap
