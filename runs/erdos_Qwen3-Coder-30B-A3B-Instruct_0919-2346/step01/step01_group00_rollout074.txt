import numpy as np
import time
import random

def project_h(h, target_sum):
    """Project h onto box [0,1] and hyperplane sum=target_sum using water-filling."""
    h = h.copy()
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if np.isclose(current_sum, target_sum):
        return h
    
    # Binary search for lambda
    low, high = -1.0, 1.0
    for _ in range(50):
        mid = (low + high) / 2
        projected = np.clip(h - mid, 0, 1)
        projected_sum = np.sum(projected)
        if projected_sum > target_sum:
            low = mid
        else:
            high = mid
    
    lambda_val = (low + high) / 2
    h = np.clip(h - lambda_val, 0, 1)
    return h

def evaluate_C5(h):
    """Evaluate C5 = max_k ∫ h(x)(1-h(x+k)) dx."""
    dx = 2.0 / len(h)
    c = np.correlate(h, 1 - h, mode="full")
    c *= dx
    return np.max(c)

def compute_gradient(h, alpha):
    """Compute gradient of smoothed max correlation with respect to h."""
    N = len(h)
    dx = 2.0 / N
    
    # Compute correlation S = ∫ h(x) h(x+k) dx  
    S = np.correlate(h, h, mode="full")
    S = S[:2*N-1]  # Keep correct size
    C = 1 - dx * S
    
    # Compute weights
    w = np.exp(alpha * C)
    w = w / np.sum(w)
    
    # Compute convolution to get gradients efficiently
    w_rev = w[::-1]
    
    v_plus = np.convolve(h, w_rev, mode='same')
    v_minus = np.convolve(h, w, mode='same')
    
    grad = dx * (1 - (v_plus + v_minus))
    
    return grad

        # Select a random seed

        h = project_h(h, target_sum)

            # Adaptive alpha and learning rate
            alpha = 10 + 70 * (t / max_iter)  # Rise from 10 to 80
            eta = 0.1 * (1 + t / max_iter)  # Adaptive step size
            
            grad = compute_gradient(h, alpha)
        
            # Gradient decent step with projection
            h_new = h - eta * grad
            
            # Project onto feasible region
            h_new = project_h(h_new, target_sum)

            C5 = evaluate_C5(h)
