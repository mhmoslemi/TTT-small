import numpy as np
import time
from scipy.optimize import differential_evolution

def c5_of(h):
    """Compute the maximum overlap integral for the given step function h."""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    c = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(c) * dx
    return c5

def project_to_feasible(h):
    """Project h to satisfy the constraints: 0 <= h <= 1 and sum(h) = n/2"""
    # Clip to bounds
    h = np.clip(h, 0, 1)
    # Renormalize
    n = len(h)
    h = h * (n/2) / np.sum(h)
    return h

def optimize_with_differential_evolution(initial_h, budget_s):
    """Use differential evolution to find a better solution."""
    start = time.time()
    n = len(initial_h)
    bounds = [(0, 1) for _ in range(n)]
    
    def objective(h):
        # Normalize
        h = project_to_feasible(h)
        return c5_of(h)
    
    # Run differential evolution with time limit
    try:
        result = differential_evolution(
            objective, 
            bounds, 
            seed=42, 
            maxiter=int(5000 * budget_s / 600),  # Scale by budget
            popsize=15,
            mutation=(0.5, 1.0),
            recombination=0.7,
            atol=1e-6,
            tol=1e-6
        )
        if time.time() - start < budget_s:
            h_opt = project_to_feasible(result.x)
            return h_opt, c5_of(h_opt)
    except:
        pass
    
    # Fallback: return initial if DE failed
    return initial_h, c5_of(initial_h)

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    N = 140
    dx = 2.0 / N
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h0 = np.interp(np.linspace(0, 2, N), 
                      np.linspace(0, 2, len(initial_h_values)), 
                      initial_h_values)
        h0 = h0 * (N/2) / np.sum(h0)  # Renormalize exactly
    else:
        h0 = np.ones(N) * 0.5  # fallback
    
    # Also try a construction that's more concentrated
    h1 = np.zeros(N)
    # Create a 3-peak construction
    x_vals = np.linspace(0, 2, N)
    # Width of peaks
    w = 0.2
    # Peak positions
    p1 = 0.4
    p2 = 1.0
    p3 = 1.6
    # Heights
    h1[(x_vals >= p1 - w/2) & (x_vals < p1 + w/2)] = 0.8
    h1[(x_vals >= p2 - w/2) & (x_vals < p2 + w/2)] = 1.0
    h1[(x_vals >= p3 - w/2) & (x_vals < p3 + w/2)] = 0.8
    # Normalize
    h1 = h1 * (N/2) / np.sum(h1)
    
    # Try multiple random constructions
    best_h = h0.copy()
    best_c5 = c5_of(h0)
    counter = 0
    
    start = time.time()
    while time.time() - start < 0.8 * budget_s:
        counter += 1
        # If the last few iterations were not helpful, try random construction
        if counter % 10 == 0:
            h_rand = np.random.rand(N)
            h_rand = h_rand * (N/2) / np.sum(h_rand)
            h_rand = project_to_feasible(h_rand)
            c5_rand = c5_of(h_rand)
            if c5_rand < best_c5:
                best_h = h_rand
                best_c5 = c5_rand
        elif counter % 3 == 0:
            h, c5 = optimize_with_differential_evolution(h1, 0.1 * budget_s)
        else:
            # Run DE from h0
            h, c5 = optimize_with_differential_evolution(h0, 0.1 * budget_s)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5
    
    # More fine-tuning if needed
    if time.time() - start < 0.1 * budget_s:
        # Gradient-like local search
        for _ in range(200):
            if time.time() - start > 0.9 * budget_s:
                break
            h_test = best_h.copy()
            i = np.random.randint(0, N)
            d = np.random.uniform(-0.05, 0.05)
            # Ensure h stays feasible
            if 0 <= h_test[i] + d <= 1:
                h_test[i] += d
                # Renormalize
                h_test = h_test * (N/2) / np.sum(h_test)
                new_c5 = c5_of(h_test)
                if new_c5 < best_c5:
                    best_h = h_test
                    best_c5 = new_c5
            
    return (best_h, best_c5, N)
