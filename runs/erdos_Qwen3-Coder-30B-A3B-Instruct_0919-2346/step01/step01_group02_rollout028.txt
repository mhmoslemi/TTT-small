```python
import numpy as np
import cvxpy as cp
import time

def waterfilling(h, target_sum):
    """Project h onto the simplex with given sum using binary search."""
    h = np.asarray(h)
    low, high = 0.0, 1.0
    for _ in range(50):
        mid = (low + high) / 2
        s = np.sum(np.clip(h + mid, 0, 1))
        if s > target_sum:
            high = mid
        else:
            low = mid
    lambda_ = low
    h_proj = np.clip(h + lambda_, 0, 1)
    # Correct the sum with waterfilling adjustment
    diff = target_sum - np.sum(h_proj)
    if abs(diff) > 1e-10:
        # Bring sum to exact target using bisection on the lambda for clipping
        # This is too fragile; just project and accept small error for now
        pass
    return h_proj

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start = time.time()
    N = 140
    dx = 2.0 / N
    target_sum = N / 2

    # Precompute valid pairs for product y[i,s]
    pairs = []
    for s in range(N):
        for i in range(N - s):
            pairs.append((i, s))
    num_pairs = len(pairs)

    # Formulate and solve the LP relaxation
    # Variables: h (N), y (num_pairs), alpha (1)
    h = cp.Variable(N, name='h')
    y = cp.Variable(num_pairs, name='y')
    alpha = cp.Variable(1, name='alpha')

    constraints = []

    # Box constraints and sum constraint
    constraints += [0 <= h, h <= 1]
    constraints += [cp.sum(h) == target_sum]

    # McCormick envelopes for y[i,s] = h[i] * h[i+s]
    for idx, (i, s) in enumerate(pairs):
        constraints += [y[idx] <= h[i]]
        constraints += [y[idx] <= h[i+s]]
        constraints += [y[idx] >= h[i] + h[i+s] - 1]
        constraints += [y[idx] >= 0]

    # Overlap constraints for each possible shift (in [-N+1, N-1])
    # We can compute all shifts in [-N+1, N-1] but we only need full correlation
    # The total number of unique correlation values is 2*N-1
    # We can use full correlation: shift from -N+1 to N-1, and handle only valid ones
    # Or we use 0 to N-1 (non-negative shifts) since others will be symmetric

    # Non-negative shifts: 0 to N-1. The maximum full correlation array size is 2*N-1
    corr_indices = {}
    for s in range(N):
        indices_s = [idx for idx, (i, shift) in enumerate(pairs) if shift == s]
        corr_indices[s] = indices_s

    # Add one constraint per shift: alpha >= dx * (N/2 - sum(y[i,s]))
    for s in range(N):
        sum_y_s = cp.sum(y[corr_indices[s]])
        constraints += [alpha >= dx * (target_sum - sum_y_s)]

    # Objective: minimize alpha
    problem = cp.Problem(cp.Minimize(alpha), constraints)

    # Solve the LP
    remaining_budget = budget_s - (time.time() - start)
    if remaining_budget < 20:
        return (np.ones(N)*0.5, 0.5, N)

    problem.solve(solver=cp.ECOS, max_iters=10000)
    if problem.status not in ["optimal", "optimal_inaccurate"]:
        # fallback to approximate solution
        pass

    # Extract solution (h_LP is likely fractional)
    h_lp = h.value
    alpha_lp = alpha.value[0]
    c5_lp = alpha_lp  # Because we compute in scaled form and alpha is already C5

    if c5_lp <= 0.3807:
        # Final constraint projection
        h_final = waterfilling(h_lp, target_sum)
        c5_final = np.max(np.correlate(h_final, 1 - h_final, mode='full')) * dx
        return (h_final, c5_final, N)

    # Now try the binary MILP
    # Change h to binary variables
    h_bin = cp.Variable(N, boolean=True, name='h_bin')
    y_bin = cp.Variable(num_pairs, name='y_bin')  # Continuous
    alpha_bin = cp.Variable(1, name='alpha_bin')

    # Constraints again
    constraints_bin = [
        cp.sum(h_bin) == target_sum,
        0 <= h_bin, h_bin <= 1
    ]

    # McCormick for binary case
    for idx, (i, s) in enumerate(pairs):
        constraints_bin += [y_bin[idx] <= h_bin[i]]
        constraints_bin += [y_bin[idx] <= h_bin[i+s]]
        constraints_bin += [y_bin[idx] >= h_bin[i] + h_bin[i+s] - 1]
        constraints_bin += [y_bin[idx] >= 0]

    # Overlap constraints again
    for s in range(N):
        sum_y_s = cp.sum(y_bin[corr_indices[s]])
        constraints_bin += [alpha_bin >= dx * (target_sum - sum_y_s)]

    # Objective
    problem_bin = cp.Problem(cp.Minimize(alpha_bin), constraints_bin)

    # Solve with limited time
    remaining_budget = budget_s - (time.time() - start)
    if remaining_budget < 60:
        h_final = waterfilling(h_lp, target_sum)
        c5_final = np.max(np.correlate(h_final, 1 - h_final, mode='full')) * dx
        return (h_final, c5_final, N)

    # Use GLPK_MI for mixed integer
    try:
        # Instant solve with time limit
        problem_bin.solve(solver=cp.GLPK_MI, max_time=remaining_budget - 10)
        if problem_bin.status in ["optimal", "optimal_inaccurate"]:
            h_milp = h_bin.value
            c5_milp = np.max(np.correlate(h_milp, 1 - h_milp, mode='full')) * dx
            return (h_milp, c5_milp, N)
    except Exception:
        pass

    # If MILP fails, use rounding + hill climbing
    h_rounded = np.zeros(N)
    h_sort_idx = np.argsort(-h_lp)  # Descending order
    for i in range(int(N/2)):
        h_rounded[h_sort_idx[i]] = 1

    # Now refine with hill-climbing
    h_best = h_rounded.copy()
    c5_best = np.max(np.correlate(h_rounded, 1 - h_rounded, mode='full')) * dx

    remaining_budget = budget_s - (time.time() - start)
    if remaining_budget < 20:
        return (h_best, c5_best, N)

    # Simple refinement (limited steps)
    step_size = 0.02
    for _ in range(200):
        if time.time() - start > budget_s - 5:
            break
        # Find worst shift
        cvals = np.correlate(h_best, 1 - h_best, mode='full')
        max_idx = np.argmax(cvals)
        s_worst = max_idx - (len(cvals) - 1) // 2  # Convert full index to shift
        # Compute gradient (simplified for now)
        # this is a rough update to encourage increasing the value for shift s_worst
        h_new = h_best.copy()
        if s_worst >= 0:
            for i in range(N-s_worst):
                h_new[i] += step_size * (1 - h_new[i+s_worst])  # gradient of h[i]*h[i+s_worst]
        else:
            for i in range(-s_worst, N):
                h_new[i] += step_size * (1 - h_new[i + s_worst])

        # projection and summation constrain
        h_new = np.clip(h_new, 0, 1)
        h_new = waterfilling(h_new, target_sum)
        c5_new = np.max(np.correlate(h_new, 1 - h_new, mode='full')) * dx
        if c5_new < c5_best:
            h_best = h_new
            c5_best = c5_new

    return (h_best, c5_best, N)
```
