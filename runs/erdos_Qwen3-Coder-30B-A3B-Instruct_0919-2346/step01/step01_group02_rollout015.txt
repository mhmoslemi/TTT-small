import numpy as np
import time
from scipy.optimize import minimize
from scipy.special import logsumexp

def c5_of(h):
    """Compute the maximum overlap integral for the given step function h."""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    c = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(c) * dx
    return c5

def project_to_feasible(h, N):
    """Project h onto the feasible set: [0,1]^N and sum(h) = N/2"""
    # Water filling algorithm
    def waterfill(v, target):
        # Sort v in descending order
        v_sorted = np.sort(v)[::-1]
        # Binary search for lambda
        low, high = -1e10, 1e10
        for _ in range(50):
            mid = (low + high) / 2
            total = np.sum(np.clip(v + mid, 0, 1))
            if total < target:
                high = mid
            else:
                low = mid
        lambda_ = low
        # Compute the projected vector
        h_proj = np.clip(v + lambda_, 0, 1)
        # Adjust small errors to make sum exactly target
        diff = target - np.sum(h_proj)
        if abs(diff) > 1e-10:
            # Find indices where we can increase slightly
            free_indices = np.where((h_proj > 0) & (h_proj < 1))[0]
            if len(free_indices) > 0:
                # Distribute diff among free indices as evenly as possible
                distribute_per = diff / len(free_indices)
                for i in free_indices:
                    new_val = min(1.0, h_proj[i] + distribute_per)
                    if abs(new_val - 1.0) < 1e-10:
                        h_proj[i] = 1.0
                    else:
                        h_proj[i] = new_val
        return h_proj

    # Project the vector
    h_projected = waterfill(h, N / 2)
    return h_projected

def soft_max_corr(h, eps):
    """Compute smooth approximation to maximum correlation and its gradient"""
    dx = 2.0 / len(h)
    # Compute correlations
    c_full = np.correlate(h, 1 - h, mode='full')
    c = c_full * dx
    
    # Subtract max for numerical stability before taking exp
    max_c = np.max(c)
    c_stable = c - max_c
    exp_c = np.exp(c_stable / eps)
    log_sum_exp = max_c + eps * logsumexp(c_stable / eps)
    weights = exp_c / np.sum(exp_c)
    
    # Gradient computation
    grad = np.zeros_like(h)
    N = len(h)
    for i in range(N):
        # Accumulate contributions from all shifts
        grad_term = np.zeros_like(h)
        for s in range(-(N-1), N):
            if s >= 0:
                # h[i] contributes to shift s when i+s < N
                if i+s < N:
                    grad_term[i] += -1
                if i-s >= 0:
                    grad_term[i-s] += -1
            else:
                # h[i] contributes to shift s when i+s >= 0
                if i+s >= 0:
                    grad_term[i+s] += -1
                if i+s+N < N:
                    grad_term[i+s+N] += -1
        if i == 0 and s == 0:
            # skip adding gradient for when s=0
            continue
        else:
            # Use the one-step convolution with shifted arrays
            grad += weights[i] * grad_term
    
    return log_sum_exp, grad

def continuation_optimize(h0, max_iter=150, eps_schedule=None):
    """Optimize using continuation on smooth surrogate."""
    if eps_schedule is None:
        # Use a decreasing schedule
        eps_schedule = [0.10, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    
    h = h0.copy()
    N = len(h)
    
    for eps in eps_schedule:
        # Minimize smooth surrogate with L-BFGS-B
        def obj(x):
            val, grad = soft_max_corr(x, eps)
            return val, grad
        
        # Use L-BFGS-B, initial starting point is just h projected
        h_proj = project_to_feasible(h, N)
        res = minimize(
            obj, h_proj, method='L-BFGS-B', jac=True,
            bounds=[(0., 1.)] * N,
            options={'maxiter': max_iter, 'ftol': 1e-7, 'gtol': 1e-6}
        )
        
        if res.success:
            h = res.x
        else:
            # If optimization failed, stop and proceed to projection
            break
        
        # Project after the optimize step
        h = project_to_feasible(h, N)
    
    # Final projection to make sure sum is exact
    h = project_to_feasible(h, N)
    
    return h

def subgrad_refine(h, max_iter=200, eta=0.05):
    """Refine solution with sub-gradient steps on the max objective."""
    dx = 2.0 / len(h)
    h = h.copy()
    prev_c5 = c5_of(h)
    
    for _ in range(max_iter):
        # Compute all correlations
        c_full = np.correlate(h, 1 - h, mode='full')
        c = c_full * dx
        # Find worst shift
        s_star = np.argmax(c)
        # Compute gradient for the worst shift
        grad = np.zeros_like(h)
        # Create shifted versions
        h_left = np.roll(h, s_star)
        h_right = np.roll(h, -s_star)
        # h[i] contributes to for s_star: i -> i+s_star
        # so subset h_left[:-s_star] for i where i+s_star < N
        # and set grad[i] = -1 if that's the case, -h[i+s_star]
        for i in range(len(h)):
            # We want to build vector of grad[i] = -1 - h[i+s_star] - h[i-s_star]
            if i + s_star < len(h):
                grad[i] -= 1  # add to the (i,i+s_star) term
            else:
                grad[i] -= 0
            if i - s_star >= 0:
                grad[i] -= 1  # add to the (i,i-s_star) term
            else:
                grad[i] -= 0
                
            # Add h[i+s_star] for the shift term, which appears also in shift
            # the correct logic is to compute:
            # For all pairs (i, i+s), the factor 1-h[i+s] appears in index i
            # So we cancel out (with COOLEST evaluation below), it is:
            # grad[i] = (1-h[i+s_star]) = - h[i+s_star] + 1
            # %
            # Actually, correct is: Contribution of h[i] to shift s when h[i] and h[i+s] are part of the sum  
            # So: if shift s = s_star, then for all i such that i + s_star < N, h[i] contributes to that term
            # That is, we have a shift_left term when computing the (i, i+s_star) contribution to C_s_star
            
        # Find indices where shifting changes the correlation term
        for i in range(len(h)):
            if i + s_star < len(h):
                grad[i] -= 1
            if i - s_star >= 0:
                grad[i] -= 1
        grad = dx * grad
        # Update h and project
        h_new = h - eta * grad
        h_new = project_to_feasible(h_new, len(h))
        new_c5 = c5_of(h_new)
        # Accept step if better
        if new_c5 < prev_c5:
            h = h_new
            prev_c5 = new_c5
        else:
            # Optionally decrease stepsize
            eta = max(eta * 0.9, 1e-6)
    return h

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    N = 140
    dx = 2.0 / N
    
    # Define seed strategies
    initial_seeds = []
    
    # Uniform seed
    h_uniform = np.ones(N) * 0.5
    initial_seeds.append(h_uniform)
    
    # Two-peak seeds
    a_vals = [0.12, 0.18, 0.24, 0.30]
    for a in a_vals:
        h_t = np.ones(N) * 0.5
        x_vals = np.linspace(0, 2, N)
        mask = (x_vals < a) | (x_vals >= 2 - a)
        h_t[mask] = 0.9
        h_t[~mask] = 0.1
        h_t = h_t * (N / 2) / np.sum(h_t)
        initial_seeds.append(h_t)
    
    # Binary seeds: exactly N/2 of the entries are 1
    for _ in range(5):
        h_bin = np.zeros(N)
        indices = np.random.permutation(N)[:N//2]
        h_bin[indices] = 1
        initial_seeds.append(h_bin)
    
    # Binary symmetric two blocks
    h_binary_sym = np.zeros(N)
    h_binary_sym[:N//4] = 1
    h_binary_sym[3*N//4:] = 1
    initial_seeds.append(h_binary_sym)
    
    # Random continuous seed
    h_rand = np.random.rand(N)
    h_rand = h_rand * (N / 2) / np.sum(h_rand)
    initial_seeds.append(h_rand)
    
    low = float('inf')
    best_h = None
    
    start = time.time()
    seeds_per_func = 0.10 * budget_s  # fraction for seeds
    num_seeds = len(initial_seeds)
    
    # Go through each seed
    for s_i, h_seed in enumerate(initial_seeds):
        # Project the seed
        h_seed = project_to_feasible(h_seed, N)
        # Run continuation
        h_opt = continuation_optimize(h_seed, eps_schedule=[0.10, 0.05, 0.02, 0.01, 0.005, 0.003, 0.001])
        # Optimized from this seed
        c5_opt = c5_of(h_opt)
        if c5_opt < low:
            low = c5_opt
            best_h = h_opt.copy()
        if time.time() - start > 0.95 * budget_s:
            break
            
    # Final polishing if time remains
    if time.time() - start < 0.80 * budget_s:
        best_h = subgrad_refine(best_h, max_iter=300, eta=0.02)
        
    return best_h, c5_of(best_h), N
