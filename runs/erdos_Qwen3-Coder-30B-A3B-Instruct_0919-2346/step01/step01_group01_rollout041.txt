import numpy as np
import time
import cvxpy as cp

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using full convolution
    overlap = np.correlate(h, 1 - h, mode='full')
    # The maximum value scaled by dx
    c5 = np.max(overlap) * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Try using N = 139 (odd), which allows use of Paley difference set
    N = 139
    dx = 2.0 / N
    
    # Construct a good initial approximation using difference set
    # For prime p ≡ 1 (mod 4), there exists a Paley difference set of size (p-1)/2
    # We can use the known difference set for p = 139
    # Set k = (p-1)/2 = 69, λ = (k-1) = 68, v = p = 139
    # So λ = (k(k-1))/(v-1) = (69*68)/138 = 34
    # i.e., each non-zero lag has intersection size exactly λ=34
    # Difference set will be in Z_139 with 69 elements
    
    # As mentioned, this will create an array of length 139 with 69 ones (approx half of 139).
    # But want sum = 139/2 = 69.5. Since we are working with binary 0/1 arrays,
    # the actual sum must be integer, so use 69 or 70 ones. 
    # To meet constraint within discretization bounds, we set:
    # (sum = 69, with framing via one element at 0.5)
    
    # Start simplest approach: build initial solution from Paley subset and force to binary (sum=69 or 70)
    # Create binary vector of length 139
    # Note: We'll construct a solution that leads to reasonable overlap.
    
    # Let's take simple recursion or repeat strong pattern for binary version
    # Our key goal: make min overlap high ~ 34, so minA ≥ 34. 
    
    # Begin with a regular periodic pattern
    # A tried method: alternate blocks of "000111" and "111000" (3+3)
    # Try 140 = 5*28, but use 139 
    # Build first: Create suggested 139 element binary array: start with 1s centered
    
    # Pick exact midpoint configuration
    h_suggested = np.zeros(N)
    half_count = (N + 1) // 2  # Use integer division to keep count as 69 or 70
    # We'll make approximately half of them ones
    num_ones = 69
    ones_indices = np.arange(num_ones)  # first 69 positions are ones
    h_suggested[ones_indices] = 1
    
    # Evaluate original start
    c5_current = c5_of(h_suggested)
    best_h = h_suggested.copy()
    best_c5 = c5_current
    
    # Allow fallback time for simulated annealing if x = 139 not good
    remaining_budget = budget_s - (time.time() - start_time)
    if remaining_budget < 10:
        return (best_h, best_c5, N)
    
    # Use computed pattern as base and refine if ms left
    # Try to validate imperial approach works, at least in bounds
    
    # Let's implement a constrained MILP optimization to see if we can do better than
    # default starting pattern;
    
    # Define MILP model to improve construction:
    
    # For better C5, we still solve the peculiarity that exact 139(same as previous) is also an option
    # A exactly 139 should approach better than 140? 
    # Ultimately, the 140 case we have been using so far doesn't have a perfect λ.
    # But for aggressive improvement, let's briefly try higher-N if needed or see if x=139 yields better.
    
    # Let's directly use the MILP approach with focus on N=139 but using binary values.
    
    # Build MILP only if time permits + signal for value
   
    # Initialize
    # binary x
    x = cp.Variable(N, boolean=True)
    constraints = [cp.sum(x) == 69]  # Number of 1s = 69
    C = cp.Variable()
    
    # We aim to minimize max c_ell over ell 1..N-1
    # Let's be methodical: for each ell, define suitable terms
    total_constraints = []
    for ell in range(1,N):  # ell from 1 to N-1: leads to effective overlapping length N-ell
        # overlapping indices in [0, N-ell-1]
        H_ell = cp.sum(x[:N-ell])  # This sum over indices 0 to N-ell-1 inclusive
        # create y variable for products x_i * x_{i+ell} for i=0 to N-ell-1
        if (N - ell) == 0: continue
        y_ell = cp.Variable(N-ell, boolean=True)
        # Constraint as y_ell[i] <= x[i] for all i
        total_constraints.extend(y_ell <= x[:N-ell])
        # Constraint as y_ell[i] <= x[i+ell] for all i  
        total_constraints.extend(y_ell <= x[ell:])
        # Constraint as y_ell[i] >= x[i] + x[i+ell] - 1 for all i  
        total_constraints.extend(y_ell >= x[:N-ell] + x[ell:] - 1)
        O_ell = cp.sum(y_ell)
        c_ell = H_ell - O_ell  # overlap integral
        total_constraints.extend([c_ell <= C, c_ell >= 0])  # Ensure non-negative and bounded
    
    # Update constraints list with all
    constraints.extend(total_constraints)
    
    # Set up and solve problem
    prob = cp.Problem(cp.Minimize(C), constraints)
    
    # Use solver with timeout
    time_remaining = budget_s - (time.time() - start_time) - 5 
    if time_remaining <= 0: 
        return (best_h, best_c5, N)
    
    try:
        prob.solve(solver=cp.CBC, max_seconds=min(time_remaining, 300), verbose=False)
    except Exception as ex:
        # fallback to original method if MILP hits limits
        return (best_h, best_c5, N)
    
    # Check solution status
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:  
        # Extract solution x
        x_sol = x.value
        if x_sol is not None:
            h_sol = np.rint(x_sol).clip(0,1).astype(float)
            c5_sol = c5_of(h_sol)  # Already consistent with dx
            
            if c5_sol < best_c5:
                best_h = h_sol
                best_c5 = c5_sol
            
            return (best_h, best_c5, N)
    
    # Fallback if no MILP success
    return (best_h, best_c5, N)
