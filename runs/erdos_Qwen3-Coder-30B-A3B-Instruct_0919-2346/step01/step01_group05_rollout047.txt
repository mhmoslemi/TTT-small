import numpy as np
from scipy.optimize import minimize
from math import sqrt
import time

def build_h_from_coeffs(a_vals, n):
    """Build h from cosine coefficients a_vals."""
    M = len(a_vals)
    h = np.full(n, 0.5)
    for m in range(1, M+1):
        h += a_vals[m-1] * np.cos(2*np.pi*m*np.arange(n)/n)
    return h

def compute_overlap_and_gradient(a_vals, n, dx, gamma=1.0):
    """Compute the softened max overlap and its gradient."""
    h = build_h_from_coeffs(a_vals, n)
    # Apply soft clipping if needed
    h = np.clip(h, 0, 1)
    # Compute overlap terms for all shifts
    c = np.correlate(h, 1-h, mode='full')
    shifts = np.arange(-n+1, n)
    f_s = c * dx
    # Compute soft max
    exp_fs = np.exp(gamma * f_s)
    Z = np.sum(exp_fs)
    soft_max = np.log(Z) / gamma
    w = exp_fs / Z
    # Gradient w.r.t. a_vals
    grad = np.zeros_like(a_vals)
    for m in range(1, len(a_vals)+1):
        # Compute d h_i / d a_m
        cos_term = np.cos(2*np.pi*m*np.arange(n)/n)
        # d f_s / d a_m
        temp1 = np.zeros(n)
        temp2 = np.zeros(n)
        for i in range(n):
            if i < n - (m-1):
                temp1[i] = cos_term[i] * (1 - h[i + (m-1)])
            if i >= (m-1):
                temp2[i] = h[i - (m-1)] * cos_term[i]
        df_s_da_m = dx * (np.sum(temp1) - np.sum(temp2))
        grad[m-1] = np.sum(w * df_s_da_m)
    # Add penalty gradient
    penalty_grad = np.zeros_like(a_vals)
    for m in range(1, len(a_vals)+1):
        cos_term = np.cos(2*np.pi*m*np.arange(n)/n)
        # Penalty gradient for each component
        h_clipped = np.clip(h, 0, 1)
        diff = h - h_clipped
        penalty_grad[m-1] = 2 * 1e4 * np.sum(diff * cos_term)
    grad += penalty_grad
    return soft_max, grad

def project_to_simplex_box(h, n, target_sum):
    """Project h to the intersection of [0,1]^n and sum = target_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    # Waterfilling to adjust to target sum
    if abs(current_sum - target_sum) < 1e-8:
        return h
    # Adjust using a scalar scaling parameter
    scale = target_sum / current_sum
    h = h * scale
    # Clip again in case scaling pushed outside bounds
    h = np.clip(h, 0, 1)
    # Extra step: bisection to fix the exact sum
    low, high = 0.0, 1.0
    for _ in range(30):
        tau = (low + high) / 2.0
        h_new = np.clip(h - tau, 0, 1)
        new_sum = np.sum(h_new)
        if abs(new_sum - target_sum) < 1e-8:
            return h_new
        elif new_sum > target_sum:
            low = tau
        else:
            high = tau
    return h_new

def polish_with_projected_gradient(h_init, n, dx, steps=100):
    """Polish h with projected gradient descent on final h."""
    h = np.array(h_init)
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h) * dx
    h = h * (1.0 / current_sum)
    for _ in range(steps):
        # Make a small gradient step in direction of minus gradient
        f_full = np.correlate(h, 1 - h, mode='full')
        shifts = np.arange(-n+1, n)
        f_vals = f_full * dx
        max_f_idx = np.argmax(f_vals)
        max_f = f_vals[max_f_idx]
        # Compute gradient of max shift term
        # For shift k that gives max f_k
        k = max_f_idx - (n-1)
        # d f_k / d h_i = dx * (1 - h_{i+k}) for i where i+k < n
        #                 -dx * h_{i-k} for i where i-k >= 0
        gradk = np.zeros(n)
        for i in range(n):
            if i+k >= 0 and i+k < n:
                gradk[i] -= dx * h[i + k]
            if i - k >= 0:
                gradk[i] += dx * h[i - k]
        # Update h
        step_size = 0.05
        h -= step_size * gradk
        # Project back to [0,1] interval
        h = np.clip(h, 0, 1)
        # Project to mean = 0.5 by scaling
        h_sum = np.sum(h) * dx
        h = h * (1.0 / h_sum)
    return h

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    correlation = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(correlation) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Parameters
    n = 240
    dx = 2.0 / n
    n_points = n
    # Target sum for h: integral from 0 to 2 equals 1, so sum(h) * dx = 1
    target_sum = 1.0 / dx

    # Set up multi restart loop
    best_c5 = float('inf')
    best_h = None

    num_restarts = 6
    for restart in range(num_restarts):
        # Seed the restart
        np.random.seed(seed + restart * 100)
        # Stage 1: Low frequency representation
        M_stage1 = 8
        a_init1 = np.random.normal(0, 0.05, M_stage1)
        final_coeffs = spectral_optim(restart, a_init1, n, dx, gamma=0.5)
        # Stage 2: Extend to M=15
        M_stage2 = 15
        a_init2 = np.zeros(M_stage2)
        a_init2[:len(final_coeffs)] = final_coeffs
        final_coeffs = spectral_optim(restart, a_init2, n, dx, gamma=0.5)
        # Stage 3: Extend to M=24
        M_stage3 = 24
        a_init3 = np.zeros(M_stage3)
        a_init3[:len(final_coeffs)] = final_coeffs
        final_coeffs = spectral_optim(restart, a_init3, n, dx, gamma=0.5)
        # Final h from coefficients
        h_raw = build_h_from_coeffs(final_coeffs, n)
        # Project onto box [0,1]
        h_band = np.clip(h_raw, 0, 1)
        # Final projection ensuring sum = target_sum
        h_projected = project_to_simplex_box(h_band, n, target_sum)
        # Optional polishing
        h_final = polish_with_projected_gradient(h_projected, n, dx, steps=100)
        # Evaluate
        c5 = evaluate_erdos_solution(h_final)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_final

        if time.time() - start_time > (budget_s * 0.95):
            break

    return (best_h.tolist(), best_c5, n_points)
