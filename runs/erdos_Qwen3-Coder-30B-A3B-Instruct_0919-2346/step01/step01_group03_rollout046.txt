import numpy as np
import time
from scipy.optimize import differential_evolution, minimize
import math

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral via full convolution of h with (1-h)
    overlaps = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum of the full correlation vector
    # scaled by dx
    c5 = np.max(overlaps) * dx
    return c5

def evaluate_constraint(h):
    """Check constraint that integral equals 1"""
    dx = 2.0 / len(h)
    return abs(np.sum(h) * dx - 1.0)

def constrained_c5(h):
    """Constrainted C5 evaluation, penalize constraint violations"""
    penalty = 1e6 * max(0, evaluate_constraint(h) - 1e-10)
    return c5_of(h) + penalty

def initialize_h(n):
    """Initialize h values with multiple methods"""
    # Method 1: Initial construction if available
    try:
        h0 = np.array(initial_h_values)
        if len(h0) != n:

            x_new = np.linspace(0, 2, n)
            h0 = np.interp(x_new, x0, h0)
    except:
        # Fallback to random initialization
        h0 = np.random.rand(n) * 0.5
        h0 *= (n / 2) / np.sum(h0)
    
    return h0

def optimize_with_differential_evolution(h0, budget_s):
    """Use differential evolution to minimize C5"""
    start_time = time.time()
    
    # Define bounds (0, 1) for each variable
    bounds = [(0, 1) for _ in range(len(h0))]
    
    # Use differential evolution
    result = differential_evolution(
        constrained_c5,
        bounds,
        maxiter=int(1000 * budget_s / 600),
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.7,
        seed=42,
        disp=False,
        polish=True
    )
    
    # Apply final constraint again to make sure it's valid
    h_final = result.x
    h_final = h_final * (len(h_final) / 2) / np.sum(h_final)
    
    return h_final, c5_of(h_final)

def local_search_step(h, step_size=0.01):
    """Perform a small local optimization step"""
    h = h.copy()
    n = len(h)
    dx = 2.0 / n
    
    # Try to adjust each point slightly
    for i in range(n):
        old_val = h[i]
        # Try small perturbations
        perturbations = [-step_size, -step_size/2, step_size/2, step_size]
        best_h = h.copy()
        best_c5 = c5_of(h)
        
        for delta in perturbations:
            new_h = h.copy()
            new_h[i] = max(0, min(1, old_val + delta))
            
            # Fix normalization
            sum_h = np.sum(new_h)
            new_h = new_h * (n / 2) / sum_h
            
            new_c5 = c5_of(new_h)
            if new_c5 < best_c5:
                best_h = new_h
                best_c5 = new_c5
        
        h = best_h
    
    return h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

                                best_c5 = new_c5
                                improvement = True
        except:
            pass
    
    # Final check and normalization
    if np.sum(best_h) * dx != 1.0:
        best_h = best_h * (n / 2) / np.sum(best_h)
        
    return (best_h, best_c5, n)
