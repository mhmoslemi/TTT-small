import numpy as np
import time

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral via full convolution of h with (1-h)
    overlaps = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum of the full correlation vector
    # scaled by dx
    c5 = np.max(overlaps) * dx
    return c5

def compute_c5_and_worst_info(h):
    """Compute C5 and worst shift information"""
    N = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    max_idx = np.argmax(corr)
    max_val = corr[max_idx]
    offset = max_idx - (N - 1)
    
    # contributions vector:
    if offset >= 0:
        hi = h[:N - offset]
        hj = h[offset:]
        contrib = hi * (1 - hj)
        i_local = np.argmax(contrib)
        i_index = i_local
        j_index = i_local + offset
    else:
        off = -offset
        hi = h[off:]
        hj = h[:N - off]
        contrib = hi * (1 - hj)
        i_local = np.argmax(contrib)
        i_index = i_local + off
        j_index = i_local
    return max_val * (2.0/N), offset, i_index, j_index, hi[i_local], hj[i_local], contrib[i_local]

def propose_direct_move(h, i, j, step):
    """Propose a move by transferring mass from i to j"""
    delta = min(step, h[i], 1 - h[j])
    if delta <= 0:
        return None
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new

def targeted_descent_with_local_search(h0, time_budget, start_time):
    """Run targeted descent with local search to improve solution"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of(h)
    
    # step starts at 0.5 (max feasible change)
    step = 0.5
    fails = 0
    max_fails = 5000  # Prevent unlimited failures
    
    while time.time() - start_time < time_budget and fails < max_fails:
        # compute worst info
        _, _, i, j, hi_val, hj_val, _ = compute_c5_and_worst_info(h)
        
        # if hi and hj are both 0 or 1? We want hi high, hj low
        # compute delta
        delta = min(step, h[i], 1 - h[j])
        if delta > 0:
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            c5_new = c5_of(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
                # Increase step slightly if successful
                step = min(step * 1.05, 0.5)
                fails = 0  # Reset fails counter
                continue
                
        # Try moving in reverse direction (j to i)
        delta2 = min(step, h[j], 1 - h[i])
        if delta2 > 0:
            h_new2 = h.copy()
            h_new2[j] -= delta2
            h_new2[i] += delta2
            c5_new2 = c5_of(h_new2)
            if c5_new2 < best_c5:
                best_c5 = c5_new2
                best_h = h_new2.copy()
                h = h_new2
                step = min(step * 1.05, 0.5)
                fails = 0
                continue
        
        # If no improvement, reduce step
        step *= 0.9
        fails += 1
        
        # Occasionally, try a random move to escape local minima
        if np.random.rand() < 0.02:
            # Select random indices
            i_rand, j_rand = np.random.choice(len(h), size=2, replace=False)
            delta_r = step * min(h[i_rand], 1 - h[j_rand])
            if delta_r > 0:
                # Try to improve with random move
                h_new = h.copy()
                h_new[i_rand] -= delta_r
                h_new[j_rand] += delta_r
                c5_new = c5_of(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new
                    fails = 0  # Reset fails counter if successful

    return best_h, best_c5

def interpolate_h_values(h_old, new_len):
    """Interpolate h values to new length"""
    x_old = np.linspace(0, 2, len(h_old))
    x_new = np.linspace(0, 2, new_len)
    h_new = np.interp(x_new, x_old, h_old)
    return h_new
