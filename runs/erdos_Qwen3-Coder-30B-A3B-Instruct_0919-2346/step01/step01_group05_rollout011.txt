import numpy as np
from math import sqrt
import time

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value scaled by dx
    max_overlap = np.max(correlation) * dx
    return max_overlap

def project_onto_bounds_sum(v, target_sum, lower=0, upper=1):
    """Project vector v onto box constraints [lower,upper] and sum constraint"""
    # Binary search for the correct lambda
    lo = v.min() - upper
    hi = v.max() - lower
    
    def compute_sum(lam):
        return np.clip(v - lam, lower, upper).sum()
        
    for _ in range(60):
        mid = (lo + hi) / 2
        s = compute_sum(mid)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
            
    lam = (lo + hi) / 2
    projected = np.clip(v - lam, lower, upper)
    return projected

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use a moderately sized vector for better accuracy
    n_points = 200
    dx = 2.0 / n_points
    
    # Initialize h values using a binary approach, precise balance
    h = np.zeros(n_points)
    # Initialize with Sturmian-like sequence (modified Fibonacci)
    # Using golden ratio
    phi = (sqrt(5) - 1) / 2
    for i in range(n_points):
        h[i] = 1 if ((i * phi) % 1) < 0.5 else 0
        
    # Adjust to get exactly n/2 ones 
    current_sum = np.sum(h)
    ones_count = int(n_points / 2)
    if current_sum < ones_count:
        # Find zeros and flip some to ones
        zero_indices = np.where(h == 0)[0]
        flip_indices = zero_indices[:ones_count - int(current_sum)]
        h[flip_indices] = 1
    elif current_sum > ones_count:
        # Find ones and flip some to zeros  
        one_indices = np.where(h == 1)[0]
        flip_indices = one_indices[:int(current_sum) - ones_count]
        h[flip_indices] = 0
    
    # Ensure the integral constraint
    h = h * (1.0 / np.sum(h) * n_points / 2) if np.sum(h) > 0 else h
    h = np.clip(h, 0, 1)
    
    # Store best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Gradient descent parameters
    max_iter = 20000
    gamma_start = 1.0
    gamma_end = 50.0 
    step_size = 0.5
    
    current_h = h.copy()
    current_c5 = best_c5
    
    # Main optimization loop
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s * 0.98:
            break
            
        # Compute overlap values for all shifts
        correlation = np.correlate(current_h, 1 - current_h, mode="full")
        overlap_values = correlation * dx
        current_max_overlap = np.max(overlap_values)
        
        # Update best
        if current_max_overlap < best_c5:
            best_c5 = current_max_overlap
            best_h = current_h.copy()
        
        # Compute softmax weights
        gamma = gamma_start + (gamma_end - gamma_start) * (iteration / max_iter)
        shifted = overlap_values - current_max_overlap
        exp_shifted = np.exp(gamma * shifted)
        weights = exp_shifted / np.sum(exp_shifted)
        
        # Compute gradient
        gradient = np.zeros(n_points)
        # For each shift, compute contribution to gradient
        for shift_index in range(len(overlap_values)):
            weight = weights[shift_index]
            shift = shift_index - (n_points - 1)
            
            if shift >= 0:
                # i ranges from 0 to n_points-1-shift
                for i in range(n_points - shift):
                    gradient[i] += weight * (1 - current_h[i + shift])
                    # The term -h[i-shift] is valid only if i-shift >= 0
                    if i >= shift:
                        gradient[i] -= weight * current_h[i - shift]
            else:
                # i ranges from -shift to n_points-1
                for i in range(-shift, n_points):
                    gradient[i] += weight * (1 - current_h[i + shift])
                    # The term -h[i-shift] is valid only if i-shift <= n_points-1
                    if i - shift < n_points:
                        gradient[i] -= weight * current_h[i - shift]
        
        # Take gradient step
        new_h = current_h - step_size * gradient
        
        # Project back onto feasible set
        new_h = project_onto_bounds_sum(new_h, n_points / 2, 0, 1)
        current_h = new_h
        
        # Check stopping criteria
        if iteration % 500 == 0:
            current_c5 = evaluate_erdos_solution(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
    
    # Final check
    final_c5 = evaluate_erdos_solution(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (best_h.tolist(), best_c5, n_points)
