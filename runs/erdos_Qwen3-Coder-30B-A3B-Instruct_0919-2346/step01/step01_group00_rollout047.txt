import numpy as np
import time
import random

def compute_C_vector(h, dx):
    """Compute vector of overlaps C_τ for all shifts τ."""
    # Compute autocorrelation via FFT
    n = len(h)
    S = np.correlate(h, h, mode='full')
    # Extract the relevant part (shifts 0 to 2n-2)
    S = S[n-1:2*n-1]
    # Compute overlaps C_τ = 1 - dx * S_τ
    c = 1 - dx * S
    return c

def compute_softmax_weights(c, alpha):
    """Compute softmax weights w_τ = exp(alpha*c_τ) / sum(exp(alpha*c_τ))"""
    c_max = np.max(c)
    exp_ca = np.exp(alpha * (c - c_max))
    w = exp_ca / np.sum(exp_ca)
    return w

def compute_gradient(h, w, dx):
    """Compute gradient of smooth objective ∇f_α = -dx * (w * h + w_rev * h)"""
    # w is length 2n-1, h is length n
    # w * h means correlation with full padding, mode = 'full'
    # w_rev is reversed w
    w_rev = w[::-1]
    
    # Convolution with padding and truncation
    # Using np.convolve to get the right length for efficient computing
    # We want same size as h
    g1 = np.convolve(w, h, mode='full')  # This gives 2*n - 1 elements
    g2 = np.convolve(w_rev, h, mode='full')
    
    # But due to mode='full', we need to get indices correctly
    # Let's just take their 'same' clipped versions, then make sure they
    # correspond correctly
    g1 = np.convolve(w, h, mode='same')  # Length n
    g2 = np.convolve(w_rev, h, mode='same')
    # They both should have length n; add contribution from boundary
    g = -dx * (g1 + g2)
    return g

def waterfill_projection(h, target_sum):

    # Need argmin_{lambda} sum clip(h[i] - lambda, 0, 1) - target_sum = 0
    # Sort h in descending order to do efficient waterfilling
    h_sorted = np.sort(h)[::-1]
    
    # Early check, for efficiency
    if np.sum(h_sorted) <= target_sum:
        return np.clip(h, 0, 1)
    
    n = len(h)

    # criterion_val = lambda l: np.sum(np.clip(h_sorted - l, 0, 1)) - target_sum
    # We need to find lambda such that the sum = target_sum
    # Simple binary search over range [min(h_sorted) - 1, max(h_sorted) + 1]
    
    low, high = h_sorted[-1] - 1.0, h_sorted[0] + 1.0
    
    def criterion_val(l):
        z = np.clip(h_sorted - l, 0, 1)
        return np.sum(z) - target_sum
    
    # Binary search

        if val > 0:
            low = mid
        else:
            high = mid
    lambda_opt = (low + high) / 2.0
    
    h_new = np.clip(h - lambda_opt, 0, 1)
    
    return h_new
