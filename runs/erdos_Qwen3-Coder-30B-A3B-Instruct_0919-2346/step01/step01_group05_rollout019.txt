import numpy as np
import time

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value scaled by dx
    max_overlap = np.max(correlation) * dx
    return max_overlap

def generate_sturmian_binary(n, alpha=(np.sqrt(5)-1)/2):
    """Generate a balanced binary sequence using Sturmian word construction"""
    h = np.zeros(n)
    count_ones = 0
    for i in range(n):
        t = (i * alpha) % 1
        if t < 0.5:
            h[i] = 1
            count_ones += 1
    # Adjust to get exactly n/2 ones
    target = n // 2
    if count_ones > target:
        # Turn some 1s to 0s
        ones_indices = np.where(h == 1)[0]
        for i in range(count_ones - target):
            h[ones_indices[i]] = 0
    elif count_ones < target:
        # Turn some 0s to 1s
        zeros_indices = np.where(h == 0)[0]
        for i in range(target - count_ones):
            h[zeros_indices[i]] = 1
    return h

def compute_g_s(h):
    """Compute all g_s values for current h"""
    n = len(h)
    dx = 2.0 / n
    correlation = np.correlate(h, 1 - h, mode="full")
    g_s = correlation * dx
    return g_s

def delta_g_s(h, i, j, s):
    """Compute change in g_s if we swap h[i] and h[j] for shift s"""
    if s == 0:
        return 0
    # Position i is the primary factor to compute original value
    delta = 0
    # Original term from i: h[i] * (1 - h[i + s]) if i + s < n else 0
    # If h[i] was 1 and h[i+s] was 0, then term is 1 * (1 - 0) = 1
    # After swap h[i] = 0, so it becomes 0 * (1 - h[i+s]) = 0
    # Change is -1 if h[i] = 1 and h[i+s] = 0
    if h[i] == 1 and (i + s < n and h[i + s] == 0):
        delta -= 1
    # If h[i] was 0 and h[i+s] = 1, then term is 0 * (1 - 1) = 0
    # After swap h[i] = 1, so it becomes 1 * (1 - h[i+s]) = 1 - h[i+s]
    # Change is (1 - h[i+s]) if h[i] = 0 and h[i+s] = 1
    elif h[i] == 0 and (i + s < n and h[i + s] == 1):
        delta += (1 - h[i + s])
    
    # Original term from i as second factor: (1 - h[i]) * h[i - s] if i - s >= 0 else 0
    # If h[i] = 1, then factor is 0. If h[i] = 0, factor is 1. So the additional term comes from h[i-s].
    # Term is (1 - h[i]) * h[i - s]  
    # After swap h[i] = 0, the factor changes to 1.
    # Additional term is h[i - s] if i - s >= 0
    if i - s >= 0:
        if i - s < n:
            delta += h[i - s]
    
    # And the j term consideration:
    # Original term from j: h[j] * (1 - h[j + s]) 
    # if h[j] = 1 and h[j+s] = 0: term = 1, after swap j = 0: term = 0
    # if h[j] = 0 and h[j+s] = 1: term = 0, after swap j = 1: term = 1 - h[j+s]
    if h[j] == 1 and (j + s < n and h[j + s] == 0):
        delta += 1
    elif h[j] == 0 and (j + s < n and h[j + s] == 1):
        delta -= (1 - h[j + s])
    
    # Factor from j as second: 1 - h[j] (so 0 if j was 1, 1 if j was 0) times h[j - s]
    # Contribution of h[j-s] (j-s >= 0) increases when j=0, 0 when j=1
    if j - s >= 0:
        if j - s < n:
            delta -= h[j - s]
    
    return delta

def check_valid_swap(h, i, j):
    """Determine whether we can do a swap at i, j"""
    return h[i] != h[j] and (0 <= i < len(h)) and (0 <= j < len(h))

def local_swap_search(h, max_time):
    """Perform local swap based optimization"""
    start_time = time.time()
    n = len(h)
    dx = 2.0 / n
    
    # Precompute functions
    g_s = compute_g_s(h)
    max_g = np.max(g_s)
    max_g_idx = np.argmax(g_s)
    
    # Tabu list: store (i, j) pairs recently swapped for recent iterations
    tabu_list = []
    tabu_max_length = max(50, n // 10)
    
    swaps_no_improve = 0
    max_swaps_no_improve = 2000
    
    max_iter = 100000
    iter_count = 0
    
    while time.time() < start_time + max_time and iter_count < max_iter:
        iter_count += 1
        if iter_count % 5000 == 0:
            print(f"Iteration {iter_count}, current max: {max_g:.6f}")
            
        # Pick worst shift
        s_max = max_g_idx
        # Find positions that contribute to g_s_max
        # For large shifts only, we can find the relevant positions efficiently by using
        # the fact that we mainly care about h[i]*h[i+s] terms that are 1 * 1 = 1
        
        # Pick a good i (1) that makes h[i+s] = 0
        ones = set(np.where(h == 1)[0])
        zeros_pos = np.where(h == 0)[0]
        candidates = []
        for i in ones:
            # To improve s_max, we want to make h[i+s_max] = 1 or reduce contribution.
            # It's better to swap with a j s.t. h[j+s_max] = 1
            j_candidate = i + s_max
            if j_candidate < n and h[j_candidate] == 1:
                candidates.append((i, j_candidate))
            # Also consider reverse swap for improving relevant shifts:
            # With a j where h[j+s_max] = 0
            for j in zeros_pos:
                if j + s_max >= n:
                    continue
                if h[j + s_max] == 0:
                    candidates.append((i, j))
        
        # Randomly choose from candidates or use a greedy heuristic
        if len(candidates) > 1:
            # Use the ones where h[i+s_max] = 0 to trigger a potential improvement
            indices_good = []
            for i, j in candidates:
                if j < n and h[j+s_max] == 0:
                    indices_good.append((i, j))
            
            if indices_good:
                # Try a few random swaps from good candidates
                idx = np.random.randint(len(indices_good))
                i, j = indices_good[idx]
            else:
                # Default to arbitrary fix
                i, j = candidates[0]
        elif len(candidates) == 1:
            i, j = candidates[0]
        else:
            # No good candidates near current worst shift
            # Just pick random good pair:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) > 0 and len(zeros) > 0:
                i = np.random.choice(ones)
                j = np.random.choice(zeros)
            else:
                break  # No valid moves

        # Test tabu
        if (i, j) in tabu_list or (j, i) in tabu_list:
            continue
            
        # Save old max
        old_max = max_g
        
        # Compute the new g array
        new_g_s = g_s.copy()
        # Update all affected entries
        for s in range(2*n - 1):
            delta = delta_g_s(h, i, j, s)
            new_g_s[s] += delta
        
        # Check improvement
        new_max = np.max(new_g_s)
        
        if new_max < old_max:
            # Accept improvement
            h[i] = 1 - h[i]
            h[j] = 1 - h[j]
            g_s = new_g_s
            max_g = new_max
            max_g_idx = np.argmax(g_s)
            swaps_no_improve = 0
        else:
            swaps_no_improve += 1
            
        # Update tabu list (if not too long)
        if len(tabu_list) >= tabu_max_length:
            tabu_list.pop(0)
        tabu_list.append((i, j))
        
        # Occasionally perturb
        if swaps_no_improve > max_swaps_no_improve:
            # Mutate small block of bits to escape basin
            half = n // 2
            start_index = np.random.randint(0, n - half)
            for k in range(start_index, start_index + half):
                if np.random.rand() < 0.05:
                    h[k] = 1 - h[k]
            
            # Recompute after mutation
            g_s = compute_g_s(h)
            max_g = np.max(g_s)
            max_g_idx = np.argmax(g_s)
            swaps_no_improve = 0
        
    return h, max_g

def upscale_h(h_old, n_new):
    """Upscale step function: replicate entries"""
    n_old = len(h_old)
    rate = n_new // n_old
    upscaled = []
    for i in range(n_old):
        upscaled.extend([h_old[i]] * rate)
    # If size mismatch, pad or truncate
    if len(upscaled) < n_new:
        upscaled += [float(h_old[-1])] * (n_new - len(upscaled))
    elif len(upscaled) > n_new:
        upscaled = upscaled[:n_new]
    upscaled = np.array(upscaled)
    # Adjust to keep sum = 0.5 * n_new
    actual_sum = np.sum(upscaled)
    target_sum = n_new / 2
    if abs(actual_sum - target_sum) > 1e-5:
        # Adjust by rescaling
        upscaled = upscaled * (target_sum / actual_sum)
    return upscaled

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    n0 = 120
    h = generate_sturmian_binary(n0)
    print(f"Generated initial binary for n={n0}")
    
    # First stage: coarse local search
    remaining_time = budget_s * 0.5
    h_final, c5_final = local_swap_search(h, remaining_time)
    
    # Second stage: upscale and refine
    n1 = 240
    h_up = upscale_h(h_final, n1)
    print(f"Upscaled to n={n1}")
    h_final, c5_final = local_swap_search(h_up, remaining_time)
    
    # Final refinement as needed
    if time.time() < start_time + 0.9 * budget_s:
        # Perform small number of iterations of continuous refinement if valid
        h_final = np.clip(h_final, 0, 1)
        h_final = np.array(h_final)
    
    # Evaluate final result
    final_c5 = evaluate_erdos_solution(h_final)
    
    return (h_final.tolist(), final_c5, len(h_final))
