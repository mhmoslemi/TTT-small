import numpy as np
import time
import random
from scipy.optimize import differential_evolution, minimize


def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    overlap = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(overlap) * dx
    return c5


def propose_move(h, max_step):
    """Propose a new feasible h vector by adjusting two elements"""
    h_new = h.copy()
    N = len(h)
    
    # Try to find a valid move
    max_tries = 100
    for _ in range(max_tries):
        i = random.randint(0, N - 1)
        j = random.randint(0, N - 1)
        if i == j:
            continue
            
        # Determine feasible interval for δ
        delta_low = max(-h_new[i], h_new[j] - 1)
        delta_high = min(1 - h_new[i], h_new[j])
        
        if delta_low >= delta_high:
            continue
            
        # Sample δ from uniform distribution over feasible range
        step = min(max_step, delta_high - delta_low)
        if step <= 0:
            continue
        delta = random.uniform(delta_low, delta_high)
        
        # Apply move
        h_new[i] += delta
        h_new[j] -= delta
        
        return h_new
    
    # If no valid move found, return original
    return h.copy()


def optimize_with_de(h_start, max_iter=1000):
    """Use differential evolution to get a good starting point"""
    N = len(h_start)
    bounds = [(0, 1) for _ in range(N)]
    
    def objective(h):
        # Ensure the constraint ∫₀² h(x) dx = 1
        dx = 2.0 / N
        if abs(np.sum(h) * dx - 1.0) > 1e-6:
            # penalty for violating constraint
            penalty = 1000 * abs(np.sum(h) * dx - 1.0)
        else:
            penalty = 0
        return c5_of(h) + penalty
    
    result = differential_evolution(
        objective, 
        bounds, 
        maxiter=max_iter,
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.7,
        seed=42,
        disp=False)
    
    return result.x


def local_search(h_start, max_iter=2000, lr=0.01):
    """Run SGD with momentum to refine solution"""
    h = h_start.copy()
    N = len(h)
    dx = 2.0 / N
    best_h = h.copy()
    best_c5 = c5_of(h)
    momentum = np.zeros_like(h)
    
    for i in range(max_iter):
        if i % 100 == 0:
            # Periodically recompute h to avoid drift
            h = best_h.copy()
        c5_prev = c5_of(h)
        
        # Compute gradient and update
        eps = 1e-5
        grad = np.zeros_like(h)
        for j in range(N):
            h_plus = h.copy()
            h_plus[j] += eps
            if h_plus[j] <= 1:
                grad[j] = (c5_of(h_plus) - c5_prev) / eps
            else:
                grad[j] = 0  # prevent jumping out of bounds
                
        # Apply gradient with momentum and projection
        momentum = 0.9 * momentum + lr * grad
        h -= momentum
        
        # Project to feasible set [0, 1]
        h = np.clip(h, 0, 1)
        
        # Re-normalize
        h = h * (N / 2) / np.sum(h)
        
        c5 = c5_of(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
    return best_h, best_c5

    # 2. Slightly hold-the-line approach
    h_hold = np.full(N, 0.5)
    mid = N // 2
    h_hold[mid - 10:mid + 10] = 0.7
    
    # 3. Randomly initialized with fixed sum
    h_rand = np.random.rand(N)
    h_rand = h_rand * (N / 2) / np.sum(h_rand)
    
    # Use DE to get very good starting points
    start_points = [h_provided, h_hold, h_rand]
    best_h = None
    best_c5 = float('inf')

    for i, h_seed in enumerate(start_points):
