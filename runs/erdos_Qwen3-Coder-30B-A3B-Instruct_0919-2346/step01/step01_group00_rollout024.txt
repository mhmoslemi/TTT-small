import numpy as np
import time
import random
from scipy.optimize import minimize

def project_h(h, target_sum):
    """Project h onto box constraints [0,1] and sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Use binary search on lambda to find offset
    # such that sum of clip(h - lambda, 0, 1) = target_sum
    low, high = -1.0, 1.0
    tol = 1e-8
    
    def sum_with_lambda(lmbda):
        proj = np.clip(h - lmbda, 0, 1)
        return np.sum(proj)
    
    # Ensure bounds work
    while sum_with_lambda(low) < target_sum:
        low -= 1.0
    while sum_with_lambda(high) > target_sum:
        high += 1.0
    
    # Binary search
    while high - low > tol:
        mid = (low + high) / 2
        if sum_with_lambda(mid) >= target_sum:
            high = mid
        else:
            low = mid
    
    lmbda = (low + high) / 2
    h_proj = np.clip(h - lmbda, 0, 1)
    return h_proj

def compute_C_and_grad(h, alpha=50):
    """Compute max overlap C and its gradient."""
    N = len(h)
    dx = 2.0 / N
    
    # Full correlation of h with 1-h
    corr = np.correlate(h, 1 - h, mode="full")
    C = corr * dx
    
    # Compute max C
    max_C = np.max(C)
    
    # Compute smoothed objective
    exp_C = np.exp(alpha * C)
    w = exp_C / np.sum(exp_C)
    
    # Compute gradients via convolution of w with h
    # Compute w_rev for shift + conv
    w_rev = w[::-1]
    # v_plus = sum_k w_k * h_{i+k}
    v_plus = np.convolve(h, w_rev, mode="same")
    # v_minus = sum_k w_k * h_{i-k}
    v_minus = np.convolve(h, w, mode="same")
    
    # Gradient: g_i = dx * [1 - v_plus[i] - v_minus[i]]
    grad = dx * (1 - v_plus - v_minus)
    
    # Gradient has zero sum (should be projected)
    
    return max_C, grad

    start = time.time()

    alpha_schedule = [5, 20, 50, 80]  # Increased gradually
    alpha_idx = 0
    
    # Warm up: compute initial value
    max_C_current, _ = compute_C_and_grad(h_best, alpha_schedule[alpha_idx])
    
    while time.time() - start < budget_s - 5:
        if iteration % 100 == 0:

        max_C_current, grad = compute_C_and_grad(h_best, alpha)
        
        # Update h

        h_new = project_h(h_new, target_sum)
        
        # Compute new C5
        max_C_new, _ = compute_C_and_grad(h_new, alpha)
        
        if max_C_new <= max_C_current:

            max_C_current = max_C_new
