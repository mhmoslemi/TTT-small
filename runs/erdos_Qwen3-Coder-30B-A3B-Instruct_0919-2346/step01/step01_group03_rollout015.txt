import numpy as np
import time
from scipy.optimize import minimize

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral via full convolution of h with (1-h)
    overlaps = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum of the full correlation vector
    # scaled by dx
    c5 = np.max(overlaps) * dx
    return c5

def project_to_feasible(h):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = 1."""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to sum to 1
    h = h * (len(h) / 2) / np.sum(h)
    return h

def local_search_step(h, step_size=0.01):
    """Make a small local change to h to explore neighborhood."""
    h = h.copy()
    N = len(h)
    # Try a few random perturbations
    for _ in range(10):
        i = np.random.randint(0, N)
        delta = np.random.uniform(-step_size, step_size)
        # Adjust i and a nearby point to maintain sum constraint
        j = (i + 1) % N
        # Ensure the move is valid
        new_h_i = h[i] + delta
        new_h_j = h[j] - delta * (N - 1) / N
        if 0 <= new_h_i <= 1 and 0 <= new_h_j <= 1:
            h[i] = new_h_i
            h[j] = new_h_j
            # Project back to feasible region
            h = project_to_feasible(h)
            return h
    return h

    np.random.seed(seed)
    start = time.time()
    N = 140
    dx = 2.0 / N

        h = h0 * (N / 2) / np.sum(h0)
    else:
        # Fallback to a more structured initial construction
        h = np.ones(N) * 0.5
        h = h * (N / 2) / np.sum(h)
    
    # Optimization loop
    best_h = h.copy()
    best_c5 = c5_of(best_h)
    initial_c5 = best_c5
    
    # Gradient free optimization with local search
    T_start = 0.1
    T = T_start
    no_improvement_count = 0
    patience = 100
    
    while time.time() - start < budget_s:
        if no_improvement_count > patience:
            break
            
        # Construct a neighbor solution using local search
        h_new = local_search_step(best_h, T)
        c_new = c5_of(h_new)

        if c_new < best_c5:
            best_h = h_new
            best_c5 = c_new
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Cooling schedule
        T = max(T * 0.9995, 1e-6)
        
        # Occasionally reinitialize with a better point from known good constructions
        if np.random.rand() < 0.05 and time.time() - start < budget_s * 0.9:
            h_new = project_to_feasible(h + np.random.normal(0, 1e-3, N))
            c_new = c5_of(h_new)
            if c_new < best_c5:

                best_c5 = c_new
