from scipy.optimize import differential_evolution, minimize

def c5_of(h):
    """Compute C5 = max_k integral h(x)(1-h(x+k)) dx."""
    dx = 2.0 / len(h)
    # Compute correlation: h * (1 - h) with full mode
    c = np.correlate(h, 1 - h, mode="full")
    # Multiply by dx to get continuous integral
    c *= dx
    # Return maximum value among all shifts
    return np.max(c)

def project_to_feasible(h):
    """Project h to feasible domain [0,1] and ensure integral constraint."""
    h = np.clip(h, 0, 1)
    # Renormalize to sum to 1
    dx = 2.0 / len(h)
    h = h * (1.0 / np.sum(h)) * (2.0 / dx)
    return h
