import numpy as np
import time
import random
from scipy.optimize import minimize

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using full convolution
    overlap = np.correlate(h, 1 - h, mode='full')
    # The maximum value scaled by dx
    c5 = np.max(overlap) * dx
    return c5

def propose_move(h, max_step):
    """Propose a move that maintains sum constraint exactly"""
    h_new = h.copy()
    N = len(h)
    while True:
        i, j = random.sample(range(N), 2)
        # Compute feasible range for delta
        delta_low = max(-h_new[i], h_new[j] - 1.0)
        delta_high = min(1.0 - h_new[i], h_new[j])
        if delta_low >= delta_high:
            continue  # No valid move, try again
        break
    
    # Sample delta from feasible interval
    step = min(max_step, delta_high - delta_low)
    if step <= 0:
        delta = 0.0
    else:
        delta = random.uniform(max(-step, delta_low), min(step, delta_high))
        
    h_new[i] += delta
    h_new[j] -= delta
    return h_new

def sobol_sampling(n):
    """Generate Sobol sequence for better coverage"""
    # Using a simple low-discrepancy sequence for initialization
    return np.array([((i * 2.618033988749895) % 1) for i in range(n)])

def initialize_h(N):
    """Create initial h values using multiple strategies"""
    # 1. Cell-based approach - refine initial_h_values if available
    try:
        h_initial = initial_h_values
        if len(h_initial) != N:

            h_initial = np.interp(x_new, x_orig, h_initial)
        h_initial = h_initial * (N / 2) / np.sum(h_initial)
    except:
        h_initial = np.ones(N) * 0.5

    # 2. Add some Sobol-based structure
    sobol_vals = sobol_sampling(N)
    h_sobol = 0.3 + 0.4 * sobol_vals
    
    # 3. Smoothed version of initial
    h_smooth = np.convolve(h_initial, np.ones(5)/5, mode='same')
    h_smooth = h_smooth * (N / 2) / np.sum(h_smooth)
    
    # Combine strategies
    h_combined = 0.4 * h_smooth + 0.3 * h_initial + 0.3 * h_sobol
    h_combined = h_combined * (N / 2) / np.sum(h_combined)
    
    return h_combined

def optimize_with_cma_es(initial_h, budget_s):
    """Use evolutionary strategy for optimization"""
    start = time.time()
    h_best = initial_h.copy()
    c_best = c5_of(h_best)
    
    N = len(initial_h)
    sigma = 0.05  # Initial step size
    lambda_ = 10  # Population size
    mu = 5  # Number of parents
    
    iteration = 0
    while time.time() - start < budget_s:
        # Generate offspring
        offspring = []
        for _ in range(lambda_):
            h_offspring = initial_h.copy()
            for i in range(N):
                if random.random() < 0.5:
                    h_offspring[i] += np.random.normal(0, sigma)
                    h_offspring[i] = max(0, min(1, h_offspring[i]))
            # Renormalize to sum = N/2
            h_offspring = h_offspring * (N / 2) / np.sum(h_offspring)
            offspring.append(h_offspring)
        
        # Evaluate offspring
        fitness = [c5_of(h) for h in offspring]
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        top_offspring = [offspring[i] for i in sorted_indices[:mu]]
        top_fitness = [fitness[i] for i in sorted_indices[:mu]]
        
        # Check if we improved
        if top_fitness[0] < c_best:
            h_best = top_offspring[0]
            c_best = top_fitness[0]
        
        # Update distribution
        if mu > 0:
            # Simple update of step size based on success rate
            success_rate = sum(1 for f in top_fitness if f < c_best + 1e-8) / mu
            if success_rate > 0.2:  # If good enough
                sigma *= 1.1
            else:
                sigma *= 0.9
                
        if iteration % 5 == 0 and time.time() - start > 0.8 * budget_s:
            # Local search
            h_new = h_best.copy()
            step = min(0.02, sigma)
            for _ in range(50):
                if time.time() - start > budget_s:
                    break
                new_h = propose_move(h_new, step)
                c_new = c5_of(new_h)
                if c_new < c_best:
                    h_new = new_h
                    c_best = c_new
            h_best = h_new
        
        iteration += 1
        if time.time() - start > budget_s * 0.95:
            break
            
    return h_best, c_best

        c_prop = c5_of(h_prop)

            c_current = c_prop
    
    return (h_current, c_current, N)
