import numpy as np
import time
import random
from scipy.signal import correlate


def compute_c5(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    overlap = correlate(h, 1 - h, mode="full")
    c5 = np.max(overlap) * dx
    return c5


def project_onto_feasible(h, n_points):
    """Project h to satisfy sum(h) * dx = 1 and 0 <= h[i] <= 1"""
    dx = 2.0 / n_points
    current_sum = np.sum(h) * dx
    h = np.clip(h, 0, 1)
    
    # Adjust to make the integral exactly 1
    if abs(current_sum - 1.0) > 1e-10:
        scale = 1.0 / current_sum
        h *= scale
    return h


def compute_gradient(h, k):
    """Compute gradient w.r.t. h for shift k"""
    # h is the step function
    # Gradient of h(x)(1 - h(x+k)) w.r.t. h(x)
    # = (1 - h(x+k)) - h(x)*delta(x - (x+k)) = (1 - h(x+k))
    # This is when we're looking at the x position
    
    # Another way to think:
    # For fixed k, d/dh_i [int h(x)(1-h(x+k)) dx] 
    # = (1 - h(i*dx + k)) 
        
    # But because of convolution, d/dh_i[overlap] = 1 - h[i-k] when i-k >= 0
    # if we consider correlation as a function of h and k.
    # Actually, it's easier to compute the gradient as:
    # approximately, d/dh_i [sum over all shifts of h_j * (1 - h_{j+k}) 
    # -> j = i, so the gradient w.r.t h_i is 1 - h_{i+k}
    
    # If we want backward shift (i-k):
    # invert shift  
    # h(i) contributes 1 - h(i+k) to the overlap integral
    # So its contribution to gradient at i gives:
    # d/dh_i [h(i) * (1 - h(i+k))] = 1 - h(i+k)  
    
    # For all shifts:
    # For overlap at shift k, the contribution to d/dh_j from h_j appearing 
    # in overlap at shift k varies as (1 - h(j-k)) when j-k >= 0
    # So we have a compact way:
    # d/dh_j [overlap] = convolve(1 - h, delta_{k})(j) -> this just means 
    # the [] shifted version: (sum of (1-h_i) at k positions) at j-th position
    # But we want the k=0 case for the original minus some h_j娣盯 hd.
    
    # This is the gradient w.r.t. the full vector h!
    # d/dh_j [sum_k h_j * (1 - h_{j+k})] = 1 - h_{j+k} (this is the gradient!)
    # For k=0 is same values appropriate at shift = 0 (sum of what?)
    # What we want: Gradients for each shift?
    
    # Let's simplify:
    # For overlap defined by forward correlation of (h, 1-h), 
    # for every i,k: overlap[k] = sum_{j} h[j]* (1-h[j+k]) 
    # rather, we should compute:
    # J = sum_k int h(x)(1 - h(x + k)) dx = sum_k overlap[k]*dx = sum_k [h * (1-h)][k] * dx
    
    # d/d h_j [h * (1 - h)][k] = 
    # if i = k + j, then 1 - h(i) = 1 - h(k + j) = d/d h_j [beta_{j kd k + j}]= 1-h(k+j)
    # So: for all k, sum_j h_j * (1-h_{j+k}) 
    # d/d h_j [h * (1 - h)][k] = 1 - h_{j + k}  
    # actually  
    # sum_k [h * (1 - h)][k] 
    # d/dh_j [sum_k [h * (1 - h)][k]] = d/dh_j (sum_k sum_i h_i * (1-h_{i + k}))  
    # = sum_k (1 - h_{(j + k)}) = sum_{offset} (1 - h_{j + offset}) 
    # In the convolution, this should be equivalent to the convolution of 1-h with a unit impulse at j: 
    # But a more elegant formulation:
    # d_j [sum_k int h(x) (1 - h(x + k)) dx] = (1 - h*(delta_{j*k})(x-k)) for fixed shift k,
    # which reduces to convolution with (1-h) file. Gamma(x) = (1-h)*(1) = 1 - h.
    # So for x from 0 to 2, it will be the convolution. 
    # But really the gradient w.r.t. to each h_i of the maximum of the convolution!
    # Argmax_k [sum h_j (1 - h_{j+k}) ]
    # Important: What are we taking the derivative of?
    # C5 = max_k [int h(x)(1 - h(x+k)) dx]
    # = max_k [sum_{j=0}^N h_j (1 - h_{j+k}) dx]
    # dC5/dh_i = max_k [d/d h_i [int h(x) (1 - h(x+k)) dx]]
    # the integral max over k, so to find which k dominates?
    # Assuming k=q for some optimizing q:
    # dC5/dh_i = d/dh_i [ h(x) (1 - h(x+q)) ] = 1 - h_{i+q} if x = i*dx, else 0. 
    # This is not the correct derivative for this maximum ...
    # Really, custom C5 gradient is:
    # Let C5 = max_q [sum_i h_i (1 - h_{i+q}) dx] = max_q F(q) 
    # dC5/dh_i = 1 - h_{i+q^*}  where q^* = argmax_q F(q) 
    # That is: find the k such that 
    # dC5/dh_i depends on optimal shift k_star!
    # So: run optimization on beta = conv(1-h, h) score...
    
    # More straightforward approach for gradient estimation:
    # Assume the shift k dominates, then dC5/dh_i = (1 - h_{i+k})
    # This gradient is for shift k_star. 
    # If we are going to use a gradient step with direction
    # Do pick k_star from the current computation of overlaps.
    # For simplicity:
    overlap = correlate(h, 1 - h, mode="full")
    max_k = np.argmax(overlap)
    # so the gradient of J wrt h_i is (1 - h_{i+max_k})...
    # treat it element-wise.
    grad = np.zeros_like(h)
    # move k backwards 
    if max_k < len(h):
        grad[:len(h)-max_k] = 1 - h[max_k:]
    # if max_k > len(h), we expect to be from the end part (i+max_k >= N)
    else:
        # wrap back its reflection? 
        # Different case . 
        # Assume we add right to left:
        # For 1 - h_{i+max_k} = 1 - h_{i + max_k} 
        # Equivalent Is to: 1 - h_{i+max_k} -> 
        # Actually this is less precise. Instead just 
        # reconstruct the correct convolution of overlaps and then form the true gradient.
        
    # Do it carefully:
    # For shift k, the integral contributes to sum over i of h_i (1-h_{i+k})
    # We define partial derivatives as:
    # d(J)/d(h_i) = sum over k: d(J_k)/d(h_i) = sum_k 1-h_{i+k} 
    # since the gradient is from the k that was our maximum:
    # Each term in 'overlap' dt scale stirngs! 
    # Let the current k be fixed for maximum: 
    # dC5/dh_j = [1 - h_{j+k_star}]  where k_star is the maximizing k
    # Add the term for all positions.
    
    # Simple evaluation on 1-d:
    # d/dh_i of overlap[k] = h_{i+k} when i+k <= N-1
    # Can just evaluate that each shift contributes to term in vector.
    
    # Let's invert. 
    M = len(h)
    # Set up shift contribution for each k
    potential_max_k = np.argmax(overlap)  # max k, then 1 - h_{i+k}
    if potential_max_k < M: 
        # aj = 1 - h_{i+max_k}  for i+k < M, i.e., i < M-max_k
        # so it's a shifted 1 - h
        # Put h in appropriate array positions:
        # Now we want grad[i] = 1 - h_{i+max_k} for valid positions...
        grad = np.zeros_like(h)
        # i = 0, 1, ..., max_k-1 ->  indices in i + max_k -> max_k, ..., M-1
        # So i + max_k = j, j goes from max_k to M-1 
        # gradient{j} = 1 - h_{j - max_k}
        if max_k > 0:
            # Copy backward from 0 to M - max_k
            grad[max_k:] = 1 - h[:-max_k]
    # otherwise, it's edge case
            
    return grad


def gradient_descent_optimize(initial_h, n_steps=5000, lr=0.01, beta=1.0):
    """Run projected gradient descent"""
    h = initial_h.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    
    for step in range(n_steps):
        # Compute current C5 and the optimal shift
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        # Compute gradient approximated to optimal shift k*
        grad = np.zeros_like(h)
        if max_k > 0:
            grad[max_k:] = 1 - h[:-max_k]
        # Normalize gradient
        grad = grad / (np.linalg.norm(grad) + 1e-8)
        h = h - lr * grad 
        # Project back to feasible set
        h = project_onto_feasible(h, n_points)
        if step % 1000 == 0:
            c5 = compute_c5(h)
            # print(f"Step {step}, C5 = {c5:.6f}")
    
    # Final projection
    h = project_onto_feasible(h, n_points)
    return h


def run(seed=42, budget_s=600, **kwargs):

    random.seed(seed)

    h_b[:n_points//4] = 1.0
    h_b[n_points//4:3*n_points//4] = 0.0
    h_b[3*n_points//4:] = 1.0
    h_b = project_onto_feasible(h_b, n_points)
    seeds.append(h_b)
    
    # 4. Random sample
    h_r = np.random.uniform(0, 1, n_points)
    h_r = project_onto_feasible(h_r, n_points)
    seeds.append(h_r)
    
    # Use exactly 1/4 of time per seed
    max_time_per_seed = budget_s * 0.2  # Leave some time at the end
    best_h = None
    best_c5 = float('inf')
    
    for i, h_seed in enumerate(seeds):
        if time.time() - start >= budget_s - 10:
            break
        h_opt = gradient_descent_optimize(h_seed, n_steps=1000, lr=0.01)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            
    # Final polishing with more directed refinement using projected gradient updates
    # Track time for final polishing

    if remaining_time > 5:
        # Run short greedy optimized loop with 1000 steps
        h_final = gradient_descent_optimize(best_h, n_steps=5000, lr=0.005)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5:
            best_c5 = c5_final
