```python
import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize_scalar

def compute_c5(h):
    """Compute the maximum overlap integral for the given step function h."""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    c = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(c) * dx
    return c5

def waterfill_projection(h, target_sum):
    """Project h onto the set of vectors with sum = target_sum and bounds [0,1]."""
    n = len(h)
    # Ensure h is within [0,1]
    h = np.clip(h, 0, 1)
    # Waterfilling
    h_sorted = np.sort(h)[::-1]
    s = 0
    k = 0
    for i in range(n):
        s += h_sorted[i]
        if s > (target_sum - i * 0):
            k = i
            break
    else:
        k = n
    lambda_v = (target_sum - s) / (n - k) if k < n else 0
    result = np.maximum(h - lambda_v, 0)
    return result

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start = time.time()
    N = 140
    dx = 2.0 / N
    target_c5 = 0.3807
    
    # Build list of valid (i,s) pairs
    pairs = []
    for s in range(N):  # Only non-negative shifts for now
        for i in range(N - s):
            pairs.append((i, s))
    
    # Helper: compute shift sums quickly
    def shift_sum_indices(s):
        # Given shift s, get all i such that i+s < N
        indices = list(range(N - s))
        return indices, s
    
    # Phase 0: Setup index mapping
    if time.time() - start > 0.1 * budget_s:
        return None, float('inf'), 0
    print("Building index structures...")
    
    # Prepare data for LP formulation
    num_pairs = len(pairs)
    # For each shift, need to know which pairs contribute to it
    shift_indices = {}
    for s in range(N):
        shift_indices[s] = []
        for i in range(N - s):
            idx = pairs.index((i, s))
            shift_indices[s].append(idx)
    
    # Phase 1: Solve LP relaxation
    print("Solving LP relaxation...")
    h = cp.Variable(N, name='h')
    y = cp.Variable(num_pairs, name='y')
    alpha = cp.Variable(name='alpha')
    
    # Constraints
    constraints = [
        0 <= h,
        h <= 1,
        cp.sum(h) == N/2,
        alpha >= 0
    ]
    
    for i, s in pairs:
        # McCormick envelope
        constraints.extend([
            y[pairs.index((i,s))] <= h[i],
            y[pairs.index((i,s))] <= h[i+s],
            y[pairs.index((i,s))] >= h[i] + h[i+s] - 1,
            y[pairs.index((i,s))] >= 0
        ])
    
    # For each shift s, enforce α ≥ dx * (N/2 - Σ_y[i,s])
    for s in range(N):
        if len(shift_indices[s]) > 0:
            y_sum = cp.sum([y[j] for j in shift_indices[s]])
            constraints.append(alpha >= dx * (N/2 - y_sum))
    
    prob = cp.Problem(cp.Minimize(alpha), constraints)
    
    # Solve
    prob.solve(solver='ECOS', max_iters=20000)
    
    if prob.status != 'optimal' and prob.status != 'approximal':
        print("Warning: LP solver status:", prob.status)
    
    h_val = h.value
    alpha_val = alpha.value
    
    print("LP solution gives α =", alpha_val)
    
    if alpha_val <= target_c5:
        print("LP solution already meets target!")
        # Check actual C5
        c5_val = compute_c5(h_val)
        print("True C5 = ", c5_val)
        return (h_val, c5_val, N)
    
    # Phase 2: MILP with binary constraints
    print("Trying MILP...")
    # We must build a new model with h as binary
    h_bin = cp.Variable(N, boolean=True, name='h')
    y_bin = cp.Variable(num_pairs, name='y')
    alpha_bin = cp.Variable(name='alpha')
    
    # Constraints
    constraints_bin = [
        cp.sum(h_bin) == N/2,
        alpha_bin >= 0
    ]
    
    for i, s in pairs:
        # McCormick envelope; exact for binary variables
        constraints_bin.extend([
            y_bin[pairs.index((i,s))] <= h_bin[i],
            y_bin[pairs.index((i,s))] <= h_bin[i+s],
            y_bin[pairs.index((i,s))] >= h_bin[i] + h_bin[i+s] - 1,
            y_bin[pairs.index((i,s))] >= 0
        ])
    
    # For each shift s, enforce α ≥ dx * (N/2 - Σ_y[i,s])
    for s in range(N):
        if len(shift_indices[s]) > 0:
            y_sum = cp.sum([y_bin[j] for j in shift_indices[s]])
            constraints_bin.append(alpha_bin >= dx * (N/2 - y_sum))
    
    prob_bin = cp.Problem(cp.Minimize(alpha_bin), constraints_bin)
    
    # Solve with time limit
    remaining_time = budget_s - (time.time() - start)
    if remaining_time < 10:
        print("Too little time for MILP, returning best known")
        c5_val = compute_c5(h_val)
        print("Best C5 from LP is", c5_val)
        return (h_val, c5_val, N)
    
    try:
        prob_bin.solve(solver=cp.CBC, maxSeconds=remaining_time - 5)
        h_sol_bin = h_bin.value
        if h_sol_bin is not None and prob_bin.status in ['optimal', 'approximal']:
            h_sol = h_sol_bin
            c5_sol = compute_c5(h_sol)
            print("Binary MILP found C5 = ", c5_sol)
            return (h_sol, c5_sol, N)
        else:
            print("MILP did not find solution")
    except Exception as e:
        print("Error in MILP solve:", str(e))
        pass
    
    # Phase 3: Fallback to rounding and refinement
    print("Falling back to rounding and local refinement...")
    
    # 1. Greedy rounding of LP solution to binary
    h_rounded = np.zeros(N)
    sorted_indices = np.argsort(h_val)[::-1]
    for i in range(N//2):
        h_rounded[sorted_indices[i]] = 1
    
    # 2. Project onto sum constraint
    h_rounded = waterfill_projection(h_rounded, N/2)
    h_rounded = np.clip(h_rounded, 0, 1)
    
    # Local refinement
    def compute_c5_gradient(h, s):
        # Compute gradient of C_s w.r.t. h
        # C_s = ∫ h(x)(1 - h(x+s)) dx
        # Gradient w.r.t. h[i] = (1 - h[i+s]) if s>=0,  (1 - h[i-s]) if s<0
        # but in discrete, it's:
        n = len(h)
        grad = np.zeros(n)
        h_shift = np.roll(h, -s) if s >= 0 else np.roll(h, s)
        for i in range(n):
            if i + s < n and i + s >= 0:
                grad[i] = 1 - h[i + s]
        return grad
    
    # Refinement loop
    best_h = h_rounded.copy()
    best_c5 = compute_c5(best_h)
    max_iterations = 500
    for _ in range(max_iterations):
        if time.time() - start > budget_s * 0.95:
            break
        c5_vals = np.array([np.sum(h * (1 - np.roll(h, s))) for s in range(N)])
        worst_s = np.argmax(c5_vals)
        grad = compute_c5_gradient(best_h, worst_s)
        # Move in direction of negative gradient with step size
        step_size = 0.02
        h_new = best_h - step_size * grad
        # Project to feasible set
        h_new = np.clip(h_new, 0, 1)
        h_new = waterfill_projection(h_new, N/2)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
            print(f"Refinement step: C5 = {best_c5:.6f}")
    
    print("Final refined C5 = ", best_c5)
    return (best_h, best_c5, N)
```
