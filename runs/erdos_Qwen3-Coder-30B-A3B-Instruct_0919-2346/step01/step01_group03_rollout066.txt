import numpy as np
import time
from scipy.optimize import minimize

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral via full convolution of h with (1-h)
    overlaps = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum of the full correlation vector
    # scaled by dx
    c5 = np.max(overlaps) * dx
    return c5

def propose_move(h, max_step):
    """Propose a feasible move by adjusting two entries."""
    h = h.copy()
    N = len(h)
    # Try to find a valid pair of indices
    for _ in range(100):  # avoid infinite loop
        i, j = np.random.randint(0, N, 2)
        if i == j:
            continue
        # Determine feasible δ range
        delta_low = max(-h[i], h[j] - 1)
        delta_high = min(1 - h[i], h[j])
        if delta_low >= delta_high:
            continue
        # Sample a move with limited step size
        step = min(max_step, delta_high - delta_low)
        delta = np.random.uniform(
            max(-step, delta_low),
            min(step, delta_high)
        )
        # Apply move
        h[i] += delta
        h[j] -= delta
        return h
    # If no valid move found, return copy
    return h

def simple_gradient_descent(h, step_size=0.01, max_iter=1000):
    """Simple gradient-based descent"""
    h = h.copy()
    N = len(h)
    dx = 2.0 / N
    
    for _ in range(max_iter):
        # Compute gradient numerically
        grad = np.zeros(N)
        c0 = c5_of(h)
        eps = 1e-6
        for i in range(N):
            h_eps = h.copy()
            h_eps[i] += eps
            c_eps = c5_of(h_eps)
            grad[i] = (c_eps - c0) / eps
        
        # Constraining gradient
        # Ensure that h remains in [0,1] and sum = 1
        h_new = h - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        # Adjust for sum constraint
        h_new *= (N / 2) / np.sum(h_new)
        c_new = c5_of(h_new)
        
        if c_new < c0:
            h = h_new
        else:
            break
    return h

def anneal(initial_h, time_budget, seed):
    """Run simulated annealing from initial_h."""
    np.random.seed(seed)
    start = time.time()
    cur = initial_h.copy()
    best = initial_h.copy()
    c_cur = c5_of(cur)
    c_best = c_cur
    T_start = 0.05
    T_end = 1e-6
    step_start = 0.2
    T = T_start
    step = step_start

    while time.time() - start < time_budget:
        elapsed = time.time() - start
        tfrac = elapsed / time_budget
        T = T_start * (T_end / T_start) ** tfrac
        step = step_start * (1 - tfrac)

        # Propose a move
        h_new = propose_move(cur, step)
        c_new = c5_of(h_new)

        # Acceptance rule
        if c_new <= c_cur or np.random.rand() < np.exp(-(c_new - c_cur) / T):
            cur = h_new
            c_cur = c_new

        # Update global best if improved
        if c_new < c_best:
            best = h_new
            c_best = c_new

    return best, c_best
