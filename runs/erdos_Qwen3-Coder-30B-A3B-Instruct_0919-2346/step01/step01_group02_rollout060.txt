import numpy as np
import time

def compute_c5_and_worst_shift(h):
    """Compute C5 and return the worst shift index."""
    dx = 2.0 / len(h)
    c = np.correlate(h, 1 - h, mode='full') * dx
    max_idx = np.argmax(c)
    s = max_idx - (len(h) - 1)
    c5 = c[max_idx]
    return c5, s

def compute_gradient(h, shift):
    """Compute gradient of C5 with respect to h for a given shift."""
    N = len(h)
    dx = 2.0 / N
    g = np.full(N, 1.0) * dx
    if shift >= 0:
        if shift < N:
            g[:N-shift] -= h[shift:] * dx
            g[shift:] -= h[:N-shift] * dx
    else:
        shift = -shift
        if shift < N:
            g[shift:] -= h[:N-shift] * dx
            g[:N-shift] -= h[shift:] * dx
    return g

def project_onto_bounds_and_sum(v, target_sum):
    """Project vector v onto box [0,1] and sum = target_sum."""
    # Binary search for lambda such that sum(clip(v + lambda, 0, 1)) = target_sum
    lower = -np.max(v)
    upper = 1 - np.min(v)
    for _ in range(60):
        mid = (lower + upper) / 2.0
        w = np.clip(v + mid, 0, 1)
        s = np.sum(w)
        if s < target_sum:
            lower = mid
        else:
            upper = mid
    w = np.clip(v + (lower + upper) / 2.0, 0, 1)
    # Adjust to ensure exact sum
    diff = target_sum - np.sum(w)
    if abs(diff) > 1e-12:
        free_idx = (w > 0) & (w < 1)
        if np.any(free_idx):
            w[free_idx] += diff / np.sum(free_idx)
            w = np.clip(w, 0, 1)
        else:
            w += diff / len(w)
            w = np.clip(w, 0, 1)
    return w

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    N = 140
    dx = 2.0 / N

    # Initialize from initial_h_values if available, otherwise uniform
    if 'initial_h_values' in globals():
        h0 = np.interp(np.linspace(0, 2, N),
                      np.linspace(0, 2, len(initial_h_values)),
                      initial_h_values)
        h0 = h0 * (N/2) / np.sum(h0)
    else:
        h0 = np.ones(N) * 0.5

    # Two-peak seeding
    h1 = np.ones(N)
    x_vals = np.linspace(0, 2, N)
    a = 0.3
    u = 0.9
    v = 0.1
    mask = (x_vals < a) | (x_vals >= 2 - a)
    h1[mask] = u
    h1[~mask] = v
    h1 = h1 * (N/2) / np.sum(h1)

    # Binary seed
    h2 = np.zeros(N)
    idx = np.random.choice(N, N//2, replace=False)
    h2[idx] = 1.0

    # Try several initial candidates
    initial_candidates = [h0, h1, h2]
    best_h = None
    best_c5 = float('inf')

    for h_seed in initial_candidates:
        # Ensure sum constraint exactly
        h_seed = h_seed * (N/2) / np.sum(h_seed)
        c5, _ = compute_c5_and_worst_shift(h_seed)
        if c5 < best_c5:
            best_h = h_seed.copy()
            best_c5 = c5

    # Gradient descent optimization
    start_time = time.time()
    max_time_fraction = 0.9  # use 90% of budget for optimization
    time_limit = start_time + budget_s * max_time_fraction
    time_remaining = lambda: time_limit - time.time()
    
    while time_remaining() > 0.1:
        # Compute worst shift and gradient
        c5_current, worst_shift = compute_c5_and_worst_shift(best_h)
        
        # Compute gradient for the worst shift
        grad = compute_gradient(best_h, worst_shift)
        
        # Adaptive step size
        elapsed = time.time() - start_time
        t_frac = min(elapsed / (budget_s * max_time_fraction), 1.0)
        base_step = 2.0 + 2.0 * (1 - t_frac)  # Starts at 4, reduces to 2
        step_size = base_step * (0.01 + 0.01 * (1 - t_frac))  # Step size decreasing
        
        # Update using gradient descent
        v = best_h - step_size * grad
        new_h = project_onto_bounds_and_sum(v, N/2)
        c5_new, _ = compute_c5_and_worst_shift(new_h)
        
        # Accept if better or with probability
        if c5_new < c5_current:
            best_h = new_h.copy()
            best_c5 = c5_new
        elif np.random.rand() < 0.05:  # Occasionally accept worse solutions
            best_h = new_h.copy()
            best_c5 = c5_new

        # Occasionally random perturbation
        if np.random.rand() < 0.1:
            noise = np.random.uniform(-0.1, 0.1, size=N)
            v = best_h + noise
            best_h = project_onto_bounds_and_sum(v, N/2)
            best_c5, _ = compute_c5_and_worst_shift(best_h)

    # Final local search
    for _ in range(5000):
        if time_remaining() < 0.1:
            break
        i, j = np.random.choice(N, size=2, replace=False)
        delta_low = max(-best_h[i], best_h[j] - 1)
        delta_high = min(1 - best_h[i], best_h[j])
        if delta_low >= delta_high:
            continue
        delta = np.random.uniform(-0.05, 0.05)
        delta = np.clip(delta, delta_low, delta_high)
        h_copy = best_h.copy()
        h_copy[i] += delta
        h_copy[j] -= delta
        # Normalize
        h_copy = h_copy * (N/2) / np.sum(h_copy)
        c5_new, _ = compute_c5_and_worst_shift(h_copy)
        if c5_new < best_c5:
            best_h = h_copy
            best_c5 = c5_new

    return (best_h, best_c5, N)
