import numpy as np
import time

def c5_of(h):
    """Compute the maximum overlap integral for the given step function h."""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    c = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(c) * dx
    return c5

def worst_shift(h):
    """Find the shift that maximizes the overlap integral."""
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    return idx - (len(h) - 1)

def subgradient(h, s):
    """Compute the subgradient with respect to the worst shift s."""
    N = len(h)
    dx = 2.0 / N
    m = abs(s)
    g = np.full(N, dx)
    if m < N:
        g[:N-m] -= dx * h[m:]
        g[m:]   -= dx * h[:N-m]
    return g

def project(v, target):
    """Euclidean projection onto box [0,1]^n and sum constraint sum(x)=target."""
    lo = -np.max(v)
    hi = 1.0 - np.min(v)
    for _ in range(50):
        lam = (lo + hi) / 2.0
        w = np.clip(v + lam, 0.0, 1.0)
        if w.sum() < target:
            lo = lam
        else:
            hi = lam
    w = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        free = (w > 0) & (w < 1)
        if free.any():
            w[free] += diff / free.sum()
            w = np.clip(w, 0.0, 1.0)
    return w

def propose_move(h, max_step):
    """Propose a new feasible step function by adjusting two entries."""
    N = len(h)
    while True:
        i, j = np.random.choice(N, size=2, replace=False)
        # Determine feasible step
        delta_low = max(-h[i], h[j] - 1)
        delta_high = min(1 - h[i], h[j])
        if delta_low >= delta_high:
            continue  # Try another pair
        # Sample step size
        step_size = np.random.uniform(max(-max_step, delta_low),
                                      min(max_step, delta_high))
        # Create new h
        h_new = h.copy()
        h_new[i] += step_size
        h_new[j] -= step_size
        return h_new

def spd_loop(init_h, time_budget, N):
    """Sub-gradient projected descent loop."""
    start = time.time()
    cur = init_h.copy()
    c_cur = c5_of(cur)
    best = cur.copy()
    c_best = c_cur
    T_init = 30.0
    T_decay = 0.9995
    t = 0
    while time.time() - start < time_budget:
        s = worst_shift(cur)
        g = subgradient(cur, s)
        eta = T_init * (T_decay ** t)
        if eta < 1e-8:
            eta = 1e-8
        v = cur - eta * g
        cur = project(v, N / 2)
        c_cur = c5_of(cur)
        if c_cur < c_best:
            best = cur.copy()
            c_best = c_cur
        t += 1
        # Occasionally jitter to escape local minima
        if t % 30 == 0:
            h_jitter = cur + np.random.uniform(-0.02, 0.02, size=N)
            h_jitter = project(h_jitter, N / 2)
            c_jitter = c5_of(h_jitter)
            if c_jitter < c_best:
                best = h_jitter
                c_best = c_jitter
                cur = h_jitter
    return best, c_best

def anneal(init_h, time_budget):
    """Simulated annealing with the move function."""
    start = time.time()
    cur = init_h.copy()
    c_cur = c5_of(cur)
    best = cur.copy()
    c_best = c_cur
    T_start = 0.05
    T_end = 1e-6
    max_step_start = 0.2
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        frac = elapsed / time_budget
        T = T_start * (T_end / T_start) ** frac
        max_step = max_step_start * (1.0 - frac)
        h_new = propose_move(cur, max_step)
        c_new = c5_of(h_new)
        if c_new <= c_cur or np.random.rand() < np.exp(-(c_new - c_cur) / T):
            cur = h_new
            c_cur = c_new
            if c_new < c_best:
                best = cur.copy()
                c_best = c_new
    return best, c_best

def hill_climb(init_h, time_budget):
    """Deterministic hill climb using small perturbations."""
    start = time.time()
    cur = init_h.copy()
    c_cur = c5_of(cur)
    best = cur.copy()
    c_best = c_cur
    while time.time() - start < time_budget:
        h_new = propose_move(cur, 0.02)
        c_new = c5_of(h_new)
        if c_new < c_best:
            best = h_new
            c_best = c_new
            cur = h_new
    return best, c_best

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    N = 140
    dx = 2.0 / N
    time_budgets = [0.05 * budget_s, 0.55 * budget_s, 0.30 * budget_s, 0.10 * budget_s]
    phase_names = ["seeding", "spd", "anneal", "hill_climb"]
    best_h = None
    best_c5 = float('inf')
    global_start = time.time()

    # Phase 0: Seeding
    seeds = []
    if 'initial_h_values' in globals():
        h0 = np.interp(np.linspace(0, 2, N),
                      np.linspace(0, 2, len(initial_h_values)),
                      initial_h_values)
        h0 = h0 * (N/2) / np.sum(h0)
        seeds.append(h0)
    else:
        h0 = np.ones(N) * 0.5
        seeds.append(h0)
    # Two-peak seed 
    h1 = np.ones(N)
    x_vals = np.linspace(0, 2, N)
    a = 0.3
    u = 0.9
    v = 0.1
    mask = (x_vals < a) | (x_vals >= 2 - a)
    h1[mask] = u
    h1[~mask] = v
    h1 = h1 * (N/2) / np.sum(h1)  # Renormalize
    seeds.append(h1)
    # Random binary vectors (exactly N/2 ones)
    random_bin = np.random.binomial(1, 0.5, size=N)
    random_bin = random_bin.astype(float) * 2
    seeds.append(random_bin)
    # Uniform
    h2 = np.ones(N) * 0.5
    seeds.append(h2)
    
    for h_seed in seeds:
        h_proj = project(h_seed, N/2)
        c5 = c5_of(h_proj)
        if c5 < best_c5:
            best_h = h_proj
            best_c5 = c5

    # Phase 1: SPD
    h_spd, c_spd = spd_loop(best_h, time_budgets[1], N)
    if c_spd < best_c5:
        best_h = h_spd
        best_c5 = c_spd

    # Phase 2: Anneal
    h_anneal, c_anneal = anneal(best_h, time_budgets[2])
    if c_anneal < best_c5:
        best_h = h_anneal
        best_c5 = c_anneal

    # Phase 3: Hill Climb
    h_hc, c_hc = hill_climb(best_h, time_budgets[3])
    if c_hc < best_c5:
        best_h = h_hc
        best_c5 = c_hc

    return (best_h, best_c5, N)
