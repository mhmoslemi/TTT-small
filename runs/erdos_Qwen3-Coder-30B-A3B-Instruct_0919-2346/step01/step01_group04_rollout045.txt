import numpy as np
import time
import random
from scipy.optimize import differential_evolution, minimize


def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    overlap = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(overlap) * dx
    return c5


def project_to_feasible(h):
    """Project h to satisfy constraints: 0<=h<=1 and sum(h)*dx=1"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Renormalize
    dx = 2.0 / len(h)
    h = h * (1.0 / np.sum(h)) * (1.0 / dx)
    return h


def propose_move(h, max_step):
    """Propose a new feasible h vector by adjusting two elements"""
    h_new = h.copy()
    N = len(h)
    
    # Try to find a valid move
    max_tries = 100
    for _ in range(max_tries):
        i = random.randint(0, N - 1)
        j = random.randint(0, N - 1)
        if i == j:
            continue
            
        # Determine feasible interval for δ
        delta_low = max(-h_new[i], h_new[j] - 1)
        delta_high = min(1 - h_new[i], h_new[j])
        
        if delta_low >= delta_high:
            continue
            
        # Sample δ from uniform distribution over feasible range
        step = min(max_step, delta_high - delta_low)
        if step <= 0:
            continue
        delta = random.uniform(delta_low, delta_high)
        
        # Apply move
        h_new[i] += delta
        h_new[j] -= delta
        
        return h_new
    
    # If no valid move found, return original
    return h.copy()


def anneal(initial_h, time_budget_s):
    """Run simulated annealing on the initial vector"""
    start = time.time()
    
    # Parameters
    T_start = 0.05
    T_end = 1e-6
    step_start = 0.2
    
    # Set up the current state
    cur = initial_h.copy()
    c_cur = c5_of(cur)
    
    best = cur.copy()
    c_best = c_cur
    
    # Main loop
    while time.time() - start < time_budget_s:
        t = time.time() - start
        elapsed_fraction = t / time_budget_s
        
        # Cooling schedule
        T = T_start * (T_end / T_start) ** elapsed_fraction
        step = step_start * (1 - elapsed_fraction)
        
        # Make a proposal
        h_proposed = propose_move(cur, step)
        c_proposed = c5_of(h_proposed)
        
        # Accept the proposal
        if c_proposed <= c_cur:
            cur = h_proposed
            c_cur = c_proposed
            if c_proposed < c_best:
                best = h_proposed.copy()
                c_best = c_proposed
        else:
            # Metropolis acceptance rule
            if random.random() < np.exp(-(c_proposed - c_cur) / T):
                cur = h_proposed
                c_cur = c_proposed
                
    return best, c_best

    """Main optimization function"""
    random.seed(seed)

    start = time.time()

    # Build initial seeds

            initial_h_values
        )
        # Renormalize to exact sum
        h_provided = h_provided * (N / 2) / np.sum(h_provided)
    else:
        h_provided = np.ones(N) * 0.5
    
    # 3. Double-peak seed
    h_double = np.full(N, 0.1)
    a = 0.3
    u = 0.9
    v = 0.1
    left = int(a * N / 2)
    right = int((2 - a) * N / 2)
    h_double[:left] = u
    h_double[right:] = u
    # Normalization
    h_double = h_double * (N / 2) / np.sum(h_double)
    
    # 4. Triangular seed
    h_tri = np.zeros(N)
    mid = N // 2
    for i in range(N):
        if i <= mid:
            h_tri[i] = 2 * i / mid
        else:
            h_tri[i] = 2 * (N - i) / (N - mid)
    h_tri = h_tri * (N / 2) / np.sum(h_tri)
    
    # Test all initial seeds
    seeds = [h_uniform, h_provided, h_double, h_tri]
    
    # Evaluate seeds with a quick heuristic search
    best_c5 = float('inf')
    best_h = None
    
    # Run each seed for 0.25 * 600 seconds = 150 seconds
    remaining_time = budget_s * 0.95  # Leave some margin for final polishing
    time_per_seed = min(remaining_time / len(seeds), 150)
    
    for i, h_seed in enumerate(seeds):
        if time.time() - start >= budget_s * 0.95:
            break

            best_c5 = c5_opt
            best_h = h_opt.copy()

            # Try a random perturbation
            h_trial = propose_move(h_refined, 0.01)
            h_trial = project_to_feasible(h_trial)
            c_trial = c5_of(h_trial)
            
            if c_trial < c_refined:
