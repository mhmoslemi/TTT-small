import numpy as np
import time
import random

def project_h(h, target_sum):
    """Project h onto box constraints [0,1] and sum constraint."""
    # First clip to [0, 1]
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if np.isclose(current_sum, target_sum):
        return h
    
    # Binary search for lambda
    tol = 1e-10
    low, high = -1.0, 1.0
    while high - low > tol:
        mid = (low + high) / 2
        h_test = np.clip(h - mid, 0, 1)
        test_sum = np.sum(h_test)
        if test_sum > target_sum:
            low = mid
        else:
            high = mid
    
    lambda_val = (low + high) / 2
    h_proj = np.clip(h - lambda_val, 0, 1)
    return h_proj

def compute_C_and_gradient(h, alpha):
    """Compute C5 and gradient for smoothed objective."""
    N = len(h)
    dx = 2.0 / N
    
    # Compute correlation using FFT for speed
    fft_h = np.fft.fft(h, n=2*N-1)
    fft_comp = np.fft.fft(1-h, n=2*N-1)
    corr = np.fft.ifft(fft_h * np.conj(fft_comp))
    C = np.real(corr[:2*N-1])  # Take only the first part
    C = 1 - dx * C
    
    # Compute softmax weights
    w = np.exp(alpha * C)
    w = w / np.sum(w)
    
    # Compute gradient
    # Direct computation via convolution of w with h
    w_rev = w[::-1]
    conv1 = np.convolve(h, w_rev, mode='same')
    conv2 = np.convolve(h, w, mode='same')
    
    grad = dx * (1 - (conv1 + conv2))
    
    # Return max C and gradient
    max_C = np.max(C)
    return max_C, grad
