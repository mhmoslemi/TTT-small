import numpy as np
import time
import random

def evaluate_c5(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    overlap = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(overlap) * dx
    return c5

def propose_move(h, max_step):
    """Propose a new feasible h vector by adjusting two elements"""
    h_new = h.copy()
    N = len(h)
    
    # Try to find a valid move
    max_tries = 100
    for _ in range(max_tries):
        i = random.randint(0, N - 1)
        j = random.randint(0, N - 1)
        if i == j:
            continue
            
        # Determine feasible interval for δ
        delta_low = max(-h_new[i], h_new[j] - 1)
        delta_high = min(1 - h_new[i], h_new[j])
        
        if delta_low >= delta_high:
            continue
            
        # Sample δ from uniform distribution over feasible range
        step = min(max_step, delta_high - delta_low)
        if step <= 0:
            continue
        delta = random.uniform(delta_low, delta_high)
        
        # Apply move
        h_new[i] += delta
        h_new[j] -= delta
        
        return h_new
    
    # If no valid move found, return original
    return h.copy()

def simulated_annealing(initial_h, time_budget_s):
    """Run simulated annealing with improved schedule"""
    start = time.time()
    
    T_start = 0.1
    T_end = 1e-8
    
    cur = initial_h.copy()
    c_cur = evaluate_c5(cur)
    
    best = cur.copy()
    c_best = c_cur
    
    # Cooling schedule with exponential decay
    iteration = 0
    while time.time() - start < time_budget_s:
        elapsed = time.time() - start
        elapsed_fraction = elapsed / time_budget_s
        
        # Cooling schedule (faster cooling than before)
        T = T_start * np.exp((np.log(T_end) - np.log(T_start)) * elapsed_fraction)
        
        # Step size decreases over time
        step = 0.1 * (1 - elapsed_fraction * 0.8)
        if step < 1e-6:
            step = 1e-6
            
        h_proposed = propose_move(cur, step)
        c_proposed = evaluate_c5(h_proposed)
        
        # Always accept if better, occasionally accept worse moves
        if c_proposed <= c_cur or np.random.random() < np.exp(-(c_proposed - c_cur) / T):
            cur = h_proposed
            c_cur = c_proposed
            
        if c_proposed < c_best:
            best = h_proposed.copy()
            c_best = c_proposed
            
        iteration += 1
    
    return best, c_best

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function"""
    np.random.seed(seed)
    random.seed(seed)
    return improved_search(seed, budget_s)
