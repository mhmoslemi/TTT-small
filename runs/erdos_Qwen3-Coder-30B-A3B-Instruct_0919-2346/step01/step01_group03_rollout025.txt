import numpy as np
import time
from scipy.optimize import minimize

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral via full convolution of h with (1-h)
    overlaps = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum of the full correlation vector
    # scaled by dx
    c5 = np.max(overlaps) * dx
    return c5

def project_to_feasible(h):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = 1"""
    h = np.clip(h, 0, 1)
    h = h * (2.0 / np.sum(h))
    return h

def evaluate_h(h, dx):
    """Compute C5 for given h"""
    overlaps = np.correlate(h, 1 - h, mode="full")
    return np.max(overlaps) * dx

    def objective(h_vec):

        h = h_vec.copy()
        h = project_to_feasible(h)
        return np.sum(h) - 1.0  # Constraint: sum = 1
    
    # Use Nelder-Mead with bounds
    bounds = [(0, 1) for _ in range(len(h))]

    max_iter = 1000

    # Simple gradient ascent
    step_size = 0.005
    gradient_iterations = 1000
    
    while time.time() - start < budget_s and iteration < gradient_iterations:
        h_new = h_best.copy()
        # Simple gradient estimation
        eps = 1e-4

            h_plus[i] += eps

        if c_new < c_best:
            h_best = h_new
            c_best = c_new
            
        iteration += 1
            
    return h_best, c_best
