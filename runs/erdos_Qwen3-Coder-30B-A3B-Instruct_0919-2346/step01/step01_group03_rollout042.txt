import numpy as np
import time
from scipy.optimize import differential_evolution, minimize
import math

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral via full convolution of h with (1-h)
    overlaps = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum of the full correlation vector
    # scaled by dx
    c5 = np.max(overlaps) * dx
    return c5

def project_to_feasible(h):
    """Project h to feasible region: 0 <= h[i] <= 1 and sum(h) * dx = 1"""
    h = np.clip(h, 0, 1)
    dx = 2.0 / len(h)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / h_sum) * (2.0 / len(h))
    return h

def solve_with_optimization(initial_h, budget_s):
    """Use scipy optimization solver to find better solution"""
    start = time.time()
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def objective(h):
        return c5_of(h)
    
    def constraint(h):
        return np.sum(h) * dx - 1.0
    
    # Define bounds: [0, 1] for each variable, equality constraint for integral
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Equality constraint: integral = 1
    eq_constraint = {'type': 'eq', 'fun': constraint}
    
    # Initial feasible point
    h_initial = project_to_feasible(initial_h)
    
    # Use differential evolution
    result = differential_evolution(
        objective, 
        bounds, 
        constraints=eq_constraint,
        maxiter=int(3000 * (budget_s / 600.0)),
        seed=42,
        atol=1e-6,
        x0=h_initial
    )
    
    return result.x, result.fun

def gradient_descent_step(h, step_size=0.01):
    """Perform one gradient descent step with projection."""
    n = len(h)
    dx = 2.0 / n
    
    # Compute the gradient numerically (simplified)
    # We'll compute it approximately using finite differences
    eps = 1e-4
    grad = np.zeros(n)
    
    for i in range(n):
        h_plus = h.copy()
        h_minus = h.copy()
        
        h_plus[i] = min(1.0, h[i] + eps)
        h_minus[i] = max(0.0, h[i] - eps)
        
        c_plus = c5_of(h_plus)
        c_minus = c5_of(h_minus)
        
        grad[i] = (c_plus - c_minus) / (2 * eps)
    
    # Update with gradient descent
    h_new = h - step_size * grad
    
    # Apply bounds and constraint
    h_new = np.clip(h_new, 0, 1)
    h_new_sum = np.sum(h_new)
    if h_new_sum > 0:
        h_new = h_new * (1.0 / h_new_sum) * (2.0 / n)
    
    return h_new

def improve_solution(h, budget_s):
    """Improve solution using gradient-based methods"""
    start = time.time()
    best_h = h.copy()
    best_c5 = c5_of(h)
    
    # Try simple gradient descent steps
    for i in range(int(5000 * (budget_s / 600.0))):
        if time.time() - start > budget_s:
            break
        
        h_new = gradient_descent_step(best_h, step_size=0.005)
        c_new = c5_of(h_new)
        
        if c_new < best_c5:
            best_h = h_new
            best_c5 = c_new
    
    return best_h, best_c5

    return h
