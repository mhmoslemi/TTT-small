import numpy as np
import time
import random
from scipy.optimize import differential_evolution

def c5_of(h):
    """Compute C5 = max_k integral h(x)(1-h(x+k)) dx."""
    dx = 2.0 / len(h)
    # Compute correlation: h * (1 - h) with full mode
    c = np.correlate(h, 1 - h, mode="full")
    # Multiply by dx to get continuous integral
    c *= dx
    # Return maximum value among all shifts
    return np.max(c)

def evaluate_h(h):
    """Evaluate the C5 bound for given h."""
    return c5_of(h)

def project_to_feasible(h):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    dx = 2.0 / len(h)
    h = np.clip(h, 0, 1)
    # Renormalize
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / dx) / h_sum
    return h

def local_search(h, max_iter=100):
    """Perform local search around current solution."""
    best_h = h.copy()
    best_c5 = c5_of(best_h)
    
    for _ in range(max_iter):
        # Try small perturbations
        idx = random.randint(0, len(h) - 1)
        delta = random.uniform(-0.05, 0.05)
        
        new_h = best_h.copy()
        new_h[idx] += delta
        
        # Project to feasibility
        new_h = project_to_feasible(new_h)
        
        new_c5 = c5_of(new_h)
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5
    
    return best_h, best_c5
