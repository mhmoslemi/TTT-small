import numpy as np
import time
import cvxpy as cp

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using full convolution
    overlap = np.correlate(h, 1 - h, mode='full')
    # The maximum value scaled by dx
    c5 = np.max(overlap) * dx
    return c5

        # For each lag ell from 1 to N-1
        for ell in range(1, N):

            if length <= 0:
                continue
            # Sum of h_i in overlapping region for shift ell
            H = cp.sum(x[:length])
            # Introduce y variables for the products x_i * x_{i+ell}
            y = cp.Variable(length, boolean=True)
            # Add constraints for y = x_i * x_{i+ell} 
            constraints += [y <= x[:length]]
            constraints += [y <= x[ell:]]
            constraints += [y >= x[:length] + x[ell:] - 1]

            c5 = c5_of(h)
            return (h, c5, N)

        h_new[i] += delta
        h_new[j] -= delta

            c_prop = c5_of(h_prop)
            
            if c_prop < c_best:
                best = h_prop.copy()
                c_best = c_prop

                c_cur = c_prop
