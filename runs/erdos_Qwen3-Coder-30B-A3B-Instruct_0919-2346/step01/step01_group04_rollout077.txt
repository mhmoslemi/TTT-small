import numpy as np
import time
import random
from scipy.optimize import differential_evolution, minimize


def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    overlap = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(overlap) * dx
    return c5


def project feasibility(h):
    """Project h onto feasible region: [0,1] and integral = 1"""
    # Clamp to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to make integral = 1
    integral = np.sum(h) * 2.0 / len(h)
    if integral > 0:
        h = h / integral * 1.0
    return h


def propose_move(h, max_step):
    """Propose a new feasible h vector by adjusting two elements"""
    h_new = h.copy()
    N = len(h)
    
    # Try to find a valid move
    max_tries = 100
    for _ in range(max_tries):
        i = random.randint(0, N - 1)
        j = random.randint(0, N - 1)
        if i == j:
            continue
            
        # Determine feasible interval for δ
        delta_low = max(-h_new[i], h_new[j] - 1)
        delta_high = min(1 - h_new[i], h_new[j])
        
        if delta_low >= delta_high:
            continue
            
        # Sample δ from uniform distribution over feasible range
        step = min(max_step, delta_high - delta_low)
        if step <= 0:
            continue
        delta = random.uniform(delta_low, delta_high)
        
        # Apply move
        h_new[i] += delta
        h_new[j] -= delta
        
        return h_new
    
    # If no valid move found, return original
    return h.copy()


def flip_move(h, max_step):

    h_new = h.copy()
    N = len(h)
    max_tries = 100
    
    for _ in range(max_tries):
        # Pick a section to flip
        i = random.randint(0, N - 1)
        j = random.randint(i, N - 1)

            continue
        # Randomly choose new value to set in the region
        new_val = random.uniform(0, 1)
        # Apply the flip
        h_new[i:j+1] = new_val

        h_new = project_feasibility(h_new)
        return h_new
    
    return h.copy()


def anneal_optimize(initial_h, time_budget_s):
    """Use simulated annealing to optimize starting with initial_h"""
    start = time.time()
    N = len(initial_h)
    
    # Parameters
    T_start = 0.05
    T_end = 1e-6
    step_start = 0.2
    
    # Set up the current state
    cur = initial_h.copy()
    c_cur = c5_of(cur)
    
    best = cur.copy()
    c_best = c_cur
    
    # Main loop
    while time.time() - start < time_budget_s:
        t = time.time() - start
        elapsed_fraction = t / time_budget_s
        
        # Cooling schedule
        T = T_start * (T_end / T_start) ** elapsed_fraction
        step = step_start * (1 - elapsed_fraction)
        
        # Make a proposal - alternate between different moves
        if random.random() < 0.7:
            h_proposed = propose_move(cur, step)
        else:
            h_proposed = flip_move(cur, step)
            
        c_proposed = c5_of(h_proposed)
        
        if c_proposed <= c_cur:
            cur = h_proposed
            c_cur = c_proposed
            if c_proposed < c_best:
                best = h_proposed.copy()
                c_best = c_proposed
        else:
            # Metropolis acceptance rule
            if random.random() < np.exp(-(c_proposed - c_cur) / T):
                cur = h_proposed
                c_cur = c_proposed
                
    return best, c_best
