import numpy as np
import time

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral via full convolution of h with (1-h)
    overlaps = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum of the full correlation vector
    # scaled by dx
    c5 = np.max(overlaps) * dx
    return c5

def full_overlap(h):
    """Return the raw correlation vector (unscaled)"""
    return np.correlate(h, 1 - h, mode="full")

def project_onto_P(v, total):
    """Project v onto {x: 0 <= x_i <= 1, sum x_i = total}"""
    def f(lam):
        x = np.clip(v - lam, 0, 1)
        return np.sum(x) - total
    
    a, b = -np.max(v), 1 - np.min(v)
    while f(a) * f(b) > 0:
        a -= 1
        b += 1
    if np.abs(f(a)) < 1e-12:
        return np.clip(v - a, 0, 1)
    if np.abs(f(b)) < 1e-12:
        return np.clip(v - b, 0, 1)
        
    for _ in range(100):
        mid = (a + b) / 2
        if f(mid) > 0:
            b = mid
        else:
            a = mid
    return np.clip(v - a, 0, 1)

def smooth_gradient(h, beta):
    """Compute gradient of Φ_β(h) using weighted shifted slices"""
    N = len(h)
    F = full_overlap(h)  # For numerical stability, subtract max
    F_max = np.max(F)
    exp_F = np.exp(beta * (F - F_max))
    w = exp_F / np.sum(exp_F)
    
    # Compute the gradient ∇Φ_β(h)
    g = np.ones(N)
    for k in range(-N+1, N):
        wk = w[k + N - 1]  # Adjust index for correlation to array indexing
        if k >= 0:
            g[:N-k] -= wk * h[k:]
            g[k:] -= wk * h[:N-k]
        else:
            d = -k
            g[d:] -= wk * h[:N-d]
            g[:N-d] -= wk * h[d:]
    return g

def projected_gradient_descent(h0, time_budget, seed):
    """Run the projected gradient descent方案."""
    np.random.seed(seed)
    start = time.time()
    h = h0.copy()
    best_h = h.copy()
    c_best = c5_of(h)
    beta = 2.0
    beta_max = 30.0
    step = 0.5
    gamma_step = 0.9995
    gamma_beta = 1.5
    beta_update_interval = 50
    max_iter = 100000
    
    for i in range(max_iter):
        if time.time() - start > time_budget:
            break
            
        # Compute overlap and weights
        F = full_overlap(h)
        F_max = np.max(F)
        exp_F = np.exp(beta * (F - F_max))
        w = exp_F / np.sum(exp_F)

        # Compute gradient
        g = smooth_gradient(h, beta)
        
        # Tentative update
        v = h - step * g
        
        # Project to feasible set
        h = project_onto_P(v, len(h) / 2)
        
        # Evaluate true C5
        c_cur = c5_of(h)
        if c_cur < c_best:
            best_h = h.copy()
            c_best = c_cur
            
        # Update parameters
        step *= gamma_step
        if i % beta_update_interval == 0:
            beta = min(beta * gamma_beta, beta_max)
            
    return best_h, c_best

def targeted_refinement(h, time_budget):
    """Perform deterministic worst-pair refinement."""
    np.random.seed(0)  # Use fixed seed for reproducibility
    start = time.time()
    h = h.copy()
    c_best = c5_of(h)
    alpha = 1.0
    max_attempts = 200
    attempts = 0

    while attempts < max_attempts and time.time() - start < time_budget:
        C = full_overlap(h)
        d = np.argmax(C) - (len(h) - 1)  # The shift corresponding to max
        # Determine indices for the pair that contributes most
        N = len(h)
        if d >= 0:
            hi = h[:N-d]
            hj = h[d:]
        else:
            hi = h[-d:]
            hj = h[:N-(-d)]
        
        # Find the position where hi[i] * (1-hj[i]) is largest
        c_vals = hi * (1 - hj)
        i_star = np.argmax(c_vals)
        j_star = i_star if d >= 0 else i_star + d
        
        # Compute feasible transfer
        c = c_vals[i_star]
        delta = min(alpha * c, h[i_star], 1 - hj[i_star])
        
        # Perform transfer
        h[i_star] -= delta
        if d >= 0:
            h[j_star] += delta
        else:
            h[j_star + (-d)] += delta
            
        c_new = c5_of(h)
        if c_new < c_best:
            c_best = c_new
            attempts = 0
        else:
            h[i_star] += delta
            if d >= 0:
                h[j_star] -= delta
            else:
                h[j_star + (-d)] -= delta
            attempts += 1
            alpha *= 0.5
            if attempts > 100 and alpha < 1e-8:
                break

    return h, c_best

    else:

        h_initial *= (N / 2) / np.sum(h_initial)
    
    h_double_peak = np.full(N, 0.1)
    a = 0.3
    idxs = np.where((np.linspace(0, 2, N) < a) | (np.linspace(0, 2, N) > (2 - a)))[0]
    h_double_peak[idxs] = 0.9
    h_double_peak *= (N / 2) / np.sum(h_double_peak)
    
    # Random Dirichlet (sample from Beta(0.5, 0.5))
    h_dirichlet = np.random.beta(0.5, 0.5, N)
    h_dirichlet *= (N / 2) / np.sum(h_dirichlet)
    
    # Run projected gradient descent on all seeds
    best_h = None
    best_c5 = float('inf')
    time_per_seed = 0.3 * budget_s
    for i, h_seed in enumerate([h_uniform, h_initial, h_double_peak, h_dirichlet]):
        if time.time() - start > budget_s * 0.95:
            break
        h, c5 = projected_gradient_descent(h_seed, time_per_seed, seed + i)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5
    
    # Refinement
    if best_h is not None and time.time() - start < budget_s * 0.95:
        refunding_time = budget_s - (time.time() - start)
        h_after_refinement, c_after_refinement = targeted_refinement(best_h, refunding_time)
        if c_after_refinement < best_c5:
            best_h = h_after_refinement
            best_c5 = c_after_refinement
