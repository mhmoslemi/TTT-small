import numpy as np
import time
import random


def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    overlap = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(overlap) * dx
    return c5


def project_to_box_simplex(v, target):
    """Project onto set {x: 0<=x<=1, sum x = target}"""
    # Use bisection on lambda
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        w = np.clip(v - lam, 0, 1)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo+hi)/2.0, 0, 1)


def gradient_smooth_max(h, beta):
    """Compute gradient of smooth approximation to max correlation"""
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, h, mode='full') * dx
    # Use numerically stable softmax
    max_g = np.max(-beta * corr)
    exp_vals = np.exp(-beta * corr - max_g)
    w = exp_vals / np.sum(exp_vals)
    grad = np.zeros_like(h)
    # for each lag
    for lag in range(-(N-1), N):
        idx = lag + N - 1
        wk = w[idx]
        if wk == 0:
            continue
        if lag >= 0:
            grad[:N-lag] += wk * h[lag:]
            grad[lag:] += wk * h[:N-lag]
        else:
            k = -lag
            grad[k:] += wk * h[:N-k]
            grad[:N-k] += wk * h[k:]
    grad *= -dx
    return grad


def subgradient_opt(h, time_budget, step_start=0.5, step_end=1e-4):
    """Run subgradient descent on worst lag"""
    target = len(h)/2
    best_h = h.copy()
    best_c5 = c5_of(h)
    start = time.time()
    iter = 0
    while time.time() - start < time_budget:
        elapsed_frac = (time.time() - start) / time_budget
        step = step_start * (1 - elapsed_frac) + step_end * elapsed_frac
        # find worst lag
        N = len(h)
        dx = 2.0 / N
        corr = np.correlate(h, h, mode='full') * dx
        worst_idx = np.argmin(corr)
        lag = worst_idx - (N-1)
        grad = np.zeros_like(h)
        if lag >= 0:
            grad[:N-lag] += h[lag:]
            grad[lag:] += h[:N-lag]
        else:
            k = -lag
            grad[k:] += h[:N-k]
            grad[:N-k] += h[k:]
        grad *= dx
        h_new = h + step * grad
        h_new = project_to_box_simplex(h_new, target)
        c5 = c5_of(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
        iter += 1
    return best_h, best_c5


def smooth_optimize(seed, time_budget, beta_start=1.0, beta_end=60.0, step_start=1.0):
    """Run smoothed gradient descent"""
    h = seed.copy()
    N = len(h)
    target = N/2
    best_h = h.copy()
    best_c5 = c5_of(h)
    start = time.time()
    iter = 0
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        t_frac = elapsed / time_budget
        beta = beta_start * (beta_end / beta_start) ** t_frac
        step = step_start * (1 - t_frac) + 1e-8
        grad = gradient_smooth_max(h, beta)
        h_new = h - step * grad
        h_new = project_to_box_simplex(h_new, target)
        c5 = c5_of(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
        iter += 1
    return best_h, best_c5


def propose_move(h, max_step):
    """Propose a new feasible h vector by adjusting two elements"""
    h_new = h.copy()
    N = len(h)
    
    # Try to find a valid move
    max_tries = 100
    for _ in range(max_tries):
        i = random.randint(0, N - 1)
        j = random.randint(0, N - 1)
        if i == j:
            continue
            
        # Determine feasible interval for δ
        delta_low = max(-h_new[i], h_new[j] - 1)
        delta_high = min(1 - h_new[i], h_new[j])
        
        if delta_low >= delta_high:
            continue
            
        # Sample δ from uniform distribution over feasible range
        step = min(max_step, delta_high - delta_low)
        if step <= 0:
            continue
        delta = random.uniform(delta_low, delta_high)
        
        # Apply move
        h_new[i] += delta
        h_new[j] -= delta
        
        return h_new
    
    # If no valid move found, return original
    return h.copy()


def hill_climbing(h, time_budget):
    """Local search using pairwise moves"""
    best_h = h.copy()
    best_c5 = c5_of(h)
    start = time.time()
    while time.time() - start < time_budget:
        h_new = propose_move(h, max_step=0.2)
        c5_new = c5_of(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
    return best_h, best_c5

    # 3. Random continuous
    rand_cont = np.random.rand(N)

    # 4. Random binary
    binary = np.zeros(N)
    half = int(round(target))
    idx = np.random.choice(N, half, replace=False)
    binary[idx] = 1.0
    binary = project_to_box_simplex(binary, target)
    
    # Test all initial seeds
    seeds = [h_uniform, h_provided, rand_cont, binary]
    best_h = None
    best_c5 = float('inf')

    time_for_seeds = budget_s * 0.7
    per_seed_budget = time_for_seeds / len(seeds)
    
    for i, h_seed in enumerate(seeds):
        if time.time() - start >= budget_s * 0.9:
            break
        # Phase 1: smoothing gradient descent
        h_smooth, c5_smooth = smooth_optimize(h_seed, per_seed_budget * 0.3, beta_start=0.5, beta_end=50.0, step_start=1.0)
        # Phase 2: subgradient descent
        h_final, c5_final = subgradient_opt(h_smooth, per_seed_budget * 0.7, step_start=0.5, step_end=1e-4)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()
    
    # Final polishing phase
    time_remaining = budget_s - (time.time() - start)
