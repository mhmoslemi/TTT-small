import numpy as np
import time
import random
from scipy.optimize import differential_evolution, minimize

def c5_of(h):
    """Compute C5 = max_k integral h(x)(1-h(x+k)) dx."""
    dx = 2.0 / len(h)
    # Compute correlation: h * (1 - h) with full mode
    c = np.correlate(h, 1 - h, mode="full")
    # Multiply by dx to get continuous integral
    c *= dx
    # Return maximum value among all shifts
    return np.max(c)

def project_to_constraint(h):
    """Project h to satisfy integral constraint exactly."""
    dx = 2.0 / len(h)
    current_sum = np.sum(h) * dx
    scale = 1.0 / current_sum
    h = h * scale
    return h

def constraint_violation(h):
    """Compute absolute violation of integral constraint."""
    dx = 2.0 / len(h)
    current_sum = np.sum(h) * dx
    return abs(current_sum - 1.0)

def propose_move(h, max_step):
    """Generate a new feasible h by adjusting entries."""
    h = h.copy()
    N = len(h)
    i = random.randint(0, N-1)
    j = random.randint(0, N-1)
    
    # Determine feasible interval for delta
    delta_low = max(-h[i], 1.0 - h[j])
    delta_high = min(1.0 - h[i], h[j])
    
    if delta_low >= delta_high:
        return h  # No valid move for this pair
    
    # Sample delta within allowed range
    step = min(max_step, delta_high - delta_low)
    delta = random.uniform(-step, step)
    delta = max(delta_low, min(delta, delta_high))
    
    h[i] += delta
    h[j] -= delta
    
    return h

def local_search(h0, max_iter=10000, step_size=0.01):
    """Simple local search starting from h0."""
    h = h0.copy()
    h = project_to_constraint(h)
    c_best = c5_of(h)
    best_h = h.copy()
    
    for _ in range(max_iter):
        h_new = propose_move(h, step_size)
        c_new = c5_of(h_new)
        if c_new < c_best:
            h = h_new
            c_best = c_new
            best_h = h.copy()
    
    return best_h, c_best
