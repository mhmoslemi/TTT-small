import numpy as np
import time
import random

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using full convolution
    overlap = np.correlate(h, 1 - h, mode='full')
    # The maximum value scaled by dx
    c5 = np.max(overlap) * dx
    return c5

def propose_move(h, max_step):
    """Propose a move that maintains sum constraint exactly"""
    h_new = h.copy()
    N = len(h)
    while True:
        i, j = random.sample(range(N), 2)
        # Compute feasible range for delta
        delta_low = max(-h_new[i], h_new[j] - 1.0)
        delta_high = min(1.0 - h_new[i], h_new[j])
        if delta_low >= delta_high:
            continue  # No valid move, try again
        break
    
    # Sample delta from feasible interval
    step = min(max_step, delta_high - delta_low)
    if step <= 0:
        delta = 0.0
    else:
        delta = random.uniform(max(-step, delta_low), min(step, delta_high))
        
    h_new[i] += delta
    h_new[j] -= delta
    return h_new

def propose_move_gradient(h, max_step):
    """Propose a move using gradient direction"""
    h_new = h.copy()
    N = len(h)
    
    # Compute gradient from finite differences
    grad = np.zeros(N)
    eps = 1e-4
    for i in range(N):
        h_plus = h.copy()
        h_minus = h.copy()
        h_plus[i] += eps
        h_minus[i] -= eps
        # Ensure constraints maintained
        if h_plus[i] > 1: h_plus[i] = 1
        if h_minus[i] < 0: h_minus[i] = 0
        
        # Get neighbors that sum to same value
        # Otherwise, need to be more careful about projection
        c_plus = c5_of(h_plus)
        c_minus = c5_of(h_minus)
        grad[i] = (c_plus - c_minus) / (2 * eps)
    
    # Find max gradient
    max_grad_idx = np.argmax(np.abs(grad))
    direction = -np.sign(grad[max_grad_idx])
    
    # Limit the move
    delta = direction * min(abs(grad[max_grad_idx]) * 0.1, max_step)
    
    # Adjust only one index
    i = max_grad_idx
    delta = max(-h[i], min(1-h[i], delta))
    h_new[i] += delta
    
    return h_new

def anneal(initial_h, time_budget, use_gradient=False):
    """Simulated annealing optimization with optional gradient moves"""
    start = time.time()
    cur = initial_h.copy()
    best = initial_h.copy()
    c_cur = c5_of(cur)
    c_best = c_cur
    
    # Annealing parameters
    T_start = 0.05
    T_end = 1e-6
    step_start = 0.2
    
    iteration = 0
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T_start * (T_end / T_start) ** (elapsed / time_budget)
        step = step_start * (1.0 - elapsed / time_budget)
        
        if use_gradient and iteration % 3 == 0:
            h_prop = propose_move_gradient(cur, step)
        else:
            h_prop = propose_move(cur, step)
            
        c_prop = c5_of(h_prop)
        
        if c_prop < c_best:
            best = h_prop.copy()
            c_best = c_prop
            
        if c_prop <= c_cur or random.random() < np.exp(-(c_prop - c_cur) / T):
            cur = h_prop
            c_cur = c_prop
            
        iteration += 1
            
    return best, c_best
