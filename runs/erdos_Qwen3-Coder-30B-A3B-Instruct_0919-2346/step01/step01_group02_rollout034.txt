def c5_of(h):
    """Compute the maximum overlap integral for the given step function h."""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    c = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(c) * dx
    return c5

def project_to_feasible(h, N):
    """Euclidean projection onto the intersection of box [0,1] and sum = N/2."""
    # Use water-filling to find lambda
    v = h.copy()
    target_sum = N / 2.0
    
    def F(lam):
        # Σ_i clip(v_i + λ , 0 , 1 )
        clipped = np.clip(v + lam, 0, 1)
        return np.sum(clipped)
    
    # Binary search for lambda
    low, high = -2.0, 2.0
    for _ in range(50):
        mid = (low + high) / 2.0
        f_val = F(mid)
        if f_val > target_sum:
            high = mid
        else:
            low = mid
    
    lambda_val = (low + high) / 2.0
    proj_h = np.clip(v + lambda_val, 0, 1)
    
    # Small redistribution to fix sum exactly
    diff = target_sum - np.sum(proj_h)
    if abs(diff) > 1e-10:
        # Find indices that can be adjusted
        free_indices = np.where((proj_h > 0) & (proj_h < 1))[0]
        if len(free_indices) > 0:
            idx = free_indices[0]
            proj_h[idx] += diff
    
    return proj_h

def compute_smooth_objective(h, eps):
    """Compute F_epsilon(h) = epsilon * log(sum(exp(C_s/epsilon)))"""
    dx = 2.0 / len(h)
    # Compute all correlations
    c = np.correlate(h, 1 - h, mode='full')
    c_scaled = c * dx
    
    # Soft max computation (log-sum-exp trick for numerical stability)
    max_c = np.max(c_scaled)
    if np.isinf(max_c):
        max_c = 0
    exp_terms = np.exp((c_scaled - max_c) / eps)
    sum_exp = np.sum(exp_terms)
    
    # Return smooth objective value
    F_eps = eps * (np.log(sum_exp) + max_c)
    return F_eps

def continuation_method(h0, N, max_iter=150, eps_list=None):
    h = h0.copy()
    # Project to feasible set
    h = project_to_feasible(h, N)
    if eps_list is None:
        eps_list = [0.10, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    
    for eps in eps_list:
        def f(x):
            return compute_smooth_objective(x, eps)
        
        def grad(x):
            return compute_smooth_gradient(x, eps)
        
        # L-BFGS-B optimization
        bounds = [(0, 1) for _ in range(N)]
        res = minimize(f, h, method='L-BFGS-B', jac=grad, bounds=bounds,
                       options={'maxiter': max_iter})
        h = res.x
        h = project_to_feasible(h, N)
    
    return h

def generate_seeds(N):
    seeds = []
    
    # Uniform seed
    seeds.append(np.ones(N) * 0.5)
    
    # Two-peak seed
    h = np.ones(N)
    x_vals = np.linspace(0, 2, N)
    a = 0.3
    u = 0.9
    v = 0.1
    mask = (x_vals < a) | (x_vals >= 2 - a)
    h[mask] = u
    h[~mask] = v
    h = h * (N/2) / np.sum(h)
    seeds.append(h)
    
    # Binary seeds
    rand_binary = np.random.choice(N, size=N//2, replace=False)
    h = np.zeros(N)
    h[rand_binary] = 1.0
    h = h * (N/2) / np.sum(h)
    seeds.append(h)
    
    # Another binary
    big_binary = np.ones(N//2) * 1
    small_binary = np.zeros(N - N//2)
    h = np.concatenate([big_binary, small_binary])
    h = h * (N/2) / np.sum(h)
    seeds.append(h)
    
    # Random continuous seed
    h = np.random.uniform(0, 1, N)
    h = h * (N/2) / np.sum(h)
    seeds.append(h)
    
    return seeds

def subgradient_refinement(h, max_time=10):
    start = time.time()
    dx = 2.0 / len(h)
    while time.time() - start < max_time:
        # Compute all correlations
        c = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(c)
        max_shift = max_idx - (len(h) - 1)
        
        # Compute direction by direct subgradient
        # For straightforward gradient logic with shift model:
        g = np.zeros_like(h)  
        # Operations for C_s = dx * sum_i h[i] *(1 - h[i+s])
        # If s is shift index and zero pad ensures no overflow:
        if max_shift >= 0:  # shift right: now get h no longer i + shift but - d  
            shifted = np.concatenate([[0]*max_shift, h[:-max_shift]])
        else:  # shift left
            shifted = np.concatenate([h[-max_shift:], [0]*(-max_shift)])
        
        # Value derivative 
        g = dx * (1 - shifted)  # simplify
    
        # Apply gradient update
        step = 0.05  # adjust manually
        h_candidate = h - step * g
        h_candidate = project_to_feasible(h_candidate, len(h))
    
        c_candidate = np.max(np.correlate(h_candidate, 1-h_candidate, mode='full')) * dx
        c_current = np.max(np.correlate(h, 1-h, mode='full')) * dx
        if c_candidate < c_current:
            h = h_candidate
        else:
            break  # No progress, stop
    return h

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start = time.time()
    
    N = 140
    dx = 2.0 / N
    
    # Time allocation
    seed_time = 0.05 * budget_s
    inner_time = 0.70 * budget_s
    multi_start_time = 0.10 * budget_s
    polish_time = 0.10 * budget_s
    safety_time = 0.05 * budget_s
    
    # Get seeds and initial evaluation
    seeds = generate_seeds(N)
    best = None
    best_c5 = float('inf')
    
    for s, seed in enumerate(seeds):
        seed_start = time.time()
        h = seed.copy()
        h = project_to_feasible(h, N)
        
        h_final = continuation_method(h, N, max_iter=150)
        c5_final = c5_of(h_final)
        
        if c5_final < best_c5:
            best = h_final.copy()
            best_c5 = c5_final
            
        # Break early if time is up
        if time.time() - start > budget_s * 0.95:
            break
    
    # Final polishing
    if best is not None and (budget_s - (time.time() - start)) > polish_time:
        polish_start = time.time()
        best = subgradient_refinement(best, max_time=polish_time) 
        best_c5 = c5_of(best)
        
    return (best, best_c5, N)
