import numpy as np
import time
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral via full convolution of h with (1-h)
    overlaps = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum of the full correlation vector
    # scaled by dx
    c5 = np.max(overlaps) * dx
    return c5

def initial_h():
    """Create an initial piecewise constant function."""
    N = 140
    h = np.ones(N) * 0.5
    # Create a solution with more concentrated regions
    # We'll keep two peaks at the ends, but make them sharper
    x = np.linspace(0, 2, N)
    # Set high values at bounds
    for i in range(N):
        if x[i] < 0.3 or x[i] > 1.7:
            h[i] = 0.9
        elif 0.7 <= x[i] <= 1.3:
            h[i] = 0.1
        else:
            h[i] = 0.5
    
    # Normalize
    h *= (N / 2) / np.sum(h)
    return h

def project_to_feasible(h):
    """Project h onto the feasible set: h[i] in [0,1] and sum(h)*dx = 1"""
    h = np.clip(h, 0, 1)
    dx = 2.0 / len(h)
    h *= (2.0 / len(h)) / np.sum(h)
    return h

def gradient_descent_step(h, lr=1e-3):
    """Simple gradient descent step using finite differences to estimate gradient."""
    # We'll sample the gradient of C5 with respect to each h[i] using finite differences
    eps = 1e-5
    grad = np.zeros_like(h)
    c5_base = c5_of(h)
    
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus)
        c5_plus = c5_of(h_plus)
        grad[i] = (c5_plus - c5_base) / eps
    
    # Update step
    h_new = h - lr * grad
    h_new = project_to_feasible(h_new)
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start = time.time()
    max_iter = 100000
    
    # Initialize
    h = initial_h()
    c5 = c5_of(h)
    best_h = h.copy()
    best_c5 = c5
    last_improvement = 0
    patience = 1000  # If no improvement for many iterations, reduce learning rate
    
    # Just a quick naive gradient descent approach
    lr = 1e-2
    for i in range(max_iter):
        if time.time() - start > budget_s * 0.95:
            break
            
        # Simple gradient descent
        h_new = gradient_descent_step(h, lr)
        c5_new = c5_of(h_new)
        
        if c5_new < c5:
            h = h_new
            c5 = c5_new
            if c5_new < best_c5:
                best_h = h.copy()
                best_c5 = c5_new
                last_improvement = i
            
        # Adaptive learning rate
        if i - last_improvement > patience:
            lr *= 0.9
            last_improvement = i
    
    # Final polishing with local search
    for _ in range(5000):
        if time.time() - start > budget_s * 0.98:
            break
        idx = np.random.randint(len(h))
        # Try small perturbations around a random index
        step = np.random.uniform(-0.1, 0.1)
        h_new = h.copy()
        h_new[idx] = np.clip(h_new[idx] + step, 0, 1)
        h_new = project_to_feasible(h_new)
        c5_new = c5_of(h_new)
        if c5_new < c5:
            h = h_new
            c5 = c5_new
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
    
    return (best_h, best_c5, len(best_h))
