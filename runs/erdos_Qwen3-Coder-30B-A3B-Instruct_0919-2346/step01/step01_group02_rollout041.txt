import numpy as np
import time

def c5_of(h):
    """Compute the maximum overlap integral for the given step function h."""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    c = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(c) * dx
    return c5

def compute_gradient(h, shift):
    """Compute gradient of C5 with respect to h for worst shift."""
    N = len(h)
    dx = 2.0 / N
    g = np.full(N, 1.0) * dx
    if shift >= 0:
        if shift < N:
            # subtract right shift
            g[:N-shift] -= h[shift:] * dx
            # subtract left shift
            g[shift:] -= h[:N-shift] * dx
    else:
        shift = -shift
        if shift < N:
            g[shift:] -= h[:N-shift] * dx
            g[:N-shift] -= h[shift:] * dx
    return g

def project_onto_bounds_and_sum(v, target):
    """Project vector v onto box [0,1] and sum = target constraints."""
    N = len(v)
    lower = -np.max(v)
    upper = 1 - np.min(v)
    # Binary search for lambda
    for _ in range(60):
        mid = (lower + upper) / 2.0
        w = np.clip(v + mid, 0, 1)
        s = np.sum(w)
        if s < target:
            lower = mid
        else:
            upper = mid
    w = np.clip(v + (lower + upper) / 2.0, 0, 1)
    diff = target - np.sum(w)
    if abs(diff) > 1e-12:
        free_idx = (w > 0) & (w < 1)
        if np.any(free_idx):
            w[free_idx] += diff / np.sum(free_idx)
            w = np.clip(w, 0, 1)
        else:
            w += diff / len(w)
            w = np.clip(w, 0, 1)
    return w

def run(seed=42, budget_s=600):
    np.random.seed(seed)

    dx = 2.0 / N
    
    # Initialize best h from initial_h_values if available

                      initial_h_values)
        h0 = h0 * (N/2) / np.sum(h0)  # Renormalize exactly
    else:
        h0 = np.ones(N) * 0.5  # fallback
    
    # Try two-peak shape
    h1 = np.ones(N)
    x_vals = np.linspace(0, 2, N)
    a = 0.3
    u = 0.9
    v = 0.1
    mask = (x_vals < a) | (x_vals >= 2 - a)
    h1[mask] = u
    h1[~mask] = v
    h1 = h1 * (N/2) / np.sum(h1)  # Renormalize
    
    # Uniform
    h2 = np.ones(N) * 0.5
    
    # Initialize
    best_h = None
    best_c5 = float('inf')
    seeds = [h0, h1, h2]
    
    # Evaluate seeds
    for h_seed in seeds:
        c5 = c5_of(h_seed)
        if c5 < best_c5:
            best_h = h_seed.copy()
            best_c5 = c5
    
    # Gradient descent with adaptive step size
    start = time.time()
    
    # Run optimization for most of the time
    while time.time() - start < 0.9 * budget_s:
        cur_h = best_h.copy()
        cur_score = best_c5
        
        # Gradient descent iteration
        inner_start = time.time()
        iteration = 0
        max_iter = 500
            
        while time.time() - start < 0.9 * budget_s and iteration < max_iter:
            # Compute correlation
            corr = np.correlate(cur_h, 1 - cur_h, mode='full')
            max_idx = np.argmax(corr)
            s = max_idx - (N-1)
            
            # Compute gradient
            grad = compute_gradient(cur_h, s)
            
            # Adaptive step size
            t_frac = (time.time() - start) / (0.9 * budget_s)
            step = 15.0 * (1 - t_frac) + 0.1
            
            # Update
            v = cur_h - step * grad
            cur_h = project_onto_bounds_and_sum(v, N/2)
            
            # Evaluate
            new_score = c5_of(cur_h)
            if new_score < cur_score:
                cur_score = new_score
                if new_score < best_c5:
                    best_c5 = new_score
                    best_h = cur_h.copy()
            
            iteration += 1
        
        # Occasionally restart with a random vector
        if np.random.rand() < 0.05:
            # Generate random vector with sum=N/2
            cur_h = np.random.rand(N)
            cur_h = cur_h * (N/2) / np.sum(cur_h)
            cur_score = c5_of(cur_h)
            if cur_score < best_c5:
                best_c5 = cur_score
                best_h = cur_h.copy()
    
    # Final polishing
    start = time.time()
    for i in range(500):
        if time.time() - start > 0.1 * budget_s:
            break
        
        # Small random jitter
        h_new = best_h.copy()
        idx1, idx2 = np.random.choice(N, size=2, replace=False)
        delta = np.random.uniform(-0.02, 0.02)
        if (h_new[idx1] + delta >= 0 and h_new[idx1] + delta <= 1 and
            h_new[idx2] - delta >= 0 and h_new[idx2] - delta <= 1):
            h_new[idx1] += delta
            h_new[idx2] -= delta
            # Rescale to match sum
            h_new *= (N/2) / np.sum(h_new)
            new_c5 = c5_of(h_new)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
    
    return (best_h, best_c5, N)
