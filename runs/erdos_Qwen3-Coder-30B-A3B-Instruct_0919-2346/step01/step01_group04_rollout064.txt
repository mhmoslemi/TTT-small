import numpy as np
import time
import random
from scipy.optimize import differential_evolution, minimize


def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    overlap = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(overlap) * dx
    return c5


def project_to_feasible(h):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) * dx = 1"""
    dx = 2.0 / len(h)
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1 / h_sum) * (1 / dx)
    return np.clip(h, 0, 1)


def local_search_step(h, step_size=0.01):
    """Make a small perturbation to the function h"""
    h_new = h.copy()
    N = len(h)
    
    # Perturb a few points
    num_changes = max(1, N // 20)
    indices = random.sample(range(N), num_changes)
    
    for i in indices:
        delta = random.uniform(-step_size, step_size)
        h_new[i] += delta
        
    # Project back to feasible set
    h_new = project_to_feasible(h_new)
    return h_new


def optimize_with_differential_evolution(initial_h, time_budget_s):
    """Use differential evolution to optimize the objective"""
    start = time.time()
    
    def objective(h):
        # Return negative C5 since we're minimizing
        return -c5_of(h)
    
    # Define bounds (0 to 1 for each variable)
    bounds = [(0, 1) for _ in range(len(initial_h))]
    
    # Run differential evolution with early stopping
    res = differential_evolution(
        objective,
        bounds,
        maxiter=int(5000 * (time_budget_s / 600)),
        popsize=15,
        tol=1e-6,
        recombination=0.7,
        mutation=(0.5, 1.0),
        seed=42,
        workers=1,
        updating='immediate',
        disp=False
    )
    
    # Return the best feasible solution
    h_opt = project_to_feasible(res.x)
    c5_opt = -res.fun
    
    return h_opt, c5_opt
