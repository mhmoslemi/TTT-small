import numpy as np
import time
import cvxpy as cp

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using full convolution
    overlap = np.correlate(h, 1 - h, mode='full')
    # The maximum value scaled by dx
    c5 = np.max(overlap) * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Try solving with MILP using cvxpy
    N = 140
    dx = 2.0 / N
    
    # Create the MILP problem
    x = cp.Variable(N, boolean=True)
    C = cp.Variable()
    
    # Constraint: sum of x equals N/2
    constraints = [cp.sum(x) == N // 2]
    
    # For each lag from 1 to N-1
    for ell in range(1, N):
        length = N - ell
        if length <= 0:
            continue
            
        # Sum of h_i over first (N-ell) indices
        H = cp.sum(x[:length])
        
        # y[i] = x[i] * x[i+ell]
        y = cp.Variable(length, boolean=True)
        constraints += [
            y <= x[:length],
            y <= x[ell:],
            y >= x[:length] + x[ell:] - 1
        ]
        
        O = cp.sum(y)
        c = H - O
        
        constraints += [c <= C, c >= 0]
    
    prob = cp.Problem(cp.Minimize(C), constraints)
    
    # Allocate time for solving
    time_remaining = budget_s - (time.time() - start_time) - 5
    if time_remaining <= 0:
        # fallback to local search
        return fallback_local_search(seed, budget_s, N)
        
    try:
        prob.solve(solver=cp.CBC, max_seconds=time_remaining, verbose=False)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE, cp.FEASIBLE]:
            # Extract solution
            x_sol = np.array(x.value).round().astype(int)
            # Ensure sum constraint met exactly
            if np.sum(x_sol) != N // 2:
                # Use the closest integer solution that respects constraint
                while np.sum(x_sol) > N // 2:
                    idx = np.argmax(x_sol)
                    x_sol[idx] = 0
                while np.sum(x_sol) < N // 2:
                    idx = np.argmin(x_sol)
                    x_sol[idx] = 1
                    
            c5_bound = c5_of(x_sol)
            return (x_sol, c5_bound, N)
    except Exception as e:
        pass
    
    # Fallback to local search if MILP fails
    return fallback_local_search(seed, budget_s, N)

    def propose_move(h, max_step):
        h_new = h.copy()
        while True:
            i, j = random.sample(range(N), 2)
            delta_low = max(-h_new[i], h_new[j] - 1.0)

            if delta_low >= delta_high:
