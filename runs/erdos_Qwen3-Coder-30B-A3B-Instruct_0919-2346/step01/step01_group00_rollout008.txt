import numpy as np
import time
import random

def c5_of(h):
    """Compute C5 = max_k integral h(x)(1-h(x+k)) dx."""
    dx = 2.0 / len(h)
    # Compute correlation: h * (1 - h) with full mode
    c = np.correlate(h, 1 - h, mode="full")
    # Multiply by dx to get continuous integral
    c *= dx
    # Return maximum value among all shifts
    return np.max(c)

def feasible_projection(h):
    """Project h to feasible region [0,1] and ensure integral constraint."""
    # Clamp to [0,1]
    h = np.clip(h, 0, 1)
    # Scale to meet integral constraint
    dx = 2.0 / len(h)
    scale = (len(h) / 2) / np.sum(h)
    h = h * scale
    return h

def random_perturbation(h, max_delta=0.05):
    """Generate a random feasible perturbation."""
    h = h.copy()
    N = len(h)
    # Pick random indices to perturb
    idx1, idx2 = random.sample(range(N), 2)
    
    # Determine valid range for delta
    delta_low = max(-h[idx1], h[idx2] - 1.0)
    delta_high = min(1.0 - h[idx1], h[idx2])
    
    if delta_low >= delta_high:
        return h
    
    # Sample delta
    delta = random.uniform(max_delta * delta_low, max_delta * delta_high)
    h[idx1] += delta
    h[idx2] -= delta
    
    return feasible_projection(h)

def local_search(h, max_iter=1000):
    """Simple local search to improve solution."""
    h = h.copy()
    c_current = c5_of(h)
    
    for _ in range(max_iter):
        h_new = random_perturbation(h, 0.01)
        c_new = c5_of(h_new)
        if c_new < c_current:
            h = h_new
            c_current = c_new
    return h, c_current

                      method='SLSQP', options={'maxiter': maxiter, 'ftol': 1e-8})
    if result.success:
        return result.x
    return h0
