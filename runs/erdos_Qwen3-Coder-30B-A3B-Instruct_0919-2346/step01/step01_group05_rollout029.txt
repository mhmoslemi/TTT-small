import numpy as np

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value scaled by dx
    max_overlap = np.max(correlation) * dx
    return max_overlap

def generate_sturmian_binary(n):
    """Generate a balanced binary sequence using irrational rotation"""
    alpha = (np.sqrt(5) - 1) / 2  # Golden ratio conjugate
    h = np.zeros(n, dtype=int)
    count = 0
    for i in range(n):
        t = (i * alpha) % 1
        if t < 0.5:
            h[i] = 1
            count += 1
    # Adjust to exactly n/2 ones
    if count > n // 2:
        # Find indices with 1s to flip to 0s
        ones_indices = np.where(h == 1)[0]
        for i in range(count - n // 2):
            h[ones_indices[i]] = 0
    elif count < n // 2:
        # Find indices with 0s to flip to 1s
        zeros_indices = np.where(h == 0)[0]
        for i in range(n // 2 - count):
            h[zeros_indices[i]] = 1
    return h

def compute_overlap_vector(h):
    """Compute the overlap vector for all shifts"""
    n = len(h)
    dx = 2.0 / n
    correlation = np.correlate(h, 1 - h, mode="full")
    return correlation * dx

def delta_shift(h, i, j, s):
    """Compute the change in overlap at shift s when swapping positions i and j"""
    n = len(h)
    # Contributions to shift s from change in h_i and h_j
    delta = 0
    
    # Check if position i contributes to shift s (i, i+s)
    if 0 <= i < n and 0 <= i + s < n:
        # Contribution of i to overlap at shift s changed from h[i]*(1-h[i+s]) to 0*(1-h[i+s])
        # So delta is -h[i]*(1-h[i+s]) = -(1-h[i+s]) since h[i] = 1
        # For simplicity, we compute the direct change in objective
        delta -= 1 - h[i + s] if i + s < n else 0  # i contributes to shift s

    # Check if position j contributes to shift s (j, j+s)
    if 0 <= j < n and 0 <= j + s < n:
        # Contribution of j to overlap at shift s changed from h[j]*(1-h[j+s]) to (1-h[j])*(1-h[j+s])
        # So delta is (1-h[j])*(1-h[j+s]) - h[j]*(1-h[j+s]) = (1-2*h[j])*(1-h[j+s])
        if j + s < n:
            delta += (1 - 2 * h[j]) * (1 - h[j + s])
        else:
            delta += (1 - 2 * h[j]) * 0

    # Deals with double-counting in applying the rule: position i and j are being swapped
    # As a proxy for full two-way accounting, we try to manage it through updating later
    
    return delta

def compute_deltas(h, i, j, overlap_vector):
    """Update overlap_vector with changes due to swapping positions i and j"""
    n = len(h)
    # Only the linear component changes, we can do this efficiently
    # Create a temporary delta vector since we have many dependencies
    new_overlap = overlap_vector.copy()
    
    # For each shift s, we determine changed overlap
    # Overlap for shift s is:
    # sum_{k} h[k] * (1 - h[k+s]) over all k
    
    # The change in overlap depends upon which k positions are adjacent to i or j, through s and s+k
    # This is tricky, just directly compute
    # But since the change for any shift is:
    # - for position i: term changes from h[i] * (1 - h[i+s]) to 0 * (1 - h[i+s]) = 0
    #   so decrease by h[i] * (1 - h[i+s])
    # - for position j: term changes from h[j] * (1 - h[j+s]) to (1 - h[j]) * (1 - h[j+s])
    #   so increase by (1-h[j]) * (1 - h[j+s]) - h[j] * (1 - h[j+s]) = (1 - 2*j) * (1 - h[j+s])
    
    # Update for all s
    # The notion is:
    # For all s, shift contribution for i changed by -h[i] * (1 - h[i+s]), 
    # and for j changed by (1 - h[j]) * (1 - h[j+s]) - h[j] * (1 - h[j+s]) = (1 - 2*h[j]) * (1 - h[j+s])
    for s in range(-n + 1, n):
        k = i - s
        if 0 <= k < n:
            # contribution of pos k: k contributes h[k]* (1 - h[k+s]) 
            # Can be computed but simpler: just move dots
            # adds a - h[i]* (1 - h[i+s]) but not easy to add only step like this 
            # doing it all systematically
            pass
    
    # Instead we use an easier full overlap evaluation 
    # Copied approach from fastDirect: 
    # For每個s, we compute only the terms affected by the change
    # Here, the overlap goes from `sum h_i (1 - h_{i+s})` to `sum new_h_i (1 - new_h_{i+s})`
    # But since we only swap two points:
    # All terms (i+s) such i under window of s change impact  
    # There is a simpler approach: we go over s=1 to n-1 and 
    # The change for s where i or j map shifts (note in s, position k matches with k+s)
    # For shift s, some terms have h[k]*(1-h[k+s]), a few may be affected if s-i or s-j is in legitimate range
    
    # The most likely values are those where s is around i and j but not simple array access
    
    # Let's recompute optimized naively for accurate testing
    # Just compute change efficiently
    
    # Instead, just compute the new overlaps directly via manual BFS shift:
    # For 1 shift s update:
    # naive but explicit
    effective_n = n  # fit typical size
    
    # For a very specific change in positions i,j. we can put into full system with a O(n) reactive:
    updated_overlap = overlap_vector.copy()
    for s in range(1 - n, n):
        # Solve for actual most and least intermediate shape:  
        # Compute exact value change
        old_val = (h[i] * (1 - h[(i + s) % n]) if 0 <= (i + s) % n < n else 0) 
        old_val += (h[j] * (1 - h[(j + s) % n]) if 0 <= (j + s) % n < n else 0) 
        new_val = (0 * (1 - h[(i + s) % n]) if 0 <= (i + s) % n < n else 0) 
        new_val += (1 * (1 - h[(j + s) % n]) if 0 <= (j + s) % n < n else 0)
        diff = new_val - old_val
        updated_overlap[n-1+s] += diff
        
    return updated_overlap

def local_swap_search(h, n_points):
    """Perform systematic swap-based optimization for the worst shift"""
    dx = 2.0 / n_points
    # Initial compute of overlap vector
    overlap_vector = compute_overlap_vector(h)
    max_shift_index = np.argmax(overlap_vector)
    current_max = overlap_vector[max_shift_index]
    
    # Store indices for ones and zeros
    ones_indices = np.where(h == 1)[0]
    zeros_indices = np.where(h == 0)[0]
    
    # For quick lookup of overlap changes
    best_h = h.copy()
    best_max = current_max
    best_overlap = overlap_vector.copy()
    
    iteration = 0
    max_iter = 50000
    
    tabu_list = set()
    tabu_size = 200
    
    for _ in range(max_iter):
        # compute initial overlap
        if iteration > 0:
            overlap_vector = compute_overlap_vector(h)
        
        current_max = np.max(overlap_vector)
        max_shift_index = np.argmax(overlap_vector)
        max_shift = max_shift_index - (n_points - 1)

        if current_max < best_max:
            best_h = h.copy()
            best_max = current_max
            best_overlap = overlap_vector.copy()

        # Identify critical (frequency-based 손실) points for swapping
        # Find positions contributing to the maximum shift:  
        # These have high overlap contribution via h[i]*(1-h[i+s])
        # i where there is a jump (h[i+s] = 0 -> large term)
        # --> focus on critical pairs (i, j) where i moves from 1 to 0, j from 0 to 1
        # and affect max shift well, constrained to change g_s by positive.
        
        s_my = max_shift
        # When i between 0 and n-1, then i+s is valid if i+s < n -> i < n-s
        # Conversely, s grows so it is going negative, which means i+s >= 0 -> i >= -s
        # i = indices within a valid range for shift s
        
        # To fast list candidates for swap, precompute of integral indices
        # One simple thing: shuffle ones/Zeros to find best pair
        if len(ones_indices) > 0 and len(zeros_indices) > 0:
            # At each step, pick a swap from potential pairs
            i_idx = np.random.randint(len(ones_indices))
            j_idx = np.random.randint(len(zeros_indices))
            i_pos = ones_indices[i_idx]
            j_pos = zeros_indices[j_idx]
            
            # consolidate swaps
            if (i_pos, j_pos) in tabu_list:
                continue  # skip temporal tabu
            
            # simulate swap update
            h_temp = h.copy()
            h_temp[i_pos] = 0
            h_temp[j_pos] = 1
            
            # Calculate the new overlap vector after the swap
            temp_overlap_vector = compute_overlap_vector(h_temp)
            new_max = np.max(temp_overlap_vector)
            
            # Update swap
            if new_max < current_max:
                h = h_temp
                overlap_vector = temp_overlap_vector
                # Update tabu
                if len(tabu_list) >= tabu_size:
                    tabu_list.pop()
                tabu_list.add((i_pos, j_pos))
            else:
                # accept minor uphill moves sometimes to escape local minima
                if np.random.rand() < 0.01:
                    h = h_temp
                    overlap_vector = temp_overlap_vector
                    if len(tabu_list) >= tabu_size:
                        tabu_list.pop()
                    tabu_list.add((i_pos, j_pos))
        
        iteration += 1
    
    return best_h, best_max

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    available_time = budget_s
    
    # Resolution for coarse-grained local search
    n0 = 120
    n1 = 240    # upscale resolution  
    n2 = 480    # full resolution
    
    best_h = None
    best_c5 = float('inf')
    best_n = None
    
    def bottlenecked_time():
        return (time.time() - start_time) / available_time
    
    # Coarse resolution search
    h0 = generate_sturmian_binary(n0)
    h_opt, c5_opt = local_swap_search(h0, n0)
    if c5_opt < best_c5:
        best_h = h_opt
        best_c5 = c5_opt
        best_n = n0
    
    # Upscale and refine
    if bottlenecked_time() < 0.70:
        h1_up = np.repeat(h_opt, 2)  # Basic duplication
        h1_up = h1_up[:n1]
        if len(h1_up) < n1:
            h1_up = np.pad(h1_up, (0, n1 - len(h1_up)), mode='constant')
        elif len(h1_up) > n1:
            h1_up = h1_up[:n1]
        
        # adjust balance
        ones_count = np.sum(h1_up)
        if ones_count > n1 // 2:
            from collections import Counter
            idxs = np.where(h1_up == 1)[0]
            pick_idxs = np.random.choice(idxs, int(ones_count - n1 // 2), replace=False)
            h1_up[pick_idxs] = 0
        elif ones_count < n1 // 2:
            idxs = np.where(h1_up == 0)[0]
            pick_idxs = np.random.choice(idxs, int(n1 // 2 - ones_count), replace=False)
            h1_up[pick_idxs] = 1

        h_opt, c5_opt = local_swap_search(h1_up, n1)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
            best_n = n1
    
    # Fine-grained after 40%  for poly resolution
    if bottlenecked_time() < 0.90:
        h_fine = np.repeat(best_h, 2)[:n2]
        if len(h_fine) < n2:
            h_fine = np.pad(h_fine, (0, n2 - len(h_fine)), mode='constant')
        elif len(h_fine) > n2:
            h_fine = h_fine[:n2]
        
        # re-adjust balance  
        ones_count = np.sum(h_fine)
        if ones_count > n2 // 2:
            idxs = np.where(h_fine == 1)[0]
            pick_idxs = np.random.choice(idxs, int(ones_count - n2 // 2), replace=False)
            h_fine[pick_idxs] = 0
        elif ones_count < n2 // 2:
            idxs = np.where(h_fine == 0)[0]
            pick_idxs = np.random.choice(idxs, int(n2 // 2 - ones_count), replace=False)
            h_fine[pick_idxs] = 1
            
        h_opt, c5_opt = local_swap_search(h_fine, n2)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
            best_n = n2

    # Continuous polish on final
    if best_h is not None and not (bottlenecked_time() > 0.99):
        dx = 2.0 / best_n
        h_final = np.array(best_h, dtype=float)
        
        def objective(h_vec):
            return evaluate_erdos_solution(h_vec)
        
        # Step size truncation to control early steps
        bounds = [(0, 1) for _ in range(best_n)]
        
        # projection to sum = 1
        def proj_h(h_vec):
            adj_factor = (best_n / 2) / np.sum(h_vec)
            h_proj = h_vec * adj_factor
            return np.clip(h_proj, 0, 1)
        
        # Iterate with small gradient-like updates
        for i in range(5000):
            # take a gradient-like step
            grad = np.zeros_like(h_final)
            h_fwd = h_final.copy()
            h_bwd = h_final.copy()
            
            # Simple finite difference
            step_size = 0.0001
            for j in range(best_n):
                h_fwd[j] = min(1.0, h_final[j] + step_size)
                h_bwd[j] = max(0.0, h_final[j] - step_size)
                
                obj_fwd = evaluate_erdos_solution(h_fwd)
                obj_bwd = evaluate_erdos_solution(h_bwd)
                grad[j] = (obj_fwd - obj_bwd) / (2 * step_size)
                
                h_fwd[j] = h_final[j]
                h_bwd[j] = h_final[j]
            
            # update
            h_update = h_final - 0.001 * grad
            h_final = proj_h(h_update)
            
            # early break if improvement not seen for few iterations
            if i > 100 and i % 10 == 0:
                c5_test = evaluate_erdos_solution(h_final)
                if c5_test > best_c5:
                    break

        h_final = proj_h(h_final)
        best_c5 = evaluate_erdos_solution(h_final)
        best_h = h_final
    
    return best_h.tolist(), best_c5, best_n
