import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def compute_c5(h):
    """Compute the maximum overlap integral for the given step function h."""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    c = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(c) * dx
    return c5

def project_to_sum_constraint(h, target_sum):
    """Project h onto the constraint that sum(h) = target_sum"""
    # Clip to [0,1] first
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    if current_sum > target_sum:
        # Need to decrease values
        # Sort descending, remove largest elements until sum matches
        idxs = np.argsort(h)[::-1]
        h_copy = h.copy()
        s = current_sum
        for i in idxs:
            if s <= target_sum:
                break
            delta = min(h_copy[i], s - target_sum)
            h_copy[i] -= delta
            s -= delta
        return h_copy
    else:
        # Need to increase values
        # Sort ascending, raise smallest elements until sum matches
        idxs = np.argsort(h)
        h_copy = h.copy()
        s = current_sum
        for i in idxs:
            if s >= target_sum:
                break
            delta = min(1 - h_copy[i], target_sum - s)
            h_copy[i] += delta
            s += delta
        return h_copy

def propose_move(h, step_size):
    """Propose a new feasible step function by adjusting two entries."""
    N = len(h)
    while True:
        i, j = np.random.choice(N, size=2, replace=False)
        # Determine feasible step
        delta_low = max(-h[i], h[j] - 1)
        delta_high = min(1 - h[i], h[j])
        if delta_low >= delta_high:
            continue  # Try another pair
        # Sample step size
        step_size = np.random.uniform(max(-step_size, delta_low),
                                      min(step_size, delta_high))
        # Create new h
        h_new = h.copy()
        h_new[i] += step_size
        h_new[j] -= step_size
        return h_new

def local_refinement(h_start, max_iter=200, step_size=0.02):
    """Run local refinement starting from h_start."""
    h = h_start.copy()
    c5 = compute_c5(h)
    for _ in range(max_iter):
        # Find the worst shift
        dx = 2.0 / len(h)
        c = np.correlate(h, 1 - h, mode='full')
        worst_idx = np.argmax(c)
        # Compute gradient for that shift
        shift = worst_idx - (len(h) - 1)
        # Approximate gradient for h[i] w.r.t. overlap with shift 'shift'
        grad = np.zeros_like(h)
        if shift >= 0:
            # h[i] contributes to overlap with shift s: h[i] * (1 - h[i+s])
            # So d/dh[i] overlap = 1 - h[i+s] - h[i] * 0 = 1 - h[i+s]
            for i in range(len(h) - shift):
                grad[i] += 1 - h[i + shift]
        else:
            # shift < 0, so i+s is i - |shift|
            # h[i] contributes to overlap with shift -s: h[i] * (1 - h[i-s])
            # d/dh[i] overlap = 1 - h[i-s]
            for i in range(-shift, len(h)):
                grad[i] += 1 - h[i + shift]
        # Move opposite to gradient
        h_new = h - step_size * grad
        # Project to feasible set: 0 <= h <= 1, sum(h) = 1
        h_new = np.clip(h_new, 0, 1)
        # Project to sum = 1
        h_new = project_to_sum_constraint(h_new, len(h) / 2.0)
        c5_new = compute_c5(h_new)
        if c5_new < c5:
            h = h_new
            c5 = c5_new
        else:
            # Accept with small probability to escape local minima
            if np.random.rand() < 0.01:
                h = h_new
                c5 = c5_new
    return h, c5

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start = time.time()
    N = 120  # Reduce size for tractability within time limit
    dx = 2.0 / N

    # Step 1: Set up the linear programming relaxation
    # Build indices for the y variables: (i,s) pairs such that i+s < N
    pairs = []
    for s in range(N):
        for i in range(N - s):
            pairs.append((i, s))

    # Create CVXPY variables
    h = cp.Variable(N, name="h")
    y = cp.Variable(len(pairs), name="y")
    alpha = cp.Variable(name="alpha")

    # Constraints for every y[i,s] pair
    constraints = []
    for idx, (i, s) in enumerate(pairs):
        constraints.extend([
            y[idx] <= h[i],
            y[idx] <= h[i + s],
            y[idx] >= h[i] + h[i + s] - 1,
            y[idx] >= 0
        ])

    # Sum constraint
    constraints.append(cp.sum(h) == N / 2)

    # Define C_s for each s = 0 to N-1 (since only non-negative
    # shifts needed due to symmetry)
    for s in range(N):
        # Sum over all terms for shift s
        sum_y_s = 0
        for idx, (i, ss) in enumerate(pairs):
            if ss == s:
                sum_y_s += y[idx]
        # α must be >= dx*(N/2 - sum_y_s)
        constraints.append(alpha >= dx * (N / 2 - sum_y_s))

    # Objective
    objective = cp.Minimize(alpha)

    # Formulate the problem
    prob = cp.Problem(objective, constraints)

    # Solve the LP relaxation
    solver_options = {'max_iters': 10000}
    try:
        prob.solve(solver=cp.ECOS, **solver_options)
        h_val = h.value
        alpha_val = alpha.value
    except Exception as e:
        print("LP solver error: ", e)
        # Fallback to uniform distribution
        h_val = np.ones(N) * 0.5
        h_val = project_to_sum_constraint(h_val, N / 2)
        alpha_val = float('inf')
    if time.time() - start > 0.9 * budget_s:
        return (h_val, compute_c5(h_val), N)

    # Phase 2: Possibly solve binary MILP
    # Check if the LP solution is good enough
    c5_lp = compute_c5(h_val)
    if c5_lp <= 0.3807:
        return (h_val, c5_lp, N)

    # Phase 2.5: Try rounding approach
    # Greedy rounding to binary
    h_rounded = np.zeros(N)
    idx_order = np.argsort(h_val)[::-1]
    count = 0
    for i in idx_order:
        if count < N / 2:
            h_rounded[i] = 1
            count += 1
    h_rounded = project_to_sum_constraint(h_rounded, N / 2)
    c5_rd = compute_c5(h_rounded)
    h_best = h_rounded
    c5_best = c5_rd

    # Try to slightly improve the rounded result with hill climbing
    if time.time() - start > 0.85 * budget_s:
        return (h_best, c5_best, N)
    h_refined, c5_refined = local_refinement(h_rounded, max_iter=50)
    if c5_refined < c5_best:
        h_best = h_refined
        c5_best = c5_refined

    # Phase 3: Run binary MILP on a subset for potentially better solution
    if time.time() - start > 0.8 * budget_s:
        return (h_best, c5_best, N)
    # Rebuild problem with integer constraints on h
    h_int = cp.Variable(N, boolean=True, name="h_int")
    y_int = cp.Variable(len(pairs), name="y_int")
    alpha_int = cp.Variable(name="alpha_int")

    # Constraints (same as LP but with integer variables)
    constraints_int = []
    for idx, (i, s) in enumerate(pairs):
        constraints_int.extend([
            y_int[idx] <= h_int[i],
            y_int[idx] <= h_int[i + s],
            y_int[idx] >= h_int[i] + h_int[i + s] - 1,
            y_int[idx] >= 0
        ])

    # Sum constraint
    constraints_int.append(cp.sum(h_int) == N / 2)
    
    # Overlap constraint
    for s in range(N):
        sum_y_s = 0
        for idx, (i, ss) in enumerate(pairs):
            if ss == s:
                sum_y_s += y_int[idx]
        # α_int must be >=  dx*(N/2 - sum_y_s)
        constraints_int.append(alpha_int >= dx * (N / 2 - sum_y_s))

    # Objective
    objective_int = cp.Minimize(alpha_int)

    prob_int = cp.Problem(objective_int, constraints_int)

    # Solve with limited time
    time_left = budget_s - (time.time() - start)
    if time_left > 10:
        try:
            # Make a warm start using what we've already computed
            warm_start_h = np.round(h_val).astype(int)
            # Need to satisfy the sum
            warm_start_h = warm_start_h.astype(float)
            warm_start_h = project_to_sum_constraint(warm_start_h, N / 2)
            warm_start_h = np.round(warm_start_h).astype(int)
            h_int.value = warm_start_h
            solver_options = {
                'maxTime': time_left - 5,  # Allow 5s overhead for solver call
                'relGap': 1e-4
            }
            prob_int.solve(solver=cp.CBC, **solver_options)
            if h_int.value is not None:
                h_milp = h_int.value
                c5_milp = compute_c5(h_milp)
                if c5_milp < c5_best:
                    h_best = h_milp
                    c5_best = c5_milp
        except Exception as e:
            pass

    # Phase 4: Final local search refinement if needed
    if time.time() - start < 0.95 * budget_s:
        h_final, c5_final = local_refinement(h_best, max_iter=100)
        if c5_final < c5_best:
            h_best = h_final
            c5_best = c5_final

    return (h_best, c5_best, N)
