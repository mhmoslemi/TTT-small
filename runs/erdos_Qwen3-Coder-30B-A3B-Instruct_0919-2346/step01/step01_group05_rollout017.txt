import numpy as np
from scipy.optimize import minimize
from math import pi
import time

def build_h_from_coeffs(a, n):
    """Build h from Fourier coefficients a"""
    h = 0.5 * np.ones(n)
    M = len(a)
    for m in range(1, M+1):
        cos_term = np.cos(2 * pi * m * np.arange(n) / n)
        h += a[m-1] * cos_term
    return h

def compute_overlap_and_gradient(a, n, dx, gamma=1.0):
    """Compute overlaps for all shifts and their gradients w.r.t. a"""
    h = build_h_from_coeffs(a, n)
    # Ensure h is in [0,1]
    h = np.clip(h, 0, 1)
    
    # Compute correlation (full mode)
    corr = np.correlate(h, 1 - h, mode="full")
    # The overlap values for each shift k=-(n-1),...,n-1
    shifts = np.arange(-(n-1), n)
    overlaps = corr * dx
    # Weight by softmax
    exp_overlaps = np.exp(gamma * overlaps)
    weights = exp_overlaps / np.sum(exp_overlaps)
    
    # Compute penalty contribution
    penalty = 1e4 * (np.sum(np.maximum(0, h-1)**2) + np.sum(np.maximum(0, -h)**2))
    
    # Compute gradient of overlaps w.r.t. a_m
    grad = np.zeros(len(a))
    for m in range(1, len(a)+1):
        cos_m = np.cos(2 * pi * m * np.arange(n) / n)
        for s in range(-n+1, n):
            i = np.arange(n)
            i_s = i + s
            valid = (i_s >= 0) & (i_s < n)
            if not np.any(valid):
                continue
            # f_s = dx * sum_{i valid} h_i * (1 - h_{i+s})
            # grad f_s / grad a_m = dx * sum_{i valid} cos_m[i] * (1 - h_{i+s})
            #                      - dx * sum_{i valid} h_i * cos_m[i+s]
            grad[m-1] += weights[s + n - 1] * (
                dx * np.sum(cos_m[valid] * (1 - h[i_s[valid]])) 
                - dx * np.sum(h[valid] * np.roll(cos_m, s)[valid])
            )
    
    # Gradient of penalty
    clip_diff = h - np.clip(h, 0, 1)
    penalty_grad = 2 * 1e4 * np.sum(clip_diff[:, np.newaxis] * 
                                    np.cos(2 * pi * np.arange(1, len(a)+1)[:, np.newaxis] * 
                                           np.arange(n) / n), axis=1)
    return np.sum(weights * overlaps), grad + penalty_grad

def project_to_simplex_box(h, n, target_sum):
    """Project h onto the simplex h in [0,1]^n with sum h_i = target_sum"""
    # Bisection on the water filling parameter tau
    h = np.clip(h, 0, 1)
    h_sorted = np.sort(h)[::-1]
    h_cumsum = np.cumsum(h_sorted)
    n_sorted = np.arange(1, n+1)
    # Find the largest k such that h_sorted[k-1] >= (target_sum - h_cumsum[k-1]) / k
    k = 1
    while k < n and h_sorted[k-1] >= (target_sum - h_cumsum[k-1]) / k:
        k += 1
    # Compute tau
    if k == n:
        tau = (target_sum - h_cumsum[k-1]) / k
    else:
        tau = (target_sum - h_cumsum[k-1]) / k
    # Apply clipping
    h_projected = np.clip(h - tau, 0, 1)
    return h_projected

def spectral_optim(n=240, max_iter=10000, seed=42):
    np.random.seed(seed)
    dx = 2.0 / n
    # Stage 1: M=8
    M = 8
    a = np.random.normal(0, 0.05, M)  
    # Bounds for a
    bounds = [(-0.6, 0.6) for _ in range(M)]
    
    gamma = 0.5
    last_f = float('inf')
    
    # For this implementation, a simplified scheme
    # We will directly do one optimization round
    def obj_and_grad(a_vec):
        loss, grad = compute_overlap_and_gradient(a_vec, n, dx, gamma)
        return loss, grad
    
    # Minimize using L-BFGS-B
    result = minimize(lambda x: obj_and_grad(x)[0], 
                      a, 
                      method='L-BFGS-B', 
                      jac=lambda x: obj_and_grad(x)[1],
                      bounds=bounds,
                      options={'maxiter': max_iter})
    a_opt = result.x
    h_raw = build_h_from_coeffs(a_opt, n)
    h_feas = project_to_simplex_box(h_raw, n, n/2)
    
    # Evaluate final C5
    final_func = lambda h_new: np.max(np.correlate(h_new, 1 - h_new, mode="full")) * dx
    c5 = final_func(h_feas)
    return h_feas, c5

def run(seed=42, budget_s=600, **kwargs):
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    
    # Run multiple restarts to avoid local minima
    n_restarts = 6
    for i in range(n_restarts):
        if time.time() - start_time > budget_s * 0.95:
            break
        h, c5 = spectral_optim(seed=seed+i)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return (best_h.tolist(), best_c5, len(best_h))
