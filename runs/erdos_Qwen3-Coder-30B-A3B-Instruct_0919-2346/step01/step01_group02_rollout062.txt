import numpy as np
import time

def c5_of(h):
    """Compute the maximum overlap integral for the given step function h."""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    c = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(c) * dx
    return c5

def worst_shift(h):
    """Determine the shift that gives the maximum overlap."""
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    return idx - (len(h) - 1)

def subgradient(h, s):
    """Compute subgradient of C5 with respect to h for shift s."""
    N = len(h)
    dx = 2.0 / N
    m = abs(s)
    g = np.full(N, dx)  # 1*dx everywhere
    if m < N:
        g[:N-m] -= dx * h[m:]
        g[m:]   -= dx * h[:N-m]
    return g

def project(v, target):
    """Project v onto the feasible set: 0 <= h[i] <= 1 and sum(h) = target."""
    lo = -np.max(v)
    hi = 1.0 - np.min(v)
    for _ in range(50):
        lam = (lo + hi) / 2.0
        w = np.clip(v + lam, 0.0, 1.0)
        if w.sum() < target:
            lo = lam
        else:
            hi = lam
    w = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    # Account for rounding errors
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        free = (w > 0) & (w < 1)
        if free.any():
            w[free] += diff / free.sum()
            w = np.clip(w, 0, 1)
    return w

def spd_descent(h_start, max_iter, eta0=30.0, gamma=0.9995, time_budget=600):
    """Subgradient projected descent."""
    start = time.time()
    h = h_start.copy()
    best_val = c5_of(h)
    best_h = h.copy()
    t = 0
    while time.time() - start < time_budget and t < max_iter:
        s = worst_shift(h)
        g = subgradient(h, s)
        eta = eta0 * (gamma ** t)
        # Avoid large steps
        if eta > 50.0 * (2.0 / len(h)):
            eta = 50.0 * (2.0 / len(h))

        v = h - eta * g
        h = project(v, len(h) / 2.0)
        val = c5_of(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()
        t += 1
        # Occasionally jitter for escaping plateaux
        if t % 30 == 0:
            for i in range(len(h)):
                delta = np.random.uniform(-0.02, 0.02)
                h[i] = np.clip(h[i] + delta, 0, 1)
            h = project(h, len(h) / 2.0)
            val = c5_of(h)
            if val < best_val:
                best_val = val
                best_h = h.copy()
    return best_h, best_val

def propose_move(h, max_step):
    """Propose a new feasible step function by adjusting two entries."""
    N = len(h)
    while True:
        i, j = np.random.choice(N, size=2, replace=False)
        delta_low = max(-h[i], h[j] - 1)
        delta_high = min(1 - h[i], h[j])
        if delta_low >= delta_high:
            continue  # Try another pair
        step_size = np.random.uniform(max(-max_step, delta_low),
                                      min(max_step, delta_high))
        h_new = h.copy()
        h_new[i] += step_size
        h_new[j] -= step_size
        return h_new

def anneal(h_start, time_budget, T_start=0.05, T_end=1e-6, step_start=0.2):
    """Perform simulated annealing."""
    start = time.time()
    cur = h_start.copy()
    c_cur = c5_of(cur)
    best = cur.copy()
    c_best = c_cur
    T = T_start
    step = step_start
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        frac = elapsed / time_budget
        T = T_start * (T_end / T_start) ** frac
        step = step_start * (1.0 - frac)
        h_new = propose_move(cur, step)
        c_new = c5_of(h_new)
        if c_new <= c_cur or np.random.rand() < np.exp(-(c_new - c_cur) / T):
            cur = h_new
            c_cur = c_new
            if c_new < c_best:
                best = cur.copy()
                c_best = c_new
    return best, c_best

def hill_climb(h_start, time_budget):
    """Perform deterministic hill climb."""
    start = time.time()
    cur = h_start.copy()
    c_cur = c5_of(cur)
    best = cur.copy()
    c_best = c_cur
    while time.time() - start < time_budget:
        h_new = propose_move(cur, 0.02)
        c_new = c5_of(h_new)
        if c_new < c_best:
            best = h_new.copy()
            c_best = c_new
            cur = h_new
            c_cur = c_new
    return best, c_best

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    N = 140
    dx = 2.0 / N

    # Seeding phase: generate diverse starts
    seeds = []
    if 'initial_h_values' in globals():
        h0 = np.interp(np.linspace(0, 2, N), 
                      np.linspace(0, 2, len(initial_h_values)), 
                      initial_h_values)
        h0 = h0 * (N/2) / np.sum(h0)  # Renormalize exactly
        seeds.append(h0)
    else:
        h0 = np.ones(N) * 0.5  # fallback
        seeds.append(h0)
        
    # Two-peak seed 
    h1 = np.ones(N)
    x_vals = np.linspace(0, 2, N)
    a = 0.3
    u = 0.9
    v = 0.1
    mask = (x_vals < a) | (x_vals >= 2 - a)
    h1[mask] = u
    h1[~mask] = v
    h1 = h1 * (N/2) / np.sum(h1)  # Renormalize
    seeds.append(h1)

    # Random binary vectors
    h2 = np.random.choice([0, 1], size=N, p=[0.5, 0.5])
    h2 = h2 * (N/2) / np.sum(h2)
    seeds.append(h2)
    
    # Uniform
    h3 = np.ones(N) * 0.5
    seeds.append(h3)

    # Process seeds
    best_val = float('inf')
    best_h = None
    start = time.time()

    for i, h_seed in enumerate(seeds):
        h_seed = project(h_seed, N / 2.0)  # Ensure sum exactly N/2
        val = c5_of(h_seed)
        if val < best_val:
            best_val = val
            best_h = h_seed.copy()

    # Transition to phase 1: SPD descent
    phase_time = 0.55 * budget_s
    h_final, c_final = spd_descent(best_h, max_iter=10000, time_budget=phase_time)
    if c_final < best_val:
        best_val = c_final
        best_h = h_final

    # Phase 2: Annealing
    phase_time = 0.30 * budget_s
    h_final, c_final = anneal(best_h, time_budget=phase_time)
    if c_final < best_val:
        best_val = c_final
        best_h = h_final

    # Phase 3: Hill Climbing
    phase_time = 0.10 * budget_s
    h_final, c_final = hill_climb(best_h, time_budget=phase_time)
    if c_final < best_val:
        best_val = c_final
        best_h = h_final

    return (best_h, best_val, N)
