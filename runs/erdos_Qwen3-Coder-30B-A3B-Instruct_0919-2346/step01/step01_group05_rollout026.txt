import numpy as np

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value scaled by dx
    max_overlap = np.max(correlation) * dx
    return max_overlap

def build_h_from_coeffs(a_coeffs, n, M):
    """Build h from Fourier coefficients"""
    h = 0.5 * np.ones(n)
    for m in range(1, M+1):
        cos_vals = np.cos(2 * np.pi * m * np.arange(n) / n)
        h += a_coeffs[m-1] * cos_vals
    return np.clip(h, 0, 1)

def compute_overlap_and_gradient(a_coeffs, n, M, gamma=1.0):
    """Compute the soft-max overlap and its gradient w.r.t. coefficients"""
    h = build_h_from_coeffs(a_coeffs, n, M)
    dx = 2.0 / n
    
    # Compute the correlation
    corr = np.correlate(h, 1 - h, mode="full")
    shifts = np.arange(-n + 1, n)
    f_values = corr * dx
    
    # Soft max approximation of max
    exp_f = np.exp(gamma * f_values)
    w = exp_f / np.sum(exp_f)
    soft_max = np.log(np.sum(exp_f)) / gamma
    
    # Add penalty to soft max
    penalty = 1e4 * (np.sum(np.maximum(0, h - 1)**2) + np.sum(np.maximum(0, -h)**2))
    objective = soft_max + penalty
    
    # Gradient
    grad = np.zeros(M)
    for m in range(1, M+1):
        cos_vals = np.cos(2 * np.pi * m * np.arange(n) / n)
        grad_m = 0
        for s in range(len(f_values)):
            shift_val = shifts[s]
            i_vals = np.arange(n)
            i_s_vals = i_vals + shift_val
            valid_indices = (i_vals >= 0) & (i_s_vals < n)
            i_valid = i_vals[valid_indices]
            i_s_valid = i_s_vals[valid_indices]
            
            term1 = np.sum(cos_vals[i_valid] * (1 - h[i_s_valid]))
            term2 = np.sum(h[i_valid] * cos_vals[i_s_valid])
            grad_m += w[s] * (term1 - term2)
        grad[m-1] = grad_m * dx
    
    # Gradient of penalty
    for m in range(1, M+1):
        cos_vals = np.cos(2 * np.pi * m * np.arange(n) / n)
        clip_diff = h - np.clip(h, 0, 1)
        grad[m-1] += 2 * 1e4 * np.sum(clip_diff * cos_vals)
        
    return objective, grad

def project_to_simplex_box(h, target_sum):
    """Project h to the simplex and box constraints exactly"""
    dx = 2.0 / len(h)
    current_sum = np.sum(h) * dx
    h = h * (target_sum / current_sum)
    h = np.clip(h, 0, 1)
    # Now project to the mean constraint
    # Water filling
    h_sorted = np.sort(h)[::-1]
    n = len(h_sorted)
    cumsum = np.cumsum(h_sorted)
    # Find tau such that sum of max(h[i] - tau, 0) = target_sum
    tau = 0
    total = 0
    for i in range(n):
        if h_sorted[i] - (target_sum - cumsum[i]) / (n - i) > 0:
            tau = (target_sum - cumsum[i]) / (n - i)
            break
    
    h_proj = np.clip(h - tau, 0, 1)
    return h_proj
