import numpy as np
import time
import random
from scipy import optimize


def compute_c5(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    overlap = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(overlap) * dx
    return c5


def project_onto_feasible(h, dx):
    """Project h to satisfy sum(h) * dx = 1, and 0 <= h[i] <= 1"""
    h = np.clip(h, 0, 1)
    total = np.sum(h) * dx
    if abs(total - 1.0) > 1e-10:
        h = h * (1.0 / total)
    return h


def gradient_step(h, step_size, beta):
    """Compute gradient-based update using softmax-weighted worst shift"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlations
    corr = np.correlate(h, 1 - h, mode="full")
    # Find the worst shift (max correlation)
    max_idx = np.argmax(corr)
    # Extract shift (in units of samples) corresponding to max_idx
    shift = max_idx - (n - 1)
    # Compute gradient of h at each point w.r.t. that specific shift
    # From h(x)(1 - h(x + shift)), directional derivative is:
    # (1 - h(x + shift)) * grad h(x) - h(x) * grad h(x + shift)
    grad_h = np.zeros(n)
    for i in range(n):
        j = i + shift
        if 0 <= j < n:
            grad_h[i] += (1 - h[j])
        j = i - shift
        if 0 <= j < n:
            grad_h[i] -= h[j]
    # Clip for numerical stability
    grad_h = np.clip(grad_h, -100, 100)
    # Apply gradient step (with temperature-controlled smoothing)
    new_h = h + step_size * grad_h
    # Project back
    new_h = project_onto_feasible(new_h, dx)
    return new_h


def smooth_optimize(h, max_iter=5000, step_size=0.05, beta_schedule=None, time_budget=600):
    """Main optimization loop with adaptive beta (softmax weighting)"""
    start = time.time()
    if beta_schedule is None:
        # Use decreasing schedule
        beta_schedule = [10 ** (3 - 3 * t / max_iter) for t in range(max_iter)]
    dx = 2.0 / len(h)
    h_best = h.copy()
    c5_best = compute_c5(h_best)
    for t in range(max_iter):
        if time.time() - start > time_budget * 0.95:
            break
        beta = beta_schedule[min(t, len(beta_schedule) - 1)]
        # Smooth gradient update with beta scaling
        step = step_size / (1 + beta)
        h_new = gradient_step(h, step, beta)
        c5_new = compute_c5(h_new)
        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new
        h = h_new
    return h_best, c5_best


def local_refinement(h, time_budget):

    start = time.time()
    dx = 2.0 / len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    while time.time() - start < time_budget * 0.9:
        # Find current worst shift
        corr = np.correlate(best_h, 1 - best_h, mode="full")
        max_idx = np.argmax(corr)
        shift = max_idx - (len(best_h) - 1)
        # Identify the points involved in the worst interaction
        # Now make small local moves
        max_tries = 1000
        for _ in range(max_tries):
            if time.time() - start > time_budget * 0.9:
                break
            i1 = random.randint(0, len(best_h) - 1)
            i2 = random.randint(0, len(best_h) - 1)
            if i1 == i2:
                continue
            # Try a small mass transfer
            delta = random.uniform(-0.01, 0.01)
            if best_h[i1] + delta < 0 or best_h[i1] + delta > 1:
                continue
            # Find j that corresponds to i1 + shift
            j = i1 + shift
            if j < 0 or j >= len(best_h):
                continue
            # Ensure valid transfer to j
            if best_h[i2] - delta < 0 or best_h[i2] - delta > 1:
                continue
            new_h = best_h.copy()
            new_h[i1] += delta
            new_h[i2] -= delta
            new_h = project_onto_feasible(new_h, dx)
            c5_new = compute_c5(new_h)
            if c5_new < best_c5:
                best_h = new_h.copy()
                best_c5 = c5_new
                break

    seeds.append(("binary", h_binary))
    
    # 4. Random continuous
