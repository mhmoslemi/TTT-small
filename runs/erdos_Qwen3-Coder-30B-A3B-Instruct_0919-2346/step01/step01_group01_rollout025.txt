import numpy as np
import time
import random
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import cdist

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using full convolution
    overlap = np.correlate(h, 1 - h, mode='full')
    # The maximum value scaled by dx
    c5 = np.max(overlap) * dx
    return c5

def propose_move(h, max_step):
    """Propose a move that maintains sum constraint exactly"""
    h_new = h.copy()
    N = len(h)
    while True:
        i, j = random.sample(range(N), 2)
        # Compute feasible range for delta
        delta_low = max(-h_new[i], h_new[j] - 1.0)
        delta_high = min(1.0 - h_new[i], h_new[j])
        if delta_low >= delta_high:
            continue  # No valid move, try again
        break
    
    # Sample delta from feasible interval
    step = min(max_step, delta_high - delta_low)
    if step <= 0:
        delta = 0.0
    else:
        delta = random.uniform(max(-step, delta_low), min(step, delta_high))
        
    h_new[i] += delta
    h_new[j] -= delta
    return h_new

def initial_h_vals(n_points):
    """Generate good initial points based on known construction"""
    x = np.linspace(0, 2, n_points)
    # Modified version of initial construction with better distribution
    h = np.zeros(n_points)
    
    # Core idea: triple peak construction
    # Peaks at 0.3, 1.0, and 1.7 with normalization
    peak1 = np.exp(-10 * (x - 0.3)**2)
    peak2 = np.exp(-10 * (x - 1.0)**2)  
    peak3 = np.exp(-10 * (x - 1.7)**2)
    h = 0.5 * (peak1 + peak2 + peak3)
    
    # Normalize to integral = 1
    h = h * (2.0 / np.sum(h) / n_points)
    
    # Make sure bounds are respected
    h = np.clip(h, 0, 1)
    return h

def local_search(h0, budget_s):
    """Improve solution with local search techniques"""
    start = time.time()
    h = h0.copy()
    c_current = c5_of(h)
    
    # Different step sizes to try
    step_sizes = [0.1, 0.05, 0.02, 0.01]
    
    # Iteratively improve
    for i in range(max(1000, int(100000 * budget_s / 600))):
        if time.time() - start > budget_s - 1:
            break
            
        # Try small random moves
        h_new = h.copy()
        idx = random.randint(0, len(h)-1)
        # Perturb a single point a bit
        delta = random.uniform(-0.1, 0.1)
        new_val = h[idx] + delta
        # Respect bounds
        new_val = max(0, min(1, new_val))
        h_new[idx] = new_val
        
        # If we can improve, accept it
        c_new = c5_of(h_new)
        if c_new < c_current:
            h = h_new
            c_current = c_new
        else:
            # Sometimes accept worse moves to escape from local optima
            # Use very low probability (1e-4) 
            if random.random() < 1e-4:
                h = h_new
                c_current = c_new
                
    return h, c_current

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    np.seterr(divide='ignore', invalid='ignore')
    
    # Try fewer points to be faster
    n_points = 120
    dx = 2.0 / n_points
    
    # Generate multiple different initial strategies 
    initial_methods = []
    
    # Method 1: Modified version of the previous solution
    base = initial_h_vals(n_points)
    initial_methods.append(base.copy())
    
    # Method 2: Two-spike pattern
    h2 = np.ones(n_points) * 0.2
    x = np.linspace(0, 2, n_points)
    h2[(x >= 0.3) & (x <= 0.7)] = 0.8
    h2[(x >= 1.3) & (x <= 1.7)] = 0.8
    h2 = h2 * (2.0 / np.sum(h2) / n_points)
    h2 = np.clip(h2, 0, 1) 
    initial_methods.append(h2)
    
    # Method 3: Known good general construction
    x = np.linspace(0, 2, n_points)
    h3 = np.ones_like(x) * 0.1
    # Add high regions
    for a, b, val in [(0.4, 0.6, 0.9), (0.9, 1.1, 0.85), (1.4, 1.6, 0.9)]:
        mask = (x >= a) & (x <= b)
        h3[mask] = val
    h3 = h3 * (2.0 / np.sum(h3) / n_points)
    h3 = np.clip(h3, 0, 1)
    initial_methods.append(h3)
    
    # Try different starting configurations
    best_h = None
    best_c5 = float('inf')
    
    # Split initial budget among methods
    budget_per_method = budget_s / len(initial_methods) 
    
    for i, h_start in enumerate(initial_methods):
        # Perform local search on each
        h_improved, c_improved = local_search(h_start, budget_per_method)
        
        if c_improved < best_c5:
            best_h = h_improved.copy()
            best_c5 = c_improved
            
    # A bit more refinement at the end
    refinements = 0
    while time.time() - start < budget_s - 1 and refinements < 100:
        h_prop = best_h.copy()
        idx = random.randint(0, len(best_h)-1)
        delta = random.uniform(-0.05, 0.05)
        new_val = best_h[idx] + delta
        new_val = max(0, min(1, new_val))
        h_prop[idx] = new_val
        c_prop = c5_of(h_prop)
        if c_prop < best_c5:
            best_h = h_prop
            best_c5 = c_prop
        refinements += 1
    
    return (best_h, best_c5, n_points)
