```python
import numpy as np
import time
import cvxpy as cp
from cvxpy import ECOS, GLPK

def water_fill(h, target_sum):
    """Project h onto the simplex with sum = target_sum using binary search for lambda."""
    h = np.asarray(h)
    if np.isclose(np.sum(h), target_sum):
        return h.copy()
    
    # Binary search for lambda
    low, high = h.min() - 1, h.max() + 1
    tol = 1e-10

    while high - low > tol:
        lam = (low + high) / 2
        z = np.clip(h - lam, 0, 1)
        s = np.sum(z)
        if s > target_sum:
            low = lam
        else:
            high = lam
    lam = (low + high) / 2
    z = np.clip(h - lam, 0, 1)
    return z

def evaluate_c5(h):
    """Compute C5 = max_s ∫ h(x)(1-h(x+s)) dx """
    dx = 2.0 / len(h)
    c = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(c) * dx
    return c5

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    N = 140
    dx = 2.0 / N

    # --- Phase 0: Setup index mappings ---
    # Build all pair indices (i, s) for valid correlations
    pairs = []
    pair_to_index = {}
    idx = 0
    for s in range(N):  # For s = 0 to N-1 (non-negative shift)
        for i in range(N - s):
            pairs.append((i, s))
            pair_to_index[(i, s)] = idx
            idx += 1
    num_pairs = len(pairs)

    # Prepare indices for faster constraint construction
    shift_indices = {}  # For a given shift s, list of (i, idx) pairs
    for i, s in pairs:
        if s not in shift_indices:
            shift_indices[s] = []
        shift_indices[s].append((i, pair_to_index[(i, s)]))

    # --- Phase 1: Solve LP relaxation ---
    print("Starting LP solve...")
    # Variables
    h = cp.Variable(N, name="h")
    y = cp.Variable(num_pairs, name="y")  # indexed by pairs [(i,s)]
    alpha = cp.Variable(name="alpha")

    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == N / 2,
    ]

    # McCormick envelopes for y[i,s] = h[i]*h[i+s]
    for i, s in pairs:
        idx = pair_to_index[(i, s)]
        constraints.append(y[idx] <= h[i])
        constraints.append(y[idx] <= h[i + s])
        constraints.append(y[idx] >= h[i] + h[i + s] - 1)
        constraints.append(y[idx] >= 0)

    # Max overlap condition: alpha >= dx*(N/2 - sum y[i,s])
    for s in range(N):
        sum_y_s = cp.sum([y[pair_to_index[(i, s)]] for i in range(N - s)])
        constraints.append(alpha >= dx * (N / 2 - sum_y_s))

    # Objective
    objective = cp.Minimize(alpha)
    prob = cp.Problem(objective, constraints)

    # Solve
    time_remaining = budget_s - (time.time() - start_time)
    if time_remaining < 30:
        # Very little time left, just return fallback
        h_result = np.ones(N)*0.5
        h_result = water_fill(h_result, N/2)
        return (h_result, evaluate_c5(h_result), N)
    
    try:
        prob.solve(solver=ECOS, max_iters=10000, verbose=False)
        print(f"LP solve ended with status: {prob.status}")
        if prob.status in ["optimal", "optimal_inaccurate"]:
            h_val = h.value
            alpha_val = alpha.value
            print(f"LP solution: α = {alpha_val:.6f}")
        else:
            h_val = np.ones(N)*0.5
            h_val = water_fill(h_val, N/2)
            return (h_val, evaluate_c5(h_val), N)
    except Exception as e:
        print("LP solve failed:", e)
        h_val = np.ones(N)*0.5
        h_val = water_fill(h_val, N/2)
        return (h_val, evaluate_c5(h_val), N)

    c5_lp = evaluate_c5(h_val)
    if c5_lp <= 0.3807:
        h_result = water_fill(h_val, N/2)
        return (h_result, c5_lp, N)

    # --- Phase 2: Binary MILP ---
    print("Starting MILP solve...")
    time_remaining = budget_s - (time.time() - start_time)
    if time_remaining < 30:
        h_result = water_fill(h_val, N/2)
        return (h_result, c5_lp, N)
    
    # --- Convert to MILP with integer h ---
    h_bin = cp.Variable(N, integer=True, name="h_bin")
    y_bin = cp.Variable(num_pairs, name="y_bin")
    alpha_bin = cp.Variable(name="alpha_bin")

    # Constraints
    constraints_bin = [
        h_bin >= 0,
        h_bin <= 1,
        cp.sum(h_bin) == N / 2,
    ]

    # McCormick envelopes for y[i,s] = h[i]*h[i+s]
    for i, s in pairs:
        idx = pair_to_index[(i, s)]
        constraints_bin.append(y_bin[idx] <= h_bin[i])
        constraints_bin.append(y_bin[idx] <= h_bin[i + s])
        constraints_bin.append(y_bin[idx] >= h_bin[i] + h_bin[i + s] - 1)
        constraints_bin.append(y_bin[idx] >= 0)

    # Max overlap condition: alpha >= dx*(N/2 - sum y[i,s])
    for s in range(N):
        sum_y_s = cp.sum([y_bin[pair_to_index[(i, s)]] for i in range(N - s)])
        constraints_bin.append(alpha_bin >= dx * (N / 2 - sum_y_s))

    # Objective
    objective_bin = cp.Minimize(alpha_bin)
    prob_bin = cp.Problem(objective_bin, constraints_bin)

    try:
        # Set time limit (use solver-specific option if possible)
        # cvxpy doesn't allow maxTime directly in ECOS; set max_iters to bound
        prob_bin.solve(solver=GLPK, verbose=False)
        print(f"MILP solve ended with status: {prob_bin.status}")
        if prob_bin.status in ["optimal", "optimal_inaccurate"]:
            h_bin_val = h_bin.value
            alpha_bin_val = alpha_bin.value
            print(f"MILP solution: α = {alpha_bin_val:.6f}")
        else:
            h_bin_val = h_val  # fallback to LP
    except Exception as e:
        print("MILP solve failed:", e)
        h_bin_val = h_val

    h_feasible = water_fill(h_bin_val, N/2)
    c5_milp = evaluate_c5(h_feasible)

    # --- Phase 3: Refinement on the MILP solution ---
    print("Starting refinement...")
    time_remaining = budget_s - (time.time() - start_time)
    if time_remaining < 10:
        return (h_feasible, c5_milp, N)

    # Simple deterministic hill-climb
    h_refined = h_feasible.copy()
    best_c5 = c5_milp

    for iter in range(500):
        # Compute all C_s for current h
        c_values = []
        for s in range(N):
            shift_slice = np.roll(h_refined, -s)
            c_s = dx * np.sum(h_refined * (1 - shift_slice))
            c_values.append(c_s)
        worst_s = np.argmax(c_values)
        worst_c = c_values[worst_s]
        
        # Gradient estimate for worst C_s
        if worst_c > best_c5:
            break  # We're done
            
        # Compute gradient approximating ∂C_s/∂h
        grad = np.zeros(N)
        for i in range(N):
            # Approximate derivative by finite diff
            h_plus = h_refined.copy()
            h_plus[i] = max(0, min(1, h_refined[i] + 0.02))  # move slightly
            c_plus = dx * np.sum(h_plus * (1 - np.roll(h_plus, -worst_s)))
            
            h_minus = h_refined.copy()
            h_minus[i] = max(0, min(1, h_refined[i] - 0.02))
            c_minus = dx * np.sum(h_minus * (1 - np.roll(h_minus, -worst_s)))
            
            grad[i] = (c_plus - c_minus) / 0.04

        # Move in opposite direction of gradient
        new_h = h_refined - 0.02 * grad
        new_h = np.clip(new_h, 0, 1)
        new_h = water_fill(new_h, N/2)
        
        # Accept if improves
        new_c5 = evaluate_c5(new_h)
        if new_c5 < best_c5:
            h_refined = new_h
            best_c5 = new_c5
            if best_c5 <= 0.3807:
                break

    return (h_refined, best_c5, N)
```
