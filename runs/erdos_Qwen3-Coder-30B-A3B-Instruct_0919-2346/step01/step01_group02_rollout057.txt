import numpy as np
import time
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import pdist, squareform

def c5_of(h):
    """Compute the maximum overlap integral for the given step function h."""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    c = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(c) * dx
    return c5

def propose_move(h, max_step):
    """Propose a new feasible step function by adjusting two entries."""
    N = len(h)
    while True:
        i, j = np.random.choice(N, size=2, replace=False)
        # Determine feasible step
        delta_low = max(-h[i], h[j] - 1)
        delta_high = min(1 - h[i], h[j])
        if delta_low >= delta_high:
            continue  # Try another pair
        # Sample step size
        step_size = np.random.uniform(max(-max_step, delta_low),
                                      min(max_step, delta_high))
        # Create new h
        h_new = h.copy()
        h_new[i] += step_size
        h_new[j] -= step_size
        return h_new

def anneal(initial_h, time_budget, T_start=0.05, T_end=1e-6, step_start=0.2):
    """Perform simulated annealing to minimize C5."""
    start = time.time()
    cur = initial_h.copy()
    c_cur = c5_of(cur)
    best = cur.copy()
    c_best = c_cur
    T = T_start
    step = step_start
    while time.time() - start < time_budget:
        # Update temperature and step size
        elapsed = time.time() - start
        frac = elapsed / time_budget
        T = T_start * (T_end / T_start) ** frac
        step = step_start * (1.0 - frac)
        # Make move
        h_new = propose_move(cur, step)
        c_new = c5_of(h_new)
        # Accept or reject
        if c_new <= c_cur or np.random.rand() < np.exp(-(c_new - c_cur) / T):
            cur = h_new
            c_cur = c_new
            if c_new < c_best:
                best = cur.copy()
                c_best = c_new
    return best, c_best

def smooth_proximity_constraint(h, target_h, alpha=0.1):
    """Penalty for deviating from a target shape."""
    return alpha * np.sum((h - target_h)**2)

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    N = 140
    dx = 2.0 / N
    
    # Load initial construction if available
    if 'initial_h_values' in globals():

        h0 = h0 * (N/2) / np.sum(h0)  # Renormalize exactly
    else:
        # High-quality starting seed based on best known constructions
        h0 = np.ones(N) * 0.5
        x_vals = np.linspace(0, 2, N)
        # Create a structured shape: smooth ramp up, spike in middle, ramp down
        # Ensure it sums to 1 exactly on the discretized grid
        a = 0.3
        b = 0.7
        c = 1.0
        h0 = np.zeros(N)
        for i in range(N):
            x = x_vals[i]
            if x <= a:
                h0[i] = 0.3 + 0.4 * (x / a)  # ramp up
            elif x >= 2 - a:
                h0[i] = 0.3 + 0.4 * ((2 - x) / a)  # ramp down
            else:
                h0[i] = 0.7 + 0.3 * np.cos(np.pi * (x - a) / (b - a))
        h0 = h0 * (N/2) / np.sum(h0)  # Renormalize to sum = 1
    
    # More global optimization approach
    start_time = time.time()
    
    # Try to get a better seed via differential evolution
    def objective(h):
        h = np.clip(h, 0, 1)
        # Renormalize to sum = 1
        h = h * (N/2) / np.sum(h)
        return c5_of(h)
    
    # Define bounds (0 to 1)
    bounds = [(0, 1) for _ in range(N)]
    
    # Use DE for global search

    return (final_h, final_c5, N)
