import numpy as np
import time

def compute_c5_and_worst_shift(h):
    """Compute C5 and the index of the worst shift."""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    max_idx = np.argmax(corr)
    s = max_idx - (len(h) - 1)
    c5 = corr[max_idx]
    return c5, s

def compute_gradient(h, shift):
    """Compute gradient of C5 for the given shift."""
    N = len(h)
    dx = 2.0 / N
    g = np.full(N, 1.0) * dx
    if shift >= 0:
        if shift < N:
            g[:N-shift] -= h[shift:] * dx
            g[shift:] -= h[:N-shift] * dx
    else:
        shift = -shift
        if shift < N:
            g[shift:] -= h[:N-shift] * dx
            g[:N-shift] -= h[shift:] * dx
    return g

def project_onto_bounds_and_sum(v, target):
    """Project vector onto constraints: 0 <= x <= 1, sum = target."""
    # Binary search for lambda
    lower = -np.max(v)
    upper = 1 - np.min(v)
    
    for _ in range(60):
        mid = (lower + upper) / 2.0
        w = np.clip(v + mid, 0, 1)
        s = np.sum(w)
        if s < target:
            lower = mid
        else:
            upper = mid
    
    w = np.clip(v + (lower + upper) / 2.0, 0, 1)
    
    # Adjust to hit exact sum
    diff = target - np.sum(w)
    if abs(diff) > 1e-12:
        free_idx = (w > 0) & (w < 1)
        if np.any(free_idx):
            w[free_idx] += diff / np.sum(free_idx)
            w = np.clip(w, 0, 1)
        else:
            w += diff / len(w)
            w = np.clip(w, 0, 1)
    
    return w

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    N = 140
    dx = 2.0 / N
    
    # Generate two-peak initial shape
    h0 = np.ones(N)
    x_vals = np.linspace(0, 2, N)
    a = 0.3
    u = 0.9
    v = 0.1
    mask = (x_vals < a) | (x_vals >= 2 - a)
    h0[mask] = u
    h0[~mask] = v
    h0 = h0 * (N/2) / np.sum(h0)  # Renormalize exactly
    
    # Generate a random uniform shape
    h1 = np.random.rand(N)
    h1 = h1 * (N/2) / np.sum(h1)
    
    # Generate a binary shape
    h2 = np.zeros(N)
    idx = np.random.choice(N, N//2, replace=False)
    h2[idx] = 1.0
    
    # Start with best of seeds
    candidates = [h0, h1, h2]
    best_h = None
    best_score = float('inf')
    
    for h_seed in candidates:
        score = compute_c5_and_worst_shift(h_seed)[0]
        if score < best_score:
            best_h = h_seed.copy()
            best_score = score
    
    start = time.time()
    iterations = 0
    
    while time.time() - start < 0.9 * budget_s:
        # Set learning rate schedule
        t_frac = (time.time() - start) / (0.9 * budget_s)
        step_size = 10 * (1 - t_frac) + 0.1

        # Compute worst shift and gradient
        c5, worst_shift = compute_c5_and_worst_shift(best_h)
        grad = compute_gradient(best_h, worst_shift)
        
        # Update step
        v = best_h - step_size * grad
        
        # Project and re-evaluate
        new_h = project_onto_bounds_and_sum(v, N/2)
        new_score = compute_c5_and_worst_shift(new_h)[0]
        
        # Accept if better, or probabilistically if worse
        if new_score < best_score:
            best_h = new_h.copy()
            best_score = new_score
        elif np.random.rand() < np.exp(-(new_score - best_score) / 0.1):
            best_h = new_h.copy()
            best_score = new_score
    
    # Final hill climbing to polish
    for _ in range(1000):
        if time.time() - start > 0.95 * budget_s:
            break
            
        # Random move
        i, j = np.random.choice(N, size=2, replace=False)
        delta_low = max(-best_h[i], best_h[j] - 1)
        delta_high = min(1 - best_h[i], best_h[j])
        if delta_low >= delta_high:
            continue
        step = np.random.uniform(-0.05, 0.05)
        step = np.clip(step, delta_low, delta_high)
        
        # Apply move
        old_h = best_h.copy()
        best_h[i] += step
        best_h[j] -= step
        best_h = project_onto_bounds_and_sum(best_h, N/2)
        
        # Evaluate
        new_score = compute_c5_and_worst_shift(best_h)[0]
        if new_score >= best_score:
            best_h = old_h.copy()
    
    return (best_h, best_score, N)
