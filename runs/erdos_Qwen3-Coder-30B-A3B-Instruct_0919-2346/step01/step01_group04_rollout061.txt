import numpy as np
import time
import random
from scipy import signal


def compute_c5(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    overlap = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(overlap) * dx
    return c5


def project_h(h, dx):
    """Project h onto feasible set: 0 <= h[i] <= 1 and sum(h)*dx = 1"""
    # Ensure bounds
    h = np.clip(h, 0, 1)
    # Normalize
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / h_sum) * (2.0 / dx)
    return h


def gradient_step(h, step_size, beta=10.0):
    """Compute gradient and perform projected step using soft maximum"""
    n = len(h)
    dx = 2.0 / n
    
    # Precompute the convolution for overlap
    # We want to compute the derivative of int h(x)(1 - h(x + k)) dx
    # For each shift k, this is: \int h(x)(-h'(x+k)) dx = -\int h(x) h'(x+k) dx
    # For the max over k, we use soft max for gradient estimation
    
    # Compute all correlations
    h1 = np.concatenate([h, np.zeros(n)])
    h2 = np.concatenate([1 - h, np.zeros(n)]) 
    conv_full = signal.convolve(h1, h2, mode='full')
    c5_vals = conv_full[n-1:2*n-1] * dx  # Corresponds to k = 0 to 2
    
    # Soft max to approximate gradient direction
    c5_vals = c5_vals * beta
    exp_c5 = np.exp(c5_vals - np.max(c5_vals))
    weights = exp_c5 / np.sum(exp_c5)
    
    # Now compute what gradient would be if we tweak
    # A simpler approach is to construct the explicit gradient of a single shifted correlation:
    # For each k, we have:
    # I_k = int h(x)(1 - h(x+k)) dx = int h(x) dx - int h(x)h(x+k) dx
    # So dI_k/dh(x) = 1 - h(x+k) - h(x) (using delta function approximations, etc.)
    # But this is complex. Let's use finite difference to compute pseudo-gradients.
    # That aggressive derivative calculation isn't stable either.
    
    # Simpler approach: 
    # Compute rough gradient by shifting and computing overlap differences
    # Find the worst k
    worst_k = np.argmax(c5_vals)
    # Construct a suggestion gradient:
    # For the worst correlation (k = worst_k), compute -D_(h(k)) h(x) * (1 - h(x + k)) 
    # By linearity, if we view overlap as sum of terms over x: h(x)(1 - h(x + k))
    # d/dh(x) [h(x)(1 - h(x + k))] = (1 - h(x + k)) if x = x, else -h(x) if x = x + k
    # Tilde idea:
    # Construct a discrete 'gradient' where each element gets contribution from corresponding h
    # Take derivative w.r.t. h_i: coeff = (1 - h_{i+k}) - (1 - h_{i-k})

    # problematic shift in the worst case, with smoothness.
    
    # For cleaner logic, use finite differences on a directional subspace.
    
    # Define direction vector based on the core quantity we think dominates:
    # H is shifted by worst_k, so we build gradient by perturbing h at fixed locations
    # to affect worst shift.
    
    # Identify where h(x+k) is maximized or minimized for the k causing max overlap.
    # Approach: For best h_i, we want to reduce h_i * h_{i+k} on both sides.
    # And reduce 1 * h_{i+k}, so h_i = h_i * (-h_{i+k}) should be one component.

    # Approach: Set up a vector that computes for each x in h that matches (1 - h(x+k)) * (potential)

    k_index = worst_k  # This means shift k = 2*(k_index)/(n-1) is worst.

    shift = int(2 * k_index / (n - 1) * n)  # Roundtrip shift in samples
    if shift >= 0 and shift < n:
        for i in range(n):
            start = (i - shift) % n
            h_grad[i] += 1 - h[start]  # Soft gradient factor
            # Another contribution could be negative -h[i + shift] where proper modulates
            # So something like: 1 * (1 - h[i+shift]) + shift weight

    # Work from simple idea: dummy gradient direction
    # If k is worst, make h(x) scale down where h(x+k) is high, and vice versa  
    # Or del change l1h(x) * (1 - h(x+k)) = (1-h(x+k))''' 
    # The best move is to determine gradient based on current worst shift.
    
    # Simpler actually 
    # If for x, the 'correlation' h(x) * (1-h(x+k)) is large,
    # it means both h(x) > 0 and h(x+k) < 1.
    # If we reduce x, we reduce term. If we increase x, we increase term. 
    # The derivative being approximated is really what shift link gives the most “gain”.
    
    # So revise approach: compute the gradient more explicitly:
    # enumerate all k’s, and compute:
    # Build a kind of gradient at each i by counting how much contribution it makes to shift k_times that
    # Which is a sum over all x, how much the product (h(x)(1 - h(x+k))) changes w.r.t to h_i
    # But it's > for x = i, x = i+k, (for now we won’t include derivative of integrand).
    
    return h_grad


def update_projection(h, grad, alpha):
    """Update and project step"""
    # proposed = h - alpha * grad
    proposed = h - alpha * grad
    return project_h(proposed, dx=2.0 / len(h))


def smooth_optimize(initial_h, time_budget_s, alpha_init=0.2, beta_s=10.0):
    """Smooth gradient-based optimization"""
    start = time.time()
    h = initial_h.copy()
    c5_current = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_current
    
    # Adaptive step size
    alpha = alpha_init
    n = len(h)
    step_size = alpha
    beta = beta_s

    max_iter = 50000
    for i in range(max_iter):
        if time.time() - start > time_budget_s:
            break
            
        # Improvement: Use soft max with cool-beta for better convergence.
        # Beta function increases to zero if desired, in time
        elapsed = (time.time() - start) / time_budget_s
        beta = beta_s * np.exp(-elapsed * 5.0)
        if beta < 1e-5:
            beta = 1e-5
        # Compute approximate gradient.
        grad = gradient_step(h, step_size, beta)
        # Update step with projection.
        h_new = update_projection(h, grad, step_size)
        c5_new = compute_c5(h_new)
        
        # Accept if improves
        if c5_new <= c5_current:
            h = h_new
            c5_current = c5_new
            if c5_new < best_c5:
                best_h = h.copy()
                best_c5 = c5_new
        else:
            # Less aggressive acceptance for smoothness
            step_size *= 0.999
            if step_size < 1e-9:
                break
    
    return best_h, best_c5

def fine_tune_move(h_in, time_budget_s):

    start = time.time()
