import numpy as np
import time
import random
from scipy.optimize import dual_annealing

def c5_of(h):
    """Compute C5 = max_k integral h(x)(1-h(x+k)) dx."""
    dx = 2.0 / len(h)
    # Compute correlation: h * (1 - h) with full mode
    c = np.correlate(h, 1 - h, mode="full")
    # Multiply by dx to get continuous integral
    c *= dx
    # Return maximum value among all shifts
    return np.max(c)

def smooth_h(h, smoothing=0.05):
    """Apply smoothing to reduce sharp transitions."""
    h_smooth = h.copy()
    n = len(h)
    for i in range(n):
        neighbors = []
        for di in [-1, 0, 1]:
            j = (i + di) % n
            neighbors.append(h[j])
        h_smooth[i] = np.mean(neighbors) * (1 - smoothing) + h[i] * smoothing
    return np.clip(h_smooth, 0, 1)

def project_to_constraint(h):
    """Project h to satisfy the constraint sum(h) * dx = 1."""
    dx = 2.0 / len(h)
    current_sum = np.sum(h)
    target_sum = 1.0 / dx  # N/2
    return h * (target_sum / current_sum)

def sample_h(n_points):
    """Create candidate solutions by interpolating precomputed values."""
    # Use the same interpolation as in the previous code
    x_full = np.linspace(0, 2, len(initial_h_values))
    x_sample = np.linspace(0, 2, n_points)
    h = np.interp(x_sample, x_full, initial_h_values)
    # Re-normalize
    h = project_to_constraint(h)
    return h

def sample_h_random(n_points):
    """Create a random feasible solution."""
    h = np.random.uniform(0, 1, n_points)
    h = project_to_constraint(h)
    return h

def sample_h_sine(n_points):
    """Create a smooth sine-like solution."""
    x = np.linspace(0, 2, n_points)
    h = 0.5 + 0.3 * np.sin(np.pi * x)
    h = project_to_constraint(h)
    return h

def local_search(h, max_iter=1000):
    """Perform local search around a solution."""
    best_h = h.copy()
    best_c5 = c5_of(best_h)
    
    dx = 2.0 / len(h)
    for _ in range(max_iter):
        # Perturb two points
        i, j = random.sample(range(len(h)), 2)
        delta = random.uniform(-0.05, 0.05)
        
        # Ensure feasibility
        new_h = best_h.copy()
        if new_h[i] + delta < 0:
            delta = -new_h[i]
        elif new_h[i] + delta > 1:
            delta = 1 - new_h[i]
        
        if new_h[j] - delta < 0:
            delta = -new_h[j]
        elif new_h[j] - delta > 1:
            delta = new_h[j] - 1
            
        new_h[i] += delta
        new_h[j] -= delta
        new_h = project_to_constraint(new_h)
        
        c5 = c5_of(new_h)
        if c5 < best_c5:
            best_h = new_h
            best_c5 = c5
            
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    start_time = time.time()
    N = 200  # Increase resolution
    
    # Predefined combinations
    initial_solutions = []
    
    # 1. Interpolated from initial values
    sample_h_interp = sample_h(N)
    initial_solutions.append(sample_h_interp)
    
    # 2. Random feasible
    sample_h_rand = sample_h_random(N)
    initial_solutions.append(sample_h_rand)
    
    # 3. Sine-like
    sample_h_sin = sample_h_sine(N)
    initial_solutions.append(sample_h_sin)
    
    # 4. Heuristic construction
    h_heuristic = np.ones(N) * 0.5
    # Place a spike in the middle
    center = N // 2
    h_heuristic[center] = 1.0
    # Distribute around it
    for i in range(N):
        dist = min(abs(i - center), abs(i + N - center), abs(i - N + center))
        if dist < 20:
            h_heuristic[i] = max(0, min(1, 1.0 + (dist - 20) * 0.02))
    # Renormalize
    h_heuristic = project_to_constraint(h_heuristic)
    initial_solutions.append(h_heuristic)
    
    best_c5 = float('inf')
    best_h = None
    
    # Try each initial solution
    for i, h_seed in enumerate(initial_solutions):
        # Local search
        h, c5 = local_search(h_seed, 500)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
            
        # Local search on refined samples
        if time.time() - start_time > budget_s * 0.7:
            break
            
        # Survey of fine-grained local improvements
        for _ in range(200):
            h_new = h.copy()
            # Mutate small subset of values
            indices = random.sample(range(N), min(5, N//4))
            for idx in indices:
                noise = random.uniform(-0.02, 0.02)
                h_new[idx] = max(0, min(1, h_new[idx] + noise))
            h_new = project_to_constraint(h_new)
            c5_new = c5_of(h_new)
            if c5_new < c5:
                c5 = c5_new
                h = h_new
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h
                    
        if time.time() - start_time > budget_s * 0.95:
            break
    
    if best_h is None:
        # Fallback to fundamental random solution
        h_fallback = np.random.random(N)
        h_fallback = project_to_constraint(h_fallback)
        best_h, best_c5 = local_search(h_fallback, 1000)
    
    return (best_h, best_c5, N)
