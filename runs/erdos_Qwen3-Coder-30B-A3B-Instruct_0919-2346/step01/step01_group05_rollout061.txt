import numpy as np
from scipy.optimize import minimize
from math import sqrt
import time

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value scaled by dx
    max_overlap = np.max(correlation) * dx
    return max_overlap

def project_onto_bounds_sum(v, target_sum, lower=0, upper=1):
    """Project vector v onto box constraints and sum constraint"""
    # Binary search for the Lagrange multiplier
    left = v.min() - upper
    right = v.max() - lower
    
    for _ in range(60):
        mid = (left + right) / 2
        projected = np.clip(v - mid, lower, upper)
        if projected.sum() > target_sum:
            left = mid
        else:
            right = mid
    
    return np.clip(v - (left + right) / 2, lower, upper)

def compute_gradient(h, f, gamma):
    """Compute gradient of the smooth approximation to max overlap"""
    n = len(h)
    dx = 2.0 / n
    # Compute softmax weights
    max_f = np.max(f)
    w = np.exp(gamma * (f - max_f))
    w = w / np.sum(w)
    
    # Compute gradient
    g = np.zeros(n)
    # For each shift, compute contribution to gradient
    for l in range(-n+1, n):
        if l >= 0:
            # Compute terms for positive shifts
            start = 0
            end = n - l
        else:
            start = -l
            end = n
        # l is shift; corresponding indices for shift l are [start, end) and [start+l, end+l)
        if end > start:
            # Initialize gradient to zero
            g[start:end] += w[l + n - 1] * (1 - h[start+l:end+l])  # from 1-h[i+l]
            g[start:end] -= w[l + n - 1] * h[start:end]  # from -h[i-l] 
    return g * dx
