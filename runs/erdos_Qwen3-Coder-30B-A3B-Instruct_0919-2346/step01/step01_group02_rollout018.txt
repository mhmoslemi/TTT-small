import numpy as np
import cvxpy as cp
import time

def evaluate_c5(h):
    """Compute the maximum overlap integral C5 for the given step function h."""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    c = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(c) * dx
    return c5

def project_to_sum_constraint(h, target_sum):
    """Project h onto the constraint sum(h) = target_sum using water filling."""
    # Sort in descending order
    sorted_h = np.sort(h)[::-1]
    # Find cutoff index
    cumsum = np.cumsum(sorted_h)
    n = np.searchsorted(cumsum, target_sum, side='right')
    if n == 0:
        return np.zeros_like(h)
    else:
        # Calculate lambda
        lambda_val = (target_sum - cumsum[n-1]) / n
        # Clamp to [0,1]
        result = np.maximum(0, np.minimum(1, sorted_h - lambda_val))
        # Restore original order
        h_proj = h.copy()
        idx = np.argsort(h)[::-1]
        h_proj[idx] = result
        return h_proj

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    N = 140
    dx = 2.0 / N
    start_time = time.time()
    
    # Build list of valid (i,s) pairs
    pairs = []
    for s in range(N):
        for i in range(N - s):
            pairs.append((i, s))
    
    # Create variables
    h = cp.Variable(N, name='h')
    y = cp.Variable(len(pairs), name='y')
    alpha = cp.Variable(1, name='alpha')
    
    # Build constraints
    constraints = []
    
    # Box constraints
    constraints += [h[i] >= 0 for i in range(N)]
    constraints += [h[i] <= 1 for i in range(N)]
    
    # Total mass constraint
    constraints += [cp.sum(h) == N/2]
    
    # McCormick envelopes for products
    for k, (i, s) in enumerate(pairs):
        # y[i,s] <= h[i]
        constraints += [y[k] <= h[i]]
        # y[i,s] <= h[i+s]
        constraints += [y[k] <= h[i+s]]
        # y[i,s] >= h[i] + h[i+s] - 1
        constraints += [y[k] >= h[i] + h[i+s] - 1]
        # y[i,s] >= 0
        constraints += [y[k] >= 0]
    
    # Alpha constraints
    # Actually, we need to do it for all shift indices, but let's do it more carefully.
    # We compute sum of y for each valid shift s.
    # Shift s runs from -(N-1) to (N-1). In our full correlation, all shifts are represented via the 'full' mode.
    # Let's extract the diagonal sums of correlation for shift s starting from 0 to N-1 (via the corr)
    # We use numpy conventions for indexing of correlations in np.correlate
    # Now for shift s = 0 to N-1, we sum the y for i from 0 to N-1-s.
    # So fix a shift s, the corresponding sum_y = Sum_{i=0}^{N-1-s} y[i,s]
    # And the constraint is alpha >= dx * (N/2 - sum_y)
    # So we have to map (i,s) to the corresponding y index.
    # But in numpy correlation indexing, for shift s (negative is left, positive is right):
    # correlate(h, 1-h, mode='full') has length 2*N-1, and is indexed from -(N-1) to N-1.
    # In the to_compare numpy array, with mode='full', we look at how i and i+s are aligned.
    # For a given i (from 0 to N-1), s from 0 to N-1 (ala data from full), it's i+s for shift s.
    # So for full correlation, for shift s=0,...,N-1, valid i from 0 to N-1-s:
    for s in range(N):
        # We collect all the y terms for that shift
        # i from 0 to N-1-s inclusive
        shift_y_indices = []
        for i in range(N - s):
            # Find the index of pair (i,s) in our list
            try:
                idx = pairs.index((i, s))
                shift_y_indices.append(idx)
            except ValueError:
                pass
        if shift_y_indices:
            sum_y = cp.sum(y[shift_y_indices])
            # C_s = dx * (N/2 - sum_y) <= alpha
            constraints += [alpha >= dx * (N/2 - sum_y)]
    
    # Objective
    objective = cp.Minimize(alpha)
    
    # Create and solve the problem
    prob = cp.Problem(objective, constraints)
    
    # Try to solve the LP first
    try:
        prob.solve(solver='ECOS', max_iters=20000, verbose=False)
    except Exception:
        # Fallback to simple initialization
        pass

    # If problem solved, get value to check
    h_val = h.value if hasattr(h.value, '__len__') else np.ones(N) * 0.5
    if not np.all(np.isnan(h_val)):
        # Project h to sum constraint exactly
        h_val = project_to_sum_constraint(h_val, N/2)
        # Evaluate the true C5, goal is < 0.3807
        c5_val = evaluate_c5(h_val)
        if c5_val <= 0.3807:
            return (h_val, c5_val, N)
    
    # If we don't have good enough LP solution, or if it failed,
    # we try a more sophisticated approach: solving an MILP with binary.
    
    # Re-initialize variables for binary problem
    h_bin = cp.Variable(N, boolean=True, name='h_bin')
    y_bin = cp.Variable(len(pairs), name='y_bin')
    alpha_bin = cp.Variable(1, name='alpha_bin')
    
    constraints_bin = []
    
    # Box constraints for binary
    constraints_bin += [h_bin[i] >= 0 for i in range(N)]
    constraints_bin += [h_bin[i] <= 1 for i in range(N)]
    
    # Total mass constraint - with equal sum
    constraints_bin += [cp.sum(h_bin) == N/2]
    
    # McCormick constraints for binary case (exact now)
    for k, (i, s) in enumerate(pairs):
        constraints_bin += [y_bin[k] <= h_bin[i]]
        constraints_bin += [y_bin[k] <= h_bin[i+s]]
        constraints_bin += [y_bin[k] >= h_bin[i] + h_bin[i+s] - 1]
        constraints_bin += [y_bin[k] >= 0]
    
    # Alpha constraints using exact sum for each shift
    for s in range(N):
        shift_y_indices = []
        for i in range(N - s):
            try:
                idx = pairs.index((i, s))
                shift_y_indices.append(idx)
            except ValueError:
                pass
        if shift_y_indices:
            sum_y = cp.sum(y_bin[shift_y_indices])
            constraints_bin += [alpha_bin >= dx * (N/2 - sum_y)]
    
    objective_bin = cp.Minimize(alpha_bin)
    prob_bin = cp.Problem(objective_bin, constraints_bin)
    
    try:
        # Allocate remaining time for binary problem, with a timeout
        time_left = budget_s - (time.time() - start_time)
        if time_left > 10:
            # Use a time limit for solving the MILP
            solver_options = {'maxTime': time_left * 0.8}
            prob_bin.solve(solver=cp.CBC, verbose=False, **solver_options)
        else:
            # Return anyway if not enough time
            pass
    except Exception:
        pass
    
    # Get result from maybe finished problem
    h_sol = None
    if hasattr(h_bin.value, '__len__') and not np.any(np.isnan(h_bin.value)):
        # Binary solution
        h_sol = h_bin.value
    elif hasattr(h.value, '__len__') and not np.any(np.isnan(h.value)):
        # Floating point solution
        h_sol = project_to_sum_constraint(h.value, N/2)
    
    if h_sol is not None:
        c5_current = evaluate_c5(h_sol)
        if c5_current <= 0.3807:
            return (h_sol, c5_current, N)
    
    # Fallback to rounding
    if h_sol is None:
        if hasattr(h.value, '__len__') and not np.any(np.isnan(h.value)):
            h_sol = project_to_sum_constraint(h.value, N/2)
    
    if h_sol is not None:
        # Refine with a few hill-climbing steps
        c5_current = evaluate_c5(h_sol)
        c5_best = c5_current
        h_best = h_sol.copy()
        
        num_refine = min(1000, int((budget_s - (time.time() - start_time)) * 100))
        for _ in range(num_refine):
            # Select worst shift
            c = np.correlate(h_best, 1 - h_best, mode='full')
            s_star = np.argmax(c)
            # Approximate gradient (subgradient) with a finite difference
            # Rather than computing direction, we make a move to just improve it
            delta = 0.01  # Small move size
            # We create a perturbed version
            h_new = h_best.copy()
            # Try to move toward better point
            for i in range(N):
                # Define a platform of size dx = 2.0 / N
                h_new[i] = min(1, max(0, h_new[i] + delta))
                # Renormalize
                if np.sum(h_new) != N/2:
                    h_new = project_to_sum_constraint(h_new, N/2)
                # Check if this improves
                c5_new = evaluate_c5(h_new)
                if c5_new < c5_current:
                    c5_current = c5_new
                    h_best = h_new.copy()
                else:
                    h_new[i] = h_best[i]  # revert
            if c5_current < c5_best:
                c5_best = c5_current
                h_best = h_new.copy()
            # Break early if we're over budget
            if time.time() - start_time > budget_s - 2:
                break
        return (h_best, c5_best, N)
    
    # Final fallback
    h_default = np.ones(N) * 0.5
    h_default = project_to_sum_constraint(h_default, N/2)
    c5_default = evaluate_c5(h_default)
    return (h_default, c5_default, N)
