import numpy as np
import time
import random

def compute_C_vector(h):
    """Compute vector of overlaps C_tau = 1 - dx * S_tau(h) for all shifts."""
    dx = 2.0 / len(h)
    # Full autocorrelation of h with zero padding: S_tau = sum h_i h_{i+tau}
    S = np.correlate(h, h, mode='full')
    # Take the middle portion, which corresponds to shifts from -n+1 to n-1
    # But we want shifts from 0 to 2n-2, so we slice appropriately
    S = S[len(h)-1:]  # Now S[tau] = S_tau
    c = 1 - dx * S
    return c

def compute_softmax_weights(c, alpha):
    """Compute softmax weights w_tau = exp(alpha*c_tau) / sum exp(alpha*c_sigma)."""
    c_max = np.max(c)
    exp_alpha_c = np.exp(alpha * (c - c_max))
    w = exp_alpha_c / np.sum(exp_alpha_c)
    return w

def compute_gradient(h, w):
    """Compute gradient of smooth objective: g = -dx * (w * h + w_rev * h)"""
    dx = 2.0 / len(h)
    # Both convolutions are done with mode='same'
    # w is padded with zeros to correct length (len(w) = 2n - 1)
    # Use full convolution
    conv1 = np.correlate(w, h, mode='full')
    conv2 = np.correlate(np.flip(w), h, mode='full')
    
    # extract the part that corresponds to the correct indices in h
    start_idx = len(w) - len(h)
    g = -dx * (conv1[start_idx:start_idx+len(h)] + conv2[start_idx:start_idx+len(h)])
    return g

def project_to_feasible(h_project):

    n = len(h_project)
    target_sum = n / 2
    
    # Ensure h is within bounds
    h_project = np.clip(h_project, 0, 1)
    
    # Compute current sum
    current_sum = np.sum(h_project)
    
    # If already correct, return immediately
    if abs(current_sum - target_sum) < 1e-12:
        return h_project.copy()

    # Water-filling approach
    # Use bisection to find lambda such that the projection has the desired sum
    def phi(lam):

        # return sum(h_proj) - target_sum
        # To make this efficient, sort the values
        sorted_h = np.sort(h_project - lam)[::-1]  # descending order
        # We need to find where the clipping becomes active
        # Clipped values are max(0, h[i] - lam)
        # If lam >= max(h), then all values are 0
        # If lam <= min(h), then all values are h[i]
        # For intermediate lam, we compute index where clipping (h[i] - lam) <= 0
        
        # Element-wise, the projection is max(h[i] - lam, 0)
        # We sum this and see how it compares to target
        
        if lam >= np.max(h_project):
            return -target_sum
        if lam <= np.min(h_project):
            return current_sum - target_sum
            
        # Find the index k where clipping happens
        # sorted[0] to sorted[k-1] are > 0 after clipping
        # sorted[k] to sorted[n-1] are 0 after clipping    
        # So sum = sum(sorted[0:k]) = sum(sorted) - sum(sorted[k:])
        # sorted = [sorted[0] ... sorted[k-1], sorted[k] ... sorted[n-1]]
        #  k is such that sorted[k] <= 0 and sorted[k-1] > 0 (or k=0)
        # wlog assume not all values are positive (so no special handling needed)
        # The total number of elements that are (0 where smaller than 0)
        
        # The most methodical way is to go from largest to smallest and compute the sum
        # of clipped values until we either reach 0, or overstep the target
        
        cumsum = 0.0
        k = 0
        for i in range(len(sorted_h)):
            if sorted_h[i] <= lam:
                k = i
                break
            cumsum += sorted_h[i]
            k = i + 1
            
        # The projected sum is
        # sum(sorted_h[0:k]) + 0*(n-k) = cumsum
        # but only if we correctly count which truncation we applied

        # Correct sum = k*lam + sum(sorted_h[k:])
        # But we already know what that's: just the part that was not floored!
        for i in range(k, len(sorted_h)):
            cumsum += sorted_h[i]  # including the ones below lam, they're not clipped

        # = sum(h_project[i] for i such that h[i] >= lam) - (number of such i) * lam
        # = sum(sorted_h[j] for j in [0, k)) - k*lam
        # = cumsum_new =  
        # at the point of crossing when sorted_h[i] <= lam, 
        #    moving to the next i makes our index larger than such, 
        #    so we want: sum(getting clipped) = sum(h_project[i]) - k*lam
        #    where large ones are summed first () k = number of large elements
        # Just one approach:
        # j: 0..n-1 sorted descending
        # Initial is perfect (k=0)
        # Trial computes sum (or exact)
        total_sum = sorted_h.sum()    
        # get index where first element is <= lam
        # 0 1 2 i+1 ... n-1  
        #         |
        #         i  -> the elements at [0..i] are > lam
        # Take sum for indices: [0..i] and e.g. 1000..1000  
        # -> we care about the valid part
            
        # So we find the cutoff where sorted_h[i] < lam
        sorted_hi = sorted_h  # alias
        sum_positive = 0.
        k = 0
        for j in range(len(sorted_hi)):
            if sorted_hi[j] <= lam:
                break
            sum_positive += sorted_hi[j]
            k = j + 1
        project_sum = sum_positive - lam * k
            
        # Compute the exact one
        proj = np.maximum(h_project - lam, 0)
        actual_sum = np.sum(proj) 
        # safe check
        return project_sum - target_sum  # should be equivalent but not necessarily
                        
    # Find reasonable bounds
    min_h = np.min(h_project)  
    max_h = np.max(h_project)
    lam_low = min_h - 1.0
    lam_high = max_h + 1.0
    # Ensure right sign
    if phi(lam_low) > 0:
        lam_high = lam_low
        lam_low = lam_high - 2.0
    
    # Binary search for root of phi(lambda) = 0 (which corresponds to sum == target)
    tries = 0 
    max_tries = 1000  # avoid infinite loop
    while abs(phi((lam_low + lam_high) / 2.0)) > 1e-10 and tries < max_tries:
        mid = (lam_low + lam_high) / 2.0
        val = phi(mid)
        if val > 0:
            lam_low = mid
        else:
            lam_high = mid
        tries += 1
    
    # Final projection
    lam = (lam_low + lam_high) / 2.0
    h_proj = np.maximum(h_project - lam, 0)
    
    # Normalize: 
    current_proj_sum = np.sum(h_proj)
    h_proj = h_proj * (target_sum / current_proj_sum)
    return h_proj

def gradient_descent_phase(h_init, budget_s, n):
    """Run gradient descent with smoothing on the max-overlap objective."""
    dx = 2.0 / n
    t_start = time.time()
    
    # Initializations
    h = h_init.copy()
    best_h = h.copy()
    c = compute_C_vector(h)
    best_C5 = np.max(c)  
    # We might shadow this, but let's be precise and overwrite
    t = 0
    max_iter = 100000
    
    # Parameters
    alpha_start = 5.0
    alpha_end = 80.0
    eta_0 = 3.0  # step size scaling factor (to keep initial moves < 0.02)
    beta = 0.7  # momentum
    K_perturb = 300  # period to add noise
    
    # Initial momentum
    m = np.zeros(n)
    
    for iter in range(max_iter):
        if time.time() - t_start > budget_s * 0.99:
            break
            
        # Update alpha (warm-up schedule)
        t = iter + 1
        alpha = alpha_start + (alpha_end - alpha_start) * min(t / (max_iter * 0.3), 1.0)
        
        # Step size schedule η = η_0 / sqrt(t)
        eta = eta_0 / np.sqrt(t)
        
        # Compute overlaps
        c = compute_C_vector(h)
        C5_current = np.max(c)
        
        # Compute softmax weights
        w = compute_softmax_weights(c, alpha)
        
        # Compute gradient
        g = compute_gradient(h, w)
        
        # Accelerated step with momentum
        m = beta * m + (1 - beta) * g
        h_new = h - eta * m
        
        # Project back to feasible set
        h_new = project_to_feasible(h_new)
        
        # Evaluate current C5 (true max)
        c_new = compute_C_vector(h_new) 
        C5_new = np.max(c_new)
        
        # Take the step if it improves
        if C5_new < C5_current:
            h = h_new.copy()
            if C5_new < best_C5:
                best_C5 = C5_new
                best_h = h.copy()
        else:
            # If the step was not successful, decrease the learning rate
            eta *= 0.5
            h_new = h - eta * m  # Try with a small step
            h_new = project_to_feasible(h_new)
            c_new = compute_C_vector(h_new)
            C5_new = np.max(c_new)
            if C5_new < C5_current:
                h = h_new.copy()
                if C5_new < best_C5:
                    best_C5 = C5_new
                    best_h = h.copy()
                    
        # Random perturbation to escape local minima
        if iter > 0 and iter % K_perturb == 0:
            # Add Gaussian noise to the best solution
            noise = np.random.normal(0, 0.01, n)
            h_pert = best_h + noise
            h_pert = project_to_feasible(h_pert)
            # If it improves, update best_h
            c_test = compute_C_vector(h_pert)
            C5_test = np.max(c_test)
            if C5_test < best_C5:
                best_C5 = C5_test
                best_h = h_pert.copy()

    return best_h, best_C5

            h = h.copy()

            delta_low = max(-h[i], h[j] - 1.0)
