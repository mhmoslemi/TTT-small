import numpy as np
import time

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral via full convolution of h with (1-h)
    overlaps = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum of the full correlation vector
    # scaled by dx
    c5 = np.max(overlaps) * dx
    return c5

def full_overlap(h):
    """Compute the raw overlap correlation vector"""
    return np.correlate(h, 1 - h, mode="full")

def project_onto_P(v, total):
    """Project vector v onto box + sum constraint: 0 <= x_i <= 1, sum x_i = total"""
    # Ensure projection stays within bounds and sum constraint
    v = np.clip(v, 0, 1)
    current_sum = np.sum(v)
    if np.isclose(current_sum, total):
        return v
    
    # Bisection to find lambda
    def project_with_lambda(lam):
        x = np.clip(v - lam, 0, 1)
        return np.sum(x) - total
    
    # Bracketing
    low, high = -10.0, 10.0
    while project_with_lambda(low) > 0:
        low -= 1.0
    while project_with_lambda(high) < 0:
        high += 1.0

    # Bisection search to machine precision (approx 50 iterations)
    for _ in range(50):
        mid = (low + high) / 2
        result = project_with_lambda(mid)
        if result > 0:
            low = mid
        else:
            high = mid
    
    lambda_val = (low + high) / 2
    x = np.clip(v - lambda_val, 0, 1)
    
    # Ensure sum is exactly correct to machine precision
    diff = total - np.sum(x)
    if abs(diff) > 1e-10:
        # Adjust a small element to compensate
        idx = np.argmin(x)
        x[idx] = np.clip(x[idx] + diff, 0, 1)
    
    return x

def smooth_gradient(h, beta):
    """Compute gradient of Phi_beta(h)"""
    # Get raw overlaps
    F = full_overlap(h)
    # Compute softmax weights (subtract max for numerical stability)
    F_max = np.max(F)
    exp_term = np.exp(beta * (F - F_max))
    w = exp_term / np.sum(exp_term)
    
    # Initialize gradient
    g = np.ones_like(h)
    
    # Process each shift k
    N = len(h)
    for k in range(-N+1, N):
        wk = w[k + N - 1]  # Corresponding index in full correlation
        if k >= 0:
            # h[i] -> h[i+k] 
            # g[:N-k] -= wk * h[k:]
            g[:N-k] -= wk * h[k:]
            # g[k:] -= wk * h[:N-k]
            g[k:] -= wk * h[:N-k]
        else:
            # k < 0
            d = -k
            # g[d:] -= wk * h[:N-d]
            g[d:] -= wk * h[:N-d]
            # g[:N-d] -= wk * h[d:]
            g[:N-d] -= wk * h[d:]
    
    return g

def projected_gradient_descent(h0, time_budget, seed=42):
    """Run projected gradient descent with smooth approximation"""
    np.random.seed(seed)
    start = time.time()
    h = h0.copy()
    best_h = h.copy()
    c5_best = c5_of(h)
    N = len(h)
    dx = 2.0 / N
    total = N / 2
    
    beta = 2.0
    beta_max = 30
    beta_update_interval = 100
    gamma_beta = 1.5
    gamma_step = 0.9995
    step = 0.5
    step_min = 1e-4
    
    iteration = 0
    while time.time() - start < time_budget:
        iteration += 1
        # Compute gradient
        g = smooth_gradient(h, beta)
        # Tentative update, then project
        v = h - step * g
        h = project_onto_P(v, total)
        
        # Evaluate actual objective
        c5_current = c5_of(h)
        if c5_current < c5_best:
            c5_best = c5_current
            best_h = h.copy()
            
        # Step size decay
        step = max(step * gamma_step, step_min)
        
        # β schedule
        if iteration % beta_update_interval == 0:
            beta = min(beta * gamma_beta, beta_max)
                    
        if c5_best < 0.379:
            break
            
    return best_h, c5_best

def targeted_refinement(h, time_budget, seed=42):
    """Perform deterministic pairwise mass transfer to reduce the worst overlap"""
    np.random.seed(seed)
    start = time.time()
    h = h.copy()
    best_h = h.copy()
    c5_best = c5_of(h)
    N = len(h)
    max_iter = 200
    alpha = 1.0
    c5_old = float('inf')
    no_improvement_count = 0
    
    while time.time() - start < time_budget and no_improvement_count < max_iter:
        # Compute full overlaps
        C = full_overlap(h)
        # Find the worst shift (k) 
        d = np.argmax(C) - (N - 1)
        # Extract overlapping slices
        if d >= 0:
            hi = h[:N - d]
            hj = h[d:]
        else:
            hi = h[-d:]
            hj = h[:N - (-d)]
        # Coordinates and overlaps
        c = hi * (1 - hj)
        # Find the maximum among these products
        i_star = np.argmax(c)
        j_star = (i_star + d) if d >= 0 else (i_star - d)
        c_star = c[i_star]
        if c_star <= 0:
            break
        
        # Compute maximum feasible move
        h_i = h[i_star]
        h_j = h[j_star]
        delta = min(alpha * c_star, h_i, 1 - h_j)
        if delta <= 1e-12:
            alpha *= 0.5
            if alpha < 1e-6:
                break
            continue
            
        # Apply the transfer
        h[i_star] -= delta
        h[j_star] += delta
        
        c5_new = c5_of(h)
        if c5_new < c5_best:
            c5_best = c5_new
            best_h = h.copy()
            no_improvement_count = 0
        else:
            # Revert
            h[i_star] += delta
            h[j_star] -= delta
            no_improvement_count += 1
            
        # Reduce α if needed
        if no_improvement_count % 10 == 0 and alpha > 1e-6:
            alpha *= 0.5
            
    return best_h, c5_best
