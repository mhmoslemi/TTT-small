import numpy as np
import time
from scipy.optimize import minimize
from scipy.optimize import Bounds

def c5_of(h):
    """Compute the maximum overlap integral for the given step function h."""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    c = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(c) * dx
    return c5

def project_to_sum_constraint(h, target_sum):
    """Project h onto the hyperplane sum(h) = target_sum using water-filling."""
    v = h.copy()
    # Binary search for lambda
    lb, ub = -2.0, 2.0  # reasonable bounds
    for _ in range(50):
        lam = (lb + ub) / 2
        new_h = np.clip(v + lam, 0, 1)
        sum_new = np.sum(new_h)
        if sum_new < target_sum:
            ub = lam
        else:
            lb = lam
    # Apply the final correction to ensure exact sum
    lam = lb
    new_h = np.clip(v + lam, 0, 1)
    # Adjust to exact sum
    diff = target_sum - np.sum(new_h)
    # redistribute diff on free variables (not at bounds)
    free = np.where((new_h > 0) & (new_h < 1))[0]
    if len(free) > 0:
        to_add = diff / len(free)
        new_h[free] += to_add
        # clip to stay valid
        new_h = np.clip(new_h, 0, 1)
    return new_h

def compute_correlation(h, mode='full'):
    """Compute correlation C_s = sum h[i] * (1 - h[i+s]) for all shifts s."""
    c = np.correlate(h, 1 - h, mode=mode)
    return c

def smooth_objective(h, eps, dx):
    """Compute smooth surrogate F_eps(h) = eps * log(sum exp(C_s / eps))."""
    c = compute_correlation(h)
    c_scaled = c * dx
    # log-sum-exp: subtract max for numerical stability
    max_c = np.max(c_scaled)
    shifted = c_scaled - max_c
    log_sum_exp = max_c + np.log(np.sum(np.exp(shifted)))
    return eps * log_sum_exp

def smooth_gradient(h, eps, dx):
    """Compute gradient of the smooth surrogate."""
    c = compute_correlation(h)
    c_scaled = c * dx
    
    # Compute weights for soft-max
    max_c = np.max(c_scaled)
    shifted = c_scaled - max_c
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    
    # Gradient: sum_s w_s * grad C_s
    grad = np.zeros_like(h)
    N = len(h)
    
    # For each shift s, compute gradient contribution
    # The gradient at index k is (1 - h[k+s] - h[k-s]) * dx
    for s in range(len(c_scaled)):
        # shifts for this correlation
        if s == 0:
            # s = 0, gradient is 1 - h
            g = -dx * h
        else:
            # shift right (s > 0)
            g = np.zeros_like(h)
            if s < N:
                g[s:] = -dx * h[:-s]
            # shift left (negative index)
            g_left = np.zeros_like(h)
            if s < N:
                g_left[:-s] = -dx * h[s:]
            g = g + g_left
            
        grad += weights[s] * g
    return grad

def continuation_optimization(h_initial, dx):
    """Run continuation optimization on smooth surrogate."""
    h = h_initial.copy()
    eps_list = [0.10, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    bounds = Bounds(np.zeros_like(h), np.ones_like(h))
    target_sum = len(h) / 2

    for eps in eps_list:
        def obj(x):
            return smooth_objective(x, eps, dx)

        def grad(x):
            return smooth_gradient(x, eps, dx)

        # Run L-BFGS-B
        res = minimize(obj, h, method='L-BFGS-B', bounds=bounds, jac=grad,
                          options={'maxiter': 150, 'gtol': 1e-6})
        h = res.x
        
        # Project back to feasible set
        h = project_to_sum_constraint(h, target_sum)
    
    # Final projection to ensure exact sum
    h = project_to_sum_constraint(h, target_sum)
    return h

def subgradient_refinement(h, dx, max_iter=1000):
    """Refine solution using max-based sub-gradient descent."""
    h = h.copy()
    for _ in range(max_iter):
        c = compute_correlation(h)
        c_scaled = c * dx
        s_star = np.argmax(c_scaled)
        C_s_star = c_scaled[s_star]
        
        # Subgradient: -dx * (1 - shift_left(h,s_star) - shift_right(h,s_star))
        # For shift s_star:
        g = np.zeros_like(h)
        if s_star < len(h):
            g[s_star:] = -dx * h[:-s_star]
        if s_star > 0:
            g[:-s_star] -= dx * h[s_star:]
        # weight with C_s_star for descent direction
        # we want to minimize, so we use -g by virtue of signed gradient
        
        # Standard subgradient descent
        eta = 0.03
        h_proposed = h - eta * g
        h_proposed = project_to_sum_constraint(h_proposed, len(h) / 2)
        
        # Compute new objective (for the early stop condition)
        c_new = compute_correlation(h_proposed)
        c_new_scaled = c_new * dx
        new_c5 = np.max(c_new_scaled)
        if new_c5 < c5_of(h):
            h = h_proposed
        else:
            # Stop if no improvement
            break
    return h

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    N = 140
    dx = 2.0 / N
    target_sum = N / 2

    # Generate multiple start seeds
    seeds = []
    # Uniform
    uniform_seed = np.ones(N) * 0.5
    seeds.append(("uniform", uniform_seed))
    
    # Two-peak seeds (different a values)
    a_values = [0.12, 0.18, 0.24, 0.30]
    for a in a_values:
        h = np.ones(N)
        x_vals = np.linspace(0, 2, N)
        u = 0.9
        v = 0.1
        mask = (x_vals < a) | (x_vals >= 2 - a)
        h[mask] = u
        h[~mask] = v
        h = h * (target_sum / np.sum(h))
        seeds.append(("two_peak_a_{}".format(a), h))
    
    # Binary seeds (N/2 ones, rest zeros)
    num_binary_seeds = 10
    for i in range(num_binary_seeds):
        binary = np.zeros(N)
        ones_indices = np.random.choice(N, size=N//2, replace=False)
        binary[ones_indices] = 1.0
        binary = binary * (target_sum / np.sum(binary))
        seeds.append(("binary_{}".format(i), binary))
    
    # Binary two block seed
    binary_block = np.zeros(N)
    mid = N // 2
    binary_block[:mid] = 1.0
    binary_block[mid:] = 0.0
    binary_block = binary_block * (target_sum / np.sum(binary_block))
    seeds.append(("binary_block", binary_block))

    # Random continuous seeds
    num_rand_seeds = 8
    for i in range(num_rand_seeds):
        random_h = np.random.rand(N)
        random_h = random_h * (target_sum / np.sum(random_h))
        seeds.append(("rand_cont_{}".format(i), random_h))

    # Initial evaluation of seeds
    start_eval = time.time()
    best_c5 = float('inf')
    best_h = None
    for name, h0 in seeds:
        # project to sum constraint (just to be safe)
        h0 = project_to_sum_constraint(h0, target_sum)
        c5_eval = c5_of(h0)
        if c5_eval < best_c5:
            best_c5 = c5_eval
            best_h = h0.copy()
    
    time_taken_eval = time.time() - start_eval

    # Allocate two thirds for continuation
    continuation_budget = 0.75 * (budget_s - time_taken_eval) 
    remaining_budget = (budget_s - time_taken_eval) - continuation_budget
    
    # Run the continuation optimization for each seed
    max_runs_per_seed = 5
    run_count = 0
    start = time.time()
    
    for name, h0 in seeds:
        if time.time() - start >= continuation_budget:
            break
        # Project to feasible set (just in case)
        h0 = project_to_sum_constraint(h0, target_sum)
        
        # Run continuation optimization
        h = continuation_optimization(h0, dx)
        c5 = c5_of(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        run_count += 1
        # Early exit if time is almost up
        if time.time() - start >= continuation_budget:
            break

    # Final polishing with subgradient descent
    remaining_time = budget_s - (time.time() - start) - time_taken_eval
    if remaining_time > 0:
        polisher_start = time.time()
        best_h = subgradient_refinement(best_h, dx)
        # Re-evaluate final C5
        best_c5 = c5_of(best_h)
        
    # Final check if it's even better
    best_c5 = c5_of(best_h)
    
    return best_h, best_c5, N
