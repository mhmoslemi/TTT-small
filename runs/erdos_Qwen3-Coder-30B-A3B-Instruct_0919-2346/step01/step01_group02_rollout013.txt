import numpy as np
import time

def c5_of(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def worst_shift(h):
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    return idx - (len(h) - 1)

def subgradient(h, s):
    N = len(h)
    dx = 2.0 / N
    m = abs(s)
    g = np.full(N, dx)
    if m < N:
        g[:N-m] -= dx * h[m:]
        g[m:] -= dx * h[:N-m]
    return g

def project(v, target):
    lo = -np.max(v)
    hi = 1.0 - np.min(v)
    for _ in range(50):
        lam = (lo + hi) / 2.0
        w = np.clip(v + lam, 0.0, 1.0)
        if w.sum() < target:
            lo = lam
        else:
            hi = lam
    w = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        free = (w > 0) & (w < 1)
        if free.any():
            w[free] += diff / free.sum()
            w = np.clip(w, 0, 1)
    return w

def spd_loop(h_start, best_h, best_val, t_budget):
    start = time.time()
    h = h_start.copy()
    c = c5_of(h)
    if c < best_val:
        best_h[:] = h
        best_val = c
    t = 0
    eta0 = 30.0
    gamma = 0.9995
    while time.time() - start < t_budget:
        s = worst_shift(h)
        g = subgradient(h, s)
        eta = eta0 * (gamma ** t)
        v = h - eta * g
        h = project(v, len(h) / 2.0)
        c = c5_of(h)
        if c < best_val:
            best_h[:] = h
            best_val = c
        t += 1
        if t % 30 == 0:
            h_j = h + np.random.uniform(-0.02, 0.02, len(h))
            h = project(h_j, len(h) / 2.0)
            c = c5_of(h)
            if c < best_val:
                best_h[:] = h
                best_val = c
    return best_h, best_val

def anneal_loop(h_start, best_h, best_val, t_budget):
    start = time.time()
    h = h_start.copy()
    c = c5_of(h)
    if c < best_val:
        best_h[:] = h
        best_val = c
    T_start = 0.05
    T_end = 1e-6
    max_step_start = 0.2
    T = T_start
    step = max_step_start
    while time.time() - start < t_budget:
        elapsed = time.time() - start
        frac = elapsed / t_budget
        T = T_start * (T_end / T_start) ** frac
        step = max_step_start * (1.0 - frac)
        i, j = np.random.choice(len(h), size=2, replace=False)
        delta_low = max(-h[i], h[j] - 1)
        delta_high = min(1 - h[i], h[j])
        if delta_low >= delta_high:
            continue
        step_size = np.random.uniform(max(-step, delta_low),
                                      min(step, delta_high))
        h_new = h.copy()
        h_new[i] += step_size
        h_new[j] -= step_size
        c_new = c5_of(h_new)
        if c_new <= c or np.random.rand() < np.exp(-(c_new - c) / T):
            h = h_new
            c = c_new
            if c < best_val:
                best_h[:] = h
                best_val = c
    return best_h, best_val

def hill_climb_loop(h_start, best_h, best_val, t_budget):
    start = time.time()
    h = h_start.copy()
    c = c5_of(h)
    if c < best_val:
        best_h[:] = h
        best_val = c
    while time.time() - start < t_budget:
        i, j = np.random.choice(len(h), size=2, replace=False)
        delta_low = max(-h[i], h[j] - 1)
        delta_high = min(1 - h[i], h[j])
        if delta_low >= delta_high:
            continue
        step_size = np.random.uniform(max(-0.02, delta_low),
                                      min(0.02, delta_high))
        h_new = h.copy()
        h_new[i] += step_size
        h_new[j] -= step_size
        c_new = c5_of(h_new)
        if c_new < c:
            h = h_new
            c = c_new
            if c < best_val:
                best_h[:] = h
                best_val = c
    return best_h, best_val

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    N = 140
    dx = 2.0 / N

    # Phase 0: seeding
    seeds = []
    if 'initial_h_values' in globals():
        h_base = np.interp(np.linspace(0, 2, N),
                          np.linspace(0, 2, len(initial_h_values)),
                          initial_h_values)
        h_base = h_base * (N / 2) / np.sum(h_base)
        seeds.append(h_base.copy())
    else:
        h_base = np.ones(N) * 0.5
        seeds.append(h_base)
    
    for _ in range(3):
        h = np.random.choice([0, 1], size=N, p=[0.5, 0.5])
        h = h * (N / 2) / np.sum(h)
        seeds.append(h)

    for a in [0.12, 0.18, 0.25, 0.30, 0.35, 0.38]:
        h = np.ones(N)
        x_vals = np.linspace(0, 2, N)
        u = 0.9
        v = 0.1
        mask = (x_vals < a) | (x_vals >= 2 - a)
        h[mask] = u
        h[~mask] = v
        h = h * (N / 2) / np.sum(h)
        seeds.append(h)

    # Compute C5 for each seed and project if necessary
    c_values = []
    for i, h in enumerate(seeds):
        h = project(h, N / 2.0)
        c = c5_of(h)
        c_values.append(c)
        seeds[i] = h.copy()

    best_idx = np.argmin(c_values)
    best_h = seeds[best_idx].copy()
    best_val = c_values[best_idx]

    # Phase 1: SPD
    t1 = 0.55 * budget_s
    s_start = time.time()
    best_h, best_val = spd_loop(seeds[best_idx], best_h, best_val, t1)
    t1_elapsed = time.time() - s_start

    # Phase 2: Annealing
    t2 = 0.30 * budget_s
    s_start = time.time()
    best_h, best_val = anneal_loop(best_h, best_h, best_val, t2)
    t2_elapsed = time.time() - s_start

    # Phase 3: hill climb
    t3 = 0.10 * budget_s
    s_start = time.time()
    best_h, best_val = hill_climb_loop(best_h, best_h, best_val, t3)
    t3_elapsed = time.time() - s_start

    return (best_h.copy(), best_val, N)
