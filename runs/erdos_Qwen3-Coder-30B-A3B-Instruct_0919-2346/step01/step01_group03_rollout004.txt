import numpy as np
import time
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import cdist

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral via full convolution of h with (1-h)
    overlaps = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum of the full correlation vector
    # scaled by dx
    c5 = np.max(overlaps) * dx
    return c5

def propose_move(h, max_step):
    """Propose a feasible move by adjusting two entries."""
    h = h.copy()
    N = len(h)
    # Try to find a valid pair of indices
    for _ in range(100):  # avoid infinite loop
        i, j = np.random.randint(0, N, 2)
        if i == j:
            continue
        # Determine feasible δ range
        delta_low = max(-h[i], h[j] - 1)
        delta_high = min(1 - h[i], h[j])
        if delta_low >= delta_high:
            continue
        # Sample a move with limited step size
        step = min(max_step, delta_high - delta_low)
        delta = np.random.uniform(
            max(-step, delta_low),
            min(step, delta_high)
        )
        # Apply move
        h[i] += delta
        h[j] -= delta
        return h
    # If no valid move found, return copy
    return h

def improve_with_optimization(h, time_budget, seed):
    """Use scipy optimization to refine solution."""
    np.random.seed(seed)
    start = time.time()

    def constraint_sum(h):
        return np.sum(h) - 1.0  # Must sum to 1
    
    def constraint_bounds(h):
        return np.concatenate([h, 1 - h])  # 0 <= h <= 1
    
    # Use differential evolution to minimize C5
    bounds = [(0, 1) for _ in range(len(h))]
    result = differential_evolution(
        lambda x: c5_of(x),
        bounds,
        maxiter=1000,
        popsize=15,
        seed=seed,
        disp=False,
        tol=1e-6,
        mutation=(0.5, 1.0),
        recombination=0.7
    )
    
    return result.x, c5_of(result.x)

def anneal(initial_h, time_budget, seed):
