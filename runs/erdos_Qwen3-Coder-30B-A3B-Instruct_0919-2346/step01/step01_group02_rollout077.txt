import numpy as np
import time
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def c5_of(h):
    """Compute the maximum overlap integral for the given step function h."""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    c = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(c) * dx
    return c5

def evaluate_constraint(h):
    """Check constraint that integral equals 1."""
    dx = 2.0 / len(h)
    return np.sum(h) * dx - 1.0

def project_to_feasible(h):
    """Project h to satisfy the integral constraint."""
    dx = 2.0 / len(h)
    current_integral = np.sum(h) * dx
    if abs(current_integral - 1.0) > 1e-10:
        h = h * (1.0 / current_integral)
    # Clamp to [0, 1]
    h = np.clip(h, 0, 1)
    return h

def perturb_h(h, delta=0.02):
    """Perturb h by making small changes to two random points."""
    h_new = h.copy()
    i, j = np.random.choice(len(h), size=2, replace=False)
    # Calculate constraint bounds
    delta_low = max(-h_new[i], 1 - h_new[j])
    delta_high = min(1 - h_new[i], h_new[j])
    if delta_low < delta_high:
        step = np.random.uniform(delta_low, delta_high)
        h_new[i] += step
        h_new[j] -= step
    return h_new

def local_search(h, max_iter=1000, tolerance=1e-6):
    """Perform local search around h to improve it."""
    h = project_to_feasible(h)
    current_c5 = c5_of(h)
    for _ in range(max_iter):
        h_new = perturb_h(h)
        h_new = project_to_feasible(h_new)
        new_c5 = c5_of(h_new)
        if new_c5 < current_c5 - tolerance:
            h = h_new
            current_c5 = new_c5
    return h, current_c5
