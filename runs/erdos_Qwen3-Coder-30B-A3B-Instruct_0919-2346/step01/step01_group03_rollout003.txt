import numpy as np
import time
from scipy.optimize import differential_evolution, minimize

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral via full convolution of h with (1-h)
    overlaps = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum of the full correlation vector
    # scaled by dx
    c5 = np.max(overlaps) * dx
    return c5

def evaluate_constraints(h, n_points):
    """Check if h satisfies constraints"""
    dx = 2.0 / n_points
    integral_constraint = abs(np.sum(h) * dx - 1.0) < 1e-10
    bounds_constraint = np.all((h >= 0) & (h <= 1))
    return integral_constraint and bounds_constraint

def smooth_step(h, smoothness=0.1):
    """Smooth the function to make it more continuous"""
    smoothed = h.copy()
    for i in range(1, len(h) - 1):
        smoothed[i] = (h[i-1] + h[i] + h[i+1]) / 3.0
    # Add a bit of smoothing on top
    for i in range(len(h)):
        smoothed[i] = (1 - smoothness) * h[i] + smoothness * smoothed[i]
    return np.clip(smoothed, 0, 1)

def genetic_algorithm(initial_h, time_budget, seed):
    """Run genetic algorithm"""
    np.random.seed(seed)
    start = time.time()
    n_points = len(initial_h)
    population_size = 20
    max_generations = 1000
    mutation_rate = 0.05
    elite_size = 2
    
    # Initialize population
    population = [initial_h.copy() for _ in range(population_size)]
    for ind in population:
        # Normalize to satisfy constraint
        ind *= (n_points / 2) / np.sum(ind)
        
    best_h = initial_h.copy()
    best_c5 = c5_of(best_h)
    
    for generation in range(max_generations):
        if time.time() - start > time_budget:
            break
            
        # Evaluate fitness (lower C5 is better)
        fitness = [c5_of(ind) for ind in population]
        sorted_indices = np.argsort(fitness)
        
        # Update best
        if fitness[sorted_indices[0]] < best_c5:
            best_h = population[sorted_indices[0]].copy()
            best_c5 = fitness[sorted_indices[0]]
            
        # Create new population
        new_population = [population[i].copy() for i in sorted_indices[:elite_size]]
        
        # Crossover and mutation
        for _ in range(population_size - elite_size):
            # Tournament selection
            parent1 = population[np.random.choice(sorted_indices[:len(sorted_indices)//2])]
            parent2 = population[np.random.choice(sorted_indices[:len(sorted_indices)//2])]
            
            # Single-point crossover
            crossover_point = np.random.randint(1, n_points-1)
            child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            
            # Mutation
            if np.random.rand() < mutation_rate:
                mutate_index = np.random.randint(0, n_points)
                child[mutate_index] = np.random.rand()
                
            # Ensure constraints
            child *= (n_points / 2) / np.sum(child)
            child = np.clip(child, 0, 1)
            
            new_population.append(child)
            
        population = new_population
        
    return best_h, best_c5

def improved_hill_climb(initial_h, time_budget, seed):
    """Improve on hill climbing with more sophisticated moves"""
    np.random.seed(seed)
    start = time.time()
    best_h = initial_h.copy()
    best_c5 = c5_of(best_h)
    
    n_points = len(initial_h)
    
    def propose_move(h, step_size=0.05):
        h = h.copy()
        # Try updating several points at once
        n_updates = np.random.randint(1, min(5, n_points//2))
        indices = np.random.choice(n_points, n_updates, replace=False)
        
        for i in indices:
            old_val = h[i]
            # Determine valid range
            delta_low = max(-old_val, 0)
            delta_high = min(1 - old_val, 1)
            
            if delta_low < delta_high:
                delta = np.random.uniform(delta_low, delta_high)
                h[i] += delta
                # Ensure integral constraint is approximately maintained
                adjustment = -delta * n_updates / n_points
                h[np.random.choice(len(h), 1)] += adjustment
        
        # Normalize to maintain sum constraint
        h *= (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        return h
    
    while time.time() - start < time_budget:
        # Large improvement
        for _ in range(5):
            h_new = propose_move(best_h, 0.1)
            c_new = c5_of(h_new)
            if c_new < best_c5:
                best_h = h_new
                best_c5 = c_new
                
        # Fine tune
        for _ in range(10):
            h_new = propose_move(best_h, 0.05)
            c_new = c5_of(h_new)
            if c_new < best_c5:
                best_h = h_new
                best_c5 = c_new
                
        # Try a few tricks to diversify
        if np.random.rand() < 0.1:
            # Perturb gently and re-optimize
            noise_strength = 0.1
            noisy_h = best_h + np.random.normal(0, noise_strength, n_points)
            noisy_h = np.clip(noisy_h, 0, 1)
            noisy_h *= (n_points / 2) / np.sum(noisy_h)
            h_new = noisy_h
            c_new = c5_of(h_new)
            if c_new < best_c5:
                best_h = h_new
                best_c5 = c_new
    
    return best_h, best_c5
