import numpy as np
import time
import random

def project_h(h, target_sum):
    """Project h onto the feasible region: [0,1] and sum = target_sum."""
    # Clip to [0, 1]
    h_clipped = np.clip(h, 0, 1)
    current = np.sum(h_clipped)
    if np.isclose(current, target_sum):
        return h_clipped
    # Use binary search to find lambda such that sum of clipped values equals target
    if current > target_sum:
        # Need to decrease some values
        low, high = 0.0, np.max(h_clipped)
        for _ in range(100):
            mid = (low + high) / 2.0
            h_test = np.clip(h_clipped - mid, 0, 1)
            sum_test = np.sum(h_test)
            if sum_test >= target_sum:
                low = mid
            else:
                high = mid
        lambda_val = low
        h_proj = np.clip(h_clipped - lambda_val, 0, 1)
    else:
        # Need to increase some values
        low, high = 0.0, 1.0
        for _ in range(100):
            mid = (low + high) / 2.0
            h_test = np.clip(h_clipped + mid, 0, 1)
            sum_test = np.sum(h_test)
            if sum_test <= target_sum:
                low = mid
            else:
                high = mid
        lambda_val = low
        h_proj = np.clip(h_clipped + lambda_val, 0, 1)
    return h_proj

def compute_C_and_grad(h, alpha):
    """Compute the current C5 score and its gradient with respect to h."""
    dx = 2.0 / len(h)
    # Compute h with zero padding
    h_padded = np.pad(h, (len(h)-1, len(h)-1), mode='constant', constant_values=0)
    
    # Compute autocorrelation via FFT to efficiently evaluate h * h
    fft_h = np.fft.fft(h_padded)
    auto_correlation = np.fft.ifft(fft_h * np.conj(fft_h))
    auto_correlation = np.real(auto_correlation[:2*len(h)-1])
    
    # C_k = 1 - dx * S_k where S_k is autocorrelation of h
    C = 1 - dx * auto_correlation
    
    # Smooth the max using log-sum-exp
    w = np.exp(alpha * C)
    w_sum = np.sum(w)
    w = w / w_sum
    
    # Compute gradient: g_i = -dx * sum_k w_k * (h_{i+k} + h_{i-k})
    # But that's tricky, let's do it via convolutions for cross-correlations
    # We want sum_k w_k * h_{i+k} = (w * h)_i where * is convolution
    w_rev = w[::-1]
    conv1 = np.convolve(h, w_rev, mode='same')
    conv2 = np.convolve(h, w, mode='same')
    grad = dx * (1 - (conv1 + conv2))
    
    return np.max(C), grad

    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # We'll do a few restarts

        if time.time() - start_time > 0.9 * budget_s:
            break
