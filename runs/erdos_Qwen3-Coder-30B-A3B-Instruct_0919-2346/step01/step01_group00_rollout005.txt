import numpy as np
import time
import random
from scipy.optimize import minimize

def project_h(h, target_sum):
    """Project h onto box constraints [0,1] and sum constraint."""
    # Clip to [0, 1]
    h_clipped = np.clip(h, 0, 1)
    current = np.sum(h_clipped)
    
    if np.isclose(current, target_sum):
        return h_clipped
    
    # Binary search for lambda to adjust sum
    tol = 1e-10
    low, high = -1.0, 1.0
    
    while high - low > tol:
        mid = (low + high) / 2
        h_test = np.clip(h_clipped - mid, 0, 1)
        sum_test = np.sum(h_test)
        if sum_test > target_sum:
            low = mid
        else:
            high = mid
    
    lambda_opt = (low + high) / 2
    h_final = np.clip(h_clipped - lambda_opt, 0, 1)
    return h_final

def compute_C_and_grad(h, alpha):
    """Compute C vector and gradient"""
    N = len(h)
    dx = 2.0 / N
    
    # Autocorrelation of h with itself
    h_fft = np.fft.fft(h, n=2*N-1)
    S = np.fft.ifft(h_fft * np.conj(h_fft)).real
    
    # C_k = 1 - dx * S_k
    C = 1 - dx * S
    
    # Softmax weights for smoothing
    w = np.exp(alpha * C)
    w = w / np.sum(w)
    
    # Gradient g_i = dx * (1 - sum_k w_k (h_{i+k} + h_{i-k}))
    # For shift k, we compute sum_{i} w_k * h_{i+k}
    # Using convolution with w, w_rev
    w_rev = w[::-1]
    conv1 = np.convolve(h, w_rev, mode='same')
    conv2 = np.convolve(h, w, mode='same')
    
    grad = dx * (1 - (conv1 + conv2))
    
    # Return max C and gradient
    C_max = np.max(C)
    return C_max, grad

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    N = 140
    dx = 2.0 / N
    target_sum = N / 2.0
    
    # Refine initial construction
    # Use a slightly better version of the initial construction:
    # Double-peak with small adjustment to make it more optimal
    h_initial = np.zeros(N)
    
    # Define centers and widths for double-peak shape
    center1 = 0.3
    center2 = 1.7
    width = 0.4
    height_high = 0.95
    height_low = 0.1
    
    x = np.linspace(0, 2, N)
    
    # Create piecewise linear function
    h_initial = np.where((x >= center1 - width/2) & (x <= center1 + width/2), 
                         height_high,
                         np.where((x >= center2 - width/2) & (x <= center2 + width/2),
                                  height_high,
                                  height_low))
    
    # Renormalize to target sum
    current_sum = np.sum(h_initial)
    h_initial = h_initial * (target_sum / current_sum)
    
    # Apply gradient descent from initial point
    h_best = h_initial.copy()
    c5_best = float('inf')
    start = time.time()
    
    # Gradient descent parameters
    eta0 = 1.0
    alpha_start = 5.0
    alpha_end = 80.0
    max_iter = 20000
    beta = 0.95  # Step size decay
    
    for iter_num in range(max_iter):
        if time.time() - start > budget_s * 0.95:
            break
            
        # Adaptive alpha
        t = iter_num / max_iter
        alpha = alpha_start * (alpha_end / alpha_start) ** t
        
        # Compute C and gradient
        c5_current, grad = compute_C_and_grad(h_best, alpha)
        
        if c5_current < c5_best:
            c5_best = c5_current
            h_best = h_best.copy()
        
        # Update with step size
        eta = eta0 * (beta ** iter_num)
        h_new = h_best - eta * grad
        h_new = project_h(h_new, target_sum)
        
        h_best = h_new
    
    # Final evaluation
    # Perform final non-smoothed evaluation
    dx = 2.0 / N
    c = np.correlate(h_best, 1 - h_best, mode="full")
    c *= dx
    c5_final = np.max(c)
    
    return (h_best, c5_final, N)
