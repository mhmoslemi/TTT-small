import numpy as np
import time
import scipy.optimize as opt

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral via full convolution of h with (1-h)
    overlaps = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum of the full correlation vector
    # scaled by dx
    c5 = np.max(overlaps) * dx
    return c5

def project_to_feasible(h):
    """Project h to satisfy the integral constraint"""
    dx = 2.0 / len(h)
    current_sum = np.sum(h) * dx
    target_sum = 1.0
    h = h * (target_sum / current_sum)
    return np.clip(h, 0, 1)

def optimize_with_scipy(h_init, maxiter=1000):
    """Use scipy optimize to improve the solution"""
    n = len(h_init)
    dx = 2.0 / n
    
    def objective(h):
        return c5_of(h)
    
    def constraint(h):
        return np.sum(h) * dx - 1.0
    
    # bounds for each variable
    bounds = [(0, 1) for _ in range(n)]
    
    # constraint dictionary
    cons = {'type': 'eq', 'fun': constraint}
    
    # Optimization
    result = opt.minimize(
        objective,
        h_init,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': maxiter, 'ftol': 1e-8, 'gtol': 1e-6}
    )
    
    if result.success:
        return project_to_feasible(result.x)
    else:
        return project_to_feasible(h_init)

def evaluate_and_improve(h, time_budget, start_time):
    """Perform local search and optimization on the current solution"""
    best_h = h.copy()
    best_c5 = c5_of(h)
    elapsed = time.time() - start_time
    
    # Try optimization with scipy
    if elapsed < time_budget * 0.7:
        optimized = optimize_with_scipy(best_h, maxiter=500)
        c5_opt = c5_of(optimized)
        if c5_opt < best_c5:
            best_h = optimized
            best_c5 = c5_opt
    
    # Local search by small perturbations
    np.random.seed(int(time.time() * 1000) % 2**32)
    step = 0.01
    T = 0.05
    for _ in range(1000):
        h_new = best_h.copy()
        i = np.random.randint(0, len(h_new))
        delta = np.random.uniform(-step, step)
        h_new[i] = np.clip(h_new[i] + delta, 0, 1)
        # Project to constraint
        h_new = project_to_feasible(h_new) 
        c5_new = c5_of(h_new)
        
        if c5_new < best_c5 or np.random.rand() < np.exp(-(c5_new - best_c5) / T):
            best_h = h_new
            best_c5 = c5_new
            
        if time.time() - start_time > time_budget * 0.95:
            break
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start = time.time()
    n_points = 140
    dx = 2.0 / n_points
    
    # Try several initialization strategies
    initializations = []
    
    # 1. Initial construction
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        if len(h0) != n_points:
            x0 = np.linspace(0, 2, len(h0))
            x_new = np.linspace(0, 2, n_points)
            h0 = np.interp(x_new, x0, h0)
        h_initial = h0 * (n_points / 2) / np.sum(h0)
        initializations.append(("initial", h_initial))
    else:
        # Uniform with integral scaling
        h_uniform = np.ones(n_points) * 0.5
        h_uniform = h_uniform * (n_points / 2) / np.sum(h_uniform)
        initializations.append(("uniform", h_uniform))

    # 2. Better tailored construction 
    h_tailed = np.zeros(n_points)
    x_vals = np.linspace(0, 2, n_points)
    
    # Place weights in two regions to avoid large overlaps
    mask1 = (x_vals >= 0.4) & (x_vals <= 0.8)
    mask2 = (x_vals >= 1.2) & (x_vals <= 1.6)
    
    h_tailed[mask1] = 0.8
    h_tailed[mask2] = 0.8
    
    # Scale to satisfy integral
    h_tailed = h_tailed * (n_points / 2) / np.sum(h_tailed)
    initializations.append(("tailored", h_tailed))

    # 3. Random construction, constrained
    h_random = np.random.rand(n_points)
    h_random = h_random * (n_points / 2) / np.sum(h_random)
    initializations.append(("random", h_random))
    
    # Optimize each initialization
    best_h = None
    best_c5 = float('inf')
    
    for name, h_seed in initializations:
        if time.time() - start > budget_s * 0.95:
            break
            
        h_improved, c5_improved = evaluate_and_improve(h_seed, budget_s, start)
        
        if c5_improved < best_c5:
            best_h = h_improved
            best_c5 = c5_improved
    
    # Extra polishing in last phase
    if best_h is not None and time.time() - start < budget_s * 0.98:
        remaining_time = budget_s - (time.time() - start)
        h_final = best_h.copy()
        c_final = best_c5
        
        step = 0.005
        # Do more intense hill climbing in final phase
        for _ in range(2000):
            h_new = h_final.copy()
            i, j = np.random.randint(0, n_points, 2)
            if i == j: continue
            delta = np.random.uniform(-step, step) 
            # Ensure the move is feasible
            low = max(-h_new[i], 1 - h_new[j])
            high = min(1 - h_new[i], h_new[j])
            if low < high:
                d = np.random.uniform(low, high)
                h_new[i] += d
                h_new[j] -= d
                # Project to feasible
                h_new = project_to_feasible(h_new)
                c_new = c5_of(h_new)
                if c_new < c_final:
                    h_final = h_new
                    c_final = c_new
            if time.time() - start > budget_s * 0.995:
                break
                
        if c_final < best_c5:
            best_h = h_final
            best_c5 = c_final
    
    return (best_h, best_c5, n_points)
