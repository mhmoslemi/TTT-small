import numpy as np
from scipy.optimize import minimize
import time

def project_onto_bounds_sum(v, target_sum, lower=0, upper=1):
    """Project vector v onto box constraints [lower, upper] with sum constraint"""
    # Binary search for the projection parameter lambda
    left = v.min() - upper
    right = v.max() - lower
    
    for _ in range(60):
        mid = (left + right) / 2
        s = np.clip(v - mid, lower, upper).sum()
        if s > target_sum:
            left = mid
        else:
            right = mid
    
    return np.clip(v - (left + right) / 2, lower, upper)

def compute_max_overlap(h):
    """Compute the maximum overlap integral for given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value scaled by dx
    max_overlap = np.max(correlation) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 200
    dx = 2.0 / n_points
    
    # Initialize using a smooth approximation, then refine it
    # Try a sine-based shape with some randomization
    x = np.linspace(0, 2, n_points)
    h = 0.5 + 0.3 * np.sin(np.pi * x / 2)  # Sine-like pattern
    h = np.clip(h, 0, 1)
    
    # Adjust to satisfy sum constraint
    current_sum = np.sum(h) * dx
    h = h * (1.0 / current_sum)
    
    # Gradient descent with projected updates
    gamma = 1.0
    gamma_end = 100
    step_size = 0.1
    
    best_h = h.copy()
    best_c5 = compute_max_overlap(h)
    
    iter_count = 0
    while (time.time() - start_time) < budget_s * 0.95:
        iter_count += 1
        # Compute correlation
        correlation = np.correlate(h, 1 - h, mode="full")
        f = correlation * dx
        max_f = np.max(f)
        
        # Compute softmax weights (gradient direction)
        shifted_f = f - max_f
        exp_f = np.exp(gamma * shifted_f)
        weights = exp_f / np.sum(exp_f)
        
        # Compute gradient components
        gradient = np.zeros_like(h)
        # Gradient for each shift
        for i in range(len(h)):
            for j in range(len(h)):
                shift_l = j - i
                # Adjust shift index to match numpy correlate convention
                shift_idx = shift_l + (n_points - 1)
                if 0 <= shift_idx < len(weights):
                    weight = weights[shift_idx]
                    # Compute contribution to gradient
                    if 0 <= i + shift_l < n_points:
                        gradient[i] += weight * (1 - h[i + shift_l])
                    if 0 <= i - shift_l < n_points:
                        gradient[i] -= weight * h[i - shift_l]
        
        # Update step with scaled gradient
        h_new = h - step_size * gradient
        # Project onto constraints
        h_new = project_onto_bounds_sum(h_new, n_points / 2, 0, 1)
        h = h_new
        
        # Check new solution
        current_c5 = compute_max_overlap(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Adaptive parameters
        if iter_count % 100 == 0:
            gamma = gamma_end * (iter_count / 5000) + 1.0
            step_size = 0.05 / (1 + iter_count / 1000)
            if step_size < 0.001:
                step_size = 0.001
                
        if (time.time() - start_time) > budget_s * 0.9:
            break
            
    return (best_h.tolist(), best_c5, n_points)
