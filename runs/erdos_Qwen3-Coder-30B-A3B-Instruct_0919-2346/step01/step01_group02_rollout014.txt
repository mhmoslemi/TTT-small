import numpy as np
import time
from scipy.optimize import minimize
from scipy.optimize import Bounds

def c5_of(h):
    """Compute the maximum overlap integral for the given step function h."""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    c = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(c) * dx
    return c5

def project_to_feasible(h, N_half):
    """Project h onto the set {h : 0 ≤ h[i] ≤ 1, sum(h) = N_half}"""
    # Waterfilling projection
    def F(lmbda):
        return np.sum(np.clip(h + lmbda, 0, 1)) - N_half
    
    # Binary search for lambda
    low, high = -1.0, 1.0
    for _ in range(50):
        mid = (low + high) / 2
        if F(mid) > 0:
            high = mid
        else:
            low = mid
    lambda_opt = (low + high) / 2
    
    h_proj = np.clip(h + lambda_opt, 0, 1)
    
    # Adjust to ensure exact sum
    diff = N_half - np.sum(h_proj)
    if abs(diff) > 1e-10:
        # Redistribute diff to free variables
        free_indices = np.where((h_proj > 0) & (h_proj < 1))[0]
        if len(free_indices) > 0:
            delta = diff / len(free_indices)
            for i in free_indices:
                h_proj[i] += delta
        else:
            # All variables are at bounds - exact projection
            pass
    
    return h_proj

def smooth_objective(h, epsilon, dx):
    """Compute smooth surrogate F_epsilon(h) and its gradient."""
    # Compute all correlations
    c = np.correlate(h, 1 - h, mode='full')
    C = c * dx
    # Avoid numerical overflow
    C_max = np.max(C)
    exp_C = np.exp((C - C_max) / epsilon)
    Z = np.sum(exp_C)
    w = exp_C / Z
    # Gradient computation
    grad = np.zeros_like(h)
    for s in range(len(C)):
        shift_left = np.roll(h, s)
        shift_right = np.roll(h, -s)
        shift_left[:s] = 0
        shift_right[-s:] = 0
        grad += w[s] * dx * (1 - shift_left - shift_right)
    # Return smooth objective and gradient
    f = epsilon * (np.log(Z) + C_max / epsilon)
    return f, grad

def continuation_optimize(h0, N_half, epsilon_list, max_iter_inner=150, tol=1e-6):
    """Perform continuation optimization using smooth surrogate"""
    h = h0.copy()
    h = project_to_feasible(h, N_half)
    for epsilon in epsilon_list:
        # L-BFGS-B for this epsilon
        bounds = Bounds(np.zeros_like(h), np.ones_like(h))
        def obj_grad(x):
            f, g = smooth_objective(x, epsilon, 2.0 / len(x))
            return f, g
        
        res = minimize(
            obj_grad,
            h,
            method='L-BFGS-B',
            bounds=bounds,
            jac=True,
            options={'maxiter': max_iter_inner, 'ftol': tol, 'gtol': tol}
        )
        h = res.x
        # Project to feasible set
        h = project_to_feasible(h, N_half)
        # Final projection to exact sum
        h = project_to_feasible(h, N_half)
    return h

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    N = 140
    N_half = N / 2
    dx = 2.0 / N
    
    # Time allocation
    start = time.time()
    seed_time = 0.05 * budget_s
    opt_time = 0.70 * budget_s
    multi_start_time = 0.10 * budget_s
    polish_time = 0.10 * budget_s
    final_time = 0.05 * budget_s
    
    # Generate diverse seeds
    seeds = []
    
    # 1. Uniform seed
    h_uniform = np.ones(N) * 0.5
    seeds.append(("uniform", h_uniform))
    
    # 2. Two-peak seeds 
    x_vals = np.linspace(0, 2, N)
    a_choices = [0.12, 0.18, 0.24, 0.30]
    u = 0.9
    v = 0.1
    for a in a_choices:
        h = np.ones(N) * v
        mask = (x_vals < a) | (x_vals >= 2 - a)
        h[mask] = u
        h = h * (N_half) / np.sum(h)
        seeds.append((f"two_peak_{a:0.2f}", h))
    
    # 3. Binary seed - random selection
    h_binary = np.zeros(N)
    ones_indices = np.random.choice(N, size=int(N_half), replace=False)
    h_binary[ones_indices] = 1.0
    seeds.append(("binary_random", h_binary))
    
    # 4. Binary seed - block
    h_binary_block = np.zeros(N)
    h_binary_block[:int(N_half)] = 1.0
    seeds.append(("binary_block", h_binary_block))
    
    # 5. Random continuous
    h_random = np.random.uniform(0, 1, N)
    h_random = h_random * (N_half) / np.sum(h_random)
    seeds.append(("random", h_random))
    
    # Optimization with continuation strategy
    epsilon_list = [0.10, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    
    best_c5 = float("inf")
    best_h = None
    
    # Multi-start strategy
    remaining_time = opt_time
    for name, h_seed in seeds:
        if time.time() - start > start + remaining_time:
            break
        h_seed = project_to_feasible(h_seed, N_half)
        h_opt = continuation_optimize(h_seed, N_half, epsilon_list)
        c5_opt = c5_of(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    # Final polishing with sub-gradient descent
    remaining_time = polish_time
    start_polish = time.time()
    
    while time.time() - start_polish < remaining_time and best_c5 > 0.379:
        # Find worst shift
        c = np.correlate(best_h, 1 - best_h, mode='full')
        worst_idx = np.argmax(c)
        # Compute sub-gradient
        shift_left = np.roll(best_h, worst_idx)
        shift_right = np.roll(best_h, -worst_idx)
        shift_left[:worst_idx] = 0
        shift_right[-worst_idx:] = 0
        g = dx * (1 - shift_left - shift_right)
        # Step
        eta = 0.03
        h_new = best_h - eta * g
        h_new = project_to_feasible(h_new, N_half)
        c5_new = c5_of(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
        else:
            # Small random adjustment
            h_rand = best_h + np.random.uniform(-0.005, 0.005, N)
            h_rand = project_to_feasible(h_rand, N_half)
            c5_rand = c5_of(h_rand)
            if c5_rand < best_c5:
                best_c5 = c5_rand
                best_h = h_rand
    
    return (best_h, best_c5, N)
