import numpy as np
import time
import random
from scipy.optimize import differential_evolution, minimize


def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    overlap = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(overlap) * dx
    return c5

    if new_i < 0 or new_i > 1 or new_j < 0 or new_j > 1:
        return None, None, False

    old_sum = np.sum(h)
    new_sum = old_sum + delta - delta  # stays same

    h_new[i] = new_i
    h_new[j] = new_j
    new_c5 = c5_of(h_new)
    
    return h_new, new_c5, True


def propose_grid_move(h, n_tries=50):
    """Propose moves with grid-based search"""
    h_new = h.copy()
    N = len(h)
    
    # Try multiple random pairs
    for _ in range(n_tries):
        i = random.randint(0, N - 1)
        j = random.randint(0, N - 1)
        if i == j:
            continue
            
        # Determine feasible interval for delta
        delta_low = max(-h_new[i], h_new[j] - 1)
        delta_high = min(1 - h_new[i], h_new[j])
        
        if delta_low >= delta_high:
            continue
            
        # Sample delta
        delta = random.uniform(0, 1) * (delta_high - delta_low) + delta_low
        
        # Apply move
        h_new[i] += delta
        h_new[j] -= delta
        
        return h_new, True
    
    return h.copy(), False


def search_local(h, n_iters=500):
    """Perform local optimization"""
    old_c5 = c5_of(h)
    h_old = h.copy()
    
    for _ in range(n_iters):
        h_new, success = propose_grid_move(h_old)
        if not success:
            continue
            
        new_c5 = c5_of(h_new)
        
        if new_c5 < old_c5:
            h_old = h_new
            old_c5 = new_c5
            
    return h_old
