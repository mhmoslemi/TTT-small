import numpy as np
import time
import random
from scipy.optimize import differential_evolution, minimize

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using full convolution
    overlap = np.correlate(h, 1 - h, mode='full')
    # The maximum value scaled by dx
    c5 = np.max(overlap) * dx
    return c5

def evaluate_constraints(h, n_points):
    """Evaluate the constraint that integral equals 1"""
    dx = 2.0 / n_points
    return abs(np.sum(h) * dx - 1.0)

def propose_move(h, max_step):
    """Propose a move that maintains sum constraint exactly"""
    h_new = h.copy()
    N = len(h)
    while True:
        i, j = random.sample(range(N), 2)
        # Compute feasible range for delta
        delta_low = max(-h_new[i], h_new[j] - 1.0)
        delta_high = min(1.0 - h_new[i], h_new[j])
        if delta_low >= delta_high:
            continue  # No valid move, try again
        break
    
    # Sample delta from feasible interval
    step = min(max_step, delta_high - delta_low)
    if step <= 0:
        delta = 0.0
    else:
        delta = random.uniform(max(-step, delta_low), min(step, delta_high))
        
    h_new[i] += delta
    h_new[j] -= delta
    return h_new

def optimize_with_differential_evolution(h0, n_points, budget_s):
    """Use differential evolution to search for better solutions"""

        # Reshape to match h
        h = np.clip(x, 0, 1)
        return c5_of(h)

        # Ensure constraint sum(h) * dx = 1
        dx = 2.0 / n_points
        return np.sum(x) * dx - 1.0
    
    # Bounds: [0, 1] for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization constraints
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    # Run optimization
    start = time.time()
    try:
        result = differential_evolution(
            objective,
            bounds,
            constraints=constraints,
            maxiter=int(budget_s * 5),
            popsize=15,
            mutation=(0.5, 1),
            recombination=0.7,
            seed=42,
            disp=False
        )
        h_final = np.clip(result.x, 0, 1)
        c5_final = c5_of(h_final)
        return h_final, c5_final
    except:
        return h0, c5_of(h0)
