import numpy as np
from scipy.optimize import minimize
from math import pi
import time

def build_h_from_coeffs(a, n):
    """Build h vector from cosine coefficients"""
    h = 0.5 * np.ones(n)
    for m in range(1, len(a) + 1):
        cos_term = np.cos(2 * pi * m * np.arange(n) / n)
        h += a[m - 1] * cos_term
    return h

def compute_overlap_and_gradient(a, n, dx, gamma=1.0):
    """Compute overlap and its gradient with respect to coefficients a"""
    h = build_h_from_coeffs(a, n)
    h = np.clip(h, 0, 1)
    
    # Compute correlation using numpy
    corr = np.correlate(h, 1 - h, mode='full')
    
    # Extract all overlaps
    overlaps = []
    for s in range(-n + 1, n):
        overlap = corr[s + n - 1] * dx
        overlaps.append(overlap)
    
    overlaps = np.array(overlaps)
    
    # Softmax weight
    exp_terms = np.exp(gamma * overlaps)
    weights = exp_terms / np.sum(exp_terms)
    
    # Cost
    cost = np.log(np.sum(exp_terms)) / gamma
    
    # Penalty for exceeding bounds
    penalty = 1e4 * np.sum(np.maximum(0, h - 1)**2 + np.maximum(0, -h)**2)
    cost += penalty
    
    # Gradient
    grad = np.zeros(len(a))
    for m in range(1, len(a) + 1):
        cos_term = np.cos(2 * pi * m * np.arange(n) / n)
        for s in range(-n + 1, n):
            shift_index = s + n - 1
            # For the shift s, compute derivative
            if 0 <= shift_index < len(corr):
                h1 = h
                h2 = 1 - h
                # Compute gradient
                i_range = np.arange(0, min(n, n - abs(s)))
                if s >= 0:
                    i_range = np.arange(0, n - s)
                else:
                    i_range = np.arange(-s, n)
                
                # Project i_range to valid indices
                if s >= 0:
                    i_valid = np.arange(0, n - s)
                    s_valid = np.arange(s, n)
                else:
                    i_valid = np.arange(-s, n)
                    s_valid = np.arange(0, n + s)
                
                # select valid indices
                i_valid = np.arange(0, max(0, n - abs(s)))
                if s >= 0:
                    i_valid = np.arange(0, n - s)
                    s_valid = np.arange(s, n)
                else:
                    i_valid = np.arange(-s, n)
                    s_valid = np.arange(0, n + s)
                
                # Simplified direct gradient computation
                pattern = cos_term[i_valid] * (1 - h[s_valid]) - h[i_valid] * cos_term[s_valid]
                grad[m-1] += weights[s + n - 1] * np.sum(pattern)
                
    # Add penalty gradient term
    clipped_h = np.clip(h, 0, 1)
    diff = h - clipped_h
    for m in range(1, len(a) + 1):
        cos_term = np.cos(2 * pi * m * np.arange(n) / n)
        penalty_grad = 2 * 1e4 * np.sum(diff * cos_term)
        grad[m-1] += penalty_grad
    
    return cost, grad

def project_to_box_simplex(h, n, dx):
    """Project h onto box [0,1]^n of sum=h_sum*dx"""
    target_sum = n/2.0
    h_clip = np.clip(h, 0, 1)
    current_sum = np.sum(h_clip) * dx
    if abs(current_sum - target_sum) < 1e-12:
        return h_clip
    
    # Use bisection to find threshold
    low, high = -1.0, 1.0
    while high - low > 1e-10:
        mid = (low + high) / 2
        guess = np.clip(h_clip - mid, 0, 1)
        guess_sum = np.sum(guess) * dx
        if guess_sum > target_sum:
            low = mid
        else:
            high = mid
    tau = (low + high) / 2
    h_proj = np.clip(h_clip - tau, 0, 1)
    return h_proj

def spectral_optim(n, max_iter=1000, gamma_max=40.0, gamma_min=0.5, num_restarts=6):
    best_c5 = float('inf')
    best_h = None
    dx = 2.0 / n

    for seed in range(num_restarts):
        np.random.seed(seed * 10 + 42)
        # Initial guess with low-frequency components
        M = 8  # Initial number of Fourier coefficients
        a_init = np.random.normal(0, 0.05, size=M)
        
        # Temperature schedule using homotopy
        gamma_schedule = [
            gamma_min * (1.5 ** i) for i in range(20)
            if gamma_min * (1.5 ** i) <= gamma_max
        ] + [gamma_max] * 10000
        
        num_stages = 3
        stage_info = [(8, 1500), (15, 2000), (24, 2500)]
        current_a = a_init.copy()
        
        for stage, (M, max_iter_stage) in enumerate(stage_info):
            if stage > 0:
                # Extend coefficients to full M
                new_coeffs = np.zeros(M)
                new_coeffs[:len(current_a)] = current_a
                current_a = new_coeffs
                
            # Break if too little budget left
            if stage > 0:
                remaining = max_iter_stage
                if remaining <= 0:
                    break
            
            # Optimize for this stage
            bounds = [(-0.6, 0.6) for _ in range(M)]
            iter_count = 0
            
            def obj_and_grad(a_vec):
                val, grad = compute_overlap_and_gradient(a_vec, n, dx, gamma_schedule[0])
                return val, grad

            # Simple gradient descent with line search or L-BFGS-B
            try:
                result = minimize(
                    lambda a: compute_overlap_and_gradient(a, n, dx, gamma_schedule[0])[0],
                    current_a,
                    method='L-BFGS-B',
                    bounds=bounds,
                    jac=lambda a: compute_overlap_and_gradient(a, n, dx, gamma_schedule[0])[1],
                    options={'maxiter': max_iter_stage, 'ftol': 1e-9}
                )
                if result.success:
                    current_a = result.x
                else:
                    pass
            except Exception:
                pass
            iter_count += max_iter_stage

        # Construct final h
        h_final = build_h_from_coeffs(current_a, n)
        h_final = np.clip(h_final, 0, 1)
        h_projected = project_to_box_simplex(h_final, n, dx)
        c5_val = evaluate_erdos_solution(h_projected)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_projected.copy()

    return best_h, best_c5

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value scaled by dx
    max_overlap = np.max(correlation) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
