import numpy as np

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value scaled by dx
    max_overlap = np.max(correlation) * dx
    return max_overlap

def upsample_initial(initial_h, n):
    """Upsample initial_h to size n using linear interpolation"""
    old_n = len(initial_h)
    old_x = np.linspace(0, 2, old_n)
    new_x = np.linspace(0, 2, n)
    h_new = np.interp(new_x, old_x, initial_h)
    return h_new

def project_to_simplex(v):
    """Project vector v onto the simplex {x: sum x = n/2, 0 ≤ x_i ≤ 1}"""
    n = len(v)
    # Sort v in descending order to find the projection threshold
    v_sorted = np.sort(v)[::-1]
    # Compute cumulative sums from the largest
    cumsum = np.cumsum(v_sorted)
    # Find the threshold index
    i = np.argmax(v_sorted - (n/2 - cumsum + v_sorted) / (np.arange(n) + 1)) 
    # Threshold value
    theta = (n/2 - cumsum[i]) / (i + 1)
    # Compute the projection
    proj = np.clip(v - theta, 0, 1)
    return proj

def compute_shift_gradients(h, dx):
    """Compute all gradient components for the objective"""
    n = len(h)
    grad = np.zeros_like(h)
    for s in range(1 - n, n):
        # Compute g^{(s)}_{i}
        g = np.zeros(n)
        if s >= 0:
            # g_i = (1 - h_{i+s}) if i+s < n else 0
            g[:-s] = 1 - h[s:]
        else:
            # g_i = -h_{i-s} if i-s >= 0 else 0
            g[-s:] = -h[:-s]
        grad += dx * g
    return grad

def softmax_weights(f_vals, gamma):
    """Compute softmax weights for given function values with temperature gamma"""
    exp_vals = np.exp(gamma * f_vals)
    return exp_vals / np.sum(exp_vals)

def compute_f_all_shifts(h):
    """Compute all f_s(h) values efficiently"""
    n = len(h)
    dx = 2.0 / n
    # Compute all shifts in one go
    fs = []
    for lag in range(1 - n, n):
        # corr = np.correlate(h, 1 - h, mode="full")[j] corresponds to shift l = 1 - n + j
        if lag >= 0:
            # h_i * (1 - h_{i+lag})
            v = h[:n-lag] * (1 - h[lag:])
        else:
            # h_i * (1 - h_{i+lag})
            v = h[-lag:] * (1 - h[:-lag])
        fs.append(np.sum(v) * dx)
    return np.asarray(fs)

def projected_gradient_descent(h0, n, max_iter, alpha0, gamma_max, seed=42):
    np.random.seed(seed)
    h = h0.copy()
    
    # Annealing schedule for gamma
    gamma0 = 1.0
    direction = 1 if gamma_max > gamma0 else -1
    gamma_schedule = [gamma0 + (gamma_max - gamma0) * (i / max_iter) for i in range(max_iter)]
    
    # Epsilon for step size decay
    alpha_schedule = [alpha0 / sqrt(i+1) for i in range(max_iter)]
    
    best_h = h.copy()
    best_obj = evaluate_erdos_solution(h)
    patience_counter = 0
    patience_threshold = 2000
    restart_count = 0

    for t in range(max_iter):
        # Set gamma and alpha for current step
        gamma = gamma_schedule[t]
        alpha = alpha_schedule[t]
        if (t % 5000 == 0) or (t == max_iter - 1):
            obj = evaluate_erdos_solution(h)
            if obj < best_obj:
                best_obj = obj
                best_h = h.copy()
                patience_counter = 0
            else:
                patience_counter += 1
            
        # Compute all f_s values
        f_all = compute_f_all_shifts(h)
        
        # Compute softmax weights
        weights = softmax_weights(f_all, gamma)
        
        # Compute surrogate gradient
        grad = np.zeros_like(h)
        for lag_idx, w in enumerate(weights):
            s = 1 - n + lag_idx
            g = np.zeros(n)
            if s >= 0:
                g[:-s] = 1 - h[s:]
            else:
                g[-s:] = -h[:-s]
            grad += w * g * 2.0 / n  # dx = 2/n
        
        v = h - alpha * grad
        
        h = project_to_simplex(v)
        
        if patience_counter > patience_threshold:
            # Restart with a perturbed version of the current best solution
            h = best_h + 0.01 * np.random.randn(n)
            h = project_to_simplex(h)
            patience_counter = 0
            restart_count += 1

    obj = evaluate_erdos_solution(h)
    if obj < best_obj:
        best_obj = obj
        best_h = h.copy()
        
    return best_h, best_obj
