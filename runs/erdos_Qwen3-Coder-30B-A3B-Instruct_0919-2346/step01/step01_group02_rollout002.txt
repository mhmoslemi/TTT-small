import numpy as np
import time

def c5_of(h):
    """Compute the maximum overlap integral for the given step function h."""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    c = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(c) * dx
    return c5

def worst_shift(h):
    """Determine the shift s that maximizes the overlap integral."""
    corr = np.correlate(h, 1-h, mode='full')
    idx = np.argmax(corr)
    return idx - (len(h)-1)  # signed shift

def subgradient(h, s):
    """Compute the subgradient of C5 with respect to h for the worst shift s."""
    N = len(h)
    dx = 2.0 / N
    m = abs(s)
    
    # Base vector with dx
    g = np.full(N, dx)
    
    # Subtract terms for h[i+m] and h[i-m] if indices are valid
    if m < N:
        g[:N-m] -= dx * h[m:]
        g[m:]   -= dx * h[:N-m]
    
    return g

def project(v, target):
    """Project vector v onto the feasible set: 0 ≤ h[i] ≤ 1 and sum(h) = target."""
    lo = -np.max(v)
    hi = 1.0 - np.min(v)
    for _ in range(50):
        lam = (lo + hi) / 2.0
        w = np.clip(v + lam, 0.0, 1.0)
        if w.sum() < target:
            lo = lam
        else:
            hi = lam
    
    w = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        free = (w > 0) & (w < 1)
        if free.any():
            w[free] += diff / free.sum()
            w = np.clip(w, 0.0, 1.0)
    return w

def initialize_seeds(N):
    """Generate diverse seeds for optimization."""
    seeds = []
    
    # Seed from initial_h_values if available
    if 'initial_h_values' in globals():
        h0 = np.interp(np.linspace(0, 2, N),
                       np.linspace(0, 2, len(initial_h_values)),
                       initial_h_values)
        h0 = h0 * (N/2) / np.sum(h0)  # Renormalize exactly
        seeds.append(h0)
    
    # Two-peak template with varying parameters (a from 0.12 to 0.38)
    for a in np.linspace(0.12, 0.38, 6):
        h = np.ones(N)
        x_vals = np.linspace(0, 2, N)
        u = 0.9
        v = 0.1
        mask = (x_vals < a) | (x_vals >= 2 - a)
        h[mask] = u
        h[~mask] = v
        h = h * (N/2) / np.sum(h)  # Renormalize
        seeds.append(h)
    
    # Random binary vectors (exactly N/2 ones)
    for _ in range(3):
        h = np.random.rand(N) < 0.5
        h = h * (N/2) / np.sum(h)  # Renormalize
        seeds.append(h)
    
    # Uniform 0.5 vector
    h = np.ones(N) * 0.5
    seeds.append(h)
    
    # Project all seeds to feasible set
    for i in range(len(seeds)):
        seeds[i] = project(seeds[i], N/2)
        
    return seeds

def subgradient_descent(h_start, max_iter, eta_0=30.0, gamma=0.9995):
    """Perform projected subgradient descent from h_start."""
    h = h_start.copy()
    best_h = h.copy()
    best_val = c5_of(h)
    t = 0
    while t < max_iter:
        s = worst_shift(h)
        g = subgradient(h, s)
        eta = eta_0 * (gamma ** t)
        if eta > 50.0 * (2.0 / len(h)):
            eta = 50.0 * (2.0 / len(h))
        v = h - eta * g
        h = project(v, len(h) / 2.0)
        
        # Small random jitter every 30 iterations to escape shallow plateaux
        if t % 30 == 0:
            jitter = np.random.uniform(-0.02, 0.02, len(h))
            v = h + jitter
            h = project(v, len(h) / 2.0)
        
        val = c5_of(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()
        
        t += 1
    
    return best_h, best_val

def anneal_stochastic(h_start, max_time, T_start=0.05, T_end=1e-6):
    """Perform annealed stochastic optimization."""
    h = h_start.copy()
    c_cur = c5_of(h)
    best_h = h.copy()
    best_c5 = c_cur
    T = T_start
    start = time.time()
    while time.time() - start < max_time:
        # Sample a random pair of indices
        i, j = np.random.choice(len(h), size=2, replace=False)
        # Determine feasible step
        delta_low = max(-h[i], h[j] - 1)
        delta_high = min(1 - h[i], h[j])
        if delta_low >= delta_high:
            continue  # Try another pair
        # Sample step size
        step_size = np.random.uniform(max(-0.2, delta_low),
                                      min(0.2, delta_high))
        # Create new h
        h_new = h.copy()
        h_new[i] += step_size
        h_new[j] -= step_size
        # Project
        h_new = project(h_new, len(h) / 2.0)
        c_new = c5_of(h_new)
        # Accept or reject the move
        if c_new <= c_cur or np.random.rand() < np.exp(-(c_new - c_cur) / T):
            h = h_new
            c_cur = c_new
            if c_new < best_c5:
                best_h = h.copy()
                best_c5 = c_new
        # Update temperature
        elapsed = time.time() - start
        frac = elapsed / max_time
        T = T_start * (T_end / T_start) ** frac
    return best_h, best_c5

def hill_climb(h_start, max_time, step_size=0.02):
    """Perform deterministic hill climbing using small perturbations."""
    h = h_start.copy()
    c_cur = c5_of(h)
    start = time.time()
    while time.time() - start < max_time:
        # Try a few random moves and take the best improving one
        best_h_new = h.copy()
        best_c_new = c_cur
        for _ in range(100):
            i, j = np.random.choice(len(h), size=2, replace=False)
            delta_low = max(-h[i], h[j] - 1)
            delta_high = min(1 - h[i], h[j])
            if delta_low >= delta_high:
                continue
            step_size = np.random.uniform(max(-step_size, delta_low),
                                          min(step_size, delta_high))
            h_new = h.copy()
            h_new[i] += step_size
            h_new[j] -= step_size
            h_new = project(h_new, len(h) / 2.0)
            c_new = c5_of(h_new)
            if c_new < best_c_new:
                best_c_new = c_new
                best_h_new = h_new.copy()
        if best_c_new < c_cur:
            h = best_h_new
            c_cur = best_c_new
        else:
            break  # No improvements found
    return h, c_cur

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    N = 140
    dx = 2.0 / N
    
    # Initialize seeds for starting points
    seeds = initialize_seeds(N)
    
    # Find the best starting point
    best_val = float('inf')
    best_h = None
    for seed_h in seeds:
        val = c5_of(seed_h)
        if val < best_val:
            best_val = val
            best_h = seed_h.copy()
    
    t_start = time.time()
    best_global_h = best_h.copy()
    best_global_val = best_val
    
    # Phase 1: Sub-gradient descent
    remaining_budget = budget_s * 0.55
    phase_start = time.time()
    while time.time() - phase_start < remaining_budget and time.time() - t_start < budget_s * 0.99:
        h, val = subgradient_descent(best_global_h, max_iter=200)
        h, val = subgradient_descent(h, max_iter=100)
        if val < best_global_val:
            best_global_val = val
            best_global_h = h.copy()
    
    # Phase 2: Annealed stochastic moves
    remaining_budget = budget_s * 0.30
    phase_start = time.time()
    while time.time() - phase_start < remaining_budget and time.time() - t_start < budget_s * 0.99:
        h, val = anneal_stochastic(best_global_h, max_time=remaining_budget)
        if val < best_global_val:
            best_global_val = val
            best_global_h = h.copy()
    
    # Phase 3: Deterministic hill-climbing
    remaining_budget = budget_s * 0.10
    phase_start = time.time()
    while time.time() - phase_start < remaining_budget and time.time() - t_start < budget_s * 0.99:
        h, val = hill_climb(best_global_h, max_time=remaining_budget, step_size=0.02)
        if val < best_global_val:
            best_global_val = val
            best_global_h = h.copy()
    
    # Final evaluation
    final_val = c5_of(best_global_h)
    
    return (np.array(best_global_h), final_val, N)
