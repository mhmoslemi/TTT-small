import numpy as np
import time

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral via full convolution of h with (1-h)
    overlaps = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum of the full correlation vector
    # scaled by dx
    c5 = np.max(overlaps) * dx
    return c5

def full_overlap(h):
    """Return the raw correlation vector (unscaled)"""
    return np.correlate(h, 1 - h, mode="full")

def project_onto_P(v, total):
    """Project v onto the set {x : 0 <= x_i <= 1, sum x_i = total}"""
    # We want to solve for lambda such that sum(clip(v - lambda, 0, 1)) = total
    # This is a bisection problem on the decreasing function of lambda.
    v = np.asarray(v)
    lower = np.min(v) - 1
    upper = np.max(v)
    tol = 1e-12

    while upper - lower > tol:
        mid = (lower + upper) / 2
        candidate = np.clip(v - mid, 0, 1)
        if np.sum(candidate) > total:
            lower = mid
        else:
            upper = mid

    return np.clip(v - upper, 0, 1)

def smooth_gradient(h, beta):
    """Compute gradient of Φ_β(h) = (1/β) log(Σ_k exp(β F_k(h)))"""
    N = len(h)
    F = full_overlap(h)  # raw correlations (no scaling by dx)
    # Subtract max for numerical stability
    F_stable = F - np.max(F)
    F_stable = beta * F_stable
    expF = np.exp(F_stable)
    w = expF / np.sum(expF)  # soft-max weights

    # Compute gradient: ∇Φ_β = Σ_k w_k [1 - (h^{+k} + h^{-k})]
    g = np.ones(N)
    for k in range(-N + 1, N):
        wk = w[k + N - 1]  # shift to get the right index
        if k >= 0:
            # h shifted forward and h shifted backward
            g[:N - k] -= wk * h[k:]
        else:
            d = -k
            g[d:] -= wk * h[:N - d]
    return g

def projected_gradient_descent(h_start, time_budget, seed=42):
    np.random.seed(seed)
    start = time.time()
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = c5_of(h)
    N = len(h)
    total = N / 2
    beta = 2.0
    beta_max = 30.0
    β_update_interval = 100
    step = 0.5
    γ_step = 0.9995
    γ_beta = 1.5

    while time.time() - start < time_budget:
        # compute F and soft weights for current h
        F = full_overlap(h)
        # subtract max for numerical stability
        F_stable = F - np.max(F)
        F_stable = beta * F_stable
        expF = np.exp(F_stable)
        w = expF / np.sum(expF)

        # calculate gradient
        g = np.ones(N)
        for k in range(-N + 1, N):
            wk = w[k + N - 1]
            if k >= 0:
                g[:N - k] -= wk * h[k:]
            else:
                d = -k
                g[d:] -= wk * h[:N - d]

        # update
        v = h - step * g
        h = project_onto_P(v, total)
        cur_c5 = c5_of(h)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()

        # schedule updates
        step *= γ_step
        if np.floor((time.time() - start) / time_budget * 1000) % β_update_interval == 0:
            beta = min(beta * γ_beta, beta_max)

    return best_h, best_c5

def targeted_refinement(h_start, time_budget):
    """Perform deterministic pairwise mass transfer to improve the worst overlap"""
    np.random.seed(42)
    start = time.time()

    h = h_start.copy()
    best_h = h.copy()
    best_c5 = c5_of(h)

    alpha = 1.0  # transfer amount factor
    max_bad_iter = 200  # consecutive unsuccessful iterations
    bad_iter = 0

    while time.time() - start < time_budget:
        C = full_overlap(h)
        d = np.argmax(C) - (len(h) - 1)  # offset of maximum
        N = len(h)

        if d >= 0:
            hi = h[:N - d]
            hj = h[d:]
        else:
            hi = h[-d:]
            hj = h[:N - (-d)]

        c = hi * (1 - hj)
        i_star = np.argmax(c)
        c_max = c[i_star]

        if c_max <= 1e-10:  # no meaningful overlap left
            break

        # figure out direction of transfer
        if h[i_star] < 1 - hj[i_star]:
            delta = min(alpha * c_max, h[i_star], 1 - hj[i_star])
            h[i_star] -= delta
            hj[i_star] += delta
        else:
            delta = min(alpha * c_max, 1 - h[i_star], hj[i_star])
            h[i_star] += delta
            hj[i_star] -= delta

        # evaluate new C5
        cur_c5 = c5_of(h)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
            bad_iter = 0
        else:
            bad_iter += 1
            if bad_iter >= max_bad_iter:
                # Possibly stuck, try reducing alpha
                if alpha < 0.001:
                    break
                alpha /= 2
                bad_iter = 0

    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start = time.time()
    N = 200
    dx = 2.0 / N

    # Generate seeds
    # 1. Uniform distribution
    h_uniform = 0.5 * np.ones(N)
    h_uniform *= (N / 2) / np.sum(h_uniform)

    # 2. Initial from global variable if exists
    h_interpolated = None
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        x0 = np.linspace(0, 2, len(h0))
        x_new = np.linspace(0, 2, N)
        h0 = np.interp(x_new, x0, h0)
        h0 = np.clip(h0, 0, 1)
        h_interpolated = h0 * (N / 2) / np.sum(h0)

    # 3. Double peak
    h_double_peak = np.full(N, 0.1)
    a = 0.3
    idxs = np.where((np.linspace(0, 2, N) < a) | (np.linspace(0, 2, N) > (2 - a)))[0]
    h_double_peak[idxs] = 0.9
    h_double_peak *= (N / 2) / np.sum(h_double_peak)

    # 4. Random Dirichlet
    h_random = np.random.dirichlet([0.5] * N)
    h_random = np.clip(h_random, 0, 1)
    h_random *= (N / 2) / np.sum(h_random)

    # Run optimization on seeds
    best_h = None
    best_c5 = float('inf')
    seeds = [h_uniform, h_interpolated, h_double_peak, h_random]
    seeds = [s for s in seeds if s is not None]

    time_per_seed = 0.3 * budget_s / len(seeds)  # 30% per seed
    refined_time = 0.5 * budget_s  # remaining 50% for refinement phase

    for i, h_seed in enumerate(seeds):
        if time.time() - start > 0.9 * budget_s:
            break
        if h_seed is None:
            continue
        h_refined, c5_refined = projected_gradient_descent(h_seed, time_per_seed, seed + i)
        refined_start = time.time()
        h_final, c5_final = targeted_refinement(h_refined, refined_time)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final

    if best_h is None:
        # fallback to last best
        h = np.random.rand(N)
        h *= (N / 2) / np.sum(h)
        h, best_c5 = projected_gradient_descent(h, 0.1 * budget_s, seed)
        best_h, best_c5 = targeted_refinement(h, 0.2 * budget_s)

    return (best_h, best_c5, N)
