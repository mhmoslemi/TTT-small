import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

def build_h_from_coeffs(a_coeffs, n):
    """Build h vector from Fourier coefficients."""
    h = 0.5 * np.ones(n)
    for m, a in enumerate(a_coeffs, 1):
        h += a * np.cos(2 * np.pi * m * np.arange(n) / n)
    return np.clip(h, 0, 1)

def compute_overlap_and_gradient(h, dx, m, a_coeffs, n):
    """Compute overlap and gradient for a given shift and coeff."""
    # compute 1-h
    one_minus_h = 1 - h
    
    # compute correlations with all shifts
    corr_full = np.correlate(h, one_minus_h, mode="full")
    # shift indices s = -(n-1) to (n-1)
    # entry at index s+(n-1) is  f_s
    shifts = np.arange(-n+1, n)
    overlaps = corr_full * dx

    # gradient computation (eq. for partial f_s / partial a_m)
    cos_m_i = np.cos(2 * np.pi * m * np.arange(n) / n)
    # compute Ax = convolve(h, cos_m_i)
    # and Ay = convolve(one_minus_h, cos_m_i)
    Ax = np.correlate(h, cos_m_i, mode="full")
    Ay = np.correlate(one_minus_h, cos_m_i, mode="full")
    # values at indices 0 to n-1:
    Ax = Ax[n-1:]
    Ay = Ay[n-1:]
    # derivative = dx * (Ax - Ay)
    return overlaps, dx * (Ax - Ay)

def softmax_weights(overlaps, gamma):
    """Compute softmax weights for given gamma."""
    exps = np.exp(gamma * overlaps)
    return exps / np.sum(exps)

def penalty(h, lambda_penalty):
    """Penalty term if h out of the box [0,1]."""
    penalty_term = lambda_penalty * (np.maximum(0, h - 1) ** 2 + np.maximum(0, -h) ** 2)
    return np.sum(penalty_term)

def penalty_grad(h, a_coeffs, n, lambda_penalty):
    """Gradient of penalty."""
    h_clipped = np.clip(h, 0, 1)
    diff = h - h_clipped
    grad = np.zeros_like(a_coeffs)
    for m in range(len(a_coeffs)):
        cos_m_i = np.cos(2 * np.pi * (m + 1) * np.arange(n) / n)
        grad[m] = 2 * lambda_penalty * np.sum(diff * cos_m_i)
    return grad

def spectral_optim(h0, n, max_iter_per_stage=2000, gamma_schedule=(0.5, 40.0, 1.5), warmup=True, lambda_penalty=1e4):
    """Perform spectral optimisation."""
    cur_coeffs = np.zeros(n // 2)
    for i in range(len(h0)):
        if i < len(cur_coeffs):
            cur_coeffs[i] = h0[i]
    # Start from warm start if available
    h = build_h_from_coeffs(cur_coeffs, n)
    total_iter = 0
    # Start stages with increasing num of frequencies: M=8, 15, 24
    stages = [8, 15, 24]
    gamma_base = gamma_schedule[0]
    gamma_mult = gamma_schedule[2]
    final_gamma = gamma_schedule[1]
    # For each stage
    for stage_idx, M in enumerate(stages):
        if stage_idx > 0:
            # extend coefficients
            new_coeffs = np.zeros(M)
            new_coeffs[:len(cur_coeffs)] = cur_coeffs
            cur_coeffs = new_coeffs
        # adjust based on stage
        gamma = gamma_base if stage_idx == 0 else gamma_base * (gamma_mult ** (stage_idx - 1))
        if gamma > final_gamma:
            gamma = final_gamma
        # Use BFGS
        bounds = [(-0.6, 0.6)] * M
        # weights = []
        def obj_and_grad(a_coeffs):
            h = build_h_from_coeffs(a_coeffs, n)
            overlaps = np.correlate(h, 1 - h, mode="full") * (2.0 / n)
            # Combine using softmax
            w = softmax_weights(overlaps, gamma)
            # objective
            val = (np.sum(w * overlaps) + penalty(h, lambda_penalty))
            # gradient computation
            grad = np.zeros_like(a_coeffs)
            for m in range(len(a_coeffs)):
                _, df_da = compute_overlap_and_gradient(h, 2.0 / n, m, a_coeffs, n)
                grad[m] = np.sum(w * df_da) + penalty_grad(h, a_coeffs, n, lambda_penalty)[m]
            return val, grad
        # Optimization with projected gradient
        options = {"maxiter": max_iter_per_stage, "gtol": 1e-9}
        result = minimize(obj_and_grad, cur_coeffs, method='L-BFGSB',
                          jac=True, bounds=bounds, options=options)
        if result.success and total_iter + max_iter_per_stage < max_iter_per_stage * 3:
            cur_coeffs = result.x.copy()
            total_iter += max_iter_per_stage
        else:
            break
    return cur_coeffs

def project_to_simplex_box(h, target_sum, n):
    """Project h onto simplex (sum=h) and box [0,1]. Using water-filling."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-9:
        return h
    if current_sum > target_sum:
        # decrease values
        # sort descending
        sorted_idx = np.argsort(h)[::-1]
        idxs = sorted_idx[:n] # first n sorted
        # binary search over tau
        tau_lo, tau_hi = 0, h[sorted_idx[0]]
        while tau_hi - tau_lo > 1e-9:
            tau = (tau_lo + tau_hi) / 2
            new_h = np.clip(h - tau, 0, 1)
            if np.sum(new_h) < target_sum:
                tau_hi = tau
            else:
                tau_lo = tau
        tau = (tau_lo + tau_hi) / 2
        h = np.clip(h - tau, 0, 1)
    else:
        # increase values: equivalent to reducing negative part
        # but we just treat it via hesitant progress: increase until overflow
        # or just do full projection
        # Simple approach: decrease threshold to match sum
        # Use a normalizing trick
        diff = target_sum - current_sum
        # add diff evenly to all h
        h = np.clip(h + diff / n, 0, 1)
    return h

def polish_with_projected_gradient(h_start, n, maxiter=500):

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value scaled by dx
    max_overlap = np.max(correlation) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n = 240  # Sufficiently fine discretization
    dx = 2.0 / n
    best_C5 = float('inf')
    best_h = None
    restarts = 6
    target_sum = n / 2
    lambda_penalty = 1e4
    # Only optimize with restarts of initial random coefficient vectors
    for r in range(restarts):
        # initialize coefficient vector
        coeffs = np.random.normal(0, 0.05, 20)
        # now upgrade to sparse spectrum
        coeffs = np.zeros(24)  # Start since 24 is max M
        # fill coefficient using normal assumption
        for i in range(min(len(coeffs), 10)):
            coeffs[i] = np.random.normal(0, 0.05)
        # Generate an initial h and proceed
        h_initial = build_h_from_coeffs(coeffs[:24], n)
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = project_to_simplex_box(h_initial, target_sum, n)
        # Now perform spectral optimization on it
        coeffs_opt = spectral_optim(h_initial, n, max_iter_per_stage=500,
                                    gamma_schedule=(0.5, 40.0, 1.5),
                                    warmup=False)
        # Build and refine solution
        h_final = build_h_from_coeffs(coeffs_opt, n)
        h_final = np.clip(h_final, 0, 1)
        h_final = project_to_simplex_box(h_final, target_sum, n)
        h_opt = polish_with_projected_gradient(h_final, n, 300)
        C5 = evaluate_erdos_solution(h_opt)
        if C5 < best_C5:
            best_C5 = C5
            best_h = h_opt.copy()
    return (best_h.tolist(), best_C5, n)
