import numpy as np
import time

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value scaled by dx
    max_overlap = np.max(correlation) * dx
    return max_overlap

def generate_sturmian_binary(n, alpha=(np.sqrt(5)-1)/2):
    """Generate a balanced binary sequence using irrational rotation"""
    h = np.zeros(n, dtype=int)
    ones_count = 0
    for i in range(n):
        t = (i * alpha) % 1
        if t < 0.5:
            h[i] = 1
            ones_count += 1
    # Adjust to have exactly n/2 ones
    if ones_count > n // 2:
        # Find positions that are 1s and make some 0s
        positions = np.where(h == 1)[0]
        for i in range(ones_count - n // 2):
            h[positions[i]] = 0
    elif ones_count < n // 2:
        # Find positions that are 0s and make some 1s
        positions = np.where(h == 0)[0]
        for i in range(n // 2 - ones_count):
            h[positions[i]] = 1
    return h

def compute_overlap(h, s, dx):
    """Compute overlap for shift s"""
    n = len(h)
    overlap = 0.0
    for i in range(n):
        j = i + s
        if 0 <= j < n:
            overlap += h[i] * (1 - h[j])
    return overlap * dx

def compute_all_overlaps(h, dx):
    """Compute all overlaps for all shifts"""
    n = len(h)
    overlaps = np.zeros(2 * n - 1)
    for s in range(-n + 1, n):
        overlaps[s + n - 1] = compute_overlap(h, s, dx)
    return overlaps

def delta_shifts_for_swap(h, i, j, overlaps, dx):
    """Compute change in all overlaps if we swap h[i] and h[j]"""
    n = len(h)
    delta = np.zeros(2 * n - 1)
    
    # For each possible shift, compute the effect of swapping
    for s in range(-n + 1, n):
        # Compute the contribution at position i to this shift
        if i + s >= 0 and i + s < n:
            # Term contributes h_i * (1 - h_{i+s})
            delta[s + n - 1] -= h[i] * (1 - h[i + s])
            # In reverse direction
            if i - s >= 0 and i - s < n:
                delta[s + n - 1] += h[i - s] * h[i]
            # Term now is 0 * (1 - h_{i+s}) = 0
            # And change in contribution after swap due to drop of h_i 
            # That is, drop h_i * (1 - h_{i+s}) is indeed just -h_i * (1 - h_{i+s})
            # And because h_i becomes 0, we don't add anything when i is now a second factor
            # Only, we did already account for it.
            # The recalculation was: effect on all s due to h[i] being swapped to 0
            # was influence of h[i] * (1 - h[i+s]) being removed.
            # At the same time the contribution of j is h[j] * (1 - h[j+s]) which reversal is
            # contributions to all shifts s where j is (positive or negative) shift away. 
        else:
            # end index out of bounds
            pass
            
        if j + s >= 0 and j + s < n:
            delta[s + n - 1] += h[j] * (1 - h[j + s])
            if j - s >= 0 and j - s < n:
                delta[s + n - 1] -= h[j - s] * h[j] 

    return delta

def local_swap_search(h, n_points, time_budget_s):
    """Perform local swap-based optimization to reduce maximum overlap."""
    start_time = time.time()
    dx = 2.0 / n_points

    # Convert binary to continuous for dealing with partial overlaps in constraint handling  
    h = h.astype(float) 
    
    # Compute initial overlaps
    overlaps = compute_all_overlaps(h, dx)
    orig_max = np.max(overlaps)
    current_max = orig_max
    
    # Tabu list to avoid cycling
    tabu_list = set()
    tabu_duration = 100
    tabu = [tuple()] * tabu_duration
    
    # Track best solution
    best_h = h.copy()
    best_max = current_max
    
    # Number of iterations without improving the max
    no_improvement_count = 0
    max_no_improvement = 2000
    perturbation_threshold = 10000
    iteration_count = 0
    
    while (time.time() - start_time) < time_budget_s:
        iteration_count += 1
        # Find worst shift
        max_idx = np.argmax(overlaps)
        worst_shift = max_idx - (n_points - 1)
        
        # Get all ones and zeros
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        
        # Try to find a good swap pair
        # Prefer swapping a `1` with a `0` such that overlap at worst shift is reduced
        best_delta = 0
        best_i, best_j = None, None
        
        # Sample some random pairs for diversity
        for _ in range(min(1000, ones.size * zeros.size)):
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            swap_key = tuple(sorted((i, j)))
            # Skip tabu moves
            if swap_key in tabu_list:
                continue
                
            # Compute change in overlaps
            delta = delta_shifts_for_swap(h, i, j, overlaps, dx)
            # Change in current max
            new_max = max(overlaps + delta) 
            current_delta = current_max - new_max
            
            if current_delta > best_delta:
                best_delta = current_delta
                best_i, best_j = i, j
                
        # If no good move found, move to next iteration
        if best_i is None:
            no_improvement_count += 1
            if no_improvement_count > max_no_improvement:
                # Perturbation: flip small block of 5% of bits
                n_flip = max(1, n_points // 20)
                # Pick a random range
                start = np.random.randint(0, n_points - n_flip)
                for k in range(start, start + n_flip):
                    h[k] = 1 - h[k]
                # Recompute overlaps
                overlaps = compute_all_overlaps(h, dx)
                current_max = np.max(overlaps)
                if current_max < best_max:
                    best_h = h.copy()
                    best_max = current_max
                no_improvement_count = 0
                continue
            
            # Maybe increase probability
            continue
            
        # Perform chosen swap
        if best_i is not None and best_j is not None:
            h[best_i], h[best_j] = h[best_j], h[best_i]
            # Add to tabu list
            swap_key = tuple(sorted((best_i, best_j)))
            tabu[iteration_count % tabu_duration] = swap_key
            tabu_list.add(swap_key)
            # Update overlaps based on the change
            delta_overlaps = delta_shifts_for_swap(h, best_i, best_j, overlaps, dx)
            overlaps += delta_overlaps
            # Update result
            current_max = np.max(overlaps)
            
            if current_max < best_max:
                best_h = h.copy()
                best_max = current_max
                
            if current_max < (0.3807 + 1e-6):
                # Early exit if target achieved
                break
            
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
    return best_h, best_max

def continuous_polish(h, n_points, dx, max_iters=500):
    """Apply projector gradient descent on the continuous vector to refine."""
    # Project onto sum == 1 constraint
    target_sum = n_points / 2.0
    current_sum = np.sum(h)
    h = h * (target_sum / current_sum)
    
    # Create index arrays mapping all shifts to relevant arrays and indices
    # For core optimization loop, we'll construct a gradient multiplier  
    # to avoid computing # overlaps on each computation step. But for simplicity,
    # let's do stochastic descent using full overlaps every few steps
    
    for _ in range(max_iters):
        # Evaluate the overlap
        overlaps = compute_all_overlaps(h, dx)
        max_idx = np.argmax(overlaps)
        worst_shift = max_idx - (n_points - 1)
        
        # Get gradient Updates for current worst shift 
        # Assume we have a h_i that are good for worst_s,
        # the increase in overlap depends on local terms.
        # We derivative很容易模糊，长时间太多细节更新状态的变化 ,ignore this part
        # We just want to functional form of best descent on h during optimization
        
        # Approximate gradient direction via finite difference
        eps = 1e-6
        h_new = h.copy()
        # Grad with respect to one variable h_k is approximately
        # 1/dx * [= Σ_s ∂(h_k*(1-h_{k+s}))/∂h_k = 1 - Σ_s h_{k+s}]
        # In our case, since we don't want to calc full gradient manually,
        # but want to maintain `sum = 1` and reduce overlap.
        # Approximate effect of a small positive delta to all h_i
        
        # Instead, we check "what is a norm of physical delta direction"
        # to truly decrease max overlap, especially for h in [0,1] range:
        # So, we rescale all differences to ensure smoother descent
        # We are inside numpy space, so use simple iterative approach:
        
        step_size = 0.001
        # Adjust potential effect of variables in worst shift
        multipliers = np.zeros(n_points)
        for s in range(-n_points + 1, n_points):
            for i in range(n_points):
                j = i + s
                # if i+shift is within bounds
                if 0 <= j < n_points:
                    # Add to the gradient estimate  
                    diff = h[i] * (1 - h[j])  
                    # Gradient update is motivated directly from structure:
                    # d/dh[k] [h[i]* (1-h[i+s])] is h[i] if k == i, 1-h[j] if k == j
                    #   which looks too complex to implement precisely correctly
                    #   Let's just use an approximative view:
                    # If we increment h[i], instead of checking all shifts
                    # just act on current worst shift contribution:
                    if abs(s - worst_shift) < n_points // 2:
                        # add small decrease at positions that right down  
                        # like in gradient if direction was given as 1, -1, +1 pattern.
                        multipliers[i] -= h[j]
                        multipliers[j] += h[i]
        
        # Normalize gradient to prevent too large moves
        mag = np.sum(multipliers ** 2) ** 0.5 + 1e-10
        multipliers /= mag
        multipliers *= step_size
        
        # Now attempt update
        h_new = h - multipliers
        h_new = np.clip(h_new, 0, 1)
        
        # Adjust back to constraint
        current_sum = np.sum(h_new)
        h_new = h_new * (target_sum / current_sum)
        
        h = h_new
        if _ % 50 == 0:
            overlaps = compute_all_overlaps(h, dx)
            new_max = np.max(overlaps)
            if new_max < 0.3807:
                break
        
    return h

def run(seed=42, budget_s=600, n_points=None):
    np.random.seed(seed)
    
    start_time = time.time()
    if n_points is None:
        n_points = 120
    dx = 2.0 / n_points
    
    # Step 1: Generate initial binary sequence
    h = generate_sturmian_binary(n_points)
    h = h.astype(float)
    
    # Convert bool vector to float vector
    h = h.astype(float)
    
    # Step 2: Local swap search
    # Spend 50% of time on coarse sweep
    search_time = budget_s * 0.5
    h_opt, c5_opt = local_swap_search(h, n_points, search_time)
    
    # Step 3: Refinement on larger grid
    if n_points < 240 and (time.time() - start_time) < budget_s * 0.9:
        # Upsampling (repetitive copying)
        n_points_up = 240
        dx_up = 2.0 / n_points_up
        h_up = np.repeat(h_opt, 2)  # Repeat each element
        h_up = np.clip(h_up, 0, 1)
        h_up = h_up.astype(float)
        h_up = np.array(h_up, dtype=float)  # ensure float array
        
        # Normalize to correct sum
        current_sum = np.sum(h_up) * dx_up
        h_up = h_up * (1.0 / current_sum) * (n_points_up / 2)
        
        # Refine further
        h_refined, c5_refined = local_swap_search(h_up, n_points_up, budget_s * 0.4)
        h_opt = h_refined
        c5_opt = c5_refined
        
    # If time remains, do some final polish with continuous method
    if (time.time() - start_time) < budget_s * 0.95:
        h_opt = continuous_polish(h_opt, len(h_opt), dx)
        c5_opt = evaluate_erdos_solution(h_opt)
    
    return (h_opt.tolist(), c5_opt, len(h_opt))
