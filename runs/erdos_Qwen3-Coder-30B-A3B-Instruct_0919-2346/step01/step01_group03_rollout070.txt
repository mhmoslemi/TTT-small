import numpy as np
import time

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral via full convolution of h with (1-h)
    overlaps = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum of the full correlation vector
    # scaled by dx
    c5 = np.max(overlaps) * dx
    return c5

def compute_c5_and_worst_info(h):
    """Compute C5 and worst shift info"""
    N = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    max_idx = np.argmax(corr)
    max_val = corr[max_idx]
    offset = max_idx - (N - 1)
    
    # contributions vector:
    if offset >= 0:
        hi = h[:N - offset]
        hj = h[offset:]
        contrib = hi * (1 - hj)
        i_local = np.argmax(contrib)
        i_index = i_local
        j_index = i_local + offset
    else:
        off = -offset
        hi = h[off:]
        hj = h[:N - off]
        contrib = hi * (1 - hj)
        i_local = np.argmax(contrib)
        i_index = i_local + off
        j_index = i_local
    
    return max_val * (2.0/N), offset, i_index, j_index, hi[i_local], hj[i_local]

def propose_direct_move(h, i, j, step):
    """Propose a move"""
    delta = min(step, h[i], 1 - h[j])
    if delta <= 0:
        return None
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new

def targeted_descent(h0, time_budget, start_time):
    """Run targeted descent"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of(h)
    
    # step starts at maybe 0.5 (max feasible change)
    step = 0.5
    fails = 0
    
    while time.time() - start_time < time_budget:
        # compute worst info
        _, _, i, j, hi_val, hj_val = compute_c5_and_worst_info(h)
        
        # compute delta
        delta = min(step, h[i], 1 - h[j])
        if delta > 0:
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            c5_new = c5_of(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
                # maybe increase step
                step = min(step * 1.1, 0.5)
                fails = 0
                continue
        
        # if neither improves, reduce step
        step *= 0.9
        fails += 1
        if fails > 2000:
            # random restart
            idxs = np.random.choice(len(h), size=2, replace=False)
            delta_r = step * min(h[idxs[0]], 1 - h[idxs[1]])
            if delta_r > 0:
                h = propose_direct_move(h, idxs[0], idxs[1], step)
                fails = 0
        
        # occasionally also try random small move to escape local minima
        if np.random.rand() < 0.01:
            i_rand, j_rand = np.random.choice(len(h), size=2, replace=False)
            delta_r = step * min(h[i_rand], 1 - h[j_rand])
            if delta_r > 0:
                h_new = h.copy()
                h_new[i_rand] -= delta_r
                h_new[j_rand] += delta_r
                c5_new = c5_of(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new
    
    return best_h, best_c5
