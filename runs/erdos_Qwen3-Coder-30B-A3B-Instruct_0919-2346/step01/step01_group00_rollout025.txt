import numpy as np
import time
import random
from scipy.optimize import minimize

def project_h(h, target_sum):
    """Project h onto box constraints [0,1] and sum constraint."""
    h_clipped = np.clip(h, 0, 1)
    current = np.sum(h_clipped)
    
    if np.isclose(current, target_sum):
        return h_clipped
    
    # Use binary search to find lambda
    if current > target_sum:
        # Need to reduce sum: decrease values
        low, high = 0.0, 1.0
        while high - low > 1e-10:
            mid = (low + high) / 2.0
            h_test = np.clip(h_clipped - mid, 0, 1)
            if np.sum(h_test) < target_sum:
                high = mid
            else:
                low = mid
        lambda_val = low
        h_proj = np.clip(h_clipped - lambda_val, 0, 1)
    else:
        # Need to increase sum: increase values
        low, high = 0.0, 1.0
        while high - low > 1e-10:
            mid = (low + high) / 2.0
            h_test = np.clip(h_clipped + mid, 0, 1)
            if np.sum(h_test) > target_sum:
                high = mid
            else:
                low = mid
        lambda_val = low
        h_proj = np.clip(h_clipped + lambda_val, 0, 1)
    
    return h_proj

def compute_C_and_grad(h, alpha):
    """Compute C_5 and gradient using smoothing."""
    N = len(h)
    dx = 2.0 / N
    
    # Compute autocorrelation of h
    fft_h = np.fft.fft(h, n=2*N-1)
    S = np.fft.ifft(fft_h * np.conj(fft_h)).real[:2*N-1]
    C = 1.0 - dx * S
    
    # Compute smoothed objective and gradient
    w = np.exp(alpha * C)
    w = w / np.sum(w)
    
    # Gradient computing using convolutions - v_plus corresponds to h shifted forward
    # (imagine w_rev as w with reversed index) - doing convolution of h with w_rev
    w_rev = w[::-1]
    v_plus = np.convolve(h, w_rev, mode='same')
    v_minus = np.convolve(h, w, mode='same')
    
    grad = dx * (1.0 - (v_plus + v_minus))
    
    # Return max C and gradient
    return np.max(C), grad
