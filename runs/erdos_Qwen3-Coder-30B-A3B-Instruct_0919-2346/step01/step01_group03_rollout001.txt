import numpy as np
import time

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral via full convolution of h with (1-h)
    overlaps = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum of the full correlation vector
    # scaled by dx
    c5 = np.max(overlaps) * dx
    return c5

def compute_c5_and_worst_info(h):
    """Compute C5 and information about worst overlap"""
    N = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    max_idx = np.argmax(corr)
    max_val = corr[max_idx]
    offset = max_idx - (N - 1)
    # contributions vector:
    if offset >= 0:
        hi = h[:N - offset]
        hj = h[offset:]
        contrib = hi * (1 - hj)
        i_local = np.argmax(contrib)
        i_index = i_local
        j_index = i_local + offset
    else:
        off = -offset
        hi = h[off:]
        hj = h[:N - off]
        contrib = hi * (1 - hj)
        i_local = np.argmax(contrib)
        i_index = i_local + off
        j_index = i_local
    # return values
    return max_val * (2.0/N), offset, i_index, j_index, hi[i_local], hj[i_local], contrib[i_local]

def propose_direct_move(h, i, j, step):
    """Propose a move from i to j"""
    delta = min(step, h[i], 1 - h[j])
    if delta <= 0:
        return None
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new

def targeted_descent(h, time_budget, start_time):
    """Perform targeted descent to improve solution"""
    N = len(h)
    best_h = h.copy()
    best_c5 = c5_of(h)
    step = 0.5
    fails = 0
    max_fails = 2000
    
    while time.time() - start_time < time_budget:
        # compute worst info
        _, offset, i, j, hi_val, hj_val, _ = compute_c5_and_worst_info(h)
        
        # Try move from i to j (hi -> hj)
        delta = min(step, h[i], 1 - h[j])
        if delta > 0:
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            c5_new = c5_of(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
                step = min(step * 1.1, 0.5)
                fails = 0
                continue
        
        # Try move from j to i
        delta2 = min(step, h[j], 1 - h[i])
        if delta2 > 0:
            h_new2 = h.copy()
            h_new2[j] -= delta2
            h_new2[i] += delta2
            c5_new2 = c5_of(h_new2)
            if c5_new2 < best_c5:
                best_c5 = c5_new2
                best_h = h_new2.copy()
                h = h_new2
                step = min(step * 1.1, 0.5)
                fails = 0
                continue
                
        # If no improvement, reduce step or try random move
        step *= 0.9
        fails += 1
        if fails > max_fails:
            # Random restart
            idxs = np.random.choice(N, size=2, replace=False)
            delta_r = step * min(h[idxs[0]], 1 - h[idxs[1]])
            if delta_r > 0:
                h = propose_direct_move(h, idxs[0], idxs[1], step)
                fails = 0
        elif np.random.rand() < 0.01:
            # Occasionally try random small move
            i_rand, j_rand = np.random.choice(N, size=2, replace=False)
            delta_r = step * min(h[i_rand], 1 - h[j_rand])
            if delta_r > 0:
                h_new = h.copy()
                h_new[i_rand] -= delta_r
                h_new[j_rand] += delta_r
                c5_new = c5_of(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new
                
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start = time.time()
    
    # Use 200 points for higher resolution
    N = 200
    dx = 2.0 / N
    
    # Initialize seeds
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        if len(h0) != N:
            x0 = np.linspace(0, 2, len(h0))
            x_new = np.linspace(0, 2, N)
            h0 = np.interp(x_new, x0, h0)
        h_initial = h0 * (N / 2) / np.sum(h0)
    else:
        h_initial = np.random.rand(N) * 0.8 + 0.1  # avoid boundary values
        h_initial *= (N / 2) / np.sum(h_initial)

    # Uniform seed
    h_uniform = 0.5 * np.ones(N)
    h_uniform *= (N / 2) / np.sum(h_uniform)
    
    # Double-peak seed
    h_double_peak = np.full(N, 0.1)
    a = 0.3
    idxs = np.where((np.linspace(0, 2, N) < a) | (np.linspace(0, 2, N) > (2 - a)))[0]
    h_double_peak[idxs] = 0.9
    h_double_peak *= (N / 2) / np.sum(h_double_peak)
    
    # Run targeted descent on seeds
    best_h = None
    best_c5 = float('inf')
    
    # Each worker gets 0.8 * budget_s
    per_run_budget = 0.5 * budget_s
    seeds = [h_uniform, h_initial, h_double_peak]
    
    for i, h_seed in enumerate(seeds):
        if time.time() - start > budget_s * 0.9:
            break
        h, c5 = targeted_descent(h_seed, per_run_budget, start)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5
            
    # Final polishing with longer iteration
    if best_h is not None and time.time() - start < budget_s * 0.95:
        polishing_time = budget_s - (time.time() - start)
        h_polish, c_polish = targeted_descent(best_h, polishing_time, start)
        if c_polish < best_c5:
            best_h = h_polish
            best_c5 = c_polish

    return (best_h, best_c5, N)
