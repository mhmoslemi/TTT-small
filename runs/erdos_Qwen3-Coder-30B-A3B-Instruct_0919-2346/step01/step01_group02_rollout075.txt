import numpy as np
import time
from scipy.optimize import minimize
from scipy.special import logsumexp

def c5_of(h):
    """Compute the maximum overlap integral for the given step function h."""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    c = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(c) * dx
    return c5

def clip_v(v, a, b):
    """Clip v element-wise to [a, b]"""
    return np.clip(v, a, b)

def project_to_sum_constraint(h, target_sum):
    """Project h to the set {x: sum(x) = target_sum, 0 <= x <= 1}."""
    N = len(h)
    v = h.copy()
    # Binary search for lambda such that sum clip(v + lambda) = target_sum
    lambda_min, lambda_max = -1000.0, 1000.0
    for _ in range(50):
        lambda_mid = (lambda_min + lambda_max) / 2.0
        s = np.sum(clip_v(v + lambda_mid, 0, 1))
        if s < target_sum:
            lambda_max = lambda_mid
        else:
            lambda_min = lambda_mid
    lambda_opt = lambda_min
    h_proj = clip_v(v + lambda_opt, 0, 1)
    # Small correction to ensure exact sum
    diff = target_sum - np.sum(h_proj)
    # Find indices that can be adjusted
    free = np.where((h_proj > 0) & (h_proj < 1))[0]
    if len(free) > 0:
        i = free[0]
        h_proj[i] += diff
        h_proj = clip_v(h_proj, 0, 1)
    return h_proj

def smooth_surrogate(h, eps):
    """Evaluate the smooth surrogate F_epsilon(h) and its gradient."""
    dx = 2.0 / len(h)
    # Compute all correlations C_s = dx * sum_i h[i] * (1 - h[i+s])
    c = np.correlate(h, 1 - h, mode='full')
    c *= dx
    # Log-sum-exp with numerical stability
    max_c = np.max(c)
    c_shifted = c - max_c
    logZ = max_c + logsumexp(c_shifted / eps)
    # Gradient w.r.t. h
    w = np.exp(c_shifted / eps)
    w /= np.sum(w)
    # Gradient calculations
    grad = np.zeros_like(h)
    # This replicates the term ∇C_s = dx * ( 1 - shift_left(h,s) - shift_right(h,s) )
    # for each shift s.
    # For every s: s < len(h), i = 0,...,len(h)-1
    # Correlation term (for shift s): h[i]* (1 - h[i+s]), i<s, i+s < len(h)
    #=> gradient is 1 - h[i+s] - h[i-s]
    # This is vectorized via convolution - build full gradient matrix
    # What we want is: 
    # grad_i = sum_s w_s * dx * ( 1 - h[i+s] - h[i-s] ) for valid i,s
    # = dx * sum_s w_s - sum_s w_s * h[i+s] - sum_s w_s * h[i-s]
    # = dx * sum_s w_s - sum_s w_s * shift_left(h,s) - sum_s w_s * shift_right(h,s)
    
    # Let's compute shift-left and shift-right directly
    # The full correlation result has length 2*N-1; we want shift_left shift_right for s = 0..2N-2
    # But we only care about shifts s=0..N-1 (the negative shifts may not be needed)
    # For shift s>0, (i+s < len(h)) in the h[i]*(1-h[i+s]) term
    # So shift_left maps h[i] to h[i-s], and we use 0-padding (tail) for shifts
    # For consistency with numpy default shift convention:
    # shift_left: each entry h[i] moved to h[i-s], with h[i-s]=0 for i<s
    # shift_right: h[i] moved to h[i+s], h[i+s]=0 for i+s>=N
    
    # It's faster to compute them correctly (with zero padding) and not use convolution operations
    
    # Instead, compute directly for s from 0 to 2*N-2:
    # For i in 0..N-1:
    # For s in 0..2*N-2:
    #   contribution to grad[i] = w[s] * dx * (1 - h[i+s] - h[i-s])
    # But we must be careful with boundary conditions
    
    # But actually just use a clear counting of terms:
    grad_full = np.zeros(2 * len(h) - 1)
    for s in range(len(h)):  # originally 0 to N-1
        # Right shift term
        if s == 0:
            # h[i] -> h[i] 
            grad_full[len(h)-1:] += -w[s] * dx * h
        else:
            # h[i] -> h[i+s] - right shifted
            shifted = np.zeros((len(h),))
            shifted[s:]; shifted[:s] = 0  # == shift_right(h, s)
            grad_full[len(h)-1+len(h)-s:] += -w[s] * dx * shifted
            # Left shift term
            shifted = np.zeros((len(h),))
            shifted[:s] = 0
            shifted[s:] = h[:(len(h)-s)]
            grad_full[len(h)-1-s: len(h)-1] += -w[s] * dx * shifted

    grad = grad_full[len(h)-1:]
    return logZ, grad

def continuation_optimization(h0, N, eps_sequence, budget_s):
    """Run continuation optimization on h0."""
    dx = 2.0 / N
    h = h0.copy()
    h = project_to_sum_constraint(h, N / 2.0)
    best_h = h.copy()
    best_c5 = c5_of(h)
    
    for eps in eps_sequence:
        # Define objective and gradient for this eps
        def obj_grad(h):
            # Ensure h is in [0,1]
            h = np.clip(h, 0, 1)
            val, grad = smooth_surrogate(h, eps)
            return val, grad
        
        # Inner loop: L-BFGS-B
        bounds = [(0, 1)] * N
        result = minimize(lambda h: smooth_surrogate(h, eps)[0], h, method='L-BFGS-B',
                          bounds=bounds, jac=lambda h: smooth_surrogate(h, eps)[1],
                          options={'maxiter': 150, 'gtol': 1e-6})
        h = result.x
        h = project_to_sum_constraint(h, N / 2.0)
        cur_c5 = c5_of(h)
        if cur_c5 < best_c5:
            best_h = h.copy()
            best_c5 = cur_c5
        # One final projection for numerical stability
        h = project_to_sum_constraint(h, N / 2.0)

    return best_h, best_c5

def generate_seeds(N):
    """Generate diverse seeds for multi-start approach."""
    seeds = []
    # i) Uniform seed
    h = np.ones(N) * 0.5
    seeds.append(("uniform", h))
    
    # ii) Two-peak seeds
    a_vals = [0.12, 0.18, 0.24, 0.30]
    for a in a_vals:
        h = np.ones(N) * 0.1
        x_vals = np.linspace(0, 2, N)
        mask_left = x_vals < a
        mask_right = x_vals >= (2 - a)
        mask = mask_left | mask_right
        h[mask] = 0.9
        h = h * (N/2) / np.sum(h)  # Renormalize
        seeds.append(("two_peak_a=%.2f" % a, h))
    
    # iii) Binary seeds
    half = N // 2
    for i in range(3):
        h = np.zeros(N)
        indices = np.random.choice(N, size=half, replace=False)
        h[indices] = 1.0
        h = h * (N/2) / np.sum(h)
        seeds.append(("binary_%d" % i, h))
    
    # iv) Symmetrical binary seed
    h = np.zeros(N)
    h[:half] = 1.0
    h = h * (N/2) / np.sum(h)
    seeds.append(("binary_sym", h))
    
    # v) Random continuous seed
    for i in range(3):
        h = np.random.uniform(0, 1, N)
        h = h * (N/2) / np.sum(h)
        seeds.append(("random_%d" % i, h))
    
    return seeds

def sub_gradient_refinement(h, t_max):
    """Refine solution with sub-gradient steps on original function."""
    dx = 2.0 / len(h)
    t0 = time.time()
    while time.time() - t0 < t_max:
        # Compute all overlap integrals
        c = np.correlate(h, 1 - h, mode='full')
        s_star = np.argmax(c)
        # Compute sub-gradient for shift s_star:
        # ∂C_s/∂h_i = dx*(1 - h[i+s_star] - h[i-s_star])
        # But we need to be careful with boundaries.
        g = np.zeros_like(h)
        if s_star >= len(h):
            # Handle (reverse) shift, s = s_star - len(h)  
            shift = s_star - len(h)+1
            g[:-shift] = dx * (1 - h[shift:])
        else:
            g[s_star:] = dx * (1 - h[:-s_star])
        if s_star >= len(h):
            shift = s_star - len(h)+1
            g[shift:] += dx * (1 - h[:-shift])
        else:
            g[:-s_star] += dx * (1 - h[s_star:])
        # Project gradient to feasible direction
        g = -g  # because we want to minimize
        # Step-size = small
        eta = 0.05
        h_new = h - eta * g
        h_new = project_to_sum_constraint(h_new, len(h) / 2.0)
        c_new = c5_of(h_new)
        if c_new < c5_of(h):
            h = h_new
        else:
            break
    return h

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    N = 140
    dx = 2.0 / N
    MAX_SEEDS = 20  # Use max 20 seeds
    
    # Generate seeds
    seeds = generate_seeds(N)
    seeds = seeds[:MAX_SEEDS]  # limit number of seeds
    
    # Parameters for continuation
    eps_sequence = [0.10, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    best_h = None
    best_c5 = float('inf')
    
    # 70% of budget for continuation
    time_for_continuation = 0.7 * budget_s
    start = time.time()
    seeds_done = 0
    for name, h0 in seeds:
        if time.time() - start > time_for_continuation:
            break
        if np.sum(h0) != N/2.0:
            h0 = h0 * (N/2) / np.sum(h0)  # Renormalize if need be
        h_opt, c5_opt = continuation_optimization(h0, N, eps_sequence, time_for_continuation*0.9)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
        seeds_done += 1
    
    # 10% of budget for polishing
    polishing_time = 0.1 * budget_s
    if best_h is not None:
        best_h = sub_gradient_refinement(best_h, polishing_time)
        best_c5 = c5_of(best_h)
    
    # Return final result
    return (best_h, best_c5, N)
