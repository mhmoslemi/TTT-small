import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value scaled by dx
    max_overlap = np.max(correlation) * dx
    return max_overlap

def project_onto_bounds_sum(v, target_sum, lower=0, upper=1):
    """Project vector v onto box constraints [lower,upper] and sum=target_sum"""
    # Binary search for the threshold
    lo = v.min() - upper
    hi = v.max() - lower
    for _ in range(60):
        mid = (lo + hi) / 2
        projected = np.clip(v - mid, lower, upper)
        if projected.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2, lower, upper)

                if shift > -len(h):

            else:
