import numpy as np
import time

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral via full convolution of h with (1-h)
    overlaps = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum of the full correlation vector
    # scaled by dx
    c5 = np.max(overlaps) * dx
    return c5

def full_overlap(h):
    """Return the raw overlap correlation vector (unscaled)."""
    return np.correlate(h, 1 - h, mode="full")

def project_onto_P(v, total):
    """Project v onto the set {x : 0 <= x_i <= 1, sum x_i = total}."""
    # Apply box constraints
    v = np.clip(v, 0, 1)
    # Adjust sum by bisection
    if np.sum(v) == total:
        return v
    
    def sum_minus_total(lam):
        x = np.clip(v - lam, 0, 1)
        return np.sum(x) - total
    
    # Since sum_minus_total is decreasing, we can use bisection.
    # The function is continuous and differentiable almost everywhere.
    # It is also differentiable in the sense of subgradients.
    
    # However, a more direct approach:
    # We need to find lambda such that sum(clip(v-lambda,0,1)) = total.
    # We know that min(v) <= sum(v) <= max(v)
    a, b = np.min(v), np.max(v)
    # Normalized version.
    if np.sum(v) > total:
        # We want to increase lambda to decrease sum
        low, high = a - 1, b
        while high - low > 1e-12:
            mid = (low + high) / 2
            s = np.sum(np.clip(v - mid, 0, 1))
            if s >= total:
                low = mid
            else:
                high = mid
        lam = low
    else:
        # sum(v) < total, so we need lower lambda to increase sum
        low, high = a, b + 1
        while high - low > 1e-12:
            mid = (low + high) / 2
            s = np.sum(np.clip(v - mid, 0, 1))
            if s <= total:
                low = mid
            else:
                high = mid
        lam = low
    
    return np.clip(v - lam, 0, 1)

def smooth_gradient(h, beta):
    """Return gradient of Φ_β(h) = 1/β * log(sum_k exp(β * F_k(h)))"""
    # Compute vector of F_k(h) values (correlations)
    F_all = full_overlap(h)
    N = len(h)
    # Stable subtraction to prevent overflow
    F_max = np.max(F_all)
    exp_F = np.exp(beta * (F_all - F_max))
    Z = np.sum(exp_F)
    w = exp_F / Z  # Softmax weighted by beta
    
    # Gradient of Φ_β(h) = sum_k w_k * F_k(h)
    # Recall F_k(h) = sum_i h_i * (1 - h_{i+k})
    # and so dF_k(h)/dh_i = 1 - (h_{i+k} + h_{i-k})
    g = np.ones(N)
    
    for k in range(-N+1, N):
        wk = w[k+N-1]  # Shift index to 0-based for indexing w
        if k >= 0:
            # g[i] -= wk * h[i+k] for i in range(N-k)
            g[:-k] -= wk * h[k:]
        else:
            # g[i] -= wk * h[i+k] for i in range(-k, N) => i in range(N+k) 
            g[-k:] -= wk * h[:N+k]
    
    return g

def projected_gradient_descent(h0, time_budget, seed=None):
    if seed is not None:
        np.random.seed(seed)
    start = time.time()
    h = h0.copy()
    c_best = c5_of(h)
    best_h = h.copy()
    
    # Parameters
    beta_start = 2.0
    beta_max = 30.0
    gamma_beta = 1.5
    gamma_step = 0.999
    step_start = 0.5
    N = len(h)
    total = N / 2
    
    beta = beta_start
    step = step_start
    max_iter = 20000
    beta_update_interval = 100
    
    for iteration in range(max_iter):
        if time.time() - start > time_budget:
            break
            
        # Compute correlation F_k(h)
        F_all = full_overlap(h)
        
        # Compute softmax weights
        F_max = np.max(F_all)
        exp_F = np.exp(beta * (F_all - F_max))
        Z = np.sum(exp_F)
        w = exp_F / Z
    
        # Gradient of Φ_β(h)
        g = smooth_gradient(h, beta)
        
        # Tentative update
        v = h - step * g
        
        # Project onto feasible set
        h = project_onto_P(v, total)
        
        # Compute actual C5
        c_new = c5_of(h)
        
        if c_new < c_best:
            c_best = c_new
            best_h = h.copy()
        
        # Update schedules
        step *= gamma_step
        if iteration % beta_update_interval == 0 and beta < beta_max:
            beta = min(beta * gamma_beta, beta_max)
    
    return (best_h, c_best)

def targeted_refinement(h, time_budget):
    """Improve solution by reducing the maximum overlap."""
    start = time.time()
    h = h.copy()
    c_prev = c5_of(h)
    
    # Small scaling factor for mass transfer
    alpha = 1.0
    max_no_improvement = 200
    no_improvement = 0
    
    while time.time() - start < time_budget:
        # Get all overlaps
        C = full_overlap(h)
        d = np.argmax(C) - (len(h) - 1)  # shift index (d = argmax(C) - (N-1))
        
        # Extract overlapping slices
        N = len(h)
        if d >= 0:
            hi = h[:N-d]
            hj = h[d:]
        else:
            hi = h[-d:]
            hj = h[:N-(-d)]
        
        # Find largest contribution to overlap
        c = hi * (1 - hj)
        i_star = np.argmax(c)
        h_i_star = hi[i_star]
        h_j_star = hj[i_star]
        c_max = c[i_star]
        
        # Compute mass transfer amount
        delta = min(alpha * c_max, h_i_star, 1 - h_j_star)
        
        # Apply transfer
        h[i_star] -= delta
        h[i_star + d if d >= 0 else i_star - d] += delta
        
        # Evaluate
        c_new = c5_of(h)
        if c_new < c_prev:
            c_prev = c_new
            no_improvement = 0
            alpha = 1.0  # Reset alpha after successful step
        else:
            # Undo and decrease alpha
            h[i_star] += delta
            h[i_star + d if d >= 0 else i_star - d] -= delta
            alpha *= 0.5
            no_improvement += 1
            if alpha < 1e-8 or no_improvement >= max_no_improvement:
                break
    
    return (h, c_prev)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start = time.time()
    
    # Parameters
    N = 200
    dx = 2.0 / N
    total = N / 2
    
    # Generate seeds
    # 1. Uniform seed
    h_uniform = 0.5 * np.ones(N)
    
    # 2. Interpolated initial_h_values (if available)
    h_initial = None
    if 'initial_h_values' in globals() and len(initial_h_values) > 0:
        h0 = np.array(initial_h_values)
        x0 = np.linspace(0, 2, len(h0))
        x_new = np.linspace(0, 2, N)
        h0_interp = np.interp(x_new, x0, h0)
        h_initial = h0_interp * total / np.sum(h0_interp)
    if h_initial is None:
        h_initial = np.random.rand(N) * 0.5
        h_initial *= total / np.sum(h_initial)
    
    # 3. Double-peak seed
    h_double_peak = np.full(N, 0.1)
    a = 0.3
    idxs = np.where((np.linspace(0, 2, N) < a) | (np.linspace(0, 2, N) > (2 - a)))[0]
    h_double_peak[idxs] = 0.9
    h_double_peak *= total / np.sum(h_double_peak)
    
    # 4. Random Dirichlet-based seed
    # Sample from Dirichlet(0.5) (flatter than uniform, spread-out)
    h_random_dirichlet = np.random.dirichlet([0.5] * N)
    h_random_dirichlet = np.clip(h_random_dirichlet, 0, 1)
    h_random_dirichlet *= total / np.sum(h_random_dirichlet)
    
    seeds = [h_uniform, h_initial, h_double_peak, h_random_dirichlet]
    
    # Run projected gradient descent on all seeds
    best_h = None
    best_c5 = float('inf')
    
    # Budget per seed
    budget_per_seed = 0.3 * budget_s / len(seeds)
    
    for i, h_seed in enumerate(seeds):
        if time.time() - start > 0.9 * budget_s:
            break
        h_opt, c5_opt = projected_gradient_descent(h_seed, budget_per_seed, seed=seed+i)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
    
    if best_h is None:
        return (np.ones(N) * 0.5, 0.5, N)
    
    # Then refine with targeted method
    remaining_time = budget_s - (time.time() - start)
    if remaining_time > 10:
        h_final, c5_final = targeted_refinement(best_h, remaining_time * 0.8)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final

    return (best_h, best_c5, N)
