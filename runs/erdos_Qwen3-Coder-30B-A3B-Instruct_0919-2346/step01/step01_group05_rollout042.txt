import numpy as np
import time

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    correlation = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(correlation) * dx
    return max_overlap

def upsample_initial(h0, n_target):
    """Upsample h0 to length n_target"""
    x_old = np.linspace(0, 2, len(h0))
    x_new = np.linspace(0, 2, n_target)
    h_upsampled = np.interp(x_new, x_old, h0)
    return h_upsampled

def project_to_simplex(v, target_sum):
    """
    Project vector v onto simplex {x | sum x = target_sum, 0 <= x <= 1}
    using binary search on the Lagrange multiplier lambda.
    """
    v = np.array(v).copy()
    # Binary search for lambda such that sum(clamp(v - lambda)) == target_sum
    lbd_lo, lbd_hi = np.min(v - 1), np.max(v)
    tol = 1e-10
    for _ in range(100):
        lbd = (lbd_lo + lbd_hi) / 2
        x = np.clip(v - lbd, 0, 1)
        s = np.sum(x)
        if s > target_sum:
            lbd_lo = lbd
        else:
            lbd_hi = lbd
        if abs(s - target_sum) < tol:
            break
    return np.clip(v - lbd, 0, 1)

def softmax_weights(gamma, fs):
    """Compute soft-max weights w_s = exp(gamma * f_s) / sum(exp(gamma * f_k))"""
    fs = np.array(fs)
    exp_gf = np.exp(gamma * fs)
    return exp_gf / np.sum(exp_gf)

def compute_gradient(h, dx):
    """
    Compute ∇J_γ(h) = Σ_s w_s ∇f_s(h)
    where ∇f_s(h) = dx * g^{(s)} for each shift s.
    """
    n = len(h)
    grad = np.zeros(n)
    # For all shifts - (n-1) to (n-1)
    all_fs = []
    for s in range(-(n-1), n):
        # Compute f_s(h) for this shift
        f_s = 0.0
        g = np.zeros(n)
        for i in range(n):
            if 0 <= i + s < n:
                g[i] += 1 - h[i+s]
            if 0 <= i - s < n and s != 0:
                g[i] -= h[i-s]
        all_fs.append(np.sum(h * g) * dx)
    return all_fs, g

def soft_gradient(h, gamma, dx):
    """
    Compute the soft gradient ∇J_γ(h) = Σ_s w_s ∇f_s(h)
    as a single vector.
    """
    n = len(h)
    grad = np.zeros(n)
    fs = []
    # Compute all f_s(h)
    for s in range(-(n-1), n):
        # careful with indexing: list view
        f_s = 0.0
        for i in range(n):
            if i+s >= 0 and i+s < n:
                f_s += h[i] * (1 - h[i+s])
        fs.append(f_s * dx)

    # Compute weights
    ws = softmax_weights(gamma, fs)
    
    # Accumulate gradient
    for s, w in enumerate(ws):
        shift = s - (n-1)  # because range(-n+1, n) => shift in [-n+1, n-1]
        g = np.zeros(n)
        for i in range(n):
            if 0 <= i + shift < n:
                g[i] += 1 - h[i+shift]
            if 0 <= i - shift < n and shift != 0:
                g[i] -= h[i-shift]
        grad += w * dx * g

    return grad
