import numpy as np
import time
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral via full convolution of h with (1-h)
    overlaps = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum of the full correlation vector
    # scaled by dx
    c5 = np.max(overlaps) * dx
    return c5

def project_to_constraints(h, dx):
    """Project h to satisfy sum(h) * dx = 1 and h in [0,1]"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def propose_move(h, step_size):
    """Propose a feasible move by adjusting two entries."""
    h = h.copy()
    N = len(h)
    
    # Sample two distinct indices
    i, j = np.random.choice(N, 2, replace=False)
    
    # Determine feasible delta range for the move
    delta_low = max(-h[i], h[j] - 1)
    delta_high = min(1 - h[i], h[j])
    
    if delta_low >= delta_high:
        return h
    
    # Sample and apply move
    delta = np.random.uniform(max(-step_size, delta_low), 
                              min(step_size, delta_high))
    h[i] += delta
    h[j] -= delta
    
    # Project back to constraints
    h = project_to_constraints(h, 2.0 / len(h))
    return h

def local_search(initial_h, time_budget, seed):
    """Improved local search using gradient-like moves"""
    np.random.seed(seed)
    start = time.time()
    
    best_h = initial_h.copy()
    best_c5 = c5_of(best_h)
    
    # Start with a large step size that decreases
    step_size = 0.2
    decay_factor = 0.9995
    
    iteration = 0
    while time.time() - start < time_budget:
        # Decrease step size over time
        if iteration > 0 and iteration % 100 == 0:
            step_size *= decay_factor
            
        # Try a move
        new_h = propose_move(best_h, step_size)
        new_c5 = c5_of(new_h)
        
        # Accept better solutions
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5
            
        iteration += 1
        
    return best_h, best_c5

def construct_initial_guess(n_points):
    """Construct better initial guess for optimization"""
    x = np.linspace(0, 2, n_points)
    h = np.zeros(n_points)
    
    # A better initial construction: The idea is to spread the mass
    # but slightly sharpened in the middle
    
    # Work with a Gaussian-like shape
    center = 1.0
    sigma = 0.3
    h = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    
    # Normalize and clip
    dx = 2.0 / n_points
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
        
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start = time.time()
    
    N = 140
    dx = 2.0 / N
    
    # Get the predefined initial construction

        if len(h_initial) != N:

            x_new = np.linspace(0, 2, N)
            h_initial = np.interp(x_new, x0, h_initial)
    else:
        h_initial = construct_initial_guess(N)
    
    # First phase: Local search from initial guess
    h_final, c5_result = local_search(h_initial, 0.6 * budget_s, seed)
    
    # Second phase: Try some different seeds
    for i in range(3):
        if time.time() - start > 0.9 * budget_s:
            break
            
        # Create a different seed by perturbing the initial solution
        h_perturbed = h_initial.copy()
        noise_level = 0.05
        noise = np.random.normal(0, noise_level, N)
        h_perturbed += noise
        h_perturbed = project_to_constraints(h_perturbed, dx)
        
        h_candidate, c5_candidate = local_search(h_perturbed, 0.1 * budget_s, seed + i + 1)
        
        if c5_candidate < c5_result:
            h_final = h_candidate
            c5_result = c5_candidate
    
    # Final polishing using gradient-free method
    """Run crude gradient-free optimization using multiple restarts"""
    if time.time() - start < 0.95 * budget_s:
        # Create a set of fragmented h functions that might give better results
        def perturb_step(h, factor=0.02, steps=20):
            h1 = h.copy()
            h2 = h.copy()
            
            # Perturb one sample
            i = np.random.randint(0, len(h))
            h1[i] += factor * np.random.uniform(-1, 1)
            h1[i] = np.clip(h1[i], 0, 1)
            
            # Normalize
            h1_sum = np.sum(h1)
            if h1_sum > 0:
                h1 = h1 * (1.0 / (h1_sum * dx))
                
            return h1
        
        def eval_solution(h):
            return c5_of(h)
        
        # Start with our current best
        candidates = [h_final]
        
        # Generate some diversified candidates with noise

            candidate = perturb_step(h_final, factor=0.05)
            candidates.append(candidate)
        
        # Evaluate candidates
        scores = [eval_solution(h) for h in candidates]
        best_idx = np.argmin(scores)
        h_final = candidates[best_idx]
        c5_result = scores[best_idx]
