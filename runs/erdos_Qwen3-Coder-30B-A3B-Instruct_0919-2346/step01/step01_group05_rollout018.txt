import numpy as np
import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value scaled by dx
    max_overlap = np.max(correlation) * dx
    return max_overlap

def generate_sturmian_binary(n):
    """Generate a balanced binary sequence using irrational rotation"""
    alpha = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
    h = np.zeros(n, dtype=int)
    for i in range(n):
        t = (i * alpha) % 1
        h[i] = 1 if t < 0.5 else 0
    
    # Balance the number of 1s to exactly n/2
    ones_count = np.sum(h)
    if ones_count > n / 2:
        # Find indices of 1s to convert to 0s
        ones_indices = np.where(h == 1)[0]
        for i in range(int(ones_count - n / 2)):
            h[ones_indices[i]] = 0
    elif ones_count < n / 2:
        # Find indices of 0s to convert to 1s
        zeros_indices = np.where(h == 0)[0]
        for i in range(int(n / 2 - ones_count)):
            h[zeros_indices[i]] = 1
    
    return h

def calculate_shift_contributions(h):
    """Calculate all shift contributions efficiently"""
    n = len(h)
    dx = 2.0 / n
    # Two arrays for efficiency
    contrib = np.zeros(2 * n - 1)
    for i in range(n):
        for s in range(n):
            if i + s < n:
                contrib[s] += h[i] * (1 - h[i + s])
    return contrib * dx

def delta_shift_change(h, i, j, worst_shift, shift_contributions):
    """Calculate the change in overlap for a potential swap"""
    n = len(h)
    # Precompute indices for all shifts where i or j contribute
    delta = np.zeros(2 * n - 1)
    if h[i] == 1 and h[j] == 0:
        # i will become 0, j will become 1
        for s in range(2 * n - 1):
            if (i + s < n) and (i >= 0):
                delta[s] -= (1 - h[i + s])
            if (j + s < n) and (j >= 0):
                delta[s] += (1 - h[j + s])
            if (i - s >= 0) and (i - s < n):
                delta[s] += h[i - s]
            if (j - s >= 0) and (j - s < n):
                delta[s] -= h[j - s]
    elif h[i] == 0 and h[j] == 1:
        # i will become 1, j will become 0
        for s in range(2 * n - 1):
            if (i + s < n) and (i >= 0):
                delta[s] += (1 - h[i + s])
            if (j + s < n) and (j >= 0):
                delta[s] -= (1 - h[j + s])
            if (i - s >= 0) and (i - s < n):
                delta[s] -= h[i - s]
            if (j - s >= 0) and (j - s < n):
                delta[s] += h[j - s]
    return delta

def local_swap_search(h, n_points, time_limit):
    """Perform targeted local swaps to reduce overlap"""
    start_time = time.time()
    dx = 2.0 / n_points
    h = h.astype(float)
    max_err = 1.0
    best_h = h.copy()
    best_err = evaluate_erdos_solution(h)
    
    shift_contributions = np.correlate(h, 1 - h, mode="full") * dx
    tabu_list = set()
    tabu_time = 0
    
    moves_since_improvement = 0
    max_moves_since_improvement = 5000
    
    while True:
        if time.time() - start_time > time_limit:
            break
            
        # Find worst shift and current max overlap
        max_shift_idx = np.argmax(shift_contributions)
        current_max = shift_contributions[max_shift_idx]
        
        if current_max < max_err:
            max_err = current_max
            best_h = h.copy()
            
        # Identify good swap candidates
        ones_indices = np.where(h == 1)[0]
        zeros_indices = np.where(h == 0)[0]
        
        if len(ones_indices) == 0 or len(zeros_indices) == 0:
            break

        # Pick indices for swap
        i = np.random.choice(ones_indices)
        j = np.random.choice(zeros_indices)
        
        # Check tabu list
        if (i, j) in tabu_list and tabu_time > 0:
            continue

        delta = delta_shift_change(h, i, j, max_shift_idx, shift_contributions)
        
        new_contributions = shift_contributions + delta
        new_max = np.max(new_contributions)
        
        if new_max < current_max or np.random.rand() < 0.05:
            if new_max < current_max:
                moves_since_improvement = 0
            else:
                moves_since_improvement += 1
            
            # Update h and shift_contributions
            h[i], h[j] = h[j], h[i]
            shift_contributions = new_contributions
            tabu_time = 50
            tabu_list.add((i, j))
            if len(tabu_list) > 200:
                tabu_list.pop()
        else:
            moves_since_improvement += 1
            
        if moves_since_improvement > max_moves_since_improvement:
            # Perturb
            indices = np.random.choice(n_points, size=n_points//20, replace=False)
            for idx in indices:
                h[idx] = 1 - h[idx]
            shift_contributions = np.correlate(h, 1 - h, mode="full") * dx
            moves_since_improvement = 0

        if tabu_time > 0:
            tabu_time -= 1
            
    return best_h, max_err

def upscale_binary(h, new_n):
    """Upscale binary sequence to a finer grid"""
    old_n = len(h)
    new_h = np.zeros(new_n)
    
    # Duplicate each element to create a finer grid
    for i in range(old_n):
        new_h[2*i] = h[i]
        new_h[2*i + 1] = h[i]
    
    # Trim if necessary
    if len(new_h) > new_n:
        new_h = new_h[:new_n]
    
    return new_h

def continuous_project_and_optimize(h, time_limit):
    """Perform a brief continuous optimization to improve solution"""
    start_time = time.time()
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Set up optimization
    hnp = h.copy()
    
    def obj(hvec):
        return evaluate_erdos_solution(hvec)
    
    def cons(hvec):
        return np.sum(hvec) - n_points / 2.0
    
    # Project h onto feasible region
    hnp = np.clip(hnp, 0, 1)
    current_sum = np.sum(hnp) * dx
    hnp = hnp * (1.0 / current_sum)
    
    bounds = [(0, 1) for _ in range(n_points)]
    constr = {'type': 'eq', 'fun': cons}
    
    try:
        result = minimize(obj, hnp, method='SLSQP', bounds=bounds, 
                            constraints=[constr], options={'maxiter': 1000}, 
                            tol=1e-6)
        if result.success:
            return result.x, obj(result.x)
    except:
        pass
    
    return h, obj(h)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Starting parameters
    max_iterations = 20000
    time_per_iteration = 0.05
    
    # Coarse resolution stage
    n_points = 120
    start_time = time.time()
    
    # Generate initial binary pattern
    h = generate_sturmian_binary(n_points)
    
    # Local search on coarse grid
    h_orig = h.copy()
    h_coarse, err_coarse = local_swap_search(h, n_points, 0.4 * budget_s)
    
    # Refinement with upsampling
    n_points_refined = 240
    h_upscaled = upscale_binary(h_coarse, n_points_refined)
    h_refined, err_refined = local_swap_search(h_upscaled, n_points_refined, 0.45 * budget_s)
    
    if (time.time() - start_time) < 0.95 * budget_s:
        # Further continuous polishing
        h_polished, err_polished = continuous_project_and_optimize(h_refined, 0.1 * budget_s)
        final_h = h_polished
        final_err = err_polished
    else:
        final_h = h_refined
        final_err = err_refined
    
    return (final_h.tolist(), final_err, len(final_h))
