def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral via full convolution of h with (1-h)
    overlaps = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum of the full correlation vector
    # scaled by dx
    c5 = np.max(overlaps) * dx
    return c5

def compute_constraint_violation(h, n_points):
    """Compute constraint violation for sum(h) = n_points / 2"""
    dx = 2.0 / n_points
    return abs(np.sum(h) * dx - 1.0)

def propose_move(h, max_step, n_points):
    """Propose a feasible move by adjusting two entries."""
    h = h.copy()
    # Try to find a valid pair of indices
    for _ in range(100):  # avoid infinite loop
        i, j = np.random.randint(0, n_points, 2)
        if i == j:
            continue
        # Determine feasible δ range
        delta_low = max(-h[i], h[j] - 1)
        delta_high = min(1 - h[i], h[j])
        if delta_low >= delta_high:
            continue
        # Sample a move with limited step size
        step = min(max_step, delta_high - delta_low)
        delta = np.random.uniform(
            max(-step, delta_low),
            min(step, delta_high)
        )
        # Apply move
        h[i] += delta
        h[j] -= delta
        return h
    # If no valid move found, return copy
    return h

    best_c5 = c_cur

    while time.time() - start < time_budget:
        # Try ten different approaches based on optimization theory
        # Approach 1: Randomized hill climbing
        h_new = propose_move(cur, 0.05, n_points)
        if compute_constraint_violation(h_new, n_points) < 0.001:
            c_new = c5_of(h_new)
            if c_new < c_cur:
                cur = h_new
                c_cur = c_new

                    if c_trial < best_c5:
                        best_h = cur.copy()
                        best_c5 = c_trial                        
                
        if best_c5 < c_cur * 0.999999:
            break
            
    return best_h, best_c5
