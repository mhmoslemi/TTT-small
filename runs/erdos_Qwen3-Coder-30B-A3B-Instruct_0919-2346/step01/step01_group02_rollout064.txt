import numpy as np
import time
from scipy.optimize import min_LBFGSB
from scipy.optimize import OptimizeResult

def project_to_sum_constraint(h, target_sum):
    """Project h onto the set {x : sum(x) = target_sum, 0 <= x <= 1}."""
    # Use water-filling algorithm
    v = h.copy()
    v_sorted = np.sort(v)[::-1]
    n = len(v_sorted)
    
    # Find lambda such that sum(clamp(v + lambda, 0, 1)) == target_sum
    def F(lam):
        return np.sum(np.clip(v_sorted + lam, 0, 1))
    
    # Binary search for lambda
    lam_low, lam_high = -2.0, 2.0
    for _ in range(50):
        lam = (lam_low + lam_high) / 2.0
        f_val = F(lam)
        if f_val < target_sum:
            lam_low = lam
        else:
            lam_high = lam
    
    # Refine toward target sum by adjusting free entries
    lam = (lam_low + lam_high) / 2.0
    h_proj = np.clip(v + lam, 0, 1)
    
    # Ensure exact sum
    current_sum = np.sum(h_proj)
    diff = target_sum - current_sum
    
    # Distribute the diff to entries that are not at the bounds
    free = np.where((h_proj > 0) & (h_proj < 1))[0]
    if len(free) > 0:
        inc = diff / len(free)
        if inc > 0:
            for i in free:
                if h_proj[i] + inc <= 1:
                    h_proj[i] += inc
                else:
                    h_proj[i] = 1.0
                    diff -= (1.0 - (h_proj[i] - inc))
                    # Redistribute remaining diff
                    if diff > 0:
                        if np.sum(h_proj) < target_sum:
                            remaining = target_sum - np.sum(h_proj)
                            inc = remaining / len(free)
                            for j in free:
                                if h_proj[j] + inc <= 1:
                                    h_proj[j] += inc
                                else:
                                    h_proj[j] = 1.0
        else:
            for i in free:
                if h_proj[i] + inc >= 0:
                    h_proj[i] += inc
                else:
                    h_proj[i] = 0.0
    
    return h_proj

def compute_correlations(h):
    """Compute full correlations between h and (1-h) using np.correlate"""
    c = np.correlate(h, 1 - h, mode='full')
    return c

def compute_smooth_surrogate(h, eps):
    """Compute smooth surrogate for max over shifts."""
    N = len(h)
    dx = 2.0 / N
    c = compute_correlations(h)
    c_scaled = c * dx
    
    # Numerical stability trick: subtract maximum
    max_c = np.max(c_scaled)
    c_shifted = c_scaled - max_c
    
    # Soft-min trick (exponentiated and normalized)
    exp_c = np.exp(c_shifted / eps)
    sum_exp = np.sum(exp_c)
    weights = exp_c / sum_exp
    
    # Compute gradient via weighted sum
    grad = np.zeros(N)
    for i in range(N):
        # Accumulate contributions from all shifts
        for s in range(2*N - 1):
            contrib = weights[s] * dx
            # We compute ∂C_s/∂h_k = 1 - h[k+s] - h[k-s]
            if s < N:
                h1 = h[i]
                h2 = 0.0 if i+s >= N else h[i+s]
                h3 = 0.0 if i-s < 0 else h[i-s]
                val = contrib * (1 - h2 - h3)
                grad[i] += val
            else:
                s_shift = s - (N - 1)
                h1 = h[i]
                h2 = 0.0 if i+s_shift >= N else h[i+s_shift]
                h3 = 0.0 if i-s_shift < 0 else h[i-s_shift]
                val = contrib * (1 - h2 - h3)
                grad[i] += val
    return c_scaled, grad

def c5_of(h):
    """Compute the maximum overlap integral for the given step function h."""
    dx = 2.0 / len(h)
    c = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(c) * dx
    return c5

def run_continuation(h0, eps_list, max_iters=150):
    """Run optimization using smooth surrogate with continuation."""
    N = len(h0)
    dx = 2.0 / N
    h = h0.copy()
    
    for eps in eps_list:
        # Inner L-BFGS loop
        def fun(x):
            c_scaled, grad = compute_smooth_surrogate(x, eps)
            c5 = np.max(c_scaled)
            return c5, grad
        
        # Binary search for projection
        h_proj = project_to_sum_constraint(h, N/2.0)
        bounds = [(0, 1) for _ in range(N)]
        
        result = min_LBFGSB(fun, h_proj, bounds=bounds, maxiter=max_iters)
        h = result.x
        
        # Post-projection to ensure exact sum
        h = project_to_sum_constraint(h, N/2.0)
    
    # Final projection
    h = project_to_sum_constraint(h, N/2.0)
    
    return h

            for i in range(N):
