import numpy as np
import time
import random
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import cdist


def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    overlap = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(overlap) * dx
    return c5


def project_to_feasible(h):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h)*dx = 1"""
    # Clamp values to [0,1]
    h = np.clip(h, 0, 1)
    # Renormalize to sum = 1/dx = n_points/2
    dx = 2.0 / len(h)
    target_sum = 1 / dx
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h


def propose_move(h, step_size):
    """Propose a new feasible h vector by making a small adjustment"""
    h_new = h.copy()
    N = len(h)
    
    # Choose two random indices
    i, j = random.sample(range(N), 2)
    
    # Determine feasible interval for delta
    delta_low = max(-h_new[i], h_new[j] - 1)
    delta_high = min(1 - h_new[i], h_new[j])
    
    # If infeasible, return original
    if delta_low >= delta_high:
        return h.copy()
        
    # Sample delta uniformly
    delta = random.uniform(delta_low, delta_high)
    
    # Apply change
    h_new[i] += delta
    h_new[j] -= delta
    
    return h_new


def local_search(h_init, budget_s):
    """Run a local search with simulated annealing."""
    start = time.time()
    
    # Parameters
    T_start = 0.1
    T_end = 1e-8
    step_size = 0.1
    
    cur_h = h_init.copy()
    c_cur = c5_of(cur_h)
    best_h = cur_h.copy()
    c_best = c_cur
    
    # Main loop
    iteration = 0
    while time.time() - start < budget_s:
        t = time.time() - start
        elapsed_frac = t / budget_s
        
        # Adaptive cooling
        T = T_start * (T_end / T_start) ** elapsed_frac
        step = step_size * (1 - elapsed_frac)
        if step < 1e-6:
            step = 1e-6
            
        # Propose move
        h_proposed = propose_move(cur_h, step)
        h_proposed = project_to_feasible(h_proposed)
        c_proposed = c5_of(h_proposed)
        
        # Accept or reject
        if c_proposed <= c_cur:
            cur_h = h_proposed
            c_cur = c_proposed
            if c_proposed < c_best:
                best_h = h_proposed.copy()
                c_best = c_proposed
        else:
            if random.random() < np.exp(-(c_proposed - c_cur) / T):
                cur_h = h_proposed
                c_cur = c_proposed
                
        iteration += 1
        
    return best_h, c_best

    v = 0.2

    right = int((2 - a) * n_points / 2)
    h_bimodal[:left] = u
    h_bimodal[right:] = u
    h_bimodal = project_to_feasible(h_bimodal)
    
    # Multi-start approach
    best_h = None
    best_c5 = float('inf')
    
    # Try different starting points
    start_solutions = [h_init, h_bimodal]
    time_per_solution = (budget_s * 0.8) / len(start_solutions)
    
    for h_seed in start_solutions:
        if time.time() - start >= budget_s:
            break

            best_c5 = c5_opt
            best_h = h_opt.copy()

    if best_h is not None and time.time() - start < budget_s - 10:
        remaining_time = budget_s - (time.time() - start)
        
        # Use a simple gradient-descent-like approach

        c_current = c5_of(h_current)
