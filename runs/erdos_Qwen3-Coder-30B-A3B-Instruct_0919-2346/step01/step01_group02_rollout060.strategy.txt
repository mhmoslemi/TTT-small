def project_onto_bounds_and_sum(v, target):
    # v: vector of shape N.
    # We'll shift all entries by lambda then clip.
    # function f(lam) = sum(clip(v + lam, 0, 1)) - target
    # f(lam) monotonic increasing in lam.
    # Find lam via binary search.
    lower = -np.max(v) - 1  # ensure all entries zero after shift and clip.
    upper = 1 - np.min(v) + 1  # ensure all entries one after shift and clip.
    # But we can set lower = -np.max(v) (v+lam <=0 -> all zero)
    # and upper = 1 - np.min(v) (v+lam >=1 -> all one)
    lower = -np.max(v)
    upper = 1 - np.min(v)
    # Expand range a bit to guarantee bracket.
    for _ in range(50):
        mid = (lower + upper) / 2.0
        w = np.clip(v + mid, 0, 1)
        s = np.sum(w)
        if s < target:
            lower = mid
        else:
            upper = mid
    w = np.clip(v + (lower + upper) / 2.0, 0, 1)
    # Might need slight correction for rounding errors: adjust by distributing diff.
    # Could enforce sum by scaling residual if needed, but we hope close enough.
    # If still not exactly target (due to numeric), adjust via adding delta to entries not at bound.
    diff = target - np.sum(w)
    if abs(diff) > 1e-12:
        # Distribute diff uniformly to non-bound entries
        free_idx = (w > 0) & (w < 1)
        if np.any(free_idx):
            w[free_idx] += diff / np.sum(free_idx)
            w = np.clip(w, 0, 1)  # reclip to be safe
        else:
            # all at bounds (should not happen often); distribute to all and reclip
            w += diff / len(w)
            w = np.clip(w, 0, 1)
    return w
