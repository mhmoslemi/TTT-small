import numpy as np
import time
import random
from scipy.optimize import differential_evolution, minimize


def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    overlap = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(overlap) * dx
    return c5


def propose_move(h, max_step):
    """Propose a new feasible h vector by adjusting two elements"""
    h_new = h.copy()
    N = len(h)
    
    # Try to find a valid move
    max_tries = 100
    for _ in range(max_tries):
        i = random.randint(0, N - 1)
        j = random.randint(0, N - 1)
        if i == j:
            continue
            
        # Determine feasible interval for δ
        delta_low = max(-h_new[i], h_new[j] - 1)
        delta_high = min(1 - h_new[i], h_new[j])
        
        if delta_low >= delta_high:
            continue
            
        # Sample δ from uniform distribution over feasible range
        step = min(max_step, delta_high - delta_low)
        if step <= 0:
            continue
        delta = random.uniform(delta_low, delta_high)
        
        # Apply move
        h_new[i] += delta
        h_new[j] -= delta
        
        return h_new
    
    # If no valid move found, return original
    return h.copy()


def refine_with_optimization(h_initial, time_budget_s):
    """Refine solution using scipy optimization"""
    start = time.time()
    N = len(h_initial)
    dx = 2.0 / N
    
    # Create initial vector with proper sum
    h_current = h_initial.copy()
    h_current = h_current * (N / 2) / np.sum(h_current)
    
    # Define constraint function
    def constraint_func(x):
        return np.sum(x) - N / 2
    
    # Define bounds: [0, 1] for each element
    bounds = [(0, 1) for _ in range(N)]
    
    # Optimization using differential evolution with known constraints
    def objective(x):
        return c5_of(x)
    
    # Use a hybrid approach
    max_evaluations = 1000
    counter = 0
    
    # First, try with a few runs of differential evolution
    for i in range(3):
        if time.time() - start > time_budget_s - 5:
            break
        try:
            # Try using a different optimizer
            result = differential_evolution(
                objective, 
                bounds, 
                maxiter=200, 
                popsize=10,
                seed=random.randint(0, 10000),
                disp=False
            )
            if result.success:
                h_temp = result.x.copy()
                # Renormalize to sum = N/2
                h_temp = h_temp * (N / 2) / np.sum(h_temp)
                c_temp = c5_of(h_temp)
                if c_temp < c5_of(h_current):
                    h_current = h_temp.copy()
        except:
            pass
    
    # Local search refinement
    # Sample several local neighbors and try to improve
    for _ in range(5000):
        if time.time() - start > time_budget_s - 5:
            break
        h_proposed = propose_move(h_current, 0.05)
        c_proposed = c5_of(h_proposed)
        if c_proposed < c5_of(h_current):
            h_current = h_proposed
        counter += 1
    
    return h_current


def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function"""
    random.seed(seed)
    np.random.seed(seed)
    
    start = time.time()
    N = 140
    dx = 2.0 / N
    
    # Build initial seeds
    # 1. Uniform seed
    h_uniform = np.ones(N) * 0.5
    
    # 2. Provided initial_h_values (interpolated if needed)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_provided = np.interp(
            np.linspace(0, 2, N),
            np.linspace(0, 2, len(initial_h_values)),
            initial_h_values
        )
        # Renormalize to exact sum
        h_provided = h_provided * (N / 2) / np.sum(h_provided)
    else:
        h_provided = np.ones(N) * 0.5
    
    # 3. Double-peak seed
    h_double = np.full(N, 0.1)
    a = 0.3
    u = 0.9
    v = 0.1
    left = int(a * N / 2)
    right = int((2 - a) * N / 2)
    h_double[:left] = u
    h_double[right:] = u
    # Normalization
    h_double = h_double * (N / 2) / np.sum(h_double)
    
    # 4. Triangular shape initial
    h_triangular = np.array([min(i, N-i) for i in range(1, N+1)])
    h_triangular = (h_triangular / np.sum(h_triangular)) * (N / 2)
    
    # 5. Sine wave
    x = np.linspace(0.5, 1.5, N)
    h_sine = 0.5 + 0.5 * np.sin(2.0 * np.pi * x)
    h_sine = h_sine * (N / 2) / np.sum(h_sine)
    
    # Test all initial seeds
    seeds = [h_uniform, h_provided, h_double, h_triangular, h_sine]
    
    best_h = None
    best_c5 = float('inf')
    
    # Run refinement on the best performing initial seeds
    for i, h_seed in enumerate(seeds):
        if time.time() - start >= budget_s:
            break
            
        # Refine with local search + optimization
        h_refined = refine_with_optimization(h_seed, budget_s - (time.time() - start))
        c5_refined = c5_of(h_refined)
        
        if c5_refined < best_c5:
            best_c5 = c5_refined
            best_h = h_refined.copy()
    
    # Use simulated annealing on the best construction
    # We will now do a longer SA run
    if best_h is None:
        best_h = h_provided
        best_c5 = c5_of(best_h)
    
    if time.time() - start < budget_s - 20:
        # Perturb initial best and do one final long annealing run
        h_anneal = best_h.copy()
        c_anneal = best_c5
        
        max_tries = 100000
        for i in range(max_tries):
            if time.time() - start > budget_s - 5:
                break
            # Use a large step size
            h_proposed = propose_move(h_anneal, 0.1)
            c_proposed = c5_of(h_proposed)
            
            if c_proposed <= c_anneal:
                h_anneal = h_proposed
                c_anneal = c_proposed
                if c_proposed < best_c5:
                    best_c5 = c_proposed
                    best_h = h_proposed.copy()
    
    return (best_h, best_c5, N)
