import numpy as np
import time
import random
from scipy import signal

def compute_c5(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    overlap = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(overlap) * dx
    return c5

def project_h(h, dx=2.0/140):
    """Project h to satisfy integral constraint and bound constraints"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h) * dx
    target_sum = 1.0
    if abs(current_sum - target_sum) > 1e-10:
        # Linear adjustment to scale to target integral
        scale_factor = target_sum / current_sum
        h = h * scale_factor
        h = np.clip(h, 0, 1)
    return h

def compute_gradient(h, k):
    """Compute gradient with respect to h for a specific k shift"""
    dx = 2.0 / len(h)
    # Gradient of integral with respect to h_i is (1 - h_{i+k}) 
    # We'll compute it by convolution of h_{i+k} with ones
    gradients = np.zeros_like(h)
    if k >= 0:
        gradients[k:] = 1 - h[:-k]  # when shift is positive
    else:
        gradients[:k] = 1 - h[-k:]  # when shift is negative
    return gradients * dx

        # Compute specific gradient based on the maximum shift
        grad_i = compute_gradient(h, worst_shift)

        h_new = project_h(h_new)
        
        h = h_new
        if time.time() - start > time_budget_s:
            break
    
    return h, compute_c5(h)

def propose_move(h, max_step):
    """Propose a new feasible h vector by adjusting two elements"""
    h_new = h.copy()
    N = len(h)
    
    # Try to find a valid move
    max_tries = 100
    for _ in range(max_tries):
        i = random.randint(0, N - 1)
        j = random.randint(0, N - 1)
        if i == j:
            continue
            
        # Determine feasible interval for δ
        delta_low = max(-h_new[i], h_new[j] - 1)
        delta_high = min(1 - h_new[i], h_new[j])
        
        if delta_low >= delta_high:
            continue
            
        # Sample δ from uniform distribution over feasible range
        step = min(max_step, delta_high - delta_low)
        if step <= 0:
            continue
        delta = random.uniform(delta_low, delta_high)
        
        # Apply move
        h_new[i] += delta
        h_new[j] -= delta
        
        return h_new
    
    # If no valid move found, return original
    return h.copy()

def local_refinement(h, time_budget_s):
    """Refine using directional pairwise moves"""
    start = time.time()
    h = h.copy()
    c5 = compute_c5(h)
    
    # Heuristic loop
    while time.time() - start < time_budget_s:
        # Propose a few moves and choose best
        best_h = h.copy()
        best_c5 = c5
        
        for move_iter in range(50):
            h_proposed = propose_move(h, 0.1)
            c_proposed = compute_c5(h_proposed)
            if c_proposed < best_c5:
                best_c5 = c_proposed
                best_h = h_proposed.copy()
        
        if best_c5 < c5:
            h = best_h
            c5 = best_c5
        else:
            break  # No improvement
    
    return h, c5
