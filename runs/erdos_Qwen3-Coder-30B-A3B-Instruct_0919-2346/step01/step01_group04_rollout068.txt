import numpy as np
import time
import random
from scipy.optimize import minimize


def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    overlap = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(overlap) * dx
    return c5


def propose_move(h, max_step):
    """Propose a new feasible h vector by adjusting two elements"""
    h_new = h.copy()
    N = len(h)
    
    # Try to find a valid move
    max_tries = 100
    for _ in range(max_tries):
        i = random.randint(0, N - 1)
        j = random.randint(0, N - 1)
        if i == j:
            continue
            
        # Determine feasible interval for δ
        delta_low = max(-h_new[i], h_new[j] - 1)
        delta_high = min(1 - h_new[i], h_new[j])
        
        if delta_low >= delta_high:
            continue
            
        # Sample δ from uniform distribution over feasible range
        step = min(max_step, delta_high - delta_low)
        if step <= 0:
            continue
        delta = random.uniform(delta_low, delta_high)
        
        # Apply move
        h_new[i] += delta
        h_new[j] -= delta
        
        return h_new
    
    # If no valid move found, return original
    return h.copy()

        h = h_vals.copy()
        # Ensure normalized
        h = h * (N / 2) / np.sum(h)
        return c5_of(h)

        h = h_vals.copy()
        h = h * (N / 2) / np.sum(h)
        return np.sum(h) - N / 2
    
    # Flatten h_vals for scipy
    h0 = initial_h.copy()
    h0 = h0 * (N / 2) / np.sum(h0)
    
    # Constraint for sum
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Bounds
    bounds = [(0, 1) for _ in range(N)]
    
    # Run optimization
    result = minimize(
        objective,

        method='Nelder-Mead',
        options={'maxiter': max_iter, 'disp': False}
    )
    
    if result.success:
        best_h = result.x
        best_h = best_h * (N / 2) / np.sum(best_h)
        return best_h
    else:
        return initial_h
