import numpy as np
import time

from scipy import signal

def compute_C_vector(h, dx):
    """Compute the vector of overlaps C_τ for all shifts τ."""
    # Compute correlation h * (1 - h) in full mode
    c = np.correlate(h, 1 - h, mode="full")
    c *= dx  # Scale by dx
    # Convert to C_τ = 1 - dx * S_τ = 1 - c_τ
    c = 1 - c
    return c

def compute_softmax_weights(c, alpha):
    """Compute softmax weights w_τ for the given overlaps c and smoothing α."""
    c_max = np.max(c)
    exp_ca = np.exp(alpha * (c - c_max))
    w = exp_ca / np.sum(exp_ca)
    return w

def compute_gradient(h, w, dx):
    """Compute the gradient ∇f_α(h) for the softmax objective."""
    # Compute (w * h) and (w_rev * h) with proper handling of zero padding
    w_rev = w[::-1]
    conv1 = signal.convolve(w, h, mode="same")
    conv2 = signal.convolve(w_rev, h, mode="same")
    # g = -dx * (conv1 + conv2)
    g = -dx * (conv1 + conv2)
    return g

def project_onto_box_and_mass(h_proj, n, mass):
    """Project onto [0,1]^n and enforce sum = mass."""
    # Clip to [0,1] first
    h_proj = np.clip(h_proj, 0, 1)
    # Then water-filling to enforce the sum constraint
    if np.abs(np.sum(h_proj) - mass) < 1e-12:
        return h_proj
    # Use water-filling algorithm
    h_sorted = np.sort(h_proj)[::-1]
    tmp = h_sorted.copy()
    cumsum = np.cumsum(tmp)
    n_exceed = np.argmax(cumsum > mass) if np.any(cumsum > mass) else len(tmp)
    if n_exceed == 0:
        # All values are too small; make them equal to mass/n
        return np.full_like(h_proj, mass/n)
    # For first n_exceed elements, determine threshold
    # sum(h_sorted[:n_exceed]) - n_exceed*threshold = mass
    threshold = (cumsum[n_exceed - 1] - mass) / n_exceed
    # Apply thresholding
    h_new = np.clip(h_proj - threshold, 0, 1)
    # Normalize
    h_new = h_new * mass / np.sum(h_new)
    return h_new
