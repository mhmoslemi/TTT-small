import numpy as np
import time
from scipy.optimize import differential_evolution, minimize

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral via full convolution of h with (1-h)
    overlaps = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum of the full correlation vector
    # scaled by dx
    c5 = np.max(overlaps) * dx
    return c5

def feasible_step(h, i, j, delta):
    """Check if moving delta from i to j keeps values in [0,1]"""
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    return np.all((h_new >= 0) & (h_new <= 1))

def propose_move(h, step_size):
    """Propose a feasible move by adjusting two entries."""
    h = h.copy()
    N = len(h)
    
    # Try to find a valid pair of indices
    for _ in range(100):  # avoid infinite loop
        i, j = np.random.randint(0, N, 2)
        if i == j:
            continue
        # Determine feasible δ range
        delta_low = max(-h[i], h[j] - 1)
        delta_high = min(1 - h[i], h[j])
        if delta_low >= delta_high:
            continue
        # Sample a move with limited step size
        delta = np.random.uniform(
            max(-step_size, delta_low),
            min(step_size, delta_high)
        )
        # Apply move
        h[i] += delta
        h[j] -= delta
        return h
    # If no valid move found, return copy
    return h

def constrained_objective(h):
    """Objective function for optimization with constraints"""
    c5 = c5_of(h)
    # Add penalty for constraint violation
    integral = np.sum(h) * (2.0 / len(h))
    penalty = 1e6 * abs(integral - 1.0)
    return c5 + penalty

def optimize_with_differential_evolution(initial_h, bounds, budget_s):
    """Use differential evolution for global optimization"""
    start = time.time()
    
    def wrapper(x):
        # Reshape x to proper dimensions
        h = np.array(x)
        return constrained_objective(h)
    
    # Run differential evolution with fixed timeout
    result = differential_evolution(
        wrapper,
        bounds,
        maxiter=max(100, int(1000 * (budget_s / 600))),
        popsize=15,
        seed=42,
        mutation=(0.5, 1),
        recombination=0.7,
        atol=1e-6,
        disp=False
    )
    
    # Refine with local optimization
    h = result.x.copy()
    h = h / (np.sum(h) * 2.0 / len(h)) * (len(h) / 2)
    return h, c5_of(h)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start = time.time()
    N = 140
    dx = 2.0 / N

    # Get initial construction
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        if len(h0) != N:
            x0 = np.linspace(0, 2, len(h0))
            x_new = np.linspace(0, 2, N)
            h0 = np.interp(x_new, x0, h0)
        h_initial = h0 * (N / 2) / np.sum(h0)
    else:
        h_initial = np.random.rand(N) * 0.5
        h_initial *= (N / 2) / np.sum(h_initial)
    
    # Create bounds for optimization
    bounds = [(0, 1) for _ in range(N)]
    
    # Run differential evolution optimization
    best_h, best_c5 = optimize_with_differential_evolution(h_initial, bounds, budget_s)
    
    # Local search refinement
    if time.time() - start < budget_s * 0.95:
        # Use gradient-free local refinement
        local_h = best_h.copy()
        local_c5 = best_c5
        step_size = 0.01
        
        # Run simple local search
        for _ in range(5000):
            if time.time() - start > budget_s * 0.98:
                break
            h_new = propose_move(local_h, step_size)
            c_new = c5_of(h_new)
            if c_new < local_c5:
                local_h = h_new
                local_c5 = c_new
                
        if local_c5 < best_c5:
            best_h = local_h
            best_c5 = local_c5
    
    return (best_h, best_c5, N)
