import numpy as np
from scipy.optimize import minimize
from math import pi
import time

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value scaled by dx
    max_overlap = np.max(correlation) * dx
    return max_overlap

def build_h_from_coeffs(a, n):
    """Build h from Fourier coefficients"""
    h = 0.5 * np.ones(n)
    M = len(a)
    for m in range(1, M + 1):
        cos_term = np.cos(2 * pi * m * np.arange(n) / n)
        h += a[m - 1] * cos_term
    return h

def compute_overlap_and_gradient(a, n, M, dx, gamma, penalty_strength):
    """Compute overlap and gradient for current coefficients"""
    h = build_h_from_coeffs(a, n)
    # Apply box constraints via penalty
    h = np.clip(h, 0, 1)
    
    # Compute correlation for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    shifts = np.arange(-n + 1, n)
    overlaps = corr * dx
    
    # Softmax approximation
    max_overlap = np.max(overlaps)
    exp_overlaps = np.exp(gamma * (overlaps - max_overlap))
    weights = exp_overlaps / np.sum(exp_overlaps)
    
    # Objective value
    obj = np.log(np.sum(exp_overlaps)) / gamma + max_overlap + penalty_strength * (
        np.sum((np.maximum(h - 1, 0))**2) +
        np.sum((np.maximum(-h, 0))**2)
    )
    
    # Gradient computation
    grad = np.zeros(M)
    for m in range(1, M + 1):
        cos_term = np.cos(2 * pi * m * np.arange(n) / n)
        # Gradient of overlaps w.r.t. a_m
        grad_f = np.zeros(len(shifts))
        for s_idx, s in enumerate(shifts):
            if s >= 0:
                valid_indices = np.arange(0, n - s)
                grad_f[s_idx] = np.sum(cos_term[valid_indices] * (1 - h[valid_indices + s])) \
                              - np.sum(h[valid_indices] * cos_term[valid_indices + s])
            else:
                valid_indices = np.arange(-s, n)
                grad_f[s_idx] = np.sum(cos_term[valid_indices] * (1 - h[valid_indices + s])) \
                              - np.sum(h[valid_indices] * cos_term[valid_indices + s])
        
        grad[m-1] = np.sum(weights * grad_f) * dx + penalty_strength * (
            np.sum((h - np.clip(h, 0, 1)) * cos_term)
        )
    
    return obj, grad

def project_to_simplex_box(h, target_sum, n):
    """Project h onto the feasible set"""
    # Use water filling
    h_clipped = np.clip(h, 0, 1)
    current_sum = np.sum(h_clipped)
    
    if abs(current_sum - target_sum) < 1e-8:
        return h_clipped
    
    # Estimate threshold using binary search
    low, high = -1.0, 1.0
    while high - low > 1e-10:
        tau = (low + high) / 2
        h_new = np.clip(h_clipped - tau, 0, 1)
        new_sum = np.sum(h_new)
        if new_sum > target_sum:
            low = tau
        else:
            high = tau
    
    tau = (low + high) / 2
    h_result = np.clip(h_clipped - tau, 0, 1)
    return h_result

def spectral_optim(n, M_max, gamma_initial, gamma_final, penalty_strength, 
                   max_iter_stage1, max_iter_stage2, max_iter_stage3, 
                   gamma_increase_interval, time_budget_s):
    """Optimise using spectral representation"""
    t_start = time.time()
    best_obj = float('inf')
    best_a = None
    best_h = None
    
    # Multiple restarts
    num_restarts = 6
    for restart in range(num_restarts):
        if time.time() - t_start > time_budget_s * 0.95:
            break
        np.random.seed(restart)
        
        # Initial coefficients
        a = np.random.normal(0, 0.05, M_max)
        
        # Stage 1: M = 8
        M_stage1 = min(8, M_max)
        a_init = a[:M_stage1]
        bounds = [(-0.6, 0.6) for _ in range(M_stage1)]
        
        # Optimise in stages
        a_opt = a_init.copy()
        gamma = gamma_initial
        iter_count = 0
        iter_total = 0
        
        # Stage 1
        while iter_total < max_iter_stage1 and time.time() - t_start < time_budget_s * 0.9:
            obj, grad = compute_overlap_and_gradient(a_opt, n, M_stage1, 2.0/n, gamma, penalty_strength)
            if obj < best_obj:
                best_obj = obj
                best_a = a_opt.copy()
                
            res = minimize(lambda x: compute_overlap_and_gradient(x, n, M_stage1, 2.0/n, gamma, penalty_strength)[0],
                          a_opt, method='L-BFGS-B', jac=lambda x: compute_overlap_and_gradient(x, n, M_stage1, 2.0/n, gamma, penalty_strength)[1],
                          bounds=bounds)
            
            if res.success:
                a_opt = res.x
                iter_total += 1
                iter_count += 1
                
                if iter_count % gamma_increase_interval == 0:
                    gamma = min(gamma * 1.5, gamma_final)
                    
        # Extend to M_stage2
        if M_stage1 < M_max:
            a_opt_new = np.zeros(M_max)
            a_opt_new[:M_stage1] = a_opt
            M_stage2 = min(15, M_max)
            
            a_opt = a_opt_new[:M_stage2]
            bounds = [(-0.6, 0.6) for _ in range(M_stage2)]
            
            while iter_total < max_iter_stage1 + max_iter_stage2 and time.time() - t_start < time_budget_s * 0.9:
                obj, grad = compute_overlap_and_gradient(a_opt, n, M_stage2, 2.0/n, gamma, penalty_strength)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a_opt.copy()
                    
                res = minimize(lambda x: compute_overlap_and_gradient(x, n, M_stage2, 2.0/n, gamma, penalty_strength)[0],
                              a_opt, method='L-BFGS-B', jac=lambda x: compute_overlap_and_gradient(x, n, M_stage2, 2.0/n, gamma, penalty_strength)[1],
                              bounds=bounds)
                
                if res.success:
                    a_opt = res.x
                    iter_total += 1
                    iter_count += 1
                    
                    if iter_count % gamma_increase_interval == 0:
                        gamma = min(gamma * 1.5, gamma_final)
                        
            # Final stage: extend to M_max
            if M_stage2 < M_max:
                a_opt_new = np.zeros(M_max)
                a_opt_new[:M_stage2] = a_opt
                a_opt = a_opt_new[:]
                bounds = [(-0.6, 0.6) for _ in range(M_max)]
                
                while iter_total < max_iter_stage1 + max_iter_stage2 + max_iter_stage3 and time.time() - t_start < time_budget_s * 0.9:
                    obj, grad = compute_overlap_and_gradient(a_opt, n, M_max, 2.0/n, gamma, penalty_strength)
                    if obj < best_obj:
                        best_obj = obj
                        best_a = a_opt.copy()
                        
                    res = minimize(lambda x: compute_overlap_and_gradient(x, n, M_max, 2.0/n, gamma, penalty_strength)[0],
                                  a_opt, method='L-BFGS-B', jac=lambda x: compute_overlap_and_gradient(x, n, M_max, 2.0/n, gamma, penalty_strength)[1],
                                  bounds=bounds)
                    
                    if res.success:
                        a_opt = res.x
                        iter_total += 1
                        iter_count += 1
                        
                        if iter_count % gamma_increase_interval == 0:
                            gamma = min(gamma * 1.5, gamma_final)
        
        # Final projection to feasible set
        h_feas = build_h_from_coeffs(best_a, n)
        h_feas = project_to_simplex_box(h_feas, n / 2, n)
        final_obj = evaluate_erdos_solution(h_feas)
        
        if final_obj < best_obj:
            best_h = h_feas
            best_obj = final_obj
            
    return best_h, best_obj

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n = 240
    M_max = 24
    gamma_initial = 0.5
    gamma_final = 40.0
    penalty_strength = 1e4
    max_iter_stage1 = 1500
    max_iter_stage2 = 2000
    max_iter_stage3 = 2500
    gamma_increase_interval = 500
    
    # Optimise
    h_best, c5_bound = spectral_optim(
        n=n, M_max=M_max,
        gamma_initial=gamma_initial, gamma_final=gamma_final,
        penalty_strength=penalty_strength,
        max_iter_stage1=max_iter_stage1,
        max_iter_stage2=max_iter_stage2,
        max_iter_stage3=max_iter_stage3,
        gamma_increase_interval=gamma_increase_interval,
        time_budget_s=budget_s
    )
    
    # Ensure it's feasible and recompute C5
    dx = 2.0 / n
    if h_best is None:
        # Fallback to simple solution if nothing worked
        h_best = np.ones(n) * 0.5
        h_best = np.clip(h_best, 0, 1)
        h_best = project_to_simplex_box(h_best, n / 2, n)
        
    c5_bound = evaluate_erdos_solution(h_best)
    
    return (h_best.tolist(), c5_bound, n)
