def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using full convolution
    overlap = np.correlate(h, 1 - h, mode='full')
    # The maximum value scaled by dx
    c5 = np.max(overlap) * dx
    return c5

def propose_move(h, max_step):
    """Propose a move that maintains sum constraint exactly"""
    h_new = h.copy()
    N = len(h)
    while True:
        i, j = random.sample(range(N), 2)
        # Compute feasible range for delta
        delta_low = max(-h_new[i], h_new[j] - 1.0)
        delta_high = min(1.0 - h_new[i], h_new[j])
        if delta_low >= delta_high:
            continue  # No valid move, try again
        break
    
    # Sample delta from feasible interval
    step = min(max_step, delta_high - delta_low)
    if step <= 0:
        delta = 0.0
    else:
        delta = random.uniform(max(-step, delta_low), min(step, delta_high))
        
    h_new[i] += delta
    h_new[j] -= delta
    return h_new

def tabu_search(initial_h, time_budget, tabu_tenure=10):
    """Tabu search optimization with memory of recent moves"""
    start = time.time()
    cur = initial_h.copy()
    best = initial_h.copy()
    c_cur = c5_of(cur)
    c_best = c_cur
    
    tabu_list = []
    tabu_tenure = max(1, tabu_tenure)
    
    # Annealing parameters
    T_start = 0.05
    T_end = 1e-6
    step_start = 0.2
    
    move_count = 0
    
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T_start * (T_end / T_start) ** (elapsed / time_budget)
        step = step_start * (1.0 - elapsed / time_budget)
        
        # Try different types of moves
        move_type = random.choice(['swap', 'gaussian', 'local'])
        
        if move_type == 'swap':
            # Try swapping two values
            h_prop = cur.copy()
            N = len(h_prop)
            i, j = random.sample(range(N), 2)
            
            delta_low = max(-h_prop[i], h_prop[j] - 1.0)
            delta_high = min(1.0 - h_prop[i], h_prop[j])
            
            if delta_low >= delta_high:
                continue
                
            delta = random.uniform(max(-step, delta_low), min(step, delta_high))
            h_prop[i] += delta
            h_prop[j] -= delta
        elif move_type == 'gaussian':
            # Small Gaussian perturbation
            h_prop = cur.copy()
            N = len(h_prop)
            i = random.randint(0, N-1)
            eps = random.gauss(0, step)
            h_prop[i] = max(0, min(1, h_prop[i] + eps))
        else:  # local
            h_prop = propose_move(cur, step)
            
        c_prop = c5_of(h_prop)
        
        # Check if this move is in tabu list
        move_key = tuple(sorted([(i, h_prop[i]) for i in range(len(h_prop)) if abs(h_prop[i] - cur[i]) > 1e-6]))
        if move_key in tabu_list and move_type != 'swap':
            # Accept anyway if we're improving
            if c_prop < c_best or random.random() < 0.05:
                pass
            else:
                continue
        
        if c_prop < c_best:
            best = h_prop.copy()
            c_best = c_prop
            # Add new move to tabu list
            if len(tabu_list) >= tabu_tenure:
                tabu_list.pop(0)
            tabu_list.append(move_key)
            
        if c_prop <= c_cur or random.random() < np.exp(-(c_prop - c_cur) / T):
            cur = h_prop
            c_cur = c_prop
            
        move_count += 1
            
    return best, c_best
