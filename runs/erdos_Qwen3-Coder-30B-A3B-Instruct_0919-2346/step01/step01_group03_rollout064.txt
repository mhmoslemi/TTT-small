import numpy as np
import time
from scipy.optimize import differential_evolution, minimize

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral via full convolution of h with (1-h)
    overlaps = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum of the full correlation vector
    # scaled by dx
    c5 = np.max(overlaps) * dx
    return c5

def project_onto_constraints(h, n_points):
    """Project h onto constraints: 0 <= h <= 1 and sum(h)*dx = 1"""
    # Clamp values to [0, 1]
    h = np.clip(h, 0, 1)
    # Rescale to meet integral constraint
    dx = 2.0 / n_points
    current_sum = np.sum(h) * dx
    h = h * (1.0 / current_sum)
    return h

def optimize_with_de(h0, n_points, time_budget):

        recombination=0.7,
        atol=1e-6,
        seed=42,
        workers=1,
        updating='immediate'
    )
    
    best_h = project_onto_constraints(result.x, n_points)
    best_c5 = c5_of(best_h)
    
    return best_h, best_c5
