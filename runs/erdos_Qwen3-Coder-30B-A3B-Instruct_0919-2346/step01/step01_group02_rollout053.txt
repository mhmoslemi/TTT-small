import numpy as np
import time

def c5_of(h):
    """Compute the maximum overlap integral for the given step function h."""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    c = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(c) * dx
    return c5

def worst_shift(h):
    """Identify the shift that achieves the maximum overlap."""
    corr = np.correlate(h, 1-h, mode='full')
    idx = np.argmax(corr)
    return idx - (len(h)-1)  # signed shift

def subgradient(h, s):
    """Compute the subgradient of C5 w.r.t. h at shift s."""
    N = len(h)
    dx = 2.0 / N
    m = abs(s)
    g = np.full(N, dx)  # 1 * dx everywhere
    if m < N:
        g[:N-m] -= dx * h[m:]  # subtract h[i+m] where it exists
        g[m:]   -= dx * h[:N-m]  # subtract h[i-m] where it exists
    return g

def project(v, target):
    """Orthogonal projection of v onto [0,1]^n with l_1 sum = target."""
    # Binary search for lambda
    lo = -np.max(v)
    hi = 1.0 - np.min(v)
    for _ in range(50):
        lam = (lo + hi) / 2.0
        w = np.clip(v + lam, 0.0, 1.0)
        if w.sum() < target:
            lo = lam
        else:
            hi = lam
    w = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    # Tiny correction for floating point errors
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        free = (w > 0) & (w < 1)
        if free.any():
            w[free] += diff / free.sum()
            w = np.clip(w, 0.0, 1.0)
    return w

def generate_seeds(N):
    """Generate diverse initial feasible solutions."""
    seeds = []
    
    # 1. Use initial_h_values if available
    if 'initial_h_values' in globals():
        h0 = np.interp(np.linspace(0, 2, N),
                       np.linspace(0, 2, len(initial_h_values)),
                       initial_h_values)
        h0 = h0 * (N / 2) / np.sum(h0)
        h0 = project(h0, N / 2)  # Ensure sum = N/2
        seeds.append(h0)
    
    # 2. Two-peak template with multiple a values
    a_values = [0.12, 0.15, 0.2, 0.25, 0.3, 0.35, 0.38]
    for a in a_values:
        h = np.ones(N)
        x_vals = np.linspace(0, 2, N)
        u = 0.9
        v = 0.1
        mask = (x_vals < a) | (x_vals >= 2 - a)
        h[mask] = u
        h[~mask] = v
        h = h * (N / 2) / np.sum(h)
        h = project(h, N / 2)
        seeds.append(h)
    
    # 3. Random binary vectors (exactly N/2 ones)
    np.random.seed(42)
    for _ in range(2):
        h = np.zeros(N)
        ones = np.random.choice(N, size=N//2, replace=False)
        h[ones] = 1.0
        h = project(h, N / 2)
        seeds.append(h)
    
    # 4. Uniform 0.5 vector
    h = np.ones(N) * 0.5
    h = project(h, N / 2)
    seeds.append(h)
    
    # Compute C5 and return as (h, c5) pairs
    candidates = []
    for h in seeds:
        c5 = c5_of(h)
        candidates.append((h.copy(), c5))
    return candidates

def stochastic_step(h, c5, T, max_delta, verbose=False):
    """Make one stochastic move to escape local minimum."""
    N = len(h)
    while True:
        i, j = np.random.choice(N, size=2, replace=False)
        delta_low = max(-h[i], h[j] - 1)
        delta_high = min(1 - h[i], h[j])
        if delta_low >= delta_high:
            continue  # No valid move possible
        step_size = np.random.uniform(max(-max_delta, delta_low),
                                      min(max_delta, delta_high))
        h_new = h.copy()
        h_new[i] += step_size
        h_new[j] -= step_size
        
        # Project to unit cube with sum constraint
        h_new = project(h_new, N / 2)
        c5_new = c5_of(h_new)

        # Accept with Metropolis probability
        if c5_new <= c5 or np.random.rand() < np.exp(-(c5_new - c5) / T):
            return h_new, c5_new
        else:
            return h, c5  # reject move (bad state)

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    N = 140
    dx = 2.0 / N

    # Phase 0: Seeding
    start = time.time()
    candidates = generate_seeds(N)
    best_h, best_val = min(candidates, key=lambda x: x[1])
    
    time_left = budget_s - (time.time() - start)
    if time_left <= 0.1:
        return (best_h, best_val, N)
    
    # Phase 1: Sub-gradientProjected Descent (SPD)
    t_start = time.time()
    t_end = t_start + 0.55 * budget_s
    h = best_h.copy()
    c5 = best_val
    
    eta_0 = 30.0 * dx
    gamma = 0.9995  # decay factor for step size
    t = 0
    while time.time() < t_end:
        s_star = worst_shift(h)
        g = subgradient(h, s_star)
        eta = eta_0 * (gamma ** t)
        # limit eta to prevent too large steps
        eta = min(eta, 50 * dx)
        v = h - eta * g
        h = project(v, N / 2)
        c5 = c5_of(h)
        if c5 < best_val:
            best_val = c5
            best_h = h.copy()
        
        # Occasionally jitter to escape plateau
        if t % 30 == 0:
            h = project(h + np.random.uniform(-0.02, 0.02, size=N), N / 2)
        
        t += 1
    
    time_left = budget_s - (time.time() - start)
    if time_left <= 0.1:
        return (best_h, best_val, N)
    
    # Phase 2: Annealed Stochastic Moves
    t_start = time.time()
    t_end = t_start + 0.3 * budget_s
    h = best_h.copy()
    c5 = best_val
    T0 = 0.05
    Tf = 1e-6
    delta0 = 0.2
    
    while time.time() < t_end:
        t = time.time() - t_start
        tau = t / (t_end - t_start) if t_end > t_start else 1.0
        T = T0 * (Tf / T0) ** tau
        delta_max = delta0 * (1.0 - tau)
        h, c5 = stochastic_step(h, c5, T, delta_max)
        if c5 < best_val:
            best_val = c5
            best_h = h.copy()
    
    time_left = budget_s - (time.time() - start)
    if time_left <= 0.1:
        return (best_h, best_val, N)
    
    # Phase 3: Deterministic Hill Climbing
    t_start = time.time()
    t_end = t_start + 0.1 * budget_s
    h = best_h.copy()
    c5 = best_val
    
    max_delta = 0.02
    while time.time() < t_end:
        i, j = np.random.choice(N, size=2, replace=False)
        delta_low = max(-h[i], h[j] - 1)
        delta_high = min(1 - h[i], h[j])
        if delta_low >= delta_high:
            continue
        step_size = np.random.uniform(max(-max_delta, delta_low),
                                      min(max_delta, delta_high))
        h_new = h.copy()
        h_new[i] += step_size
        h_new[j] -= step_size
        h_new = project(h_new, N / 2)
        c5_new = c5_of(h_new)
        if c5_new < c5:
            h = h_new
            c5 = c5_new
            if c5 < best_val:
                best_val = c5
                best_h = h.copy()
    
    return (best_h, best_val, N)
