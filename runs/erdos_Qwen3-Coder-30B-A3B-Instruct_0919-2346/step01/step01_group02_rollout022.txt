import numpy as np
import time


def c5_of(h):
    dx = 2.0 / len(h)
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx


def worst_shift(h):
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    return idx - (len(h) - 1)


def subgradient(h, s):
    N = len(h)
    dx = 2.0 / N
    m = abs(s)
    g = np.full(N, dx)
    if m < N:
        g[:N - m] -= dx * h[m:]
        g[m:] -= dx * h[:N - m]
    return g


def project(v, target):
    lo = -np.max(v)
    hi = 1.0 - np.min(v)
    for _ in range(50):
        lam = (lo + hi) / 2.0
        w = np.clip(v + lam, 0.0, 1.0)
        if w.sum() < target:
            lo = lam
        else:
            hi = lam
    w = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        free = (w > 0) & (w < 1)
        if free.any():
            w[free] += diff / free.sum()
            w = np.clip(w, 0, 1)
    return w


def spd_phase(h_start, time_budget, t_start):
    cur = h_start.copy()
    c_cur = c5_of(cur)
    best = cur.copy()
    best_c5 = c_cur

    t0 = 30.0  # step size
    gamma = 0.9995  # decay
    step_size = t0
    step_idx = 0

    while time.time() - t_start < time_budget:
        s_star = worst_shift(cur)
        grad = subgradient(cur, s_star)
        step_idx += 1

        # Update
        v = cur - step_size * grad
        cur = project(v, len(cur) / 2)
        c_cur = c5_of(cur)

        if c_cur < best_c5:
            best = cur.copy()
            best_c5 = c_cur

        # Adapt step size to decay
        step_size = t0 * (gamma ** step_idx)

        # Occasionally jitter to escape plateaux
        if step_idx % 30 == 0:
            eps = 0.02
            jitter = np.random.uniform(-eps, eps, size=len(cur))
            v = cur + jitter
            cur = project(v, len(cur) / 2)
            c_cur = c5_of(cur)
            if c_cur < best_c5:
                best = cur.copy()
                best_c5 = c_cur

    return best, best_c5


def anneal_phase(h_start, time_budget, t_start):
    cur = h_start.copy()
    c_cur = c5_of(cur)
    best = cur.copy()
    best_c5 = c_cur

    T_start = 0.05
    T_end = 1e-6
    max_step_start = 0.2

    while time.time() - t_start < time_budget:
        elapsed = time.time() - t_start
        frac = elapsed / time_budget
        T = T_start * (T_end / T_start) ** frac
        max_step = max_step_start * (1.0 - frac)

        # Propose move
        N = len(cur)
        while True:
            i, j = np.random.choice(N, size=2, replace=False)
            delta_low = max(-cur[i], cur[j] - 1)
            delta_high = min(1 - cur[i], cur[j])
            if delta_low >= delta_high:
                continue
            step_size = np.random.uniform(max(-max_step, delta_low),
                                          min(max_step, delta_high))
            h_new = cur.copy()
            h_new[i] += step_size
            h_new[j] -= step_size
            break

        c_new = c5_of(h_new)
        if c_new <= c_cur or np.random.rand() < np.exp(-(c_new - c_cur) / T):
            cur = h_new
            c_cur = c_new
            if c_new < best_c5:
                best = cur.copy()
                best_c5 = c_new

    return best, best_c5


def hill_climb_phase(h_start, time_budget, t_start):
    cur = h_start.copy()
    c_cur = c5_of(cur)
    best = cur.copy()
    best_c5 = c_cur

    while time.time() - t_start < time_budget:
        # Try small moves
        N = len(cur)
        for _ in range(100):
            if time.time() - t_start >= time_budget:
                break
            i, j = np.random.choice(N, size=2, replace=False)
            delta_low = max(-cur[i], cur[j] - 1)
            delta_high = min(1 - cur[i], cur[j])
            if delta_low >= delta_high:
                continue
            step_size = np.random.uniform(max(-0.02, delta_low),
                                          min(0.02, delta_high))
            h_new = cur.copy()
            h_new[i] += step_size
            h_new[j] -= step_size
            c_new = c5_of(h_new)
            if c_new < c_cur:
                cur = h_new
                c_cur = c_new
                if c_new < best_c5:
                    best = cur.copy()
                    best_c5 = c_new

    return best, best_c5


def run(seed=42, budget_s=600):
    np.random.seed(seed)
    N = 140
    dx = 2.0 / N

    # Phase 0: Initialize seeds
    seeds = []
    t_start = time.time()
    # Interpolate if present
    if 'initial_h_values' in globals():
        h0 = np.interp(np.linspace(0, 2, N),
                       np.linspace(0, 2, len(initial_h_values)),
                       initial_h_values)
        h0 = h0 * (N / 2) / np.sum(h0)
        seeds.append(h0)
    else:
        seeds.append(np.ones(N) * 0.5)

    # Two peak template with varying parameter a
    for a in [0.15, 0.2, 0.25, 0.3, 0.35]:
        h = np.ones(N)
        x_vals = np.linspace(0, 2, N)
        u = 0.9
        v = 0.1
        mask = (x_vals < a) | (x_vals >= 2 - a)
        h[mask] = u
        h[~mask] = v
        h = h * (N / 2) / np.sum(h)
        seeds.append(h)

    # Binary seeds with exactly N/2 ones
    for _ in range(3):
        h = np.random.choice([0, 1], size=N, p=[0.5, 0.5])
        h = h * (N / 2) / np.sum(h)
        seeds.append(h)

    # Uniform seed
    h = np.ones(N) * 0.5
    seeds.append(h)

    # Project all to feasible region and evaluate
    for i in range(len(seeds)):
        seeds[i] = project(seeds[i], N / 2)
    c5_vals = [c5_of(h) for h in seeds]
    best_idx = np.argmin(c5_vals)
    best_h = seeds[best_idx].copy()
    best_c5 = c5_vals[best_idx]

    # Phase 1: SPD descent
    time_spd = (budget_s * 0.55) / 1.05
    h_spd, c5_spd = spd_phase(best_h, time_spd, t_start)
    if c5_spd < best_c5:
        best_h = h_spd
        best_c5 = c5_spd

    # Phase 2: Annealing
    time_anneal = (budget_s * 0.30) / 1.05
    h_anneal, c5_anneal = anneal_phase(best_h, time_anneal, t_start)
    if c5_anneal < best_c5:
        best_h = h_anneal
        best_c5 = c5_anneal

    # Phase 3: Hill-climbing
    time_hc = (budget_s * 0.10) / 1.05
    h_hc, c5_hc = hill_climb_phase(best_h, time_hc, t_start)
    if c5_hc < best_c5:
        best_h = h_hc
        best_c5 = c5_hc

    # Final evaluation
    best_c5 = c5_of(best_h)
    return (best_h, best_c5, N)
