import numpy as np
import time

def compute_c5(h):
    """Compute the maximum overlap integral for the given step function h."""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    c = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(c) * dx
    return c5

def compute_gradient(h, shift):
    """Compute gradient of C5 for the specified shift."""
    N = len(h)
    dx = 2.0 / N
    g = np.full(N, 1.0) * dx
    
    if shift >= 0:
        if shift < N:
            # subtract right shift
            g[:N-shift] -= h[shift:] * dx
            # subtract left shift
            g[shift:] -= h[:N-shift] * dx
    else:
        shift = -shift
        if shift < N:
            g[shift:] -= h[:N-shift] * dx
            g[:N-shift] -= h[shift:] * dx
    return g

def project_onto_bounds_and_sum(v, target):
    """Project vector v onto box [0,1] and hyperplane sum=target."""
    # Binary search for lambda
    lower = -np.max(v)
    upper = 1 - np.min(v)
    
    for _ in range(60):
        mid = (lower + upper) / 2.0
        w = np.clip(v + mid, 0, 1)
        s = np.sum(w)
        if s < target:
            lower = mid
        else:
            upper = mid
            
    w = np.clip(v + (lower + upper) / 2.0, 0, 1)
    
    # Correct sum with small adjustments
    diff = target - np.sum(w)
    if abs(diff) > 1e-12:
        free_idx = (w > 0) & (w < 1)
        if np.any(free_idx):
            w[free_idx] += diff / np.sum(free_idx)
            w = np.clip(w, 0, 1)
        else:
            w += diff / len(w)
            w = np.clip(w, 0, 1)
    
    return w

    h1 = h1 * (N/2) / np.sum(h1)
    
    # Uniform
    h2 = np.ones(N) * 0.5
    
    # Generate several random binary seeds
    h_binary_seeds = []
    for _ in range(5):
        h_bin = np.zeros(N)
        idx = np.random.choice(N, N//2, replace=False)
        h_bin[idx] = 1.0
        h_binary_seeds.append(h_bin)
    
    # Try seeds to find best initial approximation 
    best_h = None
    best_c5 = float('inf')
    
    for h_seed in [h0, h1, h2] + h_binary_seeds:
        c5 = compute_c5(h_seed)
        if c5 < best_c5:
            best_h = h_seed.copy()
            best_c5 = c5
    
    # Gradient descent with adaptive step size
    start = time.time()
    patience = 0
    patience_limit = 100
    last_improvement = 0

        elapsed = time.time() - start

            # Accept if it's better or with some probability

                current_h = new_h
                current_c5 = new_c5
                best_c5 = min(best_c5, current_c5)
            
    return (best_h, best_c5, N)
