import numpy as np
import time
import random
from scipy.optimize import differential_evolution, minimize

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    overlap = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(overlap) * dx
    return c5

def project_to_feasible(h):
    """Project h to satisfy the constraints: 0 <= h[i] <= 1 and sum(h)*dx = 1"""
    # Clip values to [0,1]
    h = np.clip(h, 0, 1)
    # Normalize to sum = 1/dx = n_points/2
    dx = 2.0 / len(h)
    h = h * (len(h) / 2) / np.sum(h)
    return h

def propose_move(h, max_step):
    """Propose a new feasible h vector by adjusting two elements"""
    h_new = h.copy()
    N = len(h)
    
    max_tries = 100
    for _ in range(max_tries):
        i = random.randint(0, N - 1)
        j = random.randint(0, N - 1)
        if i == j:
            continue
            
        # Determine feasible interval for δ
        delta_low = max(-h_new[i], h_new[j] - 1)
        delta_high = min(1 - h_new[i], h_new[j])
        
        if delta_low >= delta_high:
            continue
            
        # Sample δ from uniform distribution over feasible range
        step = min(max_step, delta_high - delta_low)
        if step <= 0:
            continue
        delta = random.uniform(delta_low, delta_high)
        
        # Apply move
        h_new[i] += delta
        h_new[j] -= delta
        
        # Ensure feasibility
        h_new = project_to_feasible(h_new)
        
        return h_new
    
    return h.copy()

def hill_climb_with_perturbation(initial_h, time_budget_s, max_iter=100000):
    """Hill climbing with random perturbations to escape local minima"""
    start = time.time()
    cur = initial_h.copy()
    c_cur = c5_of(cur)
    best = cur.copy()
    c_best = c_cur
    iter_count = 0
    
    while time.time() - start < time_budget_s and iter_count < max_iter:
        iter_count += 1
        # Occasionally do a random large perturbation
        if random.random() < 0.03:
            # Add noise with variance
            noise_scale = 0.1
            h_perturbed = cur + np.random.normal(0, noise_scale, len(cur))
            h_perturbed = project_to_feasible(h_perturbed)
            c_perturbed = c5_of(h_perturbed)
            if c_perturbed < c_cur:
                cur = h_perturbed
                c_cur = c_perturbed
                if c_perturbed < c_best:
                    best = h_perturbed.copy()
                    c_best = c_perturbed
        else:
            # Make a small step based on a random proposed move
            h_proposed = propose_move(cur, 0.05)
            c_proposed = c5_of(h_proposed)
            if c_proposed <= c_cur:  # Accept or tie
                cur = h_proposed
                c_cur = c_proposed
                if c_proposed < c_best:
                    best = h_proposed.copy()
                    c_best = c_proposed
    
    return best, c_best

def optimize_with_scipy(initial_h, time_budget_s):
    """Use scipy to optimize a small subset of variables"""
    start = time.time()
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Define bounds (project to [0,1])
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        # Project h to proper constraints
        h_proj = project_to_feasible(h)
        return c5_of(h_proj)
    
    # Use differential evolution for global optimization
    try:
        result = differential_evolution(
            objective, 
            bounds, 
            maxiter=200, 
            popsize=15, 
            mutation=(0.5, 1),
            recombination=0.7,
            seed=42,
            disp=False
        )
        if time.time() - start < time_budget_s * 0.8:
            # Refine with local method
            h_refined = project_to_feasible(result.x)
            c_refined = c5_of(h_refined)
            # Return better of result.x or refined
            if c_refined < c5_of(initial_h):
                return h_refined, c_refined
            else:
                return initial_h, c5_of(initial_h)
        else:
            return initial_h, c5_of(initial_h)
    except Exception:
        return initial_h, c5_of(initial_h)


def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function"""
    random.seed(seed)
    np.random.seed(seed)
    
    start = time.time()
    N = 140
    dx = 2.0 / N
    
    # Build initial seeds
    if 'initial_h_values' in globals():
        h_provided = np.interp(
            np.linspace(0, 2, N),
            np.linspace(0, 2, len(initial_h_values)),
            initial_h_values
        )
        # Renormalize to exact sum
        h_provided = h_provided * (N / 2) / np.sum(h_provided)
    else:
        # Create a different initial construction
        h_provided = np.ones(N) * 0.5
    
    # Some alternative ideas:
    # 1. Smooth function with max at center
    h_smooth = np.ones(N) * 0.5
    center = N // 2
    sigma = N / 10
    x = np.arange(N)
    h_smooth = np.exp(-((x - center) ** 2) / (2 * sigma ** 2))
    h_smooth = h_smooth * (N / 2) / np.sum(h_smooth)
    
    # 2. Two peak construction (different than previous)
    h_two_peak = np.full(N, 0.05)
    u = 0.8
    left1 = int(0.3 * N)
    right1 = int(0.4 * N)
    left2 = int(0.6 * N)
    right2 = int(0.7 * N)
    h_two_peak[left1:right1] = u
    h_two_peak[left2:right2] = u
    h_two_peak = h_two_peak * (N / 2) / np.sum(h_two_peak)

    seeds = [h_provided, h_smooth, h_two_peak]
    best_h = None
    best_c5 = float('inf')
    
    remaining_time = budget_s * 0.95
    time_per_seed = min(remaining_time / len(seeds), 200)
    
    for i, h_seed in enumerate(seeds):
        if time.time() - start >= budget_s * 0.9:
            break
        # Run hill climbing with perturbations
        h_opt, c5_opt = hill_climb_with_perturbation(h_seed, time_per_seed)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    # Try optimization with scipy on best result
    if time.time() - start < budget_s * 0.9:
        h_scipy, c5_scipy = optimize_with_scipy(best_h, budget_s - (time.time() - start))
        if c5_scipy < best_c5:
            best_c5 = c5_scipy
            best_h = h_scipy.copy()
    
    # Final polishing using more aggressive local search
    remaining_time = budget_s - (time.time() - start)
    if remaining_time > 10:
        # Run local search for last bit
        try:
            # Try for a quick more intense hill climb
            h_final, c5_final = hill_climb_with_perturbation(best_h, remaining_time)
            if c5_final < best_c5:
                best_c5 = c5_final
                best_h = h_final
        except:
            pass

    return (best_h, best_c5, N)
