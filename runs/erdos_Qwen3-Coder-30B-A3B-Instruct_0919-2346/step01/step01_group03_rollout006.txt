import numpy as np
import time
from scipy.optimize import minimize
from scipy.special import expit

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    overlaps = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(overlaps) * dx
    return c5

def evaluate_gradient(h, dx):
    """Compute gradient of C5 with respect to h."""
    # Construct the correlation matrix partially
    n = len(h)
    grad = np.zeros(n)
    
    # Compute correlations and their gradients
    conv = np.correlate(h, 1 - h, mode="full")
    max_idx = np.argmax(conv)
    
    # The maximum correlation occurs at index max_idx
    # This corresponds to some shift k = max_idx - (n - 1)
    # For simplicity, we'll use finite differences for gradient
    
    # Approximate gradient using finite differences
    eps = 1e-6
    base_val = c5_of(h)
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus *= (len(h) / 2) / np.sum(h_plus)  # Renormalize
        val_plus = c5_of(h_plus)
        grad[i] = (val_plus - base_val) / eps
    
    return grad

def project_to_feasible(h):
    """Project h to satisfy constraint sum(h) * dx = 1"""
    dx = 2.0 / len(h)
    current_sum = np.sum(h)
    h = h * (1.0 / current_sum)  # Scale to unit sum
    h = h * (len(h) / 2)  # Scale to required sum
    return np.clip(h, 0, 1)

    n = len(h0)

    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of(h)
    
    # Simple gradient descent with projection
    step_size = 0.01
    max_iter = 10000
    iter_count = 0
    
    while time.time() - start < budget_s and iter_count < max_iter:
        iter_count += 1
        c5 = c5_of(h)
        
        # Compute gradient
        grad = evaluate_gradient(h, dx)
        
        # Update step
        h_new = h - step_size * grad
        
        # Project to feasible set
        h_new = project_to_feasible(h_new)
        
        c5_new = c5_of(h_new)
        
        if c5_new < c5:
            h = h_new
            if c5_new < best_c5:
                best_h = h.copy()
                best_c5 = c5_new
    
    return best_h, best_c5

def simple_perturbation(h, max_delta=0.1):
    """Simple random perturbation within bounds."""
    h = h.copy()
    n = len(h)
    i = np.random.randint(0, n)
    delta = np.random.uniform(-max_delta, max_delta)
    h[i] = np.clip(h[i] + delta, 0, 1)
    return h

def walkthrough_optimization(h0, budget_s, seed):
    """Simple local search improvement."""
    np.random.seed(seed)
    start = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of(h)
    iter_count = 0
    
    while time.time() - start < budget_s:
        iter_count += 1
        h_new = simple_perturbation(h, 0.05)
        # Ensure it remains normalized
        h_new *= (len(h_new) / 2) / np.sum(h_new)
        c5_new = c5_of(h_new)
        
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
        elif np.random.rand() < 0.1:
            h = h_new
        else:
            h = h_new
            
    return best_h, best_c5
