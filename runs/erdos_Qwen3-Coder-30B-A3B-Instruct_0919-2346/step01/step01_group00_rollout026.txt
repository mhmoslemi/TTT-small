import numpy as np
import time
import random

def compute_C_vector(h, dx):
    """Compute vector of overlaps C_tau for all shifts tau."""
    # Zero-padded autocorrelation of h with itself
    S = np.correlate(h, h, mode='full')
    # Compute overlaps: C_tau = 1 - dx * S_tau
    c = 1.0 - dx * S
    return c

def compute_softmax_weights(c, alpha):
    """Compute softmax weights w_tau = exp(alpha * c_tau) / sum(exp(alpha * c_sigma))."""
    c_max = np.max(c)
    exp_c = np.exp(alpha * (c - c_max))
    w = exp_c / np.sum(exp_c)
    return w

def compute_gradient(h, w, dx):
    """Compute gradient of smooth objective using convolutions."""
    # w is a vector of length 2*n-1, h is length n
    w_rev = w[::-1]
    
    # convolve with h, same mode
    conv1 = np.convolve(w, h, mode='same')
    conv2 = np.convolve(w_rev, h, mode='same')
    
    g = -dx * (conv1 + conv2)
    return g

def project_onto_box_and_mass(h, target_sum):
    """Project h onto [0,1] and total mass = target_sum using water-filling."""
    # Note: we want h' such that 
    # 1) h'_i in [0,1] for all i, and
    # 2) sum_i h'_i = target_sum
    # By optimality condition, there exists a λ such that the solution is
    # h'_i = clip( h_i - λ, 0, 1 )
    # and sum_i clip(h_i - λ, 0, 1) = target_sum
    
    h = np.asarray(h)
    n = len(h)
    h_sorted = np.sort(h)[::-1]  # descending order
    s = np.cumsum(h_sorted)  # prefix sums
    
    # We're solving for lambda by bisection of function φ(λ) = sum(clamp(h_i - λ, 0, 1)) - target_sum
    # In the sorted list, indices k where h_sorted[k] >= λ contribute h_sorted[k] - λ, else 0
    
    def phi(λ):
        # Sum of all clipped h_sorted[k] - λ (k = 0..m-1 where h_sorted[k] >= λ)
        # = sum_{k=0}^{m-1} (h_sorted[k] - λ) = s[m] - m*λ 
        # But find the correct index m: h_sorted[m-1] >= λ but h_sorted[m] < λ
        k = 0
        while k < n and h_sorted[k] >= λ:
            k += 1
        return s[k] - k * λ - target_sum
    
    # Initial interval - could be tightened but this is safe
    λ_min = np.min(h) - 1.0
    λ_max = np.max(h) + 1.0
    
    # Bisection
    epsilon = 1e-12
    while abs(λ_max - λ_min) > epsilon:
        λ = (λ_min + λ_max) / 2.0
        if phi(λ) > 0:
            λ_min = λ
        else:
            λ_max = λ
    
    λ = (λ_min + λ_max) / 2.0
    h_new = np.clip(h - λ, 0.0, 1.0)
    return h_new

    start_time = time.time()
    elapsed = lambda: time.time() - start_time

    h_uniform, h_initial, h_double = build_initial_h(n_points)

    def evaluate_true_c5(h):
        c = compute_C_vector(h, dx)
        return np.max(c)

    best_h = None
    best_c5 = float('inf')
    best_iter = -1  # To track when improvement occurred

    # Main gradient descent function
    def gradient_descent_seed(h0):
        nonlocal best_h, best_c5
        
        # Initialize variables
        h = h0.copy()
        t = 0
        alpha = 5.0  # Start with low smoothing
        alpha_final = 80.0
        eta0 = 3.0  # Initial step size; tuned so that first step moves at most 0.02 per entry
        beta = 0.7  # Momentum parameter
        momentum = np.zeros_like(h)
        
        max_iter = 100000  # arbitrary limit, use actual time below
        iters_since_improvement = 0
        restart_every = 300
        
        while elapsed() < time_budget * 0.8:
            t += 1
            
            # Time-dependent smoothing and step size
            alpha = alpha + (alpha_final - alpha) * 0.001  # Slow linear increase
            eta = eta0 / np.sqrt(1 + t)
            
            c = compute_C_vector(h, dx)
            w = compute_softmax_weights(c, alpha)
            gradient = compute_gradient(h, w, dx)
            
            momentum = beta * momentum + (1 - beta) * gradient
            h_new = h - eta * momentum
            
            h_new = project_onto_box_and_mass(h_new, target_sum)
            
            true_c5 = evaluate_true_c5(h_new)

                best_c5 = true_c5
                best_h = h_new.copy()
                iters_since_improvement = 0
            else:
                iters_since_improvement += 1
                
            # Occasionally restart with slight perturbation of best solution if stuck
            if iters_since_improvement >= restart_every:

        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()

    # Now a polishing phase with annealing
    if elapsed() < time_budget * 0.99:
        # Noise applied to best solution for polishing
        def propose_and_eval(h, max_step):
            h_new = h.copy()
            i, j = random.sample(range(len(h)), 2)
            
            delta_low = max(-h[i], h[j] - 1.0)
            delta_high = min(1.0 - h[i], h[j])
            if delta_low > delta_high:
                return h, evaluate_true_c5(h)
            
            delta = random.uniform(delta_low, delta_high)
            h_new[i] += delta
            h_new[j] -= delta
            
            return h_new, evaluate_true_c5(h_new)

        # Annealing loop

        cur = best_h.copy()
        c_cur = evaluate_true_c5(cur)
        temp = 0.05
        temp_min = 1e-6
        
        while elapsed() < time_budget and temp > temp_min:
            h_new, c_new = propose_and_eval(cur, 0.1)
            if c_new <= c_cur or random.random() < np.exp(-(c_new - c_cur) / temp):
                cur = h_new
                c_cur = c_new
                if c_cur < best_c5:
                    best_c5 = c_cur
                    best_h = cur.copy()
            
            temp *= 0.999

    return (best_h, best_c5, n_points)
