import numpy as np
import cvxpy as cp
import time

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    overlap = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(overlap) * dx
    return c5

        prev_c5 = c5_of(best_h_prev)
        c5_upper_bound = int(np.ceil(prev_c5 * N / 2))  # Estimate of c_max from previous best
    except:
        c5_upper_bound = 27

    best_solution = None

    # Binary search on integer scale of max overlap (c_max)

    def solve_feasibility(c_max):
        # Decision variables
        x = cp.Variable(N, boolean=True)
        # Create y variables lazily (overlaps)
        y = {}
        # Sum constraint
        constraints = [cp.sum(x) == N // 2]
        
        # Construct y variables and Lagrange product constraints
        for l in range(1, N):
            # Only create relevant Y variables for this lag
            for i in range(N - l):
                y[(l, i)] = cp.Variable(boolean=True)
                constraints += [
                    y[(l, i)] <= x[i],
                    y[(l, i)] <= x[i + l],
                    y[(l, i)] >= x[i] + x[i + l] - 1
                ]
        
        # Overlap constraints with bound c_max
        for l in range(1, N):
            # H_l = sum_{i=0}^{N-l-1} x[i]
            H_l = cp.sum(x[:N-l])
            # O_l = sum_{i=0}^{N-l-1} y[l,i]
            O_l = cp.sum([y[(l, i)] for i in range(N - l)])
            # c_l = H_l - O_l <= c_max
            constraints.append(H_l - O_l <= c_max)
        
        # We only want feasibility
        obj = cp.Minimize(0)
        prob = cp.Problem(obj, constraints)
        
        # Use CBC or other integer solver
        try:
            prob.solve(solver=cp.CBC, time_limit=min(60, budget_s / 10), verbose=False)
        except:
            try:
                prob.solve(solver=cp.GLPK_MI, time_limit=min(60, budget_s / 10), verbose=False)
            except:
                prob.solve(solver=cp.SCIP, time_limit=min(60, budget_s / 10), verbose=False)
        
        # Check if solution is valid
        if prob.status in [cp.OPTIMAL, cp.FEASIBLE]:
            return x.value
        return None

    # Binary search loop
    while low < high:
        mid = (low + high) // 2
        remaining_time = budget_s - (time.time() - start_time)

            break

        solution = solve_feasibility(mid)
        if solution is not None:
            best_solution = solution
            best_c_max = mid
            high = mid

            c_prop = evaluate_c5(h_prop)
            if c_prop < cur_c5:
                h = h_prop
                cur_c5 = c_prop

    return (h, c5_of(h), N)
