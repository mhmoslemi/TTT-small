import numpy as np
import time
import random
from scipy.optimize import differential_evolution, minimize


def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    overlap = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(overlap) * dx
    return c5


def project_to_feasible(h):
    """Project h to feasible region: 0 <= h[i] <= 1 and sum(h)*dx = 1"""
    h = np.clip(h, 0, 1)
    dx = 2.0 / len(h)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / h_sum) * (2.0 / len(h))
    return h


def propose_move(h, max_step):
    """Propose a new feasible h vector by adjusting two elements"""
    h_new = h.copy()
    N = len(h)
    
    # Try to find a valid move
    max_tries = 100
    for _ in range(max_tries):
        i = random.randint(0, N - 1)
        j = random.randint(0, N - 1)
        if i == j:
            continue
            
        # Determine feasible interval for δ
        delta_low = max(-h_new[i], h_new[j] - 1)
        delta_high = min(1 - h_new[i], h_new[j])
        
        if delta_low >= delta_high:
            continue
            
        # Sample δ from uniform distribution over feasible range
        step = min(max_step, delta_high - delta_low)
        if step <= 0:
            continue
        delta = random.uniform(delta_low, delta_high)
        
        # Apply move
        h_new[i] += delta
        h_new[j] -= delta
        
        return h_new
    
    # If no valid move found, return original
    return h.copy()


def local_search(h, budget_s):
    """Perform local search with varying step sizes"""
    start = time.time()
    cur = h.copy()
    c_cur = c5_of(cur)
    best = cur.copy()
    c_best = c_cur
    
    step_sizes = [0.1, 0.05, 0.01, 0.005, 0.001]
    step_idx = 0
    
    iteration = 0
    while time.time() - start < budget_s:
        if iteration % 100 == 0:
            step_idx = min(step_idx + 1, len(step_sizes) - 1)
        step = step_sizes[step_idx]
        
        # Make a proposal
        h_proposed = propose_move(cur, step)
        c_proposed = c5_of(h_proposed)
        
        if c_proposed <= c_cur:
            cur = h_proposed
            c_cur = c_proposed
            if c_proposed < c_best:
                best = h_proposed.copy()
                c_best = c_proposed
                
        iteration += 1
        
    return best, c_best


def find_feasible_center(h, target):
    """Find the maximal central portion that can be set to target value"""
    N = len(h)
    dx = 2.0 / N
    current_sum = np.sum(h)
    
    # If we are already below target, try to increase a central region
    target_sum = target * N * dx
    need = target_sum - current_sum
    
    if need <= 0:
        return h.copy()
        
    # Try to increase a central portion to target, up to feasibility
    h_new = h.copy()
    center_start = N // 2 - 5
    center_end = N // 2 + 5
    
    # Extend to fit target
    mid_start = max(0, center_start)
    mid_end = min(N, center_end)
    
    # Find how much we can increase while staying feasible
    max_increase = min(1.0 - h_new[mid_start:mid_end].min(), 
                       (1.0 - current_sum) / (mid_end - mid_start))
    
    if max_increase > 0:
        # Distribute the increase
        for i in range(mid_start, mid_end):
            h_new[i] = min(1.0, h_new[i] + max_increase)
            
    return h_new


def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function"""
    random.seed(seed)
    np.random.seed(seed)
    
    start = time.time()
    N = 140
    dx = 2.0 / N
    
    # Get initial h from global scope
    if 'initial_h_values' in globals() and len(initial_h_values) > 0:
        h_init = np.interp(
            np.linspace(0, 2, N),
            np.linspace(0, 2, len(initial_h_values)),
            initial_h_values
        )
        h_init = project_to_feasible(h_init)
    else:
        # Construct a default initial point
        h_init = np.full(N, 0.5)
        h_init = project_to_feasible(h_init)
    
    # Try multiple starting points to avoid local optima
    initial_points = [h_init.copy()]
    
    # Add sample points for diversity
    for _ in range(3):
        h_rand = np.random.rand(N)
        h_rand = project_to_feasible(h_rand)
        initial_points.append(h_rand)

    best_h = None
    best_c5 = float('inf')
    
    # Evaluate local minima around initial points with different search strategies
    remaining_time = budget_s * 0.95
    
    # For each starting point, run local search
    for i, h_seed in enumerate(initial_points):
        if time.time() - start >= remaining_time:
            break
            
        # Perform local search
        cur_h, cur_c5 = local_search(h_seed, remaining_time / len(initial_points))
        
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = cur_h.copy()
    
    # Final optimization with pure gradient-free approach
    final_budget = min(100, budget_s - (time.time() - start))
    if final_budget > 10:
        # Try a more focused optimization with good initial guess
        try:
            # Use a simulated annealing approach with more rigid constraints
            def objective(h):
                h_corrected = project_to_feasible(h)
                return c5_of(h_corrected)
        
            bounds = [(0, 1) for _ in range(N)]
            # Try with our best current solution
            result = differential_evolution(
                objective,
                bounds,
                maxiter=500,
                popsize=15,
                mutation=(0.5, 1),
                recombination=0.7,
                seed=seed
            )
            if result.success and result.fun < best_c5:
                h_final = project_to_feasible(result.x)
                c_final = c5_of(h_final)
                if c_final < best_c5:
                    best_c5 = c_final
                    best_h = h_final
        except:
            pass
    
    # Final local search to polish best solution
    if best_h is not None:
        final_h, final_c5 = local_search(best_h, 10)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = final_h
            
    return (best_h, best_c5, N)
