import numpy as np
import time
import random

def compute_C_vector(h, dx):
    """Compute vector of overlaps C_τ for all τ."""
    # Compute full autocorrelation S_τ = sum_i h_i h_{i+τ}
    S = np.correlate(h, h, mode='full')
    # Truncate to valid range
    S = S[len(h)-1:2*len(h)-1]
    # Compute overlaps C_τ = 1 - dx * S_τ
    C = 1.0 - dx * S
    return C

def compute_softmax_weights(c, alpha):
    """Compute softmax weights from overlap vector c."""
    c_max = np.max(c)
    # log-sum-exp trick for numerical stability
    exp_c = np.exp(alpha * (c - c_max))
    w = exp_c / np.sum(exp_c)
    return w

def compute_gradient(h, w, dx):
    """Compute gradient of smooth objective: ∇f_α(h) = -dx * (w * h + w_rev * h)"""
    # Zero-padded convolutions
    g = -dx * (np.convolve(w, h, mode='same') + np.convolve(w[::-1], h, mode='same'))
    return g

def project_to_feasible(h_new, target_sum):
    """Project h_new onto feasible region [0,1] with exact sum = target_sum."""
    # Water-filling projection to box + sum constraint
    h = h_new.copy()
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    
    # Check if we're already close to target
    if abs(current_sum - target_sum) < 1e-12:
        return h
    
    # Bisection on λ to find the right threshold
    def phi(lam):
        # return sum of clip(h_i - lam, 0, 1) - target_sum
        projected = np.clip(h - lam, 0, 1)
        return np.sum(projected) - target_sum
    
    # Determine search interval
    max_h = np.max(h)
    min_h = np.min(h)
    lambda_lo = min_h - 1.0
    lambda_hi = max_h + 1.0
    
    # Refine based on actual bounds
    while phi(lambda_lo) > 0:
        lambda_lo -= 1.0
    while phi(lambda_hi) < 0:
        lambda_hi += 1.0
    
    # Bisection
    tol = 1e-12
    max_iter = 100
    for _ in range(max_iter):
        lamb = (lambda_lo + lambda_hi) / 2
        val = phi(lamb)
        if abs(val) < tol:
            break
        if val > 0:
            lambda_lo = lamb
        else:
            lambda_hi = lamb
    
    h_proj = np.clip(h - lamb, 0, 1)
    return h_proj

    h_initial = np.interp(np.linspace(0, 2, n),

    a = 0.3
    u = 0.9
    v = 0.1
    x = np.linspace(0, 2, n)
    mask = (x < a) | (x > (2 - a))
    h_double[mask] = u
    h_double[~mask] = v
    h_double = h_double * (target_sum / np.sum(h_double))
    
    initial_seeds = [h_uniform, h_initial, h_double]
    best_h = None
    best_c5 = float('inf')
    
    # Phase 1: Gradient descent with smoothing on each seed
    time_budget_per_seed = 0.80 * budget_s / 3.0
    start = time.time()
    
    for seed_idx, h_seed in enumerate(initial_seeds):
        # Reset time tracking for each seed
        iter_start = time.time()
        h = h_seed.copy()
        # Ensure mass constraint is satisfied
        h = h * (target_sum / np.sum(h))
        
        # Optimisation parameters
        alpha_start = 5.0
        alpha_end = 80.0
        eta0 = 3.0  # initial step size
        beta = 0.7  # momentum
        
        # Momentum
        m = np.zeros(n)
        
        # Iterate
        iter_count = 0
        no_improvement_count = 0
        
        while time.time() - iter_start < time_budget_per_seed:
            t = time.time() - start
            
            # Update alpha and step size
            alpha = alpha_start + (alpha_end - alpha_start) * (t / budget_s)
            eta = eta0 / np.sqrt(1.0 + iter_count)
            if iter_count > 0 and iter_count % 100 == 0:
                eta = max(1e-6, eta * 0.95)
            
            # Compute C vector and weights
            C = compute_C_vector(h, dx)
            w = compute_softmax_weights(C, alpha)
            
            # Compute gradient
            g = compute_gradient(h, w, dx)
            
            # Momentum
            m = beta * m + (1 - beta) * g
            
            # Take step
            h_new = h - eta * m
            
            # Project
            h_new = project_to_feasible(h_new, target_sum)
            
            # Evaluate true C5
            C_true = compute_C_vector(h_new, dx)
            c5 = np.max(C_true)
            
            # Check improvement
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
            
            # Restart occasionally

                h_new = h + pert

                    best_h = h_new.copy()

            h = h_new
            iter_count += 1

        h_cur = best_h.copy()
        c5_cur = best_c5
        
        while time.time() - polish_start < polished_time_budget:
            # Propose a move using the same logic as before
            h_new = h_cur.copy()
            i, j = random.sample(range(n), 2)
            
            # Determine feasible interval for delta
            delta_low = max(-h_new[i], h_new[j] - 1.0)
            delta_high = min(1.0 - h_new[i], h_new[j])
            
            if delta_low >= delta_high:
                continue
            
            step = 0.02
            delta = random.uniform(max(delta_low, -step), min(delta_high, step))
            delta = max(delta_low, min(delta, delta_high))
            
            h_new[i] += delta
            h_new[j] -= delta
            
            # Project
            h_new = project_to_feasible(h_new, target_sum)
            
            # Evaluate true C5
            C_new = compute_C_vector(h_new, dx)
            c5_new = np.max(C_new)
            
            if c5_new < c5_cur:
                h_cur = h_new
                c5_cur = c5_new
                if c5_cur < best_c5:
                    best_c5 = c5_cur
                    best_h = h_cur.copy()
    
    # Final evaluation to make sure everything is correct
    if best_h is not None:
        final_C = compute_C_vector(best_h, dx)
        final_c5 = np.max(final_C)
        best_c5 = final_c5
    
    return (best_h, best_c5, n)
