import numpy as np
import time

def c5_of(h):
    dx = 2.0 / len(h)
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def worst_shift(h):
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    return idx - (len(h) - 1)

def subgradient(h, s):
    N = len(h)
    dx = 2.0 / N
    m = abs(s)
    g = np.full(N, dx)
    if m < N:
        g[:N - m] -= dx * h[m:]
        g[m:] -= dx * h[:N - m]
    return g

def project(v, target):
    lo = -np.max(v)
    hi = 1.0 - np.min(v)
    for _ in range(50):
        lam = (lo + hi) / 2.0
        w = np.clip(v + lam, 0.0, 1.0)
        if w.sum() < target:
            lo = lam
        else:
            hi = lam
    w = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        free = (w > 0) & (w < 1)
        if free.any():
            w[free] += diff / free.sum()
            w = np.clip(w, 0, 1)
    return w

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    N = 140
    dx = 2.0 / N
    target_sum = N / 2.0
    
    # Phase 0: Seeding
    start = time.time()
    seeds = []
    if 'initial_h_values' in globals():
        h0 = np.interp(np.linspace(0, 2, N),
                      np.linspace(0, 2, len(initial_h_values)),
                      initial_h_values)
        h0 = h0 * (N / 2) / np.sum(h0)
        seeds.append(h0)
    else:
        h0 = np.ones(N) * 0.5
        seeds.append(h0)
    
    # Two-peak templates
    for a in [0.15, 0.25, 0.35]:
        h = np.ones(N)
        x_vals = np.linspace(0, 2, N)
        u = 0.9
        v = 0.1
        mask = (x_vals < a) | (x_vals >= 2 - a)
        h[mask] = u
        h[~mask] = v
        h = h * (N / 2) / np.sum(h)
        seeds.append(h)
        
    # Random binary vectors
    for _ in range(3):
        h = np.random.choice([0.0, 1.0], size=N)
        ones = np.sum(h)
        if ones == target_sum:
            seeds.append(h)
        elif ones < target_sum:
            # Fill up with 1s
            idx = np.argsort(h)
            h[idx[:int(target_sum - ones)]] = 1.0
            seeds.append(h)
        else:
            # Reduce some 1s to 0s
            idx = np.argsort(h)[int(ones - target_sum):]
            h[idx] = 0.0
            seeds.append(h)
    
    # Uniform vector
    h_uniform = np.ones(N) * 0.5
    seeds.append(h_uniform)
    
    # Project all seeds and evaluate
    best_h = None
    best_c5 = float('inf')
    for h in seeds:
        h = project(h, target_sum)
        c5 = c5_of(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

    # Phase 1: Subgradient Projected Descent
    t_now = time.time()
    if t_now - start > 0.05 * budget_s:
        return (best_h, best_c5, N)
    t_end = start + 0.55 * budget_s
    gamma = 0.9995
    eta_0 = 30 * dx
    t = 0
    h = best_h.copy()
    while time.time() < t_end:
        s = worst_shift(h)
        g = subgradient(h, s)
        eta = eta_0 * (gamma ** t)
        v = h - eta * g
        h_new = project(v, target_sum)
        c5_new = c5_of(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        h = h_new
        t += 1
        if t % 30 == 0:
            # Random jitter
            h = project(h + np.random.uniform(-0.02, 0.02, N), target_sum)
    t_now = time.time()

    # Phase 2: Annealed stochastic moves
    if t_now - start > 0.60 * budget_s:
        return (best_h, best_c5, N)
    t_end = start + 0.90 * budget_s
    T_start = 0.05
    T_end = 1e-6
    step_start = 0.2
    T = T_start
    while time.time() < t_end:
        elapsed = time.time() - start
        frac = elapsed / (0.90 * budget_s)
        T = T_start * (T_end / T_start) ** frac
        step = step_start * (1.0 - frac)
        # Propose move
        i, j = np.random.choice(N, size=2, replace=False)
        delta_low = max(-h[i], h[j] - 1)
        delta_high = min(1 - h[i], h[j])
        if delta_low >= delta_high:
            continue
        step_size = np.random.uniform(max(-step, delta_low),
                                      min(step, delta_high))
        h_new = h.copy()
        h_new[i] += step_size
        h_new[j] -= step_size
        c5_new = c5_of(h_new)
        if c5_new <= best_c5 or np.random.rand() < np.exp(-(c5_new - best_c5) / T):
            h = h_new
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
    t_now = time.time()

    # Phase 3: Deterministic hill climbing
    if t_now - start > 0.95 * budget_s:
        return (best_h, best_c5, N)
    t_end = start + 0.99 * budget_s
    while time.time() < t_end:
        i, j = np.random.choice(N, size=2, replace=False)
        delta_low = max(-best_h[i], best_h[j] - 1)
        delta_high = min(1 - best_h[i], best_h[j])
        if delta_low >= delta_high:
            continue
        step_size = np.random.uniform(max(-0.02, delta_low),
                                      min(0.02, delta_high))
        h_new = best_h.copy()
        h_new[i] += step_size
        h_new[j] -= step_size
        c5_new = c5_of(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
    return (best_h, best_c5, N)
