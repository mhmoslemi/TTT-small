import numpy as np
import time
from scipy.optimize import minimize

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral via full convolution of h with (1-h)
    overlaps = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum of the full correlation vector
    # scaled by dx
    c5 = np.max(overlaps) * dx
    return c5

def project_to_feasible(h):
    """Project h to satisfy the integral constraint."""
    dx = 2.0 / len(h)
    current_sum = np.sum(h) * dx
    h = h * (1.0 / current_sum)
    return h

def perturb_h(h, delta=0.05):
    """Create a small perturbation of h."""
    h_new = h.copy()
    # Pick two random indices
    i, j = np.random.choice(len(h), 2, replace=False)
    # Determine feasible delta range
    delta_min = max(-h_new[i], 1 - h_new[j])
    delta_max = min(1 - h_new[i], h_new[j])
    if delta_min >= delta_max:
        return h_new
    # Apply small random perturbation
    d = np.random.uniform(delta_min * 0.1, delta_max * 0.1)
    h_new[i] += d
    h_new[j] -= d
    return h_new

def local_search(initial_h, budget_s, seed):
    """Use local search with hill climbing."""
    np.random.seed(seed)
    start = time.time()
    
    h = project_to_feasible(initial_h)
    c_current = c5_of(h)
    c_best = c_current
    h_best = h.copy()
    
    while time.time() - start < budget_s:
        h_new = perturb_h(h, 0.02)
        c_new = c5_of(h_new)
        
        if c_new < c_current:
            h = h_new
            c_current = c_new
            if c_new < c_best:
                c_best = c_new
                h_best = h.copy()
        elif np.random.rand() < 0.1:  # Occasionally accept worse moves
            h = h_new
            c_current = c_new
            
    return h_best, c_best

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    N = 140
    dx = 2.0 / N
    
    # Try several initialization strategies
    initial_h_list = []
    
    # 1. Sigmoid-like shape from referenced solution
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        if len(h0) != N:
            x0 = np.linspace(0, 2, len(h0))
            x_new = np.linspace(0, 2, N)
            h0 = np.interp(x_new, x0, h0)
        initial_h_list.append(h0)
    else:
        # Create a default sigmoid-like shape
        x = np.linspace(0, 2, N)
        h = 1.0 / (1.0 + np.exp(-10 * (x - 1)))
        h = h * (N / 2) / np.sum(h)
        initial_h_list.append(h)
    
    # 2. Bump function initialization
    h_bump = np.full(N, 0.0)
    x = np.linspace(0, 2, N)
    idxs = np.where((x >= 0.4) & (x <= 1.6))[0]
    h_bump[idxs] = 1.0
    h_bump = h_bump * (N / 2) / np.sum(h_bump)
    initial_h_list.append(h_bump)
    
    # 3. Double-peak
    h_double = np.full(N, 0.1)
    a = 0.3
    idxs1 = np.where((x < a))[0]
    idxs2 = np.where((x > (2 - a)))[0]
    h_double[idxs1] = 0.9
    h_double[idxs2] = 0.9
    h_double = h_double * (N / 2) / np.sum(h_double)
    initial_h_list.append(h_double)
    
    # 4. Piece-wise linear
    h_linear = np.full(N, 0.0)
    x_mid = 1.0
    for i in range(N):
        if x[i] <= x_mid:
            h_linear[i] = 0.3 + 0.7 * x[i] / x_mid
        else:
            h_linear[i] = 1.0 - 0.7 * (x[i] - x_mid) / (2 - x_mid)
    h_linear = h_linear * (N / 2) / np.sum(h_linear)
    initial_h_list.append(h_linear)
    
    # Run local search on each initialization
    best_h = None
    best_c5 = float('inf')
    
    for i, h_seed in enumerate(initial_h_list):
        if time.time() - time.time() > 0.9 * budget_s:
            break
        h, c5 = local_search(h_seed, 0.25 * budget_s, seed + i)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5
    
    # If we haven't made progress, do another local search with
    # random initialization
    if best_c5 == float('inf'):
        h_random = np.random.rand(N)
        h_random = h_random * (N / 2) / np.sum(h_random)
        h, c5 = local_search(h_random, 0.25 * budget_s, seed + len(initial_h_list))
        if c5 < best_c5:
            best_h = h
            best_c5 = c5
    
    return (best_h, best_c5, N)
