import numpy as np
import time
import cvxpy as cp

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using full convolution
    overlap = np.correlate(h, 1 - h, mode='full')
    # The maximum value scaled by dx
    c5 = np.max(overlap) * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use N=139 (odd) with 69 ones for Paley difference set
    N = 139
    dx = 2.0 / N
    
    # Solve MILP to find optimal binary sequence
    # Variables
    x = cp.Variable(N, boolean=True)
    C = cp.Variable()
    
    # Constraints
    constraints = [cp.sum(x) == 69]  # 69 ones out of 139
    
    # For each non-zero lag
    for ell in range(1, N):
        length = N - ell
        if length <= 0:
            continue
        # Overlapping region indices
        H = cp.sum(x[:length])  # sum of first (N-ell) elements
        
        # Product variables for x[i]*x[i+ell]
        y = cp.Variable(length, boolean=True)
        constraints += [y <= x[:length], 
                        y <= x[ell:],
                        y >= x[:length] + x[ell:] - 1]
        
        O = cp.sum(y)  # Overlap sum
        
        c = H - O  # c_ell = H - O
        
        # Add constraints
        constraints += [c <= C]
        
    # Minimize maximum correlation
    prob = cp.Problem(cp.Minimize(C), constraints)
    
    # Solve
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time <= 1:
        # Fall back to local search if insufficient time
        return fallback_search(N, seed, budget_s)
        
    solver_time = min(remaining_time - 1, 300)
    prob.solve(solver=cp.CBC, max_seconds=solver_time, verbose=False)
    
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        # Extract solution
        x_sol = np.array(x.value).round().astype(int)
        h = x_sol.astype(float)
        c5 = c5_of(h)
        return (h, c5, N)
    else:
        # Fallback to local search
        return fallback_search(N, seed, budget_s)

def fallback_search(N, seed, budget_s):
    # Simple fallback: try known construction + random local search
    np.random.seed(seed)
    
    # Start with a reasonable pattern: single interval of ones
    h = np.zeros(N)
    h[0:N//2] = 1
    h = h * (N / 2) / np.sum(h)  # normalize
    
    # Use a simple hill climbing approach
    start_time = time.time()
    best_h = h.copy()
    best_c5 = c5_of(best_h)
    
    # Simple perturbation moves
    def perturb(h):
        h_new = h.copy()
        i = np.random.randint(0, N)
        j = np.random.randint(0, N)
        if h_new[i] == 1:
            h_new[i] = 0
        else:
            h_new[i] = 1
        # Normalize exactly
        s = np.sum(h_new)
        if s != N / 2:
            # Adjust back to achieve exact sum
            c = 0
            while s > N / 2:
                idx = np.random.randint(0, N)
                if h_new[idx] == 1:
                    h_new[idx] = 0
                    s -= 1
                c += 1
                if c > N:
                    break
            while s < N / 2:
                idx = np.random.randint(0, N)
                if h_new[idx] == 0:
                    h_new[idx] = 1
                    s += 1
                c += 1
                if c > N:
                    break
        return h_new
    
    # Local search
    max_iter = 10000
    for _ in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
        h_prop = perturb(best_h)
        c_prop = c5_of(h_prop)
        if c_prop < best_c5:
            best_h = h_prop
            best_c5 = c_prop
    
    return (best_h, best_c5, N)
