import numpy as np
import time
import random
from scipy.optimize import differential_evolution

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    overlap = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(overlap) * dx
    return c5

def try_move(h, i, j, delta):
    """Try to perform a move on h. Returns new h if valid, None if invalid."""
    h_new = h.copy()
    
    # Original values
    old_i = h_new[i]
    old_j = h_new[j]
    
    # Apply move
    h_new[i] += delta
    h_new[j] -= delta
    
    # Check bounds
    if h_new[i] < 0 or h_new[i] > 1 or h_new[j] < 0 or h_new[j] > 1:
        return None
    
    # Check constraint (sum = 1)
    # We're checking if the sum constraint is satisfied,
    # so we only need to verify that the modified sum matches required sum
    sum_h = np.sum(h_new)
    if abs(sum_h - (len(h_new) / 2.0)) > 1e-10:
        return None
    
    return h_new

def generate_initial_h(n):
    """Generate initial h from provided method but with direct interpolation"""
    # Let's build a good starting point based on the structure 
    # from the previous run plus our improvements
    
    # For now, just return a modified version of `initial_h_values`
    # or fallback to a basic construction.
    
    # If initial_h_values is available, interpolate
    if 'initial_h_values' in globals():
        h_provided = np.interp(np.linspace(0, 2, n),
                            np.linspace(0, 2, len(initial_h_values)),
                            initial_h_values)
        h_provided = h_provided * (n / 2) / np.sum(h_provided)
        return h_provided
    else:
        # Fallback to uniform
        return np.ones(n) * 0.5

def drop_and_add_region(h, start_indices):
    """Modify h by reducing and increasing parts"""
    h_new = h.copy()
    for idx in start_indices:
        h_new[idx] = max(0.0, h_new[idx] - 0.05)
    h_new[start_indices[0]] = min(1.0, h_new[start_indices[0]] + 0.02)
    h_new[start_indices[1]] = min(1.0, h_new[start_indices[1]] + 0.03)
    
    # Renormalize
    current_sum = np.sum(h_new)
    h_new = h_new * (len(h_new) / 2) / current_sum
    return h_new

def gradient_descent_step(h, learning_rate=0.01):
    """Simple step that modifies h to reduce C5 using gradient-based thinking"""
    # Compute gradient via finite differences
    h_new = h.copy()
    
    # Perturb one element slightly and observe change in C5
    dx = 2.0 / len(h)
    
    # Sample direction that helps reduce the cost function
    best_delta = 0.0
    best_c = float('inf')
    
    for delta in [-0.01, -0.005, 0.005, 0.01]:
        h_test = h_new.copy()
        test_idx = random.randint(0, len(h) - 1)
        
        if h_test[test_idx] + delta < 0 or h_test[test_idx] + delta > 1:
            continue
            
        h_test[test_idx] += delta
        
        # Renormalize so sum equals exactly 1
        h_test = h_test * (len(h_test) / 2) / np.sum(h_test)
        
        c_test = c5_of(h_test)
        if c_test < best_c:
            best_c = c_test
            best_delta = delta
    
    if best_delta != 0:
        test_idx = random.randint(0, len(h) - 1)
        h_new[test_idx] += best_delta
        h_new = h_new * (len(h_new) / 2) / np.sum(h_new)
    
    return h_new

def neighborhood_search(h, max_attempts=50):
    """Improved local search based on various operations"""
    best_h = h.copy()
    best_c = c5_of(best_h)
    
    # Multiple exploration moves
    for _ in range(max_attempts):
        # Type 1: Gridear perturbations
        if random.random() < 0.7:
            # One step change in one of the entries
            idx = random.randint(0, len(h)-1)
            h_new = h.copy()
            step_size = random.choice([-0.02, -0.01, 0.01, 0.02])
            h_new[idx] = np.clip(h_new[idx] + step_size, 0, 1)
            
            # Renormalize
            h_new = h_new * (len(h_new) / 2) / np.sum(h_new)
            
            c_new = c5_of(h_new)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
                
        # Type 2: Drop-and-add of two regions
        else:
            if len(h) > 20:
                i = random.randint(0, len(h)-5)
                j = random.randint(i+1, len(h)-1)
                h_new = drop_and_add_region(h, [i, j])
                c_new = c5_of(h_new)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new.copy()
    
    return best_h, best_c

def run(seed=42, budget_s=600, **kwargs):
    """Main function implementing a hybrid global-local search with 
    differential evolution for initial search and local refinement."""
    random.seed(seed)
    np.random.seed(seed)
    
    start_time = time.time()
    
    # Setup parameters
    n_points = 140
    dx = 2.0 / n_points
    time_budget = budget_s
    
    # Constraint: sum(h)*dx = 1 => sum(h) = n_points/2 = 70
    
    # Step 1: Global search using differential evolution
    def objective(h):
        # Normalize the sum
        h_norm = h * (n_points / 2) / np.sum(h)
        return c5_of(h_norm)
    
    bounds = [(0.0, 1.0)] * n_points
    
    # Run differential evolution using wider bounds
    try:
        result = differential_evolution(
            objective,
            bounds,
            maxiter=200,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1.0),
            recombination=0.7,
            seed=seed
        )
        h0 = result.x.copy()
        
        # Normalize and ensure the constraint is satisfied
        h0 = h0 * (n_points / 2) / np.sum(h0)
    except:
        # Fall back to previous method initialization strategy
        h0 = generate_initial_h(n_points)
    
    # Step 2: Local search on top of this solution
    
    # Initialize current solution
    cur_h = h0.copy()
    cur_c = c5_of(cur_h)
    
    best_h = cur_h.copy()
    best_c = cur_c
    
    # Main loop with heuristics
    iterations = 0
    last_improvement = 0
    max_idle = 500
    
    while time.time() - start_time < time_budget - 5:
        iterations += 1
        
        # Decide strategy based on recent history
        if iterations - last_improvement > max_idle:
            # Reset to best solution so far
            cur_h = best_h.copy()
            cur_c = best_c
            last_improvement = iterations
        
        # Fine-tune
        strategy = random.choice(['neighborhood', 'gradient', 'random'])
        
        if strategy == 'neighborhood':
            h_test, c_test = neighborhood_search(cur_h, 10)
        elif strategy == 'gradient':
            h_test = gradient_descent_step(cur_h)
            c_test = c5_of(h_test)
        else:  # random perturbation
            idx = random.randint(0, len(cur_h) - 1)
            h_test = cur_h.copy()
            step = random.choice([-0.02, -0.01, 0.01, 0.02])
            h_test[idx] = np.clip(h_test[idx] + step, 0, 1)
            
            # Renormalize
            h_test = h_test * (len(h_test) / 2) / np.sum(h_test)
            
            c_test = c5_of(h_test)
        
        # Accept the better solution
        if c_test <= cur_c:
            cur_h = h_test.copy()
            cur_c = c_test
            
            if c_test < best_c:
                best_c = c_test
                best_h = h_test.copy()
                last_improvement = iterations
        else:
            # Occasionally accept worse solutions to escape local optima
            if random.random() < 0.01**min(1.0, (time.time() - start_time) / time_budget):
                cur_h = h_test.copy()
                cur_c = c_test
        
        # Early stopping if not improving for too long
        if iterations - last_improvement > max_idle:
            break
    
    return (best_h, best_c, n_points)
