def c5_of(h):
    """Compute C5 = max_k integral h(x)(1-h(x+k)) dx."""
    dx = 2.0 / len(h)
    # Compute correlation: h * (1 - h) with full mode
    c = np.correlate(h, 1 - h, mode="full")
    # Multiply by dx to get continuous integral
    c *= dx
    # Return maximum value among all shifts
    return np.max(c)

def propose_move(h, max_step):
    """Generate a new feasible h by adjusting two entries."""
    h = h.copy()
    N = len(h)
    i, j = random.sample(range(N), 2)
    
    # Determine feasible interval for delta
    delta_low = max(-h[i], h[j] - 1.0)
    delta_high = min(1.0 - h[i], h[j])
    
    if delta_low >= delta_high:
        return h  # No valid move for this pair
    
    # Sample delta within allowed range
    step = min(max_step, delta_high - delta_low)
    delta = random.uniform(-step, step)
    delta = max(delta_low, min(delta, delta_high))
    
    h[i] += delta
    h[j] -= delta
    
    return h

def smooth_move(h, max_step):
    """Generate a smooth perturbation of h."""
    h = h.copy()
    N = len(h)
    # Apply small random changes to adjacent elements
    for _ in range(5):
        i = random.randint(0, N-1)
        delta = random.uniform(-max_step, max_step)
        # Make sure changes stay within bounds
        if 0 <= h[i] + delta <= 1:
            h[i] += delta
        # Ensure normalization constraint is maintained
        total = np.sum(h)
        h = h * (N/2) / total
    return h

        elif move_type == 'smooth':
            h_new = smooth_move(cur, step * 0.5)

            i = random.randint(0, len(cur)-1)

            total = np.sum(h_new)
            h_new = h_new * (len(h_new)/2) / total
            
        c_new = c5_of(h_new)
        
        if c_new <= c_cur or random.random() < np.exp(-(c_new - c_cur) / T):
            cur = h_new
            c_cur = c_new
            if c_new < c_best:
                best = h_new
                c_best = c_new
    
    return best, c_best
