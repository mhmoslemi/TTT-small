import numpy as np
import time
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import pdist

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral via full convolution of h with (1-h)
    overlaps = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum of the full correlation vector
    # scaled by dx
    c5 = np.max(overlaps) * dx
    return c5

def project_feasible(h):
    """Project h to feasible region: [0,1] and integral = 1"""
    h = np.clip(h, 0, 1)
    dx = 2.0 / len(h)
    h = h * (1.0 / np.sum(h)) * (2.0 / dx)
    return h

def propose_move(h, max_step):
    """Propose a feasible move by adjusting two entries."""
    h = h.copy()
    N = len(h)
    # Try to find a valid pair of indices
    for _ in range(100):  # avoid infinite loop
        i, j = np.random.randint(0, N, 2)
        if i == j:
            continue
        # Determine feasible δ range
        delta_low = max(-h[i], h[j] - 1)
        delta_high = min(1 - h[i], h[j])
        if delta_low >= delta_high:
            continue
        # Sample a move with limited step size
        step = min(max_step, delta_high - delta_low)
        delta = np.random.uniform(
            max(-step, delta_low),
            min(step, delta_high)
        )
        # Apply move
        h[i] += delta
        h[j] -= delta
        return h
    # If no valid move found, return copy
    return h

def perturb_solution(h, step_size=0.02):
    """Generate a neighbor solution"""
    h = h.copy()
    N = len(h)
    # Perform multiple localized moves
    for _ in range(max(1, N // 20)):
        h = propose_move(h, step_size)
    return project_feasible(h)

def hill_climb(initial_h, time_budget, seed):
    """Run hill climbing from initial_h."""
    np.random.seed(seed)
    start = time.time()
    cur = project_feasible(initial_h.copy())
    best = cur.copy()
    c_cur = c5_of(cur)
    c_best = c_cur
    
    while time.time() - start < time_budget:
        h_new = perturb_solution(cur, 0.02)
        c_new = c5_of(h_new)
        
        if c_new < c_cur:
            cur = h_new
            c_cur = c_new
            if c_new < c_best:
                best = h_new
                c_best = c_new
        else:
            # Occasionally accept worse solutions to escape local optima
            if np.random.rand() < 0.01:
                cur = h_new
                c_cur = c_new

    return best, c_best

def trust_region_optimize(initial_h, time_budget, seed):
    """Optimize using scipy's trust-constr with constraints"""
    np.random.seed(seed)
    start = time.time()
    
    # Define constraints
    def constraint1(x):
        return np.sum(x) - 1.0  # integral = 1
    
    # Wrap objective for scipy
    def func(x):
        return c5_of(x)
    
    N = len(initial_h)
    bounds = [(0, 1) for _ in range(N)]
    
    # For trust-constr, we handle the integral constraint via a simple method
    # First project the initial solution
    x0 = project_feasible(initial_h).copy()
    
    # Use differential evolution to sample good starting points
    try:
        result = differential_evolution(func, bounds, maxiter=100, seed=seed, 
                                        disp=False, popsize=10)
        if result.success:
            x0 = result.x
    except:
        pass
    
    # Now optimize with trust-region
    try:
        # Create bounds and constraints
        constraints = [{'type': 'eq', 'fun': constraint1}]
        
        # Optimize
        res = minimize(func, x0, method='trust-constr', bounds=bounds, 
                       constraints=constraints, options={'maxiter': 500},
                       callback=lambda x: None)
        
        if res.success and res.fun < c5_of(initial_h):
            return project_feasible(res.x), res.fun
    except:
        pass
    
    return initial_h, c5_of(initial_h)

    np.random.seed(seed)

    N = 140

    # Load initial construction
    if 'initial_h_values' in globals():

        if len(h0) != N:

        h_initial = h0 * (N / 2) / np.sum(h0)
    else:
        h_initial = np.random.rand(N) * 0.5
        h_initial *= (N / 2) / np.sum(h_initial)

    # Generate additional starting points
    
    # 1. Uniform seed
    h_uniform = 0.5 * np.ones(N)
    h_uniform *= (N / 2) / np.sum(h_uniform)
    
    # 2. Double-peak seed
    h_double_peak = np.full(N, 0.1)
    a = 0.3
    idxs = np.where((np.linspace(0, 2, N) < a) | (np.linspace(0, 2, N) > (2 - a)))[0]
    h_double_peak[idxs] = 0.9
    h_double_peak *= (N / 2) / np.sum(h_double_peak)

    # 3. Single peak
    h_single_peak = np.zeros(N)
    peak_x = 1.0
    peak_width = 0.3
    x_vals = np.linspace(0, 2, N)
    h_single_peak = np.exp(-0.5 * ((x_vals - peak_x) / peak_width) ** 2)
    h_single_peak = h_single_peak / np.sum(h_single_peak) * (N / 2)

    # 4. Multi-peak
    h_multi_peak = np.full(N, 0.2)
    peaks = [0.3, 0.7, 1.3, 1.7]
    for p in peaks:
        idxs = np.abs(x_vals - p) < 0.2
        h_multi_peak[idxs] = 0.8
    
    # Try different optimization approaches
    h_seeds = [h_uniform, h_initial, h_double_peak, h_single_peak, h_multi_peak]
    best_h = None
    best_c5 = float('inf')
    
    budget_per_run = budget_s * 0.7 / len(h_seeds)
    
    for i, h_seed in enumerate(h_seeds):
        if time.time() - start > budget_s * 0.9:
            break
            
        # Try trust region approach
        try:
            h_trust, c5_trust = trust_region_optimize(h_seed, budget_per_run, seed + i)
            if c5_trust < best_c5:
                best_h = h_trust
                best_c5 = c5_trust
        except:
            pass
        
        # Also tries hill climbing
        try:
            h_hc, c5_hc = hill_climb(h_seed, budget_per_run, seed + i)
            if c5_hc < best_c5:
                best_h = h_hc
                best_c5 = c5_hc
        except:

    # Final polishing with local search (more aggressive)
    if best_h is not None and time.time() - start < budget_s * 0.95:

        h_polish = best_h.copy()
        c_polish = best_c5

        iterations = 0
        while time.time() - start < budget_s:
            h_new = perturb_solution(h_polish, step_size)
            c_new = c5_of(h_new)
            iterations += 1
            
            if c_new < c_polish:
                h_polish = h_new
                c_polish = c_new
                step_size = max(0.001, step_size * 0.99)
            elif iterations % 10 == 0:
                step_size = min(0.1, step_size * 1.01)  # Sometimes increase step if stuck

        if c_polish < best_c5:
            best_h = h_polish
            best_c5 = c_polish

    return (best_h, best_c5, N)
