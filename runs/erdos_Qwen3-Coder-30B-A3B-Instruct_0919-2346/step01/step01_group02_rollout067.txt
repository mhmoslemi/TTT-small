import numpy as np
import time
from scipy.optimize import differential_evolution
from scipy.spatial.distance import cdist

def c5_of(h):
    """Compute the maximum overlap integral for the given step function h."""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    c = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(c) * dx
    return c5

def propose_move(h, max_step):
    """Propose a new feasible step function by adjusting two entries."""
    N = len(h)
    while True:
        i, j = np.random.choice(N, size=2, replace=False)
        # Determine feasible step
        delta_low = max(-h[i], h[j] - 1)
        delta_high = min(1 - h[i], h[j])
        if delta_low >= delta_high:
            continue  # Try another pair
        # Sample step size
        step_size = np.random.uniform(max(-max_step, delta_low),
                                      min(max_step, delta_high))
        # Create new h
        h_new = h.copy()
        h_new[i] += step_size
        h_new[j] -= step_size
        return h_new

def efficient_anneal(initial_h, time_budget, T_start=0.05, T_end=1e-6, step_start=0.2):
    """Perform simulated annealing to minimize C5."""
    start = time.time()
    cur = initial_h.copy()
    c_cur = c5_of(cur)
    best = cur.copy()
    c_best = c_cur
    T = T_start
    step = step_start
    iteration = 0
    while time.time() - start < time_budget:
        # Update temperature and step size
        elapsed = time.time() - start
        frac = elapsed / time_budget
        T = T_start * (T_end / T_start) ** frac
        step = step_start * (1.0 - frac)
        # Make move
        h_new = propose_move(cur, step)
        c_new = c5_of(h_new)
        # Accept or reject
        if c_new <= c_cur or np.random.rand() < np.exp(-(c_new - c_cur) / T):
            cur = h_new
            c_cur = c_new
            if c_new < c_best:
                best = cur.copy()
                c_best = c_new
        iteration += 1
    return best, c_best
