import numpy as np
import time
from scipy.optimize import minimize
from scipy.special import logsumexp

def c5_of(h):
    """Compute the maximum overlap integral for the given step function h."""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    c = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(c) * dx
    return c5

def project_to_feasible(h, N):
    """Project h to the feasible set: h[i] in [0,1] and sum(h) = N/2."""
    # Water-filling algorithm to project to sum constraint
    v = h.copy()
    target_sum = N / 2
    
    def projection_sum(lam):
        return np.sum(np.clip(v + lam, 0, 1)) - target_sum

    # Binary search for lambda
    lam_low, lam_high = -2.0, 2.0
    for _ in range(50):
        lam = (lam_low + lam_high) / 2
        if projection_sum(lam) < 0:
            lam_low = lam
        else:
            lam_high = lam

    lam = (lam_low + lam_high) / 2
    h_proj = np.clip(v + lam, 0, 1)
    
    # Adjust for potential numerical errors
    current_sum = np.sum(h_proj)
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        # Redistribute the difference among non-bound entries
        free_indices = np.where((h_proj > 0) & (h_proj < 1))[0]
        if len(free_indices) > 0:
            redistribute = diff / len(free_indices)
            h_proj[free_indices] += redistribute
            
    return h_proj

def smooth_surrogate(h, eps):
    """Compute smooth surrogate F_eps and its gradient."""
    N = len(h)
    dx = 2.0 / N
    # Compute all correlations at once
    c = np.correlate(h, 1 - h, mode='full')
    c_scaled = c * dx
    
    # Numerically stable log-sum-exp for epsilon > 0
    max_c = np.max(c_scaled)
    if max_c == -np.inf:
        # if all c_scaled are -inf, something is wrong
        w = np.ones_like(c_scaled) / len(c_scaled)
        grad = np.zeros_like(h)
    else:
        exp_c = np.exp((c_scaled - max_c) / eps)
        Z = np.sum(exp_c)
        w = exp_c / Z

        # Construct gradient using vectorized correlation
        grad = np.zeros_like(h)
        for s in range(len(c_scaled)):
            c_s = c_scaled[s]
            shift = s - N + 1  # actual shift from Python indexing (-N+1 to N-1)
            # manual correlation for gradient of C_s
            if shift == 0:
                # h_i -> -h_i for all i
                grad -= dx * np.ones_like(h)
            elif shift > 0:
                # h[i+shift] contributes to gradient of h[i]
                # h[i-shift] contributes to gradient of h[i]  
                grad[:N-shift] += dx * (-np.ones(N-shift) - h[shift:])
            else:
                # shift < 0
                grad[-shift:] += dx * (-np.ones(N+shift) - h[:-shift])
        
        grad = w @ grad  # weighted sum of all gradient contributions

    # Return function value and gradient
    f = eps * (np.log(Z) + max_c / eps)
    return f, grad  

def continuation_optimize(initial_h, N, budget_s):
    """Optimize using continuation scheme with L-BFGS-B."""
    eps_sequence = [0.10, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    h = initial_h.copy()
    h = project_to_feasible(h, N)
    
    start = time.time()
    
    for eps in eps_sequence:
        if time.time() - start > 0.95 * budget_s:
            break
            
        # Use L-BFGS-B with box constraints and custom projection
        def func_and_grad(x):
            x = project_to_feasible(x, N)
            val, grad = smooth_surrogate(x, eps)
            return val, grad
        
        # Minimal L-BFGS-B optimization
        # Set up bounds [0,1] for all entries
        bounds = [(0., 1.)] * N
        options = {'maxiter': 150, 'gtol': 1e-6}
        try:
            res = minimize(
                lambda x: smooth_surrogate(x, eps)[0],
                h,
                method='L-BFGS-B',
                bounds=bounds,
                jac=lambda x: smooth_surrogate(x, eps)[1],
                options=options
            )
            h = res.x
        except Exception:
            # Fall back to simpler update
            pass
            
        # Final projection after optimization step
        h = project_to_feasible(h, N)
    
    # Final polishing with sub-gradients
    while time.time() - start < 0.98 * budget_s:
        c = np.correlate(h, 1 - h, mode='full')
        c_scaled = c * (2.0 / N)
        s_star = np.argmax(c_scaled)
        # Compute sub-gradient with respect to h on shift s_star
        g = np.zeros_like(h)
        shift = s_star - N + 1
        if shift == 0:
            g -= 2.0 / N * np.ones_like(h)
        elif shift > 0:
            g[:-shift] -= (2.0 / N) * np.ones(N-shift)
        else:
            g[-shift:] -= (2.0 / N) * np.ones(N+shift)
        # Update with small step size
        eta = 0.02
        h_new = h - eta * g
        h_new = project_to_feasible(h_new, N)
        if c5_of(h_new) < c5_of(h):
            h = h_new
        else:
            break
            
    return h

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    N = 140
    dx = 2.0 / N

    # Strategy 5: Multi-start seeds
    seeds = []

    # Uniform seed
    h_uniform = np.ones(N) * 0.5
    h_uniform = project_to_feasible(h_uniform, N)
    seeds.append(("uniform", h_uniform.copy()))

    # Two-peak seeds
    for a in [0.12, 0.18, 0.24, 0.30]:
        u = 0.9
        v = 0.1
        h_two_peak = np.full(N, v)
        x_vals = np.linspace(0, 2, N)
        mask = (x_vals < a) | (x_vals >= 2 - a)
        h_two_peak[mask] = u
        h_two_peak = project_to_feasible(h_two_peak, N)
        seeds.append(("two_peak_a=%.2f"%a, h_two_peak.copy()))

    # Binary seeds - multiple random configurations
    for i in range(5):  # 5 random binary seeds
        h_binary = np.zeros(N)
        indices = np.random.choice(N, size=N//2, replace=False)
        h_binary[indices] = 1
        h_binary = project_to_feasible(h_binary, N)
        seeds.append(("binary_%d"%i, h_binary.copy()))

    # Symmetric binary seed
    h_sym = np.zeros(N)
    half = N // 2
    h_sym[:half] = 1
    h_sym = project_to_feasible(h_sym, N)
    seeds.append(("sym_binary", h_sym.copy()))

    # Random continuous until we get enough seeds or time run out
    random_seeds = 0
    while random_seeds < 5 and (time.time() - time.time()) < 0.75 * budget_s:
        h_rand = np.random.random(N)
        h_rand = project_to_feasible(h_rand, N)
        seeds.append(("random_%d" % random_seeds, h_rand.copy()))
        random_seeds += 1
    
    # Start optimization loop  
    best_h = None
    best_c5 = float('inf')
    start = time.time()
    for name, h_seed in seeds:
        if time.time() - start > 0.75 * budget_s:
            break
        try:
            h_opt = continuation_optimize(h_seed, N, budget_s)
            c5_opt = c5_of(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
        except Exception:
            continue
    
    if best_h is None:
        best_h = np.ones(N) * 0.5  # fallback
        best_h = project_to_feasible(best_h, N)
        best_c5 = c5_of(best_h)
    
    # Final polishing with more sub-gradients
    while time.time() - start < 0.99 * budget_s:
        c = np.correlate(best_h, 1 - best_h, mode='full')
        c_scaled = c * dx
        s_star = np.argmax(c_scaled)
        g = np.zeros_like(best_h)
        shift = s_star - N + 1
        if shift == 0:
            g -= 2*dx * np.ones_like(best_h)
        elif shift > 0:
            g[:-shift] -= (2*dx) * np.ones(N-shift)
        else:
            g[-shift:] -= (2*dx) * np.ones(N+shift)
        eta = 0.02
        h_new = best_h - eta * g
        h_new = project_to_feasible(h_new, N)
        if c5_of(h_new) < c5_of(best_h):
            best_h = h_new
        else:
            break

    return (best_h, best_c5, N)
