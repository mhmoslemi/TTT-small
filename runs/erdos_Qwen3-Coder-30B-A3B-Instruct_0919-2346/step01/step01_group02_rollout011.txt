import numpy as np
import time

def compute_c5(h):
    """Compute the maximum overlap integral for the given step function h."""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    c = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(c) * dx
    return c5

def compute_gradient(h, shift):
    """Compute gradient of C5 with respect to h for a given shift."""
    N = len(h)
    dx = 2.0 / N
    g = np.full(N, 1.0) * dx
    if shift >= 0:
        if shift < N:
            # subtract h[i+shift] for position i
            g[:N-shift] -= h[shift:] * dx
            # subtract h[i-shift] for position i+shift
            g[shift:] -= h[:N-shift] * dx
    else:
        shift = -shift
        if shift < N:
            g[shift:] -= h[:N-shift] * dx
            g[:N-shift] -= h[shift:] * dx
    return g

def project_onto_bounds_and_sum(v, target):
    """Project vector v onto the constraints: 0 <= x <= 1, sum(x) = target."""
    # Binary search for lambda such that sum(clamp(v + lambda, 0, 1)) = target
    lower = -np.max(v)
    upper = 1 - np.min(v)
    for _ in range(60):
        mid = (lower + upper) / 2.0
        w = np.clip(v + mid, 0, 1)
        s = np.sum(w)
        if s < target:
            lower = mid
        else:
            upper = mid
    w = np.clip(v + (lower + upper) / 2.0, 0, 1)
    # Further adjustment to hit target exactly
    diff = target - np.sum(w)
    if abs(diff) > 1e-12:
        free_idx = (w > 0) & (w < 1)
        if np.any(free_idx):
            w[free_idx] += diff / np.sum(free_idx)
            w = np.clip(w, 0, 1)
        else:
            w += diff / len(w)
            w = np.clip(w, 0, 1)
    return w

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    N = 140
    dx = 2.0 / N
    best_h = None
    best_c5 = float('inf')
    
    # Initializes from available seed if present
    if 'initial_h_values' in globals():
        h0 = np.interp(np.linspace(0, 2, N), 
                      np.linspace(0, 2, len(initial_h_values)), 
                      initial_h_values)
        h0 = h0 * (N/2) / np.sum(h0)
    else:
        h0 = np.ones(N) * 0.5  # fallback
    
    # Two-peak seed 
    h1 = np.ones(N)
    x_vals = np.linspace(0, 2, N)
    a = 0.3
    u = 0.9
    v = 0.1
    mask = (x_vals < a) | (x_vals >= 2 - a)
    h1[mask] = u
    h1[~mask] = v
    h1 = h1 * (N/2) / np.sum(h1)  # Renormalize
    
    # Binary seeds
    h2 = np.zeros(N)
    idx = np.random.choice(N, N//2, replace=False)
    h2[idx] = 1.0
    
    # Uniform
    h3 = np.ones(N) * 0.5
    
    seeds = [h0, h1, h2, h3]
    
    # Run gradient descent from each seed
    start = time.time()
    for h_seed in seeds:
        cur_h = h_seed.copy()
        cur_score = compute_c5(cur_h)
        if cur_score < best_c5:
            best_h = cur_h.copy()
            best_c5 = cur_score
            
        inner_iter = 0
        while time.time() - start < budget_s * 0.95:
            # Compute worst shift
            corr = np.correlate(cur_h, 1 - cur_h, mode='full')
            max_idx = np.argmax(corr)
            s = max_idx - (N-1)
            
            # Compute gradient for worst shift
            grad = compute_gradient(cur_h, s)
            
            # Update step size schedule
            t_frac = (time.time() - start) / budget_s
            step = 20.0 * (1 - t_frac) + 0.1
            
            # Update
            v = cur_h - step * grad
            cur_h = project_onto_bounds_and_sum(v, N/2)
            cur_score = compute_c5(cur_h)
            
            # Update best if improved
            if cur_score < best_c5:
                best_h = cur_h.copy()
                best_c5 = cur_score
                
            # Occasionally 5% random jitter
            if inner_iter % 30 == 0 and np.random.rand() < 0.05:
                noise = np.random.uniform(-0.05, 0.05, size=N)
                v = cur_h + noise
                cur_h = project_onto_bounds_and_sum(v, N/2)
            inner_iter += 1
            
            if time.time() - start > (budget_s * 0.95):
                break
    
    # Final polishing: local hill climbing
    start = time.time()
    best_score = compute_c5(best_h)
    
    while time.time() - start < 0.05 * budget_s:
        i, j = np.random.choice(N, size=2, replace=False)
        delta_low = max(-best_h[i], best_h[j] - 1)
        delta_high = min(1 - best_h[i], best_h[j])
        if delta_low >= delta_high:
            continue
        delta = np.random.uniform(delta_low, delta_high)
        h_new = best_h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        # Project to constraints
        h_new = project_onto_bounds_and_sum(h_new, N/2)
        c_new = compute_c5(h_new)
        if c_new < best_score:
            best_h = h_new
            best_score = c_new
    
    return (best_h, best_c5, N)
