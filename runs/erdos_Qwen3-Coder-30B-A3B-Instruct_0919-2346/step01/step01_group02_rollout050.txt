import numpy as np
import time

def c5_of(h):
    """Compute the maximum overlap integral for the given step function h."""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    c = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(c) * dx
    return c5

def compute_gradient(h, shift):
    """Compute gradient of C5 (corresponding to worst shift) with respect to h."""
    N = len(h)
    dx = 2.0 / N
    g = np.full(N, 1.0) * dx
    if shift >= 0:
        if shift < N:
            g[:N-shift] -= h[shift:] * dx
            g[shift:] -= h[:N-shift] * dx
    else:
        shift = -shift
        if shift < N:
            g[shift:] -= h[:N-shift] * dx
            g[:N-shift] -= h[shift:] * dx
    return g

def project_onto_bounds_and_sum(v, target):
    """Project vector v onto box [0,1] and hyperplane sum=target."""
    # Binary search for lambda that makes sum = target
    lower = -np.max(v) - 1
    upper = 1 - np.min(v) + 1
    tol = 1e-12
    
    for _ in range(100):
        lam = (lower + upper) / 2.0
        w = np.clip(v + lam, 0, 1)
        s = np.sum(w)
        if s < target:
            lower = lam
        else:
            upper = lam
    
    w = np.clip(v + (lower + upper) / 2.0, 0, 1)
    diff = target - np.sum(w)
    if abs(diff) > tol:
        # Distribute to non-bound entries
        free_idx = (w > 0) & (w < 1)
        if np.any(free_idx):
            w[free_idx] += diff / np.sum(free_idx)
            w = np.clip(w, 0, 1)
        else:
            # All at bounds: redistribute and reclip
            w += diff / len(w)
            w = np.clip(w, 0, 1)
    return w

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    N = 140
    dx = 2.0 / N
    
    # Initial candidate from global if exists, otherwise uniform
    if 'initial_h_values' in globals():
        h0 = np.interp(np.linspace(0, 2, N), 
                      np.linspace(0, 2, len(initial_h_values)), 
                      initial_h_values)
        h0 = h0 * (N/2) / np.sum(h0)
    else:
        h0 = np.ones(N) * 0.5

    # Two-peak shape
    h1 = np.ones(N)
    x_vals = np.linspace(0, 2, N)
    a = 0.3
    u = 0.9
    v = 0.1
    mask = (x_vals < a) | (x_vals >= 2 - a)
    h1[mask] = u
    h1[~mask] = v
    h1 = h1 * (N/2) / np.sum(h1)

    # Uniform
    h2 = np.ones(N) * 0.5

    # Binary seed
    h3 = np.zeros(N)
    idx = np.random.choice(N, N//2, replace=False)
    h3[idx] = 1.0

    # Evaluate seeds
    candidates = [h0, h1, h2, h3]
    best_h = None
    best_c5 = float('inf')
    
    for h in candidates:
        c5 = c5_of(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

    # Gradient descent with restarts and random jumps
    start = time.time()
    total_iterations = 0
    
    while time.time() - start < budget_s * 0.95:
        # Restart from the current best
        cur_h = best_h.copy()
        cur_score = best_c5
        inner_iters = 0

        # Perform gradient descent steps
        while time.time() - start < budget_s * 0.95 and inner_iters < 200:
            # Compute worst shift and its contribution
            corr = np.correlate(cur_h, 1 - cur_h, mode='full')
            max_idx = np.argmax(corr)
            shift = max_idx - (N - 1)
            
            # Compute gradient
            grad = compute_gradient(cur_h, shift)
            
            # Adaptive step size decreasing during run
            elapsed = time.time() - start
            t_frac = elapsed / budget_s
            step = 50.0 * (1 - t_frac) + 0.5
            
            # Update
            v = cur_h - step * grad
            cur_h = project_onto_bounds_and_sum(v, N/2)
            cur_score = c5_of(cur_h)
            
            # Keep track of improvement
            if cur_score < best_c5:
                best_h = cur_h.copy()
                best_c5 = cur_score
                total_iterations = 0  # Reset counter if improvement
            
            total_iterations += 1
            inner_iters += 1
            
            # Occasionally perturb the solution
            if np.random.rand() < 0.15 and inner_iters > 10:
                # Add small random noise then reproject
                noise = np.random.uniform(-0.03, 0.03, size=N)
                v = cur_h + noise
                cur_h = project_onto_bounds_and_sum(v, N/2)
                cur_score = c5_of(cur_h)

    # Final polishing with local improvements
    final_start = time.time()
    while time.time() - final_start < 0.05 * budget_s:
        # Use simple moves to improve slightly
        i, j = np.random.choice(N, size=2, replace=False)
        delta_low = max(-best_h[i], best_h[j] - 1)
        delta_high = min(1 - best_h[i], best_h[j])
        if delta_low >= delta_high:
            continue
        delta = np.random.uniform(0.005, 0.02)  # Keep small for stability
        delta = np.clip(delta, delta_low, delta_high)
        best_h[i] += delta
        best_h[j] -= delta
        # Honor sum constraint
        residual = N/2 - np.sum(best_h)
        best_h += residual / N
        best_h = np.clip(best_h, 0, 1)
        new_score = c5_of(best_h)
        if new_score < best_c5:
            best_c5 = new_score

    return (best_h, best_c5, N)
