import numpy as np
import time

def c5_of(h):
    """Compute the maximum overlap integral for the given step function h."""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    c = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(c) * dx
    return c5

def worst_shift(h):
    """Find the shift index that maximizes the overlap integral."""
    corr = np.correlate(h, 1-h, mode='full')
    idx = np.argmax(corr)
    return idx - (len(h)-1)

def subgradient(h, s):
    """Compute subgradient of C5 at shift s."""
    N = len(h)
    dx = 2.0 / N
    m = abs(s)
    g = np.full(N, dx)
    if m < N:
        g[:N-m] -= dx * h[m:]
        g[m:]   -= dx * h[:N-m]
    return g

def project(v, target):
    """Project v onto the feasible set: 0 <= h <= 1, sum(h) = target."""
    # Binary search for lambda such that sum clip(v+lambda,0,1) = target
    lo = -np.max(v)
    hi = 1.0 - np.min(v)
    for _ in range(50):
        lam = (lo + hi) / 2.0
        w = np.clip(v + lam, 0.0, 1.0)
        if w.sum() < target:
            lo = lam
        else:
            hi = lam
    w = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    # Correct for roundoff error
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        free = (w > 0) & (w < 1)
        if free.any():
            w[free] += diff / free.sum()
            w = np.clip(w, 0.0, 1.0)
    return w

def generate_seeds(N, seed):
    np.random.seed(seed)
    seeds = []
    # 1. Use provided initial values if available
    if 'initial_h_values' in globals():
        h_temp = np.interp(np.linspace(0, 2, N), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
        h_temp = h_temp * (N/2) / np.sum(h_temp)
        seeds.append(h_temp)
    else:
        seeds.append(np.ones(N) * 0.5)

    # 2. Symmetric two-peak
    h = np.ones(N)
    x_vals = np.linspace(0, 2, N)
    a = 0.25  # Fixed for consistency
    u = 0.9
    v = 0.1
    mask = (x_vals < a) | (x_vals >= 2 - a)
    h[mask] = u
    h[~mask] = v
    h = h * (N/2) / np.sum(h)
    seeds.append(h)

    # 3. Binary strings
    for _ in range(3):
        # random sum to N/2
        h = np.random.rand(N)
        # reduce to sum equal to N/2
        h = project(h, N/2)
        seeds.append(h)

    # 4. Uniform vector
    seeds.append(np.ones(N) * 0.5)
    return seeds

def spd_descent(h0, budget_s):
    """Perform subgradient projected descent."""
    N = len(h0)
    dx = 2.0 / N
    h = h0.copy()
    c0 = c5_of(h)
    best_h = h.copy()
    best_c5 = c0
    t_start = time.time()
    eta0 = 0.3  # Starting step size
    gamma = 0.999  # Decay factor
    t = 0  # iteration counter

    while time.time() - t_start < 0.55 * budget_s:
        s = worst_shift(h)
        g = subgradient(h, s)
        eta = eta0 * (gamma ** t)
        v = h - eta * g
        h = project(v, N / 2)
        c = c5_of(h)
        if c < best_c5:
            best_h = h.copy()
            best_c5 = c
        # Occasionally jitter to escape flat regions
        if t % 30 == 29:
            eps = 0.02
            temp_h = h.copy()
            i, j = np.random.choice(N, 2, replace=False)
            delta_low = max(-temp_h[i], temp_h[j] - 1)
            delta_high = min(1 - temp_h[i], temp_h[j])
            if delta_low < delta_high:
                step = np.random.uniform(delta_low, delta_high)
                temp_h[i] += step
                temp_h[j] -= step
                h = project(temp_h, N/2)
                c = c5_of(h)
                if c < best_c5:
                    best_h = h.copy()
                    best_c5 = c
        t += 1
    return best_h, best_c5

def anneal(h0, budget_s):
    """Perform simulated annealing with adaptive step sizes."""
    N = len(h0)
    dx = 2.0 / N
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of(h)
    t_start = time.time()
    T0 = 0.1
    Tf = 1e-6
    step0 = 0.2
    t = 0
    T = T0
    final_T = Tf
    step_scale = 1.0

    while time.time() - t_start < 0.30 * budget_s:
        # Adaptive temperature schedule
        elapsed = time.time() - t_start
        frac = elapsed / (0.30 * budget_s)
        T = T0 * (final_T / T0) ** frac
        step_scale = 1.0 - frac

        # Propose move
        while True:
            i, j = np.random.choice(N, size=2, replace=False)
            delta_low = max(-h[i], h[j] - 1)
            delta_high = min(1 - h[i], h[j])
            if delta_low >= delta_high:
                continue  # Try another pair
            step_size = np.random.uniform(max(-step0 * step_scale, delta_low),
                                          min(step0 * step_scale, delta_high))
            h_new = h.copy()
            h_new[i] += step_size
            h_new[j] -= step_size
            break

        c_new = c5_of(h_new)
        accept = (c_new <= best_c5 or 
                np.random.rand() < np.exp(-(c_new - best_c5) / max(T, 1e-10)))
        if accept:
            h = h_new
            if c_new < best_c5:
                best_h = h.copy()
                best_c5 = c_new
        t += 1
    return best_h, best_c5

def hill_climb(h0, budget_s):
    """Perform deterministic hill climbing."""
    N = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of(h)
    t_start = time.time()
    step = 0.02
    while time.time() - t_start < 0.10 * budget_s:
        # Try many random moves and accept if better
        n_moves = 0
        for i in range(1000):  # Limit number of moves
            if time.time() - t_start > 0.10 * budget_s:
                break
            i, j = np.random.choice(N, size=2, replace=False)
            delta_low = max(-h[i], h[j] - 1)
            delta_high = min(1 - h[i], h[j])
            if delta_low < delta_high:
                step_size = np.random.uniform(max(-step, delta_low),
                                              min(step, delta_high))
                h_new = h.copy()
                h_new[i] += step_size
                h_new[j] -= step_size
                c_new = c5_of(h_new)
                if c_new < best_c5:
                    h = h_new
                    best_h = h.copy()
                    best_c5 = c_new
                    n_moves += 1
        if n_moves == 0:
            break  # No improvements found
    return best_h, best_c5

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    N = 140
    dx = 2.0 / N

    # Generate seeds
    seeds = generate_seeds(N, seed)
    best_h = None
    best_c5 = float('inf')
    t_start = time.time()

    # Phase 0: Seeding
    for h in seeds:
        h = project(h, N/2)
        c5 = c5_of(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

    # Phase 1: Sub-gradient projected descent
    h, c5 = spd_descent(best_h, budget_s)
    if c5 < best_c5:
        best_h = h
        best_c5 = c5

    # Phase 2: Simulated Annealing
    h, c5 = anneal(best_h, budget_s)
    if c5 < best_c5:
        best_h = h
        best_c5 = c5

    # Phase 3: Hill Climbing
    h, c5 = hill_climb(best_h, budget_s)
    if c5 < best_c5:
        best_h = h
        best_c5 = c5

    return (best_h, best_c5, N)
