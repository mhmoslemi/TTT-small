import numpy as np
import time

def c5_of(h):
    """Compute the maximum overlap integral for the given step function h."""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    c = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(c) * dx
    return c5

def worst_shift(h):
    """Find the shift s that corresponds to the maximum correlation."""
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    return idx - (len(h) - 1)

def subgradient(h, s):
    """Compute the subgradient of C5 with respect to h for the worst shift s."""
    N = len(h)
    dx = 2.0 / N
    m = abs(s)
    # Start with all 1s
    g = np.full(N, dx)
    if m < N:
        g[:N-m] -= dx * h[m:]
        g[m:] -= dx * h[:N-m]
    return g

def project(v, target):
    """Project vector v onto the feasible set; sum to target, bounds [0,1]."""
    lo = -np.max(v)
    hi = 1.0 - np.min(v)
    for _ in range(50):
        lam = (lo + hi) / 2.0
        w = np.clip(v + lam, 0.0, 1.0)
        if w.sum() < target:
            lo = lam
        else:
            hi = lam
    w = np.clip(v + (lo + hi)/2.0, 0.0, 1.0)
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        free = (w > 0) & (w < 1)
        if free.any():
            w[free] += diff / free.sum()
            w = np.clip(w, 0.0, 1.0)
    return w

def propose_move(h, max_step):
    """Propose a new feasible step function by adjusting two entries."""
    N = len(h)
    while True:
        i, j = np.random.choice(N, size=2, replace=False)
        delta_low = max(-h[i], h[j] - 1)
        delta_high = min(1 - h[i], h[j])
        if delta_low >= delta_high:
            continue  # Try another pair
        step_size = np.random.uniform(max(-max_step, delta_low),
                                      min(max_step, delta_high))
        h_new = h.copy()
        h_new[i] += step_size
        h_new[j] -= step_size
        return h_new

def subgradient_descent(initial_h, budget_s, N):
    """Run gradient descent with projected steps."""
    start = time.time()
    h = initial_h.copy()
    c_old = c5_of(h)
    best_h = h.copy()
    best_val = c_old
    t = 0
    eta0 = 30.0
    gamma = 0.9995
    while time.time() - start < budget_s:
        s = worst_shift(h)
        g = subgradient(h, s)
        eta = eta0 * (gamma ** t)
        if eta < 1e-10:
            eta = 1e-10
        v = h - eta * g
        h = project(v, N / 2.0)
        c_new = c5_of(h)
        if c_new < best_val:
            best_h = h.copy()
            best_val = c_new
        if t % 30 == 0:
            # Small perturbation to escape flat regions
            h_pert = h + np.random.uniform(-0.02, 0.02, size=N)
            h_pert = project(h_pert, N / 2.0)
            c_pert = c5_of(h_pert)
            if c_pert < c_new:
                h = h_pert
                c_new = c_pert
                if c_pert < best_val:
                    best_h = h_pert.copy()
                    best_val = c_pert
        t += 1
    return best_h, best_val

def simulated_annealing(initial_h, budget_s, T_start=0.05, T_end=1e-6):
    """Run simulated annealing."""
    start = time.time()
    cur = initial_h.copy()
    c_cur = c5_of(cur)
    best = cur.copy()
    best_c5 = c_cur
    T = T_start
    while time.time() - start < budget_s:
        elapsed = time.time() - start
        T = T_start * (T_end / T_start) ** (elapsed / budget_s)
        step_size = 0.2 * (1.0 - elapsed / budget_s)
        h_new = propose_move(cur, step_size)
        c_new = c5_of(h_new)
        if c_new <= c_cur or np.random.rand() < np.exp(-(c_new - c_cur) / T):
            cur = h_new
            c_cur = c_new
            if c_new < best_c5:
                best = cur.copy()
                best_c5 = c_new
    return best, best_c5

def deterministic_hill_climb(initial_h, budget_s):
    """Optimize using local moves."""
    start = time.time()
    h = initial_h.copy()
    c = c5_of(h)
    best_h = h.copy()
    best_c = c
    while time.time() - start < budget_s:
        h_new = propose_move(h, 0.02)
        c_new = c5_of(h_new)
        if c_new < c:
            h = h_new
            c = c_new
            if c_new < best_c:
                best_h = h.copy()
                best_c = c_new
    return best_h, best_c

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    # Parameters
    N = 140
    dx = 2.0 / N

    # Seeding: create diverse starts with exact sum constraint
    seeds = []

    # Known initial if available
    if 'initial_h_values' in globals():
        h0 = np.interp(np.linspace(0, 2, N),
                      np.linspace(0, 2, len(initial_h_values)),
                      initial_h_values)
        h0 = h0 * (N / 2) / np.sum(h0)
        seeds.append(h0)

    # Two-peak template (symmetric case)
    h1 = np.ones(N)
    x_vals = np.linspace(0, 2, N)
    a = 0.25
    u = 0.9
    v = 0.1
    mask = (x_vals < a) | (x_vals >= 2 - a)
    h1[mask] = u
    h1[~mask] = v
    h1 = h1 * (N/2) / np.sum(h1)
    seeds.append(h1)

    # Binary seeds
    for _ in range(5):
        h_bin = np.random.choice([0, 1], size=N, p=[0.5, 0.5])
        h_bin = h_bin * (N/2) / np.sum(h_bin)
        seeds.append(h_bin)

    # Uniform
    h2 = np.ones(N) * 0.5
    seeds.append(h2)

    # Evaluate and track starting points
    best_h = None
    best_c5 = float('inf')
    for h_seed in seeds:
        h_proj = project(h_seed, N / 2)
        c = c5_of(h_proj)
        if c < best_c5:
            best_h = h_proj.copy()
            best_c5 = c

    # Phase 1: Subgradient descent (55% of budget)
    phase_budget = 0.55 * budget_s
    h, c = subgradient_descent(best_h, phase_budget, N)
    if c < best_c5:
        best_h = h.copy()
        best_c5 = c

    # Phase 2: Simulated Annealing (30% of budget)
    phase_budget = 0.30 * budget_s
    h, c = simulated_annealing(best_h, phase_budget)
    if c < best_c5:
        best_h = h.copy()
        best_c5 = c

    # Phase 3: Hill Climb Finalization (10% of budget)
    phase_budget = 0.10 * budget_s
    h, c = deterministic_hill_climb(best_h, phase_budget)
    if c < best_c5:
        best_h = h.copy()
        best_c5 = c

    return (best_h, best_c5, N)
