import numpy as np
import time

def c5_of(h):
    dx = 2.0 / len(h)
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def worst_shift(h):
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    return idx - (len(h) - 1)

def subgradient(h, s):
    N = len(h)
    dx = 2.0 / N
    m = abs(s)
    g = np.full(N, dx)
    if m < N:
        g[:N - m] -= dx * h[m:]
        g[m:] -= dx * h[:N - m]
    return g

def project(v, target):
    lo = -np.max(v)
    hi = 1.0 - np.min(v)
    for _ in range(50):
        lam = (lo + hi) / 2.0
        w = np.clip(v + lam, 0.0, 1.0)
        if w.sum() < target:
            lo = lam
        else:
            hi = lam
    w = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        free = (w > 0) & (w < 1)
        if free.any():
            w[free] += diff / free.sum()
            w = np.clip(w, 0, 1)
    return w

def subgradient_descent(h0, max_iter=2000, eta0=30.0, gamma=0.9995, check_freq=30):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of(h)
    for t in range(max_iter):
        s = worst_shift(h)
        g = subgradient(h, s)
        eta = eta0 * (gamma ** t)
        if eta < 1e-10:
            break
        max_step = 50 * (2.0 / len(h))
        eta = min(eta, max_step)
        v = h - eta * g
        h = project(v, len(h) / 2)
        if t % check_freq == 0:
            h_jitter = h + np.random.uniform(-0.02, 0.02, len(h))
            h_jitter = project(h_jitter, len(h) / 2)
            c5_jitter = c5_of(h_jitter)
            if c5_jitter < best_c5:
                best_h = h_jitter.copy()
                best_c5 = c5_jitter
        c5 = c5_of(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
    return best_h, best_c5

def anneal(h0, time_budget, T_start=0.05, T_end=1e-6, step_start=0.2):
    start = time.time()
    cur = h0.copy()
    c_cur = c5_of(cur)
    best = cur.copy()
    c_best = c_cur
    T = T_start
    step = step_start
    N = len(h0)
    dx = 2.0 / N
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        frac = elapsed / time_budget
        T = T_start * (T_end / T_start) ** frac
        step = step_start * (1.0 - frac)

        # Make move
        while True:
            i, j = np.random.choice(N, size=2, replace=False)
            delta_low = max(-cur[i], cur[j] - 1)
            delta_high = min(1 - cur[i], cur[j])
            if delta_low >= delta_high:
                continue
            step_size = np.random.uniform(max(-step, delta_low),
                                          min(step, delta_high))
            h_new = cur.copy()
            h_new[i] += step_size
            h_new[j] -= step_size
            break

        c_new = c5_of(h_new)
        if c_new <= c_cur or np.random.rand() < np.exp(-(c_new - c_cur) / T):
            cur = h_new
            c_cur = c_new
            if c_new < c_best:
                best = cur.copy()
                c_best = c_new
    return best, c_best

def hill_climb(h0, time_budget, step_size=0.02):
    start = time.time()
    cur = h0.copy()
    c_cur = c5_of(cur)
    best = cur.copy()
    c_best = c_cur
    N = len(h0)
    dx = 2.0 / N
    
    while time.time() - start < time_budget:
        # Make move
        while True:
            i, j = np.random.choice(N, size=2, replace=False)
            delta_low = max(-cur[i], cur[j] - 1)
            delta_high = min(1 - cur[i], cur[j])
            if delta_low >= delta_high:
                continue
            step_size = np.random.uniform(max(-step_size, delta_low),
                                          min(step_size, delta_high))
            h_new = cur.copy()
            h_new[i] += step_size
            h_new[j] -= step_size
            break

        c_new = c5_of(h_new)
        if c_new < c_best:
            best = h_new
            c_best = c_new
            cur = h_new
            c_cur = c_new
        elif c_new < c_cur:
            cur = h_new
            c_cur = c_new
    return best, c_best

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    N = 140
    dx = 2.0 / N
    best_h = None
    best_c5 = float('inf')
    
    # Phase 0: Seeding
    start = time.time()
    seeds = []

                      initial_h_values)
        h0 = h0 * (N / 2) / np.sum(h0)
        h0 = project(h0, N / 2)
        seeds.append((h0, c5_of(h0)))
    else:
        h0 = np.ones(N) * 0.5
        h0 = project(h0, N / 2)
        seeds.append((h0, c5_of(h0)))

    # Two-peak seed
    for a in [0.15, 0.25, 0.35]:
        h1 = np.ones(N)
        x_vals = np.linspace(0, 2, N)
        u = 0.9
        v = 0.1
        mask = (x_vals < a) | (x_vals >= 2 - a)
        h1[mask] = u
        h1[~mask] = v
        h1 = h1 * (N / 2) / np.sum(h1)
        h1 = project(h1, N / 2)
        seeds.append((h1, c5_of(h1)))

    # Binary seeds
    for _ in range(3):
        h_bin = np.random.choice([0, 1], size=N)
        h_bin[0] = 1; h_bin[1] = 0
        cur_sum = np.sum(h_bin)
        if cur_sum < N/2:
            # Add more ones in low indices
            ones_idx = np.where(h_bin == 0)[0][:int(N/2 - cur_sum)]
            h_bin[ones_idx] = 1
        elif cur_sum > N/2:
            # Remove some ones
            ones_idx = np.where(h_bin == 1)[0][:int(cur_sum - N/2)]
            h_bin[ones_idx] = 0
        h_bin = project(h_bin, N/2)
        seeds.append((h_bin, c5_of(h_bin)))
    
    # Find the best seed
    for h_seed, c5_seed in seeds:
        if c5_seed < best_c5:
            best_h = h_seed
            best_c5 = c5_seed

    # Phase 1: Subgradient Descent
    spent = time.time() - start
    phase1_time = 0.55 * budget_s
    if spent < phase1_time:
        h_after_descent, c5_after_descent = subgradient_descent(best_h, max_iter=2000)
        if c5_after_descent < best_c5:
            best_h = h_after_descent
            best_c5 = c5_after_descent

    # Phase 2: Annealing
    spent = time.time() - start
    phase2_time = 0.30 * budget_s
    if spent < phase1_time + phase2_time:
        h_after_anneal, c5_after_anneal = anneal(best_h, phase2_time)
        if c5_after_anneal < best_c5:
            best_h = h_after_anneal
            best_c5 = c5_after_anneal

    # Phase 3: Final Hill Climbing
    spent = time.time() - start
    phase3_time = 0.10 * budget_s
    if spent < phase1_time + phase2_time + phase3_time:
        h_final_climb, c5_final = hill_climb(best_h, phase3_time)
        if c5_final < best_c5:
            best_h = h_final_climb
            best_c5 = c5_final

    # Final re-evaluation
    extra_time = 0.05 * budget_s
    final_start = time.time()
    while time.time() - final_start < extra_time:
        h_new = best_h.copy()
        idx1, idx2 = np.random.choice(N, size=2, replace=False)
        delta = np.random.uniform(-0.02, 0.02)
        h_new[idx1] += delta
        h_new[idx2] -= delta
        h_new = project(h_new, len(h_new) / 2)
        new_c5 = c5_of(h_new)
        if new_c5 < best_c5:
            best_h = h_new
            best_c5 = new_c5

    return (best_h, best_c5, N)
