import numpy as np
import time
import random
from scipy.optimize import differential_evolution

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using full convolution
    overlap = np.correlate(h, 1 - h, mode='full')
    # The maximum value scaled by dx
    c5 = np.max(overlap) * dx
    return c5

def project_to_constraints(h, n_points):
    """Project h to satisfy sum constraint exactly"""
    dx = 2.0 / n_points
    current_sum = np.sum(h) * dx
    scale = (n_points / 2) / current_sum
    h_projected = h * scale
    # Ensure bounds
    h_projected = np.clip(h_projected, 0, 1)
    return h_projected

def propose_move(h, max_step, n_points):
    """Propose a move that maintains sum constraint exactly"""
    h_new = h.copy()
    N = len(h)
    
    # Randomly choose two indices
    i, j = random.sample(range(N), 2)
    
    # Compute feasible range for delta
    delta_low = max(-h_new[i], h_new[j] - 1.0)
    delta_high = min(1.0 - h_new[i], h_new[j])
    
    if delta_low >= delta_high:
        return h_new  # No valid move
        
    # Sample delta from feasible interval
    step = min(max_step, delta_high - delta_low)
    if step <= 0:
        delta = 0.0
    else:
        delta = random.uniform(max(-step, delta_low), min(step, delta_high))
        
    h_new[i] += delta
    h_new[j] -= delta
    
    # Project back to constraints
    h_new = project_to_constraints(h_new, n_points)
    return h_new

def local_search(h, n_points, max_iter=1000):
    """Simple local search to improve solution"""
    cur = h.copy()
    c_cur = c5_of(cur)
    
    for _ in range(max_iter):
        # Try a few proposals
        for _ in range(5):
            h_prop = propose_move(cur, 0.1, n_points)
            c_prop = c5_of(h_prop)
            if c_prop < c_cur:
                cur = h_prop
                c_cur = c_prop
    return cur, c_cur
