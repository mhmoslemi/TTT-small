import numpy as np
import time
from scipy.optimize import minimize
from scipy.special import logsumexp

def c5_of(h):
    """Compute the maximum overlap integral for the given step function h."""
    dx = 2.0 / len(h)
    c = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(c) * dx
    return c5

def soft_max_surrogate(h, eps):
    """Compute the smooth surrogate value and gradient for C5."""
    dx = 2.0 / len(h)
    # Compute all correlations
    c = np.correlate(h, 1 - h, mode='full')
    c = c * dx  # Scale by dx
    
    # Subtract max for numerical stability
    c_max = np.max(c)
    c_shifted = c - c_max
    exp_c = np.exp(c_shifted / eps)
    log_sum_exp = c_max + eps * logsumexp(c_shifted / eps)
    
    # Compute gradient (vectorized)
    N = len(h)
    grad = np.zeros(N)
    
    # For each shift s, compute gradient contribution
    shifted_left = np.concatenate([h, np.zeros(N-1)])
    shifted_right = np.concatenate([np.zeros(N-1), h])
    
    # The gradient for each shift is the shift of h[i] and contributions at fixed indices
    for s in range(0, 2*N - 1):
        if s < N:
            # shift j from right for abs(j-s) <= N-1, but also in the difference we write
            # -h[j] + h[j+s] = -h[j+s] + h[j]  -- let's use a smarter way to avoid confusing
            pos = s
            tmp = shifted_left[pos:pos+N] - shifted_right[pos:pos+N]
            sign = 1 if s < N else -1
            tmp = sign * tmp
            # the value (i) goes to tmp[i-s], i ranges from s to N-1, so for i,
            # we're looking at pos=0 and go to -s, ... slightly confusing discussion.
            # better to test!
        else:
            # This might hurt stability, so let's make sure we're accurate
            pos = s
            tmp = shifted_left[pos:pos+N] - shifted_right[pos:pos+N]
        
        # compute actual shift terms (not just left/right shifted)
        # shift s from left to right
        if s < N:
            c_shift_data = np.zeros(N)
            c_shift_data[s:] = h[:-s]
            c_shift_data[:s] = 1  # Zero weight since we're manipulating the shift
        else:
            # s >= N, so we're right shift with subtraction from h[:N-s] to h[:s]
            c_shift_data = np.zeros(N)
            inc = s - N + 1
            c_shift_data[:inc] = h[inc:]  # shift
            c_shift_data[inc:] = 0  # zeros
        c_shift_data = c_shift_data - h
        # actually, compute dC_s / dh[i] = 1 - h[i + s] - h[i - s]  
        # for i = 0 to N-1, we need:
        #   case 1: i + s < N
        #   case 2: i - s >= 0 (position in loop)
        #   case 3: i + s >= N
        # the impact of that is the bipartition of contributions
        
        # Better approach: compute Shifted version of h with right padding zero
        # and overall result is 1 - h[i] - (shifted right) - (shifted left)
        # but for one shift s shift_left[i] = h[i+s], shift_right[i] = h[i-s]
        # so we define:
        shift_left = np.concatenate([np.zeros(s), h[:-s]])
        shift_right = np.concatenate([h[s:], np.zeros(s)])
        g = 1 - shift_left - shift_right  # This is increasing derivative
        g *= dx  # Scaling
        # Combine using soft max weight
        weight = exp_c[s] / np.sum(exp_c)
        grad += weight * g

    return log_sum_exp, grad

def project_to_feasibility(h, N_half):
    """Project h to the intersection of box constraints [0,1] and sum = N_half."""
    v = h.copy()
    # This is waterfilling. Scale v to sum to N_half
    # but want to clip to [0,1] and sum=N_half
    
    # Sort by value
    sort_idx = np.argsort(v)[::-1]
    # We want a point in the intersection of:
    # 0 ≤ v_i ≤ 1  and  sum(v_i) = N_half
    
    # Binary search for lambda such that sum(clamp(v + lambda, 0, 1)) = N_half
    lbd = 0.0
    U = 1.0
    def f(x):
        return np.sum(np.clip(v + x, 0, 1))
    
    # If f(0) already over, underfill the sum
    # Otherwise to above, see the bounds automatically
    
    # Check initial state to test
    if np.sum(np.clip(v, 0, 1)) < N_half:
        # Can we increase all?
        return np.clip(v, 0, 1)
    
    # Set large enough bounds
    lo = -np.inf
    hi = +np.inf
    for _ in range(30):
        mid = (lo + hi) / 2.0
        val = f(mid)
        if val < N_half:
            hi = mid
        else:
            lo = mid
            
    if hi <= -np.inf or lo >= np.inf:
        # fallback  
        return np.clip(v, 0, 1)
        
    lambda_final = (lo + hi) / 2.0
    
    # Clean up 
    h_pro = np.clip(v + lambda_final, 0, 1)
    diff = N_half - np.sum(h_pro)
    # Redistribute evenly
    if abs(diff) > 1e-10:
        diff_per = diff / np.count_nonzero(h_pro)
        h_pro += diff_per
        h_pro = np.clip(h_pro, 0, 1)
    return h_pro

def continuation_optimize(h0, eps_seq, max_iter=150, tol=1e-6):
    """Run continuation optimization using L-BFGS-B on smooth surrogate."""
    h = h0.copy()
    dx = 2.0 / len(h)
    N = len(h)
    N_half = N / 2
    
    best_h = h.copy()
    
    for eps in eps_seq:
        # Project to feasibility
        h = project_to_feasibility(h, N_half)
        
        # Optimization to find local minimum of smooth surrogate
        def func_and_grad(h):
            val, grad = soft_max_surrogate(h, eps)
            return val, grad
        
        # Use L-BFGS-B separately, stop inserting; this is contiguous
        # So we reuse normal bounds info
        bounds = [(0.0, 1.0)] * N
        
        result = minimize(func_and_grad, h, method='L-BFGS-B', jac=True, bounds=bounds, 
                          options={'maxiter': max_iter, 'ftol': tol, 'gtol': tol})
        h = result.x
        h = project_to_feasibility(h, N_half)
        # Check if this improves the true C5
        c5 = c5_of(h)
        if c5 < c5_of(best_h):
            best_h = h.copy()
    
    # Final re-projection and evaluation
    h = project_to_feasibility(h, N_half)
    return h, c5_of(h)
