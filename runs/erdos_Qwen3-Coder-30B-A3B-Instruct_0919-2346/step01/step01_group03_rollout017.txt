import numpy as np
import time
from scipy.optimize import differential_evolution
import math

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral via full convolution of h with (1-h)
    overlaps = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum of the full correlation vector
    # scaled by dx
    c5 = np.max(overlaps) * dx
    return c5

def initial_h():
    """Return initial construction from global variable if available."""
    try:
        return np.array(initial_h_values)
    except:
        # Fallback to a smooth construction
        n = 140
        x = np.linspace(0, 2, n)
        h = 0.5 * np.ones(n)
        # Add a smooth bump in the center
        center = 1.0
        sigma = 0.2
        h += 0.3 * np.exp(-0.5 * ((x - center) / sigma)**2)
        # Normalize
        h *= (n / 2) / np.sum(h)
        return h

def objective(h):
    """Objective function to minimize."""
    return c5_of(h)

def constraint_sums(h):
    """Constraint that integral equals 1."""
    n = len(h)
    dx = 2.0 / n
    return np.sum(h) * dx - 1.0

def constraint_bounds(h):
    """Constraint that all values are in [0,1]."""
    return np.min(h), np.max(h)

def optimize_with_differential_evolution(seed=42, budget_s=600):
    """Optimize using differential evolution."""
    np.random.seed(seed)
    n = 140
    dx = 2.0 / n
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n)]
    
    # Start with initial construction
    x0 = initial_h()
    
    # Normalize
    x0 *= (n / 2) / np.sum(x0)
    
    # Use differential evolution
    result = differential_evolution(
        objective,
        bounds,
        seed=seed,
        maxiter=budget_s // 10,
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.7,
        tol=1e-6,
        init='latinhypercube'
    )
    
    return result.x, result.fun

def local_search(h, time_budget, seed):
    """Improved local search with better move proposal."""
    np.random.seed(seed)
    start = time.time()
    cur_h = h.copy()
    c_cur = c5_of(cur_h)
    best_h = cur_h.copy()
    best_c = c_cur
    
    # Try different step sizes and strategies
    step_sizes = [0.01, 0.02, 0.05, 0.1]
    num_moves = 100
    
    while time.time() - start < time_budget:
        for step in step_sizes:
            for _ in range(num_moves):
                h_new = cur_h.copy()
                
                # Select random indices to modify
                i, j = np.random.choice(len(h_new), 2, replace=False)
                
                # Calculate bounds
                delta_low = max(-h_new[i], h_new[j] - 1)
                delta_high = min(1 - h_new[i], h_new[j])
                
                if delta_low >= delta_high:
                    continue
                
                # Adjust with step size
                delta = np.random.uniform(
                    max(-step, delta_low),
                    min(step, delta_high)
                )
                
                h_new[i] += delta
                h_new[j] -= delta
                
                # Check constraint
                if np.sum(h_new) * (2.0/len(h_new)) < 0.95 or np.sum(h_new) * (2.0/len(h_new)) > 1.05:
                    continue
                    
                c_new = c5_of(h_new)
                
                # Accept better or equal solutions
                if c_new <= c_cur:
                    cur_h = h_new
                    c_cur = c_new
                    
                    if c_new < best_c:
                        best_h = h_new.copy()
                        best_c = c_new
                        
    return best_h, best_c

def run(seed=42, budget_s=600, **kwargs):
    start_time = time.time()
    
    # Use differential evolution first
    h, c5_bound = optimize_with_differential_evolution(seed, budget_s * 0.6)
    
    # Normalize
    n_points = len(h)
    h *= (n_points / 2) / np.sum(h)
    
    # Now local search
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 10:
        h, c5_bound = local_search(h, remaining_time, seed + 100)
    
    # Final check
    h = np.clip(h, 0, 1)
    h *= (n_points / 2) / np.sum(h)
    
    return (h, c5_bound, n_points)
