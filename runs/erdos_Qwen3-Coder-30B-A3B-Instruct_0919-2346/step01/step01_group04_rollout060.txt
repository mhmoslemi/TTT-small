import numpy as np
import time
import random
from scipy.signal import correlate

def compute_c5(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    overlap = correlate(h, 1 - h, mode="full")
    c5 = np.max(overlap) * dx
    return c5

def project_h(h, dx):
    """Project h to satisfy integral constraint (sum h * dx = 1)"""
    scaled_sum = np.sum(h) * dx
    h_proj = h * (1.0 / scaled_sum)
    return np.clip(h_proj, 0, 1)

def smooth_max_min(h, beta=100):
    """Compute a smooth approximation of max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    overlap = correlate(h, 1 - h, mode="full")
    # Use softmax approximation to smooth max
    max_idx = np.argmax(overlap)
    # Smoothed max using softmax over all shifts
    exp_overlap = np.exp(beta * (overlap - overlap[max_idx]))
    # We approximate max by mean of exp terms, which converges to max for large beta
    inv_sum = 1.0 / np.sum(exp_overlap)
    smooth_max = np.sum(exp_overlap * overlap) * inv_sum
    return smooth_max * dx

def smooth_max_min_grad(h, beta=100):
    """Gradient of the smooth max-min approximation"""
    dx = 2.0 / len(h)
    overlap = correlate(h, 1 - h, mode="full")
    max_idx = np.argmax(overlap)
    exp_overlap = np.exp(beta * (overlap - overlap[max_idx]))
    inv_sum = 1.0 / np.sum(exp_overlap)
    # Gradient of smooth max with respect to h
    grad = np.zeros_like(h)
    for i in range(len(h)):
        # Compute derivative of correlation term by hand
        term = 0
        for k in range(len(h)):
            # i-th element of overlap = sum_j h[j] * (1 - h[j+k])
            # We are interested in partial derivative w.r.t. h_i
            if k < len(h) - i and i < k + len(h) - 1:
                term += (1 - h[k+i]) if i == k else 0
        term *= exp_overlap[i]
        grad[i] = term
    smooth_grad = inv_sum * (np.sum(exp_overlap * grad) - smooth_max * np.sum(exp_overlap * grad) * inv_sum)
    return smooth_grad * dx

def run(seed=42, budget_s=600, **kwargs):
    """Optimize h to minimize C5 using a novel hybrid approach"""
    random.seed(seed)
    np.random.seed(seed)
    
    start = time.time()
    N = 200
    dx = 2.0 / N
    
    best_c5 = float('inf')
    best_h = None

    def run_single_optimization(h_seed, time_budget_s):
        """Run a smooth gradient ascent-style optimization"""
        cur_h = h_seed.copy()
        cur_h = project_h(cur_h, dx)
        c5 = compute_c5(cur_h)
        best_h_local = cur_h.copy()
        best_c5_local = c5
        beta = 50.0
        step_size = 0.005
        iterations = 0
        # Run until time expires
        while time.time() - start < time_budget_s and (time.time() - start) < budget_s * 0.95:
            iterations += 1
            # Grad ascent: update h = h + step_size * grad
            grad = smooth_max_min_grad(cur_h, beta)
            # Project the update to prevent overflow
            next_h = cur_h + step_size * grad
            next_h = project_h(next_h, dx)  # This enforce integral and bounds
            
            c5_next = compute_c5(next_h)
            # Accept if better or accept with probability if acceptance rule
            if c5_next < c5:
                cur_h = next_h
                c5 = c5_next
                if c5 < best_c5_local:
                    best_c5_local = c5
                    best_h_local = cur_h.copy()
            # Increase beta to make it more like real max
            beta *= 1.005
            if iterations % 10 == 0:
                # Skip gradient step for diversity if stochastic
                if random.random() > 0.7:
                    step_size *= 0.995
        return best_h_local, best_c5_local

    # Multi-start optimization with different initializations
    seeds = []
    # 1. Uniform
    uniform_h = np.ones(N) * 0.5
    seeds.append(uniform_h)
    
    # 2. Interpolated from initial_h_values if available
    if 'initial_h_values' in globals():
        provided_h = np.interp(
            np.linspace(0, 2, N),
            np.linspace(0, 2, len(initial_h_values)),
            initial_h_values
        )
        # Renormalize to exact sum
        provided_h = provided_h * (N / 2) / np.sum(provided_h)
    else:
        provided_h = np.ones(N) * 0.5
    seeds.append(provided_h)
    
    # 3. Binary-like patterns
    binary_h = np.zeros(N)
    binary_h[::2] = 1.0
    binary_h[1::2] = 0.0
    binary_h = project_h(binary_h, dx)
    seeds.append(binary_h)
    
    # 4. Random continuous noise
    random_h = np.random.rand(N)
    random_h = project_h(random_h, dx)
    seeds.append(random_h)
    
    # Run each seed separately
    remaining_time = budget_s * 0.9
    time_per_seed = remaining_time / len(seeds)
    
    for i, h_seed in enumerate(seeds):
        h_opt, c5_opt = run_single_optimization(h_seed, time_per_seed)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    # Run final polishing with a short annealing-like stair-step descent
    # This is for small perturbations that improve convergence
    if best_h is not None:
        cur_h = best_h.copy()
        c5_current = compute_c5(cur_h)
        # Greedy small moves
        for _ in range(2000):
            if time.time() - start > budget_s:
                break
            # Make small uniform change in two positions
            i, j = random.sample(range(N), 2)
            # Modify the two values in a balanced way
            delta = random.uniform(-0.01, 0.01)
            # Ensure they stay in bounds
            delta_i = min(delta, 1 - cur_h[i])
            delta_j = max(-cur_h[j], -delta_i)
            cur_h[i] += delta_i
            cur_h[j] -= delta_j
            new_c5 = compute_c5(cur_h)
            if new_c5 < c5_current:
                c5_current = new_c5
            else:
                # Restore if worse
                cur_h[i] -= delta_i
                cur_h[j] += delta_j
                
        best_h = cur_h.copy()
        best_c5 = compute_c5(cur_h)
        
    if best_h is None:
        best_h = np.ones(N) * 0.5
        best_c5 = 0.5  # Initial very loose estimate
    
    return (best_h, best_c5, N)
