import numpy as np
import time

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral via full convolution of h with (1-h)
    overlaps = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum of the full correlation vector
    # scaled by dx
    c5 = np.max(overlaps) * dx
    return c5

def compute_c5_and_worst_info(h):
    """Compute C5 and the worst shift info efficiently"""
    N = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr)
    max_val = corr[max_idx]
    offset = max_idx - (N - 1)
    # contributions vector:
    if offset >= 0:
        hi = h[:N - offset]
        hj = h[offset:]
        contrib = hi * (1 - hj)
        i_local = np.argmax(contrib)
        i_index = i_local
        j_index = i_local + offset
    else:
        off = -offset
        hi = h[off:]
        hj = h[:N - off]
        contrib = hi * (1 - hj)
        i_local = np.argmax(contrib)
        i_index = i_local + off
        j_index = i_local
    return max_val * (2.0/N), offset, i_index, j_index, hi[i_local], hj[i_local], contrib[i_local]

def propose_direct_move(h, i, j, step):
    """Propose moving a fraction of mass from i to j"""
    delta = min(step, h[i], 1 - h[j])
    if delta <= 0:
        return None
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new

def project_onto_simplex_box(v, total):
    """Project v onto box [0,1]^N and sum constraint sum = total"""
    # Basic bisection.
    lo = v.min() - 1
    hi = v.max() + 1
    for _ in range(60):
        lam = (lo + hi) / 2
        x = np.clip(v - lam, 0, 1)
        s = x.sum()
        if s > total:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo+hi)/2, 0, 1)

def subgradient_descent(h0, iters, total_sum, step_schedule):
    """Perform subgradient descent with projection"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of(h)
    N = len(h)
    
    for it in range(iters):
        # Compute worst offset 
        _, offset, _, _, _, _, _ = compute_c5_and_worst_info(h)
        
        # Compute gradient for worst shift only
        g = np.ones(N)
        d = offset
        if d >= 0:
            g[:N-d] -= h[d:]
            g[d:]   -= h[:N-d]
        else:
            off = -d
            g[off:] -= h[:N-off]
            g[:N-off] -= h[off:]
        
        # Step size
        step = step_schedule(it, iters)
        
        # Update
        v = h - step * g
        
        # Project onto feasible region
        h = project_onto_simplex_box(v, total_sum)
        
        # Evaluate
        c5 = c5_of(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

def targeted_descent(h0, time_budget, start_time):
    """Run targeted descent with efficient moves for time constraint"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of(h)
    
    N = len(h)
    step = 0.5
    fails = 0
    
    while time.time() - start_time < time_budget:
        # compute worst info
        _, _, i, j, hi_val, hj_val, _ = compute_c5_and_worst_info(h)
        
        # Try moving from i to j
        delta = min(step, h[i], 1 - h[j])
        if delta > 0:
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            c5_new = c5_of(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
                step = min(step * 1.1, 0.5)
                fails = 0
                continue
        
        # Try moving from j to i  
        delta2 = min(step, h[j], 1 - h[i])
        if delta2 > 0:
            h_new2 = h.copy()
            h_new2[j] -= delta2
            h_new2[i] += delta2
            c5_new = c5_of(h_new2)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new2.copy()
                h = h_new2
                step = min(step * 1.1, 0.5)
                fails = 0
                continue
        
        # If no improvement
        step *= 0.9
        fails += 1
        if fails > 3000:
            # Random restart to escape local minima
            idxs = np.random.choice(N, size=2, replace=False)
            delta_r = step * min(h[idxs[0]], 1 - h[idxs[1]])
            if delta_r > 0:
                h = propose_direct_move(h, idxs[0], idxs[1], step)
            fails = 0
            
        # Random small moves occasionally
        if np.random.rand() < 0.02:
            i_rand, j_rand = np.random.choice(N, size=2, replace=False)
            delta_r = step * min(h[i_rand], 1 - h[j_rand])
            if delta_r > 0:
                h_new = h.copy()
                h_new[i_rand] -= delta_r
                h_new[j_rand] += delta_r
                c5_new = c5_of(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new
    
    return best_h, best_c5

    best_h = None
    best_c5 = float('inf')
