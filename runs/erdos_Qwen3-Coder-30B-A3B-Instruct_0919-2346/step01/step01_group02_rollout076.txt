import numpy as np
import time
from scipy.optimize import differential_evolution, minimize
import cvxpy as cp

def c5_of(h):
    """Compute the maximum overlap integral for the given step function h."""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    c = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(c) * dx
    return c5

def evaluate_solution(h, dx, n_points):
    """Evaluate solution with proper normalization and check constraints."""
    # Ensure constraints
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)  # Renormalize exactly
    return c5_of(h)

def propose_move(h, max_step):
    """Propose a new feasible step function by adjusting two entries."""
    N = len(h)
    while True:
        i, j = np.random.choice(N, size=2, replace=False)
        # Determine feasible step
        delta_low = max(-h[i], h[j] - 1)
        delta_high = min(1 - h[i], h[j])
        if delta_low >= delta_high:
            continue  # Try another pair
        # Sample step size
        step_size = np.random.uniform(max(-max_step, delta_low),
                                      min(max_step, delta_high))
        # Create new h
        h_new = h.copy()
        h_new[i] += step_size
        h_new[j] -= step_size
        return h_new

def opt_local_search(h, max_iter, step_size=0.02):
    """Apply local search heuristic to refine solution."""
    current_h = h.copy()
    current_c5 = c5_of(current_h)
    for _ in range(max_iter):
        h_new = propose_move(current_h, step_size)
        c5_new = c5_of(h_new)
        if c5_new < current_c5:
            current_h = h_new
            current_c5 = c5_new
    return current_h, current_c5

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    N = 140
    dx = 2.0 / N
    
    # Try multiple initialization strategies
    initial_solutions = []
    
    # 1. Use provided initial construction
    if 'initial_h_values' in globals():
        h0 = np.interp(np.linspace(0, 2, N), 
                      np.linspace(0, 2, len(initial_h_values)), 
                      initial_h_values)
        h0 = h0 * (N/2) / np.sum(h0)
        initial_solutions.append((h0, "initial"))
    
    # 2. Two-peak construction
    h1 = np.ones(N)
    x_vals = np.linspace(0, 2, N)
    a = 0.3
    u = 0.9
    v = 0.1
    mask = (x_vals < a) | (x_vals >= 2 - a)
    h1[mask] = u
    h1[~mask] = v
    h1 = h1 * (N/2) / np.sum(h1)
    initial_solutions.append((h1, "two_peak"))
    
    # 3. Random construction 
    h2 = np.random.uniform(0.1, 0.9, N)
    h2 = h2 * (N/2) / np.sum(h2)
    initial_solutions.append((h2, "random"))
    
    # 4. Piecewise constant construction
    h3 = np.ones(N) * 0.5
    h3[:N//3] = 0.8
    h3[2*N//3:] = 0.2
    h3 = h3 * (N/2) / np.sum(h3)
    initial_solutions.append((h3, "piecewise"))
    
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Try all initial solutions with simulated annealing
    for h_seed, name in initial_solutions:
        if time.time() - start_time > 0.7 * budget_s:
            break
            
        # Simulated Annealing on this seed
        cur = h_seed.copy()
        c_cur = c5_of(cur)
        best_in_run = cur.copy()
        c_best_in_run = c_cur
        
        # Simulated annealing parameters
        T_start = 0.05
        T_end = 1e-6
        step_size = 0.1
        run_budget = min(0.2 * budget_s, 0.7 * budget_s)
        T = T_start
        
        run_start = time.time()
        while time.time() - run_start < run_budget and time.time() - start_time < budget_s:
            # Update temperature
            elapsed = time.time() - run_start
            frac = elapsed / run_budget
            T = T_start * (T_end / T_start) ** frac
            
            # Make move
            h_new = propose_move(cur, step_size)
            c_new = c5_of(h_new)
            
            # Accept or reject
            if c_new <= c_cur or np.random.rand() < np.exp(-(c_new - c_cur) / T):
                cur = h_new
                c_cur = c_new
                if c_new < c_best_in_run:
                    best_in_run = cur.copy()
                    c_best_in_run = c_new
                    
            # Gradually reduce step size
            step_size = 0.1 * (1.0 - frac)
        
        # Local refinement
        if time.time() - start_time < budget_s:
            refined_h, refined_c5 = opt_local_search(best_in_run, 1000, 0.01)
            if refined_c5 < c_best_in_run:
                best_in_run = refined_h
                c_best_in_run = refined_c5
        
        if c_best_in_run < best_c5:
            best_h = best_in_run.copy()
            best_c5 = c_best_in_run
    
    # Final optimization using deterministic search
    if best_h is not None and time.time() - start_time < 0.3 * budget_s:
        refined_h, refined_c5 = opt_local_search(best_h, 2000, 0.005)
        if refined_c5 < best_c5:
            best_h = refined_h
            best_c5 = refined_c5
    
    return (best_h, best_c5, N)
