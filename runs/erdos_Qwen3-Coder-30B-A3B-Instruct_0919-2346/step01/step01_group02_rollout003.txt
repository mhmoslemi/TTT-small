import numpy as np
import time

def c5_of(h):
    """Compute the maximum overlap integral for the given step function h."""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    c = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(c) * dx
    return c5

def worst_shift(h):
    """Return the shift index that achieves the maximum overlap."""
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    return idx - (len(h) - 1)

def subgradient(h, s):
    """Compute the subgradient of C5 at shift s."""
    N = len(h)
    dx = 2.0 / N
    m = abs(s)
    g = np.full(N, dx)
    if m < N:
        g[:N-m] -= dx * h[m:]
        g[m:]   -= dx * h[:N-m]
    return g

def project(v, target):
    """Project v onto the feasible set {h : 0 <= h <= 1, sum(h) = target}."""
    lo = -np.max(v)
    hi = 1.0 - np.min(v)
    for _ in range(50):
        lam = (lo + hi) / 2.0
        w = np.clip(v + lam, 0.0, 1.0)
        if w.sum() < target:
            lo = lam
        else:
            hi = lam
    w = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        free = (w > 0) & (w < 1)
        if free.any():
            w[free] += diff / free.sum()
            w = np.clip(w, 0, 1)
    return w

def loop_spd(h0, target_sum, max_iter, time_budget, t_start):
    """Sub-gradient projected descent loop."""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of(h)
    t0 = time.time()
    eta0 = 30.0 * (2.0 / len(h))
    gamma = 0.9995
    for t in range(max_iter):
        if time.time() - t_start > time_budget * 0.95:
            break
        s = worst_shift(h)
        g = subgradient(h, s)
        # Adaptive step size
        eta = eta0 * (gamma ** t)
        # Clip step to avoid too aggressive moves
        max_change = 0.5
        step = np.clip(-eta * g, -max_change, max_change)
        v = h + step
        h = project(v, target_sum)
        c5 = c5_of(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
        # Occasionally jitter to escape shallow plateaux
        if t % 30 == 0 and np.random.rand() < 0.3:
            jitter = np.random.uniform(-0.02, 0.02, size=len(h))
            v = h + jitter
            h = project(v, target_sum)
            c5 = c5_of(h)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
    return best_h, best_c5

def loop_anneal(h0, target_sum, n_iter, time_budget, t_start):
    """Annealed stochastic moves loop."""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of(h)
    T_start = 0.05
    T_end = 1e-6
    step_start = 0.2
    
    for i in range(n_iter):
        if time.time() - t_start > time_budget * 0.95:
            break
        elapsed = time.time() - t_start
        frac = elapsed / time_budget
        T = T_start * (T_end / T_start) ** frac
        step = step_start * (1.0 - frac)
        
        # Propose move
        while True:
            i, j = np.random.choice(len(h), size=2, replace=False)
            delta_low = max(-h[i], h[j] - 1)
            delta_high = min(1 - h[i], h[j])
            if delta_low >= delta_high:
                continue
            step_size = np.random.uniform(max(-step, delta_low),
                                          min(step, delta_high))
            h_new = h.copy()
            h_new[i] += step_size
            h_new[j] -= step_size
            break
        
        c_new = c5_of(h_new)
        if c_new <= best_c5:
            h = h_new
            best_h = h_new
            best_c5 = c_new
        elif np.random.rand() < np.exp(-(c_new - best_c5) / T):
            h = h_new
    return best_h, best_c5

def loop_hill_climb(h0, target_sum, time_budget, t_start):
    """Deterministic hill climb loop."""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of(h)
    start = time.time()
    
    while time.time() - t_start < time_budget * 0.95:
        move_made = False
        for _ in range(1000):  # Batch of moves
            if time.time() - start > (time_budget * 0.15) * 0.9:
                break
            h_new = h.copy()
            i, j = np.random.choice(len(h), size=2, replace=False)
            delta_low = max(-h[i], h[j] - 1)
            delta_high = min(1 - h[i], h[j])
            if delta_low >= delta_high:
                continue
            step_size = np.random.uniform(max(-0.02, delta_low),
                                          min(0.02, delta_high))
            h_new[i] += step_size
            h_new[j] -= step_size
            c_new = c5_of(h_new)
            if c_new < best_c5:
                h = h_new
                best_h = h_new
                best_c5 = c_new
                move_made = True
        if not move_made:
            break
    return best_h, best_c5

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    N = 140
    target_sum = N / 2.0
    dx = 2.0 / N
    
    # Phase 0: Seeds
    start = time.time()
    seeds = []
    # Initial if exists
    if 'initial_h_values' in globals():
        h = np.interp(np.linspace(0, 2, N),
                      np.linspace(0, 2, len(initial_h_values)),
                      initial_h_values)
        h = h * (N/2) / np.sum(h)  # Renormalize exactly
        seeds.append((h, "initial"))
    else:
        # Create a generic 0.5 seed
        h = np.ones(N) * 0.5
        seeds.append((h, "uniform"))
    
    # Two peak
    for a in [0.15, 0.20, 0.25, 0.30, 0.35]:
        h = np.ones(N)
        x_vals = np.linspace(0, 2, N)
        u = 0.9
        v = 0.1
        mask = (x_vals < a) | (x_vals >= 2 - a)
        h[mask] = u
        h[~mask] = v
        h = h * (N/2) / np.sum(h)
        seeds.append((h, f"two_peak_a{a}"))
    
    # Random binary
    for _ in range(2):
        indices = np.random.choice(N, N//2, replace=False)
        h = np.zeros(N)
        h[indices] = 1
        seeds.append((h, "binary"))
    
    # For each seed, compute c5
    initial_costs = []
    for h, name in seeds:
        c5 = c5_of(h)
        initial_costs.append((h, c5, name))
    
    # Find the best among seeds
    initial_costs.sort(key=lambda x: x[1])
    best_h = initial_costs[0][0]
    best_c5 = initial_costs[0][1]

    # Phase 1: SPD
    t_start = time.time()
    spd_max_iter = 5000
    best_h, best_c5 = loop_spd(best_h, target_sum, spd_max_iter, budget_s - (time.time() - start), t_start)
    
    # Phase 2: Anneal
    t_start = time.time()
    n_iter = 5000
    best_h, best_c5 = loop_anneal(best_h, target_sum, n_iter, budget_s - (time.time() - start), t_start)

    # Phase 3: Hill climb
    t_start = time.time()
    best_h, best_c5 = loop_hill_climb(best_h, target_sum, budget_s - (time.time() - start), t_start)

    # Final evaluation
    final_c5 = c5_of(best_h)
    return (best_h, final_c5, N)
