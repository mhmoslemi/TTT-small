import numpy as np
import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value scaled by dx
    max_overlap = np.max(correlation) * dx
    return max_overlap

def generate_sturmian_binary(n, alpha=None):
    """Generate a balanced binary sequence using irrational rotation"""
    if alpha is None:
        alpha = (np.sqrt(5) - 1) / 2
    h = np.zeros(n)
    count_ones = 0
    for i in range(n):
        t = (i * alpha) % 1
        if t < 0.5:
            h[i] = 1
            count_ones += 1
    # Adjust to ensure exactly n/2 ones
    if count_ones > n // 2:
        # Find the largest elements to turn to 0
        ones_indices = np.where(h == 1)[0]
        for i in range(count_ones - n // 2):
            idx = ones_indices[i]
            h[idx] = 0
    elif count_ones < n // 2:
        # Find the smallest elements to turn to 1
        zeros_indices = np.where(h == 0)[0]
        for i in range(n // 2 - count_ones):
            idx = zeros_indices[i]
            h[idx] = 1
    return h

def compute_overlap(h, dx):
    """Compute the correlation array for all shifts"""
    correlation = np.correlate(h, 1 - h, mode="full")
    return correlation * dx

def calculate_delta(h, i, j, s_list, overlap_s, n):
    """Calculate the change in overlap for swapping h[i] and h[j]"""
    delta = np.zeros(len(s_list))
    for k, s in enumerate(s_list):
        # i contributes to shift s if i + s < n, j contributes to shift s - j if s - j >= 0
        # Check if positions are involved (s is the shift from i)
        if i + s < n:
            delta[k] -= (1 - h[i + s])
        if i - s >= 0:
            delta[k] += h[i - s]
        if j + s < n:
            delta[k] += (1 - h[j + s])
        if j - s >= 0:
            delta[k] -= h[j - s]
    return delta

def local_swap_search(h, max_iter=100000, tabu_length=100):
    n = len(h)
    dx = 2.0 / n
    overlap_s = compute_overlap(h, dx)
    s_list = np.arange(-(n-1), n)
    max_overlap = np.max(overlap_s)
    best_h = h.copy()
    best_c5 = max_overlap
    tabu_list = set()
    iter_since_improvement = 0
    last_improvement_iter = 0
    
    for iteration in range(max_iter):
        # Determine the current worst shift
        max_idx = np.argmax(overlap_s)
        max_shift = s_list[max_idx]
        
        # Identify good positions for swap
        ones_positions = np.where(h == 1)[0]
        zeros_positions = np.where(h == 0)[0]
        
        if len(ones_positions) == 0 or len(zeros_positions) == 0:
            break
            
        # Choose a random 1 position and random 0 position
        i = np.random.choice(ones_positions)
        j = np.random.choice(zeros_positions) 
        
        # Skip if swap is in tabu list
        swap_key = tuple(sorted((i, j)))
        if swap_key in tabu_list:
            continue
            
        # Evaluate the impact of the change
        old_max = max_overlap
        delta = calculate_delta(h, i, j, s_list, overlap_s, n)
        new_overlaps = overlap_s + delta
        new_max = np.max(new_overlaps)
        
        # Accept if improvement
        if new_max < max_overlap:
            h[i], h[j] = h[j], h[i]  # Swap values
            overlap_s = new_overlaps
            max_overlap = new_max
            tabu_list.add(swap_key)
            iter_since_improvement = 0
            last_improvement_iter = iteration
            if max_overlap < best_c5:
                best_c5 = max_overlap
                best_h = h.copy()
        else:
            # Accept with small probability to escape local minima
            if np.random.rand() < 0.01:
                h[i], h[j] = h[j], h[i]  # swap values
                overlap_s = new_overlaps
                max_overlap = new_max
                tabu_list.add(swap_key)
            iter_since_improvement += 1
        
        # Manage tabu list
        if len(tabu_list) > tabu_length:
            tabu_list.pop()
            
        # Early stopping if no improvement for long
        if iter_since_improvement > 3000 and iteration - last_improvement_iter > 5000:
            break
    
    return best_h, best_c5

def upscale_vector(h, new_n):
    """Upscale a binary vector by duplicating entries"""
    if len(h) >= new_n:
        return h[:new_n]
    old_n = len(h)
    factor = new_n // old_n
    remainder = new_n % old_n
    # Repeat each element factor times
    a = np.repeat(h, factor)
    # Add some additional elements
    if remainder > 0:
        a = np.append(a, h[:remainder])
    # Ensure correct length
    if len(a) > new_n:
        a = a[:new_n]
    elif len(a) < new_n:
        # Fill up with some values from existing
        a = np.resize(a, new_n)
    # Ensure balance
    current_sum = np.sum(a)
    target_sum = new_n // 2
    if current_sum != target_sum:
        diff = int(target_sum - current_sum)
        if diff > 0:
            # Add ones
            indices = np.where(a == 0)[0]
            if len(indices) >= diff:
                a[indices[:diff]] = 1
        elif diff < 0:
            # Subtract ones
            indices = np.where(a == 1)[0]
            if len(indices) >= -diff:
                a[indices[: (-diff)]] = 0
    return a

def continuous_polish(h, max_iter=500):
    """Perform a small gradient-based refinement"""
    n = len(h)
    dx = 2.0 / n
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec)
    
    # Very small step size
    step_size = 1e-4
    
    for _ in range(max_iter):
        old_f = objective(h)
        # Compute gradient approximatively
        eps = 1e-6
        grad = np.zeros(n)
        for i in range(n):
            h_eps = h.copy()
            h_eps[i] += eps
            grad[i] = (objective(h_eps) - old_f) / eps
            
        # Update
        h -= step_size * grad
        # Project onto [0,1]
        h = np.clip(h, 0, 1)
        # Project onto balancing constraint ∫ h dx = 1
        current_sum = np.sum(h) * dx
        h = h * (1.0 / current_sum)
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial construction
    n_points = 120  # Start here, then upscale
    h = generate_sturmian_binary(n_points)
    
    # Local search
    h, c5_bound = local_swap_search(h, max_iter=20000, tabu_length=50)
    
    # Upscale / refine
    # we upscale to 240 and continue local search
    h = upscale_vector(h, 240)
    h, c5_bound = local_swap_search(h, max_iter=30000, tabu_length=75)
    
    # Final continuous polishing if time allows
    time_remaining = budget_s - (time.time() - start_time)
    if time_remaining > 10:
        h = continuous_polish(h, max_iter=300)
    
    # Final check
    c5_bound = evaluate_erdos_solution(h)
    
    return (h.tolist(), c5_bound, len(h))
