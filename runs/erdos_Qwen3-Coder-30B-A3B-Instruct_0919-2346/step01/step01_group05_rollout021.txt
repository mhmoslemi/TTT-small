import numpy as np

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value scaled by dx
    max_overlap = np.max(correlation) * dx
    return max_overlap

def project_onto_bounds_sum(v, target_sum, lower=0, upper=1):
    """Project vector v onto box constraints [lower,upper] and sum constraint"""
    # Binary search for the threshold lambda
    left = v.min() - upper
    right = v.max() - lower
    for _ in range(60):
        mid = (left + right) / 2
        projected = np.clip(v - mid, lower, upper)
        if projected.sum() > target_sum:
            left = mid
        else:
            right = mid
    return np.clip(v - (left + right) / 2, lower, upper)

def softmax_gradient(h, gamma, dx):
    """Compute gradient of softmax approximation to max overlap"""
    n = len(h)
    correlation = np.correlate(h, 1 - h, mode="full")
    f = correlation * dx
    max_f = np.max(f)
    # Compute weights with numerical stability
    w = np.exp(gamma * (f - max_f))
    w = w / np.sum(w)
    
    # Compute gradient using the weight matrix
    grad = np.zeros(n)
    for l in range(2*n-1):
        lag = l - (n-1)
        if lag >= 0:
            # shift is positive
            start_idx = 0
            end_idx = n - lag
        else:
            # shift is negative
            start_idx = -lag
            end_idx = n
            
        # Handle the gradient contribution for indices corresponding to l
        if lag >= 0:
            for i in range(start_idx, end_idx):
                idx = i + lag
                if idx < n:
                    grad[i] += w[l] * (1 - h[idx])
                idx = i - lag
                if idx >= 0 and idx < n:
                    grad[i] -= w[l] * h[idx]
        else:

                if idx >= 0 and idx < n:
                    grad[i] += w[l] * (1 - h[idx])
                idx = i - lag
                if idx < n:
                    grad[i] -= w[l] * h[idx]
    
    return grad * dx
