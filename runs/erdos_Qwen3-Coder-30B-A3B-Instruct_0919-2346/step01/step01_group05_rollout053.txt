def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value scaled by dx
    max_overlap = np.max(correlation) * dx
    return max_overlap

def softmax_weights(f_vals, gamma):
    """Compute softmax weights for γ-regularized maximum of f_vals"""
    if not f_vals:
        return np.array([])
    f_vals = np.array(f_vals)
    f_max = np.max(f_vals)
    exp_vals = np.exp(gamma * (f_vals - f_max))
    return exp_vals / np.sum(exp_vals)

def correlation_shifts(h, n):
    """Precompute all shifts of h and compute weights efficiently"""
    dx = 2.0 / n
    shifts = []
    all_f_vals = []
    # Consider each shift s from -n+1 to n-1
    for s in range(1-n, n):  # shift indices
        s_terms = []
        i_start = max(0, -s)
        i_stop = min(n, n-s)
        if i_start < i_stop:
            for i in range(i_start, i_stop):
                prod = h[i] * (1 - h[i+s])
                s_terms.append(prod)
            val = dx * sum(s_terms)
            all_f_vals.append(val)
        else:
            all_f_vals.append(0)
    return all_f_vals

def compute_grad_f(h, n):
    """Compute ∇f_s for all shifts s, weighted by softmax weights"""
    dx = 2.0 / n
    # We need ∇f_s(i) = dx * [partial_i (h_i - h_i h_{i+s})]
    # = dx * [1 - h_{i+s} if there is h_{i+s} else 0] - dx * [h_{i-s} if exists]
    # ∇f_s(i) = dx * (g_i^{(s)})
    all_grads = []
    for s in range(1-n, n):
        grad = np.zeros(n)
        i_start = max(0, -s)
        i_stop = min(n, n-s)
        if i_start < i_stop:
            for i in range(i_start, i_stop):
                grad[i] += 1
                if i+s >= 0 and i+s < n:
                    grad[i] -= h[i+s]
                if i-s >= 0 and i-s < n:
                    grad[i] -= h[i-s]
            all_grads.append(dx * grad)
        else:
            all_grads.append(np.zeros(n))
    return all_grads

def project_to_simplex(v, target_sum=0.5):
    """
    Project vector v onto the simplex: sum(v) = target_sum, v_i in [0,1]
    Using binary search to find lambda such that sum clip(v_i - lambda, 0, 1) = target_sum.
    """
    n = len(v)
    v = np.array(v)
    lower, upper = np.min(v), np.max(v)  # bounds for lambda
    eps = 1e-10
    lambda_tol = 1e-8

    def get_sum(lam):
        return np.sum(np.clip(v - lam, 0, 1))

    # Binary search
    for _ in range(100):
        lam = (lower + upper) * 0.5
        s = get_sum(lam)
        if abs(s - target_sum) < lambda_tol:
            break
        if s > target_sum:
            lower = lam
        else:
            upper = lam
    return np.clip(v - lam, 0, 1)

def upsample_h(h0, n_target):
    """
    Upsample h0 (length 45) to length n_target using linear interpolation.
    """
    if len(h0) == n_target:
        return np.array(h0)
    h0 = np.array(h0)
    x0 = np.linspace(0, 2, len(h0))
    x_new = np.linspace(0, 2, n_target)
    h_new = np.interp(x_new, x0, h0)
    return h_new
