import numpy as np
import time
from scipy.optimize import differential_evolution, minimize

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral via full convolution of h with (1-h)
    overlaps = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum of the full correlation vector
    # scaled by dx
    c5 = np.max(overlaps) * dx
    return c5

def project_to_feasible(h):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1"""
    dx = 2.0 / len(h)
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / h_sum) * (2.0 / len(h))
    return h

def discrete_gradient(h, eps=1e-5):
    """Compute finite difference gradient of c5 with respect to h"""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    c0 = c5_of(h)
    
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus)
        c_plus = c5_of(h_plus)
        grad[i] = (c_plus - c0) / eps
    
    return grad

    # Optimize using a combination of differential evolution and local optimization
    best_func = float('inf')

        h_projected = project_to_feasible(h)

    return (best_h, best_func, N)
