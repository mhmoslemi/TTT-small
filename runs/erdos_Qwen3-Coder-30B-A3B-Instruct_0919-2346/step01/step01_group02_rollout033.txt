import numpy as np
import time

def c5_of(h):
    dx = 2.0 / len(h)
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def worst_shift(h):
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    return idx - (len(h) - 1)

def subgradient(h, s):
    N = len(h)
    dx = 2.0 / N
    m = abs(s)
    g = np.full(N, dx)
    if m < N:
        g[:N - m] -= dx * h[m:]
        g[m:] -= dx * h[:N - m]
    return g

def project(v, target):
    lo = -np.max(v)
    hi = 1.0 - np.min(v)
    for _ in range(50):
        lam = (lo + hi) / 2.0
        w = np.clip(v + lam, 0.0, 1.0)
        if w.sum() < target:
            lo = lam
        else:
            hi = lam
    w = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        free = (w > 0) & (w < 1)
        if free.any():
            w[free] += diff / free.sum()
            w = np.clip(w, 0, 1)
    return w

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    N = 140
    dx = 2.0 / N
    max_iter = 10000
    target_sum = N / 2.0
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    # Phase 0: Seeding
    seeds = []
    # Seed 1: interpolated from initial_h_values if available
    if 'initial_h_values' in globals():
        h0_orig = np.interp(np.linspace(0, 2, N),
                            np.linspace(0, 2, len(initial_h_values)),
                            initial_h_values)
        h0 = h0_orig * (target_sum / np.sum(h0_orig))
        h0 = project(h0, target_sum)
        seeds.append(h0)
    else:
        h0 = np.ones(N) * 0.5
        h0 = project(h0, target_sum)
        seeds.append(h0)
    # Seed 2: Two peak
    h1 = np.ones(N)
    a = 0.3
    u = 0.9
    v = 0.1
    x_vals = np.linspace(0, 2, N)
    mask = (x_vals < a) | (x_vals >= 2 - a)
    h1[mask] = u
    h1[~mask] = v
    h1 = h1 * (target_sum / np.sum(h1))
    h1 = project(h1, target_sum)
    seeds.append(h1)
    # Try different a parameters
    for a in np.linspace(0.12, 0.38, 5):
        h_temp = np.ones(N)
        u = 0.9
        v = 0.1
        x_vals = np.linspace(0, 2, N)
        mask = (x_vals < a) | (x_vals >= 2 - a)
        h_temp[mask] = u
        h_temp[~mask] = v
        h_temp = h_temp * (target_sum / np.sum(h_temp))
        h_temp = project(h_temp, target_sum)
        seeds.append(h_temp)
    # Seed 3: Random binary
    for _ in range(3):
        h_rand = np.random.rand(N)
        h_rand = (h_rand > 0.5).astype(float)
        h_rand = h_rand * (target_sum / np.sum(h_rand))
        h_rand = project(h_rand, target_sum)
        seeds.append(h_rand)
    # Seed 4: Constant
    h_const = np.ones(N) * 0.5
    h_const = project(h_const, target_sum)
    seeds.append(h_const)
    # Evaluate seeds
    for h_seed in seeds:
        c5 = c5_of(h_seed)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_seed.copy()
    # Phase 1: Sub-gradient projected descent
    time_left = budget_s * 0.55
    t_start = time.time()
    h = best_h.copy()
    c_current = best_c5
    eta_start = 30.0 * dx
    gamma = 0.9995
    iteration = 0
    while time.time() - start_time < budget_s * 0.70 and iteration < max_iter:
        s_star = worst_shift(h)
        g = subgradient(h, s_star)
        eta = eta_start * (gamma ** iteration)
        # Limit step size
        eta = min(eta, 50 * dx) 
        v = h - eta * g
        h_new = project(v, target_sum)
        c_new = c5_of(h_new)
        if c_new < c_current:
            h = h_new
            c_current = c_new
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h.copy()
        iteration += 1
        if iteration % 30 == 0:
            # Add tiny jitter to escape plateaux
            h_jitter = h + np.random.normal(0, 0.02, N)
            h_jitter = project(h_jitter, target_sum)
            c_jitter = c5_of(h_jitter)
            if c_jitter < c_current:
                h = h_jitter
                c_current = c_jitter
                if c_jitter < best_c5:
                    best_c5 = c_jitter
                    best_h = h.copy()
        if time.time() - t_start > time_left:
            break
    # Phase 2: Annealing
    time_left = budget_s * 0.30
    t_start = time.time()
    T_start = 0.05
    T_min = 1e-6
    h_current = best_h.copy()
    c_current = best_c5
    T = T_start
    iteration = 0
    max_step = 0.2
    while time.time() - start_time < budget_s * 0.90 and iteration < max_iter:
        # Temperature decay
        elapsed = time.time() - t_start
        frac = elapsed / time_left
        T = T_start * (T_min / T_start) ** frac
        # Perturb pair
        i, j = np.random.choice(N, size=2, replace=False)
        delta_low = max(-h_current[i], h_current[j] - 1)
        delta_high = min(1 - h_current[i], h_current[j])
        if delta_low >= delta_high:
            iteration += 1
            continue
        step_size = np.random.uniform(max(-max_step, delta_low),
                                      min(max_step, delta_high))
        h_new = h_current.copy()
        h_new[i] += step_size
        h_new[j] -= step_size
        c_new = c5_of(h_new)
        if c_new <= c_current or np.random.rand() < np.exp(-(c_new - c_current) / T):
            h_current = h_new
            c_current = c_new
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_current.copy()
        iteration += 1
        if time.time() - t_start > time_left:
            break
    # Phase 3: Hill-climb
    time_left = budget_s * 0.10
    t_start = time.time()
    while time.time() - start_time < budget_s - 0.1 and time.time() - t_start < time_left:
        # Pair moves with step size 0.02
        i, j = np.random.choice(N, size=2, replace=False)
        delta_low = max(-best_h[i], best_h[j] - 1)
        delta_high = min(1 - best_h[i], best_h[j])
        if delta_low >= delta_high:
            continue
        step_size = np.random.uniform(max(-0.02, delta_low),
                                      min(0.02, delta_high))
        h_new = best_h.copy()
        h_new[i] += step_size
        h_new[j] -= step_size
        c_new = c5_of(h_new)
        if c_new < best_c5:
            best_h = h_new
            best_c5 = c_new
    # Final evaluation
    final_c5 = c5_of(best_h)
    return (best_h, final_c5, N)
