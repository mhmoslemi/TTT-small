import numpy as np
import time
from scipy.optimize import minimize

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    overlaps = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(overlaps) * dx
    return c5

def evaluate_constraint(h, n_points):
    """Check if integral constraint is satisfied"""
    dx = 2.0 / n_points
    integral = np.sum(h) * dx
    return abs(integral - 1.0)

def project_onto_constraints(h, n_points):
    """Project h onto feasible region: 0 <= h[i] <= 1 and integral = 1"""
    # Ensure bounds
    h = np.clip(h, 0, 1)
    # Scale to correct integral
    dx = 2.0 / n_points
    integral = np.sum(h) * dx
    if integral > 0:
        h = h * (1.0 / integral)
    return h

def random_perturbation(h, max_delta=0.1):
    """Create perturbed version of h"""
    h_new = h.copy()
    idx = np.random.randint(0, len(h))
    delta = np.random.uniform(-max_delta, max_delta)
    h_new[idx] += delta
    return project_onto_constraints(h_new, len(h))

def solve_subproblem(h, n_points, max_iter=100):
    """Solve subproblem using gradient-based optimization to improve C5"""
    def objective(h_flat):
        h = h_flat.reshape(-1)
        return c5_of(h)
    
    def constraint_integral(h_flat):
        h = h_flat.reshape(-1)
        dx = 2.0 / n_points
        return np.sum(h) * dx - 1.0
    
    def constraint_bounds(h_flat):
        h = h_flat.reshape(-1)
        return np.concatenate([h, 1 - h])  # h[i] >= 0, h[i] <= 1
    
    # Provide a better initial guess to the optimizer
    x0 = h.copy()
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization with bounds
    try:
        result = minimize(objective, x0, method='L-BFGS-B', 
                         bounds=bounds, options={'maxiter': max_iter},
                         constraints={'type': 'eq', 'fun': constraint_integral})
        if result.success:
            return result.x.reshape(-1)
    except:
        pass
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start = time.time()
    n_points = 140
    dx = 2.0 / n_points
    
    # Try multiple initial constructions
    initial_constructions = []
    
    # 1. Uniform distribution
    h_uniform = np.ones(n_points) * 0.5
    h_uniform = project_onto_constraints(h_uniform, n_points)
    initial_constructions.append(('uniform', h_uniform))
    
    # 2. Initial construction from provided variable
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        if len(h0) != n_points:
            x0 = np.linspace(0, 2, len(h0))
            x_new = np.linspace(0, 2, n_points)
            h0 = np.interp(x_new, x0, h0)
        h_initial = project_onto_constraints(h0, n_points)
        initial_constructions.append(('initial', h_initial))
    
    # 3. Triangular construction
    x = np.linspace(0, 2, n_points)
    triangle = 1 - np.abs(x - 1)  # Peaks at x=1
    h_triangle = triangle / np.sum(triangle) * (n_points / 2)
    h_triangle = project_onto_constraints(h_triangle, n_points)
    initial_constructions.append(('triangle', h_triangle))
    
    # 4. Step function with lower and upper peak
    h_step = np.ones(n_points) * 0.5
    h_step[20:60] = 0.9
    h_step[80:120] = 0.1
    h_step = project_onto_constraints(h_step, n_points)
    initial_constructions.append(('step', h_step))
    
    # 5. Random construction constrained to integral requirement
    h_random = np.random.rand(n_points) * 0.8 + 0.1
    h_random = project_onto_constraints(h_random, n_points)
    initial_constructions.append(('random', h_random))
    
    # Initialize best solution
    best_c5 = float('inf')
    best_h = None

    # Iterate through each construction
    for name, h_init in initial_constructions:
        if time.time() - start >= budget_s * 0.95:
            break
            
        # Apply subproblem solver to refine
        h_refined = solve_subproblem(h_init, n_points, max_iter=50)
        c5_refined = c5_of(h_refined)
        
        if c5_refined < best_c5:
            best_h = h_refined
            best_c5 = c5_refined

    # Local search refinement
    if best_h is not None and time.time() - start < budget_s * 0.95:
        # Neighborhood search with perturbations
        current = best_h.copy()
        current_c5 = c5_of(current)
        
        for _ in range(500):
            if time.time() - start > budget_s - 1:
                break
                
            # Small perturbation
            h_proposed = random_perturbation(current, max_delta=0.05)
            c5_proposed = c5_of(h_proposed)
            
            # Accept if improvement or with some probability
            if c5_proposed < current_c5:
                current = h_proposed
                current_c5 = c5_proposed
            elif np.random.rand() < 0.05:  # occasional uphill step
                current = h_proposed
                current_c5 = c5_proposed
                
        if current_c5 < best_c5:
            best_h = current
            best_c5 = current_c5

    return (best_h, best_c5, n_points)
