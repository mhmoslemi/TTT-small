def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using full convolution
    overlap = np.correlate(h, 1 - h, mode='full')
    # The maximum value scaled by dx
    c5 = np.max(overlap) * dx
    return c5

def propose_move(h, max_step):
    """Propose a move that maintains sum constraint exactly"""
    h_new = h.copy()
    N = len(h)
    while True:
        i, j = random.sample(range(N), 2)
        # Compute feasible range for delta
        delta_low = max(-h_new[i], h_new[j] - 1.0)
        delta_high = min(1.0 - h_new[i], h_new[j])
        if delta_low >= delta_high:
            continue  # No valid move, try again
        break
    
    # Sample delta from feasible interval
    step = min(max_step, delta_high - delta_low)
    if step <= 0:
        delta = 0.0
    else:
        delta = random.uniform(max(-step, delta_low), min(step, delta_high))
        
    h_new[i] += delta
    h_new[j] -= delta
    return h_new

def smooth_move(h, max_step):
    """Make a move with a single position change"""
    h_new = h.copy()
    N = len(h)
    i = random.randrange(N)
    
    # Compute feasible range for delta
    delta_low = max(-h_new[i], 0.0)
    delta_high = min(1.0 - h_new[i], 1.0)
    if delta_low >= delta_high:
        return h_new
    
    # Sample delta from feasible interval
    step = min(max_step, delta_high - delta_low)
    if step <= 0:
        delta = 0.0
    else:
        delta = random.uniform(max(-step, delta_low), min(step, delta_high))
        
    h_new[i] += delta
    
    # Adjust other points to preserve sum
    adjustment = -delta / (N - 1)
    for j in range(N):
        if j != i:
            h_new[j] += adjustment
            
    # Clamp values to [0,1]
    h_new = np.clip(h_new, 0, 1)
    return h_new

def hill_climb(initial_h, time_budget, step_size=0.05):
    """Hill climbing for refinement"""
    start = time.time()
    cur = initial_h.copy()
    c_cur = c5_of(cur)
    
    while time.time() - start < time_budget:
        h_prop = smooth_move(cur, step_size)
        c_prop = c5_of(h_prop)
        
        if c_prop < c_cur:
            cur = h_prop
            c_cur = c_prop
        else:
            # Only occasionally accept worse moves to escape local minima
            # but very rarely
            if random.random() < 0.01:
                cur = h_prop
                c_cur = c_prop
                
    return cur, c_cur
