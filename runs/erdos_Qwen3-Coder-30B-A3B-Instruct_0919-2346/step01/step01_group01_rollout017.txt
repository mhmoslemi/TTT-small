import numpy as np
import time
import random
from scipy.optimize import differential_evolution

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using full convolution
    overlap = np.correlate(h, 1 - h, mode='full')
    # The maximum value scaled by dx
    c5 = np.max(overlap) * dx
    return c5

def project_to_feasible(h):
    """Project h to satisfy the constraint sum(h) * dx = 1"""
    dx = 2.0 / len(h)
    current_sum = np.sum(h)
    target_sum = 1.0 / dx
    # Scale to meet sum constraint
    h = h * target_sum / current_sum
    # Clamp to [0,1]
    h = np.clip(h, 0, 1)
    return h

def initial_h():
    """Generate improved initial seed"""
    # Use a combination of two approaches
    N = 140
    h = np.zeros(N)
    
    # Sigmoid-like shape
    x = np.linspace(0, 2, N)
    # Create a smooth transition that meets the sum constraint
    h = 0.5 + 0.3 * np.tanh(5 * (x - 1))
    h = np.clip(h, 0, 1)
    
    # Adjust to make the sum exactly 1/dx
    dx = 2.0 / N
    h = h * (1.0/dx) / np.sum(h)
    
    return h

def optimize_with_differential_evolution(h0, budget_s):
    """Optimize using differential evolution which works well for this kind of continuous problem"""
    start = time.time()
    
    # Objective function (we want to minimize)
    def objective(h):
        # Ensure constraint is satisfied
        h_projected = project_to_feasible(h)
        return c5_of(h_projected)
    
    # Bounds for each variable: [0,1]
    bounds = [(0, 1) for _ in range(len(h0))]
    
    # Use differential evolution - it's good at avoiding local minima
    try:
        result = differential_evolution(
            objective,
            bounds,
            maxiter=int(2000 * budget_s / 600),
            popsize=15,
            mutation=(0.5, 1),
            recombination=0.7,
            atol=1e-6,
            rtol=1e-6,
            seed=42
        )
        
        best_h = project_to_feasible(result.x)
        best_c5 = c5_of(best_h)
        
        return best_h, best_c5
    except:
        # Fallback to basic hill climbing
        return h0, c5_of(h0)

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    # Start with a better initial construction
    best_h = initial_h()
    best_c5 = c5_of(best_h)
    
    # Allow some time for differential evolution
    start = time.time()
    remaining_budget = budget_s - (time.time() - start)
    
    # Run optimization using differential evolution
    h_opt, c5_opt = optimize_with_differential_evolution(best_h, remaining_budget)
    
    if c5_opt < best_c5:
        best_h = h_opt
        best_c5 = c5_opt
    
    # Final refinement - very small local search
    if time.time() - start < budget_s - 5:
        # Simple local search
        dx = 2.0 / len(best_h)
        step = 0.01
        for _ in range(5000):
            if time.time() - start > budget_s - 5:
                break
            h_prop = best_h.copy()
            # Perturb two random points
            i, j = random.sample(range(len(h_prop)), 2)
            delta = random.uniform(-step, step)
            h_prop[i] += delta
            h_prop[j] -= delta
            # Clamp to feasible region
            h_prop = np.clip(h_prop, 0, 1)
            # Scale to meet constraint
            h_prop = project_to_feasible(h_prop)
            c_prop = c5_of(h_prop)
            if c_prop < best_c5:
                best_h = h_prop
                best_c5 = c_prop
    
    return (best_h, best_c5, len(best_h))
