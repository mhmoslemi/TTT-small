import numpy as np
import time
from scipy.interpolate import interp1d

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value scaled by dx
    max_overlap = np.max(correlation) * dx
    return max_overlap

def upsample_initial(initial_h_values, target_n):
    """Upsample a given initial sequence to target_n by linear interpolation"""
    n_orig = len(initial_h_values)
    x_orig = np.linspace(0, 2, n_orig)
    x_target = np.linspace(0, 2, target_n)
    f = interp1d(x_orig, initial_h_values, kind='linear', fill_value='extrapolate')
    h_upsampled = f(x_target)
    return np.clip(h_upsampled, 0, 1)

def project_to_simplex(v, target_sum=0.5):
    """
    Project vector v onto the simplex {x : sum x = target_sum, x_i in [0,1]}
    Uses water-filling approach with binary search on lambda.
    """
    n = len(v)
    # Clip to [0, 1]
    v = np.clip(v, 0, 1)
    current_sum = np.sum(v)
    
    if abs(current_sum - target_sum) < 1e-10:
        return v
    
    # Binary search for the correct lambda
    def phi(lam):
        return np.sum(np.clip(v - lam, 0, 1)) - target_sum
    
    # Find interval containing the root
    lower, upper = 0.0, np.max(v) + 1e-5
    while phi(lower) > 0:
        lower -= 1.0
    while phi(upper) < 0:
        upper += 1.0

    # Binary search
    for _ in range(40):
        lam = (lower + upper) / 2
        if phi(lam) > 0:
            lower = lam
        else:
            upper = lam

    lam = (lower + upper) / 2
    projected = np.clip(v - lam, 0, 1)
    return projected

def softmax_weights(fs, gamma):
    """Compute softmax weights for current gradient computation"""
    if gamma == 0:
        return np.ones_like(fs) / len(fs)
    ef = np.exp(gamma * fs)
    return ef / np.sum(ef)

def compute_gradient(h, gamma, dx):
    """Compute the gradient of the smooth maximum objective"""
    n = len(h)
    grad = np.zeros_like(h)
    fs = []
    for s in range(-n+1, n):
        # Compute f_s(h)
        f_s = 0
        for i in range(n):
            if 0 <= i+s < n:
                f_s += h[i] * (1 - h[i+s]) * dx
        fs.append(f_s)

    # Compute weights
    weights = softmax_weights(fs, gamma)

    # Compute gradient scalars
    for s in range(-n+1, n):
        w_s = weights[s+n-1]  # index of s
        g_s = np.zeros_like(h)
        for i in range(n):
            if 0 <= i+s < n:
                g_s[i] += 1 * dx
            if 0 <= i-s < n:
                g_s[i] -= 1 * dx
        grad += w_s * g_s

    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters for coarse and fine phases
    n_coarse = 120
    n_fine = 260
    
    # Parameters for gradient descent and projections
    gamma_0 = 1.0
    gamma_max = 80.0
    alpha_0 = 0.5
    max_iter_coarse = 3000
    max_iter_fine = 6000

    # Phase 1: Coarse optimization (n=120)
    best_h_coarse = None
    best_c5_coarse = float('inf')
    
    try:
        h_upsampled = upsample_initial(initial_h_values, n_coarse)
    except NameError:
        # Fall back to Sturmian-like sequence
        alpha = (np.sqrt(5) - 1) / 2
        h_upsampled = np.array([1 if (i * alpha) % 1 < 0.5 else 0 for i in range(n_coarse)])
        h_upsampled = np.clip(h_upsampled, 0, 1)
    
    # Normalize to correct sum
    sum_h = np.sum(h_upsampled)
    h_upsampled = h_upsampled * (n_coarse / 2.0) / sum_h
    h = project_to_simplex(h_upsampled, n_coarse / 2.0)
    
    t = 0
    patience = 300
    patience_counter = 0
    last_improvement = 0
    restarts = 0
    
    while t < max_iter_coarse and (time.time() - start_time) < budget_s * 0.95:
        if (time.time() - start_time) > budget_s * 0.9:
            break

        # Compute current true objective
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5_coarse:
            best_c5_coarse = current_c5
            best_h_coarse = h.copy()

        # Update gamma and alpha
        gamma = gamma_0 + (gamma_max - gamma_0) * t / max_iter_coarse
        alpha = alpha_0 / np.sqrt(t + 1.0)

        # Compute gradient
        grad = compute_gradient(h, gamma, 2.0 / n_coarse)
        h_new = h - alpha * grad

        # Project back onto feasible set
        h = project_to_simplex(h_new, n_coarse / 2.0)

        # Update patience
        if current_c5 < best_c5_coarse:
            last_improvement = t
            patience_counter = 0
        else:
            patience_counter += 1

        # Restart if no improvement for certain steps
        if patience_counter > patience and restarts < 5:
            np.random.seed(int(time.time() * 1000) % (2**32))
            h += 0.02 * np.random.normal(0, 1, n_coarse)
            h = project_to_simplex(h, n_coarse / 2.0)
            restarts += 1
            patience_counter = 0

        t += 1

    # Phase 2: Fine optimization (n=260)
    best_h_fine = None
    best_c5_fine = float('inf')
    try:
        h_upsampled = upsample_initial(initial_h_values, n_fine)
    except NameError:
        # Use coarse solution
        h_upsampled = upsample_initial(best_h_coarse, n_fine)
    
    # Normalize to correct sum
    sum_h = np.sum(h_upsampled)
    h_upsampled = h_upsampled * (n_fine / 2.0) / sum_h
    h = project_to_simplex(h_upsampled, n_fine / 2.0)
    
    last_improvement = 0
    patience_counter = 0
    restarts = 0
    
    max_iter = max_iter_fine
    t = 0
    while t < max_iter and (time.time() - start_time) < budget_s * 0.95:
        if (time.time() - start_time) > budget_s * 0.9:
            break

        # Compute current true objective
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5_fine:
            best_c5_fine = current_c5
            best_h_fine = h.copy()

        # Update gamma and alpha
        gamma = gamma_0 + (gamma_max - gamma_0) * t / max_iter
        alpha = alpha_0 / np.sqrt(t + 1.0)

        # Compute gradient
        grad = compute_gradient(h, gamma, 2.0 / n_fine)
        h_new = h - alpha * grad

        # Project back onto feasible set
        h = project_to_simplex(h_new, n_fine / 2.0)

        # Update patience and restart condition
        if current_c5 < best_c5_fine:
            last_improvement = t
            patience_counter = 0
        else:
            patience_counter += 1

        # Restart if no improvement for certain steps
        if patience_counter > patience and restarts < 5:
            np.random.seed(int(time.time() * 1000) % (2**32))
            h += 0.02 * np.random.normal(0, 1, n_fine)
            h = project_to_simplex(h, n_fine / 2.0)
            restarts += 1
            patience_counter = 0

        t += 1

    # If the fine optimization didn't yield an improvement, return the coarse solution
    if best_h_fine is not None and best_c5_fine < best_c5_coarse:
        return (best_h_fine.tolist(), best_c5_fine, n_fine)
    if best_h_coarse is not None:
        return (best_h_coarse.tolist(), best_c5_coarse, n_coarse)
    else:
        return (h.tolist(), evaluate_erdos_solution(h), n_coarse)
