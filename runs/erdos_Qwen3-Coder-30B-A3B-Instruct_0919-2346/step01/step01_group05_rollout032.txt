import numpy as np

def upsample_initial(h0, n):
    """Upsample initial_h_values or create a balanced Sturmian word."""
    if h0 is None:
        # Create a balanced Sturmian word
        alpha = (np.sqrt(5) - 1) / 2
        h0 = np.array([1 if (i * alpha) % 1 < 0.5 else 0 for i in range(len(h0))])
    # Linear interpolation to n points
    indices_old = np.linspace(0, 1, len(h0))
    indices_new = np.linspace(0, 1, n)
    h_upsampled = np.interp(indices_new, indices_old, h0)
    return np.clip(h_upsampled, 0, 1)

def project_to_simplex(v, target_sum):
    """Project vector v onto the simplex with given target sum."""
    # Sort v in descending order
    v_sorted = np.sort(v)[::-1]
    # Compute cumulative sums
    cumsum = np.cumsum(v_sorted)
    # Find the index where the projection happens
    i_star = np.argmax(v_sorted - (target_sum - cumsum + v_sorted) / np.arange(1, len(v_sorted) + 1) >= 0)
    # Compute the threshold
    lambda_star = (cumsum[i_star] - target_sum) / (i_star + 1)
    # Project
    projection = np.clip(v - lambda_star, 0, 1)
    return projection

def compute_f_s(h, s, dx):
    """Compute the f_s scalar for a given shift s."""
    if s >= 0:
        # s >= 0 implies indices in [0, n-s-1] and [s, n-1]
        indices_i = np.arange(0, len(h) - s)
        indices_j = np.arange(s, len(h))
    else:
        # s < 0 implies indices in [0, n+s-1] and [s, n-1]
        indices_i = np.arange(-s, len(h))
        indices_j = np.arange(0, len(h) + s)
    if len(indices_i) == 0: 
        return 0.0
    return dx * np.sum(h[indices_i] * (1 - h[indices_j]))

def softmax_weights(f_vals, gamma):
    """Compute the softmax weights for a list of scalar values."""
    if gamma == np.inf:
        # Just take the one that maximizes
        max_i = np.argmax(f_vals)
        w = np.zeros_like(f_vals)
        w[max_i] = 1.0
    else:
        # Use the standard softmax approximation
        numerators = np.exp(gamma * np.array(f_vals))
        w = numerators / np.sum(numerators)
    return w

def compute_grad_f_s(h, s, dx):
    """Compute the gradient of f_s with respect to h."""
    g = np.zeros_like(h)
    if s >= 0:
        # s >= 0 implies indices in [0, n-s-1] and [s, n-1]
        indices_i = np.arange(0, len(h) - s)
        indices_j = np.arange(s, len(h))
        g[indices_i] += 1
        g[indices_j] -= 1
    else:
        # s < 0 implies indices in [0, n+s-1] and [s, n-1]
        indices_i = np.arange(-s, len(h))
        indices_j = np.arange(0, len(h) + s)
        g[indices_i] += 1
        g[indices_j] -= 1
    return dx * g

def compute_full_gradient(h, gamma, dx):
    """Compute full gradient (smooth max) using current gamma."""
    # Compute all individual f_s values
    shift_range = range(-(len(h) - 1), len(h))
    f_vals = np.array([compute_f_s(h, s, dx) for s in shift_range])
    # Get weights for the current gamma
    weights = softmax_weights(f_vals, gamma)
    # Compute gradient as weighted sum with each component's gradient
    grad = np.zeros_like(h)
    for i, w in enumerate(weights):
        s = shift_range[i]
        grad += w * compute_grad_f_s(h, s, dx)
    return grad

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value scaled by dx
    max_overlap = np.max(correlation) * dx
    return max_overlap

def simulated_annealing_fallback(h0, n_points, dx, remaining_budget):
    """Fallback simulated annealing to improve a binary solution."""
    # Convert h0 to binary with exactly n_points/2 ones
    # Simple strategy: use thresholds
    sorted_vals = np.sort(h0)[::-1]
    threshold = sorted_vals[n_points // 2]
    h_bin = (h0 >= threshold).astype(float)
    # Flip 0s and 1s to make exactly n/2 ones
    current_sum = np.sum(h_bin)
    if current_sum > n_points / 2:
        flip_indices = np.where(h_bin == 1)[0][:int(current_sum - n_points / 2)]
        h_bin[flip_indices] = 0
    elif current_sum < n_points / 2:
        flip_indices = np.where(h_bin == 0)[0][:int(n_points / 2 - current_sum)]
        h_bin[flip_indices] = 1
        
    # Try annealing with swaps
    best_h = h_bin.copy()
    best_obj = evaluate_erdos_solution(best_h)
    T = 0.5
    cooling_rate = 0.9995
    step_max = remaining_budget * 1000
    iteration = 0
    while T > 1e-8 and iteration < step_max:
        idx1 = np.random.randint(0, n_points)
        idx2 = np.random.randint(0, n_points)
        while h_bin[idx2] == h_bin[idx1]:
            idx2 = np.random.randint(0, n_points)
        h_new = h_bin.copy()
        h_new[idx1], h_new[idx2] = h_new[idx2], h_new[idx1]
        new_obj = evaluate_erdos_solution(h_new)
        # Metropolis acceptance
        if new_obj < best_obj or np.random.rand() < np.exp(-(new_obj - best_obj) / T):
            h_bin = h_new
            if new_obj < best_obj:
                best_h = h_new
                best_obj = new_obj
        T = T * cooling_rate
        iteration += 1
    return best_h, best_obj
