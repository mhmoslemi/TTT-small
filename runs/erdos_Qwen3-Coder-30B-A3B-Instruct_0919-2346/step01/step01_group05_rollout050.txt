import numpy as np

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value scaled by dx
    max_overlap = np.max(correlation) * dx
    return max_overlap

def project_onto_bounds_sum(v, target_sum, lower=0, upper=1):
    """Project vector v onto constraint: lower <= v[i] <= upper, sum(v) = target_sum"""
    # Binary search for the value to clip at
    # f(lambda) = sum(clamp(v - lambda, lower, upper)) - target_sum
    left = v.min() - upper
    right = v.max() - lower
    
    for _ in range(60):
        mid = (left + right) / 2
        clipped = np.clip(v - mid, lower, upper)
        actual_sum = np.sum(clipped)
        if actual_sum > target_sum:
            left = mid
        else:
            right = mid
            
    return np.clip(v - (left + right) / 2, lower, upper)
