import numpy as np
import time

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value scaled by dx
    max_overlap = np.max(correlation) * dx
    return max_overlap

def generate_sturmian_binary(n):
    """Generate a balanced binary sequence using irrational rotation"""
    alpha = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
    h = np.zeros(n, dtype=int)
    for i in range(n):
        t = (i * alpha) % 1
        h[i] = 1 if t < 0.5 else 0

    # Adjust to make sum = n//2
    current_sum = np.sum(h)
    target_sum = n // 2
    
    if current_sum > target_sum:
        # Need to flip some 1s to 0s
        ones_indices = np.where(h == 1)[0]
        flip_indices = ones_indices[:current_sum - target_sum]
        h[flip_indices] = 0
    elif current_sum < target_sum:
        # Need to flip some 0s to 1s
        zeros_indices = np.where(h == 0)[0]
        flip_indices = zeros_indices[:target_sum - current_sum]
        h[flip_indices] = 1
    
    return h

def compute_shift_contributions(h):
    """Compute the contribution of each shift to overlap"""
    n = len(h)
    dx = 2.0 / n
    # Compute the full correlation
    correlation = np.correlate(h, 1 - h, mode="full")
    # Scale by dx
    return correlation * dx

def update_shift_contributions(g_s, i, j, h, n, s_start_idx):
    """Calculate the change in shift contributions when swapping positions i and j"""
    delta = np.zeros_like(g_s)
    # We'll compute a few key contributions per s:
    # The update to each g_s is due to changes in h_i*(1-h_{i+s}) and h_j*(1-h_{j+s})
    # Where s is shift index corresponding to s_start_idx + s (for s in range(2*n - 1))

    for s in range(2 * n - 1):
        shift_idx = s_start_idx + s
        # Actually, we only compute efficiently the changes in indices potentially affected
        # For shift s, positions i and j may be involved if i+s-mod-n < n or j+s-mod-n < n 
        if s < n:
            # s is positive shift (future)
            # check if i or j can contribute to this shift due to overlap
            i_pos = i
            j_pos = j
            i_s_pos = (i + s) % n
            j_s_pos = (j + s) % n

            old_i = h[i_pos] * (1 - h[i_s_pos])
            old_j = h[j_pos] * (1 - h[j_s_pos])
            
            new_i = (1 - h[i_pos]) * (1 - h[i_s_pos])
            new_j = h[j_pos] * h[j_s_pos]

            delta[s] = (new_i - old_i) + (new_j - old_j)
        else:
            # For negative s, we compute backwards
            neg_s = s - (2 * n - 1)
            i_s_pos = (i - neg_s) % n
            j_s_pos = (j - neg_s) % n
            
            old_i = h[i] * (1 - h[i_s_pos])
            old_j = h[j] * (1 - h[j_s_pos])
            
            new_i = (1 - h[i]) * (1 - h[i_s_pos])
            new_j = h[j] * h[j_s_pos]

            delta[s] = (new_i - old_i) + (new_j - old_j)
    
    return delta

def local_swap_search(h, n, max_time):
    """Locally improve a binary vector by swapping positions to reduce the maximum overlap"""
    start_time = time.time()
    g_s = compute_shift_contributions(h)
    s_max = np.argmax(g_s)
    current_max = np.max(g_s)
    
    # Mapping of shift index to relative shift
    # Full correlation has 2*n - 1 elements, centered at indices n-1
    # So shift inside [0,n) maps to index s_start_idx, i.e. n-1
    s_start_idx = n - 1

    tabu_list = set()
    tabu_length = 100
    iteration = 0
    no_improvement_iterations = 0
    max_no_improvement = 30000

    while time.time() - start_time < max_time:
        iteration += 1
        if no_improvement_iterations > max_no_improvement:
            # Perturb the solution
            idx = np.where(h == 1)[0]
            if len(idx) >= 2:
                # Pick two 1s and flip to 0
                idx1, idx2 = np.random.choice(idx, 2, replace=False)
                h[idx1] = 0
                h[idx2] = 0
                # Pick 2 new 0s and flip to 1
                idx0 = np.where(h == 0)[0]
                if len(idx0) >= 2:
                    idx3, idx4 = np.random.choice(idx0, 2, replace=False)
                    h[idx3] = 1
                    h[idx4] = 1
                
            # Reset tabu
            tabu_list.clear()
            # Recompute g_s
            g_s = compute_shift_contributions(h)
            s_max = np.argmax(g_s)
            current_max = np.max(g_s)
            no_improvement_iterations = 0

        # Find 1-positions and 0-positions
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        # Pick a 1- and a 0-position at random that can be swapped
        i = np.random.choice(ones) if len(ones) > 0 else 0
        j = np.random.choice(zeros) if len(zeros) > 0 else 1
        swap_key = tuple(sorted((i, j)))
        
        # Avoid tabu
        if swap_key in tabu_list:
            continue

        # Compute change in g_s when swapping
        old_g_max = current_max
        g_s_new = g_s.copy()
        delta_g = update_shift_contributions(g_s_new, i, j, h, n, s_start_idx)
        g_s_new += delta_g
        new_max = np.max(g_s_new)
        
        if new_max < current_max:
            h[i], h[j] = h[j], h[i]
            g_s = g_s_new
            current_max = new_max
            no_improvement_iterations = 0
            # Update tabu
            if len(tabu_list) >= tabu_length:
                tabu_list.pop()
            tabu_list.add(swap_key)
        else:
            no_improvement_iterations += 1
            # Accept with small probability to escape local minima
            if np.random.rand() < 0.001:
                h[i], h[j] = h[j], h[i]
                g_s = g_s_new 
                g_s_new += delta_g
                if np.max(g_s) > old_g_max:
                    # Undo swap if it increases max too much
                    pass
                else:
                    current_max = np.max(g_s_new)
                    # Update tabu
                    if len(tabu_list) >= tabu_length:
                        tabu_list.pop()
                    tabu_list.add(swap_key)
                # Re-evaluate max
            else:
                pass  # no change to h or g_s
    
    return h, current_max

def interleave_h(h_coarse):
    """Upscale a binary vector by duplicating elements"""
    n_coarse = len(h_coarse)
    h_fine = np.zeros(2 * n_coarse, dtype=int)
    for i in range(n_coarse):
        h_fine[2*i] = h_coarse[i]
        h_fine[2*i+1] = h_coarse[i]
    return h_fine

def project_to_sum_constraint(h, target_sum):
    """Project h to target sum with clipping"""
    current_sum = np.sum(h)
    dx = 2.0 / len(h)
    scale = target_sum / (current_sum * dx) if current_sum != 0 else 1.0
    h = h * scale
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

        dx = 2.0 / current_n
