import numpy as np
import time

def compute_c5_and_worst_shift(h):
    """Compute the C5 value and the corresponding worst shift index."""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr)
    s = max_idx - (len(h) - 1)
    c5 = corr[max_idx] * dx
    return c5, s

def compute_gradient(h, shift):
    """Compute the gradient for the correlation with given shift."""
    N = len(h)
    dx = 2.0 / N
    g = np.full(N, 1.0) * dx
    if shift >= 0:
        if shift < N:
            g[:N-shift] -= h[shift:] * dx
            g[shift:] -= h[:N-shift] * dx
    else:
        shift = -shift
        if shift < N:
            g[shift:] -= h[:N-shift] * dx
            g[:N-shift] -= h[shift:] * dx
    return g

def project_onto_bounds_and_sum(v, target):
    """Project vector v onto box [0,1] and hyperplane sum = target."""
    N = len(v)
    # Binary search for lambda
    lower, upper = -np.max(v), 1 - np.min(v)
    for _ in range(60):
        mid = (lower + upper) / 2.0
        w = np.clip(v + mid, 0, 1)
        s = np.sum(w)
        if s < target:
            lower = mid
        else:
            upper = mid
    w = np.clip(v + (lower + upper) / 2.0, 0, 1)
    # Adjust to make sum exactly target
    diff = target - np.sum(w)
    if abs(diff) > 1e-12:
        free_idx = (w > 0) & (w < 1)
        if np.any(free_idx):
            w[free_idx] += diff / np.sum(free_idx)
            w = np.clip(w, 0, 1)
        else:
            w += diff / N
            w = np.clip(w, 0, 1)
    return w

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    N = 140
    dx = 2.0 / N
    
    # Compute initial values
    if 'initial_h_values' in globals():
        h0 = np.interp(np.linspace(0, 2, N), 
                      np.linspace(0, 2, len(initial_h_values)), 
                      initial_h_values)
        h0 = h0 * (N/2) / np.sum(h0)
    else:
        h0 = np.ones(N) * 0.5
    
    # Two-peak construction  
    h1 = np.ones(N)
    x_vals = np.linspace(0, 2, N)
    a = 0.3
    u = 0.9
    v = 0.1
    mask = (x_vals < a) | (x_vals >= 2 - a)
    h1[mask] = u
    h1[~mask] = v
    h1 = h1 * (N/2) / np.sum(h1)
    
    # Uniform 
    h2 = np.ones(N) * 0.5
    
    # Evaluate multiple seeds 
    seeds = [h0, h1, h2]
    best_h = None
    best_c5 = float('inf')
    
    # Total iterations count 
    total_iter = 0
    
    start = time.time()
    
    # Use hybrid approach: gradient descent + random restarts
    while time.time() - start < budget_s * 0.9:
        for h_seed in seeds:
            h = h_seed.copy()
            current_c5, _ = compute_c5_and_worst_shift(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
            
            # Gradient descent loop
            iter_count = 0
            while time.time() - start < budget_s * 0.9 and iter_count < 500:
                c5, s = compute_c5_and_worst_shift(h)
                grad = compute_gradient(h, s)
                t_frac = (time.time() - start) / budget_s
                step = 20 * (1 - t_frac) + 0.1
                
                v = h - step * grad
                h = project_onto_bounds_and_sum(v, N/2)
                
                new_c5, _ = compute_c5_and_worst_shift(h)
                if new_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = new_c5
                
                iter_count += 1
                total_iter += 1
                
                # Add random jitter occasionally
                if iter_count % 100 == 0 and np.random.rand() < 0.2:
                    noise = np.random.uniform(-0.1, 0.1, size=N)
                    v = h + noise
                    h = project_onto_bounds_and_sum(v, N/2)
    
    # Final polishing
    while time.time() - start < budget_s * 0.97:
        if best_h is None:
            break
        # Local hill climbing
        h = best_h.copy()
        c5, _ = compute_c5_and_worst_shift(h)
        for _ in range(100):
            i, j = np.random.choice(N, size=2, replace=False)
            delta_low = max(-h[i], h[j] - 1)
            delta_high = min(1 - h[i], h[j])
            if delta_low >= delta_high:
                continue
            delta = np.random.uniform(-0.05, 0.05) 
            delta = np.clip(delta, delta_low, delta_high)
            h[i] += delta
            h[j] -= delta
            c5_new, _ = compute_c5_and_worst_shift(h)
            if c5_new < c5:
                c5 = c5_new
                best_h = h.copy()
            else:
                h[i] -= delta
                h[j] += delta
    # Final evaluation 
    final_c5, _ = compute_c5_and_worst_shift(best_h)
    return (best_h, final_c5, N)
