import numpy as np
import time
from scipy.interpolate import interp1d

def softmax_weights(f_vals, gamma):
    """Compute softmax weights from function values."""
    exp_vals = np.exp(gamma * f_vals)
    return exp_vals / np.sum(exp_vals)

def gradient_single_shift(h, s, dx):
    """Compute gradient of f_s for a given shift s."""
    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        # i+s is valid for i = 0 to n-1-s
        for i in range(n - s):
            g[i] += 1.0 - h[i + s]
        # i-s is valid for i = s to n-1
        for i in range(s, n):
            g[i] -= h[i - s]
    else:
        # i+s is valid for i = -s to n-1
        for i in range(-s, n):
            g[i] += 1.0 - h[i + s]
        # i-s is valid for i = 0 to n+s-1 but only when i-s >=0
        # so i >= s. Also, i-s < n, so i < n+s
        for i in range(s, n):
            g[i] -= h[i - s]
    return g * dx

def compute_all_gradients(h, dx):
    """Compute the full gradient by summing up weighted components."""
    n = len(h)
    grad = np.zeros(n)
    f_vals = []
    shifts = list(range(-(n - 1), n))
    for s in shifts:
        f_s = np.sum(h * (1 - np.roll(h, s) * (np.abs(s) <= n - 1)))
        f_vals.append(f_s)
    f_vals = np.array(f_vals)
    weights = softmax_weights(f_vals, 1.0)  # Initial weight for γ=1
    for w, s in zip(weights, shifts):
        grad += w * gradient_single_shift(h, s, dx)
    return grad

def project_to_simplex(v, target_sum):
    """Project vector v onto simplex {x : sum x = target_sum, 0 <= x <= 1}"""
    # Use bisection to find lambda such that \sum clip(v - λ, 0, 1) == target_sum
    # First clip
    clipped = np.clip(v, 0, 1)
    sum_low = np.sum(clipped)
    if abs(sum_low - target_sum) < 1e-10:
        return clipped
    # If sum is high, then we need to decrease v (λ > 0)
    # Find upper and lower bounds
    v_low = np.min(v)
    v_high = np.max(v)
    if v_low >= 0 and v_high <= 1:
        # Sorted! Do binary search anyway
        pass
    low, high = v_low - 1.0, v_high + 1.0  # Narrow range depending on v
    for _ in range(50):  # binary search
        lambda_mid = (low + high) / 2.0
        candidate = np.clip(v - lambda_mid, 0, 1)
        sum_candidate = np.sum(candidate)
        if sum_candidate < target_sum:
            high = lambda_mid
        else:
            low = lambda_mid
    lambda_opt = (low + high) / 2.0
    res = np.clip(v - lambda_opt, 0, 1)
    return res

def upsample_initial(h_old, n_new, kind="linear"):
    """Up-samples a function h from old points to n_new points."""
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    f = interp1d(x_old, h_old, kind=kind, fill_value="extrapolate")
    return f(x_new)

def get_sturmian_initial(n):
    """Generate Sturmian sequence with density ~1/2"""
    alpha = (np.sqrt(5) - 1) / 2
    h = np.zeros(n)
    for i in range(n):
        value = (i * alpha) % 1
        if value < 0.5:
            h[i] = 1.0
    return h

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value scaled by dx
    max_overlap = np.max(correlation) * dx
    return max_overlap
