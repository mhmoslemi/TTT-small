import numpy as np
import time
import cvxpy as cp

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using full convolution
    overlap = np.correlate(h, 1 - h, mode='full')
    # The maximum value scaled by dx
    c5 = np.max(overlap) * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Set N to 139 to leverage Paley difference set
    N = 139
    dx = 2.0 / N
    
    # Build the MILP model
    x = cp.Variable(N, boolean=True)
    C = cp.Variable()
    
    # Constraint: sum of h = N/2 (implying sum = 69.5 in continuous case, 
    # We'll set two entries to 0.5 to balance)
    # First let's ensure integer constraint as best as we can.
    # But we enforce integral of h over [0,2] to be 1. 
    # With dx = 2/N, we have sum(h)*dx = 1 => sum(h) = N/2
    # So sum(x)*dx = (N/2)*dx = 1. Exactly. So we do need sum to be N/2.
    # But if N = 139, sum(h) = 139/2 = 69.5 which is not integer!
    # We can use x_i ∈ [0,1] OR just make it work for binary.
    # Let's instead use N = 140 to get exactly integer sum = 70.  
    # Though previous best used 140, we go slightly different to explore.
    
    # But to use difference sets, best N prime ≡ 1 mod 4, so use N=139 with
    # careful handling. We must allow non-binary values, but to keep solvability,
    # we'll do approx. But otherwise using N=140 will lead to full binary support.
    # Use simple iterative approach for lower N to get initial good solution.
    # We'll just try N=140 for feasible binary problem.
    
    # But recall in Arjen's approach previously, solver defined is integer.
    # For 139, we have 69.5 ones. Not allowed for all 0/1.
    # If allowing fraction in h values, for sum = 70 exactly, use 70 ones but 
    # we adjust such that sum hh = 70 and the remaining ~0.5 gets allocated.
    # Or just use 140 and stick to integer case to be fully rigorous with int constraints.
    
    # For definitive approach for ideal difference set = 139:
    # Get 69 ones, 70 zeros element is half.
    
    # So start with N=140
    N = 140
    dx = 2.0 / N
    x = cp.Variable(N, boolean=True)
    constraints = [cp.sum(x) == N//2]
    
    # Ensure C is upper bound for all c_ell
    C = cp.Variable()
    
    for ell in range(1, N):
        length = N - ell
        if length <= 0:
            break
        H_ell = cp.sum(x[:length])  # sum of first N-ell terms
        y = cp.Variable(length, boolean=True)
        
        # y_i = x_i * x_{i+ell}
        constraints += [y <= x[:length],
                        y <= x[ell:],
                        y >= x[:length] + x[ell:] - 1]
        
        O_ell = cp.sum(y)
        c_ell = H_ell - O_ell
        
        # Upper bound constraint: c_ell <= C
        constraints += [c_ell <= C, 
                        c_ell >= 0]  # c ell non-negative
    
    # Objective
    prob = cp.Problem(cp.Minimize(C), constraints)
    
    # Solve with time limit
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time < 5:
        # Fall back to old method if time is short
        from initial_h_values import initial_h_values
        h_initial = initial_h_values
        if len(h_initial) != N:
            x_orig = np.linspace(0, 2, len(h_initial))
            x_new = np.linspace(0, 2, N)
            h_initial = np.interp(x_new, x_orig, h_initial)
        h_initial = h_initial * (N / 2) / np.sum(h_initial)
        return (h_initial.astype(float), c5_of(h_initial), N)
        
    try:
        prob.solve(solver=cp.CBC, max_seconds=remaining_time-5, verbose=False)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE, cp.FEASIBLE]:
            x_vals = np.array(x.value).round().astype(int).clip(0,1)
            # Ensure exactly sum 70
            if np.sum(x_vals) != 70:
                # Manually fix if needed
                needed = 70 - np.sum(x_vals)
                if needed > 0:  # Need to set more zeros to 1s
                    indices = np.where(x_vals == 0)[0]
                    if len(indices) >= needed:
                        x_vals[indices[:needed]] = 1
                elif needed < 0:  # Need to set more ones to 0s
                    indices = np.where(x_vals == 1)[0]
                    if len(indices) >= -needed:
                        x_vals[indices[:(-needed)]] = 0
                        
            h = x_vals.astype(float)
            c5_bound = c5_of(h)
            return (h, c5_bound, N)
        else:
            raise RuntimeError("MILP solver did not find feasible solution")
    except Exception as e:
        # Fallback to old algorithm if needed
        from initial_h_values import initial_h_values
        h_initial = initial_h_values
        if len(h_initial) != N:
            x_orig = np.linspace(0, 2, len(h_initial))
            x_new = np.linspace(0, 2, N)
            h_initial = np.interp(x_new, x_orig, h_initial)
        h_initial = h_initial * (N / 2) / np.sum(h_initial)
        return (h_initial.astype(float), c5_of(h_initial), N)
