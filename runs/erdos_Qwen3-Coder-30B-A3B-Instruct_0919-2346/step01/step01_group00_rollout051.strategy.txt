import numpy as np, time, random, math

def c5_of(h):
    dx = 2.0 / len(h)
    c = np.correlate(h, 1-h, mode='full')
    c *= dx
    return np.max(c)

def worst_shift(h):
    N = len(h)
    dx = 2.0/N
    c = np.correlate(h, 1-h, mode='full')
    c *= dx
    idx = np.argmax(c)
    k = idx - (N-1)
    val = c[idx]
    return k, val, c

def subgradient(h, k):
    N = len(h)
    dx = 2.0 / N
    g = np.zeros_like(h)
    i = np.arange(N)
    j = i + k
    mask = (j >=0) & (j < N)
    ii = i[mask]
    jj = j[mask]
    # source contributions
    g[ii] += dx * (1.0 - h[jj])
    # target contributions negative
    g[jj] -= dx * h[ii]
    return g

def project_box_sum(x, target, low=0.0, high=1.0, max_iter=60):
    # water-filling projection onto [low,high] with sum target
    lo = np.min(x) - high
    hi = np.max(x) - low
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, low, high)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo+hi)/2.0, low, high)

def random_feasible(N, target):
    # random start, then project.
    x = np.random.rand(N)
    x = project_box_sum(x, target)
    return x

def gradient_descent(h0, max_iter, eta0, target):
    h = h0.copy()
    best_h = h.copy()
    best_val = c5_of(h)
    N = len(h)
    for t in range(max_iter):
        k, val, c = worst_shift(h)
        if val < best_val:
            best_val = val; best_h = h.copy()
        g = subgradient(h, k)
        eta = eta0 / np.sqrt(t+1)
        h_new = h - eta * g
        h = project_box_sum(h_new, target)
    # final best
    return best_h, best_val
