import numpy as np
import time
import random

def compute_c_vector(h, dx):
    """Compute the vector of overlaps c_τ = 1 - dx * S_τ(h) for all shifts τ."""
    # Compute auto-correlation via FFT
    n = len(h)
    # Zero-pad to avoid wraparound
    S = np.correlate(h, h, mode='full')
    # Truncate to valid range: τ = 0 to 2n-2
    S = S[n-1:2*n-1]
    # Convert to overlaps
    c = 1 - dx * S
    return c

def compute_softmax_weights(c, alpha):
    """Compute w = softmax(alpha * c) with numerical stability."""
    c_max = np.max(c)
    exp_c = np.exp(alpha * (c - c_max))
    w = exp_c / np.sum(exp_c)
    return w

def compute_gradient(h, w, dx):
    """Compute discrete convolution-based gradient: ∇f_α(h) = -dx * (w * h + w_rev * h)."""
    n = len(h)
    # w_rev = w reversed
    w_rev = w[::-1]
    # Convolve: g = dx * (w * h + w_rev * h)
    conv1 = np.convolve(w, h, mode='same')
    conv2 = np.convolve(w_rev, h, mode='same')
    g = -dx * (conv1 + conv2)
    return g

def project_to_feasible(h, target_sum):
    """Project h onto [0,1]^n subject to sum(h) = target_sum (water-filling)."""
    # Water-filling: solve for λ such that sum(clamp(h_i - λ, 0, 1)) = target_sum
    h = np.asarray(h)
    h_sorted = np.sort(h)[::-1]  # descending order
    n = len(h)
    
    # Initial guess for λ
    lambda_min = h_sorted[-1] - 1.0
    lambda_max = h_sorted[0] + 1.0
    
    def phi(lam):
        # sum_i max(0, h_i - lam)
        clipped = np.maximum(0.0, h - lam)
        return np.sum(clipped) - target_sum
    
    # bisection search for root
    eps = 1e-12
    while lambda_max - lambda_min > eps:
        lam = (lambda_min + lambda_max) / 2.0
        val = phi(lam)
        if val > 0:
            lambda_min = lam
        else:
            lambda_max = lam
            
    lambda_sol = (lambda_min + lambda_max) / 2.0
    return np.clip(h - lambda_sol, 0.0, 1.0)

def gradient_descent_step(h, w, g, eta, target_sum):
    """Perform a gradient step and project onto feasible set."""
    # Update
    h_new = h - eta * g
    # Project
    h_proj = project_to_feasible(h_new, target_sum)
    return h_proj

    x = np.linspace(0, 2, n)

    h_double[mask] = u
    h_double[~mask] = v
    h_double = h_double * (target_sum / np.sum(h_double))
    
    # 3. Uniform
    h_uniform = np.full(n, 0.5)
    h_uniform = h_uniform * (target_sum / np.sum(h_uniform))

    best_h = None
    best_C5 = float('inf')
    
    # Global parameters
    alpha_start = 5.0
    alpha_end = 80.0
    eta_0 = 3.0  # Initial step size (assumes a max step change of ~0.02)
    momentum_beta = 0.7
    K_restart = 300  # Restart every K steps
    sigma_perturb = 0.01
    
    # Allocate budget: seeding (5%), gradient descent (80%), polishing (15%)
    seeding_budget = 0.05 * budget_s
    grad_budget = 0.80 * budget_s
    polishing_budget = 0.15 * budget_s
    writeup_budget = budget_s - seeding_budget - grad_budget - polishing_budget
    writeup_budget = max(0, writeup_budget)
    
    # Initialize seeds
    seeds = []
    for i, h_seed in enumerate(experiments):
        h_clone = h_seed.copy()
        if i == 0:  # Ensure exact constraint is met
            h_clone = project_to_feasible(h_clone, target_sum)
        seeds.append(h_clone)
        
    # Begin gradient descent phase for each seed
    grad_end_time = start_time + grad_budget
    
    iter_count = 0
    for h_seed in seeds:
        if time.time() > grad_end_time:
            break
        h_last = h_seed.copy()
        c_last = compute_c_vector(h_last, dx)
        C5_last = np.max(c_last)
        if C5_last < best_C5:
            best_C5 = C5_last
            best_h = h_last.copy()
        
        # Momentum term
        m = np.zeros_like(h_last)
        
        # Main optimization loop
        while time.time() < grad_end_time:
            iter_count += 1
            # Compute quantities
            c = compute_c_vector(h_last, dx)
            
            # Compute current alpha (increase from 5 to 80)
            alpha = alpha_start + (alpha_end - alpha_start) * (iter_count / 5000)

            w = compute_softmax_weights(c, alpha)
            
            # Compute gradient
            g = compute_gradient(h_last, w, dx)
            
            # Update momentum
            m = momentum_beta * m + (1.0 - momentum_beta) * g

                best_h = h_proj.copy()

            # Periodic restart
            if iter_count % K_restart == 0:
                noise = np.random.normal(0, sigma_perturb, n)
                h_perturbed = h_last + noise

                    h_last = h_perturbed

            # Early stopping on low gain
            if iter_count % 1000 == 0 and iter_count > 1000:

        best_C5 = compute_c_vector(best_h, dx).max()

    cur_h = best_h.copy()
    cur_c5 = best_C5
    
    def propose_move(h, max_step):
        h = h.copy()
        i, j = random.sample(range(n), 2)
        
        # Determine feasible interval for delta
        delta_low = max(-h[i], h[j] - 1.0)
        delta_high = min(1.0 - h[i], h[j])
        
        if delta_low >= delta_high:
            return h
        
        # Sample delta within allowed range
        step = min(max_step, delta_high - delta_low)
        delta = random.uniform(-step, step)
        delta = max(delta_low, min(delta, delta_high))
        
        h[i] += delta
        h[j] -= delta
        
        # Project to feasible
        h = project_to_feasible(h, target_sum)
        return h
    
    while time.time() < polish_end_time:
        h_new = propose_move(cur_h, 0.02)
        c_new = compute_c_vector(h_new, dx).max()
        if c_new <= cur_c5 or random.random() < np.exp(-(c_new - cur_c5) / 0.01):
            cur_h = h_new
            cur_c5 = c_new
            if cur_c5 < best_C5:
                best_C5 = cur_c5
                best_h = cur_h.copy()
    
    return (best_h, best_C5, n)
