import numpy as np

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral via full convolution of h with (1-h)
    overlaps = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum of the full correlation vector
    # scaled by dx
    c5 = np.max(overlaps) * dx
    return c5

def evaluate_h(h):
    """Evaluate the objective function"""
    return c5_of(h)

def project_to_feasible(h):
    """Project h to satisfy constraint sum(h) * dx = 1"""
    dx = 2.0 / len(h)
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if abs(sum_h * dx - 1.0) > 1e-10:
        h = h * (1.0 / (sum_h * dx))
    return h
