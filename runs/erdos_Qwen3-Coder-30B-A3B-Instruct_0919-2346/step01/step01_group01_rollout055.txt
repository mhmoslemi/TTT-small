import numpy as np
import time
import random
from scipy.optimize import differential_evolution

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using full convolution
    overlap = np.correlate(h, 1 - h, mode='full')
    # The maximum value scaled by dx
    c5 = np.max(overlap) * dx
    return c5

def propose_move(h, max_step):
    """Propose a move that maintains sum constraint exactly"""
    h_new = h.copy()
    N = len(h)
    while True:
        i, j = random.sample(range(N), 2)
        # Compute feasible range for delta
        delta_low = max(-h_new[i], h_new[j] - 1.0)
        delta_high = min(1.0 - h_new[i], h_new[j])
        if delta_low >= delta_high:
            continue  # No valid move, try again
        break
    
    # Sample delta from feasible interval
    step = min(max_step, delta_high - delta_low)
    if step <= 0:
        delta = 0.0
    else:
        delta = random.uniform(max(-step, delta_low), min(step, delta_high))
        
    h_new[i] += delta
    h_new[j] -= delta
    return h_new

def optimize_with_differential_evolution(initial_h, n_points):
    """Use differential evolution to find a good local minimum"""
    def objective(h):

        h = h * (n_points / 2) / np.sum(h)
        return c5_of(h)
    
    # Bounds for each variable in the optimization
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Run differential evolution
    result = differential_evolution(
        objective, 
        bounds, 
        maxiter=100, 
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.7,
        seed=42,
        tol=1e-6,
        workers=1
    )
    
    h_opt = result.x
    h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
    return h_opt, objective(h_opt)
