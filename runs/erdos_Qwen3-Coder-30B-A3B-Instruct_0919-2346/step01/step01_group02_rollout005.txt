import numpy as np
import time
from scipy.optimize import minimize
from scipy.special import logsumexp

def c5_of(h):
    """Compute the maximum overlap integral for the given step function h."""
    dx = 2.0 / len(h)
    c = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(c) * dx
    return c5

def project_to_sum_constraint(h, target_sum):
    """Project h onto the constraint that sum(h) = target_sum using water-filling."""
    N = len(h)
    v = h.copy()
    # Binary search for lambda
    lower, upper = -1000.0, 1000.0
    for _ in range(50):
        lambda_ = (lower + upper) / 2.0
        projected = np.clip(v + lambda_, 0, 1)
        sum_proj = np.sum(projected)
        if sum_proj < target_sum:
            lower = lambda_
        else:
            upper = lambda_
    lambda_ = (lower + upper) / 2.0
    h_proj = np.clip(v + lambda_, 0, 1)
    # Fix any small numerical error
    diff = target_sum - np.sum(h_proj)
    # Distribute the difference among indices with non-bound values
    if np.abs(diff) > 1e-10:
        free_indices = np.where((h_proj > 0) & (h_proj < 1))[0]
        num_free = len(free_indices)
        if num_free > 0:
            delta = diff / num_free
            h_proj[free_indices] += delta
    return h_proj

def evaluate_smooth_surrogate(h, epsilon):
    dx = 2.0 / len(h)
    c = np.correlate(h, 1 - h, mode='full')
    scaled_c = c * dx
    # Compute log-sum-exp with numerical stabilization
    max_c = np.max(scaled_c)
    if np.isinf(max_c) or np.isnan(max_c):
        max_c = 0
    log_weights = scaled_c - max_c
    log_weights /= epsilon
    log_sum_exp = max_c + epsilon * logsumexp(log_weights)
    return log_sum_exp

def compute_gradient(h, epsilon):
    dx = 2.0 / len(h)
    c = np.correlate(h, 1 - h, mode='full')
    scaled_c = c * dx
    # Compute log-sum-exp weights for smoothing
    max_c = np.max(scaled_c)
    if np.isinf(max_c) or np.isnan(max_c):
        max_c = 0
    log_weights = scaled_c - max_c
    log_weights /= epsilon
    weights = np.exp(log_weights)
    sum_weights = np.sum(weights)
    weights /= sum_weights
    
    # Compute gradient: ∇F_ε = Σ_s w_s ∇C_s
    grad = np.zeros_like(h)
    N = len(h)
    # For each shift s, compute the contribution of ∇C_s
    full_c = np.correlate(h, 1 - h, mode='full')
    # The shift index k maps to full_c[s + N - 1]
    # For shift s in range(2*N-1):
    #   ∇C_s[k] = dx * (1 - h[k+s] - h[k-s]) (or 0 outside domain)
    # We compute ∇C_s directly from shifted copies of h
    for shift in range(2 * N - 1):
        # Compute ∇C_s with zero padding
        if shift < N - 1:
            # h shifted left, h shifted right
            left_shift_idx = shift
            right_shift_idx = 0
            shift_left = np.zeros(N)
            shift_right = np.zeros(N)
            shift_left[left_shift_idx:] = h[:-left_shift_idx]
            if right_shift_idx < N:
                shift_right[:N - right_shift_idx] = h[right_shift_idx:]
            elif right_shift_idx == N:
                shift_right[:] = 0
            else:
                shift_right = h[right_shift_idx:]
            grad_component = 1 - shift_left - shift_right
        elif shift == N - 1:
            grad_component = 1 - h - h
        else:
            # shift > N - 1
            left_shift_idx = 0
            right_shift_idx = 2 * N - 2 - shift
            shift_left = np.zeros(N)
            shift_right = np.zeros(N)
            if left_shift_idx < N:
                shift_left[left_shift_idx:] = h[:-left_shift_idx]
            else:
                shift_left[:] = 0
            shift_right[:N - right_shift_idx] = h[right_shift_idx:]
            grad_component = 1 - shift_left - shift_right
        # Broadcast the gradient component with proper scaling
        grad += weights[shift] * grad_component * dx
    return grad

def continuation_optimize(h_seed, target_sum, max_iter=150, tol=1e-6):
    N = len(h_seed)
    dx = 2.0 / N
    epsilon_schedule = [0.10, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    h = h_seed.copy()
    h = project_to_sum_constraint(h, target_sum)
    best_h = h.copy()
    best_c5 = c5_of(h)
    for epsilon in epsilon_schedule:
        def fun(x):
            return evaluate_smooth_surrogate(x, epsilon)
        def grad(x):
            return compute_gradient(x, epsilon)
        # Define bounds for each variable
        bounds = [(0, 1) for _ in range(N)]
        res = minimize(fun, h, method='L-BFGS-B', jac=grad, bounds=bounds, \
                       options={'maxiter': max_iter, 'gtol': tol})
        h = res.x
        h = project_to_sum_constraint(h, target_sum)
        c5 = c5_of(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
        # Now do one final projection to be very sure
        h = project_to_sum_constraint(h, target_sum)
    return best_h, best_c5

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    N = 140
    dx = 2.0 / N
    target_sum = N / 2.0
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    seeds = []
    
    # Generate diverse seeds
    # Uniform seed
    h_uni = np.ones(N) * 0.5
    h_uni = project_to_sum_constraint(h_uni, target_sum)
    seeds.append(("uniform", h_uni))
    
    # Two-peak seed with various a values
    a_values = [0.12, 0.18, 0.24, 0.30]
    for a in a_values:
        h = np.ones(N)
        x_vals = np.linspace(0, 2, N)
        u = 0.9
        v = 0.1
        mask = (x_vals < a) | (x_vals >= 2 - a)
        h[mask] = u
        h[~mask] = v
        h = project_to_sum_constraint(h, target_sum)
        seeds.append((f"two_peak_{a}", h))
        
    # Binary seeds
    np.random.seed(1000 + seed)
    for i in range(5):
        indices = np.random.permutation(N)[:N//2]
        h = np.zeros(N)
        h[indices] = 1.0
        h = project_to_sum_constraint(h, target_sum)
        seeds.append((f"binary_{i}", h))
        
    # Biased binary seed (two blocks)
    h_block = np.zeros(N)
    h_block[:N//2] = 1.0
    h_block = project_to_sum_constraint(h_block, target_sum)
    seeds.append(("block", h_block))
    
    # Random continuous seed
    h_rand = np.random.rand(N)
    h_rand = project_to_sum_constraint(h_rand, target_sum)
    seeds.append(("random", h_rand))
    
    # Continue with multi-start
    remain_time = budget_s * 0.7
    start_time = time.time()
    for seed_name, seed_h in seeds:
        if time.time() - start_time > remain_time - 5:
            break
        h_opt, c5_opt = continuation_optimize(seed_h, target_sum, max_iter=150)
        if c5_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_opt
    # Final polishing with sub-gradient descent
    remain_time = budget_s * 0.1
    start_time = time.time()
    h = best_h.copy()
    best_c5 = c5_of(h)
    while time.time() - start_time < remain_time:
        # Find worst shift
        c = np.correlate(h, 1 - h, mode='full')
        idx_max = np.argmax(c)
        c5 = c[idx_max]
        s = idx_max - N + 1 # Convert to shift index
        # Compute sub-gradient for this shift
        if s < 0:
            # shift_left is h shifted right by |s|, shift_right is h shifted left by |s|
            shift_left = np.zeros(N)
            shift_left[-s:] = h[:-(-s)]
            shift_right = np.zeros(N)
            shift_right[:s] = h[(-s):]
        elif s == 0:
            shift_left = h
            shift_right = h
        else: # s > 0
            shift_left = np.zeros(N)
            shift_left[s:] = h[:-s]
            shift_right = np.zeros(N)
            shift_right[:s] = h[s:]
        g = 1.0 - shift_left - shift_right
        g = g * dx # scale by dx 
        # Projected gradient step
        h_new = h - 0.03 * g
        h_new = project_to_sum_constraint(h_new, target_sum)
        c5_new = c5_of(h_new)
        if c5_new < best_c5 - 1e-8: # significant improvement
            best_h = h_new.copy()
            best_c5 = c5_new
            h = h_new.copy()
        else:
            break # no significant improvement, stop
    return (best_h, best_c5, N)
