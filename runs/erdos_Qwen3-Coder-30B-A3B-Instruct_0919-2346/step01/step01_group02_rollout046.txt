import numpy as np
import time
from scipy.optimize import differential_evolution

def c5_of(h):
    """Compute the maximum overlap integral for the given step function h."""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    c = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(c) * dx
    return c5

def project_to_feasible(h):
    """Project h to satisfy constraints: 0 <= h <= 1, sum(h) * dx = 1"""
    h = np.clip(h, 0, 1)
    dx = 2.0 / len(h)
    h = h * (1.0 / np.sum(h)) * (1.0 / dx)  # Renormalize to sum(h) * dx = 1
    return h

def evaluate_objective(h):
    """Objective function to minimize C5"""
    return c5_of(h)

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    N = 140
    dx = 2.0 / N
    
    # Use the provided initial construction if available
    if 'initial_h_values' in globals():
        h0 = np.interp(np.linspace(0, 2, N), 
                      np.linspace(0, 2, len(initial_h_values)), 
                      initial_h_values)
        h0 = h0 * (N/2) / np.sum(h0)  # Renormalize exactly
    else:
        h0 = np.ones(N) * 0.5  # fallback
    
    # Setup bounds for each variable
    bounds = [(0, 1) for _ in range(N)]
    
    # Use scipy's differential evolution which is a global optimizer
    start = time.time()
    
    # Set up differential evolution with appropriate bounds and constraints
    # Since we want exactly sum(h)*dx = 1, we'll add this as a constraint
    def constraint_sum(h):
        return np.sum(h) * dx - 1.0
    
    # Define the constraint dictionary
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    # Run optimization with reasonable timeout
    result = differential_evolution(
        evaluate_objective,
        bounds,
        constraint,
        seed=seed,
        maxiter=1000,
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.7,
        disp=False,
        callback=None
    )
    
    h_final = project_to_feasible(result.x)
    c5_final = c5_of(h_final)
    
    # Additional local search 
    max_iter = 20000
    current_h = h_final.copy()
    current_c5 = c5_final
    
    # Simple local search to further refine
    for i in range(max_iter):
        if time.time() - start > budget_s * 0.95:
            break
            
        # Sample fixed set of candidate moves
        idx1 = np.random.randint(0, N)
        idx2 = np.random.randint(0, N)
        
        # Get bounds for allowed moves
        h1 = current_h[idx1]
        h2 = current_h[idx2]
        
        delta_low = max(-h1, h2 - 1)
        delta_high = min(1 - h1, h2)
        
        if delta_low < delta_high:
            step_size = np.random.uniform(delta_low, delta_high)
            
            # Make move
            new_h = current_h.copy()
            new_h[idx1] += step_size
            new_h[idx2] -= step_size
            
            # Project to feasible region
            new_h = project_to_feasible(new_h)
            
            # Evaluate
            new_c5 = c5_of(new_h)
            
            # Accept if better or with some probability
            if new_c5 < current_c5:
                current_h = new_h
                current_c5 = new_c5
    
    return (current_h, current_c5, N)
