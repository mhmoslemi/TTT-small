import numpy as np
import time
import random
from scipy.optimize import differential_evolution, minimize


def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    overlap = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(overlap) * dx
    return c5


def propose_move(h, max_step):
    """Propose a new feasible h vector by adjusting two elements"""
    h_new = h.copy()
    N = len(h)
    
    # Try to find a valid move
    max_tries = 100
    for _ in range(max_tries):
        i = random.randint(0, N - 1)
        j = random.randint(0, N - 1)
        if i == j:
            continue
            
        # Determine feasible interval for δ
        delta_low = max(-h_new[i], h_new[j] - 1)
        delta_high = min(1 - h_new[i], h_new[j])
        
        if delta_low >= delta_high:
            continue
            
        # Sample δ from uniform distribution over feasible range
        step = min(max_step, delta_high - delta_low)
        if step <= 0:
            continue
        delta = random.uniform(delta_low, delta_high)
        
        # Apply move
        h_new[i] += delta
        h_new[j] -= delta
        
        return h_new
    
    # If no valid move found, return original
    return h.copy()


def simple_local_search(initial_h, time_budget_s):
    """Simple local search with guided mutations"""
    start = time.time()
    
    cur = initial_h.copy()
    c_cur = c5_of(cur)
    
    best = cur.copy()
    c_best = c_cur
    
    # Parameters for adaptive mutation
    max_step = 0.1
    step_decay = 0.9995
    
    # Main loop
    while time.time() - start < time_budget_s:
        # Adaptive step size
        if random.random() < 0.05:  # Occasionally try large steps
            step = 0.2
        else:
            step = max_step
        
        h_proposed = propose_move(cur, step)
        c_proposed = c5_of(h_proposed)
        
        # Accept the proposal
        if c_proposed <= c_cur:
            cur = h_proposed
            c_cur = c_proposed
            if c_proposed < c_best:
                best = h_proposed.copy()
                c_best = c_proposed
                max_step = min(0.2, max_step * 1.02)
        else:
            # Occasionally accept worse moves to escape local optima
            if random.random() < 0.01:
                cur = h_proposed
                c_cur = c_proposed                
        
        max_step *= step_decay
        if max_step < 1e-5:
            max_step = 1e-5
            
    return best, c_best


def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function"""
    random.seed(seed)
    np.random.seed(seed)
    
    start = time.time()
    N = 140
    dx = 2.0 / N
    
    # Build initial seeds
    # 1. Provided initial_h_values (interpolated if needed)
    if 'initial_h_values' in globals():
        h_provided = np.interp(
            np.linspace(0, 2, N),
            np.linspace(0, 2, len(initial_h_values)),
            initial_h_values
        )
        # Renormalize to exact sum
        h_provided = h_provided * (N / 2) / np.sum(h_provided)
    else:
        h_provided = np.ones(N) * 0.5
    
    # 2. Double-peak seed
    h_double = np.full(N, 0.1)
    a = 0.3
    u = 0.9
    v = 0.1
    left = int(a * N / 2)
    right = int((2 - a) * N / 2)
    h_double[:left] = u
    h_double[right:] = u
    # Normalization
    h_double = h_double * (N / 2) / np.sum(h_double)
    
    # 3. Gauss-like shape
    x = np.linspace(0, 2, N)
    sigma = 0.2
    h_gauss = np.exp(-((x - 1)**2) / (2 * sigma**2))
    h_gauss = h_gauss * (N / 2) / np.sum(h_gauss)
    
    # Test all initial seeds and improved versions
    seeds = [h_provided, h_double, h_gauss]
    
    best_h = None
    best_c5 = float('inf')
    
    # Run each seed for about 1/3 of remaining time
    remaining_time = budget_s * 0.9
    time_per_seed = min(remaining_time / len(seeds), 200)
    
    for i, h_seed in enumerate(seeds):
        if time.time() - start >= budget_s * 0.95:
            break
        h_opt, c5_opt = simple_local_search(h_seed, time_per_seed)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    # Additional optimization with more sophisticated method on best solution
    if best_h is not None and time.time() - start < budget_s * 0.95:
        # Fine-tune with differential evolution 
        bounds = [(0, 1) for _ in range(N)]
        de_result = differential_evolution(
            lambda x: c5_of(x),
            bounds,
            args=(),
            maxiter=500,
            popsize=10,
            mutation=(0.5, 1),
            recombination=0.7,
            seed=seed,
            disp=False,
            tol=1e-6
        )
        
        if de_result.success and de_result.fun < best_c5:
            best_c5 = de_result.fun
            best_h = de_result.x
            
    # Final local refinement
    if best_h is not None and time.time() - start < budget_s * 0.98:
        final_h, final_c5 = simple_local_search(best_h, budget_s * 0.02)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = final_h

    return (best_h, best_c5, N)
