import numpy as np

from scipy.optimize import minimize

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value scaled by dx
    max_overlap = np.max(correlation) * dx
    return max_overlap

def generate_sturmian_binary(n):
    """Generate a balanced binary sequence using irrational rotation."""
    alpha = (np.sqrt(5) - 1) / 2  # Golden ratio conjugate
    h = np.zeros(n)
    positions = np.mod(np.arange(n) * alpha, 1)
    h[positions < 0.5] = 1
    
    # Adjust to have exactly n/2 ones
    current_ones = int(np.sum(h))
    needed_ones = n // 2
    if current_ones > needed_ones:
        # Turn some 1s to 0s
        indices = np.where(h == 1)[0]
        flip_indices = indices[:current_ones - needed_ones]
        h[flip_indices] = 0
    elif current_ones < needed_ones:
        # Turn some 0s to 1s
        indices = np.where(h == 0)[0]
        flip_indices = indices[:needed_ones - current_ones]
        h[flip_indices] = 1
    
    return h

def compute_overlap_vector(h):
    """Compute overlap vector g_s for all shifts s."""
    n = len(h)
    dx = 2.0 / n
    correlation = np.correlate(h, 1 - h, mode="full")
    return correlation * dx

def compute_delta_g(h, i, j, overlap_vec):

    for s in range(2 * n - 1):

        # To compute just g_s: contribution of i in i+s mod n = s permutation.

        # Simple solution:
        # Use full simple direct formula:
        # Full g_s = sum_{l=0}^{n-1} h_l * (1 - h_{(l+s)%n})
        # Change after swap i,j:
        # Old: h_i * (1 - h_{(i+s)%n}) + h_j * (1 - h_{(j+s)%n})
        # New: (1-h_i) * (1 - h_{(i+s)%n}) + (1-h_j) * (1 - h_{(j+s)%n})
        # So delta s = [ (1-h_i) * (1 - h_{(i+s)%n}) - h_i * (1 - h_{(i+s)%n}) ] 
        #             + [ (1-h_j) * (1 - h_{(j+s)%n}) - h_j * (1 - h_{(j+s)%n}) ]

        # To make it correct directly, the effect of swapping i and j is:
        # Final g_s = g_s +  delta_{i}(s) + delta_j(s) 
        # where:
        # delta_i(s) = (1-2h_i) * (1-h_{(i+s) % n})
        # delta_j(s) = (1-2h_j) * (1-h_{(j+s) % n})
        
        if h[i] == 1:
            delta[s] += (1 - 2 * h[i]) * (1 - h[(i + s) % n])
        if h[j] == 1:
            delta[s] += (1 - 2 * h[j]) * (1 - h[(j + s) % n])
    
    return delta

def tabu_local_swap_search(h, n_points, time_budget_s):
    """Perform local swap search using tabu list and targeted moves."""
    start_time = time.time()
    max_c5 = evaluate_erdos_solution(h)
    current_h = h.copy()
    current_overlaps = compute_overlap_vector(current_h)
    best_h = current_h.copy()
    best_c5 = max_c5
    tabu_list = []
    tabu_length = 100
    max_iter = 10000
    iter_count = 0
    
    while (time.time() - start_time) < time_budget_s * 0.9 and iter_count < max_iter:
        # Find worst shift:
        max_shift = np.argmax(current_overlaps)
        max_overlap = current_overlaps[max_shift]

        # Find positions that are good candidates to move
        ones = np.where(current_h == 1)[0]
        zeros = np.where(current_h == 0)[0]

        # Sample some candidate swaps
        np.random.shuffle(ones)
        np.random.shuffle(zeros)
        candidates = []
        for i in ones[:50]:
            for j in zeros[:50]:
                if (i, j) not in tabu_list:
                    candidates.append((i, j))

        # Try swaps
        for i, j in candidates:
            # Compute overlap with potential swap
            temp_h = current_h.copy()
            temp_h[i] = 0
            temp_h[j] = 1
            new_c5 = evaluate_erdos_solution(temp_h)
            
            if new_c5 < max_c5:
                # Accept improvement
                current_h = temp_h
                current_overlaps = compute_overlap_vector(current_h)
                max_c5 = new_c5
                if max_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = max_c5
                # Update tabu list
                if len(tabu_list) >= tabu_length:
                    tabu_list.pop(0)
                tabu_list.append((i, j))
                break
            else:
                # Make some random swaps for diversity
                if np.random.rand() < 0.01:
                    # Do random swap
                    temp_h = current_h.copy()
                    i_new = np.random.choice(np.where(current_h == 1)[0])
                    j_new = np.random.choice(np.where(current_h == 0)[0])
                    temp_h[i_new] = 0
                    temp_h[j_new] = 1
                    new_c5 = evaluate_erdos_solution(temp_h)
                    if new_c5 < max_c5:
                        current_h = temp_h
                        current_overlaps = compute_overlap_vector(current_h)
                        max_c5 = new_c5
                        if max_c5 < best_c5:
                            best_h = current_h.copy()
                            best_c5 = max_c5
                        # Add to tabu list
                        if len(tabu_list) >= tabu_length:
                            tabu_list.pop(0)
                        tabu_list.append((i_new, j_new))
                
        iter_count += 1
    
    if best_c5 < max_c5:
        return best_h, best_c5
    else:
        return current_h, max_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h = generate_sturmian_binary(120)
    h, c5 = tabu_local_swap_search(h, 120, budget_s * 0.5)
    # Upscale to more points
    h_up = np.repeat(h, 2)
    # Do more local search on bigger grid
    h_up, c5_up = tabu_local_swap_search(h_up, 240, budget_s * 0.3)
    # Final polishing if possible  
    if (time.time() - start_time) < 0.5 * budget_s:
        # If time left, do final continuous optimization to refine the solution
        # This is inspired by the previous approach but without SLSQP which we're avoiding
        dx = 2.0 / len(h_up)
        # Return best solution
        return (h_up.tolist(), c5_up, len(h_up))
    
    return (h_up.tolist(), c5_up, len(h_up))
