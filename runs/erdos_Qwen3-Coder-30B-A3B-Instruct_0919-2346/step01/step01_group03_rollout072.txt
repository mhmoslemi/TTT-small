import numpy as np
import time
from scipy.optimize import minimize

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral via full convolution of h with (1-h)
    overlaps = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum of the full correlation vector
    # scaled by dx
    c5 = np.max(overlaps) * dx
    return c5

def project_to_feasible(h):
    """Project h to satisfy the constraints: 0 <= h <= 1 and sum(h) * dx = 1"""
    dx = 2.0 / len(h)
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h *= (1 / dx) / h_sum
    return h

def propose_move(h, max_step):
    """Propose a feasible move by adjusting two entries."""
    h = h.copy()
    N = len(h)
    
    # Try to find a valid pair of indices
    for _ in range(100):  # avoid infinite loop
        i, j = np.random.randint(0, N, 2)
        if i == j:
            continue
        # Determine feasible δ range for the move
        delta_low = max(-h[i], h[j] - 1)
        delta_high = min(1 - h[i], h[j])
        if delta_low >= delta_high:
            continue
        # Sample a move with limited step size
        step = min(max_step, delta_high - delta_low)
        delta = np.random.uniform(
            max(-step, delta_low),
            min(step, delta_high)
        )
        # Apply move
        h[i] += delta
        h[j] -= delta
        return h
    # If no valid move found, return copy
    return h

    x = np.linspace(0, 2, N)
    h4 = np.maximum(0, 1 - np.abs(x - 1))
    h4 *= (N / 2) / np.sum(h4)
    
    # Normalize and project to feasible
    h1 = project_to_feasible(h1)
    h2 = project_to_feasible(h2) if h2 is not None else project_to_feasible(np.random.rand(N)*0.5)
    h3 = project_to_feasible(h3)
    h4 = project_to_feasible(h4)
    
    return [h1, h2, h3, h4]

def anneal_with_adaptive_temperature_and_step(initial_h, time_budget, seed):
    """Run simulated annealing from initial_h."""
    np.random.seed(seed)
    start = time.time()
    cur = initial_h.copy()
    best = initial_h.copy()
    c_cur = c5_of(cur)
    c_best = c_cur
    T_start = 0.05
    T_end = 1e-6
    step_start = 0.2
    T = T_start
    step = step_start

    while time.time() - start < time_budget:
        elapsed = time.time() - start
        tfrac = elapsed / time_budget
        T = T_start * (T_end / T_start) ** tfrac
        step = step_start * (1 - tfrac)

        # Propose a move
        h_new = propose_move(cur, step)
        c_new = c5_of(h_new)

        # Acceptance rule
        if c_new <= c_cur or np.random.rand() < np.exp(-(c_new - c_cur) / T):
            cur = h_new
            c_cur = c_new

        # Update global best if improved
        if c_new < c_best:
            best = h_new
            c_best = c_new

    return best, c_best

def line_search(h, l, direction, max_alpha=0.1):
    """Simple backtracking line search"""
    dx = 2.0 / len(h)
    best_c = c5_of(h)
    best_h = h.copy()
    for alpha in [0.01, 0.02, 0.05, 0.1]:
        h_new = h + alpha * direction
        h_new = project_to_feasible(h_new)
        c_new = c5_of(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    return best_h, best_c
