def compute_c5(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    overlap = correlate(h, 1 - h, mode="full")
    c5 = np.max(overlap) * dx
    return c5


def project_h(h, dx):
    """Project h onto feasible set: sum(h) * dx = 1 and 0 <= h <= 1"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def smooth_gradient_descent(h_start, N, num_iters=2000, lr=1e-3):
    """Run smooth gradient descent using projected gradient."""
    h = h_start.copy()
    dx = 2.0 / N
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for i in range(num_iters):
        # Compute gradients numerically using finite differences
        c5 = compute_c5(h)
        # Simple gradient for small fixed shift
        grad = np.zeros(N)
        eps = 1e-5
        # Estimate gradient by perturbations in x directions
        for ip in range(N):
            htmp = h.copy()
            htmp[ip] += eps
            htmp = project_h(htmp, dx)
            c5_plus = compute_c5(htmp)
            grad[ip] = (c5_plus - c5) / eps
        # Step in gradient direction
        h -= lr * grad
        # Project to feasible set
        h = project_h(h, dx)
        # Track best
        c5_new = compute_c5(h)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
    return best_h, best_c5


def move_mass(h, i, j, delta):
    """Move mass from i to j (positive delta moves mass from i to j)"""
    h = h.copy()
    h[i] -= delta
    h[j] += delta
    return h
