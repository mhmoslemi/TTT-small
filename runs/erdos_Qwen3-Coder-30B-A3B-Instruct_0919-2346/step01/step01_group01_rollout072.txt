import numpy as np
import time
import cvxpy as cp

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using full convolution
    overlap = np.correlate(h, 1 - h, mode='full')
    # The maximum value scaled by dx
    c5 = np.max(overlap) * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Preset parameters
    N = 140  # needs even for integer constraint
    dx = 2.0 / N
    
    # Initial C5 bound from previous record
    initial_c5 = 0.381429
    max_c_unscaled = int(np.ceil(initial_c5 * N / 2))
    
    # Best solution found so far
    best_h = None
    best_c5 = float('inf')
    best_c_unscaled = float('inf')
    
    # Keep track of time spent on iterations
    remaining_time = budget_s - (time.time() - start_time)
    
    # Performance margin for fallback if primary method fails
    solver_seconds_per_trial = min(60, remaining_time * 0.15)  # 15% of budget or cap to 60s
    time_left_for_fallback = remaining_time - solver_seconds_per_trial * 10  # conservative
    
    # Set up binary search bounds
    c_low = 0
    c_high = max_c_unscaled
    best_feasible_c = c_high + 1  # To be updated if feasible solution found
    best_x_solution = None
    last_saved_c = c_high + 1
    
    # Use a list to track solver statuses for checking feasibility
    solver_succeeded = False

    # Binary search loop to find minimal feasible C
    while c_low < c_high:
        # Compute mid-point
        c_mid = (c_low + c_high) // 2
        
        # If we are close enough to the target anymore, force exact final step
        if c_high - c_low <= 1:
            c_mid = c_low if c_low == c_high else c_high

        # Try to solve feasibility for this candidate bound
        status = None
        solver_time_start = time.time()
        
        # Build the MILP model
        x = cp.Variable(N, boolean=True)
        y = {}

        # Add constraints for each lag and index pair
        for l in range(1, N):
            y[l] = {}
            for i in range(N - l):
                y[l][i] = cp.Variable(boolean=True)
                
        # Add the McCormick constraints
        constraints = []
        for l in range(1, N):
            for i in range(N - l):
                constraints.append(y[l][i] <= x[i])
                constraints.append(y[l][i] <= x[i + l])
                constraints.append(y[l][i] >= x[i] + x[i + l] - 1)

        # Add sum constraint
        constraints.append(cp.sum(x) == N // 2)

        # Add overlap constraints for the current candidate C_bound
        C_bound = c_mid  # integer
        for l in range(1, N):
            H_l = cp.sum(x[:N - l])  # Sum over valid i indices
            O_l = cp.sum([y[l][i] for i in range(N - l)])
            constraints.append(H_l - O_l <= C_bound)

        # Now define problem
        prob = cp.Problem(cp.Minimize(0), constraints)

        # Store solver time before solving
        try:
            prob.solve(solver=cp.CBC, time_limit=solver_seconds_per_trial, verbose=False)
            status = prob.status
        except:
            try:
                prob.solve(solver=cp.GLPK_MI, time_limit=solver_seconds_per_trial, verbose=False)
                status = prob.status
            except:
                status = "OTHER_ERROR"

        solver_elapsed = time.time() - solver_time_start
        
        # Check if problem was solved feasibility, and update best
        if status in ["OPTIMAL", "FEASIBLE"]:
            # Check feasibility of the solution; store the best recorded
            x_val = np.round(x.value).astype(int)
            if abs(np.sum(x_val) - N // 2) < 1e-6:
                found_c_unscaled = c_mid
                if found_c_unscaled < best_feasible_c:
                    best_feasible_c = found_c_unscaled
                    last_saved_c = found_c_unscaled
                    best_x_solution = x_val.copy()
                c_high = c_mid
                solver_succeeded = True
            else:
                # If sum is off due to numerical error, resolve or treat as infeasible
                c_low = c_mid + 1
        else:
            c_low = c_mid + 1

        # Update remaining time
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time <= 5 or c_low >= c_high:
            break

    # If we found a feasible solution
    if best_x_solution is not None and last_saved_c < float('inf'):
        h_values = best_x_solution.astype(float)
        c_unscaled = last_saved_c
        c5_bound = c_unscaled * dx
        return (h_values, c5_bound, N)
