import numpy as np
import time

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    overlaps = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(overlaps) * dx
    return c5

def full_overlap(h):
    """Return the raw correlation vector (unscaled)."""
    return np.correlate(h, 1 - h, mode="full")

def project_onto_P(v, total):
    """Project v onto the set {x : 0 ≤ x_i ≤ 1, sum x_i = total}."""
    def phi(lam):
        x = np.clip(v - lam, 0, 1)
        return np.sum(x) - total

    # Ensure initial bracketing
    lb = np.min(v) - 1.0
    ub = np.max(v) + 1.0

    while phi(lb) > 0:
        lb -= 1.0
    while phi(ub) < 0:
        ub += 1.0
    assert phi(lb) >= 0 and phi(ub) <= 0

    # Binary search for lambda
    for _ in range(100):
        lam = (lb + ub) / 2
        if phi(lam) > 0:
            lb = lam
        else:
            ub = lam
    lam = (lb + ub) / 2

    x = np.clip(v - lam, 0, 1)
    return x

def smooth_gradient(h, beta):
    """Compute ∇Φ_β(h) where Φ_β = log(sum exp(β F_k))/β."""
    # We compute the logits F_k = (N/2) - sum_i h_i h_{i+k}
    # Use device memory efficiently.
    F = full_overlap(h)
    # Center for numerical stability
    F_centered = F - np.max(F)
    exp_F = np.exp(beta * F_centered)
    w = exp_F / np.sum(exp_F)

    g = np.ones_like(h)
    N = len(h)
    for k in range(-N+1, N):
        wk = w[k + N-1]  # Offset for negative k
        if k >= 0:
            # g[i] -= wk * h[i+k] if i+k < N else 0
            g[:N-k] -= wk * h[k:]
        else:
            # g[i] -= wk * h[i+k] if i+k >= 0 else 0
            g[-k:] -= wk * h[:-k]
    return g

def projected_gradient_descent(h0, time_budget, seed):
    """Run projected gradient descent with β-schedule."""
    np.random.seed(seed)
    start = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of(h)

    # Parameters
    max_iter = 50000
    β_start = 2.0
    β_max = 30.0
    γ_beta = 1.5
    β_update_interval = 1000
    step_start = 0.5
    γ_step = 0.9995
    N = len(h)
    total = N / 2

    β = β_start
    step = step_start

    for i in range(max_iter):
        if time.time() - start > time_budget:
            break
        # Evaluate overlaps and get logits
        F = full_overlap(h)
        # Numerical stability trick: subtract maximum
        F_centered = F - np.max(F)
        exp_F = np.exp(β * F_centered)
        w = exp_F / np.sum(exp_F)

        g = np.ones_like(h)
        for k in range(-N+1, N):
            wk = w[k + N-1]
            if k >= 0:
                g[:N-k] -= wk * h[k:]
            else:
                d = -k
                g[d:] -= wk * h[:-d]
        # Gradient descent step
        v = h - step * g
        h = project_onto_P(v, total)
        # Check improvement
        cur_c5 = c5_of(h)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
        # Update schedule
        step = max(step * γ_step, 1e-6)
        if i % β_update_interval == 0:
            β = min(β * γ_beta, β_max)
    return best_h, best_c5

def targeted_refinement(h, time_budget):
    """Refine by repeatedly transferring mass from worst contributing pair."""
    start = time.time()
    best_h = h.copy()
    best_c5 = c5_of(h)

    max_iter = 20000  # Prevent infinite loop
    alpha = 1.0
    alpha_decay = 0.5  # Multiplier for alpha
    min_alpha = 1e-6

    for iteration in range(max_iter):
        if time.time() - start > time_budget:
            break
        # Compute overlaps
        C = full_overlap(best_h)
        # Find the index of maximum (for full convolution, it's offset)
        max_idx = np.argmax(C)
        d = max_idx - (len(best_h) - 1)  # The actual shift
        N = len(best_h)

        # Extract slices
        if d >= 0:
            hi = best_h[:N-d]
            hj = best_h[d:]
        else:
            d = -d
            hi = best_h[d:]
            hj = best_h[:N-d]

        # Compute the contributions
        c = hi * (1 - hj)
        ci = np.max(c)
        # Indices of the maximum
        i_star = np.argmax(c)
        if d >= 0:
            j_star = i_star + d
        else:
            j_star = i_star + d

        # Compute feasible transfer amount
        delta = min(alpha * ci, best_h[i_star], 1 - best_h[j_star])

        if delta <= 1e-10:
            # No meaningful transfer, try reducing alpha
            alpha *= alpha_decay
            if alpha < min_alpha:
                break
            continue

        # Actually perform move
        h_new = best_h.copy()
        h_new[i_star] -= delta
        h_new[j_star] += delta

        c5_new = c5_of(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
            alpha = 1.0  # reset alpha on improvement
        else:
            # revert
            pass
            # continue with same alpha or lower it
    return best_h, best_c5

        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        remaining_budget -= budget_per_seed

    # Run targeted refinement on best solution
    if best_h is not None and remaining_budget > 10:
        refine_budget = remaining_budget * 0.5
        h_refined, c5_refined = targeted_refinement(best_h, refine_budget)
        if c5_refined < best_c5:

            best_c5 = c5_refined

    if best_h is not None and remaining_budget > 10:
        h_polish = best_h.copy()
        c5_polish = best_c5
        # Small random perturbations + gradient steps
        iter_count = 0
        max_iter = 1000
        while time.time() - start < budget_s and iter_count < max_iter:
            # Perturb with very small move
            pert_i = np.random.randint(0, len(h_polish))
            pert_j = np.random.randint(0, len(h_polish))

            iter_count += 1
        if c5_polish < best_c5:
            best_h = h_polish
            best_c5 = c5_polish

    return (best_h, best_c5, N)
