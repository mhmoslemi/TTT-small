import numpy as np
import time
from scipy.optimize import dual_annealing, minimize
import math

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral via full convolution of h with (1-h)
    overlaps = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum of the full correlation vector
    # scaled by dx
    c5 = np.max(overlaps) * dx
    return c5

def constraint_sum(h):
    """Ensure the integral constraint ∫₀² h(x) dx = 1"""
    dx = 2.0 / len(h)
    return np.sum(h) * dx - 1.0

def propose_move(h, max_step):
    """Propose a feasible move by adjusting two entries."""
    h = h.copy()
    N = len(h)
    for _ in range(100):
        i, j = np.random.randint(0, N, 2)
        if i == j:
            continue
        # Determine feasible δ range
        delta_low = max(-h[i], h[j] - 1)
        delta_high = min(1 - h[i], h[j])
        if delta_low >= delta_high:
            continue
        step = min(max_step, delta_high - delta_low)
        delta = np.random.uniform(
            max(-step, delta_low),
            min(step, delta_high)
        )
        h[i] += delta
        h[j] -= delta
        return h
    return h

def local_search(h, budget_s, seed):
    """Use local search with simulated annealing to improve solution."""
    np.random.seed(seed)
    start = time.time()
    cur = h.copy()
    best = h.copy()
    c_cur = c5_of(cur)
    c_best = c_cur
    T_start = 0.05
    T_end = 1e-6
    step_start = 0.2
    T = T_start
    step = step_start

    while time.time() - start < budget_s:
        elapsed = time.time() - start
        tfrac = elapsed / budget_s
        T = T_start * (T_end / T_start) ** tfrac
        step = step_start * (1 - tfrac)

        # Propose a move
        h_new = propose_move(cur, step)
        c_new = c5_of(h_new)

        # Acceptance rule
        if c_new <= c_cur or np.random.rand() < np.exp(-(c_new - c_cur) / T):
            cur = h_new
            c_cur = c_new

        # Update global best if improved
        if c_new < c_best:
            best = h_new
            c_best = c_new

    return best, c_best

def optimize_with_cma(h_init, budget_s, seed):
    """Optimize using a direct search approach with bounds."""
    np.random.seed(seed)
    start = time.time()
    N = len(h_init)
    dx = 2.0 / N
    
    # Create bounds for variables
    bounds = [(0, 1) for _ in range(N)]
    
    # Objective function
    def objective(h):
        return c5_of(h)
    
    # Constraint function
    def constraint(h):
        return np.sum(h) * dx - 1.0
    
    # Solve
    try:
        result = minimize(objective, h_init, method='L-BFGS-B', 
                         bounds=bounds, constraints={'type': 'eq', 'fun': constraint},
                         options={'maxiter': min(1000, int(1000 * budget_s / 600))},
                         callback=lambda x: time.time() > start + budget_s)
        
        if result.success:
            h_opt = result.x
        else:
            h_opt = h_init
    except:
        h_opt = h_init
        
    c_opt = c5_of(h_opt)
    return h_opt, c_opt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start = time.time()
    N = 140
    dx = 2.0 / N
    
    # Generate different initializations
    seeds = []
    
    # Initial from known good construction
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        if len(h0) != N:
            x0 = np.linspace(0, 2, len(h0))
            x_new = np.linspace(0, 2, N)
            h0 = np.interp(x_new, x0, h0)
        h_initial = h0 * (N / 2) / np.sum(h0)
    else:
        h_initial = np.random.rand(N) * 0.5
        h_initial *= (N / 2) / np.sum(h_initial)
    seeds.append(("initial", h_initial))
    
    # Uniform
    h_uniform = 0.5 * np.ones(N)
    h_uniform *= (N / 2) / np.sum(h_uniform)
    seeds.append(("uniform", h_uniform))
    
    # Double peak construction
    h_double_peak = np.full(N, 0.1)
    a = 0.3
    idxs = np.where((np.linspace(0, 2, N) < a) | (np.linspace(0, 2, N) > (2 - a)))[0]
    h_double_peak[idxs] = 0.9
    h_double_peak *= (N / 2) / np.sum(h_double_peak)
    seeds.append(("double_peak", h_double_peak))
    
    # Gaussian-shaped
    x = np.linspace(0, 2, N)
    mu, sigma = 1.0, 0.2
    h_gaussian = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    h_gaussian *= (N / 2) / np.sum(h_gaussian)
    seeds.append(("gaussian", h_gaussian))
    
    # Run optimization on seeds
    best_h = None
    best_c5 = float('inf')
    
    # Time allocated per seed
    time_per_seed = budget_s * 0.2
    
    for label, h_seed in seeds:
        if time.time() - start > budget_s * 0.9:
            break
            
        # Local search
        h_local, c_local = local_search(h_seed, time_per_seed, seed)
        
        if c_local < best_c5:
            best_h = h_local
            best_c5 = c_local
            
        # More refined search with bounds
        if time.time() - start > budget_s * 0.95:
            break
            
        h_refined, c_refined = optimize_with_cma(h_local, time_per_seed, seed)
        
        if c_refined < best_c5:
            best_h = h_refined
            best_c5 = c_refined

    # Final polishing with hill climbing
    if best_h is not None and time.time() - start < budget_s * 0.95:
        polishing_time = budget_s - (time.time() - start)
        h_polish = best_h.copy()
        c_polish = best_c5
        T = 0.0  # No uphill moves

        while time.time() - start < budget_s and polishing_time > 0:
            h_new = propose_move(h_polish, 0.01)
            c_new = c5_of(h_new)
            if c_new <= c_polish:
                h_polish = h_new
                c_polish = c_new
            polishing_time -= 0.01

        if c_polish < best_c5:
            best_h = h_polish
            best_c5 = c_polish

    return (best_h, best_c5, N)
