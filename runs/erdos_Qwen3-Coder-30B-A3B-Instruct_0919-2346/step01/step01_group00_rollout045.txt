import numpy as np
import time
import random

def project_h(h, target_sum):
    """Project h onto box constraints [0,1] and sum constraint."""
    h = np.clip(h, 0, 1)
    current = np.sum(h)
    if np.isclose(current, target_sum):
        return h
    
    # Binary search for lambda to adjust
    tol = 1e-10
    low, high = -1.0, 1.0
    while high - low > tol:
        mid = (low + high) / 2
        h_test = np.clip(h - mid, 0, 1)
        if np.sum(h_test) > target_sum:
            low = mid
        else:
            high = mid
    
    lambda_val = (low + high) / 2
    h_proj = np.clip(h - lambda_val, 0, 1)
    return h_proj

def compute_C_and_grad(h, alpha):
    """Compute C vector and gradient."""
    N = len(h)
    dx = 2.0 / N
    
    # Compute correlation of h with (1-h) using FFT
    fft_h = np.fft.fft(h, 2*N-1)
    fft_comp = np.fft.fft(1-h, 2*N-1)
    corr = np.fft.ifft(fft_h * np.conj(fft_comp)).real
    
    # C is the overlap integral
    C = 1 - dx * corr[:2*N-1]
    
    # Compute softmax weights
    w = np.exp(alpha * C)
    w = w / np.sum(w)
    
    # Compute convolution
    conv1 = np.convolve(h, w[::-1], mode='same')
    conv2 = np.convolve(h, w, mode='same')
    
    # Gradient w.r.t h
    grad = dx * (1 - (conv1 + conv2))
    
    # Return max C and gradient
    return np.max(C), grad
