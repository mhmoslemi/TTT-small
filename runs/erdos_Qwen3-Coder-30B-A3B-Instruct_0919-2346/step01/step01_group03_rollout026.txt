import numpy as np
import time
from scipy.optimize import differential_evolution

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral via full convolution of h with (1-h)
    overlaps = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum of the full correlation vector
    # scaled by dx
    c5 = np.max(overlaps) * dx
    return c5

def constrained_sum_constraint(x, target_sum):
    """Constraint function for sum of h to equal target_sum"""
    return np.sum(x) - target_sum

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start = time.time()
    N = 140
    dx = 2.0 / N
    target_sum = N / 2  # because sum(h) * dx = 1
    
    # Get the initial construction 
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        if len(h0) != N:
            x0 = np.linspace(0, 2, len(h0))
            x_new = np.linspace(0, 2, N)
            h0 = np.interp(x_new, x0, h0)
        h_start = h0 * (N / 2) / np.sum(h0)
    else:
        h_start = np.random.rand(N) * 0.5
        h_start *= (N / 2) / np.sum(h_start)

    # Use differential evolution to optimize
    bounds = [(0, 1) for _ in range(N)]
    constraints = {'type': 'eq', 'fun': constrained_sum_constraint, 'args': (target_sum,)}

    # Only optimize for a portion of time
    remaining_budget = budget_s * 0.8
    
    # Run a few iterations in a controlled way
    best_h = h_start.copy()
    best_c5 = c5_of(best_h)
    
    # Store intermediate best results
    best_values = [best_c5]
    patience = 0
    patience_limit = 20
    max_iter = 500  # Num iterations for DE
    
    for iteration in range(max_iter):
        if time.time() - start > budget_s * 0.95:
            break
            
        # Use differential evolution with restarts to explore search space
        try:
            result = differential_evolution(
                lambda x: c5_of(x),
                bounds,
                constraints=constraints,
                maxiter=5,
                popsize=15,
                mutation=(0.5, 1),
                recombination=0.7,
                seed=seed + iteration,
                disp=False
            )
            if not result.success:
                continue
                
            # Check if new point is better
            new_h = result.x
            new_c5 = c5_of(new_h)
            
            if new_c5 < best_c5:
                best_h = new_h.copy()
                best_c5 = new_c5
                patience = 0
                best_values.append(best_c5)
            else:
                patience += 1
                
            # Periodically reinitialize to break local optima
            if iteration > 0 and iteration % 50 == 0:
                # Restart from current best with slight perturbation  
                h_perturbed = best_h + np.random.normal(0, 0.01, N)
                h_perturbed = np.clip(h_perturbed, 0, 1)
                h_perturbed *= (N / 2) / np.sum(h_perturbed)
                c_perturbed = c5_of(h_perturbed)
                if c_perturbed < best_c5:
                    best_h = h_perturbed
                    best_c5 = c_perturbed
                    
        except Exception:
            pass  # If DE fails, continue anyway
            
        if patience > patience_limit:
            break
    
    # Final polishing with hill climbing
    if time.time() - start < budget_s * 0.9:
        def propose_move(h, step_size=0.02):
            h = h.copy()
            N = len(h)
            for _ in range(10):  # Try a few times
                i, j = np.random.randint(0, N, 2)
                if i == j:
                    continue
                delta_low = max(-h[i], h[j] - 1)
                delta_high = min(1 - h[i], h[j])
                if delta_low >= delta_high:
                    continue
                delta = np.random.uniform(
                    max(-step_size, delta_low),
                    min(step_size, delta_high)
                )
                h[i] += delta
                h[j] -= delta
                # Ensure constraint still holds (small adjustment if needed)
                h_sum = np.sum(h)
                h *= (N / 2) / h_sum
                return h
            return h

        # Apply hill climbing
        c5_val = c5_of(best_h)
        progress = 0
        while time.time() - start < budget_s:
            h_new = propose_move(best_h)
            c5_new = c5_of(h_new)
            if c5_new <= c5_val:
                best_h = h_new
                c5_val = c5_new
                progress = 0
            else:
                progress += 1
                if progress > 100:  # stop if no progress made
                    break

    return (best_h, c5_val, N)
