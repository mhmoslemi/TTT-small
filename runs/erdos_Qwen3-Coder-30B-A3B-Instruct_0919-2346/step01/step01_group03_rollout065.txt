import numpy as np
import time
from scipy.optimize import differential_evolution

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral via full convolution of h with (1-h)
    overlaps = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum of the full correlation vector
    # scaled by dx
    c5 = np.max(overlaps) * dx
    return c5

def evaluate_solution(h):
    """Evaluate solution for the given h"""
    return c5_of(h)

def project_to_feasible(h):
    """Project h to satisfy constraint ∫₀² h(x) dx = 1"""
    dx = 2.0 / len(h)
    current_sum = np.sum(h) * dx
    if abs(current_sum - 1.0) > 1e-10:
        h = h * (1.0 / current_sum)
    h = np.clip(h, 0, 1)
    return h

def setup_initial_h(N):
    """Set up an initial solution that respects the constraints"""
    if 'initial_h_values' in globals() and len(initial_h_values) == N:
        h = np.array(initial_h_values)
        h = h * (N / 2) / np.sum(h)
    else:
        # Start with a reasonable approximation
        h = np.zeros(N)
        # Create a few peaks in the middle, and some smaller values
        middle_start = N // 4
        middle_end = 3 * N // 4
        h[middle_start:middle_end] = 0.8  # dense region
        # Add some small bumps
        h[0:N//8] = 0.3
        h[7*N//8:N] = 0.3
        
        h = h * (N / 2) / np.sum(h)
    return h

def optimize_with_diff_evolution(N, budget_s):
    """Use differential evolution to find a better solution"""
    # Convert to bounds: [0, 1] for all variables
    bounds = [(0, 1) for _ in range(N)]
    
    def objective(h):
        h = project_to_feasible(h)
        return c5_of(h)
    
    start_time = time.time()
    maxiter = max(100, int(budget_s * 10))
    
    # Ensure the optimization respects the constraint ∫ h dx = 1
    def constraint_func(h):
        dx = 2.0 / len(h)
        return np.sum(h) * dx - 1.0
    
    # Only optimize for about 80% of the budget to allow for final adjustments
    remaining_time = 0.8 * budget_s
    opt_res = differential_evolution(
        objective,
        bounds,
        maxiter=maxiter,
        seed=42,
        disp=False,
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1.0),
        recombination=0.7,
        callback=lambda x, convergence: (time.time() - start_time > remaining_time)
    )
    
    best_h = project_to_feasible(opt_res.x)
    return best_h

def refine_solution(h, budget_s):
    """Use a hill-climbing approach to refine"""
    start_time = time.time()
    dx = 2.0 / len(h)
    
    def get_neighbors(h, i, delta=0.02):
        """Return neighbors for position i"""
        h1 = h.copy()
        h2 = h.copy()
        h1[i] = min(1.0, h[i] + delta)
        h2[i] = max(0.0, h[i] - delta)
        return [h1, h2]
    
    def get_sum_constraint_satisfaction(h):
        """Calculate how close we are to sum = 1"""
        return abs(np.sum(h) * dx - 1.0)
    
    # Ensure start state is feasible
    h = project_to_feasible(h)
    current_score = c5_of(h)
    best_h = h.copy()
    best_score = current_score
    
    # Use a simple gradient-descent-like approach without constraints
    # Because we project to feasible set after every move
    max_iters = int(budget_s * 2000)
    for iter in range(max_iters):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Get a random index
        i = np.random.randint(0, len(h))
        # Try small moves
        deltas = [-0.01, 0.01]
        for delta in deltas:
            if abs(delta) < 0.001:
                continue
            h_trial = h.copy()
            h_trial[i] += delta
            h_trial = project_to_feasible(h_trial)
            trial_score = c5_of(h_trial)
            if trial_score < current_score:
                h = h_trial.copy()
                current_score = trial_score
                if trial_score < best_score:
                    best_h = h.copy()
                    best_score = trial_score
    return best_h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization routine"""
    np.random.seed(seed)
    start = time.time()
    N = 140
    dx = 2.0 / N
    
    # Setup initial solution
    best_h = setup_initial_h(N)
    best_score = c5_of(best_h)
    
    # Try optimization step with differential evolution
    if time.time() - start < 0.7 * budget_s:
        improved_h = optimize_with_diff_evolution(N, budget_s)
        improved_score = c5_of(improved_h)
        if improved_score < best_score:
            best_h = improved_h
            best_score = improved_score
    
    # Optimization refinement
    if time.time() - start < budget_s * 0.95:
        refined_h = refine_solution(best_h, budget_s/2)
        refined_score = c5_of(refined_h)
        if refined_score < best_score:
            best_h = refined_h
            best_score = refined_score
    
    return (best_h, best_score, N)
