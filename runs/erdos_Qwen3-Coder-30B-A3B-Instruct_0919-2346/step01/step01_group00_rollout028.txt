import numpy as np
import time
import random
from scipy.optimize import differential_evolution, minimize

def c5_of(h):
    """Compute C5 = max_k integral h(x)(1-h(x+k)) dx."""
    dx = 2.0 / len(h)
    # Compute correlation: h * (1 - h) with full mode
    c = np.correlate(h, 1 - h, mode="full")
    # Multiply by dx to get continuous integral
    c *= dx
    # Return maximum value among all shifts
    return np.max(c)

def project_feasible(h):
    """Project h to feasible region: [0,1] and sum(h) * dx = 1."""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale so that sum(h) * dx = 1
    dx = 2.0 / len(h)
    h = h * (1.0 / (np.sum(h) * dx))
    return h

def local_search_move(h, max_step=0.05):
    """Propose a small local move to h."""
    h = h.copy()
    N = len(h)
    
    # Pick two indices
    i, j = random.sample(range(N), 2)
    
    # Determine feasible range
    delta_low = max(-h[i], h[j] - 1.0)
    delta_high = min(1.0 - h[i], h[j])
    
    if delta_low >= delta_high:
        return h
    
    # Perturb one index
    delta = random.uniform(max_step * delta_low, max_step * delta_high)
    h[i] += delta
    h[j] -= delta
    
    return project_feasible(h)

        # Apply constraint that sum(h) * dx = 1 (or sum(h) = N/2)

        integral = c5_of(h)
        return integral + penalty
    
    # Create bounds for each variable
    bounds = [(0, 1) for _ in range(N)]
    
    # Use differential evolution to find good starting point
    result = differential_evolution(
        objective, bounds, seed=42, maxiter=int(budget_s/20),

    )
    
    # Improve with local minimization
    refined_result = minimize(
        objective, result.x, method='L-BFGS-B', bounds=bounds, 
        options={'maxiter': int(budget_s/20)}
    )
    
    return project_feasible(refined_result.x), refined_result.fun
