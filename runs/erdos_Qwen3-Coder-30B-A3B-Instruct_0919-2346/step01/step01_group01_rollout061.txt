import numpy as np
import time
import cvxpy as cp

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using full convolution
    overlap = np.correlate(h, 1 - h, mode='full')
    # The maximum value scaled by dx
    c5 = np.max(overlap) * dx
    return c5

    # Initialize MILP
    x = cp.Variable(N, boolean=True)
    C = cp.Variable()
    
    # Constraints
    constraints = [cp.sum(x) == N // 2]
    
    # Build constraints for all lags
    for ell in range(1, N):
        length = N - ell
        if length <= 0:
            continue
            
        # H_ell = sum of h_i over first N-ell entries
        H = cp.sum(x[:length])
        
        # Product variables y_ell[i] = x[i] * x[i+ell]
        y = cp.Variable(length, boolean=True)
        
        # Add constraint: y[i] <= x[i], y[i] <= x[i+ell], y[i] >= x[i] + x[i+ell] - 1
        constraints += [y <= x[:length],
                        y <= x[ell:],
                        y >= x[:length] + x[ell:] - 1]
        
        # O_ell = sum of y_ell[i]
        O = cp.sum(y)
        
        # c_ell = H_ell - O_ell
        c = H - O
        constraints += [c <= C,
                        c >= 0]
    
    # Objective: minimize C
    prob = cp.Problem(cp.Minimize(C), constraints)
    
    # Use CBC solver with time limit
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time <= 1:
        # If not enough time, fallback to previous method
        return run_fallback(seed, budget_s)
        
    # Solve problem
    try:
        prob.solve(solver=cp.CBC, max_seconds=remaining_time - 1, verbose=False)
    except Exception:
        return run_fallback(seed, budget_s)
    
    # Extract solution
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        try:
            x_val = np.array(x.value).astype(float)
            # Ensure it's properly binary
            x_val = np.clip(np.rint(x_val), 0, 1).astype(float)
            # Check integral constraint
            if abs(np.sum(x_val) - N/2) < 1e-6:
                c5_bound = c5_of(x_val)
                return (x_val, c5_bound, N)
        except:
            pass
    
    # Fallback strategy
    return run_fallback(seed, budget_s)
