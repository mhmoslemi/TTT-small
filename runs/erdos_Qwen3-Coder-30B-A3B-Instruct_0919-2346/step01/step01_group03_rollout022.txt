import numpy as np
import time
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral via full convolution of h with (1-h)
    overlaps = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum of the full correlation vector
    # scaled by dx
    c5 = np.max(overlaps) * dx
    return c5

def project_to_feasible(h):
    """Project h to satisfy integral constraint with exact sum"""
    dx = 2.0 / len(h)
    sum_h = np.sum(h)
    target_sum = 1.0
    if sum_h != 0:
        h = h * target_sum / sum_h
    # Ensure bounds [0,1] 
    h = np.clip(h, 0, 1)
    return h

def local_search_step(h, step_size=0.01):
    """Make a small random perturbation to h"""
    h_new = h.copy()
    idx = np.random.randint(0, len(h))
    
    # Try to move index idx
    delta = np.random.uniform(-step_size, step_size)
    h_new[idx] += delta
    
    # Project to feasible set
    h_new = project_to_feasible(h_new)
    return h_new

def trust_region_optimize(initial_h, budget_s, seed):
    """Use trust region optimization to improve solution"""
    np.random.seed(seed)
    start = time.time()
    
    # Initialize solver
    h_best = initial_h.copy()
    c_best = c5_of(h_best)
    
    # Simple gradient-free trust region optimization
    max_trust_radius = 0.1
    min_trust_radius = 1e-6
    shrink_factor = 0.5
    expand_factor = 1.5
    
    trust_radius = max_trust_radius
    
    while time.time() - start < budget_s:
        # Generate several candidates
        candidates = []
        for _ in range(50):
            h_new = local_search_step(h_best, trust_radius)
            candidates.append((h_new, c5_of(h_new)))
            
        # Select best
        candidates.sort(key=lambda x: x[1])
        h_new, c_new = candidates[0]
        
        # Accept if better
        if c_new < c_best:
            h_best = h_new
            c_best = c_new
            trust_radius = min(max_trust_radius, trust_radius * expand_factor)
        else:
            trust_radius *= shrink_factor
            
        if trust_radius < min_trust_radius:
            trust_radius = max_trust_radius
            
        # Occasionally restart
        if time.time() - start > 0.8 * budget_s:
            h_best = initial_h.copy()
            c_best = c5_of(initial_h)
            trust_radius = max_trust_radius
            
    return h_best, c_best
