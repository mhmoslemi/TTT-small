import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

def build_h_from_coeffs(a, n):
    """Build h from cosine coefficients a"""
    h = 0.5 * np.ones(n)
    for m in range(1, len(a) + 1):
        cos_term = np.cos(2 * np.pi * m * np.arange(n) / n)
        h += a[m - 1] * cos_term
    return h

def compute_overlap_and_gradient(a, n, dx, gamma, coeffs_history, max_iter):
    """Compute overlap for all shifts and gradient w.r.t. a"""
    # Construct h
    h = build_h_from_coeffs(a, n)
    
    # Use FFT for correlation
    h_conj = 1 - h
    corr_full = np.correlate(h, h_conj, mode='full')
    # Shift to align with s = 0 at center
    shifts = np.arange(-n+1, n)
    overlap_values = corr_full * dx

    # Compute soft max surrogate
    exp_vals = np.exp(gamma * overlap_values)
    w = exp_vals / np.sum(exp_vals)
    f_soft = np.sum(w * overlap_values)
    
    # Add penalty term for constraint violation
    h_clipped = np.clip(h, 0, 1)
    penalty = 1e4 * np.sum((h - h_clipped)**2)
    f_soft += penalty

    # Compute gradient
    grad = np.zeros_like(a)
    for m in range(1, len(a) + 1):
        cos_term = np.cos(2 * np.pi * m * np.arange(n) / n)
        # First sum: ∑ h_i (1 - h_{i+s}) * cos(2πmi/n)
        conv1 = np.correlate(h, cos_term * (1 - h), mode='full')
        # Second sum: ∑ h_i h_{i+s} cos(2πm(i+s)/n) = ∑ h_i h_{i+s} cos(2πms/n) cos(2πmi/n) - ∑ h_i h_{i+s} sin(2πms/n) sin(2πmi/n)
        cos_shift_term = np.cos(2 * np.pi * m * np.arange(n) / n) 
        conv2 = np.correlate(h, h * cos_shift_term, mode='full')
        # Gradient from first term
        grad[m-1] += dx * np.sum(w * (conv1 - conv2))
        # Gradient from penalty
        grad[m-1] += 2 * 1e4 * np.sum((h - h_clipped) * cos_term)
        
    return f_soft, grad

def project_to_simplex_box(h, target_sum):
    """Project h onto the simplex with sum = target_sum and h ∈ [0,1]^n"""
    h_clipped = np.clip(h, 0, 1)
    s = np.sum(h_clipped)
    if abs(s - target_sum) < 1e-10:
        return h_clipped
    
    # Water-filling method
    # Sort values in descending order
    sorted_h = np.sort(h_clipped)[::-1]
    # Find the threshold
    cumsum = np.cumsum(sorted_h)
    threshold_idx = np.where(sorted_h * np.arange(1, len(sorted_h)+1) - cumsum > target_sum - s)[0]
    if len(threshold_idx) == 0:
        # All entries should be clipped
        tau = 0
    else:
        idx = threshold_idx[-1]  # Last index satisfying condition
        # Compute intersection point
        tau = (target_sum - cumsum[idx-1]) / (idx + 1)
        
    # Apply threshold
    result = np.clip(h_clipped - tau, 0, 1)
    return result

def spectral_optim(n, M, gamma_init=0.5, gamma_final=40, max_iter=5000, seed=42):
    np.random.seed(seed)
    dx = 2.0 / n
    target_sum = n / 2.0
    
    # Start with random spectral coefficients
    a = np.random.normal(0, 0.05, M)
    
    # Homotopy on the number of modes
    for k, M_stage in enumerate([8, 15, 24]):
        if M_stage > M:
            a = np.append(a, np.zeros(M_stage - M))
            M = M_stage
        
        # Adjust gamma to a reasonable start value
        gamma = gamma_init
        gamma_step = (gamma_final / gamma_init) ** (1.0 / 10) # Small aggressive increase
        iteration_counter = 0
        prev_f = np.inf

        # Optimizer settings
        bounds = [(-0.6, 0.6)] * M
        options = {'maxiter': max_iter, 'gtol': 1e-9, 'ftol': 1e-9}
        
        try:
            # Create a function that fixes gamma for scipy
            def func_and_grad(a):
                obj_val, grad = compute_overlap_and_gradient(a, n, dx, gamma, [], max_iter)
                return obj_val, grad
            
            # Use L-BFGS-B with explicit gradient
            result = minimize(
                lambda a: compute_overlap_and_gradient(a, n, dx, gamma, [], max_iter)[0],
                a,
                method='L-BFGS-B',
                jac=lambda a: compute_overlap_and_gradient(a, n, dx, gamma, [], max_iter)[1],
                bounds=bounds,
                options={'maxiter': max_iter, 'gtol': 1e-9, 'ftol': 1e-9}
            )
            
            if result.success:
                a = result.x
            else:
                # continue to next stage even if optimization failed slightly
                pass

        except Exception:
            pass
        
        # Polishing with projected gradient
        # Smooth the coefficients
        h_smooth = build_h_from_coeffs(a, n)
        h_smooth = project_to_simplex_box(h_smooth, target_sum)
        # Compute final overlap with projected h
        corr_full = np.correlate(h_smooth, 1 - h_smooth, mode='full')
        final_overlap = np.max(corr_full) * dx
        
        if final_overlap < 0.3807:
            break

    # Final projection to ensure validity
    h_final = build_h_from_coeffs(a, n)
    h_final = project_to_simplex_box(h_final, target_sum)
    # Final evaluation
    corr_full = np.correlate(h_final, 1 - h_final, mode='full')
    final_overlap = np.max(corr_full) * dx
    
    return h_final, final_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n = 240  # Larger sample size with manageable number of Fourier modes
    best_c5 = np.inf
    best_h = None
    
    # Multiple restarts with different seeds
    restarts = 0
    max_restarts = 6
    while restarts < max_restarts and (time.time() - start_time) < budget_s * 0.95:
        try:
            h, c5 = spectral_optim(n, M=24, gamma_init=0.5, gamma_final=40, max_iter=3000, seed=seed + restarts)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
            restarts += 1
        except Exception:
            restarts += 1
            continue
    
    if best_h is None:
        # Fallback: uniform function
        best_h = np.ones(n)*0.5
        dx = 2.0 / n
        corr_full = np.correlate(best_h, 1 - best_h, mode='full')
        best_c5 = np.max(corr_full) * dx
        
    return (best_h.tolist(), best_c5, n)
