import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value scaled by dx
    max_overlap = np.max(correlation) * dx
    return max_overlap

def build_h_from_coeffs(a, n):
    """Build h from cosine coefficients a, with mean 0.5"""
    h = 0.5 * np.ones(n)
    for m in range(1, len(a) + 1):
        h += a[m-1] * np.cos(2 * np.pi * m * np.arange(n) / n)
    return np.clip(h, 0, 1)

def project_to_simplex_box(h, target_sum):
    """Project h onto the intersection of simplex and box constraints"""
    # Water-filling to adjust mean while respecting [0,1] bounds
    h = np.asarray(h)
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Use bisection to find the correct threshold tau
    lo, hi = -1.0, 1.0
    for _ in range(100):
        tau = (lo + hi) / 2.0
        h_new = np.clip(h - tau, 0, 1)
        if np.sum(h_new) > target_sum:
            lo = tau
        else:
            hi = tau
    return np.clip(h - lo, 0, 1)

def compute_overlap_and_gradient(a, n, gamma, use_penalty=True, lambda_penalty=1e4):
    """Compute soft-max overlap and its gradient w.r.t. a"""
    h = build_h_from_coeffs(a, n)
    dx = 2.0 / n

    # Compute overlaps for all shifts
    corr = np.correlate(h, 1-h, mode='full')
    overlaps = corr * dx
    overlaps = overlaps[n-1:]  # Extract positive shifts first
    overlaps = np.concatenate([overlaps, corr[:n-1]])  # Append negative shifts
    
    # Soft-max surrogate for max overlap
    max_exp = np.exp(gamma * overlaps)
    exp_sum = np.sum(max_exp)
    softmax_weights = max_exp / exp_sum
    soft_max_value = np.log(exp_sum) / gamma
    
    # Penalty term
    penalty = 0.0
    penalty_grad = np.zeros_like(a)
    if use_penalty:
        for i in range(n):
            h_i = h[i]
            # Local penalty for violations
            violation_low = max(0, -h_i)
            violation_high = max(0, h_i - 1.0)
            penalty += lambda_penalty * (violation_low**2 + violation_high**2)
            
            # Gradient of penalty
            penalty_grad += 2 * lambda_penalty * (
                (-violation_low + violation_high) * 
                np.cos(2 * np.pi * np.arange(1, len(a)+1) * i / n)
            )
    
    # Total gradient
    grad = np.zeros_like(a)
    for m in range(1, len(a) + 1):
        arg = 2 * np.pi * m * np.arange(n) / n
        cos_m = np.cos(arg)
        sin_m = np.sin(arg)
        # Compute gradient of 1st part of overlap f_s + derivative of soft-max
        for s, w in enumerate(softmax_weights):
            if s < n - 1:
                shift = s - (n - 1)
                f_s_grad = 0.0
                # Sum over all valid i such that i+shift >= 0 and < n
                valid_i = np.arange(max(0, -shift), min(n, n-shift))
                i = valid_i
                ii = i + shift
                f_s_grad += np.sum(cos_m[i] * (1 - h[ii]))
                f_s_grad -= np.sum(h[i] * cos_m[ii])
                grad[m-1] += w * f_s_grad * dx
                
    return soft_max_value + penalty, grad + penalty_grad

def spectral_optim(n, M, gamma_0=0.5, gamma_final=40.0, gamma_steps=1000, max_iter=5000, 
                   lambda_penalty=1e4, use_early_stopping=True, seed=42):
    """Optimize h in spectral domain with homotopy on gamma and M"""
    np.random.seed(seed)
    # Start with M coefficients
    a = np.random.normal(0, 0.05, M)
    bounds = [(-0.6, 0.6)] * M
    
    # Scaling helpers
    target_sum = n / 2.0
    dx = 2.0 / n
    
    # Increase gamma gradually
    gamma_schedule = np.logspace(np.log10(gamma_0), np.log10(gamma_final), gamma_steps)
    gamma_idx = 0
    
    # Keep track of best solution
    best_h = None
    best_C5 = float('inf')
    
    iterations_since_improvement = 0
    start_time = time.time()
    
    # Configurable method for scioptmin'.  Use L-BFGS-B
    opt_options = {
        'maxiter': max_iter,
        'gtol': 1e-9,
        'ftol': 1e-10
    }
    
    for i in range(max_iter):
        if use_early_stopping and i > 0 and iterations_since_improvement > 300:
            break
        if time.time() - start_time > 0.98 * 600:
            break
            
        # Adjust gamma based on step (perform homotopy over gamma)
        current_gamma = gamma_schedule[min(i // 100, len(gamma_schedule)-1)]
        
        # Get value and gradient
        try:
            func_val, grad = compute_overlap_and_gradient(a, n, current_gamma, lambda_penalty=lambda_penalty)
        except Exception as e:
            break  # Fall back to safer result
            
        a = a - 0.001 * grad
        a = np.clip(a, -0.6, 0.6)
        
        # Check if we're improving
        h = build_h_from_coeffs(a, n)
        h = project_to_simplex_box(h, target_sum)
        
        C5 = evaluate_erdos_solution(h)
        if C5 < best_C5:
            best_C5 = C5
            best_h = h.copy()
            iterations_since_improvement = 0
        else:
            iterations_since_improvement += 1
            
    if best_h is None:
        # Fallback
        h = build_h_from_coeffs(a, n)
        h = project_to_simplex_box(h, target_sum)
        best_h = h
        best_C5 = evaluate_erdos_solution(h)
        
    return best_h, best_C5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n = 240
    best_C5 = float('inf')
    best_h = None
    n_rest = 6  # number of restarts
    
    for restart in range(n_rest):
        if time.time() - start_time > 0.95 * budget_s:
            break
            
        # Run optimization with M=8, 15, 24
        # Mesh refinement in three stages
        M_vals = [8, 15, 24]
        current_a = None
        
        for mx in M_vals:
            if time.time() - start_time > 0.95 * budget_s:
                break
            if current_a is not None:
                # Continue current solution
                a = np.pad(current_a, (0, max(0, mx - len(current_a))), 'constant', constant_values=0)
            else:
                a = np.zeros(mx)
            current_a = a
            
            # Optimize using new M value
            if mx < 15:
                maxit = 1000  # Shorter for early stages
            elif mx < 24:
                maxit = 1500
            else:
                maxit = 2000
            
            h_sol, C5_sol = spectral_optim(n, mx, max_iter=maxit, seed=seed + restart)
            if C5_sol < best_C5:
                best_C5 = C5_sol
                best_h = h_sol.copy()
                
    if best_h is None:
        # Fallback: fixed pattern
        h = np.ones(n) * 0.5
        # Perturb slightly
        noise = np.random.normal(0, 0.05, n)
        h += noise
        h = np.clip(h, 0, 1)
        # Norm to unit integral
        dx = 2.0 / n
        h = h * (1.0 / np.sum(h) * dx)
        
        # Project
        h = project_to_simplex_box(h, n/2)
        best_h = h.copy()
        best_C5 = evaluate_erdos_solution(h)
        
    return (best_h.tolist(), best_C5, n)
