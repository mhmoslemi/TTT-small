import numpy as np
import time
import random
from scipy.fftpack import rfft, irfft

def compute_C_vector(h, dx):
    """Compute vector of overlaps C_τ for all τ."""
    n = len(h)
    # Zero-padded correlation: h * (1-h) with full mode
    h_pad = np.pad(h, (0, n-1), 'constant')
    h_conj_pad = np.pad(1-h, (0, n-1), 'constant')
    corr = np.correlate(h_pad, h_conj_pad, mode='full')
    # Extract the relevant part: τ = 0 to 2n-2, center at n-1
    # But we want all shifts τ from -n+1 to n-1, so:
    # τ = 0..2n-2, shift corresponding to indices n-1..3n-3,
    # but we want that shifted by -n+1 to match [0,1,2,...,2n-2] indexing
    # Is simpler to just note that with mode='full', indices are 0 .. 2n-2
    # the number of correlations is 2n-1, which we can index.
    # We want C_τ = 1 - dx * S_τ(h) where S_τ = sum_i h_i h_{i+τ}
    # The correlation indices from numpy correlate that go from τ = -(n-1) to (n-1)
    # so we want indices: 0 .. 2n-2, corresponding to τ = -n+1 .. n-1.
    # index 0 gives τ = -n+1, index 2n-2 gives τ = n-1.
    # we index correlations from index 0 (τ = -n+1) to 2n-2 (τ = n-1)
    S = corr[0:2*n-1]  # This is now φ[τ], where τ = -n+1 .. n-1
    # So C_τ = 1 - dx * S[τ + n - 1]
    # Take indices as:
    # for τ=0: 0 + n-1 = n-1 (index in S) → S[n-1] = S[0] = sum_i h_i h_{i}=sum h_i^2? No...
    # Wait: let me reconsider:
    # In correlation: numpy correlate(h,x, mode='full') gives:
    # result[k] = sum_i h[i] * x[i+k] where indices wrap
    # Note that we compute correlate(h(x), (1-h)(x+k)) to get
    # result[τ] = sum_i h_i (1 - h_{i+τ}) = sum_i h_i - sum_i h_i h_{i+τ}
    # This is summed over i, and the summation extends to n terms
    # when we pad to length 2n-1.
    # So for shift τ, the correlation index -- care: τ is the offset of h_i vs h_{i+τ}
    # if we get a result index j, then
    # j = n - 1 + τ  =>  τ = j - n + 1
    # So τ = 0 through 2n-2 mean: shift 0 to 2n-2 in the correlator
    # We have to carefully sort out τ values:
    # For τ = -n+1 to n-1:
    # corresponding indices 0 to 2n-2
    # e.g. τ = 0 corresponds to index n-1,
    # So final indexing:
    # index j = τ + n - 1, where τ = -n+1  to  n-1
    # C_τ = 1 - dx * S_τ(h) = 1 - dx * S[τ + n - 1]
    c = 1 - dx * S
    return c

def softmax(w, alpha):
    """Apply softmax with temperature-like parameter α.
    w is a vector of log-weights (specify α > 0)
    W = exp(α w) / sum( exp(α w) )
    Stability trick: subtract max before exp.
    """
    w = np.array(w)
    w_max = np.max(w)
    exp_w = np.exp(alpha * (w - w_max))
    return exp_w / np.sum(exp_w)

def compute_gradient(h, w, dx):
    """Given weights w (should have sum 1), compute ∇ f_α(h) = -dx*(w*h + w_rev*h)"""
    n = len(h)
    # Convolve: w * h with zero padding
    w_pad = np.pad(w, (0, n-1), 'constant')
    h_pad = np.pad(h, (0, n-1), 'constant')
    # Convolve (zero padded)
    conv_right = np.correlate(w_pad, h_pad, mode='full')
    conv_left = np.correlate(h_pad, w_pad[::-1], mode='full')
    # Extract relevant parts (corresponding to indices in h)
    # The full ifft gives indices 0..2n-2, we want τ = 0 to n-1
    # Actually, we want full correlation with w convolution not shifted.
    # Use full convolution and take middle part.
    # Just compute:
    w_rev = w[::-1]
    g = -dx * (np.convolve(h, w, mode='same') + np.convolve(h, w_rev, mode='same'))
    return g

def project_onto_box_mass(h, target_sum):
    """Project onto the set of h in [0,1]^n with sum = target_sum.
    Uses bisection on lagrange multiplier.
    """
    n = len(h)
    h = h.copy()
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-12:
        return h
    # Bisection for λ such that sum_clip(h - λ) = target_sum
    # which means solve: sum_i max(0, h_i - λ) = target_sum
    # Define φ(λ) = sum_i(max(0, h_i - λ)) - target_sum
    # we have a decreasing function of λ.
    # Bounds on λ
    lb = h.min() - 1.0
    ub = h.max() + 1.0
    # Clamp bounds to still be finite
    tol = 1e-12
    while True:
        if ub - lb < tol:
            lambda_val = lb
            break
        lambda_val = (lb + ub) / 2
        s_sum = np.sum(np.maximum(h - lambda_val, 0.0))
        if s_sum > target_sum:
            lb = lambda_val
        else:
            ub = lambda_val
    # Now λ is the bisection solution; only one needed
    new_h = np.maximum(h - lambda_val, 0.0)
    # Renormalize to match target sum exactly (small numerical
    # error may occur from clipping or floating point precision)
    scale = target_sum / (np.sum(new_h) + 1e-16)
    new_h *= scale
    return new_h

def compute_C5(h):
    """Compute the true C5 = max_τ C_τ = max_τ (1 - dx * S_τ)."""
    # This computes directly over discrete values.
    n = len(h)
    dx = 2.0 / n
    # Compute auto-correlations for all τ shifts (from -n+1 to n-1)
    h_pad = np.pad(h, (0, n-1), 'constant')
    h_conj_pad = np.pad(1-h, (0, n-1), 'constant')
    corr = np.correlate(h_pad, h_conj_pad, mode='full')
    S = corr[0:2*n-1]  # Same indexing
    c = 1 - dx * S  # C_τ = 1 - dx * S_τ
    return np.max(c)
