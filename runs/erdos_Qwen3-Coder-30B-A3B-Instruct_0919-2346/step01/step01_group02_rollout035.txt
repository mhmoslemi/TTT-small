import numpy as np
import time
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def c5_of(h):
    """Compute the maximum overlap integral for the given step function h."""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    c = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(c) * dx
    return c5

def project_to_feasible(h, N):
    """Project h onto the set of feasible points: 0 <= h[i] <= 1, sum(h) = N/2"""
    # Ensure h lies in [0,1]
    h = np.clip(h, 0, 1)
    # Adjust to make sum equal to N/2
    current_sum = np.sum(h)
    target_sum = N / 2
    if abs(current_sum - target_sum) < 1e-12:
        return h
    elif current_sum > target_sum:
        # Reduce values from the largest entries to make sum = target_sum
        idx = np.argsort(h)[::-1]  # indices of sorted values descending
        s = 0
        for i in idx:
            if s + (h[i] - 0) <= current_sum - target_sum:
                s += (h[i] - 0)
                h[i] = 0
            else:
                h[i] -= (current_sum - target_sum - s)
                break
    else:
        # Increase values from the smallest entries to make sum = target_sum
        idx = np.argsort(h)
        s = 0
        for i in idx:
            if s + (1 - h[i]) <= target_sum - current_sum:
                s += (1 - h[i])
                h[i] = 1
            else:
                h[i] += (target_sum - current_sum - s)
                break
    return h

def smooth_approx(h, epsilon):
    """Compute smooth surrogate function F_epsilon(h)"""
    dx = 2.0 / len(h)
    c = np.correlate(h, 1 - h, mode='full')
    Cs = c * dx
    # subtract max for numerical stability
    max_c = np.max(Cs)
    exp_terms = np.exp((Cs - max_c) / epsilon)
    sum_exp = np.sum(exp_terms)
    F = epsilon * np.log(sum_exp) + max_c
    return F

def smooth_gradient(h, epsilon):
    """Compute gradient of smooth surrogate function"""
    dx = 2.0 / len(h)
    c = np.correlate(h, 1 - h, mode='full')
    Cs = c * dx
    # subtract max for numerical stability
    max_c = np.max(Cs)
    exp_terms = np.exp((Cs - max_c) / epsilon)
    sum_exp = np.sum(exp_terms)
    weights = exp_terms / sum_exp  # soft-max weights
    
    # build gradient from the weighted contribution of each gradient
    grad = np.zeros_like(h)
    
    # These are the non-zero parts of the derivative for each shift
    # grad[i] = sum_s (weight_s * dx * (1 - h[i+s] - h[i-s]))
    # This is equivalent to a convolution with a kernel
    # We do this efficiently with two shifted vectors
    N = len(h)
    for s in range(-N+1, N):
        w = weights[s + N - 1]
        if abs(w) < 1e-15: continue
        if s == 0:
            # This contributes 1 to all entries
            grad += w * dx * (1 - 2*h)
        else:
            # constructing shift_right and shift_left correctly
            # To match what np.correlate does
            # We exclude border cases where the shift puts it out of bounds
            shift_s = abs(s)
            if s > 0:
                # left shift: h[i] gets contribution -h[i-s]
                temp_grad = np.zeros_like(h)
                temp_grad[shift_s:] = -w * dx * h[:-shift_s]
                grad += temp_grad
                temp_grad = np.zeros_like(h)
                # h[i] also gets contribution -h[i+s]
                temp_grad[:-shift_s] = -w * dx * h[shift_s:]
                grad += temp_grad
            elif s < 0:
                # right shift: h[i] gets contribution -h[i+s]
                temp_grad = np.zeros_like(h)
                temp_grad[:shift_s] = -w * dx * h[-shift_s:]
                grad += temp_grad
                temp_grad = np.zeros_like(h)
                # h[i] also gets contribution -h[i-s]
                temp_grad[-shift_s:] = -w * dx * h[:shift_s]
                grad += temp_grad
    
    return grad

def run_continuation(h0, N, epsilon_sequence, max_iter_lbfgs=150, tol_grad=1e-6):
    """Run continuation on a single starting point"""
    h = h0.copy()
    h = project_to_feasible(h, N)
    
    for epsilon in epsilon_sequence:
        # minimize using L-BFGS-B
        def func(x):
            return smooth_approx(x, epsilon)
        def grad(x):
            return smooth_gradient(x, epsilon)
        
        # propagates back to h from the optimized returned one
        try:
            res = minimize(fun=func, x0=h, method='L-BFGS-B', jac=grad,
                           bounds=[(0, 1)]*N,
                           options={'maxiter': max_iter_lbfgs, 'gtol': tol_grad})
            h = res.x
            if not res.success:
                break
        except:
            pass
        h = project_to_feasible(h, N)
    return h

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    N = 140
    dx = 2.0 / N
    
    # Generate seeds
    seeds = []
    # Uniform seed (satisfying the sum by design)
    h_uniform = np.ones(N) * 0.5
    seeds.append(("uniform", h_uniform))
    
    # Two-peak seeds
    a_values = [0.12, 0.18, 0.24, 0.30]
    for a in a_values:
        h = np.ones(N)
        x_vals = np.linspace(0, 2, N)
        u = 0.9
        v = 0.1
        mask = (x_vals < a) | (x_vals >= 2 - a)
        h[mask] = u
        h[~mask] = v
        h = h * (N/2) / np.sum(h)  # Renormalize exactly to sum=N/2
        seeds.append(("two_peak_{:.2f}".format(a), h))
    
    # Binary seeds
    for i in range(5):
        h = np.zeros(N)
        idxs = np.random.choice(N, size=N//2, replace=False)
        h[idxs] = 1.0
        seeds.append(("binary_{}".format(i), h))
    
    # Symmetric two blocks seed
    h_sym = np.zeros(N)
    mid = N // 2
    h_sym[:mid] = 1.0
    h_sym[mid:] = 0.0
    seeds.append(("symmetric_blocks", h_sym))
    
    # Random continuous seeds
    for i in range(3):
        h = np.random.rand(N)
        h = h * (N/2) / np.sum(h)
        seeds.append(("random_continuous_{}".format(i), h))
    
    # Optimize with continuation
    epsilon_sequence = [0.10, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    best_h = None
    best_c5 = float('inf')
    
    # Split up time among seeds
    time_per_seed = budget_s * 0.7 / len(seeds)
    start = time.time()
    
    for (name, h0) in seeds:
        if time.time() - start > budget_s * 0.9:
            break
        h_opt = run_continuation(h0, N, epsilon_sequence, max_iter_lbfgs=100)
        c5 = c5_of(h_opt)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt
    
    # Final polishing with directional descent on original problem
    if best_h is not None:
        h_final = best_h.copy()
        t_final = time.time() 
        max_iters = 500
        for _ in range(max_iters):
            if time.time() - t_final > budget_s * 0.1:
                break
            # compute all correlations C_s
            c = np.correlate(h_final, 1 - h_final, mode='full')
            Cs = c * dx
            s_best = np.argmax(Cs)
            # compute gradient g for this shift s_best
            g = np.zeros_like(h_final)
            # This just mirrors the derivation above but for one shift s_best
            # We use direct computation for clearer logic:
            # For each i, we calculate the gradient of C_{s_best} w.r.t. h[i]
            N = len(h_final)
            if s_best < N:
                # right shift (as in np.correlate with mode='full'): the values of h[i+s_best] are shifted to i
                g += dx * (-h_final[s_best:])
            # Left shift: 
            # The left part of h[i] receives contribution from h[i - s_best] for i-s_best >= 0
            # So for s=1, shift 1 position right and add delta weights
            g = np.zeros_like(h_final)
            if s_best < N:
                g[:-s_best] += dx * (-h_final[s_best:])
            if s_best >= 0:
                g[s_best:] += dx * (-h_final[:-s_best])
            # Correction for end points
            g[s_best:] += dx * (-h_final[:-s_best])
            g[:s_best] += dx * (-h_final[s_best:])
            # Point contribution
            g += dx * 1
            # Optimize along this direction (simple line search or small step)
            for step_size in [0.05, 0.02, 0.01]:
                h_new = h_final - step_size * g
                h_new = project_to_feasible(h_new, N)
                c5_new = c5_of(h_new)
                if c5_new < best_c5:
                    h_final = h_new
                    best_c5 = c5_new
                    break
            # Early stop if decreased
            if step_size == 0.05:  # didn't continue with smaller steps
                break
        best_h = h_final
    
    return (best_h, best_c5, N)
