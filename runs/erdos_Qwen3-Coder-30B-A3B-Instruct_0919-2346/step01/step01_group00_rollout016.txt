import numpy as np
import time
import random

def compute_C_vector(h):
    """Compute vector of overlaps C_tau for all shifts."""
    dx = 2.0 / len(h)
    # Zero-padded correlation h * (1-h)
    c = np.correlate(h, 1 - h, mode="full")
    c *= dx
    # Correct the constant term
    c += 1
    return c

def compute_softmax_weights(c, alpha):
    """Compute softmax weights for smoothing parameter alpha."""
    c_max = np.max(c)
    # Stable log-sum-exp trick
    exp_c = np.exp(alpha * (c - c_max))
    w = exp_c / np.sum(exp_c)
    return w

def compute_gradient(h, w):
    """Compute gradient of smoothed max: -dx * (w * h + w_rev * h)."""
    dx = 2.0 / len(h)
    # w * h (same length, zero-padded convolution)
    wconv_h = np.convolve(w, h, mode="same")
    # w_rev * h
    w_rev = w[::-1]
    w_rev_conv_h = np.convolve(w_rev, h, mode="same")
    g = -dx * (wconv_h + w_rev_conv_h)
    return g

def project_h(h, target_sum):
    """Project onto box constraints and sum constraint."""
    # Robust projection using bisection root finding for lambda
    h_clip = np.clip(h, 0, 1)
    s = np.sum(h_clip)
    if abs(s - target_sum) < 1e-12:
        return h_clip

    # Bisection on lambda
    l, u = np.min(h_clip) - 1.0, np.max(h_clip) + 1.0
    eps = 1e-12
    
    while u - l > eps:
        lam = (l + u) / 2.0
        x = np.clip(h_clip - lam, 0, 1)
        s_lam = np.sum(x)
        if s_lam > target_sum:
            l = lam
        else:
            u = lam

    # Final projection
    lam = (l + u) / 2.0
    x = np.clip(h_clip - lam, 0, 1)
    return x

    # Double-peak shape
    h_double = np.zeros(n)
    a = 0.3
    u = 0.9
    v = 0.1
    x = np.linspace(0, 2, n)
    mask = (x < a) | (x > (2 - a))
    h_double[mask] = u
    h_double[~mask] = v
    h_double = h_double * (n / 2) / np.sum(h_double)

    initial_seeds = [h_uniform, h_initial, h_double]

    start = time.time()
    best_c5 = float('inf')
    best_h = None

    # Set parameters
    eta_0 = 3.0
    alpha_start = 5.0
    alpha_end = 80.0
    beta = 0.7
    k_perturb = 300
    perturb_std = 0.01

    # Schedule for logging and debugging
    log_every = 500

    # Run optimization on all seeds
    for seed_idx, h_seed in enumerate(initial_seeds):
        h = h_seed.copy()
        c5 = 1.0 - np.min(np.correlate(h, 1 - h, mode="full") * dx)  # Current C5

        # Start optimization
        step_count = 0
        eta = eta_0
        momentum = np.zeros_like(h)
        c5_history = []
        last_improvement = 0
        c5_min = c5

        while time.time() - start < budget_s * 0.95:
            # Update smoothness
            alpha = alpha_start + (alpha_end - alpha_start) * (step_count / 10000.0)
            if alpha > alpha_end:
                alpha = alpha_end

            # Update step size decay
            eta = eta_0 / np.sqrt(1 + step_count)

            # Compute overlaps vector
            c = compute_C_vector(h)

            # Compute softmax weights
            w = compute_softmax_weights(c, alpha)

            # Compute gradient
            g = compute_gradient(h, w)

            # Apply momentum
            momentum = beta * momentum + (1 - beta) * g

            # Take step
            h_new = h - eta * momentum

            # Project
            h_proj = project_h(h_new, target_sum)

            # Evaluate true C5
            c_new = compute_C_vector(h_proj)
            true_c5 = np.max(c_new)
            step_count += 1

            # Store history
            c5_history.append(true_c5)
            if true_c5 < c5_min:
                c5_min = true_c5

            if true_c5 < best_c5:
                best_c5 = true_c5

                h_perturb = h_proj + noise
                h_proj = project_h(h_perturb, target_sum)

            # If no improvement in many steps, reduce learning rate
            if len(c5_history) > 2000 and np.all(np.array(c5_history[-1000:]) >= 1.01 * np.min(c5_history[:-1000])):

            h = h_proj

            # Logging
            if step_count % log_every == 0:
                pass

    # Polishing phase using local search
    if best_h is not None and time.time() - start < budget_s * 0.98:

        cur_h = best_h.copy()
        cur_c5 = best_c5
        t0 = time.time()
        while time.time() - t0 < polishing_budget and time.time() - start < budget_s:
            # Proposal move: swap two values and adjust others to preserve sum
            i, j = random.sample(range(n), 2)
            delta = random.uniform(-0.05, 0.05)
            if cur_h[i] + delta < 0 or cur_h[j] - delta < 0 or cur_h[i] + delta > 1 or cur_h[j] - delta > 1:

            cur_h[i] += delta
            cur_h[j] -= delta

            if new_c5 < cur_c5:
                cur_c5 = new_c5
                best_c5 = new_c5
                best_h = cur_h.copy()
            elif random.random() < 0.05:  # Accept some worse moves
                pass

    return (best_h, best_c5, n)
