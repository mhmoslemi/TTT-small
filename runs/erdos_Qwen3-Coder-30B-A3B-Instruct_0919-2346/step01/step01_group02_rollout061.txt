import numpy as np
import time

def c5_of(h):
    """Compute the maximum overlap integral for the given step function h."""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    c = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(c) * dx
    return c5

def compute_gradient(h, shift):
    """Compute gradient of C5 for the given shift."""
    N = len(h)
    dx = 2.0 / N
    g = np.full(N, 1.0) * dx
    if shift >= 0:
        if shift < N:
            g[:N-shift] -= h[shift:] * dx
            g[shift:] -= h[:N-shift] * dx
    else:
        shift = -shift
        if shift < N:
            g[shift:] -= h[:N-shift] * dx
            g[:N-shift] -= h[shift:] * dx
    return g

def project_onto_bounds_and_sum(v, target_sum):
    """Project vector onto box [0,1] and hyperplane sum = target_sum."""
    # Binary search for lambda
    lower = -np.max(v)
    upper = 1 - np.min(v)
    for _ in range(60):
        mid = (lower + upper) / 2.0
        w = np.clip(v + mid, 0, 1)
        s = np.sum(w)
        if s < target_sum:
            lower = mid
        else:
            upper = mid
    w = np.clip(v + (lower + upper) / 2.0, 0, 1)
    # Correct sum if needed
    diff = target_sum - np.sum(w)
    if abs(diff) > 1e-12:
        free_idx = (w > 0) & (w < 1)
        if np.any(free_idx):
            w[free_idx] += diff / np.sum(free_idx)
            w = np.clip(w, 0, 1)
        else:
            w += diff / len(w)
            w = np.clip(w, 0, 1)
    return w

def run(seed=42, budget_s=600):
    np.random.seed(seed)

    dx = 2.0 / N

    else:

    # Two-peak seed 
    h1 = np.ones(N)
    x_vals = np.linspace(0, 2, N)
    a = 0.3
    u = 0.9
    v = 0.1
    mask = (x_vals < a) | (x_vals >= 2 - a)
    h1[mask] = u
    h1[~mask] = v
    h1 = h1 * (N/2) / np.sum(h1)  # Renormalize

    # Uniform
    h2 = np.ones(N) * 0.5

    # Binary seed
    h3 = np.zeros(N)
    idx = np.random.choice(N, N//2, replace=False)
    h3[idx] = 1.0
    
    # Initialize best
    candidates = [h0, h1, h2, h3]
    best_h = None
    best_c5 = float('inf')
    
    for h in candidates:
        c5 = c5_of(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

    # Iterative improvement
    start = time.time()
    iteration = 0
    
    while time.time() - start < budget_s * 0.95:
        cur_h = best_h.copy()
        cur_c5 = best_c5
        
        # Keep track of recent performance
        recent_improvements = 0
        last_improvement = 0
        
        # Inner loop
        inner_start = time.time()
        inner_iterations = 0
        
        while time.time() - start < budget_s * 0.95 and inner_iterations < 1000:
            # Compute correlation and find worst shift
            corr = np.correlate(cur_h, 1 - cur_h, mode='full')
            max_idx = np.argmax(corr)
            s = max_idx - (N-1)
            
            # Compute gradient
            grad = compute_gradient(cur_h, s)
            
            # Adaptive step-size
            t = min(1.0, (time.time() - start) / budget_s)
            step = 10.0 * (1 - t) + 0.1
            
            # Update
            v = cur_h - step * grad
            cur_h = project_onto_bounds_and_sum(v, N/2)

            new_c5 = c5_of(cur_h)
            
            if new_c5 < cur_c5:

                recent_improvements += 1

        test_c5 = c5_of(best_h)
        if test_c5 < best_c5:
            best_c5 = test_c5

    for _ in range(5000):

            break
        i, j = np.random.choice(N, size=2, replace=False)
        # Determine feasible step
        delta_low = max(-best_h[i], best_h[j] - 1)
        delta_high = min(1 - best_h[i], best_h[j])
        if delta_low >= delta_high:
            continue
        # Sample step size
        step_size = np.random.uniform(-0.02, 0.02)
        step_size = np.clip(step_size, delta_low, delta_high)
        # Create new h
        h_new = best_h.copy()
        h_new[i] += step_size
        h_new[j] -= step_size
        # Renormalize
        h_new = h_new * (N/2) / np.sum(h_new)
        # Evaluate
        c_new = c5_of(h_new)
        if c_new < best_c5:
            best_h = h_new
            best_c5 = c_new
    
    return (best_h, best_c5, N)
