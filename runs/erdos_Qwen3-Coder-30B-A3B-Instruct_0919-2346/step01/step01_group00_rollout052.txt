import numpy as np
import time
import random
from scipy.optimize import differential_evolution

def c5_of(h):
    """Compute C5 = max_k integral h(x)(1-h(x+k)) dx."""
    dx = 2.0 / len(h)
    # Compute correlation: h * (1 - h) with full mode
    c = np.correlate(h, 1 - h, mode="full")
    # Multiply by dx to get continuous integral
    c *= dx
    # Return maximum value among all shifts
    return np.max(c)

def evaluate_h(h):
    """Evaluate the cost function (C5) for given h."""
    return c5_of(h)

def project_feasible(h):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    dx = 2.0 / len(h)
    h = np.clip(h, 0, 1)
    h = h * (1.0 / np.sum(h)) * (1.0 / dx)
    return np.clip(h, 0, 1)

    # Ensure integral constraint
    dx = 2.0 / n_points
    h = h * (1.0 / np.sum(h)) * (1.0 / dx)
    
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    
    return h

def optimize_with_de(time_budget, n_points):
    """Use differential evolution for optimization."""
    start = time.time()
    
    # Use bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return c5_of(h)
    
    # Run differential evolution with constraints
    result = differential_evolution(
        objective,
        bounds,
        maxiter=int(500 * time_budget / 600),
        popsize=10,
        mutation=(0.5, 1),
        recombination=0.7,
        seed=42,
        disp=False
    )
    
    return result.x, result.fun

def genetic_algorithm(h0, time_budget, n_points):
    """Simple genetic algorithm approach."""

        h = h.copy()
        for i in range(len(h)):
            if random.random() < rate:
                delta = random.gauss(0, 0.05)
                h[i] = max(0, min(1, h[i] + delta))
        return h
    
    def crossover(h1, h2):
        # Simple uniform crossover
        mask = np.random.random(len(h1)) > 0.5
        h = np.where(mask, h1, h2)
        return h
    
    def fitness(h):
        # We want to minimize this, so return negative
        return -c5_of(h)
    
    start = time.time()
    
    pop_size = 30
    population = [h0.copy()]
    
    # Initialize population
    for _ in range(pop_size - 1):
        h = mutate(h0.copy(), rate=0.3)
        h = project_feasible(h)
        population.append(h)
    
    generation = 0
    while time.time() - start < time_budget:
        if generation % 10 == 0:
            # Evaluate and sort
            fitnesses = [fitness(h) for h in population]
            sorted_idx = np.argsort(fitnesses)[::-1]  # Descending order
            population = [population[i] for i in sorted_idx[:pop_size//2]]
            
            # Create new generation
            while len(population) < pop_size:
                i1, i2 = random.sample(range(len(population)), 2)
                h = crossover(population[i1], population[i2])
                h = mutate(h)
                h = project_feasible(h)
                population.append(h)
        
        generation += 1
    
    # Return best from final population
    fitnesses = [fitness(h) for h in population]
    best_idx = np.argmax(fitnesses)
    return population[best_idx], -fitnesses[best_idx]
