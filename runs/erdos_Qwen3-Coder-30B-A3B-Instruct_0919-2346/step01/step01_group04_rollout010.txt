def compute_c5(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    overlap = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(overlap) * dx
    return c5


def project_h(h, dx):
    """Project h onto feasible set with sum constraint"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h) * dx
    h = h * (1.0 / current_sum)  # Scale to sum to 1
    return np.clip(h, 0, 1)

def projected_gradient_descent(h0, max_iter=10000, lr=0.02):
    """Run projected gradient descent to minimize C5"""
    h = h0.copy()
    dx = 2.0 / len(h)
    
    for iteration in range(max_iter):
        c5 = compute_c5(h)
        if c5 <= 0.379:
            break
            
        # Find worst shift
        peak_indices = np.correlate(h, 1 - h, mode="full")
        k_best = np.argmax(peak_indices)
        # Adjust padding (example for k=0: index 0 of full correlation)
        # k_best is based on full correlation: shifts 0, 1,...,N-1 of the shift  
        # so if len(h)=N then full correlation has 2*N-1 elements.
        # index k of full correlation is the overlap shifted by k.
        
        # The sum is shifted k where k=0,...,N-1.
        # The returning is full correlation of length 2N-1.  
        # index k in full correl = shift k in [0,N-1] and cares about shift 
        # If k=0, it's shift 0 in which h(0) pairs with 1-h(0).  
        # If we're indexing from 0 to 2N - 1 in the full correl, paying attention to shifts:
        # stride = 1. Let seach be over shifts {0,...,N-1}  
        # shift = k = i - (k + 1) positions? 
        # index t=0,...,2N-2, so shift is j = t - N + 1 where t in 0,2N-2 
        # if t = k + (N-1) corresponds to shift k.
        # So if k index in {0,...,N-1}  
        # compute t = 0,...,2N-2, and find those where t = Shift_k + (N-1) 
        # shift k in full = t = k + N - 1
        
        # Therefore shift k location in full correl is t = k + N - 1  
        # But within N-dimension full-CORREL:
        # shifted by k into full-node.
        # For search over multi-shift, we compute the gradient for final k.
        # Easier to contract over N with all shifts and find max overlap.
        # => shiftk = (the index where the peak appears))
        # Let m = k_best - (N - 1)
        # k_shift = m
        # If k_shift < 0 => invalid (which is should not more be the case). 
        # k_shift can be 0,...
        # index k of shift = ( N - 1 ) + shift_k  (the index which gives h(x) * (1-h(x+k)) for k shift )

        k_shift = k_best - (len(h) - 1)  # Index to where k is
        
        if k_shift < 0:
            k_shift = -k_shift  
        # Now that we know best shift, compute the gradient:
        grad = compute_grad(h, k_shift)
        
        # Gradient step
        h_new = h - lr * grad
        
        # Project
        h_new = project_h(h_new, dx)
        
        # Safety steps
        if np.allclose(h, h_new, atol=1e-8):
            lr *= 0.95  # Reduce learning rate to proceed
        else:
            h = h_new
        
    return h

    # Random continuous (uniform in [0,1])
    seeds.append(np.random.uniform(0, 1, n_points))
    
    # Random binary (0 or 1) and renormalize
    binary_mask = np.random.choice([0, 1], n_points)
    binary_mask = binary_mask * (n_points / 2) / np.sum(binary_mask)
    seeds.append(binary_mask)
    
    return seeds


def run(seed=42, budget_s=600, **kwargs):
    """Main runner"""
    random.seed(seed)
    np.random.seed(seed)
    start = time.time()
    n_points = 200  # Use 200 points for decent resolution
    dx = 2.0 / n_points
    
    # Generate seeds
    seeds = generate_seeds(n_points)
    best_h = None
    best_c5 = float('inf')
    
    # Time allocated for each seed
    allocated_time = budget_s * 0.8 / len(seeds)
    
    for i, seed_h in enumerate(seeds):
        if time.time() - start >= budget_s * 0.9:
            break
        h_opt = projected_gradient_descent(seed_h, max_iter=10000, lr=0.02)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        if best_c5 <= 0.3795:
            break
            
    # Polishing - fine tune with small SA-like pseudo annealing
    # Only accept downhill moves that are safe for time limit
    if time.time() - start < budget_s * 0.8 and best_h is not None:
        cur_h = best_h.copy()
        c_current = compute_c5(cur_h)
        old_c = c_current + 1
        for iteration in range(5000):
            if time.time() - start >= budget_s * 0.95:
                break
                
            # Simple move: random two elements
            i = random.randint(0, n_points-1)
            j = random.randint(0, n_points-1)
            
            h_new = cur_h.copy()
            delta = random.uniform(-0.1, 0.1)
            h_new[i] += delta
            h_new[j] -= delta
            
            # Project
            h_new = project_h(h_new, dx)
            
            c_new = compute_c5(h_new)
            if c_new <= c_current:
                cur_h = h_new
                c_current = c_new
            elif iteration < 1000 and random.random() < 0.1:  # Just accept small downhill but only at start
                cur_h = h_new
                c_current = c_new
        
        if c_current < best_c5:
            best_c5 = c_current
            best_h = cur_h.copy()
    
    return (best_h, best_c5, n_points)
