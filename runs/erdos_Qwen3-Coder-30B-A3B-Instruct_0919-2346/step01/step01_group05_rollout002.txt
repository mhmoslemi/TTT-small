import numpy as np
import time

def softmax_weights(gamma, fs):
    # Compute w_s = exp(γ f_s) / sum_k exp(γ f_k)
    # fs: array of f_s values for all shifts
    exp_gamma_f = np.exp(gamma * fs)
    return exp_gamma_f / np.sum(exp_gamma_f)

def compute_shifted_products(h, s, dx):
    # Compute h_i * (1 - h_{i+s}) for valid indices
    # s: shift (positive or negative)
    n = len(h)
    if s >= 0:
        # Right shift: h_i * (1 - h_{i+s}) for i in [0, n-s-1]
        return h[:n-s] * (1 - h[s:])
    else:
        # Left shift: h_i * (1 - h_{i+s}) for i in [-s, n-1]
        return h[-s:] * (1 - h[:-s])

def gradient_for_shift(h, s, dx):
    # Returns ∇ f_s(h)
    n = len(h)
    grad_s = np.zeros(n)
    if s >= 0:
        # h_i * (1 - h_{i+s}) for i = 0..n-s-1
        grad_s[:n-s] += 1  # for h_i term
        grad_s[s:] -= 1    # for h_{i+s} term
    else:
        # h_i * (1 - h_{i+s}) for i = -s..n-1
        grad_s[-s:] += 1   # for h_i term
        grad_s[:-s] -= 1   # for h_{i+s} term
    return dx * grad_s

def compute_all_gradients(h, dx):
    n = len(h)
    grad = np.zeros(n)
    for s in range(-n+1, n):
        grad_s = gradient_for_shift(h, s, dx)
        grad += grad_s
    return grad

def compute_f_s(h, s, dx):
    # Computes f_s = dx * sum h_i (1 - h_{i+s})
    # This is approximable by dot product but with slicing
    n = len(h)
    if s >= 0:
        # h_i * (1 - h_{i+s}) for i = 0..n-s-1
        return dx * np.sum(h[:n-s] - h[:n-s] * h[s:])
    else:
        # h_i * (1 - h_{i+s}) for i = -s..n-1
        return dx * np.sum(h[-s:] - h[-s:] * h[:-s])

def compute_all_f_s(h, dx):
    n = len(h)
    fs = []
    for s in range(-n+1, n):
        fs.append(compute_f_s(h, s, dx))
    return np.array(fs)

def project_to_simplex(x, target_sum):
    # Euclidean projection onto the simplex {x: sum x = target_sum, x >= 0}
    # Uses the method of Ref. [1] for binary search on lambda
    x = np.asarray(x)
    n = len(x)
    x_sorted = np.sort(x)[::-1]  # descending sort
    # t = 1,2,...,r such that sum_{j=0}^{r-1} (x_sorted[j] - t) >= target or rather
    # sum_{k} x_sorted[k] - sum_{k} min(x_sorted[k],t) >= target
    # which is sum_{k} max(0, x_sorted[k] - t) >= target
    # find least t such that sum_j max(0, x_sorted[j] - t) <= target => 
    # So we want argmin_t: sum_k max(0, x_sorted[k] - t) >= target => 
    # Find t in range [x_sorted[-1], x_sorted[0]] s.t. sum_k max(0, x_sorted[k] - t) <= target.
    
    # Actually: solve sum(max(0, x_sorted[j] - t)) = target
    # See https://math.stackexchange.com/questions/2818343/projection-onto-the-unit-simplex
    # Modified for target_sum != 1: sum_proj = target_sum
    # Let s_k = sum_{i=0}^{k-1} x_sorted[i] 
    # k * t + s_k - sum_{j=0}^{k-1} x_sorted[j]  >= target_sum
    # So t >= (target_sum - s_k) / k = t_k.  Pick smallest k s.t. t_k <= x_sorted[k]
    
    # Done properly:  
    # Define S = sum of elements > t  => S(t) = sum_j max(0, x_sorted[j] - t)
    # Needle in a haystack: We want min t such that S(t) <= target_sum  
    # And let g(t) = sum_j max(0, x_sorted[j] - t)  
    # Use the fact that the sequence of argmax_k S(x_sorted[k]) is decreasing, or that for k < k', 
    # S(x_sorted[k']) - S(x_sorted[k]) = sum_{j such that x_sorted[j] in [x_sorted[k], x_sorted[k'])} (x_sorted[k] - x_sorted[k']) >= 0
    # So it's monotonic, and so the bisection is safe.
    # Finally, let's do it like the standard algorithm:
    # Find largest k for which (target_sum - sum_{j=0}^{k-1} x_sorted[j]) / k >= x_sorted[k]
    # Let lambda = the smallest such value
    # Then result = max(x_i - lambda, 0)
    
    # This is computing how much we should "distribute" to make sum = target_sum
    # r = argmax_k s.t. (target_sum - sum_{j=0}^{k-1} x_sorted[j]) / k >= x_sorted[k]
    # And let lambda = the value of this ratio, and return x - lambda (clipped)
    
    # However we can do more efficiently: rearrange the algorithm
    # Let prefix_sums = x_sorted[0], x_sorted[0]+x_sorted[1], etc.
    # idx = argmax_k such that (target_sum - prefix_sums[k]) >= (k+1) * x_sorted[k]
    # this means (target_sum - prefix_sums[k]) / (k+1) >= x_sorted[k]
    # we want largest such k, then lambda = (target_sum - prefix_sums[k]) / (k+1)
    
    s = 0.0
    for i, x_val in enumerate(x_sorted):
        if (target_sum - s) / (i + 1) >= x_val:
            lambda_val = (target_sum - s) / (i + 1)
            result = np.maximum(x - lambda_val, 0)
            return result
        s += x_val
    # If no such i found, set lambda to (sum - target_sum) / n, then result = x - lambda
    lambda_val = (s - target_sum) / n
    result = np.maximum(x - lambda_val, 0)
    return result

def upsample_initial(h0, n_target):
    # Simple linear interpolation, using NumPy
    n0 = len(h0)
    x0 = np.linspace(0, 2, n0)
    x_target = np.linspace(0, 2, n_target)
    h_target = np.interp(x_target, x0, h0)
    return h_target

def evaluate_erdos_solution(h):
    # Evaluate the C5 constant for a given step function h
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value scaled by dx
    max_overlap = np.max(correlation) * dx
    return max_overlap

    alpha0 = 0.8

    stagnation_count = 0

            break

        true_obj = np.max(fs)
        if true_obj < best_obj:
            best_obj = true_obj
            best_h = h_coarse.copy()
            
        # Update gamma in a linear schedule
        gamma = gamma0 + (gamma_max - gamma0) * t / max_iter_coarse

        w_s = softmax_weights(gamma, fs)

        grad_J = np.zeros_like(h_coarse)

            stagnation_count += 1
        else:

            h_coarse = best_h.copy()

        if stagnation_count > patience and restarts < max_restarts:

            noise = np.random.normal(0, 0.01, n_fine)
            h_fine += noise
            h_fine = np.clip(h_fine, 0, 1)
            h_fine = project_to_simplex(h_fine, n_fine / 2)
            stagnation_count = 0
            restarts += 1
            
    # Final check
    final_h = best_h
    final_c5 = evaluate_erdos_solution(final_h)

    return (final_h.tolist(), final_c5, len(final_h))
