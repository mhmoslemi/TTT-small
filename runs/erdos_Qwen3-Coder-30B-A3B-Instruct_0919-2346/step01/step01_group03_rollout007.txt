import numpy as np
import time
from scipy.optimize import differential_evolution, minimize

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral via full convolution of h with (1-h)
    overlaps = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum of the full correlation vector
    # scaled by dx
    c5 = np.max(overlaps) * dx
    return c5

def project_h_to_feasible(h):
    """Project h to satisfy integral constraint and bounds"""
    h = np.clip(h, 0, 1)
    dx = 2.0 / len(h)
    integral = np.sum(h) * dx
    if integral > 0:
        h = h * (1.0 / integral)
    return h

def optimize_with_scipy(h0, budget_s):
    """Use scipy optimization to refine the solution"""
    n = len(h0)
    dx = 2.0 / n
    
    def objective(h):
        # Since we're maximizing the overlap, we minimize its negative
        return -c5_of(h)
    
    # Constraint that integral equals 1
    def constraint_integral(h):
        return np.sum(h) * dx - 1.0
    
    # Bounds for each variable [0, 1]
    bounds = [(0, 1) for _ in range(n)]
    
    # Add constraints
    constraints = {'type': 'eq', 'fun': constraint_integral}
    
    # Use differential evolution for global optimization
    result = differential_evolution(
        objective, bounds, 
        constraints=constraints,
        maxiter=int(budget_s * 5),  # More iterations for more time
        seed=42,
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.7,
        atol=1e-6,
        rtol=1e-6
    )
    
    return result.x

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

    N = 140

    else:
        # Create a simple starting point
        h0 = np.ones(N) * 0.5
    
    # Ensure it's feasible
    h0 = project_h_to_feasible(h0)
    
    # Optimize with scipy
    best_h = optimize_with_scipy(h0, budget_s)
    best_c5 = c5_of(best_h)
    
    # Final polishing
    polishing_time = max(0, budget_s - (time.time() - start))
    
    if polishing_time > 0.1:
        # Do local search from the scipy result
        def local_optimization(h_start):
            h = h_start.copy()
            dx = 2.0 / len(h)
            
            # Very simple local search
            for _ in range(5000):
                if time.time() - start > budget_s * 0.95:
                    break
                i = np.random.randint(0, len(h))
                neighbor = h.copy()
                delta = np.random.uniform(-0.05, 0.05)
                neighbor[i] = np.clip(h[i] + delta, 0, 1)
                
                # Adjust others to maintain integral
                old_sum = np.sum(h) * dx
                new_sum = np.sum(neighbor) * dx
                if new_sum > 0:
                    scale_factor = old_sum / new_sum
                    for j in range(len(neighbor)):
                        if j != i:
