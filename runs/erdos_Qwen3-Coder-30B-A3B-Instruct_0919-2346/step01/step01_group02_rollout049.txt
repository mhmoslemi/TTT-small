```python
import numpy as np
import time
import cvxpy as cp
from cvxpy import ECOS, GLPK, CBC
from scipy.optimize import minimize_scalar

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start = time.time()
    N = 140
    dx = 2.0 / N
    max_time = budget_s
    
    # --- Phase 0: Build index sets ---
    # Pairs of (i, s) such that 0 <= i <= N-1, 0 <= i+s <= N-1
    P = []
    shift_map = {}  # (i,s) -> index in y vector
    y_idx = 0
    for s in range(-N+1, N):  # All possible shifts 
        for i in range(max(0, -s), min(N, N - s)):
            P.append((i, s))
            shift_map[(i, s)] = y_idx
            y_idx += 1
    num_y = len(P)

    # --- Phase 1: Solve LP relaxation ---
    
    # Variables
    h = cp.Variable(N)
    y = cp.Variable(num_y)
    alpha = cp.Variable()
    
    # Constraints
    constraints = [
        0 <= h, h <= 1,
        cp.sum(h) == N/2,
        alpha >= 0
    ]
    
    # Add McCormick constraints for each (i,s) pair
    for idx, (i, s) in enumerate(P):
        # y[i,s] <= h[i]
        constraints.append(y[idx] <= h[i])
        # y[i,s] <= h[i+s]
        constraints.append(y[idx] <= h[i+s])
        # y[i,s] >= h[i] + h[i+s] - 1
        constraints.append(y[idx] >= h[i] + h[i+s] - 1)
        # y[i,s] >= 0
        constraints.append(y[idx] >= 0)
        
    # Add α >= dx * (N/2 - sum_y[s]) for all shifts s.
    # But since y for each shift needs to be grouped:
    # For shift s, the indices of overlapping rows in h are i from 0 to N-1-s (if s>=0) or from 0 to N-1+s (if s<0)
    # So for s >= 0, we get indices i in [0, N-1-s]
    # For s < 0, we get indices i in [0, N-1+s] i.e. [N-1+s+1, N-1]
    
    # We want to compute sum_y_s = sum_{i: 0 <= i <= N-1, 0 <= i+s <= N-1} y[i,s]
    # which equals sum_{i: 0 <= i <= N-1-s} y[i,s] for s>=0
    # and sum_{i: -s <= i <= N-1} y[i,s] for s<0
    
    # To make it generic:
    # Find all (i,s) pairs from P where s equals a given shift t
    # And compute sum of y[i,s] for that shift
    
    from collections import defaultdict
    shift_sum_map = defaultdict(list)
    for i, s in P:
        shift_sum_map[s].append(shift_map[(i, s)])
    
    for s, y_indices in shift_sum_map.items():
        # The real sum is over all elements of indices in y
        sum_y_s = cp.sum(y[y_indices])
        # Apply constraint: alpha >= dx * (N/2 - sum_y_s)
        constraints.append(alpha >= dx * (N/2 - sum_y_s))

    # Objective
    objective = cp.Minimize(alpha)
    
    prob = cp.Problem(objective, constraints)
    
    # Solve LP
    try:
        prob.solve(solver=ECOS, max_iters=20000, verbose=False)
        if prob.status in ["optimal", "optimal_inaccurate"]:
            h_val = h.value
            alpha_val = alpha.value
            print(f"LP solution gives α={alpha_val}, expecting actual C5 around it.")
        else:
            print(f"LP status: {prob.status}")
            # Fall back to uniform
            h_val = np.ones(N) * 0.5
            h_val = h_val * (N/2) / np.sum(h_val)  # Ensure sum == N/2
            alpha_val = float('inf')
    except Exception as e:
        print(f"LP solver error: {e}")
        h_val = np.ones(N) * 0.5
        h_val = h_val * (N/2) / np.sum(h_val)  # Ensure sum == N/2
        alpha_val = float('inf')

    # --- Phase 2: Check if LP is sufficient ---
    def c5_of(h_arr):
        c = np.correlate(h_arr, 1 - h_arr, mode='full')
        return np.max(c) * dx

    # Compute true C5 from LP
    if h_val is not None and abs(np.sum(h_val) - N/2) < 1e-6:
        h_lp = h_val.copy()
        h_lp = np.clip(h_lp, 0, 1)
        c5_lp = c5_of(h_lp)
        print(f"LP soln C5 = {c5_lp}")
        if c5_lp <= 0.3807:
            return (h_lp, c5_lp, N)

    # --- Phase 3: Binary MILP with CBC on a potentially smaller N ---
    # If we don't have enough time, we can just make some quick binary solution
    if time.time() - start > 0.9 * max_time:
        print("Not enough time for MILP.")
        h_final = h_lp.copy()
        c5_final = c5_lp
        return (h_final, c5_final, N)
    
    # Check time remaining
    time_remaining = max_time - (time.time() - start)
    
    # Test smaller N to reduce computational cost; but still use full N in final evaluation if possible.
    # We'll use N=100 for MILP if needed.
    Nsmall = 100 if N > 100 else N
    dsmall = 2.0 / Nsmall
    
    # Quick copy with different number of samples for MILP
    # We use the same constraint structure but reduce size 
    h_milp_size = int(Nsmall)

    # For MILP, we recompute index sets and build new ones of sufficient size
    P_milp = []
    shift_map_milp = {}
    y_idx_milp = 0
    for s in range(-Nsmall+1, Nsmall):  # All possible shifts 
        for i in range(max(0, -s), min(Nsmall, Nsmall - s)):
            P_milp.append((i, s))
            shift_map_milp[(i, s)] = y_idx_milp
            y_idx_milp += 1
    num_y_milp = len(P_milp)
    
    # Solve MILP
    h_milp = cp.Variable(h_milp_size, integer=True)
    y_milp = cp.Variable(num_y_milp)
    alpha_milp = cp.Variable()
    
    constraints_milp = [
        0 <= h_milp, h_milp <= 1,
        cp.sum(h_milp) == Nsmall/2,
        # Integer constraint added
        h_milp >= 0,
        h_milp <= 1,
        alpha_milp >= 0
    ]
    
    # Add McCormick constraints for each (i,s) pair
    for idx, (i, s) in enumerate(P_milp):
        constraints_milp.append(y_milp[idx] <= h_milp[i])
        constraints_milp.append(y_milp[idx] <= h_milp[i+s])
        constraints_milp.append(y_milp[idx] >= h_milp[i] + h_milp[i+s] - 1)
        constraints_milp.append(y_milp[idx] >= 0)
        
    # Add alpha constraints
    shift_sum_map_milp = defaultdict(list)
    for i, s in P_milp:
        shift_sum_map_milp[s].append(shift_map_milp[(i, s)])
    
    for s, y_indices in shift_sum_map_milp.items():
        sum_y_s = cp.sum(y_milp[y_indices])
        constraints_milp.append(alpha_milp >= dsmall * (Nsmall/2 - sum_y_s))

    objective_milp = cp.Minimize(alpha_milp)
    prob_milp = cp.Problem(objective_milp, constraints_milp)
    
    # Time-limited solve
    time_bud = time_remaining * 0.9
    try:
        # Since cvxpy doesn’t allow solvers to use thread limits,
        # and CBC has maxTime option
        # We just call solve() and assume solver respects time limits via internal/threading
        prob_milp.solve(solver=CBC, verbose=False, maxTime=time_bud)
        if prob_milp.status in ["optimal", "optimal_inaccurate"]:
            h_milp_sol = h_milp.value.copy()
            h_milp_sol = np.clip(h_milp_sol, 0, 1)
            h_milp_sol = h_milp_sol * (Nsmall/2) / np.sum(h_milp_sol) if np.sum(h_milp_sol) > 0 else h_milp_sol
            # Interpolate re-sized solution back to N=140
            h_milp_big = np.interp(np.linspace(0, 2, N),
                                   np.linspace(0, 2, Nsmall),
                                   h_milp_sol)
            h_milp_big = h_milp_big * (N/2) / np.sum(h_milp_big)
            c5_milp = c5_of(h_milp_big)
            print(f"MILP soln C5 = {c5_milp}")
            if c5_milp <= 0.3807:
                return (h_milp_big, c5_milp, N)
        else:
            print(f"MILP status: {prob_milp.status}")
    except Exception as e:
        print(f"MILP solver error: {e}")
        h_milp_big = h_lp.copy()
        c5_milp = c5_lp
        
    # --- Phase 4: Local refinement ---
    def proj_sum(h_vals):
        # Clip to [0,1], then adjust to sum to N/2 via water filling
        h_proj = h_vals.copy()
        h_proj = np.clip(h_proj, 0, 1)
        target_sum = N / 2.0
        current_sum = np.sum(h_proj)
        if abs(current_sum - target_sum) < 1e-10:
            return h_proj
        
        # Simple projection onto the sum constraint
        diff = (target_sum - current_sum) / N  # approximate equal adjustments
        return np.clip(h_proj + diff, 0, 1)
    
    def proj_box(h_vals):
        return np.clip(h_vals, 0, 1)
    
    def project_h_to_sum(h_vals, target_sum):
        # Water-filling like approach to preserve order
        h_proj = np.copy(h_vals)
        h_proj = np.clip(h_proj, 0, 1)
        if abs(np.sum(h_proj) - target_sum) < 1e-10:
            return h_proj
        # Sort to move things in order
        sorted_idx = np.argsort(h_proj) 
        h_sorted = h_proj[sorted_idx]
        # Set values, adjust from small to large until target met
        current_sum = np.sum(h_sorted)
        if current_sum < target_sum:
            # Add to largest first
            for k in range(N-1, -1, -1):  # from high to low index
                if current_sum >= target_sum:
                    break
                delta = min(target_sum - current_sum, 1 - h_sorted[k])
                h_sorted[k] += delta
                current_sum += delta
        else:
            # Subtract from largest first
            for k in range(N-1, -1, -1):
                if current_sum <= target_sum:
                    break
                delta = min(current_sum - target_sum, h_sorted[k])
                h_sorted[k] -= delta
                current_sum -= delta
        h_proj[sorted_idx] = h_sorted
        return h_proj
    
    # --- Optimize from the best known feasible solution ---
    h_refine = h_milp_big.copy()
    c5_refined = c5_milp
    
    # Hill climb for 1000 steps or until no improvement
    last_c5 = c5_refined
    counter = 0
    for step in range(1000):
        if time.time() - start > 0.95 * max_time:
            break
        # Compute current c5 values for each shift
        c_vals = np.array([np.sum(h_refine[i:] * (1 - h_refine[:len(h_refine) - i])) * dx for i in range(N)])
        worst_idx = np.argmax(c_vals)
        
        if worst_idx == 0:
            # Easy case (no need for shift)
            # Compute gradient via difference at endpoints
            # Only change the difference, think of change to optimize via finite difference by looking at variation
            # This is coarse gradient estimate.
            grad = np.zeros_like(h_refine)
            for i in range(N):
                if i+1 < N:
                    grad[i] += (h_refine[i+1] - h_refine[i]) * dx
                if i > 0:
                    grad[i] += (h_refine[i-1] - h_refine[i]) * dx
            # Simple view of it's influence on C_s
            # For detecting poor shifts and making them better
            epsilon = 0.002
            h_new = proj_box(h_refine - epsilon * grad)
            h_new = project_h_to_sum(h_new, N/2)
            c5_new = c5_of(h_new)  
            if c5_new <= c5_refined:
                h_refine = h_new
                c5_refined = c5_new
            last_c5 = c5_new
            counter = 0
        else:
            # Make small change and evaluate - let's just do greedy drop at worst shift
            epsilon = 0.005
            # Try model similar change to what done before
            delta_h = np.zeros_like(h_refine)
            # Trick: drop mass at some indices
            drop_idx = (np.random.random_integers(1, N-2))  # Avoid edges
            # Add mass to either side of the drop
            if drop_idx > 1 and drop_idx < N-2:
                i1 = drop_idx-1; i2 = drop_idx+1
                # Change at post. potential project update
                delta_h[i1] += epsilon
                delta_h[i2] -= epsilon
                # Also modify central variable 
                delta_h[drop_idx] -= epsilon * 2
            else:
                i1 = drop_idx
                delta_h[i1] -= epsilon
                delta_h[(i1+1) % N] += epsilon/2
            
            h_new = proj_box(h_refine + delta_h)
            h_new = project_h_to_sum(h_new, N/2)
            c5_new = c5_of(h_new)
            if c5_new <= c5_refined:
                h_refine = h_new
                c5_refined = c5_new
            counter = 0
        if abs(c5_refined - last_c5) < 1e-8:
            counter += 1
        if counter > 20:
            break

    if c5_refined <= 0.3807:
        return (h_refine, c5_refined, N)

    # Final fallback
    return (h_refine, c5_refined, N)
```
