def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value scaled by dx
    max_overlap = np.max(correlation) * dx
    return max_overlap

def project_onto_bounds_sum(v, target_sum, lower=0, upper=1):
    """
    Project vector v onto the feasible set: lower <= v[i] <= upper, sum(v) = target_sum
    Using binary search for the projection threshold
    """
    v = np.array(v)
    lo = v.min() - upper
    hi = v.max() - lower
    
    def sum_clipped(lam):
        return np.clip(v - lam, lower, upper).sum()
        
    # Binary search for the right lambda
    for _ in range(60):
        mid = (lo + hi) / 2
        s = sum_clipped(mid)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    
    return np.clip(v - (lo + hi)/2, lower, upper)

def compute_gradient(h, correlation, weights, dx):
    """Compute gradient of weighted sum of overlaps"""
    n = len(h)
    grad = np.zeros(n)
    
    # The correlation is computed with mode='full', giving 2n-1 elements
    # Index in correlation = lag + (n-1)
    # So for lag l, index = l + (n-1)
    
    # Compute weighted gradient
    for l in range(-(n-1), n):
        if l >= 0:
            # Forward shift: h[i] * (1 - h[i+l])
            # gradient w.r.t. h[i] gets: (1 - h[i+l]) if i+l < n
            # gradient w.r.t. h[i+l] gets: -h[i] if i >= 0
            w = weights[l + (n-1)]
            if l == 0:
                # Only the first term applies (i and i+l same)
                for i in range(n):
                    grad[i] += w * (1 - h[i])
            else:
                # Contribution from first term
                for i in range(n-l):
                    grad[i] += w * (1 - h[i+l])
                # Contribution from second term (negative)
                for i in range(l, n):
                    grad[i] -= w * h[i-l]
        else:
            # Backward shift: h[i] * (1 - h[i+l])
            # gradient w.r.t. h[i] gets: (1 - h[i+l]) if i+l >= 0
            # gradient w.r.t. h[i+l] gets: -h[i] if i+l < n
            w = weights[l + (n-1)]
            # Contribution from first term
            for i in range(-l, n):
                grad[i] += w * (1 - h[i+l])
            # Contribution from second term (negative)
            for i in range(n+l):
                grad[i] -= w * h[i-l]
    
    return grad * dx
