import numpy as np
import time
import random

def compute_autocorrelation(h):
    """Compute the autocorrelation of h with zero padding."""
    n = len(h)
    # Use FFT for efficiency
    H = np.fft.rfft(h, 2*n-1)
    autocorr = np.fft.irfft(H * np.conj(H), 2*n-1)
    return autocorr

def compute_C_vector(h, dx):
    """Compute vector of overlaps C_τ = 1 - dx * S_τ(h) for all shifts."""
    S = compute_autocorrelation(h)
    return 1 - dx * S

def compute_softmax_weights(c, alpha):
    """Compute softmax weights w_τ = exp(α * c_τ) / sum(exp(α * c_σ))."""
    # Center for numerical stability
    c_max = np.max(c)
    exp_c = np.exp(alpha * (c - c_max))
    return exp_c / np.sum(exp_c)

def compute_gradient(h, w, dx):
    """Compute gradient of smoothed objective: g_i = -dx * sum(w_τ * (h_{i+τ} + h_{i-τ}))."""
    n = len(h)
    # For convolution, we want to treat indices as periodic with wrap-around
    # but for shift 0, we don't want to wrap
    # hence let's use `mode='same'` on a zero-padded version?
    # Actually, firing up FFT for it is cleaner
    w_rev = w[::-1]  # Reverse
    
    # Convolve with h
    g1 = np.convolve(w, h, mode='same')
    g2 = np.convolve(w_rev, h, mode='same')
    
    # The convolution is right-aligned in the middle, so we need to double-check indices
    # Also, since the shifts are from `-(n-1)` to `n-1` and we're using zero padding,
    # the base convolution works correctly for our needs.
    
    return -dx * (g1 + g2)

def project_onto_feasible(h, target_sum):
    """Project h onto box constraints [0,1] and sum = target_sum."""
    # First, clip
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    
    if abs(current_sum - target_sum) < 1e-12:
        return h
    
    # Water-filling algorithm to project onto sum constraint
    # Algorithm: sort entries in descending order.
    # Find threshold t such that sum of (h_i - t)_+ = target_sum
    # Then clip all entries at t.
    
    h_sorted_desc = np.sort(h)[::-1]
    cumulative_sum = np.cumsum(h_sorted_desc)
    
    # Find where to place threshold
    # cumulative_sum[i] - (i+1) * t = target_sum
    # t = (cumulative_sum[i] - target_sum) / (i+1)
    
    # Define the boundary: where sum of clipped values is under target
    pos = len(h_sorted_desc)
    for i in range(len(h_sorted_desc)):
        if cumulative_sum[i] - (i + 1) * h_sorted_desc[i] < target_sum:
            pass  # continue
        else:
            pos = i
            break
    
    # a = cumulative_sum[pos] - target_sum
    a = cumulative_sum[pos] - target_sum
    
    if pos >= len(h_sorted_desc):
        # All are negatives after threshold
        threshold = h_sorted_desc[pos-1] - a / (len(h_sorted_desc))
    elif a == 0:
        threshold = h_sorted_desc[pos - 1] if pos > 0 else 0
    else:
        threshold = a / (pos + 1)
    
    # Final projection
    h_proj = np.clip(h - threshold, 0, 1)
    
    # Ensure sum <= target_sum - enforce final sum
    h_proj_scaled = h_proj * (target_sum / np.sum(h_proj))
    return h_proj_scaled

def compute_C5(h):
    """Compute the actual C5 (true max overlap) without smoothing."""
    n = len(h)
    dx = 2.0 / n
    c = compute_C_vector(h, dx)
    return np.max(c)

    np.random.seed(seed)
    start_time = time.time()

    n = 140
    dx = 2.0 / n
    target_sum = n / 2

    # 1. Uniform
    h_uniform = np.ones(n) * 0.5

    h_initial = h_initial * (target_sum / np.sum(h_initial))
    
    # 3. Double-peak
    h_double = np.zeros(n)
    a = 0.3
    u = 0.9
    v = 0.1
    x = np.linspace(0, 2, n)
    mask = (x < a) | (x > (2 - a))
    h_double[mask] = u
    h_double[~mask] = v
    h_double = h_double * (target_sum / np.sum(h_double))
    
    seeds = [h_uniform, h_initial, h_double]
    
    best_h = None
    best_c5 = float('inf')

    # Gradient based optimization
    # Budget allocation: 80% for gradient descent, 15% for polishing, 5% for initialization
    
    gradient_time = 0.80 * budget_s
    polishing_time = 0.15 * budget_s
    
    # Run multiple gradient descent runs from seeds
    for seed_idx, h_seed in enumerate(seeds):
        if time.time() - start_time > gradient_time:
            break
        
        # Initialize parameters
        h = h_seed.copy()
        h = project_onto_feasible(h, target_sum)
        
        # Iteration parameters
        eta0 = 3.0  # Initial step size (> 0) 
        max_iter = 10000
        alpha_start = 5.0
        alpha_end = 80.0
        momentum_decay = 0.7
        perturb_freq = 300
        noise_std = 0.01
        alpha_t = alpha_start
        eta_t = eta0
        m = np.zeros_like(h)

        iteration = 0
        last_improvement = 0
        no_improvement_counter = 0

        while iteration < max_iter and time.time() - start_time < gradient_time:
            c = compute_C_vector(h, dx)
            alpha_t = alpha_start + (alpha_end - alpha_start) * (iteration / max_iter)
            
            w = compute_softmax_weights(c, alpha_t)
            g = compute_gradient(h, w, dx)

            # Momentum
            m = momentum_decay * m + (1 - momentum_decay) * g
            # Step with decaying eta
            eta_t = eta0 / np.sqrt(1 + iteration)
            
            h_new = h - eta_t * m

            # Project back onto feasible set
            h_new = project_onto_feasible(h_new, target_sum)

            h = h_new
            
            # Evaluate true C5
            c5_new = compute_C5(h)
            
            # Update best solution
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                last_improvement = iteration
                no_improvement_counter = 0
            else:
                no_improvement_counter += 1

            # Periodic perturbation to escape saddle points
            if iteration % perturb_freq == 0 and iteration > 0:
                noise = np.random.normal(0.0, noise_std, n)
                h = h + noise
                h = project_onto_feasible(h, target_sum)

            iteration += 1

    # Polishing stage
    if best_h is None:
        best_h = h_uniform
        best_c5 = compute_C5(best_h)

    polishing_timer = time.time()
    while time.time() - polishing_timer < polishing_time:
        # Simple pairwise move as in previous codes
        h_new = best_h.copy()
        idx_i, idx_j = random.sample(range(n), 2)
        delta_low = max(-h_new[idx_i], h_new[idx_j] - 1.0)
        delta_high = min(1.0 - h_new[idx_i], h_new[idx_j])
        
        if delta_low < delta_high:
            step = 0.02
            delta = random.uniform(-step, step)
            delta = max(delta_low, min(delta, delta_high))
            h_new[idx_i] += delta
            h_new[idx_j] -= delta
            
            # Re-project
            h_new = project_onto_feasible(h_new, target_sum)
            c5_new = compute_C5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    return (best_h, best_c5, n)
