import numpy as np
import time
import random
from scipy.optimize import differential_evolution, minimize


def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    overlap = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(overlap) * dx
    return c5


def propose_move(h, max_step):
    """Propose a new feasible h vector by adjusting two elements"""
    h_new = h.copy()
    N = len(h)
    
    # Try to find a valid move
    max_tries = 100
    for _ in range(max_tries):
        i = random.randint(0, N - 1)
        j = random.randint(0, N - 1)
        if i == j:
            continue
            
        # Determine feasible interval for δ
        delta_low = max(-h_new[i], h_new[j] - 1)
        delta_high = min(1 - h_new[i], h_new[j])
        
        if delta_low >= delta_high:
            continue
            
        # Sample δ from uniform distribution over feasible range
        step = min(max_step, delta_high - delta_low)
        if step <= 0:
            continue
        delta = random.uniform(delta_low, delta_high)
        
        # Apply move
        h_new[i] += delta
        h_new[j] -= delta
        
        return h_new
    
    # If no valid move found, return original
    return h.copy()


def anneal(initial_h, time_budget_s):
    """Run simulated annealing on the initial vector"""
    start = time.time()
    
    # Parameters
    T_start = 0.05
    T_end = 1e-6
    step_start = 0.2
    
    # Set up the current state
    cur = initial_h.copy()
    c_cur = c5_of(cur)
    
    best = cur.copy()
    c_best = c_cur
    
    # Main loop
    while time.time() - start < time_budget_s:
        t = time.time() - start
        elapsed_fraction = t / time_budget_s
        
        # Cooling schedule
        T = T_start * (T_end / T_start) ** elapsed_fraction
        step = step_start * (1 - elapsed_fraction)
        
        # Make a proposal
        h_proposed = propose_move(cur, step)
        c_proposed = c5_of(h_proposed)
        
        # Accept the proposal
        if c_proposed <= c_cur:
            cur = h_proposed
            c_cur = c_proposed
            if c_proposed < c_best:
                best = h_proposed.copy()
                c_best = c_proposed
        else:
            # Metropolis acceptance rule
            if random.random() < np.exp(-(c_proposed - c_cur) / T):
                cur = h_proposed
                c_cur = c_proposed
                
    return best, c_best

    random.seed(seed)

    start = time.time()

            initial_h_values
        )
        # Renormalize to exact sum
        h_provided = h_provided * (N / 2) / np.sum(h_provided)
    else:
        h_provided = np.ones(N) * 0.5
    
    # 2. Double-peak seed
    h_double = np.full(N, 0.1)
    a = 0.3
    u = 0.9
    v = 0.1
    left = int(a * N / 2)
    right = int((2 - a) * N / 2)
    h_double[:left] = u
    h_double[right:] = u
    # Normalization
    h_double = h_double * (N / 2) / np.sum(h_double)
    
    # 3. Uniform seed
    h_uniform = np.ones(N) * 0.5

    # 4. Linear gradient seed
    h_linear = np.linspace(0.1, 0.9, N)
    h_linear = h_linear * (N / 2) / np.sum(h_linear)
    
    # 5. Step function seed
    h_step = np.ones(N) * 0.5
    h_step[:N//3] = 0.9
    h_step[2*N//3:] = 0.1
    h_step = h_step * (N / 2) / np.sum(h_step)

    # Test all initial seeds
    seeds = [h_provided, h_double, h_uniform, h_linear, h_step]
    best_h = None
    best_c5 = float('inf')
    
    # Run each seed for 0.3 * 600 seconds = 180 seconds
    remaining_time = budget_s * 0.95  # Leave some margin for final polishing
    time_per_seed = min(remaining_time / len(seeds), 200)
    
    for i, h_seed in enumerate(seeds):
        if time.time() - start >= budget_s * 0.9:
            break
        h_opt, c5_opt = anneal(h_seed, time_per_seed)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # This might give a slightly better result than annealing...
    # But only if we have time left and a good solver strategy

    # Use differential evolution if time allows
    remaining_time = budget_s - (time.time() - start)

            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)

        except Exception:
            pass  # Skip if differential evolution fails
            
    # Final polishing phase (with zero temperature for maximum certainty)
    # Optimized steps in greedy moves to escape local minima
    remaining_time = budget_s - (time.time() - start)
    if remaining_time > 10 and best_h is not None:
        cur = best_h.copy()
        c_cur = c5_of(cur)
        
        # Greedy hill-climbing phase
        max_iterations = min(50000, int(remaining_time * 100))
        for _ in range(max_iterations):
            if time.time() - start >= budget_s:
                break
            # Small random perturbations
            h_proposed = cur.copy()
            idx = random.randint(0, N - 1)
            perturbation = random.uniform(-0.02, 0.02)
            h_proposed[idx] = max(0, min(1, h_proposed[idx] + perturbation))
            # Renormalize
            h_proposed = h_proposed * (N / 2) / np.sum(h_proposed)
            
            c_proposed = c5_of(h_proposed)
            
            if c_proposed < c_cur:
                cur = h_proposed
                c_cur = c_proposed
                if c_proposed < best_c5:
                    best_c5 = c_proposed
                    best_h = cur.copy()

    # Ensure final normalization
    if best_h is not None:
        best_h = best_h * (N / 2) / np.sum(best_h)

    return (best_h, best_c5, N)
