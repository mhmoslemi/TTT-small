import numpy as np
import time
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def project_to_box_and_sum(h, dx, N, target_sum=0.5*N):
    """
    Project h onto the intersection of [0,1]^N and {sum(h) = target_sum}
    using water filling.
    """
    # Sort values in descending order to facilitate water filling
    v = h.copy()
    v = np.clip(v, 0, 1)
    v_sorted = np.sort(v)[::-1]
    # Binary search for lambda
    low, high = -1000, 1000
    for _ in range(50):
        mid = (low + high) / 2
        total = np.sum(np.clip(v_sorted + mid, 0, 1))
        if total > target_sum:
            low = mid
        else:
            high = mid
    lam = (low + high) / 2
    # Apply projection
    h_proj = np.clip(v + lam, 0, 1)
    # Small adjustment to enforce exact sum
    diff = target_sum - np.sum(h_proj)
    if abs(diff) > 1e-10:
        # Redistribute on free variables
        free = np.where((h_proj > 0) & (h_proj < 1))[0]
        if len(free) > 0:
            to_add = diff / len(free)
            h_proj[free] += to_add
            h_proj = np.clip(h_proj, 0, 1)
    return h_proj

def c5_of(h):
    """Compute the maximum overlap integral for the given step function h."""
    dx = 2.0 / len(h)
    c = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(c) * dx
    return c5

def smooth_c5(h, eps):
    """Compute smooth approximation to C5 using log-sum-exp."""
    dx = 2.0 / len(h)
    c = np.correlate(h, 1 - h, mode='full')
    c = c * dx
    # For numerical stability, subtract max
    c_max = np.max(c)
    c_adj = c - c_max
    exps = np.exp(c_adj / eps)
    total = np.sum(exps)
    if total == 0:
        total = 1e-100 # avoid log(0)
    logsumexp = c_max + eps * np.log(total)
    return logsumexp

def grad_smooth_c5(h, eps):
    """Compute gradient of smooth C5 approximation."""
    dx = 2.0 / len(h)
    c = np.correlate(h, 1 - h, mode='full')
    c = c * dx
    c_max = np.max(c)
    c_adj = c - c_max
    exps = np.exp(c_adj / eps)
    total = np.sum(exps)
    if total == 0:
        total = 1e-100
    weights = exps / total  # Softmax weights
    # Compute gradient as weighted sum of subgradients
    grad = np.zeros_like(h)
    N = len(h)
    # Weights are over full correlation, i.e., len(c) = 2*N - 1
    for s in range(len(c)):
        # Penalties from left and right, masking
        idx_left = np.arange(s)  # 0 to s-1 (inclusive)
        idx_right = np.arange(s + 1, N)  # s+1 to N-1
        if len(idx_left) > 0:
            grad[idx_left] += weights[s] * (-h[s])
        if len(idx_right) > 0:
            grad[idx_right] += weights[s] * (-h[s])
        # Gradient from the current shift
        idx = s
        if idx < N:
            grad[idx] += weights[s] * (1 - h[idx])
    return grad

def continuation_optimize(h0, eps_list, max_iter=150, tol=1e-6):
    """Perform continuation optimization using smooth surrogate."""
    h = h0.copy()
    dx = 2.0 / len(h)
    N = len(h)
    for eps in eps_list:
        # Inner L-BFGS-B optimization
        def obj(h_in):
            return smooth_c5(h_in, eps)
        def grad_obj(h_in):
            return grad_smooth_c5(h_in, eps)
        # Optimization step
        res = minimize(obj, h, method='L-BFGS-B', jac=grad_obj,
                       bounds=[(0, 1)] * N,
                       options={'maxiter': max_iter, 'gtol': tol})
        h = res.x
        # Project to feasible set
        h = project_to_box_and_sum(h, dx, N)
    # Final projection to guarantee sum
    h = project_to_box_and_sum(h, dx, N)
    return h

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    N = 140
    dx = 2.0 / N
    start = time.time()
    target_sum = 0.5 * N

    # Define seeds with varied strategies  
    seeds = []

    # Uniform seed
    h0 = np.ones(N) * 0.5
    h0 = project_to_box_and_sum(h0, dx, N)
    seeds.append(h0.copy())

    # Two-peak seed with different peak widths
    for a in [0.12, 0.18, 0.24, 0.30]:
        h1 = np.ones(N)
        x_vals = np.linspace(0, 2, N)
        u = 0.9
        v = 0.1
        mask = (x_vals < a) | (x_vals >= 2 - a)
        h1[mask] = u
        h1[~mask] = v
        h1 = project_to_box_and_sum(h1, dx, N)
        seeds.append(h1.copy())
    
    # Binary seeds
    for b in range(3):
        h_bin = np.zeros(N)
        idxs = np.random.choice(N, size=N//2, replace=False)
        h_bin[idxs] = 1
        h_bin = project_to_box_and_sum(h_bin, dx, N)
        seeds.append(h_bin.copy())
    
    # Symmetric binary
    h_sym = np.zeros(N)
    h_sym[:N//2] = 1
    h_sym = project_to_box_and_sum(h_sym, dx, N)
    seeds.append(h_sym.copy())

    # Random continuous
    for r in range(2):
        h_rand = np.random.rand(N)
        h_rand = project_to_box_and_sum(h_rand, dx, N)
        seeds.append(h_rand.copy())

    # Choose smaller subset (original settings + few extra) to limit resources
    selected_seeds = seeds[:6]

    # Continue with large epsilon to escape local minima early
    eps_list = [0.10, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]

    best_h = None
    best_c5 = float('inf')

    # Run optimization from each seed
    for h_seed in selected_seeds:
        remaining = budget_s - (time.time() - start)
        if remaining <= 5:
            break
        h_opt = continuation_optimize(h_seed, eps_list, max_iter=100, tol=1e-6)
        c5 = c5_of(h_opt)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()

    # Final polishing with a max-based sub-gradient method
    remaining = budget_s - (time.time() - start)
    if remaining <= 5:
        return (best_h, best_c5, N)
    
    # Max-based sub-gradient descent
    def subgrad_descent_step(h, eta=0.03):
        dx = 2.0 / len(h)
        c = np.correlate(h, 1-h, mode='full')
        c = c * dx
        # argmax over s
        s_star = np.argmax(c)
        # Construct subgradient
        g = np.zeros_like(h)
        g[s_star] = -dx  # Simple subgradient
        # More effective per shift, we track left/right terms
        # Left contribution from h[0:s_star]
        left_idx = np.arange(s_star)
        g[left_idx] -= dx * h[left_idx]  # shifting from left
        # If there is no left term, that's Imaginary
        
        # Right contribution
        idx_right = np.arange(s_star + 1, len(h))
        g[idx_right] -= dx * h[idx_right]
        
        # Apply subgradient step
        h_new = h - eta * g
        h_new = project_to_box_and_sum(h_new, dx, N)
        return h_new

    # Run final polishing
    for _ in range(500):  # Max iterations
        if time.time() - start > budget_s - 5:
            break
        better_h = subgrad_descent_step(best_h, eta=0.02)
        c5_new = c5_of(better_h)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = better_h.copy()
    
    return (best_h, best_c5, N)
