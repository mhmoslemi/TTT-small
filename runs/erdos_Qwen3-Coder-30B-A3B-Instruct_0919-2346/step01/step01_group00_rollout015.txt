import numpy as np
import time
import random

def project_h(h, target_sum):
    """Project h onto [0,1] and sum=target_sum using water-filling."""
    h = np.array(h, dtype=float)
    h_clipped = np.clip(h, 0, 1)
    current_sum = np.sum(h_clipped)
    
    if np.isclose(current_sum, target_sum):
        return h_clipped.copy()
    
    # Binary search for lambda
    def sum_with_lambda(lam):
        return np.sum(np.clip(h_clipped - lam, 0, 1))
    
    if current_sum > target_sum:
        # Reduce some entries
        low, high = 0.0, np.max(h_clipped)
        while high - low > 1e-10:
            mid = (low + high) / 2.0
            if sum_with_lambda(mid) >= target_sum:
                low = mid
            else:
                high = mid
        lambda_val = low
        h_new = np.clip(h_clipped - lambda_val, 0, 1)
    else:
        # Increase some entries
        low, high = 0.0, 1.0
        while high - low > 1e-10:
            mid = (low + high) / 2.0
            if sum_with_lambda(mid) <= target_sum:
                low = mid
            else:
                high = mid
        lambda_val = low
        h_new = np.clip(h_clipped + lambda_val, 0, 1)
        
    # Correction to ensure exact sum
    diff = target_sum - np.sum(h_new)
    if abs(diff) > 1e-10:
        # Distribute the difference
        valid_indices = np.where((h_new > 0) & (h_new < 1))[0]
        if len(valid_indices) > 0:
            # Adjust uniformly among valid entries
            adjustment = diff / len(valid_indices)
            h_new[valid_indices] += adjustment
            h_new = np.clip(h_new, 0, 1)
    
    return h_new

def compute_C_and_grad(h, alpha=50):
    """Compute max C_k and gradient of smoothed C using FFT."""
    N = len(h)
    dx = 2.0 / N
    
    # Compute correlation using FFT
    # h needs padding to ensure full correlation
    h_padded = np.pad(h, (0, N-1), mode='constant', constant_values=0)
    h_conj = np.pad(1 - h, (0, N-1), mode='constant', constant_values=0)
    
    # FFT correlation using ifft(fft(h) * conj(fft(1-h)))
    corr_fft = np.fft.ifft(np.fft.fft(h_padded) * np.conj(np.fft.fft(h_conj)))
    corr = np.real(corr_fft[:2*N-1])
    
    # C_k = 1 - dx * corr_k
    C = 1 - dx * corr
    
    # Smoothing
    exp_C = np.exp(alpha * C)
    w = exp_C / np.sum(exp_C)  # Softmax
    
    # Compute gradient with respect to h
    # For each i, ∂C_k/∂h_i = dx * (1 - (h_{i+k} + h_{i-k}))
    # But we use reverse convolutions to get the weighted sum correctly.
    
    # w_rev = w[::-1] for mirror in correlation
    w_rev = w[::-1]
    # Correlations convolutions: v_plus = conv(h, w_rev), v_minus = conv(h, w)
    # v_plus = np.convolve(h, w_rev, mode='valid') with proper indexing.
    # Use full mode since length sufficient: conv(h, w_rev, mode='full')[N-1:N-1+N] 
    # But a more direct way is to compute convolution sum manually.
    
    v_plus = np.zeros(N)
    v_minus = np.zeros(N)
    
    # Properly compute w * h with shift
    for i in range(N):
        # ∑_k w_k h_{i+k} considering zero padding
        for k in range(-N+1, N):
            j = i + k
            if 0 <= j < N:
                v_plus[i] += w[N-1+k] * h[j]
                v_minus[i] += w[N-1+k] * h[j]
    
    # Gradient = dx * (1 - (v_plus + v_minus)) 
    # v_plus = conv(h, w_rev)[i], v_minus = conv(h, w)[i]
    # But use full convolve and slice appropriately:
    # Using convolution of extended arrays:
    
    # Make v_plus = w_rev * h using proper indexing
    # w_rev has length 2N-1 centered at N-1
    # conv = w_rev * h where we account for wraps and zero padding correctly over all i
    
    v_plus_full = np.convolve(h, w_rev, mode='full')
    v_minus_full = np.convolve(h, w, mode='full')
    
    v_plus = v_plus_full[N-1:N+((N-1)*2)]  # Extract same length
    v_minus = v_minus_full[N-1:N+((N-1)*2)]    
    # Actually let's follow a direct method:
    
    v_plus = np.zeros(N)
    v_minus = np.zeros(N)
    for i in range(N):
        term_plus = 0.0
        term_minus = 0.0
        for k in range(-(N-1), N):  # All shift indices in range
            if (i+k) >= 0 and (i+k) < N:
                term_plus += w[N-1+k] * h[i+k]  # w index shifts from 0 to 2N-2
                term_minus += w[N-1+k] * h[i+k]
        v_plus[i] = term_plus
        v_minus[i] = term_minus
    
    # Gradient = dx*(1 - v_plus - v_minus)
    grad = dx * (1.0 - v_plus - v_minus)
    
    # Return max C and gradient (remove mirror terms used for convolution but preserve properties)
    actual_C = C[:N]  # Take first N values as only diagonals useful for non-shift
    return np.max(actual_C), grad

def compute_C(h):
    """Compute actual max C, exact (not smoothed)."""
    N = len(h)
    dx = 2.0 / N
    # h and 1-h correlation with mode='full'
    corr = np.correlate(h, 1-h, mode="full")
    C = corr * dx
    return np.max(C)
