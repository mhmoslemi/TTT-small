import numpy as np

def project_h(h, target_sum):
    """Project h onto the feasible set: sum=h, 0<=h<=1."""
    h = np.array(h, dtype=float)
    # Clip to bounds [0, 1]
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Binary search for lambda
    low, high = -1.0, 1.0
    for _ in range(100):
        mid = (low + high) / 2.0
        temp_h = np.clip(h - mid, 0, 1)
        if np.sum(temp_h) > target_sum:
            low = mid
        else:
            high = mid
    
    lambda_val = (low + high) / 2.0
    h_new = np.clip(h - lambda_val, 0, 1)
    return h_new

def compute_C_and_gradient(h, alpha=50.0):
    """Compute C vector and gradient of smoothed objective."""
    N = len(h)
    dx = 2.0 / N
    
    # Compute correlation via FFT for efficiency
    h_fft = np.fft.fft(h, n=2*N-1)
    one_minus_h_fft = np.fft.fft(1-h, n=2*N-1)
    corr = np.fft.ifft(h_fft * np.conj(one_minus_h_fft)).real
    
    # Take only the first 2N-1 values (corresponds to shifts)
    C = corr[:2*N-1]
    # Compute C_k = integral h(x) * (1 - h(x+k)) dx = 1 - dx * sum_i h_i * h_{i+k}
    # Note: corr[k] corresponds to shift k from -(N-1) to N-1
    C = 1 - dx * C
    
    # Compute smooth approximation of max: log-sum-exp
    C_smooth = np.exp(alpha * C)
    Z = np.sum(C_smooth)
    w = C_smooth / Z
    
    # Compute gradient of smoothed objective
    # ∂f/∂h_i = dx * Σ_k w_k * (1 - h_{i+k} - h_{i-k})
    w_rev = w[::-1]
    conv1 = np.convolve(h, w_rev, mode='same')  # h * w_rev
    conv2 = np.convolve(h, w, mode='same')      # h * w
    grad = dx * (1 - (conv1 + conv2))
    
    # Compute actual max C for comparison
    C_max = np.max(C)
    
    return C_max, grad
