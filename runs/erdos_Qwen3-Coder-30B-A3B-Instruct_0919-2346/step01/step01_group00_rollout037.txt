import numpy as np
import time
import random

def compute_C_vector(h):
    """Compute the full vector of overlaps C_τ(h) for all shifts τ."""
    dx = 2.0 / len(h)
    # Use FFT-based correlation for efficiency
    H = np.fft.rfft(h, 2*len(h)-1)
    S = np.fft.irfft(H * np.conj(H), 2*len(h)-1)
    # S is the autocorrelation of h, zero-padded to 2n-1 length
    # Extract only the valid shifts
    c = 1.0 - dx * S[:2*len(h)-1]
    return c

def compute_softmax_weights(c, alpha):
    """Compute w_τ = exp(alpha * c_τ) / sum(exp(alpha * c_σ))."""
    c_max = np.max(c)
    # Numerically stable softmax with log-sum-exp trick
    exp_c = np.exp(alpha * (c - c_max))
    w = exp_c / np.sum(exp_c)
    return w

def compute_gradient(h, w):
    """Compute gradient of smoothed objective ∇f_α(h) = -dx * (w * h + w_rev * h)."""
    dx = 2.0 / len(h)
    # Convolution with w (circularly padded, same size)
    # Both convs use 'same' to preserve length
    w_rev = w[::-1]
    # Note: correlation with reversed w for the second term
    g1 = np.convolve(h, w, mode='same')
    g2 = np.convolve(h, w_rev, mode='same')
    g = -dx * (g1 + g2)
    return g

def project_onto_box_and_mass(h, target_sum):
    """Project h onto the box [0,1]^n with exactly sum=h_sum."""
    h = np.asarray(h).copy()
    h_sum = np.sum(h)
    if abs(h_sum - target_sum) < 1e-12:
        # Already exactly the right sum, just clip
        h = np.clip(h, 0, 1)
        return h
    
    # Use water-filling method
    # We need to find lambda such that sum_i clip(h_i - lambda, 0, 1) = target_sum
    
    # First check direct clipping
    h_clipped = np.clip(h, 0, 1)
    h_clipped_sum = np.sum(h_clipped)
    if abs(h_clipped_sum - target_sum) < 1e-12:
        return h_clipped
    
    # Bisection on lambda
    # Lower and upper bounds for lambda
    lbd = np.min(h) - 1.0
    ubd = np.max(h) + 1.0
    eps = 1e-15
    max_iter = 1000
    
    def phi(lam):
        # Return sum_i clip(h_i - lam, 0, 1) - target_sum
        # This is a convex function (monotonic decreasing)
        clipped = np.clip(h - lam, 0, 1)
        return np.sum(clipped) - target_sum
    
    for _ in range(max_iter):
        mid = (lbd + ubd) / 2.0
        val = phi(mid)
        if abs(val) < eps:
            break
        if val > 0:
            lbd = mid
        else:
            ubd = mid
    
    h_proj = np.clip(h - (lbd + ubd) / 2.0, 0, 1)
    return h_proj

            t_frac = min(1.0, iter_count / (max_iter_per_seed * 0.3))

            eta = eta_0 / np.sqrt(1.0 + iter_count)

            c = compute_C_vector(h)
            
            # Compute softmax weights
            w = compute_softmax_weights(c, alpha)
            
            # Compute gradient
            g = compute_gradient(h, w)
            
            # Momentum update
            m = momentum_beta * m + (1 - momentum_beta) * g

            h_try = h_try.copy()
            N = len(h_try)
            i, j = random.sample(range(N), 2)
            
            delta_low = max(-h_try[i], h_try[j] - 1.0)
            delta_high = min(1.0 - h_try[i], h_try[j])
            
            if delta_low >= delta_high:
                return h_try
            
            step = min(max_step, delta_high - delta_low)
            delta = random.uniform(-step, step)
            delta = max(delta_low, min(delta, delta_high))
            
            h_try[i] += delta
            h_try[j] -= delta
            
            return h_try
            
        t_start = time.time()

            h_new = propose_move(cur, 0.02)
            c_new = np.max(compute_C_vector(h_new))
            if c_new <= c_cur or random.random() < 0.05:
                c_cur = c_new
                cur = h_new
                if c_cur < best_C5:
                    best_C5 = c_cur
                    best_h = cur.copy()
    
    return (best_h, best_C5, n)
