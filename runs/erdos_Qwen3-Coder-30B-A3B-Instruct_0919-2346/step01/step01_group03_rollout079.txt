import numpy as np
import time

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral via full convolution of h with (1-h)
    overlaps = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum of the full correlation vector
    # scaled by dx
    c5 = np.max(overlaps) * dx
    return c5

def full_overlap(h):
    """Return the raw overlap correlation vector (unscaled)"""
    return np.correlate(h, 1 - h, mode="full")

def project_onto_P(v, total):
    """Project vector v onto the set {x : 0 <= x[i] <= 1, sum x = total}"""
    # Using bisection to find lambda
    low, high = np.min(v) - 1, np.max(v) + 1
    tol = 1e-12
    while high - low > tol:
        mid = (low + high) / 2
        x = np.clip(v - mid, 0, 1)
        if np.sum(x) > total:
            low = mid
        else:
            high = mid
    x = np.clip(v - high, 0, 1)
    return x

def smooth_gradient(h, beta):
    """Compute gradient of the smooth approximation log(sum(exp(beta*F_k)))"""
    N = len(h)
    F = full_overlap(h)
    # For numerical stability, subtract max
    F_shifted = F - np.max(F)
    w = np.exp(beta * F_shifted)
    w = w / np.sum(w)
    
    g = np.ones(N)
    for k in range(len(F)):
        wk = w[k]
        if k >= N-1:
            # shift to the right
            d = k - (N-1)
            if d < N:
                g[d:] -= wk * h[:N-d]
                g[:N-d] -= wk * h[d:]
        else:
            # shift to the left
            d = N-1-k
            if d < N:
                g[:N-d] -= wk * h[d:]
                g[d:] -= wk * h[:N-d]
    return g

def projected_gradient_descent(h0, time_budget, seed=42):
    """Run projected gradient descent with smooth surrogate and β-schedule"""
    np.random.seed(seed)
    start = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of(h)
    total = np.sum(h0)
    
    # Parameters
    beta = 2.0
    beta_max = 30.0
    beta_update_interval = 50
    gamma_beta = 1.5
    step = 0.5
    gamma_step = 0.9995
    max_iter = 20000

    for i in range(max_iter):
        if time.time() - start > time_budget:
            break
            
        # Compute overlap for gradient 
        F = full_overlap(h)
        # For numerical stability
        F_shifted = F - np.max(F)
        w = np.exp(beta * F_shifted)
        w = w / np.sum(w)
        
        # Compute gradient
        g = np.ones(len(h))
        for k in range(len(F)):
            wk = w[k]
            if k >= len(h) - 1:
                # shift to the right
                d = k - (len(h) - 1)
                if d < len(h):
                    g[d:] -= wk * h[:len(h)-d]
                    g[:len(h)-d] -= wk * h[d:]
            else:
                # shift to the left
                d = len(h) - 1 - k
                if d < len(h):
                    g[:len(h)-d] -= wk * h[d:]
                    g[d:] -= wk * h[:len(h)-d]
        
        # Tentative update
        v = h - step * g
        # Project
        h = project_onto_P(v, total)
        
        # Record best
        c5 = c5_of(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
        # Schedule updates
        step *= gamma_step
        if i % beta_update_interval == 0 and beta < beta_max:
            beta = min(beta * gamma_beta, beta_max)
            
    return best_h, best_c5

def targeted_refinement(h0, time_budget, alpha=1.0):
    """Do deterministic refinement of the worst pair"""
    start = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of(h)
    
    # How long to run
    max_tries = 100000
    tries = 0
    while time.time() - start < time_budget and tries < max_tries:
        C = full_overlap(h)
        d = np.argmax(C) - (len(h) - 1)
        threshold = np.max(C) * 0.999  # Only consider pairs that are almost max
        C_s = C[C >= threshold]
        if len(C_s) > 1:
            d_s = np.argsort(C_s)[-1] - (len(h) - 1)
            if abs(d) > abs(d_s):
                d = d_s
            else:
                # If d itself is close to a very big value, make sure we 
                # consider the one with max(C) value
                pass
        i_star = np.argmax(C)  # At least one item in the maximum is in the correlation
        if i_star == len(h):
            i_star = 0
        if i_star == len(h) - 1:
            d = 0
            shift_to_left = True  # Equivalent to shift = len(h)-1
            shift = len(h) - 1  # so from 0 to len(h) -1 
            hi = h[:len(h) - shift]
            hj = h[shift:]
        elif i_star < len(h) - 1:
            shift = i_star + (len(h) - 1) - np.argmax(C)  # Fixing d s.t. i is based on best overlap position
            if shift < 0:
                shift = 0
            d = i_star - (len(h) - 1)
            hi = h[:len(h) - shift]
            hj = h[shift:]
        else:
            # Less likely, but handle
            d = 0
            shift = 0
            hi = h[:len(h)]
            hj = h[:len(h)]
        if shift >= len(h) or abs(shift) >= len(h):
            shift = len(h) // 2
        if d >= 0:
            hi = h[:len(h) - d]
            hj = h[d:]
        else:
            # Negative shift means shift left
            d = abs(d)
            hi = h[d:]
            hj = h[:len(h) - d]
        
        if len(hi) != len(hj):
            min_len = min(len(hi), len(hj))
            hi = hi[:min_len]
            hj = hj[:min_len]
            
        # Find pair that maximizes the product hi * (1 - hj)
        # Choose the element that gives highest reduction
        # i.e. max over all hi[i] * (1 - hj[i])
        if len(hi) > 0:
            c = hi * (1 - hj)
            max_idx = np.argmax(c)
            # Use the actual indices in h
            if d >= 0:
                i = max_idx
                j = max_idx + d
            else:
                i = max_idx - d
                j = max_idx
            if i >= len(h) or j >= len(h) or i < 0 or j < 0:
                tries += 1
                continue
            c_val = c[max_idx]
            if c_val <= 0:
                tries += 1
                continue
            delta = min(alpha * c_val, h[i], 1 - h[j])
            if delta <= 0:
                tries += 1
                continue
            # Make sure there's no constraint violation
            save_h = h.copy()
            h[i] -= delta
            h[j] += delta
            # Make sure no constraint violated
            if h[i] < 0 or h[j] > 1:
                h = save_h
                alpha *= 0.5
                if alpha < 1e-8:
                    break
                tries += 1
                continue
            new_c5 = c5_of(h)
            if new_c5 <= best_c5:
                best_c5 = new_c5
                best_h = h.copy()
            else:
                h = save_h
                alpha *= 0.5
                if alpha < 1e-8:
                    break
            tries += 1
        else:
            tries += 1
    return best_h, best_c5

    else:
        h_initial = np.random.rand(N) * 0.5
        h_initial *= (N / 2) / np.sum(h_initial)

    # 3. Double-peak seed
    h_double_peak = np.full(N, 0.1)
    a = 0.3
    idxs = np.where((np.linspace(0, 2, N) < a) | (np.linspace(0, 2, N) > (2 - a)))[0]
    h_double_peak[idxs] = 0.9
    h_double_peak *= (N / 2) / np.sum(h_double_peak)

    # 4. Random Dirichlet seed (concentration 0.5)
    h_random = np.random.dirichlet([0.5] * N)
    h_random = np.clip(h_random, 0, 1)
    h_random *= (N / 2) / np.sum(h_random)

    best_h = None
    best_c5 = float('inf')

    seeds = [h_uniform, h_initial, h_double_peak, h_random]
    remaining_budget = budget_s
    budget_per_seed = 0.3 * budget_s / len(seeds)
    
    # Run projected GD on each seed
    for i, h_seed in enumerate(seeds):
        if remaining_budget <= 1.0:
            break
        h_new, c5_new = projected_gradient_descent(h_seed, budget_per_seed, seed + i)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
        remaining_budget -= budget_per_seed

    # Then apply targeted refinement 
    if best_h is not None and remaining_budget > 1.0:
        refinement_time = remaining_budget * 0.6
        best_h, best_c5 = targeted_refinement(best_h, refinement_time)
    
    # Final check of the C5 value to make sure it's actually valid
    if best_h is not None:
        actual_c5 = c5_of(best_h)
        if actual_c5 < best_c5:
            best_c5 = actual_c5
    
    return (best_h, best_c5, N)
