import numpy as np
import time

def c5_of(h):
    """Compute the maximum overlap integral for the given step function h."""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    c = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(c) * dx
    return c5

def compute_gradient(h, shift):
    """Compute the gradient of C5 for a given shift s."""
    N = len(h)
    dx = 2.0 / N
    g = np.full(N, 1.0) * dx
    if shift >= 0:
        if shift < N:
            # subtract right shift
            g[:N-shift] -= h[shift:] * dx
            # subtract left shift
            g[shift:] -= h[:N-shift] * dx
    else:
        shift_abs = -shift
        if shift_abs < N:
            g[shift_abs:] -= h[:N-shift_abs] * dx
            g[:N-shift_abs] -= h[shift_abs:] * dx
    return g

def project_onto_bounds_and_sum(v, target):
    """Project vector v onto box [0,1] and hyperplane sum = target."""
    # binary search for lambda
    lower = -np.max(v)
    upper = 1 - np.min(v)
    for _ in range(60):
        mid = (lower + upper) / 2.0
        w = np.clip(v + mid, 0, 1)
        s = np.sum(w)
        if s < target:
            lower = mid
        else:
            upper = mid
    w = np.clip(v + (lower + upper) / 2.0, 0, 1)
    # Adjust if needed to ensure exact sum
    diff = target - np.sum(w)
    if abs(diff) > 1e-12:
        free_idx = (w > 0) & (w < 1)
        if np.any(free_idx):
            w[free_idx] += diff / np.sum(free_idx)
            w = np.clip(w, 0, 1)
        else:
            w += diff / len(w)
            w = np.clip(w, 0, 1)
    return w

def run(seed=42, budget_s=600):
    np.random.seed(seed)

    # initial candidate from globals, two-peak shape, or uniform

    else:

    # Two-peak seed 
    h1 = np.ones(N)
    x_vals = np.linspace(0, 2, N)
    a = 0.3
    u = 0.9
    v = 0.1
    mask = (x_vals < a) | (x_vals >= 2 - a)
    h1[mask] = u
    h1[~mask] = v
    h1 = h1 * (N/2) / np.sum(h1)
    
    # Uniform
    h2 = np.ones(N) * 0.5
    
    # try different initializations
    seeds = [h0, h1, h2]
    
    # Binary seeds
    for _ in range(3):
        h_bin = np.zeros(N)
        idx = np.random.choice(N, N//2, replace=False)
        h_bin[idx] = 1.0
        seeds.append(h_bin)
    
    # Track the best solution
    best_h = None
    best_c5 = float('inf')
    
    # Run parallel independent gradient descent from each seed
    start = time.time()
    
    for seed_h in seeds:
        cur_h = seed_h.copy()
        cur_score = c5_of(cur_h)
        if cur_score < best_c5:
            best_h = cur_h.copy()
            best_c5 = cur_score

    # Main loop - iterative improvement
    while time.time() - start < budget_s * 0.99:
        # Compute current worst shift for the best solution
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_idx = np.argmax(corr)
        s = max_idx - (N - 1)
        
        # Compute gradient and update
        grad = compute_gradient(best_h, s)
        # Decreasing step size
        t_frac = (time.time() - start) / budget_s
        step = 5.0 * (1 - t_frac) + 0.1
        
        # Update h
        v = best_h - step * grad
        new_h = project_onto_bounds_and_sum(v, N/2)
        
        # Evaluate new solution
        new_score = c5_of(new_h)
        
        if new_score < best_c5:
            best_h = new_h.copy()
            best_c5 = new_score
        else:
            # Accept occasionally
            if np.random.rand() < 0.05:
                best_h = new_h.copy()
                best_c5 = new_score
                
        # Try with random restart occasionally
        if np.random.rand() < 0.02:
            # random restart
            cur_h = np.random.rand(N)
            cur_h = cur_h * (N/2) / np.sum(cur_h)
            cur_score = c5_of(cur_h)
            if cur_score < best_c5:
                best_h = cur_h.copy()
                best_c5 = cur_score
    
    # Final local refinement - hill climbing
    for _ in range(5000):
        if time.time() - start > budget_s * 0.99:
            break
            
        # propose small move
        i, j = np.random.choice(N, size=2, replace=False)
        # Sample step size
        delta = np.random.uniform(-0.05, 0.05)
        # Ensure feasibility
        delta_low = max(-best_h[i], best_h[j] - 1)
        delta_high = min(1 - best_h[i], best_h[j])
        if delta_low >= delta_high:
            delta = 0
        else:
            delta = np.clip(delta, delta_low, delta_high)
            
        if delta != 0:
            test_h = best_h.copy()
            test_h[i] += delta
            test_h[j] -= delta
            # Reproject to sum constraint
            test_h = project_onto_bounds_and_sum(test_h, N/2)
            test_score = c5_of(test_h)
            if test_score < best_c5:
                best_h = test_h
                best_c5 = test_score
                
    return (best_h, best_c5, N)
