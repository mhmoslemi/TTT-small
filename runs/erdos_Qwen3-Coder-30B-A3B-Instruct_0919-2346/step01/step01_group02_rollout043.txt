import numpy as np
import time

def c5_of(h):
    dx = 2.0 / len(h)
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def worst_shift(h):
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    return idx - (len(h) - 1)

def subgradient(h, s):
    N = len(h)
    dx = 2.0 / N
    m = abs(s)
    g = np.full(N, dx)
    if m < N:
        g[:N - m] -= dx * h[m:]
        g[m:] -= dx * h[:N - m]
    return g

def project(v, target):
    lo = -np.max(v)
    hi = 1.0 - np.min(v)
    for _ in range(50):
        lam = (lo + hi) / 2.0
        w = np.clip(v + lam, 0.0, 1.0)
        if w.sum() < target:
            lo = lam
        else:
            hi = lam
    w = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        free = (w > 0) & (w < 1)
        if free.any():
            w[free] += diff / free.sum()
            w = np.clip(w, 0, 1)
    return w

def spd_phase(h, target_sum, t_start, t_end):
    while time.time() < t_end:
        s = worst_shift(h)
        g = subgradient(h, s)
        # Adaptive step size
        t_elapsed = time.time() - t_start
        t_total = t_end - t_start
        ratio = min(t_elapsed / t_total, 1.0)
        eta = 30.0 * (0.9995 ** (ratio * 10000))  # Very slow decay
        eta = min(eta, 50.0 * (2.0 / len(h)))
        v = h - eta * g
        h_new = project(v, target_sum)
        # Occasionally jitter to escape plateaux
        if int(t_elapsed * 100) % 30 == 0:
            idx = np.random.randint(0, len(h_new))
            step = np.random.uniform(-0.02, 0.02)
            h_new[idx] = np.clip(h_new[idx] + step, 0, 1)
            h_new = project(h_new, target_sum)
        h = h_new
    return h

def anneal_phase(h, target_sum, t_start, t_end):
    best_h = h.copy()
    best_c5 = c5_of(h)
    T = 0.05
    T_min = 1e-6
    T_decay = 0.9999
    max_step_size = 0.2
    while time.time() < t_end:
        T = max(T * T_decay, T_min)
        idx1, idx2 = np.random.randint(0, len(h), size=2)
        orig_val1, orig_val2 = h[idx1], h[idx2]
        delta_low = max(-h[idx1], h[idx2] - 1)
        delta_high = min(1 - h[idx1], h[idx2])
        if delta_low >= delta_high:
            continue
        step_size = np.random.uniform(max(-max_step_size, delta_low),
                                      min(max_step_size, delta_high))
        h[idx1] += step_size
        h[idx2] -= step_size
        # Ensure feasibility by projection
        h = project(h, target_sum)
        new_c5 = c5_of(h)
        if new_c5 <= best_c5 or np.random.rand() < np.exp(-(new_c5 - best_c5) / T):
            best_c5 = new_c5
            best_h = h.copy()
        else:
            # restore original state
            h[idx1] = orig_val1
            h[idx2] = orig_val2
    return best_h, best_c5

def hill_climb_phase(h, target_sum, t_start, t_end):
    best_h = h.copy()
    best_c5 = c5_of(h)
    max_step_size = 0.02
    while time.time() < t_end:
        t_elapsed = time.time() - t_start
        if t_elapsed > (t_end - t_start) * 0.95:
            break
        idx1, idx2 = np.random.randint(0, len(h), size=2)
        orig_val1, orig_val2 = h[idx1], h[idx2]
        delta_low = max(-h[idx1], h[idx2] - 1)
        delta_high = min(1 - h[idx1], h[idx2])
        step_size = np.random.uniform(max(-max_step_size, delta_low),
                                      min(max_step_size, delta_high))
        h[idx1] += step_size
        h[idx2] -= step_size
        # Ensure feasibility by projection
        h = project(h, target_sum)
        new_c5 = c5_of(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        else:
            h[idx1] = orig_val1
            h[idx2] = orig_val2
    return best_h, best_c5

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    N = 140
    dx = 2.0 / N
    target_sum = N / 2
    t_start = time.time()
    # Generate seeds
    seeds = []

    if 'initial_h_values' in globals():

                      initial_h_values)
        h = h * (N / 2) / np.sum(h)
        h = project(h, target_sum)
        seeds.append(h)
    # 2. Two-peak template with different peaks
    h2 = np.ones(N)
    a_vals = np.linspace(0.12, 0.38, 10)
    u, v = 0.9, 0.1
    for a in a_vals:
        h = np.ones(N)
        x_vals = np.linspace(0, 2, N)
        mask = (x_vals < a) | (x_vals >= 2 - a)
        h[mask] = u
        h[~mask] = v
        h = h * (N / 2) / np.sum(h)
        h = project(h, target_sum)
        seeds.append(h)
    # 3. Random binary vectors
    for _ in range(5):
        h = (np.random.rand(N) > 0.5).astype(float)
        h = h * (N / 2) / np.sum(h)
        h = project(h, target_sum)
        seeds.append(h)
    # 4. Uniform half
    h = np.ones(N) * 0.5
    h = project(h, target_sum)
    seeds.append(h)
    # Evaluate seeds
    best_h = None
    best_c5 = float('inf')
    for h_seed in seeds:
        c5 = c5_of(h_seed)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_seed.copy()
    assert best_h is not None
    # Phase 0 - Initialization (5%)
    t_phase_end = t_start + 0.05 * budget_s
    # No code needed: seeds already created
    # Phase 1 - Sub-gradient projected descent (55%)
    t_start_spd = t_phase_end
    t_phase_end = t_start + 0.60 * budget_s
    best_h = spd_phase(best_h, target_sum, t_start_spd, t_phase_end)
    best_c5 = c5_of(best_h)
    # Phase 2 - Annealing (30%)
    t_start_anneal = t_phase_end
    t_phase_end = t_start + 0.90 * budget_s
    best_h, best_c5 = anneal_phase(best_h, target_sum, t_start_anneal, t_phase_end)
    # Phase 3 - Hill climb (10%)
    t_start_hill = t_phase_end
    t_phase_end = t_start + 0.99 * budget_s
    best_h, best_c5 = hill_climb_phase(best_h, target_sum, t_start_hill, t_phase_end)
    # Final evaluation
    best_c5 = c5_of(best_h)
    return (best_h.copy(), best_c5, N)
