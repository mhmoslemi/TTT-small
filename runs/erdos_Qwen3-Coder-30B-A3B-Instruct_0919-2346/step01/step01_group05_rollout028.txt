import numpy as np
import time
from math import sqrt

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value scaled by dx
    max_overlap = np.max(correlation) * dx
    return max_overlap

def sturmian_binary_sequence(n, alpha=None):
    """Generate a balanced binary sequence using Sturmian word construction."""
    if alpha is None:
        alpha = (sqrt(5) - 1) / 2  # golden ratio conjugate
    h = np.zeros(n, dtype=int)
    for i in range(n):
        t = (i * alpha) % 1
        h[i] = 1 if t < 0.5 else 0
    ones_count = np.sum(h)
    target = n // 2
    if ones_count < target:
        # Need to flip some zeros to ones
        zero_indices = np.where(h == 0)[0]
        flip_indices = zero_indices[:target - ones_count]
        h[flip_indices] = 1
    elif ones_count > target:
        # Need to flip some ones to zeros
        one_indices = np.where(h == 1)[0]
        flip_indices = one_indices[:ones_count - target]
        h[flip_indices] = 0
    return h

def compute_overlap_vector(h):
    """Compute overlap values for all shifts."""
    n = len(h)
    dx = 2.0 / n
    correlation = np.correlate(h, 1 - h, mode="full")
    return correlation * dx

def compute_delta_shifts(h, i, j):
    """Compute delta changes in all overlap values for a swap of positions i, j."""
    n = len(h)
    delta = np.zeros(2 * n - 1)
    # i is currently 1, becomes 0
    # j is currently 0, becomes 1
    # Compute changes to overlap values from shifting
    for s in range(2 * n - 1):
        # The shift corresponds to position k in full correlation
        # which represents shift (k - n + 1)
        shift = s - n + 1
        
        # Compute contribution from i as first factor
        if i + shift >= 0 and i + shift < n:
            delta[s] -= (1 - h[i + shift])
        # Compute contribution from i as second factor
        if i - shift >= 0 and i - shift < n:
            delta[s] += h[i - shift]
        # Compute contribution from j as first factor
        if j + shift >= 0 and j + shift < n:
            delta[s] += (1 - h[j + shift])
        # Compute contribution from j as second factor
        if j - shift >= 0 and j - shift < n:
            delta[s] -= h[j - shift]
    
    return delta

def local_swap_search(h, n_points, budget_s):
    """Perform local swap search to minimize max overlap."""
    start_time = time.time()
    
    # Initialize overlap vector
    g = compute_overlap_vector(h)
    current_max = np.max(g)
    best_h = h.copy()
    best_max = current_max
    best_g = g.copy()
    
    # Tabu list for recently swapped pairs
    tabu_list = set()
    tabu_duration = 50
    
    # Maintaining count of 1s for constraint checking
    ones_count = np.sum(h)
    
    iteration = 0
    no_improvement_streak = 0
    max_no_improvement = 10000
    
    while time.time() - start_time < budget_s * 0.95:
        # Find current worst shift
        s_max = np.argmax(g)
        max_val = g[s_max]
        
        if max_val <= best_max * 0.9999:
            no_improvement_streak = 0
        else:
            no_improvement_streak += 1
        
        if no_improvement_streak > max_no_improvement:
            # Perturb by flipping random block preserving balance
            # Find 1s and 0s
            ones_indices = np.where(h == 1)[0]
            zeros_indices = np.where(h == 0)[0]
            # Flip a block
            block_size = max(1, int(len(ones_indices) * 0.05))
            p_indices = np.random.choice(ones_indices, block_size, replace=False)
            n_indices = np.random.choice(zeros_indices, block_size, replace=False)
            h[p_indices] = 0
            h[n_indices] = 1
            no_improvement_streak = 0  # reset
            g = compute_overlap_vector(h)
            current_max = np.max(g)
            if current_max < best_max:
                best_h = h.copy()
                best_max = current_max
                best_g = g.copy()
            iteration += 1
            continue
            
        # Sample potential swaps
        ones_indices = np.where(h == 1)[0]
        zeros_indices = np.where(h == 0)[0]
        
        # Find ones and zeros that would decrease the overlap significantly
        if len(ones_indices) > 0 and len(zeros_indices) > 0:
            # Pick a random 1 and a random 0
            i = np.random.choice(ones_indices)
            j = np.random.choice(zeros_indices)
            
            # Check tabu status
            pair = tuple(sorted([i, j]))
            if pair in tabu_list:
                iteration += 1
                continue
            
            # Compute effect of this swap
            delta = compute_delta_shifts(h, i, j)
            
            # Compute new max overlap
            new_g = g + delta
            new_max = np.max(new_g)
            
            # Accept swap if it reduces the max or if it's a valid tabu move with large Δ
            if new_max < max_val or (new_max < max_val + 1e-6 and np.random.rand() < 0.05):
                # Apply swap
                h[i] = 0
                h[j] = 1
                
                # Update g
                g = new_g
                current_max = new_max
                
                # Check if better
                if current_max < best_max:
                    best_h = h.copy()
                    best_max = current_max
                    best_g = g.copy()
                    
                # Update tabu list
                tabu_list.add(pair)
                if len(tabu_list) > 1000:
                    tabu_list.pop()  # Remove random element
            else:
                # Periodically remove old tabu entries
                if iteration % 100 == 0 and len(tabu_list) > 0:
                    tabu_list.pop()  # Remove random element
                    
        iteration += 1
    
    return best_h, best_max

def continuous_polish(initial_h, max_iter=100):

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with Coarse resolution
    n_points = 120
    
    # Generate initial balanced binary sequence using irrational rotation
    h = sturmian_binary_sequence(n_points)
    
    # Local search for the current resolution
    h_best, c5_best = local_swap_search(h, n_points, budget_s * 0.6)
    
    # Upsample to finer grid
    n_points_fine = 240
    h_upsampled = np.repeat(h_best, 2)  # Duplicate for upsampling
    
    # Re-normalize for integration constraint
    dx = 2.0 / n_points_fine
    current_sum = np.sum(h_upsampled) * dx
    h_upsampled = h_upsampled * (1.0 / current_sum)
    h_upsampled = np.clip(h_upsampled, 0, 1)
    
    # Perform second round of local swap search at finer grid
    h_best_fine, c5_best_fine = local_swap_search(h_upsampled, n_points_fine, budget_s * 0.3)
    
    # Optional final continuous refinement
    if time.time() - start_time < budget_s * 0.9:
        h_smooth = continuous_polish(h_best_fine, 50)
        c5_final = evaluate_erdos_solution(h_smooth)
        if c5_final < c5_best_fine:
            h_best_fine = h_smooth
            c5_best_fine = c5_final
            
    # Return the final result
    return (h_best_fine.tolist(), c5_best_fine, len(h_best_fine))
