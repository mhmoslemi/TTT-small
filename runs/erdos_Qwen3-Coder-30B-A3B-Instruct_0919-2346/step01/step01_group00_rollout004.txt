import numpy as np
import time
import random
from scipy.optimize import minimize

def compute_C_and_grad(h, alpha):
    """Computes the maximum overlap integral and its gradient."""
    N = len(h)
    dx = 2.0 / N
    
    # Compute correlation C_k = 1 - dx * S_k where S_k = sum of h_i * h_{i+k}
    # Using FFT for efficiency
    h_padded = np.pad(h, (0, N-1), mode='constant', constant_values=0)
    h_conj = np.pad(1-h, (0, N-1), mode='constant', constant_values=0)
    
    # Correlation using FFT: correlation of h and (1-h)
    corr = np.fft.fft(h_padded) * np.conj(np.fft.fft(h_conj))
    corr = np.fft.ifft(corr)
    C = 1.0 - dx * np.real(corr[:2*N-1])
    
    # Compute softmax weights
    w = np.exp(alpha * C)
    w = w / np.sum(w)
    
    # Compute gradient with respect to h
    # Each C_k = 1 - dx * sum_i h_i * h_{i+k} 
    # So dC_k/dh_i = -dx * (h_{i+k} + h_{i-k})
    # Gradient of smoothed objective f = (1/alpha)*log(sum(exp(alpha*C_k)))
    # df/dh_i = (1/alpha) * sum_k w_k * dC_k/dh_i = -dx * sum_k w_k * (h_{i+k} + h_{i-k})
    
    # Use convolution to compute h_{i+k} and h_{i-k}
    # For h_{i+k}: convolve h with w_rev (reflect w)
    w_rev = w[::-1]
    v_plus = np.convolve(h, w_rev, mode='same')
    v_minus = np.convolve(h, w, mode='same')
    
    # Compute gradient
    grad = dx * (1.0 - (v_plus + v_minus))
    
    return np.max(C), grad

def project_h(h, target_sum):
    """Project h onto [0,1] and sum=target_sum using binary search for lambda."""
    h_clipped = np.clip(h, 0, 1)
    current_sum = np.sum(h_clipped)
    
    if np.isclose(current_sum, target_sum):
        return h_clipped
    
    # Binary search for lambda
    lower, upper = -1.0, 1.0
    tolerance = 1e-10
    
    for _ in range(100):
        lambda_mid = (lower + upper) / 2
        h_candidate = np.clip(h_clipped - lambda_mid, 0, 1)
        candidate_sum = np.sum(h_candidate)
        
        if abs(candidate_sum - target_sum) < tolerance:
            return h_candidate
        elif candidate_sum > target_sum:
            lower = lambda_mid
        else:
            upper = lambda_mid
    
    return h_candidate

    h_initial = h_initial * (N / 2) / np.sum(h_initial)
    
    # Double-peak shape from previous work
    h_double = np.zeros(N)
    a = 0.3
    u = 0.9
    v = 0.1
    x = np.linspace(0, 2, N)
    mask = (x < a) | (x > (2 - a))
    h_double[mask] = u
    h_double[~mask] = v
    h_double = h_double * (N / 2) / np.sum(h_double)
    
    # Prepare seeds
    seeds = [h_uniform, h_initial, h_double]
    best_h = None
    best_c5 = float('inf')
    
    start = time.time()
    time_left = budget_s
    
    for i, h_seed in enumerate(seeds):
        if time.time() - start > budget_s - 5:
            break
            
        h = h_seed.copy()
        # Simulate gradient descent with restarts and random perturbations
        # Using simple learning rate
        eta = 0.2
        alpha_start = 5.0
        alpha_end = 80.0
        max_iter = 2000  # limit iterations per specific seed to prevent overuse
        alpha_schedule = lambda t: alpha_start + (alpha_end - alpha_start) * t / max_iter
        iter_used = 0
        
        for t in range(max_iter):
            if time.time() - start > budget_s - 2:
                break
            
            iter_used += 1
            alpha = alpha_schedule(t)
            c5, grad = compute_C_and_grad(h, alpha)
            
            # Update using gradient
            h_new = h - eta * grad
            # Project: clip and recompute sum
            h_new = project_h(h_new, target_sum)
            
            # Check if c5 improved and store best
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
            
            h = h_new
            
        # Occasionally add some noise to escape local minima
        if iter_used % 500 == 0 and iter_used < max_iter:
            noise = np.random.normal(0, 0.01, N)
            h_noisy = h + noise
            h = project_h(h_noisy, target_sum)
    
    # Final polishing with a small subset of finer tuning
    if time.time() - start < budget_s - 10:

            h = h_vec
            dx = 2.0 / len(h)
            # Compute C using standard numpy correlate
            c = np.correlate(h, 1 - h, mode="full")
            c *= dx
            return -np.max(c)  # negative because we minimize, but minimize routine minimizes
        
        # Do one pass using scipy minimize with bounds and sum constraint
        bounds = [(0, 1) for _ in range(N)]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        
        # Sample from current best
        x0 = best_h.copy()
        
        res = minimize(objective, x0, bounds=bounds, constraints=constraints, 
                       method='SLSQP', options={'maxiter': 50})
        
        if not res.success:

            final_c5 = best_c5
            h_final = best_h.copy()
        else:
            h_final = res.x
            c5_vec = np.correlate(h_final, 1 - h_final, mode="full") * dx
            final_c5 = np.max(c5_vec)
            
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = h_final
    
    return (best_h, best_c5, N)
