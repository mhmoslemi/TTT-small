import numpy as np
import time
from scipy.interpolate import interp1d

def upsample_initial(h0, n):
    """Upsample initial_h_values to size n using linear interpolation."""
    if len(h0) == n:
        return np.array(h0)
    x_orig = np.linspace(0, 2, len(h0))
    x_new = np.linspace(0, 2, n)
    f = interp1d(x_orig, h0, kind='linear', fill_value='extrapolate')
    h_up = f(x_new)
    h_up = np.clip(h_up, 0, 1)
    return h_up

def project_to_simplex(v, target_sum):
    """Project vector v onto the simplex {x: sum(x)=target_sum, x>=0}."""
    # sort in descending order
    v_sorted = np.sort(v)[::-1]
    # find the threshold lambda
    cumsum = np.cumsum(v_sorted)
    tmp = (cumsum - target_sum) / np.arange(1, len(v_sorted) + 1)
    # find the largest i such that lambda_i <= v_sorted[i]
    idx = np.where(v_sorted > tmp)[0]
    if len(idx) == 0:
        lambda_val = tmp[-1]
        result = np.maximum(v - lambda_val, 0)
    else:
        i = idx[-1]
        lambda_val = tmp[i]
        result = np.maximum(v - lambda_val, 0)
    return result

def softmax_weights(gamma, f_vals):
    """Compute softmax weights for current function values."""
    f_vals = np.array(f_vals)
    exp_term = np.exp(gamma * f_vals)
    return exp_term / np.sum(exp_term)

def compute_gradient(h, dx):
    """Compute ∇J_γ(h) = Σ_s w_s ∇f_s(h), where ∇f_s(h) is easy to compute."""
    n = len(h)
    grad = np.zeros(n)
    # Collect all shifts from -n+1 to n-1
    shifts = range(-n+1, n)
    f_vals = []
    for s in shifts:
        f_s = 0.0
        for i in range(n):
            if 0 <= i+s < n:
                f_s += h[i] * (1 - h[i+s])
        f_s *= dx
        f_vals.append(f_s)
    return f_vals

def compute_exact_objective(f_vals):
    """Extract the actual objective maximum from function values."""
    return np.max(f_vals)
