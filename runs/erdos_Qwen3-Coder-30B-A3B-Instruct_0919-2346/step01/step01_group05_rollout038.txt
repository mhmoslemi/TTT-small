import numpy as np
import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value scaled by dx
    max_overlap = np.max(correlation) * dx
    return max_overlap

def generate_sturmian_binary(n, alpha=(np.sqrt(5)-1)/2):
    """Generate a balanced binary vector using irrational rotation"""
    t = np.mod(np.arange(n) * alpha, 1.0)
    h = (t < 0.5).astype(float)
    current_sum = np.sum(h)
    target_sum = n / 2.0
    if current_sum > target_sum:
        # Need to flip some 1s to 0s
        ones = np.where(h == 1)[0]
        num_flip = int(current_sum - target_sum)
        flip_indices = ones[:num_flip]
        h[flip_indices] = 0
    elif current_sum < target_sum:
        # Need to flip some 0s to 1s
        zeros = np.where(h == 0)[0]
        num_flip = int(target_sum - current_sum)
        flip_indices = zeros[:num_flip]
        h[flip_indices] = 1
    return h

def compute_g_s(h):
    """Compute all g_s values given h"""
    n = len(h)
    dx = 2.0 / n
    correlation = np.correlate(h, 1 - h, mode="full")
    # Convert to g_s = correlation * dx without the dx factor step
    g_s = correlation * dx
    return g_s

def compute_deltas_for_swap(h, i, j, g_s):
    """Compute changes in g_s values for swapping h[i] and h[j]"""
    n = len(h)
    delta = np.zeros(2 * n - 1)
    # For each shift s, where we debate adding or removing elements
    for s in range(-(n-1), n):
        # Find overlap contribution indexes (positions affected by shift s)
        # index k in h contributes to s when position k satisfies k = i - s  or k = j - s
        if i - s >= 0 and i - s < n:
            # i is in overlapping region with shift s
            delta[s + n - 1] -= 1 - h[i]
        if j - s >= 0 and j - s < n:
            # j is in overlapping region with shift s
            delta[s + n - 1] += 1 - h[j]
        
        # When i becomes 0, any positions k such that i = k + s 
        # contribute additional loss: h[k] * (1 - 0) = h[k]
        if i + s >= 0 and i + s < n:
            delta[s + n - 1] += h[i + s]
        # When j becomes 1, any positions k such that j = k + s
        # contribute additional gain: h[k] * (1 - 1) = 0
        # but also the second effect - contributed by j as factor
        if j + s >= 0 and j + s < n:
            delta[s + n - 1] -= h[j + s]
        
    return delta

def local_swap_search(h, time_limit):
    """Perform local swap search to reduce max g_s"""
    start_time = time.time()
    n = len(h)
    dx = 2.0 / n
    # Precompute g_s values
    g_s = compute_g_s(h)
    max_g_s = np.max(g_s)
    current_h = h.copy()

    # Tabu list to keep track of recent swaps
    tabu_list = set()
    tabu_size = 100
    stagnation_count = 0
    max_stagnation = 10000
    last_improvement = 0
    
    # early stop criterion
    while time.time() - start_time < time_limit:
        if stagnation_count > max_stagnation:
            # Random perturbation to escape
            perturb_indices = np.random.choice(n, size=int(n * 0.05), replace=False)
            for idx in perturb_indices:
                current_h[idx] = 1 - current_h[idx]
            g_s = compute_g_s(current_h)
            max_g_s = np.max(g_s)
            stagnation_count = 0
            last_improvement = 0
            continue
            
        # Identify indices with ones and zeros
        ones_idx = np.where(current_h == 1)[0]
        zeros_idx = np.where(current_h == 0)[0]
        
        # Get the worst shift
        worst_shift_idx = np.argmax(g_s)
        if g_s[worst_shift_idx] < max_g_s * 0.95:
            # Below threshold, reduce the swap search range
            max_swaps_to_try = 1000
        else:
            max_swaps_to_try = 5000
        counter = 0
        while counter < max_swaps_to_try and time.time() - start_time < time_limit:
            # Select random 1 and 0 position to swap
            if len(ones_idx) == 0 or len(zeros_idx) == 0:
                break
                
            i = np.random.choice(ones_idx)
            j = np.random.choice(zeros_idx)
            
            # Check tabu list
            if (i,j) in tabu_list or (j,i) in tabu_list:
                counter += 1
                continue
            
            # Compute proposed change in g_s if swapping i and j
            delta = compute_deltas_for_swap(current_h, i, j, g_s)
            new_g_s = g_s + delta
            new_max = np.max(new_g_s)
            
            if new_max < max_g_s:
                # Accept the swap
                current_h[i] = 0
                current_h[j] = 1
                g_s = new_g_s
                max_g_s = new_max
                last_improvement = 0
                # Add to tabu list
                if len(tabu_list) >= tabu_size:
                    tabu_list.pop()
                tabu_list.add((i, j))
                counter = 0  # Reset counter
                # Update ones_idx and zeros_idx
                ones_idx = np.where(current_h == 1)[0]
                zeros_idx = np.where(current_h == 0)[0]
            else:
                # Occasionally accept worse moves
                if np.random.rand() < 0.02:
                    current_h[i] = 0
                    current_h[j] = 1
                    g_s = new_g_s
                    # Add to tabu list
                    if len(tabu_list) >= tabu_size:
                        tabu_list.pop()
                    tabu_list.add((i, j))
                else:
                    counter += 1
                
            last_improvement += 1
            stagnation_count += 1
            
    return current_h, max_g_s

def upscale_vector(h_coarse):
    """Duplicate each element to upscale the resolution"""
    n = len(h_coarse)
    h_fine = np.repeat(h_coarse, 2)
    # Adjust to ensure sum(n/2.0)
    delta_sum = np.sum(h_fine) - n
    if delta_sum > 0:
        # We have excess ones, delete some
        one_indices = np.where(h_fine == 1)[0]
        num_to_delete = int(delta_sum)
        delete_indices = np.random.choice(one_indices, size=num_to_delete, replace=False)
        h_fine[delete_indices] = 0
    elif delta_sum < 0:
        # We have too few ones
        zero_indices = np.where(h_fine == 0)[0]
        num_to_add = int(-delta_sum)
        add_indices = np.random.choice(zero_indices, size=num_to_add, replace=False)
        h_fine[add_indices] = 1
    return h_fine

def continuous_polish(h, max_iter=100):
    """Perform projected gradient descent to slightly refine solution"""
    n = len(h)
    dx = 2.0 / n
    equivalent = False
    for _ in range(max_iter):
        # Compute derivative w.r.t. h for a fixed shift
        # derivative approximation using finite differences or computation
        # For now, use a simple noise-based approach
        # Slightly perturb
        h_original = h.copy()
        noise = np.random.normal(0, 0.005, n)
        h = h + noise
        h = np.clip(h, 0, 1)
        # Project back to sum exactly n/2
        current_sum = np.sum(h) * dx
        h = h * (n / 2.0) / current_sum
        
        # If max improvement is small, stop
        c5_now = evaluate_erdos_solution(h)
        if not equivalent:
            if c5_now > 0.382:
                return h_original  # do not apply if it does not lead to much improvement
            else:
                equivalent = True
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n0 = 120
    n1 = 240
    
    # Step 1: Generate initial binary vector
    h0 = generate_sturmian_binary(n0)
    
    # Step 2: Local search on coarse grid
    time_remaining = budget_s * 0.5
    h1, c5_1 = local_swap_search(h0, time_remaining)
    
    # Step 3: Upscale and refine
    h1_up = upscale_vector(h1)
    time_remaining = budget_s * 0.3
    h2, c5_2 = local_swap_search(h1_up, time_remaining)
    
    # Step 4: Optionally polish with continuous approach
    if budget_s - (time.time() - start_time) > 10:
        h_final = continuous_polish(h2, 300)
        final_c5 = evaluate_erdos_solution(h_final)
    else:
        h_final = h2
        final_c5 = c5_2
        
    return (h_final.tolist(), final_c5, len(h_final))
