import numpy as np
import time
import random


def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    overlap = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(overlap) * dx
    return c5


def project_to_box_simplex(v, target):
    # Project onto set {x: 0<=x<=1, sum x = target}
    # use bisection on lambda.
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        w = np.clip(v - lam, 0, 1)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo+hi)/2.0, 0, 1)


def gradient_smooth_max(h, beta):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, h, mode='full') * dx
    # numerical stability for softmax
    max_g = np.max(-beta * corr)
    exp_vals = np.exp(-beta * corr - max_g)
    w = exp_vals / np.sum(exp_vals)
    grad = np.zeros_like(h)
    # for each lag
    for lag in range(-(N-1), N):
        idx = lag + N - 1
        wk = w[idx]
        if wk == 0:
            continue
        if lag >= 0:
            grad[:N-lag] += wk * h[lag:]
            grad[lag:] += wk * h[:N-lag]
        else:
            k = -lag
            grad[k:] += wk * h[:N-k]
            grad[:N-k] += wk * h[k:]
    grad *= -dx
    return grad


def subgradient_step(h, step_size):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, h, mode='full') * dx
    # find worst shift (minimum correlation)
    worst_idx = np.argmin(corr)
    # compute shift value: lag = worst_idx - (N-1)
    lag = worst_idx - (N-1)
    # compute contributions
    grad = np.zeros_like(h)
    if lag >= 0:
        grad[:N-lag] += h[lag:]
        grad[lag:] += h[:N-lag]
    else:
        k = -lag
        grad[k:] += h[:N-k]
        grad[:N-k] += h[k:]
    grad *= dx
    # update
    h_new = h + step_size * grad
    # project
    h_new = project_to_box_simplex(h_new, target_sum=N/2)
    return h_new


def smooth_optimize(seed, time_budget, beta_start=1.0, beta_end=60.0, step_start=1.0):
    h = seed.copy()
    N = len(h)
    target = N/2
    best_h = h.copy()
    best_c5 = c5_of(h)
    start = time.time()
    iter = 0
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        t_frac = elapsed / time_budget
        beta = beta_start * (beta_end / beta_start) ** t_frac
        step = step_start * (1 - t_frac) + 1e-8
        grad = gradient_smooth_max(h, beta)
        h_new = h - step * grad
        h_new = project_to_box_simplex(h_new, target)
        c5 = c5_of(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
        iter +=1
    return best_h, best_c5


def subgradient_opt(h, time_budget, step_start=0.5, step_end=1e-4):
    target = len(h)/2
    best_h = h.copy()
    best_c5 = c5_of(h)
    start = time.time()
    while time.time() - start < time_budget:
        elapsed_frac = (time.time() - start) / time_budget
        step = step_start * (1 - elapsed_frac) + step_end * elapsed_frac
        # find worst lag
        N = len(h)
        dx = 2.0 / N
        corr = np.correlate(h, h, mode='full') * dx
        worst_idx = np.argmin(corr)
        lag = worst_idx - (N-1)
        grad = np.zeros_like(h)
        if lag >= 0:
            grad[:N-lag] += h[lag:]
            grad[lag:] += h[:N-lag]
        else:
            k = -lag
            grad[k:] += h[:N-k]
            grad[:N-k] += h[k:]
        grad = grad * dx
        h_new = h + step * grad
        h_new = project_to_box_simplex(h_new, target)
        c5 = c5_of(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        # always accept move for simplicity
        h = h_new
    return best_h, best_c5


def propose_move(h, max_step):
    """Propose a new feasible h vector by adjusting two elements"""
    h_new = h.copy()
    N = len(h)
    
    # Try to find a valid move
    max_tries = 100
    for _ in range(max_tries):
        i = random.randint(0, N - 1)
        j = random.randint(0, N - 1)
        if i == j:
            continue
            
        # Determine feasible interval for δ
        delta_low = max(-h_new[i], h_new[j] - 1)
        delta_high = min(1 - h_new[i], h_new[j])
        
        if delta_low >= delta_high:
            continue
            
        # Sample δ from uniform distribution over feasible range
        step = min(max_step, delta_high - delta_low)
        if step <= 0:
            continue
        delta = random.uniform(delta_low, delta_high)
        
        # Apply move
        h_new[i] += delta
        h_new[j] -= delta
        
        return h_new
    
    # If no valid move found, return original
    return h.copy()


def hill_climbing(h, time_budget):
    best_h = h.copy()
    best_c5 = c5_of(h)
    start = time.time()
    while time.time() - start < time_budget:
        h_new = propose_move(h, max_step=0.2)
        c5_new = c5_of(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
    return best_h, best_c5


def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function"""
    random.seed(seed)

    # Build seeds:
    seeds = []
    # Uniform
    seeds.append(np.full(N, 0.5))
    # Interpolated if available
    if 'initial_h_values' in globals():

        init = project_to_box_simplex(init.copy(), target)

    # Random continuous

    rand_cont = project_to_box_simplex(rand_cont, target)
    seeds.append(rand_cont)
    # Random binary
    binary = np.zeros(N)
    half = int(round(target))
    idx = np.random.choice(N, half, replace=False)
    binary[idx] = 1.0
    seeds.append(binary)
    
    # Determine per seed budget: allocate 70% of total budget to seeds
    time_for_seeds = budget_s * 0.7
    per_seed_budget = time_for_seeds / len(seeds)
    best_h = None
    best_c5 = np.inf
    
    for seed_vec in seeds:
        # Phase1: smoothing
        time_left = per_seed_budget
        # in each seed allocate smoothing and subgradient ratio: e.g. 30% smoothing, 70% subgradient
        smooth_time = time_left * 0.3
        subgrad_time = time_left * 0.7
        h_smooth, c5_smooth = smooth_optimize(seed_vec, smooth_time, beta_start=0.5, beta_end=50.0, step_start=1.0)
        h_sub, c5_sub = subgradient_opt(h_smooth, subgrad_time, step_start=0.5, step_end=1e-4)
        # record best
        if c5_sub < best_c5:
            best_c5 = c5_sub
            best_h = h_sub.copy()
    
    # Final polishing with hill climbing
    time_remaining = budget_s - (time.time() - start)
    if time_remaining > 5 and best_h is not None:
        best_h, best_c5 = hill_climbing(best_h, time_remaining)
    
    return (best_h, best_c5, N)
