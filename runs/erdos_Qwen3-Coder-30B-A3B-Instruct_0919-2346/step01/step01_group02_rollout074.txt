def evaluate_c5(h):
    """Compute the maximum overlap integral for the given step function h."""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    c = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(c) * dx
    return c5

def project_to_feasible(h, N):
    """Project h onto the feasible set: 0 <= h_i <= 1 and sum(h) = N/2."""
    # This is a water-filling algorithm
    # Build a vector of (value, index) pairs
    v = h.copy()
    # Define the target sum
    target_sum = N / 2.0
    # Try to strictly enforce the sum constraint with projection
    # Since we are under the scenario where the constraint is enforced
    # through re-normalization to mean of 1/2, let's simply ensure it's normalized correctly
    # We do it differently: for feasibility, we correct the sum
    current_sum = np.sum(v)
    if np.abs(current_sum - target_sum) < 1e-10:
        return v.copy()
    # Adjust the values to make sum correct and still in [0,1]
    # Water-filling
    def F(lam):
        return np.sum(np.clip(v + lam, 0, 1)) - target_sum
    
    # Binary search to find lambda such that F(lambda) = 0
    # F is increasing in lambda since clip is monotonic increasing
    lambda_min, lambda_max = -np.inf, np.inf
    for i in range(50):
        lambda_mid = (lambda_min + lambda_max) / 2.0
        f_val = F(lambda_mid)
        if f_val <= 0:
            lambda_min = lambda_mid
        else:
            lambda_max = lambda_mid
    # At this point, lambda_min approximately gives us the right value
    # We now make the final correction of link:
    # Make sure the result is feasible and exactly satisfying the sum
    lambda_opt = lambda_min
    h_proj = np.clip(v + lambda_opt, 0, 1)
    scale = target_sum / np.sum(h_proj)
    h_proj = h_proj * scale
    return h_proj

def smooth_objective(h, eps):
    """Evaluate smooth surrogate objective."""
    dx = 2.0 / len(h)
    c = np.correlate(h, 1 - h, mode='full')
    c_scaled = c * dx
    # log-sum-exp trick
    c_max = np.max(c_scaled)
    c_shifted = c_scaled - c_max
    exp_c = np.exp(c_shifted / eps)
    Z = np.sum(exp_c)
    # The value of the surrogate
    F = eps * np.log(Z) + c_max
    return F

def multistart_continuation(h_list, N, max_iter=100):
    """Multistart L-BFGS with smoothing continuation."""
    start_time = time.time()
    beta = 0.001  # final epsilon 
    eps_list = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, beta]
    epsilon = eps_list
    best_C5 = float('inf')
    best_h = None
    # Multi-start strategy
    for seed_num, h in enumerate(h_list):
        h_initial = h.copy()
        h_projected = project_to_feasible(h_initial, N)
        h_current = h_projected.copy()
        C5_current = evaluate_c5(h_current)
        if C5_current < best_C5:
            best_C5 = C5_current
            best_h = h_current.copy()
        for eps in epsilon:
            # Inner L-BFGS-B optimization
            # Define the bounds (keep feasible)
            bounds = Bounds(np.zeros(N), np.ones(N))
            # Optimize with projected gradient method after each LBFGS step
            for iter_lbfgs in range(150):
                if time.time() - start_time > 0.90 * 600:
                    break
                # Evaluate smooth objective and gradient
                F = smooth_objective(h_current, eps)
                grad = smooth_gradient(h_current, eps)
                # L-BFGS-B minimize Fi - projection
                h_after_step = h_current - 0.01 * grad
                h_projected = project_to_feasible(h_after_step, N)
                h_current = h_projected.copy()
                # Early termination when gradient norm small
                norm_grad = np.linalg.norm(grad) 
                if norm_grad < 1e-6:
                    break
                C5 = evaluate_c5(h_current)
                if C5 < best_C5:
                    best_C5 = C5
                    best_h = h_current.copy()
        # Project one last time to be sure
        try:
            h_projected = project_to_feasible(h_current, N) 
            final_C5 = evaluate_c5(h_projected)
            if final_C5 < best_C5:
                best_C5 = final_C5
                best_h = h_projected.copy()
        except Exception as e:
            print("Final projection failed in one start", e)

    return best_h, best_C5

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    N = 140
    dx = 2.0 / N
    # Create multiple seeds
    h_list = []
    
    # Uniform starting_h
    h_uniform = np.ones(N) * 0.5
    h_list.append(h_uniform)
    
    # Two peak: put two humps 
    h_two_peak = np.ones(N)
    x_vals = np.linspace(0, 2, N)
    a = 0.24  # Can be very small or large
    u = 0.9  # High value
    v = 0.1  # Low value
    mask = (x_vals < a) | (x_vals >= 2 - a)
    h_two_peak[mask] = u
    h_two_peak[~mask] = v
    h_two_peak = h_two_peak * (N/2) / np.sum(h_two_peak)  # Renormalize exactly
    h_list.append(h_two_peak)
    
    # Two symmetric blocks 
    h_block = np.zeros(N)
    m = N // 2  # Half size
    h_block[:m] = 1.
    h_block[m:] = 0.
    h_block = h_block * (N/2) / np.sum(h_block)
    h_list.append(h_block)

    # Uniform seed
    h_seed = np.random.rand(N)
    h_seed = h_seed * (N/2) / np.sum(h_seed)
    h_list.append(h_seed)
    
    # Use a precomputed initial solution if exists (otherwise continue)
    if 'initial_h_values' in globals():
        h_input = np.interp(np.linspace(0, 2, N), 
                           np.linspace(0, 2, len(initial_h_values)), 
                           initial_h_values)
        h_input = h_input * (N/2) / np.sum(h_input)  # Renormalize exactly
    else:
        h_input = np.ones(N) * 0.5  # fallback
    h_list.append(h_input)
    
    # Run optimization
    best_h, best_c5 = multistart_continuation(h_list, N) 
    
    # Final polishing using max-based sub-gradient
    start = time.time()
    while time.time() - start < 0.10 * budget_s:
        # Find worst shift s*
        c = np.correlate(best_h, 1 - best_h, mode='full')
        best_s = np.argmax(c)
        # Compute sub-gradient
        s_star = best_s - (N - 1)  # Convert from 0-based position to actual shift offset
        g = np.zeros_like(best_h)
        # Construct sub-gradient using manual computations
        # Use shift back to inverse
        right_shift = np.zeros_like(best_h) 
        left_shift = np.zeros_like(best_h)
        # Avoid confusion by manually computing handle shifts in vector
        if s_star > 0:
            left_shift[s_star:] = best_h[:-s_star]
            right_shift[:-s_star] = best_h[s_star:]
        elif s_star < 0:
            right_shift[-s_star:] = best_h[:s_star]
            left_shift[:s_star] = best_h[-s_star:]
        else:
            left_shift = best_h.copy()
            right_shift = best_h.copy()
        # scale dx
        dx = 2.0 / len(best_h)
        g = dx * (1 - left_shift - right_shift)
        # Step 
        h_new = best_h - 0.03 * g
        # Project to feasible
        h_new = project_to_feasible(h_new, N)
        c_new = evaluate_c5(h_new)
        if c_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c_new
        else:
            break  # Stop if no improvement
    
    return (best_h, best_c5, N)
