import numpy as np
import time
from scipy.optimize import differential_evolution, minimize

def c5_of(h):
    """Compute the maximum overlap integral for the given step function h."""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    c = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(c) * dx
    return c5

def evaluate_constraint(h, n_points):
    """Check if the constraint ∫₀² h(x) dx = 1 is satisfied."""
    dx = 2.0 / n_points
    integral = np.sum(h) * dx
    return abs(integral - 1.0)

def project_to_feasible(h, n_points):
    """Project h to satisfy the constraint ∫₀² h(x) dx = 1."""
    dx = 2.0 / n_points
    current_integral = np.sum(h) * dx
    if abs(current_integral - 1.0) > 1e-10:
        scale = 1.0 / current_integral
        h = h * scale
    return h

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 140
    dx = 2.0 / n_points
    
    # Start with initial solution if available
    if 'initial_h_values' in globals():
        h0 = np.interp(np.linspace(0, 2, n_points), 
                      np.linspace(0, 2, len(initial_h_values)), 
                      initial_h_values)
        h0 = h0 * (n_points/2) / np.sum(h0)  # Renormalize exactly
    else:
        # Create an initial good construction
        h0 = np.ones(n_points) * 0.5
        # Add a two-peak structure that tends to reduce overlap
        x_vals = np.linspace(0, 2, n_points)
        a = 0.3
        u = 0.9
        v = 0.1
        mask = (x_vals < a) | (x_vals >= 2 - a)
        h0[mask] = u
        h0[~mask] = v
        h0 = h0 * (n_points/2) / np.sum(h0)  # Renormalize exactly
        
    # Define bounds for each variable (0 to 1)
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define the objective function
    def objective(h):
        return c5_of(h)
    
    # Define constraint function
    def constraint(h):
        return np.sum(h) * dx - 1.0
    
    # Use differential evolution to explore global minimum
    start_time = time.time()
    
    # Run optimization
    result = differential_evolution(
        objective,
        bounds,
        seed=seed,
        maxiter=int(5000 * budget_s / 600),
        popsize=15,
        tol=1e-6,
        recombination=0.7,
        mutation=(0.5, 1.0),
        constraints={'type': 'eq', 'fun': constraint}
    )
    
    # Extract optimal solution
    h_opt = result.x
    h_opt = project_to_feasible(h_opt, n_points)
    c5_opt = c5_of(h_opt)
    
    # Fine-tune with local optimization if time allows
    if time.time() - start_time < 0.9 * budget_s:
        # Use local optimization to refine
        try:
            result_local = minimize(
                objective,
                h_opt,
                method='L-BFGS-B',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint},
                options={'maxiter': 1000}
            )
            h_opt_fine = project_to_feasible(result_local.x, n_points)
            c5_opt_fine = c5_of(h_opt_fine)
            if c5_opt_fine < c5_opt:
                h_opt = h_opt_fine
                c5_opt = c5_opt_fine
        except:
            pass

    # Final check
    h_opt = project_to_feasible(h_opt, n_points)
    c5_opt = c5_of(h_opt)
    
    return (h_opt, c5_opt, n_points)
