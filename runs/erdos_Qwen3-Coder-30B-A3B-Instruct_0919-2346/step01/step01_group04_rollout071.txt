import numpy as np
import time
import random
from scipy import signal


def compute_c5(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Use full correlation to get all shifts
    overlap = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(overlap) * dx
    return c5


def project_h(h, dx):
    """Project h to satisfy integral constraint and bounds"""
    h = np.clip(h, 0, 1)
    # Adjust to make integral = 1
    current_sum = np.sum(h) * dx
    if abs(current_sum - 1.0) > 1e-10:
        scale = 1.0 / current_sum
        h = h * scale
    return h


def gradient_step(h, step_size, beta=1.0):
    """Compute gradient and take a step"""
    N = len(h)
    dx = 2.0 / N
    
    # Compute h * (1 - h) for use in gradient estimates
    h1 = h * (1 - h)
    
    # Convolution with 1 - h

    # Convolution with -h
    cross_correlation2 = signal.convolve(h1, -np.ones(N), mode='full')[:2*N-1]
    # Get all shift values
    # Compute full correlation using fft and avoid overwriting 
    corr_full = signal.correlate(h, 1 - h, mode='full')
    
    # Compute the effective gradient (approximate)
    # For a shift k, we also have 1-h shifted left by k, looking at overlaps
    # The gradient part is more directly derived from finite difference or analytical
    # For now, let's compute a simplified gradient approximation based on:
    # d/dh C5 = max_k d/dh int h(x) (1 - h(x+k)) dx = max_k (1 - h(x+k))dy/dx
    # More simply, assume we know which shift gives max, and use finite differences:
    max_k_idx = np.argmax(corr_full)
    max_k = max_k_idx - (N - 1)
    
    # d/dh [ h(x) (1 - h(x+k)) ] = (1 - h(x+k)) - h(x+k) * dh/dx at x = x shift
    
    if max_k >= 0:
        grad = h - np.roll(1 - h, max_k)
    else:
        grad = h - np.roll(1 - h, max_k)
    
    # Use softmax weight version to approximate coarse direction:
    # weight becomes more concentrated as beta increases
    beta_factor = 1000.0 if beta > 1000 else beta
    weights = np.exp(beta_factor * (corr_full - np.max(corr_full))) 
    weights = weights / np.sum(weights)
    
    # Compute weighted gradient direction
    grad = np.zeros_like(h)
    for i in range(N):
        k = i - (N - 1)
        if 0 <= i < N:
            if k >= 0:
                grad[i] += weights[i] * (1 - h[i] if k == 0 else 1 - h[i-k])
            else:  # k < 0
                if i - k < N:
                    grad[i] += weights[i] * (1 - h[i-k] if i-k >= 0 else 1)
                else:
                    grad[i] += weights[i] * 1  # Add default if it rolls over (unbounded shift)
    
    # Just use a crude gradient to push values either up or down based on axis:
    grad = np.ones_like(h) - 2 * h  # Simplified direction: 
    # 1-2h favored, so sat values become essentially fixed
    
    h_new = h - step_size * grad
    h_new = project_h(h_new, dx)
    
    return h_new


def local_improve(h, time_budget_s, n_iter=1000):
    """Perform smooth local improvement without uphill steps"""
    start = time.time()
    
    c_current = compute_c5(h)
    h_current = h.copy()
    
    # We do a crude local search with HR-like adjustments
    # Random exchange moves if we find anyone violating bounds significantly
    for it in range(n_iter):
        if time.time() - start > time_budget_s:
            break
            
        try:
            # Pick random index
            i = random.randint(0, len(h) - 1)
            j = random.randint(0, len(h) - 1)
            if i == j:
                continue
            
            # Try small adjustment between i and j
            old_i, old_j = h[i], h[j]
            
            # Adjust values toward direction of reducing the worst shift
            # Assume cutoff changes by roughly 1/20 per step 
            mu = 0.01  # Adjustment scale
            old_c5 = compute_c5(h_current)
            
            h_new = h_current.copy()
            h_new[i] = old_i + mu * (1 - old_i * 2)  # Move towards middle
            h_new[j] = old_j - mu * (1 - old_j * 2)
            h_new = project_h(h_new, 2.0 / len(h_new))
            
            new_c5 = compute_c5(h_new)
            
            if new_c5 < old_c5:
                h_current = h_new.copy()
                c_current = new_c5
        except:
            pass
    
    return h_current, c_current

    h_binary = np.zeros(N)
    h_binary[0:N//3] = 1
    h_binary = project_h(h_binary, dx)

    # Seed 3: Two-hump pattern
    h_hump = np.full(N, 0.2)
    h_hump[20:60] = 0.8
    h_hump[140:180] = 0.8
    h_hump = project_h(h_hump, dx)

    # Run a sequence of primary optimizations:
    best_h = None
    best_c5 = float('inf')
    
    seeds = [h_provided, h_binary, h_hump]
    
    # Time division: spread evenly across seeds
    for i, h_seed in enumerate(seeds):
        remaining_time = budget_s - (time.time() - start)
        if remaining_time <= 5:
            break
            
        # Do single-phase gradient ascent
        h_cur = h_seed.copy()
        c_cur = compute_c5(h_cur)
        
        # Adaptive learning rate schedule
        num_iters = 1000
        step_size_schedule = [0.01 * (0.98 ** k) for k in range(num_iters)]
        beta_schedule = [min(100, 1 + 10 * k / num_iters) for k in range(num_iters)]
        
        for k in range(num_iters):
            if time.time() - start > budget_s:
                break
            t_elapsed = time.time() - start
            if t_elapsed > budget_s * 0.9:
                break
                
            step_size = step_size_schedule[k] * (1 - min(0.8*k/num_iters, 1))
            beta = beta_schedule[k]
            
            h_new = gradient_step(h_cur, step_size, beta)
            c_new = compute_c5(h_new)
            
            if c_new < c_cur:
                h_cur = h_new
                c_cur = c_new
                
        # After coarse gradient phase, do local improvement
        h_done, c_done = local_improve(h_cur, budget_s - (time.time() - start), 3000)  
        
        if c_done < best_c5:
            best_c5 = c_done
            best_h = h_done.copy()
    
    # Final polishing with SA-like refinement (no uphill)
    if best_h is not None and (time.time() - start) < budget_s - 10:
        final_time_left = budget_s - (time.time() - start) - 5
        if final_time_left > 0:
            # Do carefully tuned local moves, retry upon convergence
            h_final, c_final = local_improve(best_h, final_time_left, 2000)
            if c_final < best_c5:
                best_c5 = c_final
                best_h = h_final.copy()
                
    # Ensure final projection
    if best_h is not None:
        best_h = project_h(best_h, dx)
    
    return (best_h, best_c5, N)
