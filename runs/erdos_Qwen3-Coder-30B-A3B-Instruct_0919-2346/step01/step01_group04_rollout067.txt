import numpy as np
import time
import random
from scipy.optimize import differential_evolution, minimize


def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    overlap = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(overlap) * dx
    return c5


def enforce_constraints(h, n_points):
    """Enforce sum(h) * dx = 1, with dx = 2/n_points"""
    dx = 2.0 / n_points
    current_sum = np.sum(h)
    target_sum = n_points / 2
    scale_factor = target_sum / current_sum
    h_constrained = np.clip(h * scale_factor, 0, 1)
    return h_constrained


def propose_move(h, max_step):
    """Propose a new feasible h vector by adjusting two elements"""
    h_new = h.copy()
    N = len(h)
    
    # Try to find a valid move
    max_tries = 100
    for _ in range(max_tries):
        i = random.randint(0, N - 1)
        j = random.randint(0, N - 1)
        if i == j:
            continue
            
        # Determine feasible interval for δ
        delta_low = max(-h_new[i], h_new[j] - 1)
        delta_high = min(1 - h_new[i], h_new[j])
        
        if delta_low >= delta_high:
            continue
            
        # Sample δ from uniform distribution over feasible range
        step = min(max_step, delta_high - delta_low)
        if step <= 0:
            continue
        delta = random.uniform(delta_low, delta_high)
        
        # Apply move
        h_new[i] += delta
        h_new[j] -= delta
        
        # Ensure constraints are satisfied
        h_new = enforce_constraints(h_new, N)
        
        return h_new
    
    # If no valid move found, return original
    return h.copy()


def optimize_with_differential_evolution(initial_h, time_budget_s):
    """Try DE first to get a better starting point"""
    start = time.time()
    
    # Set bounds
    bounds = [(0, 1) for _ in range(len(initial_h))]
    
    def objective(h):
        return c5_of(h)
    
    try:
        # Use differential evolution with bounds
        result = differential_evolution(
            objective,
            bounds,
            maxiter=200,
            popsize=15,
            tol=1e-6,
            recombination=0.7,
            seed=42,
            callback=None
        )
        h = result.x
        h = enforce_constraints(h, len(initial_h))
        return h, c5_of(h)
    except:
        return initial_h, c5_of(initial_h)


def local_search(initial_h, time_budget_s):
    """Refine with local search starting from a good solution"""
    start = time.time()
    
    # Use refine with fixed-step greedy hill climbing
    # Propose from the initial_h across the spectrum
    
    cur = initial_h.copy()
    c_cur = c5_of(cur)
    
    best = cur.copy()
    c_best = c_cur
    
    # Main local search loop
    max_iter = 200000
    for iteration in range(max_iter):
        if time.time() - start > time_budget_s * 0.95:
            break
            
        # Use the same proposal mechanism but make it more local
        step = 0.05 * (1 - iteration/max_iter) + 0.005
        h_proposed = propose_move(cur, step)
        c_proposed = c5_of(h_proposed)
        
        if c_proposed < c_cur:
            c_cur = c_proposed
            cur = h_proposed
            if c_proposed < c_best:
                c_best = c_proposed
                best = h_proposed.copy()
        elif iteration % 100 == 0:  # Occasionally accept worse moves for escape
            if random.random() < 0.01:
                cur = h_proposed
                c_cur = c_proposed
                
    return best, c_best


def run(seed=42, budget_s=600, **kwargs):

    # 2. Step-like
    h_step = np.zeros(n_points)
    h_step[:n_points//3] = 0.8
    h_step[n_points//3:2*n_points//3] = 0.2
    h_step[2*n_points//3:] = 0.8
    h_step = enforce_constraints(h_step, n_points)

    # 3. Interpolated from initial_h_values if available

        h_provided = np.interp(

            initial_h_values
        )
        h_provided = enforce_constraints(h_provided, n_points)
    else:
        h_provided = np.ones(n_points) * 0.5
        
    # 4. Optimized version via DE to use as seed
    print("Starting differential evolution...")
    h_de, c5_de = optimize_with_differential_evolution(h_provided.copy(), budget_s * 0.4)
    print(f"DE found solution with C5 = {c5_de}")
    
    # 5. Find best known seed
    seeds = [h_sine, h_step, h_provided, h_de]
    best_h = None
    best_c5 = float('inf')

    time_per_seed = min(budget_s * 0.8 / len(seeds), 150)

    # Perform final refinement with more cycles

    if final_refinement_time > 10:
        h_final, c5_final = local_search(best_h, final_refinement_time)
