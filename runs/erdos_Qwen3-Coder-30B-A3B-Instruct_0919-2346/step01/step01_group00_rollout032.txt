import numpy as np
import time
import random
from scipy.optimize import differential_evolution, minimize

def c5_of(h):
    """Compute C5 = max_k integral h(x)(1-h(x+k)) dx."""
    dx = 2.0 / len(h)
    # Compute correlation: h * (1 - h) with full mode
    c = np.correlate(h, 1 - h, mode="full")
    # Multiply by dx to get continuous integral
    c *= dx
    # Return maximum value among all shifts
    return np.max(c)

def project_to_feasible(h):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) * dx = 1."""
    dx = 2.0 / len(h)
    h = np.clip(h, 0, 1)
    h = h * (1 / np.sum(h)) * (1 / dx)
    return h

def perturb_h(h, delta=0.05):
    """Apply a small perturbation to h."""
    h = h.copy()
    n = len(h)
    # Pick two random points
    i, j = random.sample(range(n), 2)
    # Determine valid range for delta
    delta_low = max(-h[i], 1.0 - h[j])
    delta_high = min(1.0 - h[i], h[j])
    
    if delta_low >= delta_high:
        return h
    
    # Ensure we don't violate bounds
    actual_delta = random.uniform(max(-delta, delta_low), min(delta, delta_high))
    
    h[i] += actual_delta
    h[j] -= actual_delta
    return h

def generate_initial_h(n):
    """Generate an initial feasible h."""
    # Start with the known good construction
    if len(initial_h_values) >= n:
        h = np.interp(np.linspace(0, 2, n),
                      np.linspace(0, 2, len(initial_h_values)),
                      initial_h_values)
    else:
        # If not enough points, we'll use a step function
        h = np.ones(n) * 0.5
    
    # Force rescaling to make integral = 1
    dx = 2.0 / n
    h = h * (1.0 / np.sum(h)) / dx
    
    # Ensure bounds
    h = np.clip(h, 0, 1)
    return h

def scipy_optimize_approach(h0, n_points, time_budget):
    """Try optimization with scipy."""
    start_time = time.time()
    
    # We will use a simple method to directly optimize
    # This approach is more structured than annealing
    
    def objective(h):
        # Reshape h into the right form
        h_reshaped = np.array(h)
        return c5_of(h_reshaped)
    
    # Constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Sum constraint (on the actual h values)
    def constraint_sum(h):
        dx = 2.0 / n_points
        return 1 - np.sum(h) * dx  # Should be 0
    
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    # Use differential evolution: it's robust and works well here
    try:
        result = differential_evolution(
            objective,
            bounds,
            constraints=constraints,
            maxiter=200,
            popsize=15,
            seed=42
        )
        
        if result.success and time.time() - start_time < 0.8 * time_budget:
            h_result = project_to_feasible(result.x)
            return h_result, c5_of(h_result)
    except Exception:
        pass
    
    # Fallback to basic optimization method
    return h0, c5_of(h0)

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    n_points = 140
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Generate the initial solution
    h_current = generate_initial_h(n_points)
    c_current = c5_of(h_current)
    
    # Track best solution
    h_best = h_current.copy()
    c_best = c_current
    
    # Try alternative approach with optimization
    try:
        h_opt, c_opt = scipy_optimize_approach(h_current, n_points, budget_s)
        if c_opt < c_best:
            h_best = h_opt
            c_best = c_opt
    except Exception:
        pass

    # Perform some additional perturbations and refining
    remaining_time = budget_s * 0.7
    i = 0
    while time.time() - start_time < remaining_time:
        # Try even more fine-tuning
        h_new = perturb_h(h_current, 0.03)
        h_new = project_to_feasible(h_new)
        c_new = c5_of(h_new)
        
        if c_new < c_current or random.random() < 0.3:
            h_current = h_new
            c_current = c_new
            
        if c_current < c_best:
            h_best = h_current.copy()
            c_best = c_current
            
        i += 1

    # Final polishing with more targeted approach
    polish_time = time.time() + (budget_s * 0.1)
    while time.time() < polish_time:
        h_candidate = perturb_h(h_best, 0.02)
        h_candidate = project_to_feasible(h_candidate)
        c_candidate = c5_of(h_candidate)
        
        if c_candidate < c_best:
            h_best = h_candidate
            c_best = c_candidate

    return (h_best, c_best, n_points)
