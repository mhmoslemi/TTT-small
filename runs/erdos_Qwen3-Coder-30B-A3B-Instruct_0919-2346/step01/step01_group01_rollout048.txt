import numpy as np
import time
import random
from scipy.optimize import differential_evolution, minimize

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using full convolution
    overlap = np.correlate(h, 1 - h, mode='full')
    # The maximum value scaled by dx
    c5 = np.max(overlap) * dx
    return c5

def propose_move(h, max_step):
    """Propose a move that maintains sum constraint exactly"""
    h_new = h.copy()
    N = len(h)
    while True:
        i, j = random.sample(range(N), 2)
        # Compute feasible range for delta
        delta_low = max(-h_new[i], h_new[j] - 1.0)
        delta_high = min(1.0 - h_new[i], h_new[j])
        if delta_low >= delta_high:
            continue  # No valid move, try again
        break
    
    # Sample delta from feasible interval
    step = min(max_step, delta_high - delta_low)
    if step <= 0:
        delta = 0.0
    else:
        delta = random.uniform(max(-step, delta_low), min(step, delta_high))
        
    h_new[i] += delta
    h_new[j] -= delta
    return h_new

def solve_with_differential_evolution(h_init, budget_s):
    """Use differential evolution for global optimization"""
    start = time.time()
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(len(h_init))]
    
    def objective(h):
        # Ensure constraint sum(h) = 1
        dx = 2.0 / len(h)
        h_scaled = h * (1.0 / np.sum(h) * len(h) / 2)  
        return c5_of(h_scaled)
        
    # Use differential evolution with constraints
    result = differential_evolution(
        objective,
        bounds,
        maxiter=int(budget_s * 10),  # Adjust based on time budget
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.7,
        seed=42,
        disp=False,
        callback=lambda x, convergence: time.time() - start > budget_s
    )
    
    best_h = result.x * (1.0 / np.sum(result.x) * len(result.x) / 2)
    return best_h, c5_of(best_h)

def advanced_annealing(initial_h, time_budget):
    """Enhanced simulated annealing with adaptive steps"""
    start = time.time()
    cur = initial_h.copy()
    best = initial_h.copy()
    c_cur = c5_of(cur)
    c_best = c_cur
    
    # Annealing parameters
    T_start = 0.1
    T_end = 1e-6
    step_start = 0.3
    step_decay = 0.99
    
    # Track improvement
    last_improvement = 0
    patience = 50
    
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T_start * (T_end / T_start) ** (elapsed / time_budget)
        step = step_start * (step_decay ** (elapsed / time_budget))
        
        # Adaptive step size
        if elapsed - last_improvement > 15:
            step *= 0.9
            
        h_prop = propose_move(cur, step)
        c_prop = c5_of(h_prop)
        
        if c_prop < c_best:
            best = h_prop.copy()
            c_best = c_prop
            last_improvement = elapsed
            
        # Metropolis acceptance
        if c_prop <= c_cur or random.random() < np.exp(-(c_prop - c_cur) / T):
            cur = h_prop
            c_cur = c_prop
            
        # Early stopping if no improvement
        if elapsed - last_improvement > patience:
            break
            
    return best, c_best
