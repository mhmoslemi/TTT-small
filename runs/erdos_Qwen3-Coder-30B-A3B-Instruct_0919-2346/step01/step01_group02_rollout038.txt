import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize_scalar

def waterfill(h, target_sum):
    """Project h onto the simplex with sum = target_sum using waterfilling."""
    h = np.asarray(h)
    # We want sum(h_proj) = target_sum and minimize sum((h_proj - h)^2)
    # Equivalent to: compute lambda such that sum clip(h + lambda, 0, 1) = target_sum
    def f(lambda_val):
        return np.sum(np.clip(h + lambda_val, 0, 1)) - target_sum

    # Binary search for lambda
    low, high = -2, 2
    for _ in range(50):
        mid = (low + high) / 2
        val = f(mid)
        if val > 0:
            high = mid
        else:
            low = mid
    lambda_opt = (low + high) / 2
    h_proj = np.clip(h + lambda_opt, 0, 1)
    return h_proj

def build_lp_model(N, dx, symmetry=False, monotonic=False):
    # Build variables
    h = cp.Variable(N)
    y = cp.Variable(N * (N + 1) // 2)  # One y for each (i,s) pair where 0 <= i <= N-1-s
    alpha = cp.Variable()

    # Build index structure for pairs (i,s)
    pairs = []
    y_idx_map = {}
    for i in range(N):
        for s in range(N - i):
            pairs.append((i, s))
            y_idx_map[(i, s)] = len(pairs) - 1
    num_pairs = len(pairs)

    # Add constraints
    constraints = []

    # Box constraints
    constraints.extend([h[i] >= 0 for i in range(N)])
    constraints.extend([h[i] <= 1 for i in range(N)])

    # Total mass constraint
    constraints.append(cp.sum(h) == N / 2)

    # McCormick constraints for each y[i,s]
    for (i, s) in pairs:
        j = i + s
        idx = y_idx_map[(i, s)]
        constraints.append(y[idx] <= h[i])
        constraints.append(y[idx] <= h[j])
        constraints.append(y[idx] >= h[i] + h[j] - 1)
        constraints.append(y[idx] >= 0)

    # Alpha constraints for each shift (s from 0 to N-1)
    # For shift s:
    # C_s = dx * sum(h[i] - y[i,s]) = dx * (sum(h) - sum(y[i,s]))
    # sum(h) = N/2 fixed
    # sum(y[i,s]) is sum over i=0..N-1-s of y[i,s]
    for s in range(N):
        # Compute the sum of y[i,s] for i=0..N-1-s
        sum_y_s = cp.sum([y[y_idx_map[(i, s)]] for i in range(N - s)])
        constraints.append(alpha >= dx * (N/2 - sum_y_s))

    # Symmetry constraint
    if symmetry:
        for i in range(N // 2):
            constraints.append(h[i] == h[N-1-i])

    # Monotonicity constraint (non-increasing from endpoints)
    if monotonic:
        for i in range(N // 2 - 1):
            constraints.append(h[i] >= h[i+1])

    # Objective
    objective = cp.Minimize(alpha)

    # Create problem
    prob = cp.Problem(objective, constraints)
    return prob, h, y, alpha, y_idx_map

def solve_lp(N, dx, symmetry=False, monotonic=False, verbose=False):
    try:
        prob, h, y, alpha, y_idx_map = build_lp_model(N, dx, symmetry, monotonic)
        prob.solve(solver=cp.ECOS, verbose=verbose)
        if prob.status in ["optimal", "optimal_inaccurate"]:
            h_val = h.value
            return h_val, alpha.value
        else:
            return None, None
    except Exception as e:
        if verbose:
            print(f"LP solve failed with error: {e}")
        return None, None

def c5_of(h):
    """Compute the maximum overlap integral for the given step function h."""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    c = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(c) * dx
    return c5

def propose_move_binary(h, max_step=0.05):
    """Propose a new feasible binary step function by adjusting two entries."""
    N = len(h)
    while True:
        i, j = np.random.choice(N, size=2, replace=False)
        # Determine feasible step
        delta_low = max(-h[i], 1 - h[j])
        delta_high = min(1 - h[i], h[j])
        if delta_low >= delta_high:
            continue  # Try another pair
        # Sample step size
        step_size = np.random.uniform(max(-max_step, delta_low),
                                      min(max_step, delta_high))
        # Create new h
        h_new = h.copy().astype(float)
        h_new[i] += step_size
        h_new[j] -= step_size
        # Project back to binary
        h_new = np.round(h_new).astype(int)
        # Ensure sum is exactly N/2
        sum_h = np.sum(h_new)
        if sum_h == N / 2:
            return h_new
        # If not, adjust the rounding
        diff = int(sum_h - N / 2)
        if diff != 0:
            # Find index to flip
            if diff > 0:
                # Find indices that are 1 to flip to 0
                candidates = np.where(h_new == 1)[0]
                if len(candidates) > 0:
                    idx = np.random.choice(candidates)
                    h_new[idx] = 0
        return h_new

def local_optimization_step(h0, max_iter=50):
    h = h0.copy()
    for it in range(max_iter):
        c5 = c5_of(h)
        # Identify worst shift
        dx = 2.0 / len(h)
        # Flip the sign of h (1-h)
        c = np.correlate(h, 1 - h, mode='full')
        worst_shift_idx = np.argmax(c)
        worst_shift = worst_shift_idx - len(h) + 1  # Convert index to shift

        # Compute gradient
        grad = np.zeros_like(h)
        # Gradient to max overlap is the derivative of correlation
        # Approximate by finite differences to flip sign of derivative
        # Let's use a simple way: adjust the elements being part of worst correlation
        if worst_shift > 0:
            for i in range(len(h) - worst_shift):
                grad[i] += (1 - h[i + worst_shift]) - h[i]  # Derivative of y[i,s] = h[i]*h[i+s] 
        else:
            for j in range(-worst_shift, len(h)):
                i = j + worst_shift
                if i >= 0:
                    grad[i] += (1 - h[j]) - h[i]
        # Simple gradient ascent 
        step_size = 0.02
        h_new = h + step_size * grad
        # Project back  
        h_new = np.clip(h_new, 0, 1)
        h_new = waterfill(h_new, len(h) / 2)
        if c5_of(h_new) < c5:
            h = h_new
        else:
            break
    return h
