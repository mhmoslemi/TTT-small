import numpy as np
import time
import random
from scipy.optimize import differential_evolution, minimize
import math

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    overlap = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(overlap) * dx
    return c5

def propose_move(h, max_step):
    """Propose a new feasible h vector by adjusting two elements"""
    h_new = h.copy()
    N = len(h)
    
    # Try to find a valid move
    max_tries = 100
    for _ in range(max_tries):
        i = random.randint(0, N - 1)
        j = random.randint(0, N - 1)
        if i == j:
            continue
            
        # Determine feasible interval for δ
        delta_low = max(-h_new[i], h_new[j] - 1)
        delta_high = min(1 - h_new[i], h_new[j])
        
        if delta_low >= delta_high:
            continue
            
        # Sample δ from uniform distribution over feasible range
        step = min(max_step, delta_high - delta_low)
        if step <= 0:
            continue
        delta = random.uniform(delta_low, delta_high)
        
        # Apply move
        h_new[i] += delta
        h_new[j] -= delta
        
        return h_new
    
    # If no valid move found, return original
    return h.copy()

def sample_h(h_start, n):
    """Sample from continuous function at discrete points"""
    x = np.linspace(0, 2, n)
    return np.interp(x, np.linspace(0, 2, len(h_start)), h_start)

def perturb_step(h, step_size):
    """Make small random perturbations to h"""
    h_new = h.copy()
    n = len(h)
    # Add small random movements to many points
    for i in range(n):
        # Perturb by small amount
        delta = np.random.normal(0, step_size)
        # Ensure saturation constraints - project back to feasible region
        h_new[i] = max(0, min(1, h_new[i] + delta))
    
    # Manual renormalization for numerical stability
    h_new = h_new * (len(h_new) / 2) / np.sum(h_new)
    return h_new

def improved_anneal(initial_h, time_budget_s):
    """Run improved simulated annealing with better strategies"""
    start = time.time()
    
    # Parameters
    T_start = 0.1
    T_end = 1e-6
    step_start = 0.1
    
    # Set up the current state
    cur = initial_h.copy()
    c_cur = c5_of(cur)
    
    best = cur.copy()
    c_best = c_cur
    
    # Main loop
    while time.time() - start < time_budget_s:
        t = time.time() - start
        elapsed_fraction = t / time_budget_s
        
        # Cooling schedule
        T = T_start * (T_end / T_start) ** elapsed_fraction
        step = step_start * (1 - elapsed_fraction) * 0.5
        
        # Alternating types of moves
        if random.random() < 0.7:
            # Standard two-point adjustment
            h_proposed = propose_move(cur, step)
        else:
            # Random perturbation
            h_proposed = perturb_step(cur, step*0.5)
            
        c_proposed = c5_of(h_proposed)
        
        # Accept the proposal
        if c_proposed <= c_cur:
            cur = h_proposed
            c_cur = c_proposed
            if c_proposed < c_best:
                best = h_proposed.copy()
                c_best = c_proposed
        else:
            # Metropolis acceptance rule
            if random.random() < np.exp(-(c_proposed - c_cur) / T):
                cur = h_proposed
                c_cur = c_proposed
                
    return best, c_best

def find_suitable_initials(n):
    """Generate multiple starting points for optimization"""
    initials = []
    
    # 1. Uniform
    initials.append(np.ones(n) * 0.5)
    
    # 2. Provided initial_h_values (if exists)
    if 'initial_h_values' in globals():
        h_provided = np.interp(
            np.linspace(0, 2, n),

            initial_h_values
        )
        h_provided = h_provided * (n / 2) / np.sum(h_provided)
        initials.append(h_provided)
    
    # 3. Step-like pattern
    h_step = np.full(n, 0.2)
    h_step[int(0.2*n):int(0.6*n)] = 0.8
    h_step = h_step * (n / 2) / np.sum(h_step)
    initials.append(h_step)
    
    # 4. Double-peak
    h_double = np.full(n, 0.1)
    a = 0.3
    u = 0.9
    v = 0.1
    left = int(a * n / 2)
    right = int((2 - a) * n / 2)
    h_double[:left] = u
    h_double[right:] = u
    h_double = h_double * (n / 2) / np.sum(h_double)
    initials.append(h_double)
    
    # 5. Single peak (Gaussian-like)
    x = np.linspace(0, 2, n)
    sigma = 0.2
    center = 1.0
    h_peak = np.exp(-0.5 * ((x - center) / sigma)**2)
    h_peak = h_peak * (n / 2) / np.sum(h_peak)
    initials.append(h_peak)
    
    return initials

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function"""
    random.seed(seed)
    np.random.seed(seed)
    
    start = time.time()
    n_points = 150  # Slightly more resolution
    dx = 2.0 / n_points
    
    # Find good starting solutions
    initials = find_suitable_initials(n_points)
    
    best_h = None
    best_c5 = float('inf')
    
    # Run multiple restarts with different strategies
    remaining_time = budget_s * 0.90
    time_per_restart = min(remaining_time / len(initials), 100)
    
    for i, h_seed in enumerate(initials):
        if time.time() - start >= budget_s * 0.95:
            break
            
        # Run optimization
        h_opt, c5_opt = improved_anneal(h_seed, time_per_restart)
        
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    # Post-processing: attempt gradient-based local optimization around best
    if best_h is not None and time.time() - start < budget_s - 10:
        # Refine using scipy minimize, only if we found a good solution
        try:
            # Define objective for scipy minimize (the negative of our C5 since we want to minimize)
            def objective(h):
                return c5_of(h)
            
            # Use boundary constraints
            bounds = [(0, 1) for _ in range(n_points)]
            
            # Due to solver limitations for high-dim nonconvex problems, 
            # do a simpler approach with bound constraints and contractive scaling
            
            # Just try a few gradient-based steps on top of best solution
            current_h = best_h.copy()
            current_c5 = c5_of(current_h)
            
            step_size = 0.01  # small step
            
            for _ in range(500):
                if time.time() - start >= budget_s - 5:

                # Small random steps with constraint checking
