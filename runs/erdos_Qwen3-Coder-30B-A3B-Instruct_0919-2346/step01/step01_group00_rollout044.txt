import numpy as np
import time
import random

def project_h(h, target_sum):
    """Project h onto [0,1] and sum = target_sum using water-filling."""
    h = h.copy()
    h = np.clip(h, 0, 1)
    current = np.sum(h)
    if np.isclose(current, target_sum):
        return h
    if current > target_sum:
        # Need to reduce some entries
        # Binary search for lambda
        low, high = 0, 1.0
        while high - low > 1e-10:
            mid = (low + high) / 2
            h_test = np.clip(h - mid, 0, 1)
            if np.sum(h_test) < target_sum:
                high = mid
            else:
                low = mid
        lambda_ = low
        h = np.clip(h - lambda_, 0, 1)
    else:
        # Need to increase some entries
        diff = target_sum - current
        # Distribute diff among entries that are not 1
        indices = np.where(h < 1)[0]
        if len(indices) == 0:
            return h
        # Assign diff equally
        diff_per = diff / len(indices)
        h[indices] = np.minimum(h[indices] + diff_per, 1)
    return h

def compute_C_and_grad(h, alpha):
    """Compute C vector and gradient with smoothed max."""
    # Compute correlation of h with (1-h) via FFT
    N = len(h)
    dx = 2.0 / N
    # Use full mode correlation
    c = np.correlate(h, 1 - h, mode='full')
    C = c * dx
    # Compute smoothed max (log-sum-exp)
    w = np.exp(alpha * C)
    w = w / np.sum(w)
    # Compute gradient: -dx * sum_i w_k * (h_{i+k} + h_{i-k})
    # Use convolution to compute v_plus = sum w_k * h_{i+k} and v_minus = sum w_k * h_{i-k}
    # w_rev = w[::-1]
    # v_plus = np.convolve(h, w_rev, mode='same')
    # v_minus = np.convolve(h, w, mode='same')
    # Note: np.convolve assumes zero-padding
    # Actually let's do more carefully compute w_k * h_{i+k}
    v_plus = np.zeros(N)
    v_minus = np.zeros(N)
    for k in range(-N+1, N):
        k_idx = k + N - 1
        if 0 <= k_idx < 2*N-1:
            w_k = w[k_idx]
            # k represents shift; need to sum h_{i+k} with appropriate boundary handling
            # For positive shift: h_{i+k} if i+k < N, else 0
            # For negative shift: h_{i+k} if i+k >= 0, else 0
            # Use np.roll for more efficient:
            if k >= 0:
                v_plus += w_k * np.roll(h, -k)
            else:
                v_plus += w_k * np.roll(h, -k)
    # Similarly for v_minus
    for k in range(-N+1, N):
        k_idx = k + N - 1
        if 0 <= k_idx < 2*N-1:
            w_k = w[k_idx]
            if k >= 0:
                v_minus += w_k * np.roll(h, k)
            else:
                v_minus += w_k * np.roll(h, k)
    # Or more simply: compute convolutions (but be careful of indexing)
    # It's easier to use convolve with full mode and extract the right portion:
    # Blunt approach: from full correlation compute overlaps:
    v_minus = np.convolve(h, w, mode='same')
    v_plus = np.convolve(h, w[::-1], mode='same')
    grad = dx * (1 - (v_plus + v_minus))
    # C_max = np.max(C)
    # Just return the gradient and C
    return C, grad
