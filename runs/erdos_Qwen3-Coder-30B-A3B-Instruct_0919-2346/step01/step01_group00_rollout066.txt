import numpy as np
import time
import random
from scipy.fft import rfft, irfft

def compute_C_vector(h):
    """Compute vector of overlaps C_tau = 1 - dx * S_tau(h) as a function of shift."""
    n = len(h)
    dx = 2.0 / n
    # Autocorrelation via FFT. Zero-pad to length 2n-1 to get full correlation
    h_pad = np.pad(h, (0, n - 1), mode='constant', constant_values=0)
    S = irfft(rfft(h_pad) * rfft(h_pad.conj()), n=2*n-1)
    C = 1 - dx * S[:2*n-1]
    return C

def compute_softmax_weights(c, alpha):
    """Compute softmax weights w_i = exp(alpha*c_i) / sum(exp(alpha*c_j))."""
    c_max = np.max(c)
    exp_c = np.exp(alpha * (c - c_max))
    return exp_c / np.sum(exp_c)

def compute_gradient(h, w):
    """Compute normalized gradient for smoothed objective using convolutions."""
    # Flip the weights for the second convolution (i->i-tau shift)
    w_rev = w[::-1]
    n = len(h)
    # Use real FFT for efficiency, mode='same' handles edge effects
    g1 = irfft(rfft(w, n=2*n-1) * rfft(h, n=2*n-1), n=2*n-1)
    g2 = irfft(rfft(w_rev, n=2*n-1) * rfft(h, n=2*n-1), n=2*n-1)
    # Extract the relevant part (proper indices for same-size convolution)
    g = -(g1[:n] + g2[:n])
    return g

def project_onto_box_and_mass(h, target_mass):
    """Project h onto [0,1] and enforce exact mass."""
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Water-filling to enforce sum = target_mass
    current_sum = np.sum(h)
    if abs(current_sum - target_mass) < 1e-12:
        return h
    # Linear search for lambda
    h_sorted = np.sort(h)[::-1]  # descending
    h_cumsum = np.cumsum(h_sorted)
    n = len(h)
    # Find index where sum > target_mass
    for i in range(n):
        if i == 0:
            s = h_sorted[i]
        else:
            s = h_sorted[i] + (h_sorted[i-1] - h_sorted[i]) * (i - 1) + h_cumsum[i-1]
        if s > target_mass:
            break
    else:
        i = n
    if i == 0:
        threshold = h_sorted[0] + (target_mass - h_cumsum[-1]) / n
    else:
        # Interpolate
        if i == 1:
            threshold = h_sorted[0] + (target_mass - h_sorted[0]) / n
        else:
            threshold = h_sorted[i - 1] - (h_cumsum[i-1] - target_mass) / (n - i + 1)
    x = np.clip(h - threshold, 0, 1)
    return x
