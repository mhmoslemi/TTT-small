import numpy as np
import time
from scipy.optimize import differential_evolution, minimize

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral via full convolution of h with (1-h)
    overlaps = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum of the full correlation vector
    # scaled by dx
    c5 = np.max(overlaps) * dx
    return c5

def modify_h(h, delta):
    """Apply small perturbations to h while respecting bounds and integral constraint."""
    h_new = h.copy()
    n = len(h)
    # Select a random point and perturb it
    i = np.random.randint(0, n)
    new_val = h[i] + delta
    # Ensure new value is in [0, 1]
    new_val = np.clip(new_val, 0, 1)
    # Adjust other values to preserve integral constraint
    diff = new_val - h[i]
    # Distribute the difference among other points randomly
    if abs(diff) > 1e-10:
        for j in np.random.permutation(n):
            if j != i:
                # For simplicity, use one point for adjustment
                h_new[j] = np.clip(h_new[j] - diff / (n - 1), 0, 1)
                break
    h_new[i] = new_val
    # Ensure that integral constraint is approximately satisfied
    h_new *= (len(h) / 2) / np.sum(h_new)
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start = time.time()
    N = 140
    dx = 2.0 / N

    # Start with initial construction if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        if len(h0) != N:
            x0 = np.linspace(0, 2, len(h0))
            x_new = np.linspace(0, 2, N)
            h0 = np.interp(x_new, x0, h0)
        h0 = h0 * (N / 2) / np.sum(h0)
    else:
        # Generate random initial function
        h0 = np.random.rand(N)
        h0 *= (N / 2) / np.sum(h0)

    # Initial solution
    best_h = h0.copy()
    best_c5 = c5_of(best_h)

    # Try a different approach: optimize using a different optimizer
    remaining_time = budget_s - (time.time() - start)
    if remaining_time > 20:
        # Use scipy.optimize differential evolution
        bounds = [(0, 1)] * N
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) * dx - 1.0}]

        # Function to minimize for differential evolution
        def obj(x):
            return c5_of(x)

        try:
            result = differential_evolution(
                obj,
                bounds,
                constraints=constraints,
                maxiter=500,
                seed=seed,
                disp=False
            )
            if not result.success:
                h_opt = best_h
            else:
                h_opt = result.x
                h_opt *= (N / 2) / np.sum(h_opt)
                c5_opt = c5_of(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except Exception:
            pass

        remaining_time = budget_s - (time.time() - start)
        if remaining_time > 10:
            # Try a hill climbing version - gradient-free optimization
            h_cur = best_h.copy()
            c_cur = best_c5
            T = 0.01
            n_steps = 0
            max_steps = 50000  # roughly 600s with ~0.01s per step

            while n_steps < max_steps and time.time() - start < budget_s - 1:
                # Make a small random displacement
                h_new = modify_h(h_cur, np.random.uniform(-T, T))
                c_new = c5_of(h_new)
                if c_new < c_cur or np.random.rand() < np.exp((c_cur - c_new) / T):
                    h_cur = h_new
                    c_cur = c_new
                if c_new < best_c5:
                    best_c5 = c_new
                    best_h = h_cur.copy()
                n_steps += 1
                # Reduce temperature
                T = max(0.001, T * 0.9999)

    return (best_h, best_c5, N)
