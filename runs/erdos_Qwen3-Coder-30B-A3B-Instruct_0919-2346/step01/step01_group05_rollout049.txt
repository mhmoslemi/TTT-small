import numpy as np
import time
from math import sqrt

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value scaled by dx
    max_overlap = np.max(correlation) * dx
    return max_overlap

def generate_sturmian_binary(n):
    """Generate a balanced binary sequence using Sturmian word with golden ratio"""
    phi = (sqrt(5) - 1) / 2
    h = np.zeros(n, dtype=int)
    
    # Generate the Sturmian word with parameter phi
    for i in range(n):
        t = (i * phi) % 1
        h[i] = 1 if t < 0.5 else 0
    
    # Adjust to have exactly n/2 ones
    ones_count = np.sum(h)
    if ones_count > n // 2:
        # Turn some 1s to 0s
        indices = np.where(h == 1)[0]
        for i in range(ones_count - n // 2):
            h[indices[i]] = 0
    elif ones_count < n // 2:
        # Turn some 0s to 1s
        indices = np.where(h == 0)[0]
        for i in range(n // 2 - ones_count):
            h[indices[i]] = 1
            
    return h

def compute_g_s(h):
    """Pre-compute the overlap sum for each shift. Returns array of g_s values."""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h): More efficient for each shift
    correlation = np.correlate(h, 1 - h, mode="full")
    return correlation * dx

def compute_delta_s(h, g_s, i, j, s_max):
    """Compute the change in g_s for all s when swapping h[i] and h[j]"""
    delta = np.zeros(len(g_s))
    
    # Find all shifts s where h[i] affects the overlap: s in [0, n-i) if i < n and s in [0, i) if i > 0
    # For (i+k) is affected by h[i], and position k is in 0..i-1 if trying to find where i is added in the formula:
    # g_s = sum_i h_i * (1 - h_{i+s})
    # If we flip h[i] from 1 to 0:
    # Contribution for shift s is - (1 - h_{i+s})
    # Consider all positions where i appears in the sum, and where i+s appears: 
    # If s = i + k => k = s - i, so the position is k which appears in overlap term with i:
    # i appears in terms involving h_{i+s} (where i+s < n) and h_{k} (where k appears in the sum).
    # Let's be more precise: 
    # h[i] contributes to terms g_s for all shifts where i appears, i.e., where we have h_i*(1-h_{i+s}). 
    # Breaking the formula: 
    # In g_s = sum h_i (1 - h_{i+s}) over all valid i, we care about:
    # 1) The term where i is current term (h_i)  - then it contributes h_i*(1-h_{i+s})
    # 2) The term where i+s is the current factor (i+s) - then it contributes (1-h_{i+s})*h_{i+s} but we have h_{i+s} in formula so we want contribution of h_{i+s} to all the terms
    # 
    # Let's compute delta correctly:
    # As h[i] is flipped:
    # - i contributes - (1 - h_{i+s}) to each shift (if i+s < n)
    # - h_{i+s} contributes + h_i to each shift (if i+s < n) from (i+s) term
    # - as h[j] is flipped from 0 to 1:
    #   - j contributes + (1 - h_{j+s}) to each shift  (if j+s < n)
    #   - h_{j+s} contributes - h_j = 0 to each shift (if j+s < n) 
    
    # Update Efficiently: break into blocks
    # The change type for a shift s due to swap (i,j):
    # - If i in range of overlap with s (i in [0, n-s)), contribution changes from + h_i * (1 - h_{i+s}) to + h_i' * (1 - h_{i+s}) = 0 * (1 - h_{i+s}) = 0
    # - If j in range of overlap with s (j in [0, n-s)), contribution changes from + h_j * (1 - h_{j+s}) to + h_j' * (1 - h_{j+s}) = 1 * (1 - h_{j+s}) = 1 - h_{j+s} 
    # That is, change due to i:  
    #   term =  0 - h_i*(1 - h_{i+s}) = -(1-h_{i+s})  if i+s < n
    #   term =  0 - h_{i+s-i}*(1 - h_{s}) = -(1-h_{s}) if s+i >= n, and i in [0,n-1], then term is sum from poses where we compute g_s 
    # The term contributes to sum of pair s.
    # Let's write overflow cases blow
    
    for s in range(len(g_s)):
        if i+s < n:
            delta[s] -= (1 - h[i+s])   # h_i contribution (before: 1, now 0)
        if j+s < n:
            delta[s] += (1 - h[j+s])   # h_j contribution (before: 0, now 1)
        # But we're double-counting contributions from first and second terms
        # This change is from h_i being active in (i+s) and h_j being active in (j+s)
        # That influence is:
        #   h[i] * (1 - h[i+s]) --> 0 after flip
        #   h[j] * (1 - h[j+s]) --> 1 - h[j+s] after flip 
        # So delta(s) = (1 - h[j+s]) - (1 - h[i+s])
        #  = h[i+s] - h[j+s] but we are global variable, so instead we must realize 
        # when we do h[i] becomes 0, h[j] becomes 1, it is g_s for all s.
        # Actually, let me recompute this more systematically:
        
        # Let assume delta due to swap of i, j at s 
        # Old g_s = sum over all i: h[i]*(1-h[i+s])
        # New g_s = sum h'[i] * (1 - h'[i+s])  = (h[i] + dh[i]) * (1 - h[i+s] - dh[i+s]) | dh[i] = -1, dh[j] = +1, dh[k] = 0
    
    return delta

def local_swap_search(h0, n_points, budget_s):
    """Local search using targeted swaps"""
    if n_points < 20:
        n_points = 20
    if n_points > 300:
        n_points = 300
    
    dx = 2.0 / n_points
    h = h0.astype(float)  # Convert to float for continuous optimization later
    
    # Compute g_s for all shifts
    g_s = compute_g_s(h)
    current_max = np.max(g_s)
    
    # Track best solution
    best_h = h.copy()
    best_max = current_max
    
    # Track recently swapped pairs to avoid cycles
    tabu_list = set()
    tabu_size = 100
    
    iteration = 0
    start_time = time.time()
    
    # Optimization parameters
    max_swaps = 100000
    swaps_since_improvement = 0
    max_no_improvement = 2000
    perturbation_interval = 1000
    block_flip_size = max(1, n_points // 20)  # ~5% size
    
    while iteration < max_swaps and (time.time() - start_time) < budget_s:
        # Find current worst shift
        s_max = np.argmax(g_s) - n_points + 1  # Convert index to shift
        
        # Identify good 0s and 1s to swap
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
            
        # Pick a random 1 and 0 to swap
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Skip if swap is tabu
        if (i, j) in tabu_list:
            iteration += 1
            continue
            
        # Compute numerical changes
        original_i = h[i]
        original_j = h[j]
        
        h[i] = 0
        h[j] = 1
        new_g_s = compute_g_s(h)  # Expensive but necessary for precision
        new_max = np.max(new_g_s)
        
        # Accept if better
        if new_max < current_max:
            # Update info
            g_s = new_g_s
            current_max = new_max
            best_h = h.copy()
            best_max = current_max
            swaps_since_improvement = 0
        else:
            # Revert
            h[i] = original_i
            h[j] = original_j
            # Reuse old g_s computed earlier
        
        # Add to tabu list
        tabu_list.add((i, j))
        if len(tabu_list) > tabu_size:
            # Remove oldest
            tabu_list.pop()
            
        iteration += 1
        swaps_since_improvement += 1
        
        # Perturbation to avoid local minima
        if swaps_since_improvement > perturbation_interval:
            # Flip a random block
            start = np.random.randint(0, n_points - block_flip_size)
            for k in range(start, min(start + block_flip_size, n_points)):
                if h[k] == 1:
                    h[k] = 0
                    # Flip to 0
                else:
                    h[k] = 1
            g_s = compute_g_s(h)
            current_max = np.max(g_s)
            swaps_since_improvement = 0
        
        # Early exit for very small margin
        if best_max < 0.379:
            break
    
    return best_h, best_max

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    n_points = 120  # Start with coarse grid
    best_h = None
    best_c5 = float('inf')
    
    # Generate initial binary sequence (Sturmian pattern as before)
    h0 = generate_sturmian_binary(n_points)
    h0 = h0.astype(float)
    
    # Step 1: Local swap search on coarse resolution
    if (time.time() - start_time) < budget_s * 0.5:
        # Leave room for higher resolution
        search_time = budget_s * 0.4
        
        # Run local search on current resolution
        h1, c5_1 = local_swap_search(h0, n_points, search_time)
        
        # Keep best found
        if c5_1 < best_c5:
            best_h = h1
            best_c5 = c5_1
    
    # Step 2: Upscale and re-search
    # Based on time remaining
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 5 and best_h is not None:
        n_points_up = 240  # Double resolution
        dx_up = 2.0 / n_points_up
        
        # Oversample coarse solution
        h2 = np.zeros(n_points_up)
        for i in range(n_points):
            h2[2*i] = h1[i]
            h2[2*i + 1] = h1[i]
        
        # Allow clearer range for alignment test
        h2 = h2.astype(float)
        
        # Ensure balance
        current_sum = np.sum(h2) * dx_up
        if abs(current_sum - 1.0) > 1e-6:
            # Normalize to exact sum
            h2 = h2 * (1.0 / current_sum)
        
        # Local search on higher resolution
        h3, c5_2 = local_swap_search(h2, n_points_up, remaining_time * 0.7)
        
        # Keep best
        if c5_2 < best_c5:
            best_h = h3
            best_c5 = c5_2
    
    # Post-processing: Finish with continuous refinement if needed
    if (time.time() - start_time) < 0.5 * budget_s:
        # Project onto feasible region and do rapid SGD
        # Normalize to keep sum = 1
        dx = 2.0 / len(best_h)
        current_sum = np.sum(best_h) * dx
        
        # Back to exact constraint accumulation
        best_h = best_h * (1.0 / current_sum)
        
        # Quick continuous gradient step, no gradient now, just rounding a bit
        pass  # Currently no functional gradient added
    
    # Evaluate final result
    final_c5 = evaluate_erdos_solution(best_h)
    
    # Ensure exact number of points
    while len(best_h) > 200:
        best_h = best_h[::2]
    
    return (best_h.tolist(), final_c5, len(best_h))
