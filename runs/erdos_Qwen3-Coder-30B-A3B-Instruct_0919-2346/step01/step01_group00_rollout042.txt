def c5_of(h):
    """Compute C5 = max_k integral h(x)(1-h(x+k)) dx."""
    dx = 2.0 / len(h)
    # Compute correlation: h * (1 - h) with full mode
    c = np.correlate(h, 1 - h, mode="full")
    # Multiply by dx to get continuous integral
    c *= dx
    # Return maximum value among all shifts
    return np.max(c)

def project_feasible(h):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = n/2"""
    h = np.clip(h, 0, 1)
    # Renormalize
    dx = 2.0 / len(h)
    scale = (len(h) / 2) / np.sum(h)
    h *= scale
    return h

def propose_move(h, max_step):
    """Generate a new feasible h by adjusting two entries."""
    h = h.copy()
    N = len(h)
    i, j = random.sample(range(N), 2)
    
    # Determine feasible interval for delta
    delta_low = max(-h[i], h[j] - 1.0)
    delta_high = min(1.0 - h[i], h[j])
    
    if delta_low >= delta_high:
        return h  # No valid move for this pair
    
    # Sample delta within allowed range
    step = min(max_step, delta_high - delta_low)
    delta = random.uniform(-step, step)
    delta = max(delta_low, min(delta, delta_high))
    
    h[i] += delta
    h[j] -= delta
    
    return project_feasible(h)

def stochastic_local_search(initial_h, time_budget):
    """Stochastic local search with random walks."""
    start = time.time()
    cur = initial_h.copy()
    best = initial_h.copy()
    c_cur = c5_of(cur)
    c_best = c_cur
    
    step = 0.1
    
    while time.time() - start < time_budget:
        h_new = propose_move(cur, step)
        c_new = c5_of(h_new)
        
        if c_new < c_cur:
            cur = h_new
            c_cur = c_new
            if c_new < c_best:
                best = h_new
                c_best = c_new
        elif random.random() < 0.05:
            cur = h_new
            c_cur = c_new
    
    return best, c_best

        new_population = []

            # Tournament

            for i in idxs[1:]:

                    best_idx = i
