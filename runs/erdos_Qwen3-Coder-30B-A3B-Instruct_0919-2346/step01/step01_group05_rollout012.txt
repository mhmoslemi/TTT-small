import numpy as np
import time

def upsample_initial(h0, target_n):
    """Upsample a given initial vector to target_n points."""
    n0 = len(h0)
    x0 = np.linspace(0, 2, n0)
    x_new = np.linspace(0, 2, target_n)
    h_new = np.interp(x_new, x0, h0)
    # Clip to [0,1]
    h_new = np.clip(h_new, 0, 1)
    # Project onto simplex
    return project_to_simplex(h_new)

def project_to_simplex(v):
    """Project vector v onto the simplex {x | sum x = n/2, x in [0,1]^n}."""
    n = len(v)
    target_sum = n / 2.
    # Clip v to [0,1]
    v = np.clip(v, 0, 1)
    # Compute projection using Lagrange multipliers
    v_sorted = np.sort(v)[::-1]
    tmp_sum = 0.0
    for i in range(n):
        tmp_sum += v_sorted[i]
        if tmp_sum > target_sum - (i+1) * 1.0:
            # We found the cutoff index
            break
    else:
        # All elements must be < 0.5 and sum ≤ n/2
        # Set lambda such that sum of clip(v[i] - lambda, 0, 1) == target_sum
        lambda_ = (np.sum(v_sorted) - target_sum) / (n - i - 1)
        if lambda_ < 0:
            lambda_ = 0.0  # fallback case
        return np.clip(v - lambda_, 0, 1)
    # General case
    lambda_ = (tmp_sum - target_sum) / (i + 1)
    return np.clip(v - lambda_, 0, 1)

def get_shifted_grad(h, s):
    """Compute the gradient of f_s(h) = sum h_i (1 - h_{i+s})."""
    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        # i+s valid for i in [0, n-s)
        for i in range(n - s):
            g[i] += (1 - h[i + s])
        # i in [0, n-s) and i-s valid for i in [s, n) is handled above
    else:
        # s < 0
        # i valid for i in [0, n+s), and i+s valid for i in [0, n)
        for i in range(-s, n):
            g[i] += 1 - h[i + s]
    return g

def softmax_weights(fs, gamma):
    """Return soft-max weights based on function values fs at temperature gamma."""
    es = np.exp(gamma * fs)
    return es / np.sum(es)

def compute_correlations(h):
    """Return all correlation terms."""
    n = len(h)
    dx = 2.0 / n
    # Use np.correlate to get all overlaps with 1-h for shifts
    # np.correlate(h, 1-h, mode='full') gives correlation for all shifts
    corr = np.correlate(h, 1 - h, mode='full')
    # Correlation corresponds to sum h_i (1 - h_{i+k}) = f_k
    # For flat array corresponding to lags from -(n-1) to (n-1)
    # We compute all shifts
    overlaps = corr * dx
    return overlaps

def gradient_descent_step(h, gamma, alpha, dx):
    """Perform one projected gradient descent step."""
    n = len(h)
    fs = compute_correlations(h)[n-1:2*n-1]  # Only the correlation terms
    ws = softmax_weights(fs, gamma)
    grad = np.zeros(n)
    for s, w in enumerate(ws):
        shifted_s = s - (n - 1)
        g = get_shifted_grad(h, shifted_s)
        grad += w * dx * g
    v = h - alpha * grad
    h_new = project_to_simplex(v)
    return h_new
