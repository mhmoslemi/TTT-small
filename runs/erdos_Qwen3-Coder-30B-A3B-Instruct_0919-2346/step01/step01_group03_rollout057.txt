import numpy as np
import time
from scipy.optimize import differential_evolution, minimize

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral via full convolution of h with (1-h)
    overlaps = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum of the full correlation vector
    # scaled by dx
    c5 = np.max(overlaps) * dx
    return c5

def project_to_constraint(h, target_sum=1.0):
    """Project h to satisfy the constraint sum(h) * dx = target_sum"""
    dx = 2.0 / len(h)
    actual_sum = np.sum(h) * dx
    if abs(actual_sum - target_sum) < 1e-10:
        return h
    return h * target_sum / actual_sum

def propose_move(h, max_step=0.05):
    """Propose a feasible move by adjusting two entries."""
    h = h.copy()
    N = len(h)
    # Try to find a valid pair of indices
    for _ in range(100):  # avoid infinite loop
        i, j = np.random.randint(0, N, 2)
        if i == j:
            continue
        # Determine feasible δ range
        delta_low = max(-h[i], h[j] - 1)
        delta_high = min(1 - h[i], h[j])
        if delta_low >= delta_high:
            continue
        # Sample a move with limited step size
        step = min(max_step, delta_high - delta_low)
        delta = np.random.uniform(
            max(-step, delta_low),
            min(step, delta_high)
        )
        # Apply move
        h[i] += delta
        h[j] -= delta
        return h
    # If no valid move found, return copy
    return h

def optimize_with_de(initial_h, maxiter=1000, seed=42):
    """Optimize using differential evolution"""
    np.random.seed(seed)
    bounds = [(0, 1) for _ in range(len(initial_h))]
    
    def objective(h):
        h = project_to_constraint(h)
        return c5_of(h)
    
    result = differential_evolution(
        objective,
        bounds,
        maxiter=maxiter,
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.7,
        seed=seed,
        disp=False
    )
    
    return result.x

    # Optimize using DE to find good starting point
    h_de = optimize_with_de(h_current, maxiter=200, seed=seed)
    h_de = project_to_constraint(h_de)
    c5_current = c5_of(h_de)
    
    # Local search using simulated annealing with better cooling
    best_h = h_de
    best_c5 = c5_current
    
    # Very aggressive simulated annealing
    max_time = budget_s * 0.8  # Keep some time for final polishing
    T_start = 0.05
    T_end = 1e-7
    step_start = 0.1
    
    T = T_start

    while time.time() - start < max_time:

        tfrac = elapsed / max_time
        T = T_start * (T_end / T_start) ** tfrac
        step = step_start * (1 - tfrac)**2
        
        h_new = propose_move(best_h, step)
