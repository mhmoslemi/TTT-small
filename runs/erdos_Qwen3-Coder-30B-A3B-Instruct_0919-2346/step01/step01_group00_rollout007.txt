def compute_C_vector(h):
    """Compute vector of overlaps C_τ(h) = 1 - dx * S_τ(h) for all shifts."""
    dx = 2.0 / len(h)
    # Compute full autocorrelation using FFT
    h_pad = np.pad(h, (0, len(h)-1), mode='constant')
    S = np.convolve(h, h, mode='full')[:len(h)+len(h)-1]
    # Convert to overlaps
    c = 1.0 - dx * S
    return c

def compute_softmax_weights(c, alpha):
    """Compute softmax weights for smooth max: w_τ = exp(alpha * c_τ) / sum(exp(alpha * c_τ))"""
    c_max = np.max(c)
    exp_c = np.exp(alpha * (c - c_max))
    w = exp_c / np.sum(exp_c)
    return w

def compute_gradient(h, w):
    """Compute gradient of smooth objective: g_i = -dx * sum(w_τ * (h_{i+τ} + h_{i-τ}))"""
    dx = 2.0 / len(h)
    w_rev = w[::-1]  # Reverse weights
    # Use convolution to compute the sums efficiently
    grad = -(dx * (np.convolve(w, h, mode='same') + np.convolve(w_rev, h, mode='same')))
    return grad

def project_onto_feasible(h, target_sum):
    """Project onto box [0,1] and exact sum target_sum using water-filling."""
    def sum_at_lambda(lam):
        # Compute sum_clip = sum_i clip(h_i - lam, 0, 1)
        x = np.clip(h - lam, 0, 1)
        return np.sum(x) - target_sum

    # Find root of sum_at_lambda using bisection
    low, high = np.min(h) - 1.0, np.max(h) + 1.0
    eps = 1e-12

    while high - low > eps:
        mid = (low + high) / 2
        if sum_at_lambda(mid) > 0:
            low = mid
        else:
            high = mid

    lam_opt = (low + high) / 2
    h_proj = np.clip(h - lam_opt, 0, 1)
    # Re-normalize to preserve exact sum
    h_proj *= target_sum / np.sum(h_proj)
    return h_proj
