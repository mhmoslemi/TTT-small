import numpy as np
import time
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import cdist

def c5_of(h):
    """Compute the maximum overlap integral for the given step function h."""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    c = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(c) * dx
    return c5

def evaluate_solution(h):
    """Evaluate a solution using correlation-based computation."""
    return c5_of(h)

def propose_moves(h, num_moves=5):
    """Propose multiple moves to explore the neighborhood."""
    N = len(h)
    new_h_list = []
    for _ in range(num_moves):
        i, j = np.random.choice(N, size=2, replace=False)
        # Compute feasible step range
        delta_low = max(-h[i], h[j] - 1)
        delta_high = min(1 - h[i], h[j])
        if delta_low >= delta_high:
            continue
        step_size = np.random.uniform(delta_low, delta_high)
        h_new = h.copy()
        h_new[i] += step_size
        h_new[j] -= step_size
        new_h_list.append(h_new)
    return new_h_list

def gradient_descent_step(h, learning_rate=0.01):
    """Perform a simple gradient descent step by adjusting values."""
    N = len(h)
    h_new = h.copy()
    # Compute the gradient approximation using finite differences
    eps = 1e-6
    base_c5 = c5_of(h)
    for i in range(N):
        h_plus = h.copy()
        h_plus[i] += eps
        c_plus = c5_of(h_plus)
        grad = (c_plus - base_c5) / eps
        h_new[i] -= learning_rate * grad
        
    # Project onto feasible region
    h_new = np.clip(h_new, 0, 1)
    # Renormalize
    dx = 2.0 / N
    h_new = h_new * (N / 2) / np.sum(h_new)
    return h_new

def local_search(h, max_iter=50):
    """Apply a local search procedure to improve the solution."""
    current_h = h.copy()
    current_c5 = c5_of(current_h)
    
    for _ in range(max_iter):
        improved = False
        moves = propose_moves(current_h, 10)
        for h_new in moves:
            c_new = c5_of(h_new)
            if c_new < current_c5:
                current_h = h_new
                current_c5 = c_new
                improved = True
                break
        if not improved:
            break
                
    return current_h, current_c5
