import numpy as np
from scipy.optimize import minimize
from math import pi
import time

def build_h_from_coeffs(a_coeffs, n):
    """Build h vector from Fourier coefficients (cosine only)"""
    h = 0.5 * np.ones(n)
    for m, a in enumerate(a_coeffs, 1):
        h += a * np.cos(2 * pi * m * np.arange(n) / n)
    return h

def compute_overlap_and_gradient(a_coeffs, n, dx, gamma=1.0):
    """Compute overlap value and gradient for current a_coeffs"""
    h = build_h_from_coeffs(a_coeffs, n)
    # Soft clip to [0,1]
    h_clipped = np.clip(h, 0, 1)
    # Compute overlaps manually for full range of shifts s
    overlaps = []
    for s in range(-n+1, n):
        if s < 0:
            # h[0:-s] * (1 - h[-s:])
            overlap = dx * np.sum(h_clipped[:n+s] * (1 - h_clipped[-s:]))
        else:
            # h[s:] * (1 - h[:n-s])
            overlap = dx * np.sum(h_clipped[s:] * (1 - h_clipped[:n-s]))
        overlaps.append(overlap)
    # Softmax over overlaps
    max_o = max(overlaps)
    exps = np.exp(gamma * np.array(overlaps) - gamma * max_o)
    weights = exps / np.sum(exps)
    # Objective value
    obj_val = (1/gamma) * np.log(np.sum(exps)) + 1e4 * np.sum(np.maximum(0, h - 1)**2) + 1e4 * np.sum(np.maximum(0, -h)**2)
    # Compute gradient
    grad = np.zeros_like(a_coeffs)
    # For each coefficient a_m
    for m in range(1, len(a_coeffs) + 1):
        cos_m_i = np.cos(2 * pi * m * np.arange(n) / n)
        for s, w in enumerate(weights):
            if s < n-1:
                shift_idx = s - (n-1)
                if shift_idx < 0:
                    # Compute gradient of f_s with respect to a_m
                    # ∂/∂a_m of ∑ h_i*(1-h_{i+s})
                    # = ∑ cos_m_i*(1-h_{i+s}) - ∑ h_i*cos_m_{i+s}
                    # Note: s is the shift as offset
                    exp_i = slice(0, n+shift_idx)
                    sh_i = slice(-shift_idx, n)
                else:
                    # Compute gradient of f_s with respect to a_m
                    exp_i = slice(shift_idx, n)
                    sh_i = slice(0, n-shift_idx)
                g = np.sum(cos_m_i[exp_i] * (1 - h_clipped[sh_i])) - np.sum(h_clipped[exp_i] * np.cos(2*pi*m*(np.arange(shift_idx, n) + shift_idx)/n))
                grad[m-1] += w * g * dx
            else:
                # out of bounds, skip
                pass
    # Add penalty gradient
    for i in range(len(a_coeffs)):
        m = i + 1
        cos_m_i = np.cos(2 * pi * m * np.arange(n) / n)
        penalty_grad = 2e4 * np.sum((h - np.clip(h, 0, 1)) * cos_m_i)
        grad[i] += penalty_grad
    return obj_val, grad

def project_to_simplex_box(h, n):
    """Project h onto the intersection of [0,1]^n and {sum h = n/2}"""
    # Water-filling approach
    h = np.clip(h, 0, 1)
    total = np.sum(h)
    # Initial guess for the threshold tau
    if total < n/2:
        # Need to increase; this is not typical with penalized optimization, but just in case
        h = np.clip(h + (n/2 - total) / n, 0, 1)
    else:
        # Need to decrease; find tau
        sort_indices = np.argsort(h)[::-1]
        h_sorted = h[sort_indices]
        cumsum = np.cumsum(h_sorted)
        M = len(h_sorted)
        # Find point m where sum > n/2
        for m in range(M):
            if cumsum[m] * (M - m) >= n/2:
                # Threshold = (n/2 - (cumsum[m] - h_sorted[m])) / (M - m)
                # But we need the cumulative value to be decreased to match the average
                break
        else:
            m = M  # No adjustment needed
        if m > 0:
            # The sum of the first m elements is too large
            # The average of the top m values should be (n/2 - cumsum[m-1]) / (M - m+1)
            # but we now allow tau to be adjusted by noting that elements lower than or equal to tau will be 1, those above are adjusted to a lower value
            # This is more tricky: let's use one pass with water-filling
            # Re-sort indices in decreasing order and adjust
            target_sum = n/2
            remaining_sum = target_sum
            for i in range(M):
                val = h_sorted[i]
                # How much we can set this value to be, without violating our assignment?
                # If all values after this we must set to 0 or 1, it becomes a computation of how much we are allowed to reduce the total
                # The total number of remaining values = M - i
                # If we set val' to be the current value, then new need sum of previous = target_sum - val'
                # In the best case, after this value, all remaining are set to 0 or 1
                # Max sum occurs if all = 1 (i.e. M-i all set to 1) but we want remaining_sum - val' = ...
                # How slowly to adjust?
                # Let's try a simplified approach:
                # The idea is to sort values and adjust from the top down
                # But it's complex to deal with two constraints
                # So let's just do simple water filling for ensuring the sum is exactly 1/2 * n

                # Use simple approach:
                # All values that are less than target_sum/(M - i) can be adjusted up to that if needed
                # But let's skip for now and just clip and rescale
                h_sorted[i] = min(val, 1)
                remaining_sum -= h_sorted[i]
                if remaining_sum <= 0:
                    break
            # If remaining_sum <= 0, rest should be set to 1 or 0
            for j in range(i+1, M):
                h_sorted[j] = min(1, max(0, h_sorted[j]))
            # Rescale if needed, but more efficient to just reset
            # Ensure sum = 1/2 * n
            total_now = np.sum(h_sorted)
            if abs(total_now - n/2) > 1e-10:
                # Simple solution: rescale
                scale = (n/2) / total_now
                h_sorted *= scale
            # Reorder back to original indices
            h[sort_indices] = h_sorted
    return h

def spectral_optimize(n_points, max_iter=2500, gamma_increase_freq=500, gamma_start=0.5, gamma_final=40.0, m_steps=[8, 15, 24]):
    """Run multi-stage spectral optimization with smooth surrogate"""
    dx = 2.0 / n_points
    # Perform random restarts
    best_c5 = float('inf')
    best_h = None
    for restart in range(8):
        np.random.seed(42 + restart)
        a_init = np.random.normal(0, 0.05, m_steps[0])
        a_coeffs = np.zeros(max(m_steps))
        a_coeffs[:len(a_init)] = a_init
        # Stage 1
        for stage in range(len(m_steps)):
            M = m_steps[stage]
            # Set coefficients to zero beyond M if needed
            if len(a_coeffs) > M:
                a_coeffs[M:] = 0
            # Current gamma
            gamma = gamma_start
            iter_count = 0
            while iter_count < (max_iter // 3 if stage < len(m_steps)-1 else max_iter):
                if iter_count > 0 and iter_count % gamma_increase_freq == 0:
                    gamma = min(gamma * 1.5, gamma_final)
                def func_grad(a_vec):
                    return compute_overlap_and_gradient(a_vec[:M], n_points, dx, gamma)
                try:
                    # L-BFGS-B is suitable for bounded convex optimization
                    res = minimize(func_grad, a_coeffs[:M], method='L-BFGS-B', jac=True, bounds=[(-0.6, 0.6)] * M, options={'maxiter': 1000})
                    if res.success or iter_count > 0:
                        a_coeffs[:M] = res.x
                        iter_count += 1
                    else:
                        break
                except:
                    break
            # Stop early if time is about to run out
            if iter_count >= max_iter:
                break
        # Final construction
        h = build_h_from_coeffs(a_coeffs, n_points)
        h_feas = project_to_simplex_box(h, n_points)
        c5_eval = evaluate_erdos_solution(h_feas)
        if c5_eval < best_c5:
            best_c5 = c5_eval
            best_h = h_feas.copy()
    return best_h, best_c5

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value scaled by dx
    max_overlap = np.max(correlation) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 240
    h_best, best_c5 = spectral_optimize(n_points, max_iter=10000)
    return (h_best.tolist(), best_c5, len(h_best))
