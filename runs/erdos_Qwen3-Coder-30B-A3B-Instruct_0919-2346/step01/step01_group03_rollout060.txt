import numpy as np
import time

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral via full convolution of h with (1-h)
    overlaps = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum of the full correlation vector
    # scaled by dx
    c5 = np.max(overlaps) * dx
    return c5

def compute_c5_and_worst_info(h):
    """Compute C5 and worst shift info"""
    N = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    max_idx = np.argmax(corr)
    max_val = corr[max_idx]
    offset = max_idx - (N - 1)
    # contributions vector:
    if offset >= 0:
        hi = h[:N - offset]
        hj = h[offset:]
        contrib = hi * (1 - hj)
        i_local = np.argmax(contrib)
        i_index = i_local
        j_index = i_local + offset
    else:
        off = -offset
        hi = h[off:]
        hj = h[:N - off]
        contrib = hi * (1 - hj)
        i_local = np.argmax(contrib)
        i_index = i_local + off
        j_index = i_local
    # return values
    return max_val * (2.0/N), offset, i_index, j_index, hi[i_local], hj[i_local], contrib[i_local]

def propose_direct_move(h, i, j, step):
    delta = min(step, h[i], 1 - h[j])
    if delta <= 0:
        return None
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new

def targeted_descent(h, time_budget, start_time):
    """Run targeted descent on h"""
    N = len(h)
    best_h = h.copy()
    best_c5 = c5_of(h)
    step = 0.5
    fails = 0
    while time.time() - start_time < time_budget:
        # Compute worst info
        _, offset, i, j, hi_val, hj_val, _ = compute_c5_and_worst_info(h)
        # compute delta
        delta = min(step, h[i], 1 - h[j])
        if delta > 0:
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            c5_new = c5_of(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
                step = min(step * 1.1, 0.5)
                fails = 0
                continue
        # Try moving in other direction (j->i) if beneficial
        delta2 = min(step, h[j], 1 - h[i])
        if delta2 > 0:
            h_new2 = h.copy()
            h_new2[j] -= delta2
            h_new2[i] += delta2
            c5_new2 = c5_of(h_new2)
            if c5_new2 < best_c5:
                best_c5 = c5_new2
                best_h = h_new2.copy()
                h = h_new2
                step = min(step * 1.1, 0.5)
                fails = 0
                continue
        # if neither improves, reduce step
        step *= 0.9
        fails += 1
        if fails > 2000:
            idxs = np.random.choice(N, size=2, replace=False)
            delta_r = step * min(h[idxs[0]], 1 - h[idxs[1]])
            if delta_r > 0:
                h = propose_direct_move(h, idxs[0], idxs[1], step)
                fails = 0
        # occasionally also try random small move to escape local minima
        if np.random.rand() < 0.01:
            i_rand, j_rand = np.random.choice(N, size=2, replace=False)
            delta_r = step * min(h[i_rand], 1 - h[j_rand])
            if delta_r > 0:
                h_new = h.copy()
                h_new[i_rand] -= delta_r
                h_new[j_rand] += delta_r
                c5_new = c5_of(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new
    return best_h, best_c5

def project_onto_simplex_box(v, total_sum):
    """Project v onto the set {x | 0 <= x_i <= 1, sum x_i = total_sum}."""
    # Basic bisection.
    lo = v.min() - 1
    hi = v.max() + 1
    for _ in range(60):
        lam = (lo + hi) / 2
        x = np.clip(v - lam, 0, 1)
        s = x.sum()
        if s > total_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo+hi)/2, 0, 1)

def compute_worst_offset(h):
    """Return the offset that corresponds to the maximum overlap."""
    N = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    max_idx = np.argmax(corr)
    offset = max_idx - (N - 1)
    return offset, corr[max_idx]

def subgradient_descent(h, iters, start_step):
    """Run subgradient descent on h"""
    N = len(h)
    total = N/2
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = c5_of(h_cur)
    for it in range(iters):
        # compute worst offset
        offset, _ = compute_worst_offset(h_cur)
        # compute gradient for worst offset only
        g = np.ones(N)
        d = offset
        if d >= 0:
            g[:N-d] -= h_cur[d:]
            g[d:]   -= h_cur[:N-d]
        else:
            off = -d
            g[off:] -= h_cur[:N-off]
            g[:N-off] -= h_cur[off:]
        
        # step size schedule (linearly decreasing)
        step = start_step * (1 - it / iters) + 0.01
        # perform gradient step
        v = h_cur - step * g
        # project onto feasible region
        h_cur = project_onto_simplex_box(v, total)
        c5 = c5_of(h_cur)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_cur.copy()
    return best_h, best_c5

            final_seed = best_h.copy()
            h_final, c5_final = targeted_descent(final_seed, final_budget, start)
