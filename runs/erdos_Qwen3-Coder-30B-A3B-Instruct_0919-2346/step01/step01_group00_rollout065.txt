import numpy as np
import time
import random

def compute_C_and_grad(h, alpha):
    """Compute C vector and gradient w.r.t. h for given alpha."""
    N = len(h)
    dx = 2.0 / N
    
    # Compute correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    C = corr * dx
    
    # Smooth the max via log-sum-exp
    w = np.exp(alpha * C)
    w = w / np.sum(w)
    
    # Compute gradient using convolutions
    # Since we're differentiating C_k = 1 - dx * sum h_i * h_{i+k}
    # dC_k / dh_i = -dx * (h_{i+k} + h_{i-k})
    # Gradient of smooth max = sum w_k * dC_k / dh_i
    
    # Compute h_{i+k} + h_{i-k} for all k for each i
    # Use convolution of h with w_rev and w to compute fast sum
    w_rev = w[::-1]
    
    # Compute np.convolve(h, w_rev, mode='same') and np.convolve(h, w, mode='same')
    # These represent sum over k of w_k * h_{i+k} and sum over k of w_k * h_{i-k}
    # Since we apply np.convolve, we get a result of length N (same as h)
    v_plus = np.convolve(h, w_rev, mode='same')
    v_minus = np.convolve(h, w, mode='same')
    
    # Compute gradient
    grad = dx * (1 - (v_plus + v_minus))
    
    # Compute max C without smoothing for reporting
    C_max = np.max(C)
    
    return C_max, grad

def project_h(h, target_sum):
    """Project h onto box constraints [0,1] and sum constraint."""
    # Start with clipped h
    h_proj = np.clip(h, 0, 1)
    
    # Adjust to target sum using water-filling
    current_sum = np.sum(h_proj)
    if abs(current_sum - target_sum) < 1e-10:
        return h_proj
    
    # Water-filling approach
    h_sorted_idx = np.argsort(h_proj)[::-1]
    h_sorted = h_proj[h_sorted_idx]
    
    # Determine threshold lambda
    sum_lo = 0.0
    sum_hi = 1.0
    threshold = 0.0
    
    while abs(sum_hi - sum_lo) > 1e-12:
        threshold = (sum_lo + sum_hi) / 2
        adjusted = np.clip(h_sorted - threshold, 0, 1)
        sum_adjusted = np.sum(adjusted)
        if sum_adjusted > target_sum:
            sum_lo = threshold
        else:
            sum_hi = threshold
    
    # Recompute final solution
    h_final = np.clip(h_sorted - threshold, 0, 1)
    
    # Map back to original indices
    h_final_full = h_proj.copy()
    h_final_full[h_sorted_idx] = h_final
    
    return h_final_full
