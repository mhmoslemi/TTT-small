def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral via full convolution of h with (1-h)
    overlaps = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum of the full correlation vector
    # scaled by dx
    c5 = np.max(overlaps) * dx
    return c5

def propose_move(h, max_step):
    """Propose a feasible move by adjusting two entries."""
    h = h.copy()
    N = len(h)
    # Try to find a valid pair of indices
    for _ in range(100):  # avoid infinite loop
        i, j = np.random.randint(0, N, 2)
        if i == j:
            continue
        # Determine feasible δ range
        delta_low = max(-h[i], h[j] - 1)
        delta_high = min(1 - h[i], h[j])
        if delta_low >= delta_high:
            continue
        # Sample a move with limited step size
        step = min(max_step, delta_high - delta_low)
        delta = np.random.uniform(
            max(-step, delta_low),
            min(step, delta_high)
        )
        # Apply move
        h[i] += delta
        h[j] -= delta
        return h
    # If no valid move found, return copy
    return h

def evolve_population(pop, fitness, mutation_rate=0.1, elite_frac=0.2):
    """Evolve a population based on fitness"""
    N = len(pop)
    elite_size = int(elite_frac * N)
    
    # Sort by fitness (lower is better)
    sort_idx = np.argsort(fitness)
    elite = [pop[i] for i in sort_idx[:elite_size]]
    
    # Select parents based on fitness (lower fitness better)
    # Use tournament selection
    new_pop = elite.copy()
    while len(new_pop) < N:
        # Tournament selection
        candidates = np.random.choice(N, 4, replace=False)
        winner = candidates[np.argmin([fitness[i] for i in candidates])]
        parent = pop[winner]
        
        # Apply mutation
        if np.random.rand() < mutation_rate:
            # Create mutated version
            h_mut = parent.copy()
            # Pick some positions to change
            mut_positions = np.random.choice(len(parent), size=max(1, len(parent) // 20), replace=False)
            for i in mut_positions:
                # Perturb the value
                pert = np.random.normal(0, 0.05)
                h_mut[i] = np.clip(h_mut[i] + pert, 0, 1)
            new_pop.append(h_mut)
        else:
            new_pop.append(parent.copy())
    
    return new_pop
