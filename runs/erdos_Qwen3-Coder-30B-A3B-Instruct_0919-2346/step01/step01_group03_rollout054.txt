import numpy as np
import time

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral via full convolution of h with (1-h)
    overlaps = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum of the full correlation vector
    # scaled by dx
    c5 = np.max(overlaps) * dx
    return c5

def compute_convolution(h):
    """Compute the full convolution of h with (1-h)"""
    return np.correlate(h, 1 - h, mode="full")

def gradient_descent_step(h, step_size=0.01):
    """Perform a gradient descent step to minimize C5"""
    h = h.copy()
    n = len(h)
    dx = 2.0 / n
    
    # Compute gradient using finite differences
    conv = compute_convolution(h)
    max_idx = np.argmax(conv)
    
    # Compute gradient of C5 with respect to h[i] 
    # by computing how each h[i] affects the maximum overlap
    grad = np.zeros(n)
    
    # For a small perturbation at index i, compute how it changes the overlap
    # We compute the effect of small changes at each position
    for i in range(n):
        # Small perturbation in h[i]
        eps = 1e-6
        h_eps = h.copy()
        h_eps[i] += eps
        conv_eps = compute_convolution(h_eps)
        grad[i] = (conv_eps[max_idx] - conv[max_idx]) / eps
    
    # Gradient ascent for minimization (negative gradient)
    h_new = h - step_size * grad
    
    # Project back to feasible region [0,1]
    h_new = np.clip(h_new, 0, 1)
    
    # Renormalize to maintain integral = 1
    h_new *= (n / 2) / np.sum(h_new)
    
    return h_new

def hill_climb(initial_h, time_budget, seed):
    """Run hill climbing optimizer with gradient-based steps"""
    np.random.seed(seed)
    start = time.time()
    
    cur = initial_h.copy()
    c_cur = c5_of(cur)
    
    best = cur.copy()
    c_best = c_cur
    
    step_size = 0.01
    max_iter = 5000
    iter_count = 0
    
    while time.time() - start < time_budget and iter_count < max_iter:
        iter_count += 1
        h_new = gradient_descent_step(cur, step_size)
        c_new = c5_of(h_new)
        
        if c_new < c_cur:
            cur = h_new
            c_cur = c_new
            
            if c_new < c_best:
                best = h_new.copy()
                c_best = c_new
        
    return best, c_best
