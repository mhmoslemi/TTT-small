import numpy as np
import time

def compute_c5_and_worst_shift(h):
    """Compute C5 and return the worst shift index."""
    dx = 2.0 / len(h)
    # Compute correlation using np.correlate
    c = np.correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(c)
    s = max_idx - (len(h) - 1)
    c5 = c[max_idx] * dx
    return c5, s

def compute_gradient(h, shift):
    """Compute gradient of C5 with respect to h for a given shift."""
    N = len(h)
    dx = 2.0 / N
    g = np.full(N, 1.0) * dx
    if shift >= 0:
        if shift < N:
            g[:N-shift] -= h[shift:] * dx
            g[shift:] -= h[:N-shift] * dx
    else:
        shift = -shift
        if shift < N:
            g[shift:] -= h[:N-shift] * dx
            g[:N-shift] -= h[shift:] * dx
    return g

def project_onto_bounds_and_sum(v, target):
    """Project vector v onto box [0,1] and sum = target."""
    # Binary search for lambda
    lower = -np.max(v)
    upper = 1 - np.min(v)
    
    for _ in range(60):
        mid = (lower + upper) / 2.0
        w = np.clip(v + mid, 0, 1)
        s = np.sum(w)
        if s < target:
            lower = mid
        else:
            upper = mid
    
    w = np.clip(v + (lower + upper) / 2.0, 0, 1)
    # Adjust sum to be exact
    diff = target - np.sum(w)
    if abs(diff) > 1e-12:
        free_idx = (w > 0) & (w < 1)
        if np.any(free_idx):
            w[free_idx] += diff / np.sum(free_idx)
            w = np.clip(w, 0, 1)
        else:
            w += diff / len(w)
            w = np.clip(w, 0, 1)
    return w

    x_vals = np.linspace(0, 2, N)
    mask = (x_vals < a) | (x_vals >= 2 - a)
    h1[mask] = u
    h1[~mask] = v
    h1 = h1 * (N/2) / np.sum(h1)
    candidates.append(h1)
    
    # Random uniform
    h2 = np.ones(N) * 0.5
    candidates.append(h2)
    
    # Binary seed
    for _ in range(3):
        h_bin = np.zeros(N)
        idx = np.random.choice(N, N//2, replace=False)
        h_bin[idx] = 1.0
        candidates.append(h_bin)
    
    # Evaluate initial candidates
    best_h = None
    best_c5 = float('inf')
    for h in candidates:
        c5 = compute_c5_and_worst_shift(h)[0]
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
    
    # Gradient descent with time management
    start = time.time()
    while time.time() - start < budget_s * 0.85:
        # Copy current best for gradient descent
        cur_h = best_h.copy()
        cur_c5 = best_c5
        
        # Perform multiple iterations
        inner_start = time.time()
        iter_count = 0
        max_iters = 2000
        
        while time.time() - inner_start < 1.0 and iter_count < max_iters:
            # Compute worst shift
            c5, s = compute_c5_and_worst_shift(cur_h)
            
            # Compute gradient
            grad = compute_gradient(cur_h, s)
            
            # Adaptive step size
            elapsed = time.time() - start

            step = 20 * (1 - t_frac) + 0.1
            
            # Update
            v = cur_h - step * grad
            cur_h = project_onto_bounds_and_sum(v, N/2)

        cur_h[j] -= delta

            if cur_c5 < best_c5:
                best_h = cur_h.copy()
                best_c5 = cur_c5
                if best_c5 < 0.379:  # Early exit
                    return (best_h, best_c5, N)
        else:
            cur_h[i] = h_save  # Revert
    return (best_h, best_c5, N)
