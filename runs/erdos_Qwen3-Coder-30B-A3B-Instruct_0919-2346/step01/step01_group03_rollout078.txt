import numpy as np
import time

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    overlaps = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(overlaps) * dx
    return c5

def full_overlap(h):
    """Return the raw overlap correlation vector"""
    return np.correlate(h, 1 - h, mode="full")

def project_onto_P(v, total):
    """
    Project vector v onto the set {x : 0 <= x_i <= 1, sum(x) = total}
    Uses bisection to find the correct lambda
    """
    v = np.array(v)
    lower, upper = 0.0, 1.0
    total = float(total)
    while True:
        lambda_ = (lower + upper) / 2
        x = np.clip(v - lambda_, 0, 1)
        sum_x = np.sum(x)
        if abs(sum_x - total) < 1e-10:
            return x
        elif sum_x > total:
            lower = lambda_
        else:
            upper = lambda_

def smooth_gradient(h, beta):
    """Compute gradient of the smooth surrogate Φ_β(h)"""
    N = len(h)
    F = full_overlap(h)
    # subtract max for numerical stability
    F_shifted = F - np.max(F)
    exp_F = np.exp(beta * F_shifted)
    w = exp_F / np.sum(exp_F)
    
    g = np.ones(N)
    for k in range(-N+1, N):
        wk = w[k + N - 1]
        if k >= 0:
            g[:N-k] -= wk * h[k:]
            g[k:] -= wk * h[:N-k]
        else:
            d = -k
            g[d:] -= wk * h[:N-d]
            g[:N-d] -= wk * h[d:]
    return g

def projected_gradient_descent(h_start, time_budget, seed=42):
    """Run projected gradient descent with β-schedule"""
    np.random.seed(seed)
    start = time.time()
    
    N = len(h_start)
    total = N / 2
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = c5_of(h)
    
    beta = 2.0
    beta_max = 30.0
    beta_update_interval = 50
    gamma_beta = 1.5
    step = 0.5
    gamma_step = 0.9995
    
    while time.time() - start < time_budget:
        F = full_overlap(h)
        F_shifted = F - np.max(F)
        exp_F = np.exp(beta * F_shifted)
        w = exp_F / np.sum(exp_F)
        
        g = smooth_gradient(h, beta)
        v = h - step * g
        h = project_onto_P(v, total)
        
        cur_c5 = c5_of(h)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
        
        step *= gamma_step
        if (time.time() - start) % beta_update_interval < 1e-3:
            beta = min(beta * gamma_beta, beta_max)
    
    return best_h, best_c5

def targeted_refinement(h_start, time_budget):
    """Perform deterministic refinement by targeted transfer of mass"""
    start = time.time()
    h = h_start.copy()
    C5 = c5_of(h)
    best_h = h.copy()
    best_c5 = C5
    alpha = 1.0
    max_no_improvement = 200
    no_improvement = 0
    
    while time.time() - start < time_budget and no_improvement < max_no_improvement:
        C = full_overlap(h)
        # The shifts are indexed from -N+1 to N-1
        # So argmax gives the index in the full vector
        max_idx = np.argmax(C)
        d = max_idx - (len(h) - 1)  # offset in [-(N-1) : N-1]
        
        N = len(h)
        if d >= 0:
            hi = h[:N-d]
            hj = h[d:]
        else:
            hi = h[-d:]
            hj = h[:N-(-d)]
        
        c = hi * (1 - hj)
        i_star = np.argmax(c)
        j_star = i_star + d if d >= 0 else i_star - d
        
        # Compute minimum transfer
        delta = min(alpha * c[i_star], h[i_star], 1 - h[j_star])
        
        if delta > 1e-10:
            h[i_star] -= delta
            h[j_star] += delta
            
            new_C5 = c5_of(h)
            if new_C5 < C5:
                C5 = new_C5
                best_c5 = new_C5
                best_h = h.copy()
                no_improvement = 0
            else:
                h[i_star] += delta
                h[j_star] -= delta
                no_improvement += 1
                alpha /= 2.0
        else:
            no_improvement += 1
            if no_improvement % 10 == 0:
                alpha /= 2.0
                
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start = time.time()
    N = 200
    dx = 2.0 / N
    total = N / 2

    # Generate seeds
    # 1. Uniform seed
    h_uniform = 0.5 * np.ones(N)
    h_uniform *= total / np.sum(h_uniform)

    # 2. Interpolated from initial_h_values if available
    h_initial = None
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        if len(h0) != N:
            x0 = np.linspace(0, 2, len(h0))
            x_new = np.linspace(0, 2, N)
            h0 = np.interp(x_new, x0, h0)
        h_initial = h0 * (total / np.sum(h0))
    else:
        h_initial = np.random.rand(N) * 0.5
        h_initial *= total / np.sum(h_initial)

    # 3. Double-peak seed
    h_double_peak = np.full(N, 0.1)
    a = 0.3
    idxs = np.where((np.linspace(0, 2, N) < a) | (np.linspace(0, 2, N) > (2 - a)))[0]
    h_double_peak[idxs] = 0.9
    h_double_peak *= total / np.sum(h_double_peak)

    # 4. Random Dirichlet seed
    h_dirichlet = np.random.dirichlet([0.5] * N)
    h_dirichlet = np.clip(h_dirichlet, 0, 1)
    h_dirichlet *= total / np.sum(h_dirichlet)

    best_h = None
    best_c5 = float('inf')
    
    seeds = [
        ("uniform", h_uniform),
        ("interpolated", h_initial),
        ("double_peak", h_double_peak),
        ("dirichlet", h_dirichlet)
    ]
    
    # Run projected gradient descent on all seeds
    time_per_run = 0.3 * budget_s / len(seeds)
    for name, h_seed in seeds:
        if time.time() - start > budget_s * 0.95:
            break
        h, c5 = projected_gradient_descent(h_seed, time_per_run, seed)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    
    # Apply targeted refinement to the best result
    if best_h is not None and time.time() - start < budget_s * 0.95:
        refine_budget = budget_s - (time.time() - start)
        h_refined, c5_refined = targeted_refinement(best_h, refine_budget)
        if c5_refined < best_c5:
            best_c5 = c5_refined
            best_h = h_refined
    
    return (best_h, best_c5, N)
