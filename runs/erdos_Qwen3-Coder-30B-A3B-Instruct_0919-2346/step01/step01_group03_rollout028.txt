import numpy as np
import time

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral via full convolution of h with (1-h)
    overlaps = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum of the full correlation vector
    # scaled by dx
    c5 = np.max(overlaps) * dx
    return c5

def full_overlap(h):
    """Return the raw overlap correlation array (unscaled)"""
    return np.correlate(h, 1 - h, mode="full")

def project_onto_P(v, total):
    """Project v onto the set {x : 0 <= x_i <= 1, sum x_i = total}"""
    # Use bisection to find lambda such that sum(clamp(v - lambda, 0, 1)) = total
    v = np.array(v)
    low, high = np.min(v), np.max(v) + 1.0
    tolerance = 1e-12
    
    def sum_at_lambda(lam):
        x = np.clip(v - lam, 0, 1)
        return np.sum(x) - total
    
    while high - low > tolerance:
        mid = (low + high) / 2
        if sum_at_lambda(mid) > 0:
            low = mid
        else:
            high = mid
    lam = (low + high) / 2
    return np.clip(v - lam, 0, 1)

def smooth_gradient(h, beta):
    """Compute the gradient of the smooth surrogate Φ_β(h)"""
    N = len(h)
    
    # Compute raw correlations
    F = full_overlap(h)
    # Center the F values to prevent overflow
    F_centered = F - np.max(F)
    exps = np.exp(beta * F_centered)
    Z = np.sum(exps)
    w = exps / Z  # Softmax weights
    
    g = np.ones(N)
    
    # Fill gradient using weighted shifts
    for k in range(-N + 1, N):
        wk = w[k + N - 1]  # offset to match correlation index
        if k >= 0:
            g[:N - k] -= wk * h[k:]
        else:
            d = -k
            g[d:] -= wk * h[:N - d]
    
    return g

def projected_gradient_descent(h0, time_budget, seed):
    """Projected gradient descent with β-schedule"""
    np.random.seed(seed)
    start = time.time()
    
    h = h0.copy()
    best_h = h.copy()
    c_best = c5_of(h)
    
    # Parameters
    beta = 2.0
    beta_max = 30.0
    beta_update_interval = 100
    gamma_beta = 1.5
    gamma_step = 0.9995
    step = 0.5
    step_min = 1e-6
    max_iter = 100000
    
    for iteration in range(max_iter):
        elapsed = time.time() - start
        if elapsed > time_budget:
            break
            
        # Step 1: compute correlations and weights
        F = full_overlap(h)
        F_centered = F - np.max(F)
        exps = np.exp(beta * F_centered)
        Z = np.sum(exps)
        w = exps / Z
        
        # Step 2: compute gradient of Φ_β
        g = smooth_gradient(h, beta)
        
        # Step 3: update and project
        v = h - step * g
        h = project_onto_P(v, len(h) / 2)
        
        # Step 4: evaluate true objective
        c_cur = c5_of(h)
        if c_cur < c_best:
            best_h = h.copy()
            c_best = c_cur
        
        # Step 5: schedule updates
        step = max(step * gamma_step, step_min)
        if iteration % beta_update_interval == 0:
            beta = min(beta * gamma_beta, beta_max)
            
    return best_h, c_best

def targeted_refinement(h0, time_budget, seed):
    """Deterministic pairwise mass transfer to refine the solution"""
    np.random.seed(seed)
    start = time.time()
    
    h = h0.copy()
    c_best = c5_of(h)
    alpha = 1.0
    max_no_improvements = 200
    no_improvements = 0
    
    while True:
        elapsed = time.time() - start
        if elapsed > time_budget:
            break
            
        # Compute full overlap
        C = full_overlap(h)
        d = np.argmax(C) - (len(h) - 1)  # shift that gives max overlap
        
        # Extract the overlapping slices
        if d >= 0:
            hi = h[:len(h) - d]
            hj = h[d:]
        else:
            hi = h[-d:]
            hj = h[:len(h) + d]
            
        # Find the pair that contributes most to overlap
        contrib = hi * (1 - hj)
        idx = np.argmax(contrib)
        c_max = contrib[idx]
        
        # Find which element to adjust
        i, j = idx, idx + d if d >= 0 else idx - d
        
        # Compute transfer amount
        delta = min(alpha * c_max, h[i], 1 - h[j])
        
        # Apply transfer
        h[i] -= delta
        h[j] += delta
        
        # Check if we got an improvement
        c_new = c5_of(h)
        if c_new < c_best:
            c_best = c_new
            no_improvements = 0
            alpha = 1.0  # reset to full scale
        else:
            # Revert transfer
            h[i] += delta
            h[j] -= delta
            no_improvements += 1
            # Reduce alpha if no improvement
            if no_improvements >= max_no_improvements:
                break
            alpha *= 0.5
    
    return h, c_best

    h_double_peak = np.full(N, 0.1)
    a = 0.3
    idxs = np.where((np.linspace(0, 2, N) < a) | (np.linspace(0, 2, N) > (2 - a)))[0]
    h_double_peak[idxs] = 0.9
    h_double_peak *= (total / np.sum(h_double_peak))
    
    # 4. RandomDirichlet seed
    h_random_dirichlet = np.random.dirichlet([0.5] * N)
    h_random_dirichlet = np.clip(h_random_dirichlet, 0, 1)
    h_random_dirichlet *= (total / np.sum(h_random_dirichlet))
    
    # Run projected gradient descent on seeds
    seeds = [h_uniform, h_initial, h_double_peak, h_random_dirichlet]
    best_h = None
    best_c5 = float('inf')
    
    # Allocate time for each seed
    time_per_seed = 0.3 * budget_s / len(seeds)
    
    for i, h_seed in enumerate(seeds):
        if time.time() - start > budget_s * 0.95:
            break
        h, c5 = projected_gradient_descent(h_seed, time_per_seed, seed + i)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5
    
    # Run deterministic refinement on best solution
    if best_h is not None and time.time() - start < budget_s * 0.95:
        refinement_time = budget_s - (time.time() - start)
        h_refined, c5_refined = targeted_refinement(best_h, refinement_time, seed + 100)
        if c5_refined < best_c5:
            best_h = h_refined
            best_c5 = c5_refined
    
    return (best_h, best_c5, N)
