import numpy as np
import time
import random
from scipy.signal import convolve


def compute_c5(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using convolution
    overlap = convolve(h, 1 - h, mode="full")
    c5 = np.max(overlap) * dx
    return c5


def project_h(h, dx):
    """Project h to satisfy integral constraint and bound constraints"""
    h = np.clip(h, 0, 1)
    # Use bisection to adjust scale to meet integral constraint
    target_sum = 1.0
    current_sum = np.sum(h) * dx
    if np.abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Binary search for correct scaling factor
    scale_low = 0.0
    scale_high = 2.0
    for _ in range(50):
        scale = (scale_low + scale_high) / 2
        h_test = h * scale
        h_test = np.clip(h_test, 0, 1)
        current_sum = np.sum(h_test) * dx
        if current_sum < target_sum:
            scale_low = scale
        else:
            scale_high = scale
    h = h * (scale_low + scale_high) / 2
    h = np.clip(h, 0, 1)
    return h

    h_random = np.random.uniform(0, 1, N)
    
    # Binary step functions
    h_binary = np.zeros(N)
    mid = N // 2
    h_binary[:mid] = 1.0
    
    return [h_uniform, h_provided, h_random, h_binary]


def compute_gradient(h, k):
    """Compute gradient of overlap with respect to h for fixed shift k"""
    dx = 2.0 / len(h)
    # Stack h and 1-h, then convolve
    h_stack = np.concatenate([h, np.zeros(len(h))])
    one_minus_h_stack = np.concatenate([1 - h, np.zeros(len(h))])
    # Forward convolution with 1-h
    d_overlaps = convolve(h_stack, one_minus_h_stack, mode="full")[:2*len(h)-1]
    # Return the part that corresponds to shift k
    grad = np.zeros_like(h)
    for i in range(len(h)):
        if i + k < len(h):
            grad[i] += (1 - h[i + k]) * dx
    return grad


def gradient_ascent_step(h, step_size, beta=1.0):
    """Perform one gradient ascent step on h"""
    dx = 2.0 / len(h)
    K = len(h)  # maximum shift needed
    
    # Compute all gradients for shifts k = 0 to K-1
    all_gradients = []
    for k in range(K):
        grad = np.zeros_like(h)
        for i in range(len(h)):
            if i + k < len(h):
                grad[i] += (1 - h[i + k]) * dx
            if i - k >= 0:
                grad[i] += (1 - h[i - k]) * dx
        all_gradients.append(grad)
    
    # Compute losses for each k
    losses = []
    for k in range(K):
        loss = 0.0
        for i in range(len(h)):
            if i + k < len(h):
                loss += h[i] * (1 - h[i + k]) * dx
        losses.append(loss)
    
    # Weighted softmax to focus on worst case
    loss_array = np.array(losses)
    weights = np.exp(beta * (loss_array - np.max(loss_array)))
    weights = weights / np.sum(weights) if np.sum(weights) > 0 else np.ones_like(weights)
    
    # Compute weighted gradient
    weighted_grad = np.zeros_like(h)
    for i, (grad, weight) in enumerate(zip(all_gradients, weights)):
        weighted_grad += weight * grad
    
    # Update h
    h_new = h + step_size * weighted_grad
    h_new = project_h(h_new, dx)  # Project to feasible set
    
    return h_new


def optimize_h_gradient(h0, time_budget_s):
    """Optimize h using gradient ascent with adaptive beta"""
    start = time.time()
    h = h0.copy()
    dx = 2.0 / len(h)
    
    # For each 100ms, update beta, step size
    remaining_time = time_budget_s
    step_size = 1e-3
    
    beta = 1.0
    beta_decay = 0.97  # exponential decay for beta
    
    while time.time() - start < time_budget_s - 0.1:
        # Change beta at regular intervals
        if (time.time() - start) % 0.1 < 0.01:
            beta *= beta_decay
            
        h = gradient_ascent_step(h, step_size, beta)
    
    return h


def direct_transfer(h, time_budget_s):
    """Perform direct move transfers to improve directionally"""
    start = time.time()
    
    # Do water-pouring or mass transfers directed by local worst shifts
    dx = 2.0 / len(h)
    patience = 1000
    count = 0
    
    while time.time() - start < time_budget_s - 0.1 and count < patience:
        # Find the worst shift k via a full overlap computation
        overlap = convolve(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        # Target is likely to be near the peak, so find neighboring peak or good place to perturb for burn-in

        # Try to shift mass to reduce that overlap
        # Find two points to swap mass

            continue
            
        # Determine delta (step size for mass transfer)
        min_h = min(h[i], 1 - h[j])
        max_h = min(1 - h[i], h[j])
        if max_h <= min_h or np.abs(max_h - min_h) < 1e-6:
            continue
            
        # Move half way in feasible range
        delta = (max_h - min_h) / 2
        
        # Make a move
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        h_new = project_h(h_new, dx)
        
        # Accept if it improves the worst overlap
        # Compute new C5
        overlap_new = convolve(h_new, 1 - h_new, mode="full")
        c5_new = np.max(overlap_new)*dx
        overlap_old = convolve(h, 1 - h, mode="full")
        c5_old = np.max(overlap_old)*dx
        
        if c5_new < c5_old - 1e-8:
            h = h_new.copy()
            count = 0
        else:
            count += 1
    
    return h
