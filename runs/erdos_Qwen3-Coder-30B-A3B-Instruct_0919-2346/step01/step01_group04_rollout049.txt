import numpy as np
import time
import random
from scipy.optimize import differential_evolution, minimize


def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    overlap = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(overlap) * dx
    return c5


def project_to_feasible(h, dx):
    """Project h to satisfy the integral constraint"""
    integral = np.sum(h) * dx
    if abs(integral - 1.0) > 1e-10:
        scale = 1.0 / integral
        h = h * scale
    # Ensure bounds
    h = np.clip(h, 0, 1)
    return h


def propose_move(h, max_step):
    """Propose a new feasible h vector by adjusting two elements"""
    h_new = h.copy()
    N = len(h)
    
    # Try to find a valid move
    max_tries = 100
    for _ in range(max_tries):
        i = random.randint(0, N - 1)
        j = random.randint(0, N - 1)
        if i == j:
            continue
            
        # Determine feasible interval for δ
        delta_low = max(-h_new[i], h_new[j] - 1)
        delta_high = min(1 - h_new[i], h_new[j])
        
        if delta_low >= delta_high:
            continue
            
        # Sample δ from uniform distribution over feasible range
        step = min(max_step, delta_high - delta_low)
        if step <= 0:
            continue
        delta = random.uniform(delta_low, delta_high)
        
        # Apply move
        h_new[i] += delta
        h_new[j] -= delta
        
        return h_new
    
    # If no valid move found, return original
    return h.copy()

    T_start = 0.1
    T_end = 1e-6
    max_step_start = 0.2
    
    # Set up the current state
    cur = project_to_feasible(initial_h.copy(), 2.0 / len(initial_h))

    best = cur.copy()
    c_best = c_cur
    
    # Main loop
    iteration = 0
    while time.time() - start < time_budget_s:
        t = time.time() - start
        elapsed_fraction = t / time_budget_s
        
        # Adaptive cooling schedule
        T = T_start * (T_end / T_start) ** (elapsed_fraction ** 0.8)
        step = max_step_start * (1 - elapsed_fraction)**0.5
        
        # Make a proposal
        h_proposed = propose_move(cur, step)
        c_proposed = c5_of(h_proposed)
        
        # Accept the proposal
        if c_proposed <= c_cur:
            cur = h_proposed
            c_cur = c_proposed
            if c_proposed < c_best:
                best = h_proposed.copy()
                c_best = c_proposed
        else:
            # Metropolis acceptance rule
            if random.random() < np.exp(-(c_proposed - c_cur) / T):
                cur = h_proposed
                c_cur = c_proposed
                
        iteration += 1
                
    return best, c_best
