def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral via full convolution of h with (1-h)
    overlaps = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum of the full correlation vector
    # scaled by dx
    c5 = np.max(overlaps) * dx
    return c5

def evaluate_constraint(h):
    """Evaluate the constraint sum(h) * dx = 1"""
    dx = 2.0 / len(h)
    return np.abs(np.sum(h) * dx - 1.0)

def solve_convex_relaxation(n_points):
    """Solve a convex relaxation of the problem using CVXPY"""
    dx = 2.0 / n_points

    # Objective: maximize the maximum correlation (approximation)
    # This is a heuristic relaxation, not exact.
    # Actually, let's just do a simpler approach to get an initial solution
    # For now, we'll use a very simple convex relaxation
    # We'll use a regularized version and solve using quadratic programming
    # or just do it exactly with a low-dimensional problem
    
    # Or better: we solve min sum(h) s.t. constraints and use it as starting point
    constraints = [h[i] >= 0 for i in range(n_points)]
    constraints.extend([h[i] <= 1 for i in range(n_points)])
    # Integral constraint
    constraints.append(cp.sum(h) == n_points / 2.0)  # sum(h) * dx = 1 => sum(h) = n_points/2
    
    # Try to minimize C5 by using good heuristic solution
    obj = cp.Minimize(cp.sum(h)**2)  # Simple quadratic heuristic for smoothness
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.SCS, verbose=False)
        if prob.status == 'optimal':
            return np.array(h.value).flatten()
    except:
        pass
    return None

def make_symmetric(h):
    """Make the solution approximately symmetric about the midpoint"""
    n = len(h)
    h_new = h.copy()

        j = n - 1 - i
        avg = (h[i] + h[j]) / 2.0
        h_new[i] = h_new[j] = avg
    return h_new

def local_search(h, dx, max_iter=500):

    # Try to make it close to symmetric
    current_h = make_symmetric(current_h)
    current_c5 = c5_of(current_h)

        # Try small perturbations
        h_new = current_h.copy()
        i = np.random.randint(0, len(h_new))
        j = np.random.randint(0, len(h_new))
        if i == j:
            continue
            
        # Compute bounds
        delta_low = max(-h_new[i], 1 - h_new[i])
        delta_high = min(1 - h_new[i], h_new[j])
        
        if delta_low >= delta_high:
            continue
            
        # Small moves
        delta = np.random.uniform(-0.05, 0.05)
        delta = max(delta_low, min(delta, delta_high))
        
        h_new[i] += delta
        h_new[j] -= delta
        
        # Project to feasible set
        for k in range(len(h_new)):
            h_new[k] = max(0, min(1, h_new[k]))
            
        # Enforce constraint
        scale = (len(h_new) / 2.0) / np.sum(h_new)
        h_new *= scale
        
        new_c5 = c5_of(h_new)
        if new_c5 < current_c5:
            current_c5 = new_c5
            current_h = h_new
    
    return current_h
