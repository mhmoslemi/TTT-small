import numpy as np
from scipy.optimize import minimize
import time

def project_onto_bounds_sum(v, target_sum, lower=0, upper=1):
    """Project vector v onto bounds [lower,upper] and sum constraint"""
    # Binary search for lambda
    def sum_at_lambda(lam):
        return np.clip(v - lam, lower, upper).sum()
    
    left = v.min() - upper
    right = v.max() - lower
    
    for _ in range(60):
        mid = (left + right) / 2
        s = sum_at_lambda(mid)
        if s > target_sum:
            left = mid
        else:
            right = mid
    
    lambda_val = (left + right) / 2
    return np.clip(v - lambda_val, lower, upper)

def compute_max_overlap(h):
    """Compute maximum overlap for given h"""
    n = len(h)
    dx = 2.0 / n
    correlation = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(correlation) * dx
    return max_overlap

        if max_f < best_c5:

        gamma = gamma_start + (gamma_end - gamma_start) * (i / max_iter)
        w = np.exp(gamma * (f - max_f))
        w = w / np.sum(w)
        
        # Compute gradient
        grad = np.zeros(n)

                i_start = 0
                i_end = n - l

                i_start = -l

                    # So contribution to gradient from l=0 is unclear. Skip correction.
                    pass
                else:
                    # Compute contribution from l
                    if l > 0:
                        # Add derivative from (1 - h[i+l])
                        grad[i_start:i_end] += w[l + n - 1] * (1 - h[i_start:i_end + l])

                    except:
                        pass

        # Gradient step
        h_new = h - step_size * grad
        
        # Project back to constraints
        h_new = project_onto_bounds_sum(h_new, n/2)
        h = h_new
        
    # Final evaluation
    final_c5 = compute_max_overlap(best_h)
    
    return (best_h.tolist(), final_c5, len(best_h))
