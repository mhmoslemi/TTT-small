import numpy as np
import time

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral via full convolution of h with (1-h)
    overlaps = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum of the full correlation vector
    # scaled by dx
    c5 = np.max(overlaps) * dx
    return c5

def full_overlap(h):
    """Return raw correlation vector (unscaled)."""
    return np.correlate(h, 1 - h, mode="full")

def project_onto_P(v, total):
    """Project v onto {x : 0 ≤ x_i ≤ 1, sum x_i = total}."""
    # Bisection to find lambda
    def proj(x):
        return np.clip(x - total, 0, 1)
    
    a, b = -100.0, 100.0
    for _ in range(100):
        lam = (a + b) / 2.0
        x = proj(v + lam)
        s = np.sum(x)
        if s > total:
            a = lam
        else:
            b = lam
    return proj(v + a)

def smooth_gradient(h, beta):
    """Compute the gradient of the smooth surrogate function Φ_β(h)."""
    N = len(h)
    # Compute F_k for all shifts k
    F = full_overlap(h)
    # For numerical stability, subtract the maximum before exponentiating
    F_max = np.max(F)
    F_shifted = F - F_max
    exp_F = np.exp(beta * F_shifted)
    Z = np.sum(exp_F)
    weights = exp_F / Z  # Softmax weights

    # Gradient computation using weighted sum of local changes
    grad = np.ones(N)
    # Loop over each k from -N+1 to N-1 (but only non-zero shifts)
    for k in range(-N+1, N):
        if k == 0:
            continue
        wk = weights[k + N - 1]  # Index adjustment as k+1 gives 0:N-1
        if k > 0:
            # Shift h forward by k
            grad[:N-k] -= wk * h[k:]
        else:
            # Shift h backward by |k|
            d = abs(k)
            grad[d:] -= wk * h[:N-d]
    return grad

def targeted_refinement(h, time_budget):
    """Perform deterministic local refinement to lower the worst overlap."""
    start = time.time()
    h = h.copy()
    best_c5 = c5_of(h)
    alpha = 1.0
    max_attempts = 1000
    failed_attempts = 0

    while time.time() - start < time_budget and failed_attempts < max_attempts:
        # Compute full overlap
        C = full_overlap(h)
        # Find worst shift d, mapping from full correlation index to shift
        d = np.argmax(C) - (len(h) - 1)
        # Extract overlapping slices according to shift d
        if d >= 0:
            hi = h[:len(h) - d]
            hj = h[d:]
        else:
            d = abs(d)
            hi = h[d:]
            hj = h[:len(h) - d]

        # Compute false overlaps for each pair
        false_overlap = hi * (1 - hj)
        # Find the maximum false overlap and its indices
        i_star, j_star = np.unravel_index(np.argmax(false_overlap), false_overlap.shape)
        if d >= 0:
            j_star += d
        else:
            i_star += d

        # Compute transfer amount
        c = false_overlap[i_star - d if d < 0 else i_star]
        delta = min(alpha * c, h[i_star], 1 - h[j_star])
        
        # Apply transfer if it is positive
        if delta > 1e-12:
            h[i_star] -= delta
            h[j_star] += delta
            new_c5 = c5_of(h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                failed_attempts = 0  # Reset failures on improvement
                alpha = 1.0  # Reset learning rate on success
            else:
                # Undo the move
                h[i_star] += delta
                h[j_star] -= delta
                alpha *= 0.5  # Reduce learning rate
                failed_attempts += 1
        else:
            alpha *= 0.5
            failed_attempts += 1

    return h, best_c5

def projected_gradient_descent(h0, time_budget, beta_start=2.0, step_start=0.5, gamma_step=0.9995, gamma_beta=1.5):
    """Perform projected gradient descent with β-scheduling and step-size decay."""
    start = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of(h)
    beta = beta_start
    step = step_start
    
    # Inner loop
    max_iter = 100000
    for iter_count in range(max_iter):
        if time.time() - start > time_budget:
            break
            
        # Schedule update
        if iter_count % 10 == 0:
            beta = min(beta * gamma_beta, 30.0)
        step = max(step * gamma_step, 1e-4)

        # Compute F_k and weights
        F = full_overlap(h)
        Fmax = np.max(F)
        F_shifted = F - Fmax
        exp_F = np.exp(beta * F_shifted)
        Z = np.sum(exp_F)
        weights = exp_F / Z

        # Compute gradient
        grad = np.ones_like(h)
        N = len(h)
        for k in range(-N+1, N):
            if k == 0:
                continue
            wk = weights[k + N - 1]
            if k > 0:
                grad[:N-k] -= wk * h[k:]
            else:
                d = -k
                grad[d:] -= wk * h[:N-d]

        # Gradient step and projection
        v = h - step * grad
        h = project_onto_P(v, len(h)/2)

        # New evaluation
        cur_c5 = c5_of(h)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()

    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start = time.time()

    # Target resolution
    N = 200
    dx = 2.0 / N

    # Reuse the initial_h_values if provided
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        x0 = np.linspace(0, 2, len(h0))
        x_new = np.linspace(0, 2, N)
        h_interp = np.interp(x_new, x0, h0)
        h_initial = h_interp * (N / 2) / np.sum(h_interp)
    else:
        h_initial = 0.5 * np.ones(N)

    # Two peaks initialization
    h_double_peak = np.full(N, 0.1)
    a = 0.3
    idxs = np.where((np.linspace(0, 2, N) < a) | (np.linspace(0, 2, N) > (2 - a)))[0]
    h_double_peak[idxs] = 0.9
    h_double_peak *= (N / 2) / np.sum(h_double_peak)

    # Uniform distribution
    h_uniform = 0.5 * np.ones(N)
    h_uniform *= (N / 2) / np.sum(h_uniform)

    # Dirichlet random (α=0.5)
    h_dirichlet = np.random.dirichlet([0.5] * N)
    h_dirichlet = np.clip(h_dirichlet, 0, 1)
    h_dirichlet *= (N / 2) / np.sum(h_dirichlet)

    # Run each algorithm with time budget
    best_h = None
    best_c5 = float('inf')
    remaining_budget = budget_s * 0.3
    seeds = [h_uniform, h_initial, h_double_peak, h_dirichlet]

    for i, h_seed in enumerate(seeds):
        if time.time() - start > budget_s * 0.95:
            break
        h, c5 = projected_gradient_descent(h_seed, remaining_budget, beta_start=2.0)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5

    # Polishing pass with targeted refinement
    if best_h is not None and time.time() - start < budget_s * 0.95:
        refine_budget = budget_s - (time.time() - start)
        h_final, c5_final = targeted_refinement(best_h, refine_budget)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final

    return (best_h, best_c5, N)
