import numpy as np
import time

def c5_of(h):
    """Compute the maximum overlap integral for the given step function h."""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    c = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(c) * dx
    return c5

def worst_shift(h):
    """Find the shift that yields the maximum overlap integral."""
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    return idx - (len(h) - 1)

def subgradient(h, s):
    """Compute the subgradient of C5 at shift s."""
    N = len(h)
    dx = 2.0 / N
    m = abs(s)
    g = np.full(N, dx)  # base vector of dx
    if m < N:
        g[:N-m] -= dx * h[m:]  # term with h[i+m]
        g[m:]   -= dx * h[:N-m]  # term with h[i-m]
    return g

def project(v, target):
    """Euclidean projection of v onto [0,1]^N with sum = target."""
    # Binary search for lambda
    lo = -np.max(v)
    hi = 1.0 - np.min(v)
    for _ in range(50):
        lam = (lo + hi) / 2.0
        w = np.clip(v + lam, 0.0, 1.0)
        if w.sum() < target:
            lo = lam
        else:
            hi = lam
    w = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix any rounding errors
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        free = (w > 0) & (w < 1)
        if free.any():
            w[free] += diff / free.sum()
            w = np.clip(w, 0.0, 1.0)
    return w

def spd_loop(initial_h, time_budget):
    """Sub-gradient projected descent loop."""
    start = time.time()
    h = initial_h.copy()
    c_prev = c5_of(h)
    best = h.copy()
    best_val = c_prev
    # Schedule parameters
    eta0 = 30.0
    gamma = 0.9995
    t = 0
    
    while time.time() - start < time_budget:
        s = worst_shift(h)
        g = subgradient(h, s)
        # Adaptive step size
        eta = eta0 * (gamma ** t)
        if eta < 1e-8:
            eta = 1e-8
        # Update
        v = h - eta * g
        h_new = project(v, len(h) / 2)
        c_new = c5_of(h_new)
        # Accept only if improving
        if c_new < c_prev:
            h = h_new
            c_prev = c_new
            if c_new < best_val:
                best = h.copy()
                best_val = c_new
        else:
            # Small jitter every 30 steps to escape shallow plateaux
            if t % 30 == 0:
                j = np.random.uniform(-0.02, 0.02, size=len(h))
                v = h + j
                h_new = project(v, len(h) / 2)
                c_new = c5_of(h_new)
                if c_new < c_prev:
                    h = h_new
                    c_prev = c_new
                    if c_new < best_val:
                        best = h.copy()
                        best_val = c_new
        t += 1
    return best, best_val

def anneal_loop(initial_h, time_budget):
    """Simulated annealing loop."""
    start = time.time()
    T_start = 0.05
    T_end = 1e-6
    step_start = 0.2
    h = initial_h.copy()
    c_best = c5_of(h)
    best = h.copy()
    best_val = c_best
    t = 0
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T_start * (T_end / T_start) ** (elapsed / time_budget)
        step = step_start * (1.0 - elapsed / time_budget)
        # Propose move
        N = len(h)
        while True:
            i, j = np.random.choice(N, size=2, replace=False)
            delta_low = max(-h[i], h[j] - 1)
            delta_high = min(1 - h[i], h[j])
            if delta_low >= delta_high:
                continue  # Try another pair
            step_size = np.random.uniform(max(-step, delta_low),
                                          min(step, delta_high))
            # Create new h
            h_new = h.copy()
            h_new[i] += step_size
            h_new[j] -= step_size
            break
        c_new = c5_of(h_new)
        # Accept or reject
        if c_new <= c_best or np.random.rand() < np.exp(-(c_new - c_best) / T):
            h = h_new
            c_best = c_new
            if c_new < best_val:
                best = h.copy()
                best_val = c_new
        t += 1
    return best, best_val

def hill_climb(initial_h, time_budget):
    """Deterministic local search/hill climb."""
    start = time.time()
    h = initial_h.copy()
    c_best = c5_of(h)
    best = h.copy()
    best_val = c_best
    while time.time() - start < time_budget:
        N = len(h)
        improved = False
        for _ in range(10):  # Sample a few moves
            # Make random pair move
            i, j = np.random.choice(N, size=2, replace=False)
            # Feasible step
            delta_low = max(-h[i], h[j] - 1)
            delta_high = min(1 - h[i], h[j])
            if delta_low >= delta_high:
                continue
            step_size = np.random.uniform(max(-0.02, delta_low),
                                          min(0.02, delta_high))
            h_new = h.copy()
            h_new[i] += step_size
            h_new[j] -= step_size
            c_new = c5_of(h_new)
            if c_new < c_best:
                h = h_new
                c_best = c_new
                best = h.copy()
                best_val = c_best
                improved = True
        if not improved:
            break
    return best, best_val

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    N = 140
    dx = 2.0 / N
    
    # Phase 0: Seeding
    starts = []
    if 'initial_h_values' in globals():
        h0 = np.interp(np.linspace(0, 2, N), 
                      np.linspace(0, 2, len(initial_h_values)), 
                      initial_h_values)
        h0 = h0 * (N/2) / np.sum(h0)
        starts.append(h0)
    else:
        # fallback
        h0 = np.ones(N) * 0.5
        starts.append(h0)
    # Two-peak seeded
    h1 = np.ones(N)
    a_range = np.linspace(0.12, 0.38, 5)
    for a_val in a_range:
        u = 0.85
        v = 0.15
        mask = (np.linspace(0, 2, N) < a_val) | (np.linspace(0, 2, N) >= 2 - a_val)
        h1[mask] = u
        h1[~mask] = v
        h1 = h1 * (N/2) / np.sum(h1)
        starts.append(h1)
    # Binary random
    h2 = np.random.choice([0, 1], size=N, p=[0.5, 0.5])
    h2 = h2 * (N/2) / np.sum(h2)
    starts.append(h2)
    # Uniform
    h3 = np.ones(N) * 0.5
    starts.append(h3)

    # Evaluate all seeds
    c_vals = [c5_of(h_start) for h_start in starts]
    best_start_idx = np.argmin(c_vals)
    best = starts[best_start_idx].copy()
    best_val = c_vals[best_start_idx]

    t_start = time.time()
    time_budget = budget_s * 0.99  # Leave a bit of headroom
    
    # Phase 1: SPD
    spd_budget = 0.55 * time_budget
    h_spd, c_spd = spd_loop(best, spd_budget)
    if c_spd < best_val:
        best = h_spd
        best_val = c_spd

    # Phase 2: Annealing
    ann_budget = 0.30 * time_budget
    h_ann, c_ann = anneal_loop(best, ann_budget)
    if c_ann < best_val:
        best = h_ann
        best_val = c_ann

    # Phase 3: Hill Climbing
    hc_budget = 0.10 * time_budget
    h_hc, c_hc = hill_climb(best, hc_budget)
    if c_hc < best_val:
        best = h_hc
        best_val = c_hc

    # Final re-evaluation
    final_c = c5_of(best)
    return (best, final_c, N)
