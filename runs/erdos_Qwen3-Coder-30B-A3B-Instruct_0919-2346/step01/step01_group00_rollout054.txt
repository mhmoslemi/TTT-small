def project_h(h, target_sum):
    """Project h onto box constraints [0,1] and sum constraint."""
    # Clip to [0, 1]
    h_clipped = np.clip(h, 0, 1)

    if np.isclose(current_sum, target_sum):
        return h_clipped

    # Hessian: sum of (1 if 0 < h_i - lambda < 1)

    # But that requires that some free variables are not at limits

    diff = current_sum - target_sum

    # Continue properly handling difference
    if diff > 0:

        # but simple: we're off by small amount, distribute
        weights = 1 - h_proj  # amount we can increase
        weights[weights < 0] = 0
        if np.sum(weights) > 1e-10:
            scale = diff / np.sum(weights)

def compute_C_and_grad(h, alpha=50.0):
    """Compute the objective and its gradient."""
    N = len(h)
    dx = 2.0 / N
    
    # Correlation of h with (1 - h)
    # This should be maximum of overlaps
    c_full = np.correlate(h, 1 - h, mode='full') * dx
    c = c_full[len(h) - 1:]  # Only positive shifts
    
    # Softmax version to allow differentiation
    c_exp = np.exp(alpha * c)
    c_sum = np.sum(c_exp)
    if c_sum < 1e-10:
        w = np.ones_like(c) / len(c)
    else:
        w = c_exp / c_sum

    # Gradient computation
    # ∂x_i of C_k = dx * (1 - h_{i+k} - h_{i-k})
    # So we compute negative Jacobian to get descent direction
    # Just using same approach for convolutions as before
    # But zero pad needed?
    shifted_plus = np.convolve(h, w[::-1], mode='same')
    shifted_minus = np.convolve(h, w, mode='same')
    g = dx * (1 - (shifted_plus + shifted_minus))

    # Compute actual value of C
    max_c = np.max(c_full)  # Custom to include full

    return max_c, g
