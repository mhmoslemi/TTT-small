import numpy as np
import time
import random

def c5_of(h):
    """Compute C5 = max_k integral h(x)(1-h(x+k)) dx."""
    dx = 2.0 / len(h)
    # Compute correlation: h * (1 - h) with full mode
    c = np.correlate(h, 1 - h, mode="full")
    # Multiply by dx to get continuous integral
    c *= dx
    # Return maximum value among all shifts
    return np.max(c)

def evaluate_h(h):
    """Evaluate a solution and return its C5 value."""
    return c5_of(h)

def project_to_feasible(h):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) * dx = 1"""
    dx = 2.0 / len(h)
    h = np.clip(h, 0, 1)
    # Renormalize
    h = h * (1.0 / np.sum(h)) * (2.0 / dx)
    return h

def perturb_solution(h, step_size=0.05):
    """Generate perturbed solution by adjusting a few points."""
    h = h.copy()
    N = len(h)
    n_perturb = max(1, N // 20)
    
    indices = random.sample(range(N), min(n_perturb, N))
    for i in indices:
        delta = random.uniform(-step_size, step_size)
        h[i] += delta
    
    return project_to_feasible(h)

def genetic_search(initial_h, time_budget, pop_size=30, generations=100):
    """Perform genetic search on the solutions."""
    start = time.time()
    population = [initial_h.copy()]
    for _ in range(pop_size - 1):
        h = perturb_solution(initial_h, 0.1)
        population.append(h)
    
    best_h = initial_h.copy()
    best_c5 = c5_of(best_h)
    
    for gen in range(generations):
        if time.time() - start > time_budget * 0.95:
            break
            
        # Evaluate population
        fitness = [c5_of(h) for h in population]
        
        # Keep best
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = fitness[best_idx]
        
        # Selection - tournament
        new_pop = [population[best_idx]]  # Elitism
        for _ in range(pop_size - 1):
            # Tournament selection
            idx1, idx2 = random.sample(range(pop_size), 2)
            parent = population[idx1] if fitness[idx1] < fitness[idx2] else population[idx2]
            child = perturb_solution(parent, 0.05)
            new_pop.append(child)
        
        population = new_pop
    
    return best_h, best_c5

def improved_local_search(initial_h, time_budget):
    """Combine several local search strategies."""
    start = time.time()
    best_h = initial_h.copy()
    best_c5 = c5_of(best_h)
    
    # First try random local search
    cur = best_h.copy()
    cur_c5 = best_c5
    
    # Simulated Annealing-like approach but more structured
    T_start = 0.1
    T_end = 1e-6
    steps = 0
    max_steps = 10000
    
    while steps < max_steps and time.time() - start < time_budget * 0.9:
        steps += 1
        # Pick two indices to modify
        i, j = random.sample(range(len(cur)), 2)
        
        # Perturbation
        old_i, old_j = cur[i], cur[j]
        new_i = old_i + random.uniform(-0.05, 0.05)
        new_j = old_j - (new_i - old_i)  # Keep sum constraint roughly
        
        # Clip and renormalize
        new_h = cur.copy()
        new_h[i] = new_i
        new_h[j] = new_j
        
        new_h = project_to_feasible(new_h)
        
        # Accept with probability
        new_c5 = c5_of(new_h)
        if new_c5 < cur_c5:
            cur = new_h
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_h = new_h.copy()
                best_c5 = new_c5
        else:
            # Accept with probability
            t = (time.time() - start) / time_budget
            if t < 1.0:
                T = T_start * (T_end / T_start) ** t
                if random.random() < np.exp(-(new_c5 - cur_c5) / T):
                    cur = new_h
                    cur_c5 = new_c5
            
    return best_h, best_c5
