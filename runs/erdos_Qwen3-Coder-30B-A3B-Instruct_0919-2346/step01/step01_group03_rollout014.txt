import numpy as np
import time
from scipy.optimize import differential_evolution, minimize

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral via full convolution of h with (1-h)
    overlaps = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum of the full correlation vector
    # scaled by dx
    c5 = np.max(overlaps) * dx
    return c5

def constrain_sum(h, target_sum):
    """Adjust h to have the target sum while keeping bounds"""
    current_sum = np.sum(h)
    if current_sum == target_sum:
        return h
    scale = target_sum / current_sum
    h_scaled = h * scale
    # Ensure bounds are respected
    h_scaled = np.clip(h_scaled, 0, 1)
    return h_scaled

def smooth_h(h, sigma=0.1):
    """Apply Gaussian smoothing to h"""
    if len(h) < 3:
        return h
    # Apply Gaussian kernel
    kernel = np.exp(-np.linspace(-2, 2, 21)**2 / (2 * sigma**2))
    kernel = kernel / np.sum(kernel)
    return np.convolve(h, kernel, mode='same')

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start = time.time()
    N = 140
    dx = 2.0 / N
    
    # Generate initial feasible solution
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        if len(h0) != N:
            x0 = np.linspace(0, 2, len(h0))
            x_new = np.linspace(0, 2, N)
            h0 = np.interp(x_new, x0, h0)
    else:
        # Create a reasonable starting point
        h0 = np.ones(N) * 0.5
        # Add some structure to break symmetry
        x = np.linspace(0, 2, N)
        h0 += 0.1 * np.sin(3 * np.pi * x)  # Some oscillation
        h0 = np.clip(h0, 0, 1)
    
    # Rescale to have integral = 1
    h0 = constrain_sum(h0, N/2)
    
    # Objective function to minimize
    def objective(h):
        return c5_of(h)
    
    # Constraint function to maintain sum
    def constraint_sum(h):
        return np.sum(h) * dx - 1.0  # Should be 0
    
    # Bound constraints: 0 <= h[i] <= 1
    bounds = [(0, 1) for _ in range(N)]
    
    # Optimization using differential evolution for global search
    # When there's little time left, do local optimization on best solution
    remaining_time = budget_s - (time.time() - start)
    
    # Use differential evolution as main optimization
    de_result = differential_evolution(
        objective,
        bounds,
        seed=seed,
        maxiter=max(100, int(remaining_time / 30)),
        popsize=10,
        mutation=(0.5, 1),
        recombination=0.7,
        atol=1e-6,
        rtol=1e-6
    )
    
    best_h = de_result.x.copy()
    best_c5 = objective(best_h)
    
    # Further refine with local optimization if time permits
    remaining_time = budget_s - (time.time() - start)
    
    if remaining_time > 5:
        # Local refinement around differential evolution result
        # Set up constraints
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Refinement using scipy.optimize.minimize
        ref_result = minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'gtol': 1e-9},
            maxiter=max(50, int(remaining_time / 10))
        )
        
        if ref_result.success and ref_result.fun < best_c5:
            best_h = ref_result.x.copy()
            best_c5 = ref_result.fun
    
    # Ensure final constraint is satisfied
    best_h = constrain_sum(best_h, N/2)
    final_c5 = c5_of(best_h)
    
    # If we have extra time, try one more local search with a different method
    remaining_time = budget_s - (time.time() - start)
    if remaining_time > 5:
        # Try with L-BFGS-B (more local)
        ref_result = minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'ftol': 1e-10, 'gtol': 1e-10},
            maxiter=max(30, int(remaining_time / 20))
        )
        
        if ref_result.success and ref_result.fun < final_c5:
            best_h = ref_result.x.copy()
            final_c5 = ref_result.fun

    return (best_h, final_c5, N)
