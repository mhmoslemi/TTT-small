```python
import numpy as np
import time
import cvxpy as cp

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using full convolution
    overlap = np.correlate(h, 1 - h, mode='full')
    # The maximum value scaled by dx
    c5 = np.max(overlap) * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Try to solve the MILP model
    N = 140
    dx = 2.0 / N
    
    # Create the optimization variables
    x = cp.Variable(N, boolean=True)
    C = cp.Variable()
    
    # Set constraints
    constraints = [cp.sum(x) == N // 2]
    
    # Compute constraints for each lag
    for ell in range(1, N):
        length = N - ell
        if length <= 0:
            continue
            
        # Compute H_ℓ: number of ones in first N-ell positions
        H = cp.sum(x[:length])
        
        # Introduce binary variables for products
        y = cp.Variable(length, boolean=True)
        
        # Add constraints ensuring y[i] = x[i] * x[i+ell]
        constraints += [y <= x[:length]]
        constraints += [y <= x[ell:]]
        constraints += [y >= x[:length] + x[ell:] - 1]
        
        # Compute O_ℓ: sum of products
        O = cp.sum(y)
        
        # Compute c_ℓ and add constraints
        c = H - O
        constraints += [c <= C]
        constraints += [c >= 0]  # c >= 0 always true
    
    # Create and solve the problem
    prob = cp.Problem(cp.Minimize(C), constraints)
    
    # Set max time for solver
    remaining_time = budget_s - (time.time() - start_time) - 5
    if remaining_time <= 0:
        return None, None, None
        
    try:
        prob.solve(solver=cp.CBC, max_seconds=remaining_time, verbose=False)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE, cp.FEASIBLE]:
            # Extract the solution
            sol_x = np.array(x.value).round().astype(int)
            sol_h = sol_x.astype(float)
            
            # Verify sum constraint
            if not np.isclose(np.sum(sol_h), N // 2):
                # Fallback to annealing if constraint not met
                return None, None, None
                
            # Compute C5 using original function
            c5_bound = c5_of(sol_h)
            
            # Return result
            return (sol_h, c5_bound, N)
        else:
            # If solving failed, fallback
            return None, None, None
    
    except:
        # Fallback to simulated annealing if MILP fails
        return None, None, None
```
