import numpy as np
import time

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral via full convolution of h with (1-h)
    overlaps = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum of the full correlation vector
    # scaled by dx
    c5 = np.max(overlaps) * dx
    return c5

def full_overlap(h):
    """Return correlation vector without scaling"""
    return np.correlate(h, 1 - h, mode="full")

def project_onto_P(v, total):
    """Project v onto {x : 0 <= x_i <= 1, sum x_i = total}"""
    # Use bisection to find lambda such that sum(clamp(v - lambda, 0, 1)) = total
    lmbda_min, lmbda_max = np.min(v - 1), np.max(v)
    tol = 1e-12
    for _ in range(100):  # at most 100 bisections
        lmbda = (lmbda_min + lmbda_max) / 2
        x = np.clip(v - lmbda, 0, 1)
        s = np.sum(x)
        if s > total:
            lmbda_min = lmbda
        elif s < total:
            lmbda_max = lmbda
        else:
            return x
    return x

def smooth_gradient(h, beta):
    """Compute gradient of Phi_beta = (1/beta) * log(sum exp(beta * F_k))"""
    # F_k = N/2 - sum h_i h_{i+k}
    # grad F_k = 1 - (h_{i+k} + h_{i-k}) where out-of-range terms ignored
    N = len(h)
    F = full_overlap(h)
    # subtract max for numerical stability
    F_stable = F - np.max(F)
    exp_F = np.exp(beta * F_stable)
    w = exp_F / np.sum(exp_F)  # soft-max weights
    
    # Compute gradient components
    g = np.ones(N)
    # Loop through each shift k
    for k in range(-N + 1, N):
        wk = w[k + N - 1]  # index matching the mode='full' convention
        if k >= 0:
            g[:N - k] -= wk * h[k:]
            g[k:] -= wk * h[:N - k]
        else:
            d = -k  # d > 0
            g[d:] -= wk * h[:N - d]
            g[:N - d] -= wk * h[d:]
    return g

def projected_gradient_descent(h0, time_budget, beta_start=2.0, beta_max=30.0,
                              step_start=0.5, gamma_step=0.9995, gamma_beta=1.5,
                              beta_update_interval=50):
    """Run projected gradient descent with smoothing to minimize the surrogate"""
    start = time.time()
    h = h0.copy()
    best_h = h.copy()
    c_cur = c5_of(h)
    best_c5 = c_cur
    beta = beta_start
    step = step_start

    N = len(h)
    total = N / 2
    
    iteration = 0
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        if iteration % beta_update_interval == 0:
            beta = min(beta * gamma_beta, beta_max)
            
        # Evaluate overlaps and compute soft-max weights
        F = full_overlap(h)
        F_stable = F - np.max(F)
        exp_F = np.exp(beta * F_stable)
        w = exp_F / np.sum(exp_F)

        # Compute gradient of Phi_beta
        g = smooth_gradient(h, beta)
        
        # Tentative update and project
        v = h - step * g
        h = project_onto_P(v, total)
        
        # Evaluate true objective
        cur_c5 = c5_of(h)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
            
        # Update step size
        step = max(step * gamma_step, 1e-5)
        
        iteration += 1
        
    return best_h, best_c5

def targeted_refinement(h, time_budget, alpha_start=1.0):
    """Apply greedy local improvement to the worst pair"""
    start = time.time()
    h = h.copy()
    c_prev = c5_of(h)
    alpha = alpha_start
    
    max_iter = 200
    no_improve_count = 0
    
    while time.time() - start < time_budget and no_improve_count < max_iter:
        # Compute correlation
        C = full_overlap(h)
        d = np.argmax(C) - (len(h) - 1)
        # Extract overlapping slices
        if d >= 0:
            hi = h[:len(h) - d]
            hj = h[d:]
        else:
            hi = h[-d:]
            hj = h[:len(h) + d]
        # Contribution of each pair
        c = hi * (1 - hj)
        # Find worst pair
        i_star = np.argmax(c)
        j_star = i_star + d if d >= 0 else i_star - d
        # Compute feasible transfer
        cap = min(alpha * c[i_star], h[i_star], 1 - h[j_star])
        if cap <= 0:
            no_improve_count += 1
            alpha *= 0.5
            if alpha < 1e-8:
                break
            continue
        # Apply transfer
        h[i_star] -= cap
        h[j_star] += cap
        # Re-evaluate
        c_new = c5_of(h)
        if c_new >= c_prev:
            # Revert
            h[i_star] += cap
            h[j_star] -= cap
            no_improve_count += 1
            alpha *= 0.5  # Decrease alpha
        else:
            c_prev = c_new
            no_improve_count = 0
            alpha = alpha_start  # Reset
            
    return h, c_prev

    a = 0.3

    h_double_peak = h_double_peak * (total / np.sum(h_double_peak))

    dirichlet = np.random.dirichlet([0.5] * N)
    h_dirichlet = np.clip(dirichlet, 0, 1)
    h_dirichlet *= total / np.sum(h_dirichlet)

    # Run the optimization pipeline
    seeds = [h_uniform, h_initial, h_double_peak, h_dirichlet]
    best_h = None
    best_c5 = float('inf')

    remaining_budget = budget_s
    # Allocate 30% for smoothing per seed, 50% to refinement, 20% to retries
    time_per_seed = 0.3 * budget_s / len(seeds)
    time_for_refinement = 0.5 * budget_s
    time_for_retry = 0.2 * budget_s

    # First pass: projected gradient with smoothing
    for h_seed in seeds:
        if time.time() - start > 0.9 * budget_s:
            break
        if remaining_budget <= 0:
            break
            
        h, c5 = projected_gradient_descent(h_seed, time_per_seed)
