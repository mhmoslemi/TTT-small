import numpy as np
import time
import random
from scipy.optimize import dual_annealing, differential_evolution


def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    overlap = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(overlap) * dx
    return c5


def propose_move(h, max_step):
    """Propose a new feasible h vector by adjusting two elements"""
    h_new = h.copy()
    N = len(h)
    
    # Try to find a valid move
    max_tries = 100
    for _ in range(max_tries):
        i = random.randint(0, N - 1)
        j = random.randint(0, N - 1)
        if i == j:
            continue
            
        # Determine feasible interval for δ
        delta_low = max(-h_new[i], h_new[j] - 1)
        delta_high = min(1 - h_new[i], h_new[j])
        
        if delta_low >= delta_high:
            continue
            
        # Sample δ from uniform distribution over feasible range
        step = min(max_step, delta_high - delta_low)
        if step <= 0:
            continue
        delta = random.uniform(delta_low, delta_high)
        
        # Apply move
        h_new[i] += delta
        h_new[j] -= delta
        
        return h_new
    
    # If no valid move found, return original
    return h.copy()


def initialize_h(N):
    """Initialize h with a smart configuration"""
    # Start with initial_h_values if available
    if 'initial_h_values' in globals() and len(initial_h_values) >= N:
        h_init = np.interp(np.linspace(0, 2, N), 
                          np.linspace(0, 2, len(initial_h_values)), 
                          initial_h_values)
    else:
        # Create a reasonable initial construction
        h_init = np.ones(N) * 0.5
    
    # Normalize to sum to N/2
    h_init = h_init * (N / 2) / np.sum(h_init)
    return h_init

                    tol=1e-6,
                    recombination=0.7,
                    atol=1e-6,
                    updating='immediate',  # Faster convergence
                    workers=1
                )
                
                h_opt = de_result.x
                h_opt = h_opt * (N / 2) / np.sum(h_opt)
                c5_opt = c5_of(h_opt)
                
                if c5_opt < best_c5:
                    best_c5 = c5_opt
