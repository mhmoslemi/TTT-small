import numpy as np
import time

def softmax_weights(f_values, gamma):
    """Compute softmax weights for a list of function values"""
    f_values = np.array(f_values)
    exp_vals = np.exp(gamma * f_values)
    return exp_vals / np.sum(exp_vals)

def compute_shifted_correlations(h, dx):
    """Compute correlation values for all lags using np.correlate"""
    # np.correlate with 'full' mode gives correlations for all lags
    # We must compute h(i)(1 - h(i + k)) for each k
    corr = np.correlate(h, 1 - h, mode='full')
    # Reshape so we have a 1D array of correlations for each lag
    return corr * dx

def compute_gradient(h, shifts, dx, weights):
    """Compute gradient ∇J_γ with current weights"""
    n = len(h)
    grad = np.zeros_like(h)
    for s, w in zip(shifts, weights):
        # Iterate over all positions i where h[i] and h[i+s] can both be accessed
        g_s = np.zeros_like(h)
        if s >= 0:
            # indices: 0, ..., n-1-s  (h[i], h[i+s] both valid)
            g_s[:n-s] = 1 - h[s:]  # g_s[i] = 1 - h[i+s]
            if s > 0:
                g_s[s:] -= h[:-s]  # g_s[i] -= h[i-s] if i >= s
        else:
            # indices: -s, ..., n-1  (h[i], h[i+s] both valid)
            g_s[-s:] = 1 - h[:n+s]  # g_s[i] = 1 - h[i+s]
            g_s[:n+s] -= h[-s:]  # g_s[i] -= h[i-s] if i - s >= 0
        grad = grad + w * dx * g_s
    return grad

def project_to_simplex(v, target_sum):
    """Projection onto simplex: adjust v to sum to target_sum with bounds [0,1]"""
    v = np.array(v)
    n = len(v)
    # Compute the full projection with Lagrange multiplier
    # Need to find lambda such that sum of clip(v_i - lambda, 0, 1) = target_sum
    # Use binary search over lambda
    lower, upper = np.min(v), np.max(v) + 1.0 # We know lambda is between these
    def phi(lam):
        return np.sum(np.clip(v - lam, 0, 1)) - target_sum
    for _ in range(100):
        mid = (lower + upper) / 2.0
        val = phi(mid)
        if val > 0:
            lower = mid
        else:
            upper = mid
    lambda_ = (lower + upper) / 2.0
    projected = np.clip(v - lambda_, 0, 1)
    return projected

def upsample_initial(h0, new_n):
    """Upsample a discrete function h0 to new_n points via linear interpolation"""
    n0 = len(h0)
    x0 = np.linspace(0, 2, n0)
    x_new = np.linspace(0, 2, new_n)
    h_up = np.interp(x_new, x0, h0)
    return h_up

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value scaled by dx
    max_overlap = np.max(correlation) * dx
    return max_overlap
