import numpy as np
from scipy.optimize import minimize
import time

def project_onto_bounds_sum(v, target_sum, lower=0, upper=1):
    """Project vector v onto box constraints [lower,upper] with sum constraint"""
    # Binary search for the correct λ
    def evaluate_lambda(lam):
        projected = np.clip(v - lam, lower, upper)
        return np.sum(projected) - target_sum

    left, right = v.min() - upper, v.max() - lower
    for _ in range(60):
        mid = (left + right) / 2
        diff = evaluate_lambda(mid)
        if diff > 0:
            left = mid
        else:
            right = mid
    
    lam = (left + right) / 2
    return np.clip(v - lam, lower, upper)

def compute_max_overlap(h):
    """Compute the maximum overlap integral C5 for step function h"""
    n = len(h)
    dx = 2.0 / n
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value scaled by dx
    max_overlap = np.max(correlation) * dx
    return max_overlap
