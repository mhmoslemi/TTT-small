import numpy as np
import time
import random
from scipy.optimize import minimize

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using full convolution
    overlap = np.correlate(h, 1 - h, mode='full')
    # The maximum value scaled by dx
    c5 = np.max(overlap) * dx
    return c5

def evaluate_constrained(h, dx):
    """Evaluate constraint and objective"""
    constraint = abs(np.sum(h) * dx - 1.0)
    return constraint

def propose_move(h, max_step):
    """Propose a move that maintains sum constraint exactly"""
    h_new = h.copy()
    N = len(h)
    while True:
        i, j = random.sample(range(N), 2)
        # Compute feasible range for delta
        delta_low = max(-h_new[i], h_new[j] - 1.0)
        delta_high = min(1.0 - h_new[i], h_new[j])
        if delta_low >= delta_high:
            continue  # No valid move, try again
        break
    
    # Sample delta from feasible interval
    step = min(max_step, delta_high - delta_low)
    if step <= 0:
        delta = 0.0
    else:
        delta = random.uniform(max(-step, delta_low), min(step, delta_high))
        
    h_new[i] += delta
    h_new[j] -= delta
    return h_new

def simulated_annealing(initial_h, time_budget):
    """Simulated annealing optimization"""
    start = time.time()
    cur = initial_h.copy()
    best = initial_h.copy()
    c_cur = c5_of(cur)
    c_best = c_cur

    T_start = 0.05
    T_end = 1e-6
    step_start = 0.2
    
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T_start * (T_end / T_start) ** (elapsed / time_budget)
        step = step_start * (1.0 - elapsed / time_budget)
        
        h_prop = propose_move(cur, step)
        c_prop = c5_of(h_prop)
        
        if c_prop < c_best:
            best = h_prop.copy()
            c_best = c_prop
            
        if c_prop <= c_cur or random.random() < np.exp(-(c_prop - c_cur) / T):
            cur = h_prop
            c_cur = c_prop
            
    return best, c_best

def hill_climb(initial_h, time_budget):
    """Simple hill climbing with constraint preserving moves"""
    start = time.time()
    cur = initial_h.copy()
    c_cur = c5_of(cur)

        c_prop = c5_of(h_prop)
        
        if c_prop < c_cur:
            cur = h_prop
            c_cur = c_prop
    
    return cur, c_cur
