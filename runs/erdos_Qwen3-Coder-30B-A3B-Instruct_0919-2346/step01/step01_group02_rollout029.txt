import numpy as np
import time
import cvxpy as cp
from cvxpy import ECOS, GLPK
from scipy.optimize import linprog
from math import ceil

def project_sum(h, target_sum):
    """Project h onto the hyperplane sum(h) = target_sum using water-filling."""
    h = np.array(h)
    total = np.sum(h)
    if abs(total - target_sum) < 1e-10:
        return h.copy()
    # Sort descending
    sorted_indices = np.argsort(-h)
    h_sorted = h[sorted_indices]
    # Water filling
    cumsum = 0.0
    for i in range(len(h_sorted)):
        cumsum += h_sorted[i]
        if cumsum >= (target_sum - (total - cumsum)):
            break
    # All elements after i, set to new value
    threshold = (target_sum - (total - cumsum)) / (len(h_sorted) - i)
    if threshold < 0:
        threshold = 0
    h_proj = h.copy()
    for j in range(i+1, len(h_sorted)):
        h_proj[sorted_indices[j]] = 0
    for j in range(i):
        h_proj[sorted_indices[j]] = min(1, h_sorted[j])  # might already be that
    # Adjust one more to get exact sum
    for j in range(len(h_sorted)):
        index = sorted_indices[j]
        if h_proj[index] > 0:
            diff = target_sum - np.sum(h_proj)
            change = min(h_sorted[j] - h_proj[index], diff)
            h_proj[index] += change
            if abs(np.sum(h_proj) - target_sum) < 1e-9:
                break
    return h_proj

def c5_of(h):
    """Compute the maximum overlap integral for the given step function h."""
    dx = 2.0 / len(h)
    c = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(c) * dx
    return c5

def run(seed=42, budget_s=600):
    start = time.time()
    np.random.seed(seed)
    N = 140
    dx = 2.0 / N
    target_c5 = 0.3807
    
    # Build index mapping for pairs (i,s) such that 0 <= i, i+s < N
    pairs = []
    pair_to_idx = {}
    idx = 0
    for s in range(N):
        for i in range(N - s):
            pairs.append((i, s))
            pair_to_idx[(i, s)] = idx
            idx += 1

    # LP model to solve relaxation
    print(f'Building LP with {len(pairs)} product variables...')
    h = cp.Variable(N, name="h")
    y = cp.Variable(len(pairs), name="y")
    alpha = cp.Variable(name="alpha")

    constraints = [h[i] >= 0 for i in range(N)]
    constraints += [h[i] <= 1 for i in range(N)]
    constraints += [cp.sum(h) == N / 2]
    for i, s in pairs:
        p_idx = pair_to_idx[(i, s)]
        constraints += [
            y[p_idx] <= h[i],
            y[p_idx] <= h[i + s],
            y[p_idx] >= h[i] + h[i + s] - 1,
            y[p_idx] >= 0
        ]
    # Apply full set of shifts (from -N+1 to N-1) but only consider shifts s=0,...,N-1 in indexing
    # For each shift s = 0,...,N-1:
    # total shift is 2*N-1 shifts, but for symmetry, only compute >=0; -s includes inverse
    # actually we compute overlap from all shifts: s = 0 to 2N-2 where s corresponds to x+k
    # full correlation uses indices 0 to 2N-2
    # but shift s = 0 corresponds to indices 0 to N-1 in h->h+s then sum elements
    # and skip index out-of-bounds
    # construction:
    # for each shift s from 0 to N-1
    #     consider sum of y[i, s] for i = 0 to N-1-s
    shift_sums = {}
    for s in range(N):
        idxs_s = [pair_to_idx[(i, s)] for i in range(N - s)]
        shift_sums[s] = cp.sum(y[idxs_s])
    # same for negative? no, we count only non-negative s due to symmetry
    # but let's make sure we cover all real cases by tolerance to Correlation full
    for s in range(N):
        constraints += [alpha >= dx * (N / 2 - shift_sums[s])]
    
    objective = cp.Minimize(alpha)
    prob = cp.Problem(objective, constraints)
    print('Solving LP relaxation...')
    try:
        prob.solve(solver=ECOS, verbose=False, max_iters=5000)
    except Exception as e:
        print('Error in LP solve:', e)
        # Fallback if LP fails
        h_val = np.ones(N) * 0.5
        h_val = project_sum(h_val, N/2)
        c5 = c5_of(h_val)
        return (h_val, c5, N)

    print(f"LP solved. alpha = {alpha.value}")
    h_val = h.value
    h_val = np.clip(h_val, 0, 1)
    c5_LP = c5_of(h_val)
    print(f"C5_LP: {c5_LP}")
    if c5_LP <= target_c5:
        # Round to nearest integer
        h_round = np.round(h_val).astype(np.float64)
        h_round = project_sum(h_round, N/2)
        c5_final = c5_of(h_round)
        return (h_round, c5_final, N)

    # Fallback to MILP if needed
    # Warm start with LP solution
    # Unary vector will be either 0 or 1?
    print('Starting MILP...')
    h_milp = cp.Variable(N, integer=True, name="h")
    y_milp = cp.Variable(len(pairs), name="y")
    alpha_milp = cp.Variable(name="alpha")

    # Binary constraint on h
    constraints_milp = [h_milp[i] >= 0 for i in range(N)]
    constraints_milp += [h_milp[i] <= 1 for i in range(N)]
    constraints_milp += [cp.sum(h_milp) == N / 2]
    for i, s in pairs:
        p_idx = pair_to_idx[(i, s)]
        constraints_milp += [
            y_milp[p_idx] <= h_milp[i],
            y_milp[p_idx] <= h_milp[i + s],
            y_milp[p_idx] >= h_milp[i] + h_milp[i + s] - 1,
            y_milp[p_idx] >= 0
        ]
    for s in range(N):
        idxs_s = [pair_to_idx[(i, s)] for i in range(N - s)]
        shift_sum = cp.sum(y_milp[idxs_s])
        constraints_milp += [alpha_milp >= dx * (N / 2 - shift_sum)]

    prob_milp = cp.Problem(cp.Minimize(alpha_milp), constraints_milp)
    time_left = budget_s - (time.time() - start)
    if time_left < 30:
        return (h_round, c5_final, N)
    try:
        # Try with limited time
        prob_milp.solve(solver=cp.CBC, verbose=False, maximumSeconds=time_left - 2)
    except Exception as e:
        print('Error in MILP solve:', e)

    if prob_milp.status == cp.OPTIMAL or prob_milp.status == cp.OPTIMAL_INACCURATE or prob_milp.status == cp.SOLVER_ERROR:
        h_milp_val = h_milp.value
        if h_milp_val is not None:
            h_milp_val = np.array(h_milp_val)
            h_final = np.round(h_milp_val).astype(np.float64)
            h_final = project_sum(h_final, N/2)
            c5_final = c5_of(h_final)
            print(f"MILP final c5 = {c5_final}")
            return (h_final, c5_final, N)
    else:
        # If got here, use LP rounded
        print("No MILP solution found, using rounded LP solution.")
        h_round = np.round(h_val).astype(np.float64)
        h_round = project_sum(h_round, N/2)
        c5_final = c5_of(h_round)
        return (h_round, c5_final, N)

    # Still not good? refine with local search
    def subgradient_step(h_in, step_size=0.01, max_iter=100):
        h = h_in.copy()
        for s in range(N):
            # Functional: C_s = dx * (N/2 - sum of prod) = dx * (N/2 - sum h[i] h[i+s])
            # Gradient is: dC_s/d h[i] = dx * (h[i+s] - 1)
            # gradient direction: -grad = dx * (1 - h[i+s])
            # Compute from full correlation
            corr = np.correlate(h, 1 - h, mode='full')
            max_s = np.argmax(corr)
            s_best = max_s - (N - 1)  # shifts range from -(N-1) to N-1
            # s_best = argmax s s.t. sum h[i] * (1 - h[i+s]) = sum h[i] - sum h[i] h[i+s]
            # so argmin sum h[i] h[i+s]
            if s_best < 0:
                continue
            # Count shift indices: i = 0 to N-1-s
            # Gradient for j: sum over i in [0,N-1-s]
            #   grad[j] = 0 if j not in [s,s+N-1-s]
            #   else, grad[j] = -dx * sum h[i+s]
            # So it's valid only for i such that both i and i+s live in interval [0,N)
            # Take subgradient in direction(opposite)
            # need update h_j +=  step_size * gradient[j]
            grad_h = np.zeros(N)
            for i in range(N - s):
                    grad_h[i] += dx * (h[i+s] - 1)
            # Project step to stay in [0,1], and keep sum = N/2
            delta = step_size * (-grad_h)
            h = h + delta
            h = np.clip(h, 0, 1)
            h = project_sum(h, N / 2)
        return h

    # Local refine
    t = time.time()
    h_local = h_round.copy()
    c5_local = c5_of(h_local)
    while time.time() < t + 0.2 * budget_s:
        h_new = subgradient_step(h_local, step_size=0.02)
        c5_new = c5_of(h_new)
        if c5_new < c5_local:
            h_local = h_new
            c5_local = c5_new
    return (h_local, c5_local, N)
