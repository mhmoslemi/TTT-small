def project_onto_bounds_sum(v, target_sum, lower=0, upper=1):
    """Project vector v onto the set of points with sum=target_sum and bounds [lower,upper]"""
    # Binary search for the optimal lambda
    def objective(lam):
        projected = np.clip(v - lam, lower, upper)
        return np.sum(projected) - target_sum
    
    # Find bounds for lambda
    lo = v.min() - upper
    hi = v.max() - lower
    
    # Binary search
    for _ in range(60):
        mid = (lo + hi) / 2
        if objective(mid) > 0:
            hi = mid
        else:
            lo = mid
    
    lam = (lo + hi) / 2
    projected = np.clip(v - lam, lower, upper)
    return projected

def compute_max_overlap(h):
    """Compute maximum overlap for given h"""
    n = len(h)
    dx = 2.0 / n
    correlation = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(correlation) * dx
    return max_overlap
