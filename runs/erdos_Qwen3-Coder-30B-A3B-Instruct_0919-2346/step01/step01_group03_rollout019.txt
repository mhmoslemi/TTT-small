import numpy as np
import time

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral via full convolution of h with (1-h)
    overlaps = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum of the full correlation vector
    # scaled by dx
    c5 = np.max(overlaps) * dx
    return c5

def full_overlap(h):
    """Compute full correlation vector of h with (1-h)"""
    return np.correlate(h, 1 - h, mode="full")

def project_onto_P(v, total):
    """Project v onto the set {x : 0 <= x_i <= 1, sum x_i = total}"""
    # Find the right lambda such that sum(clamp(v - lambda, 0, 1)) = total
    v = np.asarray(v)
    low, high = 0.0, 1.0
    # Bisection to find lambda
    for _ in range(100):
        lambda_ = (low + high) / 2.0
        x = np.clip(v - lambda_, 0, 1)
        s = np.sum(x)
        if s > total:
            low = lambda_
        else:
            high = lambda_
    x = np.clip(v - lambda_, 0, 1)
    return x

def smooth_gradient(h, beta, F):
    """Compute the gradient of the soft-max surrogate Φ_β(h)"""
    N = len(h)
    # Compute softmax weights
    F_scaled = beta * F
    # Subtract max for numerical stability
    F_scaled -= np.max(F_scaled)
    exp_F = np.exp(F_scaled)
    w = exp_F / np.sum(exp_F)
    
    # Initialize the gradient to ones
    g = np.ones(N)
    
    # Compute the weighted sum of gradients for each shift
    for k in range(-N+1, N):
        wk = w[k+N-1]  # F[k+N-1] corresponds to shift k
        if k >= 0:
            g[:N-k] -= wk * h[k:]
            g[k:] -= wk * h[:N-k]
        else:
            d = -k
            g[d:] -= wk * h[:N-d]
            g[:N-d] -= wk * h[d:]
    
    return g

def projected_gradient_descent(h0, time_budget, seed=None):
    # Set up
    if seed is not None:
        np.random.seed(seed)
    start = time.time()
    N = len(h0)
    total = N / 2.0
    h = h0.copy()
    c_best = c5_of(h)
    best_h = h.copy()
    
    # Parameters
    beta = 2.0
    beta_max = 30.0
    beta_update_interval = 200
    step = 0.5
    gamma_step = 0.9995
    gamma_beta = 1.5
    max_iter = 50000
    tfrac = 0.0
    iteration = 0
    
    while iteration < max_iter and time.time() - start < time_budget:
        tfrac = (time.time() - start) / time_budget if time_budget > 0 else 0.0
        iteration += 1
        
        # Compute overlaps (for gradient and metavariables)
        F = full_overlap(h)
        
        # Compute the smooth gradient
        g = smooth_gradient(h, beta, F)
        
        # Tentative update and projection
        v = h - step * g
        h = project_onto_P(v, total)
        
        # Evaluate true objective
        c_cur = c5_of(h)
        if c_cur < c_best:
            c_best = c_cur
            best_h = h.copy()
        
        # Schedule updates
        step *= gamma_step
        # Occasionally increase β
        if iteration % beta_update_interval == 0 and beta < beta_max:
            beta = min(beta * gamma_beta, beta_max)
    
    return best_h, c_best

def targeted_refinement(h, time_budget, seed=None):
    if seed is not None:
        np.random.seed(seed)
    start = time.time()
    h = h.copy()
    c0 = c5_of(h)
    c_best = c0
    best_h = h.copy()
    
    # Start tuning alpha
    alpha = 1.0
    alpha_decay = 0.5
    min_alpha = 1e-6
    patience = 0
    max_patience = 200
    
    while time.time() - start < time_budget:
        # Get overlaps
        C = full_overlap(h)
        # Find maximum (shift with index k=0 means shift 0)
        max_idx = np.argmax(C)
        shift = max_idx - (len(h) - 1)
        
        # Extract overlapping slices
        if shift >= 0:
            hi = h[:len(h) - shift]
            hj = h[shift:]
        else:
            hi = h[-shift:]
            hj = h[:len(h) + shift]
        
        # Compute contribution (element-wise)
        contrib = hi * (1 - hj)
        # Find worst contributor (max)
        idx = np.argmax(contrib)
        i, j = idx, idx + shift
        if shift < 0:
            j = idx - shift
        i = max(0, min(i, len(h) - 1))
        j = max(0, min(j, len(h) - 1))
        
        # Determine feasibility and transfer
        c_ij = contrib[idx]
        delta = min(alpha * c_ij, h[i], 1.0 - h[j])
        
        if delta > 0:
            # Apply transfer
            h[i] -= delta
            h[j] += delta
            
            # Save new C5
            c_new = c5_of(h)
            if c_new < c_best:
                c_best = c_new
                best_h = h.copy()
                patience = 0
            else:
                # Revert
                h[i] += delta
                h[j] -= delta
                patience += 1
        else:
            patience += 1
            
        if patience >= max_patience:
            # Reset alpha
            if alpha > min_alpha:
                alpha *= alpha_decay
                patience = 0
            else:
                break
    
    return best_h, c_best

def run(seed=42, budget_s=600, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    start = time.time()
    N = 200
    dx = 2.0 / N
    total = N / 2
    
    # Initialize seeds
    # 1. Uniform
    h_uniform = np.full(N, 0.5)
    
    # 2. Initial construct
    h_interp = None
    if 'initial_h_values' in globals():
        # Interpolate to 200 points
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_new = np.linspace(0, 2, N)
        h_interp = np.interp(x_new, x_orig, h_orig)
        h_interp *= total / np.sum(h_interp)
    
    # 3. Double-peak
    h_double_peak = np.full(N, 0.1)
    a = 0.3
    idxs = np.where((np.linspace(0, 2, N) < a) | (np.linspace(0, 2, N) > (2 - a)))[0]
    h_double_peak[idxs] = 0.9
    h_double_peak *= total / np.sum(h_double_peak)
    
    # 4. Random Dirichlet
    h_random = np.random.dirichlet([0.5]*N)
    h_random *= total / np.sum(h_random)
    
    # Evaluate seeds
    seeds = [
        ("uniform", h_uniform),
        ("interpolated", h_interp if h_interp is not None else h_uniform),
        ("double_peak", h_double_peak),
        ("random", h_random)
    ]
    
    best_h = None
    best_c5 = float('inf')
    
    # Divide time budget
    time_per_seed = 0.3 * budget_s / len(seeds)
    
    # Run smooth optimization on each seed
    for i, (name, initial_h) in enumerate(seeds):
        if time.time() - start > budget_s * 0.95:
            break
        # Skip none for interpolation
        if initial_h is None:
            continue
        h, c5 = projected_gradient_descent(initial_h, time_per_seed, seed + i)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    
    # Now refine with targeted approach
    if best_h is not None and time.time() - start < budget_s * 0.95:
        refine_time = budget_s - (time.time() - start)
        h_refined, c5_refined = targeted_refinement(best_h, refine_time, seed * 3)
        if c5_refined < best_c5:
            best_c5 = c5_refined
            best_h = h_refined
    
    # Final check
    if best_h is None:
        #Fallback uniformly distributed
        best_h = np.full(N, 0.5)
        best_c5 = c5_of(best_h)
    
    return (best_h, best_c5, N)
