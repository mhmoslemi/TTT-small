import numpy as np
from scipy.optimize import minimize
from math import pi
import time

def build_h_from_coeffs(a, n):
    """Build h from Fourier coefficients"""
    h = 0.5 * np.ones(n)
    M = len(a)
    for m in range(1, M + 1):
        h += a[m - 1] * np.cos(2 * pi * m * np.arange(n) / n)
    return h

def compute_overlap_and_gradient(a, n, dx, gamma=1.0):
    """Compute soft-max overlap and its gradient w.r.t. coefficients"""
    h = build_h_from_coeffs(a, n)
    h = np.clip(h, 0, 1)

    # Compute all overlaps via correlation
    c = np.correlate(h, 1 - h, mode='full')
    f = np.zeros(2 * n - 1)
    for s in range(2 * n - 1):
        f[s] = dx * c[s]
    f = f[n-1:]  # take positive shifts only, f[0] = f_0, f[1] = f_1, ..., f[n-1] = f_{n-1}

    # Softmax weights
    exp_gamma_f = np.exp(gamma * f)
    w = exp_gamma_f / np.sum(exp_gamma_f)

    # Compute gradient w.r.t. a
    grad = np.zeros_like(a)
    for m in range(1, len(a) + 1):
        cos_m = np.cos(2 * pi * m * np.arange(n) / n)
        # First sum for d/d a_m of h_i * (1 - h_{i+s})
        sum1 = 0
        for s in range(n):
            sum1 += cos_m[s] * (1 - h[(s + 1) % n])
        # Second sum for d/d a_m of h_{i+s} * h_i
        sum2 = 0
        for s in range(n):
            sum2 += h[s] * cos_m[(s + 1) % n]
        grad[m - 1] = dx * (sum1 - sum2)

    # Weighted sum of gradients
    g = np.zeros_like(a)
    for m, w_s in enumerate(w):
        g += w_s * grad[m]

    # Add penalty gradient on h - clip(h,0,1)
    h_clipped = np.clip(h, 0, 1)
    diff = h - h_clipped
    penalty_grad = 2e4 * np.sum(diff[:, None] * cos_m[None, :], axis=0)
    g += penalty_grad

    # Softmax objective with penalty
    soft_max_value = np.log(np.sum(exp_gamma_f)) / gamma
    objective = soft_max_value + 1e4 * np.sum(np.maximum(0, h - 1)**2) + 1e4 * np.sum(np.maximum(0, -h)**2)

    return objective, g

def project_to_simplex_box(h, n, target_sum):
    """Project h to the simplex {sum(h) = target_sum} with box constraints [0,1]^n"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if np.isclose(current_sum, target_sum):
        return h.copy()
    if current_sum > target_sum:
        def objective(tau):
            h_proj = np.clip(h - tau, 0, 1)
            return np.sum(h_proj) - target_sum
        # Binary search for tau
        low, high = 0, np.max(h)
        tol = 1e-8
        while high - low > tol:
            mid = (low + high) / 2
            if objective(mid) < 0:
                high = mid
            else:
                low = mid
        tau = (low + high) / 2
        h_proj = np.clip(h - tau, 0, 1)
        return h_proj
    else:
        # Need to increase h; we add to smallest values (to stay within box)
        # Actually, since h already clips to [0,1], only possible if all entries are 1.0
        # Or we can just use a different rescaling algorithm.
        # Water filling approach:
        # h - tau
        # Need average over 0 to 1
        # No, simpler and safe is just add area adjustment:
        adjustment = (target_sum - current_sum) / n
        h = np.clip(h + adjustment, 0, 1)
        return h

def spectral_optim(n, max_iters, gamma_schedule, m_mode, warmstart_coeffs=None):
    """Run spectral optimisation phase"""
    a_init = np.random.normal(0, 0.05, m_mode)
    if warmstart_coeffs is not None:
        for i in range(min(len(warmstart_coeffs), m_mode)):
            a_init[i] = warmstart_coeffs[i]

    bounds = [(-0.6, 0.6)] * m_mode
    dx = 2.0 / n

    # Iteration counter
    iter_count = 0

    while iter_count < max_iters:
        # Get next gamma value
        if isinstance(gamma_schedule, (list, tuple)):
            gamma = gamma_schedule[min(iter_count // 500, len(gamma_schedule) - 1)]
        else:
            gamma = gamma_schedule

        def f_and_g(a_vec):
            return compute_overlap_and_gradient(a_vec, n, dx, gamma)

        # Run one step of L-BFGS-B with bounds
        result = minimize(f_and_g, a_init, method='L-BFGS-B', bounds=bounds, jac=True)
        if result.success:
            a_init = result.x
        else:
            break

        iter_count += 1
        if iter_count % 100 == 0:
            h = build_h_from_coeffs(a_init, n)
            h = np.clip(h, 0, 1)
            h_proj = project_to_simplex_box(h, n, n / 2)
            c5 = evaluate_erdos_solution(h_proj)
            print(f"  iter {iter_count}: C5 = {c5:.6f}")

    return a_init

def polish_with_projected_gradient(h0, n, dx, max_iters=100):
    """Project gradient descent on h variables"""
    h = h0.copy()
    # reduce h to [0,1] and normalize mean
    h = np.clip(h, 0, 1)
    h *= (n / 2) / np.sum(h)
    # Project to np.sum(h) = n/2 while maintaining [0,1]
    for _ in range(max_iters):
        # gradient on h (simplified version):
        c = np.correlate(h, 1 - h, mode='full')
        overlap = np.array([c[n-1+i] for i in range(n)])
        # non-smooth approx to the gradient by perturbing h_i step by step and computing change in overlap
        # this is just an approximate direction to decrease overlap
        grad = np.zeros_like(h)
        for i in range(n):
            h_plus = h.copy()
            h_minus = h.copy()
            # Perturb h[i] by small value
            eps = 1e-4
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n/2) / np.sum(h_plus)

            h_minus[i] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n/2) / np.sum(h_minus)

            # Approximate grad
            grad[i] = (evaluate_erdos_solution(h_plus) - evaluate_erdos_solution(h_minus)) / (2 * eps)

        # Update h in direction of negative gradient
        h -= 1e-3 * grad
        h = np.clip(h, 0, 1)
        h = h * (n/2) / np.sum(h)

    return h

def evaluate_erdos_solution(h):
    """Compute C_5"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlations full
    correlation = np.correlate(h, 1 - h, mode='full')
    # Max value scaled by dx
    max_overlap = np.max(correlation) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    n = 240  # number of sample points (even for parsing)
    dx = 2.0 / n
    target_sum = n / 2

    best_c5 = np.inf
    best_h = None

    restarts = 6
    iter_used = 0

    max_iter_per_restart = 10000
    gamma_schedule = [0.5] + [0.5 * (1.5)**i for i in range(30)]
    # Livermore-like multi-stage schedule:
    stage1_max_iter = 1500
    stage2_max_iter = 2000
    stage3_max_iter = 2500

    m_stage1 = 8
    m_stage2 = 15
    m_stage3 = 24

    for r in range(restarts):
        # timestamp
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.99:
            break
        # Get a good starting signal from seed
        a_coeff = np.random.normal(0, 0.05, m_stage1)

        a = spectral_optim(n, stage1_max_iter, gamma_schedule, m_stage1, a_coeff)
        a = np.pad(a, (0, m_stage2 - m_stage1), 'constant')
        a = spectral_optim(n, stage2_max_iter, gamma_schedule, m_stage2, a)
        a = np.pad(a, (0, m_stage3 - m_stage2), 'constant')
        a = spectral_optim(n, stage3_max_iter, gamma_schedule, m_stage3, a)

        h = build_h_from_coeffs(a, n)
        h = np.clip(h, 0, 1)
        h_proj = project_to_simplex_box(h, n, target_sum)
        final_h = polish_with_projected_gradient(h_proj, n, dx)

        c5 = evaluate_erdos_solution(final_h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = final_h.copy()

        iter_used += stage1_max_iter + stage2_max_iter + stage3_max_iter

        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.97:
            break

    if best_h is None:
        # fallback to simple construction
        best_h = np.ones(n) * 0.5
        dx = 2.0 / n
        best_h = best_h * (1.0 / (np.sum(best_h) * dx))
        best_c5 = evaluate_erdos_solution(best_h)

    return (best_h.tolist(), best_c5, n)
