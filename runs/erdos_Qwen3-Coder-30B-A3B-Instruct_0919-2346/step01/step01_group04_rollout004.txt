import numpy as np
import time
import random
from scipy.optimize import differential_evolution, minimize


def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    overlap = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(overlap) * dx
    return c5


def project_to_feasible(h):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    
    # Scale to correct sum
    dx = 2.0 / len(h)
    current_sum = np.sum(h) * dx
    target_sum = 1.0
    
    if abs(current_sum) > 1e-10:
        h = h * target_sum / current_sum
    return h


def initial_construction(n_points):
    """Create a better initial construction"""
    h = np.full(n_points, 0.5)
    
    # Create a smooth, optimized shape
    x = np.linspace(0, 2, n_points)
    
    # Try to define an optimal piecewise shape
    # Use a trapezoid-like shape to minimize overlap
    a = 0.25
    b = 1.75
    midpoint = 1.0
    
    # Define three segments
    # Segment 1: x from 0 to a  
    # Segment 2: x from a to b  
    # Segment 3: x from b to 2
    
    for i in range(n_points):
        x_pos = x[i]
        
        if x_pos < a:
            # Linear increase
            h[i] = (x_pos / a) * 0.8
        elif x_pos < b:
            h[i] = 0.8
        else:
            # Linear decrease
            h[i] = 0.8 * (2 - x_pos) / (2 - b)
    
    # Ensure sum constraint (integrate over [0,2])
    h = project_to_feasible(h)
    
    return h


def optimization_step(h, temperature):
    """Make a local optimization move"""
    h_new = h.copy()
    n_points = len(h)
    
    # Choose two random indices
    i = random.randint(0, n_points - 1)
    j = random.randint(0, n_points - 1)
    if i == j:
        return h
    
    # Determine feasible change for both variables
    delta_i = random.uniform(-0.1, 0.1)
    delta_j = -delta_i
    
    # Check feasibility within bounds [0,1]
    if (h_new[i] + delta_i >= 0) and (h_new[i] + delta_i <= 1) and \
       (h_new[j] + delta_j >= 0) and (h_new[j] + delta_j <= 1):
        h_new[i] += delta_i
        h_new[j] += delta_j
    
    return h_new


def stochastic_local_search(h_guess, steps, time_limit):
    """Stochastic local search with simulated annealing flavor"""
    start_time = time.time()
    
    current = h_guess.copy()
    current_c5 = c5_of(current)
    
    best = current.copy()
    best_c5 = current_c5
    
    # Cooling schedule
    T_start = 0.1
    T_end = 1e-6
    for step in range(steps):
        if time.time() - start_time > time_limit:
            break
            
        elapsed_ratio = (time.time() - start_time) / time_limit
        T = T_start * (T_end / T_start) ** elapsed_ratio
        
        # Make move
        proposed = optimization_step(current, T)
        proposed_c5 = c5_of(proposed)
        
        # Accept or reject based on SA criteria
        if proposed_c5 < current_c5 or random.random() < np.exp(-(proposed_c5 - current_c5) / T):
            current = proposed
            current_c5 = proposed_c5
            
        if current_c5 < best_c5:
            best = current.copy()
            best_c5 = current_c5
    
    return best, best_c5


def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function"""
    random.seed(seed)
    np.random.seed(seed)
    
    start = time.time()
    n_points = 120  # Use fewer points for speed
    dx = 2.0 / n_points
    
    # Create better initial construction
    initial_h = initial_construction(n_points)
    
    # Run multiple local search instances with different initializations
    best_h = None
    best_c5 = float('inf')
    
    # Use different initialization seeds for variance
    for seed_i in [seed, seed+100, seed+200, seed+300]:
        random.seed(seed_i)
        np.random.seed(seed_i)
        
        # Perturb initial
        h_seed = initial_h.copy() + np.random.normal(0, 0.01, n_points)
        h_seed = project_to_feasible(h_seed)
        
        # Run local search
        remaining_time = budget_s - (time.time() - start)
        if remaining_time < 5:
            break
            
        h_result, c5_result = stochastic_local_search(
            h_seed, 
            steps=10000, 
            time_limit=remaining_time*0.3
        )
        
        if c5_result < best_c5:
            best_c5 = c5_result
            best_h = h_result.copy()
    
    # Final polishing with optimization method
    if best_h is not None:
        remaining_time = budget_s - (time.time() - start)
        if remaining_time > 10:
            # Try scipy.optimize approach for last refinement
            try:
                # Create a wrapper for scipy
                def objective(params):
                    h_test = project_to_feasible(params)
                    return c5_of(h_test)
                
                # Create bounds
                bounds = [(0, 1) for _ in range(n_points)]
                
                # Use differential evolution on a small subset for efficiency
                # but avoid sampling too many points in high dimension 
                subset_size = min(n_points, 60)
                indices = np.random.choice(n_points, subset_size, replace=False)
                
                # Rearrange to only optimize subset
                initial_subset = best_h[indices]
                bounds_subset = [bounds[i] for i in indices]
                
                result = differential_evolution(
                    objective,
                    bounds_subset,
                    maxiter=50,
                    popsize=10,
                    seed=seed+999
                )
                
                if result.success:
                    h_final = best_h.copy()  # Start with current
                    h_final[indices] = result.x   # Update subset
                    
                    h_final = project_to_feasible(h_final)
                    c5_final = c5_of(h_final)
                    
                    if c5_final < best_c5:
                        best_h = h_final
                        best_c5 = c5_final
                        
            except:
                pass  # Skip optimization if not successful
                
    # Final validation
    if best_h is not None:
        best_h = project_to_feasible(best_h)
        best_c5 = c5_of(best_h)
        
    return (best_h, best_c5, n_points)
