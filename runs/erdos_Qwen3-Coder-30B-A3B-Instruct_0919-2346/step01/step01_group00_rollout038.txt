import numpy as np
import time
import random
from scipy.optimize import differential_evolution, minimize

def c5_of(h):
    """Compute C5 = max_k integral h(x)(1-h(x+k)) dx."""
    dx = 2.0 / len(h)
    # Compute correlation: h * (1 - h) with full mode
    c = np.correlate(h, 1 - h, mode="full")
    # Multiply by dx to get continuous integral
    c *= dx
    # Return maximum value among all shifts
    return np.max(c)

def random_perturb(h, perturb_factor=0.05):
    """Create a new feasible h by perturbing values."""
    h_new = h.copy()
    n = len(h)
    
    # Select random subset of indices to modify
    n_change = max(1, int(n * 0.1))
    idxs = random.sample(range(n), n_change)
    
    for i in idxs:
        # Determine valid range for change
        orig = h_new[i]
        low = max(0, orig - perturb_factor)
        high = min(1, orig + perturb_factor)
        
        # Sample new value
        if low < high:
            h_new[i] = random.uniform(low, high)
    
    # Renormalize to maintain constraint
    h_new = h_new * (len(h_new) / 2) / np.sum(h_new)
    
    return h_new

    bounds = [(0, 1) for _ in range(n_points)]
