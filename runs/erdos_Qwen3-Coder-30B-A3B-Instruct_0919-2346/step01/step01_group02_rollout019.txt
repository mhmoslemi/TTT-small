import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def project_sum(h, target_sum):
    """Project h onto the sum = target_sum using water-filling."""
    h = np.array(h)
    h = np.clip(h, 0, 1)
    if np.sum(h) == target_sum:
        return h
    # Use water-filling to project onto sum constraint
    h_sorted = np.sort(h)[::-1]
    cumsum = np.cumsum(h_sorted)
    n = len(h_sorted)
    i = 0
    while i < n and cumsum[i] < target_sum:
        i += 1
    if i == 0:
        # All elements need to be scaled down
        scale = target_sum / np.sum(h)
        return h * scale
    else:
        # i-1 elements are at full capacity (1.0), others have to be scaled
        # the average of remaining elements (from i to end) should be (target_sum - (i-1)) / (n-i)
        avg = (target_sum - (i-1)) / (n-i)
        if avg < 0:
            avg = 0.0
        ret = np.zeros_like(h)
        ret[:i] = 1.0
        ret[i:] = avg
        # now we need to permute to match original indices
        indices = np.argsort(h)[::-1]
        out = np.empty_like(ret)
        out[indices] = ret
        return out

def c5_of(h):
    dx = 2.0 / len(h)
    c = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(c) * dx
    return c5

def solve_lp_relaxation(N):
    """Solve the LP relaxation using cvxpy"""
    # Build list of valid pairs (i,s)
    pairs = []
    # We'll consider all shifts s = -(N-1)...(N-1)
    # But we're only interested in the known non-negative ones to have the complete
    # set of correlations, and they are symmetric
    for s in range(N):
        for i in range(N - s):
            pairs.append((i, s))
    
    # Create variables
    h = cp.Variable(N, name='h')
    y = cp.Variable(len(pairs), name='y')
    alpha = cp.Variable(name='alpha')

    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == N / 2.0,
    ]

    # McCormick envelope constraints for each y[i,s]
    idx = 0
    for i, s in pairs:
        (i1, i2) = i, i + s
        constraints.extend([
            y[idx] <= h[i1],
            y[idx] <= h[i2],
            y[idx] >= h[i1] + h[i2] - 1,
            y[idx] >= 0
        ])
        idx += 1

    # Max overlap definition constraints
    shift_idx_map = {}
    for s in range(N):
        idxs_s = [(i, s) for i in range(N - s)]
        shift_idx_map[s] = idxs_s

    # Compute sum of y over each shift
    dx = 2.0 / N
    for s in range(N):
        idxs = shift_idx_map[s]
        # Get indices in y vector
        y_indices_s = [i for (i, pair_s) in enumerate(pairs) if pair_s in idxs]
        sum_y_s = cp.sum(y[y_indices_s])
        # Constraint: alpha >= dx * (N/2 - sum_y_s)
        # We rewrite as: alpha >= dx * N/2 - dx * sum_y_s
        # i.e., dx * sum_y_s >= dx * N/2 - alpha  
        # but we write in standard form:
        constraints.append(alpha >= dx * (N/2 - sum_y_s))

    # Objective
    objective = cp.Minimize(alpha)
    prob = cp.Problem(objective, constraints)
    
    start_time = time.time()
    try:
        # Use ECOS for fast LP solution
        prob.solve(solver=cp.ECOS, max_iters=20000, verbose=False)
        elapsed = time.time() - start_time
        h_val = h.value
        alpha_val = alpha.value
        if h_val is not None:
            return h_val, alpha_val, pairs
        else:
            return None, None, None
    except Exception as e:
        # print("LP solve failed:", e)
        return None, None, None

        # Instead, compute gradient for best shift s directly:
        # partial derivative of C_s w.r.t h[i] = 1 - 2 h[i + s] *(i<s when shift is positive) 
        # total derivative = (1 - 2 h[i + s]) + (1 - 2 h[i - s]) = 2 - 2 (h[i+s] + h[i-s])
        
        grad_direction = np.zeros(N)

                grad_direction[i] += -2 * h[i + shift]
            if i - shift >= 0 and i - shift < N:
                grad_direction[i] += -2 * h[i - shift]
            grad_direction[i] += 1  # +1 from C_i = h_i - h_i h_{i+s}

        # The exact gradient 
        # Let's go with a more pragmatic way:
        # We compute overlaps for shifted h_i to see direction of change
        # Allocate extra array to simulate moves
        step_scaled = step_size
        offset = np.zeros(N)
        for i in range(N):
            offset += step_scaled * grad_direction[i] * h[i]  # Estimate effect of shift
        
        # Try a small gradient step in the direction that reduces the objective
        h_candidate = h - step_scaled * grad_direction
        h_candidate = np.clip(h_candidate, 0, 1)
        # Re-project to fit the sum constraint
        h_candidate = project_sum(h_candidate, N / 2.0)
        new_c5 = c5_of(h_candidate)
        if new_c5 < old_c5:
            h = h_candidate
            old_c5 = new_c5
    return h, old_c5
