import numpy as np
import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value scaled by dx
    max_overlap = np.max(correlation) * dx
    return max_overlap

def generate_sturmian_binary(n):
    """Generate a balanced binary sequence using irrational rotation (Sturmian word)"""
    phi = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
    h = np.zeros(n, dtype=int)
    ones_count = 0
    for i in range(n):
        t = (i * phi) % 1
        if t < 0.5:
            h[i] = 1
            ones_count += 1
    # Adjust to ensure exactly n//2 ones
    if ones_count > n // 2:
        # Find the last 1 and turn it to 0
        for i in range(n-1, -1, -1):
            if h[i] == 1:
                h[i] = 0
                break
    elif ones_count < n // 2:
        # Find the first 0 and turn it to 1
        for i in range(n):
            if h[i] == 0:
                h[i] = 1
                break
    return h

def local_swap_search(h, n_points, time_limit):
    """Perform local swap search to reduce maximum overlap"""
    start_time = time.time()
    dx = 2.0 / n_points
    max_shifts = np.arange(1, n_points)  # valid non-zero shifts
    
    # Initialize shifted arrays efficiently
    def compute_g_s(h, s):
        # h[i] * (1 - h[i + s]) for i in range(n_points - s)
        if s == 0:
            return 0
        return np.sum(h[:n_points-s] * (1 - h[s:]))
    
    # Precompute initial g_s values
    g_s = np.zeros(2 * n_points - 1)
    for s in range(2 * n_points - 1):
        g_s[s] = compute_g_s(h, s)
    
    current_max = np.max(g_s)
    best_h = h.copy()
    best_max = current_max
    
    # Tabu list for recently swapped pairs
    tabu_list = set()
    tabu_duration = 100
    
    iteration = 0
    while time.time() - start_time < time_limit:
        iteration += 1
        # Find worst shift
        s_max = np.argmax(g_s)
        max_g = g_s[s_max]
        
        # Identify positions that contribute to this shift
        # For shift s, h[i] * (1 - h[i+s]) contributes to g_s
        # Find 1s at positions where h[i+s] = 0, and 0s where h[i+s] = 1
        contributions = []
        if s_max < n_points:
            for i in range(n_points - s_max):
                if h[i] == 1 and h[i + s_max] == 0:
                    contributions.append(i)
        else:
            s_inv = 2 * n_points - 2 - s_max
            for i in range(s_inv):
                if h[i] == 1 and h[i + s_inv] == 0:
                    contributions.append(i)
        
        if len(contributions) == 0:
            # No contribution detected, try a random swap
            # Find 1s and 0s
            ones_positions = np.where(h == 1)[0]
            zeros_positions = np.where(h == 0)[0]
            if len(ones_positions) > 0 and len(zeros_positions) > 0:
                i = np.random.choice(ones_positions)
                j = np.random.choice(zeros_positions)
                # Make sure this swap might be helpful
                if (i, j) not in tabu_list:
                    # Test this swap
                    # Directly compute the difference
                    delta = np.zeros_like(g_s)
                    # For each shift s where i or j contributes
                    for s in range(2 * n_points - 1):
                        if s < n_points - i:
                            if s >= j and j < n_points:
                                delta[s] += -(1 - h[i+s]) + (h[i-s] if i-s >= 0 else 0) + (1 - h[j+s]) - (h[j-s] if j-s >= 0 else 0)
                            elif i+s < n_points and s >= 0 and i+s >= 0:
                                # Both terms depend on i and j
                                delta[s] += -(1 - h[i+s]) + (h[i-s] if i-s >= 0 else 0) + (1 - h[j+s]) - (h[j-s] if j-s >= 0 else 0)
                        else:
                            pass  # others
                    new_g = g_s + delta
                    new_max = np.max(new_g)
                    if new_max < current_max:
                        h[i], h[j] = h[j], h[i]
                        g_s = new_g
                        current_max = new_max
                        # Update tabu list
                        tabu_list.add((i, j))
                        if len(tabu_list) > 1000:
                            # Remove some old entries
                            tabu_list = set(list(tabu_list)[:800])
                    else:
                        # Accept with probability
                        p = min(1.0, np.exp(-10.0 * (new_max - current_max)))
                        if np.random.rand() < p:
                            h[i], h[j] = h[j], h[i]
                            g_s = new_g
                            current_max = new_max
                            tabu_list.add((i, j))
                            if len(tabu_list) > 1000:
                                tabu_list = set(list(tabu_list)[:800])
        else:
            # Pick one of the positions contributing to worst
            i = np.random.choice(contributions)
            # Try to find good flip: 0 that could become 1 nearby i
            # Find 0 positions where the shift brings a 1 at that position
            zeros = np.where(h == 0)[0]
            zeros_good = []
            for j in zeros:
                if i + s_max < n_points and j < n_points and j + s_max < n_points:
                    if h[j] == 0 and h[j + s_max] == 1:
                        zeros_good.append(j)
                elif s_max >= n_points and j + (2*n_points - 2 - s_max) < n_points:
                    if h[j] == 0 and h[j + (2*n_points - 2 - s_max)] == 1:
                        zeros_good.append(j)
            if zeros_good:
                # Pick one at random
                j = np.random.choice(zeros_good)
                delta = np.zeros_like(g_s)
                # Compute change in g_s for all shifts
                for s in range(2 * n_points - 1):
                    if i < n_points:
                        if i+s < n_points:
                            delta[s] -= (1 - h[i+s])  # from 1 becoming 0
                        if i-s >= 0:
                            delta[s] += (h[i-s])    # from i-s contributing to next
                    if j < n_points:
                        if j+s < n_points:
                            delta[s] += (1 - h[j+s])  # from 0 becoming 1
                        if j-s >= 0:
                            delta[s] -= (h[j-s])   # from j-s transferring contribution
                new_g = g_s + delta
                new_max = np.max(new_g)
                if new_max < current_max:
                    h[i], h[j] = h[j], h[i]
                    g_s = new_g
                    current_max = new_max
                    tabu_list.add((i, j))
                elif np.random.rand() < 0.1:
                    # Accept with low probability
                    h[i], h[j] = h[j], h[i]
                    g_s = new_g
                    current_max = new_max
                    tabu_list.add((i, j))
                if len(tabu_list) > 1000:
                    tabu_list = set(list(tabu_list)[:800])
        
        # Update global best
        if current_max < best_max:
            best_h = h.copy()
            best_max = current_max
        
        # Occasionally shuffle
        if iteration % 100 == 0:
            np.random.shuffle(h)
    
    return best_h, best_max

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Step 1: Initialization
    n_points = 120
    h = generate_sturmian_binary(n_points).astype(float)
    
    # Ensure exactly half ones
    current_sum = np.sum(h)
    while not np.isclose(current_sum, n_points / 2.0, rtol=1e-5):
        # Adjust balance
        if current_sum > n_points / 2.0:
            # Turn one 1 to 0
            ones = np.where(h == 1)[0]
            if len(ones) > 0:
                idx = ones[0]
                h[idx] = 0
        else:
            # Turn one 0 to 1
            zeros = np.where(h == 0)[0]
            if len(zeros) > 0:
                idx = zeros[0]
                h[idx] = 1
        current_sum = np.sum(h)
    
    # Normalize to dx = 2 / n
    dx = 2.0 / n_points
    h = h
    
    # Step 2: Local swap search
    time_left = budget_s - (time.time() - start_time)
    if time_left > 10:
        h, c5_bound = local_swap_search(h, n_points, time_left * 0.9)
        if (time.time() - start_time) > 0.8 * budget_s:
            return (h.tolist(), c5_bound, len(h))
    
    # Step 3: Refinement by upsampling
    n_points_up = 240
    h_up = np.repeat(h, 2)
    # Adjust to ensure sum = n_points_up / 2
    actual_sum = np.sum(h_up)
    if not np.isclose(actual_sum, n_points_up / 2.0, rtol=1e-5):
        # slight adjustment
        diff = int(n_points_up / 2 - actual_sum)
        if diff > 0:
            # Change some 0 to 1
            zeros = np.where(h_up == 0)[0]
            if len(zeros) >= diff:
                for i in range(diff):
                    h_up[zeros[i]] = 1
            else:
                # Issue
                pass
        else:
            # Change some 1 to 0
            ones = np.where(h_up == 1)[0]
            if len(ones) >= -diff:
                for i in range(-diff):
                    h_up[ones[i]] = 0
            else:
                # Issue
                pass
    
    # Local search with upscaled version
    time_left = budget_s - (time.time() - start_time)
    if time_left > 20:
        h_up, c5_bound_up = local_swap_search(h_up, n_points_up, time_left * 0.8)
        h = h_up
        c5_bound = c5_bound_up
        if (time.time() - start_time) > 0.9 * budget_s:
            return (h.tolist(), c5_bound, len(h))
    
    # Step 4: Final polishing (with continuous variables near optimal)
    # Use scipy.optimize with SLSQP
    def objective(h_vec):
        h = np.array(h_vec)
        return evaluate_erdos_solution(h)
    
    bounds = [(0, 1) for _ in range(len(h))]
    
    def constraint_fun(h_vec):
        h = np.array(h_vec)
        return np.sum(h) - len(h) / 2.0
    
    constraint = {'type': 'eq', 'fun': constraint_fun}
    
    options = {'maxiter': 200}
    
    time_left = budget_s - (time.time() - start_time)
    if time_left > 5:
        try:
            result = minimize(objective, h, method='SLSQP', 
                            bounds=bounds, constraints=[constraint], 
                            options=options, tol=1e-6)
            if result.success:
                h = result.x
                c5_bound = result.fun
        except Exception:
            pass
    
    return (h.tolist(), c5_bound, len(h))
