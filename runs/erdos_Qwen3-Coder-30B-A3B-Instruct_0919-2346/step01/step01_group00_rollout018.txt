import numpy as np
import time

def c5_of(h):
    """Compute C5 = max_k integral h(x)(1-h(x+k)) dx."""
    dx = 2.0 / len(h)
    # Compute correlation: h * (1 - h) with full mode
    c = np.correlate(h, 1 - h, mode="full")
    # Multiply by dx to get continuous integral
    c *= dx
    # Return maximum value among all shifts
    return np.max(c)

def project_feasible(h):
    """Project h to feasible region: [0,1] and sum = n/2"""
    h = np.clip(h, 0, 1)
    dx = 2.0 / len(h)
    scale = (len(h) / 2) / np.sum(h)
    h = h * scale
    return h

def propose_move(h, step_size):
    """Generate a new feasible h by adjusting two entries."""
    h = h.copy()
    N = len(h)
    i, j = random.sample(range(N), 2)
    
    # Determine feasible interval for delta
    delta_low = max(-h[i], h[j] - 1.0)
    delta_high = min(1.0 - h[i], h[j])
    
    if delta_low >= delta_high:
        return h  # No valid move for this pair
    
    delta = random.uniform(-step_size, step_size)
    delta = max(delta_low, min(delta, delta_high))
    
    h[i] += delta
    h[j] -= delta
    
    return project_feasible(h)

def iterative_local_search(initial_h, time_budget, max_iter=1000):
    """Improved local search with adaptive step"""
    start = time.time()
    cur = project_feasible(initial_h)
    best = cur.copy()
    c_cur = c5_of(cur)
    c_best = c_cur
    
    step_size = 0.1
    patience = 0
    max_patience = 20
    
    for i in range(max_iter):
        if time.time() - start > time_budget:
            break
            
        h_new = propose_move(cur, step_size)
        c_new = c5_of(h_new)
        
        if c_new < c_cur:
            cur = h_new
            c_cur = c_new
            if c_new < c_best:
                best = h_new
                c_best = c_new
                patience = 0
                step_size = min(0.1, step_size * 1.1)
            else:
                patience += 1
                if patience > max_patience:
                    step_size *= 0.9
                    patience = 0
        else:
            # Accept with some probability (but target is minimization)
            if random.random() < 0.1:  # Small chance to accept worse
                cur = h_new
                c_cur = c_new
    
    return best, c_best

def genetic_algorithm(h0, time_budget, pop_size=50, n_generations=100):
    """Simple genetic algorithm approach"""
    start = time.time()
    N = len(h0)
    
    def mutate(h):
        h = h.copy()
        # Add some small random adjustments
        for i in range(N):
            if random.random() < 0.1:
                h[i] = max(0, min(1, h[i] + random.gauss(0, 0.01)))
        h = project_feasible(h)
        return h
    
    # Initialize population
    population = [project_feasible(h0 + random.gauss(0, 0.01) * np.random.random(N)) 
                  for _ in range(pop_size)]
    
    best_h = None
    best_c5 = float('inf')
    
    for gen in range(n_generations):
        if time.time() - start > time_budget:
            break
            
        # Evaluate fitness
        fitness = [c5_of(h) for h in population]
        
        # Find current best
        for h, c5 in zip(population, fitness):
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Selection (tournament)
        new_population = []
        for _ in range(pop_size):
            # Tournament selection
            idxs = random.sample(range(pop_size), 3)
            tournament = [population[i] for i in idxs]
            tournament_fitness = [fitness[i] for i in idxs]
            winner = tournament[np.argmin(tournament_fitness)]
            new_population.append(mutate(winner))
            
        population = new_population
    
    return best_h, best_c5
