def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value scaled by dx
    max_overlap = np.max(correlation) * dx
    return max_overlap

def sturmian_binary_sequence(n):
    """Generate a balanced binary sequence using irrational rotation"""
    alpha = (sqrt(5) - 1) / 2  # Golden ratio conjugate
    h = np.zeros(n, dtype=int)
    positions = np.array([(i * alpha) % 1 for i in range(n)])
    # Assign 1s to lower half of positions, 0s to upper half
    sorted_indices = np.argsort(positions)
    num_ones = n // 2
    h[sorted_indices[:num_ones]] = 1
    return h

def compute_g_s(h):
    """Precompute all g_s values for current h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation = sum h[i] * (1 - h[i+s]) for all shifts
    correlation = np.correlate(h, 1 - h, mode="full")
    # Scale by dx
    g_s = correlation * dx
    return g_s

def compute_delta_s(h, s, i, j):
    """Compute change in g_s when swapping h[i] and h[j]"""
    if i >= len(h) or j >= len(h):
        return 0
    delta = 0
    # For shift s, we compute how overlapping terms change
    # When changing h[i] and h[j]
    # i contributes to g_s where i in [0, n-1] and (i-s) in [0, n-1]
    # d = i - s => i = s + d; d ranges from 0 to n-1, so s+d in [0,n-1]
    # so s ranges from -d to n-1-d
    # Specifically for shift s:
    # At position i, contribution to shift s is term = h[i] * (1 - h[i+s])
    # If h[i]=1 and h[i+s]=0, that contributes 1
    # If h[i]=0 or h[i+s]=1, the contribution is 0
    # When swapped: h[i] changes from 1 to 0, h[j] from 0 to 1 
    # Contribution of i changes: 
    # original: h[i](1-h[i+s]) = 1*(1-h[i+s]) = 1-h[i+s]
    # after swap: 0*(1-h[i+s]) = 0
    # delta = -(1-h[i+s])
    # 
    # Contribution of j changes similarly:
    # original: h[j](1-h[j+s]) 
    # after swap: h[j](1-h[j+s]) = 1*(1-0) = 1 if h[j]=0 previously, but now h[j]=1
    # So: (1-h[j+s]) (note: index shifting is corrected below)
    # Check:
    # If i+s within range [0,n-1], change = -(1-h[i+s])
    # If j+s within range [0,n-1], change = + (1-h[j+s])
    
    # Instead of handling all s directly, compute for the specific s over all affected terms.
    # But it's simpler to just sum over all valid shifts that contribute to the change:
    # Again, following shift s, we have contribution from i to g_s and from j to g_s.
    
    # For shift s, term is contributing h[i] * (1 - h[i + s]) if i + s < n
    # If i + s >= n, that element wraps (periodic extension), which is circular convolution logic.
    # But shifting by s.
    # 
    # The update function of each g_s when we swap positions i and j:
    # For shift t, if i-j contribute to change in t, we arrive at:
    # if t = i - s (i under shift t is original i)
    # then change = - (1 - h[i+s]) (since i goes from 1 to 0)
    # if t = j - s (j under shift t is original j)  
    # then change = + (1 - h[j+s]) (since j goes from 0 to 1)
    # Start with proper indexing
    
    # Only consider shift offset in the valid range relative to i and j
    # But we're updating for shift s in general.
    # General upon swap between i and j:
    # The quantity g_s = sum h_i * (1 - h_{i+s})
    # i contributes to g_s only if 0 <= i + s < n  => -s <= i < n-s where s >= 0
    # j contributes to g_s similarly  
    # Arity says i in [0,n-1], shift s such that i+s in [0,n-1]
    # For i, g_s steps forward by 1
    # For j, g_s steps backward by 1?
    
    # This feels modelled by correlations computation.
    # For fixed pair of indices i and j, we must compute how g_s changes as a function of s.
    # delta_g_s(i,j) =      - (1 - h[i+s]) (if i + s < n, otherwise 0)  
    #                     + (1 - h[j+s]) (if j + s < n, otherwise 0)
    # 
    # But the way we want to compute is:
    # At shift s, number of terms in g_s = n - |s| 
    # if we extend the domain via circular or reflexive boundary conditions.
    # The final formula will be a bit tricky due to non-circularity inconsistent with
    
    # Let's compute the simpler full list:
    change_s = np.zeros(len(h) * 2 - 1)
    
    # We know that each position contributes to multiple shifts (range of shifts)
    # Given a fixed swap (i,j), we compute change in each g_s term

    # Easiest way: focus only on the specific relevant shift, so the problem simplifies.
    # if we do shift s and say that i is relevant and j is relevant, 
    # i contributes to g_s if i is involved (i is i-s then) 
    # meaning i = s + d => d is the overlap index, see below
    # 
    # position i has 1 and becomes 0
    # h[i](1-h[i-s])  
    # becomes 0*(1-h[i-s]) = 0 (loss change of 1 - h[i-s])
    # 
    # position j has 0 and becomes 1
    # h[j](1-h[j-s])  
    # becomes 1*(1-h[j-s]) = (1-h[j-s]) (gain value of 1 - h[j-s])
    
    # For shift s, compute the delta_g_s:
    
    # Special case optimization: 
    # If 0 <= i-s < n and i-s exists (position i-s is valid), 
    # then the effect is:
    # old contribution: h[i] * (1 - h[i-s]) = 1*(1-h[i-s]) = 1-h[i-s]
    # new contribution: 0*(1 - h[i-s]) = 0 
    # hence delta = - (1-h[i-s])
    
    # If 0 <= j-s < n, i.e., j-s is valid:
    # old contribution: h[j] * (1 - h[j-s]) = 0 *(1 - h[j-s]) = 0
    # new contribution: 1*(1-h[j-s]) = (1-h[j-s])
    # hence delta = + (1-h[j-s])
    
    # So for shift s:
    # if i-s < 0 or i-s >= n: no contribution from i
    # if j-s < 0 or j-s >= n: no contribution from j
    
    # But g_s involves Σ h_i (1 - h_{i+s}) which wraps around circular arrays  
    # or avoids wraparound in numpy.correlate (it's full mode which is indeed reflexive)
    # This implies that if h has size n, the correlation h * (1-h) with full mode has size 2n-1.  
    
    # We seek: when we swap h[i] and h[j], how does this change g_s for each shift s?
    # But this is not a single vector but a vector indexed by shift. So we need for each shift s:
    
    # As a first pass:
    # Let's not worry about small update errors, since we will 
    # approximate the full new g_s array by recomputing for loop and the worst shift
    # This is computational anyway, just ensure it's up to date.
    
    # To make it simple:
    # The effect of swapping a 1@i and 0@j changes the shift contribution g_s:
    # ONLY when: i or j (in the relevant positions) are in the convolution case.
    # i contributes to g_s when i and i+s align in the convolution term.
    
    # Naively but safely: just a function that re-computes change at *start* by a loop, not exact calculus.
    
    # Try to compute how each shift value g_s changes on swap  
    if i < n and i + s < n:
        delta = - (1 - h[i + s])
    else:

    # If we swap i (1) with j (0) then we have in final pattern:
    # h[i] = 0, h[j] = 1.
    # For all shifts s:
    # g_s terms:
    # If i belongs to shift s (i = s + offset), then that term for g_s is h[i] * (1 - h[i + s])  
    # Previously it was 1*(1-h[i+s]) = 1 - h[i+s], now becomes 0*(1-h[i+s]) = 0. 
    # So change is - (1 - h[i+s]) 
    # Similar: if in shift s, j = s + d for some d,
    # previous term was 0*(1 - h[j+s]) = 0; becomes 1*(1 - h[j+s]). 
    # Change is + (1 - h[j+s])  
    # 
    # But actually implementation in full correl works like:
    # length of h if 1 then correlation [0, 2n-2] gives length 2n-1
    # so s ranges from -(n-1) to n-1.  
    # Convergence of g_s to 
    # g_s = sum over i (h[i] * (1 - h[i+s])) 
    # As numpy cosinvolves with mode="full", you get indices s = -n+1 to n-1  

    # Safer way for swinging:
    shape = 2 * len(h) - 1
    delta_g = np.zeros(shape)  # shift offset in [-n+1, n-1] => shift indices from 0 to 2n-3
    start = max(0, -s)
    end = min(2*n - 3, 2*n - 3 - s)
    
    # But to save work, just consider for the specific h:
    # shift s maps to correlation[shift_idx = n-1 + s]  
    # so possible range s = 0 to 2n-3 for correlation index,
    # Which means last i back is n-1 and 0 forward of h corresponds to h[0] 
    # The convention then is: positions run by h[] indices from [0, n-1]  
    # Correlation index runs from [0, 2n-3]. 
    # Shift s → index in range [0,2n-3], specifically s (padded with 0,1...) maps to shift = s - n+1.
    
    return delta

def local_swap_search(h, n, max_time_s):
    """Perform a local search with target swaps"""
    start_time = time.time()
    
    # Convert to binary if not
    h_binary = (h > 0.5).astype(int)  
    
    # Precompute g_s for current state
    g_s = compute_g_s(h_binary)
    current_max = np.max(g_s)
    
    # Find indices of 1s and 0s
    ones = np.where(h_binary == 1)[0]
    zeros = np.where(h_binary == 0)[0] 
    
    # Track tabu list (pairs swapped recently)  
    tabu_pairs = set()
    tabu_size = 100
    
    iteration = 0
    climb_count = 0
    same_max_count = 0
    max_same_count = 2000  # max allowed climbs where no improvement
    
    while time.time() - start_time < max_time_s:
        iteration += 1
        if iteration % 1000 == 0:
            elapsed = time.time() - start_time
            print(f"Iter {iteration}, elapsed={elapsed:.2f}s, C5={current_max:.6f}") 
            
        # Determine worst shift  
        s_max = np.argmax(g_s)
        if s_max == 0:  # due to indexing of np.correlate full
            s_max = np.argmax(g_s[1:]) + 1  # non-zero shifts
            
        # For s_max, temporarily create better set of candidates
        # Find positions that have contribution to increase maximum:
        # positions i=1 which are unstuck by s_max shift should be rare  
        # But take some copies of such pairs:
        valid_ones_s = []
        valid_zeros_s = []
        
        for i in ones:
            # Check if this contributes a '1' to s_max if (h[i] * (1-h[i+s_max]))=1,
            # i.e., h[i]=1 and h[i+s_max]=0, so that term becomes 1*1=1.  
            # That's not right. Let's look at the **contribution** of i to shift s_max
            # If h[i] = 1 and h[i+s_max] = 1: contributes 1 * 0 = 0
            # If h[i] = 1 and h[i+s_max] = 0: contributes 1 * 1 = 1 
            # If h[i] = 0: contributes 0
            shift_idx = s_max  # frame view of convolution outputs
            if (i + shift_idx < n) and (i + shift_idx >= 0):
                if h_binary[i] == 1 and h_binary[i + shift_idx] == 0:
                    valid_ones_s.append(i)
        
        for j in zeros:
            # if h[j]=0, we check if swapping it with a 1 would reduce s_max
            if j + shift_idx < n and j + shift_idx >= 0:
                if h_binary[j] == 0 and h_binary[j + shift_idx] == 1:
                    valid_zeros_s.append(j)
        
        # Make sure we have enough to experiment
        max_pairs = 100
        if len(valid_ones_s) > 0 and len(valid_zeros_s) > 0:
            try:
                # Randomly test a few swaps involving these problematic pairs
                test_try = 0
                while test_try < max_pairs:
                    test_try += 1
                    i_chosen = np.random.choice(valid_ones_s)
                    j_chosen = np.random.choice(valid_zeros_s)
                    
                    key = tuple(sorted([i_chosen, j_chosen]))
                    if key in tabu_pairs:
                        continue
                        
                    # Compute change:
                    # Full version using full sweep
                    h_copy = h_binary.copy()
                    h_copy[i_chosen] = 0
                    h_copy[j_chosen] = 1
                    
                    g_s_temp = compute_g_s(h_copy)
                    max_new = np.max(g_s_temp)
                    
                    if max_new < current_max:
                        h_binary = h_copy
                        g_s = g_s_temp
                        current_max = max_new
                        
                        # Add to tabu list
                        tabu_pairs.add(key)
                        if len(tabu_pairs) > tabu_size:
                            it = iter(tabu_pairs)
                            tabu_pairs.remove(next(it))
                        
                        climb_count += 1
                        break
                        
                    elif test_try == max_pairs - 1:
                        # Final test - if can't find a move early, force one
                        h_copy = h_binary.copy()
                        i = np.random.choice(ones)
                        j = np.random.choice(zeros) 
                        key = tuple(sorted([i, j]))
                        if key in tabu_pairs:
                            continue
                        h_copy[i] = 0
                        h_copy[j] = 1
                        
                        g_s_temp = compute_g_s(h_copy)                        
                        max_new = np.max(g_s_temp)
                        if max_new < current_max:
                            h_binary = h_copy
                            g_s = g_s_temp
                            current_max = max_new
                            
                            tabu_pairs.add(key)
                            if len(tabu_pairs) > tabu_size:
                                it = iter(tabu_pairs)
                                tabu_pairs.remove(next(it))
                        break

            except Exception:
                pass

        # Stagnation check        
        if climb_count == 0:
            same_max_count += 1
            if same_max_count >= max_same_count:
                # Perturb
                pert_len = max(1, len(ones) // 20)  # try out 5%  
                if pert_len > 0:
                    perturb_indices = np.random.choice(len(ones), size=pert_len, replace=False)
                    pos_ones = ones[perturb_indices]
                    for pos in pos_ones:
                        h_binary[pos] = 0
                    diff = 2 * pert_len  # we deleted 2*pert_len bits of 1s
                    new_ones = np.random.choice(len(h_binary), size=diff, replace=False)
                    for pos in new_ones:
                        h_binary[pos] = 1
                    g_s = compute_g_s(h_binary)
                    current_max = np.max(g_s)
                same_max_count = 0            
        else:
            climb_count = 0  
            same_max_count = 0

    return h_binary, current_max

def upscale_h(h_coarse, n_new):
    """Upscale a sequence to larger resolution"""
    h_new = np.zeros(n_new)  
    dx_old = 2.0 / len(h_coarse)
    dx_new = 2.0 / n_new
    
    ratio = n_new // len(h_coarse)
    
    # Linear interpolation
    for i in range(len(h_coarse)):
        for j in range(ratio):
            h_new[i*ratio + j] = h_coarse[i]
            
    # Correct rounding balances if needed
    current_sum = np.sum(h_new) * dx_new
    h_new = h_new * (1.0 / current_sum)
    
    return h_new

def project_sum_to_normal(h, n_points):
    """Project h to sum equal to n_points/2 using simple technique"""
    dx = 2.0 / n_points
    current_sum = np.sum(h) * dx
    h = h * (1.0 / current_sum)
    return h
