```python
import numpy as np
import time
import cvxpy as cp
from cvxpy import ECOS, GLPK, CBC

def evaluate_c5(h, dx):
    """Compute the maximum overlap integral for the given step function h."""
    c = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(c) * dx
    return c5

def water_fill(h, target_sum):
    """Project h onto the sum constraint using water filling."""
    h = np.array(h)
    target_sum = float(target_sum)
    n = len(h)
    # Ensure bounds [0,1]
    h = np.clip(h, 0, 1)
    total = np.sum(h)
    if abs(total - target_sum) < 1e-10:
        return h
    if total > target_sum:
        # Need to decrease values
        # Sort descending
        idx = np.argsort(h)[::-1]
        h_sorted = h[idx]
        # Find threshold
        cumsum = np.cumsum(h_sorted)
        # Find largest k such that cumsum[k] <= target_sum
        # Start from sorted values
        k = np.searchsorted(cumsum, target_sum, side='right')
        if k == 0:
            return np.zeros(n)
        # If not exact, subtract residual
        residual = target_sum - cumsum[k-1]
        threshold = residual / (n - k)
        h_result = h.copy()
        for i in range(k):
            h_result[idx[i]] = threshold
        for i in range(k, n):
            h_result[idx[i]] = 0
        return h_result
    else:
        # Need to increase values (we can't do this in [0,1])
        # Just return the same h, expect that sum is exactly target by construction
        return h

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start = time.time()
    N = 140
    dx = 2.0 / N

    # Build list of all valid pairs (i, s)
    pairs = []
    pair_to_idx = {}
    idx = 0
    for s in range(N):
        for i in range(N - s):
            pairs.append((i, s))
            pair_to_idx[(i, s)] = idx
            idx += 1
    num_pairs = len(pairs)

    # Try solving LP:
    print("Starting LP solve")
    # Variables
    h = cp.Variable(N, name='h')
    y = cp.Variable(num_pairs, name='y')
    alpha = cp.Variable()

    # Constraints
    constraints = []
    # Box constraints and total mass
    constraints += [0 <= h, h <= 1]
    constraints += [cp.sum(h) == N / 2]

    # McCormick envelope
    for k, (i, s) in enumerate(pairs):
        constraints += [y[k] <= h[i], y[k] <= h[i + s],
                        y[k] >= h[i] + h[i + s] - 1,
                        y[k] >= 0]

    # α constraints: α >= dx * (N/2 - sum_y[i,s])
    # Treat shifts from -N+1 to N-1
    for s_shift in range(N):
        # This calculates for the positive shift s_shift (assuming reflected s_shift in mode full)
        # In full mode of np.correlate, effective shift ranges from -(N-1) to (N-1) -- but we model only non-negative
        # Assuming a shift s from 0 to N-1, calculate sum over i: y[i,s]
        indices = []
        for i in range(N - s_shift):
            indices.append(pair_to_idx[(i, s_shift)])
        sum_y = cp.sum(y[indices])
        constraints += [alpha >= dx * (N / 2 - sum_y)]

    # Objective
    objective = cp.Minimize(alpha)
    prob = cp.Problem(objective, constraints)

    # Solve
    try:
        prob.solve(solver=ECOS, max_iters=20000, verbose=False)
    except:
        print("LP solver failed")

    if prob.status in ["optimal", "optimal_inaccurate"]:
        h_lp = h.value.copy()
        h_lp = water_fill(h_lp, N/2)
        c5_lp = evaluate_c5(h_lp, dx)
        print(f"LP yielded c5 = {c5_lp}")
    else:
        print("LP did not converge. Using fallback.")
        h_lp = np.ones(N) * 0.5
        h_lp = water_fill(h_lp, N/2)
        c5_lp = evaluate_c5(h_lp, dx)
        print(f"LP fallback c5 = {c5_lp}")

    if c5_lp <= 0.3807:
        return (h_lp, c5_lp, N)

    # If the LP is not good enough, try MILP
    print("Starting MILP solve")
    h_bin = cp.Variable(N, boolean=True, name='h_bin')
    y_bin = cp.Variable(num_pairs, name='y_bin')
    alpha_bin = cp.Variable()

    # Same constraints with binary h
    constraints_bin = []
    constraints_bin += [0 <= h_bin, h_bin <= 1]
    constraints_bin += [cp.sum(h_bin) == N / 2]

    # McCormick
    for k, (i, s) in enumerate(pairs):
        constraints_bin += [y_bin[k] <= h_bin[i], y_bin[k] <= h_bin[i + s],
                            y_bin[k] >= h_bin[i] + h_bin[i + s] - 1,
                            y_bin[k] >= 0]

    for s_shift in range(N):
        indices = []
        for i in range(N - s_shift):
            indices.append(pair_to_idx[(i, s_shift)])
        sum_y = cp.sum(y_bin[indices])
        constraints_bin += [alpha_bin >= dx * (N / 2 - sum_y)]

    objective_bin = cp.Minimize(alpha_bin)
    prob_bin = cp.Problem(objective_bin, constraints_bin)

    # Try to solve with time limit
    budget_remaining = max(5, budget_s - (time.time() - start))
    if budget_remaining < 10:
        # Use a very short time limit
        solver_options = {'maxTime': 1.0}
    else:
        solver_options = {'maxTime': min(600, budget_remaining * 0.6)}

    try:
        prob_bin.solve(solver=solver, solver_options=solver_options, verbose=False)
    except:
        print("MILP solver error")

    if prob_bin.status in ["optimal", "optimal_inaccurate"]:
        h_milp = h_bin.value.copy()
        h_milp = water_fill(h_milp, N/2)  # Will be binary anyway
        c5_milp = evaluate_c5(h_milp, dx)
        print(f"MILP yielded c5 = {c5_milp}")
        if c5_milp <= 0.3807:
            return (h_milp, c5_milp, N)
    else:
        print("MILP did not converge; using rounding fallback")
        h_milp = h_lp
        h_milp = water_fill(h_milp, N/2)
        c5_milp = evaluate_c5(h_milp, dx)
        print(f"MILP fallback c5 = {c5_milp}")

    # Local refinement: start with best vector
    best_h = h_milp
    best_c5 = c5_milp

    # Use greedy rounding if we're still not good enough
    if best_c5 > 0.3807:
        # Make a greedy binary rounding
        h_round = np.copy(h_lp)
        idxs = np.argsort(h_round)[::-1]
        new_h = np.zeros(N)
        for i in range(int(N/2)):
            new_h[idxs[i]] = 1
        best_h = water_fill(new_h, N/2)
        best_c5 = evaluate_c5(best_h, dx)
        print(f"Rounded c5 = {best_c5}")

    # Try local hill climb
    if best_c5 > 0.3807:
        print("Performing local refinement")
        num_refinements = 0
        while num_refinements < 500 and (time.time() - start) < budget_s * 0.9:
            num_refinements += 1
            # Find worst shift
            h_test = best_h.copy()
            c_vals = []
            for s in range(N):
                c = 0
                for i in range(N - s):
                    c += h_test[i] * (1 - h_test[i + s])
                c_vals.append(c * dx)
            worst_s = np.argmax(c_vals)
            # Compute gradient
            grad = np.zeros(N)
            for i in range(N - worst_s):
                grad[i] += 1 - h_test[i + worst_s]
                grad[i + worst_s] -= h_test[i]
            # Update with small step
            step_size = 0.02
            updated_h = best_h - step_size * grad
            # Clip and project
            updated_h = np.clip(updated_h, 0, 1)
            updated_h = water_fill(updated_h, N/2)
            updated_c5 = evaluate_c5(updated_h, dx)
            if updated_c5 < best_c5:
                best_c5 = updated_c5
                best_h = updated_h
                if best_c5 <= 0.3807:
                    break
        print(f"Refined c5 = {best_c5}")

    return (best_h, best_c5, N)
```
