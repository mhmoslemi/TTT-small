import numpy as np
import time
import random
from scipy.optimize import differential_evolution, minimize
import math


def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    overlap = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(overlap) * dx
    return c5


def propose_move(h, max_step):
    """Propose a new feasible h vector by adjusting two elements"""
    h_new = h.copy()
    N = len(h)
    
    # Try to find a valid move
    max_tries = 100
    for _ in range(max_tries):
        i = random.randint(0, N - 1)
        j = random.randint(0, N - 1)
        if i == j:
            continue
            
        # Determine feasible interval for δ
        delta_low = max(-h_new[i], h_new[j] - 1)
        delta_high = min(1 - h_new[i], h_new[j])
        
        if delta_low >= delta_high:
            continue
            
        # Sample δ from uniform distribution over feasible range
        step = min(max_step, delta_high - delta_low)
        if step <= 0:
            continue
        delta = random.uniform(delta_low, delta_high)
        
        # Apply move
        h_new[i] += delta
        h_new[j] -= delta
        
        return h_new
    
    # If no valid move found, return original
    return h.copy()


def optimize_with_de(h_init, budget_s):
    """Use differential evolution optimizer"""
    start = time.time()
    N = len(h_init)
    dx = 2.0 / N
    
    def objective(h):
        # Normalize manually to ensure sum = N/2
        h = h * (N / 2) / np.sum(h)
        return c5_of(h)
    
    # Bounds for each variable
    bounds = [(0, 1) for _ in range(N)]
    
    # Call differential evolution
    result = differential_evolution(
        objective,
        bounds,
        maxiter=max(100, int(budget_s * 0.3 / 2)),
        popsize=15,
        seed=42,
        init='latinhypercube',
        disp=False
    )
    
    if result.success:
        h_best = result.x * (N / 2) / np.sum(result.x)
        c5_best = result.fun
    else:
        # Fall back to using initial point
        h_best = h_init
        c5_best = c5_of(h_init)
        
    return h_best, c5_best


def improved_anneal(initial_h, time_budget_s):
    """Improved simulated annealing with multiple strategies"""
    start = time.time()
    
    # Parameters - more refined cooling schedule
    T_start = 0.1
    T_end = 1e-8
    step_start = 0.1
    
    # Set up the current state
    cur = initial_h.copy()
    c_cur = c5_of(cur)
    
    best = cur.copy()
    c_best = c_cur
    iteration = 0
    
    # Optimization parameters
    max_iter = int(time_budget_s * 500)
    
    # Main loop
    while time.time() - start < time_budget_s and iteration < max_iter:
        iteration += 1
        elapsed_fraction = min((time.time() - start) / time_budget_s, 1.0)
        
        # Cooling schedule
        T = T_start * (T_end / T_start) ** elapsed_fraction
        step = step_start * (1 - elapsed_fraction)**1.5
        
        # Try three different types of moves
        if random.random() < 0.7:  # Local perturbations
            h_proposed = propose_move(cur, step)
        elif random.random() < 0.9:  # Random interval adjustment
            h_proposed = cur.copy()
            a = random.randint(0, len(cur) - 1)
            b = random.randint(a, len(cur) - 1)
            delta = random.uniform(-step, step)
            # Ensure bounds
            if a != b:
                delta = delta / (b - a + 1)
                for i in range(a, b + 1):
                    if 0 <= h_proposed[i] + delta <= 1:
                        h_proposed[i] += delta
            else:
                h_proposed[a] = max(0, min(1, h_proposed[a] + delta))
        else:  # Random overall adjustment
            h_proposed = cur.copy()
            factor = random.uniform(0.95, 1.05)
            if np.sum(h_proposed) > 0:
                h_proposed = h_proposed * factor * (len(cur) / 2) / np.sum(h_proposed)
                
        c_proposed = c5_of(h_proposed)
        
        # Accept with Metropolis probability
        if c_proposed <= c_cur:
            cur = h_proposed
            c_cur = c_proposed
            if c_proposed < c_best:
                best = h_proposed.copy()
                c_best = c_proposed
        else:
            if random.random() < np.exp(-(c_proposed - c_cur) / T):
                cur = h_proposed
                c_cur = c_proposed
                
    return best, c_best
