import numpy as np
import time
import random
from scipy.optimize import differential_evolution, minimize


def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    overlap = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(overlap) * dx
    return c5


def evaluate_and_constrain(h, dx):
    """Evaluate C5 and ensure constraints are met"""
    # Ensure bounds
    h = np.clip(h, 0, 1)
    # Ensure integral constraint
    h = h * (1.0 / np.sum(h)) * (2.0 / dx)
    return c5_of(h)


def optimize_with_differential_evolution(initial_h, time_budget_s):
    """Use differential evolution with proper constraints"""
    start = time.time()
    N = len(initial_h)
    dx = 2.0 / N
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(N)]
    
    def objective(h):
        # Normalize to satisfy integral constraint
        h_norm = h * (1.0 / np.sum(h)) * (2.0 / dx)
        return c5_of(h_norm)
    
    # Run DE
    result = differential_evolution(
        objective,
        bounds,
        maxiter=int(1000 * (time_budget_s / 600)),
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.7,
        seed=42,
        disp=False,
        atol=1e-6,
        tol=1e-6
    )
    
    # Final normalization and evaluation
    h_final = result.x * (1.0 / np.sum(result.x)) * (2.0 / dx)
    c5_final = c5_of(h_final)
    
    return h_final, c5_final


def local_search(h, time_budget_s):
    """Perform local search using gradient-based method with constraints"""
    start = time.time()
    N = len(h)
    dx = 2.0 / N
    
    # Define constraint function for sum constraint
    def constraint_sum(h):
        return np.sum(h) * dx - 1.0
    
    # Objective with constraint handling
    def objective(h):
        h = h * (1.0 / np.sum(h)) * (2.0 / dx)
        return c5_of(h)
    
    # Use L-BFGS-B with constraints
    from scipy.optimize import minimize
    bounds = [(0, 1) for _ in range(N)]
    
    # Take multiple restarts with different initial guesses
    best_c5 = float('inf')
    best_h = h.copy()
    
    for _ in range(5):
        if time.time() - start > time_budget_s * 0.9:
            break
            
        # Slightly perturb current solution for restart
        perturbed_h = h + np.random.normal(0, 0.01, N)
        perturbed_h = np.clip(perturbed_h, 0, 1)
        
        # Normalize
        perturbed_h = perturbed_h * (1.0 / np.sum(perturbed_h)) * (2.0 / dx)
        
        res = minimize(objective, perturbed_h, method='L-BFGS-B', 
                       bounds=bounds, options={'maxiter': 500, 'ftol': 1e-6})
        
        if not res.success:
            continue
            
        # Evaluate result
        h_final = res.x * (1.0 / np.sum(res.x)) * (2.0 / dx)
        c5_final = c5_of(h_final)
        
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()
    
    return best_h, best_c5


def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function"""
    random.seed(seed)
    np.random.seed(seed)
    start = time.time()
    
    # Use 140 points as original
    N = 140
    dx = 2.0 / N
    
    # Build initial seed from provided values if available
    if 'initial_h_values' in globals():
        h_provided = np.interp(
            np.linspace(0, 2, N),
            np.linspace(0, 2, len(initial_h_values)),
            initial_h_values
        )
        # Renormalize to exact sum
        h_provided = h_provided * (N / 2) / np.sum(h_provided)
    else:
        h_provided = np.ones(N) * 0.5
    
    # Use DE to initialize, then perform local search for refinement
    h_opt, c5_opt = optimize_with_differential_evolution(h_provided, budget_s * 0.7)
    
    # Refine with local search
    h_final, c5_final = local_search(h_opt, budget_s * 0.3)
    
    return (h_final, c5_final, N)
