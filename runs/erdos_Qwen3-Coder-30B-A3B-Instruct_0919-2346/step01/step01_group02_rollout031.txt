import numpy as np
import time

def compute_c5_and_worst_shift(h):
    """Compute C5 and return the worst shift index."""
    dx = 2.0 / len(h)
    c = np.correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(c)
    s = max_idx - (len(h) - 1)
    c5 = c[max_idx] * dx
    return c5, s

def compute_gradient(h, shift):
    """Compute gradient of C5 with respect to h for a given shift."""
    N = len(h)
    dx = 2.0 / N
    g = np.full(N, 1.0) * dx
    if shift >= 0:
        if shift < N:
            g[:N-shift] -= h[shift:] * dx
            g[shift:] -= h[:N-shift] * dx
    else:
        shift = -shift
        if shift < N:
            g[shift:] -= h[:N-shift] * dx
            g[:N-shift] -= h[shift:] * dx
    return g

def project_onto_bounds_and_sum(v, target_sum):
    """Project vector v onto box [0,1] and hyperplane sum = target_sum."""
    # Binary search for lambda
    lower = -np.max(v)
    upper = 1 - np.min(v)
    
    for _ in range(60):
        mid = (lower + upper) / 2.0
        w = np.clip(v + mid, 0, 1)
        s = np.sum(w)
        if s < target_sum:
            lower = mid
        else:
            upper = mid
    
    w = np.clip(v + (lower + upper) / 2.0, 0, 1)
    
    # Adjust slightly to ensure sum is correct
    diff = target_sum - np.sum(w)
    if abs(diff) > 1e-12:
        free_idx = (w > 0) & (w < 1)
        if np.any(free_idx):
            w[free_idx] += diff / np.sum(free_idx)
            w = np.clip(w, 0, 1)
        else:
            w += diff / len(w)
            w = np.clip(w, 0, 1)
    
    return w

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    N = 140
    dx = 2.0 / N
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():

                      initial_h_values)
        h0 = h0 * (N/2) / np.sum(h0)
    else:
        h0 = np.ones(N) * 0.5  # fallback
    
    # Two-peak shape
    h1 = np.ones(N)
    x_vals = np.linspace(0, 2, N)
    a = 0.3
    u = 0.9
    v = 0.1
    mask = (x_vals < a) | (x_vals >= 2 - a)
    h1[mask] = u
    h1[~mask] = v
    h1 = h1 * (N/2) / np.sum(h1)
    
    # Uniform distribution
    h2 = np.ones(N) * 0.5
    
    # Start with initial best
    best_h = h0.copy()
    best_score, _ = compute_c5_and_worst_shift(best_h)
    
    # Check other initializations
    for h_init in [h1, h2]:
        score, _ = compute_c5_and_worst_shift(h_init)
        if score < best_score:
            best_h = h_init.copy()
            best_score = score
    
    # Binary random initializations
    for _ in range(3):
        h_bin = np.zeros(N)
        idx = np.random.choice(N, N//2, replace=False)
        h_bin[idx] = 1.0
        score, _ = compute_c5_and_worst_shift(h_bin)
        if score < best_score:
            best_h = h_bin.copy()
            best_score = score

    # Gradient descent with adaptive step size
    start = time.time()
    while time.time() - start < budget_s:
        # Run gradient descent for single step
        cur_h = best_h.copy()
        cur_score, worst_shift = compute_c5_and_worst_shift(cur_h)
        
        # Compute gradient for worst shift
        grad = compute_gradient(cur_h, worst_shift)
        
        # Step size with decreasing schedule
        elapsed = time.time() - start
        frac = elapsed / budget_s
        step = 20 * (1 - frac) + 0.1  # Start at 20, end at 0.1
        
        # Update with step
        v = cur_h - step * grad
        cur_h = project_onto_bounds_and_sum(v, N/2)
        
        # Evaluate new solution
        new_score, _ = compute_c5_and_worst_shift(cur_h)
        
        # Accept or reject with simulated annealing-like acceptance
        if new_score <= cur_score or np.random.rand() < np.exp(-(new_score - cur_score) / (0.01 + cur_score * 0.1)):
            if new_score < best_score:
                best_h = cur_h.copy()
                best_score = new_score
        else:
            cur_h = best_h.copy()  # Reset if rejected
        
        # Occasionally add jitter
        if np.random.rand() < 0.05:
            noise = np.random.uniform(-0.05, 0.05, size=N)
            v = cur_h + noise
            cur_h = project_onto_bounds_and_sum(v, N/2)
            new_score, _ = compute_c5_and_worst_shift(cur_h)
            if new_score < best_score:
                best_h = cur_h.copy()
                best_score = new_score

    return (best_h, best_score, N)
