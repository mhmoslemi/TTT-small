import numpy as np
import time

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    correlation = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(correlation) * dx
    return max_overlap

def upsample_initial(h0, n_new):
    """Upsample h0 to n_new points using linear interpolation"""
    n_old = len(h0)
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h0)
    return h_new

def project_to_simplex(v, target_sum=1.0):
    """Project vector v onto the simplex {x : sum(x) = target_sum, x >= 0}"""
    # Sort v in descending order
    v_sorted = np.sort(v)[::-1]
    # Compute cumulative sums
    cumsum = np.cumsum(v_sorted)
    # Find the index where the projection occurs
    n = len(v)
    theta = (cumsum - target_sum) / np.arange(1, n + 1)
    # Find the largest index such that v_sorted[i] > theta[i]
    # This is guaranteed to find an index in [0, n-1]
    idx = 0
    for i in range(n):
        if v_sorted[i] > theta[i]:
            idx = i
        else:
            break
    # Compute the Lagrange multiplier
    lambda_ = theta[idx]
    # Project v using lambda
    x = np.clip(v - lambda_, 0, 1)
    return x

def compute_softmax_weights(f_vec, gamma):
    """Compute weights for the soft max of f_vec with temperature gamma"""
    f_scaled = gamma * np.array(f_vec)
    f_scaled -= np.max(f_scaled)  # for numerical stability
    exp_f = np.exp(f_scaled)
    weights = exp_f / np.sum(exp_f)
    return weights

def compute_gradient(h, dx):
    """Compute ∇ J_γ(h) as weighted sum of ∇ f_s(h), s = -n+1,...,n-1"""
    n = len(h)
    grad = np.zeros_like(h, dtype=float)
    # Build the list of all valid shifts, in order of increasing |s|
    shifts = list(range(-n+1, n))  # -n+1 to n-1
    # Precompute f_s(h) for each shift = recall the full convolution
    # For speed, use numPy direct convolution with shifts manually included?
    # But now we just loop and get shifts
    all_fs = []
    for s in shifts:
        if s >= 0:
            f_s = np.sum(h[:n-s] * (1 - h[s:]))
        else:
            f_s = np.sum(h[-s:] * (1 - h[:-s]))
        all_fs.append(f_s)
    all_fs = np.array(all_fs)
    # Compute softmax weights
    weights = compute_softmax_weights(all_fs, 1.0)  # start with gamma=1
    # Compute ∇f_s = dx * (g_i^(s)) for each s
    for i, s in enumerate(shifts):
        w = weights[i]
        g = np.zeros_like(h)
        if s >= 0:
            g[:n-s] += 1 - h[s:]          # add 1-to-h[s:] shifted
            g[s:] -= h[:n-s]              # subtract h[:n-s] at s'th positions
        else:
            g[-s:] += 1 - h[:-s]          # add 1-to-h[:-s] shifted
            g[:n+s] -= h[-s:]             # subtract h[-s:] at 0th positions
        grad += w * dx * g
    return grad
