import numpy as np
import time
from scipy.optimize import differential_evolution, minimize

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral via full convolution of h with (1-h)
    overlaps = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum of the full correlation vector
    # scaled by dx
    c5 = np.max(overlaps) * dx
    return c5

def optimize_hillclimb(h0, budget_s, seed):
    """Optimize using hill climb with adaptive step size."""
    np.random.seed(seed)
    start = time.time()
    
    def propose_move(h, step_size):
        h = h.copy()
        N = len(h)
        i = np.random.randint(0, N)
        # Try to find a valid delta
        delta_low = -h[i]
        delta_high = 1 - h[i]
        if delta_low >= delta_high:
            return h
        delta = np.random.uniform(max(-step_size, delta_low), 
                                min(step_size, delta_high))
        h[i] += delta
        return h

    cur = h0.copy()
    c_cur = c5_of(cur)
    
    step_size = 0.05
    patience = 0
    max_patience = 100
    best = cur.copy()
    best_c = c_cur
    last_improvement = 0
    
    while time.time() - start < budget_s:
        if patience > max_patience:
            step_size = min(0.1, step_size * 1.2)
            patience = 0
            last_improvement = 0
        
        h_new = propose_move(cur, step_size)
        # Maintain sum constraint
        h_new = h_new * (len(h_new) / 2) / np.sum(h_new)
        c_new = c5_of(h_new)
        
        if c_new <= c_cur:
            cur = h_new
            c_cur = c_new
            patience = 0
            last_improvement = 0
        else:
            patience += 1
            last_improvement += 1
            
        if c_new < best_c:
            best = h_new.copy()
            best_c = c_new
            
    return best, best_c

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start = time.time()
    N = 140
    dx = 2.0 / N
    
    # Use initial construction if available

        if len(h0) != N:

            x_new = np.linspace(0, 2, N)
            h0 = np.interp(x_new, x0, h0)
        h_initial = h0 * (N / 2) / np.sum(h0)
    else:
        h_initial = np.random.rand(N) * 0.5
        h_initial *= (N / 2) / np.sum(h_initial)
        
    # Generate some initial candidates with different structures
    h_uniform = np.full(N, 0.5)
    h_uniform *= (N / 2) / np.sum(h_uniform)
    
    h_peak = np.full(N, 0.1)
    # Create a single peak in center
    idxs = np.where(np.abs(np.linspace(0, 2, N) - 1) < 0.3)[0]
    h_peak[idxs] = 0.9
    h_peak *= (N / 2) / np.sum(h_peak)
    
    # Generate candidate solutions
    candidates = [h_initial, h_uniform, h_peak]
    
    best_h = None
    best_c5 = float('inf')
    
    # Evaluate each candidate with hill climbing
    budget_per_candidate = budget_s / (len(candidates) + 2)
    
    for i, h_seed in enumerate(candidates):
        if time.time() - start > budget_s * 0.9:
            break
        h, c5 = optimize_hillclimb(h_seed, budget_per_candidate, seed + i)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5
            
    # Run a fine-tuning phase with simple optimization
    if best_h is not None and time.time() - start < budget_s * 0.95:
        # Use simple gradient ascent with constraints
        def objective(h):
            # Add a small penalty to keep values in bounds
            penalty = 1e6 * np.sum(np.maximum(0, h - 1) + np.maximum(0, -h))
            return c5_of(h) + penalty
            
        def constraint_sum(h):
            return np.sum(h) - N / 2
            
        cons = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(N)]
        
        # Optimize a small random subset for this phase
        subset_size = min(50, N)
        indices = np.random.choice(N, subset_size, replace=False)
        
        # Create a perturbed version of current best
        h_perturbed = best_h.copy()
        for i in indices:
            h_perturbed[i] += np.random.normal(0, 0.05)
            h_perturbed[i] = np.clip(h_perturbed[i], 0, 1)
        h_perturbed = h_perturbed * (N / 2) / np.sum(h_perturbed)
        
        try:
            result = minimize(objective, h_perturbed, method='SLSQP', bounds=bounds, constraints=cons, 
                            options={'maxiter': 100, 'ftol': 1e-6})
            if result.success and result.fun < best_c5:
                best_h = result.x.copy()
                best_c5 = result.fun
        except:
            pass  # Ignore if optimization fails
            
    # Final polishing
    if best_h is not None and time.time() - start < budget_s:
        h_final, c_final = optimize_hillclimb(best_h, 0.1 * budget_s, seed + 100)
        if c_final < best_c5:
            best_h = h_final
            best_c5 = c_final
            
    return (best_h, best_c5, N)
