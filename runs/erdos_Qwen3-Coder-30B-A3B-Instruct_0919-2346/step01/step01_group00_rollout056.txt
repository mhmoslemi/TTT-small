import numpy as np
import time
import random

def compute_C_vector(h, dx):
    """Compute vector of overlaps C_τ(h) = 1 - dx * S_τ(h)."""
    # Compute full autocorrelation S_τ(h) = sum h_i h_{i+τ}
    S = np.correlate(h, h, mode='full')
    # S is of length 2*n-1
    # Only the middle n values are useful for shift indices 0..n-1
    # But we need all 2*n-1 for proper indexing
    c = 1 - dx * S
    return c

def compute_softmax_weights(c, alpha):
    """Compute softmax weights w_τ = exp(α c_τ) / sum exp(α c_σ)."""
    c_max = np.max(c)
    # Log-sum-exp trick for numerical stability
    exp_c = np.exp(alpha * (c - c_max))
    w = exp_c / np.sum(exp_c)
    return w

def compute_gradient(h, w, dx):
    """Compute gradient of smooth objective: g_i = -dx * (w * h + w_rev * h)_i."""
    # w_rev is reversed w vector
    w_rev = w[::-1]
    # Using convolve with mode='same' is sufficient for zero padding
    # Normalization constant is automatic per design
    g = -dx * (np.convolve(w, h, mode='same') + np.convolve(w_rev, h, mode='same'))
    return g

def project_to_feasible(h, target_sum):
    """Project h to [0,1] with exact sum = target_sum."""
    # Water-filling projection
    # Clip to [0,1], then adjust via Lagrange multiplier
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-12:
        return h
    # Perform bisection to find lambda
    h_sorted = np.sort(h)[::-1]  # descending order
    h_cumsum = np.cumsum(h_sorted)
    indices = np.arange(1, len(h_sorted) + 1)
    # Find the index where threshold would be
    # We seek a lambda such that sum clip(h_i - lambda, 0, 1) = target_sum
    # Use bisection on the lambda
    lower, upper = np.min(h) - 1, np.max(h) + 1
    tol = 1e-12
    max_iter = 1000
    for _ in range(max_iter):
        lambda_ = (lower + upper) / 2
        # Compute projected sum with this lambda
        h_proj = np.clip(h - lambda_, 0, 1)
        sum_proj = np.sum(h_proj)
        if abs(sum_proj - target_sum) < tol:
            break
        elif sum_proj > target_sum:
            lower = lambda_
        else:
            upper = lambda_
    # Final projection
    h_final = np.clip(h - (lower + upper)/2, 0, 1)
    return h_final
