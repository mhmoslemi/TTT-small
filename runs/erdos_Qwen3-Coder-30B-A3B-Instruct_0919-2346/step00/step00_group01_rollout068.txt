import numpy as np
import time
from scipy import fft
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n = 60
    dx = 2.0 / n
    K0 = [1, n//2, 2, n-2]  # Initial shifts to start with
    max_seconds_per_solve = min(budget_s / 2, 30)
    
    # Global timer check
    def check_time Left():
        return time.time() - start_time < budget_s - 20
    
    # Generate shift indices efficiently
    def get_shift_indices(k, n):
        return np.arange(n) - k % n
    
    # Binary optimization using MILP with cut generation
    def solve_with_cuts(K_set):
        # Create variables
        h = Variable(n, boolean=True)
        t = Variable()
        
        # Fix h[0] to break symmetry
        constraints = [h[0] == 1, sum(h) == n / 2]
        
        # Add auxiliary variables for selected shifts
        w_vars = {}
        a_vars = {}
        
        for k in list(K_set):
            w_vars[k] = Variable(n, boolean=True)
            a_vars[k] = sum(w_vars[k])
            constraints += [
                w_vars[k] <= h,
                w_vars[k] <= h[get_shift_indices(k, n)],
                w_vars[k] >= h + h[get_shift_indices(k, n)] - 1,
                t <= a_vars[k]
            ]
            
        # Objective: maximize t
        prob = Problem(Maximize(t), constraints)
        
        try:
            prob.solve(solver=CPLEX, maxSeconds=max_seconds_per_solve)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                return h.value, t.value
        except:
            pass
        return None, None

    # Function to compute minimum autocorrelation
    def compute_min_autocorr(h_vals):
        # FFT-based autocorrelation for all shifts
        hvals = np.asarray(h_vals)
        fft_len = 2 * n - 1
        H = fft.fft(hvals, fft_len)
        A = fft.ifft(H * np.conj(H))
        A_vals = np.real(A[:n])  # First n values of circular autocorrelation
        return np.min(A_vals)

    # Initial solve with K0
    h_best, t_best = solve_with_cuts(K0)
    if h_best is None:
        # Fallback: random binary construction
        h_vals = np.zeros(n)
        h_vals[:n//2] = 1
        np.random.shuffle(h_vals)
        h_best = h_vals
        t_best = compute_min_autocorr(h_vals)
        
    C5_min = 1 - (2 / n) * t_best
    
    # Cut generation loop
    if check_time_left():
        K_active = set(K0)
        while check_time_left():
            # Compute exact autocorrelations for all shifts
            h_current = h_best
            h_array = np.array(h_current)
            fft_len = 2 * n - 1
            H = fft.fft(h_array, fft_len)
            A_all = fft.ifft(H * np.conj(H))
            A_vals = np.real(A_all[:n])  # First n values of circular autocorrelation
            
            # Find the minimum autocorrelation shift
            min_k = np.argmin(A_vals)
            k_to_add = min_k
            
            if k_to_add not in K_active:
                K_active.add(k_to_add)
                h_new, t_new = solve_with_cuts(K_active)
                if h_new is not None:
                    if t_new > t_best:
                        h_best = h_new
                        t_best = t_new
                        C5_min = 1 - (2 / n) * t_best
                else:
                    break
            else:
                break
                
    # Final exact evaluation of C5
    final_h = np.array(h_best)
    final_autocorr = compute_min_autocorr(final_h)
    final_c5 = 1 - (2 / n) * final_autocorr
    
    # Gradient polishing if time allows and final bound is too high
    if check_time_left() and final_c5 > 0.3807:
        # Use simplified gradient ascent with projection
        h_continuous = final_h.copy()
        h_continuous = np.clip(h_continuous, 0, 1)
        steps = min(200, int((budget_s - (time.time() - start_time)) * 5))
        
        for _ in range(steps):
            if not check_time_left():
                break
            k_star = np.argmin(fft.ifft(fft.fft(h_continuous) * fft.fft(h_continuous, 2*n-1).conj()).real[:n])
            grad = h_continuous[np.roll(np.arange(n), k_star)] + h_continuous[np.roll(np.arange(n), -k_star)]
            h_continuous = np.clip(h_continuous + 0.001 * grad, 0, 1)
            
            # Project back to sum = n/2 using pairwise swap
            residual = np.sum(h_continuous) - n / 2
            if abs(residual) > 1e-6:
                indices_under = np.where(h_continuous < 1)[0]
                indices_over = np.where(h_continuous > 0)[0]
                if len(indices_under) > 0 and len(indices_over) > 0:
                    i_under = indices_under[0]
                    i_over = indices_over[0]
                    move_amount = min(h_continuous[i_over], -residual)
                    if move_amount > 1e-10:
                        h_continuous[i_under] += move_amount
                        h_continuous[i_over] -= move_amount
    
        # Final evaluation
        final_c5 = 1 - (2 / n) * compute_min_autocorr(h_continuous)
        final_h = h_continuous
    
    return (final_h, final_c5, n)
