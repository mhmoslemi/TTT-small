import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_erdos_solution(h_values):
    """Evaluate the Erdős constant C_5 for given h values."""
    h = np.array(h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute correlations for all lags
    correlations = np.correlate(h, 1 - h, mode='full')
    # The maximum is the maximum correlation
    C5 = np.max(correlations) * dx
    return C5

def initial_h_values():
    """Return an initial construction of h values."""
    # Simple piecewise constant construction
    n = 45
    h = np.zeros(n)
    # First 15 points are 1
    h[:15] = 1.0
    # Next 15 points are 0
    h[15:30] = 0.0
    # Next 15 points are 1
    h[30:] = 1.0
    return h

def objective(h_values):
    """Objective function to minimize."""
    h = np.array(h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute all correlations
    correlations = np.correlate(h, 1 - h, mode='full')
    # Return the maximum correlation (negative because we minimize)
    return -np.max(correlations) * dx

def constraint_eq(h_values):
    """Equality constraint: integral of h should be 1."""
    h = np.array(h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    return np.sum(h) * dx - 1.0

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function."""
    np.random.seed(seed)
    start_time = time.time()
    
    # Get initial solution
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Set up bounds (0 to 1 for each h[i])
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraints
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Optimization parameters
    maxiter = 1000
    maxfun = 10000
    
    # Use differential evolution with custom logic
    best_h = None
    best_c5 = float('inf')
    
    # Try several different optimization approaches to avoid local minima
    try:
        # First try scipy's differential evolution
        de_result = differential_evolution(
            objective, 
            bounds, 
            constraints=cons,
            maxiter=maxiter,
            popsize=15,
            seed=seed,
            callback=lambda x, convergence: None,
            atol=1e-6,
            rtol=1e-6
        )
        
        if de_result.success:
            h_de = de_result.x
            c5_de = -objective(h_de)
            if c5_de < best_c5:
                best_h = h_de
                best_c5 = c5_de
                
        # Try a local optimizer with multiple restarts to avoid local minima
        for i in range(5):
            if time.time() - start_time > budget_s * 0.9:
                break
            # Random restart
            x0 = np.random.uniform(0, 1, n_points)
            # Clip to bounds
            x0 = np.clip(x0, 0, 1)
            # Adjust to meet constraint
            integral = np.sum(x0) * dx
            x0 = x0 * (1.0 / integral if integral > 0 else 1.0)
            x0 = np.clip(x0, 0, 1)
            
            res = minimize(
                objective,
                x0,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 500},
                callback=lambda x: None
            )
            
            if res.success:
                h_local = res.x
                c5_local = -objective(h_local)
                if c5_local < best_c5:
                    best_h = h_local
                    best_c5 = c5_local
    except Exception as e:
        pass
    
    # If we still don't have a good solution, use the initial one
    if best_h is None:
        best_h = h0
        best_c5 = evaluate_erdos_solution(h0)
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    # Return best solution found
    return (list(best_h), final_c5, n_points)
