import numpy as np
import time
from scipy.optimize import linprog
import cvxpy as cp

def evaluate(h):
    n = len(h)
    dx = 2.0 / n
    h_pad = np.concatenate([h, np.zeros(n-1)])
    corr = np.correlate(h_pad, 1-h, mode='full')
    return np.max(corr) * dx

def project_onto_capped_simplex(v, S):
    # Project v onto {h | 0 <= h_i <= 1, sum h_i = S}
    v = np.array(v)
    low, high = -2.0, 2.0
    while high - low > 1e-10:
        mid = (low + high) / 2.0
        h = np.clip(v - mid, 0, 1)
        if np.sum(h) > S:
            low = mid
        else:
            high = mid
    return np.clip(v - (low + high) / 2.0, 0, 1)

def coeffs_and_const(h):
    n = len(h)
    # generate shift indices from -n+1 to n-1
    shifts = list(range(-(n-1), n))
    a_matrix = np.zeros((len(shifts), n))
    const_vec = np.zeros(len(shifts))
    for idx, k in enumerate(shifts):
        a_row = np.zeros(n)
        const_sum = 0.0
        for i in range(n):
            if 0 <= i+k < n:
                a_row[i] = 1.0 - h[i+k] - h[i-k]
                const_sum += h[i] * h[i+k]
            elif 0 <= i < n and 0 <= i+k < n:
                # on one side, zero padding
                a_row[i] = 1.0 - h[i+k] - h[i-k]
                const_sum += h[i] * h[i+k]
        a_matrix[idx] = a_row
        const_vec[idx] = const_sum
    return a_matrix, const_vec, shifts

def linear_program(a_matrix, const_vec, shifts, n, S):
    # Variables: h[0] ... h[n-1], t
    # Minimize t s.t.:
    #   for each k:
    #     a_matrix[k] @ h + const_vec[k] <= t
    #   sum(h) = S
    #   0 <= h_i <= 1, t free but >= 0

    # Reformulate LP:
    # minimize: t
    # subject to:
    #   [a_matrix] [h] + [const_vec] <= [t]    => [a_matrix, -1] [h] <= [const_vec]
    #   [1]      [h]   [S]                   =>  [1, 0]         [h] <= [ S]
    #   0 <= h[i] <= 1 for all i

    # Build coefficients of LP:
    # Number of constraints = len(shifts) + 1 (sum constraint)
    # Number of variables = n + 1 (t included)
    
    # We want constraints of form: A @ x <= b where x=[h, t]
    A = np.vstack([np.column_stack([a_matrix, -np.ones(len(shifts))]),
                    np.column_stack([np.ones(n), 0])])
    b = np.concatenate([const_vec, [S]])
    
    # Use cvxpy for robustness
    h_vars = cp.Variable(n)
    t = cp.Variable(1)
    x = cp.hstack([h_vars, t])
    
    constraints = []
    for i in range(len(shifts)):
        constraints.append(a_matrix[i] @ h_vars + const_vec[i] <= t)
    constraints.append(cp.sum(h_vars) == S)
    constraints.append(h_vars >= 0)
    constraints.append(h_vars <= 1)
    
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status in ["optimal", "optimal_inaccurate"]:
            sol_h = h_vars.value
            sol_t = t.value if t.value is not None else 0
        else:
            sol_h = np.array([0.0]*n)
            sol_t = float('inf')
    except:
        sol_h = np.array([0.0]*n)
        sol_t = float('inf')
    
    return sol_h, sol_t

def random_feasible(n):
    v = np.random.rand(n)
    v = project_onto_capped_simplex(v, n/2.0)
    return v
