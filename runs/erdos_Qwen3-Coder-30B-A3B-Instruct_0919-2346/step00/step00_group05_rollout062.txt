import numpy as np
import time
from scipy.optimize import linprog
import cvxpy as cp

def evaluate(h):
    n = len(h)
    dx = 2.0 / n
    # compute overlaps for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_onto_capped_simplex(v, S):
    # water-filling approach to project v onto {h | 0 <= h_i <= 1, sum h_i = S}
    v = np.asarray(v)
    v = np.clip(v, 0, 1)
    if abs(np.sum(v) - S) < 1e-10:
        return v.copy()

    # bisection on lambda to find right projection
    low, high = -2.0, 2.0
    tol = 1e-12
    while high - low > tol:
        mid = (low + high) / 2.0
        candidate = np.clip(v - mid, 0.0, 1.0)
        if np.sum(candidate) > S:
            low = mid
        else:
            high = mid
    res = np.clip(v - (low + high) / 2.0, 0.0, 1.0)
    return res

def coeffs_and_const(h):
    # compute a_{i,k} and c_k for all shifts k from -(n-1) to (n-1)
    n = len(h)
    a = np.zeros((n, 2 * n - 1))
    c = np.zeros(2 * n - 1)
    for k in range(-(n - 1), n):
        if k >= 0:
            # h_i * h_{i+k} for i=0 to n-1-k
            # h_{i-k} h_i
            idx_i = np.arange(n - k)
            idx_i_plus_k = np.arange(k, n)
            # a_{i,k} = 1 - h_{i+k} - h_{i-k}
            # expand the coefficients
            a[:, k + (n - 1)] = 1.0 - h[idx_i_plus_k] - h[idx_i]
            # c_k = sum h_i h_{i+k}
            c[k + (n - 1)] = np.sum(h[idx_i] * h[idx_i_plus_k])
        else:
            # k < 0, swap index: h_{i-k} h_i  where i goes from -k to n-1
            idx_i = np.arange(-k, n)
            idx_i_plus_k = np.arange(n + k)
            a[:, k + (n - 1)] = 1.0 - h[idx_i_plus_k] - h[idx_i]
            c[k + (n - 1)] = np.sum(h[idx_i] * h[idx_i_plus_k])
    return a, c

def solve_lp(a_matrix, const_vector, n, S):
    # solve: minimize t, subject to a_matrix @ h + const_vector <= t, sum(h) = S, 0 <= h <= 1
    # Form as LP in standard form

    # variables are h (n), t (1)
    # objective: minimize t
    # constraints: (a_matrix @ h + const_vector) <= t -> (-1) * a_matrix @ h + t >= -const_vector
    #              sum(h) = S
    #              0 <= h <= 1

    num_shifts = a_matrix.shape[1]
    c = np.zeros(n + 1)
    c[-1] = 1

    # inequality: -a_matrix @ h + t >= -const_vector
    A_ub = np.hstack([-a_matrix, np.ones((num_shifts, 1))])
    b_ub = -const_vector

    # equality: sum(h) = S
    A_eq = np.ones((1, n))
    A_eq = np.hstack([A_eq, np.zeros((1, 1))])  # add column for t
    b_eq = S

    # bounds on h
    bounds = [(0, 1)] * n + [None]  # t is free but should be ≥ 0 (non-neg since it's a bound)

    try:
        # If the solution is infeasible, add small slack
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        if result.success:
            h_sol = result.x[:-1]
            t_sol = result.x[-1]
            return h_sol, t_sol
        else:
            # fallback: just return previous h and a large t
            return None, 10000
    except:
        return None, 10000

def run(seed=42, budget_s=600, n_points=70):
    np.random.seed(seed)
    start_time = time.time()
    n = n_points
    S = n / 2.0
    dx = 2.0 / n

    # init from current record if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)  # already feasible
        h0 = h0 / np.sum(h0) * S
    else:
        h0 = np.random.rand(n)
        h0 = project_onto_capped_simplex(h0, S)

    best_h = h0.copy()
    best_C5 = evaluate(best_h)

    # inner iterations count - eventually we'll do many restarts
    restart_count = 0
    iter_count = 0

    while True:
        if time.time() - start_time > 0.95 * budget_s:
            break

        iter_count += 1
        restart_count += 1

        # slightly perturb previous best for diversity
        if restart_count > 1:
            h_start = best_h.copy()
            # add some noise
            noise = 0.02 * np.random.rand(n) - 0.01
            h_start = h_start + noise
            # project
            h_start = project_onto_capped_simplex(h_start, S)
        else:
            h_start = h0.copy()

        h = h_start.copy()
        found_improvement = False
        outer_iter = 0

        while True:
            if time.time() - start_time > 0.98 * budget_s:
                break

            outer_iter += 1
            if outer_iter > 100:  # hard cap
                break

            a, const = coeffs_and_const(h)
            h_new, t_new = solve_lp(a, const, n, S)
            if h_new is None:
                # fallback
                h_new = h.copy()
                t_new = 10000

            # Under-relaxation (damping)
            lambda_d = 0.8
            h = lambda_d * h_new + (1 - lambda_d) * h

            # project again
            h = project_onto_capped_simplex(h, S)

            # evaluate true C5
            c5_current = evaluate(h)
            if c5_current < best_C5:
                best_C5 = c5_current
                best_h = h.copy()
                found_improvement = True

            # check convergence
            if t_new < 1e-6 or outer_iter > 50:
                break

        # optional fine polishing: try to make moves to decrease overlap
        if found_improvement:
            for _ in range(200):  # small number of local moves
                if time.time() - start_time > 0.99 * budget_s:
                    break
                # move mass between two indices with small and large h to reduce overlap
                i = np.random.randint(0, n)
                k = np.random.randint(-n+1, n)
                j = i + k
                if 0 <= j < n and h[i] > 0.001 and h[j] < 0.999:
                    step = min(h[i], 0.999 - h[j], 0.01)
                    h[i] -= step
                    h[j] += step
                    h = project_onto_capped_simplex(h, S)
                    new_c5 = evaluate(h)
                    if new_c5 < best_C5:
                        best_C5 = new_c5
                        best_h = h.copy()

        # early stopping if time is running out
        if time.time() - start_time > 0.95 * budget_s:
            break

    return best_h, best_C5, n
