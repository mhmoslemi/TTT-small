            h[i] = max(0, 1 - abs(x - 1))
        # Normalize to integral = 1
        h = h * (1.0 / np.sum(h) * n_points * dx)

    h = np.clip(h, 0, 1)
    h = h * (n_points / 2.0) / np.sum(h)  # correct to sum = n_points/2
    
    best_h = h.copy()

    # Simulated Annealing parameters
    temp = 1.0
    min_temp = 1e-8
    alpha = 0.9995
    max_iter = 100000
    iter_count = 0
    
    def compute_c5(h_values):
        # Compute correlation values for all shifts using np.correlate
        corr = correlate(h_values, 1 - h_values, mode='full')
        # Multiply by dx to get continuous integral approximation
        dx = 2.0 / len(h_values)
        corr *= dx
        # C5 is max over all shift values (only those with positive correlation)
        return np.max(corr)
    
    # Optimization loop
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Modify two random indices
        i, j = np.random.choice(n_points, 2, replace=False)
        
        # Compute allowable delta
        max_delta = min(1 - h[i], h[j])
        if max_delta < 1e-8:
            iter_count += 1
            continue
            
        # Random delta with bounds
        delta = np.random.uniform(-max_delta, max_delta)
        
        # Compute new values keeping sum constant
        new_h = h.copy()
        new_h[i] = h[i] + delta
        new_h[j] = h[j] - delta
        
        # Ensure bounds
        new_h[i] = np.clip(new_h[i], 0, 1)
        new_h[j] = np.clip(new_h[j], 0, 1)
        
        # Renormalize to fix sum constraint
        sum_h = np.sum(new_h)
        new_h = new_h * (n_points / 2.0) / sum_h
        
        # Compute C5 for new values
        new_c5 = compute_c5(new_h)
        
        # Accept or reject based on SA criteria
        if new_c5 < best_c5 or np.random.rand() < np.exp(-(new_c5 - best_c5) / temp):

            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        
        # Decrease temperature
        temp = max(min_temp, temp * alpha)
        iter_count += 1
    
    # Final evaluation to ensure we have best
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, len(best_h))
