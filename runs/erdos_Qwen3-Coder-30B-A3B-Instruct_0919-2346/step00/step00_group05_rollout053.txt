import numpy as np
import time
import cvxpy as cp
from scipy.optimize import linprog

def evaluate(h):
    """Returns C5 = max_k integral h(x)(1-h(x+k)) dx."""
    n = len(h)
    dx = 2.0 / n
    # Compute all overlaps using correlation
    conv = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is scaled by dx
    max_overlap = np.max(conv) * dx
    return max_overlap

def project_onto_capped_simplex(v, S):
    """Project vector v onto [0,1]^n with sum = S."""
    def eval_sum(lam):
        h = np.clip(v - lam, 0, 1)
        return np.sum(h) - S
    
    # Binary search for lambda
    low, high = -2.0, 2.0
    tol = 1e-12
    while high - low > tol:
        mid = (low + high) / 2
        if eval_sum(mid) > 0:
            low = mid
        else:
            high = mid
    
    lam = (low + high) / 2
    h = np.clip(v - lam, 0, 1)
    return h

def coeffs_and_const(h):
    """Compute coefficient matrix and constants for all shifts."""
    n = len(h)
    # Precompute indices for shift lookups
    shifts = list(range(-(n - 1), n))  # k from -(n-1) to n-1
    # Preallocate matrices
    a_matrix = np.zeros((len(shifts), n))
    const_vec = np.zeros(len(shifts))
    
    for idx, k in enumerate(shifts):
        # Compute the coefficient vector for shift k
        a_row = np.zeros(n)
        const = 0.0
        for i in range(n):
            # Kirchhoff law: h_i * h_{i+k}
            if (i + k) >= 0 and (i + k) < n:
                a_row[i] = 1 - h[i + k] - h[i - k if i - k >= 0 else i - k + n]
                const += h[i] * h[i + k]
            else:
                # Zero padded case -> nothing to add to a_row or const
                a_row[i] = 1 - h[i - k if i - k >= 0 else i - k + n] if (i - k) % n < n else 1.0
        a_matrix[idx] = a_row
        const_vec[idx] = const
    return a_matrix, const_vec, shifts

def linear_program(a_matrix, const_vec, n, S, M=1e4):
    """Solve the linear program for the next iterate."""
    # Variables: h[0..n-1] + t (scalar)
    # t: auxiliary variable representing the worst-case overlap
    h_vars = cp.Variable(n)
    t = cp.Variable()

    # Objective: minimize t
    objective = cp.Minimize(t)

    # Constraints
    constraints = []
    
    # For each shift k (from the matrix)
    for i in range(len(const_vec)):
        a_k = a_matrix[i]
        c_k = const_vec[i]
        # a_k @ h + c_k <= t
        constraints.append(a_k @ h_vars + c_k <= t)
    
    # Mass constraint
    constraints.append(cp.sum(h_vars) == S)
    
    # Bounds
    constraints.append(h_vars >= 0)
    constraints.append(h_vars <= 1)
    constraints.append(t >= 0)

    # Form and solve the problem
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status in ["optimal", "optimal_inaccurate"]:
            h_new = h_vars.value
            t_new = t.value
            return h_new, t_new
    except:
        pass
    # Fallback to a weak solution or use alternative method
    return None, None

def random_feasible(n):
    """Generate a pseudo-random feasible vector."""
    v = np.random.rand(n)
    # Project onto capped simplex with sum = n/2
    v = project_onto_capped_simplex(v, n / 2)
    return v
