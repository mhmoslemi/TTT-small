import numpy as np
import time
import random

def evaluate_c5(h):
    """Evaluate the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h) using full mode
    corr = np.correlate(h, 1 - h, mode='full')
    # Take maximum value and multiply by dx
    max_overlap = np.max(corr)
    return max_overlap * dx

def random_binary_seq(n):
    """Generate a random binary sequence with exactly n/2 ones"""
    h = np.zeros(n)
    # Choose exactly n/2 positions to set to 1
    ones_indices = np.random.choice(n, n//2, replace=False)
    h[ones_indices] = 1
    return h

def solve_binary(n, iters, init_h=None, seed=None):
    """Perform simulated annealing on binary vectors"""
    rng = np.random.default_rng(seed)
    if init_h is not None:
        h = init_h.copy()
    else:
        h = random_binary_seq(n)
    
    c5 = evaluate_c5(h)
    best_h, best_c5 = h.copy(), c5
    
    # Precompute sets of indices for efficiency
    ones = set(np.where(h > 0.5)[0])
    zeros = set(np.where(h < 0.5)[0])
    
    # Simulated annealing parameters
    T0 = 0.1
    T = T0
    cooling = 0.99995
    
    for iteration in range(iters):
        # Choose random swap
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = rng.choice(list(ones))
        j = rng.choice(list(zeros))
        
        # Propose swap
        h[i], h[j] = h[j], h[i]
        
        # Update sets
        ones.remove(i)
        zeros.remove(j)
        ones.add(j)
        zeros.add(i)
        
        # Compute new c5
        new_c5 = evaluate_c5(h)
        delta = new_c5 - c5
        
        # Accept or reject according to Metropolis criterion
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            # Accept
            c5 = new_c5
            if c5 < best_c5:
                best_c5, best_h = c5, h.copy()
        else:
            # Reject
            h[i], h[j] = h[j], h[i]
            # Restore sets
            ones.remove(j)
            zeros.remove(i)
            ones.add(i)
            zeros.add(j)
        
        # Cooling schedule
        T *= cooling
        
    return best_h, best_c5
