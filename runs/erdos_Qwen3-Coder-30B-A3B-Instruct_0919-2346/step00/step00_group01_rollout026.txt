import numpy as np
import cvxpy as cp
import time
from scipy.optimize import minimize
from scipy.fftpack import fft, ifft
from math import floor, ceil

# Global constants
MAX_N_SDP = 80
MAX_N_INTERVAL = 200
NUM_SDP_RUNS = 30
NUM_ROUNDING_ATTEMPTS_PER_SDP = 10
NUM_INTERVAL_SEARCHES = 50000

def evaluate_c5(h):
    """Return C5 = max_k int h(x)(1 - h(x+k)) dx."""
    n = len(h)
    # Use circular correlation for the overlap integral
    h_full = np.concatenate([h, h])
    h_corr = np.correlate(h_full, 1 - h, mode='full')
    # Extract the part that corresponds to shifts k = 0 to n-1 (positive shifts given by np.correlate)
    overlap_vals = h_corr[n-1:2*n-1]
    return 1.0 - (2.0 / n) * np.min(overlap_vals)

def circular_correlation(h):
    """Compute circular correlation with 1-h using FFT."""
    n = len(h)
    h_pad = np.pad(h, (0, n), 'constant', constant_values=(0, 0))
    h_comp_pad = np.pad(1 - h, (0, n), 'constant', constant_values=(0, 0))
    corr = ifft(fft(h_pad) * np.conj(fft(h_comp_pad))).real
    return corr[n-1:n-1+n]

def compute_autocorrelations(h):
    n = len(h)
    ac = np.zeros(n)
    h_full = np.concatenate([h, h])
    for k in range(n):
        ac[k] = np.sum(h * np.roll(h_full[:n], k)[:n])
    return ac

def projection_sum_1(h, target_sum=0.5):
    """Project h to have sum = target_sum by clipping and rescaling."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    if current_sum > target_sum:
        # Reduce values
        sorted_h = np.sort(h)[::-1]
        cumsum = np.cumsum(sorted_h)
        i = np.argmax(cumsum > target_sum)
        if i == 0:
            # All elements to be reduced to a common value
            val = target_sum / len(h)
            h = np.full_like(h, val)
        else:
            # Only reduce the first i elements
            threshold = (target_sum - cumsum[i-1]) / (len(h) - i)
            h = np.clip(h, 0, threshold)
    else:
        # Increase values (this is unlikely with our setup)
        raise RuntimeError("Sum too small - need to increase values above 1")
    return h

def sdp_solve(n, verbose=False):
    """Solve SDP relaxation for the problem."""
    # Define matrices
    X = cp.Variable((n, n), symmetric=True)
    h = cp.Variable(n)
    
    # Build the constraint matrices M_k for each shift
    M = []
    for k in range(n):
        Mk = np.zeros((n, n))
        for i in range(n):
            j1 = (i + k) % n
            j2 = (i - k) % n
            Mk[i, j1] = 1.0
            Mk[i, j2] = 1.0
        M.append(Mk)
    
    # Objective
    t = cp.Variable()
    
    # Constraints
    constraints = [
        X >> 0,  # X is positive semidefinite
        cp.diag(X) == h,  # Diagonal equals h
        0 <= h,  # Non-negativity
        h <= 1,  # Upper bound on h
        cp.sum(h) == n / 2,  # Sum constraint
    ]
    for k in range(n):
        constraints.append(cp.trace(M[k] @ X) >= t)
    
    # Problem and solve
    prob = cp.Problem(cp.Maximize(t), constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=verbose, feastol=1e-6, reltol=1e-6)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            X_val = X.value
            h_val = h.value
            t_val = t.value
            if not np.isfinite(t_val):
                raise RuntimeError("Infinite or NaN SDP value")
            return X_val, h_val, t_val
    except Exception as e:
        if verbose:
            print(f"SDP failed: {e}")
    return None, None, None

def round_h(X_val, h_val, n, target_sum=0.5):
    """Sample a binary vector from SDP rank-1 approximation."""
    # Use Cholesky decomposition to sample
    try:
        V = np.linalg.cholesky(X_val)
        nans = np.isnan(V)
        if np.any(nans):
            # Use eigen-decomposition
            evals, evecs = np.linalg.eigh(X_val)
            # Take only positive eigenvalues
            V = evecs @ np.diag(np.sqrt(np.maximum(evals, 0)))
            V = V.T
    except np.linalg.LinAlgError:
        # Fallback to eigenvalue decomposition
        evals, evecs = np.linalg.eigh(X_val)
        V = evecs @ np.diag(np.sqrt(np.maximum(evals, 0)))
        V = V.T

    # Sample
    g = np.random.randn(V.shape[0])
    
    # Compute scores
    scores = V @ g
    
    # Determine threshold
    sorted_scores = np.sort(scores)[::-1]
    # Want target_sum = n/2 ones
    threshold = sorted_scores[floor(n/2) - 1]
    # Adjust threshold if needed
    num_ones = np.sum(scores >= threshold)
    if num_ones > floor(n/2):
        # Adjust upward
        for i in range(len(sorted_scores), 0, -1):
            if sorted_scores[i-1] > threshold:
                threshold = sorted_scores[i-1]
            else:
                break
    elif num_ones < floor(n/2):
        # Adjust downward
        for i in range(len(sorted_scores)-1, -1, -1):
            if sorted_scores[i] < threshold:
                threshold = sorted_scores[i]
            else:
                break
    
    # Assemble binary vector
    h_bin = (scores >= threshold).astype(float)
    
    # Ensure exactly floor(n/2) ones
    current_count = np.sum(h_bin)
    if current_count != floor(n/2):
        # Adjust by flipping some values from 0 to 1 or 1 to 0
        if current_count > floor(n/2):
            # Too many 1s
            flip_indices = np.where(h_bin == 1)[0]
            to_flip = int(current_count - floor(n/2))
            should_flip = np.random.choice(flip_indices, to_flip, replace=False)
            h_bin[should_flip] = 0
        else:
            # Not enough 1s
            flip_indices = np.where(h_bin == 0)[0]
            to_flip = int(floor(n/2) - current_count)
            should_flip = np.random.choice(flip_indices, to_flip, replace=False)
            h_bin[should_flip] = 1
            
    # Project onto feasible set
    h_bin = projection_sum_1(h_bin, target_sum)
    return h_bin

def swap_improve(h, n):
    """Improve solution via pairwise swaps."""
    # Make a copy for improvement
    h_new = h.copy()
    # Try swapping 0s and 1s and compute new overlap values
    n_swap_attempts = 100 * n
    max_improvement = 0
    iter_count = 0
    has_improved = True
    while has_improved and iter_count < n_swap_attempts:
        has_improved = False
        iter_count += 1
        # Sample two locations i, j to swap
        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        if i != j:
            # Get current autocorrelations
            # For efficiency, compute using the incremental method
            # Compare current overlap with swapped version
            # A rough proxy for the change in minimum correlation is the change in overlap
            # of the worst shift
            
            # If the value at i is 1, and at j is 0, we can swap
            if h_new[i] > 0.5 and h_new[j] < 0.5:
                old_val = np.sum(h_new * np.roll(h_new, j-i))
                h_new[i], h_new[j] = h_new[j], h_new[i]
                new_val = np.sum(h_new * np.roll(h_new, j-i))
                if new_val < old_val:
                    # Accept swap if it decreases worst overlap
                    has_improved = True
                else:
                    # Revert swap
                    h_new[i], h_new[j] = h_new[j], h_new[i]
    return h_new

def gradient_fine_tune(h, n, max_iter=200):
    """Fine-tune with projected gradient ascent on the extreme value."""
    h_new = h.copy()
    lr = 0.01
    for _ in range(max_iter):
        # Compute all correlations
        corr = circular_correlation(h_new)
        min_k = np.argmin(corr)
        min_val = corr[min_k]
        # Compute gradient of the minimal overlap w.r.t. h
        grad = np.zeros(n)
        k_min = min_k
        for i in range(n):
            j1 = (i + k_min) % n
            j2 = (i - k_min) % n
            grad[i] = h_new[j1] + h_new[j2]
        # Update h
        h_new = h_new + lr * grad
        # Project back to [0,1] and enforce sum = 0.5
        h_new = projection_sum_1(h_new, 0.5)
        # Bound gradient ascent
        lr *= 0.99
    return h_new
