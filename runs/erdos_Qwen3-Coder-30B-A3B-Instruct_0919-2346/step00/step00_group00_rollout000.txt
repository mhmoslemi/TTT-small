import numpy as np
import time
import random

def compute_c5(h):
    """Compute C5 bound from step function h using np.correlate"""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return corr.max()

def propose_swap_and_update(h, corr, i, j, new_corr):
    """Update correlation array and h values when swapping h[i] and h[j]"""
    old_i = h[i]
    old_j = h[j]
    new_i = old_j
    new_j = old_i
    
    delta_i = new_i - old_i  # either +1 or -1

    n = len(h)
    new_corr[:] = corr[:]  # copy old correlation

    # Loop over k not equal to i or j
    for k in range(n):
        if k == i or k == j:
            continue
        hk = h[k]
        # contributions for shift i->k: l = k - i
        idx1 = k - i + n - 1
        new_corr[idx1] += delta_i * (1 - hk)
        # contributions for shift k->i: l = i - k
        idx2 = i - k + n - 1
        new_corr[idx2] += hk * (-delta_i)

        # contributions for shift j->k: l = k - j
        idx3 = k - j + n - 1
        new_corr[idx3] += -delta_i * (1 - hk)
        # contributions for shift k->j: l = j - k
        idx4 = j - k + n - 1
        new_corr[idx4] += hk * (delta_i)

    # handle cross pairs (i,j) and (j,i)
    idx_ij = j - i + n - 1
    new_corr[idx_ij] += delta_i
    idx_ji = i - j + n - 1
    new_corr[idx_ji] -= delta_i

    # swap values in h
    h[i], h[j] = new_j, new_i
    
    return new_corr

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    best_c5 = np.inf
    best_h = None
    best_n = None
    
    # Roll with a single large n for no restarts
    n = 140  # Even, 70 ones
    if n % 2 != 0:
        n += 1  # Make even

    # Initial h: random binary with sum = n//2
    h = np.zeros(n, dtype=float)
    ones = n // 2
    h[np.random.choice(n, ones, replace=False)] = 1.0

    # compute initial correlation
    corr = np.correlate(h, 1 - h, mode='full')
    cur_max = corr.max()
    cur_c5 = cur_max * (2.0 / n)
    cur_h = h.copy()
    
    # For temperature-based simulated annealing
    temp = 0.02
    temp_decay = 0.9995  # cooling rate

    max_iters = 60000  # Aim for ~100k iterations but pace with time
    timeout_check_interval = 1000

    for it in range(max_iters):
        now = time.time()
        if now - start > budget_s - 0.5:
            break
            
        # Sample two random distinct indices
        i = np.random.randint(n)
        j = np.random.randint(n)
        while h[i] == h[j]:
            i = np.random.randint(n)
            j = np.random.randint(n)

        # Compute new correlation and related values
        new_corr = np.empty_like(corr)
        new_corr = propose_swap_and_update(h, corr, i, j, new_corr)
        new_max = new_corr.max()
        new_c5 = new_max * (2.0 / n)
        
        # Accept or reject swap
        delta = new_c5 - cur_c5

        if delta < 0 or np.random.rand() < np.exp(-delta / (temp + 1e-12)):
            corr[:] = new_corr
            cur_c5 = new_c5
            cur_max = new_max
        else:
            # Revert h to original state
            h[i], h[j] = h[j], h[i]
        
        # Update best solution
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = cur_h.copy()
            best_n = n
            
        # Temperature update
        temp *= temp_decay
        
        # Occasionally check time
        if it % timeout_check_interval == 0:
            if time.time() - start > budget_s - 0.5:
                break

    # Fallback in case no solution was found
    if best_h is None:
        try:
            from __main__ import initial_h_values
        except Exception:
            # Minimal construction
            n = 2
            best_h = np.array([1.0, 0.0])
            best_n = 2
            best_c5 = compute_c5(best_h)
        else:
            best_h = initial_h_values.copy()
            best_n = len(best_h)
            best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
