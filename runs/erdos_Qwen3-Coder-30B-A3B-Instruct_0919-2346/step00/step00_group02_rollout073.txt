import numpy as np
import time

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx using discrete correlation."""
    dx = 2.0 / len(h)
    # Compute overlap integral for all shifts k
    overlap = np.correlate(h, 1 - h, mode="full")
    # Multiply by dx to get actual integral values
    overlap *= dx
    # C5 is the maximum of these values
    return np.max(overlap)

def propose_move(h, max_step):
    """Generate a new feasible vector by adjusting two elements."""
    h_new = h.copy()
    N = len(h)
    i, j = np.random.choice(N, size=2, replace=False)
    
    # Compute feasible range for delta
    delta_low = max(-h_new[i], h_new[j] - 1)
    delta_high = min(1 - h_new[i], h_new[j])
    
    if delta_low >= delta_high:
        # Cannot make a move with these indices, return original
        return h_new
    
    # Clip max_step to feasible range
    step = min(max_step, delta_high - delta_low)
    step = max(step, 0)
    if step <= 0:
        return h_new
        
    # Sample delta in feasible interval
    delta = np.random.uniform(low=delta_low, high=delta_high)
    h_new[i] += delta
    h_new[j] -= delta
    
    return h_new

def anneal(initial_h, time_budget, seed=None):
    """Run simulated annealing from initial_h for time_budget seconds."""
    if seed is not None:
        np.random.seed(seed)
    
    cur = initial_h.copy()
    best = initial_h.copy()
    c_cur = c5_of(cur)
    c_best = c_cur
    
    start = time.time()
    
    # Parameters for annealing
    T_start = 0.05
    T_end = 1e-6
    step_start = 0.2
    
    while time.time() - start < time_budget:
        # Compute current time fraction
        elapsed = time.time() - start
        t_frac = elapsed / time_budget
        
        # Update temperature (exponential decay)
        T = T_start * (T_end / T_start) ** t_frac
        
        # Update step size linearly
        step = step_start * (1 - t_frac)
        
        # Generate proposal
        h_new = propose_move(cur, step)
        c_new = c5_of(h_new)
        
        # Accept or reject
        if c_new <= c_cur:
            cur = h_new
            c_cur = c_new
            if c_new < c_best:
                best = h_new
                c_best = c_new
        else:
            prob = np.exp(-(c_new - c_cur) / T)
            if np.random.random() < prob:
                cur = h_new
                c_cur = c_new
                if c_new < c_best:
                    best = h_new
                    c_best = c_new

    return best, c_best
