def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)

    correlation = np.correlate(h, 1 - h, mode="full")

    # For k in [0,2], the corresponding correlation indices are from n-1 to n+1
    max_correlation = np.max(correlation[n-1:n+2])
