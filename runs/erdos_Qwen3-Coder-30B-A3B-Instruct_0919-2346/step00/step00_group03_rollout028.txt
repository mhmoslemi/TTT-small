import numpy as np
from scipy.optimize import differential_evolution
import time
from cvxpy import *
import cvxpy as cp
from scipy.signal import fftconvolve

def initial_h_to_n(n):
    """Interpolate initial_h_values to length n"""
    h0 = np.array(initial_h_values)
    x0 = np.linspace(0, 2, len(h0))
    x_new = np.linspace(0, 2, n)
    return np.interp(x_new, x0, h0)

def project_to_simplex(h, target_sum):
    """Project h onto the simplex with sum = target_sum using Dykstra's method"""
    h = np.array(h)
    current_sum = np.sum(h)
    delta = (current_sum - target_sum) / len(h)
    h = h - delta
    # Dykstra's projection: clip and re-scale
    for _ in range(5):  # few iterations suffices
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        delta = (current_sum - target_sum) / len(h)
        h = h - delta
    return h

def evaluate_overlap(h, dx):
    """Compute C5 using FFT for speed"""
    h = np.array(h)
    # Ensure h is within [0,1]
    h = np.clip(h, 0, 1)
    # Using FFT correlation (full mode)
    corr = fftconvolve(h, np.flip(1 - h), mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 180
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Start with the initial construction but interpolate to desired n
    h = initial_h_to_n(n_points)
    h = project_to_simplex(h, target_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Differential Evolution optimization
    def objective(h_vec):
        h_vec = project_to_simplex(h_vec, target_sum)
        return evaluate_overlap(h_vec, dx)
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # DE parameters
    popsize = 30
    maxiter = 10000
    maxtime = budget_s - 100  # leave 100s for final refinement
    
    # Run DE to improve initial solution
    result = differential_evolution(
        objective,
        bounds,
        maxiter=maxiter,
        popsize=popsize,
        seed=seed,
        disp=False,
        timeout=maxtime,
        strategy='best1bin',
        atol=1e-6,
    )
    
    if not result.success:
        # If DE didn't converge, fallback to current best.
        pass
    else:
        # Incorporate result if better
        h_try = project_to_simplex(result.x, target_sum)
        c5_try = evaluate_overlap(h_try, dx)
        if c5_try < best_c5:
            best_h = h_try
            best_c5 = c5_try

    # Local refinement: mismatch-swap
    max_swaps = 5000
    swap_count = 0
    
    def perform_swap(h_vec, shift):
        # Identify mismatched positions
        corr = fftconvolve(h_vec, np.flip(1 - h_vec), mode='full')
        max_corr_idx = np.argmax(corr)
        max_corr = corr[max_corr_idx]
        
        # The shift corresponds to the maximum overlap index
        shift = max_corr_idx - (n_points - 1)
        start_idx = max(0, -shift)
        end_idx = min(n_points, n_points - shift)
        
        M_1 = [i for i in range(start_idx, end_idx) if h_vec[i] >= 0.99 and h_vec[i + shift] <= 0.01]
        M_0 = [i for i in range(start_idx, end_idx) if h_vec[i] <= 0.01 and h_vec[i + shift] >= 0.99]
        
        if not M_1 or not M_0:
            return h_vec, max_corr
            
        # Pick a uniform random swap
        i = np.random.choice(M_1)
        j = np.random.choice(M_0)
        
        # Perform swap
        h_new = h_vec.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]
        
        # Re-project to simplex
        h_new = project_to_simplex(h_new, target_sum)
        return h_new, max_corr
    
    h_local = best_h.copy()
    while swap_count < max_swaps:
        c5_before = evaluate_overlap(h_local, dx)
        if c5_before < (0.3807 + 1e-5):
            break
        h_local, corr = perform_swap(h_local, 0)
        c5_after = evaluate_overlap(h_local, dx)
        if c5_after >= c5_before:
            # Undo swap
            h_local = best_h.copy()
        else:
            # Keep change
            best_h = h_local.copy()
            best_c5 = c5_after
        swap_count += 1
    
    # Final check
    final_c5 = evaluate_overlap(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
