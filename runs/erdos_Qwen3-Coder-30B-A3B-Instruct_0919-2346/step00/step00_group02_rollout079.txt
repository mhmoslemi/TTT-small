import numpy as np
from scipy.optimize import minimize
import time
from math import floor

def evaluate_c5(h):
    """Evaluate C5 for a given step function h"""
    # h is an array of length n_points
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the correlation of h with (1-h) using full mode
    # Using numpy's correlate with mode='full' gives us the full correlation
    # But we want the maximum correlation value, so let's compute manually
    # We pad one array with zeros on both sides as with convolution
    # Actually, we compute the integral of h(x)(1-h(x+k)) for k=0 to 2
    # This is a discrete approximation. We only care about correlation 
    # for shift k ∈ [0,2], so we look at correlation for all possible shifted versions
    # Full correlation will have length 2*n_points - 1
    # We need the maximum value of integral of h(x)(1-h(x+k)) which is 
    # the same as full correlation over the domain [0, 2]
    # So we do the evaluation manually or use numpy convolve correctly.
    
    # Convolution of h with (1-h) with full mode gives us all lags
    # We need the maximum value of this, scaled by dx
    h1 = 1 - h  # vectorized
    # Use full mode
    corr = np.correlate(h, h1, mode='full')  # lengths: n_points + (n_points - 1) = 2*n_points - 1
    # The elements of corr index shifts from -n_points+1 to +n_points-1
    # For our problem, we are interested in k from 0 to 2
    # k = 0,1,2 mapped to indices/order 0,1,2 of overlapping parts of h and (1-h)
    # But we only need to map k values from 0 to 2 to appropriate indices in corr
    
    # Actually let's reformulate correctly:
    # let corr[i] be integral of h(x) * (1-h(x + shift)) where shift is the corresponding lag
    # Our domain is [0,2], and we have step function of n_points over that
    # dx = 2 / n_points 
    # evaluation for shift = k* dx where k in range [0, 2/dx * n_points] ->  [0, 2 * n_points] 
    # The full correlation takes k from -n_points+1 to n_points-1 (stop at 0 for k >= 0)
    # We want max over k ∈ [0,2] => k ∈ [0, 2*dx * n_points] => we take corr[range(n_points, 2*n_points)] 
    # But shift k goes from 0 to 2 (in dx increments).
    # shift lengths we're interested in: [0*dx, dx, 2*dx], ..., up to (2/dx)*dx = 2
    # That is, shifts_index = [0, 1, ... , 2/dx] in integer steps if 2/dx is integer 
    # The actual shift values (k) are integers idx*dx multiply (0 to 2/dx) -> 0 to 2
    
    # Actually we can do this directly:
    # The correlations for 0 <= shift < 2 are those from the center with len = 2*n_points - 1
    # The middle element of corr is when shift = 0, [0, n_points-1, 2*n_points-2]
    # For shift = k*dx, we want the shift index in corr.
    # shift = 0 => corr index = n_points - 1 (middle)
    # shift = 1*dx => corr index = n_points + 0 = n_points 
    # shift = 2*dx => corr index = n_points + 1 
    # shift = 3*dx => corr index = n_points + 2 
    # ...
    # For contains all such shifts, up to about 2/dx or so, we look at corr[n_points:]. 
    # But for each valid k from 0 to 2, corresponding to increment 0 to 2*(n_points/2) = n_points 
    # The last term (shift = 2) is when shift index = 2n_points - 1  
    # So indexing from full corr:
    # It seems more direct to just compute the integral for shift k from 0 to 2 with step 2/n_points
    
    # So accuracy: iterate for k = 0, dx, 2*dx, ..., 2 
    dx = 2.0 / n_points
    max_c = 0
    for i in range(n_points + 1):
        shift = i * dx
        if shift > 2: 
            break 
        # For shift k, we calculate integral h(x) * (1 - h(x+k)) 
        # which is sum over j: h[j] * (1 - h[j+k]) * dx 
        # But we must make sure that j+k remains valid, 
        # i.e., j+k < n_points, so k < n_points-j
        corr_at_shift = 0.0
        k_int = int(round(shift / dx)) 
        # corr index = n_points - 1 + k_int 
        # valid for k_int = 0 to n_points - 1 ( e.g., shift from 0 to 2) - shifts for shift >=0
        # so valid indices are n_points-1 to 2*n_points-2, +1 for shift=0 to n_points-1  
        # actually:
        # shift = k*dx, k integer = i
        # depending on k: 0 to 2*dx*points == 2*points
        # and we are concerned with shift k iterating = 0*2/n_points, 1*2/n_points, ... 2*2/n_points 
        # and integer k = 0,1,..., (2 / dx) = 2 * n_points / 2 = n_points 
        # For shift 0 we look at index form where lag is 0 -> middle of corr = n_points-1  
        # For shift dx* 1 = 1*dx -> index = lag = -1, so we look at n_points-1-1 = n_points-2  
        # Wait, its mirror, 0 shift -> corr index = n_points-1  
        # Negative shift -> we want a correlation starting from right (at some x) 
        # so for shift 0, h(0)(1-h(0)) + h(1)(1-h(1)) + ... 
        # For shift 1*dx = 1*2/n_points, h(0)(1-h(1)) + h(1)(1-h(2)) + ...  
        # Turns out that numpy's correlation definition matches our miscalculation
        # So we do:
        # Take full correlation (0 to n_points - 1) gives all index that match our desired 0 to 2
        # We note that 0 to 2/space increases a bit so we take the relevant part. 
        # This is confusing - let's write cleaner code:
        if i <= n_points - 1:  # the first half of the full correlation (this is when shift is 0, dx, ..., 1*dx)
            # corr of h vs. 1-h with left shift = i*dx
            # e.g. shift = 0, index = n_points - 1  
            # shift = 1*dx => index = n_points - 1 + 1 = n_points 
            corr_idx = n_points - 1 + i  
            corr_val = corr[corr_idx] * dx
        else:  # else i = n_points, n_points+1, ..., 2*n_points-2  
            # This should be the case when shifts from -1*dx to - (n_points-1)*dx etc. 
            # that is for i = n_points+i where i=0 to n_points-1 
            # shifting h to the right so corr layout 
            # corr[0] is shift = -n_points+1  
            # for negative shifts, i ranges from n_points to 2*n_points-2  
            corr_idx = 2 * n_points - 2 - i
            corr_val = corr[corr_idx] * dx
        # We only care about shifts 0, 1*dx, 2*dx, ..., up to 2, corresponding to some integer k from 0 to 2/dx 
        # that are all zero or positive  
        if shift <= 2:
            # shift from 0 to 2, positive shifts
            max_c = max(max_c, corr_val)
    return max_c

def objective(h):
    """Objective function: -C5 (since we want to minimize it)"""
    return -evaluate_c5(h)

def optimize_step_function(seed=42, budget_s=600):
    """Optimization algorithm for step function h"""
    np.random.seed(seed)
    start = time.time()
    
    # Start from initial construction
    # Assume initial_h_values is a list of n_points points (45 or something)  
    # Use the provided initial_h_values (but if not defined assume we have it via global import)
    try:
        h_prev = np.array(initial_h_values)  # imported globally
    except:
        # fallback to a linear approximation:
        n_points = 50
        h_prev = np.ones(n_points) * 0.5

    n_points = len(h_prev)

    # State for optimization
    best_h = h_prev.copy()
    best_c5 = float('inf')

    # Consistent sampling
    dx = 2.0 / n_points

    # Define constraints
    def constraint_eq(x):
        return np.sum(x) * dx - 1.0

    def constraint_bounds(x):
        return np.concatenate((x, 1 - x))  # x >= 0 and x <= 1

    # Bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]

    # Vanilla gradient-based optimizations for only a few steps, then local search
    def eval_func(h):
        return evaluate_c5(h)

    # SPSA or Nelder-Mead based local search
    # Let's try Nelder-Mead on a subsample of parameters (in case local search is too slow) 
    # Or perturb whole function and check - better idea 
    # Simple hill climbers with line-search-like steps
    
    # perturb 
    def hill_climb_step(h):
        h_new = h.copy()
        # make very small change to one or few elements to find a nearby min
        for i in range(10):  # try a few random changes
            idx = np.random.randint(0, n_points)
            delta = np.random.normal(0, 0.01)
            h_new[idx] = max(0, min(1, h_new[idx] + delta))
        return h_new
    
    t_now = time.time()
    iters = 0

    # Hoisted version of hill_climb_step so it can be used inline
    def get_hill_climb_step(h):
        h_new = h.copy()
        i = np.random.randint(0, n_points)
        delta = np.random.normal(0, 0.02)  
        h_new[i] = max(0, min(1, h_new[i] + delta))
        return h_new

    # Start the local hill climb-type optimization from current  

    while (time.time() - start) < budget_s * 0.95:
        if time.time() - start > budget_s * 0.95:
            break
            
        # Perturb in small random steps
        h_trial = get_hill_climb_step(best_h)
        
        # Apply sum constraint to ensure integral = 1
        total = np.sum(h_trial) * dx
        if total > 0:
            h_trial = h_trial * (1.0 / total)
            
        c5_trial = eval_func(h_trial)
        iters += 1
        
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h_trial.copy()
            # If we're doing much better, keep going faster
            if best_c5 < 0.3795:
                continue  # early termination if goal reached

    return (best_h, best_c5, n_points)

def run(seed=42, budget_s=600, **kwargs):
    """Main entrypoint"""
    h, c5_val, n_points = optimize_step_function(seed, budget_s)
    return (h.tolist(), c5_val, n_points)
