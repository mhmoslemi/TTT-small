import numpy as np
import cvxpy as cp
from scipy.signal import correlate
import time
import random
from math import floor

def project_to_capped_simplex(h, target_sum):
    # Project h to the capped simplex: 0 <= h[i] <= 1, sum(h) = target_sum
    h = np.asarray(h)
    n = len(h)
    h_sorted = np.sort(h)[::-1]
    
    # Find the projection using the standard simplex projection algorithm
    # adapted for the capped case
    temp = h_sorted.copy()
    temp = temp * np.arange(1, n+1) - (np.cumsum(h_sorted) - target_sum)
    idx = np.where(temp >= 0)[0]
    if len(idx) == 0:
        # All are zero
        return np.zeros_like(h)
    
    # We want the last qualifying index
    idx_max = idx[-1]
    theta = (np.sum(h_sorted[:idx_max+1]) - target_sum) / (idx_max + 1)
    
    projected_h = np.maximum(0, h - theta)
    # Make sure it sums to target_sum
    diff = target_sum - np.sum(projected_h)
    if abs(diff) > 1e-10:
        # Distribute the difference
        idx_nonzero = np.where(projected_h > 0)[0]
        to_add = diff / len(idx_nonzero)
        projected_h[idx_nonzero] += to_add
    return projected_h

def evaluate_solution(h_values):
    h = np.array(h_values)
    n = len(h)
    # Compute the overlap integral
    c = correlate(h, 1 - h, mode='full')
    return np.max(c) * (2.0 / n)

def solve_milp(n, max_seconds, initial_h=None):
    # Create CVXPY variables
    h = cp.Variable(n, name='h')
    tau = cp.Variable(name='tau')

    # Constraints
    constraints = [0 <= h, h <= 1, cp.sum(h) == n/2, h[0] == 1]

    # Product variables for y[i,k] = h[i] * h[i+k]
    # For each shift k from 1 to n-1, we define y_k as a vector of length n-k
    y_list = []
    for k in range(1, n):
        y_k = cp.Variable(n - k, name=f'y_{k}')
        y_list.append((k, y_k))

    # Add McCormick envelope constraints
    for k, y_k in y_list:
        for j in range(n - k):
            # y_k[j] >= h[j] + h[j+k] - 1
            constraints.append(y_k[j] >= h[j] + h[j+k] - 1)
            # y_k[j] <= h[j]
            constraints.append(y_k[j] <= h[j])
            # y_k[j] <= h[j+k]
            constraints.append(y_k[j] <= h[j+k])
            # y_k[j] >= 0
            constraints.append(y_k[j] >= 0)
        # tau <= sum(y_k)
        constraints.append(tau <= cp.sum(y_k))

    # No need for tau <= sum(h) constraint as it is redundant due to the mass constraint.
    
    # Objective
    objective = cp.Maximize(tau)
    
    prob = cp.Problem(objective, constraints)
    
    # Solver settings
    # Using the fixed solver set: the one that would be pre-loaded in the environment
    solver = cp.CBC

    # Set time limit
    prob.solve(solver=solver, verbose=False, max_seconds=max_seconds)
    
    # Extract the solution
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        h_val = h.value
        tau_val = tau.value
    else:
        h_val = None
        tau_val = -float('inf')

    return h_val, tau_val
