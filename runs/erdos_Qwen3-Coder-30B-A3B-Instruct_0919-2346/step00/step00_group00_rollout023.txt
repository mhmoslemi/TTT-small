    """Evaluate C5 for a given step function h"""
    n = len(h)
    # Compute the overlap integral using correlation
    correlation = np.correlate(h, 1 - h, mode="full")

    # Get all the overlap values for the valid k values
    # This is from k=0 to k=2, so n terms
    max_overlap = np.max(correlation[n-1:])
    return max_overlap
