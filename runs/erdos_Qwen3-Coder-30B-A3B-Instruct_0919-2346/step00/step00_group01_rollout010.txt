import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_h(h_values):
    n = len(h_values)
    dx = 2.0 / n
    # Compute correlation of h with (1-h) using full mode
    corr = np.correlate(h_values, 1 - h_values, mode='full')
    # The max value is the upper bound C5
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initial construction using a mixture of 0 and 1 values
    # Based on known good pattern from literature
    h_initial = np.zeros(n_points)
    # Simple alternating pattern with weights that balance
    for i in range(n_points):
        if i % 4 < 2:
            h_initial[i] = 1.0
        else:
            h_initial[i] = 0.0
    
    # Force sum to be exactly n/2
    current_sum = np.sum(h_initial)
    target_sum = n_points / 2
    if current_sum > target_sum:
        # Reduce some 1s to 0s
        indices_ones = np.where(h_initial == 1)[0]
        num_to_reduce = int(current_sum - target_sum)
        to_zero = np.random.choice(indices_ones, num_to_reduce, replace=False)
        h_initial[to_zero] = 0.0
    elif current_sum < target_sum:
        # Increase some 0s to 1s
        indices_zeros = np.where(h_initial == 0)[0]
        num_to_increase = int(target_sum - current_sum)
        to_one = np.random.choice(indices_zeros, num_to_increase, replace=False)
        h_initial[to_one] = 1.0
    
    # Refine using optimization to improve C5
    best_h = h_initial.copy()
    best_c5 = evaluate_h(best_h)
    
    # Random search mechanism to escape local optima
    max_iterations = 20000
    iteration = 0
    start_time = time.time()
    
    while iteration < max_iterations and (time.time() - start_time) < budget_s:
        iteration += 1
        
        # Perturb h by small random changes
        h_new = best_h.copy()
        
        # Swap two random indices
        i, j = np.random.choice(n_points, 2, replace=False)
        delta = np.random.uniform(0.01, 0.1)
        if h_new[i] + delta <= 1 and h_new[j] - delta >= 0:
            h_new[i] += delta
            h_new[j] -= delta
            
            # Ensure sum remains correct
            current_sum = np.sum(h_new)
            residual = target_sum - current_sum
            if abs(residual) > 1e-8:
                # Adjust one element
                k = np.random.randint(n_points)
                h_new[k] = np.clip(h_new[k] + residual, 0, 1)
            
            # Check if this improves C5
            new_c5 = evaluate_h(h_new)
            if new_c5 < best_c5:
                best_h = h_new
                best_c5 = new_c5
                
    return (best_h, best_c5, n_points)
