import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_h(h_values):
    """Evaluate the overlap integral for given h values using correlation."""
    n = len(h_values)
    dx = 2.0 / n
    # Compute correlation of h with (1-h)
    corr = np.correlate(h_values, 1 - np.array(h_values), mode='full')
    # Compute max correlation value and return C5
    c5 = max(corr) * dx
    return c5

def project_h(h_values, n_points):
    """Project h_values to satisfy sum constraint."""
    h = np.array(h_values)
    target_sum = n_points / 2.0
    current_sum = np.sum(h)
    if np.isclose(current_sum, target_sum):
        return h.tolist()
    
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Adjust using water-filling
    current_sum = np.sum(h)
    if current_sum > target_sum:
        # Reduce values
        excess = current_sum - target_sum
        # Sort from low to high
        indices = np.argsort(h)
        for i in indices:
            if excess <= 0:
                break
            decrease = min(h[i], excess)
            h[i] -= decrease
            excess -= decrease
    elif current_sum < target_sum:
        # Increase values
        deficit = target_sum - current_sum
        # Sort from high to low
        indices = np.argsort(h)[::-1]
        for i in indices:
            if deficit <= 0:
                break
            increase = min(1 - h[i], deficit)
            h[i] += increase
            deficit -= increase

    return h.tolist()

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    best_h = None
    best_c5 = float('inf')
    n_points = 100  # Since we prefer less than 1k samples and want good optimize
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize with the known good construction
    h_values = initial_h_values[:n_points] if len(initial_h_values) >= n_points else initial_h_values + [0.5] * (n_points - len(initial_h_values))
    h_values = project_h(h_values, n_points)
    current_c5 = evaluate_h(h_values)
    if current_c5 < best_c5:
        best_h = h_values[:]
        best_c5 = current_c5

    # Use a simple heuristic optimization - hill climbing
    iteration = 0
    max_iter = int(budget_s * 100) if budget_s < 100 else 10000
    step_size = 0.01
    
    while time.time() - start_time < budget_s and iteration < max_iter:
        # Try small random perturbations
        h_trial = h_values[:]
        # Pick 2 indices randomly
        i, j = np.random.choice(n_points, 2, replace=False)
        delta = np.random.uniform(-step_size, step_size)
        h_trial[i] += delta
        h_trial[j] -= delta
        # Project back
        h_trial = project_h(h_trial, n_points)
        # Evaluate
        trial_c5 = evaluate_h(h_trial)
        if trial_c5 < current_c5:
            h_values = h_trial[:]
            current_c5 = trial_c5
            if current_c5 < best_c5:
                best_h = h_values[:]
                best_c5 = current_c5
        iteration += 1

    # Final evaluation
    if best_h is None:
        best_h = h_values[:]
    final_c5 = evaluate_h(best_h)
        
    return (best_h, final_c5, n_points)
