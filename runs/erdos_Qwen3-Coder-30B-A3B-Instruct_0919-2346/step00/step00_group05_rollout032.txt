import numpy as np
import time
from scipy.optimize import linprog
import cvxpy as cp

# Evaluation function
def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for the given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute overlaps for all shifts
    overlaps = np.correlate(h, 1 - h, mode='full') * dx
    # Return the maximum overlap
    return np.max(overlaps)

# Projection onto capped simplex (box + sum)
def project_onto_capped_simplex(v, S):
    """Project v onto { h | 0 <= h_i <= 1, sum h_i = S }."""
    v = np.array(v)
    # Find lambda bounds such that all h_i = min(max(v_i - lambda, 0), 1)
    # gives sum = S
    def check_lambda(lambda_val):
        h = np.clip(v - lambda_val, 0, 1)
        return np.sum(h) - S

    # We know that at lambda = -2 all h_i = 1, at lambda = 2 all h_i = 0
    lambda_low = -2
    lambda_high = 2

    # Bisection to find lambda such that sum(h) = S
    tol = 1e-9
    while lambda_high - lambda_low > tol:
        lambda_mid = (lambda_low + lambda_high) / 2
        if check_lambda(lambda_mid) > 0:
            lambda_low = lambda_mid
        else:
            lambda_high = lambda_mid
            
    lambda_opt = (lambda_low + lambda_high) / 2
    h = np.clip(v - lambda_opt, 0, 1)
    return h

# Linearized LP
def linear_program(a_matrix, const_vec, n, S):
    """
    a_matrix: 2*n-1 x n matrix where a_matrix[k,i] = a_{i,k} for k = 0..2*n-2.
    const_vec: (2*n-1,) vector where const_vec[k] = const_k.
    n: length of h.
    S: sum constraint.
    """
    # Variables: h (n), t (1)
    # Minimize t
    # Coefficients for constraints
    # Multiple shift indices correspond to different rows
    # 'a_matrix' is (2*n-1) x n, const_vec is (2*n-1,)
    # Normal constraint is: a_k^T h + c_k <= t for all k
    # To LP: minimize t subject to A h + b <= t * ones, etc.
    # Constraint matrix: [A, -1] where A = [a_matrix, ..., a_matrix]
    # Now rearrange it into a form accepted by linprog
    
    # Matrix rows for A h + c <= t * ones:
    # "[-a_k, 1]" * [h; t] <= c_k
    # The constraint becomes
    # [ -a_k^T, 1 ] * [h; t] <= c_k
    
    # So we build constraint matrix with shape (2*n-1, n+1)
    # We can also build it with a loop, or using broadcasting
    
    # Build A_con = [ [ -a_k^T, 1 ] for k ]
    # But better to use cvxpy for this step because more intuitive
    
    h = cp.Variable(n)
    t = cp.Variable()
    
    # Add the constraint A * h + b <= t * ones
    # with a_matrix shape (2*n-1, n), const_vec shape (2*n-1,)
    # So we have (2*n-1) constraints, but they can be done together
    
    # Inequality: -a_k^T * h + 1*t <= c_k
    constraints = []
    for i in range(len(const_vec)):
        constraints.append(-a_matrix[i] @ h + t >= const_vec[i])

    # Add two constraints for our bounding box
    constraints.append(cp.sum(h) == S)
    constraints.append(h >= 0)
    constraints.append(h <= 1)

    # Optimize the model
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status in ['optimal', 'optimal_inaccurate']:
            _h = h.value
            _t = t.value
            return _h, _t
        else:
            # Fallback to simple linear programming
            # Build matrix for linprog manually
            # Goal: minimize [0,...0, 1] @ [h, t]
            # Constraint A @ [h, t] <= b
            m = len(const_vec)
            A = []
            b = []
            for k in range(m):
                row = np.zeros(n+1)
                row[:n] = -a_matrix[k]
                row[-1] = 1
                A.append(row)
                b.append(const_vec[k])
            A = np.array(A)
            b = np.array(b)
            # Initial vector and bounds
            c = np.zeros(n+1)
            c[-1] = 1  # minimize t
            bounds = [(0, 1)] * n + [(0, 2)]  # h bounds, t is not constrained (clear in constraints)
            bounds[-1] = (0, None)  # t > 0
            # Linear sum constraint: sum(h) = S
            A_eq = np.zeros((1, n+1))
            A_eq[0, :n] = 1
            b_eq = [S]
            res = linprog(c, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
            if res.success:
                return res.x[:-1], res.x[-1]
            else:
                raise ValueError("Linprog failed")
    except Exception as e:
        # If cvxpy fails, fallback to basic
        # Simple fallback by minimizing t in a simple LP
        # This uses scipy

        # Use scipy linprog directly
        # Convert to standard form
        # minimize c.T * x
        # s.t. A_ub * x <= b_ub
        #      A_eq * x == b_eq
        #      bounds

        m = len(const_vec)
        # Build matrix structure manually for direct linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0

        # Inequality constraints
        A_ub = []
        b_ub = []
        for k in range(m):
            row = np.zeros(n+1)
            row[:n] = -a_matrix[k]
            row[-1] = 1
            A_ub.append(row)
            b_ub.append(const_vec[k])

        A_ub = np.array(A_ub)
        b_ub = np.array(b_ub)

        # Equality constraint (sum h = S)
        A_eq = np.zeros((1, n+1))
        A_eq[0, :n] = 1
        b_eq = [S]

        # Variable bounds
        bounds = [(0, 1)] * n + [(0, None)]  # t=None for free, but we use bounds[0] max.

        # Atoms: Use method='highs' for better performance
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs',
                      options={'disp': False, 'presolve': True})
        if res.success:
            return res.x[:-1], res.x[-1]
        else:
            raise ValueError("Failed to solve LP even with linprog")
    # Fallback is now handled inside
    # Make sure the return is minimal verbose coverage
    raise ValueError("Unexpected error in linear program")

# Coefficients and constants for linearisation
def coeffs_and_const(h):
    """Compute a_{i,k} and constant terms for all shifts k."""
    n = len(h)
    # Create full range of k from -(n-1) to n-1
    shifts = np.arange(-(n-1), n)
    # For each shift, compute a_{i,k} and c_k
    a_matrix = []
    const_vec = []
    # For each shift k, need to sum h_j * h_{j+k} over j
    for k in shifts:
        a_k = np.zeros(n)
        c_k = 0.0
        
        # Compute a_{i,k} terms using convolution-style logic:
        # to compute h_i * h_{i+k}, we shift h forward by k
        # only valid when both h_i and h_{i+k} are in bounds
        # Or you can use full convolution, taking advantage of zero padding
        for i in range(n):
            if 0 <= i+k < n:
                a_k[i] = 1 - h[i+k] - h[i-k] if 0 <= i-k < n else (1 - h[i+k])
            else:
                # Outside domain, either zero padding or need to do better
                # we have to do in the convolution step directly:
                pass
        # For the general a_{i,k} via pointwise product, it needs careful indexing
        # compute coefficients manually, include zero padding
        
        # So for a_k = 1 - h[i+k] - h[i-k], where we adjust for boundary later
        # Set up left side with h[:-k], right with h[k:], pad with zeros
        # But a bit more careful work with indexing.
        
        # OR preprocess for more clearly
        # for efficiency, do:
        # Slide h by k in the assumption that for positive shift
        # then len h = n, we have support 0 to n-1, so
        # h_j h_{j+k} contribution for j such that both j, j+k within [0,n-1]
        # use full correlation:
        for i in range(n):
            # Compute i+k and i-k: out-of-bounds -> idx cropped or zero-padded
            # What happens in empty indices? They only affect contribution to a_k[i]
            # So we compute a_{i,k} explicitly
        
        # Even simpler way:
        # Use full overlap computation for final step (but precompute conv weights)
        # a_primary = 1 - h[i+k] - h[i-k] for valid or outside, handle padding via indices
        # start with double precision correction
        
        a_k = np.zeros(n)
        for i in range(n):
            ik = i + k
            imk = i - k
            val = 1.0
            if 0 <= ik < n:
                val -= h[ik]
            if 0 <= imk < n:
                val -= h[imk]
            a_k[i] = val

        # Constant term: sum h_i * h_{i+k} for i such that both indices valid
        c_k = 0.0
        for i in range(n):
            if 0 <= i < n and 0 <= i+k < n:
                c_k += h[i] * h[i+k]

        a_matrix.append(a_k)
        const_vec.append(c_k)

    return np.array(a_matrix), np.array(const_vec)
