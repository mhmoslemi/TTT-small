def evaluate(h_vec, dx):
    """Evaluate C_5 bound using FFT for efficiency."""
    h = np.array(h_vec)
    F = fft(h)
    G = fft(1 - h)
    corr = ifft(F * np.conj(G)).real
    max_corr = max(corr)
    return max_corr * dx

def project_to_simplex(h, target_sum):
    """Project h onto the simplex with sum = target_sum."""
    mean_violation = (np.sum(h) - target_sum) / len(h)
    h = h - mean_violation
    h = np.clip(h, 0, 1)
    # Ensure exact sum with Dykstra's projection
    for _ in range(5):
        mean_violation = (np.sum(h) - target_sum) / len(h)
        h = h - mean_violation
        h = np.clip(h, 0, 1)
    return h

def initialize_h(n_points, initial_h):
    """Initialize h using interpolation or a binary seed."""
    h = np.array(initial_h)
    h = np.interp(np.linspace(0, 1, n_points), np.linspace(0, 1, len(h)), h)
    return project_to_simplex(h, n_points / 2)

def compute_mismatches(h, k):
    """Compute mismatch sets for shift k."""
    n = len(h)
    I_k = list(range(n - abs(k)))
    if k > 0:
        M1 = [i for i in I_k if h[i] > 0.99 and h[i + k] < 0.01]

        M0 = [i for i in I_k if h[i] < 0.01 and h[i + k] > 0.99]
    return M1, M0

            M1, M0 = compute_mismatches(h, k - (len(h) - 1))
        n_iter += 1
