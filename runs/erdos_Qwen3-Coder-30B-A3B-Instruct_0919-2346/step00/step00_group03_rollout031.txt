import numpy as np
import time
import random

def evaluate_c5(h):
    """Evaluate the C5 bound for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h) using full mode
    corr = np.correlate(h, 1 - h, mode="full")
    # Scale by dx and take maximum
    c5 = np.max(corr) * dx
    return c5

def random_binary_seq(n):
    """Generate a random binary sequence with exactly n/2 ones."""
    h = np.zeros(n)
    indices = np.arange(n)
    np.random.shuffle(indices)
    ones_indices = indices[:n//2]
    h[ones_indices] = 1
    return h

def solve_binary(n, iters, seed=None):
    """Perform local search on binary sequences."""
    # Set random seed
    rng = np.random.default_rng(seed)
    
    # Start with a random binary sequence
    h = random_binary_seq(n)
    c5 = evaluate_c5(h)
    best_h, best_c5 = h.copy(), c5

    # Add blocking to improve performance
    T0 = 0.1
    T = T0
    cooling = 0.99995
    step = 1
    start_time = time.time()
    
    # Set the target threshold for c5
    target_c5 = 0.3807
    target_threshold = int(np.floor(target_c5 / (2.0 / n)))
    
    for iteration in range(iters):
        # Check time limit
        if time.time() - start_time > 580:
            break
        
        # Occasionally reinitialize to escape local minima
        if iteration % 2000 == 0 and iteration > 0:
            h = random_binary_seq(n)
            c5 = evaluate_c5(h)
            best_h, best_c5 = h.copy(), c5

        # Choose two random indices
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        
        # Swap the values at these indices
        h[i], h[j] = h[j], h[i]
        
        # Evaluate the new C5
        new_c5 = evaluate_c5(h)
        delta = new_c5 - c5
        
        # Accept or reject based on simulated annealing
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            c5 = new_c5
            if new_c5 < best_c5:
                best_c5, best_h = new_c5, h.copy()
        else:
            # Revert the swap
            h[i], h[j] = h[j], h[i]
        
        # Cooling schedule
        T *= cooling

    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    """Main run function to optimize the Erdős constant C5."""
    
    # Use a moderately fine discretization
    n_points = 200
    dx = 2.0 / n_points
    
    # Use multiple restarts with different seeds for robustness
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    restarts = 0
    max_restarts = 15  # Less than 200k iterations for faster evaluation
    
    for i in range(max_restarts):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Determine seed for current restart
        current_seed = seed + i * 23
        
        # Run one local search with specified number of iterations
        h, c5 = solve_binary(n_points, 10000, seed=current_seed)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        restarts += 1
    
    # Return final result
    if best_h is None:
        # Fallback to a basic initialization
        h = random_binary_seq(n_points)
        
    return (best_h.tolist(), best_c5, n_points)
