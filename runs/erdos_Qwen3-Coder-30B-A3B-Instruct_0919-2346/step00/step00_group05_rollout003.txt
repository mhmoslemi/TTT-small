import numpy as np
import time
from cvxpy import Variable, Minimize, Problem, sum as cp_sum, Norm1
from cvxpy import ECOS

def project_onto_capped_simplex(v, S):
    """Project v onto {h | 0 <= h_i <= 1, sum h_i = S} using bisection."""
    v = np.asarray(v)
    low, high = -2, 2
    tol = 1e-12
    while high - low > tol:
        mid = (low + high) / 2
        h = np.clip(v - mid, 0, 1)
        s = np.sum(h)
        if s > S:
            low = mid
        else:
            high = mid
    h = np.clip(v - (low + high) / 2, 0, 1)
    return h

def coeffs_and_const(h, n):
    """Compute a_{i,k} and c_k for all shifts k."""
    a_matrix = np.zeros((n, 2 * n - 1))
    const_vec = np.zeros(2 * n - 1)
    for k in range(-(n - 1), n):
        k_idx = k + (n - 1)
        a = np.zeros(n)
        const = 0.0
        for i in range(n):
            # h_i * h_{i+k}
            j = i + k
            if 0 <= j < n:
                a[i] = 1.0 - h[j] - h[i]
                const += h[i] * h[j]
            elif 0 <= i < n:
                a[i] = 1.0 - h[i]
            else:
                a[i] = 1.0

        a_matrix[:, k_idx] = a
        const_vec[k_idx] = const
    return a_matrix, const_vec

def linear_program(a_matrix, const_vec, n, S):
    """Form and solve the linear program, return h and t."""
    h = Variable(n)
    t = Variable()
    # Constraints
    constraints = []
    for k_idx in range(2 * n - 1):
        constraints.append(a_matrix[:, k_idx] @ h + const_vec[k_idx] <= t)
    constraints.append(cp_sum(h) == S)
    constraints.append(h >= 0)
    constraints.append(h <= 1)
    constraints.append(t >= 0)
    # Objective
    objective = Minimize(t)
    prob = Problem(objective, constraints)
    try:
        prob.solve(solver=ECOS, verbose=False)
        h_val = h.value
        t_val = t.value
    except Exception:
        # fallback to projection
        h_val = np.clip(np.random.rand(n), 0, 1)
        t_val = np.inf
    return h_val, t_val

def evaluate(h):
    """Compute C5 via correlation."""
    n = len(h)
    dx = 2.0 / n
    # Compute overlaps for all shifts k
    overlaps = np.correlate(h, 1 - h, mode='full')
    max_overlap = overlaps.max() * dx
    return max_overlap

def random_feasible(n):
    """Generate a random feasible point with exact sum."""
    v = np.random.rand(n)
    v = np.clip(v, 0, 1)
    s = np.sum(v)
    S = n / 2.0
    if abs(s - S) > 1e-8:
        scale = S / s
        v = v * scale
        v = np.clip(v, 0, 1)
    # Make sure it's actually valid and sum matches
    v = project_onto_capped_simplex(v, S)
    return v

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n = len(initial_h_values) if 'initial_h_values' in globals() else 70
    S = n / 2.0
    dx = 2.0 / n

    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    restarts = 0
    max_restarts = 100
    max_iter = 30
    damping = 0.7

    while restarts < max_restarts:
        if time.time() - start_time > 0.95 * budget_s:
            break
        # Try to initialize from existing `initial_h_values`
        if restarts == 0 and 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            if abs(np.sum(h) - S) > 1e-8:
                h = project_onto_capped_simplex(h, S)
        else:
            h = random_feasible(n)
        # coarse polish before SLA
        c5_prev = evaluate(h)
        for _ in range(100):
            if time.time() - start_time > 0.95 * budget_s:
                break
            # try one random move
            i = np.random.randint(0, n)
            j = np.random.randint(0, n)
            if np.abs(h[i] - 1) < 1e-8 or np.abs(h[j]) < 1e-8:
                continue
            delta = np.random.uniform(0, 0.01)
            # delta_i to j, i->j, j->i
            if h[i] > delta and h[j] < 1 - delta:
                h_new = h.copy()
                h_new[i] -= delta
                h_new[j] += delta
                h_new = project_onto_capped_simplex(h_new, S)
                c5_new = evaluate(h_new)
                if c5_new < c5_prev:
                    h = h_new
                    c5_prev = c5_new
        # if solution not significantly better, skip to next restart
        if c5_prev < best_c5:
            best_h = h.copy()
            best_c5 = c5_prev
        # run ccp
        for r in range(max_iter):
            if time.time() - start_time > 0.95 * budget_s:
                break
            # compute linearisation
            a_matrix, const_vec = coeffs_and_const(h, n)
            # solve LP
            h_lp, t = linear_program(a_matrix, const_vec, n, S)
            # damping and project
            h = damping * h_lp + (1 - damping) * h
            h = project_onto_capped_simplex(h, S)
            c5_current = evaluate(h)
            # best so far?
            if c5_current < best_c5:
                best_h = h.copy()
                best_c5 = c5_current
        # next restart
        restarts += 1

    # Final polishing by targeting one shift
    if best_h is not None:
        h_final = best_h.copy()
        c5_final = evaluate(h_final)
        for _ in range(1000):
            if time.time() - start_time > 0.95 * budget_s:
                break
            i = np.random.randint(0, n)
            k = np.random.randint(-(n - 1), n)
            j = i + k
            if not (0 <= i < n) or not (0 <= j < n):
                continue
            if np.abs(h_final[i] - 0) < 1e-8 or np.abs(h_final[j] - 1) < 1e-8:
                continue
            delta = np.random.uniform(0, 0.005)
            if h_final[i] - delta >= 0 and h_final[j] + delta <= 1:
                h_new = h_final.copy()
                h_new[i] -= delta
                h_new[j] += delta
                h_new = project_onto_capped_simplex(h_new, S)
                c5_new = evaluate(h_new)
                if c5_new < c5_final:
                    h_final = h_new
                    c5_final = c5_new

        best_h = h_final
        best_c5 = c5_final

    return best_h, best_c5, n
