def autocorr(h, k, n):
    # Circular autocorrelation at shift k
    # h[i] * h[i+k]
    corr = 0.0
    for i in range(n):
        corr += h[i] * h[(i + k) % n]
    return corr


def min_autocorr(h, n):
    auto_corrs = [autocorr(h, k, n) for k in range(1, n)]
    return min(auto_corrs)


def fit_h_to_constraints(h, n):
    # Ensures sum = n/2 and values in [0,1]
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target = n / 2.0
    diff = current_sum - target
    
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        # Need to decrease some values
        # Move a bit from shadier > 0 to < 1
        # Insignificant already so just round to force constraint
        h = h.copy()
        for idx in np.argsort(h)[::-1]:
            if diff <= 0:
                break
            min_reduce = min(h[idx], diff)
            h[idx] -= min_reduce
            diff -= min_reduce
    else:
        # Need to increase some values - not too impactful
        h = h.copy()
        for idx in np.argsort(h):
            if diff >= 0:
                break
            max_add = min(1 - h[idx], -diff)
            h[idx] += max_add
            diff += max_add
    return h


def calculate_C5_vectorized(h, n):
    d = np.fft.fft(h)
    corr_real = np.fft.ifft(d * np.conj(d)).real
    return 1.0 - 2 * np.min(corr_real[1:]) / n


def gradient_ascent_step(h, k_star, n, step_size=0.01):
    # Compute gradient: ∇f(x) = h_{i+k*} + h_{i-k*}
    grad = np.zeros(n)
    for i in range(n):
        grad[i] = h[(i + k_star) % n] + h[(i - k_star) % n]
    
    # Gradient ascent step
    h_new = h + step_size * grad
    # Clip to [0,1]
    h_new = np.clip(h_new, 0, 1)
    
    return fit_h_to_constraints(h_new, n)


def polish(h_init, n, initial_bound):
    h_best = h_init.copy()
    min_c5 = initial_bound
    
    for i in range(200):
        # Calculate min autocorrelation shift
        corrs = [autocorr(h_best, k, n) for k in range(1, n)]
        k_star = np.argmin(corrs) + 1
        # Take a few gradient steps
        for _ in range(1):
            h_new = gradient_ascent_step(h_best, k_star, n, 0.01)
            new_c5 = 1 - 2 * min_autocorr(h_new, n) / n
            if new_c5 < min_c5:
                min_c5 = new_c5
                h_best = h_new
        if i % 20 == 0:
            # Periodic recheck
            corr = 1 - 2 * min_autocorr(h_best, n) / n
            if corr < min_c5:
                min_c5 = corr
    return h_best, min_c5


def initial_h_values(n):
    # Initial construction from the documentation
    # Try a symmetric binary construction
    h = np.zeros(n)
    # Place ones at symmetric positions to ensure the integral is 1
    ones = list(range(0, n//2, 2)) + list(range(1, n//2, 2))
    ones = ones[:n//2]
    for i in ones:
        h[i] = 1
    return h
