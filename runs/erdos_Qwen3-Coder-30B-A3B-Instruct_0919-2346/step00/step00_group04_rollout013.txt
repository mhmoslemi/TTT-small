import numpy as np
import time
import math
import random


def c5_bound(h):
    n = len(h)
    dx = 2.0 / n
    O = np.correlate(h, 1 - h, mode='full')
    return O.max() * dx


def random_binary(n):
    h = np.random.choice([0, 1], size=n)
    while np.sum(h) != n // 2:
        if np.sum(h) < n // 2:
            # Find zeros and flip some to ones
            zeros = np.where(h == 0)[0]
            flip_idx = np.random.choice(zeros, size=(n // 2) - np.sum(h), replace=False)
            h[flip_idx] = 1
        else:
            # Find ones and flip some to zeros
            ones = np.where(h == 1)[0]
            flip_idx = np.random.choice(ones, size=np.sum(h) - (n // 2), replace=False)
            h[flip_idx] = 0
    return h


def thue_morse(n):
    h = np.zeros(n, dtype=int)
    for i in range(n):
        h[i] = bin(i).count('1') % 2
    return h


def is_prime(n):
    if n < 2:
        return False
    if n == 2:
        return True
    if n % 2 == 0:
        return False
    for i in range(3, int(n**0.5) + 1, 2):
        if n % i == 0:
            return False
    return True


def legendre_seq(p):
    h = np.zeros(p, dtype=int)
    for i in range(p):
        # Quadratic residue modulo p
        if (i * i) % p == 0:
            h[i] = 1
    # Normalize weight to p/2
    cur_sum = np.sum(h)
    if cur_sum > p // 2:
        # Flip some 1s to 0s
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:cur_sum - p // 2]] = 0
    elif cur_sum < p // 2:
        # Flip some 0s to 1s
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:p // 2 - cur_sum]] = 1
    return h


def m_sequence(m):
    # Generate maximal LFSR sequence of length 2^m - 1
    if m < 2:
        return np.array([0] * (2**m - 1))
    # Use a simple recursive Fibonacci-style approach
    # We will use known values for m up to 5, otherwise return random
    if m == 2:
        # 0,1,1,0 -> repeated pattern. Make length 3
        return np.array([0,1,1,0,0,1,1,0])
    elif m == 3:
        # Known period 7, 0,1,0,0,1,1,1
        return np.array([0,1,0,0,1,1,1] * 11 + [0])[:2**3-1]
    elif m == 4:
        # Period 15
        return np.array([0,1,0,0,1,1,1,0,1,0,1,1,0,0,0] * 2 + [0])[:2**4-1]
    else:
        # Fallback to random pattern
        return random_binary(2**m - 1)


def initial_h_to_n(n):
    if 'initial_h_values' in globals():
        base = np.array(initial_h_values, dtype=float)
        x_old = np.linspace(0, 2, len(base), endpoint=False)
        x_new = np.linspace(0, 2, n, endpoint=False)
        h_interp = np.interp(x_new, x_old, base)
        # Convert to binary, ensure sum = n/2
        h_bin = (h_interp > 0.5).astype(int)
        # Adjust weight
        cur = h_bin.sum()
        target = n // 2
        if cur > target:
            idx = np.where(h_bin == 1)[0]
            np.random.shuffle(idx)
            h_bin[idx[:cur-target]] = 0
        elif cur < target:
            idx = np.where(h_bin == 0)[0]
            np.random.shuffle(idx)
            h_bin[idx[:target-cur]] = 1
        return h_bin
    return random_binary(n)


def swap_move(h, ones, zeros):
    if len(ones) == 0 or len(zeros) == 0:
        return h, ones, zeros, None, None
    oi = random.randint(0, len(ones) - 1)
    oj = random.randint(0, len(zeros) - 1)
    i = ones[oi]
    j = zeros[oj]
    # Perform swap
    new_h = h.copy()
    new_h[i] = 0
    new_h[j] = 1
    # Update lists
    new_ones = ones.copy()
    new_zeros = zeros.copy()
    new_ones[oi] = j
    new_zeros[oj] = i
    return new_h, new_ones, new_zeros, i, j


def local_search(h, time_budget):
    n = len(h)
    cur_val = c5_bound(h)
    best_h = h.copy()
    best_val = cur_val
    start = time.time()
    max_iters = 1000000
    temp_start = 0.1
    temp_end = 1e-5

    # Track indices that are ones and zeros for quick sampling
    ones = list(np.flatnonzero(h))
    zeros = list(np.flatnonzero(1 - h))

    for it in range(max_iters):
        if time.time() - start > time_budget:
            break
        # Temperature schedule
        temp = temp_start * (temp_end / temp_start) ** (it / max_iters)
        # Swap
        new_h, new_ones, new_zeros, i, j = swap_move(h, ones, zeros)
        val = c5_bound(new_h)
        # Accept criteria: improve or accept with probability
        delta = val - cur_val
        if delta <= 0 or np.random.rand() < math.exp(-delta / temp):
            h = new_h
            ones = new_ones
            zeros = new_zeros
            cur_val = val
            if val < best_val:
                best_val = val
                best_h = h.copy()
        else:
            # revert
            continue
    return best_h, best_val


def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start = time.time()
    best_h = None
    best_c5 = float('inf')
    best_n = None

    lengths = [64, 80, 96, 112, 128, 144, 160, 176, 192, 224, 256, 320, 384, 448, 512]
    lengths = [n for n in lengths if n <= 800]

    # Precompute candidates
    candidates = []
    for n in lengths:
        # Thue-Morse if n is power of 2
        if (n & (n - 1)) == 0:
            seq = thue_morse(n)
            candidates.append((seq, n))
        # Legendre for prime n
        if is_prime(n):
            seq = legendre_seq(n)
            candidates.append((seq, n))
        # Random sequences
        for _ in range(2):
            seq = random_binary(n)
            candidates.append((seq, n))
        # Use Replicating initial pattern
        seq = initial_h_to_n(n)
        candidates.append((seq, n))

    # Evaluate initial candidates
    for h, n in candidates:
        c5 = c5_bound(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n

    # Improve the best with local search
    improved = True
    while improved and (time.time() - start < budget_s * 0.9):
        improved = False
        # Determine how much time to spend improving
        remaining_time = budget_s - (time.time() - start)
        if remaining_time <= 1.0:
            break
        # Get top few candidates based on current best
        top_candidates = []
        for h, n in candidates:
            c5 = c5_bound(h)
            if c5 <= best_c5 * 1.05:  # Allow some slack
                top_candidates.append((h.copy(), n, c5))
        if not top_candidates:
            break

        # Sort by current value
        top_candidates.sort(key=lambda x: x[2])
        # Improve first few
        for h, n, c5 in top_candidates:
            if time.time() - start > budget_s * 0.9:
                break
            time_for_this = min(remaining_time / len(top_candidates), 20)
            h_improved, c5_improved = local_search(h, time_for_this)
            if c5_improved < best_c5:
                best_c5 = c5_improved
                best_h = h_improved.copy()
                best_n = n
                improved = True
            if time.time() - start > budget_s * 0.9:
                break

    return (best_h, best_c5, best_n)
