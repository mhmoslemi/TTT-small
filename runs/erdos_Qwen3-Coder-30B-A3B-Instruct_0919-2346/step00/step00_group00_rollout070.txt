import numpy as np
import time
import random

def compute_c5(h):
    """Compute the C5 bound for a given step function h."""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return corr.max()

def propose_swap_and_update(h, corr, i, j, dx):
    """Propose a swap between positions i and j, and update correlation accordingly.
    Returns new correlation array and new max value."""
    # Get old values
    old_i = h[i]
    old_j = h[j]
    
    # Compute delta for the swap
    delta_i = old_j - old_i
    
    # Create new correlation array
    new_corr = corr.copy()
    
    # Update contributions from swapping index i and j
    n = len(h)
    
    # For each k, update contributions where i or j as first element in pair or second element
    for k in range(n):
        if k == i or k == j:
            continue
        
        hk = h[k]
        
        # Contribution to corr where p = i, q = k (shift = k - i)
        shift1 = k - i + n - 1
        delta1 = delta_i * (1 - hk)
        new_corr[shift1] += delta1
        
        # Contribution to corr where p = k, q = i (shift = i - k)
        shift2 = i - k + n - 1
        delta2 = hk * (-delta_i)
        new_corr[shift2] += delta2
        
        # Same for j
        # Contribution to corr where p = j, q = k (shift = k - j)
        shift3 = k - j + n - 1
        delta3 = -delta_i * (1 - hk)
        new_corr[shift3] += delta3
        
        # Contribution to corr where p = k, q = j (shift = j - k)
        shift4 = j - k + n - 1
        delta4 = hk * delta_i
        new_corr[shift4] += delta4
    
    # Handle special pairs (i,j) and (j,i)
    # Pair (i,j): shift = j - i
    shift_ij = j - i + n - 1
    new_corr[shift_ij] += delta_i
    
    # Pair (j,i): shift = i - j
    shift_ji = i - j + n - 1
    new_corr[shift_ji] -= delta_i
    
    # Swap values in h
    h[i], h[j] = h[j], h[i]
    
    # Compute new maximum
    new_max = new_corr.max()
    return new_corr, new_max

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    start = time.time()
    best_c5 = np.inf
    best_h = None
    best_n = None
    
    # Candidate n values for the resolution
    n_candidates = [120, 140, 160, 180, 200]
    
    for n in n_candidates:
        if time.time() - start > budget_s - 1:
            break
        if n % 2 != 0:
            continue
            
        # Set parameters for local search
        max_iters = int(20000 / (n / 120))
        max_iters = max(max_iters, 2000)
        
        # Initialize h: random binary with exactly n/2 ones
        h = np.zeros(n, dtype=float)
        ones = n // 2
        indices = np.random.choice(n, size=ones, replace=False)
        h[indices] = 1.0
        
        # Compute initial correlation
        dx = 2.0 / n
        corr = np.correlate(h, 1 - h, mode='full') * dx
        cur_c5 = corr.max()
        cur_h = h.copy()
        cur_max = corr.max()
        
        # Temperature schedule
        T = 0.02
        T_decay = 0.9995
        
        # Assume fixed number of iterations (whatever the max_iters is set)
        for it in range(max_iters):
            if time.time() - start > budget_s - 0.5:
                break
                
            # Select two random positions to swap
            i = np.random.randint(n)
            j = np.random.randint(n)
            while h[i] == h[j]:
                i = np.random.randint(n)
                j = np.random.randint(n)
                
            # Propose swap and update correlation
            new_corr, new_max = propose_swap_and_update(h, corr, i, j, dx)
            
            # Accept or reject the swap based on energy difference and temperature
            old_max = cur_max
            if new_max < old_max or np.random.rand() < np.exp(-(new_max - old_max) / (T + 1e-12)):
                # Accept the move
                corr = new_corr
                cur_max = new_max
                cur_c5 = new_max * dx
                # Check if this is the best so far
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    best_n = n
                    
            else:
                # Revert the swap
                h[i], h[j] = h[j], h[i]
                
            # Update temperature
            T *= T_decay
            
    # Return the best found solution
    if best_h is None:
        # Fallback to simple construction if no progress
        n = 120
        h = np.zeros(n, dtype=float)
        ones = n // 2
        h[np.random.choice(n, size=ones, replace=False)] = 1.0
        best_h = h
        best_c5 = compute_c5(best_h)
        best_n = n
                
    return best_h, best_c5, best_n
