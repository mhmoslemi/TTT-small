def evaluate_h(h):
    """Evaluate the maximum overlap integral for given step function h"""

    dx = 2.0 / n

    corr = np.correlate(h, 1 - h, mode="full")

    # The correlation array has length 2*n - 1
    # Central element (index n-1) corresponds to shift 0
    max_overlap = np.max(corr[1:])  # Exclude the shift 0 case
    return max_overlap * dx
