import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h, n_points):
    """Evaluate the Erdős constant C_5 for given step function h"""
    dx = 2.0 / n_points
    # Compute the overlap integral for all shifts
    correlations = np.correlate(h, 1 - h, mode="full")
    # The maximum correlation is the C_5 value
    c5 = np.max(correlations) * dx
    return c5
