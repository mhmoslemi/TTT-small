def circular_correlation(h):
    """Compute full circular correlation using FFT for efficient evaluation."""
    n = len(h)
    padded = np.concatenate([h, np.zeros(n)])
    freq = fft(padded)
    corr = ifft(freq * np.conj(freq)).real[:n]
    return corr

def compute_C5_from_autocorr(autocorr_values):
    """Convert autocorrelation to C5 bound."""
    return 1 - (2.0 / len(autocorr_values)) * np.min(autocorr_values)

def sdp_solve(h_start=None, n=n_points, seed=42):
    """Solve SDP relaxation for the max-min problem."""
    np.random.seed(seed)
    # Create matrix M_k for each shift k
    M = []
    for k in range(n):
        M_k = np.zeros((n, n))
        for i in range(n):
            M_k[i, (i + k) % n] = 1
            M_k[(i + k) % n, i] = 1
        M.append(M_k)
    
    # Variables
    X = cp.Variable((n, n), symmetric=True)
    h = cp.Variable(n)
    t = cp.Variable()

    # Constraints
    constraints = [
        X >> 0,  # PSD
        cp.diag(X) == h,
        h >= 0,
        h <= 1,
        cp.sum(h) == n / 2,
    ]
    
    for k, M_k in enumerate(M):
        constraints.append(cp.trace(M_k @ X) >= t)

    # Objective
    objective = cp.Maximize(t)

    # Problem
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=SCS, verbose=False, max_iters=1000)
    except Exception:
        return None, None, None
    
    if prob.status in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
        h_sol = h.value
        t_sol = t.value
        X_sol = X.value
        return h_sol, X_sol, t_sol
    else:
        return None, None, None

def round_h(X, h_sdp, n=n_points, seed=42):
    """Rounding step using random hyperplane method."""
    np.random.seed(seed)
    h = np.copy(h_sdp)
    
    # Perform SVD or Cholesky decomposition
    try:
        U, s, Vt = np.linalg.svd(X)
        r = np.sum(s > 1e-10)
        if r > n:
            r = n
        V = U[:, :r]
        s_sqrt = np.sqrt(s[:r])
        V = V * s_sqrt.reshape(1, -1)
    except:
        # Fallback to eigenvalue decomposition if SVD fails
        _, V = np.linalg.eigh(X)
        V = V[:, -n:]  # Take most significant components
    
    g = np.random.randn(V.shape[1])
    scores = V @ g
    sorted_indices = np.argsort(-scores)
    
    # Set threshold to have exactly n/2 ones
    target = n // 2
    threshold = scores[sorted_indices[target]]
    
    # Generate binary vector
    h_rounded = np.zeros(n)
    for i in range(n):
        if scores[i] >= threshold:
            h_rounded[i] = 1
    
    # Adjust if necessary
    diff = np.sum(h_rounded) - target
    if diff > 0:
        # Need to set some ones to zeros
        to_zero = np.where(h_rounded == 1)[0]
        to_zero = sorted(to_zero, key=lambda i: scores[i])
        for i in to_zero[:diff]:
            h_rounded[i] = 0
    elif diff < 0:
        # Need to set some zeros to ones
        to_one = np.where(h_rounded == 0)[0]
        to_one = sorted(to_one, key=lambda i: -scores[i])
        for i in to_one[:abs(diff)]:
            h_rounded[i] = 1
    
    return h_rounded

def swap_hill_climb(h, n=n_points, max_iter=50):
    # Helper to compute correlation
    def correlation(h, k):
        cor = 0
        for i in range(n):
            cor += h[i] * h[(i + k) % n]
        return cor
    
    best = np.copy(h)
    best_corr = None
    for k in range(n):
        corr = correlation(best, k)
        if best_corr is None or corr < best_corr:
            best_corr = corr
    
    # Do swaps
    for _ in range(max_iter):
        improved = False
        for i in range(n):
            if best[i] == 1:
                for j in range(n):
                    if best[j] == 0 and i != j:
                        # Temporarily swap
                        test = np.copy(best)
                        test[i] = 0
                        test[j] = 1
                        
                        # Compute all new correlations
                        new_best_corr = None
                        for k in range(n):
                            corr = correlation(test, k)
                            if new_best_corr is None or corr < new_best_corr:
                                new_best_corr = corr
                        
                        if new_best_corr < best_corr:
                            best = test
                            best_corr = new_best_corr
                            improved = True
                            break
        if not improved:
            break
    
    return best

def gradient_fine_tune(h, n=n_points, n_iter=n_iterations_gradient, step_size=0.01):
    def autocorrelations(h):
        # Compute all correlations and return the minimum
        cor = circular_correlation(h)
        return np.min(cor)

    def gradient(h, min_k):
        # Return gradient of worst correlation with respect to h
        grad = np.zeros_like(h)
        k = min_k  # Worst shift
        # Compute gradient: ∂/∂h_i of h_i * h_{i+k}
        for i in range(n):
            grad[i] += h[(i + k) % n] + h[(i - k) % n]
        return grad

    h_finetuned = np.copy(h)
    for i in range(n_iter):
        if i > 0 and i % 20 == 0:
            step_size *= 0.95  # decay step size
        # Evaluate correlations
        cor = circular_correlation(h_finetuned)
        min_k = np.argmin(cor)
        min_val = cor[min_k]

        # Compute gradient
        grad = gradient(h_finetuned, min_k)
        # Update (project to [0,1] and normalize)
        h_new = h_finetuned + step_size * grad
        h_new = np.clip(h_new, 0, 1)
        # Ensure sum = 1
        scale = (n_points / 2) / np.sum(h_new)
        h_new *= scale
        h_finetuned = h_new.copy()
    
    return h_finetuned
