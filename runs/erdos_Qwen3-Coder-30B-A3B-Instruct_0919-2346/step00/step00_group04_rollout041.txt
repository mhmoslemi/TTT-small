def evaluate_h(h):
    """Evaluate the maximum overlap integral for step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation using numpy correlate with mode='full'

    corr = np.correlate(h, 1 - h, mode='full')
    # The overlap values are in the middle part of correlation
    # Compute the maximum overlap over all possible shifts
    max_overlap = np.max(corr)
    return max_overlap * dx
