import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
import time
import math

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlations
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum is over all k values
    max_corr = np.max(corr)
    return max_corr * dx

def initial_h_values():
    """Return initial construction for h"""
    n = 45
    h = np.zeros(n)
    # Simple construction: step function that is 1 on first half
    # and 0 on second half, but we scale it to integrate to 1
    h[:n//2] = 1.0
    h[n//2:] = 0.0
    # Scale to have integral = 1
    h *= 2.0 / n
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 45  # Start with the provided size
    dx = 2.0 / n_points
    h = initial_h_values()
    
    # Convert to a continuous form for optimization
    # Keep values in [0,1] and sum(h)*dx = 1, i.e., sum(h) = n_points/2
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Additional optimization using scipy SLSQP
    # Define constraints
    def objective(h_flat):
        # Make sure values are in [0,1] and sum to n_points/2
        h = np.clip(h_flat, 0, 1)
        # Scale to make sum = n_points/2 (this is a scaling factor, not a constraint)
        h = h * (n_points / 2) / np.sum(h)
        return evaluate_erdos_solution(h)
    
    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2
    
    def constraint_bounds(h_flat):
        return np.min(h_flat) - 0.0  # min <= 0
    def constraint_bounds2(h_flat):
        return 1.0 - np.max(h_flat)  # max <= 1
        
    constraints = [
        {'type': 'eq', 'fun': constraint_sum},
        {'type': 'ineq', 'fun': constraint_bounds},
        {'type': 'ineq', 'fun': constraint_bounds2}
    ]
    
    # Start with the initial solution
    h_init = h.copy()
    # Flatten
    h_init = h_init.flatten()
    
    x0 = h_init.copy()
    
    # Use a very small bound so the optimization is robust
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Use scipy minimize with method='SLSQP' 
    try:
        result = minimize(
            objective, 
            x0, 
            bounds=bounds,
            constraints=constraints,
            method='SLSQP',
            options={'maxiter': 2000, 'ftol': 1e-8, 'gtol': 1e-6},
            callback=lambda x: None  # Placeholder callback
        )
        
        # Extract solution
        h_opt = result.x
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = h_opt * (n_points / 2) / np.sum(h_opt)  # Normalize sum
        
        # Re-evaluate
        c5_opt = evaluate_erdos_solution(h_opt)
        
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    except Exception as e:
        pass  # If optimization fails, keep initial
    
    # Try a few more refinements with local variations and other methods
    # Sample from region near the current solution
    for i in range(20):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Slightly perturb the current best
        h_pert = best_h.copy()
        # Small random noise
        noise = np.random.normal(0, 0.05, n_points)
        h_pert += noise
        
        h_pert = np.clip(h_pert, 0, 1)
        # Rescale to make sum == n_points/2
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        
        c5_pert = evaluate_erdos_solution(h_pert)
        if c5_pert < best_c5:
            best_c5 = c5_pert
            best_h = h_pert
            
    return (best_h, best_c5, n_points)
