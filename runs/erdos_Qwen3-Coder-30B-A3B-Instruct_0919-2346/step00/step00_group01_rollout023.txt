import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given step function h."""
    # Integrate h(x)(1 - h(x+k)) over all possible k values
    # Using scipy.signal.correlate for efficient computation
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h) using full mode
    corr = correlate(h, 1 - h, mode="full")
    # Take maximum of correlation values (scaled by dx)
    c5 = np.max(corr) * dx
    return c5

def initial_h_values():
    """Initial construction with 47 samples as provided."""
    n_points = 47
    h = np.zeros(n_points)
    dx = 2.0 / n_points
    
    # Simple initial construction: piecewise constant with value 0.5
    h[:] = 0.5
    
    # Adjust to meet integral constraint
    h_sum = np.sum(h)
    h *= (1.0 / h_sum)  # Scale so that integral equals 1
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Get initial guess
    h = initial_h_values()
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Define optimization constraints
    def objective(h_vec):
        # Reshape to array of appropriate size
        h_vals = h_vec.copy()
        # Evaluate objective function
        return evaluate_erdos_solution(h_vals)
    
    # Define bounds for each h[i]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define the constraint: sum(h) * dx = 1 => sum(h) = n_points / 2
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2.0
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Use scipy.optimize.minimize with SLSQP method to optimize
    options = {'maxiter': 1000, 'ftol': 1e-8, 'gtol': 1e-8}
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    print(f"Starting with initial C5 = {best_c5:.6f}")
    
    # Optimization loop with time tracking
    elapsed = time.time() - start_time
    while elapsed < budget_s * 0.9:
        # Random perturbation to escape local minima
        h_perturbed = h.copy()
        for i in range(n_points):
            if np.random.rand() < 0.3:  # 30% chance of perturbation
                delta = (np.random.rand() - 0.5) * 0.2  # Small perturbation
                h_perturbed[i] = np.clip(h_perturbed[i] + delta, 0, 1)
        
        # Make sure integral is still correct
        h_perturbed_sum = np.sum(h_perturbed)
        h_perturbed *= (n_points / 2.0) / h_perturbed_sum
        
        c5_candidate = evaluate_erdos_solution(h_perturbed)
        elapsed = time.time() - start_time
        
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_perturbed.copy()
            print(f"Improved C5 = {best_c5:.6f}")
            
        # Early exit if we're close to target
        if best_c5 <= 0.379:
            break
            
        # Update h for next iteration
        h = h_perturbed.copy()
        
        # Allow some random jumps
        if np.random.rand() < 0.1:  # 10% chance to restart with new perturbation
            h = h_perturbed.copy()
    
    # Final refinement using scipy minimize
    def scipy_objective(h_vec):
        return evaluate_erdos_solution(h_vec)
    
    result = minimize(scipy_objective, best_h, method='SLSQP', 
                      bounds=bounds, constraints=cons,
                      options=options, 
                      callback=lambda x: None)
    
    if result.success and result.fun < best_c5:
        print(f"Scipy optimization improved C5 to {result.fun:.6f}")
        best_h = result.x
        best_c5 = result.fun
    
    # Final verification
    final_c5 = evaluate_erdos_solution(best_h)
    print(f"Final C5 = {final_c5:.6f}")
    
    return (best_h.tolist(), final_c5, n_points)
