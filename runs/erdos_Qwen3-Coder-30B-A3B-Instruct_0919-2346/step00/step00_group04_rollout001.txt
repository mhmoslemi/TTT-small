def evaluate_h(h_values):
    """
    Evaluate the C5 constant for given step function h
    """
    n = len(h_values)
    dx = 2.0 / n
    # Compute correlation: O_k = sum_i h[i] * (1 - h[i+k])
    correlation = np.correlate(h_values, 1 - h_values, mode='full')
    # The correlation array has length 2*n-1
    # The middle element (index n-1) corresponds to k=0
    # We want maximum over all shifts k != 0
    max_correlation = max(correlation[:n-1], correlation[n+1:])
    c5_bound = max_correlation * dx
