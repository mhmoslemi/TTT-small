import numpy as np
import time

def c5_of(h):
    """Compute C5 from discretized h."""
    dx = 2.0 / len(h)
    # correlate h with (1-h) - this gives overlap integrals
    # The result has length 2*N-1 with k in [-N+1, N-1]
    # But we are interested in max over all k. The max is over the middle values
    # actually np.correlate(h, 1-h, mode='full') computes the convolution h * (1-h)
    # which computes sums h[i] * (1-h[i+k]) over all valid i,j pairs where j=i+k
    # We want to compute the maximum of these over all shifts k
    # The peak corresponds to the maximum correlation.
    overlaps = np.correlate(h, 1 - h, mode='full')  # Length 2*N - 1
    c5 = max(overlaps) * dx
    return c5

def propose_move(h, max_step):
    """
    Propose a new h by adjusting two indices.
    Ensures that the sum constraint remains preserved.
    """
    h_new = h.copy()
    N = len(h)
    # Choose two distinct indices
    i, j = np.random.choice(N, size=2, replace=False)
    # Compute feasible range for delta
    delta_min = max(-h_new[i], h_new[j] - 1.0)
    delta_max = min(1.0 - h_new[i], h_new[j])
    # Check if there's any feasible delta
    if delta_min >= delta_max:
        # Cannot make this move, return unchanged
        return h_new
    # Sample delta from [delta_min, delta_max] within [max_step]
    step_size = min(max_step, delta_max - delta_min)
    if step_size <= 0:
        return h_new
    delta = np.random.uniform(-step_size, step_size)
    h_new[i] += delta
    h_new[j] -= delta
    return h_new

def anneal(initial_h, time_budget):
    start_time = time.time()
    cur = initial_h.copy()
    c_cur = c5_of(cur)
    best = cur.copy()
    c_best = c_cur

    # Annealing parameters
    T_start = 0.05
    T_end = 1e-6
    step_start = 0.2
    N = len(initial_h)
    dx = 2.0 / N

    def schedule_temperature(t):
        elapsed = time.time() - start_time
        if elapsed >= time_budget:
            return 0
        frac = elapsed / time_budget
        return T_start * (T_end / T_start) ** frac

    def schedule_stepsize(t):
        elapsed = time.time() - start_time
        if elapsed >= time_budget:
            return 0
        frac = elapsed / time_budget
        return step_start * (1.0 - frac)

    while time.time() - start_time < time_budget:
        # Temperature
        T = schedule_temperature(time.time())
        # Step size
        step = schedule_stepsize(time.time())
        # Proposal
        h_proposed = propose_move(cur, step)
        c_proposed = c5_of(h_proposed)
        # Accept/Reject
        if c_proposed < c_cur:
            cur = h_proposed
            c_cur = c_proposed
            if c_proposed < c_best:
                best = h_proposed.copy()
                c_best = c_proposed
        else:
            if T > 0 and np.random.rand() < np.exp(-(c_proposed - c_cur) / T):
                cur = h_proposed
                c_cur = c_proposed

    return best, c_best

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    N = 140
    dx = 2.0 / N
    time_per_run = 0.30 * budget_s
    best_h = None
    best_c5 = float('inf')
    time_remaining = budget_s

    # Seed 1: Uniform distribution
    h1 = 0.5 * np.ones(N)
    h1 = h1 * (N/2) / np.sum(h1)
    h1_final, c1 = anneal(h1, time_per_run)
    if c1 < best_c5:
        best_h, best_c5 = h1_final, c1

    # Seed 2: Provided initial_h_values interpolated to N points
    if 'initial_h_values' in globals():
        h0 = np.interp(np.linspace(0, 2, N), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
        # Renormalize to sum = N/2
        h0 = h0 * (N/2) / np.sum(h0)
        h0_final, c0 = anneal(h0, time_per_run)
        if c0 < best_c5:
            best_h, best_c5 = h0_final, c0

    # Seed 3: Double peak
    a = 0.3
    u = 0.9
    v = 0.1
    h2 = np.ones(N)
    x = np.linspace(0, 2, N)
    mask = (x <= a) | (x >= 2 - a)
    h2[mask] = u
    h2[~mask] = v
    h2 = h2 * (N/2) / np.sum(h2)
    h2_final, c2 = anneal(h2, time_per_run)
    if c2 < best_c5:
        best_h, best_c5 = h2_final, c2

    # Final polishing phase
    if best_h is not None:
        # Set up a modified anneal that only accepts non-worsening moves (T=0)
        def polish(h_start, time_budget):
            start_time = time.time()
            cur = h_start.copy()
            c_cur = c5_of(cur)
            best = cur.copy()
            c_best = c_cur
            while time.time() - start_time < time_budget:
                step = 0.02  # Very small step
                h_proposed = propose_move(cur, step)
                c_proposed = c5_of(h_proposed)
                if c_proposed <= c_cur:
                    cur = h_proposed
                    c_cur = c_proposed
                    if c_proposed < c_best:
                        best = h_proposed.copy()
                        c_best = c_proposed
            return best, c_best

        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time > 1:
            polished_h, polished_c5 = polish(best_h, remaining_time)
            if polished_c5 < best_c5:
                best_h, best_c5 = polished_h, polished_c5

    return (best_h, best_c5, N)
