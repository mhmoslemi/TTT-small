    """Evaluate the C5 constant for a given step function h"""
    n = len(h)

    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value of the correlation (scaled by dx) is the C5 constant
    c5 = np.max(correlation) * dx
    return c5
