def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value among all lags
    max_corr = np.max(correlation)
    return max_corr * dx

def project_to_feasible(h):
    """Project h to feasible region: [0,1] and integral = 1."""
    h = np.clip(h, 0, 1)
    n = len(h)
    dx = 2.0 / n
    integral = np.sum(h) * dx
    if integral > 0:
        h = h / integral * 1.0  # Scale to make integral = 1
    return np.clip(h, 0, 1)
