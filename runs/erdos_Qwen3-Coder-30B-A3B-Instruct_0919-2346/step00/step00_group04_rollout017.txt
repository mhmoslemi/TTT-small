def evaluate_h(h):
    """Evaluate the C5 bound for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # The correlation is of length 2*n - 1, we want the max over valid shifts
    # The valid shifts correspond to indices n-1 to 2*n-2 (inclusive)
    # But we want max of all these values, so we can just take max of corr
    max_corr = np.max(corr)
    return max_corr * dx
