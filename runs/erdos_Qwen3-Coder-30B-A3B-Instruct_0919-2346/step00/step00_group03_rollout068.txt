import numpy as np

def evaluate_overlap(h):
    # h is a vector of length n
    n = len(h)
    dx = 2.0 / n
    # Compute the full correlation using FFT
    H = fft(h)
    G = fft(1 - h)
    corr = ifft(H * np.conj(G)).real
    max_overlap = max(corr)
    return dx * max_overlap

def project_to_simplex(h, target_sum):
    # Project h onto the constraint sum(h) = target_sum
    # Using Dykstra's alternating projection method
    h = np.array(h)
    current_sum = np.sum(h)
    delta = (current_sum - target_sum) / len(h)
    h = h - delta
    # Clamp to [0, 1] and reproject if needed
    h = np.clip(h, 0, 1)
    
    # One more iteration with Dykstra's method to improve projection
    for _ in range(10):
        current_sum = np.sum(h)
        delta = (current_sum - target_sum) / len(h)
        h = h - delta
        h = np.clip(h, 0, 1)
        if abs(current_sum - target_sum) < 1e-10:
            break

    return h

def mismatch_swap_refinement(h, target_c5):
    n = len(h)
    dx = 2.0 / n
    # Compute initial correlation
    H = fft(h)
    G = fft(1 - h)
    corr = ifft(H * np.conj(G)).real
    s_max = max(corr)
    
    max_shift = np.argmax(corr)
    # identify mismatches
    i = max_shift
    M_1 = []
    M_0 = []
    for k in range(n):
        if k + i < n:
            if h[k] >= 0.99 and h[k+i] <= 0.01:
                M_1.append(k)
            elif h[k] <= 0.01 and h[k+i] >= 0.99:
                M_0.append(k)

    swap_count = 0
    max_swaps = 5000
    
    while s_max > target_c5 and swap_count < max_swaps:
        if not M_1 or not M_0:
            # if mismatches empty, perturb (exchange 2 small amounts) 
            i1 = np.argmax(h)
            i2 = np.argmin(h)
            delta = min(h[i1], 1 - h[i2]) * 0.01
            if delta > 0:
                h[i1] -= delta
                h[i2] += delta
                h = project_to_simplex(h, n / 2)
            # recompute mismatches
            H = fft(h)
            G = fft(1 - h)
            corr = ifft(H * np.conj(G)).real
            s_max = max(corr)
            max_shift = np.argmax(corr)
            M_1 = []
            M_0 = []
            for k in range(n):
                if k + max_shift < n:
                    if h[k] >= 0.99 and h[k+max_shift] <= 0.01:
                        M_1.append(k)
                    elif h[k] <= 0.01 and h[k+max_shift] >= 0.99:
                        M_0.append(k)
        else:
            # Perform swap
            k1 = M_1.pop(0)
            k2 = M_0.pop(0)
            h[k1], h[k2] = h[k2], h[k1]
            # project
            h = project_to_simplex(h, n / 2)
            # recompute correlation
            H = fft(h)
            G = fft(1 - h)
            corr = ifft(H * np.conj(G)).real
            s_max = max(corr)
            max_shift = np.argmax(corr)
            
            # Update mismatches
            M_1 = []
            M_0 = []
            for k in range(n):
                if k + max_shift < n:
                    if h[k] >= 0.99 and h[k+max_shift] <= 0.01:
                        M_1.append(k)
                    elif h[k] <= 0.01 and h[k+max_shift] >= 0.99:
                        M_0.append(k)
            swap_count += 1

    return h, s_max

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Initial n (will refine if necessary)
    n = 180
    dx = 2.0 / n

    # Load initial solution
    initial = initial_h_values
    # Interpolate to length n
    h_interp = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, len(initial)), initial)
    h_start = project_to_simplex(h_interp, n / 2)

    # Local optimization using Differential Evolution
    def objective(h):
        return evaluate_overlap(h)
    
    bounds = [(0, 1) for _ in range(n)]
    # Project is performed inside DE as needed
    
    # Adapt DE to previous best
    x0 = h_start
    best_h = x0
    best_value = float('inf')
    
    # Use the progress of DE
    eval_count = 0
    
    # DE with a fixed number of iterations as a heuristic (since we control time)
    # This is a bit simplified, better to adapt to running inside time budget
    # Use a reasonable number of generations
    max_generations = 10000
    popsize = 30
    F = 0.8
    CR = 0.9
    
    # Initial population -- load from h_start + noise and random
    pop = [x0.copy()]
    for _ in range(popsize - 1):
        h = x0 + np.random.normal(0, 0.05, n)
        h = project_to_simplex(h, n / 2)
        pop.append(h)
    pop.append(x0)
    
    # Reduce popsize for early termination
    if time.time() - start_time > 0.8 * budget_s:
        popsize = 15
    
    gen = 0
    last_improvement = 0
    while gen < max_generations and (time.time() - start_time) < budget_s - 10:
        gen += 1
        # Check elapsed time
        if (time.time() - start_time) > (budget_s - 20):
            break
        
        # Calculate new population
        new_pop = []
        for i in range(popsize):
            # Pick three distinct indices
            a, b, c = np.random.choice(popsize, 3, replace=False)
            v = pop[a] + F * (pop[b] - pop[c])
            # crossover
            u = np.copy(pop[i])
            jrand = np.random.randint(n)
            for j in range(n):
                if np.random.rand() < CR or j == jrand:
                    u[j] = v[j]
            
            # Project to feasible simplex
            u = project_to_simplex(u, n / 2)
            
            # Evaluate
            value_u = evaluate_overlap(u)
            value_i = evaluate_overlap(pop[i])
            if value_u <= value_i:
                new_pop.append(u)
            else:
                new_pop.append(pop[i])
        
        pop = new_pop
        
        # Evaluate best in population
        values = [evaluate_overlap(h) for h in pop]
        min_idx = np.argmin(values)
        if values[min_idx] < best_value:
            best_value = values[min_idx]
            best_h = pop[min_idx].copy()
            last_improvement = gen
        elif gen - last_improvement > 300:  # if stagnated, restart
            # Add some random individuals to diversity
            for _ in range(5):
                h = np.random.rand(n)
                h = project_to_simplex(h, n / 2)
                pop.append(h)
            last_improvement = gen

    if gen >= max_generations:
        print("Reached max generations")

    # Final refinement with mismatch swap
    refined_h, final_c5 = mismatch_swap_refinement(best_h, best_value)

    # Final validate
    final_c5 = evaluate_overlap(refined_h)
    print("Final C5", final_c5)

    return (refined_h.tolist(), final_c5, n)
