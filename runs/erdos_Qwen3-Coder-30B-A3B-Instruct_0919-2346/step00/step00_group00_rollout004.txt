import numpy as np
import time
from scipy import signal

def project_to_capped_simplex(h, target_sum):
    """Project h onto the capped simplex with sum = target_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Water filling
    sorted_h = np.sort(h)[::-1]
    cumsum = np.cumsum(sorted_h)
    n = len(h)
    
    # Find the threshold
    idx = -1
    for i in range(n):
        if (target_sum - cumsum[i]) / (i + 1) < sorted_h[i]:
            idx = i
            break
    
    if idx == -1:
        # All elements would be negative, set to 0
        return np.zeros_like(h)
    else:
        # idx-th largest element is threshold
        threshold = (target_sum - cumsum[idx]) / (idx + 1)
        # Apply threshold
        h_proj = np.clip(h - threshold, 0, 1)
        return h_proj

def compute_correlation_gradient(h, w):
    """Compute gradient of correlation surrogate using convolution."""
    n = len(h)
    # w * (1 - h) convolution
    c1 = signal.convolve(w, 1 - h, mode='full')
    # w * h convolution with reversed w
    w_rev = w[::-1]
    c2 = signal.convolve(w_rev, h, mode='full')
    # Take the part that matches original size
    c1 = c1[n-1:2*n-1]
    c2 = c2[n-1:2*n-1]
    return c1 - c2

def logsumexp_naive(x, axis=None):
    """Naive logsumexp to avoid numerical issues."""
    max_x = np.max(x, axis=axis, keepdims=True)
    return max_x + np.log(np.sum(np.exp(x - max_x), axis=axis, keepdims=True))

def evaluate_c5(h):
    """Evaluate C5 directly using np.correlate"""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize seeds
    init_h_list = []
    
    # Placeholder for initial solution
    if 'initial_h_values' in globals():
        init_h_list.append(initial_h_values)
    else:
        # Random initial
        init_h = np.random.rand(n_points)
        init_h = project_to_capped_simplex(init_h, target_sum)
        init_h_list.append(init_h)
    
    # Add random starting points
    for i in range(5):
        h = np.random.rand(n_points)
        h = project_to_capped_simplex(h, target_sum)
        init_h_list.append(h)
    
    best_h = None
    best_c5 = float('inf')
    
    # Multi-start line-search points
    max_seeds = len(init_h_list)
    for seed_idx in range(max_seeds):
        if time.time() - start_time > budget_s - 5:
            break
        
        h = init_h_list[seed_idx].copy()
        
        # Annealing parameters
        alpha = 0.5
        gamma = 1.3
        eta = 0.5
        eta_decay = 0.99
        max_alpha = 200
        min_improvement = 1e-5
        cycle_interval = 200
        
        last_f_alpha = float('inf')
        h_history = []
        
        iterations = 0
        while time.time() - start_time < budget_s - 5:
            iterations += 1
            
            # Compute all overlap terms
            overlaps = []
            for shift in range(1 - n_points, n_points):
                if shift >= 0:
                    o = np.sum(h[:n_points-shift] * (1 - h[shift:]))
                else:
                    o = np.sum(h[-shift:] * (1 - h[:n_points+shift]))
                overlaps.append(o)
            
            overlaps = np.array(overlaps)
            
            # Compute LSE surrogate
            f_alpha = (dx/alpha) * np.log(np.sum(np.exp(alpha * overlaps)))
            
            # Compute normalized weights for gradient
            exp_alpha_o = np.exp(alpha * overlaps)
            weights = exp_alpha_o / np.sum(exp_alpha_o)
            
            # Compute gradient
            grad = compute_correlation_gradient(h, weights)
            
            # Update h
            h = h - eta * grad
            
            # Project onto feasible set
            h = project_to_capped_simplex(h, target_sum)
            
            # Annealing
            if iterations % cycle_interval == 0:
                if f_alpha >= last_f_alpha and last_f_alpha - f_alpha < min_improvement:
                    break
                last_f_alpha = f_alpha
                if alpha >= max_alpha:
                    break
                alpha *= gamma
                eta *= eta_decay
            
            # Store for binary rounding
            if iterations % 10 == 0:
                h_history.append(h.copy())
                
            # Periodically check for early exit via time
            if time.time() - start_time > budget_s - 5:
                break
        
        # Rounding to binary mask
        sorted_indices = np.argsort(h)[::-1]
        h_bin = np.zeros_like(h)
        h_bin[sorted_indices[:int(target_sum)]] = 1.0
        
        # Evaluate exact C5
        true_c5 = evaluate_c5(h_bin)
        if true_c5 < best_c5:
            best_c5 = true_c5
            best_h = h_bin
        
        # Early termination check
        if time.time() - start_time > budget_s - 5:
            break
    
    # Optional small hill-climb refinement
    if best_h is not None:
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        best_corr_max = np.max(corr)
        
        # Perform hill climb on binary
        proposals = 0
        max_proposals = 10000
        while proposals < max_proposals and time.time() - start_time < budget_s - 5:
            # Pick two random indices
            ones = np.where(best_h == 1)[0]
            zeros = np.where(best_h == 0)[0]
            
            if len(ones) == 0 or len(zeros) == 0:
                break
                
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            
            # Manually compute updated correlation
            corr_update = 0
            for k in range(1 - n_points, n_points):
                delta = 0
                # Add effects where h[i] = 1 and h[j] = 0 change contributions
                if k >= 0:
                    if i >= k: 
                        delta -= best_h[i] * (1 - best_h[j + k]) 
                        delta += (1 - best_h[i]) * (1 - (1 - best_h[j + k]))
                    if j + k < n_points:
                        delta += best_h[i] * (1 - best_h[j + k])
                        delta -= (1 - best_h[i]) * (1 - best_h[j + k])
                else:  # k < 0
                    if j >= -k: 
                        delta -= best_h[i + (-k)] * (1 - best_h[j])
                        delta += (1 - best_h[i + (-k)]) * (1 - (1 - best_h[j]))
                    if i + (-k) < n_points:
                        delta += best_h[i + (-k)] * (1 - best_h[j])
                        delta -= (1 - best_h[i + (-k)]) * (1 - best_h[j])
                        
                corr_update += delta * dx
                
            # Accept if improvement
            if corr_update < 0:
                # Swap positions i and j
                best_h[i], best_h[j] = best_h[j], best_h[i]
                proposals += 1
                
        # Final computation
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        final_c5 = np.max(corr) * dx
        
        return (best_h, final_c5, n_points)
    
    return (np.zeros(n_points), best_c5, n_points)
