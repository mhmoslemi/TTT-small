import numpy as np

def project_onto_capped_simplex(v, S):
    """Project vector v onto the capped simplex {h : 0 <= h_i <= 1, sum h_i = S}"""
    v = np.asarray(v)
    low, high = -2.0, 2.0
    for _ in range(50):
        mid = (low + high) / 2
        h = np.clip(v - mid, 0, 1)
        s = np.sum(h)
        if s > S:
            low = mid
        else:
            high = mid
    h = np.clip(v - (low + high)/2, 0, 1)
    return h

def evaluate(h):
    """Evaluate the overlap C5 using correlation."""
    n = len(h)
    dx = 2.0 / n
    h_ext = np.concatenate([h, np.zeros(n-1)])
    h1_ext = np.concatenate([np.zeros(n-1), 1-h])
    corr = np.correlate(h_ext, h1_ext, mode='full')
    max_overlap = np.max(corr)
    return max_overlap * dx

def coeffs_and_const(h):
    """Compute coefficient matrix and const vector for LP."""
    n = len(h)
    a_matrix = []
    const_vec = []
    for k in range(-(n-1), n):
        a = np.zeros(n)
        c = 0.0
        for i in range(n):
            if 0 <= i + k < n:
                a[i] = 1.0 - h[i] - h[i + k]
                c += h[i] * h[i + k]
            else:
                a[i] = 1.0 - h[i] - 0.0  # zero pad
                # ignore c contribution  
        a_matrix.append(a)
        const_vec.append(c)
    return np.array(a_matrix), np.array(const_vec)

def linear_program(a_matrix, const_vec, n, S):
    """Solve the linear program for current iteration."""
    # Create variables
    h_vars = cp.Variable(n)
    t = cp.Variable()

    # Objective
    objective = cp.Minimize(t)

    # Constraints
    constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == S]
    for i in range(len(a_matrix)):
        constraints.append(a_matrix[i] @ h_vars + const_vec[i] <= t)

    # Solve
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status == cp.OPTIMAL:
            h_new = h_vars.value
            t_val = t.value
            return h_new, t_val
        else:
            # fallback in case optimization fails
            h_new = np.zeros(n)
            t_val = 1e10
            return h_new, t_val
    except:
        h_new = np.zeros(n)
        t_val = 1e10
        return h_new, t_val

def random_feasible(n):
    """Generate a random feasible vector."""
    v = np.random.random(n)
    h = project_onto_capped_simplex(v, n/2)
    return h
