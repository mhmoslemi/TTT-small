import numpy as np
import time
import cvxpy as cp
from scipy.fft import fft, ifft
from scipy.signal import correlate
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    n = 60
    dx = 2.0 / n
    K0 = [1, n//2, 2, n-2]  # initial shifts

    # Build index tables
    idx_mod = np.arange(n)
    idx_mod = np.concatenate([idx_mod, idx_mod])  # for circular indexing

    def get_autocorrelation(h, k):
        # compute circular autocorrelation A_k
        a_k = np.sum(h * np.roll(h, k))
        return a_k

    def solve_milp(K_set, timeout_s):
        # create variables
        h = cp.Variable(n, boolean=True)
        t = cp.Variable()

        # constraints
        constraints = [
            cp.sum(h) == n // 2,  # sum constraint
            h >= 0, h <= 1,
            h[0] == 1  # symmetry breaking
        ]

        # auxiliary variables and constraints for each k in K_set
        w_list = []
        for k in K_set:
            w = cp.Variable(n, boolean=True)
            for i in range(n):
                i1 = i
                i2 = (i + k) % n
                constraints.extend([
                    w[i] <= h[i1],
                    w[i] <= h[i2],
                    w[i] >= h[i1] + h[i2] - 1
                ])
            w_list.append(w)

        # add constraints that t <= a_k for all k
        for i, (k, w) in enumerate(zip(K_set, w_list)):
            a_k = cp.sum(w)
            constraints.append(t <= a_k)

        # solve
        prob = cp.Problem(cp.Maximize(t), constraints)
        try:
            prob.solve(solver=cp.CBC, verbose=False, maxSeconds=timeout_s)
        except:
            return None, None

        if prob.status != cp.OPTIMAL:
            return None, None

        h_sol = h.value
        if h_sol is None:
            return None, None
        return np.round(h_sol).astype(int), t.value

    def compute_exact_C5(h):
        # full autocorrelation via FFT
        H = fft(h)
        A = ifft(H * np.conj(H)).real
        A = np.concatenate([A[n//2:], A[:n//2]])  # circular shift
        A = A[:n]  # fix length
        t_min = np.min(A)
        return 1 - 2*t_min/n

    # solve initial MILP
    K_current = K0.copy()
    timeout_s = min(budget_s / 5, 30)
    h_sol, t_sol = solve_milp(K_current, timeout_s)
    
    if h_sol is None:
        # fallback to random binary
        h_sol = np.zeros(n)
        ones_positions = np.random.choice(n, n//2, replace=False)
        h_sol[ones_positions] = 1
        h_sol = h_sol.astype(int)
    
    # guardar mejor solución antes de procesar
    best_h = h_sol.copy()
    best_C5 = compute_exact_C5(h_sol)
    
    # cut generation loop
    cut_gen_timeout = max(0.1, (budget_s - (time.time() - start_time)) / 3)
    
    while True:
        if time.time() - start_time >= budget_s - 20:
            break
        # compute all autocorrelations for full evaluation
        all_A = []
        for k in range(1, n):
            a_k = get_autocorrelation(h_sol, k)
            all_A.append(a_k)
        all_A = np.array(all_A)
        a_min = np.min(all_A)
        
        # check termination condition
        if a_min >= t_sol:
            break

        # add the violating shift
        k_violating = np.argmin(all_A) + 1
        if k_violating not in K_current:
            K_current.append(k_violating)
        
        h_sol, t_sol = solve_milp(K_current, cut_gen_timeout)
        if h_sol is None:
            break
        
        # keep best
        new_C5 = compute_exact_C5(h_sol)
        if new_C5 < best_C5:
            best_h = h_sol.copy()
            best_C5 = new_C5

    # optionally refine with gradient
    def project_onto_sum(h, target_sum):
        """Project h onto hyperplane sum(h) = target_sum using pairwise swaps."""
        h = np.array(h, dtype=float)
        current_sum = np.sum(h)
        diff = target_sum - current_sum
        
        if abs(diff) < 1e-10:
            return h
        
        # find indices to adjust
        mask = h > 0
        if not np.any(mask):
            return h
        
        # move from ones from >0 indices to <1 (if needed)
        target = h.copy()
        if diff > 0:
            # need to increase sum - move from 0s to 1s
            mask_neg = h < 1
            if np.sum(mask_neg) > 0:
                idx_src = np.where(mask)[0]
                idx_dst = np.where(mask_neg)[0]
                if len(idx_dst) > 0 and len(idx_src) > 0:
                    for i in range(int(diff)):
                        if len(idx_src) == 0 or len(idx_dst) == 0:
                            break
                        j = idx_dst[i % len(idx_dst)]
                        k = idx_src[i % len(idx_src)]
                        d = min(1 - h[j], h[k])
                        if d > 1e-10:
                            h[j] += d
                            h[k] -= d
        elif diff < 0:
            # decrease sum
            idx_src = np.where(mask)[0]
            idx_dst = np.where(h < 1)[0]
            if len(idx_src) > 0:
                for i in range(int(-diff)):
                    if len(idx_src) == 0:
                        break
                    j = idx_src[i % len(idx_src)]
                    k = idx_dst[i % len(idx_dst)]
                    d = min(1 - h[j], h[k])
                    if d > 1e-10:
                        h[j] -= d
                        h[k] += d
        return h

    # gradient ascent with projected steps
    h_cont = best_h.copy().astype(float)
    t_end = time.time() + max(0, budget_s - (time.time() - start_time) - 20)

    def get_grad(h_cont, k_star):
        """Return the projected gradient for a fixed shift k_star."""
        grad = np.zeros(n)
        for i in range(n):
            grad[i] = h_cont[i] + h_cont[(i - k_star) % n]
        return grad

    for step in range(200):
        if time.time() > t_end:
            break
        # compute consensus shift (ignore 0)
        all_A = []
        for k in range(1, n):
            a_k = get_autocorrelation(h_cont, k)
            all_A.append(a_k)
        k_star = np.argmin(all_A) + 1
        grad = get_grad(h_cont, k_star)
        h_cont += 0.01 * grad
        # clip
        h_cont = np.clip(h_cont, 0, 1)
        # project
        h_cont = project_onto_sum(h_cont, n/2)
        # recompute C5
        cur_C5 = compute_exact_C5(h_cont)
        if cur_C5 < best_C5:
            best_h = h_cont.copy()
            best_C5 = cur_C5

    if time.time() - start_time < budget_s - 100 and best_C5 > 0.3807:
        # spectral fallback
        def spectral_encoding(alpha):
            if len(alpha) > 4:
                alpha = alpha[:4]
            arg = np.arange(n) * 2 * np.pi / n
            h = 0.5 + np.dot(alpha, np.cos(arg * np.arange(1, len(alpha)+1)[:, None]))
            h = np.clip(h, 0, 1)
            return h

        def obj_spectral(alpha):
            try:
                h = spectral_encoding(alpha)
                # compute lower bound on min autocorrelation
                H = fft(h)
                A = ifft(H * np.conj(H)).real
                A = np.concatenate([A[n//2:], A[:n//2]])  # circular shift
                A = A[:n]  # fix length
                # surrogate for logsumexp
                beta = 10
                lower = -1/beta * np.log(np.sum(np.exp(-beta*A)))
                return -lower  # maximize
            except:
                return np.inf

        bounds = [(-0.5, 0.5)] * 4
        opt_res = differential_evolution(
            obj_spectral, bounds, maxiter=100, popsize=30, seed=seed,
            strategy='best1bin', tol=1e-5, callback=None
        )
        opt_alpha = opt_res.x
        ref_h = spectral_encoding(opt_alpha)
        ref_h = np.clip(ref_h, 0, 1)
        # make it binary with correct sum
        ref_h_binary = np.zeros(n)
        sorted_indices = np.argsort(ref_h)
        ones_count = n // 2
        ref_h_binary[sorted_indices[int(n - ones_count):]] = 1
        cur_C5 = compute_exact_C5(ref_h_binary)
        if cur_C5 < best_C5:
            best_h = ref_h_binary
            best_C5 = cur_C5

    return (best_h, best_C5, n)
