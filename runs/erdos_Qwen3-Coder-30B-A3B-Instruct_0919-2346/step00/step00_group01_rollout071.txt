def evaluate_erdos_solution(h, n_points):
    """Evaluate the Erdos overlap constant for given step function h"""
    dx = 2.0 / n_points
    # Compute correlation of h with (1-h) using full mode
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def evaluate_c5(h, n_points):
    """Wrapper for C5 evaluation"""
    return evaluate_erdos_solution(h, n_points)

def initial_h_values(n_points=47):
    """Initial construction from problem"""
    # Create binary vector with approximately half ones
    h_initial = np.zeros(n_points)
    # fill with 1s roughly in the middle and edges to reduce overlap
    half_length = n_points // 2
    h_initial[:half_length] = 1
    return h_initial

def project_to_feasible(h, n_points):
    """Project h onto feasible region: [0,1] and sum = n/2"""
    # Clip h to [0,1]
    h = np.clip(h, 0, 1)
    # Adjust to make sum = n/2
    current_sum = np.sum(h)
    desired_sum = n_points / 2.0
    diff = desired_sum - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Adjust elements to meet sum constraint
    idxs = np.argsort(h)
    delta = diff / len(h)
    
    if diff > 0:
        # Need to increase values
        h = np.minimum(h + delta, 1.0)
    else:
        # Need to decrease values
        h = np.maximum(h + delta, 0.0)
    
    # Normalize edges to meet sum exactly
    current_sum = np.sum(h)
    diff = desired_sum - current_sum
    
    if abs(diff) > 1e-10:
        # Adjust smaller values first
        idxs = np.argsort(h)
        for i in range(len(idxs)):
            idx = idxs[i]
            if h[idx] + diff <= 1.0:
                h[idx] += diff
                diff = 0
                break
            else:
                diff -= (1.0 - h[idx])

                if h[idx] + diff >= 0.0:
                    h[idx] += diff
                    break
                else:
                    diff += h[idx]
                    h[idx] = 0.0

    return h
