import numpy as np
from scipy.optimize import minimize
import time
from math import sqrt

def evaluate_erdos_solution(h_values):
    """Evaluate the C_5 constant for given h values"""
    n = len(h_values)
    dx = 2.0 / n
    # Compute convolution of h with (1-h)
    corr = np.correlate(h_values, 1 - np.array(h_values), mode="full")
    # Find maximum value
    max_val = np.max(corr) * dx
    return max_val

def initial_h_values():
    """Initial construction for h"""
    # From the problem, we have a 47-point approximation
    n = 47
    h = np.ones(n)
    # Simple construction: a step function
    # Need integral = 1, so each step should have height 1/(2/n) = n/2
    # Let's build a piecewise constant function with 2 steps for simplicity
    # This is a simplified starting point
    h = np.ones(n) * 0.5
    # Make it look more like a step function
    mid = n // 2
    h[:mid] = 1.0
    h[mid:] = 0.0
    # Adjust heights to make integral = 1
    h = h * (n / (2 * np.sum(h)))
    return h

def run(seed=42, budget_s=600, **kwargs):

    bounds = [(0, 1) for _ in range(n_points)]

    def objective(h_vec):
        h = np.array(h_vec)

        h = h * (n_points / (2 * np.sum(h)))
        return evaluate_erdos_solution(h)
    
    def constraint_func(h_vec):
        h = np.array(h_vec)
        return np.sum(h) - n_points / 2  # This must be zero
    
    # Add constraint: integral should be 1
