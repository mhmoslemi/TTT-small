def evaluate_solution(h, dx):
    """Evaluate the overlap integral C5 for a given h."""
    n = len(h)
    # Compute correlation using FFT for efficiency
    H = fft(h)
    H_conj = np.conj(H)
    one_minus_h = np.ones_like(h) - h
    one_minus_H = fft(one_minus_h)
    correlation = ifft(H * one_minus_H).real
    # The maximum overlap is over all shifts, which corresponds to the maximum 
    # value in the full correlation (mode='full')
    max_overlap = np.max(correlation)
    return dx * max_overlap


def project_to_feasible(h, n_half):
    """Project the vector h onto the simplex ∑ h_i = n_half, and clip to [0,1]."""
    # Avoid any violations due to numerical errors
    h = np.clip(h, 0, 1)
    # Calculate the difference from the target sum
    sum_h = np.sum(h)
    diff = sum_h - n_half
    if abs(diff) > 1e-10:
        # Affine rescaling to make sum equal to n_half
        # Simple shift: subtract average violation
        scale = diff / len(h)
        h = h - scale
        # Occasionally project back to [0,1] to handle overshoot
        h = np.clip(h, 0, 1)
    return h


def de_optimize(initial_h, n, dx, budget_s):
    """Differential Evolution optimization with projected updates."""
    best_h = initial_h.copy()
    best_value = evaluate_solution(best_h, dx)
    start_time = time.time()
    
    # Parameters for DE
    popsize = 30
    maxiter = int(budget_s * 10)  # rough estimate
    F = 0.8
    CR = 0.9
    generation = 0
    
    # Population initialization
    population = [initial_h.copy() for _ in range(popsize)]
    for i in range(1, popsize):
        # add some noise and project
        noise = np.random.normal(0, 0.05, n)
        population[i] = np.clip(initial_h + noise, 0, 1)
        population[i] = project_to_feasible(population[i], n/2)
    
    # Storage for best individual
    current_best = best_h.copy()
    current_best_val = best_value
    
    # Continue for maxiter or until time runs out
    while generation < maxiter and time.time() - start_time < budget_s - 30:
        if generation % 50 == 0:
            print(f"DE: Generation {generation}, current best: {current_best_val:.6f}")
        
        next_population = []
        for i in range(popsize):
            # Select three distinct individuals
            idxs = list(range(popsize))
            idxs.remove(i)
            a, b, c = [np.array(population[j]) for j in np.random.choice(idxs, 3, replace=False)]
            
            # Differential evolution mutation
            mutant = a + F * (b - c)
            # Apply crossover
            trial = np.where(np.random.rand(n) < CR, mutant, population[i])
            # Project into feasible region
            trial = project_to_feasible(trial, n/2)
            # Evaluate
            trial_val = evaluate_solution(trial, dx)
            # Keep better solution
            if trial_val <= evaluate_solution(population[i], dx):
                next_population.append(trial)
                if trial_val < current_best_val:
                    current_best_val = trial_val
                    current_best = trial.copy()
            else:
                next_population.append(population[i])
                
        population = next_population
        generation += 1
        if current_best_val < best_value:
            best_value = current_best_val
            best_h = current_best.copy()
    
    return best_h, best_value


def round_to_binary(h, n):
    """Heuristic rounding to binary, preserving the sum."""
    h = np.clip(h, 0, 1)
    # Sort indices by descending values
    indices = np.argsort(h)[::-1]
    # Determine target number of 1s
    target_ones = int(n / 2)
    h_binary = np.zeros(n)
    count = 0
    for i in indices:
        if count < target_ones:
            h_binary[i] = 1
            count += 1
        else:
            h_binary[i] = 0
    
    # Adjust to exactly match sum
    total = np.sum(h_binary)
    if total < target_ones:
        # Add some 1s to the zero slots
        zero_indices = np.where(h_binary == 0)[0]
        for i in range(int(target_ones - total)):
            if i < len(zero_indices):
                h_binary[zero_indices[i]] = 1
    elif total > target_ones:
        # Remove some 1s from the one slots
        one_indices = np.where(h_binary == 1)[0]
        for i in range(int(total - target_ones)):
            if i < len(one_indices):
                h_binary[one_indices[-(i+1)]] = 0
    return h_binary
