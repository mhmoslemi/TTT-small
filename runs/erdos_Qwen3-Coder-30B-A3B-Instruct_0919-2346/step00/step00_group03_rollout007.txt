import numpy as np

def evaluate_erdos_solution(h):
    """Compute C5 bound for the given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    conv = np.correlate(h, 1 - h, mode='full')
    # The maximum overlap corresponds to the maximum value in the convolution
    max_overlap = np.max(conv)
    # Scale by dx to get the actual integral value
    c5 = max_overlap * dx
    return c5

    """Constraint that integral of h equals 1"""
    dx = 2.0 / n_points

def objective(h, n_points):
    """Objective function to minimize: C5"""
    # Normalize h to satisfy the constraint
    dx = 2.0 / n_points
    h_normalized = np.clip(h, 0, 1)
    # Ensure the integral constraint is exactly satisfied
    current_sum = np.sum(h_normalized) * dx
    scale_factor = 1.0 / current_sum
    h_normalized = h_normalized * scale_factor
    return evaluate_erdos_solution(h_normalized)
