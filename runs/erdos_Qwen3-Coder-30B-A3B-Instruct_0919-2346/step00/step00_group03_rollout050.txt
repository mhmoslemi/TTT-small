import numpy as np
import time
import random

def evaluate_c5(h):
    """Evaluate the C5 bound for a given h."""
    dx = 2.0 / len(h)
    correlation = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(correlation)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Parameters
    n_points = 200
    dx = 2.0 / n_points
    target_c5 = 0.3807
    
    # Start timing
    start_time = time.time()
    
    # Initialize with random binary vector
    h_values = np.random.choice([0, 1], size=n_points, p=[0.5, 0.5]).astype(float)
    h_values = h_values * (n_points / 2 / np.sum(h_values))  # normalize to sum = n_points/2
    
    # Ensure that sum is exactly n/2
    h_values = h_values * (n_points / 2 / np.sum(h_values))
    h_values = np.clip(h_values, 0, 1)
    
    # Check if we get exact sum, if not adjust one element
    current_sum = np.sum(h_values)
    if abs(current_sum - n_points / 2) > 1e-10:
        # Adjust last element to make it exact
        h_values[-1] += (n_points / 2 - current_sum)
        h_values[-1] = max(0, min(1, h_values[-1]))

    # Evaluate initial solution
    current_c5 = evaluate_c5(h_values)
    best_h_values = h_values.copy()
    best_c5 = current_c5

    # Annealing parameters
    temperature = 0.1
    cooling_rate = 0.99995
    step_size = 0.05
    
    iteration_count = 0
    
    # Main optimization loop
    while time.time() - start_time < budget_s * 0.95:
        iteration_count += 1
        
        # Propose a change
        i, j = random.sample(range(n_points), 2)
        delta = random.uniform(-step_size, step_size)
        
        # Apply change
        new_h = h_values.copy()
        old_i = new_h[i]
        old_j = new_h[j]
        
        new_h[i] = max(0, min(1, old_i + delta))
        new_h[j] = max(0, min(1, old_j - delta))
        
        # Ensure sum constraint
        current_sum = np.sum(new_h)
        if abs(current_sum - n_points / 2) > 1e-10:
            adjustment = (n_points / 2 - current_sum) * (1.0 / n_points)
            new_h += adjustment
        
        # Evaluate new solution
        new_c5 = evaluate_c5(new_h)
        
        # Accept or reject
        delta_c5 = new_c5 - current_c5
        
        if delta_c5 <= 0 or random.random() < np.exp(-delta_c5 / temperature):
            h_values = new_h
            current_c5 = new_c5
            
            if current_c5 < best_c5:
                best_h_values = h_values.copy()
                best_c5 = current_c5
        
        # Update temperature
        temperature *= cooling_rate
        
        # Periodic check for early stopping
        if iteration_count % 1000 == 0:
            if time.time() - start_time > budget_s * 0.9:
                break
    
    # Final evaluation of best solution
    final_c5 = evaluate_c5(best_h_values)
    
    return (best_h_values.tolist(), final_c5, n_points)
