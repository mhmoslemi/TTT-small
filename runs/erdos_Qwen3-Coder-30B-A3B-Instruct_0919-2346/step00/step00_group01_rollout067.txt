import math
import time
import numpy as np
from scipy import optimize
from scipy.fft import fft, ifft
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h):
    n = len(h)
    # Use FFT to compute circular correlation efficiently
    # zero-pad to length 2*n to avoid wraparound effects
    h_padded = np.concatenate([h, np.zeros(n)])
    h_shifted = np.concatenate([1-h, np.zeros(n)])
    corr = ifft(fft(h_padded) * np.conj(fft(h_shifted))).real[:n]
    # The overlap integral for shift k corresponds to corr[k] / n
    # The C5 bound is 1 - (2/n) * min(corr)
    min_corr = min(corr)
    c5 = 1 - (2 / n) * min_corr
    return c5

def sdp_solve(h0, n):
    """Solves SDP relaxation to get a vector h and t such that min autocorr >= t"""
    # Variables
    h = cp.Variable(n)
    X = cp.Variable((n, n), symmetric=True)
    t = cp.Variable()
    
    # Constraints
    constr = [
        cp.trace(X) == n / 2,
        X >> 0,
        cp.diag(X) == h,
        0 <= h, h <= 1,
        t <= cp.trace(M0 @ X),
        t <= cp.trace(M1 @ X),
        t <= cp.trace(M2 @ X),
        t <= cp.trace(M3 @ X),
        t <= cp.trace(M4 @ X),
    ]

    # For simplicity, manually create 5 shift matrices M_k (k=0..4)
    # We build the 5 matrices and add to constraints
    # Matrix M_k is a symmetric matrix with 1s at positions (i, i+k) and (i+k, i)
    def create_shift_matrix(k):
        M = np.zeros((n, n))
        for i in range(n):
            ipos = (i + k) % n
            M[i, ipos] = 1
            M[ipos, i] = 1
        return M
        
    M0 = create_shift_matrix(0)
    M1 = create_shift_matrix(1)
    M2 = create_shift_matrix(2)
    M3 = create_shift_matrix(3)
    M4 = create_shift_matrix(4)
    
    # Build complete constraints
    constr = [cp.trace(X) == n / 2,
              X >> 0,
              cp.diag(X) == h,
              0 <= h, h <= 1]
    
    for M in [M0, M1, M2, M3, M4]:
        constr.append(t <= cp.trace(M @ X))

    # Objective
    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constr)
    prob.solve(solver=ECOS, verbose=False)
    
    if prob.status in ["optimal", "optimal_inaccurate"]:
        return h.value, t.value
    
    return h0, 0.0

def round_h(X, n):
    """Truncates the SDP solution X to a binary vector using random hyperplane rounding."""
    # Decompose with Cholesky (perhaps modified)
    try:
        L = np.linalg.cholesky(X)
    except np.linalg.LinAlgError:
        return np.zeros(n)
    # Sample normal random vector
    g = np.random.randn(L.shape[1])
    # Compute thresholded vector
    scores = L.T @ g
    sorted_scores = np.sort(scores)[::-1]
    target = n // 2
    # Find threshold such that we have exactly target 1s
    threshold = sorted_scores[target - 1] if target < len(sorted_scores) else sorted_scores[-1]
    h_binary = (scores >= threshold).astype(float)
    # Ensure exact number of ones
    current_sum = int(np.sum(h_binary))
    if current_sum > target:
        # Find the indices with highest scores and set to 0
        indices_ones = np.where(h_binary == 1)[0]
        if len(indices_ones) > 0:
            scores_i = scores[indices_ones]
            idx_to_change = np.argsort(-scores_i)[:current_sum - target]
            for si in idx_to_change:
                h_binary[indices_ones[si]] = 0
    elif current_sum < target:
        # Find the indices with lowest scores and set to 1
        indices_zeros = np.where(h_binary == 0)[0]
        if len(indices_zeros) > 0:
            scores_i = scores[indices_zeros]
            idx_to_change = np.argsort(scores_i)[:target - current_sum]
            for si in idx_to_change:
                h_binary[indices_zeros[si]] = 1
    return h_binary

def swap_hill_climb(h, n):
    """Improve h via deterministic swaps to improve the min autocorrelation."""
    num_swaps = 0
    while True:
        improved = False
        # This is a very simplified version - check a few random swaps
        for i in range(1000):
            if num_swaps > 5000:
                return h
            i1 = np.random.randint(n)
            i2 = np.random.randint(n)
            if i1 == i2:
                continue
            if h[i1] == h[i2]:
                continue
            # Compute change in all autocorrelations
            old_corr = np.zeros(n)
            for k in range(n):
                old_corr[k] = h[i1] * h[(i1+k) % n] + h[i2] * h[i2+k] - (
                    h[i1] * h[(i2+k) % n] + h[i2] * h[(i1+k) % n])
                
            # Try swap
            new_h = h.copy()
            new_h[i1], new_h[i2] = new_h[i2], new_h[i1]
            new_corr = np.zeros(n)
            for k in range(n):
                new_corr[k] = new_h[i1] * new_h[(i1+k) % n] + new_h[i2] * new_h[i2+k] - (
                    new_h[i1] * new_h[(i2+k) % n] + new_h[i2] * new_h[(i1+k) % n])
                
            if all(new_corr >= old_corr):
                h = new_h
                improved = True
                num_swaps += 1
                break
        if not improved:
            break
    return h

def gradient_fine_tune(h, n, num_iter=200):
    """Perform gradient ascent on the minimal correlation."""
    h = h.copy().astype(float)
    for _ in range(num_iter):
        # Compute the circular correlation
        h_padded = np.concatenate([h, np.zeros(n)])
        h_shifted = np.concatenate([1-h, np.zeros(n)])
        corr = np.real(ifft(fft(h_padded) * np.conj(fft(h_shifted))).real[:n])
        min_idx = np.argmin(corr)
        min_corr = corr[min_idx]
        # Gradient w.r.t. h
        # grad_i = 1_{i+min_idx} + 1_{i-min_idx}
        gradient = np.zeros(n)
        gradient[(min_idx + np.arange(n)) % n] += 1
        gradient[(n + np.arange(n) - min_idx) % n] += 1
        # Update with step size 0.01 and projection
        h = h + 0.01 * gradient
        # Project to [0,1] and preserve sum
        h = np.clip(h, 0, 1)
        # Re-project to keep sum = 1
        h = h * (n/2) / np.sum(h)
    return h

def interval_optimize(npoints=200):
    """Use a few continuous intervals to approximate."""
    m = 6  # number of intervals
    # Symmetric case: a1, b1, a2, b2, a3, b3 where a_j + b_j = 2
    # We optimize 3 pairs of symmetric intervals: a,b pairs to yield 6 parameters
    bounds = [(0, 1), (0, 1), (0, 1), (1, 2), (1, 2), (1, 2)]
    def objective(x):
        a1, b1, a2, b2, a3, b3 = x
        # For inversion: (1 - a1) == a3 and (1 - b1) == b3
        assert(abs(a1 + b3 - 2) < 1e-6)
        assert(abs(b1 + a3 - 2) < 1e-6)
        # intervals: [a1, b1], [a2, b2], [a3, b3]
        lengths = [b1 - a1, b2 - a2, b3 - a3]
        if abs(sum(lengths) - 1.0) > 1e-8:
            return 1e6  # penalty
        
        # Compute overlap function for shift k
        min_overlap = 1e6
        for shift_k in range(npoints):
            # Find total overlap of h(x) and h(x+k) 
            # using interval intersection
            total_len = 0.0
            for i1, a1_val in enumerate([a1, a2, a3]):
                b1_val = [b1, b2, b3][i1]
                # Exact interval overlap logic
                # For interval [a1, b1] and [a2+k, b2+k]
                # shifted by shift_k
                a2_shift = a1_val + shift_k / npoints * 2
                b2_shift = b1_val + shift_k / npoints * 2
                # wrap between [0,2]
                if a2_shift >= 2:
                    a2_shift -= 2
                    b2_shift -= 2
                if b2_shift <= 0:
                    a2_shift += 2
                    b2_shift += 2
                    
                intersection = max(0, min(b1_val, b2_shift) - max(a1_val, a2_shift))
                total_len += intersection
                
            if total_len < min_overlap:
                min_overlap = total_len
        return min_overlap
    
    # Try optimization with multiple random initializations
    best_c5 = 1e6
    best_h = None
    for r in range(20000):
        if r % 100 == 0:
            # print(r, best_c5)
            pass
        x0 = np.zeros(6)
        for i in range(3):
            x0[2*i] = np.random.uniform(0, 1)
            x0[2*i+1] = x0[2*i] + np.random.uniform(0, 1-x0[2*i])
        # Symmetrize
        for i in range(3):
            a = x0[2*i]
            b = x0[2*i+1]
            x0[2*i+1] = min(2, b)
            x0[2*(3-i-1)] = 2 - x0[2*i+1]
            x0[2*(3-i-1)+1] = 2 - x0[2*i]
        # Mixed bounds check (0, 1) for first pairs, (1, 2) for later
        x0 = [min(max(x0[i], bounds[i][0]), bounds[i][1]) for i in range(6)]
        try:
            res = optimize.minimize(objective, x0, method="L-BFGS-B", bounds=bounds)
            if res.success:
                c5 = 1 - 2 * res.fun / npoints
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = res.x
        except:
            pass
        if best_c5 < 0.3807:
            break
    return best_c5, best_h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start = time.time()
    n = 80  # Discretization points
    # First, try interval-based heuristic as fallback
    try:
        best_c5_interval, best_h_interval = interval_optimize(npoints=200)
    except:
        best_c5_interval = 1.0
        best_h_interval = None
    
    best_h = None
    best_c5 = 1.0
    best_n = n
    if time.time() - start > budget_s - 50:
        return (best_h, best_c5, best_n)

    for iter in range(30):
        if time.time() - start >= budget_s - 15:
            break
        
        # Run SDP +
        try:
            h0 = np.ones(n) * 0.5  # reasonable initial guess
            h_sdp, t_sdp = sdp_solve(h0, n)
            h_round = round_h(np.outer(h_sdp, h_sdp), n)
            if np.sum(h_round) != n / 2:
                h_round = np.clip(h_round, 0, 1)
            h_swap = swap_hill_climb(h_round, n)
            h_fine = gradient_fine_tune(h_swap, n)
            c5 = evaluate_c5(h_fine)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_fine.copy()
        except Exception as e:
            continue
        if best_c5 <= 0.379:
            break
    
    # fallback interval if SDP doesn't work
    if best_c5 > 0.3807 and best_h_interval is not None and best_c5_interval < 0.3807:
        best_c5 = best_c5_interval
        # Convert best_h_interval to actual step function on npoints=80 grid
        a1, b1, a2, b2, a3, b3 = best_h_interval
        h_final = np.zeros(80)
        for i in range(80):
            x = 2 * i / 80
            # check intervals [a1, b1], [a2, b2], [a3, b3]
            if (a1 <= x <= b1) or (a2 <= x <= b2) or (a3 <= x <= b3):
                h_final[i] = 1
        best_h = h_final
    return (best_h, best_c5, best_n)
