import numpy as np
import time

from scipy.fftpack import fft, ifft
from scipy.spatial.distance import cdist

def project_to_simplex(h, target_sum):
    """
    Project vector h onto the simplex {x : sum(x) = target_sum, 0 <= x <= 1}
    using Dykstra's projection algorithm.
    """
    h = np.array(h, dtype=float)
    current_sum = np.sum(h)
    delta = (current_sum - target_sum) / len(h)
    h -= delta
    h = np.clip(h, 0, 1)
    # Repeat Dykstra projection steps for robustness
    for _ in range(3):
        current_sum = np.sum(h)
        delta = (current_sum - target_sum) / len(h)
        h -= delta
        h = np.clip(h, 0, 1)
    return h

def evaluate_overlap(h, dx):
    """Fast evaluation using FFT."""
    h = np.asarray(h)
    F = fft(h)
    G = fft(1 - h)
    corr = ifft(F * np.conj(G)).real
    return dx * np.max(corr)

def binary_milp_approach(n, timeout_s=30):
    """
    Exact MILP solver for binary h values using CVXPY.
    """
    if n > 70:
        return None, None  # Skip for larger n
    # Binary variables
    h_vars = cp.Variable(n, boolean=True)
    # Offline parameter for scalability
    t = cp.Variable()
    # Constraints for sum
    constraints = [cp.sum(h_vars) == n // 2, t >= 0]
    # Add bilinear constraints using McCormick relaxation
    # We accumulate constraints for all feasible shifts
    num_shifts = 2 * n - 1  # From -(n-1) to (n-1)
    y_vars = {}
    # Build up the coefficient matrix using full correlation
    total_terms = 0
    for l in range(-(n-1), n):
        if abs(l) >= n: continue
        indexes = list(range(max(0, -l), min(n, n-l)))
        total_terms += 1
        expr = cp.sum(h_vars[i] for i in indexes)
        
        y_vars[l] = cp.Variable((len(indexes),), 1)
        for i_idx, i in enumerate(indexes):
            j = i + l
            if 0 <= j < n:
                idx_y = (i, l)
                y_vars[l][i_idx] = cp.Variable(1, boolean=True)
                constraints.extend([
                    y_vars[l][i_idx] <= h_vars[i],
                    y_vars[l][i_idx] <= h_vars[j],
                    y_vars[l][i_idx] >= h_vars[i] + h_vars[j] - 1,
                ])
                expr -= y_vars[l][i_idx]
        constraints.append(expr <= t)
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.CBC, verbose=False, time_limit=timeout_s)
        return h_vars.value, float(t.value)
    except Exception:
        return None, None

def de_optimize(h_initial, n, dx, timeout_s):
    """
    Differential Evolution for continuous optimization.
    """
    # Scale initial values to [0, 1] and ensure sum correct
    h_scaled = np.array(h_initial)
    h_scaled = project_to_simplex(h_scaled, n/2)
    h_best = h_scaled.copy()
    best_c5 = evaluate_overlap(h_best, dx)

    def objective(h):
        h = project_to_simplex(h, n/2)
        return evaluate_overlap(h, dx)

    # Initial population
    popsize = 30
    population = [h_best.copy()] + [project_to_simplex(np.random.rand(n), n/2) for _ in range(popsize - 1)]
    x_best = h_best.copy()
    best_val = best_c5

    start_time = time.time()
    gen_count = 0
    # Keep track of stagnation
    last_improved = start_time
    F = 0.8  # Mutation factor
    CR = 0.9  # Crossover rate

    while time.time() - start_time < timeout_s - 30:
        gen_count += 1
        # Randomly shuffle
        np.random.shuffle(population)
        new_population = []

        for i in range(popsize):
            # Select three distinct sets
            candidates = [j for j in range(popsize) if j != i]
            a, b, c = np.random.choice(candidates, 3, replace=False)
            # Mutation
            v = population[a] + F * (population[b] - population[c])
            # Crossover
            trial = np.copy(population[i])
            for j in range(n):
                if np.random.rand() < CR:
                    trial[j] = v[j]
            # Projection
            trial = project_to_simplex(trial, n/2)
            # Evaluation
            val = objective(trial)
            if val <= evaluate_overlap(population[i], dx):
                new_population.append(trial)
            else:
                new_population.append(population[i])

        population = new_population
        # Update best
        for h in new_population:
            val = evaluate_overlap(h, dx)
            if val < best_val:
                best_val = val
                x_best = h.copy()
                last_improved = time.time()

        if time.time() - last_improved > 30:
            # Restart
            new_new_best = project_to_simplex(np.random.rand(n), n/2)
            population = [x_best] + [new_new_best] + [project_to_simplex(np.random.rand(n), n/2) for _ in range(popsize - 2)]
            last_improved = time.time()

        if time.time() - start_time > timeout_s - 30:
            break

    return x_best, best_val

def mismatch_swap(h, dx, max_swaps=5000):
    """Refine continuous solution with deterministic mismatches."""
    h = np.array(h)
    current_sum = np.sum(h)
    h = project_to_simplex(h, current_sum)
    # Compute full overlap
    F = fft(h)
    G = fft(1 - h)
    corr = ifft(F * np.conj(G)).real
    best_shift = np.argmax(corr)
    max_val = corr[best_shift]

    count = 0
    for _ in range(max_swaps):
        # Identify mismatched pairs
        M1 = []
        M0 = []
        shift = best_shift - len(h) + 1
        # Shift can be negative or positive
        if shift < 0:
            i_list = list(range(-shift))
            j_list = list(range(len(h) + shift))
        else:
            i_list = list(range(len(h) - shift))
            j_list = list(range(shift, len(h)))
        # identify mismatches
        for i, j in zip(i_list, j_list):
            if h[i] > 0.99 and h[j] < 0.01:
                M1.append(i)
            if h[i] < 0.01 and h[j] > 0.99:
                M0.append(i)

        if not M1 or not M0:
            break
        i_swap = M1[0]
        j_swap = M0[0]

        hi = h[i_swap]
        hj = h[j_swap]
        # Swap
        h[i_swap] = hj
        h[j_swap] = hi

        # Ensure sum constraint
        h = project_to_simplex(h, current_sum)
        # Recompute
        F = fft(h)
        G = fft(1 - h)
        corr = ifft(F * np.conj(G)).real
        best_shift = np.argmax(corr)
        new_max = corr[best_shift]
        max_val = new_max
        count += 1
        if new_max < 0.379:
            break

    return h, max_val
