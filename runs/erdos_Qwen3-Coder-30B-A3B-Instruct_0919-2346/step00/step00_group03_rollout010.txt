```python
import numpy as np
import time
import random

def evaluate_c5(h):
    """Evaluate the C5 bound for the given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h) using full mode
    corr = np.correlate(h, 1 - h, mode='full')
    # Take maximum value (as that's what C5 is defined as)
    c5 = np.max(corr) * dx
    return c5

def random_binary_seq(n):
    """Generate a random binary sequence of length n with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    indices = np.random.choice(n, n//2, replace=False)
    h[indices] = 1
    return h

def solve_binary(n, iters, seed=None):
    """Perform simulated annealing on binary sequence."""
    if seed is not None:
        np.random.seed(seed)
        random.seed(seed)
    
    # Initialize
    h = random_binary_seq(n)
    c5 = evaluate_c5(h)
    best_h, best_c5 = h.copy(), c5
    
    # Annealing parameters
    T0 = 0.1
    T = T0
    cooling = 0.99995
    
    # Perform iterations
    for _ in range(iters):
        # Choose two random indices
        i, j = random.sample(range(n), 2)
        
        # Swap values
        h[i], h[j] = h[j], h[i]
        
        # Compute new C5
        new_c5 = evaluate_c5(h)
        
        # Decide whether to accept
        delta = new_c5 - c5
        if delta <= 0 or random.random() < np.exp(-delta / T):
            # Accept
            c5 = new_c5
            if c5 < best_c5:
                best_c5, best_h = c5, h.copy()
        else:
            # Reject
            h[i], h[j] = h[j], h[i]
            
        # Cool down
        T *= cooling
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function that returns the best h found."""
    # Set random seeds
    np.random.seed(seed)
    random.seed(seed)
    
    # Parameters
    n_points = 200
    dx = 2.0 / n_points
    
    # Track time
    start_time = time.time()
    
    # Best solution found
    best_h = None
    best_c5 = float('inf')
    
    # Run multiple restarts
    restarts = 0
    while time.time() - start_time < budget_s * 0.95:  # Leave margin
        # Run one optimization
        h, c5 = solve_binary(n_points, 10000, seed=seed + restarts)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        restarts += 1
    
    # Return best solution
    return (best_h.tolist(), best_c5, n_points)
```
