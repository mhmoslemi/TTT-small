import numpy as np
import time

def c5_of(h):
    """Compute C5 = max_k ∫ h(x)(1 - h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    # correlate(h, 1-h, mode='full') gives ∫ h(x)(1-h(x+k)) dx for all k
    overlap = np.correlate(h, 1-h, mode="full")
    # Multiply by dx to get actual integral value
    overlap *= dx
    # Return max over all shifts k
    return np.max(overlap)

def propose_move(h, max_step):
    """Propose a new feasible h vector by moving mass between two indices"""
    h_new = h.copy()
    N = len(h)
    
    # Pick two distinct indices
    i, j = np.random.choice(N, 2, replace=False)
    
    # Compute feasible delta interval
    delta_low = max(-h_new[i], h_new[j] - 1.0)
    delta_high = min(1.0 - h_new[i], h_new[j])
    
    # If no feasible move, return unchanged
    if delta_low >= delta_high:
        return h_new
    
    # Sample delta from uniform in feasible range, clipped by max_step
    step = min(max_step, delta_high - delta_low)
    if step > 0:
        delta = np.random.uniform(max(-step, delta_low), min(step, delta_high))
    else:
        delta = 0.0
    
    h_new[i] += delta
    h_new[j] -= delta
    
    return h_new

def anneal(initial_h, time_budget, T_start=0.05, T_end=1e-6, step_start=0.2):
    """Simulated annealing optimization"""
    start = time.time()
    cur = initial_h.copy()
    c_cur = c5_of(cur)
    best = cur.copy()
    c_best = c_cur
    
    N = len(initial_h)
    dx = 2.0 / N
    
    while time.time() - start < time_budget:
        # Time-dependent parameters
        elapsed = time.time() - start
        t_frac = elapsed / time_budget
        T = T_start * (T_end / T_start) ** t_frac
        step = step_start * (1.0 - t_frac)
        
        # Get new candidate
        h_new = propose_move(cur, step)
        c_new = c5_of(h_new)
        
        # Acceptance decision
        if c_new < c_cur or np.random.rand() < np.exp(-(c_new - c_cur) / T):
            cur = h_new
            c_cur = c_new
            
            # Update best if improved
            if c_new < c_best:
                best = cur.copy()
                c_best = c_new
    
    return best, c_best

    """Main optimization routine with multi-start strategy"""
    np.random.seed(seed)
    start = time.time()
    budget = budget_s

    N = 140
    dx = 2.0 / N

    h_uniform = 0.5 * np.ones(N)

    if 'initial_h_values' in globals():

                              np.linspace(0, 2, len(initial_h_values)), 
                              initial_h_values)
        # Renormalize to sum = N/2
        h_initial = h_initial * (N / 2) / np.sum(h_initial)
    else:
        h_initial = np.ones(N) * 0.5
        
    # 3. Double-peak seed
    h_double = np.ones(N) * 0.1
    a = 0.3  # half-width of peak regions
    peak_val = 0.9
    low_val = 0.1
    # Set peak regions [0,a) and [2-a,2]
    indices1 = np.arange(0, int(np.round(a*N/2)))
    indices2 = np.arange(int(np.round((2-a)*N/2)), N)
    if len(indices1) > 0:
        h_double[indices1] = peak_val
    if len(indices2) > 0:
        h_double[indices2] = peak_val
    # Renormalize
    h_double = h_double * (N / 2) / np.sum(h_double)
    
    # 4. Random perturbation with sum constraint
    h_random = np.random.uniform(0, 1, N)
    # Project to satisfy sum = N/2
    h_random = h_random * (N / 2) / np.sum(h_random)
    
    # List of seeds to run
    seeds = [h_uniform, h_initial, h_double, h_random]
    
    # Time allocation
    budget_per_seed = 0.30 * budget
    remaining_budget = budget - 4 * budget_per_seed
    
    # Multi-start optimization
    best_h = None
    best_c5 = float('inf')
    
    for i, seed_h in enumerate(seeds):
        if time.time() - start >= budget:
            break

        h_opt, c5_opt = anneal(seed_h, budget_per_seed)

            best_h = h_opt
            best_c5 = c5_opt
            
        # Check if we can stop early

            break
    
    # Final polishing phase to reduce noise
    if best_h is not None and time.time() - start < budget:

            cur = h.copy()
