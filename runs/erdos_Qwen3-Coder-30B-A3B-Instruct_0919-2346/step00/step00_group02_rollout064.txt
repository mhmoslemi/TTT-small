import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all k
    correlation = np.correlate(h, 1 - h, mode='full')
    # The correlation size is 2*n-1. We want the middle part that corresponds 
    # to valid overlaps. The valid range is from (n-1) to (n-1) + (n-1) = 2*n-2.
    # But actually, for k = 0 to 2, we consider the overlaps for k from 0 to 2.
    # We are interested in max of (n-1) to (n-1) + (n-1) entries, but non-overlapping
    # part gives valid entries from k=0 to 2.
    # The actual max is over k = 0 to 2, so valid is from index (n-1) to (n-1 + (n-1))  
    # For k in [0, 2], the corresponding correlation index range is [n-1, n+1]
    # But since we want error for k in [0, 2], we need correlation indices in [n-1, n+1]
    # This gives us 3 values, that are max(integral for k=0 to k=2) = max(C(k))
    # Because indexing in correlate gives us offset that goes from - (n-1) to n-1
    # so we want elements from (n-1) to (n-1) + (n-1)
    # Actually for overlap k, it corresponds to correlation_index = n-1+k
    # k ranges from 0 to 2, so correlation indices is from (n-1) to (n+1) = 3 values
    correlation_values = correlation[n-1:n+2] 
    # multiply by dx to get the integral value
    c5 = np.max(correlation_values) * dx
    return c5

def objective(h):
    """Return negative of C5 because we want to minimize C5."""
    return -evaluate_c5(h)

    # For simplicity, let's try a piecewise constant function with a few segments

    # For constraint: sum(h) * dx = 1 => sum(h) = n_points / 2
    # To maintain that during optimization, we will use a penalty for constraint violation
    # But here we'll just do L1 box constraints and hope it works.
    # We define bounds for each h[i] between 0 and 1
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Better: try an approach with projected gradient descents or evolution strategy
    # Since we want to minimize max of integrals,  
    # Use scipy differential evolution which is quite robust
    
    # Let's time how long we want to run
    def wrapper(h):
        # Adjust h so that sum(h) == n_points/2
        h_adjust = np.array(h)
        actual_sum = np.sum(h_adjust)

            desired_sum = n_points / 2
