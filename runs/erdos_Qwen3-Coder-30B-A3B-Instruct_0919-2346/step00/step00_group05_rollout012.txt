import numpy as np

def evaluate(h):
    """Compute C5 = max_k integral h(x)(1-h(x+k)) dx"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlations for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # Only consider valid shifts
    valid_corr = corr[n-1:2*n-1]
    return np.max(valid_corr) * dx

def project_onto_capped_simplex(v, S):
    """Project v onto {h : 0 <= h_i <= 1, sum(h) = S}"""
    v = np.array(v)
    # Bisection to find lambda such that sum(clip(v-lambda, 0, 1)) = S
    low, high = -2.0, 2.0
    tol = 1e-10
    while high - low > tol:
        mid = (low + high) / 2.0
        h = np.clip(v - mid, 0, 1)
        s = np.sum(h)
        if s > S:
            low = mid
        else:
            high = mid
    h = np.clip(v - (low + high)/2.0, 0, 1)
    return h

def coeffs_and_const(h):
    """Compute a_{i,k} and c_k for all shifts k"""
    n = len(h)
    # Precompute shifts and zero padding
    # curl : current left and right indices for shift k
    a = np.zeros((n, 2*n-1))
    c = np.zeros(2*n-1)
    
    for k in range(-(n-1), n):
        # Use vectorized operations to compute a[i,k] = 1 - h[i+k] - h[i-k]
        k_idx = k + n - 1  # map k to [0, 2n-2]
        
        if k >= 0:
            # h[i+k] is valid for i in [0, n-k-1]
            valid_i = np.arange(n - k)
            a[valid_i, k_idx] = 1 - h[valid_i+k] - h[valid_i-k] if k > 0 else 1 - h[valid_i] - h[valid_i]
        else:
            # h[i+k] is valid for i in [-k, n-1]
            valid_i = np.arange(-k, n)
            a[valid_i, k_idx] = 1 - h[valid_i+k] - h[valid_i-k] if k < 0 else 1 - h[valid_i] - h[valid_i]

        # c_k = sum h_i * h_{i+k}
        if k >= 0:
            valid_i = np.arange(n - k)
            c[k_idx] = np.sum(h[valid_i] * h[valid_i + k])
        else:
            valid_i = np.arange(-k, n)
            c[k_idx] = np.sum(h[valid_i] * h[valid_i + k])
            
    return a, c

def linear_program(a, const, n, S):
    """Solve the LP for given a and const vectors"""
    # This is the LP:
    # minimize t
    # s.t. a[i,k] * h[i] + const[k] <= t for each k (2*n - 1 constraints)
    #      sum(h) = S
    #      h[i] in [0,1]
    
    # Prepare constraint matrix
    n_constraints = 2 * n - 1
    # Coefficients for h and t (in one augmented vector of size n+1)
    # Each constraint a[i,k] h[i] + const[k] <= t becomes (a[i,k] - 1*t) <= const[k]
    # therefore coefficient matrix:
    C = np.zeros((n_constraints + 1, n + 1))
    for k in range(n_constraints):
        C[k, :-1] = a[:, k]  # For h
        C[k, -1] = -1        # For t (a[i,k] h[i] - 1 * t <= const[k])
    
    b = -const.values() if hasattr(const, 'values') else -const
    
    # Constant term for mass constraint (+1 for each h[i], 0 for t)
    C[n_constraints, :-1] = 1
    C[n_constraints, -1] = 0  # Not relevant to mass (thus 0)
    
    # Bounds on variables: h in [0,1], t free
    bounds_h = [(0, 1) for _ in range(n)]
    bounds_t = (None, None)  # t is unbounded
    bounds = bounds_h + [bounds_t]
    
    # Minimize t (corresponds to the last variable)
    c = np.zeros(n + 1)
    c[-1] = 1  # This says: minimize the last variable (t)
    
    try:
        res = linprog(c, A_ub=C[:n_constraints], b_ub=b, bounds=bounds, method='highs')
        if res.success:
            h_solve = res.x[:-1]    # Variables except last (t)
            t_solve = res.x[-1]
            return h_solve, t_solve
        else:
            # Reject infeasible solutions; fallback to no change (could be improved)  
            return None, None
    except Exception as e:
        return None, None

            ''' Build matrix for linear programming:
             c: [0, ... ,1] (minimize t)

        h_prev = h_best.copy()
