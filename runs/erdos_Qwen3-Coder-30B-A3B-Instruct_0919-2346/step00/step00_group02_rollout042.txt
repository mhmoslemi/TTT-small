import numpy as np
import time

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # np.correlate with mode='full' gives correlation with all shifts
    # Here we want int h(x)(1-h(x+k)) dx = int h(x) dx - int h(x)h(x+k) dx
    # So C5 = 1 - min_k int h(x)h(x+k) dx
    # So we compute the auto-correlation int h(x)h(x+k) dx
    # and return 1 - min_correlation
    corr = np.correlate(h, h, mode='full')
    # correlation at shift k corresponds to corr[N-1+k] (shift 0 is at index N-1)
    # We want min of the auto-correlations over all shifts
    # but we also want the integral to reflect dx
    # The actual integral over all shifts covering the full domain
    # is the full correlation vector scaled by dx, with proper indexing
    # We want int h(x)h(x+k) dx over all k in [-N+1, N-1]
    # The correlations are indexed from -N+1 to N-1, where k=0 is at index N-1
    # Thus indices -N+1 to -1 are corr[0:N-1], and 1 to N-1 are corr[N:N-1]
    # But actually theory says we want only non-negative shift (0 to N-1) due to symmetry
    # But we should evenly sample over negative and positive, so really we want full range
    # However the problem is written so we take the max of all k including negatives,
    # and it's symmetric (the integral of h(x)h(x+k) is same as h(x+k)h(x))
    # So just take the mins over all correlation values, not making any assumptions about symmetry
    # The final value we return is: C5 = 1 - min(corr_values)
    # Our corr is full of length 2*N - 1
    min_corr = np.min(corr)
    return 1.0 - min_corr * dx

def propose_move(h, max_step):
    """Propose move by adjusting two elements while preserving sum."""
    h_new = h.copy()
    N = len(h)
    while True:
        # randomly pick two different indices
        i, j = np.random.choice(N, 2, replace=False)
        # compute delta limits
        delta_low = max(-h_new[i], h_new[j] - 1.0)
        delta_high = min(1.0 - h_new[i], h_new[j])
        if delta_low >= delta_high:
            # move not possible, continue
            continue
        # sample delta from [delta_low, delta_high]
        delta = np.random.uniform(delta_low, delta_high)
        # clip delta to max step (absolute value)
        delta = np.clip(delta, -max_step, max_step)
        # apply updates
        h_new[i] += delta
        h_new[j] -= delta
        break
    return h_new

def anneal(initial_h, time_budget):
    """Run simulated annealing optimization on the initial_h."""
    # Setup parameters
    T_start = 0.05
    T_end = 1e-6
    step_start = 0.2
    start = time.time()
    cur = initial_h.copy()
    c_cur = c5_of(cur)
    best = cur.copy()
    c_best = c_cur

    while time.time() - start < time_budget:
        elapsed = time.time() - start
        t = elapsed / time_budget
        # decay temperature
        T = T_start * (T_end / T_start) ** t
        # decay step size
        step = step_start * (1.0 - t)
        if step < 1e-6:
            step = 1e-6

        # propose new sample
        h_new = propose_move(cur, step)
        c_new = c5_of(h_new)
        
        # acceptance rule
        if c_new <= c_cur:
            cur = h_new
            c_cur = c_new
            # update global best
            if c_new < c_best:
                best = h_new.copy()
                c_best = c_new
        else:
            # Metropolis acceptance
            if np.random.rand() < np.exp(-(c_new - c_cur) / T):
                cur = h_new
                c_cur = c_new

    return best, c_best

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    N = 140
    dx = 2.0 / N
    time_budget = budget_s * 0.90  # leave 10% for final polishing
    
    # Create three different seeds
    # 1. uniform initial
    h_uniform = np.ones(N) * 0.5
    
    # 2. use initial_h_values if available, else fallback uniform
    if 'initial_h_values' in globals():
        h_initial = np.interp(np.linspace(0, 2, N), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
        # renormalize sum to be N/2
        h_initial = h_initial * (N / 2) / np.sum(h_initial)
    else:
        h_initial = np.ones(N) * 0.5
    
    # 3. two-peak shaped seed, concentrated at ends
    # Create piecewise constant pattern: h(x) = u if x in [0,a] or [2-a,2], else v
    a = 0.3
    u = 0.9
    v = 0.1
    x = np.linspace(0, 2, N)
    h_two_peak = np.where(np.logical_or(x <= a, x >= 2 - a), u, v)
    # correct sum to be N/2
    h_two_peak = h_two_peak * (N / 2) / np.sum(h_two_peak)

    # run three independent annealing runs
    best_h = None
    best_c5 = float('inf')
    
    seeds = [h_uniform, h_initial, h_two_peak]
    budget_per_seed = time_budget / 3.0
    
    for i, h_seed in enumerate(seeds):
        # Run to get best from current seed
        h_final, c5_final = anneal(h_seed, budget_per_seed)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()
    
    # Polishing on the best solution found so far
    if best_h is not None:
        # Convert to deterministic hill-climb with zero temperature
        np.random.seed(seed)
        T_final = 1e-10  # effectively zero
        step_final = 0.02
        start = time.time()
        cur = best_h.copy()
        c_cur = best_c5
        while time.time() - start < (budget_s - (time_budget + 10)):  # leave 10s for final polish
            h_new = propose_move(cur, step_final)
            c_new = c5_of(h_new)
            if c_new <= c_cur:
                cur = h_new
                c_cur = c_new
        best_h = cur
        best_c5 = c_cur
    
    return (best_h, best_c5, N)
