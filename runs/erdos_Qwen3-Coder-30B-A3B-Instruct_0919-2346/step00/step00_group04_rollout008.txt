def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C_5 for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum correlation (adjusted for normalization) is our C_5
    max_correlation = np.max(correlation)
    # Normalize by dx
    c5 = max_correlation * dx
    return c5
