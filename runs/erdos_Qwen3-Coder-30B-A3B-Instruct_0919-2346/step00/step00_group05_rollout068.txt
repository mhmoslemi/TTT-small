def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h."""
    n = len(h)

    # Compute the correlation integral
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum value is what we want
    max_val = np.max(corr) * dx
    return max_val
