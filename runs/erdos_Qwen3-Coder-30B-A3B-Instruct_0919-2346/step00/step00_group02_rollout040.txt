import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h_values):
    """Compute C5 = max_k ∫ h(x)(1 - h(x+k)) dx"""
    n_points = len(h_values)
    dx = 2.0 / n_points
    corr = correlate(h_values, 1 - h_values, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

        # Annealing
        if temp < min_temp:

        hi, hj = h_values[i], h_values[j]

        elif np.random.rand() < np.exp(-(c5_new - best_c5) / temp):
            best_c5 = c5_new
            best_h = new_h.copy()
            
        # Update temperature
        temp *= cooling_rate
