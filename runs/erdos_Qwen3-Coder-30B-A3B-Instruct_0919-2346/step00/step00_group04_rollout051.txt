def evaluate_h(h):
    """Evaluate the C5 constant for given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation with 1-h
    corr = np.correlate(h, 1 - h, mode='full')
    # The correlation values correspond to shifts from -(n-1) to (n-1)
    # The maximum is at index n-1 for shift 0, we want to exclude that
    # So we exclude the central value and take max of others
    max_overlap = np.max(corr[:n-1])  # pts 0 to n-2
    max_overlap = np.maximum(max_overlap, np.max(corr[n+1:]))
    c5 = max_overlap * dx
    return c5
