import numpy as np
import random

import time

def evaluate_h(h_values):
    n = len(h_values)
    dx = 2.0 / n
    # Compute correlation of h with (1-h) using full mode
    corr = np.correlate(h_values, 1 - np.array(h_values), mode='full')
    # The full correlation gives values for shifts -n+1 to n-1.
    # The maximum of interest corresponds to the relevant shift range.
    # The maximum value in corr array corresponds to the maximum overlap integral.
    max_corr = np.max(corr) * dx
    return max_corr

def project_to_feasible(h, n):
    """Project h to satisfy sum(h) = n/2 and bounds [0,1]"""
    # Clip values
    h = np.clip(h, 0, 1)
    # Adjust to sum to n/2
    current_sum = np.sum(h)
    residual = (n / 2) - current_sum
    if abs(residual) < 1e-12:
        return h
    
    # Find indices that can be adjusted (not at bounds)
    candidates = np.where((h > 0) & (h < 1))[0]
    if len(candidates) == 0:
        # If all values are at bounds, return clipped version
        return h
    
    # Compute adjustment per candidate
    adjustment = residual / len(candidates)
    for i in candidates:
        new_val = h[i] + adjustment
        h[i] = np.clip(new_val, 0, 1)
    
    return h
