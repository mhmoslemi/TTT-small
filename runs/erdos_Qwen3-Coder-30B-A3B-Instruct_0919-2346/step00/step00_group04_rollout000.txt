    n_points = len(h_values)

    c5_bound = max_overlap * dx
    return c5_bound

def perturb_h(h_values, n_points):
    """Apply small perturbation to h_values to create new candidate."""
    new_h = h_values.copy()
    # Swap two random positions: one 1 and one 0
    ones = np.where(new_h == 1)[0]
    zeros = np.where(new_h == 0)[0]
    
    if len(ones) > 0 and len(zeros) > 0:
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        new_h[i] = 0
        new_h[j] = 1
    return new_h

def optimize_h(start_h, n_points, budget_s):
    """Optimize the step function h to minimize C5."""
    best_h = start_h.copy()
    best_c5 = evaluate_h(best_h)
    
    start_time = time.time()
    
    # Annealing schedule
    T = 1.0
    cooling_rate = 0.99995
    min_T = 1e-6
    
    iterations = 0
    while time.time() - start_time < budget_s * 0.95 and T > min_T:
        # Create new candidate h
        new_h = perturb_h(best_h, n_points)
        new_c5 = evaluate_h(new_h)
        
        # Accept or reject
        if new_c5 < best_c5:
            best_h[:] = new_h
            best_c5 = new_c5
        else:
            # Accept with probability
            delta = new_c5 - best_c5
            if np.random.rand() < np.exp(-delta / T):
                best_h[:] = new_h
                best_c5 = new_c5
                
        # Update temperature
        T *= cooling_rate
        iterations += 1
        
        # Occasionally make large moves
        if iterations % 1000 == 0:
            # Occasionally do larger perturbations
            if np.random.rand() < 0.1:
                new_h = perturb_h(best_h, n_points)
                new_h = perturb_h(new_h, n_points)  # Double swap
                new_c5 = evaluate_h(new_h)
                if new_c5 < best_c5:
                    best_h[:] = new_h
                    best_c5 = new_c5
    return best_h, best_c5
