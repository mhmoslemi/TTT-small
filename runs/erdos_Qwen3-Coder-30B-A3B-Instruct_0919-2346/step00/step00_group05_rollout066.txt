def evaluate_erdos_solution(h_values):
    """Evaluate the C5 constant for the given h_values."""
    n = len(h_values)
    dx = 2.0 / n
    c = np.correlate(h_values, 1 - np.array(h_values), mode='full')
    return np.max(c) * dx

def project_to_capped_simplex(h_vals, target_sum):
    """Project h_vals to the capped simplex with exact sum."""
    n = len(h_vals)
    h_vals = np.array(h_vals)
    # Clip to [0,1]
    h_vals = np.clip(h_vals, 0, 1)
    # Adjust to match target sum
    current_sum = np.sum(h_vals)
    if abs(current_sum - target_sum) < 1e-10:
        return h_vals
    elif current_sum > target_sum:
        # Reduce values
        # Sort descending
        idx = np.argsort(h_vals)[::-1]
        diff = current_sum - target_sum
        h_proj = h_vals.copy()
        for i in idx:
            if diff <= 0:
                break
            reduce = min(h_proj[i], diff)
            h_proj[i] -= reduce
            diff -= reduce
    else:
        # Increase values
        idx = np.argsort(h_vals)
        diff = target_sum - current_sum
        h_proj = h_vals.copy()
        for i in idx:
            if diff <= 0:
                break
            increase = min(1 - h_proj[i], diff)
            h_proj[i] += increase
            diff -= increase
    return h_proj

def solve_milp_exact(h0, n, target_sum, max_time_s):
    """Solve the MILP formulation with exact McCormick envelopes."""
    # Create variables
    h = cp.Variable(n, name='h')
    tau = cp.Variable(name='tau')
    
    # y variables for each k (shift), y_k[j] = h[j]*h[j+k]
    # y_k is a list of variables for each k
    y_vars = {}
    constraints = []

    # Add constraints for y variables (McCormick envelopes)
    for k in range(1, n):
        # y_k[j] = h[j] * h[j+k]
        # but only for j = 0,...,n-k-1 (k is shift, so j+k <= n-1 => j <= n-k-1)
        y_k = cp.Variable(n - k, name=f'y_{k}')
        y_vars[k] = y_k
        
        for j in range(n - k):
            # McCormick envelopes:
            # y_k[j] >= h[j] + h[j+k] - 1
            # y_k[j] <= h[j]
            # y_k[j] <= h[j+k]
            # y_k[j] >= 0
            
            constraints += [
                y_k[j] >= h[j] + h[j + k] - 1,
                y_k[j] <= h[j],
                y_k[j] <= h[j + k],
                y_k[j] >= 0
            ]

    # τ ≤ Σ_j y_k[j] for all k
    for k in range(1, n):
        constraints += [tau <= cp.sum(y_vars[k])]

    # Feasibility constraints
    constraints += [cp.sum(h) == target_sum]
    constraints += [h >= 0]
    constraints += [h <= 1]
    # Fix first element to break symmetry
    constraints += [h[0] == 1]
    
    # Objective
    objective = cp.Maximize(tau)

    # Build problem
    prob = cp.Problem(objective, constraints)
    
    # Solve
    try:
        prob.solve(solver=cp.CBC, time_limit=max_time_s, verbose=False)
    except:
        # fallback to other solver or fail gracefully
        try:
            prob.solve(solver=cp.GLPK, time_limit=max_time_s, verbose=False)
        except:
            return None, float('-inf')
    
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        h_opt = h.value
        tau_opt = tau.value
        return h_opt, tau_opt
    return None, float('-inf')

def refine_binary(h_vals, max_iter=1000):
    """Try improving binary solution by swapping 1->0 with 0->1 along bad shift."""
    n = len(h_vals)
    if not all(x == 0 or x == 1 for x in h_vals):
        return h_vals
    h = h_vals.copy()
    # Find shift that maximizes correlation
    c = np.correlate(h, 1 - h, mode='full')
    worst_k = np.argmax(c)
    if worst_k >= n:
        # We're halfway into full domain:
        worst_k = worst_k - n
    # Find all pairs of (i, i+worst_k) where h[i]=1, h[i+worst_k]=0
    pairs = [(i, i+worst_k) for i in range(n) if (i + worst_k < n) and h[i] == 1 and h[i + worst_k] == 0]
    # Attempt as many valid swaps as possible
    iter = 0
    while iter < max_iter and pairs:
        # Pick one pair at random
        i, j = random.choice(pairs)
        # Swap:
        h[i], h[j] = h[j], h[i]
        c_new = np.correlate(h, 1 - h, mode='full')
        if np.max(c_new) >= np.max(c):
            # Revert
            h[i], h[j] = h[j], h[i]
        else:
            # Accept, update c and repopulate pairs
            c = c_new
            pairs = [(i, i+worst_k) for i in range(n) if (i + worst_k < n) and h[i] == 1 and h[i + worst_k] == 0]
        iter += 1
    return h
