import numpy as np

def project_to_capped_simplex(h, target_sum):
    """Project h to the capped simplex with sum = target_sum"""
    n = len(h)
    # Clamp values to [0,1]
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Waterfilling
    sorted_h = np.sort(h)[::-1]
    cumulative_sum = np.cumsum(sorted_h)
    # Find where to cut
    idx = np.argmax(sorted_h * np.arange(1, n + 1) - cumulative_sum + target_sum)
    if idx == 0 and sorted_h[0] * n - cumulative_sum[0] + target_sum <= 0:
        # All elements will be zero, but can't satisfy the mean constraint
        # Just return clamped version
        return h
    if idx == 0:
        lambda_value = (target_sum - sorted_h[0]) / (n - 1)
        # Adjusted waterfilling, but just in the first case is tricky
        lambda_value = (target_sum - cumulative_sum[0]) / n
    else:
        lambda_value = (target_sum - cumulative_sum[idx-1]) / (n - idx)
    # Apply lambda
    result = np.maximum(0, np.minimum(1, h - lambda_value))
    return result

def compute_correlation(h):
    """Compute correlation as in the function"""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    return corr * dx

def compute_f_alpha(h, alpha):
    """Compute the log-sum-exp surrogate at temperature alpha"""
    dx = 2.0 / len(h)
    O = np.zeros(2 * len(h) - 1)
    # Compute O_ell(h) for all lags
    for l in range(-len(h) + 1, len(h)):
        if l >= 0:
            O[l + len(h) - 1] = dx * np.dot(h[:len(h)-l], 1 - h[l:])
        else:
            O[l + len(h) - 1] = dx * np.dot(h[-l:], 1 - h[:len(h)+l])
    # Log-sum-exp
    exp_O = np.exp(alpha * O)
    return (dx / alpha) * np.log(np.sum(exp_O))

def compute_gradient(h, alpha):
    """Compute gradient w.r.t. h using convolution"""
    dx = 2.0 / len(h)
    # For full correlation
    O = np.zeros(2 * len(h) - 1)
    for l in range(-len(h) + 1, len(h)):
        if l >= 0:
            O[l + len(h) - 1] = dx * np.dot(h[:len(h)-l], 1 - h[l:])
        else:
            O[l + len(h) - 1] = dx * np.dot(h[-l:], 1 - h[:len(h)+l])
    # Softmax weights
    exp_O = np.exp(alpha * O)
    weights = exp_O / np.sum(exp_O)
    # Compute convolution-based gradient: implied via full form
    # grad[i] = sum_l weights[l] * ( (1 - h[i + l]) - h[i - l] )
    # That's equivalent to:
    # 1. c1 = conv(weights, 1-h) -> \sum_l weights[l] * (1-h[i+l])
    # 2. c2 = conv(weights_rev, h) -> \sum_l weights[l] * h[i-l]
    # Note we must be precise with handling indices for proper convolution
    # with mode='full'
    weights_rev = weights[::-1]
    c1 = signal.convolve(weights, 1 - h, mode="full")
    c2 = signal.convolve(weights_rev, h, mode="full")
    # Select only the middle part that corresponds to indices 0 to n-1
    grad = c1[len(h) - 1:] - c2[len(h) - 1:]
    return grad
