import cvxpy as cp

        h_padded = np.concatenate([h_vec, h_vec[:k]])
        return np.dot(h_vec, h_padded[k:k+n])

    def compute_full_autocorrelations(h_vec):
        # Fast computation using FFT
        H = fft.fft(h_vec)
        autocorr = fft.ifft(H * np.conj(H)).real
        return autocorr[:len(h_vec)]  # Only the first n values correspond to the true autocorrelations

    def exact_c5(h_vec):
        ac = compute_full_autocorrelations(h_vec)
        return 1.0 - 2.0/n * np.min(ac)

    else:
        h_init = h_init[:n]
    
    # Normalize to ensure sum = 1 with dx = 2.0/n
    total = sum(h_init) * dx
    if abs(total) < 1e-10:
        h_init = np.ones(n) * (n/2) / n
    else:
        h_init *= (n/2) / (total)

    # MILP approach
    def solve_milp_with_shifts(shifts, h0_fixed=True):
        # Binary variable h
        h_bin = cp.Variable(n, integer=True)
        # Auxiliary variables w_{i,k} for each shift
        w_vars = {}
        a_vars = {}
        # Constraint for sum = n/2
        prob = cp.Problem(cp.Minimize(0), 
                          [cp.sum(h_bin) == n//2,
                           h_bin >= 0, h_bin <= 1])
        # Add auxiliary variables
        if shifts:
            for k in shifts:
                w_vars[k] = cp.Variable(n, integer=True)
                a_vars[k] = cp.sum(w_vars[k])
                prob.constraints += [
                    w_vars[k] <= h_bin,
                    w_vars[k] <= cp.roll(h_bin, k),
                    w_vars[k] >= h_bin + cp.roll(h_bin, k) - 1,
                    w_vars[k] >= 0,
                    w_vars[k] <= 1
                ]
                prob.constraints += [cp.sum(w_vars[k]) == a_vars[k]]
            # Add constraint that t <= a_k for all k in shifts
            t = cp.Variable(integer=True) # The minimum A_k
            for k in shifts:
                prob.constraints += [t <= a_vars[k]]
            # Maximize t, so minimize -t
            prob = cp.Problem(cp.Minimize(-t), prob.constraints)
            # Fix h[0] to 1 to break symmetry
            if h0_fixed:
                prob.constraints += [h_bin[0] == 1]
            # Solve the problem
            try:
                prob.solve(solver=cp.CBC, maxSeconds=min(budget_s/2, 30))
                if prob.status == 'optimal' or prob.status == 'optimal_inaccurate':
                    sol_h = np.round(h_bin.value).astype(int)
                    return sol_h, t.value
            except Exception:
                pass
        return None, None

    timeout = time.time() + budget_s - 60  # leave 60s for polish
    for iteration in range(100):  # safety
        if time.time() > timeout - 5:  # Leave time for polish

                for i in range(min(-diff, len(one_indices))):

        if diff > 0:

            for i in range(min(diff, len(zero_indices))):
                best_h[zero_indices[i]] = 1
        elif diff < 0:
            one_indices = np.where(best_h == 1)[0]
            for i in range(min(-diff, len(one_indices))):
                best_h[one_indices[i]] = 0
    # Final evaluation
    target_h = np.array(best_h, dtype=float)
    c5_val = exact_c5(target_h)
    proposed_h = target_h.copy()
    # Optional polishing
    if time.time() < timeout - 10:

            for step in range(200):
                if time.time() > timeout - 10:
                    break
                ac_full = compute_full_autocorrelations(target_h)
                k_min = np.argmin(ac_full)
                grad = np.zeros_like(target_h)
                for i in range(n):

                delta = 0.001 * grad
                target_h += delta

                # Ensure sum = n/2
                total = sum(target_h) * dx

                if abs(residual) > 1e-5:
                    # Apply pairwise swap to maintain sum
                    indices_neg = np.where(target_h < 1)[0]
                    indices_pos = np.where(target_h > 0)[0]
                    if len(indices_neg) > 0 and len(indices_pos) > 0:
                        idx_neg = indices_neg[0]
                        idx_pos = indices_pos[0]
                        amount = min(-residual, target_h[idx_pos])
                        target_h[idx_pos] -= amount
                        target_h[idx_neg] += amount
                new_c5 = exact_c5(target_h)
                if new_c5 < c5_val:
                    c5_val = new_c5
                    proposed_h = target_h.copy()
        except Exception:

    # Final check: generate a continuous version that meets requirements
    if proposed_h.shape[0] != n:
        # Fill or trim
        if proposed_h.shape[0] < n:
            proposed_h = np.pad(proposed_h, (0, n - len(proposed_h)), 'constant', constant_values=0)
        else:
            proposed_h = proposed_h[:n]
    # Final total check
    total = sum(proposed_h) * dx
    if abs(total - 1.0) > 1e-3:
        # Adjust for sum constraint
        scale_factor = 1.0 / total
        proposed_h *= scale_factor
    # Make sure in [0,1]
    proposed_h = np.clip(proposed_h, 0, 1)
    # Final evaluate
    final_c5 = exact_c5(proposed_h)
    return (proposed_h, final_c5, n)
