import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import math

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for given h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute convolution of h with (1-h)
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum value of the convolution (scaled)
    max_val = np.max(conv) * dx
    return max_val

def initial_h_values(n_points):
    """Initial construction for h"""
    # Using a piecewise constant construction similar to known good solutions
    h = np.zeros(n_points)
    dx = 2.0 / n_points
    
    # Start with a simple pattern that's known to work well
    # For simplicity, let's start with a piecewise constant step function
    # Sample with 5 intervals to keep it small and simple
    # This is just a basic construction to start from
    h[:n_points//3] = 0.5
    h[n_points//3:2*n_points//3] = 0.3
    h[2*n_points//3:] = 0.7
    
    # Adjust to satisfy sum(h)*dx = 1
    current_sum = np.sum(h) * dx
    h = h * (1.0 / current_sum)
    return h

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction
    n_points = 45
    h = initial_h_values(n_points)
    dx = 2.0 / n_points
    
    # Define optimization variables
    h_vars = Variable(n_points)
    
    # We'll use scipy's minimize to handle constraint optimization
    # We'll reformulate this as a constrained optimization with L-BFGS-B method
    # Objective: minimize max correlation value
    # Constraints: 0 <= h[i] <= 1, sum(h)*dx = 1
    
    def objective(h_vals):
        # The problem is to minimize the maximum of the correlation
        corr = np.correlate(h_vals, 1 - h_vals, mode="full")
        return np.max(corr) * dx
    
    def constraint_eq(h_vals):
        return np.sum(h_vals) * dx - 1.0
    
    def constraint_bounds(h_vals):
        return np.concatenate([h_vals, 1 - h_vals])
    
    # Optimization using scipy minimize
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) * dx - 1.0}]
    
    # Use a simple gradient-free optimization on top of initial solution
    h_init = h.copy()
    best_h = h_init.copy()
    best_c5 = evaluate_erdos_solution(h_init)
    
    # Start with fine-grained local searches
    for i in range(500):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Do a small local perturbation
        h_perturbed = best_h.copy()
        idx = np.random.randint(0, n_points)
        delta = np.random.normal(0, 0.05)
        h_perturbed[idx] += delta
        h_perturbed[idx] = np.clip(h_perturbed[idx], 0, 1)
        
        # Renormalize
        h_perturbed = h_perturbed * (1.0 / (np.sum(h_perturbed) * dx))
        
        c5_val = evaluate_erdos_solution(h_perturbed)
        if c5_val < best_c5:
            best_h = h_perturbed.copy()
            best_c5 = c5_val
    
    # Also try some structural changes like adaptively building a step function
    best_h_final = best_h.copy()
    best_c5_final = best_c5

    # Try to use a more systematic approach with gradient-based optimizer
    # use scipy.optimize.minimize with bounds and constraints
    res = minimize(
        objective,
        best_h_final,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 1000, 'gtol': 1e-6}
    )
    
    if res.success:
        h_opt = res.x
        h_opt = h_opt * (1.0 / (np.sum(h_opt) * dx))
        c5_opt = evaluate_erdos_solution(h_opt)
        if c5_opt < best_c5_final:
            best_h_final = h_opt
            best_c5_final = c5_opt

    return (best_h_final, best_c5_final, n_points)
