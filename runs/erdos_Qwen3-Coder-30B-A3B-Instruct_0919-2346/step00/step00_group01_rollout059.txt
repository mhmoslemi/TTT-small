import numpy as np
import cvxpy as cp
import time
from scipy import fft
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Global timer
    start_time = time.time()
    
    n = 60
    dx = 2.0 / n
    target_bound = 0.3807
    
    # Initial shift set - these are the most dangerous shifts to start with
    K0 = [1, n//2]
    if n > 4:
        K0.extend([2, n-2])
    K0 = sorted(set(K0))  # Remove duplicates and sort
    
    # Helper function to compute exact autocorrelations via FFT
    def compute_autocorrelations(h):
        h_pad = np.pad(h, (0, n), mode='constant')
        A = np.fft.ifft(np.fft.fft(h_pad) * np.conj(np.fft.fft(h_pad)))[:n].real
        return A[:n]  # Return only positive lags
    
    def compute_c5_bound(h):
        A = compute_autocorrelations(h)
        min_A = np.min(A)
        return 1 - (2/n) * min_A
    
    # Binary MILP solver
    def solve_milp_with_shifts(hs, K, t, w_vars, h_vars):
        # Initialize constraints for given shift set
        constraints = []
        underlying_vars = []  # Will be set in the first solve
        
        for k in K:
            for i in range(n):
                # w[i,k] = h[i] AND h[i+k]
                w_idx = (i, k)  # This creates unique indices in the dictionary
                w_var = w_vars[w_idx]
                h_i = h_vars[i]
                h_k = h_vars[(i+k) % n]
                
                # Linearization of w = h_i AND h_{i+k}
                constraints.append(w_var <= h_i)
                constraints.append(w_var <= h_k)
                constraints.append(w_var >= h_i + h_k - 1)
                constraints.append(w_var >= 0)
                constraints.append(w_var <= 1)
        
        # Constraint t <= A_k for all k in K
        for k in K:
            A_k = sum(w_vars[(i, k)] for i in range(n))
            constraints.append(t <= A_k)
        
        # Sum constraint
        sum_h = sum(h_vars[i] for i in range(n))
        constraints.append(sum_h == n/2)
        
        # Fix h_0 = 1 to break rotational symmetry
        constraints.append(h_vars[0] == 1)
        
        # Optimization problem
        objective = cp.Maximize(t)
        prob = cp.Problem(objective, constraints)
        
        # Solve with time limit
        try:
            prob.solve(solver=cp.CBC, max_seconds=min(budget_s, 30))
            return prob.status == cp.OPTIMAL or prob.status == cp.UNBOUNDED
        except Exception:
            return False
    
    # Try to solve binary MILP first
    best_h = None
    best_c5 = float('inf')
    
    # First run with the basic shift set
    big_K = set(K0)
    # Track best solution
    best_solution = None
    best_t = 0
    best_c5_current = float('inf')
    
    # Cache for binary variables
    w_vars = {}
    h_vars = {}
    
    # Set up initial model and constraints
    try:
        # Initialize variables
        for i in range(n):
            h_vars[i] = cp.Variable(boolean=True)
        t = cp.Variable()
        
        # Initialize w variables - just create placeholders
        # will update on each iteration
        for i in range(n):
            for k in big_K:
                w_vars[(i, k)] = cp.Variable(boolean=True)
        
        # Fix h_0 = 1 to break symmetry
        h_vars[0].value = 1
        h_vars[0].fix_value(1)
        
        # Initial solve with first K0
        probs = []
        constraints = []
        
        # Setup constraints and solve
        # We only do one solve here unless things don't converge
        for k in big_K:
            for i in range(n):
                w_idx = (i, k)
                w_vars[w_idx] = cp.Variable(boolean=True)
                h_i = h_vars[i]
                h_k = h_vars[(i+k) % n]
                # Linearization (McCormick)
                constraints.append(w_vars[w_idx] <= h_i)
                constraints.append(w_vars[w_idx] <= h_k)
                constraints.append(w_vars[w_idx] >= h_i + h_k - 1)
                constraints.append(w_vars[w_idx] >= 0)
                constraints.append(w_vars[w_idx] <= 1)
                
        # t <= A_k constraints
        for k in big_K:
            A_k = sum(w_vars[(i, k)] for i in range(n))
            constraints.append(t <= A_k)
            
        # Sum constraint
        sum_h = sum(h_vars[i] for i in range(n))
        constraints.append(sum_h == n/2)
        
        # Fix h_0 = 1
        h_vars[0].fix_value(1)
        
        # Objective
        objective = cp.Maximize(t)
        prob = cp.Problem(objective, constraints)

        try:
            prob.solve(solver=cp.CBC, max_seconds=min(30, budget_s))
            if prob.status == cp.OPTIMAL:
                h_solution = np.array([float(h_vars[i].value) for i in range(n)])
                # Ensure sum ≈ 30 (n/2)
                actual_sum = np.sum(h_solution)
                
                # If sum near n/2
                if abs(actual_sum - n/2) < 2:
                    # Accurate reconstruction of h from binary variables and store
                    best_h = h_solution.copy()
                    best_t = float(t.value)
                    best_c5_current = 1 - (2/n) * best_t
                    
                    # Get worst shift
                    A_exact = compute_autocorrelations(best_h)
                    A_min = np.min(A_exact)
                    if abs(A_min - best_t) < 1e-3:
                        pass  # Already optimal with respect to this set, safe to use
                    
                    # If this is sufficient, skip to polishing
                    if best_c5_current <= target_bound:
                        return (best_h, best_c5_current, n)
                    else:
                        # Continue to optimization if we want to improve it further
                        pass
        
        except Exception:
            pass  # Likely infeasible or timeout
            
    except Exception:
        pass  # fallback below will handle
    
    # Mission-warn: The binary construction as a result from a model might not always be compatible;
    # while a valid solution, it's often suboptimal compared to more careful local search.
    # So we try initial_h_values from the environment
    
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        initial_c5 = compute_c5_bound(initial_h)
        if initial_c5 < best_c5:
            best_h = initial_h.copy()
            best_c5 = initial_c5
    else:
        # Generate a binary random solution if initial_h_values not available
        h_rand = np.zeros(n)
        ones_idx = np.random.choice(n, n//2, replace=False)
        h_rand[ones_idx] = 1
        best_h = h_rand
        best_c5 = compute_c5_bound(best_h)
    
    # Now do iterative approach: Eval worst shift, add it, re-solve until convergence or time out
    num_iterations = 0
    time_limit = budget_s - 5  # Buffer
    
    # Very conservative max number of iterations in our loop to avoid Timing Out
    for iteration in range(10):
        t_now = time.time()
        if t_now - start_time > time_limit:
            break
        num_iterations += 1
        
        # Re-compute A for current best_h
        A_exact = compute_autocorrelations(best_h)
        A_min_idx = np.argmin(A_exact)
        A_min = A_exact[A_min_idx]
        
        # Check if we're already optimal for this set of shifted autocorrelations
        if best_t >= A_min:  # If the model's lower bound is already tight
            break
            
        # Add violating shift
        big_K.add(A_min_idx)
        
        # Re-solve with updated K and extra w_vars
        constraints = []
        
        # Re-declare all variables depending on K
        for i in range(n):
            h_vars[i] = cp.Variable(boolean=True)
        
        for idx in big_K:
            for i in range(n):
                w_vars[(i, idx)] = cp.Variable(boolean=True)
        
        # Constraint definitions
        for k in big_K:
            for i in range(n):
                w_var = w_vars[(i, k)]
                h_i = h_vars[i]
                h_k = h_vars[(i+k) % n]
                
                # McCormick envelopes
                constraints.append(w_var <= h_i)
                constraints.append(w_var <= h_k)
                constraints.append(w_var >= h_i + h_k - 1)
                constraints.append(w_var >= 0)
                constraints.append(w_var <= 1)
        
        t = cp.Variable()
        
        for k in big_K:
            A_k = sum(w_vars[(i, k)] for i in range(n))
            constraints.append(t <= A_k)
        
        sum_h = sum(h_vars[i] for i in range(n))
        constraints.append(sum_h == n/2)
        
        h_vars[0].fix_value(1)
        
        objective = cp.Maximize(t)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.CBC, max_seconds=min(10, time_limit - (time.time() - start_time)))
            if prob.status in [cp.OPTIMAL, cp.UNBOUNDED]:
                current_h = np.array([float(h_vars[i].value) for i in range(n)])
                if np.sum(current_h) >= n/2 - 1e-3 and np.sum(current_h) <= n/2 + 1e-3:
                    # Ensure that best solution never gets worse
                    best_t = float(t.value)
                    best_h = current_h.copy()
                    c5_try = 1 - (2/n)*best_t
                    if c5_try < best_c5:
                        best_c5 = c5_try
                    if c5_try <= target_bound:
                        return (best_h, c5_try, n)
                else:
                    pass  # not a valid sum
            else:
                break  # likely timeout or infeasible
            
        except Exception:
            break  # stay safe and terminate
    
    # Optionally apply smoothing if still better than target
    if time.time() - start_time < time_limit:
        # Apply gradient ascent polishing
        h_polished = best_h.copy()
        best_polished_c5 = best_c5
        isim = 0
        max_iter = min(200, int((time_limit - (time.time() - start_time)) / 0.5))  # let's say 0.5s per step
        for _ in range(max_iter):
            A_all = compute_autocorrelations(h_polished)
            k_star = np.argmin(A_all)
            # Gradient step
            grad = np.zeros_like(h_polished)
            for i in range(n):
                grad[i] = h_polished[(i+k_star)%n] + h_polished[(i-k_star)%n]
            # Project and adjust sum
            h_polished += 0.005 * grad
            h_polished = np.clip(h_polished, 0, 1)
            # Adjusting to preserve exact sum
            delta = np.sum(h_polished) - n/2
            if abs(delta) > 1e-6:
                # Find a pair to swap
                increase_idx = np.argmax(h_polished)  # one with max value
                decrease_idx = np.argmin(h_polished)  # one with min value
                delta_correct = min(h_polished[increase_idx], 1 - h_polished[decrease_idx])
                if delta_correct > 0:
                    h_polished[increase_idx] -= delta_correct
                    h_polished[decrease_idx] += delta_correct
                # Make sure it is still close to target sum
                # Edge case
                if h_polished[decrease_idx] > 1:
                    h_polished[decrease_idx] = 1
                    h_polished[increase_idx] += (h_polished[decrease_idx] - 1)
            # Since we're only moving one or two items, the difference is small
            c5_new = compute_c5_bound(h_polished)
            if c5_new < best_polished_c5:
                best_polished_c5 = c5_new
                # Later iterations should improve further, so saving every improvement
                best_h = h_polished.copy()
            isim += 1
            # Early stop condition
            if best_polished_c5 <= target_bound:
                break
    
    # Final return - late return in case early termination
    if best_h is None or best_c5 == float('inf'):
        # On any fail here still return a valid binary solution (if at least 1 iteration ran)
        # Generate one using a simple construction based on uniform random binary
        h_rand = np.zeros(n)
        ones_idx = np.random.choice(n, n//2, replace=False)
        h_rand[ones_idx] = 1
        best_h = h_rand
        best_c5 = compute_c5_bound(best_h)
        
    return (best_h, best_c5, n)
