import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time
import math
from cvxpy import CVXOPT, ECOS

def circular_correlation(h):
    """Compute circular correlation using FFT"""
    n = len(h)
    # Zero pad to 2*n to avoid wraparound
    h_padded = np.pad(h, (0, n), 'constant')
    h_conj = np.pad(1 - h, (0, n), 'constant')
    corr = ifft(fft(h_padded) * np.conj(fft(h_conj))).real
    # Take the correlations for shifts 0 to n-1 (the middle part)
    return corr[n//2 : 3*n//2]

def objective_autocorr(h):
    corr = circular_correlation(h)
    return np.min(corr)

def sdp_solve(h_init, n, t):
    """Solve SDP relaxation to find a lower bound"""
    h = cp.Variable(n)
    X = cp.Variable((n, n), symmetric=True)
    t_var = cp.Variable()
    
    # Objective
    objective = cp.Maximize(t_var)
    
    # Constraints for SDP
    constraints = []
    
    # Constraints for X = hh^T
    for i in range(n):
        constraints.append(X[i, i] == h[i])
    
    # Box constraints
    for i in range(n):
        constraints.append(0 <= h[i])
        constraints.append(h[i] <= 1)
    
    # Sum constraint
    constraints.append(cp.sum(h) == n/2)
    
    # Autocorrelation constraints
    # We use a structured approach to form the M_k matrices
    for k in range(n):
        # Construct matrix representing shift k
        M_k = np.zeros((n, n))
        for i in range(n):
            j1 = (i + k) % n
            j2 = (i - k) % n
            M_k[i, j1] += 1
            M_k[i, j2] += 1
        M_k = (M_k + M_k.T)/2  # Symmetrize
        constraints.append(cp.trace(M_k @ X) >= t_var)
    
    # The missing constraint: X must be positive semidefinite
    constraints.append(X >> 0)
    
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=ECOS, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return t_var.value, h.value
        else:
            return -np.inf, h_init
    except:
        return -np.inf, h_init

def round_h(X, n):
    """Round PSD solution to a binary vector using random hyperplane"""
    try:
        U, s, Vt = np.linalg.svd(X)
        r = np.sum(s > 1e-8)
        V = U[:, :r]
        g = np.random.randn(r)
        scores = V @ g
        scores_sorted = np.sort(scores)[::-1]
        if r == 0:
            return np.array([1 if i < n//2 else 0 for i in range(n)])
        
        # find threshold such that number of 1s is exactly n//2
        k = n//2
        if k >= len(scores_sorted):
            theta = scores_sorted[-1] - 1
        else:
            theta = scores_sorted[k]
        
        h = (scores >= theta).astype(float)
        if np.sum(h) != n//2:
            # crude fix by adjusting threshold
            diff = n//2 - np.sum(h)
            if diff > 0:  # need more 1s
                h[np.argsort(scores)[:diff]] = 1
            else:  # need fewer 1s
                h[np.argsort(scores)[-diff:]] = 0
        return h
    except:
        # Fallback to a heuristic 
        diag = np.diag(X)
        sorted_indices = np.argsort(-diag)
        h = np.zeros(n)
        h[sorted_indices[:n//2]] = 1
        return h

def swap_hill_climb(h, n):
    """Perform hill climb swaps to improve the solution """
    maxcorr = circular_correlation(h)
    best = np.copy(h)
    best_corr = np.min(maxcorr)
    
    # make a small number of passes
    for it in range(20):
        improved = False
        for i in range(n):
            for j in range(n):
                if h[i] == 1 and h[j] == 0:
                    # temporarily swap
                    h[i] = 0
                    h[j] = 1
                    maxcorr = circular_correlation(h)
                    corr = np.min(maxcorr)
                    if corr > best_corr:
                        best = np.copy(h)
                        best_corr = corr
                        improved = True
                    # restore
                    h[i] = 1
                    h[j] = 0
        if not improved:
            break
    return best

def gradient_fine_tune(h, n, num_iter=200):
    """Fine-tune solution using projected gradient ascent on minimum correlation"""
    step_size = 0.0001
    for _ in range(num_iter):
        corr = circular_correlation(h)
        min_k = np.argmin(corr)
        # compute gradient of min_k
        # gradient is sum of all values at indices i where j = (i + min_k) or (i - min_k)
        # (but also reflects the structure of overlaps)
        grad = np.zeros(n)
        # based on the derivative of h(i)*h((i+k) % n) for minimum k:
        # we expect smaller counts (but negative , so increasing h where it's 0)
        for i in range(n):
            j1 = (i + min_k) % n
            j2 = (i - min_k) % n
            grad[i] = h[j1] + h[j2]
        
        # update with bounds
        h += step_size * grad
        h = np.clip(h, 0, 1)
        # Project to sum = n/2
        current_sum = np.sum(h)
        if current_sum > 0:
            h *= (n/2) / current_sum
    return h

def interval_optimize(n=200):
    """Optimize step function using interval model"""
    # Model as 6 intervals, symmetric about 1
    # (total of 3 intervals on left and 3 on right)
    # Now 6 free variables a1,b1,a2,b2,a3,b3,
    # where a1 < b1 < a2 < b2 < a3 < b3,
    # b1=2-a1, b2=2-a2, b3=2-a3
    # Total area = (b1-a1)+(b2-a2)+(b3-a3) = 1
    # So  (2-a1-a1)+(2-a2-a2)+(2-a3-a3) = 1
    # i.e. 6 - 2*(a1+a2+a3) = 1
    # i.e. a1+a2+a3 = 2.5
    def obj_func(vars):
        a1,a2,a3 = vars
        # fix symmetry
        a = [a1, a2, a3, 2-a3, 2-a2, 2-a1]
        b = [a1+b1, a2+b2, a3+b3, 2-b3, 2-b2, 2-b1]
        lengths = [b[i]-a[i] for i in range(6)]
        # total area constraint
        total = sum(lengths)
        penalty = 1e6 * (total - 1)**2
        # compute all shifts and get minimum overlap
        # for now, return area penalty, assume areas are closed form
        min_corr = np.inf
        return penalty
        
    # dummy with a short search
    min_h = np.full(n, 0.)
    min_c5 = 1.0
    return min_h, min_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_sdp = 80
    n = 200  # for integration accuracy in final step
    
    best_h = None
    best_c5 = np.inf
    
    # step 1: Try SDP with random seed
    for _ in range(30):
        t_s = time.time()
        if time.time()-start_time > budget_s - 30:
            break
        # Generate an initial h
        h_init = np.random.rand(n_sdp)
        h_init = np.clip(h_init, 0, 1)
        # normalize to sum = n/2
        h_init = h_init * (n_sdp/2) / np.sum(h_init)
        sdp_t, sdp_h = sdp_solve(h_init, n_sdp, 0)
        # if not very useful, skip
        if sdp_t < 0.01:
            continue
            
        # try 10 rounds of rounding
        for _ in range(10):
            if time.time() - start_time > budget_s - 30:
                break
            h_rounded = round_h(np.outer(sdp_h, sdp_h), n_sdp)
            h_rounded = swap_hill_climb(h_rounded, n_sdp)
            h_rounded = gradient_fine_tune(h_rounded, n_sdp, 100)
            
            # convert back to n = 200 step function
            h_final = np.zeros(n)
            idx = np.floor(np.linspace(0, n_sdp, n, endpoint=False)).astype(int)
            h_final = h_rounded[idx]
            
            if np.sum(h_final) > 0:
                c5_val = 1.0 - 2.0 / n * objective_autocorr(h_final)
                if c5_val < best_c5:
                    best_h = h_final.copy()
                    best_c5 = c5_val
    
    # step 2: Second line: Few interval heuristic
    if time.time() - start_time < budget_s - 10:
        h_interval, c5_interval = interval_optimize(n)
        if c5_interval < best_c5:
            best_h = h_interval
            best_c5 = c5_interval
            
    # Validate final solution
    if best_h is not None:
        c5_val = 1.0 - 2.0 / n * objective_autocorr(best_h)
        return (best_h, c5_val, n)
    else:
        # fallback to simplest construction
        h_simple = np.zeros(n)
        h_simple[:n//2] = 1
        return (h_simple, 0.5, n)
