import numpy as np
import time
from scipy.optimize import linprog
import cvxpy as cp

def project_onto_capped_simplex(v, S):
    """Project vector v onto {x | 0 ≤ x_i ≤ 1, Σx_i = S} using bisection."""
    v = np.clip(v, 0, 1)
    low, high = -2.0, 2.0
    tol = 1e-10
    while high - low > tol:
        mid = (low + high) / 2.0
        h = np.clip(v - mid, 0, 1)
        if np.sum(h) > S:
            low = mid
        else:
            high = mid
    return np.clip(v - (low + high) / 2.0, 0, 1)

def compute_coeffs_and_constants(h, n):
    """Compute linearization coefficients a_{i,k} and constants c_k."""
    # Index ranges
    a = np.zeros((n, 2*n - 1))
    c = np.zeros(2*n - 1)
    # Shift indices in range [-n+1, n-1]
    for k in range(-(n-1), n):
        k_idx = k + n - 1  # offset to be index in [0, 2n-2]
        # Compute a_{i,k}
        a[:, k_idx] = 1.0 - np.roll(h, k) - np.roll(h, -k)
        # Handle 'zero padding'
        if k > 0:
            a[:k, k_idx] = 0.0
        elif k < 0:
            a[k:, k_idx] = 0.0
        # Compute c_k
        if k == 0:
            c[k_idx] = np.sum(h**2)
        else:
            c[k_idx] = np.sum(h * np.roll(h, k))
    return a, c

def solve_linear_program(a, c, n, S):
    """Solve the LP for current linearization."""
    # Variables: h[0], ..., h[n-1], t
    x = cp.Variable(n + 1)
    h = x[:n]
    t = x[n]
    # Objective: minimize t
    objective = cp.Minimize(t)
    # Constraints:
    # a_k^T * h + c_k <= t for all k
    constraints = []
    for k in range(2*n - 1):
        # Add constraint a[:,k]^T * h + c[k] <= t
        constraints.append(a[:, k] @ h + c[k] <= t)
    # Mass constraint
    constraints.append(cp.sum(h) == S)
    # Box constraints
    constraints.append(h >= 0)
    constraints.append(h <= 1)
    # Optional: t >= 0
    constraints.append(t >= 0)
    # Solve it
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status in ["optimal", "optimal_inaccurate"]:
            h_new = h.value
            t_new = t.value
            return h_new, t_new
        else:
            # Fallback: project the best solution attempt we can use
            # The simplest is to return h = 0.5 * ones for fallback
            h_new = np.full(n, 0.5)
            # Reuse previous t or estimate: very conservative estimate
            t_new = 1.0
            return h_new, t_new
    except:
        # Emergency fallback
        h_new = np.full(n, 0.5)
        t_new = 1.0
        return h_new, t_new

def evaluate(h):
    """Evaluate C5 = max_k ∫ h(x)(1 - h(x+k)) dx."""
    n = len(h)
    dx = 2.0 / n
    # compute fully convolution h * (1 - h)
    corr = np.correlate(h, 1 - h, mode="full")
    # max overlap index ranges from -(n-1) to (n-1) so output is length 2*n-1
    max_idx = np.argmax(corr)
    # The center (max overlap) corresponds to k = 0 shift
    return corr[max_idx] * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    # Get the initial guess
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        n = 70
        h = np.random.rand(n)
        h = project_onto_capped_simplex(h, n / 2)
    n = len(h)
    S = n / 2

    best_h = h.copy()
    best_C5 = evaluate(h)
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed

    # Random restarts
    num_restarts = 100  # Some sampling; will not do all
    best_C5_global = best_C5
    best_h_global = best_h.copy()
    tol = 1e-6

    for restart in range(num_restarts):
        if restart > 0:
            # Random different init
            h = np.random.rand(n) * 0.8  # to avoid extreme values
            h = project_onto_capped_simplex(h, S)
        h_current = h.copy()
        old_C5 = float('inf')
        max_iter = 30 
        for iter_num in range(max_iter):
            elapsed = time.time() - start_time
            if elapsed > 0.95 * budget_s:
                break
            a, c = compute_coeffs_and_constants(h_current, n)
            h_new, t_new = solve_linear_program(a, c, n, S)
            # Apply under-relaxation
            lambda_relax = 0.7
            h_new = lambda_relax * h_new + (1 - lambda_relax) * h_current
            h_new = project_onto_capped_simplex(h_new, S)
            h_current = h_new

            new_C5 = evaluate(h_current)
            if new_C5 < best_C5_global:
                best_C5_global = new_C5
                best_h_global = h_current.copy()
                # Early termination if improvement is small
                if abs(new_C5 - old_C5) < 1e-8:
                    break
            old_C5 = new_C5
            if abs(new_C5 - old_C5) < tol:
                break

        # Post-processing: polishing with mass transfer
        polishing_moves = 1000
        h_polished = h_current.copy()
        for move in range(polishing_moves):
            if time.time() - start_time > 0.95 * budget_s:
                break
            # pick one of the worst indices based on full corr
            corr = np.correlate(h_polished, 1 - h_polished, mode='full')
            max_corr_idx = np.argmax(corr)
            worst_k = max_corr_idx - (n-1)  # translate back to shift index
            # change by moving mass a small amount
            # We adjust arbitrary pairs that have h_i > 0, h_{i+k} < 1
            delta = 0.005
            i_candidates = np.where(h_polished > 0.01)[0]
            for i in i_candidates:
                j = i + worst_k
                if j >= 0 and j < n and h_polished[j] < 0.99:
                    h_new = h_polished.copy()
                    h_new[i] -= delta
                    h_new[j] += delta
                    if np.all(0 <= h_new) and np.all(h_new <= 1):
                        h_new = project_onto_capped_simplex(h_new, S)
                        new_C5 = evaluate(h_new)
                        if new_C5 < evaluate(h_polished):
                            h_polished = h_new
                            break
        final_C5 = evaluate(h_polished)
        if final_C5 < best_C5_global:
            best_C5_global = final_C5
            best_h_global = h_polished
        if final_C5 < best_C5:
            best_C5 = final_C5
            best_h = h_polished.copy()

        elapsed = time.time() - start_time
        if elapsed >= 0.95 * budget_s:
            break

    # If we have a finer grid, try interpolation and refinement
    fine_grid = False
    if n == 70:
        fine_grid = True
        h_fine = np.zeros(140)
        for i in range(70):
            h_fine[2*i] = h[i] / 2
            h_fine[2*i+1] = h[i] / 2
        h_fine = project_onto_capped_simplex(h_fine, 70)
        # Refine
        h_refine = h_fine.copy()
        old_C5 = float('inf')
        max_iter = 30
        for iter_num in range(max_iter):
            a, c = compute_coeffs_and_constants(h_refine, 140)
            h_new, t_new = solve_linear_program(a, c, 140, 70)
            lambda_relax = 0.7
            h_new = lambda_relax * h_new + (1 - lambda_relax) * h_refine
            h_new = project_onto_capped_simplex(h_new, 70)
            h_refine = h_new
            new_C5 = evaluate(h_refine)
            if new_C5 < old_C5:
                if new_C5 < best_C5_global:
                    best_C5_global = new_C5
                    best_h_global = h_refine.copy()
                old_C5 = new_C5
                if abs(new_C5 - old_C5) < tol:
                    break
            if time.time() - start_time > 0.95 * budget_s:
                break
        final_C5 = evaluate(h_refine)
        if final_C5 < best_C5_global:
            best_C5_global = final_C5
            best_h_global = h_refine.copy()

    return best_h_global, best_C5_global, len(best_h_global)
