import numpy as np
from scipy.optimize import differential_evolution
from scipy.fftpack import fft, ifft
from cvxpy import *
import time
import math
from scipy.optimize import minimize

def evaluate_h(h, dx):
    # Compute full correlation using FFT
    F = fft(h)
    G = fft(1 - h)
    correlation = ifft(F * np.conj(G)).real
    max_overlap = np.max(correlation) * dx
    return max_overlap

def project_to_simplex(h, target_sum):
    """Project h onto the simplex sum(h) = target_sum using Dykstra's method"""
    h = np.array(h, dtype=float)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Affine scaling: h_new = h - delta * ones where delta = (current_sum - target_sum) / n
    delta = (current_sum - target_sum) / len(h)
    h = h - delta
    
    # Clamp to [0,1] and re-project if needed
    h = np.clip(h, 0, 1)
    
    # Fine-tune using Dykstra's alternating projection for better accuracy
    for _ in range(5):
        current_sum = np.sum(h)
        delta = (current_sum - target_sum) / len(h)
        h = h - delta
        h = np.clip(h, 0, 1)
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Start timer
    start_time = time.time()
    
    # Use a large enough but manageable sample size
    n_points = 180
    dx = 2.0 / n_points
    
    # Initialize with interpolated initial_h_values
    initial_h = initial_h_values
    if len(initial_h) != n_points:
        # Linear interpolation from the provided initial values to target length
        t_old = np.linspace(0, 2, len(initial_h))
        t_new = np.linspace(0, 2, n_points)
        initial_h = np.interp(t_new, t_old, initial_h)
    
    # Ensure the sum constraint is met
    h = project_to_simplex(initial_h, n_points / 2)
    
    # Final refinement using continuous optimization via differential evolution
    def objective(h_vec):
        h_vec = np.array(h_vec)
        h_vec = project_to_simplex(h_vec, n_points / 2)
        return evaluate_h(h_vec, dx)
    
    # Bounds for each x[i]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Set up DE with reduced population for faster compute
    popsize = 30
    maxiter = int((budget_s * 0.6) / 0.05)  # Roughly 600s * 0.6 / 0.05 ~ 7200 gens
    maxiter = min(maxiter, 10000)
    
    # Run DE
    res = differential_evolution(
        func=objective,
        bounds=bounds,
        maxiter=maxiter,
        popsize=popsize,
        seed=seed,
        disp=False,
        strategy='best1bin'
    )
    
    # Extract best solution from DE
    h = project_to_simplex(res.x, n_points / 2)
    
    # Local improvement via mismatch swap on the best solution found so far
    def mismatch_swap_step(h_vec, dx):
        original_h = h_vec.copy()
        h_vec = np.array(h_vec)
        F = fft(h_vec)
        G = fft(1 - h_vec)
        correlation = ifft(F * np.conj(G)).real
        max_shift = np.argmax(correlation)
        max_val = correlation[max_shift]
        
        # Compute I_{max_shift} and identify mismatch sets
        n = len(h_vec)
        if max_shift >= 0:
            # Valid indices where both h_i and h_{i+max_shift} exist
            indices = list(range(n - max_shift))
        else:
            # For negative shift, valid indices for h_{i+max_shift} to exist
            max_shift = -max_shift
            indices = list(range(max_shift, n))
            
        indices = np.array(indices)
        # Calculate mismatch sets
        M0 = []  # i where h_i ≈ 0 and h_{i+max_shift} ≈ 1
        M1 = []  # i where h_i ≈ 1 and h_{i+max_shift} ≈ 0
        
        for i in indices:
            if h_vec[i] < 0.1 and h_vec[i + max_shift] > 0.9:
                M0.append(i)
            elif h_vec[i] > 0.9 and h_vec[i + max_shift] < 0.1:
                M1.append(i)
        
        # Perform swaps to reduce overlap
        swap_count = 0
        while M0 and M1 and max_val > 1e-7:
            i = M1.pop(0)
            j = M0.pop(0)
            
            # Swap values
            h_vec[i], h_vec[j] = h_vec[j], h_vec[i]
            
            # Renormalize after swap and constraints
            h_vec = project_to_simplex(h_vec, n_points / 2)
            
            # Recompute the correlation to see if we improved
            if max_val <= 1e-7:
                break
            
            # Update mismatch sets
            M0.clear()
            M1.clear()
            # Redo checks to avoid inconsistency due to swapped values
            
            F = fft(h_vec)
            G = fft(1 - h_vec)
            correlation = ifft(F * np.conj(G)).real
            max_shift_new = np.argmax(correlation)
            max_val_new = correlation[max_shift_new]
            
            # Update mismatch sets for next loop
            if max_shift_new >= 0:
                indices_new = list(range(n - max_shift_new))
            else:
                max_shift_new = -max_shift_new
                indices_new = list(range(max_shift_new, n))
                
            indices_new = np.array(indices_new)
            for i_idx in indices_new:
                if h_vec[i_idx] < 0.1 and h_vec[i_idx + max_shift_new] > 0.9:
                    M0.append(i_idx)
                elif h_vec[i_idx] > 0.9 and h_vec[i_idx + max_shift_new] < 0.1:
                    M1.append(i_idx)
                    
            max_val = max_val_new
            
            swap_count += 1
            if swap_count >= 1000:
                break
        
        return h_vec
    
    # Apply mismatch swap
    h = mismatch_swap_step(h, dx)
    
    # Final evaluation
    final_C5 = evaluate_h(h, dx)
    
    # Return final values
    return (list(h), final_C5, n_points)
