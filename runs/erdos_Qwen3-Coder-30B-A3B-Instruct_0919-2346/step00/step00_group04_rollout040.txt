    def compute_overlap(h):
        # Compute correlation using numpy's correlate function
        corr = np.correlate(h, 1 - h, mode="full")
        max_corr = np.max(corr)
        # Convert to overlap integral value
        return max_corr * dx
    
    def compute_max_overlap(h):
        """Calculate the maximum overlap integral for given h"""
        corr = np.correlate(h, 1 - h, mode="full")
        # Get max over all non-zero lags
        max_val = np.max(corr[1:])  # ignore zero lag
        return max_val * dx
    
    # Simulated Annealing parameters
    current_h = h_values.copy()
    current_cost = compute_max_overlap(current_h)
    best_h = current_h.copy()
    best_cost = current_cost
    
    # Initial temperature
    temp = 0.1
    cooling_rate = 0.9999
    min_temp = 1e-6
    max_iter = 100000
    iter_count = 0
    
    # Continue optimizing until time expires
    while time.time() - start_time < budget_s * 0.95 and temp > min_temp:
        # Generate candidate solution by swapping two elements
        i, j = np.random.choice(n_points, 2, replace=False)
        new_h = current_h.copy()
        new_h[i], new_h[j] = new_h[j], new_h[i]
        
        # Ensure binary-like values by projecting to [0, 1]
        new_h = np.clip(new_h, 0, 1)

        current_sum = np.sum(new_h)

        if current_cost < best_cost:
            best_h = current_h.copy()
