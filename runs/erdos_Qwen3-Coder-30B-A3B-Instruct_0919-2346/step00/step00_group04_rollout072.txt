```python
import numpy as np
import time
import random
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    def evaluate(h):
        n = len(h)
        dx = 2.0 / n
        O = np.correlate(h, 1 - h, mode='full')
        return O.max() * dx

    def random_binary(n):
        h = np.random.choice([0, 1], size=n, p=[0.5, 0.5])
        while np.sum(h) != n // 2:
            if np.sum(h) > n // 2:
                idx = np.where(h == 1)[0]
                np.random.shuffle(idx)
                h[idx[0]] = 0
            else:
                idx = np.where(h == 0)[0]
                np.random.shuffle(idx)
                h[idx[0]] = 1
        return h

    def thue_morse(n):
        h = np.zeros(n, dtype=int)
        for i in range(n):
            h[i] = bin(i).count('1') % 2
        return h

    def legendre_sequence(p):
        h = np.zeros(p, dtype=int)
        for i in range(1, p):
            if pow(i, (p-1)//2, p) == 1:
                h[i] = 1
        # Ensure weight is about half
        ones = np.where(h == 1)[0]
        if len(ones) > p//2:
            idx = np.random.choice(ones, size=len(ones) - p//2, replace=False)
            h[idx] = 0
        elif len(ones) < p//2:
            zeros = np.where(h == 0)[0]
            idx = np.random.choice(zeros, size=p//2 - len(ones), replace=False)
            h[idx] = 1
        return h

    def interpolate_initial_h(n):
        if 'initial_h_values' not in globals():
            return None
        base = np.array(initial_h_values, dtype=float)
        x_old = np.linspace(0, 2, len(base), endpoint=False)
        x_new = np.linspace(0, 2, n, endpoint=False)
        h_interp = np.interp(x_new, x_old, base)
        h_bin = (h_interp > 0.5).astype(int)
        # Adjust to have sum = n//2
        current_sum = np.sum(h_bin)
        target_sum = n // 2
        if current_sum > target_sum:
            idx = np.where(h_bin == 1)[0]
            np.random.shuffle(idx)
            h_bin[idx[:current_sum - target_sum]] = 0
        elif current_sum < target_sum:
            idx = np.where(h_bin == 0)[0]
            np.random.shuffle(idx)
            h_bin[idx[:target_sum - current_sum]] = 1
        return h_bin

    # Candidate lengths (must be even)
    lengths = [64, 80, 96, 112, 128, 144, 160, 176, 192, 224, 256, 320, 384, 448, 512]
    lengths = [n for n in lengths if n <= 800 and n % 2 == 0]

    best_h = None
    best_c5 = float('inf')
    best_n = None

    # Generate candidate sequences
    candidates = []
    for n in lengths:
        # Thue-Morse for power of 2
        if (n & (n - 1)) == 0:
            seq = thue_morse(n)
            candidates.append((seq, n))
        # Legendre for prime
        if n > 2 and all(n % i != 0 for i in range(2, int(math.sqrt(n)) + 1)):
            seq = legendre_sequence(n)
            candidates.append((seq, n))
        # Interpolated initial sequence
        seq = interpolate_initial_h(n)
        if seq is not None:
            candidates.append((seq, n))
        # Random binary
        for i in range(2):
            seq = random_binary(n)
            candidates.append((seq, n))

    # Evaluate candidates
    for seq, n in candidates:
        c5_val = evaluate(seq)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = seq.copy()
            best_n = n

    # Local search for top candidates (kept top 5)
    top_candidates = []
    for seq, n in candidates:
        top_candidates.append((seq.copy(), n))
    top_candidates.sort(key=lambda x: evaluate(x[0]))
    top_candidates = top_candidates[:5]

    for h, n in top_candidates:
        if time.time() - start_time > 0.9 * budget_s:
            break
        h, c5 = local_search(h, n, budget_s - (time.time() - start_time))
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
            best_n = n

    return (best_h.tolist(), best_c5, best_n)

def local_search(h, n, time_budget):
    def evaluate(h):
        dx = 2.0 / n
        O = np.correlate(h, 1 - h, mode='full')
        return O.max() * dx

    current_val = evaluate(h)
    best_h = h.copy()
    best_val = current_val
    
    start_time = time.time()
    max_iter = 2000000
    temp_start = 0.1
    temp_end = 1e-5
    
    # Maintain index lists for efficiency
    ones_idx = list(np.where(h == 1)[0])
    zeros_idx = list(np.where(h == 0)[0])
    
    for it in range(max_iter):
        if time.time() - start_time > time_budget:
            break
            
        temp = temp_start * (temp_end / temp_start) ** (it / max_iter)
        
        # Choose random 1 and 0
        i = np.random.choice(ones_idx)
        j = np.random.choice(zeros_idx)
        
        # Swap
        h[i], h[j] = 0, 1
        ones_idx[ones_idx.index(i)] = j
        zeros_idx[zeros_idx.index(j)] = i
        
        val = evaluate(h)
        delta = val - current_val
        
        # Accept or not
        if delta <= 0 or np.random.rand() < math.exp(-delta / temp):
            current_val = val
            if val < best_val:
                best_val = val
                best_h = h.copy()
        else:
            # Revert swap
            h[i], h[j] = 1, 0
            ones_idx[ones_idx.index(j)] = i
            zeros_idx[zeros_idx.index(i)] = j
    
    return best_h, best_val
```
