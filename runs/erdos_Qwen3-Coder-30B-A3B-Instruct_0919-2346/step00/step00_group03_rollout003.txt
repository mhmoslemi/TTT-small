def evaluate_c5(h):
    """Evaluate C5 bound for given h values."""
    dx = 2.0 / len(h)

    corr = correlate(h, 1 - h, mode='full')
    # The maximum value is the upper bound C5
    c5 = np.max(corr) * dx
    return c5
