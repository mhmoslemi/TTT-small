from scipy.optimize import differential_evolution
import time
from math import sqrt

def evaluate_overlap(h, dx):
    """Compute the maximum overlap integral for given h values."""
    # Ensure h is numpy array
    h = np.array(h)
    # Compute correlation of h with (1-h) using full mode
    correlation = np.correlate(h, 1-h, mode="full")
    # Maximum value over all shifts
    max_overlap = np.max(correlation)
    return max_overlap * dx
