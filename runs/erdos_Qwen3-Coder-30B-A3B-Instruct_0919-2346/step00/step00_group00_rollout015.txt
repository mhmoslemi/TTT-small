import numpy as np
import time
from scipy.signal import convolve

def project_to_capped_simplex(h, target_sum):
    """
    Projects h to the capped simplex: 0 <= h[i] <= 1 and sum(h) = target_sum
    Uses water-filling approach.
    """
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h

    # Sort in descending order
    sorted_idx = np.argsort(h)[::-1]
    sorted_h = h[sorted_idx]
    
    # Water filling algorithm
    cumsum = np.cumsum(sorted_h)
    n = len(sorted_h)
    
    # Find where to place the water level
    for i in range(n):
        if cumsum[i] + (n - i - 1) * sorted_h[i] >= target_sum:
            break
    
    if i == n - 1:
        # All elements are active
        lambda_val = (cumsum[n-1] - target_sum) / n
    else:
        lambda_val = (cumsum[i] - target_sum) / (n - i) if i < n - 1 else 0

    new_h = np.maximum(sorted_h - lambda_val, 0)
    # Reconstruct full vector
    result = h.copy()
    result[sorted_idx] = new_h
    return result

def compute_correlation(h):
    # Compute the correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # Only consider the valid correlations (non-zero padding)
    n = len(h)
    offset = len(corr) // 2
    valid_corr = corr[offset - n + 1 : offset + n]
    return valid_corr

def evaluate_solution(h, dx):
    # Compute full correlation
    corr = compute_correlation(h)
    c5 = dx * np.max(corr)
    return c5

def compute_gradient(h, alpha, dx):
    """
    Compute the gradient of F_alpha(h) with respect to h,
    where F_alpha is log-sum-exp smooth approximation to max.
    """
    n = len(h)
    # Compute all O_ell(h) for ell in [0,2n-2]
    O_ell = []
    for k in range(-n + 1, n):
        # Compute O_k(h) = dx * sum_{i} h[i] * (1 - h[i+k]) 
        # but only sum over valid indices
        if k >= 0:
            subset_h = h[k:]
            subset_h2 = h[:len(subset_h)]
            O_k = dx * np.sum(subset_h * (1 - subset_h2))
        else:
            subset_h = h[:k]
            subset_h2 = h[-k:]
            O_k = dx * np.sum(subset_h * (1 - subset_h2))
        O_ell.append(O_k)
    
    O_ell = np.array(O_ell)
    
    # Compute soft max weights w_ell = exp(alpha * O_ell) / sum
    exp_O = np.exp(alpha * O_ell)
    w = exp_O / np.sum(exp_O)
    
    # g = conv1d(w, 1-h) - conv1d(w_rev, h)
    # w_rev is reverse of w
    w_rev = w[::-1]
    # For seamless convolution using different modes, we truncate appropriately as needed
    # Compute convolution g1 = conv(w, 1-h)
    g1 = np.convolve(w, 1 - h, mode='full')
    # Compute convolution g2 = conv(w_rev, h)  
    g2 = np.convolve(w_rev, h, mode='full')
    # The full result
    gradient = g1 - g2
    # Still need to clip to match the current length n
    gradient = gradient[:n]
        
    return gradient

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Configuration
    n = 200  # Even number of points
    dx = 2.0 / n
    target_sum = n / 2.
    alpha_start = 0.5
    alpha_end = 200.0
    gamma = 1.3
    steps_per_anneal = 200
    initial_lr = 0.01
    lr_decay = 0.99
    n_seeds = 6  # Multi-start using different seeds
    
    best_c5 = float('inf')
    best_h_binary = None
    
    print(f"Starting optimization with n={n}, budget={budget_s}s")
    
    # Multi-start: go through several seeds
    for seed_i in range(n_seeds):
        if time.time() - start_time > budget_s - 5:
            break
            
        print(f"Seed {seed_i}")
        # Initialize h
        if seed_i == 0 and 'initial_h_values' in globals():
            # Use initial constructed value
            h = np.array(initial_h_values[:n])
            # Clamp to [0,1] and mean to target
            h = np.clip(h, 0, 1)
            h = project_to_capped_simplex(h, target_sum)
        else:
            # Random initialization, project to feasible set
            h = np.random.rand(n)
            h = project_to_capped_simplex(h, target_sum)
            
        # Current value of alpha
        alpha = alpha_start
        lr = initial_lr
        old_F_alpha = float('inf')
        
        # Annealing loop
        step = 0
        while alpha <= alpha_end:
            step += 1
            if time.time() - start_time > budget_s - 5:
                # Stop early
                break
                
            # Compute F_alpha and gradient
            corr_full = compute_correlation(h)
            O = np.array([dx * np.sum(h[i:] * (1 - h[:len(h) - i])) 
                          if i >= 0 else 
                          dx * np.sum(h[:len(h)+i] * (1 - h[-i:]))
                          for i in range(-n+1, n)])
            exp_O = np.exp(alpha * O)
            Z = np.sum(exp_O)
            F_alpha = (dx / alpha) * np.log(Z)
            
            grad = compute_gradient(h, alpha, dx)  # gradient w.r.t. h
            
            # Gradient step
            h = h - lr * grad
            
            # Project back to feasible set
            h = project_to_capped_simplex(h, target_sum)
            
            # Update learning rate
            if step % 10 == 0:
                lr *= lr_decay 
                
            # Anneal temperature
            if step % steps_per_anneal == 0:
                new_alpha = alpha * gamma
                if new_alpha > alpha_end:
                    new_alpha = alpha_end
                # Check improvement
                diff = old_F_alpha - F_alpha
                old_F_alpha = F_alpha
                if diff < 1e-5:
                    break
                alpha = new_alpha 
                
    print(f"Best approximation heatmap-like variant: {best_c5:.6f}")

    # Finalize
    # We already have a best binary solution from the multiple seeds
    # But we recompute one final best case as the above loop is aggregated
        
    # Round final result to binary
    h_final = np.array(initial_h_values[:n]) if 'initial_h_values' in globals() else np.random.rand(n)
    h_final = np.clip(h_final, 0, 1)
    H_sum = np.sum(h_final)
    if abs(H_sum - target_sum) > 1e-8:
        h_final = project_to_capped_simplex(h_final, target_sum)

    # Sort indices and set top half to 1
    sort_idx = np.argsort(h_final)[::-1]
    h_binary = np.zeros(n)
    h_binary[sort_idx[:int(target_sum)]] = 1.0
    h_binary = np.array(h_binary)
        
    # Evaluate the binary solution exactly
    c5_final = evaluate_solution(h_binary, dx)
    print(f"Final rounded binary solution: C5 = {c5_final:.6f}")
    
    return h_binary, c5_final, n
