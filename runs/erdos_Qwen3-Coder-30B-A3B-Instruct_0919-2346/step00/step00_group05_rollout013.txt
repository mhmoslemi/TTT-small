import numpy as np
import time
from cvxpy import *
import cvxpy as cp

def evaluate(h):
    """Evaluate C5 using correlation."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_onto_capped_simplex(v, S):
    """Project vector v onto {h | 0≤h_i≤1, sum h_i = S} using bisection."""
    v = np.clip(v, 0, 1)
    low, high = -2.0, 2.0
    while high - low > 1e-10:
        mid = (low + high) / 2
        h = np.clip(v - mid, 0, 1)
        if np.sum(h) > S:
            low = mid
        else:
            high = mid
    return np.clip(v - (low + high) / 2, 0, 1)

def coeffs_and_const(h):
    """Compute linearization coefficients a_{i,k} and constants c_k."""
    n = len(h)
    S = n / 2.0
    a_matrix = np.zeros((2 * n - 1, n))
    const_vec = np.zeros(2 * n - 1)
    shifts = list(range(-(n - 1), n))
    
    for idx, k in enumerate(shifts):
        # shift k: from i to i+k
        # we fill in each row with coefficients for this shift
        # row index: idx, col index: i (0 to n-1)
        # coefficient for h_i = 1 - h_{i+k} - h_{i-k}
        for i in range(n):
            term1 = 0 if i + k < 0 or i + k >= n else h[i + k]
            term2 = 0 if i - k < 0 or i - k >= n else h[i - k]
            a_matrix[idx, i] = 1.0 - term1 - term2
            # handle zero padding: missing indices are effectively zero
        # compute constant for this shift
        c = 0.0
        for i in range(n):
            term1 = 0 if i + k < 0 or i + k >= n else h[i + k]
            c += h[i] * term1
        const_vec[idx] = c
    return a_matrix, const_vec, shifts

def linear_program(a_matrix, const_vec, n, S, shifts):
    """Solve the linear program: minimize t subject to linear constraints."""
    h = cp.Variable(n)
    t = cp.Variable()
    
    # Linear constraints (each shift corresponds to one constraint)
    constraints = [
        a_matrix[i] @ h + const_vec[i] <= t for i in range(len(shifts))
    ]
    
    # Mass constraint
    constraints.append(cp.sum(h) == S)
    
    # Box constraints
    constraints.append(h >= 0)
    constraints.append(h <= 1)
    
    # Objective
    objective = cp.Minimize(t)
    
    # Solve
    prob = cp.Problem(objective, constraints)
    
    # Use ECOS for fast solving, which is inside cvxpy
    prob.solve(solver=ECOS, verbose=False)
    
    if prob.status in ["optimal", "optimal_inaccurate"]:
        return h.value, t.value
    else:
        # fallback - return invalid value to indicate failure
        return None, np.inf

def random_feasible(n):
    """Generate a feasible starting vector."""
    # Start with a simple piecewise constant vector
    h = np.random.rand(n)
    # Project to feasible capped simplex
    h = project_onto_capped_simplex(h, n / 2)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n = len(h)
    except:
        # default
        n = 70
        h = np.copy(np.array(initial_h_values)) if 'initial_h_values' in globals() else np.ones(n) * 0.5
        h = project_onto_capped_simplex(h, n / 2)
    
    # Scale to match integral constraint
    if abs(np.sum(h) - n/2) > 1e-10:
        h = project_onto_capped_simplex(h, n / 2)
    
    best_h = np.copy(h)
    best_C5 = evaluate(best_h)
    
    # Common values
    S = n / 2.0
    
    # Number of restarts to try
    max_restarts = 200
    restart_count = 0
    iter_count = 0
    max_iter_per_restart = 30
    
    while time.time() - start_time < 0.95 * budget_s and restart_count < max_restarts:
        # Initialize new restart
        restart_count += 1
        
        # Start from either initial_h_values or a random vector
        if restart_count == 1:
            h = np.copy(best_h)
        else:
            # Random linear combination of best solution and slight perturbations
            h = random_feasible(n)
        
        # Run inner CCP loop
        for _ in range(max_iter_per_restart):
            if time.time() - start_time > 0.95 * budget_s:
                break
            iter_count += 1
            
            # Compute linearization point
            a_mat, const_vec, shifts = coeffs_and_const(h)
            
            # Solve linear program
            h_new, t_new = linear_program(a_mat, const_vec, n, S, shifts)
            
            if h_new is None or np.isnan(t_new) or np.isinf(t_new):
                # Use fallback
                break
            
            # Under-relaxation
            lambda_ = 0.7
            h = lambda_ * h_new + (1 - lambda_) * h
            
            # Project to feasible set
            h = project_onto_capped_simplex(h, S)
            
            # Evaluate actual function
            current_C5 = evaluate(h)
            
            if current_C5 < best_C5:
                best_h = np.copy(h)
                best_C5 = current_C5
            
            # Check convergence using t change (a proxy)
            if abs(t_new - best_C5) < 1e-6:
                break
        
        # Perform short random perturbation post-CCP on the final solution
        h_perturbed = np.copy(best_h)
        for _ in range(200):
            # pick two points and move a small amount from one to the other
            i = np.random.randint(n)
            j = np.random.randint(n)
            a = np.random.rand() * 0.05
            if h_perturbed[i] > a and h_perturbed[j] < 1 - a:
                h_perturbed[i] -= a
                h_perturbed[j] += a
        current_C5 = evaluate(h_perturbed)
        if current_C5 < best_C5:
            best_h = np.copy(h_perturbed)
            best_C5 = current_C5
    
    return best_h, best_C5, n
