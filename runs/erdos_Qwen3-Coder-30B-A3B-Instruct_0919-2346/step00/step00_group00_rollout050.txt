import numpy as np
import time
import random

def compute_c5(h):
    """Compute C5 bound from step function h"""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return corr.max()

def propose_swap_and_update(h, corr, i, j):
    """Propose a swap at indices i and j, update correlation incrementally"""
    old_i = h[i]
    old_j = h[j]
    new_i = old_j
    new_j = old_i
    
    delta_i = new_i - old_i
    delta_j = new_j - old_j
    
    n = len(h)
    corr_delta = np.zeros(2 * n - 1, dtype=float)
    
    # Loop over k not equal to i or j
    for k in range(n):
        if k == i or k == j:
            continue
        hk = h[k]
        # Term h[i] * (1 - h[k]) at shift k - i
        idx1 = k - i + n - 1
        corr_delta[idx1] += delta_i * (1 - hk)
        
        # Term h[k] * (1 - h[i]) at shift i - k
        idx2 = i - k + n - 1
        corr_delta[idx2] += hk * (-delta_i)

        # Term h[j] * (1 - h[k]) at shift k - j
        idx3 = k - j + n - 1
        corr_delta[idx3] += delta_j * (1 - hk)

        # Term h[k] * (1 - h[j]) at shift j - k
        idx4 = j - k + n - 1
        corr_delta[idx4] += hk * (-delta_j)
    
    # Handle cross term pairs (i,j) and (j,i)
    # Pair (i,j): shift j - i
    idx_ij = j - i + n - 1
    corr_delta[idx_ij] += delta_i

    # Pair (j,i): shift i - j  
    idx_ji = i - j + n - 1
    corr_delta[idx_ji] += delta_j

    # Apply changes to correlation
    new_corr = corr + corr_delta
    return new_corr

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()
    best_c5 = np.inf
    best_h = None
    best_n = None
    
    # Try various values of n
    n_candidates = [60, 80, 100, 120, 140, 160, 180, 200]
    
    for n in n_candidates:
        if time.time() - start > budget_s - 1:
            break
        if n % 2 != 0:
            continue
            
        # Allocate iterations based on n
        time_left = budget_s - (time.time() - start)
        max_iters = max(1000, int(50000 / (n/100)))
        
        # Initialize binary h
        h = np.zeros(n, dtype=float)
        ones = n // 2
        h[np.random.choice(n, ones, replace=False)] = 1.0
        
        # Compute initial correlation
        corr = np.correlate(h, 1 - h, mode='full')
        cur_max = corr.max()
        cur_c5 = cur_max * (2.0 / n)
        
        # Store best so far for this n
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
            best_n = n
        
        # Simulated annealing with increment updates
        T_start = 0.05
        T = T_start
        T_decay = 0.9995
        
        for it in range(max_iters):
            now = time.time()
            if now - start > budget_s - 0.5:
                break
                
            # Pick two distinct indices with different values
            i = np.random.randint(n)
            j = np.random.randint(n)
            while h[i] == h[j]:
                i = np.random.randint(n)
                j = np.random.randint(n)
            
            # Propose swap and compute new correlation
            new_corr = propose_swap_and_update(h, corr, i, j)
            new_max = new_corr.max()
            new_c5 = new_max * (2.0 / n)
            
            # Accept or reject
            if new_c5 < cur_c5 or np.random.rand() < np.exp(-(new_c5 - cur_c5) / (T + 1e-12)):
                corr = new_corr
                cur_max = new_max
                cur_c5 = new_c5
                h[i], h[j] = h[j], h[i]
                
                # Update global best
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    best_n = n
            else:
                # Revert the swap (but h already reverted above if accepted)
                pass
                
            T *= T_decay
            
    if best_h is None:
        # Fallback with simple pattern
        n = 120
        best_h = np.zeros(n, dtype=float)
        ones = n // 2
        best_h[:ones] = 1.0
        best_c5 = compute_c5(best_h)
        best_n = n
    
    return best_h, best_c5, best_n
