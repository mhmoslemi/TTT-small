def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for given h values"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    correlations = np.correlate(h, 1 - np.array(h), mode="full")
    # The maximum is the largest correlation
    max_corr = np.max(correlations)
    return max_corr * dx
