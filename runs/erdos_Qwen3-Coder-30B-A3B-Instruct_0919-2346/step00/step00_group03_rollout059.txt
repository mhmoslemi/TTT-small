import numpy as np
import time
from scipy.optimize import differential_evolution

def evaluate_h(h, dx):
    """Calculate C₅ for a given h using FFT for efficiency."""
    h = np.asarray(h)
    # Compute convolution using FFT
    F = np.fft.fft(h)
    G = np.fft.fft(1 - h)
    corr = np.real(np.fft.ifft(F * np.conj(G)))
    # Take the maximum overlap over all shifts
    max_overlap = np.max(corr)
    return max_overlap * dx

def project_to_simplex(h, target_sum):
    """Project h onto the simplex { x : sum(x) = target_sum, 0 <= x <= 1 } using Dykstra's method."""
    h = np.asarray(h).copy()
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = (current_sum - target_sum) / len(h)
    h -= diff
    h = np.clip(h, 0, 1)
    # Iteratively refine projection
    for _ in range(5):
        current_sum = np.sum(h)
        diff = (current_sum - target_sum) / len(h)
        h -= diff
        h = np.clip(h, 0, 1)
    return h

def mutual_swap(h, i, j, const_sum):
    """Swap h[i] and h[j] and project to sum constraint."""
    h[i], h[j] = h[j], h[i]
    h = project_to_simplex(h, const_sum)
    return h

def mismatch_swap(h, dx):
    """Perform deterministic mismatch swaps to reduce the worst shift."""
    h = np.asarray(h).copy()
    const_sum = len(h) / 2
    max_corr = evaluate_h(h, dx)
    while True:
        F = np.fft.fft(h)
        G = np.fft.fft(1 - h)
        corr = np.real(np.fft.ifft(F * np.conj(G)))
        max_shift_idx = np.argmax(corr)
        # Get the relevant indices
        shift = max_shift_idx - (len(h) - 1)
        shift = int(shift)
        if shift == 0:
            break
        start_idx = max(0, -shift)
        end_idx = min(len(h), len(h) - shift)
        if abs(shift) >= len(h):
            break
        # Define mismatch sets
        M1 = [i for i in range(start_idx, end_idx) if h[i] >= 0.99 and h[i+shift] <= 0.01]
        M0 = [i for i in range(start_idx, end_idx) if h[i] <= 0.01 and h[i+shift] >= 0.99]
        if not M1 or not M0:
            break
        i = M1[0]
        j = M0[0]
        h = mutual_swap(h, i, j, const_sum)
        new_corr = evaluate_h(h, dx)
        if new_corr >= max_corr:
            # Undo swap
            h = mutual_swap(h, i, j, const_sum)
            break
        max_corr = new_corr
    return h

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    start_time = time.time()
    n = 180  # Machine precision will be improved after this. 
    dx = 2.0 / n
    # Initial h from interpolation if not available
    h0 = initial_h_values if initial_h_values is not None else np.ones(n) * 0.5
    if len(h0) != n:
        h0 = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, len(h0)), h0)
    h0 = project_to_simplex(h0, n / 2)
    
    # DE Parameters
    popsize = 30
    maxiter = int(10000 * (budget_s / 600))  # Scale with budget
    G_restart = max(100, maxiter // 10)
    CR = 0.9
    F = 0.8

    def de_objective(h):
        h = project_to_simplex(h, n / 2)
        return evaluate_h(h, dx)

    def de_mutation(a, b, c):
        return a + F * (b - c)

    # Set bounds
    bounds = [(0, 1) for _ in range(n)]

    # Start population from h0 + noise and random
    pop = []
    pop.append(h0.copy())
    for _ in range(popsize - 1):
        x = np.random.uniform(0, 1, n)
        x = project_to_simplex(x, n / 2)
        pop.append(x)
    
    current_best = float('inf')
    best_individual = h0.copy()

    # Generation loop
    for gen in range(maxiter):
        if time.time() - start_time > budget_s - 30:
            break
        new_pop = []
        for i in range(popsize):
            # Select three distinct individuals
            a, b, c = random.sample(pop, 3)
            v = de_mutation(a, b, c)
            # Crossover
            u = []
            for j in range(n):
                if random.random() < CR:
                    u.append(v[j])
                else:
                    u.append(pop[i][j])
            u = project_to_simplex(u, n / 2)
            # Evaluate
            u_cost = de_objective(u)
            pop_i_cost = de_objective(pop[i])
            if u_cost <= pop_i_cost:
                new_pop.append(u)
            else:
                new_pop.append(pop[i])
        
        pop = new_pop
        current_best = min(de_objective(x) for x in pop)
        best_individual = min(pop, key=de_objective)
        # Check for restart
        if gen % G_restart == 0 and gen > 0:
            pop.sort(key=de_objective)
            if len(pop) > 10:  # partial replacement strategy
                top = pop[:5]  # keep top 5
                rest = np.random.uniform(0, 1, (len(pop) - 5, n))
                for i in range(len(rest)):
                    rest[i] = project_to_simplex(rest[i], n / 2)
                pop = top + [x for x in rest]

    # After DE, do mismatch swap refinement
    best_refined = best_individual.copy()
    ref_start = time.time()
    while time.time() - ref_start < 120 and time.time() - start_time < budget_s - 30:
        best_before = evaluate_h(best_refined, dx)
        best_refined = mismatch_swap(best_refined, dx)
        after = evaluate_h(best_refined, dx)
        if abs(after - best_before) < 1e-8:
            break  # No more improvement

    final_cost = evaluate_h(best_refined, dx)
    return (best_refined.tolist(), final_cost, n)
