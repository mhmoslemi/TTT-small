import numpy as np
import time
import random

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return corr.max()

def propose_swap_and_update(h, corr, i, j):
    # h is numpy array of floats (0 or 1) before swap
    old_i = h[i]
    old_j = h[j]
    delta_i = old_j - old_i   # new_i - old_i
    
    # alias for length
    n = len(h)
    
    # Make a copy of current correlation to update
    new_corr = corr.copy()
    
    # loop over k not i,j
    for k in range(n):
        if k == i or k == j:
            continue
        hk = h[k]
        # contributions for i -> k: shift = k - i
        idx1 = k - i + n - 1
        new_corr[idx1] += delta_i * (1 - hk)
        
        # contributions for k -> i: shift = i - k
        idx2 = i - k + n - 1
        new_corr[idx2] += hk * (-delta_i)  # = -hk*delta_i

        # contributions for j -> k: shift = k - j
        idx3 = k - j + n - 1
        new_corr[idx3] += -delta_i * (1 - hk)  # delta_j = -delta_i

        # contributions for k -> j: shift = j - k
        idx4 = j - k + n - 1
        new_corr[idx4] += hk * (delta_i)   # = hk*delta_i

    # handle cross pairs
    idx_ij = j - i + n - 1
    new_corr[idx_ij] += delta_i

    idx_ji = i - j + n - 1
    new_corr[idx_ji] -= delta_i

    # return updated correlation and new values of h
    return new_corr, h[i], h[j]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()
    best_c5 = np.inf
    best_h = None
    best_n = None

    # Try a few different values of n
    n_list = [60, 80, 100, 120, 140, 160, 180, 200]
    # Make them even
    n_list = [n for n in n_list if n % 2 == 0]
    
    for n in n_list:
        if time.time() - start > budget_s - 1:
            break
        # Determine how many iterations to run for this n based on size
        max_iters = max(1000, min(20000, int(60000 / (n / 120))))
        
        # Initialize h: binary with exactly n//2 ones
        h = np.zeros(n, dtype=float)
        indices = np.random.choice(n, size=n//2, replace=False)
        h[indices] = 1.0

        # Compute initial correlation
        corr = np.correlate(h, 1 - h, mode='full')
        cur_c5 = corr.max() * (2.0 / n)
        cur_h = h.copy()
        cur_corr = corr.copy()
        # temperature for simulated annealing
        T = 0.05
        T_decay = 0.9995
        
        # Track iterations without improvement for restart strategy
        no_improvement_count = 0
        max_no_improvement = max(500, int(0.05 * max_iters))

        for iteration in range(max_iters):
            now = time.time()
            if now - start > budget_s - 0.5:
                break
                
            # Accept if no improvement for too long, restart
            # But we do a full restart with new random binary for simplicity
            if no_improvement_count > max_no_improvement:
                # Reinitialize as simple binary with n/2 ones
                h = np.zeros(n, dtype=float)
                indices = np.random.choice(n, size=n//2, replace=False)
                h[indices] = 1.0
                corr = np.correlate(h, 1 - h, mode='full')
                cur_c5 = corr.max() * (2.0 / n)
                cur_h = h.copy()
                cur_corr = corr.copy()
                T = 0.05
                no_improvement_count = 0

            # Pick two random indices with different values
            i = np.random.randint(n)
            j = np.random.randint(n)
            while h[i] == h[j]:
                i = np.random.randint(n)
                j = np.random.randint(n)

            # Compute the change in correlation after a swap
            new_corr, old_i, old_j = propose_swap_and_update(cur_h, cur_corr, i, j)
            new_c5 = new_corr.max() * (2.0 / n)

            # Accept or reject swap with simulated annealing
            d = new_c5 - cur_c5
            if d < 0 or np.random.rand() < np.exp(-d / (T + 1e-12)):
                cur_corr = new_corr.copy()
                cur_c5 = new_c5
                # Apply the swap in h
                cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
                no_improvement_count = 0
            else:
                no_improvement_count += 1

            # Update best solution
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()
                best_n = n
            
            # Temperature cooling
            T *= T_decay

    # Return best found
    return best_h, best_c5, best_n
