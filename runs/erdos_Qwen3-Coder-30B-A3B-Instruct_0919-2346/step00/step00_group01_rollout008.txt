```python
import time
import numpy as np
from scipy import fft
import cvxpy as cp
from cvxpy import Minimize, Problem
from cvxpy import Variable, Bool, sum as cp_sum
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Time tracking
    start_time = time.time()
    remaining_budget = lambda: budget_s - (time.time() - start_time)
    
    # Problem parameters
    n = 60
    dx = 2.0 / n
    n_half = n // 2
    
    # Initial shift set
    K0 = [1, n_half, 2]
    
    def get_shifted_indices(shift):
        """Returns indices (i, i+shift) for all i."""
        return np.arange(n), (np.arange(n) + shift) % n
    
    # Precompute index mapping for FFT-based evaluation
    shift_indices_table = {}
    for k in range(1, n):
        i_indices, k_indices = get_shifted_indices(k)
        shift_indices_table[k] = (i_indices, k_indices)
    
    def evaluate_autocorrelation(h, k):
        """Evaluate autocorrelation at shift k."""
        i_indices, k_indices = shift_indices_table[k]
        return np.sum(h[i_indices] * h[k_indices])
    
    def evaluate_all_autocorrelations(h):
        """Return all autocorrelations using FFT."""
        h_padded = np.concatenate([h, np.zeros(n)])
        H = fft.fft(h_padded)
        autocorr = fft.ifft(H * np.conj(H)).real[:n]
        return autocorr
    
    def compute_C5(h):
        """Compute C5 based on given h."""
        all_autocorr = evaluate_all_autocorrelations(h)
        min_autocorr = np.min(all_autocorr)
        return 1 - (2 / n) * min_autocorr
    
    # Initialize best solution
    best_h = None
    best_C5 = float('inf')
    best_iter = -1
    
    # Start with a random binary solution of size n with exactly n/2 ones
    def random_binary_h():
        h = np.zeros(n)
        ones_indices = np.random.choice(n, n//2, replace=False)
        h[ones_indices] = 1
        return h
    
    # Utility to get deterministic seed-based starting point
    h_start = random_binary_h()
    initial_C5 = compute_C5(h_start)
    if initial_C5 < best_C5:
        best_C5 = initial_C5
        best_h = h_start.copy()
        best_iter = 0
    
    # Initial binary MILP setup
    # Variables
    h = Variable(n, boolean=True)
    t = Variable(integer=True)
    
    # Auxiliary variables for binary autocorrelations
    w = {}
    a = {}
    
    # Add initial constraints for K0
    constraints = []
    for k in K0:
        i_indices, k_indices = get_shifted_indices(k)
        w[k] = Variable((n,), boolean=True)
        a[k] = cp_sum(w[k])
        
        # McCormick envelope for binary multiplication w[i,k] = h[i] * h[i+k]
        for i in range(n):
            constraints.append(w[k][i] <= h[i])
            constraints.append(w[k][i] <= h[k_indices[i]])
            constraints.append(w[k][i] >= h[i] + h[k_indices[i]] - 1)
        
        constraints.append(t <= a[k])
    
    # Constraints
    constraints.append(cp_sum(h) == n//2)
    constraints.append(h >= 0)
    constraints.append(h <= 1)
    
    # Optimization problem
    prob = Problem(Minimize(-t), constraints)
    
    # Solve initial problem
    cutoff_time = remaining_budget()
    if cutoff_time > 10:
        prob.solve(solver=cp.CBC, maxSeconds=min(cutoff_time - 5, 30))
    
    # Check if feasible solution was produced
    if prob.status not in ["optimal", "optimal_inaccurate"]:
        h_new = h_start
        C5_new = initial_C5
    else:
        # Retrieve solution
        h_vals = np.array([int(round(h[i].value)) for i in range(n)])
        C5_new = compute_C5(h_vals)
        h_new = h_vals
    
    # Track best binary solution so far
    if C5_new < best_C5:
        best_C5 = C5_new
        best_h = h_new.copy()
        best_iter = 1
    
    # Cut generation loop
    current_K = set(K0)
    current_iterations = 0
    
    while True:
        if remaining_budget() < 20:
            break
        
        all_autocorr = evaluate_all_autocorrelations(h_new)
        min_autocorr_idx = np.argmin(all_autocorr)
        min_autocorr_val = all_autocorr[min_autocorr_idx]
        
        if min_autocorr_val >= int(t.value):
            break  # Already optimal for current K
        
        # Add next most violating shift
        next_k = int(min_autocorr_idx)
        if next_k not in current_K:
            current_K.add(next_k)
            current_iterations += 1
            
            # Add new constraints
            i_indices, k_indices = get_shifted_indices(next_k)
            w[next_k] = Variable((n,), boolean=True)
            a[next_k] = cp_sum(w[next_k])
            
            for i in range(n):
                constraints.append(w[next_k][i] <= h[i])
                constraints.append(w[next_k][i] <= h[k_indices[i]])
                constraints.append(w[next_k][i] >= h[i] + h[k_indices[i]] - 1)
            
            constraints.append(t <= a[next_k])
            
            # Re-solve
            prob = Problem(Minimize(-t), constraints)
            cutoff_time = remaining_budget()
            if cutoff_time > 10:
                prob.solve(solver=cp.CBC, maxSeconds=min(cutoff_time - 5, 30))
            
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_vals = np.array([int(round(h[i].value)) for i in range(n)])
                C5_new = compute_C5(h_vals)
                h_new = h_vals
                
                if C5_new < best_C5:
                    best_C5 = C5_new
                    best_h = h_new.copy()
                    best_iter = current_iterations + 2
            else:
                break  # Solver failed
        else:
            # Avoid redundancy - exit if no new constraint added
            break
    
    # Post-processing: gradient polishing on binary solution
    # This is optional and aimed at fractional smoothing
    h_best = best_h.copy().astype(float)
    h_best = np.clip(h_best, 0, 1)
    current_sum = np.sum(h_best)
    adj = (n//2 - current_sum) / 2
    
    # Perform projecting gradient steps
    gradient_iters = min(300, max(50, int(budget_s / 50 - 50))) if remaining_budget() > 100 else 0
    for _ in range(gradient_iters):
        if remaining_budget() < 5:
            break
        
        all_autocorr = evaluate_all_autocorrelations(h_best)
        min_k = np.argmin(all_autocorr)
        
        # Compute gradient
        i_indices, k_indices = get_shifted_indices(min_k)
        grad = h_best[i_indices] + h_best[k_indices]
        
        # Gradient step
        step_size = 0.01
        h_best -= step_size * grad
        
        # Project onto feasible set
        h_best = np.clip(h_best, 0, 1)
        current_sum = np.sum(h_best)
        diff = current_sum - n/2
        if abs(diff) > 1e-6:
            # Adjust by pairwise swapping to maintain sum exactly n/2
            adjust_amount = diff / 2.0
            # Find indices to swap
            idx1 = np.argmax(h_best)
            idx2 = np.argmin(h_best)
            
            if h_best[idx1] > adjust_amount and h_best[idx2] + adjust_amount <= 1:
                h_best[idx1] -= adjust_amount
                h_best[idx2] += adjust_amount
            elif h_best[idx2] < 1 - adjust_amount and h_best[idx1] + adjust_amount <= 1:
                h_best[idx2] += adjust_amount
                h_best[idx1] -= adjust_amount

        # Re-evaluate C5
        new_C5 = compute_C5(h_best)
        if new_C5 < best_C5:
            best_C5 = new_C5
            best_h = h_best.copy()
    
    # Final evaluation of estimated best choice
    final_C5 = compute_C5(best_h)
    
    return (best_h, final_C5, n)
```
