import numpy as np
import time
import random

def evaluate_c5(h):
    # Compute the overlap integral using numpy correlate
    # h is a step function represented as a 1D numpy array
    # We want to compute max_k ∫ h(x)(1 - h(x + k)) dx
    # This is equivalent to computing full correlation of h with (1-h) 
    # and taking the maximum value, multiplied by dx = 2 / len(h)
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(conv)
    return max_overlap * dx

def random_binary_seq(n):
    # Generate a random binary sequence with exactly n/2 ones
    # and n/2 zeros
    h = np.zeros(n)
    ones_indices = np.random.choice(n, n//2, replace=False)
    h[ones_indices] = 1
    return h.astype(float)

def solve_binary_hill_climber(n, max_time_s, seed=None):
    rng = np.random.default_rng(seed)
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Initial solution from provided initial_h_values if available 
    # (we'll draw from it or interpolate)
    global initial_h_values
    try:
        init_h = np.array(initial_h_values)
        init_h = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, len(init_h)), init_h)
        # Adjust to have exactly n//2 ones
        current_h = init_h.copy()
        current_h = (current_h > 0.5).astype(float)
        # Ensure exactly n/2 ones
        current_sum = np.sum(current_h)
        if current_sum != n / 2:
            # Adjust by flipping values in places that don't cause issues
            diff = int(current_sum - n / 2)
            if diff > 0:
                # Too many ones, flip some to zeros
                idxs = np.where(current_h == 1)[0]
                flip_indices = rng.choice(idxs, diff, replace=False)
                current_h[flip_indices] = 0
            else:
                # Too few ones, flip some to ones
                idxs = np.where(current_h == 0)[0]
                flip_indices = rng.choice(idxs, -diff, replace=False)
                current_h[flip_indices] = 1
        current_h = current_h.astype(float)
    except:
        # Generate completely random binary
        current_h = random_binary_seq(n)
    
    current_c5 = evaluate_c5(current_h)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = current_h.copy()

    # Hill-climbing search for a decent solution 
    MAX_ITER = 50000
    T = 0.1  # Initial temperature 
    cooling = 0.99995
    step = 0.01
    
    iteration = 0
    while time.time() - start_time < max_time_s - 5 and iteration < MAX_ITER:
        # Select two indices at random
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        
        # If they're already the same value, skip
        if current_h[i] == current_h[j]:
            iteration += 1
            T *= cooling
            continue
            
        # Create a proposed solution by swapping
        new_h = current_h.copy()
        new_h[i], new_h[j] = new_h[j], new_h[i]
        
        new_c5 = evaluate_c5(new_h)
        
        # Accept the move with a probability based on simulated annealing
        delta = new_c5 - current_c5
        if delta < 0 or rng.random() < np.exp(-delta / T):
            current_h = new_h
            current_c5 = new_c5
            # Update the best if we found a better solution
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
        
        iteration += 1
        T *= cooling
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    # We'll use n=200 for better resolution to target C5 <= 0.3807
    n_points = 200
    # Seed for reproducibility
    np.random.seed(seed)
    random.seed(seed)
    
    start_time = time.time()
    
    # Run multiple independent hill climber instances
    best_h, best_c5 = solve_binary_hill_climber(n_points, 0.8 * budget_s, seed)
    
    # Fine-tune with continuous optimization around the best binary solution 
    # No need for true continuous optimization---we'll just keep the best binary solution found
    
    # Return best solution found
    h_values = best_h.tolist()
    c5_bound = best_c5
    
    return (h_values, c5_bound, n_points)
