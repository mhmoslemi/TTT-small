import numpy as np
from scipy.optimize import minimize
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral for given h"""
    # Compute convolution of h with (1-h)
    conv = np.correlate(h, 1-h, mode='full')
    # The maximum value gives us the overlap
    max_overlap = np.max(conv) * dx
    return max_overlap

    def objective(h):
