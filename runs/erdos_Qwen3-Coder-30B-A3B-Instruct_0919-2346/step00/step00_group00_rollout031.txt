import numpy as np
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return corr.max()

def propose_swap_and_update(h, corr, i, j, new_corr):
    # h is numpy array of floats (0 or 1) before swap.
    a = h[i]
    b = h[j]  # b = 1 - a (but not enforced, we ensure i,j have different values)
    delta_i = b - a   # either +1 or -1
    
    # alias for length
    n = len(h)
    
    # loop over k not i,j
    for k in range(n):
        if k == i or k == j:
            continue
        hk = h[k]
        # contributions from i->k: l = k - i
        idx1 = k - i + n - 1
        new_corr[idx1] += delta_i * (1 - hk)
        # contributions from k->i: l = i - k
        idx2 = i - k + n - 1
        new_corr[idx2] += hk * (-delta_i)

        # contributions from j->k: l = k - j
        idx3 = k - j + n - 1
        new_corr[idx3] += -delta_i * (1 - hk)  # delta_j = -delta_i
        # contributions from k->j: l = j - k
        idx4 = j - k + n - 1
        new_corr[idx4] += hk * (delta_i)   # = hk*delta_i

    # handle cross pair i->j and j->i
    idx_ij = j - i + n - 1
    new_corr[idx_ij] += delta_i

    idx_ji = i - j + n - 1
    new_corr[idx_ji] -= delta_i

    # swap the values in h
    h[i], h[j] = b, a

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start = time.time()
    best_c5 = np.inf
    best_h = None
    best_n = None

    # candidate n's
    n_candidates = [60, 80, 100, 120, 140, 160, 180, 200]
    
    # Use larger n for better resolution
    for n in [200]:
        if n % 2 != 0:
            continue
            
        # Allocate specific time per n
        time_left = budget_s - (time.time() - start)
        if time_left <= 1:
            break
            
        # Setup
        max_iters = max(2000, int(20000 / (n/120)))
        h = np.zeros(n, dtype=float)
        ones = n // 2
        h[np.random.choice(n, ones, replace=False)] = 1.0
        corr = np.correlate(h, 1 - h, mode='full')
        cur_c5 = corr.max() * (2.0 / n)
        cur_max = corr.max()
        
        # Temperature for simulated annealing
        T = 0.05
        T_decay = 0.9999
        iterations_without_improvement = 0
        max_no_improvement = 5000
        
        # Local search
        for it in range(max_iters):
            if time.time() - start > budget_s - 1:
                break
                
            # propose swap
            i = np.random.randint(n)
            j = np.random.randint(n)
            while h[i] == h[j]:
                i = np.random.randint(n)
                j = np.random.randint(n)
                
            # compute delta
            new_corr = corr.copy()
            propose_swap_and_update(h, corr, i, j, new_corr)
            new_max = new_corr.max()
            new_c5 = new_max * (2.0 / n)
            
            d = new_c5 - cur_c5
            
            if d < 0 or np.random.rand() < np.exp(-d / (T + 1e-12)):
                # accept
                corr = new_corr
                cur_c5 = new_c5
                cur_max = new_max
                iterations_without_improvement = 0
            else:
                # revert h values
                h[i], h[j] = h[j], h[i]
                iterations_without_improvement += 1
                
            # update best
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                best_n = n
                
            # cooling
            T *= T_decay
            
            # restart if stuck
            if iterations_without_improvement > max_no_improvement:
                # Random restart
                h = np.zeros(n, dtype=float)
                h[np.random.choice(n, ones, replace=False)] = 1.0
                corr = np.correlate(h, 1 - h, mode='full')
                cur_c5 = corr.max() * (2.0 / n)
                cur_max = corr.max()
                iterations_without_improvement = 0
                T = 0.05  # Reset temperature
                
    if best_h is None:
        try:
            from __main__ import initial_h_values
            best_h = initial_h_values.copy()
            best_n = len(best_h)
            best_c5 = compute_c5(best_h)
        except Exception:
            # fallback
            n = 2
            best_h = np.array([1.0, 0.0])
            best_n = 2
            best_c5 = compute_c5(best_h)
            
    return best_h, best_c5, best_n
