```python
import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import time
import random

def evaluate_erdos_solution(h):
    """Returns the C5 bound for a given h, using the correlation-based computation"""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    C5 = np.max(c) * dx
    return C5

# Pre-setup initial values
initial_h_values = np.array([0.0]*35 + [1.0]*35)  # n=70, half 0s and half 1s

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)

    n = len(initial_h_values)
    dx = 2.0 / n
    target_sum = n / 2

    best_h = initial_h_values.copy()
    best_C5 = evaluate_erdos_solution(best_h)

    start_time = time.time()
    total_time = 0

    # Helper function to project h onto the feasible simplex
    def project_simplex(h, target_sum):
        h = np.array(h)
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if abs(current_sum - target_sum) < 1e-6:
            return h
        elif current_sum > target_sum:
            # Need to decrease values
            sorted_indices = np.argsort(h)[::-1]
            cumsum = 0
            for i in range(len(h)):
                cumsum += h[sorted_indices[i]]
                if cumsum > target_sum:
                    threshold = (cumsum - target_sum) / (i + 1)
                    h = np.clip(h - threshold, 0, 1)
                    break
        else:
            # Need to increase values
            sorted_indices = np.argsort(h)
            cumsum = 0
            for i in range(len(h)):
                cumsum += h[sorted_indices[i]]
                if cumsum < target_sum:
                    threshold = (target_sum - cumsum) / (len(h) - i)
                    h = np.clip(h + threshold, 0, 1)
                    break
        return h

    # Helper function to check and enforce feasibility
    def ensure_feasible(h):
        h = project_simplex(h, target_sum)
        return h

    # Function to solve MILP using CVXPY
    def solve_milp(h_start):
        h = cp.Variable(n, name='h')
        y = [cp.Variable(n-k, name=f'y_{k}') for k in range(1, n)]  # y_{i,k} for k=1..n-1
        tau = cp.Variable(name='tau')

        # Objective
        objective = cp.Maximize(tau)

        # Constraints
        constraints = [
            cp.sum(h) == target_sum,
            0 <= h, h <= 1,
            h[0] == 1,  # break symmetry
        ]

        # Add log sum-exp to model a single constraint per shift k
        for k in range(1, n):
            # Loop over valid i for this k
            for i in range(n - k):
                y[k-1][i] >= h[i] + h[i+k] - 1
                y[k-1][i] <= h[i]
                y[k-1][i] <= h[i+k]
                y[k-1][i] >= 0
            # tau <= sum(y[k])
            constraints.append(tau <= cp.sum(y[k-1]))

        # Solve
        prob = cp.Problem(objective, constraints)
        try:
            # Use CBC for integer programming
            prob.solve(solver=cp.CBC, verbose=False, max_seconds=0.4 * budget_s)
            h_sol = h.value if h.value is not None else h_start
            tau_sol = tau.value
            return h_sol, tau_sol
        except Exception as e:
            print(f"Error in solving MILP: {e}")
            return h_start, 0.0

    # Wrapper for solving with fairness and max time
    def attempt_milp_with_timeout(h_start, timeout_s):
        start_t = time.time()
        try:
            h_sol, tau_sol = solve_milp(h_start)
            elapsed = time.time() - start_t
            return h_sol, tau_sol, elapsed
        except Exception as e:
            return h_start, 0.0, time.time() - start_t

    attempts = []
    restarts = 10
    for i in range(restarts):
        if total_time > 0.9 * budget_s:
            break
        # Initialize a fresh random seed for each restart
        run_seed = seed + i * 1000
        random.seed(run_seed)
        np.random.seed(run_seed)

        # Generate a binary seed with exactly target_sum ones
        h_seed = np.zeros(n)
        indices = np.random.choice(n, int(target_sum), replace=False)
        h_seed[indices] = 1.0
        h_seed = ensure_feasible(h_seed)

        # Try with random binary start
        try:
            h_sol, tau_sol, elapsed = attempt_milp_with_timeout(h_seed, budget_s * 0.4)
            total_time += elapsed
            if h_sol is not None and tau_sol is not None:
                C5 = (target_sum - tau_sol) * dx
                if C5 < best_C5:
                    best_C5 = C5
                    best_h = h_sol.copy()
        except Exception as e:
            pass

    # Rounding strategy to nearest binary
    h_rounded = np.zeros(n)
    indices = np.argsort(best_h)[::-1][:int(target_sum)]
    h_rounded[indices] = 1.0

    # Final check of rounded
    c5_rounded = evaluate_erdos_solution(h_rounded)
    if c5_rounded < best_C5:
        best_C5 = c5_rounded
        best_h = h_rounded

    # Post process with local search if space allows
    if not (total_time > 0.95 * budget_s):
        try:
            # We do local improvement on the binary version
            h_best_bin = (best_h > 0.5).astype(float)
            h_best_bin = ensure_feasible(h_best_bin)
            c5_final = evaluate_erdos_solution(h_best_bin)
            if c5_final < best_C5:
                best_C5 = c5_final
                best_h = h_best_bin
        except Exception as e:
            pass

    return best_h, best_C5, n
```
