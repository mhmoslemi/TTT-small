import numpy as np
import time
from scipy import signal

def project_to_capped_simplex(h, target_sum):
    """Project h to the capped simplex: 0 <= h[i] <= 1 and sum(h) = target_sum"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Water-filling approach
    sorted_h = np.sort(h)[::-1]
    cumsum = np.cumsum(sorted_h)
    n = len(h)
    # Find threshold index
    i = 0
    while i < n and sorted_h[i] - (target_sum - cumsum[i]) / (i + 1) >= 0:
        i += 1
    lambda_val = (target_sum - cumsum[i-1]) / i if i > 0 else target_sum / n
    # Apply threshold
    h_proj = np.maximum(0, np.minimum(1, h - lambda_val))
    return h_proj

def compute_gradient(h, alpha, dx):
    """Compute gradient of F_alpha(h) using convolutions."""
    n = len(h)
    # Compute O_ell(h) for all shifts ell
    O_ell = np.zeros(2 * n - 1)  # full correlation length
    for ell in range(1 - n, n):
        # This is O_ell(h) = dx * sum h[i] * (1 - h[i + ell])
        if ell >= 0:
            h_shift = h[ell:]
            h_i = h[:len(h_shift)]
        else:
            h_shift = h[:ell]
            h_i = h[-ell:]
        if len(h_i) > 0:
            O_ell[ell + n - 1] = dx * np.sum(h_i * (1 - h_shift))
    
    # Softmax weights w_ell
    O_max = np.max(O_ell)
    exp_O = np.exp(alpha * (O_ell - O_max))
    w = exp_O / np.sum(exp_O)
    
    # Compute gradient components
    # Using convolution
    # w * (1 - h) from left to right
    c1 = signal.convolve(w, 1 - h, mode='full')
    # w_rev * h from right to left, equivalent to w * h shifted
    w_rev = w[::-1]
    c2 = signal.convolve(w_rev, h, mode='full')
    # Taking the part of the same length as h
    c1 = c1[n - 1 : n - 1 + n]
    c2 = c2[n - 1 : n - 1 + n]
    
    grad = c1 - c2
    return grad

def solve_erdos_problem(budget_s=600):
    # Constants
    n = 200   # number of samples (even number)
    dx = 2.0 / n
    target_sum = n / 2.0
    max_alpha = 200.0
    alpha_init = 0.5
    gamma = 1.3
    eta = 0.05
    eta_decay = 0.99
    steps_per_omega = 200
    n_seeds = 7
    time_budget = budget_s

    # Initialize best solution
    best_c5 = float('inf')
    best_h = None

    start_time = time.time()
    
    # Seed 0: use initial_h_values if it exists
    h_list = [initial_h_values if 'initial_h_values' in globals() else None]
    # Seeds 1 to n_seeds-1: random fractional
    for i in range(1, n_seeds):
        h_rand = np.random.rand(n)
        h_rand = project_to_capped_simplex(h_rand, target_sum)
        h_list.append(h_rand)

    for seed_idx, h_init in enumerate(h_list):
        if time.time() - start_time > time_budget - 5:
            break
        
        # Reset for each seed
        alpha = alpha_init
        h = h_init.copy() if h_init is not None else np.random.rand(n)
        h = project_to_capped_simplex(h, target_sum)
        eta_curr = eta

        # Determine number of iterations a seed needs
        remaining_time = time_budget - (time.time() - start_time)
        if seed_idx == 0:
            # More iterations for the first seed since it's used as base
            max_iter = int(0.7 * remaining_time / 10) if remaining_time > 10 else 1
        else:
            max_iter = int(0.3 * remaining_time / 10) if remaining_time > 10 else 1
            
        # Annealing loop
        for iter in range(max_iter):
            if time.time() - start_time > time_budget - 5:
                break

            grad = compute_gradient(h, alpha, dx)
            h -= eta_curr * grad
            h = project_to_capped_simplex(h, target_sum)
            
            # Periodically adjust alpha
            if (iter + 1) % steps_per_omega == 0:
                alpha *= gamma
                eta_curr *= eta_decay
                
                # Evaluation with actual C5
                corr = np.correlate(h, 1 - h, mode="full")
                c5 = dx * np.max(corr)
                
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                    
                # Stop if alpha is too big
                if alpha > max_alpha:
                    break

    # If best_h is still None, create a reasonable one
    if best_h is None:
        h = np.random.rand(n)
        h = project_to_capped_simplex(h, target_sum)
        best_h = h
        corr = np.correlate(h, 1 - h, mode="full")
        best_c5 = dx * np.max(corr)

    # Convert to binary step function with deterministic rounding
    # Sort indices by h values descending
    idx_sorted = np.argsort(best_h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx_sorted[:n//2]] = 1.0

    # Final evaluation of the true C5 for the binary mask
    corr_final = np.correlate(h_bin, 1 - h_bin, mode="full")
    final_c5 = dx * np.max(corr_final)
    
    # Local refinement via hill-climbing
    # Ensure we have time for this
    initial_time = time.time()
    # Simple hill-climbing: try to swap elements
    iter_attempts = 0
    max_attempts = 10000
    while iter_attempts < max_attempts and (time.time() - initial_time) < (time_budget - 5):
        h_temp = h_bin.copy()
        i1 = np.random.choice(np.where(h_temp == 1)[0])
        i0 = np.random.choice(np.where(h_temp == 0)[0])
        h_temp[i1] = 0
        h_temp[i0] = 1

        corr_new = np.correlate(h_temp, 1 - h_temp, mode="full")
        c5_temp = dx * np.max(corr_new)

        if c5_temp < final_c5:
            h_bin = h_temp.copy()
            final_c5 = c5_temp
        iter_attempts += 1

    return (h_bin, final_c5, n)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    return solve_erdos_problem(budget_s)
