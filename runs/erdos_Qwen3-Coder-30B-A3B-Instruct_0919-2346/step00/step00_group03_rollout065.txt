import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_erdos_solution(h, n_points=70):
    """Evaluate the Erdős constant for given step function h"""
    dx = 2.0 / n_points
    # Compute cross-correlation: h(x) * (1 - h(x+k))
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take maximum over all lags
    max_corr = np.max(correlation)
    # Scale by dx to get the integral approximation
    c5 = max_corr * dx
    return c5
