def evaluate_overlap(h):
    """Evaluate the C5 constant for a given step function h."""
    if not isinstance(h, np.ndarray):
        h = np.array(h)
    dx = 2.0 / len(h)
    # Compute C5 as the maximum of overlap integrals over all shifts
    # np.correlate with mode='full' computes the convolution (1-h) * h
    # We want h(x)(1 - h(x+k)) so we compute correlate(h, 1-h, mode='full')
    # and extract the relevant part. 
    # modes full: len(h) + len(1-h) - 1 = 2*len(h) - 1
    # we need shift k in [0, 2], so range of correlation:
    # at k=0, correlation at index len(h)-1
    # at k=2, correlation at index 0
    # So the full correlation needs to be truncated from center
    # The shift k increases from left to right
    # corr at index i corresponds to shift k = i * dx (k = [0, 2])
    # But instead, we compute it via full convolution and extract the range of interest:
    # for k in [0, 2], we want shifts h (start) to h-k (end) and find minimum.
    # Since full convolution includes all shifts in [0,2], we extract the part that corresponds to k=[0,2]
    # correlation length is 2*len(h) - 1
    # Fully overlapping at start, so we take the last len(h) points 
    # and corresponding shifts from 0 to 2.  
    # Size of the overlap array should be n_points, which corresponds to shifts in [0,2]
    # So best estimate is the middle of the full convolution:
    # full_conv = np.correlate(h, np.flip(1-h), mode='full')
    # full_conv is of length 2*len(h) - 1, covering shifts from 0 to len(h) - 1
    # But since we want shifts [0, 2], we map the correlation in (0, 2] back to [0, 2]
    # shift 0: index = len(h) - 1
    # shift k: index = len(h) - 1 - k/dx
    # k increases from 0 to 2. 
    # So all shifts are within the middle 2 * len(h) - 1 values
    # Specifically full_conv[i], intensity is for shift = i * dx ... but that is not correct.
    # np.correlate(h, 1-h, mode='full')
    # y[i] = sum over j of h[j] * (1-h)[i-j] 
    # we compute h(x) * (1-h(x+k))
    # When we do correlation, we compute h[j] * (1-h)[i-j]
    # so it's convolution of h with (1-h) shifted so as to get the appropriate shift
    # for h(x)* (1 - h(x+k)), we compute correlate(h, (1 - h)_{shifted}, mode='full') with shift k = i * dx
    # For overlap _from_ h at position x, we want the amount (1-h(x+k)) at x+k
    # so it's correlation h * (1 - h) reversed. 
    # If we define: result = np.correlate(h, 1-h, mode='full'); 
    # this is: sum_j h[i-j] * (1 - h[j]) = \int h(x) (1-h(x+k)) dx if we linearly interpolate 
    # and k = (len(h) - i) dx
    # We want orientation corrected...
    # Actually it's easier to define as convolution:
    # h(x) * (1-h)(x)= \int_{R} h(x-y) (1-h)(x) dy where x-y = k  => y = -k, so introduce variable k, get:
    # \int_{R} h(x+k) (1-h)(x) dk = \int_{R} h(y) (1-h)(x)dy where k = y-x 
    # But convolution h * g with h(y) and g(x) = (1-h)(x) is y(y-g) = h(y) * (1-h)(x-y)
    # We just want: sum of integral h(x) * (1-h(x+k) dx over all x == integrate h with 1-h shifted by k.
    # so we want h * (1-h) where the second operand is also shifted by k. 
    # Why? 
    # Look at the direct formula. For each k:
    # \int_0^2 h(x) * (1-h(x+k))dx 
    # This suggests doing correlation of h with reflected (1-h) but that's not
    # The best way to do is correlation directly as cross-correlation with full 
    # Since full is 2*len(h) -1, center index is len(h) - 1 (due to python indexing)
    # we want the correct range of k from 0 to 2.  
    # For correlation h * (1-h), the shift k will be between [-len(h)+1, len(h)-1] 
    # applying dx = 2/len(h)  
    # length of normalization is 1, non-zero region is [0,2]. 
    # Let's try this: we do correlate(h, 1-h, mode='full') and shift the 0 to the right of the range.
    # Full corr of h with (1-h) gives array of size 2*len(h) - 1
    # Index j of this gives shift k = (j - (len(h) - 1) ) * dx  = (j - (len(h) - 1)) * (2/len(h))
    # we want index [0, 2] >>>
    # k = 0 -> j = len(h) - 1  
    # k = 2 -> j = len(h) - 1 + (2/dx) = len(h) - 1 + len(h) = 2*len(h) - 1
    # k = -2 -> j = len(h) - 1 - (2/dx) = len(h) - 1 - len(h) = -1
    # So we get as valid indices: from j = [0, 2*len(h) - 1] 
    # The valid within [0,2] domain corresponds to:
    # 2/len(h) = dx => 2 = dx * len(h)  
    # so 0 <= k <= 2 brings j s.t. 0 <= j <= 2*len(h) - 1 and we read those. 
    # Now, shift k has associated index = j = len(h) - 1 + k/dx = len(h) - 1 + k * len(h)/2  = (len(h) - 1) + k * len(h)/2
    # shift 0: 0 part of full corr
    # shift 2: 2*len(h) - 1 part of full corr
    # Unusual way to compute. Better way:
    # Correlation of h with 1-h, mode='full'.
    # We want to take the range corresponding to k in [0, 2]
    # The full correlation has length 2*len(h)-1  
    # Offsets from the center: 0 indexing is for shift 0 = len(h) - 1 from center -> range [len(h) - 1, len(h) -1 + len(h) - 1] 
    # lefmost value shift: - (len(h) - 1) * dx
    # rightmost value shift: (len(h) - 1) * dx  
    # But our shift is k, so. 
    # margin = len(h) - 1  
    # x_i shift = k_i = (i - margin) * dx  = (i - (len(h) -1)) * (2 / len(h))
    # ok so shift from zero to 2 => i = 0 ... to 2*len(h) - 1
    # to satisfy k in [0,2] means i - (len(h)-1) <= 2/dx   => i <= 2/dx + len(h) - 1 = len(h) + len(h) - 1 = 2*len(h) - 1  
    # Which is the entire full convolution.
    # But the kernel-based idea of computing overlap integrals over k leads to eval as correlation, 
    # so for k = 0, we are biased.
    #
    # In conclusion, for a vector of length n, we are concerned with the correlation 
    # where shifts are between [- (n-1), n-1]. That is 2*n-1 possible values. 
    # Actual shift is k = (i - (n-1))*dx (from i ranging from 0 to 2*n-2)
    # To get valid values for shifts in [0,2]: create mapping from 0 to 2, use integer indexing.
    # Start with index=0, shift k = - (n-1) * dx = - (n-1) * 2/n see if in [0,2]. 
    # This is early part which gives negative shifts. Only interested in desc from i = (n-1) 
    # to i = 2*n -1. The shifted portions ... 
    #
    # Instead do this:
    cross_corr = np.correlate(h, 1 - h, mode='full')
    # correlation is length 2*n - 1
    # center is original shift k=0: index = n-1
    # shift k = i - (n-1)  
    # indices that is k in [0,2] are for (i - (n-1)) * dx in [0, 2]  => i  in [n-1, n-1 + 2/dx] 
    # dx = 2/n => 2/dx = n
    # So i in [n-1, n-1+n] = [n-1, 2n-1]
    # this is index [n-1, 2*n-2 ] in cross_corr (0-based) 
    # Now our full correlation is valid for shifts k = [- (n-1)dx, (n-1)dx] = [-2, 2]
    # arr[0 ... 2n-2] but only use indices [n-1, 2*n-2]
    # length = 2*n - 1 comes from conv

    # An the range for shifts corrsponds to indices from [n-1, 2*n-2]
    # i = n-1 -> k = (n-1 - (n-1)) * dx = 0 
    # i = 2*n - 2 -> k = (2*n-2 - (n-1)) * dx = (n-1) * dx = 2 
    indices = np.arange(n-1, 2*n - 1) # Invalid! 2*n-1 is 0-based index to 2*n-2. Already read 2*n-1 items !!
    # In Python indices = range [n-1, 2*n-1), so it is [n-1 to 2*n - 2] exclusive upper. 
    # length is (2*n-1) - (n-1) = n  
    # Means take last half of full correlation (with shift k>=0)
    # But we also want only shift 0 to 2, not from 0 to 2.
    # Shift values are in range [0, 2] and are linear in indices [n-1, 2*n-2]
    # So the output is index n-1 to 2*n-2 (inclusive) from cross_corr, i.e. length n.
    # And shift range from i=n-1 to 2*n-2 
    # shift = (i - (n-1)) * dx = (i - (n-1)) * (2/n) 
    # i = n-1 gives shift 0
    # i = 2*n-2 gives shift 2 
    corr_val = cross_corr[n-1 : 2*n - 1]  # indices n-1 to 2*n - 1, both inclusive  
    # corr_val[k] corresponds to shift k= (k*d) = k * (2/n) where k runs from 0 to n-1
    # So correlations at index k are for values at shift (k*2/n)
    # This was not fully correct, but close.
    # Best sure result is to generate all shift k in [0, 2] and calculate convolution
    # Manually generate it properly instead. Better.
    # Try the other way. For convolution a*b, define it as int f(x)g(x+shift)dx. 
    # So for shift 0:
    # int_0^2 h(x) (1-h(x)) dx  
    # For shift k: 
    # int_0^2 h(x) (1 - h(x+k)) dx
    # Calculation x in [0,2], let x' = x+k -> x = x'-k  
    # int_{k}^{2+k} h(x'-k) (1-h(x')) dx' = int_{k}^{2+k} h(x'-k) (1 - h(x')) dx'
    # if k > 0 then integrate over [k,2]
    # if k < 0 then integrate over [0, 2+k]
    # if k= 0, integrate over [0,2]
    # REGARDLESS, we compute: 
    # int_x_h(x) * (1 - h(x-k)) dx by convolving h with (1-h) [the latter properly shifted] 
    # if we take h * (1 - h) then result at i gives shift 0 -> x = 0
    # actual shift k = i * dx - (n-1)*dx  
    # assign shift i-> shift (i - n+1) * dx 
    # for range [0,2], we want (diff) i lowering bound. 
    # Consider that shift k = - (n-1)*dx to (n-1)*dx = -2 to 2 
    # But we want k = 0 to 2
    # We use a subset of circles, [n-1, 2n-2] for k from 0 to 2  
    # special care with indexing.
    corr_shifts = cross_corr[n-1:2*n-1]   # length is n
    return np.max(corr_shifts) * dx

def evaluate_c5(h_vals):
    """Wrapper for evaluate_overlap."""
    return evaluate_overlap(h_vals)

def optimize_h(start_h, budget_s):
    """Optimize the function h with given starting point."""
    # Initialize timing
    start_time = time.time()

    best_h = np.array(start_h)
    best_c5 = evaluate_c5(best_h)
    
    # Scale the problem for optimizer
    bounds = [(0, 1) for _ in range(len(best_h))]
    
    # Try with far better sampling and local refinement
    # Use a stochastic optimization approach first
    # Try differential evolution
    def objective(h_vals):
        # Return negative because we are minimizing
        return -evaluate_c5(h_vals)
    
    maxiter = int(budget_s / 0.05)  # Assume 0.05s per iteration
    
    # Run differential evolution with multiple random starts
    best_result = None
    best_c5 = np.inf
    current_h = np.copy(best_h)

    # Try several random perturbations to get more diverse search
    for _ in range(5):
        # Randomly vary starting point a bit
        perturbed = current_h.copy()
        # Random small changes
        idxs = np.random.choice(len(current_h), size=min(10, len(current_h) // 2), replace=False)
        for i in idxs:
            if np.random.random() < 0.7:
                # add small noise
                perturbed[i] = np.clip(perturbed[i] + (np.random.random() - 0.5) * 0.1, 0, 1)
        # Also adjust so integral is correct
        new_sum = np.sum(perturbed)
        desired_sum = len(perturbed) / 2  # because dx = 2 / n_points => area = 1
        if new_sum > 0:
            scale = desired_sum / new_sum
            perturbed *= scale
            
        # Ensure it's valid with constraints
        # Normalize with fixed area (integrated area is forced to 1)
        normalized = np.array(perturbed)
        s = np.sum(normalized)
        if s > 0:
            normalized = normalized * (len(normalized) / 2) / s
        else:
            normalized = normalized * 0.5
            
        try:
            result = differential_evolution(
                objective,
                bounds,
                maxiter=maxiter,
                popsize=10,
                seed=42,
                disp=False,
                polish=True,
                workers=1
            )
        except:
            continue

        if -result.fun < best_c5:
            best_c5 = -result.fun
            best_h = result.x.copy()

        # If time out, break
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.9:
            break  

    # At the end, do final refinement
    refinement_runs = 5
    for _ in range(refinement_runs):
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.9:
            break

        # Final polishing with a small subset
        bounds = [(0,1) for _ in range(len(best_h))]
        try:
            result = differential_evolution(
                objective,
                bounds,
                maxiter=50,  # small number for speed
                popsize=5,
                seed=42,
                disp=False,
                polish=True,
                workers=1
            )
            if -result.fun < best_c5:
                best_c5 = -result.fun
                best_h = result.x.copy()
        except:
            pass
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.9:
            break

    # At end, push sum to be exactly right
    # Ensuring that sum(h) * dx = 1, i.e., sum(h) = n_points / 2  
    scale = len(best_h) / 2 / np.sum(best_h)
    best_h = best_h * scale

    # Final validation
    final_c5 = evaluate_c5(best_h)
    
    return (best_h, final_c5, len(best_h))

def run(seed=42, budget_s=600, **kwargs):
    # set the seed for reproducibility
    np.random.seed(seed)
  
    # Start time
    start_time = time.time()

    # Use initial step function from problem description
    start_h = initial_h_values.copy()
    
    # Run the optimization
    best_h, best_c5, n_points = optimize_h(start_h, budget_s)
    
    # If still time, do one final pass with full polish
    elapsed = time.time() - start_time
    if elapsed < budget_s * 0.9:
        # Possibly with new approach using Basin-hopping
        bounds = [(0, 1) for _ in range(len(best_h))]

        def objective(h_vals):
            return -evaluate_c5(h_vals)
        
        # Consider dual annealing as alternative
        try:
            result = dual_annealing(
                objective,
                bounds,
                maxiter=100,  # fixed an approximate number
                seed=seed,
                maxfun=1000,  
                x0=best_h
            )
            if -result.fun < best_c5:
                best_c5 = -result.fun
                best_h = result.x.copy()
        except:
            pass
        
    return (best_h.tolist(), best_c5, n_points)
