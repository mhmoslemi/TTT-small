import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h_values):
    """Evaluate the C5 bound for given h_values"""
    h = np.array(h_values)
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode='full')
    # The maximum value gives the C5 bound
    c5 = np.max(corr) * dx
    return c5

def initial_h_values():
    """Initial construction of h values"""
    n = 70
    h = np.zeros(n)
    # This is a simple construction from the initial solution
    # We'll use a piecewise constant function
    # Split [0, 2] into 7 parts of width 2/7 each
    segment_width = 2.0 / 7
    for i in range(7):
        # Each segment contributes 1/7 to the integral
        # Set h = 1 for first 3 segments, 0 for last 4 segments
        start_idx = int(i * (n / 7))
        end_idx = int((i + 1) * (n / 7))
        if i < 3:
            h[start_idx:end_idx] = 1.0
        else:
            h[start_idx:end_idx] = 0.0
    # Normalize so that sum(h) * dx = 1 => sum(h) = n/2
    scale_factor = (n / 2) / np.sum(h)
    h = h * scale_factor
    return h.tolist()

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    target_sum = n_points / 2  # This ensures ∫₀² h(x) dx = 1
    
    # Function to evaluate our objective for optimization
    def objective(h):
        # Apply constraints
        h = np.array(h)
        # Keep within [0,1]
        h = np.clip(h, 0, 1)
        # Enforce the integral constraint: sum(h) * dx = 1 => sum(h) = target_sum
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * (target_sum / current_sum)
        # Compute C5 = max_k ∫ h(x)(1 - h(x+k)) dx
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    # Define optimization bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define the constraint that integral = 1
    def constraint_func(h):
        return np.sum(h) - target_sum  # Should be zero
        
    # Optimization using scipy
    # We use L-BFGS-B to respect bounds
    # Also try to get a good local minimum starting from a diversified search
    
    best_h = h0.copy()
    best_c5 = objective(best_h)
    
    # Search for better configurations using random perturbations within constraints
    n_perturbations = 1000
    for i in range(n_perturbations):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Create a slightly modified version
        h = np.array(best_h)
        # Add small random Perturbations
        idx = np.random.randint(n_points)
        perturbation = np.random.uniform(-0.1, 0.1)
        h[idx] += perturbation
        # Clip values to [0, 1]
        h = np.clip(h, 0, 1)
        # Normalize to maintain ∫h = 1
        h = h * (target_sum / np.sum(h))
        
        # Evaluate
        c5 = objective(h)
        if c5 < best_c5:
            best_h = h.tolist()
            best_c5 = c5
    
    # Final evaluation to make sure we have the best
    final_c5 = evaluate_erdos_solution(best_h)
    
    # Return the result
    return (best_h, final_c5, n_points)
