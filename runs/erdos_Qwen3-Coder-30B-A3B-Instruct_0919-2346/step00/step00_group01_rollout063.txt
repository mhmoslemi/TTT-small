def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for a given step function h"""
    n_points = len(h)

    correlation = np.correlate(h, 1 - h, mode="full")
    # Maximum value over all lags
    max_correlation = np.max(correlation)
    return max_correlation * dx
