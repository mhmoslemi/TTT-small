    # Normalize so sum(h0) * dx = 1

    best_c5 = evaluate_erdos_solution(best_h)

    print(f"Initial bound: {best_c5:.6f}")

    solver = "CBC"
    opt_timeout_s = 0.8 * budget_s
    
    # Multi-start approach
    restarts = 0
    max_restarts = 50  # finite limit to not waste too much time
    
    # Time budget check
    def check_time():
        return time.time() - start_time < budget_s * 0.98

    while check_time() and restarts < max_restarts:
        # Start with continuous relaxation
        try:
            # Define variables
            h = Variable(n_points, name="h")
            tau = Variable(name="tau")

            # Convex constraints
            constraints = [
                h >= 0,
                h <= 1,
                sum(h) == target_sum,
                h[0] == 1  # symmetry breaking
            ]
            
            # Add products constraints efficiently
            y_vars = {}
            for k in range(1, n_points):
                y_vars[k] = Variable(n_points - k, name=f"y_{k}")
            
            for k in range(1, n_points):
                yk = y_vars[k]
                n_k = n_points - k
                
                # McCormick envelope constraints
                for i in range(n_k):
                    constraints += [
                        yk[i] >= h[i] + h[i + k] - 1,
                        yk[i] <= h[i],
                        yk[i] <= h[i + k],
                        yk[i] >= 0
                    ]
            
            # Tau constraints for each shift
            for k in range(1, n_points):
                yk = y_vars[k]
                constraints += [
                    tau <= sum(yk)
                ]
            
            # Solve continuous LP relaxation
            objective = Maximize(tau)
            prob = Problem(objective, constraints)
            prob.solve(solver=solver, verbose=False, max_seconds=opt_timeout_s)
            
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_sol = h.value
                tau_sol = tau.value

                # But for pure MLIP version, enforce binary immediately
    
        except Exception as e:
            print(f"Continous relaxation failed: {e}")

        finally:
            restarts += 1

        # Now solve binary version to get stronger bounds
        try:
            # Define binary variables
            h_binary = Variable(n_points, boolean=True, name='h_binary')
            tau = Variable(name='tau')

            # Constraints
            constraints = [
                sum(h_binary) == target_sum,
                h_binary[0] == 1  # symmetry breaking
            ]

            # Define product variables y[i,k] = h[i]*h[i+k]
            y_vars = {}
            for k in range(1, n_points):
                y_vars[k] = Variable(n_points - k, name=f'y_{k}')

            # Add constraints for all shifts
            for k in range(1, n_points):
                yk = y_vars[k]
                n_k = n_points - k

                for i in range(n_k):
                    constraints += [
                        yk[i] >= h_binary[i] + h_binary[i + k] - 1,
                        yk[i] <= h_binary[i],
                        yk[i] <= h_binary[i + k],
                        yk[i] >= 0
                    ]

            # Tau lower bounds for all k
            for k in range(1, n_points):
                yk = y_vars[k]
                constraints += [
                    tau <= sum(yk)
                ]

            # Optimization goal
            objective = Maximize(tau)
            
            prob = Problem(objective, constraints)

            start_solve = time.time()

                    h_np = np.array(h_sol)

                    indices_ones = np.argsort(h_np)[-int(target_sum):]

            corr = np.correlate(h_test, 1 - h_test, mode="full")

                        new_max = np.max(new_corr) * dx

                            best_c5 = new_max
                            best_h = h_test.copy()
