def compute_c5(h):
    """Compute the C5 constant for the given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode='full')
    # The max is over all shifts
    max_overlap = np.max(corr) * dx
    return max_overlap
