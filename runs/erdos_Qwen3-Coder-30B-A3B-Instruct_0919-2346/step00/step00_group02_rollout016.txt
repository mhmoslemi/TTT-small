import numpy as np
from scipy.optimize import differential_evolution
import time
import math

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute convolution of h with (1-h)
    conv = np.correlate(h, 1 - h, mode='full')
    # The maximum value of the convolution scaled by dx
    c5 = np.max(conv) * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial construction from problem statement
    n_points = 45
    h_values = initial_h_values(n_points)
    
    # We want to minimize C5 subject to constraints
    # Use differential evolution to optimize h_values
    
    # Define bounds for each variable (0 <= h[i] <= 1)
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Objective function to minimize
    def objective(h):
        # Scale h to satisfy integral constraint
        integral = np.sum(h) * (2.0 / len(h))
        if abs(integral - 1.0) > 1e-6:
            # Apply penalty for constraint violation
            penalty = 1000 * (integral - 1.0)**2
        else:
            penalty = 0
        c5 = evaluate_c5(h)
        return c5 + penalty
    
    # Optimization with time constraint
    best_c5 = float('inf')
    best_h = h_values.copy()
    
    # Run differential evolution with time control
    try:
        # Small population size to iterate quickly
        result = differential_evolution(
            func=objective,
            bounds=bounds,
            maxiter=200,
            popsize=10,
            seed=seed,
            mutation=(0.5, 1),
            recombination=0.7,
            atol=1e-6,
            rtol=1e-6,
            callback=lambda x, convergence: None
        )
        
        if not result.success:
            # If DE failed, use the original solution
            h_values = initial_h_values(n_points)
        else:
            h_values = result.x
        
        # Now apply local refinement (basically gradient descent)
        # But keep it simple - use iterative improvement
        step_size = 0.001
        max_refinement_iters = 500
        refinement_iter = 0
        
        while refinement_iter < max_refinement_iters:
            current_c5 = evaluate_c5(h_values)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_values.copy()
                
            if time.time() - start_time > budget_s * 0.9:
                break
                
            # Check gradients by finite differences
            h_new = h_values.copy()
            for i in range(n_points):
                # Try small perturbation
                h_plus = h_values.copy()
                h_minus = h_values.copy()
                delta = 0.001
                if h_values[i] + delta <= 1:
                    h_plus[i] += delta
                else:
                    h_plus[i] = 1.0
                if h_values[i] - delta >= 0:
                    h_minus[i] -= delta
                else:
                    h_minus[i] = 0.0
                
                c5_plus = evaluate_c5(h_plus)
                c5_minus = evaluate_c5(h_minus)
                
                # Update step
                grad = (c5_plus - c5_minus) / (2 * delta)
                h_new[i] = h_values[i] - step_size * grad
                
                # Clip to bounds
                h_new[i] = max(0, min(1, h_new[i]))
            
            # Now re-scale to maintain integral = 1
            integral = np.sum(h_new) * (2.0 / n_points)
            if abs(integral) > 1e-10:
                h_new = h_new * (1.0 / integral)
            
            # If we're not improving, break
            if abs(evaluate_c5(h_new) - current_c5) < 1e-8:
                break
                
            h_values = h_new
            refinement_iter += 1
            
    except:
        # Fallback to original construction if any error occurs
        pass
    
    # Final check
    final_c5 = evaluate_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    # Ensure integral constraint is satisfied exactly
    integral = np.sum(best_h) * (2.0 / n_points)
    if abs(integral - 1) > 1e-6:
        best_h = best_h * (1.0 / integral)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h)
    
    return (best_h, final_c5, n_points)

# Define initial_h_values function since it's referenced
def initial_h_values(n):
    """Initial construction of h for the problem"""
    h = np.zeros(n)
    # Simple initial guess
    h[0:n//2] = 1.0
    h[n//2:] = 0.0
    
    # Normalize the integral to 1
    integral = np.sum(h) * (2.0 / n)
    if abs(integral) > 1e-10:
        h = h * (1.0 / integral)
    
    return h
