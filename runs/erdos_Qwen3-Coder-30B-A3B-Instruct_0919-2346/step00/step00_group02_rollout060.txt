import numpy as np
import time
from scipy.optimize import differential_evolution
import math

def evaluate_erdos_solution(h_values):
    # This is a mock function based on the problem description
    # In the actual environment, this would be provided
    n_points = len(h_values)
    dx = 2.0 / n_points
    # Compute correlation with 1-h using np.correlate
    corr = np.correlate(h_values, 1 - h_values, mode='full')
    # Compute maximum over all shifts
    max_corr = np.max(corr)
    # Return C5 = 1 - min correlation (since we want to minimize max overlap)
    return 1 - max_corr * dx

initial_h_values = np.array([0.5] * 45)  # Default starting point

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize with a better starting configuration
    n_points = 100  # We'll use 100 points for better resolution
    if len(initial_h_values) < n_points:
        # Interpolate initial_h_values to n_points
        h = np.interp(np.linspace(0, 2, n_points), 
                      np.linspace(0, 2, len(initial_h_values)), 
                      initial_h_values)
    else:
        h = initial_h_values[:n_points].copy()
    
    h = np.clip(h, 0, 1)
    # Renormalize to integral = 1
    total = np.sum(h) * 2.0 / n_points
    h = h / total * 1.0
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Simulated Annealing with many restarts
    max_iter = 100000
    # Time tracking
    t0 = time.time()
    
    for iteration in range(max_iter):
        if time.time() - t0 > budget_s - 1:
            break
            
        # Create a new candidate by perturbing two indices
        i, j = np.random.choice(n_points, 2, replace=False)
        # Random step size
        step = 0.05 * (1 - iteration/max_iter)
        delta = np.random.uniform(-step, step)
        
        # Make sure perturbation doesn't cause out of bounds
        min_delta_i = -h[i]
        max_delta_i = 1 - h[i]
        min_delta_j = -h[j]
        max_delta_j = 1 - h[j]
        
        delta_i = np.clip(delta, min_delta_i, max_delta_i)
        delta_j = -delta_i  # Keep sum constant
        
        # Create candidate
        h_new = h.copy()
        h_new[i] += delta_i
        h_new[j] += delta_j
        
        # Evaluate candidate
        c5 = evaluate_erdos_solution(h_new)
        
        # Accept with simulated annealing probability
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
            h = h_new.copy()
        elif np.random.rand() < np.exp(-(c5 - best_c5) / (0.01 * (1 - iteration/max_iter))):
            h = h_new.copy()
    
    # Re-final normalization to ensure sum constraint
    total = np.sum(best_h) * 2.0 / n_points
    best_h = best_h / total * 1.0
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
