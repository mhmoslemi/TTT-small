import numpy as np
import time
from scipy.signal import convolve

def project_to_capped_simplex(h, target_sum):
    """Project h to the capped simplex: 0 <= h <= 1, sum(h) = target_sum"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if np.isclose(current_sum, target_sum):
        return h
    elif current_sum > target_sum:
        # Water-filling with lambda
        sorted_h = np.sort(h)[::-1]
        cumsum = np.cumsum(sorted_h)
        # Find threshold lambda
        for i in range(len(sorted_h)):
            if (sorted_h[i] * (i+1) - cumsum[i]) > (target_sum - cumsum[i]):
                # We found the index
                theta = (target_sum - cumsum[i]) / (i+1)
                break
        else:
            # All elements are at their lower bound of 0
            theta = sorted_h[-1] - (target_sum - cumsum[-1]) / len(sorted_h)
        return np.maximum(0, h - theta)
    else:
        # All elements are at their upper bound of 1
        return h

def compute_gradient(h, alpha, dx):
    """Compute gradient of F_alpha(h) and return gradient vector"""
    # For each shift k, compute O_k(h) = dx * sum h[i] * (1 - h[i+k])
    # Using FFT for efficiency: conv(h, 1-h) for all shifts
    corr = convolve(h, 1 - h, mode='full')
    
    # Softmax weights w_k = exp(alpha * O_k) / sum(exp(alpha * O_k))
    exp_O = np.exp(alpha * corr)
    w = exp_O / np.sum(exp_O)
    
    # Compute gradient vector using convolutions
    # We compute c1 = conv(w, 1-h) and c2 = conv(w_rev, h)
    c1 = convolve(w, 1 - h, mode='full')
    c2 = convolve(w[::-1], h, mode='full')
    
    # Gradient = c1 - c2 -> sliced to original length
    grad = c1[:len(h)] - c2[:len(h)]
    return grad

def compute_F_alpha(h, alpha, dx):
    """Compute surrogate function F_alpha(h) = (dx/alpha) * log(sum(exp(alpha * O_k(h))))"""
    corr = convolve(h, 1 - h, mode='full')
    exp_O = np.exp(alpha * corr)
    return (dx / alpha) * np.log(np.sum(exp_O))

def deterministic_annealing_step(h, alpha, eta, dx, max_iter=1000):
    """Perform deterministic annealing with adaptive α and η"""
    for i in range(max_iter):
        grad = compute_gradient(h, alpha, dx)
        h = h - eta * grad
        h = project_to_capped_simplex(h, len(h) / 2)
        
        if i % 200 == 0 and i > 0:
            alpha *= 1.3
            eta *= 0.99

    return h, alpha, eta

def binary_rounding(h_cont, n):
    """Round continuous solution h_cont to binary vector with exactly n/2 ones"""
    # Sort by descending value
    indices = np.argsort(h_cont)[::-1]
    h_bin = np.zeros(n)
    h_bin[indices[:n//2]] = 1.0
    return h_bin

def evaluate_c5(h, dx):
    """Evaluate C5 using the actual correlation, returns exact C5"""
    corr = np.correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

    start_time = time.time()
    
    # Choose a modest even n (larger than initial for better resolution)
    n = 200
    dx = 2.0 / n
    
    # Multi-start strategy: 5 to 7 restarts
    num_seeds = 7
    best_c5 = float('inf')
    best_h_binary = None
    
    for seed_i in range(num_seeds):
        current_time = time.time()
        if current_time - start_time > budget_s - 5:
            break
            
        # Initialize h randomly (can be fractional) and project
        if seed_i == 0 and 'initial_h_values' in globals():

            # Ensure it's not too far off from valid range

        else:

        h = project_to_capped_simplex(h, n / 2)
        
        # Annealing hyperparameters
        alpha = 0.5
        eta = 0.01
        max_alpha = 200.0
        max_iter = 2000
        
        for cycle in range(50):  # Early stopping if necessary
            current_time = time.time()
            if current_time - start_time > budget_s - 5:
                break
                
            # Run deterministic annealing for this temperature
            h, alpha, eta = deterministic_annealing_step(h, alpha, eta, dx)
            
            if alpha >= max_alpha:
                break
                
        # Rounding step
        h_binary = binary_rounding(h, n)
        
        # Evaluate final accuracy
        c5 = evaluate_c5(h_binary, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h_binary = h_binary.copy()
    
    # Final refinement with hill-climbing (limited swaps)
    if best_h_binary is not None:
        current_time = time.time()
        if current_time - start_time < budget_s - 5:
            # Hill climbing on the binary variable
            corr = np.correlate(best_h_binary, 1 - best_h_binary, mode='full')
            best_corr = np.max(corr)
            for _ in range(10000):
                if time.time() - start_time > budget_s - 5:
                    break
                # Pick two random different positions, one 1, one 0
                ones = np.where(best_h_binary == 1)[0]
                zeros = np.where(best_h_binary == 0)[0]

                    break

                new_h[i] = 0
                new_h[j] = 1
                
                # Recompute max correlation (for full result)

                new_max = np.max(new_corr)
                
                if new_max < best_corr:
                    best_h_binary = new_h
                    best_corr = new_max
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h_binary, dx)
    return (best_h_binary, final_c5, n)
