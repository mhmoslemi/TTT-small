import numpy as np
from cvxpy import *
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    # Parameters
    n = 70
    dx = 2.0 / n
    target_sum = n / 2
    
    initial_h = np.array(initial_h_values)
    
    # Solver info
    solver = 'CBC'  # Good for this size
    max_time = 0.6 * budget_s
    
    best_C5 = 1.0
    best_h = initial_h.copy()
    
    # Function to project onto capped simplex
    def project_simplex(v, s=target_sum):
        v = np.asarray(v)
        u = np.sort(v)[::-1]
        rho = 0
        for i in range(len(u)):
            if u[i] + (1.0 - s) / (i + 1) > 0:
                rho = i
        theta = (np.sum(u[:rho+1]) - s) / (rho + 1)
        result = np.maximum(v - theta, 0)
        return result
    
    # Function to evaluate C5 bound
    def compute_c5(h):
        h = np.asarray(h)
        c = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(c) * dx
        return c5
    
    # MILP approach
    def solve_milp(h_initial):
        h = Variable(n, name="h")
        tau = Variable(name="tau")
        y_vars = {}
        
        # Setup product variables y[i,k]
        for k in range(1, n):
            y_vars[k] = Variable(n - k, name="y_" + str(k))
        
        constraints = [
            sum(h) == target_sum,
            0 <= h,
            h <= 1,
            h[0] == 1,  # symmetry breaking
            tau <= sum(h),
        ]
        
        # Add McCormick envelope
        for k in range(1, n):
            y = y_vars[k]
            for i in range(n - k):
                # y[i] <= h[i]
                constraints.append(y[i] <= h[i])
                # y[i] <= h[i+k]
                constraints.append(y[i] <= h[i+k])
                # y[i] >= h[i] + h[i+k] - 1
                constraints.append(y[i] >= h[i] + h[i+k] - 1)
                # y[i] >= 0
                constraints.append(y[i] >= 0)
            
            # tau <= sum(y[k])
            constraints.append(tau <= sum(y))
        
        objective = Maximize(tau)
        prob = Problem(objective, constraints)
        
        prob.solve(solver=solver, verbose=False, time_limit=max_time)
        
        if prob.status in ['optimal', 'optimal_inaccurate']:
            h_val = h.value
            tau_val = tau.value
            return h_val, tau_val, True
        else:
            return None, None, False
    
    # Multi-start logic
    start_time = time.time()
    restart_count = 0
    
    # First try the initial solution as a warm-start
    while time.time() - start_time < budget_s and restart_count < 10:
        # Random binary start
        h_start = np.zeros(n)
        ones = np.random.choice(n, size=n//2, replace=False)
        h_start[ones] = 1.0
        h_start = h_start * 2  # Scale to [0,1] approx. or project below (no preserves average)
        
        # Alternate warm-start
        if restart_count > 0:
            h_val = np.random.rand(n) * 0.5 + 0.25  # somewhat random but bounded [0.25, 0.75]
            h_val = project_simplex(h_val, target_sum)
        else:
            h_val = project_simplex(h_start, target_sum)
        
        h_val = np.clip(h_val, 0, 1)  # ensure within bounds
        
        # Add relative perturbation
        if restart_count > 0 and restart_count < 8:
            noise = np.random.normal(0, 0.01, n)
            h_val += noise
            h_val = project_simplex(h_val, target_sum)
        
        # Now solve with CVXPY MILP
        h_sol, tau_val, success = solve_milp(h_val)
        if success and h_sol is not None:
            # If it's binary, perfect
            h_bin = np.round(h_sol)
            if np.allclose(h_sol, h_bin):
                h_used = h_bin
            else:
                # Still eval with continuous version first
                h_used = h_sol
                h_used = np.clip(h_used, 0, 1)
            
            c5_value = compute_c5(h_used)
            if c5_value < best_C5:
                best_C5 = c5_value
                best_h = h_used.copy()
        
        # Also consider good starting point from the given initial
        if restart_count == 0:
            h_used = initial_h
            c5_value = compute_c5(h_used)
            if c5_value < best_C5:
                best_C5 = c5_value
                best_h = h_used.copy()
                
        restart_count += 1
        
        if time.time() - start_time > budget_s * 0.95:
            break
    
    # Final local search over best solution
    h_final = best_h.copy()
    c5_final = compute_c5(h_final)
    
    # Swap improvements on the best known binary-like solution
    def improve_solution(h, c5_bound):
        h = np.array(h)
        c = np.correlate(h, 1 - h, mode="full")
        k_max = np.argmax(c)
        current_c = c[k_max] * dx
        if current_c <= c5_bound:
            return h, c5_bound
        
        improve = True
        while improve:
            improve = False
            for i in range(n):
                j = i + k_max
                if j < n:
                    # Swap if i=1 and j=0. 
                    if h[i] == 1 and h[j] == 0:
                        h[i], h[j] = 0, 1
                        new_c = np.correlate(h, 1 - h, mode="full")
                        new_c5 = np.max(new_c) * dx
                        if new_c5 < c5_bound:
                            c5_bound = new_c5
                            improve = True
                        else:
                            # If not better, backtrack
                            h[i], h[j] = 1, 0
            if not improve:
                break
        return h, c5_bound
    
    final_h, final_c5 = improve_solution(h_final, c5_final)
    
    return (final_h.tolist(), final_c5, n)
