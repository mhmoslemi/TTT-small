import numpy as np
import time
import cvxpy as cp
from scipy.optimize import linprog

def project_onto_capped_simplex(v, S):
    """Project vector v onto {h | 0 <= h_i <= 1, sum h_i = S}"""
    def check_sum(lam):
        h = np.clip(v - lam, 0, 1)
        return np.sum(h) - S

    low, high = -2.0, 2.0
    while high - low > 1e-10:
        mid = (low + high) / 2
        if check_sum(mid) > 0:
            low = mid
        else:
            high = mid
    lam = (low + high) / 2
    h = np.clip(v - lam, 0, 1)
    return h

def coeffs_and_const(h):
    n = len(h)
    # For each shift k in range [-n+1, n-1]
    a_matrix = np.zeros((2*n - 1, n))  # For all k, each row is a_k
    const_k = np.zeros(2*n - 1)
    k_idx = np.arange(-n + 1, n)  # k values from -n+1 to n-1
    
    for i, k in enumerate(k_idx):
        # index k: k-th right shift from i
        if k < 0:
            i_start = -k  # From i=0 to i_start-1, shift is valid
            a_matrix[i, :i_start] = 1.0
            # Tally cumulative shift terms for manual mixing
            a_matrix[i, i_start:] = 1.0 - h[:n - i_start] - h[i_start:]
            # homogenized part
            const_k[i] = np.sum(h[:n - i_start] * h[i_start:])
        elif k > 0:
            i_end = n - k  # i=0 to i_end-1
            a_matrix[i, k:] = 1.0
            a_matrix[i, :k] = 1.0 - h[k:] - h[:n - k]
            const_k[i] = np.sum(h[:n - k] * h[k:])
        else:  # k == 0
            a_matrix[i, :] = 1.0 - h - h
            const_k[i] = np.sum(h * h)
    return a_matrix, const_k, k_idx

def linear_program(a_matrix, const_vec, n, S):
    """Cvxpy solution to the LP: minimize t subject to linear constraints"""
    h = cp.Variable(n)
    t = cp.Variable(1)
    # Constraints
    constraints = []

    for i in range(len(const_vec)):
        # a[i,:] @ h + const[i] <= t
        constraints.append(a_matrix[i, :].T @ h + const_vec[i] <= t)

    # Mass constraint
    constraints.append(cp.sum(h) == S)

    # Bounds
    constraints.append(h >= 0)
    constraints.append(h <= 1)

    # Minimize t
    objective = cp.Minimize(t)

    # Solve the LP
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, verbose=False)

    return h.value, t.value

def evaluate(h):
    n = len(h)
    dx = 2.0 / n
    h_corr = np.correlate(h, 1-h, mode="full") * dx
    return np.max(h_corr)

def random_feasible(n):
    v = np.random.rand(n)
    v = project_onto_capped_simplex(v, n/2)
    return v

    max_outer = 30

    for r in range(num_restarts):

        if r == 0:

            C5_new = evaluate(h_new)
            if C5_new < best_C5:

        # Move small amount from one point to another
        i_from = 0
        i_to = (i_from + k_shift) % len(best_h)
        if best_h[i_from] > 1e-4 and best_h[i_to] < 1 - 1e-4:

                best_h[i_from] += delta
