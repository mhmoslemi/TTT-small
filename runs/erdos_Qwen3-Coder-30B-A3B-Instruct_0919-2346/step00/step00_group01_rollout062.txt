import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
import time
import math

def evaluate_erdos_solution(h_values):
    """Evaluate the Erdős constant C_5 for given h values."""
    h = np.array(h_values)
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all k values
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value
    c5_bound = np.max(correlation) * dx
    return c5_bound

def initial_h_values():
    """Return the initial construction."""
    # Using the initial construction with n=47
    n = 47
    h = np.zeros(n)
    dx = 2.0 / n
    # Create a step function that's 1 on [0, 1) and 0 on [1, 2]
    for i in range(n):
        x = i * dx
        if x < 1.0:
            h[i] = 1.0
        else:
            h[i] = 0.0
    # Normalize to have integral 1
    h = h / (np.sum(h) * dx)
    return h

def run(seed=42, budget_s=600):
    """Optimize the Erdős constant C_5."""
    np.random.seed(seed)
    
    # Start with the initial construction
    h0 = initial_h_values()
    n = len(h0)
    dx = 2.0 / n
    
    # Create bounds for variables
    bounds = [(0, 1) for _ in range(n)]
    
    # Optimization parameters
    max_iter = 10000
    tol = 1e-6
    start_time = time.time()
    
    # Flatten the optimization variables
    x0 = h0.copy()
    
    # Define objective function
    def objective(x):
        # Reshape x to h
        h = np.array(x)
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(correlation) * dx
        return c5
    
    # Define constraints
    def constraint_integral(x):
        h = np.array(x)
        return np.sum(h) * dx - 1.0
    
    constraint = {'type': 'eq', 'fun': constraint_integral}
    
    # Optimization
    best_x = x0.copy()
    best_c5 = objective(x0)
    
    # Try different optimization approaches
    for i in range(5):
        # Perturb the solution slightly to escape local minima
        noise = np.random.normal(0, 0.01, n)
        x_perturbed = np.clip(x0 + noise, 0, 1)
        
        # Optimize with L-BFGS-B
        res = minimize(objective, x_perturbed, method='L-BFGS-B', bounds=bounds, 
                       constraints=constraint, options={'maxiter': 1000, 'ftol': tol},
                       callback=None)
        
        if res.success and res.fun < best_c5:
            best_x = res.x
            best_c5 = res.fun
            
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
            
    # Try local search strategies
    for _ in range(50):
        if time.time() - start_time >= budget_s * 0.95:
            break
            
        # Generate random step functions
        h_candidate = np.random.rand(n)
        h_candidate = h_candidate / (np.sum(h_candidate) * dx)
        
        c5_candidate = objective(h_candidate)
        
        if c5_candidate < best_c5:
            best_x = h_candidate
            best_c5 = c5_candidate
    
    # Refine using more precise optimization
    elapsed = time.time() - start_time
    if elapsed < budget_s * 0.8:
        # Try to optimize the best solution found so far with gradient-based method
        try:
            # Try using the evaluated solution briefly for refinement
            res = minimize(objective, best_x, method='L-BFGS-B', bounds=bounds, 
                           constraints=constraint, options={'maxiter': 500, 'ftol': tol})
            if res.success and res.fun < best_c5:
                best_x = res.x
                best_c5 = res.fun
        except:
            pass
        
    # Final evaluation
    final_h = best_x
    c5_bound = evaluate_erdos_solution(final_h)
    
    return (final_h.tolist(), c5_bound, len(final_h))
