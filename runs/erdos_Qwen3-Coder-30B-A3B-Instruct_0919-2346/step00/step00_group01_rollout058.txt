import numpy as np
import cvxpy as cp
import time
from scipy import fft
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n = 60
    dx = 2.0 / n
    
    # Initial shift set K0
    K0 = [1, n//2]
    if n > 4:
        K0.append(2)
        K0.append(n-2)
    K0 = sorted(set(K0))
    
    # Initialize best solution
    best_h = None
    best_c5 = float('inf')
    
    # Try to find feasible solution via MILP
    def solve_milp(K_current, remaining_time):
        # Binary variables h
        h = cp.Variable(n, boolean=True)
        
        # Constraint: sum(h) = n/2
        constraints = [cp.sum(h) == n//2]
        
        # Fix h[0] = 1 to break symmetry
        constraints.append(h[0] == 1)
        
        # Initialize w and a for current shifts
        w_vars = {}
        a_vars = {}
        
        # Create auxiliary binary variables w_{i,k} and constraints for each k in K_current
        for k in K_current:
            w_vars[k] = cp.Variable((n, 1), boolean=True)
            a_vars[k] = cp.sum(w_vars[k])
            
            # Add McCormick constraints: w <= h[i], w <= h[i+k], w >= h[i] + h[i+k] - 1
            for i in range(n):
                idx = (i + k) % n
                constraints.append(w_vars[k][i] <= h[i])
                constraints.append(w_vars[k][i] <= h[idx])
                constraints.append(w_vars[k][i] >= h[i] + h[idx] - 1)
        
        # Variables for min_a_k
        t = cp.Variable()
        for k in K_current:
            constraints.append(t <= a_vars[k])
        
        # Maximize t
        objective = cp.Maximize(t)
        prob = cp.Problem(objective, constraints)
        
        # Solve with time constraint
        try:
            prob.solve(solver=cp.CBC, max_seconds=remaining_time)
            if prob.status in [cp.OPTIMAL, cp.OPTIMIZED_ECP]:
                h_sol = [int(hi.value) for hi in h]
                return h_sol, t.value
        except Exception:
            pass
        return None, None
    
    # Basic binary search with few initial shifts
    current_K = K0[:]

    # Repeatedly solve MILP and add worst shift
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s - 20:
            break
        
        prev_K = current_K[:]
        remaining_time = max(1.0, budget_s - elapsed - 10)
        h_sol, t_opt = solve_milp(current_K, remaining_time)
        
        if h_sol is None:
            break
            
        # Compute full circular autocorrelations
        h_array = np.array(h_sol)
        A_full = []
        for k in range(1, n):
            A_k = np.sum(h_array * np.roll(h_array, k))
            A_full.append(A_k)
        
        # Find minimum A_k and add it to shift set if not already there
        min_A_k = min(A_full)
        min_k = np.argmin(A_full) + 1
        
        if min_A_k < t_opt - 1e-6:
            current_K.append(min_k)
            current_K = sorted(set(current_K))
        else:
            break
    
    # Final evaluation of binary solution
    if h_sol is not None:
        h_array = np.array(h_sol)
        A_full = []
        for k in range(1, n):
            A_k = np.sum(h_array * np.roll(h_array, k))
            A_full.append(A_k)
        
        min_A_k = min(A_full)
        curr_c5 = 1 - 2/n * min_A_k
        
        if curr_c5 < best_c5:
            best_h = h_sol[:]
            best_c5 = curr_c5
    
    # Try fractional polishing if feasible and time permits
    if best_h is not None and time.time() - start_time < budget_s - 50:
        # Start with best binary h
        h_cont = np.array(best_h, dtype=float)
        h_cont = np.clip(h_cont, 0, 1)
        
        # Project to fixed area = n/2
        total = np.sum(h_cont)
        adjustment = (n/2.0 - total) / n
        h_cont += adjustment
        
        # Clip to [0, 1]
        h_cont = np.clip(h_cont, 0, 1)
        
        # Project to area == n/2
        total = np.sum(h_cont)
        if abs(total - n/2) > 1e-6:
            # Adjust via swap in small steps
            diff = total - n/2
            if diff > 0:
                # Need to make some entries smaller
                idx_pos = np.where(h_cont > 0)[0]
                for idx in idx_pos:
                    if diff <= 0:
                        break
                    loss = min(h_cont[idx], diff)
                    h_cont[idx] -= loss
                    diff -= loss
            else:
                # Need to make some entries larger
                idx_neg = np.where(h_cont < 1)[0]
                for idx in idx_neg:
                    if diff >= 0:
                        break
                    gain = min(1 - h_cont[idx], -diff)
                    h_cont[idx] += gain
                    diff += gain
                    
        # Gradient step
        optimization_iterations = 200
        step_size = 0.01
        
        for _ in range(optimization_iterations):
            # Compute shifted arrays for gradient
            min_k = np.argmin([np.sum(h_cont * np.roll(h_cont, k)) for k in range(1, n)])
            grad = np.roll(h_cont, min_k) + np.roll(h_cont, -min_k)
            h_cont += step_size * grad
            h_cont = np.clip(h_cont, 0, 1)
            
            # Project back to area n/2
            total = np.sum(h_cont)
            adjustment = (n/2.0 - total) / n
            h_cont += adjustment
            h_cont = np.clip(h_cont, 0, 1)
            
            # Calculate new bound
            A_all = [np.sum(h_cont * np.roll(h_cont, k)) for k in range(1, n)]
            min_A = min(A_all)
            temp_c5 = 1 - 2/n * min_A
            if temp_c5 < best_c5:
                best_c5 = temp_c5
                best_h = h_cont.copy()
    
    # If needed, try spectral fallback
    if best_c5 > 0.3807 and time.time() - start_time < budget_s - 120:
        # Use continuous Fourier parameterization
        # Parameterize h as: h_i = 0.5 + sum(alpha_m * cos(2*pi*i*m/n))
        M = 5
        
        def compute_autocorr_from_coeffs(alpha):
            h_vals = 0.5 + np.array([sum(alpha[m] * np.cos(2 * np.pi * i * (m+1) / n) for m in range(M)) for i in range(n)])
            h_vals = np.clip(h_vals, 0, 1)
            # FFT-based computation of autocorrelations
            H_fft = fft.fft(h_vals)
            cross_corr_fft = H_fft * np.conj(H_fft)
            cross_corr = fft.ifft(cross_corr_fft).real
            A_vals = cross_corr[1:]  # Exclude zero shift
            return A_vals, h_vals

        def objective(alpha):
            A_vals, _ = compute_autocorr_from_coeffs(alpha)
            # Use smooth approximation to min
            beta = 10.0
            log_sum = np.log(np.sum(np.exp(-beta * A_vals)))
            phi = -log_sum / beta
            return -phi  # Since we maximize

        # Bound on alpha magnitudes to keep h in [0, 1]
        bounds = [(-0.5, 0.5) for _ in range(M)]
        
        # Differential evolution optimization
        result = differential_evolution(objective, bounds, maxiter=300, popsize=30, seed=seed)
        
        if result.success:
            alpha_best = result.x
            A_vals, h_final = compute_autocorr_from_coeffs(alpha_best)
            min_A = min(A_vals)
            temp_c5 = 1 - 2/n * min_A
            
            if temp_c5 < best_c5:
                best_c5 = temp_c5
                best_h = h_final

    if best_h is None:
        # Fall back to a random binary construction if needed
        h_binary = np.zeros(n)
        ones_indices = np.random.choice(n, n//2, replace=False)
        h_binary[ones_indices] = 1
        best_h = h_binary
        
        # Smooth it via simple modification
        best_h = np.clip(best_h, 0, 1)
        total = np.sum(best_h)
        diff = n/2 - total
        adjustment = diff / n
        best_h += adjustment
        best_h = np.clip(best_h, 0, 1)
        
        A_all = [np.sum(best_h * np.roll(best_h, k)) for k in range(1, n)]
        min_A = min(A_all)
        best_c5 = 1 - 2/n * min_A

    return (np.array(best_h), best_c5, n)
