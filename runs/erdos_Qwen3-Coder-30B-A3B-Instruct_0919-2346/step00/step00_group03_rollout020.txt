import numpy as np
import time
import random

def evaluate_c5(h):
    """Evaluate C5 for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h)
    conv = np.correlate(h, 1 - h, mode="full")
    # Take maximum of the correlation values
    max_corr = np.max(conv)
    return max_corr * dx

def random_binary_seq(n):
    """Generate a random binary sequence with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    ones_indices = np.random.choice(n, size=n//2, replace=False)
    h[ones_indices] = 1.0
    return h

def solve_binary(n, iters, seed=None, init_h=None):
    """Solve using simulated annealing on binary variables."""
    rng = np.random.default_rng(seed)
    if init_h is not None:
        h = init_h.copy()
    else:
        h = random_binary_seq(n)
    
    c5 = evaluate_c5(h)
    best_h, best_c5 = h.copy(), c5
    
    T0 = 0.05
    T = T0
    cooling = 0.99995
    
    # Precompute indices that are ones and zeros for efficiency
    ones = set(np.where(h > 0.5)[0])
    zeros = set(np.where(h < 0.5)[0])
    
    for iteration in range(iters):
        if len(ones) == 0 or len(zeros) == 0:
            break
            
        # Choose a random one and zero to swap
        i = rng.choice(list(ones))
        j = rng.choice(list(zeros))
        
        # Flip the values
        h[i], h[j] = h[j], h[i]
        
        # Update sets efficiently
        ones.remove(i)
        zeros.remove(j)
        if h[i] > 0.5:
            ones.add(i)
        else:
            zeros.add(i)
        if h[j] > 0.5:
            ones.add(j)
        else:
            zeros.add(j)
        
        # Compute new C5
        new_c5 = evaluate_c5(h)
        delta = new_c5 - c5
        
        # Accept or reject
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            c5 = new_c5
            if c5 < best_c5:
                best_c5, best_h = c5, h.copy()
        else:
            # Revert swap
            h[i], h[j] = h[j], h[i]
            # Revert sets - already done above
            
        T *= cooling
    
    return best_h, best_c5
