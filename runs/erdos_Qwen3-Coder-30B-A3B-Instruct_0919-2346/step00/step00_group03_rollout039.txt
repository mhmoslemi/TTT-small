def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    # Use FFT for efficient computation
    h_fft = fft(h)
    one_minus_h_fft = fft(1 - h)
    corr = ifft(h_fft * np.conj(one_minus_h_fft)).real
    # The peak overlap is the maximum over all valid shifts
    s_max = np.max(corr)
    return s_max * dx

def project_to_feasible(h, n):
    """Project h onto the feasible region: sum(h) = n/2, 0 <= h[i] <= 1"""
    # Ensure that sum == n/2 using affine scaling
    h_sum = np.sum(h)
    target_sum = n / 2.0
    mean_deviation = (h_sum - target_sum) / n
    h = h - mean_deviation
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Do one step of Dykstra's projection to fix the sum
    h_sum_new = np.sum(h)
    if abs(h_sum_new - target_sum) > 1e-10:
        correction = (target_sum - h_sum_new) / n
        h = h + correction
        h = np.clip(h, 0, 1)
    return h

def local_mismatch_swap(h, n, dx, max_iter=5000):
    """Apply deterministic mismatch swap heuristic"""
    h = h.copy()
    for _ in range(max_iter):
        # Compute max correlation
        h_fft = fft(h)
        one_minus_h_fft = fft(1 - h)
        corr = ifft(h_fft * np.conj(one_minus_h_fft)).real
        max_shift = np.argmax(corr)
        max_corr = corr[max_shift]
        # Make C5 the current upper bound
        current_c5 = max_corr * dx
        if current_c5 <= 0.379:
            break
        # Identify sets for swapping
        # For shift max_shift, indices are i, i+max_shift for i=0 to n-1-max_shift
        shift = max_shift - (n-1)  # map back to actual shift index
        min_i = max(0, -shift)
        max_i = min(n-1, n-1-shift)
        if shift >= 0:  # h[i] * (1-h[i+shift])
            M1 = [(i, i+shift) for i in range(min_i, max_i+1) if h[i] >= 0.99 and h[i+shift] <= 0.01]
            M0 = [(i, i+shift) for i in range(min_i, max_i+1) if h[i] <= 0.01 and h[i+shift] >= 0.99]
        else:  # h[i-shift] * (1-h[i])
            M1 = [(i, i-shift) for i in range(min_i, max_i+1) if h[i] >= 0.99 and h[i-shift] <= 0.01]
            M0 = [(i, i-shift) for i in range(min_i, max_i+1) if h[i] <= 0.01 and h[i-shift] >= 0.99]
        if not M1 or not M0:
            # If swap sets are empty, introduce a tiny random adjustment
            idx1, idx2 = np.random.randint(0, n, size=2)
            if idx1 == idx2: continue
            delta = 0.001
            if h[idx1] < 1.0:
                h[idx1] += delta
                h[idx2] -= delta
            else:
                h[idx1] -= delta
                h[idx2] += delta
            h = project_to_feasible(h, n)
            continue
        # Pick swap pair
        i, j = M1[0]
        # Swap
        delta_h = min(h[i], 1-h[j]) if i != j else 0.0
        if delta_h > 0:
            h[i] -= delta_h
            h[j] += delta_h
        else:
            i, j = M0[0]
            delta_h = min(h[j], 1-h[i]) if i != j else 0.0
            if delta_h > 0:
                h[i] += delta_h
                h[j] -= delta_h
        h = project_to_feasible(h, n)
    return h
