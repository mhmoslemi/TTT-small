import numpy as np
import time
from scipy.optimize import differential_evolution
from scipy.spatial.distance import cdist

def evaluate_erdos_solution(h_values):
    """Compute the C5 bound from the given h_values."""
    n_points = len(h_values)
    dx = 2.0 / n_points
    correlation = np.correlate(h_values, 1 - np.array(h_values), mode='full')
    max_correlation = np.max(correlation)
    return max_correlation * dx

def initial_h_values():
    """Initial construction from problem."""
    n = 45
    dx = 2.0 / n
    h = np.zeros(n)
    # Create a piecewise constant function that's roughly balanced
    # We'll use a design that avoids extreme clustering
    for i in range(n):
        x = i * dx
        if (x >= 0.5 and x <= 1.5):
            h[i] = 1.0
        else:
            h[i] = 0.0
    # Normalize to sum = 1
    h = h * (1.0 / np.sum(h))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    
    # Start with the initial solution
    h_base = initial_h_values()
    n_points = len(h_base)
    dx = 2.0 / n_points
    
    # Function to compute C5
    def compute_c5(h):
        # Compute overlap integral using correlation
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    # Define our optimization objective as minimizing C5
    def objective(h):
        return compute_c5(h)
    
    # Constraints
    def constraint_sum(h):
        return np.sum(h) - 1.0  # Should be 0

    def constraint_bounds(h):
        return np.concatenate([h, 1-h])  # h >= 0 and h <= 1

    # Method 1: Simple local optimization with bounds
    best_h = h_base.copy()
    best_c5 = compute_c5(best_h)
    
    # Add a slight random perturbation for diversity
    h_start = best_h.copy()
    h_start += np.random.normal(0, 0.005, n_points)
    h_start = np.clip(h_start, 0, 1)
    h_start = h_start * (1.0 / np.sum(h_start))  # re-normalize
    current_h = h_start.copy()
    current_c5 = compute_c5(current_h)
    
    # Start a hill climbing search
    max_iterations = 250000
    iter_count = 0
    
    # Starting temperature for simulated annealing
    temp_start = 0.1
    temp_min = 1e-6
    alpha = 0.9999

    while iter_count < max_iterations and (time.time() - start_time) < (budget_s - 1):
        if (time.time() - start_time) >= (budget_s - 1):
            break
            
        # Pick two random indices to modify
        i, j = np.random.choice(n_points, 2, replace=False)
        
        # Compute allowable delta
        hi, hj = current_h[i], current_h[j]
        delta_up = min(1.0 - hi, hj)  # Maximum increase of hi, decrease of hj
        delta_down = min(hi, 1.0 - hj)  # Maximum decrease of hi, increase of hj
        # Make sure it's a valid range
        if delta_up < 0 or delta_down < 0:
            iter_count += 1
            continue
            
        # Choose a random step
        delta = np.random.uniform(-delta_down, delta_up)
        
        # Apply the change
        new_h = current_h.copy()
        new_h[i] = hi + delta
        new_h[j] = hj - delta
        
        # Clipping to stay within bounds
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (1.0 / np.sum(new_h))  # Normalize
        
        # Test new solution
        new_c5 = compute_c5(new_h)
        
        # Accept or reject the change
        if new_c5 < current_c5:
            current_h = new_h
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        else:
            # Accept with simulated annealing probability
            delta_e = new_c5 - current_c5
            prob = np.exp(-delta_e / max(temp_start, temp_min))
            if np.random.rand() < prob:
                current_h = new_h
                current_c5 = new_c5

        iter_count += 1
        temp_start *= alpha
        if temp_start < temp_min:
            temp_start = temp_min

    # Final check with the best solution
    c5_bound = compute_c5(best_h)
    
    return (best_h, c5_bound, n_points)
