import numpy as np
import time

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # correlate h with (1-h) in full mode
    corr = np.correlate(h, 1 - h, mode='full')
    # multiply by dx to get actual integral values
    corr *= dx
    # return the maximum of these integrals
    return np.max(corr)

def propose_move(h, max_step):
    """Propose a new feasible h by adjusting two entries"""
    h = h.copy()
    N = len(h)
    while True:
        i, j = np.random.choice(N, 2, replace=False)
        # feasible range for δ
        delta_low = max(-h[i], h[j] - 1.)
        delta_high = min(1. - h[i], h[j])
        if delta_low >= delta_high:
            continue  # cannot move this pair; try again
        # sample δ from uniform distribution
        step = min(max_step, delta_high - delta_low)
        delta = np.random.uniform(-step, step)
        h[i] += delta
        h[j] -= delta
        return h

def anneal(initial_h, time_budget):
    """Run simulated annealing to minimize C5"""
    start = time.time()
    cur = initial_h.copy()
    best = initial_h.copy()
    c_cur = c5_of(cur)
    c_best = c_cur
    T_start = 0.05
    T_end = 1e-6
    step_start = 0.2
    T = T_start
    step = step_start

    while time.time() - start < time_budget:
        t = time.time() - start
        elapsed_frac = t / time_budget
        T = T_start * (T_end / T_start) ** elapsed_frac
        step = step_start * (1 - elapsed_frac)

        h_new = propose_move(cur, step)
        c_new = c5_of(h_new)
        
        if c_new <= c_cur:
            cur = h_new
            c_cur = c_new
            if c_new < c_best:
                best = h_new
                c_best = c_new
        else:
            # Metropolis acceptance
            if np.random.rand() < np.exp(-(c_new - c_cur) / T):
                cur = h_new
                c_cur = c_new

    return best, c_best

    a = 0.3
    u = 0.9

    edges = [int(a/dx), int((2-a)/dx)]

        x = i * dx

    h_double *= scale_factor
    
    # Run search
    best_h = None
    best_C5 = float('inf')
    remaining_budget = budget_s
    
    seeds = [h_uniform, h_initial, h_double]
    budget_per_seed = remaining_budget * 0.30
