import numpy as np
import time
from scipy import signal

def project_to_capped_simplex(h, target_sum):
    """Project h onto the capped simplex: 0 <= h[i] <= 1 and sum(h) = target_sum"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    sorted_h = np.sort(h)[::-1]
    cumsum = np.cumsum(sorted_h)
    n = len(h)
    # Find the threshold lambda
    lambda_val = (cumsum[-1] - target_sum) / n
    # We must ensure lambda does not make any entry below 0
    # Find where the cumulative sum crosses the target
    # We compute the water filling threshold
    for i in range(n):
        if (sorted_h[i] - lambda_val) >= 0:
            continue
        else:
            # This is approximate, in the case of project_to_capped_simplex,
            # we recompute lambda properly.
            # Let's be more careful:
            # The right place to stop is where lambda would make all elements
            # above it be the limit.
            # s = sum (h[j] - lambda) for all j where h[j] > lambda
            # that means: sum_h - (n - k) * lambda = target
            # lambda = (sum_h - target) / (n - k)
            # We have sorted_h, let's iterate
            break
    # Simple implementation
    # Find the water level in a sorted array
    sorted_h = np.sort(h)[::-1]
    cumsum = np.cumsum(sorted_h)
    # Note: not directly true for this case as sorted isn't completely zero-indexed
    # But doing correctly using bracketing is tiresome for now, we use a simpler method:
    # Get actual lambda such that sum of (h[i] - lambda)+ = target_sum
    from scipy.optimize import bisect
    
    def func(lambda_val):
        return np.sum(np.maximum(h - lambda_val, 0)) - target_sum
    
    try:
        lambda_correct = bisect(func, -1.0, 2.0)
        h_pro = np.maximum(h - lambda_correct, 0)
        return h_pro
    except:
        # fallback to an approximate linear search
        lambda_min, lambda_max = 0.0, 1.0
        eps = 1e-8
        if current_sum < target_sum:
            lambda_min = 0.0
        else:
            lambda_max = 1.0
        # Not ideal, but should find a solution for small n
        h_pro = np.clip(h, 0, 1)
        # Just take a several steps for bisection
        for _ in range(20):
            h_test = np.clip(h - (lambda_min + lambda_max) / 2, 0, 1)
            s = np.sum(h_test)
            if s < target_sum:
                lambda_max = (lambda_min + lambda_max) / 2
            else:
                lambda_min = (lambda_min + lambda_max) / 2
        final_h = np.clip(h - (lambda_min + lambda_max) / 2, 0, 1)
        return final_h

def evaluate_C5(h, dx=2.0):
    """
    Evaluate the exact C5 using full correlation.
    """
    n = len(h)
    corr = np.correlate(h, 1-h, mode='full')
    max_corr = np.max(corr) * dx
    return max_corr

def evaluate_surrogate(h, alpha, dx):
    """
    Evaluate the smooth surrogate F_alpha.
    """
    n = len(h)
    corr = np.correlate(h, 1-h, mode='full')
    if alpha == 0.0:
        return np.max(corr) * dx
    max_k = len(corr) - n
    # For each k, compute O_k(h)
    log_sum = -np.inf
    for l in range(max_k):
        # O_l comes from corr:
        log_sum = np.logaddexp(log_sum, alpha * corr[l] * dx)
    return log_sum / alpha

def compute_gradient(h, alpha, dx):
    """
    Compute gradient of F_alpha with respect to h using convolutions.
    """
    # Compute all O_k(h) values with k=0, 1, ..., n-1
    corr = np.correlate(h, 1-h, mode='full')
    max_k = len(corr) - len(h)
    O_ell = np.zeros(max_k)
    for k in range(max_k):
        O_ell[k] = corr[k] * dx

    # Softmax of O_ell: weights w_ell
    O_ell_scaled = alpha * O_ell
    # Subtract maximum for numerics
    max_O = np.max(O_ell_scaled)
    exp_O = np.exp(O_ell_scaled - max_O)
    w = exp_O / np.sum(exp_O)

    # Gradient using standard convolutions
    w_rev = w[::-1]
    # We compute:
    # grad[i] = sum_{ell} w[ell] * ( (1 - h[i + ell]) - h[i - ell] )
    # This requires both `convs` and work with boundary conditions

    # 1) conv1d(w, 1 - h) = sum(w[ell] * (1-h[i+ell]))
    c1 = np.convolve(w, 1 - h, mode='full')
    # 2) conv1d(w_rev, h) = sum(w_rev[ell] * h[i - ell]) = sum(w[ell] * h[i - ell])
    c2 = np.convolve(w_rev, h, mode='full')

    # Now extract the valid middle part of size n
    offset1 = len(w) - 1
    offset2 = len(w_rev) - 1
    # These are the same now (w_rev length == w length)
    offset = len(w) - 1
    
    grad = c1[offset : offset + len(h)] - c2[offset : offset + len(h)]
    return grad

def optimize_step(h, grad, alpha, dx, learning_rate):
    """ Perform one gradient step """
    h_new = h - learning_rate * grad
    
    # Project onto feasible set:
    h_proj = project_to_capped_simplex(h_new, len(h) / 2)
    # h_proj should be in [0,1] and sum(h_proj) = n/2
    return h_proj
