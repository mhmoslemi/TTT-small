import numpy as np
from scipy.optimize import minimize
import time
from math import sqrt

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value corresponds to the maximum overlap
    c5 = np.max(correlation) * dx
    return c5

def initial_h_values():
    """Return initial construction - a step function with 47 samples"""
    n = 47
    h = np.zeros(n)
    # Simple construction: step function with value 1 in first half, 0 in second half
    # But we want ∫₀² h(x) dx = 1, so we scale appropriately
    h[:24] = 2.0/24  # First 24 points with height 2/24 to make integral = 1
    h[24:] = 0
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Run optimization to find improved upper bound for C5"""
    np.random.seed(seed)
    
    # Start with initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    
    # Discretization parameters
    dx = 2.0 / n_points
    
    # Objective function to minimize
    def objective(h):
        # Ensure h is within bounds [0, 1]
        h = np.clip(h, 0, 1)
        return evaluate_erdos_solution(h)
    
    # Constraints
    def constraint_sum(h):
        # integral condition: ∫₀² h(x) dx = 1
        return np.sum(h) * dx - 1
    
    def constraint_bounds(h):
        # h ∈ [0,1] is enforced by clipping in objective
        return np.concatenate([h, 1-h])
    
    # Preserve constraint that h is in [0,1]
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Use a local optimization algorithm
    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Optimization with random restarts
    start_time = time.time()
    
    # Simple gradient-free method: randomly perturb and accept better solutions
    num_restarts = 10
    current_h = best_h.copy()
    current_c5 = best_c5
    
    # Run random perturbations
    max_iterations = 5000
    iteration = 0
    
    while time.time() - start_time < budget_s and iteration < max_iterations:
        # Random perturbation
        new_h = current_h + np.random.normal(0, 0.02, n_points)
        new_h = np.clip(new_h, 0, 1)
        
        # Ensure integral constraint
        integral = np.sum(new_h) * dx
        if abs(integral - 1) > 1e-6:
            scale = 1.0 / integral
            new_h = new_h * scale
            
        # Check if new solution is better
        new_c5 = evaluate_erdos_solution(new_h)
        if new_c5 < current_c5:
            current_h = new_h.copy()
            current_c5 = new_c5
            
        # Set best solution if improved
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            
        iteration += 1
    
    # Final check to make sure we maintain required constraint
    integral = np.sum(best_h) * dx
    if abs(integral - 1) > 1e-6:
        scale = 1.0 / integral
        best_h = best_h * scale
        best_c5 = evaluate_erdos_solution(best_h)
        
    return (best_h, best_c5, n_points)
