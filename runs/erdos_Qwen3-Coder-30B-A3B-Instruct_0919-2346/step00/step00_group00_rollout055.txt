def project_to_simplex(h, target_sum):
    """Project h onto the capped simplex: 0 <= h <= 1 and sum(h) = target_sum."""
    # Clamp to [0,1]
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Waterfilling projection
    # Sort h in descending order
    sorted_indices = np.argsort(h)[::-1]
    sorted_h = h[sorted_indices]
    
    # Find the threshold lambda
    cumsum = np.concatenate([[0], np.cumsum(sorted_h)])
    n = len(h)
    # We want sum(max(0, h[i] - lambda)) = target_sum
    # For sorted values: we can compute the lambda by finding the right threshold
    # This line is approximate and assumes we only lose a few elements
    # Let's compute precisely
    
    lambda_val = 0.0
    for i in range(n):
        # sum of elements from 0 to i-1 not affected, sum of elements i to n-1 are reduced by lambda
        total = cumsum[i] + (n - i) * (sorted_h[i] - lambda_val)
        if total <= target_sum:
            lambda_val = sorted_h[i] - (target_sum - cumsum[i]) / (n - i)
            break
    else:
        # All elements are reduced
        lambda_val = sorted_h[-1] - (target_sum - cumsum[-1]) / n
    
    # Apply threshold
    h = np.clip(h - lambda_val, 0, 1)
    return h

def evaluate_corr(h, dx):
    """Evaluate the actual correlation (not surrogate)"""
    corr = np.correlate(h, 1 - h, mode="full")
    return dx * np.max(corr)

def compute_surrogate_grad(h, alpha, dx, n):
    """Compute gradient of surrogate F_alpha"""
    # Compute O_ℓ(h) for all ℓ
    O_ell = np.zeros(2 * n - 1)
    for l in range(-n + 1, n):
        if l >= 0:
            # h[l:] * (1 - h[:n-l])
            prod = h[l:] * (1 - h[:n-l])
        else:
            # h[:n+l] * (1 - h[-l:])
            prod = h[:n+l] * (1 - h[-l:])
        O_ell[l + n - 1] = np.sum(prod)
    
    # Compute soft-max weights
    exp_O = np.exp(alpha * O_ell)
    w = exp_O / np.sum(exp_O)
    
    # Convolution: w * (1 - h) and w_rev * h
    w_rev = w[::-1]
    # Using FFT for convolutions with full mode
    c1 = np.convolve(w, 1 - h, mode='full')
    c2 = np.convolve(w_rev, h, mode='full')
    grad = c1[:n] - c2[:n]
    return grad
