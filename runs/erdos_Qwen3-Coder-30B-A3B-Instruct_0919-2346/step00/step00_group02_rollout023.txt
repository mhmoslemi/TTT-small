import numpy as np
import time
import random

def c5_of(h):
    """Computes the maximum overlap integral C_5 for discretized function h."""
    dx = 2.0 / len(h)
    # Compute overlaps using full correlation mode
    overlap = np.correlate(h, 1 - h, mode="full")
    # Scale by dx and take maximum
    c5 = np.max(overlap) * dx
    return c5

def propose_move(h, max_step):
    """Proposes a new feasible vector h' by adjusting two entries."""
    h_new = h.copy()
    N = len(h)
    i, j = random.sample(range(N), 2)
    # Calculate feasible range for δ
    delta_low = max(-h_new[i], h_new[j] - 1)
    delta_high = min(1 - h_new[i], h_new[j])
    if delta_low >= delta_high:
        # Cannot move this pair
        return h_new
    # Clip step size
    step = min(max_step, abs(delta_high - delta_low))
    delta = random.uniform(-step, step)
    h_new[i] += delta
    h_new[j] -= delta
    return h_new

def anneal(initial_h, time_budget, seed=None):
    """Simulated annealing optimization of h."""
    if seed is not None:
        random.seed(seed)
        np.random.seed(seed)
    start = time.time()
    cur = initial_h.copy()
    best = cur.copy()
    c_cur = c5_of(cur)
    c_best = c_cur

    # Parameters
    T_start = 0.05
    T_end = 1e-6
    step_start = 0.2
    N = len(initial_h)

    while time.time() - start < time_budget:
        elapsed = time.time() - start
        t_frac = elapsed / time_budget
        T = T_start * (T_end / T_start) ** t_frac
        step = step_start * (1 - t_frac)
        if step < 1e-6:
            step = 1e-6

        h_new = propose_move(cur, step)
        c_new = c5_of(h_new)
        if c_new < c_best:
            best = h_new.copy()
            c_best = c_new
        if c_new <= c_cur or random.random() < np.exp(-(c_new - c_cur) / T):
            cur = h_new
            c_cur = c_new
    return best, c_best

    random.seed(seed)
    np.random.seed(seed)
    start = time.time()
    N = 140
    dx = 2.0 / N
    # Build the three initial seeds
    # 1. Uniform seed

    # 2. Initial_h_values interpolated to N

    except NameError:

        initial_h_values = np.ones(N) * 0.5
    if len(initial_h_values) != N:
        x_orig = np.linspace(0, 2, len(initial_h_values))
        x_target = np.linspace(0, 2, N)
        h_interp = np.interp(x_target, x_orig, initial_h_values)
        # Normalize to exact sum
        sum_interp = np.sum(h_interp) * dx
        h_interp = h_interp * (N/2) / sum_interp
    else:
        h_interp = initial_h_values.copy()
    # 3. Double-peak seed
    h_double = np.ones(N) * 0.1
    h_double[:int(0.3*N)] = 0.9
    h_double[int(0.7*N):] = 0.9
    # Normalize to exact sum
    sum_double = np.sum(h_double) * dx
    h_double = h_double * (N/2) / sum_double
    # Run optimization on each seed
    best_h = None
    best_c5 = float('inf')
    time_per_seed = 0.30 * budget_s
    for i, h_seed in enumerate([uniform_h, h_interp, h_double]):
        rest_time = budget_s - (time.time() - start)
        if rest_time <= time_per_seed:
            break
        h_opt, c5_opt = anneal(h_seed, time_per_seed, seed=seed + i)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
    # Polishing phase
    remaining_time = budget_s - (time.time() - start)
    if remaining_time > 10:
        # Freeze the temperature to 0 for deterministic hill climbing
        def anneal_polish(initial_h, time_budget):

            cur = initial_h.copy()
            best = cur.copy()
