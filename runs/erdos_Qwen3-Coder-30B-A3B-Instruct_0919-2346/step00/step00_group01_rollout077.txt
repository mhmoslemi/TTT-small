import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def circular_correlation(h):
    """Compute circular correlation using FFT"""
    n = len(h)
    pad = 2 * n
    h_padded = np.pad(h, (0, n), mode='constant')
    h_shifted = np.pad(h, (n, 0), mode='constant')
    corr = correlate(h_padded, h_shifted, mode='valid')
    return corr[:n]

def evaluate_c5(h):
    """Convert h vector to C5 bound"""
    n = len(h)
    corr = circular_correlation(h)
    min_corr = np.min(corr)
    c5 = 1 - 2 * min_corr / n
    return c5

def sdp_solve(n, seed=42):
    np.random.seed(seed)
    # Symmetric Toeplitz constraints
    h = cp.Variable(n)
    t = cp.Variable()
    # Trace constraint matrix
    constraints = [cp.sum(h) == n / 2]
    constraints += [0 <= h, h <= 1]
    # Correlation constraints
    for k in range(n):
        # Build matrix M_k: non-zero element at (i, i+k) and (i+k, i)
        M_k = np.zeros((n, n))
        for i in range(n):
            j = (i + k) % n
            M_k[i, j] = 1
            M_k[j, i] = 1
        constraints += [cp.trace(M_k @ cp.Variable((n, n), PSD=True)) >= t]
    # Objective
    prob = cp.Problem(cp.Maximize(t), constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status == cp.OPTIMAL:
            return h.value, t.value
    except:
        pass
    return None, None

def round_h(X, n):
    # Cholesky decomposition X = V V^T
    try:
        V = np.linalg.cholesky(X)
    except:
        # Fallback to eigendecomposition
        u, v = np.linalg.eigh(X)
        u = np.maximum(u, 1e-10)
        V = v @ np.diag(np.sqrt(u))
    r = V.shape[1]
    g = np.random.randn(r)
    scores = V.T @ g
    # Third quartile as threshold
    theta = np.quantile(scores, 0.75)
    h = (scores >= theta).astype(float)
    # Adjust to sum = n/2
    if np.sum(h) > n / 2:
        idx = np.argpartition(-scores, int(n / 2))[:int(n / 2)]
        h = np.zeros_like(h)
        h[idx] = 1
    elif np.sum(h) < n / 2:
        idx = np.argpartition(scores, int(n / 2))[:int(n / 2)]
        h = np.zeros_like(h)
        h[idx] = 1
    return h

def swap_hill_climb(h, n):
    def update_correlations(h, shifts, corr_vals):
        # Precompute all shifts' correlations
        corr_vals[:] = np.array([np.sum(h * np.roll(h, k)) for k in shifts])
    shifts = list(range(n))
    corr_vals = np.empty(n)
    update_correlations(h, shifts, corr_vals)
    min_corr = np.min(corr_vals)
    improving = True
    while improving:
        improving = False
        # Violations and further improves
        # Look for improvement through swaps
        for i in range(n):
            if h[i] == 1:
                for j in range(n):
                    if h[j] == 0:
                        # Try swap
                        h[i], h[j] = 0, 1
                        new_corr = np.array([np.sum(h * np.roll(h, k)) for k in shifts])
                        new_min = np.min(new_corr)
                        if new_min > min_corr:
                            min_corr = new_min
                            improving = True
                        else:
                            h[i], h[j] = 1, 0
    return h

def gradient_fine_tune(h, n, max_iter=100):
    def compute_grad(i, h, corr_vals, worst_k):
        # Gradient only w.r.t. if i is involved in worst_k or not.
        # It will be h[i+worst_k] + h[i-worst_k] (both taken mod n)
        return h[(i + worst_k) % n] + h[(i - worst_k) % n]  # 0 or 2 or 1

    # Direct optimization
    h_new = h.copy()
    for _ in range(max_iter):
        # Find worst k
        corr = np.array([np.sum(h_new * np.roll(h_new, k)) for k in range(n)])
        worst_k = np.argmin(corr)
        min_corr = corr[worst_k]
        # Gradient update
        grad = np.array([compute_grad(i, h_new, corr, worst_k) for i in range(n)])
        # Project gradient into valid [0,1] with fixed sum
        h_new = h_new + 0.01 * grad
        # Ensure bounds and sum
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n/2) / np.sum(h_new)
    return h_new

def solve_interval_model(n=200, m=6):
    # Objective function for interval structure: [a1,b1], [a2,b2], ...
    # With symmetry, only first m//2 intervals need to be optimized directly
    
    # Setup variables for first m//2 intervals
    # N = 2*(m//2) = m (for symmetric pairs)
    num_vars = m if m % 2 == 0 else m - 1  # m or m-1 variables
    
    bounds = [(0, 2)] * num_vars  # bounds on interval starts (a1,...)
    a_start = 0

    def interval_correlation(a_list):
        # a_list defines a1,... (other are derived by symmetry)
        intervals = []
        if m % 2 == 0:
            # Even m, m//2 intervals on each side.
            for i in range(m//2):
                a = a_list[i]
                b = a + (2 - a_list[m//2-1-i])  # minimise from the right
                intervals.append((a, b))
            intervals = intervals + intervals[::-1]  # Reflect
        else:
            # odd m, a1,...,a_m//2, a_m//2+1 = a_(m-1)//2 if exists etc.
            for i in range(m//2):
                a = a_list[i]
                b = a + (2 - a_list[m//2-1-i])  # symmetric
                intervals.append((a, b))
            # Middle one will be centered at 1
            intervals.append((1, 1))
            intervals = intervals + intervals[::-1]
        
        # Generate intervals as step function
        f = np.zeros(n)
        dx = 2.0 / n
        # Bin intervals into array by checking coverage of samples
        for a, b in intervals:
            start_idx = int(np.floor(a/dx))
            end_idx = int(np.ceil(b/dx))
            start_idx = max(0, start_idx)
            end_idx = min(n, end_idx)
            f[start_idx:end_idx] = 1.0
        return f

    def obj_func(a_list):
        h = interval_correlation(a_list)
        corr = circular_correlation(h)
        return 1.0 - 2 * np.min(corr) / n  # C5

    def grad_func(a_list):
        # Using finite difference (very coarse)
        eps = 1e-6
        grad = np.zeros_like(a_list)
        base_val = obj_func(a_list)
        for i in range(len(a_list)):
            a_list[i] += eps
            grad[i] = (obj_func(a_list) - base_val) / eps
            a_list[i] -= eps
        return grad

    # Grid search
    best_a = None
    best_c5 = np.inf
    grid_points = 200  # By trial and error turns out to be reasonable
    for _ in range(grid_points):
        a_list = np.random.uniform(0, 2, size=m // 2)
        # Ensure sum of widths = 1 (normalized area)
        y = np.sort(np.concatenate(([0], a_list, [2])))
        widths = np.diff(y)
        if np.sum(widths) > 0:
            a_list = (a_list - y[0]) / (y[-1] - y[0]) * 2
            if np.sum(np.diff(np.concatenate(([0], a_list, [2])))) > 0:
                try:
                    if m % 2 == 0:
                        mid = np.sum(a_list) / len(a_list)
                        a_list = np.clip(a_list, 0, 1)
                        if mid > 1.0:
                            a_list -= (mid - 1)
                        else:
                            a_list += 1 - mid
                    else:
                        a_list = np.clip(a_list, 0, 1) - 0.5
                        a_list = a_list + 1

                    c5 = obj_func(a_list)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_a = a_list.copy()
                except:
                    continue

    # SQP refinement
    res = minimize(obj_func, best_a, method='SLSQP', jac=grad_func, bounds=bounds, 
                   options={'maxiter': 50, 'ftol': 1e-5})
    if res.success:
        optimized = interval_correlation(res.x)
        return optimized, evaluate_c5(optimized)

    return None, np.inf
