from scipy.optimize import differential_evolution
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for given h."""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all k
    correlation = np.correlate(h, 1 - h, mode="full")
    # The full convolution result has length 2*n - 1
    # We are interested in k = 0, 1, ..., n-1 (for the positive shifts)
    # The peak value among all k values
    max_value = max(correlation)
    return max_value * dx
