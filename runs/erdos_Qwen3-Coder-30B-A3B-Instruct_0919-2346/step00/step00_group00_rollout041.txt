import numpy as np
import time
import random

def compute_c5(h):
    """Computes the C5 bound for a given step function h."""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return corr.max()

def propose_swap_and_update(h, corr, i, j):
    """Proposes a swap of h[i] and h[j], and updates correlation array in place."""
    a = h[i]
    b = h[j]
    delta_i = b - a
    
    n = len(h)
    # Create a copy of the correlation array for computing new values
    new_corr = corr.copy()
    
    # Loop over k not equal to i or j
    for k in range(n):
        if k == i or k == j:
            continue
        hk = h[k]
        # shift i->k: l = k - i
        idx1 = k - i + n - 1
        new_corr[idx1] += delta_i * (1 - hk)
        # shift k->i: l = i - k
        idx2 = i - k + n - 1
        new_corr[idx2] += hk * (-delta_i)  # = -hk*delta_i

        # contributions for j
        # shift j->k: l = k - j
        idx3 = k - j + n - 1
        new_corr[idx3] += -delta_i * (1 - hk)  # delta_j = -delta_i
        # shift k->j: l = j - k
        idx4 = j - k + n - 1
        new_corr[idx4] += hk * (delta_i)   # = hk*delta_i

    # handle cross pair i->j and j->i
    idx_ij = j - i + n - 1
    new_corr[idx_ij] += delta_i

    idx_ji = i - j + n - 1
    new_corr[idx_ji] -= delta_i

    # Now get the new maximum value
    new_max = new_corr.max()
    
    return new_corr, new_max

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    start = time.time()
    best_c5 = np.inf
    best_h = None
    best_n = None
    
    # Candidate n values: going for moderate resolution to avoid too many large integrations
    n_candidates = [60, 80, 100, 120, 140, 160, 180, 200]
    
    for n in n_candidates:
        if time.time() - start > budget_s - 1:
            break
        if n % 2 != 0:
            continue
            
        # set max iterations based on time and problem size
        time_left = budget_s - (time.time() - start)
        max_iters = max(2000, int(40000 / (n / 120)))
        
        # initialize a random binary h with exactly n/2 ones
        h = np.zeros(n, dtype=float)
        ones = n // 2
        h[np.random.choice(n, ones, replace=False)] = 1.0
        
        # initial correlation and C5
        corr = np.correlate(h, 1 - h, mode='full')
        cur_max = corr.max()
        cur_c5 = cur_max * (2.0 / n)
        cur_h = h.copy()
        
        # temperature schedule for simulated annealing 
        T = 0.05  # Initial temperature
        T_decay = 0.9995  # Cooling rate
        
        iter_no_improve = 0  # Counter for restarts
        max_no_improve = 5000  # Max iterations before restart
        
        for it in range(max_iters):
            now = time.time()
            if now - start > budget_s - 0.5:
                break
                
            # propose a swap: select indices with different values
            i = np.random.randint(n)
            j = np.random.randint(n)
            while h[i] == h[j]:
                i = np.random.randint(n)
                j = np.random.randint(n)
            
            # compute new correlation after proposed swap
            new_corr, new_max = propose_swap_and_update(h, corr, i, j)
            new_c5 = new_max * (2.0 / n)
            
            # accept criteria
            d = new_c5 - cur_c5
            if d < 0 or np.random.rand() < np.exp(-d / T):
                # accept the swap
                corr = new_corr
                cur_max = new_max
                cur_c5 = new_c5
                h[i], h[j] = h[j], h[i]  # Swap in h
                
                iter_no_improve = 0  # Reset counter
            else:
                iter_no_improve += 1
                # Do not swap back, just keep old h
            
            # update best solution
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                best_n = n
                
            # temperature decay
            T *= T_decay
            
            # Random restart if no improvements for a while
            if iter_no_improve > max_no_improve:
                h = np.zeros(n, dtype=float)
                ones = n // 2
                h[np.random.choice(n, ones, replace=False)] = 1.0
                corr = np.correlate(h, 1 - h, mode='full')
                cur_max = corr.max()
                cur_c5 = cur_max * (2.0 / n)
                iter_no_improve = 0  # Reset counter
            
            # Early exit condition
            if time.time() - start > budget_s - 0.5:
                break
                
        # End of local search for this n
        
    # If no solution found, fallback to initial_h_values or simple solution
    if best_h is None:
        try:
            from __main__ import initial_h_values
            best_h = initial_h_values.copy()
            best_n = len(best_h)
            best_c5 = compute_c5(best_h)
        except Exception:
            # use simple binary with 1s/0s alternating
            n = 120
            best_h = np.zeros(n)
            best_h[::2] = 1.0
            best_n = n
            best_c5 = compute_c5(best_h)
            
    return best_h, best_c5, best_n
