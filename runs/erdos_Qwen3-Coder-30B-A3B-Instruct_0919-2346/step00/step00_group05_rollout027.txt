import numpy as np
from scipy.optimize import minimize
from cvxpy import *
import time
import random
from math import floor

def evaluate_erdos_solution(h_values):
    """
    Compute C5 bound for a given h_values array.
    """
    h = np.array(h_values)
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(c) * dx
    return c5

def generate_random_binary(n_points):
    """Generate a random binary vector with exactly n/2 ones."""
    h = np.zeros(n_points)
    ones_indices = random.sample(range(n_points), n_points // 2)
    h[ones_indices] = 1
    return h

def project_onto_capped_simplex(h, target_sum):
    """Project h to the capped simplex: 0 <= h[i] <= 1, sum(h) = target_sum."""
    h = np.array(h)
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = target_sum - current_sum
    
    sorted_indices = np.argsort(h)[::-1]
    for i in sorted_indices:
        if diff <= 0:
            break
        can_add = min(1 - h[i], diff)
        h[i] += can_add
        diff -= can_add
    return h

def solve_milp_with_cbc(n_points, warm_start_h=None):
    """Solve the MILP using CBC."""
    h = Variable(n_points, integer=True)
    y = {}
    tau = Variable()
    
    # Constraints
    constraints = []
    
    # Add mass constraint
    constraints.append(sum(h) == n_points / 2)
    
    # Add bounds
    for i in range(n_points):
        constraints.append(h[i] >= 0)
        constraints.append(h[i] <= 1)
    
    # Fix first entry to break symmetry
    constraints.append(h[0] == 1)
    
    # Precalculate shifts
    for k in range(1, n_points):
        num_vars = n_points - k
        y[k] = Variable(num_vars)
        
        # Set constraints for y_k
        for j in range(num_vars):
            constraints.append(y[k][j] >= 0)
            constraints.append(y[k][j] <= h[j])
            constraints.append(y[k][j] <= h[j + k])
            constraints.append(y[k][j] >= h[j] + h[j + k] - 1)
            constraints.append(y[k][j] >= 0)
        
        # tau <= sum(y_k)
        constraints.append(tau <= sum(y[k]))
    
    # Objective
    objective = Maximize(tau)
    
    # Solve
    prob = Problem(objective, constraints)
    
    # Use CBC solver
    prob.solve(solver=CPLEX, verbose=False, max_seconds=300)
    
    if prob.status in ["optimal", "optimal_inaccurate"]:
        return h.value, tau.value
    else:
        return None, None
