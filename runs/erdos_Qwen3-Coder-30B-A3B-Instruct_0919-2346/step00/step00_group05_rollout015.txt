import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    """Evaluate C5 for given h values"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation: h convolution (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    # Extract the overlap integral values for all shifts
    # The relevant part of the correlation corresponds to the full overlap
    # We want the maximum of these values
    max_overlap = np.max(corr) * dx
    return max_overlap
