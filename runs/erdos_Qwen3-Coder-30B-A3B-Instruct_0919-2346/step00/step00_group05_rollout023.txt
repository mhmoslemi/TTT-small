import numpy as np
import time
import cvxpy as cp
from scipy.optimize import linprog

def project_onto_capped_simplex(v, S):
    """Project vector v onto the capped simplex: 0 <= h <= 1, sum h = S"""
    v = np.clip(v, 0, 1)
    v_sum = np.sum(v)
    if abs(v_sum - S) < 1e-10:
        return v
    
    # Binary search for the right lambda
    low, high = -2.0, 2.0
    tol = 1e-10
    while high - low > tol:
        mid = (low + high) / 2.0
        h = np.clip(v - mid, 0, 1)
        if np.sum(h) > S:
            low = mid
        else:
            high = mid
    return np.clip(v - (low + high)/2.0, 0, 1)

def coeffs_and_const(h, n):
    """Compute linearization coefficients and constants for all shifts."""
    # Precompute all required indices (sliding window)
    a_matrix = np.zeros((n, 2*n - 1))
    const_vec = np.zeros(2*n - 1)
    
    # k from -(n-1) to n-1
    for k in range(-(n-1), n):
        # Compute coefficient vector and constant
        a = np.zeros(n)
        c = 0.0
        for i in range(n):
            # Avoid index out-of-bounds by only including valid indices
            if 0 <= i + k < n:
                a[i] = 1.0 - h[i + k] - h[i]
                c += h[i] * h[i + k]
        a_matrix[:, k + n - 1] = a
        const_vec[k + n - 1] = c
    
    return a_matrix, const_vec

def solve_linear_program(a_matrix, const_vec, n, S):
    """Solve the linear program to find new h and t."""
    # Variables
    h = cp.Variable(n)
    t = cp.Variable(1)

    # Constraints for all shifts k
    constraints = [
        a_matrix @ h + const_vec <= t,
        cp.sum(h) == S,
        h >= 0,
        h <= 1,
        t >= 0,
        t <= 2
    ]

    # Objective
    objective = cp.Minimize(t)

    # Create and solve problem
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status == cp.OPTIMAL:
            h_sol = h.value
            t_sol = t.value
            return h_sol, t_sol
        else:
            # fallback to numpy linprog with bounds
            # This fallback aims for a very conservative solution
            A_eq = np.array([np.ones(n)])
            b_eq = np.array([S])
            bounds = [(0, 1)] * n
            # These constraints are extra: a_matrix @ h + const_vec <= t
            # We can just do simple bound tightening here for all shifts
            A_ub = -a_matrix
            b_ub = -const_vec
            # Reformulate to include t into optimization with one additional
            # bound which corresponds to t = 1, but we'll go with
            # minimize(u) such that all linearized inequalities hold
            # Let's just solve with primitive approach using scipy
            c = np.zeros(n + 1)
            c[-1] = 1
            A_ub = np.hstack([A_ub, np.ones((A_ub.shape[0], 1))])
            b_ub = b_ub.reshape(-1, 1).flatten()
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds + [(0, None)], method='highs')
            if res.success:
                sol_h = res.x[:-1]
                sol_t = res.x[-1]
                return sol_h, sol_t
            else:
                # worst attempt
                return np.ones(n) * S/n, S
    except Exception as e:
        print("error during cvxpy solve:", e)
        # Fallback to scipy
        c = np.zeros(n + 1)
        c[-1] = 1
        A_eq = np.array([np.ones(n)])
        b_eq = np.array([S])
        A_ub = -a_matrix # use transpose of matrix as many shifts
        b_ub = -const_vec
        A_ub = np.hstack([A_ub, np.ones((A_ub.shape[0], 1))])
        bounds = [(0., 1.)] * n + [(0, None)]
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
            if res.success:
                sol_h = res.x[:-1]
                sol_t = res.x[-1]
                return sol_h, sol_t
        except Exception:
            return np.ones(n) * S/n, S
    return np.ones(n) * S/n, S

def evaluate_erdos_solution(h, dx=2.0, max_shift=None):
    # Input h is already discretized: [h[0], h[1], ..., h[n-1]]
    n = len(h)
    # Convolution-like computation using correlation from left to right
    h_extended = np.pad(h, (0, n-1), mode='constant', constant_values=0)  # to allow full convolution up to n-1
    # For each shift k from 0 to n-1, compute integral
    max_val = 0.0
    for k in range(n):
        # h[i] * (1 - h[i+k])
        integral = sum(h[i] * (1 - h_extended[i + k]) for i in range(n))
        integral *= dx / float(n)  # dx * (1.0 / n) per step times dx / n
        max_val = max(max_val, integral)
    return max_val
