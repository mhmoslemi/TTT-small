import numpy as np
from scipy.optimize import minimize

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for a given h function"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all shifts
    correlation = np.correlate(h, 1 - h, mode="full")
    # The full correlation is of length 2*n_points - 1
    # The valid shifts are from -1 to 1, so we need the middle part
    mid = len(correlation) // 2
    # The effective range is from 0 to n_points-1, which
    # corresponds to shifts from -1 to 1 (excluding 0 if we are
    # doing full convolution)
    # Actually, let's compute directly:
    # k represents the shift corresponding to the convolution
    # we want to compute max_k ∫ h(x) (1 - h(x+k)) dx
    # for k = 0,1,2,...,n-1 and k = -1,-2,...,-(n-1)
    # But the correlation gives us the same.
    # The maximum should be over the positive shifts.
    # In fact, it's most accurate to take the maximum over all valid
    # shifts, which are in range [-n+1, n-1] after taking the full convolution
    max_corr = np.max(correlation) * dx
    return max_corr

def optimize_h_values(initial_h, n_points, max_time):
    """Optimize h values using scipy"""
    dx = 2.0 / n_points
    start_time = time.time()
    # Formulate the optimization problem
    # Variables: h[i], subject to:
    # 0 <= h[i] <= 1
    # sum(h) * dx = 1
    # So sum(h) = n_points / 2 exact
    
    # Objective: minimize max_k int h(x)(1-h(x+k))dx
    # This is min max_i corr_i where corr_i = int h(x)(1-h(x+i)) dx
    # For k in range [-n+1 to n-1] but one-sided
    # We optimize the largest of these integrals, which correspond 
    # to correlations.
    
    # Let's first get an idea of what the problem is doing.
    # Use a derived form:
    
    # Flatten the optimization variables
    def objective(h_vals):
        # h_vals is a 1D array of length n_points
        # Normalize to satisfy constraint sum(h) = n_points / 2
        h = h_vals.copy()
        h = h * (n_points / 2) / np.sum(h)
        # Evaluate the C5 bound (max overlap integral)
        return evaluate_erdos_solution(h)
    
    # Use the method of Lagrange Multipliers or a direct non-linear optimization
    # We will use COBYLA 
    # But first generate a good starting point from the provided initial values
    h_init = np.array(initial_h)
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Define bounds for h[i]: 0 <= h[i] <= 1
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Add constraint that sum h = n_points/2 (i.e., integral = 1)
    def constraint_sum(h_vals):
        return np.sum(h_vals) - n_points / 2
    
    # Here we make a simpler approach:
    # Sample random subspace of admissible h's and sample some good ones.
    
    # However, we can also run a trust-region method for continuous optimization:
    # But since there are many local optima, let's sample several
    best_h = h_init.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Try different points in a controlled way
    # Use a heuristic: refining the grid-structure of the template
    n_iterations = 0
    while time.time() - start_time < max_time - 1:
        # Perform local optimizations with different starting points
        
        # Permute to make a new starting point
        h_new = h_init.copy()
        # Small perturbation to avoid plateaus
        h_new += (np.random.rand(n_points) - 0.5) * 0.1
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Calibrate
        new_c5 = evaluate_erdos_solution(h_new)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
        
        n_iterations += 1
        # Use a breadth-first sample approach
        # After every 10 local search, try a specific perturbation
        
        if n_iterations % 10 == 0:
            # try a little random sampling with respect to the best so far
            h_rand = np.random.rand(n_points)
            h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
            c5_rand = evaluate_erdos_solution(h_rand)
            if c5_rand < best_c5:
                best_c5 = c5_rand
                best_h = h_rand.copy()
    
    # Final test
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):

    initial_h = initial_h_values(n_points)

    # Final clean-up

    # measure final value
