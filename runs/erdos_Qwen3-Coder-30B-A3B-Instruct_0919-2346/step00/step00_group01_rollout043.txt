def evaluate_erdos_solution(h_values):
    """Evaluate the C5 bound for given h values."""
    n = len(h_values)

    # Compute overlap integral for all shifts
    correlation = np.correlate(h_values, 1 - np.array(h_values), mode="full")
    # Only consider the valid shifts (k from 0 to 2)
    valid_correlations = correlation[n-1:]  # The last n elements contain valid shifts
    c5 = np.max(valid_correlations) * dx
    return c5
