from scipy.optimize import differential_evolution
import time

def evaluate_overlap(h):
    """Evaluate C5 = max_k int h(x)(1 - h(x+k)) dx"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode="full")
    # The full correlation has 2*n-1 elements
    # The valid k values are from -n+1 to n-1
    # We're interested in k from 0 to n-1 (positive shifts)
    # but also the negative shifts should be considered as well
    # Let's compute the actual C5 value properly by taking maximum
    # of relevant correlations
    max_corr = 0
    for i in range(len(corr)):
        k = i - (n - 1)
        if k >= 0:
            # Positive shift case
            corr_val = corr[i] * dx
            if corr_val > max_corr:
                max_corr = corr_val
        elif k < 0:
            # Negative shift case, but we want the shift count to be k
            # For negative k, we're looking at the right shift of h relative to 1-h
            corr_val = corr[i] * dx
            if corr_val > max_corr:
                max_corr = corr_val
    return max_corr
