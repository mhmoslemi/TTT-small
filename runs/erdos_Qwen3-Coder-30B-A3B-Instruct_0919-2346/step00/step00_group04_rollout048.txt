import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation with 1-h
    correlation = np.correlate(h, 1 - h, mode='full')
    # The maximum value of correlation * dx is our C5
    c5 = np.max(correlation) * dx
    return c5
