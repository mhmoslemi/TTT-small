import numpy as np
import time
import random
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    def evaluate(h):
        n = len(h)
        dx = 2.0 / n
        O = np.correlate(h, 1 - h, mode='full')
        return O.max() * dx

    def random_binary(n):
        h = np.random.choice([0, 1], size=n, p=[0.5, 0.5])
        while h.sum() != n // 2:
            if h.sum() > n // 2:
                idx = np.where(h == 1)[0]
                np.random.shuffle(idx)
                h[idx[0]] = 0
            else:
                idx = np.where(h == 0)[0]
                np.random.shuffle(idx)
                h[idx[0]] = 1
        return h

    def thue_morse(n):
        if (n & (n - 1)) != 0:
            raise ValueError("Thue-Morse only for powers of 2")
        h = np.zeros(n, dtype=int)
        for i in range(n):
            h[i] = bin(i).count('1') & 1
        # Adjust to have exactly n/2 ones
        if h.sum() != n // 2:
            # Find indices to flip
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if h.sum() > n // 2:
                to_flip = h.sum() - n // 2
                idx = np.random.choice(ones, to_flip, replace=False)
                h[idx] = 0
            else:
                to_flip = n // 2 - h.sum()
                idx = np.random.choice(zeros, to_flip, replace=False)
                h[idx] = 1
        return h

    def generate_candidates(n):
        candidates = []
        # Thue-Morse if power of 2
        if (n & (n - 1)) == 0:
            try:
                h = thue_morse(n)
                candidates.append(h.copy())
            except:
                pass
        # Random
        for _ in range(2):
            h = random_binary(n)
            candidates.append(h.copy())
        # Interpolated initial_h_values
        if 'initial_h_values' in globals() and initial_h_values is not None:
            base = np.array(initial_h_values, dtype=float)
            x_old = np.linspace(0, 2, len(base), endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            h_interp = np.interp(x_new, x_old, base)
            h_bin = (h_interp > 0.5).astype(int)
            # adjust to exactly sum n/2
            cur = h_bin.sum()
            target = n // 2
            if cur > target:
                idx = np.where(h_bin == 1)[0]
                np.random.shuffle(idx)
                h_bin[idx[:cur-target]] = 0
            elif cur < target:
                idx = np.where(h_bin == 0)[0]
                np.random.shuffle(idx)
                h_bin[idx[:target-cur]] = 1
            candidates.append(h_bin)
        return candidates

    def local_search(h, time_limit):
        n = len(h)
        S = n // 2
        start = time.time()
        current_val = evaluate(h)
        best_h = h.copy()
        best_val = current_val
        temp_start = 0.1
        temp_end = 1e-5
        max_iters = int(1e7)
        
        ones = list(np.flatnonzero(h))
        zeros = list(np.flatnonzero(1-h))
        for it in range(max_iters):
            if time.time() - start > time_limit:
                break
            temp = temp_start * (temp_end / temp_start) ** (it / max_iters)
            oi = random.randint(0, len(ones)-1)
            oj = random.randint(0, len(zeros)-1)
            i = ones[oi]
            j = zeros[oj]
            h[i], h[j] = 0, 1
            new_val = evaluate(h)
            delta = new_val - current_val
            if delta <= 0 or random.random() < math.exp(-delta / temp):
                current_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best_h = h.copy()
            else:
                h[i], h[j] = 1, 0
        return best_h, best_val

    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    best_n = None
    lengths = [64, 80, 96, 112, 128, 144, 160, 176, 192, 224, 256, 320, 384]
    lengths = [n for n in lengths if n <= 800]

    candidates_eval = []
    for n in lengths:
        for i, h in enumerate(generate_candidates(n)):
            if eval_time := time.time() - start_time > budget_s * 0.8:
                break
            c5 = evaluate(h)
            candidates_eval.append((c5, h, n))
    # Sort by c5
    candidates_eval.sort()
    # Run local search for top few
    top_k = min(5, len(candidates_eval))
    time_per_search = (budget_s * 0.95 - (time.time() - start_time)) / top_k
    for c5, h, n in candidates_eval[:top_k]:
        if time.time() - start_time > budget_s * 0.95:
            break
        best_h_search, best_c5_search = local_search(h, time_per_search)
        if best_c5_search < best_c5:
            best_c5 = best_c5_search
            best_h = best_h_search.copy()
            best_n = n
    return (best_h.tolist(), best_c5, best_n)
