import numpy as np
import time
import random

def evaluate_c5(h):
    """Evaluate the maximum overlap integral for the given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h) using full mode
    corr = np.correlate(h, 1 - h, mode='full')
    # Take maximum and scale by dx
    max_corr = np.max(corr)
    c5 = max_corr * dx
    return c5

def initial_h_values(n_points):
    """Generate initial step function h with n_points samples"""
    # Start with the known record solution (n=70) and interpolate to n_points
    # Using the built-in initial_h_values if available
    
    # For this implementation, we'll use a basic heuristic:
    # Construct a sequence with 0s and 1s to balance overlap
    # Sample initial sequence from n=70 if available (but we'll use 0/1 construction here)
    import math
    # For now just construct a reasonable binary pattern
    h = np.zeros(n_points)
    ones_count = n_points // 2
    if ones_count * 2 == n_points:  # even
        # Distribute ones in a way to avoid large gaps
        for i in range(ones_count):
            h[i * 2] = 1.0
    
    if np.sum(h) != ones_count:
        # Adjust to ensure sum equals exactly ones_count
        current_sum = np.sum(h)
        diff = int(ones_count - current_sum)
        if diff > 0:
            # Add 1 to some indices
            indices = np.where(h == 0)[0]
            for i in range(diff):
                if i < len(indices):
                    h[indices[i]] = 1.0
        elif diff < 0:
            # Remove 1 from some indices
            indices = np.where(h == 1)[0]
            for i in range(-diff):
                if i < len(indices):
                    h[indices[i]] = 0.0
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Find improved upper bound for Erdős minimum overlap problem constant C₅"""
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 200  # More samples to improve bound
    dx = 2.0 / n_points
    target_c5 = 0.3807  # target bound
    
    # Determine target integer threshold for correlation counts
    # Since C5 = max_ℓ sum_i h_i * (1 - h_{i+ℓ}) * dx
    # We want max_ℓ sum_i h_i * (1 - h_{i+ℓ}) <= target_c5 / dx * n/2 = target_c5 * n/2
    target_count = int(np.floor(target_c5 / dx * n_points / 2))
    
    # Initialize best solution
    best_h = None
    best_c5 = float('inf')
    
    # Run for the given budget
    start_time = time.time()
    iteration = 0
    
    # Initialize with random binary sequence
    h = initial_h_values(n_points)
    
    # Ensure sum equals exactly n/2
    current_sum = np.sum(h)
    if current_sum < n_points / 2:
        # Add more ones
        indices = np.where(h == 0)[0]
        needed = int(n_points / 2 - current_sum)
        for i in range(needed):
            if i < len(indices):
                h[indices[i]] = 1
    elif current_sum > n_points / 2:
        # Remove some ones
        indices = np.where(h == 1)[0]
        excess = int(current_sum - n_points / 2)
        for i in range(excess):
            if i < len(indices):
                h[indices[i]] = 0
    
    current_c5 = evaluate_c5(h)
    if current_c5 < best_c5:
        best_h, best_c5 = h.copy(), current_c5
        
    # Simulated Annealing with binary swaps
    T = 0.1
    cooling_rate = 0.9999
    min_T = 1e-6
    
    # For initial phase, we also occasionally try different initialization
    iteration_reset = 10000
    
    while time.time() - start_time < budget_s:
        iteration += 1
        
        if iteration % iteration_reset == 0:
            # Occasionally reinitialize
            h = initial_h_values(n_points)
            current_sum = np.sum(h)
            if current_sum != n_points / 2:
                diff = int(n_points / 2 - current_sum)
                if diff > 0:
                    indices = np.where(h == 0)[0]
                    for i in range(min(diff, len(indices))):
                        h[indices[i]] = 1
                elif diff < 0:
                    indices = np.where(h == 1)[0]
                    for i in range(min(-diff, len(indices))):
                        h[indices[i]] = 0
            current_c5 = evaluate_c5(h)
            
        # Perform a random swap
        # Find 1's and 0's indices
        ones_indices = np.where(h == 1)[0]
        zeros_indices = np.where(h == 0)[0]
        
        if len(ones_indices) > 0 and len(zeros_indices) > 0:
            # Pick random indices to swap
            i = np.random.choice(ones_indices)
            j = np.random.choice(zeros_indices)
            
            # Perform the swap
            new_h = h.copy()
            new_h[i], new_h[j] = new_h[j], new_h[i]
            
            # Check new solution
            new_c5 = evaluate_c5(new_h)
            
            # Accept with probability based on temperature and difference
            delta = new_c5 - current_c5
            if delta < 0 or np.random.rand() < np.exp(-delta / T):
                h = new_h
                current_c5 = new_c5
                
                # Update best if better
                if current_c5 < best_c5:
                    best_h, best_c5 = h.copy(), current_c5
            
        # Reduce temperature
        T = max(min_T, T * cooling_rate)
        
        # Occasionally check progress
        if iteration % 1000 == 0:
            if best_c5 <= target_c5:
                break
    
    if best_h is None:
        # Fallback to the evaluated initial solution
        best_h = initial_h_values(n_points)
        best_c5 = evaluate_c5(best_h)
        
    return (list(best_h), best_c5, n_points)
