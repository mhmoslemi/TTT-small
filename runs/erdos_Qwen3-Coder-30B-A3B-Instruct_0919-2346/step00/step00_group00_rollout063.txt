from scipy.optimize import differential_evolution
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral for given h"""
    # Compute correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum value represents the worst-case overlap
    max_overlap = np.max(corr) * dx
    return max_overlap
