import numpy as np
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    n_points = 200  # Using larger size for better optimization
    dx = 2.0 / n_points
    target_bound = 0.379
    
    # Initialize the function with a simple pattern
    h_values = np.zeros(n_points)
    h_values[:n_points//2] = 1.0  # Start with a simple binary pattern
    
    # Randomly shuffle to get a different starting point
    indices = list(range(n_points))
    random.shuffle(indices)
    h_values = np.zeros(n_points)
    h_values[indices[:n_points//2]] = 1.0
    
    # Evaluate initial solution
    def evaluate_c5(h):
        correlation = np.correlate(h, 1 - h, mode="full")
        # Only consider non-zero shifts (excluding the center)
        valid_correlations = correlation[n_points-1:]
        max_overlap = np.max(valid_correlations)
        return max_overlap * dx
    
    best_h = h_values.copy()
    best_c5 = evaluate_c5(best_h)
    
    start_time = time.time()
    
    # Simulated Annealing parameters
    temperature = 1.0
    min_temperature = 1e-6
    cooling_rate = 0.9999
    max_iterations = 500000
    
    iteration = 0
    while time.time() - start_time < budget_s * 0.95 and iteration < max_iterations:
        # Create a neighbor solution by swapping two elements
        i = random.randint(0, n_points - 1)
        j = random.randint(0, n_points - 1)
        
        # Ensure we pick different indices
        while j == i:
            j = random.randint(0, n_points - 1)
        
        # Check if swap is valid (both positions have correct value)
        if (h_values[i] == 1 and h_values[j] == 0) or (h_values[i] == 0 and h_values[j] == 1):
            # Make a copy of current state
            new_h = h_values.copy()
            new_h[i], new_h[j] = new_h[j], new_h[i]  # Swap
            
            # Evaluate new solution
            new_c5 = evaluate_c5(new_h)
            
            # Accept with probability based on temperature
            delta = new_c5 - best_c5
            if delta <= 0 or random.random() < np.exp(-delta / temperature):
                h_values = new_h
                if new_c5 < best_c5:
                    best_h = h_values.copy()
                    best_c5 = new_c5
            
        # Cool down temperature
        temperature *= cooling_rate
        iteration += 1
        
        # Occasionally randomize if we're stuck
        if iteration % 10000 == 0 and iteration > 0:
            if best_c5 <= target_bound:
                break
            random.shuffle(h_values)
            # Re-evaluate
            current_c5 = evaluate_c5(h_values)
            if current_c5 < best_c5:
                best_h = h_values.copy()
                best_c5 = current_c5
    
    return (best_h, best_c5, n_points)
