def evaluate_c5(h):
    """Evaluate C5 bound from step function h."""
    n = len(h)
    # Compute circular correlation
    h_extended = np.concatenate([h, h])
    corr = np.correlate(h_extended, 1 - h, mode='valid')
    min_corr = np.min(corr)
    return 1.0 - (2.0 / n) * min_corr

def sdp_solve(n=n_points, seed=42):
    """Solve SDP relaxation with CVXPY."""
    np.random.seed(seed)
    # Create variables
    h = Variable(n)
    X = Variable((n, n), symmetric=True)
    t = Variable()
    
    # Matrix constraints
    M = []
    for k in range(n):
        Mk = np.zeros((n, n))
        for i in range(n):
            Mk[i, (i + k) % n] = 1
            Mk[(i + k) % n, i] = 1
        M.append(Mk)
    
    # Constraints
    constraints = [X == h @ h.T,  # This is not valid in CVXPY
                   X >> 0,        # Positive semidefinite
                   h >= 0,
                   h <= 1,
                   sum(h) == n/2,
                   [trace(M[k] @ X) >= t for k in range(n)]
                   ]
    
    # Objective
    objective = Maximize(t)
    
    # Solve
    prob = Problem(objective, constraints)
    try:
        prob.solve(solver=ECOS, verbose=False)
        if prob.status in ["optimal", "optimal_inaccurate"]:
            return h.value, t.value
        else:
            return None, None
    except:  # Catch any solver issues
        return None, None

def round_h(h_sdp, seed=42):
    """Round SDP solution using random hyperplane."""
    np.random.seed(seed)
    if h_sdp is None:
        return None
    V = np.random.randn(len(h_sdp), 10)
    h_rounded = V.T @ np.random.randn(10) > 0
    h_rounded = h_rounded.astype(float)
    ones_count = np.sum(h_rounded)
    target = int(np.floor(n_points / 2))
    
    if ones_count != target:
        # Project onto target count
        scores = V.T @ np.random.randn(10)
        sorted_idxs = np.argsort(-scores)
        if ones_count > target:
            # Need to set some 1s to 0s
            indices_to_flip = sorted_idxs[target:ones_count]
            h_rounded[indices_to_flip] = 0
        else:
            # Need to set some 0s to 1s
            indices_to_flip = sorted_idxs[ones_count:target]
            h_rounded[indices_to_flip] = 1
    
    return np.clip(h_rounded, 0, 1)

def swap_hill_climb(h, max_iter=500):
    """Perform hill climb by swapping 0s and 1s."""
    h = h.copy()
    n = len(h)
    # Precompute full autocorrelations
    corr = np.correlate(np.concatenate([h, h]), 1 - h, mode='valid')
    corr = corr[:n]
    min_corr = np.min(corr)
    
    for _ in range(max_iter):
        best_swap = None
        best_improvement = 0
        for i in range(n):
            for j in range(n):
                if h[i] != h[j]:
                    # Calculate change in min correlation
                    if i < j:
                        old_i, old_j = h[i], h[j]
                    else:
                        old_i, old_j = h[j], h[i]
                    if old_i == 1 and old_j == 0:
                        # swap 1 and 0
                        # compute impact on shifted correlations
                        h_temp = h.copy()
                        h_temp[i], h_temp[j] = h_temp[j], h_temp[i]
                        corr_temp = np.correlate(np.concatenate([h_temp, h_temp]), 1 - h_temp, mode='valid')
                        corr_temp = corr_temp[:n]
                        new_min = np.min(corr_temp)
                        improvement = min_corr - new_min
                        if improvement > best_improvement:
                            best_improvement = improvement
                            best_swap = (i, j)
        if best_swap:
            i, j = best_swap
            h[i], h[j] = h[j], h[i]
            corr = np.correlate(np.concatenate([h, h]), 1 - h, mode='valid')
            corr = corr[:n]
            min_corr = np.min(corr)
        else:
            break
    return h

def gradient_fine_tune(h, max_iter=500, eta=0.01):
    """Fine tune using projected gradient ascent on the minimum autocorrelation."""
    h = h.copy()
    n = len(h)
    corr = np.correlate(np.concatenate([h, h]), 1 - h, mode='valid')
    corr = corr[:n]
    min_corr_idx = np.argmin(corr)
    min_corr = corr[min_corr_idx]
    
    for _ in range(max_iter):
        # compute gradient
        grad = np.zeros(n)
        k_star = min_corr_idx
        for i in range(n):
            j = (i - k_star) % n
            grad[i] += 1 - h[j]
            grad[j] += 1 - h[i]
        
        # Note: This is coarse, but sufficient for demonstration
        h_new = h + eta * grad
        # Project to [0,1]
        h_new = np.clip(h_new, 0, 1)
        # Enforce sum constraint
        h_new = h_new * (n / 2) / np.sum(h_new)
        h = h_new
        corr = np.correlate(np.concatenate([h, h]), 1 - h, mode='valid')
        corr = corr[:n]
        min_corr_idx = np.argmin(corr)
        min_corr = corr[min_corr_idx]
        if eta < 1e-8:
            break
        eta *= 0.95  # Decay step size
    return h

def interval_optimize():
    """Optimize a step function defined by intervals."""
    def compute_autocorrelation_from_intervals(a, b, len_interval):
        """Compute the minimum correlation for given intervals."""
        # Compute overlaps for all shifts k
        n = n_points
        # First, create an indicator function from intervals
        h = np.zeros(n)
        dx = 2.0 / n
        for i in range(n):
            x = i * dx
            for j in range(len_interval):
                if a[j] <= x < b[j]:
                    h[i] = 1
                    break
        # Now compute autocorrelation
        h_extended = np.concatenate([h, h])
        corr = np.correlate(h_extended, 1 - h, mode='valid')
        min_corr = np.min(corr)
        return min_corr
    
    # Parameters as endpoints
    best_c5 = np.inf
    best_h = np.zeros(n_points)
    # Generate good guesses
    np.random.seed(42)
    for trial in range(n_interval_trials):
        a = np.sort(np.random.uniform(0, 1, n_interval // 2)) * 2
        b = a + np.random.uniform(0.1, 0.3, n_interval // 2)
        # enforce symmetry
        b = 2 - b[::-1]
        a = a[::-1]
        # enforce area constraint
        if np.sum(b - a) != 1:
            scale = 1 / np.sum(b - a)
            a = a * scale
            b = b * scale
        # Compute correlation
        min_corr = compute_autocorrelation_from_intervals(a, b, n_interval)
        c5 = 1 - (2.0 / n_points) * min_corr
        if c5 < best_c5:
            best_c5 = c5
            # generate final h
            h = np.zeros(n_points)
            dx = 2.0 / n_points
            for i in range(n_points):
                x = i * dx
                for j in range(n_interval // 2):
                    if a[j] <= x < b[j] or a[-(j+1)] <= x < b[-(j+1)]:
                        h[i] = 1
                        break
            best_h = h.copy()
    return best_h, best_c5

    best_h_values = h_initial.copy()

    sdp_count = 0
    for i in range(n_sdp_restarts):

        if h_sdp is None:
            continue
        for j in range(n_roundings_per_sdp):

                break
            h_rounded = round_h(h_sdp, seed=1000*i+j)
            if h_rounded is None:
                continue
            h_swapped = swap_hill_climb(h_rounded)
            h_grad = gradient_fine_tune(h_swapped)
            c5 = evaluate_c5(h_grad)

                best_c5_bound = c5
                best_h_values = h_grad.copy()
        sdp_count += 1
    
    # Phase 2: Interval heuristic
    if time.time() - start_time < budget_s - 20:
        h_interval, c5_interval = interval_optimize()
        if c5_interval < best_c5_bound:
            best_c5_bound = c5_interval
            best_h_values = h_interval
    
    # Ensure we return valid result
    if best_h_values is None:
        return h_initial, evaluate_c5(h_initial), n_points
    return best_h_values, best_c5_bound, n_points
