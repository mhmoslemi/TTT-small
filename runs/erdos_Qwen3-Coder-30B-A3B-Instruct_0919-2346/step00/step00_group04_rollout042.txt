import numpy as np
import time
import random

def c5_bound(h):
    n = len(h)
    dx = 2.0 / n
    O = np.correlate(h, 1 - h, mode='full')
    return O.max() * dx

def random_binary(n):
    h = np.random.choice([0, 1], size=n)
    while np.sum(h) != n // 2:
        if np.sum(h) > n // 2:
            idx = np.where(h == 1)[0]
            np.random.shuffle(idx)
            h[idx[0]] = 0
        else:
            idx = np.where(h == 0)[0]
            np.random.shuffle(idx)
            h[idx[0]] = 1
    return h

def thue_morse(n):
    if (n & (n - 1)) != 0:
        raise ValueError("n must be a power of 2 for Thue-Morse")
    h = np.zeros(n, dtype=int)
    for i in range(n):
        # Count number of 1s in binary representation
        count = bin(i).count('1')
        h[i] = count % 2
    return h

def legendre_sequence(p):
    # Generate Legendre sequence of length p
    if p % 2 == 0:
        raise ValueError("p must be odd for Legendre sequence")
    h = np.zeros(p, dtype=int)
    for i in range(p):
        # Legendre symbol (i/p) for quadratic residues
        # We put 1 if i is a quadratic residue mod p, 0 if i == 0, and -1 for non-residue
        if i == 0:
            h[i] = 0
        else:
            # Quadratic residue mod p
            res = pow(i, (p - 1) // 2, p) == 1
            h[i] = 1 if res else 0
    # Ensure sum is exactly p//2
    current_sum = np.sum(h)
    if current_sum > p // 2:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:current_sum - p // 2]] = 0
    elif current_sum < p // 2:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:p // 2 - current_sum]] = 1
    return h

def interpolate_from_initial(n):
    # Replicate initial_h_values to create h of length n
    if 'initial_h_values' not in globals():
        raise ValueError("initial_h_values not available")
    base = np.array(initial_h_values)
    x_old = np.linspace(0, 2, len(base), endpoint=False)
    x_new = np.linspace(0, 2, n, endpoint=False)
    h_interp = np.interp(x_new, x_old, base)
    h_bin = (h_interp > 0.5).astype(int)
    # Adjust to exact sum
    current_sum = np.sum(h_bin)
    target = n // 2
    if current_sum > target:
        idx = np.where(h_bin == 1)[0]
        np.random.shuffle(idx)
        h_bin[idx[:current_sum - target]] = 0
    elif current_sum < target:
        idx = np.where(h_bin == 0)[0]
        np.random.shuffle(idx)
        h_bin[idx[:target - current_sum]] = 1
    return h_bin

def local_search(h, time_limit, max_iter=None):
    n = len(h)
    S = n // 2
    current_val = c5_bound(h)
    best_h = h.copy()
    best_val = current_val
    start = time.time()
    it = 0
    if max_iter is None:
        max_iter = 1000000
    temp_start = 0.1
    temp_end = 1e-5
    while time.time() - start < time_limit and it < max_iter:
        it += 1
        # Temperature schedule
        temp = temp_start * (temp_end / temp_start) ** (it / max_iter)
        # Choose swap
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            continue
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # Swap
        h[i], h[j] = 0, 1
        val = c5_bound(h)
        delta = val - current_val
        if delta <= 0 or np.random.rand() < np.exp(-delta / temp):
            current_val = val
            if val < best_val:
                best_val = val
                best_h = h.copy()
        else:
            # Revert
            h[i], h[j] = 1, 0
    return best_h, best_val

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start = time.time()
    best_h = None
    best_c5 = float('inf')
    best_n = None
    
    # Candidate lengths
    lengths = [64, 80, 96, 112, 128, 144, 160, 176, 192, 224, 256, 320, 384, 448, 512]
    # Pre-filter to keep n <= 800
    lengths = [n for n in lengths if n <= 800 and n % 2 == 0]
    
    # Trial candidates
    candidates = []
    for n in lengths:
        # Use interpolated initial_h_values
        try:
            h = interpolate_from_initial(n)
            candidates.append((h, n))
        except Exception:
            pass
        # Thue-Morse if n is power of 2
        if (n & (n - 1)) == 0:
            try:
                h = thue_morse(n)
                candidates.append((h, n))
            except Exception:
                pass
        # Legendre for prime n (if prime)
        if n > 2 and all(n % i != 0 for i in range(2, int(n**0.5)+1)):
            try:
                h = legendre_sequence(n)
                candidates.append((h, n))
            except Exception:
                pass
        for _ in range(2):
            h = random_binary(n)
            candidates.append((h, n))
    
    # Evaluate all candidates
    evaluated = []
    for h, n in candidates:
        if time.time() - start > 0.9 * budget_s:
            break
        c5 = c5_bound(h)
        evaluated.append((c5, h, n))
    evaluated.sort()
    
    # Improve top candidates
    time_per_candidate = (budget_s * 0.9) / min(len(evaluated), 5)
    
    for c5, h, n in evaluated[:5]:
        if time.time() - start > 0.9 * budget_s:
            break
        improved_h, improved_c5 = local_search(h, time_per_candidate)
        if improved_c5 < best_c5:
            best_c5 = improved_c5
            best_h = improved_h.copy()
            best_n = n
        if best_c5 <= 0.3807:
            break

    if best_h is None:
        # fallback to initial solution
        return interpolate_from_initial(128), 0.38092, 128
    
    # Final check
    final_c5 = c5_bound(best_h)
    return best_h, final_c5, best_n
