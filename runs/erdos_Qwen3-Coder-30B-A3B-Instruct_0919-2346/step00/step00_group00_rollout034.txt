import numpy as np
import time
from scipy.signal import convolve

def project_to_simplex(h, target_sum):
    """Project h onto the capped simplex with sum = target_sum"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Waterfilling algorithm
    sorted_h = np.sort(h)[::-1]
    cumsum = np.cumsum(sorted_h)
    n = len(h)
    # Find the break point
    idx = -1
    for i in range(n):
        if sorted_h[i] * (i + 1) - cumsum[i] > target_sum - cumsum[i]:
            idx = i
            break
    if idx == -1:
        # Solution is within [0,1] and satisfies constraint
        return np.clip(h, 0, 1)
    else:
        # Compute lambda
        if idx == 0:
            lambda_val = (target_sum - cumsum[0]) / (n - 1)
        else:
            lambda_val = (target_sum - cumsum[idx - 1]) / (n - idx)
        h_proj = np.maximum(0, np.minimum(1, h - lambda_val))
        return h_proj

def compute_grad(h, alpha, dx):
    """Compute gradient of log-sum-exp approximation"""
    n = len(h)
    # Compute O_l(h) for all shifts
    overlaps = np.zeros(2 * n - 1)
    for l in range(1 - n, n):
        # Compute O_l(h) = dx * sum h[i] * (1 - h[i+l])
        if l >= 0:
            # Reverse shift for left part
            left_part = h[:n-l]
            right_part = h[l:]
            summand = np.dot(left_part, 1 - right_part)
        else:
            # l < 0, so shift to the right
            right_part = h[-l:]
            left_part = h[:n+l]
            summand = np.dot(left_part, 1 - right_part)
        overlaps[l + n - 1] = summand

    # Compute weights (log-sum-exp softmax)
    exp_overlaps = np.exp(alpha * overlaps)
    w = exp_overlaps / np.sum(exp_overlaps)
    
    # Evaluate the convolution parts
    w_rev = w[::-1]  # flip weights
    # convolve w and (1-h) with full mode
    con1 = convolve(w, 1 - h, mode='full')
    con2 = convolve(w_rev, h, mode='full')
    # Slice to proper length
    grad = con1[:n] - con2[:n]

    return grad

def evaluate_c5(h, dx):
    """Evaluate exact value of C5 using np.correlate"""
    corr = np.correlate(h, 1 - h, mode='full')
    # take max of the full cross-correlation
    return dx * np.max(corr)
