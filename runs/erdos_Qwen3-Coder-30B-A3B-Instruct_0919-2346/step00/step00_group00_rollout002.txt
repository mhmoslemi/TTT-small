import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_erdos_solution(h_values):
    """Evaluate the Erdős constant for given step function h"""
    h = np.array(h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute all correlations for k from 0 to 2
    correlations = np.correlate(h, 1 - h, mode="full")
    # Only take the relevant part (the overlap integrals)
    c5 = np.max(correlations[len(h)-1:])
    return c5 * dx

def initial_h_values():
    """Return the initial construction for the problem"""
    n = 45
    h = np.zeros(n)
    dx = 2.0 / n
    # Create a simple step function
    h[:n//2] = 1.0
    # Normalize the integral to 1
    h = h / (np.sum(h) * dx)
    return h.tolist()

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 45  # Starting with the initial construction
    dx = 2.0 / n_points
    
    # Initialize with the known solution
    h = np.array(initial_h_values())
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Define objective function
    def objective(h_vals):
        h = np.array(h_vals)
        # Ensure integral constraint
        integral = np.sum(h) * dx
        if abs(integral - 1.0) > 1e-6:
            # penalty for violating integral constraint
            penalty = 1000 * abs(integral - 1.0)
        else:
            penalty = 0
        # Compute C5
        c5 = evaluate_erdos_solution(h_vals)
        return c5 + penalty
    
    # Bounds for h[i]: [0, 1]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint for integral
    def constraint_integral(h_vals):
        h = np.array(h_vals)
        return np.sum(h) * dx - 1.0
    
    cons = {'type': 'eq', 'fun': constraint_integral}
    
    # Optimization parameters
    max_iter = 500
    current_iter = 0
    
    # Use differential evolution with local refinement
    while time.time() - start_time < budget_s - 5:  # leave 5s buffer
        # Differential evolution
        result_de = differential_evolution(
            objective,
            bounds,
            maxiter=200,
            popsize=10,
            seed=seed + current_iter,
            disp=False,
            polish=False  # We'll polish manually
        )
        
        if result_de.success:
            h_opt = result_de.x
            c5_opt = evaluate_erdos_solution(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                if best_c5 < 0.379:
                    break
                    
            # Local refinement
            res_local = minimize(
                objective,
                h_opt,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 100}
            )
            
            if res_local.success:
                h_refined = res_local.x
                c5_refined = evaluate_erdos_solution(h_refined)
                if c5_refined < best_c5:
                    best_c5 = c5_refined
                    best_h = h_refined.copy()
                    if best_c5 < 0.379:
                        break
                        
        current_iter += 1
        
        # Random perturbations to escape local optima
        if np.random.rand() < 0.3:  # 30% chance to perturb
            h_pert = best_h.copy()
            indices = np.random.choice(n_points, size=max(1, n_points//10), replace=False)
            for i in indices:
                h_pert[i] = np.clip(h_pert[i] + np.random.normal(0, 0.05), 0, 1)
            # Renormalize
            integral = np.sum(h_pert) * dx
            h_pert = h_pert * (1.0 / integral)
            c5_new = evaluate_erdos_solution(h_pert)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_pert.copy()
                if best_c5 < 0.379:
                    break
                    
    return (best_h.tolist(), best_c5, n_points)
