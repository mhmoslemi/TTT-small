import numpy as np
from scipy.optimize import differential_evolution, minimize
import time
from math import isclose

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C_5 for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h) using full mode
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take maximum value and scale by dx
    c5 = np.max(correlation) * dx
    return c5

def setup_initial_solution(n_points):
    """Setup initial solution using the provided initial_h_values or create a simple one."""
    try:
        # Try to use the initial_h_values if available
        h_init = np.array(initial_h_values)
        if len(h_init) == n_points:
            return h_init.copy()
    except:
        pass
    
    # Else, create a reasonable initial construction
    # Simple case: step function at center
    h = np.zeros(n_points)
    center = n_points // 2
    width = n_points // 4
    h[center - width:center + width] = 1.0
    # Normalize
    h = h * (n_points / 2) / np.sum(h)
    return h

def objective(h):
    """Objective function to minimize."""
    return evaluate_erdos_solution(h)

def constraint_eq(h):
    """Equality constraint ensuring integral is 1."""
    n_points = len(h)
    dx = 2.0 / n_points
    return np.sum(h) * dx - 1.0

def constraint_bounds(h):
    """Constraint for upper and lower bounds of h."""
    return np.min(h) - 0.0, 1.0 - np.max(h)

def optimize_h(n_points, time_budget_s):
    """Optimize h to minimize C5 with given constraints using differential evolution."""
    # Setup the bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Track best solution
    best_c5 = float('inf')
    best_h = None
    
    # Start timing
    start_time = time.time()
    
    # Sample initial solution to get bounds on C5
    h0 = setup_initial_solution(n_points)
    c0 = objective(h0)
    if c0 < best_c5:
        best_c5 = c0
        best_h = h0.copy()
        
    # Use scipy's differential evolution with callbacks to check time
    def callback(xk, convergence):
        # Recompute and see if we are better
        current_c5 = objective(xk)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = xk.copy()
        return False  # continue optimization

    # Constraints: just equality on integral
    constraints = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        result = differential_evolution(
            objective,
            bounds,
            constraints=constraints,
            seed=42,
            maxiter=1000,
            popsize=15,
            mutation=(0.5, 1.0),
            recombination=0.7,
            atol=1e-8,
            callback=callback,
            disp=False
        )
        
        if result.success:
            final_c5 = objective(result.x)
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = result.x.copy()
                
    except Exception as e:
        print(f"Error in optimization: {e}")
        
    # For cases where diff_evo didn't work, try trust-constr
    try:
        # Try a few iterations using trust-constr with bounds
        constraints = [constraints]
        # Use the best_h so far as starting point if no optimization succeeded
        if best_h is None:
            h0 = setup_initial_solution(n_points)
        else:
            h0 = best_h.copy()
            
        # Run a smaller optimization using minimize
        result = minimize(
            objective,

            options={'maxiter': 1000, 'disp': False},
            callback=lambda x, result: callback(x, None)
        )
        
        if result.success:
            final_c5 = objective(result.x)
            if final_c5 < best_c5:
                best_c5 = final_c5

    except Exception as e:
        print(f"Error in trust-constr: {e}")
        
    # Final check and soft boundary enforcement if needed
    if best_h is not None:

            best_c5 = objective(best_h)
            
    return best_h, best_c5

    # Start timing

    # Evaluate final state  
    final_c5 = evaluate_erdos_solution(h_final)
    
    return (h_final.tolist(), final_c5, n_points)
