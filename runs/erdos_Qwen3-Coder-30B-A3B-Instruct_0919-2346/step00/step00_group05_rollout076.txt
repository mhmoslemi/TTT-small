```python
import numpy as np
from scipy.optimize import minimize
from cvxpy import *
import cvxpy as cp
from cvxpy import Variable, Minimize, Problem, log, sum_squares
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h_values):
    """Evaluate C5 from an h function."""
    c = correlate(h_values, 1 - np.array(h_values), mode='full')
    return np.max(c) * (2.0 / len(h_values))

def initial_h_values():
    """Provide a good initial solution"""
    n = 70
    h = np.zeros(n)
    h[:35] = 1
    return h

def rounded_binary(h_cont, n):
    """Round continuous h to binary with right mass"""
    # Get indices of n/2 largest values
    indices = np.argsort(h_cont)[-n//2:]
    h_bin = np.zeros(n)
    h_bin[indices] = 1
    return h_bin

def project_onto_simplex(h, target_sum):
    """Project h onto the capped simplex with sum = target_sum"""
    h = np.array(h)
    h = np.clip(h, 0, 1)
    s = np.sum(h)
    if s == target_sum:
        return h
    elif s > target_sum:
        # Decrease values
        diff = s - target_sum
        for i in reversed(np.argsort(h)):
            if diff <= 0:
                break
            reduce = min(h[i], diff)
            h[i] -= reduce
            diff -= reduce
    else:  # s < target_sum
        diff = target_sum - s
        for i in sorted(range(len(h)), key=lambda x: h[x]):
            if diff <= 0:
                break
            increase = min(1 - h[i], diff)
            h[i] += increase
            diff -= increase
    return h

def solve_milp_h(h0, n_points, max_seconds):
    """Solve the MILP formulation for h using CVXPY and McCormick"""
    # Create CVXPY variables
    h = Variable(n_points, name='h')
    tau = Variable(name='tau')
    
    # Constraints to enforce product variables using McCormick
    y_vars = []
    constraints = []
    
    # Use a reference for the best τ achieved so far
    best_tau = -np.inf
    
    # Add main constraints
    constraints.append(sum(h) == n_points/2)
    constraints.append(h >= 0)
    constraints.append(h <= 1)
    # Symmetry breaking
    constraints.append(h[0] == 1)
    
    # Build product variables and constraints for each shift k = 1,...,n-1
    for k in range(1, n_points):
        # y[i,k] = h[i] * h[i+k]
        # Let's define a vector for this shift
        if k == 1:
            y_k = Variable(n_points - 1, name=f'y_k{k}')
        else:
            y_k = Variable(n_points - k, name=f'y_k{k}')
        y_vars.append(y_k)
        
        # Add the McCormick constraints
        for i in range(n_points - k):
            # y[i,k] <= h[i]
            constraints.append(y_k[i] <= h[i])
            # y[i,k] <= h[i+k]
            constraints.append(y_k[i] <= h[i+k])
            # y[i,k] >= h[i] + h[i+k] - 1
            constraints.append(y_k[i] >= h[i] + h[i+k] - 1)
            # y[i,k] >= 0
            constraints.append(y_k[i] >= 0)
        
        # tau <= sum_{i} y[i,k]
        constraints.append(tau <= sum(y_k))
    
    # Objective
    objective = Maximize(tau)
    
    # Solve the problem
    prob = Problem(objective, constraints)
    
    try:
        # Note: the mainstream CVXPY doesn't directly support integer variables
        # Thus we use continuous relaxation and attempt rounding or project
        prob.solve(solver=cp.GLPK, time_limit=max_seconds, verbose=False)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_sol = h.value
            tau_sol = tau.value
            return h_sol, tau_sol
    except Exception as e:
        print("MILP solver exception:", e)
    
    # If MILP fails, fallback to LP relaxation
    prob_relax = Problem(objective, constraints)
    try:
        prob_relax.solve(solver=cp.GLPK, time_limit=max_seconds, verbose=False)
        if prob_relax.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_sol = h.value
            tau_sol = tau.value
            return h_sol, tau_sol
    except Exception as e:
        print("LP relaxation exception:", e)
    
    return h0, -np.inf  # fallback


def local_improvement(h, n_points):
    """Apply swap-based local search to reduce C5"""
    c = correlate(h, 1 - h, mode='full')
    k_best = np.argmax(c)
    # Get the worst duality gap
    if k_best >= n_points:
        k_best = k_best - n_points
    # Attempt to make swaps improving C5
    for i in range(n_points - k_best):
        if h[i] == 1 and h[i + k_best] == 0:
            # Try placing a 1 in position i+k_best and a 0 in position i
            h_copy = h.copy()
            h_copy[i] = 0
            h_copy[i + k_best] = 1
            c_new = correlate(h_copy, 1 - h_copy, mode='full')
            if np.max(c_new) < np.max(c):
                h = h_copy.copy()
                c = c_new
    return h


def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 70
    dx = 2.0 / n_points
    
    h_best = None
    c5_best = np.inf
    
    # Start with initial_solution
    h_start = initial_h_values()
    
    # Refine start with local search
    h_refined = h_start.copy()
    h_refined = local_improvement(h_refined, n_points)
    c5_initial = evaluate_erdos_solution(h_refined)
    
    # If this is better than best, save it
    if c5_initial < c5_best:
        c5_best = c5_initial
        h_best = h_refined.copy()

    # Now run MILP solves
    max_seconds_per_solve = 100
    restart_count = 0
    
    while time.time() - start_time < budget_s * 0.8:
        restart_count += 1
        
        # Mimic warm-start by picking a potentially better starting point
        h_guess = h_start
        
        # For a few things, we are literally randomizing the start
        if restart_count >= 2:
            # Random permutation for exploration
            p = np.random.permutation(n_points)
            h_guess = h_start[p]
            h_guess = project_onto_simplex(h_guess, n_points / 2)
        
        # Perturb if we have at least one warm-start
        if restart_count >= 3:
            # Try to make a few random bit-flips
            n_flip = max(1, n_points // 50)
            flip_indices = np.random.choice(n_points, size=n_flip, replace=False)
            h_guess = h_guess.copy()
            for i in flip_indices:
                h_guess[i] = 1 - h_guess[i]
            h_guess = project_onto_simplex(h_guess, n_points / 2)

        # Solve a relaxed MILP

        h_sol, tau_sol = solve_milp_h(h_guess, n_points, max_seconds_per_solve)
        
        if np.isfinite(tau_sol) and tau_sol > -1e10:
            # If it's not too bad, check C5
            h_sol = project_onto_simplex(h_sol, n_points / 2)
            # Retry with binary rounding
            h_bin = rounded_binary(h_sol, n_points)
            c5_val = evaluate_erdos_solution(h_bin)
            
            if c5_val < c5_best:
                c5_best = c5_val
                h_best = h_bin.copy()
                
                # Try local search as well
                h_improved = local_improvement(h_bin.copy(), n_points)
                c5_improved = evaluate_erdos_solution(h_improved)
                if c5_improved < c5_best:
                    c5_best = c5_improved
                    h_best = h_improved.copy()
            
            # Prevent failure due to solver not giving a valid relaxation
            if c5_sol < c5_best:
                c5_best = c5_sol
                h_best = h_sol.copy()
        
        # FF: ignore end condition - will not break precision
        # if time.time() - start_time > budget_s * 0.8:
        #     break

    # Final recurse check on any h_best we found
    if h_best is None:
        h_best = h_start
        c5_best = evaluate_erdos_solution(h_best)

    return h_best, c5_best, n_points
```
