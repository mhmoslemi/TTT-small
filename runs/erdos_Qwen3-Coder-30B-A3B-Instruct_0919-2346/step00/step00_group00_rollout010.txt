import numpy as np
import time
import random

def compute_c5(h):
    """Compute C5 bound from step function h."""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return corr.max()

def propose_swap_and_update(h, corr, i, j):
    """Compute delta changes to correlation due to swapping h[i] and h[j]."""
    # Old values
    a = h[i]
    b = h[j]
    
    # Delta values
    delta_i = b - a
    delta_j = a - b  # = -delta_i
    
    # Store old corr to avoid rewriting over same array
    new_corr = corr.copy()
    
    n = len(h)
    
    # Loop over k not equal to i and j
    for k in range(n):
        if k == i or k == j:
            continue
        
        hk = h[k]
        # Contributions for (i,k) -> shift (k-i)
        idx1 = k - i + n - 1
        new_corr[idx1] += delta_i * (1 - hk)
        
        # Contributions for (k,i) -> shift (i-k)
        idx2 = i - k + n - 1
        new_corr[idx2] += hk * (-delta_i)
        
        # Contributions for (j,k) -> shift (k-j)
        idx3 = k - j + n - 1
        new_corr[idx3] += delta_j * (1 - hk)
        
        # Contributions for (k,j) -> shift (j-k)
        idx4 = j - k + n - 1
        new_corr[idx4] += hk * (-delta_j)
    
    # Handle cross contributions (i,j) and (j,i)
    # For (i,j): shift = j-i
    idx_ij = j - i + n - 1
    new_corr[idx_ij] += delta_i
    
    # For (j,i): shift = i-j
    idx_ji = i - j + n - 1
    new_corr[idx_ji] += delta_j
    
    # Update h array after computing delta
    h[i], h[j] = h[j], h[i]
    
    return new_corr

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    start = time.time()
    best_c5 = np.inf
    best_h = None
    best_n = None
    
    # Define candidate n values
    n_candidates = [60, 80, 100, 120, 140, 160, 180, 200]
    n_candidates = [n for n in n_candidates if n % 2 == 0]  # Must be even
    
    # Track best discovered in any previous loop
    overall_best_c5 = np.inf
    overall_best_h = None
    overall_best_n = None

    for n in n_candidates:
        if time.time() - start > budget_s - 1:
            break
            
        # Allocate iterations based on n size
        max_iters = max(2000, int(50000 / (n/100)))
        
        # Initialize h
        h = np.zeros(n, dtype=float)
        ones = n // 2
        indices = np.random.choice(n, ones, replace=False)
        h[indices] = 1.0

        # Initial correlation
        cur_c5 = compute_c5(h)
        corr = np.correlate(h, 1 - h, mode='full')

        cur_h = h.copy()
        cur_max = corr.max()
        cur_c5 = cur_max * (2.0 / n)
        
        # Temperature for simulated annealing
        T = 0.05
        T_decay = 0.9995
        
        # For restart detection
        last_improvement = 0
        no_improvement_count = 0
        
        for it in range(max_iters):
            if time.time() - start > budget_s - 1:
                break
                
            # Find indices to swap
            i = np.random.randint(n)
            j = np.random.randint(n)
            while h[i] == h[j]:
                i = np.random.randint(n)
                j = np.random.randint(n)
            
            # Compute new correlation after swap
            new_corr = propose_swap_and_update(h.copy(), corr, i, j)
            new_max = new_corr.max()
            dx = 2.0 / n
            new_c5 = new_max * dx
            
            # Acceptance decision
            d = new_c5 - cur_c5
            if d < 0 or np.random.rand() < np.exp(-d / (T + 1e-12)):
                corr = new_corr
                cur_c5 = new_c5
                cur_max = new_max
                # Update best if improved
                if cur_c5 < overall_best_c5:
                    overall_best_c5 = cur_c5
                    overall_best_h = h.copy()
                    overall_best_n = n
            else:
                # revert swap in h (already done in propose_swap_and_update)
                pass

            # Update temperature
            T *= T_decay
            
            # Optional restarts
            if cur_c5 < overall_best_c5 - 1e-6:
                last_improvement = it
                no_improvement_count = 0
            else:
                no_improvement_count += 1
                
            if no_improvement_count > 5000:
                # Restart
                h = np.zeros(n, dtype=float)
                ones = n // 2
                indices = np.random.choice(n, ones, replace=False)
                h[indices] = 1.0
                cur_c5 = compute_c5(h)
                corr = np.correlate(h, 1 - h, mode='full')
                cur_max = corr.max()
                cur_c5 = cur_max * (2.0 / n)
                no_improvement_count = 0
                T = 0.05

    # Final fallback if none yet found
    if overall_best_h is None:
        try:
            from __main__ import initial_h_values
            overall_best_h = initial_h_values.copy()
            overall_best_n = len(overall_best_h)
            overall_best_c5 = compute_c5(overall_best_h)
        except Exception:
            # Fallback
            n = 100
            h = np.zeros(n, dtype=float)
            h[:50] = 1.0
            overall_best_h = h
            overall_best_n = n
            overall_best_c5 = compute_c5(overall_best_h)
    
    # Return final solution
    return overall_best_h, overall_best_c5, overall_best_n
