import numpy as np
from scipy.optimize import differential_evolution
from scipy.spatial.distance import pdist
import time

def evaluate_erdos_solution(h_values, n_points=45):
    """Evaluate the Erdős constant for given h_values."""
    dx = 2.0 / n_points
    h = np.array(h_values)
    # Compute the full convolution of h with (1-h)
    corr = np.correlate(h, 1 - h, mode='full')
    # The maximum value scaled by dx
    max_val = np.max(corr) * dx
    return max_val

def initial_h_values(n_points=45):
    """Initial construction - a step function."""
    h = np.ones(n_points)
    mid = n_points // 2
    h[mid:] = 0
    return h

def objective(h_values, n_points=45):
    """Objective function to minimize."""
    dx = 2.0 / n_points
    h = np.array(h_values)
    # Ensure constraints
    h = np.clip(h, 0, 1)
    # Integral constraint
    if np.sum(h) * dx != 1.0:
        # Adjust to satisfy constraint
        scale = 1.0 / (np.sum(h) * dx)
        h = h * scale
    
    # Compute correlations with 1-h
    corr = np.correlate(h, 1 - h, mode='full')
    # Return negative because we minimize
    return -np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    """Run optimization to find a better h_values."""
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 45
    dx = 2.0 / n_points
    
    # Start with initial construction
    h_init = initial_h_values(n_points)
    h_best = h_init.copy()
    c5_best = evaluate_erdos_solution(h_best, n_points)
    
    # Constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Additional constraint: integral = 1
    def constraint_integral(x):
        return np.sum(x) * dx - 1.0
    
    # Optimization method: differential evolution
    # This is a global optimizer that works well for black-box functions
    def optimized_objective(x):
        # Clip for bounds
        x = np.clip(x, 0, 1)
        # Force integral constraint approximately
        # Scale to meet constraint roughly
        if np.sum(x) * dx != 0:
            scale = 1.0 / (np.sum(x) * dx)
            x = x * scale
        return -evaluate_erdos_solution(x, n_points)
    
    # Run optimization with time limit
    result = differential_evolution(
        optimized_objective,
        bounds,
        maxiter=int(1000 * (budget_s / 600)),
        popsize=15,
        seed=seed,
        disp=False
    )
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return (h_best.tolist(), c5_best, n_points)
    
    # Try to refine the result
    h_opt = np.clip(result.x, 0, 1)
    # Enforce integral constraint
    if np.sum(h_opt) * dx != 0:
        h_opt = h_opt * (1.0 / (np.sum(h_opt) * dx))
    
    c5_opt = evaluate_erdos_solution(h_opt, n_points)
    
    if c5_opt < c5_best:
        h_best = h_opt
        c5_best = c5_opt
    
    # Final refinement with a slight local optimization
    def local_search(h0):
        h = h0.copy()
        dx = 2.0 / n_points
        # Small perturbations to improve
        for i in range(n_points):
            if np.sum(h) * dx == 0:
                break
            # Perturb
            delta = np.random.normal() * 0.05
            h[i] = np.clip(h[i] + delta, 0, 1)
            # Rescale
            h = h * (1.0 / (np.sum(h) * dx))
            c5 = evaluate_erdos_solution(h, n_points)
            if c5 < c5_best:
                h_best = h.copy()
                c5_best = c5
        return h_best, c5_best
    
    local_h, local_c5 = local_search(h_best)
    if local_c5 < c5_best:
        h_best = local_h
        c5_best = local_c5
    
    return (h_best.tolist(), c5_best, n_points)
