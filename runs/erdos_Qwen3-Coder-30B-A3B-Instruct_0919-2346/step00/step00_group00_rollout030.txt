import numpy as np
import time
import random

def compute_c5(h):
    """Compute C5 bound from step function h"""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return corr.max()

def propose_swap_and_update(h, corr, i, j):
    """Propose a swap between positions i,j and update correlation array efficiently"""
    # old values
    old_i = h[i]
    old_j = h[j]
    
    delta_i = old_j - old_i   # new_i - old_i = old_j - old_i
    
    # alias for length
    n = len(h)
    
    # make a copy of correlation array to compute new one
    new_corr = corr.copy()
    
    # loop over k not i,j
    for k in range(n):
        if k == i or k == j:
            continue
        hk = h[k]
        # contributions where i is first element
        # shift i->k: l = k - i
        idx1 = k - i + n - 1
        new_corr[idx1] += delta_i * (1 - hk)
        # shift k->i: l = i - k  
        idx2 = i - k + n - 1
        new_corr[idx2] += hk * (-delta_i)
        
        # contributions where j is first element
        # shift j->k: l = k - j
        idx3 = k - j + n - 1
        new_corr[idx3] += -delta_i * (1 - hk)  # delta_j = -delta_i
        # shift k->j: l = j - k
        idx4 = j - k + n - 1
        new_corr[idx4] += hk * (delta_i)   # delta_j = -delta_i => -hk*delta_j = hk*delta_i

    # handle cross pair contributions: p=i,q=j and p=j,q=i
    # pair (i,j): old contribution a*(1-b), new contribution b*(1-a) = b*a
    # shift = j-i
    idx_ij = j - i + n - 1
    new_corr[idx_ij] += delta_i
    
    # pair (j,i): old b*(1-a), new a*(1-b) = a*b
    # shift = i-j
    idx_ji = i - j + n - 1
    new_corr[idx_ji] -= delta_i

    # compute new max
    new_max = new_corr.max()
    
    return new_corr, new_max

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()
    
    best_h = None
    best_c5 = np.inf
    best_n = None

    # Try a range of n values
    n_candidates = [60, 80, 100, 120, 140, 160, 180, 200]
    # Keep track of time spent
    time_spent = 0

    for n in n_candidates:
        if time.time() - start > budget_s - 2:
            break
            
        if n % 2 != 0:
            continue
            
        # Estimate iterations based on available time and n
        time_left = budget_s - (time.time() - start)
        max_iters = max(2000, int(20000 * (200.0 / n)))
        
        # Initialize h as binary sequence with exactly n/2 ones
        h = np.zeros(n, dtype=float)
        ones = n // 2
        h[np.random.choice(n, ones, replace=False)] = 1.0

        # Compute initial correlation and c5
        corr = np.correlate(h, 1 - h, mode='full')
        cur_max = corr.max()
        cur_c5 = cur_max * (2.0 / n)
        
        # Current best for this n
        cur_h = h.copy()
        cur_corr = corr.copy()
        
        # Simulated Annealing parameters
        T = 0.1
        T_decay = 0.9995
        no_improvement_count = 0
        max_no_improvement = 30000

        for it in range(max_iters):
            now = time.time()
            if now - start > budget_s - 1:
                break
                
            # Propose a swap
            i = np.random.randint(n)
            j = np.random.randint(n)
            while h[i] == h[j]:
                i = np.random.randint(n)
                j = np.random.randint(n)
                
            # Compute new correlation and max
            new_corr, new_max = propose_swap_and_update(h, corr, i, j)
            new_c5 = new_max * (2.0 / n)
            
            # Decision rule
            if new_c5 < cur_c5 or np.random.rand() < np.exp(-(new_c5 - cur_c5) / (T + 1e-12)):
                # Accept the move
                corr = new_corr
                cur_max = new_max
                cur_c5 = new_c5
                h[i], h[j] = h[j], h[i]
                no_improvement_count = 0
            else:
                no_improvement_count += 1

            # Update best overall
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                best_n = n
                
            # Cooling schedule
            T *= T_decay
            
            # Random restart if stuck
            if no_improvement_count > max_no_improvement:
                h = np.zeros(n, dtype=float)
                ones = n // 2
                h[np.random.choice(n, ones, replace=False)] = 1.0
                corr = np.correlate(h, 1 - h, mode='full')
                cur_max = corr.max()
                cur_c5 = cur_max * (2.0 / n)
                no_improvement_count = 0

    if best_h is None:
        # Fallback to simple pattern
        n = 60
        best_h = np.zeros(n)
        best_h[:n//2] = 1.0
        best_n = n
        best_c5 = compute_c5(best_h)
            
    return best_h, best_c5, best_n
