import numpy as np
import time

def evaluate_h(h):
    """Evaluate the maximum overlap integral for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlations: this uses np.correlate with mode='full'
    # For h and (1-h), the correlation will place the correlation at index (n-1+k)
    corr = np.correlate(h, 1 - h, mode='full')
    # Compute O_k = sum_i h[i] (1 - h[i+k]) * dx for each shift k
    max_o = 0
    for k in range(1, n):  # k from 1 to n-1
        # For shift k, index in correlation array is n-1 + k
        index = (n - 1) + k
        if index < len(corr):
            o_k = corr[index] * dx
            max_o = max(max_o, o_k)
    return max_o

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 200  # Starting reasonable size
    dx = 2.0 / n_points
    target_sum = n_points / 2  # We want sum h = n_points/2 exactly
    
    # Initial construction from the given initial_h_values global variable
    # But we can start with a better guess, or a random binary sequence with exact weight
    h = np.zeros(n_points)
    
    # Random binary initialization with exact weight
    ones_indices = np.random.choice(n_points, int(target_sum), replace=False)
    h[ones_indices] = 1
    
    best_h = h.copy()
    best_c5 = evaluate_h(h)
    
    # Annealing parameters
    T = 0.1
    cooling_rate = 0.99999
    max_iter = 1_000_000
    iter_count = 0
    
    while time.time() - start_time < budget_s and iter_count < max_iter:
        # Propose new solution: pick two indices, 1 and 0, swap
        i = np.random.randint(0, n_points)
        j = np.random.randint(0, n_points)
        
        # Ensure they are different and have different values
        if h[i] == h[j]:
            # Find a valid pair with different values
            mask_1 = (h == 1)
            mask_0 = (h == 0)
            if np.any(mask_1) and np.any(mask_0):
                i = np.random.choice(np.where(mask_1)[0])
                j = np.random.choice(np.where(mask_0)[0])
            else:
                continue  # Skip if no pair available
        
        # Create a temporary copy for swap
        h_new = h.copy()
        h_new[i] = 1 - h_new[i]  # Flip
        h_new[j] = 1 - h_new[j]  # Flip
        
        # Evaluate new solution
        new_c5 = evaluate_h(h_new)
        
        # Accept or reject
        if new_c5 < best_c5:
            h = h_new
            best_h = h_new
            best_c5 = new_c5
        elif np.random.rand() < np.exp(-(new_c5 - best_c5) / T):
            h = h_new
            
        # Cool down
        T *= cooling_rate
        iter_count += 1
        
        # Occasionally re-evaluate current best to ensure we don't lose it
        if iter_count % 100 == 0:
            current_c5 = evaluate_h(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
