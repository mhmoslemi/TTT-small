import numpy as np
import cvxpy as cp
from cvxpy import Minimize, Problem
import time
import random

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    # Compute cross-correlation h * (1 - h)
    c = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(c)
    return max_corr * dx

def project_to_simplex(v, s=1.0):
    """Projects vector v onto the capped simplex with sum = s"""
    v = np.array(v)
    n = len(v)
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - s
    rho = np.argmax(u > css / np.arange(1, n+1)) if u[0] > css[0] / 1 else 0
    theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0)

def solve_milp(n_points, h0=None, max_seconds=300, binary=True):
    # Create variables
    h = cp.Variable(n_points, name="h")
    # Tau is the min autocorrelation we want to maximize
    tau = cp.Variable(name="tau")

    constraints = []
    vars_y = {}
    for k in range(1, n_points):
        # For shift k, we only go up to n-k terms
        y = cp.Variable(n_points - k, name=f"y_{k}")
        vars_y[k] = y
        
        # McCormick envelope constraints
        for i in range(n_points - k):
            constraints += [
                y[i] >= 0,
                y[i] <= h[i],
                y[i] <= h[i + k],
                y[i] >= h[i] + h[i + k] - 1
            ]
        
        # tau <= sum(y_k)
        constraints.append(tau <= cp.sum(y))

    # Mass constraint: sum(h) = n/2
    constraints.append(cp.sum(h) == n_points / 2)

    # Bounding
    constraints += [h[i] >= 0 for i in range(n_points)]
    constraints += [h[i] <= 1 for i in range(n_points)]

    # Symmetry breaking
    constraints.append(h[0] == 1.0)

    # Binary constraint if needed
    if binary:
        constraints += [h[i] == cp.Variable(1, boolean=True) for i in range(n_points)]

    objective = cp.Maximize(tau)
    prob = Problem(objective, constraints)

    # Set solver and time limit
    start_t = time.time()
    prob.solve(solver=cp.CBC, time_limit=max_seconds, warm_start=False)
    
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        h_val = h.value
        tau_opt = tau.value
        elapsed = time.time() - start_t
        return h_val, tau_opt, elapsed
    else:
        # Fallback to solving as continuous (LP)
        if binary:
            return solve_milp(n_points, h0, max_seconds, binary=False)
        else:
            return h.value, 0, time.time() - start_t

    c5_best = evaluate_erdos_solution(h_best)
    elapsed_total = 0.0

    def mk_h0(n_points, gen_type="binary"):
        if gen_type == "binary":
            # Generate binary vector with equal ones and zeros
            h0 = np.zeros(n_points)
            ones_indices = np.random.choice(n_points, size=n_points//2, replace=False)
            h0[ones_indices] = 1.0
            return h0
        else:
            return np.random.rand(n_points)

    # Run several restarts
    max_restarts = 10
    restart_count = 0

    while elapsed_total < budget_s and restart_count < max_restarts:
        # Choose a random start strategy (could be more nuanced)
        h0_type = ["binary", "continuous", "perturbation"][restart_count % 3]

        if h0_type == "binary":
            h0 = mk_h0(n_points, "binary")
        elif h0_type == "continuous":
            h0 = mk_h0(n_points, "continuous")
        else:
            # perturbation: find current best and mutate it slightly
            h0 = h_best.copy()
            # Add small noise
            h0 += np.random.normal(0, 0.02, n_points)
            h0 = np.clip(h0, 0, 1)
            h0 = project_to_simplex(h0, s=n_points/2)

        # Solve the problem
        remaining = budget_s - elapsed_total
        if remaining < 5:
            break

        h_solved, tau_solved, elapsed = solve_milp(n_points, h0, max_seconds=min(remaining, 300))

        elapsed_total += elapsed
        
        if h_solved is not None:
            # Project back if not binary
            if not np.allclose(h_solved, h_solved.astype(int)):
                h_projected = project_to_simplex(h_solved, s=n_points/2)
                c5_solved = evaluate_erdos_solution(h_projected)
            else:
                c5_solved = evaluate_erdos_solution(h_solved)

            if c5_solved < c5_best:
                c5_best = c5_solved
                h_best = h_solved.copy()

        restart_count += 1

    return h_best, c5_best, n_points
