import time
import numpy as np
from scipy.optimize import minimize
from scipy.linalg import eigh
import cvxpy as cp
from cvxpy import SCS, ECOS

def evaluate_c5(h):
    """Evaluate the C5 constant using circular correlation."""
    n = len(h)
    # Circular correlation (FFT-based)
    corr = np.fft.irfft(np.fft.rfft(h) * np.conj(np.fft.rfft(1-h)), n=2*n)
    corr = np.real(corr[:n])
    min_corr = np.min(corr)
    return 1 - (2.0/n) * min_corr

def sdp_solve(n, max_time=120):
    """Solve relaxed SDP to get a better bound."""
    start = time.time()
    np.random.seed(42)
    h = cp.Variable(n)
    X = cp.Variable((n, n), symmetric=True)
    t = cp.Variable()
    
    # Build the constraints for the SDP relaxation
    M = []
    for k in range(n):
        M_k = np.zeros((n, n))
        for i in range(n):
            M_k[i, (i+k) % n] = 1
            M_k[(i+k) % n, i] = 1
        M.append(M_k)
    
    constraints = [
        cp.trace(M[k] @ X) >= t for k in range(n)
    ]
    constraints += [X[i, i] == h[i] for i in range(n)]
    constraints += [0 <= h[i] for i in range(n)]
    constraints += [h[i] <= 1 for i in range(n)]
    constraints += [cp.sum(h) == n/2]
    constraints += [X >> 0]  # PSD constraint
    
    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=ECOS, verbose=False, max_iters=1000)
    except Exception:
        return None, None
    
    if prob.status in ["optimal", "optimal_inaccurate"]:
        h_sol = h.value
        t_val = t.value
        return h_sol, t_val
    return None, None

def round_h(X, n):
    """Round the SDP solution to a step function."""
    try:
        # Eigenvalue decomposition to get low-rank factorization
        U, V = eigh(X)
        rank = np.sum(U > 1e-8)
        if rank == 0:
            return np.zeros(n)
        V = V[:, :rank]
        U = U[:rank]
        V = V @ np.diag(np.sqrt(U))
        g = np.random.randn(rank)
        scores = V.T @ g
        sorted_scores = np.sort(scores)[::-1]
        if len(sorted_scores) >= n//2 + 1:
            theta = sorted_scores[n//2]
        else:
            theta = sorted_scores[-1]
        h_rounded = (scores >= theta).astype(float)
        # Adjust if needed to get the exact count of 1s
        ones_count = np.sum(h_rounded)
        while ones_count < n//2:
            h_rounded[np.argmin(scores)] = 1
            ones_count += 1
        while ones_count > n//2:
            h_rounded[np.argmax(scores)] = 0
            ones_count -= 1
        return h_rounded
    except:
        return np.zeros(n)

def swap_hill_climb(h, n):
    """Hill climb using only swaps."""
    h = np.array(h.copy())
    
    def get_autocorr(h, k):
        return np.sum(h * np.roll(h, k))
    
    # Compute initial autocorrelations
    auto_corr = [get_autocorr(h, k) for k in range(n)]
    min_corr = min(auto_corr)
    
    improvement = True
    while improvement:
        improvement = False
        for i in range(n):
            for j in range(n):
                if h[i] != h[j]:
                    # Try swap at positions i and j
                    old_i = auto_corr[(i - k) % n] for k in range(n))
                    old_j = auto_corr[(j - k) % n] for k in range(n))
                    h[i], h[j] = h[j], h[i]
                    new_auto = [get_autocorr(h, k) for k in range(n)]
                    new_min = min(new_auto)
                    if new_min > min_corr:
                        min_corr = new_min
                        improvement = True
                        # Recompute all autocorrelations in a faster way
                        auto_corr = new_auto
                    else:
                        h[i], h[j] = h[j], h[i]  # undo swap
                        # Reconstruct using existing window
    return h

def gradient_fine_tune(h, n, num_iter=500):
    """Continuous gradient ascent for min autocorrelation."""
    h = h.copy()
    
    def get_autocorr(h, k):
        return np.sum(h * np.roll(h, k))
    
    for _ in range(num_iter):
        # Compute current min_autocorr index and value
        auto_corr = [get_autocorr(h, k) for k in range(n)]
        min_k = np.argmin(auto_corr)
        min_autocorr = auto_corr[min_k]
        
        # Gradient update
        grad = np.zeros_like(h)
        # This is an approximation, calculated only for the worst shift
        if min_k == 0:
            grad += h[1:] + h[:-1]
        elif min_k == 1:
            grad[0] += h[1] + 1
            grad[1] += 1 + h[0]
            for i in range(2, n):
                grad[i] += h[i-1] + h[(i + 1) % n]
        else:
            # For other shifts, we only compute relevant parts
            for i in range(n):
                grad[i] += h[(i + min_k) % n] + h[(i - min_k) % n]
        
        # Project to maintain sum and bounds
        step = 0.001
        h = h + step * grad
        h = np.clip(h, 0, 1)
        h = h * (n/2) / np.sum(h)
    
    return h

def interval_optimize(n_points=200, max_time=50):
    """Optimize via union of intervals."""
    start = time.time()
    
    def overlap(k, intervals):
        """Analytical overlap for shifted intervals."""
        total = 0
        # Simplified to sum of intersections of all pairs
        # Can be more efficiently computed, but enough for the purpose.
        for (a, b) in intervals:
            shifted_a = a + k
            shifted_b = b + k
            if shifted_b > 2:
                total += (b - a) / 2 + (2 - a) / 2  # wrap around
            else:
                # intersect with [0,2]
                total += max(0, min(shifted_b, 2) - max(shifted_a, 0))
        return total
    
    # Model as union of intervals
    def build_intervals(x):
        # Explicit interval representation
        left = x[0::2]
        right = x[1::2]
        if len(left) < len(right):
            right = right[:len(left)]
        intervals = list(zip(left, right))
        total_len = sum(b - a for a, b in intervals)
        if abs(total_len - 1) > 1e-6:
            return float('inf')
        return intervals

    def target_func(x):
        if len(x) % 2 != 0:
            x = x[:-1]
        n_intervals = len(x) // 2
        intervals = []
        for i in range(n_intervals):
            a = x[2*i]
            b = x[2*i+1]
            if a >= b: return float('inf')
            intervals.append((a, b))
        # Symmetric
        intervals = [(a, b) for a, b in intervals]
        # assess impacts on all shifts
        shifts = [i for i in range(n_points)]
        overlaps = []
        for shift in shifts:
            shift % n_points
            overlap_total = 0
            for a, b in intervals:
                tmp = (b - a) - (max(0, a + shift) - max(0, a)) - (max(0, b + shift) - max(0, b))
                factor = 1.0 if (a + shift < 2 and b + shift > 0) else 0.0
                overlap_total += factor * max(0, min(b + shift, 2) - max(a + shift, 0))
            overlaps.append(overlap_total)
        if not overlaps:
            return float('inf')
        min_overlap = min(overlaps)
        return -min_overlap

    # Try a few random initiatives
    best_x = []
    best_val = float('inf')
    for i in range(1000):
        x = np.random.rand(12)
        if time.time() - start > max_time:
            break
        # Clip to range
        x = np.clip(x, 0, 2)
        res = minimize(target_func, x, method='Nelder-Mead', bounds=[(0, 2)] * len(x))
        if res.success and -res.fun < best_val:
            best_val = -res.fun
            best_x = res.x
    # Convert this to step function discretization
    if best_x:
        h = np.zeros(n_points)
        for i in range(n_points):
            x = (i * 2.0) / n_points
            for a, b in build_intervals(best_x):
                if a <= x < b:
                    h[i] = 1
                    break
        return h
    return np.zeros(n_points)

def run(seed=42, budget_s=600):
    """Main function to solve the problem."""
    start = time.time()
    n_points = 80
    
    best_h = None
    best_c5 = float('inf')
    
    # Step 1: Interval-based optimization
    if time.time() - start < budget_s - 20:
        interval_h = interval_optimize(n_points=200, max_time=30)
        interval_c5 = evaluate_c5(interval_h)
        if interval_c5 < best_c5:
            best_c5 = interval_c5
            best_h = interval_h.copy()
            print(f"Initial interval heuristic: C5={interval_c5}")
    
    # Step 2: SDP + rounding pipeline
    sdpsub = 40
    for trial in range(30):
        if time.time() - start > budget_s - sdpsub:
            break
        h_sdp, t = sdp_solve(n_points, sdpsub)
        if h_sdp is None:
            continue
        # Round and polish
        h = round_h(np.outer(h_sdp, h_sdp), n_points)
        h = swap_hill_climb(h, n_points)
        h = gradient_fine_tune(h, n_points, 300)
        h_c5 = evaluate_c5(h)
        if h_c5 < best_c5:
            best_c5 = h_c5
            best_h = h.copy()
            print(f"New best from SDP: C5={h_c5}")
    
    # Final evaluate
    if best_h is not None:
        final_c5 = evaluate_c5(best_h)
        best_h = best_h.astype(float)
        return (best_h, final_c5, n_points)
    else:
        # fallback to a simple construction
        h_fallback = np.ones(n_points) * 0.5
        return (h_fallback, 0.5, n_points)
