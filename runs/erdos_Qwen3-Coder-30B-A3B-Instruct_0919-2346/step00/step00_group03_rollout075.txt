import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for a given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute convolution of h with (1-h)
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum value over all shifts is the C5 bound
    c5 = np.max(conv) * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Starting point - use initial construction
    h = np.array(initial_h_values)
    n_points = len(h)
    
    # Discretization step size
    dx = 2.0 / n_points
    
    def objective(h_vec):
        # Reshape to array
        h_new = h_vec.copy()
        # Evaluate C5
        conv = np.correlate(h_new, 1 - h_new, mode="full")
        c5 = np.max(conv) * dx
        return c5
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) * dx - 1.0
    
    def constraint_bounds(h_vec):
        return np.minimum(h_vec, 1.0)  # h[i] <= 1
    
    def constraint_bounds_low(h_vec):
        return np.maximum(h_vec, 0.0)  # h[i] >= 0
    
    # Setup constraints for CVXPY
    h_var = Variable(n_points)
    
    # Define the optimization problem
    constraints = [h_var >= 0, h_var <= 1, sum(h_var) == 1.0]
    
    # This is tricky to solve with CVXPY because of the max(...) term
    # So we'll use a gradient-based approach with scipy
    
    # Function to evaluate the objective and constraints
    def objective_and_grad(h_vec):
        # Ensure it's within bounds
        h_vec = np.clip(h_vec, 0, 1)
        # Compute C5
        conv = np.correlate(h_vec, 1 - h_vec, mode="full")
        c5 = np.max(conv) * dx  # This is our objective
        
        # For gradient approximation (very simple)
        # Maybe use a finite difference approach for the gradient
        # Approximate gradient using finite difference
        eps = 1e-5
        grad = np.zeros_like(h_vec)
        for i in range(n_points):
            h_vec_plus = h_vec.copy()
            h_vec_plus[i] += eps
            c5_plus = objective(h_vec_plus)
            h_vec_minus = h_vec.copy()
            h_vec_minus[i] -= eps
            c5_minus = objective(h_vec_minus)
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        return c5, grad
    
    # Basics for optimization
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Starting from initial solution
    h_current = h.copy()
    
    # Instead of complex CVXPY, use Nelder-Mead (simple to implement and fast)
    # since C5 is likely slowly varying smooth function
    
    # Record best solution
    best_c5 = evaluate_erdos_solution(h_current)
    best_h = h_current.copy()
    
    # Optimization steps - try slightly random perturbations
    max_iter = 10000
    # We can use a simple gradient descent with momentum or random search
    # Instead, let's just optimize the single var h in a simple way
    
    # More practical approach:
    # perturb h and see if we can reduce C5
    def perturb_h(h_input, amount=0.01):
        noise = np.random.randn(len(h_input)) * amount
        h_new = h_input + noise
        h_new = np.clip(h_new, 0, 1)
        # Ensure integral constraint roughly holds
        current_sum = np.sum(h_new) * dx
        scale = 1.0 / current_sum * 1.0  # Project to integral = 1

        return h_new
    
    # Begin iterative improvement
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:

        for _ in range(5):

            c5_trial = evaluate_erdos_solution(h_trial)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                h_current = h_trial.copy()
                
        # Occasionally take a larger step
        if i % 100 == 0:
            h_current = best_h.copy()  # Rewind to best
            # Add noise
            h_current = perturb_h(h_current, amount=0.02)

    for i in range(500):

        h_new = h_current + 0.001 * grad

        # Renormalize
        current_sum = np.sum(h_new) * dx
        h_new = h_new / current_sum * 1.0
        c5_new = evaluate_erdos_solution(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_current = h_new.copy()
    
    # Return best result found
    return (best_h.tolist(), best_c5, n_points)
