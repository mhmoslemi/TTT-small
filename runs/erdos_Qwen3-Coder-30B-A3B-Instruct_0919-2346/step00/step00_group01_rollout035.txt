import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import cdist
import time
import math

def evaluate_erdos_solution(h_values):
    """Evaluate the C5 constant for given h_values"""
    n = len(h_values)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    correlations = np.correlate(h_values, 1 - np.array(h_values), mode='full')
    # Take the maximum correlation (excluding trivial shifts)
    max_correlation = max(correlations[n//2:]) * dx
    return max_correlation

def initialize_h_values(n_points):
    """Initialize h_values using a reasonable starting point"""
    # Start with the initial construction if available
    try:
        return initial_h_values[:n_points]
    except:
        # Otherwise create a simple step function
        h = np.ones(n_points) * 0.5
        # Make it a step function
        h[:n_points//2] = 1.0
        return h

def objective_function(h_values):
    """Objective to minimize"""
    # Normalize to satisfy constraint sum(h)*dx = 1
    dx = 2.0 / len(h_values)
    sum_h = np.sum(h_values)
    if sum_h > 0:
        h_normalized = h_values / sum_h * (2.0 / dx)
    else:
        h_normalized = h_values
    
    # Ensure bounds [0,1]
    h_normalized = np.clip(h_normalized, 0, 1)
    
    # Evaluate C5
    c5 = evaluate_erdos_solution(h_normalized)
    return c5

def constraint_function(h_values):
    """Constraint: integral of h equals 1"""
    dx = 2.0 / len(h_values)
    return np.sum(h_values) * dx - 1.0

def bounded_optimization(h_initial, max_time_s):
    """Perform bounded optimization of h values"""
    n_points = len(h_initial)
    bounds = [(0, 1) for _ in range(n_points)]
    time_left = max_time_s
    
    # Initial best solution
    best_h = h_initial.copy()
    best_c5 = objective_function(best_h)
    
    # Use a simple gradient-based approach with bounding
    current_h = h_initial.copy()
    
    # Store time for optimization
    start_time = time.time()
    
    # Use scipy's minimize with bounds - it's robust and fast enough
    for i in range(1000):
        if time.time() - start_time > max_time_s - 0.5:
            break
            
        try:
            # Optimize using L-BFGS-B
            result = minimize(
                objective_function,
                current_h,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 50}
            )
            
            if result.success:
                current_h = result.x
                new_c5 = objective_function(current_h)
                if new_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = new_c5
            else:
                # If optimization fails, try a simple local search
                new_h = current_h.copy()
                dx = 2.0 / n_points
                # Simple random perturbations
                for _ in range(10):
                    idx = np.random.randint(0, n_points)
                    step = np.random.normal(0, 0.01)
                    new_h[idx] = np.clip(new_h[idx] + step, 0, 1)
                # Ensure constraint
                sum_h = np.sum(new_h) * dx
                if sum_h != 0:
                    new_h *= (2.0 / (dx * sum_h))
                new_h = np.clip(new_h, 0, 1)
                new_c5 = objective_function(new_h)
                if new_c5 < best_c5:
                    best_h = new_h.copy()
                    best_c5 = new_c5
                    
                current_h = new_h
                
        except Exception as e:
            # Simple local search if optimization breaks
            new_h = current_h.copy()
            idx = np.random.randint(0, n_points)
            step = np.random.normal(0, 0.01)
            new_h[idx] = np.clip(new_h[idx] + step, 0, 1)
            sum_h = np.sum(new_h) * dx
            if sum_h != 0:
                new_h *= (2.0 / (dx * sum_h))
            new_h = np.clip(new_h, 0, 1)
            new_c5 = objective_function(new_h)
            if new_c5 < best_c5:
                best_h = new_h.copy()
                best_c5 = new_c5
                
            current_h = new_h
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use a reasonable number of points
    n_points = 200  # Lower sample count to make it faster
    if n_points > 1000:
        n_points = 1000
        
    # Get initial solution
    h = initialize_h_values(n_points)
    
    # Ensure it's valid
    dx = 2.0 / n_points
    sum_h = np.sum(h) * dx
    if abs(sum_h - 1.0) > 1e-6:
        # Renormalize if needed
        h = h / sum_h * (2.0 / dx)
        h = np.clip(h, 0, 1)
        
    # Run bounded optimization
    h_opt, c5_bound = bounded_optimization(h, budget_s)
    
    # Final validation
    final_c5 = evaluate_erdos_solution(h_opt)
    
    return (h_opt, final_c5, n_points)
