import numpy as np
from scipy.optimize import differential_evolution
import time
from math import floor

def evaluate_c5(h, dx):
    """Evaluate the C5 bound for a given step function h"""
    # Compute the overlap integral for all shifts
    correlation = np.correlate(h, 1 - h, mode='full')
    # The correlation length is 2*n - 1, so we take the middle part
    # corresponding to shifts from -1 to 1
    mid = len(correlation) // 2
    # Take only the relevant part corresponding to shifts in [-1,1]
    relevant = correlation[mid - (len(h) - 1):mid + len(h)]
    return np.max(relevant) * dx

def initial_h_values(n_points=45):
    """Return initial step function suggestion for C5 optimization"""
    # We'll start with a triangular-like shape and taper on edges
    h = np.zeros(n_points)
    dx = 2.0 / n_points
    
    # Create a piecewise constant function that roughly satisfies the constraints
    # Central plateau and ramps on the edges
    
    # The middle part is filled with a constant value that we'll optimize
    center_start = n_points // 3
    center_end = 2 * n_points // 3
    
    # We'll try to make it sum to 1 by adjusting the height
    base_height = 1.0
    
    # Check if we want to linearly interpolate from 0 to base_height and back
    for i in range(n_points):
        if i < center_start:
            h[i] = i / center_start * base_height
        elif i >= center_end:
            h[i] = (n_points - 1 - i) / (n_points - 1 - center_end) * base_height
        else:
            h[i] = base_height
    
    # Normalize h such that integral=1
    mean_val = np.mean(h)
    if mean_val != 0:
        h = h / mean_val * dx * n_points / 2
    
    return h

def objective(h_vals, n_points):
    """Objective function to minimize: the maximum overlap"""
    dx = 2.0 / n_points
    h = np.array(h_vals)
    h = np.clip(h, 0, 1)
    return evaluate_c5(h, dx)

def solve_erdos_c5(n_points=45, budget_s=600):
    """Solve the Erdős minimum overlap problem for C5 using optimization"""
    
    # Start with the predefined initial construction
    h = initial_h_values(n_points)
    
    # We'll allow values between 0 and 1
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Set up the optimization
    def objective_wrapper(x):
        # Ensure x is within bounds
        x = np.clip(x, 0, 1)
        # Adjust for the constraint sum(h)*dx = 1 (i.e. sum(h) = n_points/2)
        # We enforce it via the constraint when sampling.
        return objective(x, n_points)
    
    # Optimize: use differential evolution for global search
    # Minimize C5 over the h values subject to integrality constraints
    
    # Ensure initial solution satisfies constraint
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum != 0:
        h *= (n_points / 2) / h_sum
    
    t0 = time.time()
    
    # Sample a few initial solutions that might be better
    best_h = h.copy()
    best_c5 = objective_wrapper(best_h)

    # Try random perturbations
    for _ in range(5000):
        if time.time() - t0 > budget_s * 0.95:
            break
        # Create a small perturbation around the current best
        perturb = np.random.normal(0, 0.05, n_points)
        new_h = best_h + perturb
        new_h = np.clip(new_h, 0, 1)
        new_h_sum = np.sum(new_h)
        if new_h_sum != 0:
            new_h *= (n_points / 2) / new_h_sum
        
        new_c5 = objective_wrapper(new_h)
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5
            # print(f"New best C5: {best_c5:.6f}")

    # Also try using differential evolution with bounds
    try:
        # Run optimization using differential evolution as a refinement
        result = differential_evolution(
            objective_wrapper,
            bounds,
            maxiter=200,
            popsize=15,
            mutation=(0.5, 1),
            recombination=0.7,
            seed=42,
            tol=1e-6
        )
        # Check if result is better
        if result.success and objective_wrapper(result.x) < best_c5:
            best_h = np.clip(result.x, 0, 1)
            best_c5 = objective_wrapper(best_h)
    except:
        pass
    
    # Final adjustment to ensure constraint is met
    best_h_sum = np.sum(best_h)
    if best_h_sum != 0:
        best_h = best_h * (n_points / 2) / best_h_sum
    
    return best_h, best_c5, n_points

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization loop to find the best C5 bound"""
    np.random.seed(seed)
    h_values, c5_bound, n_points = solve_erdos_c5(budget_s=budget_s)
    return h_values, c5_bound, n_points
