import numpy as np
import cvxpy as cp
import time
from scipy import fft
from scipy.optimize import differential_evolution
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n = 60
    dx = 2.0 / n
    half_n = n // 2
    
    # Precompute index modulo table for efficiency
    idx_mod = np.arange(n)
    
    # Function to compute autocorrelation using FFT
    def autocorrelation(h):
        # Pad with zeros to avoid circular convolution artifacts
        padded_h = np.pad(h, (0, n-1), mode='constant')
        # Convolution via FFT (with flipped kernel)
        fft_h = fft.fft(padded_h)
        # Autocorrelation is correlation of h with itself
        # correlation(h, h) = IFFT(FFT(h) * conj(FFT(h)))
        autocorr = fft.ifft(fft_h * np.conj(fft_h)).real
        # Take only the first n values (non-circular convolution)
        return autocorr[:n]
    
    # Integer optimization approach
    def solve_binary_milp(shifts_set, timeout=30):
        # binary variables h[i]
        h = cp.Variable(n, boolean=True)
        t = cp.Variable(integer=True)
        
        # Set fixing to break symmetry
        constraints = [h[0] == 1, cp.sum(h) == n // 2]
        
        # Auxiliary binary variables w[i,k] for each k in shifts_set
        w = {}
        for k in shifts_set:
            w[k] = cp.Variable((n,), boolean=True)
            for i in range(n):
                # XOR linearization
                wi = w[k][i]
                hi = h[i]
                hik = h[(i + k) % n]
                # w[i,k] <= h[i]
                constraints.append(wi <= hi)
                # w[i,k] <= h[i+k]
                constraints.append(wi <= hik)
                # w[i,k] >= h[i] + h[i+k] - 1
                constraints.append(wi >= hi + hik - 1)
                # w[i,k] binary (already ensured by declaration)
        # Objective: maximize t such that t <= sum(w[i,k]) for all k
        # which is equivalent to t <= A_k (the autocorrelation value) for all k
        # Initialize with current shifts
        for k in shifts_set:
            # A_k = sum of w[i,k]
            ak = cp.sum(w[k])
            constraints.append(t <= ak)
        
        obj = cp.Maximize(t)
        prob = cp.Problem(obj, constraints)
        
        # Solve with CBC
        try:
            prob.solve(solver=cp.CBC, max_seconds=timeout)
        except Exception:
            return None, None
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMIZED_INACCURATE]:
            h_sol = h.value
            t_sol = t.value
            return h_sol, t_sol
        else:
            return None, None
    
    # Initial shifts
    shifts = {1, half_n}
    sol_h = None
    best_t = -1
    
    # Solve initial problem
    sol_h, t_sol = solve_binary_milp(shifts)
    if sol_h is not None:
        best_t = t_sol
        current_h = sol_h.copy()
    else:
        # Fall back to a random construction
        current_h = np.zeros(n)
        ones_idx = np.random.choice(n, size=n//2, replace=False)
        current_h[ones_idx] = 1
        # Compute exact value of t
        A = np.array([np.sum(current_h * np.roll(current_h, k)) for k in range(1, n)])
        best_t = A.min()

    # Iteratively add worst shifts
    max_iter = 10
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 20:
            break
        
        last_h = current_h.copy()
        # Recompute full autocorrelations with all shifts
        A_full = np.zeros(n - 1)
        for k in range(1, n):
            A_full[k-1] = np.sum(current_h * np.roll(current_h, k))
        # Find the minimum (worst case) and its shift index
        min_idx = np.argmin(A_full)
        min_val = A_full[min_idx]
        
        # Check if we're already at the exact solution
        if min_val >= best_t:
            # TODO: Fix if solver said optimal - this will likely not be correct due to numerical instabilities
            pass
        else:
            shifts.add(min_idx + 1)
            sol_h, t_sol = solve_binary_milp(shifts, timeout=10)
            if sol_h is not None and t_sol > best_t:
                best_t = t_sol
                current_h = sol_h.copy()
        if time.time() - start_time > budget_s - 20:
            break
    
    # At this point, we have the best bitvector.
    # Now, optionally refine with gradient descent on the continuous domain
    refine_time_left = budget_s - (time.time() - start_time)
    if refine_time_left > 120:
        # Continue with continuous polishing
        
        # Define gradient ascent
        def compute_grad(h, k_star):
            # grad[i] = h[i+k_star] + h[i-k_star]
            grad = np.zeros_like(h)
            # Handle wrapping
            for i in range(len(h)):
                grad[i] = h[(i + k_star) % len(h)] + h[(i - k_star) % len(h)]
            return grad
        
        def project_h(h, target_sum):
            # Clip to [0,1]
            h = np.clip(h, 0, 1)
            # Fix sum to exactly target_sum using a pairwise swap
            current_sum = h.sum()
            residual = target_sum - current_sum
            if abs(residual) < 1e-8:
                return h
            
            # Move from elements > 0 to elements < 1
            positives = np.where(h > 0.999)[0]
            negatives = np.where(h < 0.001)[0]
            
            if len(negatives) == 0 or len(positives) == 0:
                return h
            
            # Adjust sum by moving mass from positives to negatives
            # Just move 1 unit at a time
            # For large residuals, this might be approximate but good enough.
            moved = 0
            for i in range(100):  # Limit iterations
                if residual > 1e-6 and len(positives) > 0:
                    idx1 = positives[0]
                    idx2 = negatives[0]
                    delta = min(h[idx1], 1.0 - h[idx2], abs(residual))
                    h[idx1] -= delta
                    h[idx2] += delta
                    residual -= delta
                    moved += delta
                    if abs(residual) < 1e-6:
                        break
                else:
                    break
            return h
        
        # Convert to continuous solution with binary values
        h_cont = current_h.copy()
        # Trim to [0,1]
        h_cont = np.clip(h_cont, 0, 1)
        # Ensure sum = n/2 = 30 exactly
        h_cont = project_h(h_cont, 30)
        best_h = h_cont.copy()
        best_c5 = 1 - (2 / n) * best_t
        
        # Now do gradient ascent with one shift step
        max_it = 200
        for _ in range(max_it):
            if time.time() - start_time > budget_s - 5:
                break
            
            # Find worst shift
            A_vals = np.zeros(n - 1)
            for k in range(1, n):
                A_vals[k-1] = np.sum(h_cont * np.roll(h_cont, k))
            k_star = np.argmin(A_vals)
            
            # Compute gradient
            grad = compute_grad(h_cont, k_star)
            lr = 0.1 * dx
            h_cont -= lr * grad
            
            # Projection step
            h_cont = project_h(h_cont, 30)
            
            # Update best solution
            A_vals = np.zeros(n - 1)
            for k in range(1, n):
                A_vals[k-1] = np.sum(h_cont * np.roll(h_cont, k))
            t_val = np.min(A_vals)
            c5_val = 1 - (2 / n) * t_val
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_cont.copy()
            
            if abs(c5_val - best_c5) < 1e-8:  # stabilization check
                break
        
        final_h = best_h.copy()
        final_c5 = best_c5
    else:
        # Convert back to step function h
        final_h = current_h
        # Try exact A because possible small numerical diff
        A_vals = np.zeros(n - 1)
        for k in range(1, n):
            A_vals[k-1] = np.sum(current_h * np.roll(current_h, k))
        t_best = np.min(A_vals)
        final_c5 = 1 - (2 / n) * t_best
    
    # Debug print
    # print(f"Final C5 = {final_c5:.6f}, n={n}")

    return (final_h, final_c5, n)
