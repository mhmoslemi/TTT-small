import numpy as np
import time
import random

def compute_c5(h):
    """Compute the maximum overlap integral C5 for step function h."""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return corr.max()

def propose_swap_and_update(h, corr, i, j):
    """Propose a swap between positions i and j, update correlation incrementally."""
    old_i = h[i]
    old_j = h[j]
    delta_i = old_j - old_i  # new_i - old_i = old_j - old_i
    
    n = len(h)
    new_corr = corr.copy()  # Copy for temporary update
    
    # Loop over k not i or j
    for k in range(n):
        if k == i or k == j:
            continue
        hk = h[k]
        # contribution from i->k: shift l = k - i
        idx1 = k - i + n - 1
        new_corr[idx1] += delta_i * (1 - hk)
        # contribution from k->i: shift l = i - k
        idx2 = i - k + n - 1
        new_corr[idx2] += hk * (-delta_i)
        
        # contribution from j->k: shift l = k - j
        idx3 = k - j + n - 1
        new_corr[idx3] += -delta_i * (1 - hk)
        # contribution from k->j: shift l = j - k
        idx4 = j - k + n - 1
        new_corr[idx4] += hk * (delta_i)
    
    # Add contributions from cross pairs
    # pair (i,j): shift = j - i
    idx_ij = j - i + n - 1
    new_corr[idx_ij] += delta_i
    # pair (j,i): shift = i - j
    idx_ji = i - j + n - 1
    new_corr[idx_ji] -= delta_i
    
    new_max = new_corr.max()
    
    return new_corr, new_max

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    start = time.time()
    best_c5 = np.inf
    best_h = None
    best_n = None
    
    # Candidate n values
    n_candidates = [60, 80, 100, 120, 140, 160, 180, 200]
    
    for n in n_candidates:
        if time.time() - start > budget_s - 1:
            break
        if n % 2 != 0:
            continue
            
        # Estimate max iterations based on n
        max_iters = max(2000, int(40000 / (n / 120)))
        
        # Initialize binary h with n/2 ones
        h = np.zeros(n, dtype=float)
        ones = n // 2
        indices = np.random.choice(n, ones, replace=False)
        h[indices] = 1.0
        
        # Initial correlation calculation
        corr = np.correlate(h, 1 - h, mode='full')
        cur_max = corr.max()
        cur_c5 = cur_max * (2.0 / n)
        cur_h = h.copy()
        
        # Temperature for simulated annealing
        T = 0.02
        T_decay = 0.9995
        
        # Track last improvement
        last_improvement = 0
        patience = 5000  # restart if no improvement
        
        for it in range(max_iters):
            now = time.time()
            if now - start > budget_s - 0.5:
                break
                
            # Pick two different positions
            i = np.random.randint(n)
            j = np.random.randint(n)
            while h[i] == h[j]:
                i = np.random.randint(n)
                j = np.random.randint(n)
            
            # Propose and evaluate swap
            new_corr, new_max = propose_swap_and_update(h, corr, i, j)
            new_c5 = new_max * (2.0 / n)
            
            # Acceptance criteria
            d = new_c5 - cur_c5
            if d < 0 or np.random.rand() < np.exp(-d / (T + 1e-8)):
                corr = new_corr
                cur_c5 = new_c5
                cur_max = new_max
                h[i], h[j] = h[j], h[i]
            else:
                # Revert swap if not accepted (h already holds original values)
                pass
                
            # Check for improvement
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                best_n = n
                last_improvement = it
                
            # Annealing
            T *= T_decay
            
            # Random restart if no improvement for patience iterations
            if it - last_improvement > patience:
                h = np.zeros(n, dtype=float)
                ones = n // 2
                indices = np.random.choice(n, ones, replace=False)
                h[indices] = 1.0
                corr = np.correlate(h, 1 - h, mode='full')
                cur_max = corr.max()
                cur_c5 = cur_max * (2.0 / n)
                last_improvement = it
                T = 0.02  # reset temperature
        
        # Early exit if time runs out
        if time.time() - start > budget_s - 1:
            break
    
    # If nothing found, fallback to simple 1-0 pattern
    if best_h is None:
        n = 120
        best_h = np.zeros(n, dtype=float)
        best_h[::2] = 1.0  # 1-0 pattern
        best_c5 = compute_c5(best_h)
        best_n = n
    
    return best_h, best_c5, best_n
