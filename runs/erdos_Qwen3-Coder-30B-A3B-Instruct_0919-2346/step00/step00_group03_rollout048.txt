import numpy as np
from scipy.optimize import differential_evolution
from scipy.fft import fft, ifft
import time
from cvxpy import *
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Parameters
    initial_n = 180
    target_C5 = 0.3807  # Lower than record
    dx = 2.0 / initial_n

    # Initial solution with interpolation
    initial_h = np.array(initial_h_values)
    if len(initial_h) != initial_n:
        x_orig = np.linspace(0, 2, len(initial_h))
        x_new = np.linspace(0, 2, initial_n)
        initial_h = np.interp(x_new, x_orig, initial_h)

    # Binary MILP for n=70 to get high-quality seed
    best_milp_h = initial_h.copy()
    best_milp_C5 = float('inf')
    try:
        best_milp_h, best_milp_C5 = solve_binary_milp(70, start_time, budget_s)
    except Exception as e:
        pass

    if best_milp_C5 < 0.5 or best_milp_C5 < target_C5:
        # If we found a good binary solution, use it with interpolation
        best_h = np.interp(np.linspace(0, 2, initial_n), np.linspace(0, 2, 70), best_milp_h)
    else:
        best_h = initial_h

    # DE optimization
    def objective(h):
        h = np.clip(h, 0, 1)
        # Compute overlap integral using FFT
        F = fft(h)
        G = fft(1 - h)
        corr = ifft(F * np.conj(G)).real
        max_c = max(corr)
        return dx * max_c

    def project_to_feasible(x):
        x = np.array(x)
        # Project to sum = n/2
        mean_violation = (np.sum(x) - initial_n/2) / initial_n
        x = x - mean_violation
        # Clip to [0,1]
        x = np.clip(x, 0, 1)
        return x

    # DE bounds
    bounds = [(0, 1)] * initial_n
    # Start with a good seed
    x0 = [best_h]
    for _ in range(20):
        x0.append(np.random.rand(initial_n))
    
    # Run DE with time limit
    result = differential_evolution(objective,
                                    bounds,
                                    seed=seed,
                                    maxiter=int(10000 if budget_s > 200 else 5000),
                                    popsize=30,
                                    mutation=(0.5, 1.0),
                                    recombination=0.9,
                                    callback=lambda x, convergence: time.time() - start_time > budget_s*0.95,
                                    disp=False,
                                    polish=True)
    
    if not result.success:
        # Fallback to using best haywired h
        current_h = best_h
    else:
        current_h = project_to_feasible(result.x)

    # Final iterations of mismatch swapping (local refinement)
    elapsed = time.time() - start_time
    max_swaps = 5000  # or maybe None
    swap_count = 0
    residue = 0.1  # nearby tolerance for near-zero and near-one

    def do_swap(h, i, j, shift):
        # Swap values
        temp = h[i]
        h[i] = h[j]
        h[j] = temp

        # Renormalize (small correction to maintain sum constraint)
        sum_diff = np.sum(h) - initial_n/2
        if np.abs(sum_diff) > 1e-10:
            # Do a minimal adjustment to maintain sum
            h[0] -= sum_diff / initial_n

        h = np.clip(h, 0, 1)
        return h

    # Find worst shift and do mismatches swap
    def eval_shift(h, shift):
        p = len(h)
        i_start = max(0, shift)
        i_end = min(p, p + shift)
        if shift < 0:
            i_range = np.arange(i_start, i_end)
            j_range = np.arange(-shift + i_start, -shift + i_end)
        else:
            i_range = np.arange(i_start, i_end)
            j_range = np.arange(shift + i_start, shift + i_end)
        vals = h[i_range] * (1 - h[j_range])
        return np.sum(vals)

    max_corr = -1.0
    best_shift = -1
    for shift in range(-initial_n+1, initial_n):
        corr = eval_shift(current_h, shift)
        if corr > max_corr:
            max_corr = corr
            best_shift = shift

    # Improve using swaps
    max_corr = float('inf')
    while swap_count < max_swaps and time.time() - start_time < budget_s - 30:
        current_corr = -1
        best_shift = 0
        for shift in range(-initial_n+1, initial_n):
            corr = eval_shift(current_h, shift)
            if corr > current_corr:
                current_corr = corr
                best_shift = shift

        # If at target, stop
        if current_corr * dx < target_C5:
            break

        # Find mismatches
        i_start = max(0, best_shift)
        i_end = min(initial_n, initial_n + best_shift)
        if best_shift < 0:
            i_range = np.arange(i_start, i_end)
            j_range = np.arange(-best_shift + i_start, -best_shift + i_end)
        else:
            i_range = np.arange(i_start, i_end)
            j_range = np.arange(best_shift + i_start, best_shift + i_end)

        # Identify M0 = heuristically good pairs to swap
        matches = []
        for i, j in zip(i_range, j_range):
            if current_h[i] > (1 - residue) and current_h[j] < residue:
                matches.append((i, j))
            elif current_h[i] < residue and current_h[j] > (1 - residue):
                matches.append((i, j))
            
        if len(matches) < 1:
            # Randomly move some small amount
            idx1 = np.random.randint(0, initial_n)
            idx2 = np.random.randint(0, initial_n)
            val = np.random.rand() * 0.01
            if idx1 != idx2:
                current_h[idx1] += val
                current_h[idx2] -= val
                current_h = np.clip(current_h, 0, 1)
        else:
            # Make swaps
            idx1, idx2 = matches[0]
            current_h = do_swap(current_h, idx1, idx2, best_shift)
            swap_count += 1

    # Final evaluation with possible interim clipping and sum renormalization
    current_h = np.clip(current_h, 0, 1)
    h_sum = np.sum(current_h)
    if abs(h_sum - initial_n/2) > 1e-10:
        adjust = (initial_n/2 - h_sum) / initial_n
        current_h += adjust
        current_h = np.clip(current_h, 0, 1)

    F = fft(current_h)
    G = fft(1 - current_h)
    corr = ifft(F * np.conj(G)).real
    final_C5 = dx * max(corr)

    # Returning final solution
    result_h = (current_h / np.sum(current_h)) * (initial_n / 2)
    return (result_h.tolist(), min(final_C5, target_C5), initial_n)
