import numpy as np
import time
import random

def compute_c5(h):
    """Compute the C5 bound for a given step function h."""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return corr.max()

def propose_swap_and_update(h, corr, i, j):
    """Compute the change in correlation after swapping h[i] and h[j]."""
    a = h[i]
    b = h[j]
    delta_i = b - a   # new_i - old_i
    n = len(h)
    
    # Create a copy of correlation to avoid modifying original
    new_corr = corr.copy()
    
    # Loop over all k (excluding i and j)
    for k in range(n):
        if k == i or k == j:
            continue
        hk = h[k]
        # Contribution to shift (k - i): h[i] * (1 - h[k]) -> new_i * (1 - h[k])
        idx1 = k - i + n - 1
        new_corr[idx1] += delta_i * (1 - hk)
        # Contribution to shift (i - k): h[k] * (1 - h[i]) -> h[k] * (1 - new_i)
        idx2 = i - k + n - 1
        new_corr[idx2] += hk * (-delta_i)

        # Contribution to shift (k - j): h[j] * (1 - h[k]) -> new_j * (1 - h[k])
        idx3 = k - j + n - 1
        new_corr[idx3] += -delta_i * (1 - hk)
        # Contribution to shift (j - k): h[k] * (1 - h[j]) -> h[k] * (1 - new_j)
        idx4 = j - k + n - 1
        new_corr[idx4] += hk * (delta_i)

    # Handle cross terms
    # Pair (i, j): shift (j - i)
    idx_ij = j - i + n - 1
    new_corr[idx_ij] += delta_i
    # Pair (j, i): shift (i - j)
    idx_ji = i - j + n - 1
    new_corr[idx_ji] -= delta_i

    return new_corr

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()
    best_c5 = np.inf
    best_h = None
    best_n = None
    
    # Candidate n values
    n_candidates = [60, 80, 100, 120, 140, 160, 180, 200]
    
    # Allocate time
    allocated_time = budget_s * 0.95  # Leave some buffer
    time_per_n = allocated_time / len(n_candidates)
    
    for n in n_candidates:
        if time.time() - start > budget_s - 1:
            break
            
        # Ensure n is even
        if n % 2 != 0:
            continue
            
        # Compute max iterations based on time
        max_iters = max(2000, int(50000 / (n / 120)))
        
        # Initialize binary h
        h = np.zeros(n, dtype=float)
        ones = n // 2
        h[np.random.choice(n, ones, replace=False)] = 1.0
        
        # Compute initial correlation
        corr = np.correlate(h, 1 - h, mode='full')
        cur_max = corr.max()
        cur_c5 = cur_max * (2.0 / n)
        
        # Track best
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
            best_n = n
            
        # Simulated Annealing
        T = 0.05
        T_decay = 0.9995
        
        iterations_without_improvement = 0
        max_no_improvement = 10000
        
        for it in range(max_iters):
            if time.time() - start > budget_s - 0.5:
                break
                
            # Make a random swap
            i = np.random.randint(n)
            j = np.random.randint(n)
            while h[i] == h[j]:
                i = np.random.randint(n)
                j = np.random.randint(n)
            
            # Compute new correlation
            new_corr = propose_swap_and_update(h, corr, i, j)
            new_max = new_corr.max()
            new_c5 = new_max * (2.0 / n)
            
            # Accept with probability based on temperature
            if new_c5 < cur_c5 or np.random.rand() < np.exp(-(new_c5 - cur_c5) / (T + 1e-12)):
                corr = new_corr
                h[i], h[j] = h[j], h[i]
                cur_c5 = new_c5
                cur_max = new_max
                iterations_without_improvement = 0
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    best_n = n
            else:
                iterations_without_improvement += 1
                
            # Restart if no improvement for too long
            if iterations_without_improvement > max_no_improvement:
                h = np.zeros(n, dtype=float)
                ones = n // 2
                h[np.random.choice(n, ones, replace=False)] = 1.0
                corr = np.correlate(h, 1 - h, mode='full')
                cur_max = corr.max()
                cur_c5 = cur_max * (2.0 / n)
                iterations_without_improvement = 0
            
            T *= T_decay
        
    # Fallback to best solution found
    if best_h is None:
        n = 2
        best_h = np.array([1.0, 0.0])
        best_n = 2
        best_c5 = compute_c5(best_h)
        
    return best_h, best_c5, best_n
