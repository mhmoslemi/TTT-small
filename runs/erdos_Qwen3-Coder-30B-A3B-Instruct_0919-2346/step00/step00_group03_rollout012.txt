def evaluate_erdos_solution(h):
    """
    Evaluates the Erdős constant C_5 for a given step function h.
    """
    # h is assumed to be normalized to integrate to 1 over [0,2]
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations via FFT for efficiency
    h_rev = h[::-1]
    conv = np.correlate(h, h_rev, mode="full")
    # The integral is the maximum of the convolution
    # We want max_k int h(x) * (1 - h(x+k)) dx
    # This is equivalent to max_k int h(x) - h(x)*h(x+k) dx = int h(x) dx - max_k int h(x+k)*h(x) dx
    # Since int h(x) dx = 1, it's 1 - min_k int h(x+k)*h(x) dx
    # However we computed conv = int h(x+k)*h(x) dx (reversed)
    # So the desired max is 1 - min_k conv_k
    # MAX_k int h(x)(1-h(x+k)) dx = 1 - min_k int h(x+k)*h(x) dx = 1 - (min of conv)
    conv_0_to_n = conv[n-1:] # This corresponds to the correlations for k = 0 to n-1, but k goes from 0 to 2 (= 2*len(h) -1)
    # We only need to look over k in [0,2]
    # But since sample indices from 0 to 2:
    # But x goes from 0 to 2 and samples go from 0 to n-1, dx = 2/n
    # This means the distance span between h[0] and h[n-1] is 2
    # The full correlation includes indices up to 2*n-2
    # To get valid time-shifts k in [0, 2], we map sample indices j to distance = (j-n+1)*dx
    # The range from j=n-1 to j=2*n-2 covers k from 0 to 2
    # So we only consider the range of conv starting at index n-1 up to 2*n-2
    # So we want: k = (j-n+1) * dx where j from n-1 to 2*n-2 -> k from 0 to 2
    # Number of actually valid terms from known k-limits: (2*n-2 - (n-1) + 1) = n 
    # That's all of them! 
    # Corresponding to the indices from n-1 to 2*n-2
    valid_vals = conv[n-1:]
    # Sensible: all the valid convolutions for k in [0,2]
    c5 = 1.0 - np.min(valid_vals) * dx
    return c5
