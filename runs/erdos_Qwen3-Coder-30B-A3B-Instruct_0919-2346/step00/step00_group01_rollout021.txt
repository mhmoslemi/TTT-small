def evaluate_h(h_values):
    """Evaluate the C5 bound for given step function h"""
    n = len(h_values)
    dx = 2.0 / n
    # Compute correlation of h with (1-h) using full mode
    # This is equivalent to summing h[i] * (1 - h[i+k]) over all valid i
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # Take maximum over all possible shifts
    max_corr = np.max(corr)
    return max_corr * dx
