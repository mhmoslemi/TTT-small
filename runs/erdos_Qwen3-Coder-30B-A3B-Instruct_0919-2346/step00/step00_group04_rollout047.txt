from scipy.optimize import differential_evolution
import time

def evaluate_c5(h):
    """Evaluate C5 bound for given h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute correlation of h with (1-h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # The full correlation has length 2*n_points - 1
    # We want the maximum value for k from 0 to 2
    # The range of k values corresponds to lags from 0 to n_points-1
    # This is the max among all lags (or equivalently all k values)
    max_correlation = np.max(correlation)
    return max_correlation * dx
