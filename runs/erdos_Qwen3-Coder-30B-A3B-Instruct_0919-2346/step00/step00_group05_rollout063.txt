import numpy as np
import time
import cvxpy as cp
from scipy.optimize import linprog

def evaluate(h):
    """Evaluate C5 using the correlation-based formula."""
    n = len(h)
    dx = 2.0 / n
    # Compute all shifts via convolution
    corr = np.correlate(h, 1 - h, mode="full")
    # Align and shift correctly
    corr = corr[n-1:2*n-1]  # Extract the shift-k terms
    return np.max(corr) * dx

def project_onto_capped_simplex(v, S):
    """Project v onto simplex {h | 0 <= h_i <= 1, sum h_i = S}."""
    # Binary search for the correct lambda
    v = np.clip(v, 0, 1)
    low, high = -2.0, 2.0
    tol = 1e-10
    while high - low > tol:
        mid = (low + high) / 2.0
        h = np.clip(v - mid, 0, 1)
        s = np.sum(h)
        if s > S:
            low = mid
        else:
            high = mid
    h = np.clip(v - (low + high) / 2.0, 0, 1)
    return h

def coeffs_and_const(h):
    """Compute a_{i,k} coefficients and constants c_k for all shifts."""
    n = len(h)
    # Precompute all shifts k from -(n-1) to (n-1)
    # We'll index the shifts as k + (n-1) -> [0, ..., 2n-2]
    a_matrix = np.zeros((2*n - 1, n))
    const_vec = np.zeros(2*n - 1)
    
    for k_idx, k in enumerate(range(-(n-1), n)):
        # For shift k, compute coefficients a_{i,k} and constant c_k
        # Handle zeros from zero-padding
        # a_{i,k} = 1 - h_{i+k} - h_{i-k}
        a_i = np.zeros(n)
        c_k = 0.0
        for i in range(n):
            term1 = 0.0  # h_{i+k}
            term2 = 0.0  # h_{i-k}
            if 0 <= i + k < n:
                term1 = h[i + k]
            if 0 <= i - k < n:
                term2 = h[i - k]
            a_i[i] = 1.0 - term1 - term2
            c_k += h[i] * h[i+k] if 0 <= i+k < n else 0.0
        a_matrix[k_idx] = a_i
        const_vec[k_idx] = c_k
    return a_matrix, const_vec

def linear_program(a_matrix, const_vec, n, S, maxiter=100):
    """Solve the LP: minimize t subject to a_matrix[i] @ h + const_vec[i] <= t."""
    # Variables
    h = cp.Variable(n, name='h')
    t = cp.Variable(name='t')
    # Objective
    obj = cp.Minimize(t)
    # Constraints
    constraints = [
        a_matrix @ h + const_vec <= t,
        cp.sum(h) == S,
        h >= 0,
        h <= 1
    ]
    # Formulate and solve
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        h_new = h.value
        t_val = t.value
    except Exception:
        # fallback: just use identity
        h_new = np.zeros(n)
        t_val = float('inf')
    
    return h_new, t_val

def random_feasible(n):
    """Generate a random feasible h in [0,1], sum = n/2."""
    h = np.random.rand(n)
    h = h / np.sum(h) * (n / 2)
    h = np.clip(h, 0, 1)
    return h

        n_list = [70, 140]

        if n_pass == 70:
            h = initial_h_values.copy() if 'initial_h_values' in globals() else random_feasible(n_pass)
        else:  # interpolate from 70
            h = np.repeat(initial_h_values, 2) / 2.0
            h = h[:n_pass]  # trim if oversized
        
        # Adjust to ensure sum = S
        h = h / np.sum(h) * S
        
        # Outer loop: Successive Linear Approximation
        max_iter_outer = 30
        damping = 0.7
        t_prev = float('inf')
        
        # Use small tolerance to stop early if improvement small
        for it in range(max_iter_outer):
            if time.time() - start_time > 0.95 * budget_s:
                break
            # Compute linearization coefficients
            a_matrix, const_vec = coeffs_and_const(h)
            # Solve linear program
            h_new, t_new = linear_program(a_matrix, const_vec, n_pass, S)
            if h_new is None or np.isnan(t_new):
                # fallback
                h_new = h.copy()
                t_new = evaluate(h_new)
            # Damping
            h = damping * h_new + (1 - damping) * h
            # Project back to feasible set (capped simplex)
            h = project_onto_capped_simplex(h, S)  
            # Evaluate true value
            C5_val = evaluate(h)
            if C5_val < best_C5:
                best_C5 = C5_val
                best_h = h.copy()
            # Convergence check
            if abs(t_new - t_prev) < 1e-6 and abs(C5_val - best_C5) < 1e-6:
                # Stop early if stagnating
                break
            t_prev = t_new
        
        # Polish with specific transfer moves (a form of hill climbing)

                                best_h = h_copy.copy()

        final_C5 = evaluate(best_h)
        if final_C5 < best_C5:
            best_C5 = final_C5
