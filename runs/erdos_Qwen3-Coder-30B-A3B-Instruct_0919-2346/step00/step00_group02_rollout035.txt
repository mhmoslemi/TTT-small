def evaluate_h(h):
    """Evaluate the C5 bound for a given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute full convolution of h with (1-h)
    conv = np.correlate(h, 1 - h, mode='full')
    # The maximum value is our C5 estimate
    c5 = np.max(conv) * dx
    return c5

def objective(h):
    """Objective function to minimize"""
    return evaluate_h(h)
