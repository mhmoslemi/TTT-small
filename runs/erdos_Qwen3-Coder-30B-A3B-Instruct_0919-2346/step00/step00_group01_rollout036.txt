import time
import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft
from cvxpy import *

def circular_correlation(h):
    """Compute circular correlation using FFT."""
    n = len(h)
    h_padded = np.concatenate((h, np.zeros(n)))
    h_shifted = np.concatenate((h, np.zeros(n)))
    corr = ifft(fft(h_padded) * np.conj(fft(h_shifted))).real
    return corr[n-1:2*n-1]

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound of a step function h."""
    n = len(h)
    corr = circular_correlation(h)
    min_corr = np.min(corr)
    c5 = 1 - (2.0 / n) * min_corr
    return c5

def sdp_solve(n, max_time_s=10):
    """Solve the SDP relaxation for the problem."""
    start_time = time.time()
    # Define the SDP variables
    h = Variable(n)
    t = Variable()
    X = Variable((n, n), symmetric=True)
    M = [Variable((n, n), symmetric=True) for _ in range(n)]
    
    # Build the constraints for the autocorrelation terms
    constraints = []
    for k in range(n):
        # Compute matrix M_k corresponding to shift k
        Mk = np.zeros((n, n))
        for i in range(n):
            i_plus_k = (i + k) % n
            Mk[i, i_plus_k] = 1
            Mk[i_plus_k, i] = 1
        M[k] = Mk
        constraints.append(trace(M[k] @ X) >= t)
    
    # Box constraints
    constraints += [0 <= h[i] for i in range(n)]
    constraints += [h[i] <= 1 for i in range(n)]
    
    # Sum constraint
    constraints.append(sum(h) == n / 2)
    
    # Diagonal constraints
    for i in range(n):
        constraints.append(X[i, i] == h[i])
    
    # PSD constraint
    constraints.append(X >> 0)
    
    # Solve the SDP
    objective = Maximize(t)
    prob = Problem(objective, constraints)
    
    try:
        prob.solve(solver=ECOS, verbose=False, max_iters=500)
        if prob.status == 'optimal':
            h_dual = h.value
            t_dual = t.value
            X_dual = X.value
            return (h_dual, t_dual, X_dual)
    except:
        pass
    return None, None, None

def round_h(X, h_sdp, n):
    """Round the SDP solution to a binary vector using hyperplane rounding."""
    # Cholesky decomposition
    try:
        V = np.linalg.cholesky(X)
    except:
        # If not PD, use eigendecomposition
        _, V = np.linalg.eigh(X)
        V = V[:, :np.sum(np.linalg.eigvals(X) > 1e-8)]
    
    # Sample a random vector
    g = np.random.randn(V.shape[1])
    
    # Compute scores
    scores = V @ g
    
    # Sort scores to find median threshold
    sorted_scores = np.sort(scores)
    threshold = sorted_scores[len(sorted_scores)//2]
    
    # Round to binary vector
    h_round = (scores >= threshold).astype(float)
    
    # Adjust to have exactly `n/2` ones
    num_ones = np.sum(h_round)
    if num_ones > n/2:
        # Turn some 1s into 0s
        ones_idx = np.where(h_round == 1)[0]
        to_turn_off = int(num_ones - n/2)
        indices_to_turn_off = np.sort(ones_idx)[-to_turn_off:]
        h_round[indices_to_turn_off] = 0
    elif num_ones < n/2:
        # Turn some 0s into 1s
        zeros_idx = np.where(h_round == 0)[0]
        to_turn_on = int(n/2 - num_ones)
        indices_to_turn_on = np.sort(zeros_idx)[:to_turn_on]
        h_round[indices_to_turn_on] = 1
    
    return h_round

def swap_hill_climb(h, n):
    """Perform hill climb with pair swaps to improve the vector."""
    def update_autocorr(h, idx1, idx2, k, corr_vals):
        """Update autocorrelation values affected by swapping h[idx1] and h[idx2] for shift k."""
        old_val1 = h[idx1] * h[(idx1 + k) % n]
        old_val2 = h[idx2] * h[(idx2 + k) % n]
        new_val1 = h[idx2] * h[(idx2 + k) % n]
        new_val2 = h[idx1] * h[(idx1 + k) % n]
        corr_vals[k] += (new_val1 - old_val1) + (new_val2 - old_val2)
        return corr_vals
    
    corr_vals = circular_correlation(h)
    min_corr = np.min(corr_vals)
    improved = True
    
    while improved:
        improved = False
        for i in range(n):
            for j in range(i+1, n):
                # Check if swap affects the minimum
                if (h[i] == 1 and h[j] == 0) or (h[i] == 0 and h[j] == 1):
                    old_min = min_corr
                    new_corr = corr_vals.copy()
                    
                    # Simulate swap
                    for k in range(n):
                        new_corr[k] += (h[j] * h[(j + k) % n] - h[i] * h[(i + k) % n]) + \
                                       (h[i] * h[(i + k) % n] - h[j] * h[(j + k) % n])
                    
                    new_min = np.min(new_corr)
                    if new_min > old_min:
                        h[i], h[j] = h[j], h[i]
                        corr_vals = new_corr
                        min_corr = new_min
                        improved = True
                        break
            if improved:
                break
    return h, min_corr

def gradient_fine_tune(h, n):
    """Perform gradient ascent to fine-tune h to minimize C5."""
    def compute_grad(h, min_idx):
        grad = np.zeros(n)
        # Find which shift improves the worst auto-correlation
        k_star = min_idx
        for i in range(n):
            if h[i] > 0:
                grad[i] += h[(i + k_star) % n]
            if h[(i + k_star) % n] > 0:
                grad[i] += h[(i + k_star) % n]
        return grad
    
    for _ in range(500):
        # Compute correlation values
        corr = circular_correlation(h)
        min_idx = np.argmin(corr)
        min_val = corr[min_idx]
        
        if min_val < 0.00001:
            break
            
        # Compute gradient
        grad = compute_grad(h, min_idx)
        
        # Normalize
        grad = grad / np.linalg.norm(grad)
        
        # Perform update with step-size
        step = 0.0001
        new_h = h + step * grad
        new_h = np.clip(new_h, 0, 1)
        # Ensure sum = n/2
        new_h = new_h * (n / 2) / np.sum(new_h)
        
        corr_new = circular_correlation(new_h)
        min_val_new = np.min(corr_new)
        
        if min_val_new > min_val:
            h = new_h
        else:
            break
    
    return h

def interval_optimize(n_points=200):
    """Optimize using interval construction."""
    # Solving this as an optimization over a convex set of intervals which
    # shifts are dictated by their intersection lengths.
    # For simplicity, create two pairs of symmetric intervals with 1 height and
    # widths that sum to 1
    class IntervalProblem:
        def __init__(self):
            self.n = n_points
        
        def eval_h_from_intervals(self, a, b):
            h = np.zeros(self.n)
            dx = 2.0 / self.n
            for i in range(self.n):
                x = i * dx
                pos = 0
                for aa, bb in zip(a, b):
                    if aa <= x <= bb:
                        pos = 1
                        break
                h[i] = pos
            return h
        
        def get_overlap(self, a, b, shift):
            # Compute intersection area between intervals and shifted copies
            total = 0
            for i, (aa, bb) in enumerate(zip(a, b)):
                for j, (cc, dd) in enumerate(zip(a, b)):
                    a1, b1 = aa, bb
                    a2, b2 = cc + shift, dd + shift
                    if a2 > 2: a2 -= 2
                    if b2 > 2: b2 -= 2
                    if b1 < a2: continue
                    if a1 > b2: continue
                    int_start = max(a1, a2)
                    int_end = min(b1, b2)
                    if int_start < int_end:
                        total += int_end - int_start
            return total / 2.0  # Normalization
            
        def cost(self, q):
            # q = [a1,b1,a2,b2,...] is a list of intervals
            a = q[:len(q)//2]
            b = q[len(q)//2:]
            # Mirror symmetry ensures a and b are symmetric around 1
            a_sym = [2 - x for x in reversed(b)]
            b_sym = [2 - x for x in reversed(a)]
            a = a_sym + a
            b = b_sym + b
            
            shifts = list(range(self.n))
            overlaps = [self.get_overlap(a, b, k) for k in shifts]
            return np.min(overlaps)
    
    # Start with a known good asymmetric set of intervals
    # Symmetrically pełne intervals
    start_points = [0.25, 0.75, 1.25, 1.75]
    par = [start_points[i] for i in range(4)]
    
    def objective(q):
        return -IntervalProblem().cost(q)
    
    bounds = [(0.001, 1.999)] * len(par)
    res = minimize(objective, par, bounds=bounds, method='L-BFGS-B', options={'maxiter': 200})
    
    if res.success:
        final_a = res.x[:len(res.x)//2]
        final_b = res.x[len(res.x)//2:]
        # Mirror the intervals for full range [0,2]
        final_a_sym = [2 - x for x in reversed(final_b)]
        final_b_sym = [2 - x for x in reversed(final_a)]
        final_a = final_a_sym + final_a
        final_b = final_b_sym + final_b
        
        h = IntervalProblem().eval_h_from_intervals(final_a, final_b)
        return h
    else:
        # fall back to uniform interval construction
        h = np.zeros(n_points)
        for i in range(n_points):
            x = i * 2.0 / n_points
            h[i] = (0.25 <= x <= 0.75) or (1.25 <= x <= 1.75)
        return h

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Choose reasonably sized discretization
    n = 80
    h_best = None
    c5_best = 1e10
    best_rounds = 0
    
    # Fit a few SDP rounds with error mitigation
    done = False
    
    # Phase 1: SDP-based approach
    for run_idx in range(30):
        if time.time() - start_time > budget_s - 20:
            done = True
            break
            
        h_dual, t_dual, X_d = sdp_solve(n)
        if h_dual is None:
            continue
            
        # Perform sampling rounds
        for _ in range(10):
            if time.time() - start_time > budget_s - 20:
                done = True
                break

            h_round = round_h(X_d, h_dual, n)
            h_final, min_autocorr = swap_hill_climb(h_round, n)
            h_final = gradient_fine_tune(h_final, n)
            
            c5 = 1 - (2.0 / n) * min_autocorr
            
            if c5 < c5_best:
                c5_best = c5
                h_best = h_final.copy()
                best_rounds = run_idx
                
            if c5 <= 0.379:
                done = True
                break
                
        if done:
            break
    
    # Phase 2: Interval heuristic fallback
    if c5_best >= 0.379:
        h_interval = interval_optimize(200)
        c5_interval = evaluate_erdos_solution(h_interval)
        
        if c5_interval < c5_best:
            c5_best = c5_interval
            h_best = h_interval
    
    # Final evaluation of best solution
    if h_best is not None:
        # Ensure mean is correct
        h_best = h_best * (n / 2) / np.sum(h_best)
        final_c5 = evaluate_erdos_solution(h_best)
        c5_best = final_c5
    
    return (h_best, c5_best, n)
