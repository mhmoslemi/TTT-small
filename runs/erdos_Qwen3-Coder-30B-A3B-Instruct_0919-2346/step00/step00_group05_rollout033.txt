import numpy as np
import time
from cvxpy import Variable, Minimize, Problem, sum as cp_sum, abs as cp_abs, norm
import cvxpy as cp
from scipy.optimize import linprog
from math import sqrt

def evaluate(h):
    """Evaluate the true overlap integral C5 using correlation."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    return np.max(corr) * dx

def project_onto_capped_simplex(v, S):
    """Project vector v onto {h | 0 <= h_i <= 1, sum h_i = S}"""
    v = np.clip(v, 0, 1)
    v_sum = np.sum(v)
    if abs(v_sum - S) < 1e-10:
        return v
    
    # Binary search for the correct lambda
    lambda_low, lambda_high = -2.0, 2.0
    tol = 1e-10
    for _ in range(100):
        lambda_mid = (lambda_low + lambda_high) / 2.0
        h_proj = np.clip(v - lambda_mid, 0, 1)
        h_sum = np.sum(h_proj)
        if h_sum > S:
            lambda_low = lambda_mid
        else:
            lambda_high = lambda_mid
        if abs(lambda_high - lambda_low) < tol:
            break
    return np.clip(v - (lambda_low + lambda_high) / 2.0, 0, 1)

def coeffs_and_const(h, n, S):
    """Compute a_{i,k} and c_k for all valid integer shifts k."""
    a_matrix = np.zeros((2*n - 1, n))
    c_vec = np.zeros(2*n - 1)
    # k indices from -(n-1) to (n-1)
    k_indices = list(range(-(n-1), n))
    
    for i, k in enumerate(k_indices):
        c = 0.0
        for j in range(n):
            jk = j + k
            if 0 <= jk < n:
                c += h[j] * h[jk]
                a_matrix[i, j] = 1.0 - h[jk] - h[j-k] if 0 <= j-k < n else 1.0 - h[jk]
            else:
                a_matrix[i, j] = 1.0
        c_vec[i] = c
    return a_matrix, c_vec, k_indices

def linear_program(a_matrix, c_vec, n, S):
    """Form and solve the linear program for the current iterate."""
    # Variables
    h = Variable(n)
    t = Variable(1)
    
    # Objective
    objective = Minimize(t)
    
    # Constraints
    constraints = []
    
    for i in range(2*n - 1):
        constraints.append(a_matrix[i] @ h + c_vec[i] <= t)
    
    constraints.append(cp_sum(h) == S)
    constraints.append(h >= 0)
    constraints.append(h <= 1)
    
    # Solve with ECOS or another fast LP solver
    prob = Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, verbose=False)
    
    if prob.status in ["optimal", "optimal_inaccurate"]:
        h_new = np.array(h.value).flatten()
        t_new = float(t.value)
    else:
        # fallback to scipy.LinearProgramming
        # We will solve coeffMatrix @ h + cVec <= t with the sum and bounds
        # along with 1 <= t <= 2 as bounds for t.
        # Linear program:
        # Minimize: t
        # Subject to: A @ h <= b   (one constraint per shift k)  
        #             sum(h) = S 
        #             0 <= h_i <= 1
        #             0 <= t <= 2
        
        num_constraints = 2*n - 1
        A = np.zeros((num_constraints + 2, n + 1))  # +1 for t variable
        b = np.zeros(num_constraints + 2)
        
        # Main inequality constraints
        for i in range(num_constraints):
            A[i, :n] = a_matrix[i]
            A[i, n] = -1.0  # coefficient for t
            b[i] = -c_vec[i]  # rewrite as A @ [h, t] <= b

        # Sum constraint
        A[num_constraints, :n] = 1.0
        b[num_constraints] = S
        A[num_constraints+1, :n] = -1.0
        b[num_constraints+1] = -S  # -sum(h) <= -S

        # bounds
        bounds = [(0, 1)] * n + [(0, 2)]

        # Solve using linprog
        # We now want to solve parametrically minimize c^T x
        # subject to A_ub @ x <= b_ub
        #           A_eq @ x == b_eq  
        #           lb <= x <= ub
        
        result = linprog(
            c=np.append([0]*n + [1], method='highs', options={'maxiter': 10000})
        )
        if result.success:
            h_val = result.x[:n]
            t_val = result.x[n]
            return project_onto_capped_simplex(h_val, S), t_val
        else:
            # fallback, revert to previous h
            return None, None
    return h_new, t_new

def random_feasible(n):
    """Generate a random vector feasible for h ∈ [0,1] with sum = n/2"""
    v = np.random.rand(n)
    v = project_onto_capped_simplex(v, n/2)
    return v

    start_time = time.time()

    h_best = None

    iter_limit = 50  # max outer iterations
    lambd = 0.7  # damping factor
    restarts = 200
    current_time = time.time()
    
    for restart_idx in range(restarts):
        # Stopping criterion -- Don't start new restarts if just a few seconds left
        if time.time() - start_time > 0.95 * budget_s:
            break
            
        # Initialize starting point
        if restart_idx == 0:
            h = initial_h_values.copy() if hasattr(initial_h_values, '__len__') and len(initial_h_values) == n else random_feasible(n)
        else:
            # Random initialization with damping/near-best
            h = random_feasible(n)
            
        # Project exactly onto feasible set
        h = project_onto_capped_simplex(h, S)
        c5_before = evaluate(h)
        
        # Begin iterative update
        for i in range(iter_limit):
            # Break early if time is almost exhausted
            if time.time() - start_time > 0.95 * budget_s:
                break
            a_matrix, c_vec, k_indices = coeffs_and_const(h, n, S)
            h_new, t_new = linear_program(a_matrix, c_vec, n, S)
            if h_new is None:
                break
            
            # Damping
            h_new = lambd * h_new + (1-lambd) * h
            h_new = project_onto_capped_simplex(h_new, S)
            
            h = h_new
            
            # Evaluate true value
            c5_current = evaluate(h) 
            if c5_current < c5_best:
                c5_best = c5_current
                h_best = h.copy()
                
        # Polishing
        polished_h = h_best.copy() if h_best is not None else h.copy()
        c5_polished = c5_best
        for _ in range(1000):
            # Select a point to shift mass from
            if time.time() - start_time > 0.95 * budget_s:
                break
            i = np.random.randint(0, n)
            k = np.random.randint(-n+1, n)
            j = i + k
            if 0 <= j < n and 0 < polished_h[i] and polished_h[j] < 1:
                delta = min(polished_h[i], 1 - polished_h[j])
                if delta > 1e-7:
                    polished_h[i] -= delta
                    polished_h[j] += delta
                    new_c5 = evaluate(polished_h)
                    if new_c5 < c5_polished:
                        c5_polished = new_c5
                        h_best = polished_h.copy()
                        c5_best = c5_polished
        current_time = time.time()
        
    return (h_best, c5_best, n)
