import time
import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
from scipy.fftpack import fft, ifft
from scipy.spatial.distance import cdist

def circular_correlation(h):
    n = len(h)
    # Zero pad to 2*n for full correlation
    h_padded = np.concatenate([h, np.zeros(n)])
    h_shifted = np.concatenate([h, np.zeros(n)])
    
    # Use FFT to compute circular correlation
    H = fft(h_padded)
    H_conj = np.conj(H)
    corr = ifft(H * H_conj).real
    # Return the first n elements (the circular correlation)
    corr = corr[:n]
    return corr

def min_autocorr(h):
    """Computes the minimum value of the circular correlation of h with itself."""
    corr = circular_correlation(h)
    return np.min(corr)

def evaluate_c5(h):
    # Scale to match the discrete formula
    n = len(h)
    min_corr = min_autocorr(h)
    # Convert to C5; C5 = 1 - (2/n) * min_autocorr
    c5 = 1 - (2.0 / n) * min_corr
    return c5

def sdp_solve(h0, n):
    # Solve the SDP relaxation to get a good starting point
    X = cp.Variable((n, n), symmetric=True)
    h = cp.Variable(n)
    t = cp.Variable()
    
    # Build the matrix constraints for circular autocorrelations
    # We'll form a few typical matrices for the constraints
    # For now just compute the constraints directly
    
    # Convex constraints
    constraints = [X >> 0]
    # Copy h from the solution
    for i in range(n):
        constraints.append(X[i,i] == h[i])
    
    # Box constraints
    for i in range(n):
        constraints.append(h[i] >= 0)
        constraints.append(h[i] <= 1)
    
    # Sum constraint
    constraints.append(cp.sum(h) == n / 2)
    
    # Autocorrelation constraints (simplified)
    # Form all valid constraint matrices for circular shifts:
    # t <= trace(M_k X) for k = 0,...,n-1
    for k in range(n):
        M = np.zeros((n, n))
        for i in range(n):
            j = (i + k) % n
            M[i, j] = 1
            M[j, i] = 1
        constraints.append(cp.trace(cp.multiply(M, X)) >= t)
        
    # Objective
    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constraints)
    
    # Solve with ECOS or SCS
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status == cp.OPTIMAL:
            h_sol = h.value
            t_sol = t.value
            return h_sol, t_sol
    except Exception:
        pass
    
    return h0, 0.0

def round_h(X, n):
    # Perform random hyperplane rounding
    U, s, Vt = np.linalg.svd(X)
    r = np.sum(s > 1e-8)
    W = U[:, :r] @ np.diag(np.sqrt(s[:r]))
    g = np.random.randn(r)
    h = (W.T @ g) > 0
    h = h.astype(float)
    
    # Project to exact number of ones
    num_ones = np.sum(h)
    target = n // 2
    if num_ones < target:
        # Set the smallest values to 1
        idx = np.argsort(h)[:target - num_ones]
        h[idx] = 1.0
        num_ones = target
    elif num_ones > target:
        # Set the largest values to 0
        idx = np.argsort(h)[num_ones - target:]
        h[idx] = 0.0
        num_ones = target
    
    return h

def swap_hill_climb(h, n):
    # Iteratively performs swaps to improve minimum autocorrelation
    best_h = h.copy()
    best_corr = min_autocorr(best_h)
    
    improved = True
    while improved:
        improved = False
        for i in range(n):
            for j in range(n):
                if i == j:
                    continue
                if h[i] == h[j]:
                    continue
                # Try a swap
                h_temp = h.copy()
                h_temp[i], h_temp[j] = h_temp[j], h_temp[i]
                new_corr = min_autocorr(h_temp)
                if new_corr > best_corr:
                    # Accept swap if it improves the minimum
                    best_corr = new_corr
                    best_h = h_temp.copy()
                    improved = True
                    h = h_temp.copy()
    
    return best_h

def gradient_fine_tune(h, n, max_iter=100):
    h = h.copy()
    step_size = 0.01
    for _ in range(max_iter):        
        # Compute the minimum correlation and the corresponding shift
        corr = circular_correlation(h)
        k_star = np.argmin(corr)
        min_corr = corr[k_star]
        
        # Compute gradient
        grad = np.zeros(n)
        # The gradient term is 1 at positions where h[i] * h[i + k_star] appears,
        # and 1 at (i - k_star) if k_star denotes the difference
        # Use the definition directly: should only contribute to the current term
        # For shift k_star: grad[i] = h[i + k_star] + h[i - k_star]
        
        # Gradient for autocorrelation term min_corr (corresponding to shift k_star)
        for i in range(n):
            # Add contribution of pair (i, i + k_star)
            j1 = (i + k_star) % n
            grad[i] += h[j1]
            # Add contribution of pair (i - k_star, i)
            j2 = (i - k_star) % n  
            grad[i] += h[j2]
        
        # update with projection
        h_new = h + step_size * grad
        # Project back to feasible set [0, 1]
        h_new = np.clip(h_new, 0, 1)
        # Keep sum fixed
        h_new = h_new * (n / 2) / np.sum(h_new)
        h = h_new
        step_size *= 0.95  # decay

    return h

def interval_optimize(n):
    # A toy interval heuristic to find a seed for SDP
    
    # Heuristic: find optimal intervals numerically
    # Model: a union of intervals [a_1, b_1]...[a_m, b_m], symmetric about x=1
    # Since we have box constraints: h(x) \in [0,1] and \int h(x) dx = 1.
    
    def objective(x):
        # x = [a_1, ..., a_m, b_1, ..., b_m], with symmetry: b_i = 2 - a_{m+1-i}, etc.
        # But for simplicity we ignore the symmetry for now and use mixed representation
        # Let's build intervals-one by one given list of endpoints:
        # [0,a_1], [a_1,b_1], ..., [b_{m-1}, b_m], [b_m, 2] with h = 1 in 2nd through (m-1)th
        # total area must be 1
        
        # Assume 3 intervals symmetrically placed
        a1, b1, a2, b2 = x[:4]
        
        # Enforce: 0 <= a1 < b1 < a2 < b2 <= 2 
        if not (0 <= a1 < b1 < a2 < b2 <= 2):
            return np.inf
            
        # Compute interval area
        area1 = b1 - a1  # 0 and 1? or 0 for both ends?
        area2 = b2 - a2  # h=1 on [b1, a2]
        area3 = 2 - b2  # trailing
        
        if area1 + area2 + area3 != 1:
            return np.inf
        # Uniform height = 1 for h on intervals
            
        # Now evaluate the circular autocorrelation using precomputed overlaps
        # This is equivalent to evaluating all additions of h(x) * h(x + k) for shifts k
        # We'll do a direct counting evaluation for this semi-analytic case.
        
        # We construct h as a step-wise function over [0,2]
        # Mask the intervals
        h_func = np.zeros(n)
        dx = 2.0 / n
        for i in range(n):
            x = i * dx
            if a1 <= x < b1 or a2 <= x < b2:
                h_func[i] = 1
            # otherwise it's 0
            
        corr = circular_correlation(h_func)
        min_corr = np.min(corr)
        
        return -min_corr  # We want to maximize the minimum correlation
    
    # Grid search over 4 endpoints with fixed pattern
    best_obj = np.inf
    best_x = None
    
    points_per_dim = 20
    x1 = np.linspace(0.01, 0.99, points_per_dim)
    x2 = np.linspace(0.01, 0.99, points_per_dim)
    
    for a1 in x1:
        for b1 in x2:
            if a1 >= b1:
                continue
            for a2 in x2:
                if b1 >= a2:
                    continue
                for b2 in x2:
                    if a2 >= b2:
                        continue
                    x0 = [a1, b1, a2, b2]
                    obj = objective(x0)
                    if obj < best_obj:
                        best_obj = obj
                        best_x = list(x0)
    
    if best_x is None:
        return None
    
    # Now pass best_x to optimize via SQP
    def jac(x):
        # Numerical Jacobian for sqp
        eps = 1e-5
        g = np.zeros_like(x)
        for i in range(len(x)):
            x_plus = x.copy()
            x_minus = x.copy()
            x_plus[i] += eps
            x_minus[i] -= eps
            g[i] = (objective(x_plus) - objective(x_minus)) / (2*eps)
        return g

    # Keep best as a viable solution
    try:
        res = minimize(objective, best_x, method='SLSQP', options={'maxiter': 50}, bounds=[(0.01, 0.99)]*4, jac=jac)
        if res.success:
            best_x = res.x
    except:
        pass
    
    # Convert final x to step function
    a1, b1, a2, b2 = best_x
    h = np.zeros(n)
    dx = 2.0 / n
    for i in range(n):
        x = i * dx
        if a1 <= x < b1 or a2 <= x < b2:
            h[i] = 1
    
    # Normalize
    h = h * (n / 2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    n = 80  # start with moderate size
    h_best = None
    c5_best = np.inf
    
    # tactic 1: start from initial construction
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        if len(h0) == n:
            h_best = h0.copy()
            c5_best = evaluate_c5(h_best)
    
    # tactic 2: use interval heuristic if needed
    elapsed = time.time() - t0
    if elapsed > budget_s - 15:
        return (h_best, c5_best, n) if h_best is not None else (np.array([0]*n), np.inf, n)
        
    h_interval = interval_optimize(n)
    if h_interval is not None:
        c5_interval = evaluate_c5(h_interval)
        if c5_interval < c5_best:
            c5_best = c5_interval
            h_best = h_interval.copy()
    
    # tactic 3: run SDP + rounding (multi-pass)
    num_sdp_runs = 0
    max_sdp_runs = 30
    elapsed = time.time() - t0
    
    while elapsed < budget_s - 15 and num_sdp_runs < max_sdp_runs:
        # redo SDP solve and rounding with random columns
        try:
            # Try to build a random PSD matrix to be used for SDP
            h0 = np.random.rand(n)
            h0 = h0 * (n / 2) / np.sum(h0)
            h0 = np.clip(h0, 0, 1)

            h_sdp, t_sdp = sdp_solve(h0, n)
            if h_sdp is None or np.isnan(h_sdp).any():
                h_sdp = h0
            
            # Rounding step
            assert len(h_sdp) == n
            
            # Prefer actual SDP solution for statistics
            for _ in range(3):
                h_round = round_h(np.outer(h_sdp, h_sdp), n)
                if h_round is None:
                    continue
                if np.sum(h_round) == n / 2 and np.all(h_round >= 0) and np.all(h_round <= 1):
                    h_round = h_round * (n / 2) / np.sum(h_round)
                    h_round = np.clip(h_round, 0, 1)
                    # polish
                    h_polish = swap_hill_climb(h_round, n)
                    c5 = evaluate_c5(h_polish)
                    if c5 < c5_best:
                        c5_best = c5
                        h_best = h_polish.copy()
                    
                    # second stage: gradient ascent
                    h_finetune = gradient_fine_tune(h_polish, n, max_iter=200)
                    c5ft = evaluate_c5(h_finetune)
                    if c5ft < c5_best:
                        c5_best = c5ft
                        h_best = h_finetune.copy()
                        
        except Exception:
            pass
        
        num_sdp_runs += 1
        elapsed = time.time() - t0
    
    # Final return
    if h_best is None:
        # Fallback
        h_best = np.zeros(n)
        h_best[:n//2] = 1
        h_best = h_best * (n / 2) / np.sum(h_best)
        
    c5_best = evaluate_c5(h_best)
    return (h_best, c5_best, n)
