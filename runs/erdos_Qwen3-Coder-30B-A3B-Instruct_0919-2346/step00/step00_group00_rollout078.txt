def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute the correlation integral
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum value over all shifts
    max_corr = np.max(corr)
    return max_corr * dx
