def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum value among all shifts (scaled by dx)
    max_overlap = np.max(conv) * dx
    return max_overlap
