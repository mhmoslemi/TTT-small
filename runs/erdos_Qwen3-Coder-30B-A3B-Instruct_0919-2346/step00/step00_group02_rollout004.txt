def evaluate_c5(h, dx):
    """Evaluate C5 bound for given h"""
    # Compute correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    # Extract the overlapping parts (length 2*len(h) - 1)
    # The maximum is from shifts of k from -1 to 1 (in terms of indices)
    # Since we're dealing with x in [0,2], and h is [0,2], 
    # the relevant overlaps are when k goes from 0 to 2
    # But we should measure the max correlation for the full range
    # The full correlation is computed over all shifts
    # But since we're constrained to k in [0,2], we integrate, so we get max over 
    # valid k region. That's a convolution with kernel (1-h), so we should look at
    # the maximum value in the resulting full convolution. 
    # The length will be 2*n - 1, so we look at the middle section for k in [0,2]
    # Actually, does it make sense to just compute the maximum?
    # For a step function with values 0,1, the correlation (with the "mirrored" 1-h)
    # might have some natural structure.
    #
    # Instead, in our discretization, h is sampled at n_points over [0,2] with dx = 2/n_points
    # So we measure correlation with kernel (1-h) for all integer shifts
    # Convolution: result[i] = sum of h[j] * (1-h[i-j]) over all j such that i-j is valid
    
    max_corr = np.max(corr)
    return max_corr * dx
