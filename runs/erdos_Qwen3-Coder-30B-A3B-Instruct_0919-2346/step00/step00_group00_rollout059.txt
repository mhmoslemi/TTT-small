def evaluate_overlap(h):
    """Evaluate the maximum overlap integral for given h"""
    # Compute convolution of h with (1-h)
    conv = np.correlate(h, 1 - h, mode='full')
    # The maximum value of the convolution gives us C5
    C5 = np.max(conv)
    return C5

    """Objective function to minimize C5"""
