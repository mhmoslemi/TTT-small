def evaluate_h(h):
    """Evaluate the overlap integral for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h) using np.correlate
    corr = np.correlate(h, 1 - h, mode='full')
    # max correlation value at non-zero shifts
    max_corr = np.max(corr[n-1:])  # Avoid the shift 0, which is trivial
    return max_corr * dx
