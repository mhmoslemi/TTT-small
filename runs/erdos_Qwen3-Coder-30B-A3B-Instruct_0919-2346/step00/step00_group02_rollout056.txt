def evaluate_h(h_values):
    """Evaluate the C5 bound for given h values."""
    n_points = len(h_values)
    dx = 2.0 / n_points
    # Compute the correlation integral
    corr = np.correlate(h_values, 1 - np.array(h_values), mode="full")
    # The maximum value is the one that corresponds to overlapping in the middle
    # We're looking at the correlation with shifted version
    # Taking the maximum of the shifted cross-correlations (excluding edge effects)
    max_corr = np.max(corr)
    return max_corr * dx
