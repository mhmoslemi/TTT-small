import numpy as np
from scipy import signal
import time

def evaluate_h(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation: correlate h with (1-h)
    corr = np.correlate(h, 1 - h, mode='full')
    # Extract the relevant part where shift k ranges from -n+1 to n-1
    # The middle (n-1) elements correspond to valid correlations
    # But we want the maximum of absolute shifts
    max_corr = np.max(corr[n-1:])  # Upper half including zero shift
    # Also include negative shift values
    max_corr = max(max_corr, np.max(corr[:n]))  # Lower half
    
    # Alternate approach: Use the full set of distinct shifts  
    # (full correlation uses n-1 shifts from -n+1 to n-1)
    # Discard the center element (which corresponds to zero shift, all pairs) 
    # Our max is across all valid non-zero shifts
    max_corr_val = max(np.max(corr[n:]), np.max(corr[:n-1]))
    return max_corr_val * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with some reasonable initial points
    n_points = 200
    dx = 2.0 / n_points
    
    # Use more refined initial construction for good start
    h_values = np.zeros(n_points)
    # Fill middle half of the function with 1s
    start_idx = n_points // 4
    end_idx = 3 * n_points // 4
    h_values[start_idx:end_idx] = 1
    
    # Ensure exactly half are ones and project onto valid range [0,1]
    current_sum = np.sum(h_values)
    if current_sum > n_points / 2:
        # Need to reduce some to 0s
        indices_to_reduce = np.argsort(h_values)[::-1][:int(current_sum - n_points / 2)]
        for idx in indices_to_reduce:
            h_values[idx] = 0
    elif current_sum < n_points / 2:
        # Need to increase some to 1s
        indices_to_increase = np.argsort(h_values)[:int(n_points / 2 - current_sum)]
        for idx in indices_to_increase:
            h_values[idx] = 1
    
    # Verify sum is exact
    assert abs(np.sum(h_values) - n_points / 2) < 1e-10

    current_c5 = evaluate_h(h_values)
    best_h = h_values.copy()
    best_c5 = current_c5

    iterations = 0
    max_iter = int(1e6)
    
    try:
        while iterations < max_iter and time.time() - start_time < budget_s - 1:  # Leave 1s for cleanup
            # Take a random pair swap move
            ones_pos = np.where(h_values == 1)[0]
            zeros_pos = np.where(h_values == 0)[0]
            
            if len(ones_pos) == 0 or len(zeros_pos) == 0:
                break
                
            # Select one 1 and one 0 to swap
            swap_i = np.random.choice(ones_pos)
            swap_j = np.random.choice(zeros_pos)
            
            # Make a temporary copy
            h_temp = h_values.copy()
            h_temp[swap_i] = 0
            h_temp[swap_j] = 1
            
            # Evaluate new function
            new_c5 = evaluate_h(h_temp)
            
            # Accept if better 
            if new_c5 < current_c5:
                h_values = h_temp
                current_c5 = new_c5
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h_values.copy()
            elif np.random.rand() < 0.1:  # Occasionally accept worse moves for escaping local minima
                h_values = h_temp
                current_c5 = new_c5
                
            iterations += 1

            if iterations % 5000 == 0:
                if time.time() - start_time > budget_s - 0.5:
                    break
    except:
        pass

    return (best_h, best_c5, n_points)
