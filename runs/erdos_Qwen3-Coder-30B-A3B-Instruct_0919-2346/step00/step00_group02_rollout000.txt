def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()

    h_base = initial_h_values()

    best_h = h.copy()

    max_iter = 500000

    min_temperature = 1e-6
    cooling_rate = 0.9999
    step_size = 0.05
    
    current_time = time.time()
    iter_count = 0
    
    while current_time - start_time < budget_s * 0.95 and iter_count < max_iter:
        # Random perturbation
        i = random.randint(0, n_points - 1)
        j = random.randint(0, n_points - 1)
        while i == j:
            j = random.randint(0, n_points - 1)
        
        # Compute valid range for delta
        hi = h[i]
        hj = h[j]
        max_delta = min(1 - hi, hj)
        min_delta = max(-hi, 1 - hj)

        if max_delta <= 0 or min_delta >= 0:
            current_time = time.time()
