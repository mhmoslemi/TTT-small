import numpy as np
import time

def evaluate(h):
    """Computes C5 = max_k integral h(x)(1-h(x+k))dx using correlation."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return np.max(c) * dx

def get_max_shifts(h):
    """Returns list of integer shifts k where correlation achieves max."""
    n = len(h)
    c = np.correlate(h, 1 - h, mode='full')
    max_val = c.max()
    indices = np.where(c == max_val)[0]
    shifts = indices - (n - 1)  # Convert index to shift value
    return shifts.tolist()

def project_to_capped_simplex(v, S):
    """Project vector v onto the capped simplex: sum h = S, 0 <= h <= 1."""
    # Clip to [0,1] first (for speed and safety)
    v = np.clip(v, 0, 1)
    lo, hi = -2.0, 2.0
    for _ in range(50):
        lam = (lo + hi) / 2
        x = np.clip(v - lam, 0, 1)
        s = x.sum()
        if s > S:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo + hi) / 2, 0, 1)
    # Adjust slightly to meet exact sum
    diff = x.sum() - S
    if abs(diff) > 1e-10:
        # Distribute the difference to nearest non-bound entries
        if diff > 0:
            # Reduce some values
            indices = np.where(x > 0)[0]
            # Sort by smallest values first to reduce small values
            sort_order = np.argsort(x[indices])
            for i in sort_order:
                idx = indices[i]
                reduce = min(diff, x[idx])
                x[idx] -= reduce
                diff -= reduce
                if diff <= 1e-10:
                    break
        else:
            # Increase some values
            indices = np.where(x < 1)[0]
            sort_order = np.argsort(1 - x[indices])
            for i in sort_order:
                idx = indices[i]
                add = min(-diff, 1 - x[idx])
                x[idx] += add
                diff += add
                if diff >= -1e-10:
                    break
    return x

def gradient_max_shifts(h, shifts):
    """Compute averaged subgradient for a set of shifts."""
    N = len(h)
    g = np.zeros(N)
    for k in shifts:
        # Compute left and right shift using array slicing
        if k >= 0:
            left = np.concatenate((np.zeros(k), h[:N-k]))
            right = np.concatenate((h[k:], np.zeros(k)))
        else:
            k_abs = -k
            left = np.concatenate((h[k_abs:], np.zeros(k_abs)))
            right = np.concatenate((np.zeros(k_abs), h[:N-k_abs]))
        g += 1 - left - right
    return g / len(shifts)

def random_feasible(n):
    """Generate a random feasible vector summing to n/2."""
    v = np.random.rand(n)
    S = n / 2.0
    return project_to_capped_simplex(v, S)

            h += np.random.uniform(-0.03, 0.03, n)

            g = gradient_max_shifts(h, K)

                alpha *= 0.5

            else:
                alpha = min(alpha * 1.02, 0.5)

        cur_c5 = evaluate(h)

            if new_c5 < cur_c5:
                h, cur_c5 = h_new, new_c5
