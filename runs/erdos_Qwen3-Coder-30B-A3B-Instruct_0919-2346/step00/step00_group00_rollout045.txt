def project_to_capped_simplex(h, target_sum):
    """Project h onto the capped simplex with sum = target_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if np.isclose(current_sum, target_sum):
        return h
    
    # Water-filling approach
    sorted_indices = np.argsort(h)[::-1]
    h_sorted = h[sorted_indices]
    
    # Find splitting point
    cumsum = np.cumsum(h_sorted)
    n = len(h_sorted)
    i = 0
    while i < n and cumsum[i] - (i+1) * h_sorted[i] > target_sum - (i+1):
        i += 1
    if i == n:
        # All elements should be set to 0
        return np.zeros_like(h)
    
    # Compute threshold lambda
    if i == 0:
        lambda_val = (cumsum[i] - target_sum) / (i + 1)
    else:
        lambda_val = (cumsum[i-1] - target_sum) / i
    
    # Apply threshold
    result = np.maximum(0, h - lambda_val)
    return result

def compute_correlation(h, dx):
    """Compute correlation to use in computing C5."""
    corr = np.correlate(h, 1-h, mode='full')
    return corr * dx

def gradient_wrt_h(h, alpha, dx):
    """Compute the gradient of F_alpha with respect to h."""
    n = len(h)
    corr = np.correlate(h, 1-h, mode='full')
    exp_term = np.exp(alpha * corr)
    weights = exp_term / np.sum(exp_term)
    
    # Compute convolutions
    c1 = np.correlate(weights, 1-h, mode='full')
    c2 = np.correlate(weights[::-1], h, mode='full')
    
    # Cut to size
    c1 = c1[:n]
    c2 = c2[:n]
    
    return (c1 - c2) / alpha

def compute_F_alpha(h, alpha, dx):
    """Compute the log-sum-exp surrogate function."""
    corr = np.correlate(h, 1-h, mode='full')
    return (dx / alpha) * np.log(np.sum(np.exp(alpha * corr)))

def deterministic_annealing(h0, n, dx, alpha0=0.5, gamma=1.3, eta=0.1, delta=0.99, 
                            max_alpha=200, min_improvement=1e-5, verbose=False):
    """Run deterministic annealing with temperature schedule."""
    h = h0.copy()
    alpha = alpha0
    current_F = compute_F_alpha(h, alpha, dx)
    last_F = current_F
    last_alpha_improvement = 0
    
    # Annealing loop
    iter_num = 0
    while alpha < max_alpha:
        # Simple gradient descent step
        grad = gradient_wrt_h(h, alpha, dx)
        h -= eta * grad
        h = project_to_capped_simplex(h, n / 2)
        
        # Reduce step size
        eta *= delta
        
        iter_num += 1
        if iter_num % 50 == 0:  # Periodically recompute objective
            current_F = compute_F_alpha(h, alpha, dx)
            if verbose:
                print(f"Alpha={alpha:.2f}: F_alpha={current_F:.6f}")
            
            # Check for convergence or improvement threshold
            improvement = last_F - current_F
            if improvement < min_improvement and alpha > 50 and iter_num > 500:
                break
            last_F = current_F
            
        # Update temperature
        if iter_num % 200 == 0:
            alpha *= gamma
            eta = 0.1  # Reset step size
            if verbose:
                print(f"Temperature increased to {alpha:.2f}")
    
    return h, compute_correlation(h, dx).max()

def binary_round(h, n):
    """Round h to binary values that maintain the required sum."""
    # Sort by values in descending order
    indices = np.argsort(h)[::-1]
    h_bin = np.zeros_like(h)
    h_bin[indices[:n//2]] = 1.0
    return h_bin

def evaluate_c5_binary(h):
    """Return the exact C5 value for a binary h."""
    corr = np.correlate(h, 1-h, mode='full')
    return corr.max()
