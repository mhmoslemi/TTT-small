import numpy as np
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start = time.time()
    
    def evaluate(h):
        n = len(h)
        dx = 2.0 / n
        O = np.correlate(h, 1 - h, mode='full')
        return O.max() * dx
    
    # Use priority list of n values for testing
    lengths = [64, 80, 96, 112, 128, 144, 160, 176, 192, 224, 256, 320]
    lengths = [n for n in lengths if n <= 800]
    
    # Precompute known good starting points
    candidates = []
    
    # 1. Use scaled version of initial_h_values if available
    if 'initial_h_values' in globals():
        base = np.array(initial_h_values, dtype=float)
        for n in lengths:
            # Interpolate to target n
            x_old = np.linspace(0, 2, len(base), endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            h_interp = np.interp(x_new, x_old, base)
            # Convert to binary with correct weight
            h_bin = (h_interp > 0.5).astype(int)
            # Adjust sum to exactly n/2
            current_sum = h_bin.sum()
            target = n // 2
            if current_sum > target:
                idx = np.where(h_bin == 1)[0]
                np.random.shuffle(idx)
                h_bin[idx[:current_sum - target]] = 0
            elif current_sum < target:
                idx = np.where(h_bin == 0)[0]
                np.random.shuffle(idx)
                h_bin[idx[:target - current_sum]] = 1
                
            candidates.append((h_bin, n))
    
    # 2. Random binary sequences  
    for n in lengths:
        for _ in range(2):
            h = np.random.choice([0, 1], size=n, p=[0.5, 0.5])
            # Ensure correct sum
            current_sum = h.sum()
            target = n // 2
            if current_sum > target:
                idx = np.where(h == 1)[0]
                np.random.shuffle(idx)
                h[idx[:current_sum - target]] = 0
            elif current_sum < target:
                idx = np.where(h == 0)[0]
                np.random.shuffle(idx)
                h[idx[:target - current_sum]] = 1
            candidates.append((h, n))
    
    # Evaluate all candidates and keep top 5
    candidate_scores = []
    for h, n in candidates:
        if time.time() - start > budget_s * 0.95:
            break
        c5 = evaluate(h)
        candidate_scores.append((c5, h, n))
    
    candidate_scores.sort()
    top_candidates = candidate_scores[:5]
    
    # Now refine each top candidate with local search
    best_h = None
    best_c5 = float('inf')
    best_n = None
    
    remaining_time = budget_s * 0.95 - (time.time() - start)
    time_per_candidate = remaining_time / max(len(top_candidates), 1)
    
    for c5, h, n in top_candidates:
        if time.time() - start > budget_s * 0.95:
            break
        
        h_refined, c5_refined = local_search(h, n, time_per_candidate)
        
        if c5_refined < best_c5:
            best_c5 = c5_refined
            best_h = h_refined
            best_n = n
        
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), float(best_c5), int(best_n))

def local_search(h, n, time_budget):
    start = time.time()
    current_h = h.copy()
    current_c5 = evaluate(current_h)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Maintain indices for speed
    ones_idx = list(np.where(current_h == 1)[0])
    zeros_idx = list(np.where(current_h == 0)[0])
    
    # Simulated Annealing with exponential cooling
    max_iter = 2000000
    temp_start = 0.05
    temp_end = 1e-4
    
    for it in range(max_iter):
        if time.time() - start > time_budget:
            break
            
        # Temperature schedule
        temp = temp_start * (temp_end / temp_start) ** (it / max_iter)
        
        # Pick two indices for swap: one 1, one 0
        i = random.choice(ones_idx)
        j = random.choice(zeros_idx)
        
        # Swap values
        current_h[i] = 0
        current_h[j] = 1
        
        # Update indices
        ones_idx[ones_idx.index(i)] = j
        zeros_idx[zeros_idx.index(j)] = i
        
        # Evaluate new solution
        new_c5 = evaluate(current_h)
        
        # Accept or reject based on SA probability
        if new_c5 <= current_c5 or random.random() < np.exp((current_c5 - new_c5) / temp):
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
        else:
            # Revert swap
            current_h[i] = 1
            current_h[j] = 0
            ones_idx[ones_idx.index(j)] = i
            zeros_idx[zeros_idx.index(i)] = j
    
    return best_h, best_c5
