import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_h(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h) using full mode
    corr = np.correlate(h, 1 - h, mode='full')
    # Maximum overlap integral
    max_corr = np.max(corr)
    return max_corr * dx
