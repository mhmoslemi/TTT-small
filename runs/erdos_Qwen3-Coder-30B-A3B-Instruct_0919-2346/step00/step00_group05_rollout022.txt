import numpy as np
import time
from scipy.optimize import linprog
import cvxpy as cp

def evaluate(h):
    """Compute C5 via correlation."""
    n = len(h)
    dx = 2.0 / n
    # Compute all overlaps for shifts k = -(n-1) to n-1
    # Use full convolution mode to get all valid overlaps
    overlap = np.correlate(h, 1 - h, mode="full")
    # Sum over all shifts, normalized
    C5 = np.max(overlap) * dx
    return C5

def project_onto_capped_simplex(v, S):
    """Project vector v onto {h : 0 ≤ h_i ≤ 1, Σ h_i = S}."""
    v = np.array(v)
    low, high = -2, 2
    tol = 1e-10
    while high - low > tol:
        mid = (low + high) / 2
        h = np.clip(v - mid, 0, 1)
        if np.sum(h) > S:
            low = mid
        else:
            high = mid
    return np.clip(v - (low + high) / 2, 0, 1)

def coeffs_and_const(h):
    """Compute linearized coefficients and constants for all shifts."""
    n = len(h)
    a = np.zeros((n, 2 * n - 1))
    c = np.zeros(2 * n - 1)
    for k in range(-(n - 1), n):
        # Index in the shift array: k -> index k + n - 1
        k_idx = k + n - 1
        c[k_idx] = 0.0
        for i in range(n):
            # Access h_{i+k} (with zero padding)
            j = i + k
            if 0 <= j < n:
                c[k_idx] += h[i] * h[j]
                a[i, k_idx] = 1.0 - h[j] - h[i - k] if 0 <= i - k < n else 1.0
            else:
                a[i, k_idx] = 1.0
    return a, c

def linear_program(a, c, n, S):
    """Solve the linearized program."""
    # Variables: h (n), t (1)
    # Objective: minimize t
    # Constraints: a * h + c <= t for all k
    #              sum(h) = S
    #              0 <= h <= 1

    # For scipy, treat it as a standard LP
    # Objective: [0,...,0,1]
    # Implementation: one variable per h_i, one for t
    # Number of constraints:
    # len(c) (for the linearization terms) + 1 (sum equality) + 2*n (bounds)
    # == len(c) + 1 + 2*n rows in A_eq (and A_ub), but simpler to construct as much as scale
    # Let us use cvxpy for numerical robustness and easier constraint handling.
    h = cp.Variable(n)
    t = cp.Variable(1)
    # Define all constraints: a * h + c <= t
    constraints = []
    for i in range(len(c)):
        # a[:,i] * h + c[i] <= t
        # Or equivalent
        expr = a[:, i] @ h + c[i]
        constraints.append(expr <= t)
    # Sum constraint
    constraints.append(cp.sum(h) == S)
    # Box constraints
    constraints.append(h >= 0)
    constraints.append(h <= 1)
    problem = cp.Problem(cp.Minimize(t), constraints)
    try:
        problem.solve(solver=cp.ECOS, verbose=False)
        return h.value, t.value
    except:
        # Fallback: use np.linalg.lstsq or solve as constrained least squares
        # Try crude residual method: solve h = argmin (a*h + c - t)^2 + α * (sum(h) - S)^2 + β*||h||^2
        # But use a simple succeeded.
        # Recover a feasible point via partial solve: ignore one constraint per iteration until feasibility or approx.
        return None, float('inf')
