```python
import time
import numpy as np
from scipy import fft
import cvxpy as cp
from cvxpy import Minimize, Problem, Variable, sum as cp_sum, maximum as cp_max, minimum as cp_min

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n = 60
    dx = 2.0 / n

    def solve_milp(hints=None):
        # Binary variable h
        h = Variable(n, boolean=True)
        # Introduce auxiliary binary variables w[i,k] for k in K
        # We'll build this incrementally
        t = Variable(integer=True)
        # Constraints
        constraints = [cp_sum(h) == n // 2]
        constraints.append(h[0] == 1)  # break symmetry
        # Add initial K
        K = [1, n // 2]
        if hints is not None:
            K = list(set(K + hints))
        # List of w_{i,k} variables and constraints
        all_w = {}
        all_a = {}
        for k in K:
            w = Variable(n, boolean=True)
            all_w[k] = w
            # linearisation f.w. w = h_i AND h_{i+k}
            for i in range(n):
                ipk = (i + k) % n
                constraints.append(w[i] <= h[i])
                constraints.append(w[i] <= h[ipk])
                constraints.append(w[i] >= h[i] + h[ipk] - 1)
            a_k = cp_sum(w)
            all_a[k] = a_k
            constraints.append(t <= a_k)
        # Optimize
        objective = Minimize(-t)  # equivalent to maximize t
        prob = Problem(objective, constraints)
        prob.solve(solver=cp.CBC, maxSeconds=min(30, budget_s / 5))
        if prob.status in ['optimal', 'optimal_inaccurate']:
            h_sol = np.round(h.value).astype(int)
            return h_sol, K
        else:
            return None, K

    def get_worst_shift(h_sol, n):
        # Compute all A_k values via circular autocorrelation
        A = np.zeros(n)
        for k in range(n):
            A[k] = np.sum(h_sol * np.roll(h_sol, k))
        # Find min A_k
        a_min = np.min(A)
        k_min = np.argmin(A)
        return a_min, k_min

    def compute_exact_C5(h_values):
        h = h_values
        # Circular autocorrelation using FFT
        H = fft.fft(h)
        autocorr = fft.ifft(H * np.conj(H))
        A = np.real(autocorr[:n])  # the first n values are relevant
        # C5 = 1 - (2/n) * min(A_k) = 1 - (2/n) * min(A)
        min_A = np.min(A)
        C5 = 1 - (2.0 / n) * min_A
        return C5

    def polish_solution(h_bin, max_iter=200, step_size=0.01):
        # Exact copy of the original h_bin
        h_cont = h_bin.astype(float)
        # Compute all A_k
        A = np.zeros(n)
        h0 = h_cont.copy()
        for k in range(n):
            A[k] = np.sum(h0 * np.roll(h0, k))
        a_min = np.min(A)
        k_min = np.argmin(A)
        a_min_prev = a_min
        step = 0
        while step < max_iter and (time.time() - start_time < budget_s - 10):
            g = np.zeros(n)
            g = np.roll(h_cont, k_min) + np.roll(h_cont, -k_min)
            h_cont = h_cont + step_size * g
            h_cont = np.clip(h_cont, 0, 1)
            residual = np.sum(h_cont) - n / 2
            if abs(residual) > 1e-6:
                # pairwise swap
                idx_l = np.where(h_cont > 1e-6)[0]
                idx_h = np.where(h_cont < 1 - 1e-6)[0]
                if len(idx_l) > 0 and len(idx_h) > 0:
                    i0 = idx_l[0]
                    i1 = idx_h[0]
                    if residual > 0:
                        d = min(residual, h_cont[i0])
                        h_cont[i0] -= d
                        h_cont[i1] += d
                    else:
                        d = min(-residual, 1 - h_cont[i1])
                        h_cont[i0] += d
                        h_cont[i1] -= d
            # Recompute A_k
            A = np.zeros(n)
            for k in range(n):
                A[k] = np.sum(h_cont * np.roll(h_cont, k))
            a_min = np.min(A)
            if a_min > a_min_prev:
                a_min_prev = a_min
                k_min = np.argmin(A)  # Update worst shift
            step += 1
        return h_cont

    # Initial MILP solve
    h_sol, K = solve_milp()
    if h_sol is None:
        # fallback to binary random
        h_sol = np.zeros(n, dtype=int)
        ones_indices = np.random.choice(n, n // 2, replace=False)
        h_sol[ones_indices] = 1
        h_sol = h_sol.astype(int)

    # Cut generation loop
    last_a_min = -1
    iteration_count = 0
    while time.time() - start_time < budget_s - 100 and iteration_count < 20:
        a_min, k_min = get_worst_shift(h_sol, n)
        if a_min >= last_a_min:
            break
        last_a_min = a_min
        K.append(k_min)
        # Re-solve
        h_sol, _ = solve_milp(hints=K)
        if h_sol is None:
            break
        iteration_count += 1

    # Final evaluation
    C5 = compute_exact_C5(h_sol)
    h_final = h_sol.astype(float)

    if C5 > 0.3807 and budget_s - (time.time() - start_time) > 120:
        # Polish
        h_final = polish_solution(h_sol)
        C5 = compute_exact_C5(h_final)

    if C5 > 0.3807 and budget_s - (time.time() - start_time) > 100:
        # Orthogonal spectral heuristic as fallback
        def solve_spectral():
            # cosine basis parameterization in 4 dimensions
            M = 4
            alpha = Variable(M)
            h = 0.5 + alpha[0] * np.cos(2 * np.pi * np.arange(n) / n) + \
                alpha[1] * np.cos(2 * np.pi * 2 * np.arange(n) / n) + \
                alpha[2] * np.cos(2 * np.pi * 3 * np.arange(n) / n) + \
                alpha[3] * np.cos(2 * np.pi * 4 * np.arange(n) / n)
            h = np.clip(h, 0, 1)
            # sum = 1
            constraints = [cp_sum(h) == n / 2]
            # max - 1 / beta * log( sum exp(-beta A_k) )
            # shape
            H = fft.fft(h)
            A = np.real(fft.ifft(H * np.conj(H)))
            beta = 10
            F = -1/beta * cp.log(cp_sum(cp.exp(-beta * A)))
            obj = cp.Maximize(F)
            # Clip constraints
            constraints.append(alpha[0] >= -0.5)
            constraints.append(alpha[0] <= 0.5)
            constraints.append(alpha[1] >= -0.5)
            constraints.append(alpha[1] <= 0.5)
            constraints.append(alpha[2] >= -0.5)
            constraints.append(alpha[2] <= 0.5)
            constraints.append(alpha[3] >= -0.5)
            constraints.append(alpha[3] <= 0.5)
            prob = Problem(obj, constraints)
            prob.solve(solver=cp.SCS, max_iters=100, verbose=False)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                h_sol = np.clip(h.value, 0, 1)
                h_sol = np.round(h_sol)
                return h_sol
            return None
        # Try spectral start
        h_spectral = solve_spectral()
        if h_spectral is not None:
            c_spectral = compute_exact_C5(h_spectral)
            if c_spectral < C5:
                h_final = h_spectral.astype(float)
                C5 = c_spectral

    return h_final, C5, n
```
