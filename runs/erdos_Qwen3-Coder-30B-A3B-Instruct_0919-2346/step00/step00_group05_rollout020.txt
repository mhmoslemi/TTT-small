import numpy as np
import time

# Helper functions

def get_max_shifts(h):
    """Returns list of integer shifts k where correlation achieves maximum."""
    N = len(h)
    c = np.correlate(h, 1 - h, mode='full')
    max_val = c.max()
    indices = np.where(c == max_val)[0]
    shifts = indices - (N - 1)  # because index N-1 corresponds to shift 0
    return shifts.tolist()

def gradient_max_shifts(h, shifts):
    """Compute averaged subgradient for a list of maximizing shifts."""
    N = len(h)
    g = np.zeros(N)
    for k in shifts:
        if k >= 0:
            left = np.concatenate((np.zeros(k), h[:N-k]))
            right = np.concatenate((h[k:], np.zeros(k)))
        else:
            k_abs = -k
            left = np.concatenate((h[k_abs:], np.zeros(k_abs)))
            right = np.concatenate((np.zeros(k_abs), h[:N-k_abs]))
        g += 1 - left - right
    g /= len(shifts)
    return g

def project_to_capped_simplex(v, S):
    """Project vector v onto capped simplex: sum = S, 0 <= v_i <= 1."""
    lo, hi = -2.0, 2.0  # wide enough interval for rootfinding
    for _ in range(50):
        lam = (lo + hi) / 2
        x = np.clip(v - lam, 0, 1)
        s = x.sum()
        if s > S:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo+hi)/2, 0, 1)
    return x

def random_feasible(N):
    """Generate a random feasible vector satisfying sum = N/2."""
    v = np.random.rand(N)
    return project_to_capped_simplex(v, N / 2)

def evaluate(h):
    """Evaluate C5 = max_k integral h(x)(1 - h(x + k)) dx = max correlation * dx."""
    N = len(h)
    dx = 2.0 / N
    c = np.correlate(h, 1 - h, mode='full')
    return np.max(c) * dx

# Main optimization routine
def run(seed=42, budget_s=600, **kwargs):

    best_h = h0.copy()
    best_c5 = evaluate(best_h)

    # Multi-restart loop
    while time.time() - start < budget_s * 0.95:
        # Initialize h for this restart
        if np.random.rand() < 0.2:
            h = random_feasible(n)
        else:
            h = best_h.copy()
            h += np.random.uniform(-0.03, 0.03, n)
            h = project_to_capped_simplex(h, S)

        # Gradient descent phase
        alpha = 0.12
        last_improve = 0
        for it in range(2000):
            if time.time() - start > budget_s * 0.90:
                break
            cur_c5 = evaluate(h)
            if cur_c5 < best_c5:
                best_c5, best_h = cur_c5, h.copy()
                last_improve = it

            # Compute shifts maximizing correlation
            max_shifts = get_max_shifts(h)
            g = gradient_max_shifts(h, max_shifts)

            # Gradient step and projection
            v = h - alpha * g
            h = project_to_capped_simplex(v, S)

            # Adaptive step size
            if it - last_improve > 200:
                alpha *= 0.5
                last_improve = it
            else:
                alpha = min(alpha * 1.02, 0.5)

        # Local targeted random improvement
        cur_c5 = evaluate(h)
        for s in range(5000):
            if time.time() - start > budget_s * 0.99:
                break
            max_shifts = get_max_shifts(h)
            k = np.random.choice(max_shifts)
            # Find donor-recipient pairs for this shift k
            donors = np.where(
                (np.arange(n) + k < n) &
                (h > 1e-6) &
                (h[np.arange(n) + k] < 1 - 1e-6)
            )[0]
            if donors.size == 0:
                continue
            i = np.random.choice(donors)
            j = i + k
            eps_max = min(h[i], 1 - h[j])
            eps = eps_max * np.random.rand() * 0.5
            h_new = h.copy()
            h_new[i] -= eps
            h_new[j] += eps
            new_c5 = evaluate(h_new)
            if new_c5 < cur_c5:
                h, cur_c5 = h_new, new_c5
                if new_c5 < best_c5:
                    best_c5, best_h = new_c5, h_new.copy()

    return best_h, float(best_c5), n
