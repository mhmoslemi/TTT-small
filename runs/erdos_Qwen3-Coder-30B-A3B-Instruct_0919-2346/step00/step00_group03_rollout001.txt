def evaluate_c5(h):
    """Evaluate C5 for given step function h"""
    n = len(h)
    dx = 2.0 / n

    conv = np.correlate(h, 1 - h, mode='full')
    # Take maximum over all shifts, multiply by dx
    c5 = np.max(conv) * dx
    return c5
