import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import differential_evolution

def evaluate_erdos_solution(h):
    """Evaluate the Erdős overlap constant for given step function h"""
    # Compute overlap integral using np.correlate
    n = len(h)
    dx = 2.0 / n
    # Compute cross-correlation of h with (1-h)
    corr = correlate(h, 1 - h, mode='full')
    # Max overlap over all shifts
    max_overlap = np.max(corr) * dx
    return max_overlap

def initial_h_values():
    """Return initial h values as provided in the environment"""
    # Return the same initial configuration as in the problem
    # Let's generate a reasonable initial solution
    # Using a truncated Gaussian shape centered at 1 with variance allowing
    # it to be bounded in [0,1] and integral 1 over [0,2]
    n = 45
    x = np.linspace(0, 2, n)
    h = np.exp(-((x - 1)**2) / 0.2)  # Gaussian centered at x=1
    h = np.clip(h, 0, 1)  # Clip to [0,1] bounds
    # Normalize to integral 1
    integral = np.sum(h) * (2.0 / n)
    h = h / integral * 1.0  # Scale so integral is exactly 1
    return h

def run(seed=42, budget_s=600, **kwargs):
    # Set random seed
    np.random.seed(seed)
    
    # Initial setup
    n_points = 150  # Reasonable size for speed and quality
    dx = 2.0 / n_points
    h = initial_h_values()
    
    # Adjust initial array size if needed
    if len(h) != n_points:
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    
    # Ensure sum constraint
    h_sum = np.sum(h)
    h = h * (n_points / 2) / h_sum
    
    # Objective: minimize max correlation = maximize min of correlations
    def objective(h_vec):
        # Ensure h remains in [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Ensure sum constraint
        h_sum = np.sum(h_vec)
        h_vec = h_vec * (n_points / 2) / h_sum
        # Compute correlations
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Return negative of minimum correlation to convert to maximization problem
        return -np.min(corr) * dx
    
    # Create bounds for gradient-based optimization
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use differential evolution to find good local minima
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Optimization loop using simulated annealing with restarts
    max_iter = 100000
    current_h = h.copy()
    current_c5 = evaluate_erdos_solution(current_h)
    
    temperature = 1.0
    cooling_rate = 0.9999
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Random perturbation
        i, j = np.random.choice(n_points, 2, replace=False)
        delta = np.random.uniform(-0.1, 0.1)
        
        # Adjust delta based on constraints
        hi = current_h[i]
        hj = current_h[j]
        max_delta = min(1 - hi, hj)  # Constraint 1: hi + delta <= 1, hj - delta <= 1
        min_delta = max(-hi, 1 - hj) # Constraint 2: hi + delta >= 0, hj - delta >= 0
        
        delta = np.clip(delta, min_delta, max_delta)
        
        # Apply change
        new_h = current_h.copy()
        new_h[i] = np.clip(hi + delta, 0, 1)
        new_h[j] = np.clip(hj - delta, 0, 1)
        
        # Renormalize to maintain sum = n_points / 2
        new_sum = np.sum(new_h)
        new_h = new_h * (n_points / 2) / new_sum
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(new_h)
        
        # Accept or reject based on simulated annealing criteria
        if new_c5 < current_c5 or np.random.rand() < np.exp((current_c5 - new_c5) / (temperature + 1e-8)):
            current_h = new_h
            current_c5 = new_c5
            
        # Update best solution
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            
        # Cool down
        temperature *= cooling_rate
    
    # Final evaluation of best solution
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
