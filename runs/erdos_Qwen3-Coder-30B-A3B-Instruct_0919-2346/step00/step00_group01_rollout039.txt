import numpy as np
import cvxpy as cp
import time
from scipy import fft
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=600, **kwargs):

    # Parameters
    n = 60
    dx = 2.0 / n
    target_bound = 0.3807
    max_iters = 15
    
    # Start timing
    start_time = time.time()
    
    # Pre-computed modulo indices for performance
    mod_indices = np.arange(n)
    
    # Initial shifts to start with
    shifts = [1, n//2]
    
    # Track best solution
    best_h = None
    best_c5 = float('inf')
    
    # Function to compute circular autocorrelation using FFT
    def circular_autocorr(h_vec):
        # Normalize by n
        h_padded = np.concatenate([h_vec, np.zeros(n)])
        corr = fft.ifft(fft.fft(h_padded) * np.conj(fft.fft(h_padded)))
        return np.real(corr[:n])
    
    # Check if we have enough time to continue
    def time_left():
        return budget_s - (time.time() - start_time)
    
    # First MILP solving with initial shifts
    while True:
        if time_left() < 20:
            break
            
        # Build model
        h_vars = cp.Variable(n, boolean=True)
        t = cp.Variable()
        
        # Fix h[0] = 1 to break symmetry
        constraints = [h_vars[0] == 1]
        
        # Sum constraint
        constraints.append(cp.sum(h_vars) == n//2)
        
        # Add auxiliary variables and constraints for current shifts
        w_vars = {}
        a_vars = {}
        
        for k_idx, k in enumerate(shifts):
            w_vars[k] = cp.Variable(n, boolean=True)
            # Setup constraints for w[i,k] = h[i] * h[i+k]
            for i in range(n):
                i_k = (i + k) % n
                constraints.extend([
                    w_vars[k][i] <= h_vars[i],
                    w_vars[k][i] <= h_vars[i_k],
                    w_vars[k][i] >= h_vars[i] + h_vars[i_k] - 1
                ])
            a_vars[k] = cp.sum(w_vars[k])
            constraints.append(t <= a_vars[k])
        
        # Objective
        objective = cp.Maximize(t)
        
        # Create and solve the model
        prob = cp.Problem(objective, constraints)
        
        # Solve with time limit
        time_limit = min(30, time_left() / 2)
        try:
            prob.solve(solver=cp.CBC, maxSeconds=time_limit)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE, cp.SOLVER_ERROR]:
                # Extract the solution
                h_sol = np.array([h_vars[i].value for i in range(n)])
                h_sol = np.round(h_sol).astype(int)
                
                # Check if this is a valid solution
                if np.abs(np.sum(h_sol) - n//2) < 1e-5:
                    # Compute actual autocorrelation for all shifts
                    autocorr_full = circular_autocorr(h_sol)
                    min_autocorr = np.min(autocorr_full[1:])  # Exclude k=0
                    
                    # Compute C5 based on this
                    c5_val = 1 - 2/n * min_autocorr
                    
                    # Store best solution so far
                    if c5_val < best_c5:
                        best_c5 = c5_val
                        best_h = h_sol.copy()
                        
                    # Check if we've already reached the target
                    if c5_val <= target_bound:
                        break
                        
                    # Need to add worst violating shift
                    worst_shift = np.argmin(autocorr_full[1:]) + 1
                    if worst_shift not in shifts:
                        shifts.append(worst_shift)
                        if len(shifts) > max_iters:
                            # Bail out if too many shifts
                            break
                else:
                    # Not valid due to sum violation
                    break
            else:
                break
                
        except Exception:

        best_c5 = 1 - 2/n * min_autocorr

            min_autocorr = np.min(autocorr_full[1:])
            return 1 - 2/n * min_autocorr

            worst_k = np.argmin(autocorr_full[1:]) + 1
            
            # Gradient update
            grad = np.zeros(n)

                grad[i] = h_cont[(i + worst_k) % n] + h_cont[(i - worst_k) % n]

            # Project to [0,1]

                    swap_idx = idx_unset[0]
                    diff = min(diff, 1 - h_cont[swap_idx])

                best_h = h_cont.copy()

            # params are the cosine coefficients
            M = len(params)
            h_vals = 0.5 * np.ones(n)
            for m in range(1, M+1):
                h_vals += params[m-1] * np.cos(2 * np.pi * m * np.arange(n) / n)
            
            # Clip to [0,1]

            min_autocorr = np.min(autocorr[1:])

        # Project to ensure sum = n/2
        current_sum = np.sum(h_spectral)
        diff = current_sum - n/2
        if abs(diff) > 1e-6:
            idx_set = np.where(h_spectral > 0)[0]
            idx_unset = np.where(h_spectral < 1)[0]
            if len(idx_set) > 0 and len(idx_unset) > 0:
                move_idx = idx_set[0]

                diff = min(diff, 1 - h_spectral[swap_idx])
                diff = max(diff, -h_spectral[move_idx])
                h_spectral[move_idx] -= diff
                h_spectral[swap_idx] += diff
        
        # Final evaluation
        autocorr_full = circular_autocorr(h_spectral)
        min_autocorr = np.min(autocorr_full[1:])
        c5_spectral = 1 - 2/n * min_autocorr
        
        if c5_spectral < best_c5:
            best_h = h_spectral
            best_c5 = c5_spectral
