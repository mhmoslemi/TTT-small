import numpy as np
import cvxpy as cp
import time
from scipy.optimize import minimize
from scipy.fft import fft, ifft
from math import floor, ceil

def evaluate_erdos_solution(h_values):
    """Evaluate C5 = max_k int h(x)(1-h(x+k)) dx for given h."""
    n = len(h_values)
    dx = 2.0 / n
    # Compute circular correlation using FFT (note: this is an approximation)
    h_padded = np.concatenate([h_values, np.zeros(n)])
    h_conj_padded = np.concatenate([1 - h_values, np.zeros(n)])
    corr = ifft(fft(h_padded) * np.conj(fft(h_conj_padded)))
    # Take the first n values of correlation (circular case)
    correlation_values = np.real(corr[:n])
    correlation_values = correlation_values * dx
    # Compute the overlap integral C5 = 1 - 2/n * min(correlation)
    min_correlation = np.min(correlation_values)
    c5 = 1 - 2 / n * min_correlation
    return c5

def circular_correlation(h, pad=True):
    """Compute circular correlation of h with 1-h, using FFT."""
    n = len(h)
    if pad:
        h_padded = np.concatenate([h, np.zeros(n)])
        h_conj_padded = np.concatenate([1 - h, np.zeros(n)])
        corr = ifft(fft(h_padded) * np.conj(fft(h_conj_padded)))
        return np.real(corr[:n])
    else:
        corr = np.correlate(h, 1 - h, mode='full')
        # Take center half: indices n-1 to 2*n-2 (which excludes padding)
        return corr[n-1:]

def sdp_solve(h0, n, max_time=10):
    start_time = time.time()
    # Problem: maximize t subject to constraints
    # Let H be vector of h values (n elements)  
    # Let X = HH^T be the rank-one constraint, so that trace(M_k*X) = sum h_i h_{i+k}
    # So we maximize t subject to: g_k >= t for all k, where g_k = sum_i h_i h_{i+k}
    # We will define matrices for autocorrelation terms
    # M_k[i,j] = 1 if i = j+k or j = i+k mod n
    # This is our matrix full of 1s on the diagonal (at shift k) and 0 elsewhere
    
    # Store all shift matrices inside SDP constraints
    M_list = []
    for k in range(n):
        Mk = np.zeros((n, n))
        for i in range(n):
            Mk[i, (i+k)%n] = 1
            Mk[(i+k)%n, i] = 1
        M_list.append(Mk)
    
    # Variables
    h = cp.Variable(n)
    t = cp.Variable()
    X = cp.Variable((n, n), symmetric=True)
    
    #constraints
    constraints = [
        X == h @ h.T,  # rank-one constraint
        cp.diag(X) == h,  # diagonal = h
        0 <= h, h <= 1,  # box constraints
        cp.sum(h) == n/2,  # sum constraint
        # Autocorrelation constraints
        *[cp.trace(M_list[k] @ X) >= t for k in range(n)]
    ]
    
    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constraints)
    
    try:
        # Solving is an expensive operation; seem to find error for high n
        # So we proxy it with a mock simpler relation
        # First, let us actually compute the first 20 shifts only for quick solving
        # Use ECOS, which is fast, but still test constraint structure
        prob.solve(solver=cp.ECOS, verbose=False, max_iters=500)
        if prob.status == cp.OPTIMAL:
            return h.value, t.value
        else:
            return h0, -1
    except Exception as e:
        return h0, -1

def round_h(X, h_true):
    """
    Given an SDP solution (X = h h^T), return a near-binary vector 
    using Cholesky decomposition approach.
    """
    n = X.shape[0]
    # Cholesky decomposition
    try:
        # Standard Cholesky
        L = np.linalg.cholesky(X)
        # Generate a random Gaussian vector
        g = np.random.randn(L.shape[0])
        scores = L.T @ g
        # Threshold to exactly n/2 ones
        sorted_scores = np.sort(scores)[::-1]
        if len(sorted_scores) >= n//2:
            threshold = sorted_scores[n//2 - 1]
            if len(sorted_scores) > n//2:
                # There's a tie; we may need extra logic
                maybe_idx = np.where(scores >= threshold)[0]
                if len(maybe_idx) < n//2: 
                    # Take more with threshold
                    threshold = sorted_scores[n//2]
                elif len(maybe_idx) > n//2:
                    # Take less with threshold
                    threshold = sorted_scores[n//2 - 1]
            h_rnd = (scores >= threshold).astype(float)
        else:
            h_rnd = (scores >= sorted_scores[-1]).astype(float)
        
        # Fixed up to ensure exactly n//2 ones
        n_ones = np.sum(h_rnd)
        while n_ones > n//2:
            # Try lowering threshold
            idx = np.where(h_rnd == 1)[0]
            delta = np.max([scores[i] for i in idx])
            idx_to_change = np.where(scores == delta)[0]
            if len(idx_to_change) > 0:
                i = idx_to_change[0]
                h_rnd[i] = 0
            n_ones = np.sum(h_rnd)
        while n_ones < n//2:
            # Need to raise threshold
            idx = np.where(h_rnd == 0)[0]
            delta = np.min([scores[i] for i in idx])
            idx_to_change = np.where(scores == delta)[0]
            if len(idx_to_change) > 0:
                i = idx_to_change[0]
                h_rnd[i] = 1
            n_ones = np.sum(h_rnd)
            
        h_rnd = np.clip(h_rnd, 0, 1)
    except:
        # fallback to basis vectors
        h_rnd = np.zeros(n)
        idxs = np.argsort(h_true)[::-1]
        for i in range(n//2):
            h_rnd[idxs[i]] = 1
        
    return h_rnd

def swap_hill_climb(h, n):
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    improved = True
    swaps = 0
    max_swaps = 1000
    while improved and swaps < max_swaps:
        improved = False
        for i in range(n):
            if h[i] == 1:
                for j in range(n):
                    if h[j] == 0:
                        # Try swapping
                        h_swap = h.copy()
                        h_swap[i] = 0
                        h_swap[j] = 1
                        c5 = evaluate_erdos_solution(h_swap)
                        if c5 < best_c5:
                            best_h = h_swap.copy()
                            best_c5 = c5
                            improved = True
                            swaps += 1
                            break
            if improved:
                break
        h = best_h.copy()
    return best_h

def gradient_fine_tune(h, n, steps=1000, step_size=1e-4):
    h_tuned = h.copy()
    use_fft = True
    # compute correlation
    corr = circular_correlation(h_tuned, pad=use_fft)
    min_idx = np.argmin(corr)
    min_corr = corr[min_idx]
    
    for i in range(steps):
        # Compute gradient of min(corr) w.r.t. h_tuned
        # This is a bit tricky since correlation is a complex function of h
        # grad[i] = partial derivative w.r.t. h[i] of min_k corr_k
        # = partial corr_{k*}/partial h[i], where k* is argmin_k corr_k
        
        # Use finite diff approximation for gradient
        # Perturb one element, compute impact on min corruption
        
        # For small perturb: compute h(i+shift and h(i-shift) for shift k* 
        # Fast gradient computation
        
        eps = 1e-6
        g = np.zeros_like(h_tuned)
        
        # finite diff over all directions
        # Efficient to achieve an approximate gradient
        min_k = np.argmin(corr)
        # perturb h[i]
        # perturb a little and find min corr shift
        for i in range(n):
            h_pert = h_tuned.copy()
            h_pert[i] += eps
            corr_pert = circular_correlation(h_pert, pad=use_fft)
            g[i] = (corr_pert[min_k] - corr[min_k]) / eps
        # update
        h_tuned = h_tuned + step_size * g

        # project onto [0,1] and ensure sum = n/2
        # Projective method:
        h_tuned = np.clip(h_tuned, 0, 1)
        diff = np.sum(h_tuned) - n/2
        if abs(diff) > 1e-12:
            # redistribute diff equally among elements
            h_tuned = h_tuned - diff / n

        # update corr 
        corr = circular_correlation(h_tuned, pad=use_fft)
        if i % 100 == 0:
            c5 = 1 - 2 / n * np.min(corr)
            pass
    return h_tuned

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 80  # moderate for SDP
    # Initialize with the initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.asarray(initial_h_values)
    else:
        # use a default construction
        h = np.zeros(n_points) 
        h[:n_points//2] = 1.0
        h = h / np.sum(h) * n_points / 2
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Run SDP + random rounding + polishing
    rounds = 40  # for quick demo
    for r in range(rounds):
        if time.time() - start_time > budget_s - 30:
            break
        h_sdp, t_sdp = sdp_solve(h, n_points, max_time=10)
        if t_sdp >= 0 and np.all(h_sdp >= 0) and np.all(h_sdp <= 1):
            h_rounded = round_h(np.outer(h_sdp, h_sdp), h_sdp)
            h_swapped = swap_hill_climb(h_rounded, n_points)
            h_final = gradient_fine_tune(h_swapped, n_points)
            c5 = evaluate_erdos_solution(h_final)
            if c5 < best_c5:
                best_h = h_final
                best_c5 = c5
        else:
            print(f"SDP failed for round {r}")
            
    # Now interval method as fallback
    if time.time() - start_time < budget_s - 30:
        # Run interval construction as secondary effort
        try:
            h_int, c5_int = interval_optimize(n_points)
            if c5_int < best_c5:
                best_h = h_int
                best_c5 = c5_int
        except:
            print("interval method failed or raised error")
    
    # At end, ensure it's normalized, because of the previous fine tune
    c5_final = evaluate_erdos_solution(best_h)
    return (best_h, c5_final, n_points)
