import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import time
import random


def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial configuration
    initial_h = np.array(initial_h_values)
    n = len(initial_h)
    dx = 2.0 / n
    target_sum = n / 2.0

    # Compute initial C5
    def compute_c5(h, dx):
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx

    # Symmetry breaking: fix h[0] = 1 (without loss of generality)
    h_best = initial_h.copy()
    h_best[0] = 1.0
    c5_best = compute_c5(h_best, dx)
    
    # Simple rounding helper
    def project_to_simplex(v, target_sum):
        """Project vector v onto the simplex with sum = target_sum."""
        v = np.array(v)
        s = np.sum(v)
        if abs(s - target_sum) < 1e-8:
            return v
        # Sort in descending order
        sorted_idx = np.argsort(v)[::-1]
        sorted_v = v[sorted_idx]
        # Find the threshold
        cumsum = np.cumsum(sorted_v)
        threshold_idx = -1
        for i in range(len(cumsum)):
            if cumsum[i] <= target_sum:
                threshold_idx = i
            else:
                break
        if threshold_idx == -1:
            # Simple case: scale down all to reach target sum
            return target_sum * v / s
        else:
            # Compute threshold
            threshold = (target_sum - cumsum[threshold_idx]) / (len(cumsum) - threshold_idx)
            result = np.zeros_like(v)
            if threshold_idx >= 0:
                result[sorted_idx[:threshold_idx+1]] = sorted_v[:threshold_idx+1]
                result[sorted_idx[threshold_idx+1:]] = max(0, sorted_v[threshold_idx+1:] - threshold)
            else:
                result = max(0, v - threshold)
            return result

    def solve_milp(h_start, n, target_sum):
        # Setup CVXPY model
        h_vars = cp.Variable(n, name="h", lb=0, ub=1)
        tau = cp.Variable(1, name="tau")
        # Auxiliary variables for y[i,k] = h[i]*h[i+k]
        # Only store for non-negative k = 1,...,n-1
        y = {}
        for k in range(1, n):
            y[k] = cp.Variable(n - k, name="y_" + str(k), lb=0)
        
        constraints = [cp.sum(h_vars) == target_sum, h_vars[0] == 1.0]  # symmetry breaking

        for k in range(1, n):
            # Get the y vector for this k
            y_k = y[k]
            for i in range(n - k):
                # y[i,k] <= h[i]
                constraints.append(y_k[i] <= h_vars[i])
                # y[i,k] <= h[i+k]
                constraints.append(y_k[i] <= h_vars[i + k])
                # y[i,k] >= h[i] + h[i+k] - 1
                constraints.append(y_k[i] >= h_vars[i] + h_vars[i + k] - 1)
                # y[i,k] >= 0 (already ensured by lb=0)
            # tau <= sum(y_k)
            constraints.append(tau[0] <= cp.sum(y_k))
            
        objective = cp.Maximize(tau[0])

        # Set h_start to warm start
        h_vars.value = h_start.copy()

        prob = cp.Problem(objective, constraints)
        try:
            # Max time before timeout
            max_seconds = 0.5 * budget_s
            prob.solve(solver=cp.CBC, max_seconds=max_seconds)
        except Exception:
            # If solver fails, return None
            return None, None

        if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
            h_sol = h_vars.value
            tau_val = tau.value[0]
            return h_sol, tau_val
        return None, None

    # Run several restarts with warm-starts
    restarts = 0
    total_restarts = 10
    best_h = h_best.copy()
    best_c5 = c5_best

    while restarts < total_restarts and (time.time() - start_time) < 0.9 * budget_s:
        # Start with current best or random binary
        if restarts == 0:
            h_start = best_h.copy()  # start with current incumbent
        else:
            # Generate a random binary vector with correct count
            h_temp = np.zeros(n)
            ones_indices = np.random.choice(n, size=int(target_sum), replace=False)
            h_temp[ones_indices] = 1
            h_start = h_temp.copy()
        h_start[0] = 1.0  # symmetry only

        h_sol, tau_sol = solve_milp(h_start, n, target_sum)

        if h_sol is not None:
            # Compute C5 for this solution
            c5_sol = (target_sum - tau_sol) * 2.0 / n
            if c5_sol < best_c5:
                best_c5 = c5_sol
                best_h = h_sol.copy()
                # Evaluate again with direct correlation
                c5_direct = compute_c5(best_h, dx)
                best_c5 = c5_direct

        # Terminate early if time exceeded
        restarts += 1
        if (time.time() - start_time) > 0.9 * budget_s:
            break

    # Final check with more refined optimization
    if (time.time() - start_time) < 0.7 * budget_s:
        # Attempt refinement in refinement loop
        h_refined = best_h.copy()
        n_iters_refine = 0
        max_iters = 1000
        while n_iters_refine < max_iters and (time.time() - start_time) < 0.9 * budget_s:
            old_c5 = compute_c5(h_refined, dx)
            best_fit = (old_c5, h_refined.copy())
            found_improvement = False

            # Try swapping 1 and 0 to minimize overlap
            for k in range(1, n):
                # Look for (1,0) pairs that could be swapped (in their shift context)
                for i in range(n):
                    if i+k < n:
                        if h_refined[i] == 1 and h_refined[i+k] == 0:
                            h_test = h_refined.copy()
                            h_test[i] = 0
                            h_test[i+k] = 1
                            test_c5 = compute_c5(h_test, dx)
                            if test_c5 < best_fit[0]:
                                best_fit = (test_c5, h_test.copy())
                                found_improvement = True
            if found_improvement:
                h_refined = best_fit[1]
            else:
                break
            n_iters_refine += 1

        final_c5 = compute_c5(h_refined, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = h_refined

    # Final projection onto feasibility if needed
    best_h = project_to_simplex(best_h, target_sum)

    # Final compute C5 from best h
    final_c5 = compute_c5(best_h, dx)

    return (best_h, final_c5, n)
