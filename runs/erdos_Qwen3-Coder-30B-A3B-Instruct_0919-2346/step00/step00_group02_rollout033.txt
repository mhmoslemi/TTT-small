import numpy as np
import time

def c5_of(h):
    # Compute dx
    dx = 2.0 / len(h)
    # Compute overlap integral using np.correlate
    overlap = np.correlate(h, 1 - h, mode='full')
    # Multiply by dx and return max
    return np.max(overlap) * dx

def propose_move(h, max_step):
    # Sample two distinct indices
    i, j = np.random.choice(len(h), size=2, replace=False)
    # Compute feasible interval for delta
    delta_low = max(-h[i], h[j] - 1)
    delta_high = min(1 - h[i], h[j])
    if delta_low >= delta_high:
        # Cannot move this pair; return original
        return h.copy()
    # Sample delta from feasible range
    step = min(max_step, delta_high - delta_low)
    delta = np.random.uniform(-step, step) if step > 0 else 0
    # Apply the move
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    return h_new

def anneal(initial_h, time_budget):
    start = time.time()
    # Initial state
    cur = initial_h.copy()
    c_cur = c5_of(cur)
    best = cur.copy()
    c_best = c_cur
    # Parameters
    T_start = 0.05
    T_end = 1e-6
    step_start = 0.2
    # Main loop
    while time.time() - start < time_budget:
        t = time.time() - start
        # Current temperature and step size
        T = T_start * (T_end / T_start) ** (t / time_budget)
        step = step_start * (1 - t / time_budget)
        # Propose new state
        h_new = propose_move(cur, step)
        c_new = c5_of(h_new)
        # Acceptance rule
        if c_new <= c_cur or np.random.rand() < np.exp(-(c_new - c_cur) / T):
            cur = h_new
            c_cur = c_new
        # Update best
        if c_new < c_best:
            best = h_new
            c_best = c_new
    return best, c_best

def interpolate_and_scale(h, n_points):
    # Interpolate to new length
    x_old = np.linspace(0, 2, len(h))
    x_new = np.linspace(0, 2, n_points)
    h_interp = np.interp(x_new, x_old, h)
    # Renormalize to sum = n_points / 2
    h_interp = h_interp * (n_points / 2) / np.sum(h_interp)
    return h_interp

    # 3. Double-peak seed
    h_double = np.ones(N) * 0.1
    x = np.linspace(0, 2, N)
    mask = (x < 0.3) | (x > 1.7)
    h_double[mask] = 0.9
    h_double = h_double * (N / 2) / np.sum(h_double)
    # Run annealing on the three seeds
    start_time = time.time()
    budget_per_seed = 0.3 * budget_s
    h_best = None
    c_best = float('inf')
    for name, h_seed in [("uniform", h_uniform), ("initial", h_initial), ("double", h_double)]:
        # Anneal
        h, c = anneal(h_seed, budget_per_seed)
        if c < c_best:
            c_best = c
            h_best = h.copy()
    # Polishing phase
    if h_best is not None:
        # Run at zero temperature
        polished_start = time.time()
        budget_remaining = budget_s - (time.time() - start_time)
        if budget_remaining > 20:
