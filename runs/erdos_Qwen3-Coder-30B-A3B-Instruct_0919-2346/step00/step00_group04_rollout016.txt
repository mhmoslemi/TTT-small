def evaluate_c5(h):
    """Evaluate C5 for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation: int h(x)(1 - h(x+k)) dx
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum value of the correlation gives us the C5 bound
    c5 = np.max(corr) * dx
    return c5
