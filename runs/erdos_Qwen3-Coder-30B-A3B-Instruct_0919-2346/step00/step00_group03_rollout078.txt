def evaluate_erdos_solution(h):
    """Compute the overlap integral for the function h"""
    n = len(h)
    dx = 2.0 / n
    # Pad for full convolution (do not pad to avoid confusion)
    # Our correlation sum over valid pairs
    corr = np.correlate(h, 1-h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_sum(h, desired_sum):
    """Project h to desired sum, keeping values in [0,1]"""
    current_sum = np.sum(h)
    diff = (desired_sum - current_sum)/len(h)
    h = h - diff
    h = np.clip(h, 0, 1)
    return h

def evaluate_fft(h):
    """Evaluate using FFT method for efficiency - same as above but with fft fft"""
    n = len(h)
    dx = 2.0 / n
    F = fft(h)
    G = fft(1 - h)
    corr = ifft(F * np.conj(G)).real
    max_corr = np.max(corr)
    return max_corr * dx

def mismatch_swap_refine(h, target_value, max_iterations=5000):
    """Apply mismatch swaps to reduce max overlap (input list of h values to modify inplace, target_value as float)"""
    n = len(h)
    dx = 2.0 / n
    
    # Find current max shift (this is cpu-intensive as it uses full convolution)
    F = fft(h)
    G = fft(1 - h)
    corr = ifft(F * np.conj(G)).real
    max_corr = np.max(corr)
    best_shift_idx = np.argmax(corr)
    best_shift = best_shift_idx - n + 1  # Convert to shift index

    # Track currently very bad indices
    if best_shift < 0:
        # Left shift
        valid_indices = list(range(-best_shift, n))
    else:
        # Right shift
        valid_indices = list(range(0, n - best_shift))
    
    count = 0
    # Majority of computation is in identifying indices
    while max_corr > target_value and count < max_iterations:
        M1 = []  # indices where h[i] ~ 1 and h[i+shift] ~ 0
        M0 = []  # indices where h[i] ~ 0 and h[i+shift] ~ 1
        for i in valid_indices:
            if h[i] > 0.9:
                if h[i + best_shift] < 0.1:
                    M1.append(i)
                elif h[i + best_shift] > 0.9:
                    M0.append(i)
            elif h[i] < 0.1:
                if h[i + best_shift] > 0.9:
                    M0.append(i)
                elif h[i + best_shift] < 0.1:
                    M1.append(i)
        
        if not M1 or not M0:
            # Swap not beneficial or we have used up all, small perturbation
            # Pick two random indices and exchange small amount
            truei = np.random.choice(range(n))
            truej = np.random.choice(list(set(range(n)) - {truei}))
            epsilon = 0.01
            if abs(h[truei] - h[truej]) < epsilon:
                continue
            if h[truei] < 0.5 and h[truej] > 0.5:
                h[truei] += epsilon
                h[truej] -= epsilon
            else:
                h[truei] -= epsilon
                h[truej] += epsilon
            h = project_sum(h, n/2)  # rebalance
        else:
            idx1, idx2 = np.random.choice(M1), np.random.choice(M0)
            # swap
            h[idx1], h[idx2] = h[idx2], h[idx1]
            h = project_sum(h, n/2)
        
        # Recompute and update
        F = fft(h)
        G = fft(1 - h)
        corr = ifft(F * np.conj(G)).real
        max_corr = np.max(corr)
        count += 1
        # Recompute best_shift_idx
        best_shift_idx = np.argmax(corr)
        best_shift = best_shift_idx - n + 1  # Convert to shift index
        if best_shift < 0:
            valid_indices = list(range(-best_shift, n))
        else:
            valid_indices = list(range(0, n - best_shift))
    
    return h

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start = time.time()
    
    # Original initial h is provided, interpolate to larger size (180 is good)
    start_h = np.array(initial_h_values)
    n_points = 180
    dx = 2.0 / n_points
    
    # Interpolate to n_points
    x_orig = np.linspace(0, 2, len(start_h))
    x_new = np.linspace(0, 2, n_points)
    h_orig = np.interp(x_new, x_orig, start_h)
    
    if time.time() - start > 0.95*budget_s:
        return list(h_orig), evaluate_fft(h_orig), n_points
    
    # Conduct Differential Evolution
    # Create a function to minimize that returns the overlap
    def obj(h):
        h_scaled = project_sum(h, n_points / 2)
        return evaluate_fft(h_scaled)
    
    # Initial population
    popsize = 30
    # Start with scaled interpolant
    pop = [np.clip(h_orig + np.random.normal(0, 0.03, n_points), 0, 1)]
    pop = [project_sum(h, n_points / 2) for h in pop]
    for _ in range(popsize - 1):
        h_random = np.random.rand(n_points)
        h_random = np.clip(h_random, 0, 1)
        h_random = project_sum(h_random, n_points / 2)
        pop.append(h_random)
    
    best_h = None
    best_c5 = float('inf')
    best_obj = time.time() - start
    
    # Omega for DE
    total_calls = 0
    max_fevals = 10000  # Limit due to time concerns
    
    de_start = time.time()
    while time.time() - start < budget_s - 5:  # Reuse 5s for final fine-tuning
        # Run DE now, use bounds only on [0,1] for each dimension
        bounds = [(0, 1)] * n_points
        result = differential_evolution(obj, bounds, maxiter=200, popsize=popsize, seed=int(time.time()), callback=None, disp=False)
        total_calls += 1
        
        # Result obtained -- make sure it's feasible (projection might be mathematically inexact)
        h_final = project_sum(result.x, n_points/2)
        c5_val = evaluate_fft(h_final)
        if c5_val < best_c5 or best_h is None:
            best_h = h_final.copy()
            best_c5 = c5_val
        
        if c5_val < 0.379:  # Goal to beat
            break
        
        # Also iterate with possible restarts in case we get stuck
        # Sample small variations of current best to restart
        if total_calls < max_fevals:
            # Re-seed from current best with small variation
            h_rot = []
            for _ in range(popsize):
                hmut = best_h + np.random.normal(0, 0.02, n_points)
                hmut = np.clip(hmut, 0, 1)
                hmut = project_sum(hmut, n_points/2)
                h_rot.append(hmut)
            pop = h_rot
        
    # Final refinement with mismatch swap if needed
    if best_h is not None and best_c5 > 0.3799:
        best_h = mismatch_swap_refine(best_h, 0.3799)
        best_c5 = evaluate_fft(best_h)
    
    return list(best_h), best_c5, n_points
