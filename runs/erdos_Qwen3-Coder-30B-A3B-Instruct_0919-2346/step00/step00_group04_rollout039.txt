def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlations for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the largest value in the correlation
    max_overlap = np.max(corr)
    return max_overlap * dx
