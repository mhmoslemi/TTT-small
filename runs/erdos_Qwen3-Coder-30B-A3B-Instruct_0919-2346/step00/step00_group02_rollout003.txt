import numpy as np
import time
import random

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral for all shifts using full correlation
    overlap = np.correlate(h, 1 - h, mode='full')
    # Multiply by dx and find maximum
    max_overlap = np.max(overlap) * dx
    return max_overlap

def propose_move(h, max_step):
    """Propose a new feasible h by adjusting two elements."""
    h_new = h.copy()
    N = len(h)
    
    while True:
        i, j = random.sample(range(N), 2)
        # Determine feasible delta range
        delta_low = max(-h_new[i], h_new[j] - 1)
        delta_high = min(1 - h_new[i], h_new[j])
        
        if delta_low >= delta_high:
            continue  # No valid move, try another pair
            
        # Sample delta
        step = min(max_step, delta_high - delta_low)
        delta = random.uniform(-step, step)
        
        h_new[i] += delta
        h_new[j] -= delta
        break
        
    return h_new

def anneal(initial_h, time_budget):
    """Perform simulated annealing to minimize C5."""
    start = time.time()
    cur = initial_h.copy()
    best = initial_h.copy()
    c_cur = c5_of(cur)
    c_best = c_cur
    
    # Annealing parameters
    T_start = 0.05
    T_end = 1e-6
    step_start = 0.2
    
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        t = elapsed / time_budget
        
        # Temperature and step size scheduling
        T = T_start * (T_end / T_start) ** t
        step = step_start * (1 - t)
        
        # Propose a move
        h_new = propose_move(cur, step)
        c_new = c5_of(h_new)
        
        # Acceptance rule
        if c_new <= c_cur or random.random() < np.exp(-(c_new - c_cur) / T):
            cur = h_new
            c_cur = c_new
            
            # Update best
            if c_new < c_best:
                best = h_new
                c_best = c_new
                
    return best, c_best

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    # Set parameters
    N = 140
    dx = 2.0 / N
    
    # Build initial seeds
    # 1. Uniform seed
    h_uniform = np.ones(N) * 0.5
    
    # 2. Interpolated provided seed
    if 'initial_h_values' in globals():
        h_provided = np.interp(np.linspace(0, 2, N), 
                              np.linspace(0, 2, len(initial_h_values)), 
                              initial_h_values)
        # Renormalize to sum = N/2
        h_provided = h_provided * (N/2) / np.sum(h_provided)
    else:
        h_provided = h_uniform.copy()
    
    # 3. Two-peak seed
    h_peak = np.ones(N) * 0.1  # Default value
    a = 0.3  # Fraction of domain for high value
    u = 0.9  # High value
    v = 0.1  # Low value
    idx1 = int(a * N)
    idx2 = int((2 - a) * N)
    h_peak[:idx1] = u
    h_peak[idx2:] = u
    h_peak[idx1:idx2] = v
    # Renormalize to sum = N/2
    h_peak = h_peak * (N/2) / np.sum(h_peak)
    
    # Run annealing on each seed
    best_h = None
    best_c5 = float('inf')
    
    # Budget portion for each run (30% each)
    budget_per_run = 0.3 * budget_s
    
    for i, h_seed in enumerate([h_uniform, h_provided, h_peak]):
        h_final, c5_final = anneal(h_seed, budget_per_run)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final
    
    # Polishing phase (10% of budget)
    polishing_time = 0.1 * budget_s
    start = time.time()
    
    while time.time() - start < polishing_time:
        h_new = propose_move(best_h, 0.02)
        c_new = c5_of(h_new)
        if c_new < best_c5:
            best_h = h_new
            best_c5 = c_new
    
    return (best_h, best_c5, N)
