import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_c5(h):
    """Evaluate C5 for given h values"""
    n = len(h)
    dx = 2.0 / n
    # Compute the full correlation
    corr = np.correlate(h, 1 - h, mode='full')
    # The maximum value over all shifts
    max_corr = np.max(corr)
    return max_corr * dx

def initial_h_values():
    """Return the initial construction"""
    # Return the initial construction with 70 points
    h = np.array([0.0]*70)
    # Simple construction: set some values to 1 in the middle
    # This is a basic starting point that can be optimized
    mid = 35
    width = 10
    for i in range(max(0, mid-width//2), min(70, mid+width//2)):
        h[i] = 1.0
    # Scale so that integral is 1
    h = h * 2.0 / np.sum(h)
    return h

def objective(h):
    """Objective function to minimize"""
    return evaluate_c5(h)

def constraint_eq(h):
    """Equality constraint: integral should be 1"""
    n = len(h)
    dx = 2.0 / n
    return np.sum(h) * dx - 1.0

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction as starting point
    h0 = initial_h_values()
    
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Define bounds for each variable: [0, 1]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraint for the integral being exactly 1
    # Using the constraint from scipy.optimize.differential_evolution
    # But we'll handle it within objective function by scaling
    
    # We'll scale h such that sum(h) * dx = 1
    # This means sum(h) == n_points / 2
    
    def constrained_objective(h):
        # This rescales h before computing objective
        scale = (n_points / 2) / np.sum(h)
        h_scaled = h * scale
        return evaluate_c5(h_scaled)
        
    def constraint_function(h):
        # What we want is sum(h) * dx = 1, so sum(h) = n_points / 2
        return np.sum(h) - n_points / 2
        
    # Use differential evolution to optimize
    result = differential_evolution(
        constrained_objective,
        bounds,
        seed=seed,
        maxiter=int(budget_s / 2),  # Rough estimation - adjust as needed
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.7,
        atol=1e-6,
        rtol=1e-6,
        # Keep track of the best solution
    )
    
    # Use the final best value
    h = result.x
    # Rescale to make sure sum(h)*dx = 1
    scale = (n_points / 2) / np.sum(h)
    h_rescaled = h * scale
    
    c5_bound = evaluate_c5(h_rescaled)
    
    return (h_rescaled.tolist(), c5_bound, n_points)
