import numpy as np
import time
from scipy.optimize import linprog
import cvxpy as cp

def evaluate(h):
    """Evaluate C5 using correlation."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    conv = np.correlate(h, 1 - h, mode='full')
    # Sum over all shifts (i.e., all indices of conv)
    max_overlap = np.max(conv) * dx
    return max_overlap

def project_onto_capped_simplex(v, S):
    """Project v onto {h | 0<=h_i<=1, sum(h)=S} using bisection."""
    v = np.array(v)
    low = -2.0
    high = 2.0
    tol = 1e-10
    
    def proj_sum(lam):
        h = np.clip(v - lam, 0, 1)
        return np.sum(h) - S
    
    while high - low > tol:
        mid = (low + high) / 2
        if proj_sum(mid) > 0:
            low = mid
        else:
            high = mid
    lam = (low + high) / 2
    h = np.clip(v - lam, 0, 1)
    return h

def coeffs_and_const(h):
    """Compute a_{i,k} matrix and c_k vector for all shifts k."""
    n = len(h)
    # Shift index range
    ks = range(-(n - 1), n)
    a_matrix = np.zeros((len(ks), n))
    const_vec = np.zeros(len(ks))
    # For each shift, we compute the coefficients and constant term
    for idx, k in enumerate(ks):
        a = np.zeros(n)
        const = 0.0
        for i in range(n):
            # h_{i+k} if valid, else 0
            j = i + k
            if 0 <= j < n:
                a[i] = 1.0 - h[j] - h[i - k] if 0 <= i - k < n else 1.0 - h[j]
            else:
                a[i] = 1.0 - h[i - k] if 0 <= i - k < n else 1.0
            # Compute the constant term as sum_i h_i h_{i+k}
            if 0 <= i + k < n:
                const += h[i] * h[i + k]
        a_matrix[idx] = a
        const_vec[idx] = const
    return a_matrix, const_vec, ks

def linear_program(a_matrix, const_vec, n, S):
    """Solve the LP using cvxpy."""
    h = cp.Variable(n)
    t = cp.Variable()
    # Add constraints to the cvxpy model
    constraints = []
    for i in range(a_matrix.shape[0]):
        # a_matrix[i,:] @ h + const_vec[i] <= t
        constraints.append(a_matrix[i, :].T @ h + const_vec[i] <= t)
    constraints.append(cp.sum(h) == S)
    constraints.append(h >= 0)
    constraints.append(h <= 1)
    constraints.append(t >= 0)
    # Create and solve the model
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)  
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
            h_vals = h.value
            t_val = t.value
            return h_vals, t_val
        else:
            # fallback to scipy.optimize.linprog
            # flatten amatrix
            A_ub = -a_matrix
            b_ub = -const_vec
            A_eq = np.array([np.ones(n)])
            b_eq = np.array([S])
            bounds = [(0, 1) for _ in range(n)]
            res = linprog(c=[1.0] + [0.0]*n, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, 
                          b_eq=b_eq, bounds=bounds, method='highs')
            if res.success:
                h_vals = res.x[1:]
                t_val = res.x[0]
                return h_vals, t_val
            else:
                # If both methods fail, fallback to safe value
                print("Linprog failed, falling back.")
                return None, None
    except Exception as e:
        print(f"CVXPY error: {e}")
        try:
            # Fallback to linprog
            A_ub = -a_matrix
            b_ub = -const_vec
            A_eq = np.array([np.ones(n)])
            b_eq = np.array([S])
            bounds = [(0, 1) for _ in range(n)]
            res = linprog(c=[1.0] + [0.0]*n, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, 
                          b_eq=b_eq, bounds=bounds, method='highs')
            if res.success:
                h_vals = res.x[1:]
                t_val = res.x[0]
                return h_vals, t_val
            else:
                return None, None
        except:
            return None, None

def random_feasible(n):
    """Generate a random feasible start point."""
    h = np.random.rand(n)
    # Project to capped simplex with sum = n/2
    h = project_onto_capped_simplex(h, n / 2)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n = len(initial_h_values) if 'initial_h_values' in globals() else 70
    S = n / 2
    best_h = initial_h_values if 'initial_h_values' in globals() else random_feasible(n)
    best_C5 = evaluate(best_h)
    # Let's also try to do a refinement step on the initial
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
        h = project_onto_capped_simplex(h, S)
        try:
            C5 = evaluate(h)
            if C5 < best_C5:
                best_C5 = C5
                best_h = h.copy()
        except:
            pass

    restarts = 0
    max_iter = 50  # To make sure it's not too slow even if we're not reusing latest values
    while time.time() - start_time < 0.95 * budget_s:
        restarts += 1
        # Start fresh each restart
        h = random_feasible(n)
        prev_C5 = float('inf')  # To detect convergence
        prev_t = float('inf')   # To detect convergence
        convergence_count = 0
        for i in range(max_iter):
            if time.time() - start_time > 0.95 * budget_s:
                break
            # Compute linearization
            a_matrix, const_vec, ks = coeffs_and_const(h)
            # Solve LP
            h_new, t = linear_program(a_matrix, const_vec, n, S)
            if h_new is None:
                # Retry with random start
                h = random_feasible(n)
                continue
            # Relaxation step
            lambda_val = 0.7
            h_new = lambda_val * h_new + (1 - lambda_val) * h
            # Projection
            h_new = project_onto_capped_simplex(h_new, S)
            # Compute actual C5
            try:
                C5 = evaluate(h_new)
                if C5 < best_C5:
                    best_C5 = C5
                    best_h = h_new.copy()
                # convergence check
                if abs(t - prev_t) < 1e-6 or abs(C5 - prev_C5) < 1e-6:
                    convergence_count += 1
                    if convergence_count > 3:
                        break  # must have converged
                else:
                    convergence_count = 0
                prev_t, prev_C5 = t, C5
                h = h_new
            except Exception as e:
                h = random_feasible(n)
                continue
    # Final polishing with local changes (small moves)
    h_polish = best_h.copy()
    for i in range(500):
        if time.time() - start_time > 0.95 * budget_s:
            break
        try:
            # Choose random sample to shift
            idx = np.random.randint(0, n)
            if h_polish[idx] <= 0:
                continue
            # Find valid destination shift
            dst = (idx + np.random.randint(1, n)) % n
            if h_polish[dst] >= 1:
                continue
            # Move a small amount
            delta = np.clip(0.001 * np.random.rand(), 1e-6, 0.1)
            # Ensure no violation of bounds
            delta = min(delta, h_polish[idx], 1 - h_polish[dst])
            if delta <= 0:
                continue
            h_test = h_polish.copy()
            h_test[idx] -= delta
            h_test[dst] += delta
            h_test = project_onto_capped_simplex(h_test, S)
            C5_test = evaluate(h_test)
            if C5_test < best_C5:
                best_C5 = C5_test
                best_h = h_test.copy()
                h_polish = h_test.copy()
            # Once we're done with polishing, break early for time
        except Exception:
            pass
    return best_h, best_C5, n
