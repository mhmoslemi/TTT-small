import numpy as np
from scipy.optimize import differential_evolution, minimize
import time
from math import ceil

def evaluate_h(h):
    """Evaluate C5 for given h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute C5 = max_k int h(x)(1 - h(x+k)) dx
    # This is equivalent to max of correlation of h with (1-h)
    h_rev = np.flip(1 - h)
    correlation = np.correlate(h, h_rev, mode="full")
    # The maximum correlation is the highest value
    max_correlation = np.max(correlation)
    return max_correlation * dx

def initial_h_values():
    """Return initial construction - 45 samples"""
    n = 45
    h0 = np.zeros(n)
    
    # Based on known good construction
    # We use a piecewise constant function with some structure
    # This is a good starting point
    h0[0:20] = 1.0
    h0[20:30] = 0.0
    h0[30:35] = 1.0
    h0[35:40] = 0.0
    h0[40:45] = 1.0
    
    # Normalize so that integral = 1
    dx = 2.0 / n
    h0 = h0 * (1.0 / (np.sum(h0) * dx))
    
    return h0

def objective(h):
    """Objective function to minimize"""
    return evaluate_h(h)

def constraint_eq(h):
    """Equality constraint: integral of h = 1"""
    n_points = len(h)
    dx = 2.0 / n_points
    return np.sum(h) * dx - 1.0

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial construction
    initial_h = initial_h_values()
    n_points = len(initial_h)
    
    # Define bounds for each variable (h[i] in [0,1])
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Start timer
    start_time = time.time()
    
    # Initial score
    best_c5 = evaluate_h(initial_h)
    best_h = initial_h.copy()
    
    # First try to improve with scipy optimization
    def func_to_minimize(x):
        return evaluate_h(x)
    
    # Constraint function
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Simple Nelder-Mead based optimization
    result = minimize(func_to_minimize, initial_h, method='Nelder-Mead', 
                      options={'maxiter': 500, 'disp': False})
    if result.success:
        current_c5 = evaluate_h(result.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x.copy()
    
    # Try a few more optimization runs with different methods
    for i in range(5):
        if time.time() - start_time > budget_s - 10:
            break
        
        # Random restart
        restart_h = np.random.rand(n_points) * 0.5
        restart_h = restart_h * (2.0 / np.sum(restart_h))
        constraint_sum = np.sum(restart_h) * (2.0 / n_points) - 1.0
        restart_h = restart_h - constraint_sum * 0.01
                
        # Fix bounds to [0,1] for clipping
        restart_h = np.clip(restart_h, 0, 1)
        restart_h = restart_h * (1.0 / (np.sum(restart_h) * (2.0 / n_points)))
        
        # Try gradient descent with bounds
        result2 = minimize(func_to_minimize, restart_h, method='L-BFGS-B', 
                           bounds=bounds, options={'maxiter': 200, 'disp': False})
        
        if result2.success:
            current_c5 = evaluate_h(result2.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result2.x.copy()
                
        # Early exit if time runs out
        if time.time() - start_time > budget_s - 10:
            break
    
    # If we found an even better solution, return it
    # Otherwise return best-known solution found so far
    if best_c5 < 0.38092:
        # Final refinement with more optimization
        result_final = minimize(func_to_minimize, best_h, method='Nelder-Mead', 
                               options={'maxiter': 1000, 'disp': False})
        
        if result_final.success:
            final_c5 = evaluate_h(result_final.x)
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = result_final.x.copy()
    
    # Ensure the constraint is met exactly
    dx = 2.0 / n_points
    sum_h = np.sum(best_h)
    best_h = best_h * (1.0 / (sum_h * dx))
    
    # Final evaluation
    final_c5 = evaluate_h(best_h)
    
    # Return best values
    return (best_h.tolist(), final_c5, n_points)
