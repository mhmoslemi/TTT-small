import numpy as np
from scipy.optimize import differential_evolution
import time
import math

def evaluate_erdos_solution(h_values):
    """Evaluate the C5 constant for given h_values"""
    n = len(h_values)
    dx = 2.0 / n
    h = np.array(h_values)
    # Compute correlation of h with (1-h) using full mode
    corr = np.correlate(h, 1 - h, mode='full')
    # The maximum of these correlations gives C5
    max_corr = np.max(corr)
    return max_corr * dx

def initial_h_values():
    """Return initial step function h as a list of values"""
    # Create initial construction as suggested in prompt
    n = 70
    h = np.zeros(n)
    # Simple piecewise constant construction
    h[:35] = 1.0
    h[35:] = 0.0
    return h.tolist()

def objective_function(h_values):
    """Objective to minimize"""
    return evaluate_erdos_solution(h_values)

def bounds_constraint(h_values):
    """Ensure sum(h) * dx == 1"""
    n = len(h_values)
    dx = 2.0 / n
    return abs(sum(h_values) * dx - 1.0)

def run(seed=42, budget_s=600, **kwargs):
    """Run optimization of h to minimize C5"""
    np.random.seed(seed)
    
    start_time = time.time()
    
    # Initialize with the known good construction
    h_init = initial_h_values()
    n_points = len(h_init)
    
    # Set up bounds: [0,1] for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization function
    def optimize_step():
        # Run optimization with Differential Evolution
        result = differential_evolution(
            objective_function,
            bounds,
            maxiter=1000,
            popsize=15,
            seed=seed,
            tol=1e-6,
            mutation=(0.5, 1.0),
            recombination=0.7,
            strategy='best1bin',
            disp=False,
            polish=True
        )
        return result.x
    
    # Start with initial solution
    best_h = np.array(h_init)
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Slightly adjust to meet the constraint
    dx = 2.0 / n_points
    current_sum = np.sum(best_h)
    target_sum = 1.0 / dx  # This is n_points / 2
    scale_factor = target_sum / current_sum
    best_h = best_h * scale_factor
    
    # Evaluate new solution
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Refine with local search starting from current state
    t0 = time.time()
    iter_count = 0
    
    last_improvement = 0
    patience = 20
    
    while time.time() - start_time < budget_s - 1:
        if time.time() - t0 > 2:  # Limit no. iterations per second
            t0 = time.time()
            iter_count += 1
            
            # Sample random perturbation and/or use optimization step
            if np.random.random() < 0.7 or iter_count % 5 == 0:
                # Use random modification to explore
                h_new = best_h.copy()
                i = np.random.randint(n_points)
                delta = np.random.normal(0, 0.1)
                h_new[i] = np.clip(h_new[i] + delta, 0, 1)
                
                # Then slightly adjust to preserve sum constraint
                current_sum = np.sum(h_new)
                scale_factor = target_sum / current_sum
                h_new = h_new * scale_factor
                
                new_c5 = evaluate_erdos_solution(h_new)
                
                if new_c5 < best_c5:
                    best_h = h_new
                    best_c5 = new_c5
                    last_improvement = 0
                else:
                    last_improvement += 1
                    
            else:
                # Try global optimization step
                try:
                    optimized = optimize_step()
                    # Adjust sum to meet constraint
                    current_sum = np.sum(optimized)
                    if abs(current_sum) > 1e-10:
                        scale_factor = target_sum / current_sum
                        optimized = optimized * scale_factor
                    new_c5 = evaluate_erdos_solution(optimized)
                    if new_c5 < best_c5:
                        best_h = optimized
                        best_c5 = new_c5
                        last_improvement = 0
                    else:
                        last_improvement += 1
                except:
                    last_improvement += 1
                    continue
                
        # Check for early stop
        if last_improvement > patience:
            break

    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
