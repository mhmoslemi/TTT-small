import numpy as np
import time
import random

def compute_c5(h):
    """Compute C5 bound using correlation."""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return corr.max()

def propose_swap_and_update(h, corr, i, j):
    """Update correlation after swapping h[i] and h[j]."""
    a = h[i]
    b = h[j]
    delta_i = b - a
    
    n = len(h)
    new_corr = corr.copy()
    
    # Loop over k not equal to i or j
    for k in range(n):
        if k == i or k == j:
            continue
        hk = h[k]
        # Contribution from (i,k): shift l = k - i
        idx1 = k - i + n - 1
        new_corr[idx1] += delta_i * (1 - hk)
        
        # Contribution from (k,i): shift l = i - k  
        idx2 = i - k + n - 1
        new_corr[idx2] += hk * (-delta_i)
        
        # Contribution from (j,k): shift l = k - j
        idx3 = k - j + n - 1
        new_corr[idx3] += -delta_i * (1 - hk)
        
        # Contribution from (k,j): shift l = j - k
        idx4 = j - k + n - 1
        new_corr[idx4] += hk * (delta_i)
    
    # Handle cross pair (i,j): shift l = j-i
    idx_ij = j - i + n - 1
    new_corr[idx_ij] += delta_i
    
    # Handle cross pair (j,i): shift l = i-j
    idx_ji = i - j + n - 1
    new_corr[idx_ji] -= delta_i
    
    return new_corr

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()
    
    best_c5 = np.inf
    best_h = None
    best_n = None
    
    # try multiple n values
    n_candidates = [60, 80, 100, 120, 140, 160, 180, 200]
    
    for n in n_candidates:
        if time.time() - start > budget_s - 1:
            break
            
        if n % 2 != 0:
            continue
            
        # Compute max iterations based on time and n 
        time_left = budget_s - (time.time() - start)
        if time_left <= 1:
            break
        
        # Allocate more iterations for smaller n
        max_iters = int(50000 / (n / 100))  # roughly 50000 for n=100
        max_iters = max(max_iters, 1000)
        
        # Initialize binary h with sum = n/2 ones
        h = np.zeros(n, dtype=float)
        ones = n // 2
        h[np.random.choice(n, ones, replace=False)] = 1.0
        
        # Compute initial correlation
        corr = np.correlate(h, 1 - h, mode='full')
        cur_c5 = corr.max() * (2.0 / n)
        cur_max = corr.max()
        
        # Temperature for simulated annealing
        temp = 0.02
        T_decay = 0.9995
        
        # Track iterations without improvement
        no_improve = 0
        max_no_improve = 15000
        
        for it in range(max_iters):
            now = time.time()
            if now - start > budget_s - 0.5:
                break
                
            # propose a swap
            i = np.random.randint(n)
            j = np.random.randint(n)
            while h[i] == h[j]:
                i = np.random.randint(n)
                j = np.random.randint(n)
            
            # Compute new correlation (reuse corr)
            new_corr = propose_swap_and_update(h, corr, i, j)
            new_max = new_corr.max()
            new_c5 = new_max * (2.0 / n)
            
            # Acceptance criteria
            d = new_c5 - cur_c5
            if d <= 0 or np.random.rand() < np.exp(-d / (temp + 1e-12)):
                corr = new_corr
                cur_c5 = new_c5
                cur_max = new_max
                h[i], h[j] = h[j], h[i]
                no_improve = 0
            else:
                no_improve += 1
                
            # Update best if improved
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                best_n = n
                
            # Temperature decay
            temp *= T_decay
            
            # Random restart if stuck
            if no_improve > max_no_improve:
                h = np.zeros(n, dtype=float)
                ones = n // 2
                h[np.random.choice(n, ones, replace=False)] = 1.0
                corr = np.correlate(h, 1 - h, mode='full')
                cur_c5 = corr.max() * (2.0 / n)
                cur_max = corr.max()
                no_improve = 0
                temp = 0.02
                
        # Check time again
        if time.time() - start > budget_s - 0.5:
            break

    # Fallback to initial_h_values if nothing found
    if best_h is None:
        try:
            from __main__ import initial_h_values
            best_h = initial_h_values.copy()
            best_n = len(best_h)
            best_c5 = compute_c5(best_h)
        except:
            # simple fallback
            best_h = np.array([1.0, 0.0])
            best_n = 2
            best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
