def evaluate_erdos_solution(h):
    """Compute C5 directly from the step function h."""
    dx = 2.0 / len(h)
    c = np.correlate(h, 1 - h, mode='full')
    return np.max(c) * dx

def project_to_capped_simplex(x, target_sum):
    """Project vector x to the capped simplex with given target sum."""
    n = len(x)
    x = np.array(x, dtype=float)
    # Ensure x is within [0,1]
    x = np.clip(x, 0, 1)
    current_sum = np.sum(x)
    if np.abs(current_sum - target_sum) < 1e-8:
        return x

    if current_sum > target_sum:
        # Need to decrease values
        sorted_indices = np.argsort(-x)
        cumulative = 0
        for i in range(n):
            idx = sorted_indices[i]
            if cumulative + x[idx] <= target_sum:
                cumulative += x[idx]
            else:
                # This index will be reduced
                x[idx] = target_sum - cumulative
                cumulative = target_sum
                break
        # Zero the rest
        for i in range(i+1, n):
            idx = sorted_indices[i]
            x[idx] = 0
    else:
        # Need to increase values
        sorted_indices = np.argsort(x)
        cumulative = 0
        for i in reversed(range(n)):
            idx = sorted_indices[i]
            if cumulative + (1 - x[idx]) <= (target_sum - current_sum):
                cumulative += (1 - x[idx])
                x[idx] = 1
            else:
                # Reduce the increase amount
                x[idx] += (target_sum - current_sum) - cumulative
                break
        # Zero the rest to keep total sum fixed
        for i in reversed(range(i)):
            idx = sorted_indices[i]
            x[idx] = 0

    return x

def solve_milp_for_autocorr(h0, n, target_sum, solver='CBC'):
    """Solve the MILP with `h` as binary, compute the best τ and return h."""
    # Boolean variables h[i] for i = 0..n-1
    h = Variable(n, boolean=True)
    
    # Precompute number of pairs for each shift k
    total_y_vars = 0
    for k in range(1, n):
        total_y_vars += (n - k)
        
    # Create y variables for each k (the number of y[i,k] is n-k for shift k)
    # Use a list of lists to store for easy indexing
    y = []
    for k in range(1, n):
        # y[k][i] = h[i] * h[i+k] for i = 0, 1, ..., n-k-1
        # So only first (n-k) variables used for shift k
        y.append(Variable(n - k, boolean=True))

    # Objective: maximize τ
    tau = Variable(1)
    
    # Build the constraints:
    # For each k = 1, ..., n-1, we compute sum(y[k]) ≤ τ
    constraints = []
    constraints.append(sum_entries(h) == target_sum)
    
    # Add y equations using McCormick envelope
    for k in range(1, n):
        for i in range(n - k):
            # y[k][i] represents h[i] * h[i+k]
            # y == h[i]*h[i+k] iff:
            #  y >= 0
            #  y >= h[i] + h[i+k] - 1
            #  y <= h[i]
            #  y <= h[i+k]
                    
            y[k - 1][i].value = max(0, h[i].value + h[i+k].value - 1)
            y[k - 1][i].value = min(h[i].value, h[i+k].value)
            
            constraints.append(y[k - 1][i] >= 0)
            constraints.append(y[k - 1][i] >= h[i] + h[i+k] - 1)
            constraints.append(y[k - 1][i] <= h[i])
            constraints.append(y[k - 1][i] <= h[i+k])
            
    # Add τ constraints
    for k in range(1, n):
        constraints.append(tau <= sum_entries(y[k - 1]))

    # Solve
    objective = Minimize(-tau)
    prob = Problem(objective, constraints)
    prob.solve(solver=solver, verbose=False, max_seconds=0.6 * 600)
    
    if prob.status in ["optimal", "optimal_inaccurate"]:
        # Extract h solution (should be binary here)
        h_result = np.array([h[i].value for i in range(n)])
        tau_val = -prob.value
        return h_result, tau_val
    else:
        # Fall back to continuous solution
        h_cont = Variable(n, integer=False)
        constraints = [h_cont >= 0, h_cont <= 1, sum_entries(h_cont) == target_sum]
        
        # Create tau variable
        tau = Variable(1)
        # The tau constraint for each k
        for k in range(1, n):
            accumulated = 0
            for i in range(n - k):
                accumulated += h_cont[i] * h_cont[i + k]
            constraints.append(tau <= accumulated)
            
        prob = Problem(Maximize(tau), constraints)
        try:
            prob.solve(solver='ECOS', verbose=False)
        except:
            prob.solve(solver='GLPK', verbose=False)
        if prob.status in ["optimal", "optimal_inaccurate"]:
            # Get h_cont values and project
            h_result = np.array([h_cont[i].value for i in range(n)])
            h_result = project_to_capped_simplex(h_result, target_sum)
            tau_val = prob.value
            return h_result, tau_val
        else:
            # Return h0 as fallback
            return h0, float('inf')

def improve_solution_with_local_swap(h, K, n):
    """Apply local swaps to improve resource-efficient solution."""
    # Find the worst shift k* that shows up in the maximal.autocorrelation
    c = np.correlate(h, 1 - h, mode="full")
    k_star = np.argmax(c)
    
    # Get the value at the lowest point
    shift_k = c[k_star]
    
    # For swap operations
    max_improvement = False
    improve_count = 0
    max_iterations = 2000
    iterations = 0

    while not max_improvement and iterations < max_iterations:
        max_improvement = True
        # Try all pairs (i, i+k*) such that h[i]=1 and h[i+k*]=0
        k_use = k_star - n + 1  # Correct index in [-n+1, n-1]
        for i in range(n - abs(k_use)):
            if h[i] == 1 and h[i + k_use] == 0:
                # Swap them
                h[i], h[i + k_use] = h[i + k_use], h[i]
                new_C5 = evaluate_erdos_solution(h)
                # Revert the swap unless it improves C5
                h[i], h[i + k_use] = h[i + k_use], h[i]
                if new_C5 < evaluate_erdos_solution(h) - 1e-8:
                    # Accept the swap
                    h[i], h[i + k_use] = h[i + k_use], h[i]
                    max_improvement = False
                    improve_count += 1
                    break
        iterations += 1
    return h
