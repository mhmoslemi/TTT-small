import numpy as np
import time
from scipy.optimize import differential_evolution
from scipy.fft import fft, ifft
from scipy.spatial.distance import cdist
import math

def evaluate_solution(h, n_points):
    dx = 2.0 / n_points
    h = np.array(h)
    F = fft(h)
    G = fft(1 - h)
    correlation = ifft(F * np.conj(G)).real
    max_overlap = max(correlation)
    return dx * max_overlap

def project_to_simplex(h, target_sum):
    """Project vector h onto the simplex sum(h) = target_sum"""
    h = np.array(h)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    delta = (current_sum - target_sum) / len(h)
    h = h - delta
    h = np.clip(h, 0, 1)
    # Re-project to ensure sum is correct
    current_sum = np.sum(h)
    correction = (target_sum - current_sum) / len(h)
    h = h + correction
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start from initial_h_values with interpolation
    h_init = np.array(initial_h_values)
    n_points = 180
    dx = 2.0 / n_points
    h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_init)), h_init)

    def objective(h_vec):
        h_vec = project_to_simplex(h_vec, n_points / 2)
        return evaluate_solution(h_vec, n_points)

    # Evaluate initial solution
    best_h = h.copy()
    best_c5 = objective(best_h)
    print(f"Initial C5: {best_c5}")

    # DE optimization
    popsize = 30
    maxiter = 10000
    F = 0.8
    CR = 0.9
    restart_generations = 150
    current_time = time.time()
    elapsed = current_time - start_time
    remaining_time = budget_s - elapsed - 30  # leave 30s for refinement
    
    if remaining_time <= 0:
        return (list(best_h), best_c5, n_points)
    
    # Initialize population based on seed
    population = [best_h.copy()]
    for _ in range(popsize - 1):
        perturbation = np.random.normal(0, 0.05, n_points)
        trial = best_h + perturbation
        trial = project_to_simplex(trial, n_points / 2)
        population.append(trial)
    
    # DE loop
    for i in range(maxiter):
        if i % restart_generations == 0:
            # Restart worst half of population
            fitnesses = [objective(p) for p in population]
            sorted_indices = np.argsort(fitnesses)
            worst_indices = sorted_indices[-popsize//2:]
            for idx in worst_indices:
                population[idx] = np.random.uniform(0, 1, n_points)
                population[idx] = project_to_simplex(population[idx], n_points / 2)
        
        for j in range(popsize):
            # Select three different individuals
            a, b, c = population[0], population[1], population[2]
            while a is population[j]:
                a = population[np.random.randint(0, popsize)]
            while b is population[j] or b is a:
                b = population[np.random.randint(0, popsize)]
            while c is population[j] or c is a or c is b:
                c = population[np.random.randint(0, popsize)]
            
            # Mutate
            v = a + F * (b - c)
            v = np.clip(v, 0, 1)
            
            # Crossover
            u = np.copy(population[j])
            for k in range(n_points):
                if np.random.rand() < CR:
                    u[k] = v[k]
            
            # Project
            u = project_to_simplex(u, n_points / 2)
            
            # Evaluate and replace
            current_fitness = objective(population[j])
            new_fitness = objective(u)
            
            if new_fitness <= current_fitness:
                population[j] = u
                
        # Update best
        current_fitnesses = [objective(p) for p in population]
        min_idx = np.argmin(current_fitnesses)
        if current_fitnesses[min_idx] < best_c5:
            best_c5 = current_fitnesses[min_idx]
            best_h = population[min_idx].copy()
            
        current_time = time.time()
        elapsed = current_time - start_time
        if elapsed > budget_s - 30:
            break
    
    # Mismatch-swap refinement
    h_refined = best_h.copy()
    h_refined = project_to_simplex(h_refined, n_points / 2)
    
    def find_mismatch_sets(h_vec, shift):
        indices = list(range(len(h_vec) - shift))
        M1 = [i for i in indices if abs(h_vec[i] - 1) < 1e-5 and abs(h_vec[i + shift] - 0) < 1e-5]
        M0 = [i for i in indices if abs(h_vec[i] - 0) < 1e-5 and abs(h_vec[i + shift] - 1) < 1e-5]
        return M1, M0
    
    def reduce_overlap(h_vec, shift):
        M1, M0 = find_mismatch_sets(h_vec, shift)
        if not M1 or not M0:
            return h_vec, False
            
        i, j = M1[0], M0[0]
        delta = min(1.0 - h_vec[i], h_vec[j])
        if delta <= 1e-10:
            return h_vec, False
            
        h_vec[i] += delta
        h_vec[j] -= delta
        h_vec = project_to_simplex(h_vec, n_points / 2)
        return h_vec, True
    
    # Apply mismatch swap
    current_c5 = evaluate_solution(h_refined, n_points)
    stable_count = 0
    max_stable = 1000
    
    while stable_count < max_stable:
        _, corr = evaluate_overlap_with_shifts(h_refined, n_points)
        max_shift_idx = np.argmax(corr)
        best_shift = max_shift_idx - (n_points - 1)
        
        new_h, improved = reduce_overlap(h_refined, abs(best_shift) if best_shift >= 0 else abs(best_shift) + 1)
        if not improved:
            stable_count += 1
            # Perturb slightly to escape stagnation
            idx1, idx2 = np.random.choice(n_points, 2, replace=False)
            delta = 0.01 * np.random.choice([-1, 1])
            new_h[idx1] = min(1.0, max(0.0, new_h[idx1] + delta))
            new_h[idx2] = min(1.0, max(0.0, new_h[idx2] - delta))
            new_h = project_to_simplex(new_h, n_points / 2)
        else:
            h_refined = new_h
            stable_count = 0
            
        if abs(evaluate_solution(h_refined, n_points) - current_c5) < 1e-6:
            stable_count += 1
        else:
            stable_count = 0
            current_c5 = evaluate_solution(h_refined, n_points)
            
        current_time = time.time()
        elapsed = current_time - start_time
        if elapsed > budget_s - 10:
            break
    
    final_c5 = evaluate_solution(h_refined, n_points)
    h_solution = list(h_refined)
    return (h_solution, final_c5, n_points)

def evaluate_overlap_with_shifts(h, n_points):
    h = np.array(h)
    F = fft(h)
    G = fft(1 - h)
    correlation = ifft(F * np.conj(G)).real
    return correlation, correlation
