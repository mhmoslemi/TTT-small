def evaluate_erdos_solution(h_values):
    """Evaluate the Erdős minimum overlap constant C₅ for given h values."""
    n_points = len(h_values)
    dx = 2.0 / n_points
    # Compute correlation of h with 1-h
    corr = correlate(h_values, 1 - h_values, mode='full')
    # The maximum is the maximum value over all shifts (excluding the index 0)
    max_overlap = np.max(corr) * dx
    return max_overlap
