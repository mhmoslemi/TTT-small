import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_erdos_solution(h_values):
    """Compute C_5 bound for given h_values."""
    n = len(h_values)
    dx = 2.0 / n
    # Compute correlation of h with (1-h)
    corr = np.correlate(h_values, 1 - np.array(h_values), mode="full")
    # The maximum of the correlation gives the C_5 bound
    c5 = np.max(corr) * dx
    return c5

def initial_h_values():
    """Initial construction of h based on known solution."""
    n = 70
    h = np.zeros(n)
    # Simple piecewise constant construction
    h[:20] = 1.0
    h[20:40] = 0.5
    h[40:] = 0.0
    # Normalize to integral = 1
    h = h * 2.0 / np.sum(h)
    return h

def objective(h_values):
    """Objective function to minimize."""
    return evaluate_erdos_solution(h_values)

def constraint_fun(h_values):
    """Constraint: sum(h) * dx = 1."""
    n = len(h_values)
    dx = 2.0 / n
    return np.sum(h_values) * dx - 1.0

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial solution
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points

    # Set bounds for each variable (0 <= h[i] <= 1)
    bounds = [(0, 1) for _ in range(n_points)]

    # Wrapper to ensure sum(h) * dx = 1 constraint
    def objective_wrapped(h):
        # Normalize h to satisfy sum constraint
        h = np.array(h)
        h = h * (n_points / 2) / np.sum(h)
        return evaluate_erdos_solution(h)

    # Optimizer parameters
    maxiter = 10000
    disp = False
    tol = 1e-6

    # Track best solution
    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(h0)
    iterations = 0

    # Use differential evolution first to get a good starting point
    res = differential_evolution(
        objective_wrapped,
        bounds,
        maxiter=500,
        popsize=15,
        seed=seed,
        disp=False,
        polish=False,
        atol=1e-6,
        rtol=1e-6
    )
    if res.success:
        # Normalize to exact constraint
        h_final = res.x * (n_points / 2) / np.sum(res.x)
        this_c5 = evaluate_erdos_solution(h_final)
        if this_c5 < best_c5:
            best_c5 = this_c5
            best_h = h_final.copy()
            
    # Then refine with local minimization
    # Use scipy minimize with SLSQP
    # Force the constraint manually by averaging
    def inner_obj(h):
        h_scaled = np.array(h)
        h_scaled = h_scaled * (n_points / 2) / np.sum(h_scaled)
        return evaluate_erdos_solution(h_scaled)
    
    constraints = {'type': 'eq', 'fun': constraint_fun}
    bounds_min = [(0, 1) for _ in range(n_points)]
    
    # Do local optimization
    result = minimize(
        inner_obj,
        h0.copy(),
        method='SLSQP',
        bounds=bounds_min,
        constraints=constraints,
        options={'maxiter': 3000, 'ftol': 1e-9, 'gtol': 1e-9},
        callback=lambda x: None,
    )

    if result.success:
        # Normalize to exact constraint
        h_final = result.x * (n_points / 2) / np.sum(result.x)
        this_c5 = evaluate_erdos_solution(h_final)
        if this_c5 < best_c5:
            best_c5 = this_c5
            best_h = h_final.copy()
    
    # Try a refined grid search
    def grid_search_step():
        nonlocal best_h, best_c5
        n = n_points
        # Try random perturbations within feasible region
        current_h = best_h.copy()
        for _ in range(1000):
            if time.time() - start_time > budget_s * 0.9:
                break
            # Generate small random perturbation
            delta = np.random.normal(0, 0.01, n)
            new_h = current_h + delta
            # Project onto feasible set
            new_h = np.clip(new_h, 0, 1)
            # Scale to correct integral
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            new_c5 = evaluate_erdos_solution(new_h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()
        return best_h, best_c5

    # Try grid search for additional refinement
    try:
        grid_h, grid_c5 = grid_search_step()
        if grid_c5 < best_c5:
            best_c5 = grid_c5
            best_h = grid_h.copy()
    except:
        pass

    return list(best_h), best_c5, n_points
