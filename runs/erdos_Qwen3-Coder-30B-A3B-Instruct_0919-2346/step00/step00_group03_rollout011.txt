import numpy as np
import time
import random

def evaluate_c5(h):
    """Compute the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h) using full mode
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)

def random_binary_seq(n):
    """Generate a random binary sequence with exactly n/2 ones"""
    h = np.zeros(n, dtype=float)
    indices = list(range(n))
    random.shuffle(indices)
    for i in range(n // 2):
        h[indices[i]] = 1.0
    return h

def solve_binary(n, iters, seed=None):
    """Run simulated annealing on binary sequence"""
    np.random.seed(seed)
    # Generate initial binary sequence
    h = random_binary_seq(n)
    c5 = evaluate_c5(h)
    best_h, best_c5 = h.copy(), c5
    
    # Precompute indices of ones and zeros for efficient swapping
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    
    # Simulated Annealing parameters
    T0 = 0.05
    T = T0
    cooling = 0.99995
    
    for iteration in range(iters):
        # Choose random swap
        if len(ones) == 0 or len(zeros) == 0:
            break
            
        i = random.choice(ones)
        j = random.choice(zeros)
        
        # Propose swap
        h[i], h[j] = h[j], h[i]
        
        # Compute new C5
        new_c5 = evaluate_c5(h)
        delta = new_c5 - c5
        
        # Accept or reject based on simulated annealing
        if delta <= 0 or random.random() < np.exp(-delta / T):
            c5 = new_c5
            # Update indices
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            if c5 < best_c5:
                best_c5, best_h = c5, h.copy()
        else:
            # Revert swap
            h[i], h[j] = h[j], h[i]
            
        # Decrease temperature
        T *= cooling
        
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    """Main function to find improved upper bound for C5"""
    start_time = time.time()
    
    # Use parameter values based on strategy
    n_points = 200
    
    # Allocate time for multiple restarts

    max_restarts = int((budget_s * 0.9) / time_per_restart)
    
    best_c5 = float('inf')
    best_h = None
    
    # For multiple restarts
    for restart in range(max_restarts):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Run simulated annealing with random seed
        h, c5 = solve_binary(n_points, 5000, seed=seed + restart)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Break early if we have a very good result
        if best_c5 <= 0.379:
            break
    
    # If no good result found, try smooth optimization approach
    if best_c5 == float('inf') or best_c5 > 0.38:
        # Try a continuous approach with gradient descent on a subsample
        n_try = min(100, n_points)  # Use fewer points for speed
        h_init = np.array([0.5] * n_try)
        # Find best positions to initialize based on some heuristics
        # Use iteratively improve on a fraction of points
        h = np.array([0.5] * n_points)
        for i in range(0, n_points - 1, 2):
            h[i] = 1.0
        eval_c5 = evaluate_c5(h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
            best_h = h.copy()
    
    # If still no good result, ensure at least one valid construction

        best_c5 = evaluate_c5(best_h)
