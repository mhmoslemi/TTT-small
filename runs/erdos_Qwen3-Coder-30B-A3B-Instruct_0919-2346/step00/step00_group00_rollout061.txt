import numpy as np
import time

def compute_c5(h):
    """Compute C5 bound for given step function h"""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return corr.max()

def propose_swap_and_update(h, corr, i, j):
    """Update correlation array after swapping h[i] and h[j]"""
    # Get old values
    old_i = h[i]
    old_j = h[j]
    
    # Compute delta
    delta_i = old_j - old_i
    
    # Alias for length
    n = len(h)
    
    # Create copy of correlation array for computation
    new_corr = corr.copy()
    
    # Loop over k not i,j to compute contributions
    for k in range(n):
        if k == i or k == j:
            continue
        hk = h[k]
        
        # contribution from p = i, q = k: h[i] * (1 - h[k])
        # shift i->k: delta = delta_i * (1 - hk)
        idx1 = k - i + n - 1
        new_corr[idx1] += delta_i * (1 - hk)
        
        # contribution from p = k, q = i: h[k] * (1 - h[i])
        # shift k->i: delta = hk * (-delta_i)
        idx2 = i - k + n - 1
        new_corr[idx2] += hk * (-delta_i)

        # contribution from p = j, q = k: h[j] * (1 - h[k])
        # shift j->k: delta = -delta_i * (1 - hk)
        idx3 = k - j + n - 1
        new_corr[idx3] += -delta_i * (1 - hk)
        
        # contribution from p = k, q = j: h[k] * (1 - h[j])
        # shift k->j: delta = hk * (delta_i)
        idx4 = j - k + n - 1
        new_corr[idx4] += hk * delta_i

    # Handle cross terms
    # Pair (i,j): old = h[i]*(1-h[j]), new = h[j]*(1-h[i])
    # delta = new - old = h[j]*(1-h[i]) - h[i]*(1-h[j]) = h[j] - h[j]*h[i] - h[i] + h[i]*h[j]
    # delta = h[j] - h[i] = delta_i
    
    # shift: j - i
    idx_ij = j - i + n - 1
    new_corr[idx_ij] += delta_i

    # Pair (j,i): old = h[j]*(1-h[i]), new = h[i]*(1-h[j])
    # delta = h[i] - h[j] = -delta_i
    
    # shift: i - j
    idx_ji = i - j + n - 1
    new_corr[idx_ji] -= delta_i
    
    return new_corr

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start = time.time()
    
    best_c5 = np.inf
    best_h = None
    best_n = None
    
    # Try several n values
    n_candidates = [60, 80, 100, 120, 140, 160, 180, 200]
    
    for n in n_candidates:
        if time.time() - start > budget_s - 1:
            break
            
        if n % 2 != 0:
            continue
            
        # Set parameters for local search
        time_left = budget_s - (time.time() - start)
        max_iters = int(60000 / (n/120))
        max_iters = max(max_iters, 2000)
        
        # Initialize binary h with n/2 ones
        h = np.zeros(n, dtype=float)
        ones = n // 2
        h[np.random.choice(n, ones, replace=False)] = 1.0
        
        # Compute initial correlation and C5
        corr = np.correlate(h, 1 - h, mode='full')
        cur_c5 = corr.max() * (2.0 / n)
        cur_max = corr.max()
        
        # Temperature schedule
        T = 0.05
        T_decay = 0.9995
        no_improvement_count = 0
        
        # Local search loop
        for it in range(max_iters):
            if time.time() - start > budget_s - 0.5:
                break
                
            # Propose a swap between a 1 and a 0
            i = np.random.randint(n)
            j = np.random.randint(n)
            while h[i] == h[j]:
                i = np.random.randint(n)
                j = np.random.randint(n)
                
            # Compute new correlation after swapping
            new_corr = propose_swap_and_update(h, corr, i, j)
            new_max = new_corr.max()
            new_c5 = new_max * (2.0 / n)
            
            # Accept or reject based on simulated annealing
            d = new_c5 - cur_c5
            if d < 0 or np.random.rand() < np.exp(-d / (T + 1e-12)):
                corr = new_corr
                h[i], h[j] = h[j], h[i]
                cur_c5 = new_c5
                cur_max = new_max
                no_improvement_count = 0
            else:
                no_improvement_count += 1
            
            # Update best
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                best_n = n
                
            # Temperature cooling
            T *= T_decay
            
            # Restart if no improvement for a while
            if no_improvement_count > 5000:
                h = np.zeros(n, dtype=float)
                ones = n // 2
                h[np.random.choice(n, ones, replace=False)] = 1.0
                corr = np.correlate(h, 1 - h, mode='full')
                cur_c5 = corr.max() * (2.0 / n)
                cur_max = corr.max()
                no_improvement_count = 0
                T = 0.05  # Reset temperature
                
    # Return best solution
    if best_h is None:
        # Fallback to initial_h_values or simple construction
        try:
            from __main__ import initial_h_values
            best_h = initial_h_values.copy()
            best_n = len(best_h)
            best_c5 = compute_c5(best_h)
        except:
            # Default simple construction
            n = 2
            best_h = np.array([1.0, 0.0])
            best_n = 2
            best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, best_n
