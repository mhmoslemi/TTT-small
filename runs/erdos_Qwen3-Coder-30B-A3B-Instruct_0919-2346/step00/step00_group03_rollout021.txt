import numpy as np
import time
import random

def evaluate_c5(h):
    """Evaluate the C5 bound for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1-h, mode="full") * dx
    return np.max(conv)

def random_binary_seq(n):
    """Generate a random binary sequence with exactly n/2 ones."""
    h = np.zeros(n)
    ones_indices = np.random.choice(n, n//2, replace=False)
    h[ones_indices] = 1
    return h

def solve_binary(n, iters, init_h=None, seed=None):
    """Perform simulated annealing on binary sequences."""
    rng = np.random.default_rng(seed)
    if init_h is not None:
        h = init_h.copy()
    else:
        h = random_binary_seq(n)
    
    c5 = evaluate_c5(h)
    best_h, best_c5 = h.copy(), c5
    
    T0 = 0.1
    T = T0
    cooling = 0.9999
    step = 1
    
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    
    for iteration in range(iters):
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = rng.choice(ones)
        j = rng.choice(zeros)
        
        # propose swap
        h[i], h[j] = h[j], h[i]
        
        # update indices
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        
        new_c5 = evaluate_c5(h)
        delta = new_c5 - c5
        
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            c5 = new_c5
            if c5 < best_c5:
                best_c5, best_h = c5, h.copy()
        else:
            h[i], h[j] = h[j], h[i]
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
        
        T *= cooling
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    """Find an improved upper bound for the Erdős minimum overlap problem constant C₅."""
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    
    # Use larger n for better resolution
    n = 200
    dx = 2.0 / n
    
    # Set iterations per restart
    iters_per_restart = 15000
    
    # Run multiple restarts
    restarts = 0
    while time.time() - start_time < budget_s * 0.95 and restarts < 1000:
        rng = np.random.default_rng(seed + restarts)
        h, c5 = solve_binary(n, iters_per_restart, seed=seed + restarts)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        restarts += 1
        
        # Adjust to avoid stagnation
        if restarts % 50 == 0:
            iters_per_restart = min(20000, iters_per_restart * 1.1)
    
    return (best_h.tolist(), best_c5, n)
