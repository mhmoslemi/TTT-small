import time
import numpy as np
from cvxpy import *
import cvxpy as cp
from scipy import signal
from scipy.fft import fft, ifft
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    def get_remaining_time():
        return budget_s - (time.time() - start_time)
        
    # Set parameters
    n = 60
    dx = 2.0 / n
    
    # Initial shift set
    K0 = [1, n//2, 2]
    
    # Precompute indices for circular shifts
    idx_mod = np.arange(n)
    
    # Build initial model
    h = cp.Variable(n, integer=True)
    t = cp.Variable(integer=True)
    w = {}
    a = {}
    
    # Constraints for the initial model
    constrs = [0 <= h, h <= 1, cp.sum(h) == n//2, h[0] == 1]  # Fix h[0] = 1 for symmetry breaking
    # Add constraints for initial shifts
    for k in K0:
        w[k] = cp.Variable(n, integer=True)
        a[k] = cp.sum(w[k])
        for i in range(n):
            i_k = (i + k) % n
            constrs.extend([
                w[k][i] <= h[i],
                w[k][i] <= h[i_k],
                w[k][i] >= h[i] + h[i_k] - 1,
                w[k][i] >= 0
            ])
        constrs.append(t <= a[k])
    
    # Objective: maximize t
    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constrs)
    
    # Solve initial problem
    try:
        prob.solve(solver=cp.CBC, max_seconds=min(30, get_remaining_time()))
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE] and not np.isnan(t.value):
            best_h = np.round(h.value).astype(int)
            t_val = t.value
        else:
            # Fallback to a random solution
            best_h = np.zeros(n, dtype=int)
            ones_indices = np.random.choice(n, size=n//2, replace=False)
            best_h[ones_indices] = 1
            t_val = 0
    except:
        # Fallback to a random solution
        best_h = np.zeros(n, dtype=int)
        ones_indices = np.random.choice(n, size=n//2, replace=False)
        best_h[ones_indices] = 1
        t_val = 0
        
    if get_remaining_time() < 20:
        # Return best integer solution
        h_continuous = best_h.astype(float)
        h_continuous /= np.sum(h_continuous) * dx
        c5_bound = 1 - (2/n) * t_val
        return (h_continuous, c5_bound, n)
    
    # Cut generation loop
    K = set(K0)
    last_improvement = 0
    max_improvement_iter = 100  # Prevent infinite running
    
    while len(K) < n-1 and max_improvement_iter > 0:
        if get_remaining_time() < 20:
            break
            
        max_improvement_iter -= 1
        
        # Compute exact autocorrelations for all k
        A = {}
        for k in range(1, n):
            A[k] = np.sum(best_h * np.roll(best_h, k))
        
        # Find minimum A_k
        min_k = min(A, key=A.get)
        min_A = A[min_k]
        
        # If already optimal or not improving
        if min_A >= t_val:
            break
            
        # Add new constraint if min_k is not already in K
        if min_k not in K:
            K.add(min_k)
            # Add new variables and constraints for this shift
            w[min_k] = cp.Variable(n, integer=True)
            a[min_k] = cp.sum(w[min_k])
            for i in range(n):
                i_k = (i + min_k) % n
                constrs.extend([
                    w[min_k][i] <= h[i],
                    w[min_k][i] <= h[i_k],
                    w[min_k][i] >= h[i] + h[i_k] - 1,
                    w[min_k][i] >= 0
                ])
            constrs.append(t <= a[min_k])
            
            # Update the problem and solve
            try:
                prob.update()
                prob.solve(solver=cp.CBC, max_seconds=min(20, get_remaining_time()))
                if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE] and not np.isnan(t.value):
                    t_val = t.value
                    best_h = np.round(h.value).astype(int)
                    last_improvement = 0
                else:
                    break
            except:
                break
            
        else:
            # No need to add new constraint
            break
    
    h_continuous = best_h.astype(float)
    h_continuous /= np.sum(h_continuous) * dx
    c5_bound = 1 - (2/n) * t_val
    
    # Optionally perform gradient polishing if we have time
    if get_remaining_time() > 150:
        def smooth_h(h_arr, k):
            return np.convolve(h_arr, np.roll(h_arr, k), mode='valid')
        
        def score(h_arr):
            # Compute circ autocorrelations
            autocorrelations = np.zeros(n)
            for k in range(1, n):
                autocorrelations[k] = np.sum(h_arr * np.roll(h_arr, k))
            return np.min(autocorrelations)
        
        def gradient(h_arr, k):
            # Gradient w.r.t. a_k
            grad = np.zeros(n)
            for i in range(n):
                grad[i] = h_arr[(i + k) % n] + h_arr[(i - k) % n]
            return grad
            
        # Perform several gradient ascent steps to smooth
        h_smooth = h_arr.copy()
        best_h_smooth = None
        best_score = score(h_smooth)
        
        for it in range(200):
            if get_remaining_time() < 10:
                break
                
            # We compute the shift where auto-correlation is currently minimal
            autocorrelations = np.zeros(n)
            for k in range(1, n):
                autocorrelations[k] = np.sum(h_smooth * np.roll(h_smooth, k))
            k_star = np.argmin(autocorrelations)
            
            # Compute gradient
            grad = gradient(h_smooth, k_star)
            step_size = 0.01
            
            # Apply gradient step
            h_new = h_smooth + step_size * grad
            
            # Project back to [0,1]
            h_new = np.clip(h_new, 0, 1)
            
            # Maintain sum fixed via pairwise swap
            if np.sum(h_new) != n/2:
                delta = n/2 - np.sum(h_new)
                # Find one index > 0 and one < 1
                pos_idx = np.where(h_new > 0)[0]
                neg_idx = np.where(h_new < 1)[0]
                if len(pos_idx) > 0 and len(neg_idx) > 0:
                    i = pos_idx[0]
                    j = neg_idx[0]
                    if delta > 0:
                        # Move delta from i to j
                        move = min(delta, h_new[i])
                        h_new[i] -= move
                        h_new[j] += move
                    else:
                        move = min(-delta, 1 - h_new[j])
                        h_new[j] += move
                        h_new[i] -= move
                        
            h_smooth = h_new
            new_score = score(h_smooth)
            if new_score > best_score:
                best_score = new_score
                best_h_smooth = h_smooth.copy()
                
        if best_h_smooth is not None:
            h_continuous = best_h_smooth
            c5_bound = 1 - (2/n) * best_score
            
    return (h_continuous, c5_bound, n)
