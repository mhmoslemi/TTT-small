def evaluate_c5(h):
    """Evaluate the C5 bound for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h) using full mode
    corr = np.correlate(h, 1 - h, mode='full')
    # Get the maximum value
    max_corr = np.max(corr)
    return max_corr * dx

def random_binary_seq(n):
    """Generate a random binary sequence of length n with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)

    h[ones_indices] = 1.0
    return h
