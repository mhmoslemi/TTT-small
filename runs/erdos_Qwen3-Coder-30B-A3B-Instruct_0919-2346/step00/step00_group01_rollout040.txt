import numpy as np
from scipy.optimize import minimize
import time

def evaluate_h(h_values):
    """Evaluate the overlap integral C5 for given step function h"""
    n = len(h_values)
    dx = 2.0 / n
    # Compute correlation of h with (1-h) using full mode
    cor = np.correlate(h_values, 1 - np.array(h_values), mode='full')
    # Maximum value is the worst-case overlap
    max_cor = np.max(cor)
    return max_cor * dx

def project_onto_constraints(h, n_points):
    """Project h onto constraints: 0 <= h[i] <= 1 and sum(h) = n_points / 2"""
    # Clip values to [0,1]
    h = np.clip(h, 0, 1)
    # Adjust to meet sum constraint
    current_sum = np.sum(h)
    desired_sum = n_points / 2
    diff = desired_sum - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Find indices that can be adjusted
    indices = np.argsort(h)
    if diff > 0:  # Need to increase sum
        for i in indices:
            if diff <= 0:
                break
            max_increase = 1 - h[i]
            actual_increase = min(max_increase, diff)
            h[i] += actual_increase
            diff -= actual_increase
    else:  # Need to decrease sum
        for i in indices[::-1]:
            if diff >= 0:
                break
            max_decrease = h[i]
            actual_decrease = min(max_decrease, -diff)
            h[i] -= actual_decrease
            diff += actual_decrease
    
    return h

    n_points = 100  # Modest number for reasonable runtime
    
    # Initialize with the provided initial values if available

    else:
        # Start with uniform distribution to get a good baseline
        h = np.random.rand(n_points)
        h[:n_points//2] = 1.0  # Set first half to 1
        h[n_points//2:] = 0.0  # Set second half to 0
        h = project_onto_constraints(h, n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_h(best_h)
    
    start_time = time.time()
    iter_count = 0
    
    while time.time() - start_time < budget_s:
        iter_count += 1
        # Make small random changes
        h_new = h.copy()
        # Pick two random indices
        i1, i2 = np.random.choice(n_points, 2, replace=False)
        # Determine maximum change
        delta = np.random.rand() * 0.1
        if h_new[i1] + delta <= 1.0 and h_new[i2] - delta >= 0.0:
            h_new[i1] += delta
            h_new[i2] -= delta
        elif h_new[i1] - delta >= 0.0 and h_new[i2] + delta <= 1.0:
            h_new[i1] -= delta
            h_new[i2] += delta
        else:
            continue
            
        # Project onto constraints
        h_new = project_onto_constraints(h_new, n_points)
        
        # Evaluate
        c5_new = evaluate_h(h_new)
        
        # Accept improvement or with small probability
        if c5_new < best_c5 or np.random.rand() < 0.01:
            h = h_new.copy()
            if c5_new < best_c5:
                best_h = h.copy()
                best_c5 = c5_new
