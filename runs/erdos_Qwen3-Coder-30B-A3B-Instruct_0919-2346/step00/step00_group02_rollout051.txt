import numpy as np
import time
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=600, **kwargs):

    n_points = len(initial_h)
    
    # Scale initial to target sum
    current_sum = np.sum(initial_h)
    target_sum = n_points / 2.0
    initial_h = initial_h * (target_sum / current_sum)
    
    # Use scipy.optimize.differential_evolution for global optimization
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Objective function to minimize
    def objective(h):
        # Compute correlation
        corr = np.correlate(h, h, mode='full')
        # Compute C5 = max_k int h(x)(1 - h(x+k)) dx
        # Which equals 1 - min_k int h(x) h(x+k) dx = 1 - min_k corr_k
        min_corr = np.min(corr)
        return 1 - min_corr
    
    # Use differential evolution with multiple restarts
    start_time = time.time()
    
    # Best solution so far
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    # Try several restarts
    max_restarts = 10
    for restart in range(max_restarts):
        # Stop if budget is almost used
        if time.time() - start_time > budget_s - 5:
            break
            
        # Set random seed for this run
        np.random.seed(seed + restart)
        
        # Run differential evolution
        try:
            result = differential_evolution(
                objective,
                bounds,
                seed=seed + restart,
                maxiter=200,
                popsize=15,
                tol=1e-6, 
                mutation=(0.5, 1),
                recombination=0.7,
                disp=False
            )
            
            if result.success:
                h = result.x
                # Make sure it satisfies constraint exactly
                h_sum = np.sum(h)
                h = h * (target_sum / h_sum)
                
                c5 = objective(h)
                if c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = c5
        except:
            continue
    
    # Final refinement with simulated annealing
    # Reset time
    start_time = time.time()
    current_h = best_h.copy()
    current_c5 = best_c5
    temperature = 1.0
    cooling_rate = 0.9999
    min_temp = 1e-6
    max_iter = 100000
    iteration = 0
    
    while temperature > min_temp and iteration < max_iter and time.time() - start_time < budget_s - 2:
        # Random pair swap with delta
        i = np.random.randint(0, n_points)
        j = np.random.randint(0, n_points)
        if i == j:
            continue
            
        # Compute max delta
        hi_up = 1.0 - current_h[i]
        hj_down = current_h[j]
        h_i_up = 1.0 - current_h[i]
        h_j_down = current_h[j]
        delta_min = max(-current_h[i], hj_down - 1.0)
        delta_max = min(hi_up, h_j_down)
        
        if delta_min >= delta_max:
            continue
        
        # Choose delta
        delta = np.random.uniform(delta_min, delta_max)
        
        # Apply change
        new_h = current_h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        
        # Compute new C5
        # Maintain exact sum constraint
        new_sum = np.sum(new_h)
        adjustment = (target_sum - new_sum) / n_points
        new_h += adjustment
        
        # Ensure constraints
        new_h = np.clip(new_h, 0, 1)
        
        # Fixed sum due to adjustment
        c5 = objective(new_h)
        
        # Accept or reject based on temperature and change in objective
        delta_c5 = c5 - current_c5
        if delta_c5 < 0 or np.exp(-delta_c5 / temperature) > np.random.rand():
            current_h = new_h
            current_c5 = c5
            
        temperature *= cooling_rate
        iteration += 1
        
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
    
    # Final evaluation
    final_c5 = objective(best_h)
    
    return (best_h, final_c5, n_points)
