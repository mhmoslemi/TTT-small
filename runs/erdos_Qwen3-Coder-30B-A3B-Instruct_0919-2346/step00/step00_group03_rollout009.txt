import numpy as np
import time
from scipy.optimize import differential_evolution
from scipy.fft import fft, ifft

def evaluate_h(h, dx):
    F = fft(h)
    G = fft(1 - h)
    corr = ifft(F * np.conj(G)).real
    return dx * np.max(corr)

def project_sum(h, target_sum):
    current_sum = np.sum(h)
    delta = (current_sum - target_sum) / len(h)
    h = h - delta
    h = np.clip(h, 0, 1)
    return h

def apply_dykstra_projection(h, target_sum, max_iter=4):
    for _ in range(max_iter):
        current_sum = np.sum(h)
        delta = (current_sum - target_sum) / len(h)
        h = h - delta
        h = np.clip(h, 0, 1)
    return h

def mismatch_swap_refinement(h, dx, max_swaps=5000):
    for _ in range(max_swaps):

        G = fft(1 - h)
        corr = ifft(F * np.conj(G)).real
        max_shift = np.argmax(corr)
        max_val = corr[max_shift]
        
        if max_val <= dx * 0.379:  # set target lower bound
            break

        M1 = []
        M0 = []
        shift_len = len(h) - max_shift if max_shift >= len(h) // 2 else max_shift
        for i in range(shift_len):
            idx1, idx2 = i, i + max_shift
            if h[idx1] > 0.99 and h[idx2] < 0.01:
                M1.append(i)
            elif h[idx1] < 0.01 and h[idx2] > 0.99:
                M0.append(i)
        
        if not M1 or not M0:
            break
            
        i = M1[0]
        j = M0[0]
        h[i], h[j] = h[j], h[i]
        h = apply_dykstra_projection(h, len(h) / 2)
    
    return h
