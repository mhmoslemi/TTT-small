import numpy as np
import time
import random

def compute_c5(h):
    """Compute C5 using correlation as defined in problem statement."""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return corr.max()

def propose_swap_and_update(h, corr, i, j):
    """Apply swap and return new correlation array."""
    a = h[i]
    b = h[j]
    delta_i = b - a
    
    n = len(h)
    new_corr = corr.copy()
    
    # Loop over k not equal to i or j
    for k in range(n):
        if k == i or k == j:
            continue
        hk = h[k]
        
        # contributions from h[i] * (1 - h[k])
        idx1 = k - i + n - 1
        new_corr[idx1] += delta_i * (1 - hk)
        
        # contributions from h[k] * (1 - h[i])
        idx2 = i - k + n - 1
        new_corr[idx2] += hk * (-delta_i)
        
        # contributions from h[j] * (1 - h[k])
        idx3 = k - j + n - 1
        new_corr[idx3] += -delta_i * (1 - hk)
        
        # contributions from h[k] * (1 - h[j])
        idx4 = j - k + n - 1
        new_corr[idx4] += hk * (delta_i)
    
    # handle cross pairs
    idx_ij = j - i + n - 1
    new_corr[idx_ij] += delta_i
    
    idx_ji = i - j + n - 1
    new_corr[idx_ji] -= delta_i
    
    # apply the swap
    h[i], h[j] = b, a
    
    return new_corr

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()
    
    # Use larger n for better resolution
    n = 140  # even number
    if n % 2 != 0:
        n += 1
    ones = n // 2
    
    best_c5 = np.inf
    best_h = None
    best_n = None
    
    # Use initial_h_values if available
    try:
        from __main__ import initial_h_values
        if len(initial_h_values) == n:
            h = initial_h_values.copy()
        else:
            # scale to n if needed
            h = np.zeros(n)
            for i in range(n):
                idx = int(float(i) * len(initial_h_values) / n)
                h[i] = initial_h_values[idx] if 0 <= idx < len(initial_h_values) else 0.0
    except Exception:
        # Generate random binary with exact half ones
        h = np.zeros(n, dtype=float)
        indices = np.random.choice(n, size=ones, replace=False)
        h[indices] = 1.0
    
    # Compute initial correlation
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    cur_c5 = corr.max() * dx
    cur_max = corr.max()
    cur_h = h.copy()
    
    # Set up simulated annealing
    temp = 0.05
    temp_decay = 0.9999
    max_iters = 50000
    accept_count = 0
    last_improvement = 0
    
    for iteration in range(max_iters):
        if time.time() - start > budget_s - 1:
            break
            
        # Early termination if no progress
        if iteration - last_improvement > 10000:
            break
            
        # Generate swap indices
        i = np.random.randint(n)
        j = np.random.randint(n)
        while h[i] == h[j]:
            i = np.random.randint(n)
            j = np.random.randint(n)
        
        # Propose swap and compute new correlation
        new_corr = propose_swap_and_update(h, corr, i, j)
        new_max = new_corr.max()
        new_c5 = new_max * dx
        
        # Accept or reject
        if new_c5 < cur_c5 or np.random.rand() < np.exp(-(new_c5 - cur_c5) / (temp + 1e-12)):
            corr = new_corr
            cur_c5 = new_c5
            cur_max = new_max
            accept_count += 1
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                best_n = n
                last_improvement = iteration
        else:
            # Revert the swap in h since we rejected
            h[i], h[j] = h[j], h[i]
        
        # Update temperature
        temp *= temp_decay
    
    # Fallback to initial_h_values if needed
    if best_h is None:
        try:
            from __main__ import initial_h_values
            best_h = initial_h_values.copy()
            best_n = len(best_h)
            best_c5 = compute_c5(best_h)
        except Exception:
            # default simplest one
            best_h = np.array([1.0, 0.0])
            best_c5 = compute_c5(best_h)
            best_n = 2

    return best_h, best_c5, best_n
