def evaluate_c5(h):
    """Evaluate the C5 bound for a given step function h."""
    # Ensure h is a numpy array
    h = np.array(h)
    n = len(h)
    # The correlation computes the overlap integral for all shifts
    # Using 'full' mode to get all possible overlaps
    correlation = np.correlate(h, 1 - h, mode='full')
    # The maximum value of the overlap integral
    c5 = np.max(correlation) * (2.0 / n)
    return c5
