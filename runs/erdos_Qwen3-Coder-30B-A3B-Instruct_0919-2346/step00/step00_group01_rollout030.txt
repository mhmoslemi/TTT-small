import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_h(h):
    """Evaluate the C5 bound for given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h) using full mode
    corr = np.correlate(h, 1 - h, mode='full')
    # We want: max_k integral_h(x) * (1 - h(x+k)) dx 
    # Each term in correlation corresponds to shift k
    # Maximum shift k is n-1, so we look at indices from n-1 to 2n-2
    # But correlation with mode 'full' returns 2n-1 elements
    # The center (n-1) element corresponds to shift 0
    # For shift k > 0, use indices from n+k to 2n-2
    # For shift k < 0, use indices from n-1+k to n-1
    # All correlation values are valid
    max_corr = np.max(corr)
    c5 = max_corr * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize with a reasonable starting point
    h = np.ones(n_points) * 0.5  # Start with constant function
    # Perturb slightly for starting solution
    h += np.random.normal(0, 0.05, n_points)
    h = np.clip(h, 0, 1)
    # Adjust to ensure integral = 1 (sum * dx = 1)
    adjustment = (1 - np.sum(h) * dx) / n_points
    h += adjustment
    h = np.clip(h, 0, 1)
    
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_h(h)
    
    # Simulated Annealing approach
    T = 1.0
    cooling_rate = 0.9999
    min_T = 1e-8
    steps_per_T = 50
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        for _ in range(steps_per_T):
            # Create a new candidate solution
            h_new = h.copy()
            
            # Make small random changes
            idx1 = np.random.randint(0, n_points)
            idx2 = np.random.randint(0, n_points)
            delta = np.random.normal(0, 0.05)
            
            # Apply change to two random indices
            h_new[idx1] += delta
            h_new[idx2] -= delta
            
            # Ensure values remain in [0,1]
            h_new = np.clip(h_new, 0, 1)
            
            # Adjust to maintain constraint sum(h) * dx = 1
            actual_sum = np.sum(h_new) * dx
            adjustment = (1 - actual_sum) / n_points
            h_new += adjustment
            h_new = np.clip(h_new, 0, 1)
            
            # Evaluate
            new_c5 = evaluate_h(h_new)
            
            # Accept or reject based on simulated annealing
            if new_c5 < best_c5 or np.random.rand() < np.exp(-(new_c5 - best_c5) / T):
                h = h_new
                if new_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = new_c5
            
        # Cool down
        T *= cooling_rate
        if T < min_T:
            T = min_T
            
        iteration += 1
        
    return (best_h, best_c5, n_points)
