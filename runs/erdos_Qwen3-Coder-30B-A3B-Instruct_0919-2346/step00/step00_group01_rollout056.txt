import time

# Predefined globals
initial_h_values = None

def evaluate_erdos_solution(h):
    n = len(h)
    # Pad for circular convolution
    h_padded = np.concatenate([h, np.zeros(n)])
    h_neg = 1 - h
    h_neg_padded = np.concatenate([h_neg, np.zeros(n)])
    corr = np.real(np.fft.ifft(np.fft.fft(h_padded) * np.conj(np.fft.fft(h_neg_padded))))
    # Take the full correlation
    corr_full = corr[:n]
    # Minimize over shifts
    c5 = 1.0 - 2.0 / n * np.min(corr_full)
    return c5

def sdp_solve(n, seed=42):
    # Set random seed for reproducibility
    np.random.seed(seed)
    
    # Variables
    h = cp.Variable(n)
    t = cp.Variable()
    
    # Constraints
    constraints = [
        cp.sum(h) == n / 2,
        h >= 0,
        h <= 1
    ]
    
    # Autocorrelation constraints from SDP: trace(M_k * X) >= t
    # We're adding the quadratic constraints in their linearized form:
    # trace(M_k * X) = h_k * h_{k+j} + h_{k+j} * h_k >= t for all k, j in the range
    # This is an approximate SDP relaxation
    
    # We'll add a direct enforcement of the autcorrelation for simpler handling
    
    # Since SDP solving with cvxpy + SCS can be slow, we compute an approximate solution
    # Here we represent a known good start solution
    
    # From initial construction
    h_penalty = np.array(initial_h_values)
    h = h_penalty.copy()
    
    # Compute the SDP bounds approximately
    # Reduces to: 
    # minimize -t s.t. h sum = n/2, 0 <= h <= 1, and for each shift k:
    #      sum_i h[i] * h[i+k] >= t
    
    # We skip the full SDP solve in this simplified script for time.
    # Instead, we will work with high-quality starting vectors like those from initial_h_values
    return h, 1.0  # dummy t value for fallback

def round_h(X_sol, seed=42):
    np.random.seed(seed)
    n = len(X_sol)
    
    # Diagonal of solution is h
    h = np.diag(X_sol)
    
    # If not rank-1, we do Cholesky eigen-decomp of outer product
    try:
        # Use eigen-decomposition with rank k approximation
        _, V = eigh(X_sol)
        V = V[:,::-1]
        V = V[:, :max(1, min(5, V.shape[1]))]
        g = np.random.randn(V.shape[1])
        scores = V.T @ g
        # Sort scores
        sorted_idx = np.argsort(scores)[::-1]
        
        # Determine threshold to get exactly n/2 ones
        median_idx = int(n/2)
        theta = scores[sorted_idx[median_idx-1]] if median_idx < len(scores) else np.inf
        
        # Assign binary value
        h_binary = (scores >= theta).astype(float)
        
        # Adjust threshold to exactly sum to n/2
        current_sum = np.sum(h_binary)
        if current_sum < n / 2:
            # Add more ones
            ones_to_add = int(n / 2 - current_sum)
            not_ones = np.where(h_binary == 0)[0]
            to_set = not_ones[:ones_to_add]
            h_binary[to_set] = 1.0
        elif current_sum > n / 2:
            # Remove some ones
            ones_to_remove = int(current_sum - n / 2)
            is_ones = np.where(h_binary == 1)[0]
            to_unset = is_ones[:ones_to_remove]
            h_binary[to_unset] = 0.0
            
        h = h_binary
        
        # Ensure sum constraint exactly met
        if abs(np.sum(h) - n / 2) > 1e-10:
            print("Warning: sum of h does not exactly match n/2 (SDP rounding)")
            
    except:
        # fallback
        h = np.random.rand(n)
        h = (h > 0.5).astype(float)
        h = h.astype(float)
        if np.sum(h) > n/2:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.sum(h) - n/2)):
                h[indices[i]] = 0
            
        if np.sum(h) < n/2:
            indices = np.argsort(h)
            for i in range(int(n/2 - np.sum(h))):
                h[indices[i]] = 1
                
    return h

def swap_hill_climb(h, n):
    """Apply hill-climb swapping with freeze-out."""
    h = h.copy()  # don't modify original
    best_h = h.copy()
    best_val = evaluate_erdos_solution(best_h)
    changed = True
    max_iter = 500  # limit iterations for modest speed
    it = 0
    while changed and it < max_iter:
        changed = False
        for i in range(n):
            # find zero position
            if not h[i]:
                continue
            for j in range(n):
                if h[j]:
                    continue
                # Try swap
                h[i], h[j] = 0, 1
                val = evaluate_erdos_solution(h)
                if val < best_val:
                    best_h = h.copy()
                    best_val = val
                    changed = True
                else:
                    h[i], h[j] = 1, 0  # revert
        it += 1
    return best_h, best_val

def gradient_fine_tune(h, n, steps=200, eta=0.01):
    """Projected gradient ascent on the worst-shifted correlation."""
    h = h.copy()
    max_iter = steps
    for _ in range(max_iter):
        # Compute all overlapping values with respect to shifts
        # Evaluate circular correlation for all shifts, get the worst using FFT
        corr_full = np.zeros((n,))  # for zero padding and performance
        # prepare h_padded
        h_full = np.concatenate([h, np.zeros(n)])
        h_neg = 1-h
        h_neg_full = np.concatenate([h_neg, np.zeros(n)])
        fft_h = np.fft.fft(h_full)
        fft_hneg = np.fft.fft(h_neg_full)
        corr = np.real(np.fft.ifft(fft_h * np.conj(fft_hneg)))
        corr_values = corr[:n]
        k_star = np.argmin(corr_values)
        # Evaluate gradient just at k_star
        grad = np.zeros(n)
        for i in range(n):
            mod_i_plus = (i + k_star) % n
            mod_i_minus = (i - k_star) % n
            if i < n and mod_i_plus < n:
                grad[i] += h[mod_i_plus]  # contribution through h_{i+k}
            if i < n and mod_i_minus < n:
                grad[i] += h[mod_i_minus]  # contribution through h_{i-k}
        
        # Update with projected gradient
        h_new = h + eta * grad
        # Project to box [0,1] and sum constraint
        # sum correction
        h_minus = np.sum(h_new) - n/2
        h_new = np.clip(h_new, 0, 1)
        # Ensure sum constraint is satisfied exactly
        scale_factor = n/2 / np.sum(h_new)
        h_new *= scale_factor
        h = h_new

    return h, evaluate_erdos_solution(h)

    # Heuristic optimal construction

            lengths.append(b[i] - a[i])
        total_sum = sum(lengths)
        if total_sum != 1.0:
            return 10000.0  # invalid solution

        # Now compute the bilateral correlations

def run(seed=42, budget_s=600):
    start = time.time()
    best_c5 = float("inf")
    best_h = None
    best_n = 0
    
    # Default n size for SDP and speed
    n = 80  # keep under budget
    
    # Later we can add fallback for larger n if needed
    
    if time.time() - start > budget_s - 10:
        # If not enough time, still compute the interval part
        h_interval, c5_interval = interval_optimize(n)
        if c5_interval < best_c5:
            best_c5 = c5_interval
            best_h = h_interval
            best_n = n

    # SDP Section
    if time.time() - start > budget_s - 50:
        pass
    else:
        # This part enters and repeated
        for seed_iter in range(30):  # try 30 SDP rounds
            if time.time() - start > budget_s - 20:
                break
            try:
                # SDP samples potentially, but in our version limited to sample good h
                h_trial, t = sdp_solve(n, seed + seed_iter)
                
                # Post-processing with random hyperplane rounding
                if time.time() - start > budget_s - 30:
                    break
                    
                h_rounded = round_h(np.outer(h_trial, h_trial), seed + seed_iter)
                
                # Swapping hill climb
                h_swapped, c5swapped = swap_hill_climb(h_rounded, n)
                
                # Final gradient fine tune
                h_final, c5 = gradient_fine_tune(h_swapped, n)
                
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_final.copy()
                    best_n = n
                    
                if time.time() - start > budget_s - 20:
                    break
            except:
                continue
    
    # Even if not fully optimized, in desperation, we use the interval method
    if best_h is None or time.time() - start > budget_s - 50:
        h_interval, c5_interval = interval_optimize(n)
        if h_interval is not None and c5_interval < best_c5:
            best_c5 = c5_interval
            best_h = h_interval
            best_n = n

    if best_h is None:
        # If nothing found, fallback to baseline
        try:
            best_h = np.array(initial_h_values)
            best_c5 = evaluate_erdos_solution(best_h)
            best_n = len(best_h)
        except:
            best_h = np.ones(n) * 0.5
            best_c5 = 1.0
            best_n = n

    return (best_h, best_c5, best_n)
