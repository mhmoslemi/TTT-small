import numpy as np
import time
from cvxpy import *
import cvxpy as cp

def evaluate(h):
    """Evaluate the overlap integral C5 = max_k ∫ h(x)(1 - h(x+k)) dx via correlation."""
    n = len(h)
    dx = 2.0 / n
    # np.correlate with mode="full" computes all lags, but we only want lags from -(n-1) to n-1
    # and we multiply by dx
    corr = np.correlate(h, 1 - h, mode="full")
    # Extract the valid lags: k from -(n-1) to n-1, centered at index n-1
    # The full array length is 2*n - 1; k = index - (n-1)
    result = np.max(corr) * dx
    return result

def project_onto_capped_simplex(v, S):
    """Project vector v onto {h | 0 ≤ h_i ≤ 1, Σ h_i = S} using bisection on λ."""
    v = np.array(v)
    low, high = -2.0, 2.0
    tol = 1e-10

    def proj_sum(lam):
        h = np.clip(v - lam, 0, 1)
        return np.sum(h)

    while high - low > tol:
        mid = (low + high) / 2.0
        s = proj_sum(mid)
        if s > S:
            low = mid
        else:
            high = mid

    h = np.clip(v - (low + high) / 2.0, 0, 1)
    return h

def coeffs_and_const(h):
    """Compute a_{i,k} and c_k for all valid shifts k."""
    n = len(h)
    # a_{i,k} = 1 - h_{i+k} - h_{i-k} (with wrap-around handled by zero padding)
    a = np.zeros((n, 2 * n - 1))  # k from -(n-1) to n-1, indices k+n-1
    c = np.zeros(2 * n - 1)

    for k in range(-(n-1), n):
        idx_k = k + n - 1
        # Apply zero padding: indices where i+k >= n or i+k < 0 do not contribute
        # So h_{i+k} = 0 if i+k out of range
        h_shifted = np.zeros(n)
        if k >= 0:
            h_shifted[k:] = h[:n-k]
        else:
            h_shifted[:k] = h[-k:]
        # coeffs for this shift
        a[:, idx_k] = 1 - h_shifted - h[::-1]  # h_{i-k} after shift
        c[idx_k] = np.sum(h * h_shifted)
    return a, c

def linear_program(a_matrix, const_vec, n, S):
    """Solve the LP defined by the linearized problem."""
    # Variables
    h = cp.Variable(n)  # h each in [0,1]
    t = cp.Variable()
    # Objective
    objective = cp.Minimize(t)
    # Constraints
    constraints = [a_matrix @ h + const_vec <= t,  # t >= a_k @ h + c_k for all k
                   cp.sum(h) == S,  # mass constraint
                   h >= 0, h <= 1]  # box constraints
    # Solve
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        return h.value, t.value
    except:
        # fallback to a simple compromise solution
        h_val = np.full(n, 0.5)  # symmetric compromise
        h_val = project_onto_capped_simplex(h_val, S)
        return h_val, np.max(const_vec) + 1.0

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n = len(initial_h_values)
    n_points = n
    S = n / 2.0
    best_h = None
    best_C5 = np.inf

    max_iter = 30
    damping = 0.7

    # Outer restarts
    num_restarts = 0
    while (time.time() - start_time) < 0.95 * budget_s:
        num_restarts += 1
        if num_restarts > 100:
            break

        # random initialization
        h = np.random.rand(n) * 0.3
        h = project_onto_capped_simplex(h, S)

        # Run CCP iterations
        for iteration in range(max_iter):
            if (time.time() - start_time) > 0.98 * budget_s:
                break

            a, c = coeffs_and_const(h)
            h_new, t_new = linear_program(a, c, n, S)
            if h_new is None:
                # fallback if linear program failed
                h_new = h.copy()

            # Apply damping
            h = damping * h_new + (1 - damping) * h

            # Project onto simplex
            h = project_onto_capped_simplex(h, S)

            # Evaluate true C5
            try:
                current_C5 = evaluate(h)
            except:
                current_C5 = np.inf

            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = h.copy()

        # optionally add polish step
        if best_h is not None:
            h_polish = best_h.copy()
            # simple variation: mass transfers
            for _ in range(500):
                if (time.time() - start_time) > 0.98 * budget_s:
                    break
                i = np.random.randint(n)
                k = np.random.randint(-(n-1), n)
                j = i + k
                if j < 0 or j >= n:
                    continue
                delta = 1e-6
                if h_polish[i] > delta and h_polish[j] < 1 - delta:
                    h_polish[i] -= delta
                    h_polish[j] += delta
                    try:
                        new_C5 = evaluate(h_polish)
                        if new_C5 < best_C5:
                            best_C5 = new_C5
                            best_h = h_polish.copy()
                        else:
                            # revert
                            h_polish[i] += delta
                            h_polish[j] -= delta
                    except:
                        h_polish[i] += delta
                        h_polish[j] -= delta

    # finalize result
    if best_h is None:
        best_h = initial_h_values
        best_C5 = evaluate(best_h)
    return (best_h, best_C5, n_points)
