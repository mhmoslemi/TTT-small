import numpy as np

def evaluate_h(h_values, n_points):
    """Evaluate the C5 constant for given h values."""
    dx = 2.0 / n_points
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    corr = np.correlate(h, 1 - h, mode='full')
    # The maximum is what we want to minimize
    c5 = max(corr) * dx
    return c5

def project_onto_constraints(h, n_points):
    """Project h onto constraints: 0 <= h[i] <= 1 and sum(h) = n_points / 2"""
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Adjust sum to be exactly n_points / 2
    current_sum = np.sum(h)
    target_sum = n_points / 2
    
    if current_sum == target_sum:
        return h
    
    # Simple adjustment by scaling
    scale_factor = target_sum / current_sum
    h = h * scale_factor
    h = np.clip(h, 0, 1)
    
    return h

def local_search_step(h, n_points, step_size=0.01):
    """Perform a local search step to improve h"""
    # Sample a few pairs to adjust
    indices = list(range(n_points))
    random.shuffle(indices)
    for i in range(min(10, len(indices) // 2)):
        idx1, idx2 = indices[2*i], indices[2*i+1]

        delta = random.uniform(-step_size, step_size)
        new_h = h.copy()

        new_h[idx2] = min(1.0, max(0.0, new_h[idx2] - delta))
        new_h = project_onto_constraints(new_h, n_points)
        if evaluate_h(new_h, n_points) < evaluate_h(h, n_points):
            h = new_h
    return h
