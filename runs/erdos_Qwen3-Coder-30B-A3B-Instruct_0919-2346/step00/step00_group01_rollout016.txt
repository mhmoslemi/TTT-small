def circular_correlation(h):
    """Compute circular correlation of h with 1-h using FFT"""
    n = len(h)
    h_padded = np.pad(h, (0, n), 'constant', constant_values=(0, 0))
    one_minus_h_padded = np.pad(1-h, (0, n), 'constant', constant_values=(0, 0))
    corr = np.real(np.fft.ifft(np.fft.fft(h_padded) * np.fft.fft(one_minus_h_padded).conj()))
    return corr[n-1:]

def evaluate_solution(h):
    """Return C5 bound given h"""
    n = len(h)
    corr = circular_correlation(h)
    min_corr = np.min(corr)
    return 1.0 - (2.0 / n) * min_corr

def sdp_solve(n, seed):
    np.random.seed(seed)
    h = Variable(n)
    t = Variable()
    X = Variable((n, n), symmetric=True)
    # Objective
    objective = Maximize(t)
    # Constraints
    constraints = [
        X == h @ h.T,
        [X[i, i] == h[i] for i in range(n)],
        [h[i] >= 0 for i in range(n)],
        [h[i] <= 1 for i in range(n)],
        Sum(h) == n / 2,
        t <= 0.5 * (h @ h),  # For the correlation bound
    ]
    # Formulate the full min-correlation problem
    M = []
    for k in range(n):
        M_k = np.zeros((n,n))
        for i in range(n):
            j = (i + k) % n
            M_k[i,j] = 1
            M_k[j,i] = 1
        M.append(M_k)

    # Introduce a simple violation baseline for illustration
    # Use full matrix constraint and min multiplication in triad fashion:
    for k in range(n):
        constraints.append(trace(M[k] @ X) >= t)

    prob = Problem(objective, constraints)
    prob.solve(solver=ECOS, verbose=False)
    if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
        sol = h.value
        return sol.clip(0, 1)  # Ensure bounds
    else:
        return np.full(n, 0.5)  # fallback

def round_h(X_sol, n, seed):
    np.random.seed(seed)
    U, s, Vt = np.linalg.svd(X_sol)
    V = U @ np.diag(np.sqrt(s))
    g = np.random.randn(V.shape[1])
    scores = V @ g
    # Sort scores and find median
    idx_sorted = np.argsort(-scores)
    n_ones = n // 2
    theta = scores[idx_sorted][n_ones]
    h = (scores >= theta).astype(float)
    # Adjust if needed
    if np.sum(h) > n_ones:
        h[idx_sorted[n_ones]] = 0
    elif np.sum(h) < n_ones:
        h[idx_sorted[np.sum(h)]] = 1
    return h

def swap_hill_climb(h, n, tol=1e-6):
    corr = circular_correlation(h)
    min_corr = np.min(corr)
    corr_improved = True
    while corr_improved:
        corr_improved = False
        for i in range(n):
            for j in range(i+1, n):
                old_corr = np.array(corr)
                # Switch h[i] and h[j]
                # Update all affected correlation values
                h_new = h.copy()
                h_new[i], h_new[j] = h_new[j], h_new[i]
                new_corr = circular_correlation(h_new)
                new_min_corr = np.min(new_corr)
                if new_min_corr > min_corr + tol:
                    min_corr = new_min_corr
                    h = h_new
                    corr = new_corr
                    corr_improved = True
                    break
            if corr_improved:
                break
    return h

def gradient_fine_tune(h, n, max_iter=200):
    # keep grad update in min(auto-correlation) direction
    for i in range(max_iter):
        corr = circular_correlation(h)
        k_star = np.argmin(corr)
        # Compute gradient for min_corr
        grad = np.zeros(n)
        k = k_star
        for i in range(n):
            # Using circular shifts
            j = (i + k) % n
            if h[i] > 0 and h[j] > 0:
                grad[i] += h[j]
                grad[j] += h[i]
        h_new = h + 0.05 * grad
        # Project to [0,1] with fixed sum
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n/2) / np.sum(h_new)
        h = h_new
    return h

def interval_optimize(n_intervals=6, n_points=200, scale=2.0):
    def interval_overlap(a_list, b_list):
        # Compute overlap of interval sets under shift
        def compute_interval_overlap(shift_val):
            total = 0.0
            for a1, b1 in zip(a_list, b_list):
                a2, b2 = a1 + shift_val, b1 + shift_val
                if a2 < 0:
                    a2 += scale
                if b2 > scale:
                    b2 -= scale
                if a2 < b1 and a1 < b2:
                    overlap = min(b1, b2) - max(a1, a2)
                    if overlap > 0:
                        total += overlap
            return total
        return compute_interval_overlap

    def gen_configs(center_val=1.0):
        configs = []
        # generate small variations for the middle pair of intervals,
        # keeping symmetry
        for a1 in np.linspace(0.01, 0.49, 15):
            for a2 in np.linspace(a1 + 0.01, 0.49, 15):
                if a1 + a2 < 0.99:  # Leave room
                    remain = 1.0 - (a1 + a2)
                    # We now fix the symmetry
                    a_list = [a1, a2, 1.0 - remain/2, 1.0 - a2, 1.0 - a1, 0.0]
                    b_list = [a1 + 0.1, a2 + 0.1, 1.0 - remain/2 + 0.1, 
                              1.0 - a2 + 0.1, 1.0 - a1 + 0.1, 1.0]
                    assert len(a_list) == len(b_list)
                    return [(a_list, b_list)]
        return configs

    best_a = best_b = None
    min_val = float('inf')
    for i in range(200):
        try:
            a_list, b_list = gen_configs()
            # Evaluate a single scenario
            f = interval_overlap(a_list, b_list)
            shifts = np.linspace(0, 2, 20)
            min_corr = float('inf')
            for shift in shifts:
                overlap = f(shift)
                if overlap < min_corr:
                    min_corr = overlap
            if min_corr < min_val:
                min_val = min_corr
                best_a, best_b = a_list, b_list
        except:
            continue

    # Convert to step function
    h = np.zeros(n_points)
    dx = scale / n_points
    for a, b in zip(best_a, best_b):
        i_start = int(np.floor(a / dx))
        i_end = int(np.ceil(b / dx))
        h[i_start:i_end] = 1.0
    h = h.clip(0, 1)
    h = h * (n_points/2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 80  # for SDP
    n_points_interval = 200  # for interval

    if budget_s < 100:
        # Fallback to interval only
        h_interval = interval_optimize(n_intervals=6, n_points=n_points_interval)
        c5_interval = evaluate_solution(h_interval)
        return (h_interval, c5_interval, n_points_interval)

    # --- Initialize ---
    # Force initial h to be valid.
    h_init = initial_h_values
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    if np.sum(h_init) != n_points / 2:
        h_init = np.roll(h_init, int((n_points/2) - np.sum(h_init)))

    best_h = h_init
    best_c5 = evaluate_solution(best_h)
    print(f"initial c5 = {best_c5}")

    # --- SDP-based approach ---
    remaining = budget_s - 10
    num_solves = 0
    num_rounds = 0
    while time.time() - start_time < budget_s - 20 and num_solves < 30:
        try:
            X_sol = sdp_solve(n_points, seed + num_solves)
            for i in range(10):
                h_candidate = round_h(X_sol, n_points, seed + num_solves * 10 + i)
                h_candidate = swap_hill_climb(h_candidate, n_points)
                h_candidate = gradient_fine_tune(h_candidate, n_points)
                c5 = evaluate_solution(h_candidate)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_candidate
                    print(f"Found better C5 = {best_c5:.6f}")
                num_rounds += 1
                if time.time() - start_time > budget_s - 20:
                    break
            num_solves += 1
        except Exception as e:
            print(f"Error in SDP/rounding: {e}")
            pass
        if time.time() - start_time > budget_s - 20:
            break

    # --- Interval fallback ---
    try:
        h_interval = interval_optimize(n_intervals=6, n_points=n_points_interval)
        c5_interval = evaluate_solution(h_interval)
        if c5_interval < best_c5:
            best_c5 = c5_interval
            best_h = h_interval
            print(f"Better interval bound: C5 = {best_c5:.6f}")
    except Exception as e:
        print(f"Interval solver failed: {e}")

    if best_h is None:
        # Fallback to initial if nothing worked
        best_h = h_init
        best_c5 = evaluate_solution(best_h)

    return (best_h, best_c5, len(best_h))
