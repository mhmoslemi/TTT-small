```python
import numpy as np
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    start = time.time()
    best_h = None
    best_c5 = float('inf')
    best_n = None
    
    def c5_bound(h):
        n = len(h)
        dx = 2.0 / n
        O = np.correlate(h, 1 - h, mode='full')
        return O.max() * dx
    
    def random_binary(n):
        h = np.random.choice([0, 1], size=n, p=[0.5, 0.5])
        while np.sum(h) != n // 2:
            if np.sum(h) > n // 2:
                idx = np.where(h == 1)[0]
                np.random.shuffle(idx)
                h[idx[0]] = 0
            else:
                idx = np.where(h == 0)[0]
                np.random.shuffle(idx)
                h[idx[0]] = 1
        return h
    
    def thue_morse(n):
        h = np.array([bin(i).count('1') % 2 for i in range(n)])
        # Adjust weight if needed to exactly n/2
        cur = np.sum(h)
        target = n // 2
        if cur > target:
            idx = np.where(h == 1)[0]
            np.random.shuffle(idx)
            h[idx[:cur-target]] = 0
        elif cur < target:
            idx = np.where(h == 0)[0]
            np.random.shuffle(idx)
            h[idx[:target-cur]] = 1
        return h
    
    def legendre_seq(p):
        # For prime p, create sequence using Legendre symbols for quadratic residues
        if p % 2 == 0:
            p = p + 1 if p % 2 == 0 else p
        h = np.zeros(p, dtype=int)
        for i in range(p):
            if (i * i) % p == 0:
                h[i] = 1
            elif (i * i) % p == 1:
                h[i] = 1
            else:
                # For prime order > 2:
                legendre = pow(i, (p - 1) // 2, p)
                h[i] = 1 if legendre == 1 else 0
        # Adjust for weight n/2, or if we are meant to pad it to desired n
        # Here, we'll just take first n entries and adjust sum
        h_final = h[:p]
        cur = np.sum(h_final)
        target = p // 2
        if cur > target:
            idx = np.where(h_final == 1)[0]
            np.random.shuffle(idx)
            h_final[idx[:cur-target]] = 0
        elif cur < target:
            idx = np.where(h_final == 0)[0]
            np.random.shuffle(idx)
            h_final[idx[:target-cur]] = 1
        return h_final
    
    def refine_pattern(initial_h_values, n):
        base = np.array(initial_h_values, dtype=float)
        x_old = np.linspace(0, 2, len(base), endpoint=False)
        x_new = np.linspace(0, 2, n, endpoint=False)
        h_interp = np.interp(x_new, x_old, base)
        h_bin = (h_interp > 0.5).astype(int)
        cur = h_bin.sum()
        target = n // 2
        if cur > target:
            idx = np.where(h_bin == 1)[0]
            np.random.shuffle(idx)
            h_bin[idx[:cur-target]] = 0
        elif cur < target:
            idx = np.where(h_bin == 0)[0]
            np.random.shuffle(idx)
            h_bin[idx[:target-cur]] = 1
        return h_bin
    
    lengths = [64, 96, 128, 160, 192, 224, 256, 320, 384]
    lengths = [n for n in lengths if n <= 800]

    candidates = []
    
    # Generate candidates
    for n in lengths:
        # Thue-Morse if n is power of 2
        if (n & (n - 1)) == 0:
            seq = thue_morse(n)
            candidates.append((seq, n))
        
        # Legendre for prime n (if prime)
        if n > 2 and all(n % i != 0 for i in range(2, int(n**0.5)+1)):
            # This is not a perfect check, but sufficient for demo purposes
            seq = legendre_seq(n)
            # ensure weight half
            cur = np.sum(seq)
            target = n // 2
            if cur > target:
                idx = np.where(seq == 1)[0]
                np.random.shuffle(idx)
                seq[idx[:cur-target]] = 0
            elif cur < target:
                idx = np.where(seq == 0)[0]
                np.random.shuffle(idx)
                seq[idx[:target-cur]] = 1
            candidates.append((seq, n))

        # random sequences
        for _ in range(2):
            seq = random_binary(n)
            candidates.append((seq, n))

        # replicate initial_h_values
        if 'initial_h_values' in globals():
            try:
                seq = refine_pattern(initial_h_values, n)
                candidates.append((seq, n))
            except:
                pass

    # Evaluate candidates
    for seq, n in candidates:
        if time.time() - start > 0.9 * budget_s:
            break
        try:
            c5 = c5_bound(seq)
            if c5 < best_c5:
                best_c5 = c5
                best_h = seq.copy()
                best_n = n
        except:
            pass

    # local search on top candidates
    top_candidates = []
    top_bounds = []
    for seq, n in candidates[:5]:
        if time.time() - start > 0.9 * budget_s:
            break
        try:
            c5 = c5_bound(seq)
            top_candidates.append((seq, n))
            top_bounds.append(c5)
        except:
            pass

    # Sort by bound
    sorted_pairs = sorted(zip(top_bounds, top_candidates))
    sorted_pairs = sorted_pairs[:3]  # top 3

    for bound, (seq, n) in sorted_pairs:
        if time.time() - start > 0.9 * budget_s:
            break

        # Convert to list for faster indexing
        ones = list(np.where(seq == 1)[0])
        zeros = list(np.where(seq == 0)[0])

        current_c5 = c5_bound(seq)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = seq.copy()
            best_n = n

        # Local SA search
        time_left = budget_s * 0.9 - (time.time() - start)
        if time_left < 5:
            break

        temp_start = 0.05
        temp_end = 1e-4
        max_iter = int(3e4 * time_left / budget_s)
        temp = temp_start

        for iteration in range(max_iter):
            if time.time() - start > 0.9 * budget_s:
                break

            # Reset mse
            temp = temp_start * (temp_end / temp_start) ** (iteration / max_iter)

            if len(ones) == 0 or len(zeros) == 0:
                continue

            # Select indices to swap
            i = ones[random.randint(0, len(ones) - 1)]
            j = zeros[random.randint(0, len(zeros) - 1)]

            # Swap bits and evaluate
            seq[i], seq[j] = 0, 1
            ones.remove(i)
            zeros.remove(j)
            ones.append(j)
            zeros.append(i)

            new_c5 = c5_bound(seq)
            delta = new_c5 - current_c5

            if delta <= 0 or random.random() < np.exp(-delta / temp):
                target_c5 = new_c5
                current_c5 = target_c5
                if target_c5 < best_c5:
                    best_c5 = target_c5
                    best_h = seq.copy()
                    best_n = n
            else:
                # Revert swap
                seq[i], seq[j] = 1, 0
                ones.remove(j)
                zeros.remove(i)
                ones.append(i)
                zeros.append(j)

    if best_h is None:
        # Fallback to initial if nothing found
        if 'initial_h_values' in globals():
            try:
                best_h = refine_pattern(initial_h_values, 256)
                best_c5 = c5_bound(best_h)
                best_n = 256
            except:
                pass
        else:
            # fallback to random binary 256
            best_h = random_binary(256)
            best_c5 = c5_bound(best_h)
            best_n = 256

    return (best_h.tolist(), best_c5, best_n)
```
