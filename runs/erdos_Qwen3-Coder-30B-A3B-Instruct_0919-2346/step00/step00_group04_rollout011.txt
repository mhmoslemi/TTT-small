def evaluate_h(h):
    """Evaluate the C5 constant for given step function h"""
    # Compute correlation using np.correlate with mode='full'
    # This calculates integral h(x)(1-h(x+k)) for all shifts k
    # The correlation gives us sum_i h[i] * (1 - h[i+k]) for each shift k
    # We want to maximize this sum across all shifts to minimize the overlap
    corr = np.correlate(h, 1 - h, mode='full')
    # The max over all shifts gives us the worst-case overlap
    # We exclude the zero shift (which would be 0 due to h[i](1-h[i]) = 0)
    max_corr = np.max(corr[1:])  # Exclude zero shift
    # Normalize by dx = 2 / len(h)
    dx = 2.0 / len(h)
    c5_val = max_corr * dx
    return c5_val
