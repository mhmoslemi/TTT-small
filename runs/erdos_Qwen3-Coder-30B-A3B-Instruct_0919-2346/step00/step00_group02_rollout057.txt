import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
import time
import math

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C_5 for a given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all possible shifts
    correlation = np.correlate(h, 1 - h, mode="full")
    # Only consider the relevant part of the correlation
    # The maximum overlap occurs when we shift by k from 0 to 2
    max_overlap = max(correlation)
    return max_overlap * dx

def initial_h_values():
    """Return initial construction"""
    # Create a smooth approximation to the optimal solution
    n_points = 45
    x = np.linspace(0, 2, n_points)
    # Start with a piecewise constant function that's 1 in the first half and 0 in the second half
    h = np.zeros(n_points)
    # Correct first 22 points to be 1, last 23 points to be 0
    # Then normalize to sum to n_points/2
    h[:22] = 1.0
    h[22:] = 0.0
    normalization = np.sum(h) / (n_points / 2)
    h = h / normalization
    return h

def run(seed=42, budget_s=600):
    """Optimize the Erdős constant using a combination of strategies"""
    np.random.seed(seed)
    
    start_time = time.time()
    n_points = 45  # Use a moderate number of points for speed
    dx = 2.0 / n_points
    
    # Initial solution
    h = initial_h_values()
    current_h = h.copy()
    current_c5 = evaluate_erdos_solution(current_h)
    
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Main optimization loop
    while time.time() - start_time < budget_s * 0.95:  # Leave some time for final evaluation
        # Try local perturbations
        h_trial = current_h.copy()
        
        # Random perturbation
        for i in range(n_points):
            if np.random.random() < 0.5:
                # Small change
                delta = np.random.normal(0, 0.05)
                h_trial[i] += delta
                # Clamp to [0, 1]
                h_trial[i] = max(0, min(1, h_trial[i]))
        
        # Project onto feasible set (sum = 1)
        actual_sum = np.sum(h_trial) * dx
        if actual_sum > 0:
            h_trial = h_trial * (n_points / 2) / actual_sum
        
        # Check the new solution
        trial_c5 = evaluate_erdos_solution(h_trial)
        
        # Accept if better
        if trial_c5 < current_c5:
            current_h = h_trial
            current_c5 = trial_c5
            
            if trial_c5 < best_c5:
                best_h = h_trial.copy()
                best_c5 = trial_c5
    
    # Refine with a simple gradient-like method
    # Zoom in on the best solution with slight adjustments
    for i in range(200):  # Additional local search
        if time.time() - start_time > budget_s:
            break
        h_trial = best_h.copy()
        
        # Perturb one point at a time
        idx = np.random.randint(0, n_points)
        h_trial[idx] += np.random.normal(0, 0.01)
        h_trial[idx] = max(0, min(1, h_trial[idx]))
        
        # Rescale to meet constraint
        actual_sum = np.sum(h_trial) * dx
        if actual_sum > 0:
            h_trial = h_trial * (n_points / 2) / actual_sum
        
        trial_c5 = evaluate_erdos_solution(h_trial)
        
        if trial_c5 < best_c5:
            best_h = h_trial.copy()
            best_c5 = trial_c5
    
    return (best_h, best_c5, n_points)
