def project_to_simplex(h, target_sum):
    """Project h onto the capped simplex: 0 <= h <= 1 and sum(h) = target_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Water-filling approach
    sorted_h = np.sort(h)[::-1]
    cumsum = np.cumsum(sorted_h)
    n = len(h)
    # Find threshold
    for i in range(n):
        if cumsum[i] <= target_sum:
            continue
        else:
            threshold = (target_sum - cumsum[i-1]) / (n - i) if i > 0 else target_sum / n
            break
    else:
        threshold = (target_sum - cumsum[n-1]) / (n - n) if n > 0 else 0
        
    # Apply threshold
    h = np.clip(h - threshold, 0, 1)
    return h

def compute_correlation(h, dx):
    """Compute the correlation and return max value."""
    corr = np.correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def compute_grad_vectorized(h, alpha, dx):

    # unroll the engine:
    # O_ell for ell from -n+1 to n-1, stored in list O_vals of length 2n-1
    # BUT here we only iterate over original size 2n-1, shifted by n-1

    # Build list of these overlaps with shifting  
    # O[l] = sum_{i} h[i] (1 - h[i + l])

    full_shifts = 2 * (n - 1) + 1  # -n+1 to n-1 which is 2(n-1) + 1 numbers
    O_vals = np.zeros(full_shifts)
    
    # Do convolution for the finite sample
    h_extended = np.concatenate([np.zeros(n-1), h, np.zeros(n-1)])
    h1_extended = np.concatenate([np.zeros(n-1), 1 - h, np.zeros(n-1)])
    
    corr = np.correlate(h_extended, h1_extended, mode='full') 
        
    # Now extract only the overlapping part: O_vals we want is [shift_from, shift_to] = [n-1, 3(n-1)]
    # But shift from 0 to 2n-2 == shift = [ -(n-1) to (n-1) ] 
    shift_from = 0
    shift_to = 2 * (n - 1) + 1 

    # Slice between -n+1 and n-1: valid for our linear region at the indices [n-1, 3(n-1)-1] 
    # Corr values from i=min shifts to i=max shift in the relaxation
    # corr has size [0..2(n-1)] = shift potentially from 2(n-1) to 0 due to indexing
    # i is not what we want – indexed from -n+1:0:n-1 

    shift_from_c = n - 1
    shift_to_c = n - 1 + full_shifts

    O_as_val = corr[shift_from_c : shift_to_c]  # should be nice;; slice it right

    # Also compute soft weights:
    # The goal is to compute:
    # w_l = exp(alpha * O_l) / Z, where Z = sum_l exp(alpha * O_l)  

    # O_vals are in shift range [-n+1 to n-1]
    # Each l in 0 to 2(n-1) corresponds to a shift index j in [-n+1 to n-1]
    # `O_vals[j + (n-1)]` = O_shift[j]

    O_scaled = alpha * O_as_val
    exp_O = np.exp(O_scaled)
    Z = np.sum(exp_O)
    w = exp_O / Z  # shape (2n - 1) 
    # But here we expect it just like [-n+1 to n-1] indexing
    # So, effectively w is shifted by n-1
    # w[t] is associated with shift [t - (n-1) - (n-1)]?

    # w will be used to get the gradient w ks
    # see reverse order, add dummy terms if needed
    
    # Do appropriate convolutions  
    c1 = convolve(w, 1 - h, mode='full')
    c2 = convolve(w[::-1], h, mode='full') 

    # Now slice to get proper computation
    offset1 = n - 1
    offset2 = n - 1
    # Length = len(w) = full_shifts = 2*(n-1) + 1.
    # Final result extended by (n-1)
    valid_start = n - 1
    valid_end = valid_start + len(h)  # n - 1 + n = 2n - 1
    
    c1_valid = c1[valid_start:valid_end]
    c2_valid = c2[valid_start:valid_end]
    
    return c1_valid - c2_valid
