    """Evaluate the C5 bound for a given step function h"""
    n = len(h)

    correlation = np.correlate(h, 1 - h, mode="full")
    # Only consider the relevant part of the correlation (k from -1 to 1)
    c5 = np.max(correlation) * dx
    return c5
