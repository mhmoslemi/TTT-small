import numpy as np
import time
import random

def compute_c5(h):
    """Compute C5 bound for given step function h."""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return corr.max()

def propose_swap_and_update(h, corr, i, j):
    """Propose a swap between positions i and j, update correlation in-place."""
    # Get old values
    a = h[i]
    b = h[j]
    
    # Compute delta
    delta_i = b - a  # new_i - old_i
    
    # Store original corr values to copy before update
    new_corr = corr.copy()
    
    n = len(h)
    
    # Loop over k not equal to i or j, update contributions from i and j
    for k in range(n):
        if k == i or k == j:
            continue
            
        hk = h[k]
        
        # Contribution from pair (i, k): shift = k - i
        idx1 = k - i + n - 1
        new_corr[idx1] += delta_i * (1 - hk)
        
        # Contribution from pair (k, i): shift = i - k
        idx2 = i - k + n - 1
        new_corr[idx2] += hk * (-delta_i)
        
        # Contribution from pair (j, k): shift = k - j
        idx3 = k - j + n - 1
        new_corr[idx3] += -delta_i * (1 - hk)
        
        # Contribution from pair (k, j): shift = j - k
        idx4 = j - k + n - 1
        new_corr[idx4] += hk * (delta_i)
    
    # Handle cross contributions (i,j) and (j,i)
    # Pair (i,j): contribution to shift = j - i
    idx_ij = j - i + n - 1
    new_corr[idx_ij] += delta_i
    
    # Pair (j,i): contribution to shift = i - j
    idx_ji = i - j + n - 1
    new_corr[idx_ji] -= delta_i
    
    # Compute new max
    new_max = new_corr.max()
    
    return new_corr, new_max

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start = time.time() 
    
    # Try multiple resolutions
    n_candidates = [60, 80, 100, 120, 140, 160, 180, 200]
    
    best_c5 = np.inf
    best_h = None
    best_n = None

    for n in n_candidates:
        if time.time() - start > budget_s - 1:
            break
        if n % 2 != 0:
            continue
            
        # Time allocation for this n
        max_iters = int(30000 / (n / 100))  # scaled with n
        max_iters = max(max_iters, 1000)
        
        # Initialize random binary h
        h = np.zeros(n, dtype=float)
        ones = n // 2
        h[np.random.choice(n, ones, replace=False)] = 1.0
        
        # Compute initial correlation and C5
        corr = np.correlate(h, 1 - h, mode='full')
        cur_c5 = corr.max() * (2.0 / n)
        cur_max = corr.max()
        
        # Simulated annealing parameters
        T = 0.05
        T_decay = 0.9996
        no_improvement_count = 0
        max_no_improvement = 10000
        
        for iteration in range(max_iters):
            if time.time() - start > budget_s - 0.5:
                break
                
            # Pick two random different indices
            i = np.random.randint(n)
            j = np.random.randint(n)
            while h[i] == h[j]:
                i = np.random.randint(n)
                j = np.random.randint(n)
            
            # Compute new correlation and max using delta update
            new_corr, new_max = propose_swap_and_update(h, corr, i, j)
            new_c5 = new_max * (2.0 / n)
            
            # Accept or reject
            d = new_c5 - cur_c5
            if d < 0 or np.random.rand() < np.exp(-d / (T + 1e-8)):
                corr = new_corr
                h[i], h[j] = h[j], h[i]
                cur_c5 = new_c5
                cur_max = new_max
                no_improvement_count = 0
            else:
                no_improvement_count += 1
                
            # Update best solution if improved
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                best_n = n
            
            # Cooling
            T *= T_decay
            
            # Random restart if no improvement for a long time
            if no_improvement_count > max_no_improvement:
                h = np.zeros(n, dtype=float)
                ones = n // 2
                h[np.random.choice(n, ones, replace=False)] = 1.0
                corr = np.correlate(h, 1 - h, mode='full')
                cur_c5 = corr.max() * (2.0 / n)
                cur_max = corr.max()
                no_improvement_count = 0

    # If nothing found, create a default
    if best_h is None:
        n = 120
        h = np.zeros(n, dtype=float)
        ones = n // 2
        h[np.random.choice(n, ones, replace=False)] = 1.0
        best_h = h
        best_n = n
        best_c5 = compute_c5(h)

    return best_h, best_c5, best_n
