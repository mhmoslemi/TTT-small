import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import time
import random
from itertools import combinations

def initial_h_values(n=70):
    # Provided initial construction
    h = np.array([1 if i < n//2 else 0 for i in range(n)])
    return h

def project_to_capped_simplex(v, target_sum):
    """Project vector v onto the capped simplex with sum = target_sum."""
    n = len(v)
    # Ensure values are in [0,1]
    v = np.clip(v, 0, 1)
    total = np.sum(v)
    if abs(total - target_sum) < 1e-9:
        return v
    elif total > target_sum:
        # Need to decrease values
        # Sort descending to find largest values to reduce
        indices = np.argsort(v)[::-1]
        remaining_sum = target_sum
        for i in indices:
            if remaining_sum <= 0:
                v[i] = 0
            else:
                v[i] = min(v[i], remaining_sum)
                remaining_sum -= v[i]
        return v
    else:
        # Need to increase values - but constrained by 1
        indices = np.argsort(v)
        remaining_sum = target_sum
        for i in indices:
            if remaining_sum <= 0:
                break
            delta = min(1 - v[i], remaining_sum)
            v[i] += delta
            remaining_sum -= delta
        return v

def compute_c5(h):
    """Compute C5 using correlation"""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(c)

def optimize_hmilp(h0, n, target_sum, time_limit):
    """Solve MILP for max tau with symmetry breaking."""
    # h0 should be in [0,1] and sum = target_sum
    h = cp.Variable(n, name='h')
    y_vars = {}
    tau = cp.Variable(name='tau')

    constraints = []
    # Box constraints
    constraints += [h[i] >= 0 for i in range(n)]
    constraints += [h[i] <= 1 for i in range(n)]
    # Mass constraint
    constraints += [cp.sum(h) == target_sum]
    # Symmetry breaking
    constraints += [h[0] == 1]
    # McCormick for y[i,k]
    for k in range(1, n):
        if k <= n - 1:
            y_vars[k] = cp.Variable(n - k, name=f'y_{k}')
            for i in range(n - k):
                y = y_vars[k][i]
                constraints += [y >= 0, y <= h[i], y <= h[i + k]]
                constraints += [y >= h[i] + h[i + k] - 1]
            constraints += [tau <= cp.sum(y_vars[k])]

    # Objective
    obj = cp.Maximize(tau)

    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.GLPK, time_limit=time_limit)
    if prob.status != cp.OPTIMAL and prob.status != cp.SOLVED:
        # As fallback, try with less strict bounds for heuristic
        try:
            prob.solve(solver=cp.CBC, time_limit=time_limit)
        except:
            pass
    h_vals = h.value
    if h_vals is not None:
        h_vals = np.array(h_vals)
        tau_opt = tau.value
        if tau_opt is not None and h_vals is not None:
            return h_vals, tau_opt
    return h0, 0.0

def refine_binary_h(h_vals, n, target_sum):
    # Round to binary if not already
    h_binary = np.zeros(n)
    indices = np.argsort(h_vals)[::-1]
    for i in range(int(target_sum)):
        h_binary[indices[i]] = 1
    return h_binary

def perturb_binary(h_binary, n):
    # Modify a few elements (flip about 10%)
    n_flip = max(1, n // 10)
    indices = list(range(n))
    random.shuffle(indices)
    for i in range(n_flip):
        idx = indices[i]
        h_binary[idx] = 1 - h_binary[idx]
    return h_binary

def run(seed=42, budget_s=600, **kwargs):
    n = 70
    target_sum = n / 2
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    solver_time_limit = 0.8 * (budget_s / 10)
    
    np.random.seed(seed)
    initial_h = initial_h_values(n)
    if len(initial_h) != n:
        initial_h = initial_h[:n] if len(initial_h) > n else np.pad(initial_h, (0, n - len(initial_h)))
    h0 = project_to_capped_simplex(initial_h, target_sum)
    
    # Multi-start strategy
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    
    for trial in range(10):  # 10 restarts
        if time.time() - start_time > budget_s * 0.95:
            break
        # Restart with binary or perturbed start
        h_trial = h0.copy()
        if trial > 0:
            # Try a binary version
            h_trial = refine_binary_h(h0, n, target_sum)
            # Perturb it
            h_trial = perturb_binary(h_trial, n)
            # Make sure it sums correctly
            h_trial = project_to_capped_simplex(h_trial, target_sum)
        
        # Solve using MILP (this is where we spend time)
        try:
            h_opt, tau_opt = optimize_hmilp(h_trial.copy(), n, target_sum, solver_time_limit)
            
            if h_opt is not None:
                c5_trial = compute_c5(h_opt)
                if c5_trial < best_c5:
                    best_c5 = c5_trial
                    best_h = h_opt.copy()
                    
        except Exception as e:
            # If MILP fails, go with current approximation
            pass

        # At least once, solve continuous version for warm start
        if trial == 0:
            try:
                h_lp, _ = optimize_hmilp(h0.copy(), n, target_sum, solver_time_limit)
                # Now use that as warm start for binary problem
                h_binary = refine_binary_h(h_lp, n, target_sum)
                c5_trial = compute_c5(h_binary)
                if c5_trial < best_c5:
                    best_c5 = c5_trial
                    best_h = h_binary
            except:
                pass

    # Final polishing with swapping if needed.
    if best_h is not None:
        n_swaps = 0
        while True:
            n_swaps += 1
            if n_swaps > 1000:
                break
            if time.time() - start_time > budget_s * 0.95:
                break
            max_shift = None
            max_val = -1
            c_max = compute_c5(best_h)
            # Check which shift k produces the maximum overlap
            for k in range(1, n//2):
                s = np.sum(best_h[:-k] * (1 - best_h[k:]))
                if s > max_val:
                    max_val = s
                    max_shift = k
            if max_shift is None or max_val < 1e-10:
                break
            max_shift = max_shift
            # Check for improving swaps along k
            changed = False
            for i in range(n - max_shift):
                if best_h[i] == 1 and best_h[i + max_shift] == 0:
                    # Try swap
                    h_trial = best_h.copy()
                    h_trial[i] = 0
                    h_trial[i + max_shift] = 1
                    c_trial = compute_c5(h_trial)
                    if c_trial < c_max - 1e-6:
                        best_h = h_trial
                        changed = True
                        break
            if not changed:
                break

    # Finally return
    if best_h is None:
        best_h = h0
    c5_final = compute_c5(best_h)
    return (best_h, c5_final, n)
