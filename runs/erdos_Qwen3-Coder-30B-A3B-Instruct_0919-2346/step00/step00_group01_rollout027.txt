def circular_correlation(h):
    # zero pad to 2*n and take real part of fft to get circular correlation
    pad_len = len(h)
    h_pad = np.pad(h, (0, pad_len), mode='constant')
    corr = np.real(np.fft.ifft(np.fft.fft(h_pad) * np.conj(np.fft.fft(h_pad)))).real
    return corr[:len(h)]  # return only first n values

def check_solution(h):
    # Check constraints
    assert 0 <= np.min(h) and np.max(h) <= 1, "h must be in [0,1]"
    assert abs(np.sum(h) - len(h)/2) < 1e-10, "sum h must equal n/2"
    
def eval_bound(h):
    # evaluate C5 = max_k int h(x)(1-h(x+k)) dx
    corr = circular_correlation(h)
    # Note that h(x)(1-h(x+k)) integrates to 1 - int h(x)h(x+k), so
    # C5 = max_k (1 - correlation_k) = 1 - min_k correlation_k
    min_correlation = np.min(corr)
    return 1 - min_correlation
    
def sdp_solve(h0, n):
    # SDP solve for relaxation
    h = Variable(n)
    t = Variable()
    
    # build M_k matrices: for each shift k = 0...n-1, M_k is a (0,1) matrix
    # that has 1s at (i, (i+k)%n) and ((i+k)%n, i)
    M = []
    for k in range(n):
        M_k = np.zeros((n, n))
        for i in range(n):
            j = (i + k) % n
            M_k[i, j] = 1
            M_k[j, i] = 1
        M.append(M_k)

    # constraints
    constraints = [
        sum(h) == n / 2,
        h >= 0,
        h <= 1,
        t <= 1
    ]
    
    # add correlation constraints: trace(M_k * X) >= t
    X = h @ h.T
    for k in range(n):
        constraints.append(trace(M[k] @ X) >= t)
    
    prob = Problem(Maximize(t), constraints)
    prob.solve(solver=ECOS, verbose=False)
    t_sdp = t.value
    if t_sdp is not None:
        return t_sdp
    else:
        return float('inf')

def round_h(X, n):
    # round PSD to binary vector h
    try:
        U, s, Vt = np.linalg.svd(X)
        s = np.maximum(s, 0)  # ensure non-negative
        r = np.sum(s > 1e-10)
        V = U[:, :r]
        s_sqrt = np.sqrt(s[:r])
        Y = V * s_sqrt  # shape is n x r
    except:
        # fallback: use eigenvalues
        evals, evecs = np.linalg.eigh(X)
        evals = np.maximum(evals, 1e-12)
        Y = evecs @ np.diag(np.sqrt(evals))
    
    # sample random Gaussian vector g
    g = np.random.randn(Y.shape[1])
    scores = Y @ g
    # find threshold that gives exactly n/2 ones
    sorted_scores = np.sort(scores)[::-1]
    if len(sorted_scores) >= n//2:
        theta = sorted_scores[n//2 - 1]
    else:
        theta = np.min(scores) - 1
    
    h = (scores >= theta).astype(float)
    # adjust to exactly n/2 ones if needed
    ones_count = np.sum(h)
    if ones_count > n//2:
        indices = np.where(h == 1)[0]
        to_zero = np.random.choice(indices, int(ones_count - n//2), replace=False)
        h[to_zero] = 0
    elif ones_count < n//2:
        indices = np.where(h == 0)[0]
        to_one = np.random.choice(indices, int(n//2 - ones_count), replace=False)
        h[to_one] = 1
    
    return h

def swap_hill_climb(h, n):
    # deterministic hill climb on integer swaps
    corr = circular_correlation(h)
    min_correlation = np.min(corr)
    improvement = True
    while improvement:
        improvement = False
        best_i = best_j = -1
        best_corr = min_correlation
        for i in range(n):
            for j in range(n):
                if h[i] == 1 and h[j] == 0:
                    # compute new correlation values if we swap i and j
                    new_corr = corr.copy()
                    # update correlations for shifts that affect these positions
                    for k in range(n):
                        # affect (i, i+k) and (j, j+k) and (i+k, i) and (j+k, j)
                        # compute change to new_corr array
                        # Actually: each shift k contributes h[i]*h[(i+k)%n] + h[j]*h[(j+k)%n] 
                        # When swapping i, j : 
                        # before: h[i]*h[(i+k)%n] + h[j]*h[(j+k)%n]
                        # after: h[j]*h[(i+k)%n] + h[i]*h[(j+k)%n]
                        idx_a = (i + k) % n
                        idx_b = (j + k) % n
                        corr_change = (h[j] - h[i]) * h[idx_a] + (h[i] - h[j]) * h[idx_b]
                        new_corr[k] += corr_change
                    new_min = np.min(new_corr)
                    if new_min > best_corr:
                        best_corr = new_min
                        best_i, best_j = i, j
                        improvement = True
        if improvement:
            h[best_i], h[best_j] = h[best_j], h[best_i]
            corr = new_corr
            min_correlation = best_corr
    return h

def gradient_fine_tune(h, n, max_iter=200):
    # gradient ascent on minimum autocorrelation
    # Compute actual current correlation values
    corr = circular_correlation(h)
    min_k = np.argmin(corr)
    
    for i in range(max_iter):
        # Find the shift k that gives the minimum correlation
        min_k = np.argmin(corr)
        # Compute gradient of min_k correlation w.r.t. h
        grad = np.zeros(n)
        # This is computed by looking at the derivative of h_{i}*h_{i+k} and h_{i}*h_{i-k} 
        # with respect to h_i: becomes h_{i+k} + h_{i-k} (for shift k)        
        grad_i = np.zeros(n)
        # for shift k at which min occurs:
        # gradient is h_{i+k} + h_{i-k} where i is where we are differentiating
        # So we compute the correlation gradient for index min_k:
        for i in range(n):
            idx1 = (i + min_k) % n
            idx2 = (i - min_k) % n
            grad[i] = h[idx1] + h[idx2]
        
        # Update: project to [0,1] and keep sum fixed
        eta = 0.01 / (1 + i // 50)
        h_new = h + eta * grad
        # Project to [0,1]
        h_new = np.clip(h_new, 0, 1)
        # Project back to sum = n/2 area
        scaled_sum = np.sum(h_new)
        if abs(scaled_sum) > 1e-10:
            h_new = h_new * (n / 2) / scaled_sum
        # ensure h_new still in [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        h = h_new
        corr = circular_correlation(h)
        min_correlation = np.min(corr)
        
    return h

def interval_optimize(n=200):
    # Parameterize h as a union of intervals [a_j,b_j]
    # Construct 6 intervals for simplicity. 
    # twice 3 intervals due to symmetry around midpoint x=1
    # Free interval components: just the first 3 of the 6 intervals
        
    def compute_overlap(shift, a, b):
        # Computes the overlap area for shift
        # A shift k means the interval [a_i, b_i] overlaps with [a_j+k, b_j+k] :
        # intersection: max(0, min(b_i, b_j+k) - max(a_i, a_j+k)) 
        overlap = 0
        for i in range(3):
            # For a point shift real k, consider [a_i, b_i] and [a_{3+i}, b_{3+i}] 
            # but for symmetry full spans are [a_i, b_i] and [2-b_{i+3}, 2-a_{i+3}]
            a1, b1 = a[i], b[i]
            a2, b2 = 2 - b[i+3], 2 - a[i+3]
            # compute length of intersection of [a,b1] and [a2+shift, b2+shift]
            overlap += max(0, min(b1, b2 + shift) - max(a1, a2 + shift))
        return overlap
    
    # slight variation to make the whole structure symmetric:
    # we expect high-quality solutions: almost symmetric with some variation
    def compute_intervals(shift, a, b):
        # Returns the length of intersection for shift 
        # involving only working intervals and their shifted forms
        # For 3 intervals [a0,b0], [a1,b1], [a2,b2]
        # and their symmetric counterparts, the full set is:
        # [a0,b0], [a1,b1], [a2,b2], [2-b2, 2-a2], [2-b1, 2-a1], [2-b0, 2-a0] 
        # lengths: b0-a0, b1-a1, b2-a2
        # then for shifts a0 shifts there are
        # overlaps with other intervals determined by shift k
        # formal structure, intake a and b of 6 intervals that define final
        # shifted version: results = [a0,b0], [a1,b1], [a2,b2], [2-b2,2-a2], [2-b1,2-a1], [2-b0, 2-a0]
        # overlap with shifted version, via explicit computation of overlaps

        def compute_single_intersection(a1, b1, a2, b2):
            return max(0, min(b1, b2) - max(a1, a2))

        # Our 6 intervals (symmetrically chosen): 
        centers = np.array([a[0], a[1], a[2]])
        widths = np.array([b[0]-a[0], b[1]-a[1], b[2]-a[2]])
        # reflect for symmetry
        ca = np.array([2 - b[2], 2 - b[1], 2 - b[0]])
        cw = widths
        
        # Length 2eps to allow claiming the full range of [0,1] (for interval who split exactly at that point)
        eps = 1e-10
        
        overlap_total = 0
        # total_len = sum(widths)  # should = 1
        for i in range(3):
            a0, w0 = a[i], widths[i]
            a01 = a0 + shift
            a1, w1 = ca[i], cw[i]
            a11 = a1 + shift
            # overlap with
            # o: (a0, a0+w0) and (a11, a11+w1) 
            o = max(0, min(a0+w0, a11+w1) - max(a0, a11))
            overlap_total += o
            # also overlaps with (a01, a01+w0) and (a1, a1+w1) for x1 = 2-shift
            o2 = max(0, min(a01+w0, a1+w1) - max(a01, a1))
            overlap_total += o2
        return overlap_total
    
    # objective function
    def obj(params):
        # params is [a0, a1, a2, b0, b1, b2]
        a = params[:3]  # [a0, a1, a2]
        b = params[3:]  # [b0, b1, b2]
        
        # Softly enforce total area = 1
        if any(b[i] <= a[i] for i in range(3)):
            return 1e6
        area = sum(b[i] - a[i] for i in range(3))
        penalty_area = 100 * abs(area - 1)

        # compute minimum overlap
        shift_values = np.linspace(0, 1, 40)  # for approximating min over contiguous shifts
        overlaps = [compute_single_overlap(shift, a, b) for shift in shift_values]  # this is incorrect approach,
        # replacing with an indirect approach, using interval logic correctly:
        # revert to more precise method of computing overlaps over shifts
        
        min_overlap = min([compute_intervals(shift, a, b) for shift in np.linspace(0, 1, 40)])
        return -min_overlap + penalty_area  # use negative for minimization artifically
    
    def compute_single_overlap(shift, a, b):
        # returns overlap of intervals [a[i], b[i]] with [a[i]+shift, b[i]+shift] 
        # for a various i ind. BUT rely on the symmetry correctly encoded
        # we assume intervals like:
        # [a0, b0], [a1, b1], [a2, b2], [2-b2, 2-a2], [2-b1, 2-a1], [2-b0, 2-a0]
        # nb: 6 intervals (3 are symmetric to other 3)  
        # Hence we don't need 6, just a0,...,a2, b0,...,b2
    
        def overlap_single_interval(i):
            a1 = a[i]
            b1 = b[i]
            a2 = a1 + shift
            b2 = b1 + shift
            # Generate the 6 intervals as final.
            a_all = [a1, a[1], a[2], 2 - b[2], 2 - b[1], 2 - b[0]]
            b_all = [b1, b[1], b[2], 2 - a[2], 2 - a[1], 2 - a[0]]
            total_len = 0
            for j in range(6):
                inter = max(0, min(b_all[j], b2) - max(a_all[j], a2))
                total_len += inter
            return total_len
        
        # To improve performance: below is more precise:
        # For each interval we compare with shifted (same) interval, e.g.
        # [a0,b0] and [a0+shift, b0+shift]  
        # [a1,b1] and [a1+shift, b1+shift] 
        # [a2,b2] and [a2+shift, b2+shift]
        # Same from paired from symmetry, so it's only computed once
        overlap = 0
        for i in range(3):
            a1, b1 = a[i], b[i]
            a2, b2 = a1 + shift, b1 + shift
            # Batch check all current intervals against their shifted versions
            # First interval in list corresponds to pair [a0,b0], [a0+shift, b0+shift]
            # Final interval includes all in the pair [a, b] and their symmetry pairs
            # But ignore [-1..]
            for i_orig in range(3):
                a_z, b_z = a[i_orig], b[i_orig]
                a_s = a_z + shift    
                b_s = b_z + shift
                mins = min(b_z, b_s)
                maxs = max(a_z, a_s)
                inter = max(0, mins - maxs)
                
                overlap += inter + overlap
        return overlap

    # Optimization using scipy minimize
    # Use heuristics to start around good places
    bounds = [(0, 0.5)] * 3 + [(0.5, 1.0)] * 3  # symmetric around x=1
    # Try to further subdivide the space and sample 4 initial good-starting points 
    init_points = [
        [0.1, 0.3, 0.5, 0.6, 0.8, 0.9],
        [0.05, 0.25, 0.45, 0.55, 0.75, 0.95],
        [0.0, 0.3, 0.6, 0.6, 0.9, 1.0],
        [0.05, 0.25, 0.45, 0.55, 0.75, 0.95],
    ]
    
    min_obj = float("inf")
    best_h = None
    
    for pt in init_points:
        try:
            res = minimize(
                obj, 
                pt,
                method='L-BFGS-B',
                bounds=bounds,
                options={'disp': False, 'maxiter': 100}
            )
            if res.success and res.fun <= min_obj:
                min_obj = res.fun  
                # build segment h
                a = res.x[:3]
                b = res.x[3:]
                h_seg = np.zeros(n)
                # later code will 0,1 - assign entire axis to 1                
                # We generate the h on grid n directly
                dx = 2.0 / n
                for i in range(n):
                    x = i * dx
                    # Check position of x under intervals [a[i], b[i]] for i=0..2
                    # Apply symmetry when pulling from other side
                    overlapped = False
                    for j in range(3):  
                        a0 = a[j]
                        b0 = b[j]
                        if a0 <= x <= b0:
                            h_seg[i] = 1
                            overlapped = True
                            break
                        elif 2 - b0 <= x <= 2 - a0:
                            h_seg[i] = 1
                            overlapped = True
                            break
                    
                if np.sum(h_seg) == n / 2:
                    h_seg_eval = eval_bound(h_seg)
                    if h_seg_eval < min_obj:
                        min_obj = h_seg_eval
                        best_h = h_seg.copy()
        except:
            pass  # ignore errors
    
    # fallback to grid search, perhaps extend it outside of limits
    return best_h, min_obj
