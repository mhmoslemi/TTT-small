import numpy as np
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    
    def evaluate(h):
        n = len(h)
        dx = 2.0 / n
        O = np.correlate(h, 1 - h, mode='full')
        return O.max() * dx
    
    def random_binary(n):
        h = np.random.choice([0, 1], size=n, p=[0.5, 0.5])
        while np.sum(h) != n // 2:
            if np.sum(h) < n // 2:
                idx = np.where(h == 0)[0]
                if len(idx) > 0:
                    i = np.random.choice(idx)
                    h[i] = 1
            else:
                idx = np.where(h == 1)[0]
                if len(idx) > 0:
                    i = np.random.choice(idx)
                    h[i] = 0
        return h.astype(float)
    
    def thue_morse(n):
        h = np.zeros(n, dtype=int)
        for i in range(n):
            h[i] = bin(i).count('1') % 2
        # Ensure weight is half
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) > n // 2:
            # Remove excess ones
            to_remove = len(ones) - n // 2
            idx = np.random.choice(ones, size=to_remove, replace=False)
            h[idx] = 0
        elif len(ones) < n // 2:
            # Add necessary zeros
            to_add = n // 2 - len(ones)
            idx = np.random.choice(zeros, size=to_add, replace=False)
            h[idx] = 1
        return h.astype(float)
    
    def legendre(n):
        if n % 2 == 0:
            m = n
        else:
            m = n - 1
        h = np.zeros(m, dtype=int)
        for i in range(1, m):
            h[i] = (i * i) % m
        h = h > m // 2
        h = h.astype(int)
        # Ensure weight is half
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) > m // 2:
            to_remove = len(ones) - m // 2
            idx = np.random.choice(ones, size=to_remove, replace=False)
            h[idx] = 0
        elif len(ones) < m // 2:
            to_add = m // 2 - len(ones)
            idx = np.random.choice(zeros, size=to_add, replace=False)
            h[idx] = 1
        return h.astype(float)

    lengths = [64, 80, 96, 112, 128, 144, 160, 176, 192, 224, 256, 320, 384, 448, 512]
    lengths = [n for n in lengths if n <= 800]
    
    candidates = []
    
    if 'initial_h_values' in globals():
        base = np.array(initial_h_values, dtype=float)
        for n in lengths:
            if n > 0:
                x_old = np.linspace(0, 2, len(base), endpoint=False)
                x_new = np.linspace(0, 2, n, endpoint=False)
                h_interp = np.interp(x_new, x_old, base)
                h_bin = (h_interp > 0.5).astype(int)
                cur_sum = np.sum(h_bin)
                target = n // 2
                if cur_sum > target:
                    idx = np.where(h_bin == 1)[0]
                    np.random.shuffle(idx)
                    h_bin[idx[:cur_sum-target]] = 0
                elif cur_sum < target:
                    idx = np.where(h_bin == 0)[0]
                    np.random.shuffle(idx)
                    h_bin[idx[:target-cur_sum]] = 1
                candidates.append((h_bin, n))

    for n in lengths:
        # Try Thue-Morse if appropriate
        if (n & (n - 1)) == 0:  # power of 2
            try:
                seq = thue_morse(n)
                candidates.append((seq, n))
            except:
                pass
        # Try Legendre if n is prime
        if n > 1 and all(n % i != 0 for i in range(2, int(n**0.5)+1)):
            try:
                seq = legendre(n)
                candidates.append((seq, n))
            except:
                pass
        # Add random binaries
        for _ in range(2):
            seq = random_binary(n)
            candidates.append((seq, n))
    
    # Evaluate initial candidates
    best_bound = float('inf')
    best_h = None
    best_n = None
    
    for h, n in candidates:
        if time.time() - start_time > budget_s * 0.95:
            break
        try:
            bound = evaluate(h)
            if bound < best_bound:
                best_bound = bound
                best_h = h.copy()
                best_n = n
        except Exception as e:
            pass

    if best_h is None:
        # Fallback to random initialization
        n = 128
        h = random_binary(n)
        best_bound = evaluate(h)
        best_h = h.copy()
        best_n = n

    # Local search phase - use time budget geographically
    remaining_time = budget_s * 0.95 - (time.time() - start_time)
    candidate_count = min(5, len(candidates))
    if remaining_time > 10:
        run_local_search = True
        time_per_candidate = max(0.1, remaining_time / (candidate_count + 1))
        for h, n in candidates:
            if time.time() - start_time > budget_s * 0.95:
                break
            h_local, bound_local = improve(h.copy(), time_per_candidate)
            if bound_local < best_bound:
                best_bound = bound_local
                best_h = h_local.copy()
                best_n = n

    return (best_h.tolist(), best_bound, best_n)

def improve(h, time_budget):
    start = time.time()
    n = len(h)
    S = n // 2
    current_val = evaluate(h)
    best_h_local = h.copy()
    best_val = current_val
    
    # Lists to hold indices for efficient swapping
    ones_idx = list(np.where(h == 1)[0])
    zeros_idx = list(np.where(h == 0)[0])
    
    # Parameters
    max_iter = int(1e7)
    temp_start = 0.1
    temp_end = 1e-5
    
    for it in range(max_iter):
        if time.time() - start > time_budget:
            break
            
        temp = temp_start * (temp_end / temp_start) ** (it / max_iter)
        
        # Choose two indices to swap
        i = np.random.choice(ones_idx)
        j = np.random.choice(zeros_idx)
        
        # Perform swap
        h[i], h[j] = 0, 1
        ones_idx[ones_idx.index(i)] = j
        zeros_idx[zeros_idx.index(j)] = i
        
        new_val = evaluate(h)
        delta = new_val - current_val

        # Simulated Annealing acceptance probability
        if delta <= 0 or np.random.rand() < np.exp(-delta / temp):
            current_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h_local = h.copy()
        else:
            # Revert swap
            h[i], h[j] = 1, 0
            ones_idx[ones_idx.index(j)] = i
            zeros_idx[zeros_idx.index(i)] = j
    
    return best_h_local, best_val
