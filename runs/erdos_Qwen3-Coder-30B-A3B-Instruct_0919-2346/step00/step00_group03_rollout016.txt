import numpy as np
from scipy.optimize import differential_evolution, minimize
import time
from math import floor

def evaluate_erdos_solution(h, dx):
    """Evaluate the C5 bound for a given step function h"""
    n = len(h)
    # Compute the overlap integral for all shifts
    correlation = np.correlate(h, 1 - h, mode='full')
    # The correlation is of length 2*n - 1, we need the middle part
    # Corresponds to shifts from -1 to 1
    # We want maximum of only 0 to 1 shift range (i.e., valid overlaps)
    # So we focus on the middle n-1 values
    middle_idx = len(correlation) // 2
    overlap_values = correlation[middle_idx - (n-1)//2 : middle_idx + (n-1)//2 + 1]
    return np.max(overlap_values) * dx

def initial_h_values():
    # Return a refined initial solution
    n = 70
    h = np.ones(n) * 0.5
    # Piecewise constant approximation of the optimal solution
    # For the Erdos problem, this should be close to a periodic solution
    # but let's start with a symmetric piecewise function
    
    # Distribution of h values across the interval [0,2]
    # Simple step function: high in middle, low at ends
    # *12 points in middle zone, *28 each at ends?
    dx = 2.0 / n
    for i in range(n):
        x = i * dx
        if 0.5 <= x <= 1.5:
            h[i] = 0.75
        else:
            h[i] = 0.25
            
    return h

def objective(h):
    """Objective function to minimize. Returns negative of the C5 bound."""
    dx = 2.0 / len(h)
    max_overlap = evaluate_erdos_solution(h, dx)
    return max_overlap

def constraint_sum(h):
    """Constraint: sum(h) * dx = 1"""
    dx = 2.0 / len(h)
    return np.sum(h) * dx - 1.0

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 70  # Smaller to avoid overfitting to local minima
    dx = 2.0 / n_points
    best_h = None
    best_c5 = float('inf')
    
    # Use initial values
    h0 = initial_h_values()
    assert len(h0) == n_points, "Initial h must have n_points elements"
    h0 = np.copy(h0)
    
    # Get starting solution
    start_time = time.time()
    
    # Optimize
    # Define bounds for h values  
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraints
    # Equality constraint on integral: sum(h)*dx = 1
    def constraint_sum_eq(h):
        return np.sum(h) * dx - 1.0
    
    constraints = {'type': 'eq', 'fun': constraint_sum_eq}
    
    # Use differential evolution for global optimization
    # Set maxiter to keep within the time budget
    maxiter = max(50, int(1000 * budget_s/600))
    
    result = differential_evolution(
        objective,
        bounds,
        constraints=constraints,
        maxiter=maxiter,
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.7,
        seed=seed,
        disp=False,
    )
    
    if not result.success:
        h_opt = h0
    else:
        h_opt = result.x
    
    # Ensure bounds of h are respected
    h_opt = np.clip(h_opt, 0, 1)
    
    # Ensure the integral constraint
    actual_sum = np.sum(h_opt)
    ratio = 1.0 / actual_sum if actual_sum > 0 else 1.0
    h_opt *= ratio
    
    # Evaluate final
    c5_bound = evaluate_erdos_solution(h_opt, dx)
    
    return (h_opt, c5_bound, n_points)
