def evaluate_c5(h):
    """Evaluate the true C5 value using np.correlate."""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_capped_simplex(h, target_sum):
    """Project h onto the capped simplex: 0 ≤ h[i] ≤ 1 and sum(h) = target_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Sort in descending order to do water filling
    sorted_indices = np.argsort(-h)
    remaining_sum = target_sum
    for i in range(len(h)):
        idx = sorted_indices[i]
        if remaining_sum <= h[idx]:
            h[idx] = remaining_sum
            break
        h[idx] = 1.0
        remaining_sum -= 1.0
    return h

def compute_soft_max_weights(h, alpha):
    """Compute soft-max weights of each shift."""
    dx = 2.0 / len(h)
    # Compute all shifts O_ell(h), for ell in [-n+1, n-1]
    n = len(h)
    # Initialize list to hold all O_ell values:
    overlap_ells = []
    # For each shift ell
    for ell in range(-n+1, n):
        # Compute O_ell = dx * sum(h[i] * (1 - h[i+ell]))
        if ell >= 0:
            # h[...]*1 - h[ell:]*h[:n-ell] = sum(h[i] - h[i]*h[i+ell])
            valid_length = n - ell
            if valid_length <= 0:
                overlap_ells.append(0.0)
                continue
            o_ell = dx * np.sum(h[:valid_length] * (1.0 - h[ell:ell+valid_length]))
        else:  # ell < 0
            # h[0:ell]*1 - h[:ell]*h[-ell:] = sum(h[i] - h[i]*h[i+ell])
            valid_length = n + ell
            if valid_length <= 0:
                overlap_ells.append(0.0)
                continue
            o_ell = dx * np.sum(h[-ell:] * (1.0 - h[:valid_length]))
        overlap_ells.append(o_ell)
    overlap_ells = np.array(overlap_ells)
    # Softmax with temperature alpha
    exp_overlap = np.exp(alpha * overlap_ells)
    weights = exp_overlap / np.sum(exp_overlap)
    return weights

def compute_gradient(h, weights):
    """Compute the gradient of the smooth surrogate F_alpha."""
    dx = 2.0 / len(h)
    # Modified manual convolution to compute: sum(w_l * (1 - h[i+l]))
    # Do this as: sum(w_l * (1 - h[i+l])) = sum(w_l) - sum(w_l * h[i+l])
    # First compute: sum(w_l * h[i+l]) using full convolution with shifted h:
    # weight vector is of size 2*(n-1)+1 = 2*n-1
    n = len(h)
    w_full = np.zeros(2*n - 1)
    weight_shift = -(n-1)
    for i, w in enumerate(weights):
        w_full[i + weight_shift] = w
    
    # inverted weight vector for second part
    w_rev = w_full[::-1]
    # h_rev = np.flip(h)
    conv1 = signal.fftconvolve(w_full, 1-h, mode='full')  # This is sum(w_l * (1 - h[i+l]))
    conv2 = signal.fftconvolve(w_rev, h, mode='full')
    grad = conv1 - conv2
    # Slice result back to size n starting from the middle
    mid = len(grad) // 2
    grad = grad[mid:mid+n]
    return grad
