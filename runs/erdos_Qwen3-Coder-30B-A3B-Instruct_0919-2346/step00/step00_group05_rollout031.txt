import numpy as np
import time

def project_to_capped_simplex(v, S):
    """Project vector v onto the capped simplex: sum x = S, 0 ≤ x_i ≤ 1"""
    # Use binary search for Lagrange multiplier lambda
    lo, hi = -2.0, 2.0
    for _ in range(50):
        lam = (lo + hi) / 2
        x = np.clip(v - lam, 0, 1)
        s = x.sum()
        if s > S:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo + hi) / 2, 0, 1)
    return x

def get_max_shifts(h):
    """Get all shifts that achieve maximum correlation."""
    N = len(h)
    c = np.correlate(h, 1 - h, mode='full')
    max_val = c.max()
    indices = np.where(np.abs(c - max_val) < 1e-9)[0]
    shifts = indices - (N - 1)  # shift k where index = N-1+k corresponds to shift k
    return shifts.tolist()

def gradient_max_shifts(h, shifts):
    """Compute averaged subgradient for all maximising shifts."""
    N = len(h)
    g = np.zeros(N)
    for k in shifts:
        # Compute shifted arrays using np.concatenate
        if k >= 0:
            left = np.concatenate((np.zeros(k), h[:N-k]))
            right = np.concatenate((h[k:], np.zeros(k)))
        else:
            k_abs = -k
            left = np.concatenate((h[k_abs:], np.zeros(k_abs)))
            right = np.concatenate((np.zeros(k_abs), h[:N-k_abs]))
        g += 1 - left - right
    return g / len(shifts)

def random_feasible(N):
    """Generate a random feasible vector (sum = N/2, 0 ≤ h ≤ 1)."""
    v = np.random.rand(N)
    S = N / 2
    return project_to_capped_simplex(v, S)

def evaluate(h):
    """Evaluate the maximum overlap C5."""
    N = len(h)
    dx = 2.0 / N
    c = np.correlate(h, 1 - h, mode='full')
    return np.max(c) * dx

        if np.random.rand() < 0.2:

        else:

            h += np.random.uniform(-0.03, 0.03, n)

        # Gradient descent phase
        alpha = 0.12

            cur_c5 = evaluate(h)
            if cur_c5 < best_c5:
                best_c5, best_h = cur_c5, h.copy()

            K = get_max_shifts(h)
            g = gradient_max_shifts(h, K)

            v = h - alpha * g

                alpha *= 0.5
                last_improve = it
            else:
                alpha = min(alpha * 1.02, 0.5)

        cur_c5 = evaluate(h)

            k = np.random.choice(K)
