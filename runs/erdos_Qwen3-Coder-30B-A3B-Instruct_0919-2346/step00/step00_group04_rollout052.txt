import numpy as np
import time
import math
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()
    
    def c5_bound(h):
        n = len(h)
        dx = 2.0 / n
        O = np.correlate(h, 1 - h, mode='full')
        return O.max() * dx
    
    def evaluate(h):
        # Return C5 bound
        return c5_bound(h)
    
    # Candidate lengths (smaller is preferred)
    lengths = [64, 80, 96, 112, 128, 144, 160, 176, 192, 224, 256, 320, 384, 448, 512]
    lengths = [n for n in lengths if n <= 800 and n % 2 == 0]

    best_h = None
    best_c5 = float('inf')
    best_n = None

    # Candidate generation functions
    def random_binary(n):
        h = np.random.choice([0, 1], size=n, p=[0.5, 0.5])
        while np.sum(h) != n // 2:
            if np.sum(h) > n // 2:
                # Find 1s to flip to 0
                ones = np.where(h == 1)[0]
                flip_idx = np.random.choice(ones)
                h[flip_idx] = 0
            else:
                # Find 0s to flip to 1
                zeros = np.where(h == 0)[0]
                flip_idx = np.random.choice(zeros)
                h[flip_idx] = 1
        return h

    def thue_morse(n):
        h = np.zeros(n, dtype=int)
        for i in range(n):
            bits = bin(i)[2:].count('1')  # count 1s in binary
            h[i] = bits % 2
        return h

    def replicate_initial_h_values(n):
        # Use initial_h_values global variable if exists
        if 'initial_h_values' in globals():
            base = np.array(initial_h_values)
            x_old = np.linspace(0, 2, len(base), endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            h_interp = np.interp(x_new, x_old, base)
            h_bin = (h_interp > 0.5).astype(int)
            # adjust weight to exactly n//2
            current_sum = h_bin.sum()
            target_sum = n // 2
            if current_sum > target_sum:
                indices = np.where(h_bin == 1)[0]
                np.random.shuffle(indices)
                for i in range(current_sum - target_sum):
                    h_bin[indices[i]] = 0
            elif current_sum < target_sum:
                indices = np.where(h_bin == 0)[0]
                np.random.shuffle(indices)
                for i in range(target_sum - current_sum):
                    h_bin[indices[i]] = 1
            return h_bin
        else:
            # Fallback to random if no initial provided
            return random_binary(n)

    # Initialize best from initial_h_values if available
    if 'initial_h_values' in globals():
        h0 = replicate_initial_h_values(42)
        c0 = evaluate(h0)
        if c0 < best_c5:
            best_c5 = c0
            best_h = h0.copy()
            best_n = 42

    # Generate initial candidates for each length
    candidates = []
    for n in lengths:
        # Generate diverse candidates but avoid duplicating function calls
        # For larger n, we can use multiple methods
        if (n & (n - 1)) == 0 and n <= 512:  # power of 2
            h = thue_morse(n)
            if h.sum() != n // 2:
                while h.sum() != n//2:
                    if h.sum() > n//2:
                        ones = np.where(h == 1)[0]
                        i = np.random.choice(ones)
                        h[i] = 0
                    else:
                        zeros = np.where(h == 0)[0]
                        i = np.random.choice(zeros)
                        h[i] = 1
            candidates.append((h, n))

        # Replicate initial pattern
        h = replicate_initial_h_values(n)
        candidates.append((h, n))

        # Random binary with correct sum
        for _ in range(2):
            h = random_binary(n)
            candidates.append((h, n))

    # Evaluate all candidates
    for h, n in candidates:
        c = evaluate(h)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()
            best_n = n

    # Local search on top candidates
    top_candidates = []
    for h, n in candidates:
        c = evaluate(h)
        top_candidates.append((c, h, n))

    top_candidates.sort(key=lambda x: x[0])
    top_candidates = top_candidates[:5]  # Top 5

    # Local search on top candidates
    for i, (c, h, n) in enumerate(top_candidates):
        if time.time() - start > budget_s * 0.95:
            break
        time_left = budget_s * 0.8 / len(top_candidates) if i < len(top_candidates) - 1 else budget_s * 0.2
        h_local, c_local = local_search(h, n, time_left)
        if c_local < best_c5:
            best_c5 = c_local
            best_h = h_local.copy()
            best_n = n

    # Return the result
    return (best_h, best_c5, best_n)

def local_search(h, n, time_budget):
    def evaluate(h):
        dx = 2.0 / len(h)
        O = np.correlate(h, 1 - h, mode='full')
        return O.max() * dx

    start = time.time()
    current_val = evaluate(h)
    best_h = h.copy()
    best_val = current_val
    current_h = h.copy()

    # Maintain indices of 1s and 0s for efficient swaps
    ones = list(np.where(current_h == 1)[0])
    zeros = list(np.where(current_h == 0)[0])

    # Temperature schedule
    temp_start = 0.05
    temp_end = 1e-5
    temp = temp_start
    max_iter = int(1e6)
    
    for it in range(max_iter):
        if time.time() - start > time_budget:
            break

        # Annealing schedule
        temp = temp_start * (temp_end / temp_start) ** (it / max_iter)

        # Choose two random indices: one 1 and one 0
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Swap them
        current_h[i] = 0
        current_h[j] = 1
        
        # Update lists
        ones.remove(i)
        zeros.remove(j)
        ones.append(j)
        zeros.append(i)
        
        # Evaluate new value
        new_val = evaluate(current_h)
        
        # Accept criteria
        delta = new_val - current_val
        if delta <= 0 or np.random.rand() < math.exp(-delta / temp):
            current_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = current_h.copy()
        else:
            # Revert
            current_h[i] = 1
            current_h[j] = 0
            ones.remove(j)
            zeros.remove(i)
            ones.append(i)
            zeros.append(j)
    return best_h, best_val
