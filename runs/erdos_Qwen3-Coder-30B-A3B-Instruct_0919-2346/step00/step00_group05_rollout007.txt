import numpy as np
import cvxpy as cp
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h, n_points):
    # Computed the same way as in the problem statement
    dx = 2.0 / n_points
    c = np.correlate(h, 1 - h, mode="full")
    return np.max(c) * dx

def solve_milp(n_points, h0=None, time_limit=100):
    # Build the model (possibly with warm start)
    h = cp.Variable(n_points, name="h")
    tau = cp.Variable(name="tau")
    # Product variables y[i, k] = h[i] * h[i+k]
    # only compute for k=1,...,n_points-1
    y_vars = {}
    # Constraints for each k shift
    constraints = [cp.sum(h) == n_points / 2]
    for i in range(n_points):
        constraints.append(0 <= h[i])
        constraints.append(h[i] <= 1)
    # Enforce symmetry breaking h[0]=1
    constraints.append(h[0] == 1)

    # Generate product variables and constraints
    for k in range(1, n_points):
        y_vars[k] = cp.Variable(n_points - k, name=f"y_{k}")
        for j in range(n_points - k):
            # These are the McCormick envelopes for h[j]*h[j+k]
            constraints.append(y_vars[k][j] >= h[j] + h[j + k] - 1)
            constraints.append(y_vars[k][j] <= h[j])
            constraints.append(y_vars[k][j] <= h[j + k])
            constraints.append(y_vars[k][j] >= 0)

    # tau <= sum of all y for each shift k
    for k in range(1, n_points):
        constraints.append(tau <= cp.sum(y_vars[k]))

    # Objective: maximize tau
    objective = cp.Maximize(tau)
    prob = cp.Problem(objective, constraints)

    # Warm-start if h0 given
    if h0 is not None:
        h.value = h0
        # Potentially also initialize y to some value?
        # Not directly supported, but could be updated as needed
        # here leave it uninitialized

    try:
        # Run with time limit
        prob.solve(solver=cp.CBC, time_limit=time_limit, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_val = h.value
            tau_val = tau.value
        else:
            # Fallback to a suboptimal but feasible solution
            h_val = h.value
            tau_val = tau.value
    except Exception as e:
        # If no solution found in reasonable time, return fallback
        h_val = None
        tau_val = -np.inf

    return h_val, tau_val

def project_to_simplex(x, s):
    """
    Project x onto the capped simplex: sum(x) = s, 0 <= x <= 1.
    This is an efficient version for keeping the sum constraint and bounds.
    """
    x = np.array(x)
    s = float(s)
    n = len(x)
    x_proj = np.clip(x, 0, 1)
    sum_x = np.sum(x_proj)
    if sum_x > s:
        # Reduce values
        x_proj -= (sum_x - s) / n
        x_proj = np.clip(x_proj, 0, 1)
    elif sum_x < s:
        # Increase values
        x_proj += (s - sum_x) / n
        x_proj = np.clip(x_proj, 0, 1)
    return x_proj

def round_to_binary(x, s):
    # Bring approximately s ones (selection of top values)
    idx = np.argsort(x)[::-1]
    x_rounded = np.zeros(len(x))
    x_rounded[idx[:int(s)]] = 1
    return x_rounded
