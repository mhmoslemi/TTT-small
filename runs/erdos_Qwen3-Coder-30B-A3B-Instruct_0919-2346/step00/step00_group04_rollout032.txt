import numpy as np
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    
    def c5_bound(h):
        n = len(h)
        dx = 2.0 / n
        O = np.correlate(h, 1 - h, mode='full')
        return O.max() * dx
    
    def random_binary(n):
        h = np.zeros(n, dtype=int)
        ones = np.random.choice(n, n//2, replace=False)
        h[ones] = 1
        return h
    
    # Generate candidates
    candidate_lengths = [64, 96, 128, 160, 192, 224, 256, 320, 384, 448, 512]
    candidate_lengths = [n for n in candidate_lengths if n <= 800]
    
    candidates = []
    
    # Use initial_h_values from global if available
    if 'initial_h_values' in globals():
        base = np.array(initial_h_values, dtype=float)
        for n in candidate_lengths:
            x_old = np.linspace(0, 2, len(base), endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            h_interp = np.interp(x_new, x_old, base)
            h_bin = (h_interp > 0.5).astype(int)
            cur = h_bin.sum()
            target = n // 2
            if cur > target:
                idx = np.where(h_bin == 1)[0]
                np.random.shuffle(idx)
                h_bin[idx[:cur-target]] = 0
            elif cur < target:
                idx = np.where(h_bin == 0)[0]
                np.random.shuffle(idx)
                h_bin[idx[:target-cur]] = 1
            candidates.append((h_bin, n))
    
    # Generate random binaries
    for n in candidate_lengths:
        for _ in range(3):
            h = random_binary(n)
            candidates.append((h, n))
    
    # Evaluate candidates
    best_candidate = None
    best_bound = float('inf')
    best_n = None
    
    for h, n in candidates:
        if time.time() - start_time > budget_s * 0.95:
            break
        bound = c5_bound(h)
        if bound < best_bound:
            best_bound = bound
            best_candidate = h.copy()
            best_n = n
    
    # Local search on best candidates
    candidate_count = min(3, len(candidates))
    if candidate_count == 0:
        # If no candidates, start with a random one
        best_candidate = random_binary(256)
        best_n = 256
        best_bound = c5_bound(best_candidate)
    
    def local_search(h, n, time_limit):
        current_bound = c5_bound(h)
        best_h = h.copy()
        best_bound = current_bound
        
        start_time = time.time()
        max_iter = 1000000
        
        # Precompute lists of indices
        ones = list(np.flatnonzero(h))
        zeros = list(np.flatnonzero(1 - h))
        
        temp_start = 0.1
        temp_end = 1e-6
        
        for it in range(max_iter):
            if time.time() - start_time > time_limit:
                break
                
            # Temperature schedule
            temp = temp_start * (temp_end / temp_start) ** (it / max_iter)
            
            # Random swap
            oi = random.randint(0, len(ones) - 1)
            oj = random.randint(0, len(zeros) - 1)
            i = ones[oi]
            j = zeros[oj]
            
            # Perform swap
            h[i] = 0
            h[j] = 1
            ones[oi] = j
            zeros[oj] = i
            
            new_bound = c5_bound(h)
            
            # Accept criteria
            delta = new_bound - current_bound
            if delta <= 0 or random.random() < np.exp(-delta / temp):
                current_bound = new_bound
                if new_bound < best_bound:
                    best_bound = new_bound
                    best_h = h.copy()
            else:
                # Revert swap
                h[i] = 1
                h[j] = 0
                ones[oi] = i
                zeros[oj] = j
                
        return best_h, best_bound
    
    # Run multi-start local search
    time_left = budget_s * 0.8
    remaining_time = time_left
    
    for i in range(3):
        if remaining_time <= 0:
            break
        candidate = random_binary(256)
        improved_h, improved_bound = local_search(candidate, 256, remaining_time / 3)
        if improved_bound < best_bound:
            best_bound = improved_bound
            best_candidate = improved_h.copy()
            best_n = 256
        remaining_time = max(0, budget_s * 0.95 - (time.time() - start_time))
    
    return (best_candidate.tolist(), best_bound, best_n)
