import numpy as np
import cvxpy as cp
import time
from scipy import signal
from scipy.fft import fft, ifft

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n = 60
    dx = 2.0 / n
    max_seconds_per_solve = min(budget_s / 2, 30)
    
    # Initialize with initial construction if available
    if 'initial_h_values' in globals():
        h_best = np.array(initial_h_values)
        h_best = h_best[:n]
        if len(h_best) < n:
            h_best = np.pad(h_best, (0, n - len(h_best)), mode='constant')
    else:
        # Default binary construction
        h_best = np.zeros(n)
        h_best[:n//2] = 1.0
        np.random.shuffle(h_best)
    
    # Ensure sum constraint
    h_best = h_best * (n / 2) / np.sum(h_best)
    
    def compute_autocorrelation(h):
        # Compute full circular autocorrelation using FFT
        H = fft(h)
        autocorr = ifft(H * np.conj(H)).real
        return autocorr[:n]  # First n elements (real part)
    
    # Integer MILP-based optimization approach
    def solve_milp_with_shifts(shifts):
        # Variables
        h_vars = cp.Variable(n, boolean=True)
        t = cp.Variable()
        # Constraint h_0 = 1 to break symmetry
        constraints = [h_vars[0] == 1]
        # Sum constraint
        constraints.append(cp.sum(h_vars) == n // 2)
        # Variables for product w_{i,k}
        w_vars = {}
        a_vars = {}
        for k in shifts:
            w_vars[k] = cp.Variable(n, boolean=True)
            a_vars[k] = cp.sum(w_vars[k])
            # McCormick envelopes
            for i in range(n):
                ip = (i + k) % n
                constraints.extend([
                    w_vars[k][i] <= h_vars[i],
                    w_vars[k][i] <= h_vars[ip],
                    w_vars[k][i] >= h_vars[i] + h_vars[ip] - 1
                ])
        
        # For each shift k add t <= a_k
        for k in shifts:
            constraints.append(t <= a_vars[k])
        
        # Objective: maximize t
        objective = cp.Maximize(t)
        prob = cp.Problem(objective, constraints)
        
        # Set solver options
        prob.solve(solver=cp.CBC, max_seconds=max_seconds_per_solve, verbose=False)
        
        success = prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]
        h_sol = None
        if success:
            h_sol = h_vars.value.astype(float)
            t_val = t.value
        else:
            print("Failure to solve MILP")
        return h_sol, t_val if success else 0
    
    # Cut generation loop
    shifts = [1, n//2, 2, n-2]
    timeout_check_interval = 10
    iteration_count = 0
    
    while True:
        if iteration_count % timeout_check_interval == 0:
            elapsed = time.time() - start_time
            remaining = budget_s - elapsed
            if remaining < 30:
                break
        
        h_sol, opt_t = solve_milp_with_shifts(shifts)
        if h_sol is None:
            break
            
        # Check if exact optimization done
        corr_vals = compute_autocorrelation(h_sol)
        a_min = np.min(corr_vals)
        
        if a_min >= opt_t:
            # Found true minimum
            h_best = h_sol.copy()
            break
        else:
            # Add the most violating shift
            worst_k = np.argmin(corr_vals)
            if worst_k not in shifts:
                shifts.append(worst_k)
        
        iteration_count += 1
    
    # Final exact evaluation with all shifts
    corr_all = compute_autocorrelation(h_best)
    t_star = np.min(corr_all)
    C5_bound = 1 - (2.0 / n) * t_star
    
    h_final = h_best.copy()
    for i in range(100):
        if time.time() - start_time > budget_s - 10:
            break
        # Projected gradient ascent
        corr_vals = compute_autocorrelation(h_final)
        worst_k = np.argmin(corr_vals)
        grad = np.zeros_like(h_final)
        # Sub-gradient
        for i in range(n):
            grad[i] = h_final[(i + worst_k) % n] + h_final[(i - worst_k) % n]
        # Gradient step
        step_size = 0.001
        h_temp = h_final + step_size * grad
        # Clip and rescale
        h_temp = np.clip(h_temp, 0, 1)
        current_sum = np.sum(h_temp)
        diff = (n / 2) - current_sum
        # Apply pairwise swap to adjust sum
        if abs(diff) > 1e-6:
            # Simple heuristic: move a small amount from higher to lower value that's not at boundary
            idxs = np.argsort(h_temp)
            idxs_high = [i for i in reversed(idxs) if h_temp[i] > 0.01]
            idxs_low = [i for i in idxs if h_temp[i] < 0.99]
            if idxs_high and idxs_low:
                idx1, idx2 = idxs_high[0], idxs_low[0]
                delta = min(abs(diff), 0.01)
                if diff > 0:
                    h_temp[idx1] -= delta
                    h_temp[idx2] += delta
                else:
                    h_temp[idx1] += delta
                    h_temp[idx2] -= delta
        
        h_final = h_temp.copy()
        # Re-evaluate
        corr = compute_autocorrelation(h_final)
        t_star_update = np.min(corr)
        C5_update = 1 - (2.0 / n) * t_star_update
        if C5_update < C5_bound:
            C5_bound = C5_update
            
    # Final check if we have sufficient improvement
    # Use continuous method otherwise if needed
    if C5_bound > 0.3807:
        # Spectral heuristics approach with small number of Fourier modes
        n_modes = 3
        def objective_fft(alpha):
            h = 0.5 + np.dot(np.cos(2 * np.pi * np.arange(n).reshape((-1, 1)) * np.arange(1, n_modes+1) / n), alpha)
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * (n / 2)
            corr_fft = compute_autocorrelation(h)
            min_corr = np.min(corr_fft)
            return min_corr
        
        def random_start():
            alpha = np.random.uniform(-0.25, 0.25, n_modes)
            return alpha
        
        def clip_alpha(alpha, max_size=0.5):
            # Use max constraint to ensure bounds
            if np.sum(np.abs(alpha)) > max_size:
                alpha = alpha * max_size / np.sum(np.abs(alpha))
            return alpha
            
        best_alpha = random_start()
        best_min_corr = 0
        best_h = None
        
        for _ in range(1000):
            if time.time() - start_time > budget_s - 20:
                break
            # Optimization using direct gradient descent with stochastic update
            alpha = best_alpha + np.random.normal(0, 0.01, n_modes)
            alpha = clip_alpha(alpha)
            min_corr = objective_fft(alpha)
            if min_corr > best_min_corr:
                best_alpha = alpha.copy()
                best_min_corr = min_corr
                # compute final h
                h_temp = 0.5 + np.dot(np.cos(2 * np.pi * np.arange(n).reshape((-1, 1)) * np.arange(1, n_modes+1) / n), alpha)
                h_temp = np.clip(h_temp, 0, 1)
                h_temp = h_temp / np.sum(h_temp) * (n / 2)
                best_h = h_temp.copy()
        
        if best_h is not None:
            corr_vals = compute_autocorrelation(best_h)
            t_opt = np.min(corr_vals)
            C5_bound = 1 - (2.0 / n) * t_opt
            h_final = best_h
    
    # Return final result
    return (h_final, C5_bound, n)
