import numpy as np
import cvxpy as cp
import time
from scipy import fft
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n = 60
    dx = 2.0 / n
    half_n = n // 2
    
    # Initial shift set
    K0 = [1, half_n]
    if n > 4:
        K0.append(2)
        K0.append(n-2)
    
    # Find best solution so far via binary optimization
    start_time = time.time()
    best_c5 = 1.0
    best_h = None
    
    # Initialize with a random binary vector
    h_bin = np.zeros(n, dtype=int)
    ones_indices = np.random.choice(n, size=n//2, replace=False)
    h_bin[ones_indices] = 1
    
    # Binary program solve helper
    def solve_milp_with_shifts(shifts):
        # Variables
        h = cp.Variable(n, integer=True)
        t = cp.Variable(integer=True)
        
        # Constraints
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) == n // 2
        ]
        
        # Add symmetry breaking
        constraints.append(h[0] == 1)
        
        # Add auxiliary variables and constraints
        w_vars = {}
        a_vars = {}
        for k in shifts:
            w_vars[k] = cp.Variable(n, integer=True)
            a_vars[k] = cp.sum(w_vars[k])
            
            for i in range(n):
                w = w_vars[k][i]
                constraints.extend([
                    w <= h[i],
                    w <= h[(i+k) % n],
                    w >= h[i] + h[(i+k) % n] - 1,
                    w >= 0,
                    w <= 1
                ])
            
            # t <= a_k
            constraints.append(t <= a_vars[k])
        
        # Objective
        objective = cp.Maximize(t)
        prob = cp.Problem(objective, constraints)
        
        # Solve
        prob.solve(solver=cp.CBC, max_seconds=budget_s/2)
        
        return h.value, prob.status == cp.OPTIMAL
    
    # Cut generation loop
    current_shifts = K0.copy()
    last_min = -1
    while True:
        if time.time() - start_time > budget_s - 30:
            break
            
        h_sol, is_optimal = solve_milp_with_shifts(current_shifts)
        
        if not is_optimal or h_sol is None:
            break
        
        h_bin = np.round(h_sol).astype(int)
        a_full = np.correlate(h_bin, h_bin, mode='full')
        a_circ = a_full[n-1:]
        a_min = np.min(a_circ)
        
        if a_min > last_min:
            last_min = a_min
        else:
            break
        
        # Compute the exact autocorrelations for all shifts
        full_auto = np.correlate(h_bin, h_bin, mode='full')
        all_corr = full_auto[n-1:]
        worst_k = int(np.argmin(all_corr))
        
        # Add new shift if not in set already
        if worst_k not in current_shifts:
            current_shifts.append(worst_k)
        else:
            break
    
    # Compute final binary solution
    if current_shifts == K0 or last_min == -1:
        # Fallback to random
        h_bin = np.zeros(n)
        ones_indices = np.random.choice(n, size=n//2, replace=False)
        h_bin[ones_indices] = 1
    
    # Perform as many optimizations as possible within time
    full_auto = np.correlate(h_bin, h_bin, mode='full')
    all_corr = full_auto[n-1:]
    t_star = np.min(all_corr)
    c5 = 1.0 - (2.0 / n) * t_star
    
    # Gradient polishing step
    if c5 > 0.3807 and time.time() - start_time < budget_s - 150:
        h_cont = h_bin.astype(float)
        # Gradient ascent with small steps
        max_iter = 1000
        step_size = 0.001
        last_best = c5
        
        for _ in range(max_iter):
            if time.time() - start_time > budget_s - 150:
                break
                
            # Compute current autocorrelations, get worst shift
            a_full = np.correlate(h_cont, h_cont, mode='full')
            a_circ = a_full[n-1:]
            worst_k = int(np.argmin(a_circ))
            t_current = np.min(a_circ)
            
            if t_current >= last_best:
                break
                
            # Compute gradient for worst shift
            grad = np.zeros(n)
            for i in range(n):
                grad[i] = h_cont[(i + worst_k) % n] + h_cont[(i - worst_k) % n]
            
            # Update
            h_cont += step_size * grad
            
            # Project back to [0,1]
            h_cont = np.clip(h_cont, 0.0, 1.0)
            
            # Correct sum (with pairwise swap)
            current_sum = np.sum(h_cont)
            diff = current_sum - n / 2
            
            if abs(diff) > 1e-6:
                # Find indices to swap amounts
                i_less = np.argmax(h_cont)
                i_more = np.argmin(h_cont)
                
                if h_cont[i_less] > 0 and h_cont[i_more] < 1:
                    swap_amount = min(h_cont[i_less], 1 - h_cont[i_more])
                    h_cont[i_less] -= swap_amount
                    h_cont[i_more] += swap_amount
            
            c5_new = 1.0 - (2.0 / n) * np.min(np.correlate(h_cont, h_cont, mode='full')[n-1:])
            if c5_new < c5:
                c5 = c5_new
                h_bin_opt = h_cont.copy()
            else:
                break
    
    return (np.array(h_bin).astype(float), c5, n)
