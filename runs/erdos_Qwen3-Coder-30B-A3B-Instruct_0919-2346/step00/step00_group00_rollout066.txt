import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant for given h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute convolution of h with (1-h)
    conv = np.correlate(h, 1-h, mode='full')
    # Extract the relevant part and multiply by dx
    c5 = np.max(conv) * dx
    return c5

def initial_h_values():
    """Provide a starting point for optimization"""
    n = 45
    h = np.zeros(n)
    # Simple initial construction: step function
    h[:n//2] = 1.0
    h[n//2:] = 0.0
    return h

def objective(h):
    """Objective function to minimize (negative because we minimize)"""
    return -evaluate_erdos_solution(h)

def constraint_eq(h):
    """Equality constraint: integral of h should be 1"""
    n_points = len(h)
    dx = 2.0 / n_points
    return np.sum(h) * dx - 1.0

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Create initial values
    h_init = initial_h_values()
    n_points = len(h_init)
    dx = 2.0 / n_points
    
    # Set up optimization bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Incorporate equality constraint using penalty method
    def penalty_objective(h):
        if not all(0 <= h_i <= 1 for h_i in h):
            return float('inf')
        c5 = evaluate_erdos_solution(h)
        penalty = 1e6 * abs(constraint_eq(h))  # Large penalty if constraint violated
        return c5 + penalty
    
    # Optimization with restarts
    best_c5 = float('inf')
    best_h = None
    
    # Try different starting points
    for restart in range(10):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Random perturb the initial solution
        h_perturbed = h_init + np.random.normal(0, 0.01, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        # Normalize
        h_perturbed = h_perturbed * (1.0 / np.sum(h_perturbed) * dx)
        
        # Optimize using differential evolution
        try:
            result = differential_evolution(
                penalty_objective,
                bounds,
                seed=seed + restart,
                maxiter=500,
                popsize=10,
                disp=False
            )
            if not result.success:
                continue
            c5 = evaluate_erdos_solution(result.x)
            if c5 < best_c5:
                best_c5 = c5
                best_h = result.x.copy()
        except:
            pass
            
        # Try also local optimization
        try:
            local_result = minimize(
                penalty_objective,
                h_perturbed,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 1000}
            )
            if local_result.success:
                c5 = evaluate_erdos_solution(local_result.x)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = local_result.x.copy()
        except:
            pass
    
    # Final fine-tune with gradient-free methods if time allows
    if time.time() - start_time < budget_s * 0.9:
        # More extreme exploration
        for i in range(10):
            if time.time() - start_time > budget_s * 0.95:
                break
            # Random construction
            h_rand = np.random.rand(n_points)
            h_rand = h_rand * (1.0 / np.sum(h_rand) * dx)
            c5 = evaluate_erdos_solution(h_rand)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_rand.copy()
    
    # Ensure we have a valid and good result
    if best_h is None:
        # Fallback to initial solution
        best_h = h_init
        best_c5 = evaluate_erdos_solution(best_h)
    
    # Final check of constraints
    if abs(np.sum(best_h) * dx - 1.0) > 1e-3:
        # Normalize to satisfy constraint
        best_h = best_h * (1.0 / np.sum(best_h) * dx)
        best_c5 = evaluate_erdos_solution(best_h)
        
    return (best_h, best_c5, n_points)
