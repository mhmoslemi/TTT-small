def evaluate_overlap(h):
    """Evaluate the maximum overlap integral for a given h."""
    n = len(h)
    dx = 2.0 / n

    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum of the correlation values
    max_overlap = np.max(corr) * dx
    return max_overlap
