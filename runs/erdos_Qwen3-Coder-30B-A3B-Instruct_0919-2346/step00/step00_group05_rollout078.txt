import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import pdist
import time
import math

def evaluate_erdos_solution(h_values):
    """Evaluate the C5 bound for given h values."""
    h = np.array(h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all shifts
    corr = np.correlate(h, 1 - h, mode='full')
    # The maximum value gives us the bound
    max_corr = np.max(corr)
    return max_corr * dx

def initial_h_values(n_points=70):
    """Return initial step function h."""
    # Create a piecewise constant function
    # Start with simple construction and then optimize
    h = np.zeros(n_points)
    # Take the existing construction as a starting point
    # This is just a placeholder; actual implementation will be different
    # But in the context, we are given this function as global variable
    # We will implement a basic one to start
    # For now, let's use a piecewise linear approximation
    # This might not be optimal, but it is a reasonable starting point
    h[:n_points//2] = 1.0
    h[n_points//2:] = 0.0
    # Normalize to have integral = 1
    h *= 2.0 / n_points
    return h

def objective(h_values):
    """Objective function that we want to minimize."""
    return -evaluate_erdos_solution(h_values)

def constraint1(h_values):
    """Constraint that integral of h equals 1."""
    n_points = len(h_values)
    dx = 2.0 / n_points
    integral = sum(h_values) * dx
    return integral - 1.0

def constraint2(h_values):
    """Constraint that all values are in [0,1]."""
    # We want all h_values to be between 0 and 1
    return np.min(h_values)  # This should be >= 0

def constraint3(h_values):
    """Constraint that all values are in [0,1]."""
    return 1.0 - np.max(h_values)  # This should be >= 0

def optimize_step_function(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Initialize with a basic construction
    n_points = 70
    h = initial_h_values(n_points)
    
    # Set up the constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraint definitions
    cons = [
        {'type': 'eq', 'fun': constraint1},
        {'type': 'ineq', 'fun': constraint2},
        {'type': 'ineq', 'fun': constraint3}
    ]
    
    # Apply optimization
    start_time = time.time()
    
    # First do a simple convergence test with Differential Evolution
    try:
        de_result = differential_evolution(
            objective, 
            bounds, 
            constraints=cons, 
            seed=seed,
            maxiter=int(budget_s / 2),
            popsize=10,
            disp=False
        )
        
        if de_result.success:
            h_opt = de_result.x
            c5_opt = -de_result.fun
            return h_opt, c5_opt, n_points
    except Exception:
        pass
    
    # If that fails, try with a different optimizer
    try:
        # Refine the solution using L-BFGS-B
        result = minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': int(budget_s * 0.7)},
            tol=1e-9
        )
        
        if result.success:
            h_opt = result.x
            c5_opt = -result.fun
            return h_opt, c5_opt, n_points
        
    except Exception:
        pass
    
    # If we couldn't optimize, return the initial guess
    c5_initial = evaluate_erdos_solution(h)
    return h, c5_initial, n_points
