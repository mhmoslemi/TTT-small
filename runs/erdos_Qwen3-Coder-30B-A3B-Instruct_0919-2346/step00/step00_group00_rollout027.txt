def evaluate_c5(h):
    """Evaluate C5 bound for given h values"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation for all shifts
    correlation = np.correlate(h, 1 - h, mode='full')
    # Compute maximum value
    max_corr = np.max(correlation)
    return max_corr * dx
