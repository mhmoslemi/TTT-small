def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C5 for a given step function h."""

    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum value gives us C5
    c5 = np.max(conv) * dx
    return c5
