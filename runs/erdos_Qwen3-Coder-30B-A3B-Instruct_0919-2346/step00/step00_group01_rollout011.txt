def evaluate_h(h_values):
    """Evaluate the C5 constant for given step function h."""
    n = len(h_values)
    dx = 2.0 / n
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h_values, 1 - h_values, mode='full')
    # The maximum overlap integral
    C5 = np.max(correlation) * dx
    return C5

def project_onto_constraints(h, n_points):
    """Project h onto constraints: 0 <= h[i] <= 1 and sum(h) = n_points / 2"""

    # Distribute the residual proportionally
    weights = np.ones_like(h)
    adjusted = h + residual * weights / np.sum(weights)
    return np.clip(adjusted, 0, 1)
