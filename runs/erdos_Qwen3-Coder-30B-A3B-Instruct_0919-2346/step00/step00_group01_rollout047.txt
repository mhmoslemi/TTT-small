def circular_correlation(h, n=None):
    if n is None:
        n = len(h)
    h_padded = np.pad(h, (0, n), mode='constant')
    h_conj = np.pad(np.conj(h), (n, 0), mode='constant')
    corr = ifft(fft(h_padded) * fft(h_conj).conj()).real
    return corr[n:2*n]

def compute_c5_bound(h):
    n = len(h)
    corr = circular_correlation(h, n)
    min_corr = np.min(corr)
    # C_5 = 1 - (2/n) * min_corr
    return 1 - (2.0 / n) * min_corr

def project_to_feasible(h, n_points):
    # h is sorted in descending order of scores
    # we want to pick exactly n_points/2 ones
    # so we set all values above threshold=sorted_h[n_points//2] to 1,
    # and below to 0.
    h_sorted = np.sort(h)[::-1]
    threshold = h_sorted[n_points // 2]
    # If there's a tie, ensure exactly n_points//2 ones
    num_ones = np.sum(h > threshold) + np.sum(h == threshold)
    if num_ones == n_points // 2:
        return np.array([1 if x >= threshold else 0 for x in h])
    else:
        # Adjust threshold until exactly n_points//2 ones
        while num_ones < n_points // 2:
            threshold -= 1e-10
            num_ones = np.sum(h > threshold) + np.sum(h == threshold)
        while num_ones > n_points // 2:
            threshold += 1e-10
            num_ones = np.sum(h > threshold) + np.sum(h == threshold)
        
        return np.array([1 if x >= threshold else 0 for x in h])

def swap_improvement(h, k):
    # compute changes in correlation for shift k
    n = len(h)
    old = h * np.roll(h, k)
    return np.sum(old)

def swap_hill_climb(h):
    # deterministic hill-climbing with swapping
    # Only swap one 1 to a 0 if it improves the worst correlation
    n = len(h)
    h = h.copy()
    best = compute_c5_bound(h)
    changed = True
    iterations = 0
    max_it = 1000
    # Generate pairs efficiently
    while changed and iterations < max_it:
        changed = False
        best_rel = -np.inf
        best_pair = None
        for i in range(n):
            if h[i] == 0:
                continue
            for j in range(n):
                if h[j] == 1:
                    continue
                # Try to swap h[i] <-> h[j]
                old = h.copy()
                old[i], old[j] = old[j], old[i]
                c5 = compute_c5_bound(old)
                if c5 < best:
                    if best_rel < c5:
                        best_rel = c5
                        best_pair = (i, j)
        
        if best_pair is not None:
            i, j = best_pair
            h[i], h[j] = h[j], h[i]
            best = compute_c5_bound(h)
            changed = True
            iterations += 1
        else:
            break
    
    return h

def gradient_fine_tune(h, max_iter=500):
    # Fine-tune continuous solution
    n = len(h)
    h = np.array(h).copy()
    min_corr = compute_c5_bound(h)
    
    def gradient(h_vec):
        # Compute gradient for current min correlation
        # subtract h from itself shifted by k
        corr = circular_correlation(h_vec, n)
        min_idx = np.argmin(corr)
        # differentiable computing
        g = np.zeros(n)
        for k in range(n):
            g[k] -= h_vec[(k + min_idx) % n]
            if k != min_idx:
                g[k] -= h_vec[(k - min_idx) % n] # reverse shift
        return g
    
    # Project to feasible
    def project(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        return h_vec

    # Gradient ascent with diminishing steps
    step = 0.005
    for i in range(max_iter):
        g = gradient(h)
        h_new = h + step * g
        h_new = project(h_new)
        # Project back to sum = n/2
        diff = np.sum(h_new) - n/2
        # Simple projection
        h_new = h_new - diff / n
        
        if compute_c5_bound(h_new) < compute_c5_bound(h):
            h = h_new
        else:
            break
        step *= 0.995  # shrink step size
    return h

def sdp_solve(n):
    # SDP relaxation
    h = cp.Variable(n, name="h")
    X = cp.Variable((n, n), symmetric=True, name="X")
    t = cp.Variable(name="t")

    # Marginals constraints
    constraints = [
        X >> 0,
        cp.diag(X) == h,
        h >= 0, h <= 1,
        cp.sum(h) == n/2
    ]

    # Constraints imposed by p = h*h_k for all shifts k
    for k in range(n):
        shift_idx = np.arange(n)
        if k != 0:
            shift_idx = np.roll(shift_idx, k)
        # bi-k or sk in code
        L = np.zeros((n, n))
        for i in range(n):
            L[i, shift_idx[i]] = 1
            L[shift_idx[i], i] = 1
        constraints.append(cp.trace(L @ X) >= t)

    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constraints)

    try:
        prob.solve(solver=cp.SCS, verbose=False, max_iters=300)
        if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
            h_sol = h.value
            return h_sol, prob.value
    except Exception as e:
        pass
    return None, None

def round_h(h_sdp, n_points):
    if h_sdp is None:
        return None
    X = np.outer(h_sdp, h_sdp)
    try:
        U, s, Vt = np.linalg.svd(X)
        s = np.sqrt(np.maximum(s, 0))
        V = U @ np.diag(s)
        g = np.random.normal(size=V.shape[1])
        scores = np.dot(V, g)
        h = scores > np.partition(scores, n_points//2)[n_points//2]
        h = h.astype(float)
        return h
    except:
        return None

def interval_optimize(n_points=200, num_intervals=6):
    """
    Parametric optimization of interval union.
    Optimizes ratio of intervals to meet target sum (1).
    """
    # We model 1D intervals [a_j, b_j] in (0,2) with symmetry around 1
    max_iter = 20000

    def compute_intersection_lengths(a):
        # a = [a1, b1, a2, b2, ...] first n/2 intervals, then b_i = 2 - a_{n-i+1}, etc.
        if len(a) != num_intervals:
            raise ValueError("num_intervals mismatch")
        b = [2 - x for x in reversed(a)]
        # construct 'h' from intervals [a_i, b_i] with height 1
        h = np.zeros(n_points)
        dx = 2.0 / n_points
        for i in range(num_intervals):
            left = max(0, a[i])
            right = min(2, b[i])
            if left < right:
                # spread to points
                left_idx = int(left / dx)
                right_idx = int(right / dx) + 1
                left_idx = max(0, left_idx)
                right_idx = min(n_points, right_idx)
                h[left_idx:right_idx] = 1

        return h

    def evaluate_constraints(a):
        # Compute overlaps for correlation ground truth
        h = compute_intersection_lengths(a)
        corr = circular_correlation(h, n_points)
        return corr, h

    np.random.seed(42)
    best_corr = np.inf
    best_a = None
    best_h = None

    # Try multiple random starts with good initialisation
    for _ in range(50):
        try:
            # Generate valid endpoints for interval construction
            # Try to assign mass nicely
            a = np.sort(np.random.rand(num_intervals // 2) * 1.9 + 0.05)
            # Add symmetry part:
            a = np.concatenate([a, [2 - x for x in reversed(a)]])
            # Ensure all (0,2) and area = 1
            a = np.clip(a, 0, 2)
            # Compute overlap and update
            corr_vals, h = evaluate_constraints(a)
            min_val = np.min(corr_vals)
            if min_val < best_corr:
                best_corr = min_val
                best_a = a.copy()
                best_h = h.copy()
        except Exception as e:
            continue

    if best_a is None:
        return None, None

    # Now refine with scipy minimize via interval overlap
    def obj(a):
        h = compute_intersection_lengths(a)  # form discrete function
        corr_vals = circular_correlation(h, n_points)
        return -np.min(corr_vals)  # minimize negative -> maximize min

    # bounds
    bounds = [(0.01, 1.99)] * num_intervals

    # Optimization using SLSQP (trust-region)
    try:
        res = minimize(obj, best_a, bounds=bounds, method='SLSQP', options={'maxiter': 1000, 'ftol': 1e-7})
        if res.success and res.fun < best_corr:
            final_h = compute_intersection_lengths(res.x)
            return final_h, -res.fun
    except:
        pass

    return best_h, best_corr

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 80
    total_time = 0

    h_best = None
    c5_best = float('inf')

    start_time = time.time()

    # Layer 1: SDP + rounding + polishing
    sdp_runs = 20  # 30 with time
    for i in range(sdp_runs):
        t0 = time.time()
        h_sdp, t_sdp = sdp_solve(n_points)
        if h_sdp is None:
            continue
        h_rounded = round_h(h_sdp, n_points)
        if h_rounded is None:
            continue
        h_swap = swap_hill_climb(h_rounded)
        h_finetuned = gradient_fine_tune(h_swap, max_iter=500)
        if h_finetuned is None:
            h_finetuned = h_swap

        c5_b = compute_c5_bound(h_finetuned)
        if c5_b < c5_best:
            c5_best = c5_b
            h_best = h_finetuned.copy()
        
        elapsed = time.time() - t0
        total_time += elapsed
        if total_time > budget_s - 40:
            break

    # Layer 2: interval method for fallback
    if total_time < budget_s - 40:
        try:
            h_interval, c5_interval = interval_optimize(n_points=200, num_intervals=6)
            if h_interval is not None and c5_interval < c5_best:
                c5_best = c5_interval
                h_best = h_interval
        except:
            pass

    # Finalize evaluation with exact FFT
    if h_best is not None:
        corr = circular_correlation(h_best, len(h_best))
        min_corr = np.min(corr)
        c5_final = 1.0 - (2.0 / len(h_best)) * min_corr
        return (h_best, c5_final, len(h_best))

    # Fallback from initial if nothing worked
    if initial_h_values is not None:
        h_init = np.array(initial_h_values)
        c5_init = compute_c5_bound(h_init)
        return (h_init, c5_init, len(h_init))

    # Default to a well-known loose bound
    h_default = np.ones(n_points) * 0.5
    h_default[:n_points//2] = 1
    h_default[n_points//2:] = 0
    return (h_default, 0.5, n_points)
