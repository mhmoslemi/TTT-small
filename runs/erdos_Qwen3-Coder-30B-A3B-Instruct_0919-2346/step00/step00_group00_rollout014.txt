import numpy as np
import time
from scipy.signal import convolve

def project_to_capped_simplex(h, target_sum):
    """Project vector h onto the capped simplex with sum = target_sum."""
    n = len(h)
    # Clamp to [0, 1]
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Water-filling
    h_sorted = np.sort(h)[::-1]
    cumsum = np.cumsum(h_sorted)
    # Find the threshold
    i = 0
    while i < n and h_sorted[i] >= (target_sum - cumsum[i]) / (i + 1):
        i += 1
    if i == 0:
        lambda_ = target_sum / n
    else:
        lambda_ = (target_sum - cumsum[i - 1]) / i
    # Apply thresholding
    result = np.maximum(0, np.minimum(1, h - lambda_))
    return result

def compute_correlation(h):
    """Compute correlation matrix for given h."""
    h_rev = h[::-1]
    corr = convolve(h, h_rev, mode='full')
    return corr

def compute_grad_F_alpha(h, alpha, dx):
    """Compute gradient of the log-sum-exp surrogate."""
    n = len(h)
    # Compute correlations for all lags
    corr_h = convolve(h, 1 - h, mode='full')
    corr_h = corr_h * dx
    # corr_h[i] corresponds to O_{i-n+1}
    
    # Compute soft-weights
    exp_O = np.exp(alpha * corr_h)
    exp_O_sum = np.sum(exp_O)
    w = exp_O / exp_O_sum

    # Now compute ∂Fα/∂h[i]
    # Part 1: Σ_ℓ w_ℓ * (1 – h[i+ℓ])
    part1 = convolve(w, 1 - h, mode='full')
    # Part 2: Σ_ℓ w_ℓ * h[i-ℓ]
    part2 = convolve(w[::-1], h, mode='full')
    # Slice to correct size
    grad = part1[n-1:2*n-1] - part2[n-1:2*n-1]
    return grad

def smooth_max_rounding(h, alpha, dx, eta, n):
    """Modify h by one gradient step."""
    # Compute gradient
    grad = compute_grad_F_alpha(h, alpha, dx)
    # Update h
    h_new = h - eta * grad
    # Project back onto feasible set
    h_new = project_to_capped_simplex(h_new, n / 2)
    return h_new

def deterministic_annealing(h0, n, alpha0=0.5, gamma=1.3, eta0=0.01, delta=0.99, max_iterations=10000, max_alpha=200):
    """Run deterministic annealing with restarts."""
    dx = 2.0 / n
    h_current = h0.copy()
    alpha = alpha0
    eta = eta0
    best_h = h_current.copy()
    best_F = compute_F_alpha(h_current, alpha, dx)
    best_c5 = np.max(convolve(h_current, 1 - h_current, mode='full')) * dx
    
    iteration = 0
    while iteration < max_iterations and alpha < max_alpha:
        # Annealing iteration
        for _ in range(200):  # Do some steps before annealing
            h_current = smooth_max_rounding(h_current, alpha, dx, eta, n)
            iteration += 1
            if iteration >= max_iterations:
                break
        # After some steps, increase alpha
        alpha *= gamma
        eta *= delta
        # Check if we improved the best
        F_current = compute_F_alpha(h_current, alpha, dx)
        if F_current < best_F:
            best_F = F_current
            best_h = h_current.copy()
        # Compute exact C5 for best so far
        corr_best = convolve(best_h, 1 - best_h, mode='full')
        c5_exact = np.max(corr_best) * dx
        if c5_exact < best_c5:
            best_c5 = c5_exact
            best_h = h_current.copy()
        if iteration >= max_iterations or alpha >= max_alpha:
            break
    return best_h, best_c5

def discretize_to_binary(h_continuous, n):
    """Convert continuous values to binary step function."""
    # Create indices sorted by h_continuous, descending
    indices = np.argsort(h_continuous)[::-1]
    h_binary = np.zeros(n)
    for i in range(n // 2):
        h_binary[indices[i]] = 1.0
    return h_binary

    n = len(h_binary)

    while (attempts < max_attempts) and (time.time() - start_time < time_limit - 5):
        # Pick two indices
        i = np.random.randint(n)
        j = np.random.randint(n)
        if h_binary[i] == 1 and h_binary[j] == 0:
            # Compute change in max correlation due to swap
            # We'll do an O(n) simple loop update here rather than exact formulas
            old_max = max_corr
            # Apply swap temporarily
            h_binary[i] = 0
            h_binary[j] = 1
            corr_new = convolve(h_binary, 1 - h_binary, mode='full')
            new_max = np.max(corr_new)
            if new_max <= old_max - 1e-10:  # Accept improvement
                max_corr = new_max
                successful_swaps += 1
            else:
                # Revert swap
                h_binary[i] = 1
                h_binary[j] = 0
        attempts += 1
    return h_binary, max_corr
