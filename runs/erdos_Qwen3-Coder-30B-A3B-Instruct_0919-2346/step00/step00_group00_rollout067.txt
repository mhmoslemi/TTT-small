import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute correlation of h with (1-h) using full mode
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value is the C5 bound
    max_corr = np.max(correlation)
    return max_corr * dx

def evaluate_erdos_solution(h_values):
    """Wrapper for existing evaluation function"""
    return evaluate_c5(h_values)

def initial_h_values():
    """Returns initial step function values"""
    # Default construction with 45 samples
    n_points = 45
    h = np.zeros(n_points)
    # Simple inspired construction that tends to perform well
    # Fill first half with 1's and second half with 0's
    h[:n_points//2] = 1.0
    return h

def objective(h):
    """Objective function to minimize - negative of C5 value"""
    return -evaluate_c5(h)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Get initial solution 
    n_points = 45
    h0 = initial_h_values()
    
    # Define bounds for optimization variables [0,1] for each h_i
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Add constraint that sum(h)*dx = 1 => sum(h) = n_points/2
    def constraint(h):
        dx = 2.0 / n_points
        return np.sum(h) - n_points / 2
    
    # Get base solution
    h_best = h0.copy()
    c5_best = evaluate_c5(h_best)
    
    # Use Differential Evolution with constraints
    # Fun: minimize C5 using bounds and constraint
    de_result = differential_evolution(
        objective,
        bounds,
        args=(),
        maxiter=int(1000),
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.7,
        seed=seed,
        atol=1e-6,
        x0=h_best
    )
    
    h_opt = de_result.x
    c5_opt = -objective(h_opt)
    
    if c5_opt < c5_best:
        h_best = h_opt
        c5_best = c5_opt
        
    # If our time allows, try a smaller variation with fewer points
    # Emphasis on fewer particles for better optimization value
    for n_try in [30, 35]:
        if time.time() - start_time > 0.8 * budget_s:
            break
            
        h_try = np.zeros(n_try)
        dx = 2.0 / n_try
        # Again try simple initial construction
        h_try[:n_try//2] = 1.0 
        bounds_try = [(0, 1) for _ in range(n_try)]
        
        # Reconstruct with fewer points
        try:
            de_try = differential_evolution(
                lambda h: -evaluate_c5(h),
                bounds_try,
                args=(),
                maxiter=int(500),
                popsize=10,
                mutation=(0.5, 1),
                recombination=0.7,
                seed=seed,
                atol=1e-6,
                x0=h_try
            )
            
            h_try_opt = de_try.x
            c5_try = -de_try.fun
            
            if c5_try < c5_best:
                h_best = h_try_opt
                c5_best = c5_try
                n_points = n_try
        except:
            continue
    
    # Further local refinement with scipy minimize, if time permits
    elapsed = time.time() - start_time
    if elapsed < 0.9 * budget_s:
        # Local optimization
        try:
            from scipy.optimize import minimize
            
            def obj_h(h):
                return evaluate_c5(h)
                
            # Create bounds and constraints for refine (for sample points)
            cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points/2}
            bnds = [(0, 1) for _ in range(n_points)]
            
            result = minimize(obj_h, h_best, method='SLSQP', bounds=bnds, constraints=cons, options={'maxiter':200})
            if result.success or not np.isnan(result.fun):
                h_refined = result.x
                c5_refined = obj_h(h_refined)
                if c5_refined < c5_best:
                    h_best = h_refined
                    c5_best = c5_refined
        except:
            pass
    
    return (h_best, c5_best, n_points)
