import numpy as np
import time
import random

def compute_c5(h):
    """Compute the C5 bound from the step function h"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return corr.max()

def propose_swap_and_update(h, corr, i, j):
    """Propose a swap and update correlation array incrementally"""
    old_i, old_j = h[i], h[j]
    delta_i = old_j - old_i
    
    n = len(h)
    
    # Copy the correlation array for temporary update
    new_corr = corr.copy()
    
    # Loop over k not i,j to compute delta
    for k in range(n):
        if k == i or k == j:
            continue
        hk = h[k]
        # shift i->k: l = k - i
        idx1 = k - i + n - 1
        new_corr[idx1] += delta_i * (1 - hk)
        # shift k->i: l = i - k
        idx2 = i - k + n - 1
        new_corr[idx2] += hk * (-delta_i)
        # shift j->k: l = k - j
        idx3 = k - j + n - 1
        new_corr[idx3] += -delta_i * (1 - hk)
        # shift k->j: l = j - k
        idx4 = j - k + n - 1
        new_corr[idx4] += hk * (delta_i)
        
    # handle cross pair i->j and j->i
    idx_ij = j - i + n - 1
    new_corr[idx_ij] += delta_i
    idx_ji = i - j + n - 1
    new_corr[idx_ji] -= delta_i
    
    # Compute new max
    new_max = new_corr.max()
    
    # Return new correlation and maximum
    return new_corr, new_max

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()
    
    # Use larger n for better resolution
    n = 200  # Even number for integer half
    dx = 2.0 / n
    
    # Initial h
    h = np.zeros(n, dtype=float)
    ones = n // 2
    indices = np.random.choice(n, size=ones, replace=False)
    h[indices] = 1.0
    
    # Initial correlation and C5
    corr = np.correlate(h, 1 - h, mode='full')
    cur_max = corr.max()
    cur_c5 = cur_max * dx
    
    best_c5 = cur_c5
    best_h = h.copy()
    best_n = n
    
    # Temperature for simulated annealing
    T = 0.02
    T_decay = 0.99995
    
    iteration = 0
    max_iters = 100000
    
    start_time = time.time()
    
    for iteration in range(max_iters):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Choose two distinct positions
        i = np.random.randint(n)
        j = np.random.randint(n)
        while h[i] == h[j]:
            i = np.random.randint(n)
            j = np.random.randint(n)
        
        # Propose swap and compute new correlation with delta update
        new_corr, new_max = propose_swap_and_update(h, corr, i, j)
        new_c5 = new_max * dx
        
        # Accept or reject
        d = new_c5 - cur_c5
        if d < 0 or np.random.rand() < np.exp(-d / (T + 1e-12)):
            corr = new_corr
            cur_max = new_max
            cur_c5 = new_c5
            h[i], h[j] = h[j], h[i]
            
            # Keep best
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                best_n = n
                
        # Update temperature
        T *= T_decay
        
    return best_h, best_c5, best_n
