import numpy as np
import time

def evaluate(h):
    """Evaluate the maximum overlap integral using correlation."""
    c = np.correlate(h, 1 - h, mode='full')
    return np.max(c) * (2.0 / len(h))

def get_max_shifts(h):
    """Get list of integer shifts where correlation attains maximum."""
    N = len(h)
    c = np.correlate(h, 1 - h, mode='full')
    max_val = c.max()
    indices = np.where(c == max_val)[0]
    shifts = indices - (N - 1)  # because index N-1 corresponds to shift 0
    return shifts.tolist()

def gradient_max_shifts(h, shifts):
    """Compute the averaged subgradient for the set of maximizing shifts."""
    N = len(h)
    g = np.zeros(N)
    for k in shifts:
        if k >= 0:
            left = np.concatenate((np.zeros(k), h[:N-k]))
            right = np.concatenate((h[k:], np.zeros(k)))
        else:
            k_abs = -k
            left = np.concatenate((h[k_abs:], np.zeros(k_abs)))
            right = np.concatenate((np.zeros(k_abs), h[:N-k_abs]))
        g += 1 - left - right
    g /= len(shifts)
    return g

def project_to_capped_simplex(v, S):
    """Project vector v onto the capped simplex {x: sum(x)=S, 0≤x≤1}."""
    # Binary search on λ such that sum(clamp(v - λ, 0, 1)) = S
    lo, hi = -2.0, 2.0
    for _ in range(50):
        lam = (lo + hi) / 2
        x = np.clip(v - lam, 0, 1)
        s = x.sum()
        if s > S:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo + hi) / 2, 0, 1)
    return x

def random_feasible(N):
    """Generate a random feasible vector (sum = N/2, 0≤x≤1)."""
    v = np.random.rand(N)
    return project_to_capped_simplex(v, N / 2)
