import numpy as np
import time
import math
import random

def evaluate(h):
    """Evaluate the C5 bound for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    O = np.correlate(h, 1 - h, mode='full')
    return O.max() * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    def local_search(h, time_limit):
        n = len(h)
        S = n // 2
        current_val = evaluate(h)
        best_h = h.copy()
        best_val = current_val
        
        # Lists to efficiently select ones and zeros
        ones_idx = list(np.flatnonzero(h))
        zeros_idx = list(np.flatnonzero(1 - h))
        
        max_iter = int(1e7)
        temp_start = 0.1
        temp_end = 1e-5
        for it in range(max_iter):
            if time.time() - start_time > budget_s * 0.95:
                break
                
            # Temperature schedule
            temp = temp_start * (temp_end / temp_start) ** (it / max_iter)
            
            # Pick a random 1 and 0
            i = random.choice(ones_idx)
            j = random.choice(zeros_idx)
            
            # Perform swap
            h[i], h[j] = 0, 1
            
            # Update lists
            oi = ones_idx.index(i)
            oj = zeros_idx.index(j)
            ones_idx[oi] = j
            zeros_idx[oj] = i
            
            # New evaluation
            new_val = evaluate(h)
            
            # Accept with probability
            if new_val <= current_val or random.random() < math.exp((current_val - new_val) / temp):
                current_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best_h = h.copy()
            else:
                # Revert
                h[i], h[j] = 1, 0
                ones_idx[oi] = i
                zeros_idx[oj] = j
        
        return best_h, best_val
    
    lengths = [64, 80, 96, 112, 128, 144, 160, 176, 192, 224, 256, 320, 384, 448, 512]
    # Filter to reasonable sizes
    lengths = [n for n in lengths if n <= 800]
    
    best_h = None
    best_c5 = float('inf')
    best_n = None

    # Use the initial_h_values to scale up if available
    if 'initial_h_values' in globals():
        base = np.array(initial_h_values, dtype=float)
        x_old = np.linspace(0, 2, len(base), endpoint=False)
        # Replicate to each n
        for n in lengths:
            x_new = np.linspace(0, 2, n, endpoint=False)
            h_interp = np.interp(x_new, x_old, base)
            # Convert to binary with adjusted weight
            h_bin = (h_interp > 0.5).astype(int)
            h_corrected = h_bin.copy()
            current_sum = h_corrected.sum()
            target_sum = n // 2
            if current_sum > target_sum:
                # Turn some 1s to 0s
                one_indices = np.where(h_corrected == 1)[0]
                np.random.shuffle(one_indices)
                for i in one_indices[:current_sum - target_sum]:
                    h_corrected[i] = 0
            elif current_sum < target_sum:
                # Turn some 0s to 1s
                zero_indices = np.where(h_corrected == 0)[0]
                np.random.shuffle(zero_indices)
                for i in zero_indices[:target_sum - current_sum]:
                    h_corrected[i] = 1
            b_h, b_val = local_search(h_corrected, (budget_s * 0.9) / len(lengths))
            if b_val < best_c5:
                best_c5 = b_val
                best_h = b_h
                best_n = n
    
    # Try other random candidates
    for _ in range(5):
        for n in lengths:
            if time.time() - start_time > budget_s * 0.95:
                break
            h = np.random.choice([0, 1], size=n, p=[0.5, 0.5])
            # Ensure sum is exactly n/2
            while h.sum() < n // 2:
                zero_indices = np.where(h == 0)[0]
                h[np.random.choice(zero_indices)] = 1
            while h.sum() > n // 2:
                one_indices = np.where(h == 1)[0]
                h[np.random.choice(one_indices)] = 0
            b_h, b_val = local_search(h, (budget_s * 0.9) / len(lengths))
            if b_val < best_c5:
                best_c5 = b_val
                best_h = b_h
                best_n = n

    return (best_h, best_c5, best_n)
