import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_erdos_solution(h_values):
    """Evaluate the Erdős constant for given h values."""
    n = len(h_values)
    dx = 2.0 / n
    # Compute the integral using correlation
    correlation = np.correlate(h_values, 1 - np.array(h_values), mode="full")
    max_overlap = np.max(correlation) * dx
    return max_overlap

def initial_h_values():
    """Initial construction: a step function with 70 samples."""
    n = 70
    h = np.zeros(n)
    # More concentrated around center
    midpoint = n // 2
    h[midpoint - 10:midpoint + 10] = 1.0
    h *= 2.0 / np.sum(h)  # Normalize so integral is 1
    return h

def objective(h_values):
    """Objective function to minimize, returns negative of C5 for scipy."""
    return -evaluate_erdos_solution(h_values)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize with the provided construction
    h0 = initial_h_values()
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Define bounds (0 <= h[i] <= 1) for all i
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Add constraint that integral = 1 (sum(h)*dx = 1 => sum(h) = n/2)
    # We'll use a penalty approach to enforce this
    
    def penalized_objective(h_values):
        # Eval the original objective
        c5_val = evaluate_erdos_solution(h_values)
        # Add penalty for not satisfying integral constraint
        integral = np.sum(h_values) * dx
        penalty = 1000 * (integral - 1.0)**2 if abs(integral - 1.0) > 1e-6 else 0
        return -c5_val + penalty
    
    # Use a local optimization approach with restarts to avoid local minima
    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(h0)
    
    # Try multiple random restarts
    max_restarts = 50
    num_restarts = 0
    
    while time.time() - start_time < budget_s * 0.95 and num_restarts < max_restarts:
        # Random perturbation
        h_trial = h0 + np.random.normal(0, 0.05, n_points)
        # Projection to [0,1]
        h_trial = np.clip(h_trial, 0, 1)
        # Rescale to make integral == 1
        integral = np.sum(h_trial) * dx
        if integral > 0:
            h_trial *= (1.0 / integral)
        
        # Optimize with bounds
        try:
            result = differential_evolution(penalized_objective, bounds, maxiter=50, popsize=15, seed=seed+num_restarts)
            if result.success:
                h_new = result.x
                # Re-adjust to correct integral
                integral_new = np.sum(h_new) * dx
                h_new *= (1.0 / integral_new)
                c5_new = evaluate_erdos_solution(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
        except:
            pass
            
        num_restarts += 1
        if time.time() - start_time > budget_s * 0.99:
            break
    
    # Final optimization on best solution found
    bounds_full = [(0, 1) for _ in range(n_points)]
    
    try:
        # Let's try another approach using scipy.optimize.minimize with bounds
        from scipy.optimize import minimize
        
        def obj_func(h):
            integral = np.sum(h) * dx
            penalty = 10000 * (integral - 1.0)**2 if abs(integral - 1.0) > 1e-6 else 0
            c5_val = evaluate_erdos_solution(h)
            return -c5_val + penalty
        
        result = minimize(obj_func, best_h, method='L-BFGS-B', bounds=bounds_full, 
                          options={'maxiter': 1000, 'ftol': 1e-9})
        if result.success:
            final_h = result.x
            final_integral = np.sum(final_h) * dx
            final_h *= (1.0 / final_integral)
            final_c5 = evaluate_erdos_solution(final_h)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    except:
        pass
    
    # Final check
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
