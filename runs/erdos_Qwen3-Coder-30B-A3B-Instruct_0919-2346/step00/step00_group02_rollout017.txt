def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n_points = len(h)

    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the correlation
    max_overlap = np.max(correlation)
    return max_overlap * dx
