import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial guess
    h_init = np.array(initial_h_values)
    n = len(h_init)
    dx = 2.0 / n
    target_sum = n / 2.0
    
    best_C5 = float('inf')
    best_h = None
    
    # Helper function to evaluate C5
    def eval_C5(h):
        c = np.correlate(h, 1 - h, mode='full')
        return np.max(c) * dx
    
    # Helper function to project onto capped simplex
    def project_simplex(v, s=1.0):
        v = np.asarray(v)
        n = len(v)
        u = np.sort(v)[::-1]
        rho = 0
        for i in range(n):
            if v[i] > u[i] - (np.sum(u[:i+1]) - s) / (i + 1):
                rho = i
                break
        else:
            rho = n - 1
        theta = (np.sum(u[:rho+1]) - s) / (rho + 1)
        return np.maximum(v - theta, 0)
    
    # Symmetry breaking: fix h[0] = 1
    def solve_with_warmstart(h_start):
        # Create variables
        h = cp.Variable(n, name='h')
        y = {}
        for k in range(1, n):
            y[k] = cp.Variable(n - k, name='y')
        tau = cp.Variable(name='tau')
        
        # Constraints
        constraints = []
        # The sum constraint
        constraints.append(cp.sum(h) == target_sum)
        # Box constraints
        constraints.append(h >= 0)
        constraints.append(h <= 1)
        # Fix h[0] (symmetry case)
        constraints.append(h[0] == 1)
        
        # Product constraints using McCormick envelopes
        for k in range(1, n):
            for i in range(n - k):
                # y_k[i] >= h[i] + h[i+k] - 1
                constraints.append(y[k][i] >= h[i] + h[i + k] - 1)
                # y_k[i] >= 0
                constraints.append(y[k][i] >= 0)
                # y_k[i] <= h[i]
                constraints.append(y[k][i] <= h[i])
                # y_k[i] <= h[i+k]
                constraints.append(y[k][i] <= h[i + k])
        
        # Tau constraints
        for k in range(1, n):
            constraints.append(tau <= cp.sum(y[k]))
        
        # Objective
        objective = cp.Maximize(tau)
        
        prob = cp.Problem(objective, constraints)
        
        # Set warm start for h variable
        if h_start is not None:
            h.value = h_start.copy()
        
        # Solve with time limit
        try:
            prob.solve(solver=cp.CBC, time_limit=budget_s * 0.8)
            h_sol = h.value
            tau_sol = tau.value
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                if h_sol is not None:
                    # Non-binary case, project to feasible region
                    h_proj = project_simplex(h_sol, target_sum)
                    c5_proj = eval_C5(h_proj)
                    return h_proj, c5_proj
        except Exception as e:
            pass
            
        return None, float('inf')
    
    # Warm start: binary solution from initial_h_values, rounded
    h_bin = (h_init >= 0.5).astype(float)
    h_bin_sum = np.sum(h_bin)
    # Ensure exact sum
    diff = target_sum - h_bin_sum
    if diff > 0:
        # Add in the ones
        indices = np.argsort(-h_bin)
        for i in range(int(diff)):
            if indices[i] < n:
                h_bin[indices[i]] = 1.0
    elif diff < 0:
        # Take out ones
        indices = np.argsort(h_bin)
        for i in range(int(-diff)):
            if indices[i] < n:
                h_bin[indices[i]] = 0.0
    temp_h, c5 = solve_with_warmstart(h_bin.copy())
    if temp_h is not None:
        if c5 < best_C5:
            best_C5 = c5
            best_h = temp_h.copy()
    
    # Continue with time remaining
    count = 0
    time_left = budget_s - (time.time() - start_time)
    while time_left > 5 and count < 20:
        # Random binary start
        h_rand = np.random.randint(0, 2, size=n).astype(float)
        h_rand_sum = np.sum(h_rand)
        diff = target_sum - h_rand_sum
        if diff > 0:
            indices = np.argsort(-h_rand)
            for i in range(int(diff)):
                if indices[i] < n:
                    h_rand[indices[i]] = 1.0
        elif diff < 0:
            indices = np.argsort(h_rand)
            for i in range(int(-diff)):
                if indices[i] < n:
                    h_rand[indices[i]] = 0.0
        # Ensure index 0 is 1
        h_rand[0] = 1.0
        temp_h, c5 = solve_with_warmstart(h_rand.copy())
        if temp_h is not None:
            if c5 < best_C5:
                best_C5 = c5
                best_h = temp_h.copy()
        time_left = budget_s - (time.time() - start_time)
        count += 1
        
    # Final check
    if best_h is None:
        best_h = h_init.copy()
        best_C5 = eval_C5(best_h)
    
    return (best_h, best_C5, n)
