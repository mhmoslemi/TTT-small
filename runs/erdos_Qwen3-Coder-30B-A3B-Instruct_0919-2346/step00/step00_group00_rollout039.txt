import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
import time
import math

def evaluate_erdos_solution(h_values, n_points):
    """Compute the C5 bound for given h_values"""
    dx = 2.0 / n_points
    # Ensure h is properly normalized
    h = np.array(h_values)
    if abs(np.sum(h) * dx - 1.0) > 1e-10:
        h = h * (1.0 / (np.sum(h) * dx)) * (n_points / 2)
    
    # Compute correlations
    conv = np.correlate(h, 1 - h, mode='full')
    # The maximum is at the center of the correlation
    max_corr = np.max(conv)
    return max_corr * dx

def initial_h_values(n_points):
    """Initial construction. This is a placeholder, actual one needed"""
    # Use a reasonable starting point: piecewise constant function
    h = np.ones(n_points) * 0.5
    # Normalize to have integral = 1
    h = h * (1.0 / (np.sum(h) * 2.0 / n_points)) * (n_points / 2)
    return h

def objective_function(h, n_points):
    """Return the negative of C5 value to minimize"""
    dx = 2.0 / n_points
    # Ensure h remains in [0, 1]
    h_clipped = np.clip(h, 0, 1)
    # Normalization constraint: sum(h) * dx = 1 => sum(h) = n_points / 2
    target_sum = n_points / 2.0
    penalty = 1e6 * (np.sum(h_clipped) - target_sum) ** 2
    # Compute the overlap integral
    conv = np.correlate(h_clipped, 1 - h_clipped, mode='full')
    max_corr = np.max(conv)
    return -(max_corr * dx) + penalty

def optimize_h(n_points, initial_guess, max_iter=5000):
    """Optimize h using scipy minimize"""
    # Objective function with constraints
    def obj(h):
        return objective_function(h, n_points)
    
    # Bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraint for integral = 1
    def constraint_func(h):
        return np.sum(h) * (2.0 / n_points) - 1.0
    
    cons = {'type': 'eq', 'fun': constraint_func}
    
    # Optimize
    result = minimize(obj, initial_guess, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': max_iter, 'ftol': 1e-8, 'gtol': 1e-8})
    
    return result.x

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with the initial construction
    n_points = 100
    h_current = initial_h_values(n_points)
    current_c5 = evaluate_erdos_solution(h_current, n_points)
    
    # Optimizing from the initial values
    max_iter = 2000
    try:
        h_new = optimize_h(n_points, h_current, max_iter)
        new_c5 = evaluate_erdos_solution(h_new, n_points)
        if new_c5 < current_c5:
            h_current, current_c5 = h_new, new_c5
    except Exception as e:
        pass
    
    # Try a few other random starts to improve
    for i in range(10):
        if time.time() - start_time > budget_s * 0.95:
            break
        try:
            # Small random perturbation of the current solution
            h_perturb = h_current + np.random.normal(0, 0.01, n_points)
            h_perturb = np.clip(h_perturb, 0, 1)
            # Renormalize
            h_perturb = h_perturb * (n_points / (2 * np.sum(h_perturb)))
            h_perturb = np.clip(h_perturb, 0, 1)
            
            # Re-optimize
            h_new = optimize_h(n_points, h_perturb, max_iter // 2)
            new_c5 = evaluate_erdos_solution(h_new, n_points)
            if new_c5 < current_c5:
                h_current, current_c5 = h_new, new_c5
        except:
            pass

    # Final refinement if we have time
    if time.time() - start_time < budget_s * 0.9:
        try:
            # Try evolutionary optimization
            from scipy.optimize import differential_evolution
            bounds = [(0, 1) for _ in range(n_points)]
            
            def obj_de(h):
                return objective_function(h, n_points)
                
            result = differential_evolution(obj_de, bounds, seed=seed,
                                            maxiter=500, popsize=15, 
                                            updating='immediate')
            h_final = result.x
            # Renormalize
            h_final = h_final * (n_points / (2 * np.sum(h_final)))
            h_final = np.clip(h_final, 0, 1)
            c5_final = evaluate_erdos_solution(h_final, n_points)
            if c5_final < current_c5:
                h_current, current_c5 = h_final, c5_final
        except:
            pass

    return (h_current, -current_c5, n_points)
