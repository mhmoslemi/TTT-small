import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral for given step function h"""
    # Compute convolution of h with (1-h) using full mode
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum value gives us the overlap constant
    max_overlap = np.max(conv) * dx
    return max_overlap
