import numpy as np
import time

def c5_of(h):
    """Compute the maximum overlap integral C5 for a given step function h."""
    dx = 2.0 / len(h)
    # Compute correlation: h * (1 - h shifted)
    correlation = np.correlate(h, 1 - h, mode='full')
    # Multiply by dx to get actual integral value
    c5 = np.max(correlation) * dx
    return c5

def propose_move(h, max_step):
    """Propose a new step function by adjusting two points to preserve sum."""
    N = len(h)
    h_new = h.copy()
    
    # Pick two distinct indices
    i, j = np.random.choice(N, 2, replace=False)
    
    # Determine bounds for delta
    delta_low = max(-h_new[i], h_new[j] - 1)
    delta_high = min(1 - h_new[i], h_new[j])
    
    # If no valid move, return original
    if delta_low >= delta_high:
        return h_new
    
    # Sample delta within feasible range
    step = min(max_step, delta_high - delta_low)
    delta = np.random.uniform(delta_low, delta_high)
    
    # Apply the move
    h_new[i] += delta
    h_new[j] -= delta
    
    return h_new

def anneal(initial_h, time_budget):
    """Perform simulated annealing to minimize C5."""
    start = time.time()
    
    cur = initial_h.copy()
    best = initial_h.copy()
    c_cur = c5_of(cur)
    c_best = c_cur
    
    # Annealing parameters
    T_start = 0.05
    T_end = 1e-6
    step_start = 0.2
    
    # Time-dependent parameters
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T_start * (T_end / T_start) ** (elapsed / time_budget)
        step = step_start * (1 - elapsed / time_budget)
        
        # Propose a move
        h_new = propose_move(cur, step)
        c_new = c5_of(h_new)
        
        # Acceptance rule
        if c_new <= c_cur or np.random.rand() < np.exp(-(c_new - c_cur) / T):
            cur = h_new
            c_cur = c_new
            
        # Update best
        if c_new < c_best:
            best = h_new
            c_best = c_new
            
    return best, c_best
