import numpy as np
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()
    best_h = None
    best_c5 = float('inf')
    best_n = None

    def c5_bound(h):
        n = len(h)
        dx = 2.0 / n
        O = np.correlate(h, 1 - h, mode='full')
        return O.max() * dx

    def random_binary(n):
        h = np.zeros(n, dtype=int)
        ones = np.random.choice(n, n//2, replace=False)
        h[ones] = 1
        return h

    def thue_morse(n):
        h = np.zeros(n, dtype=int)
        for i in range(n):
            h[i] = bin(i).count('1') % 2
        return h

    def legendre_seq(n):
        if n % 2 == 0 or not is_prime(n):
            raise ValueError(f"Legendre sequence requires odd prime n, got {n}")
        h = np.zeros(n, dtype=int)
        for i in range(n):
            if pow(i, (n-1)//2, n) == 1:
                h[i] = 1
        return h

    def is_prime(n):
        if n < 2:
            return False
        if n == 2:
            return True
        if n % 2 == 0:
            return False
        for i in range(3, int(np.sqrt(n)) + 1, 2):
            if n % i == 0:
                return False
        return True

    def replicate_initial_h(n):
        if 'initial_h_values' not in globals():
            return random_binary(n)
        base = np.array(initial_h_values, dtype=float)
        x_old = np.linspace(0, 2, len(base), endpoint=False)
        x_new = np.linspace(0, 2, n, endpoint=False)
        h_interp = np.interp(x_new, x_old, base)
        h_bin = (h_interp > 0.5).astype(int)
        # adjust exact weight
        cur = h_bin.sum()
        target = n // 2
        if cur > target:
            idx = np.where(h_bin == 1)[0]
            np.random.shuffle(idx)
            h_bin[idx[:cur-target]] = 0
        elif cur < target:
            idx = np.where(h_bin == 0)[0]
            np.random.shuffle(idx)
            h_bin[idx[:target-cur]] = 1
        return h_bin

    def local_search(h, time_budget):
        n = len(h)
        S = n // 2
        current_val = c5_bound(h)
        best_h_local = h.copy()
        best_val = current_val
        start = time.time()
        max_iter = 500000
        # Prepare indices of 1s and 0s for fast swap
        ones = list(np.flatnonzero(h))
        zeros = list(np.flatnonzero(1 - h))
        temp_start = 0.1
        temp_end = 1e-5
        it = 0
        while time.time() - start < time_budget and it < max_iter:
            it += 1
            temp = temp_start * (temp_end / temp_start) ** (it / max_iter)
            # Select random 1 and 0 to swap
            oi = random.randint(0, len(ones) - 1)
            oj = random.randint(0, len(zeros) - 1)
            i = ones[oi]
            j = zeros[oj]
            # Swap values
            h[i], h[j] = 0, 1
            # Update indices
            ones[oi], zeros[oj] = j, i
            # Compute new bound
            new_val = c5_bound(h)
            delta = new_val - current_val
            if delta <= 0 or random.random() < np.exp(-delta / temp):
                current_val = new_val
                if current_val < best_val:
                    best_val = current_val
                    best_h_local = h.copy()
            else:
                # Revert swap
                h[i], h[j] = 1, 0
                # Revert indices
                ones[oi], zeros[oj] = i, j
        return best_h_local, best_val

    # Test various candidate n values (even)
    lengths = [64, 80, 96, 112, 128, 144, 160, 176, 192, 224, 256, 320, 384, 448, 512]
    lengths = [n for n in lengths if n <= 800 and n % 2 == 0]

    candidates = []
    for n in lengths:
        # 1. Thue-Morse if n is power of 2
        if (n & (n - 1)) == 0:
            try:
                seq = thue_morse(n)
                candidates.append((seq, n))
            except:
                pass
        # 2. Replicate initial_h_values if available (clearly from n=42)
        try:
            seq = replicate_initial_h(n)
            candidates.append((seq, n))
        except:
            pass
        # 3. Random binary sequences
        for _ in range(2):
            seq = random_binary(n)
            candidates.append((seq, n))

    # Evaluate all candidates
    best_candidates = []

    for seq, n in candidates:
        if time.time() - start > 0.9 * budget_s:
            break
        val = c5_bound(seq)
        best_candidates.append((seq, val, n))

    # Sort candidates by their bound
    best_candidates.sort(key=lambda x: x[1])
    top_candidates = best_candidates[:5]

    # Now run local search for each top candidate
    remaining_budget = budget_s * 0.9 - (time.time() - start)
    if remaining_budget <= 0:
        return (best_h, best_c5, best_n) if best_h is not None else (None, float('inf'), None)

    time_per_candidate = remaining_budget / max(1, len(top_candidates))

    for seq, _, n in top_candidates:
        if time.time() - start > budget_s * 0.95:
            break
        improved_h, improved_val = local_search(seq, time_per_candidate)
        if improved_val < best_c5:
            best_c5 = improved_val
            best_h = improved_h.copy()
            best_n = n
            if best_c5 <= 0.379:
                break

    if best_h is None:
        # Return worst-performing initial solution if nothing better found
        val = c5_bound(seq)
        best_c5 = val
        best_h = seq.copy()
        best_n = n

    return (best_h, best_c5, best_n)
