import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_h(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all k values
    # Using full convolution (equivalent to correlation with 1-h)
    h_rev = h[::-1]
    corr = np.correlate(h, 1 - h_rev, mode='full')
    # The correlation has length 2*n - 1, which corresponds to k from -1 to 1
    # The maximum k values of interest are from 0 to 2, which is the latter half
    # plus some of the first half (but those are symmetric)
    # We only need max of the middle values which correspond to valid k in [0,2]
    # The valid k range covers [0, 2] and the corresponding indices
    # We actually want: all k from 0 to 2 in 2/n increments, but that's not what we compute
    # We compute all k from -1 to 1, but we're only interested in k from 0 to 2
    # Actually, the full correlation gives us k from -n+1 to n-1
    # And we want k from 0 to 2, which means k from 0 to 2 with step 2/n => k from 0 to 2
    # Problem: our data has n points in [0, 2], so dx = 2/n, so k step is 2/n
    # e.g. for n=70, dx = 2/70, k step = 2/n = 2/70
    # So for k ranging from 0 to 2, we want indices from 0 to 2/dx = n
    # But will be careful to look at the actual indices for k from 0 to 2
    # Full correlation goes from -n+1 to n-1 (index-wise), which is k from -1 to 1
    # So the values from index to n-1 correspond to k from 0 to 1
    # The values from 0 to n-1 correspond to k from -1+1=0 to 1 (that's the range we care about)
    # We want max of the values at indices n-1 to 2*n-2 to get k from 0 to 2
    # Wait no, all values of convex/one-sided correlation from -1 to 1
    # index k=-1 to 1 corresponds to indices 0 to 2*n-2
    # Let's discretize k properly to match dx, but in fact just take max correlation value within k of [0, 2]
    # The important half which reflects k in [0, 2] is from index n-1 to 2*n-1 (since k goes from 0 to 1), so for k from 0 to 2 we might want to cut the right part of that
    # Actually, full correlation gives outcomes corresponding to k from -n+1 to n-1 (in order of indices 0 to 2*n-2)
    # So k=0 goes to index n-1, k=1 goes to index n, k=2 goes to index n+1, etc.
    # But k will be in range [0, 2], which means k=0, k=1, k=2 corresponds to indices n-1, n, n+1
    # But spread out by more points since we're using the full correlation
    # Simulation leads to: indices from n-1 to 2*n-2 correspond to k from 0 to 2
    # The maximum of these when k is in [0,2] covers both full-length measurement from 0 to 2
    # So take max over indices n-1 to 2*n-2 (i.e., for k from 0 to 2 in the correlation values)
    # Plus require valid indexing
    if len(corr) < 2 * n - 1:
        raise ValueError("Correlation size mismatch")
    max_corr = np.max(corr[n-1 : 2*n-1])
    return max_corr * dx

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()

    h0 = initial_h()
    n = len(h0)

    bounds = [(0, 1) for _ in range(n)]
    
    # Use differential evolution for exploration
    result = differential_evolution(

        seed=seed,

        disp=False,
        polish=True,
    )
    best_h = result.x
    
    # Apply final constraint that sum(h)*dx = 1, which means sum(h)=n/2
    actual_sum = np.sum(best_h)
    scale_factor = (n / 2) / actual_sum
    best_h = best_h * scale_factor
    
    # Check that final constraint holds
    assert abs(np.sum(best_h) * dx - 1) < 1e-10, "Final constraint not satisfied"
    
    # Evaluate C5
    final_value = evaluate_h(best_h)
    
    # Check if we have a good improvement
    # Track the best we have got so far
    best_candidate_h = best_h
    best_candidate_c5 = final_value
    
    # Now try local optimization to refine
    # But we can only do small steps within reasonable time
    max_iter = 100

        if test_c5 < best_candidate_c5:
            best_candidate_h = test_h
            best_candidate_c5 = test_c5
