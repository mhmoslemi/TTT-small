import numpy as np
from scipy import signal
import time

def evaluate_overlap(h):
    """Compute maximum overlap integral for step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute overlap for all shifts using correlation
    # h(x)(1 - h(x+k)) = h(x) - h(x) h(x+k)
    # So overlap = integral of h(x) - h(x) h(x+k) = sum(h) - sum(h * h_shift)
    # Using correlation: np.correlate(h, 1-h, mode='full') gives sum(h[i] * (1 - h[i+k]))
    
    # For our definition, we need to compute sum(h[i] * (1 - h[i+k])) over valid i
    # np.correlate(h, 1-h, mode='full') computes sum(h[i] * (1 - h[i+k])) when the supports overlap
    # Gives length 2*n-1 with correlation at index k+n-1 corresponding to shift k
    
    h_rev = 1 - h
    correlation = signal.correlate(h, h_rev, mode='full')
    
    # Index 0 to n-1 corresponds to shift -n+1 to 0
    # Index n to 2*n-2 corresponds to shift 1 to n-1
    # We want shifts from 1 to n-1 (excluding 0)
    # Index k+n-1 in correlation corresponds to shift k (0 to n-2)
    max_overlap = 0
    for k in range(1, n):  # Shifts from 1 to n-1
        overlap = correlation[k + n - 1]  # shift k
        max_overlap = max(max_overlap, overlap)
    
    return max_overlap * dx
