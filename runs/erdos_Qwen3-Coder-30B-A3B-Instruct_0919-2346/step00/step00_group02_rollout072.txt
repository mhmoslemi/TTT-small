import numpy as np
import time

def c5_of(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    overlap = np.correlate(h, 1-h, mode='full')
    # Multiply by dx to get actual integral values
    overlap *= dx
    # Return maximum value
    return np.max(overlap)

def propose_move(h, max_step):
    """Generate a feasible pairwise move preserving sum."""
    h = h.copy()
    N = len(h)
    
    # Try up to 100 times to find a valid pair
    for _ in range(100):
        i, j = np.random.choice(N, 2, replace=False)
        
        # Compute feasible delta range
        delta_low = max(-h[i], h[j] - 1)
        delta_high = min(1 - h[i], h[j])
        
        # Skip if no valid move
        if delta_low >= delta_high:
            continue
            
        # Sample delta
        step = min(max_step, delta_high - delta_low)
        delta = np.random.uniform(delta_low, delta_high)
        
        # Apply move
        h[i] += delta
        h[j] -= delta
        
        return h
    
    # If 100 tries failed, return original
    return h

def anneal(initial_h, time_budget):
    """Run simulated annealing on the initial_h."""
    start = time.time()
    
    # Set up parameters
    T_start = 0.05
    T_end = 1e-6
    step_start = 0.2
    
    cur = initial_h.copy()
    best = initial_h.copy()
    
    c_cur = c5_of(cur)
    c_best = c_cur
    
    while time.time() - start < time_budget:
        # Update temperature and step size
        elapsed = time.time() - start
        t = elapsed / time_budget
        T = T_start * (T_end / T_start) ** t
        step = step_start * (1 - t)
        
        # Propose a move
        h_new = propose_move(cur, step)
        c_new = c5_of(h_new)
        
        # Accept or reject
        if c_new <= c_cur or np.random.uniform() < np.exp(-(c_new - c_cur) / T):
            cur = h_new
            c_cur = c_new
            
            # Update best
            if c_new < c_best:
                best = h_new
                c_best = c_new
                
    return best, c_best
