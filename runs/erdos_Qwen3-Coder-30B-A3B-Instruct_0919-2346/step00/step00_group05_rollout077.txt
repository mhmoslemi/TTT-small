def project_to_capped_simplex(h, target_sum):
    """Project h onto the capped simplex: 0 <= h[i] <= 1, sum(h) = target_sum."""
    h = np.array(h)

    if current_sum == target_sum:

        indices = np.argsort(h)[::-1]

        for i in indices:

                h[i] = 0
            else:
                if h[i] <= remaining:
                    remaining -= h[i]
                else:
                    h[i] = remaining
                    remaining = 0

        # Increase values
        indices = np.argsort(h)

            if remaining <= 0:
                break
            can_add = 1 - h[i]
            add = min(can_add, remaining)
            h[i] += add
            remaining -= add
    return h
