def evaluate_h(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n

    corr = np.correlate(h, 1 - h, mode='full')
    # The maximum overlap is the maximum of correlation values
    max_corr = np.max(corr)
    return max_corr * dx
