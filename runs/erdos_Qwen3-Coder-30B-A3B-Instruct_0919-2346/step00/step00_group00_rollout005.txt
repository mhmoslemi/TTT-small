import numpy as np
import time
from scipy import signal

def project_to_capped_simplex(h, target_sum):
    """Project h to the capped simplex: 0 <= h[i] <= 1 and sum(h) = target_sum."""
    h_clamped = np.clip(h, 0, 1)
    current_sum = np.sum(h_clamped)
    if abs(current_sum - target_sum) < 1e-10:
        return h_clamped
    
    # Water-filling approach
    sorted_h = np.sort(h_clamped)[::-1]
    cumulative = np.cumsum(sorted_h)
    
    # Find threshold index
    n = len(h_clamped)
    for i in range(n):
        if cumulative[i] <= target_sum - (n - 1 - i):
            continue
        else:
            break
    
    if i == n - 1:
        # All are above threshold
        lambda_val = (cumulative[i] - target_sum) / (n - 1 - i)
    else:
        lambda_val = (cumulative[i-1] - target_sum) / (n - i)
    
    h_proj = np.maximum(0, np.minimum(1, h_clamped - lambda_val))
    return h_proj

def compute_correlation(h, dx):
    """Compute the overlap integral as correlation."""
    corr = np.correlate(h, 1-h, mode="full")
    return dx * np.max(corr)

def compute_gradient(h, alpha, dx):
    """Compute gradient of log-sum-exp surrogate F_α(h)"""
    # Compute all overlaps O_ℓ(h)
    n = len(h)
    overlaps = np.zeros(2*n - 1)
    for k in range(1 - n, n):
        if k >= 0:
            overlap = dx * np.sum(h[k:] * (1 - h[:n-k]))
        else:
            overlap = dx * np.sum(h[:n+k] * (1 - h[-k:]))
        overlaps[k + n - 1] = overlap
    
    # Softmax weights w_ℓ
    exp_overlaps = np.exp(alpha * overlaps)
    w = exp_overlaps / np.sum(exp_overlaps)
    
    # First part: conv1d(w, 1-h) - use full mode
    c1 = signal.convolve(w, 1-h, mode='full')
    c1 = c1[n-1:2*n-1]  # Extract main part
    
    # Second part: conv1d(w_rev, h) - need to reverse w
    w_rev = w[::-1]
    c2 = signal.convolve(w_rev, h, mode='full')
    c2 = c2[n-1:2*n-1]
    
    grad = c1 - c2
    return grad
