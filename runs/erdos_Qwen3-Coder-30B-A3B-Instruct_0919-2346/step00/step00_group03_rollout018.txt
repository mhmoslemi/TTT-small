def evaluate_h(h):
    """Evaluate C5 = max_k ∫ h(x)(1 - h(x+k)) dx"""
    n = len(h)
    dx = 2.0 / n
    # Compute full correlation via FFT
    H = fft(h)
    G = fft(1 - h)
    corr = ifft(H * np.conj(G))  # This computes convolution h * (1-h)
    s_max = np.real(corr).max()  # Extract real part and find maximum
    return dx * s_max

def project_to_simplex(h, target_sum):
    """Project h onto the simplex with sum = target_sum"""
    # Ensure the sum is correct
    current_sum = np.sum(h)
    delta = (target_sum - current_sum) / len(h)
    h = h + delta
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Perform Dykstra's projection to maintain sum constraint
    for _ in range(5):
        current_sum = np.sum(h)
        delta = (target_sum - current_sum) / len(h)
        h = h + delta
        h = np.clip(h, 0, 1)
    return h

def optimize_continuous(n, budget_s):
