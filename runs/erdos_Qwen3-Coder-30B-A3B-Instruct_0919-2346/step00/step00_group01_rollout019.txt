```python
import time
import numpy as np
from scipy import fft
from cvxpy import *
import cvxpy as cp
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    max_time = budget_s
    n = 60
    dx = 2.0 / n
    K0 = [1, n//2, 2, n-2]
    K0 = sorted(set(K0))  # Ensure unique and sorted
    
    # Index table for cyclic indices
    idx_table = np.arange(n)
    
    def get_idx(i, k):
        return (i + k) % n
    
    def compute_auto_correlation(h_vals, k):
        """Compute the k-th cyclic autocorrelation."""
        h_shifted = np.roll(h_vals, k)
        return np.dot(h_vals, h_shifted)
    
    def compute_all_autocorrelations(h_vals):
        """Compute all cyclic autocorrelations using FFT."""
        h_fft = fft.fft(h_vals)
        auto_corr = fft.ifft(h_fft * np.conj(h_fft)).real
        # Reorder to match expected indexing
        # auto_corr[0] corresponds to k=0
        return auto_corr[:n]
    
    def compute_C5(h_vals):
        auto_corr = compute_all_autocorrelations(h_vals)
        min_corr = np.min(auto_corr)
        return 1.0 - (2.0 / n) * min_corr
    
    # Initial binary construction from initial_h_values if available
    initial_h = initial_h_values.copy()
    if len(initial_h) != n:
        h = np.random.choice([0, 1], size=n, p=[0.5, 0.5])
        h = h.astype(float)
    else:
        h = initial_h.copy()
        # Ensure it's binary and sum to n/2
        h = (h > 0.5).astype(float)
    
    # Convert to sum = n/2
    current_sum = np.sum(h)
    if current_sum != n/2:
        # Adjust by flipping some zeros to ones and vice versa
        diff = int((n/2) - current_sum)
        if diff > 0:
            # Add diff ones
            zeros = np.where(h == 0)[0]
            if len(zeros) >= diff:
                indices = np.random.choice(zeros, diff, replace=False)
                h[indices] = 1.0
            else:
                h[zeros] = 1.0
                # Fill up more if needed
                ones = np.where(h == 1)[0]
                other_indices = np.setdiff1d(np.arange(n), ones)
                idx = np.random.choice(other_indices, diff - len(zeros), replace=False)
                h[idx] = 1.0
        elif diff < 0:
            # Reduce diff ones
            ones = np.where(h == 1)[0]
            if len(ones) >= -diff:
                indices = np.random.choice(ones, -diff, replace=False)
                h[indices] = 0.0
            else:
                h[ones] = 0.0
                # Fill up with zeros if needed
                zeros = np.where(h == 0)[0]
                other_indices = np.setdiff1d(np.arange(n), zeros)
                idx = np.random.choice(other_indices, -diff - len(ones), replace=False)
                h[idx] = 0.0
    
    # Since we use more optimized construction, we now directly proceed to the cut-generation MILP
    # We now build a mixed-integer program tool to find a candidate
    
    # approximate approach using binary IP
    best_h = h.copy()
    best_C5 = compute_C5(best_h) if best_h is not None else float('inf')

    # For binary search using boolean variables:
    current_K = list(K0)
    t = cp.Variable(integer=True)
    w_vars = {}
    w_vars_list = []
    
    # Suggestions for constraint generation in cut loop
    used_shifts = set(current_K)
    
    # To avoid too many shifts, we limit number of iterations
    max_iterations = 20
    iteration_count = 0
    
    # Use a small time window to ensure some space left for polishing
    safety_margin = 20
    
    # The method below mimics the cut gathering, but we make sure execution is
    # within time budget.
    while iteration_count < max_iterations and (time.time() - start_time) < max_time - safety_margin:
        iteration_count += 1
        # Create model
        h_bin = cp.Variable(n, boolean=True)
        constraints = [cp.sum(h_bin) == n // 2]
        
        # Fix h_0 = 1 for symmetry breaking
        constraints.append(h_bin[0] == 1)
        
        # Variables for each auto-correlation
        A_vars = []
        auto_correlations = {}
        
        for k in current_K:
            w = cp.Variable(n, boolean=True)
            w_vars[(k, 'w')] = w
            w_vars_list.append(w)
            
            # McCormick envelopes for binary variables
            for i in range(n):
                idx = get_idx(i, k)
                constraints += [
                    w[i] <= h_bin[i],
                    w[i] <= h_bin[idx],
                    w[i] >= h_bin[i] + h_bin[idx] - 1
                ]
            a = cp.sum(w)
            auto_correlations[k] = a
            A_vars.append(a)
            
        # t <= A_k for all k
        for a in A_vars:
            constraints.append(t <= a)
            
        # Objective
        objective = cp.Maximize(t)
    
        # Build the problem
        prob = cp.Problem(objective, constraints)
        
        # Solve with timeout
        try:
            prob.solve(solver=cp.CBC, maxSeconds=30, verbose=False)
        except:
            break  # Unable to solve reliably
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_bin_sol = h_bin.value
            if h_bin_sol is not None:
                h_bin_sol = np.round(h_bin_sol).astype(int)
                h_current = h_bin_sol.astype(float)
            
                # update the best
                C5_current = compute_C5(h_current)
                if C5_current < best_C5:
                    best_C5 = C5_current
                    best_h = h_current.copy()

                # Evaluate all shifts to identify the next most violating one
                all_auto = compute_all_autocorrelations(h_current)
                min_idx = np.argmin(all_auto)
                min_val = all_auto[min_idx]
                
                if min_idx >= 1 and min_idx not in used_shifts:
                    used_shifts.add(min_idx)
                    current_K.append(min_idx)
                    
                # Should we keep going?
                if min_val >= t.value:
                    # Ensure all constraints satisfied
                    break
            else:
                break
        else:
            break

    # Final evaluation of best binary solution
    c5_bound = compute_C5(best_h)
    h_final = best_h.copy()

    # Optional polishing: gradient ascent
    remaining_time = max_time - (time.time() - start_time)
    if remaining_time > 50 and c5_bound > 0.379:
        # For polishing with continuous variables
        h_cont = h_final.copy()
        h_cont = np.clip(h_cont, 0, 1)
        # We'll do a few simple gradient steps
        for it in range(300):
            if time.time() - start_time > max_time - 10:
                break
            auto_corr = compute_all_autocorrelations(h_cont)
            k_min = np.argmin(auto_corr)
            # Compute gradient of sum over k * h(i) * h(i+k)
            dwi = np.zeros(n)
            for i in range(n):
                idx1 = (i + k_min) % n
                idx2 = (i - k_min) % n
                dwi[i] = h_cont[idx1] + h_cont[idx2]
            # update using small step
            step_size = 0.05
            h_cont += step_size * dwi
            h_cont = np.clip(h_cont, 0, 1)
            # Adjust to meet sum constraint
            diff = np.sum(h_cont) - (n / 2)
            if abs(diff) > 1e-8:
                # Use pairwise swap
                ones = np.where(h_cont > 0.5)[0]
                zeros = np.where(h_cont < 0.5)[0]
                if len(ones) > 0 and len(zeros) > 0:
                    z = np.random.choice(zeros)
                    o = np.random.choice(ones)
                    to_move = min(0.5, 0.5 + diff)
                    h_cont[z] += to_move
                    h_cont[o] -= to_move
                    h_cont = np.clip(h_cont, 0, 1)
            c5_new = compute_C5(h_cont)
            if c5_new < c5_bound:
                c5_bound = c5_new
                h_final = h_cont.copy()

    return (h_final, c5_bound, n)
```
