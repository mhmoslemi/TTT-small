import numpy as np
import time
from scipy import signal

def project_to_simplex(h, target_sum):
    """Project h onto the capped simplex: 0 <= h[i] <= 1 and sum(h) = target_sum."""
    n = len(h)
    # Clamp to [0,1]
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Water-filling approach
    sorted_h = np.sort(h)[::-1]
    cumsum = np.cumsum(sorted_h)
    
    # Find the threshold
    i = 0
    while i < n and sorted_h[i] > (target_sum - cumsum[i]) / (i + 1):
        i += 1
    
    # lambda is the amount to subtract
    if i == 0:
        lambda_ = (target_sum - cumsum[0]) / 1
    else:
        lambda_ = (target_sum - cumsum[i - 1]) / i if i > 0 else 0
    
    # Apply truncation
    result = np.maximum(0, h - lambda_)
    return result

def compute_overlaps(h, dx):
    """Compute all overlaps for shift k."""
    # Compute 1 - h for shift operations
    one_minus_h = 1.0 - h
    # Full correlation with mode='full' is O(n^2) so we'll batch compute it
    corr = np.correlate(h, one_minus_h, mode='full')
    # Only the valid shifts are needed
    # corr has 2*n - 1 elements, where corr[n-1]_corresponds to shift=0
    # Actual shifts range from -(n-1) to (n-1)
    # overlaps will correspond to shifts 1 to n-1 and 0 (same as corr[n-1])
    return corr

def compute_F_alpha(h, dx, alpha):
    """Compute the log-sum-exp surrogate F_alpha(h)."""
    corr = compute_overlaps(h, dx)
    # Extract the max shift range needed - this will depend on the lengths
    # Actually interpret this correctly
    # corr will be of length 2*n - 1.
    # corr[n - 1] corresponds to k = 0,
    # corr[n] to k = 1, corr[n-2] to k = -1, ...
    # But overlaps are for k = 0 up to 2(n-1) in the full space: 
    # shifts from - (n - 1) to (n - 1) 
    # Shift k = 0 is at index n - 1
    # k = 1 at index n  
    # k = -1 at index n - 2
    # k = +1 on full cfg we take slots from index n-1 to 2*(n-1) and n-1 to 0
    
    # Compute overlaps for shifts 0, 1, 2, ..., n-1
    overlaps = np.array([corr[n - 1 + k] for k in range(n)])  # k: 0 to n-1
    
    # Apply log-sum-exp to get the upper bound
    # F_α(h) = (dx/α) * log( Σ_ℓ exp( α·O_ℓ(h) ) )
    if alpha > 1e10:  # Numerical safety
        return dx * np.max(overlaps)
        
    max_overlaps = np.max(overlaps)
    exp_terms = np.exp(alpha * (overlaps - max_overlaps))
    log_sum = max_overlaps + np.log(np.sum(exp_terms)) / alpha  
    return dx * log_sum

def compute_gradient(h, dx, alpha, corr=None):
    """Compute gradient of F_alpha using convolutions."""
    if corr is None:
        corr = compute_overlaps(h, dx)
    
    n = len(h)
    
    # Construct overlaps from corr
    overlaps = np.array([corr[n - 1 + k] for k in range(n)])
    
    # Softmax of the overlaps unnormalized:
    # w_ℓ = exp( α·O_ℓ(h) ) / Σ_j exp( α·O_j(h) ) 
    if alpha > 1e10:
        # Use max soft approximation
        w = np.zeros_like(overlaps)
        max_idx = np.argmax(overlaps) 
        w[max_idx] = 1.0
    else: 
        exp_O = np.exp(alpha * overlaps)
        w = exp_O / np.sum(exp_O)
    
    # Gradient computation using two convolutions
        
    # Convolution for: Σ_ℓ w_ℓ * (1 – h[i+ℓ])
    # Operate on 1-h 
    one_minus_h = 1.0 - h
    
    #corr1 = np.correlate(w, one_minus_h, mode='full')  # This is wasting time 
    # We'll compute directly 
    # Let's make this explicit for element-wise vector form
    # It is easier to compute as:
    # ∂F_α/∂h[i] = Σ_ℓ w_ℓ * ( (1 – h[i+ℓ]) – h[i-ℓ] ) 
    # Let’s break it into two convolutions:
    
    # c1 = Σ_ℓ w_ℓ * (1 – h[i+ℓ]) 
    # This is convolution: w * (1 - h) with full mode.
    # Similar for c2 = Σ_ℓ w_ℓ * (h[i-ℓ])
    
    # Efficiency note: Using scipy.signal.correlate here can be done directly
    # actually the proper form is computing the full convolution, and we handle indices carefully
    
    # indexing:
    # With 1-h: full mode convolution of w with (1-h)
    c1 = signal.correlate(w, one_minus_h, mode='full')
    c1 = c1[n-1 : n-1 + n]  # Final n elements, where index i corresponds to h[i]
    
    # For c2 = Σ_ℓ w_ℓ * h[i-ℓ], this is convolution of w_rev with h
    w_rev = w[::-1]
    c2 = signal.correlate(w_rev, h, mode='full') 
    c2 = c2[n-1 : n-1 + n]
    
    grad = c1 - c2
    return grad

def round_to_binary(h, target_sum):
    """Deterministic rounding to binary with the right sum."""
    n = len(h)
    indices = np.argsort(h)[::-1]  # descending order of h[i]
    h_bin = np.zeros_like(h)
    ones_count = int(target_sum)
    h_bin[indices[:ones_count]] = 1.0
    return h_bin

def swap_refinement(h, dx):
    """Light hill-climbing flip of entries."""
    # Assumes h is binary 0 or 1
    corr = compute_overlaps(h, dx)
    n = len(h)
    max_existing = np.max(corr)
    
    attempts = 0
    max_attempts = 10000  # Total number of flips considered
    success_count = 0
    max_success = 1000  # stop early if no progress after many iterations 
    timeout = time.time() + 1.0  # 1 seconds
    before_corr = max_existing
    
    while success_count < max_success and attempts < max_attempts and time.time() < timeout:
        # Randomly pick one 1 and one 0
        ones = np.where(h == 1)[0] 
        zeros = np.where(h == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Exchange i and j, now evaluate (this is the underlying correlation update)
        old_sum = np.sum(1 - h)  # or equivalently sum of h
        
        new_corr = corr.copy() 
        # compute the change in overlap if we swap positions i and j.
        # The change in the correlation matrix when swapping positions i and j  
        # is due to the change in shifts that i and j participate in.
        
        # Both i and j participate in overlaps with shifts.
        # For each k with |k| <= n-1, we form the term h[i]*h[j] + h[j]*h[i] -h[i]*h[i] - h[j]*h[j] 
        # and that equals: 0 for the same positions - but compute for accounting shift pairs.
        # Better approach: to compute changes in overlap function directly.

        # We can recompute the overlaps for new h[i], h[j] values.
        h_new = h.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]
        
        corr_new = compute_overlaps(h_new, dx)
        max_new = np.max(corr_new)
        
        if max_new < max_existing:
            # accept swap
            h[:] = h_new[:]
            corr = corr_new
            max_existing = max_new
            success_count += 1
            if success_count % 100 == 0:
                # Periodic print
                pass
        attempts += 1
    return h, max_existing

        else:

            h_init = project_to_simplex(h_init, target_sum)
        
        # Annealing with deterministic pass
        h = h_init.copy()
        h = project_to_simplex(h, target_sum)
        
        alpha = 0.5  # Starting temperature
        step_size = 0.1  # step size for gradient descent
        gamma = 1.3  # Annealing factor
        TMAX = 200  # max temperature  
        adaptive_ratio = 1.
        prev_F = compute_F_alpha(h, dx, alpha)
        improvement = 1e10  # Arbitrary large number
        
        iter_count = 0
        max_iter_per_seed = 10000  # avoid infinite loop
        
        while alpha < TMAX and improvement > 1e-6 and iter_count < max_iter_per_seed and time.time() - start_time < budget_s - 5:
            # Update gradient
            grad = compute_gradient(h, dx, alpha)
            
            # Update h
            h = h - step_size * grad
            
            # Project back to feasible set: [0,1] and ∑h = target_sum
            h = project_to_simplex(h, target_sum)
            
            iter_count += 1
            
            # Periodic update
            if iter_count % 200 == 0:
                # Check for improvement
                curr_F = compute_F_alpha(h, dx, alpha)
                improvement = prev_F - curr_F
                prev_F = curr_F
                alpha *= gamma
                step_size = max(1e-5, step_size * 0.99)  # decay step size
                
                # Eval exact C5 if necessary
                if alpha > 100:
                    h_bin = round_to_binary(h, target_sum)
                    corr = compute_overlaps(h_bin, dx)

        h_bin = round_to_binary(h, target_sum)

        actual_c5 = np.max(corr)

            best_h = h_bin.copy()
            
    # Refine with swap hill climbing:
    if best_h is not None:
        best_h, refined_c5 = swap_refinement(best_h, dx)
        if refined_c5 < best_c5:
            best_c5 = refined_c5
