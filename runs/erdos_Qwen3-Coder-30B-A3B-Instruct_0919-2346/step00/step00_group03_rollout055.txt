def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant for given h"""
    n_points = len(h)

    # Compute correlation of h with (1-h) using full mode
    corr = np.correlate(h, 1 - h, mode='full')
    # Take maximum value scaled by dx
    max_corr = np.max(corr) * dx
    return max_corr
