import numpy as np
import time
import random

def evaluate_c5(h):
    """Evaluate the C5 bound for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h) using full mode
    corr = np.correlate(h, 1 - h, mode='full') * dx
    # Return maximum overlap
    return np.max(corr)

def run(seed=42, budget_s=600, **kwargs):
    # Set random seed
    np.random.seed(seed)
    random.seed(seed)
    
    # Parameters
    n_points = 200  # Using higher resolution for better chance of improvement
    dx = 2.0 / n_points
    target_c5 = 0.3807  # Target bound
    target_s = int(np.floor(target_c5 / dx))  # Target integer value for correlation
    
    # Initialize timing
    start_time = time.time()
    
    # Initialize best solution
    best_h = None
    best_c5 = float('inf')
    best_s = float('inf')
    
    # Run multiple restarts
    max_restarts = 50
    restart_count = 0
    
    while restart_count < max_restarts and (time.time() - start_time) < budget_s * 0.9:
        # Random binary initialization
        h = np.zeros(n_points)
        ones_indices = np.random.choice(n_points, n_points // 2, replace=False)
        h[ones_indices] = 1
        
        # Local improvement using hill climbing
        current_s = evaluate_c5(h)
        temp_s = int(np.ceil(current_s / dx))
        
        local_optimum = False
        local_iter = 0
        max_local_iter = 10000  # Limit local iterations
        
        while not local_optimum and local_iter < max_local_iter and (time.time() - start_time) < budget_s * 0.9:
            local_iter += 1
            
            # Find the shift with maximum mismatch
            corr = np.correlate(h, 1 - h, mode='full')
            max_index = np.argmax(corr)
            max_shift = max_index - (n_points - 1)
            max_corr = corr[max_index] / dx  # Convert back to continuous units
            
            # Early termination if target reached
            if max_corr <= target_c5:
                local_optimum = True  # Already good enough
                break
                
            # Determine which indices are mismatched at max_shift
            if max_shift >= 0:
                # Positive shift case
                mismatched_ones = np.where((h[:n_points - max_shift] == 1) & 
                                          (h[max_shift:] == 0))[0]
                matching_zeros = np.where((h[:n_points - max_shift] == 0) & 
                                         (h[max_shift:] == 1))[0]
            else:
                # Negative shift case
                max_shift = -max_shift
                mismatched_ones = np.where((h[max_shift:] == 1) & 
                                          (h[:n_points - max_shift] == 0))[0]
                matching_zeros = np.where((h[max_shift:] == 0) & 
                                         (h[:n_points - max_shift] == 1))[0]
                
            # If we have potential swaps, make one
            if len(mismatched_ones) > 0 and len(matching_zeros) > 0:
                # Pick random mismatched one and matching zero
                i = np.random.choice(mismatched_ones)
                j = np.random.choice(matching_zeros)
                
                # Attempt the swap
                old_h = h.copy()
                h[i], h[j] = h[j], h[i]
                
                new_s = evaluate_c5(h)
                new_s_int = int(np.ceil(new_s / dx))
                
                # Accept if improvement or probabilistically
                if new_s_int < temp_s or random.random() < 0.1:
                    temp_s = new_s_int
                else:
                    # Revert the swap
                    h[:] = old_h[:]
            else:
                # Random swap to escape local minimum
                i, j = np.random.choice(n_points, 2, replace=False)
                old_h = h.copy()
                h[i], h[j] = h[j], h[i]
                
                new_s = evaluate_c5(h)
                new_s_int = int(np.ceil(new_s / dx))
                
                if new_s_int < temp_s or random.random() < 0.01:
                    temp_s = new_s_int
                else:
                    h[:] = old_h[:]
        
        # Update best solution if current solution is better
        current_s = evaluate_c5(h)
        current_s_int = int(np.ceil(current_s / dx))
        if current_s_int < best_s:
            best_h = h.copy()
            best_s = current_s_int
            best_c5 = current_s
        
        restart_count += 1
    
    # Final evaluation of best solution
    final_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
