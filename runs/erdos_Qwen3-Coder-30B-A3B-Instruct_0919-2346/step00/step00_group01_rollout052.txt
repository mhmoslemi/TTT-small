def evaluate_c5(h):
    """Evaluate C5 for given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode='full')
    # Only consider the relevant part of the correlation (k from -1 to 1)
    c5_vals = corr[len(corr)//2:]
    c5 = np.max(c5_vals) * dx
    return c5
