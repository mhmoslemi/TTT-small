import numpy as np
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    best_n = None
    
    def c5_bound(h):
        n = len(h)
        dx = 2.0 / n
        O = np.correlate(h, 1 - h, mode='full')
        return O.max() * dx
    
    def random_binary(n):
        h = np.random.choice([0, 1], size=n, p=[0.5, 0.5])
        while np.sum(h) != n // 2:
            if np.sum(h) > n // 2:
                idx = np.where(h == 1)[0]
                np.random.shuffle(idx)
                h[idx[0]] = 0
            else:
                idx = np.where(h == 0)[0]
                np.random.shuffle(idx)
                h[idx[0]] = 1
        return h
    
    def thue_morse(n):
        if (n & (n - 1)) != 0:  # not power of two
            return random_binary(n)
        h = np.zeros(n, dtype=int)
        for i in range(n):
            h[i] = bin(i).count('1') & 1
        # adjust to exact weight
        current_weight = np.sum(h)
        target_weight = n // 2
        if current_weight > target_weight:
            idx = np.where(h == 1)[0]
            np.random.shuffle(idx)
            h[idx[:current_weight - target_weight]] = 0
        elif current_weight < target_weight:
            idx = np.where(h == 0)[0]
            np.random.shuffle(idx)
            h[idx[:target_weight - current_weight]] = 1
        return h
    
    def replicate_initial_h(n):
        if 'initial_h_values' not in globals():
            return random_binary(n)
        h_orig = np.array(initial_h_values, dtype=float)
        x_old = np.linspace(0, 2, len(h_orig), endpoint=False)
        x_new = np.linspace(0, 2, n, endpoint=False)
        h_interp = np.interp(x_new, x_old, h_orig)
        h_bin = (h_interp > 0.5).astype(int)
        # adjust weight
        current_weight = np.sum(h_bin)
        target_weight = n // 2
        if current_weight > target_weight:
            idx = np.where(h_bin == 1)[0]
            np.random.shuffle(idx)
            h_bin[idx[:current_weight - target_weight]] = 0
        elif current_weight < target_weight:
            idx = np.where(h_bin == 0)[0]
            np.random.shuffle(idx)
            h_bin[idx[:target_weight - current_weight]] = 1
        return h_bin
    
    # Handle multiple n values
    lengths = [64, 80, 96, 112, 128, 144, 160, 176, 192, 224, 256]
    lengths = [n for n in lengths if n <= 800]
    
    candidates = []
    for n in lengths:
        # Thue-Morse
        if (n & (n - 1)) == 0:  # power of two
            seq = thue_morse(n)
        else:
            seq = random_binary(n)
        candidates.append((seq, n))
        
        # Replicate initial pattern
        seq = replicate_initial_h(n)
        candidates.append((seq, n))
        
        # Random
        for _ in range(2):
            seq = random_binary(n)
            candidates.append((seq, n))
    
    # Evaluate candidates
    for h, n in candidates:
        if time.time() - start_time > 0.95 * budget_s:
            break
        c5_val = c5_bound(h)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
            best_n = n
    
    # Local search for top candidates (simplified version)
    top_candidates = []
    for h, n in candidates[:5]:  # top few candidates based on preliminary eval
        if time.time() - start_time > 0.95 * budget_s:
            break
        top_candidates.append((h.copy(), n))
    
    # Produce search footprint and target improvement
    time_per_candidate = (budget_s * 0.95) / max(1, len(top_candidates))
    for h, n in top_candidates:
        if time.time() - start_time > 0.95 * budget_s:
            break
        # First time eval (if new)
        current_c5 = c5_bound(h)  
        best_c5_local = current_c5
        best_h_local = h.copy()
        
        # Simple local search with limited iterations
        max_iterations = 100000
        temp_start = 0.1
        temp_end = 1e-5
        temp_schedule = lambda it, max_iter: temp_start * (temp_end / temp_start) ** (it / max_iter)
        
        for iteration in range(max_iterations):
            if time.time() - start_time > 0.95 * budget_s:
                break
                
            temp = temp_schedule(iteration, max_iterations)
            
            # Find ones and zeros
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            
            if len(ones) == 0 or len(zeros) == 0:
                continue
                
            # Pick two random positions to swap
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            
            # Swap
            h[i], h[j] = 0, 1
            
            # Evaluate new bound
            new_c5 = c5_bound(h)
            
            # Accept or reject
            delta = new_c5 - current_c5
            
            if delta <= 0 or np.random.rand() < np.exp(-delta / temp):
                current_c5 = new_c5
                if current_c5 < best_c5_local:
                    best_c5_local = current_c5
                    best_h_local = h.copy()
            else:
                # Revert swap
                h[i], h[j] = 1, 0
                
        if best_c5_local < best_c5:
            best_c5 = best_c5_local
            best_h = best_h_local.copy()
            best_n = n
            
    return (best_h.tolist(), best_c5, best_n)
