def evaluate_c5(h):
    """Evaluate the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h) using full mode
    conv = np.correlate(h, 1 - h, mode="full")
    # Multiply by dx to account for discretization
    conv *= dx
    # Return maximum value
    return np.max(conv)
