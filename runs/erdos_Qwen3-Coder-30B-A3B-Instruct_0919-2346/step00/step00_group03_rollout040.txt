import numpy as np
import time
import random

def evaluate_c5(h):
    # Compute full correlation of h with (1-h) using numpy correlate
    conv = np.correlate(h, 1 - h, mode="full")
    # Multiply by dx = 2/n to get proper scaling
    dx = 2.0 / len(h)
    conv *= dx
    # Return maximum value of correlation over all lags (including negative)
    return np.max(conv)

def random_binary_seq(n):
    # Create a random binary sequence of length n with exactly n/2 ones
    h = np.zeros(n, dtype=float)
    indices = np.random.choice(n, size=n//2, replace=False)
    h[indices] = 1.0
    return h

def solve_binary(n, max_iter, init_h=None, seed=None):
    if seed is not None:
        np.random.seed(seed)
        random.seed(seed)
    
    # Initialize h: if no initial, create random binary
    if init_h is not None:
        h = init_h.copy()
    else:
        h = random_binary_seq(n)
    
    c5 = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = c5
    
    # Cooling schedule parameters
    T0 = 0.1
    T = T0
    cooling_rate = 0.9999
    
    for iteration in range(max_iter):
        # Choose two random indices
        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        
        if i == j:
            continue
            
        # Save old values
        old_h_i = h[i]
        old_h_j = h[j]
        
        # Swap h[i] and h[j]
        h[i], h[j] = h[j], h[i]
        
        # Compute new c5 value
        new_c5 = evaluate_c5(h)
        
        # Accept with probability based on change in c5 and current temperature T
        delta = new_c5 - c5
        if delta <= 0 or np.random.rand() < np.exp(-delta / T):
            # Accept the move
            c5 = new_c5
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            # Reject the move, revert
            h[i], h[j] = old_h_i, old_h_j
            
        # Cool down temperature
        T *= cooling_rate
        
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    # Start timing
    start_time = time.time()
    
    # Set parameters
    n_points = 200  # Use 200 points for better resolution
    
    # Initialize best solution
    best_h = None
    best_c5 = float('inf')
    
    # Number of restarts we could do within budget
    # Approximate time settings: 
    # For each restart for 10000 iterations (including overhead): ~ 0.1-0.2 seconds
    # Budget: 600 seconds
    # Let's allow multiple runs around 200 restarts with 10000 iterations each
    # But to be safe, we'll do ~200 Restart, for each 10000 steps.
    # Would be about 20k steps total; 20k * 0.0001 = 2 seconds... But let's make
    # a safe budget of 80% of time for polling and checking.
    
    # Go for 150 restarts of 2000 iterations each
    total_iterations = 150 * 2000  # 300k
    step_per_restart = 2000
    
    increments = 0
    
    while time.time() - start_time < budget_s * 0.95:
        # Choose random seed for this restart
        restart_seed = seed + increments
        
        # Run optimization on this restart
        h, c5 = solve_binary(n_points, step_per_restart, init_h=None, seed=restart_seed)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        increments += 1
    
    # Optionally try a local refinement with continuous variables around best binary
    # This part uses gradient ascent to make small adjustments, but we'll do simple refinement.
    # Purpose: possibly fine-tune best binary solution with small continuous tweaks
    
    if best_h is not None:
        # Continuous refinement
        max_iter_refinement = 5000
        # Refine with gradients
        h_refined = best_h.copy()
        c5_refined = evaluate_c5(h_refined)
        
        T_refine = 0.01
        cooling_refine = 0.999
        
        for i in range(max_iter_refinement):
            # Make small random changes to h_refined maintaining sum constraint
            # Pick two indices, slightly adjust one up, other down, keep sum fixed
            idx1 = np.random.randint(0, n_points)
            idx2 = np.random.randint(0, n_points)
            if idx1 == idx2:
                continue
                
            change = np.random.uniform(-0.03, 0.03)
            # Ensure no overflow
            if h_refined[idx1] + change > 1:
                change = 1 - h_refined[idx1]
            elif h_refined[idx1] + change < 0:
                change = -h_refined[idx1]
                
            h_refined[idx1] += change
            h_refined[idx2] -= change
            
            # Clip values to boundaries
            h_refined[idx1] = np.clip(h_refined[idx1], 0, 1)
            h_refined[idx2] = np.clip(h_refined[idx2], 0, 1)
            
            # Re-normalize to exactly sum = n/2
            true_sum = np.sum(h_refined)
            diff = true_sum - n_points / 2
            if abs(diff) > 1e-8:
                # Adjust by subtracting average offset
                offset = diff / n_points
                h_refined -= offset
                h_refined = np.clip(h_refined, 0, 1)
                
            new_c5 = evaluate_c5(h_refined)
            
            if new_c5 < c5_refined:
                c5_refined = new_c5
            else:
                # Reject if dangerous, perhaps accept probabilistically with temperature-based method
                # For this simplified version, just keep previous good value
                
            # Cool down
            T_refine *= cooling_refine
        
        # Use refined version if better
        if c5_refined < best_c5:
            best_c5 = c5_refined
            best_h = h_refined.copy()
    
    # Return the best found solution
    return (best_h.tolist(), best_c5, n_points)
