import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import time
import random

def evaluate_erdos_solution(h_values):
    """Evaluate the current h and return C5"""
    n = len(h_values)
    dx = 2.0 / n
    # compute overlap integrals for all shifts
    c = np.correlate(h_values, 1 - h_values, mode='full')
    C5 = np.max(c) * dx
    return C5

def initial_h_values():
    """Initial construction, a good starting guess"""
    n = 70
    h = np.zeros(n)
    # fill with a step pattern, approximating the initial construction from the paper
    h[:35] = 1
    return h

def project_to_capped_simplex(h, target_sum):
    """Project vector h onto the simplex with sum = target_sum, clipping to [0,1]"""
    h = np.array(h, dtype=float)
    # clamp to [0,1]
    h = np.clip(h, 0, 1)
    s = np.sum(h)
    if s == target_sum:
        return h
    elif s < target_sum:
        # Need to distribute more mass
        delta = target_sum - s
        # Add evenly to entries with least value
        idxs = np.argsort(h)
        h[idxs] = np.clip(h[idxs] + delta / len(h), 0, 1)
    else:
        # Need to remove mass
        delta = s - target_sum
        # Remove evenly from entries with largest value
        idxs = np.argsort(h)[::-1]
        h[idxs] = np.clip(h[idxs] - delta / len(h), 0, 1)
    return h

def random_binary_h(n, target_sum):
    """Generate a random binary vector with exactly target_sum ones."""
    h = np.zeros(n)
    indices = np.random.choice(n, target_sum, replace=False)
    h[indices] = 1
    return h

def get_autocorr_at_shift(h, k):
    """Get Σ_i h[i]·h[i+k] for a given shift k."""
    n = len(h)
    if k >= n:
        return 0
    return np.sum(h[:n-k] * h[k:])

def improve_with_local_swaps(h):
    """Perform local swaps to reduce C5 by flipping optimally."""
    n = len(h)
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h)
    
    # Try different shifts
    for k in range(1, n):
        # for each pair (i, i+k) such that h[i]=1 and h[i+k]=0, swap
        for i in range(n-k):
            if h[i] == 1 and h[i+k] == 0:
                # make swap
                h[i] = 0
                h[i+k] = 1
                new_C5 = evaluate_erdos_solution(h)
                if new_C5 < best_C5:
                    best_C5 = new_C5
                    best_h = h.copy()
                # revert back
                h[i] = 1
                h[i+k] = 0
    return best_h, best_C5

def solve_milp(h_init, n_points, max_seconds):
    # solver parameters
    solver = cp.CBC
    # This is done for clarity but the full conditional isn't needed in simplify
    # We always solve in exact integer mode but we limit the solution space through symmetry breaking
    # So all variables are integers here, hence it's a straight MILP
    
    # variables
    h = cp.Variable(n_points, integer=True)
    # new τ variable: maximum that all autocorrelation sums are ≥ τ
    tau = cp.Variable()
    
    constraints = []
    
    # initial bounds on h
    constraints += [0 <= h[i] for i in range(n_points)]
    constraints += [h[i] <= 1 for i in range(n_points)]
    
    # sum constraint
    constraints += [cp.sum(h) == n_points // 2]
    # symmetry breaking
    constraints += [h[0] == 1]
    
    # auxiliary variables for y[i,k] = h[i]*h[i+k]
    y_vars = {}
    y_constraints = []
    
    # for each non-negative shift k
    for k in range(1, n_points):
        # Create vector (n-k) long for that shift
        y_vec = cp.Variable(n_points - k, integer=True)
        y_vars[k] = y_vec
        
        # Set up y[i,k] constraints
        for i in range(n_points - k):
            # y[i,k] >= 0
            y_constraints.append(y_vec[i] >= 0)
            # y[i,k] <= h[i]
            y_constraints.append(y_vec[i] <= h[i])
            # y[i,k] <= h[i+k]
            y_constraints.append(y_vec[i] <= h[i+k])
            # y[i,k] >= h[i] + h[i+k] - 1
            y_constraints.append(y_vec[i] >= h[i] + h[i+k] - 1)
        
        # now add constraint: tau <= sum of y_vec
        y_constraints.append(tau <= cp.sum(y_vec))
    
    constraints += y_constraints

    # objective to maximize τ, i.e., minimize -τ
    prob = cp.Problem(cp.Maximize(tau), constraints)
    
    # solve within time limit
    start_time = time.time()
    try:
        prob.solve(solver=solver, time_limit=max_seconds)
        end_time = time.time()
        print(f"Solved in {end_time - start_time:.2f} seconds")
    except Exception as e:
        print(f"Solver crashed: {e}")
        return h_init, np.inf

    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        sol_h = np.array(h.value).round().astype(int)
        # It is still an integer solution
        # check if it is feasible (only global constraints)
        if np.sum(sol_h) == n_points // 2 and sol_h[0] == 1:
            new_C5 = evaluate_erdos_solution(sol_h)
            return sol_h, new_C5
        else:
            # fix to known feasible
            print("Failing to satisfy global constraints post-solve; using fallback.")
            return h_init, evaluate_erdos_solution(h_init)
    else:
        print("Solver done, problem is infeasible or another issue.")
        return h_init, evaluate_erdos_solution(h_init)

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()
    best_h = None
    best_C5 = np.inf
    n_points = 70
    target_sum = n_points // 2

    h_base = initial_h_values()
    h_base = project_to_capped_simplex(h_base, target_sum)

    print("Starting base solution C5:", evaluate_erdos_solution(h_base))
    best_h = h_base.copy()
    best_C5 = evaluate_erdos_solution(h_base)

    while time.time() - start_time < budget_s * 0.7:
        # Try random binary start
        h_rand = random_binary_h(n_points, target_sum)
        h_rand = project_to_capped_simplex(h_rand, target_sum)
        h_sol, C5_sol = solve_milp(h_rand, n_points, budget_s - (time.time() - start_time))
        if C5_sol < best_C5:
            best_C5 = C5_sol
            best_h = h_sol.copy()
            print(f"Best found: {best_C5:.6f}, with {time.time() - start_time:.1f}s")
        
        # Try using continuous solution as warm start
        # Solve without integrality first (LP relaxation)
        h_relaxed = np.array(h_base, dtype=float)
        try:
            # This requires to make a method that solves LP version, but seems tightly related 
            # We continue with current method - rely on heuristics from best frequent restarts.
        except:
            pass

    # refine with local swaps if we found at least one binary solution
    if best_h is not None and all(x in [0,1] for x in best_h):
        best_h, refined_C5 = improve_with_local_swaps(best_h)
        if refined_C5 < best_C5:
            best_C5 = refined_C5

    return (best_h, best_C5, n_points)
