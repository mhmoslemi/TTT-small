import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for a given step function h"""
    # h is a numpy array of values in [0,1]
    # We compute the maximum overlap integral over all shifts k
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h) using full mode
    # This effectively computes the integral for all possible shifts
    correlation = np.correlate(h, 1-h, mode="full")
    # The maximum value gives us the worst-case overlap
    max_overlap = np.max(correlation)
    c5 = max_overlap * dx
    return c5

def initial_h_values():
    """Return the initial construction for h"""
    # Return the initial construction from the problem
    # This is a 70 point initial guess
    n = 70
    h = np.zeros(n)
    # Create a simple construction that satisfies constraints
    # Use a step function: 0.5 on the first 35 points, 0.5 on the last 35 points
    # But more careful construction to satisfy ∫ h(x) dx = 1
    h[:35] = 1.0
    h[35:] = 0.0
    return h

def objective(h):
    """Objective to minimize: C5 value"""
    return evaluate_erdos_solution(h)

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function to find improved C5 bound"""
    start_time = time.time()
    
    # Use a reasonable number of points for optimization
    n_points = 70  # Start with the provided starting construction
    
    # Initialize with a simple construction
    h_init = initial_h_values()
    h_init = h_init.copy()
    
    # Ensure it satisfies the integral constraint
    dx = 2.0 / n_points
    current_sum = np.sum(h_init)
    h_init = h_init * (n_points / 2.0) / current_sum
    
    # Define bounds for h[i] in [0, 1]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization with multiple starting points to escape local minima
    best_h = h_init.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Time budget for optimization
    elapsed = time.time() - start_time
    
    # Use differential evolution for global optimization
    try:
        result = differential_evolution(
            objective,
            bounds,
            seed=seed,
            maxiter=int(1000 * (budget_s / 600.0)),
            disp=False,
            tol=1e-6,
            mutation=(0.5, 1.0),
            recombination=0.7,
            strategy='best1bin'
        )
        
        if result.success:
            # Check if the result is better (lower) than current best
            h_result = result.x
            h_result = np.clip(h_result, 0, 1)
            # Enforce the integral constraint
            current_sum = np.sum(h_result) * dx
            if current_sum != 0:
                h_result = h_result * (n_points / 2.0) / current_sum
            
            c5_result = evaluate_erdos_solution(h_result)
            if c5_result < best_c5:
                best_c5 = c5_result
                best_h = h_result.copy()
    except:
        pass
    
    # Try local optimization on best so far if time remains
    remaining_time = budget_s - elapsed
    if remaining_time > 1.0:
        try:
            # Try different starting points for more exploration
            for trial in range(5):
                # Generate random start point
                h_trial = np.random.rand(n_points)
                h_trial = np.clip(h_trial, 0, 1)
                # Enforce integral constraint
                current_sum = np.sum(h_trial) * dx
                if current_sum != 0:
                    h_trial = h_trial * (n_points / 2.0) / current_sum
                
                # Local minimization using L-BFGS-B
                result = minimize(
                    objective,
                    h_trial,
                    method='L-BFGS-B',
                    bounds=bounds,
                    tol=1e-6,
                    options={'maxiter': 500}
                )
                
                if result.success:
                    h_test = result.x
                    h_test = np.clip(h_test, 0, 1)
                    # Enforce integral constraint
                    current_sum = np.sum(h_test) * dx
                    if current_sum != 0:
                        h_test = h_test * (n_points / 2.0) / current_sum
                    
                    c5_test = evaluate_erdos_solution(h_test)
                    if c5_test < best_c5:
                        best_c5 = c5_test
                        best_h = h_test.copy()
                        
        except:
            pass
    
    # Final check and return
    return (best_h, best_c5, n_points)
