import numpy as np
import time
from scipy.signal import convolve

def project_to_capped_simplex(h, target_sum):
    """Project h onto the capped simplex with sum = target_sum."""
    n = len(h)
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Water-filling approach
    # Sort in descending order
    indices = np.argsort(h)[::-1]
    h_sorted = h[indices]
    
    # Find where to cut
    cumsum = np.cumsum(h_sorted)
    for i in range(n):
        if i == 0:
            sum_i = h_sorted[i]
        else:
            sum_i = cumsum[i-1] + h_sorted[i]
        if sum_i + (n - i - 1) >= target_sum:
            break
    else:
        # Should not happen if h is feasible
        return h
    
    threshold = (target_sum - sum_i + h_sorted[i]) / (n - i)
    h_result = np.zeros_like(h)
    for j in range(i):
        h_result[indices[j]] = 1
    for j in range(i, n):
        h_result[indices[j]] = np.clip(h_sorted[j] - threshold, 0, 1)
    
    return h_result

def compute_gradient(h, alpha, dx):
    """Compute the gradient of F_alpha using convolutions."""
    n = len(h)
    # Compute O_l(h) for all l
    O = np.zeros(2 * n - 1)
    for l in range(-n + 1, n):
        if l >= 0:
            temp = h[l:] * (1 - h[:n - l])
        else:
            temp = h[:n + l] * (1 - h[-l:])
        O[l + n - 1] = np.sum(temp)
    
    # Compute soft-max weights w_l
    exp_O = np.exp(alpha * O)
    w = exp_O / np.sum(exp_O)
    
    # Compute c1 = conv(w, 1-h) and c2 = conv(w_rev, h)
    c1 = convolve(w, 1 - h, mode='full')
    c2 = convolve(w[::-1], h, mode='full')
    
    # Extract relevant part
    grad = c1[n - 1:n - 1 + n] - c2[n - 1:n - 1 + n]
    
    return grad

def compute_F_alpha(h, alpha, dx):
    """Compute the surrogate F_alpha(h)."""
    n = len(h)
    # Compute all O_l(h)
    O = np.zeros(2 * n - 1)
    for l in range(-n + 1, n):
        if l >= 0:
            temp = h[l:] * (1 - h[:n - l])
        else:
            temp = h[:n + l] * (1 - h[-l:])
        O[l + n - 1] = np.sum(temp)
    
    # Compute F_alpha using logsumexp
    exp_O = np.exp(alpha * O)
    log_sum_exp = np.log(np.sum(exp_O))
    return (dx / alpha) * log_sum_exp

def evaluate_c5(h, dx):
    """Evaluate true C5 using np.correlate."""
    corr = np.correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def initial_h(n):
    """Initialize h from the global initial_h_values or start from flat."""
    try:
        if len(initial_h_values) == n:
            return np.array(initial_h_values)
        else:
            # interpolate if needed
            y = np.array(initial_h_values)
            x_old = np.linspace(0, 2, len(y))
            x_new = np.linspace(0, 2, n)
            y_new = np.interp(x_new, x_old, y)
            return y_new
    except NameError:
        # Return flat function with sum = n/2
        return np.ones(n) * 0.5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n = 200  # even number
    dx = 2.0 / n
    target_sum = n / 2  # required sum
    
    # Multi-start: run with different initializations
    best_c5 = float('inf')
    best_h_binary = None
    
    # Seeds to try
    seeds = [seed] + list(range(1, min(8, budget_s // 50 + 1)))
    start_time = time.time()
    
    for s in seeds:
        np.random.seed(s)
        
        # Initialize h
        h = initial_h(n)
        # Ensure we clip initially to [0,1]
        h = np.clip(h, 0, 1)
        # Then project to valid sum
        h = project_to_capped_simplex(h, target_sum)
        
        # Annealing parameters
        alpha = 0.5
        gamma = 1.3
        eta = 0.1
        eta_decay = 0.99
        max_alpha = 200.0
        min_improvement = 1e-5
        
        last_F_alpha = float('inf')
        last_alpha = 0.0
        
        iter_count = 0
        while alpha <= max_alpha and time.time() - start_time < budget_s - 5:
            # Compute the gradient
            grad = compute_gradient(h, alpha, dx)
            
            # Gradient step
            h = h - eta * grad
            h = project_to_capped_simplex(h, target_sum)
            
            # Update eta
            eta *= eta_decay
            
            iter_count += 1
            if iter_count % 200 == 0:
                F_alpha = compute_F_alpha(h, alpha, dx)
                improvement = abs(last_F_alpha - F_alpha)
                if improvement < min_improvement and alpha > last_alpha:
                    break
                last_F_alpha = F_alpha
                last_alpha = alpha
                alpha *= gamma
                
        # After annealing, round to binary
        h_sorted_indices = np.argsort(h)[::-1]
        h_binary = np.zeros(n)
        h_binary[h_sorted_indices[:n//2]] = 1.0
        
        # Evaluate exact C5
        c5 = evaluate_c5(h_binary, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h_binary = h_binary.copy()
    
    # Local hill-climbing refinement
    if best_h_binary is not None and time.time() - start_time < budget_s - 5:
        h_refined = best_h_binary.copy()
        corr = np.correlate(h_refined, 1 - h_refined, mode='full')
        current_max = np.max(corr)
        improvement_threshold = 1e-8
        
        swaps_tried = 0
        max_swaps = 10000
        
        while swaps_tried < max_swaps and time.time() - start_time < budget_s - 5:
            indices1 = np.where(h_refined == 1)[0]
            indices0 = np.where(h_refined == 0)[0]
            
            if len(indices1) == 0 or len(indices0) == 0:
                break
                
            i = np.random.choice(indices1)
            j = np.random.choice(indices0)
            
            # Compute change in correlation if swap occurs
            old_corr = np.zeros(2 * n - 1)
            old_corr[:] = corr
            
            # Inefficient way to compute difference due to swap at positions i,j
            # Make a change in the set of ones z (only the indices w.r.t to shifted
            # convolution)  
            
            # Optimized transaction for swapping:
            # Manually recompute correlation using O(n) method
            # This is easier but rather inefficient;
            # Do incremental update carefully:
            # We modify the inner product by replacing h[i] and h[j]
            # This affects correlation values for shifts that are close to {i,j}
            # So we can calculate only those
            
            # Find correlation elements affected by i -> 0 and j -> 1
            # For shift 0:
            # We lose a contribution from h[i] * (1 - h[j])
            # We gain a contribution from h[j] * (1 - h[i])
            # For a shift l:
            # Old contribution to corr[l] from indices i, i+l and j, j+l
            
            # The simple direct method:
            temp_h = h_refined.copy()
            temp_h[i] = 0
            temp_h[j] = 1
            new_corr = np.correlate(temp_h, 1 - temp_h, mode='full')
            new_max = np.max(new_corr)
            
            if new_max < current_max:
                # Accept the swap
                h_refined = temp_h
                corr = new_corr
                current_max = new_max
                swaps_tried += 1
            else:
                swaps_tried += 1
        
        # Final evaluation
        corr = np.correlate(h_refined, 1 - h_refined, mode='full')
        final_c5 = dx * np.max(corr)
    
        return (h_refined, final_c5, n)
    
    if best_h_binary is not None:
        corr = np.correlate(best_h_binary, 1 - best_h_binary, mode='full')
        final_c5 = dx * np.max(corr)
        return (best_h_binary, final_c5, n)
    
    # Fallback to base construction if no good one found
    try:
        h_init = initial_h(n)
        corr = np.correlate(h_init, 1 - h_init, mode='full')
        final_c5 = dx * np.max(corr)
        return (h_init, final_c5, n)
    except Exception:
        # Return a pseudo random pattern with tight sum
        h_rand = np.random.rand(n)
        h_rand = project_to_capped_simplex(h_rand, target_sum)
        corr = np.correlate(h_rand, 1 - h_rand, mode='full')
        final_c5 = dx * np.max(corr)
        return (h_rand, final_c5, n)
