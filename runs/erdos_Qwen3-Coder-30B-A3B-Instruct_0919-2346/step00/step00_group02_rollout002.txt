import numpy as np
import time
import random

def c5_of(h):
    """Compute C5 = max_k ∫ h(x)(1 - h(x+k)) dx"""
    dx = 2.0 / len(h)
    # correlation of h with (1-h), full mode
    c = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(c)

def propose_move(h, max_step):
    """Propose a new feasible h by adjusting two indices"""
    h_new = h.copy()
    N = len(h)
    # sample two different indices
    i, j = random.sample(range(N), 2)
    
    # compute feasible range for δ
    low = max(-h_new[i], h_new[j] - 1)
    high = min(1 - h_new[i], h_new[j])
    
    if low >= high:
        # can't move, try another pair
        return h_new
    
    # sample δ in [max(-max_step, low), min(max_step, high)]
    step = min(max_step, high - low)
    delta = random.uniform(-step, step)
    
    h_new[i] += delta
    h_new[j] -= delta
    
    return h_new

def anneal(initial_h, time_budget):
    """Run simulated annealing from initial_h"""
    start = time.time()
    
    cur = initial_h.copy()
    c_cur = c5_of(cur)
    
    best = cur.copy()
    c_best = c_cur
    
    # parameters
    T_start = 0.05
    T_end = 1e-6
    step_start = 0.2
    
    # time-distributed cooling schedule
    while time.time() - start < time_budget:
        t = time.time() - start
        T = T_start * (T_end / T_start) ** (t / time_budget)
        step = step_start * (1 - t / time_budget)
        
        # propose a move
        h_new = propose_move(cur, step)
        c_new = c5_of(h_new)
        
        # acceptance probability
        if c_new <= c_cur or random.random() < np.exp(-(c_new - c_cur) / T):
            cur = h_new
            c_cur = c_new
            
            if c_new < c_best:
                best = cur.copy()
                c_best = c_new
    
    return best, c_best
