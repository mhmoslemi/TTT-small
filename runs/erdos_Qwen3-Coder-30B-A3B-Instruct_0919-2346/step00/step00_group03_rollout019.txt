def evaluate_overlap(h):
    """Evaluate the overlap C₅ using FFT."""
    n = len(h)
    dx = 2.0 / n
    # Compute full correlation: h * (1 - h) with full mode
    H = fft(h)
    G = fft(1 - h)
    corr = ifft(H * np.conj(G)).real
    max_overlap = np.max(corr)
    return max_overlap * dx

def project_to_feasible(h, target_sum):
    """Project vector h onto the simplex with sum = target_sum."""
    h = np.array(h)
    delta = (np.sum(h) - target_sum) / len(h)
    h = h - delta
    # Dykstra's projection for box constraints [0,1]
    for _ in range(5):
        h = np.clip(h, 0, 1)
        old_sum = np.sum(h)
        delta = (old_sum - target_sum) / len(h)
        h = h - delta
    return h.tolist()

def compute_mismatches(h, shift):
    """Identify mismatched pairs for a given shift."""
    n = len(h)
    i_start = 0
    i_end = n - abs(shift)
    if shift < 0:
        i_end = n + shift
    else:
        i_start = shift
    # Mismatch: h[i]=1 and h[i+shift]=0, or vice versa
    M1 = []
    M0 = []
    for i in range(i_start, i_end):
        if h[i] >= 0.99 and h[i+shift] <= 0.01:
            M1.append(i)
        elif h[i] <= 0.01 and h[i+shift] >= 0.99:
            M0.append(i)
    return M1, M0

def swap_mismatches(h, s_max, target_bound):
    """Perform mismatch swaps to reduce maximum overlap."""
    n = len(h)
    dx = 2.0 / n
    # Find shift corresponding to max overlap
    H = fft(h)
    G = fft(1 - h)
    corr = ifft(H * np.conj(G)).real
    max_shift = np.argmax(corr)
    if max_shift > n // 2:
        max_shift = max_shift - n  # Convert to signed shift
    max_overlap = corr[max_shift + n // 2]  # Center index shift
    # Swap pairs that reduce the mismatch
    M1, M0 = compute_mismatches(h, max_shift)
    old_corr = np.copy(corr)
    count = 0
    while M1 and M0 and corr[max_shift + n // 2] > target_bound / dx and count < 5000:
        i = M1.pop(0)
        j = M0.pop(0)
        # Perform the swap and adjust
        h[i], h[j] = h[j], h[i]
        # Project to ensure sum constraint and box constraints
        h = project_to_feasible(h, n / 2)
        # Recompute correlation
        H = fft(h)
        G = fft(1 - h)
        new_corr = ifft(H * np.conj(G)).real
        if new_corr[max_shift + n // 2] >= max_overlap:
            # Bad swap, undo and try again
            h[i], h[j] = h[j], h[i]
            h = project_to_feasible(h, n / 2)
        else:
            max_overlap = new_corr[max_shift + n // 2]
            # Update maps
            M1, M0 = compute_mismatches(h, max_shift)
        count += 1
    return h
