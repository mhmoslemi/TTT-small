import numpy as np
import time
import random

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return corr.max()

def propose_swap_and_update(h, corr, i, j):
    old_i = h[i]
    old_j = h[j]
    new_i = old_j
    new_j = old_i

    delta_i = new_i - old_i

    n = len(h)
    new_corr = corr.copy()
    
    # loop over k not i,j
    for k in range(n):
        if k == i or k == j:
            continue
        hk = h[k]
        # shift i->k: l = k - i
        idx1 = k - i + n - 1
        new_corr[idx1] += delta_i * (1 - hk)
        # shift k->i: l = i - k
        idx2 = i - k + n - 1
        new_corr[idx2] += hk * (-delta_i)

        # contributions for j
        # shift j->k: l = k - j
        idx3 = k - j + n - 1
        new_corr[idx3] += -delta_i * (1 - hk)
        # shift k->j: l = j - k
        idx4 = j - k + n - 1
        new_corr[idx4] += hk * (delta_i)

    # handle cross pair i->j and j->i
    idx_ij = j - i + n - 1
    new_corr[idx_ij] += delta_i

    idx_ji = i - j + n - 1
    new_corr[idx_ji] -= delta_i

    new_max = new_corr.max()
    return new_corr, new_max

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()
    
    best_c5 = np.inf
    best_h = None
    best_n = None
    
    # Use n = 180 as it's acceptable for time not too large
    n = 180
    if n % 2 != 0:
        n -= 1
    
    ones_count = n // 2
    max_iters = 200000
    
    # Use existing initial_h_values if it's roughly size 45, else random uniform
    try:
        from __main__ import initial_h_values
        if len(initial_h_values) == 45 and np.isclose(np.sum(initial_h_values), 22.5):
            h = initial_h_values.copy()
            if len(h) != n:
                # Upscale by replicating
                prev_len = len(h)
                h = np.tile(h, int(np.ceil(n / prev_len)))
                h = h[:n]
                # Adjust to match sum
                current_sum = np.sum(h)
                excess = current_sum - ones_count
                # Adjust by changing values from 1 to 0 or 0 to 1
                indices_ones = np.where(h == 1)[0]
                indices_zeros = np.where(h == 0)[0]
                # Change from 1 to 0 to reduce sum
                change_count = int(np.round(excess))
                for i in range(change_count):
                    if len(indices_ones) > 0:
                        j = random.choice(indices_ones)
                        h[j] = 0
                        indices_ones = np.where(h == 1)[0]
                        indices_zeros = np.where(h == 0)[0]
                        
                # Move backwards to make exact
                if np.sum(h) < ones_count:
                    # Need to add more 1s
                    for i in range(int(ones_count - np.sum(h))):
                        if len(indices_zeros) > 0:
                            j = random.choice(indices_zeros)
                            h[j] = 1
                            indices_ones = np.where(h == 1)[0]
                            indices_zeros = np.where(h == 0)[0]
                # restore ones count exactly by possibly adjustment
                current_sum = np.sum(h)
                if np.isclose(current_sum, ones_count):
                    pass  # Already correct
                else:
                    # Do exact adjustment
                    more_to_add = int(ones_count - current_sum)
                    if more_to_add > 0:
                        # change zeros to ones
                        zero_indices = np.where(h == 0)[0]
                        select = np.random.choice(zero_indices, size=more_to_add, replace=False)
                        h[select] = 1
                    else:
                        # change ones to zero
                        one_indices = np.where(h == 1)[0]
                        select = np.random.choice(one_indices, size=-more_to_add, replace=False)
                        h[select] = 0
        else:
            # initialize deterministic pattern
            h = np.zeros(n, dtype=float)
            h[::2] = 1
            if ones_count != np.sum(h):
                # adjust to exact sum
                ones_indices = np.where(h == 1)[0]
                zero_indices = np.where(h == 0)[0]
                diff = int(ones_count - np.sum(h))
                for i in range(abs(diff)):
                    if diff > 0:
                        # Change one 0 to 1
                        j = random.choice(zero_indices)
                        h[j] = 1
                        zero_indices = np.where(h == 0)[0]
                        ones_indices = np.where(h == 1)[0]
                    else:
                        # Change one 1 to 0
                        j = random.choice(ones_indices)
                        h[j] = 0
                        ones_indices = np.where(h == 1)[0]
                        zero_indices = np.where(h == 0)[0]
    except ImportError:
        # Fallback to random construction
        h = np.zeros(n, dtype=float)
        indices = np.random.choice(n, size=ones_count, replace=False)
        h[indices] = 1.0
    
    # Compute correlation for initial h
    corr = np.correlate(h, 1 - h, mode='full')
    cur_max = corr.max()
    cur_c5 = cur_max * (2.0 / n)
    best_c5 = cur_c5
    best_h = h.copy()
    best_n = n
    
    # Annealing parameters
    temp = 0.02
    temp_decay = 0.9999
    
    iterations_without_improvement = 0
    restart_threshold = 10000
    
    for it in range(max_iters):
        now = time.time()
        if now - start > budget_s - 0.5:
            break

        # Propose swap
        i = np.random.randint(n)
        j = np.random.randint(n)
        while h[i] == h[j]:
            i = np.random.randint(n)
            j = np.random.randint(n)

        # Update correlation with delta
        new_corr, new_max = propose_swap_and_update(h, corr, i, j)
        new_c5 = new_max * (2.0 / n)
        
        # Accept or reject
        if new_c5 < cur_c5 or np.random.rand() < np.exp(-(new_c5 - cur_c5) / (temp + 1e-12)):
            corr = new_corr.copy()
            h[i], h[j] = h[j], h[i]
            cur_c5 = new_c5
            cur_max = new_max
            iterations_without_improvement = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                best_n = n
        else:
            iterations_without_improvement += 1
            # Restart
            if iterations_without_improvement > restart_threshold:
                h = np.zeros(n, dtype=float)
                indices = np.random.choice(n, size=ones_count, replace=False)
                h[indices] = 1.0
                corr = np.correlate(h, 1 - h, mode='full')
                cur_max = corr.max()
                cur_c5 = cur_max * (2.0 / n)
                iterations_without_improvement = 0
                temp = 0.02  # reset temperature

        temp *= temp_decay
    
    return best_h, best_c5, best_n
