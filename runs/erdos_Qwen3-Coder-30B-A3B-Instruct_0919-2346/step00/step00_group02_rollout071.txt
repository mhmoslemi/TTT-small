import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_erdos_solution(h_values):
    """Compute C5 from h_values (step function)"""
    n_points = len(h_values)
    dx = 2.0 / n_points
    corr_full = np.correlate(h_values, 1 - h_values, mode="full")
    c5 = np.max(corr_full) * dx
    return c5

# Initial construction from problem
initial_h_values = [0.5] * 45

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Start with initial solution
    h_original = np.array(initial_h_values)
    n_points = len(h_original)
    dx = 2.0 / n_points
    
    # Compute initial C5
    c5_original = evaluate_erdos_solution(h_original)
    
    # Helper function to enforce constraints
    def project_to_constraint(h):
        """Project h onto sum = n_points / 2 while maintaining bounds [0,1]"""
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        target_sum = n_points / 2
        if abs(current_sum - target_sum) < 1e-10:
            return h
        if current_sum > target_sum:
            # Need to reduce by subtracting from larger values
            diff = current_sum - target_sum
            # Sort indices in descending order of values
            sort_idx = np.argsort(h)[::-1]
            reduce_amount = 0
            h_new = h.copy()
            for i in sort_idx:
                if reduce_amount >= diff:
                    break
                max_to_reduce = min(h_new[i], diff - reduce_amount)
                h_new[i] -= max_to_reduce
                reduce_amount += max_to_reduce
        else:
            # Need to increase by adding to smaller values
            diff = target_sum - current_sum
            # Sort indices in ascending order of values
            sort_idx = np.argsort(h)
            increase_amount = 0
            h_new = h.copy()
            for i in sort_idx:
                if increase_amount >= diff:
                    break
                max_to_increase = min(1 - h_new[i], diff - increase_amount)
                h_new[i] += max_to_increase
                increase_amount += max_to_increase
        return h_new

    # Optimization parameters
    max_iter = 100000
    T_start = 1.0  # Initial temperature
    T_end = 1e-6
    alpha = 0.99  # Cooling rate
    max_neighbors = 1000
    current_time = time.time()

    best_h = h_original.copy()
    best_c5 = c5_original
    current_h = h_original.copy()
    current_c5 = c5_original

    # Annealing iteration
    T = T_start
    iter_count = 0
    last_improvement = 0
    improvement_threshold = 1000

    while iter_count < max_iter and (time.time() - current_time) < budget_s - 5:
        # Make a move: pick two random indices, change by small delta
        i = np.random.randint(0, n_points)
        j = np.random.randint(0, n_points)
        if i == j:
            continue

        # Compute bounds for delta
        hi = current_h[i]
        hj = current_h[j]
        delta_max = min(1 - hi, hj)
        delta_min = max(-hi, 1 - hj)
        if delta_min >= delta_max:
            continue
        delta = np.random.uniform(delta_min, delta_max)

        # Propose new solution
        h_new = current_h.copy()
        h_new[i] = hi + delta
        h_new[j] = hj - delta
        h_new = project_to_constraint(h_new)

        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(h_new)

        # Accept or reject
        delta_c5 = new_c5 - current_c5
        if delta_c5 < 0:
            # Always accept if better
            current_h = h_new
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = new_c5
        else:
            # Accept worse with probability based on temperature
            if np.random.rand() < np.exp(-delta_c5 / T):
                current_h = h_new
                current_c5 = new_c5
        
        # Update temperature
        T *= alpha
        iter_count += 1
        if iter_count % 1000 == 0:
            last_improvement = iter_count

    # Return final solution
    return (best_h.tolist(), best_c5, len(best_h))
