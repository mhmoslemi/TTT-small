import numpy as np
from scipy.optimize import differential_evolution
from scipy.fft import fft, ifft
import time
from math import floor
import cvxpy as cp

def evaluate_c5(h, dx):
    """Evaluate C5 using FFT for efficiency."""
    F = fft(h)
    G = fft(1 - h)
    corr = ifft(F * np.conj(G)).real
    return dx * max(corr)

def project_to_simplex(h, target_sum):
    """Project h to the simplex using Dykstra's alternating projection."""
    h = np.array(h).copy()
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h

    # Iterate to project using Dykstra's method
    for _ in range(10):
        delta = (current_sum - target_sum) / len(h)
        h = h - delta
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if abs(current_sum - target_sum) < 1e-10:
            break
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    allowed_time = budget_s - 30  # Leave 30s for final refinement

    # Initial parameters
    n = 180  # Good balance between resolution and time

    target_sum = n / 2.0

    h = project_to_simplex(h_interp, target_sum)

    # Global optimization via Differential Evolution
    def objective(h_vec):

        return evaluate_c5(h_vec, dx)

    # DE with custom bounds
    bounds = [(0, 1)] * n
    popsize = 30
    maxiter = allowed_time // 0.05

    # Initial population with seed and random vectors
    pop0 = [h.copy()]
    for _ in range(popsize - 1):
        p = np.random.rand(n)
        p = project_to_simplex(p, target_sum)
        pop0.append(p.copy())

    )

    if not result.success:
        best_h = h
    else:
        best_h = project_to_simplex(result.x, target_sum)

    # Post-processing: mismatch swap refinement
    def swap_refine(h_vec, dx, target_sum, timeout=30):
        start_refine = time.time()
        while time.time() - start_refine < timeout and True:
            F = fft(h_vec)
            G = fft(1 - h_vec)
            corr = ifft(F * np.conj(G)).real
            max_shift = np.argmax(corr)
            max_val = corr[max_shift]

            if max_val <= 0.3807 / dx + 1e-6:
                break

            # Identify mismatched pairs
            half_n = n // 2
            if max_shift <= half_n:
                # Positive shift
                i_set = [i for i in range(n - max_shift) if h_vec[i] > 0.99 and h_vec[i + max_shift] < 0.01]
                j_set = [j for j in range(n - max_shift) if h_vec[j] < 0.01 and h_vec[j + max_shift] > 0.99]
            else:
                # Negative shift
                shift = max_shift - n
                i_set = [i for i in range(-shift, n) if h_vec[i] > 0.99 and h_vec[i + shift] < 0.01]
                j_set = [j for j in range(-shift, n) if h_vec[j] < 0.01 and h_vec[j + shift] > 0.99]

            if not i_set or not j_set:
                # Perform small random move
                i = np.random.randint(0, n)
                j = np.random.randint(0, n)
                delta = np.clip(np.random.rand() * 0.02, 0.001, 0.01)

                    h_vec[i] += delta
                    h_vec[j] -= delta
                if j_set and i_set:
                    continue
                else:
                    h_vec = project_to_simplex(h_vec, target_sum)

                    F = fft(h_vec)
                    G = fft(1 - h_vec)
                    corr = ifft(F * np.conj(G)).real
                    max_val = corr[np.argmax(corr)]
                    if max_val <= 0.3807 / dx + 1e-6:
                        break
                continue

            # Perform a swap
            i = np.random.choice(i_set)
            j = np.random.choice(j_set)
            # Swap and adjust slightly to maintain feasibility
            val_i, val_j = h_vec[i], h_vec[j]
            delta = min(val_i - 0.01, 1.0 - val_j, 0.01)
            h_vec[i] = np.clip(h_vec[i] - delta, 0, 1)
            h_vec[j] = np.clip(h_vec[j] + delta, 0, 1)

            # Renormalize
            h_vec = project_to_simplex(h_vec, target_sum)

    best_h = swap_refine(best_h, dx, target_sum, timeout=60)
    final_c5 = evaluate_c5(best_h, dx)

    return list(best_h), final_c5, n
