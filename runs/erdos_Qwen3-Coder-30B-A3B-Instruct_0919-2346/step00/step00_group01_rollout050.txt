def evaluate_h(h_values):
    """Evaluate C5 for given h values using linear correlation"""
    n = len(h_values)
    dx = 2.0 / n
    # Compute correlation of h with (1-h) using linear correlation
    corr = np.correlate(h_values, 1 - h_values, mode='full')
    # The correlation values are for shifts from -(n-1) to (n-1)
    # We want the maximum over all shifts where overlap length is meaningful
    c5 = np.max(corr) * dx
    return c5

        return h_values

    idx_sorted = np.argsort(h_values)

    if abs(current_sum - target_sum) > 1e-10:
