import numpy as np
import time
import random
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()
    
    def c5_bound(h):
        n = len(h)
        dx = 2.0 / n
        O = np.correlate(h, 1 - h, mode='full')
        return O.max() * dx
    
    def random_binary(n):
        h = np.random.choice([0, 1], size=n, p=[0.5, 0.5])
        # Adjust to have exactly n/2 ones
        current_sum = h.sum()
        target_sum = n // 2
        if current_sum < target_sum:
            idx = np.where(h == 0)[0]
            np.random.shuffle(idx)
            h[idx[:target_sum - current_sum]] = 1
        elif current_sum > target_sum:
            idx = np.where(h == 1)[0]
            np.random.shuffle(idx)
            h[idx[:current_sum - target_sum]] = 0
        return h
    
    def thue_morse(n):
        h = np.zeros(n, dtype=int)
        for i in range(n):
            h[i] = bin(i).count('1') % 2
        return h
    
    def replicate_initial(n):
        # Use initial_h_values global variable
        if 'initial_h_values' not in globals():
            return random_binary(n)
        base = np.array(initial_h_values, dtype=float)
        # Interpolate to new length
        x_old = np.linspace(0, 2, len(base), endpoint=False)
        x_new = np.linspace(0, 2, n, endpoint=False)
        h_interp = np.interp(x_new, x_old, base)
        # Convert to binary and adjust weight
        h_bin = (h_interp > 0.5).astype(int)
        # adjust to exact weight
        current_sum = h_bin.sum()
        target_sum = n // 2
        if current_sum < target_sum:
            idx = np.where(h_bin == 0)[0]
            np.random.shuffle(idx)
            h_bin[idx[:target_sum - current_sum]] = 1
        elif current_sum > target_sum:
            idx = np.where(h_bin == 1)[0]
            np.random.shuffle(idx)
            h_bin[idx[:current_sum - target_sum]] = 0
        return h_bin
    
    # length list
    lengths = [64, 80, 96, 112, 128, 144, 160, 176, 192, 224, 256, 320, 384, 448, 512]
    lengths = [n for n in lengths if n <= 800 and n % 2 == 0]
    
    # Get initial candidates
    candidates = []
    for n in lengths:
        # Thue-Morse if power of 2
        if (n & (n - 1)) == 0:
            seq = thue_morse(n)
            candidates.append((seq, n))
        # replicate initial
        seq = replicate_initial(n)
        candidates.append((seq, n))
        # Random
        for _ in range(2):
            seq = random_binary(n)
            candidates.append((seq, n))
    
    # Evaluate candidates to find best ones
    best_candidates = []
    for h, n in candidates:
        if time.time() - start > budget_s * 0.9:
            break
        bound = c5_bound(h)
        best_candidates.append((bound, h, n))
    
    best_candidates.sort(key=lambda x: x[0])
    # Use top 5 to local search
    top_candidates = best_candidates[:5]
    
    best_h = None
    best_c5 = float('inf')
    best_n = None
    
    # Local search on top candidates
    remaining_budget = budget_s * 0.9
    time_per_candidate = remaining_budget / len(top_candidates)
    
    for bound, h, n in top_candidates:
        if time.time() - start > budget_s:
            break
            
        # Local optimization
        current_bound = bound
        current_h = h.copy()
        best_h_local = h.copy()
        best_bound_local = bound
        
        max_iter = 500000  # limit the iterations
        temp_start = 0.1
        temp_end = 1e-6
        
        for it in range(max_iter):
            if time.time() - start > budget_s:
                break
            temp = temp_start * (temp_end / temp_start) ** (it / max_iter)
            # pick random 1 and random 0 to swap
            ones_idx = np.where(current_h == 1)[0]
            zeros_idx = np.where(current_h == 0)[0]
            if len(ones_idx) == 0 or len(zeros_idx) == 0:
                break

            j = np.random.choice(zeros_idx)
            # swap
            current_h[i], current_h[j] = 0, 1
            new_bound = c5_bound(current_h)
            delta = new_bound - current_bound
            if delta <= 0 or random.random() < math.exp(-delta / temp):
                # accept
                current_bound = new_bound
                if new_bound < best_bound_local:
                    best_bound_local = new_bound
                    best_h_local = current_h.copy()
            else:
                # reject
                current_h[i], current_h[j] = 1, 0
                
            if best_bound_local < best_c5:
                best_c5 = best_bound_local
                best_h = best_h_local.copy()
                best_n = n
                
            # Early stopping
            if best_c5 < 0.38:
                break

    # Return final best result
    if best_h is None:
        # fallback to take best from initial candidates
        best_c5, best_h, best_n = best_candidates[0]
        best_h = best_h.copy()
    
    return (best_h, best_c5, best_n)
