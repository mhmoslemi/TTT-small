import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time
from itertools import product

def circular_correlation(h, n):
    """Compute circular correlation using FFT."""
    padded = np.pad(h, (0, n), 'constant')
    corrf = fft(padded)
    corri = ifft(corrf * np.conj(corrf))
    corr = np.real(corri[:n])
    return corr

def evaluate_c5(h):
    """Evaluate C5 from discrete h."""
    n = len(h)
    corr = circular_correlation(h, n)
    min_corr = np.min(corr)
    return 1 - 2 * min_corr / n

def sdp_solve(n, seed=0):
    """Solve the SDP relaxation of the problem."""
    np.random.seed(seed)
    # Create the matrix variable
    X = cp.Variable((n, n), symmetric=True)
    h = cp.Variable(n)
    t = cp.Variable()
    constraints = []
    M = []
    # Build M_k matrices and constraints
    for k in range(n):
        Mk = np.zeros((n, n))
        for i in range(n):
            Mk[i, (i + k) % n] = 1
            Mk[(i + k) % n, i] = 1
        M.append(Mk)
        constraints.append(cp.trace(Mk @ X) >= t)

    # Add constraints for h
    for i in range(n):
        constraints.append(X[i, i] == h[i])
        constraints.append(0 <= h[i])
        constraints.append(h[i] <= 1)
    constraints.append(cp.sum(h) == n / 2)
    constraints.append(X >> 0)  # X is positive semidefinite

    prob = cp.Problem(cp.Maximize(t), constraints)
    try:
        prob.solve(solver=cp.SCS, verbose=False, eps=1e-5)
        if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
            return h.value, t.value
        else:
            return None, None
    except:
        return None, None

def round_h(x, n, seed=0):
    """Round the SDP solution to binary with exact sum."""
    np.random.seed(seed)
    if x is None:
        return None
    # Use SVD of the Cholesky factor
    try:
        V = np.linalg.eigh(x)[1][:, :1]
    except:
        V = np.random.randn(n,1)
    g = np.random.randn(V.shape[1])
    scores = V.T @ g
    sorted_scores = np.sort(scores)[::-1]
    # Adjust threshold to get exactly n/2 ones
    target_ones = n // 2
    thresh = sorted_scores[target_ones - 1] if target_ones > 0 else sorted_scores[0]
    h = (scores >= thresh).astype(float)
    # Ensure exact count
    actual_ones = np.sum(h)
    if actual_ones < target_ones:
        ones = np.where(h == 0)[0]
        if len(ones) > 0:
            h[ones[:target_ones - actual_ones]] = 1
    elif actual_ones > target_ones:
        zeros = np.where(h == 1)[0]
        if len(zeros) > 0:
            h[zeros[:actual_ones - target_ones]] = 0
    return h

def swap_hill_climb(h, n, max_iter=500):
    """Apply deterministic swaps to improve autocorrelation."""
    h = h.copy()
    corr = circular_correlation(h, n)
    current_best = np.min(corr)
    for _ in range(max_iter):
        best_improvement = 0
        best_swap = None
        for i in range(n):
            for j in range(n):
                if h[i] != h[j]:  # Only consider swaps between different values
                    old_corr_vals = corr.copy()
                    # Simulate updated correlation
                    for k in range(n):
                        if i == (k - j) % n or j == (k - i) % n:
                            old_corr_vals[k] += h[i] - h[j]
                        elif i == (k - i) % n or j == (k - j) % n:
                            old_corr_vals[k] += h[i] - h[j]
                    new_min = np.min(old_corr_vals)
                    improvement = new_min - current_best
                    if improvement > best_improvement:
                        best_improvement = improvement
                        best_swap = (i, j)
        if best_swap is None or best_improvement <= 0:
            break
        i, j = best_swap
        h[i], h[j] = h[j], h[i]
        corr = circular_correlation(h, n)
        current_best = np.min(corr)
    return h

def gradient_fine_tune(h, n, n_iter=200):
    """Use projected gradient ascent on the minimum correlation."""
    h = h.copy()
    for i in range(n_iter):
        corr = circular_correlation(h, n)
        k_star = np.argmin(corr)
        # Compute gradient
        grad = np.zeros_like(h)
        for k in range(n):
            if k == k_star:
                grad[(k - k_star) % n] += 1
                grad[(k + k_star) % n] += 1
        # Apply projection: maintain sum=1 and bounds [0,1]
        step_size = 0.1 / (1 + i / 100)
        new_h = h + step_size * grad
        # Project to [0,1]
        new_h = np.clip(new_h, 0, 1)
        # Project to sum=1
        total = np.sum(new_h)
        if total > 0:
            new_h = new_h * (n / 2) / total
        h = new_h
    return h

def optimize_intervals(n=200, max_evals=2e5):
    """Optimize an interval union representation."""
    def compute_overlap_intervals(a, b, k):
        # Compute the total overlap length of interval union with shifted copy
        # a: array of start points
        # b: array of end points
        n_intervals = len(a)
        overlap = 0
        for i in range(n_intervals):
            for j in range(n_intervals):
                left = max(a[i], (a[j] + k) % 2)
                right = min(b[i], (b[j] + k) % 2)
                if left < right:
                    overlap += right - left
        return overlap

    def objective(x):
        # x: array of interval endpoints (a1, a2, ... bm) where bm = 2 - a1, etc.
        n = len(x)
        if n % 2 == 0:
            # Even number of intervals
            a = x[:n // 2]
            b = np.concatenate([2 - a[::-1]]), (x[n // 2:]))
        else:
            # Odd number of intervals
            a = x[:n // 2]
            b = np.concatenate((2 - a[::-1], [(x[n // 2] + 1) / 2]))
        # Normalize to area = 1 => sum(b - a) = 1
        lengths = b - a
        if np.sum(lengths) != 1:
            return 1e9
        # Compute overlaps for all shifts
        overlaps = []
        for k in range(1, n):
            overlap = compute_overlap_intervals(a, b, k)
            overlaps.append(overlap)
        if len(overlaps) == 0:
            return 1e9
        min_overlap = min(overlaps)
        return 1 - 2 * min_overlap / n

    # Optimizer using grid + refinement
    best_val = 2.0  # upper bound on C5
    best_h = None
    # Try different combinations
    seeds = np.random.randint(0, 2**32, 100)  # few seeds to speed up
    for seed in seeds:
        np.random.seed(seed)
        # Random interval setup with 3 intervals for simplicity
        a = np.sort(np.random.rand(3)) * 2
        b = a + (1 - np.sum(a)) / 3  # adjust b so that total length = 1
        x0 = np.hstack([a, b]).copy()
        # Fix symmetric constraints manually
        x0[0] = 0.0
        x0[1] = 0.5
        x0[2] = 1.0
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=[(0, 2)] * len(x0),

            if not res.success:
                continue
            val = objective(res.x)
            if val < best_val:
                best_val = val
                # Convert back to discrete h
                h = np.zeros(n)
                for a_i, b_i in zip(res.x[:3], res.x[3:]):
                    h = h + (a_i <= np.linspace(0, 2, n) - 1e-10) & (np.linspace(0, 2, n) - 1e-10 < b_i)
                best_h = h
        except:
            continue
    if best_h is not None:
        # Convert continuous to step function
        h_step = np.round(best_h).astype(int)
        h_step = h_step.astype(float)
        # Adjust the sum to exactly 1
        diff = (n / 2) - np.sum(h_step)
        if diff < 0:
            # Remove some 1's
            to_zero = int(-diff)
            ones = np.where(h_step == 1)[0]
            h_step[ones[:to_zero]] = 0
        elif diff > 0:
            # Add some 1's
            to_one = int(diff)
            zeros = np.where(h_step == 0)[0]
            h_step[zeros[:to_one]] = 1
        return h_step, best_val
