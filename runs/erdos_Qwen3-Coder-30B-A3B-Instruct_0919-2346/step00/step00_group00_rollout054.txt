def apply_gradient_step(h, grad, eta, target_sum):
    """Apply an SGD-style step"""
    h_new = h - eta * grad
    return project_to_capped_simplex(h_new, target_sum)

def compute_overlaps(h):
    """Compute all overlap values for h as a list."""
    corr = np.correlate(h, 1 - h, mode='full')
    return corr

def evaluate_overlap(h, dx):
    """Evaluate C5 for h according to the problem"""
    max_overlap = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    return max_overlap

def smooth_max(x, alpha):
    """Smooth approximation to max for vector x"""
    return np.log(np.sum(np.exp(alpha * x))) / alpha

def build_weights(h, alpha):
    """Compute soft weights"""
    overlaps = compute_overlaps(h) 
    overlaps = overlaps * np.ones_like(overlaps)  # Ensure 1D
    # Rather than exp( alpha * overlaps), use LSE normalization
    weights = np.exp(alpha * overlaps)
    norm = np.sum(weights)
    return weights / norm if norm != 0 else weights

def compute_gradient(h, alpha):
    """Compute the gradient of the smooth surrogate"""
    n = len(h)
    overlaps = compute_overlaps(h) 
    weights = build_weights(h, alpha)
    
    # Get weights vector as a 1D array
    weights_full = weights
    
    # conv1d for vectors to approximate convolution via convolution of weights and 1-h
    w_1mh = signal.convolve(weights_full, 1 - h, mode='full')
    w_h = signal.convolve(weights_full[::-1], h, mode='full')
    
    # Take results within original length n
    grad = w_1mh[:n] - w_h[:n]
    return grad

def deterministic_annealing(h, dx, alpha_start=0.5, gamma=1.3, max_alpha=200.0,
                           eta=0.01, decay_factor=0.99, steps_per_anneal=200,
                           max_iterations=30000, target_sum=0.5 * len(h)):
    """Run deterministic annealing with gradient steps."""
    h_current = h.copy()
    alpha = alpha_start
    eta_current = eta
    best_h = h_current.copy()
    best_c5 = evaluate_overlap(h_current, dx)
    it = 0
    last_alpha = -1
    
    while it < max_iterations and alpha < max_alpha:
        # Make steps without reshaping 
        for i in range(steps_per_anneal):
            if it >= max_iterations:
                break
            grad = compute_gradient(h_current, alpha)
            h_current = apply_gradient_step(h_current, grad, eta_current, target_sum)
            
            # Ensure sum compliance after step (emergency fix)
            h_current = project_to_capped_simplex(h_current, target_sum) 
            
            it += 1
            
        # Anneal
        alpha *= gamma
        eta_current *= decay_factor
        # Evaluate current at this alpha level
        current_c5 = evaluate_overlap(h_current, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_current.copy()
            
        if alpha >= max_alpha or last_alpha == alpha:
            break
        last_alpha = alpha
            
    return best_h, best_c5

def binary_rounding(h_cont, n_points):
    """Rounding the continuous h to binary t such that sum(h) * dx = 1"""
    # h should already sum to n_points/2 exactly from the projection phase
    # This would suggest the right number of 1s should be present to match unit area with dx=2/n
    # So sum(h) = n_points/2
    # The rounding will set top n_points/2 entries to 1, rest to 0
    # (Forcing N points to sum to n_points/2.)  
    # Also, this changes the physical thing anyway to a discrete mask.
    # The result is already such that sum(h) = n_points/2 (when rounded)
    
    h_ranked = np.sort(h_cont)[::-1]  
    threshold = h_ranked[len(h_ranked) // 2]
    
    h_bin = (h_cont >= threshold).astype(float) 
    # Ensure the sum is exactly n_points / 2
    actual_sum = np.sum(h_bin)
    if actual_sum != n_points / 2:
        # Find indices that can be flipped to adjust exact number
        diff = int(n_points / 2 - actual_sum)
        if diff > 0:
            # Need to set some 0s to 1s
            # Pick lowest elements (<= threshold) to set to 1
            lower_indices = np.where(h_cont < threshold)[0]
            # Take smallest diff of those indices according to their h_cont values
            # They are already sorted descending when picked from thresholded set
            # no, we want indices of h_cont < threshold!
            lower_indices = np.argsort(h_cont)[::-1]  # descending 
            to_flip = []
            for i in range(len(lower_indices)):
                if len(to_flip) >= diff:
                    break
                idx = lower_indices[i]
                if h_bin[idx] == 0:
                    to_flip.append(idx)
            for idx in to_flip:
                h_bin[idx] = 1
        elif diff < 0:
            # Pick highest 1s to flip to 0
            ones_indices = np.where(h_bin == 1)[0]
            # Flip diff elements from highest values starting from sorted 
            ones_sorted = sorted(ones_indices, key=lambda i: h_cont[i], reverse=True)
            to_flip = ones_sorted[:abs(diff)]
            for idx in to_flip:
                h_bin[idx] = 0
    
    # Double-check sum
    actual_sum = np.sum(h_bin)
    assert abs(actual_sum - n_points / 2) < 1e-12, f"Sum mismatch: got {actual_sum}, expected {n_points/2}"
    return h_bin

def local_search(h_bin, dx, max_swaps=10000):
    """Apply basic hill climbing swap heuristic to binary mask h_bin"""
    n = len(h_bin)
    
    # Compute initial correlations
    corr = np.correlate(h_bin, 1 - h_bin, mode='full')
    best_corr = np.max(corr)
    
    # Convert corr to arrays for fast index access
    corr = corr.copy()  # make it writable
    corr_max = best_corr
    
    # Only run for a limited time to avoid timeout
    start_time = time.time()
    swaps = 0
    
    while swaps < max_swaps and (time.time() - start_time) < 590:
        # Find indices where element is 1 or 0
        indices_1 = np.where(h_bin == 1)[0]
        indices_0 = np.where(h_bin == 0)[0]
        
        if len(indices_1) == 0 or len(indices_0) == 0:
            break
            
        # Randomly pick one from 1s and one from 0s
        i1 = np.random.choice(indices_1)
        i0 = np.random.choice(indices_0)
        
        # Compute change in max correlation using fast update trick,
        # only fast-add changes, not full recomputation,
        # but full correlations can be very expensive to update.
        
        # Instead, do axial swap updates:
        # When indices i1 and i0 swap in value,
        # we must update the CORRELATION for all k where:
        # - shift j into either i1 or i0 must contribute
        # This is NOT simply updating a small region as in original.
        
        # Naive but practical:
        # Test using a new candidate, see if it's better
        h_new = h_bin.copy()
        h_new[i1] = 0
        h_new[i0] = 1
        
        new_corr = np.correlate(h_new, 1 - h_new, mode='full')
        new_max = np.max(new_corr)
        
        if new_max < corr_max:
            h_bin = h_new
            corr_max = new_max
            swaps += 1
        else:
            swaps += 1
    
    return h_bin, corr_max

    np.random.seed(seed)
    
    start_time = time.time()
    
    # Starting vector

    dx = 2.0 / n_points
    target_sum = n_points / 2  # sum(h)*dx = 1
    
    # History of runs
    best_h = np.zeros(n_points)
    best_c5 = np.inf
    
    # Multi start: 1 seed from initial + 6 random ones
    seeds = [seed] + [np.random.randint(1000000) for _ in range(6)]
    
    counter = 0
    for s in seeds:
        np.random.seed(s)
        
        # Initialize with something in [0,1]
        h = np.random.rand(n_points)
        h = project_to_capped_simplex(h, target_sum)
        
        # Or on first run initialize with given initial_h_values if available
        if counter == 0 and 'initial_h_values' in globals():
            h = initial_h_values.copy()
            # normalize and project if needed
            sum_h = np.sum(h)
            if abs(sum_h - n_points/2) > 1e-12:
                h = h * (target_sum / sum_h)  
                h = project_to_capped_simplex(h, target_sum)
        
        # Run deterministic annealing
        h_opt, c5_opt = deterministic_annealing(
            h, dx,
            alpha_start=0.5, gamma=1.3, 
            max_alpha=200,
            eta=0.01, decay_factor=0.99,
            steps_per_anneal=200, max_iterations=10000, target_sum=target_sum
        )
        
        # Binary rounding
        h_bin = binary_rounding(h_opt, n_points)
        
        # Evaluate the true C5
        c5_evaluated = evaluate_overlap(h_bin, dx)
        
        if c5_evaluated < best_c5:
            best_c5 = c5_evaluated
            best_h = h_bin.copy()
            
        counter += 1
        
        if time.time() - start_time > budget_s - 5:
            break
    
    # Final refinement with local search if there is time
    if time.time() - start_time < budget_s - 10:
        try:
            _, final_c5 = local_search(best_h, dx, max_swaps=2000)
            if final_c5 < best_c5:
                best_c5 = final_c5
        except:
            pass
    
    # Final evaluation
    final_c5 = evaluate_overlap(best_h, dx)
    
    return (best_h, final_c5, n_points)
