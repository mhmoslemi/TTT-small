def compute_c5(h_values):
    # This function should compute C5 from given step function h
    # The theory/definition says we search across all shifts k to find out maximum 
    # of ∫ h(x) (1 - h(x+k)) dx  
    # Formulation-wise
    # Direct hack with what we're using. 
    n_points = len(h_values)
    dx = 2.0 / n_points
    # Since we're told direct computable h_values, we can't use external function
    corr_full = correlate(h_values, 1 - h_values, mode='full')  
    # Optimal computation range: only need shift -2 to +2 for valid shift contribution
    # Mapping: i index => shift = i - (n_points-1)
    # Want i where -2 <= i-(n_points-1) <= 2 => n_points-3 <= i <= n_points+1 
    # But i in [0, 2*n_points-2]  
    # So proper restricted indices to examine  
    # shift=0: index i = (n_points-1)  
    # shift=1: index i = n_points 
    # shift=2: index i = n_points + 1  
    # shift = -2: index i = n_points - 3  
    # Those = in [n_points-3, n_points+1]  
    # But actually for range from -2, 2 inside domain, the end are 197, 201 (for N=200)

    # do it cleanly, let's be sure we correctly understand implied discrete integration operands...
    # Assume h_values defined with multiple entries discretely for 0 to 2 with dx step (and 2/dx = n_points)
    # viz h_values[0], h_values[1], ..., h_values[n_points - 1].
    # each entry weight is dx = 2/n_points  

    # corr_full = np.correlate(h, 1-h, mode='full') produces 2*n_points-1 values -]  
    # we want: shift from -2 to 2 in domain, such that this lies in boundary but we must still evaluate integration region.
    # So points shift = 0: index = n_points - 1 (which means actual shift from 0 to 0 is initialized shifted like that)
    # So we collect points for indices from n_points-3, n_points-2, ..., n_points+1  
    # which cover range -2 to 2 precisely  
    start_i = n_points - 3  
    end_i   = n_points + 1  
    corr_values = corr_full[start_i : end_i + 1]    
    # Apply dx according to definition used for the integral expression
    corr_values = corr_values * dx
    # Change of sign convention ab/c
    # ∫ h(x) (1 - h(x+k)) dx = ∫ h(x) dx - ∫ h(x) h(x+k) dx = 1 - corr(k).
    C_values = 1 - corr_values
    # Final result is max of C_k across all shifts in aperture (shift k ∈ [-2,2])  
    C5 = np.max(C_values)
    return C5

    initial_sum = np.sum(initial_h)
    scale = 1.0 / initial_sum * (N / 2.0) 
    initial_h = initial_h * scale  # soft default scaling
    # Clip to keep in [0,1], only need scaling aspect
    initial_h.clip(0, 1, out=initial_h)
    
    # To advance: use Nx array x with values in [0,1], and sum = N/2, so it's valid
    # Using simulated annealing for local search of candidate
    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)
    
    # Optimization runs for budget time
    start_time = time.time()
    current_h = initial_h.copy()
    current_c5 = best_c5

    # For efficient random index selection and swap clamp operations
    def propose_delta(h, idx1, idx2):
        # Ensure both entries are valid before change
        hi, hj = h[idx1], h[idx2] 
        # Guarantee new values in [0,1]
        max_up_i = 1 - hi
        max_down_i = hi 
        max_up_j = 1 - hj
        max_down_j = hj
        # Restrict to valid moves to maintain feasibility
        max_delta_pos = min(max_up_i, max_down_j)
        max_delta_neg = min(max_down_i, max_up_j)
        delta = np.random.uniform(-max_delta_neg, max_delta_pos)
        return delta
    
    # Annealing parameter
    temp = 1.0
    decay_rate = 0.9999
    min_temp = 1e-8
    timeout_factor = 0.95
    iterations = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s * timeout_factor:
            break
        
        # Pick two different random indices
        idx1 = np.random.randint(0, N)
        idx2 = np.random.randint(0, N)
        while idx1 == idx2:
            idx2 = np.random.randint(0, N)
        
        delta = propose_delta(current_h, idx1, idx2)
        new_h = current_h.copy()
        new_h[idx1] += delta
        new_h[idx2] -= delta
        new_c5 = compute_c5(new_h)
        
        # Accept or reject
        if new_c5 < current_c5:  # Better
            current_h = new_h
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        else:
            # Possibly accept worse solution with probability
            # (also included in faster decline) 
            if np.exp(-(new_c5 - current_c5) / temp) > np.random.rand():
                current_h = new_h
                current_c5 = new_c5
                
        # Decrease temperature
        temp = max(min_temp, temp * decay_rate)
        iterations += 1
    
    # Final evaluation and output
    final_c5 = best_c5
    return (best_h, final_c5, N)
