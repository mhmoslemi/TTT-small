import numpy as np
import random
import time
import math

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h) using full mode
    correlation = np.correlate(h, 1 - h, mode='full')
    # Find the maximum value
    max_corr = np.max(correlation)
    return max_corr * dx

def project_onto_constraints(h, n):
    """Project h onto the constraints: 0 <= h[i] <= 1 and sum(h) = n/2"""
    # Clip values to [0,1]
    h = np.clip(h, 0, 1)
    # Adjust to meet sum constraint
    current_sum = np.sum(h)
    desired_sum = n / 2.0
    if abs(current_sum - desired_sum) < 1e-10:
        return h
    
    # Use a simple projection method
    diff = desired_sum - current_sum
    if diff > 0:  # Need to increase
        # Find indices that can be increased
        mask = (h < 1)
        indices = np.where(mask)[0]
        if len(indices) > 0:
            increase_per_idx = diff / len(indices)
            h[indices] += increase_per_idx
            h[indices] = np.clip(h[indices], 0, 1)
    else:  # Need to decrease  
        # Find indices that can be decreased
        mask = (h > 0)
        indices = np.where(mask)[0]
        if len(indices) > 0:
            decrease_per_idx = -diff / len(indices)
            h[indices] -= decrease_per_idx
            h[indices] = np.clip(h[indices], 0, 1)
    
    return h

def generate_initial_h(n):
    """Generate an initial h function"""
    # Use the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) == n:
            return h
        elif len(h) < n:
            # Pad with zeros
            h = np.pad(h, (0, n - len(h)), 'constant', constant_values=0)
        else:
            # Truncate
            h = h[:n]
        h = project_onto_constraints(h, n)
        return h
    
    # Fallback: random binary construction
    h = np.zeros(n)
    ones_indices = np.random.choice(n, n//2, replace=False)
    h[ones_indices] = 1
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Run optimization to find improved upper bound for C5"""
    random.seed(seed)
    np.random.seed(seed)
    
    start_time = time.time()
    
    # Use moderate size
    n_points = 100
    max_iter = 100000
    current_time = time.time()
    
    # Initialize h
    h_best = generate_initial_h(n_points)
    best_c5 = evaluate_c5(h_best)
    
    print(f"Initial C5: {best_c5:.6f}")
    
    # Simple local search with perturbation
    for it in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Propose small change
        h_new = h_best.copy()
        
        # Pick two indices to swap values
        i, j = np.random.choice(n_points, 2, replace=False)
        delta = np.random.uniform(-0.05, 0.05)
        
        # Adjust both indices while maintaining constraints
        old_val_i = h_new[i]
        old_val_j = h_new[j]
        
        new_val_i = old_val_i + delta
        new_val_j = old_val_j - delta
        
        # Clip to valid range
        new_val_i = max(0, min(1, new_val_i))
        new_val_j = max(0, min(1, new_val_j))
        
        h_new[i] = new_val_i
        h_new[j] = new_val_j
        
        # Project onto constraints
        h_new = project_onto_constraints(h_new, n_points)
        
        # Evaluate
        c5_new = evaluate_c5(h_new)
        
        # Accept if better
        if c5_new < best_c5:
            h_best = h_new
            best_c5 = c5_new
            if it % 1000 == 0:
                print(f"Iteration {it}: C5 = {best_c5:.6f}")
    
    # Final evaluation
    final_c5 = evaluate_c5(h_best)
    
    print(f"Final C5: {final_c5:.6f}")
    
    return (h_best.tolist(), final_c5, n_points)
