import numpy as np

def evaluate_erdos_solution(h):
    """Returns the C5 value for a given h."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(c) * dx
    return c5

def project_to_capped_simplex(h, target_sum):
    """Projects h onto the capped simplex with sum = target_sum."""
    h = np.array(h)
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if np.isclose(current_sum, target_sum):
        return h
    if current_sum > target_sum:
        # Need to reduce
        # Sort in descending order
        sorted_h = np.sort(h)[::-1]
        cumulative = np.cumsum(sorted_h)
        # Find the index where cumulative sum exceeds target_sum
        # But we ravel it like: sum(sorted_h[0:i]) is <= target_sum
        idx = np.searchsorted(cumulative, target_sum, side='right')
        if idx == 0:
            return np.zeros_like(h)
        threshold = sorted_h[idx-1] if idx < len(sorted_h) else 0
        # Set all h_vals >= threshold to threshold, and adjust the one below it
        h_new = np.minimum(h, threshold)
        diff = target_sum - np.sum(h_new)
        if diff > 0 and idx < len(sorted_h):
            # Find the first element that we can increase
            for i in range(len(h_new)-1, -1, -1):
                if h_new[i] < threshold:
                    increase = min(diff, threshold - h_new[i])
                    h_new[i] += increase
                    diff -= increase
                    if diff <= 0:
                        break
        return h_new
    else:
        # Need to increase the sum (can only add where h[i] < 1)
        # But in discrete case with constraints, project to feasible simplex
        # This is a bit tricky so we take a simpler approach:
        # 1. Ensure all h[i] is in [0,1], that's already done.
        # 2. Add enough mass to reach target
        diff = target_sum - current_sum
        indices = np.argsort(h)
        for i in range(len(h)):
            j = indices[i]
            if diff <= 0:
                break
            add = min(1 - h[j], diff)
            h[j] += add
            diff -= add
        return h

        # Create optimization variables
        h = cp.Variable(n, name='h')
        tau = cp.Variable(name='tau')
        # For k = 1..n-1, construct the y variables
        products = {}
        constraints = []
        
        for k in range(1, n):
            num_valid = n - k
            y = cp.Variable(num_valid, name=f'y_{k}')
            products[k] = y
            # McCormick envelope constraints
            for i in range(num_valid):
                j = i + k
                constraints += [
                    y[i] >= 0,
                    y[i] <= h[i],
                    y[i] <= h[j],
                    y[i] >= h[i] + h[j] - 1
                ]
            # Constraint that tau <= sum(y_k)
            constraints += [tau <= cp.sum(y)]
        
        # Constraints
        constraints += [cp.sum(h) == target_sum]
        constraints += [h >= 0]
        constraints += [h <= 1]

        objective = cp.Maximize(tau)
