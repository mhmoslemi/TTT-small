def project_to_sum_and_bounds(v, total):
    # Simple algorithm:
    # 1) Clip to [0,1]
    # 2) If sum after clipping equals total, done.
    # 3) Else adjust free entries:
    w = np.clip(v, 0, 1)
    deficit = total - np.sum(w)
    if abs(deficit) < 1e-9:
        return w
    # Determine free indices where 0<w_i<1
    free = np.where((w > 0) & (w < 1))[0]
    while True:
        if len(free) == 0:
            # No free entries, cannot adjust further, just return w
            return w
        delta = deficit / len(free)
        w[free] += delta
        # Clip again:
        over_upper = w > 1
        under_lower = w < 0
        if np.any(over_upper):
            deficit += np.sum(w[over_upper] - 1)  # overshoot amount to add back
            w[over_upper] = 1
        if np.any(under_lower):
            deficit += np.sum(w[under_lower])  # undershoot amount to add back (negative)
            w[under_lower] = 0
        # Recompute free indices:
        free = np.where((w > 0) & (w < 1))[0]
        deficit = total - np.sum(w)
        if abs(deficit) < 1e-9:
            break
    return w
```

def binary_hill_climb(h, max_iters=5000, patience=200, time_limit=120):
    start = time.time()
    best = h.copy()
    c_best = c5_of(best)
    cur = h.copy()
    c_cur = c_best
    it_without_improve = 0
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        N = len(cur)
        # compute correlation
        corr = np.correlate(cur, cur, mode='full')
        # index of minimum correlation
        idx_min = np.argmin(corr)
        shift = idx_min - (N - 1)
        # gather candidate pairs
        A = []
        B = []
        if shift >= 0:
            for i in range(N - shift):
                if cur[i] == 1 and cur[i+shift] == 0:
                    A.append(i)
                elif cur[i] == 0 and cur[i+shift] == 1:
                    B.append(i)
        else:
            # shift negative: i+shift = i - |shift|
            for i in range(-shift, N):
                j = i + shift
                if cur[i] == 1 and cur[j] == 0:
                    A.append(i)
                elif cur[i] == 0 and cur[j] == 1:
                    B.append(i)
        # If both sets nonempty, evaluate swaps
        improved = False
        if A and B:
            # limited number of samples
            sample_A = A if len(A) <= 20 else random.sample(A, 20)
            sample_B = B if len(B) <= 20 else random.sample(B, 20)
            best_swap = None
            best_c = c_cur
            for i in sample_A:
                for j in sample_B:
                    if cur[i] == cur[j]:
                        continue
                    # swap
                    cur_swapped = cur.copy()
                    cur_swapped[i], cur_swapped[j] = cur_swapped[j], cur_swapped[i]
                    c_new = c5_of(cur_swapped)
                    if c_new < best_c:
                        best_c = c_new
                        best_swap = (i, j)
            if best_swap is not None:

                cur[i], cur[j] = cur[j], cur[i]
                c_cur = best_c
                improved = True
                it_without_improve = 0

            zeros = np.where(cur == 0)[0]
            i = int(np.random.choice(ones))
