    z = z - z.max()

    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        return np.ones(len(z)) / len(z)
    return e / s

                a_new = a_new / sn

            s_new = 2.0 * n * float(_conv(a_new, fs).max())

            if np.isfinite(s_new) and s_new < best_score - 1e-14:
                best_score = s_new
                best_a = a_new.copy()
                base_a = a_new.copy()
                stall = 0
            else:
                stall += 1

            a = a_new

            # Periodic perturbation if stalled
            if stall > 100:
                z = np.log(np.maximum(base_a, 1e-12))
                z = z + rng.normal(0, 0.15 + 0.05 * (stall // 200), n)
                z = z - z.mean()
                a = _softmax(z)
                stall = 0

            if stall > 500:
                # Bigger restart
                z = np.log(np.maximum(best_a, 1e-12))
                z = z + rng.normal(0, 0.5, n)
                z = z - z.mean()
                a = _softmax(z)
                base_a = a.copy()
                stall = 0

    # ===== Phase 2: Gradient polish =====
    if time.monotonic() < fs_phase2_end:
        z = np.log(np.maximum(best_a, 1e-12))
        z = z - z.mean()
        gs = {"t": 0, "m": np.zeros(n), "v": np.zeros(n)}
        lr = 0.4
        tau_frac = 0.35
        stall = 0

        a = best_a.copy()
        while time.monotonic() < fs_phase2_end - 0.5:
            z, b = _grad_step(a, z, gs, fs, tau_frac, lr)
            a = _softmax(z)
            s = 2.0 * n * float(b.max())
            if np.isfinite(s) and s < best_score - 1e-14:
                best_score = s
                best_a = a.copy()
                stall = 0
            else:
                stall += 1

            if stall % 30 == 0:
                tau_frac = max(0.05, tau_frac * 0.97)
            if stall > 40:
                lr = max(0.05, lr * 0.9)
            if stall > 200:
                z = np.log(np.maximum(best_a, 1e-12)) + rng.normal(0, 0.4, n)
                z = z - z.mean()
                gs = {"t": 0, "m": np.zeros(n), "v": np.zeros(n)}
                lr = 0.35
                tau_frac = 0.3
                stall = 0

    # ===== Phase 3: Fine local search =====
    if time.monotonic() < deadline - 1.0:
        a = best_a.copy()
        z = np.log(np.maximum(a, 1e-12))
        z = z - z.mean()
        sigma = 0.1
        stuck = 0
        n_ph3 = int((deadline - 1.0 - time.monotonic()) * 200)  # ~200/s evaluations
        n_ph3 = max(1, min(n_ph3, 500000))

        for _ in range(n_ph3):
            if time.monotonic() >= deadline - 0.5:
                break

            r = rng.random()
            if r < 0.4:
                zc = z + rng.normal(0, sigma, n)
            elif r < 0.6:
                i = int(rng.integers(0, n))
                w = max(1, min(n // 10, int(rng.integers(1, max(2, n // 20)))))
                i2 = min(n, i + w)
                zc = z.copy()
                zc[i:i2] += rng.normal(0, sigma * 0.5, i2 - i)
            elif r < 0.75:
                k = int(rng.integers(1, max(2, n // 4)))
                ph = rng.uniform(0, 2 * math.pi)
                zc = z + sigma * 0.5 * np.sin(2 * math.pi * k * np.arange(n) / n + ph)
            elif r < 0.85:
                # Small LP nudge
                g, b = _fw_lp(a, n, fs)
                if g is not None:
                    zc = np.log(np.maximum(0.9 * a + 0.1 * g, 1e-12))
                else:
                    zc = z + rng.normal(0, sigma, n)
            else:
                zc = z * float(rng.uniform(0.5, 1.8))

            zc = zc - zc.mean()
            a2 = _softmax(zc)
            s2 = 2.0 * n * float(_conv(a2, fs).max())

            if np.isfinite(s2) and s2 < best_score - 1e-14:
                best_score = s2
                best_a = a2.copy()
                a = a2.copy()
                z = zc.copy()
                stuck = 0
                sigma = max(0.02, sigma * 0.95)
            else:
                stuck += 1
                if stuck % 500 == 0:
                    sigma = min(0.5, sigma * 1.1)
                if stuck > 3000:
                    z = np.log(np.maximum(best_a, 1e-12)) + rng.normal(0, 0.3, n)
                    z = z - z.mean()
                    a = _softmax(z)
                    stuck = 0

    # ===== Final validation with exact evaluator =====
    try:
        cand = [float(v) for v in best_a]
        fs_exact = float(evaluate_sequence(cand))
        if not np.isfinite(fs_exact):
            cand = [1.0]
            fs_exact = 2.0
    except Exception:
        cand = [1.0]

    # Safety: ensure non-empty, finite, nonnegative
    if not cand or len(cand) == 0:
        cand = [1.0]
    for i in range(len(cand)):
        if not np.isfinite(cand[i]) or cand[i] < 0:
            cand[i] = 0.0
        if cand[i] > 1000:
            cand[i] = 1000.0
    if all(c == 0 for c in cand):
        cand = [1.0]

    return cand
