def _project_simplex(v):
    """Project v onto the (nonnegative) simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    v = np.maximum(v, 0)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    # Sort and find threshold

    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]

    if s < 1e-15:

    ex = np.exp(x - x.max())
    a = ex / ex.sum()
    c = _fft_conv(a)
    Kc = K * c
    mc = Kc.max()
    log_s = np.log(np.sum(np.exp(Kc - mc)))
    val = 2.0 * n * (mc + log_s) / K
    return val

def _smooth_grad(x, K, n):
    ex = np.exp(x - x.max())
    a = ex / ex.sum()
    c = _fft_conv(a)
    Kc = K * c
    mc = Kc.max()
