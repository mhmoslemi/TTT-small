**Method 1: LP-based iteration**
At each step:
1. Current `a` (normalized, sum=1)
2. `b = conv(a,a)`, `M = max(b)`
3. Build `A` such that `(A@g)[k] = (conv(a,g))[k]`
4. Solve: max `sum(g)` s.t. `A@g <= M*(1-eps)`, `g >= 0`
5. Scale `g` to sum 1
6. For t in {0.1, 0.2, ..., 1.0}: compute `a_t = (1-t)*a + t*g`, track best
7. Update `a` to best `a_t`

This is essentially a "best response" iteration for the LP primal-dual.

**Method 2: Projected subgradient descent**
At each step:
1. `b = conv(a,a)`, find active set `K = {k : b[k] >= 0.99*max(b)}`
2. For each `k in K`, compute `grad_k[m]`:
   - For `m` where `|k-m| < n`: `grad_k[m] = a[|k-m|]` (with factor 2 if `k=m`)
   - Else: 0
3. Average gradients: `g = (1/|K|) * sum_{k in K} grad_k`
4. Project: `g_proj = g - mean(g)` (projection onto hyperplane sum=0)
5. For alpha in {0.001, 0.005, 0.01, ...}: `a_new = a - alpha * g_proj`, clip to [0,∞), normalize
6. Take best alpha

**Method 3: Simulated annealing**
- Anneal with random perturbations
- At each step, generate a perturbation, accept if better or with probability exp(-delta/T)
- Cool T over time

                a, local = nc, nv
                improved = True
                ub(a)
                break
        if not improved:
            for frac in (0.999, 0.995, 0.99, 0.98):
                nc, nv = do_grad(a, frac)
                if nv < local * (1 - 1e-12):
                    a, local = nc, nv
                    improved = True
                    ub(a)
                    break
        if improved:
            stag = 0
        else:
            stag += 1
            if stag % 20 == 0:
                for sc2 in (0.03, 0.02, 0.01):
                    ap = np.clip(a + sc2 * rng.standard_normal(n_p), 0, None)
                    sp = float(ap.sum())
                    if sp > 1e-15:
                        ap /= sp
                        vp = sc(ap)
                        if vp < local:
                            a, local = ap, vp
                            stag = 0
                            ub(a)
                            break
            if stag > 60:
                break

    # ── SMOOTH-APPROXIMATION PROJECTION (Boyd-Chanda) ──
    def smooth_grad(arr, p=100.0):
        n = len(arr)
        b = np.convolve(arr, arr)
        bpos = np.maximum(b, 1e-30)
        bp = bpos ** p
        Sp = float(bp.sum())
        if Sp < 1e-20:
            return np.zeros(n)
        wts = bpos ** (p - 1.0)
        g = np.zeros(n)
        for k in range(len(b)):
            w = wts[k]
            if w < bmax_thresh:
                continue
            jlo, jhi = max(0, k - n + 1), min(n - 1, k)
            if jlo <= jhi:
                js = np.arange(jlo, jhi + 1)
                g[js] += w * 2.0 * arr[np.abs(k - js)]
        return g * (Sp ** (1.0 / p - 1.0))

    bmax_thresh = 1e-25
    old_local = local
    bo = 0
    for it in range(8000):
        if time.monotonic() >= deadline - 100:
            break
        gsd = smooth_grad(a, 100.0)
        gsd -= gsd.mean()
        gs = float(np.linalg.norm(gsd))
        if gs < 1e-15:
            break
        gsd /= gs
        bse = float(np.dot(a, gsd))
        if abs(bse) < 1e-15:
            break
        dt = -bse / gs
        if bo < 10:
            a2 = np.clip(a - dt * gsd, 0, None)
            bmean = (a - dt * gsd).mean()
            bsum = float(a2.sum())
            if bsum > 1e-15:
                a2 /= bsum
        else:
            t_i = t_p * (2 * ro - 1)
            if t_i > dt:
                t_i = 0.5 * dt
            if t_i < -dt:
                t_i = -dt
            a2 = np.clip(a - t_i * gsd, 0, None)
            bmean = (a - t_i * gsd).mean()
            bsum = float(a2.sum())
            if bsum > 1e-15:
                a2 /= bsum
        a3 = a2.copy()
        a3 -= a3.mean()
        t_p = np.dot(a3, gsd) / gs
        if dt > 1e-10:
            ro = (0.5 * (2.0 - ro)) * (t_p - dt * 0.8) / (t_p - dt) + ro * 0.5
            ro = max(0.8, min(ro, 1.0))
        a = a2
        v = sc(a)
        if v < local:
            local = v
            ub(a)
        bo += 1

    # ── SIMULATED ANNEALING ──
    T = (local - best_score * 0.995) * 0.3
    T = max(T, 1e-5)
    for it in range(4000):
        if time.monotonic() >= deadline - 50:
            break
        T *= 0.9997
        ns = T / max(local, 1e-10)
        ap = np.clip(a + ns * rng.standard_normal(n_p), 0, None)
        sp = float(ap.sum())
        if sp < 1e-15:
            continue
        ap /= sp
        vp = sc(ap)
        if vp < local or (T > 1e-14 and rng.random() < math.exp(-(vp - local) / max(T, 1e-14))):
            a, local = ap, vp
            ub(a)

    # ── FINAL GRADIENT POLISH ──
    for it in range(5000):
        if time.monotonic() >= deadline - 15:
            break
        nc, nv = do_grad(a, 0.9999)
        if nv < local * (1 - 1e-13):
            a, local = nc, nv
            ub(a)
        elif it > 500:
            break

    # ── TRY A FEW OTHER LENGTHS ──
    def interp_len(src, n):
        if src.size == n:
            return src.copy()
        ox = np.linspace(0.0, 1.0, src.size, endpoint=True)
        nx = np.linspace(0.0, 1.0, n, endpoint=True)
        out = np.interp(nx, ox, src)
        out = np.clip(out, 0, None)
        s = float(out.sum())
        return out / s if s > 1e-15 else np.ones(n) / n

    other_lens = [n_p - 8, n_p - 4, n_p + 4, n_p + 8, 288, 300, 336, 344, 360, 384]
    for L in sorted(set(l for l in other_lens if l >= 64 and l <= 512)):
        if time.monotonic() >= deadline - 10:
            break
        alloc = min(25.0, (deadline - 10 - time.monotonic()))
        if alloc < 8:
            break
        ai = interp_len(a, L)
        ls = sc(ai)
        ub(ai)
        for it in range(3000):
            if time.monotonic() >= deadline - 8:
                break
            imp = False
            for mg in (1.0, 0.99, 0.98):
                b_x = np.convolve(ai, ai)
                M_x = float(b_x.max())
                if M_x <= 0:
                    break
                rl2, cl2 = _struct(L)
                si2 = rl2 - cl2
                try:
                    C2 = sps.csr_matrix((ai[si2], (rl2, cl2)), shape=(2*L-1, L))
                    res2 = linprog(-np.ones(L), A_ub=C2, b_ub=M_x*mg+1e-14,
                                   bounds=[(0.0,None)]*L, method='highs',
                                   options={'presolve':True})
                except Exception:
                    continue
                if res2 is not None and getattr(res2,'x',None) is not None:
                    g2 = np.clip(res2.x,0,None)
                    sg2 = float(g2.sum())
                    if sg2 > 1e-15:
                        g2 /= sg2
                        for t in (0.1,0.3,0.5,1.0):
                            c2 = np.clip((1-t)*ai+t*g2,0,None)
                            s2=float(c2.sum())
                            if s2>1e-15:
                                c2/=s2
                                v2=sc(c2)
                                if v2<ls:
                                    ai,ls=c2,v2
                                    imp=True
                                    ub(ai)
            if time.monotonic() < deadline - 8:
                nc2, nv2 = do_grad(ai, 0.999)
                if nv2 < ls*(1-1e-12):
                    ai, ls = nc2, nv2
                    imp = True
                    ub(ai)
            if not imp:
                if it > 200:
                    break

    # ── FINAL VALIDATION ──
    ub(a)
    if best_arr is None:
        best_arr = np.ones(96, dtype=np.float64)
        ub(best_arr)

    final = np.clip(best_arr, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(96, dtype=np.float64)
        ub(final)

    return [float(x) for x in final]
