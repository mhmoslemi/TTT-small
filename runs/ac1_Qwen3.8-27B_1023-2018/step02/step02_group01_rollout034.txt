import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, next_fast_len


def _convolve_square(a):
    """Compute a * a (full convolution) using FFT."""
    n = a.size
    if n == 0:
        return np.array([0.0])
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _exact_score(a):
    """Compute the exact evaluator score."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    n = a.size
    b = _convolve_square(a)
    return 2.0 * n * float(b.max()) / (s * s)


def _build_A(a):
    """Build constraint matrix A of shape (2n-1, n) where A[k,i] = a[k-i] if valid."""
    n = a.size
    m = 2 * n - 1
    k_idx = np.arange(m, dtype=np.int64)[:, None]
    i_idx = np.arange(n, dtype=np.int64)[None, :]
    j = k_idx - i_idx
    mask = (j >= 0) & (j < n)
    safe_j = np.clip(j, 0, n - 1)
    A = np.where(mask, a[safe_j], 0.0)
    return A


def _lp_best_response(a, M, timeout_s=1.0):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = a.size

    b_ub = np.full(2 * n - 1, M)

                      method='highs',

        if time.monotonic() > stop_time - 0.5:
            break

        b = _convolve_square(a)
        M = float(b.max())
        if M < best_M * (1e-12):
            best_M = M
            best_a = a.copy()

        # Subgradient: find argmax lag
        k_star = int(np.argmax(b))
        # Gradient of (a*a)[k_star] w.r.t. a[i] is a[k_star - i] (for valid i)
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = a[j]

        # Also add a small regularization to keep components positive
        grad += 0.01 / np.maximum(a, 1e-8)

        # Negative gradient step
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        a_new_sum = a_new.sum()
        if a_new_sum <= 0:
            break
        a_new = a_new / a_new_sum

        cM = float(_convolve_square(a_new).max())
        if cM < M * (1.0 - 1e-14):
            a = a_new
        else:
            lr *= 0.8

    return best_a


def _polish(a0, stop_time, max_iters=5000):
    """Final polishing with very small LP steps."""
    n = a0.size
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    a = a / a.sum()

    best_a = a.copy()
    best_M = float(_convolve_square(a).max())
    stuck = 0

    for it in range(max_iters):
        if time.monotonic() > stop_time - 0.5:
            break

        b = _convolve_square(a)
        M = float(b.max())
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        if stuck > 500:
            break

        g, _ = _lp_best_response(a, M * 1.0001, timeout_s=0.5)
        if g is None:
            break
        g = g / g.sum()

        # Very small step
        t = 0.05 if stuck < 100 else 0.02
        cand = (1 - t) * a + t * g
        cand = np.clip(cand, 0.0, 1000.0)
        cs = cand.sum()
        if cs <= 0:
            break
        cand = cand / cs
        cM = float(_convolve_square(cand).max())
        if cM < M * (1 - 1e-13):
            a = cand

    return best_a


def _make_ramp(n, power=1.0):
    """Create a power-law ramp of length n."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** power
    return a / a.sum()


def _make_gaussian(n, center_frac, width_frac):
    """Create a Gaussian-like sequence."""
    x = np.arange(n, dtype=np.float64)
    center = center_frac * n
    width = width_frac * n
    a = np.exp(-0.5 * ((x - center) / width) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = float(budget_s)
    deadline = t0 + budget - 12.0  # 12s headroom

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        if p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p
    except Exception:
        pass

    # Initialize best
    best_seq = None
    best_score = float('inf')

    def _update_best(a):
        nonlocal best_seq, best_score
        try:
            sc = _exact_score(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a.copy()
        except Exception:
            pass

    # Generate initializations of various types and lengths
    rng = np.random.default_rng(seed)
    inits = []

    # Parent
    if parent is not None:
        inits.append(('parent', parent / parent.sum(), parent.size))

    # Power-law ramps for various lengths
    for n in [64, 96, 128, 160, 192, 224, 256, 320, 400, 512]:
        for pwr in [0.5, 1.0, 1.5, 2.0, 3.0]:
            inits.append((f'_{n}_p{pwr}', _make_ramp(n, pwr), n))

    # Gaussians
    for n in [128, 192, 256, 384]:
        for cf, wf in [(0.3, 0.2), (0.5, 0.3), (0.7, 0.2), (0.5, 0.15)]:
            inits.append((f'_{n}_g{cf}_{wf}', _make_gaussian(n, cf, wf), n))

    # Exponential
    for n in [128, 192, 256, 384]:
        for alpha in [1.0, 2.0, 5.0, 10.0]:
            x = np.arange(n, dtype=np.float64)
            a = np.exp(alpha * x / n)
            inits.append((f'_{n}_e{alpha}', a / a.sum(), n))

    # Evaluate all inits quickly
    scored_inits = []
    for name, init, n in inits:
        try:
            sc = _exact_score(init)
            if np.isfinite(sc):
                scored_inits.append((sc, name, init, n))
        except Exception:
            pass

    scored_inits.sort(key=lambda x: x[0])

    # Update best with the best initialization
    if scored_inits:
        _update_best(scored_inits[0][2])

    if best_seq is None:
        # Fallback: simple ramp
        best_seq = _make_ramp(192, 1.0)
        best_score = _exact_score(best_seq)

    # Phase 1: LP optimization from top initializations
    # Allocate time based on number of inits
    phase1_end = min(deadline - 60, t0 + budget * 0.45)

    # Pick top unique (different n and type) initializations
    selected = []
    seen_types = set()
    for sc, name, init, n in scored_inits[:30]:
        key = (n, name.split('_')[1][:5] if len(name) > 5 else name)
        if key not in seen_types:
            seen_types.add(key)
            selected.append((sc, name, init, n))
        if len(selected) >= 8:
            break

    # If we have less than 8, add more
    for sc, name, init, n in scored_inits[:30]:
        if (sc, name, init, n) not in selected:
            selected.append((sc, name, init, n))
        if len(selected) >= 8:
            break

    per_init_time = max(3.0, (phase1_end - t0) / max(1, len(selected)))

    for idx, (sc, name, init, n) in enumerate(selected):
        alloc_end = min(t0 + 5.0 + (idx + 1) * per_init_time, phase1_end - 2)
        if time.monotonic() > alloc_end - 1:
            break
        try:
            result = _lp_iterate(init, stop_time=alloc_end, max_iters=500)
            _update_best(result)
        except Exception:
            pass

    # Phase 2: Polish the best result found
    phase2_end = min(deadline - 30, t0 + budget * 0.65)
    if best_seq is not None and time.monotonic() < phase2_end - 5:
        try:
            polished = _lp_iterate(best_seq, stop_time=phase2_end, max_iters=5000)
            _update_best(polished)
        except Exception:
            pass

    # Phase 3: Very fine polishing
    phase3_end = min(deadline - 15, t0 + budget * 0.85)
    if best_seq is not None and time.monotonic() < phase3_end - 3:
        try:
            polished = _polish(best_seq, stop_time=phase3_end, max_iters=10000)
            _update_best(polished)
        except Exception:
            pass

    # Phase 4: Try a few specific lengths with dedicated optimization
    if time.monotonic() < deadline - 20:
        for n_try in [best_seq.size - 1, best_seq.size + 1, best_seq.size + 2]:
            if time.monotonic() > deadline - 10:
                break
            if n_try < 8:
                continue
            # Start from a ramp of this length
            init_new = _make_ramp(n_try, 1.0)
            try:
                result = _lp_iterate(init_new, stop_time=time.monotonic() + 15, max_iters=300)
                _update_best(result)
            except Exception:
                pass

    # Final validation with the actual evaluator
    if best_seq is not None:
        try:
            final_score = float(evaluate_sequence([float(x) for x in best_seq]))
            if not np.isfinite(final_score):
                best_seq = _make_ramp(192, 1.0)
                final_score = float(evaluate_sequence([float(x) for x in best_seq]))
        except Exception:
            best_seq = _make_ramp(192, 1.0)
            try:
                final_score = float(evaluate_sequence([float(x) for x in best_seq]))
            except Exception:
                best_seq = [1.0 / 32.0] * 32

    # Sanity check
    if best_seq is None or len(best_seq) == 0:
        best_seq = [1.0 / 32.0] * 32

    # Verify finite
    result_list = [float(x) for x in best_seq]
    if not all(np.isfinite(x) for x in result_list):
        result_list = [1.0 / 32.0] * 32

    return result_list
