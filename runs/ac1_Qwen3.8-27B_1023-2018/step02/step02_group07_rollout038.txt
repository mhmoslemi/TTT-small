4. Take a step in the direction that reduces c_{k*} while keeping sum(a) = 1 and a >= 0.
5. Repeat.

More specifically:
- Let d = ∂c_{k*}/∂a (a vector of length n).
- We want to decrease c_{k*} while maintaining sum(a) = 1.
- The constrained gradient (tangent to the hyperplane sum(a) = 1) is: d_proj = d - mean(d) * 1.
- Take a: a_new = a - α * d_proj / ||d_proj||.
- Use a small α to ensure a stays non-negative.
- Check if c_{k*} actually decreased (and no other c_k increased above the new max).

This is essentially projected gradient descent on the active peak.

But this might be too slow (sublinear convergence). Let me try a different approach:

**Nesterov-style accelerated gradient**: Use momentum to accelerate the convergence.

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    total = float(budget_s)
    hard_deadline = t0 + total - 6.0

    # Get parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            if p.size > 0 and np.all(np.isfinite(p)):
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_a = None
    best_score = math.inf

    def try_update(a, score):
        nonlocal best_a, best_score
        try:
            a = np.asarray(a, dtype=np.float64).ravel()
            if a.size == 0 or not np.all(np.isfinite(a)):
                return
            a = np.clip(a, 0.0, 1000.0)
            if a.sum() < 1e-10:
                return
            if np.isfinite(score) and score < best_score:
                best_a = a.copy()
                best_score = float(score)
        except Exception:
            pass

    # Seed with parent
    if parent is not None and parent.size > 0:
        pn = parent / parent.sum()
        s = evaluate_sequence([float(v) for v in pn])
        if np.isfinite(s):
            try_update(pn, s)

    if best_a is None:
        best_a = np.ones(192) / 192.0
        best_score = 2.0

    # FFT helpers
    def autocorr(a, fs):
        n = a.size
        F = rfft(a, fs)
        return irfft(F * F, fs)[:2 * n - 1]

    def cross_grad(p, a, fs):
        n = a.size
        ar = a[::-1]
        c = irfft(rfft(ar, fs) * rfft(p, fs), fs)
        return c[n - 1:2 * n - 1]

    def score_norm(a):
        n = a.size
        fs = next_fast_len(2 * n - 1)
        b = autocorr(a, fs)
        m = float(b.max())
        return 2.0 * n * m if m > 0 else math.inf

    def run_adam(n, z_init, end_time, seed_val, tau0=0.35, lr0=0.45, verbose=False):
        fs_c = next_fast_len(2 * n - 1)
        fs_g = next_fast_len(3 * n - 2)
        z = np.array(z_init, dtype=np.float64).copy()
        z -= z.mean()
        rng = np.random.default_rng(seed_val)

        # Compute initial
        a = np.exp(z - z.max())
        a /= a.sum()
        b = autocorr(a, fs_c)
        bmax = float(b.max())
        best_local = 2.0 * n * bmax
        best_z = z.copy()
        best_a_loc = a.copy()

        m_acc = np.zeros(n)
        v_acc = np.zeros(n)
        t_ad = 0
        lr = lr0
        tau = tau0
        stuck = 0
        it = 0

        while time.monotonic() < end_time:
            a = np.exp(z - z.max())
            a /= a.sum()
            b = autocorr(a, fs_c)
            bmax = float(b.max())
            if not np.isfinite(bmax) or bmax <= 0:
                break
            cur = 2.0 * n * bmax

            if cur < best_local - 1e-10:
                best_local = cur
                best_z = z.copy()
                best_a_loc = a.copy()
                stuck = 0
            else:
                stuck += 1

            # Soft-max weights
            eff_tau = tau * bmax
            bt = b / eff_tau
            bt -= bt.max()
            e = np.exp(np.clip(bt, -80, 80))
            se = float(e.sum())
            if se < 1e-300 or not np.isfinite(se):
                p = np.zeros_like(b)
                p[int(np.argmax(b))] = 1.0
            else:
                p = e / se

            # Gradient
            g_a = cross_grad(p, a, fs_g)
            grad_z = a * (g_a - float(np.dot(a, g_a)))

            gn = float(np.abs(grad_z).max())
            if not np.isfinite(gn) or gn < 1e-15:
                if stuck > 30:
                    z = best_z + rng.normal(0, 0.4, n)
                    z -= z.mean()
                    m_acc[:] = 0
                    v_acc[:] = 0
                    t_ad = 0
                    lr = lr0
                    tau = tau0
                    stuck = 0
                it += 1
                continue

            grad_z /= (gn + 1e-30)
            t_ad += 1
            beta1, beta2 = 0.9, 0.999
            m_acc = beta1 * m_acc + (1 - beta1) * grad_z
            v_acc = beta2 * v_acc + (1 - beta2) * grad_z ** 2
            b1c = 1 - beta1 ** t_ad
            b2c = 1 - beta2 ** t_ad
            step = (m_acc / b1c) / (np.sqrt(v_acc / b2c) + 1e-8)
            smax = float(np.abs(step).max())
            if smax > 0 and np.isfinite(smax):
                step /= smax
            z = z - lr * step
            z -= z.mean()
            z = np.clip(z, -40, 40)
            z[~np.isfinite(z)] = 0

            # Adapt
            if it % 8 == 0:
                tau = max(0.015, tau * 0.975)
            if stuck > 15:
                lr = max(0.03, lr * 0.92)
            if stuck > 80:
                z = best_z + rng.normal(0, 0.3, n)
                z -= z.mean()
                m_acc[:] = 0
                v_acc[:] = 0
                t_ad = 0
                lr = lr0
                tau = min(tau0, 0.3)
                stuck = 0
            if it % 100 == 0 and stuck < 5:
                lr = min(lr0, lr * 1.01)

            it += 1

        a_out = np.exp(best_z - best_z.max())
        a_out /= a_out.sum()
        s_out = 2.0 * n * float(autocorr(a_out, fs_c).max())
        return a_out, s_out

    # Time allocation
    explore_end = t0 + 0.50 * total
    focus_end = t0 + 0.85 * total

    rng_main = np.random.default_rng(seed)

    # Build optimization jobs
    jobs = []
    job_seed = 0

    # Parent-based jobs at parent length
    if parent is not None and parent.size >= 32:
        pn = parent / parent.sum()
        zp = np.log(pn + 1e-15)
        zp -= zp.mean()
        jobs.append((parent.size, zp, 0.35, 0.45))
        # Perturbed versions
        for _ in range(3):
            zp2 = zp + rng_main.normal(0, 0.15, parent.size)
            zp2 -= zp2.mean()
            jobs.append((parent.size, zp2, 0.30, 0.40))
        job_seed += 4

    # Other lengths
    target_lengths = []
    if parent is not None:
        pl = parent.size
        target_lengths = [l for l in [pl // 2, pl, pl * 2, pl * 3, pl * 4] if 64 <= l <= 4096]
    else:
        target_lengths = [128, 192, 256, 384, 512, 768, 1024]

    seen_lengths = set()
    for n in target_lengths:
        if n in seen_lengths or n < 32:
            continue
        seen_lengths.add(n)
        # Parent-resized init
        if parent is not None:
            P = parent.size
            idx = np.clip((np.arange(n) * P // n).astype(int), 0, P - 1)
            a0 = parent[idx].astype(np.float64)
            a0 = np.maximum(a0, 1e-6)
            a0 /= a0.sum()
            z0 = np.log(a0)
            z0 -= z0.mean()
            jobs.append((n, z0, 0.35, 0.45))
            job_seed += 1
        # Uniform init
        z0 = np.zeros(n) + rng_main.normal(0, 0.1, n)
        z0 -= z0.mean()
        jobs.append((n, z0, 0.40, 0.50))
        job_seed += 1
        # Oscillatory
        t_arr = np.arange(n) / n
        a0 = 0.6 + 0.4 * np.cos(2 * np.pi * float(1 + job_seed % 5) * t_arr + float(rng_main.uniform(0, 2 * np.pi)))
        a0 = np.maximum(a0, 0.01)
        a0 /= a0.sum()
        z0 = np.log(a0)
        z0 -= z0.mean()
        jobs.append((n, z0, 0.35, 0.45))
        job_seed += 1

    # Phase 1: Round-robin exploration
    job_time = max(0.3, 30.0 / max(1, len(jobs)))

    while time.monotonic() < explore_end:
        for (n, z0, tau0, lr0) in jobs:
            if time.monotonic() >= explore_end:
                break
            rem = explore_end - time.monotonic()
            if rem < 0.3:
                break
            chunk = min(job_time, rem)
            end_t = time.monotonic() + chunk
            try:
                a_out, s_out = run_adam(n, z0, end_t, seed + job_seed * 31337, tau0=tau0, lr0=lr0)
                try_update(a_out, s_out)
            except Exception:
                pass
            job_seed += 1

    # Phase 2: Focused optimization on best
    if best_a is not None and time.monotonic() < focus_end:
        n_best = best_a.size
        a_best = best_a / best_a.sum()
        z_best = np.log(np.maximum(a_best, 1e-15))
        z_best -= z_best.mean()

        # Run multiple focused optimizations
        for attempt in range(4):
            if time.monotonic() >= focus_end - 3.0:
                break
            rem = focus_end - 3.0 - time.monotonic()
            if rem < 1.0:
                break
            chunk = min(rem / max(1, 4 - attempt), 200.0)
            end_t = time.monotonic() + chunk
            if attempt == 0:
                z_try = z_best.copy()
            else:
                z_try = z_best + rng_main.normal(0, 0.05 + 0.05 * attempt, n_best)
                z_try -= z_try.mean()
            try:
                a_out, s_out = run_adam(n_best, z_try, end_t, seed + 555000 + attempt * 777, tau0=0.15, lr0=0.30)
                try_update(a_out, s_out)
            except Exception:
                pass

    # Phase 3: Fine local search
    if best_a is not None and time.monotonic() < hard_deadline - 2.0:
        a = best_a / best_a.sum()
        n = a.size
        fs_c = next_fast_len(2 * n - 1)
        z = np.log(np.maximum(a, 1e-15))
        z -= z.mean()
        rng_ls = np.random.default_rng(seed + 888999)
        sigma = 0.08
        local_best = score_norm(a)
        local_best_z = z.copy()
        stuck = 0
        it_ls = 0

        while time.monotonic() < hard_deadline - 2.0:
            r = rng_ls.random()
            if r < 0.40:
                zt = z + rng_ls.normal(0, sigma, n)
            elif r < 0.60:
                ii = int(rng_ls.integers(0, n))
                ww = max(1, min(n, int(rng_ls.integers(1, max(2, n // 20 + 1)))))
                ii2 = min(n, ii + ww)
                zt = z.copy()
                zt[ii:ii2] += rng_ls.normal(0, sigma * 0.7, ii2 - ii)
            elif r < 0.75:
                k = int(rng_ls.integers(1, max(2, n // 6)))
                ph = rng_ls.uniform(0, 2 * np.pi)
                zt = z + sigma * np.sin(2 * np.pi * k * np.arange(n) / n + ph)
            elif r < 0.85:
                zt = z * rng_ls.uniform(0.6, 1.4)
            else:
                zt = z + sigma * rng_ls.normal(0, 1, n)
                # Smooth it
                if n > 4:
                    kernel = np.ones(3) / 3
                    zt = np.convolve(zt, kernel, mode='same')

            zt -= zt.mean()
            at = np.exp(zt - zt.max())
            at /= at.sum()
            st = 2.0 * n * float(autocorr(at, fs_c).max())

            if np.isfinite(st) and st < local_best:
                local_best = st
                local_best_z = zt.copy()
                z = zt.copy()
                stuck = 0
                sigma = max(0.01, sigma * 0.98)
            else:
                stuck += 1
                sigma = min(0.3, sigma * 1.03)
                if stuck > 40:
                    stuck = 0
                    sigma = 0.15
                    z = local_best_z + rng_ls.normal(0, 0.1, n)
                    z -= z.mean()

            it_ls += 1
            try_update(at, st)

        # Update best from local search
        a_final = np.exp(local_best_z - local_best_z.max())
        a_final /= a_final.sum()
        s_final = score_norm(a_final)
        try_update(a_final, s_final)

    # Final validation and return
    if best_a is not None:
        best_a = np.clip(best_a, 0.0, 1000.0)
        cand = [float(v) for v in best_a]
        try:
            fs = float(evaluate_sequence(cand))
            if np.isfinite(fs):
                return cand
        except Exception:
            pass

    # Absolute fallback
    return [1.0]
