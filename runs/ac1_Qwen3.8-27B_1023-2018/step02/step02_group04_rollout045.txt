        k = int(np.argmax(b))
        g = np.zeros(n)
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                g[j] = f[idx]
        gn = g.sum()
        if gn < 1e-30:
            return None
        return g / gn

    def line_search(a, d, cur_s, ntries=25):
        best_t = 0.0
        best_sc = cur_s
        for t in np.linspace(0.0, 1.0, ntries):
            cand = a * (1.0 - t) + d * t
            cand = np.clip(cand, 0, None)
            cs = score(cand)
            if cs < best_sc:
                best_sc = cs
                best_t = t
        return best_t, best_sc

    def lp_iterate(a0, time_limit, rng, max_iters=5000):
        a = norm1(a0.copy())
        n = a.size
        cur = score(a)
        update(a)
        stagnant = 0
        for it in range(max_iters):
            if time.monotonic() > time_limit:
                break
            # Try LP direction
            d = lp_direction(a, time_limit=max(5, int(time_limit - time.monotonic())))
            if d is None:
                # Fallback to subgradient
                d = subgrad_direction(a)
                if d is None:
                    stagnant += 1
                    if stagnant > 15:
                        break
                    continue
            t, new_s = line_search(a, d, cur, ntries=25)
            if t > 1e-8 and new_s < cur * (1.0 - 1e-12):
                a = norm1(a * (1.0 - t) + d * t)
                cur = new_s
                stagnant = 0
                update(a)
            else:
                stagnant += 1
                if stagnant % 10 == 0:
                    # Perturbation
                    pa = norm1(a * np.exp(0.05 * rng.standard_normal(n)))
                    ps = score(pa)
                    if ps < cur:
                        a, cur, stagnant = pa, ps, 0
                        update(a)
                if stagnant > 60:
                    # Try subgradient directly
                    d2 = subgrad_direction(a)
                    if d2 is not None:
                        t2, s2 = line_search(a, d2, cur, ntries=20)
                        if t2 > 1e-8 and s2 < cur:
                            a = norm1(a * (1.0 - t2) + d2 * t2)
                            cur = s2
                            stagnant = 0
                            update(a)
                        else:
                            break
                    else:
                        break
        return a, cur

    def make_starts(n, rng, par_ref):
        starts = []
        x = np.linspace(0, 1, n)
        def add(a):
            a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
            a = np.clip(a, 0, None)
            if a.size == n and a.sum() > 1e-10:
                starts.append(a / a.sum())
        add(np.ones(n))
        add(1.0 + 0.5 * np.sin(2 * np.pi * x))
        add(1.0 + 0.3 * np.sin(4 * np.pi * x))
        add(np.exp(2.0 * (x - 0.5) ** 2 * 4))
        add(np.exp(-4.0 * (x - 0.5) ** 2))
        add(1.0 + 0.4 * np.cos(2 * np.pi * x - 0.5))
        add(1.0 + 0.25 * np.sin(6 * np.pi * x))
        for _ in range(3):
            add(rng.dirichlet(np.full(n, 2.0)))
        if par_ref is not None and len(par_ref) >= 2:
            tp = np.linspace(0, 1, len(par_ref))
            add(np.interp(x, tp, par_ref))
            add(np.interp(x, tp, par_ref) * np.exp(0.1 * rng.standard_normal(n)))
        if not starts:
            starts.append(np.ones(n) / n)
        starts.sort(key=score)
        return starts

    # Phase 1: LP on parent length
    if par is not None:
        n_par = par.size
        if time.monotonic() < deadline - 400:
            starts = make_starts(n_par, rng, best_seq)
            for i, s in enumerate(starts[:5]):
                if time.monotonic() > deadline - 380:
                    break
                alloc = min(120.0, (deadline - 380 - time.monotonic()) * 0.4 + 10)
                if alloc < 10:
                    break
                lp_iterate(s, time.monotonic() + alloc, rng, max_iters=3000)

    # Phase 2: LP on n=128
    if time.monotonic() < deadline - 300:
        starts = make_starts(128, rng, best_seq)
        for i, s in enumerate(starts[:4]):
            if time.monotonic() > deadline - 280:
                break
            alloc = min(80.0, (deadline - 280 - time.monotonic()) * 0.35 + 8)
            if alloc < 8:
                break
            lp_iterate(s, time.monotonic() + alloc, rng, max_iters=4000)

    # Phase 3: LP on n=256
    if time.monotonic() < deadline - 220:
        starts = make_starts(256, rng, best_seq)
        for i, s in enumerate(starts[:3]):
            if time.monotonic() > deadline - 200:
                break
            alloc = min(65.0, (deadline - 200 - time.monotonic()) * 0.3 + 8)
            if alloc < 8:
                break
            lp_iterate(s, time.monotonic() + alloc, rng, max_iters=2500)

    # Phase 4: Refine best with more iterations
    if time.monotonic() < deadline - 120:
        nb = best_seq.size
        alloc = min(100.0, deadline - 120 - time.monotonic())
        if alloc > 20:
            lp_iterate(best_seq.copy(), time.monotonic() + alloc, rng, max_iters=5000)

    # Phase 5: Try interpolated lengths
    if time.monotonic() < deadline - 80:
        for nt in (160, 200, 224):
            if time.monotonic() > deadline - 60:
                break
            x = np.linspace(0, 1, nt)
            tp = np.linspace(0, 1, len(best_seq))
            a_init = np.interp(x, tp, best_seq)
            a_init = np.clip(a_init, 0, None)
            alloc = min(40.0, deadline - 60 - time.monotonic())
            if alloc < 10:
                break
            lp_iterate(a_init, time.monotonic() + alloc, rng, max_iters=2000)

    # Phase 6: Final gradient polish on the best
    if time.monotonic() < deadline - 25:
        a = norm1(best_seq.copy())
        n = a.size
        cur = score(a)
        alloc_end = deadline - 25
        for it in range(2000):
            if time.monotonic() > alloc_end:
                break
            b = np.convolve(a, a)
            ks = int(np.argmax(b))
            g = np.zeros(n)
            for j in range(n):
                idx = ks - j
                if 0 <= idx < n:
                    g[j] = a[idx]
            g = g / g.sum()
            t, ns = line_search(a, g, cur, ntries=20)
            if t > 1e-8 and ns < cur:
                a = norm1(a * (1 - t) + g * t)
                cur = ns
                update(a)
            else:
                # Try LP direction one more time
                d = lp_direction(a, time_limit=5)
                if d is not None:
                    t2, ns2 = line_search(a, d, cur, ntries=20)
                    if t2 > 1e-8 and ns2 < cur:
                        a = norm1(a * (1 - t2) + d * t2)
                        cur = ns2
                        update(a)
                        continue
                break

    # Final validation
    fin = np.asarray(best_seq, dtype=np.float64).ravel().copy()
    fin = np.nan_to_num(fin, nan=0.0, posinf=0.0, neginf=0.0)
    fin = np.clip(fin, 0, 1000)
    if fin.size < 1 or fin.sum() < 0.01:
        fin = np.array([1.0, 1.0])
    v = evaluate_sequence([float(x) for x in fin])
    if not math.isfinite(v):
        fin = np.array([1.0, 1.0])
    return [float(x) for x in fin]
