import numpy as np
import time
import math
from scipy.optimize import linprog
import scipy.sparse as sps

def propose_candidate(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 10.0
    
    # Scoring
    def sc(a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0, None)
        n = a.size
        if n == 0:
            return float('inf')
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        b = np.convolve(a, a)
        return 2.0 * n * b.max() / (s * s)
    
    # Best tracking
    best_score = float('inf')
    best_arr = None
    
    def update_best(a):
        nonlocal best_score, best_arr
        a = np.clip(np.asarray(a, dtype=np.float64).ravel(), 0, 1000)
        if a.size == 0 or a.sum() < 0.01:
            return
        v = float(sc(a))
        if math.isfinite(v) and v < best_score:
            best_score = v
            best_arr = a.copy()
    
    # Initialize from parent
    parent = None
    try:
        h = globals().get('height_sequence_1', None)
        if h is not None:
            parent = np.clip(np.asarray(h, dtype=np.float64).ravel(), 0, 1000)
            if parent.size > 0 and parent.sum() > 0.01:
                update_best(parent)
    except:
        pass
    
    if best_arr is None:
        best_arr = np.ones(96) / 96
        update_best(best_arr)
    
    n = best_arr.size
    
    # Convolution gradient
    def conv_grad_max(a, c, J):
        """Gradient of max_{k in J} c_k w.r.t. a, where c = conv(a,a)"""
        m = len(a)
        grad = np.zeros(m)
        for k in J:
            lo = max(0, k - m + 1)
            hi = min(m - 1, k)
            js = np.arange(lo, hi + 1)
            grad[js] += 2.0 * a[k - js]
        return grad
    
    # Projected gradient descent
    def proj_grad_descent(a, max_iters, stop, lr0=0.01, lr_decay=0.999, soft_sigma=0.0):
        a = a.copy()
        a = np.clip(a, 0, None)
        s = a.sum()
        if s < 1e-15:
            return a
        a /= s
        local = sc(a)
        update_best(a)
        lr = lr0
        stale = 0
        
        for it in range(max_iters):
            if time.monotonic() >= stop:
                break
            
            c = np.convolve(a, a)
            M = c.max()
            s_sum = a.sum()
            
            if soft_sigma > 0:
                # Soft max
                scaled = c / soft_sigma
                max_scaled = scaled.max()
                exp_scaled = np.exp(scaled - max_scaled)
                weights = exp_scaled / exp_scaled.sum()
                grad_M = np.zeros_like(a)
                for k in range(len(c)):
                    if weights[k] > 1e-10:
                        lo = max(0, k - a.size + 1)
                        hi = min(a.size - 1, k)
                        js = np.arange(lo, hi + 1)
                        grad_M[js] += 2.0 * weights[k] * a[k - js]
            else:
                J = np.where(c == M)[0]
                grad_M = conv_grad_max(a, c, J) / len(J)
            
            # Full gradient of f = 2n*M/s^2
            n_a = a.size
            grad_f = (2.0 * n_a / s_sum**2) * grad_M - (4.0 * n_a * M / s_sum**3)
            
            # Projected gradient step
            best_a_new = None
            best_v = local
            for lr_t in (lr, lr*0.5, lr*2):
                a_try = a - lr_t * grad_f
                a_try = np.clip(a_try, 0, None)
                s_try = a_try.sum()
                if s_try < 1e-15:

                a_try /= s_try
                v = sc(a_try)
                if v < best_v:
                    best_v = v
                    best_a_new = a_try
            
            if best_a_new is not None:
                a = best_a_new
                if best_v < local:
                    stale = 0
                else:
                    stale += 1
                local = best_v
                update_best(a)
            else:
                stale += 1
                lr *= 0.5
            
            if stale > 200:

                a_pert = a * np.exp(0.05 * rng.standard_normal(a.size))
                a_pert = np.clip(a_pert, 0, None)
                if a_pert.sum() > 1e-15:
                    a_pert /= a_pert.sum()
                    if sc(a_pert) < local:
                        a = a_pert
                        local = sc(a)
                        stale = 0
                        update_best(a)
                if stale > 500:
                    break
            
            lr = max(1e-6, lr * lr_decay)
        
        return a

    def mkA(a):
        m = len(a)
        rows = np.empty(m * m, dtype=np.int32)
        cols = np.empty(m * m, dtype=np.int32)
        vals = np.empty(m * m, dtype=np.float64)
        cnt = 0
        for k in range(2 * m - 1):
            lo = max(0, k - m + 1)
            hi = min(m - 1, k)
            for j in range(lo, hi + 1):
                rows[cnt] = k
                cols[cnt] = j
                vals[cnt] = a[k - j]
                cnt += 1
        return sps.csr_matrix((vals[:cnt], (rows[:cnt], cols[:cnt])), shape=(2*m-1, m))
    
    def lp_alternating(a, n, max_iters, stop, margins=(1.0, 0.99, 0.97)):
        a = a.copy()
        a = np.clip(a, 0, None)
        s = a.sum()
        if s < 1e-15:
            return a
        a /= s
        local = sc(a)
        update_best(a)
        stag = 0
        
        for ite in range(max_iters):
            if time.monotonic() >= stop:
                break
            
            c = np.convolve(a, a)
            M = c.max()
            if M <= 0:

            A = mkA(a)
            improved = False
            
            for m_mult in margins:

                b_ub = np.full(2*n-1, M * m_mult + 1e-12)
                c_obj = -np.ones(n)
                
                try:
                    res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                                  bounds=[(0.0, None)]*n, method='highs',
                                  options={'presolve': True})
                except:
                    continue
                
                if res is None or not hasattr(res, 'x') or res.x is None:
                    continue
                
                g = np.clip(res.x, 0, None)
                sg = g.sum()
                if sg < 1e-15:
                    continue
                g /= sg
                
                best_sc_t, best_c_t = local, None
                for t in (0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0):
                    cand = (1-t)*a + t*g
                    v = sc(cand)
                    if v < best_sc_t:
                        best_sc_t, best_c_t = v, cand
                
                if best_c_t is not None and best_sc_t < local * (1 - 1e-10):
                    a = best_c_t
                    local = best_sc_t
                    stag = 0
                    improved = True
                    update_best(a)
                    break
            
            if not improved:
                stag += 1
                if stag > 15:
                    # Perturb
                    a_p = a * np.exp(0.1 * rng.standard_normal(n))
                    a_p = np.clip(a_p, 0, None)
                    if a_p.sum() > 1e-15:
                        a_p /= a_p.sum()
                        if sc(a_p) < local:
                            a, local = a_p, sc(a_p)
                            stag = 0
                            update_best(a)
                if stag > 60:
                    break
        
        return a
    
    # Interpolation
    def interp_len(src, n):
        if src.size == n:
            return src.copy()
        ox = np.linspace(0, 1, src.size)
        nx = np.linspace(0, 1, n)
        out = np.interp(nx, ox, src)
        out = np.clip(out, 0, None)
        s = out.sum()
        if s > 1e-15:
            out /= s
        else:
            out = np.ones(n) / n
        return out
    
    # === MAIN ===
    a = best_arr.copy()
    a /= a.sum()
    update_best(a)
    
    # Phase 1: LP at parent length
    if time.monotonic() < deadline - 100:
        p = lp_alternating(a, a.size, 200, time.monotonic() + min(120, deadline - time.monotonic() - 100),
                          margins=(1.0, 0.995, 0.99, 0.98))
        update_best(p)
    
    # Phase 2: Gradient polish
    if time.monotonic() < deadline - 60:
        p = proj_grad_descent(a, 50000, deadline - 50, lr0=0.02, soft_sigma=0.0)
        update_best(p)
    
    # Phase 3: More LP
    if time.monotonic() < deadline - 80:
        p = lp_alternating(a, a.size, 150, time.monotonic() + min(100, deadline - time.monotonic() - 80),
                          margins=(1.0, 0.998, 0.995, 0.99))
        update_best(p)
