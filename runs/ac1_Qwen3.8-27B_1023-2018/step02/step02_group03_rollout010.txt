                if new_score < cur_score * (1 - 1e-12):
                    a = a_new
                    if new_score < best_local:
                        best_local = new_score
                        stale = 0
                        update_best(a)
                        # Decay LR
                        lr = max(0.001, lr * 0.999)
                    else:
                        # Score improved from this step but not all-time best
                        stale = min(stale + 1, 100)
                else:
                    stale += 1
                    # Try smaller step
                    for sc in (0.5, 0.2, 0.1):
                        a_try = a - sc * lr * m_hat / (np.sqrt(v_hat) + eps)
                        a_try = np.clip(a_try, 0.0, 1000.0)
                        if a_try.sum() > 1e-10:
                            a_try /= a_try.sum()
                            if fast_score(a_try) < cur_score:
                                a = a_try
                                best_local = fast_score(a)
                                stale = 0
                                update_best(a)
                                break

            if stale > 200:
                # Perturbation
                a_pert = a + 0.05 * rng.standard_normal(n)
                a_pert = np.clip(a_pert, 0, None)
                if a_pert.sum() > 1e-10:
                    a_pert /= a_pert.sum()
                    ps = fast_score(a_pert)
                    if ps < best_local:
                        a = a_pert
                        best_local = ps
                        stale = 0
                        update_best(a)
                if stale > 500:
                    break

        update_best(a)
        return a
