import time

        return float(2.0 * a.size * float(b.max()) / (s * s))

        a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)

            return

            best[1] = a.copy()

        s = float(a.sum())

            return np.ones(a.size, dtype=np.float64) / a.size

            return None

        b = np.convolve(f, f)
        M = float(b.max())
        if M < 1e-300 or not math.isfinite(M):
            return None

            A[j:j + n, j] = f

        try:

            idx = k_idx - ji
            valid = (idx >= 0) & (idx < n)
            grad[valid] += wt * 2.0 * f[idx[valid]]
        return grad

        return bs, bt

    # Phase 1: Optimize at parent length with multiple starts
