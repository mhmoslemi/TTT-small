def _get_parent():
    try:
        h = globals().get("height_sequence_1")
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            if p.size > 0 and np.isfinite(p).all():

def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(191, dtype=np.float64) / 191.0
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def _conv_square(a, direct=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([0.0], dtype=np.float64)
    if direct is None:

    else:

        F = rfft(a, L)
        b = irfft(F * F, L)
        b = b[:2 * n - 1]
        np.maximum(b, 0.0, out=b)
    return b


def _score_norm(a, direct=None):
    a = _normalize(a)
    if a.size == 0:
        return float("inf")
    b = _conv_square(a, direct=direct)
    mb = float(b.max())
    if mb < 0.0:
        mb = 0.0
    return float(2.0 * a.size * mb)


def _exact_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()

        return float("inf")
    a = np.where(np.isfinite(a), a, 0.0)

def _resample(a, m):
    a = _normalize(a)
    m = int(m)
    if m <= 0:
        return a
    if a.size == m:
        return a
    x_old = np.linspace(0.0, 1.0, a.size)
    x_new = np.linspace(0.0, 1.0, m)
    return _normalize(np.interp(x_new, x_old, a, kind="linear"))


def _build_A(a):
    a = _normalize(a)
    n = a.size
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for i in range(n):

    return A


def _lp_solve(a, M, timeout=0.7):
    a = _normalize(a)
    n = a.size
    if n == 0 or M <= 0.0:
        return None
    A = _build_A(a)
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(2 * n - 1, float(M), dtype=np.float64)
    bounds = [(0.0, None)] * n

    last = None
    for opts in (
        {
            "time_limit": float(timeout),
            "primal_feasibility_tolerance": 1e-11,
            "dual_feasibility_tolerance": 1e-11
        },
        None

            if opts is None:
                res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs")

                g = np.where(g > 0.0, g, 0.0)
                g = np.where(np.isfinite(g), g, 0.0)
                s = float(g.sum())
                if s > 0.0 and np.isfinite(s):
                    return g / s
        except Exception as e:
            last = e
    return None

    return best


def _cvx_trust_step(x, radius):
    if not _HAVE_CVXPY:
        return None
    x = _normalize(x)
    n = x.size
    if n < 4 or n > 512:
        return None

    d = _conv_square(x, direct=True)
    A = _build_A(x)
    M = float(d.max())

    try:
        y = cp.Variable(n)
        t = cp.Variable()
        cons = [
            cp.sum(y) == 1,
            y >= 0,
            A @ y <= 0.5 * (t + d),
            t <= M + 1e-12,
            cp.norm(y - x) <= radius
        ]
        prob = cp.Problem(cp.Minimize(t), cons)
        for solver in ("ECOS", "CVXOPT", "SCS", "CLARABEL"):
            try:
                prob.solve(solver=solver, verbose=False)
                if prob.status in ("optimal", "optimal_inaccurate"):
                    yv = np.asarray(y.value, dtype=np.float64)
                    if yv.size == n and np.isfinite(yv).all():
                        yv = _normalize(yv)
                        return yv
            except Exception:
                pass
    except Exception:
        pass
    return None


def _cvx_polish(a0, stop_time):
    if not _HAVE_CVXPY:
        return _normalize(a0)

    a = _normalize(a0)
    n = a.size
    if n > 512:
        return a

    direct = (n <= 512)
    best = a.copy()
    best_M = _score_norm(a, direct=direct)

    radii = [0.8, 0.5, 0.3, 0.18, 0.11, 0.07, 0.045, 0.03, 0.02, 0.012, 0.007, 0.004, 0.002]
    ridx = 0
    no_imp = 0

    while time.monotonic() < stop_time - 2.0 and no_imp < 28:
        r = radii[ridx]
        y = _cvx_trust_step(a, r)
        if y is None:
            if ridx + 1 < len(radii):
                ridx += 1
                continue
            break

        chosen = None
        chosen_M = best_M
        chosen_lam = 0.0

        for lam in (1.0, 0.8, 0.6, 0.4, 0.25, 0.12, 0.05):
            cand = _normalize((1.0 - lam) * a + lam * y)
            cm = _score_norm(cand, direct=direct)
            if cm < chosen_M * (1.0 - 1e-13):
                chosen = cand
                chosen_M = cm
                chosen_lam = lam

        if a.min() > 1e-10 and y.min() > 1e-10:
            la = np.log(a)
            ly = np.log(y)
            for lam in (1.0, 0.7, 0.4):
                w = (1.0 - lam) * la + lam * ly
                w -= float(w.max())
                cand = _normalize(np.exp(w))
                cm = _score_norm(cand, direct=direct)
                if cm < chosen_M * (1.0 - 1e-13):
                    chosen = cand
                    chosen_M = cm
                    chosen_lam = lam

        if chosen is not None:
            a = chosen.copy()
            best = a.copy()
            best_M = chosen_M
            no_imp = 0
            if chosen_lam < 1.0 or r < 0.25:
                ridx = min(ridx + 1, len(radii) - 1)
        else:
            no_imp += 1
            if ridx < len(radii) - 1:
                ridx += 1
            else:
                break

    return best


def _overlap_sums(a):
    n = a.size
    c = np.empty(n + 1, dtype=np.float64)
    c[0] = 0.0
    np.cumsum(a, out=c[1:])
    k = np.arange(2 * n - 1, dtype=np.int64)
    lo = np.maximum(0, k - (n - 1))
    hi = np.minimum(n - 1, k)
    return c[hi + 1] - c[lo]


def _cross_weights(a, p):
    n = a.size
    G = np.zeros(n, dtype=np.float64)
    for i in range(n):
        ai = a[i]
        if ai > 1e-15:
            G += ai * p[i:i + n]
    return G


def _make_log_params(a):
    a = _normalize(a)
    n = a.size
    loga = np.log(np.maximum(a, 1e-300))
    f = int(np.argmin(np.abs(loga - float(loga.mean()))))
    order = np.array([i for i in range(n) if i != f], dtype=int)
    z = loga[order] - loga[f]
    z = np.clip(z, -25.0, 25.0)
    return f, order, z


def _softmax_from_log_params(z, f, order):
    n = len(order) + 1
    full = np.zeros(n, dtype=np.float64)
    full[order] = np.asarray(z, dtype=np.float64)
    full[f] = 0.0
    m = float(full.max())
    e = np.exp(full - m)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(n, dtype=np.float64) / n
    return e / s


def _lbfgs_polish(a0, stop_time):
    a = _normalize(a0)
    n = a.size
    direct_conv = (n <= 384)
    direct_score = (n <= 512)
    best = a.copy()
    best_M = _score_norm(a, direct=direct_score)

    if n <= 384:
        factors = [4, 10, 25, 60]
        maxfun = 1100
        maxit = 80
    else:
        factors = [4, 12, 30]
        maxfun = 420
        maxit = 45

    for factor in factors:
        if time.monotonic() > stop_time - 5.0:
            break

        f, order, z0 = _make_log_params(a)
        beta = float(factor) * float(n)

        def objgrad(z):
            z = np.asarray(z, dtype=np.float64)
            full = np.empty(n, dtype=np.float64)
            full[order] = z
            full[f] = 0.0

            m = float(full.max())
            e = np.exp(full - m)
            s = float(e.sum())
            if not np.isfinite(s) or s <= 0.0:
                return 1e100, np.zeros(n - 1, dtype=np.float64)

            aa = e / s
            b = _conv_square(aa, direct=direct_conv)
            bm = float(b.max())
            if not np.isfinite(bm) or bm <= 0.0:
                return 0.0, np.zeros(n - 1, dtype=np.float64)

            xw = beta * (b - bm)
            xm = float(xw.max())
            ew = np.exp(xw - xm)
            S = float(ew.sum())
            if (not np.isfinite(S)) or S <= 0.0:
                p = np.zeros_like(b)
                p[np.argmax(b)] = 1.0
                val = bm
            else:
                p = ew / S
                val = bm + (1.0 / beta) * math.log(S)

            if not np.isfinite(val):
                return 1e100, np.zeros(n - 1, dtype=np.float64)

            r = _overlap_sums(aa)
            G = _cross_weights(aa, p)
            Sr = float(np.dot(p, r))
            grad_full = 2.0 * (G - aa * Sr)
            if not np.isfinite(grad_full).all():
                grad_full = np.where(np.isfinite(grad_full), grad_full, 0.0)
            grad_full -= float(grad_full.sum()) / float(n)

            return float(val), grad_full[order].astype(np.float64, copy=False)

        try:
            res = minimize(
                objgrad,
                z0,
                jac=True,
                method="L-BFGS-B",
                bounds=[(-25.0, 25.0)] * (n - 1),
                options={
                    "maxiter": int(maxit),
                    "maxfun": int(maxfun),
                    "gtol": 1e-7,
                    "ftol": 1e-11,
                    "iprint": 0
                }
            )
        except Exception:
            continue

        if res.x is not None and np.isfinite(np.asarray(res.x)).all():
            an = _softmax_from_log_params(res.x, f, order)
            an = _maybe_symmetrize(an)
            M = _score_norm(an, direct=direct_score)
            if M < best_M * (1.0 - 1e-12):
                a = an.copy()
                best = a.copy()
                best_M = M

    return best


def _lowdim_search(n, stop_time, seed=123):
    n = int(n)
    if n < 16 or time.monotonic() >= stop_time:
        return []

    out = []
    x = np.linspace(0.0, 1.0, n)
    lst = []
    for m in (1, 3, 5):
        lst.append(np.sin(2.0 * np.pi * m * x))
    for m in (1, 3):
        lst.append(np.cos(2.0 * np.pi * m * x))
    B = np.vstack(lst).astype(np.float64)
    K = B.shape[0]

    rng = np.random.default_rng(seed + 17)

    def make_arr(vals):
        arr = np.zeros(K, dtype=np.float64)
        for i, v in enumerate(vals):
            if i < K:
                arr[i] = float(v)
        return arr

    starts = [
        np.zeros(K),
        make_arr([0.8, 0.0, -0.4, 0.5, 0.1]),
        make_arr([-0.8, 0.0, 0.4, -0.5, -0.1]),
        make_arr([1.0, -0.6, 0.2, 0.3, -0.2]),
        rng.normal(0.0, 0.65, size=K),
        rng.normal(0.0, 0.65, size=K)
    ]

    direct = (n <= 384)

    for kind in (0, 1, 2):
        def make_a(c):
            c = np.asarray(c, dtype=np.float64)
            V = c @ B
            if kind == 0:
                raw = 1.0 + np.tanh(V)
            elif kind == 1:
                raw = np.exp(np.clip(V, -2.5, 2.5))
            else:
                raw = np.maximum(1.0 + V, 1e-3)
            return _normalize(raw)

        def fun(c):
            return _score_norm(make_a(c), direct=direct)

        for x0 in starts:
            if time.monotonic() > stop_time:
                break
            try:
                res = minimize(
                    fun,
                    np.asarray(x0, dtype=np.float64),
                    method="Nelder-Mead",
                    options={
                        "maxiter": 160,
                        "xatol": 1e-5,
                        "fatol": 1e-8,
                        "adaptive": True
                    }
                )
                if res.x is not None and np.isfinite(res.x).all():
                    a = make_a(res.x)
                    a = _maybe_symmetrize(a)
                    if _score_norm(a, direct=direct) < 1.95:
                        out.append(a.copy())
            except Exception:
                pass
    return out


def _quick_variants(parent, seed):
    parent = _normalize(parent)
    n = parent.size
    out = []

    def add(a):
        a = _normalize(a)
        if a.size == 0:
            return
        a = _maybe_symmetrize(a)
        out.append(a)

    add(parent)
    add(parent ** 0.7)
    add(parent ** 1.5)
    add(parent ** 2.5)
    add(parent ** 4.0)
    add(parent[::-1] ** 1.5)
    add(parent[::-1] ** 3.0)

    x = np.linspace(0.0, 1.0, n)
    for ph in (0.0, 0.35, 0.9, 1.7):
        add(parent * (1.0 + 0.45 * np.sin(2.0 * np.pi * x + ph)))
    for ph in (0.0, 0.7, 1.5):
        add(parent * (0.85 + 0.35 * np.cos(2.0 * np.pi * x + ph)))

    rng = np.random.default_rng(seed + 31)
    for _ in range(8):
        V = np.zeros(n)
        for m, amp in ((1, rng.uniform(0.1, 0.6)), (2, rng.uniform(0.05, 0.4)),
                       (3, rng.uniform(0.05, 0.35)), (6, rng.uniform(0.0, 0.25))):
            V += amp * np.sin(2.0 * np.pi * m * x + rng.uniform(0.0, 2.0 * np.pi))
        add(parent * np.exp(V))

    return out


def _improve_candidate(a0, stop_time, do_cvx=True, do_lbfgs=True, do_lp=True):
    a = _normalize(a0)

    if stop_time <= time.monotonic():
        return a

    n = a.size
    direct = (n <= 512)

    def accept(cand):
        nonlocal a
        cand = _normalize(cand)
        cand = _maybe_symmetrize(cand)
        if _score_norm(cand, direct=direct) < _score_norm(a, direct=direct) * (1.0 - 1e-13):
            a = cand.copy()
        return a

    if do_lp and time.monotonic() < stop_time - 30.0:
        now = time.monotonic()
        rem = max(0.0, stop_time - now)
        t1 = min(stop_time - 30.0, now + min(120.0, max(20.0, 0.30 * rem)))
        accept(_lp_polish(a, t1))
        a = _maybe_symmetrize(a)

    if do_cvx and _HAVE_CVXPY and n <= 512 and time.monotonic() < stop_time - 50.0:
        now = time.monotonic()
        rem = max(0.0, stop_time - now)
        t2 = min(stop_time - 50.0, now + min(350.0, max(40.0, 0.50 * rem)))
        accept(_cvx_polish(a, t2))
        a = _maybe_symmetrize(a)

    if do_lbfgs and time.monotonic() < stop_time - 25.0:
        now = time.monotonic()
        rem = max(0.0, stop_time - now)
        t3 = min(stop_time - 25.0, now + min(250.0, max(30.0, 0.35 * rem)))
        accept(_lbfgs_polish(a, t3))
        a = _maybe_symmetrize(a)

    if do_lp and time.monotonic() < stop_time - 25.0:
        now = time.monotonic()
        rem = max(0.0, stop_time - now)
        t4 = min(stop_time - 25.0, now + min(120.0, max(20.0, 0.30 * rem)))
        accept(_lp_polish(a, t4))
        a = _maybe_symmetrize(a)

    if do_lbfgs and time.monotonic() < stop_time - 15.0:
        now = time.monotonic()
        rem = max(0.0, stop_time - now)
        t5 = min(stop_time - 15.0, now + min(120.0, max(15.0, 0.40 * rem)))
        accept(_lbfgs_polish(a, t5))

    return a


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = float(budget_s)
    reserve = min(35.0, max(5.0, 0.05 * budget))
    deadline = t0 + max(10.0, budget - reserve)

    parent = _get_parent()
    best_seq = parent.copy()
    best_score = _exact_score(best_seq)
    if not np.isfinite(best_score):
        best_seq = np.ones(191, dtype=np.float64) / 191.0
        best_score = 2.0

    n0 = int(best_seq.size)
    rng = np.random.default_rng(seed)
    _ = rng  # seed consumed by helpers

    candidates = []

    def add_candidate(arr):
        a = _normalize(arr)
        if a.size == 0:
            return
        for b in candidates:
            if b.size == a.size and float(np.linalg.norm(a - b)) < 1e-7:
                return
        candidates.append(a.copy())

    for a in _quick_variants(parent, seed):
        add_candidate(a)

    if time.monotonic() < deadline - 320.0:
        for a in _lowdim_search(n0, stop_time=min(deadline - 320.0, time.monotonic() + 110.0), seed=seed + 1):
            add_candidate(a)

    n_high = min(1025, max(512, 4 * n0))
    if time.monotonic() < deadline - 320.0:
        for a in _lowdim_search(n_high, stop_time=min(deadline - 320.0, time.monotonic() + 130.0), seed=seed + 2):
            add_candidate(_resample(a, n0))

    for a in candidates:
        sc = _exact_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()

    if not candidates:
        candidates = [best_seq.copy()]

    candidates.sort(key=lambda z: _score_norm(z, direct=(z.size <= 512)))
    selected = candidates[:8]

    alt_idx = 0

    def next_alt():
        nonlocal alt_idx
        if alt_idx < len(selected):
            x = selected[alt_idx]
            alt_idx += 1
            return x
        return best_seq.copy()

    rounds = [
        ("alt", 450),
        ("global", 350),
        ("alt", 240),
        ("global", 250),
        ("alt", 180),
        ("global", 200),
        ("alt", 120)
    ]

    for kind, cap in rounds:
        now = time.monotonic()
        if now > deadline - 120.0:
            break
        if kind == "global":
            init = best_seq.copy()
        else:
            init = next_alt()

        alloc = min(float(cap), max(0.0, deadline - now - 100.0))
        if alloc < 35.0:
            break

        improved = _improve_candidate(
            init,
            stop_time=now + alloc,
            do_cvx=(best_seq.size <= 512 and _HAVE_CVXPY),
            do_lbfgs=True,
            do_lp=True
        )
        sc = _exact_score(improved)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = improved.copy()

    if time.monotonic() < deadline - 160.0:
        deltas = [1, 7, 16, 32, 64, 128, 192]
        tried = set()
        for d in deltas:
            now = time.monotonic()
            if now > deadline - 150.0:
                break
            m = n0 + int(d)
            if m in tried or m < 20 or m > 512:
                continue
            tried.add(m)

            inits = [best_seq.copy(), _resample(best_seq, m)]
            if now < deadline - 220.0:
                inits.extend(_lowdim_search(m, stop_time=min(deadline - 220.0, now + 35.0), seed=seed + m))

            best_init = None
            best_init_score = float("inf")
            for a in inits:
                a = _normalize(a)
                sc = _score_norm(a, direct=(a.size <= 512))
                if sc < best_init_score:
                    best_init_score = sc
                    best_init = a

            if best_init is None:
                best_init = _resample(best_seq, m)

            alloc = min(90.0, max(0.0, deadline - now - 120.0))
            if alloc < 30.0:
                break

            improved = _improve_candidate(
                best_init,
                stop_time=now + alloc,
                do_cvx=(m <= 512 and _HAVE_CVXPY),
                do_lbfgs=True,
                do_lp=True
            )
            sc = _exact_score(improved)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = improved.copy()

    now = time.monotonic()
    if now < deadline - 70.0:
        alloc = min(260.0, max(0.0, deadline - now - 50.0))
        if alloc > 60.0:
            improved = _improve_candidate(
                best_seq,
                stop_time=now + alloc,
                do_cvx=(best_seq.size <= 512 and _HAVE_CVXPY),
                do_lbfgs=True,
                do_lp=True
            )
            sc = _exact_score(improved)
            if np.isfinite(sc) and sc < best_score:
                best_seq = improved.copy()
                best_score = sc

    best_seq = _normalize(best_seq)
    if not best_seq.size:
        best_seq = np.ones(32, dtype=np.float64)

    final_list = [float(x) for x in best_seq]
    try:
        fs = float(evaluate_sequence(final_list))
    except Exception:
        fs = _exact_score(final_list)

    if not np.isfinite(fs):
        fallback = [1.0] * 32
        try:
            fs2 = float(evaluate_sequence(fallback))
            if np.isfinite(fs2):
                return fallback
        except Exception:
            pass
        return [1.0] * 32

    return final_list
