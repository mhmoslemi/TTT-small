            if not res.success or res.x is None:
                break
            g = np.maximum(res.x.astype(np.float64).ravel(), 0.0)
            gs = float(g.sum())
            if gs < 1e-14:
                break
            g /= gs

            if float(np.abs(g - f).max()) < 5e-10:
                break

            # Line search
            best_t = 0.0
            best_s = sc_cur
            # Coarse
            for t in (1.0, 0.7, 0.5, 0.3, 0.15, 0.05):
                h = (1.0 - t) * f + t * g
                s_h = sc(h, n, conv)
                if s_h < best_s:
                    best_s = s_h
                    best_t = t
            # Fine
            if best_t > 0.0:
                lo = max(0.0, best_t - 0.15)
                hi = min(1.0, best_t + 0.15)
                for t in np.linspace(lo, hi, 9):
                    h = (1.0 - t) * f + t * g
                    s_h = sc(h, n, conv)
                    if s_h < best_s:
                        best_s = s_h
                        best_t = t

            if best_t > 0.0:
                f = (1.0 - best_t) * f + best_t * g
                sc_cur = best_s
                if sc_cur < best_local_sc - 1e-15:
                    best_local_sc = sc_cur
                    best_local = f.copy()
            else:
                break

        return best_local, best_local_sc

    # Phase 1: LP from parent
    t_cap1 = min(deadline - 200.0, t0 + 400.0)
    f1, sc1 = lp_descent(f0, n_par, conv_par, max_iter=5000, t_cap=t_cap1)
    if sc1 < best_sc:
        best_sc = sc1
        best_f = f1
        best_n = n_par

    # Phase 2: Projected gradient with subgradient
    if time.monotonic() < deadline - 80.0:
        a = best_f / best_f.sum()
        n_a = best_n
        conv_a = mk_conv(n_a) if n_a != n_par else conv_par
        sc_a = sc(a, n_a, conv_a)
        lr = 0.3
        stuck = 0
        t_cap2 = deadline - 80.0

        for _ in range(15000):
            if time.monotonic() > t_cap2:
                break
            if stuck > 400:
                break

            c = conv_a(a)
            M = float(c.max())
            sc_a = 2.0 * n_a * M
            if sc_a < best_sc - 1e-15:
                best_sc = sc_a
                best_f = a.copy()
                best_n = n_a
                stuck = 0
            else:
                stuck += 1

            k_star = int(np.argmax(c))
            # Subgradient: d[j] = a[k_star - j] if valid
            d = np.zeros(n_a)
            lo = max(0, k_star - n_a + 1)
            hi = min(n_a - 1, k_star)
            d[lo:hi + 1] = a[k_star - hi:k_star - lo + 1][::-1]

            # Project onto tangent space of simplex
            d = d - d.mean()
            dn = float(np.linalg.norm(d))
            if dn < 1e-15:
                stuck += 3
                if stuck % 30 == 0 and stuck < 380:
                    a = np.maximum(a * np.exp(rng.normal(0, 0.05, n_a)), 1e-12)
                    a /= a.sum()
                continue

            improved = False
            for alpha in (lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 0.05):
                cand = a - alpha * d / dn
                cand = np.maximum(cand, 0.0)
                cs = cand.sum()
                if cs < 1e-10:
                    continue
                cand /= cs
                sc_c = sc(cand, n_a, conv_a)
                if sc_c < sc_a - 1e-15:
                    a = cand
                    sc_a = sc_c
                    improved = True
                    break

            if improved:
                stuck = 0
                lr = min(lr * 1.03, 3.0)
            else:
                stuck += 1
                lr = max(0.02, lr * 0.85)
                if stuck % 25 == 0 and stuck < 380:
                    a = np.maximum(a * np.exp(rng.normal(0, 0.08, n_a)), 1e-12)
                    a /= a.sum()

    # Phase 3: Adaptive stochastic search
    if time.monotonic() < deadline - 30.0:
        a = best_f / best_f.sum()
        n_a = best_n
        conv_a = mk_conv(n_a) if n_a != n_par else conv_par
        sc_a = sc(a, n_a, conv_a)
        sigma = 0.12
        rng2 = np.random.default_rng(seed + 999)
        t_cap3 = deadline - 30.0
        best_in_phase = sc_a

        for it in range(12000):
            if time.monotonic() > t_cap3:
                break
            if sc_a - best_in_phase < 0 and it % 500 == 499:
                if it > 5000:
                    break

            r = rng2.random()
            if r < 0.30:
                z = np.log(np.maximum(a, 1e-12))
                z += rng2.normal(0.0, sigma, n_a)
                z -= z.max()
                cand = np.exp(z)
                cand /= cand.sum()
            elif r < 0.50:
                cand = a.copy()
                k = rng2.integers(0, n_a)
                w = max(2, n_a // 40)
                lo = max(0, k - w)
                hi = min(n_a, k + w + 1)
                cand[lo:hi] *= abs(np.exp(rng2.normal(0, 0.18)))
                cand = np.maximum(cand, 0)
                cs = cand.sum()
                if cs > 1e-10:
                    cand /= cs
                else:
                    cand = a.copy()
            elif r < 0.65:
                cand = a.copy()
                i1 = rng2.integers(0, n_a)
                i2 = rng2.integers(0, n_a)
                amt = 0.015 * float(a.sum()) * rng2.uniform(0.5, 2.5)
                cand[i1] += amt
                cand[i2] = max(0.0, cand[i2] - amt)
                cand /= cand.sum()
            elif r < 0.80:
                w = rng2.uniform(0.1, 0.4)
                cand = (1 - w) * a + w * a[::-1]
                cand = np.maximum(cand, 0)
                cs = cand.sum()
                if cs > 1e-10:
                    cand /= cs
                else:
                    cand = a.copy()
            else:
                # Re-run short LP from perturbed point
                cand = a.copy()
                idx = rng2.integers(0, n_a, size=max(2, n_a // 6))
                cand[idx] *= np.exp(rng2.normal(0, 0.4, len(idx)))
                cand = np.maximum(cand, 1e-12)
                cand /= cand.sum()

            sc_c = sc(cand, n_a, conv_a)
            delta = sc_c - sc_a
            if delta < -1e-15 or (delta > 0 and rng2.random() < np.exp(-10 * delta / max(sigma ** 2, 1e-8))):
                a = cand.copy()
                sc_a = sc_c
                if sc_a < best_in_phase:
                    best_in_phase = sc_a
                if sc_a < best_sc - 1e-15:
                    best_sc = sc_a
                    best_f = a.copy()
                    best_n = n_a
                sigma = max(0.02, sigma * 0.97)
            else:
                sigma = min(0.5, sigma * 1.03)

    # Phase 4: Try different lengths
    if time.monotonic() < deadline - 50.0:
        for n_try in [n_par + 64, n_par - 64, n_par + 128, n_par + 32]:
            if n_try < 32 or n_try > 480:
                continue
            if time.monotonic() > deadline - 40.0:
                break
            conv_try = mk_conv(n_try)
            # Interpolate from best
            x_old = np.linspace(0, 1, best_n)
            x_new = np.linspace(0, 1, n_try)
            f_int = np.interp(x_new, x_old, best_f / best_f.sum())
            f_int = np.maximum(f_int, 0)
            f_int /= f_int.sum()
            f_opt, sc_opt = lp_descent(f_int, n_try, conv_try, max_iter=300,
                                        t_cap=deadline - 40.0)
            if sc_opt < best_sc:
                best_sc = sc_opt
                best_f = f_opt
                best_n = n_try

    # Phase 5: Final LP polish on the best
    if time.monotonic() < deadline - 15.0:
        conv_best = mk_conv(best_n)
        f_pol, sc_pol = lp_descent(best_f / best_f.sum(), best_n, conv_best,
                                    max_iter=500, t_cap=deadline - 15.0)
        if sc_pol < best_sc:
            best_sc = sc_pol
            best_f = f_pol
            best_n = best_n

    # --- Final validation ---
    best_f = np.clip(best_f, 0.0, 1000.0)
    if best_f.sum() < 0.01:
        best_f = np.ones(n_par) / n_par

    result = [float(x) for x in best_f]
    try:
        final_sc = float(evaluate_sequence(result))
    except Exception:
        final_sc = float('inf')

    if not np.isfinite(final_sc) or final_sc < 0:
        result = [1.0 / n_par] * n_par
        try:
            final_sc = float(evaluate_sequence(result))
        except Exception:
            pass
        if not np.isfinite(final_sc):
            result = [1.0]

    return result
