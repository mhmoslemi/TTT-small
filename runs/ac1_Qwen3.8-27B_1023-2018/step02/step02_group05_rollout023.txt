import numpy as np

TARGET = 1.50274349


def _sc(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.clip(a, 0.0, 1000.0)
    n = a.size
    if n == 0:
        return float('inf')
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    b = np.convolve(a, a)
    mb = float(b.max())
    if not math.isfinite(mb):
        return float('inf')
    return 2.0 * n * mb / (s * s)


def _proj_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1} via bisection on offset."""
    n = v.size
    if n == 0:
        return v
    v = np.sort(v)[::-1]
    cssv = np.cumsum(v)
    rho = np.nonzero(v * np.arange(1, n + 1) > (cssv[v.size - 1] - 1))[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0) if False else None  # placeholder


def _proj_simplex_fast(v):
    """Project onto probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv[-1] - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (float(rho + 1))
    return np.maximum(v - theta, 0.0)


def _lp_step(a, M, n, time_limit):
    """Solve LP: max sum(g) s.t. (a*g_conv)[k] <= M, g >= 0."""
    # Build dense Toeplitz matrix A where A[k,j] = a[k-j] if 0<=k-j<n
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = a
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M + 1e-15)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'time_limit': max(1.0, time_limit), 'presolve': True})
    except Exception:
        return None
    if res is not None and getattr(res, 'status', 1) == 0 and getattr(res, 'x', None) is not None:
        g = np.asarray(res.x, dtype=np.float64).ravel()
        g = np.clip(g, 0.0, None)
        sg = float(g.sum())
        if sg > 1e-15 and math.isfinite(sg):
            return g / sg
    return None

def _subgradient_polish(a_init, n, alloc_s, best, rng):
    """Subgradient descent on max autocorrelation with simplex projection."""
    a = np.asarray(a_init, dtype=np.float64).ravel().copy()
    if a.size < n:
        a = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, a.size), a)
    elif a.size > n:
        a = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, a.size), a)
    a = np.clip(a, 0.0, None)
    a = a / a.sum()

    local = _sc(a)
    if math.isfinite(local) and local < best[0]:
        best[0] = local
        best[1] = a.copy()

    eta = 0.01
    t_end = time.monotonic() + alloc_s - 5
    stale = 0

    for it in range(20000):
        if time.monotonic() > t_end:
            break
        if local < TARGET and it > 5:
            break

        b = np.convolve(a, a)
        M = float(b.max())
        eps = 0.05 * M
        active = np.where(b >= M - eps)[0]
        if len(active) == 0:
            break

        # Compute subgradient
        n_active = len(active)
        grad = np.zeros(n)
        for k in active:
            k = int(k)
            # grad_k[j] = a[k-j] if 0<=k-j<n else 0, PLUS a[j-k] if 0<=j-k<n else 0
            for j in range(n):
                v1 = k - j
                v2 = j - k
                g_val = 0.0
                if 0 <= v1 < n:
                    g_val += a[v1]
                if 0 <= v2 < n:
                    g_val += a[v2]
                grad[j] += g_val
        grad /= n_active

        # Projected subgradient step
        for a_try in (eta, eta * 0.5, eta * 0.2, eta * 2.0):
            v = a - a_try * grad / a.sum()
            a_new = _proj_simplex_fast(v)
            sc = _sc(a_new)
            if math.isfinite(sc) and sc < local:
                a = a_new
                local = sc
                stale = 0
                eta = min(eta * 1.05, 0.05)
                if local < best[0]:
                    best[0] = local
                    best[1] = a.copy()
                break
        else:
            stale += 1
            eta *= 0.8
            if eta < 1e-6:
                # Random perturbation
                pa = a + 0.02 * rng.standard_normal(n)
                pa = _proj_simplex_fast(pa)
                ps = _sc(pa)
                if math.isfinite(ps) and ps < local:
                    a = pa
                    local = ps
                    eta = 0.01
                    stale = 0
                    if local < best[0]:
                        best[0] = local
                        best[1] = a.copy()
                if stale > 50:
                    break

    return local


def _nelder_mead_polish(a_init, n, alloc_s, best):

    # Use scipy minimize with Nelder-Mead on the full vector (slow for n=320)
    # Instead, do coordinate descent on random subsets
    rng = np.random.default_rng(12345)
    t_end = time.monotonic() + alloc_s - 3
    stale = 0

    for it in range(5000):
        if time.monotonic() > t_end:
            break
        if local < TARGET:
            break

        # Pick a random pair of coordinates to optimize
        i, j = rng.integers(0, n, 2)
        if i == j:
            j = (i + 1) % n

        # 1D optimization over alpha: a[i] += alpha*(a[j]-a[i]), a[j] -= alpha*(a[j]-a[i])
        diff = a[j] - a[i]
        if abs(diff) < 1e-15:
            continue

        best_alpha = 0.0
        best_sc = local
        for alpha in np.linspace(-0.5, 0.5, 21):
            ai_new = a[i] + alpha * diff
            aj_new = a[j] - alpha * diff
            if ai_new < -1e-10 or aj_new < -1e-10:
                continue
            a_mod = a.copy()
            a_mod[i] = max(0.0, ai_new)
            a_mod[j] = max(0.0, aj_new)
            sm = a_mod.sum()
            if sm < 1e-15:
                continue
            a_mod = a_mod / sm
            sc = _sc(a_mod)
            if math.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_alpha = alpha

        if best_alpha != 0.0 and best_sc < local:
            diff = a[j] - a[i]
            a[i] = max(0.0, a[i] + best_alpha * diff)
            a[j] = max(0.0, a[j] - best_alpha * diff)
            a = a / a.sum()
            local = best_sc
            stale = 0
            if local < best[0]:
                best[0] = local
                best[1] = a.copy()
        else:
            stale += 1
            if stale > 200:
                break

    return local
