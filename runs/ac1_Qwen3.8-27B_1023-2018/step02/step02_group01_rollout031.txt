        bsc = float(evaluate_sequence([float(v) for v in bs]))
    except Exception:
        bsc = float('inf')
    if not np.isfinite(bsc) or bsc >= 999:
        bs = np.ones(64, dtype=np.float64)
        bsc = 2.0
    n0 = par.size
    ph1e = min(t0 + 0.45 * B, t0 + 500.0)
    inits = _mk_init(par, n0, seed)
    for idx in range(len(inits)):
        if time.monotonic() > ph1e - 8:
            break
        if idx == 0:
            ae = min(ph1e - 6.0, time.monotonic() + 150.0)
        elif idx <= 3:
            ae = min(ph1e - 4.0, time.monotonic() + 50.0)
        else:
            ae = min(ph1e - 2.0, time.monotonic() + 25.0)
        try:
            r = _opt(inits[idx], ae, mx=100000, sd=idx)
            ss = _sc(r)
            if np.isfinite(ss) and ss < bsc:
                bsc = ss
                bs = r.copy()
        except Exception:
            pass
    if time.monotonic() < ph1e - 5:
        try:
            r = _opt(bs, max(time.monotonic() + 120.0, ph1e - 2.0), mx=100000, sd=999)
            ss = _sc(r)
            if np.isfinite(ss) and ss < bsc:
                bsc = ss
                bs = r.copy()
        except Exception:
            pass
    ph2e = min(t0 + 0.70 * B, t0 + 850.0)
    if time.monotonic() < ph2e - 10:
        try:
            r = _opt(bs, ph2e - 3.0, mx=300000, sd=7777)
            ss = _sc(r)
            if np.isfinite(ss) and ss < bsc:
                bsc = ss
                bs = r.copy()
        except Exception:
            pass
    if time.monotonic() < dl - 50:
        for d in (-2, -1, 1, 2, -3, 3, -4, 4, -5, 5):
            nt = n0 + d
            if nt < 8:
                continue
            if time.monotonic() > dl - 20:
                break
            ix = np.linspace(0, 1, nt)
            jx = np.linspace(0, 1, bs.size)
            ini = np.interp(ix, jx, bs)
            ini = np.maximum(ini, 1e-8)
            try:
                r = _opt(ini, min(dl - 15, time.monotonic() + 40), mx=50000, sd=200 + d)
                ss = _sc(r)
                if np.isfinite(ss) and ss < bsc:
                    bsc = ss
                    bs = r.copy()
            except Exception:
                pass
    if time.monotonic() < dl - 30:
        try:
            r = _opt(bs[::-1], min(dl - 15, time.monotonic() + 120), mx=100000, sd=55555)
            ss = _sc(r)
            if np.isfinite(ss) and ss < bsc:
                bsc = ss
                bs = r.copy()
        except Exception:
            pass
    if time.monotonic() < dl - 12:
        try:
            r = _opt(bs, dl - 6.0, mx=200000, sd=12345)
            ss = _sc(r)
            if np.isfinite(ss) and ss < bsc:
                bsc = ss
                bs = r.copy()
        except Exception:
            pass
    finally:
        cs = [float(x) for x in np.clip(bs, 0.0, 1000.0)]
        try:
            fv = float(evaluate_sequence(cs))
        except Exception:
            fv = float('inf')
        if not np.isfinite(fv):
            cs = [1.0] * 64
        if time.monotonic() > dl - 2.0:
            pass
        return cs
