    t = 0.25
    stall = 0
    max_stall = 300
    iter_ct = 0
    while time.monotonic() < stop_time - 3.0:
        b = _conv_full(x)
        M = float(b.max())
        g = _lp_dir(x, n, M)
        if g is None:
            stall += 8
            if stall > max_stall:
                x = np.clip(x + rng.normal(0, 0.05, n), 0, None)
                s2 = x.sum()
                x = x / (s2 if s2 > 1e-15 else 1)
                stall = 0
                t = 0.25
            iter_ct += 1
            continue
        dist = float(np.linalg.norm(g - x))
        if dist < 1e-9:
            # Switch to gradient projection for a while
            gx, gsc = _grad_proj(x, n, min(stop_time - 3.0, time.monotonic() + 30.0), seed + iter_ct)
            if gsc < best_sc - 1e-14:
                x = gx
                best_x = x.copy()
                best_sc = gsc
                stall = 0
            else:
                stall += 5
            iter_ct += 1
            continue
        improved = False
        for _ in range(20):
            xn = (1.0 - t) * x + t * g
            xn = np.clip(xn, 0, None)
            s2 = xn.sum()
            if s2 < 1e-12:
                t = max(0.01, t * 0.5)
                continue
            xn = xn / s2
            sc = _score_norm(xn)
            if sc < best_sc - 1e-14:
                x = xn
                best_x = x.copy()
                best_sc = sc
                improved = True
                t = min(0.75, t * 1.06)
                stall = 0
                break
            else:
                t = max(0.005, t * 0.55)
        if not improved:
            stall += 1
            if stall >= max_stall:
                x = np.clip(x + rng.normal(0, 0.08, n), 0, None)
                s2 = x.sum()
                x = x / (s2 if s2 > 1e-15 else 1)
                t = 0.25
                stall = 0
        iter_ct += 1
    return best_x, best_sc


def _polish_coord(a, stop_time, seed=99):
    n = len(a)
    best = np.clip(np.asarray(a, float).ravel(), 0, None).copy()
    fs = best.sum()
    if fs < 1e-12:
        return best, math.inf
    best = best / fs
    best_sc = _score_norm(best)
    rng = np.random.default_rng(seed)
    no_imp = 0
    while time.monotonic() < stop_time - 2.0:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 2.0:
                break
            orig = best[i]
            bsc = best_sc
            bval = orig
            for mlt in (0.2, 0.4, 0.6, 0.8, 1.2, 1.5, 2.0, 0.0):
                best[i] = orig * mlt if orig > 1e-14 else (0.001 if mlt > 0 else 0.0)
                s2 = best.sum()
                if s2 < 1e-12:
                    best[i] = orig
                    continue
                sc = _score_norm(best / s2)
                if sc < bsc - 1e-14:
                    bsc = sc
                    bval = best[i]
            best[i] = bval
            if bsc < best_sc - 1e-14:
                best_sc = bsc
                improved = True
        # 2-coord moves
        for _ in range(80):
            if time.monotonic() > stop_time - 2.0:
                break
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            for mi in (0.5, 0.75, 1.3, 1.7):
                for mj in (0.5, 0.75, 1.3, 1.7):
                    best[i] = oi * mi if oi > 1e-14 else 0.001
                    best[j] = oj * mj if oj > 1e-14 else 0.001
                    s2 = best.sum()
                    if s2 < 1e-12:
                        best[i], best[j] = oi, oj
                        continue
                    sc = _score_norm(best / s2)
                    if sc < best_sc - 1e-14:
                        best_sc = sc
                        improved = True
            best[i], best[j] = oi, oj
        if not improved:
            no_imp += 1
            if no_imp >= 8:
                break
        else:
            no_imp = 0
    return best, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 15.0

    parent = _get_parent()
    n_p = len(parent)

    fallback = parent.copy()
    try:
        fsc = float(evaluate_sequence([float(x) for x in fallback]))
    except Exception:
        fsc = math.inf
    if not math.isfinite(fsc):
        fallback = np.ones(32, dtype=np.float64)
        fsc = float(evaluate_sequence([float(x) for x in fallback]))

    best_seq = fallback.copy()
    best_score = fsc

    def _try_update(arr, sc_norm):
        nonlocal best_seq, best_score
        if arr is None or not np.all(np.isfinite(arr)):
            return
        arr = np.clip(arr, 0, 1000)
        if arr.sum() < 0.01:
            return
        try:
            exact = float(evaluate_sequence([float(x) for x in arr]))
        except Exception:
            return
        if math.isfinite(exact) and exact < best_score:
            best_seq = arr.copy()
            best_score = exact

    rng_main = np.random.default_rng(seed)

    # Phase 1: LP optimization on parent length
    phase1_end = t0 + budget * 0.65
    n = n_p

    inits = [parent.copy()]
    # Add diverse initializations
    ir = np.arange(n, dtype=np.float64)
    inits.append(0.5 + 0.5 * np.cos(2 * math.pi * ir / n))
    inits.append(np.exp(-0.5 * ((ir - n / 2) / (n / 4)) ** 2) + 0.05)
    sup = np.ones(n) * 0.2
    for fr in (1, 2, 3, 5, 7):
        ph = rng_main.uniform(0, 2 * math.pi)
        sup += 0.4 * np.cos(2 * math.pi * fr * ir / n + ph)
    inits.append(np.abs(sup) + 0.02)
    inits.append(np.abs(rng_main.normal(0, 1, n)) + 0.05)
    inits.append(np.clip(parent + rng_main.normal(0, 0.3 * max(1.0, np.mean(parent)), n), 0, None) + 0.01)

    # Evaluate all inits
    for idx, init in enumerate(inits):
        if time.monotonic() > phase1_end - 10:
            break
        init_n = np.clip(init, 0, None)
        s0 = init_n.sum()
        if s0 < 1e-12:
            init_n = np.ones(n) / n
        else:
            init_n = init_n / s0
        sc0 = _score_norm(init_n)
        if idx == 0 or sc0 < best_score:
            _try_update(init_n, sc0)

    # LP optimization rounds
    time_per_round = max(10.0, (phase1_end - t0 - 5) / len(inits))
    for idx, init in enumerate(inits):
        stop = min(phase1_end - 5.0, time.monotonic() + time_per_round)
        if time.monotonic() > stop - 2:
            continue
        init_n = np.clip(init, 0, None)
        s0 = init_n.sum()
        if s0 < 1e-12:
            init_n = np.ones(n) / n
        else:
            init_n = init_n / s0
        try:
            res, rsc = _lp_optimize(init_n, n, stop, seed + 1000 * (idx + 1))
        except Exception:
            continue
        if rsc < best_score:
            _try_update(res, rsc)
        # Gradient projection follow-up
        if time.monotonic() < phase1_end - 10:
            try:
                gres, gsc = _grad_proj(res, n, min(phase1_end - 5.0, time.monotonic() + 60.0), seed + 70000 + idx)
            except Exception:
                pass
            if gsc < best_score:
                _try_update(gres, gsc)

    # Phase 2: Other lengths
    phase2_end = t0 + budget * 0.82
    other_lens = [128, 192, 256, 320, 384, 448, 512]
    if n_p not in other_lens:
        other_lens.append(n_p)
    other_lens = sorted(set(l for l in other_lens if 64 <= l <= 600))
    time_per_len = max(8.0, (phase2_end - phase1_end) / max(1, len(other_lens)))
    for L in other_lens:
        if time.monotonic() > phase2_end - 5:
            break
        stop = min(phase2_end - 3.0, time.monotonic() + time_per_len)
        if L == n_p and best_score < 1.51:
            init_l = best_seq.copy()
        else:
            if L == n_p:
                init_l = parent.copy()
            else:
                init_l = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, n_p), parent)
            init_l = np.clip(init_l, 0, None) + rng_main.uniform(0, 0.05, L)
        sl = init_l.sum()
        if sl < 1e-12:
            init_l = np.ones(L) / L
        else:
            init_l = init_l / sl
        try:
            res2, sc2 = _lp_optimize(init_l, L, stop, seed + 50000 + L)
        except Exception:
            continue
        if sc2 < best_score:
            _try_update(res2, sc2)

    # Phase 3: Polish best candidate
    polish_end = deadline - 5.0
    if time.monotonic() < polish_end - 15:
        try:
            polished, psc = _polish_coord(best_seq, polish_end, seed=4242)
            if psc < best_score:
                _try_update(polished, psc)
        except Exception:
            pass

    # Final gradient projection on best
    if time.monotonic() < deadline - 5:
        n_b = len(best_seq)
        try:
            gb, gsb = _grad_proj(best_seq, n_b, deadline - 8.0, seed=888)
            if gsb < best_score:
                _try_update(gb, gsb)
        except Exception:
            pass

    # Final polish
    if time.monotonic() < deadline - 3:
        try:
            polished2, _ = _polish_coord(best_seq, deadline - 2.0, seed=7777)
            _try_update(polished2, _score_norm(polished2 / polished2.sum()))
        except Exception:
            pass

    # Final validation
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf
    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in fallback]

    return [float(x) for x in best_seq]
