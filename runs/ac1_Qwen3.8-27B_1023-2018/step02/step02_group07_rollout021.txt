import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import minimize

def _conv(a, n, fs):
    F = rfft(a, fs)
    return irfft(F * F, fs)[:2*n-1]

def _cross_corr(p, a, n, fs2):
    """ Compute sum_m a[m] * p[m+j] for j=0..n-1 """
    ar = a[::-1]
    c = irfft(rfft(ar, fs2) * rfft(p, fs2), fs2)
    return c[n-1:2*n-1]

def _soft_max(b, tau):
    bm = b.max()
    if not np.isfinite(bm) or bm <= 0:
        return 0.0, b.max()
    x = np.clip((b - bm) / tau, -50, 50)
    e = np.exp(x)
    s = e.sum()
    if s <= 0:
        return bm, np.zeros_like(b)
    p = e / s
    return bm + tau * math.log(s), p

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import minimize


def _conv(a, n, fs):
    F = rfft(a, fs)
    return irfft(F * F, fs)[:2 * n - 1]


def _cross_corr(p, a, n, fs2):
    """Compute sum_m a[m]*p[m+j] for j=0..n-1 via FFT."""
    ar = a[::-1]
    c = irfft(rfft(ar, fs2) * rfft(p, fs2), fs2)
    return c[n - 1:2 * n - 1]


def _soft_max_and_weights(b, tau):
    bm = float(b.max())
    if not np.isfinite(bm) or bm <= 0:
        return 0.0, np.zeros_like(b)
    x = np.clip((b - bm) / tau, -50.0, 50.0)
    e = np.exp(x)
    s = float(e.sum())
    if s <= 0:
        p = np.zeros_like(b)
        p[int(np.argmax(b))] = 1.0
        return bm, p
    p = e / s
    return bm + tau * math.log(s), p


def _score(a, n, fs_c):
    b = _conv(a, n, fs_c)
    m = float(b.max())
    if not np.isfinite(m) or m < 0:
        return math.inf, 0.0
    return 2.0 * n * m, m

            zp = np.log(np.maximum(a, 1e-12))
            f = int(rng.integers(1, max(2, n // 4)))
            ph = rng.uniform(0, 2 * math.pi)
            zp += sigma * 0.4 * np.sin(2 * math.pi * f * np.arange(n) / n + ph)
        else:
            # Sparse random blip
            zp = np.log(np.maximum(a, 1e-12))
            idx = rng.integers(0, n, max(1, n // 30))
            zp[idx] += rng.normal(0, sigma * 0.8, len(idx))

        zp = zp - zp.mean()
        a_new = np.exp(np.clip(zp, -30, 30))
        a_new = a_new / max(1e-300, a_new.sum())
        b_new = _conv(a_new, n, fs_c)
        val = 2.0 * n * float(b_new.max())

        if np.isfinite(val) and val < best_val - 1e-14:
            best_val = val
            best = a_new.copy()
            a = a_new.copy()
            stuck = 0
            sigma = max(0.02, sigma * 0.96)
        else:
            stuck += 1
            sigma = min(0.6, sigma * 1.04)
            if stuck > 120:
                stuck = 0
                sigma = 0.2

    return best, best_val


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    total = max(1.0, float(budget_s))
    deadline = t0 + total - 4.0

    rng = np.random.default_rng(int(seed))

    # --- Get parent ---
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            if p.size > 0 and np.isfinite(p).all():
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    # --- Initialize best ---
    best_a = np.ones(32) / 32.0
    best_score = 2.0
    if parent is not None and parent.size > 0:
        try:
            cand = [float(max(0.0, v)) for v in parent]
            s = float(evaluate_sequence(cand))
            if np.isfinite(s):
                best_a = np.array(cand, dtype=np.float64)
                best_score = s
        except Exception:
            pass

    n_primary = parent.size if (parent is not None and parent.size > 4) else 192
    n_primary = max(16, min(n_primary, 4096))

    fs_c = next_fast_len(2 * n_primary - 1)
    fs2 = next_fast_len(3 * n_primary - 2)

    def _update(a, s):
        nonlocal best_a, best_score
        try:
            a = np.asarray(a, dtype=np.float64).ravel()
            if a.size == 0 or not np.isfinite(a).all():
                return
            a = np.clip(a, 0, 1000)
            if a.sum() < 1e-6:
                return
            a = a / a.sum()
            if np.isfinite(s) and s < best_score - 1e-14:
                best_score = float(s)
                best_a = a.copy()
        except Exception:
            pass

    # --- Phase 1: Adam from multiple initializations ---
    phase1_end = t0 + 0.55 * total

    inits = []
    if parent is not None and parent.size == n_primary:
        pp = parent.astype(np.float64).copy()
        pp = np.clip(pp, 1e-9, 1000)
        pp = pp / pp.sum()
        inits.append(pp)
    elif parent is not None and parent.size > 4:
        # Resample parent to n_primary
        P = parent.size
        idx = (np.arange(n_primary) * P // n_primary).astype(int)
        pp = parent[idx].astype(np.float64).copy()
        pp = np.clip(pp, 1e-9, 1000)
        pp = pp / pp.sum()
        inits.append(pp)

    t_arr = np.linspace(0, 1, n_primary)
    # Hann
    h1 = 0.5 * (1 + np.cos(2 * np.pi * t_arr))
    h1 = np.maximum(h1, 1e-8)
    h1 = h1 / h1.sum()
    inits.append(h1.copy())
    # Raised cosine (Tukey 0.5)
    tu = np.zeros(n_primary)
    m = n_primary // 4
    for i in range(n_primary):
        if i < m:
            tu[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / m - 1)))
        elif i >= n_primary - m:
            tu[i] = 0.5 * (1 + np.cos(np.pi * (2 * (i - (n_primary - m)) / m - 1)))
        else:
            tu[i] = 1.0
    tu = np.maximum(tu, 1e-8)
    tu = tu / tu.sum()
    inits.append(tu.copy())
    # Skewed Gaussian
    sg = np.exp(-1.5 * ((t_arr - 0.35) / 0.25) ** 2)
    sg = sg / sg.sum()
    inits.append(sg.copy())
    # Exponential
    ex = np.exp(-2.5 * t_arr)
    ex = ex / ex.sum()
    inits.append(ex.copy())
    # Bump at 1/3
    bm = np.exp(-8 * ((t_arr - 0.33) / 0.15) ** 2)
    bm = bm / bm.sum()
    inits.append(bm.copy())
    # Random
    rr = np.abs(rng.normal(0, 1, n_primary)) + 0.1
    rr = rr / rr.sum()
    inits.append(rr.copy())

    for i_init, init in enumerate(inits):
        if time.monotonic() > phase1_end - 2:
            break
        # Allocate more time to first init (parent)
        frac = 0.35 if i_init == 0 else max(0.08, 0.25 / (i_init + 1))
        phase_time = min(phase1_end - 2, time.monotonic() + frac * (phase1_end - t0))
        try:
            a_out, s_out = _adam_phase(init, n_primary, fs_c, fs2,
                                       phase_time, deadline, rng,
                                       tau_init=0.35 + 0.1 * i_init,
                                       lr_init=0.45)
            _update(a_out, s_out)
        except Exception:
            pass

    # --- Phase 2: L-BFGS-B refinement ---
    phase2_end = t0 + 0.78 * total
    if time.monotonic() < phase2_end - 2:
        a_cur = best_a.copy()
        a_cur = np.clip(a_cur, 1e-10, 1000)
        a_cur = a_cur / a_cur.sum()
        n_cur = a_cur.size
        fs_c2 = next_fast_len(2 * n_cur - 1)
        fs2_2 = next_fast_len(3 * n_cur - 2)

        try:
            a_ref, s_ref = _lbfgs_refine(a_cur, n_cur, fs_c2, fs2_2,
                                         phase2_end, deadline,
                                         base_taus=[0.12, 0.06, 0.03, 0.015])
            _update(a_ref, s_ref)
        except Exception:
            pass

        # Second L-BFGS round from a slightly perturbed best
        if time.monotonic() < phase2_end - 2:
            a_pert = best_a + rng.normal(0, 0.01, n_cur)
            a_pert = np.clip(a_pert, 1e-10, 1000)
            a_pert = a_pert / a_pert.sum()
            try:
                a_ref2, s_ref2 = _lbfgs_refine(a_pert, n_cur, fs_c2, fs2_2,
                                               phase2_end, deadline,
                                               base_taus=[0.08, 0.04, 0.02])
                _update(a_ref2, s_ref2)
            except Exception:
                pass

    # --- Phase 3: Adam again from best (short burst) ---
    phase3_end = t0 + 0.88 * total
    if time.monotonic() < phase3_end - 2:
        a_cur = best_a.copy()
        a_cur = np.clip(a_cur, 1e-10, 1000)
        a_cur = a_cur / a_cur.sum()
        try:
            a_out, s_out = _adam_phase(a_cur, a_cur.size, fs_c, fs2,
                                       phase3_end, deadline, rng,
                                       tau_init=0.15, lr_init=0.25)
            _update(a_out, s_out)
        except Exception:
            pass

    # --- Phase 4: Local search polish ---
    phase4_end = deadline - 1
    if time.monotonic() < phase4_end - 2:
        a_cur = best_a.copy()
        a_cur = np.clip(a_cur, 1e-10, 1000)
        a_cur = a_cur / a_cur.sum()
        n_cur = a_cur.size
        fs_c4 = next_fast_len(2 * n_cur - 1)
        fs2_4 = next_fast_len(3 * n_cur - 2)
        try:
            a_ls, s_ls = _local_search(a_cur, n_cur, fs_c4, fs2_4,
                                       phase4_end, deadline, rng)
            _update(a_ls, s_ls)
        except Exception:
            pass

    # --- Final validation ---
    try:
        cand = [float(v) for v in np.clip(best_a, 0, 1000)]
        if not cand or any(not np.isfinite(v) for v in cand):
            raise ValueError
        s_final = float(evaluate_sequence(cand))
        if not np.isfinite(s_final):
            raise ValueError
    except Exception:
        cand = [1.0]
        try:
            float(evaluate_sequence(cand))
        except Exception:
            pass

    return cand
