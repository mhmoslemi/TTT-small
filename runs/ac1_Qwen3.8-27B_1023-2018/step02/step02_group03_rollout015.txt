                break
            a_new /= sn

            v_new = sc(a_new)
            if v_new < local * (1.0 - 1e-10):
                a = a_new
                local = v_new
                a_rev = a[::-1]
                stale = 0
                lr *= 0.998
                ub(a)
            else:
                stale += 1
                vel *= 0.5
                if stale % 40 == 0:
                    lr *= 0.92
                if stale > 150:
                    ap = a + 0.03 * rng.standard_normal(n)
                    ap = np.clip(ap, 0, None)
                    sp = ap.sum()
                    if sp > 1e-15:
                        ap /= sp
                        if sc(ap) < local:
                            a = ap
                            a_rev = a[::-1]
                            local = sc(a)
                            stale = 0
                            lr = lr_init
                            vel *= 0.1
                            ub(a)
                    if stale > 250:
                        break
        return a

    def interp_len(src, n):
        if src.size == n:
            return src.copy()
        ox = np.linspace(0.0, 1.0, src.size)
        nx = np.linspace(0.0, 1.0, n)
        out = np.interp(nx, ox, src)
        out = np.clip(out, 0, None)
        s = float(out.sum())
        if s > 1e-15:
            out /= s
        else:
            out = np.ones(n, dtype=np.float64) / n
        return out

    # Asymmetric initializations
    def make_ramp(n, alpha=1.0):
        x = np.linspace(0, 1, n)
        a = 0.5 + alpha * (x - 0.5)
        a = np.clip(a, 0.01, None)
        a /= a.sum()
        return a

    def make_two_hump(n, pos=0.3):
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - pos) / 0.15) ** 2) + 0.7 * np.exp(-((x - (1 - pos)) / 0.15) ** 2)
        a = np.clip(a, 0, None)
        a /= a.sum()
        return a

    def make_polynomial(n, p=3):
        x = np.linspace(0, 1, n)
        a = x ** p + 0.5 * (1 - x) ** p
        a = np.clip(a, 1e-6, None)
        a /= a.sum()
        return a

    # === MAIN OPTIMIZATION ===
    p = parent if parent is not None else best_arr
    n_p = p.size
    p_norm = p / p.sum()
    a_main = p_norm.copy()

    # Phase 1: LP at parent length (primary workhorse)
    if linprog is not None and sps is not None:
        t_rem = deadline - time.monotonic()
        alloc = min(350.0, t_rem * 0.32)
        if alloc > 30:
            a_main = lp_optimize(a_main, 300, time.monotonic() + alloc,
                                margins=[1.0, 0.99, 0.975, 0.96, 0.94, 0.92])
            ub(a_main)

    # Phase 2: Gradient refinement
    if time.monotonic() < deadline - 60:
        t_rem = deadline - time.monotonic()
        alloc = min(200.0, t_rem * 0.22)
        if alloc > 20:
            a_main = grad_optimize(a_main, 80000, time.monotonic() + alloc, lr_init=0.004)
            ub(a_main)

    # Phase 3: Second LP pass (tighter margins)
    if time.monotonic() < deadline - 50 and linprog is not None and sps is not None:
        t_rem = deadline - time.monotonic()
        alloc = min(180.0, t_rem * 0.2)
        if alloc > 20:
            a_main = lp_optimize(a_main, 200, time.monotonic() + alloc,
                                margins=[1.0, 0.995, 0.99, 0.98, 0.97, 0.955])
            ub(a_main)

    # Phase 4: Gradient polish
    if time.monotonic() < deadline - 35:
        a_main = grad_optimize(a_main, 40000, deadline - 28, lr_init=0.003)
        ub(a_main)

    # Phase 5: Explore other lengths
    lengths_to_try = []
    for off in [-10, -5, -2, 2, 5, 10, 20, 40, 60]:
        L = n_p + off
        if 32 <= L <= 640:
            lengths_to_try.append(L)
    for L in [100, 128, 160, 192, 256, 448, 512, 576]:
        if L != n_p and 32 <= L <= 640:
            lengths_to_try.append(L)
    lengths_to_try = sorted(set(lengths_to_try))

    for L in lengths_to_try:
        if time.monotonic() >= deadline - 35:
            break
        t_rem = deadline - time.monotonic() - 28
        if t_rem < 14:
            break
        alloc = min(40.0, t_rem * 0.12)
        if alloc < 10:
            break

        a_init = interp_len(a_main, L)

        if linprog is not None and sps is not None:
            a_w = lp_optimize(a_init, max(8, int(alloc * 0.4)),
                             time.monotonic() + alloc * 0.55,
                             margins=[1.0, 0.97, 0.94])
            ub(a_w)
        else:
            a_w = a_init

        if time.monotonic() < deadline - 25:
            a_w = grad_optimize(a_w, max(3000, int(alloc * 4)), deadline - 20)
            ub(a_w)

    # Phase 6: Diverse restarts at parent length
    if time.monotonic() < deadline - 30:
        n_restarts = 12
        inits = [
            lambda: a_main * np.exp(0.1 * rng.standard_normal(n_p)),
            lambda: a_main[::-1].copy(),
            lambda: make_ramp(n_p, 1.2) * (a_main / a_main.mean()),
            lambda: make_two_hump(n_p),
            lambda: make_polynomial(n_p, 4),
            lambda: a_main * np.exp(0.2 * rng.standard_normal(n_p)),
        ]
        for r in range(n_restarts):
            if time.monotonic() >= deadline - 20:
                break
            t_rem = deadline - time.monotonic() - 15
            alloc = min(30.0, t_rem / max(1, n_restarts - r))
            if alloc < 7:
                break

            try:
                a_init = inits[r % len(inits)]()
            except Exception:
                a_init = a_main * np.exp(0.1 * rng.standard_normal(n_p))
            a_init = np.clip(a_init, 0, None)
            si = float(a_init.sum())
            if si < 1e-15:
                a_init = a_main.copy()
                si = 1.0
            a_init /= si

            if linprog is not None and sps is not None:
                a_w = lp_optimize(a_init, max(6, int(alloc * 0.5)),
                                  time.monotonic() + alloc * 0.6,
                                  margins=[1.0, 0.96, 0.93])
                ub(a_w)
            else:
                a_w = a_init

            if time.monotonic() < deadline - 15:
                a_w = grad_optimize(a_w, int(alloc * 3), deadline - 12, lr_init=0.003)
                ub(a_w)

    # Phase 7: Final polish on best
    if time.monotonic() < deadline - 8 and best_arr is not None:
        n_b = best_arr.size
        a_pol = best_arr.copy()
        sp = a_pol.sum()
        if sp > 1e-15:
            a_pol /= sp
        a_pol = grad_optimize(a_pol, 8000, deadline - 5, lr_init=0.002)
        ub(a_pol)

    # Final fallback
    if best_arr is None:
        best_arr = np.ones(320, dtype=np.float64)
        ub(best_arr)

    final = np.clip(best_arr, 0.0, 1000.0)
    s_final = float(final.sum())
    if s_final < 0.01:
        final = np.ones(320, dtype=np.float64)
        ub(final)

    # Verify with exact evaluator
    v_check = evaluate_sequence([float(x) for x in best_arr])
    if not math.isfinite(v_check):
        # Fallback to a valid sequence
        fb = np.ones(320, dtype=np.float64)
        v_fb = evaluate_sequence([float(x) for x in fb])
        if math.isfinite(v_fb):
            return [float(x) for x in fb]
        # Last resort
        return [1.0] * 320

    return [float(x) for x in best_arr]
