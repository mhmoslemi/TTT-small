    """Full self-convolution via FFT."""
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2*n-1]
    c[~np.isfinite(c)] = 0.0
    return c

    """Compute the AC1 upper bound (score)."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = _fft_conv(a)
    return 2.0 * n * c.max() / (s * s)

    n = len(a)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):

        jhi = min(n - 1, k)
        for j in range(jlo, jhi + 1):
            A[k, j] = a[k - j]
    return A

    ag = _fft_conv_(a, g)  # cross-convolution a * g
    gg = _fft_conv(g)
