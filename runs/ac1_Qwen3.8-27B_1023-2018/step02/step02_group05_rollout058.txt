```python
def _proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}"""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w / w.sum()
```

            for t in ts_list:
                cand = (1.0 - t) * a + t * g
                cand = np.clip(cand, 0.0, None)
                sc = _score(cand)
                if math.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_cand = cand

            if best_cand is not None and best_sc < local * (1.0 - 1e-9):
                a = np.clip(best_cand, 0.0, None)
                s = float(a.sum())
                if s > 1e-15:
                    a = a / s
                local = best_sc
                stagnant = 0
                _update_best(a, best)
            else:
                stagnant += 1
        else:
            stagnant += 1

        # Subgradient refinement every few iterations
        if stagnant > 5 and substep_count < 200:
            a, local = _subgrad_refine(a, 0.05, 20, deadline)
            substep_count += 20
            _update_best(a, best)
            if local < TARGET:
                break
            stagnant = 0
            substep_count += 1  # just for counting

        if stagnant >= max_stag:
            # Random restart
            pa = a * np.exp(0.1 * rng.standard_normal(n))
            pa = np.clip(pa, 0.0, None)
            ps = float(pa.sum())
            if ps > 1e-15:
                pa = pa / ps
                pscore = _score(pa)
                if math.isfinite(pscore) and pscore < local:
                    a = pa
                    local = pscore
                else:
                    # Try anti-palindromic
                    half = n // 2
                    v = np.clip(0.3 * rng.standard_normal(half), -0.5, 0.5)
                    fa = np.ones(n)
                    fa[:half] = 1.0 + v
                    fa[half:] = 1.0 - v[::-1]
                    fa = np.clip(fa, 0.01, None)
                    fa = fa / fa.sum()
                    fa_score = _score(fa)
                    if math.isfinite(fa_score) and fa_score < local:
                        a = fa
                        local = fa_score
            stagnant = 0

    return a, local
