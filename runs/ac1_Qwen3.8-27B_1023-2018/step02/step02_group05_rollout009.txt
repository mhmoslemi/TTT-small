            parent = np.clip(parent, 0.0, 1000.0)
            if parent.size > 1:
                _update(parent, best)
    except Exception:
        parent = None

    def _rem():
        return max(0.0, deadline - time.monotonic())

    n_par = parent.size if parent is not None else 0

    # Phase 1: Main LP on parent length (heavy investment)
    if parent is not None and n_par >= 8:
        alloc = min(550.0, _rem() * 0.5)
        if alloc > 30:
            _lp_loop(parent, n_par, alloc, rng, best)

    # Phase 2: LP on perturbed parent (same length)
    if parent is not None and n_par >= 8 and _rem() > 80:
        x = np.linspace(0.0, 1.0, n_par)
        for asym in (0.8, -0.6, 1.5):
            if _rem() < 40:
                break
            pa = parent * (1.0 + asym * x * 0.5)
            pa = np.clip(pa, 0, None)
            ps = float(pa.sum())
            if ps > 0:
                pa /= ps
                alloc = min(150.0, _rem() * 0.3)
                if alloc > 20:
                    _lp_loop(pa, n_par, alloc, rng, best)

    # Phase 3: Different lengths
    for ntr in (256, 384, 224, 300):
        if _rem() < 60:
            break
        if parent is not None and n_par > 1:
            tp = np.linspace(0.0, 1.0, n_par)
            xn = np.linspace(0.0, 1.0, ntr)
            a_init = np.interp(xn, tp, parent)
        else:
            a_init = np.ones(ntr)
        x = np.linspace(0.0, 1.0, ntr)
        a_init = a_init * (1.0 + 0.4 * x)
        a_init = np.clip(a_init, 0, None)
        s = float(a_init.sum())
        if s > 0:
            a_init /= s
        alloc = min(70.0, _rem() * 0.18)
        if alloc > 15:
            _lp_loop(a_init, ntr, alloc, rng, best)

    # Phase 4: Final polish on best found
    if _rem() > 25:
        best_arr = np.asarray(best[1], dtype=np.float64).ravel()
        nb = best_arr.size
        if nb < 4:
            nb = 320
            best_arr = np.ones(nb) if parent is None else parent[:nb]
        alloc = min(50.0, _rem() * 0.6)
        if alloc > 10:
            _lp_loop(best_arr, nb, alloc, rng, best)

    # Final validation
    final = np.asarray(best[1], dtype=np.float64).ravel()
    final = np.nan_to_num(final, nan=0.0, posinf=0.0, neginf=0.0)
    final = np.clip(final, 0.0, 1000.0)
    if final.size == 0:
        final = np.ones(16, dtype=np.float64)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(16, dtype=np.float64)

    fin_sc = evaluate_sequence([float(x) for x in final])
    if not math.isfinite(fin_sc) or fin_sc > best[0] + 0.01:
        final = np.ones(16, dtype=np.float64)
        fin_sc = evaluate_sequence([float(x) for x in final])

    return [float(x) for x in final]
