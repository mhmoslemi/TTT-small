import numpy as np
import time
import math
from scipy.optimize import linprog
import scipy.sparse as sps

def propose_candidate(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    try:
        budget = max(10.0, float(budget_s))
    except:
        budget = 1300.0
    deadline = t0 + budget - 15.0

    def score(a):
        a = np.asarray(a, dtype=np.float64).ravel()
        a = np.clip(a, 0.0, 1000.0)
        n = a.size
        if n < 1:
            return float('inf')
        s = float(a.sum())
        if s < 0.01:
            return float('inf')
        b = np.convolve(a, a)
        return float(2.0 * n * b.max() / (s * s))

    best_score = float('inf')
    best_arr = None

    def update_best(a):
        nonlocal best_score, best_arr
        try:
            a = np.asarray(a, dtype=np.float64).ravel()
            a = np.clip(a, 0.0, 1000.0)
            if a.size == 0:
                return
            s = float(a.sum())
            if s < 0.01:
                return
            v = evaluate_sequence([float(x) for x in a])
            if math.isfinite(v) and v < best_score:
                best_score = v
                best_arr = a.copy()
        except:
            pass

    # Get parent
    parent = None
    try:
        h = globals().get('height_sequence_1', None)
        if h is not None:
            parent = np.asarray(h, dtype=np.float64).ravel().copy()
            parent = np.clip(parent, 0.0, 1000.0)
            if parent.size > 0 and parent.sum() > 0.01:
                update_best(parent)
    except:
        pass

    if best_arr is None:
        best_arr = np.ones(96, dtype=np.float64)
        update_best(best_arr)

    n_p = best_arr.size

    # === LP-based optimizer ===
    def lp_iterate(a, stop_time, max_iters=1000, margins=(1.0, 0.98, 0.96, 0.94, 0.92)):
        nonlocal best_score, best_arr
        a = np.asarray(a, dtype=np.float64).ravel().copy()
        a = np.clip(a, 0, None)
        s = a.sum()
        if s < 1e-15:
            return a
        a /= s
        n = a.size
        local = score(a)
        update_best(a)
        stag = 0

        for it in range(max_iters):
            if time.monotonic() >= stop_time - 3:
                break

            b_conv = np.convolve(a, a)
            M = float(b_conv.max())
            if not math.isfinite(M) or M <= 0:
                break

            # Build sparse convolution matrix: A[k,j] = a[k-j] if 0<=k-j<n
            # Shape: (2n-1) x n
            # Each row k has nonzeros at j = max(0,k-n+1) .. min(n-1, k)
            rows = []
            cols = []
            vals = []
            for k in range(2 * n - 1):
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                for j in range(lo, hi + 1):
                    rows.append(k)
                    cols.append(j)
                    vals.append(a[k - j])
            A = sps.coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()

            improved = False
            for m in margins:
                if time.monotonic() >= stop_time - 3:
                    break
                b_ub = M * m + 1e-10
                c = -np.ones(n)
                try:
                    res = linprog(c, A_ub=A, b_ub=b_ub,
                                  bounds=[(0, None)] * n,
                                  method='highs',
                                  options={'presolve': True, 'time_limit': max(1, stop_time - time.monotonic() - 5)})
                except:
                    continue
                if res is None or not res.success:
                    # Try anyway if we got a solution
                    if res is None or getattr(res, 'x', None) is None:
                        continue
                g = np.clip(res.x, 0, None)
                sg = float(g.sum())
                if sg < 1e-15:
                    continue
                g /= sg

                # Line search
                best_sc_t = local
                best_c_t = None
                for t in (0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 0.9, 1.0):
                    cand = (1.0 - t) * a + t * g
                    v = score(cand)
                    if v < best_sc_t:
                        best_sc_t = v
                        best_c_t = cand

                if best_c_t is not None and best_sc_t < local - 1e-12:
                    a = best_c_t / best_c_t.sum()
                    local = best_sc_t
                    stag = 0
                    improved = True
                    update_best(a)
                    break

            if not improved:
                stag += 1
                if stag > 5:
                    # Perturbation
                    ap = a * np.exp(0.1 * rng.standard_normal(n))
                    ap = np.clip(ap, 0, None)
                    if ap.sum() > 1e-15:
                        ap /= ap.sum()
                        if score(ap) < local:
                            a, local = ap, score(ap)
                            stag = 0
                            update_best(a)
                if stag > 30:
                    break

        return a

    # === Gradient-based optimizer ===
    def grad_iterate(a, stop_time, max_iters=200000):
        nonlocal best_score, best_arr
        a = np.asarray(a, dtype=np.float64).ravel().copy()
        a = np.clip(a, 0, None)
        s = a.sum()
        if s < 1e-15:
            return a
        a /= s
        n = a.size
        local = score(a)
        update_best(a)
        stale = 0

        for it in range(max_iters):
            if time.monotonic() >= stop_time - 2:
                break
            if it % 1000 == 0 and time.monotonic() >= stop_time:
                break

            b_conv = np.convolve(a, a)
            M = float(b_conv.max())
            if not math.isfinite(M):
                break

            # Subgradient: use all k's within 1% of max
            thr = M * 0.999
            active = np.where(b_conv >= thr)[0]

            # Gradient of max by weighted sum
            weights = (b_conv[active] - thr + 1e-10)
            weights /= weights.sum()
            
            # grad_m = sum_{k in active} w_k * 2 * a[k-m]
            # This is: grad = 2 * (weighted sum of shifted a's)
            grad = np.zeros(n)
            for w_k, k in zip(weights, active):
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                js = np.arange(lo, hi + 1)
                grad[js] += 2.0 * w_k * a[k - js]

            # Project gradient: subtract mean to maintain sum invariance
            grad -= grad.mean()

            gn = np.linalg.norm(grad)
            if gn < 1e-15:
                break
            grad /= gn

            # Line search
            best_sc_t = local
            best_c_t = None
            best_alpha = 0
            for alpha in (0.001, 0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2):
                cand = a - alpha * grad
                cand = np.clip(cand, 0, None)
                cs = float(cand.sum())
                if cs < 1e-15:
                    continue
                cand /= cs
                v = score(cand)
                if v < best_sc_t:
                    best_sc_t = v
                    best_c_t = cand
                    best_alpha = alpha

            if best_c_t is not None and best_sc_t < local - 1e-13:
                a = best_c_t
                local = best_sc_t
                stale = 0
                if it % 100 == 0:
                    update_best(a)
            else:
                stale += 1
                if stale > 50:
                    ap = a + 0.02 * rng.standard_normal(n)
                    ap = np.clip(ap, 0, None)
                    if ap.sum() > 1e-15:
                        ap /= ap.sum()
                        v = score(ap)
                        if v < local:
                            a, local = ap, v
                            stale = 0
                            update_best(a)
                if stale > 500:
                    break

        update_best(a)
        return a

    # === Interpolation ===
    def interp_len(src, n):
        if src.size == n:
            return src.copy()
        ox = np.linspace(0.0, 1.0, src.size)
        nx = np.linspace(0.0, 1.0, n)
        out = np.interp(nx, ox, src)
        out = np.clip(out, 0, None)
        s = float(out.sum())
        if s > 1e-15:
            out /= s
        return out

    # === MAIN ===
    a_work = best_arr.copy() / best_arr.sum()

    # Phase 1: LP at parent length
    t1 = min(500.0, (deadline - time.monotonic()) * 0.45)
    if t1 > 30:
        a_work = lp_iterate(a_work, time.monotonic() + t1, max_iters=500)

    # Phase 2: Gradient polish
    if time.monotonic() < deadline - 60:
        a_work = grad_iterate(a_work, deadline - 50, max_iters=500000)

    # Phase 3: Second LP pass
    if time.monotonic() < deadline - 40:
        t3 = min(200.0, (deadline - time.monotonic()) - 30)
        if t3 > 20:
            a_work = lp_iterate(a_work, time.monotonic() + t3, max_iters=300,
                              margins=(1.0, 0.99, 0.97, 0.95))
            if time.monotonic() < deadline - 15:
                a_work = grad_iterate(a_work, deadline - 10, max_iters=200000)

    # Phase 4: Try nearby lengths
    for L in [n_p - 4, n_p - 2, n_p + 2, n_p + 4, n_p - 8, n_p + 8,
              280, 300, 340, 360, 380, 400]:
        if time.monotonic() >= deadline - 30:
            break
        t_rem = deadline - time.monotonic() - 30
        if t_rem < 15 or L < 8 or L > 600:
            continue
        if L == n_p:
            continue
        alloc = min(30.0, t_rem * 0.15)
        if alloc < 8:
            continue

        a_init = interp_len(best_arr, L)
        a_w = lp_iterate(a_init, time.monotonic() + alloc * 0.6, max_iters=50)
        if time.monotonic() < deadline - 15:
            a_w = grad_iterate(a_w, deadline - 12, max_iters=50000)

    # Phase 5: Final polish
    if time.monotonic() < deadline - 8 and best_arr is not None:
        a_pol = best_arr.copy() / best_arr.sum()
        a_pol = grad_iterate(a_pol, deadline - 4, max_iters=100000)
        update_best(a_pol)

    # Return
    if best_arr is None:
        best_arr = np.ones(96)
        update_best(best_arr)

    final = np.clip(best_arr, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(96)
    update_best(final)

    return [float(x) for x in best_arr]

```python
def lp_project(a, M):
    """Find b closest to a (in L1) satisfying A b <= M, b >= 0."""
    n = len(a)
    # Variables: b_0, ..., b_{n-1}, u_0, ..., u_{n-1}
    # Objective: min sum u_j
    # Constraints:
    #   b_j - a_j <= u_j  for j = 0,...,n-1
    #   a_j - b_j <= u_j  for j = 0,...,n-1
    #   (A b)_k <= M      for k = 0,...,2n-2
    #   b_j >= 0           for j = 0,...,n-1  (handled by bounds)
    #   u_j >= 0           (handled by bounds)
    
    # c = [0, ..., 0, 1, ..., 1] (first n are 0 for b, next n are 1 for u)
    c = np.concatenate([np.zeros(n), np.ones(n)])
    
    # Build constraint matrix
    # (1) b_j - u_j <= a_j  =>  row: e_j^T_b - e_j^T_u, rhs: a_j
    # (2) -b_j - u_j <= -a_j  =>  row: -e_j^T_b - e_j^T_u, rhs: -a_j
    # (3) (A b)_k <= M  =>  row: A_k, zeros, rhs: M
    # Total rows: 2n + (2n-1) = 4n-1
    
    # Build sparse
    nvar = 2 * n
    nrows = 2 * n + (2 * n - 1)
    
    # Start with the A matrix (2n-1 rows, n cols for b)
    A_conv = build_conv_mat(a)  # (2n-1) x n
    
    # For constraint (3), pad with zeros for u variables
    A3 = sps.hstack([A_conv, sps.csr_matrix((2*n-1, n))], format='csr')  # (2n-1) x 2n
    
    # For constraints (1) and (2)
    I = sps.eye(n, format='csr')
    A1 = sps.hstack([I, -I])  # (n) x 2n
    A2 = sps.hstack([-I, -I])  # (n) x 2n
    b1 = a.copy()
    b2 = -a.copy()
    
    A_full = sps.vstack([A1, A2, A3], format='csr')
    b_full = np.concatenate([b1, b2, np.full(2*n-1, M)])
    
    bounds = [(0, None)] * n + [(0, None)] * n
    
    res = linprog(c, A_ub=A_full, b_ub=b_full, bounds=bounds, method='highs')
    
    if res.success and res.x is not None:
        return np.clip(res.x[:n], 0, None)
    return None
```
