def _lp_direction_from_A(A, rows=None, time_limit=10.0):

def _subgradient_candidates(a, b, frac=0.05, max_out=3):
    try:
        n = int(a.size)
        if n < 2 or b.size == 0:
            return []
        M = float(np.max(b))
        if not math.isfinite(M) or M <= 1e-300:
            return []

        active = np.where(b >= M - frac * M)[0]
        if active.size == 0:
            active = np.array([int(np.argmax(b))], dtype=np.int64)
        if active.size > 120:
            active = np.argpartition(b, -120)[-120:]

        grad = np.zeros(n, dtype=np.float64)
        for k in active:
            k = int(k)
            jmin = max(0, k - (n - 1))
            jmax = min(n - 1, k)
            if jmin > jmax:
                continue
            idx = np.arange(jmin, jmax + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]
        grad /= float(active.size)

        scale = float(np.max(grad))
        if not math.isfinite(scale) or scale <= 1e-300:
            return []

        out = []
        scores = []
        for eta in (0.01, 0.02, 0.05, 0.10, 0.20, 0.50):
            h = np.clip(a - eta * grad / scale, 0.0, None)
            h = _normalize(h)
            if h is None:
                continue
            sc = _score_fast(h)
            if math.isfinite(sc):
                out.append(h)
                scores.append(sc)

        if not out:
            return []
        order = np.argsort(np.asarray(scores, dtype=np.float64))[:min(int(max_out), len(scores))]
        return [out[int(i)] for i in order]
    except Exception:
        return []

                if rows10.size > 120:
                    rows10 = np.argpartition(b, -120)[-120:]
                variants.append(rows10)
                if m > 50:
                    variants.append(np.argpartition(b, -50)[-50:])
                rows25 = np.where(b >= M - 0.25 * M)[0]
                if rows25.size > 120:
                    rows25 = np.argpartition(b, -120)[-120:]
                variants.append(rows25)

                for rs in variants:
                    rs = np.asarray(rs, dtype=np.int64)
                    rs = rs[(rs >= 0) & (rs < m)]
                    if rs.size == 0 or rs.size >= m:
                        continue
                    rs = np.unique(rs)
                    g_act = _lp_direction_from_A(A, rows=rs, time_limit=min(20.0, tl_full * 0.6))
                    if g_act is not None:
                        directions.append(g_act)
                    if len(directions) >= 5:
                        break

            directions.extend(_subgradient_candidates(a, b, frac=0.05, max_out=2))

            best_cand = None
            best_fast = local_fast
            for h in directions:
                if time.monotonic() >= deadline:
                    break
                cand, fsc = _line_search_fast(a, h, local_fast)
                if cand is not None and math.isfinite(fsc) and fsc < best_fast - 1e-14:
                    best_fast = float(fsc)
                    best_cand = cand

            if best_cand is not None:
                esc = _eval_exact(best_cand)
                if math.isfinite(esc) and esc < local_exact - 1e-12:
                    a = np.asarray(best_cand, dtype=np.float64).ravel()
                    local_fast = float(best_fast)
                    local_exact = float(esc)
                    _update_best(best, a)
                    stagnant = 0
                    continue

            pert = _random_step(a, local_fast, rng)
            if pert is not None:
                pes = _eval_exact(pert)
                pf = _score_fast(pert)
                if math.isfinite(pes) and pes < local_exact - 1e-12:
                    a = np.asarray(pert, dtype=np.float64).ravel()
                    local_exact = float(pes)
                    local_fast = float(pf)
                    _update_best(best, a)
                    stagnant = 0
                    continue

            stagnant += 1
            if stagnant >= 12:
                base = _normalize(_resample(best[1], n)) if best[1] is not None and len(best[1]) else None
                if base is None:
                    base = np.ones(n, dtype=np.float64) / float(n)
                restarted = False
                for sig in (0.01, 0.02, 0.05, 0.10, 0.20):
                    c = _normalize(np.clip(base + sig * rng.standard_normal(n), 0.0, None))
                    if c is None:
                        continue
                    cf = _score_fast(c)
                    ce = _eval_exact(c)
                    if math.isfinite(cf):
                        a = np.asarray(c, dtype=np.float64).ravel()
                        local_fast = float(cf)
                        local_exact = float(ce) if math.isfinite(ce) else float("inf")
                        _update_best(best, a)
                        restarted = True
                        break
                stagnant = 5 if restarted else stagnant

            if stagnant > 320:
                break

        _update_best(best, a)
    except Exception:
        pass


def _de_phase(init_arrays, n_target, time_limit, rng, best):
    try:
        n = int(n_target)
        if n < 16 or n > 1000:
            return
        if time_limit is None or float(time_limit) <= 0:
            return
        deadline = time.monotonic() + max(0.0, float(time_limit)) - 10.0
        P = 14 if n <= 400 else 8

        def softmax(x):
            x = np.asarray(x, dtype=np.float64).ravel()
            x = x - float(np.max(x))
            ex = np.exp(np.clip(x, -700.0, 700.0))
            s = float(ex.sum())
            if not math.isfinite(s) or s <= 1e-300:
                return np.ones(n, dtype=np.float64) / float(n)
            return ex / s

        def logits(a):
            a = _normalize(a)
            if a is None:
                a = np.ones(n, dtype=np.float64) / float(n)
            a = np.clip(a, 1e-300, None)
            x = np.log(a)
            x -= float(np.mean(x))
            return np.clip(x, -30.0, 30.0)

        pop = []

        def add(a):
            c = _normalize(_resample(a, n))
            if c is None:
                return
            x = logits(c)
            sc = _score_fast(c)
            if math.isfinite(sc):
                pop.append([float(sc), x, np.asarray(c, dtype=np.float64).ravel()])

        seen_arr = []

        def add_once(a):
            try:
                c = _normalize(_resample(a, n)).astype(np.float64)
                key = tuple(c[::max(1, c.size // 16)])
                for q in seen_arr:
                    if q.shape == c.shape and float(np.max(np.abs(q - c))) < 1e-12:
                        return
                seen_arr.append(c)
                add(a)
            except Exception:
                add(a)

        if init_arrays:
            for arr in init_arrays:
                add_once(arr)
                if len(pop) >= P:
                    break
        add_once(np.ones(n, dtype=np.float64))
        for _ in range(4):
            r = _random_shape(n, rng)
            if len(pop) >= P:
                break
            add_once(r)
        while len(pop) < P:
            r = _random_shape(n, rng)
            add_once(r)

        noimp = 0
        while time.monotonic() < deadline:
            rem = deadline - time.monotonic()
            if rem < 10.0:
                break
            if best[0] <= TARGET + 1e-12:
                break

            for _ in range(P):
                if time.monotonic() >= deadline:
                    break
                if len(pop) < 3:
                    r = _random_shape(n, rng)
                    add_once(r)
                    if len(pop) < 3:
                        break
                    continue

                r1, r2, r3 = rng.choice(len(pop), size=3, replace=False)
                F = float(rng.uniform(0.4, 0.9))
                xmut = pop[int(r1)][1] + F * (pop[int(r2)][1] - pop[int(r3)][1])
                if rng.random() < 0.5:
                    ib = min(range(len(pop)), key=lambda ii: pop[ii][0])
                    xmut = 0.6 * pop[ib][1] + 0.4 * xmut
                xmut = xmut + float(rng.uniform(0.05, 0.25)) * rng.standard_normal(n)
                xmut = np.clip(xmut, -30.0, 30.0)
                xmut = xmut - float(np.mean(xmut))
                am = softmax(xmut)
                sm = _score_fast(am)
                if not math.isfinite(sm):
                    noimp += 1
                    continue

                iw = max(range(len(pop)), key=lambda ii: pop[ii][0])
                if sm < float(pop[iw][0]) - 1e-12:
                    pop[iw] = [float(sm), np.asarray(xmut, dtype=np.float64).ravel(),
                               np.asarray(am, dtype=np.float64).ravel()]
                    if sm < best[0] - 1e-10:
                        _update_best(best, pop[iw][2])
                        noimp = 0
                    else:
                        noimp += 1
                else:
                    noimp += 1

            if noimp > 60:
                for i in range(max(1, P // 2)):
                    r = _random_shape(n, rng)
                    if len(pop) <= i:
                        if r is not None:
                            x = logits(r)
                            sc = _score_fast(r)
                            pop.append([float(sc), x, np.asarray(r, dtype=np.float64).ravel()])
                    else:
                        if r is not None:
                            x = logits(r)
                            sc = _score_fast(r)
                            pop[i] = [float(sc), x, np.asarray(r, dtype=np.float64).ravel()]
                noimp = 0

        for sc, x, a in pop:
            if math.isfinite(sc) and sc < best[0] + 1e-7:
                _update_best(best, a)
    except Exception:
        pass


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    try:
        seed_i = int(seed)
    except Exception:
        seed_i = 42
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1300.0
    if not math.isfinite(budget) or budget <= 0.0:
        budget = 1300.0

    t_start = time.monotonic()
    head = max(5.0, min(60.0, 0.05 * budget))
    deadline = t_start + budget - head

    rng = np.random.default_rng(seed_i)
    best = [float("inf"), np.ones(320, dtype=np.float64)]
    _update_best(best, best[1])

    parent = None
    try:
        h = globals().get("height_sequence_1")
        if h is not None:
            parent = _clean(h)
            if parent.size > 0:
                _update_best(best, parent)
    except Exception:
        parent = None

    def rem():
        return max(0.0, deadline - time.monotonic())

    seed_lengths = []
    parent_n = int(parent.size) if parent is not None and parent.size > 0 else 320
    for n in (parent_n, 64, 96, 128, 160, 192, 256, 320, 384, 448, 512, 640, 768):
        if 16 <= n <= 1000 and n not in seed_lengths:
            seed_lengths.append(n)

    for n in seed_lengths:
        if rem() <= 3.0:
            break
        if parent is not None and parent.size > 1:
            init = _resample(parent, n)
        else:
            init = np.ones(n, dtype=np.float64)
        a = _normalize(init)
        if a is not None:
            _update_best(best, a)
        x = np.linspace(0.0, 1.0, n)
        a2 = _normalize(init * (1.0 + 0.5 * x))
        if a2 is not None:
            _update_best(best, a2)
        a3 = _normalize(np.exp(0.5 * x) * init)
        if a3 is not None:
            _update_best(best, a3)

    if not math.isfinite(best[0]):
        fb = np.ones(320, dtype=np.float64)
        sc = _eval_exact(fb)
        best[0] = float(sc) if math.isfinite(sc) else 2.0
        best[1] = fb

    n_best = int(best[1].size) if best[1] is not None and len(best[1]) else 320
    if n_best > 800:
        n_best = 768
    if rem() > 120.0:
        alloc = min(0.55 * budget, rem())
        _lp_search(best[1], n_best, alloc, rng, best)

    n_best = int(best[1].size) if best[1] is not None and len(best[1]) else 320
    alt_lengths = []
    for cand in (n_best + 64, n_best + 32, n_best - 32,
                 320, 384, 448, 512, 640, 768,
                 256, 192, 128):
        if 32 <= cand <= 800 and cand not in alt_lengths:
            alt_lengths.append(cand)

    for alt in alt_lengths[:3]:
        if rem() < 60.0:
            break
        alloc = min(0.12 * budget, rem())
        _lp_search(best[1], alt, alloc, rng, best)

    if rem() > 60.0:
        n_best = int(best[1].size) if best[1] is not None and len(best[1]) else 320
        if n_best > 800:
            n_best = 768
        alloc = min(0.15 * budget, rem())
        _lp_search(best[1], n_best, alloc, rng, best)

    if rem() > 40.0:
        n_de = int(best[1].size) if best[1] is not None and len(best[1]) else 320
        if n_de > 900:
            n_de = 768
        alloc = min(0.12 * budget, rem())
        seeds = [best[1]]
        if parent is not None:
            seeds.append(parent)
        _de_phase(seeds, n_de, alloc, rng, best)

    if rem() > 20.0:
        n_fin = int(best[1].size) if best[1] is not None and len(best[1]) else 320
        if n_fin > 800:
            n_fin = 768
        alloc = min(0.05 * budget, rem())
        _lp_search(best[1], n_fin, alloc, rng, best)

    final = _clean(best[1]) if best[1] is not None else np.array([], dtype=np.float64)
    if final.size == 0:
        final = np.ones(320, dtype=np.float64)

    s = float(final.sum())
    if not math.isfinite(s) or s <= 1e-300:
        final = np.ones(320, dtype=np.float64)
        s = 320.0
    else:
        final = final / s

    sc = _eval_exact(final)
    if not math.isfinite(sc):
        final = np.ones(320, dtype=np.float64)
        s = float(final.sum())
        if s > 1e-300:
            final = final / s
        sc = _eval_exact(final)
        if not math.isfinite(sc):
            final = np.ones(1, dtype=np.float64)

    return [float(x) for x in np.asarray(final, dtype=np.float64).ravel()]
