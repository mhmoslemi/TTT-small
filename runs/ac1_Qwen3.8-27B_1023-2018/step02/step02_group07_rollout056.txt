def lp_update(a, M, method='highs'):
    n = len(a)
    c = -np.ones(n)  # minimize -sum b = maximize sum b
    # Constraints: (a*b)_k <= M for k = 0, ..., 2n-2
    # (a*b)_k = sum_j a_j * b_{j+k} (with b_j = 0 for j<0 or j>=n)
    # For k=0: sum_j a_j b_j <= M
    # For k=1: sum_j a_j b_{j+1} <= M => sum_j a_{j-1} b_j <= M
    # ...
    # For general k: sum_j a_{j-k} b_j <= M (with a_i = 0 for i outside [0,n-1])
    
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            i = j - k
            if 0 <= i < n:
                A_ub[k, j] = a[i]
    
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return np.ones(n) / n

```python
def _grad_logspace(a, b, w, n):
    # g_i = 2 * sum_{k=0}^{2n-2} w_k * a_{k-i}
    # For each i, the valid range of k is: k-i in [0, n-1], i.e., k in [i, i+n-1]
    # g_i = 2 * sum_{k=i}^{i+n-1} w_k * a_{k-i}
    #      = 2 * sum_{j=0}^{n-1} w_{i+j} * a_j
    
    g = np.zeros(n)
    for i in range(n):
        # w[i:i+n] * a[0:n]
        g[i] = 2.0 * np.dot(w[i:i+n], a)
    
    mean_g = np.dot(a, g)
    grad = a * (g - mean_g)
    return grad
```

This is O(n^2) with small constant, fine for n=192.

idx_j = np.arange(n)[None, :]
