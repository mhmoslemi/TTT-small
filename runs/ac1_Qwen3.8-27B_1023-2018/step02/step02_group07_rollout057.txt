import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _conv(a, b, n):
    """Compute full convolution of a and b, both of length n."""
    fs = next_fast_len(2 * n - 1)
    return irfft(rfft(a, fs) * rfft(b, fs), fs)[:2*n-1]

def _hard_score(a):
    n = len(a)
    b = _conv(a, a, n)
    m = float(b.max())
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    return 2.0 * n * m / (s * s)

def _lp_step(a, M, n):
    """
    Solve LP: max sum(b) subject to convolve(a, b)[k] <= M for all k, b >= 0.
    Returns the optimal b.
    """
    # Variables: b[0], ..., b[n-1]
    # Objective: maximize sum(b) => minimize -sum(b)
    c = -np.ones(n)
    
    # Constraints: for k = 0, ..., 2n-2
    # sum_j a[j] * b[k-j] <= M
    # where j ranges over valid indices: 0 <= j <= n-1, 0 <= k-j <= n-1
    # So j ranges from max(0, k-(n-1)) to min(n-1, k)
    
    A_ub = np.zeros((2*n-1, n))
    b_ub = np.full(2*n-1, M)
    
    for k in range(2*n-1):
        j_min = max(0, k - (n-1))
        j_max = min(n-1, k)
        for j in range(j_min, j_max+1):
            A_ub[k, k-j] = a[j]
    
    # b >= 0 is handled by bounds
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    
    if res.success:
        return res.x
    else:
        return None

def _iterative_lp_optimize(a, max_iter, time_limit, seed=42):
    n = len(a)
    a = a.copy()
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 1e-10:

    else:
        a = a / s  # normalize to sum=1
    
    best_a = a.copy()
    best_score = _hard_score(best_a)
    
    t = 0.1  # step size
    min_t = 0.01
    
    for it in range(max_iter):
        if time.monotonic() > time_limit:
            break
        
        b_conv = _conv(a, a, n)
        M = float(b_conv.max())
        
        # Solve LP

        if b_opt is None or not np.isfinite(b_opt).all():
            break
        
        b_opt = np.clip(b_opt, 0, 1000)
        s_opt = b_opt.sum()
        if s_opt < 1e-10:

        b_opt = b_opt / s_opt  # normalize to sum=1
        
        # Update

        a_new = np.clip(a_new, 0, 1000)

        a_new = a_new / s_new

        if score < best_score:

            best_score = score
            t = min(t * 1.02, 0.5)  # increase step size slightly

def _conv_full(a, b, n):
    """Full convolution of two length-n arrays."""
    fs = next_fast_len(2 * n - 1)
    Fa = rfft(a, fs)
    Fb = rfft(b, fs)
    return irfft(Fa * Fb, fs)[:2 * n - 1]
