With softmax: `a[j] = exp(z[j]) / Z` where `Z = sum_k exp(z[k])`.

Score (with sum a = 1): `S = 2n * max(b) = 2n * b[m*]`

`b[m*] = sum_j a[j] * a[m*-j]`

`∂b[m*]/∂a[j] = 2 * a[m*-j]` (if 0 <= m*-j < n, else 0) [as derived above]

`∂a[j]/∂z[k] = a[j](δ[j,k] - a[k])`

`∂S/∂z[k] = sum_j (∂S/∂a[j]) * (∂a[j]/∂z[k])`
`= 2n * sum_j 2*a[m*-j] * a[j] * (δ[j,k] - a[k])`
`= 4n * (a[m*-k] * a[k] - a[k] * sum_j a[j] * a[m*-j])`
`= 4n * a[k] * (a[m*-k] - b[m*])`

where we interpret `a[m*-k] = 0` if m*-k is out of [0, n-1].

The gradient sums to zero, which makes sense for a scale-invariant objective (adding a constant to z doesn't change a).

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len


def _score(a, fs):
    """Compute the score for a normalized sequence (sum=1)."""
    n = len(a)
    F = rfft(a, fs)
    b = irfft(F * F, fs)[:2 * n - 1]
    max_b = float(b.max())
    if not np.isfinite(max_b):
        return math.inf
    return 2.0 * n * max_b


def _grad_z(a, m_star, max_b, n):
    """Compute gradient of 2n*max(b) w.r.t. z in softmax parameterization."""
    c = np.zeros(n)
    lo = max(0, m_star - n + 1)
    hi = min(n - 1, m_star)
    idx = m_star - np.arange(lo, hi + 1)
    c[lo:hi + 1] = a[idx]
    return 4.0 * n * a * (c - max_b)


def _smooth_grad_z(a, b, n, fs, tau):
    """Smooth gradient using log-sum-exp approximation."""
    bt = b / tau
    bt = bt - bt.max()
    e = np.exp(np.clip(bt, -50, 50))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        p = np.zeros_like(b)
        p[int(np.argmax(b))] = 1.0
    else:
        p = e / s
    # Gradient: 2n * sum_k p[k] * 2*a[m*-k style]
    # = 4n * convolve(p_weighted, a) in a specific way
    # Actually: sum_k p[k] * 2 * a[k_j - k] for each j
    # This is 2 * (p flipped) convolved with a, evaluated at the right positions
    # More directly: grad_a[j] = 2 * sum_k p[k] * a[k-j] (if valid) + 2 * sum_k p[k] * a[j-k] (if valid)
    # = 2 * (p * a_cross_corr)
    # Let's compute: for each j, sum_k p[k]*a[k-j] (valid range)
    fs2 = next_fast_len(3 * n - 2)
    # convolve reversed p with a gives cross-correlation
    p_rev = p[::-1]
    cross = irfft(rfft(p_rev, fs2) * rfft(a, fs2), fs2)
    grad_a = 2.0 * 2.0 * n * (cross[n - 1:2 * n - 1])
    # Now transform to z-space
    grad_z = a * (grad_a - float(np.dot(a, grad_a)))
    return grad_z


def _adam_update(z, grad, m, v, t, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """One Adam step, returns updated z, m, v."""
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad * grad
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
    return z, m, v

            z_best = np.log(best / best.sum())
            z_best = z_best - z_best.mean()
            z_new = (1 - mixture) * z + mixture * z_best + sigma * 0.2 * rng.normal(0, 1, n)

        z_new = z_new - z_new.mean()
        z_new = np.clip(z_new, -40, 40)
        a_new = np.exp(z_new)
        a_new = a_new / a_new.sum()

        s_new = _score(a_new, fs)

        if np.isfinite(s_new) and s_new < best_score - 1e-12:
            best = a_new.copy()
            best_score = s_new
            a = a_new.copy()
            stuck = 0
            sigma = max(0.03, sigma * 0.97)
        else:
            stuck += 1
            sigma = min(0.6, sigma * 1.03)
            if stuck > 40:
                # Jump
                z = local_best_z if 'local_best_z' in dir() else np.log(a / a.sum())
                a = best + 0.05 * rng.normal(0, 1, n)
                a = np.clip(a, 1e-10, 1000.0)
                a = a / a.sum()
                sigma = 0.2
                stuck = 0

    return best, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    total = max(0.0, float(budget_s))
    hard_deadline = t0 + total - 4.0

    # Get parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            if p.size > 0 and np.isfinite(p).all():
                parent = np.clip(p, 0.0, 1000.0).astype(np.float64)
    except Exception:
        pass

    # Initialize best
    best_a = None
    best_score = math.inf

    def update_best(a, s):
        nonlocal best_a, best_score
        if a is None or s is None:
            return
        a = np.asarray(a, dtype=np.float64).ravel()
        if a.size == 0 or not np.isfinite(a).all():
            return
        a = np.clip(a, 0.0, 1000.0)
        if a.sum() < 0.01:
            return
        if s < best_score - 1e-14:
            best_a = a.copy()
            best_score = float(s)

    # Evaluate parent
    if parent is not None and parent.size > 0:
        fs_p = next_fast_len(2 * parent.size - 1)
        s_p = _score(parent / parent.sum(), fs_p)
        # Account for scale invariance: score is same regardless of scale
        update_best(parent, s_p)

    # If no good parent, use uniform
    if best_a is None or best_score > 1.9:
        n_def = parent.size if parent is not None else 192
        u = np.ones(n_def) / n_def
        fs_u = next_fast_len(2 * n_def - 1)
        s_u = _score(u, fs_u)
        update_best(u, s_u)
        if parent is None:
            parent = u.copy()

    n_best = best_a.size if best_a is not None else 192
    fs_best = next_fast_len(2 * n_best - 1)

    # Build diverse starting points
    starts = []
    starts.append(("parent", best_a.copy()))

    t_arr = np.arange(n_best) / n_best
    x_arr = 2 * t_arr - 1

    # Various shapes (non-symmetric to allow beating uniform)
    shapes = {}
    shapes['ramp'] = np.clip(1.0 - x_arr, 0.1, 2.0)  # decreasing
    shapes['ramp2'] = np.clip(1.0 + 0.5 * x_arr, 0.1, 2.0)
    shapes['sine'] = np.abs(np.sin(1.5 * np.pi * t_arr)) + 0.1
    shapes['two_peak'] = np.maximum(0, 1 - 4 * (x_arr - 0.4) ** 2) + np.maximum(0, 1 - 4 * (x_arr + 0.4) ** 2) + 0.1
    shapes['gaussian_off'] = np.exp(-6 * (x_arr - 0.3) ** 2) + 0.05
    shapes['gaussian_off2'] = np.exp(-6 * (x_arr + 0.3) ** 2) + 0.05
    shapes['cos_asym'] = 0.5 + 0.5 * np.cos(2 * np.pi * (t_arr - 0.1)) + 0.1
    shapes['linear_wiggle'] = 1.0 + 0.3 * np.sin(3 * np.pi * t_arr) + 0.2 * np.cos(5 * np.pi * t_arr) + 0.3
    shapes['exp_decay'] = np.exp(-2.0 * t_arr) + 0.1
    shapes['exp_decay_r'] = np.exp(-2.0 * (1 - t_arr)) + 0.1

    for name, shape in shapes.items():
        shape = np.clip(shape, 0.01, None).astype(np.float64)
        shape = shape / shape.sum()
        starts.append((name, shape))

    # Add random starts
    rng_master = np.random.default_rng(seed)
    for i in range(4):
        rshape = np.abs(rng_master.normal(1.0, 0.5, n_best)) + 0.1
        rshape = rshape / rshape.sum()
        starts.append((f"rand_{i}", rshape))

    # Phase 1: Multi-start optimization
    n_starts = len(starts)
    phase1_end = t0 + 0.65 * total
    per_start = (phase1_end - t0) / n_starts

    for i, (name, s) in enumerate(starts):
        if time.monotonic() >= phase1_end:
            break
        if time.monotonic() >= hard_deadline - 60:
            break
        end = min(t0 + (i + 1) * per_start, phase1_end)
        if end - time.monotonic() < 2.0:
            end = time.monotonic() + 2.0

        try:
            a_opt, s_opt = _optimize_from(s, end, seed=seed + i * 7919, use_smooth=True)
            update_best(a_opt, s_opt)
        except Exception:
            pass

        # Also try without smooth
        if (i + 1) * per_start * 0.5 < time.monotonic() and i % 2 == 0:
            end2 = min(end, time.monotonic() + per_start * 0.3)
            if end2 > time.monotonic() + 1.0:
                try:
                    a_opt2, s_opt2 = _optimize_from(s, end2, seed=seed + i * 7919 + 1, use_smooth=False)
                    update_best(a_opt2, s_opt2)
                except Exception:
                    pass

    # Phase 2: Refined optimization from best
    phase2_end = t0 + 0.85 * total
    if time.monotonic() < phase2_end - 5 and best_a is not None:
        try:
            a_ret, s_ret = _optimize_from(best_a, phase2_end, seed=seed + 55555, use_smooth=True)
            update_best(a_ret, s_ret)
        except Exception:
            pass

    # Phase 2b: Second round without smooth
    if time.monotonic() < phase2_end + 0.05 * total and best_a is not None:
        end2b = min(phase2_end + 0.05 * total, hard_deadline - 30)
        if end2b > time.monotonic() + 3:
            try:
                a_ret2, s_ret2 = _optimize_from(best_a, end2b, seed=seed + 66666, use_smooth=False)
                update_best(a_ret2, s_ret2)
            except Exception:
                pass

    # Phase 3: Local polish
    phase3_end = hard_deadline - 5
    if time.monotonic() < phase3_end - 3 and best_a is not None:
        try:
            a_pol, s_pol = _local_polish(best_a, phase3_end, seed=seed + 77777)
            update_best(a_pol, s_pol)
        except Exception:
            pass

    # Final: one more quick gradient pass
    if time.monotonic() < hard_deadline - 2 and best_a is not None:
        try:
            a_fin, s_fin = _optimize_from(best_a, hard_deadline - 1, seed=seed + 88888, use_smooth=False)
            update_best(a_fin, s_fin)
        except Exception:
            pass

    # Final validation
    if best_a is None:
        best_a = np.ones(32) / 32.0
        best_score = 2.0

    cand = [float(v) for v in best_a]

    # Validate with exact evaluator
    try:
        fs_final = float(evaluate_sequence(cand))
        if not np.isfinite(fs_final):
            cand = [1.0]
            fs_final = float(evaluate_sequence(cand))
            if not np.isfinite(fs_final):
                return [1.0]
    except Exception:
        return [1.0]

    return cand
