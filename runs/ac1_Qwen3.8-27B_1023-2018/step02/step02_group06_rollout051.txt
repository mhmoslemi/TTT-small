    return best_a, 2.0 * n * best_M


def _gen_init(n, rng, variant):
    i = np.arange(n, dtype=np.float64)
    v = int(variant) % 8
    if v == 0:
        f1 = int(rng.integers(1, n // 2 + 1))
        f2 = int(rng.integers(1, max(2, n // 2 + 1)))
        a = 1.0 + 0.5 * np.sin(2 * np.pi * f1 * i / n + rng.uniform(0, 6.28)) + \
            0.3 * np.cos(2 * np.pi * f2 * i / n + rng.uniform(0, 6.28))
    elif v == 1:
        a = np.exp(rng.normal(0, 0.6, n))
    elif v == 2:
        a = 0.5 + 0.5 * np.abs(np.sin(np.pi * i / n * rng.uniform(1, 5)))
    elif v == 3:
        a = np.abs(rng.normal(0, 1, n)) + 0.05
    elif v == 4:
        a = np.ones(n)
        for _ in range(3):
            c = rng.uniform(0, n)
            w = rng.uniform(n // 6, n // 3)
            a += rng.uniform(0.3, 0.9) * np.exp(-(i - c) ** 2 / (2 * w ** 2))
    elif v == 5:
        a = 1.0 + 0.6 * np.sin(2 * np.pi * rng.integers(2, max(3, n // 3)) * i / n)
        a += 0.25 * np.cos(2 * np.pi * rng.integers(4, max(5, n // 2)) * i / n)
    elif v == 6:
        a = rng.uniform(0.3, 1.5, n)
    else:
        a = 1.0 + 0.4 * np.sin(2 * np.pi * i / n) + 0.2 * np.sin(4 * np.pi * i / n)

    a = np.maximum(a, 0.01)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(0.0, float(budget_s))
    deadline = t0 + budget - 18.0

    parent = _get_parent()
    n = len(parent)

    try:
        best_score = float(evaluate_sequence([float(x) for x in np.clip(parent, 0, 1000)]))
    except Exception:
        best_score = math.inf

    best_a = np.clip(parent, 0.0, 1000.0).copy()
    if not np.isfinite(best_score) or best_score >= 100:
        best_a = np.ones(n) / n
        best_score = 2.0

    rng = np.random.default_rng(seed)

    def _try_update(cand_a, threshold=1e-10):
        nonlocal best_a, best_score
        sc = _fast_score(cand_a)
        if sc < best_score - threshold:
            try:
                exact = float(evaluate_sequence([float(x) for x in np.clip(cand_a, 0, 1000)]))
            except Exception:
                exact = math.inf
            if np.isfinite(exact) and exact < best_score:
                best_a = np.clip(cand_a, 0, 1000).copy()
                best_score = exact

    # Phase 1: LP optimization from parent (48% of budget)
    p1_end = min(t0 + budget * 0.48, deadline - 60)
    if time.monotonic() < p1_end:
        try:
            a0 = _norm1(best_a.copy())
            res_a, res_sc = _optimize_lp(a0, p1_end, rng, lp_every=2)
            _try_update(res_a)
        except Exception:
            pass

    # Phase 2: Local refinement (22% of budget)
    p2_end = min(t0 + budget * 0.70, deadline - 40)
    if time.monotonic() < p2_end:
        try:
            res_a, res_sc = _local_refine(best_a, p2_end, np.random.default_rng(seed + 1111),
                                           max_iter=80000)
            _try_update(res_a)
        except Exception:
            pass

    # Phase 3: Another LP pass on best (15% of budget)
    p3_end = min(t0 + budget * 0.85, deadline - 25)
    if time.monotonic() < p3_end:
        try:
            a0 = _norm1(best_a.copy())
            res_a, res_sc = _optimize_lp(a0, p3_end, rng, lp_every=3)
            _try_update(res_a)
        except Exception:
            pass

    # Phase 4: Random restarts with short LP + local (if time)
    if time.monotonic() < deadline - 25:
        for rid in range(8):
            if time.monotonic() > deadline - 15:
                break
            a_init = _gen_init(n, rng, variant=rid + 10)
            sub_dl = min(deadline - 10, time.monotonic() + min(25.0,
                          max(8.0, (deadline - 15 - time.monotonic()) / max(1, 8 - rid))))
            try:
                res_a, res_sc = _optimize_lp(a_init, sub_dl, rng, lp_every=3)
                if time.monotonic() < deadline - 8:
                    res_a2, res_sc2 = _local_refine(res_a, deadline - 6,
                                                     np.random.default_rng(seed + 2000 + rid),
                                                     max_iter=20000)
                    if res_sc2 < res_sc:
                        res_a = res_a2
                _try_update(res_a)
            except Exception:
                pass

    # Phase 5: Final local polish
    if time.monotonic() < deadline - 5:
        try:
            res_a, res_sc = _local_refine(best_a, deadline - 3,
                                           np.random.default_rng(seed + 9999),
                                           max_iter=30000)
            _try_update(res_a)
        except Exception:
            pass

    # Final validation
    final = [float(x) for x in np.clip(best_a, 0.0, 1000.0)]
    try:
        fs = float(evaluate_sequence(final))
        if not np.isfinite(fs) or fs > 10000:
            final = [1.0]
            fs2 = float(evaluate_sequence(final))
            if not np.isfinite(fs2):
                final = [1.0, 1.0, 1.0, 1.0]
    except Exception:
        final = [1.0]

    # Ensure non-empty, finite
    if not final or any(not np.isfinite(x) for x in final):
        final = [1.0]

    return final
