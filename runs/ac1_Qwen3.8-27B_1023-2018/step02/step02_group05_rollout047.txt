                if not improved and no_improve_count > 60:
                    # Bigger restart: mix with random
                    ra = np.abs(rng.standard_normal(n)) + 0.1
                    ra = ra / ra.sum()
                    for mix in (0.3, 0.5, 0.7):
                        ca = (1.0 - mix) * a + mix * ra
                        ca = np.clip(ca, 0.0, None)
                        cs = float(ca.sum())
                        if cs > 1e-15:
                            ca = ca / cs
                            csc = _score(ca)
                            if math.isfinite(csc) and csc < local:
                                a = ca
                                local = csc
                                best_local = min(best_local, local)
                                _update_best(a, best)
                                stagnant = 0
                                no_improve_count = 0
                                break
                if no_improve_count > 150:
                    break


def _subgradient_refine(a_init, n_target, time_limit, rng, best):
    """Subgradient descent on the max-convolution objective with smoothing."""
    n = int(n_target)
    if n < 4 or time_limit < 15:
        return
    t0 = time.monotonic()
    deadline = t0 + time_limit - 3.0

    a = np.asarray(a_init, dtype=np.float64).ravel().copy()
    if a.size != n:
        if a.size > 1:
            xo = np.linspace(0.0, 1.0, a.size)
            xn = np.linspace(0.0, 1.0, n)
            a = np.interp(xn, xo, a)
        else:
            a = np.ones(n)
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if s < 1e-15:
        a = np.ones(n)
    else:
        a = a / s

    local = _score(a)
    _update_best(a, best)
    best_local = local

    alpha = 0.05
    stale = 0
    use_weights = False
    K = 5  # number of top active constraints

    for it in range(5000):
        if time.monotonic() >= deadline:
            break
        rem = deadline - time.monotonic()
        if rem < 3.0:
            break
        if local < TARGET and it > 5:
            break

        b = np.convolve(a, a)
        M = float(b.max())
        if not math.isfinite(M) or M <= 1e-30:
            break

        # Select active set: top K lags
        sorted_lags = np.argsort(b)[::-1]
        active = sorted_lags[:K]
        tol = 0.05 * M
        active = sorted_lags[b[sorted_lags] >= M - tol]
        active = active[:min(len(active), 20)]

        # Compute gradient of max b over active set (uniform weights = average)
        grad = np.zeros(n)
        n_active = len(active)
        if n_active == 0:
            break
        for k in active:
            k = int(k)
            # db[k]/da[j] = a[k-j] if 0 <= k-j < n
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                grad[j] += a[k - j]
        grad /= n_active

        # Projected subgradient step
        norm_g = np.linalg.norm(grad)
        if norm_g < 1e-15:
            break

        improved = False
        for a_try in (alpha, alpha * 0.5, alpha * 2.0, alpha * 0.25):
            cand = a - a_try * grad / norm_g
            cand = np.clip(cand, 0.0, None)
            cs = float(cand.sum())
            if cs < 1e-15:
                continue
            cand = cand / cs
            sc = _score(cand)
            if math.isfinite(sc) and sc < local:
                a = cand
                local = sc
                stale = 0
                alpha = min(alpha * 1.05, 0.3)
                if local < best_local:
                    best_local = local
                _update_best(a, best)
                improved = True
                break

        if not improved:
            stale += 1
            alpha *= 0.85
            if alpha < 1e-6:
                alpha = 0.03
                vp = a * np.exp(0.05 * rng.standard_normal(n))
                vp = np.clip(vp, 0.0, None)
                vs = float(vp.sum())
                if vs > 1e-15:
                    vp = vp / vs
                    vsc = _score(vp)
                    if math.isfinite(vsc) and vsc < local:
                        a = vp
                        local = vsc
                        _update_best(a, best)
            if stale > 80:
                break

        # Sometimes LP-assist
        if it % 50 == 40 and rem > 20:
            g = _lp_solve(a, M, n, min(8.0, rem - 5.0))
            if g is not None:
                for t in (0.1, 0.2, 0.3):
                    cand = (1.0 - t) * a + t * g
                    cand = np.clip(cand, 0.0, None)
                    cs = float(cand.sum())
                    if cs > 1e-15:
                        cand = cand / cs
                        sc = _score(cand)
                        if math.isfinite(sc) and sc < local:
                            a = cand
                            local = sc
                            _update_best(a, best)
                            stale = 0
                            break


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    try:
        remaining = max(10.0, float(budget_s))
    except Exception:
        remaining = 1300.0
    deadline = t_start + remaining - 20.0

    best = [float('inf'), np.array([1.0] * 16, dtype=np.float64)]
    _update_best(best[1], best)

    # Get parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            parent = np.asarray(h, dtype=np.float64).ravel()
            parent = np.nan_to_num(parent, nan=0.0, posinf=0.0, neginf=0.0)
            parent = np.clip(parent, 0.0, 1000.0)
            if parent.size > 1:
                _update_best(parent, best)
    except Exception:
        parent = None

    def _rem():
        return max(0.0, deadline - time.monotonic())

    # Phase 1: Main LP iteration at parent length (320)
    if parent is not None and parent.size >= 4:
        np_ = parent.size
        alloc = min(550.0, _rem() * 0.5)
        if alloc > 30:
            _lp_iterate(parent, np_, alloc, rng, best)

    # Phase 2: LP at n=256
    if _rem() > 120:
        n256 = 256
        a256 = np.ones(256)
        if parent is not None and parent.size > 1:
            tp = np.linspace(0.0, 1.0, parent.size)
            a256 = np.interp(np.linspace(0.0, 1.0, 256), tp, parent)
        alloc = min(150.0, _rem() * 0.25)
        if alloc > 20:
            _lp_iterate(a256, 256, alloc, rng, best)

    # Phase 3: LP at n=384
    if _rem() > 120:
        n384 = 384
        a384 = np.ones(384)
        if parent is not None and parent.size > 1:
            tp = np.linspace(0.0, 1.0, parent.size)
            a384 = np.interp(np.linspace(0.0, 1.0, 384), tp, parent)
        alloc = min(150.0, _rem() * 0.25)
        if alloc > 20:
            _lp_iterate(a384, 384, alloc, rng, best)

    # Phase 4: Subgradient refinement on best
    if _rem() > 60:
        best_arr = np.asarray(best[1], dtype=np.float64).ravel()
        nb = best_arr.size
        alloc = min(180.0, _rem() * 0.4)
        if alloc > 20:
            _subgradient_refine(best_arr, nb, alloc, rng, best)

    # Phase 5: Final LP polish on best
    if _rem() > 30:
        best_arr = np.asarray(best[1], dtype=np.float64).ravel()
        nb = best_arr.size
        alloc = min(80.0, _rem() * 0.5)
        if alloc > 10:
            _lp_iterate(best_arr, nb, alloc, rng, best)

    # Phase 6: If still time, try parent length again with fresh perturbations
    if _rem() > 50 and parent is not None:
        np_ = parent.size
        perturbed = parent * np.exp(0.05 * rng.standard_normal(np_))
        perturbed = np.clip(perturbed, 0.0, None)
        ps = float(perturbed.sum())
        if ps > 1e-15:
            perturbed = perturbed / ps
        alloc = min(60.0, _rem() * 0.5)
        if alloc > 15:
            _lp_iterate(perturbed, np_, alloc, rng, best)

    # Final validation
    final = np.asarray(best[1], dtype=np.float64).ravel()
    final = np.nan_to_num(final, nan=0.0, posinf=0.0, neginf=0.0)
    final = np.clip(final, 0.0, 1000.0)
    if final.size == 0:
        final = np.ones(16, dtype=np.float64)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(16, dtype=np.float64)

    fin_sc = evaluate_sequence([float(x) for x in final])
    if not math.isfinite(fin_sc):
        final = np.ones(16, dtype=np.float64)
        fin_sc = evaluate_sequence([float(x) for x in final])

    # Ensure we return something valid
    if not math.isfinite(fin_sc) or fin_sc > best[0] + 0.01:
        # Try parent directly
        if parent is not None and parent.size > 1:
            p = np.clip(parent, 0.0, 1000.0)
            psc = evaluate_sequence([float(x) for x in p])
            if math.isfinite(psc):
                final = p
                fin_sc = psc
        if not math.isfinite(fin_sc):
            final = np.ones(16, dtype=np.float64)

    return [float(x) for x in final]
