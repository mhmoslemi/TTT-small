def _precompute_pattern(n):
    total_nnz = n * n  # each of n^2 (k,j) pairs
    # Actually, it's not exactly n^2. Let me compute:
    # For k=0: 1 entry (j=0)
    # For k=1: 2 entries (j=0,1)
    # ...
    # For k=n-1: n entries
    # For k=n: n-1 entries
    # ...
    # For k=2n-2: 1 entry
    # Total: n^2
    # Yes, total is n^2!
    
    rows = np.empty(n*n, dtype=np.int32)
    cols = np.empty(n*n, dtype=np.int32)
    
    idx = 0
    for k in range(2*n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        cnt = j_max - j_min + 1
        rows[idx:idx+cnt] = k
        cols[idx:idx+cnt] = np.arange(j_min, j_max+1)
        idx += cnt
    
    return rows, cols

def _make_A(rows, cols, a):
    vals = a[rows - cols]
    return sps.csr_matrix((vals, (rows, cols)), shape=(2*len(a)-1, len(a)))
