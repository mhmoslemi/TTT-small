            _lp_iterate(best[1], t0 + alloc, best, rng, max_it=300000)

    # Phase 4: Diverse starts at parent length
    if time.monotonic() < deadline - 40:
        starts = _gen_starts(n_par, parent, rng)
        for s in starts[1:8]:
            if time.monotonic() >= deadline - 40:
                break
            alloc = min(60.0, deadline - 40 - t0)
            if alloc < 15:
                break
            _lp_iterate(s, t0 + alloc, best, rng, max_it=100000)
            _subgrad(best[1], t0 + min(10.0, deadline - 40 - t0), best, rng, max_it=20000)

    # Phase 5: Try other lengths
    for nb in (128, 256, 64):
        if time.monotonic() >= deadline - 25:
            break
        alloc_total = min(40.0, deadline - 25 - t0)
        if alloc_total < 10:
            break
        starts_nb = _gen_starts(nb, best[1], rng)
        for s in starts_nb[:3]:
            if time.monotonic() >= deadline - 25:
                break
            a2 = min(15.0, deadline - 25 - t0)
            if a2 < 5:
                break
            _lp_iterate(s, t0 + a2, best, rng, max_it=50000)

    # Phase 6: Final LP + subgradient polish
    if time.monotonic() < deadline - 10:
        alloc = min(20.0, deadline - 10 - t0)
        if alloc > 5:
            _lp_iterate(best[1], t0 + alloc, best, rng, max_it=100000)
    if time.monotonic() < deadline - 5:
        alloc = min(8.0, deadline - 5 - t0)
        if alloc > 2:
            _subgrad(best[1], t0 + alloc, best, rng, max_it=30000)

    # Final validation
    fin = np.asarray(best[1], dtype=np.float64).ravel().copy()
    fin = np.nan_to_num(fin, nan=0.0, posinf=0.0, neginf=0.0)
    fin = np.clip(fin, 0.0, 1000.0)
    if fin.size < 1 or float(fin.sum()) < 0.01:
        fin = np.array([1.0, 1.0], dtype=np.float64)
    _upd(fin, best)
    fin = np.asarray(best[1], dtype=np.float64).ravel()
    v = evaluate_sequence([float(x) for x in fin])
    if not math.isfinite(v):
        fin = np.array([1.0, 1.0], dtype=np.float64)
    return [float(x) for x in fin]
