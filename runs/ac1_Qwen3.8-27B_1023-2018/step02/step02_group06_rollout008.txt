    a=np.where(np.isfinite(a), a, 0.0)
    a=np.maximum(a,0.0)
    s=a.sum()
    if s<=1e-300:
        return np.ones(a.size)/a.size
    return a/s

def _conv_self(a):
    # direct convolution, ensure nonneg
    c = np.convolve(a, a, mode='full')
    np.maximum(c, 0.0, out=c)
    c[~np.isfinite(c)] = 0.0
    return c

def _exact_M(a):
    return float(_conv_self(a).max())

def _exact_score(a):
    n = len(a)
    return 2.0*n*_exact_M(a)

def _conv_self(a):

        F=np.fft.rfft(a,N)
        c=np.fft.irfft(F*F,N)[:L]

def _a_from_z(z, p, var_idx, n):
    if n <= 0:
        return np.empty(0, dtype=np.float64)
    z = np.asarray(z, dtype=np.float64).ravel()
    if n == 1:
        return np.ones(1, dtype=np.float64)
    if z.size != n - 1 or not np.all(np.isfinite(z)):
        return np.ones(n, dtype=np.float64) / float(n)
    mz = float(np.max(z))
    if mz < 0.0:
        mz = 0.0
    if mz > 700.0:
        mz = 700.0
    ez = np.exp(z - mz)
    ep = math.exp(-mz)
    a = np.empty(n, dtype=np.float64)
    a[var_idx] = ez
    a[p] = ep
    Z = float(ez.sum()) + float(ep)
    if Z <= 0.0 or not math.isfinite(Z):
        return np.ones(n, dtype=np.float64) / float(n)
    a /= Z
    if not np.all(np.isfinite(a)):
        return np.ones(n, dtype=np.float64) / float(n)
    return a


def _a_to_z(a, p, var_idx):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n <= 1 or var_idx.size == 0:
        return np.empty(0, dtype=np.float64)
    ap = float(a[p])
    if ap <= 0.0 or not math.isfinite(ap):
        ap = 1e-300
    r = np.maximum(a[var_idx], 1e-300) / ap
    z = np.log(r)
    z = np.where(np.isfinite(z), z, 0.0)
    return np.clip(z, -80.0, 80.0).astype(np.float64)


def _smooth_score_grad(z, p, var_idx, n, s):
    z = np.asarray(z, dtype=np.float64).ravel()
    grad_zero = np.zeros(max(0, n - 1), dtype=np.float64)
    if z.size != max(0, n - 1) or not np.all(np.isfinite(z)):
        return 1e100, grad_zero
    a = _a_from_z(z, p, var_idx, n)
    c = _conv_self(a)
    if not np.all(np.isfinite(c)):
        return 1e100, grad_zero
    q = float(n) * c
    m = float(q.max())
    if not math.isfinite(m):
        return 1e100, grad_zero
    scale = max(float(s), 1e-12)
    e = np.exp((q - m) / scale)
    mean_e = float(e.mean())
    if mean_e <= 0.0 or not math.isfinite(mean_e):
        mean_e = 1e-300
    val = 2.0 * (m + scale * math.log(mean_e))
    psum = float(e.sum())
    if psum <= 0.0 or not math.isfinite(psum):
        w = np.ones(q.size, dtype=np.float64) / float(q.size)
    else:
        w = e / psum
    v = _weights_dot(w, a)
    if not np.all(np.isfinite(v)):
        return val, grad_zero
    ga = 4.0 * float(n) * v
    total = float(np.dot(a, ga))
    grad = np.zeros(n, dtype=np.float64)
    if var_idx.size:
        grad[var_idx] = a[var_idx] * (ga[var_idx] - total)
    reg = 1e-10
    val += reg * float(np.dot(z, z))
    if var_idx.size:
        grad[var_idx] += 2.0 * reg * z
    grad_out = grad[var_idx]
    if not np.all(np.isfinite(grad_out)):
        grad_out = np.zeros(grad_out.size, dtype=np.float64)
    if not math.isfinite(val):
        val = 1e100
    return float(val), grad_out.astype(np.float64, copy=False)


def _lbfgs_refine(a0, s_schedule, maxfun_per, deadline, best, eval_every=20):
    a = _normalize(a0)
    n = len(a)
    if n < 2:
        sc = _exact_score(a)
        if math.isfinite(sc) and sc < best[0]:
            best[0] = sc
            best[1] = a.copy()
        return [sc, a]

    a = np.maximum(a, 1e-300)
    a = _normalize(a)
    local = [math.inf, a.copy()]

    def _upd(a_t):
        try:
            a_t = _normalize(a_t)
            if a_t is None or a_t.size != n:
                return
            sc = _exact_score(a_t)
            if math.isfinite(sc):
                if sc < local[0] - 1e-15:
                    local[0] = sc
                    local[1] = a_t.copy()
                if sc < best[0] - 1e-15:
                    best[0] = sc
                    best[1] = a_t.copy()
        except Exception:
            pass

    _upd(a)
    bounds = [(-80.0, 80.0)] * (n - 1)
    current = a.copy()

    for s in s_schedule:
        if _now() >= deadline:
            break
        if not np.all(np.isfinite(current)):
            current = local[1]
        p = int(np.argmax(current))
        if p < 0:
            p = 0
        var_idx = np.concatenate((np.arange(0, p), np.arange(p + 1, n))).astype(int)
        z = _a_to_z(current, p, var_idx)

        state = {"cnt": 0, "last": z.copy()}

        def fun(xk, p=p, var_idx=var_idx, s=s):
            if _now() > deadline:
                return 1e100, np.zeros(n - 1, dtype=np.float64)
            val, grad = _smooth_score_grad(xk, p, var_idx, n, s)
            state["cnt"] += 1
            state["last"] = np.asarray(xk, dtype=np.float64).copy()
            return val, grad

        def cb(xk, p=p, var_idx=var_idx):
            if _now() > deadline:
                return True
            if state["cnt"] > 0 and state["cnt"] % max(1, int(eval_every)) == 0:
                try:
                    ac = _a_from_z(np.asarray(xk, dtype=np.float64), p, var_idx, n)
                    _upd(ac)
                except Exception:
                    pass
            return False

        options = {
            "maxiter": int(max(10, maxfun_per)),
            "maxfun": int(max(30, 3 * maxfun_per)),
            "ftol": 1e-12,
            "gtol": 1e-8,
            "maxcor": 10,
            "maxls": 50,
        }
        try:
            res = minimize(
                fun,
                z,
                method="L-BFGS-B",
                jac=True,
                options=options,
                bounds=bounds,
                callback=cb,
            )
            z_new = np.asarray(res.x, dtype=np.float64)
        except Exception:
            z_new = state["last"]

        if z_new is None or not np.all(np.isfinite(np.asarray(z_new, dtype=np.float64))):
            z_new = state["last"]
        if z_new is None or not np.all(np.isfinite(np.asarray(z_new, dtype=np.float64))):
            z_new = _a_to_z(current, p, var_idx)

        current = _a_from_z(np.asarray(z_new, dtype=np.float64), p, var_idx, n)
        current = _normalize(current)
        _upd(current)

    _upd(current)
    _upd(local[1])
    return local


def _generate_initials(base, rng, max_count=45):
    n = len(base)
    out = []

    def add(a):
        try:
            y = _normalize(np.asarray(a, dtype=np.float64).ravel())
        except Exception:
            return
        if y.size == 0 or not np.all(np.isfinite(y)):
            return
        y = np.maximum(y, 1e-12)
        y /= y.sum()
        out.append(y)

    add(base)
    add(base[::-1])
    if n > 1:
        for s in range(0, n, max(1, n // 12)):
            add(np.roll(base, s))

    for sigma in (0.05, 0.1, 0.2, 0.35, 0.6, 1.0):
        add(base * np.exp(sigma * rng.standard_normal(n)))

    for mix in (0.2, 0.4, 0.7):
        add((1.0 - mix) * base + (mix / float(n)))

    u = np.linspace(0.0, 1.0, n)
    i = np.arange(n, dtype=np.float64)
    left = (i + 0.5) / float(n)
    right = (n - i - 0.5) / float(n)

    for alpha, beta in (
        (0.3, 0.7),
        (0.7, 0.3),
        (0.2, 1.0),
        (1.5, 0.5),
        (0.5, 0.5),
        (0.1, 2.0),
    ):
        add(np.maximum((left**alpha) * (right**beta), 1e-12))

    for _ in range(6):
        amp = float(rng.uniform(0.4, 1.5))
        ph = float(rng.uniform(0.0, 2.0 * math.pi))
        add(np.exp(amp * np.sin(2.0 * math.pi * u + ph)))
        ph2 = float(rng.uniform(0.0, 2.0 * math.pi))
        add(np.exp(0.7 * np.cos(2.0 * math.pi * u + ph) + 0.3 * np.sin(4.0 * math.pi * u + ph2)))

    for _ in range(8):
        terms = []
        for m in range(1, 6):
            coeff = float(rng.normal(0.0, 1.0)) * (1.0 / (0.5 + m))
            terms.append(coeff * np.sin(2.0 * math.pi * m * u + float(rng.uniform(0.0, 2.0 * math.pi))))
        add(np.exp(0.6 * sum(terms)))

    for _ in range(8):
        c1 = float(rng.uniform(0.0, 1.0))
        c2 = float(rng.uniform(0.0, 1.0))
        w1 = float(rng.uniform(0.05, 0.3))
        w2 = float(rng.uniform(0.05, 0.3))
        am2 = float(rng.uniform(0.2, 1.0))
        add(np.exp(-((u - c1) / w1) ** 2) + am2 * np.exp(-((u - c2) / w2) ** 2))

    add(np.exp(-((u - float(rng.uniform(0.2, 0.8))) / float(rng.uniform(0.08, 0.25))) ** 2))

    if len(out) > max_count:
        idx = np.linspace(0, len(out) - 1, max_count, dtype=int)
        out = [out[int(j)] for j in idx]
    return out


def _simple_transforms(base, consider):
    base = _normalize(base)
    n = base.size
    if n == 0:
        return

    bases = [base]
    for eps in (1e-4, 1e-5):
        try:
            y = base.copy()
            y[y < eps] = 0.0
            consider(y)
            bases.append(y)
        except Exception:
            pass

    for arr in bases:
        arr = _normalize(arr)
        for s in range(n):
            ar = np.roll(arr, s) if s else arr
            consider(ar)
            consider(ar[::-1])


def _polish_exact(a0, deadline, rng, max_iters=1500):
    a = _normalize(a0)
    n = len(a)
    if n == 0:
        return np.empty(0, dtype=np.float64), math.inf

    best = a.copy()
    best_M = _exact_M(a)
    t = 0.01
    no = 0
    fracs = (1e-4, 3e-4, 1e-3, 3e-3, 1e-2, 3e-2, 0.1)

    for it in range(max_iters):
        if _now() >= deadline:
            break
        c = _conv_self(a)
        M = float(c.max())
        if M < best_M - 1e-13:
            best_M = M
            best = a.copy()

        q = float(n) * c
        m = float(q.max())
        if not math.isfinite(m):
            break
        e = np.exp((q - m) / 0.005)
        es = float(e.sum())
        if es > 0.0 and np.isfinite(es):
            w = e / es
        else:
            w = np.full(q.size, 1.0 / q.size, dtype=np.float64)

        tol = 1e-4 + 1e-5 * m
        top = q >= m - tol
        if top.any():
            wt = top.astype(np.float64)
            ts = float(wt.sum())
            if ts > 0:
                w = 0.5 * w + 0.5 * (wt / ts)

        v = _weights_dot(w, a)
        if not np.all(np.isfinite(v)):
            break
        grad = 4.0 * float(n) * v

        best_y = None
        best_step = 0.0
        best_M_step = M
        for frac in fracs:
            if _now() >= deadline:
                break
            step = t * frac
            if step <= 0:
                continue
            y = _proj_simplex(a - step * grad, 1.0)
            Mn = _exact_M(y)
            if Mn < best_M_step:
                best_M_step = Mn
                best_y = y.copy()
                best_step = step

        if best_y is not None and best_M_step < M - 1e-12:
            a = best_y
            t = best_step * 1.2 if best_step > 0.0 else t
            if t > 1.0:
                t = 1.0
            no = 0
        else:
            no += 1
            t = max(1e-5, t * 0.85)
            if no >= 4:
                order = np.argsort(v)
                low = order[:5]
                high = order[-5:]
                done = False
                for k in range(min(6, max(1, 5 * 5))):
                    i = int(high[k % max(1, len(high))])
                    j = int(low[k % max(1, len(low))])
                    if i == j:
                        continue
                    for frac in (0.005, 0.02, 0.05, 0.1):
                        delta = frac * float(a[i])
                        if delta <= 0.0 or delta >= float(a[i]):
                            break
                        y = a.copy()
                        y[i] -= delta
                        y[j] += delta
                        if y.min() < 0.0:
                            continue
                        if _now() >= deadline:
                            break
                        Mn = _exact_M(y)
                        if Mn < M - 1e-12:
                            a = y.copy()
                            a = _normalize(a)
                            done = True
                            no = 0
                            break
                    if done:
                        break
                if not done:
                    if no > 40 and rng is not None and it % 5 == 0:
                        a = _proj_simplex(np.maximum(a + rng.normal(0.0, 1e-4, n), 0.0), 1.0)
                    if no > 300:
                        break

    best = _normalize(best)
    sc = _exact_score(best)
    return best, sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = _now()
    total = max(1.0, float(budget_s))
    head = max(5.0, min(20.0, 0.02 * total))
    hard_deadline = t0 + total
    deadline = max(t0 + 1.0, hard_deadline - head)

    parent = _get_parent()
    rng = np.random.default_rng(seed)

    best = [math.inf, parent.copy()]

    def consider(a):
        try:
            y = _normalize(a)
            if y is None or y.size == 0:
                return
            sc = _exact_score(y)
            if math.isfinite(sc) and sc < best[0]:
                best[0] = sc
                best[1] = y.copy()
        except Exception:
            pass

    consider(parent)

    deep_sched = (2.0, 1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.012, 0.004)
    quick_sched = (1.5, 0.5, 0.2)
    medium_sched = (1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.012, 0.004)

    candidates = []

    p_end = min(deadline - 10.0, t0 + min(max(30.0, 0.30 * total), 300.0))
    if _now() < p_end and best[1] is not None:
        try:
            loc = _lbfgs_refine(best[1], deep_sched, 80, p_end, best, eval_every=20)
            if math.isfinite(loc[0]):
                candidates.append(loc)
        except Exception:
            pass

    q_end = min(deadline - 90.0, t0 + 0.72 * total)
    if _now() < q_end:
        try:
            starts = _generate_initials(best[1] if best[1] is not None else parent, rng, max_count=45)
        except Exception:
            starts = []

        scored = []
        for a in starts:
            try:
                sc = _exact_score(a)
                if math.isfinite(sc):
                    scored.append((sc, a))
            except Exception:
                pass
        scored.sort(key=lambda x: x[0])

        top_limit = min(len(scored), 30)
        for idx, (sc, a) in enumerate(scored[:top_limit]):
            if _now() >= q_end:
                break
            if best[0] < math.inf and idx > 2 and sc > best[0] + 0.18:
                break
            sub_deadline = min(q_end, _now() + 10.0)
            if sub_deadline <= _now():
                break
            try:
                loc = _lbfgs_refine(a, quick_sched, 35, sub_deadline, best, eval_every=50)
                if math.isfinite(loc[0]):
                    candidates.append(loc)
            except Exception:
                pass

    d_end = min(deadline - 70.0, t0 + 0.88 * total)
    if _now() < d_end and candidates:
        candidates.sort(key=lambda x: x[0])
        done = 0
        for sc, a in candidates:
            if _now() >= d_end or done >= 3:
                break
            if not math.isfinite(sc):
                continue
            if best[0] < math.inf and sc > best[0] + 0.12:
                continue
            sub_deadline = min(d_end, _now() + 150.0)
            if sub_deadline <= _now():
                break
            try:
                loc = _lbfgs_refine(a, medium_sched, 70, sub_deadline, best, eval_every=30)
                if math.isfinite(loc[0]):
                    candidates.append(loc)
            except Exception:
                pass
            done += 1

    if _now() < deadline - 3 and best[1] is not None:
        _simple_transforms(best[1], consider)

    polish_end = deadline - 5.0
    if _now() < polish_end and best[1] is not None:
        try:
            pol, pol_sc = _polish_exact(
                best[1],
                polish_end,
                np.random.default_rng(seed + 999),
                max_iters=1500,
            )
            consider(pol)
        except Exception:
            pass

    if _now() < deadline - 3 and best[1] is not None:
        _simple_transforms(best[1], consider)

    if _now() < deadline - 10 and best[0] > 1.50:
        micro_sched = (0.25, 0.1, 0.04, 0.015)
        try:
            loc = _lbfgs_refine(
                best[1] if best[1] is not None else parent,
                micro_sched,
                30,
                deadline - 6.0,
                best,
                eval_every=25,
            )
            if math.isfinite(loc[0]):
                candidates.append(loc)
        except Exception:
            pass

    if _now() < deadline - 2 and best[1] is not None:
        _simple_transforms(best[1], consider)

    a_final = best[1] if best[1] is not None else parent
    try:
        a_final = _normalize(a_final)
    except Exception:
        a_final = np.ones(192, dtype=np.float64) / 192.0

    try:
        final_list = [float(x) for x in a_final]
    except Exception:
        final_list = [1.0]

    eval_fn = globals().get("evaluate_sequence")
    if eval_fn is not None:
        try:
            es = float(eval_fn(final_list))
            if math.isfinite(es):
                best[0] = es
        except Exception:
            pass

    if not math.isfinite(best[0]) or len(final_list) == 0:
        fb = np.ones(192, dtype=np.float64) / 192.0
        final_list = [float(x) for x in fb]

    return final_list
