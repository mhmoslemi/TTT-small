import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len

def _fft_conv(a, b, fs=None):
    """Linear convolution via FFT."""
    n = len(a) + len(b) - 1
    if fs is None:
        fs = next_fast_len(n)
    return irfft(rfft(a, fs) * rfft(b, fs), fs)[:n]

def _autocorr(a, fs=None):
    """Linear autocorrelation (a * a)."""
    n = len(a)
    total = 2*n - 1
    if fs is None:
        fs = next_fast_len(total)
    F = rfft(a, fs)
    return irfft(F * F, fs)[:total]

def _cross_corr(a, q, fs=None):
    """Compute c[j] = sum_i a[i] * q[j+i] for j=0,...,len(a)-1."""
    n = len(a)
    # c[j] = (a * q_flip)[2n-2-j] for j=0,...,n-1
    q_flip = q[::-1]
    total = n + len(q) - 1  # = n + (2n-1) - 1 = 3n-2
    if fs is None:
        fs = next_fast_len(total)
    conv = irfft(rfft(a, fs) * rfft(q_flip, fs), fs)[:total]
    return conv[n-1:2*n-1][::-1]

def _score(a):
    """Compute the score: 2*n*max(b)/(sum(a))^2."""
    n = len(a)
    b = _autocorr(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return math.inf
    return 2.0 * n * float(b.max()) / (s * s)

def _gradient(a, p):
    """Compute gradient of sum(p[k]*b[k]) with respect to a.
    
    grad_a[j] = 2*sum_i a[i]*p[j+i] - a[j]*p[2j]
    """
    n = len(a)
    c = _cross_corr(a, p)
    return 2.0 * c - a * p[0:2*n-1:2]

def _z_gradient(z, p, w=0.0, b=None):
    """Compute gradient in z-space for the softmax parameterization."""
    a = _softmax(z)
    grad_a = _gradient(a, p)
    if w > 0.0 and b is not None:
        grad_b = _gradient(a, b)
        grad_a = grad_a + w * grad_b
    # Project to tangent space of simplex
    mean_grad = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - mean_grad)
    return a, grad_z

def _softmax(z):
    z = z - z.max()
    e = np.exp(np.clip(z, -50, 50))
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

def _softmax(z):
    z = z - z.max()
    e = np.exp(np.clip(z, -50.0, 50.0))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z, dtype=np.float64) / z.size
    return e / s

    F = rfft(a, fs)
    return irfft(F * F, fs)[: 2 * len(a) - 1]


def _cross_corr(a, q, fs):
    n = len(a)
    qf = q[::-1]
    c = irfft(rfft(a, fs) * rfft(qf, fs), fs)[: 3 * n - 2]
    return c[n - 1 : 2 * n - 1][::-1]


def _score_norm(a, fs):
    b = _autocorr(a, fs)
    return 2.0 * len(a) * float(b.max())
