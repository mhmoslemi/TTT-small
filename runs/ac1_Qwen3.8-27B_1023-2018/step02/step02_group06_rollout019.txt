def _fft_conv_self(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2*n-1]
    np.maximum(c, 0.0, out=c)
    c[~np.isfinite(c)] = 0.0
    return c

def _score_fast(a):
    n = len(a)
    c = _fft_conv_self(a)
    mb = float(c.max())
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    return 2.0 * n * mb / (s * s)

def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def _lp_refine(a, deadline, max_iters=200):
    """LP-based subgradient refinement."""
    n = len(a)
    a = _normalize(a.copy())
    best_a = a.copy()
    best_score = _score_fast(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = _fft_conv_self(a)
        M = float(c.max())
        k_star = int(np.argmax(c))
        
        # Build constraint matrix A_conv (2n-1) x n
        # A_conv[k, j] = a[k-j] if 0 <= k-j < n
        g = cp.Variable(n, nonneg=True)
        
        # Build A matrix efficiently
        k_idx = np.arange(2*n-1)[:, None]
        j_idx = np.arange(n)[None, :]
        diff = k_idx - j_idx  # shape (2n-1, n)
        valid = (diff >= 0) & (diff < n)  # (2n-1, n) boolean
        A_mat = np.zeros((2*n-1, n))
        A_mat[valid] = a[diff[valid].astype(np.int64)]
        
        constraints = [A_mat @ g <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        
        solved = False
        for solver in [cp.CBC, cp.GLPK]:
            try:
                prob.solve(solver=solver, verbose=False)
                if prob.status in ['optimal', 'optimal_inaccurate']:
                    gv = g.value
                    if gv is not None and np.isfinite(gv).all():
                        gv = np.maximum(gv, 0)
                        s = gv.sum()
                        if s > 1e-15:
                            g = gv / s
                            solved = True
                            break
            except Exception:
                continue
        
        if not solved:
            # Subgradient fallback
            grad = np.zeros(n)
            for m in range(n):
                j = k_star - m
                if 0 <= j < n:
                    grad[m] = 2.0 * a[j]
            # Projected gradient step
            for step in [0.01, 0.05, 0.1, 0.2, 0.5]:
                a_new = a - step * grad

            if cos_sim > 0.9999:

                a += np.random.default_rng(it*7919).normal(0, 0.02, n)
                a = _normalize(a)
                continue

            best_t = 0.0

        sc = _score_fast(a)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
