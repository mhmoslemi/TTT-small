def build_conv_matrix(a):
    """A @ b = convolve(a, b) restricted to nonnegative indices."""
    n = len(a)
    m = 2*n - 1
    indptr = np.zeros(m+1, dtype=np.int32)
    for k in range(m):
        count = min(k+1, n, 2*n-1-k)  # number of valid j
        # Actually: j ranges from max(0, k-n+1) to min(n-1, k)
        lo = max(0, k-n+1)
        hi = min(n-1, k)
        count = hi - lo + 1
        indptr[k+1] = indptr[k] + count
    
    total = indptr[-1]
    data = np.zeros(total, dtype=np.float64)
    cols = np.zeros(total, dtype=np.int32)
    pos = 0
    for k in range(m):
        lo = max(0, k-n+1)
        hi = min(n-1, k)
        for j in range(lo, hi+1):
            data[pos] = a[k-j]
            cols[pos] = j
            pos += 1
    
    return sps.csr_matrix((data, cols, indptr), shape=(m, n))
