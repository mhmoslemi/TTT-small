Let me think about what happens at the LP optimum. By duality, the LP has:
- Primal: max `1^T g` s.t. `A g <= M * 1`, `g >= 0`
- Dual: min `M * 1^T y` s.t. `A^T y >= 1`, `y >= 0`

where `A` is the `(2n-1) x n` matrix with `A[k,j] = a_{k-j}` if valid.

If `1^T g* = M * 1^T y* > sum a`, we can improve.
If `1^T g* = sum a`, we're at equilibrium.

The dual variables `y*` tell us which constraints are "active" (which `k` values are important).

This gives a nice interpretation: the LP finds a "dual certificate" that proves the current `M` is optimal (if `sum g* = sum a`) or gives a direction to improve.

    return best_a, best_mb


def _init_variants(n, rng, variant):
    i = np.arange(n, dtype=np.float64)
    v = int(variant) % 12
    if v == 0:
        a = 1.0 + 0.4 * np.sin(2 * np.pi * rng.integers(1, n // 2 + 1) * i / n + rng.uniform(0, 6.28)) \
            + 0.3 * np.cos(2 * np.pi * rng.integers(1, n // 2 + 1) * i / n + rng.uniform(0, 6.28))
    elif v == 1:
        a = np.exp(rng.normal(0, 0.7, n))
    elif v == 2:
        a = 0.5 + 0.5 * np.abs(np.sin(np.pi * i / n * rng.uniform(1, 4)))
    elif v == 3:
        a = np.abs(rng.normal(0, 1, n)) + 0.1
    elif v == 4:
        a = np.ones(n)
        for _ in range(4):
            c = rng.uniform(0, n)
            w = rng.uniform(n // 8, n // 3)
            a += rng.uniform(0.2, 0.8) * np.exp(-(i - c) ** 2 / (2 * w ** 2))
    elif v == 5:
        a = 1.0 + 0.6 * np.sin(2 * np.pi * i / n * rng.integers(2, max(3, n // 4)))
        a += 0.2 * np.sin(2 * np.pi * i / n * rng.integers(5, max(6, n // 2)))
    elif v == 6:
        a = rng.uniform(0.3, 1.5, n)
    elif v == 7:
        a = (i + 1) / n ** 2 * 2.0
    elif v == 8:
        a = np.exp(-i / (n * rng.uniform(0.1, 0.5)))
    elif v == 9:
        a = np.abs(np.sin(np.pi * i / n)) + 0.1 * np.cos(2 * np.pi * np.pi * i / n)
    elif v == 10:
        a = 0.5 + 0.3 * (i / n ** 2) + 0.2 * np.random.default_rng(variant * 31).uniform(0, 1, n)
    else:
        a = np.concatenate([np.exp(-i[:n//2] / (n * 0.1)), np.ones(n // 2) * 0.3])
    a = np.maximum(a, 0.01)
    return a / a.sum()


def _evaluate_fast(a):
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return np.inf
    return 2.0 * n * _maxb(a) / (S * S)


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(1.0, float(budget_s))
    deadline = t0 + budget - 15.0

    parent = _get_parent()
    n_parent = len(parent)

    best_seq = np.clip(parent, 0, 1000).astype(np.float64).copy()
    try:
        best_score = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        best_score = math.inf
    if not np.isfinite(best_score):
        best_seq = np.ones(32)
        best_score = 2.0

    rng = np.random.default_rng(seed)

    def _try_update(candidate):
        nonlocal best_seq, best_score
        try:
            fast = _evaluate_fast(candidate)
            if fast < best_score - 1e-9:
                exact = float(evaluate_sequence([float(x) for x in candidate]))
                if np.isfinite(exact) and exact < best_score:
                    best_seq = candidate.copy()
                    best_score = exact
                    return True
        except Exception:
            pass
        return False

    # Phase 1: LP from parent
    if time.monotonic() < deadline - 60:
        dl1 = min(deadline - 40, t0 + budget * 0.35)
        a0 = _norm1(best_seq.copy())
        try:
            res_a, _ = _lp_optimize(a0, dl1, rng)
            _try_update(res_a)
        except Exception:
            pass

    # Phase 2: LP from diverse inits at parent length
    if time.monotonic() < deadline - 40:
        dl2 = min(deadline - 30, t0 + budget * 0.60)
        n_restarts = 20
        for rid in range(n_restarts):
            if time.monotonic() > dl2 - 5 or best_score < 1.50:
                break
            a_init = _init_variants(n_parent, rng, variant=rid + 30)
            remaining = max(3.0, dl2 - time.monotonic() - 5)
            sub_dl = time.monotonic() + min(30.0, remaining / max(1, n_restarts - rid))
            try:
                res_a, _ = _lp_optimize(a_init, sub_dl, rng)
                _try_update(res_a)
            except Exception:
                pass

    # Phase 3: Smooth optimization on current best
    if time.monotonic() < deadline - 25:
        a_cached = _norm1(best_seq.copy() / (best_seq.sum() + 1e-30))
        dl3 = min(deadline - 20, time.monotonic() + min(60.0, (deadline - time.monotonic()) * 0.4))
        try:
            res_a, _ = _smooth_optimize(a_cached, dl3, rng)
            _try_update(res_a)
        except Exception:
            pass

    # Phase 4: More LP restarts with asymmetric inits
    if time.monotonic() < deadline - 20:
        dl4 = min(deadline - 15, time.monotonic() + min(80.0, (deadline - time.monotonic()) * 0.5))
        for rid in range(15):
            if time.monotonic() > dl4 - 5:
                break
            if best_score < 1.50:
                break
            # Asymmetric inits specifically
            a_init = _init_variants(n_parent, rng, variant=rid + 60)
            # Make it more asymmetric
            n_half = n_parent // 2
            boost = rng.uniform(0.7, 2.0)
            a_init[:n_half] *= boost
            a_init[n_half:] /= boost
            a_init = _norm1(a_init)
            sub_dl = time.monotonic() + min(20.0, max(3.0, (dl4 - time.monotonic()) / max(1, 15 - rid)))
            try:
                res_a, _ = _lp_optimize(a_init, sub_dl, rng)
                _try_update(res_a)
            except Exception:
                pass

    # Phase 5: Local search polish on best
    if time.monotonic() < deadline - 10:
        a_polish = _norm1(best_seq.copy())
        dl5 = deadline - 8
        try:
            res_a, _ = _local_search(a_polish, dl5, np.random.default_rng(seed + 12345))
            _try_update(res_a)
        except Exception:
            pass

    # Phase 6: Try a few different lengths
    if time.monotonic() < deadline - 10:
        for L in [128, 256, 96, 320]:
            if time.monotonic() > deadline - 8:
                break
            if L == n_parent:
                continue
            a_init = _init_variants(L, rng, variant=L % 12)
            # Asymmetric
            n_half = L // 2
            a_init[:n_half] *= rng.uniform(0.8, 1.5)
            a_init[n_half:] /= rng.uniform(0.8, 1.5)
            a_init = _norm1(a_init)
            sub_dl = time.monotonic() + max(3.0, min(15.0, deadline - time.monotonic() - 6))
            try:
                res_a, _ = _lp_optimize(a_init, sub_dl, rng)
                _try_update(res_a)
                if time.monotonic() < deadline - 5:
                    res_a2, _ = _local_search(res_a, deadline - 4, np.random.default_rng(seed + L))
                    _try_update(res_a2)
            except Exception:
                pass

    # Final: one more local search with whatever time remains
    if time.monotonic() < deadline - 3:
        a_final = _norm1(best_seq.copy())
        try:
            res_a, _ = _local_search(a_final, deadline - 2, np.random.default_rng(seed + 99999))
            _try_update(res_a)
        except Exception:
            pass

    # Final validation
    try:
        final_score = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_score = math.inf

    if not np.isfinite(final_score):
        fb = np.ones(32)
        return [float(x) for x in fb]

    return [float(x) for x in best_seq]
