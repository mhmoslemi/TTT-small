minimize max(a[0]², 2a[0]a[1], 2a[0]a[2]+a[1]², ..., ||a||², ..., a[0]²)
subject to sum(a) = 1, a ≥ 0, a[i] = a[n-1-i]

Since ||a||² is likely the max, we want to minimize ||a||² subject to sum(a) = 1, which gives a[i] = 1/n for all i, with ||a||² = 1/n.

But the other constraints also matter! For example, c[k] for small k involves only a few terms and might be constrained.

import numpy as np
import time
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog
import math


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(0.0, float(budget_s))
    deadline = t0 + budget - 18.0

    # Get parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
    except Exception:
        parent = np.ones(192, dtype=np.float64)
    parent = np.clip(parent, 0.0, 1000.0)
    if parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(192, dtype=np.float64)

    n = parent.size
    rng = np.random.default_rng(seed)
    fsize = next_fast_len(2 * n - 1)

    def conv(a):
        F = rfft(a, fsize)
        c = irfft(F * F, fsize)[:2 * n - 1]
        np.maximum(c, 0.0, out=c)
        c[~np.isfinite(c)] = 0.0
        return c

    def fast_score(a):
        s = a.sum()
        if s < 1e-15:
            return np.inf
        b = conv(a)
        return 2.0 * n * b.max() / (s * s)

    def project_simplex(v):
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s < 1e-15:
            return np.ones(n) / n
        return v / s

    def palindromize(a):
        return 0.5 * (a + a[::-1])

    def subgradient(a):
        b = conv(a)
        bmax = b.max()
        tol = 1e-9 * bmax
        active = np.where(b >= bmax - tol)[0]
        grad = np.zeros(n)
        for k in active:
            k = int(k)
            if k == 0:
                grad += 2.0 * a
            elif k == 2 * n - 2:
                grad[n - 1] += 2.0 * a[n - 1]
            # General: (a*a)[k] = sum_j a[j]*a[k-j], d/d a[i] = a[k-i] (if valid) + a[i] (if i==k-i, i.e., 2i=k)... 
            # Actually: (a*a)[k] = sum_{j: 0<=j<n, 0<=k-j<n} a[j]*a[k-j]
            # d/d a[i]: term where j=i gives a[i]*a[k-i] if 0<=k-i<n
            #           term where k-j=i i.e. j=k-i gives a[k-i]*a[i] if 0<=k-i<n
            # If i == k-i (2i=k): contribution is 2*a[i]*a[i]... no, just a[i]*a[i] from j=i AND a[k-i]*a[i] from j=k-i, but j=i=k-i so it's the same term.
            # More carefully: (a*a)[k] = sum_j a[j]*a[k-j]
            # d/d a[i] = a[k-i] (from j=i term) + a[i] (from k-j=i term, j=k-i)
            # If k-i == i (i.e., 2i=k), both terms are the same: d/d a[i] = 2*a[i]... no, it's a[i]*a[i] differentiated = 2a[i] only if we treat it as product.
            # Actually: (a*a)[k] = sum_j a[j]*a[k-j]. The term involving a[i] appears when j=i (giving a[i]*a[k-i]) and when k-j=i i.e. j=k-i (giving a[k-i]*a[i]).
            # If i == k-i, these are the same term: a[i]*a[i], derivative = 2*a[i].
            # If i != k-i, we get two terms: a[k-i] + a[i].
            # So: d(a*a)[k]/d a[i] = a[k-i] (if 0<=k-i<n) + a[i] (if 0<=k-i<n) when i != k-i
            #                                         = 2*a[i] when 2i=k and i is in range
            # Simplified: = a[k-i]*[0<=k-i<n] + a[i]*[0<=k-i<n] = (a[k-i]+a[i])*[0<=k-i<n]
            # But wait that's wrong when 2i=k: it should be 2*a[i], and a[k-i]+a[i] = a[i]+a[i] = 2*a[i]. OK same.
            if 0 <= k - i < n if False:  # placeholder
                pass
            # Vectorized version:
            # For each i in 0..n-1: contribution is a[k-i] if 0<=k-i<n, plus a[i] if 0<=k-i<n
            # i.e., for valid range of i where 0<=k-i<n, i.e., k-n+1 <= i <= k (and 0<=i<n)
            # So: for i in [max(0,k-n+1), min(n-1,k)]: grad[i] += a[k-i] + a[i]
            lo_i = max(0, k - (n - 1))
            hi_i = min(n - 1, k)
            if lo_i <= hi_i:
                idx = np.arange(lo_i, hi_i + 1)
                grad[idx] += a[k - idx] + a[idx]
        return grad

    def solve_lp(a, M):
        # Maximize sum(g) s.t. (a*g)[k] <= M for all k, g >= 0
        # (a*g)[k] = sum_j a[j]*g[k-j], linear in g
        # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
        k_idx = np.arange(2 * n - 1)[:, None]
        j_idx = np.arange(n)[None, :]
        diff = k_idx - j_idx
        A = np.zeros((2 * n - 1, n))
        valid = (diff >= 0) & (diff < n)
        A[valid] = a[diff[valid].astype(np.int64)]

        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                          bounds=[(0, None)] * n,
                          method='highs',
                          options={'presolve': True, 'time_limit': 5.0})
            if res.success and res.x is not None:
                g = np.maximum(res.x, 0.0)
                s = g.sum()
                if s > 1e-12:
                    return g / s
        except Exception:
            pass
        return None

    def line_search(a, direction, base_mb):
        best_a = a.copy()
        best_mb = base_mb
        # Binary-ish search over step sizes
        for t in [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.5, 0.7]:
            a_new = (1.0 - t) * a + t * direction
            a_new = palindromize(a_new)
            a_new = project_simplex(a_new)
            mb = conv(a_new).max()
            if mb < best_mb:
                best_mb = mb
                best_a = a_new.copy()
        return best_a, best_mb

    # Initialize
    a = project_simplex(parent.copy())
    a = palindromize(a)
    a = project_simplex(a)

    best_a = a.copy()
    best_mb = float(conv(best_a).max())
    best_score = float(fast_score(best_a))

    # Phase 1: FW iterations (fast, ~O(n) each)
    p1_end = min(deadline, t0 + budget * 0.25)
    fw_count = 0
    fw_no_improve = 0
    while time.monotonic() < p1_end:
        grad = subgradient(a)
        i_min = int(np.argmin(grad))
        v = np.zeros(n)
        v[i_min] = 1.0
        # FW step size: 2/(t+2) for linear convergence, or fixed
        t_fw = 2.0 / (fw_count + 3)
        a_new = (1.0 - t_fw) * a + t_fw * v
        a_new = palindromize(a_new)
        a_new = project_simplex(a_new)
        mb = conv(a_new).max()
        if mb < best_mb - 1e-13:
            best_mb = mb
            best_a = a_new.copy()
            best_score = fast_score(a_new)
            fw_no_improve = 0
        else:
            fw_no_improve += 1
        a = a_new
        fw_count += 1
        if fw_no_improve > 800:
            a = palindromize(a + rng.normal(0, 0.03, n))
            a = project_simplex(a)
            fw_no_improve = 0

    # Phase 2: LP-based optimization
    p2_end = min(deadline, t0 + budget * 0.60)
    lp_count = 0
    lp_no_improve = 0
    while time.monotonic() < p2_end:
        b = conv(a)
        M = float(b.max())

        g = solve_lp(a, M)
        if g is not None:
            g = palindromize(g)
            g = project_simplex(g)
            cos_sim = float(np.dot(a, g) / (np.linalg.norm(a) * np.linalg.norm(g) + 1e-30))
            if cos_sim < 0.9999:
                a_new, new_mb = line_search(a, g, M)
                if new_mb < M - 1e-12:
                    a = a_new
                if conv(a).max() < best_mb - 1e-12:
                    best_mb = float(conv(a).max())
                    best_a = a.copy()
                    best_score = fast_score(a)
                    lp_no_improve = 0
                else:
                    lp_no_improve += 1
            else:
                lp_no_improve += 1
        else:
            # Subgradient fallback
            grad = subgradient(a)
            gn = float(np.linalg.norm(grad))
            if gn > 1e-15:
                for step in [0.02, 0.05, 0.1, 0.2]:
                    a_new = a - step * grad / gn
                    a_new = palindromize(a_new)
                    a_new = project_simplex(a_new)
                    if conv(a_new).max() < M:
                        a = a_new
                        break
                if conv(a).max() < best_mb - 1e-12:
                    best_mb = float(conv(a).max())
                    best_a = a.copy()
                    best_score = fast_score(a)
                    lp_no_improve = 0
                else:
                    lp_no_improve += 1
            else:
                lp_no_improve += 5

        lp_count += 1
        if lp_no_improve > 80:
            a = palindromize(a + rng.normal(0, 0.04, n))
            a = project_simplex(a)
            lp_no_improve = 0

    # Phase 3: Local random search polishing
    p3_end = min(deadline, t0 + budget * 0.85)
    sigma = 0.025
    ls_no_improve = 0
    ls_count = 0
    while time.monotonic() < p3_end:
        r = rng.random()
        if r < 0.35:
            a_new = a + rng.normal(0, sigma, n)
        elif r < 0.55:
            a_new = a.copy()
            idx = int(rng.integers(0, n))
            w = int(rng.integers(1, max(2, n // 8)))
            lo = max(0, idx - w)
            hi = min(n, idx + w)
            a_new[lo:hi] *= np.exp(rng.normal(0, 0.2, hi - lo))
        elif r < 0.70:

        a_new = project_simplex(a_new)

            best_mb = mb
            best_a = a_new.copy()
            best_score = fast_score(a_new)

            ls_no_improve = 0

            v[i_min] = 1.0

            a_new = project_simplex(a_new)
            if conv(a_new).max() < conv(a).max():

            best_mb = float(conv(a).max())
            best_a = a.copy()
            best_score = fast_score(a)

    # Final validation with exact evaluator
    try:

            final_score = float(evaluate_sequence(candidate_list))
    except Exception:
        candidate_list = [1.0 / 32.0] * 32
        try:
            final_score = float(evaluate_sequence(candidate_list))

        candidate_list = [1.0 / 192.0] * 192
