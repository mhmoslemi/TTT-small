import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _get_parent():

        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(320, dtype=np.float64)
    if p.size == 0:

    p = np.clip(p, 0.0, 1000.0)

def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)

    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A

def _lp_direction(f, n, M_target):
    """Solve LP to find descent direction."""
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M_target)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                      method='highs', options={'presolve': True, 'time_limit': 5})

def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    Fa = rfft(a, fsize)
    c = irfft(Fa * Fa, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score_fast(a):
    """Fast score computation (no evaluator call)."""
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    return 2.0 * n * m / (s * s)

    """Normalize f to have sum = n, clip to nonnegative."""
    f = np.clip(f, 0, None)
    s = f.sum()
    if s < 1e-15:
        return np.ones(n) / n * n  # sum = n
    return f * n / s

    for j in range(n):
        A[j:j + n, j] = f
    return A

    """Solve LP: max sum(g) s.t. A@g <= M_target, g >= 0."""
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M_target)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                      method='highs', options={'presolve': True, 'time_limit': 10})

    """Subgradient + proximal optimization."""
    rng = np.random.default_rng(seed)
    f = _normalize(f0.copy(), n)
    best_f = f.copy()

    while time.monotonic() < stop_time - 5.0:
        conv = _conv_full(f)
        M = float(conv.max())
        tol = 0.02 * M + 1e-14
        active = np.where(conv >= M - tol)[0]
        
        # Compute subgradient
        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            # d conv[k]/d a[j] = 2*a[k-j] for j in [lo, hi]
            grad[lo:hi+1] += 2.0 * f[k - np.arange(lo, hi + 1)]

        f_new = f - lr * grad

        if f_new.max() <= 0:
            f_new = np.ones(n)
            lr *= 0.5

            # Cumulative sum approach for projection
            cumsum = np.cumsum(vals)
            # Find largest i such that vals[i] - (cumsum[i] - n)/(i+1) > 0
            theta = (cumsum - n) / np.arange(1, n + 1)
            valid = vals > theta
            if not valid.any():

            else:
                i_star = np.where(valid)[0][-1]
                lam = (cumsum[i_star] - n) / (i_star + 1)
                f_new = np.maximum(f_new - lam, 0)

            f = f_new

        if time.monotonic() > p1_end - 8.0:
            break
        stop = min(p1_end - 5.0, time.monotonic() + tpi)
        try:
            res, sc = _lp_loop(arr, n, stop, seed + 1000*(idx+1))
            _try(res, f"lp_{nm}")
        except Exception:
            pass
    
    # Phase 2: Gradient refinement
    if time.monotonic() < deadline - 80.0:
        p2_end = min(deadline - 40.0, time.monotonic() + budget*0.13)
        try:
            res, sc = _grad_loop(best_seq, len(best_seq), p2_end, seed+50000)
            _try(res, "grad")
        except Exception:
            pass
