    if time.monotonic() < p2 - 5:
        try:
            a2 = _grad_opt(best_seq, p2)
            try_update(a2)
        except Exception:
            pass

    # Phase 3: LP from best
    p3 = t0 + budget * 0.80
    if time.monotonic() < p3 - 5:
        try:
            a3 = _lp_opt(best_seq, p3)
            try_update(a3)
        except Exception:
            pass

    # Phase 4: Local search
    p4 = t0 + budget * 0.92
    if time.monotonic() < p4 - 10:
        try:
            a4 = _local(best_seq, p4, rng)
            try_update(a4)
        except Exception:
            pass

    # Phase 5: Final LP polish + gradient
    p5 = t0 + budget * 0.97
    if time.monotonic() < p5 - 5:
        try:
            a5 = _lp_opt(best_seq, t0 + budget * 0.95)
            try_update(a5)
        except Exception:
            pass
        if time.monotonic() < p5 - 3:
            try:
                a6 = _grad_opt(best_seq, p5)
                try_update(a6)
            except Exception:
                pass

    # Final validation
    try:
        fs = float(evaluate_sequence([float(x) for x in best_seq]))
        if not np.isfinite(fs):
            raise ValueError
    except Exception:
        return [1.0] * 32

    return [float(x) for x in best_seq]
