```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
import cvxpy as cp


def _fft_autocorr(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    c[~np.isfinite(c)] = 0.0
    return c


def _score_fast(a, n):
    return 2.0 * n * float(_fft_autocorr(a).max())


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-14:
        return np.ones(len(a)) / len(a)
    return a / s


def _build_A(a):
    n = len(a)
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n))
    A[valid] = a[diff[valid].astype(np.int64)]
    return A


def _lp_solve(a, M, n, deadline):
    """Solve LP: max sum(g) s.t. g>=0, (a*g)[k] <= M."""
    g = cp.Variable(n, nonneg=True)
    A = _build_A(a)
    constraints = [A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    for solver in [cp.GLOP, cp.CBC, cp.GLPK, cp.PDCO, cp.SCS]:
        if time.monotonic() > deadline:
            break
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                gv = g.value
                if gv is not None and np.all(np.isfinite(gv)):
                    gv = np.maximum(gv, 0.0)
                    s = gv.sum()
                    if s > 1e-12:
                        return gv / s
        except Exception:
            continue
    return None


def _subgrad(a, n):
    """Subgradient of max autocorrelation at current point."""
    b = _fft_autocorr(a)
    k_star = int(np.argmax(b))
    grad = np.zeros(n)
    for m in range(n):
        j = k_star - m
        if 0 <= j < n:
            grad[m] = 2.0 * a[j]
    return grad


def _multi_lag_grad(a, n, top_k=5):
    """Gradient using top-k autocorrelation lags."""
    b = _fft_autocorr(a)
    top_idx = np.argsort(b)[-top_k:]
    grad = np.zeros(n)
    for k in top_idx:
        k = int(k)
        for m in range(n):
            j = k - m
            if 0 <= j < n:
                grad[m] += 2.0 * a[j]
    return grad


def _line_search(a, g, n, max_evals=25):
    """Golden section line search on (1-t)a + t*g."""
    def f(t):
        x = (1.0 - t) * a + t * g
        x = _normalize(x)
        return _score_fast(x, n)

    gr = (math.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    a1 = hi - gr * (hi - lo)
    a2 = lo + gr * (hi - lo)
    f1, f2 = f(a1), f(a2)
    for _ in range(max(0, max_evals - 2)):
        if f1 < f2:
            hi, a2, f2 = a2, a1, f1
            a1 = hi - gr * (hi - lo)
            f1 = f(a1)
        else:
            lo, a1, f1 = a1, a2, f2
            a2 = lo + gr * (hi - lo)
            f2 = f(a2)
    t_opt = (lo + hi) / 2.0
    s_opt = min(f1, f2)
    # Check t=1
    s1 = f(1.0)
    if s1 < s_opt:

    return t_opt, s_opt

        M = float(b.max())
        cur_sc = 2.0 * n * M

        if cur_sc < best_sc * (1.0 - 1e-11):
            best_sc = cur_sc
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1

        if no_imp > patience:
            # Aggressive perturbation
            mode = rng.integers(0, 4)
            if mode == 0:
                a = a + rng.normal(0, 0.06, n)
            elif mode == 1:
                f = rng.uniform(1.0, 8.0)
                ph = rng.uniform(0, 2 * np.pi)
                i = np.arange(n)
                a = a + 0.08 * np.sin(2 * np.pi * f * i / n + ph)
            elif mode == 2:
                a = 0.8 * a + 0.2 * a[::-1] + rng.normal(0, 0.03, n)
            else:
                lo = int(rng.integers(0, n // 3))
                w = int(rng.integers(n // 15, n // 4))
                hi = min(n, lo + w)
                a = a.copy()
                a[lo:hi] *= rng.uniform(0.4, 2.0)
            a = _normalize(a)
            no_imp = 0
            patience = max(20, patience + 3)
            continue

        # LP direction
        g = _lp_solve(a, M, n, deadline - 3.0)
        if g is None:
            grad = _subgrad(a, n)
            gn = float(np.linalg.norm(grad))
            if gn < 1e-14:
                break
            g = _normalize(a - 0.15 * grad / gn)

        cos_sim = float(np.dot(a, g) / (np.linalg.norm(a) * np.linalg.norm(g) + 1e-30))
        if cos_sim > 0.999:
            no_imp += 5
            continue

        t, new_sc = _line_search(a, g, n, max_evals=20)
        if new_sc < cur_sc * (1.0 - 1e-9):
            a = _normalize((1.0 - t) * a + t * g)
        else:
            no_imp += 2

    return best_a, best_sc


def _local_search(a, n, deadline, rng):
    a = _normalize(a.copy())
    best_a = a.copy()
    best_sc = _score_fast(a, n)
    sigma = 0.025
    no_imp = 0

    while time.monotonic() < deadline - 1.0:
        r = rng.random()
        if r < 0.25:
            cand = a + rng.normal(0, sigma, n)
        elif r < 0.40:
            cand = a.copy()
            lo = int(rng.integers(0, max(1, n // 4)))
            w = int(rng.integers(max(2, n // 25), n // 4))
            hi = min(n, lo + w)
            cand[lo:hi] *= rng.uniform(0.5, 1.8)
        elif r < 0.50:
            cand = 0.5 * (a + a[::-1]) + rng.normal(0, sigma * 0.4, n)
        elif r < 0.60:
            f = rng.uniform(0.3, 6.0)
            ph = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.03, 0.2)
            i = np.arange(n)
            cand = a + amp * np.sin(2 * np.pi * f * i / n + ph)
        elif r < 0.70:
            cand = a.copy()
            n_swap = int(rng.integers(2, 6))
            for _ in range(n_swap):
                i1, i2 = rng.integers(0, n, 2)
                d = rng.uniform(-0.04, 0.04)
                cand[i1] += d
                cand[i2] -= d
        elif r < 0.80:
            w = rng.uniform(0.0, 0.25)
            cand = (1.0 - w) * a + w * np.ones(n) / n
        elif r < 0.90:
            # Exponential smoothing blend
            alpha = rng.uniform(0.4, 0.8)
            sm = np.zeros(n)
            sm[0] = a[0]
            for idx in range(1, n):
                sm[idx] = alpha * a[idx] + (1 - alpha) * sm[idx - 1]
            cand = 0.5 * a + 0.5 * sm
        else:
            # Multi-point random
            cand = a.copy()
            for _ in range(int(rng.integers(3, 10))):
                idx = int(rng.integers(0, n))
                cand[idx] *= rng.uniform(0.6, 1.6)

        cand = _normalize(cand)
        sc = _score_fast(cand, n)
        if sc < best_sc * (1.0 - 1e-10):
            best_sc = sc
            best_a = cand.copy()
            a = cand.copy()
            no_imp = 0
            sigma = max(0.004, sigma * 0.97)
        else:
            no_imp += 1
            if no_imp > 120:
                sigma = min(0.25, sigma * 1.4)
                no_imp = 0

    return best_a, best_sc


def _init_variant(n, rng, variant):
    i = np.arange(n, dtype=np.float64)
    v = int(variant) % 8
    if v == 0:
        f1 = int(rng.integers(1, n // 2 + 1))
        f2 = int(rng.integers(1, n // 2 + 1))
        a = 1.0 + 0.4 * np.sin(2 * np.pi * f1 * i / n + rng.uniform(0, 6.28)) + \
            0.3 * np.cos(2 * np.pi * f2 * i / n + rng.uniform(0, 6.28))
    elif v == 1:
        a = np.exp(rng.normal(0, 0.7, n))
    elif v == 2:
        a = 0.5 + 0.5 * np.abs(np.sin(np.pi * i / n * rng.uniform(1, 5)))
    elif v == 3:
        a = np.abs(rng.normal(0, 1, n)) + 0.1
    elif v == 4:
        a = np.ones(n)
        for _ in range(4):
            c = rng.uniform(0, n)
            w = rng.uniform(n // 10, n // 3)
            a += rng.uniform(0.2, 0.8) * np.exp(-(i - c) ** 2 / (2 * w * w))
    elif v == 5:
        a = 1.0 + 0.5 * np.sin(2 * np.pi * i / n * int(rng.integers(2, max(3, n // 5))))
        a += 0.2 * np.sin(2 * np.pi * i / n * int(rng.integers(4, max(5, n // 3))))
    elif v == 6:
        a = np.maximum(0.3, 1.0 + 0.5 * np.cos(2 * np.pi * i / n * 3.0) +
                        0.3 * np.cos(2 * np.pi * i / n * 7.0))
    else:
        a = rng.uniform(0.3, 1.5, n)
    a = np.maximum(a, 0.01)
    return _normalize(a)


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(1.0, float(budget_s))
    deadline = t0 + budget - 12.0

    # Get parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
    except Exception:
        parent = np.ones(192, dtype=np.float64)
    parent = np.clip(parent, 0.0, 1000.0)
    if parent.size == 0 or parent.sum() < 0.01:
        parent = np.ones(192, dtype=np.float64) / 192.0

    n = len(parent)
    a0 = _normalize(parent.copy())

    # Exact evaluation of parent
    best_seq = list(parent)
    best_score = float(evaluate_sequence(best_seq))
    if not np.isfinite(best_score):
        best_score = float('inf')
        best_seq = [1.0] * 32
        best_score = 2.0

    rng = np.random.default_rng(seed)

    # Phase 1: LP optimization from parent (50% budget)
    p1_dl = min(deadline - 80, t0 + budget * 0.50)
    if time.monotonic() < p1_dl:
        try:
            res_a, res_sc = _lp_phase(a0, n, p1_dl, rng)
            # Scale back to original magnitude for exact eval
            scale = parent.sum()
            cand = res_a * scale
            exact = float(evaluate_sequence([float(x) for x in cand]))
            if np.isfinite(exact) and exact < best_score:
                best_score = exact
                best_seq = [float(x) for x in cand]
        except Exception:
            pass

    # Phase 2: LP from diverse initializations (20% budget)
    p2_dl = min(deadline - 60, t0 + budget * 0.72)
    if time.monotonic() < p2_dl:
        n_restarts = 8
        for rid in range(n_restarts):
            if time.monotonic() > p2_dl - 8:
                break
            if best_score < 1.505:
                break
            a_init = _init_variant(n, rng, variant=rid + 50)
            sub_dl = time.monotonic() + max(4.0, (p2_dl - time.monotonic() - 6.0) / max(1, n_restarts - rid))
            sub_dl = min(sub_dl, p2_dl - 5.0)
            try:
                res_a, res_sc = _lp_phase(a_init, n, sub_dl, rng)
                scale = parent.sum()
                cand = res_a * scale
                exact = float(evaluate_sequence([float(x) for x in cand]))
                if np.isfinite(exact) and exact < best_score:
                    best_score = exact
                    best_seq = [float(x) for x in cand]
            except Exception:
                pass

    # Phase 3: Local search refinement (15% budget)
    p3_dl = min(deadline - 40, t0 + budget * 0.87)
    if time.monotonic() < p3_dl:
        try:
            a_cur = _normalize(np.array(best_seq, dtype=np.float64))
            res_a, res_sc = _local_search(a_cur, n, p3_dl, rng)
            scale = np.array(best_seq).sum()
            cand = res_a * scale
            exact = float(evaluate_sequence([float(x) for x in cand]))
            if np.isfinite(exact) and exact < best_score:
                best_score = exact
                best_seq = [float(x) for x in cand]
        except Exception:
            pass

    # Phase 4: Another LP round on best (5% budget)
    if time.monotonic() < deadline - 20:
        try:
            a_cur = _normalize(np.array(best_seq, dtype=np.float64))
            res_a, res_sc = _lp_phase(a_cur, n, deadline - 18, np.random.default_rng(seed + 999))
            scale = np.array(best_seq).sum()
            cand = res_a * scale
            exact = float(evaluate_sequence([float(x) for x in cand]))
            if np.isfinite(exact) and exact < best_score:
                best_score = exact
                best_seq = [float(x) for x in cand]
        except Exception:
            pass

    # Phase 5: Final local search polish
    if time.monotonic() < deadline - 12:
        try:
            a_cur = _normalize(np.array(best_seq, dtype=np.float64))
            res_a, res_sc = _local_search(a_cur, n, deadline - 11, np.random.default_rng(seed + 7777))
            scale = np.array(best_seq).sum()
            cand = res_a * scale
            exact = float(evaluate_sequence([float(x) for x in cand]))
            if np.isfinite(exact) and exact < best_score:
                best_score = exact
                best_seq = [float(x) for x in cand]
        except Exception:
            pass

    # Final validation
    try:
        final = float(evaluate_sequence(list(best_seq)))
        if not np.isfinite(final):
            raise ValueError("non-finite")
    except Exception:
        # Fallback: return a known valid sequence
        fb = [1.0] * 32
        try:
            fs = float(evaluate_sequence(fb))
            if np.isfinite(fs):
                return fb
        except Exception:
            pass
        return [1.0]

    return list(best_seq)
```
