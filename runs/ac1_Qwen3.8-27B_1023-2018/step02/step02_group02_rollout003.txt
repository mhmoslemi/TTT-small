def simplex_project(v):
    v = np.asarray(v, dtype=float).ravel()
    if not np.all(np.isfinite(v)):
        return np.full(n, 1.0/n)
    # If sum close, normalize? projection to sum=1 nonnegative.
    # Standard algorithm:
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    # positive rho where u > cssv/(1..n)
    idx = np.nonzero(u > cssv / np.arange(1, n+1))[0]
    if idx.size == 0:
        # all v make negative? return uniform
        return np.full(n, 1.0/n)

    w = np.maximum(v - theta, 0.0)
    ss = w.sum()
    if ss <= 0 or not finite: return uniform
    # due numerical, normalize to exact 1

def _proj_simplex(v, n):
    v=np.asarray(v,dtype=float).ravel()
    if v.size!=n: v=np.resize(v,n)
    v=np.where(np.isfinite(v),v,0.0)
    if np.all(v <= 0): return np.full(n,1.0/n)
    u=np.sort(v)[::-1]
    cssv=np.cumsum(u)-1.0
    # rho = np.nonzero(u - cssv/np.arange(1,n+1) > 0)[0]
    tmp = u - cssv/np.arange(1,n+1)
    rho = np.nonzero(tmp > 0)[0]
    if rho.size==0:
        return np.full(n,1.0/n)
    R = rho[-1]
    theta = cssv[R]/(R+1)
    w = np.maximum(v - theta, 0.0)
    s=w.sum()
    if s <= 0 or not np.isfinite(s): return np.full(n,1.0/n)
    return w/s
```

def _normalize(a, n):

    if n <= 0:

def _resample(x, m):
    """Piecewise-linear resampling to length m, preserving nonnegativity and sum."""
    if m <= 0:
        return np.array([], dtype=np.float64)
    try:
        x = np.asarray(x, dtype=np.float64).ravel()
    except Exception:

    s = float(x.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(m, 1.0 / m, dtype=np.float64)

def _proj_simplex(v, n):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    if n <= 0:
        return np.array([], dtype=np.float64)
    try:
        v = np.asarray(v, dtype=np.float64).ravel()
    except Exception:
        return np.full(n, 1.0 / n, dtype=np.float64)
    if v.size != n:
        v = np.resize(v, n)
    v = np.where(np.isfinite(v), v, 0.0)
    if np.all(v <= 0.0):
        return np.full(n, 1.0 / n, dtype=np.float64)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    denom = np.arange(1, n + 1, dtype=np.float64)
    rho = np.nonzero(u - cssv / denom > 0.0)[0]
    if rho.size == 0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    R = int(rho[-1])
    theta = cssv[R] / (R + 1.0)
    w = np.maximum(v - theta, 0.0)
    sw = float(w.sum())
    if not np.isfinite(sw) or sw <= 1e-300:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return w / sw


def _make_conv(n):
    """Return a fast self-convolution operator for length n."""
    L = 2 * n - 1
    fs = next_fast_len(L)

    def conv(a):
        F = rfft(a, fs)
        F *= F
        c = irfft(F, fs)
        if c.size != L:
            c = c[:L]
        c = np.asarray(c, dtype=np.float64)
        np.maximum(c, 0.0, out=c)
        c[~np.isfinite(c)] = 0.0
        return c

    return conv

        else:
            mode = it % 4
            cyc = (it // 7) % 7
            p = make_weights(b, M, mode, cyc)

        h = gradient_from_p(p, current)
        nh = float(np.linalg.norm(h))
        improved = False

        if nh > 1e-14 and M > 0.0:
            cand = current.copy()
            cand_score = score
            cand_b = b

            # Multiplicative (simplex-compatible) steps.
            denom = max(M, 1e-18)
            for em in (mult_lr, 0.5 * mult_lr, 0.25 * mult_lr, 0.125 * mult_lr):
                raw = current * np.exp(-em * h / denom)
                c = fix(raw)
                sc, bb2 = score_local(c)
                if sc < cand_score - 1e-14:
                    cand = c
                    cand_score = sc
                    cand_b = bb2

            # Euclidean projected steps with simplex projection.
            for length in (dim_lr, 0.5 * dim_lr, 0.25 * dim_lr, 0.125 * dim_lr):
                raw = current - (length / nh) * h
                c = _proj_simplex(raw, n)
                if symmetric:
                    c = fix(c)
                else:
                    s = float(c.sum())
                    if s > 1e-300 and np.isfinite(s):
                        c = c / s
                    else:
                        c = np.full(n, 1.0 / n, dtype=np.float64)
                sc, bb2 = score_local(c)
                if sc < cand_score - 1e-14:
                    cand = c
                    cand_score = sc
                    cand_b = bb2

            if cand_score < score - 1e-14:
                current = cand
                b = cand_b
                score = cand_score
                if score < best_score - 1e-14:
                    best = current.copy()
                    best_score = score
                improved = True
                stuck = 0
                dim_lr = min(0.55, dim_lr * 1.03)
                mult_lr = min(2.0, mult_lr * 1.02)
            else:
                stuck += 1
                dim_lr = max(0.03, dim_lr * 0.90)
                mult_lr = max(0.05, mult_lr * 0.92)
        else:
            stuck += 1

        # Escape slow local minima by controlled perturbation / basin switching.
        if not improved and stuck > 110:
            sig = min(0.28, 0.035 + 0.00025 * stuck)
            alt = best if (not alt_inits) else _normalize(alt_inits[alt_idx % len(alt_inits)], n)
            if symmetric:
                alt = fix(alt)

            pert_cands = []
            for src in (current, best):
                for sgm in (sig, 0.4 * sig):
                    z = np.log(np.maximum(src, 1e-12)) + rng.normal(0.0, sgm, n)
                    pert_cands.append(fix(np.exp(z)))
                for mix in (0.85, 0.70, 0.50):
                    pert_cands.append(fix(mix * src + (1.0 - mix) * alt))

            bp = None
            bp_score = float("inf")
            bp_b = None
            for c in pert_cands:
                sc, bb2 = score_local(c)
                if sc < bp_score - 1e-15:
                    bp = c
                    bp_score = sc
                    bp_b = bb2

            if bp is not None and (bp_score < score - 1e-14 or
                                   (bp_score < best_score + 1e-8 and rng.random() < 0.25)):
                current = bp
                b = bp_b
                score = bp_score
                if score < best_score - 1e-14:
                    best = current.copy()
                    best_score = score
            else:
                current = best.copy()
                b = conv(current)
                score = best_score

            alt_idx += 1
            stuck = 0
            dim_lr = max(0.10, dim_lr * 0.75)
            mult_lr = max(0.20, mult_lr * 0.75)

        # LP-based direction, the state-of-the-art move.  Used moderately because
        # dense LPs are expensive; reserved for the main length.
        if max_lp > 0 and lp_calls < max_lp and (it % 70 == 0 or stuck > 80):
            rem = float(end_time - time.monotonic())
            if rem > 18.0:
                lp_cand, lp_score, lp_pdual, lp_b = lp_try(
                    current, b, score, M, aggressive=(stuck > 50)
                )
                lp_calls += 1
                if lp_cand is not None and lp_score < score - 1e-14:
                    current = lp_cand
                    b = lp_b
                    score = lp_score
                    if score < best_score - 1e-14:
                        best = current.copy()
                        best_score = score
                    stuck = 0
                if lp_pdual is not None:
                    force_p = lp_pdual
                    force_until = it + 40

    return best, float(best_score)


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 1300.0
    budget_s = max(5.0, budget_s)
    hard_deadline = t0 + budget_s - 12.0
    if hard_deadline < t0 + 20.0:
        hard_deadline = t0 + max(5.0, budget_s - 2.0)

    rng = np.random.default_rng(seed)

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        p = np.where(np.isfinite(p), p, 0.0)
        p = np.maximum(p, 0.0)
        if p.size >= 2:
            parent = p
    except Exception:
        parent = None

    if parent is not None and 8 <= parent.size <= 768:
        n_main = int(parent.size)
        parent_use = parent
    elif parent is not None:
        n_main = 192
        parent_use = _resample(parent, n_main)
    else:
        n_main = 192
        parent_use = None

    total = max(5.0, float(hard_deadline - t0))
    if total > 1000.0:
        main_end = hard_deadline - 340.0
    elif total > 500.0:
        main_end = hard_deadline - 140.0
    else:
        main_end = hard_deadline - 18.0
    if main_end < t0 + 15.0:
        main_end = t0 + max(5.0, budget_s * 0.75)

    # Initial candidate pool for the main length.
    inits = _initial_candidates(n_main, parent_use, rng)
    conv_main = _make_conv(n_main)
    scored = []
    for a in inits:
        aa = _normalize(a, n_main)
        bb = conv_main(aa)
        sc = float(2.0 * n_main * bb.max())
        if np.isfinite(sc):
            scored.append((sc, aa))
    if not scored:
        scored.append((2.0, np.ones(n_main, dtype=np.float64) / n_main))
    scored.sort(key=lambda z: z[0])

    init_main = scored[0][1].copy()
    alt_inits = [a.copy() for _, a in scored[:10]]

    best_array = init_main.copy()
    best_local = float(scored[0][0])

    # Main unsymmetric basin.
    a_main, sc_main = _optimize_length(
        n_main, init_main, alt_inits, min(main_end, hard_deadline - 30.0),
        rng, target=_TARGET, max_lp=60, symmetric=False
    )
    if a_main is not None and a_main.size == n_main and np.isfinite(a_main).all():
        if sc_main < best_local - 1e-12:
            best_array = a_main.copy()
            best_local = float(sc_main)

    # Symmetric basin.  The objective is invariant under reversal, and a large
    # portion of the search time is often better spent in the symmetric subspace.
    if best_local > _TARGET + 1e-9:
        now = time.monotonic()
        remaining = float(hard_deadline - now - 80.0)
        if remaining > 70.0:
            sym_time = min(160.0, remaining - 20.0)
            if sym_time > 30.0:
                alt_sym = []
                for a in alt_inits[:6]:
                    alt_sym.append(_normalize(0.5 * (a + a[::-1]), n_main))
                a_sym, sc_sym = _optimize_length(
                    n_main, _normalize(0.5 * (best_array + best_array[::-1]), n_main),
                    alt_sym, now + sym_time,
                    rng, target=_TARGET, max_lp=20, symmetric=True
                )
                if a_sym is not None and a_sym.size == n_main and np.isfinite(a_sym).all():
                    if sc_sym < best_local - 1e-12:
                        best_array = a_sym.copy()
                        best_local = float(sc_sym)

    # If the target is already met, do not risk otherwise; but we may still use
    # remaining time carefully to push lower.  Also explore a few nearby lengths.
    if best_local > _TARGET - 1e-10:
        extra_ns = []
        P = int(best_array.size)
        for nn in (2 * P, int(math.floor(4.0 * P / 3.0)), 3 * P):
            if 64 <= nn <= 768 and nn not in extra_ns and nn != P:
                extra_ns.append(nn)
        # Prefer larger, but not too large for dense LPs.
        for nn in sorted(extra_ns, reverse=True):
            if len([x for x in extra_ns if x > nn]) < 3:
                pass

        # At most three extra lengths.
        for nn in extra_ns[:3]:
            now = time.monotonic()
            remaining = float(hard_deadline - now - 50.0)
            if remaining < 25.0:
                break
            chunk = min(110.0, remaining - 40.0)
            if chunk < 20.0:
                break
            init_ex = _normalize(_resample(best_array, nn), nn)
            alt_ex = []
            if parent_use is not None:
                alt_ex.append(_normalize(_resample(parent_use, nn), nn))
            alt_ex.append(_normalize(np.ones(nn, dtype=np.float64), nn))
            max_lp_ex = 12 if nn <= 384 else 4
            a_ex, sc_ex = _optimize_length(
                nn, init_ex, alt_ex, now + chunk,
                rng, target=_TARGET, max_lp=max_lp_ex, symmetric=False
            )
            if a_ex is not None and a_ex.size == nn and np.isfinite(a_ex).all():
                if sc_ex < best_local - 1e-12:
                    best_array = a_ex.copy()
                    best_local = float(sc_ex)
            if best_local <= _TARGET + 1e-10:
                break

    # Final exact validation using the supplied evaluator.  Keep several fallbacks
    # so that any numerical discrepancy between local scoring and the evaluator
    # is handled safely.
    exact_candidates = []

    def add_exact(arr):
        try:
            m = int(arr.size)
        except Exception:
            return
        if m < 1 or m > 8192:
            if m > 8192:
                m = 192
                arr = _resample(arr, m)
            else:
                return
        aa = _normalize(arr, m)
        if aa.size >= 1 and np.all(np.isfinite(aa)):
            exact_candidates.append(aa)

    add_exact(best_array)
    if parent_use is not None:
        add_exact(parent_use)
    add_exact(np.ones(n_main, dtype=np.float64) / n_main)
    if parent_use is None:
        add_exact(np.ones(192, dtype=np.float64) / 192.0)

    exact_best_list = None
    exact_best_val = float("inf")

    for arr in exact_candidates:
        lst = [float(x) for x in arr]
        try:
            val = float(evaluate_sequence(lst))
        except Exception:
            val = float("inf")
        if np.isfinite(val) and val < exact_best_val:
            exact_best_val = val
            exact_best_list = lst

    if exact_best_list is None or len(exact_best_list) == 0:
        fallback_n = max(2, n_main)
        exact_best_list = [1.0 / fallback_n for _ in range(fallback_n)]
        try:
            val = float(evaluate_sequence(exact_best_list))
            if not np.isfinite(val):
                exact_best_list = [1.0]
        except Exception:
            exact_best_list = [1.0]

    return exact_best_list
