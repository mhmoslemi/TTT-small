import numpy as np

_LINPROG = None


def _get_linprog():
    global _LINPROG
    if _LINPROG is None:
        try:
            from scipy.optimize import linprog
            _LINPROG = linprog
        except Exception:
            _LINPROG = False

def _fast_eval(seq):
    """Call the authoritative evaluator on an already-valid list of floats."""
    try:
        sc = evaluate_sequence(seq)
    except Exception:
        return math.inf
    try:
        sc = float(sc)
    except Exception:

def _clean_eval(seq):
    """Finite/nonnegative/<=1000 preprocessing, then authoritative evaluation."""
    try:
        vals = seq.tolist() if isinstance(seq, np.ndarray) else list(seq)
    except Exception:
        return math.inf

    if not vals:
        return math.inf

    out = []
    total = 0.0
    for x in vals:
        try:
            y = float(x)
        except Exception:
            return math.inf
        if not math.isfinite(y):
            return math.inf
        if y < 0.0:
            y = 0.0
        elif y > 1000.0:
            y = 1000.0
        out.append(y)
        total += y

    if total < 0.01:
        return math.inf

    return _fast_eval(out)


def _try_shape_scales(shape, best_seq, best_score):
    """Try several global scalings of a nonnegative shape; keep the best valid one."""
    try:
        shp = np.asarray(shape, dtype=np.float64).ravel()
    except Exception:
        return best_seq, best_score

    if shp.size == 0:
        return best_seq, best_score

    shp = np.where(np.isfinite(shp), shp, 0.0)
    shp = np.clip(shp, 0.0, 1000.0)

    s = float(shp.sum())
    if not math.isfinite(s) or s <= 1e-300:
        return best_seq, best_score

    shp = shp / s
    m = float(shp.max())
    if m <= 0.0:
        return best_seq, best_score

    max_scale = 999.999 / m
    raw_scales = [
        0.02, 0.1, 0.5, 1.0, 10.0, 100.0,
        float(shp.size),
        0.5 * float(shp.size),
        2.0 * float(shp.size),
        max_scale,
    ]

    seen = set()
    for sc in raw_scales:
        try:
            sc = float(sc)
        except Exception:
            continue
        if not math.isfinite(sc) or sc < 0.01 or sc > max_scale:
            continue
        key = round(sc, 12)
        if key in seen:
            continue
        seen.add(key)

        arr = shp * sc
        arr_score = _clean_eval(arr.tolist())
        if arr_score < best_score - 1e-15:
            best_score = arr_score
            best_seq = arr.tolist()

    return best_seq, best_score


def _local_pass(cur, cur_score, stop, rng):
    """One local-search pass: coordinate multipliers, pairs, and small Gaussian repair."""
    n = int(cur.size)
    if n == 0:
        return cur, cur_score, False

    improved = False

    order = np.arange(n) if n <= 600 else rng.choice(n, min(n, 1200), replace=False)

    base_mults = (0.0, 1.0 / 3.0, 0.5, 0.75, 1.25, 1.5, 2.0, 3.0)
    fine_mults = (0.8, 0.9, 1.1, 1.2, 1.4)
    abs_scale = 1.0 / n
    abs_vals = (0.0, 1e-4 * abs_scale, 1e-3 * abs_scale, 1e-2 * abs_scale, 5e-2 * abs_scale)

    for i in order:
        if time.monotonic() > stop - 12.0:
            break

        orig = float(cur[i])
        best_v = orig
        best_s = cur_score

        vals = []
        if orig > 1e-14:
            for mf in base_mults:
                vals.append(0.0 if mf == 0.0 else orig * mf)
            for mf in fine_mults:
                vals.append(orig * mf)
        else:
            vals.append(0.0)
        vals.extend(abs_vals)

        seen = set()
        for v in vals:
            v = float(v)
            if v < 0.0:
                v = 0.0
            if abs(v - orig) <= 1e-18:
                continue
            key = round(v, 18)
            if key in seen:
                continue
            seen.add(key)

            cur[i] = v
            s = float(cur.sum())
            if s <= 1e-300:
                continue

            trial = cur / s
            sc = _fast_eval(trial.tolist())
            if sc < best_s - 1e-15:
                best_s = sc
                best_v = v

        if best_v != orig:
            cur[i] = best_v
            s = float(cur.sum())
            if s > 1e-300:
                cur = cur / s
                cur_score = best_s
                improved = True
            else:
                cur[i] = orig
        else:
            cur[i] = orig

    pair_iters = 30 if n <= 600 else 10
    for _ in range(pair_iters):
        if time.monotonic() > stop - 12.0:
            break

        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue

        oi = float(cur[i])
        oj = float(cur[j])
        best_vi = oi
        best_vj = oj
        best_s = cur_score

        if oi > 1e-14:
            vi_opts = [0.0, oi * 0.25, oi * 0.5, oi * 1.5, oi * 2.0]
        else:
            vi_opts = [0.0, 1e-3 / n, 1e-2 / n]

        if oj > 1e-14:
            vj_opts = [0.0, oj * 0.25, oj * 0.5, oj * 1.5, oj * 2.0]
        else:
            vj_opts = [0.0, 1e-3 / n, 1e-2 / n]

        for vi in vi_opts:
            for vj in vj_opts:
                if vi == oi and vj == oj:
                    continue
                if abs(vi - oi) <= 1e-18 and abs(vj - oj) <= 1e-18:
                    continue

                cur[i] = vi
                cur[j] = vj
                s = float(cur.sum())
                if s <= 1e-300:
                    continue

                trial = cur / s
                sc = _fast_eval(trial.tolist())
                if sc < best_s - 1e-15:
                    best_s = sc
                    best_vi = vi
                    best_vj = vj

        if best_s < cur_score - 1e-15:
            cur[i] = best_vi
            cur[j] = best_vj
            s = float(cur.sum())
            if s > 1e-300:
                cur = cur / s
                cur_score = best_s
                improved = True
            else:
                cur[i], cur[j] = oi, oj
        else:
            cur[i], cur[j] = oi, oj

    for _ in range(5):
        if time.monotonic() > stop - 12.0:
            break
        sigma = float(rng.choice((0.03, 0.07, 0.12))) * max(float(cur.mean()), 1e-5)
        trial = np.clip(cur + rng.normal(0.0, sigma, n), 0.0, None)
        s = float(trial.sum())
        if s <= 1e-300:
            continue
        trial = trial / s
        sc = _fast_eval(trial.tolist())
        if sc < cur_score - 1e-15:
            cur = trial
            cur_score = sc
            improved = True

    return cur, cur_score, improved


def _conv_full(a):
    a = np.asarray(a, dtype=np.float64)
    return np.convolve(a, a, mode="full")


def _lp_direction(f, exclude_center=False):
    """
    One-step auxiliary direction from:
        maximize sum(g)
        s.t. f*g <= M, g >= 0
    where M is the current relevant maximum convolution level.
    """
    linprog = _get_linprog()
    if linprog is None:
        return None

    n = int(f.size)
    if n == 0 or n > 800:
        return None
    if exclude_center and n < 2:
        return None

    f = np.asarray(f, dtype=np.float64).ravel()
    f = np.where(np.isfinite(f), f, 0.0)
    f = np.clip(f, 0.0, None)
    s = float(f.sum())
    if s <= 1e-15:
        return None
    f = f / s

    b = _conv_full(f)
    if b.size == 0 or not np.all(np.isfinite(b)):
        return None

    if exclude_center:
        idx = np.concatenate((np.arange(0, n - 1), np.arange(n, 2 * n - 1)))
        M = float(np.max(b[idx])) if idx.size else float(b.max())
    else:
        idx = None
        M = float(b.max())

    if not math.isfinite(M) or M <= 1e-15:
        return None

    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f

    if idx is not None:
        A = A[idx, :]

    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(A.shape[0], M, dtype=np.float64)

    try:
        res = linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=[(0.0, None)] * n,
            method="highs",
            options={"presolve": True},
        )
    except Exception:
        return None

    if res is None or getattr(res, "status", None) != 0:
        return None

    x = getattr(res, "x", None)
    if x is None:
        return None

    g = np.asarray(x, dtype=np.float64).ravel()
    if g.size != n or not np.all(np.isfinite(g)):
        return None

    g = np.clip(g, 0.0, None)
    gs = float(g.sum())
    if gs <= 1e-15:
        return None

    return g / gs


def _lp_improve(f0, stop, seed=0, exclude_center=False):
    """Use LP-generated directions with exact-score scalar line search."""
    try:
        f = np.asarray(f0, dtype=np.float64).ravel().copy()
    except Exception:
        return None, math.inf

    if f.size == 0:
        return None, math.inf

    f = np.where(np.isfinite(f), f, 0.0)
    f = np.clip(f, 0.0, None)
    s = float(f.sum())
    if s <= 1e-15:
        return None, math.inf
    f = f / s

    cur_score = _fast_eval(f.tolist())
    if not math.isfinite(cur_score):
        return None, math.inf

    best_score = cur_score
    best_f = f.copy()
    n = int(f.size)
    rng = np.random.default_rng(seed)

    stall = 0
    max_stall = 50

    while time.monotonic() < stop - 15.0 and stall < max_stall:
        g = _lp_direction(f, exclude_center=exclude_center)

        if g is None:
            noise = np.clip(f + rng.normal(0.0, 0.08 * max(float(f.mean()), 1e-4), n), 0.0, None)
            ns = float(noise.sum())
            if ns > 1e-15:
                noise = noise / ns
                sc = _fast_eval(noise.tolist())
                if math.isfinite(sc) and sc <= cur_score + 0.05:
                    f = noise
                    cur_score = sc
            stall += 1
            continue

        dist = float(np.linalg.norm(g - f))
        if dist < 1e-8:
            noise = np.clip(f + rng.normal(0.0, 0.10 * max(float(f.mean()), 1e-4), n), 0.0, None)
            ns = float(noise.sum())
            if ns > 1e-15:
                noise = noise / ns
                sc = _fast_eval(noise.tolist())
                if math.isfinite(sc) and sc <= cur_score + 0.05:
                    f = noise
                    cur_score = sc
            stall += 1
            continue

        best_cand = None
        best_cand_score = cur_score

        for t in (0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0):
            cand = (1.0 - t) * f + t * g
            cand = np.clip(cand, 0.0, None)
            cs = float(cand.sum())
            if cs <= 1e-15:
                continue
            cand = cand / cs
            sc = _fast_eval(cand.tolist())
            if math.isfinite(sc) and sc < best_cand_score - 1e-15:
                best_cand_score = sc
                best_cand = cand

        if best_cand is not None:
            f = best_cand
            cur_score = best_cand_score
            if cur_score < best_score - 1e-15:
                best_score = cur_score
                best_f = f.copy()
            stall = 0
            continue

        accepted = False
        for t in (0.05, 0.10, 0.20):
            cand = (1.0 - t) * f + t * g
            cand = np.clip(cand, 0.0, None)
            cs = float(cand.sum())
            if cs <= 1e-15:
                continue
            cand = cand / cs
            sc = _fast_eval(cand.tolist())
            if math.isfinite(sc) and sc < cur_score - 1e-15:
                f = cand
                cur_score = sc
                if cur_score < best_score - 1e-15:
                    best_score = cur_score
                    best_f = f.copy()
                stall = 0
                accepted = True
                break

        if not accepted:
            stall += 1

    return best_f, best_score


def _improve(start_seq, stop, seed=0):
    """
    Black-box heuristic. It is only useful if the runtime evaluator reports a
    score below the nonnegative Cauchy-Schwarz bound of 2.
    """
    try:
        arr = np.asarray(start_seq, dtype=np.float64).ravel()
    except Exception:
        return start_seq, math.inf

    if arr.size == 0:
        return start_seq, math.inf

    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if s < 0.01:
        arr = np.ones(max(1, int(arr.size)), dtype=np.float64)
        s = 1.0

    best_seq = list(start_seq)
    best_score = _clean_eval(best_seq)

    shape = arr / s
    cur = shape.copy()
    cur_score = _fast_eval(cur.tolist())

    best_shape = cur.copy()
    best_shape_score = cur_score

    if math.isfinite(cur_score) and cur_score < best_score - 1e-15:
        best_score = cur_score
        best_seq = cur.tolist()

    if cur.size <= 1500 and time.monotonic() < stop - 60.0:
        remaining = stop - time.monotonic()
        local_stop = min(stop - 60.0, time.monotonic() + min(300.0, max(20.0, remaining * 0.30)))
        rng = np.random.default_rng(seed)
        no_imp = 0
        passes = 0

        while time.monotonic() < local_stop and no_imp < 50 and passes < 400:
            cur, cur_score, imp = _local_pass(cur, cur_score, local_stop, rng)
            passes += 1

            if imp:
                if math.isfinite(cur_score):
                    if cur_score < best_shape_score - 1e-15:
                        best_shape = cur.copy()
                        best_shape_score = cur_score
                    if cur_score < best_score - 1e-15:
                        best_score = cur_score
                        best_seq = cur.tolist()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp in (5, 15, 25, 35):
                    sigma = 0.05 if no_imp == 5 else (0.10 if no_imp <= 15 else 0.20)
                    trial = np.clip(
                        best_shape + rng.normal(
                            0.0,
                            sigma * max(float(best_shape.mean()), 1e-5),
                            best_shape.size,
                        ),
                        0.0,
                        None,
                    )
                    ts = float(trial.sum())
                    if ts > 1e-15:
                        trial = trial / ts
                        sc = _fast_eval(trial.tolist())
                        if math.isfinite(sc) and sc < cur_score + 0.05:
                            cur = trial
                            cur_score = sc
                    no_imp = 0
                elif no_imp >= 50:
                    break

    if cur.size <= 800 and time.monotonic() < stop - 60.0:
        for excl, offset in ((False, 1000), (True, 2000)):
            if time.monotonic() > stop - 60.0:
                break
            remaining = stop - time.monotonic()
            lp_stop = min(
                stop - 60.0,
                time.monotonic() + min(180.0, max(30.0, remaining * 0.35)),
            )
            if time.monotonic() >= lp_stop:
                continue

            try:
                f, fs = _lp_improve(best_shape, lp_stop, int(seed) + offset, exclude_center=excl)
            except Exception:
                f, fs = None, math.inf

            if f is not None and math.isfinite(fs):
                if fs < best_score - 1e-15:
                    best_score = fs
                    best_seq = list(f)
                sf = float(np.asarray(f).sum())
                if sf > 1e-15 and fs < best_shape_score - 1e-15:
                    best_shape = np.asarray(f, dtype=np.float64) / sf
                    best_shape_score = fs

    for shp in (best_shape, cur, shape):
        try:
            best_seq, best_score = _try_shape_scales(shp, best_seq, best_score)
        except Exception:
            pass

    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    """
    For the authoritative evaluator, all coefficients are clipped to be
    nonnegative. Let a be any nonnegative vector of length n with sum S.

        score = 2 n max_k (a*a)_k / S^2.

    The central convolution term is c_0 = sum_i a_i^2, and Cauchy-Schwarz gives
        c_0 >= S^2 / n.
    Hence
        score >= 2 n c_0 / S^2 >= 2.

    A constant vector attains exactly 2, so 2 is the exact minimum.  Return an
    optimal constant vector.  The parent is still consulted: if the supplied
    runtime evaluator reports a strictly smaller score (e.g. due to a different
    active scoring behavior), a bounded heuristic searches for better candidates
    and never returns anything worse than the best verified candidate.
    """
    t0 = time.monotonic()
    try:
        budget = max(10.0, float(budget_s))
    except Exception:
        budget = 1300.0

    hard_stop = t0 + budget - 15.0

    best_seq = [1.0]
    best_score = _clean_eval(best_seq)

    try:
        h = height_sequence_1
        arr = np.asarray(h, dtype=np.float64).ravel()
        if 0 < arr.size <= 5000:
            arr = np.where(np.isfinite(arr), arr, 0.0)
            arr = np.clip(arr, 0.0, 1000.0)
            if float(arr.sum()) >= 0.01:
                cand = [float(x) for x in arr]
                sc = _clean_eval(cand)
                if math.isfinite(sc) and sc < best_score - 1e-15:
                    best_seq = cand
                    best_score = sc
    except Exception:
        pass

    opt_stop = min(hard_stop - 10.0, t0 + budget - 25.0)
    if (
        math.isfinite(best_score)
        and best_score < 1.99
        and time.monotonic() < opt_stop - 30.0
    ):
        try:
            best_seq, best_score = _improve(best_seq, opt_stop, int(seed))
        except Exception:
            pass

    try:
        best_seq, best_score = _try_shape_scales(
            np.asarray(best_seq, dtype=np.float64),
            best_seq,
            best_score,
        )
    except Exception:
        pass

    try:
        fs = _clean_eval(best_seq)
        if not math.isfinite(fs):
            best_seq = [1.0]
            best_score = _clean_eval(best_seq)
    except Exception:
        best_seq = [1.0]
        try:
            best_score = _clean_eval(best_seq)
        except Exception:
            best_score = math.inf

    return [float(x) for x in best_seq]
