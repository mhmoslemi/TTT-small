import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len

def _fft_conv(a):
    n = len(a)
    size = next_fast_len(2 * n - 1)
    A = rfft(a, size)
    c = irfft(A * A, size)[:2 * n - 1]
    return c

def _exact_score(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = _fft_conv(a)
    return float(2.0 * n * c.max() / (s * s))

def _smooth_obj_grad(a, tau):
    n = len(a)
    c = _fft_conv(a)
    c_max = c.max()
    exp_c = np.exp(tau * (c - c_max))
    S = exp_c.sum()
    obj = c_max + math.log(S) / tau
    p = exp_c / S
    corr = np.correlate(p, a, mode='full')
    grad = corr[:n].copy()
    return obj, grad

def _subgradient_direction(a, c):
    n = len(a)
    k_star = int(np.argmax(c))
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = a[j]
    return grad

def _normalize_nonneg(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def _build_lp_matrix(a):
    n = len(a)
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    A = np.zeros((2 * n - 1, n))
    valid = (diff >= 0) & (diff < n)
    A[valid] = a[diff[valid].astype(int)]
    return A

    best_exact = _exact_score(a)

    for tau in tau_list:

            if time.monotonic() > deadline - 30:

            obj, grad = _smooth_obj_grad(a, tau)

            # Nesterov momentum
            v = beta * v - lr * grad
            a_new = a + v

def _lp_refine(a, deadline, max_iters=30):
    from scipy.optimize import linprog
    n = len(a)
    best_a = a.copy()
    best_exact = _exact_score(a)
    a = a.copy()
    A_mat = _build_lp_matrix(a)

            for i in idx:
                cand[i] *= rng.uniform(0.5, 1.5)
            cand = cand + rng.normal(0, sigma * 0.3, n)
        else:
            # Palindrome push
            cand = 0.5 * (a + a[::-1]) + rng.normal(0, sigma * 0.5, n)

        cand = _normalize_nonneg(cand)
        c_try = _fft_conv(cand)
        m_try = float(c_try.max())

            if exact < best_exact:
                best_exact = exact
                best_a = a.copy()
            sigma = max(0.0005, sigma * 0.997)
        else:

    return best_a, best_exact
