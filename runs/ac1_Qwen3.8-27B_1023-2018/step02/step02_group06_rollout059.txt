import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
        if p.size == 0 or not np.isfinite(p).all():
            raise ValueError
        return np.clip(p, 0.0, 1000.0)
    except Exception:
        return np.ones(192, dtype=np.float64)

def _conv(a):
    n = len(a)
    fsize = next_fast_len(4 * n)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    c[~np.isfinite(c)] = 0.0
    return c

def _score_fast(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    b = _conv(a)
    return 2.0 * n * b.max() / (s * s)

def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def _build_A(a):
    n = len(a)
    k_idx = np.arange(2 * n - 1, dtype=np.int64)[:, None]
    j_idx = np.arange(n, dtype=np.int64)[None, :]
    diff = k_idx - j_idx
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    valid = (diff >= 0) & (diff < n)
    A[valid] = a[diff[valid]]
    return A

def _lp_direction(a, M, n):
    c_obj = -np.ones(n)
    A_ub = _build_A(a)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                  bounds=[(0.0, None)] * n, method='highs',
                  options={'presolve': True, 'time_limit': 5.0})
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        s = g.sum()
        if s > 1e-15:
            return g / s
    return None

def _random_init(n, rng, variant):
    i = np.arange(n, dtype=np.float64)
    v = int(variant) % 10
    if v == 0:
        f1 = rng.integers(1, n // 2 + 1)
        f2 = rng.integers(1, n // 2 + 1)
        a = 1.0 + 0.5 * np.sin(2 * np.pi * f1 * i / n + rng.uniform(0, 6.28)) + \
            0.3 * np.cos(2 * np.pi * f2 * i / n + rng.uniform(0, 6.28))
    elif v == 1:
        a = np.exp(rng.normal(0, 1.0, n))
    elif v == 2:
        c1 = rng.uniform(0.2, 0.8) * n
        c2 = rng.uniform(0.2, 0.8) * n
        w = rng.uniform(n / 6, n / 3)
        a = np.exp(-(i - c1) ** 2 / (2 * w ** 2)) + 0.5 * np.exp(-(i - c2) ** 2 / (2 * w ** 2))
    elif v == 3:
        a = np.abs(rng.normal(0, 1, n)) + 0.1
    elif v == 4:
        a = np.ones(n)
        for _ in range(3):
            c = rng.uniform(0, n)
            w = rng.uniform(n / 10, n / 4)
            a += rng.uniform(0.3, 1.0) * np.exp(-(i - c) ** 2 / (2 * w ** 2))
    elif v == 5:
        # Linear ramp
        a = 1.0 + rng.uniform(0.5, 2.0) * (i / n - 0.5)
    elif v == 6:
        # Two-block
        split = rng.integers(n // 4, 3 * n // 4)
        a = np.ones(n)
        a[:split] *= rng.uniform(1.0, 3.0)
        a[split:] *= rng.uniform(0.2, 1.0)
    elif v == 7:
        # Periodic
        p = rng.integers(3, max(4, n // 10))
        a = 1.0 + 0.6 * np.sin(2 * np.pi * i / p + rng.uniform(0, 6.28))
    elif v == 8:
        # Sparse peaks
        a = np.zeros(n)
        npeaks = rng.integers(max(3, n // 20), max(5, n // 8))
        peaks = rng.integers(0, n, npeaks)
        for p in peaks:
            w = rng.integers(1, max(2, n // 15))
            lo = max(0, p - w)
            hi = min(n, p + w + 1)
            a[lo:hi] += rng.uniform(0.5, 2.0) * np.exp(-(i[lo:hi] - p) ** 2 / (2 * w * w))
        a += 0.05
    else:
        a = rng.uniform(0.2, 2.0, n)
    a = np.maximum(a, 0.01)
    return a / a.sum()

def _lp_iterate(a, n, deadline, rng):
    best_a = _normalize(a.copy())
    best_score = _score_fast(best_a)
    a = best_a.copy()
    no_improve = 0
    step_size = 0.3

    for it in range(100000):
        if time.monotonic() > deadline:
            break

        b = _conv(a)
        M = b.max()
        cur_score = 2.0 * n * M

        if cur_score < best_score * (1 - 1e-11):
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            step_size = min(0.5, step_size * 1.02)
        else:
            no_improve += 1
            if no_improve % 50 == 0:
                step_size = max(0.05, step_size * 0.9)

        if no_improve > 300:
            a = _normalize(a + rng.normal(0, 0.08, n))
            no_improve = 0
            continue

        g = _lp_direction(a, M, n)
        if g is None:
            k_star = int(np.argmax(b))
            grad = np.zeros(n)
            for m in range(n):
                j = k_star - m
                if 0 <= j < n:
                    grad[m] = 2.0 * a[j]
            gn = np.linalg.norm(grad)
            if gn < 1e-15:
                break
            a = _normalize(a - 0.3 * grad / gn)
            continue

        cos_sim = float(np.dot(a, g) / (np.linalg.norm(a) * np.linalg.norm(g) + 1e-30))
        if cos_sim > 0.9999:
            no_improve += 10
            a = _normalize(a + rng.normal(0, 0.05, n))
            continue

        # Line search
        best_t = 0.0
        best_mb = M
        for t in [0.05, 0.15, 0.3, 0.5, 0.7, 1.0]:
            cand = _normalize((1 - t) * a + t * g)
            mb = _conv(cand).max()
            if mb < best_mb * (1 - 1e-12):
                best_mb = mb
                best_t = t

        if best_t > 0:
            a = _normalize((1 - best_t) * a + best_t * g)
        else:
            a = _normalize(0.97 * a + 0.03 * g)

    return best_a, best_score

def _local_search(a, n, deadline, rng):
    best_a = _normalize(a.copy())
    best_score = _score_fast(best_a)
    a = best_a.copy()
    sigma = 0.03
    no_improve = 0
    max_no_improve = 100

    while time.monotonic() < deadline:
        r = rng.random()

        if r < 0.30:
            # Global noise
            cand = a + rng.normal(0, sigma, n)
        elif r < 0.50:
            # Coordinate perturbation
            cand = a.copy()
            ncoord = rng.integers(1, max(2, n // 30))
            idxs = rng.integers(0, n, ncoord)
            cand[idxs] *= rng.uniform(0.3, 3.0, ncoord)
        elif r < 0.65:
            # Block rescale
            cand = a.copy()
            lo = int(rng.integers(0, max(1, n - 5)))
            width = int(rng.integers(max(2, n // 40), max(3, n // 6)))
            hi = min(n, lo + width)
            cand[lo:hi] *= rng.uniform(0.4, 2.5)
        elif r < 0.80:
            # Neighbor swap
            cand = a.copy()
            i = int(rng.integers(1, n - 1))
            eps = sigma * rng.uniform(0.5, 2.0)
            cand[i] += eps
            cand[i - 1] -= eps * 0.5
            cand[i + 1] -= eps * 0.5
        else:
            # Palindrome blend + noise
            cand = 0.7 * a + 0.3 * a[::-1]
            cand += rng.normal(0, sigma * 0.5, n)

        cand = _normalize(cand)
        sc = _score_fast(cand)

        if sc < best_score * (1 - 1e-12):
            best_score = sc
            best_a = cand.copy()
            a = cand.copy()
            no_improve = 0
            sigma = max(0.003, sigma * 0.97)
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                sigma = min(0.5, sigma * 1.8)
                no_improve = 0
                if sigma > 0.4:
                    no_improve = 0
                    max_no_improve = int(max(100, max_no_improve * 1.5))

    return best_a, best_score
