    b = np.convolve(a, a)
    M = float(b.max())
    if M < 1e-15:
        return None
    
    # Build constraint matrix A (2n-1) x n
    # A[k, i] = a[i] if 0 <= k-i < n, else 0
    # i.e., A[k, i] = a[min(i, k)] ... no
    # A[k, i] = a[k-i] if 0 <= k-i < n, else 0
    # Wait, the constraint is (a*g)_k <= M
    # (a*g)_k = sum_j a[j] * g[k-j]
    # So the coefficient of g[i] in constraint k is a[k-i] if 0 <= k-i < n
    # i.e., A[k, i] = a[k-i] if 0 <= k-i < n, else 0
    
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A[k, i] = a[j]
    
    # Actually, let me think about this differently.
    # (a * g)_k = sum_i a[k-i] * g[i] (for valid k-i)
    # So row k of A has: A[k, i] = a[k-i] if 0 <= k-i < n
    # For k=0: A[0,i] = a[-i] -> only i=0 gives a[0]
    # For k=n-1: A[n-1,i] = a[n-1-i] for all i -> reversed a
    # For k=2n-2: A[2n-2,i] = a[2n-2-i] -> only i=n-1 gives a[n-1]
    
    # This is a banded-toeplitz-like matrix
    # Let me build it more efficiently
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        lo = max(0, k - (n-1))
        hi = min(k, n-1)
        for i in range(lo, hi+1):
            A[k, i] = a[k-i]
    
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n,
                    method='highs', options={'time_limit': time_limit})
        if r.success and r.x is not None:
            g = np.clip(r.x, 0, None)
            sg = g.sum()
            if sg > 1e-10:
                return g / sg
    except Exception:
        pass

def _norm(a):

def _build_A(a):

    A = np.zeros((2 * n - 1, n), dtype=np.float64)

def _lp_solve(a, time_limit=8.0):
