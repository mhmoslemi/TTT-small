        lengths = sorted(set([n_p - 2, n_p + 2, n_p - 4, n_p + 4, n_p - 6, n_p + 6,
                               n_p - 8, n_p + 8]))
        for L in lengths:
            if time.monotonic() >= deadline - 15:
                break
            if L < 16 or L > 512:
                continue
            a_try = interp_to_len(a_work, L)
            update_best(a_try)
            # Quick LP at this length
            if time.monotonic() < deadline - 12:
                n_t = L
                try:
                    A_t = build_A(a_try)
                    ac_t = np.convolve(a_try, a_try)
                    M_t = float(ac_t.max())
                    if M_t > 0:
                        res = linprog(-np.ones(n_t), A_ub=A_t,
                                     b_ub=M_t * np.ones(2*n_t-1),
                                     bounds=[(0.0, None)]*n_t, method='highs',
                                     options={'time_limit': 3.0})
                        if res is not None and res.x is not None:
                            b_t = np.clip(res.x, 0, None)
                            if b_t.sum() > 1e-15:
                                b_t /= b_t.sum()
                                for t in (0.3, 0.5, 0.7):
                                    cand = (1-t)*a_try + t*b_t
                                    cand = np.clip(cand, 0, None)
                                    cs = cand.sum()
                                    if cs > 1e-15:
                                        cand /= cs
                                        update_best(cand)
                except Exception:
                    pass

    # Phase 5: Final random restarts at best length
    if time.monotonic() < deadline - 15:
        n_b = best_arr.size if best_arr is not None else n_p
        n_restarts = 5
        for r in range(n_restarts):
            if time.monotonic() >= deadline - 10:
                break
            if r == 0:
                a_r = best_arr / best_arr.sum()
            else:
                a_r = a_work * np.exp(0.2 * rng.standard_normal(n_lp))
                a_r = np.clip(a_r, 0, None)
                if a_r.sum() < 1e-15:
                    continue
                a_r /= a_r.sum()
            # Quick LP
            if time.monotonic() < deadline - 8:
                try:
                    A_r = build_A(a_r)
                    ac_r = np.convolve(a_r, a_r)
                    M_r = float(ac_r.max())
                    if M_r > 0:
                        res = linprog(-np.ones(n_lp), A_ub=A_r,
                                     b_ub=M_r * np.ones(2*n_lp-1),
                                     bounds=[(0.0, None)]*n_lp, method='highs',
                                     options={'time_limit': 3.0})
                        if res is not None and res.x is not None:
                            b_r = np.clip(res.x, 0, None)
                            if b_r.sum() > 1e-15:
                                b_r /= b_r.sum()
                                for t in (0.4, 0.6, 0.8, 1.0):
                                    cand = (1-t)*a_r + t*b_r
                                    cand = np.clip(cand, 0, None)
                                    cs = cand.sum()
                                    if cs > 1e-15:
                                        cand /= cs
                                        update_best(cand)
                except Exception:
                    pass

    # Final validation
    if best_arr is None:
        best_arr = np.ones(96, dtype=np.float64)
        update_best(best_arr)

    final = np.clip(best_arr, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones_like(final)
    update_best(final)

    return [float(x) for x in best_arr]
