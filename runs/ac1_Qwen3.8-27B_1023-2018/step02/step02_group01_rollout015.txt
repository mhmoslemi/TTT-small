    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s <= 0:
        return np.ones_like(a) / a.size
    return a / s


def _lp_optimize(a0, stop_t, max_it=20000):
    """LP best-response iteration with line search."""
    a = _normalize(a0).copy()
    n = a.size
    best_a = a.copy()
    best_M = float(_conv(a).max())
    stuck = 0

    for it in range(max_it):
        if time.monotonic() > stop_t:
            break

        b = _conv(a)
        M = float(b.max())

        if M < best_M * (1.0 - 1e-14):
            best_M = M
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        if stuck > 300:
            break

        # Try both LP formulations
        g, MP = _lp_minimax(a)
        if g is None:
            g = _lp_original(a, M)
            if g is None:
                break
            g = g / g.sum()

        # Line search
        improved = False
        for t in (1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1, 0.05, 0.01):
            cand = _normalize((1.0 - t) * a + t * g)
            cM = float(_conv(cand).max())
            if cM < M * (1.0 - 1e-14):
                a = cand
                improved = True
                break

        if not improved:
            stuck += 10

    return best_a


def _mirror_descent(a0, stop_t, max_it=100000):
    """Exponential weighting / mirror descent."""
    a = _normalize(a0).copy()
    n = a.size
    best_a = a.copy()
    best_M = float(_conv(a).max())

    for it in range(max_it):
        if time.monotonic() > stop_t:
            break

        b = _conv(a)
        M = float(b.max())

        if M < best_M * (1.0 - 1e-14):
            best_M = M
            best_a = a.copy()

        # Soft-max weights
        eps = max(0.005 * M, 1e-10)
        w = np.exp((b - M) / eps)
        w /= w.sum()

        # Gradient: grad_j = 2 * sum_l a_l * w_{j+l}
        grad = 2.0 * np.correlate(w, a, 'valid')

        # Multiplicative update
        eta = 0.5 / n
        a_new = a * np.exp(-eta * grad)
