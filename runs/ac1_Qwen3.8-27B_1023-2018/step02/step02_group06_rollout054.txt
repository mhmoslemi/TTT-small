            delta = rng.normal(0, sigma) * max(a[i], a[j])
            cand = a.copy()
            cand[i] += delta
            cand[j] -= delta
        elif r < 0.8:
            # Block modification
            lo = int(rng.integers(0, max(1, n//4)))
            w = int(rng.integers(max(1, n//20), n//4))
            hi = min(n, lo + w)
            cand = a.copy()
            cand[lo:hi] *= np.exp(rng.normal(0, 0.3, hi-lo))
        else:
            # Palindromic push
            cand = 0.5 * (a + a[::-1]) + rng.normal(0, sigma*0.5, n)
        
        cand = _normalize(cand)
        b = _fft_conv(cand)
        mb = b.max() / (cand.sum()**2)
        
        if mb < best_mb * (1 - 1e-12):
            best_mb = mb
            best_a = cand.copy()
            a = cand.copy()
            no_improve = 0
            sigma = max(0.001, sigma * 0.98)
        else:
            no_improve += 1
            if no_improve > 100:
                sigma = min(0.2, sigma * 1.5)
                no_improve = 0
            if no_improve > 500:
                # Big jump
                cand = a + rng.normal(0, 0.3, n)
                cand = _normalize(cand)
                mb = _fft_conv(cand).max() / (cand.sum()**2)
                if mb < best_mb:
                    best_mb = mb
                    best_a = cand.copy()
                    a = cand.copy()
                no_improve = 0
    
    return best_a

def _lp_optimize_lite(a_init, deadline, n_lp_steps=20):
    """LP-based optimization (simplified version of previous program)."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_mb = _fft_conv(a).max()
    
    for step in range(n_lp_steps):
        if time.monotonic() > deadline:
            break
        
        b = _fft_conv(a)
        M = float(b.max())
        
        # Build constraint matrix A: A[k,m] = a[k-m] if 0<=k-m<n
        # Constraint: A @ g <= M for all k
        # But we want max sum(g) s.t. A@G <= M, g >= 0
        
        # For the central lag, A[k,m] = a[k-m]
        # This matrix has a banded structure
        
        # Solve: max sum(g) s.t. sum_j a_{k-j} * g_j <= M for each k, g >= 0
        # This is a LP with 2n-1 constraints and n variables
        
        k_idx = np.arange(2*n-1)
        # A[k, m] = a[k-m] if 0 <= k-m < n else 0
        A = np.zeros((2*n-1, n))
        for m in range(n):
            lo = max(0, m)
            hi = min(2*n-1, m + n)
            A[lo:hi, m] = a[m - (np.arange(lo, hi) - m)]
        
        # Hmm, let me redo this. A[k,m] = a[k-m] if 0<=k-m<n
        # So for fixed m, k must be in [m, m+n-1] ∩ [0, 2n-2]
        A = np.zeros((2*n-1, n))
        for m in range(n):
            ks = np.arange(max(0, m), min(2*n-1, m+n))
            j_vals = ks - m  # = a[k-m] = a[j_vals]
            A[ks, m] = a[j_vals]
        
        g = cp.Variable(n, nonneg=True)
        constraints = [A @ g <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        
        solved = False
        for solver in [cp.ECOS, cp.SCS, cp.CBC, cp.GLPK]:
            try:
                prob.solve(solver=solver, verbose=False)
                if prob.status in ['optimal', 'optimal_inaccurate']:
                    gv = g.value
                    if gv is not None and np.isfinite(gv).all():
                        gv = np.maximum(gv, 0)
                        s = gv.sum()
                        if s > 1e-15:
                            g_norm = gv / s
                            solved = True
                            break
            except Exception:
                continue
        
        if not solved:
            break
        
        # Line search
        best_t = 0
        best_new_mb = M
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_new = _normalize((1-t)*a + t*g_norm)
            mb = _fft_conv(a_new).max()
            if mb < best_new_mb:
                best_new_mb = mb
                best_t = t
        
        if best_t > 0:
            a = _normalize((1-best_t)*a + best_t*g_norm)
        
        if M < best_mb * (1 - 1e-10):
            best_mb = M
            best_a = a.copy()
    
    return best_a
