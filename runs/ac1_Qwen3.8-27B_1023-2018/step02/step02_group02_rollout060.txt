            for frac in (0.85, 1.15, 0.9, 1.1, 0.95, 1.05):
                t2 = best_t * frac
                if 0 < t2 < 1.0:
                    h = (1.0 - t2) * f_curr + t2 * g
                    m_h = float(conv(h).max())
                    if m_h < best_m - 1e-15:
                        best_m = m_h
                        best_t = t2
        return best_t, best_m

    # === PHASE 1: LP-based descent ===
    B = build_B(f)
    M = float(conv(f).max())
    stagnant = 0
    it_count = 0

    while time.monotonic() < deadline - 80.0:
        it_count += 1
        g = do_lp(f, B, M)
        if g is None:
            stagnant += 1
            if stagnant > 20:
                break
            # Perturb and retry
            rng_p = np.random.default_rng(seed * 31 + stagnant)
            f = f + rng_p.normal(0, 0.02, n)
            f = np.maximum(f, 0.0)
            ss = f.sum()
            if ss < 1e-12:
                break
            f /= ss
            B = build_B(f)
            M = float(conv(f).max())
            continue

        diff = float(np.abs(g - f).max())
        if diff < 1e-9:
            stagnant += 1
            if stagnant > 30:
                break
            # Small perturbation to escape
            rng_p = np.random.default_rng(seed * 17 + stagnant)
            f = f + rng_p.normal(0, 0.015, n)
            f = np.maximum(f, 0.0)
            f /= f.sum()
            B = build_B(f)
            M = float(conv(f).max())
            continue

        best_t, best_m = line_search(f, g, M)
        if best_t <= 0:
            stagnant += 1
            if stagnant > 30:
                break
            continue

        f = (1.0 - best_t) * f + best_t * g
        M = best_m
        B = build_B(f)
        stagnant = 0

        sc = 2.0 * n * M
        if sc < best_score - 1e-13:
            best_score = sc
            best_f = f.copy()

        # Periodic large perturbation to explore
        if it_count % 500 == 0 and time.monotonic() < deadline - 100.0:
            rng_p = np.random.default_rng(seed * 7 + it_count)
            f_pert = f + rng_p.normal(0, 0.04, n)
            f_pert = np.maximum(f_pert, 0.0)
            f_pert /= f_pert.sum()
            sc_pert = 2.0 * n * float(conv(f_pert).max())
            if sc_pert < 2.0 * n * M * 1.5:  # Not too bad
                f = f_pert
                M = float(conv(f).max())
                B = build_B(f)

    # === PHASE 2: Smoothed gradient descent ===
    if time.monotonic() < deadline - 40.0:
        a = best_f.copy()
        alpha = 0.05
        rng = np.random.default_rng(seed + 7777)
        stuck = 0

        for it in range(30000):
            if time.monotonic() > deadline - 40.0:
                break
            if stuck > 300:
                break

            c = conv(a)
            M = float(c.max())
            sc = 2.0 * n * M

            if sc < best_score - 1e-13:
                best_score = sc
                best_f = a.copy()
                stuck = 0

            # Soft weights
            tau = max(1e-8, 0.025 * M)
            w = np.exp(np.clip((c - M) / tau, -60.0, 0.0))
            ws = float(w.sum())
            if ws < 1e-30:
                w = np.zeros_like(c)
                w[int(c.argmax())] = 1.0
            else:
                w /= ws

            # Gradient: grad[j] = 2 * sum_{k} w[k] * a[k-j] (valid range)
            grad = np.empty(n, dtype=np.float64)
            for j in range(n):
                grad[j] = 2.0 * float(np.dot(w[j:j + n], a))

            # Projected step (tangent to simplex)
            hc = float(np.dot(grad, a))
            step = grad - hc
            sn = float(np.linalg.norm(step))
            if sn < 1e-14:
                stuck += 1
                if stuck > 100:
                    a = a + rng.normal(0, 0.03, n)
                    a = np.maximum(a, 0.0)
                    a /= a.sum()
                    alpha = 0.05
                    stuck = 0
                continue

            improved = False
            for scale in (alpha, alpha * 0.5, alpha * 0.25, alpha * 0.1):
                cand = a - scale * step / sn
                cand = np.maximum(cand, 0.0)
                cs = float(cand.sum())
                if cs < 1e-10:
                    continue
                cand /= cs
                cand_sc = 2.0 * n * float(conv(cand).max())
                if cand_sc < sc - 1e-14:
                    a = cand
                    sc = cand_sc
                    improved = True
                    break

            if improved:
                stuck = 0
                alpha = min(0.3, alpha * 1.02)
            else:
                stuck += 1
                alpha = max(0.003, alpha * 0.92)
                if stuck % 50 == 0:
                    a = a + rng.normal(0, 0.02, n)
                    a = np.maximum(a, 0.0)
                    a /= a.sum()
                    alpha = 0.04
                    stuck = 0

    # === PHASE 3: Local stochastic refinement ===
    if time.monotonic() < deadline - 10.0:
        a = best_f.copy()
        sc = best_score
        rng2 = np.random.default_rng(seed + 424242)
        sigma = 0.06

        for _ in range(8000):
            if time.monotonic() > deadline - 10.0:
                break

            r = rng2.random()
            if r < 0.35:
                z = np.log(np.clip(a, 1e-12, None))
                z += rng2.normal(0.0, sigma, n)
                z -= z.max()
                cand = np.exp(z)
                cand /= cand.sum()
            elif r < 0.65:
                cand = a.copy()
                k = int(rng2.integers(0, n))
                width = max(1, n // 40)
                lo = max(0, k - width)
                hi = min(n, k + width + 1)
                cand[lo:hi] *= np.exp(rng2.normal(0, 0.08))
                cand = np.maximum(cand, 0.0)
                cand /= cand.sum()
            elif r < 0.85:
                cand = a.copy()
                k1 = int(rng2.integers(0, n))
                k2 = int(rng2.integers(0, n))
                amt = 0.015 * float(a.max())
                cand[k1] += amt
                cand[k2] -= amt
                cand = np.maximum(cand, 0.0)
                cand /= cand.sum()
            else:
                # Mirror blend
                t = rng2.uniform(0.05, 0.25)
                cand = (1.0 - t) * a + t * a[::-1]
                cand /= cand.sum()

            cand_sc = 2.0 * n * float(conv(cand).max())
            if cand_sc < sc - 1e-13:
                a = cand
                sc = cand_sc
                if sc < best_score - 1e-13:
                    best_score = sc
                    best_f = a.copy()
                sigma = max(0.015, sigma * 0.97)
            else:
                sigma = min(0.3, sigma * 1.04)

    # === FINAL VALIDATION ===
    best_f = np.clip(best_f, 0.0, 1000.0)
    if not np.isfinite(best_f).all() or best_f.sum() < 0.01:
        best_f = np.ones(n) / n

    final_list = [float(x) for x in best_f]
    try:
        fs = float(evaluate_sequence(final_list))
    except Exception:
        fs = float('inf')

    if not np.isfinite(fs):
        final_list = [1.0 / n] * n
        try:
            fs = float(evaluate_sequence(final_list))
        except Exception:
            pass
        if not np.isfinite(fs):
            return [1.0, 1.0, 1.0]

    return final_list
