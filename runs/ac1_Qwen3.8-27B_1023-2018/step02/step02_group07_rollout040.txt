import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _conv_self(a):
    n = len(a)
    fs = next_fast_len(2 * n - 1)
    F = rfft(a, fs)
    return irfft(F * F, fs)[:2*n-1]


def _score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return math.inf
    b = _conv_self(a)
    m = b.max()
    return 2.0 * n * m / (s * s)


def _lp_solve(a, M):
    n = len(a)
    A_ub = np.zeros((2*n-1, n))
    for i in range(n):
        A_ub[i:i+n, i] = a
    b_ub = np.full(2*n-1, M)
    c = -np.ones(n)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if res.success:
        g = np.maximum(0, res.x)
        gs = g.sum()
        if gs > 1e-10:
            return g / gs
    return None


def _grad_log(a, tau):
    """Gradient of soft-max of conv(a,a), projected to log-space."""
    n = len(a)
    b = _conv_self(a)
    bt = b / tau
    bt -= bt.max()
    e = np.exp(bt)
    p = e / e.sum()
    
    # grad_a[i] = 2 * sum_k p[k] * a[k-i] for valid k-i
    grad_a = np.zeros(n)
    for i in range(n):
        lo = max(0, i)
        hi = min(2*n-2, i+n-1)
        if lo <= hi:
            grad_a[i] = 2.0 * np.dot(p[lo:hi+1], a[lo-i:hi-i+1])
    
    # Project to log-space: grad_z[i] = a_i * (grad_a[i] - sum_j a_j * grad_a[j])
    mean_ag = np.dot(a, grad_a)
    return a * (grad_a - mean_ag)


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + float(budget_s) - 5.0
    
    # --- Get parent ---
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            if p.size > 0 and np.isfinite(p).all():
                p = np.clip(p, 0, 1000)
                if p.sum() > 0.01:
                    parent = p / p.sum()
    except:
        pass
    
    # --- Initialize ---
    rng = np.random.default_rng(seed)
    
    if parent is not None:
        a = parent.copy()
        n = a.size
    else:
        n = 192
        a = np.ones(n) / n
    
    best_a = a.copy()
    best_score = _score(a)
    
    # --- Phase 1: LP-based optimization ---
    lp_end = t0 + 0.65 * (deadline - t0)
    t_blend = 0.3
    stuck = 0
    lp_iter = 0
    
    while time.monotonic() < lp_end:
        b = _conv_self(a)
        M = b.max()
        
        g = _lp_solve(a, M)
        if g is None:
            break
        
        improved = False
        for t in [t_blend, t_blend*0.5, t_blend*0.2, 0.05]:
            a_try = (1-t)*a + t*g
            s_try = _score(a_try)
            if s_try < best_score - 1e-12:
                best_a = a_try.copy()
                best_score = s_try
                a = a_try.copy()
                improved = True
                break
        elif s_try < _score(a) - 1e-12:
            # Accept even if not better than absolute best (curriculum)
            a = a_try.copy()
            improved = True
        
        if improved:
            stuck = 0
        else:
            stuck += 1
            t_blend = max(0.02, t_blend * 0.8)
            if stuck > 30:
                # Random restart from best
                a = best_a + rng.normal(0, 0.05, n)
                a = np.maximum(a, 1e-8)
                a = a / a.sum()
                t_blend = 0.3
                stuck = 0
        
        lp_iter += 1
        if lp_iter % 200 == 0:
            t_blend = 0.3  # Reset blend periodically
    
    # --- Phase 2: Gradient refinement ---
    grad_end = min(lp_end + 0.25 * (deadline - t0), deadline - 30)
    
    a = best_a.copy()
    z = np.log(np.maximum(a, 1e-12))
    z = z - z.mean()

    v_buf = np.zeros(n)
    t_count = 0

        v_hat = v_buf / (1 - 0.999**t_count)

            update = update / un

            best_score = s_new

            z = z_new
        else:

def _conv_self(a):
    n = len(a)

    F = rfft(a, fs)
    return irfft(F * F, fs)[:2 * n - 1]

    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return math.inf
    b = _conv_self(a)
    m = float(b.max())
    if not np.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _lp_solve(a, M):
    n = len(a)
    A_ub = np.zeros((2 * n - 1, n))
    for i in range(n):
        A_ub[i:i + n, i] = a
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs')
        if res.success:
            g = np.maximum(0, res.x)
            gs = float(g.sum())
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None
