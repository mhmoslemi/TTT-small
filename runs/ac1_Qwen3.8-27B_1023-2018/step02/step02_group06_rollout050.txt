from scipy.fft import rfft, irfft, next_fast_len

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(192, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    if p.size == 0 or not np.isfinite(p).all():
        p = np.ones(192, dtype=np.float64)
    return p

def _fft_conv_self(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2*n-1]
    np.maximum(c, 0.0, out=c)
    c[~np.isfinite(c)] = 0.0
    return c

def _score_fast(a):
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    b = _fft_conv_self(a)
    return 2.0 * n * b.max() / (s * s)

def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones_like(a) / len(a)
        return a
    return a / s

def _lp_phase(a_init, deadline, rng):
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_mb = float(_fft_conv_self(a).max())
    no_improve = 0
    max_no_improve = 25
    epoch = 0
    
    while time.monotonic() < deadline:
        epoch += 1
        b = _fft_conv_self(a)
        M = float(b.max())
        
        if M < best_mb * (1 - 1e-9):
            best_mb = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > max_no_improve:
            sigma = min(0.3, 0.02 * math.sqrt(no_improve / max_no_improve))
            a = _normalize(a + rng.normal(0, sigma, n))
            no_improve = 0
            max_no_improve = max(15, int(max_no_improve * 1.2))
            continue
        
        # Active lags: where b is close to max
        frac = 0.95
        active = np.where(b >= frac * M)[0]
        
        g = _build_lp_and_solve(a, M, active, deadline)
        
        if g is None:
            # Subgradient fallback
            k_star = int(np.argmax(b))
            grad = np.zeros(n)
            for j in range(n):
                i = k_star - j
                if 0 <= i < n:
                    grad[j] = 2.0 * a[i]
            gn = float(np.linalg.norm(grad))
            if gn < 1e-15:
                break
            g = _normalize(a - 0.05 / gn * grad)
        
        cos_sim = float(np.dot(a, g) / (np.linalg.norm(a) * np.linalg.norm(g) + 1e-30))
        if cos_sim > 0.999:
            no_improve += 2
            a = _normalize(a + rng.normal(0, 0.02, n))
            continue
        
        t, new_M = _line_search(a, g, M, deadline)
        
        if t > 0 and new_M < M * (1 - 1e-10):
            a = _normalize((1.0 - t) * a + t * g)
        else:
            no_improve += 1
    
    return best_a, best_mb

def _local_search(a, deadline, rng):
    n = len(a)
    a = _normalize(a.copy())
    best_a = a.copy()
    best_mb = float(_fft_conv_self(a).max())
    sigma = 0.02
    no_improve = 0
    
    while time.monotonic() < deadline:
        r = rng.random()
        if r < 0.35:
            cand = a + rng.normal(0, sigma, n)
        elif r < 0.55:
            idx = rng.integers(0, n, size=max(1, n // 15))
            cand = a.copy()
            cand[idx] *= rng.uniform(0.4, 2.5, len(idx))
        elif r < 0.7:
            lo = int(rng.integers(0, n))
            w = int(rng.integers(3, max(4, n // 8)))
            hi = min(n, lo + w)
            cand = a.copy()
            cand[lo:hi] *= np.exp(rng.normal(0, 0.3, hi - lo))
        elif r < 0.85:
            cand = 0.5 * (a + a[::-1]) + rng.normal(0, sigma * 0.3, n)
        else:
            cand = (1 - sigma) * a + sigma * np.ones(n) / n + rng.normal(0, sigma * 0.1, n)
        
        cand = _normalize(cand)
        mb = float(_fft_conv_self(cand).max())
        
        if mb < best_mb * (1 - 1e-10):
            best_mb = mb
            best_a = cand.copy()
            a = cand.copy()
            no_improve = 0
            sigma = max(0.005, sigma * 0.97)
        else:
            no_improve += 1
            if no_improve > 150:
                sigma = min(0.25, sigma * 1.7)
                no_improve = 0
    
    return best_a, best_mb
