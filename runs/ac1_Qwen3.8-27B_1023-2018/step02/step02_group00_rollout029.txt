    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(320, dtype=np.float64)
    return p


def _conv_full(a):
    n = len(a)
    if n <= 256:
        return np.convolve(a, a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[: 2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _conv_matrix(f, n):
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j : j + n, j] = f
    return A


def _lp_direction(f, n, M):
    A = _conv_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(
            c, A_ub=A, b_ub=b_ub,
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True, "time_limit": 10.0},
        )
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    gs = float(g.sum())
    if gs < 1e-15:
        return None
    return g / gs


def _lp_optimize(f0, n, stop_time, seed):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel(), 0, None).copy()
    if f.size < n:
        f = np.concatenate([f, np.zeros(n - f.size)])
    f = f[:n]
    fs = f.sum()
    if fs < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / fs

    best_f = f.copy()
    best_score = _score(f)

    t = 0.3
    t_min = 0.015
    t_max = 0.65
    stall = 0
    max_stall = 100
    consecutive_no_iter = 0

    while time.monotonic() < stop_time - 2.0:
        b = _conv_full(f)
        M = float(b.max())
        if M < 1e-18:
            break

        g_dir = _lp_direction(f, n, M)
        if g_dir is None:
            stall += 12
            if stall > max_stall:
                break
            f = np.clip(f + rng.normal(0, 0.015, n), 0, None)
            fs2 = f.sum()
            f = f / fs2 if fs2 > 1e-10 else best_f.copy()
            continue

        dist = float(np.linalg.norm(g_dir - f))
        if dist < 1e-9:
            stall += 5
            if stall > max_stall:
                break
            f = np.clip(f + rng.normal(0, 0.03, n), 0, None)
            fs2 = f.sum()
            f = f / fs2 if fs2 > 1e-10 else best_f.copy()
            continue

        improved = False
        for _ in range(14):
            f_new = (1.0 - t) * f + t * g_dir
            f_new = np.clip(f_new, 0, None)
            fs_new = f_new.sum()
            if fs_new < 1e-10:
                t = max(t_min, t * 0.4)
                continue
            f_new = f_new / fs_new
            sc = _score(f_new)
            if sc < best_score - 1e-14:
                f = f_new
                best_f = f.copy()
                best_score = sc
                improved = True
                t = min(t_max, t * 1.06)
                stall = 0
                consecutive_no_iter = 0
                break
            else:
                t = max(t_min, t * 0.5)

        if not improved:
            stall += 1
            if stall >= max_stall:
                f = np.clip(f + rng.normal(0, 0.04, n), 0, None)
                fs2 = f.sum()
                f = f / fs2 if fs2 > 1e-10 else best_f.copy()
                t = 0.3
                stall = max_stall // 3

        consecutive_no_iter += 1
        if consecutive_no_iter > 5000:
            break

    return best_f, best_score


def _coord_descent(a, stop_time, seed=12345):
    n = len(a)
    rng = np.random.default_rng(seed)
    a = np.clip(a.copy(), 0, None)
    s = a.sum()
    if s < 1e-10:
        return a, math.inf
    b = _conv_full(a)
    best_sc = 2.0 * n * b.max() / (s * s)
    best_a = a.copy()

    no_imp = 0
    multipliers = (0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.6, 2.2, 0.0, 0.1)

    while time.monotonic() < stop_time - 2.0 and no_imp < 12:
        no_imp += 1
        pass_improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > stop_time - 3.0:
                break
            v = a[i]
            if v < 1e-14:
                cands = (0.01, 0.03, 0.08)
            else:
                cands = multipliers

            best_cand_sc = None
            best_cand_val = None
            best_cand_aff = None
            best_cand_delta = None

            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            if k_lo > k_hi:
                continue
            idx = np.arange(k_lo, k_hi + 1) - i
            prefix_max = b[:k_lo].max() if k_lo > 0 else 0.0
            suffix_max = b[k_hi + 1 :].max() if k_hi + 1 < len(b) else 0.0

            for alpha in cands:
                new_val = v * alpha if v > 1e-14 else alpha
                delta = new_val - v
                if abs(delta) < 1e-16:
                    continue

                affected = b[k_lo : k_hi + 1] + 2.0 * delta * a[idx]
                if 2 * i >= k_lo and 2 * i <= k_hi:
                    affected = affected.copy()
                    affected[2 * i - k_lo] += delta * delta

                aff_max = float(affected.max())
                new_max = max(float(prefix_max), aff_max, float(suffix_max))
                new_sum = s + delta
                if new_sum < 0.01:
                    continue
                sc = 2.0 * n * new_max / (new_sum * new_sum)
                if best_cand_sc is None or sc < best_cand_sc:
                    best_cand_sc = sc
                    best_cand_val = new_val
                    best_cand_aff = affected
                    best_cand_delta = delta

            if best_cand_sc is not None and best_cand_sc < best_sc - 1e-14:
                a[i] = best_cand_val
                b[k_lo : k_hi + 1] = best_cand_aff
                s += best_cand_delta
                best_sc = best_cand_sc
                best_a = a.copy()
                pass_improved = True

        # 2-coordinate moves
        for _ in range(50):
            if time.monotonic() > stop_time - 3.0:
                break
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            oi, oj = a[i], a[j]
            if oi < 1e-14 and oj < 1e-14:
                continue
            for mi in (0.5, 1.5, 0.0):
                for mj in (0.5, 1.5, 0.0):
                    ni = oi * mi if oi > 1e-14 else 0.01
                    nj = oj * mj if oj > 1e-14 else 0.01
                    a[i], a[j] = ni, nj
                    ns = s + (ni - oi) + (nj - oj)
                    if ns < 0.01:
                        a[i], a[j] = oi, oj
                        continue
                    nb = _conv_full(a)
                    sc = 2.0 * n * nb.max() / (ns * ns)
                    if sc < best_sc - 1e-14:
                        best_sc = sc
                        best_a = a.copy()
                        b = nb
                        s = ns
                        pass_improved = True
                    else:
                        a[i], a[j] = oi, oj

        if pass_improved:
            no_imp = 0

    return best_a, best_sc


def _block_refine(a, stop_time, seed=777):
    """Try shifting mass between adjacent blocks."""
    n = len(a)
    rng = np.random.default_rng(seed)
    a = a.copy()
    s = a.sum()
    if s < 1e-10:
        return a, math.inf
    best_a = a.copy()
    best_sc = _score(a)
    no_imp = 0

    while time.monotonic() < stop_time - 2.0 and no_imp < 8:
        no_imp += 1
        improved = False
        for _ in range(80):
            if time.monotonic() > stop_time - 3.0:
                break
            i = int(rng.integers(0, n - 1))
            # Move mass from i to i+1 (or vice versa)
            a2 = a.copy()
            for frac in (0.1, 0.3, 0.5, 0.7, 0.9):
                amt = a[i] * frac
                a2[i] = a[i] - amt
                a2[i + 1] = a[i + 1] + amt
                if a2[i] < 0:
                    a2[i] = 0
                ns = a2.sum()
                if ns < 0.01:
                    a2 = a.copy()
                    continue
                sc = _score(a2)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_a = a2.copy()
                    a = a2.copy()
                    improved = True
                a2 = a.copy()
            if not improved:
                a2 = a.copy()
                amt = a[i + 1] * 0.3
                a2[i + 1] -= amt
                a2[i] += amt
                if a2[i + 1] < 0:
                    continue
                sc = _score(a2)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_a = a2.copy()
                    a = a2.copy()
                    improved = True

        if not improved:
            no_imp += 1
        else:
            no_imp = 0

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(20.0, float(budget_s))
    deadline = t0 + budget - 15.0

    parent = _get_parent()
    n_p = len(parent)

    best_seq = parent.copy()
    try:
        best_score = _score(best_seq)
    except Exception:
        best_score = math.inf
    if not math.isfinite(best_score):
        best_seq = np.ones(32)
        best_score = _score(best_seq)

    exact_best = float(evaluate_sequence([float(x) for x in best_seq]))
    if not math.isfinite(exact_best):
        exact_best = best_score

    def _accept(arr, sc):
        nonlocal best_seq, best_score, exact_best
        if sc < best_score - 1e-13:
            try:
                ex = float(evaluate_sequence([float(x) for x in arr]))
            except Exception:
                ex = math.inf
            if math.isfinite(ex) and ex < exact_best - 1e-13:
                best_seq = np.clip(arr, 0, 1000).copy()
                best_score = ex
                exact_best = ex

    def _run_lp(init, n, stop, sd):
        nonlocal best_seq, best_score, exact_best
        try:
            res, sc = _lp_optimize(init, n, stop, sd)
        except Exception:
            return
        if not np.all(np.isfinite(res)):
            return
        res = np.clip(res, 0, 1000)
        if res.sum() < 0.01:
            return
        if sc < best_score:
            _accept(res, sc)

    # === Phase 1: LP optimization on parent length ===
    n = n_p
    i_arr = np.arange(n, dtype=np.float64)
    rng_main = np.random.default_rng(seed)

    inits = []
    inits.append(parent.copy())
    inits.append(np.ones(n) / n)
    inits.append(0.5 + 0.5 * np.cos(2 * math.pi * i_arr / n))
    sup = np.ones(n) * 0.25
    for fr in (1, 2, 3, 5, 7, 11):
        ph = rng_main.uniform(0, 2 * math.pi)
        sup += 0.25 * np.cos(2 * math.pi * fr * i_arr / n + ph)
    inits.append(np.abs(sup) + 0.03)
    inits.append(np.abs(rng_main.normal(0, 1, n)) + 0.03)
    inits.append(np.exp(rng_main.normal(0, 0.6, n)) + 0.01)
    tri = np.minimum(i_arr, n - 1 - i_arr) / (n / 2.0)
    inits.append(tri + 0.1)
    inits.append(np.abs(parent + rng_main.normal(0, 0.15 * max(1.0, np.mean(parent)), n)) + 0.01)
    # Alternating pattern
    alt = np.ones(n) * 0.5
    alt[::2] += 0.3
    alt[1::2] -= 0.1
    inits.append(np.abs(alt) + 0.05)

    phase1_end = t0 + budget * 0.50
    time_per_init = max(12.0, (phase1_end - t0) / len(inits))

    for idx, init in enumerate(inits):
        if time.monotonic() > phase1_end - 10.0:
            break
        if exact_best < 1.5025:
            break
        stop = min(phase1_end - 5.0, time.monotonic() + time_per_init)
        _run_lp(init, n, stop, seed + 100 * (idx + 1))

    # === Phase 2: Other lengths ===
    phase2_end = t0 + budget * 0.70
    other_lengths = [192, 256, 384, 448, 512]
    if n_p not in other_lengths:
        other_lengths.append(n_p)
    other_lengths = sorted(set(other_lengths))
    time_per_len = max(8.0, (phase2_end - phase1_end) / max(1, len(other_lengths)))

    for L in other_lengths:
        if time.monotonic() > phase2_end - 10.0:
            break
        if exact_best < 1.5025:
            break
        stop = min(phase2_end - 5.0, time.monotonic() + time_per_len)
        if L == len(best_seq) and best_score < 2.0:
            init_l = best_seq.copy()
        else:
            init_l = np.interp(
                np.linspace(0, 1, L), np.linspace(0, 1, n_p), parent
            )
        _run_lp(init_l, L, stop, seed + 50000 + L)

    # === Phase 3: Coordinate descent + block refinement ===
    polish_end = deadline - 8.0
    if time.monotonic() < polish_end - 15.0:
        try:
            refined, ref_sc = _coord_descent(best_seq, polish_end - 10.0, seed=555)
            if ref_sc < best_score:
                _accept(refined, ref_sc)
        except Exception:
            pass

    if time.monotonic() < polish_end - 5.0:
        try:
            refined2, ref_sc2 = _block_refine(best_seq, polish_end - 3.0, seed=666)
            if ref_sc2 < best_score:
                _accept(refined2, ref_sc2)
        except Exception:
            pass

    # Another round of coord descent if time remains
    if time.monotonic() < polish_end - 3.0:
        try:
            refined3, ref_sc3 = _coord_descent(best_seq, polish_end - 1.0, seed=777)
            if ref_sc3 < best_score:
                _accept(refined3, ref_sc3)
        except Exception:
            pass

    # === Final validation ===
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        fallback = np.ones(n_p)
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in fallback]

    return [float(x) for x in best_seq]
