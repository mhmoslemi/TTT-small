import time
import math
from scipy.fft import rfft, irfft, next_fast_len
import cvxpy as cp

def _get_parent():

def _fft_conv(a):
    """Compute full convolution of a with itself using FFT."""
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)
    # The full convolution is in c[0:2*n-1]
    c = c[:2*n-1]
    np.maximum(c, 0.0, out=c)
    c[~np.isfinite(c)] = 0.0
    return c

def _score(a):
    """Fast score computation (scale-invariant form)."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return math.inf
    b = _fft_conv(a)
    return 2.0 * n * b.max() / (s * s)

def _normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def _build_conv_matrix(a):
    """Build the (2n-1) x n matrix A where (A@g)[k] = (a*g)[k]."""
    n = len(a)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(max(0, k-n+1), min(n, k+1)):
            A[k, j] = a[k-j]
    return A

def _solve_lp(a, M):
    """Solve max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns normalized g."""
    n = len(a)
    g = cp.Variable(n, nonneg=True)
    A = _build_conv_matrix(a)
    constraints = [A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    
    for solver in [cp.GLOP, cp.CBC, cp.GLPK, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
                gv = g.value
                if gv is not None and np.isfinite(gv).all():
                    gv = np.maximum(gv, 0)
                    s = gv.sum()
                    if s > 1e-15:
                        return gv / s
        except Exception:
            continue
    return None

def _subgradient(a):
    """Compute subgradient of max conv(a,a) at a."""
    n = len(a)
    b = _fft_conv(a)
    k_star = int(np.argmax(b))
    grad = np.zeros(n)
    for j in range(n):
        k_j = k_star - j
        if 0 <= k_j < n:
            grad[j] = 2.0 * a[k_j]
    return grad, k_star

def _lp_optimize(n, a_init, deadline, rng, max_iters=5000):

            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > max_no_improve:

            a = _normalize(a)
            no_improve = 0
            max_no_improve = int(max(50, max_no_improve * 1.1))
            if max_no_improve > 500:

            continue
        
        # Solve LP

        if g is None:

            gn = np.linalg.norm(grad)
            if gn < 1e-15:

            g = a - 0.1 * grad / gn
            g = _normalize(g)

            a = _normalize(a)

        best_t = 0.0
        best_mb = M
        for t in [0.1, 0.3, 0.5, 0.7, 1.0]:

    n = len(a)

    c = irfft(F * F, fsize)[:2 * n - 1]

def _normalize(a):

def _build_A(a):

    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    A = np.zeros((2 * n - 1, n))
    valid = (diff >= 0) & (diff < n)
    A[valid] = a[diff[valid].astype(np.int32)]
    return A

    for solver in [cp.GLOP, cp.CBC, cp.GLPK, cp.CLARABEL, cp.SCS]:
        try:

                    best_seq = res_a.copy()

                pass

    # Phase 3: Local search polishing (remaining time)
    if time.monotonic() < deadline - 10:
        a_norm = _normalize(best_seq.copy())
        local_dl = deadline - 12
        try:
            res_a, res_sc = _local_search(a_norm, local_dl, np.random.default_rng(seed + 9999))
            exact = float(evaluate_sequence([float(x) for x in res_a]))
            if np.isfinite(exact) and exact < best_score:
                best_score = exact
                best_seq = res_a.copy()
        except Exception:
            pass

    # Phase 4: Quick LP on best if time remains
    if time.monotonic() < deadline - 5:
        n_best = len(best_seq)
        a_norm = _normalize(best_seq.copy())
        quick_dl = deadline - 3
        try:
            res_a, res_sc = _lp_optimize(n_best, a_norm, quick_dl, np.random.default_rng(seed + 55555), max_iters=2000)
            exact = float(evaluate_sequence([float(x) for x in res_a]))
            if np.isfinite(exact) and exact < best_score:
                best_score = exact
                best_seq = res_a.copy()
        except Exception:
            pass

    # Final validation
    try:
        final_score = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_score = math.inf

    if not np.isfinite(final_score) or final_score > 50:
        fb = np.ones(32) / 32.0
        try:
            fs = float(evaluate_sequence([float(x) for x in fb]))
            if np.isfinite(fs):
                return [float(x) for x in fb]
        except Exception:
            pass
        return [1.0]

    return [float(x) for x in best_seq]
