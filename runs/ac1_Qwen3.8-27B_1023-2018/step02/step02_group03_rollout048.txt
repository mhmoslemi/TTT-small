            if j_max >= 0:
                grad[m] = np.dot(a[:j_max + 1], p[m:m + j_max + 1])
            # Second term
            if 2 * m < 2 * n - 1:
                grad[m] += a[m] * p[2 * m]
        return grad

    # --- Smooth gradient descent ---
    def smooth_grad_descent(a, max_iters, stop_time, beta_start=5.0, beta_end=50.0):
        nonlocal best_score, best_arr
        a = np.asarray(a, dtype=np.float64).ravel().copy()
        a = np.clip(a, 0, None)
        s = float(a.sum())
        if s < 1e-15:
            return a
        a = a / s  # normalize to sum = 1
        n = a.size
        local = sc(a)
        update_best(a)
        stale = 0

        for it in range(max_iters):
            if time.monotonic() >= stop_time:
                break
            # Adaptive beta
            progress = min(1.0, it / max(1, max_iters * 0.5))
            beta = beta_start + (beta_end - beta_start) * progress

            c = np.convolve(a, a)
            c_max = c.max()
            # Numerical stability
            exp_bc = np.exp(beta * (c - c_max))
            p = exp_bc / exp_bc.sum()

            # Compute gradient
            grad = np.zeros(n)
            for m in range(n):
                j_max = min(n - 1, 2 * n - 2 - m)
                if j_max >= 0:
                    grad[m] = np.dot(a[:j_max + 1], p[m:m + j_max + 1])
                if 2 * m < 2 * n - 1:
                    grad[m] += a[m] * p[2 * m]

            # Project gradient to tangent space of simplex
            grad_centered = grad - np.mean(grad)
            gnorm = np.linalg.norm(grad_centered)
            if gnorm < 1e-14:
                break

            # Adaptive step size
            best_sc_t, best_a_t = local, None
            for lr in (0.001, 0.002, 0.005, 0.01, 0.02, 0.05, 0.1):
                cand = a - lr * grad_centered
                cand = np.clip(cand, 0, None)
                cs = float(cand.sum())
                if cs < 1e-15:
                    continue
                cand = cand / cs
                v = sc(cand)
                if v < best_sc_t:
                    best_sc_t, best_a_t = v, cand

            if best_a_t is not None and best_sc_t < local * (1 - 1e-10):
                a = best_a_t
                local = best_sc_t
                stale = 0
                if it % 50 == 0:
                    update_best(a)
            else:
                stale += 1
                if stale % 50 == 0:
                    # Perturb
                    ap = a * np.exp(0.05 * rng.standard_normal(n))
                    ap = np.clip(ap, 0, None)
                    aps = float(ap.sum())
                    if aps > 1e-15:
                        ap /= aps
                        if sc(ap) < local:
                            a, local = ap, sc(ap)
                            stale = 0
                            update_best(a)
                if stale > 200:
                    break

        update_best(a)
        return a
