def compute_subgradient(a, n):
    """Compute subgradient of f(a) = 2n*max_b/S² at point a."""
    b = np.convolve(a, a)
    M = b.max()
    S = a.sum()
    active = np.where(b >= M - 1e-10)[0]
    
    grad = np.zeros(n)
    for k in active:
        k = int(k)
        # ∂b[k]/∂a[i] = 2*a[k-i] if 0 <= k-i < n
        # So the contribution to grad[i] from position k is 2*a[k-i]
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for i in range(lo, hi + 1):
            grad[i] += 2.0 * a[k - i]
    
    # Average over active set
    grad /= len(active)
    
    # ∂f/∂a[i] = 2n * grad[i] / S² - 4n * M / S³
    grad = 2.0 * n * grad / (S * S) - 4.0 * n * M / (S ** 3)
    return grad, M
