    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]

def _score(a):

    if not math.isfinite(m) or m < 0:

    return 2.0 * n * m / (s * s)

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(320, dtype=np.float64)
    if p.size == 0:
        p = np.ones(320, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(320, dtype=np.float64)
    return p


def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score_fast(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _solve_lp(A, M, n, time_limit=20):
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
    except Exception:
        return None
    if res is None:
        return None
    if hasattr(res, 'x') and res.x is not None:
        g = np.clip(res.x, 0.0, None)
        if g.sum() > 1e-15:
            return g
    return None


def _lp_optimize(f0, n, stop_time, seed=42):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel()[:n].copy(), 0, None)
    while f.size < n:
        f = np.concatenate([f, np.zeros(n - f.size)])
    f = f[:n]
    fs = float(f.sum())
    if fs < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / fs

    best_f = f.copy()
    best_score = _score_fast(f)
    t = 0.35
    t_min = 0.02
    t_max = 0.7
    stall = 0
    max_stall = 180
    it = 0
    last_improve_it = 0

    while time.monotonic() < stop_time - 3.0:
        it += 1
        if it - last_improve_it > 800:
            # Large perturbation to escape
            f = np.clip(f + rng.normal(0, 0.1, n), 0, None)
            fs = f.sum()
            f = f / fs if fs > 1e-10 else np.ones(n) / n
            t = 0.35
            last_improve_it = it
            continue

        b = _conv_full(f)
        M = float(b.max())
        if M < 1e-15:
            break

        A = _build_A(f)
        g = _solve_lp(A, M, n)

        if g is None:
            stall += 5
            if stall > max_stall:
                break
            f = np.clip(f + rng.normal(0, 0.03, n), 0, None)
            fs = f.sum()
            f = f / fs if fs > 1e-10 else np.ones(n) / n
            continue

        g = g / g.sum()
        dist = float(np.linalg.norm(g - f))
        if dist < 1e-8:
            stall += 3
            if stall > max_stall:
                break
            f = np.clip(f + rng.normal(0, 0.04, n), 0, None)
            fs = f.sum()
            f = f / fs if fs > 1e-10 else np.ones(n) / n
            continue

        improved = False
        # Bisection-style line search
        for attempt in range(14):
            f_new = (1.0 - t) * f + t * g
            f_new = np.clip(f_new, 0.0, None)
            sc = _score_fast(f_new)
            if sc < best_score - 1e-13:
                f = f_new
                best_f = f.copy()
                best_score = sc
                improved = True
                last_improve_it = it
                t = min(t_max, t * 1.04)
                stall = 0
                break
            else:
                t = max(t_min, t * 0.5)

        if not improved:
            stall += 1
            t = max(t_min, t * 0.7)
            if stall >= max_stall:
                f = np.clip(f + rng.normal(0, 0.06, n), 0, None)
                fs = f.sum()
                f = f / fs if fs > 1e-10 else np.ones(n) / n
                t = 0.35
                stall = 0

    return best_f, best_score


def _polish(a, stop_time, seed=4242):

    best = np.clip(np.asarray(a, dtype=np.float64).ravel(), 0, None).copy()
    fs = best.sum()
    if fs < 1e-10:
        return best, math.inf
    best = best / fs
    best_sc = _score_fast(best)
    rng = np.random.default_rng(seed)
    no_imp = 0
    max_no_imp = 10

    while time.monotonic() < stop_time - 2.0:
        improved = False
        # 1-coordinate moves with more multipliers
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 2.0:
                break
            orig = best[i]
            best_val = orig
            best_val_sc = best_sc
            for mlt in (0.3, 0.5, 0.75, 1.0, 1.25, 1.5, 2.0, 0.0):
                new_val = orig * mlt if orig > 1e-12 else (0.01 if mlt > 0 else 0.0)
                if abs(new_val - orig) < 1e-14:
                    continue
                best[i] = new_val
                if best.sum() < 0.01:
                    best[i] = orig
                    continue
                sc = _score_fast(best)
                if sc < best_val_sc - 1e-13:
                    best_val_sc = sc
                    best_val = new_val
            best[i] = best_val
            if best_val_sc < best_sc - 1e-13:
                best_sc = best_val_sc
                improved = True

        # 2-coordinate moves
        for _ in range(50):
            if time.monotonic() > stop_time - 2.0:
                break
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            for mi in (0.5, 1.5):
                for mj in (0.5, 1.5):
                    best[i] = oi * mi if oi > 1e-12 else 0.01
                    best[j] = oj * mj if oj > 1e-12 else 0.01
                    if best.sum() < 0.01:
                        best[i], best[j] = oi, oj
                        continue
                    sc = _score_fast(best)
                    if sc < best_sc - 1e-13:
                        best_sc = sc
                        improved = True
            best[i], best[j] = oi, oj

        # 3-coordinate moves (less frequent)
        for _ in range(15):
            if time.monotonic() > stop_time - 2.0:
                break
            i, j, k = int(rng.integers(n)), int(rng.integers(n)), int(rng.integers(n))
            if i == j or j == k or i == k:
                continue
            oi, oj, ok = best[i], best[j], best[k]
            # Random perturbation of 3 coords
            best[i] = max(0, oi + rng.normal(0, 0.05))
            best[j] = max(0, oj + rng.normal(0, 0.05))
            best[k] = max(0, ok + rng.normal(0, 0.05))
            if best.sum() < 0.01:
                best[i], best[j], best[k] = oi, oj, ok
                continue
            sc = _score_fast(best)
            if sc < best_sc - 1e-13:
                best_sc = sc
                improved = True
            else:
                best[i], best[j], best[k] = oi, oj, ok

        if not improved:
            no_imp += 1
            if no_imp >= max_no_imp:
                break
        else:
            no_imp = 0

    return best, best_sc


def _peak_targeted_polish(a, stop_time, seed=7777):
    """Target the specific lags where max autocorrelation occurs."""
    n = len(a)
    best = np.clip(np.asarray(a, dtype=np.float64).ravel(), 0, None).copy()
    fs = best.sum()
    if fs < 1e-10:
        return best, math.inf
    best = best / fs
    best_sc = _score_fast(best)
    rng = np.random.default_rng(seed)
    no_imp = 0

    while time.monotonic() < stop_time - 2.0:
        b = _conv_full(best)
        M = float(b.max())
        peak_lags = np.where(b >= M - 1e-10)[0]

        # For each peak lag, compute gradient of that lag's autocorrelation
        # and do a descent step
        grad = np.zeros(n)
        for k in peak_lags:
            k = int(k)
            for j in range(n):
                jj = k - j
                if 0 <= jj < n:
                    if jj != j:
                        grad[j] += 2.0 * best[jj]
                    else:
                        grad[j] += best[jj]
        grad /= len(peak_lags)

        # Subgradient of the score (scaled properly)
        # score ~ M / (sum)^2 * 2n, gradient w.r.t. a[j]:
        # d(M)/da[j] - 2M/sum => sign determines increase/decrease
        s = float(best.sum())
        direction = grad - 2.0 * M / s  # coefficient for each a[j]
        # Descent: a[j] -= eta * direction[j], project to [0, inf]
        eta = 0.01
        for _ in range(5):
            f_new = np.clip(best - eta * direction, 0, None)
            if f_new.sum() < 0.01:
                eta *= 0.5
                continue
            sc = _score_fast(f_new)
            if sc < best_sc - 1e-14:
                best = f_new
                best_sc = sc
                no_imp = 0
                break
            else:
                eta *= 0.5
        else:
            # Try random perturbation
            f_new = np.clip(best + rng.normal(0, 0.03, n), 0, None)
            if f_new.sum() > 0.01:
                sc = _score_fast(f_new)
                if sc < best_sc - 1e-14:
                    best = f_new
                    best_sc = sc
                    no_imp = 0
                else:
                    no_imp += 1
            else:
                no_imp += 1

        if no_imp >= 30:
            break

    return best, best_sc
