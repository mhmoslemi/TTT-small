import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
import cvxpy as cp

def _fft_conv(a):
    """Compute full convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    fsize = next_fast_len(2*n - 1 + 1)  # +1 for safety
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)
    c = c[:2*n-1]
    np.maximum(c, 0.0, out=c)
    return c

def _build_conv_matrix(a):
    """Build the matrix A such that (A @ g)_k = (a*g)_k for the LP."""
    n = len(a)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                A[k, j] = a[i]
    return A

def _solve_lp_direction(a, M, n):
    """Solve max sum(g) s.t. (a*g)_k <= M, g >= 0."""
    A = _build_conv_matrix(a)
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    for solver in [cp.MOSEK, cp.ECOS, cp.CBC, cp.GLPK, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                gv = g.value
                if gv is not None and np.isfinite(gv).all():
                    gv = np.maximum(gv, 0)
                    s = gv.sum()
                    if s > 1e-15:
                        return gv / s
        except Exception:
            continue
    return None

    """Subgradient of conv(a,a)[k_star] wrt a."""

    if s < 1e-15:

    return a / s
