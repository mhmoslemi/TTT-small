    # Phase 1: Primary LP at parent length
    if time.monotonic() < dl - 100.0:
        p1_end = dl - 100.0
        starts = _mk(n_par, rng, init_par)
        for i, s in enumerate(starts[:12]):
            if time.monotonic() >= p1_end:
                break
            remaining = p1_end - time.monotonic()
            alloc = min(100.0, remaining * 0.35)
            if alloc < 8.0:
                break
            _lpi(s, time.monotonic() + alloc, best, rng, max_it=2000, tl=7)

    # Phase 2: Subgradient + more LP from best
    if time.monotonic() < dl - 80.0:
        p2_end = dl - 80.0
        if best[1].size == n_par:
            _subgrad(best[1], time.monotonic() + min(40.0, (p2_end - time.monotonic()) * 0.4), best, rng)
        if time.monotonic() < p2_end:
            for _ in range(4):
                if time.monotonic() >= p2_end:
                    break
                nb = best[1].size
                pert = best[1].astype(np.float64) * np.exp(0.04 * rng.standard_normal(nb))
                pert = np.clip(pert, 0.0, None)
                ps = float(pert.sum())
                if ps < 1e-15:
                    break
                pert = pert / ps
                alloc = min(50.0, p2_end - time.monotonic())
                if alloc < 8.0:
                    break
                _lpi(pert, time.monotonic() + alloc, best, rng, max_it=1000, tl=7)

    # Phase 3: Try n=128
    if time.monotonic() < dl - 70.0:
        p3_end = dl - 70.0
        starts128 = _mk(128, rng, best[1])
        for i, s in enumerate(starts128[:6]):
            if time.monotonic() >= p3_end:
                break
            alloc = min(60.0, (p3_end - time.monotonic()) * 0.3)
            if alloc < 8.0:
                break
            _lpi(s, time.monotonic() + alloc, best, rng, max_it=1500, tl=7)

    # Phase 4: Subgradient on best (whatever length)
    if time.monotonic() < dl - 50.0:
        alloc = min(40.0, dl - 50.0 - time.monotonic())
        if alloc > 10.0:
            _subgrad(best[1], time.monotonic() + alloc, best, rng)

    # Phase 5: More diverse LP starts at parent length
    if time.monotonic() < dl - 40.0:
        p5_end = dl - 40.0
        starts2 = _mk(n_par, rng, best[1])
        for i, s in enumerate(starts2[:8]):
            if time.monotonic() >= p5_end:
                break
            alloc = min(40.0, (p5_end - time.monotonic()) * 0.3)
            if alloc < 6.0:
                break
            _lpi(s, time.monotonic() + alloc, best, rng, max_it=800, tl=5)

    # Phase 6: Final LP polish on best
    if time.monotonic() < dl - 12.0:
        alloc = min(25.0, dl - 12.0 - time.monotonic())
        if alloc > 5.0:
            _lpi(best[1], time.monotonic() + alloc, best, rng, max_it=500, tl=5)
            _subgrad(best[1], time.monotonic() + min(10.0, dl - 12.0 - time.monotonic()), best, rng)

    # Final validation
    fin = np.asarray(best[1], dtype=np.float64).ravel().copy()
    fin = np.nan_to_num(fin, nan=0.0, posinf=0.0, neginf=0.0)
    fin = np.clip(fin, 0.0, 1000.0)
    if fin.size < 1:
        fin = np.array([1.0], dtype=np.float64)
    if float(fin.sum()) < 0.01:
        fin = np.ones(fin.size, dtype=np.float64)
    _bu(fin, best)
    fin = np.asarray(best[1], dtype=np.float64).ravel()
    v = evaluate_sequence([float(x) for x in fin])
    if not math.isfinite(v):
        fin = np.array([1.0], dtype=np.float64)
    return [float(x) for x in fin]
