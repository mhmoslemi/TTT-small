        if v < best_v:
            best_v = v
            best_t = t
    if best_t == 0.0:
        return 0.0, cur_M
    # Ternary refinement
    lo, hi = 0.0, 1.0
    # Narrow to bracket around best_t
    delta = 1.0 / 15.0
    lo = max(0.0, best_t - delta)
    hi = min(1.0, best_t + delta)
    for _ in range(15):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        c1 = (1.0 - m1) * a + m1 * g_n
        c1 = c1 / c1.sum()
        v1 = _peak_of(c1)
        c2 = (1.0 - m2) * a + m2 * g_n
        c2 = c2 / c2.sum()
        v2 = _peak_of(c2)
        if v1 < v2:
            hi = m2
        else:
            lo = m1
        if v1 < best_v:
            best_v = v1
            best_t = m1
        if v2 < best_v:
            best_v = v2
            best_t = m2
    # Final check at midpoint
    tm = (lo + hi) / 2.0
    cm = (1.0 - tm) * a + tm * g_n
    cm = cm / cm.sum()
    vm = _peak_of(cm)
    if vm < best_v:
        best_v = vm
        best_t = tm
    return best_t, best_v


def _lp_optimize(a0, stop_time, max_iters=5000, perturb_sigma=0.12):
    n = a0.size
    a = np.clip(a0.astype(np.float64).copy(), 0.0, 1000.0)
    s = a.sum()
    if s <= 0:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_M = _peak_of(a)

    rng = np.random.default_rng(777)
    stuck = 0
    it = 0

    while it < max_iters:
        it += 1
        if time.monotonic() > stop_time - 2.0:
            break

        b = _conv_square(a)
        M = float(b.max())

        if M < best_M * (1.0 - 1e-12):
            best_M = M
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        if stuck > 60:
            sigma = perturb_sigma * min(1.0, stuck / 120.0)
            a = a * np.exp(rng.normal(0.0, sigma, n))
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            if stuck > 200:
                # More aggressive restart
                a = a * np.exp(rng.normal(0.0, 0.3, n))
                a = np.maximum(a, 1e-10)
                a = a / a.sum()
            stuck = 0
            continue

        # Solve LP
        M_relax = M * (1.0 + 1e-8)
        g = _lp_solve(a, M_relax, timeout_s=1.5)
        if g is None:
            M_relax = M * (1.0 + 1e-5)
            g = _lp_solve(a, M_relax, timeout_s=1.0)
        if g is None:
            stuck += 10
            continue

        gs = g.sum()
        if gs <= 0:
            continue
        g = g / gs

        t, new_M = _best_t_line(a, g, M)
        if t > 1e-8 and new_M < M * (1.0 - 1e-10):
            a = (1.0 - t) * a + t * g
            a = np.clip(a, 0.0, 1000.0)
            a = a / a.sum()
        else:
            # Try a small fixed step
            t_s = 0.02
            cand = (1.0 - t_s) * a + t_s * g
            cand = np.clip(cand, 0.0, 1000.0)
            cs = cand.sum()
            if cs > 0:
                cand = cand / cs
                cm = _peak_of(cand)
                if cm < M:
                    a = cand
                else:
                    stuck += 3

    return best_a


def _gradient_polish(a0, stop_time, max_iters=8000):
    n = a0.size
    a = np.clip(a0.astype(np.float64).copy(), 0.0, 1000.0)
    a = a / a.sum()
    best_a = a.copy()
    best_M = _peak_of(a)
    step = 0.02
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > stop_time - 2.0:
            break

        b = _conv_square(a)
        M = float(b.max())
        if M < best_M * (1.0 - 1e-13):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 500:
            break

        # Find all near-peak positions (within 1% of max)
        threshold = M * 0.99
        peak_idxs = np.where(b >= threshold)[0]

        # Compute subgradient (average over near-peaks)
        grad = np.zeros(n)
        for kpk in peak_idxs:
            for j in range(n):
                kj = kpk - j
                if 0 <= kj < n:
                    grad[j] += a[j] + a[kj]
        grad /= max(1, len(peak_idxs))

        # Projected subgradient step
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new <= 0:
            step *= 0.5
            continue
        a_new = a_new / s_new

        new_M = _peak_of(a_new)
        if new_M < M:
            a = a_new
            step = min(step * 1.03, 0.15)
            no_improve = 0
        else:
            step *= 0.8
            if step < 1e-7:
                step = 0.02
                no_improve += 50

    return best_a


def _resample_linear(arr, new_n):
    old_n = arr.size
    if new_n == old_n:
        return arr.copy()
    x_old = np.arange(old_n, dtype=np.float64)
    x_new = np.linspace(0, old_n - 1, new_n)
    return np.interp(x_new, x_old, arr)


def _quick_lp(a0, stop_time, max_iters=800):
    """Shorter LP run for length exploration."""
    return _lp_optimize(a0, stop_time, max_iters=max_iters, perturb_sigma=0.1)


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = float(budget_s)
    deadline = t0 + budget - 8.0

    parent = _get_parent()
    n_parent = parent.size

    best_a = np.clip(parent.astype(np.float64), 0.0, 1000.0)
    bs = best_a.sum()
    if bs < 0.01:
        best_a = np.ones(n_parent) / n_parent
    else:
        best_a = best_a / bs
    best_score = _score(best_a)

    # Phase 1: Main LP optimization (60% budget)
    phase1_end = t0 + budget * 0.58
    try:
        result = _lp_optimize(best_a, stop_time=phase1_end, max_iters=8000)
        sc = _score(result)
        if np.isfinite(sc) and sc < best_score:
            best_a = result
            best_score = sc
    except Exception:
        pass

    # Phase 2: Gradient polish (12% budget)
    phase2_end = t0 + budget * 0.72
    if time.monotonic() < phase2_end - 3.0:
        try:
            result = _gradient_polish(best_a, stop_time=phase2_end)
            sc = _score(result)
            if np.isfinite(sc) and sc < best_score:
                best_a = result
                best_score = sc
        except Exception:
            pass

    # Phase 3: LP again after gradient (8% budget)
    phase3_end = t0 + budget * 0.80
    if time.monotonic() < phase3_end - 3.0:
        try:
            result = _lp_optimize(best_a, stop_time=phase3_end, max_iters=3000,
                                  perturb_sigma=0.08)
            sc = _score(result)
            if np.isfinite(sc) and sc < best_score:
                best_a = result
                best_score = sc
        except Exception:
            pass

    # Phase 4: Try nearby lengths (12% budget)
    phase4_end = t0 + budget * 0.92
    if time.monotonic() < phase4_end - 10.0:
        remaining = phase4_end - time.monotonic()
        lengths = [n_parent - 2, n_parent - 1, n_parent + 1, n_parent + 2]
        per_len = max(3.0, remaining / len(lengths))
        for n_try in lengths:
            if time.monotonic() > phase4_end - 3.0 or per_len < 4.0:
                break
            if n_try < 10:
                continue
            try:
                init = _resample_linear(best_a, n_try)
                init = np.clip(init, 0.0, 1000.0)
                init_s = init.sum()
                if init_s > 0.01:
                    init = init / init_s
                    alloc_end = min(time.monotonic() + per_len * 0.9, phase4_end - 2)
                    result = _quick_lp(init, stop_time=alloc_end, max_iters=500)
                    sc = _score(result)
                    if np.isfinite(sc) and sc < best_score:
                        best_a = result
                        best_score = sc
            except Exception:
                pass

    # Phase 5: Final LP polish (remaining time)
    if time.monotonic() < deadline - 5.0:
        try:
            result = _lp_optimize(best_a, stop_time=deadline - 3.0,
                                  max_iters=5000, perturb_sigma=0.06)
            sc = _score(result)
            if np.isfinite(sc) and sc < best_score:
                best_a = result
                best_score = sc
        except Exception:
            pass

    # Final gradient polish if time permits
    if time.monotonic() < deadline - 3.0:
        try:
            result = _gradient_polish(best_a, stop_time=deadline - 1.0, max_iters=3000)
            sc = _score(result)
            if np.isfinite(sc) and sc < best_score:
                best_a = result
                best_score = sc
        except Exception:
            pass

    # Final validation with exact evaluator
    final_list = [float(x) for x in best_a]
    try:
        fs = float(evaluate_sequence(final_list))
    except Exception:
        fs = float('inf')

    if not np.isfinite(fs):
        # Fallback: return uniform
        return [1.0] * 192

    return final_list
