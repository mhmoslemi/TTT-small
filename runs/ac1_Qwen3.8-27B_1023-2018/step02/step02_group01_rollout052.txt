from scipy.fft import rfft, irfft, next_fast_len


def _get_parent():

        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(191, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    if p.size == 0 or not np.isfinite(p).all():
        p = np.ones(191, dtype=np.float64)
    return p


def _conv_sq(a):
    """Fast convolution a*a via FFT."""
    n = a.size
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score_exact(a):
    """Exact evaluator score."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    n = a.size
    b = _conv_sq(a)
    return 2.0 * n * float(b.max()) / (s * s)


def _score_norm(a):
    """Score assuming sum(a)=1. Returns 2n*max(conv)."""
    n = a.size
    b = _conv_sq(a)
    return 2.0 * n * float(b.max())


def _build_A(a):
    """Build convolution constraint matrix A of shape (2n-1, n)."""
    n = a.size
    m = 2 * n - 1
    k_idx = np.arange(m, dtype=np.int64)[:, None]
    j_idx = np.arange(n, dtype=np.int64)[None, :]
    d = k_idx - j_idx
    mask = (d >= 0) & (d < n)
    safe_d = np.clip(d, 0, n - 1)
    A = np.where(mask, a[safe_d], 0.0)
    return A


def _lp_br(a, M, tl=2.0):
    """LP best response: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = a.size
    A = _build_A(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'time_limit': tl})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    if not np.isfinite(g).all() or g.sum() <= 0:
        return None
    return g


def _lp_optimize(a0, stop_time, max_iters=5000):
    """LP best-response iteration with line search and perturbation."""
    n = a0.size
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()

    rng = np.random.default_rng(999)
    stuck = 0

    for it in range(max_iters):
        if time.monotonic() > stop_time - 0.3:
            break

        b = _conv_sq(a)
        M = float(b.max())

        if M < best_M - 1e-15:
            best_M = M
            best_a = a.copy()
            stuck = 0

        g = _lp_br(a, M, tl=1.5)
        if g is None:
            # Try with relaxed M
            g = _lp_br(a, M * 1.001, tl=1.0)
            if g is None:
                stuck += 1
                if stuck > 50:
                    break
                # Perturb
                a = a * np.exp(rng.normal(0, 0.1, n))
                a = np.maximum(a, 1e-10)
                a = a / a.sum()
                continue

        gs = float(g.sum())
        if gs <= 1.0 + 1e-10:

            if stuck > 100:
                # Bigger perturbation
                a = a * np.exp(rng.normal(0, 0.3, n))
                a = np.maximum(a, 1e-10)

        for t in (1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05):

            if cM < M - 1e-15:
                a = cand
                improved = True
                break

        if not improved:

            if stuck > 200:
                break

    return best_a
