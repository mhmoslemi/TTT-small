                pa = a * (1.0 + sz * rng.standard_normal(n))
                pa = np.clip(pa, 0.0, None)
                ps = float(pa.sum())
                if ps > 1e-15:
                    pa = pa / ps
                    pscore = _score(pa)
                    if math.isfinite(pscore) and pscore < local:
                        a = pa
                        local = pscore
                        stagnant = 0
                        stale_sg = 0
                        alpha_sg = 0.05
                        _update(a, best)
                        improved = True
                        break
            if not improved and stagnant > 60:
                # More aggressive restart
                ra = np.abs(rng.standard_normal(n)) + 0.1
                ra = ra / ra.sum()
                for mix in (0.2, 0.35, 0.5, 0.7):
                    ca = (1.0 - mix) * a + mix * ra
                    ca = np.clip(ca, 0.0, None)
                    cs = float(ca.sum())
                    if cs > 1e-15:
                        ca = ca / cs
                        csc = _score(ca)
                        if math.isfinite(csc) and csc < local:
                            a = ca
                            local = csc
                            stagnant = 0
                            stale_sg = 0
                            alpha_sg = 0.05
                            _update(a, best)
                            break
                if stagnant > 100:
                    break


def _subgradient_refine(a_init, n, time_limit, rng, best, target_score=1.50274349):
    """Pure subgradient descent for final polish."""
    if n < 4 or time_limit < 3:
        return
    t0 = time.monotonic()
    deadline = t0 + time_limit - 2.0

    a = np.asarray(a_init, dtype=np.float64).ravel().copy()
    if a.size != n:
        if a.size > 1:
            a = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, a.size), a)
        else:
            a = np.ones(n)
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if s > 1e-15:
        a = a / s
    local = _score(a)
    _update(a, best)

    alpha = 0.02
    best_alpha = alpha
    patience = 0
    max_patience = 50

    for it in range(50000):
        if time.monotonic() >= deadline:
            break
        if local < target_score:
            break

        b = np.convolve(a, a)
        M = float(b.max())
        if not math.isfinite(M) or M < 1e-30:
            break

        # Active set with adaptive threshold
        eps = max(0.01, 0.05 / (1.0 + 0.01 * it)) * M
        active = np.where(b >= M - eps)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        grad = np.zeros(n)
        for k in active:
            k = int(k)
            dk = k if k < n else (2 * n - 2 - k)
            if dk == 0:
                grad += 2.0 * a
            else:
                for i in range(n):
                    v = 0.0
                    if i + dk < n:
                        v += a[i + dk]
                    if i >= dk:
                        v += a[i - dk]
                    grad[i] += v
        grad /= len(active)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-14:
            break

        # Polyak-style step size
        scale = float(a.std()) + 0.03 * float(a.mean())
        step = alpha * scale / gn

        cand = a - step * grad
        cand = np.clip(cand, 0.0, None)
        cs = float(cand.sum())
        if cs < 1e-15:
            break
        cand = cand / cs
        sc = _score(cand)

        if math.isfinite(sc) and sc < local * (1.0 - 1e-12):
            a = cand
            local = sc
            patience = 0
            best_alpha = alpha
            alpha = min(alpha * 1.05, 0.3)
            _update(a, best)
        else:
            patience += 1
            alpha *= 0.9
            if patience > max_patience:
                # Restart with different step
                alpha = best_alpha * 0.5
                patience = 0
                # Small random kick
                a = a + 0.02 * rng.standard_normal(n)
                a = np.clip(a, 0.0, None)
                a = a / a.sum()
                local = _score(a)
                _update(a, best)
                if patience > 200:
                    break


def _try_multiple_inits(a_base, n, time_per, rng, best, target_score=1.50274349):
    """Try several perturbation directions from the base."""
    if time_per < 5:
        return
    t0 = time.monotonic()
    deadline = t0 + time_per - 3.0

    # Different perturbation strategies
    strategies = []

    # 1. Linear ramp asymmetry
    x = np.linspace(0.0, 1.0, n)
    for asym in (0.3, 0.7, 1.2, 2.0, -0.5, -1.5):
        a = a_base * (1.0 + asym * x)
        a = np.clip(a, 0.0, None)
        if a.sum() > 1e-15:
            strategies.append(a / a.sum())

    # 2. Quadratic shaping
    for c in (0.5, 1.0, 2.0, -0.5, -1.0):
        a = a_base * (1.0 + c * x * x)
        a = np.clip(a, 0.0, None)
        if a.sum() > 1e-15:
            strategies.append(a / a.sum())

    # 3. Sine modulation
    for freq in (1.0, 2.0, 3.0):
        for amp in (0.2, 0.5, 1.0):
            a = a_base * (1.0 + amp * np.sin(2 * np.pi * freq * x))
            a = np.clip(a, 0.0, None)
            if a.sum() > 1e-15:
                strategies.append(a / a.sum())

    # 4. Random perturbations
    for _ in range(4):
        a = a_base * (1.0 + 0.3 * rng.standard_normal(n))
        a = np.clip(a, 0.0, None)
        if a.sum() > 1e-15:
            strategies.append(a / a.sum())

    # Sort by score, take best few
    scored = []
    for sa in strategies:
        sc = _score(sa)
        if math.isfinite(sc):
            scored.append((sc, sa))
    scored.sort(key=lambda x: x[0])

    # Run LP on top candidates
    for sc, sa in scored[:6]:
        if time.monotonic() >= deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break
        _lp_lp_hybrid_iterate(sa, n, remaining, rng, best, target_score)


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    try:
        remaining = max(0.0, float(budget_s))
    except Exception:
        remaining = 1300.0
    deadline = t_start + remaining - 20.0

    best = [float('inf'), np.array([1.0, 1.0], dtype=np.float64)]
    _update(best[1], best)

    target_score = 1.50274349

    # Get parent sequence
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            parent = np.asarray(h, dtype=np.float64).ravel()
            parent = np.nan_to_num(parent, nan=0.0, posinf=0.0, neginf=0.0)
            parent = np.clip(parent, 0.0, 1000.0)
            if parent.size > 1:
                _update(parent, best)
    except Exception:
        parent = None

    def _rem():
        return max(0.0, deadline - time.monotonic())

    # Determine working length
    if parent is not None and parent.size >= 4:
        n_work = parent.size
        a_work = parent.copy()
        s = float(a_work.sum())
        if s > 1e-15:
            a_work = a_work / s
    else:
        n_work = 320
        a_work = np.ones(n_work) / n_work

    # Phase 1: Main LP iteration on parent (most of the budget)
    if _rem() > 60:
        alloc = min(_rem() * 0.55, 600.0)
        _lp_lp_hybrid_iterate(a_work, n_work, alloc, rng, best, target_score)

    # Phase 2: Try perturbed initializations
    if _rem() > 60:
        alloc = min(_rem() * 0.25, 300.0)
        _try_multiple_inits(a_work, n_work, alloc, rng, best, target_score)

    # Phase 3: Subgradient polish on best found
    if _rem() > 20:
        best_arr = np.asarray(best[1], dtype=np.float64).ravel()
        nb = best_arr.size
        if nb < 4:
            nb = n_work
        alloc = min(_rem() * 0.3, 120.0)
        _subgradient_refine(best_arr, nb, alloc, rng, best, target_score)

    # Phase 4: Final LP polish
    if _rem() > 15:
        best_arr = np.asarray(best[1], dtype=np.float64).ravel()
        nb = best_arr.size
        if nb < 4:
            nb = n_work
        alloc = min(_rem() * 0.4, 60.0)
        _lp_lp_hybrid_iterate(best_arr, nb, alloc, rng, best, target_score)

    # Phase 5: Subgradient on final best
    if _rem() > 10:
        best_arr = np.asarray(best[1], dtype=np.float64).ravel()
        nb = best_arr.size
        alloc = min(_rem() * 0.5, 30.0)
        _subgradient_refine(best_arr, nb, alloc, rng, best, target_score)

    # Final validation
    final = np.asarray(best[1], dtype=np.float64).ravel()
    final = np.nan_to_num(final, nan=0.0, posinf=0.0, neginf=0.0)
    final = np.clip(final, 0.0, 1000.0)
    if final.size == 0:
        final = np.ones(32, dtype=np.float64)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(32, dtype=np.float64)
        s = 32.0

    fin_sc = evaluate_sequence([float(x) for x in final])
    if not math.isfinite(fin_sc) or fin_sc > best[0] + 0.01:
        # Fallback to known-good
        if parent is not None and parent.size > 1:
            final = parent
            fin_sc = evaluate_sequence([float(x) for x in final])
        if not math.isfinite(fin_sc):
            final = np.ones(32, dtype=np.float64)
            fin_sc = evaluate_sequence([float(x) for x in final])

    return [float(x) for x in final]
