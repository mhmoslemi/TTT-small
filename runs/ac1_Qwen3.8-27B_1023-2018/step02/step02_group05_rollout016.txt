def _smooth_gradient(a, tau):
    """Compute smooth approximation and its gradient."""
    n = a.size
    c = np.convolve(a, a)  # length 2n-1
    S = float(a.sum())
    if S < 1e-15:
        return float('inf'), None
    # Soft max
    c_shifted = c / tau - np.max(c) / tau
    exp_c = np.exp(c_shifted)
    sum_exp = float(exp_c.sum())
    if sum_exp < 1e-300:
        return float('inf'), None
    p = exp_c / sum_exp  # softmax weights
    L = tau * (float(np.max(c)) / tau + math.log(sum_exp))  # = tau * logsumexp(c/tau)
    f_val = 2.0 * n * L / (S * S)
    
    # Gradient: df/da_i = (2n/S^2) * (2/tau) * h[i] - 4n*L/S^3
    # where h[i] = sum_k p[k] * a[k-i] for valid k-i
    # h[i] = sum_{j=0}^{n-1} p[i+j] * a[j]
    h = np.zeros(n)
    for i in range(n):
        k_start = i
        k_end = min(i + n, 2 * n - 1)
        if k_start < 2 * n - 1 and k_end > 0:
            h[i] = float(np.dot(p[k_start:k_end], a[:k_end - k_start]))
    grad = (4.0 * n / (S * S * tau)) * h - (4.0 * n * L) / (S * S * S)
    return f_val, grad

        xp = np.linspace(0.0, 1.0, parent.size)
        xn = np.linspace(0.0, 1.0, n)
        a0 = np.interp(xn, xp, parent)
    s0 = float(a0.sum())
    if s0 > 1e-15:
        a0 = a0 / s0
    
    # Phase 1: LP refinement on parent (largest share)
    if _rem() > 100:
        alloc = min(500.0, _rem() * 0.45)
        if alloc > 30:
            a1, _ = _lp_refine(a0, n, alloc, rng, best, use_minmax=False)
    
    # Phase 2: Smooth gradient refinement
    if _rem() > 80:
        best_arr = np.asarray(best[1], dtype=np.float64).ravel()
        if best_arr.size != n:
            best_arr = a1 if parent is not None else np.ones(n)
            if best_arr.size != n:
                best_arr = np.interp(np.linspace(0,1,n), np.linspace(0,1,best_arr.size), best_arr)
        alloc = min(200.0, _rem() * 0.2)
        if alloc > 15:
            a2, _ = _smooth_refine(best_arr, n, alloc, rng, best)
    
    # Phase 3: More LP refinement from best
    if _rem() > 60:
        best_arr = np.asarray(best[1], dtype=np.float64).ravel()
        if best_arr.size != n:
            best_arr = np.interp(np.linspace(0,1,n), np.linspace(0,1,best_arr.size), best_arr)
        alloc = min(200.0, _rem() * 0.3)
        if alloc > 15:
            a3, _ = _lp_refine(best_arr, n, alloc, rng, best, use_minmax=True)
    
    # Phase 4: Final LP polish with small steps
    if _rem() > 30:
        best_arr = np.asarray(best[1], dtype=np.float64).ravel()
        if best_arr.size != n:
            best_arr = np.interp(np.linspace(0,1,n), np.linspace(0,1,best_arr.size), best_arr)
        alloc = min(80.0, _rem() * 0.4)
        if alloc > 10:
            a4, _ = _lp_refine(best_arr, n, alloc, rng, best, use_minmax=False)
    
    # Phase 5: Try a perturbed variant
    if _rem() > 40:
        best_arr = np.asarray(best[1], dtype=np.float64).ravel()
        if best_arr.size != n:
            best_arr = np.interp(np.linspace(0,1,n), np.linspace(0,1,best_arr.size), best_arr)
        pa = best_arr * np.exp(0.02 * rng.standard_normal(n))
        pa = np.clip(pa, 0.0, None)
        ps = float(pa.sum())
        if ps > 1e-15:
            pa = pa / ps
            _update_best(pa, best)
        best_arr = np.asarray(best[1], dtype=np.float64).ravel()
        if best_arr.size != n:
            best_arr = np.interp(np.linspace(0,1,n), np.linspace(0,1,best_arr.size), best_arr)
        alloc = min(40.0, _rem() * 0.5)
        if alloc > 8:
            a5, _ = _lp_refine(best_arr, n, alloc, rng, best, use_minmax=False)
    
    # Final validation
    final = np.asarray(best[1], dtype=np.float64).ravel()
    final = np.nan_to_num(final, nan=0.0, posinf=0.0, neginf=0.0)
    final = np.clip(final, 0.0, 1000.0)
    if final.size == 0:
        final = np.array([1.0, 1.0], dtype=np.float64)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(16, dtype=np.float64)
    
    fin_sc = evaluate_sequence([float(x) for x in final])
    if not math.isfinite(fin_sc) or fin_sc > best[0] + 0.005:
        final = np.ones(16, dtype=np.float64)
        fin_sc = evaluate_sequence([float(x) for x in final])
    
    return [float(x) for x in final]
