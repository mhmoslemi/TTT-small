def _fft_conv_self(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2*n-1]
    np.maximum(c, 0.0, out=c)
    c[~np.isfinite(c)] = 0.0
    return c

def _build_A(a):
    """Build the (2n-1) x n matrix A where A[k,j] = a[k-j] if 0<=k-j<n"""
    n = len(a)
    # Vectorized construction
    k = np.arange(2*n-1)

    # A[k,j] = a[k-j] if 0 <= k-j < n
    # This is equivalent to: A is a banded matrix
    # Let me use a different approach - build it as sum of shifted identity matrices
    A = np.zeros((2*n-1, n))
    for d in range(n):
        # a[d] contributes to A[k, k-d] = a[d] when 0 <= k-d < n, i.e., d <= k < d+n
        lo = d
        hi = min(2*n-1, d+n)
        if lo < hi:
            rows = np.arange(lo, hi)
            cols = rows - d
            A[rows, cols] = a[d]
    return A

def _project_simplex(a):
    """Project onto {x >= 0, sum(x) = 1}"""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    c[~np.isfinite(c)] = 0.0
    return c

    d = k - j

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
import cvxpy as cp

def _conv(a):
    n = len(a)
    L = next_fast_len(2 * n - 1)
    F = rfft(a, L)
    c = irfft(F * F, L)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    c[~np.isfinite(c)] = 0.0
    return c

def _Amat(a):
    n = len(a)
    k = np.arange(2 * n - 1)[:, None]
    j = np.arange(n)[None, :]
    d = k - j
    A = np.zeros((2 * n - 1, n))
    msk = (d >= 0) & (d < n)
    A[msk] = a[d[msk].astype(np.intp)]
    return A

def _proj(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-18:
        return np.ones(len(a)) / len(a)
    return a / s

def _lpdir(a, M):
    n = len(a)
    A = _Amat(a)
    g = cp.Variable(n, nonneg=True)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [A @ g <= M])
    for solver in [cp.CBC, cp.GLPK, cp.GLOP, cp.ECOS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ("optimal", "optimal_inaccurate"):
                v = g.value
                if v is not None and np.all(np.isfinite(v)):
                    v = np.clip(v, 0.0, None)
                    s = v.sum()
                    if s > 1e-15:
                        return v / s
        except Exception:
            continue
    return None

def _sg(a):
    b = _conv(a)
    ks = int(np.argmax(b))
    n = len(a)
    m = np.arange(n)
    j = ks - m
    g = np.zeros(n)
    ok = (j >= 0) & (j < n)
    g[ok] = 2.0 * a[j[ok]]
    return g

def _lpopt(a0, dl, rng):
    n = len(a0)
    a = _proj(a0.copy())
    ba, bm = a.copy(), float(_conv(a).max())
    ni, mni = 0, 35
    while time.monotonic() < dl:
        M = float(_conv(a).max())
        if M < bm * (1 - 1e-14):
            bm, ba, ni = M, a.copy(), 0
        else:
            ni += 1
        if ni > mni:
            a = _proj(a + rng.normal(0, 0.035, n))
            ni, mni = 0, max(20, mni + 5)
            continue
        g = _lpdir(a, M)
        if g is None:
            sg = _sg(a)
            ng = np.linalg.norm(sg)
            if ng < 1e-15:
                break
            a = _proj(a - 0.06 / ng * sg)
            continue
        sim = float(np.dot(a, g) / (np.linalg.norm(a) * np.linalg.norm(g) + 1e-30))
        if sim > 0.9999:
            a = _proj(a + rng.normal(0, 0.02, n))
            ni += 3
            continue
        bt, bml = 0.0, M
        for t in (0.03, 0.07, 0.12, 0.18, 0.25, 0.35, 0.5, 0.65, 0.8, 1.0):
            at = _proj((1.0 - t) * a + t * g)
            mt = float(_conv(at).max())
            if mt < bml * (1 - 1e-14):
                bml, bt = mt, t
        if bt > 0:
            a = _proj((1.0 - bt) * a + bt * g)
        else:
            sg = _sg(a)
            ng = np.linalg.norm(sg)
            if ng > 1e-15:
                a = _proj(a - 0.04 / ng * sg)
            ni += 1
    return ba, bm

def _loc(a0, dl, rng):
    n = len(a0)
    a = _proj(a0.copy())
    ba, bm = a.copy(), float(_conv(a).max())
    sig, ni = 0.012, 0

        if r < 0.28:

            c = a.copy()

        elif r < 0.62:

        elif r < 0.76:

        else:
