def _project_simplex(v):
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rng = np.arange(1, n + 1, dtype=np.float64)
    cond = u * rng > (cssv - 1.0)

        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)

                    break
            if not improved:
                stagnant += 1
                alpha *= 0.85
                if stagnant > 40:
                    # Perturbation
                    for scale in (0.01, 0.03, 0.05, 0.1, 0.2, 0.4):
                        a_p = _project_simplex(a * np.exp(scale * rng.standard_normal(n)))
                        sc_p = _score(a_p)
                        if math.isfinite(sc_p) and sc_p < local:
                            a = a_p
                            local = sc_p
                            stagnant = 0
                            alpha = 0.03
                            break
                    if stagnant > 100:
                        break

    return best_a, best_local


def _lp_phase(a_init, n, time_limit, rng, best, target_score):
    """LP-based polishing with adaptive constraint set."""
    if linprog is None or sps is None:
        return None

    t0 = time.monotonic()
    deadline = t0 + time_limit - 2.0

    a = np.asarray(a_init, dtype=np.float64).ravel().copy()
    if a.size != n:
        if a.size > 1:
            a = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, a.size), a)
        else:
            a = np.ones(n) / n
    a = np.clip(a, 0.0, None)
    s = a.sum()
    a = a / s if s > 1e-15 else np.ones(n) / n

    local = _score(a)
    _update_best(a, best)
    best_local = local
    best_a = a.copy()
    stagnant = 0

    # Precompute index arrays for LP matrix
    col_idx = np.repeat(np.arange(n, dtype=np.int64), n)
    row_idx = col_idx + np.tile(np.arange(n, dtype=np.int64), n)

    for it in range(500):
        if time.monotonic() >= deadline:
            break
        if local < target_score:
            break

        b = np.convolve(a, a)
        M = float(b.max())
        if not math.isfinite(M) or M < 1e-30:
            break

        # Adaptive: use all constraints early, top-k later
        if it < 30:
            use_rows = slice(None)
            n_rows = 2 * n - 1
        elif it < 80:
            top_k = min(40, 2 * n - 1)
            row_order = np.argsort(b)[::-1]
            use_rows = row_order[:top_k]
            n_rows = len(use_rows)
        else:
            top_k = 20
            row_order = np.argsort(b)[::-1]
            use_rows = row_order[:top_k]
            n_rows = len(use_rows)

        # Build sparse A for the selected rows
        if use_rows is slice(None):
            rows = np.arange(2 * n - 1)
        else:
            rows = np.asarray(use_rows, dtype=np.int64)

        # Filter the precomputed indices to selected rows
        mask = np.isin(row_idx, rows)
        r_idx = row_idx[mask]
        c_idx = col_idx[mask]
        vals = np.tile(a, n)[mask]

        # Remap row indices to [0, n_rows)
        row_map = np.searchsorted(np.sort(rows), r_idx)
        A = sps.csr_matrix((vals, (row_map, c_idx)), shape=(n_rows, n))

        c_obj = -np.ones(n)
        b_ub = np.full(n_rows, M * (1.0 + 1e-12))

        tl = min(8.0, max(0.5, deadline - time.monotonic()))
        if tl < 1.0:
            break

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                          bounds=[(0.0, None)] * n,
                          method='highs',
                          options={'presolve': True, 'time_limit': tl})
        except Exception:
            stagnant += 1
            if stagnant > 8:
                break
            continue

        if res is None or getattr(res, 'x', None) is None:
            stagnant += 1
            if stagnant > 8:
                break
            continue

        g = np.asarray(res.x, dtype=np.float64).ravel()
        g = np.clip(g, 0.0, None)
        sg = float(g.sum())
        if sg < 1e-15 or not math.isfinite(sg):
            stagnant += 1
            if stagnant > 8:
                break
            continue
        g = g / sg

        # Line search over step sizes
        best_sc = local
        best_cand = None
        ts = (0.02, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5)
        for t in ts:
            cand = (1.0 - t) * a + t * g
            cand = np.clip(cand, 0.0, None)
            cs = float(cand.sum())
            if cs < 1e-15:
                continue
            cand = cand / cs
            sc = _score(cand)
            if math.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_cand = cand

        if best_cand is not None and best_sc < local * (1 - 1e-11):
            a = best_cand
            local = best_sc
            stagnant = 0
            if local < best_local:
                best_local = local
                best_a = a.copy()
                _update_best(a, best)
        else:
            stagnant += 1
            if stagnant > 25:
                # Perturb and retry
                a_p = _project_simplex(a * np.exp(0.05 * rng.standard_normal(n)))
                sc_p = _score(a_p)
                if math.isfinite(sc_p) and sc_p < local:
                    a = a_p
                    local = sc_p
                    stagnant = 0
                elif stagnant > 40:
                    break

    return best_a, best_local


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    try:
        remaining = max(1.0, float(budget_s))
    except Exception:
        remaining = 1300.0
    deadline = t_start + remaining - 20.0

    best = [float('inf'), np.array([1.0], dtype=np.float64)]

    # Get parent
    parent = None
    n = 320
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            parent = np.asarray(h, dtype=np.float64).ravel()
            parent = np.nan_to_num(parent, nan=0.0, posinf=0.0, neginf=0.0)
            parent = np.clip(parent, 0.0, 1000.0)
            if parent.size > 1 and parent.sum() > 0.01:
                _update_best(parent, best)
                n = parent.size
    except Exception:
        parent = None

    if parent is None:
        parent = np.ones(n, dtype=np.float64)
        _update_best(parent, best)

    def _rem():
        return max(0.0, deadline - time.monotonic())

    # Phase 1: Subgradient on parent
    if _rem() > 100:
        alloc = min(350.0, _rem() * 0.4)
        sa, sl = _subgrad_phase(parent, n, alloc, rng, best, TARGET)
        if sa is not None:
            parent = sa

    # Phase 2: LP polish on best from subgradient
    if _rem() > 80:
        best_arr = np.asarray(best[1], dtype=np.float64).ravel()
        nb = best_arr.size
        if nb < 4:
            nb = n
        alloc = min(300.0, _rem() * 0.35)
        la, ll = _lp_phase(best_arr, nb, alloc, rng, best, TARGET)
        if la is not None:
            best_arr = la

    # Phase 3: Another subgradient pass from LP result
    if _rem() > 60:
        best_arr = np.asarray(best[1], dtype=np.float64).ravel()
        nb = best_arr.size
        if nb < 4:
            nb = n
        alloc = min(200.0, _rem() * 0.3)
        sa2, sl2 = _subgrad_phase(best_arr, nb, alloc, rng, best, TARGET)

    # Phase 4: Final LP polish
    if _rem() > 30:
        best_arr = np.asarray(best[1], dtype=np.float64).ravel()
        nb = best_arr.size
        if nb < 4:
            nb = n
        alloc = min(150.0, _rem() * 0.4)
        la2, ll2 = _lp_phase(best_arr, nb, alloc, rng, best, TARGET)

    # Phase 5: Try a different initialization (anti-symmetric) if time
    if _rem() > 50:
        n_alt = n
        # Build anti-palindromic-like init: high at ends, low in middle
        x = np.linspace(0, 1, n_alt)
        a_anti = 0.3 + 0.7 * (0.5 + 0.5 * np.cos(np.pi * x))
        a_anti = a_anti / a_anti.sum()
        # Quick subgradient
        alloc = min(_rem() * 0.35, 80.0)
        if alloc > 10:
            _subgrad_phase(a_anti, n_alt, alloc, rng, best, TARGET)
        if _rem() > 30:
            b_arr = np.asarray(best[1], dtype=np.float64).ravel()
            nb = b_arr.size
            if nb < 4:
                nb = n_alt
            alloc2 = min(_rem() * 0.4, 50.0)
            if alloc2 > 10:
                _lp_phase(b_arr, nb, alloc2, rng, best, TARGET)

    # Phase 6: Final short LP if time
    if _rem() > 15:
        best_arr = np.asarray(best[1], dtype=np.float64).ravel()
        nb = best_arr.size
        if nb < 4:
            nb = n
        alloc = min(_rem() * 0.5, 30.0)
        if alloc > 5:
            _lp_phase(best_arr, nb, alloc, rng, best, TARGET)

    # Final validation
    final = np.asarray(best[1], dtype=np.float64).ravel()
    final = np.nan_to_num(final, nan=0.0, posinf=0.0, neginf=0.0)
    final = np.clip(final, 0.0, 1000.0)
    if final.size == 0 or final.sum() < 0.01:
        final = np.ones(16, dtype=np.float64)

    fin_sc = evaluate_sequence([float(x) for x in final])
    if not math.isfinite(fin_sc):
        final = np.ones(16, dtype=np.float64)
        fin_sc = evaluate_sequence([float(x) for x in final])

    return [float(x) for x in final]
