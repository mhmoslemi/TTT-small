def _subgradient_descent(a0, stop_time, max_iters=5000):
    """Subgradient descent on max_k (a*a)[k] with sum(a)=1, a>=0."""
    n = a0.size
    a = np.maximum(a0, 0.0).astype(np.float64).copy()
    a = a / a.sum()
    
    best_a = a.copy()
    best_M = float(_convolve_square(a).max())
    
    step = 0.05
    momentum = np.zeros(n)
    rho = 0.9
    
    for it in range(max_iters):
        if time.monotonic() > stop_time - 1.0:
            break
        
        b = _convolve_square(a)
        M = float(b.max())
        
        if M < best_M - 1e-14:
            best_M = M
            best_a = a.copy()
        
        # Find active peaks
        tol = M * 1e-6
        K = np.where(b >= M - tol)[0]
        
        # Compute subgradient
        g = np.zeros(n)
        for k in K:
            # db[k]/da[j] = 2*a[k-j] if 0<=k-j<n else 0
            j_start = max(0, k - (n-1))
            j_end = min(n-1, k)
            for j in range(j_start, j_end+1):
                g[j] += 2.0 * a[k - j]
        g /= len(K)
        
        # Project: we need to move in direction -g while maintaining sum(a)=1, a>=0
        # Use projected gradient: a_new = max(a - step*g, 0), then normalize
        # But we need to handle the projection onto {a>=0, sum(a)=1} properly
        
        momentum = rho * momentum + (1-rho) * g
        direction = -momentum
        
        a_new = a + step * direction
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s <= 0:
            continue
        a_new = a_new / s
        
        M_new = float(_convolve_square(a_new).max())
        if M_new < M:
            a = a_new
            # Optionally increase step
            step = min(step * 1.02, 0.3)
        else:
            step = max(step * 0.8, 0.001)
    
    return best_a

    """A[k,i] = a[k-i] if 0<=k-i<n else 0. Shape (2n-1, n)."""
    n = a.size

        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A[k, i] = a[j]
    return A

    cols = np.arange(n)[None, :]  # shape (1, n)
    rows = np.arange(m)[:, None]  # shape (m, 1)
    j = rows - cols  # A[k,i] uses a[k-i], so j = k-i

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, next_fast_len


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(96, dtype=np.float64) / 96.0
    p = np.clip(p, 0.0, 1000.0)
    if p.size == 0 or not np.isfinite(p).all():
        p = np.ones(96, dtype=np.float64) / 96.0
    return p


def _auto_corr(a):
    n = a.size
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _score(a):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = a.size
    return 2.0 * n * float(_auto_corr(a).max()) / (s * s)


def _build_A(a):
    n = a.size
    m = 2 * n - 1
    cols = np.arange(n)[None, :]
    rows = np.arange(m)[:, None]
    j = rows - cols
    mask = (j >= 0) & (j < n)
    A = np.where(mask, a[np.clip(j, 0, n - 1)], 0.0)
    return A


def _lp_solve(a, M, timeout=3.0):
    n = a.size
    A = _build_A(a)
    c_vec = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_vec, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': timeout,
                               'primal_feasibility_tolerance': 1e-11,
                               'dual_feasibility_tolerance': 1e-11})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    if not np.isfinite(g).all() or g.sum() <= 0:
        return None
    return g


def _lp_iterate(a0, stop_time, max_iters=5000, verbose=False):
    n = a0.size
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s <= 0:
        a = np.ones(n) / n
    a = a / a.sum()

    best_a = a.copy()
    best_M = float(_auto_corr(a).max())
    stuck = 0

    for it in range(max_iters):
        remaining = stop_time - time.monotonic()
        if remaining < 1.5:
            break

        b = _auto_corr(a)
        M = float(b.max())
        if M < best_M - 1e-14:
            best_M = M
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        if stuck > 250:
            break

        t_limit = min(2.5, max(0.5, remaining * 0.15))
        g = _lp_solve(a, M, timeout=t_limit)
        if g is None:
            break

        gs = g.sum()
        if gs <= 0:
            break
        g = g / gs

        improved = False
        for t in [0.5, 0.4, 0.3, 0.25, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02]:
            cand = (1.0 - t) * a + t * g
            cand = np.maximum(cand, 0.0)
            cs = cand.sum()
            if cs <= 0:
                continue
            cand = cand / cs
            cM = float(_auto_corr(cand).max())
            if cM < M * (1.0 - 1e-13):
                a = cand
                improved = True
                break

        if not improved:
            stuck += 5

    return best_a


def _smoothed_max_grad(a, M_active_frac=0.0):
    """Gradient of smoothed max of auto-correlation. Returns (grad, M)."""
    n = a.size
    b = _auto_corr(a)
    M = float(b.max())

    if M_active_frac > 0:
        # Hard subgradient: average over active peaks
        thresh = M - M_active_frac * M
        K = np.where(b >= thresh)[0]
        if K.size == 0:
            K = np.array([np.argmax(b)])
        g = np.zeros(n)
        for k in K:
            j_start = max(0, k - (n - 1))
            j_end = min(n - 1, k)
            g[j_start:j_end + 1] += 2.0 * a[k - j_start:k - j_end + 1][::-1].copy()
        g /= K.size
        return g, M
    else:
        # Softmax smoothed
        eps = 0.01 * M + 1e-15
        shifted = b / eps
        shifted -= shifted.max()
        p = np.exp(shifted)
        p /= p.sum()
        g = np.zeros(n)
        K_all = np.arange(2 * n - 1)
        for k in K_all:
            if p[k] < 1e-10:
                continue
            j_start = max(0, k - (n - 1))
            j_end = min(n - 1, k)
            g[j_start:j_end + 1] += 2.0 * p[k] * a[k - j_start:k - j_end + 1][::-1].copy()
        return g, M


def _subgradient_polish(a0, stop_time, max_iters=20000, verbose=False):
    n = a0.size
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    a = a / a.sum()

    best_a = a.copy()
    best_M = float(_auto_corr(a).max())

    step = 0.03
    momentum = np.zeros(n)
    rho = 0.85
    no_improve = 0

    for it in range(max_iters):
        remaining = stop_time - time.monotonic()
        if remaining < 1.5:
            break

        g, M = _smoothed_max_grad(a, M_active_frac=0.001)

        if M < best_M - 1e-15:
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 2000:
            step = min(step * 1.5, 0.08)
            no_improve = 0

        momentum = rho * momentum + (1 - rho) * g
        a_new = a - step * momentum
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s <= 0:
            step *= 0.9
            continue
        a_new = a_new / s
        M_new = float(_auto_corr(a_new).max())
        if M_new < M * (1 - 1e-14):
            a = a_new
            step = min(step * 1.005, 0.08)
        else:
            step = max(step * 0.93, 0.002)

    return best_a


def _block_perturb_polish(a0, stop_time, rng, max_iters=3000):
    """Perturb blocks of the sequence and use LP to recover."""
    n = a0.size
    a = a0.copy()
    best_a = a.copy()
    best_M = float(_auto_corr(a).max())
    no_improve = 0

    for it in range(max_iters):
        remaining = stop_time - time.monotonic()
        if remaining < 3.0:
            break

        r = rng.random()
        if r < 0.4:
            # Scale a block
            width = rng.integers(3, max(4, n // 8))
            start = rng.integers(0, n - width)
            factor = np.exp(rng.normal(0, 0.15))
            cand = a.copy()
            cand[start:start + width] *= factor
        elif r < 0.6:
            # Gaussian bump
            center = rng.uniform(0, n)
            width = rng.uniform(n * 0.05, n * 0.3)
            amp = rng.uniform(-0.3, 0.3)
            x = np.arange(n, dtype=np.float64)
            bump = amp * np.exp(-0.5 * ((x - center) / width) ** 2)
            cand = a * (1.0 + bump)
        elif r < 0.75:
            # Swap mass between two regions
            i, j = rng.integers(0, n, 2)
            delta = rng.uniform(-0.15, 0.15) * a.mean()
            cand = a.copy()
            cand[i] += delta
            cand[j] -= delta
        elif r < 0.9:
            # Full Gaussian perturbation
            z = np.log(np.maximum(a, 1e-10))
            z += rng.normal(0, 0.06, n)
            cand = np.exp(z)
            cand /= cand.sum()
        else:
            # LP-guided step
            M = float(_auto_corr(a).max())
            g = _lp_solve(a, M, timeout=1.0)
            if g is None:
                continue
            g /= g.sum()
            t = rng.uniform(0.1, 0.4)
            cand = (1 - t) * a + t * g

        cand = np.maximum(cand, 0.0)
        cs = cand.sum()
        if cs <= 0:
            continue
        cand = cand / cs
        M_cand = float(_auto_corr(cand).max())

        if M_cand < best_M - 1e-13:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                a = a * np.exp(rng.normal(0, 0.2, n))
                a = np.maximum(a, 1e-8)
                a = a / a.sum()
                no_improve = 0

    return best_a


def _try_length(a_best, n_try, stop_time):
    """Try a different sequence length by resampling."""
    n_orig = a_best.size
    x_old = np.linspace(0, 1, n_orig)
    x_new = np.linspace(0, 1, n_try)
    init = np.interp(x_new, x_old, a_best)
    init = np.maximum(init, 1e-8)
    init = init / init.sum()
    return _lp_iterate(init, stop_time=stop_time, max_iters=800)
