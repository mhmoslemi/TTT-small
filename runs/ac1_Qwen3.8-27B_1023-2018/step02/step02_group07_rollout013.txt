import time

from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _conv(a, n, fs):
    F = rfft(a, fs)
    return irfft(F * F, fs)[:2 * n - 1]


def _score_norm(a, n, fs):
    """Score assuming sum(a)=1: 2*n*max_b"""
    b = _conv(a, n, fs)
    m = float(b.max())
    if not np.isfinite(m) or m <= 0:
        return math.inf
    return 2.0 * n * m


def _softmax(x, T=1.0):
    x = np.asarray(x, dtype=np.float64).ravel()
    x = x - x.max()
    e = np.exp(np.clip(x / T, -50, 50))
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0:
        return np.full(x.size, 1.0 / x.size)
    return e / s


def _cc(q, a, n, fs2):
    """Cross-correlation: return c[n-1:2n-1] where c = irfft(rfft(a[::-1],fs2)*rfft(q,fs2))"""
    ar = a[::-1]
    c = irfft(rfft(ar, fs2) * rfft(q, fs2), fs2)
    return c[n - 1:2 * n - 1]


def _lp_flatten(f, M, n):
    """Solve LP: max sum(b) s.t. b>=0, (f*b)[k]<=M for k=0..2n-2. Return b/sum(b) or None."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f[j]
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True, 'time_limit': 2})

        s = float(x.sum())

            return x / s
    return None

    return seeds


def _adam_optimize(a, n, fs_c, fs2, rng, end_time, hard_deadline, init_z=None):
    """Adam optimizer in log-space to minimize 2*n*max_b (sum(a)=1)."""
    if init_z is not None:
        z = init_z.copy()
    else:
        z = np.log(np.maximum(a, 1e-8))
        z = z - z.mean()

    best_z = z.copy()
    best_score = _score_norm(_softmax(z), n, fs_c)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t_adam = 0
    tau = 0.4 * _conv(_softmax(z), n, fs_c).max()
    lr = 0.5
    stuck = 0
    it = 0

    while time.monotonic() < min(end_time, hard_deadline) - 0.5:
        a_c = _softmax(np.clip(z, -30, 30))
        b = _conv(a_c, n, fs_c)
        bmax = float(b.max())
        if not np.isfinite(bmax) or bmax <= 0:
            break
        cur_score = 2.0 * n * bmax

        # Smooth max gradient
        tau = max(1e-9, 0.35 * bmax)
        p = _softmax(b, tau)
        # Gradient of smooth max: d/d a_j [sum_k p_k * b_k]
        g = 2.0 * _cc(p, a_c, n, fs2)
        # Project to simplex (log-param): grad_log = a * (g - sum(a*g))
        grad = a_c * (g - float(np.dot(a_c, g)))

        gn = float(np.abs(grad).max())
        if not np.isfinite(gn) or gn < 1e-15:
            stuck += 1
            if stuck > 30:
                z = best_z + rng.normal(0, 0.5, n)
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)
                t_adam = 0
                lr = 0.45
                stuck = 0
            continue

        grad = grad / (gn + 1e-30)
        t_adam += 1
        m_buf = 0.9 * m_buf + 0.1 * grad
        v_buf = 0.999 * v_buf + 0.001 * grad * grad
        mh = m_buf / (1 - 0.9 ** t_adam)
        vh = v_buf / (1 - 0.999 ** t_adam)
        upd = mh / (np.sqrt(vh) + 1e-8)
        um = float(np.abs(upd).max())
        if not np.isfinite(um) or um <= 0:
            continue
        upd = upd / (um + 1e-30)

        z = z - lr * upd
        z = z - z.mean()
        z = np.clip(z, -30, 30)

        a2 = _softmax(z)
        b2 = _conv(a2, n, fs_c)
        score2 = 2.0 * n * max(float(b2.max()), 0)

        if np.isfinite(score2) and score2 < best_score - 1e-11:
            best_score = score2
            best_z = z.copy()
            stuck = 0
        else:
            stuck += 1

        it += 1
        if it % 50 == 0:
            lr = max(0.05, lr * 0.9)
        if stuck > 20:
            lr = max(0.05, lr * 0.9)
        if stuck > 120:
            z = best_z + rng.normal(0, 0.4, n)
            z = z - z.mean()
            m_buf = np.zeros(n)
            v_buf = np.zeros(n)
            t_adam = 0
            lr = 0.45
            stuck = 0

    a_out = _softmax(np.clip(best_z, -30, 30))
    s_out = _score_norm(a_out, n, fs_c)
    return a_out, s_out, best_z


def _local_polish(a, n, fs_c, rng, end_time, hard_deadline):
    """Local search polish."""
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 1e-9)
    a = a / a.sum()
    best = a.copy()
    bs = _score_norm(best, n, fs_c)
    sigma = 0.15
    stuck = 0

    while time.monotonic() < min(end_time, hard_deadline) - 0.3:
        z = np.log(np.maximum(a, 1e-12))
        r = rng.random()
        if r < 0.3:
            zc = z + rng.normal(0, sigma, n)
        elif r < 0.45:
            i = int(rng.integers(0, n))
            w = max(1, int(rng.integers(1, max(2, n // 20))))
            i2 = min(n, i + w)
            zc = z.copy()
            zc[i:i2] += rng.normal(0, sigma * 0.5, i2 - i)
        elif r < 0.6:
            zv = z - z.mean()
            zc = zv * float(rng.uniform(0.5, 1.8))
        elif r < 0.75:
            a2 = a.copy()
            i = int(rng.integers(0, n))
            w = max(1, int(rng.integers(1, max(2, n // 25))))
            i2 = min(n, i + w)
            a2[i:i2] *= float(np.exp(rng.normal(0, 0.3)))
            zc = np.log(np.maximum(a2, 1e-12))
        else:
            k = float(rng.uniform(0.3, 4.0))
            ph = float(rng.uniform(0, 2 * math.pi))
            zc = z + sigma * np.sin(k * np.pi * np.arange(n) / n + ph)

        zc = zc - zc.mean()
        a2 = _softmax(zc)
        s2 = _score_norm(a2, n, fs_c)
        if np.isfinite(s2) and s2 < bs - 1e-12:
            best = a2.copy()
            bs = s2
            a = a2.copy()
            stuck = 0
            sigma = max(0.03, sigma * 0.95)
        else:
            stuck += 1
            sigma = min(0.6, sigma * 1.04)
            if stuck > 50:
                a = best + rng.normal(0, 0.1, n)
                a = np.maximum(a, 1e-9)
                a = a / a.sum()
                stuck = 0
                sigma = 0.2

    best = np.maximum(best, 1e-9)
    best = best / best.sum()
    return best, _score_norm(best, n, fs_c)


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    total = max(100.0, float(budget_s))
    hard_deadline = t0 + total - 5.0

    # Get parent sequence
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            if p.size > 0:
                p = np.clip(p, 0.0, 1000.0)
                if np.all(np.isfinite(p)) and float(p.sum()) > 0:
                    parent = p / p.sum()
    except Exception:
        pass

    best_a = None
    best_score = math.inf

    def _try_update(a, s):
        nonlocal best_a, best_score
        try:
            a = np.asarray(a, dtype=np.float64).ravel()
            if a.size == 0 or not np.all(np.isfinite(a)):
                return
            a = np.clip(a, 0.0, 1000.0)
            sm = float(a.sum())
            if sm < 1e-6:
                return
            a = a / sm
            if np.isfinite(s) and s < best_score - 1e-13:
                best_a = a.copy()
                best_score = float(s)
        except Exception:
            pass

    # Initialize with parent
    if parent is not None and parent.size > 0:
        pa = parent.copy()
        fs_init = next_fast_len(2 * parent.size - 1)
        s_init = _score_norm(pa, parent.size, fs_init)
        _try_update(pa, s_init)

    if best_a is None:
        best_a = np.ones(32) / 32.0
        best_score = 2.0

    rng = np.random.default_rng(seed)
    n_main = parent.size if parent is not None else 192
    fs_main = next_fast_len(2 * n_main - 1)
    fs2_main = next_fast_len(3 * n_main - 2)

    # === Phase 1: LP-based iteration on main size (55% budget) ===
    lp_end = t0 + 0.55 * total
    try:
        a = best_a.copy() if best_a.size == n_main else np.ones(n_main) / n_main
        t_blend = 0.35
        no_improve = 0
        for it in range(50000):
            if time.monotonic() > lp_end - 2:
                break
            b = _conv(a, n_main, fs_main)
            M = float(b.max())
            if not np.isfinite(M) or M <= 1e-12:
                break
            try:
                b_opt = _lp_flatten(a, M, n_main)
            except Exception:
                break
            if b_opt is None:
                break
            a_new = (1 - t_blend) * a + t_blend * b_opt
            a_new = np.maximum(a_new, 1e-10)
            a_new = a_new / a_new.sum()
            s_new = _score_norm(a_new, n_main, fs_main)
            if np.isfinite(s_new):
                if s_new < best_score - 1e-13:
                    _try_update(a_new, s_new)
                    no_improve = 0
                else:
                    no_improve += 1
            a = a_new
            t_blend = max(0.1, t_blend * 0.997)
            if no_improve > 200:
                break
    except Exception:
        pass

    # === Phase 2: Adam gradient optimization (25% budget) ===
    grad_end = t0 + 0.80 * total
    try:
        # Multiple starting points
        if best_a.size == n_main:
            a_start = best_a.copy()
        else:
            a_start = np.ones(n_main) / n_main

        z_init = np.log(np.maximum(a_start, 1e-8))
        z_init = z_init - z_init.mean()

        a_g, s_g, z_g = _adam_optimize(
            a_start, n_main, fs_main, fs2_main, rng,
            grad_end - 30, hard_deadline, init_z=z_init
        )
        _try_update(a_g, s_g)

        # Second Adam run from perturbed state
        if time.monotonic() < grad_end - 10:
            z_pert = z_g + rng.normal(0, 0.3, n_main)
            z_pert = z_pert - z_pert.mean()
            a_g2, s_g2, z_g2 = _adam_optimize(
                _softmax(z_pert), n_main, fs_main, fs2_main, rng,
                grad_end, hard_deadline, init_z=z_pert
            )
            _try_update(a_g2, s_g2)
    except Exception:
        pass

    # === Phase 3: Local search polish (15% budget) ===
    polish_end = min(hard_deadline, t0 + 0.95 * total)
    try:
        if best_a.size == n_main:
            a_pol = _local_polish(best_a, n_main, fs_main, rng, polish_end, hard_deadline)
            _try_update(a_pol[0], a_pol[1])
    except Exception:
        pass

    # === Phase 4: Try other sizes if time remains (10% budget) ===
    other_end = hard_deadline
    try:
        for n_try in [256, 320, 384, 512]:
            if time.monotonic() > other_end - 30:
                break
            fs_t = next_fast_len(2 * n_try - 1)
            fs2_t = next_fast_len(3 * n_try - 2)
            # Interpolate parent to new size
            if parent is not None:
                idx = (np.arange(n_try) * parent.size / n_try).astype(int)
                idx = np.clip(idx, 0, parent.size - 1)
                a_t = parent[idx].copy()
            else:
                a_t = np.ones(n_try) / n_try
            a_t = np.maximum(a_t, 1e-6)
            a_t = a_t / a_t.sum()
            # Quick Adam on this size
            t_sub = min(other_end - 15, time.monotonic() + 60)
            a_r, s_r, _ = _adam_optimize(
                a_t, n_try, fs_t, fs2_t,
                np.random.default_rng(seed + n_try), t_sub, other_end
            )
            _try_update(a_r, s_r)
    except Exception:
        pass

    # === Final validation ===
    try:
        cand = [float(v) for v in best_a]
        fs_score = float(evaluate_sequence(cand))
        if not np.isfinite(fs_score) or fs_score > best_score + 0.1:
            raise ValueError("validation failed")
    except Exception:
        # Fallback: constant sequence
        cand = [1.0]
        try:
            fs_score = float(evaluate_sequence(cand))
        except Exception:
            pass

    return cand
