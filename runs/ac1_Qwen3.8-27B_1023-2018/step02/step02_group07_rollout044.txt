            pert = 1.0 + 0.1 * np.sin(2.0 * np.pi * rng.integers(1, max(2, n // 4)) *
                                       np.arange(n) / n + rng.uniform(0, 2 * np.pi))
            a = np.maximum(pert, 1e-10)
            s = _uniform_score(1)  # placeholder, use evaluator directly
            try:
                s = float(evaluate_sequence([float(v) for v in a]))
            except Exception:
                s = math.inf
            _try_update(a, s)

    # 4. Final: ensure we have a valid candidate
    if best_a is None or best_a.size == 0:
        best_a = np.ones(192, dtype=np.float64)

    # 5. Final exact validation with the authoritative evaluator
    try:
        cand = [float(v) for v in best_a]
    except Exception:
        cand = [1.0]

    try:
        fs = float(evaluate_sequence(cand))
    except Exception:
        fs = math.inf

    if not np.isfinite(fs):
        return [1.0]

    return cand
