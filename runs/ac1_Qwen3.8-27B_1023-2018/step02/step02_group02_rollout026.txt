                if time.monotonic() > deadline - 60.0:
                    break
                c = conv_full(a)
                M = float(c.max())
                sc2 = 2.0 * n * M
                if sc2 < best_score - 1e-14:
                    best_score = sc2
                    best_f = a.copy()

                tau2 = max(0.02 * M, 1e-12)
                x2 = (c - M) / tau2
                ex2 = np.exp(np.clip(x2, -70, 0))
                s_ex2 = float(ex2.sum())
                if s_ex2 < 1e-30:
                    p2 = np.zeros(2 * n - 1)
                    p2[int(np.argmax(c))] = 1.0
                else:
                    p2 = ex2 / s_ex2

                p2_rev = p2[::-1]
                A2 = rfft(a, total_grad_fsize)
                Pr2 = rfft(p2_rev, total_grad_fsize)
                ap2 = irfft(A2 * Pr2, total_grad_fsize)
                grad2 = ap2[n - 1:2 * n - 1][::-1].copy()

                step2 = grad2 - grad2.mean()
                sn2 = float(np.linalg.norm(step2))
                if sn2 < 1e-14:
                    break

                improved2 = False
                alpha2 = lr2
                for _ in range(10):
                    cand2 = np.maximum(a - alpha2 * step2 / sn2, 0.0)
                    cs2 = float(cand2.sum())
                    if cs2 < 1e-8:
                        alpha2 *= 0.5
                        continue
                    cand2 /= cs2
                    m2 = float(peak_conv(cand2))
                    if m2 < M - 1e-15:
                        a = cand2
                        M = m2
                        improved2 = True
                        break
                    alpha2 *= 0.5
                if not improved2:
                    break

    # =========================================================
    # PHASE 4: Stochastic local search / fine-tuning
    # =========================================================
    if time.monotonic() < deadline - 15.0:
        a = best_f.copy()
        sc = best_score
        rng2 = np.random.default_rng(seed + 7777)
        sigma = 0.04

        for iters in range(30000):
            if time.monotonic() > deadline - 15.0:
                break

            r = rng2.random()
            if r < 0.25:
                # Log-normal multiplicative perturbation
                z = np.log(np.maximum(a, 1e-12))
                z += rng2.normal(0.0, sigma, n)
                z -= z.max()
                cand = np.exp(z)
                cand = np.maximum(cand, 1e-12)
                cand /= cand.sum()
            elif r < 0.45:
                # Gaussian additive (on log scale)
                z = np.log(np.maximum(a, 1e-12))
                z += rng2.normal(0.0, sigma * 1.5, n)
                z -= z.max()
                cand = np.exp(z)
                cand = np.maximum(cand, 1e-12)
                cand /= cand.sum()
            elif r < 0.65:
                # Local bump
                cand = a.copy()
                k = int(rng2.integers(0, n))
                w = max(2, n // 40)
                lo, hi = max(0, k - w), min(n, k + w + 1)
                scale = math.exp(rng2.normal(0, 0.08))
                cand[lo:hi] *= scale
                cand = np.maximum(cand, 0.0)
                cs = cand.sum()
                if cs > 1e-8:
                    cand /= cs
                else:
                    continue
            elif r < 0.80:
                # Two-point swap with noise
                cand = a.copy()
                k1 = int(rng2.integers(0, n))
                k2 = int(rng2.integers(0, n))
                spread = rng2.uniform(0.005, 0.03) * float(a.max())
                cand[k1] += spread
                cand[k2] -= spread
                cand = np.maximum(cand, 0.0)
                cs = cand.sum()
                if cs > 1e-8:
                    cand /= cs
                else:
                    continue
            elif r < 0.90:
                # Blend with reversed
                t_rev = rng2.uniform(0.02, 0.15)
                cand = (1 - t_rev) * a + t_rev * a[::-1]
                cand = np.maximum(cand, 1e-12)
                cand /= cand.sum()
            else:
                # Smooth neighbor
                kernel = np.array([0.1, 0.8, 0.1])
                cand = np.convolve(a, kernel, mode='same')
                cand = np.maximum(cand, 1e-12)
                cand /= cand.sum()

            cand_sc = score_of(cand)
            if cand_sc < sc - 1e-14:
                a = cand.copy()
                sc = cand_sc
                sigma = max(0.008, sigma * 0.985)
            else:
                sigma = min(0.25, sigma * 1.015)

        if sc < best_score - 1e-14:
            best_score = sc
            best_f = a.copy()

    # =========================================================
    # PHASE 5: Very fine perturbation (final polish)
    # =========================================================
    if time.monotonic() < deadline - 5.0:
        a = best_f.copy()
        sc = best_score
        rng3 = np.random.default_rng(seed + 31313)
        for _ in range(20000):
            if time.monotonic() > deadline - 5.0:
                break
            # Tiny random perturbation
            idx = rng3.integers(0, n, size=rng3.integers(1, 5))
            cand = a.copy()
            for ii in idx:
                cand[ii] *= math.exp(rng3.normal(0, 0.005))
            cand = np.maximum(cand, 0.0)
            cs = cand.sum()
            if cs > 1e-8:
                cand /= cs
            else:
                continue
            cand_sc = score_of(cand)
            if cand_sc < sc - 1e-15:
                a = cand.copy()
                sc = cand_sc

    if sc < best_score - 1e-15:
        best_score = sc
        best_f = a.copy()

    # =========================================================
    # Final validation with exact evaluator
    # =========================================================
    best_f = np.clip(best_f, 0.0, 1000.0)
    best_f[~np.isfinite(best_f)] = 0.0
    if best_f.sum() < 0.01:
        best_f = np.ones(n) / n
        best_f[0] = 1.0

    final_list = [float(x) for x in best_f]
    try:
        fs = float(evaluate_sequence(final_list))
    except Exception:
        fs = float('inf')

    if not np.isfinite(fs):
        # Fallback: return the parent (which is valid)
        fallback = [float(x) for x in np.clip(parent, 0, 1000)]
        try:
            if evaluate_sequence(fallback) == float('inf'):
                fallback = [1.0 / n] * n
        except Exception:
            fallback = [1.0]
        return fallback

    return final_list
