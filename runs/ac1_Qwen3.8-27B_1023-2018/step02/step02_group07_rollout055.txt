import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len


def _score_and_grad(a, tau):
    """Compute score and gradient in z-space (a=exp(z) normalized)."""
    n = len(a)
    fs = next_fast_len(2 * n - 1)
    F = rfft(a, fs)
    c = irfft(F * F, fs)[:2 * n - 1]
    S = a.sum()
    M = c.max()
    score = 2.0 * n * M / (S * S)
    
    # Soft argmax weights
    c_max = c.max()
    exp_c = np.exp(np.clip((c - c_max) / tau, -100, 0))
    Z = exp_c.sum()
    if Z < 1e-300:
        p = np.zeros_like(c)
        p[int(np.argmax(c))] = 1.0
    else:
        p = exp_c / Z
    
    # Soft max value (for gradient computation)
    M_soft = c_max + tau * math.log(max(Z, 1e-300))
    
    # h[j] = sum_m a[m] * p[j+m] for j=0,...,n-1
    # h = conv(a[::-1], p)[n-1 : 2*n-1]
    fs2 = next_fast_len(3 * n - 2)
    h = irfft(rfft(a[::-1], fs2) * rfft(p, fs2), fs2)[n - 1:2 * n - 1]
    
    # Gradient in z-space: 4n*a*(h - M*a/S) / S^2 ... wait
    # score = 2n*M_soft/S^2
    # d(score)/d(a_j) = 2n/S^2 * d(M_soft)/d(a_j) - 2n*M_soft*2*(1/S^2)*(1/S)*a_j
    # = 2n/S^2 * 2*h[j] - 4n*M_soft*a_j/S^3
    # = 4n/S^2 * (h[j] - M_soft*a_j/S)
    # d(score)/d(z_j) = d(score)/d(a_j) * (a_j - a_j*S_j... )
    
    # Wait, a depends on z through a_i = exp(z_i)/S where S = sum(exp(z)).
    # da_i/dz_j = a_i*(delta_ij - a_j)
    # d(score)/dz_j = sum_i d(score)/da_i * da_i/dz_j
    # = sum_i [4n/S^2 * (h_i - M_soft*a_i/S)] * a_i*(delta_ij - a_j)
    # = 4n/S^2 * [a_j*(h_j - M_soft*a_j/S) - a_j*sum_i(h_i*a_i - M_soft*a_i^2/S)]
    # = 4n/S^2 * a_j * [(h_j - M_soft*a_j/S) - (sum_i h_i*a_i - M_soft*sum_i a_i^2/S)]
    
    # For simplicity (and since the score is scale-invariant), let's work with
    # unnormalized a = exp(z) and score = 2n*max(conv(a,a))/(sum a)^2.
    
    # d(score)/d(a_j) = 4n/S^2 * (h[j] - M_soft*a_j/S)
    # d(score)/d(z_j) = a_j * d(score)/d(a_j) = 4n*a_j/S^2 * (h[j] - M_soft*a_j/S)
    
    grad_z = 4.0 * n * a / (S * S) * (h - M_soft * a / S)
    return grad_z, score, c, p, M_soft, S

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
