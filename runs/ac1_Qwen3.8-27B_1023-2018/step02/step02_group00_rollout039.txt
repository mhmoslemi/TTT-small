    c = -np.ones(n)

    g = np.clip(g, 0, None)

    if gs < 1e-15:

def _polish(a, stop_time, max_rounds=100):

    best = np.clip(np.asarray(a, dtype=np.float64).ravel().copy(), 0, None)
    fs = best.sum()

    best = best / fs
    best_sc = _score(best)
    rng = np.random.default_rng(999)
    no_imp = 0

    for rnd in range(max_rounds):
        if time.monotonic() > stop_time:

        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time:
                break
            orig = best[i]
            best_sc_i = best_sc
            best_val = orig
            # Try multiplicative changes
            for mlt in (0.1, 0.25, 0.5, 0.75, 1.33, 1.5, 2.0, 4.0, 0.0):

                    best[i] = orig
                    continue

                if sc < best_sc_i - 1e-14:
                    best_sc_i = sc
                    best_val = candidate
            best[i] = best_val
            if best_sc_i < best_sc - 1e-14:
                best_sc = best_sc_i
                improved = True

        # Try pairwise moves
        for _ in range(40):
            if time.monotonic() > stop_time:
                break
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue

                    sc = _score(best)

            grad_tan = grad - mean_grad
            f_new = f - 0.01 * grad_tan
            f_new = np.clip(f_new, 0, None)
            fs2 = f_new.sum()
            if fs2 > 1e-10:
                f_new = f_new / fs2
                sc = _score(f_new)
                if sc < best_score - 1e-12:
                    f = f_new
                    best_f = f.copy()
                    best_score = sc
                    stall = 0
                else:
                    stall += 2
            else:
                stall += 2
            if stall > 300:
                # Larger perturbation
                f = np.clip(f + rng.normal(0, 0.05, n), 0, None)
                fs2 = f.sum()
                if fs2 > 1e-10:
                    f = f / fs2
                else:
                    f = best_f.copy()
                stall = 100
            continue

        dist = float(np.linalg.norm(g_dir - f))
        if dist < 1e-8:
            # At fixed point of LP: use subgradient to escape
            grad, _ = _subgradient_step(f, 100.0, n)
            mean_grad = grad.mean()
            grad_tan = grad - mean_grad
            gnorm = np.linalg.norm(grad_tan)
            if gnorm > 1e-10:
                step_dir = -grad_tan / gnorm
                # Project onto feasible direction
                step_dir -= step_dir.mean()
                for step_size in (0.05, 0.02, 0.01, 0.005):
                    f_new = f + step_size * step_dir
                    f_new = np.clip(f_new, 0, None)
                    fs2 = f_new.sum()
                    if fs2 > 1e-10:
                        f_new = f_new / fs2
                        sc = _score(f_new)
                        if sc < best_score - 1e-12:
                            f = f_new
                            best_f = f.copy()
                            best_score = sc
                            stall = 0
                            break
            stall += 3
            if stall > 300:
                f = np.clip(f + rng.normal(0, 0.06, n), 0, None)
                fs2 = f.sum()
                if fs2 > 1e-10:
                    f = f / fs2
                else:
                    f = best_f.copy()
                t = 0.35
                stall = 100
            continue

        # Line search for step size
        improved = False
        for _ in range(20):
            f_new = (1.0 - t) * f + t * g_dir
            f_new = np.clip(f_new, 0, None)
            fs_new = float(f_new.sum())
            if fs_new < 1e-10:
                t = max(0.01, t * 0.5)
                continue
            sc = _score(f_new)
            if sc < best_score - 1e-12:
                f = f_new
                best_f = f.copy()
                best_score = sc
                improved = True
                t = min(0.7, t * 1.05)
                stall = 0
                break
            else:
                t = max(0.01, t * 0.6)

        if not improved:
            stall += 1
            if stall > 250:
                f = np.clip(f + rng.normal(0, 0.07, n), 0, None)
                fs2 = f.sum()
                if fs2 > 1e-10:
                    f = f / fs2
                else:
                    f = best_f.copy()
                t = 0.35
                stall = 80

    return best_f, best_score


def _polish(a, stop_time, max_rounds=200):
    n = len(a)
    best = np.clip(np.asarray(a, dtype=np.float64).ravel().copy(), 0, None)
    fs = best.sum()
    if fs < 1e-10:
        return best, math.inf
    best = best / fs
    best_sc = _score(best)
    rng = np.random.default_rng(777)
    no_imp = 0

    for rnd in range(max_rounds):
        if time.monotonic() > stop_time:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time:
                break
            orig = best[i]
            best_sc_i = best_sc
            best_val = orig
            for mlt in (0.0, 0.1, 0.2, 0.35, 0.5, 0.7, 1.3, 1.5, 2.0, 3.0, 5.0):
                if orig < 1e-14:
                    candidate = 0.1 / n if mlt > 1.0 else 0.0
                else:
                    candidate = orig * mlt
                best[i] = candidate
                s = best.sum()
                if s < 0.01:
                    best[i] = orig
                    continue
                sc = _score(best)
                if sc < best_sc_i - 1e-14:
                    best_sc_i = sc
                    best_val = candidate
            best[i] = best_val
            if best_sc_i < best_sc - 1e-14:
                best_sc = best_sc_i
                improved = True

        # Pairwise moves
        for _ in range(50):
            if time.monotonic() > stop_time:
                break
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            for mi in (0.5, 0.8, 1.25, 1.6):
                for mj in (0.5, 0.8, 1.25, 1.6):
                    best[i] = oi * mi if oi > 1e-14 else 0.1 / n
                    best[j] = oj * mj if oj > 1e-14 else 0.1 / n
                    if best.sum() < 0.01:
                        best[i], best[j] = oi, oj
                        continue
                    sc = _score(best)
                    if sc < best_sc - 1e-14:
                        best_sc = sc
                        improved = True
            best[i], best[j] = oi, oj

        if not improved:
            no_imp += 1
            if no_imp >= 6:
                break
        else:
            no_imp = 0

    return best, best_sc
