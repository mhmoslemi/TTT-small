import numpy as np
import time
import math
from scipy.optimize import linprog

TARGET = 1.50274349

def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 0.0, 1000.0)
    n = a.size
    if n == 0:
        return float('inf')
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    b = np.convolve(a, a)
    mb = float(b.max())
    if not math.isfinite(mb):
        return float('inf')
    return 2.0 * n * mb / (s * s)

def _build_A(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A

def _solve_lp(a, M, tl=3.0):
    n = len(a)
    A = _build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'time_limit': tl, 'presolve': True})
    except Exception:
        return None
    if res is not None and getattr(res, 'x', None) is not None:
        g = np.clip(res.x, 0.0, None)
        sg = float(g.sum())
        if sg > 1e-12 and math.isfinite(sg):
            return g / sg
    return None

def _solve_lp_v2(a, b_conv, M, tl=3.0):

    A = _build_A(a)
    c = -np.ones(n)

def _subgrad(a, active):
    n = len(a)
    grad = np.zeros(n)
    for k in active:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if hi < lo:
            continue
        lo2 = k - hi
        hi2 = k - lo
        grad[lo:hi + 1] += 2.0 * a[lo2:hi2 + 1][::-1]
    return grad
