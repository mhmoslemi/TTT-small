    """Pre-build row/col indices for convolution matrix (a fixed structure)."""
    rows = []

    """LP: max sum(g) s.t. convolve(a,g) <= M, g >= 0"""

    vals = np.empty(len(rows), dtype=np.float64)

        jj = k - cols[i]
        vals[i] = a[jj]

def _gradient_active(a, M, rel_tol=0.02):
    """Gradient of max autocorrelation w.r.t. a at active set."""
    n = len(a)
    b = np.convolve(a, a)
    thresh = M - rel_tol * M
    active = np.where(b >= thresh)[0]
    if len(active) == 0:
        return None, 0
    grad = np.zeros(n)

        if dk == 0:
            grad += 2.0 * a
        else:
            grad[:n - dk] += a[dk:]
            grad[dk:] += a[:n - dk]
    grad /= len(active)
    return grad, len(active)

    if not math.isfinite(fin_sc) or fin_sc > best[0] + 0.01:
        final = np.ones(16)

    return [float(x) for x in final]
