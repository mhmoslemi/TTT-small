    b = irfft(rfft(a, fs)**2, fs)

            v_stat[:] = 0
            t = 0
            lr = 0.3
            tau_frac = 0.3
            stuck = 0
    
    return best_z, best_val

def _polish(a, end_time):
    """Local search polish."""
    n = a.size
    fs = next_fast_len(2*n-1)
    best = a.copy()
    bs = _score_norm(best, fs)
    rng = np.random.default_rng(424242)
    sigma = 0.15
    stuck = 0
    
    while time.monotonic() < end_time:
        z = np.log(np.maximum(best, 1e-15))
        r = rng.random()
        
        if r < 0.3:
            zc = z + rng.normal(0, sigma, n)
        elif r < 0.5:
            i = rng.integers(0, n)
            w = max(1, min(n, rng.integers(1, max(2, n//20+1))))
            i2 = min(n, i+w)
            zc = z.copy()
            zc[i:i2] += rng.normal(0, sigma*0.5, i2-i)
        elif r < 0.7:
            # Scale a segment
            i = rng.integers(0, n)
            w = max(1, min(n, rng.integers(1, max(2, n//15+1))))
            i2 = min(n, i+w)
            zc = z.copy()
            zc[i:i2] += rng.normal(0, 0.3)
        elif r < 0.85:
            # Global tilt
            t = np.arange(n)/max(1, n-1)
            zc = z + rng.uniform(-sigma, sigma) * (2*t - 1)
        else:
            # Random frequency perturbation
            f = rng.integers(1, max(2, n//3))
            amp = rng.uniform(0.1, 0.5) * sigma
            ph = rng.uniform(0, 2*np.pi)
            zc = z + amp * np.sin(2*np.pi*f*np.arange(n)/n + ph)
        
        zc -= zc.mean()
        a2 = _sm(zc)
        s2 = _score_norm(a2, fs)
        
        if s2 < bs - 1e-11:
            best = a2.copy()
            bs = s2
            stuck = 0
            sigma = max(0.03, sigma * 0.9)
        else:
            stuck += 1
            sigma = min(0.8, sigma * 1.03)
            if stuck > 80:
                stuck = 0
                sigma = 0.3
    
    return best, bs
