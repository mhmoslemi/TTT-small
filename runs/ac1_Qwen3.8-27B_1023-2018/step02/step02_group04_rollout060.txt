def _score(a):
    a = np.clip(np.asarray(a, dtype=np.float64).ravel(), 0, 1000)
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    b = np.convolve(a, a)
    return float(2.0 * n * float(b.max()) / (s * s))


def _norm(a):
    a = np.clip(np.asarray(a, dtype=np.float64).ravel(), 0, None)
    s = float(a.sum())
    if s < 1e-300:
        return np.ones(a.size, dtype=np.float64)
    return a / s


def _lp_dir(a, M=None, time_limit=5.0):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0. Returns normalized b or None."""
    n = a.size
    if M is None:
        M = float(np.convolve(a, a).max())
    if not math.isfinite(M) or M < 1e-300:
        return None
    # Build A: (2n-1) x n, A[k,j] = a[j] if 0 <= k-j < n
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = a
    b_ub = np.full(2 * n - 1, M * (1.0 + 1e-9), dtype=np.float64)
    c_obj = -np.ones(n, dtype=np.float64)
    try:
        r = linprog(c_obj, A_ub=A, b_ub=b_ub,
                    bounds=[(0.0, None)] * n,
                    method='highs',
                    options={'presolve': True, 'time_limit': time_limit})
        if r is not None and r.success and r.x is not None:
            b = np.nan_to_num(r.x, nan=0.0, posinf=0.0, neginf=0.0)
            b = np.clip(b, 0, None)
            sb = float(b.sum())
            if sb > 1e-300:
                return b / sb
    except Exception:
        pass
    return None


def _line_search(a, b, loc_s):
    """Geometric line search. Returns (best_a, best_s, best_t)."""
    best_t, best_s = 0.0, loc_s

    if best_t == 0.0:

    return _norm((1.0 - best_t) * a + best_t * b), best_s, best_t

def _grad_refine(a, stop_t, best, rng, max_iter=5000):
    """Subgradient refinement for max of convolution."""
    a = _norm(a)
    n = a.size
    loc_s = _score(a)

        # Actually: dc[k]/da[j] = a[k-j] if 0<=k-j<n

            if 0 <= idx < n:

            idx = kstar - j
            if 0 <= idx < n:
                grad[j] += a[idx]
        # Projected gradient: minimize max c[k] s.t. sum(a)=1, a>=0
        gn = float(np.linalg.norm(grad))

            break

def _gen_starts(n, rng, ref, count=12):
    """Generate diverse starting sequences of length n."""
    x = np.linspace(0, 1, n)
    starts = [np.ones(n, dtype=np.float64)]
    # Interpolate reference
    if ref is not None and len(ref) >= 2:
        r = _norm(ref)
        xp = np.linspace(0, 1, len(r))
        g = np.interp(x, xp, r)
        starts.append(_norm(g))
        starts.append(_norm(g * np.exp(0.1 * rng.standard_normal(n))))
        starts.append(_norm(g * np.exp(0.3 * rng.standard_normal(n))))
    # Exponentials
    for c in (0.5, 1.0, 2.0, 3.0, -0.5, -1.0, -2.0):
        starts.append(_norm(np.exp(c * (x - 0.5))))
    # Sine modulations
    for k in (1, 2, 3, 4):
        amp = 0.5
        starts.append(_norm(1.0 + amp * np.sin(2 * np.pi * k * x)))
        starts.append(_norm(1.0 - amp * np.sin(2 * np.pi * k * x)))
    # Random Dirichlet

        try:
            starts.append(_norm(rng.dirichlet(np.full(n, alpha))))
        except Exception:
            pass
    # Sort by score
    starts.sort(key=_score)
    return starts[:count]
