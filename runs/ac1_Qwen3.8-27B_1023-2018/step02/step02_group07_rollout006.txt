```
def _resample(arr,n,log=False):
   arr=np.asarray(arr,float).ravel()
   P=arr.size
   if n<=0: return np.ones(1)
   if P==0: return np.ones(n)
   if P==n: return arr.copy()
   x_old=(np.arange(P)+0.5)/P
   x_new=(np.arange(n)+0.5)/n
   if P==1:
       vals=np.full(n, float(arr[0]))
   else:
       if log:
          ya=np.log(np.maximum(arr,1e-300))
          vals=np.exp(np.interp(x_new,x_old,ya))
       else:
          vals=np.interp(x_new,x_old,np.maximum(arr,0.0))
   return vals
```

```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len

try:
    from scipy.optimize import linprog
except Exception:
    linprog = None


def _softmax(x):
    x = np.asarray(x, dtype=np.float64).ravel()
    if x.size == 0:
        return np.array([1.0])
    x = x - float(np.max(x))
    e = np.exp(np.clip(x, -50.0, 50.0))
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full(x.size, 1.0 / x.size, dtype=np.float64)
    return e / s


def _normalize(a, floor=0.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, float(floor))
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full(a.size, 1.0 / a.size, dtype=np.float64)
    return a / s


def _resample(arr, n, log=False):
    arr = np.asarray(arr, dtype=np.float64).ravel()
    if n <= 0:
        return np.ones(1, dtype=np.float64)
    P = arr.size
    if P == 0:
        return np.ones(n, dtype=np.float64)
    if P == n:
        return arr.copy()
    x_old = (np.arange(P) + 0.5) / float(P)
    x_new = (np.arange(n) + 0.5) / float(n)
    if P == 1:
        vals = np.full(n, float(arr[0]), dtype=np.float64)
    else:
        if log:
            ya = np.log(np.maximum(arr, 1e-300))
            vals = np.exp(np.interp(x_new, x_old, ya))
        else:
            vals = np.interp(x_new, x_old, np.maximum(arr, 0.0))
    return np.asarray(vals, dtype=np.float64)


def _init_variant(n, kind, rng):
    t = (np.arange(n) + 0.5) / float(n)
    x = 2.0 * t - 1.0
    k = int(kind) % 9
    if k == 0:
        a = np.ones(n, dtype=np.float64)
    elif k == 1:
        a = 1.0 - np.abs(x)
    elif k == 2:
        a = 1.0 - x * x
    elif k == 3:
        a = np.exp(-2.5 * np.abs(x))
    elif k == 4:
        a = 0.5 + 0.5 * np.sin(np.pi * t)
    elif k == 5:
        a = np.exp(-3.0 * t)
    elif k == 6:
        a = np.exp(-3.0 * (1.0 - t))
    elif k == 7:
        a = np.abs(np.sin(np.pi * t)) + 0.2
    else:
        a = np.abs(rng.normal(0.0, 1.0, n)) + 0.2
    a = np.maximum(a, 1e-3)
    a = a + 0.05 * np.abs(rng.normal(0.0, 1.0, n))
    return _normalize(a, 1e-12)


def _score(a, n, fs=None):
    if n <= 0:
        return 0.0, np.zeros(1, dtype=np.float64)
    if n <= 128:
        b = np.convolve(a, a)
    else:
        if fs is None:
            fs = next_fast_len(2 * n - 1)
        F = rfft(a, fs)
        b = irfft(F * F, fs)
        b = b[: (2 * n - 1)]
    b = np.asarray(b, dtype=np.float64)
    if b.size < 2 * n - 1:
        b = np.pad(b, (0, max(0, 2 * n - 1 - b.size)))
    b = np.maximum(b, 0.0)
    m = float(b.max()) if b.size else 0.0
    if not np.isfinite(m) or m < 0.0:
        m = 0.0
    return m, b


def _cc(p, a, n, fs2=None):
    if n <= 64:
        c = np.zeros(n, dtype=np.float64)
        for s in range(n):
            av = float(a[s])
            if av != 0.0:
                c += av * p[s : s + n]
        return c
    if fs2 is None:
        fs2 = next_fast_len(3 * n - 2)
    ar = a[::-1]
    c = irfft(rfft(ar, fs2) * rfft(p, fs2), fs2)
    c = c[n - 1 : 2 * n - 1]
    if c.size < n:
        c = np.pad(c, (0, n - c.size))
    return np.asarray(c[:n], dtype=np.float64)


def _soft_weights(b, M, tau_frac):
    sz = b.size
    if sz == 0:
        return np.array([1.0])
    if M <= 0.0:
        p = np.zeros(sz, dtype=np.float64)
        p[int(np.argmax(b))] = 1.0
        return p
    denom = max(1e-300, tau_frac * M)
    x = (b - M) / denom
    x = np.clip(x, -60.0, 60.0)
    e = np.exp(x)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.zeros(sz, dtype=np.float64)
        p[int(np.argmax(b))] = 1.0
        return p
    return e / s


def _line_search_mix(a, b, M, fs, n, max_evals=8, deadline=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    b = np.asarray(b, dtype=np.float64).ravel()
    best_m = float(M)
    best_lam = 0.0
    best_a = a.copy()
    lams = (1.0, 0.7, 0.5, 0.3, 0.15, 0.075, 0.0375, 0.01875)
    done = 0
    for lam in lams:
        if done >= max_evals:
            break
        if deadline is not None and time.monotonic() >= deadline:
            break
        c = (1.0 - lam) * a + lam * b
        c = _normalize(c, 1e-12)
        m2, _ = _score(c, n, fs)
        if m2 < best_m - 1e-15:
            best_m = m2
            best_lam = lam
            best_a = c
        done += 1

    if best_lam < 0.25 and done < max_evals:
        if best_lam > 0.0:
            extras = (max(best_lam / 4.0, 1e-4), max(best_lam / 16.0, 1e-5))
        else:
            extras = (3e-3, 1e-3)
        for lam in extras:
            if done >= max_evals:
                break
            if deadline is not None and time.monotonic() >= deadline:
                break
            c = (1.0 - lam) * a + lam * b
            c = _normalize(c, 1e-12)
            m2, _ = _score(c, n, fs)
            if m2 < best_m - 1e-15:
                best_m = m2
                best_lam = lam
                best_a = c
            done += 1
    return best_lam, best_m, best_a


def _lp_direction(a, M, delta, keep, cap_factor, n, deadline=None):
    if linprog is None or n > 1024 or M <= 0.0:
        return None, None
    A = np.zeros((2 * n, n), dtype=np.float64)
    for i, val in enumerate(a):
        if val > 0.0:
            A[i : i + n, i] = val
    A[2 * n - 1, :] = 1.0

    c = -np.ones(n, dtype=np.float64)
    maxa = float(np.max(a)) if n else 0.0
    cap = max(8.0 / float(n), float(cap_factor) * maxa + 1e-9, maxa + 1e-9)
    if not np.isfinite(cap) or cap <= 0.0:
        cap = 1.0
    lo = np.maximum(float(keep) * a, 0.0)
    bounds = np.empty((n, 2), dtype=np.float64)
    bounds[:, 0] = lo
    bounds[:, 1] = cap

    x0 = None
    if np.all(lo <= a + 1e-12) and np.all(a <= cap + 1e-12):
        x0 = np.clip(a, lo, cap)

    res = None
    for attempt in range(2):
        if deadline is not None and time.monotonic() >= deadline:
            return None, None
        tol = (M * 1e-7 + 1e-10) * (1.0 if attempt == 0 else 100.0)
        b_ub = np.empty(2 * n, dtype=np.float64)
        b_ub[: 2 * n - 1] = M + tol
        b_ub[2 * n - 1] = 1.0 + float(delta) + 1e-12
        kw = dict(c=c, A_ub=A, b_ub=b_ub, bounds=bounds, options={"presolve": True})
        if x0 is not None:
            kw["x0"] = x0
        try:
            res = linprog(method="highs", **kw)
        except Exception:
            try:
                kw2 = dict(kw)
                kw2.pop("options", None)
                kw2.pop("method", None)
                if x0 is not None:
                    kw2["x0"] = x0
                res = linprog(**kw2)
            except Exception:
                return None, None
        if res is not None and getattr(res, "success", False):
            break

    if res is None or not getattr(res, "success", False):
        return None, None
    try:
        g = np.asarray(res.x, dtype=np.float64)
    except Exception:
        return None, None
    if g.size != n or not np.all(np.isfinite(g)):
        return None, None
    g = np.maximum(g, 0.0)
    S = float(g.sum())
    if not np.isfinite(S) or S <= 1.0 + 1e-5:
        return None, None
    return g / S, S

    lp_keep = 0.0

        if score < best_score - 1e-12:

                    except Exception:

    Mcur, _ = _score(a, n, fs)
    scur = 2.0 * n * max(Mcur, 0.0)

    return best_a, best_score


def _make_inits(size, best_a, parent, rng):
    cands = []
    if best_a is not None:
        xa = np.asarray(best_a, dtype=np.float64).ravel()
        if xa.size > 0 and np.all(np.isfinite(xa)):
            base = _resample(xa, size, False)
            cands.append(_normalize(base, 1e-12))
            cands.append(_normalize(_resample(xa, size, True), 1e-12))
            cands.append(_normalize(base * np.exp(rng.normal(0.0, 0.05, size)), 1e-12))
    if parent is not None:
        xp = np.asarray(parent, dtype=np.float64).ravel()
        if xp.size > 0 and np.all(np.isfinite(xp)):
            cands.append(_normalize(_resample(xp, size, False), 1e-12))
            cands.append(_normalize(_resample(xp, size, True), 1e-12))
    for k in (0, 1, 2, 4, 9):
        cands.append(_init_variant(size, int(k + rng.integers(0, 3)), rng))

    scored = []
    for c in cands:
        try:
            m, _ = _score(c, size, None)
            scored.append((m, c))
        except Exception:
            continue
    if not scored:
        c = _normalize(_init_variant(size, 0, rng), 1e-12)
        return c, [c]
    scored.sort(key=lambda q: q[0])
    init = scored[0][1]
    extras = [scored[i][1] for i in range(1, min(5, len(scored)))]
    return init, extras

            if s < 0.01:
                return math.inf
            a = a / s
            return float(evaluate_sequence([float(x) for x in a]))
        except Exception:
            return math.inf

    def _try_update(arr):
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
            if a.size == 0 or not np.all(np.isfinite(a)):
                return
            a = np.clip(a, 0.0, 1000.0)
            s = float(a.sum())
            if s < 0.01:
                return
            a = a / s
            sc = _exact_score(a)
            if np.isfinite(sc) and (not np.isfinite(best["score"]) or sc < best["score"] - 1e-12):
                best["a"] = a.copy()
                best["score"] = float(sc)
        except Exception:
            return

    if parent is not None:
        _try_update(parent)
    _try_update(fallback)

    eff = max(0.0, hard_deadline - t0)
    if eff < 20.0:
        try:
            out = [float(v) for v in best["a"]]
            if out and all(np.isfinite(v) for v in out):
                return out
        except Exception:
            pass
        return [1.0]

    if eff > 500.0:
        t1 = min(eff * 0.32, 450.0)
        t2 = eff * 0.40
        t3 = max(20.0, eff - t1 - t2)
    else:
        t1 = eff * 0.45
        t2 = eff * 0.30
        t3 = max(20.0, eff - t1 - t2)
        if t1 + t2 + t3 > eff:
            t1 = max(5.0, eff * 0.40)
            t2 = max(5.0, eff * 0.25)
            t3 = max(5.0, eff - t1 - t2)

    phase1_end = min(hard_deadline, t0 + t1)
    if phase1_end > time.monotonic() + 5.0:
        a1, _ = _optimize(
            best["a"], phase1_end, hard_deadline, use_lp=True, seed=seed + 17
        )
        _try_update(a1)

    base_len = int(np.asarray(best["a"]).size)
    other_sizes = []
    for s in (128, 192, 256, 384, 512, 768, 1024):
        if s != base_len:
            other_sizes.append(s)

    weights = []
    for s in other_sizes:
        if s <= 128:
            w = 0.55
        elif s <= 192:
            w = 1.0
        elif s <= 256:
            w = 1.35
        elif s <= 384:
            w = 1.65
        elif s <= 512:
            w = 1.25
        elif s <= 768:
            w = 0.90
        else:
            w = 0.70
        weights.append((s, w))
    weights.sort(key=lambda q: q[1], reverse=True)

    phase2_alloc_end = min(hard_deadline, t0 + t1 + t2)
    now = time.monotonic()
    if phase2_alloc_end > now + 15.0 and weights:
        phase2_total = max(0.0, phase2_alloc_end - now)
        if phase2_total < 60.0:
            weights = weights[:1]
        elif phase2_total < 150.0:
            weights = weights[:2]
        elif phase2_total < 300.0:
            weights = weights[:4]

        tot_w = float(sum(w for _, w in weights))
        rng_top = np.random.default_rng(seed + 12345)
        for s, w in weights:
            now = time.monotonic()
            if now >= phase2_alloc_end - 5.0:
                break
            alloc = phase2_total * (w / tot_w)
            alloc = max(10.0, min(alloc, 220.0))
            alloc = min(alloc, max(5.0, phase2_alloc_end - now - 2.0))
            end = now + alloc
            if end < now + 5.0:
                continue
            init, extras = _make_inits(int(s), best["a"], parent, rng_top)
            use_lp = int(s) <= 512
            a2, _ = _optimize(
                init, end, hard_deadline, use_lp=use_lp, seed=seed + int(s) * 13 + 7, extras=extras
            )
            _try_update(a2)

    final_end = hard_deadline - 2.0
    now = time.monotonic()
    if final_end > now + 10.0:
        a3, _ = _optimize(
            best["a"],
            final_end,
            hard_deadline,
            use_lp=int(np.asarray(best["a"]).size) <= 512,
            seed=seed + 987653,
            extras=None,
        )
        _try_update(a3)

    if parent is not None and time.monotonic() < hard_deadline - 2.0:
        _try_update(parent)
    _try_update(fallback)
    if not np.isfinite(best["score"]):
        best["a"] = fallback.copy()
        best["score"] = _exact_score(fallback)

    try:
        out = [float(v) for v in best["a"]]
    except Exception:
        out = []
    if not out:
        out = [float(v) for v in fallback]

    if time.monotonic() < hard_deadline - 1.0:
        sc = _exact_score(out)
        if not np.isfinite(sc):
            out2 = [1.0]
            sc2 = _exact_score(out2)
            if np.isfinite(sc2):
                out = out2
    return out
```
