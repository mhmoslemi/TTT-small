def _conv_full(a):
    n = len(a)
    if n <= 256:
        b = np.convolve(a, a, mode='full')
    else:
        fsize = next_fast_len(2*n - 1)
        F = rfft(a, fsize)
        b = irfft(F * F, fsize)[:2*n-1]
        np.maximum(b, 0, out=b)
    return b

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)

def _lp_step(f, stop_time):
    """One LP-based direction finding and return the direction."""
    n = len(f)
    b = _conv_full(f)
    M = float(b.max())
    if M < 1e-15:
        return None, M, b
    
    # Build cross-conv matrix A: (f*g)_k = sum_l A[k,l] * g_l
    # A[k,l] = f[k-l] if 0 <= k-l < n else 0
    A = np.zeros((2*n-1, n))
    for l in range(n):
        A[l:l+n, l] = f
    
    # LP: max sum(g) s.t. A*g <= M, g >= 0
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, 
                      bounds=[(0, None)]*n,
                      method='highs',
                      options={'presolve': True, 'time_limit': 5})
    except Exception:
        return None, M, b
    
    if res.status != 0 or res.x is None:
        return None, M, b
    
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0, None)
    g_sum = float(g.sum())
    if g_sum < 1e-12:
        return None, M, b
    
    return g / g_sum, M, b

def _subgrad_polish(f, stop_time, max_iters=5000):
    """Polish using projected subgradient descent on the max-conv term."""
    n = len(f)
    f = f.copy()
    f_sum = f.sum()
    if f_sum < 1e-10:
        return f, math.inf
    f = f / f_sum
    best_f = f.copy()
    best_sc = _score(f)
    
    alpha = 0.05  # initial step size
    alpha_min = 1e-5
    alpha_max = 0.5
    patience = 0
    max_patience = 200
    
    for it in range(max_iters):
        if time.monotonic() > stop_time:
            break
        
        b = _conv_full(f)
        k_star = int(np.argmax(b))
        M = b[k_star]
        
        # Subgradient of (f*f)_{k*} w.r.t. f
        # d/d f_l [(f*f)_{k*}] = 2*f_{k*-l} (if 0 <= k*-l < n), 
        # minus f_{k*/2} if k* even and l = k*/2
        s = np.zeros(n)
        for l in range(n):
            idx = k_star - l
            if 0 <= idx < n:
                s[l] = 2.0 * f[idx]
        if k_star % 2 == 0:
            l_mid = k_star // 2
            if 0 <= l_mid < n:
                s[l_mid] -= f[l_mid]  # subtract the double-counted term
        
        # Projected subgradient step
        # minimize 2n*M / (sum f)^2, but since sum is fixed at 1, just minimize M
        # step: f_new = f - alpha * s, then project to {a>=0, sum=1}
        f_trial = f - alpha * s
        
        # Project: find tau such that sum(max(0, f_trial - tau)) = 1
        if f_trial.max() <= 0:
            f = np.ones(n) / n
        else:
            lo = f_trial.min() - 0.01
            hi = f_trial.max() - 0.01
            for _ in range(50):
                mid = (lo + hi) / 2
                val = np.maximum(0, f_trial - mid).sum()
                if val > 1.0:
                    lo = mid
                else:
                    hi = mid
            tau = (lo + hi) / 2
            f = np.maximum(0, f_trial - tau)
            fs = f.sum()
            if fs > 1e-10:
                f = f / fs
        
        sc = _score(f)
        if sc < best_sc - 1e-10:
            best_f = f.copy()
            best_sc = sc
            patience = 0
            alpha = min(alpha_max, alpha * 1.02)
        else:
            patience += 1
            alpha = max(alpha_min, alpha * 0.95)
            if patience >= max_patience:
                alpha = max(alpha_min, alpha * 0.5)
                patience = 0
                if alpha < alpha_min * 2:
                    break
    
    return best_f, best_sc

def _coord_polish(f, stop_time, max_rounds=50):

    n = len(f)
    f = f.copy()
    sc = _score(f)
    rng = np.random.default_rng(12345)
    no_imp = 0
    
    for rnd in range(max_rounds):

            break

                if sc_i < best_sc_i - 1e-12:
                    best_val = f[i]
                    best_sc_i = sc_i
            f[i] = best_val

                sc = best_sc_i
                improved = True

            no_imp += 1
            if no_imp >= 5:

        else:
            no_imp = 0
