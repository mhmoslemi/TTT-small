```python
import numpy as np
import time
import math
from scipy.optimize import linprog
import scipy.sparse as sps

TARGET = 1.50274349

def _score(a):
    """Fast score computation."""
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 0.0, 1000.0)
    n = a.size
    if n == 0:
        return float('inf')
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    b = np.convolve(a, a)
    mb = float(b.max())
    return 2.0 * n * mb / (s * s)

def _lp_best_response(a, M, n, time_limit=10.0):
    """Solve the LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    # Build sparse constraint matrix
    # A[k, j] = a[j] if 0 <= k-j < n, else 0
    # So A[k, j] = a[j] for max(0, k-n+1) <= j <= min(n-1, k)
    cols = []
    rows = []
    vals = []
    for k in range(2*n-1):
        j_min = max(0, k - n + 1)
        j_max = min(n-1, k)
        for j in range(j_min, j_max+1):
            cols.append(j)
            rows.append(k)
            vals.append(a[j])
    
    A = sps.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M + 1e-12)
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, None) for _ in range(n)],
                      method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-8,
                               'primal_feasibility_tolerance': 1e-8})
    except Exception:
        return None
    
    if res is not None and getattr(res, 'x', None) is not None:
        g = np.asarray(res.x, dtype=np.float64).ravel()
        g = np.clip(g, 0.0, None)
        sg = float(g.sum())
        if sg > 1e-15:
            return g / sg
    return None
```

```python
def _build_structure(n):
    """Build the row/col indices for the convolution matrix (structure only)."""
    n2 = n * n
    row_idx = np.zeros(n2, dtype=np.int32)
    col_idx = np.zeros(n2, dtype=np.int32)
    for j in range(n):
        start = j * n
        row_idx[start:start+n] = np.arange(j, j+n)
        col_idx[start:start+n] = j
    return row_idx, col_idx

def _make_conv_matrix(a, row_idx, col_idx, n):
    """Create the convolution matrix given pre-built structure."""
    # Values for entry (k, j) where k = j + offset, offset = 0..n-1
    # The value is always a[j]
    vals = np.repeat(a, n)
    return sps.csr_matrix((vals, (row_idx, col_idx)), shape=(2*n-1, n))
```

                    cand = np.clip(cand, 0.0, None)
                    cs = float(cand.sum())
                    if cs > 1e-15:
                        cand = cand / cs
                        sc = _score(cand)
                        if math.isfinite(sc) and sc < local:
                            if best_trial is None or sc < _score(best_trial):
                                best_trial = cand
                                best_trial_sc = sc

                if best_trial is not None:
                    a = best_trial
                    local = best_trial_sc
                    lr = min(lr * 1.05, 0.1)
                    _update(a)
                else:
                    lr *= 0.7
                    if lr < 1e-6:
                        break

    # Final: try one more LP polish on the absolute best
    if _rem() > 30:
        nb = best_arr.size
        if nb >= 8:
            alloc = min(60.0, _rem() * 0.5)
            if alloc > 10:
                _lp_iterate(best_arr, nb, alloc, rng)

    # Final validation
    final = np.asarray(best_arr, dtype=np.float64).ravel()
    final = np.nan_to_num(final, nan=0.0, posinf=0.0, neginf=0.0)
