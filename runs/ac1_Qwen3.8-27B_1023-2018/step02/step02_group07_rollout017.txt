def _fft_conv(a, fs):
    F = rfft(a, fs)
    return irfft(F * F, fs)


def _fft_cc(p, a, n, fs):
    ar = a[::-1]
    c = irfft(rfft(ar, fs) * rfft(p, fs), fs)
    return c[n - 1:2 * n - 1]


def _softmax_weights(b, tau_frac):
    bmax = float(b.max())
    if bmax <= 0:
        p = np.zeros(len(b))
        p[int(b.argmax())] = 1.0
        return p, bmax
    tau = max(1e-10, tau_frac * bmax)
    bt = (b - b.max()) / tau
    e = np.exp(np.clip(bt, -200, 200))
    s = e.sum()
    if not np.isfinite(s) or s < 1e-300:
        p = np.zeros(len(b))
        p[int(b.argmax())] = 1.0
    else:
        p = e / s
    return p, bmax
