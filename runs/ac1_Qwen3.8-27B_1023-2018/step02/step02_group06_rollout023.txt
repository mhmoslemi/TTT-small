```python
        # b_soft = logsumexp(s * b) / s
        # grad = (sum_k (2 * R(k) * z) * softmax(s * b)) / (sum_k softmax(s * b))

        sb = s * b
        sb_max = sb.max()
        exp_sb = np.exp(sb - sb_max)
        weights = exp_sb / exp_sb.sum()
        
        # grad[m] = sum_k weights[k] * 2 * z[k-m] for valid k-m
        grad = np.zeros(n)
        for k in range(2*n-1):
            km = k - np.arange(n)  # k - m for m = 0..n-1
            valid = (km >= 0) & (km < n)
            grad[valid] += weights[k] * 2 * z[km[valid]]
        
        # Gradient step
        lr = 0.01 / (np.linalg.norm(grad) + 1e-10)
        z_new = z - lr * grad
        z_new = np.maximum(z_new, 0)
        z_new = z_new / z_new.sum()
        
        # Evaluate
        b_new = fft_conv(z_new)
        score_new = 2 * n * b_new.max()
        
        if score_new < best_score:
            best_score = score_new
            best_z = z_new.copy()
        
        z = z_new
        
        # Adapt s
        s = min(s * 1.02, 500.0)
    
    return best_z
```
