import time
import math
from scipy.fft import rfft, irfft, next_fast_len

    import cvxpy as cp
    HAS_CVXPY = True
except ImportError:
    HAS_CVXPY = False


def _build_T(a, n):
    """Build the (2n-1) x n matrix T where (T @ b)_k = (a*b)_k."""
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def _full_conv(a, b):
    """Full convolution via FFT."""
    n = len(a) + len(b) - 1
    fs = next_fast_len(n)
    return irfft(rfft(a, fs) * rfft(b, fs), fs)[:n]


def _score_inner(a):
    """Compute score without calling evaluate_sequence (for speed)."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return math.inf
    b = _full_conv(a, a)
    return 2.0 * n * float(b.max()) / (S * S)

    """Solve: max 1^T b s.t. T b <= M, b >= 0."""

    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    for solver in solver_list:
        try:
            prob.solve(solver=solver)
            if b_var.value is not None and np.all(np.isfinite(b_var.value)):
                return np.maximum(b_var.value, 0.0)

            a_new = np.maximum(a_new, 1e-12)
            a_new = a_new / float(a_new.sum())
        a = a_new
        it += 1
    
    return best_a

            p[int(np.argmax(b))] = 1.0

            p = e / s

        # ∂L/∂a_j = (2/tau) * cross_corr(p, a)_j

        grad = a_cur * (2.0 * g_corr / tau - float(np.dot(a_cur, 2.0 * g_corr / tau)) )

        # ∂F/∂z_j = a_j * (∂F/∂a_j - dot(a, ∂F/∂a))

        t += 1

        z = np.clip(z, -30, 30)

        b2 = irfft(rfft(a2, fs_c) * rfft(a2, fs_c), fs_c)[:2*n-1]
        s2 = 2.0 * n * float(b2.max())

            m_buf[:] = 0
            v_buf[:] = 0
            t = 0
            lr = 0.3
            tau_frac = 0.25
            stall = 0
    
    return best_a

    if n not in _FSMAP:

    ns = len(a) + len(b) - 1
