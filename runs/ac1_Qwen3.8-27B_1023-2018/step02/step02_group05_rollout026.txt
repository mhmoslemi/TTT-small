def _lp_direction(a, n, time_limit=10):
    """Find g >= 0, sum(g)=1, minimizing max_k (a*g)_k."""
    # A matrix: A[k, j] = a[k-j] if 0 <= k-j < n
    col_idx = np.repeat(np.arange(n), n)
    row_idx = col_idx + np.tile(np.arange(n), n)
    vals = np.tile(a, n)
    # Filter valid entries
    valid = (row_idx >= 0) & (row_idx < 2*n - 1) & (col_idx >= 0) & (col_idx < n)
    row_idx = row_idx[valid]
    col_idx = col_idx[valid]
    vals = vals[valid]
    
    A = sps.csr_matrix((vals, (row_idx, col_idx)), shape=(2*n-1, n))
    
    # Minimize t
    # Variables: g_0, ..., g_{n-1}, t
    # c = [0, 0, ..., 0, 1] (minimize t)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Constraints: A @ g - t <= 0  =>  A @ g + (-1) @ t <= 0
    #              sum(g) = 1
    #              g >= 0, t unbounded
    
    # A_ub @ x <= b_ub
    # [A, -1] @ [g, t] <= 0
    A_aug = sps.hstack([A, sps.csr_matrix(np.zeros((2*n-1, 1)))]).toarray()
    A_aug[:, -1] = -1.0  # wait, this is wrong
    
    # Actually: A @ g <= t  =>  A @ g - t <= 0
    # In the augmented space [g, t]: [A, -I] @ [g, t] <= 0
    # But t is a scalar, so: [A, -1] @ [g, t] <= 0 (the -1 multiplies t)
    
    # Equality: sum(g) = 1 => [1, 1, ..., 1, 0] @ [g, t] = 1
    A_eq = np.hstack([np.ones(n), [0.0]])
    b_eq = np.array([1.0])
    
    # Inequality: A @ g - t <= 0
    A_ub = np.hstack([A.toarray(), -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n-1)
    
    bounds = [(0, None)] * n + [(None, None)]  # g >= 0, t free
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'time_limit': time_limit, 'presolve': True})
    
    if res is not None and res.status == 0 and res.x is not None:
        g = res.x[:n]
        g = np.clip(g, 0, None)
        s = g.sum()
        if s > 1e-15:
            return g / s
    return None

Wait, there's an issue. The formulation "minimize $\max_k (a * g)_k$" gives us a $g$ that has small "cross-convolution" with $a$. But we want $g$ to have small "auto-convolution" ($g * g$). These are different objectives.

However, the mixing step $a_{new} = (1-t)a + tg$ gives:
$a_{new} * a_{new} = (1-t)^2 (a*a) + 2t(1-t)(a*g) + t^2 (g*g)$

So if both $a*g$ and $g*g$ are small, the new max is small. The LP minimizes $a*g$, which helps. And if $g$ is "flat" (spread out), then $g*g$ should also be relatively flat.
