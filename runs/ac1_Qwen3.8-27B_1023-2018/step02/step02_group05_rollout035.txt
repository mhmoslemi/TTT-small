        t_col = np.full(2*n-1, -1.0)

        b_eq = 1.0
        
        c = np.zeros(n+1)
        c[n] = 1.0  # minimize t
        
        bounds = [(0, None)] * n + [(None, None)]  # g >= 0, t free
        
        try:
            res = linprog(c, A_ub=A_ub_full, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'time_limit': time_limit, 'presolve': True})
        except:
            return None, None
        
        if res is not None and res.x is not None and res.status in (0, 1, 3):
            g = np.clip(res.x[:n], 0, None)
            if g.sum() > 1e-15:
                g = g / g.sum()
                return g, float(res.x[n])
        return None, None
