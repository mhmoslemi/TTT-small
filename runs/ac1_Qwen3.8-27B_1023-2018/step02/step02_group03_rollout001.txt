                F0 = M + tau * math.log(max(ws, 1e-300)) if ws > 0.0 else M

                d = cross_corr(w, a)
                d = d - float(d.mean())
                if float(np.linalg.norm(d)) < 1e-15:
                    break

                best_cand = None
                best_val = F0
                for al in alphas_add:
                    cand = a - al * d
                    cand = np.clip(cand, 0.0, None)
                    if float(cand.sum()) < 1e-15:
                        continue
                    cand = _norm1(cand)
                    c2 = np.convolve(cand, cand)
                    F2 = _lse_value(c2, tau)
                    if F2 < best_val - 1e-15:
                        best_val = F2
                        best_cand = cand

                if best_cand is not None:
                    a = best_cand
                    record(a)
                    no_improve = 0
                    continue

                base = float(np.mean(np.abs(d)))
                if base > 1e-15:
                    ds = d / base
                    best_cand = None
                    best_val = F0
                    for eta in alphas_mult:
                        trial = a * np.exp(-eta * np.clip(ds, -20.0, 20.0))
                        trial = np.clip(trial, 0.0, None)
                        if float(trial.sum()) < 1e-15:
                            continue
                        trial = _norm1(trial)
                        c2 = np.convolve(trial, trial)
                        F2 = _lse_value(c2, tau)
                        if F2 < best_val - 1e-15:
                            best_val = F2
                            best_cand = trial
                            break
                    if best_cand is not None:
                        a = best_cand
                        record(a)
                        no_improve = 0
                        continue

                no_improve += 1
                if no_improve >= 50:
                    break

        return a

    def active_refine(a, stop, max_iters=800):
        a = _norm1(a)
        if a.size == 0:
            return a
        deltas = (0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.0)
        alphas = (0.05, 0.1, 0.2, 0.35, 0.5, 0.8, 1.2)

        for delta in deltas:
            if time.monotonic() >= stop - 0.2:
                break
            no_improve = 0
            for _ in range(max_iters):
                if time.monotonic() >= stop - 0.2:
                    break
                c = np.convolve(a, a)
                M = float(c.max()) if c.size else 0.0
                if not math.isfinite(M) or M <= 0.0:
                    break
                if delta > 0.0:
                    mask = c >= M * (1.0 - float(delta))
                    if not np.any(mask):
                        mask[int(np.argmax(c))] = True
                else:
                    K = max(3, int(0.02 * c.size))
                    mask = np.zeros(c.size, dtype=bool)
                    if K >= c.size:
                        mask[:] = True
                    else:
                        idx = np.argsort(c)[-K:]
                        mask[idx] = True
                w = mask.astype(np.float64)
                ws = float(w.sum())
                if ws <= 0.0:
                    w[int(np.argmax(c))] = 1.0
                    ws = 1.0
                else:
                    w /= ws

                d = cross_corr(w, a)
                d = d - float(d.mean())
                if float(np.linalg.norm(d)) < 1e-15:
                    break

                h0 = float(np.dot(w, c))
                best_cand = None
                best_h = h0
                for al in alphas:
                    cand = a - al * d
                    cand = np.clip(cand, 0.0, None)
                    if float(cand.sum()) < 1e-15:
                        continue
                    cand = _norm1(cand)
                    c2 = np.convolve(cand, cand)
                    h2 = float(np.dot(w, c2))
                    if h2 < best_h - 1e-15:
                        best_h = h2
                        best_cand = cand

                if best_cand is not None:
                    a = best_cand
                    record(a)
                    no_improve = 0
                    continue
                no_improve += 1
                if no_improve >= 35:
                    break
        return a

    def small_search(small_stop):
        nmax = 10
        if time.monotonic() >= small_stop:
            return
        per_n = max(1.0, (small_stop - time.monotonic()) / float(nmax))

        for n in range(1, nmax + 1):
            if time.monotonic() >= small_stop:
                break
            if n == 1:
                record(np.ones(1, dtype=np.float64))
                continue
            if n == 2:
                record(np.array([0.5, 0.5], dtype=np.float64))
                continue

            starts = []
            if n == 3:
                starts.append(np.array([0.5, 0.0, 0.5], dtype=np.float64))
            starts.append(np.ones(n, dtype=np.float64))
            starts.append(np.linspace(1.0, 0.2, n, dtype=np.float64))
            starts.append(np.linspace(0.2, 1.0, n, dtype=np.float64))
            starts.append(np.exp(-np.linspace(0.0, 2.0, n, dtype=np.float64)))
            starts.append(np.exp(np.linspace(0.0, 2.0, n, dtype=np.float64)))
            starts.append(np.concatenate([np.ones(n // 2, dtype=np.float64),
                                          np.zeros(n - n // 2, dtype=np.float64)]))
            starts.append(np.concatenate([np.zeros(n // 2, dtype=np.float64),
                                          np.ones(n - n // 2, dtype=np.float64)]))

            n_rand = 12 if n <= 6 else 6
            for _ in range(n_rand):
                v = np.abs(rng.normal(0.0, 1.0, n))
                if float(v.sum()) > 1e-12:
                    starts.append(v)

            n_budget = min(per_n, max(0.0, small_stop - time.monotonic()))
            if n_budget < 0.5:
                break
            per_start = max(0.2, n_budget / float(max(1, len(starts))))

            for a0 in starts:
                if time.monotonic() + 0.05 >= small_stop:
                    break
                a = _norm1(a0)
                stop1 = min(small_stop, time.monotonic() + per_start)
                a = smooth_refine(
                    a,
                    stop1,
                    max_iters=2500,
                    tau_fracs=(0.2, 0.1, 0.05, 0.02, 0.01),
                )
                stop2 = min(small_stop, time.monotonic() + per_start)
                a = active_refine(a, stop2, max_iters=500)
                record(a)

    parent = None
    try:
        parent = globals().get("height_sequence_1", None)
    except Exception:
        parent = None
    if parent is not None:
        try:
            record(parent)
        except Exception:
            pass

    record(np.array([0.5, 0.0, 0.5], dtype=np.float64))

    small_stop = min(deadline - 1.0, t0 + min(90.0, 0.25 * budget))
    if time.monotonic() < small_stop:
        try:
            small_search(small_stop)
        except Exception:
            pass

    if parent is not None and time.monotonic() < deadline - 5.0:
        try:
            p = _norm1(parent)
            p_stop = min(deadline - 4.0, time.monotonic() + min(60.0, 0.25 * budget))
            if p_stop > time.monotonic() + 2.0:
                p = smooth_refine(
                    p,
                    p_stop,
                    max_iters=5000,
                    tau_fracs=(0.1, 0.05, 0.02, 0.01, 0.005),
                )
                p_stop2 = min(deadline - 3.0, time.monotonic() + min(30.0, 0.10 * budget))
                if p_stop2 > time.monotonic() + 1.0:
                    p = active_refine(p, p_stop2, max_iters=1000)
                record(p)
        except Exception:
            pass

    if best_arr is None:
        best_arr = np.array([0.5, 0.0, 0.5], dtype=np.float64)
        best_score = _fast_score(best_arr)

    final = np.asarray(best_arr, dtype=np.float64).ravel()
    final = np.where(np.isfinite(final), final, 0.0)
    final = np.clip(final, 0.0, 1000.0)
    if final.size == 0 or float(final.sum()) < 0.01:
        final = np.array([0.5, 0.0, 0.5], dtype=np.float64)

    try:
        fs = _fast_score(final)
        if math.isfinite(fs) and (not math.isfinite(best_score) or fs + 1e-9 < best_score):
            seq = [float(x) for x in final]
            try:
                es = evaluate_sequence(seq)
            except Exception:
                es = fs
            if math.isfinite(es) and es < best_score:
                best_score = float(es)
                best_arr = final.copy()
                final = best_arr
    except Exception:
        pass

    return [float(x) for x in final]
