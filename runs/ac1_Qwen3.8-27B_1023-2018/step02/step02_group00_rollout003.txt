    grad = np.zeros(n)

                        low = max(0.0, bw - width)
                        high = min(hi, bw + width)
                    if high <= low:
                        low, high = 0.0, hi
                    arr = np.linspace(low, high, 25)
                    for ww in arr:
                        ww = float(ww)
                        if abs(ww - bw) <= 1e-15:
                            continue
                        m = eval_w(ww)
                        if m < best_m - 1e-15:
                            best_m = m
                            best_w = ww

            if best_m < cur_mu - 1e-12:
                s = (1.0 - best_w) / den
                x_new = x * s
                x_new[i] = best_w
                sxn = float(x_new.sum())
                if sxn <= 0.0 or not math.isfinite(sxn):
                    continue
                x_new = x_new / sxn
                c_new = _conv_full(x_new)
                m_new = float(c_new.max())
                sc_new = 2.0 * n * m_new / (sx * sx)
                if math.isfinite(sc_new) and sc_new < cur_sc - 1e-12:
                    x = x_new
                    c = c_new
                    sx = sxn
                    cur_mu = m_new / (sx * sx)
                    cur_sc = sc_new
                    improved = True
                    if sc_new < best_score - 1e-12:
                        best_x = x.copy()
                        best_score = float(sc_new)

        return x, c, improved

    def local_optimize(x0, stop, max_lp):
        nonlocal best_x, best_score
        x = _normalize(x0, n).astype(np.float64).copy()
        if x.size != n:
            x = _normalize(best_x, n).astype(np.float64).copy()
        c = _conv_full(x)
        sx = float(x.sum())
        if sx <= 0.0 or c.size != 2 * n - 1:
            x = _normalize(best_x, n).astype(np.float64).copy()
            c = _conv_full(x)
            sx = float(x.sum())

        cur_sc = 2.0 * n * float(c.max()) / (sx * sx)
        if not math.isfinite(cur_sc):
            cur_sc = best_score
        update_best(x)

        stall = 0
        lp_frac = 0.0
        lp_calls = 0

        while time.monotonic() < stop - 1.5:
            if time.monotonic() >= deadline - 5.0:
                break

            substop = min(stop - 1.0, time.monotonic() + 120.0)
            x, c, imp = coordinate_pass(x, c, substop)
            sx = float(x.sum())
            if sx > 0.0 and c.size == 2 * n - 1:
                cur_sc = 2.0 * n * float(c.max()) / (sx * sx)
                if not math.isfinite(cur_sc):
                    cur_sc = best_score
                update_best(x)
            else:
                x = _normalize(best_x, n).astype(np.float64).copy()
                c = _conv_full(x)
                sx = float(x.sum())
                cur_sc = 2.0 * n * float(c.max()) / (sx * sx)

            if imp:
                stall = 0
                continue

            if time.monotonic() < stop - 12.0:
                gstop = min(stop - 4.0, time.monotonic() + 45.0)
                y = _try_gradient_move(x, c, cur_sc, gstop)
                if y is not None:
                    x = _normalize(y, n).astype(np.float64).copy()
                    c = _conv_full(x)
                    sx = float(x.sum())
                    cur_sc = 2.0 * n * float(c.max()) / (sx * sx)
                    update_best(x)
                    stall = 0
                    continue

            if (
                int(max_lp) > 0
                and time.monotonic() < stop - 70.0
                and lp_calls < int(max_lp)
            ):
                if n <= 600:
                    y = _try_lp_move(x, cur_sc, min(stop - 5.0, time.monotonic() + 60.0), lp_frac)
                    if y is not None:
                        x = _normalize(y, n).astype(np.float64).copy()
                        c = _conv_full(x)
                        sx = float(x.sum())
                        cur_sc = 2.0 * n * float(c.max()) / (sx * sx)
                        update_best(x)
                        stall = 0
                        lp_frac = 0.0
                        lp_calls += 1
                        continue
                    lp_calls += 1
                    lp_frac = min(0.04, lp_frac * 2.0 + 1e-4)
                    if lp_frac > 0.02:
                        lp_calls = int(max_lp)
                else:
                    lp_calls = int(max_lp)

            if time.monotonic() >= stop - 6.0:
                break

            sigma = 0.035 + 0.017 * min(stall, 6)
            y = np.clip(x * (1.0 + sigma * rng_main.standard_normal(n)), 0.0, None)
            y = _normalize(y, n).astype(np.float64).copy()
            scy = _score_array(y)

            if math.isfinite(scy) and scy < best_score + 0.08 and scy < max(cur_sc, 1e-8) + 0.2:
                x = y
                c = _conv_full(x)
                sx = float(x.sum())
                cur_sc = 2.0 * n * float(c.max()) / (sx * sx)
            else:
                x = _normalize(best_x, n).astype(np.float64).copy()
                c = _conv_full(x)
                sx = float(x.sum())
                cur_sc = 2.0 * n * float(c.max()) / (sx * sx)

            stall += 1
            if stall > 12:
                break

        update_best(x)
        return x, cur_sc

    stop1 = min(deadline - 25.0, t0 + max(30.0, 0.55 * budget))
    if time.monotonic() < stop1 - 3.0:
        try:
            local_optimize(best_x.copy(), stop1, max_lp=20)
        except Exception:
            pass

    stop2 = min(deadline - 45.0, t0 + max(30.0, 0.78 * budget))
    if time.monotonic() < stop2 - 4.0:
        try:
            x2 = np.clip(
                best_x * (1.0 + 0.045 * rng_main.standard_normal(n)),
                0.0,
                None,
            ).astype(np.float64)
            x2 = _normalize(x2, n)
            sc2 = _score_array(x2)
            if math.isfinite(sc2) and sc2 < best_score + 0.12:
                local_optimize(x2, stop2, max_lp=10)
            else:
                local_optimize(best_x.copy(), stop2, max_lp=10)
        except Exception:
            pass

    if time.monotonic() < deadline - 6.0:
        try:
            local_optimize(best_x.copy(), deadline - 6.0, max_lp=5)
        except Exception:
            pass

    try:
        update_best(best_x, tol=0.0)
    except Exception:
        pass

    final = best_x
    if final is None or final.size == 0:
        final = np.ones(max(1, n), dtype=np.float64)
    final = np.asarray(final, dtype=np.float64).ravel().copy()
    final = np.where(np.isfinite(final), final, 0.0)
    final = np.clip(final, 0.0, 1000.0)
    if final.size == 0 or float(final.sum()) < 0.01:
        final = np.ones(max(1, n), dtype=np.float64)

    try:
        exact = float(evaluate_sequence([float(v) for v in final]))
    except Exception:
        exact = math.inf

    if (not math.isfinite(exact)) or (exact <= 0.0 and _score_array(final) > 0.0):
        try:
            fb = _normalize(parent if parent is not None else best_x, n).astype(np.float64)
            fb_exact = float(evaluate_sequence([float(v) for v in fb]))
            if math.isfinite(fb_exact):
                final = fb
                exact = fb_exact
        except Exception:
            pass

    if (not math.isfinite(exact)) or exact <= 0.0:
        final = np.ones(max(1, n), dtype=np.float64) / max(1, n)

    return [float(v) for v in final]
