```python
def _smooth_max_gradient(a, T):
    """Compute gradient of smooth max of a*a."""
    n = len(a)
    b = np.convolve(a, a)  # length 2n-1
    # Smooth max
    m = b.max()
    b_shifted = b - m  # prevent overflow
    weights = np.exp(T * b_shifted)
    weights /= weights.sum()
    # Gradient: d/d a_j of (a*a)_k = 2*a[k-j] for valid k-j
    grad = np.zeros(n)
    for k in range(2*n-1):
        if weights[k] < 1e-15:
            continue
        # a[k-j] for j = max(0,k-n+1) to min(n-1, k)
        lo = max(0, k-n+1)
        hi = min(n-1, k)
        for j in range(lo, hi+1):
            grad[j] += 2.0 * weights[k] * a[k-j]
    return grad
```
