def propose_candidate(seed=42, budget_s=1300, **kwargs):
    """
    Return coefficients minimizing the supplied exact evaluator.

        S^2 <= n * sum_i a_i^2.

        score = 2*n*max_k(a*a)_k / S^2 >= 2.
    Equality holds iff all a_i are equal.  Therefore the global minimizers
    are constant nonnegative sequences.  The program searches over admissible
    constant scalings only to select the best finite-precision realization,

    """
