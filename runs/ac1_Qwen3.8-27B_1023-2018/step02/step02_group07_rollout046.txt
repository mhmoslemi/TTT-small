```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _conv_full(a, b):
    """Full convolution of a and b."""
    if len(a) == 0 or len(b) == 0:
        return np.array([])
    n, m = len(a), len(b)
    fs = next_fast_len(n + m - 1)
    Fa = rfft(a, fs)
    Fb = rfft(b, fs)
    return irfft(Fa * Fb, fs)[:n + m - 1]

def _score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    b = _conv_full(a, a)
    mb = float(b.max())
    sa = float(a.sum())
    if sa < 0.01:
        return float('inf')
    return 2.0 * n * mb / (sa * sa)

def _lp_direction(f, M):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)
    # Build Toeplitz matrix T where T[k,j] = f[k-j] if valid
    # conv(f, g)[k] = sum_j T[k,j] * g[j]
    # T has shape (2n-1, n)
    T = np.zeros((2*n - 1, n))
    for k in range(2*n - 1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                T[k, j] = f[idx]
    
    c = -np.ones(n)  # minimize -sum(g)
    b_ub = np.full(2*n - 1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None:
        return np.maximum(res.x, 0)
    return None

def _fw_iterate(f, M, t=0.1):
    """One Frank-Wolfe iteration."""
    g = _lp_direction(f, M)
    if g is None:
        return f
    S = float(f.sum())
    sg = float(g.sum())
    if sg < 1e-15:
        return f
    g = S / sg * g  # rescale to match normalization
    f_new = (1 - t) * f + t * g
    f_new = np.maximum(f_new, 0)
    return f_new
```

```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _fast_conv(a, b):
    n, m = len(a), len(b)
    fs = next_fast_len(n + m - 1)
    return irfft(rfft(a, fs) * rfft(b, fs), fs)[:n+m-1]

def _fast_autocorr(a):
    n = len(a)
    fs = next_fast_len(2*n - 1)
    F = rfft(a, fs)
    return irfft(F * F, fs)[:2*n-1]

def _compute_score(a):
    n = len(a)
    b = _fast_autocorr(a)
    mb = float(b.max())
    sa = float(a.sum())
    if sa < 0.01:
        return float('inf')
    return 2.0 * n * mb / (sa * sa)

def _build_conv_matrix(f):
    n = len(f)
    T = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        lo = max(0, k - n + 1)
        hi = min(n-1, k)
        T[k, lo:hi+1] = f[k-hi:k-lo+1]
    return T

def _fw_step(f, M, step_size):
    n = len(f)
    T = _build_conv_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return f, False
    g = np.maximum(res.x, 0)
    S = float(f.sum())
    sg = float(g.sum())
    if sg < 1e-15:
        return f, False
    g = S / sg * g
    f_new = (1 - step_size) * f + step_size * g
    return np.maximum(f_new, 0), True

            best_f = f.copy()
        
        t = t0 / (1 + 0.1 * k)  # decreasing step size
        f, ok = _fw_step(f, M, t)
        if not ok:
```
