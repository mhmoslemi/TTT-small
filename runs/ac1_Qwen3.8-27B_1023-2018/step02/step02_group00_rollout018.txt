def _conv_full(a):

def _build_A(f, n):

def _lp_solve(f, n, M, time_limit=20.0):

    try:
        res = linprog(
            c, A_ub=A, b_ub=b_ub,
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True, "time_limit": time_limit},
        )
    except Exception:

    if res.status not in (0, 3) or res.x is None:
        return None, None
