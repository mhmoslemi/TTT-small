import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import minimize, linprog


def _conv_full(a):
    """Full convolution of a with itself."""
    n = len(a)
    if n <= 256:
        c = np.convolve(a, a, mode='full')
    else:
        fsize = next_fast_len(2 * n - 1)
        F = rfft(a, fsize)
        c = irfft(F * F, fsize)[:2*n-1]
    np.maximum(c, 0.0, out=c)
    return c


def _conv2(a, b):
    """Full convolution of a and b via FFT."""
    na, nb = len(a), len(b)
    fsize = next_fast_len(na + nb - 1)
    return irfft(rfft(a, fsize) * rfft(b, fsize), fsize)[:na+nb-1]


def _score(a):
    """Compute the AC1 upper bound score."""
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)

def _grad_softmax(theta, beta, n):
    """Compute soft-max objective value and gradient w.r.t. theta."""
    # Safe softmax
    t = theta - theta.max()
    e = np.exp(t)
    a = e / e.sum()
    
    # Convolution
    c = _conv_full(a)
    
    # Soft-max weights
    bc = beta * c
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    p = exp_bc / exp_bc.sum()
    
    # Objective
    val = (bc_max + math.log(float(exp_bc.sum()))) / beta

    r = pc[n-1:2*n-1]
    
    r_bar = float(np.dot(r, a))
    grad = 2.0 * a * (r - r_bar)
    
    return val, grad, a

    """Run L-BFGS-B optimization on the softmax parameterization."""

def _lp_direction(f, n, M):

        return best

    rng = np.random.default_rng(seed)
    no_imp = 0
    
    while time.monotonic() < stop_time - 2.0:
        improved = False
        
        # Single-coordinate moves
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 2.0:
                break
            orig = best[i]
            best_val = orig
            best_val_sc = best_sc

                    continue

                if sc < best_val_sc - 1e-14:
                    best_val = best[i]
                    best_val_sc = sc
            best[i] = best_val
            if best_val_sc < best_sc - 1e-14:
                best_sc = best_val_sc
                improved = True

            if no_imp >= 5:

        else:
