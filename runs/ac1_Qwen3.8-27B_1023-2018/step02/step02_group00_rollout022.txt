        p = np.ones(320, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(320, dtype=np.float64)
    return p

def _conv(a):
    n = len(a)
    if n <= 256:
        c = np.convolve(a, a, mode='full')
    else:
        fsize = next_fast_len(2*n-1)
        F = rfft(a, fsize)
        c = irfft(F*F, fsize)[:2*n-1]
        np.maximum(c, 0, out=c)
    return c

def _score(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return math.inf
    b = _conv(a)
    m = b.max()
    return 2.0 * n * m / (s*s)

def _build_A(f, n):
    # A[k, j] = f[j] if 0 <= k-j < n, else 0
    # i.e., A is the convolution matrix
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = f
    return A

def _lp_solve(f, n, M):
    """Solve: max sum(g) s.t. (f*g)_k <= M, g >= 0"""
    A = _build_A(f, n)
    c = -np.ones(n)
    try:
        res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M),
                      bounds=[(0, None)]*n, method='highs',
                      options={'presolve': True, 'time_limit': 5})
        if res.status != 0:
            return None
        g = res.x
        gs = g.sum()
        if gs < 1e-15:
            return None
        return g / gs  # normalize to sum=1
    except Exception:
        return None

def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(20.0, float(budget_s))
    deadline = t0 + budget - 15  # 15s buffer for return
    
    parent = _get_parent()
    n = len(parent)
    rng = np.random.default_rng(seed)
    
    # Initialize from parent
    f = np.clip(parent.astype(np.float64), 0, 1000).copy()
    fs = f.sum()
    if fs < 0.01:
        f = np.ones(n, dtype=np.float64) / n
    else:
        f = f / fs
    
    best_f = f.copy()
    best_score = _score(f)
    
    # Validate with exact evaluator
    try:
        exact = evaluate_sequence([float(x) for x in best_f])
        if exact < best_score:
            best_score = exact
    except:
        pass
    
    # Phase 1: LP iterations
    t_lp_end = t0 + budget * 0.70
    step = 0.25
    stall = 0
    
    conv_cache = _conv(f)
    
    while time.monotonic() < t_lp_end:
        # Check deadline
        if time.monotonic() > deadline - 20:
            break
        
        M = conv_cache.max()
        if M < 1e-15:
            break
        
        # Solve LP
        g = _lp_solve(f, n, M)
        if g is None:
            stall += 5
            if stall > 200:
                break
            # Perturb
            f = np.clip(f + rng.normal(0, 0.01, n), 0, None)
            fs = f.sum()
            if fs > 1e-10:
                f /= fs
            continue
        
        # Check if g ≈ f (convergence)
        dist = np.linalg.norm(g - f)
        if dist < 1e-8:
            stall += 2
            if stall > 100:
                break
            # Small perturbation
            idx = rng.integers(n)
            f[idx] *= (1 + 0.1 * rng.standard_normal())
            f = np.clip(f, 0, None)
            fs = f.sum()
            if fs > 1e-10:
                f /= fs
            else:
                f = best_f.copy()
            continue
        
        # Try step sizes
        improved = False
        for _ in range(15):
            f_new = (1 - step) * f + step * g
            f_new = np.maximum(f_new, 0)
            fs_new = f_new.sum()
            if fs_new < 1e-10:
                step = max(0.01, step * 0.5)
                continue
            f_new /= fs_new  # normalize to sum=1
            
            conv_new = _conv(f_new)
            M_new = conv_new.max()
            sc_new = 2.0 * n * M_new  # since sum=1, score = 2n*M
            
            if sc_new < best_score - 1e-12:
                f = f_new
                conv_cache = conv_new
                best_f = f.copy()
                best_score = sc_new
                step = min(0.5, step * 1.1)
                stall = 0
                improved = True
                break
            elif sc_new < 2.0 * n * M:  # at least not much worse
                step = max(0.01, step * 0.7)
            else:
                step = max(0.01, step * 0.5)
        
        if not improved:
            stall += 1
            if stall > 300:
                # Reset with perturbation
                f = best_f.copy()
                f = np.clip(f + rng.normal(0, 0.05, n), 0, None)
                fs = f.sum()
                if fs > 1e-10:
                    f /= fs
                else:
                    f = best_f.copy()
                conv_cache = _conv(f)
                step = 0.25
                stall = 0
                if stall > 500:  # absolute max perturb
                    break
    
    # Phase 2: Subgradient refinement
    sg_end = t0 + budget * 0.85
    if time.monotonic() < sg_end:
        f = best_f.copy()
        for _ in range(100):
            if time.monotonic() > sg_end:
                break
            conv_f = _conv(f)
            k_star = conv_f.argmax()
            # Subgradient: (f * delta_{k*})_i = f[k*-i] if 0 <= k*-i < n
            sg = np.zeros(n)
            lo = max(0, k_star - n + 1)
            hi = min(n - 1, k_star)
            for i in range(lo, hi + 1):
                sg[i] = f[k_star - i]
            
            # Adj-Hessian-like for step size
            eta = 1.0 / (np.linalg.norm(sg) + 1e-10)
            
            f_new = f - eta * 0.5 * sg
            f_new = np.maximum(f_new, 0)
            fs = f_new.sum()
            if fs > 1e-10:
                f_new /= fs
