```python
def lp_solve(f, M, time_limit=20):
    n = f.size
    # Build constraint matrix: A[k, j] = f[k-j] if 0 <= k-j < n
    # Total constraints: 2n-1
    # Use scipy linprog with HiGHS
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = f
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M * 1.001)
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.clip(r.x, 0, None)
            if g.sum() > 1e-10:
                return g / g.sum()
    except:
        pass
    return None
```
