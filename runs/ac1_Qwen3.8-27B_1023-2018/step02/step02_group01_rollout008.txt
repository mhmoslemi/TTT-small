Let me think about the LP more carefully. The problem is:
max 1^T g
s.t. A_a * g ≤ M * 1  (where A_a is the convolution operator)
     g ≥ 0

The constraint (a⊕g)[k] ≤ M for all k means:
sum_j a[k-j] * g[j] ≤ M for k = 0, ..., 2n-2

So A[k, j] = a[k-j] if 0 ≤ k-j ≤ n-1, else 0.

This is a banded matrix (bandwidth n in a 2n-1 × n matrix). For n=191, it's 381×191, which is trivial for HiGHS.

Now, for the gradient-based approach, I want to minimize:
f(a) = log(sum_k exp(β * c[k])) + λ * (sum(a) - 1)² + μ * max(0, -a)² 

or better, use the constraint directly:
minimize f(a) = log(sum_k exp(β * c[k]))
subject to a ≥ 0, sum(a) = 1

Using L-BFGS-B with bounds [0, None] for each coordinate, and handling the equality constraint by parameterizing a = softmax(z) or by projecting.

Actually, for simplicity, let me work with the normalization built in:
- Let a be the variable, with a ≥ 0.
- The score is 2n*max(c)/S² where S = sum(a).
- Since score is scale-invariant, we can fix S = 1 and minimize 2n*max(c).
- But during optimization, we can let S vary and just minimize the ratio.

For the gradient of the soft-max objective with respect to a (without normalization):
L = log(sum_k exp(β*c[k]))
dL/da[j] = β * sum_k w_k * (a[k-j] + a[j-k]) where w_k = exp(β*c[k]) / sum exp(β*c[l])
(and a[x] = 0 for x outside [0,n-1]).

This is O(n²) to compute, which is fine.

```python
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, next_fast_len

def _convolve_square(a):
    n = a.size
    if n <= 1:
        return a * a
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _build_convolve_mat(a):
    """Build matrix A where (A @ g) = a ⊕ g (full convolution)."""
    n = a.size
    m = 2 * n - 1
    k_idx = np.arange(m)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    mask = (diff >= 0) & (diff < n)
    A = np.where(mask, a[np.clip(diff, 0, n-1)], 0.0)
    return A

def _lp_best_response(a, M, timeout_s=1.0):
    n = a.size
    A = _build_convolve_mat(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0.0, None)] * n,
                      method='highs', options={'time_limit': timeout_s,
                      'dual_feasibility_tolerance': 1e-10,
                      'primal_feasibility_tolerance': 1e-10})
    except Exception:
        return None
    if res.status != 0:
        return None
    g = res.x
    if g is None or not np.isfinite(g).all():
        return None
    if g.sum() <= 0:
        return None
    return g

def _lp_iterate(a0, stop_time, max_iters=200, relax=1.0):

            break

        M = float(c.max()) * relax

            best_a = a.copy()
            stuck = 0

            stuck += 1

            break
        
        g = _lp_best_response(a, M, timeout_s=1.0)
        if g is None:
            break
        
        g = g / g.sum()
        
        # Line search for best blend
        improved = False
        for t in (1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1, 0.05):
            cand = (1 - t) * a + t * g
            cand = np.maximum(cand, 0.0)
            cs = cand.sum()
            if cs <= 0:
                continue
            cand = cand / cs
            cM = float(_convolve_square(cand).max())
            if cM < M * (1 - 1e-12):
                a = cand
                improved = True
                break
        
        if not improved:
            if stuck > 50:
                break
    
    return best_a
```

```python
    m = 2 * n - 1

    w = exp_bc / exp_bc.sum()
    
    # grad[j] = β * (sum_k w[k]*a[k-j] + sum_k w[k]*a[j-k])
    
    # Part 1: P1[j] = sum_k w[k]*a[k-j] for valid k, k-j
    # k ranges [0, m-1], k-j ranges [0, n-1]
    # So k-j = i => k = j+i, i in [0, n-1], k in [j, j+n-1] ∩ [0, m-1]
    # P1[j] = sum_{i=0}^{n-1} w[j+i] * a[i] (where j+i < m)
    # This is a Toeplitz matrix: P1 = T @ a where T[j,i] = w[j+i] if j+i<m else 0
    
    # Build T (n x n):
    # T[j, i] = w[j+i] if j+i <= m-1 else 0
    idx = np.arange(n)
    T = w[np.arange(n)[:, None] + np.arange(n)[None, :]]
    T[np.arange(n)[:, None] + np.arange(n)[None, :] >= m] = 0
    
    P1 = T @ a
    
    # Part 2: P2[j] = sum_k w[k]*a[j-k] for valid k, j-k
    # j-k = i => k = j-i, i in [0, n-1], k in [j-(n-1), j] ∩ [0, m-1]
    # P2[j] = sum_{i=0}^{n-1} w[j-i] * a[i] (where j-i >= 0)
    # This is: T2[j, i] = w[j-i] if j-i >= 0 else 0
    T2 = np.where(np.arange(n)[None, :] <= np.arange(n)[:, None], 
                  w[np.clip(np.arange(n)[:, None] - np.arange(n)[None, :], 0, m-1)], 0)
    # Wait: T2[j, i] = w[j-i] if 0 <= j-i <= m-1 else 0
    # j-i >= 0 means i <= j
    diff = np.arange(n)[:, None] - np.arange(n)[None, :]  # diff[j,i] = j-i
    T2 = np.where((diff >= 0) & (diff < m), w[np.clip(diff, 0, m-1)], 0)
    P2 = T2 @ a
    
    grad = beta * (P1 + P2)
    return grad
```

```python
import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, next_fast_len


def _convolve_square(a):
    n = a.size
    if n <= 1:
        return a * a
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _build_convolve_mat(a):
    n = a.size
    m = 2 * n - 1
    k_idx = np.arange(m)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    mask = (diff >= 0) & (diff < n)
    A = np.where(mask, a[np.clip(diff, 0, n - 1)], 0.0)
    return A


def _lp_solve(a, M, timeout_s=1.5):
    n = a.size
    A = _build_convolve_mat(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'time_limit': timeout_s,
                               'dual_feasibility_tolerance': 1e-11,
                               'primal_feasibility_tolerance': 1e-11})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    if not np.isfinite(g).all() or g.sum() <= 1e-15:
        return None
    return g

        c = _convolve_square(a)

        improved = False

            cand = (1.0 - t) * a + t * g

        c = _convolve_square(a)
        cur_M = float(c.max())
        if cur_M < best_M * (1.0 - 1e-13):

            best_a = a.copy()
            no_improve = 0
        else:

        if no_improve > 8000:
            a = a * np.exp(rng.normal(0, 0.05, n))
            a = np.maximum(a, 1e-12)
            a = a / a.sum()
            momentum = np.zeros(n)

        c_max = c.max()

        idx_n = np.arange(n)

        a_new = np.maximum(a_new, 0.0)
```
