    s = float(parent.sum())

        s = float(n)
    f = parent / s

    def convolve_full(a):
        F = rfft(a, fsize)
        c = irfft(F * F, fsize)[:2 * n - 1]
        return np.maximum(c, 0.0)

    def obj(a):
        return float(2.0 * n * convolve_full(a).max())

    best_score = obj(f)
    best_f = f.copy()

    def remaining():
        return deadline - time.monotonic()

    phi = (np.sqrt(5.0) - 1.0) / 2.0
    prev_sc = best_score
    no_progress = 0

    for it in range(10000):
        if remaining() < 180.0:
            break

        M = float(convolve_full(f).max())
        sc = 2.0 * n * M
        if sc < best_score - 1e-14:
            best_score = sc
            best_f = f.copy()

        # Build constraint matrix: B[k, j] = f[j] if 0 <= k-j < n else 0
        B = np.zeros((2 * n - 1, n), dtype=np.float64)
        for j in range(n):
            B[j:j + n, j] = f

        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

                          options={'presolve': True, 'time_limit': 30})
        except Exception:
            break

            break

            break
