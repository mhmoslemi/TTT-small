    if top_k >= n_full if (n_full := 2 * n - 1) else n:
        top_k = n
    # Use argpartition for speed
    if top_k >= len(b):
        idxs = np.argsort(-b)[:min(top_k, len(b))]
    else:
        part = np.argpartition(-b, top_k)[:top_k]
        idxs = part[np.argsort(-b[part])]
    sg = np.zeros(n)
    for p in idxs:
        p = int(p)
        for j in range(n):
            idx = p - j
            if 0 <= idx < n:
                sg[j] += 2.0 * a[idx]
    sg /= len(idxs)
    return sg, b


def _subgrad_step(a, eta, top_k=5):
    n = a.size
    sg, b = _subgrad_dir(a, top_k)
    sgnorm = float(np.linalg.norm(sg))
    if sgnorm < 1e-15:
        return a
    a_new = a - eta * sg / sgnorm * (float(np.std(a)) + 1e-6)
    a_new = np.clip(a_new, 0.0, None)
    s = float(a_new.sum())
    if s < 1e-15:
        return a
    return a_new / s
