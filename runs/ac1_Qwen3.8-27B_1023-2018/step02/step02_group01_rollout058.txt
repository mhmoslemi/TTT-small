            a = a / a.sum()

    return best_a


def _lp_optimize_aggressive(a0, stop_time, max_iters=20000):
    """More aggressive LP with relaxed constraints and dual info."""
    n = a0.size
    a = np.clip(np.asarray(a0, dtype=np.float64).copy(), 0.0, 1000.0)
    s = a.sum()
    if s <= 0:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    best_M = float(_conv_sq(a).max())
    best_a = a.copy()
    rng = np.random.default_rng(777)
    stagnant = 0

    for it in range(max_iters):
        if time.monotonic() > stop_time - 2.0:
            break

        b = _conv_sq(a)
        M = float(b.max())
        if M < best_M * (1.0 - 1e-10):
            best_M = M
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1

        if stagnant > 200:
            a = a * np.exp(rng.normal(0, 0.1, n))
            a = np.clip(a, 1e-8, 1000.0)
            a = a / a.sum()
            stagnant = 0
            continue

        # Use slightly relaxed M to get a more useful direction
        M_relax = M * (1.0 + 1e-5)
        g = _lp_best_resp(a, M_relax, timeout_s=2.0)
        if g is None:
            break
        g = g / g.sum()

        # Try multiple step sizes
        improved = False
        for t in [1.0, 0.7, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05]:
            cand = (1.0 - t) * a + t * g
            cand = np.clip(cand, 0.0, 1000.0)
            cs = cand.sum()
            if cs < 1e-6:
                continue
            cand = cand / cs
            cM = float(_conv_sq(cand).max())
            if cM < M * (1.0 - 1e-10):
                a = cand
                improved = True
                break

        if not improved:
            # Subgradient step on the max lag
            max_idx = int(np.argmax(b))
            # Direction to reduce (a*a)[max_idx]: decrease a[j] proportional to a[max_idx-j]
            direction = np.zeros(n)
            for j in range(n):
                k = max_idx - j
                if 0 <= k < 2 * n - 1:
                    direction[j] = -a[k]
            a = a + 0.01 * direction
            a = np.clip(a, 1e-8, 1000.0)
            a = a / a.sum()

    return best_a


def _smooth_gradient_step(a, p=20, step=0.1, stop_time=None):
    """One step of gradient descent on Lp-smoothed max."""
    n = a.size
    b = _conv_sq(a)
    b_pos = np.maximum(b, 1e-15)
    
    # f_p = (sum b_k^p)^(1/p)
    f_p = np.sum(b_pos ** p) ** (1.0 / p)
    if f_p < 1e-15:
        return a
    
    # Gradient: 2C * sum_k b_k^(p-1) * a[k-j] for valid k-j
    # grad[j] = 2 * (sum b^p)^(1/p - 1) * sum_k b_k^(p-1) * a[k-j]
    C = 2.0 * (np.sum(b_pos ** p)) ** (1.0 / p - 1.0)
    
    w = b_pos ** (p - 1)  # length 2n-1
    grad = np.zeros(n)
    for j in range(n):
        # sum over k where 0 <= k-j < n => k in [j, j+n-1]
        klo = j
        khi = min(j + n, 2 * n - 1)
        grad[j] = 2.0 * np.sum(w[klo:khi] * a[0:khi - klo])
    grad *= C / (a.sum() ** 2)  # normalize by sum^2
    
    # Projected gradient step
    a_new = a - step * grad
    a_new = np.clip(a_new, 0.0, 1000.0)
    a_new = a_new / a_new.sum()
    return a_new


def _smooth_optimize(a0, stop_time, max_iters=5000, p=20):
    """Optimize using Lp-smoothed objective with adaptive step."""
    n = a0.size
    a = np.clip(np.asarray(a0, dtype=np.float64).copy(), 0.0, 1000.0)
    s = a.sum()
    if s <= 0:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    
    best_M = float(_conv_sq(a).max())
    best_a = a.copy()
    step = 0.5
    stagnant = 0
    
    for it in range(max_iters):
        if time.monotonic() > stop_time - 2.0:
            break
        
        b = _conv_sq(a)
        M = float(b.max())
        if M < best_M * (1.0 - 1e-10):
            best_M = M
            best_a = a.copy()
            stagnant = 0
            step = min(step * 1.2, 2.0)
        else:
            stagnant += 1
            step = max(step * 0.8, 0.01)
        
        if stagnant > 200:
            break
        
        a_new = _smooth_gradient_step(a, p=p, step=step)
        
        # Check improvement
        cM = float(_conv_sq(a_new).max())
        if cM < M * (1.0 - 1e-12):
            a = a_new
        else:
            step *= 0.7
    
    return best_a


def _try_length(a_best, n_target, stop_time):
    """Resample/reconstruct at a different length and optimize."""
    n_orig = a_best.size
    if n_target == n_orig:
        return a_best
    
    # Interpolate
    x_orig = np.linspace(0, 1, n_orig)
    x_new = np.linspace(0, 1, n_target)
    a_new = np.interp(x_new, x_orig, a_best)
    a_new = np.clip(a_new, 0.0, 1000.0)
    s = a_new.sum()
    if s > 0:
        a_new = a_new / s
    
    result = _lp_optimize(a_new, stop_time=min(stop_time, time.monotonic() + 20))
    return result


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = float(budget_s)
    deadline = t0 + budget - 10.0  # 10s headroom
    
    # Get parent
    parent = _get_parent()
    n_parent = parent.size
    
    # Initialize best
    best_a = np.clip(parent.copy(), 0.0, 1000.0)
    s = best_a.sum()
    if s > 0:
        best_a = best_a / s
    else:
        best_a = np.ones(191) / 191.0
    
    try:
        best_score = float(evaluate_sequence([float(x) for x in best_a]))
    except Exception:
        best_score = float('inf')
    
    rng = np.random.default_rng(seed)
    
    # Phase 1: LP optimization from parent (main workhorse)
    phase1_end = t0 + min(budget * 0.40, 480.0)
    if time.monotonic() < phase1_end - 5:
        result = _lp_optimize(best_a, phase1_end, max_iters=15000)
        sc = _score_quick(result)
        if np.isfinite(sc) and sc < best_score:
            best_a = result
            best_score = sc
    
    # Phase 2: Aggressive LP (relaxed constraints)
    phase2_end = t0 + min(budget * 0.55, 650.0)
    if time.monotonic() < phase2_end - 5:
        result = _lp_optimize_aggressive(best_a, phase2_end, max_iters=20000)
        sc = _score_quick(result)
        if np.isfinite(sc) and sc < best_score:
            best_a = result
            best_score = sc
    
    # Phase 3: Smooth gradient refinement
    phase3_end = t0 + min(budget * 0.65, 840.0)
    if time.monotonic() < phase3_end - 5:
        for p_val in [30, 20, 10]:
            if time.monotonic() > phase3_end - 10:
                break
            result = _smooth_optimize(best_a, min(phase3_end, time.monotonic() + 60),
                                      max_iters=3000, p=p_val)
            sc = _score_quick(result)
            if np.isfinite(sc) and sc < best_score:
                best_a = result
                best_score = sc
    
    # Phase 4: Another round of LP on the refined solution
    phase4_end = t0 + min(budget * 0.75, 960.0)
    if time.monotonic() < phase4_end - 5:
        result = _lp_optimize(best_a, phase4_end, max_iters=15000)
        sc = _score_quick(result)
        if np.isfinite(sc) and sc < best_score:
            best_a = result
            best_score = sc
    
    # Phase 5: Try nearby lengths
    phase5_end = t0 + min(budget * 0.88, 1130.0)
    if time.monotonic() < phase5_end - 30:
        for n_try in [n_parent - 1, n_parent + 1, n_parent - 2, n_parent + 2]:
            if time.monotonic() > phase5_end - 10:
                break
            if n_try < 4:
                continue
            alloc = min(25.0, phase5_end - time.monotonic() - 5)
            if alloc < 5:
                continue
            try:
                result = _try_length(best_a, n_try, t0 + alloc)
                sc = _score_quick(result)
                if np.isfinite(sc) and sc < best_score:
                    best_a = result
                    best_score = sc
            except Exception:
                pass
    
    # Phase 6: Final polish LP
    phase6_end = deadline - 5.0
    if time.monotonic() < phase6_end - 5:
        result = _lp_optimize(best_a, phase6_end, max_iters=10000)
        sc = _score_quick(result)
        if np.isfinite(sc) and sc < best_score:
            best_a = result
            best_score = sc
    
    # Final validation with exact evaluator
    try:
        final_score = float(evaluate_sequence([float(x) for x in best_a]))
    except Exception:
        final_score = float('inf')
    
    if not np.isfinite(final_score) or final_score < 0:
        # Fallback: uniform sequence
        return [1.0] * 32
    
    # Verify best_a is valid
    if time.monotonic() > deadline - 3.0:
        pass  # Just return what we have
    
    return [float(x) for x in best_a]
