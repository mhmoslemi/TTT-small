import numpy as np
import time
import math
from scipy.optimize import linprog


def _build_A(a):
    """Build convolution constraint matrix A: (2n-1) x n."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _solve_lp(a, M):
    """Solve: max sum(b) s.t. (a*b)_k <= M for all k, b >= 0."""
    n = len(a)
    A = _build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='simplex')
            if res.success and res.x is not None:
                return np.maximum(res.x, 0.0)
        except Exception:
            pass
    return None


def _lp_optimize(a_init, deadline, start_t=0.3, max_iters=50000):
    """Run LP-based iterations to minimize max(a*a) with sum(a)=1."""
    n = len(a_init)
    a = np.maximum(a_init, 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_M = float(np.convolve(a, a).max())

    t = start_t
    stuck = 0
    it = 0

    while it < max_iters:
        if time.monotonic() > deadline:
            break
        it += 1

        b_conv = np.convolve(a, a)
        M = float(b_conv.max())

        b_star = _solve_lp(a, M)
        if b_star is None:
            break

        sum_b = float(b_star.sum())
        if sum_b < 1e-15:
            break

        b_star = b_star / sum_b

        a_new = (1.0 - t) * a + t * b_star
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        b_new = np.convolve(a_new, a_new)
        M_new = float(b_new.max())

        if M_new < best_M - 1e-14:
            best_M = M_new
            best_a = a_new.copy()
            stuck = 0
            t = min(0.95, t * 1.01)
        else:
            stuck += 1
            t = max(0.03, t * 0.97)
            if stuck > 200:
                t = 0.3
                stuck = 0

        a = a_new

    return best_a, best_M


def _gradient_polish(a_init, deadline, max_iters=50000):
    """Polish using projected gradient on the simplex."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()

    best_a = a_init.copy()
    best_M = float(np.convolve(a_init, a_init).max())

    lr = 0.5
    mu = 0.9
    m = np.zeros(n)
    v = np.zeros(n)
    tstep = 0
    stuck = 0

    while time.monotonic() < deadline and max_iters > 0:
        max_iters -= 1
        a = np.exp(np.clip(z, -50, 50))
        a = a / a.sum()

        b_conv = np.convolve(a, a)
        k_star = int(np.argmax(b_conv))
        M = b_conv[k_star]

        # Gradient of b_{k*} w.r.t. a: grad_m = 2 * a_{k*-m} (where a_i=0 outside [0,n-1])
        grad = np.zeros(n)
        for m_idx in range(n):
            idx = k_star - m_idx
            if 0 <= idx < n:
                grad[m_idx] = 2.0 * a[idx]
        # Project onto simplex tangent: subtract mean
        grad = grad - grad.mean()

        tstep += 1
        m = mu * m + (1 - mu) * grad
        v = 0.999 * v + 0.001 * grad * grad
        m_hat = m / (1 - mu ** tstep)
        v_hat = v / (1 - 0.999 ** tstep)
        update = m_hat / (np.sqrt(v_hat) + 1e-8)
        umax = float(np.abs(update).max()) if update.size else 0.0
        if umax < 1e-18:
            broken = True
        else:
            update = update / umax

        z = z - lr * update
        z = z - z.mean()
        z = np.clip(z, -50, 50)

        a2 = np.exp(z)
        a2 = a2 / a2.sum()
        b2 = np.convolve(a2, a2)
        M2 = float(b2.max())

        if M2 < best_M - 1e-14:
            best_M = M2
            best_a = a2.copy()
            stuck = 0
        else:
            stuck += 1
            lr = max(0.05, lr * 0.95)

                z = z - z.mean()
                m = np.zeros(n)
                v = np.zeros(n)
                lr = 0.4
                stuck = 0
                tstep = 0

    return best_a, best_M
