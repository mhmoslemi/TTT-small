def project_simplex(v, s):
    n=v.size
    v=np.asarray(v,float)
    # If v contains nan, fallback
    u=np.sort(v)[::-1]
    cv=np.cumsum(u)
    # Find rho
    # condition u * np.arange(1,n+1) > cv - s
    conds = u * np.arange(1,n+1) - (cv - s)
    rho = np.nonzero(conds > 0)[0]
    if rho.size == 0:
        return np.full(n, s/n)
    r = rho[-1]
    theta = (cv[r] - s)/(r+1)
    w = v - theta
    w[w < 0] = 0
    ss = w.sum()
    if ss <= 0: return np.full(n,s/n)
    w *= s/ss
    return w
```
Standard.

    def project_simplex(v, s):
        v = np.asarray(v, dtype=np.float64).ravel()
        if v.size != n: ...
        if not np.isfinite(v).all(): return np.full(n, s/n)
        u = np.sort(v)[::-1]
        cv = np.cumsum(u)
        r = np.arange(1, n+1, dtype=np.float64)
        cond = u * r - (cv - s)
        idx = np.nonzero(cond > 0.0)[0]
        if idx.size == 0:
            return np.full(n, s/n)
        rho = idx[-1]
        theta = (cv[rho] - s) / (rho + 1.0)
        w = v - theta
        w = np.maximum(w, 0.0)
        ss = float(w.sum())
        if ss <= 1e-15:
            return np.full(n, s/n)
        w *= (s / ss)
        return w
```

        parent = np.ones(192, dtype=np.float64)

    MAX_N = 512
    if parent.size > MAX_N:
        old_n = int(parent.size)
        new_n = MAX_N
        old_idx = np.arange(old_n, dtype=np.float64)
        new_idx = np.linspace(0.0, old_n - 1.0, new_n)
        parent = np.interp(new_idx, old_idx, parent)

    n = int(parent.size)
    if n < 2:
        n = 2
        parent = np.ones(n, dtype=np.float64)

    # Use the normalization sum(a) = sqrt(2n).  Then the evaluator score
    # is equal to max(a*a) up to floating-point roundoff.
    S = math.sqrt(2.0 * n)
    m_len = 2 * n - 1
    fsize = int(next_fast_len(m_len))

    # P_idx[i, j] = i + j, used to build the gradient of weighted
    # self-convolution terms: c_i = (a*a)_i.
    P_idx = np.add.outer(np.arange(n, dtype=np.int64), np.arange(n, dtype=np.int64))

    # ------------------------------------------------------------------
    # Basic helpers
    # ------------------------------------------------------------------
    def normalize(v):
        v = np.asarray(v, dtype=np.float64).ravel()
        if v.size != n:
            tmp = np.zeros(n, dtype=np.float64)
            take = min(n, v.size)
            if take > 0:
                tmp[:take] = v[:take]
            v = tmp
        v = np.where(np.isfinite(v), v, 0.0)
        v = np.maximum(v, 0.0)
        ss = float(v.sum())
        if ss < 1e-15:
            return np.full(n, S / n, dtype=np.float64)
        v *= (S / ss)
        return v

    def conv_vec(a):
        Fa = rfft(a, fsize)
        c = irfft(Fa * Fa, n=fsize)[:m_len]
        return np.maximum(c, 0.0)

    def M_of(a):
        c = conv_vec(a)
        if c.size == 0:
            return 0.0
        return float(c.max())

    def exact_array_score(a):
        try:
            arr = np.asarray(a, dtype=np.float64).ravel()
            if arr.size == 0:
                return float("inf")
            arr = np.where(np.isfinite(arr), arr, 0.0)
            arr = np.clip(arr, 0.0, 1000.0)
            if float(arr.sum()) < 0.01:
                return float("inf")
            return float(evaluate_sequence([float(x) for x in arr]))
        except Exception:
            return float("inf")

    # ------------------------------------------------------------------
    # Best-candidate bookkeeping
    # ------------------------------------------------------------------
    best_score = float("inf")
    best_a = None
    best_M = float("inf")
    pool = []  # list of (score, numpy array)

    def consider_candidate(a, add_pool=True):
        nonlocal best_score, best_a, best_M
        try:
            a = normalize(a)
            if not np.isfinite(a).all():
                return
            sc = exact_array_score(a)
            if not np.isfinite(sc):
                return
            if add_pool:
                dup = False
                for i, (s, arr) in enumerate(pool):
                    if np.linalg.norm(a - arr) < max(1e-7 * S, 1e-9):
                        if sc < s - 1e-15:
                            pool[i] = (sc, arr.copy())
                        dup = True
                        break
                if not dup:
                    pool.append((sc, a.copy()))
                    pool.sort(key=lambda z: z[0])
                    if len(pool) > 24:
                        pool.pop()
            if sc < best_score - 1e-13:
                best_score = sc
                best_a = a.copy()
                best_M = M_of(a)
        except Exception:
            return

    # ------------------------------------------------------------------
    # LP direction from the "state of the art" iteration
    # ------------------------------------------------------------------
    def build_B(a):
        B = np.zeros((m_len, n), dtype=np.float64)
        for j in range(n):
            B[j:j + n, j] = a
        return B

    def lp_direction(a, M, M_factor=1.0, time_limit=0.3):
        try:
            Mcur = float(M)
            if not np.isfinite(Mcur) or Mcur <= 1e-14:
                return None
            Mb = max(1e-12, Mcur * float(M_factor))
            B = build_B(a)
            g_ub = max(
                float(np.max(a)) * 1.1 + 1e-8,
                3.0 * math.sqrt(max(Mb, 0.0)) + 0.5,
                10.0 * S / n
            )
            bounds = [(0.0, float(g_ub))] * n
            try:
                res = linprog(
                    -np.ones(n, dtype=np.float64),
                    A_ub=B,
                    b_ub=np.full(m_len, Mb, dtype=np.float64),
                    bounds=bounds,
                    method="highs",
                    options={"time_limit": float(time_limit), "presolve": True},
                )
            except Exception:
                return None
            x = getattr(res, "x", None)
            if x is None:
                return None
            g = np.asarray(x, dtype=np.float64).ravel()
            if g.size != n or not np.isfinite(g).all():
                return None
            g = np.maximum(g, 0.0)
            V = float(g.sum())
            if V < 1e-12:
                return None
            g *= (S / V)
            if not np.isfinite(g).all():
                return None
            if float(np.abs(g - a).sum()) < 1e-7 * S:
                return None
            return g
        except Exception:
            return None

    # ------------------------------------------------------------------
    # Exact line search along a -> (1-t)a + t g using FFT structure
    # ------------------------------------------------------------------
    def line_search(a, g, Mcur):
        try:
            a = normalize(a)
            g = normalize(g)
            Fa = rfft(a, fsize)
            Fg = rfft(g, fsize)
            ca = np.maximum(irfft(Fa * Fa, n=fsize)[:m_len], 0.0)
            cag = np.maximum(irfft(Fa * Fg, n=fsize)[:m_len], 0.0)
            cg = np.maximum(irfft(Fg * Fg, n=fsize)[:m_len], 0.0)

            def vals(ts):
                t = np.asarray(ts, dtype=np.float64).reshape(-1, 1)
                one = 1.0 - t
                c = (one * one) * ca[None, :] + (2.0 * t * one) * cag[None, :] + (t * t) * cg[None, :]
                c = np.maximum(c, 0.0)
                return c.max(axis=1)

            ts = np.linspace(0.0, 1.0, 201, dtype=np.float64)
            v = vals(ts)
            v[0] = float(Mcur)
            i1 = int(np.argmin(v))
            if v[i1] >= float(Mcur) - 1e-12:
                return None
            t0 = float(ts[i1])

            lo = max(0.0, t0 - 0.05)
            hi = min(1.0, t0 + 0.05)
            ts2 = np.linspace(lo, hi, 101, dtype=np.float64)
            v2 = vals(ts2)
            i2 = int(np.argmin(v2))
            tb = float(ts2[i2])
            mb = float(v2[i2])
            if mb >= float(Mcur) - 1e-12:
                return None
            return tb, mb
        except Exception:
            return None

    # ------------------------------------------------------------------
    # Simplex projection and projected-gradient / LSE refinement
    # ------------------------------------------------------------------
    def project_simplex(v, s):
        v = np.asarray(v, dtype=np.float64).ravel()
        if v.size != n:
            tmp = np.zeros(n, dtype=np.float64)
            take = min(n, v.size)
            if take > 0:
                tmp[:take] = v[:take]
            v = tmp
        if not np.isfinite(v).all():
            return np.full(n, s / n, dtype=np.float64)
        u = np.sort(v)[::-1]
        cv = np.cumsum(u)
        r = np.arange(1, n + 1, dtype=np.float64)
        cond = u * r - (cv - s)
        idx = np.nonzero(cond > 0.0)[0]
        if idx.size == 0:
            return np.full(n, s / n, dtype=np.float64)
        rho = int(idx[-1])
        theta = (cv[rho] - s) / float(rho + 1)
        w = v - theta
        w = np.maximum(w, 0.0)
        ww = float(w.sum())
        if ww <= 1e-15:
            return np.full(n, s / n, dtype=np.float64)
        w *= (s / ww)
        return w

    def grad_refine(a, M, dl, max_iters=4000):
        try:
            a = normalize(a)
            M = M_of(a)
            consider_candidate(a)
            alpha = 0.1
            tau = 0.05 * max(M, 1e-6)
            stuck = 0

            for it in range(max_iters):
                if time.monotonic() > dl:
                    break

                c = conv_vec(a)
                Mc = float(c.max())
                if Mc < M - 1e-12:
                    M = Mc
                    consider_candidate(a)

                if it % 4 == 0:
                    thresh = Mc * 0.965
                    p = (c >= thresh).astype(np.float64)
                    if float(p.sum()) < 1e-12:
                        p = np.zeros(m_len, dtype=np.float64)
                        p[int(np.argmax(c))] = 1.0
                else:
                    tau = max(0.015 * M, min(0.25 * M, tau))
                    z = c / max(tau, 1e-12)
                    zm = float(z.max())
                    ez = np.exp(z - zm)
                    ez = np.where(np.isfinite(ez), ez, 0.0)
                    ssum = float(ez.sum())
                    if ssum < 1e-300:
                        p = np.zeros(m_len, dtype=np.float64)
                        p[int(np.argmax(c))] = 1.0
                    else:
                        p = ez / ssum

                # Gradient of sum_i p_i * (a*a)_i.
                h = 2.0 * (p[P_idx] @ a)
                lam = float(np.dot(a, h)) / S
                d = h - a * lam
                nd = float(np.linalg.norm(d))

                if nd < 1e-14:
                    a = normalize(np.maximum(a + rng.normal(0.0, 0.02 * S / n, n), 1e-10))
                    M = M_of(a)
                    stuck += 1
                    continue

                bm = Mc
                ba = a
                for mult in (alpha, alpha * 0.5, alpha * 2.0, alpha * 0.1):
                    y = a - mult * d
                    w = project_simplex(y, S)
                    mw = M_of(w)
                    if mw < bm - 1e-13:
                        bm = mw
                        ba = w

                if bm < Mc - 1e-13:
                    a = ba
                    M = bm
                    stuck = 0
                    alpha = min(2.0, alpha * 1.07)
                    consider_candidate(a)
                else:
                    stuck += 1
                    alpha = max(0.003, alpha * 0.8)
                    tau = min(0.25 * M, tau * 1.25)
                    if stuck > 80:
                        a = normalize(np.maximum(a + rng.normal(0.0, 0.05 * S / n, n), 1e-10))
                        M = M_of(a)
                        stuck = 20
                        consider_candidate(a)
            consider_candidate(a)
            return a, M
        except Exception:
            return a, M

    # ------------------------------------------------------------------
    # L-BFGS-B on a smooth LSE surrogate, in a simplex-compatible log map
    # ------------------------------------------------------------------
    def lbfgs_refine(a, M, dl):
        try:
            a = normalize(a)
            M = M_of(a)
            consider_candidate(a)
            cur_a = a
            cur_M = M

            for tau in (0.02, 0.05, 0.10):
                if time.monotonic() > dl - 3.0:
                    break

                def ftheta(theta, tau=tau):
                    theta = np.asarray(theta, dtype=np.float64).ravel()
                    z = theta - float(theta.max())
                    ea = np.exp(z)
                    ssum = float(ea.sum())
                    if ssum <= 0.0:
                        return 1e300
                    aa = S * ea / ssum
                    c = conv_vec(aa)
                    zc = c / tau
                    zm = float(zc.max())
                    tmp = zc - zm
                    return float(tau * (zm + np.log(np.sum(np.exp(tmp)))))

                def gtheta(theta, tau=tau):
                    theta = np.asarray(theta, dtype=np.float64).ravel()
                    z = theta - float(theta.max())
                    ea = np.exp(z)
                    ssum = float(ea.sum())
                    if ssum <= 0.0:
                        return np.zeros(n, dtype=np.float64)
                    aa = S * ea / ssum

                    c = conv_vec(aa)
                    zc = c / tau
                    zm = float(zc.max())
                    ez = np.exp(zc - zm)
                    ez = np.where(np.isfinite(ez), ez, 0.0)
                    ssum2 = float(ez.sum())
                    if ssum2 <= 1e-300:
                        p = np.zeros(m_len, dtype=np.float64)
                        p[int(np.argmax(c))] = 1.0
                    else:
                        p = ez / ssum2

                    h = 2.0 * (p[P_idx] @ aa)
                    lam = float(np.dot(aa, h)) / S
                    return h - aa * lam

                theta0 = np.log(np.maximum(cur_a, 1e-8 * S / n)) - np.log(S / n)
                rem = max(5.0, dl - time.monotonic())
                maxit = int(min(70, max(20, rem / 2.0)))
                try:
                    res = minimize(
                        ftheta,
                        theta0,
                        jac=gtheta,
                        method="L-BFGS-B",
                        bounds=[(-30.0, 30.0)] * n,
                        options={
                            "maxiter": maxit,
                            "ftol": 1e-12,
                            "gtol": 1e-8,
                            "maxls": 20,
                        },
                    )
                except Exception:
                    res = None

                if res is not None and getattr(res, "x", None) is not None:
                    theta = np.asarray(res.x, dtype=np.float64).ravel()
                    if theta.size == n and np.isfinite(theta).all():
                        z = theta - float(theta.max())
                        ea = np.exp(z)
                        ssum = float(ea.sum())
                        if ssum > 0.0:
                            aa = S * ea / ssum
                        else:
                            aa = cur_a
                        aa = normalize(aa)
                        M2 = M_of(aa)
                        if np.isfinite(M2) and M2 < cur_M - 1e-12:
                            cur_a = aa
                            cur_M = M2
                            consider_candidate(aa)

            return cur_a, cur_M
        except Exception:
            return a, M

    # ------------------------------------------------------------------
    # Local optimization routine for one start
    # ------------------------------------------------------------------
    def optimize_start(start_a, phase_dl, do_lbfgs=True):
        try:
            a = normalize(start_a)
            M = M_of(a)
            consider_candidate(a)
            no_improve = 0

            while time.monotonic() < phase_dl - 30.0:
                rem = phase_dl - time.monotonic()
                if rem < 15.0:
                    break
                tl = min(0.8, max(0.05, rem / 120.0))

                g = lp_direction(a, M, 1.0, tl)
                improved = False

                if g is None and rem > 40.0:
                    fac = 0.98
                    if np.isfinite(best_score) and best_score < M * 0.999:
                        fac = max(0.85, min(fac, best_score / (M + 1e-12)))
                    g = lp_direction(a, M, fac, min(tl, 0.4))

                if g is not None:
                    # Occasionally the LP endpoint by itself is useful.
                    Mg = M_of(g)
                    if Mg < M - 1e-12 or (np.isfinite(best_score) and Mg < best_score - 1e-12):
                        consider_candidate(g)

                    ls = line_search(a, g, M)
                    if ls is not None:
                        t, mv = ls
                        a_new = normalize((1.0 - t) * a + t * g)
                        M_new = M_of(a_new)
                        if M_new < M - 1e-12:
                            a = a_new
                            M = M_new
                            no_improve = 0
                            improved = True
                            consider_candidate(a)

                if not improved:
                    no_improve += 1
                    if no_improve > 70:
                        break
                # If LP is stuck, nudge the iterate to escape weak fixed points.
                if no_improve > 30 and no_improve % 10 == 0:
                    a = normalize(np.maximum(a + rng.normal(0.0, 0.02 * S / n, n), 1e-10))
                    M = M_of(a)
                    no_improve = 0
                    consider_candidate(a)

            if time.monotonic() < phase_dl - 20.0:
                a, M = grad_refine(a, M, phase_dl - 20.0)
            if do_lbfgs and time.monotonic() < phase_dl - 10.0:
                a, M = lbfgs_refine(a, M, phase_dl - 10.0)

            consider_candidate(a)
            return a, M
        except Exception:
            return start_a, M_of(start_a)

    # ------------------------------------------------------------------
    # Final stochastic local search
    # ------------------------------------------------------------------
    def stochastic_refine(a, M, dl):
        try:
            a = normalize(a)
            M = M_of(a)
            consider_candidate(a)
            sigma = 0.04
            stuck = 0

            while time.monotonic() < dl:
                r = rng.random()

                if r < 0.30:
                    lag = np.log(np.maximum(a, 1e-8))
                    z = lag + sigma * rng.normal(0.0, 1.0, n)
                    z -= float(z.max())
                    ez = np.exp(z)
                    ssum = float(ez.sum())
                    cand = S * ez / ssum if ssum > 0.0 else a
                elif r < 0.55:
                    cand = a.copy()
                    k = int(rng.integers(0, n))
                    width = int(rng.integers(1, max(2, n // 20) + 1))
                    lo = max(0, k - width)
                    hi = min(n, k + width + 1)
                    if hi > lo:
                        cand[lo:hi] *= math.exp(float(rng.normal(0.0, sigma * 0.7)))
                    cand = normalize(cand)
                elif r < 0.75 and len(pool) > 1:
                    b = pool[int(rng.integers(0, len(pool)))][1]
                    s = float(rng.uniform(0.05, 0.5))
                    cand = normalize((1.0 - s) * a + s * b)
                else:
                    s = float(rng.uniform(0.05, 0.5))
                    cand = normalize((1.0 - s) * a + s * a[::-1])

                if not np.isfinite(cand).all():
                    continue
                mc = M_of(cand)
                if mc < M - 1e-13:
                    a = cand
                    M = mc
                    consider_candidate(a)
                    stuck = 0
                    sigma = max(0.005, sigma * 0.98)
                else:
                    stuck += 1
                    sigma = min(0.25, sigma * 1.05)

                if stuck > 400:
                    sigma = min(0.35, max(0.08, sigma * 2.0))
                    a = normalize(np.maximum(a + rng.normal(0.0, sigma * S / n, n), 1e-10))
                    M = M_of(a)
                    stuck = 0
                    consider_candidate(a)

            consider_candidate(a)
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Build a diverse initial pool
    # ------------------------------------------------------------------
    parent_norm = normalize(parent)
    uniform = np.full(n, S / n, dtype=np.float64)

    consider_candidate(parent_norm)
    consider_candidate(uniform)
    consider_candidate(parent_norm[::-1])

    if n > 2:
        i = np.arange(n, dtype=np.float64)
        x = i / float(max(1, n - 1))

        # Antiperiodic / periodic shape perturbations.
        for eps in (0.20, 0.40, 0.70, 1.00):
            base = 1.0 + eps * np.cos(2.0 * np.pi * i / n)
            consider_candidate(np.maximum(base, 1e-8))

        base = 0.5 + 0.5 * np.cos(2.0 * np.pi * i / n)
        for p in (1.0, 2.0):
            consider_candidate(np.maximum(base ** p, 1e-8))

        # Mid-bell shapes.
        base = np.sin(np.pi * (i + 0.5) / n)
        for p in (0.5, 1.0, 2.0, 4.0):
            consider_candidate(np.maximum(base ** p, 1e-8))

        # Ramps.
        for alpha in (-3.0, -1.0, -0.3, 0.3, 1.0, 3.0):
            consider_candidate(np.exp(alpha * x))

        # Blends with uniform.
        for al in (0.2, 0.5, 0.8):
            consider_candidate(al * parent_norm + (1.0 - al) * uniform)

        # Smooth random low-frequency shapes.
        for _ in range(12):
            z = np.zeros(n, dtype=np.float64)
            for freq in range(1, 4):
                amp = float(rng.uniform(0.2, 1.0))
                ph = float(rng.uniform(0.0, 2.0 * np.pi))
                z += amp * np.cos(2.0 * np.pi * freq * i / n + ph)
            denom = 1.0 + float(np.max(np.abs(z))) + 1e-9
            base = 1.0 + z / denom
            consider_candidate(np.maximum(base, 1e-8))

    if best_a is None:
        best_a = uniform.copy()
        best_score = exact_array_score(best_a)
        best_M = M_of(best_a)
        pool = [(best_score, best_a.copy())]

    # ------------------------------------------------------------------
    # Run optimization from the best diverse starts
    # ------------------------------------------------------------------
    starts = []
    for sc, arr in pool:
        dup = False
        for s in starts:
            if np.linalg.norm(arr - s) < max(1e-6 * S, 1e-8):
                dup = True
                break
        if not dup:
            starts.append(arr)
        if len(starts) >= 3:
            break

    search_end = max(t0 + 5.0, min(deadline, deadline - 30.0))

    for idx, st in enumerate(starts):
        now = time.monotonic()
        if now >= search_end:
            break
        rem = search_end - now
        if idx == 0:
            alloc = 0.55 * rem
        elif idx == 1:
            alloc = 0.30 * rem
        else:
            alloc = rem
        phase_dl = min(search_end, now + alloc)
        optimize_start(st, phase_dl, do_lbfgs=(idx < 2))

    # ------------------------------------------------------------------
    # Stochastic polish and final exact validation
    # ------------------------------------------------------------------
    if best_a is not None and time.monotonic() < deadline - 2.0:
        stochastic_refine(best_a, best_M, deadline - 2.0)

    final_candidates = []
    if best_a is not None:
        final_candidates.append(best_a)
    for sc, arr in pool[:16]:
        final_candidates.append(arr)
    final_candidates.append(parent_norm)

    final_arr = None
    final_score = float("inf")
    for cand in final_candidates:
        sc = exact_array_score(cand)
        if np.isfinite(sc) and sc < final_score:
            final_score = sc
            final_arr = cand

    if final_arr is None:
        final_arr = np.full(n, S / n, dtype=np.float64)

    final_arr = np.asarray(final_arr, dtype=np.float64).ravel()
    final_arr = np.where(np.isfinite(final_arr), final_arr, 0.0)
    final_arr = np.clip(final_arr, 0.0, 1000.0)
    if final_arr.sum() < 0.01:
        final_arr = np.full(n, S / n, dtype=np.float64)

    final_list = [float(v) for v in final_arr]
    try:
        fs = float(evaluate_sequence(final_list))
    except Exception:
        fs = float("inf")

    if not np.isfinite(fs):
        final_list = [1.0]
        try:
            evaluate_sequence(final_list)
        except Exception:
            pass

    return final_list
