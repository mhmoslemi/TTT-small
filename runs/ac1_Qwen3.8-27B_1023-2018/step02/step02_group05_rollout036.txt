    col_idx = np.repeat(np.arange(n), n)  # each column j appears n times
    row_idx = []

For simplex projection, I'll use the standard algorithm:
```python
def _project_simplex(v):
    """Project v onto the probability simplex {x: x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not any(cond):
        return np.ones(n) / n
    rho = np.max(ind[cond])
    theta = cssv[rho-1] / rho
    return np.maximum(v - theta, 0)
```

def _score_norm(p):
    """Score for normalized p (sum=1): 2n * max(p*p)."""
    n = p.size
    b = np.convolve(p, p)
    return 2.0 * n * float(b.max())

def _build_conv_matrix(p, n):
    """Build sparse convolution matrix T(p) of shape (2n-1, n)."""
    col_idx = np.repeat(np.arange(n), n)
    row_idx = col_idx + np.tile(np.arange(n), n)
    vals = np.tile(p, n)
    return sps.csr_matrix((vals, (row_idx, col_idx)), shape=(2*n-1, n))

def _solve_lp(p, M, n, time_limit=10):
    """Solve: max sum(q) s.t. T(p)q <= M, q >= 0."""
    A = _build_conv_matrix(p, n)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M + 1e-14)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)]*n, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-9})
    except Exception:
        return None
    if res is not None and getattr(res, 'x', None) is not None:
        q = np.asarray(res.x, dtype=np.float64)
        q = np.clip(q, 0, None)
        s = float(q.sum())
        if s > 1e-15 and math.isfinite(s):
            return q / s
    return None

def _project_simplex(v):
    """Project v onto probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n+1, dtype=np.float64)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho = int(np.max(ind[cond]))
    theta = cssv[rho-1] / rho
    return np.maximum(v - theta, 0.0)

def _subgrad_step(p, lr, n):
    """One subgradient step on the simplex."""
    b = np.convolve(p, p)

    grad = np.zeros(n)
    for k in active:
        ki = int(k)
        j_lo = max(0, ki - n + 1)
        j_hi = min(n - 1, ki)
        if j_lo <= j_hi:
            p_idx = ki - np.arange(j_lo, j_hi + 1)
            grad[j_lo:j_hi+1] += 2.0 * p[p_idx]
    grad /= len(active)
    
    p_new = _project_simplex(p - lr * grad)
    return p_new, M
