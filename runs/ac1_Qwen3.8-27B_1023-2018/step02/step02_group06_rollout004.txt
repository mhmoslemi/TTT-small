import math
from scipy.fft import rfft, irfft, next_fast_len
import cvxpy as cp

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(192, dtype=np.float64)
    try:
        p = np.clip(p, 0.0, 1000.0)
    except Exception:
        p = np.ones(192, dtype=np.float64)
    if p.size == 0 or not np.isfinite(p).all():
        p = np.ones(192, dtype=np.float64)
    return p

def _fft_conv_self(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2*n-1]
    np.maximum(c, 0.0, out=c)
    c[~np.isfinite(c)] = 0.0
    return c

def _score_fast(a):
    s = a.sum()
    if s < 1e-15:
        return math.inf
    b = _fft_conv_self(a)
    return 2.0 * len(a) * b.max() / (s * s)

def _build_A(a):
    n = len(a)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        # (a*g)(k) = sum_j a[k-j] * g_j, valid when 0<=k-j<n and 0<=j<n
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            # a[k-j] for j in [j_lo, j_hi]
            # indices of a: k - j for j in [j_lo, j_hi] => [k-j_hi, k-j_lo]
            a_idx = k - np.arange(j_hi - j_lo + 1)
            A[k, j_lo:j_hi+1] = a[a_idx]
    return A

def _solve_lp(a, M, n):
    g_var = cp.Variable(n, nonneg=True)
    A = _build_A(a)
    constraints = [A @ g_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
    for solver in [cp.GLOP, cp.GLPK, cp.CBC]:
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ['optimal', 'optimal_inaccurate']:
