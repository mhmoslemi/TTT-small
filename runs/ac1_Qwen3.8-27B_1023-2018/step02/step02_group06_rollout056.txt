    n = len(a)

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break
        
        # Backtracking line search
        step = step0 / (1 + it * 0.001)
        improved = False
        for _ in range(20):
            a_new = a - step * grad
            a_new = _normalize(a_new)
            buf_new = float(_fft_conv_self(a_new).max())
            if buf_new < buf * (1 - 1e-10 * step):
                a = a_new
                improved = True
                break
            step *= 0.5
        if not improved:
            # Try smaller step
            a = a - 0.001 * grad / gnorm
            a = _normalize(a)
        
        if no_improve > 500:
            break
    
    return best_a, best_score
