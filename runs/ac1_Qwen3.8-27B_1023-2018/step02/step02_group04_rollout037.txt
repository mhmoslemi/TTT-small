def _smooth_max_opt(a0, stop_t, best, rng, beta=100.0, lr=0.01):
    """Optimize using smooth max approximation."""
    a = a0.copy().astype(np.float64)
    a = np.clip(np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0), 0.0, None)
    s = a.sum()

    n = a.size

    _upd(a, best)

            break

        # Smooth max gradient
        bm = b - b.max()
        ew = np.exp(beta * bm)
        w = ew / ew.sum()
        
        # grad = 2 * correlate(w, a, 'full')[n-1:2n-1]
        grad = 2.0 * np.correlate(w, a, mode='full')[n-1:2*n-1]
        
        # Projected gradient step (project onto simplex)
        # a_new = a - lr * grad, then project to {x >= 0, sum(x) = 1}
        a_new = a - lr * grad
        # Project onto simplex
        # Sort a_new
        a_sorted = np.sort(a_new)[::-1]
        cumsum = np.cumsum(a_sorted)
        idx = np.arange(1, n+1)
        mask = (a_sorted - (cumsum - 1) / idx) > 0
        rho = np.sum(mask)
        theta = (cumsum[rho-1] - 1) / rho if rho > 0 else 0
        a_new = np.maximum(a_new - theta, 0)
        a_new = a_new / a_new.sum()
        
        new_loc = _sc(a_new)
        if math.isfinite(new_loc) and new_loc < loc:
            a = a_new
            loc = new_loc
            _upd(a, best)
        else:
            lr *= 0.9
            if lr < 1e-8:
                break
    return a
