def _conv(a, n, fs):
    a = np.asarray(a, dtype=np.float64).ravel()
    F = rfft(a, fs)
    c = irfft(F * F, fs)
    m = 2 * n - 1
    if c.size < m:
        c = np.convolve(a, a)
        return c[:m]
    return c[:m]


def _grad_h(p, a, n, fs2):
    p = np.asarray(p, dtype=np.float64).ravel()
    a = np.asarray(a, dtype=np.float64).ravel()
    m = 2 * n - 1
    if p.size < m:
        fullp = np.zeros(m, dtype=np.float64)
        fullp[:p.size] = p
        p = fullp
    else:
        p = p[:m]
    ar = a[::-1]
    c = irfft(rfft(ar, fs2) * rfft(p, fs2), fs2)
    if c.size < m:
        conv = np.convolve(a, p[::-1])
        return conv[n - 1:2 * n - 1][::-1]
    return c[n - 1:2 * n - 1]

def _lp_blend(a, mask, update_func, hard_deadline):
    try:
        from scipy.optimize import linprog
    except Exception:
        return

    a = _normalize(a)
    if a is None or a.size < 3 or a.size > 256:
        return
    if time.monotonic() > hard_deadline - 25.0:
        return

    n = a.size
    L = 2 * n - 1
    mask = np.asarray(mask, dtype=bool)
    active = np.nonzero(mask)[0]
    if active.size < 2:
        return

    fs = next_fast_len(2 * n - 1)
    b = _conv(a, n, fs)
    b = np.maximum(b, 0.0)
    M = float(np.max(b[active]))
    if not np.isfinite(M) or M <= 1e-15:
        return

    A = np.zeros((active.size, n), dtype=np.float64)
    idx = active
    for j in range(n):
        sel = (idx >= j) & (idx <= j + n - 1)
        if np.any(sel):
            A[sel, j] = a[idx[sel] - j]

    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(active.size, M, dtype=np.float64)

    res = None
    try:
        res = linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=[(0.0, 5.0)] * n,
            method="highs",
            options={"presolve": True},
        )
    except Exception:
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=(0.0, 5.0))
        except Exception:
            res = None

    if res is None:
        return
    if getattr(res, "status", None) != 0:
        return

    g = np.asarray(res.x, dtype=np.float64).ravel()
    g = np.where(np.isfinite(g), g, 0.0)
    g = np.maximum(g, 0.0)
    gs = float(g.sum())
    if gs <= 1e-12:
        return
    g /= gs

    for t in (0.2, 0.4, 0.6, 0.8):
        if time.monotonic() > hard_deadline - 10.0:
            break
        try:
            update_func((1.0 - t) * a + t * g)

            pass

        except Exception:
            pass

    trigger = (math.isfinite(best_score) and best_score < 2.0 - 1e-4) or \
              (math.isfinite(parent_score) and parent_score < 2.0 - 1e-4)

    if not trigger:
        if best_a is None:
            return [1.0]
        cand = [float(x) for x in np.clip(best_a, 0.0, 1000.0)]
        if not cand or not all(np.isfinite(v) for v in cand):
            cand = [1.0]
        return cand

    ref = parent_for_init
    if ref is None or ref.size < 4:
        ref = best_a
    if ref is None or ref.size < 4:
        ref = np.ones(64, dtype=np.float64) / 64
    ref = _normalize(ref)
    if ref is None or ref.size < 4:
        ref = np.ones(64, dtype=np.float64) / 64

    mask_name = _infer_mask(ref)
    if mask_name not in ("off", "off_in", "all"):
        mask_name = "off"

    surrog_end = t0 + 0.68 * total
    local_end = t0 + 0.88 * total
    surrog_end = min(surrog_end, hard_deadline - 10.0)
    local_end = min(local_end, hard_deadline - 5.0)
    if surrog_end < t0 + 5.0:
        surrog_end = hard_deadline - 5.0
    if local_end < surrog_end:
        local_end = surrog_end + max(1.0, 0.1 * max(1.0, total))
    local_end = min(local_end, hard_deadline - 3.0)

    selected = []
    if parent is not None and parent.size > 1:
        selected.append(int(parent.size))
    for L in (128, 192, 256, 384, 512, 96, 64):
        if len(selected) >= 5:
            break
        if L not in selected and 16 <= L <= 512:
            selected.append(L)
    if not selected:
        selected = [64, 128, 192]

    jobs = []
    job_seed_base = int(seed)
    for n in selected[:5]:
        n = int(n)
        if n < 2 or n > 512:
            continue
        if parent is not None and n == parent.size:
            types = [0, 0, 1, 2, 3, 4]
        else:
            types = [0, 1]
        for rep, typ in enumerate(types):
            if time.monotonic() > surrog_end - 2.0:
                break
            try:
                rng = np.random.default_rng(job_seed_base * 1000 + n * 31 + rep * 7 + 11)
                a0 = _init_shape(n, typ, rng, parent_for_init)
                mask = _mask_by_name(n, mask_name)
                if not np.any(mask):
                    continue
                w = 0.0 if (typ == 0 and parent is not None and n == parent.size) else 0.02
                jobs.append(_make_job(n, a0, mask, job_seed_base * 17 + n * 7 + rep, w))
                jobs = jobs[:12]
            except Exception:
                pass

    if jobs:
        order = list(range(len(jobs)))
        while time.monotonic() < surrog_end:
            for j in order:
                now = time.monotonic()
                if now >= surrog_end:
                    break
                end = min(now + 1.5, surrog_end)
                try:
                    _run_job(jobs[j], end, hard_deadline, update_candidate)
                except Exception:
                    pass

    if best_a is not None and 3 <= best_a.size <= 256 and time.monotonic() < local_end - 20.0:
        try:
            _lp_blend(best_a, _mask_by_name(best_a.size, mask_name), update_candidate, hard_deadline)
        except Exception:
            pass

    polish_lengths = []
    if best_a is not None:
        polish_lengths.append(int(best_a.size))
    if parent is not None:
        polish_lengths.append(int(parent.size))
    for L in selected:
        if L not in polish_lengths:
            polish_lengths.append(L)
        if len(polish_lengths) >= 4:
            break

    seen = set()
    for n in polish_lengths:
        n = int(n)
        if n in seen or n < 2 or n > 512:
            continue
        seen.add(n)
        if time.monotonic() >= local_end:
            break
        try:
            if best_a is not None:
                init_a = _resize(best_a, n)
            else:
                rngp = np.random.default_rng(31 + n)
                init_a = _init_shape(n, 0, rngp, parent_for_init)
            if parent is not None and n == parent.size:
                pn = _normalize(parent)
                if pn is not None:
                    blended = 0.5 * init_a + 0.5 * pn
                    nb = _normalize(blended)
                    if nb is not None:
                        init_a = nb
            _local_polish(init_a, local_end, hard_deadline, mask_name, update_candidate)
        except Exception:
            pass

    if best_a is None:
        cand = [1.0]
    else:
        bb = _normalize(best_a)
        if bb is None:
            cand = [1.0]
        else:
            cand = [float(x) for x in bb]
            if not all(np.isfinite(v) for v in cand):
                cand = [1.0]

    fs = _exact_score(cand, normalize=False)
    if not np.isfinite(fs) or fs > 1e12:
        cand = [1.0]
        fs = _exact_score(cand, normalize=False)
        if not np.isfinite(fs) or fs > 1e12:
            return [1.0]

    return cand
