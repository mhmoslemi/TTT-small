def _build_A(f, n):

                      bounds=[(0, None)] * n,
                      method='highs',
                      options={"presolve": True, "time_limit": time_limit})

        return None, M
    g = np.asarray(res.x, dtype=np.float64)

    inits.append(np.clip(parent + rng_main.normal(0, 0.3 * np.std(parent), n_p), 0, None) + 0.01)
    # Cosine bump
    inits.append(0.5 + 0.5 * np.cos(2 * math.pi * i_arr / n_p))
    # Multi-freq
    sup = np.ones(n_p) * 0.5
    for fr in (1, 2, 4):
        ph = rng_main.uniform(0, 2 * math.pi)
        sup += 0.3 * np.cos(2 * math.pi * fr * i_arr / n_p + ph)
    inits.append(np.clip(sup, 0, None) + 0.02)

    time_per_init = max(8.0, (phase1_end - t0) / len(inits))

    for idx, init in enumerate(inits):
        if time.monotonic() > phase1_end - 5.0:
            break
        if best_score < 1.502:
            break
        # First init (parent) gets more time
        multiplier = 2.5 if idx == 0 else 1.0
        stop = min(phase1_end - 3.0, time.monotonic() + time_per_init * multiplier)
        _try(n, init, stop, seed + 1000 * (idx + 1))

    # Phase 2: Try other lengths (15% of budget)
    phase2_end = t0 + budget * 0.87
    other_lengths = [256, 320, 352, 384, 448, 512]
    if n_p not in other_lengths:
        other_lengths.append(n_p)
    other_lengths = sorted(set(other_lengths))
    time_per_len = max(5.0, (phase2_end - phase1_end) / max(1, len(other_lengths)))

    for L in other_lengths:
        if time.monotonic() > phase2_end - 5.0:
            break
        if best_score < 1.502:
            break
        if L == n_p:
            continue
        stop = min(phase2_end - 3.0, time.monotonic() + time_per_len)
        if abs(L - len(best_seq)) <= 48 and best_score < 2.0:
            init_l = np.interp(
                np.linspace(0, 1, L), np.linspace(0, 1, len(best_seq)), best_seq
            )
        else:
            init_l = np.interp(
                np.linspace(0, 1, L), np.linspace(0, 1, len(parent)), parent
            )
        _try(L, init_l, stop, seed + 50000 + L)

    # Phase 3: Polish and local search (remaining budget)
    polish_end = deadline
    if time.monotonic() < polish_end - 15.0:
        # Round of LP on the best found
        if time.monotonic() < polish_end - 40.0:
            _try(len(best_seq), best_seq.copy(), min(polish_end - 15.0, time.monotonic() + 30.0), seed + 99999)

        # Local search
        if time.monotonic() < polish_end - 20.0:
            try:
                best_seq, best_score = _local_search(best_seq, min(polish_end - 10.0, time.monotonic() + 40.0), seed + 555)
            except Exception:
                pass

        # Coordinate polish
        if time.monotonic() < polish_end - 5.0:
            try:
                best_seq, best_score = _polish(best_seq, polish_end)
            except Exception:
                pass

    # Final validation
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in fallback]

    if final_sc < best_score - 1e-12:
        best_score = final_sc

    return [float(x) for x in best_seq]
