        s = 1.0
    a = a / s

    b = _conv_fft(a, n, fs)
    best_a = a.copy()
    best_score = 2.0 * n * b.max()
    if not np.isfinite(best_score):
        return best_a

    sigma = 0.15
    stuck = 0
    base = a.copy()

    while time.monotonic() < min(end_time, hard_deadline) - 0.5:
        z = np.log(np.clip(a, 1e-20, None))
        r = float(rng.random())

        if r < 0.30:
            zc = z + rng.normal(0.0, sigma, n)
        elif r < 0.45:
            i = int(rng.integers(0, n))
            w = max(1, min(n, int(rng.integers(1, max(2, n // 20 + 1)))))
            i2 = min(n, i + w)
            zc = z.copy()
            zc[i:i2] += rng.normal(0.0, sigma * 0.7, i2 - i)
        elif r < 0.55:
            # Targeted: reduce peak
            peak_k = int(np.argmax(b))
            gk = np.zeros(n)
            lo = max(0, peak_k - n + 1)
            hi = min(n, peak_k + 1)
            for j in range(lo, hi):
                gk[j] = a[peak_k - j]
            pert = -gk + gk.mean()
            zc = z + sigma * 0.5 * pert / (np.std(pert) + 1e-15)
        elif r < 0.65:
            # Scale
            zc = z * float(rng.uniform(0.5, 1.8)) + float(rng.uniform(-0.1, 0.1))
        elif r < 0.80:
            # Mix with base
            zc = (1.0 - float(rng.uniform(0.1, 0.5))) * z + float(rng.uniform(0.1, 0.5)) * np.log(np.clip(base, 1e-20, None))
        else:
            # Gaussian window perturbation
            center = float(rng.uniform(0, n))
            width = float(rng.uniform(n * 0.05, n * 0.4))
            tt = np.arange(n)
            win = np.exp(-0.5 * ((tt - center) / width) ** 2)
            zc = z + sigma * win * rng.normal(0.0, 1.0, n)

        zc = zc - zc.mean()
        a2 = _softmax(zc)
        b2 = _conv_fft(a2, n, fs)
        s2 = 2.0 * n * b2.max()

        if np.isfinite(s2) and s2 < best_score - 1e-12:
            best_score = s2
            best_a = a2.copy()
            a = a2.copy()
            b = b2
            stuck = 0
        else:
            stuck += 1
            sigma = min(0.5, sigma * 1.04)
            if stuck > 80:
                stuck = 0
                sigma = 0.2
                if rng.random() < 0.3:
                    a = base + rng.normal(0.0, 0.05, n)
                    a = np.clip(a, 1e-6, None)
                    a = a / a.sum()
                    b = _conv_fft(a, n, fs)

    return best_a


def _init_variants(n, rng, parent):
    """Generate diverse initializations."""
    t = (np.arange(n) + 0.5) / n
    x = 2.0 * t - 1.0
    variants = []

    # Uniform
    variants.append(np.ones(n))
    # Triangular
    variants.append(1.0 - np.abs(x))
    # Parabolic
    variants.append(1.0 - x * x)
    # Cosine bell
    variants.append(0.5 * (1.0 + np.cos(np.pi * x)))
    # Cubic taper
    variants.append((1.0 - np.abs(x)) ** 3)
    # Exponential
    variants.append(np.exp(-2.0 * np.abs(x)))
    # Sine bump
    variants.append(0.5 + 0.5 * np.sin(np.pi * t))
    # Cosine window
    variants.append(0.5 + 0.5 * np.cos(2.0 * np.pi * t))
    # Raised cosine (Hann-like)
    variants.append(0.5 * (1.0 + np.cos(2.0 * np.pi * (t - 0.5))))
    # Flat top
    ft = np.ones(n)
    edge = max(1, n // 8)
    ft[:edge] = 0.5 * (1 + np.cos(np.pi * np.arange(edge) / edge))
    ft[-edge:] = ft[:edge][::-1]
    variants.append(ft)
    # Power law taper
    variants.append((1.0 + np.cos(np.pi * x)) ** 0.7)

    # Random with smooth structure
    for _ in range(5):
        a = np.ones(n)
        for _ in range(4):
            f = int(rng.integers(1, max(2, n // 3)))
            ph = float(rng.uniform(0, 2 * math.pi))
            amp = float(rng.uniform(0.1, 0.5))
            a += amp * np.sin(2 * math.pi * f * t + ph)
        variants.append(np.abs(a) + 0.1)

    # Parent resampled if available
    if parent is not None and parent.size > 1:
        P = parent.size
        idx = (np.arange(n) * P // n).astype(int)
        variants.append(parent[idx].copy())

    # Random
    variants.append(np.abs(rng.normal(0, 1, n)) + 0.2)

    result = []
    for v in variants:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 1e-6)
        v = v / v.sum()
        result.append(v)
    return result


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    total = max(1.0, float(budget_s))
    hard_deadline = t0 + total - 8.0

    # Get parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            if p.size > 0 and np.isfinite(p).all():
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        parent = None

    # Initial best
    best_a = None
    best_score = float('inf')

    if parent is not None and parent.size > 0:
        try:
            fs_val = float(evaluate_sequence([float(v) for v in parent]))
            if np.isfinite(fs_val):
                best_a = parent.copy()
                best_score = fs_val
        except Exception:
            pass

    if best_a is None:
        best_a = np.ones(32) / 32.0
        try:
            best_score = float(evaluate_sequence([1.0]))
        except Exception:
            best_score = 2.0
    if not np.isfinite(best_score):
        best_a = np.ones(32) / 32.0
        best_score = 2.0

    n_parent = best_a.size if parent is not None else 192

    # Time allocation
    phase1_end = t0 + 0.45 * total  # Global search
    phase2_end = t0 + 0.75 * total  # LP refinement
    phase3_end = hard_deadline      # Final polish

    # Phase 1: Global search - try multiple lengths and inits
    lengths_to_try = [n_parent]
    if n_parent >= 100:
        lengths_to_try.extend([max(64, n_parent // 2), n_parent * 2, n_parent * 3 // 2])
    # Deduplicate and sort
    lengths_to_try = sorted(set(max(64, int(l)) for l in lengths_to_try if l >= 64))[:5]

    rng_main = np.random.default_rng(int(seed))

    for n in lengths_to_try:
        if time.monotonic() > phase1_end:
            break
        try:
            fs_c = next_fast_len(2 * n - 1)
            inits = _init_variants(n, rng_main, parent)
            # Focus on parent for the parent length
            if n == n_parent and parent is not None:
                inits.insert(0, parent.copy() / parent.sum())

            # Short budget per init
            budget_per_init = max(2.0, (phase1_end - time.monotonic()) / max(1, len(inits)))
            for idx_init, a0 in enumerate(inits):
                if time.monotonic() > phase1_end:
                    break
                a0 = np.asarray(a0, dtype=np.float64).ravel()
                a0 = np.clip(a0, 1e-6, None)
                a0 = a0 / a0.sum()
                fs_c = next_fast_len(2 * a0.size - 1)

                # Quick Adam run
                end_this = min(phase1_end, time.monotonic() + budget_per_init * 1.5)
                try:
                    a_ref, s_ref = _adam_refine(a0, a0.size, fs_c, end_this, hard_deadline,
                                                seed=int(seed) + idx_init * 13 + n)
                    if np.isfinite(s_ref) and s_ref < best_score:
                        best_score = s_ref
                        best_a = a_ref.copy()
                except Exception:
                    pass
        except Exception:
            continue

    # Phase 2: LP cutting-plane refinement on best
    if time.monotonic() < phase2_end - 1.0:
        try:
            n_best = best_a.size
            fs_best = next_fast_len(2 * n_best - 1)
            a_best_norm = best_a.copy()
            a_best_norm = np.clip(a_best_norm, 0, 1000)
            a_best_norm = a_best_norm / max(1e-30, a_best_norm.sum())

            # Multiple LP passes with different active set sizes
            for lp_pass in range(3):
                if time.monotonic() > phase2_end - 2.0:
                    break
                end_lp = min(phase2_end - 2.0, time.monotonic() + 0.25 * (phase2_end - phase1_end))
                a_lp = _lp_refine(a_best_norm, n_best, fs_best,
                                  max_iters=200, deadline=end_lp)
                s_lp = 2.0 * n_best * _conv_fft(a_lp, n_best, fs_best).max()
                if np.isfinite(s_lp) and s_lp < best_score:
                    best_score = s_lp
                    best_a = a_lp.copy()
                    a_best_norm = a_lp.copy()
        except Exception:
            pass

    # Phase 3: Final local polish
    if time.monotonic() < phase3_end - 2.0:
        try:
            n_best = best_a.size
            fs_best = next_fast_len(2 * n_best - 1)
            # One more short Adam pass
            end_adam = min(phase3_end - max(30.0, 0.3 * (phase3_end - t0)),
                           time.monotonic() + 0.15 * (phase3_end - t0))
            if end_adam > time.monotonic() + 5.0:
                a_ad, s_ad = _adam_refine(best_a, n_best, fs_best, end_adam, hard_deadline,
                                          seed=int(seed) + 999)
                if np.isfinite(s_ad) and s_ad < best_score:
                    best_score = s_ad
                    best_a = a_ad.copy()

            # LP one more time
            if time.monotonic() < phase3_end - 5.0:
                n_best = best_a.size
                fs_best = next_fast_len(2 * n_best - 1)
                a_best_n = best_a.copy()
                a_best_n = a_best_n / max(1e-30, a_best_n.sum())
                end_lp2 = min(phase3_end - 5.0, time.monotonic() + 15.0)
                a_lp2 = _lp_refine(a_best_n, n_best, fs_best, max_iters=100,
                                   deadline=end_lp2)
                s_lp2 = 2.0 * n_best * _conv_fft(a_lp2, n_best, fs_best).max()
                if np.isfinite(s_lp2) and s_lp2 < best_score:
                    best_score = s_lp2
                    best_a = a_lp2.copy()

            # Final local search
            n_best = best_a.size
            fs_best = next_fast_len(2 * n_best - 1)
            end_ls = min(phase3_end - 2.0, time.monotonic() + max(10.0, 0.2 * (phase3_end - t0)))
            a_ls = _local_search(best_a, n_best, fs_best, end_ls, hard_deadline,
                                 seed=int(seed) + 55555)
            s_ls = 2.0 * n_best * _conv_fft(a_ls, n_best, fs_best).max()
            if np.isfinite(s_ls) and s_ls < best_score:
                best_score = s_ls
                best_a = a_ls.copy()
        except Exception:
            pass

    # Final validation with exact evaluator
    try:
        cand = [float(v) for v in best_a]
        fs_final = float(evaluate_sequence(cand))
        if not np.isfinite(fs_final):
            cand = [1.0]
            fs_final = float(evaluate_sequence(cand))
        if not np.isfinite(fs_final):
            return [1.0]
        if fs_final > best_score + 0.1:
            # Mismatch - re-evaluate parent
            if parent is not None and parent.size > 0:
                cand = [float(v) for v in parent]
                fs_parent = float(evaluate_sequence(cand))
                if np.isfinite(fs_parent):
                    return cand
            return [1.0]
        return cand
    except Exception:
        return [1.0]
