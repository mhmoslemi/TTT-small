```python
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(320, dtype=np.float64)

def _conv(a):
    """Full linear convolution via FFT."""
    n = len(a)
    if n <= 256:
        return np.convolve(a, a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2*n-1]
    return np.maximum(c, 0.0)

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv(a)
    m = float(b.max())
    return 2.0 * n * m / (s * s)

def _build_conv_matrix(f):
    """Build the convolution matrix A where (A @ g)[k] = sum_j f[j]*g[k-j]."""
    n = len(f)
    # A is (2n-1) x n
    # A[k, j] = f[j] if max(0, k-n+1) <= j <= min(k, n-1), else 0
    # More precisely: A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    # Wait: (f*g)[k] = sum_j f[j]*g[k-j] = sum_j g[j]*f[k-j]
    # So A[k,j] = f[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2*n-1, n))
    for j in range(n):
        # A[k, j] = f[k-j] for k-j in [0, n-1], i.e., k in [j, j+n-1]
        A[j:j+n, j] = f
    return A

def _lp_solve(f, M, n):
    """Solve the LP: max sum(g) s.t. (f*g)_k <= M, g >= 0."""
    A = _build_conv_matrix(f)
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b_ub = np.full(2*n-1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n,
                     method='highs', options={'presolve': True, 'time_limit': 5})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.maximum(g, 0)
    if g.sum() < 1e-15:
        return None
    return g / g.sum()  # normalize to sum=1

def _subgradient_step(f, alpha):
    """One subgradient step to reduce max autocorrelation."""
    n = len(f)
    b = _conv(f)
    k_star = int(np.argmax(b))
    # Subgradient of (f*f)_{k_star} w.r.t. f:
    # s_i = 2 * f[k_star - i] (with f_j = 0 outside [0,n-1])
    # We want to DECREASE (f*f)_{k_star}, so move AGAINST s:
    # f_new = f - alpha * s
    # But we also want to maintain sum f = 1
    s = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            s[i] = 2.0 * f[j]
    # Normalize s to unit norm for stable step
    sn = np.linalg.norm(s)
    if sn < 1e-15:
        return f
    f_new = f - alpha * s / sn
    f_new = np.maximum(f_new, 0)
    fs = f_new.sum()
    if fs < 1e-10:
        return f
    return f_new / fs

def _coordinate_polish(f, stop_time, max_rounds=50):
    """Coordinate descent polishing."""
    n = len(f)
    f = f.copy()
    f = f / f.sum()
    best_sc = _score(f)
    rng = np.random.default_rng(12345)

        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time:

            orig = f[i]
            best_val = orig
            best_local_sc = best_sc
            for mlt in (0.1, 0.3, 0.5, 0.75, 1.25, 1.6, 2.0, 3.0, 0.0):

                if sc < best_local_sc - 1e-12:

            f[i] = best_val
            if best_local_sc < best_sc - 1e-12:
                best_sc = best_local_sc
                improved = True
        if not improved and rnd > 5:
            break
    return f, best_sc

def _lp_optimize(f0, n, stop_time, seed):
    """Main LP-based optimization loop."""
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    while f.size < n:

    f = f[:n]
    fs = f.sum()
    if fs < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / fs
    
    best_f = f.copy()
    best_score = _score(f)
    t = 0.3
    stall = 0
    
    # Pre-build A (it changes when f changes, so we rebuild each iteration)
    it = 0
    last_lp_time = time.monotonic()
    
    while time.monotonic() < stop_time - 1.0:
        it += 1
        
        # Every 3rd iteration, do an LP step
        if it % 3 == 1:
            b = _conv(f)
            M = float(b.max())
            g = _lp_solve(f, M, n)
            last_lp_time = time.monotonic()
            
            if g is not None:
                # Line search on t
                improved = False
                for _ in range(8):
                    f_new = (1-t)*f + t*g
                    f_new = np.maximum(f_new, 0)
                    fs_new = f_new.sum()
                    if fs_new < 1e-10:
                        t = max(0.05, t*0.5)
                        continue
                    f_new = f_new / fs_new
                    sc = _score(f_new)
                    if sc < best_score - 1e-13:
                        f = f_new
                        best_f = f.copy()
                        best_score = sc
                        improved = True
                        t = min(0.6, t * 1.1)
                        stall = 0
                        break
                    else:
                        t = max(0.05, t * 0.6)
                if not improved:
                    stall += 2
            else:
                stall += 5
        else:
            # Subgradient step
            alpha = 0.02 / math.sqrt(max(1, it))
            f_new = _subgradient_step(f, alpha)
            sc = _score(f_new)
            if sc < best_score:
                f = f_new
                best_f = f.copy()
                best_score = sc
                stall = 0
            else:
                # Try smaller alpha

                if sc < best_score:
                    f = f_new

            t = 0.3
            stall = 0
    
    return best_f, best_score
```

Actually, let me reconsider the problem. For the LP:
max sum(g) s.t. (f*g)_k <= M, g >= 0

The key insight is that M = max(f*f) is the current peak. The LP finds the "best" g that can be added while keeping the *linearized* correlation bounded. If the optimal g has sum > 1 (after normalizing f to sum=1), it means we can "spread out" the mass.

But there's a subtlety: the constraint (f*g)_k <= M is a CONSTRAINT on the cross-correlation, not on the self-correlation of the new sequence. The actual self-correlation of f' = (1-t)f + t*g is:
f'*f' = (1-t)^2 f*f + 2t(1-t) f*g + t^2 g*g

So we need ALL of these terms to be bounded. The LP ensures f*g is bounded, but we also need g*g to be bounded (and it is, since g >= 0 and sum(g) is bounded).

For small t, the t^2 term is negligible, and the key constraint is on f*g. So the LP is a valid first-order approximation.

```python
def _lp_direction_v2(f, n):
    """
    Solve: min t s.t. (f*g)_k <= t, sum(g)=1, g>=0
    Returns (g*, t*)
    """
    A = _build_conv_matrix(f)  # (2n-1) x n, A[k,j] = f[k-j] if valid
    # Variables: [g_0, ..., g_{n-1}, t]
    # Objective: minimize t (last variable)
    c = np.zeros(n+1)
    c[n] = 1.0  # minimize t
    
    # Constraints:
    # (f*g)_k - t <= 0 for k=0,...,2n-2
    # A_ub: [A | -I] where A is (2n-1)x n
    A_ub = np.hstack([A, -np.eye(2*n-1)])
    b_ub = np.zeros(2*n-1)
    
    # sum(g) = 1 -> as two inequalities: sum(g) <= 1 and -sum(g) <= -1
    # Or use A_eq
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)]*n + [(0, None)]  # g_i >= 0, t >= 0
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                 bounds=bounds, method='highs', options={'presolve': True})
    
    if res.status != 0:
        return None, None
    g = res.x[:n]
    t_star = res.x[n]
    return g, t_star
```

Wait, I need to double-check the matrix. (f*g)_k = sum_j f[j]*g[k-j]. In matrix form: (f*g)_k = sum_j A[k,j]*g[j] where A[k,j] = f[k-j] if 0<=k-j<n else 0.

Let me verify: (f*g)[k] = sum_j f[j] * g[k-j] where j goes from max(0, k-n+1) to min(k, n-1).
Alternatively: (f*g)[k] = sum_j g[j] * f[k-j] where j goes from max(0, k-n+1) to min(k, n-1).

So A[k,j] = f[k-j] if 0 <= k-j <= n-1, else 0. Which means A[k,j] = f[k-j] if j <= k <= j+n-1.

For a fixed j, A[k,j] = f[k-j] for k = j, j+1, ..., j+n-1. So column j has f[0], f[1], ..., f[n-1] in rows j through j+n-1, and 0 elsewhere. This is a "convolution matrix" (sometimes called a "Toeplitz" matrix but it's actually "anti-diagonal" within each column).

```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(320, dtype=np.float64)
    if p.size == 0:
        p = np.ones(320, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(320, dtype=np.float64)
    return p


def _conv_ff(a):
    n = len(a)
    if n <= 256:
        return np.maximum(np.convolve(a, a), 0.0)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_ff(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_min_peak(f, n):
    """
    Solve: min t s.t. (f*g)_k <= t for all k, sum(g)=1, g>=0.
    Returns (g_normalized, t_star) or (None, None).
    """
    A = _build_A(f)
    # Variables: [g_0,...,g_{n-1}, t]
    c = np.zeros(n + 1)
    c[n] = 1.0
    # Inequality: A@g - t <= 0
    A_ub = np.hstack([A, -np.eye(2 * n - 1)])
    b_ub = np.zeros(2 * n - 1)
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 8.0})
    except Exception:
        return None, None
    if res.status != 0 or res.x is None:
        return None, None
    g = np.maximum(res.x[:n], 0.0)
    gs = g.sum()
    if gs < 1e-12:
        return None, None
    g = g / gs
    t_star = float(res.x[n])
    return g, t_star


def _lf_optimize(f0, n, stop_time, seed):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    while f.size < n:
        f = np.concatenate([f, np.zeros(n - f.size)])
    f = f[:n]
    fs = f.sum()
    if fs < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / fs

    best_f = f.copy()
    best_score = _score(f)
    t = 0.25
    stall = 0
    max_stall = 80
    it = 0

    while time.monotonic() < stop_time - 2.0:
        it += 1

        if it % 2 == 1:
            # LP direction finding
            g, t_star = _lp_min_peak(f, n)
            if g is None:
                stall += 5
            else:
                M_cur = float(_conv_ff(f).max())
                if t_star >= M_cur - 1e-10:
                    # No improvement possible from this direction
                    stall += 2
                else:
                    # Line search
                    improved = False
                    for _ in range(10):
                        f_new = (1.0 - t) * f + t * g
                        f_new = np.maximum(f_new, 0.0)
                        fns = f_new.sum()
                        if fns < 1e-10:
                            t = max(0.03, t * 0.5)
                            continue
                        f_new = f_new / fns
                        sc = _score(f_new)
                        if sc < best_score - 1e-12:
                            f = f_new
                            best_f = f.copy()
                            best_score = sc
                            improved = True
                            t = min(0.55, t * 1.08)
                            stall = 0
                            break
                        else:
                            t = max(0.03, t * 0.6)
                    if not improved:
                        stall += 1
        else:
            # Subgradient step (cheap)
            b = _conv_ff(f)
            k_star = int(np.argmax(b))
            s = np.zeros(n)
            for i in range(n):
                j = k_star - i
                if 0 <= j < n:
                    s[i] = 2.0 * f[j]
            sn = np.linalg.norm(s)
            if sn > 1e-15:
                alpha = 0.015 / math.sqrt(1.0 + it * 0.01)
                f_new = f - alpha * s / sn
                f_new = np.maximum(f_new, 0.0)
                fns = f_new.sum()
                if fns > 1e-10:
                    f_new = f_new / fns
                    sc = _score(f_new)
                    if sc < best_score - 1e-12:
                        f = f_new
                        best_f = f.copy()
                        best_score = sc
                        stall = 0
                    else:
                        stall += 1
                else:
                    stall += 1
            else:
                stall += 1

        if stall > max_stall:
            f = best_f + rng.normal(0, 0.03, n)
            f = np.maximum(f, 0.0)
            fns = f.sum()
            if fns > 1e-10:
                f = f / fns
            else:
                f = best_f.copy()
            t = 0.25
            stall = 0

    return best_f, best_score


def _polish(f, stop_time, seed=999):
    n = len(f)
    best = np.clip(np.asarray(f, dtype=np.float64).ravel(), 0, None).copy()
    bsum = best.sum()
    if bsum < 1e-10:
        return best, _score(best)
    best = best / bsum
    best_sc = _score(best)
    rng = np.random.default_rng(seed)
    no_imp = 0

    while time.monotonic() < stop_time - 1.5:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 1.5:
                break
            orig = best[i]
            bv = orig
            bs_local = best_sc
            for mlt in (0.1, 0.3, 0.5, 0.7, 1.3, 1.7, 2.5, 0.0):
                best[i] = orig * mlt if orig > 1e-12 else (0.03 if mlt > 0 else 0.0)
                if best.sum() < 0.01:
                    best[i] = orig
                    continue
                sc = _score(best)
                if sc < bs_local - 1e-11:
                    bs_local = sc
                    bv = best[i]
            best[i] = bv
            if bs_local < best_sc - 1e-11:
                best_sc = bs_local
                improved = True

        # 2-coord moves
        for _ in range(40):
            if time.monotonic() > stop_time - 1.5:
                break
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            for mi in (0.4, 1.5):
                for mj in (0.4, 1.5):
                    best[i] = oi * mi if oi > 1e-12 else 0.03
                    best[j] = oj * mj if oj > 1e-12 else 0.03
                    if best.sum() < 0.01:
                        best[i], best[j] = oi, oj
                        continue
                    sc = _score(best)
                    if sc < best_sc - 1e-11:
                        best_sc = sc
                        improved = True
            best[i], best[j] = oi, oj

        if not improved:
            no_imp += 1
            if no_imp >= 5:
                break
        else:
            no_imp = 0

    return best, best_sc

            except Exception:

        except Exception:
            pass

    # Final validation

            return [1.0]
        return [float(x) for x in fallback]

    if final_sc < best_score:
        best_score = final_sc

    return [float(x) for x in best_seq]
```
