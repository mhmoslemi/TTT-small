                        if v < local:

                            stale = 0
                            momentum = np.zeros(n, dtype=np.float64)
                            update_best(a)
                if stale > 400:
                    break

        update_best(a)
        return a

    # Interpolation
    def interp_len(src, n_target):
        if src.size == n_target:
            return src.copy()
        ox = np.linspace(0.0, 1.0, src.size)
        nx = np.linspace(0.0, 1.0, n_target)
        out = np.interp(nx, ox, src)
        out = np.clip(out, 0, None)
        s = float(out.sum())
        if s > 1e-15:
            out = out / s
        else:
            out = np.ones(n_target, dtype=np.float64) / n_target
        return out

    # === MAIN OPTIMIZATION ===
    p = parent if parent is not None else best_arr
    n_p = p.size
    p_norm = p / p.sum()

    # Phase 1: Intensive optimization at parent length
    if n_p >= 4:
        # 1a: LP
        t_rem = deadline - time.monotonic()
        alloc_lp1 = min(420.0, t_rem * 0.38)
        if alloc_lp1 > 20 and linprog is not None and sps is not None:
            a1 = lp_opt(p_norm.copy(), n_p, time.monotonic() + alloc_lp1, max_iter=350)
            update_best(a1)

        # 1b: Smooth gradient
        if time.monotonic() < deadline - 100:
            t_rem = deadline - time.monotonic() - 100
            alloc_g1 = min(200.0, t_rem * 0.5)
            if alloc_g1 > 15:
                a1 = smooth_opt(a1, n_p, time.monotonic() + alloc_g1, max_iter=20000)
                update_best(a1)

        # 1c: Second LP pass (with tighter margins)
        if time.monotonic() < deadline - 80 and linprog is not None:
            alloc_lp2 = min(180.0, (deadline - time.monotonic() - 80) * 0.6)
            if alloc_lp2 > 15:
                a1 = lp_opt(a1, n_p, time.monotonic() + alloc_lp2, max_iter=250)
                update_best(a1)

        # 1d: More gradient
        if time.monotonic() < deadline - 40:
            a1 = smooth_opt(a1, n_p, deadline - 35, max_iter=15000)
            update_best(a1)

    # Phase 2: Try nearby lengths
    if time.monotonic() < deadline - 50:
        lengths_to_try = []
        for L in [n_p - 8, n_p - 4, n_p - 2, n_p + 2, n_p + 4, n_p + 8, n_p + 16]:
            if 64 <= L <= 600 and L != n_p:
                lengths_to_try.append(L)
        for L in [256, 280, 304, 336, 352, 384, 416, 448, 480, 512]:
            if 64 <= L <= 600 and L != n_p and L not in lengths_to_try:
                lengths_to_try.append(L)
        lengths_to_try = sorted(set(lengths_to_try))

        for L in lengths_to_try:
            if time.monotonic() >= deadline - 35:
                break
            t_rem = deadline - time.monotonic() - 35
            alloc = min(35.0, t_rem * 0.12)
            if alloc < 8:
                continue

            a_init = interp_len(p_norm, L)

            if linprog is not None and sps is not None:
                a_w = lp_opt(a_init, L, time.monotonic() + alloc * 0.55, max_iter=40)
                update_best(a_w)

            if time.monotonic() < deadline - 30:
                a_w = smooth_opt(a_w, L, min(time.monotonic() + alloc * 0.4, deadline - 28), max_iter=3000)
                update_best(a_w)

    # Phase 3: Restarts at parent length
    if time.monotonic() < deadline - 30 and linprog is not None:
        n_restarts = 10
        for r in range(n_restarts):
            if time.monotonic() >= deadline - 20:
                break
            t_rem = deadline - time.monotonic() - 20
            alloc = min(22.0, t_rem / max(1, n_restarts - r))
            if alloc < 5:
                break

            rtype = r % 5
            if rtype == 0:
                a_init = p_norm * np.exp(0.15 * rng.standard_normal(n_p))
            elif rtype == 1:
                a_init = p_norm[::-1].copy()
            elif rtype == 2:
                a_init = p_norm * np.exp(0.35 * rng.standard_normal(n_p))
                a_init = np.clip(a_init, 0, None)
            elif rtype == 3:
                # Perturb a random subset
                a_init = p_norm.copy()
                mask = rng.random(n_p) < 0.3
                a_init[mask] *= np.exp(0.3 * rng.standard_normal(mask.sum()))
                a_init = np.clip(a_init, 0, None)
            else:
                # Create a "bump" perturbation
                center = int(rng.integers(n_p // 4, 3 * n_p // 4))
                width = max(3, n_p // 10)
                a_init = p_norm.copy()
                for j in range(max(0, center - width), min(n_p, center + width + 1)):
                    a_init[j] *= 0.5 + 0.5 * rng.random()
                a_init = np.clip(a_init, 0, None)

            s_init = float(a_init.sum())
            if s_init > 1e-15:
                a_init = a_init / s_init

            a_w = lp_opt(a_init, n_p, time.monotonic() + alloc * 0.6, max_iter=30)
            update_best(a_w)
            if time.monotonic() < deadline - 15:
                a_w = smooth_opt(a_w, n_p, deadline - 12, max_iter=2000)
                update_best(a_w)

    # Phase 4: Final polish
    if time.monotonic() < deadline - 8 and best_arr is not None:
        n_b = best_arr.size
        a_pol = best_arr.copy() / best_arr.sum()

        if linprog is not None and time.monotonic() < deadline - 12:
            a_pol = lp_opt(a_pol, n_b, deadline - 10, max_iter=25)
            update_best(a_pol)

        if time.monotonic() < deadline - 5:
            a_pol = smooth_opt(a_pol, n_b, deadline - 3, max_iter=8000)
            update_best(a_pol)

    # Return
    if best_arr is None:
        best_arr = np.ones(320, dtype=np.float64)
        update_best(best_arr)

    final = np.clip(best_arr, 0.0, 1000.0)
    s_fin = float(final.sum())
    if s_fin < 0.01:
        final = np.ones_like(final)
    update_best(final)

    return [float(x) for x in best_arr]
