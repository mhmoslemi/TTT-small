import cvxpy as cp
import numpy as np

def solve_ac1(n, solver=None):
    a = cp.Variable(n, nonneg=True)
    t = cp.Variable()
    
    constraints = [cp.sum(a) == 1]
    
    for k in range(2*n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        q = cp.zeros_scalar  # or start at 0
        terms = []
        for j in range(lo, hi + 1):
            terms.append(cp.multiply(a[j], a[k-j]))
        q = cp.sum(terms)
        constraints.append(q <= t)
    
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=solver or cp.MOSEK)

            return float('inf')

        F = rfft(a, fs)
