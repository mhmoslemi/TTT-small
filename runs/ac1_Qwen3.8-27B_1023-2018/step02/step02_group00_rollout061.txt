                best_sc = bs
        except Exception:

    # ---- Phase 1b: LP from alternative initializations ----
    p1b_end = t0 + budget * 0.58
    i_arr = np.arange(n, dtype=np.float64)
    inits = []
    # Cosine bump
    inits.append(0.5 + 0.5 * np.cos(2 * math.pi * i_arr / n))
    # Multi-frequency

    for fr in (1, 2, 3, 5, 7):
        sup += 0.3 * np.cos(2 * math.pi * fr * i_arr / n + rng.uniform(0, 2 * math.pi))
    inits.append(np.abs(sup) + 0.05)
    # Random
    inits.append(np.abs(rng.normal(0, 1, n)) + 0.1)
    # Best so far + noise

        if time.monotonic() > p3_end - 5:
            break
        # Interpolate parent to length L
        fL = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, n), p)
        sL = fL.sum()
        if sL < 1e-10:
            fL = np.ones(L) / L
        else:
            fL = fL / sL

        stop = min(p3_end - 3, time.monotonic() + max(5.0, (p3_end - time.monotonic()) / 4))
        try:

            if bs < best_sc:

                best_sc = bs

                    bsc = sc

                    break
