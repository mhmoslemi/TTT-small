    """One step of smooth gradient descent on max(a*a)."""
    n = len(a)
    # Compute a * a
    F = rfft(a, n_fft)
    c = irfft(F * F, n_fft)[:2*n-1]
    
    # Compute softmax weights
    t_c = t * c - t * c.max()  # For numerical stability
    p = np.exp(t_c)
    p /= p.sum()
    
    # Compute gradient: grad[j] = 2 * sum_m p[j+m] * a[m]
    # This is cross-correlation
    # FFT approach: 
    # FFT of p (padded to n_fft)
    Pf = rfft(p, n_fft)
    # FFT of a (padded to n_fft)
    Af = rfft(a, n_fft)
    # Cross-correlation: ifft(conj(Pf) * Af) gives sum_k conj(p[k]) * a[k+j]
    # We want sum_m p[j+m] * a[m] = sum_m a[m] * p[m+j]

def _optimize_smooth(a0, t_start, t_end, n_iter, stop_time):
    n = len(a0)
    a = a0.copy()
    n_fft = next_fast_len(3*n - 2)
    
    for i in range(n_iter):
        if time.monotonic() > stop_time:
            break
        t = t_start * (t_end/t_start) ** (i/n_iter)  # Anneal t
        
        # Compute a*a
        F = rfft(a, n_fft)
        c = irfft(F*F, n_fft)[:2*n-1]
        
        # Softmax
        tc = t * c - t * c.max()
        p = np.exp(tc)
        p /= p.sum()
        
        # Gradient
        Af = rfft(a, n_fft)
        Pf = rfft(p, n_fft)
        grad_full = irfft(np.conj(Af) * Pf, n_fft).real
        grad = 2.0 * grad_full[:n]
        
        # Step size (backtracking line search)
        c_max = c.max()
        for lr in [0.01, 0.005, 0.001, 0.0005, 0.0001]:
            a_new = a - lr * grad
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s <= 0:
                continue
            a_new /= s
            F_new = rfft(a_new, n_fft)
            c_new = irfft(F_new*F_new, n_fft)[:2*n-1]
            c_new_max = c_new.max()
            if c_new_max < c_max:
                a = a_new
                break
    
    return a
