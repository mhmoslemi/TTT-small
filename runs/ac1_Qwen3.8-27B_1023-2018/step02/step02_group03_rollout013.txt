    def precompute_pattern(n):
        rows_parts, cols_parts, aidx_parts = [], [], []
        for k in range(2 * n - 1):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            js = np.arange(lo, hi + 1)
            rows_parts.append(np.full(len(js), k, dtype=np.int32))
            cols_parts.append(js.astype(np.int32))
            aidx_parts.append((k - js).astype(np.int32))
        return np.concatenate(rows_parts), np.concatenate(cols_parts), np.concatenate(aidx_parts)
    
    pattern_cache = {}
    def get_pattern(n):
        if n not in pattern_cache:
            pattern_cache[n] = precompute_pattern(n)
        return pattern_cache[n]
    
    def solve_lp(a, M, margin):
        n = len(a)
        rows, cols, aidx = get_pattern(n)
        vals = a[aidx]
        A = sps.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
        b_ub = M * margin + 1e-14
        try:
            res = linprog(-np.ones(n), A_ub=A, b_ub=b_ub,
                         bounds=[(0, None)]*n, method='highs',
                         options={'presolve': True, 'time_limit': 5,
                                  'dual feasibility tolerance': 1e-10,
                                  'primal feasibility tolerance': 1e-10})
        except:
            return None
        if res is None or getattr(res, 'x', None) is None:
            return None
        g = np.clip(res.x, 0, None)
        sg = float(g.sum())
        if sg < 1e-15:
            return None
        return g / sg
