def _gradient_refine(f, n, stop_time, seed):
    """Gradient-based refinement using soft-max smoothing of the max."""
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f, dtype=np.float64).ravel(), 0, None).copy()
    f = f / max(f.sum(), 1e-15)
    
    best_f = f.copy()
    best_score = _score(f)
    beta = 80.0
    lr = 0.3
    no_improve = 0
    
    while time.monotonic() < stop_time - 2.0 and no_improve < 200:
        c = _conv_full(f)
        max_c = c.max()
        exp_bc = np.exp(beta * (c - max_c))
        total_exp = exp_bc.sum()
        if total_exp < 1e-300:
            break
        weights = exp_bc / total_exp
        
        # Gradient: g_i = sum_k weights[k] * f[k-i]
        g = np.zeros(n)
        for i in range(n):
            # k ranges from i to i+n-1 (so that k-i ranges 0 to n-1)
            lo = max(i, 0)
            hi = min(i + n, 2*n - 1)
            if hi > lo:
                g[i] = np.dot(weights[lo:hi], f[:hi-lo])
        
        S = float(np.dot(weights, c))
        S_sum = float(f.sum())
        
        # dF/da_i where F = 2n * S / S_sum^2
        grad = 2.0 * n * (g * S_sum - 2.0 * S) / (S_sum ** 3)
        
        # Normalized gradient (since sum is fixed)
        # Remove mean component to stay on simplex
        grad_centered = grad - np.mean(grad)
        
        improved = False
        for lr_try in [lr, lr*0.5, lr*0.25, lr*0.1]:
            f_new = f - lr_try * grad_centered
            f_new = np.clip(f_new, 0, None)
            f_new_sum = f_new.sum()
            if f_new_sum < 1e-10:
                continue
            f_new = f_new / f_new_sum
            sc = _score(f_new)
            if sc < best_score - 1e-14:
                best_f = f_new.copy()
                best_score = sc
                f = f_new
                improved = True
                no_improve = 0
                lr = min(3.0, lr * 1.05)
                break
            else:
                continue
        if not improved:
            lr = max(0.05, lr * 0.6)
            no_improve += 1
            # Occasionally perturb
            if no_improve % 30 == 0:
                f = np.clip(f + rng.normal(0, 0.02, n), 0, None)
                f = f / max(f.sum(), 1e-15)
    
    return best_f, best_score

def _polish(a, stop_time):
    """Coordinate descent polish"""
    n = len(a)
    best = np.clip(np.asarray(a, dtype=np.float64).ravel(), 0, None).copy()
    best = best / max(best.sum(), 1e-15)
    best_sc = _score(best)
    rng = np.random.default_rng(12345)
    no_imp = 0
    
    while time.monotonic() < stop_time - 2.0 and no_imp < 20:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 2.0:
                break
            orig = best[i]
            best_val = orig
            best_val_sc = best_sc
            for mlt in (0.3, 0.5, 0.75, 1.0, 1.25, 1.5, 2.0, 0.0):
                best[i] = orig * mlt if orig > 1e-12 else (0.01 if mlt > 0 else 0.0)
                if best.sum() < 0.01:
                    best[i] = orig
                    continue
                sc = _score(best)
                if sc < best_val_sc - 1e-14:
                    best_val = best[i]
                    best_val_sc = sc
                    best[i] = best[i]  # keep
            best[i] = best_val
            if best_val_sc < best_sc - 1e-14:
                best_sc = best_val_sc
                improved = True
        
        # 2-coordinate moves
        for _ in range(40):
            if time.monotonic() > stop_time - 2.0:
                break
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            for mi in (0.5, 1.0, 1.5):
                for mj in (0.5, 1.0, 1.5):
                    best[i] = oi * mi
                    best[j] = oj * mj
                    if best.sum() < 0.01:
                        best[i], best[j] = oi, oj
                        continue
                    sc = _score(best)
                    if sc < best_sc - 1e-14:
                        best_sc = sc
                        improved = True
            best[i], best[j] = oi, oj
        
        if not improved:
            no_imp += 1
        else:
            no_imp = 0
    
    return best, best_sc

            if stall >= max_stall:
                f = np.clip(f + rng.normal(0, 0.07, n), 0, None)
                f = f / max(f.sum(), 1e-15)
                t = 0.3
                stall = 0

        iteration += 1

    return best_f, best_score


def _gradient_refine(f, n, stop_time, seed, beta_init=50.0):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f, dtype=np.float64).ravel(), 0, None).copy()
    f = f / max(f.sum(), 1e-15)

    best_f = f.copy()
    best_score = _score(f)
    beta = beta_init
    lr = 0.4
    no_improve = 0

    while time.monotonic() < stop_time - 2.0 and no_improve < 300:
        c = _conv_full(f)
        max_c = float(c.max())
        exp_bc = np.exp(beta * (c - max_c))
        total_exp = float(exp_bc.sum())
        if total_exp < 1e-300:
            break
        weights = exp_bc / total_exp

        # Gradient via cross-correlation
        g = np.correlate(weights, f, mode="valid")

        S = float(np.dot(weights, c))
        S_sum = float(f.sum())

        grad = 2.0 * n * (g * S_sum - 2.0 * S) / (S_sum ** 3)
        grad_cent = grad - np.mean(grad)

        improved = False
        for lr_try in (lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 0.05):
            f_new = f - lr_try * grad_cent
            f_new = np.clip(f_new, 0, None)
            f_new_sum = f_new.sum()
            if f_new_sum < 1e-10:
                continue
            f_new = f_new / f_new_sum
            sc = _score(f_new)
            if sc < best_score - 1e-14:
                best_f = f_new.copy()
                best_score = sc
                f = f_new
                improved = True
                no_improve = 0
                lr = min(4.0, lr * 1.04)
                break

        if not improved:
            lr = max(0.03, lr * 0.65)
            no_improve += 1
            if no_improve % 40 == 0:
                f = np.clip(f + rng.normal(0, 0.015, n), 0, None)
                f = f / max(f.sum(), 1e-15)
            if no_improve % 80 == 0:
                beta = max(10.0, beta * 0.6)

    return best_f, best_score


def _polish(a, stop_time):
    n = len(a)
    best = np.clip(np.asarray(a, dtype=np.float64).ravel(), 0, None).copy()
    best = best / max(best.sum(), 1e-15)
    best_sc = _score(best)
    rng = np.random.default_rng(54321)
    no_imp = 0

    while time.monotonic() < stop_time - 2.0 and no_imp < 25:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 2.0:
                break
            orig = best[i]
            best_val = orig
            best_val_sc = best_sc
            for mlt in (0.25, 0.5, 0.7, 1.0, 1.3, 1.6, 2.0, 0.0):
                best[i] = orig * mlt if orig > 1e-12 else (0.005 if mlt > 0 else 0.0)
                if best.sum() < 0.01:
                    best[i] = orig
                    continue
                sc = _score(best)
                if sc < best_val_sc - 1e-14:
                    best_val = best[i]
                    best_val_sc = sc
            best[i] = best_val
            if best_val_sc < best_sc - 1e-14:
                best_sc = best_val_sc
                improved = True

        # 2-coordinate moves
        for _ in range(30):
            if time.monotonic() > stop_time - 2.0:
                break
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            for mi in (0.4, 0.7, 1.0, 1.4, 2.0):
                for mj in (0.4, 0.7, 1.0, 1.4, 2.0):
                    best[i] = oi * mi
                    best[j] = oj * mj
                    if best.sum() < 0.01:
                        best[i], best[j] = oi, oj
                        continue
                    sc = _score(best)
                    if sc < best_sc - 1e-14:
                        best_sc = sc
                        improved = True
            best[i], best[j] = oi, oj

        if not improved:
            no_imp += 1
        else:
            no_imp = 0

    return best, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 15.0

    parent = _get_parent()
    n_p = len(parent)

    fallback = parent.copy()
    try:
        fallback_sc = float(evaluate_sequence([float(x) for x in fallback]))
    except Exception:
        fallback_sc = math.inf
    if not math.isfinite(fallback_sc):
        fallback = np.ones(max(n_p, 32)) / max(n_p, 32)
        try:
            fallback_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            fallback_sc = math.inf
    if not math.isfinite(fallback_sc):
        return [1.0]

    best_seq = fallback.copy()
    best_score = fallback_sc

    def _try_update(arr):
        nonlocal best_seq, best_score
        if arr is None or len(arr) == 0:
            return
        if not np.all(np.isfinite(arr)):
            return
        arr = np.clip(np.asarray(arr, dtype=np.float64), 0, 1000)
        if arr.sum() < 0.01:
            return
        try:
            exact = float(evaluate_sequence([float(x) for x in arr]))
        except Exception:
            exact = math.inf
        if math.isfinite(exact) and exact < best_score:
            best_seq = arr.copy()
            best_score = exact

    n = n_p
    rng_main = np.random.default_rng(seed)

    # Build diverse initializations
    i_arr = np.arange(n, dtype=np.float64)
    inits = [parent.copy()]
    p_std = max(1e-6, float(np.std(parent)))
    inits.append(np.abs(parent + rng_main.normal(0, 0.3 * p_std, n)))
    inits.append(np.abs(parent * (1 + 0.4 * np.cos(2 * math.pi * 5 * i_arr / n)) + 0.01))
    sup = np.ones(n) * 0.15
    for fr in (1, 2, 3, 5, 7, 11):
        sup += 0.12 * np.cos(2 * math.pi * fr * i_arr / n + rng_main.uniform(0, 2 * math.pi))
    inits.append(np.abs(sup) + 0.01)
    inits.append(np.abs(rng_main.normal(0, 1, n)) + 0.05)

    # Phase 1: LP optimization (50% budget)
    phase1_end = t0 + budget * 0.50
    time_per_init = max(8.0, (phase1_end - t0) / len(inits))

    for idx, init in enumerate(inits):
        if time.monotonic() > phase1_end - 5.0:
            break
        stop = min(phase1_end - 3.0, time.monotonic() + time_per_init)
        try:
            res, _ = _lp_optimize(init, n, stop, seed + 1000 * (idx + 1))
        except Exception:
            continue
        _try_update(res)

    # Phase 2: Gradient refinement (20% budget)
    phase2_end = t0 + budget * 0.72
    if time.monotonic() < phase2_end - 10.0:
        try:
            res, _ = _gradient_refine(best_seq, n, phase2_end - 2.0, seed + 7777, beta_init=60.0)
            _try_update(res)
        except Exception:
            pass

    if time.monotonic() < phase2_end - 10.0:
        try:
            res, _ = _gradient_refine(best_seq, n, phase2_end - 2.0, seed + 6666, beta_init=20.0)
            _try_update(res)
        except Exception:
            pass

    # Phase 3: Additional LP from best (12% budget)
    phase3_end = t0 + budget * 0.84
    if time.monotonic() < phase3_end - 5.0:
        try:
            res, _ = _lp_optimize(best_seq, n, phase3_end - 2.0, seed + 5555)
            _try_update(res)
        except Exception:
            pass

    # Phase 4: Gradient + LP interleaved (7% budget)
    phase4_end = t0 + budget * 0.91
    if time.monotonic() < phase4_end - 5.0:
        try:
            res, _ = _gradient_refine(best_seq, n, phase4_end - 2.0, seed + 4444, beta_init=40.0)
            _try_update(res)
        except Exception:
            pass

    # Phase 5: Coordinate polish (remaining)
    polish_end = deadline - 3.0
    if time.monotonic() < polish_end - 5.0:
        try:
            res, _ = _polish(best_seq, polish_end)
            _try_update(res)
        except Exception:
            pass

    # Final validation
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in fallback]

    return [float(x) for x in best_seq]
