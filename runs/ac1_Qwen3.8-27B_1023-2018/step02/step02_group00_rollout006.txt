    return grad_half[i] = sum_j p[j] * w[i+j], for i=0..n-1.
    Used for gradients of quadratic convolution objectives.
    """

    floor = 1e-12

        fpos /= float(fpos.sum())

    b = _conv(fpos)

        return None

        ts = np.asarray(ts, dtype=np.float64).ravel()
        t1 = 1.0 - ts
        t2 = ts * ts
        t12 = 2.0 * t1 * ts
        v = (t1[:, None] ** 2) * bf[None, :]
        v += t12[:, None] * bfg[None, :]
        v += t2[:, None] * bg[None, :]
        maxv = v.max(axis=1)
        sc = 2.0 * n * maxv
        return ts, sc

    ts, sc = vals_for_ts(np.linspace(0.0, 1.0, 192))
    i = int(np.argmin(sc))
    t_best = float(ts[i])
    sc_best = float(sc[i])

    def refine(lo, hi, count=57):
        nonlocal t_best, sc_best
        if hi <= lo + 1e-13:
            return
        ts2, sc2 = vals_for_ts(np.linspace(lo, hi, count))
        j = int(np.argmin(sc2))
        if sc2[j] < sc_best - 1e-15:
            sc_best = float(sc2[j])
            t_best = float(ts2[j])

    if 0 < i < len(ts) - 1:
        refine(ts[i - 1], ts[i + 1])
    elif i == 0:
        refine(0.0, ts[1])
    else:
        refine(ts[-2], 1.0)

    h = (1.0 - t_best) * f + t_best * g
    hs = float(h.sum())
    if not math.isfinite(hs) or hs <= 1e-300:
        h = f.copy()
        t_best = 0.0
        sc_best = cur_score
    else:
        h /= hs
    return t_best, float(sc_best), h


def _gradient_step(f, lr, cur_score, custom_w=None, beta=8.0):
    f = np.asarray(f, dtype=np.float64).ravel()
    n = f.size
    b = _conv(f)
    bmax = float(b.max())
    if not math.isfinite(bmax) or bmax <= 1e-15:
        return f, cur_score

    w = None
    if custom_w is not None:
        ww = np.asarray(custom_w, dtype=np.float64).ravel()
        ww = np.where(np.isfinite(ww), ww, 0.0)
        ws = float(ww.sum())
        if math.isfinite(ws) and ws > 1e-300:
            w = ww / ws

    if w is None:
        rel = np.clip(b / bmax, 0.0, 1.0)
        w = np.exp(beta * (rel - 1.0))
        ws = float(w.sum())
        if not math.isfinite(ws) or ws <= 1e-300:
            w = b.copy()
            ws = float(w.sum())
        if ws > 1e-300:
            w /= ws

    gh = _hankel(f, w)
    grad = 2.0 * gh
    d = grad - float(grad.mean())
    scale = float(np.max(np.abs(d))) if d.size else 0.0
    if scale <= 1e-15:
        return f, cur_score
    d /= scale

    for tlr in (lr, 0.5 * lr, 0.25 * lr, 2.0 * lr):
        if tlr <= 0.0 or not math.isfinite(tlr):
            continue
        p = f * np.exp(-tlr * d)
        s = float(p.sum())
        if not math.isfinite(s) or s <= 1e-300:
            continue
        p /= s
        sc = _score(p)
        if math.isfinite(sc) and sc < cur_score - 1e-13:
            return p, sc

    return f, cur_score


def _coordinate_polish(a, stop, rng, max_passes=25):
    n = a.size
    if n < 2:
        return _normalize(a, n), _score(a)

    a = _normalize(a, n).copy()
    b = _conv(a)
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    passes = 0
    while time.monotonic() < stop - 2.0 and passes < max_passes:
        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > stop - 2.0:
                break
            vi = float(a[i])
            avg = 1.0 / n
            maxa = float(a.max())

            cands = []
            for mlt in (0.0, 0.1, 0.25, 0.5, 0.75, 1.5, 2.0, 4.0):
                cands.append(vi * mlt)
            cands.extend([
                0.25 * avg, 0.5 * avg, avg, 2.0 * avg, 4.0 * avg,
                0.25 * maxa, 0.5 * maxa, maxa, 2.0 * maxa
            ])

            start = i
            end = i + n
            ln = b.size
            if end > ln:
                end = ln

            out_left = float(b[:start].max()) if start > 0 else 0.0
            out_right = float(b[end:].max()) if end < ln else 0.0
            outside = max(0.0, out_left, out_right)
            old_aff = b[start:end]

            best_nv = vi
            best_delta = 0.0
            best_sc = cur_score
            seen = set()

            for nv in cands:
                if not isinstance(nv, (int, float)):
                    continue
                nv = float(nv)
                if not math.isfinite(nv) or nv < 0.0:
                    continue
                key = round(nv, 12)
                if key in seen:
                    continue
                seen.add(key)

                delta = nv - vi
                if abs(delta) <= 1e-15:
                    continue
                dvec = 2.0 * delta * a
                dvec[i] += delta * delta
                new_aff = old_aff + dvec
                aff_max = float(new_aff.max())
                new_max = outside if outside > aff_max else aff_max
                new_sum = 1.0 + delta
                if new_sum <= 1e-300:
                    continue
                sc = 2.0 * n * new_max / (new_sum * new_sum)
                if sc < best_sc - 1e-13:
                    best_sc = sc
                    best_nv = nv
                    best_delta = delta

            if best_delta != 0.0:
                delta = best_delta
                dvec = 2.0 * delta * a
                dvec[i] += delta * delta
                b[start:end] += dvec
                a[i] = best_nv
                S = 1.0 + delta
                if S > 1e-300:
                    a *= 1.0 / S
                    b *= 1.0 / (S * S)
                    cur_score = best_sc
                    if cur_score < best_score - 1e-14:
                        best_score = cur_score
                        best_a = a.copy()
                    improved = True

        if not improved:
            break
        passes += 1

    try:
        a = _normalize(a, n)
        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
    except Exception:
        pass
    return best_a, float(best_score)


def _make_inits(parent, n, rng, keep=5):
    i = np.arange(n, dtype=np.float64)
    inits = []

    if parent is not None and parent.size > 0:
        p = np.asarray(parent, dtype=np.float64).ravel()
        p = np.where(np.isfinite(p), p, 0.0)
        p = np.clip(p, 0.0, 1000.0)
        if p.size>0 and p.size != n:
            xold = np.linspace(0.0, 1.0, p.size)
            xnew = np.linspace(0.0, 1.0, n)
            inits.append(np.interp(xnew, xold, p))
        else:
            inits.append(p.astype(np.float64))

    # One-sided inverse-square-root style initializations.
    for p_exp in (0.40, 0.45, 0.50, 0.55):
        for alpha in (0.5, 1.0, 1.5):
            inits.append((i + max(alpha, 0.25)) ** (-p_exp))

    # Exponential / rational decays.
    denom = max(1, n - 1)
    for lam in (0.5, 1.0, 2.0, 4.0):
        inits.append(np.exp(-lam * i / denom))
    for lam in (0.75, 1.5, 3.0):
        inits.append(1.0 / (1.0 + lam * i / denom))

    # Smooth bump that is large on the left.
    if n > 1:
        inits.append(0.25 + 0.75 * np.cos(np.pi * 0.5 * i / denom))

    inits.append(np.ones(n, dtype=np.float64))
    inits.append(np.abs(rng.normal(0.0, 1.0, n)) + 0.1)

    if parent is not None and parent.size > 0:
        p = np.asarray(parent, dtype=np.float64).ravel()
        if p.size != n:
            p = np.interp(np.linspace(0.0, 1.0, n),
                          np.linspace(0.0, 1.0, p.size), p)
        inits.append(np.abs(p + 0.05 * rng.normal(0.0, 1.0, n)) + 0.01)

    scored = []
    seen = []
    for arr in inits:
        arr = np.asarray(arr, dtype=np.float64).ravel()
        arr = np.where(np.isfinite(arr), arr, 0.0)
        arr = np.clip(arr, 0.0, 1000.0)
        if arr.size != n or arr.sum() <= 0:
            continue
        arr = _normalize(arr, n)
        key = tuple(np.round(arr, 6))
        if any(np.allclose(arr, other, atol=1e-8) for other in seen):
            continue
        seen.append(arr)
        sc = _score(arr)
        if math.isfinite(sc):
            scored.append((sc, arr))

    scored.sort(key=lambda x: x[0])
    return [a for _, a in scored[:max(1, keep)]]


def _optimize(f0, n, stop, seed, use_lp=True, target=TARGET_UPPER):
    rng = np.random.default_rng(seed)
    f = _normalize(f0, n)
    cur_score = _score(f)
    best_f = f.copy()
    best_score = cur_score
    sscore_exact = None

    lr = 0.55
    stall = 0
    it = 0
    last_polish = time.monotonic()

    while time.monotonic() < stop - 6.0:
        rem = stop - time.monotonic()
        if rem < 8.0:
            break

        improved = False

        # Main LP-direction step.
        if use_lp and (it % 2 == 0) and rem > 35.0:
            factors = (1.0,)
            if it % 8 == 0:
                factors = (1.0, 0.92, 0.85)

            for factor in factors:
                g = _lp_direction(f, stop, factor=factor)
                if g is None:
                    continue
                _, sc, h = _line_search(f, g, cur_score)
                if math.isfinite(sc) and sc < cur_score - 1e-12:
                    f = h
                    cur_score = sc
                    improved = True
                    stall = 0
                    break
                elif time.monotonic() > stop - 20.0:
                    break

        if improved and cur_score < best_score - 1e-14:
            best_score = cur_score
            best_f = f.copy()
            it += 1
            continue

        # Gradient/log-peak step.
        f_new, sc = _gradient_step(f, lr, cur_score)
        if math.isfinite(sc) and sc < cur_score - 1e-12:
            f = f_new
            cur_score = sc
            lr = min(2.2, lr * 1.12)
            stall = 0
            improved = True
        else:
            lr = max(0.07, lr * 0.78)
            stall += 1

        if improved and cur_score < best_score - 1e-14:
            best_score = cur_score
            best_f = f.copy()

        # Periodic coordinate polish.
        if (it % 18 == 17 or stall >= 12) and time.monotonic() - last_polish > 12.0 and rem > 45.0:
            try:
                p_stop = time.monotonic() + max(8.0, min(35.0, 0.12 * rem))
                fp, scp = _coordinate_polish(f, p_stop, rng, max_passes=8)
                if math.isfinite(scp) and scp < cur_score - 1e-14:
                    f = fp
                    cur_score = scp
                    improved = True
                    stall = 0
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_f = f.copy()
            except Exception:
                pass
            last_polish = time.monotonic()

        # Escape / reset.
        if stall > 40:
            f = best_f.copy()
            f *= np.exp(rng.normal(0.0, 0.12, n))
            f = np.clip(f, 0.0, None)
            fs = float(f.sum())
            if fs <= 1e-300:
                f = best_f.copy()
                cur_score = best_score
            else:
                f /= fs
                cur_score = _score(f)
                if cur_score >= best_score:
                    f = best_f.copy()
                    cur_score = best_score
            stall = 0
            lr = 0.55

        it += 1

        if best_score < target - 1e-10:
            break

    # Final polish inside this run.
    if time.monotonic() < stop - 8.0:
        try:
            fp, scp = _coordinate_polish(best_f, stop, rng, max_passes=25)
            if math.isfinite(scp) and scp < best_score:
                best_f = fp
                best_score = scp
        except Exception:
            pass

    return _normalize(best_f, n), float(best_score)


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(20.0, float(budget_s))
    headroom = max(8.0, min(35.0, 0.05 * budget))
    work_end = t0 + max(10.0, budget - headroom)
    total = max(1.0, work_end - t0)

    rng = np.random.default_rng(seed)
    parent = _get_parent()
    n_parent = int(parent.size)
    if n_parent < 8:
        n_parent = 320

    best_seq = None
    best_exact = math.inf

    def consider(a):
        nonlocal best_seq, best_exact
        a = np.asarray(a, dtype=np.float64).ravel()
        if a.size == 0:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        s = float(a.sum())
        if not math.isfinite(s) or s < 0.01:
            return
        a = a / s
        sc = _exact(a)
        if math.isfinite(sc) and sc < best_exact:
            best_exact = sc
            best_seq = a.copy()

    # Strong small fallback, independent of parent.
    consider(np.array([2.0 / 3.0, 1.0 / 3.0], dtype=np.float64))
    consider(np.array([1.0], dtype=np.float64))
    if parent.size > 0:
        consider(parent)

    last_exact_update = t0

    # Primary search at the parent length.
    init_list = _make_inits(parent, n_parent, rng, keep=4)
    for arr in init_list:
        consider(arr)

    if len(init_list) and time.monotonic() < work_end:
        stop_a = t0 + 0.52 * total
        if math.isfinite(best_exact) and best_exact > TARGET_UPPER - 1e-8:
            try:
                res, _ = _optimize(init_list[0], n_parent, stop_a, seed + 1, use_lp=True)
                consider(res)
            except Exception:
                pass

        # Second sth initialization at parent length.
        if len(init_list) > 1:
            stop_b = t0 + 0.64 * total
            if time.monotonic() < stop_b - 20.0 and best_exact > TARGET_UPPER - 1e-8:
                try:
                    res, _ = _optimize(init_list[1], n_parent, stop_b, seed + 101, use_lp=True)
                    consider(res)
                except Exception:
                    pass

    # Other lengths, especially slightly larger ones.
    lengths = set()
    lengths.add(n_parent)
    for mult in (1.5, 2.0, 3.0):
        lengths.add(int(round(n_parent * mult)))
    for L in (128, 192, 256, 320, 384, 512, 768, 1024):
        lengths.add(L)
    lengths = [L for L in lengths if 16 <= L <= 1024]
    other_lengths = [L for L in lengths if L != n_parent]
    other_lengths.sort(key=lambda L: (abs(L - n_parent), L))
    other_lengths = other_lengths[:4]

    phase_end = t0 + 0.90 * total
    if other_lengths and time.monotonic() < phase_end - 25.0:
        n_runs = max(1, len([L for L in other_lengths if (phase_end - time.monotonic()) > 40]))
        time_per = max(25.0, (phase_end - time.monotonic()) / max(1, n_runs))
        for L in other_lengths:
            if time.monotonic() >= phase_end - 25.0:
                break
            if best_exact < TARGET_UPPER - 1e-10:
                break
            if time.monotonic() + time_per > phase_end - 5.0:
                break
            try:
                inits = _make_inits(parent, L, rng, keep=3)
                for arr in inits:
                    consider(arr)
                if not inits:
                    continue
                stop_l = min(phase_end - 5.0, time.monotonic() + time_per)
                res, _ = _optimize(
                    inits[0],
                    L,
                    stop_l,
                    seed + 1000 + L,
                    use_lp=(L <= LP_MAX_N),
                )
                consider(res)
            except Exception:
                pass

    # Final global polish.
    if best_seq is not None and time.monotonic() < work_end - 6.0:
        final_stop = work_end - 4.0
        try:
            fp, scp = _coordinate_polish(best_seq, final_stop, rng, max_passes=60)
            consider(fp)
        except Exception:
            pass

    if best_seq is None or not math.isfinite(best_exact):
        fb = np.array([2.0 / 3.0, 1.0 / 3.0], dtype=np.float64)
        sc = _exact(fb)
        if math.isfinite(sc):
            best_seq = fb
            best_exact = sc
        else:
            best_seq = np.array([1.0], dtype=np.float64)

    # Final validation with the authoritative evaluator.
    try:
        fb = _exact(best_seq)
        if math.isfinite(fb) and fb < best_exact:
            best_exact = fb
    except Exception:
        pass

    if not math.isfinite(best_exact):
        fb = _exact(np.array([1.0], dtype=np.float64))
        if math.isfinite(fb):
            best_seq = np.array([1.0], dtype=np.float64)
            best_exact = fb

    arr = np.asarray(best_seq, dtype=np.float64).ravel()
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not math.isfinite(s) or s < 0.01:
        arr = np.array([1.0], dtype=np.float64)
    else:
        arr = arr / s

    if time.monotonic() > work_end - 2.0:
        # Return immediately with headroom; no more work.
        pass

    return [float(x) for x in arr]
