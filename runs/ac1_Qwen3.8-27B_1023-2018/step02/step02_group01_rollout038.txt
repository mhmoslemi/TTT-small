    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    return c

def _exact_score(a):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    n = a.size
    b = _conv(a)
    mb = float(b.max())
    return 2.0 * n * mb / (s * s)

def _build_A(a, active_indices=None):
    """Build constraint matrix. If active_indices given, use only those rows."""
    n = a.size
    m = 2 * n - 1
    if active_indices is not None:
        k_idx = active_indices
    else:
        k_idx = np.arange(m, dtype=np.int64)
    
    i_idx = np.arange(n, dtype=np.int64)[None, :]
    j = k_idx[:, None] - i_idx
    mask = (j >= 0) & (j < n)
    safe_j = np.clip(j, 0, n - 1)
    A = np.where(mask, a[safe_j], 0.0)
    return A

def _lp_step(a, M, top_k=None, timeout_s=1.0):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0.
    Optionally use only top_k constraints (those with largest A row sums).
    """
    n = a.size
    m = 2 * n - 1
    
    if top_k is not None and top_k < m:
        # Use only the constraints that are most likely active
        # Heuristic: rows with larger sum are more constraining
        A_full = _build_A(a)
        row_sums = A_full.sum(axis=1)
        top_idx = np.argsort(row_sums)[-top_k:]
        A = A_full[top_idx]
        b_ub = np.full(len(top_idx), M)
    else:
        A = _build_A(a)
        b_ub = np.full(m, M)
    
    c_obj = -np.ones(n)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, 
                     bounds=[(0.0, None)] * n, 
                     method='highs',
                     options={'time_limit': timeout_s,
                              'dual_feasibility_tolerance': 1e-11,
                              'primal_feasibility_tolerance': 1e-11})
    except Exception:
        return None
    
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    if not np.isfinite(g).all() or g.sum() <= 0:
        return None
    return g

def _soft_max_gradient(a, t=0.01):
    """Compute gradient of soft-max of convolution."""
    n = a.size
    c = _conv(a)
    # Soft max: eps * log(sum exp(c[m]/eps))
    c_max = c.max()
    weights = np.exp((c - c_max) / t)
    weights /= weights.sum()
    
    # Gradient of c[m] with respect to a[i]:
    # dc[mm]/da[i] = a[mm-i] + a[i-mm] (with zero padding)
    # Gradient of soft-max = sum_m weights[m] * dc[mm]/da[i]
    # = sum_m weights[m] * (a[mm-i] + a[i-mm])
    
    # For the first term: sum_m weights[m] * a[m-i] for valid m-i
    # = correlation of weights with a (shifted)
    # For the second term: sum_m weights[m] * a[i-m] for valid i-m
    # = correlation of weights with a (shifted other direction)
    
    # Actually, let's just compute it directly using the full convolution structure
    # grad[i] = sum_m weights[m] * a[m-i] + sum_m weights[m] * a[i-m]
    # First sum: for fixed i, sum over m where 0 <= m-i < n, i.e., i <= m <= i+n-1
    # = sum_{m=i}^{min(i+n-1, 2n-2)} weights[m] * a[m-i]
    # This is the cross-correlation of weights and a at lag i
    
    # Let's compute using FFT
    # First term: (conv of reversed weights with a) ... hmm this is getting complex
    # Let me just do it with the identity: grad = 2 * conv(weighted_profile, a) approximately
    
    # Actually, the cleanest way:
    # c[m] = sum_j a[j] * a[m-j]
    # dc[m]/da[i] = a[m-i] + a[i-m] (with a[k]=0 for k<0 or k>=n)
    # 
    # grad[i] = sum_m w[m] * (a[m-i] + a[i-m])
    # = sum_m w[m] * a[m-i] + sum_m w[m] * a[i-m]
    #
    # First term: for m from i to i+n-1 (clamped): w[m] * a[m-i]
    #   = sum_{k=0}^{n-1} w[i+k] * a[k] (where w is zero-padded)
    # Second term: for m from 0 to i (clamped): w[m] * a[i-m]
    #   = sum_{k=0}^{i} w[m] * a[i-m] where m = i-k
    #   = sum_{k=0}^{min(i, 2n-2)} w[i-k] * a[k]
    
    # So grad[i] = sum_k a[k] * (w[i+k] + w[i-k]) (with w zero-padded)
    # This is a convolution of a with the "symmetrized and shifted" weights!
    
    # Let's define v[i] = sum_k a[k] * (w[i+k] + w[i-k])
    # = (a * w_shifted)[i] + (a * w_shifted_2)[i]
    
    # Actually, the cleanest computation:
    # Define W[m] = weights[m] for m in [0, 2n-2], 0 otherwise
    # Then grad[i] = sum_m W[m] * a[m-i] + sum_m W[m] * a[i-m]
    # First sum: (W_conv_a)[i] where W_conv_a[i] = sum_m W[m] * a[m-i]
    #   This is the cross-correlation of W and a
    # Second sum: (W_conv_a_rev)[i] where we reverse W
    #   = sum_m W[m] * a[i-m]
    #   = sum_k W[2n-2-k] * a[i-(2n-2-k)] ... getting confusing
    
    # Let me just use a simple O(n^2) approach for n=191 (it's fast enough)
    grad = np.zeros(n)
    for i in range(n):
        s1 = 0.0
        s2 = 0.0
        for m in range(2 * n - 1):
            # a[m-i] valid if 0 <= m-i < n
            if 0 <= m - i < n:
                s1 += weights[m] * a[m - i]
            # a[i-m] valid if 0 <= i-m < n
            if 0 <= i - m < n:
                s2 += weights[m] * a[i - m]
        grad[i] = s1 + s2
    
    return grad, c.max()

def _soft_max_gradient_fast(a, t=0.01):
    """Faster gradient using FFT."""
    n = a.size
    c = _conv(a)
    c_max = c.max()
    weights = np.exp((c - c_max) / t)
    weights /= weights.sum()
    
    # grad[i] = sum_m weights[m] * a[m-i] + sum_m weights[m] * a[i-m]
    # First term: this is the cross-correlation of weights (length 2n-1) with a (length n)
    # cross_corr[i] = sum_m weights[m] * a[m-i]
    # For fixed i: sum over m where 0 <= m-i < n, i.e., m in [i, i+n-1]
    # = sum_{k=0}^{n-1} weights[i+k] * a[k]
    # This is the "sliding dot product" of weights[i:i+n] with a
    # = correlation (not convolution)
    
    # Let's use the FFT approach:
    # cross_corr[i] = sum_k a[k] * weights[i+k] for k=0..n-1 (where weights is defined on [0, 2n-2])
    # This is the cross-correlation of a and weights at lag i.
    # Using FFT: cross_corr = ifft(fft(a) * conj(fft(weights_padded)))
    # But we need to be careful with the definition.
    
    # Actually, let me just use np.correlate or convolve
    # cross_corr[i] = sum_k a[k] * weights[i+k]
    # = (a correlated with weights)[i] (in the signal processing sense)
    
    # np.correlate(weights, a, 'valid') gives sum_k a[k] * weights[k+i] for i=0..len(weights)-n
    # That's exactly what we want for the first term!
    
    first_term = np.correlate(weights, a, mode='valid')  # length (2n-1) - n + 1 = n
    # first_term[i] = sum_k a[k] * weights[i+k] for k=0..n-1
    
    # Second term: sum_m weights[m] * a[i-m] for m where 0 <= i-m < n, i.e., m in [max(0,i-n+1), min(i, 2n-2)]
    # = sum_k weights[i-k] * a[k] for k=0..min(i, n-1) (since a[k]=0 for k>=n)
    # This is (a correlated with reversed weights) ... 
    # = sum_k a[k] * weights[i-k]
    # This is the convolution of a with weights at position i!
    # (a * weights)[i] = sum_k a[k] * weights[i-k]
    
    # But weights has length 2n-1 and a has length n
    conv_result = np.convolve(a, weights, mode='full')  # length 3n-2
    second_term = conv_result[:n]  # positions 0 to n-1
    
    # Wait, (a * weights)[i] = sum_k a[k] * weights[i-k] for k where both are defined
    # a[k] is defined for k in 0..n-1
    # weights[i-k] is defined for i-k in 0..2n-2, i.e., k in [i-(2n-2), i]
    # So the sum is over k in [max(0, i-2n+2), min(n-1, i)]
    # For i in 0..n-1: k in [0, min(n-1, i)] = [0, i]
    # So second_term[i] = sum_{k=0}^{i} a[k] * weights[i-k]
    
    # And we want: sum_m weights[m] * a[i-m] = sum_{m=0}^{i} weights[m] * a[i-m] (for i < n)
    # = sum_{k=0}^{i} a[k] * weights[i-k] (substituting k = i-m)
    # Yes! So second_term[i] = conv_result[i] for i = 0..n-1. ✓
    
    grad = first_term + second_term
    return grad, c.max()
