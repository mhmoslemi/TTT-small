import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len

def _softmax(x, max_clip=80.0):
    x = np.asarray(x, dtype=np.float64)
    x = x - x.max()
    x = np.clip(x, -max_clip, max_clip)
    e = np.exp(x)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.full_like(x, 1.0/x.size)
    return e / s

def _fft_conv(a, fs):
    n = len(a)
    F = rfft(a, fs)
    return irfft(F * F, fs)[:2*n-1]

def _fft_corr(p, a, fs):
    """Compute cross-correlation: result[j] = sum_m p[m+j]*a[m], j=0..len(a)-1"""
    P = rfft(p, fs)
    A = rfft(a, fs)
    return irfft(P * np.conj(A), fs)[:len(a)]

def _grad_softmax(b, a, tau, fs):
    """Gradient of soft_max_tau(b) w.r.t. a, where b = conv(a,a)."""
    # p = softmax(b / tau)
    bt = b / tau
    bt = bt - bt.max()
    bt = np.clip(bt, -80, 80)
    e = np.exp(bt)
    s = e.sum()
    if s < 1e-300:
        p = np.zeros_like(b)
        p[int(np.argmax(b))] = 1.0
    else:
        p = e / s
    # g = 2 * correlate(p, a)
    g = 2.0 * _fft_corr(p, a, fs)
    return g

def _opt(n, a0, tau0, lr0, n_iter, end_time, rng, sym=False):
    """Run Adam optimization for n_iter steps or until end_time."""
    fs = next_fast_len(2*n - 1)
    z = np.log(np.maximum(a0, 1e-12))
    z = z - z.mean()
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = lr0
    tau = tau0
    best_z = z.copy()
    best_score = np.inf
    stuck = 0
    
    for it in range(n_iter):
        if time.monotonic() >= end_time:
            break
        
        a = _softmax(z)
        b = _fft_conv(a, fs)
        bmax = b.max()
        score = 2.0 * n * bmax  # sum(a)=1 by construction
        
        if score < best_score - 1e-13:
            best_score = score
            best_z = z.copy()
            stuck = 0
        else:
            stuck += 1
        
        # Compute gradient
        tau = max(1e-6, tau0 * (0.95 ** it))
        g = _grad_softmax(b, a, tau, fs)
        
        # Project to tangent space of simplex
        g = g - g.mean()
        
        # Adam
        t += 1
        m = 0.9 * m + 0.1 * g
        v = 0.999 * v + 0.001 * g**2
        mhat = m / (1 - 0.9**t)
        vhat = v / (1 - 0.999**t)
        
        step = mhat / (np.sqrt(vhat) + 1e-8)
        step_norm = np.abs(step).max()
        if step_norm < 1e-15:
            continue
        step = step / step_norm * lr
        
        z = z - step
        z = z - z.mean()
        z = np.clip(z, -50, 50)
        
        # Adaptive lr
        if stuck > 50:
            lr = max(0.01, lr * 0.9)
        if stuck > 200:
            # Restart
            z = best_z + rng.normal(0, 0.5, n)
            z = z - z.mean()
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            lr = lr0
            stuck = 0
        
        if sym and it % 10 == 0:
            a = _softmax(z)
            a = (a + a[::-1]) / 2
            z = np.log(np.maximum(a, 1e-12))
            z = z - z.mean()
    
    a = _softmax(best_z)
    b = _fft_conv(a, fs)
    score = 2.0 * n * b.max()
    return a, score
