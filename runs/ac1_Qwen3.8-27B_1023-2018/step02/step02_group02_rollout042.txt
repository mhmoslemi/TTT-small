    A = convolve(a,a), B = convolve(a,g), C = convolve(g,g)

    # (1-t)^2*A[k] + 2t(1-t)*B[k] + t^2*C[k] <= M for all k

    # = A[k] - 2t*A[k] + t^2*A[k] + 2t*B[k] - 2t^2*B[k] + t^2*C[k] <= M
    # = t^2*(A[k] - 2*B[k] + C[k]) + t*(-2*A[k] + 2*B[k]) + (A[k] - M) <= 0
    
    lo_t, hi_t = 0.0, 1.0

    M_lo = 0.0  # will be infeasible (or barely)
    M_hi = max(A.max(), C.max())  # feasible

    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate ≈ 0.618
    
    def obj(t):
        """max_k c(t)[k]"""
        c = (1-t)**2 * A + 2*t*(1-t) * B + t**2 * C

        F = rfft(a, fsize)
        c = irfft(F * F, fsize)[:2 * n - 1]
        return np.maximum(c, 0.0)

        Fa = rfft(a, fsize)
        Fb = rfft(b, fsize)

        def obj(t):
            ct = (1.0 - t) * (1.0 - t) * A + 2.0 * t * (1.0 - t) * Bc + t * t * Cc
            return float(np.maximum(ct, 0.0).max())
