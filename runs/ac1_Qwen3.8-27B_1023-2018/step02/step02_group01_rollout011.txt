    if p.size == 0 or not np.isfinite(p).all():
        p = np.ones(192, dtype=np.float64) / 192.0
    return p

def _autocorr(a):
    """Compute full convolution a*a."""
    n = a.size
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a.astype(np.float64), fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    return c

def _score_fn(a):
    """Score with scale invariance handled externally."""
    n = a.size
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    b = _autocorr(a)
    return 2.0 * n * b.max() / (s * s)

def _exact_score(a):
    """Exact evaluator score."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = a.size
    b = _autocorr(a)
    return float(2.0 * n * b.max() / (s * s))

def _build_toeplitz(a):
    """Build the Toeplitz matrix A of size (2n-1) x n where A[k,i] = a[k-i] if valid."""
    n = a.size
    m = 2 * n - 1
    # A[k, i] = a[k-i] if 0 <= k-i < n, else 0
    # Vector form: first column is [a_0, 0, 0, ..., 0], first row is [a_0, a_1, ..., a_{n-1}]
    # Actually, A[k,i] = a[k-i] for max(0, k-n+1) <= i <= min(n-1, k)
    # This is a Toeplitz matrix: column i has a_i at row i, a_{i-1} at row i-1, etc.
    # More precisely: A[k,i] = a[k-i] if 0 <= k-i <= n-1
    # Column 0: [a_0, a_0? No...]
    # A[0,0] = a[0]
    # A[1,0] = a[1], A[1,1] = a[0]
    # A[2,0] = a[2], A[2,1] = a[1], A[2,2] = a[0]
    # etc.
    # So column i has a[0] at row i, a[1] at row i+1, ..., a[n-1-i] at row i+n-1-i = n-1... wait
    # A[k,i] = a[k-i] for 0 <= k-i < n, i.e., i <= k <= i+n-1
    # In column i, non-zero entries are at rows i, i+1, ..., i+n-1
    # Row i: A[i,i] = a[0]
    # Row i+1: A[i+1,i] = a[1]
    # ...
    # Row i+n-1: A[i+n-1,i] = a[n-1]
    
    # This is a Toeplitz matrix. Let's just build it directly.
    idx = np.arange(n)
    k = np.arange(m)
    # A[k,i] = a[k-i] if 0 <= k-i < n
    j = k[:, None] - idx[None, :]
    mask = (j >= 0) & (j < n)
    safe_j = np.clip(j, 0, n-1)
    A = np.where(mask, a[safe_j], 0.0)
    return A

def _lp_best_response(a, M, timeout=1.0):
    """Solve LP: max sum(g) s.t. A @ g <= M * ones, g >= 0."""
    n = a.size
    A = _build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                      method='highs', options={'time_limit': timeout})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    if not np.isfinite(g).all() or g.sum() <= 0:
        return None
    return g

def _optimize_lp(a0, stop_time):
    """Run LP best-response iteration."""
    a = np.clip(a0.astype(np.float64).copy(), 0.0, 1000.0)
    s = a.sum()
    if s <= 0:
        a = np.ones(a.size) / a.size
        s = 1.0
    a = a / s
    
    best_a = a.copy()
    best_M = float(_autocorr(a).max())
    
    stuck = 0
    n = a.size
    
    for it in range(100000):
        if time.monotonic() > stop_time - 2.0:
            break
        
        b = _autocorr(a)
        M = float(b.max())
        
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1
        
        if stuck > 500:
            # Perturb
            eps = 0.1 / np.sqrt(stuck / 100.0)
            a = a * np.exp(np.random.default_rng(it).normal(0, eps, n))
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            stuck = 0
            continue
        
        g = _lp_best_response(a, M * (1 + 1e-9), timeout=0.5)
        if g is None:
            break
        
        gs = g.sum()
        g = g / gs
        
        # Check if g is very different from a
        diff = np.max(np.abs(g - a))
        
        # Line search
        improved = False
        for t in (0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01):
            cand = (1 - t) * a + t * g
            cand = np.maximum(cand, 0.0)
            cs = cand.sum()
            if cs <= 0:
                continue
            cand /= cs
            cM = float(_autocorr(cand).max())
            if cM < M * (1 - 1e-12):
                a = cand
                improved = True
                break
        
        if not improved:
            stuck += 1
    
    return best_a

def _gradient_descent_subgrad(a0, stop_time):
    """Subgradient/coordinate descent approach."""
    n = a0.size
    a = np.clip(a0.astype(np.float64).copy(), 0.0, 1000.0)

        b = _autocorr(a)

        if M < best_M * (1 - 1e-12):

            best_a = a.copy()
        
        # Subgradient of max_k (a*a)_k w.r.t. a
        # (a*a)_k = sum_i a[i]*a[k-i]

        # The subgradient of max is the subgradient of the active component
        grad = np.zeros(n)
        for j in range(n):
            i1 = k_star - j  # a[k_star - j] if 0 <= k_star - j < n
            i2 = j - (n - 1 - (k_star - (n - 1)))  # Hmm, this is getting complicated
            # Actually (a*a)_{k} = sum_{i: 0<=i<n, 0<=k-i<n} a[i] * a[k-i]
            # d/d a[j] = a[k-j] (if 0 <= k-j < n)
            # But also, if j appears as k-i, then i = k-j, and we get a[k-j] * 1
            # Wait, (a*a)_k = sum_i a[i]*a[k-i]. d/d a[j] = a[k-j] (if valid) + ... 
            # Actually, a[i]*a[k-i]: if i=j, term is a[j]*a[k-j]; if k-i=j, i.e., i=k-j, term is a[k-j]*a[j].
            # These are the same term (both give a[j]*a[k-j]).
            # So d (a*a)_k / d a[j] = a[k-j] if 0 <= k-j < n, else 0.
            # But wait, this double-counts when j = k-j, i.e., 2j = k.
            # Actually no: (a*a)_k = sum_i a[i]*a[k-i]. This is a sum over i.
            # d/d a[j] = sum_i [d a[i]/d a[j] * a[k-i] + a[i] * d a[k-i]/d a[j]]

            # = 2*a[k-j] if 0 <= k-j < n and j != k-j
            # = a[k-j] if j = k-j (i.e., 2j = k)
            # Hmm, but actually (a*a)_k is not a quadratic form in a. It's a product a[i]*a[k-i].
            # Let me just compute the gradient numerically or use the formula.
            pass
        
        # Actually, let me just do coordinate descent
        j = rng.integers(0, n)
        for delta in [0.01, -0.01, 0.005, -0.005, 0.001, -0.001]:
            cand = a.copy()
            cand[j] = max(0, a[j] + delta * a[j])  # relative change
            cs = cand.sum()
            if cs <= 0:
                continue
            cand /= cs
            cM = float(_autocorr(cand).max())
