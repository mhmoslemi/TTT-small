        k = rows[i]
        j = cols[i]
        vals[i] = a[k - j]
    return sps.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))

                                    a = a_pert
                                    local = sc(a)
                                    stag = 0
                                    update_best(a)
                                    break
                    if stag >= 3 and local >= sc(a):
                        # Random kick
                        a_pert = a * np.exp(0.05 * rng.standard_normal(n))
                        a_pert = np.clip(a_pert, 0, None)
                        sp2 = float(a_pert.sum())
                        if sp2 > 1e-15:
                            a_pert /= sp2
                            if sc(a_pert) < local:
                                a = a_pert
                                local = sc(a)
                                update_best(a)
                if stag > 30:
                    break

        return a

    # Gradient refinement with bundle
    def grad_refine(a, n, max_iters, stop):
        nonlocal best_score, best_arr
        a = np.asarray(a, dtype=np.float64).ravel().copy()
        a = np.clip(a, 0, None)
        s = a.sum()
        if s < 1e-15:
            return a
        a /= s
        local = sc(a)
        update_best(a)
        stale = 0
        alpha = 0.03
        bundle = np.zeros(n, dtype=np.float64)
        bundle_count = 0

        for it in range(max_iters):
            if time.monotonic() >= stop:
                break

            b_conv = np.convolve(a, a)
            M = float(b_conv.max())
            if not math.isfinite(M) or M <= 0:
                break

            thr = M * 0.995
            active = np.where(b_conv >= thr)[0]
            grad = np.zeros(n, dtype=np.float64)
            for k in active:
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                if lo <= hi:
                    js = np.arange(lo, hi + 1)
                    grad[js] += a[k - js]

            # Bundle: exponential moving average
            bundle = 0.9 * bundle + 0.1 * grad
            if bundle_count > 10:
                grad_use = bundle
            else:
                grad_use = grad
            bundle_count += 1

            grad_use -= grad_use.mean()
            gn = float(np.linalg.norm(grad_use))
            if gn < 1e-14:
                break
            grad_use /= gn

            # Backtracking
            best_sc_t = local
            best_a_t = None
            for a_try in (alpha * 0.25, alpha * 0.5, alpha, alpha * 2.0, alpha * 4.0):
                cand = np.clip(a - a_try * grad_use, 0, None)
                cs = float(cand.sum())
                if cs < 1e-15:
                    continue
                cand /= cs
                v = sc(cand)
                if v < best_sc_t:
                    best_sc_t = v
                    best_a_t = cand

            if best_a_t is not None and best_sc_t < local * (1.0 - 1e-10):
                a = best_a_t
                local = best_sc_t
                stale = 0
                alpha = max(0.001, alpha * 0.97)
                if it % 50 == 0:
                    update_best(a)
            else:
                stale += 1
                alpha = max(0.001, alpha * 0.92)
                if stale % 100 == 0:
                    a_pert = np.clip(a - 0.01 * alpha * rng.standard_normal(n), 0, None)
                    sp2 = float(a_pert.sum())
                    if sp2 > 1e-15:
                        a_pert /= sp2
                        if sc(a_pert) < local:
                            a = a_pert
                            local = sc(a)
                            stale = 0
                            update_best(a)
                if stale > 500:
                    break

        return a

    def interp_len(src, n):
        if src.size == n:
            return src.copy()
        ox = np.linspace(0.0, 1.0, src.size)
        nx = np.linspace(0.0, 1.0, n)
        out = np.interp(nx, ox, src)
        out = np.clip(out, 0, None)
        s = float(out.sum())
        if s > 1e-15:
            out /= s
        else:
            out = np.ones(n, dtype=np.float64) / n
        return out

    # === MAIN OPTIMIZATION ===
    p = parent if (parent is not None) else best_arr
    # If parent not available, use best_arr
    if 'parent' not in dir() or parent is None:
        p = best_arr.copy()
    n_p = p.size
    p_norm = p / p.sum()

    # Phase 1: Intensive LP at parent length
    if n_p >= 8 and linprog is not None and sps is not None:
        t_rem = deadline - time.monotonic()
        alloc1 = min(550.0, t_rem * 0.45)
        if alloc1 > 30:
            a1 = p_norm.copy()
            a1 = lp_refine(a1, n_p, 2000, time.monotonic() + alloc1)
            update_best(a1)

    # Phase 2: Gradient refinement
    if time.monotonic() < deadline - 50:
        a1 = grad_refine(a1, n_p, 80000, deadline - 45)
        update_best(a1)

    # Phase 3: Second LP pass (polish)
    if time.monotonic() < deadline - 50 and linprog is not None and sps is not None:
        t_rem = deadline - time.monotonic() - 45
        alloc = min(250.0, t_rem * 0.5)
        if alloc > 20:
            a1 = lp_refine(a1, n_p, 1500, time.monotonic() + alloc)
            update_best(a1)
            if time.monotonic() < deadline - 25:
                a1 = grad_refine(a1, n_p, 30000, deadline - 20)
                update_best(a1)

    # Phase 4: Try nearby lengths
    if time.monotonic() < deadline - 35 and linprog is not None and sps is not None:
        lengths_to_try = [n_p - 8, n_p - 4, n_p - 2, n_p + 2, n_p + 4, n_p + 8, n_p + 16]
        for L in lengths_to_try:
            if time.monotonic() >= deadline - 25:
                break
            t_rem = deadline - time.monotonic() - 25
            if t_rem < 15:
                break
            if L < 8:
                continue
            alloc = min(30.0, t_rem * 0.2)
            if alloc < 8:
                continue
            a_init = interp_len(p, L)
            a_w = lp_refine(a_init, L, max(20, int(alloc // 0.3)), time.monotonic() + alloc * 0.7)
            update_best(a_w)
            if time.monotonic() < deadline - 20:
                a_w = grad_refine(a_w, L, max(2000, int(alloc * 3)), deadline - 18)
                update_best(a_w)

    # Phase 5: Multiple restarts at parent length
    if time.monotonic() < deadline - 30 and linprog is not None and sps is not None:
        n_restarts = 8
        for r in range(n_restarts):
            if time.monotonic() >= deadline - 22:
                break
            t_rem = deadline - time.monotonic() - 22
            alloc = min(25.0, t_rem / max(1, n_restarts - r))
            if alloc < 5:
                break

            rtype = r % 4
            if rtype == 0:
                a_init = p_norm * np.exp(0.08 * rng.standard_normal(n_p))
            elif rtype == 1:
                a_init = p_norm[::-1].copy()
            elif rtype == 2:
                # Asymmetric perturbation
                a_init = p_norm * np.exp(0.1 * np.linspace(-1, 1, n_p))
            else:
                # Bimodal
                a_init = p_norm.copy()
                mid = n_p // 2
                a_init[:mid] *= 0.8
                a_init[mid:] *= 1.2

            s_init = float(np.clip(a_init, 0, None).sum())
            if s_init > 1e-15:
                a_init = np.clip(a_init, 0, None) / s_init
                a_w = lp_refine(a_init, n_p, max(15, int(alloc // 0.3)), time.monotonic() + alloc * 0.6)
                update_best(a_w)
                if time.monotonic() < deadline - 18:
                    a_w = grad_refine(a_w, n_p, int(alloc * 2), deadline - 15)
                    update_best(a_w)

    # Phase 6: Final intensive polish on best
    if time.monotonic() < deadline - 10 and best_arr is not None:
        n_b = best_arr.size
        a_pol = best_arr.copy() / best_arr.sum()
        if linprog is not None and sps is not None and time.monotonic() < deadline - 15:
            a_pol = lp_refine(a_pol, n_b, 300, deadline - 12)
        if time.monotonic() < deadline - 8:
            a_pol = grad_refine(a_pol, n_b, 50000, deadline - 5)
        update_best(a_pol)

    # Final validation
    if best_arr is None:
        best_arr = np.ones(320, dtype=np.float64)
        update_best(best_arr)

    final = np.clip(best_arr, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones_like(best_arr, dtype=np.float64)
    update_best(final)

    return [float(x) for x in best_arr]
