def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.clip(a, 0.0, 1000.0)
    n = a.size
    if n == 0:
        return float('inf')
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    b = np.convolve(a, a)
    mb = float(b.max())
    if not math.isfinite(mb):
        return float('inf')
    return 2.0 * n * mb / (s * s)

def _proj_simplex(v):
    """Project v onto {x: sum(x)=1, x>=0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    ind = np.arange(1, n+1)
    cond = u - (cssv - 1.0) / ind > 0
    i = ind[cond][-1]
    rho = (cssv[i-1] - 1.0) / i
    return np.maximum(v - rho, 0.0)

def _smooth_grad(a, theta):
    """Compute gradient of smooth max autocorrelation and the score."""
    n = len(a)
    h = np.convolve(a, a)  # length 2n-1
    max_h = h.max()
    # log-sum-exp for stability
    shifted = theta * (h - max_h)
    log_sum_exp = np.log(np.sum(np.exp(shifted)))
    w = np.exp(shifted) / np.sum(np.exp(shifted))
    # smooth objective value (up to additive constant)
    smooth_val = max_h + np.log(np.sum(np.exp(shifted))) / theta
    # Gradient: grad[i] = 2 * sum_j w[i+j] * a[j]
    # Build the weight matrix M[i,j] = w[i+j]
    idx = np.arange(n)
    M = w[idx[:,None] + idx[None,:]]  # n x n, M[i,j] = w[i+j]
    grad = 2.0 * (M @ a)
    return grad, smooth_val, h

def _lp_direction(a, M_target, n, time_limit=10):
    """Solve LP to find direction g."""
    col_idx = np.repeat(np.arange(n, dtype=np.int64), n)
    row_idx = col_idx + np.tile(np.arange(n, dtype=np.int64), n)
    vals = np.tile(a, n).astype(np.float64)
    A = sps.csr_matrix((vals, (row_idx, col_idx)), shape=(2*n-1, n))
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M_target + 1e-14)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, None)*n],
                      method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
    except Exception:
        return None
    if res is not None and getattr(res, 'x', None) is not None:
        g = np.asarray(res.x, dtype=np.float64).ravel()
        g = np.clip(g, 0.0, None)
        sg = float(g.sum())
        if sg > 1e-15 and math.isfinite(sg):
            return g / sg  # normalize to sum=1
    return None
