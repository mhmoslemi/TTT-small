import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, next_fast_len


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(191, dtype=np.float64)
    if p.size == 0 or not np.isfinite(p).all():
        p = np.ones(191, dtype=np.float64)
    return np.clip(p, 0.0, 1000.0)


def _conv2(a):
    """Full linear convolution a*a via FFT."""
    n = a.size
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    return irfft(F * F, fsize)[:2 * n - 1]


def _score(a):
    """Exact score: 2n * max(a*a) / (sum a)^2."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = a.size
    return 2.0 * n * _conv2(a).max() / (s * s)


def _norm1(a):
    """Normalize to sum=1, nonnegative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0:
        return np.ones(a.size) / a.size
    return a / s


def _conv_matrix(a):
    """Build (2n-1) x n matrix where A[k,i] = a[k-i] if valid."""
    n = a.size
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for k in range(m):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            A[k, idx] = a[k - idx]
    return A


def _lp_solve(a, M, tlimit=3.0):
    """LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0. Returns g or None."""
    n = a.size
    A = _conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'time_limit': tlimit,
                               'primal_feasibility_tolerance': 1e-10,
                               'dual_feasibility_tolerance': 1e-10})
    except Exception:
        return None

        return None
    g = np.asarray(res.x, dtype=np.float64)
    if not np.isfinite(g).all() or g.sum() <= 1e-15:
        return None
    return g


def _subgradient(a, kstar):
    """Subgradient of (a*a)[kstar] w.r.t. a."""
    n = a.size
    sg = np.zeros(n)
    j_range = np.arange(n)
    k_minus_j = kstar - j_range
    valid = (k_minus_j >= 0) & (k_minus_j < n)
    sg[valid] = 2.0 * a[k_minus_j[valid]]
    # Adjust for j = kstar/2 (the diagonal term)
    if kstar % 2 == 0:
        j = kstar // 2
        if 0 <= j < n:
            sg[j] = a[j]  # Only one term, not two
    return sg

                        break
                    # Perturb

                    a = _norm1(a)
                    continue

            # Line search
            improved = False
            for t in [1.0, 0.5, 0.3, 0.1, 0.05]:
                cand = (1 - t) * a + t * g
                cand = _norm1(cand)
                if _conv2(cand).max() < M * (1 - 1e-11):
                    a = cand
                    improved = True
                    break
            if not improved:

                    a = a * np.exp(rng.normal(0, 0.05, n))
                    a = _norm1(a)

                cand = _norm1(cand)
                if _conv2(cand).max() < M * (1 - 1e-10):
                    a = cand
                    break

from scipy.optimize import linprog
from scipy.fft import rfft, irfft, next_fast_len


def _get_parent():

        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:

def _conv2(a):
    n = a.size
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    return irfft(F * F, fsize)[:2 * n - 1]


def _score(a):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = a.size
    return 2.0 * n * float(_conv2(a).max()) / (s * s)


def _norm1(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / s


def _conv_matrix(a):
    n = a.size
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for k in range(m):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            A[k, idx] = a[k - idx]
    return A


def _lp_solve(a, M, tlimit=3.0):
    n = a.size
    A = _conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'time_limit': tlimit,
                               'primal_feasibility_tolerance': 1e-11,
                               'dual_feasibility_tolerance': 1e-11})
    except Exception:
        return None, None
    if res.status != 0 or res.x is None:
        return None, None
    g = np.asarray(res.x, dtype=np.float64)
    if not np.isfinite(g).all() or g.sum() <= 1e-15:
        return None, None
    return g, float(-res.fun)


def _subgrad_step(a, b, kstar):
    n = a.size
    sg = np.zeros(n)
    j_range = np.arange(n)
    kmj = kstar - j_range
    valid = (kmj >= 0) & (kmj < n)
    sg[valid] = 2.0 * a[kmj[valid]]
    if kstar % 2 == 0:
        j = kstar // 2
        if 0 <= j < n:
            sg[j] = a[j]
    return sg
