            return float('inf')
        s = float(a.sum())
        if s < 1e-15:
            return float('inf')
        b = np.convolve(a, a)
        return 2.0 * n * float(b.max()) / (s * s)

    best_score = float('inf')
    best_arr = None

    def ub(a):
        nonlocal best_score, best_arr
        a = np.asarray(a, dtype=np.float64).ravel()
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0 or float(a.sum()) < 0.01:
            return
        try:
            v = evaluate_sequence([float(x) for x in a])
            if math.isfinite(v) and v < best_score:
                best_score = v
                best_arr = a.copy()
        except Exception:
            pass

    # Get parent construction
    parent = None
    h = globals().get('height_sequence_1', None)
    if h is not None:
        parent = np.asarray(h, dtype=np.float64).ravel().clip(0, 1000)
        if parent.size > 0 and float(parent.sum()) > 0.01:
            ub(parent)

    if best_arr is None:
        best_arr = np.ones(96, dtype=np.float64)
        ub(best_arr)

    p = (parent if parent is not None else best_arr).astype(np.float64).copy()
    p = p / p.sum()
    n_p = p.size

    def build_A(a):
        """Build convolution matrix: A[k,j] = a[k-j] if 0<=k-j<n"""
        n = len(a)
        A = np.zeros((2 * n - 1, n), dtype=np.float64)
        for j in range(n):
            A[j:j + n, j] = a
        return A

    def lp_solve(a, M, margin=1.0, tlim=15):
        """Solve LP: max sum(g) s.t. A_a @ g <= M*margin, g >= 0"""
        n = len(a)
        A = build_A(a)
        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M * margin + 1e-14)
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub,
                          bounds=[(0.0, None)] * n,
                          method='highs',
                          options={'time_limit': tlim, 'presolve': True})
            if res.x is not None and res.status == 0:
                return np.clip(res.x, 0, None)
        except Exception:
            pass
        return None

    def smooth_max_direction(a, lam=40.0):
        """Compute descent direction using smooth max of b_k."""
        n = len(a)
        b = np.convolve(a, a)
        b_shift = b - b.max()
        w = np.exp(lam * b_shift)
        w_sum = w.sum()
        if w_sum < 1e-30:
            return None
        w /= w_sum

        grad = np.zeros(n, dtype=np.float64)
        # Only iterate over significant weights
        sig = np.where(w > 1e-10)[0]
        for k in sig:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            grad[idx] += 2.0 * w[k] * a[k - idx]

        grad -= grad.mean()
        gn = float(np.linalg.norm(grad))
        if gn < 1e-14:
            return None
        return grad / gn

    def optimize(a, n, stop_time):
        """Main iterative optimization for sequence of length n."""
        a = np.asarray(a, dtype=np.float64).ravel().copy()
        a = np.clip(a, 0, None)
        s = float(a.sum())
        if s < 1e-15:
            return a
        a /= s
        local = sc(a)
        ub(a)
        stal = 0
        iter_count = 0

        while time.monotonic() < stop_time - 4:
            iter_count += 1
            b = np.convolve(a, a)
            M = float(b.max())

            improved = False
            # Try LP with progressively tighter margins
            for margin in [1.0, 0.997, 0.995, 0.99, 0.985, 0.98]:
                if time.monotonic() >= stop_time - 4:
                    break
                g = lp_solve(a, M, margin)
                if g is None:
                    continue
                sg = float(g.sum())
                if sg <= 1.0 + 1e-8:
                    continue
                g = g / sg

                # Line search with several t values
                bt, bsc = 0.0, local
                for t in [0.3, 0.5, 0.6, 0.7, 0.8, 0.9, 1.0]:
                    cand = (1 - t) * a + t * g
                    cand = np.clip(cand, 0, None)
                    cs = float(cand.sum())
                    if cs < 1e-15:
                        continue
                    cand /= cs
                    v = sc(cand)
                    if v < bsc:
                        bsc, bt = v, t

                if bt > 0 and bsc < local * (1 - 5e-7):
                    a = (1 - bt) * a + bt * g
                    a = np.clip(a, 0, None)
                    a /= a.sum()
                    local = bsc
                    improved = True
                    ub(a)
                    break

            if not improved:
                stal += 1
                # Gradient step using smooth max
                if stal <= 30:
                    direction = smooth_max_direction(a, lam=40.0)
                    if direction is not None:
                        for alpha in [0.001, 0.002, 0.005, 0.01, 0.02, 0.03, 0.05]:
                            cand = a - alpha * direction
                            cand = np.clip(cand, 0, None)
                            cs = float(cand.sum())
                            if cs < 1e-15:
                                continue
                            cand /= cs
                            v = sc(cand)
                            if v < local * (1 - 1e-7):
                                a = cand
                                local = v
                                ub(a)
                                stal = 0
                                break
                # Perturbation to escape local minima
                elif stal > 30 and stal % 20 == 0:
                    strength = 0.02 + 0.01 * min(stal // 20, 5)
                    pert = a + strength * rng.standard_normal(n)
                    pert = np.clip(pert, 0, None)
                    ps = float(pert.sum())
                    if ps > 1e-15:
                        pert /= ps
                        vp = sc(pert)
                        if vp < local:
                            a = pert
                            local = vp
                            ub(a)
                            stal = 0
                elif stal > 80:
                    break
            else:
                stal = 0

            # Periodic exact evaluation
            if iter_count % 100 == 0:
                ub(a)

        return a

    # === Phase 1: Optimize at parent length (primary) ===
    stop1 = deadline - max(80.0, budget * 0.12)
    a_main = optimize(p, n_p, stop1)
    ub(a_main)

    # === Phase 2: Try nearby and other lengths ===
    other_lengths = sorted(set([
        n_p - 40, n_p - 20, n_p + 20, n_p + 40,
        n_p + 80, n_p + 120,
        160, 200, 256, 384, 448, 512, 640, 768, 896
    ]))
    other_lengths = [L for L in other_lengths if 8 <= L <= 1024 and L != n_p]

    for L in other_lengths:
        if time.monotonic() >= deadline - 25:
            break
        remaining = deadline - 25 - time.monotonic()
        if remaining < 12:
            break
        alloc = min(35.0, remaining * 0.3)
        if alloc < 10:
            continue

        # Interpolate parent to new length
        ox = np.linspace(0.0, 1.0, n_p)
        nx = np.linspace(0.0, 1.0, L)
        a_L = np.interp(nx, ox, p).clip(0, None)
        sL = float(a_L.sum())
        if sL > 1e-15:
            a_L /= sL
        else:
            a_L = np.ones(L, dtype=np.float64) / L
        ub(a_L)

        a_L = optimize(a_L, L, time.monotonic() + alloc)
        ub(a_L)

    # === Phase 3: Restart with different initializations at parent length ===
    if time.monotonic() < deadline - 35:
        inits = []
        # Mirrored parent
        am = p[::-1].copy()
        inits.append(am)
        # Damped oscillation
        ao = p * np.exp(0.3 * rng.standard_normal(n_p))
        ao = ao.clip(0, None)
        if ao.sum() > 1e-15:
            inits.append(ao / ao.sum())
        # Smoother version
        from scipy.ndimage import uniform_filter1d
        as_ = uniform_filter1d(p, size=7)
        as_ = as_.clip(0, None)
        if as_.sum() > 1e-15:
            inits.append(as_ / as_.sum())
        # Random
        ar = np.abs(rng.standard_normal(n_p)) + 0.05
        inits.append(ar / ar.sum())

        for a_init in inits:
            if time.monotonic() >= deadline - 25:
                break
            remaining = deadline - 25 - time.monotonic()
            if remaining < 12:
                break
            alloc = min(30.0, remaining * 0.4)
            if alloc < 10:
                continue
            a_w = optimize(a_init, n_p, time.monotonic() + alloc)
            ub(a_w)

    # === Phase 4: Final gradient polish on best found ===
    if time.monotonic() < deadline - 8 and best_arr is not None:
        n_b = best_arr.size
        a_pol = best_arr.astype(np.float64).copy()
        a_pol = np.clip(a_pol, 0, None)
        sp_ = float(a_pol.sum())
        if sp_ > 1e-15:
            a_pol /= sp_
        # Short gradient optimization
        local = sc(a_pol)
        for _ in range(5000):
            if time.monotonic() >= deadline - 4:
                break
            direction = smooth_max_direction(a_pol, lam=60.0)
            if direction is None:
                break
            improved = False
            for alpha in [0.0005, 0.001, 0.002, 0.005]:
                cand = a_pol - alpha * direction
                cand = np.clip(cand, 0, None)
                cs = float(cand.sum())
                if cs < 1e-15:
                    continue
                cand /= cs
                v = sc(cand)
                if v < local * (1 - 1e-8):
                    a_pol = cand
                    local = v
                    improved = True
                    break
            if not improved:
                break
        ub(a_pol)

    # === Final validation and return ===
    if best_arr is None:
        best_arr = np.ones(96, dtype=np.float64)
        ub(best_arr)

    final = np.clip(best_arr, 0.0, 1000.0)
    if float(final.sum()) < 0.01:
        final = np.ones_like(final)
        ub(final)
    else:
        ub(final)

    return [float(x) for x in best_arr]
