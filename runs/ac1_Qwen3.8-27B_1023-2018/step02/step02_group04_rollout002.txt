def _grad_conv(a, w):
    a = np.asarray(a, dtype=np.float64).ravel()
    w = np.asarray(w, dtype=np.float64).ravel()
    n = a.size
    need = 2 * n - 1

        w = ww

        return np.zeros(n, dtype=np.float64)
    ar = a[::-1]
    tmp = np.convolve(w, ar)
    return 2.0 * tmp[n - 1:2 * n - 1]

    for j in range(n):

    return A

    s = float(ex.sum())
    if not math.isfinite(s) or s <= 0.0:
        return np.ones(x.size, dtype=np.float64) / x.size
    return ex / s

                if math.isfinite(sc) and sc < bs:
                    bs, bc = sc, cand
            if bc is not a and math.isfinite(bs) and bs < loc * (1.0 - 1e-12):
                a = _norm_vec(bc)
                loc = bs
                n = a.size
                stag = 0
                improved = True
                _eval_update(a, best)

        if not improved:
            if g is None:
                fails += 1
            stag += 1
            if fails > 5 or stag > 12:
                break
            if stag % 2 == 0 and (stop_t - time.monotonic()) > 20.0:
                accepted = False
                for sigma in (0.03, 0.08, 0.15):
                    pa = _norm_vec(a * np.exp(sigma * rng.standard_normal(n)))
                    if pa is None:
                        continue
                    sp = _score_vec(pa)
                    if pa.size % 2 == 0 and pa.size >= 8:
                        ap = _anti_projection(pa)
                        if ap is not None:
                            sp2 = _score_vec(ap)
                            if sp2 < sp:
                                pa, sp = ap, sp2
                    if math.isfinite(sp) and sp < loc:
                        a = pa
                        loc = sp
                        n = a.size
                        stag = 0
                        _eval_update(a, best)
                        accepted = True
                        break
                if not accepted and best[1].size == n and best[0] < loc:
                    a = _norm_vec(best[1])
                    loc = best[0]
    return a


def _smooth_full(a0, stop_t, best, rng):
    a = _norm_vec(a0)
    if a is None or minimize is None:
        return a
    if a.size > 512:
        a = _resize(a, 512)
    n = a.size
    if n < 2:
        return a
    _eval_update(a, best)
    loc = _score_vec(a)
    cur = a.copy()

    cap = 700 if n <= 256 else (520 if n <= 384 else 320)
    bounds = [(-55.0, 55.0)] * n
    taus = (0.4, 0.2, 0.1, 0.05, 0.02, 0.008)

    for tau_rel in taus:
        if time.monotonic() > stop_t - 2.0:
            break
        b = np.convolve(cur, cur)
        Mq = float(n * np.max(b))
        if not math.isfinite(Mq) or Mq <= 1e-14:
            break
        tau = max(1e-10, tau_rel * Mq)

        for call in range(2):
            mc = time.monotonic()
            if mc > stop_t - 3.0:
                break
            mf = int(max(100, min(cap, max(0.0, stop_t - mc) * 20)))
            if mf < 80:
                break

            x0 = np.log(np.clip(cur, 1e-15, None))
            if call:
                x0 = x0 + 0.04 * rng.standard_normal(n)
            x0 = np.clip(x0, -54.0, 54.0)

            def fun(x, tau=tau, n=n):
                xm = x - np.max(x)
                ex = np.exp(np.clip(xm, -70.0, 70.0))
                exs = float(ex.sum())
                if exs <= 0.0 or not math.isfinite(exs):
                    return 1e100, np.zeros(n, dtype=np.float64)
                p = ex / exs
                b2 = np.convolve(p, p)
                q = n * b2
                sh = float(np.max(q))
                if not math.isfinite(sh):
                    return 1e100, np.zeros(n, dtype=np.float64)
                z = (q - sh) / tau
                m2 = float(np.max(z))
                ez = np.exp(z - m2)
                ew = float(ez.sum())
                if ew <= 0.0 or not math.isfinite(ew):
                    return 1e100, np.zeros(n, dtype=np.float64)
                wv = ez / ew
                s = n * _grad_conv(p, wv)
                val = sh + tau * (m2 + math.log(ew))
                grad = p * (s - float(np.dot(p, s)))
                if not np.all(np.isfinite(grad)):
                    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
                return float(val), grad

            try:
                res = minimize(fun, x0, jac=True, method="L-BFGS-B",
                               bounds=bounds,
                               options={"maxiter": mf, "maxfun": mf,
                                        "ftol": 1e-14, "gtol": 1e-7})
                xres = getattr(res, "x", None)
                if xres is None:
                    continue
                cand = _norm_vec(_softmax(xres))
                if cand is None:
                    continue
                sc = _score_vec(cand)
                if math.isfinite(sc) and sc < loc - 1e-13:
                    cur = cand
                    loc = sc
                    n = cur.size
                    _eval_update(cur, best)
            except Exception:
                pass

    return cur


def _smooth_anti(a0, stop_t, best, rng):
    a = _norm_vec(a0)
    if a is None or minimize is None:
        return a
    n = a.size
    if n % 2 != 0 or n < 8 or n > 512:
        return a
    h = n // 2
    c = 1.0 / float(n)

    curv = _anti_v_from_array(a, n)
    cur = _norm_vec(_make_anti_v(n, curv))
    if cur is None:
        return a
    _eval_update(cur, best)
    loc = _score_vec(cur)

    cap = 950 if n <= 256 else (700 if n <= 384 else 420)
    bounds = [(-1.0, 1.0)] * h
    taus = (0.4, 0.2, 0.1, 0.05, 0.02, 0.008)

    for tau_rel in taus:
        if time.monotonic() > stop_t - 2.0:
            break
        aa0 = _make_anti_v(n, curv)
        b = np.convolve(aa0, aa0)
        Mq = float(n * np.max(b))
        if not math.isfinite(Mq) or Mq <= 1e-14:
            break
        tau = max(1e-10, tau_rel * Mq)

        for call in range(2):
            mc = time.monotonic()
            if mc > stop_t - 3.0:
                break
            mf = int(max(120, min(cap, max(0.0, stop_t - mc) * 22)))
            if mf < 100:
                break

            v0 = curv.copy()
            if call:
                v0 = v0 + 0.05 * rng.standard_normal(h)
            v0 = np.clip(v0, -1.0, 1.0)

            def fun(vv, tau=tau, n=n, h=h, c=c):
                vv = np.clip(vv, -1.0, 1.0)
                aa = _make_anti_v(n, vv)
                b2 = np.convolve(aa, aa)
                q = n * b2
                sh = float(np.max(q))
                if not math.isfinite(sh):
                    return 1e100, np.zeros(h, dtype=np.float64)
                z = (q - sh) / tau
                m2 = float(np.max(z))
                ez = np.exp(z - m2)
                ew = float(ez.sum())
                if ew <= 0.0 or not math.isfinite(ew):
                    return 1e100, np.zeros(h, dtype=np.float64)
                wv = ez / ew
                s = n * _grad_conv(aa, wv)
                val = sh + tau * (m2 + math.log(ew))
                gv = c * (s[:h] - s[::-1][:h])
                if not np.all(np.isfinite(gv)):
                    gv = np.nan_to_num(gv, nan=0.0, posinf=0.0, neginf=0.0)
                return float(val), gv

            try:
                res = minimize(fun, v0, jac=True, method="L-BFGS-B",
                               bounds=bounds,
                               options={"maxiter": mf, "maxfun": mf,
                                        "ftol": 1e-14, "gtol": 1e-7})
                vres = getattr(res, "x", None)
                if vres is None:
                    continue
                vnew = np.clip(np.asarray(vres, dtype=np.float64).ravel()[:h], -1.0, 1.0)
                cand = _norm_vec(_make_anti_v(n, vnew))
                if cand is None:
                    continue
                sc = _score_vec(cand)
                if math.isfinite(sc) and sc < loc - 1e-13:
                    curv = vnew
                    cur = cand
                    loc = sc
                    _eval_update(cur, best)
            except Exception:
                pass

    return cur


def _full_starts(n, par, best_arr, rng, count=12):
    n = int(n)
    if n < 2:
        n = 2
    S = []

    def add(a):
        a = _norm_vec(a)
        if a is None:
            return
        try:
            sc = _score_vec(a)
        except Exception:
            return
        if math.isfinite(sc):
            S.append((sc, a))

    x = np.linspace(0.0, 1.0, n)
    add(np.ones(n, dtype=np.float64))

    if par is not None and par.size >= 2:
        add(_resize(par, n))
        if n % 2 == 0 and n >= 8:
            add(_anti_projection(_resize(par, n)))
    if best_arr is not None and len(best_arr) >= 2:
        add(_resize(best_arr, n))
        if n % 2 == 0 and n >= 8:
            add(_anti_projection(_resize(best_arr, n)))

    for cc in (-3.0, -1.5, -0.5, 0.5, 1.5, 3.0):
        add(np.exp(cc * (x - 0.5)))

    for m in range(1, 5):
        for amp in (0.2, 0.5, 0.8):
            add(1.0 + amp * np.sin(2.0 * np.pi * m * x))
            add(1.0 + amp * np.cos(2.0 * np.pi * m * x))
            add(np.abs(np.sin(np.pi * m * x)))

    for al in (0.3, 1.0, 5.0, 20.0):
        try:
            add(rng.dirichlet(np.full(n, al)))
        except Exception:
            pass

    for _ in range(6):
        y = rng.standard_normal(n)
        kern = max(3, n // 16)
        y = np.convolve(y, np.ones(kern) / kern, mode="same")
        y = np.exp(np.clip(0.5 * y, -5.0, 5.0))
        add(y)

    if n % 2 == 0 and n >= 8:
        h = n // 2
        r = (np.arange(h) + 0.5) / float(h)

        def add_v(v):
            add(_make_anti_v(n, v))

        if par is not None and par.size >= 2:
            add_v(_anti_v_from_array(_resize(par, n), n))
        if best_arr is not None and len(best_arr) >= 2:
            add_v(_anti_v_from_array(_resize(best_arr, n), n))

        add_v(np.zeros(h, dtype=np.float64))

        amps = (0.25, 0.45, 0.65, 0.85)
        ms = (1, 2, 3, 4)
        for amp in amps:
            for m in ms:
                base = np.sin(2.0 * np.pi * m * (r - 0.5))
                add_v(amp * base)
                add_v(-amp * base)
                base2 = np.cos(2.0 * np.pi * m * (r - 0.5))
                add_v(amp * base2)
                add_v(-amp * base2)

        betas = (1.5, 2.5, 4.0, 7.0, 11.0, 18.0)
        for beta in betas:
            base = np.tanh(beta * (r - 0.5))
            for amp in amps:
                add_v(amp * base)
                add_v(-amp * base)

        for _ in range(8):
            y = rng.standard_normal(h)
            kern = max(3, h // 16)
            y = np.convolve(y, np.ones(kern) / kern, mode="same")
            nm = float(np.max(np.abs(y)))
            if math.isfinite(nm) and nm > 0:
                add_v(0.9 * y / nm)

    S.sort(key=lambda z: z[0])
    out = [a for _, a in S[:max(1, int(count))]]
    if not out:
        out = [np.ones(n, dtype=np.float64)]
    return out


def _improve_one(a0, stop_t, best, rng):
    a = _norm_vec(a0)
    if a is None:
        return a
    if a.size > 512:
        a = _resize(a, 512)
    n = a.size
    _eval_update(a, best)
    loc = _score_vec(a)
    no_improve_cycles = 0

    while True:
        now = time.monotonic()
        rem = stop_t - now
        if rem < 25.0:
            break

        improved_this_cycle = False

        if n % 2 == 0 and n >= 8 and rem > 55.0:
            cand = _smooth_anti(a, min(stop_t, now + min(45.0, rem - 25.0)), best, rng)
            if cand is not None:
                sc = _score_vec(cand)
                if math.isfinite(sc) and sc < loc - 1e-12:
                    a = _norm_vec(cand)
                    loc = sc
                    n = a.size
                    improved_this_cycle = True
        _eval_update(a, best)
        if best[1].size == n and best[0] < loc - 1e-12:
            a = _norm_vec(best[1])
            loc = best[0]
            n = a.size
            improved_this_cycle = True

        if time.monotonic() > stop_t - 35.0:
            break
        cand = _smooth_full(a, min(stop_t, time.monotonic() + min(45.0, stop_t - time.monotonic() - 25.0)), best, rng)
        if cand is not None:
            sc = _score_vec(cand)
            if math.isfinite(sc) and sc < loc - 1e-12:
                a = _norm_vec(cand)
                loc = sc
                n = a.size
                improved_this_cycle = True
        _eval_update(a, best)
        if best[1].size == n and best[0] < loc - 1e-12:
            a = _norm_vec(best[1])
            loc = best[0]
            n = a.size
            improved_this_cycle = True

        if time.monotonic() > stop_t - 40.0:
            break
        cand = _lp_polish(a, stop_t, best, rng)
        if cand is not None:
            sc = _score_vec(cand)
            if math.isfinite(sc) and sc < loc - 1e-12:
                a = _norm_vec(cand)
                loc = sc
                n = a.size
                improved_this_cycle = True
        _eval_update(a, best)
        if best[1].size == n and best[0] < loc - 1e-12:
            a = _norm_vec(best[1])
            loc = best[0]
            n = a.size
            improved_this_cycle = True

        if not improved_this_cycle:
            no_improve_cycles += 1
            if no_improve_cycles >= 4:
                break
        else:
            no_improve_cycles = 0

    return a


def _fallback_from_parent():
    best = [float("inf"), np.array([1.0, 1.0], dtype=np.float64)]
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            _eval_update(h, best)
    except Exception:
        pass
    fin = _norm_vec(best[1])
    if fin is None:
        fin = np.array([1.0], dtype=np.float64)
    return [float(x) for x in fin]


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        B = float(budget_s)
    except Exception:
        B = 1300.0
    if not math.isfinite(B) or B <= 0.0:
        B = 1300.0
    if B < 60.0:
        return _fallback_from_parent()

    deadline = t0 + max(0.0, B) - 40.0
    rng = np.random.default_rng(seed)

    best = [float("inf"), np.array([1.0, 1.0], dtype=np.float64)]
    _eval_update(best[1], best)

    par = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = _clean_vec(h)
            if p.size >= 1:
                _eval_update(p, best)
                par = p
    except Exception:
        par = None

    if par is None or par.size < 2:
        par = best[1].copy()

    if par.size > 512:
        par_search = _resize(par, 512)
    else:
        par_search = par
    n_main = par_search.size if par_search.size >= 8 else 192
    n_main = int(max(24, min(512, n_main)))

    # Main phase: high-quality starts at the parent's main length.
    if time.monotonic() < deadline - 220.0:
        main_budget = min(900.0, max(0.0, (deadline - 220.0 - time.monotonic()) * 0.80))
        if main_budget > 80.0:
            starts = _full_starts(n_main, par_search, best[1], rng, count=6)
            main_end = time.monotonic() + main_budget
            allocs = (0.55, 0.25, 0.15, 0.05)
            for i, s in enumerate(starts[:4]):
                if time.monotonic() >= main_end - 35.0:
                    break
                frac = allocs[i] if i < len(allocs) else 0.05
                alloc = min(frac * main_budget, main_end - time.monotonic() - 5.0, 550.0)
                if alloc < 35.0:
                    break
                _improve_one(s, time.monotonic() + alloc, best, rng)

    # Secondary phases: other genuine resolutions from compact templates.
    sec_end = deadline - 140.0
    if time.monotonic() < sec_end - 70.0:
        sec_total = min(320.0, max(0.0, sec_end - time.monotonic() - 30.0))
        if sec_total > 90.0:
            lengths = []
            for ns in (384, 256, 128, 96, 64):
                if 24 <= ns <= 512 and ns != n_main:
                    lengths.append(ns)
            fracs = (0.32, 0.30, 0.20, 0.10, 0.10)
            for i, ns in enumerate(lengths[:4]):
                if time.monotonic() >= sec_end - 35.0:
                    break
                frac = fracs[i] if i < len(fracs) else 0.05
                alloc = min(frac * sec_total, sec_end - time.monotonic() - 25.0, 150.0)
                if alloc < 40.0:
                    continue
                starts = _full_starts(ns, par_search, best[1], rng, count=3)
                if starts:
                    _improve_one(starts[0], time.monotonic() + alloc, best, rng)
                if time.monotonic() >= sec_end - 35.0:
                    break

    # Final polish on the current global best length.
    final_end = deadline - 35.0
    if time.monotonic() < final_end - 35.0:
        _improve_one(best[1], final_end, best, rng)

    # Small final cleanup: try zeroing numerically negligible entries.
    fin = _norm_vec(best[1])
    if fin is None:
        fin = np.array([1.0], dtype=np.float64)
    for th in (1e-14, 1e-12):
        f2 = fin.copy()
        f2[f2 < th] = 0.0
        f2 = _norm_vec(f2)
        if f2 is not None:
            _eval_update(f2, best)
    fin = _norm_vec(best[1])
    if fin is None:
        fin = np.array([1.0], dtype=np.float64)

    v = _exact_score([float(x) for x in fin])
    if not math.isfinite(v):
        fin = np.array([1.0], dtype=np.float64)
        v = _exact_score([float(x) for x in fin])
    if not math.isfinite(v):
        return [1.0]
    return [float(x) for x in fin]
