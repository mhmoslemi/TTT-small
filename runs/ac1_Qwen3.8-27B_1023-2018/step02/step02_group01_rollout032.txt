    j = k_idx - i_idx
    mask = (j >= 0) & (j < n)
    safe_j = np.clip(j, 0, n - 1)

    if rng is None:
        rng = np.random.default_rng(n * 7 + 13)

    stuck = 0
    perturb_count = 0
    for it in range(max_iters):
        if time.monotonic() > stop_time - 2.0:
            break

        # Compute current autocorrelation
        Fa = rfft(a, fsize)
        b = irfft(Fa * Fa, fsize)[:2 * n - 1]
        M = float(b.max())

        if M < best_M - 1e-15:
            best_M = M
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        if stuck > 300:
            # Escalating perturbation
            perturb_count += 1
            sigma = min(0.1 * perturb_count, 0.5)
            a = a * (1.0 + rng.normal(0, sigma, n))
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            stuck = 0
            continue

        # Build constraint matrix A where A[k,i] = a[k-i] if valid
        A = np.where(mask, a[safe_j], 0.0)

        # Solve LP: max sum(g) s.t. A@g <= M, g >= 0
        try:
            res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2 * n - 1, M),
                          bounds=[(0.0, None)] * n, method='highs',
                          options={'time_limit': 1.0})
        except Exception:
            break

        if res.status != 0 or res.x is None:
            break
        g = np.asarray(res.x, dtype=np.float64)
        gs = float(g.sum())
        if gs <= 1e-12 or not np.isfinite(g).all():
            break
        g = g / gs

        # Check if g is essentially a (converged)
        if float(np.abs(g - a).max()) < 1e-12:
            break

        # Line search: minimize max_k [(a+t(g-a))*(a+t(g-a))][k] over t in (0,1]
        d = g - a
        Fd = rfft(d, fsize)
        cross = irfft(Fa * Fd, fsize)[:2 * n - 1]  # a*d
        dd = irfft(Fd * Fd, fsize)[:2 * n - 1]  # d*d

        # F(t) = max_k [b[k] + 2t*cross[k] + t^2*dd[k]]
        best_t, best_val = 0.0, M
        # Coarse then fine grid
        for t in np.linspace(0.02, 1.0, 30):
            val = float((b + 2.0 * t * cross + t * t * dd).max())
            if val < best_val:
                best_val = val
                best_t = t

        if best_t <= 0 or best_val >= M - 1e-14:
            # Try very small steps
            for t in [0.005, 0.002, 0.001, 0.0005]:
                val = float((b + 2.0 * t * cross + t * t * dd).max())
                if val < M - 1e-14:
                    best_t = t
                    break
            if best_t <= 0:
                break

        a = a + best_t * d
        a = np.maximum(a, 0.0)
        a = a / a.sum()

    return best_a

def _smoothed_max_gradient(a, beta=500.0):
    """Compute gradient of smoothed max of a*a using soft-max."""
    n = a.size
    b = _conv(a)
    # Soft weights
    b_shifted = b - b.max()
    w = np.exp(beta * b_shifted)
    w = w / w.sum()
    # Gradient of w @ b with respect to a: sum_k w[k] * 2 * (R_k a)
    # (R_k a)[i] = a[k-i] if valid
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                grad[i] += 2.0 * w[k] * a[j]
    return grad

def _projected_gradient(a0, stop_time, max_iters=50000, beta=200.0, lr=None):
    """Projected gradient descent on smoothed objective."""
    n = a0.size
    a = np.clip(a0.astype(np.float64).copy(), 0.0, 1000.0)
    s = a.sum()
    if s <= 0:
        a = np.ones(n) / n
        return a
    a = a / s

    best_a = a.copy()
    best_M = float(_conv(a).max())

    if lr is None:
        lr = 1.0 / (beta * n) * 2.0

    for it in range(max_iters):
        if time.monotonic() > stop_time - 2.0:
            break

        b = _conv(a)
        M = float(b.max())
        if M < best_M - 1e-14:
            best_M = M
            best_a = a.copy()

        grad = _smoothed_max_gradient(a, beta)
        a = a - lr * grad
        a = np.maximum(a, 0.0)
        a = a / a.sum()

        # Decay learning rate
        lr *= 0.999

    return best_a

def _generate_initializations(parent, n, seed):
    """Generate diverse initializations."""
    rng = np.random.default_rng(seed)
    inits = []

    # Parent
    p = np.clip(parent.astype(np.float64), 0.0, 1000.0)
    if p.size == n and p.sum() > 0:
        inits.append(p / p.sum())

    x = np.arange(n, dtype=np.float64)
    xn = x / max(n - 1, 1)

    # Cosine windows
    for phase in [0.0, 0.25, 0.5]:
        c = np.maximum(np.cos(np.pi * (xn + phase)), 0.01)
        inits.append(c / c.sum())

    # Linear ramps
    ramp = np.linspace(0.05, 1.0, n)
    inits.append(ramp / ramp.sum())
    inits.append(ramp[::-1] / ramp.sum())

    # Triangular
    tri = np.minimum(xn, 1 - xn)
    tri = np.maximum(tri, 0.01)
    inits.append(tri / tri.sum())

    # Gaussians
    for center in [0.25, 0.35, 0.5, 0.65, 0.75]:
        for width in [0.08, 0.15, 0.25, 0.4]:
            g = np.exp(-0.5 * ((xn - center) / width) ** 2)
            inits.append(g / g.sum())

    # Exponentials
    for rate in [0.5, 1.0, 2.0, 4.0, 8.0]:
        e = np.exp(-rate * xn)
        inits.append(e / e.sum())
        e2 = np.exp(-rate * (1 - xn))
        inits.append(e2 / e2.sum())

    # Power laws
    for p in [0.3, 0.5, 1.0, 2.0, 4.0]:
        q = xn ** p
        inits.append(q / q.sum())
        q2 = (1 - xn) ** p
        inits.append(q2 / q2.sum())

    # Sines
    for freq in [0.5, 1.0, 1.5, 2.0, 3.0]:
        s = np.maximum(0.5 + 0.5 * np.sin(2 * np.pi * freq * xn), 0.01)
        inits.append(s / s.sum())
        s2 = np.maximum(0.5 + 0.5 * np.cos(2 * np.pi * freq * xn), 0.01)
        inits.append(s2 / s2.sum())

    # Bimodal
    for split in [0.3, 0.4, 0.5]:
        for ratio in [2.0, 5.0, 10.0]:
            two = np.ones(n)
            two[x < int(split * n)] = ratio
            inits.append(two / two.sum())
            two2 = np.ones(n)
            two2[x >= int(split * n)] = ratio
            inits.append(two2 / two2.sum())

    # Random smooth
    for smoothing in [3, 7, 15]:
        r = rng.normal(0, 1, n)
        k = np.ones(smoothing) / smoothing
        r = np.convolve(r, k, mode='same')
        r = np.maximum(r, 0.01)
        inits.append(r / r.sum())

    # Parent with noise
    if p.size == n and p.sum() > 0:
        pn = p / p.sum()
        for lvl in [0.1, 0.3, 0.5, 0.8]:
            noisy = pn * np.exp(rng.normal(0, lvl, n))
            noisy = np.maximum(noisy, 1e-8)
            inits.append(noisy / noisy.sum())

    return inits

def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = float(budget_s)
    deadline = t0 + budget - 15.0

    parent = _get_parent()
    rng = np.random.default_rng(seed)

    # Initialize best with parent
    best_seq = parent.copy()
    best_score = float(_score(best_seq))
    if not np.isfinite(best_score):
        best_seq = np.ones(64, dtype=np.float64)
        best_score = 2.0

    n = parent.size

    # Phase 1: FW-LP from parent (most promising start)
    p1_end = min(t0 + budget * 0.20, deadline - 200.0)
    if time.monotonic() < p1_end:
        result = _fw_lp(parent, stop_time=p1_end, max_iters=50000, rng=rng)
        sc = _score(result)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = result.copy()

    # Phase 2: Diverse initializations with FW-LP
    inits = _generate_initializations(parent, n, seed)

    p2_end = min(t0 + budget * 0.60, deadline - 100.0)
    per_init = max(2.0, (p2_end - t0) / len(inits))
    for idx, init in enumerate(inits):
        if time.monotonic() > p2_end - 5.0:
            break
        alloc_end = min(t0 + (idx + 1) * per_init + 2.0, p2_end)
        if time.monotonic() > alloc_end - 1.0:
            continue
        try:
            result = _fw_lp(init, stop_time=alloc_end, max_iters=5000,
                           rng=np.random.default_rng(seed * 100 + idx))
            sc = _score(result)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = result.copy()
        except Exception:
            pass

    # Phase 3: Polish best with extended FW-LP
    p3_end = min(t0 + budget * 0.82, deadline - 50.0)
    if time.monotonic() < p3_end - 10:
        try:
            result = _fw_lp(best_seq, stop_time=p3_end, max_iters=100000, rng=rng)
            sc = _score(result)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = result.copy()
        except Exception:
            pass

    # Phase 4: Smoothed gradient polish
    if time.monotonic() < deadline - 30:
        try:
            p4_end = min(time.monotonic() + 30, deadline - 20)
            result = _projected_gradient(best_seq, stop_time=p4_end,
                                         max_iters=20000, beta=500.0)
            sc = _score(result)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = result.copy()
        except Exception:
            pass

    # Phase 5: Try different lengths
    if time.monotonic() < deadline - 25:
        n_candidates = [n - 4, n - 2, n - 1, n + 1, n + 2, n + 4,
                        n // 2, n // 2 + 1, n // 2 - 1,
                        int(n * 1.25), int(n * 0.75), 128, 129, 130, 160, 161]
        seen = set()
        for n_try in n_candidates:
            if time.monotonic() > deadline - 15:
                break
            if n_try < 8 or n_try > 600 or n_try in seen:
                continue
            seen.add(n_try)
            # Resample parent to new length
            idxs = np.clip((np.arange(n_try) * n // max(n_try, 1)).astype(int), 0, n - 1)
            init_new = np.maximum(parent[idxs].astype(np.float64), 1e-8)
            init_new = init_new / init_new.sum()

            alloc = min(20.0, deadline - 12 - time.monotonic())
            if alloc < 4.0:
                break
            try:
                result = _fw_lp(init_new, stop_time=time.monotonic() + alloc,
                                max_iters=5000, rng=np.random.default_rng(seed + n_try))
                sc = _score(result)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = result.copy()
            except Exception:
                pass

    # Final FW-LP polish on the overall best
    if time.monotonic() < deadline - 8:
        try:
            result = _fw_lp(best_seq, stop_time=deadline - 5, max_iters=20000, rng=rng)
            sc = _score(result)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = result.copy()
        except Exception:
            pass

    # Final validation with exact evaluator
    try:
        final_score = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_score = float('inf')

    if not np.isfinite(final_score):
        final_score = _score(best_seq)

    if not np.isfinite(final_score) or best_seq.size == 0:
        return [1.0] * 32

    return [float(x) for x in np.clip(best_seq, 0.0, 1000.0)]
