def _local_evaluate(seq):
    """Fallback mirroring the stated evaluator if evaluate_sequence is absent."""
    if not seq:
        return float("inf")
    try:
        seq = [float(x) for x in seq]
    except Exception:
        return float("inf")

    # The supplied validator enforces finite, non-bool numeric entries.
    if any(isinstance(x, bool) or not math.isfinite(x) for x in seq):
        return float("inf")

def _lp_solve(a, M, n, time_limit):
    """
    Solve
        max sum(g)
        s.t. a * g <= M, g >= 0
    where * is full linear convolution.

    Used only if the live evaluator reports a score below the analytic lower
    bound of the stated evaluator, which would indicate a nonstandard live score.
    """
    if linprog is None or sps is None:
        return None
    if n < 2 or n > LP_MAX_N:
        return None
    M = float(M)
    if not math.isfinite(M) or M <= 1e-30:
        return None
    if time_limit < 1.0:
        return None

    try:
        col_idx = np.repeat(np.arange(n, dtype=np.int64), n)
        row_idx = col_idx + np.tile(np.arange(n, dtype=np.int64), n)
        vals = np.tile(a, n).astype(np.float64, copy=False)
        A = sps.csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))

        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M + 1e-12, dtype=np.float64)

        res = linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=[(0.0, None)] * n,
            method="highs",
            options={"presolve": True, "time_limit": max(1.0, float(time_limit))},
        )
    except Exception:
        return None

    x = getattr(res, "x", None)

        return None

    try:
        g = np.asarray(x, dtype=np.float64).ravel()
    except Exception:
        return None

    return None


def _autocorr_gradient_peak(a, rel_eps=0.05):
    """
    Subgradient of a smoothed max of the autocorrelation peak of a * a.

    Only nonnegative lags 0..n-1 are used; the full autocorrelation is symmetric.
    """
    n = int(a.size)
    if n < 1:
        return a.copy(), 0.0

    b = np.convolve(a, a)
    if b.size < n:
        b = np.pad(b, (0, n - b.size), mode="constant")
    b_pos = b[:n]
    M = float(b_pos.max())
    if not math.isfinite(M) or M <= 1e-30:
        return np.zeros(n), M

    active = np.where(b_pos >= M - rel_eps * M)[0]
    if active.size == 0:
        grad = np.zeros(n, dtype=np.float64)
        return grad, M

    grad = np.zeros(n, dtype=np.float64)
    for k in active:
        d = int(k)
        if d == 0:
            grad += 2.0 * a
        else:
            if d < n:
                grad[: n - d] += a[d:]
                grad[d:] += a[: n - d]

    grad /= float(active.size)
    return grad, M
