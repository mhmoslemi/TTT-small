                    best_cand = cand

            if best_cand is not None and best_sc < local * (1.0 - 1e-12):
                a = best_cand
                local = best_sc
                update_best(a)
                stall = 0
            else:
                stall += 1
                if stall > 60:
                    # Perturbation to escape
                    pert = a * (1.0 + 0.05 * rng.standard_normal(n))
                    pert = np.clip(pert, 0, None)
                    ps = float(pert.sum())
                    if ps > 1e-15:
                        pert = pert / ps
                        if fscore(pert) < local:
                            a = pert
                            local = fscore(pert)
                            update_best(a)
                            stall = 0
                    if stall > 120:
                        break

        return a

    # Softmax-smoothed gradient descent
    def soft_grad_opt(a, stop_time, max_iters=30000):
        nonlocal best_score, best_arr
        n = len(a)
        a = np.array(a, dtype=np.float64).ravel().copy()
        a = np.clip(a, 0, None)
        s = float(a.sum())
        if s < 1e-15:
            return a
        a = a / s
        local = fscore(a)
        update_best(a)

        tau = 0.1  # softness parameter
        lr = 0.05
        stall = 0

        for it in range(max_iters):
            if time.monotonic() >= stop_time:
                break

            b = np.convolve(a, a)
            S = float(a.sum())

            # Softmax weights over convolution values
            z = b / tau
            z -= z.max()
            exp_z = np.exp(z)
            p = exp_z / exp_z.sum()

            # Gradient of soft-max objective w.r.t. a_i
            grad = np.zeros(n, dtype=np.float64)
            for k in range(2 * n - 1):
                pk = p[k]
                if pk < 1e-10:
                    continue
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                js = np.arange(lo, hi + 1)
                grad[js] += 2.0 * pk * a[k - js]

            # Normalize gradient
            gn = float(np.linalg.norm(grad))
            if gn < 1e-14:
                break

            # Projected gradient step (on the simplex a>=0, sum=1)
            best_sc = local
            best_cand = None
            for alpha in (lr * 0.5, lr, lr * 2.0):
                cand = a - alpha * grad / gn
                cand = np.clip(cand, 0, None)
                cs = float(cand.sum())
                if cs < 1e-15:
                    continue
                cand = cand / cs
                sc = fscore(cand)
                if sc < best_sc:
                    best_sc = sc
                    best_cand = cand

            if best_cand is not None and best_sc < local * (1.0 - 1e-10):
                a = best_cand
                local = best_sc
                update_best(a)
                stall = 0
                # Decay learning rate slowly
                if stall == 0 and it % 500 == 0:
                    lr *= 0.95
            else:
                stall += 1
                lr *= 0.9
                if it % 200 == 0 and stall > 100:
                    pert = a + 0.02 * rng.standard_normal(n)
                    pert = np.clip(pert, 0, None)
                    ps = float(pert.sum())
                    if ps > 1e-15:
                        pert = pert / ps
                        if fscore(pert) < local:
                            a = pert
                            local = fscore(pert)
                            update_best(a)
                            stall = 0
                if stall > 500:
                    break

            # Periodically adapt tau
            if it % 1000 == 0:
                tau = max(0.01, tau * 0.9)

        return a

    # Interpolate to different lengths
    def interp_len(src, n_target):
        if src.size == n_target:
            return src.copy()
        ox = np.linspace(0.0, 1.0, src.size)
        nx = np.linspace(0.0, 1.0, n_target)
        out = np.interp(nx, ox, src)
        out = np.clip(out, 0, None)
        s = float(out.sum())
        if s > 1e-15:
            out = out / s
        else:
            out = np.ones(n_target, dtype=np.float64) / n_target
        return out

    # === MAIN OPTIMIZATION ===
    p = parent if parent is not None else best_arr
    n_p = p.size
    p_norm = p / p.sum()

    # Phase 1: Heavy LP at parent length
    if n_p >= 4 and linprog is not None and sps is not None:
        t1 = min(600.0, (deadline - time.monotonic()) * 0.6)
        if t1 > 30:
            a1 = lp_optimize(p_norm, time.monotonic() + t1, max_iters=50000)
            update_best(a1)

    # Phase 2: Soft-gradient polish at parent length
    if time.monotonic() < deadline - 30:
        a2 = soft_grad_opt(a1, deadline - 20, max_iters=20000)
        update_best(a2)

    # Phase 3: Second LP pass (with current best as start)
    if time.monotonic() < deadline - 25 and linprog is not None and sps is not None:
        t3 = min(200.0, (deadline - time.monotonic()) - 20)
        if t3 > 15:
            a3 = lp_optimize(a2, time.monotonic() + t3, max_iters=20000)
            update_best(a3)

    # Phase 4: Try nearby lengths
    nearby_lengths = []
    for L in [n_p - 8, n_p - 4, n_p - 2, n_p + 2, n_p + 4, n_p + 8,
              256, 288, 300, 340, 360, 384, 400]:
        if L >= 8 and L != n_p and L <= 512:
            nearby_lengths.append(L)
    nearby_lengths = sorted(set(nearby_lengths))

    for L in nearby_lengths:
        if time.monotonic() >= deadline - 25:
            break
        t_rem = deadline - time.monotonic() - 20
        if t_rem < 10:
            break
        alloc = min(30.0, t_rem * 0.15)
        if alloc < 5:
            continue

        a_init = interp_len(best_arr, L)

        if linprog is not None and sps is not None:
            a_w = lp_optimize(a_init, time.monotonic() + alloc * 0.6, max_iters=2000)
            update_best(a_w)

        if time.monotonic() < deadline - 15:
            a_w = soft_grad_opt(a_w, deadline - 10, max_iters=3000)
            update_best(a_w)

    # Phase 5: Final gradient polish on best
    if time.monotonic() < deadline - 10 and best_arr is not None:
        a_pol = best_arr.copy() / best_arr.sum()
        a_pol = soft_grad_opt(a_pol, deadline - 5, max_iters=10000)
        update_best(a_pol)

    # Final validation
    if best_arr is None:
        best_arr = np.ones(320, dtype=np.float64)
        update_best(best_arr)

    final = np.clip(best_arr, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones_like(final)
    update_best(final)

    return [float(x) for x in best_arr]
