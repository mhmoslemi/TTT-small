def _build_A(f, n):
    """Build the convolution matrix A where (A @ g)[k] = (f*g)[k]."""
    # Using sparse format
    col_idx = np.repeat(np.arange(n, dtype=np.int64), n)
    row_idx = col_idx + np.tile(np.arange(n, dtype=np.int64), n)
    # Clip to valid range
    mask = (row_idx >= 0) & (row_idx < 2*n-1)
    col_idx = col_idx[mask]
    row_idx = row_idx[mask]
    vals = np.tile(f, n)[mask].astype(np.float64)
    A = sps.csr_matrix((vals, (row_idx, col_idx)), shape=(2*n-1, n))
    return A

def _subgrad_step(a, n, alpha, rng):

    M = float(b.max())
    if M < 1e-30:
        return a, M
    
    # Active set
    tol = 0.1 * M
    active = np.where(b >= M - tol)[0]
    
    # Compute subgradient
    grad = np.zeros(n)
    for k in active:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi+1] += 2.0 * a[k-hi:k-lo]  # a[k-j] for j in [lo, hi]
    
    # Wait, let me redo this. For lag k:
    # ∂(a*a)[k]/∂a_j = 2*a[k-j] if 0 <= k-j < n
    # j ranges from max(0, k-n+1) to min(n-1, k)
    # So grad[j] += 2*a[k-j] for j in [max(0,k-n+1), min(n-1,k)]
    
    grad = np.zeros(n)
    for k in active:

        lo = max(0, k - n + 1)
        hi = min(n - 1, k)

            grad[j] += 2.0 * a[k-j]
    
    grad /= len(active)
    
    # Project out the (1,...,1) direction (to maintain constant sum)
    mean_g = grad.mean()
    grad -= mean_g
    
    # Step
    cand = a - alpha * grad
    cand = np.clip(cand, 1e-10, None)
    cand = cand / cand.sum()
    
    return cand, M

def _score(a):

    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.clip(a, 0.0, 1000.0)

def _build_lp_data(n):
    """Precompute sparse matrix indices for convolution operator."""
    col_idx = np.repeat(np.arange(n, dtype=np.int64), n)
    row_offset = np.tile(np.arange(n, dtype=np.int64), n)
    row_idx = col_idx + row_offset
    mask = (row_idx >= 0) & (row_idx < 2 * n - 1)
    return col_idx[mask], row_idx[mask]


def _lp_solve(f, M, n, col_idx, row_idx, time_limit=10.0):
    """Solve max sum(g) s.t. f*g <= M, 0 <= g <= 10."""
    vals = np.zeros(2 * n - 1, dtype=np.float64)
    np.add.at(vals, row_idx, np.tile(f, n)[0:len(vals)//n * n + n if len(vals) < n*(2*n-1)//n + n else len(vals)://1//n*n + n])
    # Actually let me just build it directly
    # For column j, rows j..j+n-1 (clamped), values are f[j] repeated
    # Total entries = len(col_idx)
    vals = np.full(len(col_idx), 0.0, dtype=np.float64)
    # Each entry in column j has value f[j]
    vals = np.tile(f, n)[:len(col_idx)] if len(col_idx) <= n * n else np.repeat(f, n)[:len(col_idx)]
    # vals[j-th column entries] = f[j]. Since col_idx is [0]*n + [1]*n + ... , 
    # the value for position p in the array is f[col_idx[p]]
    vals = f[col_idx]
    A = sps.csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M * 1.0000001 + 1e-12)
    try:

                      bounds=[(0.0, 10.0)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
    except Exception:
        return None
    if res is not None and getattr(res, 'x', None) is not None:
        g = np.asarray(res.x, dtype=np.float64).ravel()
        g = np.clip(g, 0.0, None)
        sg = float(g.sum())
        if sg > 1e-12 and math.isfinite(sg):

    return None

def _gradient_descent(a_init, n_target, time_limit, rng, best):

            a = np.ones(n)

    a = a / a.sum()
    local = _score(a)

    stale = 0

        if local < TARGET:
            break
        b = np.convolve(a, a)

        # Subgradient of max_k (a*a)[k]

            k = int(k)
            lo = max(0, k - n + 1)

            idx = np.arange(lo, hi + 1)
            grad[idx] += 2.0 * a[k - idx]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        step = alpha / gnorm * (np.std(a) + 1e-6)
        improved = False
        for mult in (1.0, 0.5, 0.25, 2.0):

            cand = np.clip(cand, 1e-15, None)

                local = sc
                stale = 0
                alpha = min(alpha * 1.03, 10.0 / np.sqrt(n))
