import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, next_fast_len

def _conv(a):
    """Compute linear convolution a*a."""
    n = a.size
    L = next_fast_len(2*n-1)
    A = rfft(a, L)
    c = irfft(A*A, L)[:2*n-1]
    np.maximum(c, 0, out=c)
    return c

def _score(a):
    """Compute the exact score (scale-invariant, assume sum=1)."""
    n = a.size
    c = _conv(a)
    return 2.0 * n * c.max()

def _build_A(a):
    """Build the Toeplitz-like matrix for convolve(a, g) <= M."""
    n = a.size
    m = 2*n - 1
    # c[k] = sum_j a[j] * g[k-j], valid when 0<=j<n and 0<=k-j<n
    # So for each k, the valid j are max(0, k-(n-1)) <= j <= min(n-1, k)
    # We build A such that A @ g = c
    # A[k, j] = a[j] if max(0,k-(n-1)) <= j <= min(n-1, k), else 0
    A = np.zeros((m, n))
    for k in range(m):
        j_lo = max(0, k - (n-1))
        j_hi = min(n-1, k)
        A[k, j_lo:j_hi+1] = a[j_lo:j_hi+1]
    return A

import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, next_fast_len

def _conv(a):
    n = a.size
    L = next_fast_len(2*n - 1)
    A = rfft(a, L)
    c = irfft(A * A, L)[:2*n-1]
    np.maximum(c, 0.0, out=c)
    return c

def _build_conv_mat(a):
    """A[k,j] = a[j] if 0<=k-j<n else 0, so (A@g)[k] = conv(a,g)[k]."""
    n = a.size
    m = 2*n - 1
    # For each k (0 to 2n-2), the valid j are max(0, k-(n-1)) to min(n-1, k)
    # A[k, j] = a[j] for valid j
    A = np.zeros((m, n))

        jlo = max(0, k - n + 1)
        jhi = min(n - 1, k)
        A[k, jlo:jhi+1] = a[jlo:jhi+1]
