```
def _make_lambda(b,t):

   bm=float(b.max())
   if not math.isfinite(bm): return np.ones(m)/m
```

```
def _solve_hankel(env,lam):
   n=env.n
   try:
       H=lam[env.pair]
       H=0.5*(H+H.T)
       normH=float(np.linalg.norm(H, ord=1))
       if not math.isfinite(normH) or normH <= 0: return None
       I=env.eye
       x=None
       for ridge in (1e-12, 1e-10, 1e-8, 1e-6, 1e-4):
           try:
               xx=np.linalg.solve(H + ridge*normH*I, env.ones)
           except np.linalg.LinAlgError:
               continue
           if np.all(np.isfinite(xx)):
               mxx=float(np.max(np.abs(xx)))
               amxx=float(np.mean(np.abs(xx))) + 1e-300
               if mxx <= 1e6*amxx:
                   x=xx; break
       if x is None:
           try:
               x,_,_,_ = np.linalg.lstsq(H, env.ones, rcond=1e-10)
           except Exception:
               x=None
       if x is None or not np.all(np.isfinite(x)):
           return None
       # cap huge
       fin=x[np.isfinite(x)]
       if fin.size:
          med=float(np.median(np.abs(fin)))
          cap = max(1e6, 1e4*(med + 1e-12))
          x=np.clip(x, -cap, cap)
       return x
   except Exception:
       return None
```

```
def _lp_direction(env,a,b,stop_time=None):
    if stop_time is not None and time.monotonic() > stop_time: return None,None
    try:
      n=env.n; m=env.m
      A=np.zeros((m,n+1), dtype=float)
      A[:, n] = -1.0
      for k,row in enumerate(env.lp_rows):
          lo,hi,idx=row
          A[k, lo:hi+1] = 2.0 * a[idx]
      bmax=float(b.max())
      b_ub=b + (1e-12 + 1e-12*bmax)
      A_eq=np.zeros((1,n+1), dtype=float); A_eq[0,:n]=1.0
      c=np.zeros(n+1); c[n]=1.0
      bounds=[(0.0,None)]*n + [(-np.inf, np.inf)]
      res=linprog(c, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=[1.0], bounds=bounds, method='highs')
      if res is not None and res.success:
         g=np.asarray(res.x[:n],float)
         g=np.where(np.isfinite(g),g,0.0)
         g=np.maximum(g,0.0)
         s=g.sum()
         if s > 0:
             g=g/s
             pred=float(res.x[n])
             return g,pred
    except Exception:
         pass
    return None,None
```

```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


TARGET_UPPER_BOUND = 1.50274349


def _softmax(y):
    y = np.asarray(y, dtype=np.float64).ravel()
    if y.size == 0:
        return np.ones(1, dtype=np.float64)
    m = float(np.max(y))
    e = np.exp(y - m)
    s = float(np.sum(e))
    if not math.isfinite(s) or s <= 0.0:
        return np.ones_like(y) / y.size
    return e / s


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if not math.isfinite(s) or s <= 1e-15:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / s


def _exact_score(a):
    try:
        arr = np.asarray(a, dtype=np.float64).ravel()
        if arr.size == 0:
            return math.inf
        if not np.all(np.isfinite(arr)):
            return math.inf
        s = float(np.sum(arr))
        if s <= 0.01:
            arr = _normalize(arr)
            s = float(np.sum(arr))
            if s <= 0.01:
                return math.inf
        lst = arr.tolist()
        val = evaluate_sequence(lst)
        val = float(val)
        return val if math.isfinite(val) else math.inf
    except Exception:
        return math.inf


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = None

    if p is None or p.size == 0:
        p = np.ones(192, dtype=np.float64)
    else:
        p = np.where(np.isfinite(p), p, 0.0)
        p = np.clip(p, 0.0, 1000.0)
        s = float(np.sum(p))
        if not math.isfinite(s) or s <= 1e-12:
            p = np.ones(192, dtype=np.float64)
    return p


class ConvEnv:
    __slots__ = ("n", "m", "fsize", "pair", "eye", "ones", "lp_rows")

    def __init__(self, n):
        self.n = int(n)
        self.m = 2 * self.n - 1
        self.fsize = next_fast_len(self.m)
        ar = np.arange(self.n)
        self.pair = (ar[:, None] + ar[None, :]).astype(np.int32)
        self.eye = np.eye(self.n, dtype=np.float64)
        self.ones = np.ones(self.n, dtype=np.float64)

        self.lp_rows = []
        nn = self.n
        for k in range(self.m):
            lo = max(0, k - nn + 1)
            hi = min(nn - 1, k)
            idx = k - np.arange(lo, hi + 1)
            self.lp_rows.append((lo, hi, idx))

    def conv(self, a):
        a = np.asarray(a, dtype=np.float64).ravel()
        if a.size != self.n:
            if a.size == 0:
                a = np.ones(self.n, dtype=np.float64) / max(1, self.n)
            else:
                a = _normalize(a)
                if a.size != self.n:
                    if a.size == 1:
                        a = np.full(self.n, float(a[0]) / self.n, dtype=np.float64)
                    else:
                        x0 = np.linspace(0.0, 1.0, a.size)
                        x1 = np.linspace(0.0, 1.0, self.n)
                        a = np.interp(x1, x0, a)
                        a = _normalize(a)

        F = rfft(a, self.fsize)
        c = irfft(F * F, self.fsize)[:self.m]
        np.maximum(c, 0.0, out=c)
        if not np.all(np.isfinite(c)):
            c = np.where(np.isfinite(c), c, 0.0)
        return c

    def M(self, a):
        return float(np.max(self.conv(a)))


def _make_lambda(b, t):
    b = np.asarray(b, dtype=np.float64).ravel()
    m = b.size
    if m == 0:
        return np.ones(1, dtype=np.float64)

    bm = float(np.max(b))
    if not math.isfinite(bm):
        return np.full(m, 1.0 / m, dtype=np.float64)

    if t <= 1e-8:
        return np.full(m, 1.0 / m, dtype=np.float64)

    z = float(t) * (b - bm)
    ez = np.exp(z)
    s = float(np.sum(ez))
    if not math.isfinite(s) or s <= 0.0:
        lam = np.full(m, 1.0 / m, dtype=np.float64)
    else:
        lam = ez / s

    sl = float(np.sum(lam))
    if not math.isfinite(sl) or sl <= 0.0:
        lam = np.full(m, 1.0 / m, dtype=np.float64)
    else:
        lam /= sl

    rngv = float(np.max(b) - np.min(b))
    tol = max(1e-12, 1e-6 * abs(bm), 1e-4 * max(0.0, rngv))
    top = b >= (bm - tol)
    cnt = int(np.count_nonzero(top))
    if 0 < cnt < m:
        lam_top = np.zeros(m, dtype=np.float64)
        lam_top[top] = 1.0 / cnt
        lam = 0.7 * lam + 0.3 * lam_top
        lam /= np.sum(lam)
    return lam


def _solve_hankel(env, lam):
    try:
        n = env.n
        H = lam[env.pair]
        H = 0.5 * (H + H.T)
        normH = float(np.linalg.norm(H, ord=1))
        if not math.isfinite(normH) or normH <= 0.0:
            return None

        I = env.eye
        x = None

        for ridge in (1e-12, 1e-10, 1e-8, 1e-6, 1e-4):
            try:
                A = H + (ridge * normH) * I
                xx = np.linalg.solve(A, env.ones)
            except Exception:
                continue

            if np.all(np.isfinite(xx)):
                mxx = float(np.max(np.abs(xx)))
                amxx = float(np.mean(np.abs(xx))) + 1e-300
                if mxx <= 1e6 * amxx:
                    x = xx
                    break

        if x is None:
            try:
                x, _, _, _ = np.linalg.lstsq(H, env.ones, rcond=1e-10)
            except Exception:
                x = None

        if x is None or not np.all(np.isfinite(x)):
            return None

        fin = x[np.isfinite(x)]
        if fin.size:
            med = float(np.median(np.abs(fin)))
            cap = max(1e6, 1e4 * (med + 1e-12))
            x = np.clip(x, -cap, cap)
        return x
    except Exception:
        return None


def _kkt_trial(env, a, b, lam):
    a = np.asarray(a, dtype=np.float64).ravel()
    b = np.asarray(b, dtype=np.float64).ravel()
    best_a = a
    best_M = float(np.max(b))
    best_b = b

    x = _solve_hankel(env, lam)
    if x is None:
        return best_a, best_M, best_b

    candidate_vectors = []

    xp = np.maximum(x, 0.0)
    sx = float(np.sum(xp))
    if math.isfinite(sx) and sx > 1e-12:
        candidate_vectors.append(xp / sx)

    xa = np.abs(x)
    sx = float(np.sum(xa))
    if math.isfinite(sx) and sx > 1e-12:
        hx = xa / sx
        if len(candidate_vectors) == 0 or not np.allclose(hx, candidate_vectors[0], rtol=1e-8, atol=1e-12):
            candidate_vectors.append(hx)

    if not candidate_vectors:
        return best_a, best_M, best_b

    for xn in candidate_vectors:
        for beta in (0.10, 0.25, 0.50, 0.75, 1.0):
            c = (1.0 - beta) * a + beta * xn
            sc = float(np.sum(c))
            if sc <= 1e-15:
                continue
            c = c / sc
            bc = env.conv(c)
            mc = float(np.max(bc))
            if mc < best_M - 1e-15:
                best_a = c
                best_M = mc
                best_b = bc

    return best_a, best_M, best_b


def _lp_direction(env, a, b, stop_time=None):
    if stop_time is not None and time.monotonic() > stop_time:
        return None, None
    try:
        n = env.n
        m = env.m
        A = np.zeros((m, n + 1), dtype=np.float64)
        A[:, n] = -1.0

        for k, (lo, hi, idx) in enumerate(env.lp_rows):
            A[k, lo:hi + 1] = 2.0 * a[idx]

        bmax = float(np.max(b))
        b_ub = b + (1e-12 + 1e-12 * max(0.0, bmax))

        A_eq = np.zeros((1, n + 1), dtype=np.float64)
        A_eq[0, :n] = 1.0

        c = np.zeros(n + 1, dtype=np.float64)
        c[n] = 1.0

        bounds = [(0.0, None) for _ in range(n)] + [(-np.inf, None)]

        res = linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=[1.0],
            bounds=bounds,
            method="highs",
        )

        if res is not None and res.success:
            g = np.asarray(res.x[:n], dtype=np.float64)
            g = np.where(np.isfinite(g), g, 0.0)
            g = np.maximum(g, 0.0)
            s = float(np.sum(g))
            if s > 1e-15:
                g /= s
                pred = float(res.x[n])
                return g, pred
    except Exception:
        pass
    return None, None


def _line_search_lp(env, a, g, Mcur):
    a = np.asarray(a, dtype=np.float64).ravel()
    g = np.asarray(g, dtype=np.float64).ravel()
    g = np.where(np.isfinite(g), g, 0.0)
    g = np.maximum(g, 0.0)
    s = float(np.sum(g))
    if s <= 1e-15:
        return a, Mcur
    g = g / s

    best_a = a
    best_M = Mcur

    for step in (0.03, 0.05, 0.10, 0.20, 0.40, 0.70, 1.0):
        c = (1.0 - step) * a + step * g
        sc = float(np.sum(c))
        if sc <= 1e-15:
            continue
        c = c / sc
        cb = env.conv(c)
        cm = float(np.max(cb))
        if cm < best_M - 1e-15:
            best_M = cm
            best_a = c
    return best_a, best_M


def _perturb(a, rng, sigma):
    a = _normalize(a)
    y = np.log(np.maximum(a, 1e-12))
    n = a.size

    if rng.random() < 0.45 and n > 4:
        i = np.arange(n, dtype=np.float64)
        two_pi = 2.0 * math.pi
        f1 = int(rng.integers(1, max(2, n // 3)))
        ph1 = float(rng.uniform(0.0, two_pi))
        amp1 = float(sigma * rng.uniform(0.7, 1.8))
        y = y + amp1 * np.sin(two_pi * f1 * i / n + ph1)

        if rng.random() < 0.6:
            f2 = int(rng.integers(1, max(2, n // 5)))
            ph2 = float(rng.uniform(0.0, two_pi))
            amp2 = float(sigma * rng.uniform(0.3, 1.0))
            y = y + amp2 * np.cos(two_pi * f2 * i / n + ph2)

    y = y + rng.normal(0.0, float(sigma), a.size)
    return _softmax(y)


def _improve_kkt(env, a0, deadline, rng, max_perturb=60):
    a = _normalize(a0)
    b = env.conv(a)
    M = float(np.max(b))

    best_a = a.copy()
    best_M = M
    best_b = b.copy()

    t_cycle = (10.0, 25.0, 75.0, 250.0, 800.0)
    step = 0
    no_improve = 0
    perturb_count = 0
    lp_calls = 0
    last_lp = 0.0

    if env.n <= 192:
        max_lp = 300
    elif env.n <= 256:
        max_lp = 120
    elif env.n <= 384:
        max_lp = 45
    else:
        max_lp = 0
    use_lp = max_lp > 0

    while time.monotonic() < deadline:
        now = time.monotonic()
        if now > deadline - 1.0:
            break

        lam = _make_lambda(b, t_cycle[step % len(t_cycle)])
        step += 1

        cand, cM, cB = _kkt_trial(env, a, b, lam)

        if cM < best_M * (1.0 - 1e-13):
            best_a = cand
            best_M = cM
            best_b = cB
            a = cand
            M = cM
            b = cB
            no_improve = 0
        elif cM < M * (1.0 - 1e-12):
            a = cand
            M = cM
            b = cB
            no_improve += 1
        else:
            no_improve += 1

            if (
                use_lp
                and lp_calls < max_lp
                and no_improve >= 3
                and (now - last_lp) > 0.15
            ):
                g, pred = _lp_direction(env, a, b, deadline - 2.0)
                lp_calls += 1
                last_lp = time.monotonic()

                if g is not None and pred < M * (1.0 - 1e-12):
                    an, Mn = _line_search_lp(env, a, g, M)
                    if Mn < best_M * (1.0 - 1e-12):
                        best_a = an
                        best_M = Mn
                        best_b = env.conv(an)
                        a = an
                        M = Mn
                        b = best_b
                        no_improve = 0
                    elif Mn < M * (1.0 - 1e-10):
                        a = an
                        M = Mn
                        b = env.conv(an)
                        no_improve += 1

            if no_improve >= 7:
                sigma = 0.03 + 0.02 * (perturb_count % 6)
                if sigma > 0.25:
                    sigma = 0.25

                base = best_a if M > best_M * 1.0005 else a
                a = _perturb(base, rng, sigma)
                b = env.conv(a)
                M = float(np.max(b))

                if M < best_M * (1.0 - 1e-13):
                    best_a = a
                    best_M = M
                    best_b = b

                step = 0
                no_improve = 0
                perturb_count += 1
                if perturb_count > max_perturb:
                    break

    return best_a, best_M, best_b


def _resample(parent, L):
    p = _normalize(parent)
    L = int(L)
    if L <= 0:
        return np.ones(1, dtype=np.float64)
    if p.size == L:
        return p.copy()
    if p.size == 0:
        return np.ones(L, dtype=np.float64) / L
    x0 = np.linspace(0.0, 1.0, p.size)
    x1 = np.linspace(0.0, 1.0, L)
    v = np.interp(x1, x0, p)
    v = _normalize(v)
    return v


def _random_init(n, rng, style, parent=None):
    n = int(n)
    i = np.arange(n, dtype=np.float64)
    two_pi = 2.0 * math.pi
    v = int(style) % 8

    base = None
    if parent is not None:
        base = _resample(parent, n)

    if v == 0:
        a = np.ones(n, dtype=np.float64)
        for _ in range(4):
            f = int(rng.integers(1, max(2, n // 3)))
            ph = float(rng.uniform(0.0, two_pi))
            amp = float(rng.uniform(0.1, 0.5))
            a = a + amp * np.sin(two_pi * f * i / n + ph)

    elif v == 1:
        ld = float(rng.uniform(0.1, 1.5))
        rd = float(rng.uniform(0.1, 1.5))
        a = (
            np.exp(-ld * i / max(1.0, n - 1))
            + np.exp(-rd * (n - 1 - i) / max(1.0, n - 1))
            + 0.1
        )
        if n > 4:
            f = int(rng.integers(1, max(2, n // 4)))
            a = a * (1.0 + 0.4 * np.sin(two_pi * f * i / n + float(rng.uniform(0.0, two_pi))))

    elif v == 2:
        y = np.zeros(n, dtype=np.float64)
        for f in (1, 2, 3, 5, 8, 13):
            amp = float(rng.uniform(0.0, 1.0)) * math.exp(-0.2 * f)
            y = y + amp * np.sin(two_pi * f * i / n + float(rng.uniform(0.0, two_pi)))
        a = np.exp(y)

    elif v == 3:
        a = np.full(n, 0.1, dtype=np.float64)
        for _ in range(int(rng.integers(3, 7))):
            c = float(rng.uniform(0.0, n))
            w = float(rng.uniform(n / 10.0, n / 3.0))
            amp = float(rng.uniform(0.2, 1.0))
            a = a + amp * np.exp(-0.5 * (((i - c) / max(1e-6, w)) ** 2))

    elif v == 4:
        logc = float(rng.uniform(-0.05, 0.05))
        logc2 = float(rng.uniform(-0.05, 0.05))
        a = np.exp(logc * i) + np.exp(logc2 * (n - 1 - i)) + 0.2

    elif v == 5:
        d = np.minimum(i, n - 1 - i) + 0.5
        a = 1.0 / (1.0 + float(rng.uniform(0.2, 1.2)) * d) + 0.2
        if n > 4:
            f = int(rng.integers(1, max(2, n // 5)))
            a = a + 0.2 * np.abs(np.sin(two_pi * f * i / n + float(rng.uniform(0.0, two_pi))))

    elif v == 6:
        f1 = int(rng.integers(1, max(2, n // 2)))
        f2 = int(rng.integers(1, max(2, n // 2)))
        a = (
            1.0
            + 0.4 * np.sin(two_pi * f1 * i / n + float(rng.uniform(0.0, two_pi)))
            + 0.3 * np.cos(two_pi * f2 * i / n + float(rng.uniform(0.0, two_pi)))
        )
    else:
        a = rng.uniform(0.2, 1.0, n)

    a = np.maximum(a, 1e-6)
    if base is not None:
        w = float(rng.uniform(0.15, 0.45))
        a = w * base + (1.0 - w) * a

    return _normalize(a)


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1300.0
    if not math.isfinite(budget) or budget <= 0.0:
        budget = 0.1

    safety = max(2.0, min(20.0, budget * 0.02 + 1.0))
    deadline = t0 + max(0.2, budget - safety)

    try:
        rng = np.random.default_rng(int(seed) % (2**32))
    except Exception:
        rng = np.random.default_rng(42)

    parent = _get_parent()
    parent_norm = _normalize(parent)

    best_seq = parent_norm.copy()
    best_score = _exact_score(best_seq)

    if not math.isfinite(best_score):
        best_seq = np.ones(192, dtype=np.float64) / 192.0
        best_score = _exact_score(best_seq)

    if not math.isfinite(best_score):
        return [1.0]

    n0 = int(best_seq.size)
    can_optimize = (0 < n0 <= 512)

    env0 = ConvEnv(n0) if can_optimize else None

    def consider(env, a, force=False):
        nonlocal best_seq, best_score
        if a is None:
            return
        try:
            arr = np.asarray(a, dtype=np.float64).ravel()
            if arr.size == 0:
                return
            arr = np.where(np.isfinite(arr), arr, 0.0)
            arr = _normalize(arr)
        except Exception:
            return

        if env is not None and arr.size == env.n:
            try:
                b = env.conv(arr)
                mx = float(np.max(b))
                s = float(np.sum(arr))
                fast = (2.0 * env.n * mx) / max(1e-300, s * s) if s > 0 else math.inf
            except Exception:
                fast = math.inf
        else:
            fast = math.nan
            try:
                fast = _exact_score(arr)
            except Exception:
                fast = math.inf

        if force or (math.isfinite(fast) and fast < best_score - 1e-9):
            sc = _exact_score(arr)
            if math.isfinite(sc) and sc < best_score - 1e-12:
                best_score = sc
                best_seq = arr.copy()

    if can_optimize:
        try:
            main_dl = min(deadline, t0 + budget * 0.50)
            if time.monotonic() < main_dl:
                a_res, _, _ = _improve_kkt(env0, best_seq, main_dl, rng, max_perturb=90)
                consider(env0, a_res, force=True)

            restart_end = min(deadline - 30.0, t0 + budget * 0.72)
            if budget > 250.0 and time.monotonic() < restart_end:
                for r in range(24):
                    now = time.monotonic()
                    if now >= restart_end:
                        break
                    rem = restart_end - now
                    if rem < 10.0:
                        break

                    dur = max(8.0, min(30.0, rem / max(1, 24 - r)))

                    if r % 4 == 0:
                        init = _perturb(best_seq, rng, 0.05 + 0.02 * (r % 5))
                    elif r % 4 == 1:
                        init = _random_init(n0, rng, r, best_seq)
                    elif r % 4 == 2:
                        init = _random_init(n0, rng, r, None)
                    else:
                        init = _resample(best_seq, n0)
                        init = _perturb(init, rng, 0.10)

                    sub_dl = time.monotonic() + dur
                    try:
                        a_res, _, _ = _improve_kkt(env0, init, sub_dl, rng, max_perturb=25)
                        consider(env0, a_res, force=True)
                    except Exception:
                        pass
        except Exception:
            pass

        if budget > 450.0 and time.monotonic() < deadline - 30.0:
            alt_len_list = []
            for L in (256, 128, 191, 288, 384):
                if L != n0 and L not in alt_len_list:
                    alt_len_list.append(L)

            for L in alt_len_list:
                if time.monotonic() > deadline - 35.0:
                    break
                if L > 512:
                    continue
                rem_total = deadline - time.monotonic() - 15.0
                if rem_total < 20.0:
                    break

                try:
                    env = ConvEnv(L)
                except Exception:
                    continue

                inits = []
                baser = _resample(best_seq, L)
                inits.append(baser)
                inits.append(_perturb(baser, rng, 0.08))
                inits.append(_random_init(L, rng, L % 8, best_seq))
                inits.append(_random_init(L, rng, (L + 1) % 8, None))

                for idx, init in enumerate(inits):
                    now = time.monotonic()
                    if now > deadline - 20.0:
                        break
                    rem_now = deadline - now - 10.0
                    if rem_now < 8.0:
                        break

                    dur = max(6.0, min(40.0, rem_now / max(1, len(inits) - idx)))
                    sub_dl = now + dur

                    try:
                        a_res, _, _ = _improve_kkt(env, init, sub_dl, rng, max_perturb=18)
                        consider(env, a_res, force=True)
                    except Exception:
                        pass
                try:
                    env = None
                except Exception:
                    pass

    try:
        final_score = _exact_score(best_seq)
    except Exception:
        final_score = math.inf

    if not math.isfinite(final_score):
        try:
            fb = _normalize(_get_parent())
            fb_score = _exact_score(fb)
            if math.isfinite(fb_score):
                best_seq = fb
                final_score = fb_score
        except Exception:
            pass

    if not math.isfinite(final_score):
        best_seq = np.ones(192, dtype=np.float64) / 192.0
        final_score = _exact_score(best_seq)

    if not math.isfinite(final_score):
        return [1.0]

    out = [float(x) for x in best_seq]
    if not out:
        out = [1.0]
    return out
```
