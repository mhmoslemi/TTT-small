The solution is: `a_new[i] = a[i] * exp(-eta * grad[i]) / Z` (same as MWU!).

import math

def _soft_score(a, n, tau, fs):
    """Compute soft-max surrogate score and the probability vector."""
    b = irfft(rfft(a, fs) * rfft(a, fs), fs)[:2*n-1].real
    bmax = b.max()
    bt = (b - bmax) / tau
    e = np.exp(np.clip(bt, -80, 80))
    se = e.sum()
    if se < 1e-300:
        p = np.zeros(2*n-1)
        p[int(np.argmax(b))] = 1.0
    else:
        p = e / se

    ar = a[::-1]

        a = a_new
        iter_count += 1

    return local_best, local_best_score


def _opt_cycling(a_init, t_end, seed_val, n, fs, fs2):
    """Cycling method: minimize the active constraint one at a time."""
    rng = np.random.default_rng(seed_val)
    a = a_init / max(a_init.sum(), 1e-300)
    local_best = a.copy()
    local_best_score = 2.0 * n * np.convolve(a, a).max()
    eta = 0.02
    no_improve = 0

    while time.monotonic() < t_end:
        b = np.convolve(a, a)
        k_star = int(np.argmax(b))
        bval = b[k_star]

        # Exact gradient at active index
        grad = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        idx = np.arange(lo, hi + 1)
        grad[idx] = 2.0 * a[k_star - idx]

        # Projected gradient step on simplex
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-300:
            break
        a_new = a_new / s

        # Line search (backtracking)
        best_trial = a
        best_trial_score = 2.0 * n * np.convolve(a, a).max()
        for ls in range(10):
            b_ls = np.convolve(a_new, a_new)
            sc_ls = 2.0 * n * b_ls.max()
            if sc_ls < best_trial_score:
                # Check if k_star is still active or if another index dominates
                if b_ls[k_star] <= b_ls.max() + 1e-10:
                    best_trial = a_new
                    best_trial_score = sc_ls
                    break
            a_new = a + 0.5 * (a_new - a)

        if best_trial_score < local_best_score - 1e-13:
            local_best = best_trial.copy()
            local_best_score = best_trial_score
            no_improve = 0
            a = best_trial
        else:
            a = best_trial
            no_improve += 1
            if no_improve > 50:
                eta = max(0.001, eta * 0.7)
            if no_improve > 300:
                a = local_best + rng.normal(0, 0.005, n)
                a = np.clip(a, 1e-12, None)
                a = a / a.sum()
                eta = 0.02
                no_improve = 0

    return local_best, local_best_score


def _polish_local(a_init, t_end, n, seed_val):
    """Fine local search with small random perturbations."""
    rng = np.random.default_rng(seed_val)
    a = a_init.copy()
    n = len(a)
    a = a / a.sum()
    best = a.copy()
    best_sc = 2.0 * n * np.convolve(a, a).max()
    sigma = 0.002
    stuck = 0

    while time.monotonic() < t_end - 1:
        r = rng.random()
        if r < 0.4:
            a_new = a + rng.normal(0, sigma, n)
        elif r < 0.6:
            i = rng.integers(0, n)
            w = max(1, int(rng.integers(1, max(2, n // 20 + 1))))
            i2 = min(n, i + w)
            a_new = a.copy()
            a_new[i:i2] *= np.exp(rng.normal(0, sigma, i2 - i))
        elif r < 0.8:
            # Sinusoidal perturbation
            f = int(rng.integers(1, max(2, n // 4)))
            ph = rng.uniform(0, 2 * math.pi)
            a_new = a * (1 + sigma * 0.5 * np.sin(2 * math.pi * f * np.arange(n) / n + ph))
        else:
            # Swap two entries
            i, j = rng.integers(0, n, 2)
            a_new = a.copy()
            delta = (a_new[i] - a_new[j]) * 0.5
            a_new[i] -= delta
            a_new[j] += delta

        a_new = np.clip(a_new, 0, None)
        s = a_new.sum()
        if s < 1e-300:
            continue
        a_new = a_new / s

        sc = 2.0 * n * np.convolve(a_new, a_new).max()
        if sc < best_sc - 1e-14:
            best = a_new.copy()
            best_sc = sc
            a = a_new
            stuck = 0
            sigma = max(0.0005, sigma * 0.98)
        else:
            a = a_new
            stuck += 1
            if stuck > 100:
                a = best.copy()
                sigma = min(0.02, sigma * 1.5)
                stuck = 0

    return best, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    hard_deadline = t0 + max(1.0, float(budget_s)) - 8.0

    # Get parent
    parent = None
    try:
        h = globals().get('height_sequence_1')
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            if p.size > 0 and np.all(np.isfinite(p)):
                p = np.clip(p, 0, 1000)
                if p.sum() >= 0.01:
                    parent = p
    except Exception:
        parent = None

    # Initial best
    best_a = None
    best_score = float('inf')

    def _update(a, sc):
        nonlocal best_a, best_score
        if a is None or sc is None:
            return
        try:
            a = np.asarray(a, dtype=np.float64).ravel()
            if a.size == 0 or not np.all(np.isfinite(a)):
                return
            if sc < best_score - 1e-14:
                best_a = a.copy()
                best_score = float(sc)
        except Exception:
            pass

    if parent is not None:
        n_p = parent.size
        a_p = parent / parent.sum()
        sc_p = 2.0 * n_p * np.convolve(a_p, a_p).max()
        _update(a_p, sc_p)

    if best_a is None:
        best_a = np.ones(192) / 192.0
        best_score = 2.0

    rng = np.random.default_rng(seed)

    # Phase 1: MWU optimization from parent with various hyperparameters
    phase1_end = t0 + 0.50 * (budget_s - 10)
    a_parent = parent / parent.sum() if parent is not None else None

    if a_parent is not None:
        n_p = a_parent.size
        fs_p = next_fast_len(2 * n_p - 1)
        fs2_p = next_fast_len(3 * n_p - 2)
        configs = [
            (0.05, 0.4, 1),
            (0.1, 0.5, 2),
            (0.2, 0.5, 3),
            (0.5, 0.4, 4),
            (1.0, 0.3, 5),
            (0.03, 0.6, 6),
        ]
        for ci, (tau0, eta0, sid) in enumerate(configs):
            if time.monotonic() > phase1_end - 3:
                break
            t_end_ci = min(phase1_end - 1, time.monotonic() + max(5, (phase1_end - time.monotonic()) / max(1, len(configs) - ci)))
            try:
                a_opt, sc = _opt_mwu(a_parent, tau0, eta0, t_end_ci, seed + sid * 1000, fs_p, fs2_p, n_p)
                _update(a_opt, sc)
            except Exception:
                pass
            # Also try cycling from the result
            if time.monotonic() < phase1_end - 5:
                try:
                    a_cyc, sc_c = _opt_cycling(a_parent, phase1_end - 3, seed + sid * 777 + 5, n_p, fs_p, fs2_p)
                    _update(a_cyc, sc_c)
                except Exception:
                    pass

    # Phase 2: Try other n values
    phase2_end = t0 + 0.80 * (budget_s - 10)
    sizes_to_try = [192, 224, 256, 320, 384]
    for n_try in sizes_to_try:
        if time.monotonic() > phase2_end - 10:
            break
        # Skip if we already have a good result at this size
        for trial in range(3):
            if time.monotonic() > phase2_end - 5:
                break
            if trial == 0:
                a_init = np.ones(n_try) / n_try
            elif trial == 1:
                a_init = np.abs(rng.normal(1.0, 0.4, n_try)) + 0.1
                a_init = a_init / a_init.sum()
            else:
                t_arr = np.arange(n_try) / n_try
                a_init = (0.5 + 0.5 * np.cos(2 * math.pi * t_arr)) ** 2 + 0.05
                a_init = a_init / a_init.sum()

            fs_t = next_fast_len(2 * n_try - 1)
            fs2_t = next_fast_len(3 * n_try - 2)
            t_end_t = min(phase2_end - 2, time.monotonic() + max(5, (phase2_end - time.monotonic()) / 4))
            try:
                a_opt, sc = _opt_mwu(a_init, 0.3, 0.5, t_end_t, seed + n_try * 10 + trial, fs_t, fs2_t, n_try)
                _update(a_opt, sc)
            except Exception:
                pass

    # Phase 3: Fine polish from global best
    phase3_end = hard_deadline - 5
    if time.monotonic() < phase3_end - 10:
        n_best = best_a.size
        a_pol = best_a.copy()
        a_pol = a_pol / a_pol.sum()
        try:
            a_fin, sc_fin = _polish_local(a_pol, phase3_end, n_best, seed + 424242)
            _update(a_fin, sc_fin)
        except Exception:
            pass

        # One more MWU pass with very small tau
        if time.monotonic() < phase3_end - 15:
            fs_b = next_fast_len(2 * n_best - 1)
            fs2_b = next_fast_len(3 * n_best - 2)
            try:
                a_fin2, sc_fin2 = _opt_mwu(best_a, 0.02, 0.2, phase3_end - 3, seed + 88888, fs_b, fs2_b, n_best)
                _update(a_fin2, sc_fin2)
            except Exception:
                pass

        # Cycling polish
        if time.monotonic() < phase3_end - 10:
            try:
                a_cyc2, sc_cyc2 = _opt_cycling(best_a, phase3_end - 2, seed + 55555, n_best, fs_b, fs2_b)
                _update(a_cyc2, sc_cyc2)
            except Exception:
                pass

    # Final validation with exact evaluator
    try:
        cand = [float(max(0.0, min(1000.0, v))) for v in best_a]
    except Exception:
        cand = [1.0]

    try:
        fs = float(evaluate_sequence(cand))
        if not np.isfinite(fs):
            cand = [1.0]
            fs = float(evaluate_sequence(cand))
    except Exception:
        cand = [1.0]

    if not np.isfinite(fs):
        return [1.0]
    return cand
