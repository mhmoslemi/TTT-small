def proj_simplex(v, target=1.0):
   v=np.asarray(v,float).ravel()
   n=v.size
   if n==1: return np.ones(1) if target>0 else [0]
   u=np.sort(v)[::-1]
   css=np.cumsum(u)
   rho=np.nonzero(u*(np.arange(1,n+1))+target - css > 0)[0]
   if rho.size==0: return np.ones(n)/n
   idx=rho[-1]

   w=np.maximum(v-theta,0)

   if s <= target*1e-15: return np.ones(n)/n
   if s != target: w*=target/s
   return w

def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()

        horizon = max(10.0, float(budget_s))
    except Exception:

    def to_arr(a):
        try:
            arr = np.asarray(a, dtype=np.float64).ravel()
        except Exception:
            return None
        if arr.size == 0:
            return None
        if not np.all(np.isfinite(arr)):
            return None
        arr = np.clip(arr, 0.0, 1000.0)
        return arr

    def normalize(a):
        arr = to_arr(a)
        if arr is None:
            return np.ones(96, dtype=np.float64)
        s = float(arr.sum())
        if not math.isfinite(s) or s <= 1e-15:
            n = arr.size if arr.size > 0 else 96
            return np.ones(n, dtype=np.float64) / max(1, n)
        arr = arr / s
        if not np.all(np.isfinite(arr)):
            return np.ones(arr.size, dtype=np.float64) / max(1, arr.size)
        return arr

    def interp_len(src, n):
        n = int(max(2, n))
        arr = to_arr(src)
        if arr is None or arr.size == 0:
            return np.ones(n, dtype=np.float64)

            return normalize(arr)
        if arr.size == 1:
            out = np.full(n, float(arr[0]), dtype=np.float64)
            return normalize(out)
        ox = np.linspace(0.0, 1.0, arr.size)
        nx = np.linspace(0.0, 1.0, n)
        out = np.interp(nx, ox, arr)
        out = np.clip(out, 0.0, None)
        if float(out.sum()) <= 1e-15:
            out = np.ones(n, dtype=np.float64)
        return normalize(out)

    def proj_simplex(v, target=1.0):
        v = np.asarray(v, dtype=np.float64).ravel()
        v = np.nan_to_num(v, nan=0.0, posinf=1e8, neginf=-1e8)
        n = v.size
        if n == 0:
            return np.ones(1, dtype=np.float64) / max(1, int(target))
        if n == 1:
            return np.array([float(target)], dtype=np.float64)

        u = np.sort(v)[::-1]
        css = np.cumsum(u)
        cond = u * (np.arange(1, n + 1, dtype=np.float64)) + target - css
        rho = np.nonzero(cond > 0.0)[0]
        if rho.size == 0:
            w = np.ones(n, dtype=np.float64) / n
        else:
            idx = int(rho[-1])
            theta = (css[idx] - target) / (idx + 1.0)
            w = np.maximum(v - theta, 0.0)
        s = float(w.sum())
        if not math.isfinite(s) or s <= target * 1e-15:
            w = np.ones(n, dtype=np.float64) / n
        else:

        return w

    def line_search_dir(a, v, loc, etas):

        mode_i = 0
        stag = 0

        for _ in range(max_iters):
            if now() >= stop:
                break

            c = conv_auto(arr)
            M = float(c.max())
            if not math.isfinite(M) or M <= 0.0:
                break

            if stag > 8 and (stag % 4 == 0):
                top = min(12, c.size)
                if top <= 0:
                    top = 1
                if top >= c.size:
                    idx = np.arange(c.size)
                else:
                    idx = np.argpartition(-c, top - 1)[:top]
                w = rng.dirichlet(np.ones(top, dtype=np.float64))
                p = np.zeros(c.size, dtype=np.float64)
                p[idx] = w
                mode_i += 1
            else:
                mtype, val = modes[mode_i % len(modes)]
                p = make_weights(c, mtype, val)
                mode_i += 1

            if p is None:
                continue

            G = grad_weights(p, arr)
            v = G - float(G.mean())
            vv = float(np.linalg.norm(v, 1))
            if not math.isfinite(vv) or vv < 1e-13:
                improved = False
                for al in (2, 10, 50, 200):
                    p2 = make_weights(c, "alpha", al)
                    if p2 is None:
                        continue
                    G2 = grad_weights(p2, arr)
                    v2 = G2 - float(G2.mean())
                    if float(np.linalg.norm(v2, 1)) < 1e-13:
                        continue
                    cand, sc = line_search_dir(arr, v2, loc, etas)
                    if cand is not None and sc < loc:
                        arr = cand
                        loc = sc
                        stag = 0
                        improved = True
                        if loc < best_loc:
                            best_loc = loc
                            best_a = arr.copy()
                            update_best(arr, loc)
                        break
                if not improved:
                    stag += 1
                    if stag > 40:
                        break
                continue

            cand, sc = line_search_dir(arr, v, loc, etas)
            if cand is not None and sc < loc - 1e-14:
                arr = cand
                loc = sc
                stag = 0
                if loc < best_loc:
                    best_loc = loc
                    best_a = arr.copy()
                    update_best(arr, loc)
            else:
                stag += 1
                if stag == 10:
                    improved = False
                    for _r in range(5):
                        top = min(12, c.size)
                        if top <= 0:
                            top = 1
                        if top >= c.size:
                            idx = np.arange(c.size)
                        else:
                            idx = np.argpartition(-c, top - 1)[:top]
                        w = rng.dirichlet(np.ones(top, dtype=np.float64))
                        pr = np.zeros(c.size, dtype=np.float64)
                        pr[idx] = w
                        Gr = grad_weights(pr, arr)
                        vr = Gr - float(Gr.mean())
                        if float(np.linalg.norm(vr, 1)) < 1e-13:
                            continue
                        cand_r, sc_r = line_search_dir(arr, vr, loc, etas)
                        if cand_r is not None and sc_r < loc:
                            arr = cand_r
                            loc = sc_r
                            stag = 0
                            improved = True
                            if loc < best_loc:
                                best_loc = loc
                                best_a = arr.copy()
                                update_best(arr, loc)
                            break
                    if not improved:
                        eps = max(1e-6, 0.05 * float(arr.mean()))
                        aj = proj_simplex(arr + eps * rng.standard_normal(n))
                        sj = score_fast(aj)
                        if math.isfinite(sj) and sj < loc:
                            arr = aj
                            loc = sj
                            stag = 0
                            if loc < best_loc:
                                best_loc = loc
                                best_a = arr.copy()
                                update_best(arr, loc)
                        else:
                            stag += 5
                if stag > 40:
                    break

        if best_loc < loc:
            return best_a
        return best_a

    def lin_dir(a, R, tl):
        if not HAVE_LP:
            return None
        n = len(a)
        if n > 800:
            return None
        c = conv_auto(a)
        m = 2 * n - 1

        A = np.zeros((m + 2 * n, n + 1), dtype=np.float64)
        b = np.empty(m + 2 * n, dtype=np.float64)

        for j in range(n):
            A[j:j + n, j] = 2.0 * a

        A[:m, n] = -1.0
        b[:m] = -c

        idx = np.arange(n)
        r0 = m
        A[r0:r0 + n, idx] = -1.0
        b[r0:r0 + n] = a

        r1 = r0 + n
        A[r1:r1 + n, idx] = 1.0
        b[r1:r1 + n] = float(R)

        cobj = np.zeros(n + 1, dtype=np.float64)
        cobj[n] = 1.0
        Aeq = np.zeros((1, n + 1), dtype=np.float64)
        Aeq[0, :n] = 1.0
        beq = np.array([0.0], dtype=np.float64)
        bounds = [(None, None)] * (n + 1)

        opts = {"presolve": True}
        if tl is not None and float(tl) > 0.0:
            opts["time_limit"] = float(min(5.0, max(0.05, float(tl))))

        res = None
        try:
            res = linprog(cobj, A_ub=A, b_ub=b, A_eq=Aeq, b_eq=beq,
                          bounds=bounds, method="highs", options=opts)
        except Exception:
            try:
                res = linprog(cobj, A_ub=A, b_ub=b, A_eq=Aeq, b_eq=beq,
                              bounds=bounds, method="highs", options={"presolve": True})
            except Exception:
                res = None

        if res is None or getattr(res, "status", 1) != 0:
            return None
        x = getattr(res, "x", None)
        if x is None or len(x) != n + 1:
            return None
        delta = np.asarray(x[:n], dtype=np.float64)
        if not np.all(np.isfinite(delta)):
            return None
        if float(np.linalg.norm(delta, 1)) < 1e-15:
            return None
        return delta

    def linlp_local(a, stop, max_iters=500):
        arr = normalize(a)
        n = arr.size
        if not HAVE_LP or n > 800:
            return arr
        loc = score_fast(arr)
        update_best(arr, loc)
        best_a = arr.copy()
        best_loc = loc

        stag = 0
        R_idx = 0
        Rfactors = (1.0, 3.0, 10.0, 35.0, 120.0)
        scales = (0.02, 0.1, 0.3, 0.6, 1.0, 1.5)

        for _ in range(max_iters):
            if now() >= stop - 1.0:
                break
            R = max(1e-8, Rfactors[R_idx % len(Rfactors)] * float(arr.mean()))
            tl = max(0.15, min(2.5, float(stop - now() - 0.5)))
            if tl <= 0.05:
                break

            delta = lin_dir(arr, R, tl)
            if delta is None:
                stag += 1
                R_idx = (R_idx + 1) % len(Rfactors)
                if stag > 8:
                    break
                continue

            best_cand = arr
            best_sc = loc
            for s in scales:
                cand = arr + float(s) * delta
                cand = proj_simplex(cand)
                sc = score_fast(cand)
                if math.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_cand = cand

            if best_sc < loc - 1e-13:
                arr = best_cand
                loc = best_sc
                stag = 0
                R_idx = 0
                if loc < best_loc:
                    best_loc = loc
                    best_a = arr.copy()
                    update_best(arr, loc)
            else:
                stag += 1
                R_idx = (R_idx + 1) % len(Rfactors)
                if stag == 4:
                    eps = max(1e-6, 0.03 * float(arr.mean()))
                    aj = proj_simplex(arr + eps * rng.standard_normal(n))
                    sj = score_fast(aj)
                    if math.isfinite(sj) and sj < loc:
                        arr = aj
                        loc = sj
                        stag = 0
                        R_idx = 0
                        if loc < best_loc:
                            best_loc = loc
                            best_a = arr.copy()
                            update_best(arr, loc)
                if stag > 10:
                    break

        if best_loc < loc:
            return best_a
        return best_a

    def cross_lp_local(a, stop, max_iters=80):
        arr = normalize(a)
        n = arr.size
        if not HAVE_LP or n > 800:
            return arr

        loc = score_fast(arr)
        update_best(arr, loc)
        best_a = arr.copy()
        best_loc = loc

        stag = 0
        cobj = -np.ones(n, dtype=np.float64)
        bounds = [(0.0, 10.0)] * n
        margins = (1.0, 0.99, 0.97, 0.94, 0.90, 0.84)

        for _ in range(max_iters):
            if now() >= stop - 1.0:
                break
            c = conv_auto(arr)
            M = float(c.max())
            if not math.isfinite(M) or M <= 0.0:
                break

            A = np.zeros((2 * n - 1, n), dtype=np.float64)
            for j in range(n):
                A[j:j + n, j] = arr

            improved = False
            tl = max(0.2, min(2.5, float(stop - now() - 0.5)))
            opts = {"presolve": True, "time_limit": tl}

            for mg in margins:
                if now() >= stop - 0.5:
                    break
                b_ub = np.full(2 * n - 1, M * float(mg) + 1e-12, dtype=np.float64)
                res = None
                try:
                    res = linprog(cobj, A_ub=A, b_ub=b_ub, bounds=bounds,
                                  method="highs", options=opts)
                except Exception:
                    try:
                        res = linprog(cobj, A_ub=A, b_ub=b_ub, bounds=bounds,
                                      method="highs", options={"presolve": True})
                    except Exception:
                        res = None

                if res is None or getattr(res, "status", 1) != 0:
                    continue
                g = getattr(res, "x", None)
                if g is None:
                    continue
                g = np.asarray(g, dtype=np.float64)
                g = np.clip(g, 0.0, 1000.0)
                sg = float(g.sum())
                if not math.isfinite(sg) or sg <= 1e-12:
                    continue
                g = g / sg

                best_cand = arr
                best_sc = loc
                for t in (0.1, 0.2, 0.35, 0.5, 0.7, 0.9, 1.0):
                    cand = (1.0 - float(t)) * arr + float(t) * g
                    cand = proj_simplex(cand)
                    sc = score_fast(cand)
                    if math.isfinite(sc) and sc < best_sc:
                        best_sc = sc
                        best_cand = cand

                if best_cand is not None and best_sc < loc - 1e-13:
                    arr = best_cand
                    loc = best_sc
                    improved = True
                    if loc < best_loc:
                        best_loc = loc
                        best_a = arr.copy()
                        update_best(arr, loc)
                    break

            if not improved:
                stag += 1
                if stag == 3:
                    eps = max(1e-6, 0.03 * float(arr.mean()))
                    aj = proj_simplex(arr + eps * rng.standard_normal(n))
                    sj = score_fast(aj)
                    if math.isfinite(sj) and sj < loc:
                        arr = aj
                        loc = sj
                        stag = 0
                        if loc < best_loc:
                            best_loc = loc
                            best_a = arr.copy()
                            update_best(arr, loc)
                if stag > 12:
                    break
            else:
                stag = 0

        if best_loc < loc:
            return best_a
        return best_a

    def generate_perturb(arr, sigma=0.15):
        arr = normalize(arr)
        n = arr.size
        kind = int(rng.integers(0, 5))
        if kind == 0:
            a2 = arr * np.exp(float(sigma) * rng.standard_normal(n))
        elif kind == 1:
            a2 = arr * np.exp(2.0 * float(sigma) * rng.standard_normal(n))
            a2 = np.maximum(a2, 0.0)
        elif kind == 2:
            scale = max(1.0 / max(1, n), float(arr.mean()))
            a2 = arr + float(sigma) * scale * rng.standard_normal(n)
        elif kind == 3:
            rnd = np.exp(rng.normal(0.0, float(sigma), n))
            a2 = 0.5 * arr + 0.5 * rnd
        else:
            base = arr.copy()
            k = max(1, n // 10)
            idx = rng.choice(n, size=k, replace=False)
            base[idx] *= 0.0
            base += 1e-6 + 0.05 * rng.uniform(0.0, 1.0, n)
            a2 = 0.5 * arr + 0.5 * base
        return proj_simplex(a2)

    def run_phase(func, curr, alloc, *args):
        if now() >= deadline - 45.0:
            return curr
        stop = min(deadline - 40.0, now() + float(alloc))
        if stop <= now() + 1.0:
            return curr
        try:
            out = func(curr, stop, *args)
        except Exception:
            return curr
        if out is None:
            return curr
        out = normalize(out)
        try:
            sc = score_fast(out)
        except Exception:
            return curr
        if math.isfinite(sc):
            update_best(out, sc, force=True)
        return out

    # Load optional parent directly, without copying its literal values.
    parent = None
    try:
        h = height_sequence_1
        if h is not None:
            p = to_arr(np.asarray(h, dtype=np.float64).ravel())
            if p is not None:
                if p.size > 1200:
                    p = interp_len(p, 1024)
                parent = normalize(p)
                update_best(parent, score_fast(parent), force=True)
    except Exception:
        parent = None

    if best_arr is None or parent is None:
        n_def = 320
        a_def = np.ones(n_def, dtype=np.float64)
        a_def = normalize(a_def)
        update_best(a_def, score_fast(a_def), force=True)

    n_base = int(best_arr.size if best_arr is not None else 320)
    if n_base < 8:
        n_base = 32
        a_tmp = np.ones(n_base, dtype=np.float64)
        a_tmp = normalize(a_tmp)
        update_best(a_tmp, score_fast(a_tmp), force=True)
        best_arr = a_tmp
        n_base = int(best_arr.size)

    a_start = normalize(best_arr)
    best_init_score = score_fast(a_start)

    # A few cheap alternative initializations in the same length family.
    init_list = [a_start]
    xgrid = np.linspace(0.0, 1.0, n_base)
    for g in (0.5, 1.0, 2.0, 4.0):
        init_list.append(np.exp(g * xgrid))
        init_list.append(np.exp(-g * xgrid))
        init_list.append(np.exp(-4.0 * (xgrid - 0.2) ** 2))
        init_list.append(np.exp(-4.0 * (xgrid - 0.8) ** 2))
        init_list.append(np.exp(-6.0 * (xgrid - 0.5) ** 2))
    for _ in range(4):
        init_list.append(np.exp(rng.normal(0.0, 0.5, n_base)))

    for ain in init_list:
        an = normalize(ain)
        try:
            sc = score_fast(an)
        except Exception:
            continue
        if math.isfinite(sc):
            if sc < best_init_score - 1e-12:
                best_init_score = sc
                a_start = an.copy()
            update_best(an, sc, force=False)

    a_cur = a_start.copy()

    # Main optimizations at the primary length.
    if HAVE_LP:
        a_cur = run_phase(soft_local, a_cur, min(180.0, horizon * 0.12), 20000)
        a_cur = run_phase(linlp_local, a_cur, min(420.0, horizon * 0.30), 700)
        a_cur = run_phase(soft_local, a_cur, min(220.0, horizon * 0.16), 30000)
        a_cur = run_phase(cross_lp_local, a_cur, min(130.0, horizon * 0.09), 100)
        a_cur = run_phase(linlp_local, a_cur, min(150.0, horizon * 0.11), 350)
    else:
        a_cur = run_phase(soft_local, a_cur, min(560.0, horizon * 0.42), 120000)
        a_cur = run_phase(soft_local, a_cur, min(300.0, horizon * 0.22), 80000)

    update_best(a_cur, score_fast(a_cur), force=True)

    # Explore nearby and related lengths.
    if now() < deadline - 70.0:
        base_len = int(best_arr.size if best_arr is not None else a_cur.size)
        cand_lengths = set()
        for off in (-160, -128, -96, -72, -64, -48, -40, -32, -24, -16,
                    -12, -8, -4, -2, 2, 4, 8, 12, 16, 24, 32, 40, 48,
                    64, 72, 96, 128, 160, 256):
            L = base_len + off
            if 96 <= L <= 1024:
                cand_lengths.add(int(L))
        for L in (96, 128, 160, 192, 224, 256, 320, 384, 448,
                  512, 640, 768, 896, 1024):
            if 96 <= L <= 1024:
                cand_lengths.add(L)

        cand_lengths = [L for L in cand_lengths if L != base_len and 96 <= L <= 1024]
        cand_lengths = sorted(set(cand_lengths), key=lambda L: (abs(L - base_len), L))
        cand_lengths = cand_lengths[:14]

        for L in cand_lengths:
            if now() >= deadline - 60.0:
                break
            rem = deadline - 65.0 - now()
            if rem < 18.0:
                break
            left = max(1, len(cand_lengths))
            alloc = min(22.0, max(8.0, rem / max(1, left) / 0.8))
            alloc = min(alloc, max(6.0, rem * 0.45))
            if alloc < 6.0:
                break

            init_a = interp_len(best_arr if best_arr is not None else a_cur, L)
            if HAVE_LP and L <= 800:
                stop1 = min(deadline - 55.0, now() + alloc * 0.45)
                a_l = linlp_local(init_a, stop1, max(40, int(alloc * 2)))
                stop2 = min(deadline - 50.0, now() + alloc * 0.55)
                a_l = soft_local(a_l, stop2, max(1500, int(alloc * 60)))
            else:
                stop1 = min(deadline - 50.0, now() + alloc)
                a_l = soft_local(init_a, stop1, max(1500, int(alloc * 60)))

            if a_l is not None:
                a_l = normalize(a_l)
                sc_l = score_fast(a_l)
                if math.isfinite(sc_l):
                    update_best(a_l, sc_l, force=True)

    # Restart-like local searches around the current best.
    if now() < deadline - 55.0:
        a_base = normalize(best_arr if best_arr is not None else a_cur)
        n_base_r = int(a_base.size)
        for r in range(8):
            if now() >= deadline - 50.0:
                break
            rem = deadline - 55.0 - now()
            if rem < 15.0:
                break
            alloc = min(22.0, max(8.0, rem / max(1, 8 - r) / 0.8))
            init_a = generate_perturb(a_base, sigma=0.06 + 0.045 * r)
            if HAVE_LP and n_base_r <= 800:
                stop1 = min(deadline - 50.0, now() + alloc * 0.45)
                a_r = linlp_local(init_a, stop1, max(30, int(alloc * 2)))
                stop2 = min(deadline - 45.0, now() + alloc * 0.55)
                a_r = soft_local(a_r, stop2, max(1200, int(alloc * 50)))
            else:
                stop1 = min(deadline - 45.0, now() + alloc)
                a_r = soft_local(init_a, stop1, max(1200, int(alloc * 50)))
            if a_r is not None:
                a_r = normalize(a_r)
                sc_r = score_fast(a_r)
                if math.isfinite(sc_r):
                    update_best(a_r, sc_r, force=True)

    # Final polish on the best exact candidate found so far.
    if best_arr is not None and now() < deadline - 18.0:
        a_fin = normalize(best_arr)
        stop_soft = min(deadline - 12.0, now() + min(40.0, horizon * 0.03))
        a_fin = soft_local(a_fin, stop_soft, 25000)
        update_best(a_fin, score_fast(a_fin), force=True)

        if HAVE_LP and len(a_fin) <= 800 and now() < deadline - 10.0:
            stop_lin = min(deadline - 8.0, now() + min(20.0, horizon * 0.015))
            a_fin = linlp_local(a_fin, stop_lin, 120)
            update_best(a_fin, score_fast(a_fin), force=True)

    # Fallback if nothing valid was stored.
    if best_arr is None:
        fb = np.ones(96, dtype=np.float64)
        fb = normalize(fb)
        update_best(fb, score_fast(fb), force=True)

    final = best_arr if best_arr is not None else np.ones(96, dtype=np.float64)
    final = to_arr(final)
    if final is None or final.size == 0 or float(final.sum()) < 0.01:
        final = np.ones(96, dtype=np.float64)

    # Final exact validation.
    try:
        final_v = evaluate_sequence([float(x) for x in final])
        if not math.isfinite(final_v):
            final = np.ones(96, dtype=np.float64)
    except Exception:
        final = np.ones(96, dtype=np.float64)

    # Ensure finite, nonnegative list.
    final = np.clip(np.asarray(final, dtype=np.float64).ravel(), 0.0, 1000.0)
    if final.size == 0:
        final = np.ones(96, dtype=np.float64)
    try:
        if float(final.sum()) < 0.01:
            final = np.ones(96, dtype=np.float64)
    except Exception:
        final = np.ones(96, dtype=np.float64)

    return [float(x) for x in final]
