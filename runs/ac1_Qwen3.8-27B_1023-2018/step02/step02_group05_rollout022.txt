import numpy as np
import time
import math

try:
    from scipy.optimize import linprog
    import scipy.sparse as sps
except Exception:
    linprog = None
    sps = None

TARGET = 1.50274349

def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.clip(a, 0.0, 1000.0)
    n = a.size
    if n == 0:
        return float('inf')
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    b = np.convolve(a, a)
    mb = float(b.max())
    if not math.isfinite(mb):
        return float('inf')
    return 2.0 * n * mb / (s * s)

def _update_best(seq, best):
    try:
        arr = np.asarray(seq, dtype=np.float64).ravel()
        arr = np.nan_to_num(arr, nan=0.0, posinf=0.0, neginf=0.0)
        arr = np.clip(arr, 0.0, 1000.0)
        if arr.size == 0:
            return
        s = float(arr.sum())
        if s < 0.02:
            return
        sc = evaluate_sequence([float(x) for x in arr])
        if math.isfinite(sc) and sc < best[0]:
            best[0] = float(sc)
            best[1] = arr.copy()
    except Exception:
        pass

def _build_conv_matrix(a):
    n = a.size
    # Column j has nonzeros at rows j to min(j+n-1, 2n-2), all with value a[j]
    row_parts = []
    col_parts = []
    val_parts = []
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        cnt = k_end - k_start + 1
        row_parts.append((k_start, cnt))
        col_parts.append(np.full(cnt, j, dtype=np.int32))
        val_parts.append(np.full(cnt, a[j], dtype=np.float64))
    rows = np.concatenate([np.arange(rs, rs+cnt, dtype=np.int32) for rs, cnt in row_parts])
    cols = np.concatenate(col_parts)
    vals = np.concatenate(val_parts)
    return sps.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))

def _lp_direction(a, M, time_limit=10):
    n = a.size
    if linprog is None or sps is None:
        return None
    A = _build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M + 1e-12)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, None)]*n,
                      method='highs',
                      options={'presolve': True, 'time_limit': time_limit, 'mip_rel_gap': 0.1})
    except Exception:
        return None
    if res is not None and getattr(res, 'x', None) is not None:
        g = np.asarray(res.x, dtype=np.float64).ravel()
        g = np.clip(g, 0.0, None)
        sg = float(g.sum())
        if sg > 1e-15 and math.isfinite(sg):
            return g / sg
    return None
