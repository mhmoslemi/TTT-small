Rescale g_0 to match the normalization, g = sqrt(2n)/sum(b) * g_0, and update f <- (1-t)f + t*g. Repeating this step produces a sequence with nonincreasing ||f*f||_∞.

So the normalization is sum(f) = sqrt(2n), not 1. But since the score is scale-invariant, this doesn't matter.

The key property: "nonincreasing ||f*f||_∞". This is guaranteed because:
- f has peak M
- g has (f*g)[k] <= M for all k
- The blend (1-t)f + t*g has convolution (1-t)²(f*f) + t(1-t)(f*g) + t²(g*g)
- At the peak of f*f, the blend's convolution is (1-t)²M + t(1-t)(f*g)[k*] + t²(g*g)[k*]
- Since (f*g)[k*] <= M, the first two terms give (1-t)²M + t(1-t)*M = M(1-t)(1-t+1-t)... wait, that's not right.

((1-t)f + t*g) * ((1-t)f + t*g) = (1-t)²(f*f) + 2t(1-t)(f*g) + t²(g*g)

At the peak location k* of f*f:
= (1-t)² * M + 2t(1-t) * (f*g)[k*] + t² * (g*g)[k*]
<= (1-t)² * M + 2t(1-t) * M + t² * max(g*g)
= M * (1-t)² * (1 + 2t/(1-t)) + t² * max(g*g)

Actually, the claim is that ||f_new * f_new||_∞ <= M where f_new = (1-t)f + t*g and M = ||f*f||_∞. Let me verify:

||f_new * f_new||_∞ = max_k [(1-t)²(f*f)[k] + 2t(1-t)(f*g)[k] + t²(g*g)[k]]

For the first two terms: (1-t)²(f*f)[k] + 2t(1-t)(f*g)[k] <= (1-t)² M + 2t(1-t) M = M(1-t)(1-t + 2t) = M(1-t)(1+t)

```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(1.0, float(budget_s))
    deadline = t0 + budget - 5.0

    # --- Initialize ---
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
    except Exception:
        parent = np.ones(192, dtype=np.float64)
    parent = np.clip(parent, 0.0, 1000.0)
    if parent.size < 2 or not np.isfinite(parent).all() or parent.sum() < 0.01:
        parent = np.ones(192, dtype=np.float64)

    n = int(parent.size)
    f = parent.astype(np.float64).copy()
    f /= f.sum()

    # --- FFT convolution ---
    fsize = next_fast_len(2 * n - 1)

    def conv(a):
        F = rfft(a, fsize)
        c = irfft(F * F, fsize)[:2 * n - 1]
        c[~np.isfinite(c)] = 0.0
        return c

    def pk(c):
        return float(c.max())

    # --- Score ---
    def score_of(a):
        return 2.0 * n * pk(conv(a))

    # --- Build B matrix: (f*g)[k] = sum_j B[k,j]*g[j] ---
    def build_B(fc):
        B = np.zeros((2 * n - 1, n), dtype=np.float64)
        for j in range(n):
            B[j:j + n, j] = fc
        return B

    # --- LP: max sum(g) s.t. B@g <= target, g >= 0 ---
    def lp_max_sum(fc, target):
        B = build_B(fc)
        try:
            res = linprog(-np.ones(n), A_ub=B, b_ub=np.full(2 * n - 1, target),
                          bounds=[(0.0, None)] * n, method='highs',
                          options={'presolve': True, 'time_limit': 15.0})
        except Exception:
            return None
        if not res.success or res.x is None:
            return None
        g = np.maximum(np.asarray(res.x, dtype=np.float64), 0.0)
        gs = g.sum()
        if gs < 1e-12:
            return None
        return g / gs

    # --- Line search ---
    def line_search(fc, g, M):
        best_t, best_m = 0.0, M
        for t in (1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05, 0.02, 0.01):
            h = (1.0 - t) * fc + t * g
            mh = pk(conv(h))
            if mh < best_m - 1e-15:
                best_m, best_t = mh, t
        return best_t, best_m

    # --- LP descent loop ---
    def lp_descent(fc, max_iter=300, alpha=1.0):
        fc = fc.copy()
        for _ in range(max_iter):
            if time.monotonic() > deadline - 60.0:
                break
            M = pk(conv(fc))
            target = alpha * M
            g = lp_max_sum(fc, target)
            if g is None:
                break
            if np.linalg.norm(g - fc, 1) < 1e-9:
                break
            bt, bm = line_search(fc, g, M)
            if bt == 0.0:
                break
            fc = (1.0 - bt) * fc + bt * g
            fc = np.maximum(fc, 0.0)
            fc /= fc.sum()
        return fc

    # --- Track best ---
    best_score = score_of(f)
    best_f = f.copy()

    def try_update(fc):
        nonlocal best_score, best_f
        s = score_of(fc)
        if s < best_score - 1e-14:
            best_score = s
            best_f = fc.copy()

    # === Phase 1: LP descent from parent ===
    f1 = lp_descent(f, max_iter=400, alpha=1.0)
    try_update(f1)

    # === Phase 2: LP descent with tightened constraint ===
    if time.monotonic() < deadline - 180.0:
        f2 = best_f.copy()
        for alpha in (0.999, 0.998, 0.995, 0.99):
            if time.monotonic() > deadline - 180.0:
                break
            f2 = lp_descent(f2, max_iter=100, alpha=alpha)
            try_update(f2)

    # === Phase 3: Template initializations ===
    if time.monotonic() < deadline - 120.0:
        templates = []
        # Cosine
        j = np.arange(n)
        templates.append(np.cos(np.pi * j / (n - 1)) ** 2)
        # Sine
        templates.append(np.sin(np.pi * (j + 1) / (n + 1)) ** 2)
        # Triangular
        templates.append(np.maximum(0, 1.0 - np.abs(2.0 * j / (n - 1) - 1.0)))
        # Gaussian (narrow)
        templates.append(np.exp(-((j - n/2) ** 2) / (2.0 * (n/4) ** 2)))
        # Gaussian (wide)
        templates.append(np.exp(-((j - n/2) ** 2) / (2.0 * (n/3) ** 2)))
        # Plateau
        w = n // 3
        t_plateau = np.zeros(n)
        t_plateau[w:n-w] = 1.0
        templates.append(t_plateau)
        # Parent reversed (if not symmetric)
        if np.linalg.norm(f - f[::-1]) > 1e-6:
            templates.append(f[::-1])

        for tmpl in templates:
            if time.monotonic() > deadline - 120.0:
                break
            tmpl = np.maximum(tmpl, 0.0)
            if tmpl.sum() < 1e-8:
                continue
            tmpl = tmpl / tmpl.sum()
            # Blend with current best to keep some structure
            a = 0.5 * tmpl + 0.5 * best_f
            a = a / a.sum()
            a = lp_descent(a, max_iter=150, alpha=1.0)
            try_update(a)

    # === Phase 4: Stochastic perturbation + re-optimize ===
    rng = np.random.default_rng(seed)
    a = best_f.copy()
    for round_num in range(30):
        if time.monotonic() > deadline - 60.0:
            break
        # Perturb
        r = rng.random()
        if r < 0.25:
            # Gaussian noise
            a_p = a + rng.normal(0, 0.03, n)
        elif r < 0.45:
            # Block scale
            a_p = a.copy()
            k = int(rng.integers(0, n))
            w = max(2, n // 24)
            lo, hi = max(0, k - w), min(n, k + w)
            a_p[lo:hi] *= abs(np.exp(rng.normal(0, 0.12)))
        elif r < 0.60:
            # Blend with reversed
            a_p = (1 - 0.15 * rng.random()) * a + 0.15 * rng.random() * a[::-1]
        elif r < 0.75:
            # Endpoint modification
            a_p = a.copy()
            w = max(1, n // 20)
            a_p[:w] *= np.exp(rng.normal(0, 0.15))
            a_p[-w:] *= np.exp(rng.normal(0, 0.15))
        else:
            # Two-point swap
            a_p = a.copy()
            i, j = int(rng.integers(0, n)), int(rng.integers(0, n))
            if i != j:
                amt = 0.01 * a.max()
                a_p[i] += amt
                a_p[j] -= amt
        a_p = np.maximum(a_p, 0.0)
        if a_p.sum() < 1e-8:
            continue
        a_p /= a_p.sum()
        # Re-optimize
        a_opt = lp_descent(a_p, max_iter=80, alpha=1.0)
        try_update(a_opt)
        if score_of(a_opt) < score_of(a) + 1e-14:
            a = a_opt.copy()

    # === Phase 5: Gradient refinement ===
    if time.monotonic() < deadline - 30.0:
        a = best_f.copy()
        lr = 0.1
        for it in range(3000):
            if time.monotonic() > deadline - 30.0:
                break
            c = conv(a)
            M = pk(c)
            s = 2.0 * n * M
            
            # Soft-LSE subgradient
            tau = max(1e-10, 0.02 * M)
            bm = float(np.max(c))
            x = np.clip((c - bm) / tau, -50.0, 50.0)
            ex = np.exp(x)
            s_ex = ex.sum()
            p = ex / s_ex

            # h[j] = sum_k p[k] * a[max(0, min(n-1, k-j))]
            # h[j] = sum_m p[m+j] * a[m] for m where 0<=m<n and 0<=m+j<2n-1
            h = np.zeros(n)
            for j in range(n):
                h[j] = np.dot(p[j:j+n], a)

            # Projected gradient step
            hf = np.dot(h, a)
            step = h - hf
            sn = np.linalg.norm(step)
            if sn < 1e-14:
                break
            
            improved = False
            for scale in (lr, lr*0.5, lr*0.25, lr*0.1):
                cand = a - scale * step / sn
                cand = np.maximum(cand, 1e-12)
                cs = cand.sum()
                if cs < 1e-8:
                    continue
                cand /= cs
                cs_score = 2.0 * n * pk(conv(cand))
                if cs_score < s - 1e-14:
                    a = cand
                    s = cs_score
                    improved = True
                    break
            
            if improved:
                try_update(a)
                lr = min(lr * 1.05, 2.0)
            else:
                lr = max(0.01, lr * 0.85)
                if lr < 0.02:
                    break

    # === Final validation ===
    best_f = np.clip(best_f, 0.0, 1000.0)
    if best_f.sum() < 0.01 or not np.isfinite(best_f).all():
        best_f = np.ones(n) / n

    result = [float(x) for x in best_f]
    try:
        fs = float(evaluate_sequence(result))
    except Exception:
        fs = float('inf')

    if not np.isfinite(fs):
        result = [1.0 / n] * n
        try:
            fs = float(evaluate_sequence(result))
        except Exception:
            return [1.0]

    return result
```
