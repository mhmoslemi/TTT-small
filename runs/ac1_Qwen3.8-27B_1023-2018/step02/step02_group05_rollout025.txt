import time
import math
from scipy.optimize import linprog

def _score(a):
    """Compute the exact score."""
    a = np.clip(a, 0, 1000)
    n = len(a)
    if n == 0:
        return float('inf')
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = np.convolve(a, a)
    mb = b.max()
    return 2.0 * n * mb / (s * s)

def _alt_lp(a, time_limit=10):
    """Alternating LP: fix b=a, find new a minimizing max(a*b)."""
    n = len(a)
    b = np.convolve(a, a)
    # Convolution matrix: A[k,j] = b[k-j] if 0<=k-j<n else 0
    # We want: min t s.t. A @ a_new + t... 
    # Actually: min t s.t. (conv_matrix(b)) @ new_a <= t, new_a >= 0, sum(new_a) = 1
    
    # Build constraint matrix
    # A_ub[k, j] = b[k-j] if 0 <= k-j < n
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A_ub[k, j] = b[idx]
    
    # Objective: minimize t (last variable)
    # Variables: [a_0, ..., a_{n-1}, t]
    c = np.zeros(n + 1)
    c[-1] = 1  # minimize t
    
    # A_ub @ [a; t] <= 0 → A_conv @ a - t <= 0
    A_full = np.column_stack([A_ub, -np.ones(2*n-1)])
    b_ub = np.zeros(2*n-1)
    
    # sum(a) = 1 → A_eq @ a = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_full, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success and res.x is not None:
            new_a = np.clip(res.x[:n], 0, None)
            s = new_a.sum()
            if s > 1e-15:
                new_a /= s
            return new_a
    except Exception:
        pass
    return None

Let me also think about whether I should use the "original" LP form (max sum(g) s.t. conv(a,g) <= M) or the "alternating" LP (min t s.t. conv(b,a_new) <= t).

The original LP: Given a and M = max(conv(a,a)), find g maximizing sum(g) subject to conv(a,g) <= M, g >= 0. Then blend a with normalized g.

The alternating LP: Given a, let b = conv(a,a). Find a_new minimizing max_lag conv(a_new, b) subject to a_new >= 0, sum(a_new) = 1. Then blend.

The alternating LP directly targets the objective, so it should be better. But the original LP might have better properties (it's a "polar" type step).

I'll implement both and use whichever works better.

def _alt_lp_step(a, time_limit=20):
    """One alternating LP step: min max_k (a_new * b)_k s.t. a_new>=0, sum=1."""
    n = len(a)
    b = np.convolve(a, a)
    # Build A: A[k,j] = b[k-j] if 0<=k-j<n
    A = np.zeros((2*n-1, n))

        jhi = min(k, n-1)
        if jlo <= jhi:
            A[k, jlo:jhi+1] = b[k-jhi:k-jlo+1]

    c = np.zeros(n+1); c[n] = 1.0

    b_ub = np.zeros(2*n-1)
    A_eq = np.array([[1.0]*n + [0.0]])
    b_eq = np.array([1.0])
    bounds = [(0, None)]*n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs',
                     options={'presolve': True, 'time_limit': time_limit})
