- Constraints: `(convolve(a, g))[k] <= M` for `k = 0, ..., 2n-2`

    # C is (2n-1) x n
    # C[k, j] = a[k-j] if 0 <= k-j < n else 0
    # Column j: nonzero at rows j, j+1, ..., j+n-1 with values a[0], a[1], ..., a[n-1]
    col_idx = []
    row_idx = []
    vals = []

        rows = np.arange(j, j+n)
        col_idx.extend([j]*n)
        row_idx.extend(rows.tolist())
        vals.extend(a.tolist())
    A = sps.csr_matrix((vals, (row_idx, col_idx)), shape=(2*n-1, n))
    return A

One issue with the LP approach is that it can get stuck at non-optimal points. The LP gives the best "direction" given the current `a`, but the blending step might not be optimal.

Let me think about what the LP is actually doing. At each step:
- We have current `a` with `M = max(convolve(a,a))`
- We solve for `g` that maximizes `sum(g)` subject to `convolve(a,g) <= M, g >= 0`
- We update `a <- (1-t)*a + t*g`

The key property: after the update, `convolve(a_new, a_new)` should have a smaller max. This is because `g` is "flatter" than `a` (it maximizes its sum while keeping convolution with `a` bounded).

The theoretically optimal step size would be found by line search, but in practice, fixed or adaptive step sizes work.

**Improvement idea**: Instead of a single LP iteration followed by blending, do multiple LP iterations with the SAME `a` but different `M` values (progressively tighter). Or better: after finding `g`, compute the exact `t` that minimizes the new objective.

Actually, the cleanest approach is:
1. Compute `b = convolve(a, a)`, `M = max(b)`
2. Solve LP to get `g` (normalized to sum 1)
3. For a grid of `t` values, compute `score((1-t)*a + t*g)` and pick the best
4. Repeat

            # For this k: grad += 2 * a[k-i] for i where 0 <= k-i < n
            # i.e., i in [max(0, k-n+1), min(k, n-1)]
            i_start = max(0, k - n + 1)
            i_end = min(k, n - 1)
            if i_end >= i_start:
                grad[i_start:i_end + 1] += 2.0 * a[k - i_end:k - i_start + 1][::-1]
        grad /= len(active)

        # Normalize and step
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break

        # Try different step sizes
        improved = False
        for a_try in (alpha, alpha * 0.5, alpha * 2.0, alpha * 0.25):
            step = a_try * grad / gnorm
            # Scale step relative to the magnitude of a
            scale = float(a.mean()) + 1e-8
            step = step * scale
            cand = a - step
            cand = np.clip(cand, 0.0, None)
            sn = float(cand.sum())
            if sn > 1e-15:
                cand = cand / sn
                sc = _score(cand)
                if math.isfinite(sc) and sc < local:
                    a = cand
                    local = sc
                    _update_best(a, best)
                    stale = 0
                    alpha = min(alpha * 1.05, 0.5)
                    improved = True
                    break

        if not improved:
            stale += 1
            alpha *= 0.85
            if alpha < 0.001:
                # Random perturbation
                pa = a + 0.05 * rng.standard_normal(n)
                pa = np.clip(pa, 0.0, None)
                psn = float(pa.sum())
                if psn > 1e-15:
                    pa = pa / psn
                    ps = _score(pa)
                    if math.isfinite(ps) and ps < local:
                        a = pa
                        local = ps
                        _update_best(a, best)
                alpha = 0.05
            if stale > 50:
                break

    return a, local


def _coordinate_descent(a, time_limit, rng, best):
    """Cycle through coordinates, optimize each one with 1D line search."""
    t0 = time.monotonic()
    deadline = t0 + time_limit - 2.0
    n = a.size

    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if s > 1e-15:
        a = a / s
    local = _score(a)
    _update_best(a, best)

    order = list(range(n))
    stale = 0

    for it in range(2000):
        if time.monotonic() >= deadline:
            break
        if local < TARGET and it > 5:
            break

        improved_any = False
        for i in order:
            if time.monotonic() >= deadline:
                break
            # Try perturbing coordinate i
            for delta in (-0.05, -0.02, 0.02, 0.05):
                cand = a.copy()
                cand[i] = max(0.0, cand[i] + delta * a.max())
                sn = float(cand.sum())
                if sn > 0.01:
                    sc = _score(cand)
                    if math.isfinite(sc) and sc < local:
                        a = cand
                        local = sc
                        _update_best(a, best)
                        improved_any = True
                        break

        if improved_any:
            stale = 0
            # Shuffle order
            rng.permutation(order)
        else:
            stale += 1
            if stale > 5:
                break

    return a, local


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    try:
        remaining = max(0.0, float(budget_s))
    except Exception:
        remaining = 1300.0
    deadline = t_start + remaining - 25.0

    best = [float('inf'), np.array([1.0, 1.0], dtype=np.float64)]
    _update_best(best[1], best)

    # Get parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            parent = np.asarray(h, dtype=np.float64).ravel()
            parent = np.nan_to_num(parent, nan=0.0, posinf=0.0, neginf=0.0)
            parent = np.clip(parent, 0.0, 1000.0)
            if parent.size > 1:
                _update_best(parent, best)
    except Exception:
        parent = None

    def _rem():
        return max(0.0, deadline - time.monotonic())

    def _resample(arr, n_new):
        if arr.size == n_new:
            return arr.copy()
        if arr.size > 1:
            xo = np.linspace(0.0, 1.0, arr.size)
            xn = np.linspace(0.0, 1.0, n_new)
            return np.interp(xn, xo, arr)
        return np.ones(n_new) / n_new

    # Phase 1: Heavy LP on parent length
    if parent is not None and parent.size >= 4:
        np_ = parent.size
        alloc = min(500.0, _rem() * 0.45)
        if alloc > 30:
            a, _ = _lp_iterate(parent, np_, alloc, rng, best)
            # Subgradient polish
            if _rem() > 30:
                a, _ = _subgradient_polish(a, min(60.0, _rem() * 0.3), rng, best)

    # Phase 2: LP at various sizes
    sizes_to_try = [96, 128, 160, 192, 256, 320, 384]
    if parent is None:
        sizes_to_try = [64, 96, 128, 160, 192, 256, 320]

    for ntr in sizes_to_try:
        if _rem() < 60:
            break
        if parent is not None:
            base = _resample(parent, ntr)
        else:
            base = np.ones(ntr)

        # Create a few variants
        variants = [base]
        x = np.linspace(0.0, 1.0, ntr)
        variants.append(base * (1.0 + 0.5 * x))
        variants.append(base * (1.0 - 0.3 * x))

        for v in variants:
            if _rem() < 40:
                break
            a = np.clip(v, 0.0, None)
            sv = float(a.sum())
            if sv < 1e-15:
                a = np.ones(ntr) / ntr
            else:
                a = a / sv
            sc = _score(a)
            _update_best(a, best)

            alloc = min(80.0, _rem() * 0.2)
            if alloc > 15:
                a, _ = _lp_iterate(a, ntr, alloc, rng, best)

    # Phase 3: Best sequence refinement
    if _rem() > 30:
        best_arr = np.asarray(best[1], dtype=np.float64).ravel()
        nb = best_arr.size
        if nb < 4:
            nb = int(parent.size) if parent is not None and parent.size > 4 else 96
            best_arr = _resample(np.ones(nb), nb)

        # LP polish
        alloc = min(120.0, _rem() * 0.3)
        if alloc > 15:
            a, _ = _lp_iterate(best_arr, nb, alloc, rng, best)

        # Subgradient polish
        if _rem() > 20:
            best_arr = np.asarray(best[1], dtype=np.float64).ravel()
            alloc = min(80.0, _rem() * 0.25)
            if alloc > 10:
                a, _ = _subgradient_polish(best_arr, alloc, rng, best)

        # Coordinate descent
        if _rem() > 20:
            best_arr = np.asarray(best[1], dtype=np.float64).ravel()
            alloc = min(40.0, _rem() * 0.2)
            if alloc > 5:
                a, _ = _coordinate_descent(best_arr, alloc, rng, best)

    # Phase 4: Final LP on best
    if _rem() > 20:
        best_arr = np.asarray(best[1], dtype=np.float64).ravel()
        nb = best_arr.size
        if nb >= 4:
            alloc = min(60.0, _rem() * 0.4)
            if alloc > 10:
                _lp_iterate(best_arr, nb, alloc, rng, best)

    # Phase 5: Very final check
    if _rem() > 10:
        best_arr = np.asarray(best[1], dtype=np.float64).ravel()
        nb = best_arr.size
        if nb >= 4:
            # One more round of LP with the best
            alloc = min(30.0, _rem() * 0.3)
            if alloc > 5:
                _lp_iterate(best_arr, nb, alloc, rng, best)

    # Final validation
    final = np.asarray(best[1], dtype=np.float64).ravel()
    final = np.nan_to_num(final, nan=0.0, posinf=0.0, neginf=0.0)
    final = np.clip(final, 0.0, 1000.0)
    if final.size == 0:
        final = np.ones(16, dtype=np.float64)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(16, dtype=np.float64)

    fin_sc = evaluate_sequence([float(x) for x in final])
    if not math.isfinite(fin_sc) or fin_sc > best[0] + 0.01:
        final = np.ones(16, dtype=np.float64)
        fin_sc = evaluate_sequence([float(x) for x in final])

    return [float(x) for x in final]
