        return None
    return g


def _lp_iterate(p, stop_time, max_iters=100):
    """LP best-response iteration on normalized p."""
    n = p.size
    best_p = p.copy()
    best_M = float(_conv_square(p).max())
    stuck = 0
    
    for it in range(max_iters):
        if time.monotonic() > stop_time - 3.0:
            break
        
        b = _conv_square(p)
        M = float(b.max())
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_p = p.copy()
            stuck = 0
        else:
            stuck += 1
        
        if stuck > 150:
            break
        
        g = _lp_best_response(p, M * 1.0001, timeout_s=3.0)
        if g is None:
            break
        g /= g.sum()
        
        improved = False
        for t in (1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1):
            cand = (1 - t) * p + t * g
            cand = np.maximum(cand, 0.0)
            cs = cand.sum()
            if cs <= 0:
                continue
            cand /= cs
            cM = float(_conv_square(cand).max())
            if cM < M * (1 - 1e-13):
                p = cand
                improved = True
                break
    
    return best_p


def _optimize_one(p0, stop_time):
    """Full optimization pipeline for one initialization."""
    remaining = stop_time - time.monotonic()
    if remaining < 10:
        return p0
    
    # Phase 1: Gradient descent (fast, many iterations)
    t1 = time.monotonic() + remaining * 0.35
    p1 = _grad_descent(p0, t1)
    
    # Phase 2: LP iterations (slower, better global moves)
    t2 = time.monotonic() + min(remaining * 0.35, (stop_time - time.monotonic()))
    p2 = _lp_iterate(p1, t2)
    
    # Phase 3: More gradient descent on LP result
    p3 = _grad_descent(p2, stop_time - 0.5)
    
    # Return best of all phases
    candidates = [p1, p2, p3]
    scores = [float(_conv_square(c).max()) for c in candidates]
    best_idx = int(np.argmin(scores))
    return candidates[best_idx]


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = float(budget_s)
    deadline = t0 + budget - 15.0
    
    parent = _get_parent()
    n_parent = parent.size
    
    # Validate parent
    best_seq = parent.copy().astype(np.float64)
    try:
        best_score = float(evaluate_sequence([float(x) for x in parent]))
    except Exception:
        best_score = float('inf')
    if not np.isfinite(best_score):
        best_seq = np.ones(32, dtype=np.float64) / 32.0
        best_score = 2.0
    
    rng = np.random.default_rng(seed)
    
    # Normalize parent
    p = np.clip(parent, 0.0, 1000.0).astype(np.float64)
    ps = p.sum()
    if ps <= 0:
        p = np.ones(n_parent) / n_parent
        ps = 1.0
    p_norm = p / ps
    
    # Generate diverse initializations
    inits = [p_norm]
    
    # Small perturbation
    noisy = p_norm * np.exp(rng.normal(0, 0.1, n_parent))
    noisy = np.maximum(noisy, 1e-8)
    inits.append(noisy / noisy.sum())
    
    # Gaussian smoothing at different scales
    for sigma in [1.0, 2.5, 4.0, 7.0]:
        sm = gaussian_filter1d(p_norm, sigma=sigma)
        sm = np.maximum(sm, 1e-8)
        inits.append(sm / sm.sum())
    
    # Flipped
    inits.append(p_norm[::-1] / p_norm[::-1].sum())
    
    # Symmetrized
    sym = 0.5 * (p_norm + p_norm[::-1])
    inits.append(sym / sym.sum())
    
    # Strongly perturbed
    strong_noisy = p_norm * np.exp(rng.normal(0, 0.4, n_parent))
    strong_noisy = np.maximum(strong_noisy, 1e-8)
    inits.append(strong_noisy / strong_noisy.sum())
    
    # Allocate time: reserve 80s for final polish
    n_inits = len(inits)
    avail = max(60.0, (deadline - t0) - 80)
    per_init = max(15.0, avail / n_inits)
    
    for idx, init in enumerate(inits):
        if time.monotonic() > deadline - 80:
            break
        alloc_end = min(t0 + (idx + 1) * per_init + 5, deadline - 80)
        if time.monotonic() > alloc_end - 3:
            continue
        try:
            result = _optimize_one(init, alloc_end)
            sc = _score(result)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = result.copy()
        except Exception:
            pass
    
    # Final extended polish on the best solution
    if time.monotonic() < deadline - 5:
        try:
            polished = _optimize_one(best_seq, deadline - 3.0)
            sc = _score(polished)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = polished.copy()
        except Exception:
            pass
    
    # Final validation with exact evaluator
    try:
        final_score = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_score = float('inf')
    
    if not np.isfinite(final_score):
        return [1.0] * 32
    
    return [float(x) for x in best_seq]
