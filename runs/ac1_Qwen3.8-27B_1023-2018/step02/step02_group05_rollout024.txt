            bounds = [(0.0, None)] * n
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True, 'time_limit': tl})
            if res is not None and getattr(res, 'x', None) is not None:
                g = np.asarray(res.x, dtype=np.float64).ravel()
                g = np.clip(g, 0.0, None)
                sg = float(g.sum())
                if sg > 1e-15 and math.isfinite(sg):
                    return g / sg
        except Exception:
            pass
        return None

    # Subgradient of max(convolve(a,a))
    def subgrad(a, n):
        b = np.convolve(a, a)
        M = float(b.max())
        if M < 1e-30:
            return np.zeros(n)
        eps = 0.01 * M
        active = np.where(b >= M - eps)[0]
        if len(active) == 0:
            return np.zeros(n)
        grad = np.zeros(n)
        for m in active:
            m = int(m)
            lo = max(0, m - n + 1)
            hi = min(n - 1, m)
            if lo <= hi:
                idx = m - np.arange(lo, hi + 1)
                grad[lo:hi + 1] += a[idx]
        grad *= 2.0 / len(active)
        return grad

    def subgrad_step(a, n, step):
        g = subgrad(a, n)
        a_new = a - step * g
        a_new = np.clip(a_new, 0.0, None)
        s = float(a_new.sum())
        if s > 1e-15:
            a_new /= s
        else:
            a_new = a.copy()
        return a_new

    # Main Frank-Wolfe optimization
    def fw_optimize(a_init, n, time_budget, rng):
        t0 = time.monotonic()
        dl = t0 + time_budget - 2.0

        a = np.asarray(a_init, dtype=np.float64).ravel().copy()
        if a.size != n:
            if a.size > 1:
                xo = np.linspace(0.0, 1.0, a.size)
                xn = np.linspace(0.0, 1.0, n)
                a = np.interp(xn, xo, a)
            else:
                a = np.ones(n)
        a = np.clip(a, 0.0, None)
        s = float(a.sum())
        if s > 1e-15:
            a /= s
        local = score_fast(a)
        try_update(a)

        stagnant = 0
        maxstag = 35
        it = 0

        while time.monotonic() < dl:
            if local < 1.5028:
                break

            b = np.convolve(a, a)
            M = float(b.max())
            if not math.isfinite(M) or M <= 1e-30:
                break

            remaining = dl - time.monotonic()
            tl = min(5.0, max(0.8, remaining * 0.3))
            if tl < 1.0:
                break

            g = lp_solve(a, M, n, tl)
            if g is None:
                stagnant += 1
                if stagnant > 10:
                    break
                continue

            # Line search
            best_sc = local
            best_cand = None
            tbl = [(1.0 - t) * a + t * g for t in (0.05, 0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5, 0.6, 0.7, 0.8, 0.9, 1.0)]
            for cand in tbl:
                c = np.clip(cand, 0.0, None)
                sc = score_fast(c)
                if math.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_cand = c

            if best_cand is not None and best_sc < local * (1.0 - 1e-12):
                a = best_cand.copy()
                s = float(a.sum())
                if s > 1e-15:
                    a /= s
                local = best_sc
                stagnant = 0
                try_update(a)
            else:
                stagnant += 1
                # Subgradient escape
                if stagnant % 5 == 3:
                    improved = False
                    for ss in (0.005, 0.01, 0.02, 0.03, 0.05):
                        a_sg = subgrad_step(a, n, ss)
                        sc_sg = score_fast(a_sg)
                        if math.isfinite(sc_sg) and sc_sg < local * (1 - 1e-10):
                            a = a_sg
                            local = sc_sg
                            stagnant = 0
                            try_update(a)
                            improved = True
                            break
                    if not improved and stagnant % 10 == 5:
                        for sz in (0.02, 0.04, 0.08, 0.15):
                            pa = a * np.exp(sz * rng.standard_normal(n))
                            pa = np.clip(pa, 0.0, None)
                            ps = float(pa.sum())
                            if ps > 1e-15:
                                pa /= ps
                                psc = score_fast(pa)
                                if math.isfinite(psc) and psc < local:
                                    a = pa
                                    local = psc
                                    stagnant = 0
                                    try_update(a)
                                    improved = True
                                    break
                    if not improved and stagnant > maxstag:
                        break
            it += 1
        return a, local

    # Determine n
    if parent is not None and parent.size >= 4:
        n = parent.size
        a0 = parent.copy().ravel()
    else:
        n = 320
        a0 = np.ones(n)

    # Phase 1: Main LP optimization on parent
    alloc1 = min(600.0, (deadline - time.monotonic()) * 0.5)
    if alloc1 > 30:
        _, _ = fw_optimize(a0, n, alloc1, rng)

    # Phase 2: Continue from best
    if time.monotonic() < deadline - 30:
        b_arr = np.asarray(best_arr, dtype=np.float64).ravel()
        if b_arr.size == n:
            alloc2 = min(350.0, (deadline - time.monotonic()) * 0.4)
            if alloc2 > 20:
                _, _ = fw_optimize(b_arr, n, alloc2, rng)

    # Phase 3: Perturbed restarts
    for trial in range(8):
        if time.monotonic() >= deadline - 40:
            break
        sz = 0.05 + 0.05 * trial
        pa = np.asarray(best_arr, dtype=np.float64).ravel()
        pa = pa * (1.0 + sz * rng.standard_normal(pa.size))
        pa = np.clip(pa, 0.0, None)
        ps = float(pa.sum())
        if ps > 1e-15:
            pa /= ps
        alloc3 = min(60.0, (deadline - time.monotonic()) * 0.2)
        if alloc3 > 15:
            _, _ = fw_optimize(pa, pa.size, alloc3, rng)

    # Phase 4: Final polish
    if time.monotonic() < deadline - 20:
        b_arr = np.asarray(best_arr, dtype=np.float64).ravel()
        if b_arr.size == n:
            alloc4 = min(80.0, (deadline - time.monotonic()) * 0.5)
            if alloc4 > 10:
                _, _ = fw_optimize(b_arr, n, alloc4, rng)

    # Final validation
    final = np.asarray(best_arr, dtype=np.float64).ravel()
    final = np.nan_to_num(final, nan=0.0, posinf=0.0, neginf=0.0)
    final = np.clip(final, 0.0, 1000.0)
    if final.size == 0 or final.sum() < 0.01:
        final = np.ones(320, dtype=np.float64)

    fin_sc = evaluate_sequence([float(x) for x in final])
    if not math.isfinite(fin_sc):
        final = np.ones(320, dtype=np.float64)

    return [float(x) for x in final]
