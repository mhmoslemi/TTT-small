def _make_conv_matrix(f, n):
    # Column j has f[0..n-1] at rows j..j+n-1
    idx = np.arange(n, dtype=np.int64)
    rows = np.repeat(idx, n) + np.tile(idx, n)
    cols = np.tile(idx, n)
    vals = np.tile(f, n)
    return sps.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))

def _lp_solve(f, n, time_limit):

        return None

def _subgrad_step(a, n):
    """Compute subgradient of max_k (a*a)_k and return it."""
    b = np.convolve(a, a)
    M = float(b.max())
    if not math.isfinite(M) or M <= 0:
        return None, M
    # Active set with tolerance
    tol = 0.01 * M
    active = np.where(b >= M - tol)[0]
    if len(active) == 0:
        active = np.where(b == M)[0]
    # h_k = 1/|active| for k in active
    h = np.zeros(2*n-1)
    h[active] = 1.0 / len(active)
    # grad[j] = 2 * sum_k h[k] * a[k-j] for valid k-j
    # = 2 * convolve(h, a[::-1])[j + n - 1]
    conv = np.convolve(h, a[::-1])
    grad = 2.0 * conv[n-1 : n-1+n]
    return grad, M
