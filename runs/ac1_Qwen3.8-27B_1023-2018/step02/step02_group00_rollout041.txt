import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(320, dtype=np.float64)
    if p.size == 0:
        p = np.ones(320, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(320, dtype=np.float64)
    return p


def _conv_full(a):
    n = len(a)
    if n <= 64:
        c = np.convolve(a, a, mode="full")
    else:
        fsize = next_fast_len(2 * n - 1)
        F = rfft(a, fsize)
        c = irfft(F * F, fsize)[:2 * n - 1]
        np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n):
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            j = np.arange(j_lo, j_hi + 1)
            i = k - j
            A[k, j] = f[i]
    return A


def _lp_solve(f, n):
    b = _conv_full(f)
    M = float(b.max())
    if M < 1e-15:
        return None, M, b
    
    A = _build_A(f, n)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                     bounds=[(0, None)] * n,
                     method='highs',
                     options={'presolve': True, 'time_limit': 10})
    except Exception:
        return None, M, b
    
    if res.status != 0 or res.x is None:
        return None, M, b
    
    g = np.asarray(res.x, dtype=np.float64)
    g_sum = float(g.sum())
    if g_sum < 1e-15:
        return None, M, b
    
    return g / g_sum, M, b


def _lp_optimize(f0, n, stop_time, seed):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel()[:n].copy(), 0, None)
    while f.size < n:
        f = np.concatenate([f, np.zeros(n - f.size)])
    f = f[:n]
    f_sum = float(f.sum())
    if f_sum < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / f_sum
    
    best_f = f.copy()
    best_score = _score(f)
    t = 0.3
    t_min = 0.01
    stall_count = 0
    max_stall = 500
    iteration = 0
    
    while time.monotonic() < stop_time - 3.0:
        # Compute LP direction
        g_dir, M, b = _lp_solve(f, n)
        
        if g_dir is None:
            # LP failed - add noise and retry
            f = np.clip(f + rng.normal(0, 0.02, n), 0, None)
            fs = f.sum()
            if fs > 1e-10:
                f = f / fs
            stall_count += 5
            if stall_count > max_stall:
                break
            continue
        
        dist = np.linalg.norm(g_dir - f)
        if dist < 1e-8:
            stall_count += 2
            if stall_count > max_stall:
                break
            # Perturb
            f = np.clip(f + rng.normal(0, 0.05, n), 0, None)
            fs = f.sum()
            if fs > 1e-10:
                f = f / fs
            continue

        improved = False
        t_candidates = []
        t_cur = t
        for _ in range(8):
            t_candidates.append(t_cur)
            t_cur = max(t_min, t_cur * 0.6)

            fs_new = f_new.sum()
            if fs_new < 1e-10:

            sc = _score(f_new)
            if sc < best_score - 1e-12:

                best_f = f.copy()

            elif sc < _score(f) - 1e-12:

    n = len(a)
    best = np.clip(np.asarray(a, dtype=np.float64).ravel(), 0, None).copy()
    fs = best.sum()
    if fs < 1e-10:
        return best, math.inf
    best = best / fs
    best_sc = _score(best)
    rng = np.random.default_rng(seed)
    no_improve_rounds = 0
    max_no_improve = 8
    
    multipliers = (0.2, 0.4, 0.6, 0.8, 1.25, 1.5, 2.0, 3.0, 0.0)
    
    while time.monotonic() < stop_time - 3.0:
        improved = False
        
        # Coordinate descent with line search
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 3.0:
                break
            orig = best[i]
            best_val = orig
            best_val_sc = best_sc
            for mlt in multipliers:
                cand = orig * mlt if orig > 1e-12 else (0.01 if mlt > 0 else 0.0)
                best[i] = cand
                if best.sum() < 0.01:
                    best[i] = orig
                    continue
                sc = _score(best)
                if sc < best_val_sc - 1e-13:
                    best_val = cand
                    best_val_sc = sc
            best[i] = best_val
            if best_val_sc < best_sc - 1e-13:
                best_sc = best_val_sc
                improved = True

        n_pairs = min(200, n)
        for _ in range(n_pairs):

                break
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            for mi in (0.3, 0.7, 1.4, 2.5):
                for mj in (0.3, 0.7, 1.4, 2.5):
                    ci = oi * mi if oi > 1e-12 else 0.01
                    cj = oj * mj if oj > 1e-12 else 0.01
                    best[i], best[j] = ci, cj
                    if best.sum() < 0.01:
                        best[i], best[j] = oi, oj
                        continue
                    sc = _score(best)
                    if sc < best_sc - 1e-13:
                        best_sc = sc
                        improved = True
