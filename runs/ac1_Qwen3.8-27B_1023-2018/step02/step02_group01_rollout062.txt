def _proj_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n+1)
    rho = np.nonzero(u * idx > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    theta = (cssv[rho[-1]] - 1) / float(rho[-1]+1)
    return np.maximum(v - theta, 0)

def _lp_optimize(a0, stop_time, verbose=False):
    n = a0.size
    a = np.clip(a0.astype(np.float64), 0, 1000).copy()

    if s <= 0:
        a[:] = 1.0 / n
        s = 1.0
    a /= s
    
    best = a.copy()
    best_M = float(_conv_sq(a).max())
    
    rng = np.random.default_rng(0)
    no_improve = 0
    
    for it in range(10000):
        if time.monotonic() > stop_time - 1.0:
            break
        
        c = _conv_sq(a)
        M = float(c.max())
        
        if M < best_M * (1 - 1e-14):
            best_M = M
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 100:
            # Perturb
            a *= np.exp(rng.normal(0, 0.15, n))
            a = np.maximum(a, 1e-10)

            no_improve = 0
