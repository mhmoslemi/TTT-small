    return None

def _line_search_bisect(a, g, n, max_iter=20):
    """Bisection line search: find best t in [0,1]."""
    best_t = 0.0
    best_mb = float(_fft_conv_self(a).max())
    
    lo, hi = 0.0, 1.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2
        a_new = _normalize((1 - mid) * a + mid * g)
        mb = float(_fft_conv_self(a_new).max())
        if mb < best_mb:
            best_mb = mb
            best_t = mid
        # Check if we should go higher or lower
        # Since the function is not necessarily convex in t, use bracketing
        a_mid_plus = _normalize((1 - min(1.0, mid + 0.1)) * a + min(1.0, mid + 0.1) * g)
        mb_plus = float(_fft_conv_self(a_mid_plus).max())
        if mb_plus > mb:
            hi = mid + 0.1 if mid + 0.1 <= 1.0 else hi
            break
        else:
            lo = mid
    return best_t, best_mb

def _lp_optimize(n, a_init, deadline, rng, solver_list=None):
    """Main LP-based optimization loop."""
    a = _normalize(np.maximum(a_init, 0).copy())
    best_a = a.copy()
    best_score = float(_score(a))
    
    no_improve = 0
    max_no_improve = 50

        b = _fft_conv_self(a)
        M = float(b.max())
        current_score = 2.0 * n * M  # since sum(a) = 1
        
        if current_score < best_score * (1 - 1e-10):
            best_score = current_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > max_no_improve:
            # Perturb
            a = a + rng.normal(0, 0.05, n)
            a = _normalize(np.maximum(a, 0))
            no_improve = 0
            max_no_improve = int(max(30, max_no_improve * 1.15))
            continue
        
        # Solve LP
        g = _solve_lp(a, M, n, solver_list)
        
        if g is None:
            # Fallback: subgradient
            k_star = int(np.argmax(b))
            grad = np.zeros(n)
            for j in range(n):
                idx = k_star - j
                if 0 <= idx < n:
                    grad[j] = 2.0 * a[idx]
            gn = np.linalg.norm(grad)
            if gn < 1e-15:
                break
            a = a - 0.01 * grad / gn
            a = _normalize(np.maximum(a, 0))
            continue
        
        # Check similarity
        cos_sim = float(np.dot(a, g) / (np.linalg.norm(a) * np.linalg.norm(g) + 1e-30))
        if cos_sim > 0.9995:
            no_improve += 2
            a = a + rng.normal(0, 0.03, n)
            a = _normalize(np.maximum(a, 0))
            continue
        
        # Line search
        t, new_mb = _line_search_bisect(a, g, n)
        
        if new_mb < M * (1 - 1e-10):
            a = _normalize((1 - t) * a + t * g)
        else:
            no_improve += 1
    
    # Final check
    final_score = float(_score(best_a))
    if final_score < best_score:
        best_score = final_score
    
    return best_a, best_score
