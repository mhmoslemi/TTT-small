    except Exception:

def _fft_conv_self(a):
    """Compute a * a (full convolution) using FFT."""
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)

def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def _build_A(a):
    n = len(a)
    # A[k, j] = a[k-j] if 0 <= k-j < n
    # Row k: entries a[k], a[k-1], ..., a[k-n+1] (clipped to valid range)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        # For j in [lo, hi-1], A[k,j] = a[k-j]
        # k-j ranges from k-hi+1 to k-lo
        # = from k-min(n,k+1)+1 to k-max(0,k-n+1)
        # For j=lo: k-lo = k - max(0, k-n+1) = min(k, n-1)
        # For j=hi-1: k-(hi-1) = k-hi+1 = k - min(n,k+1) = max(0, k-n+1)
        A[k, lo:hi] = a[k - np.arange(hi - lo)[::-1]]
    return A

def _solve_lp_direction(a, M, n):
    """Solve LP: max sum(g) s.t. A@(g) <= M, g >= 0"""
    A = _build_A(a)
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    for solver in [cp.ECOS, cp.CVXOPT, cp.CBC, cp.GLOP, cp.GLPK]:
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                gv = g.value
                if gv is not None and np.isfinite(gv).all():
                    gv = np.maximum(gv, 0)
                    s = gv.sum()
                    if s > 1e-15:
                        return gv / s
        except Exception:
            continue
    return None

def _init_sinusoidal(n, rng):
    """Initialize with sinusoidal pattern."""
    i = np.arange(n, dtype=np.float64)
    f1 = rng.uniform(0.3, 2.0)

    phi2 = rng.uniform(0, 2*np.pi)

    amp2 = rng.uniform(0.1, 0.5)
    a = 1.0 + amp1 * np.sin(2*np.pi*f1*i/n + phi1) + amp2 * np.sin(2*np.pi*f2*i/n + phi2)
    a = np.maximum(a, 0.01)
    return a / a.sum()

def _init_cosine(n, rng):

    a = 0.5 * (1 + np.cos(2*np.pi*beta*i/n))

    return a / a.sum()

```python
def _project_simplex(x):
    n = len(x)
    if n == 0:
        return x
    # Sort x in descending order
    x_sorted = np.sort(x)[::-1]
    # Find the threshold
    cssv = np.cumsum(x_sorted) - 1.0
    # Find largest i such that x_sorted[i] + (cssv[i])/i > 0
    cond = x_sorted + cssv / np.arange(1, n+1) > 0
    rho = np.sum(cond)  # number of positive components
    theta = cssv[rho-1] / rho
    return np.maximum(x - theta, 0)
```

This is the standard algorithm. Let me verify: we want to find the closest point to x on the simplex. The projection is max(x - theta, 0) where theta is chosen so that sum(max(x-theta, 0)) = 1.
