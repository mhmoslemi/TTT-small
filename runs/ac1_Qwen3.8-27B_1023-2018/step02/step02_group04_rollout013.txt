def _conv_mat(f):
    n = f.size

def _solve_lp(f, tighten=0.0, tlimit=3):
    n = f.size
    b = np.convolve(f, f)
    M = float(b.max())

        return None

                    options={'presolve': True, 'time_limit': tlimit})

def _subgrad_step(a, deadline, best, rng):

    M = float(b[k_star])

    if S < 0.01 or M < 1e-300:
        return a
    # Subgradient of 2n*M/S^2 w.r.t a[m]
    grad = np.zeros(n)
    for m in range(n):
        g_m = a[m]  # from the a[i] * da[k*-i]/da[m] term
        j = k_star - m
        if 0 <= j < n:
            g_m += a[j]
        grad[m] = 2.0 * n * (g_m * S - 2.0 * M) / (S * S * S)
    # Projected subgradient descent
    best_a = a.copy()
