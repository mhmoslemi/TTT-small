from scipy.fft import rfft, irfft, next_fast_len

def _convolve_square(a):
    """Compute full linear convolution of a with itself."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    """Compute the exact evaluator score."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    n = a.size
    b = _convolve_square(a)
    mb = float(b.max())
    return 2.0 * n * mb / (s * s)

def _build_A(a):
    """Build constraint matrix for the LP: (a*g)[k] for k=0..2n-2."""
    n = a.size
    m = 2 * n - 1
    # A[k, i] = a[k-i] if 0 <= k-i < n, else 0
    # This is a banded matrix
    k = np.arange(m)
    i = np.arange(n)
    # Use broadcasting
    diff = k[:, None] - i[None, :]
    mask = (diff >= 0) & (diff < n)
    A = np.where(mask, a[diff.clip(0, n-1)], 0.0)
    return A

    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0.0, None)] * n, method='highs',
                  options={'time_limit': timeout_s})
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)

Let me also think about whether there's a clever trick to get below 4/3.

Consider: what if a is not monotone? What if it goes up, stays flat, then goes down? Or what if it has a "gap" in the middle?

For a "U-shape" (high at both ends, low in middle): a = [1, 1, ..., 1, 0, 0, ..., 0, 1, 1, ..., 1]
The autocorrelation would have contributions from the two blocks. The max might be at the self-correlation of one block or the cross-correlation.

Actually, for a "bump" shape (high in middle, low at ends): a = [0, 0, ..., 1, 1, ..., 1, 0, 0, ...]
This is like a truncated uniform, which gives a trapezoidal autocorrelation. The max is at the center, and the score would be close to 2 (worse than the ramp).

I think for nonnegative sequences, the monotone (ramp) shape is near-optimal because it maximally "spreads out" the autocorrelation. A non-monotone shape would concentrate the autocorrelation at certain lags.

Let me just go with the ramp and LP refinement.

import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, next_fast_len

def _convolve_square(a):
    n = a.size
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    n = a.size
    b = _convolve_square(a)
    mb = float(b.max())
    return 2.0 * n * mb / (s * s)

def _build_A(a):
    n = a.size
    m = 2 * n - 1
    k = np.arange(m, dtype=np.int64)
    i = np.arange(n, dtype=np.int64)
    diff = k[:, None] - i[None, :]
    mask = (diff >= 0) & (diff < n)
    A = np.where(mask, a[np.clip(diff, 0, n-1)], 0.0)
    return A

def _lp_best_response(a, M, timeout_s=3.0):
    n = a.size
    A = _build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0.0, None)] * n, method='highs',
                  options={'time_limit': timeout_s, 'dual_feasibility_tolerance': 1e-9,
                           'primal_feasibility_tolerance': 1e-9})
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    if not np.isfinite(g).all() or g.sum() <= 0:
        return None
    return g

        if time.monotonic() > stop_time - 2.0:

        M = float(_convolve_square(a).max())

            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1
        
        if stuck > 200:

        g = _lp_best_response(a, M * (1 + 1e-9), timeout_s=2.0)
        if g is None:

        gs = float(g.sum())

            cand = np.maximum(cand, 0.0)
            cs = cand.sum()
            if cs <= 0:

            cand = cand / cs
            cM = float(_convolve_square(cand).max())
            if cM < M * (1.0 - 1e-10):

                improved = True
                break
