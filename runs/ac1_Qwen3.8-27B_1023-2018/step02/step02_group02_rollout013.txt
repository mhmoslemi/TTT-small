                cand = np.exp(z - z.max())
            elif r < 0.5:
                # Local window scaling
                cand = a.copy()
                k = int(rng2.integers(0, n))
                w = max(2, n // 50)
                lo, hi = max(0, k - w), min(n, k + w)
                cand[lo:hi] *= abs(np.exp(rng2.normal(0, 0.1)))
            elif r < 0.7:
                # Pair transfer
                cand = a.copy()
                i1 = int(rng2.integers(0, n))
                i2 = int(rng2.integers(0, n))
                amt = 0.005 * a.sum()
                cand[i1] += amt
                cand[i2] = max(0, cand[i2] - amt)
            elif r < 0.85:
                # Mirror blend
                alpha = 0.1 + 0.2 * rng2.random()
                cand = (1 - alpha)
