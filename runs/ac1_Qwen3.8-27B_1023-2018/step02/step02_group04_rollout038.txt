            t_max = min(t_max, t_root)
        elif A < -1e-12:
            # Concave
            disc = B*B + 4.0*A*(M - b[k])
            # 4*A*(M-b[k]): A < 0, M-b[k] >= 0, so this <= 0
            if disc < 0:
                continue  # Always feasible
            sqrt_disc = np.sqrt(disc)
            # Smaller root: (-B + sqrt_disc) / (2A), since 2A < 0
            t_small = (-B + sqrt_disc) / (2.0*A)
            if t_small >= 0:
                t_max = min(t_max, t_small)
        else:
            # Linear
            if B > 1e-12:
                t_max = min(t_max, -C / B)
    
    return min(max(t_max, 0.0), 1.0)

def _lp_solve(f, M):
    """Solve LP: max sum(g) s.t. f*g <= M, g >= 0"""
    if linprog is None:
        return None
    n = f.size
    # Build constraint matrix: (2n-1) x n
    # (f*g)_k = sum_j f_j * g_{k-j} for valid j
    # For fixed k, valid j are max(0, k-n+1) <= j <= min(n-1, k)
    # (f*g)_k = sum over j of f_{k-j} * g_j (using full convolution convention)
    # Actually np.convolve(f, g)_k = sum_j f_j * g_{k-j}
    # For the constraint matrix, row k, column j: coefficient is f_{k-j} if 0 <= k-j < n, else 0
    
    # More efficiently: the constraint matrix is a banded Toeplitz-like matrix
    # Row k has nonzeros at columns j where 0 <= k-j < n, i.e., max(0,k-n+1) <= j <= min(n-1, k)
    # The entry at (k,j) is f[k-j]
    
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for j in range(n):
        # Column j: f[k-j] for k = j, j+1, ..., j+n-1
        A[j:j+n, j] = f
    
    c_obj = -np.ones(n)  # minimize -sum(g)
    b_ub = np.full(2*n-1, M)
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success and res.x is not None:
            g = np.clip(res.x, 0.0, None)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None

def _lp_iterate(f, stop_t, best, rng, max_it=5000):
    """LP-based iteration with exact line search"""
    n = f.size
    f = _normalize(f)
    loc_score = _score(f)
    _update_best(f, best)
    
    for it in range(max_it):

            break

        if not math.isfinite(M) or M < 1e-300:
            break

        if g is None:
            if it > 20:

            continue
        
        # Normalize g to same sum as f
        g = _normalize(g)
        
        # Compute cross terms for line search
        c_cross = np.convolve(f, g)
        d = np.convolve(g, g)
        
        # Find max feasible t
        t_max = _exact_line_search(b, c_cross, d, M)
        
        if t_max < 1e-8:
            # Can't move in this direction
            # Try perturbation
            if it % 10 == 0:
                noise = np.abs(rng.standard_normal(n)) * 0.05 * f.std()
                f_pert = _normalize(f + noise)
                if _score(f_pert) < loc_score * 1.001:
                    f = f_pert
                    loc_score = _score(f)
                    _update_best(f, best)
            continue
        
        # Take the step
        t = t_max
        f_new = (1.0 - t) * f + t * g
        f_new = _normalize(f_new)
        new_score = _score(f_new)
        
        if math.isfinite(new_score) and new_score < loc_score * (1.0 - 1e-12):
            f = f_new
            loc_score = new_score
            _update_best(f, best)
        else:

                f_try = _normalize((1.0 - t_try) * f + t_try * g)

                    f = f_try
                    loc_score = s_try
                    _update_best(f, best)
                    break
    
    return f

    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.clip(a, 0.0, 1000.0)

    return float(2.0 * n * float(b.max()) / (s * s))

    return a * (a.size / s)

def _exact_ls(b, c, d, M):
    """Exact line search: max t in [0,1] s.t. (1-t)^2 b + 2t(1-t) c + t^2 d <= M elementwise"""
    t_max = 1.0
    for k in range(len(b)):
        A = b[k] - 2.0*c[k] + d[k]
        B = 2.0*(c[k] - b[k])  # = -2*(b[k] - c[k])
        
        if A > 1e-10:
            disc = B*B + 4.0*A*(M - b[k])
            if disc < 0:
                disc = 0.0
            t_root = (-B + math.sqrt(disc)) / (2.0*A)
            if t_root < t_max:
                t_max = t_root
        elif A < -1e-10:
            disc = B*B + 4.0*A*(M - b[k])
            if disc < 0:
                continue
            sqrt_d = math.sqrt(disc)
            t_small = (-B + sqrt_d) / (2.0*A)  # 2A < 0, so this is the smaller root
            if t_small >= 0 and t_small < t_max:
                t_max = t_small
        else:
            if B > 1e-10:
                t_lin = (M - b[k]) / B  # -C/B where C = b[k]-M
                if t_lin < t_max:
                    t_max = t_lin

def _grad_iterate(f_init, stop_t, best, rng, max_it=5000):
    n = f_init.size
    if n < 10 or n % 2 != 0:
        return f_init
    h = n // 2
    
    # Antipalindromic parametrization: f[i] = 1+v[i], f[n-1-i] = 1-v[i]
    # Initialize v from f_init
    f = _norm_sum_n(f_init)
    v = np.zeros(h)
    for i in range(h):
        v[i] = (f[i] - 1.0)  # since sum(f) = n, average is 1
    
    def get_f(v):
        f = np.ones(n)
        f[:h] = 1.0 + v
        f[h:] = 1.0 - v[::-1]
        return np.clip(f, 0.0, None)
    
    loc = _score(get_f(v))
    _update_best(get_f(v), best)
    
    stale = 0
    for it in range(max_it):
        if time.monotonic() > stop_t - 3.0:
            break
        
        fv = get_f(v)
        b = np.convolve(fv, fv)
        k_star = int(np.argmax(b))
        
        # Gradient: d/dv[i] = 2*f[k_star-i] - 2*f[k_star-n+1+i] (with bounds checking)
        grad = np.zeros(h)
        for i in range(h):
            gi = k_star - i
            gj = k_star - n + 1 + i
            val = 0.0
            if 0 <= gi < n:
                val += fv[gi]
            if 0 <= gj < n:
                val -= fv[gj]
            grad[i] = 2.0 * val
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        
        # Line search (backtracking)
        direction = -grad / gnorm
        step = 0.5
        best_v, best_s = None, loc
        
        for trial in range(20):
            v_new = np.clip(v + step * direction, -0.999, 0.999)
            s_new = _score(get_f(v_new))
            if math.isfinite(s_new) and s_new < best_s:
                best_s, best_v = s_new, v_new
            step *= 0.7
        
        if best_v is not None:
            v = best_v
            loc = best_s
            _update_best(get_f(v), best)
            stale = 0
        else:
            stale += 1
            if stale % 10 == 0:
                v_pert = np.clip(v + 0.02 * rng.standard_normal(h), -0.999, 0.999)
                s_pert = _score(get_f(v_pert))
                if math.isfinite(s_pert) and s_pert < loc:
                    v = v_pert
                    loc = s_pert
                    _update_best(get_f(v), best)
                    stale = 0
            if stale > 50:
                break
    
    return get_f(v)
