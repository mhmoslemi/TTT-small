def _proj_simplex(v, s=1.0):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v.copy()
    if n == 1:
        return np.array([float(s)], dtype=np.float64)
    if not np.all(np.isfinite(v)):
        return np.full(n, float(s) / float(n), dtype=np.float64)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u * ind - cssv + s > 0.0
    if np.any(cond):
        r = int(np.nonzero(cond)[0][-1])
        theta = (cssv[r] - s) / float(r + 1)
    else:
        theta = (float(v.sum()) - s) / float(n)

    w = np.maximum(v - theta, 0.0)
    sw = float(w.sum())
    if not np.isfinite(sw) or sw <= 1e-15:
        return np.full(n, float(s) / float(n), dtype=np.float64)
    w = w * (s / sw)
    return w

def _lp_best_response(a, M, timeout=0.5):

        return None
    a = _sanitize(a)
    n = a.size
    if n == 0 or not np.isfinite(M) or M <= 0.0:
        return None

    a_lp = np.maximum(a, 1e-8)
    a_lp = a_lp / a_lp.sum()
    A = _build_A(a_lp)
    b_ub = np.full(2 * n - 1, float(M), dtype=np.float64)
    b_ub = b_ub * (1.0 + 1e-9) + 1e-12

    try:
        res = linprog(
            -np.ones(n),
            A_ub=A,
            b_ub=b_ub,
            bounds=[(0.0, None)] * n,
            method="highs",
            options={"time_limit": timeout, "presolve": True},
        )
    except Exception:
        return None

def _random_init(n, rng):

    x = np.arange(n, dtype=np.float64)
    t = x / float(n - 1)
    mode = int(rng.integers(0, 4))

    if mode == 0:
        z = rng.normal(0.0, 1.0, size=n)
        w = max(3, n // 60)
        sigma = max(1.0, w // 3)
        off = np.arange(-w, w + 1, dtype=np.float64)
        kernel = np.exp(-0.5 * (off / sigma) ** 2)
        kernel /= kernel.sum()
        v = np.convolve(z, kernel, mode="same")
        v = np.maximum(v, 0.01 * max(1e-300, float(v.max())) + 1e-8)
    elif mode == 1:
        base = np.sin(np.pi * (x + 1.0) / (n + 1.0))
        alpha = float(rng.uniform(0.4, 2.8))
        v = np.power(np.maximum(base, 1e-8), alpha)
    elif mode == 2:
        c1 = float(rng.uniform(0.08, 0.38))

        w1 = float(rng.uniform(0.03, 0.12))

            np.exp(-0.5 * ((t - c1) / w1) ** 2)
            + np.exp(-0.5 * ((t - c2) / w2) ** 2)

        )
    else:
        f = float(rng.choice([1.0, 2.0, 3.0]))

        v = 1.0 + alpha * np.cos(2.0 * np.pi * f * t)

    v = np.asarray(v, dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)
    v = np.maximum(v, 0.0)
    s = float(v.sum())
    if s <= 1e-15:

        if n > 8 and rng.random() < 0.90:
            sigma = 0.02 + 0.08 * float(rng.random())
            init = init * np.exp(rng.normal(0.0, sigma, size=n))

    init = _sanitize(init)
    return init


def _alternative_inits(base, n, rng):
    n = int(max(8, min(1024, int(n))))
    base = _sanitize(base)
    out = []

    for p in (0.7, 2.0):
        v = np.power(np.maximum(base, 1e-12), float(p))
        out.append(_sanitize(v))

    w = max(2, n // 30)
    sigma = max(1.0, w // 3)
    off = np.arange(-w, w + 1, dtype=np.float64)
    kernel = np.exp(-0.5 * (off / sigma) ** 2)
    kernel /= kernel.sum()
    out.append(_sanitize(np.convolve(base, kernel, mode="same")))

    out.append(_random_init(n, rng))

    uniq = []
    for v in out:
        v = _sanitize(v)
        dup = False
        for u in uniq:
            if v.size == u.size and np.linalg.norm(v - u) <= 1e-12:
                dup = True
                break
        if not dup:
            uniq.append(v)
        if len(uniq) >= 3:
            break
    return uniq


def _optimize_sequence(a0, stop_time, seed=0, effort=1.0, do_smooth=True, smooth_stages=6):
    a = _sanitize(a0)
    n = a.size
    current = a.copy()
    best_a = current.copy()
    best_M = _M(current)

    if stop_time < time.monotonic() + 1.0:
        return best_a, best_M

    if do_smooth and minimize is not None and n <= 1200:
        cur, bt, btM = _smooth_sequence(current, stop_time, effort=effort, stages=smooth_stages)
        if np.isfinite(btM) and btM < best_M - 1e-14:
            best_M = btM
            best_a = bt.copy()

        curM = _M(cur)
        if np.isfinite(curM) and curM <= max(1.25 * best_M, best_M + 1e-9):
            current = cur.copy()
        else:
            current = best_a.copy()

        if np.isfinite(curM) and curM < best_M - 1e-14:
            best_M = curM
            best_a = cur.copy()

    rng = np.random.default_rng(seed + 1000003)

    if n <= 256:
        loop_cap = int(65000 * max(0.1, effort))
        lp_every = 28
    elif n <= 512:
        loop_cap = int(30000 * max(0.1, effort))
        lp_every = 70
    else:
        loop_cap = int(12000 * max(0.1, effort))
        lp_every = 200

    lp_enabled = n <= 700 and linprog is not None

    if effort >= 0.70:
        temp_fracs = (0.03, 0.08, 0.20, 0.60, 1.50)
        mirror_mfac = (0.2, 0.6, 1.5)
        proj_mfac = (0.3, 1.0)
    elif effort >= 0.40:
        temp_fracs = (0.05, 0.20, 0.80)
        mirror_mfac = (0.3, 1.0)
        proj_mfac = (0.5,)
    else:
        temp_fracs = (0.10, 0.50)
        mirror_mfac = (0.5, 1.5)
        proj_mfac = ()

    no_improve = 0
    cand = current.copy()
    cand_M = best_M

    for it in range(loop_cap):
        if time.monotonic() > stop_time - 3.0:
            break

        if not np.all(np.isfinite(current)) or current.sum() <= 1e-12:
            current = best_a.copy()

        c = _conv_sq(current)
        c = np.where(np.isfinite(c), c, 0.0)
        M = float(c.max()) if c.size else 0.0
        if not np.isfinite(M) or M < 0.0:
            M = 0.0

        if M < best_M - 1e-14:
            best_M = M
            best_a = current.copy()
            no_improve = 0
        else:
            no_improve += 1

        cand = current.copy()
        cand_M = M
        Ms = max(M, 1e-15)

        for tf in temp_fracs:
            y = _softmax_vec(c, max(1e-15, tf * Ms))
            h = _correlate_y_a(y, current)
            h = np.where(np.isfinite(h), h, 0.0)
            sh = float(np.std(h))
            if sh <= 1e-16:
                continue
            base_eta = 1.0 / sh
            hmean = float(h.mean())

            for mf in mirror_mfac:
                eta = base_eta * float(mf)
                delta = eta * (h - hmean)
                delta = np.clip(delta, -35.0, 35.0)
                tmp = current * np.exp(-delta)
                cand, cand_M = _consider_candidate(tmp, cand, cand_M)

            for mf in proj_mfac:
                eta = base_eta * float(mf)
                v = current - eta * (h - hmean)
                tmp = _proj_simplex(v, 1.0)
                cand, cand_M = _consider_candidate(tmp, cand, cand_M)

        active_tol = max(1e-12, 1e-5 * M)
        mask = c >= M - active_tol
        if np.count_nonzero(mask) >= 2 and np.count_nonzero(mask) < c.size:
            y = np.zeros_like(c)
            y[mask] = 1.0
            y /= y.sum()
            h = _correlate_y_a(y, current)
            h = np.where(np.isfinite(h), h, 0.0)
            sh = float(np.std(h))
            if sh > 1e-16:
                base_eta = 1.0 / sh
                hmean = float(h.mean())
                for mf in (0.5, 1.5):
                    eta = base_eta * float(mf)
                    delta = eta * (h - hmean)
                    delta = np.clip(delta, -35.0, 35.0)
                    tmp = current * np.exp(-delta)
                    cand, cand_M = _consider_candidate(tmp, cand, cand_M)

        if lp_enabled and no_improve > 0:
            lp_timeout = 0.25 if n <= 256 else 0.45
            if no_improve > 180:
                lp_timeout *= 2.0
            if no_improve > 600:
                lp_timeout = min(2.0, lp_timeout + 0.5)

            if (it % (2 * lp_every) == 0 or no_improve > 45) and time.monotonic() < stop_time - 6.0:
                yd = _lp_dual_weights(current, M, timeout=lp_timeout)
                if yd is not None:
                    h = _correlate_y_a(yd, current)
                    h = np.where(np.isfinite(h), h, 0.0)
                    sh = float(np.std(h))
                    if sh > 1e-16:
                        base_eta = 1.0 / sh
                        hmean = float(h.mean())
                        for mf in (0.4, 1.0, 2.2):
                            eta = base_eta * float(mf)
                            delta = eta * (h - hmean)
                            delta = np.clip(delta, -35.0, 35.0)
                            tmp = current * np.exp(-delta)
                            cand, cand_M = _consider_candidate(tmp, cand, cand_M)

            if (it % (4 * lp_every) == 0 or no_improve > 90) and time.monotonic() < stop_time - 6.0:
                g = _lp_best_response(current, M, timeout=lp_timeout)
                if g is not None:
                    for t in (1.0, 0.8, 0.6, 0.4, 0.25, 0.12):
                        tmp = (1.0 - t) * current + t * g
                        cand, cand_M = _consider_candidate(tmp, cand, cand_M)
                    if g.size == n:
                        for t in (0.35, 0.55, 0.75):
                            pg = np.power(np.maximum(current, 1e-30), 1.0 - t)
                            pg *= np.power(np.maximum(g, 1e-30), t)
                            cand, cand_M = _consider_candidate(pg, cand, cand_M)

        if (it % 80 == 0 or no_improve > 70) and time.monotonic() < stop_time - 4.0:
            unin = np.full(n, 1.0 / float(n), dtype=np.float64)
            for alpha in (0.05, 0.15):
                cand, cand_M = _consider_candidate((1.0 - alpha) * current + alpha * unin, cand, cand_M)
            cand, cand_M = _consider_candidate(0.5 * current + 0.5 * current[::-1], cand, cand_M)
            for p in (0.7, 1.45):
                cand, cand_M = _consider_candidate(np.power(np.maximum(current, 1e-12), float(p)), cand, cand_M)

        if no_improve > 25 and time.monotonic() < stop_time - 4.0:
            sigma = 0.04 + 0.18 * min(1.0, no_improve / 400.0)
            tmp = current * np.exp(rng.normal(0.0, sigma, size=n))
            cand, cand_M = _consider_candidate(tmp, cand, cand_M)

        if cand_M < M - 1e-13:
            current = cand.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 600 and time.monotonic() < stop_time - 8.0:
                sigma = 0.20 + 0.25 * min(1.0, no_improve / 2500.0)
                tmp = best_a * np.exp(rng.normal(0.0, sigma, size=n))
                tmp = np.maximum(tmp, 1e-12)
                ss = float(tmp.sum())
                if ss > 1e-12:
                    current = tmp / ss
                else:
                    current = best_a.copy()
                no_improve = 0

    for arr in (current, cand, best_a):
        try:
            m = _M(arr)
        except Exception:
            continue
        if np.isfinite(m) and m < best_M - 1e-14:
            best_M = m
            best_a = arr.copy()

    return best_a, best_M


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = float(budget_s)
    if budget <= 25.0:
        try:
            p = _get_parent()
        except Exception:
            p = np.ones(64, dtype=np.float64) / 64.0
        return [float(x) for x in _sanitize(p)]

    headroom = max(18.0, 0.03 * budget)
    deadline = t0 + max(1.0, budget - headroom)

    try:
        parent = _get_parent()
    except Exception:
        parent = np.ones(64, dtype=np.float64) / 64.0

    best_seq = _sanitize(parent)
    best_score = _exact_score_simple(best_seq)
    if not np.isfinite(best_score):
        best_seq = np.ones(64, dtype=np.float64) / 64.0
        best_score = 2.0

    rng = np.random.default_rng(int(seed))
    bn = best_seq.size
    tried_lengths = set()

    rem0 = max(1.0, deadline - t0)

    # Phase 1: main length, deeper search.
    main_stop = t0 + 0.30 * rem0
    if main_stop > time.monotonic() + 8.0:
        try:
            a, _ = _optimize_sequence(
                best_seq,
                main_stop,
                seed=seed,
                effort=1.0,
                do_smooth=True,
                smooth_stages=7,
            )
            best_seq, best_score = _maybe_update(a, best_seq, best_score)
        except Exception:
            pass
    tried_lengths.add(int(best_seq.size))

    # Phase 2: extra parent-length initializations / basins.
    alt_stop = main_stop + 0.10 * rem0
    if alt_stop > time.monotonic() + 8.0:
        try:
            inits = _alternative_inits(best_seq, bn, rng)
        except Exception:
            inits = []
        if inits:
            used = 0.0
            per = max(6.0, (alt_stop - time.monotonic()) / float(len(inits) + 0.0))
            for k, init in enumerate(inits):
                if time.monotonic() > alt_stop - 5.0:
                    break
                stop_i = time.monotonic() + per
                try:
                    a, _ = _optimize_sequence(
                        init,
                        min(stop_i, alt_stop - 4.0),
                        seed=seed + 101 + k,
                        effort=0.55,
                        do_smooth=True,
                        smooth_stages=4,
                    )
                    best_seq, best_score = _maybe_update(a, best_seq, best_score)
                except Exception:
                    pass

    # Phase 3: nearby lengths.
    near_stop = alt_stop + 0.18 * rem0
    if near_stop > time.monotonic() + 8.0:
        near_lengths = []
        for L in (bn - 2, bn - 1, bn + 1, bn + 2, bn - 3, bn + 3):
            L = int(L)
            if 8 <= L <= 1200 and L not in tried_lengths:
                near_lengths.append(L)
                tried_lengths.add(L)
        if near_lengths:
            per = max(6.0, (near_stop - time.monotonic()) / float(len(near_lengths)))
            for L in near_lengths:
                if time.monotonic() > near_stop - 5.0:
                    break
                try:
                    init = _make_init(best_seq, L, rng)
                    a, _ = _optimize_sequence(
                        init,
                        min(time.monotonic() + per, near_stop - 4.0),
                        seed=seed + 1000 + L,
                        effort=0.65,
                        do_smooth=True,
                        smooth_stages=4,
                    )
                    best_seq, best_score = _maybe_update(a, best_seq, best_score)
                except Exception:
                    pass

    # Phase 4: broader length exploration.
    other_stop = near_stop + 0.17 * rem0
    if other_stop > time.monotonic() + 8.0:
        candidates = [
            int(0.75 * bn),
            int(0.9 * bn),
            int(1.15 * bn),
            int(1.35 * bn),
            int(1.6 * bn),
            128,
            191,
            255,
            320,
            512,
            768,
            1024,
        ]
        others = []
        for L in candidates:
            L = int(max(8, min(1024, L)))
            if L not in tried_lengths:
                others.append(L)
                tried_lengths.add(L)
        if others:
            per = max(5.0, (other_stop - time.monotonic()) / float(len(others) + 0.0))
            for L in others:
                if time.monotonic() > other_stop - 5.0:
                    break
                try:
                    init = _make_init(best_seq, L, rng)
                    a, _ = _optimize_sequence(
                        init,
                        min(time.monotonic() + per, other_stop - 4.0),
                        seed=seed + 5000 + L,
                        effort=0.35,
                        do_smooth=True,
                        smooth_stages=3,
                    )
                    best_seq, best_score = _maybe_update(a, best_seq, best_score)
                except Exception:
                    pass

    # Phase 5: final polish of the globally best candidate.
    if time.monotonic() < deadline - 8.0:
        try:
            a, _ = _optimize_sequence(
                best_seq,
                deadline - 5.0,
                seed=seed + 999999,
                effort=0.85,
                do_smooth=False,
                smooth_stages=0,
            )
            best_seq, best_score = _maybe_update(a, best_seq, best_score)
        except Exception:
            pass

    # Final exact validation with the authoritative evaluator.
    cand = _sanitize(best_seq)
    try:
        exact = float(evaluate_sequence([float(x) for x in cand]))
    except Exception:
        exact = float("inf")

    if not np.isfinite(exact):
        cand = np.ones(64, dtype=np.float64) / 64.0
        try:
            exact2 = float(evaluate_sequence([float(x) for x in cand]))
        except Exception:
            exact2 = 2.0
        if np.isfinite(exact2):
            return [float(x) for x in cand]

    if time.monotonic() > budget - 0.5:
        return [float(x) for x in cand]

    return [float(x) for x in cand]
