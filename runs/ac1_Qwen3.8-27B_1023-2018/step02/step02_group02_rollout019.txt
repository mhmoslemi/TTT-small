def proj_simplex(v):

    n = len(v)

    theta = (cssv[rho] - 1) / (rho + 1)

    def proj_simplex(v):
        """Project v onto simplex {x >= 0, sum x = 1}."""
        n_v = len(v)
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u)
        rho = np.nonzero(u * np.arange(1, n_v+1) > (cssv - 1))[0]
        if len(rho) == 0:
            return np.ones(n_v) / n_v
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    
    def subgrad_step(f, c, active_frac=0.1):
        """Compute subgradient and projected step."""
        max_c = c.max()
        threshold = max_c * (1.0 - active_frac)
        active = np.where(c >= threshold)[0]
        n_act = len(active)
        
        grad = np.zeros(n)
        for m in active:
            k_lo = max(0, m - n + 1)
            k_hi = min(m, n - 1)
            if k_lo <= k_hi:
                # grad[k] += 2*f[m-k] for k in [k_lo, k_hi]
                idx = m - np.arange(k_lo, k_hi+1)
                grad[k_lo:k_hi+1] += 2.0 * f[idx]
        grad /= n_act
        
        # Project gradient onto tangent space of simplex
        grad_tan = grad - grad.mean()
        return grad_tan

        """Solve LP to find direction g that reduces peak."""
