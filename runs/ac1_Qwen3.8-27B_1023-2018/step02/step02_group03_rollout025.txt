                    break
        
        return a
    
    def grad_polish(a, max_iters, stop):
        """Gradient/subgradient polish."""
        n = a.size
        a = a.copy()
        a = np.clip(a, 0, None)
        a /= a.sum()
        local = fs(a)
        ub(a)
        stale = 0
        beta = 3.0  # smoothing parameter
        
        for it in range(max_iters):
            if time.monotonic() >= stop:
                break
            
            b = np.convolve(a, a)
            m = float(b.max())
            if m <= 0:
                break
            
            # Smooth weights
            e = np.exp(beta * (b - m))
            w = e / e.sum()
            
            # Compute weighted gradient
            g = np.zeros(n)
            for k in range(2 * n - 1):
                wk = w[k]
                if wk < 1e-12:
                    continue
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                idx = np.arange(lo, hi + 1)
                g[idx] += wk * 2.0 * a[k - idx]
            
            # Normalize gradient
            gn = float(np.linalg.norm(g))
            if gn < 1e-14:
                break
            g /= gn
            
            # Line search
            best_sc = local
            best_t = 0
            for alpha in [0.001, 0.002, 0.005, 0.01, 0.02, 0.05, 0.1]:
                v = a - alpha * g
                v = np.clip(v, 0, None)
                vs = v.sum()
                if vs < 1e-15:
                    continue
                v /= vs
                sc = fs(v)
                if sc < best_sc:
                    best_sc = sc
                    best_t = alpha
            
            if best_t > 0 and best_sc < local * (1 - 1e-12):
                a -= best_t * g
                a = np.clip(a, 0, None)
                a /= a.sum()
                local = best_sc
                stale = 0
                ub(a)
            else:
                stale += 1
                if stale % 20 == 0:
                    a_p = a + 0.02 * rng.standard_normal(n)
                    a_p = np.clip(a_p, 0, None)
                    if a_p.sum() > 1e-15:
                        a_p /= a_p.sum()
                        if fs(a_p) < local:
                            a = a_p
                            local = fs(a)
                            stale = 0
                            ub(a)
                if stale > 80:
                    break
        
        return a
    
    def interp_len(src, n_target):
        if src.size == n_target:
            return src.copy()
        ox = np.linspace(0.0, 1.0, src.size)
        nx = np.linspace(0.0, 1.0, n_target)
        out = np.interp(nx, ox, src)
        out = np.clip(out, 0, None)
        s = out.sum()
        if s > 1e-15:
            out /= s
        else:
            out = np.ones(n_target) / n_target
        return out
    
    # ===== MAIN OPTIMIZATION =====
    
    p = parent.copy()
    n_p = p.size
    s_p = p.sum()
    if s_p > 1e-15:
        p /= s_p
    else:
        p = np.ones(n_p) / n_p
    
    # Phase 1: Intensive LP at parent length
    n_p_actual = p.size
    if n_p_actual >= 4:
        t_rem = deadline - time.monotonic()
        alloc = min(500.0, t_rem * 0.4)
        if alloc > 30:
            t_stop = time.monotonic() + alloc * 0.85
            p = lp_refine(p, 2000, t_stop, 
                         [1.0, 0.99, 0.98, 0.97, 0.96, 0.95])
            ub(p)
        
        if time.monotonic() < deadline - 60:
            p = grad_polish(p, 30000, deadline - 50)
            ub(p)
        
        if time.monotonic() < deadline - 40:
            t_rem2 = deadline - time.monotonic() - 30
            alloc2 = min(200.0, t_rem2 * 0.6)
            if alloc2 > 15:
                p = lp_refine(p, 1000, time.monotonic() + alloc2 * 0.85,
                             [1.0, 0.995, 0.99, 0.985, 0.98])
                ub(p)
                if time.monotonic() < deadline - 20:
                    p = grad_polish(p, 20000, deadline - 15)
                    ub(p)
    
    # Phase 2: Nearby lengths
    lengths_to_try = []
    for delta in [-8, -6, -4, -3, -2, -1, 1, 2, 3, 4, 6, 8, 10]:
        L = n_p_actual + delta
        if 20 <= L <= 512:
            lengths_to_try.append(L)
    # Some fixed lengths
    for L in [256, 288, 320, 352, 384]:
        if L != n_p_actual and 20 <= L <= 512:
            lengths_to_try.append(L)
    lengths_to_try = sorted(set(lengths_to_try))
    
    for L in lengths_to_try:
        if time.monotonic() >= deadline - 30:
            break
        t_rem = deadline - time.monotonic() - 25
        alloc = min(40.0, t_rem * 0.15)
        if alloc < 10:
            break
        
        a_init = interp_len(p, L)
        a_w = lp_refine(a_init, max(50, int(alloc)), time.monotonic() + alloc * 0.6,
                       [1.0, 0.98, 0.96])
        ub(a_w)
        
        if time.monotonic() < deadline - 20:
            a_w = grad_polish(a_w, int(alloc * 2), deadline - 15)
            ub(a_w)
    
    # Phase 3: Multi-restart at parent length
    n_restarts = 20
    for r in range(n_restarts):
        if time.monotonic() >= deadline - 25:
            break
        t_rem = deadline - time.monotonic() - 20
        alloc = min(25.0, t_rem / max(1, n_restarts - r))
        if alloc < 5:
            break
        
        r_type = r % 5
        if r_type == 0:
            a_init = p * np.exp(0.15 * rng.standard_normal(n_p_actual))
        elif r_type == 1:
            a_init = p[::-1].copy()
        elif r_type == 2:
            a_init = p * np.exp(0.3 * rng.standard_normal(n_p_actual))
        elif r_type == 3:
            a_init = p + 0.05 * rng.standard_normal(n_p_actual)
        else:
            # Compress/expand
            shift = rng.integers(-3, 4)
            a_init = np.roll(p, shift)
        
        a_init = np.clip(a_init, 0, None)
        s_init = a_init.sum()
        if s_init > 1e-15:
            a_init /= s_init
        
        a_w = lp_refine(a_init, max(10, int(alloc * 0.5)),
                       time.monotonic() + alloc * 0.45,
                       [1.0, 0.97])
        ub(a_w)
        if time.monotonic() < deadline - 15:
            a_w = grad_polish(a_w, int(alloc), deadline - 12)
            ub(a_w)
    
    # Phase 4: Final polish on best
    if time.monotonic() < deadline - 10 and best_arr is not None:
        a_pol = best_arr.copy()
        s_pol = a_pol.sum()
        if s_pol > 1e-15:
            a_pol /= s_pol
        a_pol = grad_polish(a_pol, 10000, deadline - 5)
        ub(a_pol)
        if time.monotonic() < deadline - 5:
            a_pol = lp_refine(a_pol, 200, deadline - 3, [1.0, 0.995])
            ub(a_pol)
    
    # Final: ensure valid return
    if best_arr is None:
        arr = np.ones(96)
        ub(arr)
    
    final = np.clip(best_arr, 0.0, 1000.0)
    if final.sum() < 0.01:
        final = np.ones(96)
    ub(final)
    
    return [float(x) for x in final]
