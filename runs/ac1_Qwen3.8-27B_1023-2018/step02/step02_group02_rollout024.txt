import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def propose_candidate(seed=42, budget_s=1300, **kwargs):

    deadline = t0 + max(1.0, float(budget_s)) - 10.0

        parent = np.ones(192, dtype=np.float64)

    n = int(parent.size)
    fsize = next_fast_len(2 * n - 1)

    def convolve_full(a):
        F = rfft(a, fsize)
        c = irfft(F * F, fsize)[:2 * n - 1]
        c = np.maximum(c, 0.0)
        c[~np.isfinite(c)] = 0.0
        return c

    def peak(c):
        return float(c.max()) if c.size else 0.0

    def score_norm(a):
        """Score assuming sum(a)=1"""
        return 2.0 * n * peak(convolve_full(a))

    def proj_simplex(v):
        """Project v onto {x >= 0, sum(x) = 1} via Euclidean projection."""
        n_v = len(v)
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u)
        idx = np.arange(1, n_v + 1)
        cond = u * idx > (cssv - 1.0)
        if not np.any(cond):
            return np.ones(n_v) / n_v
        rho = int(np.nonzero(cond)[0][-1])
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
        w = np.maximum(w, 0.0)
        ws = float(w.sum())
        if ws < 1e-30:
            return np.ones(n_v) / n_v
        w /= ws
        return w

    # Normalize parent to sum=1
    s_parent = float(parent.sum())
    if s_parent < 0.01:
        parent = np.ones(n, dtype=np.float64)
        s_parent = 1.0
    f_init = parent / s_parent

    # Evaluate initial
    M0 = peak(convolve_full(f_init))
    best_score = 2.0 * n * M0
    best_f = f_init.copy()

    # Build constraint matrix once per LP step (a is the "f" in f*g)
    def build_B(f_curr):
        B = np.zeros((2 * n - 1, n), dtype=np.float64)
        for j in range(n):
            B[j:j + n, j] = f_curr
        return B

    def lp_step(f_curr, B=None):
        """Solve LP for direction g, then line-search."""
        c_curr = convolve_full(f_curr)
        M = peak(c_curr)
        if M < 1e-15:
            return None, M

        if B is None:
            B = build_B(f_curr)

        c_obj = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        bounds = [(0.0, None)] * n

        try:
            res = linprog(c_obj, A_ub=B, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True, 'time_limit': 5.0})
        except Exception:
            return None, M

        if not res.success or res.x is None:
            return None, M

        g = np.maximum(np.asarray(res.x, dtype=np.float64).ravel(), 0.0)
        gs = float(g.sum())
        if gs < 1e-12:
            return None, M
        g /= gs

        if float(np.linalg.norm(g - f_curr, 1)) < 1e-9:
            return None, M

        best_t = 0.0
        best_m = M

        t_list = np.linspace(0.02, 1.0, 25)
        for t in t_list:
            h = (1.0 - t) * f_curr + t * g
            m_h = peak(convolve_full(h))
            if m_h < best_m - 1e-15:
                best_m = m_h
                best_t = t

        if best_t > 0.0:
            t_lo = max(0.0, best_t - 0.08)
            t_hi = min(1.0, best_t + 0.08)
            for _ in range(12):

                t1 = t_lo + (1.0 - gr) * (t_hi - t_lo)
                t2 = t_lo + gr * (t_hi - t_lo)
                m1 = peak(convolve_full((1.0 - t1) * f_curr + t1 * g))
                m2 = peak(convolve_full((1.0 - t2) * f_curr + t2 * g))
                if m1 < best_m - 1e-15:
                    best_m = m1
                    best_t = t1
                if m2 < best_m - 1e-15:
                    best_m = m2
                    best_t = t2
                if m1 < m2:
                    t_hi = t2
                else:
                    t_lo = t1

        if best_t > 0.0 and best_m < M - 1e-15:
            return (1.0 - best_t) * f_curr + best_t * g, best_m
        return None, M

    # PHASE 1: LP-based descent
    # =========================================================
    f = f_init.copy()
    B = build_B(f)

        if time.monotonic() > deadline - 300.0:
            break

                f = proj_simplex(f + rng_k.normal(0, 0.02, n))
                B = build_B(f)
                continue
            break

            best_f = f.copy()

        sc = score_norm(a)
        lr = 0.02
        stuck = 0
        rng = np.random.default_rng(seed + 1000)

            if time.monotonic() > deadline - 80.0:
                break
            if stuck > 800:
                break

            c = convolve_full(a)
            M = peak(c)
            sc = 2.0 * n * M
            if sc < best_score - 1e-14:
                best_score = sc
                best_f = a.copy()
                stuck = 0

            # Soft-LSE weights
            tau = max(1e-10, 0.008 * M)
            bm = float(np.max(c))
            x = np.clip((c - bm) / tau, -80.0, 80.0)
            ex = np.exp(x)
            s_ex = float(ex.sum())
            if s_ex < 1e-30:
                idx_m = int(np.argmax(c))
                p = np.zeros_like(c)
                p[idx_m] = 1.0

                p = ex / s_ex

            # Build sliding window matrix: P[j, m] = p[m+j] for m=0..n-1
            # P has shape (n, n), P[j] = p[j:j+n]
            P_mat = np.ascontiguousarray(p[:n*2 - 1 * 0])  # p has length 2n-1
            # Actually p has length 2n-1
            # For j in 0..n-1: P[j] = p[j:j+n] (length n)
            Pm = np.empty((n, n), dtype=np.float64)
            for j in range(n):
                Pm[j, :] = p[j:j+n]
            h = Pm @ a  # h[j] = sum_m p[j+m]*a[m]

            # Tangent projection: step = h - mean(h) (for sum(a)=1 constraint)
            step = h - float(np.mean(h))

            sn = float(np.linalg.norm(step))
            if sn < 1e-14:
                stuck += 1
                if stuck % 100 == 0 and stuck < 800:

                cand = a - scale * step / sn
                cand = proj_simplex(cand)
                cand_sc = score_norm(cand)
                if cand_sc < sc - 1e-14:
                    a = cand
                    sc = cand_sc
                    improved = True
                    break

            if improved:
                stuck = 0
                lr = min(lr * 1.03, 0.5)
            else:
                stuck += 1
                lr = max(0.001, lr * 0.92)

    # =========================================================
    # PHASE 3: Fine stochastic polish
    # =========================================================
    if time.monotonic() < deadline - 15.0:
        a = best_f.copy()
        sc = best_score

        sigma = 0.015

                break

            if r < 0.35:

                cand /= cand.sum()
            elif r < 0.60:

                cand = a.copy()
                k = int(rng2.integers(0, n))
                width = max(1, n // 50)
                lo = max(0, k - width)
                hi = min(n, k + width)
                cand[lo:hi] *= float(np.exp(rng2.normal(0, 0.04)))
                cand = np.maximum(cand, 1e-12)
                cand /= cand.sum()
            elif r < 0.80:
                # Pair transfer
                cand = a.copy()
                k1 = int(rng2.integers(0, n))
                k2 = int(rng2.integers(0, n))
                if k1 != k2:
                    amt = 0.005 * float(a.sum()) * float(rng2.random())
                    cand[k1] += amt
                    cand[k2] -= amt
                    cand = np.maximum(cand, 1e-12)
                    cand /= cand.sum()
            elif r < 0.90:
                # Mirror blend
                blend = 0.05 * rng2.random()
                cand = (1.0 - blend) * a + blend * a[::-1]
                cand = np.maximum(cand, 1e-12)
                cand /= cand.sum()
            else:
                # Very small Gaussian
                cand = a + rng2.normal(0, sigma * 0.05, n)
                cand = np.maximum(cand, 1e-12)
                cand /= cand.sum()

            cand_sc = score_norm(cand)
            if cand_sc < sc - 1e-14:
                a = cand.copy()
                sc = cand_sc
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_f = a.copy()
                sigma = max(0.002, sigma * 0.99)
            else:
                sigma = min(0.2, sigma * 1.01)

    # =========================================================
    # PHASE 4: Final LP polish on best_f
    # =========================================================
    if time.monotonic() < deadline - 5.0:
        a = best_f.copy()
        B_final = build_B(a)
        for _ in range(200):
            if time.monotonic() > deadline - 5.0:
                break
            new_a, new_M = lp_step(a, B_final)
            if new_a is None:
                break
            a = new_a
            B_final = build_B(a)
            sc = 2.0 * n * new_M
            if sc < best_score - 1e-14:
                best_score = sc
                best_f = a.copy()

    # =========================================================
    # Final validation
    # =========================================================
    best_f = np.clip(best_f, 0.0, 1000.0)
    if not np.isfinite(best_f).all() or float(best_f.sum()) < 0.01:
