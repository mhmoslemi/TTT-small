                         'maxls': 50}
            )
            if res.x is not None:
                f_res = np.exp(res.x - res.x.max())
                f_res /= f_res.sum()
                sc_res = score_of(f_res)
                if sc_res < best_score:
                    best_score = sc_res
                    best_f = f_res.copy()
        except Exception:
            pass

        # Second L-BFGS pass with higher alpha
        if time.monotonic() < deadline - 150.0:
            x0b = np.log(np.maximum(best_f, 1e-12))
            x0b = x0b - x0b.max()
            alpha_val2 = 200.0
            try:
                res2 = minimize(
                    lambda x: lse_obj_grad(x, alpha_val2)[0],
                    x0b,
                    jac=lambda x: lse_obj_grad(x, alpha_val2)[1],
                    method='L-BFGS-B',
                    options={'maxiter': 200, 'maxfun': 400, 'gtol': 1e-12,
                             'maxls': 40}
                )
                if res2.x is not None:
                    f_res2 = np.exp(res2.x - res2.x.max())
                    f_res2 /= f_res2.sum()
                    sc_res2 = score_of(f_res2)
                    if sc_res2 < best_score:
                        best_score = sc_res2
                        best_f = f_res2.copy()
            except Exception:
                pass

    # =========================================================
    # PHASE 3: Mirror descent refinement with steepening alpha
    # =========================================================
    if time.monotonic() < deadline - 80.0:
        a = best_f.copy()
        lr = 0.3
        alpha = 100.0
        stuck = 0
        rng = np.random.default_rng(seed)

        for it in range(10000):
            if time.monotonic() > deadline - 80.0:
                break
            if stuck > 500:
                a = np.maximum(a + rng.normal(0, 0.04, n), 1e-10)
                a /= a.sum()
                alpha = max(50.0, alpha * 0.6)
                lr = 0.3
                stuck = 0
                continue

            c = convolve_full(a)
            bm = float(c.max())
            sc = 2.0 * n * bm

            # Softmax weights
            x_s = np.clip((c - bm) * alpha, -60.0, 60.0)
            ex = np.exp(x_s)
            s_ex = float(ex.sum())
            if s_ex < 1e-30:
                p = np.zeros_like(c)
                p[int(np.argmax(c))] = 1.0
            else:
                p = ex / s_ex

            # Gradient
            pf = rfft(p, fsize)
            ff = rfft(a, fsize)
            corr = irfft(np.conj(pf) * ff, fsize)
            grad_f = 2.0 * corr[:n]
            grad_f -= float(np.dot(grad_f, a))  # project to simplex tangent
            gn = float(np.linalg.norm(grad_f))

            if gn < 1e-15:
                stuck += 1
                continue

            # Mirror descent step
            im_ratio = lr * alpha / (n * max(bm, 1e-12))
            update = np.exp(-im_ratio * grad_f / gn)
            a_new = a * update
            su = float(a_new.sum())
            if su < 1e-300:
                stuck += 1
                lr = max(0.05, lr * 0.8)
                continue
            a_new /= su

            sc_new = 2.0 * n * peak(convolve_full(a_new))
            if sc_new < sc - 1e-15:
                a = a_new
                stuck = 0
                lr = min(lr * 1.03, 3.0)
                alpha = min(alpha * 1.015, 1000.0)
                if sc_new < best_score - 1e-14:
                    best_score = sc_new
                    best_f = a.copy()
            else:
                stuck += 1
                lr = max(0.05, lr * 0.92)

    # =========================================================
    # PHASE 4: Stochastic local search
    # =========================================================
    if time.monotonic() < deadline - 20.0:
        a = best_f.copy()
        sc = best_score
        rng2 = np.random.default_rng(seed + 12345)
        sigma = 0.04

        for _ in range(8000):
            if time.monotonic() > deadline - 20.0:
                break

            r = rng2.random()
            if r < 0.30:
                z = np.log(np.clip(a, 1e-12, None))
                z += rng2.normal(0.0, sigma, n)
                z -= z.max()
                cand = np.exp(z)
                cand /= cand.sum()
            elif r < 0.55:
                cand = a.copy()
                k = int(rng2.integers(0, n))
                width = max(1, n // 100)
                lo = max(0, k - width)
                hi = min(n, k + width)
                cand[lo:hi] *= math.exp(rng2.normal(0, 0.08))
                cand /= cand.sum()
            elif r < 0.75:
                cand = a.copy()
                k1 = int(rng2.integers(0, n))
                k2 = int(rng2.integers(0, n))
                if k1 != k2:
                    amt = 0.005 * float(a.max())
                    cand[k1] += amt
                    cand[k2] = max(0.0, cand[k2] - amt)
                    cand = np.maximum(cand, 1e-12)
                    cand /= cand.sum()
            elif r < 0.90:
                w = rng2.uniform(0.05, 0.3)
                cand = (1.0 - w) * a + w * a[::-1]
                cand /= cand.sum()
            else:
                # Slight global reshape
                base = 1.0 + rng2.normal(0, sigma * 0.5)
                cand = a * base
                cand = np.maximum(cand, 1e-12)
                cand /= cand.sum()

            cand_sc = score_of(np.maximum(cand, 0))
            if cand_sc < sc - 1e-14:
                a = cand.copy()
                sc = cand_sc
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_f = a.copy()
                sigma = max(0.008, sigma * 0.985)
            else:
                sigma = min(0.25, sigma * 1.015)

    # =========================================================
    # PHASE 5: Final LP polish
    # =========================================================
    if time.monotonic() < deadline - 10.0:
        f_final = best_f.copy()
        for it in range(80):
            if time.monotonic() > deadline - 10.0:
                break
            c_curr = convolve_full(f_final)
            M = peak(c_curr)
            curr_score = 2.0 * n * M
            if curr_score < best_score - 1e-14:
                best_score = curr_score
                best_f = f_final.copy()

            g = lp_direction(f_final, M)
            if g is None:
                break
            if float(np.linalg.norm(g - f_final, 1)) < 1e-9:
                break
            t_best, M_best = line_search(f_final, g, M)
            if t_best < 1e-10:
                break
            f_final = (1.0 - t_best) * f_final + t_best * g

    # =========================================================
    # Final validation with exact evaluator
    # =========================================================
    best_f = np.clip(best_f, 0.0, 1000.0)
    best_f = np.maximum(best_f, 0.0)
    if not np.isfinite(best_f).all() or best_f.sum() < 0.01:
        best_f = np.ones(n) / n

    final_list = [float(x) for x in best_f]
    try:
        fs = float(evaluate_sequence(final_list))
    except Exception:
        fs = float('inf')

    if not np.isfinite(fs):
        final_list = [1.0 / n] * n
        try:
            fs = float(evaluate_sequence(final_list))
        except Exception:
            pass
        if not np.isfinite(fs):
            return [1.0] * n

    return final_list
