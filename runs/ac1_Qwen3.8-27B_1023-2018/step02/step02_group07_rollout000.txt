def _lse_off_grad(a, K, fs, fs2, n):

      A=rfft(a,fs)
      b=irfft(A*A,fs)[:2*n-1]
      vals=b[1:n]

      x = np.minimum(x, 0.0)

      if not np.isfinite(s) or s <= 0.0:

          p[int(np.argmax(vals))] = 1.0

def _lp_ray(a, M, n, timeout_margin=20):
    try:
       from scipy.optimize import linprog
       # M maybe max b; if M<=0 return None
       # To avoid infeasibility due to rounding, M *= (1+1e-8)
       A = np.zeros((2*n-1, n), dtype=np.float64)
       r = np.arange(2*n-1)[:,None]
       cidx = np.arange(n)[None,:]
       d = r - cidx
       mask = (d >=0) & (d < n)
       # Need assign a[d] for mask. But d is 2D; A[mask] = a[d[mask]]
       A[mask] = a[d[mask]]
       res = linprog(np.full(n, -1.0), A_ub=A, b_ub=np.full(2*n-1, M*(1+1e-8)), bounds=[(0,None)]*n, method='highs')
       if res is not None and res.success and res.x is not None:
           g=np.asarray(res.x,dtype=np.float64)
           g=np.maximum(g,0)
           s=g.sum()
           if s>1e-12:
              return g/s

def _interpolate(par, n):
   if par is None: return None
   P=par.size
   if P == n: return par.copy()
   if P == 0 or n < 1: return None
   idx=(np.arange(n)*P)//n
   a=par[idx]
   return _norm(a)

def _random_init(n, seed, par):

    return e / s

                    K = min(K_end, K * 1.12 + 0.5)
                    beta = max(beta1, beta * 0.97)

            _update_int(best_a, best_hard)
            _exact_score(best_a)
            _exact_score(cur_a)
        except Exception:
            pass

    def _interpolate(par_arr, n):
        if par_arr is None or par_arr.size == 0 or n < 1:
            return None
        P = par_arr.size
        if P == n:
            return par_arr.copy()
        idx = (np.arange(n) * P) // n
        a = par_arr[idx]
        return _norm(a)

    def _random_init(n, job_seed, par_arr):
        rng = np.random.default_rng(job_seed)
        a = None
        if par_arr is not None and rng.random() < 0.6:
            a = _interpolate(par_arr, n)
            if a is not None:
                a = a + rng.normal(0.0, 0.10, n)
        if a is None:
            t = (np.arange(n) + 0.5) / n
            x = 2.0 * t - 1.0
            k = int(rng.integers(0, 5))
            if k == 0:
                a = np.ones(n)
            elif k == 1:
                a = np.maximum(0.05, 1.0 - np.abs(x))
            elif k == 2:
                a = np.exp(-2.0 * x * x)
            elif k == 3:
                a = np.maximum(0.05, 0.5 + 0.5 * np.cos(2.0 * math.pi * float(rng.random()) * t))
            else:
                a = np.maximum(1e-3, 1.0 + 0.5 * rng.normal(0.0, 1.0, n))
        a = a + 1e-3 * np.abs(rng.normal(0.0, 1.0, n))
        return _norm(a)

    def _polish(a0, end_time, job_seed, do_lp=True):
        try:
            a = _norm(a0)
            if a is None:
                return
            n = a.size
            if n < 2 or time.monotonic() >= end_time:
                return
            fs = next_fast_len(2 * n - 1)
            z = np.log(np.maximum(a, 1e-300))
            z = z - float(z.mean())
            z = np.clip(z, -45.0, 45.0)

            cur_a = a.copy()
            cur_h = _hard_fast(cur_a, fs)
            best_a = cur_a.copy()
            best_h = cur_h
            _update_int(best_a, best_h)

            rng = np.random.default_rng(job_seed)
            sigma = 0.10
            stuck = 0
            lp_attempts = 0

            if do_lp and n <= 512 and time.monotonic() < end_time - 60.0:
                smooth, _, lmax = _grad_smooth(cur_a, 50.0, 0.0, fs, fs2 := next_fast_len(3 * n - 2))
                if np.isfinite(lmax):
                    for _ in range(2):
                        if time.monotonic() >= end_time - 30.0 or lp_attempts >= 2:
                            break
                        g = _lp_ray(best_a, lmax, n)
                        lp_attempts += 1
                        if g is None:
                            continue
                        improved = False
                        for tmix in (0.03, 0.08, 0.15, 0.30, 0.55, 0.80):
                            mix = (1.0 - tmix) * best_a + tmix * g
                            mix = _norm(mix)
                            if mix is None:
                                continue
                            hm = _hard_fast(mix, fs)
                            if np.isfinite(hm) and hm < best_h - 1e-12:
                                best_a = mix.copy()
                                best_h = hm
                                _update_int(best_a, best_h)
                                improved = True
                                break
                        if not improved:
                            break

            while time.monotonic() < end_time - 0.6:
                r = float(rng.random())
                z2 = z.copy()

                if r < 0.38:
                    i = int(rng.integers(0, n))
                    z2[i] += rng.normal(0.0, sigma)
                elif r < 0.55:
                    w = max(1, min(n, int(rng.integers(1, max(2, n // 30 + 1)))))
                    i = int(rng.integers(0, max(1, n - w + 1)))
                    i = max(0, min(i, n - w))
                    z2[i : i + w] += rng.normal(0.0, 0.6 * sigma, w)
                elif r < 0.72:
                    f = float(rng.uniform(0.4, 4.5))
                    phase = float(rng.uniform(0.0, 2.0 * math.pi))
                    z2 += 0.55 * sigma * np.sin(2.0 * math.pi * f * np.arange(n) / float(n) + phase)
                elif r < 0.86:
                    z2 = (z - float(z.mean())) * float(rng.uniform(0.70, 1.35))
                else:
                    z2 = z + 0.4 * sigma * rng.normal(0.0, 1.0, n)

                z2 = z2 - float(z2.mean())
                z2 = np.clip(z2, -45.0, 45.0)
                a2 = _softmax(z2)
                h2 = _hard_fast(a2, fs)

                if np.isfinite(h2) and h2 < cur_h - 1e-13:
                    z = z2
                    cur_a = a2
                    cur_h = h2
                    stuck = 0
                    if h2 < best_h - 1e-14:
                        best_a = a2.copy()
                        best_z_local = z.copy()
                        best_h = h2
                        _update_int(best_a, best_h)
                        sigma = max(0.015, sigma * 0.985)
                else:
                    # Small-temperature acceptance for escape.
                    if np.isfinite(h2) and h2 < best_h * 1.001:
                        gap = max(0.0, h2 - cur_h)
                        temp = max(1e-8, 2e-5 * max(1.0, best_h))
                        if float(rng.random()) < math.exp(-gap / temp):
                            z = z2
                            cur_a = a2
                            cur_h = h2
                    else:
                        stuck += 1
                        sigma = min(1.25, sigma * 1.035)
                        if stuck > 350:
                            z = z + rng.normal(0.0, 0.35, n)
                            z = z - float(z.mean())
                            z = np.clip(z, -45.0, 45.0)
                            a2 = _softmax(z)
                            h2 = _hard_fast(a2, fs)
                            if np.isfinite(h2):
                                cur_a = a2
                                cur_h = h2
                            stuck = 0
                            sigma = 0.30

            _update_int(best_a, best_h)
            _exact_score(best_a)
            _exact_score(cur_a)
        except Exception:
            pass

    # Build search jobs.
    base_n = int(par.size) if par is not None else 192
    base_n = int(min(max(base_n, 32), 1024))

    jobs = []

    if par is not None:
        a0 = _interpolate(par, base_n)
        if a0 is None:
            a0 = _norm(par)
        if a0 is None:
            a0 = np.ones(base_n, dtype=np.float64) / base_n
        jobs.append((a0, 45.0, int(seed) + 101))
    else:
        a0 = _random_init(base_n, int(seed) + 101, None)
        if a0 is None:
            a0 = np.ones(base_n, dtype=np.float64) / base_n
        jobs.append((a0, 45.0, int(seed) + 101))

    a_noise = _random_init(base_n, int(seed) + 102, par)
    if a_noise is not None:
        jobs.append((a_noise, 16.0, int(seed) + 102))

    if par is not None:
        pbase = _interpolate(par, base_n)
        if pbase is not None:
            psharp = _norm(np.power(pbase, 2.0) + 1e-9)
            if psharp is not None:
                jobs.append((psharp, 10.0, int(seed) + 103))

    cand_sizes = []
    for s in (
        max(32, base_n // 2),
        min(1024, 2 * base_n),
        128,
        192,
        256,
        384,
        512,
    ):
        s = int(min(max(s, 32), 1024))
        if s not in cand_sizes:
            cand_sizes.append(s)
        if len(cand_sizes) >= 7:
            break

    weights = [10.0, 8.0, 7.0, 5.0, 4.0, 3.0, 3.0]
    for i, s in enumerate(cand_sizes[: len(weights)]):
        if s == base_n and par is not None:
            continue
        a0 = _random_init(int(s), int(seed) + 300 + int(s), par)
        if a0 is not None:
            jobs.append((a0, float(weights[i]), int(seed) + 300 + int(s)))

    # Time allocation.
    polish_time = min(220.0, max(15.0, 0.18 * max(0.0, hard - t0)))
    opt_end = hard - polish_time
    if opt_end < t0 + 10.0:
        opt_end = hard - 2.0

    target = 1.50274349
    total_opt = max(0.0, opt_end - t0)
    if total_opt > 5.0 and jobs:
        total_w = sum(float(w) for _, w, _ in jobs)
        if total_w > 0.0:
            used_frac = 0.0
            for idx, (a0, w, job_seed) in enumerate(jobs):
                if time.monotonic() >= opt_end - 2.0:
                    break
                if idx > 0 and np.isfinite(best_exact_s) and best_exact_s < target:
                    break
                frac = float(w) / total_w
                alloc = max(2.0, frac * total_opt)
                end_time = min(opt_end, time.monotonic() + alloc)
                if end_time - t0 < 1.0:
                    continue
                njob = int(a0.size)
                mainish = idx == 0
                exact_every = 35 if njob <= 256 else (160 if njob <= 512 else 800)
                max_lp = 3 if (mainish and njob <= 512) else (1 if njob <= 256 else 0)
                _run_adam(
                    a0,
                    end_time,
                    job_seed,
                    K_start=(8.0 if mainish else 6.0),
                    K_end=(600.0 if mainish else 350.0),
                    beta0=(0.45 if mainish else 0.40),
                    beta1=(0.02 if mainish else 0.05),
                    lr0=(0.50 if mainish else 0.55),
                    exact_every=exact_every,
                    max_lp=max_lp,
                )
                used_frac += frac
                if used_frac > 0.65 and np.isfinite(best_exact_s) and best_exact_s < target:
                    break

    # Focused refinement of the best exact-score candidate.
    refine_start = time.monotonic()
    hard_end = hard - 1.5
    if refine_start < hard_end - 5.0:
        refine_end = min(hard_end, refine_start + max(5.0, 0.40 * max(0.0, hard_end - refine_start)))
        seed_ref = int(seed) + 777
        if np.isfinite(best_exact_s):
            refine_a = best_exact_a
        else:
            refine_a = best_int_a
        nref = int(refine_a.size) if refine_a is not None else 192
        _run_adam(
            refine_a,
            refine_end,
            seed_ref,
            K_start=25.0,
            K_end=900.0,
            beta0=0.12,
            beta1=0.0,
            lr0=0.22,
            exact_every=20 if nref <= 256 else 120,
            max_lp=2 if nref <= 512 else 0,
        )

    # Local polish.
    if time.monotonic() < hard_end - 2.0:
        polish_a = best_exact_a if np.isfinite(best_exact_s) else best_int_a
        _polish(polish_a, hard_end, int(seed) + 9999, do_lp=True)

    # Final exact validation.
    for arr in (best_int_a, best_exact_a, np.array([1.0], dtype=np.float64)):
        _exact_score(arr, update=True)

    if not np.isfinite(best_exact_s):
        return [1.0]

    try:
        cand = [float(v) for v in best_exact_a]
    except Exception:
        cand = [1.0]

    try:
        fs = float(evaluate_sequence(cand))
    except Exception:
        fs = math.inf

    if not np.isfinite(fs) or not all(np.isfinite(float(x)) for x in cand):
        return [1.0]

    return cand
