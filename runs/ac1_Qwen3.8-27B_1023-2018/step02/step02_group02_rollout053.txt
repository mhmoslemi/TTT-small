            best_f = f_new.copy()
        
        # Check for convergence
        if sc_new >= M - 1e-13:
            break
        
        f = f_new
    
    # Phase 2: LSE-smoothed projected gradient
    if time.monotonic() < deadline - 40.0:
        a = best_f.copy()
        rng = np.random.default_rng(seed)
        lr = 0.1
        stuck = 0
        
        for it in range(20000):
            if time.monotonic() > deadline - 40.0:
                break
            if stuck > 300:
                break
            
            c = do_conv(a)
            M = float(np.max(c))
            sc = 2.0 * n * M
            
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_f = a.copy()
                stuck = 0
            
            # Try multiple tau values
            improved = False
            for tau_mult in [0.01, 0.03, 0.1, 0.3]:
                tau = max(1e-10, tau_mult * M)
                x = (c - M) / tau
                ex = np.exp(np.clip(x, -50.0, 50.0))
                p = ex / ex.sum()
                
                # Gradient: grad[i] = 2 * sum_k p[k] * a[k-i] (valid range)
                grad = np.zeros(n)
                for i in range(n):
                    grad[i] = 2.0 * float(np.dot(p[i:i + n], a))
                
                # Center for simplex
                grad = grad - float(np.dot(grad, a))
                
                gn = float(np.linalg.norm(grad))
                if gn < 1e-14:
                    continue

                    step = scale * grad / gn
                    cand = a - step
                    cand = np.maximum(cand, 1e-14)
                    cs = cand.sum()
                    if cs < 1e-10:
                        continue
                    cand /= cs
                    cand_sc = 2.0 * n * float(np.max(do_conv(cand)))
                    if cand_sc < sc - 1e-14:
                        a = cand
                        sc = cand_sc
                        improved = True
                        break
                
                if improved:
                    lr = min(lr * 1.03, 2.0)
                    stuck = 0
                    break
            
            if not improved:
                stuck += 1
                lr = max(0.005, lr * 0.85)
                if stuck % 100 == 0 and stuck < 300:
                    a = np.maximum(a + rng.normal(0, 0.03, n), 1e-12)
                    a /= a.sum()
    
    # Phase 3: Stochastic local search
    if time.monotonic() < deadline - 15.0:
        a = best_f.copy()
        sc = best_sc
        rng2 = np.random.default_rng(seed + 777)
        sigma = 0.05
        
        for it in range(20000):

                z = np.log(np.clip(a, 1e-12, None))
                z += rng2.normal(0, sigma, n)
                z -= z.max()
                cand = np.exp(z)
                cand /= cand.sum()
            elif r < 0.5:

                cand = a.copy()
                k = int(rng2.integers(0, n))
                w = max(2, n // 50)
                lo = max(0, k - w)
                hi = min(n, k + w + 1)
                factor = np.exp(rng2.normal(0, 0.1))
                cand[lo:hi] *= factor
                cand /= cand.sum()
            elif r < 0.7:
                # Pair swap
                cand = a.copy()
                i1 = int(rng2.integers(0, n))
                i2 = int(rng2.integers(0, n))
                if i1 != i2:
                    amt = 0.01 * a.max() * rng2.uniform(0.5, 1.5)
                    cand[i1] += amt
                    cand[i2] -= amt
                    cand = np.maximum(cand, 1e-12)
                    cand /= cand.sum()
                else:
                    continue
            elif r < 0.85:
                # Mirror blend
                w = rng2.uniform(0.05, 0.2)
                cand = (1 - w) * a + w * a[::-1]
                cand /= cand.sum()
            else:
                # Random coordinate perturbation
                cand = a.copy()
                idx = rng2.integers(0, n, size=5)
                for i in idx:
                    cand[i] *= np.exp(rng2.normal(0, 0.15))
                cand = np.maximum(cand, 1e-12)
                cand /= cand.sum()
            
            cand_sc = 2.0 * n * float(np.max(do_conv(cand)))
            if cand_sc < sc - 1e-14:
                a = cand
                sc = cand_sc
                if sc < best_sc:
                    best_sc = sc
                    best_f = a.copy()
                sigma = max(0.01, sigma * 0.97)
            else:
                sigma = min(0.3, sigma * 1.05)

        try:
            fs = evaluate_sequence(final_list)
        except Exception:
            pass
    
    return final_list
