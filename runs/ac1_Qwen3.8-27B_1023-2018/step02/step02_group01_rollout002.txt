def _perturb(a, sigma, rng):
    base = np.maximum(a, 1e-12)
    z = np.log(base)
    z = z + rng.normal(0.0, float(sigma), size=a.size)
    z = np.clip(z, z.max() - 35.0, z.max())
    b = np.exp(z)
    b = np.maximum(b, 1e-12)
    s = float(b.sum())
    if s <= 0.0:
        return a.copy()
    return b / s


def _blend(a, b, alpha):
    c = (1.0 - alpha) * a + alpha * b
    s = float(c.sum())
    if s <= 0.0:
        return a.copy()
    return c / s

    tau = max(1e-6 * M, 1e-14)
    try:
        shifted = (np.asarray(c, dtype=np.float64) - M) / tau
        shifted -= np.max(shifted)
        w = np.exp(shifted)
    except Exception:
        return None
    ws = float(w.sum())
    if ws <= 0.0 or not np.isfinite(ws):
        return None
    w = w / ws
    d = 2.0 * (A.T @ w)
    dscale = float(np.max(np.abs(d)))
    if dscale <= 0.0 or not np.isfinite(dscale):
        return None
    for eta in (0.2, 0.5, 1.0, 2.0, 4.0, 8.0):
        try:
            z = np.log(np.maximum(a, 1e-12)) - (eta * d / dscale)
            z = np.clip(z, z.max() - 35.0, z.max())
            b = np.exp(z)
            b = np.maximum(b, 1e-12)
            b = b / b.sum()
            mv = float(np.convolve(b, b).max())
            if np.isfinite(mv) and mv < M * (1.0 - 1e-12):
                return b
        except Exception:
            continue
    return None


def _update_state(state, a, m=None):
    a = _norm(a)
    if a is None:
        return
    if m is None:
        c = np.convolve(a, a)
        m = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(m):
        return
    if m < state["m"] - 1e-15:
        state["m"] = float(m)
        state["a"] = a.copy()
        sc = _eval_exact([float(x) for x in a])
        if np.isfinite(sc) and sc < state["score"] - 1e-14:
            state["score"] = float(sc)


def _local_search(a_start, stop_time, seed, state):
    a = _norm(a_start)
    if a is None:
        a = np.ones(191, dtype=np.float64) / 191.0
    _update_state(state, a)
    rng = np.random.default_rng(seed)

    stuck_best = 0
    while True:
        if time.monotonic() > stop_time - 2.0:
            break

        c = np.convolve(a, a)
        M = float(np.max(c)) if c.size else 0.0

        if np.isfinite(M) and M < state["m"] - 1e-14:
            _update_state(state, a, M)
            stuck_best = 0
        else:
            stuck_best += 1

        if stuck_best > 35:
            base = state["a"]
            sigma = min(0.5, 0.02 + 0.015 * (stuck_best // 20))
            a = _perturb(base, sigma, rng)
            if rng.random() < 0.25:
                rev = base[::-1].copy()
                srev = float(rev.sum())
                if srev > 0.0:
                    a = _blend(a, rev / srev, 0.12)
            _update_state(state, a)
            stuck_best = 0
            continue

        A = _build_A(a)
        moved = False
        remain = stop_time - time.monotonic()

        if remain > 4.0:
            timeout = max(0.05, min(1.2, float(remain - 1.0)))
            delta = _slp_direction(A, a, c, M, timeout)
            if delta is not None:
                delta = np.asarray(delta, dtype=np.float64)
                sm = float(delta.sum())
                if abs(sm) > 1e-14:
                    delta -= sm / a.size
                tmax = _tmax(a, delta)
                res = _line_search(a, c, delta, tmax, M)
                if res is not None:
                    a, Mn = res
                    _update_state(state, a, Mn)
                    moved = True

        if (not moved) and remain > 4.0:
            timeout = max(0.05, min(1.2, float(remain - 1.0)))
            target = M
            if stuck_best > 8:
                target = max(1e-15, min(0.995 * M, 0.995 * state["m"]))
            g = _lp_response(A, target, timeout)
            if g is not None:
                g = _norm(g)
                if g is not None and g.size == a.size:
                    delta = g - a
                    delta = delta - delta.sum() / a.size
                    tmax = min(1.0, _tmax(a, delta))
                    res = _line_search(a, c, delta, tmax, M)
                    if res is not None:
                        a, Mn = res
                        _update_state(state, a, Mn)
                        moved = True

        if not moved and remain > 1.5:
            if rng.random() < 0.25:
                cand = _multi_update(a, A, c, M, rng)
                if cand is not None:
                    a = cand
                    _update_state(state, a)
                    moved = True
            if not moved:
                sigma = 0.02 if stuck_best < 18 else 0.05
                a = _perturb(a, sigma, rng)

        # If we are much worse than the known best, pull back toward best occasionally.
        if stuck_best > 10 and rng.random() < 0.1:
            base = state["a"]
            a = _blend(a, base, 0.25)
            _update_state(state, a)


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1300.0
    if not np.isfinite(budget) or budget <= 0:
        budget = 1300.0

    headroom = max(5.0, min(60.0, 0.10 * budget))
    deadline = t0 + max(1.0, budget - headroom)
    hard_end = t0 + max(1.0, budget - 2.0)

    parent = _get_parent()
    fallback = parent.copy()

    state = {
        "a": fallback.copy(),
        "m": _M(fallback),
        "score": _eval_exact([float(x) for x in fallback]),
    }

    if not np.isfinite(state["score"]) or state["score"] >= 2.0:
        uniform = np.ones(fallback.size if fallback.size > 0 else 191, dtype=np.float64)
        _update_state(state, uniform)

    rng = np.random.default_rng(seed)

    # A few inexpensive variants to seed the local search.
    base = state["a"].copy()
    variants = [base]
    try:
        variants.append(base[::-1] / base[::-1].sum())
    except Exception:
        pass
    for sg in (0.03, 0.08, 0.15):
        variants.append(_perturb(base, sg, rng))
    n = base.size
    try:
        variants.append(_blend(base, np.ones(n) / n, 0.20))
        a05 = np.maximum(base, 1e-12) ** 0.5
        variants.append(a05 / a05.sum())
        a2 = np.maximum(base, 1e-12) ** 2.0
        variants.append(a2 / a2.sum())
    except Exception:
        pass
    for v in variants:
        try:
            vv = _norm(v)
            if vv is not None and vv.size == n:
                _update_state(state, vv)
        except Exception:
            pass

    try:
        if time.monotonic() < deadline - 3.0:
            _local_search(state["a"].copy(), deadline, int(seed) + 1, state)
    except Exception:
        pass

    # Final validation and tiny scale search.
    best_arr = None
    best_score = float("inf")
    candidates = []
    try:
        candidates.append(np.asarray(state["a"], dtype=np.float64).ravel())
    except Exception:
        pass
    try:
        candidates.append(np.asarray(fallback, dtype=np.float64).ravel())
    except Exception:
        pass

    for cand in candidates:
        cand = _norm(cand)
        if cand is None or cand.size == 0:
            continue
        s = float(cand.sum())
        if not np.isfinite(s) or s <= 0:
            continue
        candidates.append(cand / s)

    seen = set()
    valid_candidates = []
    for cand in candidates:
        if cand is None:
            continue
        cand = np.asarray(cand, dtype=np.float64).ravel()
        if cand.size in seen:
            continue
        seen.add(cand.size)
        valid_candidates.append(cand)

    for cand in valid_candidates:
        if cand is None or cand.size == 0:
            continue
        scales = [1.0]
        s = float(cand.sum())
        if np.isfinite(s) and s > 0.0:
            scales.append(1.0 / s)
        for d in (1e-8, 1e-7, 1e-6):
            scales.extend([1.0 + d, 1.0 - d])
        for sc in scales:
            if not np.isfinite(sc) or sc <= 0.0:
                continue
            arr = cand * sc
            arr = np.where(np.isfinite(arr), arr, 0.0)
            arr = np.clip(arr, 0.0, 1000.0)
            if arr.size == 0 or float(arr.sum()) < 0.01:
                continue
            seq = [float(x) for x in arr]
            trial = _eval_exact(seq)
            if np.isfinite(trial) and trial < best_score - 1e-15:
                best_score = trial
                best_arr = arr

    if best_arr is None or best_arr.size == 0 or not np.isfinite(best_score):
        # Safe fallback: integer ones gives exact score 2.0 for many thresholds.
        if fallback is not None and fallback.size >= 1 and fallback.size <= 2048:
            best_arr = np.clip(np.asarray(fallback, dtype=np.float64).ravel(), 0.0, 1000.0)
            seq = [float(x) for x in best_arr]
            sc = _eval_exact(seq)
            if np.isfinite(sc):
                best_score = sc
            else:
                best_arr = np.ones(191, dtype=np.float64)
        else:
            best_arr = np.ones(191, dtype=np.float64)

    best_arr = np.asarray(best_arr, dtype=np.float64).ravel()
    if best_arr.size == 0 or not np.all(np.isfinite(best_arr)):
        best_arr = np.ones(191, dtype=np.float64)

    best_arr = np.clip(best_arr, 0.0, 1000.0)
    if best_arr.size > 4096:
        best_arr = best_arr[:191]

    # If timing is close, do not risk a long final conversion.
    if time.monotonic() > hard_end - 1.0 and best_arr.size > 512:
        best_arr = np.ones(191, dtype=np.float64)

    return [float(x) for x in best_arr]
