```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len


def _full_conv(a, L):
    """Full self-convolution via FFT."""
    F = rfft(a, L)
    return irfft(F * F, L)[:len(a) * 2 - 1]


def _cross_corr(w, a, L):
    """c[i] = sum_j w[i+j]*a[j] for i=0..n-1 via FFT."""
    n = len(a)
    Fw = rfft(w, L)
    Fa = rfft(a[::-1], L)
    result = irfft(Fw * Fa, L)
    return result[n - 1:2 * n - 1]


def _normalize(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def _score(a):
    """Compute score matching evaluator (scale-invariant)."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return math.inf
    L = next_fast_len(2 * n - 1)
    conv = _full_conv(a, L)
    return 2.0 * n * conv.max() / (s * s)


def _score_norm(a, L):
    """Score for sum(a)=1: 2n*max(conv)."""
    return 2.0 * len(a) * _full_conv(a, L).max()


def _optimize(a0, n, stop_time, seed):
    """Adaptive log-sum-exp gradient method with momentum and restarts."""
    rng = np.random.default_rng(seed)
    L_conv = next_fast_len(2 * n - 1)
    L_cross = next_fast_len(3 * n - 2)

    a = np.clip(np.asarray(a0, dtype=np.float64).ravel(), 0.0, None)
    if a.size < n:
        a = np.concatenate([a, np.zeros(n - a.size)])
    a = a[:n]
    a = _normalize(a)

    best_a = a.copy()
    best_score = _score_norm(a, L_conv)

    alpha = 25.0
    alpha_max = 800.0
    lr = 0.08
    momentum = 0.8
    v = np.zeros(n)
    no_improve = 0
    it = 0

    while time.monotonic() < stop_time:
        # Compute gradient of log-sum-exp(alpha * conv)
        conv = _full_conv(a, L_conv)
        max_c = conv.max()
        shifted = alpha * (conv - max_c)
        exp_s = np.exp(shifted)
        p = exp_s / exp_s.sum()
        grad = 2.0 * alpha * _cross_corr(p, a, L_cross)

        # Momentum
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-15:
            direction = -grad / gnorm
        else:
            break
        v = momentum * v + direction
        vnorm = np.linalg.norm(v)
        if vnorm > 0:
            v = v / vnorm

        # Backtracking line search
        improved = False
        for bt in range(15):
            step = lr * (0.55 ** bt)
            a_new = a + step * v
            a_new = _normalize(a_new)
            sc = _score_norm(a_new, L_conv)
            if sc < best_score - 1e-14:
                a = a_new
                best_a = a.copy()
                best_score = sc
                no_improve = 0
                improved = True
                break

        if not improved:
            no_improve += 1
            if no_improve % 20 == 0:
                v = direction  # reset momentum

        # Adapt alpha
        if no_improve > 40:
            alpha = min(alpha * 1.12, alpha_max)

        # Escape mechanism
        if no_improve > 180:
            escaped = False
            for _ in range(40):
                scale = rng.uniform(0.01, 0.12)
                a_p = _normalize(a + rng.normal(0, scale, n))
                sc = _score_norm(a_p, L_conv)
                if sc < best_score:
                    a = a_p
                    best_a = a.copy()
                    best_score = sc
                    escaped = True
                    break
            if not escaped:
                # Big jump
                a = _normalize(a + rng.normal(0, 0.2, n))
            v = np.zeros(n)
            no_improve = 0
            alpha = max(10.0, alpha * 0.4)
            lr = 0.08

        # Gradual parameter drift
        it += 1
        if it % 300 == 0:
            lr = max(0.005, lr * 0.99)

    return best_a, best_score


def _optimize_adam(a0, n, stop_time, seed):
    """Adam-based gradient descent on log-sum-exp."""
    rng = np.random.default_rng(seed)
    L_conv = next_fast_len(2 * n - 1)
    L_cross = next_fast_len(3 * n - 2)

    a = np.clip(np.asarray(a0, dtype=np.float64).ravel(), 0.0, None)
    if a.size < n:
        a = np.concatenate([a, np.zeros(n - a.size)])
    a = a[:n]
    a = _normalize(a)

    best_a = a.copy()
    best_score = _score_norm(a, L_conv)

    alpha = 30.0
    alpha_max = 600.0
    lr = 0.02
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    vv = np.zeros(n)
    t = 0
    no_improve = 0

    while time.monotonic() < stop_time:
        conv = _full_conv(a, L_conv)
        max_c = conv.max()
        shifted = alpha * (conv - max_c)
        exp_s = np.exp(shifted)
        p = exp_s / exp_s.sum()
        grad = 2.0 * alpha * _cross_corr(p, a, L_cross)

        t += 1
        m = beta1 * m + (1 - beta1) * grad
        vv = beta2 * vv + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = vv / (1 - beta2 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = _normalize(a_new)
        sc = _score_norm(a_new, L_conv)

        if sc < best_score - 1e-14:
            a = a_new
            best_a = a.copy()
            best_score = sc
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                lr = max(0.001, lr * 0.85)
            alpha = min(alpha * 1.05, alpha_max)

        if no_improve > 200:
            # Escape
            for _ in range(30):
                scale = rng.uniform(0.01, 0.1)
                a_p = _normalize(a + rng.normal(0, scale, n))
                sc = _score_norm(a_p, L_conv)
                if sc < best_score:
                    a = a_p
                    best_a = a.copy()
                    best_score = sc
                    break
            v = np.zeros(n)
            m = np.zeros(n)
            vv = np.zeros(n)
            t = 0
            no_improve = 0
            alpha = max(15.0, alpha * 0.5)
            lr = 0.02

    return best_a, best_score


def _polish(a, stop_time, seed):
    """Coordinate descent polish."""
    rng = np.random.default_rng(seed)
    n = len(a)
    L_conv = next_fast_len(2 * n - 1)

    a = _normalize(a.copy())
    best_a = a.copy()
    best_score = _score_norm(a, L_conv)
    stall = 0

    while time.monotonic() < stop_time and stall < 12:
        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > stop_time:
                break
            orig = a[i]
            if orig < 1e-14:
                continue

            for factor in (0.4, 0.6, 0.8, 1.2, 1.5, 2.0, 0.2, 3.0, 0.0):
                a[i] = orig * factor if factor > 0 else 0.0
                s = a.sum()
                if s < 1e-10:
                    a[i] = orig
                    continue
                a_n = a / s
                sc = _score_norm(a_n, L_conv)
                if sc < best_score - 1e-14:
                    a = a_n
                    best_a = a.copy()
                    best_score = sc
                    improved = True
                    orig = a[i]
                    break
            a[i] = orig

        # Pair moves
        for _ in range(40):
            if time.monotonic() > stop_time:
                break
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            oi, oj = a[i], a[j]
            for mi, mj in ((0.5, 2.0), (2.0, 0.5), (0.7, 1.4), (1.4, 0.7)):
                a[i] = oi * mi if oi > 1e-14 else 0.1
                a[j] = oj * mj if oj > 1e-14 else 0.1
                s = a.sum()
                if s < 1e-10:
                    a[i], a[j] = oi, oj
                    continue
                a_n = a / s
                sc = _score_norm(a_n, L_conv)
                if sc < best_score - 1e-14:
                    a = a_n
                    best_a = a.copy()
                    best_score = sc
                    improved = True
            a[i], a[j] = oi, oj

        if not improved:
            stall += 1
        else:
            stall = 0

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(30.0, float(budget_s))
    deadline = t0 + budget - 15.0

    # Get parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
    except Exception:
        parent = np.ones(198, dtype=np.float64)

    if parent.size == 0:
        parent = np.ones(198, dtype=np.float64)
    parent = np.clip(parent, 0.0, 1000.0)
    parent = np.where(np.isfinite(parent), parent, 0.0)
    if parent.size == 0:
        parent = np.ones(198, dtype=np.float64)

    # Fallback
    fallback = parent.copy()
    fallback_sc = _score(fallback)
    if not math.isfinite(fallback_sc):
        fallback = np.ones(32) / 32
        fallback_sc = _score(fallback)

    best_seq = fallback.copy()
    best_score = fallback_sc

    def _try(a_raw):
        nonlocal best_seq, best_score
        a_raw = np.clip(np.asarray(a_raw, dtype=np.float64).ravel(), 0.0, None)
        if a_raw.size == 0 or a_raw.sum() < 0.01:
            return
        sc = _score(a_raw)
        if math.isfinite(sc) and sc < best_score - 1e-15:
            best_seq = a_raw.copy()
            best_score = sc

    n = len(parent)
    rng_main = np.random.default_rng(seed)

    # Build initializations
    i_arr = np.arange(n, dtype=np.float64)
    inits = [
        parent.copy(),
        np.ones(n) / n,
        np.maximum(0.5 + 0.5 * np.cos(2 * math.pi * i_arr / n), 0.01),
        np.maximum(1.0 + 0.4 * np.cos(2 * math.pi * 2 * i_arr / n)
                   + 0.3 * np.cos(2 * math.pi * 5 * i_arr / n), 0.01),
        np.abs(rng_main.normal(0, 1, n)) + 0.05,
    ]

    # Phase 1: Optimize parent length with multiple inits
    phase1_end = min(t0 + budget * 0.48, deadline - 70.0)
    time_per_init = max(8.0, (phase1_end - t0) / len(inits))

    for idx, init in enumerate(inits):
        if time.monotonic() > phase1_end - 5.0:
            break
        stop = min(phase1_end - 3.0, time.monotonic() + time_per_init)
        if idx == 0:
            # Primary: use momentum method from parent
            res, sc = _optimize(init, n, stop, seed=seed + 100)
        else:
            # Alternating: use Adam for diversity
            res, sc = _optimize_adam(init, n, stop, seed=seed + 200 + idx * 111)
        _try(res)

    # Phase 2: Try other lengths
    phase2_end = min(t0 + budget * 0.74, deadline - 45.0)
    other_lengths = [L for L in [192, 256, 384, 512] if L != n]

    if other_lengths and time.monotonic() < phase2_end - 10.0:
        time_per_len = max(3.0, (phase2_end - time.monotonic()) / len(other_lengths))
        for L in other_lengths:
            if time.monotonic() > phase2_end - 5.0:
                break
            stop = min(phase2_end - 3.0, time.monotonic() + time_per_len)
            if len(best_seq) >= L:
                init_l = np.interp(
                    np.linspace(0, 1, L),
                    np.linspace(0, 1, len(best_seq)),
                    best_seq
                )
            else:
                init_l = np.interp(
                    np.linspace(0, 1, L),
                    np.linspace(0, 1, len(parent)),
                    parent
                )
            res, sc = _optimize(init_l, L, stop, seed=seed + 5000 + L)
            _try(res)

    # Phase 3: Polish the best candidate
    polish_end = deadline - 10.0
    if time.monotonic() < polish_end - 5.0:
        res, sc = _polish(best_seq, polish_end, seed=seed + 99999)
        _try(res)

    # Final validation with exact evaluator
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in fallback]

    if final_sc < best_score:
        best_score = final_sc

    return [float(x) for x in best_seq]
```
