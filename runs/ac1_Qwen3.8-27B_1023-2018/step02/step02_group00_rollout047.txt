def _conv_full(a):
    n = len(a)
    if n <= 256:
        c = np.convolve(a, a, mode="full")
    else:
        fsize = next_fast_len(2 * n - 1)
        F = rfft(a, fsize)
        c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

    """Build the (2n-1) x n convolution matrix for convolve(f, g)."""
    # (convolve(f, g))[k] = sum_j f[j] * g[k-j]
    # For k = 0..2n-2, j must satisfy 0<=j<n and 0<=k-j<n
    # So A[k,j] = f[j] if 0 <= k-j < n, else 0
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # A[k,j] = f[j] when 0 <= k-j < n, i.e., j <= k <= j+n-1
        A[j:j+n, j] = f
    return A

    A = _build_A(f, n)
