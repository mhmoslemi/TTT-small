import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, next_fast_len


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(192, dtype=np.float64) / 192.0
    p = np.clip(p, 0.0, 1000.0)
    if p.size == 0 or not np.isfinite(p).all():
        p = np.ones(192, dtype=np.float64) / 192.0
    return p


def _conv_sq(a):
    """Full self-convolution via FFT."""
    n = a.size
    fs = next_fast_len(2 * n - 1)
    F = rfft(a, fs)
    c = irfft(F * F, fs)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _build_A(a):
    """Toeplitz matrix A[k,i] = a[k-i] if 0<=k-i<n else 0. Shape (2n-1, n)."""
    n = a.size
    idx = np.arange(2 * n - 1)[:, None] - np.arange(n)[None, :]
    return np.where((idx >= 0) & (idx < n), a[np.clip(idx, 0, n - 1)], 0.0)


def _lp_br(a, M):
    """Solve LP: max 1^T g s.t. A g <= M, g >= 0. Return g or None."""
    n = a.size
    A = _build_A(a)
    try:
        res = linprog(
            -np.ones(n),
            A_ub=A,
            b_ub=np.full(2 * n - 1, M),
            bounds=[(0.0, None)] * n,
            method="highs",
            options={"time_limit": 6.0, "presolve": "always"},
        )
    except Exception:
        return None
    if res.x is None or res.status != 0:
        return None
    g = np.maximum(res.x, 0.0)
    s = g.sum()
    if s < 1e-12:
        return None
    return g / s


def _lp_optimize(a0, t_end, max_iters=500000, seed=0):
    """LP best-response iteration with adaptive step and perturbation restart."""
    n = a0.size
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a.sum() < 1e-10:
        a = np.ones(n) / n

    rng = np.random.default_rng(seed)
    best_a = a.copy()
    best_M = float(_conv_sq(a).max())
    no_improve = 0
    last_improve_M = best_M

    for it in range(max_iters):
        if time.monotonic() > t_end:
            break

        M = float(_conv_sq(a).max())
        if M < best_M * (1.0 - 1e-13):

            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

            sigma = 0.05 * min(1.0, no_improve / 500.0)
            a = a * np.exp(rng.normal(0, sigma, n))
            a = np.maximum(a, 1e-8)
            a /= a.sum()
            no_improve = 0
            last_improve_M = best_M
            continue

        g = _lp_br(a, M)
        if g is None:
            break

        # Adaptive step size
        gap = (best_M - M) / (best_M + 1e-30)
        if gap < 1e-6:
            t = 0.03
        elif gap < 1e-4:
            t = 0.08
        elif gap < 1e-3:
            t = 0.15
        elif gap < 1e-2:
            t = 0.3
        else:
            t = 0.6

        c = np.clip(c, 0.0, 1000.0)
        cs = c.sum()
        if cs < 1e-10:
            continue
        c /= cs
        cM = float(_conv_sq(c).max())

        if cM < M * (1.0 - 1e-14):
            a = c
            accepted = True
        else:

            delta = rng.normal(0, 0.01, 3)

def _pg_polish(a0, t_end, seed=0):
    """Projected gradient with log-sum-exp smoothing for fine-tuning."""
    n = a0.size
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a.sum() < 1e-10:
        a = np.ones(n) / n
    a /= a.sum()

    best_a = a.copy()
    best_M = float(_conv_sq(a).max())

    rng = np.random.default_rng(seed)
    beta = 50.0
    lr = 0.05
    stall = 0

    while time.monotonic() < t_end:
        b = _conv_sq(a)
        M = float(b.max())
        if M < best_M * (1 - 1e-14):
            best_M = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 200:
                # Perturb and reduce learning rate
                a = a * np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 1e-8)
                a /= a.sum()
                lr *= 0.7
                stall = 0
                if lr < 1e-5:
                    break

        # Compute smoothed gradient
        w = np.exp(beta * (b - b.max()))
        w /= w.sum()

        # grad_i = sum_k w_k * d(a*a)_k / da_i
        # d(a*a)_k / da_i = a_{k-i} (if valid, not i=k-i) + a_{k-i} (if valid, i≠k-i)
        # = 2*a_{k-i} - a_i * [k==2i]  (with bounds)
        grad = np.zeros(n)
        for i in range(n):
            k_lo = max(0, i)
            k_hi = min(2 * n - 2, i + n - 1)
            ks = np.arange(k_lo, k_hi + 1)
            js = ks - i
            vals = 2.0 * a[js]
            # Correction for k = 2i (the diagonal term counted twice)
            if 0 <= 2 * i < 2 * n - 1:
                vals[js == i] = a[i]  # not 2*a[i]
            grad[i] = np.dot(w[ks], vals)

        # Projected gradient step on the simplex
        a_new = a - lr * grad
        # Project onto {x >= 0, sum x = 1}
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        # Simple: if sum > 1 after clipping, rescale
        a_new = np.maximum(a_new, 0.0)
        a_new /= a_new.sum()

        cM = float(_conv_sq(a_new).max())
        if cM <= M + 1e-14:
            a = a_new
        else:
            # Backtrack
            a_new2 = a - 0.5 * lr * grad
            a_new2 = np.maximum(a_new2, 0.0)
            a_new2 /= a_new2.sum()
            if _conv_sq(a_new2).max() <= M:
                a = a_new2

    return best_a


def _resample(parent, n):
    """Resample parent to length n via interpolation."""
    if n == parent.size:
        return parent.copy()
    idx = np.linspace(0, parent.size - 1, n)
    return np.interp(idx, np.arange(parent.size), parent)

def _exact_score_fast(a):
    """Fast exact score (same as evaluator formula)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
