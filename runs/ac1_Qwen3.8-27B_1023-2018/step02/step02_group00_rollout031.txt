            break

        # Project gradient to maintain sum=1: subtract mean
        grad_centered = grad - grad.mean()
        # Update
        f_new = best_f - lr * grad_centered
        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()

        else:
            f_new = f_new / s
        sc = _score_norm(f_new)

            best_sc = sc

            lr = min(lr * 1.05, 0.15)

                f_new = f_new / f_new.sum()
                best_f = f_new
                best_sc = _score_norm(best_f)
                lr = lr0
                stall = 0
    return best_f, best_sc
