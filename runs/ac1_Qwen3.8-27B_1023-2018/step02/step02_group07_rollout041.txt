        if not np.isfinite(z).all():
            z = best_z.copy()
            m1[:] = 0; m2[:] = 0; t_ad = 0
            continue

        # Evaluate
        zc = np.clip(z, -50, 50)
        zc -= zc.max()
        an = np.exp(zc)
        an /= an.sum()
        bn = _autocorr(an, fs)
        sn = 2.0 * n * float(bn.max())
        n_iter += 1

        if math.isfinite(sn) and sn < best_s - 1e-12:
            best_s = sn
            best_a = an.copy()
            best_z = z.copy()
            stuck = 0
            tau = max(tau_min, tau * 0.996)
            lr = min(lr_max, lr * 1.005)
        else:
            stuck += 1
            if stuck % 50 == 0:
                lr = max(0.15, lr * 0.85)

        if stuck > 200:
            z = best_z + rng.normal(0, 0.3, n)
            z -= z.mean()
            m1[:] = 0; m2[:] = 0; t_ad = 0
            lr = 1.0; tau = max(tau_min, 0.35); stuck = 0

    return best_a, best_s


def _polish(a_best, end_time, hard_dl, seed):
    """Local polishing with multiplicative perturbations."""
    n = len(a_best)
    fs = next_fast_len(2*n-1)
    rng = np.random.default_rng(seed)
    best_a = a_best.copy()
    best_s = _score_unit(best_a, fs)
    sigma = 0.15
    no_imp = 0

    while time.monotonic() < min(end_time, hard_dl) - 0.1:
        a_try = best_a.copy()
        r = rng.random()

        if r < 0.35:
            # Single coordinate
            i = int(rng.integers(0, n))
            a_try[i] *= np.exp(rng.normal(0, sigma))
        elif r < 0.60:
            # Small block (1-5 coords)
            i = int(rng.integers(0, n))
            w = int(rng.integers(1, min(5, n)))
            for j in range(i, min(n, i+w)):
                a_try[j] *= np.exp(rng.normal(0, sigma*0.7))
        elif r < 0.75:
            # Paired swap-like
            i1 = int(rng.integers(0, n))
            i2 = int(rng.integers(0, n))
            f = np.exp(rng.normal(0, sigma*0.8))
            a_try[i1] *= f
            a_try[i2] *= 1.0/f
        elif r < 0.90:
            # Global gentle
            a_try *= np.exp(rng.normal(0, sigma*0.25, n))
        else:
            # Symmetric perturbation (palindromic)
            half = n // 2
            pert = rng.normal(0, sigma*0.4, half+1)
            for j in range(half+1):
                a_try[j] *= np.exp(pert[j])
                a_try[n-1-j] *= np.exp(pert[j])

        a_try = np.maximum(a_try, 1e-15)
        s_try = _score_unit(a_try, fs)

        if math.isfinite(s_try) and s_try < best_s - 1e-13:
            best_a = a_try.copy()
            best_s = s_try
            no_imp = 0
            sigma = min(0.4, sigma * 1.01)
        else:
            no_imp += 1
            sigma = min(0.8, sigma * 1.04)
            if no_imp > 80:
                no_imp = 0
                sigma = 0.15

    return best_a, best_s


def _multiplicative_step(a, n, fs, fs2, tau):
    """One KKT-inspired multiplicative update."""
    b = _autocorr(a, fs)
    bg = b / tau
    bg -= bg.max()
    w = np.exp(bg)
    ws = float(w.sum())
    if ws < 1e-30 or not math.isfinite(ws):
        w = np.zeros_like(b)
        w[int(np.argmax(b))] = 1.0
    else:
        w = w / ws

    g = 2.0 * _xcorr(w, a, fs2)
    g = np.maximum(g, 1e-15)

    lam = 1.0 / float(np.sum(a / g))
    a_new = a * (lam / g)
    a_new = np.maximum(a_new, 1e-15)
    a_new = a_new / float(a_new.sum())
    return a_new


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    total = max(1.0, float(budget_s))
    hard_dl = t0 + total - 3.0

    # Get parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            if p.size > 0 and np.isfinite(p).all():
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    # Initial best
    best_a = np.array([1.0], dtype=np.float64)
    best_score = 2.0

    if parent is not None and parent.size > 0:
        try:
            s = float(evaluate_sequence([float(v) for v in parent]))
            if math.isfinite(s):
                best_a = parent.astype(np.float64).copy()
                best_score = s
        except Exception:
            pass

    if not math.isfinite(best_score):
        best_a = np.array([1.0], dtype=np.float64)
        best_score = 2.0

    def _try_update(a, s):
        nonlocal best_a, best_score
        try:
            a = np.asarray(a, dtype=np.float64).ravel()
            if a.size == 0 or not np.isfinite(a).all():
                return
            a = np.maximum(a, 1e-15)
            if float(a.sum()) < 1e-6:
                return
            a = a / float(a.sum())
        except Exception:
            return
        if math.isfinite(s) and s < best_score - 1e-13:
            best_a = a.copy()
            best_score = float(s)

    # Phase 1: Exploration
    explore_end = t0 + min(0.72 * total, total - 60.0)
    explore_end = max(explore_end, t0 + 30.0)

    # Build job list
    sizes = [192, 128, 256, 384, 512, 768, 1024, 1536]
    jobs = []
    job_i = 0
    for sz in sizes:
        n_variants = 3 if sz <= 384 else 2
        for vi in range(n_variants):
            variant = (job_i * 7 + seed) % 20
            a0 = _init_sequence(sz, variant, np.random.default_rng(seed*1000 + job_i*13), parent)
            jobs.append((sz, a0, seed*37 + job_i*101))
            job_i += 1

    # If parent available, also add it as a job at its own size
    if parent is not None and parent.size > 0:
        pn = parent.size
        a0 = parent.astype(np.float64).copy()
        a0 = np.maximum(a0, 1e-15)
        a0 = a0 / float(a0.sum())
        jobs.insert(0, (pn, a0, seed*999 + 7))

    # Round-robin exploration
    rng_master = np.random.default_rng(seed + 55555)
    active = list(range(len(jobs)))
    job_times_left = {}
    # Allocate time per job inversely proportional to size (smaller sizes get more relative time)
    for ji in range(len(jobs)):
        sz = jobs[ji][0]
        # Time per job: more time for smaller sizes (easier to converge), less for large
        t_alloc = max(5.0, (explore_end - t0) * 0.4 / (1.0 + 0.15*math.log2(max(2, sz/64))))
        job_times_left[ji] = min(t_alloc, explore_end - t0 - 1.0)

    while time.monotonic() < explore_end:
        for ji in range(len(active)):
            if time.monotonic() >= explore_end:
                break
            j = active[ji]
            if time.monotonic() >= explore_end:
                break
            rem_total = explore_end - time.monotonic()
            rem_job = job_times_left.get(j, 0.0)
            if rem_job <= 0.5 or rem_total < 0.3:
                continue
            sz, a0, jd = jobs[j]
            slice_end = time.monotonic() + min(1.5, rem_job, rem_total)
            job_times_left[j] = rem_job - min(1.5, rem_job, rem_total)

            try:
                a_res, s_res = _run_opt(a0, sz, slice_end, hard_dl, jd)
                _try_update(a_res, s_res)
            except Exception:
                pass

    # Phase 2: Refinement on best found
    refine_end = t0 + min(0.88 * total, total - 25.0)
    refine_end = max(refine_end, explore_end + 10.0)

    try:
        n_best = len(best_a)
        fs_b = next_fast_len(2*n_best-1)
        # Re-run optimization from best for refinement
        n_refine = int(refine_end - time.monotonic())
        if n_refine > 5:
            a_res, s_res = _run_opt(best_a, n_best, refine_end, hard_dl, seed*777)
            _try_update(a_res, s_res)

            # Also try multiplicative updates
            fs2_b = next_fast_len(3*n_best-2)
            a_m = best_a.copy()
            a_m = np.maximum(a_m, 1e-15)
            a_m = a_m / float(a_m.sum())
            tau_m = 0.3
            for _ in range(5000):
                if time.monotonic() >= refine_end:
                    break
                a_m = _multiplicative_step(a_m, n_best, fs_b, fs2_b, tau_m)
                s_m = _score_unit(a_m, fs_b)
                if math.isfinite(s_m) and s_m < best_score - 1e-13:
                    best_a = a_m.copy()
                    best_score = s_m
                tau_m = max(0.05, tau_m * 0.995)
    except Exception:
        pass

    # Phase 3: Polishing
    polish_end = hard_dl - 1.0
    try:
        if time.monotonic() < polish_end - 1.0:
            a_pol, s_pol = _polish(best_a, polish_end, hard_dl, seed*12345)
            _try_update(a_pol, s_pol)
    except Exception:
        pass

    # Final validation
    try:
        cand = [float(v) for v in best_a]
        fs_val = float(evaluate_sequence(cand))
        if not math.isfinite(fs_val) or fs_val > best_score + 0.1:
            # Fallback to parent or [1.0]
            if parent is not None and parent.size > 0:
                cand = [float(v) for v in parent]
                try:
                    fs_val = float(evaluate_sequence(cand))
                except Exception:
                    fs_val = math.inf
            if not math.isfinite(fs_val):
                cand = [1.0]
    except Exception:
        cand = [1.0]

    return cand
