def _parent():

        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:

def _fc(a):
    """FFT-based self-convolution"""
    n = len(a)
    fs = next_fast_len(2*n-1)
    F = rfft(a, fs)
    c = irfft(F*F, fs)[:2*n-1]
    c[~np.isfinite(c)] = 0.0
    np.maximum(c, 0.0, out=c)
    return c


def _mc(a):
    return float(_fc(a).max())


def _norm(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a/s if s > 1e-15 else np.ones(len(a))/len(a)


def _bA(a, n):
    ki = np.arange(2*n-1)[:,None]
    ji = np.arange(n)[None,:]
    d = ki - ji
    v = (d >= 0) & (d < n)
    A = np.zeros((2*n-1, n))
    A[v] = a[d[v].astype(np.intp)]
    return A


def _lpd(a, M, n):
    try:
        A = _bA(a, n)
        r = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M),
                    bounds=[(0.0, None)]*n, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            s = g.sum()
            if s > 1e-15: return g/s
    except Exception:
        pass
    return None


def _ag(a, b, n):
    M = b.max()
    act = np.where(b >= M*(1-1e-8))[0]
    g = np.zeros(n)
    for k in act:
        lo, hi = max(0, k-n+1), min(k, n-1)
        ix = np.arange(lo, hi+1)
        g[ix] += 2.0*a[k-ix]
    return g/max(1, len(act))
