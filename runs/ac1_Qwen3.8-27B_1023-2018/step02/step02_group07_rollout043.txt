import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len


def _softmax(x):
    x = np.asarray(x, dtype=np.float64).ravel()
    x = x - x.max()
    e = np.exp(np.clip(x, -40.0, 40.0))
    s = float(e.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return np.full(x.size, 1.0 / x.size, dtype=np.float64)
    return e / s


def _score_from_conv(c, n):
    """Compute 2*n*max(c) assuming sum(a)=1."""
    m = float(c.max())
    if not np.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m


def _run_gradient(a_init, n, end_time, hard_deadline, rng):
    """Run Adam gradient descent on the smoothed max objective."""
    fs = next_fast_len(3 * n - 2)
    z = np.log(np.clip(a_init, 1e-15, 1000.0))
    z = z - z.mean()
    
    best_z = z.copy()
    best_score = math.inf
    
    # Initial score
    a = _softmax(z)
    F = rfft(a, fs)
    c = irfft(F * F, fs)[:2 * n - 1]
    best_score = _score_from_conv(c, n)
    
    # Adam state
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    t = 0
    lr = 0.5
    tau_frac = 0.35
    w_var = 0.12
    stuck = 0
    best_stuck = 0
    
    while time.monotonic() < min(end_time, hard_deadline):
        # Check deadline frequently
        if time.monotonic() > hard_deadline - 2.0:
            break
        
        a = _softmax(np.clip(z, -25.0, 25.0))
        
        # Compute convolution
        F = rfft(a, fs)
        c = irfft(F * F, fs)[:2 * n - 1]
        cmax = float(c.max())
        if not np.isfinite(cmax) or cmax <= 0:
            break
        cur_score = 2.0 * n * cmax
        
        # Compute smoothed max distribution
        tau = max(1e-8, tau_frac * cmax)
        bt = c / tau

        A = rfft(a, fs)

        grad_a = (2.0 / tau) * g1 + w_var * (4.0 / (2 * n - 1)) * g2
        
        # Project to simplex: grad_z = a * (grad_a - <a, grad_a>)
        ga_dot = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - ga_dot)

def _local_search(a, n, end_time, hard_deadline, rng):
    """Local search on the simplex with various move types."""
    fs = next_fast_len(3 * n - 2)
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = np.clip(a, 0.0, 1000.0)
    a = a / max(1e-30, float(a.sum()))
    
    def _eval(a):
        F = rfft(a, fs)
        c = irfft(F * F, fs)[:2 * n - 1]
        return 2.0 * n * float(c.max())
    
    best_score = _eval(a)
    if not np.isfinite(best_score):
        best_score = 2.0 * n * a.max()  # fallback
    best_a = a.copy()
    
    sigma = 0.15
    stuck = 0
    n_it = 0
    
    while time.monotonic() < min(end_time, hard_deadline) - 0.5:
        n_it += 1
        r = rng.random()
        
        if r < 0.30:
            # Global Gaussian perturbation
            a_new = a + rng.normal(0, sigma, n)
        elif r < 0.45:
            # Block perturbation
            w = int(rng.integers(2, max(3, n // 20)))
            i0 = int(rng.integers(0, max(1, n - w)))
            a_new = a.copy()
            a_new[i0:i0+w] += rng.normal(0, sigma * 0.5, w)
        elif r < 0.55:
            # Coordinate move: shift mass from one coordinate to another
            a_new = a.copy()
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                amt = min(a[i], a[j] + 1) * rng.uniform(0.05, 0.3)
                a_new[i] -= amt
                a_new[j] += amt
        elif r < 0.65:
            # Smooth perturbation: add a sine wave
            freq = rng.uniform(0.5, 5.0)
            phase = rng.uniform(0, 2 * math.pi)
            delta = sigma * 0.3 * np.sin(2 * math.pi * freq * np.arange(n) / n + phase)
            a_new = a + delta
        elif r < 0.75:
            # Exponential scaling of a block
            w = int(rng.integers(3, max(4, n // 10)))
            i0 = int(rng.integers(0, max(1, n - w)))
            a_new = a.copy()
            a_new[i0:i0+w] *= np.exp(rng.normal(0, 0.3, w))
        elif r < 0.85:
            # Top-peak reduction: identify peak lag and redistribute
            F = rfft(a, fs)
            c = irfft(F * F, fs)[:2 * n - 1]
            k_peak = int(np.argmax(c))
            # Perturb coordinates that contribute most to c[k_peak]
            contrib = np.zeros(n)
            for j in range(n):
                if 0 <= k_peak - j < n:
                    contrib[j] = a[k_peak - j]
            cmax_c = contrib.max()
            if cmax_c > 1e-15:
                contrib = contrib / cmax_c
                a_new = a + sigma * 0.2 * (rng.normal(0, 1, n) * (1 - contrib))
            else:
                a_new = a + rng.normal(0, sigma, n)
        else:
            # Blend with uniform (relaxation)
            alpha = rng.uniform(0.02, 0.15)
            a_new = (1 - alpha) * a + alpha / n
        
        # Project to simplex
        a_new = np.maximum(a_new, 0.0)
        s = float(a_new.sum())
        if s < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s
        
        # Evaluate
        score_new = _eval(a_new)
        
        if np.isfinite(score_new) and score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new.copy()
            stuck = 0
            sigma = max(0.03, sigma * 0.95)
        else:
            stuck += 1
            sigma = min(0.8, sigma * 1.03)
            if stuck > 40:
                stuck = 0
                sigma = 0.3
                # Try a larger perturbation
                a = best_a + rng.normal(0, 0.5, n)
                a = np.maximum(a, 0)
                a = a / max(1e-15, a.sum())
    
    return best_a, best_score
