        c2 = lo + gr*(hi-lo)
        f1, f2 = mt(c1), mt(c2)
        for _ in range(25):
            if f1 < f2:
                hi, c2, f2 = c2, c1, f1
                c1 = hi - gr*(hi-lo)
                f1 = mt(c1)
            else:
                lo, c1, f1 = c1, c2, f2
                c2 = lo + gr*(hi-lo)
                f2 = mt(c2)
        t_opt = (lo+hi)/2
        return t_opt, mt(t_opt)

    # Fallback
    fallback = parent.copy()
    fsc = score(fallback)
    if not math.isfinite(fsc):
        fallback = np.ones(64)/64.0
        fsc = score(fallback)
    best_seq = fallback.copy()
    best_sc = fsc

    def try_update(f, sc):
        nonlocal best_seq, best_sc
        f = np.clip(f, 0, 1000)
        if f.sum() < 0.01:
            return
        s = score(f)
        if math.isfinite(s) and s < best_sc:
            best_sc = s
            best_seq = f.copy()

    def run_lp_opt(f0, n, stop_t, sd):
        nonlocal best_seq, best_sc
        rng = np.random.default_rng(sd)
        f = np.clip(np.asarray(f0, dtype=np.float64).ravel(), 0, None)
        if f.size < n:
            f = np.concatenate([f, np.zeros(n - f.size)])
        f = f[:n]
        f = norm1(f)
        bf, bs = f.copy(), score(f)
        t_val = 0.15
        stall = 0
        
        while time.monotonic() < stop_t - 2.0:
            b_f = conv(f)
            M = float(b_f.max())
            if M < 1e-15:
                break
            
            improved = False
            # Try standard LP
            g = lp_dir(f, n, M)
            used_g = False
            if g is not None and np.linalg.norm(g - f) > 1e-7:
                used_g = True
                t_opt, mt_val = golden_t(f, g, n)
                f_new = (1 - t_opt) * f + t_opt * g
                f_new = norm1(f_new)
                sc = score(f_new)
                if sc < bs - 1e-14:
                    f, bf, bs = f_new, f_new.copy(), sc
                    improved = True
                    try_update(f, sc)
                    t_val = min(0.6, t_opt * 1.1)
                    stall = 0
                    continue
            
            # Try tighter LP
            if not used_g:
                g2 = lp_dir(f, n, M, tighten=0.7)
                if g2 is not None and np.linalg.norm(g2 - f) > 1e-7:
                    t_opt, mt_val = golden_t(f, g2, n)
                    f_new = (1 - t_opt) * f + t_opt * g2
                    f_new = norm1(f_new)
                    sc = score(f_new)
                    if sc < bs - 1e-14:
                        f, bf, bs = f_new, f_new.copy(), sc
                        improved = True
                        try_update(f, sc)
                        t_val = 0.15
                        stall = 0
                        continue
            
            if not improved:
                stall += 1
                if stall > 120:
                    f = norm1(np.clip(f + rng.normal(0, 0.2, n), 0, None))
                    t_val = 0.15
                    stall = 0
                elif stall > 60:
                    f = norm1(np.clip(f + rng.normal(0, 0.08, n), 0, None))
                    t_val = max(0.05, t_val * 0.7)
        
        try_update(bf, bs)
    
    # Phase 1: LP at parent length
    n = n_p
    rng_m = np.random.default_rng(seed)
    ia = np.arange(n, dtype=np.float64)
    
    inits = [parent.copy(),
             0.5 + 0.5*np.cos(2*math.pi*ia/n),
             np.linspace(0.1, 2.0, n),
             np.abs(rng_m.standard_normal(n)) + 0.1,
             np.abs(parent + 0.2*np.std(parent)*rng_m.standard_normal(n)) + 0.01]
    
    # Add cosine multi-freq
    mf = np.ones(n)*0.5
    for fr in (2, 5, 13):
        mf += 0.35*np.cos(2*math.pi*fr*ia/n + rng_m.uniform(0, 2*math.pi))
    inits.append(np.abs(mf) + 0.05)
    
    p1_end = t0 + budget * 0.50
