```python
import math

    if x.size == 0:

    F = rfft(a, fs)

def _cross(p, a, n, fs2):
    """Compute cross_correlate(p, a)[j] = sum_m p[j+m]*a[m] for j=0..n-1"""
    ar = a[::-1]
    c = irfft(rfft(ar, fs2) * rfft(p, fs2), fs2)

    """
    Solve: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0
    A is (2n-1) x n: A[k,j] = f[k-j] if 0 <= k-j < n, else 0

        jlo = max(0, k - n + 1)
        jhi = min(n - 1, k)
        if jlo <= jhi:

        A_rows.append(row)
    A_ub = np.array(A_rows)
    b_ub = np.full(2*n-1, M)
    c = np.full(n, -1.0)

        s = g.sum()
```

```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _softmax(x):
    x = np.asarray(x, dtype=np.float64).ravel()
    if x.size == 0:
        return np.array([1.0])
    x = x - np.max(x)
    e = np.exp(np.clip(x, -80.0, 80.0))
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full(x.size, 1.0 / x.size)
    return e / s


def _conv(a, n, fs):
    F = rfft(a, fs)
    return irfft(F * F, fs)[:2 * n - 1]


def _cross(p, a, n, fs2):
    ar = a[::-1]
    c = irfft(rfft(ar, fs2) * rfft(p, fs2), fs2)
    return c[n - 1:2 * n - 1]


def _score_cached(a, n, fs):
    b = _conv(a, n, fs)
    m = float(b.max())
    return 2.0 * n * m, b


def _lp_step(f, n):
    """Solve max sum(g) s.t. (f*g)[k] <= M, g >= 0. Return normalized g or None."""
    M = float(_conv(f, n, next_fast_len(2 * n - 1)).max())
    if M <= 0 or not np.isfinite(M):
        return None
    # Build A: (2n-1) x n, A[k,j] = f[j] if 0 <= k-j < n
    A_rows = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        jlo = max(0, k - n + 1)
        jhi = min(n - 1, k)
        if jlo <= jhi:
            A_rows[k, jlo:jhi + 1] = f[jlo:jhi + 1]
    c_vec = np.full(n, -1.0)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_vec, A_ub=A_rows, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 2.0})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            s = float(g.sum())
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return None


def _adam_run(z0, n, end_time, tau_init, lr_init, seed):
    """Run Adam gradient descent on log-softmax parameters."""
    fs_c = next_fast_len(2 * n - 1)
    fs2 = next_fast_len(3 * n - 2)
    rng = np.random.default_rng(seed)
    z = z0.copy().astype(np.float64)
    z = z - z.mean()
    best_z = z.copy()
    a = _softmax(z)
    cur_score, b = _score_cached(a, n, fs_c)
    best_score = cur_score
    if not np.isfinite(best_score):
        best_score = 2.0 * n

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    tau = tau_init
    lr = lr_init
    stuck = 0
    lp_every = 40
    it = 0

    while time.monotonic() < end_time:
        a = _softmax(np.clip(z, -30.0, 30.0))
        b = _conv(a, n, fs_c)
        bmax = float(b.max())
        if not np.isfinite(bmax) or bmax <= 0:
            break
        cur_score = 2.0 * n * bmax

        tau = max(0.005 * bmax, tau * 0.997)
        bt = np.clip(b / tau, -60.0, 60.0)
        e = np.exp(bt)
        s_e = float(e.sum())
        if not np.isfinite(s_e) or s_e <= 0:
            p = np.zeros_like(b)
            p[int(np.argmax(b))] = 1.0
        else:
            p = e / s_e

        g = 2.0 * _cross(p, a, n, fs2)
        grad = a * (g - float(np.dot(a, g)))
        gnorm = float(np.abs(grad).max())

        if gnorm < 1e-18:
            stuck += 1
            if stuck > 30:
                z = best_z + rng.normal(0.0, 0.4, n)
                z -= z.mean()
                m[:] = 0; v[:] = 0; t = 0; stuck = 0; lr = lr_init
            continue

        grad /= (gnorm + 1e-300)
        t += 1
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * grad * grad
        mh = m / (1.0 - 0.9 ** t)
        vh = v / (1.0 - 0.999 ** t)
        upd = mh / (np.sqrt(vh) + 1e-8)
        um = float(np.abs(upd).max())
        if um < 1e-18:
            continue
        upd /= (um + 1e-300)

        z_new = z - lr * upd
        z_new -= z_new.mean()
        z_new = np.clip(z_new, -30.0, 30.0)
        a_new = _softmax(z_new)
        b_new = _conv(a_new, n, fs_c)
        s_new = 2.0 * n * max(0.0, float(b_new.max()))

        if np.isfinite(s_new) and s_new < best_score - 1e-12:
            best_score = s_new
            best_z = z_new.copy()
            z = z_new
            stuck = 0
        else:
            z = z_new
            stuck += 1

        it += 1
        if it % 20 == 0:
            lr = max(0.1, lr * 0.97)
        if stuck > 30:
            lr = max(0.1, lr * 0.85)
        if stuck > 150:
            z = best_z + rng.normal(0.0, 0.5, n)
            z -= z.mean()
            m[:] = 0; v[:] = 0; t = 0; stuck = 0; lr = lr_init; tau = tau_init

        # Periodic LP step
        if it % lp_every == 0 and it < 600:
            a_cur = _softmax(best_z)
            g_lp = _lp_step(a_cur, n)
            if g_lp is not None:
                mix = 0.3
                a_mix = (1.0 - mix) * a_cur + mix * g_lp
                a_mix = np.maximum(a_mix, 1e-9)
                a_mix /= a_mix.sum()
                s_mix, _ = _score_cached(a_mix, n, fs_c)
                if np.isfinite(s_mix) and s_mix < best_score - 1e-12:
                    best_score = s_mix
                    best_z = np.log(np.maximum(a_mix, 1e-12))
                    best_z -= best_z.mean()
                    z = best_z.copy()
                    m[:] = 0; v[:] = 0; t = 0; stuck = 0
                    lp_every = max(lp_every + 10, 80)

    a_best = _softmax(best_z)
    s_best, _ = _score_cached(a_best, n, fs_c)
    return a_best, s_best if np.isfinite(s_best) else 2.0 * n


def _local_polish(a0, n, end_time, seed):
    """Random perturbation local search."""
    fs_c = next_fast_len(2 * n - 1)
    rng = np.random.default_rng(seed)
    a = np.asarray(a0, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 0.0)
    s0 = float(a.sum())
    if s0 < 1e-10:
        return a, 2.0 * n
    a /= s0
    best = a.copy()
    best_score, _ = _score_cached(best, n, fs_c)
    sigma = 0.15
    stuck = 0

    while time.monotonic() < end_time - 0.5:
        r = rng.random()
        z = np.log(np.maximum(a, 1e-12))
        if r < 0.30:
            z += rng.normal(0.0, sigma, n)
        elif r < 0.45:
            i = int(rng.integers(0, n))
            w = max(1, int(rng.integers(1, max(2, n // 20))))
            i2 = min(n, i + w)
            z[i:i2] += rng.normal(0.0, sigma * 0.7, i2 - i)
        elif r < 0.55:
            z = z * float(rng.uniform(0.6, 1.5))
        elif r < 0.70:
            k = int(rng.integers(1, max(2, n // 10)))
            ph = rng.uniform(0, 2 * math.pi)
            z += sigma * 0.5 * np.sin(2.0 * math.pi * k * np.arange(n) / n + ph)
        else:
            i1 = int(rng.integers(0, n))
            i2 = int(rng.integers(0, n))
            tmp = z[i1]
            z[i1] = z[i2]
            z[i2] = tmp

        z -= z.mean()
        a2 = _softmax(z)
        s2, _ = _score_cached(a2, n, fs_c)
        if np.isfinite(s2) and s2 < best_score - 1e-12:
            best_score = s2
            best = a2.copy()
            a = a2.copy()
            stuck = 0
            sigma = max(0.03, sigma * 0.95)
        else:
            stuck += 1
            sigma = min(0.6, sigma * 1.04)
            if stuck > 80:
                stuck = 0
                sigma = 0.3
                a = best.copy()

    return best, best_score


def _try_update(best_a, best_score, a, s):
    nonlocal_val = [best_score, best_a]
    if a is None or s is None or not np.isfinite(s):
        return nonlocal_val
    try:
        a = np.asarray(a, dtype=np.float64).ravel()
        if a.size == 0 or not np.isfinite(a).all():
            return nonlocal_val
        a = np.clip(a, 0.0, 1000.0)
        s_sum = float(a.sum())
        if s_sum < 1e-6:
            return nonlocal_val
        if s < nonlocal_val[0] - 1e-13:
            nonlocal_val[0] = float(s)
            nonlocal_val[1] = a.copy()
    except Exception:
        pass
    return nonlocal_val
```
