        F = rfft(a, fsize)
        c = irfft(F * F, fsize)[:2 * n - 1]

        c[~np.isfinite(c)] = 0.0
        return c

        return float(c.max()) if c.size else 0.0

        n_local = v.size
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u)
        rho = np.nonzero(u * np.arange(1, n_local + 1) > (cssv[-1] - 1))[0][-1]
        theta = (cssv[rho] - 1) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)

            return np.zeros(n), M
