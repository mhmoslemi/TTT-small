    if n == 0:
        return np.array([0.0])
    fsize = max(2 * n - 1, 1)
    # Use next_fast_len for efficiency
    N = next_fast_len(2 * n - 1)
    F = rfft(a, N)
    c = irfft(F * F, N)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _exact_score(a):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    n = a.size
    b = _convolve_square(a)
    mb = float(b.max())
    return 2.0 * n * mb / (s * s)

def _normalize(a):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 0:
        return np.ones_like(a) / a.size
    return a / s

def _build_A(a):
    n = a.size
    m = 2 * n - 1
    k_idx = np.arange(m, dtype=np.int64)[:, None]
    i_idx = np.arange(n, dtype=np.int64)[None, :]
    j = k_idx - i_idx
    mask = (j >= 0) & (j < n)
    safe_j = np.clip(j, 0, n - 1)
    A = np.where(mask, a[safe_j], 0.0)
    return A

def _lp_best_response(a, M, timeout_s=2.0):
    n = a.size
    A = _build_A(a)
    c_vec = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=[(0.0, None)] * n,
                      method='highs', options={'time_limit': timeout_s})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    if not np.isfinite(g).all() or g.sum() <= 0:
        return None
    return g

def _soft_max_and_weights(c, t):
    """Compute soft max and weights for f(a) = t*log(sum exp(c/t))."""
    c_max = c.max()
    exp_c = np.exp((c - c_max) / t)
    sum_exp = exp_c.sum()
    weights = exp_c / sum_exp
    soft_max_val = c_max + t * np.log(sum_exp)
    return soft_max_val, weights

def _compute_gradient(a, weights):
    """Compute gradient of sum_k weights[k]*c_k w.r.t. a.
    c_k = sum_i a_i * a[k-i] (linear conv, a_ext outside [0,n-1] = 0)
    dc_k/da_j = a_ext[k-j] + a_ext[j-k]
    """
    n = a.size
    m = 2 * n - 1
    j = np.arange(n)[:, None]
    k = np.arange(m)[None, :]
    
    d = k - j  # (n, m)
    mask1 = (d >= 0) & (d < n)
    safe_d = np.clip(d, 0, n - 1)
    
    # term1[j,k] = mask1[j,k] * a[d]
    vals = np.where(mask1, a[safe_d], 0.0)
    # term2[j,k] = mask2[j,k] * a[-d] where mask2 is (-d >= 0) & (-d < n)
    mask2 = (~mask1) & (d > -n) & (d < 0)
    # Wait, d = k - j. mask2: (j - k >= 0) & (j - k < n) means -d >= 0 and -d < n
    # So d <= 0 and d > -n
    mask2 = (d <= 0) & (d > -n)
    safe_d2 = np.clip(-d, 0, n - 1)
    vals2 = np.where(mask2, a[safe_d2], 0.0)
    
    g = (vals + vals2) @ weights
    return g

def _lp_gradient_descent(a0, stop_time, max_iters=5000, t_init=0.1, t_min=0.01):
    """Gradient descent on soft-max objective."""
    n = a0.size
    a = _normalize(a0)
    best_a = a.copy()
    best_M = float(_convolve_square(a).max())
    
    t = t_init
    lr = 0.1
    t0 = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() > stop_time - 1.0:
            break
        
        c = _convolve_square(a)
        M = float(c.max())
        
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
        
        # Compute soft-max weights
        _, weights = _soft_max_and_weights(c, t)
        
        # Compute gradient
        g = _compute_gradient(a, weights)
        
        # Normalize gradient
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        
        g = g / g_norm
        
        # Projected gradient step with backtracking
        a_trial = a - lr * g
        a_trial = np.maximum(a_trial, 0.0)
        s = a_trial.sum()
        if s <= 0:
            lr *= 0.5
            continue
        a_trial = a_trial / s
        
        c_trial = _convolve_square(a_trial).max()
        
        if c_trial < M * (1 - 1e-14):
            a = a_trial
            # Increase learning rate slightly
            lr = min(lr * 1.05, 0.5)
        else:
            lr *= 0.7
            if lr < 1e-10:
                # Reduce temperature and try again
                t = max(t_min, t * 0.8)
                lr = 0.05
        
        # Anneal temperature
        if it % 100 == 0 and it > 0:
            t = max(t_min, t * 0.95)
    
    return best_a

def _lp_optimize(a0, stop_time, max_iters=200):
    """Run LP best-response iteration."""
    n = a0.size
    a = _normalize(a0)
    best_a = a.copy()
    best_M = float(_convolve_square(a).max())
    
    stuck = 0
    for it in range(max_iters):
        if time.monotonic() > stop_time - 0.5:
            break
        
        c = _convolve_square(a)
        M = float(c.max())
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1
        
        if stuck > 100:
            break
        
        g = _lp_best_response(a, M * 1.0001, timeout_s=1.5)
        if g is None:
            break
        
        gs = float(g.sum())
        if gs <= 0:
            break
        g = g / gs
        
        improved = False
        for t_val in (0.3, 0.2, 0.1, 0.05, 0.02):
            cand = (1.0 - t_val) * a + t_val * g
            cand = np.maximum(cand, 0.0)
            cs = cand.sum()
            if cs <= 0:
                continue
            cand = cand / cs
            cM = float(_convolve_square(cand).max())
            if cM < M * (1.0 - 1e-12):
                a = cand
                improved = True
                break
        
        if not improved:
            if stuck > 50:
                break
    
    return best_a

def _generate_initializations(parent, seed, n):
    rng = np.random.default_rng(seed)
    inits = []
    
    # Parent
    p = np.clip(parent.astype(np.float64), 0.0, 1000.0)
    if p.size == n:
        ps = p.sum()
        if ps > 0:
            inits.append(p / ps)
    
    # Ramp up
    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits.append(ramp / ramp.sum())
    
    # Ramp down
    inits.append(ramp[::-1] / ramp.sum())
    
    # Exponential
    for alpha in (0.5, 1.0, 2.0):
        exps = np.exp(-alpha * np.arange(n) / n)
        inits.append(exps / exps.sum())
        inits.append(exps[::-1] / exps.sum())
    
    # U-shape
    x = np.arange(n, dtype=np.float64)
    u = 1.0 / (0.1 + np.abs(x - n/2) / (n/2) * 5)
    inits.append(u / u.sum())
    
    # Gaussian
    center = n * 0.5
    width = n * 0.25
    gauss = np.exp(-0.5 * ((x - center) / width) ** 2)
    inits.append(gauss / gauss.sum())
    
    # Two-hump
    c1, c2 = n * 0.3, n * 0.7
    w = n * 0.15
    two = np.exp(-0.5 * ((x - c1) / w) ** 2) + np.exp(-0.5 * ((x - c2) / w) ** 2)
    inits.append(two / two.sum())
    
    # Random smooth
    r = rng.normal(0, 1, n)
    k = np.array([0.1, 0.8, 1.0, 0.8, 0.1])
    k /= k.sum()
    smooth = np.convolve(r, k, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())
    
    # Parent with noise
    if p.size == n:
        ps = p.sum()
        if ps > 0:
            pf = p / ps
            noisy = pf * np.exp(rng.normal(0, 0.2, n))
            noisy = np.maximum(noisy, 1e-6)
            inits.append(noisy / noisy.sum())
    
    return inits

def _conv_square(a, fsize=None):
    n = a.size
    if fsize is None:
        fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    return a / s if s > 0 else np.ones_like(a) / a.size

def _score(a):
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * a.size * float(_conv_square(a).max()) / (s * s)

def _lp_response(a, M, tl=2.0):
    n = a.size
    k = np.arange(2*n-1)[:,None] - np.arange(n)[None,:]
    A = np.where((k >= 0) & (k < n), a[np.clip(k, 0, n-1)], 0.0)
    try:
        r = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M),
                     bounds=[(0,None)]*n, method='highs',
                     options={'time_limit': tl})
    except: return None
    if r.status != 0 or r.x is None: return None
    g = np.asarray(r.x, float)
    return g if np.isfinite(g).all() and g.sum() > 0 else None

def _grad(a, w):
    """Gradient via FFT. w[k] for k=0..2n-2."""
    n = a.size
    m = 2*n - 1
    wr = w[::-1]
    N = next_fast_len(n + m - 1)
    conv = irfft(rfft(a, N) * rfft(wr, N), N)[:n+m-1]
    return 2.0 * conv[n-1 : 2*n-1][::-1]

def _soft_weights(c, t):
    cm = c.max()
    e = np.exp((c - cm) / t)
    return e / e.sum()

def _opt_lp(a, t_end, iters=300):

        c = _conv_square(a); M = c.max()
        if M < bestM * (1-1e-12):

        ok = False
        for t in (0.3, 0.15, 0.08, 0.04, 0.02):

            if _conv_square(r).max() < M*(1-1e-12):
                a, ok = r, True; break
        if not ok and stk > 40: break
    return best

def _opt_grad(a, t_end, iters=10000, t0=0.05, tmin=0.003):

    best, bestM = a.copy(), float(_conv_square(a).max())

        w = _soft_weights(c, t)

        if _ % 200 == 199:
            t = max(tmin, t*0.97)
    return best

    p = np.clip(p, 0.0, 1000.0)
    if p.size == 0 or not np.isfinite(p).all():
        p = np.ones(191, dtype=np.float64) / 191.0
    return p


def _conv_sq(a):
    n = a.size
    N = next_fast_len(2 * n - 1)
    F = rfft(a, N)
    c = irfft(F * F, N)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _norm(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    return a / s if s > 0 else np.ones_like(a) / a.size


def _score(a):
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * a.size * float(_conv_sq(a).max()) / (s * s)


def _lp_resp(a, M, tl=2.0):
    n = a.size
    k = np.arange(2 * n - 1)[:, None] - np.arange(n)[None, :]
    A = np.where((k >= 0) & (k < n), a[np.clip(k, 0, n - 1)], 0.0)
    try:
        r = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2 * n - 1, M),
                    bounds=[(0, None)] * n, method='highs',
                    options={'time_limit': tl})
    except Exception:
        return None
    if r.status != 0 or r.x is None:
        return None
    g = np.asarray(r.x, dtype=np.float64)
    return g if np.isfinite(g).all() and g.sum() > 1e-15 else None


def _grad_fft(a, w):
    n = a.size
    m = 2 * n - 1
    wr = w[::-1]
    N = next_fast_len(n + m - 1)
    conv = irfft(rfft(a, N) * rfft(wr, N), N)[:n + m - 1]
    return 2.0 * conv[n - 1:2 * n - 1][::-1]


def _soft_w(c, t):
    cm = c.max()
    e = np.exp((c - cm) / t)
    return e / e.sum()


def _opt_lp(a, t_end, iters=500):
    a = _norm(a)
    best = a.copy()
    bestM = _conv_sq(a).max()
    stk = 0
    for _ in range(iters):
        if time.monotonic() > t_end - 0.5:
            break
        c = _conv_sq(a)
        M = c.max()
        if M < bestM * (1 - 1e-13):
            best = a.copy()
            bestM = M
            stk = 0
        else:
            stk += 1
        if stk > 100:
            break
        g = _lp_resp(a, M * 1.00005, 1.0)
        if g is None:
            break
        g /= g.sum()
        ok = False
        for t in (0.25, 0.12, 0.06, 0.03, 0.015):
            r = _norm((1 - t) * a + t * g)
            if _conv_sq(r).max() < M * (1 - 1e-13):
                a = r
                ok = True
                break
        if not ok and stk > 50:
            break
    return best


def _opt_grad(a0, t_end, iters=20000, t_init=0.05, t_min=0.002):
    a = _norm(a0)
    best = a.copy()
    bestM = _conv_sq(a).max()
    t = t_init
    lr = 0.5
    no_impr = 0
    for it in range(iters):
        if time.monotonic() > t_end - 0.3:
            break
        c = _conv_sq(a)
        M = c.max()
        if M < bestM * (1 - 1e-13):
            best = a.copy()
            bestM = M
            no_impr = 0
        else:
            no_impr += 1
        
        if no_impr > 500:
            t = max(t_min, t * 0.8)
            lr = 0.3
            no_impr = 0
            if t <= t_min * 1.01:
                break
        
        w = _soft_w(c, t)
        g = _grad_fft(a, w)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        
        accepted = False
        for _ in range(25):
            r = _norm(a - lr * g)
            if _conv_sq(r).max() < M * (1 - 1e-14):
                a = r
                accepted = True
                break
            lr *= 0.65
        if not accepted:
            t = max(t_min, t * 0.9)
            lr = 0.2
        
        if it % 500 == 499:
            t = max(t_min, t * 0.95)
    
    return best
