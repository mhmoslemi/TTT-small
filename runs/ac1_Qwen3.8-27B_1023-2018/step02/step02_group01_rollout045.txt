import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, next_fast_len


def _conv_sq(a):
    """Compute full auto-convolution a*a via FFT."""
    n = a.size
    if n == 0:
        return np.array([0.0])
    fs = next_fast_len(2 * n - 1)
    F = rfft(a, fs)
    c = irfft(F * F, fs)[:2 * n - 1]
    return c


def _score(a):
    """Compute exact evaluator score for nonnegative a."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    n = a.size
    b = _conv_sq(a)
    mb = float(b.max())
    return 2.0 * n * mb / (s * s)


def _normalize(a):
    """Normalize so sum = 1."""
    a = np.clip(a, 0.0, 1000.0).copy()
    s = a.sum()
    if s <= 0:

    else:

def _build_conv_matrix(a):
    """Build (2n-1) x n matrix A where A[k,i] = a[k-i] if valid."""
    n = a.size

    j = k - i

    rng = np.random.default_rng(777)

    for it in range(max_iters):
        if time.monotonic() > stop_time - 2.0:
            break

        b = _conv_sq(a)
        M = float(b.max())
        if M < best_M * (1 - 1e-13):
            best_M = M
            best_a = a.copy()
            best_score = _score(best_a)
            stuck = 0
        else:
            stuck += 1

        if stuck > 300:
            # Perturb to escape local optimum
            a = a * np.exp(rng.normal(0, 0.08, n))
            a = np.maximum(a, 1e-10)
            a = _normalize(a)
            stuck = 0
            if stuck > 600:
                break
            continue

        # Solve LP with exact M (allow tiny slack for numerical safety)
        g = _lp_best_response(a, M * (1 + 1e-9), timeout=1.0)
        if g is None:
            # Try with slightly looser constraint
            g = _lp_best_response(a, M * (1 + 1e-5), timeout=0.5)
            if g is None:
                break
        g = g / g.sum()

        # Line search: try multiple step sizes
        improved = False
        for t in (1.0, 0.9, 0.75, 0.6, 0.45, 0.3, 0.2, 0.1, 0.05):
            cand = (1.0 - t) * a + t * g
            cand = np.maximum(cand, 0.0)
            cs = cand.sum()
            if cs < 1e-12:
                continue
            cand /= cs
            cM = float(_conv_sq(cand).max())
            if cM < M - 1e-14:
                a = cand
                improved = True
                break

        if not improved:
            if stuck > 50:
                # Smaller step
                for t in (0.02, 0.01, 0.005):
                    cand = (1.0 - t) * a + t * g
                    cand = np.maximum(cand, 0.0)
                    cs = cand.sum()
                    if cs > 0:
                        cand /= cs
                        cM = float(_conv_sq(cand).max())
                        if cM < M:
                            a = cand
                            improved = True
                            break
            if not improved and stuck > 100:
                # Try gradient step
                active = np.where(b >= M * (1 - 0.01))[0]
                if len(active) == 1:
                    k_star = active[0]
                    grad = np.zeros(n)
                    j = k_star - np.arange(n)
                    mask = (j >= 0) & (j < n)
                    grad[mask] = 2.0 * a[j[mask]]
                    lr = 0.1 / (np.linalg.norm(grad) + 1e-10)
                    a = a - lr * grad
                    a = _normalize(a)
                else:
                    if stuck > 200:
                        break

    return best_a


def _gradient_polish(a0, stop_time, max_iters=5000):
    """Projected subgradient method to polish."""
    n = a0.size
    a = _normalize(a0.astype(np.float64).copy())
    best_a = a.copy()
    best_score = _score(a)
    rng = np.random.default_rng(4242)
    stuck = 0
    lr = 0.05

    for it in range(max_iters):
        if time.monotonic() > stop_time - 2.0:
            break

        b = _conv_sq(a)
        M = float(b.max())
        sc = 2.0 * n * M
        if sc < best_score * (1 - 1e-12):
            best_score = sc
            best_a = a.copy()
            stuck = 0
            lr = min(lr * 1.2, 0.3)
        else:
            stuck += 1
            lr = max(lr * 0.95, 0.005)

        if stuck > 500:
            break

        # Subgradient of max at active constraints
        active = np.where(b >= M * (0.999))[0]
        n_active = len(active)
        grad = np.zeros(n)
        for k in active:
            j = k - np.arange(n)
            mask = (j >= 0) & (j < n)
            grad[mask] += 2.0 * a[j[mask]]
        grad /= n_active

        # Gradient descent with barrier for non-negativity
        a = a - lr * grad
        a = np.maximum(a, 0.0)
        a = _normalize(a)

    return best_a


def _hann_window(n):
    """Hann window of length n."""
    x = np.arange(n, dtype=np.float64)
    w = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / (n - 1)))
    return w / w.sum()


def _raised_cosine(n, beta):
    """Raised cosine power window."""
    x = np.arange(n, dtype=np.float64)
    t = x / (n - 1) - 0.5
    w = np.cos(np.pi * t / 1.0) ** beta
    w = np.maximum(w, 0.0)
    if w.sum() < 1e-15:
        w = np.ones(n)
    return w / w.sum()


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = float(budget_s)
    deadline = t0 + budget - 10.0  # 10s safety margin

    # Get parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
    except Exception:
        parent = np.ones(192) / 192.0
    parent = np.clip(parent, 0.0, 1000.0)
    if parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(192) / 192.0

    n_parent = parent.size
    rng = np.random.default_rng(seed)

    # Initial best
    best_seq = _normalize(parent.copy())
    best_score = _score(best_seq)

    # Try different initializations
    inits = []
    # 1. Parent normalized
    inits.append(('parent', _normalize(parent.copy())))
    # 2. Parent + multiplicative noise
    noisy = parent * np.exp(rng.normal(0, 0.15, n_parent))
    inits.append(('parent_noise', _normalize(np.maximum(noisy, 1e-8))))
    # 3. Hann window
    inits.append(('hann', _hann_window(n_parent)))
    # 4. Raised cosine beta=4
    inits.append(('rc4', _raised_cosine(n_parent, 4.0)))
    # 5. Raised cosine beta=8
    inits.append(('rc8', _raised_cosine(n_parent, 8.0)))
    # 6. Linear ramp (asymmetric)
    ramp = np.arange(1, n_parent + 1, dtype=np.float64)
    inits.append(('ramp_up', ramp / ramp.sum()))
    ramp2 = np.arange(n_parent, 0, -1, dtype=np.float64)
    inits.append(('ramp_down', ramp2 / ramp2.sum()))
    # 7. Exponential decay
    exps = np.exp(-2.0 * np.arange(n_parent) / n_parent)
    inits.append(('exp', exps / exps.sum()))
    # 8. Two-sided: high in middle
    xm = np.arange(n_parent, dtype=np.float64)
    mid = (n_parent - 1) / 2.0
    gauss = np.exp(-0.5 * ((xm - mid) / (n_parent * 0.25)) ** 2)
    inits.append(('gauss', gauss / gauss.sum()))

    # Phase 1: LP iteration from each init (time-limited)
    n_inits = len(inits)
    per_init_time = min(30.0, max(8.0, (budget * 0.4) / n_inits))
    phase1_deadline = t0 + budget * 0.45

    for name, init in inits:
        if time.monotonic() > phase1_deadline - 5:
            break
        init_end = min(time.monotonic() + per_init_time, phase1_deadline)
        try:
            result = _lp_iteration(init, stop_time=init_end, max_iters=2000)
            sc = _score(result)
            if np.isfinite(sc) and sc < best_score:
                best_seq = result.copy()
                best_score = sc
        except Exception:
            pass

    # Phase 2: Extended LP polish on best
    phase2_deadline = t0 + budget * 0.65
    if time.monotonic() < phase2_deadline - 10:
        try:
            polished = _lp_iteration(best_seq, stop_time=phase2_deadline, max_iters=5000)
            sc = _score(polished)
            if np.isfinite(sc) and sc < best_score:
                best_seq = polished.copy()
                best_score = sc
        except Exception:
            pass

    # Phase 3: Gradient polish
    phase3_deadline = t0 + budget * 0.75
    if time.monotonic() < phase3_deadline - 10:
        try:
            upgraded = _gradient_polish(best_seq, stop_time=phase3_deadline, max_iters=10000)
            sc = _score(upgraded)
            if np.isfinite(sc) and sc < best_score:
                best_seq = upgraded.copy()
                best_score = sc
        except Exception:

    # Phase 4: Final LP polish
    phase4_deadline = t0 + budget * 0.88
    if time.monotonic() < phase4_deadline - 15:
        try:
            polished2 = _lp_iteration(best_seq, stop_time=phase4_deadline, max_iters=5000)
            sc = _score(polished2)

                best_score = sc
        except Exception:
            pass

    # Phase 5: Try nearby lengths
    if time.monotonic() < deadline - 40:

            if time.monotonic() > deadline - 20:
                break
            if n_try < 8:
                continue
            # Resample from parent
            idxs = np.linspace(0, n_parent - 1, n_try).astype(int)
            new_init = np.clip(parent[idxs], 0.0, 1000.0).astype(np.float64)
            new_init = _normalize(new_init)
            try:

                sc = _score(result)
                if np.isfinite(sc) and sc < best_score:
                    best_seq = result.copy()
                    best_score = sc
            except Exception:
                pass

        # Also try Hann at different lengths
        for n_try in [190, 192, 193, 195, 200]:
            if time.monotonic() > deadline - 15:
                break
            try:
                hann_init = _hann_window(n_try)
                result = _lp_iteration(hann_init, stop_time=time.monotonic() + 10, max_iters=400)
                sc = _score(result)

        fallback = [1.0] * 64
        try:
            fs = evaluate_sequence(fallback)
        except Exception:
            fs = float('inf')
        if np.isfinite(fs):
            return fallback
        return [1.0] * 32

    return [float(x) for x in best_seq]
