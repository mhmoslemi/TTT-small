                                    pert2 = normalize(a - lr * gv * 0.5, nt)
                                    ps2 = score(pert2)
                                    if ps2 < cur * (1 - 1e-12):
                                        a = pert2
                                        cur = ps2
                                        stale = 0
                                        update(a)
                                        break
                if stale > 70:
                    break
        return a

    def make_inits(n, ref, rng_obj):
        """Generate diverse initializations of length n"""
        inits = []
        x = np.linspace(0.0, 1.0, n)
        # Constant
        inits.append(np.ones(n))
        # Exponential ramps
        for c in (0.5, 1.0, 2.0, 3.0, -1.0, -2.0):
            inits.append(np.exp(c * (x - 0.5)))
        # Sine harmonics
        for k in range(1, min(6, n // 2)):
            for amp in (0.3, 0.6, 0.9):
                inits.append(1.0 + amp * np.sin(2 * np.pi * k * (x - 0.5)))
                inits.append(1.0 + amp * np.cos(2 * np.pi * k * (x - 0.5)))
        # From reference
        if ref is not None and ref.size >= 2:
            ref_i = np.interp(x, np.linspace(0, 1, ref.size), ref)
            inits.append(ref_i)
            inits.append(ref_i * np.exp(0.1 * rng_obj.standard_normal(n)))
            inits.append(ref_i * np.exp(0.3 * rng_obj.standard_normal(n)))
        # Random from Dirichlet
        for al in (0.1, 0.3, 1.0):
            try:
                inits.append(rng_obj.dirichlet(np.full(n, al)))
            except:
                pass
        # Clean and score
        cleaned = []
        for ini in inits:
            ini = np.clip(np.nan_to_num(np.asarray(ini, dtype=np.float64)), 0, None)
            if ini.size == n and ini.sum() > 0.01:
                cleaned.append(ini / ini.sum())
        cleaned.sort(key=score)
        return cleaned

    # === Main optimization ===

    n_main = parent.size
    # Phase 1: Primary LP at parent length (most time)
    t1_end = t0 + min(550.0, deadline - t0 - 350.0)
    if t1_end > t0 + 30:
        lp_optimize(parent, t1_end, max_iters=30000)

    # Phase 2: LP from perturbed best + fresh inits
    phase2_end = t0 + min(850.0, deadline - 200.0)
    for trial in range(12):
        rem = phase2_end - time.monotonic()
        if rem < 25:
            break
        alloc = min(80.0, rem * 0.3)
        if alloc < 15:
            break
        if trial < 6:
            # Perturb best
            p = best_seq * np.exp(0.06 * rng.standard_normal(best_seq.size))
            p = np.clip(p, 0, 1000)
            if p.sum() < 0.01:
                p = best_seq.copy()
            start_a = p
        else:
            inits = make_inits(n_main, best_seq, rng)
            start_a = inits[trial % len(inits)]
        lp_optimize(start_a, time.monotonic() + alloc, max_iters=8000)

    # Phase 3: Try different lengths
    phase3_end = deadline - 80.0
    for n_try in (128, 160, 256, 224):
        if time.monotonic() > phase3_end:
            break
        rem = phase3_end - time.monotonic()
        if rem < 30:
            break
        nx_old = np.linspace(0, 1, best_seq.size)
        nx_new = np.linspace(0, 1, n_try)
        init = np.interp(nx_new, nx_old, best_seq)
        alloc = min(90.0, rem * 0.6)
        if alloc < 20:
            break
        lp_optimize(init, time.monotonic() + alloc, max_iters=10000)

    # Phase 4: Final polish on best
    rem = deadline - time.monotonic() - 5.0
    if rem > 15:
        lp_optimize(best_seq, time.monotonic() + min(50.0, rem), max_iters=8000)

    # Final validation
    fin = np.asarray(best_seq, dtype=np.float64).ravel().copy()
    fin = np.nan_to_num(fin, nan=0.0, posinf=0.0, neginf=0.0)
    fin = np.clip(fin, 0.0, 1000.0)
    if fin.size < 1 or float(fin.sum()) < 0.01:
        fin = np.ones(4, dtype=np.float64)
    update(fin)

    result = evaluate_sequence([float(x) for x in best_seq])
    if not math.isfinite(result):
        return [1.0, 1.0, 1.0, 1.0]
    return [float(x) for x in best_seq]
