        a = np.ones(320, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if s <= 1e-12:
        a = np.ones(a.size, dtype=np.float64) / a.size
    else:
        a = a / s
    return a


def _conv_full(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    N = 2 * n - 1
    if n <= 256:
        c = np.convolve(a, a, mode="full")
        c = np.maximum(c, 0.0)
        return c
    fsize = next_fast_len(N)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:N]
    c = np.maximum(c, 0.0)
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    if n == 0:
        return math.inf
    s = float(a.sum())
    if s <= 1e-12:
        return math.inf
    b = _conv_full(a)
    m = float(b.max()) if b.size else math.inf
    if not math.isfinite(m) or m < 0.0:
        return math.inf
    return float(2.0 * n * m / (s * s))


def _max_conv(a):
    b = _conv_full(a)
    m = float(b.max()) if b.size else math.inf
    return b, m


def _build_A(p, n):
    N = 2 * n - 1
    A = np.zeros((N, n), dtype=np.float64)
    for j in range(n):
        A[j : j + n, j] = p
    return A


def _line_search(p, d, base_sc, stop_time=None):
    d = np.asarray(d, dtype=np.float64).ravel()
    d = np.where(np.isfinite(d), d, 0.0)
    d = np.clip(d, 0.0, None)
    s = float(d.sum())
    if s <= 1e-12:
        return None, base_sc
    d = d / s
    if float(np.linalg.norm(d - p)) < 1e-11:
        return None, base_sc

    best_x = None
    best_s = base_sc
    for alpha in (1.0, 0.8, 0.6, 0.4, 0.25, 0.12, 0.05, 0.02):
        if stop_time is not None and time.monotonic() > stop_time:
            break
        x = (1.0 - alpha) * p + alpha * d
        x = np.clip(x, 0.0, None)
        tx = float(x.sum())
        if tx <= 1e-12:
            continue
        x = x / tx
        sc = _score(x)
        if sc < best_s - 1e-12:
            best_s = sc
            best_x = x.copy()
    if best_x is None:
        return None, base_sc
    return best_x, best_s


def _line_search_delta(p, d, base_sc, stop_time=None):
    d = np.asarray(d, dtype=np.float64).ravel()
    d = np.where(np.isfinite(d), d, 0.0)
    d = d - float(d.mean())
    if float(np.max(np.abs(d))) < 1e-12:
        return None, base_sc

    max_alpha = 2.0
    neg = d < -1e-12
    if neg.any():
        limits = -p[neg] / d[neg]
        max_alpha = min(max_alpha, float(np.min(limits)))
        if not math.isfinite(max_alpha) or max_alpha <= 1e-9:
            return None, base_sc

    best_x = None
    best_s = base_sc
    for frac in (1.0, 0.75, 0.5, 0.3, 0.15, 0.05):
        if stop_time is not None and time.monotonic() > stop_time:
            break
        alpha = frac * max_alpha
        if alpha <= 0.0:
            continue
        x = p + alpha * d
        if np.any(x < -1e-10):
            continue
        x = np.clip(x, 0.0, None)
        tx = float(x.sum())
        if tx <= 1e-12:
            continue
        x = x / tx
        sc = _score(x)
        if sc < best_s - 1e-12:
            best_s = sc
            best_x = x.copy()
    if best_x is None:
        return None, base_sc
    return best_x, best_s


def _run_linprog(c, A_ub, b_ub, A_eq=None, b_eq=None, bounds=None):
    try:
        return linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
            options={"presolve": True},
        )
    except Exception:
        try:
            return linprog(
                c,
                A_ub=A_ub,
                b_ub=b_ub,
                A_eq=A_eq,
                b_eq=b_eq,
                bounds=bounds,
            )
        except Exception:
            return None


def _lp_maxsum(A, M, n):
    if M <= 1e-300:
        return None, 1.0
    N = int(A.shape[0])
    As = A / M
    B = max(1.0, float(N) * M) + 1e-8
    cvec = -np.ones(n, dtype=np.float64)
    b = np.ones(N, dtype=np.float64)
    bounds = np.zeros((n, 2), dtype=np.float64)
    bounds[:, 1] = B
    res = _run_linprog(cvec, As, b, None, None, bounds)
    if res is None or getattr(res, "status", 1) != 0:
        return None, 1.0
    g = np.asarray(res.x, dtype=np.float64).ravel()
    g = np.clip(g, 0.0, None)
    R = float(g.sum())
    if R <= 1e-12:
        return None, 1.0
    return g, R


def _lp_mint(A, M, n):
    if M <= 1e-300:
        return None, M
    N = int(A.shape[0])
    As = A / M
    B = max(1.0, float(N) * M) + 1e-8
    cvec = np.zeros(n + 1, dtype=np.float64)
    cvec[n] = 1.0

    A2 = np.empty((N, n + 1), dtype=np.float64)
    A2[:, :n] = As
    A2[:, n] = -1.0
    b = np.zeros(N, dtype=np.float64)

    Aeq = np.zeros((1, n + 1), dtype=np.float64)
    Aeq[0, :n] = 1.0
    beq = np.array([1.0], dtype=np.float64)

    bounds = np.zeros((n + 1, 2), dtype=np.float64)
    bounds[:n, 1] = B
    bounds[n, 1] = 1.0 + 1e-8

    res = _run_linprog(cvec, A2, b, Aeq, beq, bounds)
    if res is None or getattr(res, "status", 1) != 0:
        return None, M
    x = np.asarray(res.x, dtype=np.float64).ravel()
    g = np.clip(x[:n], 0.0, None)
    gs = float(g.sum())
    if gs <= 1e-12:
        return None, M
    g = g / gs
    tau = float(x[n])
    return g, tau * M


def _lp_eta(A, c_lags, M, p, n):
    if M <= 1e-300:
        return None, 0.0
    N = int(A.shape[0])
    rhs = (M - c_lags) / M
    rhs = np.maximum(rhs, 0.0)

    A2 = np.empty((N, n + 1), dtype=np.float64)
    A2[:, :n] = (2.0 / M) * A
    A2[:, n] = 1.0

    cvec = np.zeros(n + 1, dtype=np.float64)
    cvec[n] = -1.0

    Aeq = np.ones((1, n + 1), dtype=np.float64)
    Aeq[0, n] = 0.0
    beq = np.array([0.0], dtype=np.float64)

    bounds = np.zeros((n + 1, 2), dtype=np.float64)
    bounds[:n, 0] = -np.maximum(p, 1e-12)
    bounds[:n, 1] = 1.0
    bounds[n, 0] = 0.0
    bounds[n, 1] = 1.0

    res = _run_linprog(cvec, A2, rhs, Aeq, beq, bounds)
    if res is None or getattr(res, "status", 1) != 0:
        return None, 0.0
    x = np.asarray(res.x, dtype=np.float64).ravel()
    d = x[:n]
    eta = float(x[n])
    if eta <= 1e-8:
        return None, 0.0
    d = d - float(d.mean())
    if float(np.max(np.abs(d))) < 1e-12:
        return None, 0.0
    return d, eta


def _perturb(p, rng, sigma):
    base = np.maximum(p, 1e-10)
    y = rng.normal(0.0, sigma, p.size)
    q = base * np.exp(np.clip(y, -8.0, 8.0))
    q = np.clip(q, 0.0, None)
    s = float(q.sum())
    if s <= 1e-12:
        return p.copy()
    return q / s


def _local_lp(p0, stop_time, seed, target=_TARGET):
    cur = _clean(p0)
    n = int(cur.size)
    cur_sc = _score(cur)
    if not math.isfinite(cur_sc):
        cur = _fallback_sequence(320)
        n = cur.size
        cur_sc = _score(cur)

    best = cur.copy()
    best_sc = cur_sc
    rng = np.random.default_rng(seed)
    stall = 0
    pert = 0
    it = 0
    last_best = time.monotonic()

    while time.monotonic() < max(time.monotonic(), stop_time) - 2.0:
        if best_sc < target - 1e-9:
            break
        c, M = _max_conv(cur)
        if not math.isfinite(M) or M <= 1e-300:
            break
        A = _build_A(cur, n)
        improved = False

        g, R = _lp_maxsum(A, M, n)
        if g is not None and R > 1.0 + 1e-5:
            d = g / max(1e-12, float(g.sum()))
            cand, sc = _line_search(cur, d, cur_sc, stop_time)
            if cand is not None:
                cur, cur_sc = cand, sc
                improved = True

        if not improved and (it % 3 == 1):
            g, tstar = _lp_mint(A, M, n)
            if g is not None and tstar < M * (1.0 - 1e-7):
                d = g / max(1e-12, float(g.sum()))
                cand, sc = _line_search(cur, d, cur_sc, stop_time)
                if cand is not None:
                    cur, cur_sc = cand, sc
                    improved = True

        if not improved and (it == 0 or it % 12 == 0):
            d, _eta = _lp_eta(A, c, M, cur, n)
            if d is not None:
                cand, sc = _line_search_delta(cur, d, cur_sc, stop_time)
                if cand is not None:
                    cur, cur_sc = cand, sc
                    improved = True

        if improved:
            stall = 0
            if cur_sc < best_sc - 1e-12:
                best = cur.copy()
                best_sc = cur_sc
                last_best = time.monotonic()
        else:
            stall += 1
            if stall >= 8:
                sigma = (0.03, 0.06, 0.12, 0.20, 0.35)[min(4, stall // 8)]
                cur = _perturb(cur, rng, sigma)
                cur_sc = _score(cur)
                stall = 0
                pert += 1
                if cur_sc < best_sc - 1e-12:
                    best = cur.copy()
                    best_sc = cur_sc
                    last_best = time.monotonic()

        it += 1
        if pert > 40:
            break
        if it > 50 and (time.monotonic() - last_best) > 160.0:
            break

    if not math.isfinite(_score(best)):
        best = _fallback_sequence(320)
        best_sc = _score(best)
    return best, best_sc


def _corr_fft(p, w):
    n = int(p.size)
    N = 2 * n - 1
    if n == 1:
        return np.array([2.0 * w[0] * p[0]], dtype=np.float64)
    L = next_fast_len(N)

    A = np.zeros(L, dtype=np.float64)
    A[:n] = p

    Y = np.zeros(L, dtype=np.float64)
    Y[0] = w[0]
    if N > 1:
        idx = L - np.arange(1, N)
        Y[idx] = w[1:]

    Z = irfft(rfft(A, L) * rfft(Y, L), L)
    out_idx = np.mod(L - np.arange(n), L)
    return 2.0 * Z[out_idx]


def _smooth_opt(p0, stop_time, seed, target=_TARGET):
    cur = _clean(p0)
    n = int(cur.size)
    if n > 1536:
        sc = _score(cur)
        return cur, sc

    cur_sc = _score(cur)
    best = cur.copy()
    best_sc = cur_sc
    rng = np.random.default_rng(seed)

    tau_factor = 0.05
    stall = 0
    perturb_count = 0
    it = 0
    last_best = time.monotonic()
    last_progress = time.monotonic()

    while time.monotonic() < max(time.monotonic(), stop_time) - 2.0:
        if best_sc < target - 1e-9:
            break

        c, m = _max_conv(cur)
        if not math.isfinite(m) or m <= 1e-300:
            break

        tau = max(1e-14, float(tau_factor) * m)
        z = (c - m) / tau
        if not np.all(np.isfinite(z)):
            break
        w = np.exp(np.clip(z, -700.0, 0.0))
        sw = float(w.sum())
        if sw <= 1e-12:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = w / sw

        G = _corr_fft(cur, w)
        if not np.all(np.isfinite(G)):
            break
        G = np.where(G > 0.0, G, 0.0)
        maxG = float(G.max())
        if maxG <= 1e-15:
            tau_factor = min(1.0, tau_factor * 2.0)
            stall += 1
            if stall > 60:
                break
            continue

        base_eta = 1.0 / maxG
        best_x = None
        best_s = cur_sc
        for f in (1.2, 0.8, 0.4, 0.2, 0.1, 0.05, 0.02):
            if time.monotonic() > stop_time:
                break
            eta = f * base_eta
            q = cur * np.exp(-np.clip(eta * G, -700.0, 700.0))
            q = np.clip(q, 0.0, None)
            s = float(q.sum())
            if s <= 1e-12:
                continue
            q = q / s
            sc = _score(q)
            if sc < best_s - 1e-12:
                best_s = sc
                best_x = q.copy()

        if best_x is not None:
            cur, cur_sc = best_x, best_s
            last_progress = time.monotonic()
            if cur_sc < best_sc - 1e-12:
                best = cur.copy()
                best_sc = cur_sc
                last_best = time.monotonic()
                stall = 0
            else:
                stall = 0
            tau_factor = max(0.001, tau_factor * 0.8)
        else:
            tau_factor = min(1.0, tau_factor * 1.6)
            stall += 1
            if stall >= 12:
                sigma = 0.04 + 0.01 * min(8, stall // 4)
                cur = _perturb(cur, rng, sigma)
                cur_sc = _score(cur)
                stall = 0
                perturb_count += 1
                last_progress = time.monotonic()
                if cur_sc < best_sc - 1e-12:
                    best = cur.copy()
                    best_sc = cur_sc
                    last_best = time.monotonic()

        it += 1
        if perturb_count > 36:
            break
        if (time.monotonic() - last_best) > 180.0 and (time.monotonic() - last_progress) > 60.0:
            break

    if not math.isfinite(_score(best)):
        best = _fallback_sequence(320)
        best_sc = _score(best)
    return best, best_sc


def _interpolate(a, L):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if a.size == 0 or s <= 1e-12:
        return np.ones(max(1, L), dtype=np.float64) / max(1, L)
    a = a / s
    n = int(a.size)
    L = int(max(1, L))
    if n == 1:
        return np.full(L, 1.0 / L, dtype=np.float64)
    xo = np.linspace(0.0, 1.0, n)
    xn = np.linspace(0.0, 1.0, L)
    b = np.interp(xn, xo, a)
    b = np.clip(b, 0.0, None)
    sb = float(b.sum())
    if sb <= 1e-12:
        return np.ones(L, dtype=np.float64) / L
    return b / sb


def _alternative_lengths(base, best, best_sc):
    n = int(base.size)
    candidates = [96, 128, 160, 192, 256, 320, 384, 448, 512, 640, n // 2, n * 2]
    seen = set()
    for L in candidates:
        L = int(L)
        if L < 16 or L > 1024 or L in seen or L == n:
            continue
        seen.add(L)
        init = _interpolate(base, L)
        sc = _score(init)
        if math.isfinite(sc) and sc < best_sc - 1e-10:
            best = init.copy()
            best_sc = sc
    return best, best_sc


def _make_valid_list(x):
    a = np.asarray(x, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    if a.size == 0 or float(a.sum()) <= 0.0:
        a = _fallback_sequence(320)
    return a


def _exact_score(seq):
    try:
        return float(evaluate_sequence([float(x) for x in seq]))
    except Exception:
        return math.inf


def _final_answer(seq, fallback):
    arr = _make_valid_list(seq)
    scor = _exact_score(arr)
    fb_arr = _make_valid_list(fallback)
    fb_sc = _exact_score(fb_arr)
    if not math.isfinite(scor):
        if math.isfinite(fb_sc):
            return [float(x) for x in fb_arr]
        return [float(x) for x in _fallback_sequence(320)]
    if math.isfinite(fb_sc) and fb_sc < scor:
        return [float(x) for x in fb_arr]
    return [float(x) for x in arr]


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard_stop = t0 + max(10.0, budget - 12.0)

    parent = _get_parent()
    best = _clean(parent)
    best_sc = _score(best)
    if not math.isfinite(best_sc):
        best = _fallback_sequence(320)
        best_sc = _score(best)
    fallback = best.copy()

    if budget > 30.0 and best_sc >= _TARGET:
        stop1 = min(hard_stop - 50.0, t0 + 0.45 * budget)
        if stop1 > time.monotonic() + 1.0:
            best, best_sc = _local_lp(best, stop1, int(seed), _TARGET)
            if not math.isfinite(best_sc):
                best = fallback
                best_sc = _score(fallback)

    if best_sc >= _TARGET and budget > 60.0:
        best, best_sc = _alternative_lengths(best, best, best_sc)
        if not math.isfinite(best_sc):
            best = fallback
            best_sc = _score(fallback)

    if best_sc >= _TARGET and budget > 100.0:
        stop2 = min(hard_stop - 35.0, t0 + 0.75 * budget)
        if stop2 > time.monotonic() + 1.0 and len(best) <= 1536:
            best, best_sc = _smooth_opt(best, stop2, int(seed) + 7919, _TARGET)
            if not math.isfinite(best_sc):
                best = fallback
                best_sc = _score(fallback)

    if best_sc >= _TARGET and budget > 160.0:
        stop3 = hard_stop - 10.0
        if stop3 > time.monotonic() + 1.0 and len(best) <= 768:
            best, best_sc = _local_lp(best, stop3, int(seed) + 104729, _TARGET)
            if not math.isfinite(best_sc):
                best = fallback
                best_sc = _score(fallback)

    return _final_answer(best, fallback)
