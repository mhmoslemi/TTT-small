        M = float(b.max())
        # Find all lags within 1% of max
        thresh = M * 0.99
        active = np.nonzero(b >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        sg = np.zeros(n)
        for k in active:
            k = int(k)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    sg[i] += a[j]
        sg /= len(active)

        v = a - lr * sg
        proj = _simplex_proj(v)
        s_new = _score(proj)
        if math.isfinite(s_new) and s_new < loc * (1.0 - 1e-13):
            a, loc, stall = proj, s_new, 0
            _upd(a, best)
        else:
            stall += 1
            if stall % 20 == 10:
                ap = _norm(a * np.exp(0.03 * rng.standard_normal(n)))
                sp = _score(ap)
                if sp < loc:
                    a, loc, stall = ap, sp, 0
                    _upd(a, best)
            if stall > 150:
                break
    return a


def _inits(n, rng, parent):
    out = []
    x = np.linspace(0, 1, n)

    def add(a):
        a = np.asarray(a, dtype=np.float64).ravel()
        a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
        a = np.clip(a, 0.0, None)
        if a.size == n and a.sum() > 1e-10:
            out.append(_norm(a))

    add(np.ones(n))
    for c in [-2, -1, -0.5, 0.5, 1, 2, 3, -3]:
        add(np.exp(c * (x - 0.5)))
    for k in range(1, min(12, n // 3 + 1)):
        for amp in [0.3, 0.5, 0.7, 0.9]:
            add(np.maximum(1.0 + amp * np.sin(2 * np.pi * k * x), 0.01))
            add(np.maximum(1.0 + amp * np.cos(2 * np.pi * k * x), 0.01))
    for sig in [0.1, 0.15, 0.25, 0.35]:
        add(np.exp(-0.5 * ((x - 0.5) / sig) ** 2))
    # Triangular
    add(np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0))
    add(np.abs(x - 0.5))

    if parent is not None and len(parent) >= 4:
        tp = np.linspace(0, 1, len(parent))
        base = np.interp(x, tp, parent)
        add(base)
        for sc in [0.05, 0.1, 0.2, 0.3, 0.5]:
            add(base * np.exp(sc * rng.standard_normal(n)))
        # Reshape parent with different interpolations
        for m in [16, 32, 64]:
            if m < len(parent):
                pm = parent[np.linspace(0, len(parent)-1, m).astype(int)]
                im = np.interp(x, np.linspace(0, 1, m), pm)
                add(im)

    for _ in range(8):
        add(rng.dirichlet(np.ones(n) * rng.uniform(0.3, 2.0)))

    out.sort(key=_score)
    return out


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    try:
        rem = max(30.0, float(budget_s))
    except Exception:
        rem = 1300.0
    deadline = t0 + rem - 15.0

    best = [float('inf'), np.array([1.0, 1.0])]
    _upd(best[1], best)

    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            parent = np.asarray(h, dtype=np.float64).ravel()
            parent = np.nan_to_num(parent, nan=0.0, posinf=0.0, neginf=0.0)
            parent = np.clip(parent, 0.0, 1000.0)
            if parent.size >= 2:
                _upd(parent, best)
    except Exception:
        pass

    n_par = parent.size if parent is not None else 192

    # Phase 1: LP at parent length - main workhorse
    if time.monotonic() < deadline - 130.0:
        inits = _inits(n_par, rng, parent)
        for idx, s in enumerate(inits[:12]):
            if time.monotonic() >= deadline - 130.0:
                break
            remaining = deadline - 130.0 - time.monotonic()
            if remaining < 10.0:
                break
            alloc = min(100.0, remaining * 0.3)
            if alloc < 8.0:
                break
            _lp_refine(s, time.monotonic() + alloc, best, rng, max_iter=3000)

    # Phase 2: LP at n=128
    if time.monotonic() < deadline - 100.0:
        inits128 = _inits(128, rng, best[1])
        for s in inits128[:8]:
            if time.monotonic() >= deadline - 100.0:
                break
            remaining = deadline - 100.0 - time.monotonic()
            if remaining < 8.0:
                break
            alloc = min(70.0, remaining * 0.3)
            if alloc < 6.0:
                break
            _lp_refine(s, time.monotonic() + alloc, best, rng, max_iter=2000)

    # Phase 3: LP at n=256
    if time.monotonic() < deadline - 80.0:
        inits256 = _inits(256, rng, best[1])
        for s in inits256[:4]:
            if time.monotonic() >= deadline - 80.0:
                break
            remaining = deadline - 80.0 - time.monotonic()
            if remaining < 8.0:
                break
            alloc = min(60.0, remaining * 0.3)
            if alloc < 6.0:
                break
            _lp_refine(s, time.monotonic() + alloc, best, rng, max_iter=1500)

    # Phase 4: More LP at parent length with diverse starts
    if time.monotonic() < deadline - 55.0:
        inits4 = _inits(n_par, rng, best[1])
        for s in inits4[8:20]:
            if time.monotonic() >= deadline - 55.0:
                break
            remaining = deadline - 55.0 - time.monotonic()
            if remaining < 8.0:
                break
            alloc = min(45.0, remaining * 0.3)
            if alloc < 6.0:
                break
            _lp_refine(s, time.monotonic() + alloc, best, rng, max_iter=1500)

    # Phase 5: Subgradient polish on best candidate
    if time.monotonic() < deadline - 30.0:
        alloc = min(35.0, deadline - 30.0 - time.monotonic())
        if alloc > 5.0:
            _subgrad_polish(best[1], time.monotonic() + alloc, best, rng, max_iter=5000)

    # Phase 6: Multi-active subgradient polish
    if time.monotonic() < deadline - 18.0:
        alloc = min(15.0, deadline - 18.0 - time.monotonic())
        if alloc > 3.0:
            _multi_active_sg(best[1], time.monotonic() + alloc, best, rng, max_iter=2000)

    # Phase 7: Final LP polish
    if time.monotonic() < deadline - 8.0:
        alloc = min(10.0, deadline - 8.0 - time.monotonic())
        if alloc > 3.0:
            _lp_refine(best[1], time.monotonic() + alloc, best, rng, max_iter=500)

    # Final validation
    fin = np.asarray(best[1], dtype=np.float64).ravel().copy()
    fin = np.nan_to_num(fin, nan=0.0, posinf=0.0, neginf=0.0)
    fin = np.clip(fin, 0.0, 1000.0)
    if fin.size < 1 or fin.sum() < 0.01:
        fin = np.array([1.0])
    _upd(fin, best)

    out = np.asarray(best[1], dtype=np.float64).ravel()
    v = evaluate_sequence([float(x) for x in out])
    if not math.isfinite(v):
        out = np.array([1.0])

    return [float(x) for x in out]
