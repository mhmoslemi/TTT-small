def _conv_full(a):
    n = len(a)
    if n <= 128:
        c = np.convolve(a, a, mode="full")
    else:
        fsize = next_fast_len(2 * n - 1)
        F = rfft(a, fsize)
        c = irfft(F * F, fsize)[: 2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _conv_mat(f, n):
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j : j + n, j] = f
    return A


def _lp_solve(f, n, M):
    A = _conv_mat(f, n)
    c_vec = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)

            c_vec, A_ub=A, b_ub=b_ub,
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True, "time_limit": 60,
                     "primal_feasibility_tolerance": 1e-9,
                     "dual_feasibility_tolerance": 1e-9},
        )

        return None

        return None
    g = np.asarray(res.x, dtype=np.float64)

    if g_sum < 1e-12:
        return None
    return g / g_sum

                break

        if not improved:
            stall += 1
            t = max(0.015, t * 0.55)
            if stall >= 60:
                # Structured restart
                f = np.clip(f + rng.normal(0, 0.06, n), 0, None)
                # Try flipping symmetry
                if rng.random() < 0.3:
                    f = 0.5 * f + 0.5 * f[::-1]
                f /= max(f.sum(), 1e-10)
                t = 0.25
                stall = 0

    return best_f, best_sc


def _subgrad_polish(f0, n, stop_time, seed, max_steps=20000):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel(), 0, None).copy()
    f /= max(f.sum(), 1e-10)
    best_f = f.copy()
    best_sc = _score(f)
    f_ave = f.copy()
    alpha = 0.3
    decay = 0.9997

    for step in range(max_steps):
        if time.monotonic() > stop_time:
            break

        b = _conv_full(f)
        k_star = int(np.argmax(b))

        # Subgradient
        grad = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        idx = np.arange(lo, hi + 1)
        grad[idx] = 2.0 * f[k_star - idx]
        if k_star % 2 == 0:
            mid = k_star // 2
            if 0 <= mid < n:
                grad[mid] = f[mid]

        f = f - alpha * grad

        # Project onto simplex (fast approximation: clip + renormalize)
        f = np.clip(f, 0, None)
        fns = float(f.sum())
        if fns < 1e-10:
            f = best_f.copy()
        else:
            f = f / fns

        # Nesterov average
        f_ave = (step / (step + 1.0)) * f_ave + (1.0 / (step + 1.0)) * f

        # Evaluate
        sc = _score(f)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_f = f.copy()

        if step % 1000 == 999:
            sc_ave = _score(f_ave)
            if sc_ave < best_sc - 1e-14:
                best_sc = sc_ave
                best_f = f_ave.copy()

        alpha *= decay
        if alpha < 1e-6:
            alpha = 0.05
            decay = 0.9999

    return best_f, best_sc


def _local_polish(f0, stop_time, seed):
    n = len(f0)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel(), 0, None).copy()
    f /= max(f.sum(), 1e-10)
    best_sc = _score(f)
    rng = np.random.default_rng(seed)
    no_imp = 0

    while time.monotonic() < stop_time:
        improved = False

        # Single-coordinate scaling
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time:
                break
            orig = f[i]
            for mlt in (0.4, 0.6, 0.8, 1.2, 1.5, 2.0, 0.0, 0.2):
                f[i] = orig * mlt if orig > 1e-13 else (0.01 if mlt > 0 else 0.0)
                fns = f.sum()
                if fns < 0.01:
                    f[i] = orig
                    continue
                sc = _score(f / fns)
                if sc < best_sc - 1e-13:
                    best_sc = sc
                    f = f / fns
                    f = f.copy()
                    improved = True
                    break
            f[i] = f[i]  # keep as-is (may have been updated)

        # Paired moves
        for _ in range(40):
            if time.monotonic() > stop_time:
                break
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            oi, oj = f[i], f[j]
            for mi, mj in ((0.5, 1.5), (1.5, 0.5), (0.7, 1.3), (1.3, 0.7), (0.3, 2.0)):
                f[i] = oi * mi if oi > 1e-13 else 0.01
                f[j] = oj * mj if oj > 1e-13 else 0.01
                fns = f.sum()
                if fns < 0.01:
                    f[i], f[j] = oi, oj
                    continue
                sc = _score(f / fns)
                if sc < best_sc - 1e-13:
                    best_sc = sc
                    f = (f / fns).copy()
                    improved = True
            f[i], f[j] = oi, oj

        if improved:
            no_imp = 0
        else:
            no_imp += 1
            if no_imp >= 10:
                break

    f /= max(f.sum(), 1e-10)
    return f, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(20.0, float(budget_s))
    deadline = t0 + budget - 12.0

    parent = _get_parent()
    n_p = len(parent)
    n = n_p

    # Validate parent
    f_parent = np.clip(parent, 0, None).copy()
    f_parent /= max(f_parent.sum(), 1e-10)

    # Fallback
    fallback = f_parent.copy()
    try:
        fb_sc = float(evaluate_sequence([float(x) for x in fallback]))
    except Exception:
        fb_sc = math.inf
    if not math.isfinite(fb_sc):
        fallback = np.ones(64) / 64
        fb_sc = float(evaluate_sequence([float(x) for x in fallback]))

    best_f = fallback.copy()
    best_score = fb_sc

    def _try_update(res, sc_est):
        nonlocal best_f, best_score
        if not np.all(np.isfinite(res)):
            return
        res = np.clip(res, 0, 1000.0)
        if res.sum() < 0.01:
            return
        if sc_est < best_score - 1e-15:
            try:
                exact = float(evaluate_sequence([float(x) for x in res]))
            except Exception:
                exact = math.inf
            if math.isfinite(exact) and exact < best_score - 1e-14:
                best_f = res.copy()
                best_score = exact

    # Phase 1: LP optimization at parent length (55% budget)
    phase1_end = t0 + budget * 0.55
    rng_main = np.random.default_rng(seed)
    i_arr = np.arange(n, dtype=np.float64)

    initializations = [
        f_parent.copy(),
        np.clip(((i_arr / max(n - 1, 1)) ** 1.3) + 0.05, 0, None),
        np.clip(np.exp(-2.0 * i_arr / n) + 0.02, 0, None),
        np.clip(1.0 - i_arr / n + 0.05, 0, None),
        np.clip(0.5 + 0.5 * np.cos(2 * math.pi * i_arr / n) + 0.03, 0, None),
        np.clip(parent + rng_main.normal(0, 0.15 * max(1.0, parent.mean()), n), 0, None),
    ]
    for init in initializations:
        init = np.clip(init, 0, None)
        init /= max(init.sum(), 1e-10)

    n_inits = len(initializations)
    for idx, init in enumerate(initializations):
        if time.monotonic() > phase1_end - 15:
            break
        if best_score < 1.503:
            break
        time_per = max(15.0, (phase1_end - t0 - 15.0) / max(1, n_inits - idx))
        stop = min(phase1_end - 10, time.monotonic() + time_per)
        sd = seed + 10000 * (idx + 1)
        try:
            res, sc = _lp_iterate(init, n, stop, sd)
        except Exception:
            continue
        _try_update(res, sc)

    # Phase 2: Other lengths (15% budget)
    phase2_end = t0 + budget * 0.72
    other_lengths = [256, 384, 448]
    if n_p not in other_lengths:
        other_lengths.append(n_p)
    other_lengths = sorted(set(other_lengths))
    n_alt = len(other_lengths)

    for L in other_lengths:
        if time.monotonic() > phase2_end - 10:
            break
        if best_score < 1.503:
            break
        time_per = max(10.0, (phase2_end - t0 - 10.0) / max(1, n_alt))
        stop = min(phase2_end - 8, time.monotonic() + time_per)
        # Interpolate
        init_L = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, n_p), parent)
        init_L = np.clip(init_L, 0, None)
        init_L /= max(init_L.sum(), 1e-10)
        sd = seed + 50000 + L
        try:
            res, sc = _lp_iterate(init_L, L, stop, sd, max_iters=300)
        except Exception:
            continue
        _try_update(res, sc)

    # Phase 3: Polish the best (23% budget)
    polish_start = max(time.monotonic(), t0 + budget * 0.72)
    polish_end = deadline - 5.0

    if polish_end - polish_start > 10:
        best_n = len(best_f)
        # Subgradient polishing
        sg_stop = min(polish_end - 8, polish_start + (polish_end - polish_start) * 0.5)
        if sg_stop - time.monotonic() > 5:
            try:
                best_f, best_score = _subgrad_polish(best_f, best_n, sg_stop, seed + 7777)
            except Exception:
                pass
            _try_update(best_f, best_score)

        # Local coordinate polish
        lp_stop = min(polish_end - 3, time.monotonic() + (polish_end - polish_start) * 0.35)
        if lp_stop - time.monotonic() > 3:
            try:
                best_f, best_score = _local_polish(best_f, lp_stop, seed + 8888)
            except Exception:
                pass
            _try_update(best_f, best_score)

        # One more short LP run for final squeeze
        if time.monotonic() < polish_end - 10:
            try:
                res, sc = _lp_iterate(best_f, len(best_f), polish_end - 5, seed + 99999, max_iters=200)
            except Exception:
                res, sc = best_f, best_score
            _try_update(res, sc)

    # Final validation
    final_list = [float(x) for x in best_f]
    try:
        final_sc = float(evaluate_sequence(final_list))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc) or any(x < 0 or x > 1000 or not math.isfinite(x) for x in final_list):
        final_list = [float(x) for x in fallback]
        try:
            final_sc = float(evaluate_sequence(final_list))
        except Exception:
            final_list = [1.0]
            final_sc = float(evaluate_sequence(final_list))

    # Ensure we return before deadline
    if time.monotonic() > deadline:
        pass  # Already out of time, just return

    return final_list
