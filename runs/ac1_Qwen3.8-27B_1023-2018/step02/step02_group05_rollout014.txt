        a_new = a - lr * grad
        a_new = _project_simplex(a_new)
        a_new = np.clip(a_new, 1e-15, None)
        a_new = a_new / a_new.sum()

        sc = _score(a_new)
        if math.isfinite(sc) and sc < local:
            a = a_new
            local = sc
            stale = 0
            if it % 200 == 0:
                _update_best(a, best)
            else:
                _update_best(a, best)
        else:
            stale += 1
            lr *= 0.85
            if lr < 1e-10:
                break
        if stale > 200:
            break

    _update_best(a, best)


def _anti_pal_optimize(a_init, n, time_limit, rng, best):
    """Optimize in the anti-palindromic subspace: a[j]+a[n-1-j]=c."""
    if n < 6:
        return
    if n % 2 == 1:
        n -= 1
    half = n // 2
    t0 = time.monotonic()
    deadline = t0 + time_limit - 3.0

    def f_from_v(v):
        v = np.clip(np.asarray(v, dtype=np.float64), -0.999, 0.999)
        f = np.ones(n)
        f[:half] = 1.0 + v
        f[half:] = 1.0 - v[::-1]
        return np.clip(f, 0.0, None)

    def v_from_f(f):
        v = 0.5 * (f[:half] - f[n - 1: n - half - 1: -1])
        return np.clip(v, -0.999, 0.999)

    a = np.asarray(a_init, dtype=np.float64).ravel()
    if a.size != n:
        a = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, a.size), a)
    a = np.clip(a, 0.0, None)
    a = a / a.sum()
    v = v_from_f(a)
    f = f_from_v(v)
    f = f / f.sum()

    local = _score(f)
    _update_best(f, best)

    lr = 0.05
    stale = 0
    for it in range(30000):
        if time.monotonic() >= deadline:
            break
        if local < TARGET:
            break

        b = np.convolve(f, f)
        M = float(b.max())
        eps = 0.01 * M + 1e-15
        active = np.where(b >= M - eps)[0]
        if len(active) == 0:
            break

        grad_f = np.zeros(n)
        for k in active:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            for m in range(lo, hi + 1):
                grad_f[m] += 2.0 * f[k - m]
        grad_f /= len(active)

        grad_v = grad_f[:half] - grad_f[half:][::-1]
        gn = float(np.linalg.norm(grad_v))
        if gn < 1e-14:
            break

        improved = False
        for lr_try in (lr, lr * 0.5, lr * 2.0):
            v_new = v - lr_try * grad_v / gn * (float(np.std(v)) + 0.02)
            v_new = np.clip(v_new, -0.999, 0.999)
            f_new = f_from_v(v_new)
            f_new = f_new / f_new.sum()
            sc = _score(f_new)
            if math.isfinite(sc) and sc < local:
                v = v_new
                f = f_new
                local = sc
                stale = 0
                lr = min(lr * 1.05, 0.5)
                _update_best(f, best)
                improved = True
                break

        if not improved:
            stale += 1
            lr *= 0.7
            if lr < 1e-7:
                vp = v + 0.03 * rng.standard_normal(half)
                vp = np.clip(vp, -0.999, 0.999)
                fp = f_from_v(vp)
                fp = fp / fp.sum()
                sp = _score(fp)
                if math.isfinite(sp) and sp < local:
                    v = vp
                    f = fp
                    local = sp
                    _update_best(f, best)
                lr = 0.05
                if stale > 50:
                    break

    _update_best(f, best)


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    try:
        remaining = max(0.0, float(budget_s))
    except Exception:
        remaining = 1300.0
    deadline = t_start + remaining - 22.0

    best = [float('inf'), np.array([1.0, 1.0], dtype=np.float64)]
    _update_best(np.ones(16), best)

    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            parent = np.asarray(h, dtype=np.float64).ravel()
            parent = np.nan_to_num(parent, nan=0.0, posinf=0.0, neginf=0.0)
            parent = np.clip(parent, 0.0, 1000.0)
            if parent.size > 1 and float(parent.sum()) > 0.02:
                _update_best(parent, best)
    except Exception:
        parent = None

    def _rem():
        return max(0.0, deadline - time.monotonic())

    # Phase 1: LP on parent length (primary)
    if parent is not None and parent.size >= 4:
        np_ = parent.size
        alloc = min(480.0, _rem() * 0.38)
        if alloc > 30:
            _lp_iterate(parent, np_, alloc, rng, best)

    # Phase 2: Anti-palindromic optimization on best
    if _rem() > 80:
        ba = np.asarray(best[1], dtype=np.float64).ravel()
        nb = ba.size
        alloc = min(150.0, _rem() * 0.18)
        if alloc > 20:
            _anti_pal_optimize(ba, nb, alloc, rng, best)

    # Phase 3: LP polish on current best
    if _rem() > 60:
        ba = np.asarray(best[1], dtype=np.float64).ravel()
        nb = ba.size
        alloc = min(120.0, _rem() * 0.18)
        if alloc > 15:
            _lp_iterate(ba, nb, alloc, rng, best)

    # Phase 4: Try other lengths
    for ntr in (96, 128, 192, 256, 160):
        if _rem() < 70:
            break
        a_init = np.ones(ntr)
        if parent is not None and parent.size > 1:
            a_init = np.interp(np.linspace(0, 1, ntr),
                               np.linspace(0, 1, parent.size), parent)
        x = np.linspace(0, 1, ntr)
        a_init = a_init * (1.0 + 0.4 * x)
        alloc = min(60.0, _rem() * 0.12)
        if alloc > 15:
            _lp_iterate(a_init, ntr, alloc, rng, best)

    # Phase 5: Gradient polish
    if _rem() > 30:
        ba = np.asarray(best[1], dtype=np.float64).ravel()
        nb = ba.size
        alloc = min(50.0, _rem() * 0.3)
        if alloc > 8:
            _grad_polish(ba, nb, alloc, best)

    # Phase 6: Final short LP
    if _rem() > 20:
        ba = np.asarray(best[1], dtype=np.float64).ravel()
        nb = ba.size
        alloc = min(30.0, _rem() * 0.5)
        if alloc > 8:
            _lp_iterate(ba, nb, alloc, rng, best)

    # Final validation
    final = np.asarray(best[1], dtype=np.float64).ravel()
    final = np.nan_to_num(final, nan=0.0, posinf=0.0, neginf=0.0)
    final = np.clip(final, 0.0, 1000.0)
    if final.size == 0:
        final = np.ones(16, dtype=np.float64)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(16, dtype=np.float64)

    fin_sc = evaluate_sequence([float(x) for x in final])
    if not math.isfinite(fin_sc) or fin_sc > best[0] + 0.01:
        # Try parent fallback
        if parent is not None and parent.size > 1:
            p_sc = evaluate_sequence([float(x) for x in parent])
            if math.isfinite(p_sc) and p_sc < fin_sc:
                final = parent.copy()
                fin_sc = p_sc
        if not math.isfinite(fin_sc):
            final = np.ones(32, dtype=np.float64)
            fin_sc = evaluate_sequence([float(x) for x in final])

    return [float(x) for x in final]
