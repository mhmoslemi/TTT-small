    relax_cycle = 0
    for it in range(max_it):
        if time.monotonic() - t0 > limit:
            break
        tl = max(1.0, min(8.0, limit - (time.monotonic() - t0) - 0.5))
        if tl < 1.0:
            break
        # Cycle through LP variants
        if relax_cycle % 5 < 3:
            g = _lp(a, tl)
        elif relax_cycle % 5 == 3:
            g = _lp_relax(a, 0.005, tl)
        else:
            g = _lp_relax(a, 0.001, tl)
        relax_cycle += 1
        if g is None:
            stag += 1
            if stag > 30:
                break
            continue
        t, ns = _line_search(a, g, cur)
        if t > 1e-4 and ns < cur * (1.0 - 1e-12):
            a = _norm((1.0 - t) * a + t * g)
            cur = ns
            stag = 0
            _upd(a, best)
        else:
            stag += 1
            if stag % 15 == 0:
                # Intelligent perturbation
                eps = 0.03 + 0.02 * (stag // 15 % 4)
                pa = a * np.exp(eps * rng.standard_normal(n))
                pa = _norm(pa)
                ps = _score(pa)
                if math.isfinite(ps) and ps < cur:
                    a, cur, stag = pa, ps, 0
                    _upd(a, best)
            if stag > 150:
                break
    return a, cur


def _coord_desc(a0, limit, best):
    a = _norm(np.asarray(a0, dtype=np.float64).ravel().copy())
    n = a.size
    cur = _score(a)
    _upd(a, best)
    t0 = time.monotonic()
    factors = np.array([0.4, 0.6, 0.75, 0.85, 0.9, 1.0, 1.1, 1.15, 1.3, 1.5, 2.0, 0.0])
    sweep = 0
    while time.monotonic() - t0 < limit:
        if sweep > 0 and sweep % 5 == 0:
            # Occasionally use finer factors
            factors = np.array([0.9, 0.95, 1.0, 1.05, 1.1, 1.0])
        else:
            factors = np.array([0.4, 0.6, 0.75, 0.85, 0.9, 1.0, 1.1, 1.15, 1.3, 1.5, 2.0, 0.0])
        any_imp = False
        for i in range(n):
            if time.monotonic() - t0 > limit - 1.0:
                break
            old = a[i]
            bs = cur
            bf = None
            for f in factors:
                a[i] = old * f
                s = _score(a)
                if math.isfinite(s) and s < bs:
                    bs = s
                    bf = f
            if bf is not None and bs < cur * (1.0 - 1e-10):
                a[i] = old * bf
                cur = bs
                any_imp = True
                _upd(a, best)
            else:
                a[i] = old
        sweep += 1
        if not any_imp:
            break
    return a, cur


def _gradient_ref(a0, limit, best, rng):
    """Subgradient-based refinement targeting the peak lag."""
    a = _norm(np.asarray(a0, dtype=np.float64).ravel().copy())
    n = a.size
    if n < 10:
        return a, _score(a)
    cur = _score(a)
    _upd(a, best)
    t0 = time.monotonic()
    stale = 0
    for it in range(20000):
        if time.monotonic() - t0 > limit - 1.0:
            break
        b = np.convolve(a, a)
        S = float(a.sum())
        ks = int(np.argmax(b))
        # Subgradient: direction to reduce b[ks]
        sg = np.zeros(n, dtype=np.float64)
        for i in range(n):
            j = ks - i
            if 0 <= j < n:
                sg[i] += a[j]
                if j != i:
                    sg[i] += a[j]
        # Normalize and project (ensure nonnegativity)
        gn = float(np.linalg.norm(sg))
        if gn < 1e-15:
            break
        d = sg / gn
        # Try several step sizes
        bs, bv = cur, None
        for alpha in (0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5):
            cand = a - alpha * d
            cand = np.clip(cand, 0.0, None)
            cs = float(cand.sum())
            if cs < 1e-300:
                continue
            cand = cand / cs
            s = _score(cand)
            if math.isfinite(s) and s < bs:
                bs, bv = s, cand
        if bv is not None and bs < cur * (1.0 - 1e-11):
            a, cur, stale = bv, bs, 0
            _upd(a, best)
        else:
            stale += 1
            if stale % 20 == 0:
                pa = a * np.exp(0.02 * rng.standard_normal(n))
                ps = _score(_norm(pa))
                if math.isfinite(ps) and ps < cur:
                    a, cur, stale = _norm(pa), ps, 0
                    _upd(a, best)
            if stale > 200:
                break
    return a, cur


def _gen_starts(n, rng, parent=None):
    """Generate diverse initializations."""
    S = []
    x = np.linspace(0.0, 1.0, n)

    def add(a):
        a = np.asarray(a, dtype=np.float64).ravel()
        a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
        a = np.clip(a, 0.0, None)
        if a.size == n and float(a.sum()) > 1e-300:
            S.append(a / a.sum())

    add(np.ones(n))
    add(np.exp(1.0 * (x - 0.5)))
    add(np.exp(-1.0 * (x - 0.5)))
    add(np.exp(3.0 * (x - 0.5)))
    add(np.exp(-3.0 * (x - 0.5)))
    add(np.exp(5.0 * (x - 0.5)))
    for k in range(1, min(6, n // 2 + 1)):
        add(1.0 + 0.5 * np.sin(2.0 * np.pi * k * x))
        add(1.0 + 0.8 * np.sin(2.0 * np.pi * k * x))
        add(1.0 + 0.5 * np.cos(2.0 * np.pi * k * x))
    for alpha in (0.1, 0.3, 0.5, 1.0, 2.0):
        try:
            add(rng.dirichlet(np.full(n, alpha)))
        except Exception:
            pass
    # Structured: alternating patterns
    alt = np.ones(n)
    alt[1::2] = 0.3
    add(alt)
    alt2 = np.ones(n)
    alt2[1::2] = 0.0
    add(alt2)
    alt3 = np.ones(n)
    alt3[2::3] = 0.2
    add(alt3)
    # Smooth bumps
    for center in (0.25, 0.5, 0.75):
        add(np.exp(-50.0 * (x - center) ** 2))
    # Parent interpolation
    if parent is not None and len(parent) >= 2 and float(parent.sum()) > 0:
        tp = np.linspace(0.0, 1.0, len(parent))
        for noise in (0.0, 0.05, 0.1, 0.2, 0.35):
            base = np.interp(x, tp, parent)
            if noise > 0:
                base = base * np.exp(noise * rng.standard_normal(n))
            add(base)
        # Also try truncated/wrapped versions
        if n >= 64:
            h = n // 2
            half = np.interp(np.linspace(0, 1, h), tp[:len(tp)//2 + 1] if len(tp)//2+1 < len(tp) else tp, parent)
            full = np.concatenate([half, half[::-1]])
            if full.size == n:
                add(full)
    if not S:
        S.append(np.ones(n) / n)
    S.sort(key=_score)
    return S


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    try:
        rem = max(0.0, float(budget_s))
    except Exception:
        rem = 1300.0
    deadline = t0 + rem - 20.0  # 20s buffer

    best = [float('inf'), np.array([1.0, 1.0], dtype=np.float64)]
    _upd(best[1], best)

    # Get parent
    par = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            par = np.asarray(h, dtype=np.float64).ravel()
            par = np.nan_to_num(par, nan=0.0, posinf=0.0, neginf=0.0)
            par = np.clip(par, 0.0, 1000.0)
            if par.size >= 2:
                _upd(par, best)
    except Exception:
        par = None

    n_par = par.size if par is not None else 192
    print(f"Starting with parent length {n_par}, initial score {best[0]:.8f}")

    # === Phase 1: LP refinement at parent length (primary) ===
    if time.monotonic() < deadline - 300.0:
        starts = _gen_starts(n_par, rng, best[1])
        for idx, s in enumerate(starts[:10]):
            if time.monotonic() >= deadline - 300.0:
                break
            alloc = min(180.0, (deadline - 300.0 - time.monotonic()) * 0.30)
            if alloc < 15.0:
                break
            _lp_iter(s, alloc, best, rng, max_it=20000)
            print(f"  LP start {idx}: best={best[0]:.8f}")

    # === Phase 2: LP at n=128 ===
    if time.monotonic() < deadline - 200.0:
        starts128 = _gen_starts(128, rng, best[1])
        for s in starts128[:6]:
            if time.monotonic() >= deadline - 200.0:
                break
            alloc = min(100.0, (deadline - 200.0 - time.monotonic()) * 0.35)
            if alloc < 10.0:
                break
            _lp_iter(s, alloc, best, rng, max_it=15000)
            print(f"  LP n=128: best={best[0]:.8f}")

    # === Phase 3: More LP at parent length with different starts ===
    if time.monotonic() < deadline - 150.0:
        starts3 = _gen_starts(n_par, rng, best[1])
        for s in starts3[5:14]:
            if time.monotonic() >= deadline - 150.0:
                break
            alloc = min(80.0, deadline - 150.0 - time.monotonic())
            if alloc < 8.0:
                break
            _lp_iter(s, alloc, best, rng, max_it=15000)
            print(f"  LP phase3: best={best[0]:.8f}")

    # === Phase 4: Gradient refinement ===
    if time.monotonic() < deadline - 100.0:
        for ng in (n_par, 128):
            if time.monotonic() >= deadline - 100.0:
                break
            alloc = min(50.0, (deadline - 100.0 - time.monotonic()) * 0.5)
            if alloc < 5.0:
                break
            _gradient_ref(best[1], alloc, best, rng)
            print(f"  Grad n={ng}: best={best[0]:.8f}")

    # === Phase 5: Coordinate descent polish ===
    if time.monotonic() < deadline - 50.0:
        nb = best[1].size
        alloc = min(40.0, deadline - 50.0 - time.monotonic())
        if alloc > 3.0:
            _coord_desc(best[1], alloc, best)
            print(f"  CoordDesc: best={best[0]:.8f}")

    # === Phase 6: Final LP polish on best ===
    if time.monotonic() < deadline - 15.0:
        alloc = min(30.0, deadline - 15.0 - time.monotonic())
        if alloc > 3.0:
            _lp_iter(best[1], alloc, best, rng, max_it=5000)
            print(f"  Final LP: best={best[0]:.8f}")

    # === Phase 7: Final coordinate descent ===
    if time.monotonic() < deadline - 8.0:
        alloc = min(8.0, deadline - 8.0 - time.monotonic())
        if alloc > 1.0:
            _coord_desc(best[1], alloc, best)

    # Final validation
    fin = np.asarray(best[1], dtype=np.float64).ravel().copy()
    fin = np.nan_to_num(fin, nan=0.0, posinf=0.0, neginf=0.0)
    fin = np.clip(fin, 0.0, 1000.0)
    if fin.size < 1:
        fin = np.array([1.0], dtype=np.float64)
    if float(fin.sum()) < 0.01:
        fin = np.ones(fin.size, dtype=np.float64)
    _upd(fin, best)
    fin = np.asarray(best[1], dtype=np.float64).ravel()
    v = evaluate_sequence([float(x) for x in fin])
    if not math.isfinite(v):
        fin = np.array([1.0], dtype=np.float64)
    print(f"FINAL: score={v:.10f}, length={fin.size}")
    return [float(x) for x in fin]
