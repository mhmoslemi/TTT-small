        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(192, dtype=np.float64)
    try:

    except Exception:
        p = np.ones(192, dtype=np.float64)
    if p.size == 0 or not np.isfinite(p).all():
        p = np.ones(192, dtype=np.float64)
    return p

def _fft_conv(a):
    """Compute full convolution of a with itself (or autobuf)."""
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2*n-1]
    np.maximum(c, 0.0, out=c)
    c[~np.isfinite(c)] = 0.0
    return c

def _convolve_xy(a, g):
    """Compute conv(a, g) for vectors of same length."""
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    Fa = rfft(a, fsize)
    Fg = rfft(g, fsize)
    c = irfft(Fa * Fg, fsize)[:2*n-1]
    c[~np.isfinite(c)] = 0.0
    return c

def _build_A(a):
    """Build the (2n-1) x n matrix where A[k,j] = a[k-j] if 0<=k-j<n."""
    n = len(a)
    k_idx = np.arange(2*n-1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    A = np.zeros((2*n-1, n))
    valid = (diff >= 0) & (diff < n)
    A[valid] = a[diff[valid].astype(int)]
    return A

def _paceflex_lp(a, M, n):
    """Solve: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    A = _build_A(a)
    # linprog: min c@x s.t. A_ub @ x <= b_ub, x >= 0
    c = -np.ones(n)  # minimize -sum(g)
    A_ub = A
    b_ub = np.full(2*n-1, M)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if res.status == 0:
        g = res.x
        g = np.maximum(g, 0)
        s = g.sum()
        if s > 1e-15:
            return g / s
    return None

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(192, dtype=np.float64)
    try:
        p = np.clip(p, 0.0, 1000.0)
    except Exception:
        p = np.ones(192, dtype=np.float64)
    if p.size == 0 or not np.isfinite(p).all():
        p = np.ones(192, dtype=np.float64)
    return p


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def _fft_conv_self(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    c[~np.isfinite(c)] = 0.0
    return c


def _score_fast(a):
    n = len(a)
    b = _fft_conv_self(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return float(2.0 * n * b.max() / (s * s))


def _precompute_structure(n):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(k - j)
    return (np.array(rows, dtype=np.int32),
            np.array(cols, dtype=np.int32),
            np.array(vals, dtype=np.int32))


def _solve_lp_direction(a, M, n, struct):
    rows, cols, vals = struct
    A = np.zeros((2 * n - 1, n))
    A[rows, cols] = a[vals]
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs',
                      options={"presolve": True})
        if res.status == 0 and res.x is not None:
            g = np.maximum(res.x, 0)
            s = g.sum()
            if s > 1e-12:
                return g / s, res
    except Exception:
        pass
    return None, None


def _subgradient(a, n, k_star):
    b = np.zeros(n)
    for m in range(n):
        if m + k_star < n:
            b[m] += a[m + k_star]
        if m - k_star >= 0:
            b[m] += a[m - k_star]
    return b


def _line_search(a, direction, n, M0):
    best_t = 0.0
    best_M = M0
    ts = [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]
    best_a = a.copy()
    for t in ts:
        a_new = (1 - t) * a + t * direction
        a_new = _normalize(a_new)
        b = _fft_conv_self(a_new)
        M = float(b.max())
        if M < best_M:
            best_M = M
            best_t = t
            best_a = a_new.copy()
    return best_t, best_M, best_a


def _lp_phase(a, n, struct, deadline, rng):
    a = _normalize(a.copy())
    best_a = a.copy()
    best_M = float(_fft_conv_self(a).max())
    it = 0
    no_improve = 0
    max_no_improve = 60

    while time.monotonic() < deadline:
        it += 1
        b = _fft_conv_self(a)
        M = float(b.max())

        if M < best_M * (1 - 1e-11):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            scale = 0.05 + 0.05 * (no_improve // max_no_improve)
            a = a + rng.normal(0, scale, n)
            a = _normalize(a)
            no_improve = 0
            max_no_improve = int(max(30, max_no_improve * 1.15))
            if max_no_improve > 500:
                break
            continue

        k_star = int(np.argmax(b))
        g_lp, _ = _solve_lp_direction(a, M, n, struct)

        if g_lp is None:
            grad = _subgradient(a, n, k_star)
            gn = np.linalg.norm(grad)
            if gn < 1e-15:
                a = a + rng.normal(0, 0.02, n)
                a = _normalize(a)
                continue
            g_lp = _normalize(a - 0.3 * grad / gn)

        cos_sim = float(np.dot(a, g_lp) / (np.linalg.norm(a) * np.linalg.norm(g_lp) + 1e-30))
        if cos_sim > 0.9995:
            a = a + rng.normal(0, 0.03, n)
            a = _normalize(a)
            no_improve += 2
            continue

        t, new_M, a_new = _line_search(a, g_lp, n, M)

        if new_M < M * (1 - 1e-12):
            a = a_new
        else:
            grad = _subgradient(a, n, k_star)
            gn = np.linalg.norm(grad)
            if gn > 1e-15:
                a_try = _normalize(a - (0.1 / gn) * grad)
                M_try = float(_fft_conv_self(a_try).max())
                if M_try < M:
                    a = a_try
                else:
                    no_improve += 1

        if it % 500 == 0 and time.monotonic() > deadline:
            break

    return best_a


def _local_search(a, n, deadline, rng):
    a = _normalize(a.copy())
    best_a = a.copy()
    best_M = float(_fft_conv_self(a).max())
    sigma = 0.03
    no_improve = 0

    while time.monotonic() < deadline:
        r = rng.random()
        if r < 0.30:
            cand = a + rng.normal(0, sigma, n)
        elif r < 0.50:
            cand = a.copy()
            idx = int(rng.integers(0, n))
            width = int(rng.integers(1, max(2, n // 25)))
            lo = max(0, idx - width // 2)
            hi = min(n, lo + width)
            cand[lo:hi] *= np.exp(rng.normal(0, 0.3, hi - lo))
        elif r < 0.65:
            cand = a.copy()
            lo = int(rng.integers(0, max(1, n // 6)))
            width = int(rng.integers(max(1, n // 30), max(2, n // 5)))
            hi = min(n, lo + width)
            cand[lo:hi] *= rng.uniform(0.3, 2.0)
        elif r < 0.80:
            cand = a.copy()
            for _ in range(int(rng.integers(2, 8))):
                idx = int(rng.integers(0, n))
                cand[idx] *= rng.uniform(0.5, 1.8)
        elif r < 0.90:
            cand = 0.5 * (a + a[::-1])
            cand += rng.normal(0, sigma * 0.4, n)
        else:
            cand = 0.93 * a + 0.07 / n
            cand += rng.normal(0, sigma * 0.3, n)

        cand = _normalize(cand)
        M = float(_fft_conv_self(cand).max())

        if M < best_M * (1 - 1e-11):
            best_M = M
            best_a = cand.copy()
            a = cand.copy()
            no_improve = 0
            sigma = max(0.003, sigma * 0.96)
        else:
            no_improve += 1
            if no_improve > 100:
                sigma = min(0.35, sigma * 1.5)
                no_improve = 0

    return best_a, best_M


def _init_variant(n, rng, variant):
    i = np.arange(n, dtype=np.float64)
    v = int(variant) % 8
    if v == 0:
        f1 = rng.uniform(0.1, 0.7)
        f2 = rng.uniform(0.1, 0.8)
        a = 1.0 + 0.5 * np.sin(2 * np.pi * f1 * i / n + rng.uniform(0, 2 * np.pi)) \
              + 0.3 * np.cos(2 * np.pi * f2 * i / n + rng.uniform(0, 2 * np.pi))
    elif v == 1:
        A1 = rng.uniform(0.5, 8.0)
        A2 = rng.uniform(0.5, 8.0)
        f1 = rng.uniform(0.05, 0.6)
        f2 = rng.uniform(0.05, 0.7)
        p1 = rng.uniform(0, 2 * np.pi)
        p2 = rng.uniform(0, 2 * np.pi)
        a = np.exp(A1 * np.sin(2 * np.pi * f1 * i / n + p1) + A2 * np.sin(2 * np.pi * f2 * i / n + p2))
    elif v == 2:
        a = np.exp(rng.normal(0, 0.7, n))
    elif v == 3:
        a = 0.5 + 0.5 * np.abs(np.sin(np.pi * i / n * rng.uniform(1, 5)))
    elif v == 4:
        a = np.abs(rng.normal(0, 1, n)) + 0.05
    elif v == 5:
        a = np.ones(n)
        for _ in range(int(rng.integers(3, 8))):
            c = rng.uniform(0, n)
            w = rng.uniform(n / 10, n / 4)
            a += rng.uniform(0.2, 1.0) * np.exp(-(i - c) ** 2 / (2 * w ** 2))
    elif v == 6:
        a = 1.0 + 0.6 * np.sin(2 * np.pi * i / n * int(rng.integers(2, max(3, n // 3))))
        a += 0.25 * np.sin(2 * np.pi * i / n * int(rng.integers(3, max(4, n // 2))))
    else:
        a = rng.uniform(0.3, 1.5, n)
    a = np.maximum(a, 0.01)
    return a / a.sum()
