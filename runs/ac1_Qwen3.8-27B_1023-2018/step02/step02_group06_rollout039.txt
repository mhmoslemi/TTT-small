```python
def _build_conv_matrix(a):
    n = len(a)
    # Use a Toeplitz-like construction
    # A[k,j] = a[max(0, min(k-j, 2n-2-k-j))] ... no that's not right.
    # A[k,j] = a[k-j] if 0 <= k-j < n
    # Let d = k-j. For row k, d ranges from k-(n-1) to k.
    # Valid d: 0 <= d < n.
    # So for row k, j ranges from max(0, k-n+1) to min(n-1, k).
    # And A[k,j] = a[k-j] for valid j.
    
    k_idx = np.arange(2*n-1)[:, None]  # (2n-1, 1)
    j_idx = np.arange(n)[None, :]       # (1, n)
    d = k_idx - j_idx                    # (2n-1, n)
    valid = (d >= 0) & (d < n)
    A = np.zeros((2*n-1, n))
    A[valid] = a[d[valid]]
    return A
```

```python
    """Quick score computation (scale-invariant form)."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return 2.0 * n  # fallback
    b = _conv_self(a)
    return 2.0 * n * b.max() / (s * s)

    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)

def _local_search(a_init, deadline, rng, max_iters=5000):
    """Local search with adaptive noise."""
    n = len(a_init)
    a = _normalize(np.maximum(a_init, 0.0).copy())
    b = _conv_self(a)
    best_mb = float(b.max())
    best_a = a.copy()
    
    sigma = 0.02
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Generate candidate
        r = rng.random()
        if r < 0.3:
            cand = a + rng.normal(0, sigma, n)
        elif r < 0.5:
            # Block mutation
            cand = a.copy()
            width = int(rng.integers(3, max(4, n // 8)))
            lo = int(rng.integers(0, n - width))
            cand[lo:lo+width] *= np.exp(rng.normal(0, 0.3, width))
        elif r < 0.65:
            # Single point
            cand = a.copy()
            idx = int(rng.integers(0, n))
            cand[idx] *= rng.uniform(0.3, 3.0)
        elif r < 0.8:
            # Smooth perturbation
            cand = a.copy()
            kernel_size = int(rng.integers(5, max(6, n // 10)))
            kernel = rng.normal(0, 1, kernel_size)
            kernel /= (np.abs(kernel).sum() + 1e-15)
            for offset in range(-kernel_size//2, kernel_size//2+1):
                for j in range(n):
                    jj = j + offset
                    if 0 <= jj < n:
                        cand[j] += kernel_size * kernel[offset + kernel_size//2] * a[jj] * 0.1
        else:
            # Sinusoidal perturbation
            cand = a * (1.0 + 0.3 * sigma * np.sin(2*np.pi*rng.uniform(1, 5)*np.arange(n)/n + rng.uniform(0, 6.28)))
        
        np.maximum(cand, 0.0, out=cand)
        cand = _normalize(cand)
        mb = float(_conv_self(cand).max())
        
        if mb < best_mb - 1e-13:
            best_mb = mb
            best_a = cand.copy()
            a = cand.copy()
            no_improve = 0
            sigma = max(0.002, sigma * 0.95)
        else:
            no_improve += 1
            if no_improve > 200:
                sigma = min(0.3, sigma * 1.5)
                no_improve = 0
    
    return best_a, best_mb

def _init_variants(n, rng, variant=0):
    i = np.arange(n, dtype=np.float64)
    v = int(variant) % 10
    
    if v == 0:
        # Two sinusoids
        f1 = rng.uniform(0.5, 5.0)
        f2 = rng.uniform(5.0, 20.0)
        a = 1.0 + 0.3*np.sin(2*np.pi*f1*i/n) + 0.2*np.cos(2*np.pi*f2*i/n)
    elif v == 1:
        # Random exponential
        a = np.exp(rng.normal(0, 0.5, n))
    elif v == 2:
        # Raised cosine variant
        a = 0.5 + 0.5*np.cos(np.pi*(i - (n-1)/2)/((n-1)/2))
        a = 1.0 - 0.5*a  # inverted
    elif v == 3:
        # Multi-scale Gaussian mixture
        a = np.ones(n) * 0.5
        for _ in range(int(rng.integers(2, 5))):
            c = rng.uniform(0, n)
            w = rng.uniform(n//10, n//3)

    elif v == 4:
        # Random walk (smooth)
        a = np.cumsum(rng.normal(0, 1, n))
        a = a - a.min() + 0.1
    elif v == 5:
        # Periodic with random phases
        a = 1.0
        for f in [rng.uniform(1,3), rng.uniform(3,7), rng.uniform(7,15)]:
            a += 0.2 * np.sin(2*np.pi*f*i/n + rng.uniform(0, 6.28))

        # Uniform (baseline)
        a = np.ones(n) * rng.uniform(0.8, 1.2) + rng.normal(0, 0.1, n)

    else:
        a = rng.uniform(0.5, 1.5, n)

    return _normalize(a)
```

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
import cvxpy as cp

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(128, dtype=np.float64)
    try:
        p = np.clip(p, 0.0, 1000.0)
    except Exception:
        p = np.ones(128, dtype=np.float64)
    if p.size == 0 or not np.isfinite(p).all():
        p = np.ones(128, dtype=np.float64)
    return p

def _conv_self(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2*n-1]
    np.maximum(c, 0.0, out=c)
    c[~np.isfinite(c)] = 0.0
    return c

def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def _build_A(a):
    n = len(a)
    k = np.arange(2*n-1)[:, None]
    j = np.arange(n)[None, :]
    d = k - j
    valid = (d >= 0) & (d < n)
    A = np.zeros((2*n-1, n))
    A[valid] = a[d[valid]]
    return A

def _solve_lp(a, M):
    n = len(a)
    g = cp.Variable(n, nonneg=True)
    A = _build_A(a)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [A @ g <= M])
    for solver in [cp.GLOP, cp.CBC, cp.GLPK, cp.MOSEK]:
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                gv = g.value
                if gv is not None and np.all(np.isfinite(gv)):
                    gv = np.maximum(gv, 0)
                    s = gv.sum()
                    if s > 1e-15:
                        return gv / s
        except Exception:
            continue
    return None

            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 60:
            a = a + rng.normal(0, 0.04, n)
            np.maximum(a, 0, out=a)
            a = _normalize(a)
            no_improve = 0
            continue
        
        g = _solve_lp(a, M)
        if g is None:
            no_improve += 5
            continue
        
        # Check convergence
        na = np.linalg.norm(a)
        ng = np.linalg.norm(g)
        if na * ng > 1e-15:
            cos = float(np.dot(a, g) / (na * ng))
        else:
            cos = 1.0
        
        if cos > 0.9999:

            if no_improve > 100:

                a = _normalize(a)
                no_improve = 0

def _local_search(a_init, deadline, rng):
    n = len(a_init)
    a = _normalize(np.maximum(a_init, 0.0))
    best_a = a.copy()
    best_mb = float(_conv_self(a).max())
    sigma = 0.015
    no_improve = 0
    
    while time.monotonic() < deadline:
        r = rng.random()
        if r < 0.4:
            cand = a + rng.normal(0, sigma, n)
        elif r < 0.6:
            cand = a.copy()
            w = int(rng.integers(2, max(3, n//10)))
            lo = int(rng.integers(0, n-w))
            cand[lo:lo+w] *= np.exp(rng.normal(0, 0.2, w))
        elif r < 0.75:
            cand = a.copy()
            idx = int(rng.integers(0, n))
            cand[idx] *= rng.uniform(0.5, 2.0)
        elif r < 0.85:
            cand = 0.5*(a + a[::-1]) + rng.normal(0, sigma*0.5, n)
        else:
            cand = (1-sigma)*a + sigma*np.ones(n)/n
        
        np.maximum(cand, 0, out=cand)
        cand = _normalize(cand)
        mb = float(_conv_self(cand).max())
        
        if mb < best_mb - 1e-13:
            best_mb = mb
            best_a = cand.copy()
            a = cand.copy()
            no_improve = 0
            sigma = max(0.003, sigma*0.97)
        else:
            no_improve += 1
            if no_improve > 300:
                sigma = min(0.2, sigma*1.8)
                no_improve = 0
    
    return best_a, best_mb

def _init(n, rng, v=0):
    i = np.arange(n, dtype=float)
    if v % 5 == 0:
        a = 1.0 + 0.3*np.sin(2*np.pi*rng.uniform(1,5)*i/n) + 0.2*np.cos(2*np.pi*rng.uniform(5,20)*i/n)
    elif v % 5 == 1:
        a = np.exp(rng.normal(0, 0.6, n))
    elif v % 5 == 2:
        a = np.ones(n)
        for _ in range(3):
            c = rng.uniform(0, n)
            w = rng.uniform(n//8, n//3)
            a += 0.5*np.exp(-(i-c)**2/(2*w**2))
    elif v % 5 == 3:
        a = 1.0 + 0.4*np.abs(np.sin(np.pi*rng.uniform(1,4)*i/n))
    else:
        a = rng.uniform(0.5, 1.5, n)
    a = np.maximum(a, 0.01)
    return _normalize(a)

    t0 = time.monotonic()
    budget = max(1.0, float(budget_s))
    deadline = t0 + budget - 15.0
    
    rng = np.random.default_rng(seed)
    
    parent = _get_parent()
    n_p = len(parent)
    
    # Evaluate parent
    best_seq = np.clip(parent, 0, 1000).copy()
    try:
        best_score = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        best_score = 2.0 * n_p * float(_conv_self(_normalize(best_seq)).max())
    
    if not np.isfinite(best_score):
        best_seq = np.ones(64)
        best_score = 2.0
    
    # Phase 1: LP on parent (most promising)
    dl1 = t0 + budget * 0.40
    if time.monotonic() < dl1:
        try:
            ra, rmb = _lp_refine(best_seq.copy(), dl1, rng)
            sc = 2.0 * n_p * rmb
            if sc < best_score - 1e-10:
                es = float(evaluate_sequence([float(x) for x in ra]))
                if np.isfinite(es) and es < best_score:
                    best_seq, best_score = ra.copy(), es
        except Exception:
            pass
    
    # Phase 2: Restarts at parent length
    dl2 = t0 + budget * 0.65
    if time.monotonic() < dl2:

                break
            ai = _init(n_p, rng, rid+1)
            sdl = time.monotonic() + max(10, min(50, (dl2-time.monotonic())/max(1,6-rid) - 5))
            try:
                ra, rmb = _lp_refine(ai, sdl, rng)
                sc = 2.0 * n_p * rmb
                if sc < best_score - 1e-10:
                    es = float(evaluate_sequence([float(x) for x in ra]))

            except Exception:
                pass
    
    # Phase 3: Local search polish
    dl3 = t0 + budget * 0.82
    if time.monotonic() < dl3:
        try:
            ra, rmb = _local_search(best_seq.copy(), dl3, np.random.default_rng(seed+5000))
            sc = 2.0 * len(ra) * rmb
            if sc < best_score - 1e-12:
