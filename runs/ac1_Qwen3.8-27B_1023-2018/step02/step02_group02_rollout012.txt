                # Two-point transfer
                cand = a.copy()
                k1 = int(rng2.integers(0, n))
                k2 = int(rng2.integers(0, n))
                if k1 == k2:
                    continue
                amt = 0.005 * a.max() * rng2.uniform(0.5, 2.0)
                cand[k1] += amt
                cand[k2] -= amt
                cand = np.maximum(cand, 1e-12)
                cand /= cand.sum()
            elif r < 0.80:
                # Blend with reversed
                t_mix = rng2.uniform(0.03, 0.15)
                cand = (1 - t_mix) * a + t_mix * a[::-1]
                cand /= cand.sum()
            else:
                # Single-point nudge
                cand = a.copy()
                k = int(rng2.integers(0, n))
                delta = rng2.normal(0, 0.01 * a.max())
                cand[k] = max(1e-12, cand[k] + delta)
                cand /= cand.sum()

            cand = np.maximum(cand, 0.0)
            cand_sum = float(cand.sum())
            if cand_sum < 1e-8 or not np.isfinite(cand).all():
                continue
            cand /= cand_sum
            cand_sc = 2.0 * n * peak(convolve_full(cand))
            if cand_sc < sc - 1e-14:
                a = cand.copy()
                sc = cand_sc
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_f = a.copy()
                sigma = max(0.005, sigma * 0.985)
            else:
                sigma = min(0.25, sigma * 1.015)

    # =========================================================
    # PHASE 4: One more LP pass from best found
    # =========================================================
    if time.monotonic() < deadline - 20.0:
        f = best_f.copy()
        no_improve2 = 0
        for it in range(2000):
            if time.monotonic() > deadline - 20.0 or no_improve2 >= 20:
                break
            c_curr = convolve_full(f)
            M = peak(c_curr)
            new_f, new_M = lp_step(f, M)
            if new_f is None:
                no_improve2 += 1
                if no_improve2 % 5 == 0:
                    rng_q = np.random.default_rng(seed + 50000 + it)
                    f = f + rng_q.normal(0, 0.02 / math.sqrt(n), n)
                    f = np.maximum(f, 1e-12)
                    f /= f.sum()
                continue
            f = new_f
            no_improve2 = 0
            sc = 2.0 * n * new_M
            if sc < best_score - 1e-14:
                best_score = sc
                best_f = f.copy()

    # =========================================================
    # Final validation with exact evaluator
    # =========================================================
    best_f = np.clip(best_f, 0.0, 1000.0)
    if not np.isfinite(best_f).all() or best_f.sum() < 0.01:
        best_f = np.ones(n) / n

    final_list = [float(x) for x in best_f]
    try:
        fs = float(evaluate_sequence(final_list))
    except Exception:
        fs = float('inf')

    if not np.isfinite(fs):
        # Fallback: uniform
        final_list = [1.0 / n] * n
        try:
            fs = float(evaluate_sequence(final_list))
        except Exception:
            pass
        if not np.isfinite(fs):
            return [1.0]

    return final_list
