def _project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

def _eval(a):
    a = np.clip(a, 0.0, 1000.0)
    if a.size == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    b = np.convolve(a, a)
    return float(2.0 * a.size * float(b.max()) / (s * s))


def _project_simplex(v):
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if rho.size == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _lp_solve(f, time_limit=8.0):
    if linprog is None:
        return None
    n = f.size
    b = np.convolve(f, f)
    M = float(b.max())
    if not math.isfinite(M) or M < 1e-300:
        return None
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    c = -np.ones(n, dtype=np.float64)
    ub = np.full(2 * n - 1, M * 1.0 + 1e-12, dtype=np.float64)
    try:
        r = linprog(c, A_ub=A, b_ub=ub, bounds=[(0.0, None)] * n,
                    method='highs', options={'presolve': True, 'time_limit': time_limit})
        if r is not None and r.success and r.x is not None:
            g = np.clip(np.nan_to_num(r.x, nan=0.0, posinf=0.0, neginf=0.0), 0.0, None)
            sg = float(g.sum())

    except Exception:

    return None

            continue

        na = float(np.linalg.norm(a))

        cos_sim = ga / (na * ng + 1e-30)

def _fista(a, stop_t, best, max_iter=50000):
    n = a.size
    a = _project_simplex(a.astype(np.float64).copy())
    y = a.copy()
    t_k = 1.0
    loc = _eval(a)
    _update_best(a, best)
    L = 4.0
    eta = 1.0 / L
    back = 0
    
    for it in range(max_iter):
        if time.monotonic() > stop_t - 2.0:
            break
        b = np.convolve(y, y)
        k_star = int(np.argmax(b))
        grad = np.zeros(n)
        for i in range(n):
            j1 = k_star - i
            if 0 <= j1 < n:
                grad[i] += y[j1]
            j2 = i - k_star
            if 0 <= j2 < n:
                grad[i] += y[j2]
        # Step
        a_new = _project_simplex(y - eta * grad)
        # FISTA momentum
        t_new = 0.5 * (1.0 + math.sqrt(1.0 + 4.0 * t_k * t_k))

        t_k = t_new

            back = 0
