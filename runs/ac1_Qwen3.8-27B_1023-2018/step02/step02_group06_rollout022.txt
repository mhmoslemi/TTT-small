def _optimize(a, deadline, rng):
    n = len(a)
    a = a / a.sum()  # normalize sum to 1
    best_a = a.copy()
    best_score = _score_fast(a)
    
    # Main loop
    momentum = np.zeros(n)
    mu = 0.9  # momentum coefficient
    lr = 0.1
    
    while time.monotonic() < deadline:
        b = _fft_conv(a)
        M = b.max()
        k_star = b.argmax()
        S_a = a.sum()
        score = 2.0 * n * M / (S_a * S_a)
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Subgradient
        grad = _subgrad_single(a, k_star)
        # Adjust for the sum term in the objective
        # ∂S/∂a_m ∝ (grad[m] * S_a - 2*M) / S_a (dropping 2n factor)
        sg = grad * S_a - 2.0 * M
        
        # Momentum update
        momentum = mu * momentum - lr * sg
        a_new = a + momentum
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        # Occasionally perturb
        if rng.random() < 0.05:
            a_new += rng.normal(0, 0.02, n)
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
        
        a = a_new
    
    return best_a, best_score
