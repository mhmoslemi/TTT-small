import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(320, dtype=np.float64)
    if p.size == 0:
        p = np.ones(320, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0 or p.sum() < 0.01:
        p = np.ones(320, dtype=np.float64)
    return p


def _conv_fft(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    S = float(a.sum())
    if S < 0.01:
        return math.inf
    M = float(_conv_fft(a).max())
    if not math.isfinite(M):
        return math.inf
    return 2.0 * n * M / (S * S)


def _build_A(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_direction(f, M):
    n = len(f)
    if M < 1e-15:
        return None
    A = _build_A(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={"presolve": True, "dual_feasibility_tolerance": 1e-8,
                               "primal_feasibility_tolerance": 1e-8})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    s = float(g.sum())
    if s < 1e-10:
        return None
    return g / s

    else:

    best_score = _score(f)
    best_conv = _conv_fft(f).copy()
    best_M = float(best_conv.max())

    t_min = 0.02
    t_max = 0.6
    stall = 0
    max_stall = 150
    iteration = 0
    last_improve_iter = 0

    while time.monotonic() < stop_time - 1.0 and iteration < max_iters:
        conv = _conv_fft(f)
        M = float(conv.max())
        score = 2.0 * n * M / 1.0  # f is normalized to sum 1

        if g is None:
            stall += 5
            if stall > max_stall:

            # Random perturbation

def _polish(a, stop_time, seed=4242):
    n = len(a)
    best = np.clip(np.asarray(a, dtype=np.float64).ravel(), 0, None).copy()
    fs = best.sum()
    if fs < 1e-10:
        return best, math.inf
    best = best / fs
    best_sc = _score(best)
    rng = np.random.default_rng(seed)
    no_imp = 0
    max_no_imp = 8

    multipliers_1d = [0.2, 0.4, 0.6, 0.8, 1.2, 1.5, 2.0, 3.0, 0.0]

    while time.monotonic() < stop_time - 1.0:
        improved = False

        # 1-coordinate moves
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 1.0:
                break
            orig = best[i]
            best_val = orig
            best_val_sc = best_sc
            for mlt in multipliers_1d:
                new_val = orig * mlt if orig > 1e-12 else (0.01 if mlt > 0 else 0.0)
                best[i] = new_val
                if best.sum() < 0.01:
                    best[i] = orig
                    continue
                sc = _score(best)
                if sc < best_val_sc - 1e-13:
                    best_val = new_val
                    best_val_sc = sc
            best[i] = best_val
            if best_val_sc < best_sc - 1e-13:
                best_sc = best_val_sc
                improved = True

        # 2-coordinate moves
        n_2coord = 80
        for _ in range(n_2coord):
            if time.monotonic() > stop_time - 1.0:
                break
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            for mi in (0.5, 1.5, 0.0, 2.0):
                for mj in (0.5, 1.5, 0.0, 2.0):
                    best[i] = oi * mi if oi > 1e-12 else (0.01 if mi > 0 else 0.0)
                    best[j] = oj * mj if oj > 1e-12 else (0.01 if mj > 0 else 0.0)
                    if best.sum() < 0.01:
                        best[i], best[j] = oi, oj
                        continue
                    sc = _score(best)
                    if sc < best_sc - 1e-13:
                        best_sc = sc
                        improved = True
            best[i], best[j] = oi, oj

        if not improved:
            no_imp += 1
            if no_imp >= max_no_imp:
                break
        else:
            no_imp = 0

    return best, best_sc


def _gradient_refine(f, stop_time, seed=99):
    """Use subgradient information to refine."""
    n = len(f)
    f = np.clip(f.copy(), 0, None)
    fs = f.sum()
    if fs < 1e-10:
        return f, math.inf
    f = f / fs
    best_f = f.copy()
    best_sc = _score(f)
    rng = np.random.default_rng(seed)

    lr = 0.05
    for it in range(200):
        if time.monotonic() > stop_time - 2.0:
            break
        conv = _conv_fft(f)
        M = float(conv.max())
        # Find active constraints (within 5% of max)
        thresh = 0.95 * M
        active = np.where(conv >= thresh)[0]
        # Subgradient: average of gradients of active constraints
        # d conv[k]/da[i] = 2*a[k-i] if 0 <= k-i < n
        subgrad = np.zeros(n)
        for k in active:
            for i in range(n):
                idx = k - i
                if 0 <= idx < n:
                    subgrad[i] += 2.0 * f[idx]
        subgrad /= len(active)

        # Gradient step with projection
        f_new = f - lr * subgrad
        f_new = np.clip(f_new, 0, None)
        fs_new = f_new.sum()
        if fs_new < 1e-10:
            break
        f_new = f_new / fs_new
        sc = _score(f_new)
        if sc < best_sc - 1e-12:
            best_f = f_new.copy()
            best_sc = sc
            f = f_new
            lr = min(0.1, lr * 1.02)
        else:
            lr = max(0.005, lr * 0.7)

    return best_f, best_sc

    other_lengths = [128, 160, 192, 256, 384, 448, 512]
    if n_p not in other_lengths:
        other_lengths.append(n_p)
    other_lengths = sorted(set(other_lengths))
    phase2_dur = phase2_end - max(phase1_end, time.monotonic())
    time_per_len = max(3.0, phase2_dur / max(1, len(other_lengths)))

    for L in other_lengths:
        if time.monotonic() > phase2_end - 3.0:
            break
        stop = min(phase2_end - 2.0, time.monotonic() + time_per_len)
        if L == n:
            init_l = best_seq.copy()
        else:
            # Interpolate parent to new length
            init_l = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, len(parent)), parent)
        _run_lp(init_l, L, stop, seed + 50000 + L)

    # Phase 3: Polish the best candidate (remaining time)
    polish_end = deadline - 5.0
    if time.monotonic() < polish_end - 10.0:
        try:
            polished, pol_sc = _polish(best_seq, polish_end)
            _track(polished, pol_sc)
        except Exception:
            pass

    # Final gradient refinement if time allows
    if time.monotonic() < deadline - 8.0:
        try:
            ref_f, ref_sc = _gradient_refine(best_seq, deadline - 6.0, seed + 1234)
            _track(ref_f, ref_sc)
        except Exception:
            pass

    # Final polish
    if time.monotonic() < deadline - 5.0:
        try:
            polished, pol_sc = _polish(best_seq, deadline - 4.0)
            _track(polished, pol_sc)
        except Exception:
            pass

    # Final validation
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in fallback]

    return [float(x) for x in best_seq]
