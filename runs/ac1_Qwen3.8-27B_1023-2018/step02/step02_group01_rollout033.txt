        best_seq = np.ones(64, dtype=np.float64) / 64.0
        best_score = 2.0

    n = n_parent

    # Phase 1: Main LP+gradient optimization at parent length (65% of time)
    phase1_end = t0 + (budget - 25) * 0.65
    try:
        result, _ = _main_optimize(best_seq, stop_time=phase1_end, seed=seed)
        sc = _score(result)
        if np.isfinite(sc) and sc < best_score:
            best_seq = result.copy()
            best_score = sc
    except Exception:
        pass

    # Phase 2: Gradient polish (15% of time)
    phase2_end = t0 + (budget - 25) * 0.80
    if time.monotonic() < phase2_end - 10:
        try:
            result = _gradient_polish(best_seq, stop_time=phase2_end)
            sc = _score(result)
            if np.isfinite(sc) and sc < best_score:
                best_seq = result.copy()
                best_score = sc
        except Exception:
            pass

    # Phase 3: Final LP polish (10% of time)
    phase3_end = t0 + (budget - 25) * 0.92
    if time.monotonic() < phase3_end - 10:
        try:
            result, _ = _main_optimize(best_seq, stop_time=phase3_end, seed=seed + 1)
            sc = _score(result)
            if np.isfinite(sc) and sc < best_score:
                best_seq = result.copy()
                best_score = sc
        except Exception:
            pass

    # Phase 4: Try nearby lengths (remaining time)
    if time.monotonic() < deadline - 25:
        for n_try in [n - 2, n + 2, n - 5, n + 5]:
            if time.monotonic() > deadline - 18:
                break
            if n_try < 20:
                continue
            remaining = deadline - time.monotonic() - 15
            if remaining < 25:
                break
            # Resample to new length
            x_old = np.linspace(0, 1, n)
            x_new = np.linspace(0, 1, n_try)
            init = np.interp(x_new, x_old, best_seq)
            init = np.maximum(init, 0.0)
            s = init.sum()
            if s > 0:
                init = init / s
            else:
                init = np.ones(n_try) / n_try
            try:
                result, _ = _main_optimize(init, stop_time=time.monotonic() + min(remaining, 60),
                                           seed=seed + n_try)
                sc = _score(result)
                if np.isfinite(sc) and sc < best_score:
                    best_seq = result.copy()
                    best_score = sc
            except Exception:
                pass

    # Final validation with exact evaluator
    try:
        final_score = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_score = float('inf')

    if not np.isfinite(final_score):
        return [1.0] * 64

    return [float(x) for x in best_seq]
