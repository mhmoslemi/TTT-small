def _fft_conv(a, b=None):
    """Compute full convolution using FFT."""
    if b is None:

    fsize = next_fast_len(n + m - 1)
    F_a = rfft(a, fsize)
    F_b = rfft(b, fsize)

    j_idx = np.arange(n)[None, :]

    """
    Minimize f(t) = max_k [(1-t)^2 * b_a[k] + 2t(1-t)*c[k] + t^2 * b_g[k]]
    over t in [0, 1] using golden section search.

        vals = (1-t)**2 * b_a + 2*t*(1-t) * c + t**2 * b_g
        return float(np.max(vals))

    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1, f2 = f(c1), f(c2)

            c2 = c1
            f2 = f1

            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)

    else:
        a = a / s
    
    b_a = _fft_conv(a)
    best_mb = float(b_a.max())
    best_a = a.copy()
    
    no_improve = 0
    max_no_improve = 30
    
    while time.monotonic() < deadline:
        b_a = _fft_conv(a)
        M = float(b_a.max())
        
        if M < best_mb * (1 - 1e-12):
            best_mb = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > max_no_improve:
            # Perturb
            a = a + rng.normal(0, 0.05, n)
            a = np.maximum(a, 0)
            a = a / a.sum()
            no_improve = 0
            max_no_improve = int(max(15, max_no_improve * 1.3))
            continue
        
        # Solve LP
        g = _solve_lp(a, M, n)
        
        if g is None:
            # Fallback: subgradient
            k_star = int(np.argmax(b_a))
            grad = np.zeros(n)
            for m in range(n):
                j = k_star - m
                if 0 <= j < n:
                    grad[m] = 2.0 * a[j]
            norm = np.linalg.norm(grad)
            if norm < 1e-15:
                break
            g = a - 0.05 * grad / norm
            g = np.maximum(g, 0)
            g = g / g.sum()
        
        # Check direction quality
        cos_sim = float(np.dot(a, g) / (np.linalg.norm(a) * np.linalg.norm(g) + 1e-30))
        if cos_sim > 0.9999:
            no_improve += 2
            # Small perturbation
            a = a + rng.normal(0, 0.02, n)
            a = np.maximum(a, 0)
            a = a / a.sum()
            continue
        
        # Compute line search quantities
        b_g = _fft_conv(g)
        c = _fft_conv(a, g)
        
        # Golden section line search
        t_opt, f_opt = _golden_section_line_search(b_a, c, b_g)
        
        # Apply update
        if f_opt < M * (1 - 1e-12):
            a_new = (1 - t_opt) * a + t_opt * g
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            a = a_new
        else:
            no_improve += 1
    
    return best_a, best_mb

def _local_polish(a, deadline, rng):
    """Final local polishing with small random perturbations."""
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_mb = float(_fft_conv(a).max())
    
    sigma = 0.02
    no_improve = 0
    
    while time.monotonic() < deadline:
        if no_improve > 200:
            sigma = min(0.3, sigma * 2)
            no_improve = 0
        
        cand = a + rng.normal(0, sigma, n)
        cand = np.maximum(cand, 0)
        cand = cand / cand.sum()
        
        mb = float(_fft_conv(cand).max())
        if mb < best_mb * (1 - 1e-12):
            best_mb = mb
            best_a = cand.copy()
            a = cand.copy()
            no_improve = 0
            sigma = max(0.003, sigma * 0.95)
        else:
            no_improve += 1
    
    return best_a, best_mb
