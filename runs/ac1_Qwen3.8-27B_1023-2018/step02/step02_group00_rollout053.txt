    """Build the (2n-1) x n matrix where A[k,j] = f[k-j] if 0<=k-j<n, else 0."""
    A = np.zeros((2 * n - 1, n))

def _lp_direction(f, n, M):
    """Find g >= 0 maximizing sum(g) s.t. (f*g)[k] <= M for all k."""
    A = _build_A(f, n)
    c = -np.ones(n)  # minimize -sum = maximize sum
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': 5.0})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0, 1000)
    if g.sum() < 1e-15:
        return None
    return g

            if best_val_sc < best_sc - 1e-15:
                best_sc = best_val_sc
                improved = True

        # Two-coordinate
        for _ in range(100):
            if time.monotonic() > stop_time - 4.0:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            for mi in (0.3, 0.5, 0.7, 1.3, 1.8, 0.0):
                for mj in (0.3, 0.5, 0.7, 1.3, 1.8, 0.0):
                    if mi == 1.0 and mj == 1.0:
                        continue
                    best[i] = oi * mi if oi > 1e-15 else 0.0
                    best[j] = oj * mj if oj > 1e-15 else 0.0
                    if best.sum() < 0.01:
                        best[i], best[j] = oi, oj
                        continue
                    sc = _score(best)
                    if sc < best_sc - 1e-15:
                        best_sc = sc
                        improved = True
            best[i], best[j] = oi, oj

        # Small random perturbations
        if not improved:
            for _ in range(30):
                if time.monotonic() > stop_time - 4.0:
                    break
                k = int(rng.integers(n))
                pert = rng.normal(0, 0.02 * max(1.0, best.mean()))
                old = best[k]
                best[k] = max(0.0, old + pert)
                if best.sum() < 0.01:
                    best[k] = old
                    continue
                sc = _score(best)
                if sc < best_sc - 1e-15:
                    best_sc = sc
                    improved = True
                else:
                    best[k] = old

        if not improved:
            no_imp += 1
            if no_imp >= max_no_imp:
                break
        else:
            no_imp = 0

    return best, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 15.0

    parent = _get_parent()
    n_p = len(parent)

    fallback = parent.copy()
    fallback_sc = _score(fallback)
    if not math.isfinite(fallback_sc):
        fallback = np.ones(320) / 320.0
        fallback_sc = _score(fallback)

    best_seq = fallback.copy()
    best_score = fallback_sc

    def _try(n, init_arr, stop, sd):
        nonlocal best_seq, best_score
        try:
            res, sc = _lp_optimize(init_arr, n, stop, sd)
        except Exception:
            return
        if res is None or len(res) == 0:
            return
        if not np.all(np.isfinite(res)):
            return
        res = np.clip(res, 0, 1000)
        if res.sum() < 0.01:
            return
        if sc < best_score - 1e-14:
            try:
                exact = float(evaluate_sequence([float(x) for x in res]))
            except Exception:
                exact = math.inf
            if math.isfinite(exact) and exact < best_score - 1e-14:
                best_seq = res.copy()
                best_score = exact

    # Phase 1: LP on parent length with multiple inits
    n = n_p
    phase1_end = t0 + budget * 0.42
    rng_main = np.random.default_rng(seed)
    i_arr = np.arange(n, dtype=np.float64)

    inits = []
    inits.append(parent.copy())

    # Raised cosine
    inits.append(0.5 * (1.0 + np.cos(2.0 * math.pi * i_arr / n)))

    # Multi-frequency superposition
    sup = np.ones(n) * 0.2
    for fr in (1, 2, 3, 5, 8):
        ph = rng_main.uniform(0, 2 * math.pi)
        sup += 0.2 * np.cos(2.0 * math.pi * fr * i_arr / n + ph)
    inits.append(np.abs(sup) + 0.03)

    # Random abs-normal
    inits.append(np.abs(rng_main.normal(0, 1, n)) + 0.05)

    # Parent + noise
    sig = 0.15 * max(1.0, float(np.mean(parent)))
    inits.append(np.abs(parent + rng_main.normal(0, sig, n)) + 0.01)

    # Centered Gaussian bump
    center = (n - 1) / 2.0
    width = n / 4.0
    inits.append(np.exp(-0.5 * ((i_arr - center) / width) ** 2))

    # Two-lobe
    two = np.exp(-0.5 * ((i_arr - n * 0.3) / (n * 0.15)) ** 2) + np.exp(-0.5 * ((i_arr - n * 0.7) / (n * 0.15)) ** 2)
    inits.append(two)

    time_per_init = max(12.0, (phase1_end - t0) / len(inits))

    for idx, init in enumerate(inits):
        if time.monotonic() > phase1_end - 6.0:
            break
        if best_score < 1.40:
            break
        stop = min(phase1_end - 4.0, time.monotonic() + time_per_init)
        _try(n, init, stop, seed + 7777 * (idx + 1))

    # Phase 2: Other lengths
    phase2_end = t0 + budget * 0.68
    other_lengths = [128, 160, 192, 224, 256, 288, 320, 352, 384, 416, 448, 480, 512]
    other_lengths = [L for L in other_lengths if time.monotonic() < phase2_end - 10]
    time_per_len = max(6.0, (phase2_end - phase1_end) / max(1, len(other_lengths)))

    for L in other_lengths:
        if time.monotonic() > phase2_end - 6.0:
            break
        if best_score < 1.40:
            break
        stop = min(phase2_end - 4.0, time.monotonic() + time_per_len)
        if abs(L - len(best_seq)) <= 4 and best_score < 2.0:
            init_l = best_seq.copy()
        else:
            init_l = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, len(parent)), parent)
        _try(L, init_l, stop, seed + 50000 + L)

    # Phase 3: Second LP pass on best (warm start from best)
    phase3_end = t0 + budget * 0.85
    if time.monotonic() < phase3_end - 20.0:
        stop_p3 = min(phase3_end - 4.0, time.monotonic() + (phase3_end - time.monotonic()))
        _try(len(best_seq), best_seq.copy(), stop_p3, seed + 999999)

    # Phase 4: Polish
    polish_end = deadline - 5.0
    if time.monotonic() < polish_end - 10.0:
        try:
            best_seq, best_score = _polish(best_seq, polish_end)
        except Exception:
            pass

    # Second polish with different seed if time allows
    if time.monotonic() < deadline - 10.0:
        try:
            best_seq2, sc2 = _polish(best_seq, deadline - 6.0, seed=12345)
            if sc2 < best_score - 1e-14:
                best_seq = best_seq2
                best_score = sc2
        except Exception:
            pass

    # Final validation with exact evaluator
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in fallback]

    if final_sc >= best_score - 1e-12:
        best_score = final_sc

    return [float(x) for x in best_seq]
