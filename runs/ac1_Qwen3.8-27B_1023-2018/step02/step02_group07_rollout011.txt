import math
from scipy.fft import rfft, irfft, next_fast_len

def _softmax(x):
    x = np.asarray(x, dtype=np.float64).ravel()
    if x.size == 0:
        return np.array([1.0])
    x = x - np.max(x)
    e = np.exp(np.clip(x, -80.0, 80.0))
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0:
        return np.full(x.size, 1.0 / x.size, dtype=np.float64)
    return e / s

def _conv(a, n, fs):
    F = rfft(a, fs)
    return irfft(F * F, fs)[:2 * n - 1]

    ar = a[::-1]
    c = irfft(rfft(ar, fs) * rfft(q, fs), fs)
    return c[n-1:2*n-1]
