        for t in (1.0, 0.9, 0.8, 0.7, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01):
            h = (1.0 - t) * f_curr + t * g
            m_h = float(convolve_full(h).max())
            if m_h < best_m - 1e-14:
                best_m = m_h
                best_t = t
        
        if best_t > 0:
            return (1.0 - best_t) * f_curr + best_t * g, best_m
        return None, M_curr
    
    for _ in range(50):
        if time.monotonic() > deadline - 120.0:
            break
        c_curr = convolve_full(f)
        M_curr = float(c_curr.max())
        new_f, new_M = lp_step(f, c_curr, M_curr)
        if new_f is None:
            break
        f = new_f
        sc = 2.0 * n * new_M
        if sc < best_score - 1e-13:
            best_score = sc
            best_f = f.copy()
    
    if time.monotonic() < deadline - 60.0:
        a = f.copy()
        sc = 2.0 * n * float(convolve_full(a).max())
        if sc < best_score:
            best_score = sc
            best_f = a.copy()
        
        lr = 0.5
        stuck = 0
        rng = np.random.default_rng(seed)
        
        for it in range(20000):
            if time.monotonic() > deadline - 60.0:
                break
            if stuck > 500:
                a = best_f + 0.005 * rng.normal(0, 1, n)
                a = np.maximum(a, 0.0)
                sa = float(a.sum())
                if sa < 1e-12:
                    a = best_f.copy()
                else:
                    a = a / sa
                sc = 2.0 * n * float(convolve_full(a).max())
                if sc < best_score:
                    best_score = sc
                    best_f = a.copy()
                stuck = 0
                continue
            
            c = convolve_full(a)
            M = float(c.max())
            sc = 2.0 * n * M
            if sc < best_score - 1e-13:
                best_score = sc
                best_f = a.copy()
                stuck = 0
            else:
                stuck += 1
            
            tau = max(1e-10, 0.01 * M)
            x = (c - M) / tau
            x = np.clip(x, -700.0, 0.0)
            ex = np.exp(x)
            s_ex = float(ex.sum())
            if s_ex < 1e-30:
                p = np.zeros_like(c)
                p[int(np.argmax(c))] = 1.0
            else:
                p = ex / s_ex
            
            R = np.correlate(p, a, mode='full')
            h = R[n-1::-1]
            
            grad_amb = 2.0 * n * h
            mean_grad = float(grad_amb.mean())
            step = - (grad_amb - mean_grad)
            
            improved = False
            for scale in (lr, lr * 0.5, lr * 0.25, lr * 0.125, 0.01, 0.005):
                cand = a + scale * step
                cand = np.maximum(cand, 0.0)
                cs = float(cand.sum())
                if cs < 1e-12:
                    continue
                cand /= cs
                cand_sc = 2.0 * n * float(convolve_full(cand).max())
                if cand_sc < sc - 1e-12:
                    a = cand
                    sc = cand_sc
                    improved = True
                    break
            
            if improved:
                lr = min(lr * 1.02, 10.0)
            else:
                lr = max(0.001, lr * 0.5)
    
    if time.monotonic() < deadline - 15.0:
        a = best_f.copy()
        sc = best_score
        rng2 = np.random.default_rng(seed + 31337)
        sigma = 0.05
        
        for _ in range(20000):
            if time.monotonic() > deadline - 15.0:
                break
            
            r = rng2.random()
            if r < 0.5:
                z = np.log(np.clip(a, 1e-12, None))
                z += rng2.normal(0.0, sigma, n)
                z -= z.max()
                cand = np.exp(z)
            elif r < 0.8:
                cand = a.copy()
                k = int(rng2.integers(0, n))
                width = max(1, n // 50)
                lo = max(0, k - width)
                hi = min(n, k + width)
                cand[lo:hi] *= abs(float(np.exp(rng2.normal(0, 0.1))))
            else:
                cand = a.copy()
                k1 = int(rng2.integers(0, n))
                k2 = int(rng2.integers(0, n))
                amt = 0.01 * a[0]
                cand[k1] += amt
                cand[k2] -= amt
            
            cand = np.maximum(cand, 0.0)
            cs = float(cand.sum())
            if cs < 1e-12:
                continue
            cand /= cs
            cand_sc = 2.0 * n * float(convolve_full(cand).max())
            if cand_sc < sc - 1e-13:
                a = cand.copy()
                sc = cand_sc
                if sc < best_score - 1e-13:
                    best_score = sc
                    best_f = a.copy()
                sigma = max(0.01, sigma * 0.97)
            else:
                sigma = min(0.3, sigma * 1.03)
    
    best_f = np.clip(best_f, 0.0, 1000.0)
    if not np.isfinite(best_f).all() or best_f.sum() < 0.01:
        best_f = np.ones(n) / n
        s = float(best_f.sum())
        if s < 0.01:
            best_f = np.ones(192) / 192
    
    final_list = [float(x) for x in best_f]
    try:
        fs = float(evaluate_sequence(final_list))
        if not np.isfinite(fs):
            final_list = [1.0 / n] * n
            fs = float(evaluate_sequence(final_list))
    except Exception:
        final_list = [1.0 / n] * n
        try:
            fs = float(evaluate_sequence(final_list))
        except Exception:
            pass
        if not np.isfinite(fs):
            return [1.0]
    
    return final_list
