            if time.monotonic() < deadline - 15:
                a1 = grad_opt_fft(a1, 200000, deadline - 10,
                                 eps0=0.005, eps_min=0.0002)

    # Phase 4: Nearby and standard lengths
    src = best_arr if best_arr is not None else p
    lengths_to_try = sorted(set(
        [L for L in range(max(6, n_p - 14), min(513, n_p + 15), 2) if L != n_p] +
        [160, 192, 224, 256, 288, 320, 352, 384, 416, 448, 480, 512]
    ))

    for L in lengths_to_try:
        if time.monotonic() >= deadline - 35:
            break
        t_rem = deadline - time.monotonic() - 35
        if t_rem < 12:
            break
        alloc = min(28.0, t_rem * 0.1)
        if alloc < 6:
            break

        a_init = interp_len(src, L)

        if linprog is not None:
            a_w = lp_opt(a_init, max(10, int(alloc * 0.5)),
                        time.monotonic() + alloc * 0.5,
                        margins=(1.0, 0.97, 0.94))
        else:
            a_w = a_init

        if time.monotonic() < deadline - 20:
            a_w = grad_opt_fft(a_w, max(10000, int(alloc * 8)),
                             deadline - 15, eps0=0.008, eps_min=0.001)

    # Phase 5: Diverse restarts at parent length
    if time.monotonic() < deadline - 40:
        n_restarts = 30
        for r in range(n_restarts):
            if time.monotonic() >= deadline - 20:
                break
            t_rem = deadline - time.monotonic() - 20
            alloc = min(18.0, t_rem / max(1, n_restarts - r))
            if alloc < 4:
                break

            r_type = r % 6
            if r_type == 0:
                a_init = p_norm * np.exp(0.06 * rng.standard_normal(n_p))
            elif r_type == 1:
                a_init = p_norm[::-1].copy()
            elif r_type == 2:
                a_init = p_norm * np.exp(0.12 * rng.standard_normal(n_p))
            elif r_type == 3:
                # Smooth low-frequency perturbation
                freqs = np.fft.rfftfreq(n_p)
                mask = np.exp(-freqs * 5.0)
                pert = np.fft.irfft(rng.standard_normal(len(freqs)) * mask, n=n_p)
                a_init = p_norm + 0.03 * pert
            elif r_type == 4:
                a_init = (best_arr if best_arr is not None else p_norm).copy()
            else:
                # Block perturbation
                a_init = p_norm.copy()
                blk = max(4, n_p // 8)
                start = int(rng.integers(0, n_p - blk))
                a_init[start:start + blk] *= np.exp(0.2 * rng.standard_normal(blk))

            a_init = np.clip(a_init, 0, None)
            s_init = float(a_init.sum())
            if s_init > 1e-15:
                a_init /= s_init
            else:
                a_init = p_norm.copy()

            if linprog is not None:
                a_w = lp_opt(a_init, max(8, int(alloc * 0.45)),
                            time.monotonic() + alloc * 0.45,
                            margins=(1.0, 0.96, 0.93))
            else:
                a_w = a_init

            if time.monotonic() < deadline - 12:
                a_w = grad_opt_fft(a_w, int(alloc * 5), deadline - 8,
                                  eps0=0.006, eps_min=0.001)

    # Phase 6: Final polish with very fine steps
    if time.monotonic() < deadline - 8 and best_arr is not None:
        a_pol = best_arr.copy() / best_arr.sum()
        a_pol = grad_opt_fft(a_pol, 100000, deadline - 4,
                           eps0=0.003, eps_min=0.0001)
        ub(a_pol)

    # Final validation
    if best_arr is None:
        best_arr = np.ones(96, dtype=np.float64)
        ub(best_arr)

    final = np.clip(best_arr, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones_like(final)
    ub(final)

    return [float(x) for x in best_arr]
