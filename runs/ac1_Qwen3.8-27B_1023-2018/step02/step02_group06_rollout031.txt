import time
import math
from scipy.fft import rfft, irfft, next_fast_len
import cvxpy as cp

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(192, dtype=np.float64)
    try:
        p = np.clip(p, 0.0, 1000.0)
    except Exception:
        p = np.ones(192, dtype=np.float64)
    if p.size == 0 or not np.isfinite(p).all():
        p = np.ones(192, dtype=np.float64)
    return p

def _fft_conv(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2*n-1]
    np.maximum(c, 0.0, out=c)
    c[~np.isfinite(c)] = 0.0
    return c

def _score(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = _fft_conv(a)
    mb = float(b.max())
    return 2.0 * n * mb / (s * s)

def _build_A(a):
    n = len(a)
    k_idx = np.arange(2*n-1)[:, None]
    j_idx = np.arange(n)[None, :]
    d = k_idx - j_idx
    valid = (d >= 0) & (d < n)
    A = np.zeros((2*n-1, n), dtype=np.float64)
    A[valid] = a[d[valid].astype(np.intp)]
    return A

def _solve_lp(a, M, n, deadline):
    g = cp.Variable(n, nonneg=True)
    A = _build_A(a)
    constraints = [A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    for solver in [cp.ECOS, cp.SCS, cp.GLPK, cp.CBC]:
        try:
            if time.monotonic() > deadline:
                return None
            prob.solve(solver=solver, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                gv = g.value
                if gv is not None and np.isfinite(gv).all():
                    gv = np.maximum(gv, 0.0)
                    s = gv.sum()
                    if s > 1e-12:
                        return gv / s
        except Exception:
            continue
    return None

def _subgrad(a, n):
    b = _fft_conv(a)
    k_star = int(np.argmax(b))
    g = np.zeros(n)
    for m in range(n):
        j = k_star - m
        if 0 <= j < n:
            g[m] = 2.0 * a[j]
    return g

def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def _line_search(a, direction, n, current_mb, max_steps=20):
    best_t = 0.0
    best_mb = current_mb
    t = 0.05
    for _ in range(max_steps):
        a_new = _normalize((1 - t) * a + t * direction)
        b = _fft_conv(a_new)
        mb = float(b.max())
        s_new = a_new.sum()
        s_old = a.sum()
        # Compare score-like quantity (mb * n / s^2)
        scale_factor = (s_old / s_new) ** 2
        eff_mb = mb * scale_factor
        if eff_mb < best_mb - 1e-14:
            best_mb = eff_mb
            best_t = t
            t = min(1.0, t * 1.3)
        else:
            t = t * 0.6
            if t < 0.001:
                break
    return best_t, best_mb

def _lp_iteration(a, n, deadline):
    """One LP-based Sion iteration."""
    b = _fft_conv(a)
    M = float(b.max())
    s = a.sum()
    if s < 1e-15:
        return a, M
    # M needs to be scaled: constraint is (a*g)_k <= M where g is normalized
    # Actually the constraint is (a*g)_k <= M, and we want g with sum(g)=1
    # So M should be the current max of (a*a)
    g = _solve_lp(a, M, n, deadline)
    if g is None:
        return a, M
    # Check improvement potential
    dot = _fft_conv(a @ np.ones(1) * 1.0)  # just a*a
    # Line search
    t, _ = _line_search(a, g, n, M)
    if t > 0.001:
        a_new = _normalize((1 - t) * a + t * g)
    else:
        a_new = a.copy()
    return a_new, float(_fft_conv(a_new).max())

def _sion_optimize(a_init, n, deadline, rng, max_iters=500):
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_score = _score(a)
    no_improve = 0
    patience = 60
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = _fft_conv(a)
        M = float(b.max())
        cur_score = _score(a)
        
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > patience:
            # Perturb to escape
            strength = 0.03 * (1 + no_improve / patience)
            a = a + rng.normal(0, strength, n)
            a = _normalize(a)
            no_improve = 0
            patience = max(30, patience * 1.3)
            continue
        
        # LP direction
        if time.monotonic() > deadline - 5:
            break
        g = _solve_lp(a, M, n, deadline - 2)
        
        if g is not None:
            # Check if g gives improvement
            cos_sim = float(np.dot(a, g) / (np.linalg.norm(a) * np.linalg.norm(g) + 1e-30))
            if cos_sim < 0.9999:
                t, _ = _line_search(a, g, n, M)
                if t > 0.005:
                    a_new = _normalize((1 - t) * a + t * g)
                    if _score(a_new) < cur_score:
                        a = a_new
                    else:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 2
                # Try subgradient
                sg = _subgrad(a, n)
                sg -= sg.mean()
                a_new = _normalize(a - 0.05 * sg)
                if _score(a_new) < cur_score:
                    a = a_new
        else:
            # LP failed, use subgradient
            sg = _subgrad(a, n)
            sg -= sg.mean()
            gn = np.linalg.norm(sg)
            if gn > 1e-10:
                for lr in [0.1, 0.05, 0.02, 0.01]:
                    a_new = _normalize(a - lr * sg / gn)
                    if _score(a_new) < cur_score:
                        a = a_new
                        break
                else:
                    no_improve += 1
            else:
                no_improve += 5
    
    return best_a, best_score

def _coordinate_descent(a, n, deadline, rng, max_rounds=200):
    """Polish via coordinate descent on the score."""
    a = _normalize(a.copy())
    best_a = a.copy()
    best_score = _score(a)
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        # Try each coordinate (subset for large n)
        if n <= 200:
            indices = np.arange(n)
        else:
            indices = np.random.default_rng(rnd).choice(n, size=min(n, 64), replace=False)
        
        for idx in indices:
            if time.monotonic() > deadline:
                break
            orig = a[idx]
            # Try a few values
            for factor in [0.5, 0.7, 0.85, 1.0 + 0.05, 1.15, 1.3, 1.5]:
                a[idx] = orig * factor
                a = _normalize(a)
                sc = _score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                else:
                    a[idx] = orig
                    a = _normalize(a)
                    break
            # If orig*factor was worse, check lower
            if not improved and a[idx] != orig:
                a[idx] = orig
                a = _normalize(a)
    
    if not improved:
        # Random perturbation
        a = a + rng.normal(0, 0.01, n)
        a = _normalize(a)
    
    return best_a, best_score

def _multistart(a_init, n, deadline, rng):
    """Run Sion from multiple starting points."""
    best_a = _normalize(a_init.copy())
    best_score = _score(best_a)
    
    # Try several perturbations of the initial point
    attempts = [a_init.copy()]
    for _ in range(5):
        pert = a_init + rng.normal(0, 0.15, n)
        attempts.append(_normalize(pert))
    # Smooth version
    from scipy.ndimage import uniform_filter1d
    attempts.append(_normalize(uniform_filter1d(a_init, size=5, mode='wrap')))
    # Exp re-encoding
    log_a = np.log(a_init + 1e-6)
    log_noised = log_a + rng.normal(0, 0.3, n)
    attempts.append(_normalize(np.exp(log_noised)))
    
    for i, a_try in enumerate(attempts):
        if time.monotonic() > deadline - 10:
            break
        sub_dl = min(deadline - 5, time.monotonic() + max(20, (deadline - time.monotonic()) / max(1, len(attempts) - i)))

        try:
            res_a, res_sc = _sion_optimize(a_try, n, sub_dl, rng)
            if res_sc < best_score:
                best_score = res_sc
                best_a = res_a.copy()
        except Exception:
            pass
    
    return best_a, best_score
