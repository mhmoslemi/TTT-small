- Start with normalized f (sum=1)
- Each iteration:
  1. Compute b = f*f, M = max(b), k* = argmax(b)
  2. Solve LP: max 1^T g s.t. (f*g)[k] <= M, g >= 0
  3. Normalize g to sum=1
  4. Line search for t in [0, 0.5]: minimize score((1-t)f + tg)
  5. Update f
  6. If improvement < threshold, perturb and continue

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(320, dtype=np.float64)

    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(320, dtype=np.float64)
    return p


def _conv_fft(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[: 2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_fft(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n):
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j : j + n, j] = f
    return A


def _lp_solve(f, n, M):
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(
            c, A_ub=A, b_ub=b_ub,
            bounds=[(0.0, None)] * n,
            method="highs",
            options={"presolve": True, "time_limit": 30.0},
        )
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    gs = float(g.sum())
    if gs < 1e-14:
        return None
    return g / gs


def _line_search(f, g_dir, n, best_sc, t_init=0.30, stop_time=None):
    t_lo, t_hi = 0.0, t_init
    f_lo = f
    for it in range(18):
        if stop_time and time.monotonic() > stop_time:
            break
        t_mid = 0.5 * (t_lo + t_hi)
        fn = (1.0 - t_mid) * f + t_mid * g_dir
        np.maximum(fn, 0.0, out=fn)
        fs = float(fn.sum())
        if fs < 1e-10:
            t_hi = t_mid
            continue
        sc = _score(fn)
        if sc < best_sc - 1e-14:
            t_lo = t_mid
            f_lo = fn.copy()
        else:
            t_hi = t_mid
        if t_hi - t_lo < 2e-4:
            break
    if t_lo < 1e-5:
        return None, None
    return t_lo, f_lo


def _lp_loop(f0, n, stop_time, seed):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    fs = float(f.sum())
    if fs < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / fs
    best_f = f.copy()
    best_sc = _score(f)
    t_cur = 0.30
    stall = 0
    max_stall = 60
    it = 0

    while time.monotonic() < stop_time - 2.0:
        it += 1
        b = _conv_fft(f)
        M = float(b.max())
        if M < 1e-15:
            break

        g_dir = _lp_solve(f, n, M)
        if g_dir is None:
            stall += 5
            if stall > max_stall:
                break
            f = np.clip(f + rng.normal(0, 0.05, n), 0, None)
            fs2 = f.sum()
            f = f / fs2 if fs2 > 1e-10 else best_f.copy()

def _subgrad_polish(f0, n, stop_time, seed):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    fs = float(f.sum())
    f = f / fs
    best_f = f.copy()
    best_sc = _score(f)
    lr = 0.01
    momentum = np.zeros(n)
    beta = 0.85
    no_imp = 0

    for it in range(200000):
        if time.monotonic() > stop_time - 2.0:
            break

        b = _conv_fft(f)
        M = float(b.max())
        # Find top-K lags for better subgradient
        topk = min(5, 2 * n - 1)
        idx = np.argpartition(b, -topk)[-topk:]
        weights = b[idx].copy()
        weights = weights - weights.min() + 1e-10
        weights /= weights.sum()

        grad_M = np.zeros(n)
        for w, k in zip(weights, idx):
            lo = max(0, int(k) - n + 1)
            hi = min(n, int(k) + 1)
            for i in range(lo, hi):
                if 0 <= int(k) - i < n:
                    grad_M[i] += w * 2.0 * f[int(k) - i]

        s = float(f.sum())
        grad_F = (2.0 * n / (s * s)) * grad_M - (4.0 * n * M / (s * s * s)) * np.ones(n)

        momentum = beta * momentum - lr * grad_F
        f_new = f + momentum
        f_new = np.maximum(f_new, 0.0)
        fs_new = float(f_new.sum())
        if fs_new < 1e-10:
            f_new = f.copy()
        else:
            f_new = f_new / fs_new

        sc = _score(f_new)
        if sc < best_sc - 1e-14:
            f = f_new
            best_f = f.copy()
            best_sc = sc
            no_imp = 0
            lr = min(0.05, lr * 1.01)
        else:
            no_imp += 1
            lr = max(0.001, lr * 0.97)
            if no_imp > 300:
                f = f_new.copy()
                no_imp = 0
                lr = 0.01
                beta = 0.7
                momentum = np.zeros(n)

    return best_f, best_sc


def _coord_polish(f0, n, stop_time, seed):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    fs = float(f.sum())
    f = f / fs

    no_imp = 0
    multipliers = (0.2, 0.4, 0.6, 0.8, 1.25, 1.5, 2.0, 0.0)

    while time.monotonic() < stop_time - 2.0:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 2.0:
                return f, best_sc
            orig = f[i]
            for mlt in multipliers:
                f[i] = orig * mlt if orig > 1e-12 else (0.01 if mlt > 0 else 0.0)
                sc = _score(f)
                if sc < best_sc - 1e-13:
                    best_sc = sc
                    improved = True
                    orig = f[i]  # keep improvement
                    orig_base = f[i]
                else:
                    f[i] = orig
            # Try fine tune around best

                    f[i] = max(0.0, orig + dt * orig)
                    sc = _score(f)
                    if sc < best_sc - 1e-13:
                        best_sc = sc
                        improved = True
        # 2-coord moves

                    if sc < best_sc - 1e-13:

            no_imp += 1
            if no_imp >= 10:
                break
        else:
            no_imp = 0

    return f, best_sc
