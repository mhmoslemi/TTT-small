def _subgradient_polish(a, stop_time, seed=123):
    """Subgradient method for fine-tuning."""
    n = len(a)
    f = np.clip(np.asarray(a, dtype=np.float64).ravel(), 0, None).copy()
    fs = f.sum()
    if fs < 1e-10:
        return f, _score(f)
    f = f / fs
    best_f = f.copy()
    best_sc = _score(f)
    
    lr = 1.0 / (n ** 0.5)
    decay = 0.999
    rng = np.random.default_rng(seed)
    
    for it in range(5000):
        if time.monotonic() > stop_time - 3.0:
            break
        
        c = _conv_full(f)
        k_star = int(np.argmax(c))
        M = float(c[k_star])
        
        # Subgradient of max(c) w.r.t. f_j:
        # ∂c[k_star]/∂f_j = f[k_star - j] if 0 <= k_star - j <= n-1
        # Plus the symmetric term (but for full conv, it's just f[k_star-j])
        # Actually: c[k] = sum_i f[i]*f[k-i]
        # ∂c[k]/∂f[j] = sum_i (δ_{ij}*f[k-i] + f[i]*δ_{k-i,j})
        # = f[k-j] + f[k-j] = 2*f[k-j] if 0 <= k-j <= n-1
        # Wait, the second term: sum_i f[i]*δ_{k-i,j} = f[k-j] if k-j is valid
        # And the first term: sum_i δ_{ij}*f[k-i] = f[k-j] if k-j is valid
        # So total = 2*f[k-j] if 0 <= k-j <= n-1, else 0.
        
        # Hmm wait, that's the subgradient of c[k_star] specifically.
        # For the max, the subgradient is in the convex hull of subgradients of active components.
        # If only one k is active (k_star), then subgrad = 2*f[shifted by k_star].
        
        sg = np.zeros(n)
        # 2 * f[k_star - j] for j = 0..n-1 where 0 <= k_star - j <= n-1
        # i.e., j ranges from max(0, k_star-(n-1)) to min(k_star, n-1)
        j_min = max(0, k_star - (n - 1))
        j_max = min(k_star, n - 1)
        sg[j_min:j_max+1] = 2.0 * f[k_star - j_max : k_star - j_min - 1 : -1]
        
        # Normalize subgradient
        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-15:
            break
        sg = sg / sg_norm
        
        # Projected gradient step (project onto sum=1, f>=0)
        f_new = f - lr * sg
        f_new = np.clip(f_new, 0, None)
        fs_new = f_new.sum()
        if fs_new > 1e-10:
            f_new = f_new / fs_new
        
        sc = _score(f_new)
        if sc < best_sc - 1e-14:
            best_f = f_new.copy()
            best_sc = sc
        else:
            # Keep best, but still move (to explore)
            f = f_new
        
        lr *= decay
    
    return best_f, best_sc
