Let me now think about the overall algorithm more carefully:

**Main loop:**
1. Normalize a so sum(a) = 1
2. Compute b = conv(a, a), M = max(b)
3. Solve LP: min M' s.t. A(a)g <= M', sum(g)=1, g>=0
4. Get g from LP solution
5. Line search: try t in {0.1, 0.2, ..., 1.0}, compute score of (1-t)a + t*g, pick best
6. If improved, update a; else, try different approaches (gradient, perturbation)
7. Repeat until time runs out or convergence

**Convergence detection:**
- If the LP optimal value M' is very close to M (max(a*a)), we're near a fixed point
- If line search doesn't find improvement, we're in a local minimum

**Escape from local minima:**
- Add random perturbation
- Try a different n
- Use gradient descent with different step sizes

Let me also add a "soft" gradient step as an alternative when the LP doesn't help:
- Compute the gradient of the log-sum-exp approximation
- Take a step in that direction
- Project onto the feasible set (a >= 0, sum(a) = 1)

One more thought: the LP I described minimizes M' = max(A(a)g) subject to sum(g)=1, g>=0. This is a standard "bottleneck" LP. The optimal M' tells us the best we can do with a single "response" to a.

def propose_candidate(seed=42, budget_s=1300, **kwargs):
    from scipy.optimize import linprog
    import scipy.sparse as sps

    rng = np.random.default_rng(seed)

    def ub(a):

            a = np.asarray(a, dtype=np.float64).ravel()
            a = np.clip(a, 0.0, 1000.0)
            if a.size == 0 or a.sum() < 0.01:

            v = evaluate_sequence([float(x) for x in a])

    try:
        h = globals().get('height_sequence_1', None)

            parent = np.clip(parent, 0, 1000)

                ub(parent)
            else:
                parent = np.ones(n_parent)
        else:
            parent = np.ones(n_parent)

        parent = np.ones(n_parent)

    if best_arr is None:
        ub(parent)

    def score_norm(a, n):
        """Score assuming sum(a)=1: 2n * max(conv(a,a))"""
        b = np.convolve(a, a)
        return 2.0 * n * float(b.max())

    def optimize_lp(a_init, n, time_stop, max_stag=30):
        """LP-based fixed-point iteration"""
        a = np.asarray(a_init, dtype=np.float64).ravel().copy()
        a = np.clip(a, 0, None)
        s = a.sum()
        if s < 1e-15:
            return a
        a /= s
        local = score_norm(a, n)
        ub(a)

        # Precompute sparsity pattern
        total_nz = n * n
        rows = np.empty(total_nz, dtype=np.int32)
        cols = np.empty(total_nz, dtype=np.int32)
        cnt = 0
        for k in range(2 * n - 1):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            nn = hi - lo + 1
            rows[cnt:cnt + nn] = k
            cols[cnt:cnt + nn] = np.arange(lo, hi + 1)
            cnt += nn

        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        b_ub_vec = np.zeros(2 * n - 1)
        A_eq_row = np.zeros((1, n + 1))
        A_eq_row[0, :n] = 1.0
        b_eq_vec = np.array([1.0])
        bounds_lp = [(0, None)] * n + [(None, None)]
        neg_I = -sps.eye(2 * n - 1, format='csr')

        stag = 0
        while time.monotonic() < time_stop:
            # Build A(a) values
            vals = np.empty(cnt, dtype=np.float64)
            idx = 0
            for k in range(2 * n - 1):
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                nn = hi - lo + 1
                vals[idx:idx + nn] = a[k - hi:k - lo + 1][::-1]
                idx += nn

            A_conv = sps.csr_matrix((vals, (rows[:cnt], cols[:cnt])), shape=(2 * n - 1, n))
            A_ub_mat = sps.bmat([[A_conv, neg_I]], format='csr')

            try:
                res = linprog(c_obj, A_ub=A_ub_mat, b_ub=b_ub_vec,
                              A_eq=A_eq_row, b_eq=b_eq_vec,
                              bounds=bounds_lp, method='highs',
                              options={'presolve': True, 'time_limit': 10})
            except Exception:

                    break
                continue

            g = np.clip(res.x[:n], 0, None)
            sg = g.sum()
            if sg < 1e-15:
                stag += 1
                continue
            g /= sg

            best_sc = local
            best_c = None
            for t in (0.5, 0.3, 0.7, 1.0, 0.1, 0.2, 0.4, 0.6, 0.8, 0.9):
                cand = (1.0 - t) * a + t * g
                cs = cand.sum()
                if cs < 1e-15:

                cand /= cs
                sc = score_norm(cand, n)

                a = best_c
                local = best_sc

                                local = psc
                                stag = 0
                                ub(a)
                                break

    def optimize_grad(a_init, n, time_stop):
        """Gradient-based polishing with subgradient of max"""
        a = np.asarray(a_init, dtype=np.float64).ravel().copy()
        a = np.clip(a, 0, None)
        s = a.sum()

            return a

                js = np.arange(lo, hi + 1)

            gn = float(np.linalg.norm(grad))
            if gn < 1e-14:
