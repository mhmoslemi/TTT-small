        if time.monotonic() < deadline - 25:
            t_rem2 = deadline - time.monotonic() - 18
            alloc2 = min(200.0, t_rem2 * 0.5)
            if alloc2 > 15:
                a1 = lp_iterate(a1, n_p, 1000, time.monotonic() + alloc2)
                ub(a1)
                if time.monotonic() < deadline - 12:
                    a1 = subgrad_opt(a1, n_p, 15000, deadline - 8)
                    ub(a1)

    # Phase 2: Try nearby and other lengths
    if time.monotonic() < deadline - 50:
        lengths_to_try = []
        for L in [n_p - 16, n_p - 8, n_p - 4, n_p - 2, n_p + 2, n_p + 4, n_p + 8, n_p + 16,
                  128, 160, 192, 224, 256, 288, 320, 352, 384, 448, 512, 576]:
            if 8 <= L <= 640 and L != n_p:
                lengths_to_try.append(L)
        lengths_to_try = sorted(set(lengths_to_try))

        for L in lengths_to_try:
            if time.monotonic() >= deadline - 30:
                break
            t_rem = deadline - time.monotonic() - 30
            if t_rem < 18:
                break
            alloc = min(35.0, t_rem * 0.14)
            if alloc < 10:
                continue

            a_init = interp_len(p, L)
            a_w = lp_iterate(a_init, L, max(15, int(alloc)), time.monotonic() + alloc * 0.6)
            ub(a_w)
            if time.monotonic() < deadline - 20:
                a_w = subgrad_opt(a_w, L, max(2000, int(alloc * 3)), deadline - 15)
                ub(a_w)

    # Phase 3: Multiple restarts at parent length
    if time.monotonic() < deadline - 35:
        n_restarts = 12
        for r in range(n_restarts):
            if time.monotonic() >= deadline - 20:
                break
            t_rem = deadline - time.monotonic() - 20
            alloc = min(22.0, t_rem / max(1, n_restarts - r))
            if alloc < 6:
                break

            if r % 4 == 0:
                a_init = p_norm * np.exp(0.12 * rng.standard_normal(n_p))
            elif r % 4 == 1:
                a_init = p_norm[::-1].copy()
            elif r % 4 == 2:
                # Gaussian-smoothed
                kernel = np.exp(-0.5 * (np.arange(n_p) - n_p/2)**2 / (n_p/6)**2)
                a_init = p_norm * (0.7 + 0.3 * kernel)
            else:
                a_init = p_norm * np.exp(0.2 * rng.standard_normal(n_p))

            a_init = np.clip(a_init, 0, None)
            s_init = float(a_init.sum())
            if s_init > 1e-15:
                a_init /= s_init
            else:
                continue

            a_w = lp_iterate(a_init, n_p, max(8, int(alloc * 0.6)), time.monotonic() + alloc * 0.55)
            ub(a_w)
            if time.monotonic() < deadline - 15:
                a_w = subgrad_opt(a_w, n_p, int(alloc * 1.2), deadline - 10)
                ub(a_w)

    # Phase 4: Final polish of best found
    if best_arr is not None and time.monotonic() < deadline - 8:
        n_b = best_arr.size
        a_pol = norm_sum(best_arr.copy())

        if time.monotonic() < deadline - 12:
            a_pol = lp_iterate(a_pol, n_b, 100, time.monotonic() + min(12, deadline - time.monotonic() - 4))
            ub(a_pol)

        if time.monotonic() < deadline - 4:
            a_pol = subgrad_opt(a_pol, n_b, 10000, deadline - 2)
            ub(a_pol)

    # Final validation
    if best_arr is None:
        best_arr = np.ones(96, dtype=np.float64)
    final = np.clip(best_arr, 0.0, 1000.0)
    s_final = float(final.sum())
    if s_final < 0.01:
        final = np.ones_like(final) * 0.02
    ub(final)

    return [float(x) for x in best_arr]
