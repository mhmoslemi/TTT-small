```python
def line_search(f, g, ff, fg, gg):
    """Find t in [0,1] minimizing max_k ((1-t)f+tg)^2[k]"""
    # Precompute the three convolutions
    # quad_k(t) = (1-t)^2 * ff[k] + 2t(1-t) * fg[k] + t^2 * gg[k]
    #          = ff[k] + t*(2fg[k] - 2ff[k]) + t^2*(gg[k] - 2fg[k] + ff[k])
    
    # For each k, this is a quadratic in t.
    # The max of these quadratics is convex in t.
    # Find the minimum using golden section.
    
    a, b = 0.0, 1.0
    gr = (math.sqrt(5) - 1) / 2  # golden ratio
    
    def obj(t):
        quad = (1-t)**2 * ff + 2*t*(1-t) * fg + t**2 * gg
        return float(quad.max())
    
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = obj(c)
    fd = obj(d)
    
    for _ in range(40):
        if fc < fd:
            b = d; d = c; fd = fc
            c = b - gr * (b - a)
            fc = obj(c)
        else:
            a = c; c = d; fc = fd
            d = a + gr * (b - a)
            fd = obj(d)
    
    t_best = (a + b) / 2
    return t_best, obj(t_best)
```
