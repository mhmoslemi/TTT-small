import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def propose_candidate(seed=42, budget_s=1300, **kwargs):

    deadline = t0 + max(1.0, float(budget_s)) - 5.0
    rng = np.random.default_rng(seed)

    # --- Retrieve parent ---
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
    except Exception:
        parent = np.ones(192, dtype=np.float64)

        parent = np.ones(192, dtype=np.float64)

    global_best_sc = float('inf')

    def track_best(seq, n_seq):
        nonlocal global_best_seq, global_best_sc
        seq = np.clip(seq, 0, 1000).astype(np.float64)
        s = seq.sum()
        if s < 0.01:
            return
        c = _conv(seq, n_seq)
        sc = 2.0 * n_seq * float(c.max()) / (s * s)
        if sc < global_best_sc:
            global_best_sc = sc
            global_best_seq = seq.copy()
        return sc

    def _conv(a, n_a):
        fs = next_fast_len(2 * n_a - 1)
        F = rfft(a, fs)
        c = irfft(F * F, fs)[:2 * n_a - 1]
        np.maximum(c, 0.0, out=c)
        c[~np.isfinite(c)] = 0.0
        return c

    # Optimize a single sequence of given length
    # ============================================================
    def optimize_sequence(f_init, n, time_limit_frac):
        """Run LP + gradient + perturbation. Returns best (f, score)."""
        f = f_init.astype(np.float64).copy()
        f = np.maximum(f, 0)
        fs = f.sum()
        if fs < 1e-15:
            f = np.ones(n) / n
            fs = 1.0
        f = f / fs

        fsize = next_fast_len(2 * n - 1)

        def conv(a):
            F = rfft(a, fsize)
            c = irfft(F * F, fsize)[:2 * n - 1]
            np.maximum(c, 0.0, out=c)
            c[~np.isfinite(c)] = 0.0
            return c

        M0 = float(conv(f).max())

            M = float(c_curr.max())

                break

            B = np.zeros((2 * n - 1, n), dtype=np.float64)
            for j in range(n):
                B[j:j + n, j] = f

            bounds = [(0.0, None)] * n

            try:

                              method='highs', options={'presolve': True, 'time_limit': 3.0})
            except Exception:
                break

            if not res.success or res.x is None:
                break

            g = np.maximum(res.x, 0.0)

            if gs < 1e-14:

                        track_best(best_f, n)
