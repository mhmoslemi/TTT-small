```python
    # Variables: [g_0, ..., g_{n-1}, t]
    # c = [0, 0, ..., 0, 1] (minimize t)
    # A_ub: [A_mat | -1] @ [g; t] <= 0
    # A_eq: [1^T | 0] @ [g; t] = 1
    # bounds: g_i >= 0, t free (or t >= 0)
    
    A_conv = np.zeros((2*n-1, n))
    for j in range(n):
        A_conv[j:j+n, j] = a
    
    nv = n + 1  # variables
    c = np.zeros(nv)
    c[n] = 1.0  # minimize t
    
    A_ub = np.zeros((2*n-1, nv))
    A_ub[:, :n] = A_conv
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2*n-1)
    
    A_eq = np.zeros((1, nv))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]  # t >= 0
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs', options={'presolve': True})
    except:
        return None, None
    if res.status != 0:
        return None, res.fun if res.fun is not None else None
    g = np.clip(res.x[:n], 0, None)
    gs = g.sum()
    t_star = res.x[n]
    if gs < 1e-12:
        return None, None
    g = g / gs
    return g, t_star
```

```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(320, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(320, dtype=np.float64)
    return p


def _conv_fft(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_fft(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1} via water-filling bisection."""
    n = len(v)
    if n == 0:
        return v
    lo = float(v.min()) - 1.0
    hi = float(v.max())
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.maximum(v - mid, 0.0).sum()
        if s > 1.0:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    x = np.maximum(v - tau, 0.0)
    xsum = x.sum()
    if xsum < 1e-15:
        return np.full(n, 1.0 / n)
    return x / xsum


def _lp_direction(a, n):
    """
    Solve: min t
           s.t. (a * g)[k] <= t  for k = 0..2n-2
                sum(g) = 1
                g >= 0
    Returns (g_normalized, t_star) or (None, None).
    """
    A_conv = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_conv[j:j + n, j] = a

    nv = n + 1
    c = np.zeros(nv)
    c[n] = 1.0

    A_ub = np.zeros((2 * n - 1, nv))
    A_ub[:, :n] = A_conv
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, nv))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-8})
    except Exception:
        return None, None

    if res.status != 0:
        return None, None

    g = np.clip(res.x[:n], 0.0, None)
    gs = float(g.sum())
    t_star = float(res.x[n])
    if gs < 1e-12:
        return None, t_star
    g = g / gs
    return g, t_star


def _subgrad_step(a, fsize):
    """One subgradient step on max_k (a*a)[k]. Returns (new_a, k_star, peak)."""
    n = len(a)
    F = rfft(a, fsize)
    b = irfft(F * F, fsize)[:2 * n - 1]
    k_star = int(b.argmax())
    peak = float(b[k_star])

    # Gradient of b[k_star] w.r.t. a
    idx = k_star - np.arange(n)
    grad = np.zeros(n)
    valid = (idx >= 0) & (idx < n)
    grad[valid] = 2.0 * a[idx[valid]]

    return grad, k_star, peak


def _coord_descent(a, stop_time, max_rounds=50):
    """Coordinate descent polish."""
    n = len(a)
    best = a.copy()
    best_sc = _score(best)
    multipliers = (0.3, 0.5, 0.7, 0.9, 1.1, 1.3, 1.6, 2.5, 0.0)
    no_improve = 0

    for _round in range(max_rounds):
        if time.monotonic() > stop_time:
            break
        improved = False
        for i in range(n):
            if time.monotonic() > stop_time:
                break
            orig = best[i]
            best_val = orig
            best_val_sc = best_sc
            for mlt in multipliers:
                best[i] = orig * mlt if orig > 1e-13 else (0.005 if mlt > 0 else 0.0)
                s = best.sum()
                if s < 0.01:
                    best[i] = orig
                    continue
                sc = _score(best)
                if sc < best_val_sc - 1e-14:
                    best_val = best[i]
                    best_val_sc = sc
            best[i] = best_val
            if best_val_sc < best_sc - 1e-14:
                best_sc = best_val_sc
                improved = True

        # 2-coord moves
        rng = np.random.default_rng(42 + _round)
        for _ in range(40):
            if time.monotonic() > stop_time:
                break
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            for mi in (0.5, 1.5):
                for mj in (0.5, 1.5):
                    best[i] = oi * mi if oi > 1e-13 else 0.005
                    best[j] = oj * mj if oj > 1e-13 else 0.005
                    if best.sum() < 0.01:
                        best[i], best[j] = oi, oj
                        continue
                    sc = _score(best)
                    if sc < best_sc - 1e-14:
                        best_sc = sc
                        improved = True
            best[i], best[j] = oi, oj

        if not improved:
            no_improve += 1
            if no_improve >= 5:
                break
        else:
            no_improve = 0

    return best, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(30.0, float(budget_s))
    deadline = t0 + budget - 15.0

    parent = _get_parent()
    n = len(parent)

    # Initialize from parent
    a = np.clip(parent.astype(np.float64), 0.0, None)
    s = a.sum()
    if s < 0.01:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_sc = _score(a)

    if not math.isfinite(best_sc):
        a = np.ones(n) / n
        best_a = a.copy()
        best_sc = _score(a)

    rng = np.random.default_rng(seed)
    fsize = next_fast_len(2 * n - 1)

    # === PHASE 1: LP-based iteration ===
    lp_end = t0 + budget * 0.65
    t = 0.25
    t_min = 0.02
    t_max = 0.50
    stall = 0
    max_stall = 150

    while time.monotonic() < lp_end:
        if stall > max_stall:
            # Restart with perturbation
            a = _project_simplex(best_a + rng.normal(0, 0.03, n))
            t = 0.2
            stall = 0
            if time.monotonic() > lp_end:
                break

        # Compute current peak
        F = rfft(a, fsize)
        b = irfft(F * F, fsize)[:2 * n - 1]
        M_current = float(b.max())

        # Solve LP
        g, t_star = _lp_direction(a, n)

        if g is None:
            stall += 5
            a = _project_simplex(a + rng.normal(0, 0.02, n))
            continue

        # Check if direction is meaningful
        dist = float(np.linalg.norm(g - a))
        if dist < 1e-7:
            stall += 3
            if stall > max_stall // 2:
                a = _project_simplex(a + rng.normal(0, 0.04, n))
                t = 0.15
                stall = 0
            continue

        # Line search: try several alpha values
        improved = False
        for trial in range(15):
            alpha = t if trial == 0 else t * (0.65 ** trial)
            if alpha < 0.005:
                break
            a_new = (1.0 - alpha) * a + alpha * g
            a_new = np.maximum(a_new, 0.0)
            sn = a_new.sum()
            if sn < 0.01:
                continue
            a_new = a_new / sn
            sc = _score(a_new)
            if sc < best_sc - 1e-13:
                a = a_new
                best_a = a.copy()
                best_sc = sc
                improved = True
                t = min(t_max, t * 1.06)
                stall = 0
                break
            else:
                t = max(t_min, t * 0.55)

        if not improved:
            stall += 1
            if stall >= 80 and stall < max_stall:
                # Try smaller step in opposite direction or random
                noise_scale = 0.01 * min(1.0, stall / 40.0)
                a = _project_simplex(a + rng.normal(0, noise_scale, n))
                t = 0.15
                stall = 0

    # === PHASE 2: Subgradient refinement ===
    sg_end = t0 + budget * 0.85
    if time.monotonic() < sg_end:
        a = best_a.copy()
        sc = best_sc
        lr = 0.05
        no_improve = 0

        while time.monotonic() < sg_end and no_improve < 300:
            grad, k_star, peak = _subgrad_step(a, fsize)
            if peak < 1e-15:
                break

            # Adaptive learning rate based on peak
            lr_eff = lr / max(peak, 1e-8) * 0.01
            lr_eff = min(lr_eff, 0.5)

            a_new = a - lr_eff * grad
            a_new = _project_simplex(a_new)
            sc_new = _score(a_new)

            if sc_new < sc - 1e-13:
                a = a_new
                sc = sc_new
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
                no_improve = 0
                lr = min(0.2, lr * 1.01)
            else:
                lr = max(0.005, lr * 0.85)
                no_improve += 1

    # === PHASE 3: Coordinate descent polish ===
    polish_end = deadline - 5.0
    if time.monotonic() < polish_end - 5.0:
        try:
            polished, pol_sc = _coord_descent(best_a, polish_end, max_rounds=80)
            if pol_sc < best_sc:
                best_a = polished
                best_sc = pol_sc
        except Exception:
            pass

    # Final validation with exact evaluator
    try:
        final_list = [float(x) for x in best_a]
        final_sc = float(evaluate_sequence(final_list))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        # Ultimate fallback: uniform
        return [1.0 / n] * n

    return [float(x) for x in best_a]
```
