- a_1 = (1-t) * a_0 + t * g_1 (normalized)

The question is whether this converges to the optimum. Intuitively, yes - it's like a version of the method of conditional gradients (Frank-Wolfe) adapted for this problem.

For the step size, I'll use:
- Start with t = 0.5
- If the score increases, halve t
- Clip t to [0.01, 1.0]

- Compute the subgradient at the argmax of conv(a,a)
- Take a step in the negative subgradient direction
- Project to simplex (sum = 1, a ≥ 0)

This is essentially what the parent does, but I'll combine it more carefully with the LP.

One more thought: the evaluator clips to [0, 1000]. So coefficients can be up to 1000. This means very "spiky" sequences are allowed. But spiky sequences have high peak autocorrelation, so they're bad. The optimal should have relatively smooth coefficients.

def _fft_conv(a, b=None):
    """Compute full linear convolution using FFT."""
    if b is None:
        b = a
    n = len(a) + len(b) - 1
    fsize = next_fast_len(n)
    A = rfft(a, fsize)
    B = rfft(b, fsize)
    c = irfft(A * B, fsize)[:n]
    return c

def _score(a, n):
    """Compute the score (using FFT for speed)."""
    c = _fft_conv(a, a)
    max_c = float(c.max())
    s = float(a.sum())
    if s < 1e-15:
        return float('inf')
    return 2.0 * n * max_c / (s * s)

def _normalize(a):
    """Normalize to sum = 1, nonnegative."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def _build_conv_matrix(a, n):
    """Build matrix A such that (A @ g)_k = conv(a, g)_k."""
    # A is (2n-1) x n
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = a[idx]
    return A

def _build_conv_matrix_fast(a, n):
    """Build matrix A such that (A @ g)_k = conv(a, g)_k using vectorization."""
    A = np.zeros((2*n-1, n))
    # For row k, A[k,j] = a[k-j] if 0<=k-j<n
    k_idx = np.arange(2*n-1)[:, None]  # (2n-1, 1)
    j_idx = np.arange(n)[None, :]  # (1, n)
    diff = k_idx - j_idx  # (2n-1, n)
    valid = (diff >= 0) & (diff < n)
    A[valid] = a[diff[valid]]
    return A

def _solve_lp_direction(a, M, n, solver_list=None):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    if solver_list is None:
        solver_list = [cp.GLOP, cp.CBC, cp.CPLEX, cp.GLPK]
    
    A = _build_conv_matrix_fast(a, n)
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M, cp.sum(g) <= 1e10]  # M bounds, also cap sum
    # Actually, we want max sum(g) s.t. A@g <= M, g >= 0
    # The sum is automatically bounded since A@g <= M implies each row gives a bound
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    
    for solver in solver_list:
        try:
            prob.solve(solver=solver, verbose=False, threads=1)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                gv = g.value
                if gv is not None and np.isfinite(gv).all():
                    gv = np.maximum(gv, 0)
                    s = gv.sum()
                    if s > 1e-12:
                        return gv / s  # normalized to sum = 1
        except Exception:
            continue
    return None

def _subgradient_direction(a, n):
    """Compute subgradient of max(conv(a,a)) w.r.t. a."""
    c = _fft_conv(a, a)
    k_star = int(np.argmax(c))
    # d/d a_i [conv(a,a)_{k_star}] = a_{k_star - i} (with bounds)
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = a[j]
    # Since conv(a,a) is symmetric, the actual gradient of the max is 2*grad if we consider
    # both a and a as variables. But since we're differentiating w.r.t. a only:
    # d/da_i sum_j a_j * a_{k-i} = a_{k-i} + a_{k-i} (from both terms) 
    # Actually no: conv(a,a)_k = sum_{i+j=k} a_i a_j
    # d/d a_i of this = a_{k-i} + (a_{k-i} if k-i != i, else a_{k-i})
    # = 2 * a_{k-i} if i != k-i, a_{k-i} if i = k-i
    # But for subgradient purposes, 2*grad is a valid subgradient of max.
    return 2.0 * grad

        # Try LP direction
        g = _solve_lp_direction(a, M, n)

            # Check if g provides a useful direction
            diff = g - a
            if np.linalg.norm(diff) > 1e-6:

                best_t = 0.0
                best_s = current_score

                _subgradient_step(a, n)

def _subgradient_step(a, n):

    grad = np.zeros(n)

                best_a = cand.copy()
                no_improve = 0
                sigma = max(0.003, sigma * 0.97)
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        # Annealing schedule
        temp *= 0.9985
        if no_improve > 150:
            sigma = min(0.35, sigma * 1.4)
            temp = max(temp, 0.008)
            no_improve = 0
    
    return best_a, best_score

def _try_different_length(L, deadline, rng, best_score):
    i = np.arange(L, dtype=np.float64)
    a = 1.0 + 0.3 * np.sin(2 * np.pi * i / L * 3) + 0.2 * np.cos(2 * np.pi * i / L * 7)
    a = np.maximum(a, 0.05)
    a = _normalize(a)
    
    sub_dl = min(time.monotonic() + 40, deadline - 5)
    if time.monotonic() > sub_dl:
        return None, best_score
    
    res_a, res_sc = _optimize_lp(a, L, sub_dl, rng, max_iters=1000)
    sc = 2.0 * L * float(_fft_conv(res_a, res_a).max()) / float(res_a.sum())**2
    
    if sc < best_score - 1e-6:
        seq = [float(x) for x in res_a]
        exact = float(evaluate_sequence(seq))
        if np.isfinite(exact) and exact < best_score:
            return res_a, exact
    
    return None, best_score

def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + budget_s - 12.0
    
    rng = np.random.default_rng(seed)
    parent = _get_parent()
    n = len(parent)
    parent = np.clip(parent.astype(np.float64), 0, 1000)
    
    if parent.sum() < 0.01:
        parent = np.ones(192) / 192.0
        n = 192
    
    try:
        best_score = float(evaluate_sequence([float(x) for x in parent]))
    except Exception:
        best_score = float('inf')
    best_seq = parent.copy()
    
    if not np.isfinite(best_score):
        best_seq = np.ones(32) / 32.0
        best_score = 2.0
        n = 32
    
    # Phase 1: LP optimization from parent (40% budget)
    p1_end = t0 + 0.40 * budget_s
    if time.monotonic() < p1_end - 5:
        a_init = _normalize(best_seq.copy())
        try:
            res_a, res_sc = _optimize_lp(a_init, n, p1_end - 5, rng, max_iters=10000)
            seq_list = [float(x) for x in res_a]
            exact = float(evaluate_sequence(seq_list))
            if np.isfinite(exact) and exact < best_score:
                best_score = exact
                best_seq = res_a.copy()
        except Exception:
            pass
    
    # Phase 2: Local search / annealing (30% budget)
    p2_end = t0 + 0.70 * budget_s
    if time.monotonic() < p2_end - 5:
        a_init = _normalize(best_seq.copy())
        try:
            res_a, res_sc = _local_search(a_init, n, p2_end - 5, rng)
            seq_list = [float(x) for x in res_a]
            exact = float(evaluate_sequence(seq_list))
            if np.isfinite(exact) and exact < best_score:
                best_score = exact
                best_seq = res_a.copy()
        except Exception:
            pass
    
    # Phase 3: Second LP pass on improved solution (20% budget)
    p3_end = t0 + 0.90 * budget_s
    if time.monotonic() < p3_end - 5:
        a_init = _normalize(best_seq.copy())
        try:
            res_a, res_sc = _optimize_lp(a_init, n, p3_end - 5, np.random.default_rng(seed + 777), max_iters=5000)
            seq_list = [float(x) for x in res_a]
            exact = float(evaluate_sequence(seq_list))
            if np.isfinite(exact) and exact < best_score:
                best_score = exact
                best_seq = res_a.copy()
        except Exception:
            pass
    
    # Phase 4: Final polish with tiny perturbations
    p4_end = deadline
    if time.monotonic() < p4_end - 5:
        a_init = _normalize(best_seq.copy())
        try:
            res_a, res_sc = _local_search(a_init, n, p4_end - 3, np.random.default_rng(seed + 4242))
            seq_list = [float(x) for x in res_a]
            exact = float(evaluate_sequence(seq_list))
            if np.isfinite(exact) and exact < best_score:
                best_score = exact
                best_seq = res_a.copy()
        except Exception:
            pass
    
    # Final validation
    try:
        final_score = float(evaluate_sequence([float(x) for x in best_seq]))
        if not np.isfinite(final_score):
            raise ValueError("non-finite")
    except Exception:
        fb = np.ones(32) / 32.0
        return [float(x) for x in fb]
    
    return [float(x) for x in best_seq]
