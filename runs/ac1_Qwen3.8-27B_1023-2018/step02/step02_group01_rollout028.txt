import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, next_fast_len

def _convolve_square(a):
    """Full linear convolution of a with a."""
    n = a.size
    if n == 0:
        return np.array([0.0])
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)  # numerical safety
    return c

def _score(a):
    """Compute the exact score."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    n = a.size
    b = _convolve_square(a)
    return 2.0 * n * float(b.max()) / (s * s)

def _build_A(a):
    """Build the (2n-1) x n constraint matrix for LP."""
    n = a.size
    m = 2 * n - 1
    k_idx = np.arange(m)[:, None].astype(np.int64)
    i_idx = np.arange(n)[None, :].astype(np.int64)
    j = k_idx - i_idx
    mask = (j >= 0) & (j < n)
    safe_j = np.clip(j, 0, n - 1)
    A = np.where(mask, a[safe_j], 0.0)
    return A

def _lp_solve(a, M, timeout=3.0):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = a.size
    A = _build_A(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), 
                  bounds=[(0, None)]*n, method='highs',
                  options={'time_limit': timeout, 'presolve': True})
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    if g.sum() <= 0:
        return None
    return g

def _soft_max_gradient(a, b, T):
    """Compute soft-max weighted gradient of max(b)."""
    # w[k] = exp(T*b[k]) / sum(exp(T*b))
    Tb = T * (b - b.max())  # for numerical stability
    w = np.exp(Tb)
    w /= w.sum()
    
    n = a.size
    grad = np.zeros(n)
    # grad[i] = 2 * sum_j a[j] * w[i+j] (for valid i+j in [0, 2n-2])
    for i in range(n):
        j_end = min(n, 2*n-1-i)
        if j_end > 0:
            grad[i] = 2.0 * np.dot(a[:j_end], w[i:i+j_end])
    return grad

def _gradient_descent(a0, stop_time, T_init=10.0, T_final=100.0, 
                      lr_init=0.05, max_iters=5000):
    """Soft-max gradient descent."""
    n = a0.size
    a = np.clip(a0.astype(np.float64), 0, 1000).copy()
    s = a.sum()
    if s <= 0:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_bmax = float(_convolve_square(a).max())
    
    lr = lr_init
    T = T_init
    
    for it in range(max_iters):
        if time.monotonic() > stop_time - 1:
            break
        
        b = _convolve_square(a)
        bmax = float(b.max())
        
        if bmax < best_bmax * (1 - 1e-12):
            best_bmax = bmax
            best_a = a.copy()
        
        # Anneal temperature and learning rate
        frac = min(1.0, it / max(1, max_iters * 0.5))
        T_cur = T_init + frac * (T_final - T_init)
        lr_cur = lr_init * (0.8 ** (it // 100))  # geometric decay
        
        # Compute soft-max gradient
        grad = _soft_max_gradient(a, b, T_cur)
        gnorm = np.sqrt(np.dot(grad, grad))
        if gnorm < 1e-15:
            break
        
        # Gradient step
        a_new = a - lr_cur * grad / gnorm  # normalized gradient direction
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-10:
            break
        a_new = a_new / sn
        
        # Check improvement
        b_new = _convolve_square(a_new)
        bmax_new = float(b_new.max())
        
        if bmax_new < bmax * (1 - 1e-13):
            a = a_new
        else:
            # Backtrack
            for factor in [0.5, 0.25, 0.1]:
                a_test = a - factor * lr_cur * grad / gnorm
                a_test = np.maximum(a_test, 0.0)
                st = a_test.sum()
                if st < 1e-10:

                a_test = a_test / st
                b_test = _convolve_square(a_test)
                if b_test.max() < bmax * (1 - 1e-14):
                    a = a_test

    return best_a

def _lp_iterate(a0, stop_time, max_iters=300):
    """LP best-response iteration."""
    n = a0.size
    a = np.clip(a0.astype(np.float64), 0, 1000).copy()
    s = a.sum()
    if s <= 0:
        a = np.ones(n) / n
    else:
        a = a / s

    best_bmax = float(_convolve_square(a).max())
    stuck = 0
    
    for it in range(max_iters):
        if time.monotonic() > stop_time - 2:

        b = _convolve_square(a)
        M = float(b.max())

            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        g = _lp_solve(a, M * 1.0001, timeout=2.0)
        if g is None:

        g = g / g.sum()

                improved = True
                break

Actually, I realize I should also think about whether the "LP best-response" is really helping. Let me think about what it does geometrically.

Given a with max(a*a) = M, the LP finds g with max(a*g) <= M that maximizes sum(g). The LP solution is an extreme point where many (a*g)[k] = M constraints are active.

The update a -> (1-t)a + t*g/Sg moves a toward g. The intuition is: g is "optimal" in the sense that it makes a*g as flat as possible (at level M) while maximizing its total mass. By moving a toward g, we're trying to make a*a more flat.

But there's a subtlety: the LP constraint is (a*g)[k] <= M, not (a*a)[k] <= M. And we're computing M from a*a, not a*g. So the LP is solving a different (but related) problem.

A more direct approach: solve the LP with the constraint (a*a)[k] <= M... but a*a is quadratic in a, so this is not an LP. The LP is only valid when one factor is fixed.

So the LP approach is: fix a, find the best g. This gives a direction for a. It's a heuristic, not an exact method. But it works well in practice.

An alternative: use the dual of the LP. The dual is:
min sum_k M*y_k s.t. sum_k A[k,i]*y_k >= 1 for all i, y_k >= 0

where A[k,i] = a[k-i] (if valid). The dual gives "weights" y_k on the constraints. At the optimum, the dual objective equals the primal objective.

The KKT conditions at the optimum:
- Primal feasibility: A*g <= M, g >= 0
- Dual feasibility: A^T*y >= 1, y >= 0
- Complementary slackness: g_i > 0 → (A^T*y)_i = 1; y_k > 0 → (A*g)_k = M

The dual variable y_k can be interpreted as the "importance" of constraint k. At the optimum, the active constraints (where (A*g)_k = M) have y_k > 0.

This gives us the "active set" of lags. We can use this to focus our gradient optimization on the active lags.

def _conv2(a):
    n = a.size

        return np.array([0.0])
    fs = next_fast_len(2 * n - 1)
    F = rfft(a, fs)
    return np.maximum(irfft(F * F, fs)[:2 * n - 1], 0.0)


def _sc(a):
