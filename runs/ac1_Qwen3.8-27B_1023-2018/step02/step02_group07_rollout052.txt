import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len


def _softmax(x):
    x = np.asarray(x, dtype=np.float64).ravel()
    if x.size == 0:
        return np.array([1.0])
    x = x - x.max()
    e = np.exp(np.clip(x, -80.0, 80.0))
    s = e.sum()
    if not np.isfinite(s) or s <= 0.0:
        return np.full(x.size, 1.0 / x.size, dtype=np.float64)
    return e / s


def _full_conv(a, fs):
    F = rfft(a, fs)
    return irfft(F * F, fs)[: 2 * len(a) - 1]


def _xcorr_weights(a, p, fs):
    """Compute g[i] = sum_k p[k] * a[k-i] for valid k-i."""
    n = len(a)
    ar = a[::-1]
    c = irfft(rfft(ar, fs) * rfft(p, fs), fs)
    return c[n - 1 : 2 * n - 1]


def _score_from_b(b, n):
    """Score = 2n * max(b) assuming sum(a) = 1."""
    m = b.max()
    return 2.0 * n * max(m, 0.0)


def _active_set_step(a, top_k=5, n_max=2048):
    """
    One active-set refinement step.
    Solves the KKT linear system for the top-k peaks.
    Returns the refined a (or original if it fails).
    """
    n = len(a)
    if n > n_max or n < 2:
        return a.copy()
    
    b = _full_conv(a, next_fast_len(2 * n - 1))
    # Find top-k peaks
    k = min(top_k, len(b))
    idx = np.argsort(b)[::-1][:k]
    # Weights proportional to peak values
    bv = b[idx]
    w = bv / max(bv.sum(), 1e-30)
    
    # Build M matrix
    M = np.zeros((n, n), dtype=np.float64)
    for mi in range(k):
        ki = idx[mi]
        wi = w[mi]
        # M[i, ki - i] += wi for valid ki - i
        for i in range(n):
            j = ki - i
            if 0 <= j < n:
                M[i, j] += wi
    
    # Augmented system: M @ a - lambda * 1 = 0, sum(a) = 1
    A = np.zeros((n + 1, n + 1), dtype=np.float64)
    A[:n, :n] = M
    A[:n, n] = -1.0
    A[n, :n] = 1.0
    # A[n, n] = 0
    rhs = np.zeros(n + 1)
    rhs[n] = 1.0
    
    try:
        sol = np.linalg.solve(A, rhs)
        a_new = sol[:n]
    except np.linalg.LinAlgError:
        try:
            sol = np.linalg.lstsq(A, rhs, rcond=None)[0]
            a_new = sol[:n]
        except:
            return a.copy()
    
    if not np.isfinite(a_new).all():
        return a.copy()
    
    # Check non-negativity
    if a_new.min() >= -1e-10:
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-15:
            a_new /= s
            return a_new
    else:
        # Clip negative values and renormalize
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-15:
            a_new /= s
            return a_new
        else:
            return a.copy()

    tau_frac = 0.5  # temperature fraction of bmax
    eta = 0.9  # momentum
    rho = 0.999  # secondary momentum
    
    stuck = 0
    total_iter = 0
    
    while time.monotonic() < end_time:
        # Softmax
        a = _softmax(np.clip(z, -30, 30))
        b = _full_conv(a, fs_c)
        bmax = b.max()
        
        if not np.isfinite(bmax) or bmax <= 0:
            break
        
        cur_score = _score_from_b(b, n)
        
        # Soft-max weights
        tau = max(1e-10, tau_frac * bmax)
        bt = np.clip(b / tau, -80, 80)
        e = np.exp(bt)
        s = e.sum()
        if not np.isfinite(s) or s <= 0:
            p = np.zeros_like(b)
            p[np.argmax(b)] = 1.0
        else:
            p = e / s
        
        # Gradient: g = xcorr(p, a)
        g = 2.0 * _xcorr_weights(a, p, fs2)
        
        # Variance regularization
        if w_reg > 0:
            q = b - b.mean()
            vg = 2.0 * _xcorr_weights(a, q, fs2)
            # Scale to be comparable
            vg = vg * (w_reg * bmax)
            g = g + vg
        
        # Project to simplex tangent (softmax parameterization)
        ag = np.dot(a, g)
        grad = a * (g - ag)
        
        # Normalize gradient
        gn = np.abs(grad).max()
        if not np.isfinite(gn) or gn < 1e-20:
            stuck += 1
            if stuck > 50:
                # Perturb
                z = best_z + rng.normal(0, 0.5, n)
                z = z - z.mean()
                m_adam = np.zeros(n)
                v_adam = np.zeros(n)
                t_step = 0
                lr = 0.5
                tau_frac = 0.4
                stuck = 0
            continue
        
        grad = grad / gn
        
        # Adam update
        t_step += 1
        m_adam = eta * m_adam + (1 - eta) * grad
        v_adam = rho * v_adam + (1 - rho) * (grad * grad)
        m_hat = m_adam / (1 - eta ** t_step)
        v_hat = v_adam / (1 - rho ** t_step)
        update = m_hat / (np.sqrt(v_hat) + 1e-8)
        
        # Scale update
        um = np.abs(update).max()
        if not np.isfinite(um) or um <= 0:
            continue
        update = update / um
        
        z = z - lr * update
        z = z - z.mean()
        z = np.clip(z, -30, 30)
        z[~np.isfinite(z)] = 0.0
        
        # Evaluate
        a2 = _softmax(np.clip(z, -30, 30))
        b2 = _full_conv(a2, fs_c)
        score2 = _score_from_b(b2, n)
        
        if np.isfinite(score2) and score2 < best_score - 1e-14:
            best_score = score2
            best_z = z.copy()
            stuck = 0
        else:
            stuck += 1
        
        total_iter += 1
        
        # Adaptive schedule
        if total_iter % 20 == 0:
            tau_frac = max(0.02, tau_frac * 0.97)
        if stuck > 30:
            lr = max(0.05, lr * 0.85)
        if stuck > 100:
            # Periodic active-set refinement
            if t_step > 10:
                a_asyn = _active_set_step(a2, top_k=5)
                z_asyn = np.log(np.maximum(a_asyn, 1e-12))
                z_asyn = z_asyn - z_asyn.mean()
                a_check = _softmax(z_asyn)
                b_check = _full_conv(a_check, fs_c)
                s_check = _score_from_b(b_check, n)
                if np.isfinite(s_check) and s_check < best_score:
                    best_score = s_check
                    best_z = z_asyn.copy()
            
            z = best_z + rng.normal(0, 0.5, n)
            z = z - z.mean()
            m_adam = np.zeros(n)
            v_adam = np.zeros(n)
            t_step = 0
            lr = 0.5
            tau_frac = 0.4
            stuck = 0
    
    # Return best
    a_best = _softmax(np.clip(best_z, -30, 30))
    b_best = _full_conv(a_best, fs_c)
    s_best = _score_from_b(b_best, n)
    return a_best, s_best, total_iter


def _local_search(a, n, end_time, seed):
    """
    Local search / perturbation-based polish.
    """
    fs = next_fast_len(2 * n - 1)
    rng = np.random.default_rng(seed)
    
    a = np.clip(a, 1e-12, None).astype(np.float64)
    a = a / a.sum()
    best = a.copy()
    b_best = _full_conv(best, fs)
    best_score = _score_from_b(b_best, n)
    
    sigma = 0.15
    stuck = 0
    
    while time.monotonic() < end_time:
        z = np.log(np.maximum(a, 1e-15))
        r = rng.random()
        
        if r < 0.25:
            # Global perturbation
            zc = z + rng.normal(0, sigma, n)
        elif r < 0.45:
            # Local perturbation
            i = rng.integers(0, n)
            w = max(1, min(n, rng.integers(1, max(2, n // 20))))
            i2 = min(n, i + w)
            zc = z.copy()
            zc[i:i2] += rng.normal(0, sigma * 0.5, i2 - i)
        elif r < 0.60:
            # Scale perturbation
            zv = z - z.mean()
            zc = zv * rng.uniform(0.5, 2.0)
        elif r < 0.75:
            # Multiplicative perturbation on a
            a2 = a.copy()
            i = rng.integers(0, n)
            w = max(1, min(n, rng.integers(1, max(2, n // 25))))
            i2 = min(n, i + w)
            a2[i:i2] *= np.exp(rng.normal(0, 0.3))
            zc = np.log(np.maximum(a2, 1e-15))
        elif r < 0.90:
            # Sine perturbation
            f = rng.uniform(0.5, 5.0)
            ph = rng.uniform(0, 2 * math.pi)
            zc = z + sigma * np.sin(2 * math.pi * f * np.arange(n) / n + ph)
        else:
            # Active set step
            zc = np.log(np.maximum(_active_set_step(a, top_k=5), 1e-12))
        
        zc = zc - zc.mean()
        a2 = _softmax(np.clip(zc, -30, 30))
        b2 = _full_conv(a2, fs)
        s2 = _score_from_b(b2, n)
        
        if np.isfinite(s2) and s2 < best_score - 1e-14:
            best = a2.copy()
            best_score = s2
            a = a2.copy()
            stuck = 0
            sigma = max(0.03, sigma * 0.95)
        else:
            stuck += 1
            sigma = min(0.8, sigma * 1.03)
            if stuck > 80:
                stuck = 0
                sigma = 0.3
    
    b_best = _full_conv(best, fs)
    s_best = _score_from_b(b_best, n)
    return best, s_best
