    def lp_optimize(a, max_iters=3000, stop_time=None):
        n = len(a)
        a = np.array(a, dtype=np.float64).copy()
        a = np.clip(a, 0, None)
        s = float(a.sum())
        if s < 1e-15:
            return a
        a = a / s
        local = score_fn(a)
        update_best(a)

        c_vec = -np.ones(n, dtype=np.float64)
        bounds = [(0.0, None)] * n
        stag = 0
        no_improve_count = 0

        for it in range(max_iters):
            if stop_time is not None and time.monotonic() >= stop_time:
                break

            b = np.convolve(a, a)
            M = float(b.max())
            if not math.isfinite(M) or M <= 0:
                break

            try:
                A = build_conv_matrix(a)
            except Exception:
                break

            improved = False
            margins = (1.0, 0.999, 0.995, 0.99, 0.98, 0.97, 0.95)

            for mg in margins:
                if stop_time is not None and time.monotonic() >= stop_time:
                    break
                b_ub = np.full(2 * n - 1, M * mg)
                try:
                    res = linprog(c_vec, A_ub=A, b_ub=b_ub,
                                  bounds=bounds, method='highs',
                                  options={'presolve': True, 'time_limit': 10})
                except Exception:
                    continue

                if res is None or not hasattr(res, 'x') or res.x is None:
                    continue
                if res.status not in (0, 2):
                    if res.status == 1:  # infeasible
                        continue
                    # Try to use solution if available
                    if res.x is None:
                        continue

                g = np.clip(np.asarray(res.x), 0, None)
                sg = float(g.sum())
                if sg < 1e-15:
                    continue
                g = g / sg

                # Check if g is essentially the same as a (no progress)
                if np.linalg.norm(g - a) < 1e-10:
                    continue

                # Line search
                best_t_sc = local
                best_t_cand = None
                for t in (0.5, 0.3, 0.7, 1.0, 0.2, 0.8, 0.4, 0.6, 0.9, 0.1):
                    cand = (1.0 - t) * a + t * g
                    cand = np.clip(cand, 0, None)
                    cs = float(cand.sum())
                    if cs < 1e-15:
                        continue
                    cand = cand / cs
                    v = score_fn(cand)
                    if v < best_t_sc:
                        best_t_sc = v
                        best_t_cand = cand

                if best_t_cand is not None and best_t_sc < local * (1 - 1e-10):
                    a = best_t_cand
                    local = best_t_sc
                    stag = 0
                    improved = True
                    update_best(a)
                    break

            if not improved:
                stag += 1
                no_improve_count += 1
                if stag > 30:
                    # Perturbation to escape local optimum
                    pert_strength = min(0.3, 0.05 + 0.02 * (stag // 30))
                    a_p = a * np.exp(pert_strength * rng.standard_normal(n))
                    a_p = np.clip(a_p, 0, None)
                    if a_p.sum() > 1e-15:
                        a_p = a_p / a_p.sum()
                        if score_fn(a_p) < local:
                            a = a_p
                            local = score_fn(a)
                            stag = 0
                            no_improve_count = 0
                            update_best(a)
                if no_improve_count > 150:
                    break

        return a

    def grad_optimize(a, max_iters=200000, stop_time=None, tau=40.0):
        n = len(a)
        a = np.array(a, dtype=np.float64).copy()
        a = np.clip(a, 0, None)
        s = float(a.sum())
        if s < 1e-15:
            return a
        a = a / s
        local = score_fn(a)
        update_best(a)
        stale = 0

        for it in range(max_iters):
            if stop_time is not None and time.monotonic() >= stop_time:
                break

            b = np.convolve(a, a)
            M = float(b.max())
            if not math.isfinite(M):
                break

            # Soft-max gradient
            shifted = b - M
            exp_b = np.exp(tau * shifted)
            p_sum = exp_b.sum()
            if p_sum < 1e-30:
                break
            p = exp_b / p_sum

            # Compute gradient: g_j = 2 * sum_k p_k * a[k-j]
            g = np.zeros(n, dtype=np.float64)
            # Only process significant p_k for efficiency
            significant = p > 1e-8
            for k in np.where(significant)[0]:
                pk = p[k]
                lo = max(0, int(k) - n + 1)
                hi = min(n - 1, int(k))
                if lo <= hi:
                    js = np.arange(lo, hi + 1)
                    g[js] += 2.0 * pk * a[int(k) - js]

            gn = float(np.linalg.norm(g))
            if gn < 1e-15:
                break
            g = g / gn

            # Line search
            best_a_new = None
            best_sc = local
            for alpha in (0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.001, 0.002, 0.04, 0.03):
                cand = a - alpha * g
                cand = np.clip(cand, 0, None)
                cs = float(cand.sum())
                if cs < 1e-15:
                    continue
                cand = cand / cs
                v = score_fn(cand)
                if v < best_sc:
                    best_sc = v
                    best_a_new = cand

            if best_a_new is not None and best_sc < local * (1 - 1e-12):
                a = best_a_new
                local = best_sc
                stale = 0
                if it % 200 == 0:
                    update_best(a)
            else:
                stale += 1
                if stale % 100 == 0:
                    # Small perturbation
                    a_p = a + (0.01 + 0.005 * min(stale // 100, 5)) * rng.standard_normal(n)
                    a_p = np.clip(a_p, 0, None)
                    if a_p.sum() > 1e-15:
                        a_p = a_p / a_p.sum()
                        if score_fn(a_p) < local:
                            a = a_p
                            local = score_fn(a)
                            stale = 0
                            update_best(a)
                if stale > 2000:
                    break

        update_best(a)
        return a

    def subgrad_optimize(a, max_iters=200000, stop_time=None):
        """Subgradient method using active set."""
        n = len(a)
        a = np.array(a, dtype=np.float64).copy()
        a = np.clip(a, 0, None)
        s = float(a.sum())
        if s < 1e-15:
            return a
        a = a / s
        local = score_fn(a)
        update_best(a)
        stale = 0

        for it in range(max_iters):
            if stop_time is not None and time.monotonic() >= stop_time:
                break

            b = np.convolve(a, a)
            M = float(b.max())
            if not math.isfinite(M):
                break

            # Active set: lags within 2% of max
            thr = M * 0.98
            active = np.where(b >= thr)[0]
            if len(active) == 0:
                active = np.array([int(np.argmax(b))])

            # Averaged subgradient
            g = np.zeros(n, dtype=np.float64)
            for k in active:
                lo = max(0, int(k) - n + 1)
                hi = min(n - 1, int(k))
                if lo <= hi:
                    js = np.arange(lo, hi + 1)
                    g[js] += 2.0 * a[int(k) - js]
            g /= len(active)

            gn = float(np.linalg.norm(g))
            if gn < 1e-15:
                break
            g = g / gn

            best_a_new = None
            best_sc = local
            for alpha in (0.005, 0.01, 0.02, 0.05, 0.1, 0.02, 0.03, 0.04):
                cand = a - alpha * g
                cand = np.clip(cand, 0, None)
                cs = float(cand.sum())
                if cs < 1e-15:
                    continue
                cand = cand / cs
                v = score_fn(cand)
                if v < best_sc:
                    best_sc = v
                    best_a_new = cand

            if best_a_new is not None and best_sc < local * (1 - 1e-12):
                a = best_a_new
                local = best_sc
                stale = 0
                if it % 200 == 0:
                    update_best(a)
            else:
                stale += 1
                if stale % 100 == 0:
                    a_p = a + 0.015 * rng.standard_normal(n)
                    a_p = np.clip(a_p, 0, None)
                    if a_p.sum() > 1e-15:
                        a_p = a_p / a_p.sum()
                        if score_fn(a_p) < local:
                            a = a_p
                            local = score_fn(a)
                            stale = 0
                            update_best(a)
                if stale > 3000:
                    break

        update_best(a)
        return a

    def interp_len(src, n_new):
        if src.size == n_new:
            return src.copy()
        ox = np.linspace(0.0, 1.0, src.size)
        nx = np.linspace(0.0, 1.0, n_new)
        out = np.interp(nx, ox, src)
        out = np.clip(out, 0, None)
        s = float(out.sum())
        if s > 1e-15:
            out = out / s
        else:
            out = np.ones(n_new, dtype=np.float64) / n_new
        return out

    # === MAIN OPTIMIZATION ===
    p = parent if parent is not None else best_arr
    n_p = p.size
    p_norm = p / p.sum()

    # Phase 1: LP optimization at parent length (primary effort)
    t1_stop = t0 + budget * 0.42
    a_work = p_norm.copy()
    if linprog is not None and sps is not None and time.monotonic() < t1_stop - 5:
        a_work = lp_optimize(a_work, max_iters=5000, stop_time=t1_stop)
    update_best(a_work)

    # Phase 2: Gradient polish (soft-max)
    t2_stop = t0 + budget * 0.62
    if time.monotonic() < t2_stop - 5:
        a_work = grad_optimize(a_work, max_iters=300000, stop_time=t2_stop, tau=40.0)
    update_best(a_work)

    # Phase 3: Another LP pass (refined)
    t3_stop = t0 + budget * 0.75
    if linprog is not None and sps is not None and time.monotonic() < t3_stop - 5:
        a_work = lp_optimize(a_work, max_iters=3000, stop_time=t3_stop)
    update_best(a_work)

    # Phase 4: Subgradient polish
    t4_stop = t0 + budget * 0.85
    if time.monotonic() < t4_stop - 5:
        a_work = subgrad_optimize(a_work, max_iters=200000, stop_time=t4_stop)
    update_best(a_work)

    # Phase 5: Try an alternative length
    if time.monotonic() < deadline - 50:
        n_alt = min(400, max(200, n_p + 32))
        if n_alt != n_p:
            a_alt = interp_len(p_norm, n_alt)
            t5_stop = min(deadline - 30, time.monotonic() + 60)
            if linprog is not None and sps is not None:
                a_alt = lp_optimize(a_alt, max_iters=200, stop_time=t5_stop)
            update_best(a_alt)
            if time.monotonic() < deadline - 20:
                a_alt = grad_optimize(a_alt, max_iters=50000,
                                      stop_time=deadline - 10, tau=30.0)
                update_best(a_alt)

    # Phase 6: Try another alternative length (shorter)
    if time.monotonic() < deadline - 35 and n_p > 200:
        n_alt2 = max(160, n_p - 64)
        if n_alt2 != n_p and n_alt2 != 400:
            a_alt2 = interp_len(p_norm, n_alt2)
            t6_stop = min(deadline - 15, time.monotonic() + 30)
            if linprog is not None and sps is not None:
                a_alt2 = lp_optimize(a_alt2, max_iters=100, stop_time=t6_stop)
            update_best(a_alt2)
            if time.monotonic() < deadline - 10:
                a_alt2 = grad_optimize(a_alt2, max_iters=20000,
                                       stop_time=deadline - 5, tau=30.0)
                update_best(a_alt2)

    # Phase 7: Final gradient polish on the best found
    if best_arr is not None and time.monotonic() < deadline - 5:
        n_b = best_arr.size
        a_final = best_arr.copy()
        sb = float(a_final.sum())
        if sb > 1e-15:
            a_final = a_final / sb
        a_final = grad_optimize(a_final, max_iters=100000,
                                stop_time=deadline - 3, tau=25.0)
        update_best(a_final)

    # Final validation
    if best_arr is None:
        best_arr = np.ones(96, dtype=np.float64)
        update_best(best_arr)

    final = np.clip(best_arr, 0.0, 1000.0)
    if final.sum() < 0.01:
        final = np.ones_like(final)
        update_best(final)

    # One last check
    update_best(best_arr)

    return [float(x) for x in best_arr]
