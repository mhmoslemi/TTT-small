        local = n * np.convolve(a, a).max()
        ub(a)
        stale = 0

        for it in range(max_iters):
            if time.monotonic() > stop_time:
                break
            b = np.convolve(a, a)
            M = float(b.max())
            if M <= 0:
                break

            # Build sparse convolution matrix
            try:
                import scipy.sparse as sps
                from scipy.optimize import linprog
            except Exception:
                break

            rows = []
            cols = []
            vals = []
            for k in range(2 * n - 1):
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                for j in range(lo, hi + 1):
                    rows.append(k)
                    cols.append(j)
                    vals.append(a[k - j])
            A = sps.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

            improved = False
            for margin in [1.0, 0.99, 0.98, 0.96, 0.94]:
                if time.monotonic() > stop_time - 2:
                    break
                b_ub = np.full(2 * n - 1, M * margin + 1e-14)
                c_vec = -np.ones(n)
                try:
                    res = linprog(c_vec, A_ub=A, b_ub=b_ub,
                                  bounds=[(0.0, None)] * n, method='highs',
                                  options={'presolve': True, 'max_iter': 50000})
                except Exception:
                    continue
                if res is None or getattr(res, 'x', None) is None:
                    continue
                g = np.clip(res.x, 0, None)
                sg = float(g.sum())
                if sg < 1e-14:
                    continue
                g /= sg

                best_t_val = local
                best_t_cand = None
                for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
                    cand = (1.0 - t) * a + t * g
                    cand = np.clip(cand, 0, None)
                    cs = float(cand.sum())
                    if cs < 1e-12:
                        continue
                    cand /= cs
                    cv = n * np.convolve(cand, cand).max()
                    if cv < best_t_val:
                        best_t_val = cv
                        best_t_cand = cand

                if best_t_cand is not None and best_t_val < local * (1.0 - 1e-10):
                    a = best_t_cand
                    local = best_t_val
                    stale = 0
                    improved = True
                    ub(a)
                    break

            if not improved:
                stale += 1
                if stale > 15:
                    a_p = a * np.exp(0.1 * rng.standard_normal(n))
                    a_p = np.clip(a_p, 0, None)
                    ps = float(a_p.sum())
                    if ps > 1e-12:
                        a_p /= ps
                        if n * np.convolve(a_p, a_p).max() < local:
                            a = a_p
                            local = n * np.convolve(a, a).max()
                            stale = 0
                    if stale > 40:
                        break
        return a

    # === Interpolation helper ===
    def interp_to_len(src, target_n):
        if src.size == target_n:
            return src.copy()
        ox = np.linspace(0.0, 1.0, src.size)
        nx = np.linspace(0.0, 1.0, target_n)
        out = np.interp(nx, ox, src)
        out = np.clip(out, 0, None)
        s = float(out.sum())
        if s > 1e-15:
            out /= s
        else:
            out = np.ones(target_n) / target_n
        return out

    # === MAIN OPTIMIZATION LOOP ===
    p = parent if parent is not None else best_arr
    n_p = p.size
    p_norm = p / p.sum()

    # Phase 1: Aggressive soft-payoff gradient on parent length (primary workhorse)
    if time.monotonic() < deadline - 60:
        stop1 = min(deadline - 120, time.monotonic() + 500.0)
        a1 = softmax_grad_opt(p_norm, n_p, 50000, stop1, tau_start=2.0, tau_end=200.0)
        ub(a1)

    # Phase 2: LP refinement on the result
    if time.monotonic() < deadline - 40:
        stop2 = min(deadline - 60, time.monotonic() + 180.0)
        a2 = lp_refine(a1, n_p, 100, stop2)
        ub(a2)

    # Phase 3: Another gradient pass (tuned tau)
    if time.monotonic() < deadline - 30:
        stop3 = min(deadline - 30, time.monotonic() + 150.0)
        a3 = softmax_grad_opt(a2, n_p, 30000, stop3, tau_start=20.0, tau_end=200.0)
        ub(a3)

    # Phase 4: Try nearby lengths
    try_lengths = sorted(set([
        n_p - 8, n_p - 6, n_p - 4, n_p - 2,
        n_p + 2, n_p + 4, n_p + 6, n_p + 8,
        256, 280, 300, 340, 360, 400, 448, 512
    ]))
    try_lengths = [L for L in try_lengths if 16 <= L <= 600 and L != n_p]

    for L in try_lengths:
        if time.monotonic() > deadline - 25:
            break
        t_rem = deadline - time.monotonic() - 25
        if t_rem < 15:
            break
        alloc = min(35.0, t_rem * 0.15)
        if alloc < 10:
            break

        a_init = interp_to_len(p, L)
        stop_L = time.monotonic() + alloc * 0.6
        a_L = softmax_grad_opt(a_init, L, 5000, stop_L, tau_start=2.0, tau_end=100.0)
        ub(a_L)

        if time.monotonic() < deadline - 20:
            stop_L2 = min(deadline - 20, time.monotonic() + alloc * 0.35)
            a_L = lp_refine(a_L, L, 30, stop_L2)
            ub(a_L)

    # Phase 5: Multi-restart at parent length
    if time.monotonic() < deadline - 30:
        n_restarts = 12
        for r in range(n_restarts):
            if time.monotonic() > deadline - 20:
                break
            t_rem = deadline - time.monotonic() - 20
            alloc = min(25.0, t_rem / max(1, n_restarts - r))
            if alloc < 6:
                break

            style = r % 5
            if style == 0:
                a_init = p_norm * np.exp(0.12 * rng.standard_normal(n_p))
            elif style == 1:
                a_init = p_norm[::-1].copy()
            elif style == 2:
                a_init = p_norm * np.exp(0.25 * rng.standard_normal(n_p))
                a_init = np.clip(a_init, 0, None)
            elif style == 3:
                # Ramp initialization
                a_init = np.linspace(0.1, 2.0, n_p)
                a_init = a_init + 0.1 * rng.standard_normal(n_p)
                a_init = np.clip(a_init, 0, None)
            else:
                # Gaussian window
                x = np.linspace(0, 1, n_p)
                a_init = np.exp(-8.0 * (x - 0.5) ** 2) + 0.05 * rng.standard_normal(n_p)
                a_init = np.clip(a_init, 0, None)

            s_init = float(a_init.sum())
            if s_init > 1e-12:
                a_init /= s_init

            stop_r = time.monotonic() + alloc * 0.7
            a_r = softmax_grad_opt(a_init, n_p, 3000, stop_r, tau_start=2.0, tau_end=100.0)
            ub(a_r)

    # Phase 6: Final polish on best found
    if time.monotonic() < deadline - 10 and best_arr is not None:
        n_b = best_arr.size
        a_pol = best_arr.copy()
        a_pol = np.clip(a_pol, 0, None)
        s_pol = float(a_pol.sum())
        if s_pol > 1e-12:
            a_pol /= s_pol
        stop_pol = deadline - 5
        a_pol = softmax_grad_opt(a_pol, n_b, 20000, stop_pol, tau_start=30.0, tau_end=300.0)
        ub(a_pol)

        # LP polish on best
        if time.monotonic() < deadline - 5:
            stop_lp = deadline - 3
            a_pol = lp_refine(a_pol, n_b, 20, stop_lp)
            ub(a_pol)

    # Final validation and return
    if best_arr is None:
        best_arr = np.ones(320, dtype=np.float64)
        ub(best_arr)

    final = np.clip(best_arr, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones_like(final)
    ub(final)

    # Verify
    try:
        v = evaluate_sequence([float(x) for x in final])
        if not math.isfinite(v) or v >= best_score:
            # Fallback to best_arr
            final = best_arr
    except Exception:
        final = best_arr

    return [float(x) for x in np.clip(final, 0.0, 1000.0)]
