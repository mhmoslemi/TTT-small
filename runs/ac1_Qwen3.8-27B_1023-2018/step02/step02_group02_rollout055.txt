```python
def lp_min_peak(f, n):
    """Minimize max_k (f*g)[k] s.t. sum(g)=1, g>=0. Returns (g, min_peak)."""
    # Variables: g[0],...,g[n-1], t
    # Minimize t
    # (f*g)[k] <= t for k=0,...,2n-2
    # sum(g) = 1
    # g[j] >= 0
    
    c_obj = np.zeros(n+1)
    c_obj[-1] = 1  # minimize t
    
    # Inequality constraints: (f*g)[k] - t <= 0
    A_ub = np.zeros((2*n-1, n+1))
    for k in range(2*n-1):
        for j in range(max(0, k-n+1), min(k+1, n)):
            A_ub[k, j] = f[j]
        A_ub[k, -1] = -1  # -t
    
    b_ub = np.zeros(2*n-1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1
    b_eq = [1.0]
    
    bounds = [(0, None)] * n + [(None, None)]  # t is unbounded below
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'time_limit': 10})
    
    if res.success:
        g = res.x[:n]
        t = res.x[n]
        return g, t
    return None, None
```
