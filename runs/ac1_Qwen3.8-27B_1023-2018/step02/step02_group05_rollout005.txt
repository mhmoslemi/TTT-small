        if not math.isfinite(s) or s <= 0.0:
            return float("inf")
        b = np.convolve(a, a)
        mb = float(b.max())
        if not math.isfinite(mb) or mb < 0.0:
            return float("inf")
        return float(2.0 * a.size * mb / (s * s))
    except Exception:
        return float("inf")


def _softmax(y):
    try:
        y = np.asarray(y, dtype=np.float64).ravel()
        if y.size == 0 or not np.all(np.isfinite(y)):
            return None
        yy = y - np.mean(y)
        yy = yy - yy.max()
        e = np.exp(yy)
        s = float(e.sum())
        if s <= 0.0 or not math.isfinite(s):
            return None
        a = e / s
        if not np.all(np.isfinite(a)) or a.min() < 0.0:
            return None
        return a
    except Exception:
        return None


def _y_from_a(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 1e-300)
    y = np.log(a)
    y -= y.mean()
    if not np.all(np.isfinite(y)):
        y = np.zeros_like(a)
    return y


def _grad_from_weights(a, w, idx):
    # gradient of sum_k w_k c_k where c = a * a, with ordered-pair convention
    W = w[idx]
    T = W.dot(a)
    extra = a * w[0::2]
    g = T + extra
    if not np.all(np.isfinite(g)):
        g = np.zeros_like(a)
    return g


def _anti_from_delta(delta, n):
    n = int(n)
    if n % 2:
        return None
    m = n // 2
    mean = 1.0 / n
    d = np.asarray(delta, dtype=np.float64).ravel()
    if d.size != m:
        if d.size < m:
            d = np.pad(d, (0, m - d.size))
        else:
            d = d[:m]
    d = np.clip(d, -mean, mean)
    a = np.empty(n, dtype=np.float64)
    a[:m] = mean + d
    a[m:] = (mean - d)[::-1]
    s = float(a.sum())
    if s <= 0.0:
        return None
    return a / s


def _anti_from_z(z, n):
    n = int(n)
    if n % 2:
        return None
    z = np.asarray(z, dtype=np.float64).ravel()
    t = np.tanh(z)
    d = (1.0 / n) * t
    return _anti_from_delta(d, n)


def _interp_vec(a, n):
    n = int(n)
    if a is None:
        return np.ones(n, dtype=np.float64) / n
    a = _make_vec(a)
    if a is None:
        return np.ones(n, dtype=np.float64) / n
    if a.size == n:
        return a.copy()
    if a.size < 2:
        return np.full(n, 1.0 / n, dtype=np.float64)
    xo = np.linspace(0.0, 1.0, a.size)
    xn = np.linspace(0.0, 1.0, n)
    return np.interp(xn, xo, a)


def _gen_candidates(n, base, rng):
    out = []
    base = _make_vec(base)
    if base is None:
        base = np.ones(n, dtype=np.float64) / n
    out.append(base.copy())
    out.append(np.ones(n, dtype=np.float64) / n)

    if n == 2:
        out.append(np.array([2.0 / 3.0, 1.0 / 3.0], dtype=np.float64))
        out.append(np.array([1.0 / 3.0, 2.0 / 3.0], dtype=np.float64))

    if n % 2 == 0 and n >= 4:
        m = n // 2
        mean = 1.0 / n
        t = np.linspace(0.0, 1.0, m, dtype=np.float64)
        patterns = [
            t - 0.5,
            np.sin(2.0 * np.pi * t),
            np.cos(2.0 * np.pi * t),
            np.sin(4.0 * np.pi * t),
            np.sin(2.0 * np.pi * t + np.pi / 3.0),
            np.zeros(m, dtype=np.float64),
        ]
        for r in (0.20, 0.45, 0.70, 0.92):
            for p in patterns:
                a = _anti_from_delta(r * mean * np.asarray(p, dtype=np.float64), n)
                if a is not None:
                    out.append(a)
        for scale in (0.35, 0.85, 1.5):
            a = _anti_from_z(rng.normal(0.0, scale, m), n)
            if a is not None:
                out.append(a)

    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    for slope in (-1.5, -0.75, -0.35, 0.35, 0.75, 1.5, 3.0):
        out.append(base * (1.0 + slope * x))

    bpos = np.maximum(base, 1e-10)
    for p in (0.5, 2.0, 4.0):
        out.append(bpos ** p)

    if n > 5:
        k = np.array([1.0, 2.0, 1.0], dtype=np.float64)
        k /= k.sum()
        try:
            out.append(np.convolve(base, k, mode="same"))
        except Exception:
            pass

    for alpha in (0.05, 0.20, 0.50, 1.0, 2.0, 5.0, 15.0):
        try:
            out.append(rng.dirichlet(np.full(n, alpha, dtype=np.float64)))
        except Exception:
            pass

    return out


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    try:
        budget = max(10.0, float(budget_s))
    except Exception:
        budget = 1300.0

    t0 = time.monotonic()
    deadline = t0 + budget - 30.0

    def time_left():
        return max(0.0, deadline - time.monotonic())

    eval_func = globals().get("evaluate_sequence", None)
    if not callable(eval_func):
        eval_func = _fallback_evaluator

    def _exact(a):
        try:
            sc = eval_func([float(x) for x in a])
        except Exception:
            sc = float("inf")
        try:
            sc = float(sc)
        except Exception:
            sc = float("inf")
        if math.isfinite(sc):
            return sc
        return float("inf")

    best_score = float("inf")
    best_vec = None

    def _exact_consider(a, update=True):
        nonlocal best_score, best_vec
        arr = _make_vec(a)
        if arr is None:
            return float("inf"), None
        sc = _exact(arr)
        if update and math.isfinite(sc) and sc < best_score - 1e-15:
            best_score = sc
            best_vec = arr.copy()
        return (sc if math.isfinite(sc) else float("inf")), arr

    rng = np.random.default_rng(seed)

    # Basic valid fallbacks.
    for fb in ([1.0, 1.0], np.ones(16, dtype=np.float64) / 16.0, np.ones(320, dtype=np.float64) / 320.0):
        _exact_consider(fb)

    parent = None
    try:
        raw = globals().get("height_sequence_1", None)
        if raw is not None:
            p = _make_vec(raw)
            if p is not None and p.size >= 2:
                parent = p
                _exact_consider(parent)
    except Exception:
        parent = None

    pool = []
    generated = 0
    max_generated = 900

    if time_left() > 60:
        lengths = []
        if parent is not None:
            lengths.append(parent.size)
            for d in (-32, -16, 16, 32, 64):
                lengths.append(parent.size + d)
        for n in (2, 4, 8, 16, 32, 64, 128, 256, 320, 448, 512, 640):
            lengths.append(n)

        uniq = []
        seen = set()
        for n in lengths:
            n = int(n)
            if 2 <= n <= 1024 and n not in seen:
                seen.add(n)
                uniq.append(n)
        if parent is not None:
            uniq.sort(key=lambda n: (0 if n == parent.size else 1, abs(n - parent.size)))
        else:
            uniq.sort()
        uniq = uniq[:14]

        for n in uniq:
            if time_left() < 35 or generated >= max_generated:
                break
            base = None
            if parent is not None:
                base = _interp_vec(parent, n)
            if base is None:
                base = np.ones(n, dtype=np.float64) / n
            for cand in _gen_candidates(n, base, rng):
                if time_left() < 35 or generated >= max_generated:
                    break
                sc, arr = _exact_consider(cand)
                if math.isfinite(sc) and arr is not None:
                    pool.append((sc, arr))
                generated += 1

    # Choose diverse starting points for refinement.
    starts = []

    def _add_start(a):
        arr = _make_vec(a)
        if arr is None or arr.size < 2:
            return
        if len(starts) >= 5:
            return
        # Very lightweight duplicate filter: same length and same first head empirically enough.
        for s in starts:
            if s.size == arr.size and np.linalg.norm(s - arr) < 1e-3:
                return
        starts.append(arr)

    _add_start(best_vec)
    _add_start(parent)
    for _, arr in sorted(pool, key=lambda x: x[0])[:12]:
        _add_start(arr)
        if len(starts) >= 5:
            break
    if not starts:
        _add_start(np.ones(320, dtype=np.float64) / 320.0)

    def _run_lp(a_start, max_time):
        if linprog is None:
            return
        a = _make_vec(a_start)
        if a is None:
            return
        n = a.size
        if n < 2 or n > 768:
            return
        lp_start = time.monotonic()
        lp_deadline = min(deadline, lp_start + max_time)
        try:
            idx, mask = _get_lp_data(n)
        except Exception:
            return

        local = _score_internal(a)
        if not math.isfinite(local):
            return
        stagnant = 0

        while True:
            now = time.monotonic()
            if now >= lp_deadline - 2.0 or now >= deadline - 20.0:
                break
            if stagnant > 7:
                break

            try:
                c = np.convolve(a, a)
                M = float(c.max())
            except Exception:
                break
            if not math.isfinite(M) or M <= 1e-15:
                break

            A = np.zeros((2 * n - 1, n), dtype=np.float64)
            try:
                A[mask] = a[idx[mask]]
            except Exception:
                break

            b_ub = np.full(2 * n - 1, M * (1.0 + 1e-10) + 1e-14, dtype=np.float64)
            tl = max(1.0, min(30.0, lp_deadline - now - 5.0))
            g = None
            try:
                res = linprog(
                    -np.ones(n, dtype=np.float64),
                    A_ub=A,
                    b_ub=b_ub,
                    bounds=[(0.0, None)] * n,
                    method="highs",
                    options={"presolve": True, "time_limit": tl},
                )
            except Exception:
                res = None

            if res is not None and getattr(res, "x", None) is not None:
                x = np.asarray(res.x, dtype=np.float64).ravel()
                if x.size == n and np.all(np.isfinite(x)):
                    x = np.clip(x, 0.0, None)
                    if float(x.sum()) > 1e-12:
                        try:
                            viol = float(np.max(A @ x - b_ub))
                        except Exception:
                            viol = float("inf")
                        if viol < 1e-5:
                            g = x / float(x.sum())

            if g is None:
                stagnant += 1
                if stagnant > 1:
                    # Try a slightly relaxed feasibility tolerance once.
                    b_ub2 = np.full(2 * n - 1, M * (1.0 + 1e-6) + 1e-13, dtype=np.float64)
                    try:
                        res = linprog(
                            -np.ones(n, dtype=np.float64),
                            A_ub=A,
                            b_ub=b_ub2,
                            bounds=[(0.0, None)] * n,
                            method="highs",
                            options={"presolve": True, "time_limit": max(1.0, min(20.0, lp_deadline - now - 4.0))},
                        )
                    except Exception:
                        res = None
                    if res is not None and getattr(res, "x", None) is not None:
                        x = np.asarray(res.x, dtype=np.float64).ravel()
                        if x.size == n and np.all(np.isfinite(x)):
                            x = np.clip(x, 0.0, None)
                            if float(x.sum()) > 1e-12:
                                try:
                                    viol = float(np.max(A @ x - b_ub2))
                                except Exception:
                                    viol = float("inf")
                                if viol < 1e-4:
                                    g = x / float(x.sum())
                if g is None:
                    if stagnant >= 8:
                        break
                    continue

            best_cand = a
            best_sc = local
            for t in (0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.65, 0.8, 1.0):
                cand = (1.0 - t) * a + t * g
                cand = np.clip(cand, 0.0, None)
                sc = _score_internal(cand)
                if math.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_cand = cand

            if best_sc < local * (1.0 - 1e-10):
                na = _make_vec(best_cand)
                if na is None:
                    break
                a = na
                local = best_sc
                stagnant = 0
                _exact_consider(a)
            else:
                stagnant += 1
                if stagnant % 3 == 0:
                    pa = a * np.exp(0.10 * rng.standard_normal(n))
                    pn = _make_vec(pa)
                    if pn is not None:
                        psc = _score_internal(pn)
                        if math.isfinite(psc) and psc < local:
                            a = pn
                            local = psc
                            stagnant = 0
                            _exact_consider(a)

    def _run_full_smooth(a_start, max_time):
        if minimize is None:
            return
        a = _make_vec(a_start)
        if a is None:
            return
        n = a.size
        if n < 2 or time_left() < 35:
            return
        try:
            idx = _get_grad_idx(n)
        except Exception:
            idx = None
            if idx is None:
                return

        y0 = _y_from_a(a)
        sc0 = _score_internal(a)
        if not math.isfinite(sc0):
            sc0 = 2.0
        P = max(sc0 / (2.0 * n), 1.0 / max(n, 1))
        factors = (4.0, 20.0, 90.0, 350.0, 1400.0, 5600.0)
        betas = []
        for f in factors:
            b = float(f / max(P, 1e-15))
            b = min(1.0e7, max(1.0e-3, b))
            betas.append(b)

        total = min(float(max_time), max(20.0, time_left() - 15.0))
        if total < 25.0:
            return
        per = max(6.0, total / float(len(betas)))
        y_cur = y0.copy()

        for beta in betas:
            if time_left() < 25.0:
                break
            y_best = [y_cur.copy()]
            f_best = [float("inf")]
            stage_deadline = min(deadline, time.monotonic() + min(per, max(5.0, time_left() - 10.0)))

            def obj(y_in, beta=beta):
                if time.monotonic() > stage_deadline:
                    raise _BudgetTimeout()
                try:
                    y_in = np.asarray(y_in, dtype=np.float64).ravel()
                    if y_in.size != n or not np.all(np.isfinite(y_in)):
                        return 1.0e100, np.zeros(n, dtype=np.float64)
                    yy = y_in - np.mean(y_in)
                    aa = _softmax(yy)
                    if aa is None:
                        return 1.0e100, np.zeros(n, dtype=np.float64)
                    c = np.convolve(aa, aa)
                    cm = float(c.max())
                    u = beta * (c - cm)
                    em = np.exp(u)
                    se = float(em.sum())
                    if se <= 0.0 or not math.isfinite(se):
                        return 1.0e100, np.zeros(n, dtype=np.float64)
                    fval = 2.0 * n * (cm + math.log(se) / beta)
                    w = em / se
                    g = (2.0 * n) * _grad_from_weights(aa, w, idx)
                    ga = float(np.dot(aa, g))
                    grad = aa * (g - ga)
                    grad = grad - np.mean(grad)
                    if not np.all(np.isfinite(grad)):
                        grad = np.zeros(n, dtype=np.float64)
                    if math.isfinite(fval) and fval < f_best[0]:
                        f_best[0] = fval
                        y_best[0] = yy.copy()
                    return fval, grad
                except _BudgetTimeout:
                    raise
                except Exception:
                    return 1.0e100, np.zeros(n, dtype=np.float64)

            try:
                minimize(
                    obj,
                    y_cur.copy(),
                    method="L-BFGS",
                    jac=True,
                    options={
                        "maxiter": 260,
                        "maxfun": 1200,
                        "ftol": 1e-12,
                        "gtol": 1e-10,
                        "maxls": 20,
                    },
                )
            except _BudgetTimeout:
                pass
            except Exception:
                pass

            a_new = _softmax(y_best[0])
            if a_new is not None:
                _exact_consider(a_new)
                y_cur = _y_from_a(a_new)

    def _run_anti_smooth(a_start, max_time):
        if minimize is None:
            return
        a = _make_vec(a_start)
        if a is None:
            return
        n = a.size
        if n < 8 or (n % 2) or time_left() < 35:
            return
        m = n // 2
        mean = 1.0 / n
        try:
            idx = _get_grad_idx(n)
        except Exception:
            return

        def _z_from(a_now):
            left = a_now[:m]
            right = a_now[n - 1:m - 1:-1]
            ratio = (left - right) / (2.0 * mean)
            ratio = np.clip(ratio, -0.999999, 0.999999)
            z = np.arctanh(ratio)
            return np.where(np.isfinite(z), z, 0.0)

        z_seeds = [_z_from(a)]
        z_seeds.append(np.zeros(m, dtype=np.float64))
        z_seeds.append(rng.normal(0.0, 0.65, m))

        sc0 = _score_internal(a)
        if not math.isfinite(sc0):
            sc0 = 2.0
        P = max(sc0 / (2.0 * n), 1.0 / max(n, 1))
        factors = (8.0, 40.0, 160.0, 640.0, 2560.0)
        betas = [min(1.0e7, max(1e-3, float(f / max(P, 1e-15)))) for f in factors]

        total = min(float(max_time), max(20.0, time_left() - 15.0))
        if total < 25.0:
            return
        per_seed = max(12.0, total / float(len(z_seeds)))

        for seed_z in z_seeds:
            if time_left() < 25.0:
                break
            z_cur = np.asarray(seed_z, dtype=np.float64).ravel().copy()
            seed_deadline = min(deadline, time.monotonic() + per_seed)

            for beta in betas:
                if time.monotonic() >= seed_deadline - 1.0 or time_left() < 25.0:
                    break
                z_best = [z_cur.copy()]
                f_best = [float("inf")]
                stage_deadline = min(seed_deadline, deadline, max(time.monotonic() + 4.0, time.monotonic() + max(2.0, (seed_deadline - time.monotonic()) / 6.0)))

                def obj(z_in, beta=beta):
                    if time.monotonic() > stage_deadline:
                        raise _BudgetTimeout()
                    try:
                        z_in = np.asarray(z_in, dtype=np.float64).ravel()
                        if z_in.size != m or not np.all(np.isfinite(z_in)):
                            return 1.0e100, np.zeros(m, dtype=np.float64)
                        tt = np.tanh(z_in)
                        sech2 = 1.0 - tt * tt
                        d = mean * tt
                        aa = np.empty(n, dtype=np.float64)
                        aa[:m] = mean + d
                        aa[m:] = (mean - d)[::-1]
                        ss = float(aa.sum())
                        if ss <= 0.0 or not np.all(np.isfinite(aa)):
                            return 1.0e100, np.zeros(m, dtype=np.float64)
                        aa = aa / ss
                        c = np.convolve(aa, aa)
                        cm = float(c.max())
                        u = beta * (c - cm)
                        em = np.exp(u)
                        se = float(em.sum())
                        if se <= 0.0 or not math.isfinite(se):
                            return 1.0e100, np.zeros(m, dtype=np.float64)
                        fval = 2.0 * n * (cm + math.log(se) / beta)
                        w = em / se
                        g = (2.0 * n) * _grad_from_weights(aa, w, idx)
                        gz = mean * sech2 * (g[:m] - g[m:][::-1])
                        gz = gz - np.mean(gz)
                        if not np.all(np.isfinite(gz)):
                            gz = np.zeros(m, dtype=np.float64)
                        if math.isfinite(fval) and fval < f_best[0]:
                            f_best[0] = fval
                            z_best[0] = z_in.copy()
                        return fval, gz
                    except _BudgetTimeout:
                        raise
                    except Exception:
                        return 1.0e100, np.zeros(m, dtype=np.float64)

                try:
                    minimize(
                        obj,
                        z_cur.copy(),
                        method="L-BFGS",
                        jac=True,
                        options={
                            "maxiter": 240,
                            "maxfun": 1100,
                            "ftol": 1e-12,
                            "gtol": 1e-10,
                            "maxls": 20,
                        },
                    )
                except _BudgetTimeout:
                    pass
                except Exception:
                    pass

                a_new = _anti_from_z(z_best[0], n)
                if a_new is not None:
                    _exact_consider(a_new)
                    z_cur = z_best[0].copy()

    def _run_subgrad(a_start, max_time):
        a = _make_vec(a_start)
        if a is None:
            return
        n = a.size
        if n < 2 or time_left() < 20:
            return
        try:
            idx = _get_grad_idx(n)
        except Exception:
            idx = None
            if idx is None:
                return

        y = _y_from_a(a)
        local = _score_internal(a)
        stagnant = 0
        end = min(deadline, time.monotonic() + float(max_time))

        for _ in range(20000):
            if time.monotonic() >= end - 1.0 or time_left() < 15.0:
                break
            if stagnant >= 25:
                break
            try:
                c = np.convolve(a, a)
                M = float(c.max())
            except Exception:
                break
            if not math.isfinite(M) or M <= 1e-15:
                break
            order = np.argsort(c)[-24:]
            active = order[c[order] >= M - max(0.10 * M, 1e-12)]
            if active.size == 0:
                active = order[:1]
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[active] = 1.0 / float(active.size)
            g = (2.0 * n) * _grad_from_weights(a, w, idx)
            ga = float(np.dot(a, g))
            d = a * (g - ga)
            d = d - np.mean(d)
            nd = float(np.linalg.norm(d))
            if not math.isfinite(nd) or nd < 1e-14:
                break
            direction = d / nd
            improved = False
            for step in (0.01, 0.03, 0.1, 0.3, 1.0, 3.0, 10.0, 30.0):
                y2 = y - step * direction
                a2 = _softmax(y2)
                if a2 is None:
                    continue
                sc = _score_internal(a2)
                if math.isfinite(sc) and sc < local * (1.0 - 1e-12):
                    y = y2
                    a = a2
                    local = sc
                    improved = True
                    _exact_consider(a)
                    break
            if improved:
                stagnant = 0
            else:
                stagnant += 1
                if stagnant % 5 == 0:
                    yp = y + 0.05 * rng.standard_normal(n)
                    ap = _softmax(yp)
                    if ap is not None:
                        sp = _score_internal(ap)
                        if math.isfinite(sp) and sp < local:
                            y = yp
                            a = ap
                            local = sp
                            stagnant = 0
                            _exact_consider(a)

    # Main search schedule.
    if linprog is not None and time_left() > 150:
        for st in starts[:2]:
            if time_left() < 90:
                break
            _run_lp(st, min(260.0, time_left() * 0.30))

    if minimize is not None and time_left() > 220:
        for st in starts[:3]:
            if time_left() < 110:
                break
            _run_full_smooth(st, min(320.0, time_left() * 0.35))

    if minimize is not None and time_left() > 180:
        for st in starts[:2]:
            if time_left() < 100:
                break
            if st is not None and st.size % 2 == 0 and st.size >= 8:
                _run_anti_smooth(st, min(240.0, time_left() * 0.35))

    if linprog is not None and time_left() > 110:
        _run_lp(best_vec if best_vec is not None else starts[0], min(180.0, time_left() * 0.30))

    if minimize is not None and time_left() > 110:
        _run_full_smooth(best_vec if best_vec is not None else starts[0], min(200.0, time_left() * 0.35))

    if time_left() > 70:
        _run_subgrad(best_vec if best_vec is not None else starts[0], min(90.0, time_left() * 0.25))

    # Explore nearby lengths only if time remains.
    optimized_lengths = set(s.size for s in starts if s is not None)
    trial_lengths = []
    if parent is not None:
        pl = int(parent.size)
        for d in (-32, -16, 16, 32, 64):
            ntr = pl + d
            if 2 <= ntr <= 768 and ntr not in optimized_lengths:
                trial_lengths.append(ntr)
    for ntr in (128, 256, 320, 384, 448, 512, 640):
        if ntr not in optimized_lengths:
            trial_lengths.append(ntr)

    for ntr in trial_lengths[:4]:
        if time_left() < 95:
            break
        ntr = int(ntr)
        if ntr in optimized_lengths or ntr < 4 or ntr > 768:
            continue
        a0 = _interp_vec(parent if parent is not None else (best_vec if best_vec is not None else None), ntr)
        if a0 is None:
            a0 = np.ones(ntr, dtype=np.float64) / ntr
        _exact_consider(a0)
        if minimize is not None:
            _run_full_smooth(a0, min(90.0, time_left() * 0.25))
        if minimize is not None and ntr % 2 == 0 and ntr >= 8:
            _run_anti_smooth(a0, min(70.0, time_left() * 0.20))
        optimized_lengths.add(ntr)

    # Final boundary/simple structures.
    if best_vec is not None and time_left() > 8:
        a = _make_vec(best_vec)
        if a is not None:
            _exact_consider(0.5 * (a + a[::-1]))
            _exact_consider(a[::-1])
            for frac in (0.05, 0.10, 0.20, 0.30, 0.50):
                try:
                    q = float(np.quantile(a, frac))
                except Exception:
                    q = 0.0
                _exact_consider(np.where(a > q, a, 0.0))
            for keep in (1, 2, 3, 4, 5, 8, 12, 16, 24, 32, 48, 64, 96):
                if keep >= a.size:
                    idxs = np.arange(a.size)
                else:
                    idxs = np.argsort(a)[-keep:]
                z = np.zeros(a.size, dtype=np.float64)
                z[idxs] = 1.0
                _exact_consider(z)

    # Ensure a valid exact-scored final fallback.
    if best_vec is None or not _make_vec(best_vec):
        _exact_consider([1.0, 1.0])
        _exact_consider(np.ones(2, dtype=np.float64) / 2.0)

    if parent is not None:
        _exact_consider(parent)

    # Final validation.
    final_score, final_vec = _exact_consider(best_vec, update=True)
    if final_vec is None or not math.isfinite(final_score):
        final_score, final_vec = _exact_consider(np.ones(2, dtype=np.float64) / 2.0, update=True)

    if final_vec is None:
        final_vec = np.ones(2, dtype=np.float64) / 2.0

    final_vec = _make_vec(final_vec)
    if final_vec is None:
        final_vec = np.ones(2, dtype=np.float64) / 2.0

    return [float(x) for x in final_vec]
