```python
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
import cvxpy as cp

def _get_parent():

        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(192, dtype=np.float64)
    try:
        p = np.clip(p, 0.0, 1000.0)
    except Exception:
        p = np.ones(192, dtype=np.float64)
    if p.size == 0 or not np.isfinite(p).all():
        p = np.ones(192, dtype=np.float64)
    return p

def _fft_conv_self(a):
    """Compute linear convolution of a with itself using FFT."""
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2*n-1]
    np.maximum(c, 0.0, out=c)
    c[~np.isfinite(c)] = 0.0
    return c

def _build_A(a):
    """Build the matrix A such that (A @ g)_k = (a * g)_k for all k."""
    n = len(a)
    # (a * g)_k = sum_j a[j] * g[k-j] for valid j
    # = sum_j g[j] * a[k-j] for valid j (symmetric form)
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    k_idx = np.arange(2*n-1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    A = np.zeros((2*n-1, n))
    valid = (diff >= 0) & (diff < n)
    A[valid] = a[diff[valid].astype(np.int64)]
    return A

def _solve_lp(a, M, n, deadline):
    """Solve the LP: max sum(g) s.t. A@g <= M, g >= 0."""
    try:
        A = _build_A(a)
        g = cp.Variable(n, nonneg=True)
        constraints = [A @ g <= M]
        obj = cp.Maximize(cp.sum(g))
        prob = cp.Problem(obj, constraints)
        
        for solver in [cp.ECOS, cp.SCS, cp.GLPK, cp.CBC, cp.CLARABEL]:
            try:
                if time.monotonic() > deadline:
                    return None
                prob.solve(solver=solver, verbose=False)
                if prob.status in ['optimal', 'optimal_inaccurate']:
                    gv = g.value
                    if gv is not None and np.isfinite(gv).all():
                        gv = np.maximum(gv, 0)
                        s = gv.sum()
                        if s > 1e-15:
                            return gv / s
            except Exception:
                continue
    except Exception:
        pass
    return None
```

```python
def _solve_kkt(n, K, S, max_iter=100):
    """Solve the KKT system given active peak set K and support S."""
    S_arr = np.array(S, dtype=int)
    K_arr = np.array(K, dtype=int)
    s = len(S_arr)
    k = len(K_arr)
    
    # Build the system: for m in S, (2/k) * sum_{ki in K} a[|ki-m|] - lambda = 0
    # Plus: sum(a[m] for m in S) = 1
    # Variables: a[S_arr[0]], ..., a[S_arr[s-1]], lambda
    # Total: s+1 variables
    
    # Build matrix A_sys of size (s+1) x (s+1)
    A_sys = np.zeros((s+1, s+1))
    b_sys = np.zeros(s+1)
    
    # Map from value to index in S
    s_set = set(S_arr.tolist())
    
    for i, m in enumerate(S_arr):
        # Row i: (2/k) * sum_{ki in K} a[|ki-m|] - lambda = 0
        for ki in K_arr:
            idx = abs(ki - m)
            if idx in s_set:
                j = np.where(S_arr == idx)[0][0]  # This is O(s) - could be O(1) with a dict
                A_sys[i, j] += 2.0 / k
        A_sys[i, s] = -1.0  # lambda term
        b_sys[i] = 0.0
    
    # Last row: sum of a[S] = 1
    A_sys[s, :s] = 1.0
    b_sys[s] = 1.0
    
    # Solve
    try:
        sol = np.linalg.solve(A_sys, b_sys)
    except np.linalg.LinAlgError:
        sol = np.linalg.lstsq(A_sys, b_sys, rcond=None)[0]
    
    a = np.zeros(n)
    for i, m in enumerate(S_arr):
        a[m] = max(0, sol[i])
    lambda_val = sol[s]
    
    return a, lambda_val
```

```python
def _get_parent():

def _fft_conv_self(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2*n-1]
    np.maximum(c, 0.0, out=c)
    c[~np.isfinite(c)] = 0.0
    return c

def _project_simplex(a, c=1.0):
    """Project a onto {x >= 0, sum(x) = c}."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-30:
        n = len(a)
        return np.full(n, c / n)
    return c * a / s

def _build_A(a):
    """Build matrix A where (A@g)_k = (a*g)_k."""
    n = len(a)
    k_idx = np.arange(2*n-1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    A = np.zeros((2*n-1, n))
    valid = (diff >= 0) & (diff < n)
    A[valid] = a[diff[valid].astype(np.int64)]
    return A

def _solve_lp_minmax(a, n, deadline):
    """Solve: min z s.t. (a*g)_k <= z for all k, sum(g)=1, g>=0.
    Returns (g_normalized, z_optimal) or (None, inf)."""
    try:

        g = cp.Variable(n, nonneg=True)
        z = cp.Variable()
        constraints = [A @ g <= z, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(z), constraints)
        for solver in [cp.ECOS, cp.CLARABEL, cp.SCS, cp.GLPK, cp.CBC]:

def _subgradient(a, b):
    """Compute subgradient of max(b) w.r.t. a. Returns vector of length len(a)."""
    n = len(a)

    grad *= 2.0 / len(active)
    return grad

    """Solve the KKT system with active peak set K_list.
    Returns (a, score) or (None, inf)."""
    S = list(range(n))  # full support
    K = list(K_list)
    
    for iteration in range(max_iter):
        if deadline and time.monotonic() > deadline:
            return None, np.inf
        if len(K) == 0:
            return None, np.inf

        p = len(K)

        # Plus: sum(a_m for m in S) = 1
        A_sys = np.zeros((s + 1, s + 1))
        b_sys = np.zeros(s + 1)
        
        s_to_idx = {v: i for i, v in enumerate(S)}
        
        for i, m in enumerate(S):
            for k in K:
                d = abs(k - m)
                if d < n and d in s_to_idx:
                    A_sys[i, s_to_idx[d]] += 2.0 / p
                elif d < n:
                    # d not in support, so a_d = 0, no contribution
                    pass
            A_sys[i, s] = -1.0
        
        A_sys[s, :s] = 1.0
        b_sys[s] = 1.0
        
        # Solve
        try:
            sol = np.linalg.solve(A_sys, b_sys)
        except np.linalg.LinAlgError:
            try:
                sol = np.linalg.lstsq(A_sys, b_sys, rcond=1e-12)[0]
            except Exception:
                return None, np.inf
        
        a = np.zeros(n)
        for i, m in enumerate(S):
            a[m] = sol[i]

            # Remove the most negative from S
            worst = int(np.argmin(a))
            if worst in s_to_idx:
                S.remove(worst)
            continue
        
        # Clip small negatives to zero
        a = np.maximum(a, 0)

        sa = a.sum()
        if sa < 1e-15:
            return None, np.inf
        a = a / sa

        residual = np.max(np.abs((2.0/p) * np.array([sum(a[abs(k-m)] for k in K) for m in range(n)]) - lambda_val))
        if residual < 1e-10:

    rng = np.random.default_rng(seed)
    
    # Evaluate parent
    try:
        best_score = float(evaluate_sequence([float(x) for x in np.clip(parent, 0, 1000)]))
    except Exception:
        best_score = math.inf
    best_seq = np.clip(parent, 0, 1000).astype(np.float64).copy()
    
    if not np.isfinite(best_score):
        best_seq = np.ones(32)
        best_score = float(evaluate_sequence([1.0] * 32))
    
    def _update_best(a, n):
        nonlocal best_seq, best_score
        fast_sc = 2.0 * n * float(_fft_conv_self(a).max())
        if fast_sc < best_score - 1e-12:
            try:
                exact = float(evaluate_sequence([float(x) for x in a]))
                if np.isfinite(exact) and exact < best_score:
                    best_seq = a.copy()
                    best_score = exact
            except Exception:
                pass
    
    # Phase 1: KKT method on parent (20% of budget)
    phase1_dl = t0 + budget * 0.20
    if time.monotonic() < phase1_dl:
        a_norm = _project_simplex(best_seq.copy(), 1.0)
        b = _fft_conv_self(a_norm)
        k_star = int(np.argmax(b))
        # Try K = {k_star} and K = {k_star, n-1} and nearby
        for K_try in [[k_star], [k_star, n-1] if k_star != n-1 else [k_star, n-2], 
                       [k_star, max(0, k_star-1), min(2*n-2, k_star+1)]]:
            if time.monotonic() > phase1_dl - 2:
                break
            a_kkt, _ = _kkt_solve(n_parent, K_try, deadline=phase1_dl - 2)
            if a_kkt is not None:
                _update_best(a_kkt, n_parent)
    
    # Phase 2: LP optimization on parent (35% of budget)
    phase2_dl = t0 + budget * 0.55
    if time.monotonic() < phase2_dl:
        a_norm = _project_simplex(best_seq.copy(), 1.0)
        sub_dl = min(phase2_dl - 3, time.monotonic() + budget * 0.30)
        try:
            res_a, res_mb = _lp_optimize(a_norm, n_parent, sub_dl, rng)
            _update_best(res_a, n_parent)
        except Exception:
            pass
    
    # Phase 3: Multiple restarts with LP (25% of budget)
    phase3_dl = t0 + budget * 0.80
    if time.monotonic() < phase3_dl:
        num_restarts = 12
        base_score = best_score
        for rid in range(num_restarts):
            if time.monotonic() > phase3_dl - 5:
                break
            if best_score < 1.50:
                break
            a_init = _init_variants(n_parent, rng, variant=rid + 50)
            remaining = max(3.0, phase3_dl - time.monotonic() - 5)
            per_restart_tl = min(40.0, remaining / max(1, num_restarts - rid) * 1.2)
            sub_dl = time.monotonic() + per_restart_tl
            try:
                res_a, _ = _lp_optimize(a_init, n_parent, sub_dl, rng)
                _update_best(res_a, n_parent)
            except Exception:
                pass
        # Also try KKT on the best found so far
        if time.monotonic() < phase3_dl - 3:
            a_norm = _project_simplex(best_seq.copy(), 1.0)
            b = _fft_conv_self(a_norm)
            peaks = np.argsort(b)[::-1][:5]
            for K_try in [list(peaks[:1]), list(peaks[:2]), list(peaks[:3])]:
                if time.monotonic() > phase3_dl - 2:
                    break
                a_kkt, _ = _kkt_solve(n_parent, K_try, deadline=phase3_dl - 2)
                if a_kkt is not None:
                    _update_best(a_kkt, n_parent)
    
    # Phase 4: Local search polishing (20% of budget)
    phase4_dl = deadline - 5
    if time.monotonic() < phase4_dl:
        a_norm = _project_simplex(best_seq.copy(), 1.0)
        # Multiple local search runs with different rng seeds
        for ls_i in range(3):
            if time.monotonic() > phase4_dl - 3:
                break
            sub_dl = time.monotonic() + min(30.0, (phase4_dl - time.monotonic()) / (3 - ls_i))
            try:
                res_a, _ = _local_search(a_norm, sub_dl, np.random.default_rng(seed + 1000 + ls_i * 7))
                _update_best(res_a, len(res_a))
            except Exception:
                pass
    
    # Phase 5: Try different lengths (if time remains)
    if time.monotonic() < deadline - 8:
        for L in [128, 256]:
            if time.monotonic() > deadline - 6:
                break
            a_init = _init_variants(L, rng, variant=L % 8)
            sub_dl = time.monotonic() + max(3.0, min(25.0, deadline - time.monotonic() - 8))
            try:
                res_a, _ = _lp_optimize(a_init, L, sub_dl, rng)
                _update_best(res_a, L)
            except Exception:
                pass
    
    # Final KKT polish on best
    if time.monotonic() < deadline - 3:
        a_norm = _project_simplex(best_seq.copy(), 1.0)
        n_best = len(a_norm)
        b = _fft_conv_self(a_norm)
        k_star = int(np.argmax(b))
        for K_try in [[k_star], [k_star, n_best-1] if k_star != n_best-1 else [k_star, 0]]:
            if time.monotonic() > deadline - 2:
                break
            a_kkt, _ = _kkt_solve(n_best, K_try, deadline=deadline - 2)
            if a_kkt is not None:
                _update_best(a_kkt, n_best)
    
    # Final validation
    try:
        final_score = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_score = math.inf
    
    if not np.isfinite(final_score):
        fb = np.ones(32)
        return [float(x) for x in fb]
    
    return [float(x) for x in best_seq]
```
