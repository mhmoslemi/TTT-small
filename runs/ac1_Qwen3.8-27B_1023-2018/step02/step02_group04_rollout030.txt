    n = f.size
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = f

    b = np.convolve(f, f)
    M = b.max()
    if M < 1e-300: return None
    A = _conv_mat(f)
    try:
        r = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M*(1+1e-10)),
                    bounds=[(0,None)]*n, method='highs',
                    options={'time_limit':tl, 'presolve':True})
        if r.success and r.x is not None:
            g = np.clip(r.x, 0, None)
            s = g.sum()
            return g/s if s > 1e-300 else None
    except: pass
    return None

def _lp_search(a0, stop_t, best, rng):
    a = np.clip(a0, 0, None); a = a/a.sum()
    n = a.size; loc = _sc(a); _upd(a, best)
    stag = 0
    while time.monotonic() < stop_t:
        g = _lp_dir(a)
        if g is None:
            stag += 5
            if stag > 100: break
            continue
        bs, bt = loc, 0
        for t in (0.1, 0.2, 0.3, 0.5, 0.7, 1.0):
            c = np.clip((1-t)*a + t*g, 0, None)
            c = c/c.sum()
            s = _sc(c)
            if s < bs: bs, bt = s, t
        if bt > 0 and bs < loc*(1-1e-12):
            a = np.clip((1-bt)*a + bt*g, 0, None); a = a/a.sum()
            loc = bs; stag = 0; _upd(a, best)
        else:
            stag += 1
            if stag > 50 and stag % 25 == 0:
                p = a * np.exp(0.08*rng.standard_normal(n))
                p = np.clip(p, 0, None)
                if p.sum() > 1e-6:
                    sp = _sc(p/p.sum())
                    if sp < loc: a = p/p.sum(); loc = sp; stag = 0; _upd(a, best)
            if stag > 500: break
    return a

def _subgrad(a, stop_t, best, rng):
    a = a / a.sum(); n = a.size; loc = _sc(a); _upd(a, best)
    eta = 0.02
    for it in range(50000):
        if time.monotonic() > stop_t: break
        b = np.convolve(a, a)
        M = b.max()
        # Active lags: those within 1% of max
        active = np.where(b >= M * 0.99)[0]
        sg = np.zeros(n)
        for k in active:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    sg[i] += 2.0 * a[j]
        sg /= len(active)
        sg -= sg.mean()  # Remove mean for sum constraint
        c = a - eta * sg
        c = np.clip(c, 1e-10, None)
        c = c / c.sum()
        sc = _sc(c)
        if sc < loc * (1 - 1e-14):
            a = c; loc = sc; _upd(a, best)
            # AdaGrad-style: reduce eta slowly
            eta *= 0.999
        else:
            eta *= 0.9
            if eta < 1e-6: break
    return a

Actually, I realize I should test whether the LP actually improves the score consistently. Let me trace through the logic:

1. Start with normalized a (sum=1)
2. Compute b = a*a, M = max(b)
3. Solve LP: max sum(g) s.t. (a*g) <= M, g >= 0
4. The LP solution g has sum > 0 (by feasibility, g=0 works but the max doesn't)
5. Normalize g: g = g/sum(g)
6. Blend: c = (1-t)*a + t*g, normalize
7. Check if sc(c) < sc(a)

Step 6-7: does this always help? Not necessarily. The LP ensures (a*g) <= M, but the blended autocorrelation is ((1-t)*a+t*g) * ((1-t)*a+t*g), which is not simply related to (a*g).

Actually, ((1-t)*a+t*g) * ((1-t)*a+t*g) = (1-t)^2 * (a*a) + 2t(1-t)*(a*g) + t^2 * (g*g)

So the blended autocorrelation has three terms. The LP constrains (a*g) <= M, but (g*g) is not constrained. If (g*g) has a large peak, the blend might not help.

This is a known limitation of the LP surrogate. The parent program handles it with the line search (trying small t if the blend doesn't help).

            a = np.asarray(cand, dtype=np.float64).ravel()

            a = np.clip(a, 0.0, 1000.0)

        """Fast internal score."""
        a = np.clip(a, 0.0, 1000.0)
        n = a.size
        s = float(a.sum())
        if s < 1e-300:
            return float('inf')
        bm = float(np.max(np.convolve(a, a)))
        return 2.0 * n * bm / (s * s)

    def convmat(f):
        """Convolution matrix: (A @ g) = np.convolve(f, g)."""
        n = f.size
        A = np.zeros((2 * n - 1, n), dtype=np.float64)
        for j in range(n):
            A[j:j + n, j] = f
        return A

    def lpdir(f, use_prof=False):

        n = f.size
        b = np.convolve(f, f)
        M = float(b.max())
        if M < 1e-300:
            return None
        A = convmat(f)
        if use_prof:
            ub = np.maximum(b * 1.0001, M * 0.005)
        else:

                -np.ones(n, dtype=np.float64),
                A_ub=A, b_ub=ub,
                bounds=[(0.0, None)] * n,
                method='highs',
                options={'presolve': True, 'time_limit': 5}
            )
            if r.success and r.x is not None:
