def _lp_direction(a, M, eps=0.02):
    n = len(a)
    if n <= 1 or n > 512:

        j = np.arange(n)[None, :]
        d = k - j
        valid = (d >= 0) & (d < n)
        rows, cols = np.where(valid)
        dflat = d[valid].astype(np.int64)

    rows, cols, dflat = _lp_cache[key]

    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    A[rows, cols] = a[dflat]

    ncon = 2 * n - 1
    Afull = np.zeros((ncon + n, n + 1), dtype=np.float64)
    Afull[:ncon, :n] = A

    idx = np.arange(n)
    Afull[ncon + idx, idx] = 1.0
    Afull[ncon + idx, n] = -1.0

    c[:n] = -1.0
    c[n] = float(eps)

    try:
        res = linprog(
            c,
            A_ub=Afull,
            b_ub=b,
            bounds=[(0.0, None)] * (n + 1),
            method="highs",
            options={"presolve": True},

    except Exception:
        try:
            res = linprog(

        return None

    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(np.sum(x))
    if (not np.isfinite(s)) or s <= 1e-12:
        return None
    return x / s

        for amp, sgn in [(0.4, 1.0), (0.4, -1.0), (0.7, 1.0), (0.7, -1.0)]:
            m = 1.0 + amp * sgn * (2.0 * x)
            out.append(_normalize(np.maximum(a * m, 1e-9)))

        i = np.arange(n)
        modes = min(6, max(1, n // 2))
        for _ in range(4):
            base = np.ones(n, dtype=np.float64)
            for mode in range(1, modes + 1):
                if rng.random() < 0.7:
                    amp = rng.uniform(0.05, 0.45)
                    ph = rng.uniform(0.0, 2.0 * math.pi)
                    base += amp * np.cos(2.0 * math.pi * mode * i / float(n) + ph)
                if rng.random() < 0.5:
                    amp = rng.uniform(0.05, 0.35)
                    ph = rng.uniform(0.0, 2.0 * math.pi)
                    base += amp * np.sin(2.0 * math.pi * mode * i / float(n) + ph)
            base = np.maximum(base, 0.05)
            out.append(_normalize(base))
            out.append(_normalize(a * base))

    return out[:max_count]


def _stochastic_polish(a0, time_limit, rng, max_iters=200000):
    a = _normalize(a0)
    n = len(a)
    best = a.copy()
    best_sc = _fast_score(a)
    tstart = time.monotonic()
    tlim = tstart + max(0.0, float(time_limit))
    if n <= 1 or (tlim - tstart) <= 1.0:
        return best, best_sc

    meanv = 1.0 / float(n)
    sigma = 0.01
    stale = 0
    it = 0
    i_base = np.arange(n)

    while time.monotonic() < tlim and it < int(max_iters):
        it += 1
        a = best.copy()
        r = rng.random()
        improved = False

        if r < 0.18:
            c = _conv_self(best)
            if c.size:
                cm = float(np.max(c))
                if cm > 0.0 and np.isfinite(cm):
                    scale = max(10.0, 50.0 * float(n))
                    e = np.exp(scale * (c - cm))
                    se = float(np.sum(e))
                    if se > 0.0 and np.isfinite(se):
                        p = e / se
                        g = _grad_self(best, p)
                        gn = float(np.linalg.norm(g))
                        if gn > 1e-30:
                            for alpha in (0.05, 0.02, 0.005, 0.001):
                                cand = _normalize(best - (alpha / gn) * g)
                                sc = _fast_score(cand)
                                if sc < best_sc:
                                    best = cand
                                    best_sc = sc
                                    improved = True
                                    break
            if improved:
                stale = 0
                sigma = max(0.002, sigma * 0.97)
            continue

        if r < 0.45:
            loga = np.log(np.maximum(a, 1e-300))
            loga = loga + sigma * rng.normal(size=n)
            loga -= loga.max()
            cand = np.exp(loga)
            s = float(np.sum(cand))
            cand = cand / s if s > 0.0 else a
        elif r < 0.65:
            seg_lo = max(2, min(n, n // 8))
            seg_hi = max(seg_lo + 1, min(n, n // 2 + 2))
            seg = int(rng.integers(seg_lo, seg_hi + 1))
            seg = max(2, min(n, seg))
            lo = int(rng.integers(0, max(1, n - seg + 1)))
            cand = a.copy()
            cand[lo:lo + seg] *= np.exp(sigma * rng.normal(size=seg))
            cand = np.maximum(cand, 0.0)
            s = float(np.sum(cand))
            cand = cand / s if s > 0.0 else a
        elif r < 0.75:
            ii = int(rng.integers(0, n))
            jj = int(rng.integers(0, n))
            if ii == jj:
                jj = (jj + 1) % n
            step = sigma * meanv
            cand = a.copy()
            cand[ii] += step
            cand[jj] -= step
            cand = np.maximum(cand, 0.0)
            s = float(np.sum(cand))
            cand = cand / s if s > 0.0 else a
        elif r < 0.85:
            base = np.ones(n, dtype=np.float64)
            modes = min(5, max(1, n // 2))
            for mode in range(1, modes + 1):
                amp = sigma * rng.uniform(0.5, 2.0)
                ph = rng.uniform(0.0, 2.0 * math.pi)
                base += amp * np.cos(2.0 * math.pi * mode * i_base / float(n) + ph)
                amp = sigma * rng.uniform(0.5, 2.0)
                ph = rng.uniform(0.0, 2.0 * math.pi)
                base += amp * np.sin(2.0 * math.pi * mode * i_base / float(n) + ph)
            base = np.maximum(base, 0.1)
            cand = a * base
            s = float(np.sum(cand))
            cand = cand / s if s > 0.0 else a
        else:
            cand = 0.5 * a + 0.5 * a[::-1]
            cand = cand + (sigma * meanv) * rng.normal(size=n)
            cand = np.maximum(cand, 0.0)
            s = float(np.sum(cand))
            cand = cand / s if s > 0.0 else a

        cand = _normalize(cand)
        sc = _fast_score(cand)
        if sc < best_sc:
            best = cand
            best_sc = sc
            stale = 0
            sigma = max(0.002, sigma * 0.97)
        else:
            stale += 1
            if stale % 120 == 0:
                sigma = min(0.10, sigma * 1.15)
            if stale > 8000:
                break

    return _normalize(best), _fast_score(best)


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1300.0
    if budget <= 2.0:
        return [1.0]

    hard_finish = t0 + max(2.0, budget) - 1.0
    headroom = min(35.0, max(3.0, 0.035 * max(1.0, budget)))
    deadline = t0 + max(2.0, budget) - headroom
    if deadline <= t0 + 1.0:
        deadline = t0 + max(2.0, budget) - 0.5

    parent = _get_parent()
    if parent is None or parent.size == 0:
        n = 192
        x = np.linspace(-0.5, 0.5, n)
        p = 1.0 + 0.35 * np.cos(2.0 * math.pi * 3.0 * x) + 0.25 * np.sin(2.0 * math.pi * 5.0 * x)
        p = np.maximum(p, 0.05)
        parent = p / float(np.sum(p))

    best = _normalize(parent)
    best_sc = _fast_score(best)
    rng = np.random.default_rng(seed)
    pool = []

    def _add_pool(arr):
        if arr is None:
            return
        arr = _normalize(arr)
        if len(pool) < 120:
            pool.append(arr.copy())

    _add_pool(best)

    def _update(a, sc=None):
        nonlocal best, best_sc
        a = _normalize(a)
        if sc is None:
            sc = _fast_score(a)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best = a.copy()
        _add_pool(a)

    # Phase 1: local smoothing around parent and related variants.
    if time.monotonic() < deadline - 10.0:
        variants = _make_variants(best, rng, max_count=7)
        base_req = max(8.0, min(240.0, 0.28 * budget))
        lim = min(base_req, deadline - time.monotonic() - 8.0)
        if variants and lim > 4.0:
            ra, sa = _smooth_local(variants[0], lim, rng, noise=0.004)
            _update(ra, sa)

        var_req = max(4.0, min(45.0, 0.035 * budget))
        for v in variants[1:]:
            if time.monotonic() >= deadline - 25.0:
                break
            lim = min(var_req, deadline - time.monotonic() - 6.0)
            if lim < 3.0:
                break
            ra, sa = _smooth_local(v, lim, rng, noise=0.010)
            _update(ra, sa)

    # Phase 2: LP-style polish at modest lengths.
    if best.size <= 512 and time.monotonic() < deadline - 20.0:
        lim = min(80.0, 0.055 * budget, deadline - time.monotonic() - 12.0)
        if lim > 8.0:
            ra, sa = _lp_polish(best, lim, rng, eps=0.02, max_iter=35)
            _update(ra, sa)

    # Phase 3: length exploration by smooth resampling.
    if time.monotonic() < deadline - 50.0:
        base_len = int(best.size)
        lens = []
        for L in [64, 96, 128, 160, base_len, 192, 224, 256, 320, 384, 512, 640, 768, 1024, 1280, 2048, 3584, 4096]:
            if 8 <= L <= 4096:
                if not lens or L != lens[-1]:
                    lens.append(L)

        high_best = None
        high_sc = math.inf
        for L in lens:
            if time.monotonic() >= deadline - 25.0:
                break
            arr = _resample(best, L)
            sc = _fast_score(arr)
            _add_pool(arr)
            if np.isfinite(sc) and sc < best_sc:
                best = arr.copy()
                best_sc = sc
            if L != base_len and sc < high_sc:
                high_sc = sc
                high_best = arr.copy()

        if high_best is not None and high_best.size <= 2048 and time.monotonic() < deadline - 50.0:
            lim = min(200.0, 0.18 * budget, deadline - time.monotonic() - 20.0)
            if lim > 12.0:
                ra, sa = _smooth_local(high_best, lim, rng, noise=0.003)
                _update(ra, sa)

    # Phase 4: another smooth pass on the current best.
    if time.monotonic() < deadline - 35.0:
        lim = min(160.0, 0.12 * budget, deadline - time.monotonic() - 15.0)
        if lim > 10.0:
            ra, sa = _smooth_local(best, lim, rng, noise=0.007)
            _update(ra, sa)

    # Phase 5: broad local stochastic polishing with time to spare.
    if time.monotonic() < deadline - 20.0:
        lim = max(0.0, deadline - time.monotonic() - 8.0)
        if lim > 5.0:
            ra, sa = _stochastic_polish(best, lim, rng)
            _update(ra, sa)

    # Final exact selection among the short list of candidates.
    cands = list(pool)
    cands.append(best.copy())
    scored = []
    for arr in cands:
        arr = _normalize(arr)
        if arr.size == 0:
            continue
        sc = _fast_score(arr)
        if np.isfinite(sc):
            scored.append((sc, arr))

    if scored:
        scored.sort(key=lambda z: z[0])
        exact_best = None
        exact_best_sc = math.inf

        max_n = max(a.size for _, a in scored[:12])
        if max_n <= 256:
            topk = min(10, len(scored))
        elif max_n <= 512:
            topk = min(6, len(scored))
        elif max_n <= 1024:
            topk = min(4, len(scored))
        elif max_n <= 2048:
            topk = min(3, len(scored))
        else:
            topk = min(2, len(scored))

        for _, arr in scored[:topk]:
            if time.monotonic() > hard_finish - 0.6:
                break
            sc = _exact_eval(arr)
            if np.isfinite(sc) and sc < exact_best_sc:
                exact_best_sc = sc
                exact_best = arr.copy()

        if exact_best is None:
            sc = _exact_eval(best)
            if np.isfinite(sc):
                exact_best = best.copy()
                exact_best_sc = sc
    else:
        exact_best = None
        exact_best_sc = math.inf

    if exact_best is None:
        fb = np.ones(32, dtype=np.float64)
        sc = _exact_eval(fb)
        if np.isfinite(sc):
            exact_best = fb
            exact_best_sc = sc
        else:
            return [1.0]

    out = _normalize(exact_best)
    try:
        out_seq = out.astype(float).tolist()
        final_sc = float(evaluate_sequence(out_seq))
    except Exception:
        out_seq = [1.0]
        final_sc = math.inf

    if (not np.isfinite(final_sc)) or len(out_seq) == 0:
        fb = np.ones(32, dtype=np.float64).astype(float).tolist()
        try:
            fs = float(evaluate_sequence(fb))
        except Exception:
            fs = math.inf
        if np.isfinite(fs):
            return fb
        return [1.0]

    return out_seq
