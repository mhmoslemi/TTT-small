import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
import cvxpy as cp


def _softmax(x):
    x = np.asarray(x, dtype=np.float64).ravel()
    if x.size == 0:
        return np.array([1.0])
    x = x - np.max(x)
    e = np.exp(np.clip(x, -80.0, 80.0))
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full(x.size, 1.0 / x.size, dtype=np.float64)
    return e / s


def _conv(a, n, fs):
    F = rfft(a, fs)
    return irfft(F * F, fs)[: (2 * n - 1)]


def _cc(p, a, n, fs2):
    ar = a[::-1]
    c = irfft(rfft(ar, fs2) * rfft(p, fs2), fs2)
    return c[n - 1 : 2 * n - 1]


def _score(a, n, fs):
    b = _conv(a, n, fs)
    m = float(b.max())
    s = float(a.sum())
    if s < 0.01 or not np.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _compute_grad(a, n, fs_c, fs2, tau, w):
    b = _conv(a, n, fs_c)
    bmax = float(b.max())
    if bmax <= 0 or not np.isfinite(bmax):
        return None, bmax
    tau = max(1e-10, tau)
    bt = np.clip(b / tau, -80.0, 80.0)
    e = np.exp(bt)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0:
        p = np.zeros_like(b)
        p[int(np.argmax(b))] = 1.0
    else:
        p = e / s
    grad = 2.0 * _cc(p, a, n, fs2)
    if w > 0.0:
        mean_b = float(b.sum()) / (2 * n - 1.0)
        q = b - mean_b
        vg = 4.0 * _cc(q, a, n, fs2)
        grad = grad + w * vg
    return grad, bmax


def _adam_run(z, n, fs_c, fs2, end_time, hard_deadline, seed, tau_start=50.0,

    z = z.copy()
    best_z = z.copy()
    a0 = _softmax(z)
    fs_init = next_fast_len(2 * n - 1)
    b0 = _conv(a0, n, fs_init)
    best_score = 2.0 * n * max(float(b0.max()), 0.0) / max(1e-300, float(a0.sum())**2)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    st = 0
    it = 0
    tau = tau_start
    tau_decay = 0.995
    tau_frac = 0.5

    while time.monotonic() < min(end_time, hard_deadline) - 0.3:
        a = _softmax(np.clip(z, -30.0, 30.0))
        s_a = float(a.sum())
        if s_a < 0.01:
            break
        # normalize a for computation
        an = a / s_a
        n_actual = n

        grad, bmax = _compute_grad(an, n_actual, fs_c, fs2, tau, w)
        if grad is None:
            break

        cur_score = 2.0 * n_actual * bmax / (s_a * s_a)

        gn = float(np.abs(grad).max()) if grad.size else 0.0
        if not np.isfinite(gn) or gn < 1e-15:

                m = np.zeros(n)

        # Log-space gradient: a * (grad - dot(a, grad))

def _lp_refine(f, n, end_time, hard_deadline, max_iters=40):
    fs = next_fast_len(2 * n - 1)
    best = f.copy()
    best_score = _score(best, n, fs)
    fs2 = next_fast_len(3 * n - 2)

    # Pre-build A matrix: A[k,j] = f[k-j] if 0<=k-j<n
    # A shape: (2n-1, n)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f

    blend = 0.25
    for it in range(max_iters):
        if time.monotonic() > min(end_time, hard_deadline) - 1.0:
            break
        b = _conv(best, n, fs)
        M = float(b.max())
        if not np.isfinite(M) or M <= 0:
            break

        # Update A to use current best
        for j in range(n):
            A[j:j + n, j] = best

        g_var = cp.Variable(n)
        constraints = [g_var >= 0, A @ g_var <= M]
        obj = cp.Maximize(cp.sum(g_var))
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.CBC, time_limit=8)
        except Exception:
            try:
                prob.solve(solver=cp.GLPK)
            except Exception:
                try:
                    prob.solve(solver=cp.PDLPS)
                except Exception:
                    break

        if g_var.value is None or not np.isfinite(g_var.value).all():
            break

        g_val = np.clip(g_var.value, 0, None)
        sg = float(g_val.sum())
        if sg < 1e-8:
            break
        g_rescaled = g_val / sg * float(best.sum())

        # Try several blend factors
        improved = False
        for t_blend in [blend, blend * 0.5, blend * 2.0]:
            new_f = (1.0 - t_blend) * best + t_blend * g_rescaled
            new_f = np.clip(new_f, 1e-10, None)
            nscore = _score(new_f, n, fs)
            if np.isfinite(nscore) and nscore < best_score - 1e-12:
                best = new_f
                best_score = nscore
                improved = True
                break
        if not improved:
            # slight decrease blend
            blend = max(0.05, blend * 0.9)

    return best, best_score

            zc = z + rng.normal(0.0, sigma * 0.3, n)

        zc = zc - zc.mean()
        a2 = _softmax(zc)
        s2 = _score(a2, n, fs)
        if np.isfinite(s2) and s2 < bs - 1e-11:
            best = a2.copy()
            bs = s2
            stuck = 0
            sigma = min(sigma, 0.35)
        else:
            stuck += 1
            sigma = min(0.8, sigma * 1.03)
            if stuck > 80:
                stuck = 0
                sigma = 0.4

    return best, bs


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    total = max(0.0, float(budget_s))
    hard_deadline = t0 + max(0.0, total) - 5.0

    seed = int(seed)
    rng_global = np.random.default_rng(seed)

    # Get parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            if p.size > 0 and np.isfinite(p).all() and float(p.sum()) > 0.01:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    # Initialize best
    best_a = None
    best_score = math.inf

    if parent is not None and parent.size >= 2:
        pa = parent / max(float(parent.sum()), 1e-300)
        pa = np.clip(pa, 0, 1000.0)
        try:
            sc = float(evaluate_sequence([float(v) for v in pa]))
            if np.isfinite(sc):
                best_a = pa.copy()
                best_score = sc
        except Exception:
            n_p = len(pa)
            fs_p = next_fast_len(2 * n_p - 1)
            best_score = _score(pa, n_p, fs_p)
            if np.isfinite(best_score):
                best_a = pa.copy()

    if best_a is None:
        best_a = np.ones(32) / 32.0
        best_score = 2.0

    def _try_update(a, s):
        nonlocal best_a, best_score
        if a is None or s is None:
            return
        try:
            a = np.asarray(a, dtype=np.float64).ravel()
            if a.size == 0 or not np.isfinite(a).all():
                return
            a = np.clip(a, 0.0, 1000.0)
            if float(a.sum()) < 0.01:
                return
            if np.isfinite(s) and s < best_score - 1e-10:
                best_a = a.copy()
                best_score = float(s)
        except Exception:
            pass

    # Budget allocation
    n_parent = best_a.size if best_a is not None else 192
    lp_end = t0 + min(60.0, 0.08 * total)
    adam1_end = t0 + min(0.55 * total, 700.0)
    adam2_end = t0 + min(0.75 * total, 950.0)
    polish_end = t0 + max(0.0, total) - 15.0

    try:
        # ===== Phase 1: LP-based refinement at parent length =====
        if parent is not None and parent.size >= 8 and time.monotonic() < hard_deadline - 30:
            n_lp = min(parent.size, 384)  # cap for LP feasibility
            a_lp = parent[:n_lp] / max(float(parent[:n_lp].sum()), 1e-300)
            a_lp, s_lp = _lp_refine(a_lp, n_lp, lp_end, hard_deadline, max_iters=30)
            _try_update(a_lp, s_lp)

    except Exception:
        pass

    try:
        # ===== Phase 2: Adam optimization at parent length =====
        if time.monotonic() < adam1_end - 5:
            n_opt = n_parent
            fs_c = next_fast_len(2 * n_opt - 1)
            fs2 = next_fast_len(3 * n_opt - 2)

            a_start = best_a.copy()
            a_start = np.clip(a_start, 1e-10, 1000.0)
            a_start = a_start / max(float(a_start.sum()), 1e-300)
            z_start = np.log(a_start)
            z_start = z_start - z_start.mean()

            # Multiple Adam runs with different params
            for run_idx in range(3):
                if time.monotonic() >= adam1_end - 5:
                    break
                w_val = [0.0, 0.1, 0.2][run_idx]
                tau_s = [40.0, 20.0, 10.0][run_idx]
                lr_val = [0.5, 0.4, 0.35][run_idx]
                seed_r = seed * 100 + run_idx * 77 + 13
                rem = adam1_end - time.monotonic()
                if rem < 10:
                    break
                a_r, s_r, z_r = _adam_run(z_start, n_opt, fs_c, fs2,
                                          adam1_end - run_idx * 30, hard_deadline,
                                          seed_r, tau_start=tau_s, lr=lr_val, w=w_val)
                _try_update(a_r, s_r)
                if np.isfinite(s_r) and s_r < best_score:
                    a_start = a_r.copy()
                    a_start = a_start / max(float(a_start.sum()), 1e-300)
                    z_start = np.log(np.maximum(a_start, 1e-12))
                    z_start = z_start - z_start.mean()

    except Exception:
        pass

    try:
        # ===== Phase 3: Try slightly longer sequences =====
        if time.monotonic() < adam2_end - 10 and n_parent >= 64:
            for n_try in [n_parent + 32, n_parent + 64, n_parent * 2]:
                if time.monotonic() >= adam2_end - 20:
                    break
                if n_try > 1024:
                    break
                # Interpolate parent to new length
                if parent is not None and parent.size > 1:
                    npad = np.arange(n_try) * parent.size / n_try
                    idx0 = np.floor(npad).astype(int)
                    idx1 = np.minimum(idx0 + 1, parent.size - 1)
                    frac = npad - idx0
                    a_interp = parent[idx0] * (1 - frac) + parent[idx1] * frac
                else:
                    t_arr = (np.arange(n_try) + 0.5) / n_try
                    a_interp = 1.0 - np.abs(2 * t_arr - 1)
                    a_interp = a_interp + 0.1 * np.random.default_rng(seed + n_try).normal(0, 0.1, n_try)
                    a_interp = np.maximum(a_interp, 0.01)
                a_interp = a_interp / max(float(a_interp.sum()), 1e-300)
                fs_c2 = next_fast_len(2 * n_try - 1)
                fs2_2 = next_fast_len(3 * n_try - 2)
                z_try = np.log(np.maximum(a_interp, 1e-12))
                z_try = z_try - z_try.mean()
                rem2 = adam2_end - time.monotonic()
                if rem2 < 15:
                    break
                a_t2, s_t2, _ = _adam_run(z_try, n_try, fs_c2, fs2_2,
                                          adam2_end - 10, hard_deadline,
                                          seed * 200 + n_try, tau_start=20.0, lr=0.45, w=0.05)
                _try_update(a_t2, s_t2)

    except Exception:
        pass

    try:
        # ===== Phase 4: Polish =====
        if time.monotonic() < polish_end - 3:
            a_pol = best_a.copy()
            a_pol = np.clip(a_pol, 1e-12, 1000.0)
            a_pol = a_pol / max(float(a_pol.sum()), 1e-300)
            a_pol, s_pol = _polish(a_pol, a_pol.size, polish_end, hard_deadline, seed + 54321)
            _try_update(a_pol, s_pol)

    except Exception:
        pass

    # ===== Final validation =====
    try:
        cand = np.asarray(best_a, dtype=np.float64).ravel()
        cand = np.clip(cand, 0.0, 1000.0)
        if cand.size == 0 or float(cand.sum()) < 0.01:
            cand = np.ones(32) / 32.0
        cand = cand / max(float(cand.sum()), 1e-300) * max(float(cand.sum()), 1.0)
        # Actually keep it as-is, score is scale invariant
        cand_list = [float(v) for v in cand]
        fs_final = float(evaluate_sequence(cand_list))
        if not np.isfinite(fs_final) or fs_final > best_score + 1.0:
            cand_list = [1.0]
            fs_final = float(evaluate_sequence(cand_list))
            if not np.isfinite(fs_final):
                return [1.0]
    except Exception:
        cand_list = [1.0]

    return cand_list
