            if b_t > 0:
                lo = max(0.0, b_t - 0.1)
                hi = min(1.0, b_t + 0.1)
                for _ in range(15):
                    t1 = lo + (hi - lo) * (3 - math.sqrt(5)) / 2
                    t2 = lo + (hi - lo) * (sqrt5m3 = (math.sqrt(5) - 1) / 2)
                    # Actually just do a few more points
                    for t in np.linspace(lo, hi, 7):
                        cand = norm((1.0 - t) * a + t * g)
                        sv = sc(cand)
                        if math.isfinite(sv) and sv < b_val:
                            b_val, b_t = sv, t
            if b_t > 0 and b_val < loc * (1.0 - 1e-12):
                a = norm((1.0 - b_t) * a + b_t * g)
                loc = b_val
                stagnation = 0
                upd(a)
                if loc < best_local[0]:
                    best_local[0] = loc
                    best_local[1] = a.copy()
            else:
                stagnation += 1
                if stagnation % 10 == 0:
                    a = norm(a * np.exp(0.08 * rng.standard_normal(n)))
                    new_loc = sc(a)
                    if math.isfinite(new_loc) and new_loc < loc:
                        loc = new_loc
                        stagnation = 0
                        upd(a)
                        if loc < best_local[0]:
                            best_local[0] = loc
                            best_local[1] = a.copy()
                if stagnation > 60:
                    break
        return a, loc

    # Smooth-max gradient descent
    def grad_refine(a0, stop_time, best_local, tau0=50.0):
        a = norm(a0)
        n = a.size
        loc = sc(a)
        upd(a)
        if loc < best_local[0]:
            best_local[0] = loc
            best_local[1] = a.copy()
        tau = tau0
        stale = 0
        for it in range(5000):
            if time.monotonic() > stop_time:
                break
            b = np.convolve(a, a)
            # Softmax weights
            b_shifted = b - b.max()
            tb = tau * b_shifted
            p = np.exp(tb)
            ps = p.sum()
            if ps < 1e-300 or not math.isfinite(ps):
                p = np.zeros_like(b)
                p[b.argmax()] = 1.0
            else:
                p = p / ps
            # Gradient: 2 * sum_k p_k * reflect_shift(a, k)
            # (grad F)_j = 2 * sum_k p_k * a[k-j]
            # This is: 2 * convolve(p_within_range, a) ... let's compute directly
            grad = np.zeros(n, dtype=np.float64)
            for j in range(n):
                # sum over k where 0 <= k-j < 2n-1 and 0 <= k-j < n
                k_lo = max(j, 0)
                k_hi = min(j + n - 1, 2 * n - 2)
                if k_lo <= k_hi:
                    grad[j] = 2.0 * np.dot(p[k_lo:k_hi + 1], a[k_hi - j: -1 if k_lo >= k_hi else k_lo - j][-n:])
            # Simpler: grad[j] = 2 * sum_{k=j}^{j+n-1} p[k] * a[k-j] for valid k
            grad = np.zeros(n, dtype=np.float64)
            for j in range(n):
                ks = np.arange(j, min(j + n, 2 * n - 1))
                js = ks - j  # indices into a: 0, 1, ..., len(ks)-1
                grad[j] = 2.0 * np.dot(p[ks], a[js])

            gn = np.linalg.norm(grad)
            if not math.isfinite(gn) or gn < 1e-15:
                break

            # Normalized gradient step
            # We want to minimize F subject to sum(a)=1, a>=0
            # Projected gradient: a_new = proj_simplex(a - step * grad / gn)
            # But we also need to handle the constraint properly
            # Actually since we normalize (divide by sum), we can do:
            # a_candidate = proj_simplex(a - lr * grad)
            step_sizes = np.array([0.01, 0.03, 0.1, 0.3, 1.0, 3.0, 10.0])
            best_step = 0.0
            best_sv = loc
            for s in step_sizes:
                cand = proj_simplex(a - s * grad / (n * gn))
                sv = sc(cand)
                if math.isfinite(sv) and sv < best_sv:
                    best_sv = best_step_sv = sv
                    best_step = s
            if best_step > 0 and best_sv < loc * (1.0 - 1e-14):
                a = norm(proj_simplex(a - best_step * grad / (n * gn)))
                loc = sc(a)
                stale = 0
                upd(a)
                if loc < best_local[0]:
                    best_local[0] = loc
                    best_local[1] = a.copy()
            else:
                stale += 1
                tau = max(10.0, tau * 0.95)
                if stale % 8 == 0:
                    a = norm(a * np.exp(0.04 * rng.standard_normal(n)))
                    nl = sc(a)
                    if math.isfinite(nl) and nl < loc * 1.001:
                        loc = nl
                        stale = 0
                        upd(a)
                        if loc < best_local[0]:
                            best_local[0] = loc
                            best_local[1] = a.copy()
                if stale > 100:
                    break
        return a, loc

    # Generate diverse initializations
    def gen_starts(n, par_arr, count=12):
        starts = []
        x = np.linspace(0.0, 1.0, n)

        def add(v):
            v = np.asarray(v, dtype=np.float64).ravel()
            v = np.nan_to_num(v, nan=0.0, posinf=0.0, neginf=0.0)
            v = np.clip(v, 0.0, None)
            if v.size == n and v.sum() > 1e-300:
                starts.append(v / v.sum())

        # Uniform
        add(np.ones(n))
        # Exponential decays
        for rate in (0.3, 0.7, 1.0, 1.5, 2.0, 3.0, 5.0, 8.0):
            add(np.exp(-rate * x))
            add(np.exp(-rate * (1 - x)))
        # Power law
        for p in (0.3, 0.5, 0.7, 1.0, 1.5):
            add(1.0 - p * 0.8 * x)
            add(1.0 - p * 0.8 * (1 - x))
        # Sine perturbations
        for freq in (1, 2, 3, 5):
            for amp in (0.2, 0.4, 0.6):
                add(1.0 + amp * np.sin(2 * np.pi * freq * x))
                add(1.0 - amp * np.sin(2 * np.pi * freq * x))
        # Parent interpolated
        if par_arr is not None and par_arr.size >= 2:
            add(np.interp(x, np.linspace(0, 1, par_arr.size), par_arr))
            for noise in (0.05, 0.1, 0.2):
                add(np.interp(x, np.linspace(0, 1, par_arr.size), par_arr) *
                    np.exp(noise * rng.standard_normal(n)))
        # Random
        for _ in range(4):
            add(rng.dirichlet(np.full(n, rng.choice([0.5, 1.0, 2.0, 5.0]))))
        # Sort by score
        starts.sort(key=lambda v: sc(v))
        return starts[:count]

    # ============ MAIN OPTIMIZATION ============

    # Phase 1: LP on parent length with multiple starts
    n_work = n0
    bl = [float('inf'), None]
    starts = gen_starts(n_work, par, count=10)
    for i, s in enumerate(starts[:6]):
        if time.monotonic() >= deadline - 300.0:
            break
        alloc = min(120.0, (deadline - 300.0 - time.monotonic()) * 0.3)
        if alloc < 15.0:
            break
        _, _ = lp_iterate(s, time.monotonic() + alloc, bl)

    # Phase 2: Gradient refinement on best found
    if time.monotonic() < deadline - 200.0:
        cur = bl[1] if bl[1] is not None else best_seq
        alloc = min(100.0, deadline - 200.0 - time.monotonic())
        if alloc > 20.0:
            _, _ = grad_refine(cur, time.monotonic() + alloc, bl, tau0=100.0)

    # Phase 3: More LP with fresh starts
    if time.monotonic() < deadline - 150.0:
        starts2 = gen_starts(n_work, bl[1] if bl[1] is not None else par, count=8)
        for i, s in enumerate(starts2[2:7]):
            if time.monotonic() >= deadline - 150.0:
                break
            alloc = min(80.0, (deadline - 150.0 - time.monotonic()) * 0.3)
            if alloc < 10.0:
                break
            _, _ = lp_iterate(s, time.monotonic() + alloc, bl)

    # Phase 4: Try n=128 (smaller, might find different structures)
    if time.monotonic() < deadline - 120.0:
        starts128 = gen_starts(128, par, count=6)
        for s in starts128[:4]:
            if time.monotonic() >= deadline - 120.0:
                break
            alloc = min(60.0, (deadline - 120.0 - time.monotonic()) * 0.3)
            if alloc < 10.0:
                break
            _, _ = lp_iterate(s, time.monotonic() + alloc, bl)

    # Phase 5: Try n=256 (larger)
    if time.monotonic() < deadline - 100.0:
        starts256 = gen_starts(256, par, count=4)
        for s in starts256[:3]:
            if time.monotonic() >= deadline - 100.0:
                break
            alloc = min(50.0, (deadline - 100.0 - time.monotonic()) * 0.3)
            if alloc < 10.0:
                break
            _, _ = lp_iterate(s, time.monotonic() + alloc, bl)

    # Phase 6: Gradient refinement on overall best
    if time.monotonic() < deadline - 60.0:
        cur = best_seq
        alloc = min(80.0, deadline - 60.0 - time.monotonic())
        if alloc > 20.0:
            _, _ = grad_refine(cur, time.monotonic() + alloc, bl, tau0=50.0)

    # Phase 7: Final LP polish on best
    if time.monotonic() < deadline - 30.0:
        cur = best_seq
        n_cur = cur.size
        alloc = min(60.0, deadline - 30.0 - time.monotonic())
        if alloc > 15.0:
            _, _ = lp_iterate(cur, time.monotonic() + alloc, bl)

    # Phase 8: One more gradient pass
    if time.monotonic() < deadline - 10.0:
        cur = best_seq
        alloc = min(30.0, deadline - 10.0 - time.monotonic())
        if alloc > 10.0:
            _, _ = grad_refine(cur, time.monotonic() + alloc, bl, tau0=20.0)

    # Final validation
    fin = np.asarray(best_seq, dtype=np.float64).ravel().copy()
    fin = np.nan_to_num(fin, nan=0.0, posinf=0.0, neginf=0.0)
    fin = np.clip(fin, 0.0, 1000.0)
    if fin.size < 1 or fin.sum() < 0.01:
        fin = np.ones(192, dtype=np.float64)
    upd(fin)
    fin = np.asarray(best_seq, dtype=np.float64).ravel()
    try:
        v = evaluate_sequence([float(x) for x in fin])
        if not math.isfinite(v):
            fin = np.ones(192, dtype=np.float64)
    except:
        fin = np.ones(192, dtype=np.float64)
    return [float(x) for x in fin]
