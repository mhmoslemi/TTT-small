```python
def _anderson_ext(F_vals, x_vals, m):
    """Compute Anderson extrapolation.
    F_vals: list of F(x_i) for last m+1 points
    x_vals: list of x_i for last m+1 points
    Returns: new x
    """
    F_curr = F_vals[-1]
    x_curr = x_vals[-1]
    
    m_eff = min(m, len(x_vals) - 1)
    if m_eff == 0:
        return F_curr
    
    # D_i = x_i - F(x_i) for i in last m_eff+1
    D = []
    for i in range(-m_eff-1, 0):
        D.append(x_vals[i] - F_vals[i])
    D = np.array(D)  # (m_eff+1, n)
    
    # Solve: min ||D @ omega||^2 s.t. sum(omega) = 1, omega >= 0 (optional)
    # Without nonnegativity: G = D @ D.T ( (m+1) x (m+1) ), solve G @ omega = 1 with equality constraint
    # Using the standard formula with Lagrange multiplier:
    # G @ omega + mu * 1 = 0, 1^T @ omega = 1
    G = D @ D.T
    G = np.maximum(G, 1e-10)  # Regularize diagonal
    # Actually, just solve the unconstrained problem with equality constraint
    # [G  1] [omega]   [0]
    # [1^T 0] [ mu  ] = [1]
    m1 = G.shape[0]
    A = np.zeros((m1+1, m1+1))
    A[:m1, :m1] = G
    A[:m1, m1] = 1.0
    A[m1, :m1] = 1.0
    b = np.zeros(m1+1)
    b[m1] = 1.0
    try:
        sol = np.linalg.solve(A, b)
        omega = sol[:m1]
    except np.linalg.LinAlgError:
        omega = np.ones(m1) / m1
    
    # x_new = F(x_curr) + D @ omega
    x_new = F_curr + D @ omega
    return x_new
```
