When we blend f and g, the new autocorrelation is:

We know (f*g) <= M, and (f*f) <= M (by definition of M). If (g*g) is also <= M (which we don't guarantee), then the blend would have max <= M. But (g*g) might exceed M.

                        sp_val = _score(fp)
                        if math.isfinite(sp_val) and sp_val < local:
                            v = vp
                            f = fp
                            local = sp_val
                            _update(f, best)
                    stale = 0
                    alpha = 0.04
                if stale > 40:
                    break

    # === MAIN EXECUTION ===

    # Phase 1: Heavy LP at parent length (320)
    if parent is not None and parent.size >= 8:
        np_ = parent.size
        alloc = min(500.0, _rem() * 0.5)
        if alloc > 30:
            _lp_iterate(parent.copy(), np_, alloc, rng)

    # Phase 2: LP at n=512 (interpolate parent)
    if _rem() > 80:
        n512 = 512
        a512 = np.ones(n512)
        if parent is not None:
            a512 = np.interp(np.linspace(0,1,n512), np.linspace(0,1,parent.size), parent)
        x = np.linspace(0, 1, n512)
        a512 = a512 * (1.0 + 0.4 * x)
        alloc = min(250.0, _rem() * 0.3)
        if alloc > 20:
            _lp_iterate(a512, n512, alloc, rng)

    # Phase 3: LP at n=320 from best-so-far
    if _rem() > 60:
        best_arr = best[1]
        n320 = 320
        a320 = best_arr
        if best_arr.size != n320:
            a320 = np.interp(np.linspace(0,1,n320), np.linspace(0,1,best_arr.size), best_arr)
        alloc = min(200.0, _rem() * 0.25)
        if alloc > 15:
            _lp_iterate(a320.copy(), n320, alloc, rng)

    # Phase 4: Anti-palindromic optimization
    if _rem() > 50:
        best_arr = best[1]
        alloc = min(150.0, _rem() * 0.3)
        if alloc > 20:
            _anti_pal_opt(best_arr, max(best_arr.size, 64), alloc, rng)

    # Phase 5: Final LP polish on the very best
    if _rem() > 20:
        best_arr = best[1]
        alloc = min(50.0, _rem() * 0.6)
        if alloc > 10:
            _lp_iterate(best_arr.copy(), best_arr.size, alloc, rng)

    # Final validation
    final = np.asarray(best[1], dtype=np.float64).ravel()
    final = np.nan_to_num(final, nan=0.0, posinf=0.0, neginf=0.0)
    final = np.clip(final, 0.0, 1000.0)
    if final.size == 0 or final.sum() < 0.01:
        final = np.ones(64, dtype=np.float64)

    fin_sc = evaluate_sequence([float(x) for x in final])
    if not math.isfinite(fin_sc) or fin_sc > best[0] + 0.005:
        final = np.ones(64, dtype=np.float64)
        fin_sc = evaluate_sequence([float(x) for x in final])

    return [float(x) for x in final]
