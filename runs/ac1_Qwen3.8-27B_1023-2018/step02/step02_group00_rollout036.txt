import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _fft_conv_a(a):
    """a * a (full linear convolution)."""
    n = len(a)
    size = next_fast_len(2 * n - 1)
    A = rfft(a, size)
    c = irfft(A * A, size)[:2 * n - 1]
    return c

def _fft_conv_ab(a, b):
    """a * b (full linear convolution)."""
    n = len(a)
    m = len(b)
    size = next_fast_len(n + m - 1)
    A = rfft(a, size)
    B = rfft(b, size)
    return irfft(A * B, size)[:n + m - 1]

def _score(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return math.inf
    c = _fft_conv_a(a)
    m = float(c.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)

def _lp_direction(f, n, M, max_cg=200):
    """LP direction via constraint generation."""
    c_ff = _fft_conv_a(f)
    S = [int(np.argmax(c_ff))]
    prev_margin = -1e18
    
    g = None
    for cg_it in range(max_cg):
        # Build constraint rows for active lags S
        A = np.zeros((len(S), n))
        for idx, k in enumerate(S):
            i_lo = max(0, k - n + 1)
            i_hi = min(k, n - 1)
            # A[idx, i] = f[k - i] for i in [i_lo, i_hi]
            # f[k - i_hi], f[k - i_hi + 1], ..., f[k - i_lo]
            A[idx, i_lo:i_hi + 1] = f[k - i_hi:k - i_lo + 1][::-1]
        
        b_ub = np.full(len(S), M)
        res = linprog(-np.ones(n), A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True})
        
        if res.status != 0 or res.x is None:
            return None
        
        g = np.asarray(res.x, dtype=np.float64)
        if g.sum() < 1e-15:
            return None
        
        # Check all constraints
        fg = _fft_conv_ab(f, g)
        max_viol = float(fg.max()) - M
        if max_viol < 1e-9:
            return g / g.sum()
        else:
            k_new = int(np.argmax(fg))
            if k_new not in S:
                S.append(k_new)
            else:
                # Already in S but violated? Shouldn't happen for LP feasible solution
                # Might be numerical. Add anyway if not duplicate.
                if len(S) < max_cg:
                    S.append(k_new)
                break
    
    if g is not None and g.sum() > 1e-15:
        return g / g.sum()
    return None

                f = np.clip(f + 0.1 * rng.standard_normal(n), 0, None)
                f = f / max(f.sum(), 1e-10)
                t = 0.25
                stall = 0
    
    return best_f, best_score


def _polish(a, stop_time, seed=99999):
    n = len(a)
    best = np.clip(np.asarray(a, dtype=np.float64).ravel(), 0, None).copy()
    fs = best.sum()
    if fs < 1e-10:
        return best, math.inf
    best = best / fs
    best_sc = _score(best)
    rng = np.random.default_rng(seed)
    no_imp = 0
    
    while time.monotonic() < stop_time - 5.0 and no_imp < 20:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 5.0:
                break
            orig = best[i]
            best_val = orig
            best_val_sc = best_sc
            for mlt in (0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0, 3.0, 0.0, 0.1, 4.0):
                val = orig * mlt if orig > 1e-14 else (0.01 * n if mlt > 0 else 0.0)
                best[i] = val
                sc = _score(best)
                if sc < best_val_sc - 1e-14:
                    best_val = val
                    best_val_sc = sc
            best[i] = best_val
            if best_val_sc < best_sc - 1e-14:
                best_sc = best_val_sc
                improved = True
        
        # 2-coordinate
        for _ in range(60):
            if time.monotonic() > stop_time - 5.0:
                break
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            for mi in (0.4, 1.5, 0.0):
                for mj in (0.4, 1.5, 0.0):
                    best[i] = oi * mi if oi > 1e-14 else (0.01 * n if mi > 0 else 0.0)
                    best[j] = oj * mj if oj > 1e-14 else (0.01 * n if mj > 0 else 0.0)
                    sc = _score(best)
                    if sc < best_sc - 1e-14:
                        best_sc = sc
                        improved = True
            best[i], best[j] = oi, oj
        
        if not improved:
            no_imp += 1
        else:
            no_imp = 0
    
    return best, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard_deadline = t0 + budget - 10.0  # 10s safety margin
    
    # Get parent
    try:
        parent = np.asarray(globals()["height_sequence_1"], dtype=np.float64).ravel()
    except Exception:
        parent = np.ones(320, dtype=np.float64)
    parent = np.clip(parent, 0, 1000)
    parent = np.where(np.isfinite(parent), parent, 0.0)
    if parent.size == 0:
        parent = np.ones(320)
    n_parent = len(parent)
    
    # Fallback
    fallback = parent.copy()
    fallback_sc = _score(fallback)
    if not math.isfinite(fallback_sc):
        fallback = np.ones(32) / 32
        fallback_sc = _score(fallback)
    
    best_seq = fallback.copy()
    best_score = fallback_sc
    
    def _update(res, sc):
        nonlocal best_seq, best_score
        if not np.all(np.isfinite(res)):
            return
        res = np.clip(res, 0, 1000)
        if res.sum() < 0.01:
            return
        if sc < best_score:
            try:
                exact = float(evaluate_sequence([float(x) for x in res]))
            except Exception:
                exact = math.inf
            if math.isfinite(exact) and exact < best_score:
                best_seq = res.copy()
                best_score = exact
    
    def _make_inits(n, seed_val):
        rng = np.random.default_rng(seed_val)
        inits = []
        # Parent interpolated
        if n_parent != n:
            inits.append(np.interp(np.linspace(0, 1, n), np.linspace(0, 1, n_parent), parent))
        else:
            inits.append(parent.copy())
        # Cosine window
        x = np.arange(n) / n
        inits.append(0.5 + 0.5 * np.cos(2 * math.pi * x - math.pi))
        # Hann
        inits.append(0.5 * (1 + np.cos(2 * math.pi * x - math.pi)))
        # Superposition of cosines
        sup = np.ones(n) * 0.5
        for k in range(1, 6):
            sup += 0.3 / k * np.cos(2 * math.pi * k * x + rng.uniform(0, 2 * math.pi))
        inits.append(np.clip(sup, 0.01, None))
        # Random
        inits.append(np.abs(rng.standard_normal(n)) + 0.1)
        # Best so far (if different length, interpolate)
        if len(best_seq) != n:
            inits.append(np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(best_seq)), best_seq))
        else:
            inits.append(best_seq.copy())
        return inits
    
    # Phase 1: LP on parent length, multiple inits
    n = n_parent
    phase1_end = t0 + budget * 0.45
    inits = _make_inits(n, seed)
    time_per = max(5.0, (phase1_end - t0) / len(inits))
    
    for idx, init in enumerate(inits):
        if time.monotonic() > phase1_end - 10.0:
            break
        stop = min(phase1_end - 5.0, time.monotonic() + time_per)
        if time.monotonic() > stop - 2.0:
            continue
        try:
            res, sc = _lp_optimize(init, n, stop, seed + 100 * (idx + 1))
            _update(res, sc)
        except Exception:
            pass
    
    # Phase 2: Other lengths
    phase2_end = t0 + budget * 0.70
    other_n = [192, 256, 320, 384, 448, 512]
    if n_parent not in other_n:
        other_n.append(n_parent)
    other_n = sorted(set(other_n))
    time_per_len = max(5.0, (phase2_end - phase1_end) / len(other_n))
    
    for L in other_n:
        if time.monotonic() > phase2_end - 10.0:
            break
        stop = min(phase2_end - 5.0, time.monotonic() + time_per_len)
        if time.monotonic() > stop - 2.0:
            continue
        init = _make_inits(L, seed + L)
        try:
            res, sc = _lp_optimize(init[0], L, stop, seed + 10000 + L)
            _update(res, sc)
        except Exception:
            pass
    
    # Phase 3: Polish
    polish_end = hard_deadline - 5.0
    if time.monotonic() < polish_end - 10.0:
        try:
            res, sc = _polish(best_seq, polish_end)
            _update(res, sc)
        except Exception:
            pass
    
    # Final validation
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf
    
    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in fallback]
    
    return [float(x) for x in best_seq]
