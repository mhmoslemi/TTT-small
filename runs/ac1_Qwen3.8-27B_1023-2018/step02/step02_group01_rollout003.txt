def _lp_solve(A,M,n,timeout=1.0):
   if M <= 0 or not np.isfinite(M): return None
   cvec = -np.ones(n)
   bvec = np.full(2*n-1, M, dtype=np.float64)
   bounds = [(0.0, 1e6)]*n
   try:
      opts={'time_limit': max(0.05, float(timeout)),
            'primal_feasibility_tolerance':1e-9,
            'dual_feasibility_tolerance':1e-9}
      res=linprog(cvec, A_ub=A, b_ub=bvec, bounds=bounds, method='highs', options=opts)
   except Exception:
      return None
   if res.status not in (0,) or res.x is None:
      return None
   g=np.asarray(res.x,dtype=np.float64)
   g=np.where(np.isfinite(g),g,0.0)
   g=np.clip(g,0.0,1e6)
   s=g.sum()
   if s <= 1e-12:
      return None

def _lambda_for_active(a, active):
   n=a.size; m=active.size
   D=np.zeros((n,m), dtype=np.float64)
   for idx,k in enumerate(active):
      k=int(k)
      lo=max(0,k-n+1); hi=min(n-1,k)
      length=hi-lo+1
      if length<=0: continue
      end=k-hi-1
      if end >= 0:
         vals=a[k-lo:end:-1]
      else:
         vals=a[k-lo::-1]
      # length should match slice length
      D[lo:hi+1, idx]=vals[:length]
   D_aug = np.column_stack((D, -np.ones(n, dtype=np.float64)))
   scale=float(np.linalg.norm(D_aug, ord=np.inf))
   if scale <= 0 or not np.isfinite(scale): return None
   D_aug = D_aug / scale
   E=np.zeros((1,m+1), dtype=np.float64)
   E[0,:m]=1.0
   beta=20.0
   A=np.vstack((D_aug, beta*E))
   bvec=np.concatenate((np.zeros(n, dtype=np.float64), np.array([beta], dtype=np.float64)))
   try:
       y, *_ = np.linalg.lstsq(A, bvec, rcond=None)
   except Exception:
       return None
   lam=np.clip(np.asarray(y[:m], dtype=np.float64), 0.0, 1e6)
   s=lam.sum()
   if s <= 1e-10:
       return np.ones(m)/m
   lam /= s
   # blend with uniform to avoid pathological sparse
   lam = 0.85*lam + 0.15*(1.0/m)
   lam /= lam.sum()
   return lam

def _normalize_pos(a, use_floor=True):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    n = a.size
    if s <= 1e-300:
        a = np.ones(n, dtype=np.float64)
        s = float(n)
    a = a / s
    if use_floor:
        floor = 1e-14 / max(1, n)
        if (a < floor).any():
            a = np.maximum(a, floor)
            a = a / a.sum()
    return a


def _conv_square(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([0.0], dtype=np.float64)
    if n <= 64:
        c = np.convolve(a, a)
    else:
        L = next_fast_len(2 * n - 1)
        F = rfft(a, L)
        c = irfft(F * F, L)[: 2 * n - 1]
    c = np.asarray(c, dtype=np.float64)
    return np.maximum(c, 0.0)


def _unit_score(a):
    c = _conv_square(a)
    if c.size == 0:
        return float("inf")
    return float(2.0 * a.size * c.max())

def _build_A(a):
    n = int(a.size)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for k in range(n):
        A[k, 0 : k + 1] = a[k :: -1]
    for k in range(n, m):
        l = k - n + 1
        if l < n:
            A[k, l:n] = a[n - 1 : k - n : -1]
    return A


def _lp_solve(A, M, n, timeout=1.0):
    if M <= 0.0 or not np.isfinite(M):
        return None
    cvec = -np.ones(n, dtype=np.float64)
    bvec = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0.0, 1.0e6)] * n
    opts = {
        "time_limit": max(0.05, float(timeout)),
        "primal_feasibility_tolerance": 1e-9,
        "dual_feasibility_tolerance": 1e-9,
    }
    try:
        res = linprog(cvec, A_ub=A, b_ub=bvec, bounds=bounds, method="highs", options=opts)
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.where(np.isfinite(g), g, 0.0)
    g = np.clip(g, 0.0, 1.0e6)
    s = float(g.sum())
    if s <= 1e-12:
        return None
    return g / s


def _lp_step(a, M, cur_score, time_left, budget):
    n = int(a.size)
    if n < 2:
        return None
    if time_left < 1.5 or budget < 0.2:
        return None
    try:
        A = _build_A(a)
    except Exception:
        return None

    base = M * (1.0 + 1.0e-7)
    tries = [base]
    if budget > 0.7:
        tries.append(base * (1.0 + 1.0e-5))
    if budget > 1.3:
        tries.append(base * (1.0 + 5.0e-4))

    per = max(0.05, budget / max(1, len(tries)))
    start = time.monotonic()
    for Mt in tries:
        if time.monotonic() > time_left - 1.0:
            break
        if time.monotonic() - start > budget - 0.05:
            break
        g = _lp_solve(A, Mt, n, timeout=per)
        if g is None:
            continue
        if float(np.linalg.norm(g - a, ord=1)) <= 1e-8:
            continue
        cand, sc = _search_blend(a, g, cur_score, cap_t=1.0)
        if cand is not None:
            return cand
    return None


def _select_active(c, rng, max_active=120):
    m = int(c.size)
    if m <= 0:
        return np.array([], dtype=np.int64)
    M = float(c.max())
    if M <= 0.0 or not np.isfinite(M):
        return np.arange(m, dtype=np.int64)

    near = np.where(c >= M - max(1e-6 * M, 1e-12))[0]
    top_count = min(m, max(8, int(0.22 * m)))
    top = np.argsort(c, kind="stable")[-top_count:][::-1]
    extra_count = max(1, min(m, 5))
    extra = rng.choice(m, size=extra_count, replace=False)
    active = np.unique(np.concatenate((near, top, extra)).astype(np.int64))

    if active.size > max_active:
        order = np.argsort(c[active], kind="stable")[::-1]
        active = active[order[:max_active]]
    return active.astype(np.int64, copy=False)


def _lambda_for_active(a, active):
    n = int(a.size)
    m = int(active.size)
    if m == 0:
        return None
    D = np.zeros((n, m), dtype=np.float64)
    for idx, k in enumerate(active):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        length = hi - lo + 1
        if length <= 0:
            continue
        end = k - hi - 1
        if end >= 0:
            vals = a[k - lo : end : -1]
        else:
            vals = a[k - lo :: -1]
        D[lo : hi + 1, idx] = vals[:length]

    D_aug = np.column_stack((D, -np.ones(n, dtype=np.float64)))
    scale = float(np.linalg.norm(D_aug, ord=np.inf))
    if scale <= 0.0 or not np.isfinite(scale):
        return None
    D_aug = D_aug / scale

    E = np.zeros((1, m + 1), dtype=np.float64)
    E[0, :m] = 1.0
    beta = 20.0
    A = np.vstack((D_aug, beta * E))
    bvec = np.concatenate((np.zeros(n, dtype=np.float64), np.array([beta], dtype=np.float64)))
    try:
        y, *_ = np.linalg.lstsq(A, bvec, rcond=None)
    except Exception:
        return None

    lam = np.clip(np.asarray(y[:m], dtype=np.float64), 0.0, 1e6)
    s = float(lam.sum())
    if s <= 1e-10:
        return np.ones(m, dtype=np.float64) / m
    lam = lam / s
    lam = 0.85 * lam + 0.15 * (1.0 / m)
    lam = lam / lam.sum()
    return lam


def _direction_from_lambda(lam, n):
    idx = np.add.outer(np.arange(n, dtype=np.int64), np.arange(n, dtype=np.int64))
    H = lam[idx]
    b = np.ones(n, dtype=np.float64)

    if n <= 250:
        try:
            x, _ = nnls(H, b)
        except Exception:
            try:
                x = np.linalg.lstsq(H, b, rcond=None)[0]
            except Exception:
                return None
    else:
        try:
            Hreg = H + 1e-6 * np.eye(n, dtype=np.float64)
            x = np.linalg.solve(Hreg, b)
        except Exception:
            try:
                x, _ = nnls(H, b)
            except Exception:
                return None

    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s <= 1e-12:
        try:
            Hreg = H + 1e-4 * np.eye(n, dtype=np.float64)
            x = np.linalg.solve(Hreg, b)
            x = np.where(np.isfinite(x), x, 0.0)
            x = np.maximum(x, 0.0)
            s = float(x.sum())
        except Exception:
            pass
    if s <= 1e-12:
        return None
    return x / s


def _h_step(a, c, M, cur_score, rng):
    n = int(a.size)
    if n < 2:
        return None
    active = _select_active(c, rng, max_active=min(120, max(20, n // 2)))
    if active.size == 0:
        return None
    lam_active = _lambda_for_active(a, active)
    if lam_active is None:
        return None

    lam = np.zeros(c.size, dtype=np.float64)
    lam[active] = lam_active

    dirx = _direction_from_lambda(lam, n)
    if dirx is not None:
        if float(np.linalg.norm(dirx - a, ord=1)) > 1e-7:
            cand, sc = _search_blend(a, dirx, cur_score, cap_t=1.0)
            if cand is not None:
                return cand

    g = np.zeros(n, dtype=np.float64)
    for k in active:
        w = float(lam[k])
        if w <= 0.0:
            continue
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        length = hi - lo + 1
        if length <= 0:
            continue
        end = k - hi - 1
        if end >= 0:
            vals = a[k - lo : end : -1]
        else:
            vals = a[k - lo :: -1]
        g[lo : hi + 1] += 2.0 * w * vals[:length]

    g -= float(g.mean())
    if float(np.max(np.abs(g))) > 1e-15:
        cand, sc = _search_additive(a, g, cur_score, cap_t=0.05)
        if cand is not None:
            return cand
    return None


def _random_candidate(a, rng):
    n = int(a.size)
    if n < 3:
        z = np.log(np.maximum(a, 1e-12)) + rng.normal(0.0, 0.1, n)
        return _normalize_pos(np.exp(z))

    typ = int(rng.integers(0, 6))
    if typ == 0:
        z = np.log(np.maximum(a, 1e-12)) + rng.normal(0.0, 0.08, n)
        return _normalize_pos(np.exp(z))

    if typ == 1:
        k = np.array([0.2, 0.6, 1.0, 0.6, 0.2], dtype=np.float64)
        k /= k.sum()
        cand = np.convolve(a, k, mode="same")
        return _normalize_pos(cand)

    if typ == 2:
        Lmax = max(3, min(n, n // 4 + 10))
        L = int(rng.integers(3, min(Lmax, n) + 1))
        start = int(rng.integers(0, n - L + 1))
        factor = math.exp(float(np.clip(rng.normal(0.0, 0.35), -4.0, 4.0)))
        cand = a.copy()
        cand[start : start + L] *= factor
        return _normalize_pos(cand)

    if typ == 3:
        i, j = rng.choice(n, size=2, replace=False)
        i = int(i)
        j = int(j)
        amt = 0.15 * float(np.max(a))
        if a[i] < a[j]:
            i, j = j, i
        cand = a.copy()
        cand[i] -= amt
        cand[j] += amt
        return _normalize_pos(cand)

    if typ == 4:
        t = float(rng.uniform(0.2, 0.5))
        cand = (1.0 - t) * a + t * a[::-1]
        return _normalize_pos(cand)

    Lmax = max(3, min(n, n // 5 + 8))
    L = int(rng.integers(3, min(Lmax, n) + 1))
    s1 = int(rng.integers(0, n - L + 1))
    s2 = int(rng.integers(0, n - L + 1))
    cand = a.copy()
    cand[s1 : s1 + L] *= math.exp(float(np.clip(rng.normal(0.0, 0.30), -3.5, 3.5)))
    cand[s2 : s2 + L] *= math.exp(float(np.clip(rng.normal(0.0, 0.30), -3.5, 3.5)))
    return _normalize_pos(cand)


def _perturb(a, rng, sigma):
    z = np.log(np.maximum(a, 1e-12)) + rng.normal(0.0, sigma, a.size)
    return _normalize_pos(np.exp(z))


def _intensive_search(a0, stop_time, seed, use_H=True):
    a = _normalize_pos(a0)
    n = int(a.size)
    if n < 3:
        return a.copy(), _unit_score(a)

    best = a.copy()
    best_score = _unit_score(a)
    rng = np.random.default_rng(seed)
    no_improve = 0
    it = 0
    if use_H:
        pattern = ["lp", "lp", "lp", "h", "rand"]
    else:
        pattern = ["lp", "lp", "rand"]

    while time.monotonic() < stop_time - 1.5:
        c = _conv_square(a)
        M = float(c.max())
        cur = 2.0 * n * M
        if not np.isfinite(cur) or cur <= 0.0:
            break

        if cur < best_score - 1e-12:
            best = a.copy()
            best_score = cur
            no_improve = 0
        else:
            no_improve += 1

        tleft = stop_time - time.monotonic()
        if tleft < 1.0:
            break

        method = pattern[it % len(pattern)]
        improved = False

        if method == "lp":
            budget = min(1.2, max(0.3, 0.015 * tleft))
            if tleft < 12.0:
                budget = 0.25
            cand = _lp_step(a, M, cur, tleft, budget)
            if cand is not None:
                a = cand
                improved = True
                no_improve = 0

        elif method == "h" and use_H:
            cand = _h_step(a, c, M, cur, rng)
            if cand is not None:
                a = cand
                improved = True
                no_improve = 0

        else:
            for _ in range(5):
                if time.monotonic() > stop_time - 1.5:
                    break
                cand = _random_candidate(a, rng)
                sc = _unit_score(cand)
                if sc < cur - 1e-12:
                    a = cand
                    improved = True
                    no_improve = 0
                    break

        if not improved:
            for _ in range(2):
                if time.monotonic() > stop_time - 1.5:
                    break
                cand = _random_candidate(a, rng)
                sc = _unit_score(cand)
                if sc < cur - 1e-12:
                    a = cand
                    no_improve = 0
                    break

            if time.monotonic() <= stop_time - 1.5 and a is not best:
                pass

            if float(_unit_score(a)) < best_score - 1e-12:
                best = a.copy()
                best_score = _unit_score(a)
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 15:
                    sigma = 0.05 + 0.25 * min(1.0, no_improve / 40.0)
                    a = _perturb(best, rng, sigma)
                    no_improve = 0

        it += 1

    return best, best_score


def _softmax(u):
    u = np.asarray(u, dtype=np.float64)
    u = u - float(np.max(u))
    p = np.exp(u)
    s = float(p.sum())
    if s <= 1e-300 or not np.isfinite(s):
        n = u.size
        return np.ones(n, dtype=np.float64) / n
    return p / s


def _lbfgs_polish(a0, stop_time, seed):
    a = _normalize_pos(a0)
    n = int(a.size)
    if n < 3:
        return a.copy()

    state = {"a": a.copy(), "M": float(_conv_square(a).max())}
    u0 = np.log(state["a"] + 1e-300)
    u0 = u0 - float(np.mean(u0))
    u = u0.copy()

    facs = [0.50, 0.28, 0.16, 0.09, 0.05, 0.028, 0.015, 0.008, 0.004]
    if n > 300:
        opts = {"maxiter": 25, "maxfun": 50, "ftol": 1e-13, "gtol": 1e-11, "maxls": 15}
    else:
        opts = {"maxiter": 55, "maxfun": 110, "ftol": 1e-14, "gtol": 1e-12, "maxls": 18}

    for fac in facs:
        if time.monotonic() > stop_time - 6.0:
            break
        tau = max(1e-9, float(state["M"]) * fac)

        def fun(u):
            a_cur = _softmax(u)
            c = _conv_square(a_cur)
            Mc = float(c.max())
            if np.isfinite(Mc) and Mc < state["M"] - 1e-15:
                state["M"] = Mc
                state["a"] = a_cur.copy()

            z = c / tau
            zm = float(np.max(z))
            ez = np.exp(z - zm)
            se = float(ez.sum())
            if se <= 1e-300 or not np.isfinite(se):
                p = np.ones(c.size, dtype=np.float64) / c.size
                F = tau * (zm + 1e-300)
            else:
                p = ez / se
                F = tau * (zm + math.log(se + 1e-300))

            half = np.zeros(n, dtype=np.float64)
            for k in range(c.size):
                pk = float(p[k])
                if pk <= 1e-16:
                    continue
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                length = hi - lo + 1
                if length <= 0:
                    continue
                end = k - hi - 1
                if end >= 0:
                    vals = a_cur[k - lo : end : -1]
                else:
                    vals = a_cur[k - lo :: -1]
                half[lo : hi + 1] += pk * vals[:length]
            g = 2.0 * half
            gu = g * a_cur - a_cur * float(np.dot(g, a_cur))
            gu = gu + 1e-12 * (u - float(np.mean(u)))
            return F, gu

        try:
            minimize(fun, u, method="L-BFGS-B", jac=True, options=dict(opts))
        except Exception:
            pass

        try:
            u = np.log(state["a"] + 1e-300)
            u = u - float(np.mean(u))
        except Exception:
            u = u0.copy()

    return state["a"]


def _resample_unit(a, n_new):
    a = _normalize_pos(a)
    n0 = int(a.size)
    n_new = int(max(1, n_new))
    if n_new == n0:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n0, dtype=np.float64)
    x_new = np.linspace(0.0, 1.0, n_new, dtype=np.float64)
    new = np.interp(x_new, x_old, a)
    return _normalize_pos(new)


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1300.0
    horizon = t0 + max(10.0, budget - 20.0)

    parent = _get_parent()
    best_unit = _normalize_pos(parent)
    best_exact = _exact_score(best_unit)
    best_raw = best_unit.copy()

    def _consider(a, require_exact=True):
        nonlocal best_raw, best_exact
        a = np.asarray(a, dtype=np.float64).ravel()
        if a.size == 0:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        s = float(a.sum())
        if s <= 0.0:
            return
        a = a / s
        if require_exact:
            sc = _exact_score(a)
        else:
            sc = _unit_score(a)
        if np.isfinite(sc) and sc < best_exact - 1e-15:
            best_raw = a.copy()
            best_exact = sc

    ptry = _exact_score(parent)
    if np.isfinite(ptry) and ptry < best_exact:
        aclean = np.clip(parent, 0.0, 1000.0)
        s0 = float(aclean.sum())
        if s0 > 0.0:
            best_raw = (aclean / s0).copy()
            best_exact = ptry

    # Main phase: intensive LP/H/random search from the parent.
    stop1 = min(horizon, t0 + max(60.0, 0.55 * budget))
    try:
        a_main, _ = _intensive_search(best_raw, stop1, int(seed), use_H=best_raw.size <= 320)
        _consider(a_main, True)
    except Exception:
        pass

    # Smooth first-order polish.
    stop2 = min(horizon, stop1 + max(40.0, 0.25 * budget))
    if time.monotonic() < stop2 - 15.0:
        try:
            pol = _lbfgs_polish(best_raw, stop2, int(seed) + 1)
            _consider(pol, True)
        except Exception:
            pass

    # Final intensive polish.
    stop3 = horizon - 35.0
    if time.monotonic() < stop3 - 20.0:
        try:
            a_fin, _ = _intensive_search(best_raw, stop3, int(seed) + 2, use_H=best_raw.size <= 320)
            _consider(a_fin, True)
        except Exception:
            pass

    # Nearby lengths and, if time remains, a modest upsample smooth polish.
    if time.monotonic() < horizon - 70.0:
        base = _normalize_pos(best_raw)
        n0 = int(base.size)
        for nt in (n0 + 1, n0 - 1, n0 + 2, n0 - 2, n0 + 4, n0 - 4):
            if time.monotonic() > horizon - 45.0:
                break
            if nt < 16 or nt > 600 or nt == n0:
                continue
            alloc_end = min(horizon - 35.0, time.monotonic() + 18.0)
            if alloc_end - time.monotonic() < 8.0:
                continue
            try:
                init = _resample_unit(base, nt)
                ra, _ = _intensive_search(init, alloc_end, int(seed) + 100 + nt, use_H=nt <= 250)
                _consider(ra, True)
            except Exception:
                pass

        n_big = 2 * n0
        if n_big <= 700 and time.monotonic() < horizon - 60.0:
            try:
                init_big = _resample_unit(base, n_big)
                pol_big = _lbfgs_polish(
                    init_big,
                    min(horizon - 30.0, time.monotonic() + 45.0),
                    int(seed) + 999,
                )
                _consider(pol_big, True)
            except Exception:
                pass

    # Clean candidate and final validation.
    out = _normalize_pos(best_raw)
    candidates = []
    candidates.append(out.copy())

    c0 = out.copy()
    c0[np.abs(c0) < 1e-14] = 0.0
    s0 = float(c0.sum())
    if s0 > 1e-300:
        candidates.append(c0 / s0)

    candidates.append(out[::-1].copy())

    for cand in candidates:
        try:
            sc = _exact_score(cand)
        except Exception:
            sc = float("inf")
        if np.isfinite(sc) and sc < best_exact:
            best_raw = cand
            best_exact = sc

    # Use the supplied exact evaluator for final acceptance.
    try:
        final_list = [float(x) for x in np.clip(np.asarray(best_raw, dtype=np.float64), 0.0, 1000.0)]
        fs = evaluate_sequence(final_list)
        if np.isfinite(fs):
            return final_list
    except Exception:
        pass

    fin = np.clip(np.asarray(best_raw, dtype=np.float64), 0.0, 1000.0)
    s = float(fin.sum())
    if fin.size == 0 or s < 0.01 or not np.isfinite(s):
        return [1.0] * 32
    fin = fin / s
    fin[np.abs(fin) < 1e-15] = 0.0
    if fin.size == 0:
        return [1.0] * 32
    return [float(x) for x in fin]
