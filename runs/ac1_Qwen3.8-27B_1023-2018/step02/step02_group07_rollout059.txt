import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len


def _softmax(x):
    x = np.asarray(x, dtype=np.float64).ravel()
    if x.size == 0:
        return np.array([1.0])
    x = x - np.max(x)
    x = np.clip(x, -90.0, 90.0)
    e = np.exp(x)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full(x.size, 1.0 / x.size, dtype=np.float64)
    return e / s


def _conv(a, fs):
    F = rfft(a, fs)
    return irfft(F * F, fs)[: 2 * len(a) - 1]


def _cc(q, a, fs2):
    """Cross-correlation: h[j] = sum_l a[l]*q[j+l] for j=0..n-1"""
    n = len(a)
    ar = a[::-1]
    c = irfft(rfft(ar, fs2) * rfft(q, fs2), fs2)
    return c[n - 1: 2 * n - 1]


def _score_and_conv(a, fs):
    b = _conv(a, fs)
    n = len(a)
    sa = float(a.sum())
    if sa < 0.01:
        return math.inf, b
    return 2.0 * n * float(b.max()) / (sa * sa), b


def _init_variants(n, seed, parent=None):
    rng = np.random.default_rng(seed)
    t = (np.arange(n) + 0.5) / n
    x = 2.0 * t - 1.0
    variants = []

    variants.append(np.maximum(0.01, 1.0 - np.abs(x)))
    variants.append(0.5 * (1.0 + np.cos(np.pi * x)))
    variants.append(np.exp(-2.5 * x * x))
    variants.append(np.abs(np.sin(np.pi * t)) + 0.05)
    variants.append(np.maximum(0.01, (1.0 - np.abs(x)) ** 2.5))
    variants.append(np.exp(-4.0 * np.abs(x - 0.2)))
    variants.append(np.exp(-4.0 * np.abs(x + 0.2)))
    a = np.ones(n)
    for freq in range(1, 6):
        ph = rng.uniform(0, 2 * math.pi)
        amp = rng.uniform(0.15, 0.6) / freq
        a += amp * np.cos(2 * math.pi * freq * t + ph)
    variants.append(np.maximum(a, 0.01))
    variants.append(np.abs(rng.normal(0, 1, n)) + 0.1)
    # Symmetric random
    r = np.abs(rng.normal(0, 1, n // 2)) + 0.05
    if n % 2 == 1:
        r = np.append(r, r[:: -1])
    else:
        r = np.concatenate([r, r[:: -1]])
    variants.append(r)

    if parent is not None and parent.size > 0:
        p = np.asarray(parent, dtype=np.float64).ravel()
        if p.size == n:
            variants.append(p + 0.02 * rng.normal(0, 1, n))
            variants.append(p + 0.05 * rng.normal(0, 1, n))
        else:
            idx = (np.arange(n) * p.size // n).astype(int)
            variants.append(p[idx] + 0.02 * rng.normal(0, 1, n))

    result = []
    for v in variants:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 1e-6)
        v = v / v.sum()
        result.append(v)
    return result


def _optimize(a0, n, max_time, seed, fs, fs2):
    """Run soft-gradient Adam optimization on log-space parameterization."""
    rng = np.random.default_rng(seed)
    z = np.log(np.maximum(a0, 1e-12)).copy()
    z -= z.mean()

    best_z = z.copy()
    best_score = _score_and_conv(_softmax(z), fs)[0]

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = 0.5
    tau = 0.35
    stuck = 0
    start = time.monotonic()

    while time.monotonic() - start < max_time:
        a = _softmax(np.clip(z, -40, 40))
        b = _conv(a, fs)
        bmax = float(b.max())
        if not np.isfinite(bmax) or bmax <= 0:
            break
        sa = float(a.sum())
        cur = 2.0 * n * bmax / (sa * sa)

        if cur < best_score - 1e-12:
            best_score = cur
            best_z = z.copy()
            stuck = 0
        else:
            stuck += 1

        # Temperature schedule: start soft, get harder
        tau_eff = max(0.02, tau * bmax)

        # Soft distribution over lags
        bt = b / tau_eff
        bt -= bt.max()
        e = np.exp(np.clip(bt, -80, 80))
        s = e.sum()
        if not np.isfinite(s) or s <= 0:
            p = np.zeros_like(b)
            p[int(np.argmax(b))] = 1.0
        else:
            p = e / s

        # Gradient: h[j] = sum_k p[k] * a[k-j] (valid)
        h = _cc(p, a, fs2)
        grad = a * (h - float(a @ h))

        # Optional: add small variance regularization
        if tau > 0.2:
            q = b - 1.0 / (2 * n - 1)
            vg = _cc(q, a, fs2)
            grad += 0.05 * (a * (vg - float(a @ vg)))

        if not np.isfinite(gn) or gn < 1e-14:
            if stuck > 30:
                z = best_z + 0.8 * rng.normal(0, 1, n)
                z -= z.mean()
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
                lr = 0.5
                stuck = 0
            continue

        grad /= gn
        t += 1
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * grad * grad
        mh = m / (1.0 - 0.9 ** t)
        vh = v / (1.0 - 0.999 ** t)
        upd = mh / (np.sqrt(vh) + 1e-8)
        um = float(np.abs(upd).max())
        if not np.isfinite(um) or um <= 0:
            continue
        upd /= um

        z -= lr * upd
        z -= z.mean()
        z = np.clip(z, -40, 40)
        z[~np.isfinite(z)] = 0.0

        # Adaptive scheduling
        if t % 25 == 0:
            tau = max(0.08, tau * 0.97)
        if stuck > 20:
            lr = max(0.1, lr * 0.9)
        if stuck > 120:
            z = best_z + rng.normal(0, 0.7, n)
            z -= z.mean()
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            lr = 0.5
            tau = max(0.15, tau)
            stuck = 0

    a_best = _softmax(best_z)
    s_best = _score_and_conv(a_best, fs)[0]
    return a_best, s_best


def _coordinate_polish(a, max_time, seed):
    """Coordinate descent polish: try small perturbations on subsets."""
    rng = np.random.default_rng(seed)
    n = len(a)
    fs = next_fast_len(2 * n - 1)
    best = a.copy()
    bs, _ = _score_and_conv(best, fs)
    if not np.isfinite(bs):
        return best, bs
    start = time.monotonic()
    sigma = 0.15
    stuck = 0

    while time.monotonic() - start < max_time:
        a2 = best.copy()
        # Random subset perturbation
        k = max(1, int(rng.integers(1, max(2, n // 10 + 1))))
        idx = rng.choice(n, size=k, replace=False)
        a2[idx] *= np.exp(rng.normal(0, sigma, k))
        a2 = np.maximum(a2, 1e-12)
        a2 = a2 / a2.sum()
        s2, _ = _score_and_conv(a2, fs)
        if np.isfinite(s2) and s2 < bs - 1e-11:
            best = a2
            bs = s2
            stuck = 0
            sigma = max(0.05, sigma * 0.92)
        else:
            stuck += 1
            sigma = min(0.5, sigma * 1.08)
            if stuck > 40:
                stuck = 0
                sigma = 0.3

    return best, bs


def _fourier_polish(a, max_time, seed):
    """Try structured Fourier perturbations."""
    rng = np.random.default_rng(seed)
    n = len(a)
    fs = next_fast_len(2 * n - 1)
    best = a.copy()
    bs, _ = _score_and_conv(best, fs)

    t = np.arange(n) / n
    start = time.monotonic()

    for freq in list(range(1, min(n // 2, 30))):

            if time.monotonic() - start > max_time:

            a2 = best * (1.0 + eps * np.cos(2 * math.pi * freq * t + ph))
            a2 = np.maximum(a2, 1e-12)
            a2 = a2 / a2.sum()
            s2, _ = _score_and_conv(a2, fs)
            if np.isfinite(s2) and s2 < bs - 1e-12:
                best = a2
                bs = s2

    return best, bs
