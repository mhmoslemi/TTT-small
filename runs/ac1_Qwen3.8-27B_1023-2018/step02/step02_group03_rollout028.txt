import time
import math

try:
    from scipy.optimize import linprog
    import scipy.sparse as sps
    from scipy.fft import fft, ifft
except ImportError:
    linprog = None
    sps = None
    fft = np.fft.fft
    ifft = np.fft.ifft

def propose_candidate(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    try:
        budget = max(10.0, float(budget_s))
    except:
        budget = 1300.0
    deadline = t0 + budget - 10.0
    
    # Scoring function
    def score(a):
        a = np.asarray(a, dtype=np.float64).ravel()
        a = np.clip(a, 0, 1000)
        n = a.size
        if n == 0:
            return float('inf')
        s = float(a.sum())
        if s < 0.01:
            return float('inf')
        b = np.convolve(a, a)
        return float(2.0 * n * b.max() / (s * s))
    
    # Best tracking
    best_score = float('inf')
    best_arr = None
    
    def update_best(a):
        nonlocal best_score, best_arr
        a = np.asarray(a, dtype=np.float64).ravel()
        a = np.clip(a, 0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        v = score(a)
        if math.isfinite(v) and v < best_score:
            best_score = v
            best_arr = a.copy()
    
    # Get parent
    parent = None
    try:
        h = globals().get('height_sequence_1', None)
        if h is not None:
            parent = np.asarray(h, dtype=np.float64).ravel()
            parent = np.clip(parent, 0, 1000.0)
            if parent.size > 0:
                update_best(parent)
    except:
        pass
    
    if best_arr is None:
        best_arr = np.ones(96) / 96.0
        update_best(best_arr)
    
    # Simplex projection: project x onto {x>=0, sum x=1}
    def simplex_proj(x):
        x = np.asarray(x, dtype=np.float64).ravel()
        n = x.size
        if n == 0:
            return x
        # Binary search for threshold
        lo, hi = float(x.min()), float(x.max())
        for _ in range(60):
            mid = (lo + hi) / 2
            s = float(np.sum(np.maximum(x - mid, 0)))
            if s > 1.0:
                lo = mid
            else:
                hi = mid
        theta = (lo + hi) / 2
        proj = np.maximum(x - theta, 0)
        s = float(proj.sum())
        if s > 1e-15:
            proj /= s
        else:
            proj = np.ones(n) / n
        return proj
    
    # Gradient descent with softmax temperature
    def grad_desc(a, n, T_init, T_min, n_iters, stop_time):
        a = simplex_proj(a)
        local_sc = score(a)
        update_best(a)
        T = T_init
        T_decay = (T_min / T_init) ** (1.0 / max(1, n_iters))
        
        for it in range(n_iters):
            if time.monotonic() > stop_time:
                break
            
            c = np.convolve(a, a)
            cm = float(c.max())
            
            # Softmax weights
            T_eff = max(T, 1e-6)
            c_shifted = c - cm  # for numerical stability
            exp_c = np.exp(c_shifted / T_eff * n)  # scale by n for resolution
            w = exp_c / exp_c.sum()
            
            # Gradient: dL/da_i ≈ 2 * sum_k w_k * a_{k-i}
            grad = np.zeros(n)
            for k in range(2 * n - 1):
                wk = w[k]
                if wk < 1e-15:
                    continue
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                for j in range(lo, hi + 1):
                    grad[j] += wk * a[k - j]
                    if k - j != j:
                        grad[k - j] += wk * a[j]
            grad *= 2.0
            
            # Center the gradient (invariance to constant shifts... actually not needed on simplex)
            # Step size
            gn = float(np.linalg.norm(grad))
            if gn < 1e-15:
                break
            
            # Adaptive step size
            best_alpha = None
            best_sc_t = local_sc
            for alpha in [0.1, 0.05, 0.02, 0.01, 0.005, 0.001]:
                cand = simplex_proj(a - alpha * grad / gn)
                sc_t = score(cand)
                if sc_t < best_sc_t:
                    best_sc_t = sc_t
                    best_alpha = cand
            
            if best_alpha is not None:
                a = best_alpha
                local_sc = best_sc_t
                update_best(a)
            else:
                # No improvement, try smaller step or break
                break
            
            T *= T_decay
            T = max(T, T_min)
        
        return a
    
    # LP-based refinement
    def lp_refine(a, n, max_iters, stop_time, margin=0.99):
        a = simplex_proj(a)
        local_sc = score(a)
        update_best(a)
        stag = 0
        
        for it in range(max_iters):
            if time.monotonic() > stop_time:
                break
            
            c = np.convolve(a, a)
            M = float(c.max())
            if not math.isfinite(M) or M <= 0:
                break
            
            # Build sparse convolution matrix
            rows = np.empty(n * n, dtype=np.int32)
            cols = np.empty(n * n, dtype=np.int32)
            vals = np.empty(n * n, dtype=np.float64)
            cnt = 0
            for k in range(2 * n - 1):
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                for j in range(lo, hi + 1):
                    rows[cnt] = k
                    cols[cnt] = j
                    vals[cnt] = a[k - j]
                    cnt += 1
            
            A = sps.csr_matrix((vals[:cnt], (rows[:cnt], cols[:cnt])), shape=(2 * n - 1, n))
            
            # LP: max sum g s.t. A@g <= M*margin, g >= 0
            b_ub = np.full(2 * n - 1, M * margin + 1e-12)
            c_obj = -np.ones(n)
            
            try:
                res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                             bounds=[(0, None)] * n, method='highs',
                             options={'presolve': True, 'time_limit': 30})
            except:
                continue
            
            if res is None or getattr(res, 'x', None) is None:
                continue
            g = np.clip(res.x, 0, None)
            sg = float(g.sum())
            if sg < 1e-15:
                continue
            g /= sg
            
            # Line search
            best_cand = None
            best_sc_t = local_sc
            for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
                cand = simplex_proj((1 - t) * a + t * g)
                sc_t = score(cand)
                if sc_t < best_sc_t - 1e-12:
                    best_sc_t = sc_t
                    best_cand = cand
            
            if best_cand is not None:
                a = best_cand
                local_sc = best_sc_t
                update_best(a)
                stag = 0
            else:
                stag += 1
                if stag > 15:
                    break
        
        return a
    
    # Length interpolation
    def interp_len(src, n):
        if src.size == n:
            return src.copy()
        ox = np.linspace(0, 1, src.size)
        nx = np.linspace(0, 1, n)
        out = np.interp(nx, ox, src)
        out = np.clip(out, 0, None)
        s = out.sum()
        if s > 1e-15:
            out /= s
        else:
            out = np.ones(n) / n
        return out
    
    # === MAIN OPTIMIZATION ===
    p = parent if parent is not None else best_arr
    n_p = p.size
    p_norm = p / p.sum()
    
    # Phase 1: Gradient descent at parent length
    t_rem = deadline - time.monotonic()
    alloc1 = min(400.0, t_rem * 0.35)

                      n_iters=20000, stop_time=time.monotonic() + alloc1)
        update_best(a1)
    
    # Phase 2: LP refinement at parent length
    if time.monotonic() < deadline - 30:
        t_rem2 = deadline - time.monotonic() - 25
        alloc2 = min(200.0, t_rem2 * 0.4)
        if alloc2 > 15:
            a2 = lp_refine(a1, n_p, max_iters=50, stop_time=time.monotonic() + alloc2,
                          margin=0.99)
            update_best(a2)
    
    # Phase 3: Gradient polish after LP
    if time.monotonic() < deadline - 20:
        a3 = grad_desc(a2, n_p, T_init=n_p, T_min=n_p*0.05,
                      n_iters=10000, stop_time=deadline - 15)
        update_best(a3)
    
    # Phase 4: Try different lengths
    if time.monotonic() < deadline - 30:
        for L in [n_p - 20, n_p + 20, n_p + 40, 160, 240, 280, 360, 400, 480]:
            if time.monotonic() >= deadline - 25:
                break
            if L < 8 or L > 600:
                continue
            t_rem3 = deadline - time.monotonic() - 20
            alloc3 = min(30.0, t_rem3 / 5)
            if alloc3 < 10:
                continue
            
            a_init = interp_len(p, L)
            a_w = grad_desc(a_init, L, T_init=L*2, T_min=L*0.1,
                          n_iters=5000, stop_time=time.monotonic() + alloc3)
            update_best(a_w)
            
            if time.monotonic() < deadline - 20:
                a_w = lp_refine(a_w, L, max_iters=10, stop_time=deadline - 15,
                              margin=0.99)
                a_w = grad_desc(a_w, L, T_init=L, T_min=L*0.05,
                              n_iters=3000, stop_time=deadline - 12)
                update_best(a_w)
    
    # Phase 5: Multiple restarts
    if time.monotonic() < deadline - 25:
        for r in range(20):
            if time.monotonic() >= deadline - 15:
                break
            t_rem4 = deadline - time.monotonic() - 15
            alloc4 = min(20.0, t_rem4 / max(1, 20 - r))
            if alloc4 < 5:
                break
            
            # Different initializations
            rtype = r % 5
            if rtype == 0:
                a_init = p_norm * np.exp(0.2 * rng.standard_normal(n_p))
            elif rtype == 1:
                a_init = p_norm[::-1].copy()
            elif rtype == 2:
                # Wedge shape
                a_init = np.abs(np.linspace(-1, 1, n_p) + rng.normal(0, 0.2, n_p))
            elif rtype == 3:
                # Concentrated
                a_init = np.exp(-0.5 * ((np.arange(n_p) - rng.uniform(0.2, 0.8) * n_p) / (0.1 * n_p))**2)
            else:
                a_init = p_norm + 0.15 * rng.standard_normal(n_p)
            
            a_init = np.clip(a_init, 0, None)
            s_init = a_init.sum()
