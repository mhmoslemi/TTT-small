```python
from scipy.fft import rfft, irfft, next_fast_len

def _fft_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2*n-1]
    np.maximum(c, 0, out=c)  # numerical safety
    return c

def _project_simplex(v):
    """Project v onto the probability simplex {a: a>=0, sum(a)=1}."""
    n = len(v)
    v = v.copy()
    mu = np.zeros(n)
    for _ in range(10):  # bisection
        # Sort
        sv = np.sort(v)[::-1]
        cumsum = np.cumsum(sv)
        rho = np.where(sv > (cumsum - 1) / np.arange(1, n+1))[0]
        if len(rho) == 0:
            rho = np.array([n-1])
        rho = rho[-1] + 1  # 1-indexed count
        theta = (cumsum[rho-1] - 1) / rho
        result = np.maximum(v - theta, 0)
        return result
    return np.ones(n) / n
```

```python
def _project_simplex(v, target_sum=1.0):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - target_sum)
    rho = cond.sum()
    if rho == 0:
        return np.full(n, target_sum / n)
    theta = (cssv[rho-1] - target_sum) / rho
    return np.maximum(v - theta, 0)
```

```python
def _compute_grad(a, w):
    """Compute gradient of sum_k w[k] * b[k] where b = conv(a,a)."""
    n = len(a)
    # grad[i] = 2 * sum_j w[i+j] * a[j] - w[2i] * a[i] (if 2i < 2n-1)
    # First term: use matrix W where W[i,j] = w[i+j]
    w_ext = np.zeros(2*n)
    w_ext[:len(w)] = w
    W = w_ext[np.arange(n)[:, None] + np.arange(n)[None, :]]
    grad = 2.0 * (W @ a)
    # Diagonal correction
    diag_idx = 2 * np.arange(n)
    valid = diag_idx < len(w)
    grad[valid] -= w[diag_idx[valid]] * a[valid]
    return grad
```

```python
def _fft_conv(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2*n-1]
    np.maximum(c, 0, out=c)
    c[~np.isfinite(c)] = 0
    return c

def _project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1.0)
    rho = int(cond.sum())
    if rho == 0:
        return np.full(n, 1.0/n)
    theta = (cssv[rho-1] - 1.0) / rho
    return np.maximum(v - theta, 0)

    n = len(a)
    return 2.0 * n * _fft_conv(a).max()

def _compute_grad(a, w):
    n = len(a)
    w_ext = np.zeros(2 * n)
    w_ext[:len(w)] = w
    W = w_ext[np.arange(n)[:, None] + np.arange(n)[None, :]]
    grad = 2.0 * (W @ a)
    diag_idx = 2 * np.arange(n)
    valid = diag_idx < len(w)
    grad[valid] -= w[diag_idx[valid]] * a[valid]
    return grad

    n = len(a)
```

```python
def _compute_grad_fast(a, w):
    n = len(a)
    # c[i] = sum_j w[i+j] * a[j]
    # c[i] = (w_rev * a)[2n-2-i] where w_rev = w[::-1]
    
    w_rev = w[::-1]  # length 2n-1
    conv_size = (2*n-1) + n - 1  # = 3n-2
    fsize = next_fast_len(conv_size)
    
    W_rev = rfft(w_rev, fsize)
    A = rfft(a, fsize)
    d = irfft(W_rev * A, fsize)[:conv_size]
    
    # c[i] = d[2n-2-i] for i=0,...,n-1
    c = d[2*n-2 :: -1][:n]  # d[2n-2], d[2n-3], ..., d[n-1]
    
    # grad = 2*c - diag_correction
    grad = 2.0 * c
    diag_idx = 2 * np.arange(n)
    valid = diag_idx < len(w)
    grad[valid] -= w[diag_idx[valid]] * a[valid]
    
    return grad
```
