from scipy.optimize import linprog

def _get_parent():

        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(320, dtype=np.float64)
    if p.size == 0:
        p = np.ones(320, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)

    if p.size == 0:
        p = np.ones(320, dtype=np.float64)
    return p

def _conv_full(a):
    """Compute full convolution a*a using FFT for efficiency."""
    n = len(a)
    if n <= 256:
        return np.convolve(a, a, mode="full")
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)  # numerical roundoff
    return c

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)

def _lp_direction(f, n, M=None):
    """Find LP direction: max sum(g) s.t. f*g <= M, g >= 0."""
    if M is None:
        b = _conv_full(f)
        M = float(b.max())
    if M < 1e-15:
        return None, M
    
    # Build constraint matrix A where (A @ g) = f * g (convolution)
    # A has shape (2n-1, n)
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j+n, j] = f
    
    c_vec = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    
    try:
        res = linprog(c_vec, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': 30})
    except Exception:
        return None, M
    
    if res.status != 0 or res.x is None:
        return None, M
    
    g = np.asarray(res.x, dtype=np.float64)
    g_sum = float(g.sum())
    if g_sum < 1e-15:

def _get_parent():

        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:

def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):

    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())

def _make_inits(parent, n, seed):

    inits = []
    # Parent
    inits.append(parent.copy())
    # Cosine bump
    x = np.linspace(0, 2*math.pi, n, endpoint=False)
    inits.append(0.5 + 0.5 * np.cos(x))
    # Superposition of cosines

        if s > 1e-10:

def _gradient_refine(a_in, n, stop_time, seed=123):
    """Gradient-based refinement using logsumexp smoothing."""
    rng = np.random.default_rng(seed)
    a = np.clip(a_in.ravel()[:n].copy(), 0, None)
    s = a.sum()
    if s < 1e-10:
        return a, Mathf.inf
    a = a / s
    
    best_a = a.copy()
    best_score = _score(a)
    
    tau = 0.03
    lr = 0.01
    no_improve = 0
    
    idx = np.arange(n)
    
    while time.monotonic() < stop_time - 1.0:
        conv = _conv_full(a)
        max_c = conv.max()
        shifted = (conv - max_c) / tau
        # Clip for numerical stability
        shifted = np.clip(shifted, -50, 0)
        weights = np.exp(shifted)
        w_sum = weights.sum()
        if w_sum < 1e-300:
            break
        weights /= w_sum
        
        # Gradient: grad[m] = (2/tau) * sum_j weights[m+j] * a[j]
        W = weights[idx[:, None] + idx[None, :]]
        grad = (2.0 / tau) * (W @ a)
        
        # Gradient step with projection
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-10:
            a_new = a
        else:
            a_new = a_new / sn

            a = a_new
            best_a = a.copy()
            best_score = sc
            no_improve = 0
            lr = min(0.03, lr * 1.02)
        else:
            no_improve += 1
            lr = max(0.002, lr * 0.92)
            if no_improve % 30 == 0:

                a = np.maximum(a + rng.normal(0, 0.005, n), 0)
                a /= max(a.sum(), 1e-10)
                tau = max(0.005, tau * 0.9)
        
        tau = max(0.005, tau * 0.9995)
    
    return best_a, best_score

import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _get_parent():

        h = globals()["height_sequence_1"]

def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _make_inits(parent, n, seed):
