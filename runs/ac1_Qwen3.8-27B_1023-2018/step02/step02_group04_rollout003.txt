def _softmax(z):
    z = np.asarray(z,float).ravel()
    z = z - np.max(z)
    e = np.exp(z)

def _soft_obj_grad(z,beta):
   z=np.asarray(z,float).ravel()
   n=z.size; a=_softmax(z)
   c=np.convolve(a,a)
   mc=float(c.max())
   x=beta*(c-mc)
   x=np.clip(x,-700,700)
   ex=np.exp(x)
   Z=ex.sum()

   else:
      J=(mc + math.log(Z)/beta)
      p=ex/Z

   az=float(np.dot(a,ga))
   gz=a*ga - a*az
   if not finite J: J=1e100
   if not finite gz: gz=np.zeros(n)

    if not math.isfinite(s) or s < 0.01:
        return float("inf")
    try:
        b = np.convolve(a, a)
        m = float(np.max(b))
    except Exception:
        return float("inf")
    if not math.isfinite(m):
        return float("inf")
    if m < 0.0:
        m = 0.0
    return float(2.0 * n * m / (s * s))


def _update_best(a, best):
    try:
        a = _normalize(a)
        v = _score(a)
        if math.isfinite(v) and v < best[0]:
            best[0] = v
            best[1] = a.copy()
    except Exception:
        pass


def _interpolate(template, n):
    try:
        a = _clean(template)
        n = int(max(2, n))
        if a.size < 2:
            return np.ones(n, dtype=np.float64) / n
        xo = np.linspace(0.0, 1.0, a.size)
        xn = np.linspace(0.0, 1.0, n)
        return np.clip(np.interp(xn, xo, a), 0.0, None)
    except Exception:
        return np.ones(int(max(2, n)), dtype=np.float64) / int(max(2, n))


def _softmax(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size == 0:
        return np.ones(2, dtype=np.float64) * 0.5
    z = z - float(np.max(z))
    e = np.exp(np.clip(z, -700.0, 700.0))
    s = float(e.sum())
    if not math.isfinite(s) or s <= 1e-300:
        return np.ones(z.size, dtype=np.float64) / z.size
    return e / s


def _soft_obj_grad(z, beta):
    z = np.asarray(z, dtype=np.float64).ravel()
    n = z.size
    a = _softmax(z)
    c = np.convolve(a, a)
    mc = float(np.max(c))
    x = beta * (c - mc)
    x = np.clip(x, -700.0, 700.0)
    ex = np.exp(x)
    Z = float(ex.sum())
    if not math.isfinite(Z) or Z <= 1e-300:
        J = mc
        p = np.full(c.size, 1.0 / c.size, dtype=np.float64)
    else:
        J = mc + math.log(Z) / beta
        p = ex / Z
    if not math.isfinite(J):
        J = 1e100
    try:
        h = np.correlate(p, a, mode="valid")
    except Exception:
        h = np.zeros(n, dtype=np.float64)
    ga = 2.0 * h
    az = float(np.dot(a, ga))
    gz = a * ga - a * az
    if not math.isfinite(az):
        gz = np.zeros(n, dtype=np.float64)
    else:
        gz = np.nan_to_num(gz, nan=0.0, posinf=0.0, neginf=0.0)
    return float(J), gz.astype(np.float64, copy=False)


def _coordinate_refine(a0, limit, best, rng):
    a = _normalize(a0)
    n = a.size
    if n < 2:
        a = np.ones(2, dtype=np.float64) * 0.5
        n = a.size

    try:
        c = np.convolve(a, a)
        s = float(2.0 * n * max(0.0, float(c.max())))
    except Exception:
        return _normalize(a)

    if not math.isfinite(s):
        return _normalize(a)
    _update_best(a, best)

    best_a = a.copy()
    best_s = s
    stall = 0
    sweep = 0

    try:
        idx_lists = [i + np.arange(n, dtype=np.int64) for i in range(n)]
    except Exception:
        return best_a

    while time.monotonic() < limit - 3.0:
        sweep += 1
        if sweep > 100000:
            break

        improved_this_sweep = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() >= limit - 3.0:
                return best_a

            v = float(a[i])
            d = 1.0 - v
            if not math.isfinite(d) or d <= 1e-10:
                continue

            if d < 1e-4:
                lo_y = max(0.0, v - 1e-3)
            else:
                lo_y = 0.0
            hi_y = 1.0
            if v < lo_y:
                lo_y = v

            k = idx_lists[i]
            twi = 2 * i
            v2 = v * v

            a_other = a.copy()
            a_other[i] = 0.0
            coef = 2.0 * a_other
            c_local = c

            def val_y(y, d=d, k=k, coef=coef, c_local=c_local, v=v, v2=v2,
                      twi=twi, n=n):
                if y < -1e-14 or y > 1.0 + 1e-14:
                    return float("inf")
                y = min(1.0, max(0.0, float(y)))
                alpha = (1.0 - y) / d
                if not math.isfinite(alpha) or alpha < -1e-14:
                    return float("inf")
                if alpha < 0.0:
                    alpha = 0.0
                alpha2 = alpha * alpha
                if not math.isfinite(alpha2):
                    return float("inf")

                cn = c_local * alpha2
                f = alpha * y - alpha2 * v
                if f != 0.0:
                    cn[k] = cn[k] + coef * f
                cn[twi] = cn[twi] + y * y - alpha2 * v2

                m = float(cn.max())
                if not math.isfinite(m):
                    return float("inf")
                if m < 0.0:
                    m = 0.0
                return float(2.0 * n * m)

            grid_n = 33
            try:
                ys = np.linspace(lo_y, hi_y, grid_n, dtype=np.float64)
            except Exception:
                continue

            best_y = v
            best_val = s
            vals = []

            for yy in ys:
                yy = float(yy)
                vv = val_y(yy)
                if math.isfinite(vv) and vv < best_val:
                    best_val = vv
                    best_y = yy
                vals.append((vv, yy))

            # Refine a few promising grid intervals.
            try:
                vals.sort()
            except Exception:
                vals = []

            step_y = (hi_y - lo_y) / max(1, grid_n - 1)
            top = 3
            gr_inv = (math.sqrt(5.0) - 1.0) / 2.0

            for vv0, yy0 in vals[:top]:
                if not math.isfinite(vv0):
                    continue
                if abs(yy0 - v) < 1e-9 and vv0 >= s:
                    continue

                lo_r = max(lo_y, float(yy0) - step_y)
                hi_r = min(hi_y, float(yy0) + step_y)
                if hi_r - lo_r <= 1e-14:
                    continue

                g1 = hi_r - (hi_r - lo_r) * gr_inv
                g2 = lo_r + (hi_r - lo_r) * gr_inv
                f1 = val_y(g1)
                f2 = val_y(g2)

                for _ in range(16):
                    if f1 < f2:
                        hi_r = g2
                        g2, f2 = g1, f1
                        g1 = hi_r - (hi_r - lo_r) * gr_inv
                        f1 = val_y(g1)
                    else:
                        lo_r = g1
                        g1, f1 = g2, f2
                        g2 = lo_r + (hi_r - lo_r) * gr_inv
                        f2 = val_y(g2)

                for cand in (g1, g2, 0.5 * (lo_r + hi_r), float(yy0)):
                    cand = float(cand)
                    vc = val_y(cand)
                    if math.isfinite(vc) and vc < best_val:
                        best_val = vc
                        best_y = cand

            if math.isfinite(best_val) and best_val < s * (1.0 - 1e-11) and best_y != v:
                alpha = (1.0 - best_y) / d
                if not math.isfinite(alpha) or alpha < 0.0:
                    continue

                a[i] = best_y
                if alpha != 1.0:
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    a[mask] *= alpha

                a = np.clip(a, 0.0, None)
                ss = float(a.sum())
                if not math.isfinite(ss) or ss <= 1e-300:
                    a = np.ones(n, dtype=np.float64) / n
                    c = np.convolve(a, a)
                    s = float(2.0 * n * max(0.0, float(c.max())))
                    continue
                a = a / ss

                c = np.convolve(a, a)
                s = float(2.0 * n * max(0.0, float(c.max())))
                if not math.isfinite(s):
                    a = best_a.copy()
                    c = np.convolve(a, a)
                    s = best_s
                    continue

                if s < best_s * (1.0 - 1e-12):
                    best_s = s
                    best_a = a.copy()
                    _update_best(a, best)
                improved_this_sweep = True

        if improved_this_sweep:
            stall = 0
            continue

        stall += 1
        if stall >= 2:
            explored = False
            base = best_a
            for sig in (0.03, 0.08, 0.18, 0.35, 0.80):
                if time.monotonic() >= limit - 3.0:
                    break
                try:
                    w = np.exp(np.clip(rng.normal(0.0, sig, n), -5.0, 5.0))
                    pa = base * w + 1e-9
                    pa = _normalize(pa)
                    ps = _score(pa)
                except Exception:
                    continue
                if not math.isfinite(ps):
                    continue
                if ps < best_s * (1.0 - 1e-12):
                    a = pa.copy()
                    c = np.convolve(a, a)
                    s = ps
                    best_s = ps
                    best_a = a.copy()
                    _update_best(a, best)
                    stall = 0
                    explored = True
                    break
                if ps < s and ps < best_s * 1.10:
                    a = pa.copy()
                    c = np.convolve(a, a)
                    s = ps
                    stall = 0
                    explored = True
                    break

            if not explored and stall >= 5:
                break

    return best_a


def _smooth_refine(a0, limit, best, rng):
    if minimize is None:
        return None
    a = _normalize(a0)
    s = _score(a)
    _update_best(a, best)
    best_a = a.copy()
    best_s = s
    n = best_a.size

    betas = (15.0, 35.0, 75.0, 150.0, 300.0, 600.0, 1200.0)

    for trial in range(3):
        if time.monotonic() >= limit - 12.0:
            break
        if trial == 0:
            base = best_a
        else:
            sig = 0.05 * trial
            try:
                base = _normalize(best_a *
                                  np.exp(np.clip(rng.normal(0.0, sig, n), -5.0, 5.0)) +
                                  1e-9)
            except Exception:
                base = best_a

        z = np.log(base + 1e-10)
        z = z - float(np.mean(z))

        for beta in betas:
            rem = limit - time.monotonic()
            if rem < 12.0:
                break

            def fun(zz, bb=beta):
                if time.monotonic() >= limit - 3.0:
                    raise _Timeout
                J, _ = _soft_obj_grad(zz, bb)
                return J

            def jac(zz, bb=beta):
                if time.monotonic() >= limit - 3.0:
                    raise _Timeout
                _, g = _soft_obj_grad(zz, bb)
                return g

            maxiter = int(min(220, max(40, rem * 0.25)))
            try:
                res = minimize(
                    fun,
                    z,
                    jac=jac,
                    method="L-BFGS-B",
                    options={
                        "maxiter": maxiter,
                        "ftol": 1e-14,
                        "gtol": 1e-9,
                        "maxls": 40,
                    },
                )
            except _Timeout:
                break
            except Exception:
                break

            if res is None or not hasattr(res, "x") or res.x is None:
                break

            try:
                ar = _softmax(res.x)
                sr = _score(ar)
            except Exception:
                break

            if not math.isfinite(sr):
                break

            if sr < best_s * (1.0 - 1e-12):
                best_s = sr
                best_a = ar.copy()
                _update_best(ar, best)
                z = np.log(best_a + 1e-10)
                z = z - float(np.mean(z))
            elif sr < best_s * 1.01:
                z = np.log(ar + 1e-10)
                z = z - float(np.mean(z))

    return best_a


def _line_search_convex(f, g, score0):
    n = f.size
    f = _normalize(np.asarray(f, dtype=np.float64).ravel())
    g = _normalize(np.asarray(g, dtype=np.float64).ravel())

    try:
        C = np.convolve(f, f)
        FG = np.convolve(f, g)
        GG = np.convolve(g, g)
    except Exception:
        return f, score0

    def val(alpha):
        alpha = float(alpha)
        if alpha < 0.0:
            alpha = 0.0
        elif alpha > 1.0:
            alpha = 1.0
        o = 1.0 - alpha
        q = (o * o) * C + (2.0 * alpha * o) * FG + (alpha * alpha) * GG
        m = float(np.max(q))
        if not math.isfinite(m) or m < 0.0:
            return float("inf")
        return float(2.0 * n * m)

    best_a = f.copy()
    best_s = score0
    best_alpha = 0.0

    grid_n = 41
    try:
        alphas = np.linspace(0.0, 1.0, grid_n, dtype=np.float64)
    except Exception:
        return best_a, best_s

    vals = []
    for aa in alphas:
        aa = float(aa)
        ss = val(aa)
        vals.append((ss, aa))
        if math.isfinite(ss) and ss < best_s:
            best_s = ss
            best_alpha = aa
            best_a = (1.0 - aa) * f + aa * g
            best_a = np.clip(best_a, 0.0, None)
            if best_a.sum() > 0.0:
                best_a = best_a / best_a.sum()

    try:
        vals.sort()
    except Exception:
        vals = []

    step = 1.0 / max(1, grid_n - 1)
    gr_inv = (math.sqrt(5.0) - 1.0) / 2.0

    for _, aa0 in vals[:3]:
        lo = max(0.0, float(aa0) - step)
        hi = min(1.0, float(aa0) + step)
        if hi - lo <= 1e-14:
            continue
        g1 = hi - (hi - lo) * gr_inv
        g2 = lo + (hi - lo) * gr_inv
        f1 = val(g1)
        f2 = val(g2)
        for _ in range(16):
            if f1 < f2:
                hi = g2
                g2, f2 = g1, f1
                g1 = hi - (hi - lo) * gr_inv
                f1 = val(g1)
            else:
                lo = g1
                g1, f1 = g2, f2
                g2 = lo + (hi - lo) * gr_inv
                f2 = val(g2)
        for cc in (g1, g2, 0.5 * (lo + hi), float(aa0)):
            cc = float(cc)
            sc = val(cc)
            if math.isfinite(sc) and sc < best_s:
                best_s = sc
                best_alpha = cc
                best_a = (1.0 - cc) * f + cc * g
                best_a = np.clip(best_a, 0.0, None)
                if best_a.sum() > 0.0:
                    best_a = best_a / best_a.sum()
                else:
                    best_a = f.copy()

    return _normalize(best_a), best_s


def _lp_direction(f, alpha0, tl):
    if linprog is None:
        return None
    f = _normalize(np.asarray(f, dtype=np.float64).ravel())
    n = f.size
    L = 2 * n - 1
    F = f * float(n)

    try:
        A = np.zeros((L, n), dtype=np.float64)
        for j in range(n):
            A[j:j + n, j] = F

        I = np.eye(L, dtype=np.float64)
        c_obj = np.zeros(n + 1, dtype=np.float64)
        c_obj[-1] = 1.0

        if abs(float(alpha0)) < 1e-8:
            A_ub = np.hstack((A, -I))
            b_ub = np.zeros(L, dtype=np.float64)
        else:
            aa = min(0.99, max(0.001, float(alpha0)))
            coef = 2.0 * aa * (1.0 - aa)
            cf = np.convolve(f, f)
            base = float(n) * ((1.0 - aa) * (1.0 - aa)) * cf
            A_ub = np.hstack((coef * A, -I))
            b_ub = -base

        A_eq = np.zeros((1, n + 1), dtype=np.float64)
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0], dtype=np.float64)
        bounds = [(0.0, None)] * n + [(0.0, None)]

        try:
            r = linprog(
                c_obj,
                A_ub=A_ub,
                b_ub=b_ub,
                A_eq=A_eq,
                b_eq=b_eq,
                bounds=bounds,
                method="highs",
                options={"presolve": True, "time_limit": float(max(1.0, tl))},
            )
        except Exception:
            r = linprog(
                c_obj,
                A_ub=A_ub,
                b_ub=b_ub,
                A_eq=A_eq,
                b_eq=b_eq,
                bounds=bounds,
                method="highs",
            )

        x = getattr(r, "x", None)
        if x is None:
            return None
        g = np.asarray(x[:n], dtype=np.float64)
        g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
        g = np.clip(g, 0.0, None)
        sg = float(g.sum())
        if not math.isfinite(sg) or sg <= 1e-300:
            return None
        return g / sg
    except Exception:
        return None


def _lp_refine(a0, limit, best, rng):
    if linprog is None:
        return None
    a = _normalize(a0)
    s = _score(a)
    _update_best(a, best)
    best_a = a.copy()
    best_s = s
    stall = 0
    it = 0
    alphas = (0.0, 0.05, 0.10, 0.20, 0.35)

    while time.monotonic() < limit - 5.0:
        it += 1
        aa = alphas[it % len(alphas)]
        rem = limit - time.monotonic()
        tl = min(20.0, max(2.0, rem * 0.18 if it < 4 else rem * 0.10))

        g = _lp_direction(a, aa, tl)
        if g is None and it % 2 == 0:
            g = _lp_direction(a, 0.0, max(1.0, tl * 0.6))

        if g is not None:
            cand, cs = _line_search_convex(a, g, s)
            if math.isfinite(cs) and cs < s * (1.0 - 1e-12):
                a = _normalize(cand)
                s = cs
                if s < best_s:
                    best_s = s
                    best_a = a.copy()
                _update_best(a, best)
                stall = 0
                continue

        stall += 1
        if stall >= 3:
            sigma = 0.08
            try:
                pa = _normalize(a * np.exp(np.clip(rng.normal(0.0, sigma, a.size), -5.0, 5.0)) + 1e-9)
                ps = _score(pa)
                if math.isfinite(ps) and ps < max(s, best_s) * (1.0 - 1e-12):
                    a = pa
                    s = ps
                    if s < best_s:
                        best_s = s
                        best_a = a.copy()
                    _update_best(a, best)
                    stall = 0
                    continue
            except Exception:
                pass

        if stall > 18:
            break

    return best_a


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        rem = float(budget_s)
        if not math.isfinite(rem) or rem <= 0.0:
            rem = 1300.0
    except Exception:
        rem = 1300.0
    rem = max(1.0, min(rem, 1300.0))
    head = min(30.0, max(1.0, 0.03 * rem))
    deadline = t0 + max(1.0, rem - head)

    rng = np.random.default_rng(int(seed) if seed is not None else 42)
    best = [float("inf"), np.array([1.0, 1.0], dtype=np.float64)]

    par = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            par = _normalize(h)
            if par.size >= 2:
                _update_best(par, best)
    except Exception:
        par = None

    if best[0] == float("inf") or not math.isfinite(best[0]):
        a0 = np.ones(192, dtype=np.float64) / 192.0
        _update_best(a0, best)
        par = a0

    cur = _normalize(best[1])

    # Primary local search: exact one-coordinate exact descent.
    reserve = min(380.0, max(80.0, 0.30 * rem))
    if time.monotonic() < deadline - reserve:
        cur = _coordinate_refine(cur, deadline - reserve, best, rng)
        if best[1] is not None:
            cur = _normalize(best[1])

    # Brief escape attempts by noisy coordinate restarts.
    for sig in (0.04, 0.10, 0.22):
        if time.monotonic() >= deadline - 250.0:
            break
        try:
            start = _normalize(cur * np.exp(np.clip(rng.normal(0.0, sig, cur.size), -5.0, 5.0)) + 1e-9)
            _update_best(start, best)
        except Exception:
            continue
        lim = min(deadline - 250.0, time.monotonic() + 35.0)
        if lim > time.monotonic() + 5.0:
            _coordinate_refine(start, lim, best, rng)
        cur = _normalize(best[1])

    # Smoothed nonconvex refinement.
    if time.monotonic() < deadline - 130.0:
        lim = min(deadline - 130.0, time.monotonic() + 180.0)
        if lim > time.monotonic() + 10.0 and minimize is not None:
            _smooth_refine(best[1], lim, best, rng)
        cur = _normalize(best[1])

    # Linearized linear-programming direction refinement.
    if time.monotonic() < deadline - 90.0:
        lim = min(deadline - 90.0, time.monotonic() + 120.0)
        if lim > time.monotonic() + 10.0 and linprog is not None:
            _lp_refine(best[1], lim, best, rng)
        cur = _normalize(best[1])

    # Final exact coordinate polish.
    if time.monotonic() < deadline - 20.0:
        _coordinate_refine(best[1], deadline - 20.0, best, rng)

    # If there is a little more time, try a nearby resolution.
    if time.monotonic() < deadline - 60.0:
        try:
            n_cur = int(best[1].size)
            for nl in (n_cur + 16, n_cur + 48, n_cur + 64):
                if time.monotonic() >= deadline - 60.0:
                    break
                if nl <= n_cur:
                    continue
                start = _interpolate(best[1], nl)
                _update_best(start, best)
                lim = min(deadline - 60.0, time.monotonic() + 35.0)
                if lim > time.monotonic() + 5.0:
                    _coordinate_refine(start, lim, best, rng)
        except Exception:
            pass

    fin = _normalize(best[1])
    fin = np.asarray(fin, dtype=np.float64).ravel()
    fin = np.nan_to_num(fin, nan=0.0, posinf=0.0, neginf=0.0)
    fin = np.clip(fin, 0.0, 1000.0)
    if fin.size < 1 or float(fin.sum()) < 0.01:
        fin = np.ones(192, dtype=np.float64) / 192.0

    try:
        v = evaluate_sequence([float(x) for x in fin])
        if not math.isfinite(v):
            fin = np.ones(192, dtype=np.float64) / 192.0
    except Exception:
        v = _score(fin)
        if not math.isfinite(v):
            fin = np.ones(192, dtype=np.float64) / 192.0

    return [float(x) for x in fin]
