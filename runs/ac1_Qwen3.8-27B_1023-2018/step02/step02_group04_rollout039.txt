        a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
        a = np.clip(a, 0.0, None)
        if a.size == n and float(a.sum()) > 1e-300:
            starts.append(a / a.sum())

    _add(np.ones(n))
    for c in (0.5, 1.0, 2.0, 3.0, -0.5, -1.0, -2.0, -3.0):
        _add(np.exp(c * (x - 0.5)))
    for k in range(1, min(10, n // 2 + 1)):
        for amp in (0.3, 0.5, 0.7, 0.9):
            _add(1.0 + amp * np.sin(2.0 * np.pi * k * (x - 0.5)))
    for al in (0.2, 0.5, 1.0):
        try:
            _add(rng.dirichlet(np.full(n, al)))
        except Exception:
            pass
    if par is not None and len(par) >= 2:
        tp = np.linspace(0.0, 1.0, len(par))
        _add(np.interp(x, tp, par))
        _add(np.interp(x, tp, par) * np.exp(0.05 * rng.standard_normal(n)))
    if not starts:
        starts.append(np.ones(n) / n)
    starts.sort(key=_sc)
    return starts


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    try:
        rem = max(0.0, float(budget_s))
    except Exception:
        rem = 1300.0
    deadline = t0 + rem - 20.0

    best = [float('inf'), np.array([1.0, 1.0], dtype=np.float64)]
    _upd(best[1], best)

    par = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            par = np.asarray(h, dtype=np.float64).ravel()
            par = np.nan_to_num(par, nan=0.0, posinf=0.0, neginf=0.0)
            par = np.clip(par, 0.0, 1000.0)
            if par.size >= 2:
                _upd(par, best)
    except Exception:
        par = None

    n_par = par.size if par is not None else 192

    # Phase 1: Aggressive LP at parent length (primary)
    if time.monotonic() < deadline - 350.0:
        d1 = deadline - 350.0
        starts = _gen_starts(n_par, rng, best[1])
        for idx, s in enumerate(starts[:6]):
            if time.monotonic() >= d1 - 5.0:
                break
            alloc = min(100.0, (d1 - 5.0 - time.monotonic()) * 0.45)
            if alloc < 8.0:
                break
            _lp_iterate(s, time.monotonic() + alloc, best, rng, max_it=100000)
        # Continue LP from best
        if time.monotonic() < d1 - 5.0:
            _lp_iterate(best[1], d1 - 2.0, best, rng, max_it=100000)

    # Phase 2: V-gradient at parent length
    if time.monotonic() < deadline - 200.0:
        d2 = deadline - 200.0
        _v_grad_descent(best[1], d2 - 2.0, best, rng, n_target=n_par)
        # LP from the v-gradient result
        if time.monotonic() < d2 - 2.0:
            _lp_iterate(best[1], d2 - 1.0, best, rng, max_it=50000)

    # Phase 3: LP at n=128
    if time.monotonic() < deadline - 100.0:
        d3 = deadline - 100.0
        starts128 = _gen_starts(128, rng, best[1])
        for s in starts128[:4]:
            if time.monotonic() >= d3 - 5.0:
                break
            alloc = min(60.0, (d3 - 5.0 - time.monotonic()) * 0.4)
            if alloc < 8.0:
                break
            _lp_iterate(s, time.monotonic() + alloc, best, rng, max_it=50000)
        if time.monotonic() < d3 - 2.0:
            _lp_iterate(best[1], d3 - 1.0, best, rng, max_it=30000)

    # Phase 4: V-gradient at best length
    if time.monotonic() < deadline - 60.0:
        d4 = deadline - 60.0
        _v_grad_descent(best[1], d4 - 2.0, best, rng, n_target=None)
        if time.monotonic() < d4 - 1.0:
            _lp_iterate(best[1], d4 - 0.5, best, rng, max_it=20000)

    # Phase 5: Final LP polish
    if time.monotonic() < deadline - 5.0:
        _lp_iterate(best[1], deadline - 2.0, best, rng, max_it=20000)

    # Validate and return
    fin = np.asarray(best[1], dtype=np.float64).ravel().copy()
    fin = np.nan_to_num(fin, nan=0.0, posinf=0.0, neginf=0.0)
    fin = np.clip(fin, 0.0, 1000.0)
    if fin.size < 1:
        fin = np.array([1.0])
    if float(fin.sum()) < 0.01:
        fin = np.ones(fin.size)
    _upd(fin, best)
    fin = np.asarray(best[1], dtype=np.float64).ravel()
    v = evaluate_sequence([float(x) for x in fin])
    if not math.isfinite(v):
        fin = np.array([1.0])
    return [float(x) for x in fin]
