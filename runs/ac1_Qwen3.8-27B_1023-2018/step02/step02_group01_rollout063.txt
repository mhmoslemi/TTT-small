    while time.monotonic() < stop - 0.3:
        b = _cvs(a)
        M = float(b.max())
        # Smooth subgradient: weighted average over near-max positions
        sig = max(M * 0.01, 1e-10)
        w = np.exp((b - M) / sig)
        wsum = w.sum()
        if wsum < 1e-10:
            w = np.zeros_like(b); w[b.argmax()] = 1.0
        else:
            w /= wsum
        # Compute weighted subgradient
        grad = np.zeros(n)
        for k in range(len(w)):
            if w[k] < 1e-8: continue
            t1i, t2i = k - j, j - k
            o1 = (t1i >= 0) & (t1i < n)
            o2 = (t2i >= 0) & (t2i < n)
            grad += w[k] * (np.where(o1, a[np.clip(t1i,0,n-1)], 0.0) +
                           np.where(o2, a[np.clip(t2i,0,n-1)], 0.0))
        # Mirror descent
        eta = max(1e-4, 0.03 * (1+stuck*0.01)**(-1))
        an = a * np.exp(-eta * grad)
        an = np.maximum(an, 1e-15)
        an /= an.sum()
        cm = float(_cvs(an).max())
        if cm < M - 1e-16:
            a = an
            if cm < bestM - 1e-15:
                bestM, best, stuck = cm, a.copy(), 0
            else: stuck += 1
        else:
            stuck += 1
            if stuck % 500 == 0:
                a = a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 1e-10); a /= a.sum()
            if stuck > 3000: break
    return best
