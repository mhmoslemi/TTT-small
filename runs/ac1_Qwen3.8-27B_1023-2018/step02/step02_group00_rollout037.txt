    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n):
    rows = np.arange(2 * n - 1)[:, None]
    cols = np.arange(n)[None, :]
    idx = rows - cols
    mask = (idx >= 0) & (idx < n)
    A = np.zeros((2 * n - 1, n))
    A[mask] = f[idx[mask]]
    return A


def _lp_solve(f, n, M):
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        with warnings.catch_warnings():
            warnings.simplefilter("ignore")
            res = linproj = linprog(c, A_ub=A, b_ub=b_ub,
                                    bounds=[(0.0, None)] * n,
                                    method='highs',
                                    options={'presolve': True})
    except Exception:
        return None, 0.0
    if res is None or res.status != 0 or res.x is None:
        return None, 0.0
    g = np.clip(np.asarray(res.x, dtype=np.float64), 0, None)
    gs = float(g.sum())
    if gs < 1e-12:
        return None, 0.0
    return g / gs, gs


def _optimize_lp(f0, n, stop_time, seed):
    rng = np.random.default_rng(seed)
    f = np.clip(f0.astype(np.float64).copy(), 0, None)
    fs = f.sum()
    if fs < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / fs

    best_f = f.copy()
    best_score = _score(f)

    alpha = 0.35
    alpha_min, alpha_max = 0.02, 0.65
    stall = 0
    max_stall = 1000
    it = 0
    last_improve_it = 0

    while time.monotonic() < stop_time - 2.0:
        M = float(_conv_full(f).max())
        g, gs = _lp_solve(f, n, M)

        if g is None:
            stall += 8
            if stall > max_stall:
                break
            f = np.clip(f + rng.normal(0, 0.02, n), 0, None)
            fs = f.sum()
            f = f / fs if fs > 1e-10 else best_f.copy()
            continue

        if np.linalg.norm(g - f) < 1e-9:
            stall += 2
            if stall > max_stall:
                break
            f = np.clip(f + rng.normal(0, 0.04, n), 0, None)
            fs = f.sum()
            f = f / fs if fs > 1e-10 else best_f.copy()
            continue

        improved = False
        for _ in range(20):
            f_new = (1.0 - alpha) * f + alpha * g
            f_new = np.clip(f_new, 0, None)
            fs_new = f_new.sum()
            if fs_new < 1e-10:
                alpha = max(alpha_min, alpha * 0.5)
                continue
            f_new = f_new / fs_new
            sc = _score(f_new)
            if sc < best_score - 1e-14:
                f = f_new
                best_f = f.copy()
                best_score = sc
                improved = True
                alpha = min(alpha_max, alpha * 1.03)
                stall = 0
                last_improve_it = it
                break
            else:
                alpha = max(alpha_min, alpha * 0.55)

        if not improved:
            stall += 1
            if stall >= max_stall:
                f = np.clip(f + rng.normal(0, 0.08, n), 0, None)
                fs = f.sum()
                f = f / fs if fs > 1e-10 else best_f.copy()
                alpha = 0.35
                stall = 0
        it += 1

    return best_f, best_score


def _subgradient_polish(f0, stop_time, seed):
    rng = np.random.default_rng(seed)
    f = np.clip(f0.astype(np.float64).copy(), 0, None)
    fs = f.sum()
    if fs < 1e-10:
        return f, math.inf
    f = f / fs

    best_f = f.copy()
    best_score = _score(f)
    step = 0.08
    no_imp = 0
    max_no_imp = 300

    while time.monotonic() < stop_time - 2.0 and no_imp < max_no_imp:
        b = _conv_full(f)
        M = float(b.max())
        if M < 1e-15:
            break

        # Active lags
        frac = max(0.95, 1.0 - 0.5 * min(1.0, no_imp / 100.0))
        threshold = M * frac
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        # Compute subgradient: sg[j] = (1/|A|) * sum_{k in active} 2*f[k-j]
        n = len(f)
        sg = np.zeros(n)
        for k in active:
            j_lo = max(0, k - n + 1)
            j_hi = min(k, n - 1)
            if j_lo <= j_hi:
                sg[j_lo:j_hi + 1] += 2.0 * f[k - j_hi:k - j_lo - 1:-1]
        sg /= len(active)

        # Projected gradient step
        f_new = f - step * sg
        f_new = np.maximum(f_new, 0)
        fs_new = f_new.sum()
        if fs_new > 1e-10:
            f_new = f_new / fs_new
            sc = _score(f_new)
            if sc < best_score - 1e-14:
                f = f_new
                best_f = f.copy()
                best_score = sc
                no_imp = 0
                step = min(0.4, step * 1.05)
            else:
                step *= 0.75
                no_imp += 1
                if step < 1e-8:
                    break
        else:
            break

    return best_f, best_score


def _coordinate_polish(f0, stop_time, seed):
    n = len(f0)
    best = np.clip(f0.astype(np.float64).copy(), 0, None)
    fs = best.sum()
    if fs < 1e-10:
        return best, _score(best)
    best = best / fs
    best_sc = _score(best)
    rng = np.random.default_rng(seed)
    no_imp = 0

    while time.monotonic() < stop_time - 2.0 and no_imp < 25:
        improved = False

        # 1D coordinate moves
        for i in rng.permutation(n):
            if time.monotonic() > stop_time - 2.0:
                break
            orig = best[i]
            best_val = orig
            best_val_sc = best_sc
            for mlt in (0.0, 0.25, 0.5, 0.7, 0.85, 1.15, 1.35, 1.7, 2.5):
                if mlt == 0.0:
                    best[i] = 0.0
                elif orig > 1e-12:
                    best[i] = orig * mlt
                else:
                    best[i] = 0.01
                if best.sum() < 0.01:
                    best[i] = orig
                    continue
                sc = _score(best)
                if sc < best_val_sc - 1e-14:
                    best_val = best[i]
                    best_val_sc = sc
            best[i] = best_val
            if best_val_sc < best_sc - 1e-14:
                best_sc = best_val_sc
                improved = True

        # 2D moves
        for _ in range(60):
            if time.monotonic() > stop_time - 2.0:
                break
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            for mi, mj in [(0.5, 1.5), (1.5, 0.5), (0.3, 2.0), (2.0, 0.3),
                           (0.7, 1.3), (1.3, 0.7), (0.0, 1.2), (1.2, 0.0)]:
                best[i] = oi * mi if (oi > 1e-12 and mi > 0) else (0.01 if mi > 0 else 0.0)
                best[j] = oj * mj if (oj > 1e-12 and mj > 0) else (0.01 if mj > 0 else 0.0)
                if best.sum() < 0.01:
                    best[i], best[j] = oi, oj
                    continue
                sc = _score(best)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    improved = True
            best[i], best[j] = oi, oj

        no_imp = 0 if improved else no_imp + 1

    return best, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 15.0

    parent = _get_parent()
    n_p = len(parent)

    # Fallback
    fallback = parent.copy()
    fallback_sc = _score(fallback)
    if not math.isfinite(fallback_sc):
        fallback = np.ones(32)
        fallback_sc = _score(fallback)

    best_seq = fallback.copy()
    best_score = fallback_sc

    def _try_update(res):
        nonlocal best_seq, best_score
        if res is None:
            return
        res = np.asarray(res, dtype=np.float64)
        if not np.all(np.isfinite(res)) or res.sum() < 0.01:
            return
        res = np.clip(res, 0, 1000)
        try:
            exact = float(evaluate_sequence([float(x) for x in res]))
        except Exception:
            return
        if math.isfinite(exact) and exact < best_score:
            best_seq = res.copy()
            best_score = exact

    # Phase 1: LP optimization on parent length (55% of budget)
    phase1_end = t0 + budget * 0.55
    rng = np.random.default_rng(seed)

    # Primary: parent initialization (allocate most time)
    stop1 = min(phase1_end - 5, time.monotonic() + budget * 0.38)
    try:
        res, sc = _optimize_lp(parent, n_p, stop1, seed)
        _try_update(res)
    except Exception:
        pass

    # Secondary initializations
    inits = []
    i_arr = np.arange(n_p, dtype=np.float64)
    inits.append(0.5 + 0.5 * np.cos(2 * math.pi * i_arr / n_p))
    sup = 0.5 + 0.35 * np.cos(2 * math.pi * i_arr / n_p) + 0.2 * np.cos(4 * math.pi * i_arr / n_p + 0.7)
    inits.append(np.abs(sup) + 0.01)
    inits.append(np.abs(rng.normal(0, 1, n_p)) + 0.05)
    inits.append(np.abs(parent + rng.normal(0, 0.3 * max(1.0, np.mean(parent)), n_p)) + 0.01)

    for idx, init in enumerate(inits):
        if time.monotonic() > phase1_end - 10:
            break
        if best_score < 1.503:
            break
        remaining = phase1_end - time.monotonic()
        per_init = max(8.0, remaining / (len(inits) - idx))
        stop = min(phase1_end - 5, time.monotonic() + per_init)
        try:
            res, sc = _optimize_lp(init, n_p, stop, seed + 5000 + idx * 100)
            _try_update(res)
        except Exception:
            pass

    # Phase 2: Subgradient polish (15% of budget)
    phase2_end = t0 + budget * 0.72
    if time.monotonic() < phase2_end - 10:
        sg_time = min(phase2_end - time.monotonic(), budget * 0.15)
        stop_sg = time.monotonic() + sg_time * 0.9
        try:
            res, sc = _subgradient_polish(best_seq, stop_sg, seed + 9000)
            _try_update(res)
        except Exception:
            pass

    # Phase 3: Coordinate polish + final subgradient (remaining - headroom)
    polish_end = deadline - 5.0

    if time.monotonic() < polish_end - 10:
        coord_time = min(polish_end - time.monotonic() - 5, budget * 0.12)
        stop_coord = time.monotonic() + coord_time
        try:
            res, sc = _coordinate_polish(best_seq, stop_coord, seed + 8000)
            _try_update(res)
        except Exception:
            pass

    # Final subgradient pass
    if time.monotonic() < polish_end - 5:
        stop_final = polish_end - 3
        try:
            res, sc = _subgradient_polish(best_seq, stop_final, seed + 9500)
            _try_update(res)
        except Exception:
            pass

    # One more coordinate pass if time remains
    if time.monotonic() < polish_end - 3:
        try:
            res, sc = _coordinate_polish(best_seq, polish_end - 1, seed + 8500)
            _try_update(res)
        except Exception:
            pass

    # Final validation
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        return [float(x) for x in fallback]

    return [float(x) for x in best_seq]
