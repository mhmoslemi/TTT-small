    # Inverse position
    variants.append((1 - t) ** 0.5)

    # Sinc-like
    variants.append(np.abs(np.sin(np.pi * x) / (np.pi * x + 1e-6)) + 0.1)

    # Bumps at edges
    e = 0.15
    a = np.ones(n)
    a[:int(e * n)] *= 2.5
    a[-int(e * n):] *= 2.5
    variants.append(a)

    # Valley in middle
    a = np.ones(n)
    mid = n // 2
    w = n // 4
    a[mid - w:mid + w] *= 0.4
    variants.append(a)

    for v in variants:
        v = np.maximum(v, 1e-6)
        v = v / v.sum()
    return variants


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    total = max(1.0, float(budget_s))
    hard_deadline = t0 + total - 4.0

    seed = int(seed)

    # Get parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            if p.size > 0 and np.isfinite(p).all():
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        parent = None

    # Initial best
    best_a = None
    best_score = math.inf

    # Try parent first
    if parent is not None and parent.size > 0:
        try:
            cand = [float(v) for v in parent]
            fs = float(evaluate_sequence(cand))
            if np.isfinite(fs):
                best_a = parent.copy()
                best_score = fs
        except Exception:
            pass

    if best_a is None:
        best_a = np.ones(32) / 32.0
        best_score = 2.0
        try:
            fs = float(evaluate_sequence([1.0 / 1.0]))
            if np.isfinite(fs):
                best_a = np.array([1.0])
                best_score = fs
        except Exception:
            pass

    # Determine which lengths to try
    # The parent length is preferred; also try a couple others
    sizes = []
    if parent is not None and parent.size > 0:
        sizes.append(parent.size)
    for s in [128, 192, 256, 320]:
        if s not in sizes:
            sizes.append(s)

    # Phase 1: Multi-start optimization (70% of budget)
    phase1_end = t0 + 0.65 * total
    # Phase 2: Best polish (30% of budget)
    phase2_end = t0 + 0.93 * total

    n_best = best_a.size
    w_list = [0.0, 0.05, 0.1, 0.2, 0.0]
    tau_list = [0.1, 0.15, 0.2, 0.08, 0.12]

    job_count = 0
    all_jobs = []

    for size in sizes:
        variants = _init_variants(size, seed + size)
        if parent is not None and size == parent.size:
            # Include parent as first variant
            vp = np.maximum(parent, 1e-6)
            vp = vp / vp.sum()
            variants.insert(0, vp.copy())

        for vi, v in enumerate(variants[:8]):  # limit to 8 per size
            w = w_list[job_count % len(w_list)]
            tau = tau_list[job_count % len(tau_list)]
            all_jobs.append((size, v, w, tau, seed + job_count * 31))
            job_count += 1

    # Round-robin over jobs
    job_results = [None] * len(all_jobs)
    attempts_per_job = 3

    for attempt in range(attempts_per_job):
        if time.monotonic() >= phase1_end:
            break
        for ji in range(len(all_jobs)):
            if time.monotonic() >= phase1_end:
                break
            size, v, w, tau, s = all_jobs[ji]
            # Time slice: give each job a fair share
            remaining = phase1_end - time.monotonic()
            n_jobs_remaining = (len(all_jobs) - ji) + (attempts_per_job - 1 - attempt) * len(all_jobs)
            slice_time = max(2.0, min(60.0, remaining / max(1, n_jobs_remaining)))
            end_time = time.monotonic() + slice_time

            try:
                a_res, s_res = _optimize(v, end_time, hard_deadline, w=w, tau_init=tau, seed=s)
                if s_res < best_score:
                    best_a = a_res.copy()
                    best_score = s_res
                if job_results[ji] is None or s_res < job_results[ji][1]:
                    job_results[ji] = (a_res, s_res)
            except Exception:
                pass

    # Also try the best from each job with w=0 (pure peak reduction)
    if time.monotonic() < phase1_end:
        for ji, (a_j, s_j) in enumerate(job_results):
            if time.monotonic() >= phase1_end:
                break
            if a_j is not None:
                size = a_j.size
                end_time = min(phase1_end, time.monotonic() + 30.0)
                try:
                    a_res, s_res = _optimize(a_j, end_time, hard_deadline, w=0.0, tau_init=0.05, seed=seed + 999 + ji)
                    if s_res < best_score:
                        best_a = a_res.copy()
                        best_score = s_res
                except Exception:
                    pass

    # Phase 2: Polish the best
    if time.monotonic() < phase2_end - 2.0:
        try:
            a_pol, s_pol = _polish(best_a, phase2_end, hard_deadline)
            if s_pol < best_score:
                best_a = a_pol.copy()
                best_score = s_pol
        except Exception:
            pass

    # One more optimization pass from the polished result
    if time.monotonic() < hard_deadline - 5.0:
        try:
            end_t = min(hard_deadline - 3.0, time.monotonic() + max(5.0, 0.1 * total))
            a_res, s_res = _optimize(best_a, end_t, hard_deadline, w=0.0, tau_init=0.03, seed=seed + 5555)
            if s_res < best_score:
                best_a = a_res.copy()
                best_score = s_res
        except Exception:
            pass

    # Final polish
    if time.monotonic() < hard_deadline - 2.0:
        try:
            a_pol, s_pol = _polish(best_a, hard_deadline - 1.0, hard_deadline)
            if s_pol < best_score:
                best_a = a_pol.copy()
                best_score = s_pol
        except Exception:
            pass

    # Final validation with exact evaluator
    try:
        cand = [float(v) for v in best_a]
    except Exception:
        cand = [1.0]

    try:
        fs = float(evaluate_sequence(cand))
    except Exception:
        fs = math.inf

    if not np.isfinite(fs):
        # Fallback
        try:
            cand = [1.0 / i for i in range(1, 3)]  # [1, 0.5]
            fs = float(evaluate_sequence(cand))
        except Exception:
            return [1.0]

    # If the evaluator score is much worse than our internal score, something went wrong
    if np.isfinite(fs) and fs < best_score + 0.1:
        best_score = fs

    return cand
