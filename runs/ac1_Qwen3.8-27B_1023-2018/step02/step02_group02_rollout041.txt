def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + max(1.0, float(budget_s)) - 10.0

    # --- Retrieve parent ---
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
    except Exception:
        parent = np.ones(192, dtype=np.float64)
    parent = np.clip(parent, 0.0, 1000.0)
    if parent.size == 0 or not np.isfinite(parent).all() or parent.size < 2:
        parent = np.ones(192, dtype=np.float64)

    n = int(parent.size)
    s = float(parent.sum())
    if s < 0.01:
        parent = np.ones(n, dtype=np.float64)
        s = 1.0
    f = parent / s  # normalize to sum = 1

    fsize = next_fast_len(2 * n - 1)

    def convolve_full(a):
        F = rfft(a, fsize)
        c = irfft(F * F, fsize)[:2 * n - 1]
        c = np.maximum(c, 0.0)
        c[~np.isfinite(c)] = 0.0
        return c

    def peak(c):
        return float(c.max())

    best_score = 2.0 * n * peak(convolve_full(f))
    best_f = f.copy()
    rng = np.random.default_rng(seed)

    # --- LP direction computation ---
    def build_B(f_curr):
        B = np.zeros((2 * n - 1, n), dtype=np.float64)
        for i in range(n):
            B[i:i + n, i] = f_curr
        return B

    def lp_direction(f_curr, tl=2.0):
        c_curr = convolve_full(f_curr)
        M = peak(c_curr)
        if M < 1e-16:
            return None, M

        B = build_B(f_curr)
        c_obj = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        bounds = [(0.0, None)] * n

        try:
            res = linprog(c_obj, A_ub=B, b_ub=b_ub, bounds=bounds,
                          method='highs',
                          options={'presolve': True, 'time_limit': tl})
        except Exception:
            return None, M

        if not res.success or res.x is None:
            return None, M

        g = np.maximum(np.asarray(res.x, dtype=np.float64).ravel(), 0.0)
        gs = float(g.sum())
        if gs < 1e-12:
            return None, M
        g = g / gs

        if float(np.linalg.norm(g - f_curr, 1)) < 1e-9:
            return None, M

        return g, M

    # --- Line search ---
    def line_search(f_curr, g, M_curr):
        best_t = 0.0
        best_M = M_curr

        # Try a range of t values
        for t in (1.0, 0.9, 0.8, 0.7, 0.6, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05, 0.02):
            h = (1.0 - t) * f_curr + t * g
            M_h = peak(convolve_full(h))
            if M_h < best_M - 1e-15:
                best_M = M_h
                best_t = t

        if best_t == 0.0:
            return 0.0, M_curr

        # Refine with golden section around best_t
        lo = max(0.01, best_t - 0.05)
        hi = min(0.99, best_t + 0.05)
        gr = (math.sqrt(5.0) - 1.0) / 2.0

        for _ in range(12):
            if hi - lo < 1e-5:
                break
            d = hi - lo
            c1 = hi - gr * d
            c2 = lo + gr * d
            h1 = (1.0 - c1) * f_curr + c1 * g
            h2 = (1.0 - c2) * f_curr + c2 * g
            M1 = peak(convolve_full(h1))
            M2 = peak(convolve_full(h2))
            if M1 < M2:
                hi = c2
            else:
                lo = c1
            mid = (lo + hi) / 2.0
            h_mid = (1.0 - mid) * f_curr + mid * g
            M_mid = peak(convolve_full(h_mid))
            if M_mid < best_M - 1e-15:
                best_M = M_mid
                best_t = mid

        return best_t, best_M

    # ===================================================================
    # PHASE 1: LP-based descent (main workhorse)
    # ===================================================================
    f = best_f.copy()
    stuck = 0
    best_streak = 0

    for it in range(100000):
        if time.monotonic() > deadline - 300.0:
            break

        remaining = deadline - time.monotonic()
        tl = min(3.0, max(0.3, remaining * 0.01))

        g, M_curr = lp_direction(f, tl=tl)

        if g is None:
            stuck += 1
            if stuck > 50:
                # Perturb to escape
                sigma = min(0.15, 0.02 + 0.01 * stuck / 50.0)
                f = np.maximum(f * np.exp(rng.normal(0.0, sigma, n)), 1e-15)
                f /= f.sum()
                stuck = 0
            continue

        stuck = 0
        best_t, best_M = line_search(f, g, M_curr)

        if best_t > 0:
            f = (1.0 - best_t) * f + best_t * g
            sc = 2.0 * n * best_M
            if sc < best_score - 1e-14:
                best_score = sc
                best_f = f.copy()
                best_streak = 0
            else:
                best_streak += 1

            # If no improvement over many steps, perturb
            if best_streak > 200:
                f = np.maximum(f * np.exp(rng.normal(0.0, 0.03, n)), 1e-15)
                f /= f.sum()
                best_streak = 0
        else:
            # LP gave a direction but line search found no improvement
            stuck += 1
            if stuck > 30:
                f = np.maximum(f * np.exp(rng.normal(0.0, 0.02, n)), 1e-15)
                f /= f.sum()
                stuck = 0

    # ===================================================================
    # PHASE 2: Gradient-based refinement with soft-max weights
    # ===================================================================
    if time.monotonic() < deadline - 60.0:
        a = best_f.copy()
        sc = 2.0 * n * peak(convolve_full(a))
        if sc < best_score - 1e-14:
            best_score = sc
            best_f = a.copy()

        lr = 0.1
        stuck = 0

        for it in range(100000):
            if time.monotonic() > deadline - 60.0:
                break
            if stuck > 600:
                a = np.maximum(a * np.exp(rng.normal(0.0, 0.04, n)), 1e-15)
                a /= a.sum()
                stuck = 0
                lr = 0.1
                continue

            c = convolve_full(a)
            M = peak(c)
            sc = 2.0 * n * M
            if sc < best_score - 1e-14:
                best_score = sc
                best_f = a.copy()
                stuck = 0

            # Soft-max weights over peaks
            tau = max(1e-12, 0.005 * M)
            x = np.clip((c - M) / tau, -40.0, 0.0)
            w = np.exp(x)
            w_sum = float(w.sum())
            if w_sum < 1e-30:
                w = np.zeros(2 * n - 1)
                w[int(np.argmax(c))] = 1.0
            else:
                w /= w_sum

            # Subgradient: grad[i] = sum_k w[k] * a[k-i] (with bounds)
            grad = np.zeros(n, dtype=np.float64)
            for i in range(n):
                max_j = min(n - 1, 2 * n - 2 - i)
                grad[i] = float(np.dot(w[i:i + max_j + 1], a[:max_j + 1]))

            # Project onto simplex tangent space
            grad_centered = grad - float(np.mean(grad))
            sn = float(np.linalg.norm(grad_centered))

            if sn < 1e-14:
                stuck += 1
                if stuck % 50 == 0:
                    a = np.maximum(a * np.exp(rng.normal(0.0, 0.02, n)), 1e-15)
                    a /= a.sum()
                    stuck = 0
                continue

            improved = False
            for scale in (lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 0.05, lr * 0.01):
                cand = a - scale * grad_centered / sn
                cand = np.maximum(cand, 1e-15)
                cs = float(cand.sum())
                if cs < 1e-10:
                    continue
                cand /= cs
                cand_sc = 2.0 * n * peak(convolve_full(cand))
                if cand_sc < sc - 1e-14:
                    a = cand
                    sc = cand_sc
                    improved = True
                    break

            if improved:
                stuck = 0
                lr = min(lr * 1.03, 3.0)
            else:
                stuck += 1
                lr = max(0.005, lr * 0.9)

    # ===================================================================
    # PHASE 3: Final LP polish from best solution
    # ===================================================================
    if time.monotonic() < deadline - 15.0:
        f = best_f.copy()
        for _ in range(500):
            if time.monotonic() > deadline - 15.0:
                break
            remaining = deadline - time.monotonic()
            tl = min(2.0, max(0.2, remaining * 0.2))
            g, M_curr = lp_direction(f, tl=tl)
            if g is None:
                break

            best_t, best_M = line_search(f, g, M_curr)
            if best_t > 0:
                f = (1.0 - best_t) * f + best_t * g
                sc = 2.0 * n * best_M
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_f = f.copy()
            else:
                break

    # ===================================================================
    # PHASE 4: Short stochastic local search for final polish
    # ===================================================================
    if time.monotonic() < deadline - 5.0:
        a = best_f.copy()
        sc = best_score
        rng2 = np.random.default_rng(seed + 7777)
        sigma = 0.01
        for _ in range(5000):
            if time.monotonic() > deadline - 5.0:
                break

            r = rng2.random()
            if r < 0.5:
                # Multiplicative perturbation
                cand = a * np.exp(rng2.normal(0.0, sigma, n))
            elif r < 0.8:
                # Two-index swap
                cand = a.copy()
                i = int(rng2.integers(0, n))
                j = int(rng2.integers(0, n))
                amt = sigma * a.sum()
                cand[i] += amt
                cand[j] -= amt
                cand = np.maximum(cand, 1e-15)
            else:
                # Local bump
                cand = a.copy()
                k = int(rng2.integers(0, n))
                width = max(1, n // 50)
                lo = max(0, k - width)
                hi = min(n, k + width + 1)
                cand[lo:hi] *= float(np.exp(rng2.normal(0.0, sigma * 2)))

            cand = np.maximum(cand, 1e-15)
            cand /= cand.sum()
            cand_sc = 2.0 * n * peak(convolve_full(cand))
            if cand_sc < sc - 1e-14:
                a = cand
                sc = cand_sc
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_f = a.copy()
                sigma = max(0.003, sigma * 0.95)
            else:
                sigma = min(0.1, sigma * 1.05)

    # ===================================================================
    # Final validation
    # ===================================================================
    best_f = np.clip(best_f, 0.0, 1000.0)
    if not np.isfinite(best_f).all() or best_f.sum() < 0.01:
        best_f = np.ones(n) / n

    final_list = [float(x) for x in best_f]
    try:
        fs = float(evaluate_sequence(final_list))
    except Exception:
        fs = float('inf')

    if not np.isfinite(fs):
        final_list = [1.0 / n] * n
        try:
            fs = float(evaluate_sequence(final_list))
        except Exception:
            pass
        if not np.isfinite(fs):
            return [1.0]

    return final_list
