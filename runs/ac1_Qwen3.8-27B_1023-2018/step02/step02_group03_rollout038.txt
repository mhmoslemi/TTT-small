- B = convolve(a, g) - computed once per LP
- C = convolve(g, g) - computed once per LP

def line_search(a, g, A, B, C, n, local_score):
    """Find best t minimizing score of (1-t)*a + t*g.
    A = a*a, B = a*g, C = g*g (all precomputed)"""
    # Coefficients: Q_k(t) = (1-t)^2 A_k + 2t(1-t) B_k + t^2 C_k
    #             = A_k - 2A_k*t + A_k*t^2 + 2B_k*t - 2B_k*t^2 + C_k*t^2
    #             = A_k + (2B_k - 2A_k)*t + (A_k - 2B_k + C_k)*t^2
    
    a_coeffs = A  # constant term
    b_coeffs = 2*(B - A)  # linear term
    c_coeffs = (A - 2*B + C)  # quadratic term
    
    best_t = 0.0
    best_val = local_score
    
    # Coarse search
    for t in np.arange(0.05, 1.01, 0.05):
        vals = a_coeffs + b_coeffs * t + c_coeffs * t * t
        v = 2.0 * n * float(vals.max())
        if v < best_val:
            best_val = v
            best_t = t
    
    # Golden section refinement
    if 0 < best_t < 1:
        gr = (1 + math.sqrt(5)) / 2
        lo = max(0, best_t - 0.08)
        hi = min(1, best_t + 0.08)
        for _ in range(30):
            if hi - lo < 1e-10:
                break
            m1 = hi - (hi - lo) / gr
            m2 = lo + (hi - lo) / gr
            v1 = 2.0 * n * float((a_coeffs + b_coeffs*m1 + c_coeffs*m1*m1).max())
            v2 = 2.0 * n * float((a_coeffs + b_coeffs*m2 + c_coeffs*m2*m2).max())
            if v1 < v2:
                hi = m2
            else:
                lo = m1
            # Update best
            if v1 < best_val:
                best_val = v1
                best_t = m1
            if v2 < best_val:
                best_val = v2
                best_t = m2
    
    # Final check at t=1
    v1 = 2.0 * n * float(C.max())
    if v1 < best_val:
        best_val = v1
        best_t = 1.0
    
    return best_t, best_val

        update_best(a)

        stale = 0
        alpha = 0.1

        for it in range(max_iters):
            if time.monotonic() >= stop_time - 3:
                break

            b = np.convolve(a, a)
            M = float(b.max())
            if M <= 0:
                break

            # Active set: lags within 2% of max
            thr = M * 0.98
            active = np.where(b >= thr)[0]

            # Subgradient: sum over active lags of 2*a[k-j]
            grad = np.zeros(n, dtype=np.float64)
            for k in active:
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                if lo <= hi:
                    js = np.arange(lo, hi + 1)
                    grad[js] += 2.0 * a[k - js]

            # Normalize to remove mean (stays on simplex)
            grad -= grad.mean()
            gn = float(np.linalg.norm(grad))
            if gn < 1e-14:
                break
            grad /= gn

            # Backtracking line search
            improved = False
            for _ in range(30):
                cand = a - alpha * grad
                cand = np.clip(cand, 0, None)
                cs = float(cand.sum())
                if cs < 1e-15:
                    alpha *= 0.5
                    continue
                cand /= cs
                v = score(cand)
                if v < local_score * (1 - 1e-12):
                    a = cand
                    local_score = v
                    improved = True
                    stale = 0
                    update_best(a)
                    alpha = min(alpha * 1.1, 0.5)
                    break
                else:
                    alpha *= 0.7

            if not improved:
                stale += 1
                if stale % 50 == 0 and stale < 500:
                    a_p = a + 0.02 * rng.standard_normal(n)
                    a_p = np.clip(a_p, 0, None)
                    if a_p.sum() > 1e-15:
                        a_p /= a_p.sum()
                        if score(a_p) < local_score:
                            a = a_p
                            local_score = score(a_p)
                            stale = 0
                            update_best(a)
                if stale > 500:
                    break

        return a

    # === MAIN OPTIMIZATION ===
    n = parent.size
    a = parent / parent.sum()

    # Phase 1: Intensive LP at parent length (60% of budget)
    t1 = time.monotonic()
    phase1_time = min(600.0, (deadline - t1) * 0.6)
    stop1 = t1 + phase1_time

    if linprog is not None and sps is not None:
        a = lp_optimize(a, n, stop1, max_iters=10000)
    update_best(a)

    # Phase 2: Subgradient polish (10% of budget)
    if time.monotonic() < deadline - 100:
        t2 = time.monotonic()
        phase2_time = min(120.0, (deadline - t2) * 0.1)
        stop2 = t2 + phase2_time
        a = subgradient_optimize(a, n, stop2, max_iters=50000)
        update_best(a)

    # Phase 3: Second LP pass after gradient (15% of budget)
    if time.monotonic() < deadline - 80 and linprog is not None:
        t3 = time.monotonic()
        phase3_time = min(150.0, (deadline - t3) * 0.15)
        stop3 = t3 + phase3_time
        a = lp_optimize(a, n, stop3, max_iters=5000)
        update_best(a)

    # Phase 4: Try nearby lengths (10% of budget)
    if time.monotonic() < deadline - 60:
        lengths = [n+2, n-2, n+4, n-4, n+8, n-8, 400, 448, 512, 384]
        lengths = [L for L in lengths if 16 <= L <= 1024 and L != n]
        lengths = sorted(set(lengths))

        for L in lengths:
            if time.monotonic() >= deadline - 40:
                break
            t_rem = deadline - time.monotonic() - 40
            if t_rem < 20:
                break
            alloc = min(40.0, t_rem * 0.25)
            if alloc < 10:
                break

            # Interpolate
            ox = np.linspace(0.0, 1.0, n)
            nx = np.linspace(0.0, 1.0, L)
            a_init = np.interp(nx, ox, a)
            a_init = np.clip(a_init, 0, None)
            si = float(a_init.sum())
            if si > 1e-15:
                a_init /= si
            else:
                a_init = np.ones(L) / L

            if linprog is not None and sps is not None and alloc > 15:
                a_w = lp_optimize(a_init, L, time.monotonic() + alloc * 0.6, max_iters=2000)
                update_best(a_w)
            if time.monotonic() < deadline - 30:
                a_w = subgradient_optimize(a_w, L, deadline - 25, max_iters=10000)
                update_best(a_w)

    # Phase 5: Restart with perturbations at parent length (5% of budget)
    if time.monotonic() < deadline - 30 and linprog is not None:
        n_restart = 8
        for r in range(n_restart):
            if time.monotonic() >= deadline - 20:
                break
            t_rem = deadline - time.monotonic() - 20
            if t_rem < 15:
                break
            alloc = min(25.0, t_rem / max(1, n_restart - r))
            if alloc < 8:
                break

            if r % 2 == 0:
                a_init = a * np.exp(0.1 * rng.standard_normal(n))
            else:
                a_init = a * np.exp(0.2 * rng.standard_normal(n))
            a_init = np.clip(a_init, 0, None)
            si = float(a_init.sum())
            if si > 1e-15:
                a_init /= si

            a_w = lp_optimize(a_init, n, time.monotonic() + alloc * 0.7, max_iters=1000)
            update_best(a_w)
            if time.monotonic() < deadline - 15:
                a_w = subgradient_optimize(a_w, n, deadline - 12, max_iters=5000)
                update_best(a_w)

    # Phase 6: Final polish
    if time.monotonic() < deadline - 10 and best_arr is not None:
        n_b = best_arr.size
        a_pol = best_arr / best_arr.sum()
        a_pol = subgradient_optimize(a_pol, n_b, deadline - 5, max_iters=20000)
        update_best(a_pol)

        if linprog is not None and time.monotonic() < deadline - 8:
            a_pol = lp_optimize(a_pol, n_b, deadline - 3, max_iters=500)
            update_best(a_pol)

    # === Final validation ===
    if best_arr is None:
        best_arr = np.ones(96, dtype=np.float64)
        update_best(best_arr)

    final = np.clip(best_arr, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones_like(final)
    update_best(final)

    return [float(x) for x in best_arr]
