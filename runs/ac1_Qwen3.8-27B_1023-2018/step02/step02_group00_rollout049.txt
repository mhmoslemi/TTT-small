def _conv_full(a):
    n = len(a)
    if n <= 256:
        return np.maximum(np.convolve(a, a, mode="full"), 0.0)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score_fast(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)

def _build_A(f, n):
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j+n, j] = f
    return A

def _lp_direction(f, n, M=None):
    if M is None:
        b = _conv_full(f)
        M = float(b.max())
    if M < 1e-15:
        return None
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True, 'mip_rel_gap': 1e-8})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g_sum = float(g.sum())
    if g_sum < 1e-15:
        return None
    return g / g_sum

        p = np.ones(320, dtype=np.float64)
    return p

def _conv_full(a):
    n = len(a)
    if n <= 256:
        return np.maximum(np.convolve(a, a, mode="full"), 0.0)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score_fast(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)
