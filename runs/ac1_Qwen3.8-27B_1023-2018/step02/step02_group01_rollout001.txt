def _derivative_for_lag(a, lag):
    n = a.size
    v = np.zeros(n, dtype=np.float64)
    lag = int(lag)

    if lag >= 0:
        if 0 <= lag < n:
            v[: n - lag] += a[lag:]
            v[lag:] += a[: n - lag]
    else:
        l = -lag
        if 0 <= l < n:
            v[l:] += a[: n - l]
            v[: n - l] += a[l:]

    return v


def _smooth_kernel(width, sigma):
    width = int(max(1, width))

        noise = rng.normal(0.0, 1.0, n)
        kern = np.convolve(noise, np.ones(3, dtype=np.float64) / 3.0, mode="same")
        sigma = float(rng.choice([0.05, 0.1, 0.2, 0.3]))
        cand = a * np.exp(sigma * kern)

    return _normalize_nonnegative(cand)


def _population_search(current_best, current_score, scored_inits, offpeak, stop_time, seed):
    rng = np.random.default_rng(seed)
    current_best = _normalize_nonnegative(current_best)

    items = [(float(current_score), current_best.copy())]
    for sc, arr in scored_inits:
        if math.isfinite(sc):
            items.append((float(sc), _normalize_nonnegative(arr)))

    items = [(sc, arr) for sc, arr in items if math.isfinite(sc)]
    if not items:
        return current_best, float(current_score) if math.isfinite(current_score) else float("inf")

    items.sort(key=lambda z: z[0])
    pop = items[:8]
    pop_sc = [float(z[0]) for z in pop]
    pop_arr = [z[1].copy() for z in pop]

    best_arr = pop_arr[0].copy()
    best_sc = pop_sc[0]

    cache_uniform = {}
    max_evals = 30000
    evals = 0

    while time.monotonic() < stop_time and evals < max_evals:
        if float(rng.random()) < 0.75:
            parent = best_arr
        else:
            idx = int(rng.integers(len(pop_arr)))
            parent = pop_arr[idx]

        n = parent.size
        uniform = cache_uniform.get(n)
        if uniform is None:
            uniform = np.ones(n, dtype=np.float64) / float(n)
            cache_uniform[n] = uniform

        cand = _mutate(parent, rng, uniform)
        sc = _score_array(cand)
        evals += 1
        if not math.isfinite(sc):
            continue

        worst = max(range(len(pop_sc)), key=lambda i: pop_sc[i])

        if sc < best_sc:
            best_arr = cand.copy()
            best_sc = sc
            pop_sc[worst] = sc
            pop_arr[worst] = cand.copy()
        elif sc < pop_sc[worst]:
            pop_sc[worst] = sc
            pop_arr[worst] = cand.copy()

        if evals % 700 == 0:
            sigma = float(rng.choice([0.1, 0.2, 0.35]))
            noise = rng.normal(0.0, 1.0, best_arr.size)
            kern = np.ones(3, dtype=np.float64) / 3.0
            noise = np.convolve(noise, kern, mode="same")
            cand2 = best_arr * np.exp(sigma * noise)
            sc2 = _score_array(cand2)
            evals += 1
            if math.isfinite(sc2):
                worst2 = max(range(len(pop_sc)), key=lambda i: pop_sc[i])
                if sc2 < best_sc:
                    best_arr = cand2.copy()
                    best_sc = sc2
                    pop_sc[worst2] = sc2
                    pop_arr[worst2] = cand2.copy()
                elif sc2 < pop_sc[worst2]:
                    pop_sc[worst2] = sc2
                    pop_arr[worst2] = cand2.copy()

    return best_arr, best_sc


def _gradient_polish(a, offpeak, stop_time):
    best = _normalize_nonnegative(a)
    best_sc = _score_array(best)
    current = best.copy()
    cur_sc = best_sc
    n = current.size
    rng = np.random.default_rng(98765)
    stuck = 0

    for _ in range(10000):
        if time.monotonic() > stop_time:
            break

        lags, _ = _active_lags(current, offpeak, k=3 if offpeak else 1)
        v = np.zeros(n, dtype=np.float64)
        for lag in lags:
            v += _derivative_for_lag(current, lag)
        if len(lags) > 1:
            v /= float(len(lags))

        w = v - float(v.mean())
        w_std = float(np.std(w))
        improved = False

        if w_std > 1e-10:
            z = np.clip(w / w_std, -4.0, 4.0)
            for t in (0.05, 0.1, 0.2, 0.4, 0.8, 1.5, 3.0):
                cand = current * np.exp(-t * z)
                cand = np.maximum(cand, 0.0)
                s = float(cand.sum())
                if s <= 1e-12:
                    continue
                cand /= s
                sc = _score_array(cand)
                if math.isfinite(sc) and sc < cur_sc - 1e-12:
                    current = cand
                    cur_sc = sc
                    improved = True
                    break

        if not improved:
            w_norm = float(np.linalg.norm(w))
            if w_norm > 1e-12:
                scale = max(1.0 / float(n), float(np.std(current)))
                w_unit = w / w_norm
                for t in (0.05, 0.1, 0.3, 1.0, 3.0, 10.0):
                    cand = current - t * scale * w_unit
                    cand = np.maximum(cand, 0.0)
                    s = float(cand.sum())
                    if s <= 1e-12:
                        continue
                    cand /= s
                    sc = _score_array(cand)
                    if math.isfinite(sc) and sc < cur_sc - 1e-12:
                        current = cand
                        cur_sc = sc
                        improved = True
                        break

        if improved:
            stuck = 0
            if cur_sc < best_sc - 1e-12:
                best = current.copy()
                best_sc = cur_sc
        else:
            stuck += 1
            if stuck in (8, 16, 32) or stuck >= 64:
                sigma = 0.02 if stuck < 16 else (0.05 if stuck < 32 else 0.1)
                noise = rng.normal(0.0, 1.0, n)
                kern = np.ones(3, dtype=np.float64) / 3.0
                noise = np.convolve(noise, kern, mode="same")
                cand = current * np.exp(sigma * noise)
                cand = np.maximum(cand, 0.0)
                s = float(cand.sum())
                if s > 1e-12:
                    cand /= s
                    sc = _score_array(cand)
                    if math.isfinite(sc):
                        current = cand
                        cur_sc = sc
                        stuck = 0
                        if cur_sc < best_sc - 1e-12:
                            best = current.copy()
                            best_sc = cur_sc
                else:
                    stuck = 0

    return best, best_sc


def _coordinate_polish(a, offpeak, stop_time, seed):
    best = _normalize_nonnegative(a)
    best_sc = _score_array(best)
    n = best.size
    uniform = np.ones(n, dtype=np.float64) / float(n)
    rng = np.random.default_rng(seed + 123456)

    evals = 0
    max_evals = 60000
    no_improve = 0

    while time.monotonic() < stop_time and evals < max_evals:
        evals += 1
        r = float(rng.random())

        if r < 0.25:
            t = float(rng.choice([0.01, 0.02, 0.05, 0.1, 0.2, 0.4, 0.8, 1.0]))
            cand = (1.0 - t) * best + t * uniform
        elif r < 0.45:
            w = int(rng.choice([1, 1, 2, 3, 4, 6, 8, 12, 16, 24]))
            w = min(w, n)
            st = int(rng.integers(0, max(1, n - w + 1)))
            sigma = float(rng.choice([0.05, 0.1, 0.2, 0.3, 0.5]))
            factor = math.exp(float(rng.normal(0.0, sigma)))
            cand = best.copy()
            cand[st:st + w] *= factor
        elif r < 0.65:
            i = int(rng.integers(0, n))
            sigma = float(rng.choice([0.05, 0.1, 0.2, 0.35]))
            factor = math.exp(float(rng.normal(0.0, sigma)))
            cand = best.copy()
            cand[i] *= factor
        elif r < 0.80:
            cand = best.copy()
            if n >= 2:
                i, j = rng.choice(n, size=2, replace=False)
                delta = float(rng.normal(0.0, 0.02))
                cand[int(i)] += delta
                cand[int(j)] -= delta
        elif r < 0.90:
            k = int(rng.choice([2, 3, 5, 7, 9]))
            kern = np.ones(k, dtype=np.float64) / float(k)
            cand = np.convolve(best, kern, mode="same")
        else:
            shift = int(rng.integers(-max(1, n // 10), max(1, n // 10) + 1))
            cand = np.roll(best, shift)

        cand = _normalize_nonnegative(cand)
        sc = _score_array(cand)
        if math.isfinite(sc) and sc < best_sc - 1e-12:
            best = cand
            best_sc = sc
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 4000:
                noise = rng.normal(0.0, 1.0, n)
                kern = np.ones(3, dtype=np.float64) / 3.0
                noise = np.convolve(noise, kern, mode="same")
                cand2 = best * np.exp(0.25 * noise)
                sc2 = _score_array(cand2)
                if math.isfinite(sc2) and sc2 < best_sc:
                    best = cand2
                    best_sc = sc2
                no_improve = 0

    return best, best_sc


def _build_convolve_matrix(a, exclude):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    m = 2 * n - 1
    ks = np.arange(m, dtype=np.int64)
    if exclude is not None and 0 <= int(exclude) < m:
        ks = np.delete(ks, int(exclude))

    k_idx = ks[:, None]
    i_idx = np.arange(n, dtype=np.int64)[None, :]
    j = k_idx - i_idx
    mask = (j >= 0) & (j < n)
    safe_j = np.clip(j, 0, n - 1)
    A = np.where(mask, a[safe_j], 0.0).astype(np.float64)
    return A


def _lp_polish(a, offpeak, stop_time):
    best = _normalize_nonnegative(a)
    best_sc = _score_array(best)

    if linprog is None or best.size > 350:
        return best, best_sc

    n = best.size

    for _ in range(60):
        if time.monotonic() > stop_time:
            break

        conv = _fast_conv(best)
        m = 2 * n - 1
        if offpeak:
            vals = conv.copy()
            if 0 <= n - 1 < vals.size:
                vals[n - 1] = -np.inf
            M = float(np.max(vals)) if vals.size else 0.0
            exclude = n - 1
        else:
            M = float(np.max(conv)) if conv.size else 0.0
            exclude = None

        if not math.isfinite(M) or M <= 1e-300:
            break

        A = _build_convolve_matrix(best, exclude)
        if A.size == 0:
            break

        c = -np.ones(n, dtype=np.float64)
        b = np.full(A.shape[0], M * (1.0 + 1e-8), dtype=np.float64)

        try:
            res = linprog(
                c,
                A_ub=A,
                b_ub=b,
                bounds=[(0.0, None)] * n,
                method="highs",
                options={
                    "time_limit": 0.45,
                    "primal_feasibility_tolerance": 1e-10,
                    "dual_feasibility_tolerance": 1e-10,
                },
            )
        except Exception:
            break

        x = getattr(res, "x", None)
        if x is None:
            break

        g = np.asarray(x, dtype=np.float64).ravel()
        g = np.where(np.isfinite(g), g, 0.0)
        g = np.clip(g, 0.0, None)
        gs = float(g.sum())
        if gs <= 1e-15:
            break
        g /= gs

        improved = False
        for t in (1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01):
            cand = (1.0 - t) * best + t * g
            cand = _normalize_nonnegative(cand)
            sc = _score_array(cand)
            if math.isfinite(sc) and sc < best_sc - 1e-12:
                best = cand
                best_sc = sc
                improved = True
                break

        if not improved:
            break

    return best, best_sc


def _resample_nonnegative(a, m):
    a = _normalize_nonnegative(a)
    m = int(m)
    if m <= 0:
        return None
    if m == a.size:
        return a.copy()

    old_x = np.linspace(0.0, 1.0, a.size)
    new_x = np.linspace(0.0, 1.0, m)
    b = np.interp(new_x, old_x, a)
    return _normalize_nonnegative(b)


def _length_sweep(a, offpeak, stop_time):
    best = _normalize_nonnegative(a)
    best_sc = _score_array(best)
    n0 = best.size

    candidates = set()
    for m in [
        n0,
        max(8, n0 - 1),
        n0 + 1,
        max(8, n0 // 2),
        min(1024, n0 * 2),
        64,
        96,
        128,
        160,
        191,
        192,
        224,
        256,
        320,
        384,
        512,
        640,
        768,
        1024,
    ]:
        if 8 <= int(m) <= 1024:
            candidates.add(int(m))

    ordered = sorted(candidates, key=lambda z: abs(z - n0))
    for m in ordered:
        if time.monotonic() > stop_time:
            break

        cand = _resample_nonnegative(best, m)
        if cand is not None:
            sc = _score_array(cand)
            if math.isfinite(sc) and sc < best_sc - 1e-12:
                best = cand
                best_sc = sc

        uni = np.ones(m, dtype=np.float64) / float(m)
        sc_uni = _score_array(uni)
        if math.isfinite(sc_uni) and sc_uni < best_sc - 1e-12:
            best = uni
            best_sc = sc_uni

    return best, best_sc


def _finalize_sequence(a):
    arr = _normalize_nonnegative(a)
    if arr.size == 0:
        arr = np.ones(16, dtype=np.float64) / 16.0

    sc = _score_array(arr)
    if not math.isfinite(sc):
        arr = np.ones(16, dtype=np.float64) / 16.0
        sc = _score_array(arr)

    if not math.isfinite(sc):
        return [1.0]

    arr = np.clip(arr, 0.0, 1000.0)
    return [float(x) for x in arr]


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1300.0

    if not math.isfinite(budget) or budget <= 0:
        budget = 1300.0
    deadline = t0 + max(0.0, budget - 10.0)

    parent_raw = _parent_raw()
    parent_nn = _normalize_nonnegative(parent_raw)
    n0 = parent_nn.size
    uniform_nn = np.ones(n0, dtype=np.float64) / float(n0)

    offpeak, lower_factor = _calibrate_evaluator()

    best = parent_nn.copy()
    best_sc = _score_array(best)
    uni_sc = _score_array(uniform_nn)
    if math.isfinite(uni_sc) and uni_sc < best_sc:
        best = uniform_nn.copy()
        best_sc = uni_sc

    if lower_factor is not None and best_sc <= lower_factor + 1e-7:
        return _finalize_sequence(best)

    inits = _generate_inits(parent_raw, integer.get(seed, seed)) if hasattr(seed, "get") else _generate_inits(parent_raw, seed)

    scored = []
    for arr in inits:
        arr = _normalize_nonnegative(arr)
        sc = _score_array(arr)
        scored.append((sc, arr))
        if math.isfinite(sc) and sc < best_sc:
            best = arr.copy()
            best_sc = sc

    if lower_factor is not None and best_sc <= lower_factor + 1e-7:
        return _finalize_sequence(best)

    stop1 = min(max(t0, deadline - 120.0), t0 + max(30.0, 0.15 * budget))
    if time.monotonic() < stop1:
        best, best_sc = _population_search(best, best_sc, scored, offpeak, stop1, int(seed) if np.isscalar(seed) else 42)

    if lower_factor is not None and best_sc <= lower_factor + 1e-7:
        return _finalize_sequence(best)

    stop2 = min(max(stop1, deadline - 60.0), stop1 + max(60.0, 0.30 * budget))
    if time.monotonic() < stop2:
        best, best_sc = _gradient_polish(best, offpeak, stop2)

    if lower_factor is not None and best_sc <= lower_factor + 1e-7:
        return _finalize_sequence(best)

    stop3 = min(max(stop2, deadline - 30.0), stop2 + max(60.0, 0.30 * budget))
    if time.monotonic() < stop3:
        best, best_sc = _coordinate_polish(best, offpeak, stop3, int(seed) if np.isscalar(seed) else 42)

    if lower_factor is not None and best_sc <= lower_factor + 1e-7:
        return _finalize_sequence(best)

    if time.monotonic() < deadline - 25.0 and best.size <= 350:
        stop_lp = min(max(stop3, deadline - 20.0), stop3 + max(30.0, 0.10 * budget))
        if time.monotonic() < stop_lp:
            best, best_sc = _lp_polish(best, offpeak, stop_lp)

    if time.monotonic() < deadline - 12.0:
        stop_len = max(t0 + 5.0, deadline - 8.0)
        best, best_sc = _length_sweep(best, offpeak, stop_len)

    uni_final = np.ones(best.size, dtype=np.float64) / float(best.size)
    sc_uni_final = _score_array(uni_final)
    if math.isfinite(sc_uni_final) and sc_uni_final < best_sc:
        best = uni_final.copy()
        best_sc = sc_uni_final

    sc_parent_final = _score_array(parent_nn)
    if math.isfinite(sc_parent_final) and sc_parent_final < best_sc:
        best = parent_nn.copy()
        best_sc = sc_parent_final

    return _finalize_sequence(best)
