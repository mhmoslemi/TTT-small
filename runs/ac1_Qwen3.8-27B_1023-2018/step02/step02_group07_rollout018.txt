        conv = irfft(rfft(a, fs_c) * rfft(p[::-1], fs_c), fs_c)
        g = conv[n-1 : 2*n-1]
        return 2.0 * g


def _score_and_grad(z, n, fs_c, tau):
    """Compute smoke-sensitive score and gradient w.r.t. z.
    Returns (score, grad_z) where grad_z = ∂score/∂z.
    """
    a = _softmax(z)
    b = _conv(a, fs_c)
    S = float(a.sum())  # should be 1
    if S < 1e-15:
        return math.inf, np.zeros(n)
    
    bmax = float(b.max())
    if not np.isfinite(bmax):
        return math.inf, np.zeros(n)
    score = 2.0 * n * bmax / (S * S)
    
    # Softmax weights over b
    bt = np.clip(b / tau, -80.0, 80.0)
    e = np.exp(bt)
    se = float(e.sum())
    if se <= 0 or not np.isfinite(se):
        p = np.zeros_like(b)
        p[int(np.argmax(b))] = 1.0
    else:
        p = e / se
    
    # Gradient of b w.r.t. a: 2 * cross_correlate(p, a)
    # grad_b[k] = 2 * sum_i p[i] * a[i-k]
    g = _cross_grad(p, a, fs_c)
    
    # ∂score/∂a_k = (2n/S²) * g[k] - (4n*bmax/S³) * 1
    # = (2n/S²) * (g[k] - bmax/S)
    # But in softmax parameterization:
    # ∂score/∂z_k = a_k * (∂score/∂a_k - sum_j a_j * ∂score/∂a_j)
    
    grad_a = (2.0 * n / (S * S)) * (g - (bmax / S))
    # Wait, let me redo this.
    # score = 2n * max_b / S²
    # ∂score/∂a_k = 2n * ∂max_b/∂a_k / S² - 2n * max_b * 2*S / S⁴
    # = 2n/S² * ∂max_b/∂a_k - 4n*max_b/S³
    # = (2n/S²) * (∂max_b/∂a_k - 2*max_b/S)
    
    # Hmm, that doesn't look right dimensionally. Let me redo.
    # score = 2n * max_b / S²
    # ∂(1/S²)/∂a_k = -2/S³
    # ∂score/∂a_k = 2n * ∂max_b/∂a_k / S² + 2n * max_b * (-2/S³)
    # = 2n/S² * ∂max_b/∂a_k - 4n*max_b/S³
    # = (2n/S²) * (∂max_b/∂a_k - 2*max_b/S)
    
    # In the softmax parameterization (a = softmax(z)), sum(a) = 1 always.
    # So S = 1, and score = 2n * max_b.
    # ∂score/∂a_k = 2n * ∂max_b/∂a_k (since S=1 is fixed by parameterization)
    # But ∂max_b/∂a_k in the softmax space:
    # ∂max_b/∂z_k = sum_j (∂max_b/∂a_j)(∂a_j/∂z_k)
    # ∂a_j/∂z_k = a_j(δ_{jk} - a_k)
    # ∂max_b/∂z_k = a_k * ∂max_b/∂a_k - a_k * sum_j a_j * ∂max_b/∂a_j
    
    if S == 1.0 or abs(S - 1.0) < 1e-10:
        # Since we use softmax, S should be 1
        grad_b_k = g  # = 2 * sum_i p[i]*a[i-k], the gradient of smoothed max_b w.r.t. a
        mean_grad = float(np.dot(a, grad_b_k))
        grad_z = a * (grad_b_k - mean_grad) * 2.0 * n
    else:
        grad_a_full = (2.0 * n / (S * S)) * (g - 2.0 * bmax / S)
        mean_grad = float(np.dot(a, grad_a_full))
        grad_z = a * (grad_a_full - mean_grad)
    
    return score, grad_z


def _run_adam(z_init, n, tau_init, lr_init, end_time, hard_deadline, seed=0):
    """Run Adam optimization on log-amplitude z."""
    fs_c = next_fast_len(2 * n - 1)
    z = z_init.copy()
    best_z = z.copy()
    best_score = math.inf
    
    # Evaluate initial
    s0, _ = _score_and_grad(z, n, fs_c, tau_init)
    best_score = s0
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = lr_init
    tau = tau_init
    stuck = 0
    tau_min = 0.02
    tau_max = 0.8
    tau_decay = 0.995
    lr_decay = 0.97
    lr_min = 0.05
    
    rng = np.random.default_rng(seed)
    
    while time.monotonic() < min(end_time, hard_deadline):
        t += 1
        # Compute gradient
        score, grad_z = _score_and_grad(z, n, fs_c, tau)
        
        if not np.isfinite(score):
            score = best_score
            grad_z = np.zeros(n)
        
        if score < best_score - 1e-14:
            best_score = score
            best_z = z.copy()
            stuck = 0
        else:
            stuck += 1
        
        # Normalize gradient
        gn = float(np.abs(grad_z).max())
        if not np.isfinite(gn) or gn < 1e-18:
            if stuck > 30:
                z = best_z + rng.normal(0, 0.5, n)
                z -= z.mean()
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
                lr = lr_init
                tau = max(tau_min, tau * 0.5)
                stuck = 0
            continue
        
        grad_z = grad_z / gn
        
        # Adam update
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * (grad_z * grad_z)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        update = m_hat / (np.sqrt(v_hat) + eps)
        
        z = z - lr * update
        z -= z.mean()
        z = np.clip(z, -25.0, 25.0)
        z[~np.isfinite(z)] = 0.0
        
        # Adaptive parameters
        if t % 50 == 0:
            tau = max(tau_min, tau * tau_decay)
        if stuck > 40:
            lr = max(lr_min, lr * lr_decay)
        if stuck > 200:
            z = best_z + rng.normal(0, 0.7, n)
            z -= z.mean()
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            lr = lr_init
            tau = max(tau_min, tau * 0.7)
            stuck = 0
    
    # Final evaluation
    a = _softmax(best_z)
    s = _score_and_grad(best_z, n, fs_c, tau)[0]
    return a, s, best_z


def _polish_local(a, end_time, hard_deadline, seed=12345):
    """Local search polish."""
    n = a.size
    fs_c = next_fast_len(2 * n - 1)
    best = a.copy()
    bs = 2.0 * n * float(_conv(best, fs_c).max())
    rng = np.random.default_rng(seed)
    sigma = 0.15
    stuck = 0
    z_best = np.log(np.clip(best, 1e-15, 1000.0))
    z_best -= z_best.mean()
    
    while time.monotonic() < min(end_time, hard_deadline) - 0.3:
        r = rng.random()
        if r < 0.3:
            z = z_best + rng.normal(0, sigma, n)
        elif r < 0.5:
            i = int(rng.integers(0, n))
            w = max(1, int(rng.integers(1, max(2, n // 20 + 1))))
            i2 = min(n, i + w)
            z = z_best.copy()
            z[i:i2] += rng.normal(0, sigma * 0.5, i2 - i)
        elif r < 0.7:
            z = z_best * rng.uniform(0.5, 1.8)
        else:
            z = z_best + rng.uniform(-sigma, sigma) * np.sin(2*np

.pi*np.arange(n)/n + rng.uniform(0, 2*np.pi))
        
        z -= z.mean()
        a2 = _softmax(z)
        b2 = _conv(a2, fs_c)
        s2 = 2.0 * n * float(b2.max())
        
        if np.isfinite(s2) and s2 < bs - 1e-12:
            best = a2.copy()
            bs = s2
            z_best = np.log(np.clip(best, 1e-15, 1000.0))
            z_best -= z_best.mean()
            stuck = 0
            sigma = max(0.03, sigma * 0.97)
        else:
            stuck += 1
            sigma = min(1.0, sigma * 1.03)
            if stuck > 80:
                stuck = 0
                sigma = max(0.1, sigma * 0.5)
    
    a = np.clip(best, 1e-15, 1000.0)
    s = 2.0 * n * float(_conv(a, fs_c).max())
    return a, s
