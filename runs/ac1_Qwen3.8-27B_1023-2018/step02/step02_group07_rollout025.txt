    F = rfft(a, fs)

    return e / s

def _fftc(a, fs):
    F = rfft(a, fs)
    out = irfft(F * F, fs)
    return out[:2*len(a)-1]

def _fftcx(p, a, fs):
    n = len(a)
    pr = p[::-1]
    out = irfft(rfft(pr, fs) * rfft(a, fs), fs)
    return out[n-1:2*n-1]

def _score(a):
    b = _fftc(a, next_fast_len(2*len(a)-1))
    m = float(b.max())
    if m < 0: m = 0.0
    return 2.0 * len(a) * m
