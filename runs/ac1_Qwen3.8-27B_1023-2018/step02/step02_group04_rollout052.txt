def _build_A(n):
    """Build the convolution constraint matrix A (2n-1 x n).
    Row k: A[k, i] = 1 if 0 <= k-i < n, so (A @ g)[k] = sum_i g[i]*f[k-i] = (f*g)[k]
    Wait, actually we need: (f*g)[k] = sum_i f[k-i] * g[i] for valid i.
    So A[k, i] = f[k-i] if 0<=k-i<n else 0.
    But f changes each iteration, so we build A each time.
    """
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        # A[k, i] where i ranges over valid indices of g: 0<=i<n, and f[k-i] valid: 0<=k-i<n
        # So i ranges from max(0, k-n+1) to min(k, n-1)
        for i in range(lo, hi + 1):
            A[k, i] = 1  # placeholder, actual f values set externally
    return A


def _solve_lp(f, M, time_limit=5.0):
    """Solve: max sum(g) s.t. g>=0, (f*g)[k] <= M for all k.
    (f*g)[k] = sum_i f[k-i] * g[i], so row k of A has entries f[k-i].
    """
    if linprog is None:
        return None
    n = f.size
    m = 2 * n - 1
    # Build A: A[k, i] = f[k-i] if 0 <= k-i < n, else 0
    A = np.zeros((m, n), dtype=np.float64)
    for k in range(m):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for i in range(lo, hi + 1):
            A[k, i] = f[k - i]
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(m, M * 1.0000001 + 1e-12, dtype=np.float64)
    bounds = [(0.0, None)] * n
    for attempt in range(2):
        try:
            opts = {'presolve': True, 'time_limit': time_limit}
            if attempt == 1:
                opts['dual_feasibility_tolerance'] = 1e-9
                opts['primal_feasibility_tolerance'] = 1e-9
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options=opts)
            if r is not None and r.x is not None:
                g = np.clip(np.nan_to_num(r.x, nan=0.0, posinf=0.0, neginf=0.0), 0.0, None)
                sg = float(g.sum())
                if sg > 1e-12:
                    return g / sg
        except Exception:
            pass
    return None


def _golden_line_search(a, g, loc, best, n_alphas=12):
    """Golden section search for blend parameter t in (0,1)."""
    if loc is None or not math.isfinite(loc):
        loc = _score(a)
    lo, hi = 0.001, 0.999
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _score(_norm((1.0 - c1) * a + c1 * g))
    f2 = _score(_norm((1.0 - c2) * a + c2 * g))
    for _ in range(n_alphas):
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = _score(_norm((1.0 - c1) * a + c1 * g))
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = _score(_norm((1.0 - c2) * a + c2 * g))
    t_best = (lo + hi) / 2.0
    f_best = min(f1, f2)
    # Also check t=0 and a few extra points
    for t in (0.0, 0.05, 0.1, 0.3, 0.5, 1.0):
        fc = _score(_norm((1.0 - t) * a + t * g))
        if math.isfinite(fc) and fc < f_best:
            f_best = fc
            t_best = t
    return t_best, f_best


def _lp_refine(a0, deadline, best, rng, max_it=5000, verbose=False):

    a = _norm(a0.copy())

            continue

        if t_best > 0.0 and f_best < loc * (1.0 - 1e-10):

def _grad_step(a, deadline, best, rng, max_steps=200):
    """Subgradient-based refinement at the active convolution lag."""
    a = _norm(a.copy())
    n = a.size
    loc = _score(a)
    _update(a, best)
    stale = 0
    for step in range(max_steps):
        if time.monotonic() > deadline - 3.0:
            break
        b = np.convolve(a, a)
        S = float(a.sum())
        k_star = int(np.argmax(b))
        # Subgradient of conv[k_star] w.r.t. a[i]:
        # conv[k] = sum_j a[j]*a[k-j], d/da[i] = a[k-i] + a[i-k] (adjusted for boundaries)
        grad = np.zeros(n, dtype=np.float64)
        for i in range(n):
            j1 = k_star - i
            j2 = i - k_star
            if 0 <= j1 < n:
                grad[i] += a[j1]
            if 0 <= j2 < n:
                grad[i] += a[j2]
        # Normalize score gradient: d(score)/da[i] = 2n * grad[i] / S^2 - 4n*b_max*a[i]/S^3
        # For direction: minimize, so descend
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        d = -grad / gn
        # Line search
        bs, bt = loc, 0.0
        for alpha in (0.01, 0.02, 0.05, 0.1, 0.2, 0.4, 0.7, 1.0, 2.0, 5.0, 10.0):
            cand = _norm(a + alpha * d)
            sc = _score(cand)
            if math.isfinite(sc) and sc < bs:
                bs, bt = sc, alpha
        if bt > 0.0 and bs < loc * (1.0 - 1e-10):
            a = _norm(a + bt * d)
            loc = bs
            stale = 0
            _update(a, best)
        else:
            stale += 1
            if stale % 10 == 0:
                pa = _norm(a + 0.02 * rng.standard_normal(n))
                psa = _score(pa)
                if math.isfinite(psa) and psa < loc:
                    a, loc, stale = pa, psa, 0
                    _update(a, best)
            if stale > 50:
                break
    return a


def _mk_starts(n, rng, parent):
    """Generate diverse starting sequences of length n."""
    starts = []
    x = np.linspace(0.0, 1.0, n)
    if n == 2:
        x = np.array([0.0, 1.0])

    def add(a):
        a = np.asarray(a, dtype=np.float64).ravel()
        a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
        a = np.clip(a, 0.0, None)
        if a.size == n and float(a.sum()) > 1e-300:
            starts.append(_norm(a))

    # Uniform
    add(np.ones(n))
    # Exponentials
    for c in (0.5, 1.0, 2.0, 3.0, -0.5, -1.0, -2.0, -3.0, -5.0, 5.0):
        add(np.exp(c * (x - 0.5)))
    # Sine waves
    for k in range(1, min(6, n // 2 + 1)):
        for amp in (0.3, 0.6, 0.9):
            add(np.maximum(0.01, 1.0 + amp * np.sin(2.0 * np.pi * k * x)))
            add(np.maximum(0.01, 1.0 - amp * np.sin(2.0 * np.pi * k * x)))
    # Cosine bell
    add(0.5 * (1.0 + np.cos(np.pi * (2.0 * x - 1.0))))
    # Quasi-exponential
    add(np.exp(-4.0 * (x - 0.3) ** 2))
    add(np.exp(-4.0 * (x - 0.7) ** 2))
    # Random
    for _ in range(3):
        add(rng.dirichlet(np.full(n, 0.5) + 0.1))
    # Parent interpolated
    if parent is not None and len(parent) >= 2:
        pp = np.asarray(parent, dtype=np.float64).ravel()
        pp = np.nan_to_num(pp, nan=0.0, posinf=0.0, neginf=0.0)
        pp = np.clip(pp, 0.0, None)
        old_x = np.linspace(0.0, 1.0, len(pp))
        gi = np.interp(x, old_x, pp)
        add(gi)
        add(gi * np.exp(0.1 * rng.standard_normal(n)))
        add(gi * np.exp(0.3 * rng.standard_normal(n)))
        add(gi * (1.0 + 0.2 * np.sin(4.0 * np.pi * x)))

    if not starts:
        starts.append(np.ones(n, dtype=np.float64) / n)
    starts.sort(key=lambda s: _score(s))
    return starts

    a = np.asarray(a, dtype=np.float64).ravel()
    old_x = np.linspace(0.0, 1.0, len(a))
    new_x = np.linspace(0.0, 1.0, new_n)
    return np.interp(new_x, old_x, a)


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    try:
        rem = max(0.0, float(budget_s))
    except Exception:
        rem = 1300.0
    deadline = t0 + rem - 30.0  # 30s buffer

    best = [float('inf'), np.array([1.0, 1.0], dtype=np.float64)]

    # Get parent
    par = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            par = np.asarray(h, dtype=np.float64).ravel()
            par = np.nan_to_num(par, nan=0.0, posinf=0.0, neginf=0.0)
            par = np.clip(par, 0.0, 1000.0)
            if par.size >= 2 and float(par.sum()) > 0.01:
                _update(par, best)
    except Exception:
        par = None

    n_par = par.size if par is not None else 192

    # === Phase 1: LP refinement at parent length (main workhorse) ===
    # Allocate ~50% of time
    phase1_end = min(deadline, t0 + rem * 0.48)
    if phase1_end > t0 + 10:
        starts = _mk_starts(n_par, rng, par)
        # Try top diverse starts
        tried = 0
        for idx, s in enumerate(starts[:12]):
            if time.monotonic() > phase1_end - 10:
                break
            # Vary by seed offset for diversity
            srng = np.random.default_rng(seed + idx * 997)
            # Slightly perturb start
            s2 = _norm(s * np.exp(0.02 * srng.standard_normal(n_par)))
            alloc_end = min(phase1_end - 5, time.monotonic() + 80.0)
            a_lm, _ = _lp_refine(s2 if _score(s2) < _score(s) else s, alloc_end, best, srng, max_it=3000)
            tried += 1

    # === Phase 2: LP at different lengths ===
    phase2_end = min(deadline, t0 + rem * 0.70)
    if phase2_end > time.monotonic() + 15:
        for n_try in (128, 256, 160):
            if time.monotonic() > phase2_end - 15:
                break
            a_resized = _resize(best[1], n_try) if best[1].size > 1 else np.ones(n_try)
            starts2 = _mk_starts(n_try, rng, a_resized)
            for s in starts2[:4]:
                if time.monotonic() > phase2_end - 10:
                    break
                srng = np.random.default_rng(seed + 5000 + n_try)
                alloc_end = min(phase2_end - 5, time.monotonic() + 50.0)
                _lp_refine(s, alloc_end, best, srng, max_it=1500)

    # === Phase 3: Gradient polish at best length ===
    phase3_end = min(deadline, t0 + rem * 0.82)
    if phase3_end > time.monotonic() + 10:
        nb = best[1].size
        grng = np.random.default_rng(seed + 7777)
        _grad_step(best[1], phase3_end - 5, best, grng, max_steps=300)

    # === Phase 4: Final LP polish ===
    phase4_end = min(deadline, t0 + rem * 0.92)
    if phase4_end > time.monotonic() + 8:
        frng = np.random.default_rng(seed + 12345)
        _lp_refine(best[1], phase4_end - 3, best, frng, max_it=2000)

    # === Phase 5: One more gradient pass ===
    if time.monotonic() < deadline - 5:
        grng2 = np.random.default_rng(seed + 31415)
        _grad_step(best[1], deadline - 3, best, grng2, max_steps=100)

    # Final validation
    fin = np.asarray(best[1], dtype=np.float64).ravel().copy()
    fin = np.nan_to_num(fin, nan=0.0, posinf=0.0, neginf=0.0)
    fin = np.clip(fin, 0.0, 1000.0)
    if fin.size < 1:
        fin = np.array([1.0], dtype=np.float64)
    if float(fin.sum()) < 0.01:
        fin = np.ones(max(fin.size, 2), dtype=np.float64)
    _update(fin, best)
    fin = np.asarray(best[1], dtype=np.float64).ravel()
    v = evaluate_sequence([float(x) for x in fin])
    if not math.isfinite(v):
        fin = np.ones(4, dtype=np.float64)
        evaluate_sequence([float(x) for x in fin])
    return [float(x) for x in fin]
