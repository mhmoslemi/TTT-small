        kd = dst + r
        mask = kd < L
        if mask.any():
            garr[kd[mask]] += 2.0 * a_mat[r[mask]]

        ks = src + r
        mask = ks < L
        if mask.any():
            garr[ks[mask]] -= 2.0 * a_mat[r[mask]]

        if 2 * dst < L:
            darr[2 * dst] += 1.0
        if 2 * src < L:
            darr[2 * src] += 1.0
        if src + dst < L:
            darr[src + dst] -= 2.0

        nz = np.nonzero((garr != 0.0) | (darr != 0.0))[0]
        if nz.size == 0:
            return None

        base = b[nz].astype(np.float64, copy=False)
        gv = garr[nz]
        dv = darr[nz]
        cand = b.copy()
        best_t = 0.0
        bestM = M

        fracs = np.array(
            [1e-4, 3e-4, 1e-3, 3e-3, 1e-2, 3e-2, 1e-1,
             2e-1, 3e-1, 5e-1, 7e-1, 9e-1, 1.0],
            dtype=np.float64,
        )

        for fr in fracs:
            t = float(fr) * max_t
            if t <= 0.0:
                continue
            vals = base + gv * t + dv * t * t
            np.maximum(vals, 0.0, out=vals)
            cand[nz] = vals
            Mnew = float(cand.max())
            if Mnew < bestM:
                bestM = Mnew
                best_t = t

        if best_t <= 0.0 or bestM >= M - 1e-12:
            return None

        best_frac = best_t / max_t
        if best_frac <= 0.01:
            lo = 0.0
            hi = max(0.03, best_frac * 3.0)
        elif best_frac >= 0.9:
            lo = min(1.0, best_frac - 0.2)
            hi = 1.0
        else:
            lo = max(0.0, best_frac - 0.20 * best_frac)
            hi = min(1.0, best_frac + 0.20 * best_frac)

        for fr in np.linspace(lo, hi, 25, dtype=np.float64):
            t = float(fr) * max_t
            if t <= 0.0:
                continue
            vals = base + gv * t + dv * t * t
            np.maximum(vals, 0.0, out=vals)
            cand[nz] = vals
            Mnew = float(cand.max())
            if Mnew < bestM:
                bestM = Mnew
                best_t = t

        if best_t <= 0.0 or bestM >= M - 1e-12:
            return None

        out = a_mat.copy()
        out[src] -= best_t
        out[dst] += best_t
        return out, bestM

    def coordinate_attempt(a_mat, b, M):
        h = compute_h_active(a_mat, b, M, K=64)
        hmax = float(h.max()) if h.size else 0.0

        if hmax <= 1e-30:
            src_pool = np.array([int(rng.integers(0, n)) for _ in range(3)], dtype=np.intp)
            dst_pool = np.array([int(rng.integers(0, n)) for _ in range(3)], dtype=np.intp)
        else:
            k4 = min(n, 4)
            k3 = min(n, 3)
            sh = np.argpartition(-h, k4)[:k4] if k4 > 0 else np.empty(0, dtype=np.intp)
            dh = np.argpartition(h, k3)[:k3] if k3 > 0 else np.empty(0, dtype=np.intp)
            src_pool = np.unique(np.concatenate((sh, rng.integers(0, n, 2))))[:4]
            dst_pool = np.unique(np.concatenate((dh, rng.integers(0, n, 2))))[:3]

        best = None
        bestM = M
        for s in src_pool:
            s = int(s)
            if a_mat[s] <= 1e-14:
                continue
            for d in dst_pool:
                d = int(d)
                if d == s:
                    continue
                res = move_search(a_mat, b, M, s, d)
                if res is not None and res[1] < bestM:
                    best = res
                    bestM = res[1]

        return best

    def coord_polish(start, time_end, max_moves=5000):
        a_mat = _clean(start)
        if a_mat is None:
            return best_a
        b = conv(a_mat)
        M = float(b.max()) if b.size else 0.0
        local_best = a_mat.copy()
        local_best_M = M
        stuck = 0
        it = 0

        while time.monotonic() < time_end - 2.0 and it < max_moves:
            it += 1
            if time.monotonic() > time_end - 4.0:
                break

            if M > local_best_M * 1.10 and stuck > 6:
                a_mat = local_best.copy()
                b = conv(a_mat)
                M = float(b.max()) if b.size else 0.0

            cand = coordinate_attempt(a_mat, b, M)
            if cand is not None:
                a_new, _ = cand
                a_mat = _clean(a_new)
                if a_mat is not None:
                    b = conv(a_mat)
                    M = float(b.max()) if b.size else 0.0
                    if M < local_best_M - 1e-12:
                        local_best = a_mat.copy()
                        local_best_M = M
                        try_accept(a_mat)
                    stuck = 0
            else:
                stuck += 1
                if stuck % 25 == 0:
                    sm = smooth_step(a_mat, b, M)
                    if sm is not None:
                        a_mat = _clean(sm)
                        b = conv(a_mat)
                        M = float(b.max()) if b.size else 0.0
                        if M < local_best_M - 1e-12:
                            local_best = a_mat.copy()
                            local_best_M = M
                            try_accept(a_mat)
                    else:
                        z = np.log(local_best + 1e-300) + rng.normal(0.0, 0.05, n)
                        z = np.clip(z - z.max(), -30.0, 0.0)
                        a_mat = _clean(np.exp(z))
                        if a_mat is None:
                            a_mat = local_best.copy()
                        b = conv(a_mat)
                        M = float(b.max()) if b.size else 0.0
                if stuck > 120:
                    break

        try_accept(local_best, force=True)
        return local_best

    def make_starts(base):
        starts = []
        if n > 1:
            xx = np.arange(n, dtype=np.float64) / float(n - 1)
        else:
            xx = np.zeros(n, dtype=np.float64)

        for sig in (0.07, 0.12, 0.20, 0.32):
            z = np.log(base + 1e-300) + rng.normal(0.0, sig, n)
            z = np.clip(z - z.max(), -30.0, 0.0)
            starts.append(np.exp(z))

        for m_f, amp, ph, slope in (
            (1, 0.40, 0.0, 0.50),
            (2, 0.60, 1.0, -0.60),
            (3, 0.50, 2.0, 0.20),
        ):
            v = 1.0 + amp * np.cos(2.0 * np.pi * m_f * xx + ph) + slope * xx
            v = np.maximum(v, 1e-4)
            starts.append(v)

        for lam in (-2.0, -0.7, 0.7, 2.0):
            v = np.exp(lam * xx)
            starts.append(v)

        r = rng.dirichlet(np.full(n, 0.6))
        starts.append(0.75 * base + 0.25 * r)

        return starts[:6]

    # ------------------------------------------------------------------
    # Main search schedule
    # ------------------------------------------------------------------
    try_accept(best_a, force=True)

    total = max(5.0, hard - t0)
    p1 = t0 + 0.40 * total
    p2 = t0 + 0.58 * total
    p3 = t0 + 0.72 * total
    p4 = t0 + 0.88 * total

    # Phase 1: LP-based descent from parent.
    if time.monotonic() < p1:
        a_cur = lp_polish(best_a, p1, max_stuck=18)
        try_accept(a_cur, force=True)

    # Phase 2: coordinate / smooth polishing.
    if time.monotonic() < p2:
        a_cur = coord_polish(best_a, p2, max_moves=6000)
        try_accept(a_cur, force=True)

    # Phase 3: another LP polish.
    if time.monotonic() < p3:
        a_cur = lp_polish(best_a, p3, max_stuck=12)
        try_accept(a_cur, force=True)

    # Phase 4: bounded multi-start escapes.
    if time.monotonic() < p4 - 10.0:
        starts = make_starts(best_a)
        for i, st in enumerate(starts):
            if time.monotonic() > p4 - 5.0:
                break
            rem = max(0.0, p4 - 5.0 - time.monotonic())
            left_starts = max(1, len(starts) - i)
            chunk_end = min(p4 - 4.0, time.monotonic() + rem / left_starts)
            if chunk_end <= time.monotonic() + 2.0:
                continue

            lp_end = time.monotonic() + 0.55 * (chunk_end - time.monotonic())
            if use_lp:
                a_st = lp_polish(st, max(lp_end, time.monotonic() + 1.0), max_stuck=5)
            else:
                a_st = _clean(st)
                if a_st is None:
                    a_st = best_a.copy()

            a_st = coord_polish(a_st, chunk_end, max_moves=500)
            try_accept(a_st, force=True)

    # Phase 5: final local polishing.
    if time.monotonic() < hard - 7.0:
        a_cur = coord_polish(best_a, hard - 5.0, max_moves=5000)
        try_accept(a_cur, force=True)

    if time.monotonic() < hard - 4.0 and use_lp:
        a_cur = lp_polish(best_a, hard - 3.0, max_stuck=6)
        try_accept(a_cur, force=True)

    # ------------------------------------------------------------------
    # Final safe return
    # ------------------------------------------------------------------
    final_a = np.asarray(best_a, dtype=np.float64).ravel()
    if final_a.size == 0 or not np.all(np.isfinite(final_a)) or float(np.sum(final_a)) < 0.01:
        final_a = np.ones(1, dtype=np.float64)

    final_list = [float(x) for x in final_a]
    try:
        fs = float(evaluate_sequence(final_list))
        if not np.isfinite(fs):
            final_list = [1.0] * max(1, final_a.size)
    except Exception:
        final_list = [float(x) for x in np.ones(max(1, final_a.size))]

    return final_list
