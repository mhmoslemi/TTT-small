def fast_lp(f, n, M, A_buf, tlimit=2.0):
    """Solve LP using preallocated buffer."""
    # Update A_buf in place
    A_buf[:] = 0
    for j in range(n):
        A_buf[j:j+n, j] = f
    
    cvec = -np.ones(n)  # preallocate this too
    bvec = M * 1.0000001
    
    try:
        r = linprog(cvec, A_ub=A_buf, b_ub=np.full(2*n-1, M*(1+1e-10)),
                   bounds=[(0, None)]*n, method='highs',
                   options={'time_limit': tlimit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.clip(np.nan_to_num(r.x, nan=0, posinf=0, neginf=0), 0, None)
            sg = g.sum()
            if sg > 1e-30:
                return g / sg
    except:
        pass
    return None

Each iteration:
