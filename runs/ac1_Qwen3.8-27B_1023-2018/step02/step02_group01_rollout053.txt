import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, next_fast_len

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(191, dtype=np.float64) / 191.0
    p = np.clip(p, 0.0, 1000.0)
    if p.size == 0 or not np.isfinite(p).all():
        p = np.ones(191, dtype=np.float64) / 191.0
    return p

def _convolve_square(a):
    n = a.size
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score_normalized(a):
    """Score assuming sum(a)=1. Returns 2*n*max(a*a)."""
    b = _convolve_square(a)
    return 2.0 * a.size * float(b.max())

def _exact_score(a):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    n = a.size
    b = _convolve_square(a)
    mb = float(b.max())
    return 2.0 * n * mb / (s * s)

def _build_A(a):
    n = a.size
    m = 2 * n - 1
    k_idx = np.arange(m, dtype=np.int64)[:, None]
    i_idx = np.arange(n, dtype=np.int64)[None, :]
    j = k_idx - i_idx
    mask = (j >= 0) & (j < n)
    safe_j = np.clip(j, 0, n - 1)
    A = np.where(mask, a[safe_j], 0.0)
    return A

def _lp_best_response(a, M, timeout_s=1.0):
    n = a.size
    A = _build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0.0, None)] * n, method='highs',
                      options={'time_limit': timeout_s})

Let me also think about a key insight: **the best response LP can be reformulated**.

The LP: max sum(g) s.t. (a*g)[k] ≤ M, g ≥ 0.

If we let h = g/M, we get: max sum(h) s.t. (a*h)[k] ≤ 1, h ≥ 0. The solution is g = M*h.

The dual: min sum(λ) s.t. A^T λ ≥ 1, λ ≥ 0.

At the optimum of the dual, λ_k > 0 only for active k (where (a*g)[k] = M).

This dual is independent of M! So we can solve the dual once and get the active set, then compute g from it.

Actually wait, the dual is: min sum(λ) s.t. sum_k λ_k * A[k,i] ≥ 1 for all i, λ ≥ 0.

A[k,i] = a[k-i] (if valid). So the constraint is: sum_k λ_k * a[k-i] ≥ 1 for all i.

This is: (a*λ)[i] ≥ 1 for all i (where λ is extended by 0 to length 2n-1).

So the dual is: min sum(λ) s.t. (a*λ)[i] ≥ 1 ∀i, λ ≥ 0, λ has length 2n-1.

This is also an LP but with 2n-1 variables and n constraints. For n=191, that's 381 vars and 191 constraints.

The primal has n=191 vars and 2n-1=381 constraints.
The dual has 2n-1=381 vars and n=191 constraints.

The dual might be slightly easier (fewer constraints). But both are small.

Let me just go with the primal formulation as before, since it's what the parent code uses.

**Key optimization: Avoid rebuilding A from scratch each iteration.**

Actually, A depends on a, so it changes every iteration. But building A is O(n^2) = 36k ops, which is trivial.

**Better line search:**

Instead of trying fixed step sizes, use a golden section or bisection search on t ∈ [0, 1]:
f(t) = max_k ((1-t)a + tg) * ((1-t)a + tg)[k]

Each evaluation is O(n log n) (for the FFT convolution). Bisection with 20 iterations gives precision 1e-6, costing 20 * O(n log n).

Actually, we want to find the largest t such that f(t) < f(0) = M (current max). If f(t) is unimodal in [0,1] (which it might not be for non-convex problems), bisection works. Otherwise, we just need to find ANY t that improves.

Let me use a simpler approach: try t = 1, 0.9, 0.8, ..., 0.1 and take the first that improves. Or better: ternary search / golden section.

Actually, for the LP best-response, t=1 (full step) sometimes works, sometimes overshoots. A good strategy is to try t=1 first, then bisect down.

    n = a.size

def _subgradient_step(a, alpha):
    """One subgradient step for min max_k (a*a)[k]."""
    b = _convolve_square(a)
    k_star = int(np.argmax(b))
    # Gradient of (a*a)[k_star] w.r.t. a[i] = 2*a[k_star - i] if valid
    n = a.size
    g = np.zeros(n)
    i_range = np.arange(n)
    ji = k_star - i_range

    a_new = np.clip(a_new, 0.0, None)
    s = a_new.sum()

    return a_new

        v /= v.sum()
        try:
            result, M = _lp_optimize(v, stop_time=time.monotonic() + time_per)
            sc = _exact_score_safe(result)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = result.copy()
        except Exception:
            pass

    # Phase 3: Extended LP polish on best
    phase3_end = deadline - 15
    if time.monotonic() < phase3_end - 10:
        result, M = _lp_optimize(best_seq, stop_time=phase3_end)
        sc = _exact_score_safe(result)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = result.copy()

    # Final validation
    try:
        final_score = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_score = float('inf')

    if not np.isfinite(final_score) or final_score >= 2.0:
        return [1.0] * 32

    return [float(x) for x in best_seq]


def _exact_score_safe(a):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    n = a.size
    b = _convolve_square(a)
    mb = float(b.max())
    return 2.0 * n * mb / (s * s)
