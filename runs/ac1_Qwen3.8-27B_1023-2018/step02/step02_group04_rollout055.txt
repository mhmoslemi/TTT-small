Actually, let me think about what the LP actually does. Given f, we solve:
maximize sum(g)
subject to (f*g)(k) <= M for k = 0,...,2n-2
g >= 0

where M = max(f*f).

If f is uniform, f = 1/n * 1, then M = 1/n. The LP becomes:
maximize sum(g)
subject to (1/n) * sum of g in a window <= 1/n
i.e., sum of g in each window of overlapping length <= 1
g >= 0

def _sc(a):

    n = a.size
    s = float(a.sum())

def _lp_step(f, time_limit=10):
    if linprog is None:
        return None
    n = f.size
    b = np.convolve(f, f)
    M = float(b.max())
    if M < 1e-300:
        return None
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    c = -np.ones(n, dtype=np.float64)
    ub = np.full(2 * n - 1, M, dtype=np.float64)
    for _try in range(2):
        try:
            opts = {'time_limit': max(1, int(time_limit))}
            if _try == 1:
                opts = {'presolve': True, 'time_limit': max(1, int(time_limit))}
            r = linprog(c, A_ub=A, b_ub=ub, bounds=[(0.0, None)] * n,
                        method='highs', options=opts)
            if r.success and r.x is not None:
                g = np.clip(np.nan_to_num(r.x, nan=0.0, posinf=0.0, neginf=0.0), 0.0, None)
                sg = float(g.sum())
                if sg > 1e-300:

        except Exception:

    return None
