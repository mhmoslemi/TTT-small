import time
import math
from scipy.fft import rfft, irfft, next_fast_len


def _softmax(x):
    x = np.asarray(x, dtype=np.float64).ravel()
    if x.size == 0:
        return np.array([1.0])
    x = x - float(x.max())
    e = np.exp(np.clip(x, -80.0, 80.0))
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full(x.size, 1.0 / x.size, dtype=np.float64)

    n = a.size
    return irfft(F * F, fs)[:2 * n - 1]

    """Cross-correlation (p ⋆ a)_i = sum_k p_k * a_{k-i}, i=0..n-1"""
    n = a.size

    mh = m / (1 - beta1 ** t)

    elif v == 7:
        a = np.exp(-3.0 * t)
    elif v == 8:
        a = np.exp(-3.0 * (1.0 - t))
    elif v == 9:
        a = 0.5 + 0.5 * np.cos(2.0 * np.pi * t)
    elif v == 10:
        a = 0.5 + 0.5 * np.cos(np.pi * x)

        a = np.abs(np.sin(np.pi * t)) + 0.1
    elif v == 12:
        a = 1.0 / (1.0 + np.exp(15.0 * x))

        idx = (np.arange(n) * P // n).astype(int)
        a = parent[idx].copy()

        a = np.abs(rng.normal(0.0, 1.0, n)) + 0.3
    a += 1e-3 * np.abs(rng.normal(0.0, 1.0, n))

def _optimize(n, a0, w_reg, seed, end_time, hard_deadline, init_tau=0.3, min_tau=0.01):
    """Adam optimization in log-space with smooth max."""
    fs_c = next_fast_len(2 * n - 1)
    fs2 = next_fast_len(3 * n - 2)
    
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    a0 = np.clip(a0, 1e-8, 1000.0)
    a0 = a0 / float(a0.sum())
    
    z = np.log(a0)
    z = z - z.mean()
    
    best_z = z.copy()
    best_score = _score(a0, fs_c, n)
    
    if not np.isfinite(best_score):
        best_score = 2.0 * n  # fallback
    
    rng = np.random.default_rng(seed)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = 0.4
    tau = init_tau
    stuck = 0
    total_iters = 0
    
    while time.monotonic() < min(end_time, hard_deadline):
        a = _softmax(np.clip(z, -30, 30))
        b = _conv_fft(a, fs_c)
        bmax = float(b.max())
        if bmax <= 0 or not np.isfinite(bmax):
            break
        cur = 2.0 * n * bmax
        
        # Smooth max
        tau_eff = max(min_tau, tau)
        bt = b / tau_eff
        bt_shift = bt - bt.max()
        e = np.exp(np.clip(bt_shift, -80, 80))
        s_e = float(e.sum())
        if s_e <= 0 or not np.isfinite(s_e):
            p = np.zeros_like(b)
            p[int(np.argmax(b))] = 1.0
        else:
            p = e / s_e
        
        # Gradient of smooth max
        g = 2.0 * _ccos(p, a, fs2)
        
        # Add flattening regularization
        if w_reg > 0:
            q = b - 1.0 / (2 * n - 1)
            vg = 4.0 * _ccos(q, a, fs2)
            g = g + w_reg * vg
        
        # Log-space gradient
        grad = a * (g - float(np.dot(a, g)))
        
        # Normalize
        gn = float(np.abs(grad).max()) if grad.size > 0 else 0.0
        if not np.isfinite(gn) or gn < 1e-15:
            stuck += 1
            if stuck > 50:
                z = best_z + rng.normal(0, 0.5, n)
                z -= z.mean()
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
                lr = 0.4
                tau = max(min_tau * 2, init_tau * 0.5)
                stuck = 0
            continue
        
        grad = grad / (gn + 1e-30)
        t += 1
        z, m, v = _adam_step(z, grad, m, v, t, lr)
        z -= z.mean()
        z = np.clip(z, -30, 30)
        z[~np.isfinite(z)] = 0.0
        
        a2 = _softmax(z)
        b2 = _conv_fft(a2, fs_c)
        score2 = 2.0 * n * max(float(b2.max()), 0.0)
        
        if np.isfinite(score2) and score2 < best_score - 1e-12:
            best_score = score2
            best_z = z.copy()
            stuck = 0
        else:
            stuck += 1
        
        total_iters += 1
        
        # Adaptive parameter updates
        if total_iters % 20 == 0:
            tau = max(min_tau, tau * 0.95)
        if stuck > 30:
            lr = max(0.05, lr * 0.85)
        if stuck > 150:
            z = best_z + rng.normal(0, 0.5, n)
            z -= z.mean()
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            lr = 0.4
            tau = max(min_tau * 2, init_tau * 0.7)
            stuck = 0
    
    a_best = _softmax(best_z)
    return a_best, best_score


def _polish(a, end_time, hard_deadline):
    """Local search polish."""
    n = a.size
    fs_c = next_fast_len(2 * n - 1)
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 1e-9, 1000.0)
    a = a / float(a.sum())
    
    best = a.copy()
    best_s = _score(a, fs_c, n)
    
    rng = np.random.default_rng(42)
    sigma = 0.2
    stuck = 0
    
    while time.monotonic() < min(end_time, hard_deadline) - 0.3:
        z = np.log(np.maximum(a, 1e-12))
        r = float(rng.random())
        
        if r < 0.3:
            zc = z + rng.normal(0, sigma, n)
        elif r < 0.5:
            i = int(rng.integers(0, n))
            w = max(1, min(n // 10 + 1, int(rng.integers(1, n // 20 + 2))))
            i2 = min(n, i + w)
            zc = z.copy()
            zc[i:i2] += rng.normal(0, sigma * 0.5, i2 - i)
        elif r < 0.65:
            zc = z * float(rng.uniform(0.5, 2.0))
        elif r < 0.8:
            f = int(rng.integers(1, max(2, n // 4)))
            ph = float(rng.uniform(0, 2 * math.pi))
            amp = float(rng.uniform(0, sigma * 0.3))
            zc = z + amp * np.sin(2 * math.pi * f * np.arange(n) / n + ph)
        else:
            # Mirror flip
            zc = z.copy()
            i = int(rng.integers(0, n // 2))
            j = n - 1 - i
            zc[i], zc[j] = zc[j], zc[i]
        
        zc -= zc.mean()
        a2 = _softmax(zc)
        s2 = _score(a2, fs_c, n)
        
        if np.isfinite(s2) and s2 < best_s - 1e-12:
            best = a2.copy()
            best_s = s2
            a = a2.copy()
            stuck = 0
            sigma = max(0.05, sigma * 0.9)
        else:
            stuck += 1
            sigma = min(0.8, sigma * 1.03)
            if stuck > 40:
                a = best.copy()
                sigma = 0.3
                stuck = 0
    
    return best, best_s


def _score(a, fs, n):
    b = _conv_fft(a, fs)
    return 2.0 * n * max(float(b.max()), 0.0)

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len


def _softmax_safe(x):
    x = np.asarray(x, dtype=np.float64).ravel()
    if x.size == 0:
        return np.array([1.0])
    x = x - float(np.max(x))
    e = np.exp(np.clip(x, -80.0, 80.0))
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full(x.size, 1.0 / x.size, dtype=np.float64)
    return e / s


def _conv_full(a, fs):
    n = a.size
    F = rfft(a, fs)
    return irfft(F * F, fs)[:2 * n - 1]


def _xcorr(p, a, fs2):
    """Compute (p ⋆ a)_i = sum_k p_k * a_{k-i} for i=0..n-1."""
    n = a.size
    ar = a[::-1]
    c = irfft(rfft(ar, fs2) * rfft(p, fs2), fs2)
    return c[n - 1:2 * n - 1]


def _eval_score(a, fs):
    b = _conv_full(a, fs)
    return 2.0 * a.size * float(np.max(b))


def _grad_smooth_max(a, b, fs2, tau):
    """Gradient of smooth max of b w.r.t. a, with temperature tau."""
    bt = b / tau
    bt = bt - float(np.max(bt))
    e = np.exp(np.clip(bt, -80, 80))
    s = float(e.sum())
    if s <= 0 or not np.isfinite(s):
        p = np.zeros_like(b)
        p[int(np.argmax(b))] = 1.0
    else:
        p = e / s
    g = 2.0 * _xcorr(p, a, fs2)
    return g, p


def _adam_update(z, g, m, v, t, lr):
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = b1 * m + (1 - b1) * g
    v = b2 * v + (1 - b2) * g * g
    mh = m / (1 - b1 ** t)
    vh = v / (1 - b2 ** t)
    z_new = z - lr * mh / (np.sqrt(vh) + eps)
    return z_new, m, v


def _make_init(n, kind, rng, parent=None):
    t = (np.arange(n) + 0.5) / n
    x = 2.0 * t - 1.0
    k = int(kind)
    if k == 0:
        a = np.ones(n, dtype=np.float64)
    elif k == 1:
        a = 1.0 - np.abs(x)
    elif k == 2:
        a = 1.0 - x ** 2
    elif k == 3:
        a = 0.5 * (1.0 + np.cos(np.pi * x))
    elif k == 4:
        a = (1.0 - np.abs(x)) ** 3
    elif k == 5:
        a = np.exp(-2.5 * np.abs(x))
    elif k == 6:
        a = 0.5 + 0.5 * np.sin(np.pi * t)
    elif k == 7:
        a = 0.5 + 0.5 * np.cos(2.0 * np.pi * t)
    elif k == 8:
        a = 0.5 + 0.5 * np.cos(np.pi * x)
    elif k == 9:
        a = np.abs(np.sin(np.pi * t)) + 0.05
    elif k == 10:
        a = 0.5 * (1.0 - np.abs(x)) + 0.5 * (1.0 / (1.0 + np.exp(18.0 * x)))
    elif k == 11 and parent is not None and parent.size > 1:
        P = parent.size
        if n == P:
            a = parent.copy()
        else:
            idx = (np.arange(n) * P // n).astype(int)
            a = parent[idx].astype(np.float64)
    elif k == 12:
        # Multi-harmonic
        a = np.ones(n, dtype=np.float64)
        for h in range(1, 4):
            f = h * 2
            ph = float(rng.uniform(0, 2 * math.pi))
            amp = float(rng.uniform(0.1, 0.3)) / h
            a += amp * np.sin(2 * math.pi * f * t + ph)
    else:
        a = np.abs(rng.normal(0, 1, n)) + 0.2
    a += 1e-4 * np.abs(rng.normal(0, 1, n))
    a = np.maximum(a, 1e-8)
    a = a / float(a.sum())
    return a


def _run_opt(n, a0, w_reg, seed, t_end, t_hard, tau0=0.25, tau_min=0.005):
    fs_c = next_fast_len(2 * n - 1)
    fs2 = next_fast_len(3 * n - 2)
    
    a0 = np.asarray(a0, np.float64).ravel()
    a0 = np.clip(a0, 1e-8, 1000.0)
    a0 = a0 / max(1e-300, float(a0.sum()))
    
    z = np.log(a0)
    z -= z.mean()
    
    best_z = z.copy()
    best_s = _eval_score(a0, fs_c)
    
    if not np.isfinite(best_s):
        best_s = 2.0 * n
    
    rng = np.random.default_rng(seed)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = 0.35
    tau = tau0
    stuck = 0
    it = 0
    
    while time.monotonic() < min(t_end, t_hard):
        a = _softmax_safe(np.clip(z, -30, 30))
        b = _conv_full(a, fs_c)
        bm = float(b.max())
        if bm <= 0 or not np.isfinite(bm):
            break
        cur = 2.0 * n * bm
        
        # Smooth max gradient
        g, p = _grad_smooth_max(a, b, fs2, max(tau, 1e-10))
        
        # Flattening regex
        if w_reg > 0:
            q = b - 1.0 / (2 * n - 1)
            g += w_reg * 4.0 * _xcorr(q, a, fs2)
        
        # Log-space gradient
        gdot = float(np.dot(a, g))
        grad = a * (g - gdot)
        
        gn = float(np.abs(grad).max()) if grad.size else 0.0
        if not np.isfinite(gn) or gn < 1e-14:
            stuck += 1
            if stuck > 60:
                z = best_z + rng.normal(0, 0.4, n)
                z -= z.mean()
                m = v = np.zeros(n)
                t = 0
                lr = 0.35
                tau = max(tau_min * 3, tau0 * 0.4)
                stuck = 0
            continue
        
        grad /= (gn + 1e-30)
        t += 1
        z, m, v = _adam_update(z, grad, m, v, t, lr)
        z -= z.mean()
        z = np.clip(z, -30, 30)
        z[~np.isfinite(z)] = 0.0
        
        a2 = _softmax_safe(z)
        s2 = _eval_score(a2, fs_c)
        it += 1
        
        if np.isfinite(s2) and s2 < best_s - 1e-13:
            best_s = s2
            best_z = z.copy()
            stuck = 0
        else:
            stuck += 1
        
        # Schedule updates
        if it % 15 == 0:
            tau = max(tau_min, tau * 0.96)
        if stuck > 25:
            lr = max(0.03, lr * 0.88)
        if stuck > 120:
            z = best_z + rng.normal(0, 0.4, n)
            z -= z.mean()
            m = v = np.zeros(n)
            t = 0
            lr = 0.35
            tau = max(tau_min * 3, tau0 * 0.6)
            stuck = 0
    
    a_best = _softmax_safe(best_z)
    return a_best, best_s


def _polish_local(a, t_end, t_hard):
    n = a.size
    fs_c = next_fast_len(2 * n - 1)
    a = np.asarray(a, np.float64).ravel()
    a = np.clip(a, 1e-9, 1000.0)
    a = a / float(a.sum())
    
    best = a.copy()
    bs = _eval_score(best, fs_c)
    
    rng = np.random.default_rng(777)
    sigma = 0.15
    stuck = 0
    
    while time.monotonic() < min(t_end, t_hard) - 0.3:
        z = np.log(np.maximum(a, 1e-12))
        r = float(rng.random())
        
        if r < 0.35:
            zc = z + rng.normal(0, sigma, n)
        elif r < 0.55:
            i = int(rng.integers(0, n))
            w = max(2, int(n * rng.uniform(0.02, 0.15)))
            i2 = min(n, i + w)
            zc = z.copy()
            zc[i:i2] += rng.normal(0, sigma * 0.6, i2 - i)
        elif r < 0.70:
            zc = z * float(rng.uniform(0.5, 2.0))
        elif r < 0.85:
            f = int(rng.integers(1, max(2, n // 5)))
            ph = float(rng.uniform(0, 2 * math.pi))
            zc = z + float(rng.uniform(0, sigma * 0.4)) * np.sin(
                2 * math.pi * f * np.arange(n) / n + ph)
        else:
            # Mirror pair swap
            zc = z.copy()
            i = int(rng.integers(0, n // 2))
            j = n - 1 - i
            zc[i], zc[j] = zc[j] + rng.normal(0, sigma * 0.3), zc[i] + rng.normal(0, sigma * 0.3)
        
        zc -= zc.mean()
        a2 = _softmax_safe(zc)
        s2 = _eval_score(a2, fs_c)
        
        if np.isfinite(s2) and s2 < bs - 1e-13:
            best = a2.copy()
            bs = s2
            a = a2.copy()
            stuck = 0
            sigma = max(0.03, sigma * 0.92)
        else:
            stuck += 1
            sigma = min(0.6, sigma * 1.04)
            if stuck > 50:
                a = best.copy()
                z = np.log(np.maximum(a, 1e-12))
                sigma = 0.2
                stuck = 0
    
    return best, bs
