    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n):
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_min_peak(f, n):
    """Minimize max_k (f*g)_k s.t. sum(g)=1, g>=0. Returns g* or None."""
    A = _build_A(f, n)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [1.0]
    bounds = [(0, None)] * n + [(-1, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 10.0})
    except Exception:
        return None
    if res.status != 0:
        return None
    g = np.clip(res.x[:n], 0, None)
    gs = g.sum()
    if gs < 1e-15:
        return None
    return g / gs


def _lp_max_sum(f, n):
    """Maximize sum(g) s.t. (f*g)_k <= M, g>=0. Returns normalized g* or None."""
    b = _conv_full(f)
    M = float(b.max())
    if M < 1e-15:
        return None
    A = _build_A(f, n)
    c = -np.ones(n)
    try:
        res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                      bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True, 'time_limit': 10.0})
    except Exception:
        return None
    if res.status != 0:
        return None
    g = np.clip(res.x, 0, None)
    gs = g.sum()
    if gs < 1e-15:
        return None
    return g / gs


def _lp_optimize(f0, n, stop_time, rng):
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel()[:n].copy(), 0, None)
    if f.size < n:
        f = np.concatenate([f, np.zeros(n - f.size)])
    f = f[:n]
    fs = f.sum()
    if fs < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / fs

    best_f = f.copy()
    best_score = _score(f)
    t = 0.3
    t_min, t_max = 0.005, 0.95
    stall, max_stall = 0, 250
    it = 0

    while time.monotonic() < stop_time - 1.0:
        it += 1
        g = _lp_min_peak(f, n)
        if g is None and it % 5 == 0:
            g = _lp_max_sum(f, n)
        if g is None:
            stall += 6
            if stall > max_stall:
                f = np.clip(f + rng.normal(0, 0.08, n), 0, None)
                s = f.sum()
                f = f / s if s > 1e-10 else best_f.copy()
                stall, t = 0, 0.3
            continue
        if np.linalg.norm(g - f) < 1e-7:
            stall += 3
            if stall > max_stall:
                f = np.clip(f + rng.normal(0, 0.12, n), 0, None)
                s = f.sum()
                f = f / s if s > 1e-10 else best_f.copy()
                stall, t = 0, 0.3
            continue

        best_t, best_loc = 0.0, _score(f)
        for ct in [t, t*0.5, t*1.5, 0.1, 0.4, 0.7, 0.9]:
            if not (t_min <= ct <= t_max):
                continue
            fn = np.clip((1 - ct) * f + ct * g, 0, None)
            sn = fn.sum()
            if sn < 1e-10:
                continue
            fn /= sn
            sc = _score(fn)
            if sc < best_loc:
                best_loc, best_t = sc, ct

        if best_t > 0 and best_loc < best_score:
            f = np.clip((1 - best_t) * f + best_t * g, 0, None)
            s = f.sum()
            f /= s if s > 1e-10 else (f := best_f.copy())
            best_f, best_score = f.copy(), _score(f)
            t = min(t_max, t * 1.04)
            stall = 0
        else:
            stall += 1
            t = max(t_min, t * 0.55)
            if stall >= max_stall:
                f = np.clip(f + rng.normal(0, 0.1, n), 0, None)
                s = f.sum()
                f = f / s if s > 1e-10 else best_f.copy()
                stall, t = 0, 0.3
    return best_f, best_score


def _polish(a, stop_time):
    n = len(a)
    best = np.clip(np.asarray(a, dtype=np.float64).ravel(), 0, None).copy()
    fs = best.sum()
    if fs < 1e-10:
        return best, _score(best)
    best /= fs
    best_sc = _score(best)
    rng = np.random.default_rng(99999)
    no_imp = 0
    while time.monotonic() < stop_time - 1.0:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 1.0:
                break
            orig = best[i]
            cv = _conv_full(best)
            s_val = best.sum()
            sc0 = 2.0 * n * cv.max() / (s_val * s_val)
            if orig < 1e-15:
                for v in [0.001, 0.005, 0.01, 0.05, 0.1, 0.2]:
                    best[i] = v
                    sc = _score(best)
                    if sc < best_sc - 1e-14:
                        best_sc, improved = sc, True
                        orig = v
                best[i] = orig
            else:
                mlt_b, sc_b = 1.0, best_sc
                for mlt in [0.25, 0.4, 0.6, 0.8, 1.2, 1.5, 2.0, 3.0, 0.0]:
                    best[i] = orig * mlt if mlt > 0 else 0.0
                    sc = _score(best)
                    if sc < sc_b - 1e-14:
                        sc_b, mlt_b = sc, mlt
                if sc_b < best_sc - 1e-14:
                    best[i] = orig * mlt_b if mlt_b > 0 else 0.0
                    best_sc, improved = sc_b, True
        if not improved:
            no_imp += 1
            if no_imp >= 30:
                break
        else:
            no_imp = 0
    return best, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 15.0
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
    except Exception:
        parent = np.ones(320) / 320
    parent = np.clip(np.where(np.isfinite(parent), parent, 0.0), 0, 1000)
    if parent.size == 0:
        parent = np.ones(320) / 320
    n_p = len(parent)
    rng = np.random.default_rng(seed)

    best_seq, best_score = parent.copy(), _score(parent)
    if not math.isfinite(best_score):
        best_seq = np.ones(320) / 320
        best_score = _score(best_seq)

    i_arr = np.arange(n_p, dtype=np.float64)
    inits = [parent.copy(),
             0.5 + 0.5 * np.cos(2 * np.pi * i_arr / n_p)]
    sup = np.ones(n_p) * 0.2
    for fr in [1, 2, 3, 5]:
        sup += 0.3 * np.cos(2 * np.pi * fr * i_arr / n_p + rng.uniform(0, 2 * np.pi))
    inits.append(np.abs(sup) + 0.05)
    inits.append(np.abs(rng.normal(0, 1, n_p)) + 0.1)
    inits.append(np.abs(parent + rng.normal(0, 0.2 * max(1.0, np.mean(parent)), n_p)) + 0.01)

    p1_end = t0 + budget * 0.50
    for idx, init in enumerate(inits):
        if time.monotonic() > p1_end - 5 or best_score < 1.50:
            break
        stop = min(p1_end - 3, time.monotonic() + max(15.0, (p1_end - t0) / 5))
        try:
            res, sc = _lp_optimize(init, n_p, stop, rng)
            res = np.clip(res, 0, 1000)
            if res.sum() < 0.01:
                continue
            ex = float(evaluate_sequence([float(x) for x in res]))
            if math.isfinite(ex) and ex < best_score:
                best_seq, best_score = res.copy(), ex
        except Exception:
            pass

    p2_end = t0 + budget * 0.78
    oths = sorted(set([256, 320, 384, 448, 512, n_p]))
    tper = max(8.0, (p2_end - p1_end) / len(oths))
    for L in oths:
        if time.monotonic() > p2_end - 5 or best_score < 1.50:
            break
        stop = min(p2_end - 3, time.monotonic() + tper)
        il = best_seq.copy() if L == len(best_seq) else np.interp(
            np.linspace(0, 1, L), np.linspace(0, 1, n_p), parent)
        try:
            res, sc = _lp_optimize(il, L, stop, rng)
            res = np.clip(res, 0, 1000)
            if res.sum() < 0.01:
                continue
            ex = float(evaluate_sequence([float(x) for x in res]))
            if math.isfinite(ex) and ex < best_score:
                best_seq, best_score = res.copy(), ex
        except Exception:
            pass

    if time.monotonic() < deadline - 5:
        try:
            best_seq, best_score = _polish(best_seq, deadline)
        except Exception:
            pass

    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf
    if not math.isfinite(final_sc):
        return [1.0]
    return [float(x) for x in best_seq]
