                best_f = a.copy()
                stuck = 0

            # Soft-max weights
            tau = max(1e-12, 0.02 * M)
            xw = np.clip((c - M) / tau, -50.0, 50.0)
            w = np.exp(xw)
            ws = w.sum()
            if ws < 1e-30:
                w = np.zeros(nconv)
                w[int(np.argmax(c))] = 1.0
            else:
                w /= ws

            # Gradient: grad[j] = 2 * sum_m a[m] * w[m+j]
            grad = np.zeros(n)
            for j in range(n):
                # sum_{m=0}^{n-1} a[m] * w[m+j]
                # w indices: j, j+1, ..., j+n-1 (all < 2n-1 since j <= n-1, j+n-1 <= 2n-2)
                grad[j] = 2.0 * float(np.dot(a, w[j:j + n]))

            # Projected gradient step
            improved = False
            for scale in [lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 0.05]:
                a_new = a - scale * grad
                a_new = np.maximum(a_new, 0)
                a_new = normalize(a_new)
                cand_sc = score_of(a_new)
                if cand_sc < sc - 1e-14:
                    a = a_new
                    sc = cand_sc
                    improved = True
                    break
            if improved:
                lr = min(lr * 1.03, 1.5)
                stuck = 0
            else:
                stuck += 1
                lr = max(0.005, lr * 0.92)

    # ============ PHASE 3: Simulated annealing ============
    if time.monotonic() < deadline - 40:
        a = best_f.copy()
        sc = best_score
        rng = np.random.default_rng(seed + 999)
        T = 0.002
        T_min = 1e-7
        cool = 0.998

        for it in range(20000):
            if time.monotonic() > deadline - 40:
                break
            if T < T_min:
                break

            r = rng.random()
            if r < 0.35:
                # Log-normal perturbation
                z = np.log(np.clip(a, 1e-12, None))
                z += rng.normal(0, 0.08, n)
                z -= z.max()
                cand = np.exp(z)
                cand = normalize(cand)
            elif r < 0.60:
                # Block scaling
                cand = a.copy()
                k = int(rng.integers(0, n - 5))
                width = int(rng.integers(3, max(4, n // 15)))
                width = min(width, n - k)
                factor = exp(rng.normal(0, 0.1))
                cand[k:k + width] *= factor
                cand = normalize(cand)
            elif r < 0.80:
                # Coordinate swap
                cand = a.copy()
                i1 = int(rng.integers(0, n))
                i2 = int(rng.integers(0, n))
                amt = 0.01 * a.sum()
                cand[i1] += amt
                cand[i2] = max(0, cand[i2] - amt)
                cand = normalize(cand)
            else:
                # Mirror blend
                alpha = rng.uniform(0.05, 0.3)
                cand = (1 - alpha) * a + alpha * a[::-1]
                cand = normalize(cand)

            cand_sc = score_of(cand)
            delta = cand_sc - sc
            if delta < -1e-14:
                a = cand
                sc = cand_sc
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_f = a.copy()
            else:
                # Accept with probability exp(-delta/T)
                if delta < 50 * T:
                    if rng.random() < math.exp(-delta / T):
                        a = cand
                        sc = cand_sc

            T *= cool

    # ============ PHASE 4: Final fine-tuning ============
    if time.monotonic() < deadline - 15:
        a = best_f.copy()
        sc = best_score
        lr = 0.02
        for it in range(5000):
            if time.monotonic() > deadline - 15:
                break

            c = convolve_full(a)
            M = float(c.max())
            sc = 2.0 * n * M

            if sc < best_score - 1e-15:
                best_score = sc
                best_f = a.copy()

            # Sharp weights (small tau for precision)
            tau = max(1e-14, 0.005 * M)
            xw = np.clip((c - M) / tau, -40.0, 40.0)
            w = np.exp(xw)
            ws = w.sum()
            if ws < 1e-30:
                break
            w /= ws

            grad = np.zeros(n)
            for j in range(n):
                grad[j] = 2.0 * float(np.dot(a, w[j:j + n]))

            a_new = a - lr * grad
            a_new = np.maximum(a_new, 0)
            a_new = normalize(a_new)
            cand_sc = score_of(a_new)
            if cand_sc < sc - 1e-15:
                a = a_new
                sc = cand_sc
            else:
                lr = max(0.001, lr * 0.9)

    # ============ Final validation ============
    best_f = np.clip(best_f, 0.0, 1000.0)
    if not np.isfinite(best_f).all() or best_f.sum() < 0.01:
        best_f = np.ones(n) / n
        best_score = 2.0

    final_list = [float(x) for x in best_f]
    try:
        fs = float(evaluate_sequence(final_list))
    except Exception:
        fs = float('inf')

    if not np.isfinite(fs):
        final_list = [1.0 / n] * n
        try:
            fs = float(evaluate_sequence(final_list))
        except Exception:
            pass
        if not np.isfinite(fs):
            return [1.0]

    return final_list
