        local = fast_score(a)
        update_best(a)
        best_local = local

        alpha = 0.015
        stall = 0
        v = np.zeros(n)
        momentum = 0.85

        while time.monotonic() < stop_time:
            b = np.convolve(a, a)
            M = float(b.max())
            if not math.isfinite(M) or M <= 0:
                break

            # Active set: within 99.8% of max
            thresh = M * 0.998
            active = np.where(b >= thresh)[0]

            grad = np.zeros(n)
            for k in active:
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                if lo <= hi:
                    grad[lo:hi + 1] += 2.0 * a[k - lo:k - lo + (hi - lo)]
            grad /= max(1, len(active))

            v = momentum * v + (1 - momentum) * grad
            a_trial = a - alpha * v
            a_trial = np.clip(a_trial, 0.0, None)
            st = a_trial.sum()
            if st < 1e-15:
                stall += 5
                alpha *= 0.5
                continue
            a_trial /= st
            sc = fast_score(a_trial)

            if sc < best_local * (1.0 - 1e-10):
                a = a_trial
                best_local = sc
                stall = 0
                alpha = min(alpha * 1.02, 0.2)
                if len(active) < 10:
                    momentum = min(momentum + 0.01, 0.95)
                update_best(a)
            else:
                found = False
                for frac in [0.5, 0.25, 0.1]:
                    a_t = a - alpha * frac * v
                    a_t = np.clip(a_t, 0.0, None)
                    s2 = a_t.sum()
                    if s2 < 1e-15:
                        continue
                    a_t /= s2
                    sc2 = fast_score(a_t)
                    if sc2 < best_local * (1.0 - 1e-10):
                        a = a_t
                        best_local = sc2
                        alpha *= 0.8
                        stall = 0
                        update_best(a)
                        found = True
                        break
                if not found:
                    alpha *= 0.6
                    v *= 0.5
                    stall += 1

            if stall > 50:
                a_p = a * np.exp(0.03 * rng.standard_normal(n))
                a_p = np.clip(a_p, 0.0, None)
                sp = a_p.sum()
                if sp > 1e-15:
                    a_p /= sp
                    a = a_p
                    v = np.zeros(n)
                    best_local = fast_score(a)
                    stall = 0
                    update_best(a)
                if stall > 100:
                    break
        return a

    # === Interpolation to different lengths ===
    def interp_len(src, n):
        if src.size == n:
            return src.copy()
        ox = np.linspace(0.0, 1.0, src.size)
        nx = np.linspace(0.0, 1.0, n)
        out = np.interp(nx, ox, src)
        out = np.clip(out, 0.0, None)
        s = float(out.sum())
        if s > 1e-15:
            out /= s
        else:
            out = np.ones(n, dtype=np.float64) / n
        return out

    # === MAIN EXECUTION ===
    p = parent if parent is not None else best_arr
    n_p = p.size
    p_norm = p / p.sum()

    # Phase 1: Soft-max gradient (main workhorse, most time)
    t1 = time.monotonic()
    alloc1 = min(500.0, (deadline - t1) * 0.42)
    if alloc1 > 40:
        a1 = soft_max_optimize(p_norm, n_p, t1 + alloc1, beta_init=10.0, beta_max=150.0)
        update_best(a1)
        base_a = a1 if a1 is not None else p_norm
    else:
        base_a = p_norm

    # Phase 2: Hard-max gradient for precision (tighter active set)
    if time.monotonic() < deadline - 80:
        t2 = time.monotonic()
        alloc2 = min(350.0, (deadline - t2) * 0.3)
        if alloc2 > 30:
            a2 = hard_max_optimize(base_a, n_p, t2 + alloc2)
            update_best(a2)
            if a2 is not None:
                base_a = a2

    # Phase 3: LP refinement
    if time.monotonic() < deadline - 60:
        t3 = time.monotonic()
        alloc3 = min(200.0, (deadline - t3) * 0.25)
        if alloc3 > 20:
            a3 = lp_refine(base_a, t3 + alloc3)
            update_best(a3)
            if a3 is not None:
                base_a = a3
            # Quick gradient polish after LP
            if time.monotonic() < deadline - 30:
                t3b = time.monotonic()
                a3b = hard_max_optimize(base_a, n_p, min(t3b + 30, deadline - 20))
                update_best(a3b)
                if a3b is not None:
                    base_a = a3b

    # Phase 4: Try neighboring lengths
    for L in [n_p - 32, n_p - 16, n_p + 16, n_p + 32, 160, 192, 256, 384, 448]:
        if time.monotonic() >= deadline - 45:
            break
        if L < 64 or L > 640 or L == n_p:
            continue
        a_init = interp_len(base_a, L)
        t4 = time.monotonic()
        alloc4 = min(40.0, (deadline - t4) * 0.1)
        if alloc4 < 10:
            continue
        a_L = soft_max_optimize(a_init, L, t4 + alloc4, beta_init=12.0, beta_max=120.0)
        if a_L is not None:
            update_best(a_L)
            if time.monotonic() < deadline - 25:
                a_L2 = hard_max_optimize(a_L, L, min(time.monotonic() + 15, deadline - 20))
                update_best(a_L2)

    # Phase 5: Final polish on best found
    if best_arr is not None and time.monotonic() < deadline - 25:
        n_b = best_arr.size
        a_pol = best_arr / best_arr.sum()
        t5 = time.monotonic()
        alloc5 = min(60.0, deadline - t5 - 10)
        if alloc5 > 10:
            a_pol = hard_max_optimize(a_pol, n_b, t5 + alloc5)
            update_best(a_pol)

    # Final validation
    if best_arr is None:
        best_arr = np.ones(320, dtype=np.float64)
        update_best(best_arr)

    final = np.clip(best_arr, 0.0, 1000.0)
    if float(final.sum()) < 0.01:
        final = np.ones(320, dtype=np.float64)
        update_best(final)

    return [float(x) for x in best_arr]
