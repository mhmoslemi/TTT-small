import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _softmax(x):
    x = np.asarray(x, dtype=np.float64).ravel()
    if x.size == 0:
        return np.array([1.0])
    x = x - np.max(x)
    e = np.exp(np.clip(x, -80.0, 80.0))
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full(x.size, 1.0 / x.size, dtype=np.float64)
    return e / s

def _conv(a, fs):
    F = rfft(a, fs)
    n = len(a)
    return irfft(F * F, fs)[:2*n-1]

def _cross_corr(p, a, fs2):
    """Compute gradient of sum_k p[k]*(a*a)[k] w.r.t. a"""
    ar = a[::-1]
    c = irfft(rfft(ar, fs2) * rfft(p, fs2), fs2)
    n = len(a)
    return c[n-1:2*n-1]

def _score(a):
    n = len(a)
    b = _conv(a, next_fast_len(2*n-1))
    return 2.0 * n * float(b.max())

def _lp_refine(a, max_iters=20, tol=1e-8):
    """LP-based refinement using the state-of-the-art approach"""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = a / max(1e-300, float(a.sum()))
    
    for it in range(max_iters):
        b = _conv(a, next_fast_len(2*n-1))
        M = float(b.max())
        if M < 1e-15:
            break
        
        # Build LP: min t s.t. (a*b)[k] <= t, sum(b)=1, b>=0
        # (a*b)[k] = sum_i a[i]*b[k-i]
        # This is a linear function of b for each k
        # A[k, j] = coefficient of b[j] in (a*b)[k]
        # (a*b)[k] = sum_j a[k-j]*b[j] (where 0<=k-j<n, 0<=j<n)
        # So A[k, j] = a[k-j] if 0 <= k-j < n else 0
        
        # For k = 0, ..., 2n-2 and j = 0, ..., n-1
        # A[k, j] = a[k-j] if k-j is in [0, n-1] else 0
        
        # This is a (2n-1) x n matrix, but it's structured (like a Toeplitz matrix)
        # For efficiency, I'll construct it explicitly for small n
        
        # Actually, for n=192, 2n-1=383 constraints, n=192 variables.
        # The constraint matrix is sparse (each row has at most n nonzeros, most have fewer).
        
        # Let me construct A explicitly
        A_rows = []
        for k in range(2*n-1):
            row = np.zeros(n)
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    row[j] = a[idx]
            A_rows.append(row)
        A_conv = np.array(A_rows)  # shape (2n-1, n)
        
        # LP: min t
        # s.t. A_conv @ b - t <= 0  (2n-1 constraints)
        #      sum(b) = 1            (1 equality)
        #      b >= 0                 (n bounds)
        
        # Variables: [b_0, ..., b_{n-1}, t]
        # Objective: [0, ..., 0, 1] (minimize t)
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        
        # Inequality constraints: A_conv @ b - t <= 0
        A_ub = np.hstack([A_conv, -np.ones((2*n-1, 1))])
        b_ub = np.zeros(2*n-1)
        
        # Equality: sum(b) = 1
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.ones(1)
        
        # Bounds: b_j >= 0, t >= 0
        bounds = [(0, None)] * n + [(0, None)]
        
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                      bounds=bounds, method='highs')
        
        if not res.success:
            break
        
        b_opt = res.x[:n]
        t_opt = res.x[n]
        
        if t_opt >= M - tol:
            break  # No improvement possible
        
        # Mix
        t_mix = 0.5
        a_new = (1 - t_mix) * a + t_mix * b_opt
        a_new = np.maximum(a_new, 0)
        s = float(a_new.sum())
        if s < 1e-6:
            break
        a_new /= s
        
        # Check if it actually improved
        b_new = _conv(a_new, next_fast_len(2*n-1))
        M_new = float(b_new.max())
        if M_new < M - 1e-12:
            a = a_new
        else:
            # Try smaller step
            t_mix = 0.1
            a_new = (1 - t_mix) * a + t_mix * b_opt
            a_new = np.maximum(a_new, 0)
            s = float(a_new.sum())
            if s < 1e-6:
                break
            a_new /= s
            b_new = _conv(a_new, next_fast_len(2*n-1))
            M_new = float(b_new.max())
            if M_new < M - 1e-12:
                a = a_new
            else:
                break
    
    a = a / max(1e-300, float(a.sum()))
    return a

    stuck = 0

        a = _softmax(np.clip(z, -30, 30))

        bmax = float(b.max())
        if not np.isfinite(bmax) or bmax <= 0:
            break
        cur_score = 2.0 * n * bmax

            zc[i:i2] += rng.normal(0, sigma * 0.5, i2 - i)
