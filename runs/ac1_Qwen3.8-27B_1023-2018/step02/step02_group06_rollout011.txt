import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
import cvxpy as cp

def _fft_autocorr(a):
    """Compute full autocorrelation of a using FFT."""
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2*n-1]
    np.maximum(c, 0.0, out=c)
    c[~np.isfinite(c)] = 0.0
    return c

def _make_A(a):
    """Build the convolution matrix A where (A @ g)[k] = (a*g)[k]."""
    n = len(a)
    # A is (2n-1) x n
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    k_idx = np.arange(2*n-1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    A = np.zeros((2*n-1, n))
    valid = (diff >= 0) & (diff < n)
    # For valid entries, A[k,j] = a[diff]
    diff_valid = diff[valid].astype(int)
    A[valid] = a[diff_valid]
    return A

def _normalize(a):
    """Normalize so sum = 1, clip to [0, 1000]."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    a = a / s
    a = np.minimum(a, 1000.0)
    return a

def _score_fast(a):
    """Fast score computation (assumes sum=1, so score = 2n * max_autocorr)."""
    n = len(a)
    b = _fft_autocorr(a)
    return 2.0 * n * float(b.max())

def _solve_lp(a, M, n, solver=None):
    """Solve: max sum(g) s.t. A@x <= M, g >= 0."""
    A = _make_A(a)
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    
    solvers = solver if solver else [cp.GLOP, cp.CBC, cp.GLPK, cp.CBC]
    for s in solvers:
        try:
            prob.solve(solver=s, verbose=False)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                gv = g.value
                if gv is not None:
                    gv = np.asarray(gv, dtype=np.float64)
                    if np.isfinite(gv).all():
                        gv = np.maximum(gv, 0)
                        sv = gv.sum()
                        if sv > 1e-15:
                            return gv / sv
        except Exception:
            continue
    return None

def _lp_iterate(a, n, deadline, rng, max_iters=200, patience=25):
    """Main LP-based optimization loop."""
    a = _normalize(a.copy())
    best_a = a.copy()
    best_score = _score_fast(a)
    
    no_improve = 0
    it = 0
    
    while time.monotonic() < deadline and it < max_iters:
        it += 1
        b = _fft_autocorr(a)
        M = float(b.max())
        cur_score = 2.0 * n * M
        
        if cur_score < best_score * (1 - 1e-10):
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > patience:
            # Perturb
            a = a + rng.normal(0, 0.1, n)
            a = _normalize(a)
            no_improve = 0
            patience = max(15, patience + 5)
            continue
        
        # Solve LP for direction

        if g is None:

                a = a + rng.normal(0, 0.05, n)
                a = _normalize(a)

        # Check similarity

        if cos_sim > 0.9995:

def _fft_autocorr(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    c[~np.isfinite(c)] = 0.0
    return c

def _make_A(a):
    n = len(a)
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    A = np.zeros((2 * n - 1, n))
    valid = (diff >= 0) & (diff < n)
    A[valid] = a[diff[valid].astype(int)]
    return A

def _norm(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s
