        g = _lpsolve(a)
        if g is None:
            stall += 5
            if stall > 50:
                break
            a = _nf(a * np.exp(0.05 * rng.standard_normal(n)))
            loc = _sc(a)
            _upd(a, best)
            continue
        gn = _nf(g)
        # Quick check: is g close to a?
        if np.allclose(gn, a, atol=1e-7, rtol=1e-7):
            stall += 5
            if stall > 50:
                break
            a = _nf(a * np.exp(0.03 * rng.standard_normal(n)))
            loc = _sc(a)
            _upd(a, best)
            continue
        # Bisection line search
        lo, hi = 0.0, 1.0
        best_t, best_s = 0.0, loc
        for _ in range(18):
            mid = 0.5 * (lo + hi)
            cand = _nf((1.0 - mid) * a + mid * gn)
            s = _sc(cand)
            if math.isfinite(s) and s < loc:
                if s < best_s:
                    best_t, best_s = mid, s
                lo = mid
            else:
                hi = mid
        if best_t > 0 and best_s < loc * (1.0 - 1e-11):
            a = _nf((1.0 - best_t) * a + best_t * gn)
            loc = best_s
            stall = 0
            _upd(a, best)
        else:
            stall += 1
            if stall % 10 == 0:
                improved = False
                for _try in range(5):
                    scale = 0.02 + 0.01 * min(stall / 10.0, 3.0)
                    pa = _nf(a * np.exp(scale * rng.standard_normal(n)))
                    ps = _sc(pa)
                    if math.isfinite(ps) and ps < loc * (1.0 - 1e-12):
                        a, loc, stall = pa, ps, 0
                        _upd(a, best)
                        improved = True
                        break
                if not improved and stall > 120:
                    break
    return a

def _smooth_obj_grad(v, beta):
    """Smooth max objective and gradient for scale-invariant score."""
    n = v.size
    s = float(v.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    x = np.convolve(v, v)
    m = float(x.max())
    sh = x - m
    W = np.exp(beta * sh)
    wsum = float(W.sum())
    if wsum < 1e-300:
        return 1e10, np.zeros(n)
    w = W / wsum
    smax = m + np.log(wsum) / beta
    score = 2.0 * n * smax / (s * s)
    # Gradient
    grad = np.zeros(n)
    for i in range(n):
        grad[i] = 2.0 * n * float(np.dot(w[i:i+n], v)) / (s * s) - 4.0 * n * smax / (s * s * s)
    return score, grad

    if _sp_min is None:

    a = _nf(f0.copy())
    n = a.size
