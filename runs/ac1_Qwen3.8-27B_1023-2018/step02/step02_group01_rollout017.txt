        sorted_raw = np.sort(raw)[::-1]
        cumsum = np.cumsum(sorted_raw)
        rho = np.where(sorted_raw > (cumsum - 1.0) / np.arange(1, n + 1))[0]
        if len(rho) > 0:
            r = rho[-1]
            tau = (cumsum[r] - 1.0) / (r + 1)
        else:
            tau = raw.max() + 0.1
        a_new = np.maximum(raw - tau, 0.0)
        a_new /= a_new.sum()

        a = a_new
        if it % 10 == 0:
            pct *= 0.97

    return best_a

def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = float(budget_s)
    deadline = t0 + max(0.0, budget - 12.0)

    parent = _get_parent()
    n = parent.size

    # Precompute indices
    safe_j, mask, m = _precompute_indices(n)

    best_seq = parent.copy()
    try:
        best_sc = float(evaluate_sequence([float(x) for x in parent]))
    except Exception:
        best_sc = float('inf')
    if not np.isfinite(best_sc):
        best_seq = np.ones(32, dtype=np.float64) / 32.0
        best_sc = 2.0

    a0 = np.clip(parent, 0.0, 1000.0).astype(np.float64)
    s0 = a0.sum()
    if s0 < 1e-12:
        a0 = np.ones(n) / n
    else:
        a0 = a0 / s0

    rng = np.random.default_rng(seed)

    # Phase 1: Main LP iteration from parent (45% of budget)
    phase1_end = t0 + budget * 0.45
    best_a1, best_M1 = _lp_iteration(a0, phase1_end, safe_j, mask, m, rng)
    sc1 = _score(best_a1)
    if np.isfinite(sc1) and sc1 < best_sc:
        best_sc = sc1
        best_seq = best_a1.copy()

    # Phase 2: Alternative initializations (25% of budget)
    if time.monotonic() < deadline - 60:
        inits = []
        # Linear ramps
        ramp = np.arange(1, n + 1, dtype=np.float64)
        inits.append(ramp / ramp.sum())
        inits.append(ramp[::-1] / ramp[::-1].sum())
        # Exponential decays
        for al in (0.5, 1.0, 1.5, 2.0):
            e = np.exp(-al * np.arange(n, dtype=np.float64) / n)
            inits.append(e / e.sum())
        # Sine / cosine
        x = np.linspace(0, 1, n)
        s1 = np.maximum(0.3 + 0.7 * np.sin(np.pi * x), 0.01)
        inits.append(s1 / s1.sum())
        s2 = np.maximum(0.3 + 0.7 * np.sin(np.pi * 2 * x), 0.01)
        inits.append(s2 / s2.sum())
        # Noisy parent
        for ns in (0.1, 0.25, 0.5):
            nz = a0 * np.exp(rng.normal(0, ns, n))
            nz = np.maximum(nz, 1e-10)
            inits.append(nz / nz.sum())
        # Two-block
        split = n // 2
        tb = np.ones(n)
        tb[:split] = 2.0
        tb[split:] = 0.5
        inits.append(tb / tb.sum())
        tb2 = np.ones(n)
        tb2[:split] = 0.5
        tb2[split:] = 2.0
        inits.append(tb2 / tb2.sum())

        phase2_per = max(5.0, min(60.0, (deadline - 60 - time.monotonic()) / len(inits)))
        for ini in inits:
            if time.monotonic() > deadline - 50:
                break
            p2_end = min(time.monotonic() + phase2_per, deadline - 50)
            ba, bM = _lp_iteration(ini, p2_end, safe_j, mask, m, rng)
            sc = _score(ba)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_seq = ba.copy()

    # Phase 3: Subgradient polish on best (15% of budget)
    if time.monotonic() < deadline - 20:
        p3_end = min(deadline - 15, time.monotonic() + budget * 0.1)
        pol_a = _subgradient_polish(best_seq, p3_end, rng)
        sc = _score(pol_a)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_seq = pol_a.copy()

    # Phase 4: Final LP polish (remaining time)
    if time.monotonic() < deadline - 5:
        ba4, bM4 = _lp_iteration(best_seq, deadline, safe_j, mask, m, rng)
        sc = _score(ba4)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_seq = ba4.copy()

    # Final validation
    try:
        fs = float(evaluate_sequence([float(x) for x in best_seq]))
        if not np.isfinite(fs) or len(best_seq) == 0:
            return [1.0] * 32
    except Exception:
        return [1.0] * 32

    return [float(x) for x in best_seq]
