import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog, minimize


class _Timeout(Exception):
    pass


_fft_len_cache = {}


def _fft_len(m: int) -> int:
    m = int(m)
    if m <= 0:
        return 1
    v = _fft_len_cache.get(m)
    if v is None:
        v = int(next_fast_len(m))
        _fft_len_cache[m] = v
    return v


def _conv_same(a: np.ndarray) -> np.ndarray:
    n = int(a.size)
    if n == 0:
        return np.array([], dtype=np.float64)
    m = 2 * n - 1
    L = _fft_len(m)
    fa = rfft(a, L)
    c = irfft(fa * fa, L)[:m]
    c = np.maximum(c, 0.0)
    return c


def _conv2(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    n1 = int(a.size)
    n2 = int(b.size)
    if n1 == 0 or n2 == 0:
        return np.array([], dtype=np.float64)
    m = n1 + n2 - 1
    L = _fft_len(m)
    c = irfft(rfft(a, L) * rfft(b, L), L)[:m]
    c = np.maximum(c, 0.0)
    return c


def _valid_conv(d: np.ndarray, a: np.ndarray) -> np.ndarray:
    # s_j = sum_{r=0}^{n-1} d[j+r] a[r], j=0..n-1
    n = int(a.size)
    if n == 0:
        return np.array([], dtype=np.float64)
    ar = a[::-1].copy()
    c = _conv2(d, ar)
    return c[n - 1:2 * n - 1]


def _normalize(a, fallback_len: int = None) -> np.ndarray:
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        n = int(fallback_len) if fallback_len is not None else 32
        return np.ones(n, dtype=np.float64) / float(n)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not math.isfinite(s) or s < 1e-12:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def _score_internal(a) -> float:
    try:
        a = np.asarray(a, dtype=np.float64).ravel()
        if a.size == 0:
            return math.inf
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        s = float(a.sum())
        if not math.isfinite(s) or s < 1e-12:
            return math.inf
        a = a / s
        n = int(a.size)
        c = _conv_same(a)
        m = float(c.max())
        if not math.isfinite(m) or m < 0.0:
            return math.inf
        return 2.0 * float(n) * m
    except Exception:
        return math.inf


def _exact_score(a) -> float:

        fn = globals().get("evaluate_sequence", None)
        if fn is None:
            return math.inf
        lst = [float(x) for x in np.asarray(a, dtype=np.float64).ravel()]

    except Exception:

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
    except Exception:
        return None
    if h is None:
        return None
    try:

def _interp_parent(parent: np.ndarray, L: int) -> np.ndarray:
    L = int(L)
    if L <= 0:
        return np.array([], dtype=np.float64)
    if parent is None:
        return np.ones(L, dtype=np.float64) / float(L)
    parent = np.asarray(parent, dtype=np.float64).ravel()
    if parent.size == 0:
        return np.ones(L, dtype=np.float64) / float(L)

    x0 = np.linspace(0.0, 1.0, parent.size)
    x1 = np.linspace(0.0, 1.0, L)
    v = np.interp(x1, x0, parent)
    return _normalize(v)


def _softmax_baseline(y: np.ndarray, b: int, n: int) -> np.ndarray:
    y = np.asarray(y, dtype=np.float64).ravel()
    y = np.nan_to_num(y, nan=-30.0, posinf=30.0, neginf=-30.0)
    y = np.clip(y, -30.0, 30.0)
    ex = np.exp(y)
    Z = 1.0 + float(ex.sum())
    if not math.isfinite(Z) or Z <= 0.0:
        return np.ones(n, dtype=np.float64) / float(n)
    a = np.empty(n, dtype=np.float64)
    a[:b] = ex[:b] / Z
    a[b] = 1.0 / Z
    if b + 1 < n:
        a[b + 1:] = ex[b:] / Z
    s = float(a.sum())
    if s <= 0.0 or not math.isfinite(s):
        return np.ones(n, dtype=np.float64) / float(n)
    return a / s


def _best_t_quadratic(cf: np.ndarray, cfs: np.ndarray, cg: np.ndarray):
    base = cf.astype(np.float64, copy=False)
    bcoef = 2.0 * (cfs - cf)
    gcoef = cf - 2.0 * cfs + cg

    ts = np.linspace(0.0, 1.0, 33)
    vals = np.max(base + bcoef * ts[:, None] + gcoef * (ts[:, None] ** 2), axis=1)
    i = int(np.argmin(vals))
    lo = ts[max(0, i - 2)]
    hi = ts[min(len(ts) - 1, i + 2)]
    if hi <= lo:
        hi = min(1.0, lo + 0.1)
        lo = max(0.0, hi - 0.1)

    ts2 = np.linspace(lo, hi, 121)
    vals2 = np.max(base + bcoef * ts2[:, None] + gcoef * (ts2[:, None] ** 2), axis=1)
    j = int(np.argmin(vals2))
    return float(ts2[j]), float(vals2[j])


def _lbfgs_refine(arr, eta_list, stop: float, maxfun_per: int = 800):
    f = _normalize(arr)
    n = int(f.size)
    best = f.copy()
    best_sc = _score_internal(f)
    if n < 2:
        return best, best_sc

    for eta in eta_list:
        if time.monotonic() > stop - 1.0:
            break
        c = _conv_same(f)
        M = float(c.max())
        if not math.isfinite(M) or M <= 1e-15:
            break
        beta = float(eta) / M

        b = int(np.argmax(f))
        idx = np.array([i for i in range(n) if i != b], dtype=int)
        base = max(float(f[b]), 1e-300)
        y0 = np.log(np.maximum(f[idx], 1e-300) / base)
        y0 = np.clip(np.nan_to_num(y0, nan=-30.0, posinf=30.0, neginf=-30.0), -30.0, 30.0)

        def fg(y, _beta=beta, _b=b, _idx=idx, _n=n):
            if time.monotonic() > stop:
                raise _Timeout
            y = np.asarray(y, dtype=np.float64)
            if not np.all(np.isfinite(y)):
                return 1e100, np.zeros_like(y)
            a = _softmax_baseline(y, _b, _n)
            c = _conv_same(a)
            maxc = float(c.max())
            if not math.isfinite(maxc) or maxc <= 1e-300:
                return 1e100, np.zeros_like(y)
            x = _beta * (c - maxc)
            x = np.clip(x, -30.0, 0.0)
            w = np.exp(x)
            s = float(w.sum())
            if s <= 0.0 or not math.isfinite(s):
                w = np.ones_like(c) / float(c.size)
                s = float(c.size)
            fun = _beta * maxc + math.log(s)
            d = w / s
            g = (2.0 * _beta) * _valid_conv(d, a)
            if not np.all(np.isfinite(g)):
                g = np.zeros(_n, dtype=np.float64)
            dot = float(np.dot(a, g))
            rg = a[_idx] * (g[_idx] - dot)
            rg = np.where(np.isfinite(rg), rg, 0.0)
            return float(fun), rg

        try:
            res = minimize(
                fg,
                y0,
                jac=True,
                method="L-BFGS-B",
                options={"maxiter": 220, "maxfun": int(maxfun_per), "ftol": 1e-12, "gtol": 1e-9, "maxls": 50},
            )
        except _Timeout:
            break
        except Exception:
            break

        if not hasattr(res, "x") or res.x is None:
            break
        a_new = _softmax_baseline(np.asarray(res.x, dtype=np.float64), b, n)
        sc = _score_internal(a_new)
        if math.isfinite(sc) and sc < best_sc:
            best = a_new.copy()
            best_sc = sc
            f = best
        elif (
            getattr(res, "success", False)
            and math.isfinite(sc)
            and sc < best_sc + 1e-8
        ):
            f = a_new.copy()

    return best, best_sc


def _lp_update(f0: np.ndarray, n: int, stop: float, seed: int = 0):

            else:

            sc_f = _score_internal(f)
            if math.isfinite(sc_f) and sc_f < best_sc:

                best_sc = sc_f

def _make_inits(L: int, parent, rng):
    inits = []
    if parent is not None:
        inits.append(_interp_parent(parent, L))
    inits.append(np.ones(L, dtype=np.float64) / float(L))

    inits.append(_normalize(rng.exponential(1.0, size=L)))

    x = np.ones(L, dtype=np.float64)
    i = np.arange(L, dtype=np.float64)
    for m in range(1, 6):
        amp = rng.uniform(0.25, 1.0) / (m + 1.0)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        x += amp * np.cos((2.0 * np.pi * m * i / L) + ph)
    inits.append(_normalize(np.abs(x) + 0.05))

    inits.append(_normalize(np.abs(rng.normal(0.0, 1.0, size=L)) + 0.05))

    v = np.arange(1, L + 1, dtype=np.float64)
    for p in (0.25, 0.5, 1.0):
        inits.append(_normalize(np.power(v, p)))
        inits.append(_normalize(np.power(L - v + 1.0, p)))
    inits.append(_normalize(v))
    inits.append(_normalize(np.zeros(L, dtype=np.float64)))

    keep = []
    seen = set()
    for a in inits:
        a = _normalize(a)
        key = tuple(np.round(a, 10).tolist())
        if key in seen or a.size != L:
            continue
        seen.add(key)
        keep.append(a)
    return keep[:10]

            base = orig if orig > 1e-15 else default_scale
            best_val = orig
            best_val_sc = best_sc

            vals = {orig}
            for m in mults:
                val = base * float(m)
                if 0.0 <= val <= 1000.0:
                    vals.add(val)
            if orig <= default_scale:
                for m in (1.0, 2.0, 4.0, 8.0):
                    val = default_scale * float(m)
                    if 0.0 <= val <= 1000.0:
                        vals.add(val)

            for val in vals:
                best[idx] = val
                sc = _score_internal(best)
                if math.isfinite(sc) and sc < best_val_sc:
                    best_val_sc = sc
                    best_val = val

            best[idx] = best_val
            if best_val_sc < best_sc:
                best_sc = best_val_sc
                improved = True

        if not improved:
            break

    return best, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = max(10.0, float(budget_s))
    except Exception:
        budget = 1300.0
    hard_deadline = t0 + budget
    if budget > 180.0:
        deadline = hard_deadline - 30.0
    elif budget > 60.0:
        deadline = hard_deadline - 12.0
    else:
        deadline = hard_deadline - 2.5

    parent = _get_parent()
    if parent is None:
        parent = _normalize(np.ones(320, dtype=np.float64))
    else:
        parent = _normalize(parent)

    best_seq = parent.copy()
    fallback = parent.copy()

    ps = _exact_score(best_seq)
    if not math.isfinite(ps):
        ps = _score_internal(best_seq)
    if not math.isfinite(ps):
        best_seq = _normalize(np.ones(320, dtype=np.float64))
        fallback = best_seq.copy()
        ps = _exact_score(best_seq)
        if not math.isfinite(ps):
            ps = _score_internal(best_seq)
    best_score = ps

    rng = np.random.default_rng(seed)

    def _try_candidate(arr):
        nonlocal best_seq, best_score
        if arr is None:
            return
        a = _normalize(arr)
        if a.size == 0:
            return
        sc_int = _score_internal(a)
        if not math.isfinite(sc_int):
            return
        if sc_int >= best_score - 1e-10:
            return
        sc = _exact_score(a)
        if not math.isfinite(sc):
            sc = sc_int
        if math.isfinite(sc) and sc < best_score - 1e-12:
            best_score = sc
            best_seq = a.copy()

    def _finalize():
        nonlocal best_seq, best_score
        seq = np.clip(np.asarray(best_seq, dtype=np.float64).ravel(), 0.0, 1000.0)
        if seq.size == 0 or float(seq.sum()) < 1e-8:
            seq = np.ones(32, dtype=np.float64) / 32.0
        lst = [float(x) for x in seq]
        sc = _exact_score(lst)
        if not math.isfinite(sc):
            fb = np.clip(np.asarray(fallback, dtype=np.float64).ravel(), 0.0, 1000.0)
            if fb.size == 0 or float(fb.sum()) < 1e-8:
                fb = np.ones(32, dtype=np.float64) / 32.0
            fb_lst = [float(x) for x in fb]
            sc2 = _exact_score(fb_lst)
            if math.isfinite(sc2):
                return fb_lst
        if not math.isfinite(sc):
            # If the evaluator is unavailable somehow, return the internally valid sequence.
            return lst
        return lst

    n_base = int(parent.size)

    # Very small budget: return quickly after a tiny local polish.
    if budget < 80.0:
        if time.monotonic() < deadline - 2.0:
            arr, _ = _coordinate_polish(best_seq, deadline - 2.0, seed=123)
            _try_candidate(arr)
        return _finalize()

    # Phase 1: quick local refinement of a few initializations at the base length.
    phase1_stop = min(deadline - 65.0, t0 + 0.22 * budget)
    inits = _make_inits(n_base, parent, rng)
    scored = []
    for a in inits:
        if time.monotonic() > phase1_stop - 4.0:
            break
        sc = _score_internal(a)
        if math.isfinite(sc):
            scored.append((sc, a))
    scored.sort(key=lambda z: z[0])

    for _, a in scored[:4]:
        if time.monotonic() > phase1_stop - 4.0:
            break
        if best_score < 1.45:
            break
        if _score_internal(a) > best_score + 0.35:
            continue
        eta_stop = min(phase1_stop - 3.0, time.monotonic() + 55.0)
        if eta_stop - time.monotonic() > 10.0:
            arr, _ = _lbfgs_refine(a, (12.0, 45.0), eta_stop, maxfun_per=550)
            _try_candidate(arr)
        if time.monotonic() < phase1_stop - 18.0 and n_base <= 900:
            lp_stop = min(phase1_stop - 3.0, time.monotonic() + 45.0)
            arr, _ = _lp_update(arr, n_base, lp_stop, seed=int(rng.integers(1, 10_000_000)))
            _try_candidate(arr)

    # Phase 2: main search at base length.
    core_stop = min(deadline - 90.0, t0 + 0.58 * budget)
    arr = best_seq.copy()

    s1_stop = min(core_stop - 130.0, time.monotonic() + 110.0)
    if s1_stop - time.monotonic() > 15.0:
        arr, _ = _lbfgs_refine(arr, (60.0, 180.0), s1_stop, maxfun_per=850)
        _try_candidate(arr)

    lp_stop = min(core_stop - 90.0, time.monotonic() + max(60.0, core_stop - time.monotonic() * 0.6))
    if lp_stop - time.monotonic() > 20.0 and n_base <= 900:
        arr, _ = _lp_update(arr, n_base, lp_stop, seed=int(rng.integers(1, 10_000_000)))
        _try_candidate(arr)

    s2_stop = core_stop - 8.0
    if s2_stop - time.monotonic() > 15.0:
        arr, _ = _lbfgs_refine(arr, (300.0, 700.0), s2_stop, maxfun_per=750)
        _try_candidate(arr)

    # Phase 3: other lengths and a randomized restart at the current best length.
    other_stop = min(deadline - 35.0, t0 + 0.90 * budget)
    cur_n = int(best_seq.size)

    # Random restart at current best length if useful.
    rr_stop = min(other_stop - 50.0, time.monotonic() + 90.0)
    if rr_stop - time.monotonic() > 25.0:
        rnd_init = _normalize(rng.exponential(1.0, size=cur_n))
        rnd_init = _lbfgs_refine(rnd_init, (35.0, 120.0), rr_stop, maxfun_per=500)[0]
        _try_candidate(rnd_init)
        if (_score_internal(rnd_init) < best_score + 0.12) and time.monotonic() < rr_stop - 15.0:
            rnd_init, _ = _lp_update(rnd_init, cur_n, rr_stop, seed=int(rng.integers(1, 10_000_000)))
            _try_candidate(rnd_init)

    cur_n = int(best_seq.size)
    other_lengths = [512, 384, 768, 256, 1024]
    other_lengths = [L for L in other_lengths if L != cur_n]
    other_lengths = sorted(other_lengths)[:4]

    if other_lengths:
        per_alloc = max(35.0, (other_stop - max(time.monotonic(), t0)) / float(len(other_lengths)))
    for L in other_lengths:
        if time.monotonic() > other_stop - 8.0:
            break
        if best_score < 1.45:
            break
        stopL = min(other_stop - 5.0, time.monotonic() + max(30.0, per_alloc))
        arr = _interp_parent(best_seq, L)
        if L <= 512:
            etas = (45.0, 140.0)
            maxfun = 650
        elif L <= 768:
            etas = (45.0, 160.0)
            maxfun = 600
        else:
            etas = (45.0, 140.0)
            maxfun = 520

        sub_stop = min(stopL - 8.0, time.monotonic() + max(25.0, 0.55 * max(1.0, stopL - time.monotonic())))
        if sub_stop - time.monotonic() > 12.0:
            arr, _ = _lbfgs_refine(arr, etas, sub_stop, maxfun_per=maxfun)
            _try_candidate(arr)

        if L <= 768 and time.monotonic() < stopL - 18.0:
            lpL = min(stopL - 3.0, time.monotonic() + max(15.0, 0.35 * max(1.0, stopL - time.monotonic())))
            arr, _ = _lp_update(arr, L, lpL, seed=int(rng.integers(1, 10_000_000)))
            _try_candidate(arr)

        if time.monotonic() < stopL - 6.0:
            arr, _ = _lbfgs_refine(arr, (300.0,), stopL - 2.0, maxfun_per=350)
            _try_candidate(arr)

    # Final polish.
    fin_stop = deadline - 8.0
    if time.monotonic() < fin_stop - 18.0:
        cur_n = int(best_seq.size)
        if cur_n <= 1100:
            arr, _ = _lbfgs_refine(best_seq, (500.0, 900.0), min(fin_stop - 12.0, time.monotonic() + 80.0), maxfun_per=600)
            _try_candidate(arr)

    if time.monotonic() < fin_stop - 3.0:
        cur_n = int(best_seq.size)
        if cur_n <= 1100:
            arr, _ = _coordinate_polish(best_seq, fin_stop - 2.0, seed=987654321)
            _try_candidate(arr)

    # One last small safeguard: if we are extremely close to the timer, return immediately.
    if time.monotonic() >= hard_deadline - 12.0:
        return _finalize()

    return _finalize()
