```python
def _fft_conv_self(a):
    """Compute conv(a,a) using FFT."""
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2*n-1]
    np.maximum(c, 0.0, out=c)
    return c

def _build_A(a):
    """Build the constraint matrix for the LP.
    A[k,j] = a[k-j] if 0<=k-j<n else 0
    """
    n = len(a)
    # A has shape (2n-1, n)
    # A[k,j] = a[k-j] if 0 <= k-j <= n-1
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        lo = max(0, k - (n-1))
        hi = min(n-1, k)
        for j in range(lo, hi+1):
            A[k, j] = a[k-j]
    return A

def _solve_lp(a, M, n, solver_list=None):
    """Solve the LP: max sum(g) s.t. A@g <= M, g >= 0"""
    if solver_list is None:
        solver_list = [cp.CBC, cp.GLPK, cp.GLOP, cp.PDLp]
    
    A = _build_A(a)
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    
    for solver in solver_list:
        try:
            prob.solve(solver=solver, verbose=False)
            gv = g.value
            if prob.status in ['optimal', 'optimal_inaccurate'] and gv is not None:
                gv = np.asarray(gv, dtype=np.float64).ravel()
                if np.isfinite(gv).all():
                    gv = np.maximum(gv, 0)
                    s = gv.sum()
                    if s > 1e-15:
                        return gv / s, gv
        except Exception:
            continue
    return None, None
```

```python
                if vv is not None and tv is not None:
                    vv = np.asarray(vv, dtype=np.float64).ravel()
                    tv = float(tv)
                    if np.isfinite(vv).all() and abs(vv.sum() - 1) < 1e-6:
                        vv = vv / vv.sum()  # renormalize
                        return vv, tv
        except Exception:
            continue
    return None, None
```

This is better because:
1. It directly gives the minimum achievable peak (t*)
2. The direction v is automatically normalized (sum = 1)
3. We can compare t* with current M to decide if improvement is possible

The line search then becomes: find the best t in [0,1] such that max conv((1-t)*a + t*v, (1-t)*a + t*v) is minimized.

```python
def _optimize_lp_v2(n, a_init, deadline, rng):
    a = _normalize(np.maximum(a_init.copy(), 0))
    best_a = a.copy()
    b = _fft_conv_self(a)
    best_mb = float(b.max())
    
    solvers = [cp.CBC, cp.GLPK, cp.GLOP, cp.PDLp]
    no_imp = 0
    max_no_imp = 20
    it = 0
    
    while time.monotonic() < deadline:
        it += 1
        b = _fft_conv_self(a)
        M = float(b.max())
        
        if M < best_mb * (1 - 1e-10):
            best_mb = M
            best_a = a.copy()
            no_imp = 0
        
        v, t_star = _solve_lp_min_peak(a, n, solvers)
        
        if v is None:
            # Fallback: subgradient
            k_star = int(np.argmax(b))
            grad = _subgrad_direction(a, k_star)
            gn = np.linalg.norm(grad)
            if gn < 1e-15:
                break
            step = 1.0 / (it + 10)
            a_new = _normalize(np.maximum(a - step * grad / gn, 0))
            mb_new = float(_fft_conv_self(a_new).max())
            if mb_new < M:
                a = a_new
            else:
                no_imp += 1
        else:
            # t_star is the best linearized peak
            # If t_star >= M, we're at a local optimum (for this linearization)
            if t_star >= M * (1 - 1e-6):
                no_imp += 1
                if no_imp > 5:
                    a = _normalize(a + rng.normal(0, 0.05, n))
                    no_imp = 0
            else:
                # Line search along direction v
                best_t = 0.0
                best_new = M
                for ts in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                    a_new = (1-ts)*a + ts*v
                    a_new = _normalize(np.maximum(a_new, 0))
                    mb_new = float(_fft_conv_self(a_new).max())
                    if mb_new < best_new:
                        best_new = mb_new
                        best_t = ts
                if best_t > 0 and best_new < M * (1 - 1e-10):
                    a = (1-best_t)*a + best_t*v
                    a = _normalize(np.maximum(a, 0))
                else:
                    no_imp += 1
        
        if no_imp > max_no_imp:
            a = _normalize(a + rng.normal(0, 0.1, n))
            no_imp = 0
            max_no_imp = max(15, int(max_no_imp * 1.3))
    
    return best_a, 2.0 * n * best_mb
```

This is cleaner. The key insight: t_star < M means there exists a normalized v with conv(a, v)[k] <= t_star < M for all k. Moving toward v should reduce the peak.

Now, one concern: the LP `min t s.t. A@v <= t, sum(v)=1, v>=0` might have t_star very close to M (within numerical precision), making it hard to tell if we're truly stuck. In that case, we perturb.

Let me also think about the compute budget more carefully:
- Building A: O(n^2) = O(37000) - fast, maybe 0.5ms
- Solving LP with CBC: maybe 5-50ms for n=192 (depends on the solver)
- FFT: negligible (~0.1ms)
- Line search (7 evaluations): ~0.7ms

Total per iteration: ~10-60ms. In 500 seconds: ~8,000-50,000 iterations.

This should be more than enough to converge.

One more thought: I should also consider whether the "min t" LP is the right one, or if I should use "max sum(v) s.t. A@v <= M*(1-eps)". The former directly minimizes the peak, while the latter maximizes the "volume" under the peak constraint. They should give similar directions but the former is more direct.

Actually, there's a subtle difference. The "min t" LP finds the v that minimizes the worst-case conv(a,v) value. This is like finding the "best" direction to move. The "max sum" LP with a fixed M finds the most "spread out" v that still fits under the M constraint. Both are valid, but I think "min t" is more natural for our purpose.

Let me go with "min t" and add the alternative as a fallback.

```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
import cvxpy as cp

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        if p.size == 0 or not np.isfinite(p).all():
            p = np.ones(192, dtype=np.float64)
        return p
    except Exception:
        return np.ones(192, dtype=np.float64)

def _fft_conv_self(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _build_A(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        if lo <= hi:
            A[k, lo:hi + 1] = a[k - np.arange(lo, hi + 1)]
    return A

def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def _subgrad_direction(a, k_star):
    n = len(a)
    grad = np.zeros(n)
    for m in range(n):
        j = k_star - m
        if 0 <= j < n:
            grad[m] = a[j]
    return 2.0 * grad

def _solve_lp_min_peak(a, n, solvers=None):
    """Solve: min t s.t. A@v <= t, sum(v)=1, v>=0."""
    if solvers is None:
        solvers = [cp.CBC, cp.GLPK, cp.GLOP, cp.PDLp]
    A = _build_A(a)
    v = cp.Variable(n, nonneg=True)
    t = cp.Variable()
    constraints = [A @ v <= t, cp.sum(v) == 1]
    prob = cp.Problem(cp.Minimize(t), constraints)
    for s in solvers:
        try:
            prob.solve(solver=s, verbose=False)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                vv = v.value
                if vv is not None:
                    vv = np.asarray(vv, dtype=np.float64).ravel()
                    if np.isfinite(vv).all():
                        vv = np.maximum(vv, 0)
                        s_val = vv.sum()
                        if s_val > 1e-12:
                            vv /= s_val
                            return vv, float(prob.value)
        except Exception:
            continue
    return None, None

def _solve_lp_max_vol(a, M, n, solvers=None):
    """Solve: max sum(v) s.t. A@v <= M, v>=0. Returns normalized v."""
    if solvers is None:
        solvers = [cp.CBC, cp.GLPK, cp.GLOP, cp.PDLp]
    A = _build_A(a)
    v = cp.Variable(n, nonneg=True)
    prob = cp.Problem(cp.Maximize(cp.sum(v)), [A @ v <= M])
    for s in solvers:
        try:
            prob.solve(solver=s, verbose=False)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                vv = v.value
                if vv is not None:
                    vv = np.asarray(vv, dtype=np.float64).ravel()
                    if np.isfinite(vv).all():
                        vv = np.maximum(vv, 0)
                        s_val = vv.sum()
                        if s_val > 1e-12:
                            return vv / s_val
        except Exception:
            continue
    return None

def _optimize_lp(n, a_init, deadline, rng):
    """Core LP-based optimizer."""
    a = _normalize(np.maximum(a_init.copy(), 0))
    best_a = a.copy()
    b = _fft_conv_self(a)
    best_mb = float(b.max())
    
    solvers = [cp.CBC, cp.GLPK, cp.GLOP, cp.PDLp]
    no_imp = 0
    max_no_imp = 20
    it = 0
    
    while time.monotonic() < deadline:
        it += 1
        b = _fft_conv_self(a)
        M = float(b.max())
        
        if M < best_mb * (1 - 1e-10):
            best_mb = M
            best_a = a.copy()
            no_imp = 0
        
        # Primary: LP min-peak
        v, t_star = _solve_lp_min_peak(a, n, solvers)
        
        if v is None:
            # Secondary: LP max-volume with tightened M
            v = _solve_lp_max_vol(a, M * 0.999, n, solvers)
        
        if v is None:
            # Fallback: subgradient
            k_star = int(np.argmax(b))
            grad = _subgrad_direction(a, k_star)
            gn = float(np.linalg.norm(grad))
            if gn < 1e-15:
                break
            step = 2.0 / (float(np.linalg.norm(a))**2 * (min(it, 200) + 10))
            a_new = _normalize(np.maximum(a - step * grad, 0))
            mb_new = float(_fft_conv_self(a_new).max())
            if mb_new < M:
                a = a_new
            else:
                no_imp += 1
        else:
            cos_sim = float(np.dot(a, v) / (np.linalg.norm(a) * np.linalg.norm(v) + 1e-30))
            if cos_sim > 0.9999:
                no_imp += 1
                a = _normalize(a + rng.normal(0, 0.04, n))
            else:
                # Check if LP says improvement is possible
                if t_star is not None and t_star >= M * (1 - 1e-5):
                    no_imp += 1
                    if no_imp % 5 == 0:
                        a = _normalize(a + rng.normal(0, 0.06, n))
                else:
                    # Line search
                    best_t = 0.0
                    best_new = M
                    for ts in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                        a_new = _normalize((1 - ts) * a + ts * v)
                        mb_new = float(_fft_conv_self(a_new).max())
                        if mb_new < best_new:
                            best_new = mb_new
                            best_t = ts
                    if best_t > 0 and best_new < M * (1 - 1e-10):
                        a = _normalize((1 - best_t) * a + best_t * v)
                    else:
                        no_imp += 1
        
        if no_imp > max_no_imp:
            a = _normalize(a + rng.normal(0, 0.1, n))
            no_imp = 0
            max_no_imp = max(15, int(max_no_imp * 1.25))
    
    return best_a, 2.0 * n * best_mb

def _local_search(a, deadline, rng):
    """Simple local search for final polishing."""
    n = len(a)
    a = _normalize(np.maximum(a.copy(), 0))
    best_a = a.copy()
    best_mb = float(_fft_conv_self(a).max())
    sigma = 0.015
    no_imp = 0
    it = 0
    
    while time.monotonic() < deadline:
        it += 1
        r = rng.random()
        if r < 0.4:
            cand = a + rng.normal(0, sigma, n)
        elif r < 0.6:
            idx = rng.integers(0, n, size=3)
            cand = a.copy()
            cand[idx] *= rng.uniform(0.6, 1.5, 3)
        elif r < 0.8:
            lo = rng.integers(0, max(1, n // 4))
            w = rng.integers(2, max(3, n // 8))
            hi = min(n, lo + w)
            cand = a.copy()
            cand[lo:hi] *= rng.uniform(0.5, 1.8)
        else:
            cand = 0.9 * a + 0.1 * np.ones(n) / n
            cand += rng.normal(0, sigma * 0.5, n)
        
        cand = _normalize(np.maximum(cand, 0))
        mb = float(_fft_conv_self(cand).max())
        sc = 2.0 * n * mb
        
        if sc < best_mb * (1 - 1e-10):
            best_mb = sc
            best_a = cand.copy()
            a = cand.copy()
            no_imp = 0
            sigma = max(0.003, sigma * 0.95)
        else:
            no_imp += 1
            if no_imp > 150:
                sigma = min(0.2, sigma * 1.6)
                no_imp = 0
    
    return best_a, 2.0 * n * best_mb
```
