from scipy.fft import rfft, irfft, next_fast_len

    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

    n = a.size
    b = _convolve_square(a)
    mb = float(b.max())
    return 2.0 * n * mb / (s * s)


def _build_A(a):

    m = 2 * n - 1
    k_idx = np.arange(m, dtype=np.int64)[:, None]
    j_idx = np.arange(n, dtype=np.int64)[None, :]
    diff = k_idx - j_idx
    mask = (diff >= 0) & (diff < n)
    safe_diff = np.clip(diff, 0, n - 1)
    A = np.where(mask, a[safe_diff], 0.0)
    return A


def _lp_best_response(a, M, timeout_s=3.0):
    n = a.size
    A = _build_A(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(
            c_obj, A_ub=A, b_ub=b_ub,
            bounds=[(0.0, None)] * n,
            method='highs',
            options={'time_limit': timeout_s}
        )
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    if not np.isfinite(g).all() or g.sum() <= 0:
        return None
    return g


def _lp_best_response_with_perturbation(a, M, timeout_s=3.0, perturb=0.0):

    n = a.size
    A = _build_A(a)
    if perturb > 0:

def _run_lp_iterations(a0, stop_time, max_iters=5000, verbose=False):
    """Main LP best-response optimization loop."""
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    s = float(a.sum())
    if s <= 0:
        a = np.ones(a.size) / a.size
        s = 1.0
    a = a / s

    best_a = a.copy()
    best_M = float(_convolve_square(a).max())
    
    t_step = 0.5  # initial step size
    stuck = 0
    total_iters = 0
    
    for it in range(max_iters):
        if time.monotonic() > stop_time - 1.0:
            break
        
        b = _convolve_square(a)
        M = float(b.max())
        
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1
        
        if stuck > 200:
            break
        
        # Solve LP
        g = _lp_best_response(a, M, timeout_s=2.0)
        if g is None:
            # Try with perturbation
            g = _lp_best_response_with_perturbation(a, M, timeout_s=2.0, perturb=0.01)
            if g is None:
                break
        
        gs = float(g.sum())
        if gs <= 0:
            break
        g = g / gs
        
        # Check if g is very different from a
        diff = np.linalg.norm(g - a)
        if diff < 1e-10:
            # At fixed point
            if stuck > 50:
                break
            stuck += 10
            continue
        
        # Line search with backtracking
        improved = False
        for t in [t_step, t_step * 0.5, t_step * 0.25, t_step * 0.1, 0.05, 0.01]:
            cand = (1.0 - t) * a + t * g
            cand = np.clip(cand, 0.0, 1000.0)
            cs = float(cand.sum())
            if cs <= 0:
                continue
            cand = cand / cs
            cM = float(_convolve_square(cand).max())
            if cM < M * (1.0 - 1e-11):
                a = cand
                improved = True
                # If we used a full step, keep t_step; otherwise reduce
                if t == t_step:
                    pass  # keep current step
                elif t < t_step * 0.5:
                    t_step = max(0.05, t_step * 0.5)
                break
        
        if not improved:
            t_step = max(0.01, t_step * 0.5)
            if t_step <= 0.01 and stuck > 100:
                break
        
        total_iters += 1
    
    return best_a, best_M

def _make_initializations(parent, n, seed):
    """Create diverse initializations."""
    rng = np.random.default_rng(seed)
    inits = []
    
    # 1. Parent normalized
    p = np.clip(parent.astype(np.float64), 0.0, 1000.0)
    if p.size == n and p.sum() > 0:
        inits.append(p / p.sum())
    
    # 2. Uniform
    inits.append(np.ones(n) / n)
    
    # 3. Cosine (smooth, peaks in middle)
    x = np.linspace(0, np.pi, n)
    cos_seq = np.cos(x) ** 2 + 0.01
    inits.append(cos_seq / cos_seq.sum())
    
    # 4. Sine squared (peaks in middle)
    sin_seq = np.sin(x) ** 2 + 0.01
    inits.append(sin_seq / sin_seq.sum())
    
    # 5. Exponential concentration at center
    center = n / 2
    exps = np.exp(-0.5 * (np.arange(n) - center) ** 2 / (n * 0.2))
    inits.append(exps / exps.sum())
    
    # 6. Two-block (high-low or low-high)
    block = np.ones(n)
    block[:n//2] = 2.0
    inits.append(block / block.sum())
    
    block2 = np.ones(n)
    block2[n//2:] = 2.0
    inits.append(block2 / block2.sum())
    
    # 7. Alternating high/low
    alt = np.ones(n)
    alt[::2] = 1.5
    alt[1::2] = 0.5
    inits.append(alt / alt.sum())
    
    # 8. Random smooth
    r = rng.normal(0, 1, n)
    k = np.array([0.25, 0.5, 1.0, 0.5, 0.25])
    k = k / k.sum()
    smooth = np.convolve(r, k, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())
    
    # 9. Parent with small noise
    if p.size == n and p.sum() > 0:
        pn = p / p.sum()
        noisy = pn * np.exp(rng.normal(0, 0.2, n))
        noisy = np.maximum(noisy, 1e-8)
        inits.append(noisy / noisy.sum())

    x = np.linspace(-3, 3, n)
