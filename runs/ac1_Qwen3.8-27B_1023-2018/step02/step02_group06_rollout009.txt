    n = len(a)

    F = rfft(a, fsize)

class _ConvSolver:
    def __init__(self, n):
        self.n = n
        self.fsize = next_fast_len(2 * n - 1)
    
    def conv_self(self, a):
        F = rfft(a, self.fsize)
        c = irfft(F * F, self.fsize)[:2*self.n-1]
        np.maximum(c, 0.0, out=c)
        c[~np.isfinite(c)] = 0.0
        return c
    
    def conv_cross(self, a, b):
        Fa = rfft(a, self.fsize)
        Fb = rfft(b, self.fsize)
        c = irfft(Fa * Fb, self.fsize)[:2*self.n-1]
        c[~np.isfinite(c)] = 0.0
        return c
    
    def build_matrix(self, a):
        n = self.n
        A = np.zeros((2*n-1, n))
        for k in range(2*n-1):
            jlo = max(0, k - n + 1)
            jhi = min(n - 1, k)
            if jlo <= jhi:

        return A
