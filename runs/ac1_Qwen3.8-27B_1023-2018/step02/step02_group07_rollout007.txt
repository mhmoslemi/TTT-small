def _softmax(x):
    x = x - x.max()
    e = np.exp(np.clip(x, -50, 50))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(x) / x.size
    return e / s


def _fft_conv(a, n, fs):
    F = rfft(a, fs)
    return irfft(F * F, fs)[:2 * n - 1]


def _fft_xcorr(a, p, n, fs):
    Fa = rfft(a, fs)
    Fp = rfft(p, fs)
    return irfft(Fa * np.conj(Fp), fs)[:n]


def _exact_score(a):

    mb = float(np.max(b))
    sa = float(np.sum(a))
    if sa < 0.01:

    return 2.0 * n * mb / (sa * sa)

def _coord_descent(a, n, fs_c, end_time, rng=None):
    """Coordinate descent: adjust pairs of elements to reduce max b."""
    if rng is None:
        rng = np.random.default_rng(99)

        kmax = int(np.argmax(b))
        bmax = float(b[kmax])
        
        # Find the pair (i, kmax-i) that contributes most to b[kmax]
        lo = max(0, kmax - n + 1)
        hi = min(n, kmax + 1)
        best_pair = None
        best_contrib = -1
        for j in range(lo, hi):
            kk = kmax - j
            if 0 <= kk < n:
                contrib = 2.0 * a[j] * a[kk]
                if contrib > best_contrib:
                    best_contrib = contrib
                    best_pair = (j, kk)
        
        if best_pair is None:
            break
        i, k = best_pair
        
        # Try reducing the larger of a[i], a[k]
        if i == k:
            # Can't reduce a single element without changing sum
            # Skip

                best_sc = sc_p
                best_a = a_p.copy()

    explore_end = t0 + 0.65 * total
    polish_end = t0 + 0.90 * total
    final_deadline = t0 + total - 8.0
    
    # Run Adam on each variant
    n_variants = min(len(variants), 8)
    time_per_variant = max(5.0, (explore_end - t0) / n_variants)
    
    for vi in range(n_variants):
        if time.monotonic() >= explore_end - 2:
            break
        v_end = min(t0 + (vi + 1) * time_per_variant, explore_end - 2)
        if v_end - time.monotonic() < 3:
            break
        
        z0 = np.log(np.maximum(variants[vi], 1e-10))
        z0 = z0 - z0.mean()
        z0 = np.clip(z0, -20, 20)
        
        try:
            a_res, sc_res = _adam_run(
                z0, n_primary, fs_c, fs_x, v_end,
                tau0=0.3 + 0.1 * master_rng.random(),
                lr0=0.5 + 0.2 * master_rng.random(),
                seed=seed + vi * 101
            )
            try_update(a_res)
        except Exception:
            pass
    
    # Phase 1b: Try a couple other lengths
    other_lengths = [256, 320]
    if time.monotonic() < explore_end - 10:
        for nl in other_lengths:
            if time.monotonic() >= explore_end - 5:
                break
            fs_c_nl = next_fast_len(2 * nl - 1)
            fs_x_nl = next_fast_len(2 * nl - 1)
            z0 = (np.arange(nl) + 0.5) / nl
            z0p = np.log(np.maximum(1 - np.abs(2 * z0 - 1) + 0.1, 0.01))
            z0p = z0p - z0p.mean()
            end_nl = min(time.monotonic() + 30, explore_end - 2)
            try:
                a_res, sc_res = _adam_run(z0p, nl, fs_c_nl, fs_x_nl, end_nl,
                                          tau0=0.3, lr0=0.5, seed=seed + 9000 + nl)
                try_update(a_res)
            except Exception:
                pass
    
    # Phase 2: Polish the best found
    if time.monotonic() < polish_end - 5:
        # Re-run Adam from best
        a_polish = best_a.copy()
        if a_polish.size != n_primary:
            a_polish = a_polish / a_polish.sum()
            n_primary = a_polish.size
            fs_c = next_fast_len(2 * n_primary - 1)
            fs_x = next_fast_len(2 * n_primary - 1)
        
        z_best = np.log(np.maximum(a_polish, 1e-12))
        z_best = z_best - z_best.mean()
        
        adam_polish_end = min(time.monotonic() + 0.4 * (polish_end - t0), polish_end - 20)
        try:
            a_res, sc_res = _adam_run(z_best, n_primary, fs_c, fs_x, adam_polish_end,
                                      tau0=0.15, lr0=0.3, seed=seed + 5000)
            try_update(a_res)
        except Exception:
            pass
        
        # Exact gradient polish
        if time.monotonic() < polish_end - 10:
            try:
                a_res, sc_res = _exact_grad_polish(
                    a_polish, a_polish.size,
                    next_fast_len(2 * a_polish.size - 1),
                    min(time.monotonic() + 30, polish_end - 5),
                    rng=np.random.default_rng(seed + 7777)
                )
                try_update(a_res)
            except Exception:
                pass
        
        # Coordinate descent
        if time.monotonic() < polish_end - 3:
            try:
                a_res, sc_res = _coord_descent(
                    a_polish, a_polish.size,
                    next_fast_len(2 * a_polish.size - 1),
                    min(time.monotonic() + 30, polish_end - 1),
                    rng=np.random.default_rng(seed + 8888)
                )
                try_update(a_res)
            except Exception:
                pass
    
    # Phase 3: Final local search with small perturbations
    if time.monotonic() < final_deadline:
        rng_final = np.random.default_rng(seed + 31415)
        n_final = best_a.size
        fs_final = next_fast_len(2 * n_final - 1)
        sigma = 0.08
        for _ in range(20000):
            if time.monotonic() >= final_deadline:
                break
            z_p = np.log(np.maximum(best_a, 1e-12)) + rng_final.normal(0, sigma, n_final)
            a_p = _softmax(z_p)
            b_p = _fft_conv(a_p, n_final, fs_final)
            sc_p = 2.0 * n_final * float(b_p.max())
            if sc_p < best_score - 1e-12:
                best_a = a_p.copy()
                best_score = sc_p
                sigma = max(0.01, sigma * 0.95)
            else:
                sigma = min(0.2, sigma * 1.03)
    
    # Final validation
    try:
        cand = [float(v) for v in best_a]
        fs = float(evaluate_sequence(cand))
        if not np.isfinite(fs) or fs > best_score + 0.1:
            # Fallback
            cand = [1.0]
            fs = float(evaluate_sequence(cand))
            if not np.isfinite(fs):
                return [1.0]
    except Exception:
        return [1.0]
    
    return cand
