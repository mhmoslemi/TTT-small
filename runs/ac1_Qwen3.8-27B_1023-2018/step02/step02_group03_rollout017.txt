        alloc3 = min(200.0, deadline - t3 - 30)
        if alloc3 > 15:
            margins_tight = (1.0, 0.999, 0.998, 0.997, 0.995, 0.993, 0.99, 0.985)
            a3 = lp_opt(best_arr.copy() / best_arr.sum(),
                        best_arr.size, t3 + alloc3, margins=margins_tight)
            ub(a3)
            if time.monotonic() < deadline - 15:
                a3 = grad_opt(a3, a3.size, 20000, deadline - 10)
                ub(a3)

    # Phase 4: Try nearby lengths
    if time.monotonic() < deadline - 30:
        nearby_lengths = []
        for d in (-8, -6, -4, -3, -2, -1, 1, 2, 3, 4, 6, 8, 12, 16, 24, 32):
            L = n_p + d
            if 48 <= L <= 512 and L != n_p:
                nearby_lengths.append(L)
        nearby_lengths.sort()

        for L in nearby_lengths:
            if time.monotonic() >= deadline - 20:
                break
            t_rem = deadline - time.monotonic() - 20
            if t_rem < 15:
                break
            alloc = min(35.0, t_rem * 0.3)
            if alloc < 8:
                break

            a_init = interp_len(p, L)
            if linprog is not None and sps is not None:
                a_w = lp_opt(a_init, L, time.monotonic() + alloc * 0.6,
                            margins=(1.0, 0.99, 0.98, 0.96))
                ub(a_w)
            if time.monotonic() < deadline - 15:
                a_w = grad_opt(a_w, L, 5000, deadline - 12)
                ub(a_w)

    # Phase 5: Multiple restarts at parent length
    if time.monotonic() < deadline - 25 and linprog is not None:
        n_restarts = 10
        for r in range(n_restarts):
            if time.monotonic() >= deadline - 18:
                break
            t_rem = deadline - time.monotonic() - 18
            alloc = min(25.0, t_rem / max(1, n_restarts - r))
            if alloc < 5:
                break

            rtype = r % 5
            if rtype == 0:
                a_init = p_norm * np.exp(0.12 * rng.standard_normal(n_p))
            elif rtype == 1:
                a_init = p_norm[::-1].copy()
            elif rtype == 2:
                a_init = p_norm * np.exp(0.25 * rng.standard_normal(n_p))
                a_init = np.clip(a_init, 0, None)
            elif rtype == 3:
                a_init = p_norm + 0.08 * rng.standard_normal(n_p)
                a_init = np.clip(a_init, 0, None)
            else:
                # Smooth perturbation
                filt = np.fft.rfft(p_norm)
                filt *= np.exp(0.1 * rng.standard_normal(filt.size) + 0j * rng.standard_normal(filt.size))
                a_init = np.fft.irfft(filt, n_p)
                a_init = np.clip(a_init, 0, None)

            s_init = float(a_init.sum())
            if s_init > 1e-15:
                a_init /= s_init

            a_w = lp_opt(a_init, n_p, time.monotonic() + alloc * 0.6,
                        margins=(1.0, 0.99, 0.98))
            ub(a_w)
            if time.monotonic() < deadline - 12:
                a_w = grad_opt(a_w, n_p, 3000, deadline - 10)
                ub(a_w)

    # Phase 6: Final gradient polish on best
    if time.monotonic() < deadline - 5 and best_arr is not None:
        n_b = best_arr.size
        a_pol = best_arr.copy() / best_arr.sum()
        a_pol = grad_opt(a_pol, n_b, 50000, deadline - 2)
        ub(a_pol)

        # One more LP pass if time allows
        if time.monotonic() < deadline - 5 and linprog is not None:
            t6 = time.monotonic()
            alloc6 = deadline - t6 - 3
            if alloc6 > 10:
                margins_final = (1.0, 0.9995, 0.999, 0.998, 0.997)
                a_pol = lp_opt(a_pol, n_b, t6 + alloc6, margins=margins_final)
                ub(a_pol)
                if time.monotonic() < deadline - 2:
                    a_pol = grad_opt(a_pol, n_b, 5000, deadline - 1)
                    ub(a_pol)

    # === Final validation and return ===
    if best_arr is None:
        best_arr = np.ones(96, dtype=np.float64)
        ub(best_arr)

    final = np.clip(best_arr, 0.0, 1000.0)
    fs = float(final.sum())
    if fs < 0.01:
        final = np.ones_like(final, dtype=np.float64)
        ub(final)
    else:
        ub(final)

    return [float(x) for x in best_arr]
