Detailed code:
```
import numpy as np, time, math
try:
    from scipy.optimize import linprog
except Exception:
    linprog = None

TARGET = 1.50274349

def _as_clean_seq(a):
    a=np.asarray(a, dtype=np.float64).ravel()
    a=np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a=np.clip(a,0.0,1000.0)
    return a

def _score_general(a):
    a=_as_clean_seq(a)
    if a.size==0: return inf
    s=a.sum()
    if s<0.01: return inf
    if not np.all(np.isfinite(a)): return inf
    c=np.convolve(a,a)
    m=float(np.max(c))
    if not math.isfinite(m): return inf
```

```python
def _score_p(p):
    """Exact evaluator-equivalent score for a numpy array already in valid range."""
    try:
        p = np.asarray(p, dtype=np.float64).ravel()
        if p.size == 0:
            return float("inf")

def _weighted_grad(p, w):
    n = int(p.size)
    if n == 0:
        return p.copy()
    h = np.zeros(n, dtype=np.float64)
    if w.size == 0:
        return h
    w = np.asarray(w, dtype=np.float64).ravel()
    if w.size != 2 * n - 1:
        w = _resample_vector(w, 2 * n - 1)
        w = np.clip(w, 0.0, None)
        sw = float(w.sum())
        if sw > 0:
            w = w / sw
    idx = np.flatnonzero(w > 1e-16)
    if idx.size == 0:
        return h

    if idx.size <= max(8, n // 10):
        for kk in idx:
            kk = int(kk)
            wt = float(w[kk])
            if wt <= 0.0:
                continue
            t0 = kk - (n - 1)
            if t0 < 0:
                t0 = 0
            t1 = kk
            if t1 > n - 1:
                t1 = n - 1
            if t1 < t0:
                continue
            L = t1 - t0 + 1
            start = kk - t0
            if L == 1:
                h[start] += wt * p[t0]
            else:
                h[start:start - L:-1] += wt * p[t0:t1 + 1]
        return h
    else:
        c = np.convolve(p, w[::-1])
        return c[n - 1:2 * n - 1][::-1].copy()


def _soft_grad(p, alpha=12.0):
    """
    Returns (soft_max_value, gradient wrt p) for M = max convolve(p,p).
    The gradient is for M, not for 2n*M.
    """
    p = np.asarray(p, dtype=np.float64).ravel()
    n = int(p.size)
    if n == 0:
        return 0.0, np.zeros(0, dtype=np.float64)
    c = np.convolve(p, p)
    cmax = float(np.max(c))
    if not math.isfinite(cmax) or cmax <= 0.0:
        return cmax, np.zeros(n, dtype=np.float64)
    beta = max(1.0, float(alpha)) / cmax
    e = np.exp(np.clip(beta * (c - cmax), -80.0, 0.0))
    S = float(e.sum())
    if S <= 0.0 or not math.isfinite(S):
        w = np.zeros_like(c, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / S
    h = _weighted_grad(p, w)
    Msoft = cmax + (math.log(S) / beta if S > 0.0 else 0.0)
    g = 2.0 * h
    if not np.all(np.isfinite(g)):
        g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
    return float(Msoft), g

def _lp_escape(p, deadline, best):
    if _sp_linprog is None:
        return
    n = int(np.size(p))
    if n < 4 or n > 512:
        return
    t0 = time.monotonic()
    rem = deadline - t0
    if rem < 4.0:
        return
    p = _normalize_p(p)
    if p is None:
        return
    c = np.convolve(p, p)
    M = float(np.max(c))
    if not math.isfinite(M) or M <= 0.0:
        return

    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for i, pi in enumerate(p):
        if pi <= 0.0:
            continue
        k0 = i
        k1 = min(2 * n - 2, i + n - 1)
        k = np.arange(k0, k1 + 1, dtype=np.int64)
        A[k, i] = p[k - i]

    tl = max(1.0, min(5.0, rem - 1.5))
    try:
        res = _sp_linprog(
            -np.ones(n, dtype=np.float64),
            A_ub=A,
            b_ub=np.full(2 * n - 1, M * (1.0 + 1e-12), dtype=np.float64),
            bounds=[(0.0, None)] * n,
            method="highs",
            options={"presolve": True, "time_limit": float(tl)}
        )
    except Exception:
        return

    if res is None or getattr(res, "x", None) is None:
        return
    q = np.asarray(res.x, dtype=np.float64).ravel()
    q = np.nan_to_num(q, nan=0.0, posinf=0.0, neginf=0.0)
    q = np.clip(q, 0.0, None)
    s = float(q.sum())
    if not math.isfinite(s) or s <= 1e-15:
        return
    q = q / s

    for mix in (0.1, 0.2, 0.3, 0.5, 0.7, 1.0):
        cand = (1.0 - mix) * p + mix * q
        sc = cand.sum()
        if sc <= 1e-15:
            continue
        cand = cand / sc
        _update_best(best, cand, _score_p(cand))
        if time.monotonic() > deadline - 2.0:
            break

            p = np.ones(n0, dtype=np.float64) / float(n0)
        parent = _normalize_p(p)
        _update_best(best, parent)

    parent_n = int(parent.size)

    candidate_sizes = []
    def add_size(s, priority=0):
        try:
            s = int(s)
        except Exception:
            return
        if s < 64 or s > 1500:
            return
        found = False
        for x, pr in candidate_sizes:
            if x == s:
                if priority < pr:
                    x = s, priority
                found = True
                break
        if not found:
            candidate_sizes.append((s, priority))

    add_size(parent_n, 0)
    preferred = [512, 640, 1024, 448, 384, 256, parent_n * 2, parent_n // 2]
    for idx, s in enumerate(preferred):
        add_size(s, 10 + idx)
    candidate_sizes.sort(key=lambda x: x[1])
    sizes = [s for s, _ in candidate_sizes]
    if sizes:
        sizes = [sizes[0]] + [s for s in sizes if s != parent_n]
    else:
        sizes = [parent_n]
    sizes = sizes[:7]

    jobs = []
    for s in sizes:
        p_s = _resample_p(parent, s)
        v_s = _anti_v_from_p(p_s, s)
        if s == parent_n:
            a_auc, a_gen = 240.0, 100.0
        elif s in (512, 640):
            a_auc, a_gen = 150.0, 55.0
        elif s in (1024,):
            a_auc, a_gen = 100.0, 35.0
        elif s < parent_n:
            a_auc, a_gen = 45.0, 12.0
        else:
            a_auc, a_gen = 70.0, 20.0
        jobs.append(("anti", v_s, s, a_auc, 0))
        if s in (parent_n, 512, 640, 1024) or a_gen > 0:
            jobs.append(("general", p_s, s, a_gen, 0))

    top_for_fourier = []
    top_for_fourier.append(parent_n)
    for s in sizes:
        if s != parent_n and abs(s - 512) <= 128:
            top_for_fourier.append(s)
            break
    for j, s in enumerate(top_for_fourier[:2]):
        jobs.append(("fourier", None, s, 70.0, 10000 + 17 * parent_n + 71 * s + 11 * j))

    top_for_random = [parent_n]
    for s in sizes:
        if s != parent_n and (s >= 400 or s in (512, 640)):
            top_for_random.append(s)
            break
    for j, s in enumerate(top_for_random[:2]):
        for r in range(2):
            rv = _random_antipal_v(s, rng)
            jobs.append(("anti", rv, s, 35.0, 0))

    for typ, init, n, alloc, fdseed in jobs:
        now = time.monotonic()
        if now >= deadline:
            break
        rem = deadline - now
        if rem < 8.0:
            break
        use_alloc = min(alloc, max(5.0, rem * 0.32))
        job_deadline = min(deadline, now + use_alloc)
        if job_deadline - now < 3.0:
            continue
        try:
            if typ == "anti":
                _antipal_local(init, n, job_deadline, rng, best)
            elif typ == "general":
                _general_local(init, n, job_deadline, rng, best)
            elif typ == "fourier":
                _antipal_fourier_job(n, fdseed, job_deadline, rng, best)
        except Exception:
            pass

    lp_calls = 0
    polish_iter = 0
    while time.monotonic() < deadline - 8.0 and polish_iter < 60:
        polish_iter += 1
        now = time.monotonic()
        rem = deadline - now
        if rem < 12.0:
            break
        pbest = np.asarray(best[1], dtype=np.float64).ravel()
        pbest = _normalize_p(pbest)
        if pbest is None:
            break
        nbest = int(pbest.size)

        if rem > 30:
            vbest = _anti_v_from_p(pbest, nbest)
            dl = min(deadline, now + min(30.0, rem * 0.30))
            try:
                _antipal_local(vbest, nbest, dl, rng, best)
            except Exception:
                pass

        now = time.monotonic()
        if now < deadline - 12.0:
            rem = deadline - now
            dl = min(deadline, now + min(25.0, rem * 0.30))
            try:
                _general_local(_normalize_p(best[1]), int(np.size(best[1])), dl, rng, best)
            except Exception:
                pass

        now = time.monotonic()
        if lp_calls < 2 and now < deadline - 18.0:
            pb = np.asarray(best[1], dtype=np.float64).ravel()
            if np.size(pb) <= 512:
                try:
                    _lp_escape(pb, min(deadline, now + 10.0), best)
                except Exception:
                    pass
                lp_calls += 1

    final = _clean_vec(best[1])
    if final.size == 0:
        final = np.ones(16, dtype=np.float64)

    try:
        fs = evaluate_sequence([float(x) for x in final])
    except Exception:
        fs = float("inf")

    if not math.isfinite(fs) or fs >= best[0] + 1e-6 or final.size == 0:
        fb = np.asarray(best[1], dtype=np.float64).ravel()
        if fb.size > 0:
            try:
                fs2 = evaluate_sequence([float(x) for x in fb])
                if math.isfinite(fs2):
                    final = fb
                    fs = fs2
            except Exception:
                pass
        if not math.isfinite(fs):
            final = np.ones(16, dtype=np.float64)
            try:
                fs = evaluate_sequence([float(x) for x in final])
            except Exception:
                fs = float("inf")

    final = _clean_vec(final)
    sfin = float(final.sum())
    if final.size == 0 or sfin < 0.01:
        final = np.ones(16, dtype=np.float64)

    return [float(x) for x in final]
```
