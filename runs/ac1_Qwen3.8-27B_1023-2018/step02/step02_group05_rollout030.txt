        sol_tl = min(10.0, max(0.5, rem * 0.6))

        # Solve LP at standard M and a slightly tighter M
        g_list = []
        g = _lp(a, M, n, ci, sol_tl)
        if g is not None:
            # Optionally post-scale g so its own autocorrelation is also bounded
            g_conv = np.convolve(g, g)
            gm = float(g_conv.max())
            if gm > M * 1.05 and gm > 1e-30:
                alpha = math.sqrt(M / gm)
                g2 = alpha * g
                sg2 = float(g2.sum())
                if sg2 > 1e-15:
                    g_list.append(g2 / sg2)
            g_list.append(g)

        rem = dl - time.monotonic()
        if rem > 2.0 and it % 3 == 0:
            g_t = _lp(a, M * 0.95, n, ci, min(8.0, rem * 0.5))
            if g_t is not None:
                g_list.append(g_t)

        if not g_list:
            stag += 1
            if stag > 12:
                break
            continue

        best_sc, best_c = loc, None
        ts = (0.02, 0.05, 0.08, 0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5, 0.6, 0.7, 0.8, 0.9, 1.0)
        for g in g_list:
            for t in ts:
                c = np.clip((1.0 - t) * a + t * g, 0.0, None)
                sc = _score(c)
                if math.isfinite(sc) and sc < best_sc:
                    best_sc, best_c = sc, c

        if best_c is not None and best_sc < loc * (1.0 - 1e-12):
            a = best_c
            s2 = float(a.sum())
            a = a / s2 if s2 > 1e-15 else np.ones(n) / n
            loc = best_sc
            stag = 0
            _eval(a, best)
        else:
            stag += 1
            if stag >= 25:
                imp = False
                for ps_ in (0.05, 0.1, 0.2, 0.5, 1.0):
                    p = np.clip(a * np.exp(ps_ * rng.standard_normal(n)), 0.0, None)
                    pv = float(p.sum())
                    if pv > 1e-15:
                        p /= pv
                        sc = _score(p)
                        if math.isfinite(sc) and sc < loc:
                            a, loc = p, sc
                            _eval(a, best)
                            stag = 0
                            imp = True
                            break
                if not imp and stag > 60:
                    break
            elif stag > 10:
                conv2 = np.convolve(a, a)
                if abs(float(conv2.max()) - M) / max(M, 1e-15) < 1e-8 and stag > 15:
                    break

def _local_search(a0, n, tl, rng, best):
    n = int(n)
    if n < 4 or tl < 5:
        return
    a = np.asarray(a0, dtype=np.float64).ravel().copy()
    if a.size != n:
        if a.size > 1:
            a = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, a.size), np.maximum(a, 0))
        else:
            a = np.ones(n)
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    a = a / s if s > 1e-15 else np.ones(n) / n
    loc = _score(a)
    _eval(a, best)

    t0 = time.monotonic()
    dl = t0 + tl - 3
    no_imp = 0
    scale = 0.02
    npert = max(3, n // 10)

    while time.monotonic() < dl:
        if loc < TARGET:
            break
        improved = False
        for _ in range(40):
            if time.monotonic() >= dl:
                break
            idx = rng.integers(0, n, size=npert)
            p = a.copy()
            p[idx] *= np.exp(scale * rng.standard_normal(size=npert))
            p = np.clip(p, 0.0, None)
            pv = float(p.sum())
            if pv < 1e-15:
                continue
            p /= pv
            sc = _score(p)
            if math.isfinite(sc) and sc < loc:
                a, loc = p, sc
                _eval(a, best)
                improved = True
                no_imp = 0
                break
        if not improved:
            no_imp += 1
            scale *= 0.85
            if scale < 0.002:
                scale = 0.03
                no_imp = 0
            if no_imp > 150:
                break

def propose_candidate(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    try:
        rem_total = max(0.0, float(budget_s))
    except Exception:
        rem_total = 1300.0
    deadline = t_start + rem_total - 25

    best = [float('inf'), np.ones(8, dtype=np.float64)]
    _eval(best[1], best)

    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            parent = np.asarray(h, dtype=np.float64).ravel()
            parent = np.clip(np.nan_to_num(parent, nan=0.0, posinf=0.0, neginf=0.0), 0.0, 1000.0)
            if parent.size > 1:
                _eval(parent, best)
    except Exception:
        pass

    def _rem():
        return max(0.0, deadline - time.monotonic())

    # Phase 1: Heavy LP on parent
    if parent is not None and parent.size >= 8:
        np_ = parent.size
        alloc = min(_rem() * 0.50, 650.0)
        if alloc > 30:
            _lp_iter(parent, np_, alloc, rng, best)

    # Phase 2: Different lengths
    tried = set()
    if parent is not None:
        tried.add(parent.size)
    for n_try in (256, 384, 448, 200, 512, 192, 320, 288, 360, 440):
        if _rem() < 50 or n_try in tried or n_try < 16:
            continue
        tried.add(n_try)
        if parent is not None and parent.size > 1:
            a_init = np.interp(np.linspace(0, 1, n_try), np.linspace(0, 1, parent.size), np.maximum(parent, 0))
        else:
            x = np.linspace(0, 1, n_try)
            a_init = 1.0 + 0.6 * x
        alloc = min(_rem() * 0.10, 100.0)
        if alloc > 15:
            _lp_iter(a_init, n_try, alloc, rng, best)

    # Phase 3: Local search
    if _rem() > 20:
        ba = best[1]
        alloc = min(_rem() * 0.15, 100.0)
        if alloc > 10:
            _local_search(ba, ba.size, alloc, rng, best)

    # Phase 4: Final LP polish on best
    if _rem() > 15:
        ba = best[1]
        alloc = min(_rem() * 0.25, 80.0)
        if alloc > 10:
            _lp_iter(ba, ba.size, alloc, rng, best)

    # Phase 5: Final local search
    if _rem() > 10:
        ba = best[1]
        alloc = min(_rem() * 0.35, 40.0)
        if alloc > 5:
            _local_search(ba, ba.size, alloc, rng, best)

    # Final validation
    fa = np.asarray(best[1], dtype=np.float64).ravel().copy()
    fa = np.clip(np.nan_to_num(fa, nan=0.0, posinf=0.0, neginf=0.0), 0.0, 1000.0)
    if fa.size == 0:
        fa = np.ones(16)
    if float(fa.sum()) < 0.01:
        fa = np.ones(16)

    fs = evaluate_sequence([float(x) for x in fa])
    if not math.isfinite(fs) or fs > best[0] + 0.01:
        fa = np.ones(16)
        fs = evaluate_sequence([float(x) for x in fa])

    return [float(x) for x in fa]
