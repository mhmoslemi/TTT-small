        A[i:i+n, i] = f
    return A

def _lp_best_response(f, M, time_limit=20):
    """Solve max sum(b) s.t. conv(f,b) <= M, b >= 0."""
    n = f.size
    A = _make_conv_mat(f)
    c = -np.ones(n)
    ub = np.full(2*n - 1, M * (1 + 1e-9))
    try:
        r = linprog(c, A_ub=A, b_ub=ub, bounds=[(0, None)]*n,
                    method='highs', options={'presolve': True, 'time_limit': time_limit})
        if r.success and r.x is not None:
            b = np.clip(np.nan_to_num(r.x), 0, None)
            sb = float(b.sum())
            if sb > 1e-15:
                return b / sb
    except Exception:
        pass
    return None

def _lp_iterate(a0, stop_time, best, rng, max_iter=10000):
    """LP game iteration."""
    a = np.asarray(a0, dtype=np.float64).copy()
    a = np.clip(a, 0, None)
    s = float(a.sum())
    if s < 1e-15:
        a = np.ones_like(a)
        s = a.size
    a = a / s
    n = a.size
    cur = _score(a)
    _update_best(best, a)
    stall = 0
    for it in range(max_iter):
        if time.monotonic() > stop_time:
            break
        M = float(np.convolve(a, a).max())
        g = _lp_best_response(a, M, time_limit=10)
        if g is None:
            stall += 1
            if stall > 30:
                break
            continue
        # Geometric line search
        best_s, best_t = cur, 0.0
        ts = [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.25, 0.2, 0.15, 0.1, 0.05, 0.02]
        for t in ts:
            cand = (1 - t) * a + t * g
            sc = cand.sum()
            if sc < 1e-15:
                continue
            cand = cand / sc
            s = _score(cand)
            if s < best_s:
                best_s, best_t = s, t
        if best_t > 0 and best_s < cur * (1 - 1e-13):
            a = ((1 - best_t) * a + best_t * g) / (((1 - best_t) * a + best_t * g).sum())
            cur = best_s
            stall = 0
            _update_best(best, a)
        else:
            stall += 1
            if stall > 10 and stall % 5 == 0:
                # Perturb
                noise = np.exp(0.05 * rng.standard_normal(n))
                pa = a * noise
                pa = pa / pa.sum()
                ps = _score(pa)
                if ps < cur:
                    a, cur, stall = pa, ps, 0
                    _update_best(best, a)
            if stall > 60:
                break
    return a

def _subgrad_polish(a, stop_time, best):
    """Subgradient descent on the score function."""
    n = a.size
    a = a / a.sum()
    cur = _score(a)

        # db[k]/da[j] = d/da[j] sum_i a[i]*a[k-i]
        #            = (d/da[j])[a[j]*a[k-j]] + sum_{i != j} a[i]*d/da[j][a[k-i]]
        # The second sum: d/da[j][a[k-i]] = 1 if k-i=j, i.e., i = k-j.
        # If k-j != j (i.e., k != 2j), then the term a[k-j]*a[j] appears once in the sum,
        # and its derivative w.r.t. a[j] is a[k-j].
        # Similarly, the term a[j]*a[k-j] has derivative a[k-j] w.r.t. a[j].

        #             = a[k-j] if k-j == j (i.e., k=2j) [because the term is a[j]*a[j] and derivative is 2a[j]]
        #             Wait no: if k=2j, then the term in the sum at i=j is a[j]*a[j], and db/da[j] of this = 2a[j].
        #             But also, is there another term? i=k-j=j, same term. So no double counting.
        #             Actually: db[k]/da[j] = a[k-j] + a[k-j] = 2a[k-j] when k!=2j.

        # So the gradient of b[k_star] w.r.t. a is: grad[j] = 2*a[k_star - j] for j in [max(0,k_star-n+1), min(n-1,k_star)]

        # d(score)/da = (2n/s^2) * grad - (4n*b[k_star]/s^3) * ones

            break

            cur = s_new
            _update_best(best, a)
            lr *= 1.05
        else:
            lr *= 0.85
            if lr < 1e-8:
                break
    return a

def _score(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(np.nan_to_num(a, nan=0.0, posinf=1000.0, neginf=0.0), 0, 1000)
    n = a.size

        return float('inf')
    s = float(a.sum())
    if s < 1e-300:
        return float('inf')
    b = np.convolve(a, a)
    mb = float(b.max())
    return float(2 * n * mb / (s * s))

    try:
        a = np.asarray(cand, dtype=np.float64)
        a = np.clip(np.nan_to_num(a, nan=0.0, posinf=1000.0, neginf=0.0), 0, 1000)

def _gen_starts(n, rng, par):
    S = []
    x = np.linspace(0, 1, n)
    def add(a):
        a = np.asarray(a, dtype=np.float64)

        if a.size == n and a.sum() > 1e-15:

    # Uniform
    add(np.ones(n))
    # Exponentials
    for c in (0.5, 1.0, 2.0, 3.0, -0.5, -1.0, -2.0, -3.0):

            add(1.0 + amp*np.sin(2*np.pi*k*(x-0.5)))

    for alpha in (0.1, 0.5, 1.0, 5.0):
        try:
            add(rng.dirichlet(np.full(n, alpha)))

            pass

    add(1.0 + 0.5*np.cos(2*np.pi*x))
    # Parent interpolation
    if par is not None and par.size >= 2:

        add(np.interp(x, tp, par) * np.exp(0.05*rng.standard_normal(n)))
    # Random
    for _ in range(3):
        add(rng.dirichlet(np.full(n, rng.uniform(0.5, 3.0))))
    if not S:
        S.append(np.ones(n)/n)
    # Sort by score
    S.sort(key=_score)
    return S
