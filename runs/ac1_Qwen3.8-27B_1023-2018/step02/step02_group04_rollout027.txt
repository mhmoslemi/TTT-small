def _golden_search(a, g, n_evals=12):
    """Golden section search for best t in [0,1]."""
    inv_phi = (1 - math.sqrt(5)) / 2  # ≈ -0.618
    # We want to minimize f(t) = _scoring((1-t)*a + t*g)
    lo, hi = 0.0, 1.0
    # ... golden section
    gr = (math.sqrt(5) - 1) / 2  # ≈ 0.618
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = _scoring((1-c)*a + c*g)
    fd = _scoring((1-d)*a + d*g)
    for _ in range(n_evals - 2):
        if fc < fd:
            hi = d; d = c; fd = fc
            c = hi - gr*(hi-lo)
            fc = _scoring((1-c)*a + c*g)
        else:
            lo = c; c = d; fc = fd
            d = lo + gr*(hi-lo)
            fd = _scoring((1-d)*a + d*g)
    t_best = (lo + hi) / 2
    return t_best, _scoring((1-t_best)*a + t_best*g)
