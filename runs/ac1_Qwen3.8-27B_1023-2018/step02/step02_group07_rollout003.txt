def lp_refine(f, n, M, n_iter=100, time_limit=60):
    """
    LP-based refinement: iteratively find g that maximizes sum(g)
    subject to (f*g)(x) <= M for all x, g >= 0.
    Then blend f and g.
    """
    from scipy.optimize import linprog
    f = np.asarray(f, dtype=np.float64)
    n = len(f)
    
    # Build constraint matrix A: (2n-1) x n
    # A[x, i] = f[i] if max(0, x-n+1) <= i <= min(n-1, x)
    A = np.zeros((2*n-1, n))
    for x in range(2*n-1):
        i_lo = max(0, x - n + 1)
        i_hi = min(n-1, x)
        A[x, i_lo:i_hi+1] = f[i_lo:i_hi+1]
    
    # Objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: A @ g <= M
    b_ub = np.full(2*n-1, M)
    
    # g >= 0 (bounds)
    bounds = [(0, None)] * n
    
    f_old = f.copy()
    for it in range(n_iter):
        if time.monotonic() > time_limit:
            break
        
        M = max(np.convolve(f_old, f_old))
        b_ub = np.full(2*n-1, M)
        
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        
        if res.success and res.x is not None:
            g = res.x
            g_sum = g.sum()
            if g_sum > 0:
                g = g / g_sum * f_old.sum()
                # Blend
                t = 0.5
                f_new = (1-t) * f_old + t * g
                # Check if it improves
                new_M = max(np.convolve(f_new, f_new))
                if new_M < M:
                    f_old = f_new

    F = rfft(a, fs)
    return irfft(F * F, fs)[:2*n-1]

def _cc_fft(a, b, n, fs):
    """Cross-correlation: result[k] = sum_i a[i]*b[i+k] for k=0..n-1"""
    ar = a[::-1]
    c = irfft(rfft(ar, fs) * rfft(b, fs), fs)
    return c[n-1:2*n-1]

def _score(a, n, fs):
    b = _conv_fft(a, n, fs)
    m = float(b.max())
    if not np.isfinite(m):
        return math.inf
    return 2.0 * n * m

def _softmax(x):
    x = x - x.max()
    e = np.exp(np.clip(x, -50, 50))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(x) / len(x)
    return e / s

def _grad_step(z, n, fs_c, fs2, tau_frac, w_reg, rng):
    """One gradient step on log-space variable z."""
    a = _softmax(np.clip(z, -30, 30))
    b = _conv_fft(a, n, fs_c)
    bmax = float(b.max())
    if not np.isfinite(bmax) or bmax <= 0:
        return z, 0.0, math.inf
    
    # Softmax-weighted peak indicator
    tau = max(1e-12, tau_frac * bmax)
    bt = b / tau
    bt = bt - bt.max()
    e = np.exp(bt)
    s = e.sum()
    p = e / s if s > 0 else np.zeros_like(b)
    
    # Gradient of sum(p_k * b_k) wrt a
    # db_k/da_i = a_{k-i} + a_{k+i} (for the self-convolution)
    # Actually, b_k = sum_j a_j a_{k-j}
    # db_k/da_i = a_{k-i} + a_{k+i} (with the convention a_outside = 0)
    # So grad_a_i = sum_k p_k (a_{k-i} + a_{k+i}) = 2 * (p * a)_i (cross-correlation)
    g = 2.0 * _cc_fft(p, a, n, fs2)
    
    # Project to tangent space (sum a = 1 constraint)
    grad = a * (g - float(np.dot(a, g)))
    
    # Regularizer: penalize deviation of b from uniform
    if w_reg > 0:
        q = b - 1.0/(2*n-1)
        vg = 4.0 * _cc_fft(q, a, n, fs2)
        grad = grad + w_reg * a * (vg - float(np.dot(a, vg)))
    
    return z, grad, 2.0*n*bmax

import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _conv_fft(a, n, fs):
    F = rfft(a, fs)
    return irfft(F * F, fs)[:2 * n - 1]


def _cc_fft(p, a, n, fs):
    ar = a[::-1]
    c = irfft(rfft(ar, fs) * rfft(p, fs), fs)
    return c[n - 1:2 * n - 1]


def _score_conv(b, n):
    m = float(b.max())
    if not np.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m


def _softmax(x):
    x = np.asarray(x, dtype=np.float64).ravel()
    x = x - x.max()
    e = np.exp(np.clip(x, -50.0, 50.0))
    s = float(e.sum())
    if not np.isfinite(s) or s < 1e-300:
        return np.ones_like(x) / len(x)
    return e / s


def _init_variants(n, rng, parent=None):

def _lp_refine(n, f_init, end_time, seed=0):
    """LP-based iterative refinement. Returns best (a, score)."""
    fs_c = next_fast_len(2 * n - 1)
    f = np.asarray(f_init, dtype=np.float64).copy()
    f = np.maximum(f, 1e-12)
    f = f / f.sum()
    rng = np.random.default_rng(seed)

    best_f = f.copy()
    b0 = _conv_fft(f, n, fs_c)
    best_score = _score_conv(b0, n)
    bmax = float(b0.max())

    iter_count = 0
    no_progress = 0
    alpha = 0.4

    while time.monotonic() < end_time - 1.0:
        # Rebuild constraint matrix
        b = _conv_fft(f, n, fs_c)
        M = float(b.max())
        if not np.isfinite(M) or M < 1e-15:
            break

        # A[x, j] = f[x-j] if 0 <= x-j <= n-1
        A = np.zeros((2 * n - 1, n), dtype=np.float64)
        for j in range(n):
            A[j:j + n, j] = f

        c = -np.ones(n)
        try:
            res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                          bounds=[(0, None)] * n, method='highs',
                          options={'time_limit': 5})
        except Exception:
            break

        if not res.success or res.x is None:
            no_progress += 1
            if no_progress > 20:
                break
            continue

        g = res.x
        g_sum = float(g.sum())
        if g_sum < 1e-15:
            break

        # Rescale g to match f's sum
        g = g / g_sum * f.sum()

        # Try blending
        improved = False
        for t in [alpha, alpha * 0.5, alpha * 0.1, 0.01]:
            f_new = (1 - t) * f + t * g
            f_new = np.maximum(f_new, 1e-12)
            b_new = _conv_fft(f_new, n, fs_c)
            M_new = float(b_new.max())
            if np.isfinite(M_new) and M_new < M - 1e-14:
                f = f_new
                bmax = M_new
                score = _score_conv(b_new, n)
                improved = True
                if score < best_score - 1e-13:
                    best_score = score
                    best_f = f.copy()
                break

        if not improved:
            no_progress += 1
            if no_progress > 30:
                # Perturb and retry
                f = f + rng.normal(0, 0.02 * f.mean(), n)
                f = np.maximum(f, 1e-10)
                f = f / f.sum()
                no_progress = 0

                    break

    b = _conv_fft(a, n, fs_c)

    lp2_end = t0 + 0.94 * total
    try:
        if time.monotonic() < lp2_end - 5:
            a_lp2, s_lp2 = _lp_refine(n, best_a, lp2_end, seed=seed + 7777)
            _try_update(a_lp2, s_lp2)
    except Exception:
        pass

    # Phase 5: Local search (5%)
    ls_end = t0 + total - 10.0
    try:
        if time.monotonic() < ls_end - 2:
            a_ls, s_ls = _local_search(best_a, ls_end, seed=seed + 31415)
            _try_update(a_ls, s_ls)
    except Exception:
        pass

    # Final validation
    try:
        cand = [float(v) for v in best_a]
        fs = float(evaluate_sequence(cand))
        if not np.isfinite(fs):
            cand = [1.0]
            fs = float(evaluate_sequence(cand))
    except Exception:
        cand = [1.0]
        fs = float(evaluate_sequence(cand))
        if not np.isfinite(fs):
            return [1.0]

    return cand
