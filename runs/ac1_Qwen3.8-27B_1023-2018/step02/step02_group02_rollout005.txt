        if stop <= time.monotonic():
            return a_in.copy()

        a = np.asarray(a_in, dtype=np.float64).copy()

        if DENSE_QUAD:
            a = newton_refine(a, stop - 0.45 * budget)

        if time.monotonic() < stop - 0.22 * budget:
            a = lse_refine(a, stop - 0.28 * budget)

        if DENSE_QUAD:
            a = newton_refine(a, stop - 0.15 * budget)

        if time.monotonic() < stop - 0.07 * budget:
            a = lp_refine(a, stop - 0.10 * budget)

        if time.monotonic() < stop - 1.0:
            a = stochastic_refine(a, stop - 1.0)

        return a

    starts = []

    def add_start(cand):
        try:
            arr = np.asarray(cand, dtype=np.float64).ravel()
        except Exception:
            return
        if arr.size != n:
            return
        arr = np.clip(arr, 0.0, 1000.0)
        ss = float(np.sum(arr))
        if ss < 1e-8:
            return
        arr /= ss
        try:
            pk = peak(arr)
        except Exception:
            return
        if not np.isfinite(pk):
            return
        for _, prev in starts:
            if np.linalg.norm(arr - prev) < 1e-6:
                return
        starts.append((pk, arr))

    add_start(a0)

    rotstep = max(1, n // 96)
    rot_scores = []
    for sh in range(0, n, rotstep):
        rr = np.roll(a0, sh)
        try:
            rot_scores.append((peak(rr), sh))
        except Exception:
            pass
    rot_scores.sort()
    added_sh = set()
    for _, sh in rot_scores[:10]:
        if sh not in added_sh:
            added_sh.add(sh)
            add_start(np.roll(a0, sh))

    add_start(np.ones(n, dtype=np.float64) / float(n))

    for sig in (0.05, 0.10, 0.20, 0.35, 0.55):
        z = np.log(np.maximum(a0, 1e-12)) + sig * rng.normal(size=n)
        z -= float(np.max(z))
        add_start(np.exp(z))

    idx = np.arange(n, dtype=np.float64)
    for f in (1, 2, 3, 5, 8):
        for amp in (0.1, 0.2):
            ph = float(rng.uniform(0.0, 2.0 * math.pi))
            cand = a0 * np.exp(amp * np.sin(2.0 * math.pi * f * idx / float(n) + ph))
            cand = np.maximum(cand, 0.0)
            add_start(cand)

    for amp in (0.2, 0.4, 0.7):
        z = np.zeros(n, dtype=np.float64)
        for f in range(1, 4):
            ph = float(rng.uniform(0.0, 2.0 * math.pi))
            z += amp * (float(rng.normal()) * np.cos(2.0 * math.pi * f * idx / float(n) + ph))
        z -= float(np.max(z))
        add_start(np.exp(z))

    z = rng.normal(size=n)
    z -= float(np.max(z))
    add_start(np.exp(z))

    ramp = 1.0 + 0.4 * np.linspace(-1.0, 1.0, n, dtype=np.float64)
    add_start(a0 * ramp)

    starts.sort()
    if not starts:
        fallback = np.ones(n, dtype=np.float64) / float(n)
        starts = [(peak(fallback), fallback)]

    best_pk = math.inf
    best_pk_a = np.ones(n, dtype=np.float64) / float(n)
    best_exact = math.inf
    best_exact_a = np.ones(n, dtype=np.float64) / float(n)

    def consider(a):
        nonlocal best_pk, best_pk_a, best_exact, best_exact_a
        if a is None:
            return
        try:
            arr = np.asarray(a, dtype=np.float64).ravel()
        except Exception:
            return
        if arr.size != n:
            return
        arr = np.clip(arr, 0.0, 1000.0)
        ss = float(np.sum(arr))
        if ss < 1e-12:
            return
        arr /= ss

        try:
            pk = peak(arr)
        except Exception:
            return
        if np.isfinite(pk) and pk < best_pk:
            best_pk = pk
            best_pk_a = arr.copy()

        ex = exact_score([float(x) for x in arr])
        if np.isfinite(ex) and ex < best_exact:
            best_exact = ex
            best_exact_a = arr.copy()

    max_initial_exact = 20 if n > 256 else 60
    for i, (_, arr) in enumerate(starts):
        if i >= max_initial_exact:
            break
        consider(arr)

    work = [arr.copy() for _, arr in starts[:5]]
    extra_uniform = np.ones(n, dtype=np.float64) / float(n)
    z = rng.normal(size=n)
    z -= float(np.max(z))
    extra_random = np.exp(z)
    extra_random /= float(np.sum(extra_random))

    extra = [extra_uniform, extra_random]
    work.extend(extra)

    unique_work = []
    for cand in work:
        cand = np.asarray(cand, dtype=np.float64).ravel().copy()
        ss = float(np.sum(cand))
        if ss <= 0.0:
            continue
        cand /= ss
        ok = True
        for prev in unique_work:
            if np.linalg.norm(cand - prev) < 1e-5:
                ok = False
                break
        if ok:
            unique_work.append(cand)
    work = unique_work[:6]

    fracs = [0.38, 0.20, 0.13, 0.09, 0.07, 0.05]

    for i, cand in enumerate(work):
        rem = deadline - time.monotonic()
        if rem < 30.0:
            break
        frac = fracs[i] if i < len(fracs) else 0.03
        b = rem * frac
        if b < 5.0:
            b = max(1.0, rem - 5.0)
        b = min(b, rem - 5.0)
        if b <= 0.0:
            break

        polished = polish(cand.copy(), b)
        consider(polished)

    work-base = best_exact_a if np.isfinite(best_exact) else best_pk_a
    work_base = np.array(work-base if False else (best_exact_a if np.isfinite(best_exact) else best_pk_a), copy=True)

    rem = deadline - time.monotonic()
    if rem > 50.0:
        a_fin = lse_refine(work_base, deadline - 25.0)
        consider(a_fin)

    rem = deadline - time.monotonic()
    if rem > 25.0:
        if DENSE_QUAD:
            a_fin = newton_refine(work_base, deadline - 12.0)
        else:
            a_fin = lse_refine(work_base, deadline - 12.0)
        consider(a_fin)

    rem = deadline - time.monotonic()
    if rem > 12.0:
        a_fin = stochastic_refine(work_base, deadline - 6.0)
        consider(a_fin)

    final_candidates = [
        best_exact_a if np.isfinite(best_exact) else None,
        best_pk_a,
        work_base,
    ]

    final_list = None
    final_score = math.inf

    for arr in final_candidates:
        if arr is None:
            continue
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size != n:
            continue
        arr = np.clip(arr, 0.0, 1000.0)
        ss = float(np.sum(arr))
        if ss < 0.01:
            continue
        arr /= ss
        lst = [float(x) for x in arr]
        sc = exact_score(lst)
        if np.isfinite(sc) and sc < final_score:
            final_score = sc
            final_list = lst

    if final_list is None:
        fallback = np.ones(n, dtype=np.float64) / float(n)
        final_list = [float(x) for x in fallback]
        try:
            exact_score(final_list)
        except Exception:
            pass

    return final_list
