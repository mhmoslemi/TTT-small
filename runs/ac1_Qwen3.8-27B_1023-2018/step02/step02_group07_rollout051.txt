Actually wait, the second term: da_{k-j}/da_i = 1 if k-j = i, i.e., j = k-i. So the second sum gives a_{k-i} (the value of a at index k-i when j = k-i).

And the first term: da_j/da_i = 1 if j = i. So the first sum gives a_{k-i} (the value of a_{k-i}).

So indeed db_k/da_i = 2*a_{k-i} (if 0 <= k-i <= n-1, else 0).

Now, the gradient of L:
dL/da_i = tau * (1/sum(exp)) * sum_k exp(b_k/tau) * (b_k/tau) * db_k/da_i
         = sum_k p_k * 2*a_{k-i} (where p_k = softmax(b/tau)_k)
         = 2 * sum_k p_k * a_{k-i}

This is 2 * (anti-convolution of p and a) evaluated at index i.

The anti-convolution (correlation) of p and a: sum_k p_k * a_{k-i} = sum_j p_{j+i} * a_j (reindexing).

This can be computed via FFT: it's the cross-correlation of p and a.

In the previous code, `_cc(q, a, n, fs2)` computes this:
```python
def _cc(q, a, n, fs2):
    ar = a[::-1]
    c = irfft(rfft(ar, fs2) * rfft(q, fs2), fs2)
    return c[n - 1 : 2 * n - 1]
```

This computes sum_k q[k] * a[k+i-(n-1)] for i=0,...,n-1, which is the cross-correlation. Let me verify:
- ar = a[::-1] = [a_{n-1}, ..., a_0]
- conv(ar, q) at position m = sum_j ar[j] * q[m-j] = sum_j a_{n-1-j} * q[m-j]
- irfft(rfft(ar)*rfft(q)) gives conv(ar, q)
- c[n-1 : 2n-1] gives positions n-1 to 2n-2

At position n-1+i: sum_j a_{n-1-j} * q[n-1+i-j] = sum_j a_{n-1-j} * q[n-1+i-j]
Let k = n-1-j: = sum_k a_k * q[i+k] = sum_k a_k * q_{i+k}

So _cc(q, a)[i] = sum_k a_k * q_{i+k}. This is the "anti-convolution" or one-sided correlation.

And the gradient is 2 * _cc(p, a) (with appropriate indexing). The previous code uses `g = 2.0 * _cc(p, a, n, fs2)` which matches.

Then the "unconstrained gradient" (in log-space with softmax parameterization) is:
ugrad = a * (g - dot(a, g))

This is the gradient of L with respect to log(a) projected to the hyperplane sum(a) = constant.

        z = z - lr * step
        z = z - z.mean()
        z = np.clip(z, -30.0, 30.0)
        z[~np.isfinite(z)] = 0.0

        a2 = _softmax(z)
        b2 = _conv(a2, fs_c)
        score2 = 2.0 * n * float(b2.max())

        if np.isfinite(score2) and score2 < best_score - 1e-12:
            best_score = score2
            best_z = z.copy()
            best_a = a2.copy()
            stuck = 0
        else:
            stuck += 1

        if t % 12 == 0:
            tau_frac = max(0.015, tau_frac * 0.94)
        if stuck > 35:
            lr = max(0.05, lr * 0.87)
        if stuck > 120:
            z = best_z + rng.normal(0, 0.5, n)
            z = z - z.mean()
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            lr = 0.55
            tau_frac = 0.35
            stuck = 0

    a = _softmax(best_z)
    b = _conv(a, fs_c)
    s = 2.0 * n * float(b.max())
    return a, s


def _polish(a, end_time, hard_deadline):
    n = a.size
    fs = next_fast_len(2 * n - 1)
    a = np.maximum(np.asarray(a, dtype=np.float64).ravel(), 1e-12)
    a = a / a.sum()
    best = a.copy()
    best_s = 2.0 * n * float(_conv(best, fs).max())

    rng = np.random.default_rng(777)
    sigma = 0.12
    stuck = 0

    while time.monotonic() < min(end_time, hard_deadline) - 0.3:
        z = np.log(np.maximum(a, 1e-12))
        r = rng.random()
        if r < 0.25:
            zc = z + rng.normal(0, sigma, n)
        elif r < 0.45:
            i = int(rng.integers(0, n))
            w = max(1, min(n, int(rng.integers(1, n // 25 + 1))))
            i2 = min(n, i + w)
            zc = z.copy()
            zc[i:i2] += rng.normal(0, sigma * 0.5, i2 - i)
        elif r < 0.60:
            zc = (z - z.mean()) * float(rng.uniform(0.4, 2.0))
        elif r < 0.75:
            zc = z.copy()
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            d = float(rng.normal(0, sigma))
            zc[i] += d
            zc[j] -= d
        else:
            f = float(rng.uniform(1, 3))
            ph = float(rng.uniform(0, 2 * math.pi))
            zc = z + sigma * 0.3 * np.sin(2 * math.pi * f * np.arange(n) / n + ph)

        zc = zc - zc.mean()
        a2 = _softmax(zc)
        b2 = _conv(a2, fs)
        s2 = 2.0 * n * float(b2.max())

        if np.isfinite(s2) and s2 < best_s - 1e-11:
            best = a2.copy()
            best_s = s2
            a = a2.copy()
            stuck = 0
            sigma = max(0.02, sigma * 0.96)
        else:
            stuck += 1
            sigma = min(1.2, sigma * 1.06)
            if stuck > 100:
                stuck = 0
                sigma = 0.25

    return best, best_s


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    total = max(0.0, float(budget_s))
    hard_deadline = t0 + total - 4.0

    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            if p.size > 0 and np.isfinite(p).all():
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_a = None
    best_score = math.inf

    def _try_update(a, s):
        nonlocal best_a, best_score
        if a is None or s is None or not np.isfinite(s):
            return
        a = np.asarray(a, dtype=np.float64).ravel()
        if a.size == 0:
            return
        a = np.clip(a, 0.0, 1000.0)
        if a.sum() < 0.01:
            return
        if s < best_score:
            best_a = a.copy()
            best_score = float(s)

    # Evaluate parent
    if parent is not None and parent.size > 0:
        try:
            ps = float(evaluate_sequence([float(v) for v in parent]))
            if np.isfinite(ps):
                _try_update(parent, ps)
        except Exception:
            pass

    # Fallback
    if best_a is None:
        best_a = np.ones(64) / 64.0
        best_score = 2.0

    n_par = parent.size if parent is not None else 192

    # Phase 1: optimize at parent length
    p1_end = t0 + min(0.40 * total, 500.0)
    try:
        init1 = best_a.copy() if best_a.size == n_par else np.ones(n_par) / n_par
        if parent is not None and parent.size == n_par:
            init1 = parent.copy()
        a1, s1 = _optimize(init1, n_par, p1_end, hard_deadline, seed)
        _try_update(a1, s1)
    except Exception:
        pass

    # Phase 2: try other lengths
    p2_end = t0 + 0.82 * total
    other_sizes = []
    for mult in [2, 3]:
        sz = n_par * mult
        if sz <= 4096:
            other_sizes.append(sz)
    for sz in [256, 384, 512]:
        if sz != n_par and sz <= 4096 and sz not in other_sizes:
            other_sizes.append(sz)
    other_sizes = sorted(set(other_sizes))[:3]

    if other_sizes and time.monotonic() < p2_end - 2:
        per_sz = max(10.0, (p2_end - time.monotonic() - 2.0) / len(other_sizes))
        for si, sz in enumerate(other_sizes):
            end_t = min(time.monotonic() + per_sz, p2_end - 1.0)
            if time.monotonic() >= end_t - 1:
                continue
            # Upsample best
            a_up = np.interp(np.linspace(0, 1, sz), np.linspace(0, 1, best_a.size), best_a)
            a_up = np.maximum(a_up, 1e-6)
            try:
                a2, s2 = _optimize(a_up, sz, end_t, hard_deadline, seed + 2000 + si)
                _try_update(a2, s2)
            except Exception:
                pass

    # Phase 3: polish
    p3_end = hard_deadline - 2.0
    if time.monotonic() < p3_end - 3:
        try:
            a_p, s_p = _polish(best_a, p3_end, hard_deadline)
            _try_update(a_p, s_p)
        except Exception:
            pass

    # Final validation
    try:
        cand = [float(max(0.0, v)) for v in best_a]
        if not cand or any(not np.isfinite(x) for x in cand):
            cand = [1.0]
        fs = float(evaluate_sequence(cand))
        if not np.isfinite(fs):
            cand = [1.0]
            fs = float(evaluate_sequence(cand))
    except Exception:
        cand = [1.0]

    return cand
