import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len

def _fft_conv(a):
    """Compute full linear convolution a*a using FFT."""
    n = len(a)
    fsize = next_fast_len(2*n-1)
    F = rfft(a, fsize)
    c = irfft(F*F, fsize)[:2*n-1]
    np.maximum(c, 0.0, out=c)
    c[~np.isfinite(c)] = 0.0
    return c

def _score(a):
    """Compute the score (without the evaluate_sequence call)."""
    n = len(a)
    b = _fft_conv(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    return 2.0 * n * b.max() / (s*s)

def _normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def _build_A(a):
    """Build the constraint matrix A where A[k,j] = a[k-j] if 0<=k-j<n."""
    n = len(a)
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for j in range(n):
        A[j:j+n, j] = a
    return A

def _solve_lp(a, M):
    """Solve LP: max 1^T g s.t. A g <= M, g >= 0."""
    import cvxpy as cp
    n = len(a)
    g = cp.Variable(n, nonneg=True)
    A = _build_A(a)
    constraints = [A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    
    for solver in [cp.ECOS, cp.SCS, cp.CBC, cp.GLPK, cp.GLOP]:
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                gv = g.value
                if gv is not None and np.isfinite(gv).all():
                    gv = np.maximum(gv, 0)
                    s = gv.sum()
                    if s > 1e-15:
                        return gv / s
        except Exception:
            continue
    return None

Hmm, I realize I should think more carefully about the LP. The constraint A @ g <= M means for each k:
sum_j A[k,j] * g[j] <= M
sum_j a[k-j] * g[j] <= M (where the sum is over valid j)

This is the same as (a * g)[k] <= M.

At the current point a (normalized), (a*a)[k] <= M by definition (M is the max). So a is a feasible point. But the LP maximizes sum(g), so the solution g* will have sum(g*) >= sum(a) = 1.

After normalizing g* to sum 1, we get h. The question is whether h has a lower max autocorrelation.

Let me think about this geometrically. The feasible set {g >= 0 : (a*g) <= M} is a polytope. The LP finds the point in this polytope that maximizes 1^T g. This is the point "furthest" in the all-ones direction.

If a is already at the vertex in the 1^T direction, then g* = a (up to scaling), and no improvement is possible. Otherwise, g* is different, and moving toward it should help.

The tricky part is: (a*g) <= M doesn't guarantee (g*g) <= M. So the LP is a relaxation. The iteration might oscillate or converge slowly.

To address this, I can:
1. Use a **shrinkage**: a_new = (1-t)*a + t*g* with small t.
2. **Check** the actual score after each step and only accept if it improves.
3. **Perturb** if stuck.

Let me also consider a **quadratic** refinement. Once the LP iteration converges (g* ≈ a), I can try:
- Move along the subgradient direction.
- Do a local QP: minimize (a*a)[k] for the active lag k*, subject to a >= 0, sum a = 1.

The QP for a fixed active lag k*:
minimize sum_j a[j]*a[k*-j] (with the understanding that this is (a*a)[k*])
subject to a >= 0, sum a = 1.

This is a quadratic minimization with linear constraints. The objective is a quadratic form: a^T Q_k a where Q_k is a permutation-like matrix.

For k* = n-1 (center): (a*a)[n-1] = sum_j a[j]*a[n-1-j] = a^T P a where P is the reversal matrix. P is symmetric, so this is a standard QP.

But the max might be achieved at multiple lags, making the problem more complex.

Define f(a) = max_k (a*a)[k]. The subgradient is:

For the **Nesterov acceleration**:
- Maintain momentum: v ← β*v + ∇f(a)
- a ← project_simplex(a - step * v)
- β is the momentum parameter (e.g., 0.9).

This can converge much faster than plain subgradient.

Actually, I want to think about one more thing: the **dual** of the LP.

Primal: max 1^T g s.t. A g <= M, g >= 0.
Dual: min M * 1^T y s.t. A^T y >= 1, y >= 0.

A^T y >= 1 means: for each j, sum_k A[k,j] * y[k] >= 1, i.e., sum_k a[k-j] * y[k] >= 1.

Or equivalently: (a * y)[k-j] >= 1 for all j, where the convolution is taken appropriately.

The dual has 2n-1 variables (y) and n constraints. For n=192, that's 383 variables and 192 constraints. Also small.

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len

def _fft_conv(a):
    n = len(a)
    fsize = next_fast_len(2*n-1)
    F = rfft(a, fsize)
    c = irfft(F*F, fsize)[:2*n-1]
    np.maximum(c, 0.0, out=c)
    c[~np.isfinite(c)] = 0.0
    return c

def _build_A(a):
    n = len(a)
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for j in range(n):
        A[j:j+n, j] = a
    return A

def _solve_lp_dir(a, M):
    """Solve LP to get a descent direction."""
    try:
        import cvxpy as cp
        n = len(a)
        g = cp.Variable(n, nonneg=True)
        A = _build_A(a)
        prob = cp.Problem(cp.Maximize(cp.sum(g)), [A @ g <= M])
        for solver in [cp.ECOS, cp.SCS, cp.CBC, cp.GLPK]:
            try:
                prob.solve(solver=solver, verbose=False)
                if prob.status in ['optimal', 'optimal_inaccurate']:
                    gv = np.asarray(g.value, dtype=np.float64)
                    if gv.shape == (n,) and np.isfinite(gv).all():
                        gv = np.maximum(gv, 0)
                        s = gv.sum()
                        if s > 1e-10:
                            return gv / s
            except Exception:
                continue
    except Exception:
        pass
    return None

def _subgrad_dir(a, S):
    """Subgradient of max_k (a*a)[k] using active set S."""
    n = len(a)
    grad = np.zeros(n)
    for k in S:
        for j in range(n):
            d = k - j
            if 0 <= d < n:
                grad[j] += 2.0 * a[d]
    return grad / max(1, len(S))

def _lp_iterate(a, deadline, rng, max_iters=500):
    """LP-based iteration to minimize max(a*a)."""
    n = len(a)
    a = _normalize(a.copy())
    best_a = a.copy()
    best_M = float(_fft_conv(a).max())
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = _fft_conv(a)
        M = float(b.max())
        
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
        
        # Find active lags
        thresh = M * 0.999
        S = np.where(b >= thresh)[0].tolist()
        
        # LP direction
        g = _solve_lp_dir(a, M)
        
        if g is None:
            # Subgradient fallback
            grad = _subgrad_dir(a, S)
            gn = np.linalg.norm(grad)
            if gn < 1e-15:
                break
            g = a - 0.1 * grad / gn
            g = _normalize(g)
        
        # Check similarity
        cos_sim = float(np.dot(a, g) / (np.linalg.norm(a) * np.linalg.norm(g) + 1e-30))
        if cos_sim > 0.9999:
            # Too similar, perturb
            a = _normalize(a + rng.normal(0, 0.05, n))
            continue
        
        # Line search
        best_t = 0.0
        best_mb = M
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_new = _normalize((1-t)*a + t*g)
            mb = float(_fft_conv(a_new).max())
            if mb < best_mb:
                best_mb = mb
                best_t = t
        
        if best_t > 0:
            a = _normalize((1-best_t)*a + best_t*g)
        else:
            # Try subgradient step
            grad = _subgrad_dir(a, S)
            gn = np.linalg.norm(grad)
            if gn > 1e-15:
                for step in [0.01, 0.05, 0.1, 0.5]:
                    a_new = _normalize(a - step * grad / gn)
                    mb = float(_fft_conv(a_new).max())
                    if mb < M:
                        a = a_new
                        break
    
    return best_a, best_M
