- (sum f)^2 is convex (square of a linear function, which is convex)

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(198, dtype=np.float64)
    if p.size == 0:
        p = np.ones(198, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(198, dtype=np.float64)
    return p


def _conv_full(a):
    n = len(a)
    if n <= 600:
        return np.convolve(a, a, mode="full")
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[: 2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n):
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j : j + n, j] = f
    return A


def _lp_direction(f, n, time_limit=8.0):
    b = _conv_full(f)
    M = float(b.max())
    if M < 1e-15:
        return None, M, b
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(
            c, A_ub=A, b_ub=b_ub,
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True, "time_limit": time_limit},
        )
    except Exception:
        return None, M, b
    if res.status != 0 or res.x is None:
        return None, M, b
    g = np.asarray(res.x, dtype=np.float64)
    g_sum = float(g.sum())
    if g_sum < 1e-15:
        return None, M, b
    return g / g_sum, M, b


def _lp_optimize(f0, n, stop_time, seed, max_iters=500):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    while f.size < n:
        f = np.concatenate([f, np.zeros(n - f.size)])
    f = f[:n]
    f_sum = float(f.sum())
    if f_sum < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / f_sum

    best_f = f.copy()
    best_score = _score(f)
    conv = _conv_full(f)
    current_max_b = float(conv.max())

    t = 0.20
    t_min = 0.01
    t_max = 0.6
    stall = 0
    max_stall = 150
    iteration = 0

    while time.monotonic() < stop_time - 3.0 and iteration < max_iters:
        g_dir, M, b = _lp_direction(f, n, time_limit=6.0)
        if g_dir is None:
            stall += 5
            if stall > max_stall:
                break
            f = np.clip(f + rng.normal(0, 0.04, n), 0, None)
            fs = f.sum()
            if fs > 1e-10:
                f = f / fs
            else:
                f = best_f.copy()
            continue

        dist = float(np.linalg.norm(g_dir - f))
        if dist < 1e-7:
            stall += 4
            if stall > max_stall:
                break
            f = np.clip(f + rng.normal(0, 0.05, n), 0, None)
            fs = f.sum()
            if fs > 1e-10:
                f = f / fs
            else:
                f = best_f.copy()
            continue

        improved = False
        for _ in range(12):
            f_new = (1.0 - t) * f + t * g_dir
            f_new = np.clip(f_new, 0, None)
            fs_new = float(f_new.sum())
            if fs_new < 1e-10:
                t = max(t_min, t * 0.5)
                continue
            sc = _score(f_new)
            if sc < best_score - 1e-13:
                f = f_new
                best_f = f.copy()
                best_score = sc
                conv = _conv_full(f)
                current_max_b = float(conv.max())
                improved = True
                t = min(t_max, t * 1.06)
                stall = 0
                break
            else:
                t = max(t_min, t * 0.5)

        if not improved:
            stall += 1
            if stall >= max_stall:
                f = np.clip(f + rng.normal(0, 0.07, n), 0, None)
                fs = f.sum()
                if fs > 1e-10:
                    f = f / fs
                else:
                    f = best_f.copy()
                t = 0.2
                stall = 0
        iteration += 1

    return best_f, best_score


def _coord_polish(f, stop_time, max_sweeps=40, seed=777):
    """Coordinate descent with exact 1D optimization for each coordinate."""
    n = len(f)
    f = np.clip(f.astype(np.float64).copy(), 0, None)
    S = float(f.sum())
    if S < 0.01:
        return f, _score(f)
    conv = _conv_full(f)
    best_score = 2.0 * n * conv.max() / (S * S)
    rng = np.random.default_rng(seed)

    for sweep in range(max_sweeps):
        if time.monotonic() > stop_time - 5.0:
            break
        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > stop_time - 5.0:
                break

            v = f[i]
            S_rest = S - v

            # Build influence: infl[k] = 2*f[k-i] for k-i in [0,n-1]
            infl = np.zeros(2 * n - 1)
            start = i
            end = min(i + n, 2 * n - 1)
            infl[start:end] = 2.0 * f[:end - start]

            # Determine search range
            x_max = max(5.0 * v, 1.0) if v > 1e-8 else 2.0
            x_max = min(x_max, 200.0)

            # Coarse grid
            n_pts = 130
            x_grid = np.linspace(0.0, x_max, n_pts)

            best_x = v
            best_obj = best_score

            for x in x_grid:
                S_new = S_rest + x
                if S_new < 0.01:
                    continue
                w = x - v
                if abs(w) < 1e-13:
                    m_new = conv.max()
                else:
                    # new_conv[k] = conv[k] + w*infl[k] for k in [start, end)
                    # correction at k=2i: add w^2
                    tmp = conv.copy()
                    tmp[start:end] += w * infl[start:end]
                    if 0 <= 2 * i <= 2 * n - 2:
                        tmp[2 * i] += w * w
                    m_new = float(tmp.max())

                obj = 2.0 * n * m_new / (S_new * S_new)
                if obj < best_obj - 1e-15:
                    best_obj = obj
                    best_x = x

            # Fine grid around best_x
            if best_x != v:
                dw = x_max / n_pts
                x_fine = np.linspace(max(0, best_x - 3 * dw), min(x_max, best_x + 3 * dw), 60)
                for x in x_fine:
                    S_new = S_rest + x
                    if S_new < 0.01:
                        continue
                    w = x - v
                    if abs(w) < 1e-14:
                        continue
                    tmp = conv.copy()
                    tmp[start:end] += w * infl[start:end]
                    if 0 <= 2 * i <= 2 * n - 2:
                        tmp[2 * i] += w * w
                    m_new = float(tmp.max())
                    obj = 2.0 * n * m_new / (S_new * S_new)
                    if obj < best_obj - 1e-15:
                        best_obj = obj
                        best_x = x

            # Accept
            if best_x != v and best_obj < best_score - 1e-14:
                w = best_x - v
                f[i] = best_x
                S += w
                conv[start:end] += w * infl[start:end]
                if 0 <= 2 * i <= 2 * n - 2:
                    conv[2 * i] += w * w
                best_score = best_obj
                improved = True

        if not improved:
            break

    # Verify with full recompute
    f = np.clip(f, 0, 1000)
    final_score = _score(f)
    return f, final_score


def _pair_polish(f, stop_time, max_rounds=30, seed=9999):
    """Try 2-coordinate moves to escape 1D local optima."""
    n = len(f)
    f = f.copy()
    best_score = _score(f)
    rng = np.random.default_rng(seed)

    for rnd in range(max_rounds):
        if time.monotonic() > stop_time - 5.0:
            break
        improved = False
        for _ in range(80):
            if time.monotonic() > stop_time - 5.0:
                break
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            oi, oj = f[i], f[j]
            multipliers_i = (0.3, 0.6, 1.0, 1.5, 2.5, 0.0)
            multipliers_j = (0.3, 0.6, 1.0, 1.5, 2.5, 0.0)
            best_pair = (oi, oj)
            best_pair_sc = best_score
            for mi in multipliers_i:
                for mj in multipliers_j:
                    f[i] = oi * mi if oi > 1e-12 else (0.02 if mi > 0 else 0.0)
                    f[j] = oj * mj if oj > 1e-12 else (0.02 if mj > 0 else 0.0)
                    if f.sum() < 0.01:
                        f[i], f[j] = oi, oj
                        continue
                    sc = _score(f)
                    if sc < best_pair_sc - 1e-14:
                        best_pair_sc = sc
                        best_pair = (f[i], f[j])
            f[i], f[j] = best_pair
            if best_pair_sc < best_score - 1e-14:
                best_score = best_pair_sc
                improved = True

        if not improved:
            break

    return f, best_score
