```python
def build_constraint_matrix(f, n):
    """Build A where A[k,i] = f[k-i] if 0<=k-i<=n-1"""
    # A has shape (2n-1, n)
    # For row k, the non-zero entries are at columns i where 0<=k-i<=n-1
    # i.e., max(0, k-n+1) <= i <= min(n-1, k)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        j_vals = k - np.arange(i_min, i_max + 1)
        A[k, i_min:i_max+1] = f[j_vals]
    return A
```

```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _build_A(f, n):
    """Build constraint matrix A[k,i] = f[k-i] if 0<=k-i<n else 0. Shape (2n-1, n)."""
    k = np.arange(2 * n - 1)[:, None]
    i = np.arange(n)[None, :]
    j = k - i
    mask = (j >= 0) & (j < n)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    A[mask] = f[np.clip(j, 0, n - 1)[mask]]
    return A


def _conv_fft(a, n):
    fs = next_fast_len(2 * n - 1)
    F = rfft(a, fs)
    return irfft(F * F, fs)[:2 * n - 1]


def _score_normalized(a, n):
    """Score = 2n * max(a*a) assuming sum(a)=1."""
    b = _conv_fft(a, n)
    m = float(b.max())
    if not np.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m


def _lp_step(f, M, n, time_limit=2.0):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)_k <= M for all k. Returns (g, S_star)."""
    A = _build_A(f, n)
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c_obj = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    t_start = time.monotonic()
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                  options={'time_limit': time_limit, 'presolve': True})
    if not res.success or res.x is None:
        return None, 0.0
    g = np.maximum(res.x, 0.0)
    S_star = float(g.sum())
    return g, S_star

def _softmax(x):
    x = np.asarray(x, dtype=np.float64).ravel()
    x = x - x.max()
    e = np.exp(np.clip(x, -70.0, 70.0))
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0:
        return np.full(x.size, 1.0 / x.size)
    return e / s


def _cross_corr(p, a, n, fs2):
    """Compute cross-correlation (p * a_rev) at the relevant lags."""
    ar = a[::-1]
    c = irfft(rfft(ar, fs2) * rfft(p, fs2), fs2)
    return c[n - 1:2 * n - 1]


def _gradient_refine(a, n, end_time, seed=0):
    """Adam-based gradient refinement using soft-max approximation."""
    fs_c = next_fast_len(2 * n - 1)
    fs2 = next_fast_len(3 * n - 2)
    rng = np.random.default_rng(seed)
    
    z = np.log(np.maximum(a, 1e-300))
    z = z - z.mean()
    best_z = z.copy()
    best_score = _score_normalized(a, n)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t_count = 0
    lr = 0.3
    tau_frac = 0.4
    stuck = 0
    it = 0
    
    while time.monotonic() < end_time - 0.1:
        a = _softmax(np.clip(z, -25.0, 25.0))
        b = _conv_fft(a, n)
        bmax = float(b.max())
        if not np.isfinite(bmax) or bmax <= 0:
            break
        cur = 2.0 * n * bmax
        
        # Soft-max approximation of argmax
        tau = max(1e-10, tau_frac * bmax)
        bt = np.clip(b / tau, -70.0, 70.0)
        e = np.exp(bt)
        s_e = float(e.sum())
        if not np.isfinite(s_e) or s_e <= 0:
            p = np.zeros_like(b)
            p[int(np.argmax(b))] = 1.0
        else:
            p = e / s_e
        
        # Gradient via cross-correlation
        g = 2.0 * _cross_corr(p, a, n, fs2)
        ugrad = a * (g - float(np.dot(a, g)))
        
        gn = float(np.abs(ugrad).max())
        if not np.isfinite(gn) or gn < 1e-15:
            stuck += 1
            if stuck > 50:
                z = best_z + rng.normal(0, 0.5, n)
                z = z - z.mean()
                m, v = np.zeros(n), np.zeros(n)
                t_count = 0
                stuck = 0
            continue
        
        ugrad = ugrad / (gn + 1e-300)
        t_count += 1
        m = 0.9 * m + 0.1 * ugrad
        v = 0.999 * v + 0.001 * (ugrad * ugrad)
        mh = m / (1.0 - 0.9 ** t_count)
        vh = v / (1.0 - 0.999 ** t_count)
        upd = mh / (np.sqrt(vh) + 1e-8)
        um = float(np.abs(upd).max())
        if not np.isfinite(um) or um <= 0:
            continue
        upd = upd / (um + 1e-300)
        
        z = z - lr * upd
        z = z - z.mean()
        z = np.clip(z, -25.0, 25.0)
        z[~np.isfinite(z)] = 0.0
        
        a2 = _softmax(z)
        sc2 = _score_normalized(a2, n)
        
        if np.isfinite(sc2) and sc2 < best_score - 1e-12:
            best_score = sc2
            best_z = z.copy()
            stuck = 0
        else:
            stuck += 1
        
        it += 1
        if it % 15 == 0:
            tau_frac = max(0.05, tau_frac * 0.95)
        if stuck > 30:
            lr = max(0.05, lr * 0.8)
        if stuck > 120:
            z = best_z + rng.normal(0, 0.5, n)
            z = z - z.mean()
            m, v = np.zeros(n), np.zeros(n)
            t_count = 0
            lr = 0.3
            tau_frac = 0.35
            stuck = 0
    
    return _softmax(best_z), best_score


def _local_search(a, n, end_time, seed=12345):
    """Random local search with various move types."""
    fs_c = next_fast_len(2 * n - 1)
    rng = np.random.default_rng(seed)
    best = a.copy()
    bs = _score_normalized(best, n)
    sigma = 0.15
    stuck = 0
    
    while time.monotonic() < end_time - 0.1:
        r = float(rng.random())
        if r < 0.3:
            # Gaussian perturbation
            a2 = best + rng.normal(0, sigma, n)
        elif r < 0.5:
            # Block perturbation
            a2 = best.copy()
            i = int(rng.integers(0, max(1, n - 1)))
            w = int(rng.integers(1, max(2, n // 10)))
            i2 = min(n, i + w)
            a2[i:i2] *= np.exp(rng.normal(0, sigma, i2 - i))
        elif r < 0.7:
            # Scaling
            a2 = best * np.exp(float(rng.normal(0, sigma * 0.5)))
        elif r < 0.85:
            # Swap-like
            a2 = best.copy()
            i1 = int(rng.integers(0, n))
            i2 = int(rng.integers(0, n))
            w = min(3, n - max(i1, i2))
            a2[i1:i1+w], a2[i2:i2+w] = a2[i2:i2+w].copy(), a2[i1:i1+w].copy()
        else:
            #Mirror perturbation
            a2 = best.copy()
            a2 = a2 * (1.0 + sigma * rng.normal(0, 1, n))
        
        a2 = np.maximum(a2, 1e-12)
        s2 = float(a2.sum())
        if s2 < 1e-10:
            continue
        a2 = a2 / s2
        sc = _score_normalized(a2, n)
        
        if np.isfinite(sc) and sc < bs - 1e-12:
            best = a2.copy()
            bs = sc
            stuck = 0
            sigma = max(0.02, sigma * 0.95)
        else:
            stuck += 1
            sigma = min(0.5, sigma * 1.03)
            if stuck > 100:
                stuck = 0
                sigma = 0.3
    
    return best, bs


def _init_variant(n, variant, rng):
    t = (np.arange(n) + 0.5) / n
    x = 2.0 * t - 1.0
    v = variant % 12
    if v == 0:
        a = np.ones(n)
    elif v == 1:
        a = 1.0 - np.abs(x)
    elif v == 2:
        a = 1.0 - x * x
    elif v == 3:
        a = 0.5 * (1.0 + np.cos(np.pi * x))
    elif v == 4:
        a = (1.0 - np.abs(x)) ** 2
    elif v == 5:
        a = np.exp(-2.0 * x * x)
    elif v == 6:
        a = np.abs(np.sin(np.pi * t))
    elif v == 7:
        a = 0.5 + 0.5 * np.cos(2.0 * np.pi * t)
    elif v == 8:
        a = np.exp(-1.5 * np.abs(x))
    elif v == 9:
        a = 1.0 / (1.0 + np.exp(8.0 * x))
    elif v == 10:
        a = 1.0 + 0.3 * np.sin(2.0 * np.pi * 2 * t) + 0.2 * np.cos(2.0 * np.pi * 3 * t)
    else:
        a = np.abs(rng.normal(0, 1, n)) + 0.5
    a = np.maximum(a, 1e-6)
    a = a / a.sum()
    return a


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    total = max(1.0, float(budget_s))
    hard_deadline = t0 + total - 5.0
    
    # Get parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            if p.size > 0 and np.isfinite(p).all():
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        parent = None
    
    # Initialize with parent or fallback
    if parent is not None and parent.size > 0:
        best_a = parent.astype(np.float64).copy()
        best_a = np.maximum(best_a, 0.0)
        ps = float(best_a.sum())
        if ps < 1e-10:
            best_a = np.ones(192) / 192.0
            ps = 1.0
        best_a = best_a / ps
        n = best_a.size
        best_score = _score_normalized(best_a, n)
    else:
        n = 192
        best_a = np.ones(n) / n
        best_score = 2.0  # uniform gives score 2
    
    # Try to validate with evaluator
    try:
        fs = float(evaluate_sequence([float(v) for v in best_a]))
        if np.isfinite(fs) and fs < best_score:
            best_score = fs
    except Exception:
        pass
    
    rng_main = np.random.default_rng(seed)
    
    # Phase 1: LP-based alternating method on parent
    if time.monotonic() < hard_deadline - 60:
        lp_end1 = t0 + 0.45 * total
        try:
            a_lp, s_lp, n_it = _lp_alternating(best_a, n, lp_end1)
            if np.isfinite(s_lp) and s_lp < best_score:
                best_a = a_lp.copy()
                best_score = s_lp
        except Exception:
            pass
    
    # Phase 2: Try different sizes with LP method
    if time.monotonic() < hard_deadline - 40:
        sizes_to_try = [256, 384, 512]
        lp_end2 = t0 + 0.70 * total
        for sz in sizes_to_try:
            if time.monotonic() > lp_end2 - 5:
                break
            # Initialize by interpolating parent
            if parent is not None and parent.size > 1:
                idx = np.linspace(0, parent.size - 1, sz)
                a0 = np.interp(idx, np.arange(parent.size), parent)
            else:
                a0 = _init_variant(sz, 0, rng_main)
            a0 = np.maximum(a0, 1e-8)
            a0 = a0 / a0.sum()
            sc0 = _score_normalized(a0, sz)
            if sc0 >= best_score * 0.95:
                # Only try if it's competitive
                pass
            try:
                a_lp, s_lp, _ = _lp_alternating(a0, sz, lp_end2)
                if np.isfinite(s_lp) and s_lp < best_score:
                    best_a = a_lp.copy()
                    best_score = s_lp
                    n = sz
            except Exception:
                pass
    
    # Phase 3: Gradient refinement on best found
    if time.monotonic() < hard_deadline - 20:
        grad_end = t0 + 0.88 * total
        n_best = best_a.size
        try:
            a_gr, s_gr = _gradient_refine(best_a, n_best, grad_end, seed=seed + 100)
            if np.isfinite(s_gr) and s_gr < best_score:
                best_a = a_gr.copy()
                best_score = s_gr
        except Exception:
            pass
    
    # Phase 4: Local search
    if time.monotonic() < hard_deadline - 5:
        n_best = best_a.size
        try:
            a_ls, s_ls = _local_search(best_a, n_best, hard_deadline - 2, seed=seed + 200)
            if np.isfinite(s_ls) and s_ls < best_score:
                best_a = a_ls.copy()
                best_score = s_ls
        except Exception:
            pass
    
    # Final: one more LP pass if time allows
    if time.monotonic() < hard_deadline - 3:
        n_best = best_a.size
        try:
            a_lp2, s_lp2, _ = _lp_alternating(best_a, n_best, hard_deadline - 1)
            if np.isfinite(s_lp2) and s_lp2 < best_score:
                best_a = a_lp2.copy()
                best_score = s_lp2
        except Exception:
            pass
    
    # Final validation with exact evaluator
    best_a = np.asarray(best_a, dtype=np.float64).ravel()
    best_a = np.clip(best_a, 0.0, 1000.0)
    s_sum = float(best_a.sum())
    if s_sum < 0.01:
        best_a = np.ones(192) / 192.0
        s_sum = 1.0
    
    cand = [float(v) for v in best_a]
    try:
        fs = float(evaluate_sequence(cand))
        if not np.isfinite(fs):
            cand = [1.0]
            fs = float(evaluate_sequence(cand))
    except Exception:
        cand = [1.0]
    
    return cand
```
