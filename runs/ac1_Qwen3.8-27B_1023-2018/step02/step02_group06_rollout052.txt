    fs = next_fast_len(2 * n - 1)
    F = rfft(a, fs)
    c = irfft(F * F, fs)[:2 * n - 1]
    c[~np.isfinite(c)] = 0.0
    np.maximum(c, 0.0, out=c)
    return c

def _build_cross_conv_matrix(a):
    """Build matrix A such that A @ g = convolve(a, g) (full)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        k_lo = j
        k_hi = min(m - 1, j + n - 1)
        length = k_hi - k_lo + 1
        A[k_lo:k_hi + 1, j] = a[:length]
    return A

def _lp_direction(a, M, n, time_limit=3.0):
    """Solve LP: max sum(g) s.t. g>=0, conv(a,g)<=M. Return normalized g or None."""
    try:
        A = _build_cross_conv_matrix(a)
        c = -np.ones(n)
        b_ub = M * np.ones(2 * n - 1)
        bounds = [(0.0, None)] * n
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return None

def _score_fast(a):
    """Fast score: 2n * max(conv(a,a)) / sum(a)^2"""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    c = _fft_conv_self(a)
    return 2.0 * n * c.max() / (s * s)

def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s
