        if t_alloc4 > 15:
            a_main = smooth_opt(a_main, n_p, time.monotonic() + t_alloc4)
            ub(a_main)

    # Phase 5: Alternate LP + gradient
    for cycle in range(4):
        if time.monotonic() >= deadline - 25:
            break
        if linprog is not None:
            t_alloc = min(60.0, (deadline - time.monotonic() - 25) * 0.6)
            if t_alloc > 10:
                a_main = lp_opt(a_main, n_p, time.monotonic() + t_alloc, max_passes=100)
                ub(a_main)
        if time.monotonic() < deadline - 20:
            t_alloc_g = min(60.0, (deadline - time.monotonic() - 20) * 0.5)
            if t_alloc_g > 10:
                a_main = grad_opt(a_main, n_p, time.monotonic() + t_alloc_g, max_iters=30000)
                ub(a_main)

    # Phase 6: Explore nearby lengths
    for L in [n_p - 4, n_p - 2, n_p + 2, n_p + 4, n_p - 8, n_p + 8, 300, 340]:
        if time.monotonic() >= deadline - 20:
            break
        if L < 8 or L > 512 or L == n_p:
            continue
        t_rem = deadline - time.monotonic() - 20
        if t_rem < 12:
            break

        a_L = interp_len(a_main, L)
        if linprog is not None:
            t_alloc = min(30.0, t_rem * 0.4)
            if t_alloc > 8:
                a_L = lp_opt(a_L, L, time.monotonic() + t_alloc, max_passes=50)
                ub(a_L)
        if time.monotonic() < deadline - 15:
            a_L = grad_opt(a_L, L, min(deadline - 10, time.monotonic() + 40), max_iters=15000)
            ub(a_L)

    # Phase 7: Multiple restarts at parent length
    if time.monotonic() < deadline - 30 and linprog is not None:
        n_restarts = 10
        for r in range(n_restarts):
            if time.monotonic() >= deadline - 20:
                break
            t_rem_r = deadline - time.monotonic() - 20
            alloc_r = min(25.0, t_rem_r / max(1, n_restarts - r))
            if alloc_r < 5:
                break

            rtype = r % 5
            if rtype == 0:
                a_init = a_main * np.exp(0.12 * rng.standard_normal(n_p))
            elif rtype == 1:
                a_init = a_main[::-1].copy()
            elif rtype == 2:
                a_init = a_main * (1.0 + 0.2 * np.sin(2 * np.pi * rng.integers(2, 8) * np.arange(n_p) / n_p))
            elif rtype == 3:
                a_init = a_main * np.exp(0.25 * rng.standard_normal(n_p))
            else:
                a_init = a_main + 0.08 * rng.standard_normal(n_p)
            a_init = np.clip(a_init, 0, None)
            si = float(a_init.sum())
            if si > 1e-15:
                a_init /= si
                a_init = lp_opt(a_init, n_p, time.monotonic() + alloc_r * 0.6, max_passes=30)
                ub(a_init)
                if time.monotonic() < deadline - 12:
                    a_init = grad_opt(a_init, n_p, min(deadline - 10, time.monotonic() + alloc_r * 0.4),
                                      max_iters=8000)
                    ub(a_init)

    # Phase 8: Final gradient polish on best found
    if best_arr is not None and time.monotonic() < deadline - 5:
        n_b = best_arr.size
        a_pol = best_arr.copy()
        sp_ = a_pol.sum()
        if sp_ > 1e-15:
            a_pol /= sp_
            a_pol = grad_opt(a_pol, n_b, min(deadline - 2, time.monotonic() + 25), max_iters=20000)
            ub(a_pol)

    # === Final validation and return ===
    if best_arr is None:
        best_arr = np.ones(320, dtype=np.float64)
        ub(best_arr)

    final = np.clip(best_arr, 0.0, 1000.0)
    sf = float(final.sum())
    if sf < 0.01:
        final = np.ones(320, dtype=np.float64)
        ub(final)
        best_arr = final

    return [float(x) for x in best_arr]
