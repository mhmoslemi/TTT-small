import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, next_fast_len

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(191, dtype=np.float64) / 191.0
    p = np.clip(p, 0.0, 1000.0)
    if p.size == 0 or not np.isfinite(p).all():
        p = np.ones(191, dtype=np.float64) / 191.0
    return p

def _conv(a, b=None):
    """FFT-based convolution."""
    if b is None:
        b = a
    n = a.size
    m = b.size
    fsize = next_fast_len(n + m - 1)
    F = rfft(a, fsize)
    G = rfft(b, fsize)
    return irfft(F * G, fsize)[:n + m - 1]

def _score(a):
    """Compute exact score."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = a.size
    b = _conv(a, a)
    return 2.0 * n * b.max() / (s * s)

def _build_A(a):
    """Build (2n-1) x n matrix where A[k,j] = a[k-j] if valid."""
    n = a.size
    m = 2 * n - 1
    k = np.arange(m)[:, None]
    j = np.arange(n)[None, :]
    d = k - j
    mask = (d >= 0) & (d < n)
    A = np.where(mask, a[np.clip(d, 0, n-1)], 0.0)
    return A

def _solve_lp(a, M, timeout=1.0):
    """Solve LP: max sum(g) s.t. A@ g <= M, g >= 0."""
    n = a.size
    A = _build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n,
                     method='highs', options={'time_limit': timeout})
    except Exception:
        return None
    if res.status != 0:
        return None
    g = np.clip(res.x, 0, None)
    if g.sum() <= 0:

    return g

            # grad_z_j = sum_i grad_a_i * da_i/dz_j = sum_i grad_a_i * a_i * (delta_ij - a_j)
            # = a_j * (sum_i grad_a_i * a_i - sum_i grad_a_i * a_i * a_j)
            # = a_j * (sum_i grad_a_i * a_i - a_j * sum_i grad_a_i * a_i)
            # Hmm, let me just do it directly
            grad_z = a_soft * (grad_a - a_soft @ grad_a)
            
            return f, grad_z
        
        # Initial z
        z = np.log(a + 1e-15)
        
        try:
            res = minimize(obj_and_grad, z, method='L-BFGS-B',
                          options={'maxiter': 200, 'maxfun': 500,
                                   'ftol': 1e-15, 'gtol': 1e-12},
                          timeout=max(1, stop_time - time.monotonic() - 5))
        except Exception:
            break
        
        a_res = np.exp(res.x - res.x.max())
        a_res /= a_res.sum()
        a_res = np.clip(a_res, 1e-10, 1000)
        
        sc = _exact_score(a_res)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_res.copy()
        
        # Reduce epsilon
        eps *= 0.5  # or some other reduction
        
    return best_a
