import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len


def _softmax(x):
    x = np.asarray(x, dtype=np.float64).ravel()
    if x.size == 0:
        return np.array([1.0])
    xm = x - x.max()
    e = np.exp(np.clip(xm, -70.0, 70.0))
    s = float(e.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(x.size, 1.0 / x.size, dtype=np.float64)
    return e / s


def _conv_fft(a, n, fs):
    F = rfft(a, fs)
    return irfft(F * F, fs)[:2 * n - 1]


def _cross_corr(q, a, n, fs):
    ar = a[::-1]
    c = irfft(rfft(ar, fs) * rfft(q, fs), fs)
    return c[n - 1:2 * n - 1]


def _score_2n_maxb(a, n, fs):
    b = _conv_fft(a, n, fs)
    m = float(b.max())
    if not np.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m


def _gradient_mxb(a, n, fs2, tau):
    """Gradient of max(b) w.r.t. a using soft-max temperature tau.
    Returns grad of length n. grad[m] = 2 * sum_k p[k] * a[k-m] (bounded)."""
    b = _conv_fft(a, n, next_fast_len(2 * n - 1))
    bmax = float(b.max())
    if bmax <= 0:
        return np.zeros(n)
    bt = np.clip(b / max(tau, 1e-300), -60.0, 60.0)
    e = np.exp(bt)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0:
        p = np.zeros_like(b)
        p[int(np.argmax(b))] = 1.0
    else:
        p = e / s
    g = 2.0 * _cross_corr(p, a, n, fs2)
    return g, bmax

        if not np.isfinite(s_e) or s_e <= 0:

def _coord_descent(a_in, n, fs_c, time_budget, seed_cd):

    return best_a, best
