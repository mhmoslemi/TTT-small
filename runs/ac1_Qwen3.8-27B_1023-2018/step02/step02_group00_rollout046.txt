    # Random phase spectrum
    freqs = np.fft.rfftfreq(n)
    phases = rng.uniform(0, 2*np.pi, len(freqs))

    a = np.fft.irfft(F, n)

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _safe_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(320, dtype=np.float64)
    if p.size < 2:
        p = np.ones(320, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.sum() < 1e-10:
        p = np.ones(len(p) if len(p) > 0 else 320)
    return p


def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_conv_mat(f, n):
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_direction(f, n, M, tlimit=4.0):
    A = _build_conv_mat(f, n)
    c_vec = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_vec, A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': tlimit})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    np.maximum(g, 0.0, out=g)
    gs = float(g.sum())
    if gs < 1e-14:
        return None
    return g / gs


def _lp_loop(f0, n, deadline, seed, tlimit=4.0):
    rng = np.random.default_rng(seed)
    f = np.clip(f0.astype(np.float64).ravel(), 0, None).copy()
    if f.size < n:
        f = np.concatenate([f, np.zeros(n - f.size)])
    elif f.size > n:
        f = f[:n]
    fs = float(f.sum())
    if fs < 1e-12:
        f = np.ones(n, dtype=np.float64) / n
    else:
        f = f / fs

    best_f = f.copy()
    best_sc = _score(f)
    best_b = _conv_full(f)
    M = float(best_b.max())

    t = 0.30
    stall = 0
    max_stall = 60
    it = 0

    while time.monotonic() < deadline - 1.0:
        g = _lp_direction(f, n, M, tlimit)
        if g is None:
            stall += 6
            if stall > max_stall:
                break
            f = np.clip(f + rng.normal(0, 0.03, n), 0, None)
            fs2 = f.sum()
            f = f / fs2 if fs2 > 1e-12 else best_f.copy()
            continue

        dist = float(np.linalg.norm(g - f))
        if dist < 1e-9:
            stall += 3
            if stall > max_stall:
                break
            f = np.clip(f + rng.normal(0, 0.06, n), 0, None)
            fs2 = f.sum()
            f = f / fs2 if fs2 > 1e-12 else best_f.copy()
            continue

        improved = False
        for _ in range(14):
            f_new = np.clip((1.0 - t) * f + t * g, 0, None)
            sn = float(f_new.sum())
            if sn < 1e-12:
                t = max(0.015, t * 0.5)
                continue
            f_new = f_new / sn
            sc = _score(f_new)
            if sc < best_sc - 1e-13:
                f = f_new
                best_f = f.copy()
                best_sc = sc
                b_new = _conv_full(f)
                M = float(b_new.max())
                improved = True
                t = min(0.65, t * 1.04)
                stall = 0
                break
            else:
                t = max(0.015, t * 0.55)

        if not improved:
            stall += 1
            if stall >= max_stall:
                f = np.clip(f + rng.normal(0, 0.1, n), 0, None)
                fs2 = f.sum()
                f = f / fs2 if fs2 > 1e-12 else best_f.copy()
                t = 0.3
                stall = 0
        it += 1

    return best_f, best_sc


def _gradient_polish(a, deadline, seed=77):
    """Steepest-descent-like polish targeting peak lags."""
    rng = np.random.default_rng(seed)
    n = len(a)
    best = np.clip(a.copy(), 0, None)
    s = best.sum()
    best = best / s if s > 1e-12 else best
    best_sc = _score(best)
    no_imp = 0

    while time.monotonic() < deadline - 1.0 and no_imp < 8:
        b = _conv_full(best)
        M = float(b.max())
        # Find lags near the peak
        thresh = M * 0.95
        peak_lags = np.where(b >= thresh)[0]
        if len(peak_lags) < 1:
            peak_lags = np.array([np.argmax(b)])

        # Gradient of conv[k] w.r.t. a[i] = a[k-i] (for valid i)
        grad = np.zeros(n)
        for k in peak_lags:
            for i in range(max(0, k - n + 1), min(n, k + 1)):
                j = k - i
                if 0 <= j < n:
                    grad[i] += a[j] if i != j else 2.0 * a[j]
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-14:
            grad = grad / gnorm

        # Try decreasing peak-contributing elements
        improved_any = False
        for i in range(n):
            if time.monotonic() > deadline - 1.0:
                break
            if grad[i] < 5e-5:
                continue
            orig = best[i]
            for frac in (0.7, 0.5, 0.3, 0.1, 0.0):
                best[i] = orig * frac
                ss = best.sum()
                if ss < 0.01:
                    best[i] = orig
                    continue
                sc = _score(best)
                if sc < best_sc - 1e-13:
                    best_sc = sc
                    improved_any = True
                    break
            else:
                best[i] = orig

        # Also try 2-coordinate moves
        for _ in range(80):
            if time.monotonic() > deadline - 1.0:
                break
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            found = False
            for mi in (0.3, 0.5, 0.7, 1.3, 1.7):
                for mj in (0.3, 0.5, 0.7, 1.3, 1.7):
                    best[i] = oi * mi if oi > 1e-14 else 0.03
                    best[j] = oj * mj if oj > 1e-14 else 0.03
                    if best.sum() < 0.01:
                        best[i], best[j] = oi, oj
                        continue
                    sc = _score(best)
                    if sc < best_sc - 1e-13:
                        best_sc = sc
                        improved_any = True
                        found = True
                        break
                if found:
                    break
            if not found:
                best[i], best[j] = oi, oj

        if not improved_any:
            no_imp += 1
        else:
            no_imp = 0

    return best, best_sc


def _random_flat_init(n, seed):
    """Approximately flat-spectrum non-negative initialization."""
    rng = np.random.default_rng(seed)
    m = n // 2 + 1
    phases = rng.uniform(0, 2 * np.pi, m)
    F = np.exp(1j * phases)
    F[0] = float(m)
    a = np.fft.irfft(F, n)
    a = np.abs(a) + 0.005
    return a / a.sum()


def _blackman_harris(n):
    k = np.arange(n, dtype=np.float64)
    a0, a1, a2, a3 = 0.35875, 0.48829, 0.14128, 0.01168
    return a0 - a1 * np.cos(2 * np.pi * k / (n - 1)) + a2 * np.cos(4 * np.pi * k / (n - 1)) - a3 * np.cos(6 * np.pi * k / (n - 1))
