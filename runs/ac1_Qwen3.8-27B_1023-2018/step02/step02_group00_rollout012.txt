                stall = 0
                break
            else:
                t_step *= 0.5
        
        if not improved:
            stall += 1
            if stall >= 30:
                # Perturb and reset
                f = np.clip(f + rng.normal(0, 0.08, n), 0, None)

                t_step = 0.3
                stall = 0
            if stall > max_stall:
                break
    
    return best_f, best_score
