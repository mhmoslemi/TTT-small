    n = len(a)

    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    c[~np.isfinite(c)] = 0.0
    return c

def _score_fast(a):

def _normalize(a):

    S = a.sum()
    if S < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / S

def _build_C(a):
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        if j_lo <= j_hi:
            js = np.arange(j_lo, j_hi + 1)
            C[k, js] = a[k - js]
    return C

def _solve_lp(a, M, n, time_limit=10.0):
    C = _build_C(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:

                      bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})

            if s > 1e-12:

    except Exception:
        pass
    return None
