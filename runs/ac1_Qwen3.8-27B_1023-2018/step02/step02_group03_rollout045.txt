import scipy.sparse as sps

def propose_candidate(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    try:
        budget = max(10.0, float(budget_s))
    except Exception:
        budget = 1300.0
    deadline = t0 + budget - 10.0

    def sc(a):
        a = np.asarray(a, dtype=np.float64).ravel()
        a = np.clip(a, 0.0, None)
        n = a.size
        if n == 0:
            return float('inf')
        s = float(a.sum())
        if s < 1e-15:
            return float('inf')
        b = np.convolve(a, a)
        return 2.0 * n * float(b.max()) / (s * s)

    best_score = float('inf')
    best_arr = None

    def ub(a):
        nonlocal best_score, best_arr
        try:
            a = np.asarray(a, dtype=np.float64).ravel()
            a = np.clip(a, 0.0, 1000.0)
            if a.size == 0:
                return
            s = float(a.sum())
            if s < 0.01:
                return
            v = evaluate_sequence([float(x) for x in a])
            if math.isfinite(v) and v < best_score:
                best_score = v
                best_arr = a.copy()
        except Exception:
            pass

    # Get parent
    parent = None
    try:
        h = globals().get('height_sequence_1', None)
        if h is not None:
            parent = np.asarray(h, dtype=np.float64).ravel()
            parent = np.clip(parent, 0.0, 1000.0)
            if parent.size > 0:
                ub(parent)
    except Exception:
        pass

    if best_arr is None:
        best_arr = np.ones(96, dtype=np.float64)
        ub(best_arr)

    # === Build sparse convolution constraint matrix ===
    def build_conv_matrix(a):
        n = len(a)
        rows = []
        cols = []
        vals = []
        for k in range(2 * n - 1):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                rows.append(k)
                cols.append(j)
                vals.append(a[k - j])
        return sps.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

    # === LP optimization step ===
    def lp_step(a, n, M, margin=1.0):
        A = build_conv_matrix(a)
        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M * margin + 1e-12, dtype=np.float64)
        bounds = [(0.0, None)] * n
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 30})
        except Exception:
            return None
        if res is None or getattr(res, 'x', None) is None or not res.success:
            return None
        g = np.clip(res.x, 0, None)
        sg = float(g.sum())
        if sg < 1e-15:
            return None
        return g / sg

    # === LP-based optimization loop ===
    def lp_opt(a, n, max_iters, stop_time, margins_seq=None):
        a = np.asarray(a, dtype=np.float64).ravel().copy()
        a = np.clip(a, 0, None)
        s = a.sum()
        if s < 1e-15:
            return a
        a /= s
        local = sc(a)
        ub(a)
        stag = 0

        for it in range(max_iters):
            if time.monotonic() >= stop_time - 2:
                break
            b = np.convolve(a, a)
            M = float(b.max())
            if not math.isfinite(M) or M <= 0:
                break

            improved = False
            # Try multiple margins, starting from tightest
            margins = margins_seq if margins_seq else [1.0, 0.98, 0.95, 0.90]
            for m in margins:
                if time.monotonic() >= stop_time - 2:
                    break
                g = lp_step(a, n, M, m)
                if g is None:
                    continue
                # Line search
                best_sc_t, best_c_t = local, None
                for t in (0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0):
                    cand = (1.0 - t) * a + t * g
                    cand = np.clip(cand, 0, None)
                    cs = cand.sum()
                    if cs < 1e-15:
                        continue
                    v = sc(cand)
                    if v < best_sc_t:
                        best_sc_t, best_c_t = v, cand
                if best_c_t is not None and best_sc_t < local * (1.0 - 1e-10):
                    a = best_c_t / best_c_t.sum()
                    local = best_sc_t
                    stag = 0
                    improved = True
                    ub(a)
                    break

            if not improved:
                stag += 1
                if stag % 8 == 0:
                    # Perturbation
                    ap = a * np.exp(0.15 * rng.standard_normal(n))
                    ap = np.clip(ap, 0, None)
                    asp = ap.sum()
                    if asp > 1e-15:
                        ap /= asp
                        if sc(ap) < local:
                            a, local = ap, sc(ap)
                            stag = 0
                            ub(a)
                if stag > 30:
                    break
        return a

    # === Smooth-max gradient descent ===
    def smooth_conv_obj_and_grad(x, t_sm):
        """Objective: 2n * soft_max(conv(x,x)) / sum(x)^2, with gradient."""
        n = x.size
        b = np.convolve(x, x)
        s = float(x.sum())
        if s < 1e-15:
            return 1e10, np.zeros(n)
        # Soft max
        bm = b.max()
        exp_b = np.exp((b - bm) / t_sm)
        p = exp_b / exp_b.sum()
        soft_max_val = bm + t_sm * np.log(np.sum(exp_b))
        obj = 2.0 * n * soft_max_val / (s * s)
        # Gradient of soft_max w.r.t. x_j
        grad_sm = np.zeros(n, dtype=np.float64)
        for k in range(len(b)):
            if p[k] > 1e-12:
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                j = np.arange(lo, hi + 1)
                grad_sm[j] += 2.0 * p[k] * x[k - j]
        # Full gradient: d/dx[j] [2n * sm / s^2]
        # = (2n/s^2) * grad_sm[j] - (4n * sm / s^3)
        grad = (2.0 * n / (s * s)) * grad_sm - (4.0 * n * soft_max_val / (s * s * s))
        return obj, grad

    def smooth_grad_opt(a, n, stop_time, max_iters=5000):
        a = np.asarray(a, dtype=np.float64).ravel().copy()
        a = np.clip(a, 0, None)
        s = a.sum()
        if s < 1e-15:
            return a
        a /= s
        local = sc(a)
        ub(a)
        stale = 0

        for it in range(max_iters):
            if time.monotonic() >= stop_time - 2:
                break
            # Adaptive temperature
            t_sm = max(0.01, 0.1 * (1.0 - min(1.0, it / 200.0)) + 0.01)
            obj_val, grad = smooth_conv_obj_and_grad(a, t_sm)
            gn = float(np.linalg.norm(grad))
            if gn < 1e-14:
                break
            grad_n = grad / gn

            # Backtracking line search
            improved = False
            for alpha in (0.001, 0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2):
                cand = a - alpha * grad_n
                cand = np.clip(cand, 0, None)
                cs = cand.sum()
                if cs < 1e-15:
                    continue
                cand /= cs
                v = sc(cand)
                if v < local * (1.0 - 1e-11):
                    a = cand
                    local = v
                    improved = True
                    stale = 0
                    break
            if improved:
                ub(a)
            else:
                stale += 1
                if stale % 20 == 0:
                    ap = a + 0.03 * rng.standard_normal(n)
                    ap = np.clip(ap, 0, None)
                    asp = ap.sum()
                    if asp > 1e-15:
                        ap /= asp
                        if sc(ap) < local:
                            a, local = ap, sc(ap)
                            stale = 0
                            ub(a)
                if stale > 100:
                    break
        return a

    # === Interpolation for different lengths ===
    def interp_len(src, n):
        if src.size == n:
            return src.copy()
        ox = np.linspace(0.0, 1.0, src.size)
        nx = np.linspace(0.0, 1.0, n)
        out = np.interp(nx, ox, src)
        out = np.clip(out, 0, None)
        s = float(out.sum())
        if s > 1e-15:
            out /= s
        else:
            out = np.ones(n, dtype=np.float64) / n
        return out

    # === MAIN OPTIMIZATION ===
    p = parent if parent is not None else best_arr
    n_p = p.size
    p_norm = p / p.sum()

    # Phase 1: Intensive LP at parent length
    if n_p >= 4:
        alloc1 = min(500.0, (deadline - time.monotonic()) * 0.45)
        if alloc1 > 30:
            a1 = lp_opt(p_norm.copy(), n_p, 800, time.monotonic() + alloc1,
                       margins_seq=[1.0, 0.99, 0.97, 0.94, 0.90])
            ub(a1)

        # Gradient refinement
        if time.monotonic() < deadline - 60:
            a1 = smooth_grad_opt(a1, n_p, time.monotonic() + min(200, (deadline - time.monotonic()) * 0.15))
            ub(a1)

        # Second LP pass
        if time.monotonic() < deadline - 50:
            alloc2 = min(200.0, (deadline - time.monotonic()) * 0.15)
            if alloc2 > 20:
                a1 = lp_opt(a1, n_p, 500, time.monotonic() + alloc2,
                           margins_seq=[1.0, 0.995, 0.985, 0.97, 0.95])
                ub(a1)

        # Final gradient polish at parent length
        if time.monotonic() < deadline - 30:
            a1 = smooth_grad_opt(a1, n_p, deadline - 20)
            ub(a1)

    # Phase 2: Try nearby and other lengths
    lengths_to_try = []
    for L in [n_p - 8, n_p - 4, n_p - 2, n_p + 2, n_p + 4, n_p + 8,
              64, 80, 96, 128, 160, 192, 224, 256, 288, 320, 352, 384, 416, 448, 480]:
        if L >= 8 and L != n_p and L <= 512:
            lengths_to_try.append(L)
    lengths_to_try = sorted(set(lengths_to_try))

    for L in lengths_to_try:
        if time.monotonic() >= deadline - 40:
            break
        t_rem = deadline - time.monotonic() - 40
        if t_rem < 20:
            break
        alloc = min(50.0, t_rem * 0.25)
        if alloc < 10:
            break

        a_init = interp_len(p, L)
        # LP for this length
        a_w = lp_opt(a_init, L, max(20, int(alloc * 0.6)), time.monotonic() + alloc * 0.55,
                    margins_seq=[1.0, 0.97, 0.93])
        ub(a_w)
        # Gradient
        if time.monotonic() < deadline - 30:
            a_w = smooth_grad_opt(a_w, L, min(time.monotonic() + alloc * 0.4, deadline - 25))
            ub(a_w)

    # Phase 3: Multi-restart at best length
    if time.monotonic() < deadline - 40:
        n_b = best_arr.size
        b_norm = best_arr / best_arr.sum()
        n_restarts = 10
        for r in range(n_restarts):
            if time.monotonic() >= deadline - 30:
                break
            t_rem = deadline - time.monotonic() - 30
            alloc = min(40.0, t_rem / max(1.0, n_restarts - r))
            if alloc < 8:
                break

            init_type = r % 5
            if init_type == 0:
                a_init = b_norm * np.exp(0.2 * rng.standard_normal(n_b))
                a_init = np.clip(a_init, 0, None)
            elif init_type == 1:
                a_init = b_norm[::-1].copy()
            elif init_type == 2:
                a_init = b_norm * np.exp(0.4 * rng.standard_normal(n_b))
                a_init = np.clip(a_init, 0, None)
            elif init_type == 3:
                # Blend with uniform
                a_init = 0.7 * b_norm + 0.3 * np.ones(n_b) / n_b
            else:
                a_init = b_norm + 0.08 * rng.standard_normal(n_b)
                a_init = np.clip(a_init, 0, None)

            s_init = float(a_init.sum())
            if s_init > 1e-15:
                a_init /= s_init

            a_w = lp_opt(a_init, n_b, max(15, int(alloc * 0.5)),
                        time.monotonic() + alloc * 0.45, margins_seq=[1.0, 0.96, 0.92])
            ub(a_w)
            if time.monotonic() < deadline - 20:
                a_w = smooth_grad_opt(a_w, n_b, deadline - 15)
                ub(a_w)

    # Phase 4: Final gradient polish on best
    if time.monotonic() < deadline - 10 and best_arr is not None:
        n_b = best_arr.size
        a_pol = best_arr.copy() / best_arr.sum()
        a_pol = smooth_grad_opt(a_pol, n_b, deadline - 5)
        ub(a_pol)

    # Phase 5: One more LP polish on absolute best
    if time.monotonic() < deadline - 8 and best_arr is not None:
        n_b = best_arr.size
        a_pol = best_arr.copy() / best_arr.sum()
        a_pol = lp_opt(a_pol, n_b, 100, deadline - 4, margins_seq=[1.0, 0.999, 0.995])
        ub(a_pol)

    # === Return ===
    if best_arr is None:
        best_arr = np.ones(96, dtype=np.float64)
        ub(best_arr)

    final = np.clip(best_arr, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones_like(final, dtype=np.float64)
    ub(final)

    return [float(x) for x in best_arr]
