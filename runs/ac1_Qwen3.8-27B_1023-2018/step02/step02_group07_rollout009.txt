def _normalize_nonneg(a, positive=False):

def _softmax(z):
    z = np.asarray(z, dtype=np.float64).ravel()

                A[d - 1] = np.roll(a, d)

        else:
            return None

        c = -np.ones(n, dtype=np.float64)
        try:
            res = linprog(c, A_ub=A, b_ub=bub, bounds=[(0.0, None)] * n, method="highs")
        except Exception:
            try:

                return None
        if res is None or not getattr(res, "success", False):
            return None

        if x.size != n:
            return None

                    F0 = s2
                    accepted = True
                    stuck = 0
                    if s2 < best_s - 1e-12:
                        best_s = s2
                        best_a = a2.copy()
                        _try_exact_update(ctx, best_a)
                    break
            if not accepted:
                stuck += 1
                lr = max(0.008, lr * 0.85)
                if stuck > 30:
                    z = _z_of(best_a) + rng.normal(0.0, float(rng.uniform(0.25, 0.8)), n)
                    z = z - z.mean()
                    m[:] = 0.0
                    v[:] = 0.0
                    t = 0
                    lr = 0.04
                    stuck = 0

        _try_exact_update(ctx, best_a)
    except Exception:
        pass


def _infer_mode(pool_scores):
    dist = {"full": 0.0, "lin_off": 0.0, "cyc_off": 0.0}
    used = 0
    for a, ex in pool_scores:
        if not np.isfinite(ex):
            continue
        n = a.size
        if n == 0 or n > 2048:
            continue
        try:
            est = {
                "full": _score_direct(a, "full"),
                "lin_off": _score_direct(a, "lin_off"),
                "cyc_off": _score_direct(a, "cyc_off"),
            }
        except Exception:
            continue
        for k, e in est.items():
            if np.isfinite(e):
                dist[k] += abs(ex - e) / (1.0 + max(abs(ex), abs(e)))
        used += 1
    if used == 0:
        return "full"
    # If a full match is exact and no clearly sub-2 evidence exists, prefer full.
    if dist["full"] < 1e-8:
        return "full"
    return min(dist, key=dist.get)


def _try_exact_update(ctx, a):
    try:
        a = _normalize_nonneg(a, positive=False)
        if a.size == 0:
            return
        sc = _exact_score(a)
        if not np.isfinite(sc):
            return
        if sc < ctx["best_s"] - 1e-12:
            ctx["best_s"] = float(sc)
            ctx["best_a"] = a.copy()
        elif abs(sc - ctx["best_s"]) <= 1e-12:
            cur = ctx["best_a"]
            if cur is None or a.size > cur.size:
                ctx["best_a"] = a.copy()
    except Exception:
        pass


def _to_list(a):
    try:
        a = np.asarray(a, dtype=np.float64).ravel()
        if a.size == 0:
            return [1.0]
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if float(a.sum()) < 0.01:
            a = np.ones(a.size, dtype=np.float64)
        out = [float(x) for x in a]
        return out if out else [1.0]
    except Exception:
        return [1.0]


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    total = max(0.0, float(budget_s))
    hard = t0 + max(0.0, total) - 5.0
    if hard < t0 + 0.5:
        hard = t0 + 0.5

    try:
        seed_i = int(seed) & 0xFFFFFFFF
    except Exception:
        seed_i = 42
    rng = np.random.default_rng(seed_i + 1)

    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            arr = np.asarray(h, dtype=np.float64).ravel()
            if arr.size > 0 and np.any(np.isfinite(arr)):
                parent = _normalize_nonneg(arr, positive=False)
    except Exception:
        parent = None

    if parent is not None and parent.size > 0:
        search_n = parent.size if 2 <= parent.size <= 1024 else 192
    else:
        search_n = 192
    if search_n < 2:
        search_n = 192

    mode_n = search_n if 2 <= search_n <= 512 else 192

    pool = []
    seen = []

    def add_pool(a):
        try:
            a = _normalize_nonneg(a, positive=False)
            if a.size == 0:
                return
            if a.size <= 4096:
                for b in seen:
                    if b.size == a.size and np.allclose(a, b, rtol=1e-9, atol=1e-11):
                        return
            pool.append(a)
            if len(seen) < 80 and a.size <= 4096:
                seen.append(a.copy())
        except Exception:
            pass

    if parent is not None and parent.size > 0:
        add_pool(parent)
    add_pool(np.ones(search_n, dtype=np.float64))
    add_pool(np.ones(mode_n, dtype=np.float64))
    add_pool(np.ones(1, dtype=np.float64))

    if 2 <= mode_n <= 512:
        sp = np.zeros(mode_n, dtype=np.float64)
        sp[0] = 1.0
        add_pool(sp)

    if 2 <= mode_n <= 512:
        rr = np.random.default_rng((seed_i + 7919) & 0xFFFFFFFF)
        t = (np.arange(mode_n) + 0.5) / mode_n
        x = 2.0 * t - 1.0
        variants = [
            np.ones(mode_n),
            np.maximum(1.0 - np.abs(x), 1e-3),
            np.maximum(1.0 - x * x, 1e-3),
            0.5 + 0.5 * np.cos(2.0 * np.pi * t),
            0.5 + 0.5 * np.cos(4.0 * np.pi * t),
            0.5 + 0.5 * np.sin(2.0 * np.pi * t),
            np.exp(-2.0 * np.abs(x)),
            np.exp(-2.0 * t) + np.exp(-2.0 * (1.0 - t)),
        ]
        for vv, base in enumerate(variants):
            try:
                base = np.asarray(base, dtype=np.float64)
                base = base + 0.02 * np.abs(rr.normal(0.0, 1.0, mode_n))
                add_pool(base)
            except Exception:
                pass

    ctx = {"best_a": None, "best_s": math.inf}

    pool_scores = []
    for a in pool:
        sc = math.inf
        try:
            if a.size <= 4096:
                sc = _exact_score(a)
        except Exception:
            sc = math.inf
        pool_scores.append((a, sc))
        if np.isfinite(sc):
            _try_exact_update(ctx, a)

    if ctx["best_a"] is None or not np.isfinite(ctx["best_s"]):
        fallback = np.ones(mode_n, dtype=np.float64)
        sc = _exact_score(fallback)
        if np.isfinite(sc):
            ctx["best_a"] = fallback.copy()
            ctx["best_s"] = float(sc)
        else:
            ctx["best_a"] = fallback.copy()
            ctx["best_s"] = 2.0

    mode = _infer_mode(pool_scores)

    if ctx["best_s"] <= 1e-12:
        return _to_list(ctx["best_a"])

    if mode in ("lin_off", "cyc_off"):
        a_start = None
        s_min = math.inf
        for a, sc in pool_scores:
            if not np.isfinite(sc) or a.size < 2:
                continue
            try:
                ss = _score_direct(a, mode) if a.size <= 512 else _surrogate_score(a, mode)
            except Exception:
                ss = math.inf
            if np.isfinite(ss) and ss < s_min:
                s_min = ss
                a_start = a
        if a_start is None:
            a_start = ctx["best_a"]

        if a_start is not None and 2 <= a_start.size <= 1024:
            end_smooth = min(hard - 20.0, t0 + 0.45 * total)
            if time.monotonic() < end_smooth - 5.0:
                _smooth_run(_normalize_nonneg(a_start, positive=True), mode, end_smooth, rng, ctx)

        end_local = min(hard - 10.0, t0 + 0.78 * total)
        if time.monotonic() < end_local - 3.0:
            start_local = ctx["best_a"] if ctx["best_a"] is not None else a_start
            _local_search(start_local, ctx, end_local, rng, lambda x: _surrogate_score(x, mode))

        if time.monotonic() < hard - 6.0:
            best_a = ctx["best_a"]
            if best_a is not None:
                nbest = best_a.size
                if nbest <= 256:
                    limit = max(5.0, 0.20 * total)
                elif nbest <= 512:
                    limit = max(5.0, 0.10 * total)
                elif nbest <= 1024:
                    limit = max(3.0, min(30.0, 0.02 * total))
                else:
                    limit = 0.0
                if limit > 3.0:
                    end_exact_local = min(hard - 3.0, time.monotonic() + limit)
                    _local_search(best_a, ctx, end_exact_local, rng, _exact_score)
    elif mode == "full":
        # For the exact evaluator as stated, nonnegative sequences have minimum 2.
        # Do only light exact polishing to respect possible numerical variants.
        if time.monotonic() < hard - 6.0 and ctx["best_a"] is not None:
            nbest = ctx["best_a"].size
            limit = min(30.0, max(3.0, 0.05 * total)) if nbest <= 256 else 0.0
            if limit > 2.0:
                _local_search(ctx["best_a"], ctx, time.monotonic() + limit, rng, _exact_score)
    else:
        if ctx["best_a"] is not None and 2 <= ctx["best_a"].size <= 512:
            end_fd = min(hard - 45.0, t0 + 0.50 * total)
            if time.monotonic() < end_fd - 10.0:
                _generic_fd(_normalize_nonneg(ctx["best_a"], positive=True), ctx, end_fd, rng)

        if time.monotonic() < hard - 5.0 and ctx["best_a"] is not None:
            nbest = ctx["best_a"].size
            if nbest <= 256:
                limit = min(hard - 3.0, time.monotonic() + max(10.0, 0.35 * total))
            elif nbest <= 512:
                limit = min(hard - 3.0, time.monotonic() + max(5.0, 0.18 * total))
            elif nbest <= 1024:
                limit = min(hard - 3.0, time.monotonic() + max(3.0, min(30.0, 0.05 * total)))
            else:
                limit = time.monotonic()
            if time.monotonic() < limit:
                _local_search(ctx["best_a"], ctx, limit, rng, _exact_score)

    # Final exact fine-check with safe, inexpensive candidates.
    final_candidates = []
    if ctx["best_a"] is not None:
        final_candidates.append(ctx["best_a"])
    final_candidates.append(np.ones(search_n, dtype=np.float64))
    final_candidates.append(np.ones(1, dtype=np.float64))
    if parent is not None and parent.size <= 4096:
        final_candidates.append(parent)
    for a in final_candidates:
        _try_exact_update(ctx, a)

    cand = _to_list(ctx["best_a"] if ctx["best_a"] is not None else np.ones(1))
    try:
        if time.monotonic() < hard - 0.4:
            fs = float(evaluate_sequence(cand))
            if not np.isfinite(fs):
                cand = _to_list(np.ones(search_n, dtype=np.float64))
                fs = float(evaluate_sequence(cand))
                if not np.isfinite(fs):
                    cand = [1.0]
    except Exception:
        if cand is None or not cand:
            cand = [1.0]
    return cand
