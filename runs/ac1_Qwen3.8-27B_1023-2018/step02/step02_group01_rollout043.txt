def _convolve_square(a):
    n = a.size
    if n == 0:
        return np.array([0.0])
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    n = a.size
    b = _convolve_square(a)
    return 2.0 * n * float(b.max()) / (s * s)


def _build_A(a):
    n = a.size
    m = 2 * n - 1
    k_idx = np.arange(m, dtype=np.int64)[:, None]
    i_idx = np.arange(n, dtype=np.int64)[None, :]
    j = k_idx - i_idx
    mask = (j >= 0) & (j < n)
    safe_j = np.clip(j, 0, n - 1)
    A = np.where(mask, a[safe_j], 0.0)
    return A


def _lp_best_response(a, M, timeout_s=3.0):

    A = _build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0.0, None)] * n, method='highs',
                      options={'time_limit': timeout_s})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)

        return None

    n = a.size
    b = _conv_square(a)
    return 2.0 * n * float(b.max()) / (s * s)


def _build_T(a):
    n = a.size
    m = 2 * n - 1
    k = np.arange(m, dtype=np.int64)[:, None]
    i = np.arange(n, dtype=np.int64)[None, :]
    j = k - i
    mask = (j >= 0) & (j < n)
    safe_j = np.clip(j, 0, n - 1)
    T = np.where(mask, a[safe_j], 0.0)
    return T


def _lp_response(a, M, time_limit=2.0):
    n = a.size
    T = _build_T(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=T, b_ub=b_ub, bounds=[(0, None)] * n,
                      method='highs', options={'time_limit': time_limit,
                                               'primal_feasibility_tolerance': 1e-10,
                                               'dual_feasibility_tolerance': 1e-10})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    if not np.isfinite(g).all() or g.sum() < 1e-12:
        return None
    return g


def _lp_optimize(a_input, stop_time, max_iters=50000):
    """LP best-response iteration with adaptive stepping."""
    a = np.clip(a_input.astype(np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones_like(a)
    a /= s
    n = a.size

    best_a = a.copy()
    best_M = float(_conv_square(a).max())
    stuck = 0
    step_scale = 1.0

    for it in range(max_iters):
        if time.monotonic() > stop_time - 0.3:
            break

        M = float(_conv_square(a).max())
        if M < best_M - 1e-16:
            best_M = M
            best_a = a.copy()
            stuck = 0
            step_scale = 1.0
        else:
            stuck += 1
            step_scale = max(0.1, step_scale * 0.98)

        if stuck > 300:
            break

        t_lim = max(0.2, min(2.0, stop_time - time.monotonic() - 0.5))
        g = _lp_response(a, M, time_limit=t_lim)
        if g is None:
            break
        g /= g.sum()

        base_t = 0.3 * step_scale
        improved = False
        for t in (base_t, base_t * 0.5, base_t * 0.25, base_t * 0.1, 0.01, 0.005):
            cand = np.maximum((1.0 - t) * a + t * g, 0.0)
            cs = cand.sum()
            if cs < 1e-12:
                continue
            cand /= cs
            cM = float(_conv_square(cand).max())
            if cM < M - 1e-16:
                a = cand
                improved = True
                break

        if not improved and stuck > 150:
            break

    return best_a


def _gradient_polish(a_input, stop_time, max_iters=100000):
    """Polish using smooth log-sum-exp gradient with projected steps."""
    a = np.clip(a_input.astype(np.float64), 0, 1000).copy()
    a = a / a.sum()
    n = a.size

    best_a = a.copy()
    best_M = float(_conv_square(a).max())
    rng = np.random.default_rng(7777)
    stuck = 0

    beta = 30.0
    lr = 0.05

    fsize = next_fast_len(3 * n - 2)

    for it in range(max_iters):
        if time.monotonic() > stop_time - 0.3:
            break

        b = _conv_square(a)
        M = float(b.max())

        if M < best_M - 1e-15:
            best_M = M
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        if stuck > 2000:
            a = a * np.exp(rng.normal(0, 0.15, n))
            a = np.maximum(a, 1e-10)
            a /= a.sum()
            stuck = 0
            beta = max(15.0, beta * 0.7)
            lr = max(lr, 0.02)
            continue

        # Compute gradient of LSE(beta * b)
        shifted = beta * (b - M)
        exp_b = np.exp(shifted)
        weights = exp_b / exp_b.sum()

        # grad[i] = 2*beta * sum_{j=0}^{n-1} weights[i+j] * a[j]
        # = 2*beta * (fa * weights)[i + n - 1] where fa = a[::-1]
        fa = a[::-1]
        Fa = rfft(fa, fsize)
        Fw = rfft(weights, fsize)
        conv = irfft(Fa * Fw, fsize)
        grad = 2.0 * beta * conv[n - 1:n - 1 + n]

        # Projected gradient step
        candidate = a - lr * grad
        candidate = np.maximum(candidate, 0.0)
        cs = candidate.sum()
        if cs > 1e-12:
            candidate /= cs
            cM = float(_conv_square(candidate).max())
            if cM < M - 1e-17:
                a = candidate
                lr = min(lr * 1.01, 0.1)
            else:
                lr *= 0.9
                if lr < 1e-7:
                    lr = 1e-7
        else:
            lr *= 0.8

        # Slowly increase beta
        if it % 500 == 499:
            beta = min(beta * 1.03, 150.0)

    return best_a


def _polish_fine_lp(a_input, stop_time, max_iters=50000):
    """Very fine LP polishing with minimal steps."""
    a = np.clip(a_input.astype(np.float64), 0, 1000).copy()
    a = a / a.sum()
    n = a.size

    best_a = a.copy()
    best_M = float(_conv_square(a).max())
    rng = np.random.default_rng(31415)
    stuck = 0

    for it in range(max_iters):
        if time.monotonic() > stop_time - 0.2:
            break

        M = float(_conv_square(a).max())
        if M < best_M - 1e-17:
            best_M = M
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        if stuck > 2000:
            a = a * np.exp(rng.normal(0, 0.03, n))
            a = np.maximum(a, 1e-12)
            a /= a.sum()
            stuck = 0
            continue

        t = max(0.001, 0.05 * (0.997 ** (stuck // 5)))
        t_lim = max(0.15, min(1.0, stop_time - time.monotonic() - 0.3))
        g = _lp_response(a, M * (1 + 1e-14), time_limit=t_lim)
        if g is None:
            break
        g /= g.sum()

        cand = np.maximum((1 - t) * a + t * g, 0.0)
        cs = cand.sum()
        if cs > 1e-12:
            cand /= cs
            cM = float(_conv_square(cand).max())
            if cM < M - 1e-18:
                a = cand
            else:
                for t2 in (t * 0.3, t * 0.1, t * 0.03):
                    cand2 = np.maximum((1 - t2) * a + t2 * g, 0.0)
                    cs2 = cand2.sum()
                    if cs2 > 1e-12:
                        cand2 /= cs2
                        cM2 = float(_conv_square(cand2).max())
                        if cM2 < M - 1e-18:
                            a = cand2
                            break

    return best_a


def _resample(a, n_new):
    """Resample sequence to new length using linear interpolation."""
    n_old = a.size
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    result = np.interp(x_new, x_old, a)
    return np.maximum(result, 0.0)


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + budget_s - 12.0

    parent = _get_parent()
    n_p = parent.size
    rng = np.random.default_rng(seed)

    # Initialize best
    best_seq = parent.copy()
    best_score = float(evaluate_sequence([float(x) for x in parent]))
    if not np.isfinite(best_score):
        best_seq = np.ones(n_p) / n_p
        best_score = 2.0

    # === Phase 1: LP optimization at parent length ===
    p1_end = min(t0 + budget_s * 0.18, t0 + 200)
    res = _lp_optimize(best_seq, stop_time=p1_end)
    sc = _score_of(res)
    if np.isfinite(sc) and sc < best_score:
        best_seq, best_score = res.copy(), sc

    # === Phase 2: Gradient polish ===
    p2_end = min(t0 + budget_s * 0.30, t0 + 380)
    if time.monotonic() < p2_end - 3:
        res = _gradient_polish(best_seq, stop_time=p2_end)
        sc = _score_of(res)
        if np.isfinite(sc) and sc < best_score:
            best_seq, best_score = res.copy(), sc

    # === Phase 3: Fine LP polish at parent length ===
    p3_end = min(t0 + budget_s * 0.42, t0 + 550)
    if time.monotonic() < p3_end - 3:
        res = _polish_fine_lp(best_seq, stop_time=p3_end)
        sc = _score_of(res)
        if np.isfinite(sc) and sc < best_score:
            best_seq, best_score = res.copy(), sc

    # === Phase 4: Try different lengths (key insight: shorter n reduces 2n factor) ===
    # Priority: n_p-1 first since it gives ~0.5% just from the factor
    length_order = [n_p - 1, n_p - 2, n_p + 1, n_p - 3, n_p + 2, n_p - 4, n_p + 3]

    for nl in length_order:
        if time.monotonic() > deadline - 25:
            break
        if nl < 20:
            continue
        alloc = min(35.0, deadline - time.monotonic() - 15)
        if alloc < 8:
            break

        init = _resample(best_seq, nl)
        s = init.sum()
        if s > 1e-12:
            init /= s
        else:
            init = np.ones(nl) / nl

        try:
            res = _lp_optimize(init, stop_time=time.monotonic() + alloc)
            sc = _score_of(res)
            if np.isfinite(sc) and sc < best_score:
                best_seq, best_score = res.copy(), sc
        except Exception:
            pass

    # === Phase 5: Gradient + fine LP polish on best found ===
    if time.monotonic() < deadline - 30:
        alloc = min(60.0, deadline - time.monotonic() - 20)
        if alloc > 10:
            res = _gradient_polish(best_seq, stop_time=time.monotonic() + alloc)
            sc = _score_of(res)
            if np.isfinite(sc) and sc < best_score:
                best_seq, best_score = res.copy(), sc

    p5_end = deadline - 20
    if time.monotonic() < p5_end - 5:
        res = _polish_fine_lp(best_seq, stop_time=p5_end)
        sc = _score_of(res)
        if np.isfinite(sc) and sc < best_score:
            best_seq, best_score = res.copy(), sc

    # === Phase 6: If time remains, try a couple more lengths ===
    if time.monotonic() < deadline - 20:
        for nl in [n_p - 5, n_p + 4, n_p - 6]:
            if time.monotonic() > deadline - 18:
                break
            if nl < 20:
                continue
            alloc = min(20.0, deadline - time.monotonic() - 12)
            if alloc < 5:
                break
            init = _resample(best_seq, nl)
            s = init.sum()
            if s > 1e-12:
                init /= s
            else:
                init = np.ones(nl) / nl
            try:
                res = _lp_optimize(init, stop_time=time.monotonic() + alloc)
                sc = _score_of(res)
                if np.isfinite(sc) and sc < best_score:
                    best_seq, best_score = res.copy(), sc
            except Exception:
                pass

    # Final validation with the exact evaluator
    try:
        final_score = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_score = float('inf')

    if not np.isfinite(final_score) or final_score > 2.0:
        # Fallback: return the parent or a safe default
        try:
            parent_score = float(evaluate_sequence([float(x) for x in parent]))
            if np.isfinite(parent_score):
                return [float(x) for x in parent]
        except Exception:
            pass
        return [1.0] * max(2, n_p)

    return [float(x) for x in best_seq]
