            total_iters += 1

    # =========================================================
    # PHASE 3: Stochastic perturbation (wide search)
    # =========================================================
    if time.monotonic() < deadline - 60.0:
        a = best_f.copy()
        sc = best_sc
        rng2 = np.random.default_rng(seed + 104729)
        sigma = 0.06

        for _ in range(200000):
            if time.monotonic() > deadline - 60.0:
                break

            r = rng2.random()
            if r < 0.30:
                z = np.log(np.clip(a, 1e-12, None))
                z += rng2.normal(0.0, sigma, n)
                z -= z.max()
                cand = np.exp(z)
                cand /= cand.sum()
            elif r < 0.50:
                cand = a.copy()
                k = int(rng2.integers(0, n))
                width = max(1, n // 30)
                lo = max(0, k - width)
                hi = min(n, k + width)
                cand[lo:hi] *= float(np.exp(rng2.normal(0, 0.18)))
                cand = np.maximum(cand, 0.0)
                cand /= cand.sum()
            elif r < 0.65:
                cand = a.copy()
                k1 = int(rng2.integers(0, n))
                k2 = int(rng2.integers(0, n))
                if k1 != k2:
                    amt = 0.008 * float(a.max())
                    cand[k1] += amt
                    cand[k2] = max(cand[k2] - amt, 0.0)
                    cand /= cand.sum()
            elif r < 0.80:
                # Blend with reversed
                w = 0.05 + 0.15 * rng2.random()
                cand = (1 - w) * a + w * a[::-1]
                cand /= cand.sum()
            elif r < 0.90:
                # Blend with parent
                w = 0.02 + 0.1 * rng2.random()
                cand = (1 - w) * a + w * (parent / parent.sum())
                cand /= cand.sum()
            else:
                # Larger random jump
                z = np.log(np.clip(a, 1e-12, None))
                z += rng2.normal(0.0, 0.4, n)
                z -= z.max()
                cand = np.exp(z)
                cand /= cand.sum()

            cand_sc = 2.0 * n * float(do_conv(np.maximum(cand, 0.0)).max())
            if cand_sc < sc - 1e-14:
                a = cand.copy()
                sc = cand_sc
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_f = a.copy()
                sigma = max(0.008, sigma * 0.97)
            else:
                sigma = min(0.6, sigma * 1.05)

    # =========================================================
    # PHASE 4: Final LP polish
    # =========================================================
    if time.monotonic() < deadline - 20.0:
        f = best_f.copy()
        for _ in range(200):
            if time.monotonic() > deadline - 20.0:
                break
            new_f, new_M = lp_step(f)
            if new_f is None:
                break
            f = new_f
            sc = 2.0 * n * new_M
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_f = f.copy()

    # =========================================================
    # PHASE 5: Final gradient polish
    # =========================================================
    if time.monotonic() < deadline - 12.0:
        a = best_f.copy()
        lr = 0.1
        for it in range(5000):
            if time.monotonic() > deadline - 12.0:
                break
            c = do_conv(a)
            M = float(c.max())
            sc = 2.0 * n * M
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_f = a.copy()

            tau = max(1e-5, 0.005 * M * np.exp(-it / 2000.0))
            x = np.clip((c - M) / tau, -40.0, 40.0)
            ex = np.exp(x)
            s_ex = float(ex.sum())
            if s_ex < 1e-30:
                p = np.zeros(2 * n - 1)
                p[int(np.argmax(c))] = 1.0
            else:
                p = ex / s_ex

            grad = fft_gradient(a, p)
            step = grad - float(np.dot(grad, a))
            sn = float(np.linalg.norm(step))
            if sn < 1e-14:
                break
            step /= sn

            improved = False
            for trial in range(6):
                scale = lr * (0.5 ** trial)
                cand = np.maximum(a - scale * step, 0.0)
                cs = float(cand.sum())
                if cs < 1e-8:
                    continue
                cand /= cs
                cs_sc = 2.0 * n * float(do_conv(cand).max())
                if cs_sc < sc - 1e-15:
                    a = cand
                    improved = True
                    break
            if not improved:
                lr = max(0.001, lr * 0.7)

    # =========================================================
    # Final validation with exact evaluator
    # =========================================================
    best_f = np.clip(best_f, 0.0, 1000.0)
    if not np.isfinite(best_f).all() or best_f.sum() < 0.01:
        best_f = np.ones(n, dtype=np.float64) / n

    final_list = [float(x) for x in best_f]
    try:
        fs = float(evaluate_sequence(final_list))
    except Exception:
        fs = float('inf')

    if not np.isfinite(fs):
        final_list = [1.0 / n] * n
        try:
            fs = float(evaluate_sequence(final_list))
        except Exception:
            pass
        if not np.isfinite(fs):
            return [1.0]

    return final_list
