    for size in sizes_to_try:
        n = size
        fs = next_fast_len(2 * n - 1)
        fs2 = next_fast_len(3 * n - 2)
        
        # Primary job: from parent (if matching) or uniform
        if parent is not None and parent.size == n:
            a0 = parent.copy()
            a0 = a0 / max(1e-300, float(a0.sum()))
        else:
            a0 = _init_diverse(n, 0, np.random.default_rng(seed + 100))
        z0 = np.log(np.clip(a0, 1e-30, 1000.0))
        z0 = z0 - z0.mean()
        jobs.append(('adam', z0, n, fs, fs2, seed + 200))
        
        # Secondary jobs with different inits
        for vi in range(1, 4):
            a0 = _init_diverse(n, vi, rng_master, parent)
            z0 = np.log(np.clip(a0, 1e-30, 1000.0))
            z0 = z0 - z0.mean()
            jobs.append(('adam', z0, n, fs, fs2, seed + 300 + vi * 100))
    
    # Allocate time: 70% for Adam jobs, 15% for L-BFGS polish, 15% for local search
    explore_end = t0 + 0.65 * (total - 15.0)
    lbfgs_end = t0 + 0.85 * (total - 15.0)
    
    def _try_update(a, s, n):
        nonlocal best_a, best_score
        if a is None or not np.isfinite(s):
            return
        try:
            a = np.asarray(a, dtype=np.float64).ravel()
            if a.size == 0 or not np.isfinite(a).all():
                return
            a = np.clip(a, 0.0, 1000.0)
            if float(a.sum()) < 0.01:
                return
        except Exception:
            return
        if s < best_score - 1e-15:
            best_a = a.copy()
            best_score = float(s)
    
    # Phase 1: Run Adam jobs
    try:
        # Run the parent-seeded job first (most likely to improve)
        parent_done = False
        for ji in range(len(jobs)):
            if time.monotonic() >= explore_end:
                break
            job = jobs[ji]
            if job[0] != 'adam':
                continue
            _, z0, n, fs, fs2, js = job
            end_time = min(explore_end, time.monotonic() + 60.0)
            try:
                a, s = _run_adam(z0, n, fs, fs2, end_time, hard_deadline, js)
                _try_update(a, s, n)
            except Exception:
                pass
            if parent is not None and n == parent.size and not parent_done:
                parent_done = True
    except Exception:
        pass
    
    # Phase 2: L-BFGS polish on best candidate
    try:
        if time.monotonic() < lbfgs_end - 5.0:
            a_best = best_a.copy()
            n_best = a_best.size
            a_best = a_best / max(1e-300, float(a_best.sum()))
            fs_best = next_fast_len(2 * n_best - 1)
            fs2_best = next_fast_len(3 * n_best - 2)
            
            end_time = min(lbfgs_end, hard_deadline)
            a, s = _run_lbfgs(a_best, n_best, fs_best, fs2_best, end_time, hard_deadline)
            _try_update(a, s, n_best)
    except Exception:
        pass
    
    # Phase 3: Local search polish
    try:
        if time.monotonic() < hard_deadline - 2.0:
            a_best = best_a.copy()
            n_best = a_best.size
            a_best = a_best / max(1e-300, float(a_best.sum()))
            fs_best = next_fast_len(2 * n_best - 1)
            fs2_best = next_fast_len(3 * n_best - 2)
            
            a, s = _local_search(a_best, n_best, fs_best, fs2_best, hard_deadline, hard_deadline, seed + 999)
            _try_update(a, s, n_best)
    except Exception:
        pass
    
    # Final validation
    try:
        cand = [float(v) for v in best_a]
        if len(cand) == 0:
            cand = [1.0]
    except Exception:
        cand = [1.0]
    
    try:
        fs = float(evaluate_sequence(cand))
    except Exception:
        fs = math.inf
    
    if not math.isfinite(fs):
        cand = [1.0]
        try:
            fs = float(evaluate_sequence(cand))
        except Exception:
            pass
        if not math.isfinite(fs):
            return [1.0]
    
    return cand
