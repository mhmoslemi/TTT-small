import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog
from scipy.linalg import toeplitz


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(320, dtype=np.float64)
    if p.size == 0:
        p = np.ones(320, dtype=np.float64)

    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(320, dtype=np.float64)
    return p


_FFT_SIZE = None

def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    return float(2.0 * n * m / (s * s))


def _build_A(f, n):
    c = np.zeros(2 * n - 1)
    c[:n] = f
    r = np.zeros(n)
    r[0] = f[0]
    return toeplitz(c, r)


def _lp_direction(f, n, M, time_limit=5.0):
    A = _build_A(f, n)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
    except Exception:
        return None
    if res.status != 0:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    gs = float(g.sum())
    if gs < 1e-15:
        return None
    return g / gs

    def h(t):
        u = (1 - t) * f + t * g
        return _score(u)
    
    fc = h(c_)
    fd = h(d_)
    
    for _ in range(max_iter):
        if fc < fd:
            b = d_
            d_ = c_
            fd = fc
            c_ = b - inv_phi * (b - a)
            fc = h(c_)
        else:
            a = c_
            c_ = d_
            fc = fd
            d_ = a + inv_phi * (b - a)
            fd = h(d_)
    
    return (a + b) / 2.0

def _polish(a, stop_time, seed=4242):
    n = len(a)
    best = np.clip(np.asarray(a, dtype=np.float64).ravel(), 0, None).copy()
    fs = float(best.sum())

        return best, math.inf
    best = best / fs
    best_sc = _score(best)

    no_imp = 0
    max_no_imp = 10

    while time.monotonic() < stop_time - 3.0:
        improved = False

        # Single coordinate
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 3.0:
                break
            orig = best[i]
            muls = (0.1, 0.3, 0.5, 0.7, 0.9, 1.1, 1.3, 1.6, 2.0, 3.0, 0.0)
            bval = orig
            bsc = best_sc
            for m in muls:
                best[i] = orig * m if orig > 1e-14 else (0.01 if m > 0 else 0.0)
                sc = _score(best)
                if sc < bsc - 1e-14:
                    bval = best[i]
                    bsc = sc
            best[i] = bval
            if bsc < best_sc - 1e-14:
                best_sc = bsc
                improved = True

        # 2-coordinate
        for _ in range(60):
            if time.monotonic() > stop_time - 3.0:
                break
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            for mi in (0.4, 0.7, 1.4, 2.0, 0.0):
                for mj in (0.4, 0.7, 1.4, 2.0, 0.0):
                    best[i] = oi * mi if oi > 1e-14 else (0.01 if mi > 0 else 0.0)
                    best[j] = oj * mj if oj > 1e-14 else (0.01 if mj > 0 else 0.0)
                    sc = _score(best)
                    if sc < best_sc - 1e-14:
                        best_sc = sc
                        improved = True
            best[i], best[j] = oi, oj

        if not improved:
            no_imp += 1
            if no_imp >= max_no_imp:
                break
        else:
            no_imp = 0

    return best, best_sc

        return [float(x) for x in fallback]

    if final_sc < best_score:
        best_score = final_sc

    return [float(x) for x in best_seq]
