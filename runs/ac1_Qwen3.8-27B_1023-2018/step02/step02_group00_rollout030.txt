import time
import math
from scipy.fft import rfft, irfft, next_fast_len

def _fft_conv(a, b):
    n = len(a)
    m = len(b)
    fsize = next_fast_len(n + m - 1)
    A = rfft(a, fsize)
    B = rfft(b, fsize)
    return irfft(A * B, fsize)[:n + m - 1]

def _simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv[-1] - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    tau = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - tau, 0)

def _score_scaled(a):
    # a is normalized (sum = 1)
    n = len(a)
    c = _fft_conv(a, a)
    return 2.0 * n * float(c.max())

def _evaluate(a_raw):
    # Use the authoritative evaluator
    return float(evaluate_sequence([float(x) for x in a_raw]))

def _grad_smooth(a, s, c):
    # Compute gradient of smooth approximation
    # w = softmax(s * c), grad = 2 * convolve(w, a)
    c_clipped = s * c
    c_clipped -= c_clipped.max()  # for numerical stability
    w = np.exp(c_clipped)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    grad = 2.0 * _fft_conv(w, a)
    # Truncate to length of a
    return grad[:len(a)]

import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _fft_conv(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    fsize = next_fast_len(n + m - 1)
    A = rfft(a, fsize)
    B = rfft(b, fsize)
    return irfft(A * B, fsize)[:n + m - 1]

def _fft_autocorr(a):
    """Full autocorrelation via FFT (a convolved with itself)."""
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    A = rfft(a, fsize)
    return irfft(A * A, fsize)[:2 * n - 1]

def _simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cssv[-1] - 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    tau = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - tau, 0.0)

def _score_from_conv(a, c):
    """Compute score given normalized a (sum=1) and its autocorrelation c."""
    n = len(a)
    return 2.0 * n * float(c.max())

def _compute_grad(a, c, s):
    """Compute gradient of smooth approximation f_s = (1/s)*log(sum exp(s*c_k))."""
    n = len(a)
    # Softmax weights
    sc = s * c
    sc -= sc.max()
    w = np.exp(sc)
    wsum = w.sum()
    if wsum < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= wsum
    
    # grad[j] = 2 * sum_k w[k] * a[k-j] (with a out-of-range = 0)
    # = 2 * (w * B_rev)[2n-2 + j] where B[k] = a[k] for k<n, 0 else
    B = np.zeros(2 * n - 1)
    B[:n] = a
    B_rev = B[::-1]
    conv = _fft_conv(w, B_rev)
    grad = 2.0 * conv[2*n-2 : 3*n-2]
    return grad

    n = len(a)

def _autocorr(a):
    n = len(a)
    fs = _fftsz(2 * n - 1)
    A = rfft(a, fs)
    c = irfft(A * A, fs)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)  # numerical safety

def _simplex_project(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    ones = np.arange(1, n + 1, dtype=np.float64)

    rho = np.nonzero(cond)[0]
    if len(rho) == 0:

    r = rho[-1]
    tau = (cssv[r] - 1.0) / (r + 1.0)

            s_sm = max(5.0, s_sm * 0.5)
            m_active = max(3, m_active - 1)
            stall = 0
            max_stall = min(max_stall + 100, 3000)
            iteration += 1
            continue
        
        grad = _grad_topm(a, c, m_active, s_sm)
        
        # Nesterov-like momentum
        v = beta * v + grad
        
        # Backtracking line search
        improved = False
        for _ in range(30):
            a_new = _simplex_project(a - alpha * v)
            c_new = _autocorr(a_new)
            sc_new = 2.0 * n * float(c_new.max())
            if sc_new < sc - 1e-14:
                a = a_new
                c = c_new
                max_c = sc_new / (2.0 * n)
                sc = sc_new
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            # Small random perturbation
            a = _simplex_project(a + 0.01 * rng.standard_normal(n))
        
        # Schedule
        s_sm = min(s_sm * 1.001, 50.0)
        m_active = min(int(m_active + 0.01), 10)
        # beta = max(0.85, beta - 0.00005)
        
        # Reset alpha if it got too small
        if alpha < 0.001:
            alpha = 0.5
            v = np.zeros(n)
        
        iteration += 1
    
    return best_a, best_sc

def _lp_min_peak(a, M_current):
    """LP: find direction g that minimizes max_k (a*g)_k.
    Returns normalized g or None."""
    n = len(a)
    # Build A: A[k, j] = a[k] for row k = j+j' (j' from 0 to n-1)
    # Actually: (a * g)_k = sum_j a[k-j] * g[j]
    # A[k, j] = a[k-j] if 0 <= k-j < n else 0
    # Column j: rows j to j+n-1 contain a[0] to a[n-1]
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    # A@g gives the full convolution of a and g
    
    # LP: minimize M' subject to A@g - M' <= 0, sum(g)=1, g>=0
    # Variables: [g_0, ..., g_{n-1}, M']
    # Objective: [0, ..., 0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality: A@g - M' <= 0 → [A, -1] @ [g; M'] <= 0
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5.0})
    except Exception:
        return None
    
    if res.status != 0 or res.x is None:
        return None
    g = res.x[:n]
    g = np.clip(g, 0, None)
    gs = g.sum()
    if gs < 1e-15:
        return None
    g /= gs
    return g

def _lp_refine(a, stop_time, seed):
    """Iterative LP-based refinement."""
    rng = np.random.default_rng(seed)
    n = len(a)
    a = np.clip(a.astype(np.float64).ravel().copy(), 0, None)
    sv = a.sum()
    if sv < 1e-10:
        a = np.ones(n) / n
    else:
        a /= sv
    
    best_a = a.copy()
    best_sc = _score_norm(a)
    t = 0.3  # step size for blend
    stall = 0
    max_stall = 100
    
    while time.monotonic() < stop_time - 1.0:
        c = _autocorr(a)
        M = float(c.max())
        
        g = _lp_min_peak(a, M)
        if g is None:
            stall += 5
            if stall > max_stall:
                break
            a = _simplex_project(a + 0.03 * rng.standard_normal(n))
            continue
        
        dist = float(np.linalg.norm(g - a))
        if dist < 1e-8:
            stall += 3
            if stall > max_stall:
                break
            a = _simplex_project(a + 0.05 * rng.standard_normal(n))
            continue
        
        # Line search
        improved = False
        for _ in range(15):
            a_new = a + t * (g - a)  # = (1-t)*a + t*g
            a_new = _simplex_project(a_new)
            sc_new = _score_norm(a_new)
            if sc_new < best_sc - 1e-14:
                a = a_new
                best_a = a.copy()
                best_sc = sc_new
                improved = True
                t = min(0.6, t * 1.05)
                stall = 0
                break
            else:
                t *= 0.6
        
        if not improved:
            t = max(0.05, t * 0.5)
            stall += 1
            if stall >= max_stall:
                a = _simplex_project(a + 0.08 * rng.standard_normal(n))
                best_from_here = _score_norm(a)
                if best_from_here < best_sc:
                    best_sc = best_from_here
                    best_a = a.copy()
                t = 0.3
                stall = 0
                max_stall = min(max_stall + 50, 500)
    
    return best_a, best_sc

def _polish_coarse(a, stop_time):
    """2-coordinate swap polish."""
    rng = np.random.default_rng(4242)
    n = len(a)
    best = np.clip(a.astype(np.float64).ravel().copy(), 0, None)
    sv = best.sum()
    if sv < 1e-10:
        return best, _score_norm(best)
    best /= sv
    best_sc = _score_norm(best)
    no_imp = 0
    
    while time.monotonic() < stop_time - 1.0 and no_imp < 8:
        no_imp += 1
        improved = False
        
        # Single-coordinate moves
        for i in range(n):
            if time.monotonic() > stop_time - 1.0:
                    break
            orig = best[i]
            for mlt in (0.3, 0.6, 0.8, 1.2, 1.5, 2.0):
                best[i] = orig * mlt if orig > 1e-12 else 0.01
                sv = best.sum()
                if sv < 0.01:
                    best[i] = orig
                    continue
                a_t = best / sv
                sc = _score_norm(a_t)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    improved = True
                    break
            best[i] = orig
            if not improved:
                # try the best multiplier
                best_vals = []
                for mlt in (0.3, 0.5, 0.7, 0.9, 1.1, 1.3, 1.6, 2.0):
                    best[i] = orig * mlt if orig > 1e-12 else 0.01
                    sv = best.sum()
                    if sv < 0.01:
                        continue
                    a_t = best / sv
                    sc = _score_norm(a_t)
                    best_vals.append((sc, mlt))
                if best_vals:
                    best_sc_mv, best_mlt = min(best_vals, key=lambda x: x[0])
                    if best_sc_mv < best_sc - 1e-14:
                        best[i] = orig * best_mlt if orig > 1e-12
