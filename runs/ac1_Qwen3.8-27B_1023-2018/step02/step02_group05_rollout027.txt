import math

def _score_direct(a):
    """Fast score computation."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return float('inf')
    s = float(a.sum())
    if s < 1e-15:
        return float('inf')
    b = np.convolve(a, a)
    mb = float(b.max())
    return 2.0 * n * mb / (s * s)

def _subgradient(a, eps_rel=0.01):
    """Compute subgradient of max(convolve(a,a)) at a."""
    n = a.size
    c = np.convolve(a, a)
    M = c.max()
    if M <= 0:
        return None, 0.0
    active = np.where(c >= M * (1 - eps_rel))[0]
    if len(active) == 0:
        active = np.where(c >= M - 1e-10)[0]
    # g_i = 2/|S| * sum_{k in S, 0<=k-i<n} a[k-i]
    g = np.zeros(n)
    for k in active:
        k = int(k)
        for j in range(n):
            if 0 <= k - j <= n - 1:
                g[j] += 2.0 * a[k - j]
    g /= len(active)
    return g, M

def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64).ravel().copy()
    n = v.size
    if n == 0:
        return v
    # Find the projection onto the probability simplex
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    s = w.sum()
    if s > 1e-15:
        w /= s
    else:
        w = np.ones(n) / n
    return w
