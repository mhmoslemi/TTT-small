    if a is None or a.size < 2:
        a = np.ones(192, dtype=np.float64)
    a = norm(a)
    ubd(a)

    n = a.size

    def build_L(f, nn):
        ks = np.arange(2 * nn - 1)
        js = np.arange(nn)
        diff = ks[:, None] - js[None, :]
        mask = (diff >= 0) & (diff < nn)
        L = np.zeros((2 * nn - 1, nn), dtype=np.float64)
        L[mask] = f[diff[mask]]
        return L

    def lp_dir(f, nn):
        b = np.convolve(f, f)
        M = float(b.max())
        if M < 1e-300 or not math.isfinite(M):
            return None
        L = build_L(f, nn)
        c = -np.ones(nn)
        ub = M * (1.0 + 1e-9) + 1e-12
        try:
            res = linprog(c, A_ub=L, b_ub=np.full(2 * nn - 1, ub),
                          bounds=[(0.0, None)] * nn, method='highs',
                          options={'time_limit': 4, 'presolve': True})
            if res.success and res.x is not None:
                g = np.clip(np.nan_to_num(res.x, nan=0.0, posinf=0.0, neginf=0.0), 0.0, None)
                sg = float(g.sum())
                if sg > 1e-300:
                    return g / sg
        except Exception:
            pass
        return None

    def line_search(f, g, cur_sc):
        bs, bt, bf = cur_sc, 0.0, None
        for t in (0.15, 0.3, 0.5, 0.7, 1.0):
            cand = norm((1.0 - t) * f + t * g)
            cs = sc(cand)
            if cs < bs:
                bs, bt, bf = cs, t, cand
        if bf is not None:
            for t in (bt * 0.5, bt * 1.3, bt * 2.0, bt * 0.5, bt * 0.7):
                t = min(max(t, 0.01), 1.0)
                cand = norm((1.0 - t) * f + t * g)
                cs = sc(cand)
                if cs < bs:
                    bs, bt, bf = cs, t, cand
        return bf, bs if bf is not None else cur_sc

    # Phase 1: Main LP iteration
    cur = a.copy()
    cur_score = sc(cur)
    stg = 0
    lp_count = 0

    while time.monotonic() < deadline - 50:
        if stg > 150:
            break

        g = lp_dir(cur, n)
        lp_count += 1

        if g is None:
            stg += 1
            if stg % 15 == 0:
                p = norm(cur * np.exp(0.05 * rng.standard_normal(n)))
                ps = sc(p)
                if ps < cur_score:
                    cur, cur_score, stg = p, ps, 0
                    ubd(cur)
            continue

        # Check if direction is meaningful
        diff_norm = np.linalg.norm(g - cur / cur.sum())
        if diff_norm < 1e-8:
            stg += 1
            continue

        bf, bs = line_search(cur, g, cur_score)
        if bf is not None and bs < cur_score * (1.0 - 1e-11):
            cur, cur_score = bf, bs
            stg = 0
            ubd(cur)
        else:
            stg += 1
            if stg % 10 == 0:
                p = cur.copy()
                idx = rng.choice(n, size=n // 4, replace=False)
                p[idx] *= np.exp(0.08 * rng.standard_normal(len(idx)))
                p = norm(np.clip(p, 0, None))
                ps = sc(p)
                if ps < cur_score:
                    cur, cur_score, stg = p, ps, 0
                    ubd(cur)

    # Phase 2: Gradient refinement
    if time.monotonic() < deadline - 35:
        g_stg = 0
        for _ in range(20000):
            if time.monotonic() > deadline - 35:
                break
            if g_stg > 300:
                break
            b = np.convolve(cur, cur)
            S = float(cur.sum())
            M = float(b.max())
            ks = int(np.argmax(b))
            j_arr = ks - np.arange(n)
            valid = (j_arr >= 0) & (j_arr < n)
            direction = np.zeros(n)
            direction[valid] = cur[j_arr[valid]] * S - 2.0 * M
            # For non-zero: direction[i] = cur[ks-i]*S; for zero: -2M
            direction[~valid] = -2.0 * M
            alpha = 0.3 * S / (np.max(np.abs(direction)) + 1e-10)
            new_a = np.clip(cur - alpha * direction, 0.0, None)
            new_a = norm(new_a)
            ns = sc(new_a)
            if ns < cur_score * (1.0 - 1e-12):
                cur, cur_score = new_a, ns
                g_stg = 0
                ubd(cur)
            else:
                g_stg += 1
                if g_stg % 20 == 0:
                    p = norm(cur * np.exp(0.02 * rng.standard_normal(n)))
                    ps = sc(p)
                    if ps < cur_score:
                        cur, cur_score, g_stg = p, ps, 0
                        ubd(cur)

    # Phase 3: Another LP pass on the refined sequence
    if time.monotonic() < deadline - 30:
        cur = best[1].copy() if best[1] is not None else cur
        cur_score = sc(cur)
        stg = 0
        for _ in range(10000):
            if time.monotonic() > deadline - 30:
                break
            if stg > 80:
                break
            g = lp_dir(cur, n)
            if g is None:
                stg += 1
                if stg % 10 == 0:
                    p = norm(cur * np.exp(0.03 * rng.standard_normal(n)))
                    ps = sc(p)
                    if ps < cur_score:
                        cur, cur_score, stg = p, ps, 0
                        ubd(cur)
                continue
            bf, bs = line_search(cur, g, cur_score)
            if bf is not None and bs < cur_score * (1.0 - 1e-11):
                cur, cur_score = bf, bs
                stg = 0
                ubd(cur)
            else:
                stg += 1

    # Phase 4: Try adjacent lengths with interpolation
    if time.monotonic() < deadline - 20:
        for nn in (n - 16, n + 16, n - 32, n + 32):
            if nn < 16 or nn > 512:
                continue
            if time.monotonic() > deadline - 20:
                break
            src = best[1] if best[1] is not None else a
            old_x = np.arange(src.size, dtype=np.float64)
            new_x = np.linspace(0.0, src.size - 1, nn)
            ac = np.interp(new_x, old_x, src)
            ac = norm(ac * nn)
            if sc(ac) > 2.5:
                continue
            # Quick LP refinement at new length
            s_score = sc(ac)
            for _ in range(80):
                if time.monotonic() > deadline - 20:
                    break
                g = lp_dir(ac, nn)
                if g is None:
                    break
                bf, bs = line_search(ac, g, s_score)
                if bf is not None and bs < s_score * (1.0 - 1e-11):
                    ac, s_score = bf, bs
                else:
                    break
            ubd(ac)

    # Final LP polish on best
    if time.monotonic() < deadline - 10 and best[1] is not None:
        bseq = best[1].copy()
        bn = bseq.size
        bscore = sc(bseq)
        stg = 0
        for _ in range(2000):
            if time.monotonic() > deadline - 10:
                break
            if stg > 30:
                break
            g = lp_dir(bseq, bn)
            if g is None:
                stg += 1
                continue
            bf, bs = line_search(bseq, g, bscore)
            if bf is not None and bs < bscore * (1.0 - 1e-11):
                bseq, bscore = bf, bs
                stg = 0
            else:
                stg += 1
        ubd(bseq)

    # Final validation and return
    if best[1] is None:
        best[1] = np.ones(n, dtype=np.float64)
    final = np.asarray(best[1], dtype=np.float64).ravel().copy()
    final = np.nan_to_num(final, nan=0.0, posinf=0.0, neginf=0.0)
    final = np.clip(final, 0.0, 1000.0)
    if final.size < 1 or final.sum() < 0.01:
        final = np.ones(n, dtype=np.float64)
    v = evaluate_sequence([float(x) for x in final])
    if not math.isfinite(v):
        final = np.ones(n, dtype=np.float64)
        best[1] = final
        best[0] = evaluate_sequence([float(x) for x in final])
    return [float(x) for x in best[1]]
