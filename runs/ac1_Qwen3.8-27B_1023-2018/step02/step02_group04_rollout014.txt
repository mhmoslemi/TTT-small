def smooth_grad_scoring(a, beta):
    """Compute smooth score and gradient for normalized a (sum=1)."""
    n = len(a)
    b = np.convolve(a, a)
    # softmax
    max_b = b.max()
    exp_b = np.exp(beta * (b - max_b))
    w = exp_b / exp_b.sum()
    smooth_score = 2*n * (max_b + np.log(exp_b.sum()) / beta)
    # gradient: grad_j = 2 * sum_i w_{i+j} * a_i
    grad = np.zeros(n)
    for j in range(n):
        # sum over i=0..n-1 of a[i] * w[i+j]
        # i+j ranges from j to j+n-1
        grad[j] = 2.0 * np.dot(a, w[j:j+n])
    # Scale by 2n
    grad *= 2*n
    # Project onto sum=0 hyperplane
    grad -= grad.mean()
    return smooth_score, grad
