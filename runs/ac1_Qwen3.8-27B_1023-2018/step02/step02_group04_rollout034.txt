import time
import math
from scipy.optimize import linprog

def _score(a):
    """Compute the AC1 score."""
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 0.0, 1000.0)
    n = a.size
    s = float(a.sum())
    if s < 1e-12:
        return float('inf')
    b = np.convolve(a, a)
    return float(2.0 * n * float(b.max()) / (s * s))

def _normalize(a):
    """Normalize to sum = 1."""
    s = float(a.sum())
    if s < 1e-12:
        return np.ones_like(a) / a.size
    return a / s

def _simplex_project(v):
    """Project v onto the simplex {x: sum(x)=1, x>=0}."""
    n = v.size
    if n == 1:
        return np.array([1.0])
    # Algorithm from duchi et al. 2008
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.argmax(u * np.arange(1, n+1) - (cssv - 1) > 0)
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _conv_matrix(f, n):
    """Build the convolution matrix A where A @ g = convolve(f, g)."""
    A = np.zeros((2*n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j+n, j] = f
    return A

def _lp_direction(f, M):
    """Solve the LP to find direction g."""
    n = f.size
    A = np.zeros((2*n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j+n, j] = f
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(2*n - 1, M, dtype=np.float64)
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0.0, None)]*n,
                    method='highs', options={'presolve': True, 'time_limit': 10})
        if r.success and r.x is not None:
            g = np.clip(r.x, 0.0, None)
            if g.sum() > 1e-12:
                return g / g.sum()
    except Exception:
        pass
    return None

def _gradient_descent(a, budget_end, best, rng):
    """Projected gradient descent on the simplex."""
    a = _normalize(a.copy())
    n = a.size
    current_score = _score(a)
    
    for it in range(100000):
        if time.monotonic() > budget_end:
            break
        
        b = np.convolve(a, a)
        k_star = int(np.argmax(b))
        
        # Compute gradient of C[k_star] w.r.t. a
        grad = np.zeros(n, dtype=np.float64)
        for j in range(n):
            k_j = k_star - j
            if 0 <= k_j < n:
                grad[j] = 2.0 * a[k_j]
        
        # Projected gradient step
        # Try multiple step sizes
        best_step = None
        best_sc = current_score
        for alpha in np.logspace(-4, 0, 20):
            v = a - alpha * grad
            proj = _simplex_project(v)
            sc = _score(proj)
            if sc < best_sc:
                best_sc = sc
                best_step = proj
        
        if best_step is not None:
            a = best_step
            current_score = best_sc
            # Update best
            if current_score < best[0]:
                best[0] = current_score
                best[1] = a.copy()
        else:
            # Add noise and try again
            if rng.random() < 0.1:
                a = _normalize(a + 0.01 * rng.standard_normal(n))
            else:
                break
    
    return a

The issue is that `C[k*]` is a quadratic form `a^T Q_{k*} a`, and its gradient is `2 Q_{k*} a` (if `Q` is symmetric). Let me verify:

`C[k] = sum_{i} a[i] * a[k-i]` where the sum is over `i` such that `0 <= i < n` and `0 <= k-i < n`.

Let's fix `j` and ask: what is `dC[k]/da[j]`?

`a[j]` appears in the sum in the following ways:
1. As `a[i]` with `i = j`: this term is `a[j] * a[k-j]`, contributed when `0 <= j < n` and `0 <= k-j < n`. Derivative w.r.t. `a[j]` is `a[k-j]`.
2. As `a[k-i]` with `k-i = j`, i.e., `i = k-j`: this term is `a[k-j] * a[j]`, contributed when `0 <= k-j < n` and `0 <= j < n`. Derivative w.r.t. `a[j]` is `a[k-j]`.

So if `0 <= j < n` and `0 <= k-j < n`:
`dC[k]/da[j] = a[k-j] + a[k-j] = 2*a[k-j]`.

But wait, if `j = k-j` (i.e., `2j = k`), then cases 1 and 2 are the same term `a[j]^2`, and the derivative is `2*a[j]`, which equals `2*a[k-j]` since `k-j = j`. So the formula `2*a[k-j]` works in all cases.

If either condition fails, `a[j]` doesn't appear in the sum, so `dC[k]/da[j] = 0`.

Great, so the gradient is:
