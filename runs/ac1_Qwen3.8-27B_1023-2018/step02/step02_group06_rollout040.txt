def _conv_self_fft(a):
    """Compute full convolution a*a using FFT."""
    n = len(a)
    fsize = next_fast_len(2*n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2*n-1]
    # Clean up small negatives from numerical error
    np.maximum(c, 0.0, out=c)
    return c

    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return math.inf
    a_norm = a / s
    b = _conv_self_fft(a_norm)
    return 2.0 * n * b.max()

def project_simplex(v):
    """Project v onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    # Algorithm from Wang & Carreira-Perpinan (2013)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

    c = irfft(F * F, fsize)[:2*n-1]
    np.maximum(c, 0.0, out=c)
    c[~np.isfinite(c)] = 0.0
    return c

def _score(a):
    """Compute score = 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return math.inf
    b = _conv_self(a)
    return 2.0 * n * b.max() / (s * s)

def _build_A(a):
    """A[k,j] = a[k-j] if 0<=k-j<n else 0. Shape: (2n-1, n)."""
    n = len(a)
    k = np.arange(2*n-1)
    j = np.arange(n)
    diff = k[:, None] - j[None, :]  # (2n-1, n)
    A = np.zeros((2*n-1, n))
    mask = (diff >= 0) & (diff < n)
    A[mask] = a[diff[mask].astype(np.int64)]
    return A

def _solve_lp_direction(a, M, timeout_s=10):
    """Solve max sum(g) s.t. A(a)@g <= M, g>=0. Return g or None."""
    n = len(a)
    A = _build_A(a)
    
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    
    for solver in [cp.CVXOPT, cp.ECOS, cp.SCS, cp.PDLP, cp.CBC]:
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.asarray(g.value).ravel()
                if np.isfinite(gv).all():
                    gv = np.maximum(gv, 0.0)
                    if gv.sum() > 1e-12:
                        return gv
        except Exception:
            continue
    return None

def _subgradient(a, k_star):
    """Subgradient of max conv at coordinate a, using argmax k_star."""
    n = len(a)
    grad = np.zeros(n)
    # grad[m] = 2*a[k_star-m] if 0<=k_star-m<n else 0
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    if lo <= hi:
        grad[lo:hi+1] = 2.0 * a[k_star - hi:k_star - lo + 1][::-1]
    return grad

def _project_simplex(v):
    """Projection onto {x>=0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n+1)
    cond = u * idx > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    ws = w.sum()
    if ws < 1e-15:
        return np.ones(n) / n
    return w / ws

def _lp_optimize(a_init, deadline, max_iters=300):
    """LP-based optimization. Returns (best_a, best_score)."""
    n = len(a_init)
    a = np.maximum(a_init, 0.0)
    a = _project_simplex(a)
    
    best_a = a.copy()
    best_score = _score(a)
    best_max_b = float(_conv_self(a).max())
    
    no_improve = 0
    max_no_improve = 50
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = _conv_self(a)
        M = float(b.max())
        s = a.sum()
        cur_score = 2.0 * n * M / (s*s)
        
        if cur_score < best_score - 1e-10:
            best_score = cur_score
            best_a = a.copy()
            best_max_b = M
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > max_no_improve:
            break
        
        # Solve LP direction
        g = _solve_lp_direction(a, M)
        if g is None:
            break
        
        g = g / g.sum()
        
        # Check direction quality
        cos_sim = float(np.dot(a, g) / (np.linalg.norm(a) * np.linalg.norm(g) + 1e-30))
        if cos_sim > 0.999:
            no_improve += 5
            continue
        
        # Line search
        best_t = 0.0
        best_mb = M
        for t in np.linspace(0.05, 1.0, 12):
            a_cand = (1-t)*a + t*g
            a_cand = np.maximum(a_cand, 0)
            sc = a_cand.sum()
            if sc < 1e-15:
                continue
            a_cand = a_cand / sc
            b_cand = _conv_self(a_cand)
            mb = float(b_cand.max())
            if mb < best_mb:
                best_mb = mb
                best_t = t
        
        if best_t > 0:
            a = (1-best_t)*a + best_t*g
            a = np.maximum(a, 0)
            a = _project_simplex(a)
        else:
            no_improve += 1
    
    return best_a, best_score

def _subgradient_optimize(a_init, deadline, max_iters=5000):
    """Subgradient descent for local polishing."""
    n = len(a_init)
    a = _project_simplex(np.maximum(a_init, 0.0))
    
    best_a = a.copy()
    best_score = _score(a)
    
    step_size = 0.02
    best_step = step_size
    patience = 0
    max_patience = 200
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = _conv_self(a)
        M = float(b.max())
        k_star = int(np.argmax(b))
        s = a.sum()
        cur_score = 2.0 * n * M / (s*s)
        
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        if patience > max_patience:
            break
        
        grad = _subgradient(a, k_star)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        
        a_new = a - step_size * grad / gn
        a_new = _project_simplex(a_new)
        
        b_new = _conv_self(a_new)
        M_new = float(b_new.max())
        
        if M_new < M * (1 - 1e-12):
            a = a_new
            step_size = best_step = min(0.05, best_step * 1.1)
        else:
            step_size *= 0.9
            if step_size < 1e-6:
                step_size = best_step * 0.5
                # Try a different k_star (second worst)
                b_sorted = np.sort(b)[::-1]
                if len(b_sorted) > 1 and b_sorted[1] > 0.99 * b_sorted[0]:
                    # Multiple near-active constraints
                    k_star2 = int(np.argsort(b)[::-1][1])
                    grad2 = _subgradient(a, k_star2)
                    grad = (grad + grad2) / 2.0
                    gn = np.linalg.norm(grad)
                    if gn > 1e-15:
                        a_new = a - step_size * grad / gn
                        a_new = _project_simplex(a_new)
                        b_new2 = _conv_self(a_new)
                        if b_new2.max() < M:
                            a = a_new
    
    return best_a, best_score

    # Power of sine (controls edge steepness)
    for p in [0.5, 1.0, 2.0, 3.0]:
        inits.append(np.abs(np.sin(np.pi * i / n)) ** p + 1e-8)

    # Parabolic taper
    inits.append(1.0 - ((i / (n - 1)) - 0.5) ** 2 * 0.8)

    # Smooth random
    for _ in range(4):
        x = rng.uniform(0, 1, n)
        # Simple box smoothing
        kernel = np.ones(max(3, n // 15)) / max(3, n // 15)
        x = np.convolve(x, kernel, mode='same')
        inits.append(np.maximum(x, 0.01))

    # Double bump
    c1, c2 = n * 0.3, n * 0.7
    inits.append(np.exp(-((i - c1) ** 2) / (2 * (n / 8) ** 2)) +
                 np.exp(-((i - c2) ** 2) / (2 * (n / 8) ** 2)))

    return inits

def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(1.0, float(budget_s))
    deadline = t0 + budget - 20.0

    parent = _get_parent()
    n = len(parent)

    try:
        parent_score = float(evaluate_sequence([float(x) for x in parent]))
    except Exception:
        parent_score = math.inf

    best_seq = np.maximum(parent, 0.0).astype(np.float64).copy()
    best_score = parent_score

    if not np.isfinite(best_score):
        best_seq = np.ones(n, dtype=np.float64)
        best_score = 2.0

    rng = np.random.default_rng(seed)

    # Phase 1: LP from parent (main optimization)
    p1_dl = t0 + budget * 0.45
    if time.monotonic() < p1_dl:
        try:
            res_a, res_sc = _lp_optimize(best_seq, p1_dl, max_iters=250)
            if res_sc < best_score - 1e-9:
                exact = float(evaluate_sequence([float(x) for x in res_a]))
                if np.isfinite(exact) and exact < best_score:
                    best_score = exact
                    best_seq = res_a.copy()
        except Exception:
            pass

    # Phase 2: Multiple starts + LP
    p2_dl = t0 + budget * 0.75
    if time.monotonic() < p2_dl and best_score > 1.05:
        inits = _make_inits(n, rng)
        for idx, init in enumerate(inits):
            if time.monotonic() > p2_dl - 10:
                break
            if best_score < 1.05:
                break
            sub_dl = min(p2_dl - 5, time.monotonic() + 35.0)
            try:
                res_a, res_sc = _lp_optimize(init, sub_dl, max_iters=80)
                if res_sc < best_score - 1e-9:
                    exact = float(evaluate_sequence([float(x) for x in res_a]))
                    if np.isfinite(exact) and exact < best_score:
                        best_score = exact
                        best_seq = res_a.copy()
            except Exception:
                pass

    # Phase 3: Subgradient polishing
    p3_dl = deadline - 12
    if time.monotonic() < p3_dl:
        try:
            res_a, res_sc = _sg_optimize(best_seq, p3_dl, max_iters=10000)
            if res_sc < best_score - 1e-9:
                exact = float(evaluate_sequence([float(x) for x in res_a]))
                if np.isfinite(exact) and exact < best_score:
                    best_score = exact
                    best_seq = res_a.copy()
        except Exception:
            pass

    # Phase 4: Final LP round on best
    if time.monotonic() < deadline - 15:
        sub_dl = deadline - 15
        try:
            res_a, res_sc = _lp_optimize(best_seq, sub_dl, max_iters=80)
            if res_sc < best_score - 1e-9:
                exact = float(evaluate_sequence([float(x) for x in res_a]))
                if np.isfinite(exact) and exact < best_score:
                    best_score = exact
                    best_seq = res_a.copy()
        except Exception:
            pass

    # Final subgradient if time remains
    if time.monotonic() < deadline - 8:
        try:
            res_a, res_sc = _sg_optimize(best_seq, deadline - 8, max_iters=5000)
            if res_sc < best_score - 1e-9:
                exact = float(evaluate_sequence([float(x) for x in res_a]))
                if np.isfinite(exact) and exact < best_score:
                    best_score = exact
                    best_seq = res_a.copy()
        except Exception:
            pass

    # Final validation
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    try:
        final_score = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_score = math.inf

    if not np.isfinite(final_score):
        fb = np.ones(32, dtype=np.float64) / 32.0
        try:
            fs = float(evaluate_sequence([float(x) for x in fb]))
            if np.isfinite(fs):
                return [float(x) for x in fb]
        except Exception:
            pass
        return [1.0, 1.0, 1.0, 1.0]

    return [float(x) for x in best_seq]
