        if time.monotonic() > deadline:
            break
        if nt < 1:
            continue
        try_candidate(np.ones(nt, dtype=np.float64))

    # 4) If everything somehow failed, fall back to a safe uniform
    if best_seq is None:
        best_seq = np.ones(n_parent, dtype=np.float64)
        try:
            best_score = float(evaluate_sequence([float(x) for x in best_seq]))
        except Exception:
            best_score = 2.0

    # Final sanity check with the exact evaluator
    try:
        fs = float(evaluate_sequence([float(x) for x in best_seq]))
        if not np.isfinite(fs):
            best_seq = np.ones(191, dtype=np.float64)
    except Exception:
        best_seq = np.ones(191, dtype=np.float64)

    return [float(x) for x in best_seq]
