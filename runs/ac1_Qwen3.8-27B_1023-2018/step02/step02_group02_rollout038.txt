import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + max(1.0, float(budget_s)) - 5.0
    rng = np.random.default_rng(seed)

    # Retrieve parent
    try:

    except Exception:
        parent = None
    if parent is not None and parent.size >= 2 and np.isfinite(parent).all() and parent.sum() > 0.01:
        parent = np.clip(parent, 0.0, 1000.0)
        n = int(parent.size)
        f = parent / parent.sum()
    else:
        n = 192
        f = np.ones(n) / n

    fsize = next_fast_len(2 * n - 1)

    def convolve_full(a):
        F = rfft(a, fsize)
        c = irfft(F * F, fsize)[:2 * n - 1]
        c = np.maximum(c, 0.0)
        c[~np.isfinite(c)] = 0.0
        return c

    def peak(c):
        return float(c.max())

    c0 = convolve_full(f)
    best_score = 2.0 * n * peak(c0)
    best_f = f.copy()

    def build_B(f_curr):
        B = np.zeros((2 * n - 1, n), dtype=np.float64)
        for j in range(n):
            B[j:j + n, j] = f_curr
        return B

    # ============ LP-based descent ============
    def lp_descent(f_curr, time_budget_s):
        t_start = time.monotonic()
        f = f_curr.copy()
        c_curr = convolve_full(f)
        M = peak(c_curr)
        B = build_B(f)
        sc = 2.0 * n * M
        j_arr = np.arange(n)

        for it in range(1000):
            if time.monotonic() - t_start > time_budget_s:
                break
            if M < 1e-18:
                break

            # Active constraints: use those with c[k] >= 0.85*M
            thresh = 0.85 * M
            active = c_curr >= thresh
            if active.sum() < 5:
                active = c_curr >= 0.5 * M
            if active.sum() < 3:
                active = np.ones(2 * n - 1, dtype=bool)

            B_act = B[active, :]
            b_ub_act = np.full(int(active.sum()), M, dtype=np.float64)
            c_obj = -np.ones(n, dtype=np.float64)
            bounds = [(0.0, None)] * n

            try:
                res = linprog(c_obj, A_ub=B_act, b_ub=b_ub_act, bounds=bounds,
                              method='highs', options={'presolve': True})
            except Exception:
                break

            if not res.success or res.x is None:
                break

            g = np.maximum(np.asarray(res.x, dtype=np.float64).ravel(), 0.0)
            gs = float(g.sum())
            if gs < 1e-12:
                break
            g /= gs

            if float(np.linalg.norm(g - f, 1)) < 1e-9:
                break

            # Careful line search
            best_t = 0.0
            best_m = M
            for t in (1.0, 0.9, 0.8, 0.7, 0.6, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05, 0.02):
                h = (1.0 - t) * f + t * g
                m_h = peak(convolve_full(h))
                if m_h < best_m:
                    best_m = m_h
                    best_t = t

            if best_t <= 0:
                break

                hi = min(1.0, best_t + 0.08)

                    m2 = lo + gr * (hi - lo)

                    fm2 = peak(convolve_full((1 - m2) * f + m2 * g))
                    if fm1 < fm2:
                        hi = m2
                    else:
                        lo = m1
                    t_mid = (lo + hi) / 2.0
                    fm = peak(convolve_full((1 - t_mid) * f + t_mid * g))
                    if fm < best_m:
                        best_m = fm
                        best_t = t_mid

            f = (1.0 - best_t) * f + best_t * g
            c_curr = convolve_full(f)
            M = peak(c_curr)
            B = build_B(f)
            sc = 2.0 * n * M

        return f, sc

            if stuck > 400:

                stuck = 0
                lr = 0.5

                grad[valid] += 2.0 * a[k2 - j_arr[valid]]
            grad /= len(near)

                continue

                if cand_sum < 1e-10:

                    sc_cur = cand_sc

                stuck += 1
                lr = max(0.05, lr * 0.88)

    # ============ Stochastic local search ============
    if time.monotonic() < deadline - 10.0:

        sc = best_score

            elif r < 0.70:
                cand = a.copy()
                k1, k2 = int(rng2.integers(0, n)), int(rng2.integers(0, n))
                if k1 == k2:
                    continue
                amt = 0.008 * float(a.max())
                cand[k1] += amt
                cand[k2] = max(cand[k2] - amt, 1e-12)
                cand /= cand.sum()
            elif r < 0.82:
                # Pair swap
                cand = a.copy()
                i1, i2 = int(rng2.integers(0, n)), int(rng2.integers(0, n))

                    continue
                t_s = rng2.uniform(0.15, 0.85)
                v1, v2 = cand[i1], cand[i2]
                cand[i1] = t_s * v1 + (1 - t_s) * v2
                cand[i2] = (1 - t_s) * v1 + t_s * v2
                cand = np.maximum(cand, 1e-12)

            elif r < 0.90:

                t_mix = rng2.uniform(0.03, 0.20)

            else:

                kernel = np.exp(-0.5 * (np.arange(-3, 4) / 2.0) ** 2)

                cand = np.convolve(a, kernel, mode='same')

            if cand_sc < sc - 1e-14:
                a = cand.copy()
                sc = cand_sc
                if sc < best_score - 1e-14:
