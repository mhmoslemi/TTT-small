        if src.size == n_target:
            return src.copy()
        ox = np.linspace(0.0, 1.0, src.size)
        nx = np.linspace(0.0, 1.0, n_target)
        out = np.interp(nx, ox, src)
        out = np.clip(out, 0, None)
        s = float(out.sum())
        if s > 1e-15:
            out /= s
        else:
            out = np.ones(n_target, dtype=np.float64) / n_target
        return out

    # === MAIN OPTIMIZATION PHASES ===
    p = parent if parent is not None else best_arr
    n_p = p.size
    p_norm = p / p.sum()

    # Phase 1: Intensive optimization at parent length (majority of time)
    t_rem = deadline - time.monotonic()
    if t_rem > 60:
        alloc1 = min(t_rem * 0.45, 600.0)
        a1 = optimize(p_norm, time.monotonic() + alloc1, max_iter=80000)
        ub(a1)

    # Phase 2: Continue optimizing best found
    if time.monotonic() < deadline - 40 and best_arr is not None:
        t_rem2 = deadline - time.monotonic() - 40
        if t_rem2 > 20:
            alloc2 = min(t_rem2 * 0.55, 350.0)
            a2 = optimize(best_arr.copy(), time.monotonic() + alloc2, max_iter=50000)
            ub(a2)

    # Phase 3: Try nearby lengths
    if time.monotonic() < deadline - 20:
        tried = set()
        for dL in (-2, 2, -1, 1, -3, 3, -4, 4, -6, 6, -5, 5, -8, 8, -10, 10, -4, 4):
            if time.monotonic() >= deadline - 15:
                break
            L = n_p + dL
            if L < 6 or L > 700 or L in tried:
                continue
            tried.add(L)
            t_rem3 = deadline - time.monotonic() - 15
            if t_rem3 < 10:
                break
            alloc3 = min(25.0, t_rem3 * 0.35)
            if alloc3 < 6:
                break
            a_init = interp_len(p, L)
            a_w = optimize(a_init, time.monotonic() + alloc3, max_iter=8000)
            ub(a_w)

    # Phase 4: Multi-restart quality at parent length
    if time.monotonic() < deadline - 20:
        for r in range(12):
            if time.monotonic() >= deadline - 15:
                break
            t_rem4 = deadline - time.monotonic() - 15
            if t_rem4 < 8:
                break
            alloc4 = min(20.0, t_rem4 / max(1, 12 - r))
            if alloc4 < 5:
                break

            pert_scale = 0.05 + 0.03 * r
            a_init = p_norm * np.exp(pert_scale * rng.standard_normal(n_p))
            a_init = np.clip(a_init, 0, None)
            si = float(a_init.sum())
            if si > 1e-15:
                a_init /= si
            else:
                a_init = p_norm.copy()

            a_w = optimize(a_init, time.monotonic() + alloc4, max_iter=5000)
            ub(a_w)

    # Phase 5: Final intensive polish on best
    if time.monotonic() < deadline - 8 and best_arr is not None:
        ap = best_arr.copy()
        sp_ = float(ap.sum())
        if sp_ > 1e-15:
            ap /= sp_
        ap = optimize(ap, deadline - 4, max_iter=50000)
        ub(ap)

    # Final validation and return
    if best_arr is None:
        best_arr = np.ones(50, dtype=np.float64)
        ub(best_arr)

    final = np.clip(best_arr, 0.0, 1000.0)
    if float(final.sum()) < 0.01:
        final = np.ones(50, dtype=np.float64)
        ub(final)
    ub(final)

    return [float(x) for x in best_arr]
