(a*a)[k] = sum_j a[j]*a[k-j]

For the anti-palindromic case, there's a relation between lags k and 2n-2-k:
(a*a)[2n-2-k] = sum_j a[j]*a[2n-2-k-j]
Let i = n-1-j: a[n-1-i]*a[2n-2-k-n+1+i] = a[n-1-i]*a[n-1-k+i]
This is the same as sum_i a[n-1-i]*a[n-1-k+i] = (a*a)[n-1-k]... hmm, not exactly.

Actually, if a is palindromic (a[j] = a[n-1-j]), then (a*a)[k] = (a*a)[2n-2-k]. This reduces the number of distinct lags.

    def _f_from_v(v):
        v = np.clip(np.asarray(v, dtype=np.float64).ravel(), 0.0, 1.0)
        f = np.ones(n)
        f[:half] = v
        f[half:] = (1.0 - v)[::-1]
        return f

    def _v_from_f(f):
        f = np.asarray(f, dtype=np.float64).ravel()
        v = f[:half].copy()
        v = np.clip(v, 0.0, 1.0)
        return v

    a = np.asarray(a_init, dtype=np.float64).ravel().copy()
    if a.size != n:
        if a.size > 1:
            xo = np.linspace(0, 1, a.size)
            xn = np.linspace(0, 1, n)
            a = np.interp(xn, xo, a)
        else:
            a = np.ones(n)
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if s > 1e-15:
        a = a / s
    v = _v_from_f(a)
    f = _f_from_v(v)
    local = _score(f)
    _update(f, best)

    step = 0.1
    patience = 0

    for it in range(3000):
        if time.monotonic() >= deadline:
            break
        if local <= TARGET * 1.0005:
            break

        b = np.convolve(f, f)
        M = float(b.max())
        eps = 0.01 * M + 1e-12
        active = np.where(b >= M - eps)[0]
        if len(active) == 0:
            break

        grad_f = np.zeros(n)
        for k in active:
            k = int(k)
            for i in range(n):
                gval = 0.0
                ki = k - i
                if 0 <= ki < n:
                    gval += f[ki]
                ik = i - k
                if 0 <= ik < n:
                    gval += f[ik]
                grad_f[i] += gval
        grad_f /= float(len(active))

        grad_v = grad_f[:half] - grad_f[half:][::-1]
        gn = float(np.linalg.norm(grad_v))
        if gn < 1e-15:
            break

        improved = False
        for s_try in (step, step * 0.5, step * 2.0):
            vc = v - s_try * grad_v / gn
            vc = np.clip(vc, 0.0, 1.0)
            fc = _f_from_v(vc)
            sc = _score(fc)
            if math.isfinite(sc) and sc < local:
                v = vc
                f = fc
                local = sc
                patience = 0
                _update(f, best)
                step = min(step * 1.2, 1.0)
                improved = True
                break

        if not improved:
            patience += 1
            step *= 0.7
            if patience > 40:
                vp = np.clip(v + 0.05 * rng.standard_normal(half), 0.0, 1.0)
                fp = _f_from_v(vp)
                sp = _score(fp)
                if math.isfinite(sp) and sp < local:
                    v = vp
                    f = fp
                    local = sp
                    _update(f, best)
                    step = 0.1
                    patience = 0
                if patience > 50:
                    break

    return local


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    try:
        remaining = max(0.0, float(budget_s))
    except Exception:
        remaining = 1300.0
    deadline = t_start + remaining - 12.0

    best = [float('inf'), np.ones(4, dtype=np.float64)]
    _update(best[1], best)

    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            parent = np.asarray(h, dtype=np.float64).ravel()
            parent = np.nan_to_num(parent, nan=0.0, posinf=0.0, neginf=0.0)
            parent = np.clip(parent, 0.0, 1000.0)
            if parent.size > 1:
                _update(parent, best)
    except Exception:
        parent = None

    def _rem():
        return max(0.0, deadline - time.monotonic())

    # Phase 1: Heavy LP on parent
    if parent is not None and parent.size >= 4 and _rem() > 80:
        np_ = parent.size
        alloc = min(450.0, _rem() * 0.42)
        if alloc > 20:
            _lp_iterate(parent, np_, alloc, rng, best)

    # Phase 2: Subgradient on best
    if _rem() > 50:
        ba = np.asarray(best[1], dtype=np.float64).ravel()
        nb = ba.size if ba.size >= 4 else parent.size if parent is not None else 320
        alloc = min(180.0, _rem() * 0.2)
        if alloc > 10:
            _subgradient_refine(best[1], nb, alloc, rng, best)

    # Phase 3: LP polish on best
    if _rem() > 40:
        ba = np.asarray(best[1], dtype=np.float64).ravel()
        nb = ba.size if ba.size >= 4 else 320
        alloc = min(180.0, _rem() * 0.25)
        if alloc > 10:
            _lp_iterate(best[1], nb, alloc, rng, best)

    # Phase 4: Anti-palindromic LP
    if _rem() > 40:
        ba = np.asarray(best[1], dtype=np.float64).ravel()
        nb = ba.size if ba.size >= 8 else parent.size if parent is not None else 320
        alloc = min(100.0, _rem() * 0.2)
        if alloc > 10:
            _anti_pal_lp_iterate(best[1], nb, alloc, rng, best)

    # Phase 5: Try nearby lengths
    for ntr in (256, 384, 280, 340):
        if _rem() < 35:
            break
        a_init = np.ones(ntr)
        if parent is not None and parent.size > 1:
            tp = np.linspace(0, 1, parent.size)
            xn = np.linspace(0, 1, ntr)
            a_init = np.interp(xn, tp, parent)
        elif best[1].size > 1:
            tp = np.linspace(0, 1, best[1].size)
            xn = np.linspace(0, 1, ntr)
            a_init = np.interp(xn, tp, best[1])
        a_init = np.clip(a_init, 0.01, None)
        a_init = a_init / a_init.sum()
        alloc = min(50.0, _rem() * 0.12)
        if alloc > 8:
            _lp_iterate(a_init, ntr, alloc, rng, best)

    # Phase 6: Final LP polish on best
    if _rem() > 15:
        ba = np.asarray(best[1], dtype=np.float64).ravel()
        nb = ba.size if ba.size >= 4 else 320
        alloc = min(30.0, _rem() * 0.6)
        if alloc > 5:
            _lp_iterate(best[1], nb, alloc, rng, best)

    # Final subgradient polish
    if _rem() > 10:
        ba = np.asarray(best[1], dtype=np.float64).ravel()
        nb = ba.size if ba.size >= 4 else 320
        alloc = min(15.0, _rem() * 0.7)
        if alloc > 3:
            _subgradient_refine(best[1], nb, alloc, rng, best)

    # Final validation
    final = np.asarray(best[1], dtype=np.float64).ravel()
    final = np.nan_to_num(final, nan=0.0, posinf=0.0, neginf=0.0)
    final = np.clip(final, 0.0, 1000.0)
    if final.size < 1:
        final = np.ones(16, dtype=np.float64)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(16, dtype=np.float64)

    fin_sc = evaluate_sequence([float(x) for x in final])
    if not math.isfinite(fin_sc) or fin_sc > best[0] + 0.01:
        final = np.ones(16, dtype=np.float64)
        fin_sc = evaluate_sequence([float(x) for x in final])

    return [float(x) for x in final]
