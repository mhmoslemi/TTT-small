    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M + 1e-12)
    bounds = [(0.0, None)] * n
    opts = {'presolve': True, 'time_limit': time_limit}
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options=opts)
    except Exception:
        return None
    if res is not None and getattr(res, 'x', None) is not None:
        g = np.asarray(res.x, dtype=np.float64).ravel()
        g = np.clip(g, 0.0, None)
        sg = float(g.sum())
        if sg > 1e-15 and math.isfinite(sg):
            return g / sg
    return None


def _lp_iterate(a_init, n_target, time_limit, rng, best):
    """LP-based iteration to minimize the AC1 bound."""
    n = int(n_target)
    if n < 2 or time_limit < 5:
        return
    t0 = time.monotonic()
    deadline = t0 + time_limit - 2.0

    a = np.asarray(a_init, dtype=np.float64).ravel().copy()
    if a.size != n:
        if a.size > 1:
            xo = np.linspace(0.0, 1.0, a.size)
            xn = np.linspace(0.0, 1.0, n)
            a = np.interp(xn, xo, a)
        else:
            a = np.ones(n) / n
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    local = _score(a)
    _update(a, best)

    stagnant = 0
    max_stag = 25
    prev_local = local

    for it in range(10000):
        if time.monotonic() >= deadline:
            break
        if local <= TARGET:
            break

        b = np.convolve(a, a)
        M = float(b.max())
        if not math.isfinite(M) or M <= 1e-30:
            break

        tl = min(30.0, max(1.0, deadline - time.monotonic()))
        if tl < 2.0:
            break

        g = _lp_solve(a, M, n, tl)
        if g is None:
            stagnant += 1
            if stagnant > 8:
                break
            continue

        best_sc = local
        best_cand = None
        ts_list = [0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5, 0.6, 0.7, 0.8, 0.9, 1.0]
        for t in ts_list:
            cand = (1.0 - t) * a + t * g
            cand = np.clip(cand, 0.0, None)
            cs = float(cand.sum())
            if cs < 1e-15:
                continue
            cand = cand / cs
            sc = _score(cand)
            if math.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_cand = cand

        if best_cand is not None and best_sc < local * (1.0 - 1e-10):
            a = best_cand
            local = best_sc
            stagnant = 0
            _update(a, best)
            prev_local = local
        else:
            stagnant += 1
            if stagnant >= max_stag:
                improved = False
                for sz in (0.03, 0.05, 0.1, 0.15, 0.25, 0.4):
                    pa = a * np.exp(sz * rng.standard_normal(n))
                    pa = np.clip(pa, 0.0, None)
                    ps = float(pa.sum())
                    if ps > 1e-15:
                        pa = pa / ps
                        pscore = _score(pa)
                        if math.isfinite(pscore) and pscore < local:
                            a = pa
                            local = pscore
                            _update(a, best)
                            stagnant = 0
                            prev_local = local
                            improved = True
                            break
                if not improved:
                    # Try mixing with a structured perturbation
                    x = np.linspace(0.0, 1.0, n)
                    for mix in (0.1, 0.2, 0.3):
                        structured = a * (1.0 + 0.3 * np.sin(np.pi * x))
                        structured = np.clip(structured, 0.0, None)
                        ca = (1.0 - mix) * a + mix * structured / max(structured.sum(), 1e-15)
                        ca = np.clip(ca, 0.0, None)
                        cs = float(ca.sum())
                        if cs > 1e-15:
                            ca = ca / cs
                            csc = _score(ca)
                            if math.isfinite(csc) and csc < local:
                                a = ca
                                local = csc
                                _update(a, best)
                                stagnant = 0
                                improved = True
                                break
                if not improved and stagnant > 40:
                    break


def _subgrad_iterate(a_init, n_target, time_limit, rng, best):
    """Subgradient descent refinement."""
    n = int(n_target)
    if n < 4 or time_limit < 5:
        return
    t0 = time.monotonic()
    deadline = t0 + time_limit - 1.0

    a = np.asarray(a_init, dtype=np.float64).ravel().copy()
    if a.size != n:
        if a.size > 1:
            xo = np.linspace(0.0, 1.0, a.size)
            xn = np.linspace(0.0, 1.0, n)
            a = np.interp(xn, xo, a)
        else:
            a = np.ones(n) / n
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    local = _score(a)
    _update(a, best)

    alpha = 0.05
    no_improve = 0

    for it in range(5000):
        if time.monotonic() >= deadline:
            break
        if local <= TARGET:
            break

        b = np.convolve(a, a)
        M = float(b.max())
        if M <= 1e-30:
            break

        # Find active set with relative tolerance
        eps_rel = 0.005 + 0.005 * (it / 500.0 + 1e-9)
        threshold = M * (1.0 - eps_rel)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            break

        # Compute subgradient: g_j = (2/|A|) * sum_{k in A} a[k-j] * [0<=k-j<n]
        # = (2/|A|) * sum_{k in A, j in [max(0,k-n+1), min(n-1,k)]} a[k-j]
        g = np.zeros(n)
        for k in active:
            k = int(k)
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            if j_min <= j_max:
                # g[j_min..j_max] += a[k-j_max .. k-j_min] (reversed)
                g[j_min:j_max + 1] += a[k - j_max:k - j_min + 1][::-1]
        g *= 2.0 / len(active)

        # Project to sum(a)=1 feasible direction
        g_proj = g - g.mean()

        improved = False
        for a_coeff in (alpha, alpha * 0.5, alpha * 2.0, alpha * 0.25, alpha * 0.1):
            new_a = a - a_coeff * g_proj / (np.linalg.norm(g_proj) + 1e-15)
            new_a = np.clip(new_a, 0.0, None)
            ns = float(new_a.sum())
            if ns < 1e-15:
                continue
            new_a = new_a / ns
            sc = _score(new_a)
            if math.isfinite(sc) and sc < local:
                a = new_a
                local = sc
                _update(a, best)
                alpha = min(alpha * 1.05, 0.3)
                improved = True
                no_improve = 0
                break

        if not improved:
            no_improve += 1
            alpha *= 0.7
            if alpha < 1e-6:
                # Perturbation
                pa = a * np.exp(0.03 * rng.standard_normal(n))
                pa = np.clip(pa, 0.0, None)
                ps = float(pa.sum())
                if ps > 1e-15:
                    pa = pa / ps
                    pscore = _score(pa)
                    if math.isfinite(pscore) and pscore < local:
                        a = pa
                        local = pscore
                        _update(a, best)
                        alpha = 0.05
                        no_improve = 0
                if no_improve > 50:
                    break


def _local_search_refine(a_init, n_target, time_limit, rng, best):
    """Coordinate-descent style local refinement."""
    n = int(n_target)
    if n < 4 or time_limit < 5:
        return
    t0 = time.monotonic()
    deadline = t0 + time_limit - 1.0

    a = np.asarray(a_init, dtype=np.float64).ravel().copy()
    if a.size != n:
        if a.size > 1:
            xo = np.linspace(0.0, 1.0, a.size)
            xn = np.linspace(0.0, 1.0, n)
            a = np.interp(xn, xo, a)
        else:
            a = np.ones(n) / n
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    local = _score(a)
    _update(a, best)

    # Row-wise and column-wise adjustments
    order = rng.permutation(n)
    max_passes = 200
    for p in range(max_passes):
        if time.monotonic() >= deadline:
            break
        improved = False
        for idx in order:
            if time.monotonic() >= deadline:
                break
            # Try scaling a[idx]
            for frac in (0.7, 0.85, 1.15, 1.3):
                new_a = a.copy()
                new_a[idx] = a[idx] * frac
                new_a = np.clip(new_a, 0.0, None)
                ns = float(new_a.sum())
                if ns < 1e-15:
                    continue
                new_a = new_a / ns
                sc = _score(new_a)
                if math.isfinite(sc) and sc < local:
                    a = new_a
                    local = sc
                    improved = True
                    break
        _update(a, best)
        if not improved and p > 5:
            # Random perturbation
            pa = a * np.exp(0.05 * rng.standard_normal(n))
            pa = np.clip(pa, 0.0, None)
            ps = float(pa.sum())
            if ps > 1e-15:
                pa = pa / ps
                pscore = _score(pa)
                if math.isfinite(pscore) and pscore < local:
                    a = pa
                    local = pscore
                    _update(a, best)
                    improved = True
        if not improved and p > 30:
            break


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    try:
        remaining = max(0.0, float(budget_s))
    except Exception:
        remaining = 1300.0
    deadline = t_start + remaining - 25.0  # 25s buffer for final validation

    best = [float('inf'), np.array([1.0, 1.0], dtype=np.float64)]
    _update(best[1], best)

    # Get parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            parent = np.asarray(h, dtype=np.float64).ravel()
            parent = np.nan_to_num(parent, nan=0.0, posinf=0.0, neginf=0.0)
            parent = np.clip(parent, 0.0, 1000.0)
            if parent.size > 1:
                _update(parent, best)
    except Exception:
        parent = None

    def _rem():
        return max(0.0, deadline - time.monotonic())

    # Phase 1: Heavy LP iteration on parent length (most promising)
    if parent is not None and parent.size >= 4:
        np_ = parent.size
        alloc = min(550.0, _rem() * 0.5)
        if alloc > 40:
            _lp_iterate(parent, np_, alloc, rng, best)

    # Phase 2: Subgradient refinement on current best
    if _rem() > 50:
        best_arr = np.asarray(best[1], dtype=np.float64).ravel()
        nb = best_arr.size
        alloc = min(120.0, _rem() * 0.15)
        if alloc > 15:
            _subgrad_iterate(best_arr, nb, alloc, rng, best)

    # Phase 3: LP at other lengths for diversity
    for ntr in (256, 384, 192, 448):
        if _rem() < 50:
            break
        a_init = np.ones(ntr) / ntr
        if parent is not None and parent.size > 1:
            tp = np.linspace(0.0, 1.0, parent.size)
            xn = np.linspace(0.0, 1.0, ntr)
            a_init = np.interp(xn, tp, parent)
            a_sum = float(a_init.sum())
            if a_sum > 1e-15:
                a_init = a_init / a_sum
        x = np.linspace(0.0, 1.0, ntr)
        a_init = a_init * (1.0 + 0.3 * np.sin(np.pi * x))
        a_init = np.clip(a_init, 0.0, None)
        a_sum = float(a_init.sum())
        if a_sum > 1e-15:
            a_init = a_init / a_sum

        alloc = min(60.0, _rem() * 0.12)
        if alloc > 15:
            _lp_iterate(a_init, ntr, alloc, rng, best)

    # Phase 4: Subgradient on best again (with tighter tolerance)
    if _rem() > 40:
        best_arr = np.asarray(best[1], dtype=np.float64).ravel()
        nb = best_arr.size
        alloc = min(100.0, _rem() * 0.2)
        if alloc > 10:
            _subgrad_iterate(best_arr, nb, alloc, rng, best)

    # Phase 5: Final LP polish on the best candidate
    if _rem() > 30:
        best_arr = np.asarray(best[1], dtype=np.float64).ravel()
        nb = best_arr.size
        alloc = min(80.0, _rem() * 0.35)
        if alloc > 10:
            _lp_iterate(best_arr, nb, alloc, rng, best)

    # Phase 6: Local search refinement
    if _rem() > 20:
        best_arr = np.asarray(best[1], dtype=np.float64).ravel()
        nb = best_arr.size
        alloc = min(40.0, _rem() * 0.3)
        if alloc > 5:
            _local_search_refine(best_arr, nb, alloc, rng, best)

    # Final validation
    final = np.asarray(best[1], dtype=np.float64).ravel()
    final = np.nan_to_num(final, nan=0.0, posinf=0.0, neginf=0.0)
    final = np.clip(final, 0.0, 1000.0)
    if final.size == 0:
        final = np.ones(16, dtype=np.float64)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(16, dtype=np.float64)

    fin_sc = evaluate_sequence([float(x) for x in final])
    if not math.isfinite(fin_sc) or fin_sc > best[0] + 0.005:
        # Fallback to a safe candidate
        if parent is not None and parent.size > 1:
            final = parent
        else:
            final = np.ones(16, dtype=np.float64)
        fin_sc = evaluate_sequence([float(x) for x in final])

    return [float(x) for x in final]
