import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(320, dtype=np.float64)
    if p.size == 0:
        p = np.ones(320, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(320, dtype=np.float64)
    return p


def _conv_full(a):
    n = len(a)
    if n <= 256:
        return np.convolve(a, a, mode="full")
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[: 2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score_fast(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _precompute_pattern(n):
    rows = []
    cols = []
    f_idx = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
            f_idx.append(k - j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32), np.array(f_idx, dtype=np.int32)


def _make_A(f, pattern):
    rows, cols, f_idx = pattern
    data = f[f_idx]
    return csr_matrix((data, (rows, cols)), shape=(2 * len(f) - 1, len(f)))


def _lp_direction(f, n, pattern):
    b = _conv_full(f)
    M = float(b.max())
    if M < 1e-15:
        return None, M
    A = _make_A(f, pattern)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(
            c, A_ub=A, b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True,
                     "dual_feasibility_tolerance": 1e-11,
                     "primal_feasibility_tolerance": 1e-11,
                     "time_limit": 5.0}
        )
    except Exception:
        try:
            A_d = A.toarray()
            res = linprog(c, A_ub=A_d, b_ub=b_ub, bounds=bounds, method="highs")
        except Exception:
            return None, M
    if res.status != 0 or res.x is None:
        return None, M
    g = np.asarray(res.x, dtype=np.float64)
    g_sum = float(g.sum())
    if g_sum < 1e-15:
        return None, M
    return g / g_sum, M


def _line_search(f, g_dir, best_score, t_start, stop_time):
    t = min(0.6, max(0.05, t_start))
    best_t = 0.0
    best_sc = best_score
    for _ in range(12):
        if time.monotonic() > stop_time - 1.0:
            break
        f_new = np.clip((1.0 - t) * f + t * g_dir, 0, None)
        fs = float(f_new.sum())
        if fs < 1e-10:
            t *= 0.5
            continue
        f_new = f_new / fs
        sc = _score_fast(f_new)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_t = t
            t = min(0.9, t * 1.3)
        else:
            t *= 0.5
            if t < 0.005:
                break
    if best_t > 0:
        f_new = np.clip((1.0 - best_t) * f + best_t * g_dir, 0, None)
        fs = float(f_new.sum())
        if fs > 1e-10:
            f_new = f_new / fs
        return f_new, best_sc, best_t
    return None, best_score, t_start


def _lp_optimize(f0, n, stop_time, seed):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel()[:n].copy(), 0, None)
    while f.size < n:
        f = np.concatenate([f, np.zeros(n - f.size)])
    f = f[:n]
    f_sum = float(f.sum())
    if f_sum < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / f_sum

    best_f = f.copy()
    best_score = _score_fast(f)
    pattern = _precompute_pattern(n)
    t = 0.2
    stall = 0
    max_stall = 60
    iteration = 0

    while time.monotonic() < stop_time - 3.0:
        iteration += 1
        g_dir, M = _lp_direction(f, n, pattern)
        if g_dir is None:
            stall += 5
            if stall > max_stall:
                f = np.clip(f + rng.normal(0, 0.02, n), 0, None)
                fs = f.sum()
                if fs > 1e-10:
                    f = f / fs
                else:
                    f = best_f.copy()
                stall = 0
            continue

        dist = float(np.linalg.norm(g_dir - f))
        if dist < 1e-8:
            stall += 4
            if stall > max_stall:
                f = np.clip(f + rng.normal(0, 0.04, n), 0, None)
                fs = f.sum()
                if fs > 1e-10:
                    f = f / fs
                else:
                    f = best_f.copy()
                t = 0.2
                stall = 0
            continue

        f_new, sc, t = _line_search(f, g_dir, best_score, t, stop_time)
        if f_new is not None:
            f = f_new
            if sc < best_score - 1e-13:
                best_f = f.copy()
                best_score = sc
            stall = 0
        else:
            stall += 1
            if stall > max_stall:
                f = np.clip(f + rng.normal(0, 0.05, n), 0, None)
                fs = f.sum()
                if fs > 1e-10:
                    f = f / fs
                else:
                    f = best_f.copy()
                t = 0.2
                stall = 0

    return best_f, best_score


def _subgradient_polish(a, stop_time):
    n = len(a)
    best = np.clip(np.asarray(a, dtype=np.float64).ravel(), 0, None).copy()
    fs = best.sum()
    if fs < 1e-10:
        return best, _score_fast(best)
    best = best / fs
    best_sc = _score_fast(best)
    rng = np.random.default_rng(777)
    eta = 0.05
    no_imp = 0
    max_no_imp = 30

    for _iter in range(20000):
        if time.monotonic() > stop_time - 2.0:
            break
        conv = _conv_full(best)
        M = float(conv.max())
        # Subgradient: average over active lags
        thresh = M * 0.9999
        active = np.where(conv >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(conv))])
        grad = np.zeros(n)
        for k in active:
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    grad[j] += best[idx]
                if 2 * j == k:
                    grad[j] += best[j]
        grad /= len(active)

        # Projected gradient step onto simplex {a>=0, sum=1}
        b = best - eta * grad
        # Project onto simplex
        sorted_b = np.sort(b)[::-1]
        cumsum = np.cumsum(sorted_b)
        rho = np.where(sorted_b > 0)[0]
        if len(rho) == 0:
            best = np.ones(n) / n
            best_sc = _score_fast(best)
            continue
        tau_candidates = (cumsum[rho] - 1.0) / (rho + 1)
        valid = tau_candidates < sorted_b[rho]
        if np.any(valid):
            rho_idx = rho[valid][-1]
            tau = (cumsum[rho_idx] - 1.0) / (rho_idx + 1)
        else:
            tau = sorted_b[0] - 1.0 / n
        new_a = np.maximum(b - tau, 0.0)
        s = new_a.sum()
        if s < 1e-10:
            new_a = np.ones(n) / n
        else:
            new_a = new_a / s
        sc = _score_fast(new_a)
        if sc < best_sc - 1e-14:
            best = new_a
            best_sc = sc
            no_imp = 0
            eta = min(eta * 1.1, 0.3)
        else:
            eta *= 0.8
            no_imp += 1
            if no_imp > max_no_imp:
                # Coordinate descent kick
                for _cd in range(50):
                    if time.monotonic() > stop_time - 2.0:
                        break
                    i = int(rng.integers(n))
                    orig = best[i]
                    for mlt in (0.3, 0.5, 0.7, 1.3, 1.7, 2.5, 0.0):
                        best[i] = orig * mlt if orig > 1e-12 else (0.01 if mlt > 0 else 0.0)
                        if best.sum() < 0.01:
                            best[i] = orig
                            continue
                        sc2 = _score_fast(best)
                        if sc2 < best_sc - 1e-14:
                            best_sc = sc2
                            no_imp = 0
                        else:
                            best[i] = orig
                    if no_imp > max_no_imp * 2:
                        break
                break
    return best, best_sc


def _coord_polish(a, stop_time):
    n = len(a)
    best = np.clip(np.asarray(a, dtype=np.float64).ravel(), 0, None).copy()
    fs = best.sum()
    if fs < 1e-10:
        return best, _score_fast(best)
    best = best / fs
    best_sc = _score_fast(best)
    rng = np.random.default_rng(9999)
    no_imp = 0

    while time.monotonic() < stop_time - 2.0:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 2.0:
                break
            orig = best[i]
            best_val = orig
            best_val_sc = best_sc
            multipliers = (0.2, 0.4, 0.6, 0.8, 1.2, 1.5, 2.0, 3.0, 0.0)
            for mlt in multipliers:
                best[i] = orig * mlt if orig > 1e-12 else (0.01 if mlt > 0 else 0.0)
                if best.sum() < 0.01:
                    best[i] = orig
                    continue
                sc = _score_fast(best)
                if sc < best_val_sc - 1e-14:
                    best_val = best[i]
                    best_val_sc = sc
            best[i] = best_val
            if best_val_sc < best_sc - 1e-14:
                best_sc = best_val_sc
                improved = True

        # 2-coordinate
        for _ in range(40):
            if time.monotonic() > stop_time - 2.0:
                break
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            for mi in (0.5, 1.5, 2.0):
                for mj in (0.5, 1.5, 2.0):
                    best[i] = oi * mi if oi > 1e-12 else 0.01
                    best[j] = oj * mj if oj > 1e-12 else 0.01
                    if best.sum() < 0.01:
                        best[i], best[j] = oi, oj
                        continue
                    sc = _score_fast(best)
                    if sc < best_sc - 1e-14:
                        best_sc = sc
                        improved = True
            best[i], best[j] = oi, oj

        if not improved:
            no_imp += 1
            if no_imp >= 5:
                break
        else:
            no_imp = 0
    return best, best_sc
