        s_out = 2.0 * n * float(_convolve(a_out, n, fs_c).max())
        return a_out, s_out

    explore_end = t0 + budget * 0.70
    polish_end = t0 + budget * 0.92

            np.exp(-3 * (2*t_arr - 1)**2),  # Gaussian
            1.0 / (1.0 + np.exp(10*x)),     # sigmoid
            np.abs(np.sin(np.pi * t_arr)),   # sine abs
        ]
        for vi, v in enumerate(variants):
            v = v + 1e-4 * np.abs(rng.normal(0, 1, n_try))
            v = np.maximum(v, 1e-6)
            inits.append((n_try, v, f"v{vi}_{n_try}"))

    n_inits = len(inits)
    time_per_init = max(2.0, (explore_end - t0) / n_inits)
    
    for ii, (n_i, a_i, tag) in enumerate(inits):
        if time.monotonic() >= explore_end:
            break
        dl = min(t0 + time.monotonic() - t0 + time_per_init, explore_end)
        try:
            a_res, s_res = run_opt(n_i, a_i, dl, tau_frac0=0.3, )
            if np.isfinite(s_res) and s_res < best_s:
                best_s = float(s_res)
                best_a = a_res.copy()
        except:
            pass
    
    # Phase 2: Refine best
    if time.monotonic() < polish_end:
        try:
            a_res, s_res = run_opt(best_a.size, best_a, polish_end, tau_frac0=0.1)
            if np.isfinite(s_res) and s_res < best_s:
                best_s = float(s_res)
                best_a = a_res.copy()
        except:
            pass

    if time.monotonic() < hard_dl - 2:

        fs = float(evaluate_sequence(cand))
        if not np.isfinite(fs):
            cand = [1.0]
        return cand
    except:
        return [1.0]

**Coordinate-wise polish**: After the global optimization, try adjusting each coordinate (or small groups) to reduce the max autocorrelation. Since the autocorrelation is `b_k = sum_j a_j a_{k-j}`, changing one `a_i` affects multiple `b_k` values. The effect is:
- `b_k` changes by `a_i * (a_{k-i} + a_{k-n+1-i})`... wait, for full convolution of length-n with itself:
- `b_k = sum_{j=0}^{n-1} a_j a_{k-j}` (with a_m=0 for m<0 or m>=n)
- If we change `a_i → a_i + δ`, then `b_k` changes by `δ * a_{k-i}` for each k.
- The new `a_i` also appears as `a_{k-i'} ` for `k-i' = i`, i.e., `k = i+i'`.

So changing `a_i` by `δ` changes `b_k` by `δ * a_{k-i}` for all k. To minimize the max, we want to reduce the largest `b_k` values. If `a_i` is large and it contributes to the peak, reducing it helps.

A simple approach: for each i, try a few values of `a_i` (scale up/down) and pick the one that minimizes max(b). This is O(n * n * n * n) naively, but with FFT it's O(n * n * n log n). For n=192, that's about 192 * 192 * 800 ≈ 30M operations per full sweep. Feasible in a few seconds.

def _sm(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - z.max()
    e = np.exp(np.clip(z, -80.0, 80.0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / z.size
    return e / s


def _conv(a, n, fs):
    F = rfft(a, fs)
    return irfft(F * F, fs)[:2 * n - 1]


def _xcorr(p, a, n, fs):
    """c_i = sum_k p_k * a_{k-i} for i=0..n-1; p len 2n-1, a len n."""
    ar = a[::-1]
    c = irfft(rfft(ar, fs) * rfft(p, fs), fs)
    return c[n - 1: 2 * n - 1]


def _sc(a, n, fs):
    b = _conv(a, n, fs)
