def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))

    parent = _get_parent()
    n = len(parent)
    rng = np.random.default_rng(seed)

    # FFT setup
    fsize = next_fast_len(2 * n - 1)
    half = 2 * n - 1

    def _conv(a):
        F = rfft(a, fsize)
        return irfft(F * F, fsize)[:half]

    def _sc(a):
        s = a.sum()
        if s < 0.01:
            return math.inf
        return 2.0 * n * float(_conv(a).max()) / (s * s)

    # Initialize from parent
    f = np.clip(parent, 0, None).astype(np.float64)
    if f.sum() < 1e-10:
        f = np.ones(n) / n
    f = f / f.sum()

    best_f = f.copy()
    best_sc = _sc(f)

    # Pre-allocate constraint matrix
    A = np.zeros((half, n))

    def _fill_A(a):
        A[:] = 0.0
        for j in range(n):
            A[j:j + n, j] = a

    def _lp_iteration(a, use_elementwise=True):
        """Solve the LP direction-finding step. Returns normalized g or None."""
        b_cur = _conv(a)
        M = float(b_cur.max())
        if M < 1e-15:
            return None, 0.0

        _fill_A(a)
        b_ub = b_cur.copy() if use_elementwise else np.full(half, M)

        try:
            res = linprog(-np.ones(n), A_ub=A, b_ub=b_ub,
                          bounds=[(0, None)] * n,
                          method='highs', options={'presolve': True})
        except Exception:
            return None, M

        if res.status != 0 or res.x is None:
            return None, M

        g = res.x.copy()
        gs = g.sum()
        if gs < 1e-12:
            return None, M
        return g / gs, M

    # === Phase 1: Main LP loop ===
    lp_deadline = t0 + budget * 0.72
    t = 0.35
    stall = 0
    max_stall = 300

    while time.monotonic() < lp_deadline:
        g, M = _lp_iteration(f, use_elementwise=True)

        if g is None:
            stall += 5
            if stall > max_stall:
                break
            f = np.clip(f + rng.normal(0, 0.04, n), 0, None)
            f /= max(f.sum(), 1e-10)
            continue

        d = float(np.abs(g - f).max())
        if d < 1e-10:
            stall += 2
            if stall > max_stall:
                break
            f = np.clip(f + rng.normal(0, 0.08, n), 0, None)
            f /= max(f.sum(), 1e-10)
            t = 0.35
            max_stall = max(100, int(max_stall * 0.92))
            continue

        # Step size search
        improved = False
        t_candidates = sorted({max(0.005, min(1.0, v)) for v in
                               (t, 0.5 * t, 0.25 * t, 0.6, 0.15)}, reverse=True)
        for t_try in t_candidates:
            fn = (1.0 - t_try) * f + t_try * g
            np.clip(fn, 0, None, out=fn)
            if fn.sum() < 1e-10:
                continue
            sc = _sc(fn)
            if sc < best_sc:
                f = fn
                best_f = fn.copy()
                best_sc = sc
                stall = 0
                improved = True
                t = min(0.65, t_try * 1.03)
                break
            t = max(0.005, t * 0.55)

        if not improved:
            stall += 1
            if stall >= max_stall:
                f = np.clip(best_f + rng.normal(0, 0.1, n), 0, None)
                f /= max(f.sum(), 1e-10)
                t = 0.35
                stall = 0
                max_stall = max(80, int(max_stall * 0.82))

    # === Phase 2: Small-step refinement ===
    refine_deadline = t0 + budget * 0.86
    if time.monotonic() < refine_deadline:
        f = best_f.copy()
        t = 0.06
        stall = 0
        max_stall = 200

        while time.monotonic() < refine_deadline:
            g, M = _lp_iteration(f, use_elementwise=True)
            if g is None:
                stall += 4
                if stall > max_stall:
                    break
                f = np.clip(f + rng.normal(0, 0.02, n), 0, None)
                f /= max(f.sum(), 1e-10)
                continue

            d = float(np.abs(g - f).max())
            if d < 1e-10:
                stall += 2
                if stall > max_stall:
                    break
                f = np.clip(f + rng.normal(0, 0.03, n), 0, None)
                f /= max(f.sum(), 1e-10)
                continue

            improved = False
            for t_try in (t, 0.5 * t, 0.2 * t, 0.04, 0.01):
                t_try = max(0.002, min(1.0, t_try))
                fn = (1.0 - t_try) * f + t_try * g
                np.clip(fn, 0, None, out=fn)
                if fn.sum() < 1e-10:
                    continue
                sc = _sc(fn)
                if sc < best_sc:
                    f = fn
                    best_f = fn.copy()
                    best_sc = sc
                    stall = 0
                    improved = True
                    t = min(0.12, t_try * 1.04)
                    break
                t = max(0.002, t * 0.5)

            if not improved:
                stall += 1
                if stall >= max_stall:
                    f = np.clip(best_f + rng.normal(0, 0.04, n), 0, None)
                    f /= max(f.sum(), 1e-10)
                    t = 0.06
                    stall = 0
                    max_stall = max(60, int(max_stall * 0.85))

    # === Phase 3: Coordinate descent polishing ===
    polish_end = t0 + budget - 12.0
    if time.monotonic() < polish_end - 3.0:
        a = best_f.copy()
        a_sc = best_sc
        no_imp = 0
        muls_1d = (0.25, 0.4, 0.55, 0.7, 0.82, 0.92, 1.08, 1.18, 1.35, 1.6, 2.2, 0.0)

        while time.monotonic() < polish_end:
            improved = False
            order = rng.permutation(n)
            for i in order:
                if time.monotonic() > polish_end:
                    break
                orig = a[i]
                bv, bvs = orig, a_sc
                for m in muls_1d:
                    val = orig * m if orig > 1e-12 else (0.003 if m > 0 else 0.0)
                    a[i] = val
                    if a.sum() < 0.01:
                        a[i] = orig
                        continue
                    s = _sc(a)
                    if s < bvs - 1e-15:
                        bv, bvs = val, s
                a[i] = bv
                if bvs < a_sc - 1e-15:
                    a_sc = bvs
                    improved = True

            # 2-coordinate moves
            for _ in range(60):
                if time.monotonic() > polish_end:
                    break
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j:
                    continue
                oi, oj = a[i], a[j]
                for mi in (0.35, 0.6, 0.8, 1.25, 1.5, 2.0):
                    for mj in (0.35, 0.6, 0.8, 1.25, 1.5, 2.0):
                        a[i] = oi * mi if oi > 1e-12 else 0.003
                        a[j] = oj * mj if oj > 1e-12 else 0.003
                        if a.sum() < 0.01:
                            a[i], a[j] = oi, oj
                            continue
                        s = _sc(a)
                        if s < a_sc - 1e-15:
                            a_sc = s
                            improved = True
                a[i], a[j] = oi, oj

            if not improved:
                no_imp += 1
                if no_imp >= 14:
                    break
            else:
                no_imp = 0

        if a_sc < best_sc:
            best_f = a
            best_sc = a_sc

    # Final validation with exact evaluator
    try:
        fs = float(evaluate_sequence([float(x) for x in best_f]))
        if math.isfinite(fs):
            return [float(x) for x in best_f]
    except Exception:
        pass

    # Fallback
    if np.all(np.isfinite(best_f)) and best_f.sum() > 0.01:
        return [float(x) for x in best_f]
    return [1.0] * 32
