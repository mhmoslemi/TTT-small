import math

def _score(a):
    """Fast internal score matching the evaluator."""
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)

def _lp_solve_dense(a, M, n, time_limit=10):
    """Solve the LP: max sum(g) s.t. convolve(a,g) <= M, g >= 0."""
    if linprog is None or sps is None:
        return None
    # Build sparse convolution matrix
    col_idx = np.repeat(np.arange(n, dtype=np.int64), n)
    row_idx = col_idx + np.tile(np.arange(n, dtype=np.int64), n)
    vals = np.tile(a, n).astype(np.float64)
    A = sps.csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M + 1e-12)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
    except Exception:
        return None
    if res is not None and getattr(res, 'x', None) is not None:
        g = np.asarray(res.x, dtype=np.float64).ravel()
        g = np.clip(g, 0.0, None)
        sg = float(g.sum())
        if sg > 1e-15 and math.isfinite(sg):
            return g / sg
    return None


def _lp_iterate(a_init, time_limit, rng, best, target=1.50):
    """LP-based iteration to minimize the score."""
    n = int(a_init.size)
    if n < 4 or time_limit < 5:
        return
    t0 = time.monotonic()
    deadline = t0 + time_limit - 2

    a = np.asarray(a_init, dtype=np.float64).ravel().copy()
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    else:
        a = a / s

    local = _score(a)
    _update(best, a)

    stagnant = 0
    max_stagnant = 30

    for it in range(5000):
        if time.monotonic() >= deadline:
            break
        if local <= target:
            break

        b = np.convolve(a, a)
        M = float(b.max())
        if not math.isfinite(M) or M <= 1e-30:
            break

        tl = min(20.0, max(1.0, deadline - time.monotonic()))
        if tl < 2.0:
            break

        g = _lp_solve_dense(a, M, n, tl)
        if g is None:
            stagnant += 1
            if stagnant > 5:
                # Try perturbation
                pa = a * np.exp(0.1 * rng.standard_normal(n))
                pa = np.clip(pa, 0.0, None)
                ps = float(pa.sum())
                if ps > 1e-15:
                    pa = pa / ps
                    pscore = _score(pa)
                    if math.isfinite(pscore) and pscore < local:
                        a = pa
                        local = pscore
                        _update(best, a)
                        stagnant = 0
                    else:
                        stagnant += 2
                if stagnant > max_stagnant:
                    break
            continue

        # Line search
        best_sc = local
        best_cand = None
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            cand = (1.0 - t) * a + t * g
            cand = np.clip(cand, 0.0, None)
            cs = float(cand.sum())
            if cs < 1e-15:
                continue
            cand = cand / cs
            sc = _score(cand)
            if math.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_cand = cand

        if best_cand is not None and best_sc < local * (1.0 - 1e-8):
            a = best_cand
            local = best_sc
            stagnant = 0
            _update(best, a)
        else:
            stagnant += 1
            if stagnant >= max_stagnant:
                break


def _subgradient_descent(a_init, time_limit, rng, best, target=1.50):
    """Subgradient descent on the max-convolution."""
    n = int(a_init.size)
    if n < 4 or time_limit < 5:
        return
    t0 = time.monotonic()
    deadline = t0 + time_limit - 2

    a = np.asarray(a_init, dtype=np.float64).ravel().copy()
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if s < 1e-15:
        a = np.ones(n)
        s = float(n)
    else:
        a = a / s

    local = _score(a)
    _update(best, a)

    if local <= target:
        return

    alpha = 1.0 / math.sqrt(n)
    stale = 0

    for it in range(10000):
        if time.monotonic() >= deadline:
            break
        if local <= target:
            break

        b = np.convolve(a, a)
        M = float(b.max())
        if not math.isfinite(M) or M <= 1e-30:
            break

        # Find active lags (within epsilon of max)
        eps = 0.01 * M
        active = np.where(b >= M - eps)[0]
        if len(active) == 0:
            break

        # Compute subgradient (average over active lags)
        grad = np.zeros(n)
        for k in active:
            k = int(k)
            if k < n:
                # (a*a)_k = sum_j a_j * a_{j+k}
                # d/d a_j of (a*a)_k:
                # if j < n-k: contributes a_{j+k}
                # if j >= k: contributes a_{j-k}
                dk = k
                for j in range(n):
                    if j + dk < n:
                        grad[j] += a[j + dk]
                    if j >= dk:
                        grad[j] += a[j - dk]
            else:
                dk = 2 * n - 2 - k
                for j in range(n):
                    if j + dk < n:
                        grad[j] += a[j + dk]
                    if j >= dk:
                        grad[j] += a[j - dk]

        grad /= len(active)

        # Normalize and apply
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        # Try several step sizes
        improved = False
        scale = alpha * gnorm
        for factor in [1.0, 0.5, 0.1, 2.0, 0.05]:
            step = alpha * factor
            a_try = a - step * grad / gnorm
            a_try = np.clip(a_try, 0.0, None)
            ts = float(a_try.sum())
            if ts < 1e-15:
                continue
            a_try = a_try / ts
            sc = _score(a_try)
            if math.isfinite(sc) and sc < local:
                a = a_try
                local = sc
                stale = 0
                improved = True
                alpha = min(alpha * 1.05, 1.0)
                _update(best, a)
                break

        if not improved:
            stale += 1
            alpha *= 0.8
            if alpha < 1e-6:
                # Random perturbation
                a_try = a * np.exp(0.05 * rng.standard_normal(n))
                a_try = np.clip(a_try, 0.0, None)
                ts = float(a_try.sum())
                if ts > 1e-15:
                    a_try = a_try / ts
                    sc = _score(a_try)
                    if math.isfinite(sc) and sc < local:
                        a = a_try
                        local = sc
                        _update(best, a)
                        stale = 0
                        alpha = 1.0 / math.sqrt(n)
                if stale > 50:
                    break


def _coordinate_descent(a_init, time_limit, rng, best, target=1.50):
    """Coordinate-wise optimization."""
    n = int(a_init.size)
    if n < 4 or time_limit < 5:
        return
    t0 = time.monotonic()
    deadline = t0 + time_limit - 2

    a = np.asarray(a_init, dtype=np.float64).ravel().copy()
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if s < 1e-15:
        a = np.ones(n)
    else:
        a = a / s

    local = _score(a)
    _update(best, a)

    if local <= target:
        return

    order = rng.permutation(n)
    no_improve = 0

    for it in range(20000):
        if time.monotonic() >= deadline:
            break
        if local <= target:
            break

        idx = int(order[it % n])
        old_val = a[idx]
        if old_val < 1e-15:
            # Try small positive values
            for v in [0.01, 0.1, 0.5, 1.0]:
                a[idx] = v
                sc = _score(a)
                if math.isfinite(sc) and sc < local:
                    local = sc
                    _update(best, a)
                    no_improve = 0
                    break
            else:
                a[idx] = old_val
                no_improve += 1
            continue

        # Try scaling a[idx]
        best_v = old_val
        best_sc = local
        for scale_f in [0.5, 0.7, 1.3, 1.5, 2.0, 0.2, 3.0, 0.01]:
            a[idx] = old_val * scale_f
            a[idx] = min(a[idx], 1000.0)
            sc = _score(a)
            if math.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_v = a[idx]
        a[idx] = best_v
        if best_sc < local * (1 - 1e-10):
            local = best_sc
            no_improve = 0
            _update(best, a)
        else:
            no_improve += 1

        if no_improve > n * 3:
            # Reshuffle and add noise
            a = a * np.exp(0.05 * rng.standard_normal(n))
            a = np.clip(a, 0.0, None)
            s2 = float(a.sum())
            if s2 > 1e-15:
                a = a / s2
            local = _score(a)
            _update(best, a)
            order = rng.permutation(n)
            no_improve = 0


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()

    try:
        remaining = max(0.0, float(budget_s))
    except Exception:
        remaining = 1300.0
    deadline = t_start + remaining - 15.0

    best = [float('inf'), np.ones(320, dtype=np.float64) / 320.0]

    def _rem():
        return max(0.0, deadline - time.monotonic())

    # Phase 0: Evaluate uniform sequences and parent
    for n in [16, 32, 64, 128, 192, 256, 320, 384, 512]:
        _update(best, np.ones(n, dtype=np.float64) / n)

    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            parent = np.asarray(h, dtype=np.float64).ravel()
            parent = np.nan_to_num(parent, nan=0.0, posinf=0.0, neginf=0.0)
            parent = np.clip(parent, 0.0, 1000.0)
            if parent.size > 0:
                _update(best, parent)
    except Exception:
        parent = None

    # If we already have score <= 2.0 + tiny, we're essentially optimal
    # (theoretical minimum for non-negative sequences is 2.0)
    if best[0] <= 2.0 + 1e-9:
        # Try a few more things for numerical precision
        for n in [320, 128, 64]:
            _update(best, np.ones(n, dtype=np.float64))
        final = np.asarray(best[1], dtype=np.float64).ravel()
        final = np.clip(final, 0.0, 1000.0)
        if final.size == 0:
            final = np.ones(320, dtype=np.float64) / 320.0
        return [float(x) for x in final]

    # Phase 1: LP iteration on parent (if available and has decent score)
    if parent is not None and parent.size >= 8 and _rem() > 60:
        alloc = min(300.0, _rem() * 0.35)
        _lp_iterate(parent, alloc, rng, best)

    # Phase 2: Subgradient descent from best
    if _rem() > 40:
        best_arr = np.asarray(best[1], dtype=np.float64).ravel()
        if best_arr.size >= 8:
            alloc = min(200.0, _rem() * 0.25)
            _subgradient_descent(best_arr, alloc, rng, best)

    # Phase 3: Try LP on various sizes
    for ntr in [96, 128, 160, 192, 256]:
        if _rem() < 30:
            break
        a_init = np.ones(ntr, dtype=np.float64)
        if parent is not None and parent.size > 1:
            tp = np.linspace(0.0, 1.0, parent.size)
            xn = np.linspace(0.0, 1.0, ntr)
            a_init = np.interp(xn, tp, parent)
        x = np.linspace(0.0, 1.0, ntr)
        a_init = a_init * (1.0 + 0.5 * x)
        alloc = min(40.0, _rem() * 0.15)
        if alloc > 10:
            _lp_iterate(a_init, alloc, rng, best)

    # Phase 4: Coordinate descent polish
    if _rem() > 20:
        best_arr = np.asarray(best[1], dtype=np.float64).ravel()
        if best_arr.size >= 8:
            alloc = min(80.0, _rem() * 0.3)
            _coordinate_descent(best_arr, alloc, rng, best)

    # Phase 5: Final subgradient polish on best
    if _rem() > 15:
        best_arr = np.asarray(best[1], dtype=np.float64).ravel()
        if best_arr.size >= 8:
            alloc = min(60.0, _rem() * 0.4)
            _subgradient_descent(best_arr, alloc, rng, best)

    # Phase 6: LP polish on best
    if _rem() > 10:
        best_arr = np.asarray(best[1], dtype=np.float64).ravel()
        if best_arr.size >= 8:
            alloc = min(40.0, _rem() * 0.5)
            _lp_iterate(best_arr, alloc, rng, best)

    # Final validation
    final = np.asarray(best[1], dtype=np.float64).ravel()
    final = np.nan_to_num(final, nan=0.0, posinf=0.0, neginf=0.0)
    final = np.clip(final, 0.0, 1000.0)
    if final.size == 0:
        final = np.ones(320, dtype=np.float64) / 320.0

    fin_sc = evaluate_sequence([float(x) for x in final])
    if not math.isfinite(fin_sc):
        final = np.ones(320, dtype=np.float64) / 320.0
        fin_sc = evaluate_sequence([float(x) for x in final])

    # Also check uniform as fallback
    unif = np.ones(320, dtype=np.float64) / 320.0
    unif_sc = evaluate_sequence([float(x) for x in unif])
    if math.isfinite(unif_sc) and unif_sc < fin_sc:
        final = unif
        fin_sc = unif_sc

    return [float(x) for x in final]
