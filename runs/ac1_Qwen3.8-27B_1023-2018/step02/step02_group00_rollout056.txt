def _polish(a, stop_time, seed=4242):
    n = len(a)
    best = np.clip(a, 0, None).copy()
    best = best / best.sum()
    best_sc = _score_fast(best)
    rng = np.random.default_rng(seed)
    no_imp = 0
    
    while time.monotonic() < stop_time - 2:
        improved = False
        
        # Single coordinate moves
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 2:
                break
            orig = best[i]
            best_i_sc = best_sc
            for mlt in (0.1, 0.3, 0.5, 0.7, 1.3, 1.5, 2.0, 0.0):
                best[i] = orig * mlt if orig > 1e-12 else (0.05 if mlt > 0 else 0.0)
                sc = _score_fast(best)
                if sc < best_i_sc:
                    best_i_sc = sc
            if best_i_sc < best_sc:
                best_sc = best_i_sc
                improved = True
            else:
                best[i] = orig
        
        # Paired coordinate moves
        for _ in range(100):
            if time.monotonic() > stop_time - 2:
                break
            i, j = rng.integers(n), rng.integers(n)
            if i == j:
                continue
            oi, oj = best[i], best[j]
            pair_best_sc = best_sc
            for mi in (0.3, 0.7, 1.3, 0.0):
                for mj in (0.3, 0.7, 1.3, 0.0):
                    best[i] = oi * mi if oi > 1e-12 else (0.01 if mi > 0 else 0.0)
                    best[j] = oj * mj if oj > 1e-12 else (0.01 if mj > 0 else 0.0)
                    sc = _score_fast(best)
                    if sc < pair_best_sc:
                        pair_best_sc = sc
                        pair_best_i, pair_best_j = best[i], best[j]
            best[i], best[j] = oi, oj
            if pair_best_sc < best_sc:
                best[i], best[j] = pair_best_i, pair_best_j
                best_sc = pair_best_sc
                improved = True
        
        if not improved:
            no_imp += 1
            if no_imp >= 8:
                break
        else:
            no_imp = 0
    
    return best, best_sc
