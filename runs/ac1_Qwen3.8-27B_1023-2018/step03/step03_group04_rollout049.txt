                new_val = orig * mlt if orig > 1e-12 else (0.01 if mlt > 0 else 0.0)
                best[i] = new_val

                best_sc = best_val_sc
                improved = True

                break
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue

            for mi in (0.0, 0.5, 1.5, 2.0):

                    best[i] = oi * mi if oi > 1e-12 else (0.01 if mi > 0 else 0.0)
                    best[j] = oj * mj if oj > 1e-12 else (0.01 if mj > 0 else 0.0)

            if no_imp >= max_no_imp:

        else:

def _gradient_refine(f, stop_time, seed=99):
    """Subgradient refinement."""
    n = len(f)
    f = np.clip(f.copy(), 0, None)
    fs = f.sum()
    if fs < 1e-10:
        return f, math.inf
    f = f / fs
    best_f = f.copy()
    best_sc = _score(f)
    rng = np.random.default_rng(seed)
    lr = 0.03
    
    for it in range(500):
        if time.monotonic() > stop_time - 2.0:
            break
        conv = _conv(f)
        M = float(conv.max())
        thresh = M * (1.0 - 0.05)
        active = np.where(conv >= thresh)[0]
        
        subgrad = np.zeros(n)
        for k in active:
            for i in range(n):
                idx = k - i
                if 0 <= idx < n:
                    subgrad[i] += 2.0 * f[idx]
        subgrad /= len(active)
        
        f_new = f - lr * subgrad
        f_new = np.clip(f_new, 0, None)
        fs_new = f_new.sum()
        if fs_new < 1e-10:
            break
        f_new = f_new / fs_new
        sc = _score(f_new)
        if sc < best_sc - 1e-12:
            best_f = f_new.copy()
            best_sc = sc
            f = f_new
            lr = min(0.08, lr * 1.02)
        else:
            lr = max(0.005, lr * 0.75)
    
    return best_f, best_sc
