    rng = np.random.default_rng(seed)
    f = np.clip(f0.copy(), 0, None).astype(np.float64)
    S = float(f.sum())
    if S < 1e-10:
        f[:] = 1.0/n
        S = 1.0

        f /= S
        S = 1.0
    
    best_f = f.copy()
    best_score = _score_fast(f)
    conv = _conv_full(f)
    
    t = 0.15
    t_min = 0.03
    t_max = 0.5
    stall = 0
    max_stall = 200
    
    for it in range(max_iters):
        if time.monotonic() > stop_time - 3.0:
            break
        
        M = float(conv.max())
        if M < 1e-15:
            break
        
        # Build constraint matrix
        A = _build_A(f)
        b_ub = np.full(2*n-1, M)
        c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
        
        tl = min(8.0, max(1.0, stop_time - time.monotonic() - 5.0))
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n,
                         method='highs', options={'presolve': True, 'time_limit': tl, 'mip_rel_gap': 0.1})
        except Exception:
            stall += 3
            if stall > max_stall: break
            continue
        
        if res.status != 0 or res.x is None:
            stall += 3
            if stall > max_stall: break
            # Perturb and continue
            f = np.clip(f + rng.normal(0, 0.03, n), 0, None)
            f /= max(1e-10, f.sum())
            conv = _conv_full(f)
            continue
        
        g = np.clip(res.x, 0, None)
        g_sum = float(g.sum())
        if g_sum < 1e-10:
            stall += 3
            if stall > max_stall: break
            continue
        
        g /= g_sum  # Normalize to sum 1
        
        # Check if g is different from f
        dist = np.linalg.norm(g - f)
        if dist < 1e-8:
            stall += 2
            if stall > max_stall: break
            # Try perturbation
            f = np.clip(f + rng.normal(0, 0.05, n), 0, None)
            f /= max(1e-10, f.sum())
            conv = _conv_full(f)
            continue
        
        # Line search for step size
        improved = False
        for _ in range(15):
            f_new = (1-t)*f + t*g
            f_new = np.clip(f_new, 0, None)
            S_new = float(f_new.sum())
            if S_new < 1e-10:
                t = max(t_min, t*0.5)
                continue
            conv_new = _conv_full(f_new)
            sc_new = 2.0*n*float(conv_new.max())/(S_new*S_new)
            if sc_new < best_score - 1e-12:
                f = f_new
                conv = conv_new
                best_f = f.copy()
                best_score = sc_new
                t = min(t_max, t*1.1)
                stall = 0
                improved = True
                break
            else:
                t = max(t_min, t*0.5)
        
        if not improved:
            stall += 1
            if stall >= max_stall:
                f = np.clip(f + rng.normal(0, 0.08, n), 0, None)
                f /= max(1e-10, f.sum())
                conv = _conv_full(f)
                t = 0.15
                stall = 0
    
    return best_f, best_score

def _conv(a):
    """Full autocorrelation via FFT."""
    n = len(a)
    if n <= 256:
        return np.convolve(a, a, mode='full')
    fsize = next_fast_len(2*n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2*n - 1]
    np.maximum(c, 0.0, out=c)
    return c
