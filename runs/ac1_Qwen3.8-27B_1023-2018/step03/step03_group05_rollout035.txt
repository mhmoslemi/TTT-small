def project_simplex(v, target=1.0):
    n = v.size
    if n == 0:
        return np.array([1.0])
    # Project v onto {a >= 0, sum(a) = target}
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    idx = np.arange(1, n+1, dtype=float)
    cond = (css - target) / idx < u
    rho = np.nonzero(cond)[0]
    if rho.size == 0:
        return np.full(n, target/n)
    theta = (css[rho[-1]] - target) / (rho[-1] + 1)
    return np.maximum(v - theta, 0.0)
