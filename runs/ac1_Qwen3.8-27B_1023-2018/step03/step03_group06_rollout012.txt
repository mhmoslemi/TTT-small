class _Timeout(Exception):

_FFT_CACHE = {}


def _fft_len(m: int) -> int:
    m = int(m)
    if m <= 0:
        return 1
    v = _FFT_CACHE.get(m)
    if v is None:
        v = int(next_fast_len(m))
        _FFT_CACHE[m] = v
    return v


def _conv_same(a: np.ndarray) -> np.ndarray:
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.array([], dtype=np.float64)
    L = 2 * n - 1
    F = _fft_len(L)
    fa = rfft(a, F)
    c = irfft(fa * fa, F)[:L]
    return np.maximum(c, 0.0)


def _conv2(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n1, n2 = int(a.size), int(b.size)
    if n1 == 0 or n2 == 0:
        return np.array([], dtype=np.float64)
    L = n1 + n2 - 1
    F = _fft_len(L)
    c = irfft(rfft(a, F) * rfft(b, F), F)[:L]
    return c


def _valid_conv(p: np.ndarray, a: np.ndarray) -> np.ndarray:
    """Return g_i = sum_j p[i+j] * a[j]. Used for gradients of lag-weighted convolution."""
    a = np.asarray(a, dtype=np.float64)
    p = np.asarray(p, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.array([], dtype=np.float64)
    c = _conv2(a[::-1], p)
    return c[n - 1:2 * n - 1]


def _normalize(a) -> np.ndarray:
    a = np.asarray(a, dtype=np.float64).ravel()

def _line_min_quad(A, B, C, lo=0.0, hi=1.0):
    """Minimize max_k (C_k + B_k t + A_k t^2) over t in [lo, hi]."""
    C = np.asarray(C, dtype=np.float64)
    A = np.asarray(A, dtype=np.float64)
    B = np.asarray(B, dtype=np.float64)
    L = int(C.size)
    if L == 0:
        return float(lo), math.inf
    if hi <= lo + 1e-15:
        return float(lo), float(np.max(C))
    A = np.nan_to_num(A, nan=0.0, posinf=0.0, neginf=0.0)
    B = np.nan_to_num(B, nan=0.0, posinf=0.0, neginf=0.0)
    C = np.nan_to_num(C, nan=0.0, posinf=0.0, neginf=0.0)

    N = 81
    ts = np.linspace(lo, hi, N)
    ts2 = ts * ts
    vals = C[None, :] + B[None, :] * ts[:, None] + A[None, :] * ts2[:, None]
    mx = np.max(vals, axis=1)
    if not np.all(np.isfinite(mx)):
        tc = np.array([lo, 0.5 * (lo + hi), hi], dtype=np.float64)
        vals = C[None, :] + B[None, :] * tc[:, None] + A[None, :] * (tc * tc)[:, None]
        mx = np.max(vals, axis=1)
        j = int(np.argmin(mx))
        return float(tc[j]), float(mx[j])

    i = int(np.argmin(mx))
    lo2 = lo if i == 0 else float(ts[i - 1])
    hi2 = hi if i == N - 1 else float(ts[i + 1])

        tsr = np.linspace(lo2, hi2, 49)
        tsr2 = tsr * tsr
        vals = C[None, :] + B[None, :] * tsr[:, None] + A[None, :] * tsr2[:, None]
        mxr = np.max(vals, axis=1)
        if np.all(np.isfinite(mxr)):
            j = int(np.argmin(mxr))
            return float(tsr[j]), float(mxr[j])
    return float(ts[i]), float(mx[i])


def _line_mixture_only(a, b, Ca):
    """Line search a_t = (1-t)a + t b, using Ca=conv(a,a)."""
    a = _normalize(a)
    b = _normalize(b)
    Cb = _conv_same(b)
    Cab = _conv2(a, b)
    A = Ca - 2.0 * Cab + Cb
    B = 2.0 * (Cab - Ca)
    t, m = _line_min_quad(A, B, Ca)
    if not math.isfinite(m):
        return 0.0, float(Ca.max())
    return float(t), float(m)


def _swap_line(a, Ca, d, r):
    """Line search for moving all mass from coordinate d to r; returns (a_new, M_new) if beneficial."""
    a = np.asarray(a, dtype=np.float64)
    Ca = np.asarray(Ca, dtype=np.float64)
    n = int(a.size)
    L = 2 * n - 1
    if d == r or d < 0 or d >= n or r < 0 or r >= n:
        return None
    md = float(a[d])
    if md <= 1e-15:
        return None

    V = np.zeros(L, dtype=np.float64)
    V[r:r + n] = a
    V[d:d + n] -= a

    Q = np.zeros(L, dtype=np.float64)
    Q[2 * d] += 1.0
    Q[2 * r] += 1.0
    Q[d + r] -= 2.0

    A = (md * md) * Q
    B = (2.0 * md) * V
    t, m = _line_min_quad(A, B, Ca)
    if t <= 1e-14 or not math.isfinite(m):
        return None
    if m >= float(Ca.max()) - 1e-14:
        return None
    a_new = a.copy()
    delta = t * md
    a_new[d] -= delta
    a_new[r] += delta
    a_new = np.clip(a_new, 0.0, None)
    a_new = _normalize(a_new)
    return a_new, m


def _dense_endpoint(a, g, max_delta=1.0):
    """Endpoint a + delta*_projected_gradient_minus, used with mixture line search."""
    a = np.asarray(a, dtype=np.float64)
    g = np.asarray(g, dtype=np.float64)
    n = int(a.size)
    if n < 2 or g.size != n:
        return None
    d = g.mean() - g
    d = np.where(np.isfinite(d), d, 0.0)
    tiny = a < 1e-14
    d[tiny] = np.maximum(d[tiny], 0.0)
    d -= d.mean()
    sc = float(np.max(np.abs(d))) if d.size else 0.0
    if sc <= 1e-15:
        return None
    d /= sc
    neg = d < -1e-15
    if neg.any():
        delta = float(np.min(a[neg] / (-d[neg])))
    else:
        delta = float(max_delta)
    delta = min(float(max_delta), delta)
    if delta <= 1e-15:
        return None
    b = a + delta * d
    b = np.clip(b, 0.0, None)
    b = _normalize(b)
    return b


def _top_weights(C, K):
    C = np.asarray(C, dtype=np.float64)
    L = int(C.size)
    if L == 0:
        return np.array([], dtype=np.float64)
    K = int(K)
    if K <= 0 or K >= L:
        return np.ones(L, dtype=np.float64) / float(L)
    idx = np.argpartition(C, -K)[-K:]
    top = C[idx]
    M = float(np.max(top))
    scale = M - float(np.min(top))
    if scale <= 1e-300:
        w = np.ones_like(top, dtype=np.float64)
    else:
        w = np.exp(-((M - top) / scale) * 2.0)
    s = float(w.sum())
    if not math.isfinite(s) or s <= 0.0:
        w = np.ones_like(top, dtype=np.float64)
        s = float(max(1.0, top.size))
    p = np.zeros(L, dtype=np.float64)
    p[idx] = w / s
    return p


def _quad_endpoint_from_diag(P, maxP, mu, ones):
    A = P.copy()
    n = int(A.shape[0])
    if n <= 0:
        return None
    A.flat[::n + 1] += mu * maxP
    try:
        y = np.linalg.solve(A, ones)
    except Exception:
        try:
            y = np.linalg.lstsq(A, ones, rcond=None)[0]
        except Exception:
            return None
    y = np.where(np.isfinite(y), y, 0.0)
    b = np.clip(y, 0.0, None)
    s = float(b.sum())
    if s <= 1e-12:
        b = np.abs(y)
        s = float(b.sum())
        if s <= 1e-12:
            return None
    b = b / s
    return b


def _smooth_refine(a0, stop, maxfun_per=600, betas=None):
    a = _normalize(a0)
    n = int(a.size)
    if n < 2:
        return a, float(_conv_same(a).max())
    best = a.copy()
    best_M = float(_conv_same(best).max())
    if not math.isfinite(best_M):
        best_M = float("inf")

    sc = max(0.75, min(3.0, float(n) / 256.0))
    if betas is None:
        betas = (40.0 * sc, 180.0 * sc, 700.0 * sc, 2800.0 * sc)
    betas = tuple(float(b) for b in betas if math.isfinite(b) and b > 0.0)
    if not betas:
        betas = (50.0, 250.0, 1000.0)

    x = np.log(np.maximum(a, 1e-12))
    x = np.clip(x, -40.0, 40.0)
    bounds = [(-40.0, 40.0)] * n

    def softmax_x(z):
        z = np.asarray(z, dtype=np.float64)
        z = np.nan_to_num(z, nan=-30.0, posinf=30.0, neginf=-30.0)
        z = np.clip(z, -40.0, 40.0)
        m = float(np.max(z))
        ez = np.exp(z - m)
        s = float(ez.sum())
        if not math.isfinite(s) or s <= 0.0:
            return np.ones(n, dtype=np.float64) / float(n)
        return ez / s

    for beta in betas:
        now = time.monotonic()
        if now >= stop - 1.0:
            break
        maxfun = int(min(maxfun_per, max(40, 4.0 * max(1.0, stop - now))))
        maxfun = max(20, maxfun)

        def fg(xx, _beta=beta):
            if time.monotonic() > stop:
                raise _Timeout
            z = np.asarray(xx, dtype=np.float64)
            z = np.nan_to_num(z, nan=-30.0, posinf=30.0, neginf=-30.0)
            z = np.clip(z, -40.0, 40.0)
            m = float(np.max(z))
            ez = np.exp(z - m)
            s = float(ez.sum())
            if not math.isfinite(s) or s <= 0.0:
                return 1e100, np.zeros_like(z)
            aa = ez / s
            C = _conv_same(aa)
            Mc = float(C.max())
            if not math.isfinite(Mc) or Mc <= 0.0:
                return 0.0, np.zeros_like(z)
            args = np.clip(_beta * (C - Mc), -745.0, 0.0)
            w = np.exp(args)
            sw = float(w.sum())
            if not math.isfinite(sw) or sw <= 0.0:
                p = np.zeros(C.size, dtype=np.float64)
                p[int(np.argmax(C))] = 1.0
            else:
                p = w / sw
            val = _beta * Mc + math.log(sw)
            g = 2.0 * _beta * _valid_conv(p, aa)
            if not np.all(np.isfinite(g)):
                g = np.zeros(n, dtype=np.float64)
            dot = float(np.dot(aa, g))
            grad = aa * (g - dot)
            grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
            return float(val), grad

        def cb(xx, _res):
            if time.monotonic() > stop:
                raise _Timeout

        try:
            res = minimize(
                fg,
                x,
                jac=True,
                method="L-BFGS-B",
                bounds=bounds,
                options={
                    "maxiter": maxfun,
                    "maxfun": maxfun,
                    "ftol": 1e-13,
                    "gtol": 1e-9,
                    "maxcor": 10,
                    "maxls": 50,
                    "disp": False,
                },
                callback=cb,
            )
        except _Timeout:
            break
        except Exception:
            break

        if not hasattr(res, "x") or res.x is None:
            break
        x_new = np.nan_to_num(np.asarray(res.x, dtype=np.float64), nan=-20.0, posinf=20.0, neginf=-20.0)
        a_new = softmax_x(x_new)
        C_new = _conv_same(a_new)
        M_new = float(C_new.max())
        if not math.isfinite(M_new):
            break
        if M_new < best_M - 1e-14:
            best = a_new.copy()
            best_M = M_new
            x = x_new
            a = best
        elif getattr(res, "success", False):
            x = x_new
            a = a_new

    M_end = float(_conv_same(a).max())
    if math.isfinite(M_end) and M_end < best_M - 1e-14:
        best = a.copy()
        best_M = M_end
    return best, best_M


def _local_minimax(a0, stop, max_iter=6000, use_quad=True, seed=0):
    a = _normalize(a0)
    n = int(a.size)
    if n < 2:
        return a, float(_conv_same(a).max())
    Ca0 = _conv_same(a)
    best = a.copy()
    best_M = float(Ca0.max())
    if not math.isfinite(best_M):
        return best, best_M

    rng = np.random.default_rng(seed)
    L = 2 * n - 1
    K = max(8, min(80, L // 16))
    ones = np.ones(n, dtype=np.float64)
    noimp = 0
    restarts = 0
    it = 0

    while it < max_iter and time.monotonic() < stop:
        it += 1
        Ca = _conv_same(a)
        M = float(Ca.max())
        if not math.isfinite(M):
            break
        if M < best_M - 1e-14:
            best = a.copy()
            best_M = M
            noimp = 0

        improved = False

        if use_quad and n <= 512:
            p = _top_weights(Ca, K)
            try:
                P = sliding_window_view(p, n).astype(np.float64, copy=False)
                maxP = float(np.max(np.abs(P)))
                maxP = max(1e-15, maxP)
                if noimp > 6:
                    mus = (1e-8, 1e-6, 1e-4, 1e-2, 0.1)
                else:
                    mus = (1e-5, 1e-4) if (it % 3 == 0) else (1e-5,)
                for mu in mus:
                    b = _quad_endpoint_from_diag(P, maxP, float(mu), ones)
                    if b is None:
                        continue
                    if np.max(np.abs(b - a)) < 1e-10:
                        continue
                    t, mnew = _line_mixture_only(a, b, Ca)
                    if t > 1e-14 and mnew < M - 1e-13:
                        a = (1.0 - t) * a + t * b
                        a = _normalize(a)
                        improved = True
                        break
            except Exception:
                pass

        if not improved:
            p = _top_weights(Ca, K)
            g = 2.0 * _valid_conv(p, a)
            if not np.all(np.isfinite(g)):
                g = np.zeros(n, dtype=np.float64)

            b = _dense_endpoint(a, g, max_delta=1.0)
            if b is not None and np.max(np.abs(b - a)) > 1e-10:
                t, mnew = _line_mixture_only(a, b, Ca)
                if t > 1e-14 and mnew < M - 1e-13:
                    a = (1.0 - t) * a + t * b
                    a = _normalize(a)
                    improved = True

            if not improved and n >= 2:
                ks = min(3, n)
                if n <= ks:
                    high = np.arange(n)
                    low = np.arange(n)
                else:
                    high = np.argpartition(g, -ks)[-ks:]
                    low = np.argpartition(g, ks)[:ks]
                pairs = []
                for hd in high:
                    for lr in low:
                        hd, lr = int(hd), int(lr)
                        if hd != lr and float(a[hd]) > 1e-14:
                            pairs.append((hd, lr))
                        if len(pairs) >= 4:
                            break
                    if len(pairs) >= 4:
                        break
                for hd, lr in pairs:
                    res = _swap_line(a, Ca, hd, lr)
                    if res is not None:
                        a_new, mnew = res
                        if mnew < M - 1e-13:
                            a = a_new
                            improved = True
                            break

        if improved:
            noimp = 0
        else:
            noimp += 1
            if noimp > 25:
                restarts += 1
                if restarts > 6:
                    break
                a = best.copy()
                base_mean = max(1e-12, float(a.mean()))
                a = a + 1e-4 * base_mean
                a = a * (1.0 + rng.normal(0.0, 0.08, n))
                a = np.clip(a, 0.0, None)
                a = _normalize(a)
                noimp = 0
                K = int(min(max(4, L // 32), max(8, L // 16)))
                K = max(2, K)
            elif noimp > 10:
                K = max(2, min(128, K // 2))
            elif noimp > 4:
                K = int(max(2, min(128, int(0.75 * K))))

    M_end = float(_conv_same(a).max())
    if math.isfinite(M_end) and M_end < best_M - 1e-14:
        best = a.copy()
        best_M = M_end
    return best, best_M


def _lp_polish(a0, stop, max_iters=25, max_n=512):
    a = _normalize(a0)
    n = int(a.size)
    if n < 2 or n > max_n:
        return a, float(_conv_same(a).max())
    Ca = _conv_same(a)
    best = a.copy()
    best_M = float(Ca.max())
    if not math.isfinite(best_M):
        return best, best_M
    noimp = 0
    it = 0

    L = 2 * n - 1
    b_eq_arr = np.array([1.0], dtype=np.float64)

    while it < max_iters and time.monotonic() < stop:
        it += 1
        if time.monotonic() >= stop - 2.0:
            break
        Ca = _conv_same(a)
        M = float(Ca.max())
        if not math.isfinite(M):
            break
        if M < best_M - 1e-14:
            best = a.copy()
            best_M = M
            noimp = 0

        T = np.zeros((L, n), dtype=np.float64)
        for j, val in enumerate(a):
            if val != 0.0:
                T[j:j + n, j] = val
        A_ub = np.empty((L, n + 1), dtype=np.float64)
        A_ub[:, :n] = T
        A_ub[:, n] = -1.0
        c = np.zeros(n + 1, dtype=np.float64)
        c[n] = 1.0
        b_ub = np.zeros(L, dtype=np.float64)
        A_eq = np.zeros((1, n + 1), dtype=np.float64)
        A_eq[0, :n] = 1.0
        bounds = [(0.0, None)] * n + [(0.0, None)]

        try:
            try:
                res = linprog(
                    c,
                    A_ub=A_ub,
                    b_ub=b_ub,
                    A_eq=A_eq,
                    b_eq=b_eq_arr,
                    bounds=bounds,
                    method="highs",
                    options={"presolve": True},
                )
            except TypeError:
                res = linprog(
                    c,
                    A_ub=A_ub,
                    b_ub=b_ub,
                    A_eq=A_eq,
                    b_eq=b_eq_arr,
                    bounds=bounds,
                    method="highs",
                )
        except Exception:
            break

        if res is None or getattr(res, "status", 1) != 0 or getattr(res, "x", None) is None:
            break

        b = np.asarray(res.x[:n], dtype=np.float64)
        b = np.where(np.isfinite(b), b, 0.0)
        b = np.clip(b, 0.0, None)
        s = float(b.sum())
        if s <= 1e-12:
            break
        b = b / s
        if np.max(np.abs(b - a)) < 1e-10:
            noimp += 1
        else:
            t, mnew = _line_mixture_only(a, b, Ca)
            if t > 1e-14 and mnew < M - 1e-13:
                a = (1.0 - t) * a + t * b
                a = _normalize(a)
                noimp = 0
            else:
                noimp += 1

        if noimp > 5:
            break

    M_end = float(_conv_same(a).max())
    if math.isfinite(M_end) and M_end < best_M - 1e-14:
        best = a.copy()
        best_M = M_end
    return best, best_M


def _random_init(L, rng):
    L = int(L)
    if L <= 1:
        return np.ones(max(2, L), dtype=np.float64) / max(2, L)
    kind = int(rng.integers(0, 6))
    ii = np.arange(L, dtype=np.float64)
    if kind == 0:
        a = rng.dirichlet(1.0, size=L)
    elif kind == 1:
        a = rng.dirichlet(2.5, size=L)
    elif kind == 2:
        a = rng.dirichlet(0.35, size=L)
    elif kind == 3:
        v = (ii + 1.0) / float(L)
        p = float(rng.uniform(0.35, 3.5))
        a = 0.15 + 0.85 * np.power(v, p)
        if rng.random() < 0.5:
            a = a[::-1]
    elif kind == 4:
        c1 = int(rng.integers(0, L))
        c2 = int(rng.integers(0, L))
        if c2 == c1:
            c2 = (c1 + L // 2 + 1) % L
        w1 = max(1.0, float(rng.uniform(0.05, 0.25)) * L)
        w2 = max(1.0, float(rng.uniform(0.05, 0.25)) * L)
        s1 = float(rng.uniform(0.6, 1.4))
        s2 = float(rng.uniform(0.6, 1.4))
        a = s1 * np.exp(-(((ii - c1) / w1) ** 2)) + s2 * np.exp(-(((ii - c2) / w2) ** 2)) + 0.02
    else:
        m = int(rng.integers(max(2, int(0.6 * L)), L + 1))
        mask = np.zeros(L, dtype=np.float64)
        mask[rng.choice(L, size=m, replace=False)] = 1.0
        a = mask * (0.25 + rng.uniform(0.0, 1.0, size=L))
    return _normalize(a)


def _make_inits(L, base, rng, max_count=10):
    L = int(L)
    if L <= 1:
        return (np.ones(L, dtype=np.float64) if L == 1 else np.ones(2, dtype=np.float64) / 2.0,)
    arrs = []
    if base is not None:
        arrs.append(_interp_parent(base, L))
    arrs.append(np.ones(L, dtype=np.float64) / float(L))
    v = (np.arange(L, dtype=np.float64) + 1.0) / float(L)
    for p in (0.4, 1.0, 2.5):
        arrs.append(_normalize(0.15 + 0.85 * np.power(v, p)))
        arrs.append(_normalize(0.15 + 0.85 * np.power(v[::-1], p)))
    for _ in range(8):
        arrs.append(_random_init(L, rng))

    scored = []
    for a in arrs:
        a = _normalize(a)
        if a.size != L:
            continue
        sc = _quick_score(a)
        if math.isfinite(sc):
            scored.append((sc, a))
    if not scored:
        return (np.ones(L, dtype=np.float64) / float(L),)
    scored.sort(key=lambda z: z[0])
    return [z[1] for z in scored[:max_count]]


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = max(10.0, float(budget_s))
    except Exception:
        budget = 1300.0
    hard_deadline = t0 + budget
    if budget > 250.0:
        deadline = hard_deadline - 30.0
    elif budget > 80.0:
        deadline = hard_deadline - 12.0
    else:
        deadline = hard_deadline - 2.5

    parent = _get_parent()
    vertex = np.random.default_rng(int(seed))
    if parent is None:
        parent = _normalize(0.3 + 0.7 * np.abs(vertex.normal(0.0, 1.0, size=256)))

    best_seq = _normalize(parent)
    fallback = _normalize(parent)
    best_score = _exact_score(best_seq)
    if not math.isfinite(best_score):
        best_score = _quick_score(best_seq)
    if not math.isfinite(best_score):
        fallback = np.ones(32, dtype=np.float64) / 32.0
        best_seq = fallback.copy()
        best_score = _exact_score(best_seq)
        if not math.isfinite(best_score):
            best_score = _quick_score(best_seq)

    rng = np.random.default_rng(int(seed) + 17)

    def _try_candidate(arr):
        nonlocal best_seq, best_score
        if arr is None:
            return
        a = _normalize(arr)
        if a.size == 0:
            return
        q = _quick_score(a)
        if not math.isfinite(q):
            return
        # Call the exact evaluator for anything close, so small FFT/internal drifts do not hide gains.
        if q >= best_score + 0.02:
            return
        s = _exact_score(a)
        if not math.isfinite(s):
            s = q
        if math.isfinite(s) and s < best_score - 1e-12:
            best_seq = a.copy()
            best_score = s

    def _finalize():
        seq = np.asarray(best_seq, dtype=np.float64).ravel()
        if seq.size == 0 or float(np.where(np.isfinite(seq), seq, 0.0).sum()) < 1e-8:
            seq = np.ones(32, dtype=np.float64) / 32.0
        seq = np.clip(np.where(np.isfinite(seq), seq, 0.0), 0.0, 1000.0)
        lst = [float(x) for x in seq]
        sc = _exact_score(seq)
        if not math.isfinite(sc):
            fb = np.asarray(fallback, dtype=np.float64).ravel()
            if fb.size > 0:
                fb = np.clip(np.where(np.isfinite(fb), fb, 0.0), 0.0, 1000.0)
                sfb = float(fb.sum())
                if math.isfinite(sfb) and sfb > 1e-8:
                    fb = fb / sfb
                    fb_lst = [float(x) for x in fb]
                    s2 = _exact_score(fb_lst)
                    if math.isfinite(s2):
                        return fb_lst
        return lst

    total = max(10.0, deadline - t0)

    def _time_left():
        return deadline - time.monotonic()

    # Small budgets: do only one fast local polish and return.
    if total < 70.0:
        if time.monotonic() < deadline - 2.0:
            a, _ = _local_minimax(
                best_seq,
                deadline - 2.0,
                max_iter=max(100, int(20.0 * total)),
                seed=int(rng.integers(1, 10_000_000)),
            )
            _try_candidate(a)
        return _finalize()

    # Phase 1: fast local refinement of the parent / current best at its length.
    stop1 = t0 + min(130.0, 0.13 * total)
    stop1 = min(stop1, deadline - max(120.0, 0.25 * total))
    if stop1 > time.monotonic() + 8.0:
        a, _ = _local_minimax(
            best_seq,
            stop1,
            max_iter=7000,
            use_quad=True,
            seed=int(rng.integers(1, 10_000_000)),
        )
        _try_candidate(a)

    # Phase 2: LP best-response polish if dimensions permit.
    base_n = int(best_seq.size)
    stop_lp1 = t0 + min(170.0, 0.24 * total)
    stop_lp1 = min(stop_lp1, deadline - max(180.0, 0.35 * total))
    if base_n <= 512 and stop_lp1 > time.monotonic() + 25.0:
        a, _ = _lp_polish(best_seq, stop_lp1, max_iters=22)
        _try_candidate(a)

    # Phase 3: multi-start smooth + local improvement at the base length.
    stop_smooth = t0 + min(750.0, 0.58 * total)
    stop_smooth = min(stop_smooth, deadline - max(220.0, 0.28 * total))
    if stop_smooth > time.monotonic() + 20.0:
        n_base = int(best_seq.size)
        inits = _make_inits(n_base, best_seq, rng, max_count=9)
        # Sort by quick internal score.
        scored = []
        for a in inits:
            a = _normalize(a)
            if a.size != n_base:
                continue
            q = _quick_score(a)
            if math.isfinite(q):
                scored.append((q, a))
        scored.sort(key=lambda z: z[0])
        for _, a in scored:
            if time.monotonic() >= stop_smooth - 15.0:
                break
            rem = stop_smooth - time.monotonic()
            if rem <= 12.0:
                break
            sub_stop = min(stop_smooth - 5.0, time.monotonic() + max(18.0, min(70.0, 0.22 * rem)))
            a2, _ = _smooth_refine(a, sub_stop, maxfun_per=520)
            _try_candidate(a2)
            if time.monotonic() < stop_smooth - 12.0:
                sub_stop2 = min(stop_smooth - 5.0, time.monotonic() + max(18.0, min(45.0, 0.18 * rem)))
                a3, _ = _local_minimax(
                    a2,
                    sub_stop2,
                    max_iter=4500,
                    use_quad=(int(a2.size) <= 512),
                    seed=int(rng.integers(1, 10_000_000)),
                )
                _try_candidate(a3)

    # Phase 4: randomized diversification at the current best length.
    stop_rand = t0 + min(1050.0, 0.82 * total)
    stop_rand = min(stop_rand, deadline - max(140.0, 0.18 * total))
    n_cur = int(best_seq.size)
    per_rand = 40.0 if n_cur <= 384 else (30.0 if n_cur <= 512 else 20.0)
    while time.monotonic() < stop_rand - 15.0:
        a = np.zeros(0)
        # Blend random with current best when lengths match to keep candidates in good basin.
        rnd = _random_init(n_cur, rng)
        if best_seq.size == n_cur:
            a = _normalize(0.45 * _normalize(best_seq) + 0.55 * rnd)
        else:
            a = rnd
        qa = _quick_score(a)
        if not math.isfinite(qa) or qa >= best_score + 0.55:
            continue
        sub_stop = min(stop_rand - 5.0, time.monotonic() + per_rand)
        a2, _ = _local_minimax(
            a,
            sub_stop,
            max_iter=4000,
            use_quad=(n_cur <= 512),
            seed=int(rng.integers(1, 10_000_000)),
        )
        _try_candidate(a2)
        if time.monotonic() < stop_rand - 20.0 and qa < best_score + 0.2:
            sub_stop2 = min(stop_rand - 5.0, time.monotonic() + max(10.0, 0.4 * per_rand))
            a3, _ = _smooth_refine(a2, sub_stop2, maxfun_per=350)
            _try_candidate(a3)

    # Phase 5: try a few other lengths. Target is scale-dependent, so this can help if the best shape extends well.
    stop_other = deadline - max(90.0, 0.10 * total)
    cur_n = int(best_seq.size)
    candidates = []
    for L in (cur_n + 128, 512, 384, 768, cur_n - 128, 1024):
        L = int(L)
        if 96 <= L <= 1024 and L != cur_n and L not in candidates:
            candidates.append(L)
    candidates.sort(key=lambda L: abs(L - cur_n))
    for L in candidates[:4]:
        if time.monotonic() >= stop_other - 35.0:
            break
        a0 = _interp_parent(best_seq, L)
        alloc = 55.0 if L <= 384 else (45.0 if L <= 512 else 30.0)
        sub_stop = min(stop_other - 15.0, time.monotonic() + alloc)
        a1, _ = _local_minimax(
            a0,
            sub_stop,
            max_iter=5000,
            use_quad=(L <= 512),
            seed=int(rng.integers(1, 10_000_000)),
        )
        _try_candidate(a1)
        if time.monotonic() < stop_other - 25.0:
            sub_stop2 = min(stop_other - 8.0, time.monotonic() + max(12.0, 0.45 * alloc))
            a2, _ = _smooth_refine(a1, sub_stop2, maxfun_per=360)
            _try_candidate(a2)

    # Final polish: use remaining time on the best exact candidate.
    final_stop = deadline - 6.0
    if time.monotonic() < final_stop - 8.0:
        rem = final_stop - time.monotonic()
        nfin = int(best_seq.size)
        local_stop = min(final_stop - 3.0, time.monotonic() + max(5.0, min(70.0, 0.55 * rem)))
        a, _ = _local_minimax(
            best_seq,
            local_stop,
            max_iter=max(2000, int(40.0 * rem)),
            use_quad=(nfin <= 512),
            seed=int(rng.integers(1, 10_000_000)),
        )
        _try_candidate(a)

    if time.monotonic() < final_stop - 8.0 and int(best_seq.size) <= 512:
        rem = final_stop - time.monotonic()
        smooth_stop = min(final_stop - 2.0, time.monotonic() + max(4.0, min(45.0, 0.5 * rem)))
        a, _ = _smooth_refine(best_seq, smooth_stop, maxfun_per=420, betas=(600.0, 2500.0, 8000.0))
        _try_candidate(a)

    if time.monotonic() < final_stop - 12.0 and int(best_seq.size) <= 512:
        rem = final_stop - time.monotonic()
        lp_stop = min(final_stop - 1.0, time.monotonic() + max(5.0, min(35.0, rem)))
        a, _ = _lp_polish(best_seq, lp_stop, max_iters=8)
        _try_candidate(a)

    # Last-minute local squeeze.
    if time.monotonic() < final_stop - 2.0:
        a, _ = _local_minimax(
            best_seq,
            final_stop - 1.0,
            max_iter=max(1000, int(30.0 * max(1.0, final_stop - time.monotonic()))),
            use_quad=(int(best_seq.size) <= 512),
            seed=int(rng.integers(1, 10_000_000)),
        )
        _try_candidate(a)

    if time.monotonic() >= hard_deadline - 8.0:
        return _finalize()

    return _finalize()
