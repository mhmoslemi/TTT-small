    return g / gs


def _parabolic_line_search(f, g_dir, n, best_sc, t_init=0.3, stop_time=None):
    """Parabolic line search for t in (1-t)*f + t*g_dir."""
    best_t, best_f, best_sc_local = 0.0, f.copy(), best_sc
    
    # Try a few initial points
    for t in [t_init * 0.2, t_init * 0.5, t_init, min(1.0, t_init * 1.5)]:
        if stop_time and time.monotonic() > stop_time:
            break
        fn = (1.0 - t) * f + t * g_dir
        np.maximum(fn, 0.0, out=fn)
        fs = float(fn.sum())
        if fs < 1e-10:
            continue
        sc = _score(fn)
        if sc < best_sc_local:
            best_sc_local = sc
            best_t = t
            best_f = fn.copy()
    
    if best_t < 1e-6:
        return None, None
    
    # Refine around best_t
    lo, hi = max(0.0, best_t - t_init * 0.3), min(1.0, best_t + t_init * 0.3)
    for _ in range(15):
        if stop_time and time.monotonic() > stop_time:
            break
        t1 = lo + (hi - lo) * 0.333
        t2 = lo + (hi - lo) * 0.667
        f1 = (1.0 - t1) * f + t1 * g_dir
        np.maximum(f1, 0.0, out=f1)
        f2 = (1.0 - t2) * f + t2 * g_dir
        np.maximum(f2, 0.0, out=f2)
        s1 = float(f1.sum())
        s2 = float(f2.sum())
        sc1 = _score(f1) if s1 > 1e-10 else math.inf
        sc2 = _score(f2) if s2 > 1e-10 else math.inf
        if sc1 < sc2:
            hi = t2
        else:
            lo = t1
        if hi - lo < 1e-4:
            break
    
    t_mid = 0.5 * (lo + hi)
    fn = (1.0 - t_mid) * f + t_mid * g_dir
    np.maximum(fn, 0.0, out=fn)
    sc = _score(fn)
    if sc < best_sc_local:
        best_sc_local = sc
        best_t = t_mid
        best_f = fn.copy()
    
    if best_t < 1e-5:
        return None, None
    return best_t, best_f


def _lp_optimize(f0, n, stop_time, seed):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    fs = float(f.sum())
    if fs < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / fs
    best_f = f.copy()
    best_sc = _score(f)
    t_cur = 0.30
    stall = 0
    max_stall = 40
    momentum_dir = np.zeros(n)
    beta_mom = 0.5
    it = 0
    
    while time.monotonic() < stop_time - 1.0:
        it += 1
        b = _conv_fft(f)
        M = float(b.max())
        if M < 1e-15:
            break
        
        # Solve LP
        g_dir = _lp_solve(f, n, M)
        if g_dir is None:
            stall += 3
            if stall > max_stall:
                # Perturb and restart
                f = np.clip(f + rng.normal(0, 0.05, n), 0, None)
                f = f / f.sum()
                t_cur = 0.25
                stall = 0
            continue
        
        dist = float(np.linalg.norm(g_dir - f))
        if dist < 1e-7:
            stall += 3
            if stall > max_stall:
                f = np.clip(f + rng.normal(0, 0.08, n), 0, None)
                f = f / f.sum()
                t_cur = 0.30
                stall = 0
                momentum_dir = np.zeros(n)
            continue
        
        # Store direction in momentum
        momentum_dir = beta_mom * momentum_dir + (1 - beta_mom) * (g_dir - f)
        effective_dir = g_dir + 0.3 * momentum_dir
        effective_dir = np.maximum(effective_dir, 0)
        ed_sum = effective_dir.sum()
        if ed_sum > 1e-10:
            effective_dir = effective_dir / ed_sum
        
        # Line search
        res_t, f_new = _parabolic_line_search(f, effective_dir, n, best_sc, t_init=min(0.5, t_cur * 1.15), stop_time=stop_time)
        if f_new is None:
            # Try with g_dir directly
            res_t, f_new = _parabolic_line_search(f, g_dir, n, best_sc, t_init=min(0.5, t_cur), stop_time=stop_time)
        
        if f_new is not None:
            sc_new = _score(f_new)
            if sc_new < best_sc - 1e-13:
                f = f_new
                best_f = f.copy()
                best_sc = sc_new
                t_cur = min(0.6, res_t * 1.2)
                stall = 0
            else:
                t_cur = max(0.02, t_cur * 0.7)
                stall += 1
        else:
            t_cur = max(0.02, t_cur * 0.5)
            stall += 1
        
        if stall >= max_stall:
            f = np.clip(f + rng.normal(0, 0.1, n), 0, None)
            f = f / f.sum()
            t_cur = 0.30
            stall = 0
            momentum_dir = np.zeros(n)
    
    return best_f, best_sc


def _grad_optimize(f0, n, stop_time, seed):
    """Subgradient optimization with momentum."""
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    f = f / f.sum()
    best_f = f.copy()
    best_sc = _score(f)
    lr = 0.02
    momentum = np.zeros(n)
    beta = 0.8
    no_imp = 0
    K = 10  # number of top lags to use for subgradient
    
    for it in range(500000):
        if time.monotonic() > stop_time - 1.0:
            break
        
        b = _conv_fft(f)
        M = float(b.max())
        topk = min(K, 2 * n - 1)
        idx = np.argpartition(b, -(2*topk))[-(2*topk):]
        m2 = b[idx]
        # Filter: keep only those above 0.95 * M
        idx = idx[m2 > 0.95 * M]
        if len(idx) == 0:
            idx = np.array([int(np.argmax(b))])
        
        weights = b[idx] - b[idx].min() + 1e-10
        weights /= weights.sum()
        
        grad = np.zeros(n)
        for w, k in zip(weights, idx):
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            for i in range(lo, hi):
                j = k - i
                if 0 <= j < n:
                    grad[i] += w * 2.0 * f[j]
        
        s = float(f.sum())
        # Gradient of score = 2n*M/s^2 - 2n*(a*a_max)/s^2 (directional)
        # Simplified: gradient of 2n*M/s^2 w.r.t. a
        grad_F = (2.0 * n / (s * s)) * grad - (4.0 * n * M / (s**3)) * np.ones(n)
        
        momentum = beta * momentum - lr * grad_F
        f_new = f + momentum
        f_new = np.maximum(f_new, 0.0)
        fs_new = float(f_new.sum())
        if fs_new < 1e-10:
            f_new = f.copy()
        else:
            f_new = f_new / fs_new
        
        sc = _score(f_new)
        if sc < best_sc - 1e-14:
            f = f_new
            best_f = f.copy()
            best_sc = sc
            no_imp = 0
            lr = min(0.08, lr * 1.005)
        else:
            no_imp += 1
            lr = max(0.002, lr * 0.98)
            if no_imp > 500:
                # Restart
                f = f_new.copy()
                no_imp = 0
                lr = 0.015
                beta = 0.7
                momentum = np.zeros(n)
    
    return best_f, best_sc


def _coord_polish(f0, n, stop_time, seed):
    """Coordinate descent with pair moves."""
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    f = f / f.sum()
    best_sc = _score(f)
    no_imp = 0
    multipliers = (0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.7)
    
    while time.monotonic() < stop_time - 1.0:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 1.0:
                return f, best_sc
            orig = f[i]
            for mlt in multipliers:
                f[i] = orig * mlt if orig > 1e-12 else (0.01 if mlt > 0 else 0.0)
                sc = _score(f)
                if sc < best_sc - 1e-13:
                    best_sc = sc
                    improved = True
                    orig = f[i]
            f[i] = orig
        
        # Pair moves
        for _ in range(min(60, n)):
            if time.monotonic() > stop_time - 1.0:
                return f, best_sc
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            oi, oj = f[i], f[j]
            for mi in (0.5, 1.5, 2.0):
                for mj in (0.5, 1.5, 0.3):
                    f[i] = oi * mi if oi > 1e-12 else 0.01
                    f[j] = oj * mj if oj > 1e-12 else 0.01
                    sc = _score(f)
