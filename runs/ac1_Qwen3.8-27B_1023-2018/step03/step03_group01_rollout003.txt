def _clean_arr(arr):

    except Exception:
        return None

        return None
    a = np.where(np.isfinite(a), a, 0.0)

        return None
    return a


def _conv_full(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([0.0], dtype=np.float64)
    L = 2 * n - 1
    N = next_fast_len(L)
    try:
        fa = rfft(a, N)
        c = irfft(fa * fa, N)[:L]
        if not np.all(np.isfinite(c)):
            raise ValueError
    except Exception:
        c = np.convolve(a, a)
    np.maximum(c, 0.0, out=c)
    return c


def _M(a):
    c = _conv_full(a)
    if c.size == 0:
        return 0.0
    m = float(c.max())
    if not math.isfinite(m):
        return math.inf
    return m

    s = float(a.sum())

    return a / s * S


def _from_d(d, S, n):
    z = np.asarray(d, dtype=np.float64).ravel()
    z = z - float(np.max(z))
    e = np.exp(z)
    es = float(e.sum())
    if es <= 0.0 or not math.isfinite(es):

    return S * (e / es)


def _log_d(a):

    z = np.log(a)
    z = z - float(np.mean(z))
    return z

    s = float(x.sum())

def _grad_conv_weighted(a, w):
    a = np.asarray(a, dtype=np.float64).ravel()
    w = np.asarray(w, dtype=np.float64).ravel()
    n = a.size
    L = 2 * n - 1
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    if L == 0:
        return np.ones(n, dtype=np.float64) * 2.0 * float(w.sum()) if w.size else np.zeros(n)
    target_len = n + L - 1
    N = next_fast_len(target_len)
    ar = a[::-1]
    wp = np.zeros(N, dtype=np.float64)
    wp[:L] = w
    try:
        cv = irfft(rfft(ar, N) * rfft(wp, N), N)[:target_len]
        grad = 2.0 * cv[n - 1 : n - 1 + n]
    except Exception:

    grad = np.where(np.isfinite(grad), grad, 0.0)
    np.maximum(grad, 0.0, out=grad)
    return grad


def _smooth_objective(d, beta, S, n):
    z = np.asarray(d, dtype=np.float64).ravel()

    e = np.exp(z)
    es = float(e.sum())

    a = S * (e / es)

    if c.size == 0:
        return 0.0, np.zeros(n, dtype=np.float64)
    cm = float(c.max())

        return 0.0, np.zeros(n, dtype=np.float64)

    u = float(beta) * (c - cm)

            continue
        ai = float(a[i])
        aj = float(a[j])
        if ai <= 1e-12 and aj <= 1e-12:
            continue

        deltas = np.linspace(-0.95 * ai, 0.95 * aj, fracs.size)
        cur = best_m
        best_d = 0.0
        for d in deltas:
            dd = float(d)
            v_i = ai + dd
            v_j = aj - dd
            if v_i < 0.0 or v_j < 0.0:
                continue
            a[i] = v_i
            a[j] = v_j
            m = _M(a)
            if m < cur - 1e-13:
                cur = m
                best_d = dd
        a[i] = ai
        a[j] = aj

        if best_d != 0.0 and cur < best_m - 1e-13:
            a[i] = ai + best_d
            a[j] = aj - best_d
            best = a.copy()
            best_m = cur
            no_improve = 0
        else:
            no_improve += 1

    return best, best_m


def _direct_polish(a0, stop_time, seed):
    n = int(np.size(a0))
    if n <= 0 or n > 900:
        return a0, math.inf
    a = np.asarray(a0, dtype=np.float64).ravel()
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-12:
        return a0, math.inf

    best = a.copy()
    best_sc = _score_direct(a)
    if not math.isfinite(best_sc):
        return best, math.inf

    rng = np.random.default_rng(seed)
    fracs = np.linspace(-0.70, 0.70, 9)
    no_improve = 0

    while time.monotonic() < stop_time - 1.0 and no_improve < 120:
        if no_improve % 50 == 30:
            i = int(np.argmax(a + 1e-9 * rng.random(n)))
            j = int(np.argmin(a + 1e-9 * rng.random(n)))
        else:
            i, j = int(rng.integers(0, n)), int(rng.integers(0, n))
        if i == j:
            continue
        ai = float(a[i])
        aj = float(a[j])
        if ai <= 1e-12 and aj <= 1e-12:
            continue

        deltas = np.linspace(-0.85 * ai, 0.85 * aj, fracs.size)
        cur = best_sc
        best_d = 0.0
        for d in deltas:
            dd = float(d)
            v_i = ai + dd
            v_j = aj - dd
            if v_i < 0.0 or v_j < 0.0:
                continue
            a[i] = v_i
            a[j] = v_j
            sc = _score_direct(a)
            if sc < cur - 1e-12:
                cur = sc
                best_d = dd
        a[i] = ai
        a[j] = aj

        if best_d != 0.0 and cur < best_sc - 1e-12:
            a[i] = ai + best_d
            a[j] = aj - best_d
            best = a.copy()
            best_sc = cur
            no_improve = 0
        else:
            no_improve += 1

    return best, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(1.0, float(budget_s))
    headroom = min(30.0, max(8.0, 0.03 * budget))
    deadline = t0 + budget - headroom
    if deadline < t0 + 5.0:
        deadline = t0 + budget - 1.0

    parent = _parent_array()
    if parent is None or parent.sum() < 0.01:
        parent = np.ones(384, dtype=np.float64)

    candidates = []
    best_arr = None
    best_fft_score = math.inf

    def consider(arr):
        nonlocal best_arr, best_fft_score
        arr = _clean_arr(arr)
        if arr is None:
            return
        try:
            sc = _score_fft(arr)
        except Exception:
            sc = math.inf
        if math.isfinite(sc):
            if sc < best_fft_score - 1e-12:
                best_fft_score = sc
                best_arr = arr.copy()
                candidates.append(arr.copy())
            elif len(candidates) < 64:
                candidates.append(arr.copy())

    consider(parent)
    if best_arr is None:
        parent = np.ones(384, dtype=np.float64)
        consider(parent)
        return [float(x) for x in parent]

    def current_normalized():
        arr = _clean_arr(best_arr)
        if arr is None:
            arr = parent
        n = arr.size
        S = math.sqrt(2.0 * n)
        return _normalize(arr, S)

    try:
        a0 = current_normalized()
        consider(a0)
    except Exception:
        a0 = _normalize(parent, math.sqrt(2.0 * parent.size))
        consider(a0)

    # Phase 1: smooth log-sum-exp polish.
    stop1 = min(deadline, t0 + 0.24 * budget)
    if time.monotonic() < stop1 - 4.0:
        try:
            a1, _ = _smooth_optimize(
                a0,
                stop1,
                seed + 1,
                betas=(5.0, 20.0, 60.0, 200.0),
                maxiter=110,
                replays=1,
            )
            consider(a1)
        except Exception:
            pass
        try:
            a0 = current_normalized()
        except Exception:
            pass

    # Phase 2: LP alternating, closely related to the previous state-of-the-art.
    stop2 = min(deadline, t0 + 0.70 * budget)
    if time.monotonic() < stop2 - 30.0:
        try:
            a2, _ = _lp_optimize(a0, stop2, seed + 101)
            consider(a2)
        except Exception:
            pass
        try:
            a0 = current_normalized()
        except Exception:
            pass

    # Phase 3: high-resolution smoothing polish.
    stop3 = min(deadline, t0 + 0.85 * budget)
    if time.monotonic() < stop3 - 8.0:
        try:
            a3, _ = _smooth_optimize(
                a0,
                stop3,
                seed + 202,
                betas=(30.0, 100.0, 300.0),
                maxiter=110,
                replays=1,
            )
            consider(a3)
        except Exception:
            pass
        try:
            a0 = current_normalized()
        except Exception:
            pass

    # Phase 4: randomized two-coordinate local search.
    stop4 = min(deadline, t0 + 0.93 * budget)
    if time.monotonic() < stop4 - 4.0:
        try:
            a4, _ = _pair_polish(a0, stop4, seed + 303)
            consider(a4)
        except Exception:
            pass
        try:
            a0 = current_normalized()
        except Exception:
            pass

    # Phase 5: try a few other lengths, if time remains.
    remaining = deadline - time.monotonic()
    if remaining > 120.0 and best_fft_score > 1.5028:
        other_lengths = [512, 448, 320, 256]
        per_length = max(30.0, min(70.0, remaining / (len(other_lengths) + 1)))
        for L in other_lengths:
            if time.monotonic() > deadline - 20.0:
                break
            stopL = min(deadline - 20.0, time.monotonic() + per_length)
            if stopL - time.monotonic() < 15.0:
                continue
            try:
                if parent.size > 1:
                    xp = np.interp(
                        np.linspace(0.0, 1.0, L),
                        np.linspace(0.0, 1.0, parent.size),
                        parent,
                    )
                else:
                    xp = np.ones(L, dtype=np.float64)
                xp = np.maximum(xp, 1e-3)
                aL, _ = _smooth_optimize(
                    xp,
                    stopL,
                    seed + 5000 + L,
                    betas=(20.0, 80.0, 250.0),
                    maxiter=70,
                    replays=1,
                )
                consider(aL)
            except Exception:
                pass

    # Phase 6: very short exact-score polish on the best internal candidate.
    try:
        arr_best = _clean_arr(best_arr)
        if arr_best is not None and len(arr_best) <= 900:
            a_exp, _ = _direct_polish(arr_best, deadline - 4.0, seed + 999)
            consider(a_exp)
    except Exception:
        pass

    # Final validation using the authoritative evaluator when possible.
    evaluator = globals().get("evaluate_sequence", None)
    final_candidates = list(candidates)
    for extra in (best_arr, parent, np.ones(384, dtype=np.float64)):
        extra = _clean_arr(extra)
        if extra is not None:
            final_candidates.append(extra)

    best_final = None
    best_final_score = math.inf

    for arr in final_candidates:
        if time.monotonic() > deadline - 2.0:
            break
        arr = _clean_arr(arr)
        if arr is None:
            continue
        lst = [float(x) for x in arr]
        if evaluator is not None:
            try:
                sc = evaluator(lst)
                sc = float(sc)
            except Exception:
                sc = math.inf
        else:
            sc = _score_direct(arr)
        if math.isfinite(sc) and sc < best_final_score:
            best_final_score = sc
            best_final = lst

    if best_final is None:
        best_final = [1.0] * 384

    return best_final
