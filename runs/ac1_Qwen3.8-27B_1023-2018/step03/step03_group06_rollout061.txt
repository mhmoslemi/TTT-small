import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog, minimize

class _Timeout(Exception): pass

_fft_cache = {}

def _fft_len(m):
    m = int(max(m, 1))
    if m not in _fft_cache:
        _fft_cache[m] = int(next_fast_len(m))
    return _fft_cache[m]

def _conv(a, b=None):
    """Full linear convolution. If b is None, compute a*a."""
    if b is None:
        b = a
    n1, n2 = len(a), len(b)
    m = n1 + n2 - 1
    L = _fft_len(m)
    return irfft(rfft(a, L) * rfft(b, L), L)[:m]

def _score(a):
    """Compute the AC1 score. Normalizes sum to 1 internally."""
    a = np.clip(np.asarray(a, dtype=np.float64).ravel(), 0, 1000)
    s = a.sum()
    if s < 1e-12: return math.inf
    a = a / s
    n = len(a)
    c = _conv(a)
    return 2.0 * n * c.max()

def _clean(a, fallback=None):
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a[~np.isfinite(a)] = 0.0
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 1e-12:
        if fallback is not None:
            return fallback.copy()
        n = max(len(a), 32)
        return np.ones(n)/n
    return a / s

def _fL(m):
    m = max(1, int(m))
    v = _fl.get(m)
    if v is None:
        v = int(next_fast_len(m))
        _fl[m] = v
    return v

def _conv(a, b=None):
    if b is None: b = a
    n = len(a); m = len(b)
    L = _fL(n + m - 1)
    return irfft(rfft(a, L) * rfft(b, L), L)[:n+m-1]

    """Score for raw (possibly unnormalized) nonnegative sequence."""
    a = np.clip(np.asarray(a, dtype=np.float64).ravel(), 0.0, 1000.0)

    except:
        return _score_raw(a)

def _lp_iter(f, n):
    """One LP iteration. Returns (g, t_opt) or None."""
    c_ff = _conv(f)
    M = float(c_ff.max())
    if M <= 1e-18: return None
    
    fp = np.maximum(f, 1e-12)
    fp = fp / fp.sum()
    
    A_ub = np.zeros((2*n-1, n+1))
    for j in range(n):
        A_ub[j:j+n, j] = fp
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2*n-1)
    
    obj = np.zeros(n+1); obj[n] = 1.0
    A_eq = np.zeros((1, n+1)); A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, 1.0)]*n + [(0.0, None)]
    
    try:
        res = linprog(obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs', options={'presolve': True, 'time_limit': 10.0})
    except:
        return None
    
    if res is None or res.status != 0 or res.x is None:
        return None
    
    g = np.clip(res.x[:n], 0.0, 1.0)
    gs = g.sum()
    if gs < 1e-12: return None
    g = g / gs
    return g

def _line_search(f, g, c_ff=None, stop_check=None):
    """Find best alpha in [0,1] minimizing max of autocorrelation of (1-alpha)f + alpha*g."""
    if c_ff is None: c_ff = _conv(f)
    c_fg = _conv(f, g)
    c_gg = _conv(g)
    
    ts = np.linspace(0.0, 1.0, 41)
    vals = (1-ts)**2 * c_ff[None,:] + 2*ts*(1-ts) * c_fg[None,:] + ts**2 * c_gg[None,:]
    pk = vals.max(axis=1)
    i0 = int(np.argmin(pk))
    lo, hi = ts[max(0,i0-2)], ts[min(40,i0+2)]
    ts2 = np.linspace(lo, hi, 81)
    vals2 = (1-ts2)**2 * c_ff[None,:] + 2*ts2*(1-ts2) * c_fg[None,:] + ts2**2 * c_gg[None,:]
    pk2 = vals2.max(axis=0)
    t_best = float(ts2[int(np.argmin(pk2))])
    return t_best

def _lbfgs_refine(f0, stop, betas=(50.0, 150.0, 400.0, 800.0), maxfun_per=600):
    a = _clean(f0)
    n = len(a)
    best = a.copy()
    best_sc = _score_norm(best)
    
    ff = np.maximum(a, 1e-15)
    y0 = np.log(ff)
    
    L_fft = _fL(2*n - 1)
    
    for beta in betas:
        if time.monotonic() > stop - 5.0: break
        
        def fg(y):
            if time.monotonic() > stop:
                raise _Timeout
            y = np.nan_to_num(np.clip(np.asarray(y, dtype=np.float64), -60, 60))
            a_s = np.exp(y - y.max())
            a_s = a_s / a_s.sum()
            c = _conv(a_s)
            M = float(c.max())
            if M < 1e-30: return 1e100, np.zeros(n)
            cs = c - M
            ex = np.exp(beta * cs)
            Z = ex.sum()
            w = ex / Z
            fun = beta * M + np.log(Z)
            # Gradient
            fa = fft(a_s, L_fft)
            fw = fft(w, L_fft)
            grad_a = 2.0 * np.real(ifft(np.conj(fa) * fw, L_fft)[:n])
            dot = float(np.dot(a_s, grad_a))
            grad_y = a_s * (grad_a - dot)
            grad_y[~np.isfinite(grad_y)] = 0.0
            return float(fun), grad_y
        
        try:
            res = minimize(fg, y0, jac=True, method='L-BFGS-B',
                         options={'maxiter': 200, 'maxfun': maxfun_per, 'gtol': 1e-8, 'ftol': 1e-12, 'maxls': 50})
        except _Timeout:
            break
        except:
            break
        
        if not hasattr(res, 'x') or res.x is None:
            break
        
        y_new = np.nan_to_num(np.clip(res.x, -60, 60))
        a_new = np.exp(y_new - y_new.max())
        a_new = a_new / a_new.sum()
        sc = _score_norm(a_new)
        
        if math.isfinite(sc) and sc < best_sc:
            best = a_new.copy()
            best_sc = sc
        y0 = y_new
    
    return best, best_sc

def _coord_descend(a0, stop, n_passes=3):
    a = _clean(a0)
    n = len(a)
    best = a.copy()
    best_sc = _score_norm(a)
    
    for _ in range(n_passes):
        if time.monotonic() > stop - 1.0: break
        improved = False
        for i in range(n):
            if time.monotonic() > stop - 1.0: break
            orig = float(a[i])
            s_rest = float(a.sum()) - orig
            
            # Candidate values
            base = max(orig, 1.0/(4*n))
            cands = {orig, 0.0, base*0.25, base*0.5, base*0.75, base*1.25, base*1.5, base*2.0, base*4.0, s_rest/n}
            cands = {max(0.0, float(v)) for v in cands if v is not None}
            
            best_v = orig
            best_sc_i = _score_norm(a)
            
            for v in sorted(cands):
                a[i] = v
                sc = _score_norm(a)
                if math.isfinite(sc) and sc < best_sc_i:
                    best_sc_i = sc
                    best_v = v
            
            if best_v != orig:
                a[i] = best_v
            else:
                a[i] = orig
            
            sc_now = _score_norm(a)
            if math.isfinite(sc_now) and sc_now < best_sc:
                best = a.copy()
                best_sc = sc_now
                improved = True
        
        if not improved: break
    
    return best, best_sc

def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = max(10.0, float(budget_s))
    except:
        budget = 1300.0
    deadline = t0 + budget - 8.0
    
    # Get parent
    parent = None
    try:
        h = globals()['height_sequence_1']
        if h is not None:
            p = _clean(h)
            if p.sum() > 0.01:
                parent = p
    except:
        pass
    
    if parent is None:
        parent = np.ones(256) / 256.0
    
    n = len(parent)
    best_seq = parent.copy()
    best_score = _exact(best_seq)
    if not math.isfinite(best_score):
        best_score = _score_norm(best_seq)
    
    fallback = parent.copy()
    rng = np.random.default_rng(seed)
    
    def try_candidate(a):
        nonlocal best_seq, best_score
        if a is None: return
        a = _clean(a, fallback)
        sc = _exact(a)
        if math.isfinite(sc) and sc < best_score:
            best_seq = a.copy()
            best_score = sc
    
    def finalize():
        seq = np.clip(np.asarray(best_seq, dtype=np.float64).ravel(), 0.0, 1000.0)
        if seq.size == 0 or seq.sum() < 1e-8:
            seq = np.ones(32) / 32.0
        return [float(x) for x in seq]
    
    # Phase 1: LP iterations (35% of budget)
    p1_stop = min(deadline - 0.45 * budget, t0 + 0.35 * budget)
    f = best_seq.copy()
    no_imp = 0
    while time.monotonic() < p1_stop - 5.0:
        g = _lp_iter(f, n)
        if g is None:
            no_imp += 1
            if no_imp >= 5: break
            break
        t_best = _line_search(f, g, stop_check=time.monotonic)
