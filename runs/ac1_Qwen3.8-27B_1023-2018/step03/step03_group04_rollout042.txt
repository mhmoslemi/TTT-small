def _conv(a):

    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]

                    best_val = new_val
                    b = b_new.copy()

                improved = True

"Max sum" LP: max 1^T g s.t. (a*g)_k ≤ M, g ≥ 0
- Finds the "largest" g compatible with current M
- Normalizing g to sum 1 gives a direction that "spreads out" the mass

"Min M" LP: min M s.t. (a*g)_k ≤ M, 1^T g = 1, g ≥ 0
- Finds the tightest M consistent with a unit-sum g
- The optimal g* might be better than the "max sum" g

For the "min M" LP:
- Variables: g_0, ..., g_{n-1}, M (n+1 variables)
- Constraints: (a*g)_k - M ≤ 0 for k = 0, ..., 2n-2 (2n-1 constraints)
- 1^T g = 1 (1 constraint)
- g ≥ 0 (n constraints)
- M ≥ 0 (1 constraint)

The objective is to minimize M. The solution gives us the minimum M achievable by any unit-sum nonnegative g whose cross-convolution with a is bounded by M.

If M* < current_M, then moving toward g* should reduce M. If M* = current_M, we're at a fixed point.

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _conv(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    return c


def _score(a):
    n = len(a)
    S = float(a.sum())
    if S < 0.01:
        return math.inf
    M = float(_conv(a).max())
    if not math.isfinite(M):
        return math.inf
    return 2.0 * n * M / (S * S)


def _build_A(a):
    """Build convolution matrix: (A @ g)_k = (a * g)_k"""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_maxsum(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M, g >= 0. Returns normalized g."""
    n = len(a)
    if M <= 1e-15:
        return None
    A = _build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={"presolve": True,
                               "dual_feasibility_tolerance": 1e-9,
                               "primal_feasibility_tolerance": 1e-9})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    s = float(g.sum())
    if s < 1e-12:
        return None
    return g / s


def _lp_minM(a):
    """LP: min M s.t. (a*g)_k <= M, sum(g)=1, g >= 0. Returns (M_star, g_star)."""
    n = len(a)
    A_conv = _build_A(a)  # (2n-1) x n
    # Variables: [g_0,...,g_{n-1}, M] -> n+1 variables
    # Constraints:
    #   (a*g)_k - M <= 0 for k=0..2n-2  ->  (2n-1) constraints
    #   sum(g) = 1  ->  A_eq
    #   g >= 0, M >= 0
    A_ub = np.hstack([A_conv, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M
    A_eq = np.hstack([np.ones(1, n), [[0.0]]])
    b_eq = [1.0]
    bounds = [(0, None)] * n + [(0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={"presolve": True,
                               "dual_feasibility_tolerance": 1e-9,
                               "primal_feasibility_tolerance": 1e-9})
    except Exception:
        return None, None
    if res.status != 0 or res.x is None:
        return None, None
    g = res.x[:n]
    M_star = res.x[n]
    s = float(g.sum())
    if s < 1e-12:
        return None, None
    g = g / s
    return M_star, g


def _line_search_convex(a, g, max_evals=16):
    """Find best t in [0,1] minimizing max(conv((1-t)a + t*g))."""
    n = len(a)

    def obj(t):
        f = (1.0 - t) * a + t * g
        return float(_conv(f).max())

    # Golden section search
    invphi = (math.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c1 = hi - invphi * (hi - lo)
    c2 = lo + invphi * (hi - lo)
    f1, f2 = obj(c1), obj(c2)
    count = 2

    for _ in range(24):
        if count >= max_evals or hi - lo < 1e-6:
            break
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - invphi * (hi - lo)
            f1 = obj(c1)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + invphi * (hi - lo)
            f2 = obj(c2)
        count += 1

    t_mid = (lo + hi) / 2.0
    # Check endpoints
    f0 = obj(0.0)
    f1e = obj(1.0)
    fm = obj(t_mid)
    count += 3

    best_t = t_mid
    best_f = fm
    if f0 <= best_f:
        best_t, best_f = 0.0, f0
    if f1e <= best_f:
        best_t, best_f = 1.0, f1e

    return best_t, best_f


def _lp_optimize(a0, stop_time, seed, use_minM=False):
    """LP-based optimization loop."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.clip(a0.astype(np.float64), 0, None).copy()
    S = a.sum()
    if S < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / S

    best_a = a.copy()
    best_M = float(_conv(a).max())
    best_score = 2.0 * n * best_M  # S=1

    stall = 0
    max_stall = 300
    prev_dist = 0.0

    while time.monotonic() < stop_time - 1.0:
        b = _conv(a)
        M = float(b.max())

        if use_minM:
            M_star, g = _lp_minM(a)
            if g is None:
                stall += 10
                if stall > max_stall:
                    break
                a = np.clip(a + rng.normal(0, 0.05, n), 0, None)
                a = a / a.sum()
                continue
        else:
            g = _lp_maxsum(a, M)
            if g is None:
                stall += 10
                if stall > max_stall:
                    break
                a = np.clip(a + rng.normal(0, 0.05, n), 0, None)
                a = a / a.sum()
                continue

        # Check staleness
        dist = float(np.linalg.norm(g - a, ord=np.inf))
        if dist < 1e-8:
            stall += 5
            if stall > max_stall:
                break
            # Perturb to escape
            scale = 0.03 + 0.02 * min(stall / 50.0, 2.0)
            a = np.clip(a + rng.normal(0, scale, n), 0, None)
            S = a.sum()
            a = a / S if S > 1e-10 else best_a.copy()
            continue

        # Line search
        t_opt, f_val = _line_search_convex(a, g, max_evals=14)

        a_new = (1.0 - t_opt) * a + t_opt * g
        a_new = np.clip(a_new, 0, None)
        S_new = a_new.sum()
        if S_new < 1e-10:
            stall += 5
            if stall > max_stall:
                break
            continue
        a_new = a_new / S_new

        M_new = float(_conv(a_new).max())
        sc_new = 2.0 * n * M_new

        if sc_new < best_score - 1e-13:
            best_a = a_new.copy()
            best_score = sc_new
            best_M = M_new
            stall = 0
        elif M_new < M - 1e-12:
            stall = 0
        else:
            stall += 1
            if stall >= max_stall:
                # Escape: try best + perturbation
                if rng.random() < 0.4:
                    a = np.clip(best_a + rng.normal(0, 0.04, n), 0, None)
                else:
                    a = np.clip(a + rng.normal(0, 0.12, n), 0, None)
                S = a.sum()
                a = a / S if S > 1e-10 else best_a.copy()
                stall = 0
                continue

        a = a_new

    return best_a, best_score


def _coordinate_polish(a, stop_time, seed=4242):
    """Coordinate descent polish with full convolution recomputation."""
    rng = np.random.default_rng(seed)
    n = len(a)
    a = np.clip(a.astype(np.float64), 0, None).copy()
    S = a.sum()
    if S < 1e-10:
        return a, math.inf
    a = a / S

    best_a = a.copy()
    best_M = float(_conv(a).max())
    best_score = 2.0 * n * best_M

    multipliers = [0.0, 0.3, 0.5, 0.7, 1.3, 1.5, 2.0]
    no_imp_sweeps = 0
    max_no_imp = 6

    while no_imp_sweeps < max_no_imp and time.monotonic() < stop_time - 2.0:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 2.0:
                break
            orig = a[i]
            best_local = orig
            best_local_M = best_M

            for mlt in multipliers:
                if orig < 1e-14:
                    new_val = 0.001 if mlt > 0 else 0.0
                else:
                    new_val = orig * mlt
                if abs(new_val - orig) < 1e-14:
                    continue
                a[i] = new_val
                M_trial = float(_conv(a).max())
                if M_trial < best_local_M - 1e-14:
                    best_local_M = M_trial
                    best_local = new_val
                else:
                    a[i] = best_local

            a[i] = best_local
            if best_local_M < best_M - 1e-14:
                best_M = best_local_M
                improved = True

        # Recompute to be safe
        if improved:
            a = np.clip(a, 0, None)
            S = a.sum()
            a = a / S if S > 1e-10 else best_a.copy()
            M_final = float(_conv(a).max())
            sc_final = 2.0 * n * M_final
            if sc_final < best_score - 1e-14:
                best_score = sc_final
                best_a = a.copy()
                best_M = M_final
            no_imp_sweeps = 0
        else:
            no_imp_sweeps += 1

    return best_a, best_score
