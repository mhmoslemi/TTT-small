    # Tapered
    taper = 0.6 + 0.4 * np.cos(2 * math.pi * i_arr / n)
    inits.append(("taper", np.clip(parent * taper, 0, None)))
    # Multi-freq perturbation
    pert = parent * (1 + 0.06 * np.sin(2 * math.pi * i_arr / 31) + 0.04 * np.cos(2 * math.pi * i_arr / 17))
    inits.append(("multi", np.clip(pert, 0, None)))
    # Ramp-modulated
    ramp = 0.8 + 0.4 * i_arr / max(n - 1, 1)
    inits.append(("ramp", np.clip(parent * ramp, 0, None)))

    time_per_init = max(20.0, (lp_end - t0 - 10) / max(len(inits), 1))
    for idx, (name, init) in enumerate(inits):
        if time.monotonic() > lp_end - 5.0:
            break
        if best_score < 1.5028:
            break
        stop = min(lp_end - 4.0, time.monotonic() + time_per_init)
        try:
            res, sc = _lp_loop(init, n, stop, seed + 1000 * (idx + 1))
            try_update(res)
        except Exception:
            pass

    # Phase 2: Subgradient polish (15%)
    sg_end = min(deadline, t0 + budget * 0.72)
    if time.monotonic() < sg_end - 10.0:
        try:
            res, sc = _subgrad_polish(best_seq, len(best_seq), sg_end, seed + 99001)
            try_update(res)
        except Exception:
            pass

    # Phase 3: Try other lengths (13%)
    other_end = min(deadline, t0 + budget * 0.85)
    if time.monotonic() < other_end - 10.0:
        other_lengths = [192, 256, 448, 128]
        for L in other_lengths:
            if time.monotonic() > other_end - 10.0:
                break
            if best_score < 1.5028:
                break
            if L == n_p:
                continue
            stop = min(other_end - 8.0, time.monotonic() + 40.0)
            lr_ = np.linspace(0, 1, L)
            rr_ = np.linspace(0, 1, n_p)
            init_l = np.interp(lr_, rr_, parent)
            try_update(init_l)
            try:
                res, sc = _lp_loop(init_l, L, stop, seed + 50000 + L)
                try_update(res)
            except Exception:
                pass

    # Phase 4: Coordinate descent polish (12%)
    polish_end = deadline - 5.0
    if time.monotonic() < polish_end - 15.0:
        try:
            res, sc = _coord_polish(best_seq, len(best_seq), polish_end, seed + 77001)
            try_update(res)
        except Exception:
            pass

    # Final subgradient burst
    if time.monotonic() < deadline - 10.0:
        try:
            res, sc = _subgrad_polish(best_seq, len(best_seq), deadline - 8.0, seed + 33001)
            try_update(res)
        except Exception:
            pass

    # Final validation with exact evaluator
    final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    try:
        sc = evaluate_sequence(final_list)
        if math.isfinite(sc) and sc > 0:
            return final_list
    except Exception:
        pass

    fb = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    if fb and all(math.isfinite(x) for x in fb) and sum(fb) > 0.01:
        return fb
    return [1.0] * n_p
