def _analytical_line_search(f, g, cf=None, cg=None, cfg=None):
    """Find alpha in [0,1] minimizing max_k c_k(alpha) along f -> g.
    c_k(alpha) = (1-alpha)^2 * cf_k + 2*alpha*(1-alpha) * cfg_k + alpha^2 * cg_k
    """
    if cf is None:
        cf = np.convolve(f, f)
    if cg is None:
        cg = np.convolve(g, g)
    if cfg is None:
        cfg = np.convolve(f, g)
    
    def obj(alpha):
        t = (1-alpha)**2 * cf + 2*alpha*(1-alpha) * cfg + alpha**2 * cg
        return t.max()
    
    # Coarse grid
    alphas = np.linspace(0, 1, 101)
    vals = np.array([obj(a) for a in alphas])
    idx = np.argmin(vals)
    
    # Golden section around the best
    if idx == 0:
        return 0.0, vals[0]
    if idx == len(alphas) - 1:
        return 1.0, vals[-1]
    
    # Bracket
    lo = alphas[max(0, idx-2)]
    hi = alphas[min(len(alphas)-1, idx+2)]
    
    # Golden section
    gr = (np.sqrt(5) - 1) / 2  # golden ratio ~0.618
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc, fd = obj(c), obj(d)
    
    for _ in range(50):
        if hi - lo < 1e-8:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = obj(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = obj(d)
    
    mid = (lo + hi) / 2
    fm = obj(mid)
    if fm < min(fc, fd):
        return mid, fm
    elif fc < fd:
        return c, fc
    else:
        return d, fd

            A_ub=A,
            b_ub=np.full(2*n-1, M),
            bounds=[(0.0, None)]*n,

            g = np.asarray(res.x, dtype=np.float64)

Actually, I realize I should also try a different approach: **direct non-smooth optimization using subgradients**.

The objective is $F(f) = \max_k (f*f)_k / (\sum f)^2$.

Since $\sum f = 1$ (by normalization), $F(f) = \max_k (f*f)_k$.

The subgradient of $\max_k c_k(f)$ is a convex combination of the gradients $\nabla c_k(f)$ for active $k$ (those with $c_k = \max$).

A simple subgradient method:
- $f_{t+1} = \text{proj}_{\mathcal{F}}(f_t - \eta_t \nabla c_{k^*}(f_t))$

where $\mathcal{F}$ is the feasible set ($f \ge 0$, $\sum f = 1$), and $\eta_t$ is a step size.

This is simpler than the LP approach but might converge slower. The LP approach effectively finds the "best" direction in the face of the active constraint polytope.

The autocorrelation in the time domain corresponds to the power spectrum in the frequency domain:

def _exact_score(a):

    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    c = np.convolve(a, a)
    return float(2.0 * a.size * np.max(c)) / (s * s)


def _build_A(f):
    n = f.size
    m = 2 * n - 1
    kk = np.arange(m)[:, None] - np.arange(n)[None, :]
    mask = (kk >= 0) & (kk < n)
    return np.where(mask, f[kk.clip(0, n - 1)], 0.0)


def _solve_lp(A, M, t_limit=1.0):
    n = A.shape[1]
    try:
        r = linprog(-np.ones(n), A_ub=A, b_ub=np.full(A.shape[0], M),
                    bounds=[(0.0, None)] * n, method='highs',
                    options={'time_limit': t_limit})
        if r.status == 0 and r.x is not None:
            g = np.array(r.x, dtype=np.float64)
            gs = float(g.sum())
            if gs > 1e-14 and np.all(np.isfinite(g)):
                return g / gs
    except Exception:
        pass
    return None


def _ls_quads(cf, cg, cfg):
    def fmax(t):
        a = 1.0 - t
        return float(np.max(a * a * cf + 2.0 * t * a * cfg + t * t * cg))
    ts = np.linspace(0.0, 1.0, 81)
    vs = np.empty(81)
    for i in range(81):
        vs[i] = fmax(ts[i])
    ib = int(np.argmin(vs))

        return 0.0, float(vs[0])

        return 1.0, float(vs[-1])
    lo = ts[max(0, ib - 2)]
    hi = ts[min(80, ib + 2)]

    d_ = lo + gr * (hi - lo)

            fc = fmax(c_)
        else:
            lo = c_
            c_ = d_
            fc = fd
            d_ = lo + gr * (hi - lo)
            fd = fmax(d_)
    tm = 0.5 * (lo + hi)

    for t in (c_, d_):
        v = fmax(t)
        if v < bv:

def _grad(f, k):
    n = f.size
    g = np.zeros(n)
    for j in range(n):
        idx = k - j
        if 0 <= idx < n:
            g[j] = 2.0 * f[idx]
    g -= g.mean()
    return g
