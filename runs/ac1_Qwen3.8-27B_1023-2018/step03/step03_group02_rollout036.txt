import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(384, dtype=np.float64)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0 or p.sum() < 0.01:
        p = np.ones(384, dtype=np.float64)
    return p

def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    # Don't clip to 0 - small negative values from FFT noise are OK
    return c

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(np.max(b))
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)

def _build_A(f, n):
    """Build the (2n-1) x n matrix where row k has f at positions corresponding to (f*g)_k."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A

def _lp_direction(f, n, M, time_limit=5.0):
    """Solve LP: max sum(g) s.t. f*g <= M, g >= 0. Returns normalized g or None."""
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                     method='highs', options={'presolve': True, 'time_limit': time_limit})

        for i in order:
            if time.monotonic() > stop_time - 1.0:
                break

            # Try various multipliers

            if orig > 1e-10:
                candidates = [orig * m for m in (0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0, 3.0)]
            else:
                candidates = [0.001, 0.005, 0.01, 0.05, 0.1, 0.5]
            candidates.append(0.0)

            best_cand_sc = best_sc
            
            for cv in candidates:
                best[i] = cv
                # Renormalize

                if s < 0.01:
                    best[i] = orig

            if best_cand_sc < best_sc - 1e-14:
                best_sc = best_cand_sc
                improved = True

            no_imp += 1
        else:
            no_imp = 0

    best_score = fallback_sc

    rng = np.random.default_rng(seed)

    # Build diverse initializations
    inits = []
    inits.append(parent.copy())
    inits.append(parent[::-1].copy())

    # Smooth random
    base = rng.normal(0, 1, n)
    kern = np.exp(-0.5 * (np.arange(-6, 7) / 2.0) ** 2)
    kern /= kern.sum()
    sm = np.convolve(base, kern, 'same')
    inits.append(np.abs(sm) + 0.01)

    # Parent + moderate noise
    ns = 0.12 * max(1.0, float(np.std(parent)))
    inits.append(np.abs(parent + rng.normal(0, ns, n)) + 0.001)

    # End-emphasized
    i_a = np.arange(n, dtype=np.float64)
    taper = 0.2 + 1.8 * (i_a / max(1, n - 1)) ** 2
    inits.append(taper + 0.01)

    # Phase 1: LP optimization (68% of time)
    phase1_end = t0 + budget * 0.68
    n_inits = len(inits)
    # Give parent the most time
    weights = [0.35, 0.15, 0.15, 0.15, 0.20]
    weights = np.array(weights[:n_inits], dtype=np.float64)
    weights /= weights.sum()
    total_p1 = phase1_end - t0

    for idx, init in enumerate(inits):
        if time.monotonic() > phase1_end - 5.0:
            break
        alloc = weights[idx] * total_p1
        stop = min(phase1_end - 3.0, time.monotonic() + alloc)
        if time.monotonic() >= stop:
            break
        try:
            res, sc = _lp_optimize(init, n, stop, seed=seed + 7919 * (idx + 1))
            if np.all(np.isfinite(res)) and res.sum() > 0.01:
                rl = np.clip(res, 0, 1000)
                exact = float(evaluate_sequence([float(x) for x in rl]))
                if math.isfinite(exact) and exact < best_score:
                    best_seq = rl.copy()
                    best_score = exact
        except Exception:
            pass

    # Phase 2: Try n=512 (15% of time)
    phase2_end = t0 + budget * 0.83
    n_long = 512
    if time.monotonic() < phase2_end - 15.0:
        x_old = np.linspace(0, 1, n)
        x_new = np.linspace(0, 1, n_long)
        parent_long = np.interp(x_new, x_old, parent)
        stop = min(phase2_end - 3.0, time.monotonic() + (phase2_end - phase1_end) * 0.85)
        if time.monotonic() < stop:
            try:
                res, sc = _lp_optimize(parent_long, n_long, stop, seed=seed + 31337)
                if np.all(np.isfinite(res)) and res.sum() > 0.01:
                    rl = np.clip(res, 0, 1000)
                    exact = float(evaluate_sequence([float(x) for x in rl]))
                    if math.isfinite(exact) and exact < best_score:
                        best_seq = rl.copy()
                        best_score = exact
            except Exception:
                pass

    # Phase 3: Polish (remaining time before deadline)
    phase3_stop = deadline - 2.0
    if time.monotonic() < phase3_stop - 5.0:
        try:
            best_seq, best_score = _polish(best_seq, phase3_stop, seed=55555)
        except Exception:
            pass

    # Final validation with exact evaluator
    final_seq = np.clip(best_seq, 0, 1000)
    try:
        final_sc = float(evaluate_sequence([float(x) for x in final_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in fallback]

    return [float(x) for x in final_seq]
