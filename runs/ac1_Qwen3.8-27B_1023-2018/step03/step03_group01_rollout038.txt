                p = p / ps2 if ps2 > 1e-12 else best_p.copy()
                momentum = np.zeros(n)
                lr = 0.004
                beta = 0.7
                no_imp = 0

    return best_p, best_sc

def _coord_polish(p0, n, stop_time, seed):
    rng = np.random.default_rng(seed)
    p = np.asarray(p0, dtype=np.float64).copy()
    ps = p.sum()
    p = p / ps if ps > 1e-12 else np.ones(n) / n

    best_sc = _score_norm(p)
    no_full_no_imp = 0
    multipliers = (0.3, 0.5, 0.7, 1.3, 1.7, 2.5, 0.0)

    while time.monotonic() < stop_time - 1.0:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 1.0:
                break
            orig = p[i]
            for mlt in multipliers:
                if mlt == 0.0 and orig < 1e-12:
                    continue
                p[i] = orig * mlt if mlt > 0 else 0.0
                if orig < 1e-12 and mlt > 0:
                    p[i] = 0.01
                ps = p.sum()
                if ps < 1e-10:
                    p[i] = orig
                    continue
                p_tmp = p / ps
                sc = _score_norm(p_tmp)
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    p = p_tmp
                    improved = True
                    orig = p[i]
            p[i] = orig if orig > 0 else p[i]
            # Rescale back
            ps = p.sum()
            if ps > 1e-12:
                p = p / ps

        if not improved:
            no_full_no_imp += 1
            if no_full_no_imp >= 5:
                break
        else:
            no_full_no_imp = 0

    return p, best_sc

def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 12.0

    parent = _get_parent()
    n_parent = len(parent)

    best_seq = parent.copy()
    best_sc = _score_norm(parent)

    def try_update(p):
        nonlocal best_seq, best_sc
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        if p.size == 0:
            return
        p = np.maximum(p, 0.0)
        s = p.sum()
        if s < 1e-10:
            return
        p = p / s
        sc = _score_norm(p)
        if math.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_seq = p.copy()

    rng_main = np.random.default_rng(seed)
    i_arr = np.arange(n_parent, dtype=np.float64)

    # Build initializations
    inits = [("parent", parent.copy())]

    # Cosine
    inits.append(("cos", 0.3 + 0.7 * (0.5 + 0.5 * np.cos(2 * math.pi * i_arr / n_parent))))

    # Multi-frequency
    sup = np.full(n_parent, 0.15)
    for fr in (1, 2, 3, 5, 8):
        ph = rng_main.uniform(0, 2 * math.pi)
        sup += 0.25 * np.abs(np.cos(2 * math.pi * fr * i_arr / n_parent + ph))
    inits.append(("mfreq", sup))

    # Random positive
    inits.append(("rand", np.abs(rng_main.normal(0, 1, n_parent)) + 0.05))

    # Parent + noise
    ns = 0.5 * max(1.0, float(np.mean(parent)))
    inits.append(("pnoise", np.maximum(parent + rng_main.normal(0, ns, n_parent), 0.01)))

    # Sinc-like
    sinc = np.sinc((i_arr - n_parent / 2) / (n_parent * 0.25))
    inits.append(("sinc", np.abs(sinc) + 0.05))

    # Triangular
    tri = 1.0 - 2.0 * np.abs(i_arr - n_parent / 2) / n_parent
    inits.append(("tri", np.maximum(tri, 0.05)))

    # Phase 1: LP iterations (60% of budget)
    lp_end = t0 + budget * 0.60
    time_per_init = max(12.0, (lp_end - t0 - 20) / len(inits))

    for idx, (name, init) in enumerate(inits):
        if time.monotonic() > lp_end - 5.0:
            break
        if best_sc < 1.5030:
            break
        stop = min(lp_end - 4.0, time.monotonic() + time_per_init)
        try:
            res, sc = _lp_loop(init, n_parent, stop, seed + 1000 * (idx + 1))
            try_update(res)
        except Exception:
            pass

    # Phase 2: Additional LP from best with different perturbations (10%)
    lp2_end = min(deadline, t0 + budget * 0.70)
    if time.monotonic() < lp2_end - 10.0 and best_sc > 1.5030:
        for rep in range(3):
            if time.monotonic() > lp2_end - 5.0:
                break
            pert = np.maximum(best_seq + rng_main.normal(0, 0.05, n_parent), 0)
            pert = pert / pert.sum()
            stop = min(lp2_end - 4.0, time.monotonic() + 30.0)
            try:
                res, sc = _lp_loop(pert, n_parent, stop, seed + 50000 + rep)
                try_update(res)
            except Exception:
                pass

    # Phase 3: Subgradient polish (12%)
    sg_end = min(deadline, t0 + budget * 0.82)
    if time.monotonic() < sg_end - 10.0:
        try:
            res, sc = _subgrad_polish(best_seq, n_parent, sg_end, seed + 99001)
            try_update(res)
        except Exception:
            pass

    # Phase 4: Coordinate descent polish (10%)
    cd_end = min(deadline, t0 + budget * 0.92)
    if time.monotonic() < cd_end - 10.0:
        try:
            res, sc = _coord_polish(best_seq, n_parent, cd_end, seed + 77001)
            try_update(res)
        except Exception:
            pass

    # Phase 5: Final subgradient burst
    if time.monotonic() < deadline - 8.0:
        try:
            res, sc = _subgrad_polish(best_seq, n_parent, deadline - 5.0, seed + 33001)
            try_update(res)
        except Exception:
            pass

    # Final validation with exact evaluator
    result = [float(min(1000.0, max(0.0, x))) for x in best_seq]
    try:
        sc = evaluate_sequence(result)
        if math.isfinite(sc) and sc > 0:
            return result
    except Exception:
        pass

    # Fallback
    fb = [float(min(1000.0, max(0.0, x))) for x in parent]
    try:
        sc = evaluate_sequence(fb)
        if math.isfinite(sc) and sc > 0:
            return fb
    except Exception:
        pass

    return [1.0] * n_parent
