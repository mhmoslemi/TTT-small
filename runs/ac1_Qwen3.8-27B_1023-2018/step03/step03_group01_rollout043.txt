def _project_simplex(z):
    """Project z onto {y: sum(y)=1, y>=0}."""
    n = len(z)
    # Sort z in decreasing order
    u = np.sort(z)[::-1]

    rho = np.where(u * np.arange(1, n+1) - cv > 0)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = cv[rho] / (rho + 1)
    return np.maximum(z - theta, 0)

    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    gs = float(g.sum())
    if gs < 1e-14:
        return None
    return g / gs


def _line_search(f, g_dir, n, best_sc, t_init=0.3, stop_time=None):
    t_lo, t_hi = 0.0, t_init
    f_lo = f.copy()
    for _ in range(14):
        if stop_time and time.monotonic() > stop_time:
            break
        t_mid = 0.5 * (t_lo + t_hi)
        fn = (1.0 - t_mid) * f + t_mid * g_dir
        np.maximum(fn, 0.0, out=fn)
        if fn.sum() < 1e-10:
            t_hi = t_mid
            continue
        sc = _score(fn)
        if sc < best_sc - 1e-14:
            t_lo = t_mid
            f_lo = fn.copy()
        else:
            t_hi = t_mid
        if t_hi - t_lo < 5e-5:
            break
    if t_lo < 1e-5:
        return None, None
    return t_lo, f_lo


def _lp_loop(f0, n, stop_time, seed):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    fs = float(f.sum())
    if fs < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / fs
    best_f = f.copy()
    best_sc = _score(f)
    t_cur = 0.25
    stall = 0
    max_stall = 60

    while time.monotonic() < stop_time - 3.0:
        b = _conv_fft(f)
        M = float(b.max())
        if M < 1e-15:
            break

        g_dir = _lp_solve_reduced(f, n, M, K=20, time_limit=12.0)

        if g_dir is None:
            stall += 4
            if stall > max_stall:
                break
            f = np.clip(f + rng.normal(0, 0.06, n), 0, None)
            s2 = f.sum()
            f = f / s2 if s2 > 1e-10 else best_f.copy()
            t_cur = 0.25
            continue

        dist = float(np.linalg.norm(g_dir - f))
        if dist < 1e-7:
            stall += 3
            if stall > max_stall:
                break
            f = np.clip(f + rng.normal(0, 0.08, n), 0, None)
            s2 = f.sum()
            f = f / s2 if s2 > 1e-10 else best_f.copy()
            t_cur = 0.25
            continue

        res_t, f_new = _line_search(f, g_dir, n, best_sc,
                                     t_init=min(0.5, max(0.05, t_cur)),
                                     stop_time=stop_time)
        if f_new is not None:
            sc_new = _score(f_new)
            if sc_new < best_sc - 1e-14:
                f = f_new
                best_f = f.copy()
                best_sc = sc_new
                t_cur = min(0.5, res_t * 1.1)
                stall = 0
            else:
                t_cur = max(0.02, res_t * 0.5)
                stall += 1
        else:
            t_cur = max(0.02, t_cur * 0.5)
            stall += 1

        if stall >= max_stall:
            f = np.clip(f + rng.normal(0, 0.1, n), 0, None)
            s2 = f.sum()
            f = f / s2 if s2 > 1e-10 else best_f.copy()
            t_cur = 0.25
            stall = 0

    return best_f, best_sc


def _subgrad_polish(f0, n, stop_time, seed, topk=12):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    fs = float(f.sum())
    f = f / fs
    best_f = f.copy()
    best_sc = _score(f)
    lr = 0.005
    momentum = np.zeros(n)
    beta = 0.82
    no_imp = 0

    while time.monotonic() < stop_time - 2.0:
        b = _conv_fft(f)
        M = float(b.max())
        k = min(topk, 2 * n - 1)
        idx = np.argpartition(b, -k)[-k:]
        weights = b[idx].copy()
        wmin = weights.min()
        weights = weights - wmin + 1e-10
        weights /= weights.sum()

        grad_M = np.zeros(n)
        for w, kk in zip(weights, idx):
            kk = int(kk)
            lo = max(0, kk - n + 1)
            hi = min(n, kk + 1)
            for i in range(lo, hi):
                j = kk - i
                if 0 <= j < n:
                    grad_M[i] += w * 2.0 * f[j]

        s = float(f.sum())
        grad_F = (2.0 * n / (s * s)) * grad_M - (4.0 * n * M / (s**3)) * np.ones(n)

        momentum = beta * momentum - lr * grad_F
        f_raw = f + momentum
        f_new = _project_simplex(f_raw)

        sc = _score(f_new)
        if sc < best_sc - 1e-15:
            f = f_new
            best_f = f.copy()
            best_sc = sc
            no_imp = 0
            lr = min(0.02, lr * 1.003)
        else:
            no_imp += 1
            if no_imp > 300:
                lr = max(0.0005, lr * 0.93)
            if no_imp > 1500:
                f = f_new.copy()
                no_imp = 0
                lr = 0.004
                beta = 0.7
                momentum = np.zeros(n)

    return best_f, best_sc


def _coord_polish(f0, n, stop_time, seed):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    fs = float(f.sum())
    f = f / fs
    best_sc = _score(f)
    no_full = 0

    while time.monotonic() < stop_time - 2.0:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 2.0:
                return f, best_sc
            orig = f[i]
            best_local = orig
            best_local_sc = best_sc
            for mlt in (0.7, 0.85, 1.15, 1.3, 0.5, 1.5):
                f[i] = orig * mlt if orig > 1e-12 else 0.01
                sc = _score(f)
                if sc < best_local_sc - 1e-14:
                    best_local_sc = sc
                    best_local = f[i]
            for frac in (0.95, 1.05, 0.9, 1.1):
                f[i] = best_local * frac
                sc = _score(f)
                if sc < best_local_sc - 1e-14:
                    best_local_sc = sc
                    best_local = f[i]
            f[i] = best_local
            if best_local_sc < best_sc - 1e-14:
                best_sc = best_local_sc
                improved = True

        for _ in range(25):
            if time.monotonic() > stop_time - 2.0:
                return f, best_sc
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            oi, oj = f[i], f[j]
            for mi in (0.6, 1.4):
                for mj in (0.6, 1.4):
                    f[i] = oi * mi if oi > 1e-12 else 0.01
                    f[j] = oj * mj if oj > 1e-12 else 0.01
                    sc = _score(f)
                    if sc < best_sc - 1e-14:
                        best_sc = sc
                        improved = True
            f[i], f[j] = oi, oj

        if not improved:
            no_full += 1
            if no_full >= 6:
                break
        else:
            no_full = 0

    return f, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 22.0

    parent = _get_parent()
    n_p = len(parent)

    best_seq = parent.copy()
    best_score = _score(best_seq)
    if not math.isfinite(best_score):
        best_seq = np.ones(384) / 384.0
        n_p = 384
        best_score = _score(best_seq)

    def _try_update(cand):
        nonlocal best_seq, best_score
        if cand is None:
            return
        cand = np.clip(np.asarray(cand, dtype=np.float64).ravel(), 0, 1000)
        if cand.size == 0 or cand.sum() < 0.01:
            return
        try:
            sc = _score(cand)
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-15:
            best_seq = cand.copy()
            best_score = sc

    n = n_p
    i_arr = np.arange(n, dtype=np.float64)
    rng_main = np.random.default_rng(seed)

    inits = []
    inits.append(("parent", parent.copy()))
    inits.append(("cos", 0.5 + 0.5 * np.cos(2 * math.pi * i_arr / n)))
    inits.append(("cos2", 0.5 + 0.5 * np.cos(4 * math.pi * i_arr / n)))
    tri = 1.0 - 2.0 * np.abs(i_arr - n / 2.0) / n
    inits.append(("tri", np.maximum(tri, 0.01)))
    x = (i_arr - n / 2.0) / (n * 0.25)
    sinc = np.sinc(x)
    inits.append(("sinc", np.abs(sinc) + 0.05))
    inits.append(("rand", np.abs(rng_main.normal(0, 1, n)) + 0.05))
    ns = 0.5 * max(1.0, float(np.mean(parent)))
    inits.append(("pnoise", np.abs(parent + rng_main.normal(0, ns, n)) + 0.01))
    g2 = np.exp(-0.5 * ((i_arr - n / 2.0) / (n * 0.18)) ** 2)
    inits.append(("gauss", g2 + 0.05))

    lp_end = t0 + budget * 0.48
    time_per_init = max(18.0, (lp_end - t0 - 12) / len(inits))

    for idx, (name, init) in enumerate(inits):
        if time.monotonic() > lp_end - 6.0:
            break
        stop = min(lp_end - 4.0, time.monotonic() + time_per_init)
        try:
            res, sc = _lp_loop(init, n, stop, seed + 1000 * (idx + 1))
            _try_update(res)
        except Exception:
            pass

    sg_end = min(deadline, t0 + budget * 0.68)
    if time.monotonic() < sg_end - 12.0:
        try:
            res, sc = _subgrad_polish(best_seq, len(best_seq), sg_end, seed + 99001, topk=15)
            _try_update(res)
        except Exception:
            pass

    other_end = min(deadline, t0 + budget * 0.83)
    if time.monotonic() < other_end - 15.0 and best_score > 1.503:
        other_lengths = sorted(set([L for L in (n - 16, n - 8, n + 8, n + 16, n + 32, n - 32) if 64 <= L <= 1000]))
        for L in other_lengths:
            if time.monotonic() > other_end - 8.0:
                break
            if L == n:
                continue
            stop = min(other_end - 5.0, time.monotonic() + 25.0)
            init_l = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, n), parent)
            try:
                res, sc = _lp_loop(init_l, L, stop, seed + 50000 + L)
                _try_update(res)
            except Exception:
                pass

    polish_end = deadline - 15.0
    if time.monotonic() < polish_end - 15.0:
        try:
            res, sc = _coord_polish(best_seq, len(best_seq), polish_end, seed + 77001)
            _try_update(res)
        except Exception:
            pass

    if time.monotonic() < deadline - 12.0:
        try:
            res, sc = _subgrad_polish(best_seq, len(best_seq), deadline - 8.0, seed + 33001, topk=20)
            _try_update(res)
        except Exception:
            pass

    if time.monotonic() < deadline - 10.0:
        try:
            res, sc = _coord_polish(best_seq, len(best_seq), deadline - 6.0, seed + 55001)
            _try_update(res)
        except Exception:
            pass

    final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    try:
        sc = evaluate_sequence(final_list)
        if math.isfinite(sc) and sc > 0:
            return final_list
    except Exception:
        pass

    fb = [float(max(0.0, min(1000.0, x))) for x in parent]
    try:
        sc = evaluate_sequence(fb)
        if math.isfinite(sc) and sc > 0:
            return fb
    except Exception:
        pass

    return [1.0] * n_p
