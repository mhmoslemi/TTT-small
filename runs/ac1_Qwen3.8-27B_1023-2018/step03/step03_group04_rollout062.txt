    c = irfft(F * F, fsize)[:2*n-1]
    np.maximum(c, 0.0, out=c)

    return 2.0 * len(a) * _conv(a).max() / (S*S)

def _subgradient(f, conv_f, thresh_frac=0.0):
    """Compute subgradient of max(conv) w.r.t. f.
    Returns subgradient vector of same length as f."""
    n = len(f)
    M = conv_f.max()
    if M < 1e-30:
        return np.zeros(n)
    # Active set: lags within thresh_frac*M of the max
    thresh = M * (1.0 - thresh_frac)
    active = np.where(conv_f >= thresh)[0]
    # For each active lag k, the gradient w.r.t. f[j] is 2*f[k-j] (if 0<=k-j<n)
    # subgrad[j] = (1/|active|) * sum_{k in active} 2*f[k-j] (if valid)
    # This is (2/|active|) * (indicator_active * f)[j] (convolution, truncated)
    ind = np.zeros(len(conv_f))
    ind[active] = 1.0
    # conv(ind, f) gives sum_k ind[k]*f[k-j] for each j
    # We only need the first n values
    fsize = next_fast_len(len(conv_f) + n - 1)
    F1 = rfft(ind, fsize)
    F2 = rfft(f, fsize)
    conv_if = irfft(F1 * F2, fsize)[:n]  # First n values
    return (2.0 / len(active)) * conv_if

def _lp_direction(f, M):
    """Find g maximizing sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)
    # Build A: A[k,j] = f[k-j] if 0<=k-j<n else 0
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = f
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    res = linprog(c, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)]*n,
                  method='highs')

        return None
    g = res.x
    s = g.sum()
    if s < 1e-15:
        return None
    return g / s

    n = len(a)
    best = np.clip(a.copy(), 0, None)
    fs = best.sum()

    best_sc = _score_fast(best)

    while time.monotonic() < stop_time - 1.0:

            orig = best[i]
            best_val = orig
            best_val_sc = best_sc

                if sc < best_val_sc - 1e-14:

                    best_val_sc = sc
            best[i] = best_val
            if best_val_sc < best_sc - 1e-14:
                best_sc = best_val_sc
                improved = True

            if no_imp >= max_no_imp:

def _conv(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score_fast(a):
    S = float(a.sum())
    if S < 0.01:
        return math.inf
    M = float(_conv(a).max())
    if not math.isfinite(M):
        return math.inf
    return 2.0 * len(a) * M / (S * S)


def _build_A(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_direction(f, M):
    n = len(f)
    if M < 1e-15:
        return None
    A = _build_A(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={"presolve": True})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    s = float(g.sum())
    if s < 1e-15:
        return None
    return g / s


def _subgradient(f, conv_f, top_frac=0.15):
    n = len(f)

    ind = np.zeros(len(conv_f))

            no_imp += 1
            if no_imp >= max_no_imp:
                break
        else:
            no_imp = 0

    return best, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 12.0  # Headroom

    parent = _get_parent()
    n = len(parent)

    # Fallback
    fallback = parent.copy()
    try:
        fallback_sc = float(evaluate_sequence([float(x) for x in fallback]))
    except Exception:
        fallback_sc = math.inf
    if not math.isfinite(fallback_sc):
        fallback = np.ones(384, dtype=np.float64)
        try:
            fallback_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            fallback_sc = math.inf

    best_seq = fallback.copy()
    best_score = fallback_sc

    def _track(f):
        nonlocal best_seq, best_score
        if f is None or not np.all(np.isfinite(f)) or len(f) == 0:
            return
        f = np.clip(f, 0.0, 1000.0)
        if f.sum() < 0.01:
            return
        try:
            exact = float(evaluate_sequence([float(x) for x in f]))
        except Exception:
            exact = math.inf
        if math.isfinite(exact) and exact < best_score:
            best_seq = f.copy()
            best_score = exact

    # Phase 1: Subgradient warm-up (10% budget)
    p1_end = t0 + budget * 0.10
    if time.monotonic() < p1_end - 2:
        try:
            res, sc = _subgrad_optimize(parent, p1_end, seed)
            _track(res)
        except Exception:
            pass

    # Phase 2: Main LP optimization (50% budget)
    p2_end = t0 + budget * 0.60
    if time.monotonic() < p2_end - 2:
        try:
            res, sc = _lp_optimize(best_seq, p2_end, seed + 1)
            _track(res)
        except Exception:
            pass

    # Phase 3: Subgradient refinement (10% budget)
    p3_end = t0 + budget * 0.70
    if time.monotonic() < p3_end - 2:
        try:
            res, sc = _subgrad_optimize(best_seq, p3_end, seed + 2)
            _track(res)
        except Exception:
            pass

    # Phase 4: Another LP round with perturbed start (12% budget)
    p4_end = t0 + budget * 0.82
    if time.monotonic() < p4_end - 3:
        rng4 = np.random.default_rng(seed + 100)
        p4_alocal = np.clip(best_seq + rng4.normal(0, 0.05, n), 0, None)
        s4 = p4_alocal.sum()
        p4_alocal = p4_alocal / s4 if s4 > 0.01 else best_seq.copy()
        try:
            res, sc = _lp_optimize(p4_alocal, p4_end, seed + 3)
            _track(res)
        except Exception:
            pass

    # Phase 5: Try alternative initializations (5% budget)
    p5_end = t0 + budget * 0.87
    if time.monotonic() < p5_end - 3:
        i_arr = np.arange(n, dtype=np.float64)
        alts = [
            np.exp(-i_arr / (n * 0.4)) + 0.005,
            0.5 + 0.5 * np.cos(2 * np.pi * i_arr / n) + 0.1,
            np.exp(-((i_arr - n * 0.2) / (n * 0.12)) ** 2) + np.exp(-((i_arr - n * 0.8) / (n * 0.12)) ** 2) + 0.005,
        ]
        for idx, alt in enumerate(alts):
            if time.monotonic() > p5_end - 3:
                break
            alt_stop = min(p5_end - 2, time.monotonic() + (p5_end - max(time.monotonic(), t0)) / (len(alts) - idx))
            if time.monotonic() >= alt_stop:
                continue
            try:
                res, sc = _lp_optimize(alt, alt_stop, seed + 200 + idx)
                _track(res)
            except Exception:
                pass

    # Phase 6: Coordinate descent polish (remaining)
    polish_end = deadline - 5.0
    if time.monotonic() < polish_end - 5:
        try:
            res, sc = _coordinate_polish(best_seq, polish_end, seed + 99)
            _track(res)
        except Exception:
            pass

    # Final subgradient micro-polish
    if time.monotonic() < deadline - 8:
        try:
            res, sc = _subgrad_optimize(best_seq, deadline - 6, seed + 777)
            _track(res)
        except Exception:
            pass

    # Final coordinate polish
    if time.monotonic() < deadline - 4:
        try:
            res, sc = _coordinate_polish(best_seq, deadline - 3, seed + 555)
            _track(res)
        except Exception:
            pass

    # Final validation
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        return [float(x) for x in fallback]

    return [float(x) for x in best_seq]
