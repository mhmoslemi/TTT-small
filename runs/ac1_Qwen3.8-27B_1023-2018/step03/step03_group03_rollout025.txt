def _get_parent():

def _autocorr(a):
    n = len(a)
    if n <= 256:
        return np.convolve(a, a, mode='full')
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _autocorr(a)
    m = float(c.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _lp_direction(a, time_limit=4.0):
    n = len(a)
    # Mickey Mouse LP: max 1^T g  s.t. A g <= 1, g >= 0
    # where A[k,j] = a[j] if k is in [j, j+n-1] (i.e., (a*g)[k] includes a[j]*g[j])
    # Actually (a*g)[k] = sum_j a[j] * g[k-j] for valid j, k-j
    # = sum_j a[k-j] * g[j] for valid k-j, j
    # So A[k,j] = a[k-j] if 0<=k-j<=n-1 and 0<=j<=n-1
    
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j+n, j] = a
    
    c_obj = -np.ones(n)
    b_ub = np.ones(2 * n - 1)
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                     bounds=[(0, None)] * n,
                     method='highs',
                     options={'presolve': True, 'time_limit': time_limit,
                               'dual_refine': True})
    except Exception:
        return None
    
    if res.status != 0 or res.x is None:
        return None
    
    g = np.asarray(res.x, dtype=np.float64)
    gs = float(g.sum())
    if gs < 1e-15:
        return None
    return g / gs

def _block_descent(a, stop_time, seed=0, max_rounds=50):
    """Try 2-coordinate (block) moves."""
    n = len(a)
    a = np.clip(a.astype(np.float64).copy(), 0, None)
    s = float(a.sum())
    if s < 1e-10:
        return a, math.inf
    best_score = _score(a)
    rng = np.random.default_rng(seed)
    
    for rnd in range(max_rounds):
        if time.monotonic() > stop_time - 3.0:
            break
        improved = False
        
        for _ in range(100):
            if time.monotonic() > stop_time - 2.0:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            
            oi, oj = a[i], a[j]
            s_rest = s - oi - oj
            
            # Try a grid of (x, y) values
            x_grid = np.linspace(0, max(oi, s/n) * 2 + s/n*0.5, 12)
            y_grid = np.linspace(0, max(oj, s/n) * 2 + s/n*0.5, 12)
            
            best_obj = best_score
            best_xy = (oi, oj)
            
            for x in x_grid:
                for y in y_grid:
                    s_new = s_rest + x + y
                    if s_new < 0.01:
                        continue
                    a[i], a[j] = x, y
                    sc = _score(a)
                    if sc < best_obj - 1e-14:
                        best_obj = sc
                        best_xy = (x, y)
            
            a[i], a[j] = best_xy
            if best_obj < best_score - 1e-14:
                best_score = best_obj
                s = s - oi - oj + best_xy[0] + best_xy[1]
                improved = True

            break

    return a, _score(a)

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(384, dtype=np.float64)

    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)

def _autocorr(a):
    n = len(a)
    if n <= 256:
        c = np.convolve(a, a, mode='full')
    else:
        fsize = next_fast_len(2 * n - 1)
        F = rfft(a, fsize)
        c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score_fast(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return math.inf
    c = _autocorr(a)
    m = c.max()
    if not np.isfinite(m) or m < 0:
        return math.inf
    return float(2.0 * n * m / (s * s))


def _lp_dir(a, tlim=4.0):
    """Mickey Mouse LP."""
    n = len(a)
    M = float(_autocorr(a).max())
    if M < 1e-15:
        return None, M
    
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M),
                      bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True, 'time_limit': tlim})
    except Exception:
        return None, M
    
    if res.status != 0 or res.x is None:
        return None, M
    
    g = np.asarray(res.x, dtype=np.float64)
    gs = g.sum()
    if gs < 1e-15:
        return None, M
    return g / gs, M


def _lp_opt(a0, t_stop, seed, max_it=80):

def _cd(a, t_stop, sweeps=300, seed=0):
    """Coordinate descent."""
    n = len(a)
    a = np.clip(a.astype(np.float64).copy(), 0, None)
    s = a.sum()
    if s < 1e-10:
        return a, math.inf
    a = a / s
    c = _autocorr(a)
    bs = 2.0 * n * float(c.max())
    rng = np.random.default_rng(seed)
    cm = 2 * n - 1
    noimp = 0
    
    for sw in range(sweeps):
        if time.monotonic() > t_stop - 3:
            break
        imp = False
        for i in rng.permutation(n):
            if time.monotonic() > t_stop - 2:
                break
            v = a[i]
            sr = s - v
            st = i
            en = min(i + n, cm)
            ai = a[np.arange(st, en) - i]
            inf = 2.0 * ai
            um = 0.0
            if st > 0:
                um = max(um, float(c[:st].max()))
            if en < cm:
                um = max(um, float(c[en:].max()))
            ti2 = 2 * i
            in_fa = st <= ti2 < en
            ca = c[st:en].copy()
            
            xh = max(10 * v, s / n * 8) if v > 1e-12 else s / n * 4
            xh = min(xh, s * 1.5)
            xh = max(xh, s / n)
            
            bx, bo = v, bs
            for x in np.linspace(0, xh, 100):
                w = x - v
                sn = sr + x
                if sn < 0.01:
                    continue
                nm = float((ca + w * inf).max())
                if in_fa:
                    nm = max(nm, ca[ti2 - st] + w * inf[ti2 - st] + w * w)
                ob = 2.0 * n * max(um, nm) / (sn * sn)
                if ob < bo - 1e-15:
                    bo, bx = ob, x
            
            if bx != v or bo < bs:
                dw = xh / 100
                lo, hi = max(0, bx - 2 * dw), min(xh, bx + 2 * dw)
                for x in np.linspace(lo, hi, 50):
                    w = x - v
                    sn = sr + x
                    if sn < 0.01:
                        continue
                    nm = float((ca + w * inf).max())
                    if in_fa:
                        nm = max(nm, ca[ti2 - st] + w * inf[ti2 - st] + w * w)
                    ob = 2.0 * n * max(um, nm) / (sn * sn)
                    if ob < bo - 1e-15:
                        bo, bx = ob, x
            
            if abs(bx - v) > 1e-13 and bo < bs - 1e-14:
                w = bx - v
                a[i] = bx
                s += w
                c[st:en] += w * inf
                if in_fa:
                    c[ti2] += w * w
                bs, imp = bo, True
        
        noimp = 0 if imp else noimp + 1
        if noimp >= 3:

def _block_cd(a, t_stop, seed=0, rounds=40):
    n = len(a)
    a = a.copy()
    bs = _score_fast(a)
    rng = np.random.default_rng(seed)
    for r in range(rounds):
        if time.monotonic() > t_stop - 3:
            break
        imp = False
        for _ in range(80):
            if time.monotonic() > t_stop - 2:
                break
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            oi, oj = a[i], a[j]
            best_xy, best_sc = (oi, oj), bs
            xg = np.linspace(0, max(oi, a.mean()) * 2 + a.mean() * 0.5, 10)
            yg = np.linspace(0, max(oj, a.mean()) * 2 + a.mean() * 0.5, 10)
            for x in xg:
                for y in yg:
                    a[i], a[j] = x, y
                    if a.sum() < 0.01:
                        a[i], a[j] = oi, oj
                        continue
                    sc = _score_fast(a)
                    if sc < best_sc - 1e-14:
                        best_sc, best_xy = sc, (x, y)
            a[i], a[j] = best_xy
            if best_sc < bs - 1e-14:
                bs = best_sc
                imp = True
        if not imp:
            break
    a = np.clip(a, 0, 1000)
    return a, _score_fast(a)
