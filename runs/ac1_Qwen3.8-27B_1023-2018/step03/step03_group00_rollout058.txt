def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(198, dtype=np.float64)
    if p.size == 0:
        p = np.ones(198, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(198, dtype=np.float64)
    return p

def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)

def _gradient_descent(a0, max_iters=1000, initial_lr=0.1, stop_time=None):
    """Gradient descent with backtracking."""
    n = len(a0)
    a = np.clip(a0.copy(), 0, None)
    S = a.sum()
    if S < 1e-10:
        return a, _score(a)
    
    best = a.copy()
    best_sc = _score(a)
    lr = initial_lr
    no_improve = 0
    
    for it in range(max_iters):
        if stop_time is not None and time.monotonic() > stop_time:
            break
        
        a_new = _gradient_step(a, lr)
        if a_new.sum() < 0.01:
            a = best.copy()
            break
        
        sc = _score(a_new)
        if sc < best_sc - 1e-12:
            best = a_new.copy()
            best_sc = sc
            a = a_new
            lr = min(lr * 1.15, 5.0)
            no_improve = 0
        else:
            lr *= 0.5
            if lr < 1e-8:
                break
            a = a_new  # still move, just smaller effective step next time
            no_improve += 1
            if no_improve > 50:
                break
    
    return best, best_sc

def _lp_direction(f, n, M, timeout=3.0):
    """LP to find direction that maximizes sum subject to f*g <= M."""
    # Build A: (f*g)[k] = sum_j f[k-j] * g[j]
    # A is (2n-1) x n, but we only need f*g (not full convolution)
    # Actually f*g (cross-convolution) has 2n-1 entries
    # (f*g)[k] = sum_j f[k-j] * g[j] (with f[m]=0 for m outside [0,n-1])
    
    # For the LP, we need: for all k in [0, 2n-2], sum_j A[k,j] * g[j] <= M
    # A[k,j] = f[k-j] if 0<=k-j<n, else 0
    
    # This is a Toeplitz-like matrix. For n up to ~500, dense is fine.
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        # A[k, j] = f[k-j] for k-j in [0, n-1], i.e., k in [j, j+n-1]
        lo = j
        hi = j + n - 1
        A[lo:hi+1, j] = f
    
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(
            c, A_ub=A, b_ub=b_ub,
            bounds=bounds, method='highs',
            options={"presolve": True, "time_limit": timeout},
        )
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0.0, 1000.0)
    if g.sum() < 1e-15:
        return None
    return g

def _lp_optimize(f0, n, stop_time, seed, max_iters=200):

    f = f[:n]
    f_sum = float(f.sum())
    if f_sum < 1e-10:
        f = np.ones(n)
    
    best_f = f.copy()
    best_score = _score(f)
    
    t = 0.3
    stall = 0
    max_stall = 100
    
    for iteration in range(max_iters):
        if time.monotonic() > stop_time:
            break
        
        b = _conv_full(f)
        M = float(b.max())
        if M < 1e-18:
            break
        
        # LP direction
        g = _lp_direction(f, n, M, timeout=2.0)
        if g is None:
            # Fallback: gradient step
            for lr in (0.5, 0.2, 0.1, 0.05):
                f_new = _gradient_step(f, lr)
                sc = _score(f_new)
                if sc < best_score - 1e-12:
                    f = f_new
                    best_f = f.copy()
                    best_score = sc
                    break
            stall += 5
            if stall > max_stall:
                break
            continue
        
        g_sum = float(g.sum())
        if g_sum < 1e-15:
            stall += 3
            continue
        g = g * (f.sum() / g_sum)
        
        dist = float(np.linalg.norm(g - f)) / max(1.0, float(np.linalg.norm(f)))
        if dist < 1e-9:
            stall += 2
            if stall > max_stall:
                break
            continue
        
        # Line search
        improved = False
        best_t = t
        for _ in range(12):
            f_new = (1.0 - t) * f + t * g
            f_new = np.clip(f_new, 0, None)
            if f_new.sum() < 0.01:
                t *= 0.5
                continue
            sc = _score(f_new)
            if sc < best_score - 1e-12:
                f = f_new.copy()
                best_f = f.copy()
                best_score = sc
                improved = True
                t = min(0.7, t * 1.2)
                stall = 0
                break
            else:
                t = max(0.02, t * 0.5)
        
        if not improved:
            # Try gradient step
            for lr in (0.3, 0.1, 0.05):
                f_new = _gradient_step(f, lr)
                sc = _score(f_new)
                if sc < best_score - 1e-12:
                    f = f_new
                    best_f = f.copy()
                    best_score = sc
                    improved = True
                    break
            if not improved:
                stall += 1
                if stall >= max_stall:
                    break
    
    return best_f, best_score

def _polish(a, stop_time, seed=4242):
    """Coordinate descent polish."""
    n = len(a)
    best = np.clip(np.asarray(a, dtype=np.float64).ravel(), 0, None).copy()
    best_sc = _score(best)
    rng = np.random.default_rng(seed)
    no_imp = 0
    
    while time.monotonic() < stop_time - 3.0:
        improved = False
        
        # Coordinate descent
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 3.0:
                break
            orig = best[i]
            if orig < 1e-14:
                # Try setting to small positive
                for val in (0.01, 0.1, 1.0):
                    best[i] = val
                    sc = _score(best)
                    if sc < best_sc - 1e-15:
                        best_sc = sc
                        improved = True
                        break
                else:
                    best[i] = 0.0
                continue
            
            best_val = orig
            best_val_sc = best_sc
            for mlt in (0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.7, 2.5, 4.0, 0.0):
                best[i] = orig * mlt
                if best.sum() < 0.01:
                    best[i] = orig
                    continue
                sc = _score(best)
                if sc < best_val_sc - 1e-15:
                    best_val = best[i]
                    best_val_sc = sc
            best[i] = best_val
            if best_val_sc < best_sc - 1e-15:
                best_sc = best_val_sc
                improved = True
        
        # Random perturbations
        if not improved:
            for _ in range(50):
                if time.monotonic() > stop_time - 3.0:
                    break
                k = int(rng.integers(n))
                old = best[k]
                best[k] = max(0.0, old + rng.normal(0, 0.05 * max(1.0, best.mean())))
                if best.sum() < 0.01:
                    best[k] = old
                    continue
                sc = _score(best)
                if sc < best_sc - 1e-15:
                    best_sc = sc
                    improved = True
                else:
                    best[k] = old
        
        if not improved:
            no_imp += 1
            if no_imp >= 8:
                break
        else:
            no_imp = 0
    
    return best, best_sc

        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(198, dtype=np.float64)
    if p.size == 0:
        p = np.ones(198, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(198, dtype=np.float64)
    return p

def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    # Small negative values from FFT roundoff
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)

def _grad_and_peak(a):
    """Compute gradient of score and peak location."""
    n = len(a)
    S = float(a.sum())
    b = _conv_full(a)
    k_star = int(np.argmax(b))
    max_b = float(b[k_star])
    
    # Mirror: mirror[i] = a[k_star - i] if 0 <= k_star - i < n, else 0
    mirror = np.zeros(n)
    i_lo = max(0, k_star - n + 1)
    i_hi = min(n - 1, k_star)
    if i_lo <= i_hi:
        # For i in [i_lo, i_hi], k_star - i is in [k_star - i_hi, k_star - i_lo] = [max(0,k_star-n+1), min(n-1,k_star)]
        # a[k_star - i] for i = i_lo gives a[k_star - i_lo], for i = i_hi gives a[k_star - i_hi]
        # So indices in a: from k_star - i_hi to k_star - i_lo
        idx_in_a = k_star - np.arange(i_lo, i_hi + 1)  # [k_star-i_lo, ..., k_star-i_hi] -- wait
        # Actually: i goes from i_lo to i_hi, k_star-i goes from k_star-i_lo down to k_star-i_hi
        # So a[k_star-i] for i=i_lo is a[k_star-i_lo], for i=i_hi is a[k_star-i_hi]
        # The array a[k_star-i_hi : k_star-i_lo+1] in reverse gives what we want
        # Or equivalently: a[k_star - [i_lo, ..., i_hi]]
        mirror[i_lo:i_hi+1] = a[k_star - np.arange(i_lo, i_hi + 1)]
    
    # Gradient of score = (4n/S^2) * mirror - (4n*max_b/S^3) * a
    g = (4.0 * n / (S * S)) * mirror - (4.0 * n * max_b / (S ** 3)) * a
    return g, k_star, max_b, b

def _gd_optimize(a0, stop_time, initial_lr=0.5, max_iters=50000):
    """Gradient descent on the AC1 score."""
    n = len(a0)
    a = np.clip(a0.astype(np.float64).copy(), 0, None)
    S = a.sum()
    if S < 1e-10:
        a = np.ones(n) / n
        return a, _score(a)
    
    # Normalize to sum=1 for cleaner optimization (score is scale-invariant)
    a = a / S
    best = a.copy()
    best_sc = _score(a)
    
    lr = initial_lr
    no_improve = 0
    patience = 500
    
    for it in range(max_iters):
        if time.monotonic() > stop_time:
            break
        
        g, k_star, max_b, b = _grad_and_peak(a)
        
        a_new = a - lr * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            lr *= 0.5
            if lr < 1e-10:
                break
            continue
        a_new = a_new / s_new  # re-normalize to sum=1
        
        sc = _score(a_new)
        if sc < best_sc - 1e-13:
            best = a_new.copy()
            best_sc = sc
            a = a_new
            lr = min(lr * 1.1, 10.0)
            no_improve = 0
        else:
            lr *= 0.7
            if lr < 1e-10:
                break
            no_improve += 1
            # Still try the step (might help in long run)
            a = a_new
            if no_improve >= patience:
                break
    
    return best, best_sc

def _lp_direction(f, n, M, timeout=3.0):
    """LP: maximize sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j+n, j] = f
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={"presolve": True, "time_limit": timeout})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.clip(np.asarray(res.x, dtype=np.float64), 0.0, 1000.0)
    if g.sum() < 1e-15:
        return None
    return g

def _lp_optimize(f0, n, stop_time, seed, max_iters=50):
    """LP-based optimization."""
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel()[:n].copy(), 0, None)
    while f.size < n:
        f = np.concatenate([f, np.zeros(n - f.size)])
    f = f[:n]
    fs = f.sum()
    if fs < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / fs  # normalize
    
    best_f = f.copy()
    best_score = _score(f)
    t = 0.3
    stall = 0
    max_stall = 50
    conv_thresh = 1e-9
    
    for iteration in range(max_iters):
        if time.monotonic() > stop_time:
            break
        
        b = _conv_full(f)
        M = float(b.max())
        if M < 1e-18:
            break
        
        g = _lp_direction(f, n, M * 1.01, timeout=2.0)
        if g is None:
            stall += 5
            if stall > max_stall:
                break
            continue
        
        gs = g.sum()
        if gs < 1e-15:
            stall += 3
            continue
        g = g / gs  # normalize to sum=1
        
        dist = float(np.linalg.norm(g - f))
        if dist < conv_thresh:
            stall += 2
            if stall > max_stall:
                break
            continue
        
        # Line search
        improved = False
        for _ in range(15):
            f_new = (1.0 - t) * f + t * g
            f_new = np.maximum(f_new, 0.0)
            s_new = f_new.sum()
            if s_new < 1e-15:
                t *= 0.5
                continue
            f_new = f_new / s_new
            sc = _score(f_new)
            if sc < best_score - 1e-13:
                f = f_new
                best_f = f_new.copy()
                best_score = sc
                improved = True
                t = min(0.7, t * 1.15)
                stall = 0
                break
            else:
                t = max(0.02, t * 0.5)
        
        if not improved:
            stall += 1
            if stall >= max_stall:
                break
    
    return best_f, best_score

def _polish(a, stop_time, seed=4242):
    """Coordinate descent polish (in normalized space)."""
    n = len(a)
    best = np.clip(np.asarray(a, dtype=np.float64).ravel(), 0, None).copy()
    s = best.sum()
    if s < 1e-10:
        return best, _score(best)
    best = best / s
    best_sc = _score(best)
    rng = np.random.default_rng(seed)
    no_imp = 0
    max_rounds = 30
    
    for rnd in range(max_rounds):
        if time.monotonic() > stop_time - 3.0:
            break
        improved = False
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() > stop_time - 3.0:
                break
            orig = best[i]
            best_val = orig
            best_val_sc = best_sc
            
            if orig < 1e-13:
                candidates = (0.01, 0.05, 0.1, 0.3, 1.0)
            else:
                candidates = tuple(orig * m for m in (0.3, 0.5, 0.7, 0.9, 1.1, 1.4, 2.0, 3.5, 0.0))
            
            for val in candidates:
                best[i] = val
                s = best.sum()
                if s < 0.01:
                    best[i] = orig
                    continue
                best[i] = best[i]  # keep as is for scoring (score handles normalization)
                sc = _score(best)
                if sc < best_val_sc - 1e-15:
                    best_val = val
                    best_val_sc = sc
            
            best[i] = best_val
            if best_val_sc < best_sc - 1e-15:
                best_sc = best_val_sc
                improved = True
        
        # Random perturbations
        if not improved:
            for _ in range(100):
                if time.monotonic() > stop_time - 3.0:
                    break
                k = int(rng.integers(n))
                old = best[k]
                best[k] = max(0.0, old + rng.normal(0, 0.02))
                if best.sum() < 0.01:
                    best[k] = old
                    continue
                sc = _score(best)
                if sc < best_sc - 1e-15:
                    best_sc = sc
                    improved = True
                else:
                    best[k] = old
        
        if not improved:
            no_imp += 1
            if no_imp >= 5:
                break
        else:
            no_imp = 0
    
    return best, best_sc
