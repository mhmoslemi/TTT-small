```python
import numpy as np
import time, math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog, minimize

_LUT = {}
def _flen(m):
    m = int(max(1, m))
    if m not in _LUT:
        _LUT[m] = int(next_fast_len(m))
    return _LUT[m]

def _conv(a):
    n = len(a)
    if n == 0:
        return np.zeros(0)
    m = 2*n - 1
    L = _flen(m)
    fa = rfft(a, L)
    c = np.real(irfft(fa * fa, L)[:m])
    return np.maximum(c, 0.0)

def _conv2(a, b):
    n1, n2 = len(a), len(b)
    if n1 == 0 or n2 == 0:
        return np.zeros(0)
    m = n1 + n2 - 1
    L = _flen(m)
    c = np.real(irfft(rfft(a, L) * rfft(b, L), L)[:m])
    return np.maximum(c, 0.0)

def _score_fast(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    n = len(a)
    c = _conv(a)
    mx = float(c.max())
    if not math.isfinite(mx):
        return math.inf
    return 2.0 * n * mx / (s * s)

def _norm(a):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0.0, 1000.0)
    a = np.where(np.isfinite(a), a, 0.0)
    s = float(a.sum())
    if s < 1e-12:
        a[:] = 1.0 / len(a)
        return a
    return a / s

def _lp_dir(f, M):
    """LP: max sum(g) s.t. (f*g)_k <= M, g >= 0."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    res = linprog(
        -np.ones(n, dtype=np.float64),
        A_ub=A,
        b_ub=np.full(2 * n - 1, M, dtype=np.float64),
        method='highs',
    )
    if res is None or getattr(res, 'status', 1) != 0:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0.0, None)
    gs = float(g.sum())
    if gs < 1e-12:
        return None
    return g / gs

def _best_t_quad(cf, cfs, cg):
    """Minimize max_k [(1-t)^2 cf + 2t(1-t) cfs + t^2 cg] over t in [0,1]."""
    # = max_k [cf + 2t(cfs-cf) + t^2(cg-2cfs+cf)]
    A_ = cf.astype(np.float64)
    B_ = 2.0 * (cfs - cf)
    C_ = cg - 2.0 * cfs + cf

    def fmax(t):
        return float(np.max(A_ + B_ * t + C_ * t * t))

    # Coarse grid
    ts = np.linspace(0.0, 1.0, 25)
    vals = np.empty(25)
    for i in range(25):
        vals[i] = fmax(ts[i])
    i0 = int(np.argmin(vals))
    lo = ts[max(0, i0 - 2)]
    hi = ts[min(24, i0 + 2)]

    # Golden section in [lo, hi]
    if hi - lo < 1e-10:
        return 0.5 * (lo + hi)
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    c_p, d_p = hi - gr * (hi - lo), lo + gr * (hi - lo)
    fc, fd = fmax(c_p), fmax(d_p)
    for _ in range(50):
        if fc < fd:
            hi, d_p, fd = d_p, c_p, fc
            c_p = hi - gr * (hi - lo)
            fc = fmax(c_p)
        else:
            lo, c_p, fc = c_p, d_p, fd
            d_p = lo + gr * (hi - lo)
            fd = fmax(d_p)
    return 0.5 * (lo + hi)

        t0 = _best_t_quad(c, cfs, cg)

def _lbfgs_smooth(a0, t_stop, taus=(1.0, 0.1, 0.01)):
    """L-BFGS-B with logsumexp smooth max."""
    a = _norm(a0)
    n = len(a)
    best = a.copy()
    best_sc = _score_fast(best)

    for tau in taus:
        if time.monotonic() > t_stop:
            break
        max_c = float(_conv(best).max())
        tau_eff = tau * max_c if max_c > 0 else 1e-6

        def obj(y):
            y = np.asarray(y, dtype=np.float64)
            a_y = np.exp(np.clip(y - y.max(), -30, 30))
            a_y = a_y / a_y.sum()
            c = _conv(a_y)
            mc = c.max()
            if not math.isfinite(mc):
                return 1e10, np.zeros_like(y)
            shifted = (c - mc) / tau_eff
            w = np.exp(np.clip(shifted, -50, 0))
            sw = w.sum()
            lse = mc + tau_eff * math.log(sw)
            # grad lse w.r.t. a_y:
            # d lse / d a_y_j = sum_k w_k/sw * (1/tau_eff) * 2*a_y[k-j]*I(0<=k-j<n)
            gc = np.zeros(n)
            for k in range(2 * n - 1):
                wk = w[k] / sw / tau_eff
                if wk < 1e-15:
                    continue
                lo = max(0, k - n + 1)
                hi = min(n, k + 1)
                idx = np.arange(lo, hi)
                offsets = k - idx
                gc[idx] += wk * 2.0 * a_y[offsets]
            # d score / d a_y_j = 2n * d_lse/da_j / s^2 - ... but s=1 since normalized
            # Actually the score is 2n * lse (since s=1)
            # Wait, score = 2n * max_b / s^2, and we're using lse as proxy for max_b
            # So the smoothed score is 2n * lse / s^2
            # Since s = 1 (normalized), this is just 2n * lse
            # But we need the gradient w.r.t. y, not a_y
            # da_y_j / dy_m = a_y_j * (delta(j,m) - a_y_m)
            # d score / dy_m = sum_j (d score / da_y_j) * (da_y_j / dy_m)
            # = sum_j 2n * gc_j * a_y_j * (delta(j,m) - a_y_m)
            # = 2n * (gc_m * a_y_m - a_y_m * sum_j gc_j * a_y_j)
            
            dot = float(np.dot(gc, a_y))
            grad = 2.0 * n * a_y * (gc - dot)
            # Wait, this is the gradient of 2n * lse (not the full score with /s^2)
            # Since we've normalized s=1 in the parameterization, this should be correct
            # But the actual score also has /s^2, and s=1, so it's just 2n*lse
            # Actually no: the original score is 2n*max_b/s^2. If we normalize a to have s=1,
            # then score = 2n*max_b. The smoothed version is 2n*lse.
            # But we need to be careful: the score is scale-invariant, so the value
            # 2n*lse with s=1 IS the score (approximately).
            
            val = 2.0 * n * lse
            grad = np.where(np.isfinite(grad), grad, 0.0)
            return float(val), grad

        y0 = np.log(np.maximum(best, 1e-10))
        y0 = y0 - y0.mean()  # center to avoid overflow
        y0 = np.clip(y0, -20, 20)

        try:
            res = minimize(
                obj, y0, jac=True, method='L-BFGS-B',
                options={'maxiter': 100, 'maxfun': 300, 'ftol': 1e-10, 'gtol': 1e-8, 'maxls': 30}
            )
        except Exception:
            continue

        if hasattr(res, 'x') and res.x is not None:
            y = np.asarray(res.x, dtype=np.float64)
            y = y - y.max()
            a_new = np.exp(np.clip(y, -30, 30))
            a_new = a_new / a_new.sum()
            sc = _score_fast(a_new)
            if math.isfinite(sc) and sc < best_sc:
                best, best_sc = a_new, sc

    return best, best_sc
