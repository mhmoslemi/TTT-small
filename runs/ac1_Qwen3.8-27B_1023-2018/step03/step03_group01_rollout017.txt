        return math.inf
    b = _conv(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0*n*m/(s*s)

def _lp_direction(f, n, M, time_limit=10.0):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0"""
    # Build constraint matrix A (2n-1) x n
    # A[k,j] = f[k-j] if 0 <= k-j < n else 0
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = f
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0,None)]*n,
                     method='highs', options={'time_limit': time_limit})
        if res.status == 0 and res.x is not None:
            g = np.asarray(res.x, dtype=np.float64)
            gs = g.sum()
            if gs > 1e-14:
                return g/gs
    except Exception:
        pass
    return None

def _build_sparse_A(f, n):
    # Banded matrix: row k has nonzeros at columns max(0, k-n+1) to min(n-1, k)
    rows = []
    cols = []
    vals = []
    for k in range(2*n-1):
        lo = max(0, k-n+1)
        hi = min(n-1, k)
        for j in range(lo, hi+1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k-j])
    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))

Let me think about a different approach to break out of local optima:

**Peak-directed perturbation**: If the convolution has a peak at lag k, we can deliberately modify the sequence to reduce that specific peak. For example, if the peak is at lag 0 (which means sum a_i^2 is large), we should make the sequence more uniform. If the peak is at a non-zero lag, we should reduce the autocorrelation at that lag.

**Spectral approach**: Compute the FFT of the sequence. The peak of the convolution at lag k relates to the integral of |A(ω)|^2 * exp(2πiωk/F). To flatten the convolution, we want |A(ω)|^2 to be as flat as possible (white spectrum). This is a spectral flattening problem.

Algorithm:
1. Compute A(ω) = FFT(a)
2. Compute |A(ω)|^2 (power spectrum)
3. Modify |A(ω)| to be flatter: |A(ω)|_new = |A(ω)|^α for α < 1 (compression)
4. Inverse FFT to get new a, clip to nonnegative
5. Iterate

This is related to the "iterative amplitude modification" (IAM) or "Gerchberg-Saxton" type algorithms, but applied to the power spectrum.

If we have a sequence of length n and we remove one element (shifting subsequent elements), the convolution changes completely. It's not a simple rescaling.

However, if we have elements that are essentially zero (like < 1e-6), removing them is approximately equivalent to just reducing n (the convolution barely changes). Let me check: if a[j] = 0 for some j, then the contribution of a[j] to the convolution is zero everywhere. So removing a zero element and shifting is EXACTLY equivalent to reducing n by 1, IF the element is exactly at the boundary. If it's in the middle, shifting changes the lags.

Actually, for the convolution, if a[j] = 0, then in the convolution sum, the terms involving a[j] are zero. So the convolution of the remaining n-1 elements (as a contiguous block) is the same as the original convolution with the zero removed and the sequence compacted. The peak might be slightly different because the lags shift, but the values are the same.

So the score changes from 2n*max_b/sum^2 to 2(n-1)*max_b'/sum^2 where max_b' is the max of the new (shorter) convolution. If the zero was at an end, max_b' = max_b (the same values, just shifted). If in the middle, max_b' could be different.

This suggests a strategy: during optimization, try to drive some elements to zero, then remove them to reduce n.
