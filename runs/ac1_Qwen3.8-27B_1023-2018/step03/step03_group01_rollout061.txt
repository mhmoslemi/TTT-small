    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_fft(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)

def _build_A(f, n):
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo = j
        hi = min(2 * n - 1, j + n)
        A[lo:hi, j] = f[:hi - lo]
    return A

def _grad_one_lag(f, k):
    n = len(f)
    g = np.zeros(n)
    if k == 0:
        g[:] = 2.0 * f
    else:
        # g[j] = f[j+k] for j < n-k (if j+k < n)
        # g[j] += f[j-k] for j >= k (if j-k >= 0)
        if n - k > 0:
            g[:n-k] += f[k:]
        if n - k > 0:
            g[k:] += f[:n-k]
    return g

def _lp_solve(f, n, M, time_limit=5.0):
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(
            c, A_ub=A, b_ub=b_ub,
            bounds=[(0.0, None)] * n,
            method='highs',
            options={'presolve': True, 'time_limit': time_limit},
        )
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    gs = float(g.sum())
    if gs < 1e-14:
        return None
    return g / gs

def _line_search(f, g_dir, n, best_sc, t_init=0.30, deadline=None):
    t_lo, t_hi = 0.0, t_init
    f_lo = f
    for it in range(20):
        if deadline and time.monotonic() > deadline:
            break
        t_mid = 0.5 * (t_lo + t_hi)
        fn = (1.0 - t_mid) * f + t_mid * g_dir
        np.maximum(fn, 0.0, out=fn)
        fs = float(fn.sum())
        if fs < 1e-10:
            t_hi = t_mid
            continue
        sc = _score(fn)
        if sc < best_sc - 1e-14:
            t_lo = t_mid
            f_lo = fn.copy()
        else:
            t_hi = t_mid
        if t_hi - t_lo < 1e-5:
            break
    if t_lo < 1e-6:
        return None, None
    return t_lo, f_lo

def _lp_loop(f0, n, deadline, seed, max_iters=5000):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    fs = float(f.sum())
    if fs < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / fs  # normalize to sum 1
    
    best_f = f.copy()
    best_sc = _score(f)
    t_cur = 0.30
    stall = 0
    max_stall = 80
    it = 0
    m_reduction_count = 0
    
    while time.monotonic() < deadline - 3.0:
        it += 1
        b = _conv_fft(f)
        M = float(b.max())
        if M < 1e-15:
            break
        
        # Adaptive M: slightly reduce to force progress when stalled
        M_eff = M * (1.0 - 0.005 * min(m_reduction_count, 20))
        if M_eff < 0.01 * M:
            M_eff = 0.01 * M
        
        g_dir = _lp_solve(f, n, M_eff, time_limit=min(5.0, max(1.0, (deadline - time.monotonic()) * 0.1)))
        if g_dir is None:
            stall += 5
            if stall > max_stall:
                break
            f = np.clip(f + rng.normal(0, 0.05, n), 0, None)
            fs2 = float(f.sum())
            f = f / fs2 if fs2 > 1e-10 else best_f.copy()
            t_cur = 0.25
            continue
        
        dist = float(np.linalg.norm(g_dir - f))
        if dist < 1e-7:
            stall += 3
            m_reduction_count += 1
            if stall > max_stall:
                break
            f = np.clip(f + rng.normal(0, 0.08, n), 0, None)
            fs2 = float(f.sum())
            f = f / fs2 if fs2 > 1e-10 else best_f.copy()
            t_cur = 0.30
            continue

        if f_new is not None:
            sc_new = _score(f_new)
            if sc_new < best_sc - 1e-13:
                f = f_new
                best_f = f.copy()
                best_sc = sc_new
                t_cur = min(0.55, res_t * 1.15)
                stall = 0
                m_reduction_count = 0
            else:
                t_cur = max(0.02, res_t * 0.6)
                stall += 1
                m_reduction_count += 1
        else:
            t_cur = max(0.02, t_cur * 0.5)
            stall += 1
            m_reduction_count += 1
        
        if stall >= max_stall:
            # Diversification
            mode = rng.integers(3)
            if mode == 0:
                f = np.clip(f + rng.normal(0, 0.1, n), 0, None)
            elif mode == 1:
                i = np.arange(n)

            fs2 = float(f.sum())
            f = f / fs2 if fs2 > 1e-10 else best_f.copy()
            t_cur = 0.30
            stall = 0
            m_reduction_count = 0
        
        if it % 100 == 0 and best_sc < 1.503:
            break
    
    return best_f, best_sc

def _subgrad_polish(f0, n, deadline, seed):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    fs = float(f.sum())
    f = f / fs
    best_f = f.copy()
    best_sc = _score(f)
    
    lr = 0.005
    momentum = np.zeros(n)
    beta = 0.8
    no_imp = 0
    topk = min(8, 2 * n - 1)
    
    for it in range(500000):
        if time.monotonic() > deadline - 3.0:
            break
        
        b = _conv_fft(f)
        M = float(b.max())
        
        # Find top-K lags
        idx = np.argpartition(b, -topk)[-topk:]
        weights = b[idx].copy()
        weights = weights - weights.min() + 1e-12
        weights /= weights.sum()
        
        # Compute weighted gradient
        grad_M = np.zeros(n)
        for w, k in zip(weights, idx):
            k = int(k)
            # Gradient of conv(f)[k] w.r.t. f
            if k == 0:
                grad_M += w * 2.0 * f
            else:
                kk = min(k, 2*n-2-k)  # use symmetry
                if kk == 0:
                    grad_M += w * 2.0 * f
                else:
                    if n - kk > 0:
                        grad_M[:n-kk] += w * f[kk:]
                        grad_M[kk:] += w * f[:n-kk]
        
        s = float(f.sum())
        # Gradient of score = 2n*M/s² - 4n*M/s³ * 1
        # = (2n/s²) * grad_M - (4n*M/s³) * ones
        grad_F = (2.0 * n / (s * s)) * grad_M - (4.0 * n * M / (s * s * s)) * np.ones(n)
        
        momentum = beta * momentum - lr * grad_F
        f_new = f + momentum
        f_new = np.maximum(f_new, 0.0)
        fs_new = float(f_new.sum())
        if fs_new < 1e-10:
            f_new = f.copy()
        else:
            f_new = f_new / fs_new
        
        sc = _score(f_new)
        if sc < best_sc - 1e-14:
            f = f_new
            best_f = f.copy()
            best_sc = sc
            no_imp = 0
            lr = min(0.02, lr * 1.005)
        else:
            no_imp += 1
            lr = max(0.0005, lr * 0.98)
            if no_imp > 200:
                f = f_new.copy()
                no_imp = 0
                lr = 0.005
                beta = 0.6
                momentum = np.zeros(n)
                break  # restart subgrad
    
    return best_f, best_sc

def _coord_polish(f0, n, deadline, seed):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    fs = float(f.sum())
    f = f / fs
    best_sc = _score(f)
    no_imp = 0
    multipliers = (0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0, 0.0)
    
    while time.monotonic() < deadline - 3.0:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > deadline - 3.0:
                return f, best_sc
            orig = f[i]
            best_val = orig
            best_local_sc = best_sc
            for mlt in multipliers:
                f[i] = orig * mlt if orig > 1e-12 else (0.01 if mlt > 0 else 0.0)
                sc = _score(f)
                if sc < best_local_sc - 1e-13:
                    best_local_sc = sc
                    best_val = f[i]
            f[i] = best_val
            if best_local_sc < best_sc - 1e-13:
                best_sc = best_local_sc
                improved = True
        
        # Fine-tune around current values
        if not improved:
            for i in order:
                if time.monotonic() > deadline - 3.0:
                    return f, best_sc
                orig = f[i]
                if orig < 1e-8:
                    continue
                for frac in (0.9, 1.1, 0.8, 1.2):
                    f[i] = orig * frac
                    sc = _score(f)
                    if sc < best_sc - 1e-13:
                        best_sc = sc
                        improved = True
            f = f  # (already updated in place)
        
        # Pair moves
        for _ in range(30):
            if time.monotonic() > deadline - 3.0:
                return f, best_sc
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            oi, oj = f[i], f[j]
            for mi in (0.5, 1.5, 2.0):
                for mj in (0.5, 1.5, 0.3):
                    f[i] = oi * mi if oi > 1e-12 else 0.01
                    f[j] = oj * mj if oj > 1e-12 else 0.01
                    sc = _score(f)
                    if sc < best_sc - 1e-13:
                        best_sc = sc
                        improved = True
            f[i], f[j] = oi, oj
        
        if not improved:
            no_imp += 1
            if no_imp >= 8:
                break
        else:
            no_imp = 0
    
    return f, best_sc

def _try_update(best, best_sc, cand, deadline):
    if cand is None:
        return best, best_sc
    cand = np.clip(np.asarray(cand, dtype=np.float64).ravel(), 0, 1000)
    if cand.size == 0 or float(cand.sum()) < 0.01:
        return best, best_sc
    try:
        sc = _score(cand)
    except Exception:
        return best, best_sc
    if sc < best_sc - 1e-14:
        return cand.copy(), sc
    return best, best_sc

For the subgradient of h(f) = max_k conv(f)[k]:
- If the max is at lag k*, the subgradient is the gradient of conv(f)[k*]
- If the max is at multiple lags k_1, ..., k_m, any convex combination of their gradients is a valid subgradient

The gradient of conv(f)[k] w.r.t. f[j]:
- For k = 0: 2*f[j]

        # i.e., j in [k-n+1, k] intersected with [0, n-1]

        hi = min(n, k + 1)  # j <= k

    return g

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(320, dtype=np.float64)
    if p.size == 0:
        p = np.ones(320, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(320, dtype=np.float64)
    return p


def _conv_fft(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_fft(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n):
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo = j
        hi = min(2 * n - 1, j + n)
        A[lo:hi, j] = f[:hi - lo]
    return A


def _grad_weighted(f, lags, weights):
    n = len(f)
    g = np.zeros(n)
    for k, w in zip(lags, weights):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            idx = np.arange(lo, hi)
            g[idx] += 2.0 * w * f[k - idx]
    return g


def _lp_solve(f, n, M, tlimit=3.0):
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(
            c, A_ub=A, b_ub=b_ub,
            bounds=[(0.0, None)] * n,
            method='highs',
            options={'presolve': True, 'time_limit': tlimit},
        )
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    gs = float(g.sum())
    if gs < 1e-14:
        return None
    return g / gs


def _quick_line_search(f, g_dir, n, best_sc, deadline):
    """Try a few step sizes and return the best."""
    best_t = 0.0
    best_fn = None
    best_s = best_sc
    for t in (0.15, 0.3, 0.5, 0.7, 0.9):
        if deadline and time.monotonic() > deadline:
            break
        fn = (1.0 - t) * f + t * g_dir
        np.maximum(fn, 0.0, out=fn)
        fs = float(fn.sum())
        if fs < 1e-10:
            continue
        sc = _score(fn)
        if sc < best_s - 1e-14:
            best_s = sc
            best_t = t
            best_fn = fn.copy()
    if best_fn is None:
        return None, None
    # Refine around best_t
    for dt in (0.05, -0.05, 0.1, -0.1):
        t2 = max(0.01, min(0.99, best_t + dt))
        fn = (1.0 - t2) * f + t2 * g_dir
        np.maximum(fn, 0.0, out=fn)
        fs = float(fn.sum())
        if fs < 1e-10:
            continue
        sc = _score(fn)
        if sc < best_s - 1e-14:
            best_s = sc
            best_t = t2
            best_fn = fn.copy()
    return best_t, best_fn


def _lp_loop(f0, n, deadline, seed):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    fs = float(f.sum())
    if fs < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / fs
    
    best_f = f.copy()
    best_sc = _score(f)
    t_cur = 0.3
    stall = 0
    max_stall = 80
    it = 0
    
    while time.monotonic() < deadline - 3.0 and it < 5000:
        it += 1
        b = _conv_fft(f)
        M = float(b.max())
        if M < 1e-15:
            break
        
        # Check deadline for LP
        rem = max(0.5, deadline - 3.0 - time.monotonic())
        lt = min(3.0, max(0.5, rem * 0.15))
        
        g_dir = _lp_solve(f, n, M, tlimit=lt)
        if g_dir is None:
            stall += 5
            if stall > max_stall:
                break
            f = np.clip(f + rng.normal(0, 0.05, n), 0, None)
            fs2 = float(f.sum())
            f = f / fs2 if fs2 > 1e-10 else best_f.copy()
            t_cur = 0.25
            continue
        
        dist = float(np.linalg.norm(g_dir - f))
        if dist < 1e-7:
            stall += 3
            if stall > max_stall:
                break
            f = np.clip(f + rng.normal(0, 0.08, n), 0, None)
            fs2 = float(f.sum())
            f = f / fs2 if fs2 > 1e-10 else best_f.copy()
            t_cur = 0.3
            continue
        
        res_t, f_new = _quick_line_search(f, g_dir, n, best_sc, deadline)
        if f_new is not None:
            sc_new = _score(f_new)
            if sc_new < best_sc - 1e-13:
                f = f_new
                best_f = f.copy()
                best_sc = sc_new
                t_cur = min(0.6, res_t * 1.15)
                stall = 0
            else:
                t_cur = max(0.02, res_t * 0.6)
                stall += 1
        else:
            t_cur = max(0.02, t_cur * 0.5)
            stall += 1
        
        if stall >= max_stall:
            mode = int(rng.integers(3))
            if mode == 0:
                f = np.clip(f + rng.normal(0, 0.1, n), 0, None)
            elif mode == 1:
                i = np.arange(n, dtype=np.float64)
                fr = float(rng.uniform(0.5, 5.0))
                ph = float(rng.uniform(0, 2 * math.pi))
                f = np.clip(f * (1 + 0.3 * np.cos(2 * math.pi * fr * i / n + ph)), 0, None)
            else:
                f = np.abs(f + 0.1 * rng.normal(0, 1, n))
            fs2 = float(f.sum())
            f = f / fs2 if fs2 > 1e-10 else best_f.copy()
            t_cur = 0.3
            stall = 0
        
        if best_sc < 1.503 and it > 20:
            break
    
    return best_f, best_sc


def _subgrad_polish(f0, n, deadline, seed):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    fs = float(f.sum())
    f = f / fs
    best_f = f.copy()
    best_sc = _score(f)
    
    lr = 0.008
    momentum = np.zeros(n)
    beta = 0.82
    no_imp = 0
    topk = 10
    
    for it in range(500000):
        if time.monotonic() > deadline - 3.0:
            break
        
        b = _conv_fft(f)
        M = float(b.max())
        
        # Top-K lags
        idx = np.argpartition(b, -topk)[-topk:]
        weights = b[idx].copy()
        weights = np.maximum(weights - weights.min(), 0) + 1e-12
        weights /= weights.sum()
        
        # Gradient
        grad_M = _grad_weighted(f, idx, weights)
        
        s = float(f.sum())
        grad_F = (2.0 * n / (s * s)) * grad_M - (4.0 * n * M / (s ** 3)) * np.ones(n)
        
        momentum = beta * momentum - lr * grad_F
        f_new = f + momentum
        f_new = np.maximum(f_new, 0.0)
        fs_new = float(f_new.sum())
        if fs_new < 1e-10:
            f_new = f.copy()
        else:
            f_new = f_new / fs_new
        
        sc = _score(f_new)
        if sc < best_sc - 1e-14:
            f = f_new
            best_f = f.copy()
            best_sc = sc
            no_imp = 0
            lr = min(0.03, lr * 1.003)
        else:
            no_imp += 1
            lr = max(0.001, lr * 0.985)
            if no_imp > 300:
                f = f_new.copy()
                no_imp = 0
                lr = 0.008
                beta = 0.7
                momentum = np.zeros(n)
    
    return best_f, best_sc


def _coord_polish(f0, n, deadline, seed):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    fs = float(f.sum())
    f = f / fs
    best_sc = _score(f)
    no_imp = 0
    mults = (0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.7, 0.0)
    
    while time.monotonic() < deadline - 3.0:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > deadline - 3.0:
                return f, best_sc
            orig = f[i]
            best_val = orig
            best_local = best_sc
            for m in mults:
                f[i] = orig * m if orig > 1e-12 else (0.01 if m > 0 else 0.0)
                sc = _score(f)
                if sc < best_local - 1e-13:
                    best_local = sc
                    best_val = f[i]
            f[i] = best_val
            if best_local < best_sc - 1e-13:
                best_sc = best_local
                improved = True
        
        if not improved:
            no_imp += 1
            if no_imp >= 10:
                break
        else:
            no_imp = 0
    
    return f, best_sc


def _try_update(best, best_sc, cand):
    if cand is None:
        return best, best_sc
    cand = np.clip(np.asarray(cand, dtype=np.float64).ravel(), 0, 1000)
    if cand.size == 0 or float(cand.sum()) < 0.01:
        return best, best_sc
    try:
        sc = _score(cand)
    except Exception:
        return best, best_sc
    if math.isfinite(sc) and sc < best_sc - 1e-14:
        return cand.copy(), sc
    return best, best_sc

    for idx, init in enumerate(remaining_inits):

    other_end = min(deadline, t0 + budget * 0.87)

    if time.monotonic() < deadline - 10.0:

            best_seq, best_score = _try_update(best_seq, best_score, res)

            pass

        pass
