    n = len(a)
    best = np.clip(np.asarray(a, dtype=np.float64).ravel(), 0, None).copy()
    fs = best.sum()
    if fs < 1e-10:
        return best, math.inf
    best = best / fs
    best_sc = _score(best)
    rng = np.random.default_rng(seed)
    no_imp = 0
    max_no_imp = 10
    
    multipliers = [0.1, 0.3, 0.5, 0.7, 1.0, 1.3, 1.7, 2.5, 4.0, 0.0]
    
    while time.monotonic() < stop_time - 1.0:
        improved = False
        
        # 1-coordinate moves
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 1.0:
                break
            orig = best[i]
            best_val = orig
            best_val_sc = best_sc
            for mlt in multipliers:
                new_val = orig * mlt if orig > 1e-12 else (0.005 if mlt > 0 else 0.0)
                best[i] = new_val
                if best.sum() < 0.01:
                    best[i] = orig
                    continue
                sc = _score(best)
                if sc < best_val_sc - 1e-14:
                    best_val = new_val
                    best_val_sc = sc
            best[i] = best_val
            if best_val_sc < best_sc - 1e-14:
                best_sc = best_val_sc
                improved = True
        
        # 2-coordinate moves
        n_2coord = max(20, n // 4)
        for _ in range(n_2coord):
            if time.monotonic() > stop_time - 1.0:
                break
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            for mi in (0.3, 0.7, 1.4, 2.5, 0.0):
                for mj in (0.3, 0.7, 1.4, 2.5, 0.0):
                    best[i] = oi * mi if oi > 1e-12 else (0.005 if mi > 0 else 0.0)
                    best[j] = oj * mj if oj > 1e-12 else (0.005 if mj > 0 else 0.0)
                    if best.sum() < 0.01:
                        best[i], best[j] = oi, oj
                        continue
                    sc = _score(best)
                    if sc < best_sc - 1e-14:
                        best_sc = sc
                        improved = True
            best[i], best[j] = oi, oj
        
        if not improved:
            no_imp += 1
            if no_imp >= max_no_imp:
                break
        else:
            no_imp = 0
    
    return best, best_sc

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _get_parent():

def _conv_fft(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    S = float(a.sum())
    if S < 0.01:
        return math.inf
    M = float(_conv_fft(a).max())
    if not math.isfinite(M):
        return math.inf
    return 2.0 * n * M / (S * S)


def _build_A(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(n):
        A[k:k + n, k] = f
    return A


def _lp_direction(f, M):
    n = len(f)
    if M < 1e-15:
        return None
    A = _build_A(f)
    c_vec = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_vec, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={"presolve": True,
                               "dual_feasibility_tolerance": 1e-9,
                               "primal_feasibility_tolerance": 1e-9})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    s = float(g.sum())
    if s < 1e-10:
        return None
    return g / s

def _smooth_optimize(a0, stop_time, seed, n_outer=40, n_inner=10):
    """Active-set gradient descent with annealing."""
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64).ravel(), 0, None).copy()
    S = a.sum()
    if S < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / S

    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(seed)

    conv0 = _conv_fft(a)
    M0 = conv0.max()
    K = min(25, 2 * n - 1)
    lr = 0.08
    stall_count = 0

    for outer in range(n_outer):
        if time.monotonic() > stop_time - 1.0:
            break
        progress = outer / max(1, n_outer - 1)
        # K_active decreases over time
        K_cur = max(5, int(K * (1.0 - 0.5 * progress)))

        for inner in range(n_inner):
            if time.monotonic() > stop_time - 1.0:
                break

            conv = _conv_fft(a)
            if K_cur < len(conv):
                top_k = np.argpartition(conv, -K_cur)[-K_cur:]
            else:
                top_k = np.arange(len(conv))

            grad = np.zeros(n)
            for k in top_k:
                lo = max(0, k - n + 1)
                hi = min(k, n - 1)
                if lo <= hi:
                    m_idx = np.arange(lo, hi + 1)
                    grad[m_idx] += 2.0 * a[k - m_idx]
            grad /= len(top_k)

            # Normalized gradient step
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-12:
                stall_count += 1
                break

            a_new = a - lr * grad / grad_norm * grad_norm**0.5  # scale
            a_new = np.clip(a_new, 0, None)
            S_new = a_new.sum()
            if S_new < 1e-10:
                a_new = a.copy()
            else:
                a_new = a_new / S_new

            sc = _score(a_new)
            if sc < best_sc - 1e-14:
                best_a = a_new.copy()
                best_sc = sc
                a = a_new
                stall_count = 0
                lr = min(0.25, lr * 1.02)
            else:
                lr = max(0.005, lr * 0.7)
                stall_count += 1

            # Occasional perturbation
            if rng.random() < 0.15 and stall_count > 3:
                a_pert = a + rng.normal(0, 0.02, n)
                a_pert = np.clip(a_pert, 0, None)
                S_p = a_pert.sum()
                if S_p > 1e-10:
                    a_pert = a_pert / S_p
                    sc_p = _score(a_pert)
                    if sc_p < best_sc - 1e-14:
                        best_a = a_pert.copy()
                        best_sc = sc_p
                        a = a_pert
                        stall_count = 0

    return best_a, best_sc
