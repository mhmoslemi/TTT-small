            else:
                cands = np.array([0.0, 0.01, 0.05, 0.1, 0.3])

            bi = orig
            bi_sc = best_sc
            for val in cands:
                best[i] = max(0.0, val)
                if best.sum() < 0.01:
                    best[i] = orig
                    continue
                sc = _score(best)
                if sc < bi_sc:
                    bi = max(0.0, val)
                    bi_sc = sc
            best[i] = bi
            if bi_sc < best_sc - 1e-14:
                best_sc = bi_sc
                improved = True

        # Pair moves
        n_pairs = min(300, n * 3)
        for _ in range(n_pairs):
            if time.monotonic() > stop_time - 2.0:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            total = oi + oj
            if total < 1e-14:
                continue
            for frac in (0.1, 0.3, 0.5, 0.7, 0.9):
                best[i] = frac * total
                best[j] = (1.0 - frac) * total
                sc = _score(best)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    improved = True
            # Also try adding/removing
            for delta in (0.01, 0.05, 0.1):
                best[i] = max(0, oi + delta)
                best[j] = max(0, oj - delta * 0.5)
                if best.sum() < 0.01:
                    best[i], best[j] = oi, oj
                    continue
                sc = _score(best)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    improved = True
            best[i], best[j] = oi, oj

        if not improved:
            no_imp += 1
        else:
            no_imp = 0

    return best, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 12.0

    parent = _get_parent()
    n_p = len(parent)

    # Fallback
    fallback = parent.copy()
    try:
        fallback_sc = _score(fallback)
    except Exception:
        fallback_sc = math.inf
    if not math.isfinite(fallback_sc):
        fallback = np.ones(n_p) / n_p
        fallback_sc = _score(fallback)

    best_seq = fallback.copy()
    best_score = fallback_sc
    n = n_p  # 384

    i_arr = np.arange(n, dtype=np.float64)
    rng_main = np.random.default_rng(seed)

    # Build diverse initializations
    inits = [parent.copy()]

    # Cosine window
    inits.append(0.5 + 0.5 * np.cos(np.pi * i_arr / max(1, n - 1)))

    # Raised cosine squared
    rc = 0.5 + 0.5 * np.cos(np.pi * i_arr / max(1, n - 1))
    inits.append(rc * rc)

    # Multi-harmonic
    sup = np.ones(n) * 0.15
    for fr in (1, 3, 5, 9):
        ph = rng_main.uniform(0, 2 * math.pi)
        sup += 0.2 * np.cos(2 * math.pi * fr * i_arr / n + ph)
    inits.append(np.abs(sup) + 0.005)

    # Smoothed random
    kern = np.exp(-0.5 * (np.arange(-8, 9) / 3.0) ** 2)
    kern /= kern.sum()
    sm = np.convolve(rng_main.normal(0, 1, n), kern, 'same')
    inits.append(np.abs(sm) + 0.01)

    # Parent perturbed
    inits.append(np.abs(parent + rng_main.normal(0, 0.12 * max(1.0, np.std(parent)), n)) + 0.001)

    # Symmetric emphasis (center peak)
    center = n / 2.0
    sym = 1.0 + 0.3 * np.cos(np.pi * (i_arr - center + 0.5) / (n / 2.0))
    inits.append(np.abs(sym) + 0.005)

    # Phase 1: LP optimization (55% of budget)
    phase1_end = t0 + budget * 0.55
    # Run the best init (parent) longest, others shorter
    time_parent = max(30.0, (phase1_end - t0) * 0.5)
    time_other = max(10.0, (phase1_end - t0 - time_parent) / max(1, len(inits) - 1))

    # Parent first (most likely to be best)
    if time.monotonic() < phase1_end - 8.0:
        stop1 = min(phase1_end - 5.0, t0 + time_parent)
        try:
            res, sc = _lp_optimize(inits[0], n, stop1, seed + 1)
            if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                rc2 = np.clip(res, 0, 1000)
                ex = float(evaluate_sequence([float(x) for x in rc2]))
                if math.isfinite(ex) and ex < best_score:
                    best_seq = rc2.copy()
                    best_score = ex
        except Exception:
            pass

    # Other inits
    for idx in range(1, len(inits)):
        if time.monotonic() > phase1_end - 6.0:
            break
        stop = min(phase1_end - 3.0, time.monotonic() + time_other)
        if time.monotonic() >= stop:
            break
        try:
            res, sc = _lp_optimize(inits[idx], n, stop, seed + 10000 * (idx + 1))
            if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                rc2 = np.clip(res, 0, 1000)
                ex = float(evaluate_sequence([float(x) for x in rc2]))
                if math.isfinite(ex) and ex < best_score:
                    best_seq = rc2.copy()
                    best_score = ex
        except Exception:
            pass

    # Phase 2: Subgradient + coordinate polish on best (15% of budget)
    phase2_end = t0 + budget * 0.70
    if time.monotonic() < phase2_end - 5.0:
        sg_stop = min(phase2_end - 2.0, time.monotonic() + (phase2_end - time.monotonic()) * 0.5)
        if sg_stop > time.monotonic() + 5.0:
            try:
                best_seq, best_score = _subgradient_polish(best_seq, sg_stop, seed=5000)
            except Exception:
                pass

        coord_stop = min(phase2_end - 1.0, time.monotonic() + (phase2_end - time.monotonic()) * 0.6)
        if coord_stop > time.monotonic() + 3.0:
            try:
                best_seq, best_score = _coordinate_polish(best_seq, coord_stop, seed=6000)
            except Exception:
                pass

    # Phase 3: More LP passes on current best (15% of budget)
    phase3_end = t0 + budget * 0.85
    if time.monotonic() < phase3_end - 5.0:
        n_passes = 3
        time_per_pass = max(8.0, (phase3_end - time.monotonic()) / n_passes)
        for p in range(n_passes):
            stop = min(phase3_end - 3.0, time.monotonic() + time_per_pass)
            if time.monotonic() >= stop - 1.0:
                break
            try:
                best_seq, best_score = _lp_optimize(best_seq, n, stop, seed + 99000 + p * 31)
            except Exception:
                pass

    # Phase 4: Final coordinate polish (remaining time)
    final_polish_stop = deadline - 3.0
    if time.monotonic() < final_polish_stop - 4.0:
        try:
            best_seq, best_score = _coordinate_polish(best_seq, final_polish_stop, seed=77777)
        except Exception:
            pass

    # Final validation with exact evaluator
    final_seq = np.clip(best_seq, 0, 1000)
    try:
        final_sc = float(evaluate_sequence([float(x) for x in final_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in fallback]

    return [float(x) for x in final_seq]
