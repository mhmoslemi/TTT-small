import numpy as np
import math
import time
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import minimize, linprog


class _Timeout(Exception):
    pass


_fft_len_cache = {}


def _fft_len(m: int) -> int:
    m = int(m)
    if m <= 0:
        return 1
    v = _fft_len_cache.get(m)
    if v is None:
        try:
            v = int(next_fast_len(m))
        except Exception:
            v = 1
            while v < m:
                v <<= 1
        if v < m:
            v = int(m)
        _fft_len_cache[m] = v
    return v


def _clean(a, target_sum: float = 1.0) -> np.ndarray:
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([1.0], dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not math.isfinite(s) or s <= 1e-300:
        return np.full(a.size, target_sum / float(a.size), dtype=np.float64)
    return a * (float(target_sum) / s)


def _conv_square(a: np.ndarray) -> np.ndarray:
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    if n == 0:
        return np.array([], dtype=np.float64)
    if n == 1:
        return np.array([float(a[0] * a[0])], dtype=np.float64)
    m = 2 * n - 1
    L = _fft_len(m)
    fa = rfft(a, L)
    c = irfft(fa * fa, L)[:m]
    c = np.asarray(c, dtype=np.float64)
    return np.maximum(c, 0.0)


def _cross(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    a = np.asarray(a, dtype=np.float64).ravel()
    b = np.asarray(b, dtype=np.float64).ravel()
    n1, n2 = int(a.size), int(b.size)
    if n1 == 0 or n2 == 0:
        return np.array([], dtype=np.float64)
    m = n1 + n2 - 1
    L = _fft_len(m)
    c = irfft(rfft(a, L) * rfft(b, L), L)[:m]
    c = np.asarray(c, dtype=np.float64)
    return np.maximum(c, 0.0)


def _valid(d: np.ndarray, a: np.ndarray) -> np.ndarray:
    d = np.asarray(d, dtype=np.float64).ravel()
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    if n == 0 or d.size == 0:
        return np.array([], dtype=np.float64)
    m = int(d.size) + n - 1
    L = _fft_len(m)
    ar = a[::-1].copy()
    c = irfft(rfft(d, L) * rfft(ar, L), L)
    c = np.asarray(c, dtype=np.float64)
    return np.maximum(c[n - 1:2 * n - 1], 0.0)


def _softmax(y: np.ndarray) -> np.ndarray:
    y = np.asarray(y, dtype=np.float64).ravel()
    if y.size == 0:
        return np.array([1.0], dtype=np.float64)
    y = np.where(np.isfinite(y), y, -50.0)
    y = np.clip(y, -50.0, 50.0)
    z = np.exp(y - float(y.max()))
    Z = float(z.sum())
    if not math.isfinite(Z) or Z <= 1e-300:
        return np.ones(y.size, dtype=np.float64) / float(y.size)
    return z / Z


def _score(a) -> float:
    try:
        a = np.asarray(a, dtype=np.float64).ravel()
        if a.size == 0:
            return math.inf
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        s = float(a.sum())
        if not math.isfinite(s) or s < 1e-300:
            return math.inf
        p = a / s
        q = _conv_square(p)
        if q.size == 0:
            return math.inf
        M = float(q.max())
        if not math.isfinite(M) or M < 0.0:
            return math.inf
        return float(2.0 * a.size * M)
    except Exception:
        return math.inf


def _exact_score(a) -> float:
    try:
        lst = [float(x) for x in np.asarray(a, dtype=np.float64).ravel()]
        val = evaluate_sequence(lst)
        return float(val)
    except Exception:
        return math.inf


def _interp(sv, L: int) -> np.ndarray:
    L = int(L)
    if L <= 0:
        return np.array([1.0], dtype=np.float64)
    sv = _clean(sv)
    if sv.size == 0:
        return np.ones(L, dtype=np.float64) / float(L)
    if sv.size == 1:
        return np.full(L, float(sv[0]) / float(L), dtype=np.float64)
    if sv.size == L:
        return sv.copy()
    x0 = np.linspace(0.0, 1.0, sv.size, dtype=np.float64)
    x1 = np.linspace(0.0, 1.0, L, dtype=np.float64)
    return _clean(np.interp(x1, x0, sv))


def _beta_settings(p: np.ndarray):
    q = _conv_square(p)
    M = float(q.max()) if q.size else 1.0 / max(1, p.size)
    if not math.isfinite(M) or M <= 1e-300:
        M = 1.0 / max(1, int(p.size))
    b = 2.0 / M
    cap = max(100.0, 50.0 * float(p.size))
    b = max(1.0, min(b, cap))
    raw = [0.25 * b, 0.75 * b, 2.0 * b, 5.0 * b, 12.0 * b, 25.0 * b]
    out = []
    for x in raw:
        x = float(np.clip(x, 5.0, 200000.0))
        if not out or abs(x - out[-1]) > 1.0:
            out.append(x)
    return out


def _maxfun(L: int) -> int:
    if L <= 256:
        return 900
    if L <= 512:
        return 700
    if L <= 768:
        return 500
    if L <= 1024:
        return 400
    return 280


def _power_profile(L: int, a: float, b: float) -> np.ndarray:
    i = np.arange(1, L + 1, dtype=np.float64)
    w = np.power(i, float(a)) * np.power(float(L + 1) - i, float(b))
    w = np.where(np.isfinite(w), w, 1.0)
    w = np.maximum(w, 1e-12)
    return _clean(w)


def _deterministic_profiles(L: int):
    out = []
    x = (np.arange(L) + 0.5) / float(L)
    for aa, bb in [(-1.0, 1.0), (1.0, -1.0), (-0.5, 0.5),
                   (0.5, -0.5), (-1.0, -1.0), (1.0, 1.0)]:
        out.append(_power_profile(L, aa, bb))
    out.append(_clean(np.sin(np.pi * x) ** 0.5 + 1e-9))
    out.append(_clean(np.cos(0.5 * np.pi * x) ** 0.5 + 1e-9))
    out.append(_clean(np.abs(np.cos(np.pi * x)) + 0.05 * np.sin(2.0 * np.pi * x)))
    out.append(_clean(1.0 + 0.8 * np.cos(2.0 * np.pi * x)))
    out.append(_clean(np.exp(-((x / 0.10) ** 2)) + np.exp(-(((1.0 - x) / 0.10) ** 2)) + 0.25))
    return out


def _random_profiles(L: int, rng, count: int):
    out = []
    x = np.linspace(0.0, 1.0, L, dtype=np.float64)
    for _ in range(int(max(0, count))):
        y = np.zeros(L, dtype=np.float64)
        K = int(rng.integers(2, 7))
        for _k in range(K):
            m = int(rng.integers(1, 9))
            amp = float(rng.uniform(0.25, 1.25))
            ph = float(rng.uniform(0.0, 2.0 * math.pi))
            y += amp * np.cos(2.0 * math.pi * m * x + ph)
        if rng.random() < 0.55:
            sigma = float(rng.uniform(0.05, 0.25))
            y += float(rng.uniform(0.35, 1.70)) * (
                np.exp(-((x / sigma) ** 2)) + np.exp(-(((1.0 - x) / sigma) ** 2))
            )
        if rng.random() < 0.30:
            y += float(rng.uniform(-0.55, 0.55)) * np.sin(2.0 * math.pi * x)
        y = np.where(np.isfinite(y), y, 0.0)
        p = np.exp(np.clip(y - float(y.max()), -50.0, 50.0))
        p = _clean(p)
        out.append(p)
    return out


def _smooth_opt(p, betas, stop: float, maxfun: int = 600, maxiter: int = 180, sym: bool = False):
    best = _clean(p)
    if sym:
        best = _clean(0.5 * (best + best[::-1]))
    best_sc = _score(best)
    best_arr = best.copy()
    n = int(best.size)
    if n < 2 or len(betas) == 0:
        return best_arr, best_sc

    for beta in betas:
        if time.monotonic() >= stop:
            break
        beta = float(beta)
        if not math.isfinite(beta) or beta <= 1e-6:
            continue

        y0 = np.log(np.maximum(best, 1e-300))
        y0 -= float(np.mean(y0))
        y0 = np.clip(y0, -50.0, 50.0)

        def fg(y):
            nonlocal best, best_sc, best_arr
            if time.monotonic() > stop:
                raise _Timeout
            p0 = _softmax(y)
            if sym:
                ps = _clean(0.5 * (p0 + p0[::-1]))
            else:
                ps = p0
            q = _conv_square(ps)
            if q.size == 0:
                return 1e100, np.zeros(n, dtype=np.float64)
            M = float(q.max())
            if not math.isfinite(M) or M <= 1e-300:
                return 1e100, np.zeros(n, dtype=np.float64)

            sc = 2.0 * float(n) * M
            if sc < best_sc - 1e-12:
                best = ps.copy()
                best_sc = float(sc)
                best_arr = ps.copy()

            u = beta * (q - M)
            u = np.clip(u, -50.0, 0.0)
            ex = np.exp(u)
            s = float(ex.sum())
            if not math.isfinite(s) or s <= 1e-300:
                d = np.full(q.size, 1.0 / float(q.size), dtype=np.float64)
                log_s = -math.log(max(1, int(q.size)))
            else:
                d = ex / s
                log_s = math.log(s)

            fun = 2.0 * float(n) * M + (2.0 * float(n) / beta) * max(-50.0, log_s)
            if not math.isfinite(fun):
                fun = 1e100

            vp = _valid(d, ps)
            g = 4.0 * float(n) * vp
            if sym:
                gp = 0.5 * (g + g[::-1])
            else:
                gp = g
            dot = float(np.dot(p0, gp))
            gy = p0 * (gp - dot)
            gy = np.where(np.isfinite(gy), gy, 0.0)
            gy = np.clip(gy, -1e6, 1e6)
            return float(fun), gy

        opts = {
            "maxiter": int(max(50, maxiter)),
            "maxfun": int(max(80, maxfun)),
            "ftol": 1e-11,
            "gtol": 1e-5,
            "maxls": 10,
        }
        try:
            res = minimize(fg, y0, jac=True, method="L-BFGS-B", options=opts, disp=False)
        except _Timeout:
            break
        except Exception:
            continue

        try:
            pnew = _softmax(np.asarray(res.x, dtype=np.float64).ravel())
            if sym:
                pnew = _clean(0.5 * (pnew + pnew[::-1]))
            scnew = _score(pnew)
            if math.isfinite(scnew) and scnew < best_sc - 1e-12:
                best = pnew.copy()
                best_sc = float(scnew)
                best_arr = best.copy()
        except Exception:
            pass

    return best_arr.copy(), best_sc


def _blend_step(p, g):
    p = _clean(p)
    g = _clean(g)
    n = int(p.size)
    if n == 1:
        return p.copy(), 0.0

    cf = _conv_square(p)
    cg = _conv_square(g)
    cfs = _cross(p, g)

    def vals(ts):
        ts = np.asarray(ts, dtype=np.float64)
        o = 1.0 - ts
        mat = (
            (o * o)[:, None] * cf[None, :]
            + (2.0 * o * ts)[:, None] * cfs[None, :]
            + (ts * ts)[:, None] * cg[None, :]
        )
        return 2.0 * float(n) * np.max(mat, axis=1)

    ts = np.linspace(0.0, 1.0, 81, dtype=np.float64)
    v = vals(ts)
    if not np.all(np.isfinite(v)):
        return p.copy(), 0.0
    i = int(np.argmin(v))
    lo = float(ts[max(0, i - 2)])
    hi = float(ts[min(len(ts) - 1, i + 2)])
    if hi <= lo:
        hi = min(1.0, lo + 0.1)
        lo = max(0.0, hi - 0.1)
    ts2 = np.linspace(lo, hi, 121, dtype=np.float64)
    v2 = vals(ts2)
    if not np.all(np.isfinite(v2)):
        return p.copy(), 0.0
    j = int(np.argmin(v2))
    t = float(ts2[j])

    if t <= 1e-12:
        return p.copy(), 0.0
    if t >= 1.0 - 1e-12:
        return g.copy(), 1.0
    arr = (1.0 - t) * p + t * g
    return _clean(arr), t


def _lp_direction(p, stop: float, full: bool = False, active_frac: float = 0.35):
    if time.monotonic() > stop:
        return None
    p = _clean(p)
    n = int(p.size)
    if n < 2:
        return None

    q = _conv_square(p)
    K = int(q.size)
    if K <= 0:
        return None

    if full:
        rows = np.arange(K, dtype=np.int64)
    else:
        frac = float(min(1.0, max(0.0, active_frac)))
        cnt = int(round(frac * K))
        cnt = max(12, min(K, cnt))
        rows = np.argsort(q)[::-1][:cnt].astype(np.int64)

    A = np.zeros((int(rows.size), n + 1), dtype=np.float64)
    for r, k in enumerate(rows):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(k, n - 1) + 1
        if hi > lo:
            A[r, lo:hi] = p[lo:hi]
        A[r, n] = -1.0

    c = np.zeros(n + 1, dtype=np.float64)
    c[n] = 1.0
    Aeq = np.zeros((1, n + 1), dtype=np.float64)
    Aeq[0, :n] = 1.0
    beq = np.array([1.0], dtype=np.float64)
    bounds = [(0.0, None)] * (n + 1)

    try:
        res = linprog(
            c,
            A_ub=A,
            b_ub=np.zeros(int(rows.size), dtype=np.float64),
            A_eq=Aeq,
            b_eq=beq,
            bounds=bounds,
            method="highs",
            options={"presolve": True},
        )
    except Exception:
        return None

    if res is None or getattr(res, "status", 1) != 0 or getattr(res, "x", None) is None:
        return None

    g = np.asarray(res.x[:n], dtype=np.float64)
    g = np.where(np.isfinite(g), g, 0.0)
    g = np.maximum(g, 0.0)
    s = float(g.sum())
    if s <= 1e-15:
        return None
    return g / s


def _lp_improve(p, stop: float, max_iters: int = 3, rng=None, sym: bool = False, max_n: int = 1200):
    best = _clean(p)
    if sym:
        best = _clean(0.5 * (best + best[::-1]))
    best_sc = _score(best)
    n = int(best.size)
    if n < 2 or n > max_n:
        return best.copy(), best_sc

    fracs = [0.25, 0.45, 1.0]
    noimp = 0
    for it in range(int(max(0, max_iters))):
        if time.monotonic() > stop - 5.0:
            break
        frac = fracs[it % len(fracs)]
        full = frac >= 1.0
        g = _lp_direction(best, stop, full=full, active_frac=frac)
        if g is None:
            break
        if sym:
            g = _clean(0.5 * (g + g[::-1]))

        arr, _t = _blend_step(best, g)
        if sym:
            arr = _clean(0.5 * (arr + arr[::-1]))
        sc = _score(arr)
        if math.isfinite(sc) and sc < best_sc - 1e-12:
            best = arr
            best_sc = sc
            noimp = 0
        else:
            noimp += 1
            if rng is not None and noimp <= 3:
                noise_scale = 0.25 * float(best.mean())
                cand = best + rng.normal(0.0, max(noise_scale, 1e-12), n)
                if sym:
                    cand =-_clean(0.5) + 0.5 * cand  # placeholder to avoid syntax only; replaced below
                if sym:
                    cand = cand + cand[::-1]
                    cand = _clean(0.5 * cand)
                else:
                    cand = _clean(cand)
                sc2 = _score(cand)
                if math.isfinite(sc2) and sc2 < best_sc:
                    best = cand
                    best_sc = sc2
                    noimp = 0
            if noimp >= 3:
                break
    return best.copy(), best_sc


def _coordinate_polish(p, stop: float, seed: int = 1, sym: bool = False, passes: int = None):
    best = _clean(p)
    if sym:
        best = _clean(0.5 * (best + best[::-1]))
    best_sc = _score(best)
    n = int(best.size)
    if n < 2:
        return best.copy(), best_sc

    rng = np.random.default_rng(int(seed))
    if passes is None:
        if sym:
            passes = 5 if n <= 300 else (3 if n <= 700 else 2)
        else:
            passes = 4 if n <= 300 else (3 if n <= 600 else (2 if n <= 1200 else 1))

    work_n = ((n + 1) // 2) if sym else n
    mults = (0.0, 0.10, 0.25, 0.50, 0.75, 1.25, 1.50, 2.00, 3.50)

    for _pass in range(int(max(0, passes))):
        if time.monotonic() > stop:
            break
        order = rng.permutation(work_n)
        improved = False
        avg = float(best.mean())

        for idx in order:
            if time.monotonic() > stop:
                break
            idx = int(idx)
            orig = float(best[idx])
            base = orig if orig > 1e-15 else avg
            vals = {orig}
            for mm in mults:
                v = base * float(mm)
                if 0.0 <= v <= 1000.0:
                    vals.add(v)
            if orig <= 1.5 * max(avg, 1e-300):
                for mm in (1.0, 2.0, 4.0, 8.0):
                    v = avg * float(mm)
                    if 0.0 <= v <= 1000.0:
                        vals.add(v)

            best_val = orig
            best_val_sc = best_sc
            mirror = n - 1 - idx if sym else -1

            for val in vals:
                val = float(val)
                if val < 0.0:
                    val = 0.0
                best[idx] = val
                if sym and mirror != idx:
                    best[mirror] = val
                sc = _score(best)
                if math.isfinite(sc) and sc < best_val_sc:
                    best_val_sc = sc
                    best_val = val

            best[idx] = best_val
            if sym and mirror != idx:
                best[mirror] = best_val

            if best_val_sc < best_sc - 1e-13:
                best = _clean(best)
                if sym:
                    best = _clean(0.5 * (best + best[::-1]))
                best_sc = float(best_val_sc)
                improved = True

        if not improved:
            break

    return best.copy(), best_sc


def _get_parent():
    try:
        h = globals().get("height_sequence_1", None)
    except Exception:
        h = None
    if h is None:
        return None
    try:
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        return None
    p = _clean(p)
    if p.size == 0:
        return None
    return p


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = max(10.0, float(budget_s))
    except Exception:
        budget = 1300.0
    hard = t0 + budget

    if budget > 600.0:
        reserve = 45.0
    elif budget > 200.0:
        reserve = 25.0
    elif budget > 60.0:
        reserve = 10.0
    else:
        reserve = max(1.0, 0.05 * budget)
    search_deadline = max(t0 + 2.0, hard - reserve)

    rng = np.random.default_rng(int(seed) if seed is not None else 42)

    parent = _get_parent()
    if parent is None:
        initial = np.ones(256, dtype=np.float64) / 256.0
        base_len = 256
    else:
        initial = parent.copy()
        base_len = int(parent.size)

    best_seq = _clean(initial)
    best_int = _score(best_seq)
    best_exact = math.inf
    fallback = best_seq.copy()

    if not math.isfinite(best_int) or best_seq.size == 0:
        best_seq = np.ones(256, dtype=np.float64) / 256.0
        best_int = _score(best_seq)
        fallback = best_seq.copy()
        best_exact = math.inf

    def _try_candidate(arr):
        nonlocal best_seq, best_int, best_exact, fallback
        if arr is None:
            return
        try:
            a = _clean(arr)
        except Exception:
            return
        if a.size == 0:
            return

        cands = [a]
        if a.size > 1:
            ar = a[::-1].copy()
            cands.append(ar)
            cands.append(_clean(0.5 * (a + ar)))

        seen = set()
        for c in cands:
            try:
                key = tuple(np.round(c[:min(64, c.size)], 8).tolist())
            except Exception:
                key = None
            if key is not None and key in seen:
                continue
            if key is not None:
                seen.add(key)

            scint = _score(c)
            if not math.isfinite(scint):
                continue
            if not (scint < best_int + 0.02 or math.isinf(best_exact)):
                continue

            scex = _exact_score(c)
            if math.isfinite(scex):
                if scex < best_exact - 1e-10:
                    best_exact = float(scex)
                    best_int = float(scint)
                    best_seq = c.copy()
                    fallback = c.copy()
            else:
                if math.isinf(best_exact) and scint < best_int + 0.05:
                    best_exact = float(scint)
                    best_int = float(scint)
                    best_seq = c.copy()
                    fallback = c.copy()

    _try_candidate(initial)

    def _process_length(L: int, stop: float, do_full: bool = False):
        nonlocal best_seq, best_int, best_exact, fallback
        now = time.monotonic()
        stop = min(float(stop), search_deadline)
        if stop - now < 40.0:
            return
        L = int(max(8, L))
        if L > 2048:
            L = 2048

        seeds = []
        if best_seq is not None and best_seq.size > 0:
            seeds.append(_interp(best_seq, L))
        if parent is not None:
            seeds.append(_interp(parent, L))
        seeds.append(np.ones(L, dtype=np.float64) / float(L))

        nrand = 10 if L <= 512 else 6
        seeds.extend(_random_profiles(L, rng, nrand))
        seeds.extend(_deterministic_profiles(L))

        scored = []
        for arr in seeds:
            try:
                a = _clean(arr)
            except Exception:
                continue
            if a.size != L:
                continue
            sc = _score(a)
            if math.isfinite(sc):
                scored.append((sc, a))

        if not scored:
            return
        scored.sort(key=lambda z: z[0])

        rem = stop - time.monotonic()
        top_n = 3 if rem > 420.0 else (2 if rem > 180.0 else 1)

        for rank, (sc0, a0) in enumerate(scored[:top_n]):
            now = time.monotonic()
            if stop - now < 35.0:
                break
            if rank > 0 and math.isfinite(best_exact) and sc0 > best_exact + 0.20 and stop - now < 220.0:
                break

            p = a0.copy()
            if float(np.std(p)) < 1e-7:
                p = _clean(p + rng.normal(0.0, 0.10 * max(float(p.mean()), 1e-15), L))

            total = max(20.0, stop - time.monotonic())
            betas = _beta_settings(p)
            low_b = betas[:3] if len(betas) >= 3 else betas[-3:]
            high_b = betas[3:] if len(betas) > 3 else betas[-2:]

            stop1 = min(stop - 20.0, time.monotonic() + 0.35 * total)
            if stop1 - time.monotonic() > 8.0:
                p, _ = _smooth_opt(p, low_b, stop1, maxfun=_maxfun(L), maxiter=180, sym=True)
                _try_candidate(p)

            if L <= 1024 and time.monotonic() < stop - 60.0:
                lp_stop = min(stop - 45.0, time.monotonic() + max(25.0, 0.25 * max(0.0, stop - time.monotonic())))
                maxit = 3 if L <= 512 else 2
                p, _ = _lp_improve(p, lp_stop, max_iters=maxit, rng=rng, sym=True)
                _try_candidate(p)

            stop2 = min(stop - 20.0, time.monotonic() + max(25.0, 0.30 * max(0.0, stop - time.monotonic())))
            if stop2 - time.monotonic() > 8.0:
                p, _ = _smooth_opt(p, high_b, stop2, maxfun=_maxfun(L), maxiter=180, sym=True)
                _try_candidate(p)

            if L <= 1200 and time.monotonic() < stop - 12.0:
                cof_stop = min(stop - 5.0, time.monotonic() + max(10.0, 0.15 * max(0.0, stop - time.monotonic())))
                p, _ = _coordinate_polish(p, cof_stop, seed=int(rng.integers(0, 10**9)), sym=True)
                _try_candidate(p)

            if do_full and L <= 600 and time.monotonic() < stop - 60.0:
                total2 = max(0.0, stop - time.monotonic())
                stopF = min(stop - 25.0, time.monotonic() + 0.45 * total2)
                bset = _beta_settings(p)
                p, _ = _smooth_opt(p, bset[:3] if len(bset) >= 3 else bset, stopF,
                                   maxfun=_maxfun(L), maxiter=180, sym=False)
                _try_candidate(p)

                if L <= 800 and time.monotonic() < stop - 25.0:
                    lpF = min(stop - 20.0, time.monotonic() + 0.30 * max(0.0, stop - time.monotonic()))
                    p, _ = _lp_improve(p, lpF, max_iters=2, rng=rng, sym=False)
                    _try_candidate(p)

                stopF2 = min(stop - 10.0, time.monotonic() + 0.30 * max(0.0, stop - time.monotonic()))
                if stopF2 - time.monotonic() > 8.0:
                    p, _ = _smooth_opt(p, bset[3:] if len(bset) > 3 else bset[-2:], stopF2,
                                       maxfun=_maxfun(L), maxiter=180, sym=False)
                    _try_candidate(p)

    planned = []
    if base_len == 256:
        planned = [(256, 0.30), (512, 0.24), (384, 0.14), (768, 0.12), (1024, 0.14), (1536, 0.06)]
    elif base_len == 512:
        planned = [(512, 0.32), (1024, 0.26), (768, 0.14), (1536, 0.18), (2048, 0.10)]
    else:
        req = [
            (base_len, 0.34),
            (base_len * 2, 0.26),
            (int(round(base_len * 1.5)), 0.16),
            (base_len * 3, 0.12),
            (min(2048, base_len * 4), 0.12),
        ]
        seen = set()
        for L, w in req:
            L = int(max(8, min(2048, L)))
            if L in seen:
                continue
            seen.add(L)
            planned.append((L, float(w)))
        if not planned:
            planned = [(base_len, 1.0)]
        if len(planned) > 5:
            planned = planned[:5]

    total_w = float(sum(w for _, w in planned))
    if total_w <= 0.0:
        planned = [(base_len, 1.0)]
        total_w = 1.0

    span = max(0.0, search_deadline - t0)
    running_w = 0.0
    for i, (L, w) in enumerate(planned):
        running_w += float(w)
        stop = min(search_deadline, t0 + span * (running_w / total_w))
        if stop - time.monotonic() < 30.0:
            break
        do_full = (i == 0 and int(best_seq.size) <= 512 and (search_deadline - time.monotonic()) > 200.0)
        _process_length(L, stop, do_full=do_full)
        if time.monotonic() >= search_deadline - 10.0:
            break

    if time.monotonic() < search_deadline - 30.0:
        L = int(best_seq.size)
        if not (8 <= L <= 2048):
            L = 256
        _process_length(L, search_deadline - 5.0, do_full=False)

    final_stop = hard - 3.0
    if best_seq.size >= 8 and best_seq.size <= 1200:
        if time.monotonic() < final_stop - 30.0:
            bset = _beta_settings(best_seq)
            p, _ = _smooth_opt(best_seq, bset[-3:] if len(bset) >= 3 else bset,
                               final_stop - 15.0, maxfun=350, maxiter=120, sym=True)
            _try_candidate(p)
        if time.monotonic() < final_stop - 10.0:
            p, _ = _coordinate_polish(best_seq, final_stop - 4.0, seed=987654321, sym=True)
            _try_candidate(p)

    def _finalize_list(arr):
        seq = np.asarray(arr, dtype=np.float64).ravel()
        if seq.size == 0:
            seq = np.ones(32, dtype=np.float64) / 32.0
        seq = np.where(np.isfinite(seq), seq, 0.0)
        seq = np.clip(seq, 0.0, 1000.0)
        if float(seq.sum()) < 1e-8:
            seq = np.ones(32, dtype=np.float64) / 32.0
        return [float(x) for x in seq]

    final_list = _finalize_list(best_seq)
    sc_final = _exact_score(final_list)

    if not math.isfinite(sc_final):
        fb_list = _finalize_list(fallback)
        sc_fb = _exact_score(fb_list)
        if math.isfinite(sc_fb):
            final_list = fb_list
        else:
            final_list = [1.0] / [1]
            final_list = [1.0]
            sc_fb2 = _exact_score(final_list)
            if not math.isfinite(sc_fb2):
                final_list = _finalize_list(np.ones(32, dtype=np.float64) / 32.0)

    if time.monotonic() > hard - 0.25:
        return final_list

    return final_list
