        except Exception:
            pass
    return np.ones(512, dtype=np.float64) / 512.0


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-300:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / s


def _fft_conv(a):
    n = a.size
    nfft = next_fast_len(2 * n - 1)
    fa = rfft(a, nfft)
    c = irfft(fa * fa, nfft)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _fft_grad(a, w):
    n = a.size
    mr = w.size
    nfft = next_fast_len(n + mr - 1)
    fa = rfft(a[::-1], nfft)
    fw = rfft(w, nfft)
    conv_full = irfft(fa * fw, nfft)[:n + mr - 1]
    g = 2.0 * conv_full[n - 1:2 * n - 1]
    return g


def _exact_score(a):
    try:
        out = [float(x) for x in a]
        return float(evaluate_sequence(out))
    except Exception:
        return float('inf')


def _lp_direction(a, M, deadline):
    n = a.size
    if n < 2 or n > 2048 or M <= 0 or not np.isfinite(M):
        return None
    rem = deadline - time.monotonic()
    if rem < 2.0:
        return None
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = a
    c_obj = -np.ones(n, dtype=np.float64)
    b_ub = np.full(m, M, dtype=np.float64)
    tl = max(0.5, min(3.0, rem - 1.0))
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0.0, None),
                      method='highs', options={'time_limit': tl, 'presolve': True})
        if res.success and res.x is not None:
            b = np.asarray(res.x, dtype=np.float64)
            b = np.maximum(b, 0.0)
            bs = b.sum()
            if bs > 1e-12:
                return b / bs
    except Exception:
        pass
    return None


def _optimize(a_init, deadline, rng):
    n = a_init.size
    a = _normalize(a_init.copy())
    c = _fft_conv(a)
    M = float(c.max())

    best_a = a.copy()
    best_M = M

    T = 0.08
    eta = 0.025
    patience = 0
    it = 0
    last_lp = time.monotonic()
    n_no_lp_ok = 0

    while time.monotonic() < deadline - 25:
        it += 1

        z = (c - M) / (T * M + 1e-30)
        z = np.clip(z, -60.0, 0.0)
        w = np.exp(z)
        sw = w.sum()
        if sw > 0 and np.isfinite(sw):
            w = w / sw
        else:
            w = np.zeros(c.shape, dtype=np.float64)
            w[int(np.argmax(c))] = 1.0

        g = _fft_grad(a, w)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            patience += 50
            if patience > 400:
                break
            continue

        a_new = a - eta * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
            a_new[int(rng.integers(n))] += 1e-5
            a_new = _normalize(a_new)
            s = 1.0
        a_new = a_new / s

        c_new = _fft_conv(a_new)
        M_new = float(c_new.max())

        if M_new < M * (1.0 - 1e-9):
            a, c, M = a_new, c_new, M_new
            patience = 0
            eta = min(eta * 1.008, 0.06)
            if M < best_M * (1.0 - 1e-12):
                best_a = a.copy()
                best_M = M
        else:
            patience += 1
            eta = max(eta * 0.93, 0.0005)
            if patience > 250:
                noise_scale = min(0.05, 0.01 * (1.0 + patience / 100.0))
                noise = rng.normal(0, noise_scale, n)
                a_try = _normalize(a + noise)
                c_try = _fft_conv(a_try)
                M_try = float(c_try.max())
                if M_try < best_M * 1.15:
                    a, c, M = a_try, c_try, M_try
                elif M_try >= best_M * 1.5:
                    a = best_a.copy()
                    c = _fft_conv(a)
                    M = best_M
                patience = 0
                eta = 0.02
                T = max(0.04, T * 1.4)

        T = max(0.008, T * 0.9997)

        do_lp = (it % 800 == 0) or (time.monotonic() - last_lp > 12.0)
        if do_lp and time.monotonic() < deadline - 55:
            last_lp = time.monotonic()
            try:
                dirb = _lp_direction(a, M, deadline - 30)
                if dirb is not None:
                    improved = False
                    for alpha in (0.1, 0.2, 0.35, 0.5, 0.7, 1.0):
                        cand = _normalize(a + alpha * (dirb - a))
                        c_cand = _fft_conv(cand)
                        M_cand = float(c_cand.max())
                        if M_cand < M * (1.0 - 1e-9):
                            a = cand
                            c = c_cand
                            M = M_cand
                            if M < best_M * (1.0 - 1e-12):
                                best_a = a.copy()
                                best_M = M
                            improved = True
                            break
                    if not improved:
                        n_no_lp_ok += 1
            except Exception:
                pass

    return best_a, best_M


def _try_lengths(a_init, deadline, rng):
    best_a = _normalize(a_init.copy())
    best_score = _exact_score(best_a)
    n_init = a_init.size

    lengths = []
    if 32 <= n_init <= 1024:
        lengths.append(n_init)
    for L in (384, 448, 512, 640):
        if L not in lengths and 32 <= L <= 1024:
            lengths.append(L)

    for L in lengths:
        rem = deadline - time.monotonic() - 20
        if rem < 80:
            break
        time_here = min(rem * 0.7, 400.0)
        if time_here < 60:
            break

        if L == n_init:
            a_L = _normalize(a_init.copy())
        else:
            a_L = _resample(best_a, L)

        sub_deadline = time.monotonic() + time_here
        try:
            a_opt, M_opt = _optimize(a_L, sub_deadline, rng)
            sc = _exact_score(a_opt)
            if np.isfinite(sc) and sc < best_score:
                best_a = a_opt.copy()
                best_score = sc
        except Exception:
            pass

    return best_a, best_score


def _resample(a, m):
    a = _normalize(a)
    n = a.size
    m = int(m)
    if m <= 0:
        return np.ones(1)
    if n == m:
        return a.copy()
    if m > n:
        idx = (np.arange(m, dtype=np.int64) * n) // m
        counts = np.bincount(idx, minlength=n).astype(np.float64)
        counts[counts <= 0] = 1.0
        b = a[idx] / counts[idx]
    else:
        b = np.zeros(m, dtype=np.float64)
        idx = (np.arange(n, dtype=np.int64) * m) // n
        np.add.at(b, idx, a)
    return _normalize(b)


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = max(0.0, float(budget_s))
    except Exception:
        budget = 1300.0

    if budget <= 3.0:
        return [1.0]

    deadline = t0 + budget - 12.0
    rng = np.random.default_rng(42)
    if isinstance(seed, (int, np.integer)):
        rng = np.random.default_rng(int(seed))
    elif isinstance(seed, (float, np.floating)) and np.isfinite(float(seed)):
        rng = np.random.default_rng(int(seed))

    parent = _get_parent()
    n = parent.size
    a = _normalize(parent.copy())

    best_a = a.copy()
    best_score = _exact_score(a)
    if not np.isfinite(best_score):
        best_a = np.ones(512, dtype=np.float64) / 512.0
        best_score = _exact_score(best_a)

    if best_score >= 2.0:
        x = np.linspace(0, 1, n, endpoint=False)
        alt = (1.0 + 0.3 * np.cos(2 * np.pi * x))
        alt = _normalize(alt)
        sc_alt = _exact_score(alt)
        if sc_alt < best_score:
            best_a = alt.copy()
            best_score = sc_alt

    main_deadline = deadline - 80.0
    if main_deadline > time.monotonic() + 100:
        try:
            a_main, M_main = _optimize(best_a, main_deadline, rng)
            sc_main = _exact_score(a_main)
            if np.isfinite(sc_main) and sc_main < best_score:
                best_a = a_main.copy()
                best_score = sc_main
        except Exception:
            pass

    if time.monotonic() < deadline - 50:
        try:
            best_a, best_score = _try_lengths(best_a, deadline - 30, rng)
        except Exception:
            pass

    if time.monotonic() < deadline - 25:
        try:
            c = _fft_conv(best_a)
            M = float(c.max())
            dirb = _lp_direction(best_a, M, deadline - 5)
            if dirb is not None:
                for alpha in (0.05, 0.1, 0.2, 0.3, 0.5, 0.8, 1.0):
                    cand = _normalize(best_a + alpha * (dirb - best_a))
                    sc = _exact_score(cand)
                    if np.isfinite(sc) and sc < best_score:
                        best_a = cand.copy()
                        best_score = sc
                        break
        except Exception:
            pass

    best_a = np.asarray(best_a, dtype=np.float64).ravel()
    best_a = np.where(np.isfinite(best_a), best_a, 0.0)
    best_a = np.clip(best_a, 0.0, 1000.0)
    best_a = _normalize(best_a)

    final_score = _exact_score(best_a)
    if not np.isfinite(final_score):
        fb = np.ones(512, dtype=np.float64) / 512.0
        fs = _exact_score(fb)
        if np.isfinite(fs):
            return [float(x) for x in fb]
        return [1.0]

    return [float(x) for x in best_a]
