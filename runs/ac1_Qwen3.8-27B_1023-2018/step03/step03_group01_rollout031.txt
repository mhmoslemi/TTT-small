    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_fft(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)

def _build_sparse_A(f, n):
    """Build sparse constraint matrix for the LP.
    A[k, j] = f[k-j] if 0 <= k-j <= n-1, else 0
    Shape: (2n-1) x n
    """
    rows = []
    cols = []
    vals = []
    for j in range(n):
        r0 = j
        r1 = min(j + n, 2 * n)  # k ranges from j to j+n-1, clipped
        for k in range(r0, r1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A

def _lp_solve(f, n, M, time_limit=20.0):
    A = _build_sparse_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:

            c, A_ub=A, b_ub=b_ub,
            bounds=[(0.0, None)] * n,
            method="highs",
            options={"presolve": True, "time_limit": time_limit, "mip_rel_gap": 0},
        )

        return None
    g = np.asarray(res.x, dtype=np.float64)

import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _conv_fft(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_fft(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _normalize(a):
    s = float(a.sum())
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def _build_sparse_A(f, n):
    """A[k, j] = f[k-j] if 0 <= k-j <= n-1, else 0. Shape (2n-1, n)."""
    nrows = 2 * n - 1
    # Preallocate
    max_nz = n * n  # upper bound
    rows = np.empty(max_nz, dtype=np.int32)
    cols = np.empty(max_nz, dtype=np.int32)
    vals = np.empty(max_nz, dtype=np.float64)
    cnt = 0
    for j in range(n):
        k_lo = j
        k_hi = min(j + n, nrows)  # k-j ranges from 0 to n-1, k ranges from j to j+n-1
        L = k_hi - k_lo
        rows[cnt:cnt+L] = np.arange(k_lo, k_hi)
        cols[cnt:cnt+L] = j
        vals[cnt:cnt+L] = f[:L]
        cnt += L
    A = csr_matrix((vals[:cnt], (rows[:cnt], cols[:cnt])), shape=(nrows, n))
    return A


def _lp_direction(f, n, M, time_limit=15.0):
    """Solve max sum(g) s.t. A@g <= M, g >= 0. Return g normalized to sum 1."""
    A = _build_sparse_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, None)] * n,
                      method="highs",
                      options={"presolve": True, "time_limit": time_limit})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    gs = float(g.sum())
    if gs < 1e-14:
        return None
    return g / gs

    f = _normalize(f)

    max_stall = 40

        if time.monotonic() > stop_time - 2.0:
            break

        b = _conv_fft(f)
        M = float(b.max())
        if M < 1e-15:

        if g_dir is None:

            if stall > max_stall:

            f = np.clip(f + rng.normal(0, 0.03, n), 0, None)
            f = _normalize(f)
            t_step = 0.2
            continue

        dist = float(np.linalg.norm(g_dir - f))
        if dist < 1e-7:
            stall += 3
            if stall > max_stall:

            f = np.clip(f + rng.normal(0, 0.05, n), 0, None)
            f = _normalize(f)
            t_step = 0.25

def _subgrad_polish(f0, n, stop_time, seed):
    """Subgradient method with top-K lag weighting and momentum."""
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    f = _normalize(f)
    best_f = f.copy()
    best_sc = _score(f)
    lr = 0.02
    beta = 0.8
    momentum = np.zeros(n)
    no_imp = 0

    for it in range(500000):
        if time.monotonic() > stop_time - 2.0:
            break
        if it % 500 == 0 and time.monotonic() > stop_time - 10.0:
            break

        b = _conv_fft(f)
        M = float(b.max())
        S = float(f.sum())

        # Top-K lags
        K = min(10, 2 * n - 1)
        idx = np.argpartition(b, -K)[-K:]
        w = b[idx].copy()
        w = np.exp((w - w.max()) * 3.0)  # soften
        w /= w.sum()

        # Gradient of M
        grad_M = np.zeros(n)
        for w_k, k in zip(w, idx):
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            for i in range(lo, hi):
                j = k - i
                if 0 <= j < n:
                    grad_M[j] += w_k * f[i]

        # Gradient of score = 2n*M/S^2
        # d(score)/dx_j = 2n * (grad_M[j] * S - 2*M) / S^3
        grad = (2.0 * n / S**3) * (grad_M * S - 2.0 * M)

        # Momentum update
        momentum = beta * momentum + grad
        f_new = f - lr * momentum
        f_new = np.maximum(f_new, 0.0)
        f_new = _normalize(f_new)

        sc = _score(f_new)
        if sc < best_sc - 1e-14:
            f = f_new
            best_f = f.copy()
            best_sc = sc
            no_imp = 0
            lr = min(0.08, lr * 1.005)
        else:
            no_imp += 1
            lr = max(0.002, lr * 0.98)
            f = f_new  # still take the step for exploration
            if no_imp > 500:
                # Reset
                lr = 0.02
                beta = 0.7
                momentum = np.zeros(n)
                no_imp = 0

    return best_f, best_sc


def _coord_polish(f0, n, stop_time, seed):
    """Coordinate descent with single and paired moves."""
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    f = _normalize(f)
    best_sc = _score(f)
    no_imp_rounds = 0

    single_factors = (0.3, 0.5, 0.75, 1.25, 1.5, 2.0, 0.1)
    pair_factors = ((0.5, 2.0), (2.0, 0.5), (0.3, 1.5), (1.5, 0.3),
                    (0.75, 1.3), (1.3, 0.75))

    while time.monotonic() < stop_time - 3.0:
        improved = False

        # Single coordinate moves
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 3.0:
                return f, best_sc
            orig = f[i]
            if orig < 1e-12:
                # Try setting to small value
                for v in (0.001, 0.01, 0.05):
                    f[i] = v
                    f = _normalize(f)
                    sc = _score(f)
                    if sc < best_sc - 1e-13:
                        best_sc = sc
                        improved = True
                    f[i] = 0.0
                    f = _normalize(f)
                continue
            for mlt in single_factors:
                f[i] = orig * mlt
                sc = _score(f)
                if sc < best_sc - 1e-13:
                    best_sc = sc
                    improved = True
                    orig = f[i]
            f[i] = orig
            # Fine-tune
            for dt in (0.02, -0.02, 0.05, -0.05):
                f[i] = max(0.0, orig * (1.0 + dt))
                sc = _score(f)
                if sc < best_sc - 1e-13:
                    best_sc = sc
                    improved = True
            f[i] = orig

        # Paired coordinate moves
        for _ in range(60):
            if time.monotonic() > stop_time - 3.0:
                return f, best_sc
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            oi, oj = f[i], f[j]
            for mi, mj in pair_factors:
                f[i] = oi * mi if oi > 1e-12 else 0.005
                f[j] = oj * mj if oj > 1e-12 else 0.005
                sc = _score(f)
                if sc < best_sc - 1e-13:
                    best_sc = sc
                    improved = True
            f[i], f[j] = oi, oj

        if improved:
            no_imp_rounds = 0
        else:
            no_imp_rounds += 1
            if no_imp_rounds >= 8:
                break

    return f, best_sc

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _conv_fft(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_fft(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _normalize(a):
    s = float(a.sum())
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


class _LPHelper:
    def __init__(self, n):
        self.n = n
        self.nrows = 2 * n - 1
        self.col_lengths = np.minimum(n, self.nrows - np.arange(n))
        total = int(self.col_lengths.sum())
        self.rows = np.empty(total, dtype=np.int32)
        self.cols = np.empty(total, dtype=np.int32)
        offset = 0
        for j in range(n):
            L = int(self.col_lengths[j])
            self.rows[offset:offset+L] = np.arange(j, j+L, dtype=np.int32)
            self.cols[offset:offset+L] = j
            offset += L
        self._bounds = [(0.0, None)] * n
        self._c = -np.ones(n)
    
    def build_A(self, f):
        vals = np.empty(int(self.col_lengths.sum()), dtype=np.float64)
        offset = 0
        for j in range(self.n):
            L = int(self.col_lengths[j])
            vals[offset:offset+L] = f[:L]
            offset += L
        return csr_matrix((vals, (self.rows, self.cols)), shape=(self.nrows, self.n))
    
    def solve(self, f, M, time_limit=10.0):
        A = self.build_A(f)
        b_ub = np.full(self.nrows, M)
        try:
            res = linprog(self._c, A_ub=A, b_ub=b_ub,
                          bounds=self._bounds,
                          method="highs",
                          options={"presolve": True, "time_limit": time_limit})
        except Exception:
            return None
        if res.status != 0 or res.x is None:
            return None
        g = np.asarray(res.x, dtype=np.float64)
        gs = float(g.sum())
        if gs < 1e-14:
            return None
        return g / gs

        dist = float(np.linalg.norm(g_dir - f))
        if dist < 1e-8:
            stall += 3
            if stall > 50:
                break
            f = _normalize(np.clip(f + rng.normal(0, 0.05, n), 0, None))
            t_step = 0.25
            continue

        t_lo, t_hi = 0.0, t_step
        f_lo = f
        best_t = 0.0
        for _ls in range(22):
            if time.monotonic() > stop_time - 2.0:
                break
            t_mid = 0.5 * (t_lo + t_hi)
            fn = (1.0 - t_mid) * f + t_mid * g_dir
            np.maximum(fn, 0.0, out=fn)
            fs = float(fn.sum())
            if fs < 1e-10:
                t_hi = t_mid
                continue
            fn = fn / fs
            sc = _score(fn)
            if sc < best_sc - 1e-15:
                t_lo = t_mid
                f_lo = fn.copy()
                best_t = t_mid
            else:
                t_hi = t_mid
            if t_hi - t_lo < 5e-5:
                break

        if best_t < 1e-6:
            consecutive_no_improve += 1
            t_step = max(0.02, t_step * 0.6)
        else:
            f = f_lo
            sc_new = _score(f)
            if sc_new < best_sc - 1e-14:
                best_f = f.copy()
                best_sc = sc_new
                t_step = min(0.55, best_t * 1.15)
                stall = 0
                consecutive_no_improve = 0
            else:
                consecutive_no_improve += 1
                t_step = max(0.03, t_step * 0.7)

        if consecutive_no_improve >= 60:
            f = _normalize(np.clip(f + rng.normal(0, 0.08, n), 0, None))
            t_step = 0.30
            stall = 0
            consecutive_no_improve = 0

    return best_f, best_sc


def _subgrad_polish(f0, n, stop_time, seed):
    rng = np.random.default_rng(seed)
    f = _normalize(np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None))
    best_f = f.copy()
    best_sc = _score(f)
    lr = 0.03
    beta = 0.80
    momentum = np.zeros(n)
    no_imp = 0
    it = 0

    while time.monotonic() < stop_time - 2.0 and it < 300000:
        it += 1
        if it % 1000 == 0 and time.monotonic() > stop_time - 10.0:
            break

        b = _conv_fft(f)
        M = float(b.max())
        S = float(f.sum())
        if S < 1e-10:
            break

        K = min(12, 2 * n - 1)
        idx = np.argpartition(b, -K)[-K:]
        w = b[idx].copy()
        w_exp = w - w.max()
        np.exp(w_exp * 2.5, out=w_exp)
        w = w_exp / w_exp.sum()

        grad_M = np.zeros(n)
        for w_k, k in zip(w, idx):
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            for i in range(lo, hi):
                j = k - i
                if 0 <= j < n:
                    grad_M[j] += w_k * f[i]

        grad = (2.0 * n / (S * S * S)) * (grad_M * S - 2.0 * M)
        momentum = beta * momentum + grad
        f_new = f - lr * momentum
        f_new = np.maximum(f_new, 0.0)
        fs = float(f_new.sum())
        if fs < 1e-10:
            f_new = f.copy()
        else:
            f_new = f_new / fs

        sc = _score(f_new)
        if sc < best_sc - 1e-15:
            f = f_new
            best_f = f.copy()
            best_sc = sc
            no_imp = 0
            lr = min(0.06, lr * 1.003)
        else:
            no_imp += 1
            lr = max(0.003, lr * 0.985)
            f = f_new
            if no_imp > 400:
                lr = 0.025
                beta = 0.70
                momentum = np.zeros(n)
                no_imp = 0
                f = _normalize(np.clip(f + rng.normal(0, 0.04, n), 0, None))

    return best_f, best_sc


def _coord_polish(f0, n, stop_time, seed):
    rng = np.random.default_rng(seed)
    f = _normalize(np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None))
    best_sc = _score(f)
    no_imp = 0

    single_facs = (0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0, 0.15)
    pair_facs = ((0.5, 2.0), (2.0, 0.5), (0.3, 1.5), (1.5, 0.3),
                 (0.7, 1.4), (1.4, 0.7), (0.8, 1.25), (1.25, 0.8))
    fine_dts = (0.03, -0.03, 0.06, -0.06, 0.01, -0.01)

    while time.monotonic() < stop_time - 3.0:
        improved = False

        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 3.0:
                return f, best_sc
            orig = f[i]
            if orig < 1e-13:
                for v in (0.005, 0.02, 0.08):
                    f[i] = v
                    sc = _score(f)
                    if sc < best_sc - 1e-14:
                        best_sc = sc
                        improved = True
                f[i] = 0.0
                continue
            for mlt in single_facs:
                f[i] = orig * mlt
                sc = _score(f)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    improved = True
                    orig = f[i]
            f[i] = orig
            for dt in fine_dts:
                f[i] = max(0.0, orig * (1.0 + dt))
                sc = _score(f)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    improved = True
            f[i] = orig

        for _ in range(80):
            if time.monotonic() > stop_time - 3.0:
                return f, best_sc
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            oi, oj = f[i], f[j]
            for mi, mj in pair_facs:
                f[i] = oi * mi if oi > 1e-13 else 0.005
                f[j] = oj * mj if oj > 1e-13 else 0.005
                sc = _score(f)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    improved = True
            f[i], f[j] = oi, oj

        if improved:
            no_imp = 0
        else:
            no_imp += 1
            if no_imp >= 6:
                break

    return f, best_sc


def _peak_reduction(f0, n, stop_time, seed):
    rng = np.random.default_rng(seed)
    f = _normalize(np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None))
    best_f = f.copy()
    best_sc = _score(f)

    for it in range(50000):
        if time.monotonic() > stop_time - 2.0:
            break

        b = _conv_fft(f)
        k_peak = int(np.argmax(b))
        lo = max(0, k_peak - n + 1)
        hi = min(n, k_peak + 1)
        contrib = []
        for i in range(lo, hi):
            j = k_peak - i
            if 0 <= j < n:
                contrib.append((i, f[i] * f[j]))
        if not contrib:
            break
        contrib.sort(key=lambda x: -x[1])

        improved = False
        for i, _cv in contrib[:6]:
            orig = f[i]
            for mlt in (0.7, 0.8, 0.9, 0.5, 0.3):
                f[i] = orig * mlt
                sc = _score(f)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_f = f.copy()
                    improved = True
                    orig = f[i]
            f[i] = orig

        picks = rng.choice(n, size=min(5, n), replace=False)
        for j in picks:
            orig = f[j]
            for mlt in (1.2, 1.5, 2.0, 0.6):
                f[j] = orig * mlt if orig > 1e-13 else 0.01
                sc = _score(f)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_f = f.copy()
                    improved = True
                    orig = f[j]
            f[j] = orig

        if not improved:
            if it > 500:
                break

    return best_f, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 18.0

    try:
        h = globals()["height_sequence_1"]
        parent = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        parent = np.ones(320)
    if parent.size == 0:
        parent = np.ones(320)
    parent = np.clip(parent, 0.0, 1000.0)
    parent = np.where(np.isfinite(parent), parent, 0.0)
    if parent.sum() < 0.01:
        parent = np.ones(320)
    n = len(parent)

    best_seq = parent.copy()
    best_score = _score(best_seq)
    if not math.isfinite(best_score):
        best_seq = np.ones(n)
        best_score = _score(best_seq)

    def _update(cand):
        nonlocal best_seq, best_score
        if cand is None:
            return
        cand = np.clip(np.asarray(cand, dtype=np.float64).ravel(), 0, 1000)
        if cand.size == 0 or cand.sum() < 0.01:
            return
        try:
            sc = _score(cand)
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score:
            best_seq = cand.copy()
            best_score = sc

    lp_helper = _LPHelper(n)
    rng = np.random.default_rng(seed)

    # ---- Phase 1: LP-based FW (55% of time) ----
    lp_end = t0 + budget * 0.55
    i_arr = np.arange(n, dtype=np.float64)
    inits = [
        parent.copy(),
        0.5 + 0.5 * np.cos(2.0 * np.pi * i_arr / n + rng.uniform(0, 2 * np.pi)),
        np.abs(rng.normal(1.0, 0.6, n)) + 0.01,
        np.abs(parent + rng.normal(0, 0.3 * max(1.0, float(np.mean(parent))), n)) + 0.01,
    ]
    sup = np.full(n, 0.15)
    for fr in (1, 2, 3, 5, 8):
        sup += 0.25 * np.cos(2.0 * np.pi * fr * i_arr / n + rng.uniform(0, 2 * np.pi))
    inits.append(np.abs(sup) + 0.02)

    n_inits = len(inits)
    time_per_init = max(12.0, (lp_end - t0 - 5.0) / n_inits)
    for idx, init in enumerate(inits):
        if time.monotonic() > lp_end - 5.0:
            break
        if best_score < 1.503:
            break
        stop = min(lp_end - 3.0, time.monotonic() + time_per_init)
        try:
            res, _sc = _fw_loop(init, n, lp_helper, stop, seed + 1000 * (idx + 1))
            _update(res)
        except Exception:
            pass

    # ---- Phase 2: Subgradient polish (20%) ----
    sg_end = min(deadline, t0 + budget * 0.75)
    if time.monotonic() < sg_end - 15.0:
        try:
            res, _sc = _subgrad_polish(best_seq, n, sg_end, seed + 99001)
            _update(res)
        except Exception:
            pass

    # ---- Phase 3: Peak reduction + coordinate descent (15%) ----
    cd_end = min(deadline, t0 + budget * 0.90)
    if time.monotonic() < cd_end - 15.0:
        try:
            res, _sc = _peak_reduction(best_seq, n, min(cd_end - 5.0, time.monotonic() + 60.0), seed + 22001)
            _update(res)
        except Exception:
            pass
    if time.monotonic() < cd_end - 5.0:
        try:
            res, _sc = _coord_polish(best_seq, n, cd_end, seed + 77001)
            _update(res)
        except Exception:
            pass

    # ---- Phase 4: Final FW burst + subgrad if time remains ----
    if time.monotonic() < deadline - 12.0:
        try:
            res, _sc = _fw_loop(best_seq, n, lp_helper, deadline - 8.0, seed + 55001, max_iter=80)
            _update(res)
        except Exception:
            pass
    if time.monotonic() < deadline - 8.0:
        try:
            res, _sc = _subgrad_polish(best_seq, n, deadline - 6.0, seed + 33001)
            _update(res)
        except Exception:
            pass

    # ---- Final validation ----
    final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    try:
        sc = evaluate_sequence(final_list)
        if math.isfinite(sc) and sc > 0:
            return final_list
    except Exception:
        pass

    fb = [float(max(0.0, min(1000.0, x))) for x in parent]
    try:
        sc = evaluate_sequence(fb)
        if math.isfinite(sc) and sc > 0:
            return fb
    except Exception:
        pass

    return [1.0] * n
