def _get_parent():

def _conv(a):
    n = len(a)
    if n <= 256:
        return np.convolve(a, a, mode='full')
    fs = next_fast_len(2*n - 1)
    F = rfft(a, fs)
    c = irfft(F * F, fs)[:2*n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _sc(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return math.inf
    b = _conv(a)
    m = b.max()
    return 2.0 * n * m / (s * s)

def _build_A(f, n):
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = f
    return A

def _lp_dir(f, n, M):
    A = _build_A(f, n)
    try:
        r = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M),
                    bounds=[(0, None)]*n, method='highs',
                    options={'presolve': True, 'time_limit': 5})
    except:
        return None
    if r.status != 0:
        return None
    g = np.clip(r.x, 0, None)
    if g.sum() < 1e-15:
        return None
    return g

def _lp_loop(f0, n, stop, seed):

    f = np.clip(f0.copy(), 0, None)

        if gs < 1e-15:

            stall += 2
