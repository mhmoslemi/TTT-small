  ```python
      for i, t in enumerate(log_T):
          tau = np.exp(t)
          # Sample new point from Gaussian perturbed by temperature
          perturbation = np.random.normal(size=n) * (tau**0.5)
          candidate = initial_sequence * (1 + perturbation)
          candidate = np.clip(candidate, 0, 1000)
          
          # Repair sum constraint if needed
          if np.sum(candidate) < 0.01:
              candidate *= max(0.01, np.sum(candidate) * 0.2)  # Scale to meet minimum sum requirement
          
          # Optimization
          fast_coef = optimize_coordinate_step(candidate)  # See optimization details below
          final_point = candidate * (0.8 + fast_coef)
          
          current_score = evaluate_sequence(final_point)
          if current_score < best_score:
              best_score = current_score
              best_sequence = final_point.copy()
              print(f"Improvement found: {best_score}")
          else:
              # May still accept occasional uphill moves
              if np.random.rand() < (Tau_independence / (1 + i/Tau_max)):
                  best_sequence = final_point
                  best_score = current_score
                    
          # Store progress
          history['score'].append(best_score)
          history['time_series'] = final_point.tolist()
          
      return best_sequence, best_score
  ```
