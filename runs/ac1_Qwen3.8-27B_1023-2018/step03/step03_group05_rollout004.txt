def _get_parent():

def _normalize(a, target=1.0):
    arr = np.asarray(a, dtype=np.float64).ravel()
    if arr.size == 0:
        arr = np.ones(1, dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    s = float(arr.sum())
    if s > 1e-300:
        return (arr / s) * target
    return np.full(arr.size, target / arr.size, dtype=np.float64)


def _conv_direct(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([0.0], dtype=np.float64)
    c = np.convolve(a, a, mode="full")
    c = np.asarray(c, dtype=np.float64)
    c = np.where(np.isfinite(c), c, 0.0)
    np.maximum(c, 0.0, out=c)
    return c


def _exact_score(a):
    try:
        a = np.asarray(a, dtype=np.float64).ravel()
        if a.size == 0:
            return float("inf")
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.maximum(a, 0.0)
        s = float(a.sum())
        if s <= 0.0:
            return float("inf")
        m = float(_conv_direct(a).max())
        if not np.isfinite(m):
            return float("inf")
        return float(2.0 * a.size * m / (s * s))
    except Exception:
        return float("inf")


def _resample(a, m):
    a = _normalize(a)
    n = int(a.size)
    m = int(m)
    if m <= 0:
        return np.ones(1, dtype=np.float64)
    if n <= 0:
        return np.ones(m, dtype=np.float64) / float(m)
    if n == m:
        return a.copy()

    if m > n:
        idx = (np.arange(m, dtype=np.int64) * n) // m
        counts = np.bincount(idx, minlength=n).astype(np.float64)
        counts[counts <= 0] = 1.0
        b = a[idx] / counts[idx]
    else:
        b = np.zeros(m, dtype=np.float64)
        idx = (np.arange(n, dtype=np.int64) * m) // n
        np.add.at(b, idx, a)
        b = np.maximum(b, 0.0)
    return _normalize(b)


def _smooth(a, alpha=0.3):
    a = _normalize(a)
    n = a.size
    if n < 3 or alpha <= 0.0:
        return a
    v = np.empty_like(a)
    v[1:-1] = 0.25 * a[:-2] + 0.5 * a[1:-1] + 0.25 * a[2:]
    v[0] = 0.5 * a[0] + 0.5 * a[1]
    v[-1] = 0.5 * a[-1] + 0.5 * a[-2]
    return _normalize((1.0 - alpha) * a + alpha * v)


def _random_smooth(n, rng, base=None):
    n = int(n)
    if n <= 0:
        return np.ones(1, dtype=np.float64)

                if val > threshold:
                    continue

            tb = float(np.clip(tb, 0.0, 1.0 - 1e-12))
            factor = (1.0 - tb) * inv_s

            a *= factor
            a[i] = tb

            p = (1.0 - tb) * (1.0 - tb)
            q = tb * (1.0 - tb)
            np.multiply(C0, p, out=newc)
            np.multiply(cross, q, out=C0)
            newc += C0
            newc[two_i] += tb * tb
            c[:] = newc
            np.maximum(c, 0.0, out=c)

            improved = True
            coords += 1

            if coords % 16 == 0:
                ss = float(a.sum())
                if ss > 0.0 and abs(ss - 1.0) > 1e-8:
                    a = a / ss
                    c = _conv_direct(a)

        if coords > 0 or sweep % 3 == 0:
            ss = float(a.sum())
            if ss > 0.0 and abs(ss - 1.0) > 1e-10:
                a = a / ss
            c = _conv_direct(a)
            mval = float(c.max()) if c.size else 0.0
            if np.isfinite(mval) and mval < best_M - max(1e-15, 1e-12 * best_M):
                best_M = mval
                best_a = a.copy()
                best_sc = 2.0 * n * best_M

        if improved:
            no_imp = 0
        else:
            no_imp += 1

        if no_imp > 0 and no_imp in (12, 25, 50, 100, 200, 400):
            sigma = min(0.22, 0.008 + 0.010 * (no_imp / 10.0))
            a = _normalize(a * np.exp(rng.normal(0.0, sigma, n)))
            r = float(rng.random())
            if r < 0.35:
                alt = _anti_start(n, rng)
                a = _normalize(0.70 * a + 0.30 * alt)
            elif r < 0.70:
                alt = _random_smooth(n, rng, base=a)
                a = _normalize(0.75 * a + 0.25 * alt)
            else:
                a = _smooth(a, 0.35)

            c = _conv_direct(a)
            mval = float(c.max()) if c.size else 0.0
            if np.isfinite(mval) and mval < best_M:
                best_M = mval
                best_a = a.copy()
                best_sc = 2.0 * n * best_M
            no_imp = 0

        if no_imp > 90:
            curM = float(c.max()) if c.size else 0.0
            if np.isfinite(curM) and curM > best_M * 1.0005:
                a = best_a.copy()
                c = _conv_direct(a)
                no_imp = 0

        if not np.isfinite(best_sc):
            break

    best_a = _normalize(best_a)
    best_sc = _exact_score(best_a)
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = max(0.0, float(budget_s))
    except Exception:
        budget = 1300.0

    try:
        seed_int = int(seed)
        if not np.isfinite(float(seed_int)):
            seed_int = 42
    except Exception:
        seed_int = 42

    if budget <= 2.0:
        return [1.0]

    head = max(8.0, min(55.0, 0.04 * max(budget, 1.0)))
    deadline = t0 + budget - head
    if deadline <= t0 + 5.0:
        deadline = t0 + max(3.0, 0.9 * budget)

    rng = np.random.default_rng(seed_int)

    parent = _get_parent()
    best_a = _normalize(parent)
    best_score = _exact_score(best_a)

    if not np.isfinite(best_score):
        best_a = np.ones(512, dtype=np.float64) / 512.0
        best_score = _exact_score(best_a)

    if not np.isfinite(best_score):
        return [1.0]

    rem0 = max(0.0, deadline - t0)
    if rem0 <= 90.0:
        search_end = deadline - max(3.0, 0.05 * rem0)
    else:
        reserve = min(300.0, max(60.0, 0.2 * rem0))
        search_end = deadline - reserve

    if search_end <= t0 + 15.0:
        search_end = t0 + max(10.0, 0.5 * rem0)

    parent_n = int(best_a.size)
    if parent_n < 16 or parent_n > 4096:
        parent_n = 512

    tasks = []
    seen = set()

    def _add_side(L, kind, w):
        L = int(L)
        if not (16 <= L <= 4096) or w <= 0.0:
            return
        key = (L, kind)
        if key in seen:
            return
        seen.add(key)
        tasks.append((L, kind, float(w)))

    _add_side(parent_n, "parent", 3.6)
    _add_side(512, "parent", 3.2)
    _add_side(1024, "parent", 2.3)
    _add_side(768, "parent", 1.1)
    _add_side(1280, "parent", 0.8)
    _add_side(1536, "parent", 0.6)
    _add_side(512, "anti", 0.9)
    _add_side(1024, "anti", 0.6)
    _add_side(768, "smooth", 0.5)

    tasks.sort(key=lambda x: -x[2])
    total_w = float(sum(w for _, _, w in tasks))
    phase_start = time.monotonic()
    phase_total = max(0.0, search_end - phase_start)
    phase_cursor = phase_start

    for L, kind, w in tasks:
        if phase_total <= 0.0:
            break
        if time.monotonic() >= search_end - 5.0:
            break

        alloc = (w / total_w) * phase_total
        task_end = phase_cursor + alloc
        if task_end > search_end - 5.0:
            task_end = search_end - 5.0
        phase_cursor = max(phase_cursor + alloc, task_end)

        if task_end - time.monotonic() < 10.0:
            continue

        try:
            base = _resample(parent, L)
            if kind == "parent":
                init = base
            elif kind == "anti":
                alt = _anti_start(L, rng)
                init = _normalize(0.62 * base + 0.38 * alt)
            else:
                alt = _random_smooth(L, rng, base=base)
                init = _normalize(0.75 * base + 0.25 * alt)

            cand, sc = _cd_refine(_normalize(init), task_end, rng)
            sc = _exact_score(cand)
            if np.isfinite(sc) and sc < best_score:
                best_a = _normalize(cand)
                best_score = float(sc)
        except Exception:
            pass

    if deadline - time.monotonic() > 12.0:
        try:
            cand, sc = _cd_refine(_normalize(best_a), deadline, rng)
            sc = _exact_score(cand)
            if np.isfinite(sc) and sc < best_score:
                best_a = _normalize(cand)
                best_score = float(sc)
        except Exception:
            pass

    best_a = _normalize(best_a)
    best_a = np.clip(best_a, 0.0, 1000.0).astype(np.float64)
    if best_a.size == 0:
        best_a = np.ones(1, dtype=np.float64)

    out = [float(x) for x in best_a]

    try:
        ev = globals().get("evaluate_sequence", None)
        if ev is not None:
            sc = ev(out)
            if np.isfinite(sc):
                return out
    except Exception:
        pass

    sc = _exact_score(out)
    if not np.isfinite(sc):
        fb = np.ones(32, dtype=np.float64) / 32.0
        return [float(x) for x in fb]

    return out
