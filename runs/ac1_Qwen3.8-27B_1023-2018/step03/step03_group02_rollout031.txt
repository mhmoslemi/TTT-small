                    bvs = sc
                    best_val = nv
            best[i] = best_val
            if bvs < best_sc - 1e-15:
                best_sc = bvs
                improved = True

        # Pair moves
        n_pairs = min(60, n)
        for _ in range(n_pairs):
            if time.monotonic() > stop_time - 1.0:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            for ratio in (0.2, 0.4, 0.6, 0.8, 1.2, 1.5, 0.0, 1.0):
                ni = max(0.0, oi * ratio)
                nj = max(0.0, oj + oi * (1.0 - ratio))
                best[i], best[j] = ni, nj
                if best.sum() < 0.01:
                    break
                sc = _score(best)
                if sc < best_sc - 1e-15:
                    best_sc = sc
                    improved = True
            best[i], best[j] = oi, oj

        # Triple moves (fewer)
        for _ in range(20):
            if time.monotonic() > stop_time - 1.0:
                break
            idx = rng.choice(n, size=3, replace=False)
            i, j, k = int(idx[0]), int(idx[1]), int(idx[2])
            oi, oj, ok = best[i], best[j], best[k]
            total = oi + oj + ok
            if total < 1e-12:
                continue
            for ri in (0.3, 0.6, 1.0, 1.5):
                for rj in (0.4, 0.6, 0.8, 1.2):
                    ni = ri * oi
                    nj = rj * oj
                    nk = total - ni - nj
                    if ni < 0 or nj < 0 or nk < 0:
                        continue
                    best[i], best[j], best[k] = ni, nj, nk
                    sc = _score(best)
                    if sc < best_sc - 1e-15:
                        best_sc = sc
                        improved = True
            best[i], best[j], best[k] = oi, oj, ok

        if not improved:
            no_imp += 1
            if no_imp >= max_no_imp:
                break
        else:
            no_imp = 0

    return best, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(20.0, float(budget_s))
    deadline = t0 + budget - 6.0

    parent = _get_parent()
    n_p = len(parent)

    best_seq = parent.copy()
    best_score = math.inf
    try:
        best_score = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        pass
    if not math.isfinite(best_score):
        best_seq = np.ones(n_p) / n_p
        best_score = _score(best_seq)

    def _try_update(a):
        nonlocal best_seq, best_score
        a_c = np.clip(np.asarray(a, dtype=np.float64), 0, 1000)
        if a_c.size == 0 or a_c.sum() < 0.01:
            return
        try:
            sc = float(evaluate_sequence([float(x) for x in a_c]))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score:
            best_seq = a_c.copy()
            best_score = sc

    n = n_p
    rng_main = np.random.default_rng(seed)

    # Phase 1: Main LP optimization (50% of time)
    phase1_end = t0 + budget * 0.50
    # First pass on parent
    stop1 = min(phase1_end - 3.0, t0 + budget * 0.22)
    try:
        res, sc = _lp_optimize(parent, n, stop1, seed + 7)
        _try_update(res)
    except Exception:
        pass

    # Second pass continuing from current best
    stop2 = min(phase1_end - 2.0, time.monotonic() + (phase1_end - time.monotonic()) * 0.8)
    if time.monotonic() < stop2 - 5.0:
        try:
            res, sc = _lp_optimize(best_seq, n, stop2, seed + 13)
            _try_update(res)
        except Exception:
            pass

    # Phase 2: Subgradient refinement (10%)
    phase2_end = t0 + budget * 0.62
    if time.monotonic() < phase2_end - 5.0:
        stop2b = min(phase2_end - 2.0, time.monotonic() + (phase2_end - time.monotonic()) * 0.85)
        try:
            res, sc = _subgrad_optimize(best_seq, n, stop2b, seed + 23)
            _try_update(res)
        except Exception:
            pass

    # Phase 3: Try nearby lengths (15%)
    phase3_end = t0 + budget * 0.77
    lengths_to_try = []
    for delta in [-3, -2, -1, 1, 2, 3, -5, 5]:
        lengths_to_try.append(n + delta)
    # Also try a couple of very different lengths
    lengths_to_try.extend([int(n * 0.75), int(n * 1.25)])
    lengths_to_try = sorted(set(L for L in lengths_to_try if 50 <= L <= 800))

    if lengths_to_try:
        time_per_len = max(8.0, (phase3_end - time.monotonic()) / len(lengths_to_try))
        for L in lengths_to_try:
            if time.monotonic() > phase3_end - 4.0:
                break
            if time.monotonic() >= phase3_end:
                break
            stop_l = min(phase3_end - 2.0, time.monotonic() + time_per_len)
            # Interpolate parent to new length
            init_l = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, n_p), parent)
            try:
                res, sc = _lp_optimize(init_l, L, stop_l, seed + 200 + L)
                _try_update(res)
            except Exception:
                pass

    # Phase 4: Refined LP on best length (10%)
    phase4_end = t0 + budget * 0.87
    if time.monotonic() < phase4_end - 5.0:
        n_best = len(best_seq)
        stop4 = min(phase4_end - 2.0, time.monotonic() + (phase4_end - time.monotonic()) * 0.85)
        if time.monotonic() < stop4 - 5.0:
            try:
                res, sc = _lp_optimize(best_seq, n_best, stop4, seed + 500)
                _try_update(res)
            except Exception:
                pass

    # Phase 5: Coordinate polish (rest of time)
    polish_start = time.monotonic()
    polish_end = deadline
    if polish_end - polish_start > 15.0:
        n_polish = min(5, max(2, int((polish_end - polish_start) / 30.0)))
        t_per = (polish_end - polish_start - 3.0) / n_polish
        for p in range(n_polish):
            stop_p = time.monotonic() + t_per
            if stop_p > polish_end:
                stop_p = polish_end
            if time.monotonic() >= stop_p - 2.0:
                break
            try:
                res, sc = _coord_polish(best_seq, stop_p, seed + 9000 + p * 17)
                _try_update(res)
            except Exception:
                pass
    elif polish_end - time.monotonic() > 8.0:
        try:
            res, sc = _coord_polish(best_seq, polish_end - 1.0, seed + 9000)
            _try_update(res)
        except Exception:
            pass

    # Final: one short subgradient pass if time allows
    if time.monotonic() < deadline - 12.0:
        try:
            res, sc = _subgrad_optimize(best_seq, len(best_seq), deadline - 2.0, seed + 99999)
            _try_update(res)
        except Exception:
            pass

    # Final validation
    final = np.clip(best_seq, 0, 1000)
    try:
        final_sc = float(evaluate_sequence([float(x) for x in final]))
    except Exception:
        final_sc = math.inf
    if not math.isfinite(final_sc) or final_sc >= best_score:
        final_sc = best_score

    return [float(x) for x in final]
