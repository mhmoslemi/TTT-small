def _gauss_kernel(width=15, sigma=None):
    width = int(max(1, width))

def _set_nearest_key(n, S, m):

            if d:
                counts[d] += 1
    R = int(counts.max()) if counts.size else 1
    if R <= 0:

    return float(R) / float(m * m)


def _sidon_search(n, time_limit, rng):
    if n < 3 or time_limit <= 0.0:
        return []
    start = time.monotonic()
    sets = []
    seen = set()

    def add_set(S):
        if len(S) < 5:
            return
        S = tuple(sorted(set(int(x) for x in S)))
        if S in seen:
            return
        seen.add(S)
        m = len(S)
        if m > n:
            return
        key = _set_nearest_key(n, list(S), m)
        sets.append((key, list(S), m))
        sets.sort(key=lambda x: x[0])
        if len(sets) > 50:
            del sets[50:]

    max_restarts = int(min(2500, max(150, time_limit * 25)))
    restart = 0
    while restart < max_restarts and time.monotonic() - start < 0.55 * time_limit:
        rlim = (1, 2, 3)[restart % 3]
        S = [0]
        counts = np.zeros(n - 1, dtype=np.int32)
        cand = list(range(1, n))
        rng.shuffle(cand)
        for x in cand:
            if time.monotonic() - start > 0.7 * time_limit:
                break
            maxc = 0
            for y in S:
                d = abs(x - y)
                if d:
                    v = int(counts[d]) + 1
                    if v > maxc:
                        maxc = v
            if maxc <= rlim:
                for y in S:
                    d = abs(x - y)
                    if d:
                        counts[d] += 1
                S.append(x)
        add_set(S)
        restart += 1

    if sets and time.monotonic() - start < 0.9 * time_limit:
        moves_per_set = int(min(20000, max(1500, time_limit * 200)))
        for _, S0, m in [(t[0], t[1], t[2]) for t in sets[:8]]:
            if time.monotonic() - start > time_limit - 3.0:
                break
            S = [int(x) for x in S0]
            if len(S) != m or len(set(S)) != m:
                continue
            inS = np.zeros(n, dtype=bool)
            for x in S:
                if 0 <= x < n:
                    inS[x] = True
            counts = np.zeros(n - 1, dtype=np.int32)
            for i in range(m):
                si = S[i]
                for j in range(i + 1, m):
                    d = abs(si - S[j])
                    if d:
                        counts[d] += 1
            best_key = float(counts.max()) / float(m * m)
            if best_key <= 0:
                best_key = 1.0

            for _ in range(moves_per_set):
                if time.monotonic() - start > time_limit - 2.0:
                    break
                out = int(rng.integers(m))
                old = S[out]
                while True:
                    new = int(rng.integers(n))
                    if not inS[new]:
                        break

                counts2 = counts.copy()
                for y in S:
                    if y == old:
                        continue
                    d = abs(old - y)
                    if d:
                        counts2[d] -= 1

                S[out] = new
                inS[old] = False
                inS[new] = True

                for y in S:
                    if y == new:
                        continue
                    d = abs(new - y)
                    if d:
                        counts2[d] += 1

                R2 = int(counts2.max())
                if R2 <= 0:
                    R2 = 1
                key2 = float(R2) / float(m * m)

                if key2 < best_key - 1e-13:
                    counts = counts2
                    best_key = key2
                    add_set(S)
                else:
                    S[out] = old
                    inS[old] = True
                    inS[new] = False
    return sets

def _active_order(a):
    n = len(a)
    try:
        c = np.convolve(a, a, mode="full")
        center = n - 1
        c0 = float(c[center]) if center < c.size else 0.0
        offmax = 0.0
        if center > 0:
            offmax = max(offmax, float(np.max(c[:center])))
        if center + 1 < c.size:
            offmax = max(offmax, float(np.max(c[center + 1:])))
        vals = c.copy()
        if offmax > 1e-12 * max(1.0, c0):
            vals[center] = 0.0
            top = np.argsort(vals)[-min(32, max(1, c.size - 1)):][::-1]
        else:
            top = np.array([center], dtype=int)
        grad = np.zeros(n, dtype=np.float64)
        for idx in top:
            idx = int(idx)
            d = abs(idx - center)
            if d >= n:
                continue
            if d == 0:
                grad += 2.0 * a
            else:
                grad[: n - d] += a[d:]
                grad[d:] += a[: n - d]
        order = np.argsort(-grad)
        return order
    except Exception:
        return np.arange(n)


def _coord_sweep(a, cur_sc, stop, rng):

        j = int(idx)
        old = float(a[j])

        if old > 1e-6:

                v = 0.0

        a[j] = local_best_val

    s = float(a.sum())
    if s > 1e-12:

    else:
        a = np.ones(n, dtype=np.float64) / float(n)

def _block_moves(a, cur_sc, stop, rng):
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    n = a.size
    if n == 0:
        return a, _INF, False
    if not math.isfinite(cur_sc):
        cur_sc = _exact_score(a)
    if not math.isfinite(cur_sc):
        return a, cur_sc, False

    best_sc = cur_sc
    improved = False
    bsz = max(1, n // 14)
    factors = (0.0, 0.25, 0.5, 0.75, 1.25, 1.6, 2.2)

    for start in range(0, n, bsz):
        if time.monotonic() > stop:
            break
        end = min(n, start + bsz)
        old_block = a[start:end].copy()
        best_f = 1.0
        local_best = best_sc

        for f in factors:
            if time.monotonic() > stop:
                break
            a[start:end] = old_block * f
            if float(a.sum()) <= 1e-12:
                continue
            sc = _exact_score(a)
            if math.isfinite(sc) and sc < local_best - 1e-13:
                local_best = sc
                best_f = f

        a[start:end] = old_block * best_f
        if local_best < best_sc - 1e-13:
            best_sc = local_best
            improved = True

    s = float(a.sum())
    if s > 1e-12:
        a = a / s
    else:
        a = np.ones(n, dtype=np.float64) / float(n)
        best_sc = _exact_score(a)
        improved = False

    return a, best_sc, improved


def _build_conv_matrix(f, n):
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_direction(a, n, mode, stop, time_limit):

        return None
    if time.monotonic() > stop - max(3.0, time_limit * 0.5):
        return None

    try:
        conv = np.convolve(a, a, mode="full")
        center = n - 1
        U = 100.0
        conv_max = float(np.max(conv))
        dot = float(np.dot(a, a))

        if mode == 0:
            rows = np.arange(2 * n - 1, dtype=int)
            M = max(conv_max, dot)

            if allrows.size == 0:
                return None
            order = np.argsort(conv[allrows])
            rows = allrows[order][-min(120, allrows.size):]
            offmax = float(np.max(conv[allrows]))
            if offmax <= 1e-14:
                return None
            M = 0.78 * offmax
        else:
            rows = np.arange(2 * n - 1, dtype=int)
            M = 0.9 * max(conv_max, dot)

        if not math.isfinite(M) or M <= 0.0:
            return None

        A = _build_conv_matrix(a, n)
        A = A[rows, :]
        cvec = -np.ones(n, dtype=np.float64)
        b_ub = np.full(A.shape[0], float(M), dtype=np.float64)
        bounds = [(0.0, U)] * n

        tl = float(max(1.0, min(20.0, time_limit)))
        res = linprog(
            cvec,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True, "time_limit": tl},
        )
        if res is None or getattr(res, "x", None) is None:
            return None
        if res.status not in (0, 3):
            return None
        g = np.asarray(res.x, dtype=np.float64)
        g = np.clip(g, 0.0, None)
        s = float(g.sum())
        if s <= 1e-12:
            return None
        return g / s
    except Exception:
        return None


def _line_search_lp(f, g, cur_sc, stop):
    if not math.isfinite(cur_sc):
        return None, _INF
    best_a = None
    best_sc = cur_sc
    for t in (0.125, 0.25, 0.5, 0.75, 1.0):
        if time.monotonic() > stop:
            break
        a = (1.0 - t) * f + t * g
        a = np.clip(a, 0.0, 1000.0)
        if float(a.sum()) <= 1e-12:
            continue
        sc = _exact_score(a)
        if math.isfinite(sc) and sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a.copy()
    if best_a is None:
        return None, cur_sc
    best_a = _normalize(best_a)
    return best_a, best_sc


def _perturb(a, rng, parent, strength=0.15):
    n = len(a)
    if n <= 0:
        return np.ones(1, dtype=np.float64)
    b = np.asarray(a, dtype=np.float64).copy()
    wd = min(15, max(3, n))
    kern = _gauss_kernel(wd)
    noise = np.convolve(rng.standard_normal(n), kern, mode="same")
    b *= 1.0 + float(strength) * noise
    b = np.clip(b, 0.0, None)

    if parent is not None and rng.random() < 0.55:
        alpha = rng.uniform(0.08, 0.35)
        pr = np.asarray(parent, dtype=np.float64).ravel()
        if pr.size != n:
            if pr.size > 1:
                pr = np.interp(
                    np.linspace(0.0, 1.0, n),
                    np.linspace(0.0, 1.0, pr.size),
                    pr,
                )
            else:
                pr = np.full(n, float(pr.sum()))
        pr = _normalize(pr)
        b = (1.0 - alpha) * b + alpha * pr

    if rng.random() < 0.25:
        pos = int(rng.integers(n))
        b[pos] += 0.08 * float(b.sum())

    b = np.clip(b, 0.0, None)
    return _normalize(b)


def _local_search(a, stop, rng, parent):
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = np.clip(a, 0.0, 1000.0)
    a = _normalize(a)
    n = a.size
    if n == 0:
        return a, _INF

    cur = a.copy()
    cur_sc = _exact_score(cur)
    if not math.isfinite(cur_sc):
        cur = np.ones(n, dtype=np.float64) / float(n)
        cur_sc = _exact_score(cur)
    best = cur.copy()
    best_sc = cur_sc
    if not math.isfinite(best_sc):
        return best, best_sc

    it = 0
    mode = 0
    stall = 0

    while time.monotonic() < stop - 3.0:
        if best_sc <= 1e-12:
            break

        improved = False

        if linprog is not None and it % 4 == 0 and time.monotonic() < stop - 25.0:
            remaining = max(1.0, stop - time.monotonic())
            tl = max(2.0, min(12.0, remaining * 0.08))
            g = _lp_direction(cur, n, mode % 4, stop - 15.0, tl)
            mode += 1
            if g is not None:
                na, nsc = _line_search_lp(cur, g, cur_sc, stop - 5.0)
                if na is not None:
                    cur = na
                    cur_sc = nsc
                    improved = True

        if time.monotonic() < stop - 3.0:
            na, nsc, imp = _coord_sweep(cur, cur_sc, stop - 2.0, rng)
            cur = na
            cur_sc = nsc
            if imp:
                improved = True

        if time.monotonic() < stop - 3.0:
            na, nsc, imp = _block_moves(cur, cur_sc, stop - 2.0, rng)
            cur = na
            cur_sc = nsc
            if imp:
                improved = True

        if math.isfinite(cur_sc) and cur_sc < best_sc - 1e-12:
            best = cur.copy()
            best_sc = cur_sc

        if not improved:
            stall += 1
            if stall >= 5:
                cand = _perturb(cur, rng, parent, strength=0.10 + 0.04 * min(stall, 10))
                csc = _exact_score(cand)
                if math.isfinite(csc) and csc < max(best_sc, cur_sc) * 1.25 + 1e-9:
                    cur = cand
                    cur_sc = csc
                else:
                    cur = best.copy()
                    cur_sc = best_sc
                stall = 0

        it += 1

    return best, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(1.0, float(budget_s))
    headroom = min(25.0, max(1.0, 0.03 * budget))
    hard = t0 + budget - headroom

    parent_raw = _get_parent_raw()
    if parent_raw.size == 0 or parent_raw.size < 2:
        parent_raw = np.ones(384, dtype=np.float64)
    elif parent_raw.size > 2048:
        parent_raw = np.interp(
            np.linspace(0.0, 1.0, 384),
            np.linspace(0.0, 1.0, parent_raw.size),
            parent_raw,
        )
    n = int(parent_raw.size)
    rng = np.random.default_rng(seed)

    uniform = np.ones(n, dtype=np.float64) / float(n)
    parent = _normalize(parent_raw)

    best_seq = uniform.copy()
    best_score = _exact_score(best_seq)
    if not math.isfinite(best_score):
        return [1.0]

    quick = [uniform, parent]
    if n >= 3:
        for pos in (0, n // 3, 2 * n // 3, n - 1):
            e = np.zeros(n, dtype=np.float64)
            e[pos] = 1.0
            quick.append(e)
    for a in quick:
        sc = _exact_score(a)
        if math.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
    if best_score <= 1e-12:
        return _to_list(best_seq)

    diag_stop = min(t0 + 18.0, hard - 8.0)
    if diag_stop > t0 + 3.0 and _is_full_evaluator(n, parent_raw, rng, diag_stop):
        final_c = [uniform, parent, best_seq]
        for pos in (0, n // 2, n - 1) if n >= 1 else ():
            e = np.zeros(n, dtype=np.float64)
            e[pos] = 1.0
            final_c.append(e)
        chosen = uniform
        chosen_sc = _INF
        for a in final_c:
            sc = _exact_score(a)
            if math.isfinite(sc) and sc < chosen_sc:
                chosen_sc = sc
                chosen = a.copy()
        return _to_list(chosen)

    support_sets = []
    if time.monotonic() < hard - 120.0:
        support_time = min(65.0, max(0.0, (hard - time.monotonic()) * 0.08))
        if support_time > 5.0:
            try:
                support_sets = _sidon_search(n, support_time, rng)
            except Exception:
                support_sets = []

    cands = []
    try:
        cands = _initial_candidates(n, parent_raw, rng, support_sets)
    except Exception:
        cands = [uniform, parent]

    scored = []
    pool_stop = max(t0 + 1.0, min(t0 + 90.0, hard - 110.0))
    for a in cands:
        if time.monotonic() > pool_stop:
            break
        try:
            arr = np.asarray(a, dtype=np.float64).ravel()
            arr = np.clip(arr, 0.0, 1000.0)
            arr = _normalize(arr)
            if arr.size != n or arr.sum() <= 1e-14:
                continue
            sc = _exact_score(arr)
            if math.isfinite(sc):
                scored.append((sc, arr.copy()))
        except Exception:
            continue

    scored.sort(key=lambda x: x[0])
    for sc, arr in scored:
        if sc < best_score:
            best_score = sc
            best_seq = arr.copy()
    if best_score <= 1e-12:
        return _to_list(best_seq)

    seed_entries = []

    def _add_seed(a, sc=None, tol=0.12):
        try:
            an = _normalize(np.asarray(a, dtype=np.float64).ravel())
        except Exception:
            return
        if an.size != n or an.sum() <= 1e-14:
            return
        if sc is None:
            sc = _exact_score(an)
        if not math.isfinite(sc):
            return
        for idx, (old_sc, old_a) in enumerate(seed_entries):
            if np.linalg.norm(an - old_a) < tol:
                if sc < old_sc - 1e-12:
                    seed_entries[idx] = (sc, an.copy())
                return
        seed_entries.append((sc, an.copy()))

    _add_seed(best_seq, best_score)
    _add_seed(parent)
    for sc, a in scored[:12]:
        _add_seed(a, sc)
    if not seed_entries:
        seed_entries = [(best_score, best_seq.copy())]

    seed_entries.sort(key=lambda x: x[0])

    search_deadline = max(time.monotonic() + 1.0, hard - 45.0)
    fracs = [0.34, 0.18, 0.14, 0.10, 0.08, 0.07, 0.06, 0.03]
    idx = 0
    while idx < len(seed_entries) and time.monotonic() < search_deadline - 25.0:
        remaining = max(0.0, search_deadline - time.monotonic())
        if remaining < 35.0:
            break
        if idx < len(fracs):
            dur = remaining * fracs[idx]
        else:
            dur = remaining * 0.025
        dur = max(18.0, min(420.0, dur))
        stop = min(search_deadline - 12.0, time.monotonic() + dur)
        if stop - time.monotonic() < 12.0:
            break

        sc0, seq = seed_entries[idx]
        try:
            lseq, lsc = _local_search(seq, stop, rng, parent)
        except Exception:
            lseq, lsc = None, _INF
        if lseq is not None and math.isfinite(lsc) and lsc < best_score:
            best_score = lsc
            best_seq = lseq.copy()
        if best_score <= 1e-12:
            break
        idx += 1

    final_local_deadline = hard - 18.0
    while time.monotonic() < final_local_deadline - 30.0 and best_score > 1e-9:
        remaining = max(0.0, final_local_deadline - time.monotonic())
        if remaining < 70.0:
            break
        stop = min(final_local_deadline - 12.0, time.monotonic() + min(120.0, remaining * 0.45))
        if stop - time.monotonic() < 25.0:
            break
        try:
            lseq, lsc = _local_search(best_seq, stop, rng, parent)
        except Exception:
            break
        if lseq is not None and math.isfinite(lsc) and lsc < best_score:
            best_score = lsc
            best_seq = lseq.copy()
        else:
            break

    final_cands = [best_seq, uniform, parent]
    if n >= 1:
        for pos in (0, n // 2, n - 1):
            e = np.zeros(n, dtype=np.float64)
            e[pos] = 1.0
            final_cands.append(e)
    for _, a in scored[:3]:
        final_cands.append(a)

    final_seq = best_seq
    final_sc = _INF
    for a in final_cands:
        try:
            arr = _normalize(np.asarray(a, dtype=np.float64).ravel())
            if arr.size != n or arr.sum() <= 1e-14:
                continue
            sc = _exact_score(arr)
            if math.isfinite(sc) and sc < final_sc:
                final_sc = sc
                final_seq = arr.copy()
        except Exception:
            continue

    if not math.isfinite(final_sc) or final_seq is None or len(final_seq) == 0:
        return [1.0]

    return _to_list(final_seq)
