This gives a natural gradient in log-space. I could use this with Adam or L-BFGS.

But the non-smoothness of the max is still an issue. I'll handle it by:
- Tracking the top few active lags
- Using the max over them for the gradient
- Or using a "smooth max" approximation: `smooth_max(b) = log(sum(exp(b/τ)))*τ` for large τ

Actually, for the gradient polishing phase, I'll use:
1. If a single lag is clearly the max (gap > ε), use its gradient
2. If multiple lags are close, use the max over all of them (subgradient)

```python
def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = np.convolve(a, a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf

def _conv(a):
    n = len(a)
    if n <= 1024:
        return np.convolve(a, a)

    F = rfft(a, fsize)
    c = irfft(F*F, fsize)[:2*n-1]
    np.maximum(c, 0, out=c)
    return c
```

For the LP direction:
```python
def _lp_direction(a, M):
    n = len(a)
    # Build constraint matrix A where (A@g)[k] = (a*g)[k]
    # A[k,j] = a[j] if 0 <= k-j < n, else 0
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = a
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)]*n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': 5, 'presolve': True})
    if res.status != 0 or res.x is None:
        return None
    g = np.clip(res.x, 0, 1000)
    if g.sum() < 1e-15:
        return None
    return g
```

```python
def _grad_optimize(a0, n, stop_time, seed):
    rng = np.random.default_rng(seed)
    a = np.clip(a0.astype(np.float64), 0, 1000).copy()
    best = a.copy()
    best_score = _score(a)
    
    lr = 0.1  # learning rate
    no_improve = 0
    
    while time.monotonic() < stop_time - 5:
        b = _conv(a)
        M = float(b.max())
        S = float(a.sum())
        k_star = int(b.argmax())
        
        # Compute gradient of score wrt a (in the region where k* is active)
        # d(score)/da[j] = 4n*(a[k*-j]*S - M) / S^3 if 0 <= k*-j < n
        #               = -4n*M / S^3 otherwise
        
        grad = -4.0 * n * M / (S**3) * np.ones(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        for j in range(lo, hi + 1):
            grad[j] = 4.0 * n * (a[k_star - j] * S - M) / (S**3)
        
        # Gradient step (descent)
        a_new = a - lr * grad
        a_new = np.clip(a_new, 0, 1000)
        
        if a_new.sum() < 0.01:
            lr *= 0.5
            continue
        
        sc = _score(a_new)
        if sc < best_score - 1e-14:
            a = a_new
            best = a.copy()
            best_score = sc
            no_improve = 0
            lr = min(1.0, lr * 1.05)
        else:
            lr = max(1e-6, lr * 0.8)

                a = np.clip(best + rng.normal(0, 0.01 * best.mean(), n), 0, None)

                lr = 0.05
```
