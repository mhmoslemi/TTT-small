                        oi, oj = f[i], f[j]
                    else:
                        f[i], f[j] = oi, oj
                        f = _normalize(f)

        if not improved:
            no_sweep_imp += 1
            if no_sweep_imp >= 8:
                break
        else:
            no_sweep_imp = 0

    return best_f, best_sc

def _mirror_descent(f0, n, stop_time, max_iter=30000):
    f = _normalize(np.maximum(f0, 0))
    best_f = f.copy()
    best_sc = _score_norm(f)
    t_k = 1.0
    topk = 6
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > stop_time:
            break

        b = _conv(f)
        M = float(b.max())
        idx = np.argpartition(b, -topk)[-topk:]
        vals = b[idx]
        vals = np.maximum(vals - vals.min() + 1e-12, 1e-12)
        wts = vals / vals.sum()

        grad = _subgradient(f, [int(k) for k in idx], wts)

        lr = 1.0 / (math.sqrt(it + 1) * (float(np.linalg.norm(grad)) + 1e-10))
        f_new = np.maximum(f * np.exp(-lr * grad), 0.0)
        f_new = _normalize(f_new)

        sc = _score_norm(f_new)
        if sc < best_sc - 1e-13:
            best_f = f_new.copy()
            best_sc = sc
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 200:
                f_new = _normalize(np.maximum(f + 0.02 * np.random.randn(n), 0))
                no_imp = 0
                best_check = _score_norm(f_new)
                if best_check < best_sc:
                    best_f = f_new.copy()
                    best_sc = best_check
        f = f_new

    return best_f, best_sc

def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 12.0

    parent = _get_parent()
    n_p = len(parent)
    best_seq = _normalize(parent)
    best_score = _score_norm(best_seq)
    if not math.isfinite(best_score):
        best_seq = np.ones(384) / 384.0
        best_score = _score_norm(best_seq)
        n_p = 384

    def _try_update(cand):
        nonlocal best_seq, best_score
        if cand is None:
            return
        cand = np.asarray(cand, dtype=np.float64).ravel()
        if cand.size == 0:
            return
        cand = np.clip(cand, 0, 1e6)
        if cand.sum() < 1e-10:
            return
        cand = _normalize(cand)
        try:
            sc = _score_norm(cand)
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-15:
            best_seq = cand.copy()
            best_score = sc

    rng = np.random.default_rng(seed)
    n = n_p

    # Build diverse initializations
    i_arr = np.arange(n, dtype=np.float64)
    inits = [best_seq.copy()]

    # Raised cosine
    cos_w = 0.5 + 0.5 * np.cos(2 * math.pi * (i_arr + 0.5) / n)
    inits.append(_normalize(cos_w))

    # Gaussian
    g1 = np.exp(-0.5 * ((i_arr - n / 2) / (n * 0.2)) ** 2)
    inits.append(_normalize(g1))

    # Asymmetric double bump
    b1 = np.exp(-0.5 * ((i_arr - n * 0.3) / (n * 0.12)) ** 2)
    b2 = np.exp(-0.5 * ((i_arr - n * 0.7) / (n * 0.12)) ** 2)
    inits.append(_normalize(b1 + 0.7 * b2))

    # Smooth random
    rs = np.abs(rng.standard_normal(n))
    from scipy.ndimage import gaussian_filter1d
    rs = gaussian_filter1d(rs, sigma=max(2, n // 15))
    inits.append(_normalize(rs))

    # Perturbed parent
    inits.append(_normalize(np.maximum(parent + 0.3 * float(np.mean(parent)) * rng.standard_normal(n), 0)))

    # Pinned parent (just in case)
    inits.append(_normalize(parent))

    # ===== Phase 1: LP-based active-set descent =====
    p1_end = t0 + budget * 0.35
    time_per_init = max(15.0, (p1_end - t0 - 5) / len(inits))

    for idx, init in enumerate(inits):
        if time.monotonic() > p1_end - 15:
            break
        stop = min(p1_end - 10, time.monotonic() + time_per_init)
        try:
            res, sc = _fw_active(init, n, stop, max_iter=200)
            _try_update(res)
        except Exception:
            pass

    # ===== Phase 2: Mirror descent =====
    p2_end = min(deadline, t0 + budget * 0.55)
    if time.monotonic() < p2_end - 10:
        try:
            res, sc = _mirror_descent(best_seq, n, p2_end - 5, max_iter=20000)
            _try_update(res)
        except Exception:
            pass

    # ===== Phase 3: Nesterov subgradient =====
    p3_end = min(deadline, t0 + budget * 0.70)
    if time.monotonic() < p3_end - 10:
        try:
            res, sc = _nest_subgrad(best_seq, n, p3_end - 5, max_iter=15000)
            _try_update(res)
        except Exception:
            pass

    # ===== Phase 4: Coordinate polish (long) =====
    p4_end = min(deadline, t0 + budget * 0.85)
    if time.monotonic() < p4_end - 10:
        try:
            res, sc = _coord_polish(best_seq, n, p4_end - 5)
            _try_update(res)
        except Exception:
            pass

    # ===== Phase 5: Try nearby lengths =====
    p5_end = min(deadline, t0 + budget * 0.93)
    if time.monotonic() < p5_end - 20 and best_score > 1.503:
        for L in [n_p - 16, n_p, n_p + 16, n_p + 32, n_p + 64]:
            if time.monotonic() > p5_end - 10:
                break
            if L <= 0 or L > 1024:
                continue
            if L == n_p:
                continue
            init_l = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, n_p), best_seq)
            init_l = _normalize(init_l)
            stop = min(p5_end - 5, time.monotonic() + 25)
            try:
                res, sc = _fw_active(init_l, L, stop, max_iter=80)
                _try_update(res)
            except Exception:
                pass

    # ===== Phase 6: Final coordinate polish =====
    if time.monotonic() < deadline - 5:
        try:
            res, sc = _coord_polish(best_seq, len(best_seq), deadline - 3)
            _try_update(res)
        except Exception:
            pass

    # ===== Phase 7: One more image/mirror burst =====
    if time.monotonic() < deadline - 5:
        try:
            res, sc = _mirror_descent(best_seq, len(best_seq), deadline - 3, max_iter=10000)
            _try_update(res)
        except Exception:
            pass

    # Final validation with exact evaluator
    final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    try:
        sc = evaluate_sequence(final_list)
        if math.isfinite(sc) and sc > 0:
            return final_list
    except Exception:
        pass

    # Fallback to parent
    fb = [float(max(0.0, min(1000.0, x))) for x in parent]
    try:
        sc = evaluate_sequence(fb)
        if math.isfinite(sc) and sc > 0:
            return fb
    except Exception:
        pass

    return [1.0] * max(1, n_p)
