```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(198, dtype=np.float64)
    if p.size == 0:
        p = np.ones(198, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(198, dtype=np.float64)
    return p

def _conv(a):
    n = len(a)
    if n <= 256:
        return np.convolve(a, a)
    fsize = next_fast_len(2*n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2*n-1]
    np.maximum(c, 0, out=c)
    return c

def _score_fast(a):
    """Score assuming a is already normalized to sum 1."""
    n = len(a)
    c = _conv(a)
    return 2.0 * n * c.max()

def _subgradient_step(a, conv, active_frac=0.05, top_k=20):
    """Compute subgradient of max(conv) using active set of peaks."""
    n = len(a)
    m = conv.max()
    if m < 1e-15:
        return np.zeros(n)
    
    # Use top-k peaks or active fraction
    n_active = max(top_k, int(active_frac * (2*n - 1)))
    n_active = min(n_active, 2*n - 1)
    
    # Find indices of top peaks
    # Use partition for efficiency
    if n_active >= 2*n - 1:
        active_mask = np.ones(2*n - 1, dtype=bool)
    else:
        # Get the top n_active values
        threshold = np.partition(conv, -(n_active))[-n_active]
        active_mask = conv >= threshold
        # If there are ties at the threshold, include all of them
        # (this is fine for subgradient)
    
    n_act = active_mask.sum()
    if n_act == 0:
        active_mask = np.zeros(2*n-1, dtype=bool)
        active_mask[np.argmax(conv)] = True
        n_act = 1
    
    # Compute gradient: g_i = (2/n_act) * sum_{k active} a_{k-i} [0<=k-i<n]
    # This is (2/n_act) * sum_{k active} a_{k-i} for valid k-i
    # = (2/n_act) * sum_{j=0}^{n-1} a_{i+j} * active_mask[i+j]
    
    # Zero-pad a to length 2n-1
    a_pad = np.zeros(2*n - 1)
    a_pad[:n] = a
    
    # g_i = (2/n_act) * sum_{j=0}^{n-1} a_pad[i+j] * active_mask[i+j]
    # This is a sliding dot product
    
    # Efficient computation using cumsum of (a_pad * active_mask)
    s = a_pad * active_mask.astype(np.float64)
    # g_i = (2/n_act) * sum_{j=0}^{n-1} s[i+j]
    # = (2/n_act) * (cumsum_s[i+n] - cumsum_s[i])
    cumsum_s = np.zeros(2*n)
    np.cumsum(s, out=cumsum_s[1:])
    
    g = (2.0 / n_act) * (cumsum_s[n:] - cumsum_s[:n])
    
    return g

def _project_normalize(a):
    """Project onto a >= 0 and sum(a) = 1."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def _peaks_minimize(a0, n, stop_time, seed=42):
    """Minimize max(a*a) using subgradient method."""
    rng = np.random.default_rng(seed)
    a = _project_normalize(a0.astype(np.float64).copy())
    
    best_a = a.copy()
    best_score = _score_fast(a)
    
    alpha_base = 0.1
    alpha_min = 1e-5
    alpha = alpha_base
    
    n_active_top = max(10, int(0.05 * (2*n - 1)))
    
    stall_count = 0
    max_stall = 200
    
    conv = _conv(a)
    
    for it in range(100000):
        if time.monotonic() > stop_time - 2.0:
            break
        
        m = conv.max()
        
        # Compute subgradient
        g = _subgradient_step(a, conv, top_k=n_active_top)
        
        # Line search
        improved = False
        for trial_alpha in [alpha, alpha*0.5, alpha*0.25, alpha*2.0, alpha*0.1]:
            if trial_alpha < alpha_min:
                continue
            a_new = a - trial_alpha * g
            a_new = _project_normalize(a_new)
            c_new = _conv(a_new)
            m_new = c_new.max()
            if m_new < m - 1e-14:
                a = a_new
                conv = c_new
                m = m_new
                alpha = min(alpha * 1.1, 0.5)
                stall_count = 0
                improved = True
                break
            else:
                alpha = max(alpha_min, trial_alpha * 0.5)
        
        if not improved:
            stall_count += 1
            alpha = max(alpha_min, alpha * 0.7)
            if stall_count >= max_stall:
                # Perturb and restart
                a = a + rng.normal(0, 0.01, n)
                a = _project_normalize(a)
                conv = _conv(a)
                stall_count = 0
                n_active_top = max(5, int(0.02 * (2*n-1)))
        
        sc = 2.0 * n * conv.max()
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
    
    return best_a, best_score
```

def _conv(a):
    n = len(a)
    if n <= 256:
        c = np.convolve(a, a)
    else:
        fsize = next_fast_len(2*n - 1)
        F = rfft(a, fsize)
        c = irfft(F * F, fsize)[:2*n-1]
    np.maximum(c, 0, out=c)
    return c
