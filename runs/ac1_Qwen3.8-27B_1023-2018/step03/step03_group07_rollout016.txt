    s = float(a.sum())

        return float('inf')
    c = np.convolve(a, a)

def _build_A(a):
    n = a.size
    m = 2 * n - 1
    k = np.arange(m)
    j = np.arange(n)
    diff = k[:, None] - j[None, :]
    mask = (diff >= 0) & (diff < n)
    A = np.zeros((m, n), dtype=np.float64)
    A[mask] = a[diff[mask]]
    return A

def _lp(A, M, n):

    try:
        res = linprog(c, A_ub=A, b_ub=np.full(A.shape[0], M),
                      bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'time_limit': 1.0})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    if not np.isfinite(g).all():
        return None
    gsum = g.sum()
    if gsum < 1e-15:
        return None
    return g / gsum
