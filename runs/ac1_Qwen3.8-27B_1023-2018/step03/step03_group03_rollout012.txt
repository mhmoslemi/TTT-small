        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method="highs",
                      options={"presolve": True, "time_limit": time_limit})
    except Exception:
        return None, None
    if res.status != 0 or res.x is None:
        return None, None
    g = np.asarray(res.x[:n], dtype=np.float64)
    t = float(res.x[n])
    g = np.maximum(g, 0)
    gs = float(g.sum())
    if gs < 1e-15:
        return None, None
    return g / gs, t


def _coord_descent(f, stop_time, max_sweeps=60, seed=777, verbose=False):
    """Coordinate descent with exact 1D optimization."""
    n = len(f)
    f = np.clip(f.astype(np.float64).copy(), 0, None)
    S = float(f.sum())
    if S < 0.01:
        return f, math.inf
    conv = _fft_conv(f)
    best_score = 2.0 * n * conv.max() / (S * S)
    rng = np.random.default_rng(seed)
    improvement = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > stop_time - 5.0:
            break
        sweep_improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > stop_time - 5.0:
                break

            v = f[i]
            S_rest = S - v

            # Influence: when f_i changes by delta, conv changes by delta * infl + delta^2 * (1 at 2i)
            infl = np.zeros(2 * n - 1)
            start = i
            end = min(i + n, 2 * n - 1)
            infl[start:end] = 2.0 * f[:end - start]

            # 1D minimization: minimize max_k(conv[k] + d*infl[k] + d^2*delta(k,2i)) / (S_rest + v + d - j + d)^2
            # where d = new_v - v, new_v = v + d
            # Objective: h(d) = 2n * max_k(conv[k] + d*infl[k] + d^2*[k==2i]) / (S+d)^2
            # where d ranges over [-(v), X_max]

            d_max = max(3.0 * v + 0.5 / n, 0.1)
            d_max = min(d_max, 50.0 / n)

            # Coarse grid
            n_coarse = 100
            d_grid = np.linspace(-v, d_max, n_coarse)
            best_d = 0.0
            best_obj = best_score

            for d in d_grid:
                S_new = S + d
                if S_new < 0.01:
                    continue
                # max of (conv + d*infl + d^2*at_2i) / S_new^2
                # To avoid full array ops, compute max of (conv[k] + d*infl[k] + d^2*eq)
                temp_max = conv[start:end] + d * infl[start:end]
                if 0 <= 2 * i <= 2 * n - 2:
                    temp_max = np.maximum(temp_max, d * d)  # only at position 2i in [start,end)
                    # Actually need to be more careful
                # For positions outside [start, end), conv doesn't change
                # The max is max(conv outside [start,end), max(conv + d*infl inside [start,end), d^2 at 2i)
                outer_max = np.max(np.concatenate([conv[:start], conv[end:]])) if (start > 0 or end < 2*n-1) else 0.0
                inner_vals = conv[start:end] + d * infl[start:end]
                # Add d^2 at position 2i (relative index 2i - start)
                ri = 2 * i - start
                if 0 <= ri < end - start:
                    inner_vals = inner_vals.copy()
                    inner_vals[ri] += d * d
                else:
                    # 2i is outside [start, end) which shouldn't happen since start=i, end<=i+n, 2i < i+n for i < n
                    outer_max = max(outer_max, d * d)
                m_new = max(outer_max, float(np.max(inner_vals)))
                obj = 2.0 * n * m_new / (S_new * S_new)
                if obj < best_obj - 1e-15:
                    best_obj = obj
                    best_d = d

            # Fine grid
            step = d_max / n_coarse
            d_fine = np.linspace(max(-v, best_d - 3 * step), min(d_max, best_d + 3 * step), 50)
            for d in d_fine:
                S_new = S + d
                if S_new < 0.01:
                    continue
                outer_max = np.max(np.concatenate([conv[:start], conv[end:]])) if (start > 0 or end < 2*n-1) else 0.0
                inner_vals = conv[start:end] + d * infl[start:end]
                ri = 2 * i - start
                if 0 <= ri < end - start:
                    inner_vals = inner_vals.copy()
                    inner_vals[ri] += d * d
                else:
                    outer_max = max(outer_max, d * d)
                m_new = max(outer_max, float(np.max(inner_vals)))
                obj = 2.0 * n * m_new / (S_new * S_new)
                if obj < best_obj - 1e-15:
                    best_obj = obj
                    best_d = d

            # Apply
            if abs(best_d) > 1e-12 and best_obj < best_score - 1e-14:
                f[i] = v + best_d
                S += best_d
                conv[start:end] += best_d * infl[start:end]
                if 0 <= 2 * i <= 2 * n - 2:
                    conv[2 * i] += best_d * best_d
                best_score = best_obj
                sweep_improved = True
                improvement += 1

        if not sweep_improved:
            break

    f = np.clip(f, 0, 1000)
    return f, _score(f)


def _main_optimize(f0, n, stop_time, seed):
    """Main LP + subgradient optimization loop."""
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    while f.size < n:
        f = np.concatenate([f, np.zeros(n - f.size)])
    f = f[:n]
    f_sum = float(f.sum())
    if f_sum < 1e-10:
        f = np.ones(n)
    else:
        f = f / f_sum

    best_f = f.copy()
    best_score = _score(f)
    t = 0.15
    stall = 0
    max_stall = 80
    iteration = 0

    while time.monotonic() < stop_time - 10.0 and iteration < 2000:
        # Get LP direction
        g_dir, M = _lp_maxsum(f, n, time_limit=3.0)

        if g_dir is None:
            # Fallback to subgradient
            sg, conv, idx = _subgradient_direction(f, top_k=5)
            g_dir = -sg
            gs = float(g_dir.sum())
            if gs > 1e-12:
                g_dir = g_dir / gs
            else:
                g_dir = f.copy()

        dist = float(np.linalg.norm(g_dir - f))
        if dist < 1e-8:
            stall += 3
            if stall > max_stall:
                break
            f = np.clip(f + rng.normal(0, 0.03, n), 0, None)
            fs = f.sum()
            if fs > 1e-10:
                f /= fs
            continue

        # Line search
        improved = False
        for _ in range(15):
            f_new = (1.0 - t) * f + t * g_dir
            f_new = np.clip(f_new, 0, None)
            fs_new = float(f_new.sum())
            if fs_new < 0.01:
                t = max(0.005, t * 0.5)
                continue
            f_new = f_new / fs_new
            sc = _score(f_new)
            if sc < best_score - 1e-13:
                f = f_new
                best_f = f.copy()
                best_score = sc
                t = min(0.5, t * 1.08)
                stall = 0
                improved = True
                break
            else:
                t = max(0.005, t * 0.5)

        if not improved:
            stall += 1
            if stall >= max_stall:
                # Perturb and reset
                f = np.clip(f + rng.normal(0, 0.05, n), 0, None)
                fs = f.sum()
                if fs > 1e-10:
                    f /= fs
                else:
                    f = best_f.copy()
                t = 0.15
                stall = 0

        iteration += 1

    return best_f, best_score


def _try_length(n, init_arr, stop_time, seed):
    """Run main optimization + polish for a given length."""
    if time.monotonic() > stop_time - 30:
        return init_arr, _score(init_arr)

    # Normalize
    f = np.clip(np.asarray(init_arr, dtype=np.float64), 0, None)
    while f.size < n:
        f = np.concatenate([f, np.zeros(n - f.size)])
    f = f[:n]
    fs = f.sum()
    if fs > 1e-10:
        f = f / fs

    # Main LP optimization
    lp_stop = stop_time - max(60.0, (stop_time - time.monotonic()) * 0.4)
    try:
        f1, sc1 = _main_optimize(f, n, lp_stop, seed)
    except Exception:
        f1, sc1 = f, _score(f)

    # Coordinate polish
    pol_stop = stop_time - 15
    try:
        f2, sc2 = _coord_descent(f1, pol_stop, max_sweeps=50, seed=seed + 1)
    except Exception:
        f2, sc2 = f1, sc1

    final_sc = _score(f2)
    return f2, final_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(30.0, float(budget_s))
    hard_deadline = t0 + budget - 15.0

    # Get parent
    try:
        h = globals()["height_sequence_1"]
        parent = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        parent = None
    if parent is None or parent.size == 0:
        parent = np.ones(198)
    parent = np.clip(parent, 0, 1000)
    parent = np.where(np.isfinite(parent), parent, 0.0)
    n_p = len(parent)
    parent_norm = parent.copy()
    ps = parent_norm.sum()
    if ps > 1e-10:
        parent_norm /= ps

    fallback = parent.copy()
    fallback_sc = _score(fallback)
    if not math.isfinite(fallback_sc):
        fallback = np.ones(64)
        fallback_sc = _score(fallback)

    best_seq = fallback.copy()
    best_score = fallback_sc

    def _update(seq, label=""):
        nonlocal best_seq, best_score
        try:
            exact = float(evaluate_sequence([float(x) for x in seq]))
        except Exception:
            exact = math.inf
        if math.isfinite(exact) and exact < best_score - 1e-14:
            best_seq = np.array(seq, dtype=np.float64).copy()
            best_score = exact
            if label:
                pass  # could log

    _update(parent, "parent")

    # === Phase 1: Main LP optimization on parent length ===
    phase1_end = t0 + budget * 0.35
    stop_lp1 = min(phase1_end, hard_deadline - 60)

    try:
        f1, sc1 = _main_optimize(parent_norm, n_p, stop_lp1, seed + 10)
    except Exception:
        f1, sc1 = parent_norm, fallback_sc
    if np.all(np.isfinite(f1)) and f1.sum() > 0.01:
        _update(f1, "lp_parent")

    # === Phase 2: Coordinate polish on best ===
    phase2_end = min(t0 + budget * 0.55, hard_deadline - 40)
    if time.monotonic() < phase2_end - 20:
        cur = best_seq.copy()
        cur_s = cur.sum()
        if cur_s > 1e-10:
            cur = cur / cur_s
        try:
            f2, sc2 = _coord_descent(cur, phase2_end, max_sweeps=60, seed=seed + 20)
        except Exception:
            f2, sc2 = cur, _score(cur)
        if np.all(np.isfinite(f2)) and f2.sum() > 0.01:
            _update(f2, "polish1")

    # === Phase 3: Retry LP with perturbation ===
    phase3_end = min(t0 + budget * 0.65, hard_deadline - 80)
    if time.monotonic() < phase3_end - 30:
        rng = np.random.default_rng(seed + 999)
        cur = best_seq.copy()
        cur_s = cur.sum()
        if cur_s > 1e-10:
            cur = cur / cur_s
        # Perturb
        cur_p = np.clip(cur + rng.normal(0, 0.04, len(cur)), 0, None)
        cur_p /= cur_p.sum()
        try:
            f3, sc3 = _main_optimize(cur_p, len(cur_p), phase3_end, seed + 200)
        except Exception:
            f3, sc3 = cur_p, _score(cur_p)
        if np.all(np.isfinite(f3)) and f3.sum() > 0.01:
            _update(f3, "lp_retry")
            # Quick polish
            pol_stop = min(phase3_end + budget * 0.05, hard_deadline - 30)
            if time.monotonic() < pol_stop - 10:
                try:
                    f3b, sc3b = _coord_descent(f3, pol_stop, max_sweeps=30, seed=seed + 300)
                    _update(f3b, "polish_retry")
                except Exception:
                    pass

    # === Phase 4: Try a few other initializations on same length ===
    phase4_end = min(t0 + budget * 0.80, hard_deadline - 60)
    if time.monotonic() < phase4_end - 40:
        n = n_p
        i_arr = np.arange(n, dtype=np.float64)
        inits = [
            0.3 + 0.7 * (i_arr / n) ** 0.7,       # ramp-like
            0.5 + 0.5 * np.cos(np.pi * i_arr / n),  # cosine taper
            np.exp(-2 * (i_arr / n - 0.6) ** 2),   # off-center gaussian
        ]
        for idx, init in enumerate(inits):
            if time.monotonic() > phase4_end - 20:
                break
            init = np.clip(init, 0, None)
            isum = init.sum()
            if isum < 1e-6:
                init = np.ones(n)
                isum = n
            init = init / isum
            stop_init = min(phase4_end - 10, time.monotonic() + budget * 0.06)
            try:
                fi, sci = _main_optimize(init, n, stop_init, seed + 1000 + idx)
                _update(fi, f"init_{idx}")
            except Exception:
                pass

    # === Phase 5: Try different lengths ===
    phase5_end = min(t0 + budget * 0.90, hard_deadline - 50)
    if time.monotonic() < phase5_end - 30:
        other_lengths = [256, 320, 384, 448, 512, 640, 768]
        if n_p not in other_lengths:
            other_lengths.append(n_p)
        other_lengths = sorted(set(other_lengths))
        time_per_len = max(10.0, (phase5_end - time.monotonic()) / max(1, len(other_lengths)))

        for L in other_lengths:
            if time.monotonic() > phase5_end - 20:
                break
            if best_score < 1.5028:
                break
            stop_L = min(phase5_end - 15, time.monotonic() + time_per_len * 1.2)
            # Interpolate best to length L
            if best_seq.size > 1:
                init_L = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, best_seq.size), best_seq)
            else:
                init_L = np.ones(L)
            sum_L = init_L.sum()
            if sum_L > 1e-6:
                init_L /= sum_L
            try:
                fL, scL = _try_length(L, init_L, stop_L, seed + 5000 + L)
                if np.all(np.isfinite(fL)) and fL.sum() > 0.01:
                    _update(fL, f"len_{L}")
            except Exception:
                pass

    # === Phase 6: Final intensive coordinate polish ===
    final_end = hard_deadline - 10.0
    if time.monotonic() < final_end - 15:
        cur = best_seq.copy()
        cur_s = cur.sum()
        if cur_s > 1e-10:
            cur = cur / cur_s
        try:
            f_fin, sc_fin = _coord_descent(cur, final_end, max_sweeps=80, seed=seed + 4000)
        except Exception:
            f_fin, sc_fin = cur, _score(cur)
        if np.all(np.isfinite(f_fin)) and f_fin.sum() > 0.01:
            _update(f_fin, "final_polish")

        # One more short LP round if time
        if time.monotonic() < final_end - 30:
            try:
                f_fin2, sc_fin2 = _main_optimize(f_fin, len(f_fin), final_end - 5, seed + 5000)
                _update(f_fin2, "final_lp")
                if time.monotonic() < final_end - 10:
                    ff, ss = _coord_descent(f_fin2, final_end, max_sweeps=20, seed=seed + 6000)
                    _update(ff, "final_polish2")
            except Exception:
                pass

    # Final validation
    best_seq = np.clip(np.asarray(best_seq, dtype=np.float64), 0, 1000)
    best_seq = np.where(np.isfinite(best_seq), best_seq, 0.0)
    if best_seq.size == 0 or best_seq.sum() < 0.01:
        best_seq = fallback.copy()

    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        best_seq = fallback.copy()

    return [float(x) for x in best_seq]
