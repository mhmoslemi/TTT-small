    c = irfft(F*F, fs)[:2*n-1]
    np.maximum(c, 0.0, out=c)

def _get_parent():

def _conv(a):
    n = len(a)
    fs = next_fast_len(2 * n - 1)
    F = rfft(a, fs)
    c = irfft(F * F, fs)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _sc(a):
    n = len(a)
    m = float(_conv(a).max())

def _norm(a):

                      bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'presolve': True})

        return None

        except Exception:
            pass

    # Validate with exact evaluator
    fl = [float(max(0.0, min(1000.0, x))) for x in best_f]
    try:
        s = evaluate_sequence(fl)
        if math.isfinite(s) and s > 0:
            return fl
    except Exception:
        pass

    # Fallback to parent
    fb = [float(max(0.0, min(1000.0, x))) for x in parent]
    try:
        s = evaluate_sequence(fb)
        if math.isfinite(s) and s > 0:
            return fb
    except Exception:
        pass

    return [1.0] * max(n, 1)
