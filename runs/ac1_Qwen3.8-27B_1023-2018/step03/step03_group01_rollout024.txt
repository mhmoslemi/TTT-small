```python
            old_val = f[i]
            old_sc = best_sc

                        best_t = old_val * ratio
            else:

def _subgrad_mirror(f, n, stop_time, seed):
    """Subgradient + mirror descent for non-negativity."""
    rng = np.random.default_rng(seed)
    f = np.maximum(f.astype(np.float64).copy(), 0.0)
    s = f.sum()
    f = f / s if s > 1e-10 else np.ones(n) / n
    
    best_f = f.copy()
    best_sc = _score(f)
    
    eta = 0.5  # step size for mirror updates
    patience = 0
    max_patience = 500
    
    while time.monotonic() < stop_time:
        b = _conv_fft(f)
        M = float(b.max())
        k_star = int(np.argmax(b))
        
        # Gradient of (f*f)_{k_star} w.r.t. f
        grad = np.zeros(n)
        for j in range(n):
            kj = k_star - j
            if 0 <= kj < n:
                if kj == j:  # k_star = 2j
                    grad[j] = 2.0 * f[j]
                else:
                    grad[j] = 2.0 * f[kj]
        
        # Gradient of score = 2n * M / s^2 w.r.t. f:
        # d(score)/d(f_j) = 2n/s^2 * dM/d(f_j) - 4n*M/s^3 * 1
        # But M is the max, so we use the gradient at the active set.
        
        s = f.sum()
        # For mirror descent, we need the "primal" gradient
        # score = 2n * M / s^2
        # grad_score_j = 2n/s^2 * grad_M_j - 4n*M/s^3
        
        grad_score = (2.0 * n / (s * s)) * grad - (4.0 * n * M / (s * s * s)) * np.ones(n)
        
        # Mirror descent: f_j <- f_j * exp(-eta * grad_score_j)
        f_new = f * np.exp(-eta * grad_score)
        f_new = np.maximum(f_new, 0.0)
        s_new = f_new.sum()
        if s_new < 1e-10:
            break
        f_new = f_new / s_new
        
        sc = _score(f_new)
        if sc < best_sc - 1e-14:
            best_f = f_new.copy()
            best_sc = sc
            patience = 0
        else:
            patience += 1
            # Decay eta
            eta *= 0.999
            if patience > max_patience:
                # Restart with random perturbation
                f = np.maximum(f + rng.normal(0, 0.02, n), 0)
                f = f / f.sum()
                patience = 0
                eta = 0.3
        
        f = f_new
    
    return best_f, best_sc

def _fw_loop(f, n, stop_time, seed):
    """Frank-Wolfe with LP direction."""
    rng = np.random.default_rng(seed)
    f = np.maximum(f.astype(np.float64).copy(), 0.0)
    s = f.sum()
    f = f / s if s > 1e-10 else np.ones(n) / n
    best_f = f.copy()
    best_sc = _score(f)
    
    gamma = 0.5  # FW step size
    stall = 0
    
    while time.monotonic() < stop_time - 2:
        b = _conv_fft(f)
        M = float(b.max())
        if M < 1e-15:
            break
        
        g_dir = _fw_lp(f, n, M, time_limit=8)
        if g_dir is None:
            stall += 5
            if stall > 30:
                break
            f = np.maximum(f + rng.normal(0, 0.05, n), 0)
            f = f / f.sum()
            continue
        
        # FW step: f_new = (1-gamma)*f + gamma*g_dir
        # Try multiple step sizes
        best_t = 0.0
        best_new_sc = best_sc
        for t in [gamma, 0.3, 0.5, 0.7, 0.2, 0.4, 0.1]:
            f_new = (1 - t) * f + t * g_dir
            f_new = np.maximum(f_new, 0)
            s_new = f_new.sum()
            if s_new < 1e-10:
                continue
            f_new = f_new / s_new
            sc = _score(f_new)
            if sc < best_new_sc:
                best_new_sc = sc
                best_t = t
        
        if best_t > 0:
            f_new = (1 - best_t) * f + best_t * g_dir
            f_new = np.maximum(f_new, 0)
            f_new = f_new / f_new.sum()
            if best_new_sc < best_sc - 1e-13:
                f = f_new
                best_f = f.copy()
                best_sc = best_new_sc
                stall = 0
                gamma = min(0.7, gamma * 1.1)
            else:
                stall += 1
                gamma *= 0.8
        else:
            stall += 1
        
        if stall > 40:
            f = np.maximum(f + rng.normal(0, 0.1, n), 0)
            f = f / f.sum()
            stall = 0
            gamma = 0.5
    
    return best_f, best_sc
```

```python
    F = rfft(a, fs)
    c = irfft(F * F, fs)[:2 * n - 1]
    return np.maximum(c, 0.0)
```

```python
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv(a)
    m = float(b.max())
    return 2.0 * n * m / (s * s)

def _build_A(f, n):
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A

def _fw_lp(f, n, M, tl=5.0):
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, None)] * n,
                      method="highs",
                      options={"presolve": True, "time_limit": tl})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    gs = float(g.sum())
    if gs < 1e-14:
        return None
    return g / gs

def _fw_step(f, n, stop_time, seed, max_iter=500):
    """Frank-Wolfe with LP direction."""
    rng = np.random.default_rng(seed)
    f = np.maximum(f.astype(np.float64).copy(), 0)
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / s
    best_f = f.copy()
    best_sc = _score(f)
    gamma = 0.5
    stall = 0
    
    for it in range(max_iter):
        if time.monotonic() > stop_time:
            break
        b = _conv(f)
        M = float(b.max())
        if M < 1e-15:
            break
        
        g_dir = _fw_lp(f, n, M, tl=5.0)
        if g_dir is None:
            stall += 3
            if stall > 20:
                break
            f = np.maximum(f + rng.normal(0, 0.03, n), 0)
            f = f / max(f.sum(), 1e-10)
            continue

        best_t = 0.0
        best_try_sc = best_sc
        for t in (gamma, 0.3, 0.5, 0.7, 0.2, 0.4, 0.1, 0.6):
            fn = (1 - t) * f + t * g_dir
            fn = np.maximum(fn, 0)
            s_new = fn.sum()
            if s_new < 1e-10:

            fn = fn / s_new
            sc = _score(fn)
            if sc < best_try_sc:

                best_t = t

            fn = np.maximum(fn, 0)
            fn = fn / fn.sum()
            f = fn
            best_f = f.copy()
            best_sc = best_try_sc
            stall = 0
            gamma = min(0.8, gamma * 1.05)
        else:
            stall += 1
            gamma *= 0.85
            if stall > 30:
                f = np.maximum(f + rng.normal(0, 0.08, n), 0)
                f = f / f.sum()
                gamma = 0.5
                stall = 0
```

```python
    """Update score after changing f[i] from old_v to new_v.
    b_cache is the full convolution before the change."""
    n = len(f)
    s_old = f.sum()
    s_new = s_old - old_v + new_v
    if s_new < 0.01:
        return math.inf

    delta = new_v - old_v
    if abs(delta) < 1e-15:
        m_new = float(b_cache.max())
        return 2.0 * n * m_new / (s_new * s_new)

    # But f[k-i] might reference f[i] itself, so we need to use old values
    # Actually, f[k-i] for k-i != i is unchanged. For k-i == i (k = 2i), 
    # it's f[i] which changes.
    
    # Let's be careful:

    # b_new[2i] = b_cache[2i] + 2*delta*f_old[i] + delta*(new_v)  ... hmm this is getting complicated.

    #   if k=2i: v^2
    #   if 0<=k-i<n and k!=2i: 2*v*f[k-i]

    #   if 0<=k-i<n and k!=2i: 2*t*f[k-i]

    # For k≠2i and 0<=k-i<n: old term is 2*f[i]*f[k-i] = 2*v*f[k-i], new is 2*t*f[k-i].
    #   Here f[k-i] is unchanged (since k-i ≠ i).
    
    # So:
    # b_new[k] = b_cache[k] + 2*delta*f[k-i] for k in [i, i+n-1], k ≠ 2i
    # b_new[2i] = b_cache[2i] + (t^2 - v^2)
    
    # ... but I need to be careful about the b_cache. The b_cache was computed
    # with f[i] = v. So b_cache already includes the contribution of f[i].
    # When I change f[i] to t:
    # For k ≠ 2i: b[k] loses 2*v*f[k-i] and gains 2*t*f[k-i] → change = 2*delta*f[k-i]
    # For k = 2i: b[2i] loses v^2 and gains t^2 → change = t^2 - v^2
    #   (Note: there's no 2*delta*f[i] term because the f[i] in f[k-i] IS the changing variable)
    
    # Wait, for k = 2i:
    # Old: f[i]*f[2i-i] = f[i]*f[i] = v^2 (one term in the sum, since j=i=k-j)
    # New: t*t = t^2
    # Change: t^2 - v^2. Correct.
    
    # For k ≠ 2i, 0 <= k-i < n:
    # Old terms involving f[i]: f[i]*f[k-i] (j=i) + f[k-i]*f[i] (j=k-i) = 2*v*f[k-i]
    # New terms: 2*t*f[k-i]
    # Change: 2*delta*f[k-i]. And f[k-i] is unchanged (k-i ≠ i).
    
    # So the update is:
    # For k in [max(0,i), min(2n-2, i+n-1)]\{2i}: b[k] += 2*delta*f[k-i]
    # For k=2i (if 0 <= 2i <= 2n-2): b[2i] += t^2 - v^2 = (t-v)(t+v) = delta*(t+v)
    
    # Note: 2i is in [i, i+n-1] iff i <= 2i <= i+n-1 iff i <= n-1, which is true.
    
    # Let me implement this.
    
    b_new = b_cache.copy()  # O(2n-1) copy
    lo = i  # k starts at i (since k-i >= 0)
    hi = min(2 * n - 2, i + n - 1)  # k-i <= n-1
    
    for k in range(lo, hi + 1):
        if k == 2 * i:
            b_new[k] += delta * (new_v + old_v)
        else:
            fj = f[k - i]  # this is unchanged (k-i != i)
            b_new[k] += 2 * delta * fj
    
    m_new = float(b_new.max())
    return 2.0 * n * m_new / (s_new * s_new)
```

Wait, this is O(n) per trial (the loop over k), plus O(n) for the b_new copy. But I'm doing this for each of ~12 trials per coordinate, so O(12n) per coordinate, O(12n²) per full pass. For n=384: 12 × 384² ≈ 1.8M operations. Very fast.
