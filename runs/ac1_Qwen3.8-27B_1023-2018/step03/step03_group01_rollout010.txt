def _conv(a):
    """Full linear convolution via FFT."""
    n = len(a)
    sz = next_fast_len(2*n - 1)
    F = rfft(a, sz)
    c = irfft(F * F, sz)[:2*n-1]
    return c

def _score(a):
    n = len(a)
    S = float(a.sum())
    if S < 0.01:
        return math.inf
    b = _conv(a)
    M = float(b.max())
    return 2.0 * n * M / (S * S)

def _build_A(f, n):
    """Build the (2n-1) x n matrix where A[k,j] = f[k-j] if valid, else 0."""
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = f
    return A

def _lp_direction(f, M, n):
    """Find g >= 0 maximizing sum(g) subject to A(g) <= M."""
    A = _build_A(f, n)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M),
                  bounds=[(0, None)]*n, method='highs',
                  options={'presolve': True, 'time_limit': 10.0})
    if res.status != 0 or res.x is None:
        return None
    g = np.clip(res.x, 0, None)
    gs = float(g.sum())
    if gs < 1e-14:
        return None
    return g / gs  # normalize to sum = 1

def _grad_polish(f0, n, time_limit, seed):
    """Gradient-based polish using subgradient of max convolution."""
    rng = np.random.default_rng(seed)
    f = np.asarray(f0, dtype=np.float64).copy()
    f = np.clip(f, 0, None)
    f /= f.sum()
    
    best_f = f.copy()
    best_score = _score(f)
    lr = 0.02
    momentum = np.zeros(n)
    beta = 0.9
    no_imp = 0
    
    while time.monotonic() < time_limit - 2.0:
        b = _conv(f)
        M = float(b.max())
        S = float(f.sum())
        
        # Find all lags where b is near max
        thresh = M * 0.999 + 1e-12
        active = np.where(b >= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        
        # Subgradient of M w.r.t. f
        grad_M = np.zeros(n)
        for k in active:
            k = int(k)
            # d(f*f)[k]/df[i] = f[k-i] if 0 <= k-i < n
            for i in range(max(0, k-n+1), min(n, k+1)):
                grad_M[i] += f[k - i]
        
        # Gradient of M/S^2 (with S=1 after normalization, but we track it)
        # d(score)/df[i] = 2n/S^2 * grad_M[i] - 4nM/S^3
        grad = (2.0 * n / (S*S)) * grad_M - (4.0 * n * M / (S*S*S))
        
        # Nesterov momentum update
        momentum = beta * momentum - lr * grad
        f_new = f + momentum
        f_new = np.clip(f_new, 0, None)
        S_new = float(f_new.sum())
        if S_new < 1e-10:
            f_new = f.copy()
        else:
            f_new /= S_new
        
        sc = _score(f_new)
        if sc < best_score - 1e-14:
            f = f_new
            best_f = f.copy()
            best_score = sc
            no_imp = 0
            lr = min(0.1, lr * 1.02)
        else:
            no_imp += 1
            lr = max(0.002, lr * 0.97)
            if no_imp > 200:
                # Restart
                f = np.clip(f + 0.05 * rng.standard_normal(n), 0, None)
                f /= max(f.sum(), 1e-10)
                lr = 0.02
                beta = 0.85
                momentum = np.zeros(n)
                no_imp = 0
    
    return best_f, best_score

def _coord_polish(f0, n, time_limit, seed):
    """Coordinate descent polish."""
    rng = np.random.default_rng(seed)
    f = np.asarray(f0, dtype=np.float64).copy()
    f = np.clip(f, 0, None)
    f /= f.sum()
    best_score = _score(f)
    
    multipliers = (0.5, 0.8, 1.2, 1.5, 2.0, 0.3, 0.7, 1.3)
    no_full_pass = 0
    
    while time.monotonic() < time_limit - 2.0:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > time_limit - 2.0:
                break
            orig = f[i]
            if orig < 1e-12:
                f[i] = 0.001
                if _score(f) < best_score - 1e-14:
                    best_score = _score(f)
                    improved = True
                else:
                    f[i] = 0.0
                continue
            for m in multipliers:
                f[i] = orig * m
                sc = _score(f)
                if sc < best_score - 1e-14:
                    best_score = sc
                    improved = True
            f[i] = orig  # restore
        
        if not improved:
            no_full_pass += 1
            if no_full_pass >= 5:
                break
        else:
            no_full_pass = 0
    
    return f, best_score

import numpy as np
import math
import time
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(384, dtype=np.float64)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    return p.copy()


def _conv(a):
    n = len(a)
    sz = next_fast_len(2 * n - 1)
    F = rfft(a, sz)
    c = irfft(F * F, sz)[:2 * n - 1]
    return c


def _score(a):
    n = len(a)
    S = float(a.sum())
    if S < 0.01:
        return math.inf
    b = _conv(a)
    M = float(b.max())
    if not math.isfinite(M):
        return math.inf
    return 2.0 * n * M / (S * S)


def _build_A(f, n):
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_dir(f, M, n, timeout=15.0):
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': timeout})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.clip(np.asarray(res.x, dtype=np.float64), 0, None)
    gs = float(g.sum())
    if gs < 1e-14:
        return None
    return g / gs


def _line_search(f, g, n, best_sc, t_start=0.3, stop_time=None):
    best_t = 0.0
    best_fn = None
    t_hi = min(0.9, t_start * 1.5)
    t_lo = 0.0
    for _ in range(16):
        if stop_time and time.monotonic() > stop_time:
            break
        t_mid = 0.5 * (t_lo + t_hi)
        fn = (1.0 - t_mid) * f + t_mid * g
        np.maximum(fn, 0.0, out=fn)
        S = float(fn.sum())
        if S < 1e-10:
            t_hi = t_mid
            continue
        fn = fn / S
        sc = _score(fn)
        if sc < best_sc - 1e-14:
            t_lo = t_mid
            best_t = t_mid
            best_fn = fn.copy()
        else:
            t_hi = t_mid
        if t_hi - t_lo < 5e-4:
            break
    if best_fn is None:
        # Try a few fixed values
        for tv in [0.05, 0.1, 0.15, t_start, 0.4, 0.5]:
            if stop_time and time.monotonic() > stop_time:
                break
            fn = (1.0 - tv) * f + tv * g
            np.maximum(fn, 0.0, out=fn)
            S = float(fn.sum())
            if S < 1e-10:
                continue
            fn = fn / S
            sc = _score(fn)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_t = tv
                best_fn = fn.copy()
    return best_t, best_fn


def _lp_loop(f0, n, stop_time, seed):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel(), 0, None).copy()
    S = float(f.sum())
    if S < 1e-10:
        f = np.ones(n)
    else:
        f = f / S

    best_f = f.copy()
    best_sc = _score(f)
    t_cur = 0.25
    stall = 0
    max_stall = 50

    while time.monotonic() < stop_time - 1.0:
        b = _conv(f)
        M = float(b.max())
        if M < 1e-15:
            break

        g = _lp_dir(f, M, n, timeout=min(12.0, max(3.0, stop_time - time.monotonic() - 2.0)))
        if g is None:
            f = np.clip(f + 0.06 * rng.standard_normal(n), 0, None)
            S2 = float(f.sum())
            f = f / S2 if S2 > 1e-10 else best_f.copy()
            stall += 4
            t_cur = 0.25
            if stall > max_stall:
                break
            continue

        dist = float(np.linalg.norm(g - f))
        if dist < 1e-8:
            stall += 4
            if stall > max_stall:
                break
            f = np.clip(f + 0.08 * rng.standard_normal(n), 0, None)
            S2 = float(f.sum())
            f = f / S2 if S2 > 1e-10 else best_f.copy()
            t_cur = 0.30
            continue

        res_t, f_new = _line_search(f, g, n, best_sc, t_start=t_cur, stop_time=stop_time)
        if f_new is not None:
            sc_new = _score(f_new)
            if sc_new < best_sc - 1e-13:
                f = f_new
                best_f = f.copy()
                best_sc = sc_new
                t_cur = min(0.6, res_t * 1.2 + 0.02)
                stall = 0
            else:
                t_cur = max(0.03, t_cur * 0.55)
                stall += 1
        else:
            t_cur = max(0.03, t_cur * 0.5)
            stall += 1

        if stall >= max_stall:
            f = np.clip(f * (1 + 0.15 * rng.standard_normal(n)), 0, None)
            S2 = float(f.sum())
            f = f / S2 if S2 > 1e-10 else best_f.copy()
            t_cur = 0
            stall = 0

    return best_f, best_sc


def _grad_opt(f0, n, stop_time, seed):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel(), 0, None).copy()
    S = float(f.sum())
    if S < 1e-10:
        f = np.ones(n)
    else:
        f = f / S

    best_f = f.copy()
    best_sc = _score(f)
    lr = 0.03
    beta = 0.9
    momentum = np.zeros(n)
    no_imp = 0

    while time.monotonic() < stop_time - 1.0:
        b = _conv(f)
        M = float(b.max())
        S = float(f.sum())

        # Subgradient of M = max_k c_k
        # Use top-K active lags for smoother gradient
        topk = min(3, 2 * n - 1)
        idx = np.argpartition(b, -topk)[-topk:]
        weights = b[idx] - b[idx].min() + 1e-12
        weights /= weights.sum()

        grad_M = np.zeros(n)
        for w, k in zip(weights, idx):
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            # ∂c_k/∂f[i] = 2*f[k-i] for valid i
            for i in range(lo, hi):
                grad_M[i] += 2.0 * w * f[k - i]

        # Gradient of score = 2n*M/S^2
        # d(score)/df[i] = (2n/S^2)*grad_M[i] - (4nM/S^3)
        grad = (2.0 * n / (S * S)) * grad_M - (4.0 * n * M / (S * S * S))

        # Nesterov momentum
        y = f + beta * momentum
        momentum = beta * momentum - lr * ((2.0 * n / (S * S)) * _grad_M_from_conv(y, b, idx, weights, n) - (4.0 * n * b[np.argmax(b)] / (S * S * S)))
        f_new = y + momentum
        f_new = np.clip(f_new, 0, None)
        S_new = float(f_new.sum())
        if S_new < 1e-10:
            f_new = f.copy()
        else:
            f_new = f_new / S_new

        sc = _score(f_new)
        if sc < best_sc - 1e-14:
            f = f_new
            best_f = f.copy()
            best_sc = sc
            no_imp = 0
            lr = min(0.12, lr * 1.015)
        else:
            no_imp += 1
            lr = max(0.003, lr * 0.975)
            if no_imp > 400:
                f = np.clip(f * (1.0 + 0.06 * rng.standard_normal(n)), 0, None)
                S2 = float(f.sum())
                f = f / S2 if S2 > 1e-10 else best_f.copy()
                lr = 0.03
                beta = 0.85
                momentum = np.zeros(n)
                no_imp = 0

    return best_f, best_sc


def _grad_M_from_conv(f, b, idx, weights, n):
    grad_M = np.zeros(n)
    for w, k in zip(weights, idx):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for i in range(lo, hi):
            grad_M[i] += 2.0 * w * f[k - i]
    return grad_M


def _grad_opt_simple(f0, n, stop_time, seed):
    """Simpler gradient method without Nesterov issues."""
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel(), 0, None).copy()
    S = float(f.sum())
    if S < 1e-10:
        f = np.ones(n)
    else:
        f = f / S

    best_f = f.copy()
    best_sc = _score(f)
    lr = 0.04
    beta = 0.88
    momentum = np.zeros(n)
    no_imp = 0

    while time.monotonic() < stop_time - 1.0:
        b = _conv(f)
        M = float(b.max())
        S = float(f.sum())

        # Top-K for subgradient
        topk = min(4, 2 * n - 1)
        idx = np.argpartition(b, -topk)[-topk:]
        wts = b[idx]
        wts = wts - wts.min() + 1e-12
        wts /= wts.sum()

        # Compute subgradient of M
        gM = np.zeros(n)
        for w, k in zip(wts, idx):
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            valid_i = np.arange(lo, hi)
            valid_j = k - valid_i
            gM[valid_i] += 2.0 * w * f[valid_j]

        # Score gradient: d(2nM/S^2)/df = (2n/S^2)*gM - (4nM/S^3)
        score_grad = (2.0 * n / (S * S)) * gM - (4.0 * n * M / (S * S * S))

        # Remove mean to maintain sum=1 (projected gradient)
        score_grad -= score_grad.mean()

        # Momentum update
        momentum = beta * momentum - lr * score_grad
        fn = f + momentum
        fn = np.clip(fn, 0, None)
        S_new = float(fn.sum())
        if S_new < 1e-10:
            fn = f.copy()
        else:
            fn = fn / S_new

        sc = _score(fn)
        if sc < best_sc - 1e-14:
            f = fn
            best_f = f.copy()
            best_sc = sc
            no_imp = 0
            lr = min(0.15, lr * 1.02)
        else:
            no_imp += 1
            lr = max(0.004, lr * 0.98)
            if no_imp > 500:
                f = np.clip(f * (1 + 0.08 * rng.standard_normal(n)), 0, None)
                S2 = float(f.sum())
                f = f / S2 if S2 > 1e-10 else best_f.copy()
                lr = 0.04
                beta = 0.85
                momentum = np.zeros(n)
                no_imp = 0

    return best_f, best_sc


def _coord_opt(f0, n, stop_time, seed):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel(), 0, None).copy()
    S = float(f.sum())
    f = f / S if S > 1e-10 else np.ones(n)
    best_sc = _score(f)

    mults = (0.3, 0.5, 0.7, 0.85, 1.2, 1.35, 1.6, 2.0)
    no_pass = 0

    while time.monotonic() < stop_time - 2.0:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 2.0:
                break
            orig = f[i]
            if orig < 1e-14:
                # Try small positive
                f[i] = 0.001
                sc = _score(f)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    orig = f[i]
                    improved = True
                else:
                    f[i] = 0.0
                continue
            found = False
            for m in mults:
                f[i] = orig * m
                sc = _score(f)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    orig = f[i]
                    found = True
                    improved = True
            if not found:
                f[i] = orig
        # 2-coord
        for _ in range(30):
            if time.monotonic() > stop_time - 2.0:
                break
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            oi, oj = f[i], f[j]
            for mi in (0.5, 1.5, 2.0):
                for mj in (0.5, 1.5, 0.3):
                    f[i] = max(oi * mi, 1e-6) if oi > 1e-12 else 0.01
                    f[j] = max(oj * mj, 1e-6) if oj > 1e-12 else 0.01
                    sc = _score(f)
                    if sc < best_sc - 1e-14:
                        best_sc = sc
                        improved = True
            f[i], f[j] = oi, oj

        if not improved:
            no_pass += 1
            if no_pass >= 8:
                break
        else:
            no_pass = 0

    return f, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 22.0

    parent = _get_parent()
    n_par = len(parent)

    best_seq = parent.copy()
    try:
        best_score = _score(best_seq)
    except Exception:
        best_score = math.inf

    def try_update(cand):
        nonlocal best_seq, best_score
        if cand is None:
            return
        cand = np.clip(np.asarray(cand, dtype=np.float64).ravel(), 0, None)
        if cand.size == 0 or cand.sum() < 0.01:
            return
        try:
            sc = _score(cand)
            if math.isfinite(sc) and sc < best_score - 1e-15:
                best_seq = cand.copy()
                best_score = sc
        except Exception:
            pass

    # --- Phase 1: LP-based optimization with multiple initializations ---
    n = n_par
    inits = []
    inits.append(parent.copy())

    i_ar = np.arange(n, dtype=np.float64)
    # Cosine bump
    inits.append(0.5 + 0.5 * np.cos(2 * math.pi * i_ar / n))
    # Gaussian
    inits.append(np.exp(-0.5 * ((i_ar - n / 2) / (n * 0.25)) ** 2))
    # Multi-peak
    peaks = np.zeros(n)
    for frac in (0.2, 0.5, 0.8):
        peaks += np.exp(-0.5 * ((i_ar - n * frac) / (n * 0.12)) ** 2)
    inits.append(peaks + 0.02)
    # Random
    rng_init = np.random.default_rng(seed)
    inits.append(np.abs(rng_init.standard_normal(n)) + 0.1)
    # Parent + noise
    ns = 0.6 * max(1.0, float(np.mean(parent)))
    inits.append(np.abs(parent + rng_init.normal(0, ns, n)) + 0.01)

    lp_end = t0 + budget * 0.48
    per_init = max(25.0, (lp_end - t0 - 5) / len(inits))

    for idx, init in enumerate(inits):
        if time.monotonic() > lp_end - 5.0:
            break
        stop = min(lp_end - 3.0, time.monotonic() + per_init)
        try:
            res, sc = _lp_loop(init, n, stop, seed + 100 * (idx + 1))
            try_update(res)
        except Exception:
            pass

    # --- Phase 2: Gradient polish ---
    g1_end = min(deadline, t0 + budget * 0.68)
    if time.monotonic() < g1_end - 12.0:
        try:
            res, sc = _grad_opt_simple(best_seq, len(best_seq), g1_end, seed + 5000)
            try_update(res)
        except Exception:
            pass

    # --- Phase 3: Second LP round on best ---
    lp2_end = min(deadline, t0 + budget *  0.82)
    if time.monotonic() < lp2_end - 12.0:
        try:
            res, sc = _lp_loop(best_seq, len(best_seq), lp2_end, seed + 15000)
            try_update(res)
        except Exception:
            pass

    # --- Phase 4: Coordinate polish ---
    c_end = min(deadline, t0 + budget * 0.92)
    if time.monotonic() < c_end - 12.0:
        try:
            res, sc = _coord_opt(best_seq, len(best_seq), c_end, seed + 25000)
            try_update(res)
        except Exception:
            pass

    # --- Phase 5: Final gradient burst ---
    if time.monotonic() < deadline - 8.0:
        try:
            res, sc = _grad_opt_simple(best_seq, len(best_seq), deadline - 6.0, seed + 35000)
            try_update(res)

            pass

    # --- Final validation ---
    final = [float(max(0.0, min(1000.0, x))) for x in best_seq]

            return fb
    except Exception:
        pass

    return [1.0] * 400
