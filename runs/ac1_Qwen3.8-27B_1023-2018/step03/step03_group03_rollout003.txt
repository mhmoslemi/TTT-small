import numpy as np
import math
import time
from scipy.optimize import minimize, linprog


def _now():
    return time.monotonic()


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64), False
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s <= 1e-15:
        return np.ones(a.size, dtype=np.float64) / a.size, False
    return a / s, True


def _conv2(a, b):
    a = np.asarray(a, dtype=np.float64).ravel()
    b = np.asarray(b, dtype=np.float64).ravel()
    if a.size == 0 or b.size == 0:
        return np.array([0.0], dtype=np.float64)
    L = a.size + b.size - 1
    if L <= 6000:
        c = np.convolve(a, b)
    else:
        m = 1 << max(1, (L - 1).bit_length())
        fa = np.fft.rfft(a, m)
        fb = np.fft.rfft(b, m)
        c = np.fft.irfft(fa * fb, m)[:L]
    c = np.where(np.isfinite(c), c, 0.0)
    return np.maximum(c, 0.0)


def _conv(a):
    return _conv2(a, a)


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return math.inf
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _conv(a)
    if c.size == 0:
        return math.inf
    m = float(c.max())
    if not math.isfinite(m) or m < 0.0:
        return math.inf
    return float(2.0 * a.size * m / (s * s))


def _exact_score(seq):
    try:
        return float(evaluate_sequence([float(x) for x in seq]))
    except Exception:
        return math.inf


def _get_parent():
    h = globals().get("height_sequence_1", None)
    if h is None:
        p = np.ones(32, dtype=np.float64)
    else:
        p = np.asarray(h, dtype=np.float64).ravel()
        p = np.where(np.isfinite(p), p, 0.0)
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if p.size == 0 or s < 0.01:
            p = np.ones(32, dtype=np.float64)
        else:

def _resample(a, L):
    a = np.asarray(a, dtype=np.float64).ravel()
    L = int(L)
    if L <= 0 or a.size == 0:
        return None
    if a.size == L:
        return a.copy()
    if a.size == 1:
        return np.full(L, float(a[0]) / L, dtype=np.float64)
    old = np.linspace(0.0, a.size - 1, a.size)
    new = np.linspace(0.0, L - 1, L)
    b = np.interp(new, old, a)
    return _normalize(b)[0]


def _perturb(a, rng, sigma=0.03):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return a
    if n > 1:
        noise = rng.normal(size=n)
        k = max(3, n // 16)
        kernel = np.ones(k, dtype=np.float64) / float(k)
        noise = np.convolve(noise, kernel, mode="same")
        sd = float(noise.std())
        if sd > 1e-12:
            noise = (noise - float(noise.mean())) / sd
        else:
            noise[:] = 0.0
        q = a * np.exp(float(sigma) * noise)
    else:
        q = a.copy()
    q = np.clip(np.where(np.isfinite(q), q, 0.0), 0.0, 1000.0)
    s = float(q.sum())
    if s < 1e-12:
        return a.copy()
    return q / s


def _phase_perturb(a, rng, mode=1, alpha=0.25):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return a
    x = np.linspace(0.0, n - 1.0, n)
    ph = float(rng.uniform(0.0, 2.0 * np.pi))
    q = a * (1.0 + float(alpha) * np.cos(2.0 * np.pi * float(mode) * x / float(n) + ph))
    q = np.clip(np.where(np.isfinite(q), q, 0.0), 0.0, 1000.0)
    s = float(q.sum())
    if s < 1e-12:
        return a.copy()
    return q / s


def _tangent_direction(p, M, time_limit=8.0):
    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    if n == 0 or not math.isfinite(M) or M <= 0.0:
        return None
    if time_limit <= 0.2:
        return None
    L = 2 * n - 1
    A = np.zeros((L, n + 1), dtype=np.float64)
    A[:, n] = -1.0
    for j in range(n):
        A[j:j + n, j] = p
    b = np.zeros(L, dtype=np.float64)
    Aeq = np.zeros((1, n + 1), dtype=np.float64)
    Aeq[0, :n] = 1.0
    beq = np.array([1.0], dtype=np.float64)
    c = np.zeros(n + 1, dtype=np.float64)
    c[n] = 1.0
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(
            c,
            A_ub=A,
            b_ub=b,
            A_eq=Aeq,
            b_eq=beq,
            bounds=bounds,
            method="highs",
            options={"presolve": True, "time_limit": time_limit},
        )
    except Exception:
        return None
    if getattr(res, "x", None) is None:
        return None
    q = np.asarray(res.x[:n], dtype=np.float64)
    if not np.all(np.isfinite(q)):
        return None
    q = np.maximum(q, 0.0)
    s = float(q.sum())
    if s < 1e-12:
        return None
    return q / s


def _fw_line_search(p, q):
    p = np.asarray(p, dtype=np.float64).ravel()
    q = np.asarray(q, dtype=np.float64).ravel()
    n = p.size
    C = 2.0 * n
    pp = _conv(p)
    pq = _conv2(p, q)
    qq = _conv(q)

    def eval_t(ts):
        a = 1.0 - ts
        b = ts
        vals = (
            (a[:, None] * a[:, None]) * pp[None, :]
            + (2.0 * a[:, None] * b[:, None]) * pq[None, :]
            + (b[:, None] * b[:, None]) * qq[None, :]
        )
        return C * vals.max(axis=1)

    ts = np.linspace(0.0, 1.0, 128, dtype=np.float64)
    obj = eval_t(ts)
    k = int(np.argmin(obj))
    best_t = float(ts[k])
    best_obj = float(obj[k])

    width = (1.0 / ts.size) * 3.0
    for _ in range(3):
        lo = max(0.0, best_t - width)
        hi = min(1.0, best_t + width)
        if hi <= lo:
            break
        ts2 = np.linspace(lo, hi, 64, dtype=np.float64)
        obj2 = eval_t(ts2)
        k2 = int(np.argmin(obj2))
        t2 = float(ts2[k2])
        o2 = float(obj2[k2])
        if math.isfinite(o2) and o2 < best_obj:
            best_obj = o2
            best_t = t2
        width *= 0.32
    return best_t, best_obj


def _fw_refine(a, stop_time, max_iters=6, seed=0):
    a = np.asarray(a, dtype=np.float64).ravel()
    a, ok = _normalize(a)
    if not ok:
        return a, math.inf
    n = a.size
    best = a.copy()
    best_sc = _score(a)
    if not math.isfinite(best_sc):
        return best, best_sc

    for it in range(max_iters):
        if _now() > stop_time - 2.0:
            break
        c = _conv(a)
        M = float(c.max())
        if not math.isfinite(M) or M <= 0.0:
            break
        remaining = max(0.0, stop_time - _now())
        if remaining < 6.0:
            break
        lp_time = min(12.0, max(2.0, 0.7 * remaining / max(1, max_iters - it)))
        q = _tangent_direction(a, M, time_limit=lp_time)
        if q is None:
            break
        if float(np.linalg.norm(q - a)) < 1e-10:
            break
        t, obj = _fw_line_search(a, q)
        if not math.isfinite(obj):
            break
        if t <= 1e-12 or obj >= best_sc - 1e-12:
            break
        p_new = (1.0 - t) * a + t * q
        p_new = np.clip(np.where(np.isfinite(p_new), p_new, 0.0), 0.0, 1000.0)
        s = float(p_new.sum())
        if s < 1e-12:
            break
        p_new /= s
        sc = _score(p_new)
        if math.isfinite(sc) and sc < best_sc - 1e-12:
            a = p_new
            best = p_new.copy()
            best_sc = sc
        else:
            break
    return best, best_sc


def _softmax(y):
    y = np.asarray(y, dtype=np.float64).ravel()
    if y.size == 0:
        return np.array([1.0], dtype=np.float64)
    y = y - float(y.max())
    e = np.exp(y)
    s = float(e.sum())
    if s <= 1e-300:
        return np.ones(y.size, dtype=np.float64) / y.size
    return e / s


def _smooth_optimize(a0, stop_time, max_iter=35):
    a = np.asarray(a0, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return a, math.inf
    a, ok = _normalize(a)
    if not ok:
        return a, math.inf
    a = np.maximum(a, 1e-12)
    a = a / float(a.sum())
    best = a.copy()
    best_sc = _score(a)
    if not math.isfinite(best_sc):
        return best, best_sc

    y = np.log(a)
    for eta in (1.0e5, 1.0e6):
        if _now() > stop_time - 20.0:
            break

        def fun(y, _eta=eta):
            p = _softmax(y)
            c = _conv(p)
            m = float(c.max())
            diff = _eta * (c - m)
            e = np.exp(diff)
            logsum = math.log(max(1e-300, float(e.sum())))
            return m + logsum / _eta

        def jac(y, _eta=eta):
            p = _softmax(y)
            c = _conv(p)
            m = float(c.max())
            diff = _eta * (c - m)
            e = np.exp(diff)
            w = e / max(1e-300, float(e.sum()))
            w = 0.5 * (w + w[::-1])
            gp = 2.0 * np.correlate(w, p, mode="valid")
            gy = gp - p * float(np.dot(p, gp))
            gy = np.where(np.isfinite(gy), gy, 0.0)
            return gy

        def cb(xk):
            nonlocal best, best_sc
            p = _softmax(xk)
            sc = _score(p)
            if math.isfinite(sc) and sc < best_sc - 1e-15:
                best = p.copy()
                best_sc = sc
            return _now() > stop_time - 20.0

        try:
            res = minimize(
                fun,
                y,
                jac=jac,
                method="L-BFGS-B",
                options={
                    "maxiter": max_iter,
                    "ftol": 1e-11,
                    "gtol": 1e-8,
                    "maxls": 20,
                    "callback": cb,
                },
            )
            y = np.asarray(res.x, dtype=np.float64)
            p = _softmax(y)
            sc = _score(p)
            if math.isfinite(sc) and sc < best_sc - 1e-15:
                best = p.copy()
                best_sc = sc
        except Exception:
            break
    return best, best_sc


def _coordinate_pass(a, c, stop_time, local_only=False, rng=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    c = np.asarray(c, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return a, c, False
    L = 2 * n - 1
    C = 2.0 * n
    if rng is None:
        rng = np.random.default_rng(0)

    s = float(a.sum())
    if s <= 1e-15:
        a = np.ones(n, dtype=np.float64) / n
        c = _conv(a)
    else:
        inv = 1.0 / s
        a = a * inv
        c = c * (inv * inv)

    c = np.where(np.isfinite(c), c, 0.0)
    np.maximum(c, 0.0, out=c)
    M0 = float(c.max())
    current_score = C * M0
    if not math.isfinite(current_score):
        current_score = _score(a)

    sqrtM = math.sqrt(max(0.0, M0))
    x_hi = max(
        2.0,
        min(10.0, 10.0 * float(a.max()) + 1.0, 5.0 * sqrtM + 1.0),
    )

    if n <= 400:
        Gc = 96
    elif n <= 600:
        Gc = 72
    elif n <= 900:
        Gc = 52
    else:
        Gc = 40
    Gf = 48

    improved_any = False
    order = rng.permutation(n)

    for i in order:
        if _now() > stop_time:
            break

        v = float(a[i])
        start = i
        end = i + n
        idx = 2 * i - start
        segL = 2.0 * a[start:end]
        c_seg = c[start:end]

        m_out = 0.0
        if start > 0:
            m_out = max(m_out, float(c[:start].max()))
        if end < L:
            m_out = max(m_out, float(c[end:]))

        if local_only:
            width = max(0.02, 0.6 * v, 0.25 * sqrtM)
            lo = max(0.0, v - width)
            hi = max(lo + 1e-10, min(x_hi, v + width))
        else:
            lo = 0.0
            hi = x_hi

        if hi <= lo:
            hi = lo + 1e-8

        xs = np.linspace(lo, hi, Gc, dtype=np.float64)
        step = max(1e-8, (hi - lo) / max(1, Gc))
        centers = [v]
        if not local_only or Gc > 1:
            # add one or two coarse minima that are clearly promising
            for center in (sqrtM, 0.0, hi):
                if center is None:
                    continue
                center = float(center)
                wdt = max(0.02, 3.0 * step)
                cl = max(lo, center - wdt)
                ch = min(hi, center + wdt)
                if ch > cl:
                    xs = np.concatenate((xs, np.linspace(cl, ch, Gf // 2, dtype=np.float64)))
        xs = np.concatenate((xs, np.array([v, lo, hi], dtype=np.float64)))
        xs = np.clip(xs, lo, hi)

        def eval_xs(xvals):
            w = xvals - v
            den = 1.0 + w
            seg = c_seg[None, :] + w[:, None] * segL[None, :]
            seg[:, idx] += w * w
            mseg = seg.max(axis=1)
            m = np.maximum(mseg, m_out)
            with np.errstate(divide="ignore", invalid="ignore"):
                obj = C * m / (den * den)
            obj[den <= 1e-12] = math.inf
            return obj

        obj = eval_xs(xs)
        if obj.size == 0 or not np.any(np.isfinite(obj)):
            continue

        order_obj = np.argsort(obj)
        min_sep = max(0.02, (hi - lo) / 40.0)
        selected = [v]
        for oi in order_obj:
            ox = float(xs[oi])
            oo = float(obj[oi])
            if not math.isfinite(oo):
                continue
            if all(abs(ox - sx) > min_sep for sx in selected):
                if oo < current_score - 1e-13 or len(selected) < 2:
                    selected.append(ox)
            if len(selected) >= 3:
                break

        best_x = v
        best_obj = current_score

        for cx in selected:
            cx = float(np.clip(cx, lo, hi))
            wdt = max(0.015, 3.0 * step, 0.04 * max(1.0, cx))
            for _lvl in range(2):
                lo2 = max(lo, cx - wdt)
                hi2 = min(hi, cx + wdt)
                if hi2 <= lo2:
                    break
                fxs = np.linspace(lo2, hi2, Gf, dtype=np.float64)
                fok = eval_xs(fxs)
                if fok.size == 0 or not np.any(np.isfinite(fok)):
                    break
                k = int(np.argmin(fok))
                fx = float(fxs[k])
                fo = float(fok[k])
                if math.isfinite(fo) and fo < best_obj - 1e-15:
                    best_obj = fo
                    best_x = fx
                wdt *= 0.3

        if math.isfinite(best_obj) and best_obj < current_score - 1e-12:
            w = best_x - v
            if abs(w) > 1e-15:
                Snew = 1.0 + w
                if Snew > 1e-12:
                    a[i] = best_x
                    c[start:end] += w * segL
                    c[2 * i] += w * w
                    inv = 1.0 / Snew
                    a *= inv
                    c *= inv * inv
                    np.maximum(c, 0.0, out=c)
                    current_score = C * float(c.max())
                    if not math.isfinite(current_score):
                        current_score = best_obj
                    improved_any = True

    return a, c, improved_any


def _coordinate_refine(a0, stop_time, seed=0):
    a, ok = _normalize(a0)
    if not ok:
        return a0, math.inf
    n = a.size
    if n == 0:
        return a, math.inf
    best = a.copy()
    best_sc = _score(a)
    if not math.isfinite(best_sc):
        best_sc = math.inf

    rng = np.random.default_rng(seed)
    if n <= 512:
        global_sweeps = 3
    elif n <= 900:
        global_sweeps = 2
    else:
        global_sweeps = 1

    sweep = 0
    no_imp = 0
    restarts = 0
    max_restarts = 10

    while _now() < stop_time - 3.0 and sweep < 50000:
        c = _conv(a)
        local = sweep >= global_sweeps
        try:
            a, c, _imp = _coordinate_pass(
                a,
                c,
                stop_time - 2.0,
                local_only=local,
                rng=rng,
            )
        except Exception:
            break

        sc = float(2.0 * n * max(0.0, float(c.max())))
        if not math.isfinite(sc):
            sc = _score(a)

        if math.isfinite(sc) and sc < best_sc - 1e-14:
            best = a.copy()
            best_sc = sc
            no_imp = 0
            restarts = 0
        else:
            no_imp += 1
            if no_imp >= 2:
                if _now() < stop_time - 30.0 and restarts < max_restarts and n <= 900:
                    a = _perturb(best, rng, sigma=0.02 + 0.015 * restarts)
                    restarts += 1
                    no_imp = 0
                elif sweep < 3 and _now() < stop_time - 30.0:
                    a = best.copy()
                    no_imp = 0
                else:
                    break
        sweep += 1

    return best, best_sc


def _improve_candidate(a0, stop_time, seed=0, use_fw=True):
    a, ok = _normalize(a0)
    if not ok:
        return a0, math.inf
    n = a.size
    if n == 0:
        return a, math.inf
    best = a.copy()
    best_sc = _score(a)

    if use_fw and n <= 512 and stop_time - _now() > 70.0:
        fw_stop = min(stop_time - 35.0, _now() + max(15.0, min(90.0, 0.25 * max(1.0, stop_time - _now()))))
        try:
            p, sc2 = _fw_refine(a, fw_stop, max_iters=6, seed=seed + 7)
            if p is not None and math.isfinite(sc2) and sc2 < best_sc - 1e-13:
                a = np.asarray(p, dtype=np.float64).ravel()
                a, _ = _normalize(a)
                best = a.copy()
                best_sc = sc2
        except Exception:
            pass

    if n <= 450 and stop_time - _now() > 160.0:
        sm_stop = min(stop_time - 70.0, _now() + max(15.0, min(80.0, 0.18 * max(1.0, stop_time - _now()))))
        try:
            p, sc2 = _smooth_optimize(a, sm_stop, max_iter=30)
            if p is not None and math.isfinite(sc2) and sc2 < best_sc - 1e-13:
                a = np.asarray(p, dtype=np.float64).ravel()
                a, _ = _normalize(a)
                best = a.copy()
                best_sc = sc2
        except Exception:
            pass

    coord_stop = stop_time - 6.0
    if coord_stop > _now() + 8.0:
        try:
            a, sc2 = _coordinate_refine(a, coord_stop, seed=seed + 17)
            if a is not None and math.isfinite(sc2) and sc2 < best_sc - 1e-14:
                a = np.asarray(a, dtype=np.float64).ravel()
                a, _ = _normalize(a)
                best = a.copy()
                best_sc = sc2
        except Exception:
            pass

    return best, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = _now()
    try:
        budget = max(20.0, float(budget_s))
    except Exception:
        budget = 1300.0
    headroom = max(15.0, min(45.0, 0.045 * budget))
    hard_deadline = t0 + budget - headroom

    parent = _get_parent()
    fallback = parent.copy()
    if fallback.size == 0:
        fallback = np.ones(1, dtype=np.float64)
    fallback, _ = _normalize(fallback)

    best_seq = fallback.copy()
    best_score = _exact_score(best_seq)
    if not math.isfinite(best_score):
        best_score = math.inf

    def _update(seq):
        nonlocal best_seq, best_score
        seq = np.asarray(seq, dtype=np.float64).ravel()
        if seq.size == 0:
            return
        seq = np.where(np.isfinite(seq), seq, 0.0)
        seq = np.clip(seq, 0.0, 1000.0)
        s = float(seq.sum())
        if s < 0.01:
            return
        seq = seq / s
        exact = _exact_score(seq)
        if math.isfinite(exact) and exact < best_score - 1e-13:
            best_seq = seq.copy()
            best_score = exact

    _update(best_seq)

    if _now() > hard_deadline - 25.0:
        return [float(x) for x in best_seq]

    # Main parent-length intensive search.
    stop1 = min(hard_deadline - 20.0, t0 + max(120.0, 0.42 * budget))
    try:
        cand, _sc = _improve_candidate(parent, stop1, seed=101, use_fw=True)
        _update(cand)
    except Exception:
        pass

    # A longer length is often useful; start from the best shape found so far.
    if _now() < hard_deadline - 110.0:
        L2 = 512
        init2 = _resample(best_seq, L2)
        if init2 is not None:
            stop2 = min(hard_deadline - 20.0, _now() + min(420.0, max(90.0, 0.25 * budget)))
            try:
                cand, _sc = _improve_candidate(init2, stop2, seed=202, use_fw=True)
                _update(cand)
            except Exception:
                pass

    tried_lengths = {fallback.size, best_seq.size, 512}
    base_len = int(parent.size)
    candidate_lengths = [
        max(2, base_len - 1),
        base_len + 1,
        max(2, base_len - 5),
        base_len + 5,
        448,
        640,
        768,
        256,
        320,
        1024,
    ]
    candidate_lengths = [x for x in candidate_lengths if x > 1 and x not in tried_lengths]
    candidate_lengths.sort(key=lambda z: (abs(z - base_len), z))

    for L in candidate_lengths:
        if _now() > hard_deadline - 90.0:
            break
        if L in tried_lengths:
            continue
        tried_lengths.add(L)
        remaining = hard_deadline - _now()
        alloc = min(180.0, max(45.0, 0.16 * remaining))
        stopL = min(hard_deadline - 18.0, _now() + alloc)
        initL = _resample(best_seq, L)
        if initL is None:
            continue
        try:
            cand, _sc = _improve_candidate(initL, stopL, seed=5000 + int(L), use_fw=(L <= 512))
            _update(cand)
        except Exception:
            pass

    # Randomized local restarts on the current best length.
    if _now() < hard_deadline - 140.0:
        rng = np.random.default_rng(int(seed))
        perturbations = []
        for i in range(4):
            perturbations.append(_perturb(best_seq, rng, sigma=0.025 + 0.05 * i))
            perturbations.append(_phase_perturb(best_seq, rng, mode=i + 1, alpha=0.15 + 0.08 * i))
        for idx, pc in enumerate(perturbations):
            if _now() > hard_deadline - 75.0:
                break
            if pc is None or pc.size == 0:
                continue
            remaining = hard_deadline - _now()
            alloc = min(110.0, max(35.0, 0.12 * remaining))
            stopP = min(hard_deadline - 16.0, _now() + alloc)
            try:
                cand, _sc = _improve_candidate(pc, stopP, seed=9000 + idx, use_fw=False)
                _update(cand)
            except Exception:
                pass

    # Final squeeze on the best exact candidate.
    if _now() < hard_deadline - 70.0:
        try:
            cand, _sc = _improve_candidate(best_seq, hard_deadline - 10.0, seed=424242, use_fw=(best_seq.size <= 512))
            _update(cand)
        except Exception:
            pass

    # Final cleanup and validation.
    final = np.asarray(best_seq, dtype=np.float64).ravel()
    final = np.where(np.isfinite(final), final, 0.0)
    final = np.clip(final, 0.0, 1000.0)
    s = float(final.sum())
    if final.size == 0 or s < 0.01:
        final = fallback.copy()
    else:
        final = final / s

    final_score = _exact_score(final)
    if not math.isfinite(final_score):
        fb_score = _exact_score(fallback)
        if math.isfinite(fb_score):
            final = fallback.copy()
            final_score = fb_score
    if final.size == 0 or not math.isfinite(final_score):
        return [1.0]
    return [float(x) for x in final]
