        ps = 256.0
    parent = parent / ps
    
    n = parent.size
    
    def conv_self(a):
        """Compute self-convolution using FFT."""
        m = 2 * len(a) - 1
        L = next_fast_len(m)
        fa = rfft(a, L)
        c = irfft(fa * fa, L)[:m]
        return np.maximum(c, 0.0)
    
    def conv_cross(a, b):
        """Compute cross-convolution using FFT."""
        m = len(a) + len(b) - 1
        L = next_fast_len(m)
        c = irfft(rfft(a, L) * rfft(b, L), L)[:m]
        return np.maximum(c, 0.0)
    
    def score(a):
        """Compute the score: 2*n*max(conv) / sum^2."""
        s = a.sum()
        if s < 1e-12:
            return math.inf
        c = conv_self(a)
        return 2.0 * n * c.max() / (s * s)

        """Score for unit-sum sequence: 2*n*max(conv)."""
        c = conv_self(a)
        return 2.0 * n * c.max()

            if time.monotonic() > deadline - 40.0:

            for _ in range(n_iter_ls // len(beta_vals)):
                if time.monotonic() > deadline - 40.0:
                    break
                c = _conv_self(f)
                cm = c.max()
                if cm < 1e-15:
                    break
                # Softmax weights over the convolution
                xc = beta * (c - cm)
                xc = np.clip(xc, -50.0, 0.0)
                w = np.exp(xc)
                ws = w.sum()
                if ws < 1e-30:
                    break
                p = w / ws  # probability distribution over lags

                # Gradient: d/d a[j] of sum_k p[k] * conv(a,a)[k]
                # = sum_k p[k] * 2*a[k-j] (for 0<=k-j<n)
                # = 2 * (valid_conv(p, a))[j]
                # valid_conv(p, a)[j] = sum_{m: 0<=j+m<n, j+m<=2n-2} p[j+m]*a[m]
                # Use FFT: this is the "valid" part of conv(p, a_reversed)
                ar = a_rev = f[::-1].copy()
                cp = _conv_cross(p, ar)
                # valid correlation: sum_m p[j+m]*a[m] for m=0..n-1
                # cp[k] = sum_i p[i]*ar[k-i] = sum_i p[i]*a[n-1-(k-i)]
                # We want sum_m p[j+m]*a[m]. Let's use a different approach.
                # grad[j] = 2 * sum_{m=0}^{n-1} p[j+m] * a[m] * 1_{j+m <= 2n-2}
                # = 2 * (p * a)[valid]
                # Compute via FFT: full_conv = conv(p, a), then valid part
                # conv(p,a)[j] = sum_{m: 0<=m<len(a), 0<=j-m<len(p)} p[j-m]*a[m]
                # Hmm, this is getting confusing. Let me just do it directly for small n.
                # Actually for n=256, a direct approach is O(n^2) = 65K, fast enough.
                # grad[j] = 2*sum_{m: 0<=m<n, 0<=j+m<2n-1} p[j+m]*a[m]
                grad = np.zeros(n)
                for j in range(n):
                    # p indices: j+m for m=0..n-1, valid if j+m <= 2n-2
                    max_m = min(n, 2 * n - 1 - j)
                    if max_m > 0:
                        grad[j] = 2.0 * np.dot(p[j:j + max_m], f[:max_m])

                # Project gradient to tangent space (sum=0) for normalization
                mean_g = grad.mean()
                grad_tangent = grad - mean_g

                # Gradient descent with adaptive step
                lr = 0.3 / (n * (1.0 + beta / 100.0))
                f_new = f - lr * grad_tangent
                f_new = np.clip(f_new, 0, None)
                if f_new.sum() < 1e-12:
                    break
                f_new /= f_new.sum()
                sc_new = _score(f_new)
                if sc_new < _score(f) - 1e-12:
                    f = f_new
                    if sc_new < best_sc:
                        best = f.copy()
                        best_sc = sc_new

    # --- Coordinate descent polish ---
    if time.monotonic() < deadline - 15.0:
        f = best.copy()
        cur_sc = best_sc
        max_passes = 4
        multis = (0.0, 0.3, 0.5, 0.7, 0.9, 1.0, 1.1, 1.3, 1.6, 2.0, 3.0)
        for pnum in range(max_passes):
            if time.monotonic() > deadline - 15.0:
                break
            order = rng.permutation(n)
            improved = False
            for j in order:
                if time.monotonic() > deadline - 15.0:
                    break
                orig = f[j]
                base = orig if orig > 1e-15 else 1.0 / n
                best_val = orig
                best_local_sc = cur_sc
                for m in multis:
                    val = base * m
                    if val < 0 or val > 1000:
                        continue
                    f[j] = val
                    s = f.sum()
                    if s < 1e-12:
                        f[j] = orig
                        continue
                    sc = _score(f)
                    if sc < best_local_sc:
                        best_local_sc = sc
                        best_val = val
                f[j] = best_val
                s = f.sum()
                if s > 1e-12:
                    f /= s
                    cur_sc = _score(f)
                if cur_sc < best_sc - 1e-12:
                    best_sc = cur_sc
                    best = f.copy()
                    improved = True
            if not improved:
                break

    # --- Final verification and return ---
    result_arr = np.clip(best, 0.0, 1000.0)
    if result_arr.sum() < 0.01:
        result_arr = np.clip(parent, 0.0, 1000.0)
    if result_arr.sum() < 0.01:
        result_arr = np.ones(32) / 32.0 * 0.01

    result = [float(x) for x in result_arr]

    # Verify with exact evaluator
    try:
        sc_final = evaluate_sequence(result)
        if not math.isfinite(sc_final):
            # Fallback to parent
            fb = [float(x) for x in np.clip(parent, 0.0, 1000.0)]
            sc_fb = evaluate_sequence(fb)
            if math.isfinite(sc_fb):
                result = fb
    except Exception:
        pass

    return result
