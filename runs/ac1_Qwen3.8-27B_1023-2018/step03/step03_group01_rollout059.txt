```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _conv(a):
    n = len(a)
    F = next_fast_len(2 * n - 1)
    Af = rfft(a, F)
    c = irfft(Af * Af, F)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _conv(a)
    m = float(c.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_direction(f, M, top_k=80, mode="max"):
    n = len(f)
    c_f = _conv(f)
    if top_k < 2 * n - 1:
        idx = np.argpartition(c_f, -top_k)[-top_k:]
    else:
        idx = np.arange(2 * n - 1)

    A_full = _build_A(f)
    A = A_full[idx, :].copy()
    b = np.full(len(idx), M, dtype=np.float64)

    if mode == "max":
        c_obj = -np.ones(n)
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b,
                          bounds=[(0.0, None)] * n, method='highs',
                          options={'time_limit': 8, 'presolve': True})
            if res.status == 0 and res.x is not None:
                g = np.maximum(res.x, 0.0)
                if g.sum() > 1e-14:
                    return g / g.sum()
        except Exception:
            pass
    elif mode == "min_dot":
        # minimize sum(f_i * g_i) s.t. (f*g)_k <= M, sum(g)=1, g>=0
        A_eq = np.ones((1, n))
        b_eq = np.array([1.0])
        c_obj = f.copy()
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                          bounds=[(0.0, None)] * n, method='highs',
                          options={'time_limit': 8, 'presolve': True})
            if res.status == 0 and res.x is not None:
                g = np.maximum(res.x, 0.0)
                if g.sum() > 1e-14:
                    return g / g.sum()
        except Exception:
            pass
    return None


def _subgrad_at(a, c_conv, n_active=8):
    n = len(a)
    M = c_conv.max()
    thresh = M - 1e-10
    active = np.where(c_conv >= thresh)[0]
    if len(active) > n_active:
        active = np.argpartition(c_conv, -n_active)[-n_active:]
    grad = np.zeros(n)
    for k in active:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for i in range(lo, hi):
            j = k - i
            if 0 <= j < n:
                grad[i] += 2.0 * a[j]
    return grad, active


def _fista_polish(f0, n, stop_time, seed):
    rng = np.random.default_rng(seed)
    f = f0.copy()
    y = f.copy()
    t_ita = 1.0
    lr = 0.05
    best_f = f.copy()
    best_sc = _score(f)
    no_imp = 0

    for it in range(100000):
        if time.monotonic() > stop_time:
            break
        c_y = _conv(y)
        grad, active = _subgrad_at(y, c_y, n_active=10)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn

        y_new = y - lr * grad
        y_new = np.maximum(y_new, 0.0)
        ys = y_new.sum()
        if ys < 1e-12:
            y_new = f.copy()
        else:
            y_new /= ys

        sc = _score(y_new)
        if sc < best_sc - 1e-14:
            best_f = y_new.copy()
            best_sc = sc
            no_imp = 0
            lr = min(0.2, lr * 1.005)
        else:
            no_imp += 1
            if no_imp > 300:
                lr = max(0.005, lr * 0.9)
                no_imp = 0
            elif no_imp > 1500:
                y_new = np.maximum(f + rng.normal(0, 0.02, n), 0)
                f = y_new / y_new.sum()
                y = f.copy()
                lr = 0.05
                no_imp = 0
                break

        t_new = 0.5 * (1.0 + math.sqrt(1.0 + 4.0 * t_ita * t_ita))
        y = y_new + ((t_ita - 1.0) / t_new) * (y_new - y)
        y = np.maximum(y, 0.0)
        ys2 = y.sum()
        if ys2 > 1e-12:
            y /= ys2
        else:
            y = best_f.copy()
        t_ita = t_new

    return best_f, best_sc


def _block_polish(f0, n, stop_time, seed):
    rng = np.random.default_rng(seed)
    f = f0.copy()
    best_f = f.copy()
    best_sc = _score(f)
    block_sizes = [8, 16, 32]
    no_imp_total = 0

    while time.monotonic() < stop_time and no_imp_total < 80:
        B = block_sizes[len(block_sizes) // 2]
        n_blocks = (n + B - 1) // B
        improved = False

        for b in range(n_blocks):
            if time.monotonic() > stop_time:
                return best_f, best_sc
            lo = b * B
            hi = min(n, lo + B)
            orig = f[lo:hi].copy()

            # Try scaling
            for s in [0.3, 0.5, 0.7, 1.3, 1.5, 2.0, 3.0]:
                f[lo:hi] = np.maximum(orig * s, 0.0)
                fs = f.sum()
                if fs < 1e-10:
                    f[lo:hi] = orig
                    continue
                sc = _score(f)
                if sc < best_sc - 1e-13:
                    best_sc = sc
                    best_f = f.copy()
                    improved = True
                    orig = f[lo:hi].copy()
                    break
            f[lo:hi] = orig

            # Try replacing with small constant
            if sum(1 for x in orig if x < 1e-8) > len(orig) // 2:
                f[lo:hi] = np.full(hi - lo, 1e-4)
                fs = f.sum()
                if fs > 1e-10:
                    sc = _score(f)
                    if sc < best_sc - 1e-13:
                        best_sc = sc
                        best_f = f.copy()
                        improved = True
                f[lo:hi] = orig

            # Random block perturbation
            if not improved:
                pert = orig + rng.normal(0, 0.03 * max(1, orig.mean()), len(orig))
                f[lo:hi] = np.maximum(pert, 0.0)
                fs = f.sum()
                if fs > 1e-10:
                    sc = _score(f)
                    if sc < best_sc - 1e-13:
                        best_sc = sc
                        best_f = f.copy()
                        improved = True
                f[lo:hi] = orig

        if not improved:
            no_imp_total += 1
        else:
            no_imp_total = 0
            f = best_f.copy()

    return best_f, best_sc


def _coord_fine(f0, n, stop_time, seed):
    rng = np.random.default_rng(seed)
    f = f0.copy()
    best_sc = _score(f)
    no_imp = 0

    while time.monotonic() < stop_time and no_imp < 200:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time:
                return f, best_sc
            orig = f[i]
            for dt_frac in [0.5, 1.5, 0.7, 1.3, 3.0, 0.1]:
                f[i] = max(0.0, orig * dt_frac)
                if abs(f[i] - orig) < 1e-15:
                    continue
                sc = _score(f)
                if sc < best_sc - 1e-13:
                    best_sc = sc
                    improved = True
                    orig = f[i]
                    break
            f[i] = orig
            if orig > 1e-10:
                for dt in [0.1 * orig, -0.1 * orig, 0.05 * orig]:
                    f[i] = max(0.0, orig + dt)
                    sc = _score(f)
                    if sc < best_sc - 1e-13:
                        best_sc = sc
                        improved = True
                        orig = f[i]
                        break
            f[i] = orig
        if not improved:
            no_imp += 1
        else:
            no_imp = 0

    return f, best_sc


def _lp_loop(f0, n, stop_time, seed):
    rng = np.random.default_rng(seed)
    f = np.maximum(np.asarray(f0, dtype=np.float64).ravel(), 0.0).copy()
    fs = f.sum()
    if fs < 1e-10:
        f = np.ones(n) / n
    else:
        f /= fs
    best_f = f.copy()
    best_sc = _score(f)
    stall = 0
    it = 0

    while time.monotonic() < stop_time:
        it += 1
        c_f = _conv(f)
        M = float(c_f.max())
        if M < 1e-15:
            break

        # Try both LP modes
        g = _lp_direction(f, M, top_k=80, mode="max")
        if g is not None and np.linalg.norm(g - f) < 1e-5:
            g2 = _lp_direction(f, M, top_k=60, mode="min_dot")
            if g2 is not None:
                g = g2

        if g is None:
            stall += 4
            if stall > 60:
                break
            f = np.maximum(f + rng.normal(0, 0.03, n), 0.0)
            fs2 = f.sum()
            f = f / fs2 if fs2 > 1e-10 else best_f.copy()
            continue

        dist = np.linalg.norm(g - f)
        if dist < 1e-6:
            stall += 3
            if stall > 60:
                break
            f = np.maximum(f + rng.normal(0, 0.04, n), 0.0)
            fs2 = f.sum()
            f = f / fs2 if fs2 > 1e-10 else best_f.copy()
            continue

        # Aggressive line search
        s_f = f.sum()
        g_normed = g * (s_f / g.sum())
        best_t = None
        best_fn = None
        best_sc_t = best_sc

        for t in [1.0, 0.7, 0.5, 0.35, 0.2, 0.1, 0.05]:
            fn = (1.0 - t) * f + t * g_normed
            fn = np.maximum(fn, 0.0)
            fsn = fn.sum()
            if fsn < 1e-10:
                continue
            sc = _score(fn)
            if sc < best_sc_t:
                best_sc_t = sc
                best_t = t
                best_fn = fn.copy()

        if best_fn is not None and best_sc_t < best_sc - 1e-14:
            f = best_fn
            best_f = f.copy()
            best_sc = best_sc_t
            stall = 0
        else:
            # Try a small step in LP direction
            t_small = max(0.01, min(0.05, dist * 0.3))
            fn = (1.0 - t_small) * f + t_small * g_normed
            fn = np.maximum(fn, 0.0)
            fsn = fn.sum()
            if fsn > 1e-10:
                sc = _score(fn)
                if sc < best_sc - 1e-14:
                    f = fn
                    best_f = f.copy()
                    best_sc = sc
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1

            if stall > 30:
                f = np.maximum(f + rng.normal(0, 0.06, n), 0.0)
                fs2 = f.sum()
                f = f / fs2 if fs2 > 1e-10 else best_f.copy()
                stall = 0

        if best_sc < 1.5028:
            break

    return best_f, best_sc
```
