        i_end = min(n - 1, k)
        for j in range(i_start, i_end + 1):
            A[r, j] = f[k - j]
    return A


def _lp_solve(f, n, M, top_k=100, time_limit=5.0):
    """Solve: max sum(g) s.t. (f*g)_k <= M for top-k active constraints, g>=0."""
    b = _conv_full(f)
    sorted_idx = np.argsort(-b)
    K = min(top_k, 2 * n - 1)
    active = sorted_idx[:K]
    A = _build_A(f, n, active)
    c = -np.ones(n)
    b_ub = np.full(K, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method="highs",
                      options={"presolve": True, "time_limit": time_limit})
    except Exception:
        return None
    if res.status not in (0, 3) or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g_sum = float(g.sum())
    if g_sum < 1e-12:
        return None
    return g / g_sum


def _lp_optimize(f0, n, stop_time, seed, top_k=100):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel()[:n].copy(), 0, None)
    f_sum = float(f.sum())
    if f_sum < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / f_sum

    best_f = f.copy()
    best_score = _score(f)
    t = 0.25
    t_min = 0.03
    t_max = 0.7
    stall = 0
    max_stall = 80
    iter_count = 0
    top_k_current = top_k

    while time.monotonic() < stop_time - 2.0:
        M = float(_conv_full(f).max())

        g_dir = _lp_solve(f, n, M, top_k_current, time_limit=4.0)

        if g_dir is None:
            stall += 3
            if stall > max_stall:
                break
            f = np.clip(f + rng.normal(0, 0.06, n), 0, None)
            fs = f.sum()
            if fs > 1e-10:
                f = f / fs
            else:
                f = best_f.copy()
            continue

        dist = float(np.linalg.norm(g_dir - f))
        if dist < 1e-8:
            stall += 2
            if stall > max_stall:
                break
            # Structured perturbation based on autocorrelation structure
            b = _conv_full(f)
            peak_idx = int(np.argmax(b))
            # Perturb coordinates that contribute to the peak
            contrib_idx = [i for i in range(n) if 0 <= peak_idx - i < n]
            n_perturb = max(5, min(20, len(contrib_idx) // 2))
            perturb_idx = rng.choice(contrib_idx, size=min(n_perturb, len(contrib_idx)), replace=False)
            f = f.copy()
            for idx in perturb_idx:
                f[idx] *= (1.0 + rng.normal(0, 0.15))
            f = np.clip(f, 0, None)
            fs = f.sum()
            if fs > 1e-10:
                f = f / fs
            else:
                f = best_f.copy()
            continue

        # Adaptive line search
        improved = False
        for _ in range(20):
            f_new = (1.0 - t) * f + t * g_dir
            f_new = np.clip(f_new, 0, None)
            fs_new = float(f_new.sum())
            if fs_new < 1e-10:
                t = max(t_min, t * 0.5)
                continue
            sc = _score(f_new)
            if sc < best_score - 1e-15:
                f = f_new
                best_f = f.copy()
                best_score = sc
                improved = True
                t = min(t_max, t * 1.06)
                stall = 0
                break
            else:
                t = max(t_min, t * 0.5)

        if not improved:
            stall += 1
            # Adaptive: increase top_k if stuck
            if stall % 20 == 0 and top_k_current < 2 * n - 1:
                top_k_current = min(top_k_current + 50, 2 * n - 1)
            if stall >= max_stall:
                # Aggressive restart strategies
                s = stall % 5
                if s == 0:
                    f = np.clip(f + rng.normal(0, 0.15, n), 0, None)
                elif s == 1:
                    # Mirror perturbation
                    half = n // 2
                    f = f.copy()
                    f[:half] = (0.6 * f[:half] + 0.4 * f[half:half + half] + rng.normal(0, 0.05, half))
                    f = np.clip(f, 0, None)
                elif s == 2:
                    # Smooth noise
                    kern = np.exp(-0.5 * (np.arange(-6, 7) / 2.5) ** 2)
                    kern /= kern.sum()
                    smooth = np.convolve(rng.normal(0, 0.25, n), kern, 'same')
                    f = np.clip(f * (1 + smooth), 0, None)
                elif s == 3:
                    # Block swap
                    blk = n // 3
                    f = f.copy()
                    perm = [blk, 0, n - blk]
                    f = np.concatenate([f[perm[ii]:perm[ii]+blk] if ii < 2 else f[perm[ii-2]:perui+blk]
                                        for ii in range(3)]) if False else f
                    # Simpler: swap random blocks
                    a1, a2 = int(rng.integers(0, n - n//4)), int(rng.integers(0, n - n//4))
                    size = n // 4
                    f[a1:a1+size], f[a2:a2+size] = f[a2:a2+size].copy(), f[a1:a1+size].copy()
                else:
                    # Exponential decay perturbation
                    decay = np.exp(-np.linspace(0, 2, n))
                    f = np.clip(f * (1 + 0.2 * (rng.normal(0, 1, n) * decay)), 0, None)
                fs = f.sum()
                if fs > 1e-10:
                    f = f / fs
                else:
                    f = best_f.copy()
                t = 0.25
                top_k_current = top_k
                stall = 0
        iter_count += 1

    return best_f, best_score


def _peak_reduction_descent(a, stop_time, seed=100):
    """Targeted coordinate descent that reduces peak autocorrelation."""
    rng = np.random.default_rng(seed)
    best = np.clip(a.astype(np.float64), 0, None).copy()
    fs = best.sum()
    if fs < 1e-10:
        return best, _score(best)
    best = best / fs
    best_sc = _score(best)
    n = len(best)
    no_imp = 0
    max_no_imp = 8

    while time.monotonic() < stop_time - 2.0:
        improved = False
        b = _conv_full(best)
        top_peaks = np.argsort(-b)[:10]

        # Pass 1: Direct peak reduction
        for k_star in top_peaks[:5]:
            if time.monotonic() > stop_time - 2.0:
                break
            # Find contributors to conv at position k_star
            contribs = []
            for i in range(n):
                j = k_star - i
                if 0 <= j < n:
                    contribs.append((i, best[j]))
            contribs.sort(key=lambda x: -x[1])
            # Try reducing top contributors and redistributing
            for i_c, val_c in contribs[:8]:
                orig = best[i_c]
                for frac in (0.5, 0.7, 0.85, 0.3, 0.0):
                    best[i_c] = orig * frac
                    # Redistribute to non-contributing positions
                    redist = orig - orig * frac
                    if redist > 1e-12:
                        # Add to positions that don't contribute to k_star
                        safe_idx = [i for i in range(n) if not (0 <= k_star - i < n)]
                        if not safe_idx:
                            safe_idx = list(range(n))
                        target = int(rng.integers(0, len(safe_idx)))
                        best[safe_idx[target]] += redist
                    best = np.clip(best, 0, None)
                    if best.sum() < 0.01:
                        best[i_c] = orig
                        continue
                    sc = _score(best)
                    if sc < best_sc - 1e-15:
                        best_sc = sc
                        improved = True
                        break
                best[i_c] = orig  # Restore if no improvement found in inner loop

        # Pass 2: General coordinate descent on random subset
        order = rng.permutation(n)
        step_size = n // 2
        for i in order[:step_size]:
            if time.monotonic() > stop_time - 2.0:
                break
            orig = best[i]
            if orig < 1e-10:
                # Try small positive values
                for nv in (0.001, 0.005, 0.01, 0.02):
                    best[i] = nv
                    if best.sum() < 0.01:
                        best[i] = 0
                        continue
                    sc = _score(best)
                    if sc < best_sc - 1e-15:
                        best_sc = sc
                        improved = True
                        break
                best[i] = orig if not improved else best[i]
                continue
            for mlt in (0.4, 0.6, 0.8, 1.1, 1.3, 1.7, 0.0):
                best[i] = orig * mlt
                if best.sum() < 0.01:
                    best[i] = orig
                    continue
                sc = _score(best)
                if sc < best_sc - 1e-15:
                    best_sc = sc
                    improved = True
                    break
            else:
                best[i] = orig

        # Pass 3: Pair moves focused on peak
        for _ in range(40):
            if time.monotonic() > stop_time - 2.0:
                break
            k_star = int(top_peaks[rng.integers(min(5, len(top_peaks)))])
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            for ratio in (0.3, 0.5, 0.7, 1.4, 2.0, 1.0):
                ni = ratio * oi
                nj = oj + (oi - ni)
                if nj < 0:
                    continue
                best[i], best[j] = ni, nj
                if best.sum() < 0.01:
                    best[i], best[j] = oi, oj
                    continue
                sc = _score(best)
                if sc < best_sc - 1e-15:
                    best_sc = sc
                    improved = True
                    break
            best[i], best[j] = oi, oj

        if not improved:
            no_imp += 1
            if no_imp >= max_no_imp:
                break
        else:
            no_imp = 0

    return best, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 25.0

    parent = _get_parent()
    n = len(parent)

    # Initialize best with parent
    p = np.clip(parent, 0, 1000).copy()
    ps = p.sum()
    if ps < 0.01:
        p = np.ones(n) / n
    else:
        p = p / ps
    best_seq = p.copy()
    best_score = float(evaluate_sequence([float(x) for x in p]))
    if not math.isfinite(best_score):
        best_seq = np.ones(n) / n
        best_score = _score(best_seq)
    if not math.isfinite(best_score):
        return [1.0]

    rng_main = np.random.default_rng(seed)
    i_arr = np.arange(n, dtype=np.float64)

    # Generate diverse initializations
    inits = [parent.copy()]

    # Sine-modulated
    for freq in (1, 2, 3):
        phase = rng_main.uniform(0, 2 * math.pi)
        sig = 0.5 + 0.5 * np.sin(2 * math.pi * freq * i_arr / n + phase)
        sig = sig + 0.01
        sig /= sig.sum()
        inits.append(sig)

    # Gaussian bump + tails
    center = rng_main.integers(n // 4, 3 * n // 4)
    width = n // 6
    bump = np.exp(-0.5 * ((i_arr - center) / width) ** 2)
    bump = bump + 0.02
    bump /= bump.sum()
    inits.append(bump)

    # Alternating blocks
    block_size = max(2, n // 16)
    alt = np.ones(n)
    for b in range(0, n, block_size * 2):
        alt[b:b + block_size] *= 1.5
        alt[b + block_size:b + 2 * block_size] *= 0.5
    alt = np.clip(alt + 0.01, 0, None)
    alt /= alt.sum()
    inits.append(alt)

    # Parent with structured noise
    kern = np.exp(-0.5 * (np.arange(-5, 6) / 2.0) ** 2)
    kern /= kern.sum()
    smooth_noise = np.convolve(rng_main.normal(0, 1, n), kern, 'same')
    pn = np.clip(parent / max(parent.sum(), 1e-10) * (1 + 0.3 * smooth_noise), 0, None)
    pn /= pn.sum()
    inits.append(pn)

    # Phase allocation
    # 55% LP, 25% polish, 10% other lengths, 10% safety
    phase1_end = t0 + budget * 0.52
    phase2_end = t0 + budget * 0.75
    phase3_end = deadline

    # Phase 1: LP-based descent
    time_per_init = max(15.0, (phase1_end - t0) / len(inits))

    for idx, init in enumerate(inits):
        if time.monotonic() > phase1_end - 10.0:
            break
        stop = min(phase1_end - 5.0, time.monotonic() + time_per_init)
        if time.monotonic() >= stop:
            break
        init_n = init[:n].copy()
        init_s = init_n.sum()
        if init_s > 1e-10:
            init_n = init_n / init_s
        try:
            res, sc = _lp_optimize(init_n, n, stop, seed + 1000 * (idx + 1), top_k=80)
            if np.all(np.isfinite(res)) and res.sum() > 0.01:
                res_clipped = np.clip(res, 0, 1000)
                exact = float(evaluate_sequence([float(x) for x in res_clipped]))
                if math.isfinite(exact) and exact < best_score:
                    best_seq = res_clipped.copy()
                    best_score = exact
        except Exception:
            pass

        # Check remaining inits
        if best_score < 1.503:
            # Continue with remaining inits briefly
            time_per_init = max(8.0, (phase1_end - time.monotonic()) / max(1, len(inits) - idx - 1))

    # Phase 2: Polishing with peak reduction
    polish_budget = phase2_end - time.monotonic()
    if polish_budget > 10.0:
        n_polish = max(2, int(polish_budget / 60))
        for pidx in range(n_polish):
            p_stop = time.monotonic() + min(55.0, (phase2_end - time.monotonic()))
            if p_stop <= time.monotonic() + 5.0:
                break
            try:
                res, sc = _peak_reduction_descent(best_seq, p_stop, seed=5000 + pidx * 7)
                if np.all(np.isfinite(res)) and res.sum() > 0.01:
                    res_clipped = np.clip(res, 0, 1000)
                    exact = float(evaluate_sequence([float(x) for x in res_clipped]))
                    if math.isfinite(exact) and exact < best_score:
                        best_seq = res_clipped.copy()
                        best_score = exact
            except Exception:
                pass

    # Phase 3: Try nearby lengths
    if time.monotonic() < phase3_end - 30.0:
        for L in [n - 16, n + 16, n - 32, n + 32]:
            if time.monotonic() > phase3_end - 25.0:
                break
            if L < 16 or L > 2048:
                continue
            time_per_L = max(5.0, min(25.0, (phase3_end - 20.0) - time.monotonic()))
            stop_L = time.monotonic() + time_per_L
            if stop_L > phase3_end - 15.0:
                stop_L = phase3_end - 15.0
            # Interpolate parent to new length
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, L)
            init_L = np.interp(x_new, x_old, parent)
            s_L = init_L.sum()
            if s_L > 1e-10:
                init_L = init_L / s_L
            else:
                init_L = np.ones(L) / L
            try:
                res, sc = _lp_optimize(init_L, L, stop_L, seed + 99983 + L, top_k=60)
                if np.all(np.isfinite(res)) and res.sum() > 0.01:
                    res_clipped = np.clip(res, 0, 1000)
                    exact = float(evaluate_sequence([float(x) for x in res_clipped]))
                    if math.isfinite(exact) and exact < best_score:
                        best_seq = res_clipped.copy()
                        best_score = exact
            except Exception:
                pass

    # Final polish
    final_stop = t0 + budget - 8.0
    if time.monotonic() < final_stop - 15.0:
        try:
            best_seq, best_score = _peak_reduction_descent(
                best_seq, final_stop, seed=777777)
        except Exception:
            pass

    # Final validation
    final_seq = np.clip(best_seq, 0, 1000)
    try:
        final_sc = float(evaluate_sequence([float(x) for x in final_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc) or final_sc > best_score + 0.01:
        # Fall back to parent
        try:
            pb = np.clip(parent, 0, 1000)
            fb = float(evaluate_sequence([float(x) for x in pb]))
            if math.isfinite(fb):
                return [float(x) for x in pb]
        except Exception:
            pass
        # Last resort
        return [float(x) for x in (np.ones(n) / n)]

    return [float(x) for x in final_seq]
