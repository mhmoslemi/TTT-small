Let me think about what the optimal structure looks like for large n. We want the convolution to be as flat as possible. In the Fourier domain, this means |A(f)|^2 should be as flat as possible.

For a nonnegative sequence, A(f) = sum a[i] e^{-2πifi/n}. The condition |A(f)|^2 = constant means the sequence is a "flat spectrum" sequence. For nonnegative sequences, this is related to the "Shannon" or "Wiener" time-frequency uncertainty principle.

The most "flat spectrum" nonnegative sequences are approximately Gaussian (by the uncertainty principle, Gaussian minimizes time-frequency concentration). But a Gaussian has a peaked autocorrelation, giving score ≈ 2.

```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(198, dtype=np.float64)
    if p.size == 0:
        p = np.ones(198, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    return p


def _conv(a):
    n = len(a)
    size = next_fast_len(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv(a)
    m = float(np.max(b))
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_conv_matrix(f, n):
    """Build the convolution matrix A where (A@x)[k] = (f*x)[k]."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_solve(f, n, M, timeout=5.0):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    A = _build_conv_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs',
                      options={'presolve': True, 'time_limit': timeout,
                               'dual_feasibility_tolerance': 1e-10})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.clip(np.asarray(res.x, dtype=np.float64), 0.0, 1000.0)
    if g.sum() < 1e-15:
        return None
    return g


def _lp_iterate(f0, n, stop_time, seed):
    """Main LP-based optimization loop."""
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel()[:n].copy(), 0, None)
    if f.size < n:
        f = np.concatenate([f, np.zeros(n - f.size)])
    if f.sum() < 1e-10:
        f = np.ones(n) / n

    best_f = f.copy()
    best_sc = _score(f)

    t = 0.3
    stall = 0
    max_stall = 150
    m_factor = 0.998  # target M as fraction of current peak

    while time.monotonic() < stop_time - 3.0:
        b = _conv(f)
        M = float(np.max(b))
        if M < 1e-18:
            break

        # Try tighter M first
        M_tight = M * m_factor
        g = _lp_solve(f, n, M_tight, timeout=4.0)

        if g is None:
            g = _lp_solve(f, n, M, timeout=4.0)

        if g is None:
            stall += 5
            if stall > max_stall:
                break
            # Perturb and retry
            f = np.clip(f + rng.normal(0, 0.02 * f.mean(), n), 0, None)
            continue

        g_sum = float(g.sum())
        f_sum = float(f.sum())
        g = g * (f_sum / g_sum)

        # Convergence check
        rel_dist = float(np.linalg.norm(g - f)) / max(1.0, float(np.linalg.norm(f)))
        if rel_dist < 1e-7:
            stall += 2
            if stall > max_stall:
                break
            # Try even tighter M
            m_factor = max(0.99, m_factor - 0.002)
            continue

        # Line search
        improved = False
        for _ in range(12):
            f_new = (1.0 - t) * f + t * g
            f_new = np.clip(f_new, 0, None)
            if f_new.sum() < 0.01:
                t = max(0.05, t * 0.5)
                continue
            sc = _score(f_new)
            if sc < best_sc - 1e-12:
                f = f_new
                best_f = f.copy()
                best_sc = sc
                improved = True
                t = min(0.85, t * 1.2)
                stall = 0
                m_factor = 0.998  # reset
                break
            else:
                t = max(0.05, t * 0.5)

        if not improved:
            stall += 1
            if stall >= max_stall:
                # Escape: perturbation
                sigma = 0.1 * max(1.0, f.mean())
                f = np.clip(f + rng.normal(0, sigma, n), 0, None)
                if f.sum() > 1e-10:
                    f = f * (best_f.sum() / f.sum())
                t = 0.3
                m_factor = 0.998
                stall = 0
                max_stall += 100
                if max_stall > 1000:
                    break

    return best_f, best_sc


def _project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _grad_polish(a, stop_time, seed, max_iters=5000):
    """Gradient-based polishing with log-sum-exp smoothing."""
    rng = np.random.default_rng(seed)
    n = len(a)
    a = np.clip(np.asarray(a, dtype=np.float64).ravel().copy(), 0, None)
    S = a.sum()
    if S < 1e-10:
        return a, _score(a)
    a = a / S

    best_a = a.copy()
    best_sc = _score(a)

    epsilon = 0.05
    lr = 0.05
    no_imp = 0

    for it in range(max_iters):
        if time.monotonic() > stop_time - 3.0:
            break

        q = _conv(a)
        max_q = float(np.max(q))

        # Log-sum-exp weights
        shifted = (q - max_q) / epsilon
        shifted = np.clip(shifted, -60, 0)
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w /= w_sum

        # Use top-15 weights for gradient
        top_idx = np.argsort(w)[-15:]
        w_top = w[top_idx]

        # Weighted gradient
        grad = np.zeros(n)
        for k_idx in top_idx:
            k = int(k_idx)
            w_k = w_top[list(top_idx).index(k_idx)]
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                grad[lo:hi + 1] += w_k * 2.0 * a[k - hi:k - lo + 1][::-1]

        # Step
        a_new = a - lr * grad
        a_new = _project_simplex(a_new)

        sc = _score(a_new)
        if sc < best_sc - 1e-12:
            a = a_new.copy()
            best_a = a.copy()
            best_sc = sc
            no_imp = 0
            epsilon = max(0.005, epsilon * 0.995)
            lr = min(0.2, lr * 1.005)
        else:
            no_imp += 1
            lr = max(0.005, lr * 0.95)
            if no_imp > 200:
                # Reset with perturbation
                a = _project_simplex(a + rng.normal(0, 0.01, n))
                no_imp = 0
                lr = 0.05
                epsilon = 0.05

    return best_a, best_sc


def _coord_polish(a, stop_time, seed):
    """Coordinate descent polishing."""
    rng = np.random.default_rng(seed)
    n = len(a)
    best = np.clip(np.asarray(a, dtype=np.float64).ravel(), 0, None).copy()
    best_sc = _score(best)

    sweeps = 0
    max_sweeps = 10

    while time.monotonic() < stop_time - 3.0 and sweeps < max_sweeps:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 3.0:
                break
            orig = best[i]
            best_val = orig
            best_val_sc = best_sc

            if orig > 1e-15:
                candidates = [orig * m for m in (0.2, 0.4, 0.6, 0.8, 1.2, 1.5, 2.0, 0.0)]
            else:
                mean_val = max(1e-3, best[best > 0].mean() if (best > 0).any() else 1.0)
                candidates = [0.0, 0.1 * mean_val, 0.3 * mean_val, 0.5 * mean_val, 1.0 * mean_val]
