def _convolve(a, b=None):
    if b is None:
        b = a
    # Use direct for small n
    return np.convolve(a, b, mode='full')

def _score(a):
    """Compute the AC1 score."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = np.convolve(a, a)
    return 2.0 * a.size * c.max() / (s * s)

def _build_conv_matrix(a):
    """Build matrix A where (A@x)_k = (a*x)_k for full convolution."""
    n = len(a)
    m = 2 * n - 1

    for k in range(m):

            idx = k - j
            if 0 <= idx < n:
                A[k, j] = a[idx]
    return A

                      method='highs',

            inner = np.dot(grad_x, x)
            grad_t = x * (grad_x - inner)
            return grad_t

            pass

The smoothed objective: $F_\epsilon(x) = \epsilon \log \sum_k \exp((x*x)_k / \epsilon)$

With the softmax parameterization $x_j = e^{t_j} / \sum e^{t_k}$:

$G_\epsilon(t) = F_\epsilon(\text{softmax}(t))$

This is smooth and we can use L-BFGS. The gradient is:
$\frac{\partial G_\epsilon}{\partial t_i} = \sum_j \frac{\partial F_\epsilon}{\partial x_j} \cdot \frac{\partial x_j}{\partial t_i}$

$= \sum_j \left(\frac{2}{\epsilon} \sum_k w_k x_{k-j}\right) \cdot x_j (\delta_{ji} - x_i)$
