def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2*n-1]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)

def _build_conv_matrix(f, n):
    """Build the (2n-1) x n matrix A where A[k,j] = f[k-j] if valid."""
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = f
    return A

def _lp_find_g(f, n, M, time_limit=3.0):
    """Find g that maximizes sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    A = _build_conv_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                     bounds=[(0, None)]*n,
                     method='highs',
                     options={'time_limit': time_limit})

        return None

        return None
