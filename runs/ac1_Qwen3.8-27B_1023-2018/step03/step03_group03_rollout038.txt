def _lp_direction(f, n, time_limit=5.0):
    """Find LP-based improvement direction."""
    b = np.convolve(f, f)
    M = float(b.max())
    if M < 1e-15:
        return None

    # Build constraint matrix: (f * g)[k] <= M for all k
    # (f*g)[k] = sum_j f[k-j] * g[j] for valid j
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f

    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method="highs",
                      options={"presolve": True, "time_limit": time_limit})
    except Exception:
        return None

    if res.status != 0 or res.x is None:
        return None

    g = np.asarray(res.x, dtype=np.float64)
    g_sum = float(g.sum())
    if g_sum < 1e-15:
        return None
    return g / g_sum


def _lp_step(f, g_dir, t):
    """Take a step towards LP direction."""
    new = (1.0 - t) * f + t * g_dir
    new = np.clip(new, 0, None)
    s = float(new.sum())
    if s < 0.01:
        return None
    return new / s

    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        return [1.0] * 32

    return [float(x) for x in best_seq]
