import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _get_parent():

def _conv(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    return irfft(F * F, fsize)[:2 * n - 1]


def _sc(a, b):
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * len(a) * m / (s * s)


def _build_A(f, n):
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_solve(f, n, M, symm, tl=4.0):
    A = _build_A(f, n)
    if symm:
        nh = n // 2
        As = A[:, :nh] + A[:, n - 1:n - 1 - nh:-1]
        c = -np.ones(nh)
        bounds = [(0, None)] * nh
    else:
        As = A
        c = -np.ones(n)
        bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=As, b_ub=np.full(2 * n - 1, M),
                      bounds=bounds, method='highs',
                      options={'time_limit': tl, 'presolve': True})
    except Exception:
        return None
    if res.status not in (0, 3) or res.x is None:
        return None
    x = np.maximum(np.asarray(res.x, dtype=np.float64), 0.0)
    if symm:
        nh = n // 2
        g = np.zeros(n)
        g[:nh] = x
        g[nh:] = x[::-1]
    else:
        g = x
    gs = float(g.sum())
    if gs < 1e-12:
        return None
    return g / gs

def _polish(a, stop_time, rng):
    n = len(a)
    best = np.maximum(np.asarray(a, dtype=np.float64).ravel(), 0.0)
    bs = best.sum()
    if bs < 1e-10:
        return best, _sc(best, _conv(best))
    best = best / bs
    best_b = _conv(best)
    best_sc = _sc(best, best_b)
    no_imp = 0
    max_no_imp = 6

    while time.monotonic() < stop_time - 2.0 and no_imp < max_no_imp:
        improved = False

        # 1D coordinate descent
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 2.0:
                break
            orig = best[i]
            if orig > 1e-8:
                mults = (0.4, 0.6, 0.8, 1.1, 1.3, 1.6, 2.2, 0.0, 0.05, 0.3)
            else:
                mults = (0.01, 0.05, 0.1, 0.3, 1.0)
            found = False
            for mlt in mults:
                best[i] = orig * mlt if orig > 1e-8 else mlt
                s = best.sum()
                if s < 0.01:
                    best[i] = orig
                    continue
                sc = _sc(best, _conv(best))
                if sc < best_sc - 1e-15:
                    best_sc = sc
                    improved = True
                    found = True
                    break
            if not found:
                best[i] = orig

        # Pair moves
        n_pairs = min(80, n)
        for _ in range(n_pairs):
            if time.monotonic() > stop_time - 2.0:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            tot = oi + oj
            if tot < 1e-12:
                continue
            found = False
            for ratio in (0.1, 0.3, 0.5, 0.7, 0.9):
                best[i] = ratio * tot
                best[j] = (1 - ratio) * tot
                sc = _sc(best, _conv(best))

                    best_sc = sc
                    improved = True
                    found = True
                    break
            if not found:
                best[i], best[j] = oi, oj

        if not improved:
            no_imp += 1
        else:
            no_imp = 0

    return best, best_sc
