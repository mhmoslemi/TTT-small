    if p.size == 0:
        p = np.ones(198, dtype=np.float64)
    return p


def _cconv(a):
    n = len(a)
    fs = next_fast_len(2 * n - 1)
    F = rfft(a, fs)
    return irfft(F * F, fs)[:2 * n - 1]


def _score(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return math.inf
    b = _cconv(a)
    m = b.max()
    return 2.0 * n * m / (s * s)


def _eval(seq):
    try:
        v = float(evaluate_sequence([float(x) for x in seq]))
        return v if math.isfinite(v) else math.inf
    except Exception:
        return math.inf


def _lp_direction(f, n, M=None, time_limit=30.0):
    if M is None:
        b = _cconv(f)
        M = float(b.max())
    # Build A: (2n-1) x n, A[k,j] = f[k-j] if 0<=k-j<n
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j+n, j] = f
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n,
                      method='highs', options={'presolve': True, 'time_limit': time_limit})
    except Exception:
        return None
    if res.x is None:
        return None
    g = np.asarray(res.x)
    gsum = g.sum()
    if gsum < 1e-12:
        return None
    return g / gsum


def _lp_optimize(f0, n, stop_time, seed, max_iters=5000):
    rng = np.random.default_rng(seed)
    f = np.clip(f0.copy(), 0, None)
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / s
    best = f.copy()
    best_sc = _score(f)
    t = 0.2
    stall = 0
    it = 0
    
    while time.monotonic() < stop_time - 2 and it < max_iters:
        b = _cconv(f)
        M = float(b.max())
        
        # LP direction
        d = _lp_direction(f, n, M, time_limit=10.0)
        
        if d is None or np.linalg.norm(d - f) < 1e-10:
            stall += 5
            if stall > 100:
                break
            f = np.clip(f + rng.normal(0, 0.05, n), 0, None)
            fs = f.sum()
            if fs > 1e-10: f /= fs
            else: f = best.copy()
            it += 1
            continue
        
        # Line search
        improved = False
        for _ in range(15):
            f_new = np.clip((1-t)*f + t*d, 0, None)
            fs = f_new.sum()
            if fs < 1e-10:
                t *= 0.5
                continue
            sc = _score(f_new)
            if sc < best_sc - 1e-12:
                f = f_new
                best = f.copy()
                best_sc = sc
                t = min(0.8, t * 1.1)
                stall = 0
                improved = True
                break
            else:
                t = max(0.01, t * 0.5)
        
        if not improved:
            stall += 1
            if stall > 80:
                break
        
        it += 1
    
    return best, best_sc
