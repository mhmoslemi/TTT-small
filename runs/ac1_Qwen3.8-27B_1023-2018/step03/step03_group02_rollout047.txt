def _optimize_coord_i(a, i, n):
    """Optimize a[i] with others fixed. Returns optimal value."""
    c_base = _conv(a)  # current convolution with a[i] = current value
    # Actually, let's compute the base (without a[i]'s contribution)
    a_zero = a.copy()

    # = c_base[k] + a[i] * (a[k-i] if 0<=k-i<n) * (1 + 1 if k-i != i else 0)
    
    def h(x):
        # Compute max convolution value with a[i] = x
        max_val = c_base.max()  # lower bound
        for k in range(2*n - 1):
            j = k - i
            contrib = 0
            if 0 <= j < n:
                if j == i:
                    contrib = x * x
                else:
                    contrib = x * a_zero[j] + a_zero[j] * x  # = 2*x*a_zero[j]
                    # Wait, a_zero[j] = a[j] since j != i
                    contrib = 2 * x * a[j]
            val = c_base[k] + contrib
            if val > max_val:

        return max_val
    
    # Minimize h(x) for x in [0, a_max]
    # h is convex (max of convex functions - actually max of affine and one quadratic)
    # Use bisection on subgradient or golden section
    
    # Golden section
    gr = (math.sqrt(5) - 1) / 2  # golden ratio
    lo, hi = 0.0, a[i] * 3  # search range
    for _ in range(50):
        x1 = hi - gr * (hi - lo)
        x2 = lo + gr * (hi - lo)
        if h(x1) < h(x2):
            hi = x2
        else:
            lo = x1
    return (lo + hi) / 2

def _conv(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)

def _score(a):

    m = float(b.max())

def _build_A(f, n, row_indices=None):

        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end >= j_start:
            A[ii, j_start:j_end + 1] = f[j_start:j_end + 1]
    return A


def _lp_direction(f, n, M, top_k=300, time_limit=5.0):

        if res.status in (0, 3) and res.x is not None:

def _subgrad_step(f, n, lr, frac=0.08):

    M = float(b.max())
    thresh = M * (1.0 - frac)

    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    grad = np.zeros(n)
    for k in active:
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end >= j_start:
            # grad[i] += 2 * f[k-i] for i in [j_start, j_end]
            indices = np.arange(j_start, j_end + 1)
            paired = k - indices
            grad[indices] += 2.0 * f[paired]
    gn = float(np.linalg.norm(grad))
    if gn < 1e-14:

    f_new = np.maximum(f_new, 0.0)
    fs = f_new.sum()
    if fs > 1e-12:

    return f_new
