        if res.status not in (0, 3) or res.x is None:
            stall += 3
            if stall > max_stall:
                break
            continue
        
        g = np.clip(np.asarray(res.x, dtype=np.float64), 0, None)
        g_sum = float(g.sum())
        if g_sum < 1e-15:
            stall += 3
            continue
        g *= a.sum() / g_sum
        
        dist = np.linalg.norm(g - a) / max(1.0, np.linalg.norm(a))
        if dist < 1e-8:
            stall += 2
            if stall > max_stall:
                break
            continue
        
        # Line search
        improved = False
        tt = t
        for _ in range(15):
            a_new = np.clip((1 - tt) * a + tt * g, 0, None)
            if a_new.sum() < 0.01:
                tt *= 0.5
                continue
            sc = _score(a_new)
            if sc < best_sc - 1e-12:
                a = a_new
                best_a = a.copy()
                best_sc = sc
                improved = True
                tt = min(0.8, tt * 1.15)
            else:
                tt *= 0.5
            if tt < 0.01:
                break
        
        if improved:
            stall = 0
        else:
            stall += 1
            if stall >= max_stall:
                a = np.clip(a + rng.normal(0, 0.03 * max(1.0, a.mean()), n), 0, None)
                stall = 0
                max_stall += 10
    
    return best_a, best_sc
