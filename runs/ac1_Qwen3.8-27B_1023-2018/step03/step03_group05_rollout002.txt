def _clean_nonneg(a):
    arr = np.asarray(a, dtype=np.float64).ravel()
    if arr.size == 0:
        arr = np.ones(1, dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    return arr


def _normalize(a, target=1.0):
    arr = _clean_nonneg(a)
    n = arr.size
    s = float(arr.sum())
    if s <= 1e-300:
        return np.full(n, float(target) / max(1, n), dtype=np.float64)
    return (arr / s) * float(target)


def _project_simplex(v, target=1.0):
    v = np.asarray(v, dtype=np.float64).ravel()
    n = v.size
    if n == 0:
        return np.ones(1, dtype=np.float64) * target
    v = np.where(np.isfinite(v), v, 0.0)
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    with np.errstate(divide="ignore", invalid="ignore"):
        cond = (css - target) / idx < u
    rho = np.nonzero(cond)[0]
    if rho.size == 0:
        return np.full(n, target / n, dtype=np.float64)
    theta = (css[rho[-1]] - target) / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    s = float(w.sum())
    if s <= 1e-300 or not np.isfinite(s):
        return np.full(n, target / n, dtype=np.float64)
    return (w / s) * target


def _conv_full_fft(a):
    arr = np.asarray(a, dtype=np.float64).ravel()
    n = arr.size
    if n == 0:
        return np.array([0.0], dtype=np.float64)
    need = 2 * n - 1
    L = next_fast_len(max(1, need))
    f = rfft(arr, L)
    c = irfft(f * f, L)[:need]
    c = np.asarray(c, dtype=np.float64)
    c = np.where(np.isfinite(c), c, 0.0)
    np.maximum(c, 0.0, out=c)
    return c

            else:

            idx = np.argpartition(c, m - top_k)[m - top_k:]
        else:
            idx = np.arange(m)
        A = _build_A(a_safe)
        AT = np.ascontiguousarray(A.T[:, idx])
        tie = rng.standard_normal(idx.size).astype(np.float64)
        if tie.size > 1:
            tie = (tie - tie.mean()) / (np.std(tie) + 1e-12)
        c_obj = np.ones(idx.size, dtype=np.float64) + 1e-6 * tie
        A_ub = -AT
        b_ub = -np.ones(n, dtype=np.float64)
        tl = max(0.5, min(5.0, 0.25 * rem))
        res = _linprog_wrap(c_obj, A_ub, b_ub, tl)
        if res is None:
            return None
        x = np.asarray(res.x, dtype=np.float64).ravel()
        if x.size != idx.size:
            return None
        x = np.where(np.isfinite(x), x, 0.0)
        x = np.maximum(x, 0.0)
        if x.sum() <= 1e-12:
            return None
        if (not getattr(res, "success", False)):
            if idx.size <= 1200:
                viol = float(np.min(AT.T @ x - 1.0))
                if not np.isfinite(viol) or viol < -1e-5:
                    return None
        w = np.zeros(m, dtype=np.float64)
        w[idx] = x
        s = float(w.sum())
        if s > 0 and np.isfinite(s):
            return w / s
        return None
    except Exception:
        return None


def _gradient_phase(start, deadline, rng, on_candidate, max_patience=30):
    a = _normalize(start)
    n = int(a.size)
    if n < 2:
        return a
    if on_candidate is not None:
        on_candidate(a)
    c = _conv_full_fft(a)
    best_a = a.copy()
    best_sc = 2.0 * n * float(c.max())
    no_improve = 0
    it = 0
    while time.monotonic() < deadline:
        c = _conv_full_fft(a)
        M = float(c.max())
        sc = 2.0 * n * M
        if np.isfinite(sc) and sc < best_sc:
            best_sc = float(sc)
            best_a = a.copy()
            if on_candidate is not None:
                on_candidate(best_a)
        q = _make_weights(c, n, frac=0.08, use_top=(n <= 1200))
        grad = _weighted_grad_fft(a, q)
        if not np.all(np.isfinite(grad)):
            grad = np.zeros(n, dtype=np.float64)
        best_cand = None
        best_cand_sc = best_sc * (1.0 - 1e-13)
        gmax = float(np.max(grad)) if grad.size else 0.0
        if gmax <= 1e-15:
            gmax = 1e-15
        etas = (0.05, 0.1, 0.25, 0.5, 1.0, 2.0)
        for eta in etas:
            if time.monotonic() >= deadline:
                break
            v = a * np.exp(-eta * grad / gmax)
            v = _normalize(v)
            cv = _conv_full_fft(v)
            scv = 2.0 * n * float(cv.max())
            if np.isfinite(scv) and scv < best_cand_sc:
                best_cand_sc = float(scv)
                best_cand = v
        alphas = (0.05, 0.1, 0.25, 0.5, 1.0, 2.0)
        for alpha in alphas:
            if time.monotonic() >= deadline:
                break
            v = _project_simplex(a - alpha * grad)
            cv = _conv_full_fft(v)
            scv = 2.0 * n * float(cv.max())
            if np.isfinite(scv) and scv < best_cand_sc:
                best_cand_sc = float(scv)
                best_cand = v
        if best_cand is not None:
            a = _normalize(best_cand)
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= max_patience:
                base = _invsqrt_seed(n, 0.5)
                if rng.random() < 0.6:
                    a = _project_simplex(best_a + rng.normal(0.0, 0.015 / math.sqrt(n), n))
                elif rng.random() < 0.5:
                    a = _random_smooth(n, rng, base=best_a)
                else:
                    a = _normalize(0.65 * best_a + 0.35 * base)
                no_improve = 0
        it += 1
        if it > 6000:
            break
    if on_candidate is not None:
        on_candidate(best_a)
    return best_a


def _smooth_refine(start, deadline, on_candidate, T_fracs=None):
    a = _normalize(start)
    n = int(a.size)
    if n < 2:
        return a
    if T_fracs is None:
        T_fracs = (0.15, 0.075, 0.0375, 0.01875)
    c = _conv_full_fft(a)
    best_a = a.copy()
    best_sc = 2.0 * n * float(c.max())
    if on_candidate is not None:
        on_candidate(best_a)
    for frac in T_fracs:
        rem = float(deadline - time.monotonic())
        if rem < 4.0:
            break
        T_abs = float(frac) * max(best_sc / max(2.0 * n, 1.0), 1e-12)
        if T_abs <= 0 or not np.isfinite(T_abs):
            continue
        state = {"best_a": best_a.copy(), "best_sc": float(best_sc)}

        def fun(u):
            uc = np.asarray(u, dtype=np.float64).ravel()
            if uc.size != n:
                uc = best_a * 0.0
            uc = np.where(np.isfinite(uc), uc, 0.0)
            uc = uc - uc.mean()
            uc = np.clip(uc, -50.0, 50.0)
            mxc = float(uc.max())
            z = np.exp(uc - mxc)
            sz = float(z.sum())
            if sz <= 1e-300 or not np.isfinite(sz):
                a_n = best_a.copy()
            else:
                a_n = z / sz
            c_n = _conv_full_fft(a_n)
            maxc = float(c_n.max())
            sc = 2.0 * n * maxc
            if np.isfinite(sc) and sc < state["best_sc"]:
                state["best_sc"] = float(sc)
                state["best_a"] = a_n.copy()
            if maxc <= 0:
                return 0.0, np.zeros(n, dtype=np.float64)
            vals = (c_n - maxc) / T_abs
            vals = np.clip(vals, -50.0, 0.0)
            ev = np.exp(vals)
            sev = float(ev.sum())
            if sev <= 0 or not np.isfinite(sev):
                val = maxc
                q = np.full(c_n.size, 1.0 / max(1, c_n.size))
            else:
                val = maxc + T_abs * float(np.log(sev))
                q = ev / sev
            G = _weighted_grad_fft(a_n, q)
            C = 2.0 * float(np.dot(q, c_n))
            g = a_n * (G - C)
            g = np.where(np.isfinite(g), g, 0.0)
            g -= float(g.mean())
            if not np.isfinite(val):
                val = maxc
                g = np.zeros(n, dtype=np.float64)
            return float(val), g

        try:
            u0 = np.log(np.maximum(best_a, 1e-300))
            u0 = u0 - float(u0.mean())
            t1 = time.monotonic()
            fun(u0)
            t_eval = max(1e-4, time.monotonic() - t1)
            maxit = int(min(50, max(8, (rem * 0.035) / t_eval)))
            if maxit < 8:
                continue
            bounds = [(-30.0, 30.0)] * n
            res = minimize(
                fun,
                u0,
                jac=True,
                method="L-BFGS-B",
                bounds=bounds,
                options={
                    "maxiter": maxit,
                    "ftol": 1e-10,
                    "gtol": 1e-5,
                    "maxls": 6 if n > 2048 else 8,
                    "maxcorrection": 10,
                },
            )
            if res is not None and getattr(res, "x", None) is not None:
                xr = np.asarray(res.x, dtype=np.float64).ravel()
                if xr.size == n and np.all(np.isfinite(xr)):
                    xr = xr - xr.mean()
                    xr = np.clip(xr, -50.0, 50.0)
                    z = np.exp(xr - xr.max())
                    sz = float(z.sum())
                    if sz > 1e-300 and np.isfinite(sz):
                        ar = z / sz
                        cr = _conv_full_fft(ar)
                        scr = 2.0 * n * float(cr.max())
                        if np.isfinite(scr) and scr < state["best_sc"]:
                            state["best_sc"] = float(scr)
                            state["best_a"] = ar.copy()
        except Exception:
            pass
        if state["best_sc"] < best_sc:
            best_sc = float(state["best_sc"])
            best_a = state["best_a"].copy()
            if on_candidate is not None:
                on_candidate(best_a)
    if on_candidate is not None:
        on_candidate(best_a)
    return best_a


def _lp_phase(start, deadline, rng, on_candidate):
    a = _normalize(start)
    n = int(a.size)
    if n < 2 or n > 600:
        return a
    c = _conv_full_fft(a)
    best_a = a.copy()
    best_sc = 2.0 * n * float(c.max())
    if on_candidate is not None:
        on_candidate(best_a)
    max_calls = int(min(9, max(2, (deadline - time.monotonic()) // 12.0)))
    call = 0
    it = 0
    no_improve = 0
    while time.monotonic() < deadline - 5.0:
        c = _conv_full_fft(a)
        M = float(c.max())
        sc = 2.0 * n * M
        if np.isfinite(sc) and sc < best_sc:
            best_sc = float(sc)
            best_a = a.copy()
            if on_candidate is not None:
                on_candidate(best_a)
        dirb = None
        wdual = None
        if time.monotonic() < deadline - 10.0:
            if it % 3 == 0 and call < max_calls:
                dirb = _lp_primal_dir(a, deadline, rng)
                call += 1
            elif it % 3 == 1 and n <= 480:
                wdual = _lp_dual_weights(a, c, deadline, rng)
        best_cand = None
        best_cand_sc = best_sc * (1.0 - 1e-13)
        if dirb is not None:
            for alpha in (0.05, 0.1, 0.2, 0.35, 0.55, 0.8, 1.0):
                v = _normalize(a + alpha * (dirb - a))
                cv = _conv_full_fft(v)
                scv = 2.0 * n * float(cv.max())
                if np.isfinite(scv) and scv < best_cand_sc:
                    best_cand_sc = float(scv)
                    best_cand = v
        if wdual is not None and np.any(wdual > 0):
            grad = _weighted_grad_fft(a, wdual)
            gmax = float(np.max(grad)) if grad.size else 0.0
            if gmax <= 1e-15:
                gmax = 1e-15
            for eta in (0.1, 0.25, 0.5, 1.0, 2.0):
                v = _normalize(a * np.exp(-eta * grad / gmax))
                cv = _conv_full_fft(v)
                scv = 2.0 * n * float(cv.max())
                if np.isfinite(scv) and scv < best_cand_sc:
                    best_cand_sc = float(scv)
                    best_cand = v
        if best_cand is None:
            q = _make_weights(c, n, frac=0.08, use_top=(n <= 600))
            grad = _weighted_grad_fft(a, q)
            gmax = float(np.max(grad)) if grad.size else 0.0
            if gmax <= 1e-15:
                gmax = 1e-15
            for eta in (0.1, 0.25, 0.5, 1.0, 2.0):
                v = _normalize(a * np.exp(-eta * grad / gmax))
                cv = _conv_full_fft(v)
                scv = 2.0 * n * float(cv.max())
                if np.isfinite(scv) and scv < best_cand_sc:
                    best_cand_sc = float(scv)
                    best_cand = v
            for alpha in (0.05, 0.15, 0.35, 0.8):
                v = _project_simplex(a - alpha * grad)
                cv = _conv_full_fft(v)
                scv = 2.0 * n * float(cv.max())
                if np.isfinite(scv) and scv < best_cand_sc:
                    best_cand_sc = float(scv)
                    best_cand = v
        if best_cand is not None:
            a = _normalize(best_cand)
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 5:
                a = _normalize(0.8 * best_a + 0.2 * _random_smooth(n, rng, base=best_a))
                no_improve = 0
        it += 1
        if it > 2000:
            break
    if on_candidate is not None:
        on_candidate(best_a)
    return best_a


def _optimize_length(start, deadline, rng, on_candidate, use_lp):
    a = _normalize(start)
    n = int(a.size)
    if on_candidate is not None:
        on_candidate(a)
    rem = float(deadline - time.monotonic())
    if rem > 22.0 and n <= 2048:
        gdeadline = min(deadline - 10.0, time.monotonic() + max(10.0, min(60.0, 0.20 * rem)))
        a = _gradient_phase(a, gdeadline, rng, on_candidate, max_patience=25)
    rem = float(deadline - time.monotonic())
    if rem > 18.0:
        smdeadline = min(deadline - 5.0, time.monotonic() + max(15.0, min(180.0, 0.48 * rem)))
        T_fracs = (0.20, 0.10, 0.05, 0.025) if n <= 1024 else (0.20, 0.08, 0.03)
        a = _smooth_refine(a, smdeadline, on_candidate, T_fracs)
    if use_lp:
        rem = float(deadline - time.monotonic())
        if rem > 35.0:
            lpdeadline = min(deadline - 10.0, time.monotonic() + max(20.0, min(140.0, 0.35 * rem)))
            a = _lp_phase(a, lpdeadline, rng, on_candidate)
    rem = float(deadline - time.monotonic())
    if rem > 12.0 and n <= 4096:
        smdeadline = min(deadline - 2.0, time.monotonic() + max(10.0, rem * 0.65))
        T_fracs = (0.04, 0.02, 0.01) if n <= 2048 else (0.04, 0.015)
        a = _smooth_refine(a, smdeadline, on_candidate, T_fracs)
    rem = float(deadline - time.monotonic())
    if rem > 8.0 and n <= 1024:
        gdeadline = min(deadline - 1.0, time.monotonic() + max(5.0, rem * 0.6))
        a = _gradient_phase(a, gdeadline, rng, on_candidate, max_patience=20)
    return a


def _exact_eval(arr):
    try:
        return float(evaluate_sequence([float(x) for x in arr]))
    except Exception:
        return float("inf")


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1300.0
    if not np.isfinite(budget) or budget <= 0.0:
        budget = 1300.0

    try:
        i_seed = int(seed)
    except Exception:
        i_seed = 42
    rng = np.random.default_rng(i_seed)

    parent = _get_parent()

    if budget <= 8.0:
        out = _normalize(parent)
        if out.size == 0:
            out = np.ones(1, dtype=np.float64)
        return [float(x) for x in np.clip(out, 0.0, 1000.0)]

    headroom = max(35.0, min(70.0, 0.05 * budget))
    deadline = t0 + budget - headroom
    if deadline <= t0 + 10.0:
        deadline = t0 + max(10.0, budget * 0.90)

    best_a = parent.copy()
    best_score = float("inf")
    exact_pool = []

    def add_pool(arr):
        try:
            a2 = _normalize(arr)
            if a2.size > 0 and np.all(np.isfinite(a2)) and len(exact_pool) < 60:
                exact_pool.append(a2.copy())
        except Exception:
            pass

    def note(a, exact=False):
        nonlocal best_a, best_score
        arr = _normalize(a)
        if arr.size == 0 or not np.all(np.isfinite(arr)):
            return
        sc = _score_arr(arr)
        if not np.isfinite(sc):
            return
        improved = False
        if sc < best_score:
            best_score = float(sc)
            best_a = arr.copy()
            improved = True
        if exact:
            esc = _exact_eval(arr)
            if np.isfinite(esc) and esc < best_score:
                best_score = float(esc)
                best_a = arr.copy()
                improved = True
        if improved:
            add_pool(best_a)
        else:
            add_pool(arr)

    note(parent, exact=True)

    if time.monotonic() >= deadline - 20.0:
        out = _normalize(best_a)
        return [float(x) for x in np.clip(out, 0.0, 1000.0)]

    parent_len = int(parent.size)
    lengths = {256, 512, 1024, 2048}
    if budget > 900.0:
        lengths.update({384, 768, 1536, 4096})
    if budget > 1150.0:
        lengths.add(8192)
    if 32 <= parent_len <= 8192:
        lengths.add(parent_len)
    lengths = sorted(int(x) for x in lengths if 32 <= int(x) <= 8192)

    seed_best = {}
    for n in lengths:
        if time.monotonic() >= deadline - 30.0:
            break
        bs = None
        bsc = float("inf")
        for seedarr in _make_seeds(n, parent, rng):
            sc = _score_arr(seedarr)
            if np.isfinite(sc) and sc < bsc:
                bsc = float(sc)
                bs = seedarr.copy()
        if bs is None:
            bs = _invsqrt_seed(n, 0.5)
        seed_best[n] = (bs, bsc if np.isfinite(bsc) else 2.0)
        note(bs, exact=False)
        add_pool(bs)

    base_w = {
        128: 0.05,
        256: 0.40,
        384: 0.20,
        512: 1.00,
        768: 0.40,
        1024: 0.80,
        1536: 0.45,
        2048: 0.75,
        3072: 0.30,
        4096: 0.35,
        8192: 0.25,
    }
    weights = {}
    for n in lengths:
        w = max(0.05, float(base_w.get(n, 0.2)))
        sc = seed_best[n][1]
        if np.isfinite(sc):
            w *= max(0.6, min(1.4, 1.5 / max(1.2, sc)))
        if n == parent_len:
            w *= 1.10
        if n > 256:
            w *= 1.0 + 0.04 * max(0.0, math.log2(float(n) / 256.0))
        weights[n] = float(w)

    order = sorted(lengths, key=lambda x: -weights.get(x, 0.0))
    remaining_w = float(sum(weights.values()))
    if remaining_w <= 0:
        remaining_w = 1.0

    for n in order:
        if time.monotonic() >= deadline - 25.0:
            break
        rem = float(deadline - time.monotonic())
        if rem < 30.0:
            break
        alloc = rem * weights.get(n, 0.0) / max(1e-12, remaining_w)
        remaining_w -= weights.get(n, 0.0)
        if alloc < 22.0:
            continue
        sub_deadline = min(deadline, time.monotonic() + alloc)
        init, _ = seed_best.get(n, (_invsqrt_seed(n, 0.5), 2.0))
        use_lp = (n <= 600 and alloc > 55.0)
        try:
            a = _optimize_length(init, sub_deadline, rng, note, use_lp)
        except Exception:
            a = init
        note(a, exact=False)
        add_pool(a)

    # Final global polish on the current best.
    rem = float(deadline - time.monotonic())
    if rem > 35.0 and best_a.size > 0:
        try:
            b = _normalize(best_a)
            if b.size <= 600 and rem > 70.0:
                b = _lp_phase(b, min(deadline - 15.0, time.monotonic() + 0.45 * rem), rng, note)
            note(b, exact=False)
            b = _smooth_refine(
                b,
                min(deadline - 5.0, time.monotonic() + 0.45 * rem),
                note,
                (0.03, 0.015, 0.008),
            )
            note(b, exact=False)
        except Exception:
            pass

    # Exact final selection.
    candidates = []
    for arr in exact_pool:
        candidates.append(arr)
    candidates.append(best_a)
    candidates.append(parent)
    fin_a = None
    fin_sc = float("inf")
    for arr in candidates:
        try:
            a2 = _normalize(arr)
            if a2.size == 0 or not np.all(np.isfinite(a2)):
                continue
            sc = _exact_eval(a2)
            if np.isfinite(sc) and sc < fin_sc:
                fin_sc = float(sc)
                fin_a = a2.copy()
        except Exception:
            continue

    if fin_a is None:
        fin_a = _normalize(best_a)
    if fin_a.size == 0 or not np.all(np.isfinite(fin_a)):
        fin_a = _normalize(parent)
    if fin_a.size == 0 or not np.all(np.isfinite(fin_a)):
        fin_a = np.ones(1, dtype=np.float64)

    fin_a = np.clip(fin_a, 0.0, 1000.0)
    fin_a = _normalize(fin_a)
    if fin_a.size == 0:
        return [1.0]
    return [float(x) for x in fin_a]
