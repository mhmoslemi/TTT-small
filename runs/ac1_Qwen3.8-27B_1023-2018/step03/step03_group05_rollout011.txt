def _project_simplex(v, target=1.0):
    v = np.asarray(v, dtype=np.float64).ravel()
    n = v.size
    if n == 0:
        return np.array([target], dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)

    css = np.cumsum(u)

    rho = np.nonzero(cond)[0]

        return np.full(n, target / n, dtype=np.float64)

    s = float(w.sum())

def _grad_soft_max(a, c, T):
    """Gradient of T*log(sum exp(c/T)) w.r.t. a."""
    n = a.size
