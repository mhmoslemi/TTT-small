        # So A[k, j] = a[k-j] for j in [lo, hi]
        for j in range(lo, hi+1):

                      options={'time_limit': tl, 'presolve': True})

            x = np.asarray(res.x, dtype=np.float64).ravel()
            if x.size == n and np.all(np.isfinite(x)):
                x = np.maximum(x, 0.0)
                s = float(x.sum())
                if s > 1e-12:
                    return x / s
    except Exception:
        pass
    return None

def _weighted_gradient(a, w):
    """Compute sum_k w[k] * 2*a[k-m] for m = 0..n-1 using FFT."""
    n = a.size
    m_full = 2*n - 1
    if w.size != m_full:
        if w.size > m_full:
            w = w[:m_full]
        else:
            w = np.pad(w, (0, m_full - w.size))
    w = np.where(np.isfinite(w), w, 0.0)
    s = float(w.sum())
    if s <= 0:
        return np.zeros(n)
    w = w / s
    
    nfft = next_fast_len(3*n - 2)
    fa = rfft(a, nfft)
    fw = rfft(w[::-1], nfft)
    cr = irfft(fa * fw, nfft)[:3*n - 2]
    cr = np.where(np.isfinite(cr), cr, 0.0)
    
    idx = 2*n - 2 - np.arange(n)
    g = 2.0 * cr[idx]
    return g  # Don't clip - the gradient can have any sign in the weighted case
    # Actually wait, since a >= 0 and w >= 0, the gradient should be >= 0.
    # But numerically there might be tiny negatives. Let's not clip for now.

def _try_step(a, direction, alphas, deadline):
    """Try a series of step sizes, return best (a_new, score, conv)."""
    n = a.size
    best_a = None
    best_M = float('inf')
    best_c = None
    
    for alpha in alphas:
        if time.monotonic() >= deadline:
            break
        cand = _simplex_project(a + alpha * direction)
        c = _conv_full(cand)
        M = float(c.max())
        if M < best_M:
            best_M = M
            best_a = cand
            best_c = c
    return best_a, best_M, best_c

def _local_refine(a, deadline, rng, use_lp=True, lp_interval=3):
    """Main local refinement loop."""
    a = _normalize(a)
    n = a.size
    c = _conv_full(a)
    M = float(c.max())
    best_a = a.copy()
    best_M = M
    
    eta = 0.1  # Initial step size
    patience = 0
    it = 0
    
    while time.monotonic() < deadline:
        it += 1
        rem = deadline - time.monotonic()
        if rem < 2.0:
            break
        
        # Update best
        if M < best_M * (1.0 - 1e-14):
            best_a = a.copy()
            best_M = M
            patience = 0
            eta = max(0.001, eta * 0.99)
        else:
            patience += 1
            eta = max(0.001, eta * 0.95)
        
        # Periodic restart if stuck
        if patience > 50:
            r = rng.random()
            if r < 0.4:
                a = _simplex_project(a + rng.normal(0, 0.02, n))
            elif r < 0.8:
                # Blend with a smooth function
                x = np.linspace(0, 1, n, endpoint=False)
                alpha_p = float(rng.uniform(0.3, 2.0))
                smooth = x ** alpha_p
                smooth = _normalize(smooth)
                a = _simplex_project(0.7 * a + 0.3 * smooth)
            else:
                a = _simplex_project(a * np.exp(0.05 * rng.standard_normal(n)))
            c = _conv_full(a)
            M = float(c.max())
            patience = 0
            eta = 0.1
            continue
        
        # Compute gradient
        # Find active lags (near-max)
        M_val = float(c.max())
        threshold = M_val * 0.995  # 0.5% of max
        active = np.where(c >= threshold)[0]
        
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])
        elif len(active) > 64:
            # Fall back to soft weights
            T = M_val * 0.01
            z = (c - M_val) / T
            z = np.clip(z, -50, 0)
            w = np.exp(z)
            ws = w.sum()
            if ws > 0:
                w = w / ws
            else:
                w = np.zeros(c.size)
                w[int(np.argmax(c))] = 1.0
        else:
            w = np.zeros(c.size)
            w[active] = 1.0 / len(active)
        
        grad = _weighted_gradient(a, w)
        
        # LP direction (periodically)
        if use_lp and it % lp_interval == 0 and rem > 10.0 and n <= 600:
            b_dir = _solve_lp_direction(a, M_val, deadline - 5.0, rng)
            if b_dir is not None:
                # Try LP direction
                a_try, M_try, c_try = _try_step(
                    a, b_dir - a, (0.1, 0.3, 0.5, 0.8, 1.0), deadline)
                if a_try is not None and M_try < M * (1 - 1e-14):
                    a = a_try
                    c = c_try
                    M = M_try
                    continue
        
        # Gradient step
        if float(np.linalg.norm(grad)) > 1e-15:
            # Project gradient onto tangent space (remove mean)
            g_tangent = grad - grad.mean()
            
            # Additive step
            a_add, M_add, c_add = _try_step(
                a, -g_tangent, (eta*0.5, eta, eta*2), deadline)
            
            # Multiplicative step
            a_mult = a * np.exp(-eta * grad / max(grad.mean(), 1e-15))
            a_mult = a_mult / a_mult.sum()
            c_mult = _conv_full(a_mult)
            M_mult = float(c_mult.max())
            
            # Pick the best
            candidates = []
            if a_add is not None:
                candidates.append((M_add, a_add, c_add))
            candidates.append((M_mult, a_mult, c_mult))
            
            best_cand = min(candidates, key=lambda x: x[0])
            if best_cand[0] < M * (1 - 1e-14):
                M, a, c = best_cand[0], best_cand[1], best_cand[2]
                eta = min(eta * 1.1, 2.0)
            else:
                # No improvement, reduce step
                eta = max(0.001, eta * 0.8)
        
        # Update convolution if a changed
        if a is not best_a:  # rough check
            c = _conv_full(a)
            M = float(c.max())
    
    return best_a, best_M

Wait, I have a problem. The gradient computation: I'm computing the gradient of the max convolution with respect to `a`, which gives a direction to DECREASE the max. So I should move in the direction `-grad` in the simplex.

But there's a subtlety: the objective is `2n * max(c) / (sum(a))²`. With `sum(a) = 1`, this is just `2n * max(c)`. The gradient of this w.r.t. `a` (on the simplex) is `2n * grad(c[k*])` where `k*` is the argmax.

Since I'm already working with sum=1 normalization, the factor `2n` doesn't affect the direction.

Now, the key question: is the gradient direction `-grad` actually good? Let's think:
- `grad[m] = 2*a[k*-m]` (approximately, for a single active lag)
- This says: to decrease `c[k*]`, reduce `a[m]` proportional to `a[k*-m]`
- In other words, the "mirror" of `a` (reflected around k*/2) tells us where to reduce

This makes intuitive sense: if `a` has a peak at position p, then `c[k*]` gets a contribution from `a[p] * a[k*-p]`. To reduce `c[k*]`, we want to reduce the product, i.e., reduce one of the factors.

OK, I think the approach is sound. Let me also add a key improvement: **annealing the active set**.

Initially, use a broad active set (many lags near the max) to get a smooth gradient. As we converge, narrow the active set to focus on the true maximum. This is similar to temperature annealing in simulated annealing.

Let me also think about whether there's a better way to handle the non-differentiability at the max. One approach is the "smoothing" trick:
`max(c) ≈ T * log(sum exp(c/T))` for small T.

The gradient of this is:
`∇(T * log(sum exp(c_k/T))) = sum_k (exp(c_k/T) / sum exp(c_j/T)) * ∇c_k`

This is a weighted average of the individual gradients, with weights proportional to `exp(c_k/T)`. As T → 0, this concentrates on the argmax.

I'll use this with adaptive T: start with large T (broad) and decrease over time.

def _simplex_project(v, target=1.0):
    v = np.asarray(v, dtype=np.float64).ravel()
    n = v.size
    if n == 0:
        return np.array([1.0])
    if n == 1:
        return np.array([max(target, 0.0)])
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = (css - target) / idx < u
    rho = np.nonzero(cond)[0]
    if rho.size == 0:
        return np.full(n, target / n)
    theta = (css[rho[-1]] - target) / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    s = float(w.sum())
    if s <= 1e-300:
        return np.full(n, target / n)
    return (w / s) * target

                    if time.monotonic() >= deadline:
                        break
                    cand = _simplex_project(a + alpha * (b_dir - a))
                    cc = _conv_full(cand)
                    mm = float(cc.max())
                    if mm < M * (1 - 1e-13):
                        a, c, M = cand, cc, mm
                        improved = True
                        break

        if not improved:
            # Additive gradient step
            g_tan = grad - grad.mean()
            best_step_M = M
            best_step_a = a
            best_step_c = c
            for scale in (0.5, 1.0, 2.0, 4.0):
                e = eta * scale
                cand = _simplex_project(a - e * g_tan)
                cc = _conv_full(cand)
                mm = float(cc.max())
                if mm < best_step_M:
                    best_step_M = mm
                    best_step_a = cand
                    best_step_c = cc

            # Multiplicative step
            gm = max(grad.mean(), 1e-15)
            for e_mult in (eta * 0.5, eta, eta * 2):
                cand = a * np.exp(-e_mult * grad / gm)
                s = float(cand.sum())
                if s > 1e-15:
                    cand = cand / s
                    cc = _conv_full(cand)
                    mm = float(cc.max())
                    if mm < best_step_M:
                        best_step_M = mm
                        best_step_a = cand
                        best_step_c = cc

            if best_step_M < M * (1 - 1e-13):
                a, c, M = best_step_a, best_step_c, best_step_M
            else:
                eta = max(0.003, eta * 0.85)

    return best_a, best_M


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = max(0.0, float(budget_s))
    except Exception:
        budget = 1300.0

    if budget <= 3.0:
        p = _get_parent()
        sc = _exact_score(p)
        if np.isfinite(sc):
            return [float(x) for x in p]
        return [1.0]

    headroom = max(10.0, min(40.0, 0.04 * budget))
    deadline = t0 + budget - headroom

    rng = np.random.default_rng(int(seed) if np.isfinite(float(seed)) else 42)

    parent = _get_parent()
    best_a = _normalize(parent)
    best_score = _exact_score(best_a)

    if not np.isfinite(best_score):
        best_a = np.ones(512, dtype=np.float64) / 512.0
        best_score = _exact_score(best_a)
        if not np.isfinite(best_score):
            return [1.0]

    n = best_a.size

    # Phase 1: Main refinement on parent (longest)
    p1_time = min(700.0, 0.60 * budget)
    p1_deadline = min(t0 + p1_time, deadline - 50.0)
    if p1_deadline - time.monotonic() > 20.0:
        try:
            loc_a, loc_M = _local_refine(best_a, p1_deadline, rng, use_lp=True, lp_interval=3)
            sc = _exact_score(_normalize(loc_a))
            if np.isfinite(sc) and sc < best_score:
                best_a = _normalize(loc_a)
                best_score = float(sc)
        except Exception:
            pass

    # Phase 2: Exploratory diverse runs
    if time.monotonic() < deadline - 40.0:
        attempts = 4
        time_per = (deadline - time.monotonic() - 30.0) / attempts
        if time_per > 20.0:
            for att in range(attempts):
                if time.monotonic() >= deadline - 35.0:
                    break
                x = np.linspace(0, 1, n, endpoint=False)
                if att == 0:
                    init = _normalize(x ** float(rng.uniform(0.4, 1.2)))
                elif att == 1:
                    init = _normalize((1 - x) ** float(rng.uniform(0.4, 1.2)))
                elif att == 2:
                    c0 = float(rng.uniform(0.3, 0.7))
                    w0 = float(rng.uniform(0.08, 0.25))
                    init = _normalize(np.exp(-((x - c0) ** 2) / (2 * w0 * w0)) + 0.05)
                else:
                    init = _random_smooth(n, rng)
                init = _simplex_project(0.5 * best_a + 0.5 * init)
                sub_deadline = min(time.monotonic() + time_per * 0.9, deadline - 35.0)
                if sub_deadline - time.monotonic() < 15.0:
                    continue
                try:
                    loc_a2, loc_M2 = _local_refine(init, sub_deadline, rng,
                                                   use_lp=(n <= 512), lp_interval=4)
                    sc2 = _exact_score(_normalize(loc_a2))
                    if np.isfinite(sc2) and sc2 < best_score:
                        best_a = _normalize(loc_a2)
                        best_score = float(sc2)
                except Exception:
                    pass

    # Phase 3: Final polish
    if time.monotonic() < deadline - 15.0:
        pol_deadline = deadline - 5.0
        if pol_deadline - time.monotonic() > 15.0:
            try:
                loc_a3, loc_M3 = _local_refine(best_a, pol_deadline, rng,
                                               use_lp=True, lp_interval=2)
                sc3 = _exact_score(_normalize(loc_a3))
                if np.isfinite(sc3) and sc3 < best_score:
                    best_a = _normalize(loc_a3)
                    best_score = float(sc3)
            except Exception:
                pass

    # Final validation
    try:
        best_a = _normalize(best_a)
        fs = _exact_score(best_a)
        if np.isfinite(fs):
            best_score = fs
        else:
            raise ValueError("invalid")
    except Exception:
        best_a = _get_parent()
        best_score = _exact_score(best_a)
        if not np.isfinite(best_score):
            return [1.0]

    out = np.clip(best_a, 0.0, 1000.0)
    if out.size == 0:
        return [1.0]
    return [float(x) for x in out]
