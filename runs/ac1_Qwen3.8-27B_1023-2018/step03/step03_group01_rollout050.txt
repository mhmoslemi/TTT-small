        A[k, lo:hi] = f[k - (hi - 1):k - lo + 1][::-1]

def _lp_solve(f, n, M, time_limit=30.0):
    A = _build_conv_matrix(f, n)
    c_vec = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(
            c_vec, A_ub=A, b_ub=b_ub,
            bounds=[(0.0, None)] * n,
            method="highs",
            options={"presolve": True, "time_limit": time_limit},
        )
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    gs = float(g.sum())
    if gs < 1e-14:
        return None
    return g / gs


def _subgrad_step(f, b, n, lr, momentum, beta, topk=8):
    M = float(b.max())
    n_lags = 2 * n - 1
    k = min(topk, n_lags)
    idx = np.argpartition(b, -k)[-k:]
    wts = b[idx].copy()
    wts = np.exp(wts - wts.max())
    wts /= wts.sum()

    grad_M = np.zeros(n)
    for w, kk in zip(wts, idx):
        kk = int(kk)
        lo = max(0, kk - n + 1)
        hi = min(n, kk + 1)
        for i in range(lo, hi):
            j = kk - i
            if 0 <= j < n:
                grad_M[i] += w * 2.0 * f[j]

    s = float(f.sum())
    if s < 1e-10:
        return f.copy(), 0.0
    grad_F = (2.0 * n / (s * s)) * grad_M - (4.0 * n * M / (s ** 3)) * np.ones(n)

    f_new = f + momentum
    f_new = np.maximum(f_new, 0.0)
    fs = float(f_new.sum())
    if fs < 1e-10:
        f_new = f.copy()
    else:

    return f_new, M

            if best_i_sc < best_sc - 1e-13:
                best_sc = best_i_sc
                improved = True

        # 2-coord moves
        for _ in range(60):
            if time.monotonic() > stop_time - 2.0:
                return f, best_sc
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            oi, oj = f[i], f[j]
            for mi in (0.5, 1.5, 2.0, 0.3):
                for mj in (0.5, 1.5, 0.3, 2.0):
                    f[i] = oi * mi if oi > 1e-12 else 0.01
                    f[j] = oj * mj if oj > 1e-12 else 0.01
                    sc = _score(f)
                    if sc < best_sc - 1e-13:
                        best_sc = sc
                        improved = True
                        break
                f[i], f[j] = oi, oj
                if f[i] != oi or f[j] != oj:
                    break
            f[i], f[j] = oi, oj

        if not improved:
            no_imp += 1
            if no_imp >= 8:
                break
        else:
            no_imp = 0

    return f, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 20.0

    parent = _get_parent()
    n_p = len(parent)
    n = n_p  # Use parent length

    best_seq = parent.copy()
    best_score = _score(best_seq)
    if not math.isfinite(best_score):
        best_seq = np.ones(384) / 384.0
        n = 384
        best_score = _score(best_seq)

    def _try_update(cand):
        nonlocal best_seq, best_score
        if cand is None:
            return
        cand = np.clip(np.asarray(cand, dtype=np.float64).ravel(), 0, 1000)
        if cand.size == 0 or cand.sum() < 0.01:
            return
        try:
            sc = _score(cand)
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-14:
            best_seq = cand.copy()
            best_score = sc

    # Phase 1: LP optimization - main workhorse
    # Use parent as primary init, plus a few alternatives
    rng_main = np.random.default_rng(seed)
    i_arr = np.arange(n, dtype=np.float64)

    inits = [
        ("parent", parent.copy()),
        ("cosine", 0.3 + 0.7 * np.cos(np.pi * i_arr / n) ** 2),
        ("gauss", np.exp(-0.5 * ((i_arr - n / 2) / (n * 0.25)) ** 2) + 0.05),
    ]
    # Add parent + small noise
    noise = rng_main.normal(0, 0.05 * max(1.0, np.mean(parent)), n)
    inits.append(("pnoise", np.abs(parent + noise) + 0.01))

    # Allocate ~55% of budget for LP
    lp_end = t0 + budget * 0.55
    n_inits = len(inits)
    time_per_lp = max(20.0, (lp_end - t0 - 10) / n_inits)

    for idx, (name, init) in enumerate(inits):
        if time.monotonic() > lp_end - 5.0:
            break
        if best_score < 1.5028:
            break
        stop = min(lp_end - 4.0, time.monotonic() + time_per_lp)
        try:
            res, sc = _lp_loop(init, n, stop, seed + 1000 * (idx + 1))
            _try_update(res)
        except Exception:
            pass

    # Phase 2: Subgradient polish
    sg_end = min(deadline, t0 + budget * 0.78)
    if time.monotonic() < sg_end - 15.0:
        try:
            res, sc = _subgrad_polish(best_seq, n, sg_end, seed + 50001)
            _try_update(res)
        except Exception:
            pass

    # Phase 3: Coordinate descent polish
    cd_end = min(deadline, t0 + budget * 0.92)
    if time.monotonic() < cd_end - 15.0:
        try:
            res, sc = _coord_polish(best_seq, n, cd_end, seed + 77001)
            _try_update(res)
        except Exception:
            pass

    # Phase 4: Final subgradient burst
    if time.monotonic() < deadline - 12.0:
        try:
            res, sc = _subgrad_polish(best_seq, n, deadline - 10.0, seed + 33001)
            _try_update(res)
        except Exception:
            pass

    # Phase 5: Final coordinate micro-polish
    if time.monotonic() < deadline - 8.0:
        try:
            res, sc = _coord_polish(best_seq, n, deadline - 6.0, seed + 44001)
            _try_update(res)
        except Exception:
            pass

    # Final validation with exact evaluator
    final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    try:
        sc = evaluate_sequence(final_list)
        if math.isfinite(sc) and sc > 0 and len(final_list) > 0:
            return final_list
    except Exception:
        pass

    # Fallback to parent
    fb = [float(max(0.0, min(1000.0, x))) for x in parent]
    try:
        sc = evaluate_sequence(fb)
        if math.isfinite(sc) and sc > 0:
            return fb
    except Exception:
        pass

    return [1.0] * n
