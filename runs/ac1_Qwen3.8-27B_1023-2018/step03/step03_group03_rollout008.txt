    # LP direction
    def _lp_dir(f):
        n_loc = len(f)
        b = np.convolve(f, f, mode='full')
        M = float(b.max())
        if M < 1e-15:
            return None
        
        # Build constraint matrix
        A = np.zeros((2*n_loc-1, n_loc))
        for j in range(n_loc):
            A[j:j+n_loc, j] = f
        
        try:
            res = linprog(-np.ones(n_loc), A_ub=A, b_ub=np.full(2*n_loc-1, M),
                          bounds=[(0, None)]*n_loc, method='highs',
                          options={'presolve': True, 'time_limit': 5.0})
        except Exception:
            return None
        
        if res.status != 0 or res.x is None:
            return None
        g = np.asarray(res.x, dtype=np.float64)
        gs = float(g.sum())
        if gs < 1e-10:
            return None
        return g / gs
