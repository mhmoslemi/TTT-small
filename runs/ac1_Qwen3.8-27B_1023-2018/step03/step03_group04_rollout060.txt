        return None

import numpy as np
import time
import math
from scipy.optimize import linprog


def _autoconv(a):
    return np.convolve(a, a)


def _score(a):
    n = len(a)
    S = float(a.sum())
    if S < 0.01:
        return math.inf
    M = float(_autoconv(a).max())
    if not math.isfinite(M):
        return math.inf
    return 2.0 * n * M / (S * S)


def _lp_direction(f, M, n, time_limit=6.0):
    """Solve LP: max sum(g) s.t. (f*cosv g)[k] <= M, g >= 0. Normalize g to sum=1."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, None)] * n,
                      method='highs',
                      options={"time_limit": time_limit,
                               "primal_feasibility_tolerance": 1e-9,
                               "dual_feasibility_tolerance": 1e-9})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    s = float(g.sum())
    if s < 1e-12:
        return None
    return g / s


def _subgrad_step(f, conv, M, n):
    """Compute subgradient of max_k b[k] and return projected step candidate."""
    thresh = 0.95 * M
    active = np.where(conv >= thresh)[0]
    if len(active) == 0:
        active = np.argmax(conv, keepdims=True)

    grad = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for j in range(lo, hi + 1):
            grad[j] += 2.0 * f[k - j]
    grad /= float(len(active))
    return grad

    """1-coordinate multiplier search."""

        if time.monotonic() > deadline - 3.0:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > deadline - 3.0:
                break
            orig = f[i]
            best_val = orig
            best_sc = cur_score
            for m in mults:
                f[i] = orig * m if orig > 1e-14 else (0.0005 if m > 0 else 0.0)
                if f.sum() < 0.01:
                    f[i] = orig
                    continue
                sc = _score(f)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_val = f[i]
            f[i] = best_val
            if best_sc < cur_score - 1e-14:
                cur_score = best_sc
                improved = True
        if not improved:
            no_imp += 1
            if no_imp >= 3:
                break
        else:
            no_imp = 0
    return f, cur_score


def _pair_polish(f, deadline, seed, n, max_rounds=50):
    """2-coordinate selective adjustment."""
