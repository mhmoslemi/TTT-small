import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    return c


def _score_fast(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _conv_full(a)
    m = float(c.max())
    if m <= 0:
        return 0.0
    return 2.0 * n * m / (s * s)


def _lp_dir(f, n, M, time_limit=5.0):

    for i in range(n):
        row_end = min(2 * n - 1, i + n)
        A[i:row_end, i] = f[i]
    neg_ones = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(
            neg_ones, A_ub=A, b_ub=b_ub,
            bounds=[(0.0, None)] * n,
            method="highs",
            options={"time_limit": time_limit, "presolve": True},

        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.maximum(res.x, 0.0)
    gs = float(g.sum())
    if gs < 1e-12:
        return None
    return g / gs

            t1 = t2
            f1 = f2
            sc1 = sc2
            t2 = lo + gr * (hi - lo)
            f2 = (1.0 - t2) * f + t2 * d
            np.maximum(f2, 0.0, out=f2)
            s2 = float(f2.sum())
            sc2 = _score_fast(f2) if s2 > 1e-10 else math.inf
        if hi - lo < 1e-5:
            break
    tm = 0.5 * (lo + hi)
    fm = (1.0 - tm) * f + tm * d
    np.maximum(fm, 0.0, out=fm)
    sm = float(fm.sum())
    scm = _score_fast(fm) if sm > 1e-10 else math.inf
    if scm < best_val:
        best_val = scm
        best_t = tm
        best_f = fm.copy()
    return best_t, best_f


def _lp_optimize(f0, n, stop, seed=42):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    fs = float(f.sum())
    f = f / fs if fs > 1e-10 else np.ones(n) / n
    best_f = f.copy()
    best_sc = _score_fast(f)
    t_cur = 0.25
    stall = 0
    max_stall = 40
    it = 0

    while time.monotonic() < stop - 3.0:
        it += 1
        conv = _conv_full(f)
        M = float(conv.max())
        if M < 1e-15:
            break

        g = _lp_dir(f, n, M, time_limit=8.0)
        if g is None:
            stall += 3
            if stall > max_stall:
                break
            f = np.clip(f + rng.normal(0, 0.04, n), 0, None)
            f = f / f.sum()
            t_cur = 0.2
            continue

        dist = float(np.linalg.norm(g - f))
        if dist < 1e-7:
            stall += 3
            if stall > max_stall:
                break
            f = np.clip(f + rng.normal(0, 0.06, n), 0, None)
            f = f / f.sum()
            t_cur = 0.25
            continue

        res_t, f_new = _line_search(f, g, n, best_sc, t_max=min(0.6, t_cur * 1.3))
        if f_new is not None:
            sc_new = _score_fast(f_new)
            if sc_new < best_sc - 1e-14:
                f = f_new
                best_f = f.copy()
                best_sc = sc_new
                t_cur = min(0.5, res_t * 1.2)
                stall = 0
            else:
                t_cur = max(0.05, res_t * 0.5)
                stall += 1
        else:
            t_cur = max(0.05, t_cur * 0.5)
            stall += 1

        if stall >= max_stall:
            f = np.clip(f + rng.normal(0, 0.08, n), 0, None)
            f = f / f.sum()
            t_cur = 0.25
            stall = 0
            max_stall = min(80, max_stall + 10)

        if best_sc < 1.503:
            break

    return best_f, best_sc


def _subgrad_optimize(f0, n, stop, seed=42):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    f = f / f.sum()
    best_f = f.copy()
    best_sc = _score_fast(f)
    lr = 0.008
    momentum = np.zeros(n)
    beta = 0.8
    no_imp = 0
    f_prev = f.copy()

    for it in range(500000):
        if time.monotonic() > stop - 3.0:
            break

        conv = _conv_full(f)
        M = float(conv.max())
        s = float(f.sum())

        # Subgradient of max_k conv_k at f
        # Use top-3 lags with softmax weights
        topk = min(3, 2 * n - 1)
        idx = np.argpartition(-conv, topk)[:topk]
        weights = conv[idx] - conv[idx].min() + 1e-10
        weights = weights * weights  # square for sharper focus
        weights /= weights.sum()

        grad_M = np.zeros(n)
        for w, k in zip(weights, idx):
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            for i in range(lo, hi):
                j = k - i
                if 0 <= j < n:
                    grad_M[i] += w * 2.0 * f[j]

        # Gradient of score = 2n*M/s^2
        grad_score = (2.0 * n / (s * s)) * grad_M - (4.0 * n * M / (s * s * s)) * np.ones(n)

        # Nesterov momentum
        v = beta * momentum - lr * grad_score
        momentum = v
        f_new = f + v
        f_new = np.maximum(f_new, 0.0)
        sn = float(f_new.sum())
        if sn < 1e-10:
            f_new = f.copy()
        else:
            f_new = f_new / sn

        sc = _score_fast(f_new)
        if sc < best_sc - 1e-15:
            f = f_new
            best_f = f.copy()
            best_sc = sc
            no_imp = 0
            lr = min(0.04, lr * 1.005)
        else:
            no_imp += 1
            lr = max(0.001, lr * 0.995)
            f = f_new  # still move (non-monotone)
            if no_imp > 200:
                lr = 0.005
                beta = 0.7
                momentum = np.zeros(n)
                no_imp = 0

        if best_sc < 1.5027:
            break

    return best_f, best_sc


def _coord_optimize(f0, n, stop, seed=42):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    f = f / f.sum()
    best_sc = _score_fast(f)
    no_sweep = 0
    sweeps = 0

    while time.monotonic() < stop - 3.0 and no_sweep < 15:
        sweeps += 1
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop - 3.0:
                break
            orig = f[i]
            if orig < 1e-10:
                # Try small values
                for v in (0.005, 0.01, 0.02):
                    f[i] = v
                    sc = _score_fast(f)
                    if sc < best_sc - 1e-14:
                        best_sc = sc
                        improved = True
                f[i] = orig
                continue
            # Try geometric scaling
            for mult in (0.4, 0.6, 0.8, 1.25, 1.5, 2.0):
                f[i] = orig * mult
                sc = _score_fast(f)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    improved = True
                    orig = f[i]
            # Fine adjustment
            for d in (0.05, -0.05, 0.1, -0.1):
                f[i] = max(0.0, orig * (1.0 + d))
                sc = _score_fast(f)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    improved = True
            f[i] = orig

        # 2-coordinate moves
        for _ in range(30):
            if time.monotonic() > stop - 3.0:
                break
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            oi, oj = f[i], f[j]
            for mi in (0.5, 1.5, 2.0, 0.3):
                for mj in (0.5, 1.5, 1.0, 0.4):
                    f[i] = max(0.0, oi * mi)
                    f[j] = max(0.0, oj * mj)
                    sc = _score_fast(f)
                    if sc < best_sc - 1e-14:
                        best_sc = sc
                        improved = True
            f[i], f[j] = oi, oj

        if improved:
            no_sweep = 0
        else:
            no_sweep += 1

    return f, best_sc


def _make_inits(parent, n, rng):
    inits = [parent.copy()]
    i = np.arange(n, dtype=np.float64)
    # Cosine window
    inits.append(0.5 + 0.5 * np.cos(2 * np.pi * i / n))
    # Tent (peak at center)
    inits.append(1.0 - np.abs(i - n / 2) / (n / 2))
    # Anti-tent (peak at edges)
    inits.append(np.abs(i - n / 2) / (n / 2) + 0.1)
    # Multi-frequency
    mf = np.full(n, 0.15)
    for fr in (1, 2, 3, 5, 8, 13):
        ph = rng.uniform(0, 2 * np.pi)
        mf += 0.25 * np.cos(2 * np.pi * fr * i / n + ph)
    inits.append(np.abs(mf) + 0.03)
    # Random positive
    inits.append(np.abs(rng.normal(0, 1, n)) + 0.02)
    # Parent + Gaussian noise
    ns = 0.4 * max(1.0, float(parent.mean()))
    inits.append(np.abs(parent + rng.normal(0, ns, n)) + 0.01)
    # Sinc-like
    sinc = np.sinc((i - n * 0.5) / (n * 0.35))
    inits.append(np.abs(sinc) + 0.04)
    # Two-lobe: high at 1/4 and 3/4
    two_lobe = np.exp(-((i - n * 0.25) / (n * 0.12))**2) + np.exp(-((i - n * 0.75) / (n * 0.12))**2)
    inits.append(two_lobe + 0.02)
    # Periodic: high at multiples of n/6
    per = np.zeros(n)
    for k in range(6):
        center = k * n // 6
        per += np.exp(-((i - center) / (n * 0.06))**2)
    inits.append(per + 0.02)
    # All normalized
    result = []
    for a in inits:
        a = np.clip(np.asarray(a, dtype=np.float64), 0, None)
        sa = a.sum()
        if sa > 0.01:
            result.append(a / sa)
        else:
            result.append(np.ones(n) / n)
    return result


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 25.0

    # Get parent sequence
    try:
        parent = np.array(height_sequence_1, dtype=np.float64).copy()
    except Exception:
        parent = np.ones(384, dtype=np.float64) / 384.0

    parent = np.clip(parent, 0.0, 1000.0)
    parent = np.where(np.isfinite(parent), parent, 0.0)
    if parent.size == 0:
        parent = np.ones(384) / 384.0
    n = len(parent)
    ps = float(parent.sum())
    if ps < 0.01:
        parent = np.ones(n) / n
    else:
        parent = parent / ps

    best = parent.copy()
    best_sc = _score_fast(parent)
    if not math.isfinite(best_sc):
        best = np.ones(n) / n
        best_sc = _score_fast(best)

    def _try_update(cand):
        nonlocal best, best_sc
        if cand is None:
            return
        cand = np.clip(np.asarray(cand, dtype=np.float64).ravel(), 0, 1000)
        if cand.size == 0:
            return
        cs = float(cand.sum())
        if cs < 0.01:
            return
        try:
            sc = _score_fast(cand)
        except Exception:
            return
        if math.isfinite(sc) and sc < best_sc - 1e-15:
            best = cand.copy()
            best_sc = sc

    rng_main = np.random.default_rng(seed)
    inits = _make_inits(parent, n, rng_main)

    # Phase 1: LP optimization on parent length (55% of time)
    lp_deadline = t0 + budget * 0.55
    time_per_init = max(12.0, (lp_deadline - t0 - 15.0) / len(inits))
    for idx, init in enumerate(inits):
        if time.monotonic() > lp_deadline - 5.0:
            break
        if best_sc < 1.5027:
            break
        stop = min(lp_deadline - 4.0, time.monotonic() + time_per_init)
        try:
            res, sc = _lp_optimize(init, n, stop, seed + 10000 * (idx + 1))
            _try_update(res)
        except Exception:
            pass

    # Phase 1b: Additional LP rounds on best (15% of time)
    lp2_deadline = t0 + budget * 0.70
    if time.monotonic() < lp2_deadline - 10.0:
        extra_inits = [best.copy()]
        # Add perturbed versions of best
        for p in (0.05, 0.1, 0.2):
            extra_inits.append(np.clip(best + rng_main.normal(0, p, n), 0, None))
        for ei in extra_inits:
            if time.monotonic() > lp2_deadline - 5.0:
                break
            if best_sc < 1.5027:
                break
            _try_update(ei)
            ei = ei / ei.sum()
            stop = min(lp2_deadline - 4.0, time.monotonic() + 20.0)
            try:
                res, sc = _lp_optimize(ei, n, stop, seed + 77000 + int(rng_main.integers(1000)))
                _try_update(res)
            except Exception:
                pass

    # Phase 2: Subgradient polish (12% of time)
    sg_deadline = t0 + budget * 0.82
    if time.monotonic() < sg_deadline - 10.0:
        try:
            res, sc = _subgrad_optimize(best, n, sg_deadline, seed + 55001)
            _try_update(res)
        except Exception:
            pass

    # Phase 3: Try nearby lengths (8% of time)
    other_deadline = t0 + budget * 0.90
    if time.monotonic() < other_deadline - 10.0 and best_sc > 1.503:
        for L in (n - 8, n - 4, n + 4, n + 8, n - 16, n + 16):
            if time.monotonic() > other_deadline - 5.0:
                break
            if L < 16:
                continue
            stop = min(other_deadline - 4.0, time.monotonic() + 15.0)
            # Interpolate parent (or best) to new length
            x_old = np.linspace(0, 1, n)
            x_new = np.linspace(0, 1, L)
            init_L = np.interp(x_new, x_old, best)
            if init_L.sum() < 0.01:
                init_L = np.ones(L) / L
            else:
                init_L = init_L / init_L.sum()
            try:
                res, sc = _lp_optimize(init_L, L, stop, seed + 99000 + L)
                _try_update(res)
            except Exception:
                pass

    # Phase 4: Coordinate descent (8% of time)
    cd_deadline = deadline - 10.0
    if time.monotonic() < cd_deadline - 10.0:
        try:
            res, sc = _coord_optimize(best, len(best), cd_deadline, seed + 33001)
            _try_update(res)
        except Exception:
            pass

    # Phase 5: Final subgradient burst if time remains
    if time.monotonic() < deadline - 15.0:
        try:
            res, sc = _subgrad_optimize(best, len(best), deadline - 12.0, seed + 88001)
            _try_update(res)
        except Exception:
            pass

    # Final coordinate polish in remaining time
    if time.monotonic() < deadline - 10.0:
        try:
            res, sc = _coord_optimize(best, len(best), deadline - 8.0, seed + 11001)
            _try_update(res)
        except Exception:
            pass

    # Final validation with exact evaluator
    final_list = [float(max(0.0, min(1000.0, x))) for x in best]
    try:
        sc = evaluate_sequence(final_list)
        if math.isfinite(sc) and sc > 0:
            return final_list
    except Exception:
        pass

    # Fallback: parent
    fb = [float(max(0.0, min(1000.0, x))) for x in parent]
    try:
        sc = evaluate_sequence(fb)
        if math.isfinite(sc) and sc > 0:
            return fb
    except Exception:
        pass

    return [1.0] * 320
