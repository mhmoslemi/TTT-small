```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(384, dtype=np.float64)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    return p


def _conv_fft(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_fft(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_conv_matrix(f, n):
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_direction(f, n, M, time_limit=15.0):
    A = _build_conv_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    gs = float(g.sum())
    if gs < 1e-14:
        return None
    return g / gs


def _line_search_bisect(f, g, best_sc, t_init, stop_time):
    t_lo, t_hi = 0.0, t_init
    f_lo = f.copy()
    for _ in range(20):
        if time.monotonic() > stop_time:
            break
        t_mid = 0.5 * (t_lo + t_hi)
        fn = (1.0 - t_mid) * f + t_mid * g
        np.maximum(fn, 0.0, out=fn)
        fs = float(fn.sum())
        if fs < 1e-10:
            t_hi = t_mid
            continue
        sc = _score(fn)
        if sc < best_sc - 1e-15:
            t_lo = t_mid
            f_lo = fn.copy()
        else:
            t_hi = t_mid
        if t_hi - t_lo < 1e-5:
            break
    if t_lo < 1e-6:
        return None, None
    return t_lo, f_lo


def _lp_optimize(f0, n, stop_time, seed):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    fs = float(f.sum())
    f = f / fs if fs > 1e-10 else np.ones(n) / n
    best_f = f.copy()
    best_sc = _score(f)
    t_cur = 0.30
    stall = 0
    max_stall = 80

    while time.monotonic() < stop_time - 3.0:
        b = _conv_fft(f)
        M = float(b.max())
        if M < 1e-15:
            break

        g_dir = _lp_direction(f, n, M, time_limit=20.0)
        if g_dir is None:
            stall += 5
            if stall > max_stall:
                break
            f = np.clip(f + rng.normal(0, 0.04, n), 0, None)
            fs2 = f.sum()
            f = f / fs2 if fs2 > 1e-10 else best_f.copy()
            t_cur = 0.25
            continue

        dist = float(np.linalg.norm(g_dir - f))
        if dist < 1e-7:
            stall += 3
            if stall > max_stall:
                break
            f = np.clip(f + rng.normal(0, 0.06, n), 0, None)
            fs2 = f.sum()
            f = f / fs2 if fs2 > 1e-10 else best_f.copy()
            t_cur = 0.30
            continue

        res_t, f_new = _line_search_bisect(f, g_dir, best_sc, min(0.5, t_cur * 1.2), stop_time)
        if f_new is not None:
            sc_new = _score(f_new)
            if sc_new < best_sc - 1e-14:
                f = f_new
                best_f = f.copy()
                best_sc = sc_new
                t_cur = min(0.55, res_t * 1.1)
                stall = 0
            else:
                t_cur = max(0.02, res_t * 0.5)
                stall += 1
        else:
            t_cur = max(0.02, t_cur * 0.5)
            stall += 1

        if stall >= max_stall:
            f = np.clip(f + rng.normal(0, 0.08, n), 0, None)
            fs2 = f.sum()
            f = f / fs2 if fs2 > 1e-10 else best_f.copy()
            t_cur = 0.30
            stall = 0

    return best_f, best_sc


def _subgrad_optimize(f0, n, stop_time, seed):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    fs = float(f.sum())
    f = f / fs if fs > 1e-10 else np.ones(n) / n
    best_f = f.copy()
    best_sc = _score(f)
    momentum = np.zeros(n)
    lr = 0.008
    beta = 0.8
    no_imp = 0
    t_total = 0.0

    while time.monotonic() < stop_time - 3.0:
        b = _conv_fft(f)
        M = float(b.max())

        # Use top-K lags with exponential weighting
        K = min(8, 2 * n - 1)
        idx = np.argpartition(b, -K)[-K:]
        idx = idx[np.argsort(b[idx])[::-1]]
        vals = b[idx]
        weights = np.exp(-3.0 * (M - vals))
        weights /= weights.sum()

        grad_M = np.zeros(n)
        for w, k in zip(weights, idx):
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            for i in range(lo, hi):
                j = k - i
                if 0 <= j < n:
                    grad_M[i] += 2.0 * w * f[j]

        s = float(f.sum())
        grad_obj = (2.0 * n / (s * s)) * grad_M - (4.0 * n * M / (s ** 3)) * np.ones(n)

        momentum = beta * momentum - lr * grad_obj
        f_new = f + momentum
        f_new = np.maximum(f_new, 0.0)
        fs_new = float(f_new.sum())
        if fs_new < 1e-10:
            f_new = f.copy()
        else:
            f_new = f_new / fs_new

        sc = _score(f_new)
        if sc < best_sc - 1e-15:
            f = f_new
            best_f = f.copy()
            best_sc = sc
            no_imp = 0
            lr = min(0.03, lr * 1.005)
        else:
            no_imp += 1
            lr = max(0.001, lr * 0.98)
            if no_imp > 400:
                f = np.clip(f + rng.normal(0, 0.03, n), 0, None)
                f = f / f.sum()
                no_imp = 0
                lr = 0.008
                beta = 0.7
                momentum = np.zeros(n)

        if no_imp > 2000:
            break

    return best_f, best_sc


def _coord_polish(f0, n, stop_time, seed):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    fs = float(f.sum())
    f = f / fs if fs > 1e-10 else np.ones(n) / n
    best_sc = _score(f)
    no_imp_rounds = 0
    multipliers = (0.3, 0.5, 0.7, 1.3, 1.5, 2.0, 0.1)

    while time.monotonic() < stop_time - 2.0 and no_imp_rounds < 15:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 2.0:
                return f, best_sc
            orig = f[i]
            for mlt in multipliers:
                f[i] = orig * mlt if orig > 1e-14 else (0.01 if mlt > 0 else 0.0)
                sc = _score(f)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    improved = True
                    orig = f[i]
                else:
                    f[i] = orig
            # Fine tune
            if orig > 1e-8:
                for dt in (0.1, -0.1, 0.2, -0.2):
                    f[i] = max(0.0, orig + dt * orig)
                    sc = _score(f)
                    if sc < best_sc - 1e-14:
                        best_sc = sc
                        improved = True
        # 2-coord moves
        for _ in range(min(60, n // 4)):
            if time.monotonic() > stop_time - 2.0:
                return f, best_sc
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            oi, oj = f[i], f[j]
            for mi in (0.5, 1.5, 0.3, 2.0):
                for mj in (0.5, 1.5, 0.3):
                    f[i] = oi * mi if oi > 1e-14 else 0.01
                    f[j] = oj * mj if oj > 1e-14 else 0.01
                    sc = _score(f)
                    if sc < best_sc - 1e-14:
                        best_sc = sc
                        improved = True
            f[i], f[j] = oi, oj

        if improved:
            no_imp_rounds = 0
        else:
            no_imp_rounds += 1

    return f, best_sc

    sg_end = min(deadline, t0 + budget * 0.85)
    if time.monotonic() < sg_end - 10.0:
        try:
            res, sc = _subgrad_optimize(best_f, n, sg_end, seed + 99001)
            _try_update(res)
        except Exception:
            pass

    # Phase 4: Coordinate polish
    cd_end = deadline - 5.0
    if time.monotonic() < cd_end - 10.0:
        try:
            res, sc = _coord_polish(best_f, n, cd_end, seed + 77001)
            _try_update(res)
        except Exception:
            pass

    # Phase 5: Final subgradient burst
    if time.monotonic() < deadline - 12.0:
        try:
            res, sc = _subgrad_optimize(best_f, n, deadline - 8.0, seed + 33001)
            _try_update(res)
        except Exception:
            pass

    # Final coordinate fine-tune
    if time.monotonic() < deadline - 8.0:
        try:
            res, sc = _coord_polish(best_f, n, deadline - 4.0, seed + 55001)
            _try_update(res)
        except Exception:
            pass

    # Final validation with exact evaluator
    final_list = [float(max(0.0, min(1000.0, x))) for x in best_f]
    try:
        sc = evaluate_sequence(final_list)
        if math.isfinite(sc) and sc > 0:
            return final_list
    except Exception:
        pass

    # Fallback
    fb = [float(max(0.0, min(1000.0, x))) for x in parent]
    try:
        sc = evaluate_sequence(fb)
        if math.isfinite(sc) and sc > 0:
            return fb
    except Exception:
        pass

    return [1.0 / n] * n
```
