    f = rfft(a, nfft)
    c = irfft(f * f, nfft)[:2 * n - 1]
    return np.maximum(np.real(c), 0.0)

def _project_simplex(v):
    n = v.size
    if n == 0:
        return np.array([1.0])
    v = np.asarray(v, dtype=np.float64).ravel()
    v = np.where(np.isfinite(v), v, 0.0)
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    rho = np.max(np.where((css - 1.0) / idx < u)[0] + 1) if np.any((css - 1.0) / idx < u) else 1
    theta = (css[int(rho) - 1] - 1.0) / rho
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-300:
        return np.ones(n) / n
    return w / s

def _build_A(a):
    n = a.size
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for k in range(m):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            A[k, js:je+1] = a[k - je : k - js + 1][::-1]
    return A

def _grad(a, w):
    """Compute gradient g_i = 2 * sum_k w_k * a_{k-i}."""
    n = a.size
    m = 2 * n - 1
    if w.size != m:
        if w.size > m:
            w = w[:m]
        else:
            w = np.concatenate([w, np.zeros(m - w.size)])
    L = next_fast_len(m + n - 1)
    W = rfft(w, L)
    A = rfft(a, L)
    r = irfft(W * np.conj(A), L)
    g = 2.0 * np.real(r[:n])
    return g

    w = np.zeros(c.size)

from scipy.optimize import linprog

def _eSeq(a):
    """Eval wrapper."""
    try:
        a = np.asarray(a, float).ravel()
        if a.size == 0 or not np.all(np.isfinite(a)):
            return float('inf')
        return float(evaluate_sequence([float(v) for v in a]))
    except:

def _conv(a):
    n = a.size
    nf = next_fast_len(2*n-1)
    f = rfft(a, nf)
    c = irfft(f*f, nf)[:2*n-1]
    return np.maximum(np.real(c), 0)

def _ps(x):
    """Project to probability simplex."""
    n = x.size
    x = np.asarray(x, float)
    x = np.where(np.isfinite(x), x, 0)
    u = np.sort(x)[::-1]
    cs = np.cumsum(u)
    rho = np.arange(1, n+1, float)
    cond = (cs - 1)/rho < u
    if not cond.any():
        return np.ones(n)/n
    r = int(np.where(cond)[0][-1]) + 1
    th = (cs[r-1] - 1)/r
    w = np.maximum(x - th, 0)
    s = w.sum()
    return w/s if s > 1e-300 else np.ones(n)/n

def _buildA(a):
    n = a.size
    m = 2*n-1
    A = np.zeros((m,n))
    for k in range(m):
        js = max(0, k-n+1)
        je = min(n-1, k)
        if js <= je:
            A[k, js:je+1] = a[k-je:k-js+1][::-1]
    return A

def _grad(a, w):
    n = a.size
    m = 2*n-1
    if w.size != m:
        w = np.pad(w, (0, m-w.size)) if w.size < m else w[:m]
    L = next_fast_len(m+n-1)
    g = 2.0 * np.real(irfft(rfft(w,L) * np.conj(rfft(a,L)), L))[:n]
    return g

def _sw(c, M, xf=0.05):
    T = max(M*xf, 1e-12)
    z = np.clip((c-M)/T, -50, 0)
    p = np.exp(z)
    s = p.sum()
    return p/s if s > 0 else np.eye(1, c.size, c.argmax())[0]

def _lp(a, M, dl, rng):

    m = 2*n-1
    if time.monotonic() > dl - 3:
        return None, None
    A = _buildA(a)
    As = A/M
    co = -np.ones(n) + 1e-8*rng.standard_normal(n)

        if r.success:
            b = np.maximum(r.x, 0)
            s = b.sum()
            if s > 1e-12: bd = b/s
    except: pass
    if time.monotonic() < dl - 3:

            r2 = linprog(np.ones(m), A_ub=A_d, b_ub=-np.ones(n)/M, bounds=(0,None),
                        method='highs', options={'time_limit':tl,'presolve':True})
            if r2.success:
                wd = np.maximum(r2.x, 0)
                s = wd.sum()
                if s > 1e-12: wd /= s
                else: wd = None
        except: wd = None
    return bd, wd

    a = np.asarray(a, float).ravel().copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-300: return a, float('inf')
    a /= s
    n = a.size

            else: a = _ps(best + rng.normal(0, 0.005, n))

def _make_inits(n, parent, rng):
    x = np.linspace(0, 1, n)
    inits = [parent.copy()]
    # Raised cosine
    rc = (1 + np.cos(2*np.pi*x))/2 + 0.01
    inits.append(rc/rc.sum())
    # Hann
    h = np.sin(np.pi*x)**2 + 0.01
    inits.append(h/h.sum())
    # Hamming
    hm = 0.54 + 0.46*np.cos(2*np.pi*x) + 0.01
    inits.append(hm/hm.sum())
    # Blackman
    bm = 0.42 + 0.5*np.cos(2*np.pi*x) + 0.08*np.cos(4*np.pi*x)
    bm = np.maximum(bm, 0.01)
    inits.append(bm/bm.sum())
    # Exponential taper
    ec = np.exp(-2*(x-0.5)**2*4) + 0.01
    inits.append(ec/ec.sum())
    # Symmetric parent
    sp = (parent + parent[::-1])/2
    sp = np.maximum(sp, 0)
    inits.append(sp/sp.sum())
    return inits

def _resample(a, m):
    n = a.size
    if n == m: return a.copy()
    if n < m:
        idx = (np.arange(m) * n) // m
        counts = np.bincount(idx, minlength=n).astype(float)
        counts[counts == 0] = 1
        b = a[idx] / counts[idx]
    else:
        b = np.zeros(m)
        idx = (np.arange(n) * m) // n
        np.add.at(b, idx, a)
    b = np.maximum(b, 0)
    s = b.sum()
    return b/s if s > 1e-300 else np.ones(m)/m

def _conv(a):
    n = a.size
    nf = next_fast_len(2 * n - 1)
    f = rfft(a, nf)
    c = irfft(f * f, nf)[:2 * n - 1]
    return np.maximum(np.real(c), 0.0)

def _ps(x):
    n = x.size
    x = np.asarray(x, float)
    x = np.where(np.isfinite(x), x, 0.0)
    u = np.sort(x)[::-1]
    cs = np.cumsum(u)
    rho = np.arange(1, n + 1, float)
    cond = (cs - 1.0) / rho < u
    if not cond.any():
        return np.ones(n) / n
    r = int(np.where(cond)[0][-1]) + 1
    th = (cs[r - 1] - 1.0) / r
    w = np.maximum(x - th, 0.0)
    s = w.sum()
    return w / s if s > 1e-300 else np.ones(n) / n

def _buildA(a):
    n = a.size
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for k in range(m):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            A[k, js:je+1] = a[k-je:k-js+1][::-1]
    return A

def _grad(a, w):
    n = a.size
    m = 2 * n - 1
    if w.size != m:
        w = np.pad(w, (0, m - w.size)) if w.size < m else w[:m]
    L = next_fast_len(m + n - 1)
    r = irfft(rfft(w, L) * np.conj(rfft(a, L)), L)
    return 2.0 * np.real(r[:n])

def _sw(c, M, xf=0.05):
    T = max(M * xf, 1e-12)
    z = np.clip((c - M) / T, -50.0, 0.0)
    p = np.exp(z)
    s = p.sum()
    if s > 0:
        return p / s
    w = np.zeros(c.size)
    w[np.argmax(c)] = 1.0
    return w

def _lp(a, M, dl, rng):
    n = a.size
    if n < 2 or n > 600 or M <= 0 or time.monotonic() > dl - 2:
        return None, None
    m = 2 * n - 1
    A = _buildA(a)
    As = A / M
    co = -np.ones(n) + 1e-8 * rng.standard_normal(n)
    tl = max(0.5, min(10.0, (dl - time.monotonic()) * 0.15))
    
    bd = None
    try:
        r = linprog(co, A_ub=As, b_ub=np.ones(m), bounds=(0, None),
                    method='highs', options={'time_limit': tl, 'presolve': True})
        if r is not None and r.success:
            b = np.maximum(np.asarray(r.x, float), 0)
            s = b.sum()
            if s > 1e-10:
                bd = b / s
    except:
        pass
