        for w_k, k in zip(w, active):
            k = int(k)
            # grad[j] += 2 * w_k * a_{k-j} for valid k-j
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            for j in range(lo, hi):
                idx = k - j
                if 0 <= idx < n:
                    grad[j] += 2.0 * w_k * f[idx]
        
        # Full subgradient of J = 2n*M/S^2
        # dJ/da = (2n/S^2) * grad_M - (4n*M/S^3) * 1
        grad_F = (2.0*n/(S*S)) * grad - (4.0*n*M/(S**3)) * np.ones(n)
        
        momentum = beta * momentum - lr * grad_F
        f_new = f + momentum
        f_new = np.maximum(f_new, 0.0)
        fs_new = float(f_new.sum())
        if fs_new < 1e-10:
            f_new = f.copy()
        else:
            f_new = f_new / fs_new
        
        sc = _score(f_new)
        if sc < best_sc - 1e-14:
            f = f_new
            best_f = f.copy()
            best_sc = sc
            no_imp = 0
            lr = min(0.08, lr * 1.02)
        else:
            no_imp += 1
            lr = max(0.002, lr * 0.97)
            if no_imp > 200:
                f = f_new.copy()
                no_imp = 0
                lr = 0.01
                beta = 0.6
                momentum = np.zeros(n)
    
    return best_f, best_sc

def _coord_descent(f0, n, stop_time, seed):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None).astype(np.float64)
    f = f / f.sum()
    best_sc = _score(f)
    no_imp_rounds = 0
    
    while time.monotonic() < stop_time - 3.0:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 3.0:
                return f, best_sc
            orig = f[i]
            # Try scaling this coordinate
            for mlt in (0.3, 0.5, 0.7, 1.3, 1.5, 2.0, 0.0):
                f[i] = orig * mlt if orig > 1e-12 else (0.01 if mlt > 0 else 0.0)
                sc = _score(f)
                if sc < best_sc - 1e-13:
                    best_sc = sc
                    improved = True
                else:
                    f[i] = orig
            # 2-coord
            if not improved and rng.random() < 0.1:
                j = int(rng.integers(n))
                if j == i:
                    continue
                oj = f[j]
                for mi in (0.5, 1.5):
                    for mj in (0.5, 1.5):
                        f[i] = orig * mi if orig > 1e-12 else 0.01
                        f[j] = oj * mj if oj > 1e-12 else 0.01
                        sc = _score(f)
                        if sc < best_sc - 1e-13:
                            best_sc = sc
                            improved = True
                f[i], f[j] = orig, oj
        
        no_imp_rounds = 0 if improved else no_imp_rounds + 1
        if no_imp_rounds >= 15:
            break
    
    return f, best_sc
