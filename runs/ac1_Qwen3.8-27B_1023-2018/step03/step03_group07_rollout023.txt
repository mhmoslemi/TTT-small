    n = len(a0)

        M = float(b.max())
        if M < best_M - 1e-14:

        j = np.arange(n)[None, :]

        A = np.zeros((2*n-1, n))
        A[mask] = a[d[mask]]
        
        # Solve LP

        b_ub = np.full(2*n-1, M)
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                          bounds=[(0, None)]*n,
                          method='highs',
                          options={'time_limit': 3.0})
        except Exception:

            a = np.maximum(a, 1e-15)
            a /= a.sum()
            continue
        
        if res.status != 0 or res.x is None:
            a += 0.01 * rng.standard_normal(n)
            a = np.maximum(a, 1e-15)
            a /= a.sum()
            continue

            a += 0.01 * rng.standard_normal(n)
            a = np.maximum(a, 1e-15)
            a /= a.sum()
            continue

    b = np.convolve(a, a)
    M = float(b.max())
    # Active set
    tol = 1e-10 * M
    active = np.where(b > M - tol)[0]
    # Subgradient: average of 2*S_k(a) for k in active
    s = np.zeros(n)
    for k in active:
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                s[i] += 2.0 * a[j]

            hi = lam

    a_new /= a_new.sum()  # (should already be 1, but normalize for safety)
    return a_new, M, b
