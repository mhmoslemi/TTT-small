import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    return c

def _cross_conv(a, b):
    """Compute a * b (full convolution) using FFT."""
    n = len(a) + len(b) - 1
    fsize = next_fast_len(n)
    F = rfft(a, fsize)
    G = rfft(b, fsize)
    return irfft(F * G, fsize)[:n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _conv_full(a)
    m = float(c.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)

def _lp_solve(f, n, M, time_limit=10):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0"""
    # Build constraint matrix
    # A[k, j] = f[j] if 0 <= k-j < n, else 0
    # This means for column j, A[k, j] = f[j] for k = j, j+1, ..., j+n-1
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo = j
        hi = min(2 * n - 1, j + n)
        A[lo:hi, j] = f[j]
    
    c_obj = -np.ones(n)  # minimize -sum(g)
    b_ub = np.full(2 * n - 1, M)
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
    except Exception:
        return None
    
    if res.status != 0 or res.x is None:
        return None
    
    g = res.x.copy()
    gs = float(g.sum())
    if gs < 1e-14:
        return None
    return g / gs

def _mw_update(f, n, eta, top_k=20):
    """Multiplicative weights update."""
    c = _conv_full(f)
    M = float(c.max())
    
    # Get top-k lags
    if top_k >= 2 * n - 1:
        idx = np.arange(2 * n - 1)
        weights = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        idx = np.argpartition(c, -top_k)[-top_k:]
        w = c[idx].copy()
        w = np.exp(eta * (w - w.max()))
        weights = w / w.sum()
    
    # Build weight vector
    w_vec = np.zeros(2 * n - 1)
    w_vec[idx] = 2.0 * weights  # factor of 2 from gradient
    
    # grad[i] = sum_k w_vec[k] * f[k-i] for valid k-i
    # This is w_vec * rev_f (cross-correlation)
    rev_f = f[::-1]
    grad = _cross_conv(w_vec, rev_f)
    # grad has length (2n-1) + n - 1 = 3n-2, we need the middle n values
    # Actually, let me think about this more carefully.
    # w_vec has length 2n-1, rev_f has length n.
    # conv(w_vec, rev_f) has length 3n-2.
    # grad[i] = sum_k w_vec[k] * rev_f[n-1-k+i]... 
    
    # Actually, let me just compute it directly for the active lags.
    grad = np.zeros(n)
    for w, k in zip(weights, idx):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for i in range(lo, hi):
            j = k - i
            if 0 <= j < n:
                grad[i] += 2.0 * w * f[j]
    
    # Multiplicative update
    f_new = f * np.exp(-0.1 * grad / M)
    fs = float(f_new.sum())
    if fs < 1e-14:
        return f.copy(), M
    return f_new / fs, M

Let me also consider: instead of the LP, I can use a simpler approach. The LP is:
max 1^T g
s.t. A g ≤ M * 1
     g ≥ 0

The KKT conditions give: A^T y = 1 - s (where s is the dual variable for g ≥ 0), Ay ≤ M (primal feasibility), y ≥ 0 (dual feasibility), complementarity.

At the optimum, the set of binding constraints {k: (Ag)_k = M} determines the solution. The number of binding constraints is at most n (by the fundamental theorem of LPs, the optimum is at a vertex with at most n binding constraints).

import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    return irfft(F * F, fsize)[:2 * n - 1]

    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _conv_full(a)
    m = float(np.max(c))
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)

def _subgrad(f, top_k=20):
    """Compute subgradient of max_k (f*f)[k]. Returns (grad, M)."""
    n = len(f)
    c = _conv_full(f)
    M = float(np.max(c))
    if M < 1e-15:
        return np.zeros(n), M
    k = min(top_k, 2 * n - 1)
    idx = np.argpartition(c, -k)[-k:] if k < 2 * n - 1 else np.arange(2 * n - 1)
    w = c[idx].copy()
    w = w - w.min() + 1e-12
    w /= w.sum()
    
    grad = np.zeros(n)
    for wi, ki in zip(w, idx):
        ki = int(ki)
        lo = max(0, ki - n + 1)
        hi = min(n, ki + 1)
        lo_j = ki - hi + 1
        hi_j = ki - lo
        grad[lo:hi] += 2.0 * wi * f[lo_j:hi_j + 1][::-1]
    return grad, M

def _lp_solve(f, n, M, time_limit=5.0):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0. Return normalized g."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j+n, j] = f[j]
    
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
    except Exception:
        return None
    
    if res.status != 0 or res.x is None:
        return None
    g = np.clip(res.x, 0, None)
    gs = float(g.sum())
    if gs < 1e-14:
        return None
    return g / gs

def _line_search(f, g, n, best_sc, t_init=0.3, deadline=None):
    """Find best t in [0, t_init] for f_new = (1-t)*f + t*g."""
    best_t = 0.0
    best_f = None

    for _ in range(15):
        if deadline and time.monotonic() > deadline:
            break
        t_mid = 0.5 * (t_lo + t_hi)
        fn = (1.0 - t_mid) * f + t_mid * g
        np.maximum(fn, 0.0, out=fn)
        fs = float(fn.sum())
        if fs < 1e-10:
            t_hi = t_mid
            continue
        sc = _score(fn)
        if sc < best_sc - 1e-14:
            best_t = t_mid
            best_f = fn.copy()
            t_lo = t_mid
        else:
            t_hi = t_mid
        if t_hi - t_lo < 1e-5:
            break
    return best_t, best_f

            f /= f.sum()
            t_cur = 0.3
            stall = 0
    
    return best_f, best_sc
