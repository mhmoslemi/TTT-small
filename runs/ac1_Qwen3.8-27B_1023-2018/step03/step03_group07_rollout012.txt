import time
import math
from scipy.optimize import linprog


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + budget_s - 25.0

    parent = np.clip(parent, 0.0, 1000.0)

    def get_score(f):
        f = np.asarray(f, dtype=np.float64)
        s = float(f.sum())
        if s < 1e-15:
            return float('inf')
        c = np.convolve(f, f)
        return 2.0 * n * float(c.max()) / (s * s)

    def normalize(f):
        s = float(f.sum())
        if s < 1e-15:
            return np.ones(n) / n
        return f / s

    f0 = normalize(parent.copy())
    best = f0.copy()

        k = np.arange(2 * n - 1)[:, None]
        j = np.arange(n)[None, :]
        idx = k - j
        mask = (idx >= 0) & (idx < n)
        return np.where(mask, f[np.clip(idx, 0, n - 1)], 0.0)

    def solve_lp(A, M):
        c = -np.ones(n)
        b = np.full(A.shape[0], M)
        try:
            res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                          method='highs',
                          options={'time_limit': 10.0,
                                   'primal_feasibility_tolerance': 1e-12,
                                   'dual_feasibility_tolerance': 1e-12,
                                   'mip_rel_gap': 0.0})
        except Exception:
            return None
        if res.status != 0 or res.x is None:
            return None
        g = np.asarray(res.x, dtype=np.float64)
        if not np.all(np.isfinite(g)):
            return None
        gs = float(g.sum())
        if gs < 1e-15:
            return None
        return g / gs

    def max_conv(h):
        return float(np.convolve(h, h).max())

    def golden_line_search(f, g, M_cur):
        def obj(t):
            h = (1.0 - t) * f + t * g
            return max_conv(h)

        val_at_1 = obj(1.0)
        if val_at_1 >= M_cur:
            return f.copy(), M_cur

        lo, hi = 0.0, 1.0
        a_ = hi - gr * (hi - lo)
        b_ = lo + gr * (hi - lo)
        fa = obj(a_)
        fb = obj(b_)

                b_ = a_
                fb = fa

                fa = obj(a_)
            else:

                a_ = b_
                fa = fb
                b_ = lo + gr * (hi - lo)
                fb = obj(b_)

        return h, val

    def subgradient_step(f, k_star, M_cur, rng_local):

        # Fallback: tiny random perturbation
        fh = f * np.exp(rng_local.normal(0, 0.003, n))
        fh = np.maximum(fh, 1e-15)
        fh = normalize(fh)
        return fh, max_conv(fh)

                stuck = 0

                f = normalize(f)
                stuck = 0
                continue

            M_lp = M * (1.0 + 1e-12)
            g = solve_lp(A, M_lp)
