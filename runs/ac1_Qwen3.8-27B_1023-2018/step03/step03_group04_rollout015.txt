import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(384, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0 or p.sum() < 0.01:
        p = np.ones(384, dtype=np.float64)
    return p

def _convolve(a):
    """Full convolution a*a using FFT"""
    n = len(a)
    fsize = next_fast_len(2*n-1)
    F = rfft(a, fsize)
    c = irfft(F*F, fsize)[:2*n-1]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    n = len(a)
    S = float(a.sum())
    if S < 0.01:
        return math.inf
    M = float(_convolve(a).max())
    if not math.isfinite(M):
        return math.inf
    return 2.0 * n * M / (S * S)

def _build_A(f):
    n = len(f)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = f
    return A

def _lp_direction(f, M):
    n = len(f)
    if M < 1e-15:
        return None
    A = _build_A(f)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)]*n,
                      method='highs',
                      options={"presolve": True})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    s = float(g.sum())
    if s < 1e-10:
        return None
    return g / s

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(384, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0 or p.sum() < 0.01:
        p = np.ones(384, dtype=np.float64)
    return p


def _conv(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    S = float(a.sum())
    if S < 0.01:
        return math.inf
    M = float(_conv(a).max())
    if not math.isfinite(M):
        return math.inf
    return 2.0 * n * M / (S * S)


def _build_A(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_dir(f, M):
    n = len(f)
    if M <= 1e-15:
        return None
    A = _build_A(f)
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2 * n - 1, M),
                      bounds=[(0, None)] * n, method='highs',
                      options={"presolve": True, "time_limit": 4,
                               "primal_feasibility_tolerance": 1e-10,
                               "dual_feasibility_tolerance": 1e-10})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    s = float(g.sum())
    if s < 1e-10:
        return None
    return g / s


def _line_search(f, g, n):
    """Golden-section search for best t in (0,1) minimizing score of (1-t)f+tg."""
    gr = (math.sqrt(5) - 1) / 2.0
    a, b = 1e-6, 1.0
    for _ in range(18):
        c = b - gr * (b - a)
        d = a + gr * (b - a)
        fc = _score((1 - c) * f + c * g)
        fd = _score((1 - d) * f + d * g)
        if fc < fd:
            b = d
        else:
            a = c
    t = (a + b) / 2.0
    f_new = np.clip((1 - t) * f + t * g, 0, None)
    f_new = f_new / max(f_new.sum(), 1e-14)
    return t, f_new, _score(f_new)


def _bem(f0, stop, seed, max_gap=100):
    rng = np.random.default_rng(seed)
    n = len(f0)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel(), 0, None).copy()
    s = f.sum()
    f = f / s if s > 1e-10 else np.ones(n) / n

    best_f = f.copy()
    best_sc = _score(f)
    stall = 0
    it = 0

    while time.monotonic() < stop - 3.0 and it < 200000:
        M = float(_conv(f).max())
        g = _lp_dir(f, M)
        if g is None:
            stall += 4
            if stall > max_gap:
                break
            f = np.clip(f + rng.normal(0, 0.04, n), 0, None)
            f = f / max(f.sum(), 1e-14)
            it += 1
            continue

        if float(np.dot(g - f, g - f)) < 1e-16:
            stall += 1
            if stall > max_gap:
                break
            it += 1
            continue

        t, f_new, sc = _line_search(f, g, n)
        if sc < best_sc - 1e-12:
            best_f = f_new.copy()
            best_sc = sc
            f = f_new
            stall = 0
        else:
            stall += 1
            if stall > max_gap:
                f = np.clip(best_f + rng.normal(0, 0.06, n), 0, None)
                f = f / max(f.sum(), 1e-14)
                stall = 0
                if rng.random() < 0.35:
                    f = np.clip(f + rng.normal(0, 0.12, n), 0, None)
                    f = f / max(f.sum(), 1e-14)
        it += 1

    return best_f, best_sc, it


def _polish(f, stop, seed=0):
    rng = np.random.default_rng(seed)
    n = len(f)
    f = np.clip(np.asarray(f, dtype=np.float64).ravel(), 0, None).copy()
    f = f / max(f.sum(), 1e-14)
    best_f = f.copy()
    best_sc = _score(f)
    no_imp = 0

    while time.monotonic() < stop - 3.0:
        improved = False
        for i in range(n):
            if time.monotonic() > stop - 3.0:
                break
            orig = f[i]
            bv = orig
            bvs = best_sc
            muls = [0.0, 0.1, 0.25, 0.5, 0.75, 1.25, 1.5, 2.0, 3.0, 0.4, 1.1, 0.9]
            for m in muls:
                f[i] = orig * m if orig > 1e-12 else (0.0005 / n if m > 0 else 0.0)
                if f.sum() < 0.01:
                    f[i] = orig
                    continue
                sc = _score(f)
                if sc < bvs - 1e-13:
                    bv = f[i]
                    bvs = sc
            f[i] = bv
            if bvs < best_sc - 1e-13:
                best_sc = bvs
                improved = True

        if not improved:
            no_imp += 1
            if no_imp >= 4:
                break
        else:
            no_imp = 0

    return f, best_sc

    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 12.0

    parent = _get_parent()
    best_seq = parent.copy()
    try:
        best_score = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        best_score = math.inf

        best_seq = np.ones(384)
        best_score = float(evaluate_sequence([1.0] * 384))

    def track(f):
        nonlocal best_seq, best_score
        f = np.clip(np.asarray(f, dtype=np.float64).ravel(), 0, 1000)
        if f.size == 0 or f.sum() < 0.01 or not np.all(np.isfinite(f)):
            return
        try:
            sc = float(evaluate_sequence([float(x) for x in f]))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score:
            best_seq = f.copy()
            best_score = sc

    # Phase 1: BEM on parent length (50% of budget)
    p1_end = min(t0 + budget * 0.50, deadline - 120)
    f_init = parent / max(parent.sum(), 1e-10)
    try:
        rf, rsc, nit = _bem(f_init, p1_end, seed)
        track(rf)
    except Exception:
        pass

    # Phase 2: Second BEM from current best (15%)
    p2_end = min(t0 + budget * 0.65, deadline - 80)
    if time.monotonic() < p2_end - 60:
        try:

        except Exception:
            pass

    p3_end = min(t0 + budget * 0.75, deadline - 60)

                     np.exp(-((i_arr - n * 0.75) / (n * 0.12)) ** 2))
