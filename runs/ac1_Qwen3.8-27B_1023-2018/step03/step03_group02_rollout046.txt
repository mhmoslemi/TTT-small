    if p.size == 0:
        p = np.ones(198, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(198, dtype=np.float64)
    return p


def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m <= 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n):
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_solve(f, n, M, top_k=None, time_limit=8.0):
    N = 2 * n - 1
    if top_k is not None and top_k < N:
        b = _conv_full(f)
        idx = np.argsort(-b)[:top_k]
    else:
        idx = np.arange(N)
    A_full = _build_A(f, n)
    A = A_full[idx]
    c = -np.ones(n)
    b_ub = np.full(len(idx), M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
    except Exception:
        return None
    if res.status not in (0, 3) or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g_sum = float(g.sum())
    if g_sum < 1e-12:
        return None
    return g / g_sum


def _subgradient_step(f, n, eta=0.01):
    """One projected subgradient step toward reducing max(f*f)."""
    b = _conv_full(f)
    b_norm = b / b.sum()  # normalize for averaging over active peaks
    active = b > 0.95 * b.max()
    if active.sum() == 0:
        active = b > 0.9 * b.max()
    w = b[active]
    w = w / w.sum()
    # Subgradient: for each active k, grad_k(f) = flip-shift of f
    grad = np.zeros(n)
    for k in np.where(active)[0]:
        # (f*f)_k = sum_j f_j * f_{k-j}
        # d/d f_i = f_{k-i} + f_{k-i} (with bounds)
        # More precisely: for i in [0,n-1], contribution from (f*f)_k is:
        #   f_{k-i} if 0<=k-i<n (from j=i term) + f_{k-i} if 0<=k-i<n (from j=k-i term)
        #   but when i=k-i, it's just f_i
        gk = np.zeros(n)
        for i in range(n):
            ji = k - i
            val = 0.0
            if 0 <= ji < n:
                val += f[ji]
            if 0 <= ji < n and ji != i:
                val += f[ji]
            elif 0 <= ji < n:
                val += f[ji]  # double when ji==i
            gk[i] = val
        grad += w[int(k)] * gk
    # Projected gradient descent
    f_new = f - eta * grad
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-10:
        f_new /= s
    else:
        f_new = f.copy()
    return f_new


def _lp_optimize(f0, n, stop_time, seed, use_focused=False):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel()[:n].copy(), 0, None)
    while f.size < n:
        f = np.concatenate([f, np.zeros(n - f.size)])
    f = f[:n]
    f_sum = float(f.sum())
    if f_sum < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / f_sum
    best_f = f.copy()
    best_score = _score(f)
    b_current = _conv_full(f)
    M = float(b_current.max())
    t = 0.35
    t_min = 0.03
    t_max = 0.9
    stall = 0
    max_stall = 120
    iter_count = 0

    while time.monotonic() < stop_time - 2.0:
        # Choose LP variant
        if use_focused and iter_count % 4 == 0:
            g_dir = _lp_solve(f, n, M, top_k=min(120, 2 * n - 1), time_limit=5.0)
        else:
            g_dir = _lp_solve(f, n, M, time_limit=8.0)

        if g_dir is None:
            stall += 5
            if stall > max_stall:
                break
            scale = 0.04 + 0.06 * min(stall / 40, 1.0)
            f = np.clip(f + rng.normal(0, scale, n), 0, None)
            fs = f.sum()
            if fs > 1e-10:
                f = f / fs
            else:
                f = best_f.copy()
            b_current = _conv_full(f)
            M = float(b_current.max())
            continue

        dist = float(np.linalg.norm(g_dir - f))
        if dist < 1e-8:
            # Try subgradient step instead
            f_alt = _subgradient_step(f, n, eta=0.02)
            sc_alt = _score(f_alt)
            if sc_alt < best_score - 1e-12:
                f = f_alt
                best_f = f.copy()
                best_score = sc_alt
                b_current = _conv_full(f)
                M = float(b_current.max())
                stall = max(0, stall - 1)
            else:
                stall += 4
                if stall > max_stall:
                    # Aggressive perturbation
                    strategy = stall % 5
                    if strategy == 0:
                        f = np.clip(f + rng.normal(0, 0.10, n), 0, None)
                    elif strategy == 1:
                        blk = n // 5
                        f2 = f.copy()
                        f2[:blk], f2[n - blk:] = f[n - blk:].copy(), f[:blk].copy()
                        f = f2
                    elif strategy == 2:
                        kern = np.exp(-0.5 * (np.arange(-5, 6) / 2.0) ** 2)
                        kern /= kern.sum()
                        noise = np.convolve(rng.normal(0, 0.25, n), kern, 'same')
                        f = np.clip(f * (1 + noise), 0, None)
                    elif strategy == 3:
                        f = np.clip(f + np.abs(rng.normal(0, 0.08, n)), 0, None)
                    else:
                        # Block swap
                        bs = rng.integers(20, n - 20)
                        bstart = rng.integers(0, n - bs)
                        f2 = f.copy()
                        f2[bstart:bstart + bs], f2[n - bstart - bs:n - bstart] = \
                            f[n - bstart - bs:n - bstart].copy(), f[bstart:bstart + bs].copy()
                        f = f2
                    fs = f.sum()
                    if fs > 1e-10:
                        f = f / fs
                    else:
                        f = best_f.copy()
                    b_current = _conv_full(f)
                    M = float(b_current.max())
                    t = 0.35
                    stall = 0
            continue

        # Adaptive line search: find best t
        improved = False
        best_t = 0
        best_sc_t = best_score
        for _ in range(20):
            f_new = (1.0 - t) * f + t * g_dir
            f_new = np.clip(f_new, 0, None)
            fs_new = float(f_new.sum())
            if fs_new < 1e-10:
                t = max(t_min, t * 0.5)
                continue
            sc = _score(f_new)
            if sc < best_sc_t - 1e-14:
                best_sc_t = sc
                best_t = t
            if sc < best_score - 1e-14:
                f = f_new
                best_f = f.copy()
                best_score = sc
                b_current = _conv_full(f)
                M = float(b_current.max())
                improved = True
                t = min(t_max, t * 1.06)
                stall = 0
                break
            else:
                t = max(t_min, t * 0.55)

        if not improved:
            # Try the best sub-step even if it didn't beat global best
            if best_t > 0:
                f = (1.0 - best_t) * f + best_t * g_dir
                f = np.clip(f, 0, None)
                fs = f.sum()
                if fs > 1e-10:
                    f = f / fs
                    sc_new = _score(f)
                    if sc_new < best_score:
                        best_f = f.copy()
                        best_score = sc_new
                        b_current = _conv_full(f)
                        M = float(b_current.max())
                        stall = 0
                    else:
                        stall += 1
                else:
                    stall += 1
            else:
                stall += 1

            if stall >= max_stall // 3:
                # Light restart
                f = np.clip(f + rng.normal(0, 0.06, n), 0, None)
                fs = f.sum()
                if fs > 1e-10:
                    f = f / fs
                b_current = _conv_full(f)
                M = float(b_current.max())
                t = 0.35

        iter_count += 1

    return best_f, best_score


def _polish(a, stop_time, seed=99):
    n = len(a)
    best = np.clip(np.asarray(a, dtype=np.float64).ravel(), 0, None).copy()
    fs = best.sum()
    if fs < 1e-10:
        return best, _score(best)
    best = best / fs
    best_sc = _score(best)
    rng = np.random.default_rng(seed)
    no_imp = 0
    max_no_imp = 8

    while time.monotonic() < stop_time - 1.5:
        improved = False

        # 1D coordinate descent with finer grid
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 1.5:
                break
            orig = best[i]
            if orig > 1e-8:
                mults = (0.3, 0.5, 0.7, 0.85, 1.0, 1.1, 1.2, 1.4, 1.7, 2.2, 0.0)
            else:
                mults = (0.005, 0.02, 0.05, 0.1, 0.2, 0.5)
            best_val = orig
            best_val_sc = best_sc
            for mlt in mults:
                nv = orig * mlt if orig > 1e-8 else mlt / n
                best[i] = nv
                if best.sum() < 0.01:
                    best[i] = orig
                    continue
                sc = _score(best)
                if sc < best_val_sc - 1e-14:
                    best_val = nv
                    best_val_sc = sc
            best[i] = best_val
            if best_val_sc < best_sc - 1e-14:
                best_sc = best_val_sc
                improved = True

        # 2-coordinate moves
        n_pairs = min(150, n * 2)
        for _ in range(n_pairs):
            if time.monotonic() > stop_time - 1.5:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            total = oi + oj
            if total < 1e-14:
                continue
            for ratio in (0.2, 0.4, 0.6, 0.8, 0.5, 1.0):
                best[i] = ratio * total
                best[j] = (1 - ratio) * total
                if best.sum() < 0.01:
                    best[i], best[j] = oi, oj
                    continue
                sc = _score(best)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    improved = True
            best[i], best[j] = oi, oj

        # 3-coordinate moves
        for _ in range(40):
            if time.monotonic() > stop_time - 1.5:
                break
            idx = rng.choice(n, size=3, replace=False)
            i, j, k = int(idx[0]), int(idx[1]), int(idx[2])
            oi, oj, ok = best[i], best[j], best[k]
            rem = oi + oj + ok
            if rem < 1e-14:
                continue
            for ri in (0.3, 0.5, 0.7, 1.0):
                for rj in (0.4, 0.6, 0.8, 1.0):
                    ni = ri * oi
                    nj = rj * oj
                    nk = max(0.0, rem - ni - nj)
                    best[i], best[j], best[k] = ni, nj, nk
                    sc = _score(best)
                    if sc < best_sc - 1e-14:
                        best_sc = sc
                        improved = True
            best[i], best[j], best[k] = oi, oj, ok

        if not improved:
            no_imp += 1
            if no_imp >= max_no_imp:
                break
        else:
            no_imp = 0

    return best, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 12.0

    parent = _get_parent()
    n_p = len(parent)
    n = n_p

    # Fallback
    fallback = parent.copy()
    try:
        fallback_sc = _score(fallback)
    except Exception:
        fallback_sc = math.inf
    if not math.isfinite(fallback_sc):
        fallback = np.ones(n_p) / n_p
        fallback_sc = _score(fallback)

    # Scale fallback for scoring (score is scale-invariant)
    best_seq = fallback.copy()
    best_score = fallback_sc

    i_arr = np.arange(n, dtype=np.float64)
    rng_main = np.random.default_rng(seed)

    # Diverse initializations
    inits = []
    inits.append(parent.copy())  # Parent

    # Cosine taper
    taper = 0.5 + 0.5 * np.cos(np.pi * i_arr / (n - 1))
    inits.append(taper)

    # Gaussian-like
    center = (n - 1) / 2.0
    gauss = np.exp(-0.5 * ((i_arr - center) / (n * 0.2)) ** 2)
    inits.append(gauss + 0.001)

    # Multi-frequency
    sup = np.ones(n) * 0.15
    for fr in (1, 2, 3, 5, 8):
        ph = rng_main.uniform(0, 2 * math.pi)
        sup += 0.25 * np.cos(2 * math.pi * fr * i_arr / n + ph)
    inits.append(np.abs(sup) + 0.005)

    # Smooth random
    base = rng_main.normal(0, 1, n)
    kern = np.exp(-0.5 * (np.arange(-8, 9) / 3.0) ** 2)
    kern /= kern.sum()
    smoothed = np.convolve(base, kern, 'same')
    inits.append(np.abs(smoothed) + 0.005)

    # Parent + noise
    noise_scale = 0.10 * max(1.0, np.std(parent))
    inits.append(np.abs(parent + rng_main.normal(0, noise_scale, n)) + 0.001)

    # Bimodal
    c1, c2 = n * 0.25, n * 0.75
    bi = np.exp(-0.5 * ((i_arr - c1) / (n * 0.1)) ** 2) + np.exp(-0.5 * ((i_arr - c2) / (n * 0.1)) ** 2)
    inits.append(bi + 0.001)

    # Zipper
    z = np.ones(n)
    z[::2] *= 1.6
    z[1::2] *= 0.5
    inits.append(z)

    # Phase 1: LP optimization from diverse inits (55% of time)
    phase1_end = t0 + budget * 0.55
    time_per_init = max(15.0, (phase1_end - t0) / len(inits))

    for idx, init in enumerate(inits):
        if time.monotonic() > phase1_end - 6.0:
            break
        if best_score < 1.503:
            break
        stop = min(phase1_end - 3.0, time.monotonic() + time_per_init)
        if time.monotonic() >= stop:
            break
        try:
            res, sc = _lp_optimize(init, n, stop, seed + 7919 * (idx + 1),
                                   use_focused=(idx > 3))
            if np.all(np.isfinite(res)) and res.sum() > 0.01:
                res_clipped = np.clip(res, 0, 1000)
                if sc < best_score:
                    exact = float(evaluate_sequence([float(x) for x in res_clipped]))
                    if math.isfinite(exact) and exact < best_score:
                        best_seq = res_clipped.copy()
                        best_score = exact
        except Exception:
            pass

    # Phase 2: More LP iterations from best so far (20% of time)
    phase2_end = t0 + budget * 0.75
    if time.monotonic() < phase2_end - 10.0 and best_score >= 1.503:
        stop2 = phase2_end - 5.0
        try:
            res, sc = _lp_optimize(best_seq, n, stop2, seed + 104729,
                                   use_focused=True)
            if np.all(np.isfinite(res)) and res.sum() > 0.01:
                res_clipped = np.clip(res, 0, 1000)
                if sc < best_score:
                    exact = float(evaluate_sequence([float(x) for x in res_clipped]))
                    if math.isfinite(exact) and exact < best_score:
                        best_seq = res_clipped.copy()
                        best_score = exact
        except Exception:
            pass

        # Second round from slightly perturbed best
        if time.monotonic() < phase2_end - 15.0 and best_score >= 1.503:
            rng2 = np.random.default_rng(seed + 555)
            perturbed = np.clip(best_seq + rng2.normal(0, 0.03, n), 0, None)
            perturbed /= perturbed.sum()
            stop2b = phase2_end - 8.0
            try:
                res, sc = _lp_optimize(perturbed, n, stop2b, seed + 209459)
                if np.all(np.isfinite(res)) and res.sum() > 0.01:
                    res_clipped = np.clip(res, 0, 1000)
                    if sc < best_score:
                        exact = float(evaluate_sequence([float(x) for x in res_clipped]))
                        if math.isfinite(exact) and exact < best_score:
                            best_seq = res_clipped.copy()
                            best_score = exact
            except Exception:
                pass

    # Phase 3: Polish (25% of remaining time)
    phase3_end = deadline - 3.0
    if time.monotonic() < phase3_end - 10.0:
        polish_time_per = max(10.0, (phase3_end - time.monotonic()) / 3)
        for pidx in range(3):
            p_stop = time.monotonic() + polish_time_per
            if p_stop > phase3_end:
                p_stop = phase3_end
            if time.monotonic() >= p_stop - 1.0:
                break
            try:
                best_seq, best_score = _polish(best_seq, p_stop, seed=5000 + pidx * 111)
            except Exception:
                pass

    # Final short polish
    final_stop = t0 + budget - 5.0
    if time.monotonic() < final_stop - 4.0:
        try:
            best_seq, _ = _polish(best_seq, final_stop, seed=77777)
        except Exception:
            pass

    # Re-evaluate with exact score
    final_seq = np.clip(best_seq, 0, 1000)
    try:
        final_sc = float(evaluate_sequence([float(x) for x in final_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in fallback]

    return [float(x) for x in final_seq]
