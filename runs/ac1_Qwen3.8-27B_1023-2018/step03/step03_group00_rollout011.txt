import numpy as np
import time
import math
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, next_fast_len


def _conv(a):
    n = len(a)
    if n <= 2048:
        return np.convolve(a, a, mode="full")
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n):
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_dir(f, n, M, timeout=4.0):
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': timeout, 'dual_feasibility_tolerance': 1e-10,
                               'primal_feasibility_tolerance': 1e-10})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.clip(np.asarray(res.x, dtype=np.float64), 0, None)
    if g.sum() < 1e-15:
        return None
    return g

    f = f.copy()
    cur = _score(f)

            else:
                base = max(1e-3, 0.5 * f.mean()) if f.mean() > 1e-12 else 1e-3
                for m in [0.5, 1.0, 2.0, 5.0]:

                    f[i] = orig
                    continue
                sc = _score(f)

                    best_v = v
            f[i] = best_v

                improved_any = True
