    if v is None:

        _fft_cache[m] = v
    return v

def _conv_same(a):
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    m = 2*n - 1
    L = _fft_len(m)
    fa = rfft(a, L)
    c = irfft(fa * fa, L)[:m]

def _LP(f, stop):
    """LP direction: minimize s s.t. (f*g)_k <= s, sum(g)=1, g>=0"""
    n = f.size
    if time.monotonic() > stop: return None
    fp = np.maximum(f, 1e-10)
    fp /= fp.sum()
    
    # Build A: (2n-1) x (n+1)
    A = np.zeros((2*n-1, n+1))
    for j in range(n):
        A[j:j+n, j] = fp
    A[:, n] = -1.0
    
    obj = np.zeros(n+1); obj[n] = 1.0
    Aeq = np.zeros((1, n+1)); Aeq[0, :n] = 1.0
    
    tl = max(0.5, stop - time.monotonic() - 5)
    if tl < 0.5: return None
    
    try:
        r = linprog(obj, A_ub=A, b_ub=np.zeros(2*n-1), A_eq=Aeq, b_eq=[1.0],
                   bounds=[(0,None)]*(n+1), method='highs',
                   options={'time_limit': tl})
    except: return None
    
    if r.status != 0 or r.x is None: return None
    g = np.clip(r.x[:n], 0, None)
    return g/g.sum() if g.sum() > 1e-12 else None

def _TS(cf, cfs, cg):
    """Best t for blending"""
    B = 2*(cfs - cf)
    Cc = cf - 2*cfs + cg
    ts = np.linspace(0, 1, 501)
    v = np.maximum(cf[None,:] + ts[:,None]*B + ts[:,None]**2*Cc, axis=1)
    i = np.argmin(v)
    lo, hi = ts[max(0,i-5)], ts[min(500,i+5)]
    ts2 = np.linspace(lo, hi, 501)
    v2 = np.maximum(cf[None,:] + ts2[:,None]*B + ts2[:,None]**2*Cc, axis=1)
    return float(ts2[np.argmin(v2)])

    pass

_FC = {}
def _FL(m):
    m = int(m)
    if m not in _FC:
        _FC[m] = int(next_fast_len(m))
    return _FC[m]

def _C(a):
    n = a.size
    L = _FL(2 * n - 1)
    fa = rfft(a, L)
    return np.maximum(irfft(fa * fa, L)[:2 * n - 1], 0.0)

def _C2(a, b):
    L = _FL(a.size + b.size - 1)
    return np.maximum(irfft(rfft(a, L) * rfft(b, L), L)[:a.size + b.size - 1], 0.0)

def _VC(p, a):
    n = a.size
    pr = p[::-1]
    full = _C2(a, pr)
    return full[2 * n - 2 :: -1][:n]

def _N(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:

def _best_t(cf, cfs, cg):

        return best, bsc

            break
        c = _C(a)
        Mc = float(c.max())
        if Mc < 1e-15:
            break

            if time.monotonic() > stop:
                raise _TO()
            u = np.asarray(u, np.float64)
            u = np.clip(np.nan_to_num(u), -60.0, 60.0)

            c = _C(al)

                return 1e100, np.zeros(_n)

            corr = np.zeros(_n)

    # Phase 4: L-BFGS again with higher beta
    p4_stop = min(deadline - 150.0, t0 + 0.70 * budget)
    if p4_stop - time.monotonic() > 10.0:
        cur_n2 = int(best_seq.size)
        f, sc = _LBFGS(best_seq, (400.0, 1200.0), p4_stop, maxfun=700)
        _try(f)

    # Phase 5: Try other lengths
    p5_stop = min(deadline - 60.0, t0 + 0.85 * budget)
    cur_best_n = int(best_seq.size)
    other_lengths = []
    for L in (192, 320, 384, 256, 512, 224, 288):
        if L != cur_best_n and 64 <= L <= 512:
            other_lengths.append(L)
    other_lengths = other_lengths[:4]
    rng2 = np.random.default_rng(seed + 999)
    for k, L in enumerate(other_lengths):
        rem = p5_stop - time.monotonic()
        if rem < 25.0:
            break
        alloc = max(25.0, min(120.0, (p5_stop - t0) * 0.15 / max(1, len(other_lengths) - k)))
        stopL = min(p5_stop - 10.0, time.monotonic() + alloc)
        # Two inits: interpolated and random
        init1 = _interp(best_seq, L)
        init2 = _N(rng2.exponential(1.0, size=L) + 0.01)
        for init in (init1, init2):
            if time.monotonic() > stopL - 8.0:
                break
            slp = min(stopL - 5.0, time.monotonic() + max(15.0, 0.45 * (stopL - time.monotonic())))
            f, sc = _LP_run(init, slp, seed=seed + L + k * 7)
            _try(f)
            if time.monotonic() < stopL - 5.0 and L <= 384:
                slp2 = min(stopL - 3.0, time.monotonic() + max(8.0, 0.3 * (stopL - time.monotonic())))
                f2, sc2 = _LBFGS(f, (60.0, 200.0), slp2, maxfun=400)
                _try(f2)

    # Phase 6: Final LP + L-BFGS polish at best length
    p6_stop = deadline - 15.0
    cur_n3 = int(best_seq.size)
    if p6_stop - time.monotonic() > 20.0 and cur_n3 <= 512:
        f, sc = _LP_run(best_seq, min(p6_stop - 15.0, time.monotonic() + max(10.0, 0.2 * (p6_stop - time.monotonic()))),
                        seed=seed + 5555)
        _try(f)
        if time.monotonic() < p6_stop - 8.0:
            f, sc = _LBFGS(best_seq, (800.0, 2000.0), p6_stop - 8.0, maxfun=500)
            _try(f)

    # Phase 7: Coordinate polish
    if time.monotonic() < deadline - 5.0:
        f, sc = _CP(best_seq, deadline - 3.0, seed=88888)
        _try(f)

    # Final verification
    seq = np.clip(np.asarray(best_seq, np.float64), 0.0, 1000.0)
    if seq.size == 0 or seq.sum() < 1e-8:
        seq = np.ones(32) / 32.0
    lst = [float(x) for x in seq]
    sc_final = _ES(lst)
    if not math.isfinite(sc_final):
        fb = _N(parent)
        lst = [float(x) for x in fb]
        sc_final = _ES(lst)
        if not math.isfinite(sc_final):
            lst = [1.0 / 32.0] * 32

    return lst
