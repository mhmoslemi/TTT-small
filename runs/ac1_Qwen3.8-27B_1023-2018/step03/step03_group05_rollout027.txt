def _normalize(a, target=1.0):

def _project_simplex(v, target=1.0):

def _build_conv_mat(a):
    """Build the convolution matrix A such that A @ b = conv(a, b)."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for k in range(n):
        A[k, :k + 1] = a[:k + 1][::-1]
    for k in range(n, m):
        L = m - k
        s = k - n + 1
        A[k, s:s + L] = a[n - 1:n - L - 1:-1]
    return A


def _lp_direction(a, M, deadline, rng):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0. Return b/sum(b) or None."""
    n = a.size
    if n > 768 or time.monotonic() >= deadline:
        return None
    try:
        A = _build_conv_mat(a) / M
        m = 2 * n - 1
        c_obj = -np.ones(n, dtype=np.float64)
        # Add tiny tie-breaking
        c_obj += 1e-6 * rng.standard_normal(n)
        b_ub = np.ones(m, dtype=np.float64)
        tl = max(0.5, min(15.0, 0.5 * (deadline - time.monotonic())))
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0.0, None),
                      method="highs", options={"time_limit": tl, "presolve": True})
        if res is not None and getattr(res, "success", False):
            x = np.asarray(res.x, dtype=np.float64)
            x = np.maximum(x, 0.0)
            s = float(x.sum())
            if s > 1e-12:
                return x / s
    except Exception:
        pass
    return None

                    M = float(c.max())

        # Update best
        sc = _fast_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            patience = 0
            step = min(0.15, step * 1.05)
            T = max(0.005, T * 0.99)
        else:
            patience += 1
            step = max(0.01, step * 0.95)

        if patience > 50:
            # Explore
            r = rng.random()
            if r < 0.5:
                a = _project_simplex(a + rng.normal(0, 0.03, n), 1.0)
            elif r < 0.8:
                x = np.linspace(0, 1, n)
                a = _project_simplex(0.6 * best_a + 0.4 * (x ** rng.uniform(0.5, 3.0)), 1.0)
            else:
                a = _project_simplex(best_a + rng.normal(0, 0.05, n), 1.0)
            patience = 0
            step = 0.05
            T = 0.03

    return best_a, min(best_score, _fast_score(best_a))


def _generate_starts(n, rng, parent):
    """Generate diverse initial points."""
    starts = []
    if parent is not None and parent.size == n:
        starts.append(_normalize(parent))
    elif parent is not None:
        # Resample parent to n
        p = _normalize(parent)
        np_ = p.size
        if np_ > 0:
            idx = (np.arange(n) * np_) // n
            b = p[np.clip(idx, 0, np_ - 1)]
            starts.append(_normalize(b))

    x = np.linspace(0, 1, n)

    # Somewhat uniform with mild modulation
    for k in (1, 2, 3):
        amp = rng.uniform(0.1, 0.4)
        ph = rng.uniform(0, 2 * np.pi)
        starts.append(_normalize(1.0 + amp * np.cos(2 * np.pi * k * x + ph)))

    # Power laws
    for alpha in (0.5, 1.5, 2.5, -0.5, -1.0):
        starts.append(_normalize(np.maximum(x ** alpha, 1e-10) + 1e-10))
        starts.append(_normalize(np.maximum((1 - x) ** alpha, 1e-10) + 1e-10))

    # Gaussian bumps
    for _ in range(4):
        c = rng.uniform(0.2, 0.8)
        w = rng.uniform(0.05, 0.3)
        starts.append(_normalize(np.exp(-((x - c) ** 2) / (2 * w * w)) + 0.05))

    # Two bumps
    for _ in range(3):
        c1 = rng.uniform(0.15, 0.4)
        c2 = 1.0 - c1
        w = rng.uniform(0.05, 0.15)
        starts.append(_normalize(
            np.exp(-((x - c1) ** 2) / (2 * w * w)) +
            np.exp(-((x - c2) ** 2) / (2 * w * w)) + 0.05))

    # Smoothed random
    for _ in range(2):
        rough = rng.uniform(0.3, 1.5, n)
        # Smooth with a simple moving average
        k = max(3, n // 20)
        kernel = np.ones(k) / k
        smooth = np.convolve(rough, kernel, mode="same")
        starts.append(_normalize(smooth + 0.1))

    return starts


def _resample_to(a, m):
    """Resample a to length m."""
    a = _normalize(a)
    n = a.size
    if n == m:
        return a.copy()
    if n == 0:
        return np.ones(m, dtype=np.float64) / m
    if m > n:
        idx = (np.arange(m) * n) // m
        counts = np.bincount(np.clip(idx, 0, n - 1), minlength=n).astype(np.float64)
        counts = np.maximum(counts, 1.0)
        b = a[np.clip(idx, 0, n - 1)] / counts[np.clip(idx, 0, n - 1)]
    else:
        b = np.zeros(m, dtype=np.float64)
        idx = (np.arange(n) * m) // n
        idx = np.clip(idx, 0, m - 1)
        np.add.at(b, idx, a)
    return _normalize(b)


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = max(1.0, float(budget_s))
    except Exception:
        budget = 1300.0

    headroom = max(8.0, min(30.0, 0.025 * budget))
    deadline = t0 + budget - headroom

    rng = np.random.default_rng(int(seed) if np.isfinite(float(seed)) else 42)

    parent = _get_parent()

    best_a = None
    best_score = float("inf")

    def _update_best(a):
        nonlocal best_a, best_score
        sc = _exact_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = _normalize(a).copy()
        elif best_a is None:
            best_a = _normalize(a).copy()
            best_score = _fast_score(best_a)

    # Initial: parent
    if parent is not None:
        _update_best(parent)

    # Fallback initializations if no parent
    if best_score >= 2.0:
        for n_fb in (64, 128, 256):
            _update_best(np.ones(n_fb, dtype=np.float64))

    if best_a is None:
        best_a = np.ones(32, dtype=np.float64) / 32.0
        best_score = _exact_score(best_a)
        if not np.isfinite(best_score):
            return [1.0]

    # Determine target lengths to try
    n_parent = best_a.size if best_a.size > 0 else 512
    lengths = [512]
    if n_parent != 512 and 32 <= n_parent <= 1024:
        lengths.append(n_parent)
    lengths.extend([384, 768])
    # Deduplicate, keep order
    seen = set()
    unique_lengths = []
    for L in lengths:
        if L not in seen and 16 <= L <= 1024:
            unique_lengths.append(L)
            seen.add(L)

    # Allocate time per length
    time_alloc = {}
    total_w = 0.0
    for L in unique_lengths:
        if L == 512:
            w = 3.0
        elif L == 768:
            w = 1.0
        elif L == 384:
            w = 1.5
        else:
            w = 0.8
        time_alloc[L] = w
        total_w += w

    phase_budget = max(0.0, deadline - time.monotonic()) * 0.95

    for L in unique_lengths:
        if time.monotonic() >= deadline - 30.0:
            break
        tL = (time_alloc[L] / total_w) * phase_budget
        tL = min(tL, deadline - time.monotonic() - 25.0)
        if tL < 15.0:
            continue
        L = int(L)
        n = L

        # Generate starts
        starts = _generate_starts(n, rng, parent)
        # Also resample current best
        starts.append(_resample_to(best_a, n))
        # Cap number of starts based on time
        max_starts = max(2, min(len(starts), int(tL / 25.0) + 1))
        starts = starts[:max_starts]

        use_lp = (n <= 640 and tL > 60.0)

        for si, s in enumerate(starts):
            sub_deadline = time.monotonic() + min(
                tL / max(1, len(starts)) - 1.0, tL * 0.45
            )
            if sub_deadline <= time.monotonic() + 5.0:
                break
            if time.monotonic() >= deadline - 25.0:
                break

            try:
                # Phase 1: Smoothed L-BFGS (short)
                t_smooth = min(20.0, (sub_deadline - time.monotonic()) * 0.3)
                if t_smooth > 5.0:
                    a_sm, sc_sm = _optimize_smoothed(
                        s, n, T_init=0.05, T_min=0.005,
                        deadline=time.monotonic() + t_smooth, rng=rng, max_inner=50
                    )
                    if sc_sm < _fast_score(s):
                        s = a_sm

                # Phase 2: Projected gradient
                t_pg = min(25.0, (sub_deadline - time.monotonic()) * 0.4)
                if t_pg > 5.0:
                    a_pg, sc_pg = _projected_grad_ascent(
                        s, n, deadline=time.monotonic() + t_pg,
                        rng=rng, T_init=0.03, max_iter=200
                    )
                    if sc_pg < _fast_score(s):
                        s = a_pg

                # Phase 3: Local refine with LP
                if time.monotonic() < sub_deadline - 3.0:
                    a_lr, sc_lr = _local_refine(
                        s, n, deadline=min(sub_deadline, time.monotonic() + (sub_deadline - time.monotonic())),
                        rng=rng, use_lp=use_lp, lp_interval=3
                    )
                    if sc_lr < _fast_score(s):
                        s = a_lr

                _update_best(s)
            except Exception:
                pass

    # Final polish on the best candidate
    if time.monotonic() < deadline - 20.0:
        pol_time = min(300.0, deadline - time.monotonic() - 12.0)
        if pol_time > 30.0:
            n_best = best_a.size
            use_lp_final = n_best <= 640
            try:
                a_pol, sc_pol = _local_refine(
                    best_a, n_best,
                    deadline=time.monotonic() + pol_time,
                    rng=rng, use_lp=use_lp_final, lp_interval=4
                )
                if sc_pol < _fast_score(best_a):
                    _update_best(a_pol)
            except Exception:
                pass

            # One more round with perturbation
            if time.monotonic() < deadline - 10.0:
                for _ in range(3):
                    if time.monotonic() >= deadline - 8.0:
                        break
                    a_pert = _project_simplex(best_a + rng.normal(0, 0.01, best_a.size), 1.0)
                    a_pol2, sc_pol2 = _local_refine(
                        a_pert, best_a.size,
                        deadline=time.monotonic() + min(60.0, deadline - time.monotonic() - 8.0),
                        rng=rng, use_lp=False, lp_interval=5
                    )
                    if sc_pol2 < _fast_score(best_a):
                        _update_best(a_pert)
                        _update_best(a_pol2)

    # Final validation
    try:
        out = np.asarray(best_a, dtype=np.float64).ravel()
        out = np.clip(out, 0.0, 1000.0)
        out = _normalize(out)
        if out.size == 0 or not np.all(np.isfinite(out)):
            raise ValueError
        fs = _exact_score(out)
        if np.isfinite(fs) and fs >= best_score:
            # Re-validate
            pass
    except Exception:
        out = np.ones(32, dtype=np.float64) / 32.0

    out = np.clip(np.asarray(out, dtype=np.float64), 0.0, 1000.0)
    out = _normalize(out)
    if out.size == 0:
        return [1.0]

    # Final safety check
    final_sc = _exact_score(out)
    if not np.isfinite(final_sc):
        fb = np.ones(64, dtype=np.float64) / 64.0
        return [float(x) for x in fb]

    return [float(x) for x in out]
