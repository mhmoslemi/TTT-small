    def lp_optimize(f0, stop_time, max_iters=5000, seed=0):
        rng_l = np.random.default_rng(seed)
        f = np.clip(np.asarray(f0, dtype=np.float64), 0, None).copy()
        fs = float(f.sum())
        if fs < 1e-10:
            f = np.ones(len(f))
        else:
            f = f / fs
        n = len(f)
        conv = fft_conv(f)
        M = float(conv.max())
        local_best = f.copy()
        local_best_score = 2.0 * n * M  # score with sum=1
        
        t = 0.3
        stall = 0
        max_stall = 200
        it = 0
        
        while time.monotonic() < stop_time - 5.0 and it < max_iters:
            g = lp_step(f, M, tl=2.0)
            if g is None:
                stall += 10
                if stall > max_stall:
                    break
                # Perturb
                f = np.clip(f + rng.normal(0, 0.02, n), 0, None)
                f /= f.sum()
                conv = fft_conv(f)
                M = float(conv.max())
                it += 1
                continue
            
            # Check if g is essentially f
            dist = float(np.linalg.norm(g - f))
            if dist < 1e-8:
                stall += 5
                if stall > max_stall:
                    break
                f = np.clip(f + rng.normal(0, 0.03, n), 0, None)
                f /= f.sum()
                conv = fft_conv(f)
                M = float(conv.max())
                it += 1
                continue
            
            # Line search for best t
            best_t = t
            best_sc = local_best_score
            for trial in range(8):
                f_new = (1 - best_t) * f + best_t * g
                f_new = np.clip(f_new, 0, None)
                fs_new = float(f_new.sum())
                if fs_new < 1e-10:
                    best_t *= 0.5
                    continue
                sc_new = 2.0 * n * float(fft_conv(f_new).max()) / (fs_new * fs_new)
                # Normalize: since we fixed sum=1 in the LP, f_new should have sum ~1
                # Actually (1-t)*f + t*g where both have sum 1, so sum of f_new = 1
                if sc_new < best_sc - 1e-14:
                    best_sc = sc_new
                else:
                    best_t *= 0.5
                    break
            
            # Accept if improved
            if best_sc < local_best_score - 1e-14:
                f = (1 - best_t) * f + best_t * g
                f = np.clip(f, 0, None)
                f /= f.sum()
                local_best = f.copy()
                local_best_score = best_sc
                conv = fft_conv(f)
                M = float(conv.max())
                stall = 0
                t = min(0.7, t * 1.05)
            else:
                stall += 1
                t = max(0.05, t * 0.5)
                if stall >= max_stall:
                    # Bigger perturbation
                    f = np.clip(f + rng.normal(0, 0.08, n), 0, None)
                    f /= f.sum()
                    conv = fft_conv(f)
                    M = float(conv.max())
                    stall = 0
                    t = 0.3
            
            it += 1
        
        return local_best, local_best_score
