                f = _normalize(f + rng.normal(0, 0.005, n))

    return best_f, best_sc


def _tightened_lp_iterate(f, n, rng, deadline, min_time_left=30.0):
    """LP with tightened constraints on near-peak lags"""
    best_f = f.copy()
    best_sc = _score1(f)
    no_improve = 0
    max_stall = 40
    tighten_schedule = [1.0, 0.98, 0.96, 0.94, 0.92, 0.90]
    ti = 0

    while time.monotonic() < deadline - min_time_left:
        c_ff = _conv_same(f)
        M = float(c_ff.max())
        if M <= 1e-15:
            break

        # Tighten constraints on near-peak lags
        frac = tighten_schedule[min(ti, len(tighten_schedule) - 1)]
        b_ub = np.where(c_ff > 0.9 * M, M * frac, M)

        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = f[j]
        c_obj = -np.ones(n)

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0.0, None)] * n,
                          method='highs', options={'presolve': True})
        except Exception:
            no_improve += 1
            if no_improve > max_stall:
                break
            f = _normalize(f + rng.normal(0, 0.01, n))
            continue

        if res.status != 0 or res.x is None:
            no_improve += 1
            ti = 0  # reset tightening
            if no_improve > max_stall:
                break
            f = _normalize(f + rng.normal(0, 0.01, n))
            continue

        g = np.maximum(res.x, 0.0)
        gs = g.sum()
        if gs < 1e-12:
            no_improve += 1
            ti = 0
            if no_improve > max_stall:
                break
            f = _normalize(f + rng.normal(0, 0.01, n))
            continue
        g = g / gs

        t_best = _line_search(f, g)
        improved = False
        if t_best > 1e-6:
            new_f = np.maximum((1.0 - t_best) * f + t_best * g, 0.0)
            ns = new_f.sum()
            if ns > 1e-12:
                new_f = new_f / ns
                new_sc = _score1(new_f)
                if new_sc < best_sc - 1e-12:
                    best_sc = new_sc
                    best_f = new_f.copy()
                    f = new_f.copy()
                    no_improve = 0
                    improved = True
        if not improved:
            no_improve += 1
            if no_improve % 10 == 0:
                ti = min(ti + 1, len(tighten_schedule) - 1)
            if no_improve > max_stall:
                f = _normalize(np.abs(f + rng.normal(0, 0.05, n)) + 1e-6)
                no_improve = 0
                ti = 0
            else:
                f = _normalize(f + rng.normal(0, 0.008, n))

    return best_f, best_sc


def _coordinate_polish(a, deadline, min_time_left=5.0, max_coord=200):
    """Coordinate descent polish"""
    a = a.copy()
    n = len(a)
    best_sc = _score1(a)
    default_val = 1.0 / n

    passes = 0
    while time.monotonic() < deadline - min_time_left and passes < max_coord:
        passes += 1
        improved = False
        for i in range(n):
            if time.monotonic() > deadline - min_time_left:
                break
            orig = a[i]
            vals = {orig, 0.0, default_val, orig * 0.5, orig * 1.5,
                    orig * 0.2, orig * 2.0, orig - 0.3 * default_val,
                    orig + 0.3 * default_val}
            best_v = orig
            best_v_sc = best_sc
            for v in vals:
                v = max(0.0, min(1000.0, v))
                a[i] = v
                s = a.sum()
                if s < 1e-12:
                    a[i] = best_v
                    continue
                an = a / s
                sc = _score1(an)
                if sc < best_v_sc:
                    best_v_sc = sc
                    best_v = v
            a[i] = best_v
            if best_v_sc < best_sc - 1e-14:
                a = a / a.sum()
                best_sc = best_v_sc
                improved = True
        if not improved:
            break
    return a, best_sc


def _subgradient_descent(f, n, deadline, min_time_left=10.0, max_iters=50000):
    """Subgradient descent on the peak autocorrelation"""
    best_f = f.copy()
    best_sc = _score1(f)
    alpha = 0.01
    iter_count = 0

    while time.monotonic() < deadline - min_time_left and iter_count < max_iters:
        iter_count += 1
        c_ff = _conv_same(f)
        M = c_ff.max()
        peak_idx = np.where(c_ff >= 0.999 * M)[0]

        # Subgradient of max at peaks
        n_sub = len(peak_idx)
        sg = np.zeros(n)
        for k in peak_idx:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    sg[i] += f[j]
        sg /= n_sub

        # Update
        f_new = f - alpha * sg
        f_new = _normalize(f_new)
        sc = _score1(f_new)
        if sc < best_sc:
            best_sc = sc
            best_f = f_new.copy()
            f = f_new
            alpha = min(alpha * 1.01, 0.05)
        else:
            alpha *= 0.95

    return best_f, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + budget_s - 5.0

    # Get parent
    f = np.array(height_sequence_1, dtype=np.float64).copy()
    f = np.clip(f, 0, 1000)
    s = f.sum()
    if s < 0.01:
        f = np.ones(256, dtype=np.float64) / 256.0
        s = 1.0
    f = f / s
    n = len(f)

    best_f = f.copy()
    best_score = _score1(f)
    rng = np.random.default_rng(seed)

    def _try_update(arr):
        nonlocal best_f, best_score
        if arr is None or arr.size != n:
            return
        a = _normalize(arr)
        sc = _score1(a)
        if sc < best_score:
            best_score = sc
            best_f = a.copy()

    # Phase 1: LP iterations (main workhorse)
    p1_stop = deadline - max(120.0, 0.25 * (budget_s - 5.0))
    f_cur = best_f.copy()
    result, sc = _lp_iterate(f_cur, n, rng, p1_stop, min_time_left=10.0)
    _try_update(result)

    # Phase 2: Tightened LP iterations
    p2_stop = deadline - max(80.0, 0.15 * (budget_s - 5.0))
    f_cur = best_f.copy()
    if time.monotonic() < p2_stop - 10:
        result, sc = _tightened_lp_iterate(f_cur, n, rng, p2_stop, min_time_left=10.0)
        _try_update(result)

    # Phase 3: More LP with different perturbation strategy
    p3_stop = deadline - max(50.0, 0.08 * (budget_s - 5.0))
    f_cur = best_f.copy()
    # Add small structured perturbation before continuing LP
    f_cur = _normalize(f_cur * (1.0 + 0.02 * np.sin(np.pi * np.arange(n) / n)))
    if time.monotonic() < p3_stop - 5:
        result, sc = _lp_iterate(f_cur, n, rng, p3_stop, min_time_left=10.0)
        _try_update(result)

    # Phase 4: Coordinate polish
    p4_stop = deadline - 8.0
    if time.monotonic() < p4_stop - 3:
        result, sc = _coordinate_polish(best_f, p4_stop, min_time_left=5.0, max_coord=100)
        _try_update(result)

    # Phase 5: Final micro-polish
    if time.monotonic() < deadline - 3.0:
        result, sc = _coordinate_polish(best_f, deadline - 2.0, min_time_left=2.0, max_coord=50)
        _try_update(result)

    # Final validation
    final_arr = np.clip(best_f, 0.0, 1000.0)
    if final_arr.size == 0 or final_arr.sum() < 0.01:
        final_arr = np.ones(256, dtype=np.float64) / 256.0
    result_lst = [float(x) for x in final_arr]
    sc = _exact_score(result_lst)
    if not math.isfinite(sc):
        # Fallback: try with explicit float list
        result_lst = [float(x) if math.isfinite(x) else 0.0 for x in final_arr]
        sc = _exact_score(result_lst)
    if not math.isfinite(sc):
        result_lst = [1.0 / n] * n

    return result_lst
