Yes! So `ifft(W * conj(A))[m] = sum_j w[m+j] a[j]` (circular).

With N ≥ 2n-1+n-1 = 3n-2, the circular sum equals the linear sum for m = 0, ..., n-1 (since w has length 2n-1 and a has length n, the linear cross-correlation at shift m involves w[m] to w[m+n-1], which fits in N ≥ 3n-2).

So grad[m] = 2 * cc[m] where cc = ifft(rfft(w, N) * conj(rfft(a, N)), N)[:n].

This is correct and efficient.

def _build_conv_matrix(a):
    n = a.size
    m = 2*n - 1
    A = np.zeros((m, n))
    for k in range(n):
        # Row k: a[k-i] for i=0..k (i.e., a[0] to a[k])
        # A[k, i] = a[k-i] for i=0..k
        for i in range(k+1):
            A[k, i] = a[k-i]
    for k in range(n, m):
        # Row k: a[k-i] for i=k-n+1..n-1
        # A[k, i] = a[k-i] for i=k-n+1..n-1
        for i in range(k-n+1, n):
            A[k, i] = a[k-i]
    return A

s.t. A^T @ lam ≥ c (i.e., for each j: sum_k A[k,j] lam[k] ≥ 1)
     lam ≥ 0

The dual variables lam give the "importance" of each lag constraint. Normalized lam can be used as weights for the gradient.

def _get_parent():

def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)

    s = a.sum()

    n = a.size

    fa = rfft(a, nfft)
    c = irfft(fa * fa, nfft)[:2*n-1]

    n = a.size
    c = _conv(a)
    s = a.sum()
    return 2.0 * n * c.max() / (s * s)

def _project_simplex(v):
    n = v.size
    v = np.asarray(v, dtype=np.float64).copy()
    v = np.where(np.isfinite(v), v, 0.0)
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    idx = np.arange(1, n+1, dtype=np.float64)
    cond = (css - 1.0) / idx < u
    rho = np.nonzero(cond)[0]
    if rho.size == 0:
        return np.full(n, 1.0/n)
    theta = (css[rho[-1]] - 1.0) / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s <= 1e-300:
        return np.full(n, 1.0/n)
    return w / s

        sw = p / ps
    else:
        sw = np.zeros(c.size)

    m = c.size

    else:
        topK = np.arange(m)

    tw[topK] = 1.0 / len(topK)
    
    w = 0.3 * tw + 0.7 * sw
    sw2 = w.sum()
    if sw2 > 0:

    return w

def _grad_wfft(a, w):
    n = a.size
    m = 2*n - 1
    if w.size < m:
        w = np.pad(w, (0, m-w.size))
    else:
        w = w[:m].copy()

    W = rfft(w, N)
    A = rfft(a, N)
    cc = irfft(W * np.conj(A), N)
    return 2.0 * cc[:n]
