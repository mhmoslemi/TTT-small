    hi2 = min(0.999, ts2[min(len(ts2)-1, idx2+1)])
    ts3 = np.linspace(lo2, hi2, 20)
    vals3 = np.array([val(t) for t in ts3])
    idx3 = np.argmin(vals3)
    
    return float(ts3[idx3])

def _lp_optimize(f0, n, stop_time, seed):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel()[:n].copy(), 0, None)
    while f.size < n:
        f = np.concatenate([f, np.zeros(n - f.size)])
    f = f[:n]
    f_sum = float(f.sum())
    if f_sum < 1e-10:
        f = np.ones(n)
    
    best_f = f.copy()
    best_score = _score(f)
    
    t = 0.4  # Initial step size (for reference, we use analytic line search)
    stall = 0
    max_stall = 400
    conv_thresh = 1e-10
    
    while time.monotonic() < stop_time - 8.0:
        b = _conv_full(f)
        M = float(b.max())
        if M < 1e-18:
            break
        
        g = _lp_direction(f, n, M, timeout=5.0)
        if g is None:
            stall += 5
            if stall > max_stall:
                break
            noise = rng.normal(0, 0.02 * max(1.0, f.mean()), n)
            f = np.clip(f + noise, 0, None)
            fs = f.sum()
            if fs > 1e-10:
                f = f / fs * best_f.sum()
            continue
        
        g_sum = float(g.sum())
        if g_sum < 1e-15:
            stall += 3
            continue
        
        # Scale g to same sum as f
        S_f = float(f.sum())
        g = g * (S_f / g_sum)
        
        # Check convergence
        dist = float(np.linalg.norm(g - f)) / max(1.0, float(np.linalg.norm(f)))
        if dist < conv_thresh:
            stall += 3
            if stall > max_stall:
                break
            continue
        
        # Precompute convolutions for analytic line search
        ff = _conv_full(f)
        gg = _conv_full(g)
        # fg = convolve(f, g)
        fsize = next_fast_len(2 * n - 1)
        Ff = rfft(f, fsize)
        Fg = rfft(g, fsize)
        fg = irfft(Ff * Fg, fsize)[:2 * n - 1]
        np.maximum(fg, 0.0, out=fg)
        
        S_g = float(g.sum())
        
        # Find optimal t
        t_opt = _analytic_line_search(ff, fg, gg, S_f, S_g, n)
        
        # Apply step
        f_new = (1.0 - t_opt) * f + t_opt * g
        f_new = np.clip(f_new, 0, None)
        sc_new = _score(f_new)
        
        if sc_new < best_score - 1e-14:
            f = f_new
            best_f = f.copy()
            best_score = sc_new
            stall = 0
        else:
            stall += 1
            if stall >= max_stall:
                f = np.clip(f + rng.normal(0, 0.05 * max(1.0, f.mean()), n), 0, None)
                fs = f.sum()
                if fs > 1e-10:
                    f = f / fs * best_f.sum()
                else:
                    f = best_f.copy()
                stall = 0
                max_stall += 50
    
    return best_f, best_score

def _polish(a, stop_time, seed=4242):
    n = len(a)
    best = np.clip(np.asarray(a, dtype=np.float64).ravel(), 0, None).copy()
    fs = best.sum()
    if fs < 1e-10:
        return best, _score(best)
    best = best / fs
    best_sc = _score(best)
    rng = np.random.default_rng(seed)
    no_imp = 0
    max_no_imp = 12
    
    while time.monotonic() < stop_time - 5.0:
        improved = False
        
        # Single-coordinate golden section
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 5.0:
                break
            orig = best[i]
            if orig < 1e-14:
                continue
            S_rest = best.sum() - orig
            
            def sc_at(val):
                best[i] = max(0.0, val)
                S = S_rest + best[i]
                if S < 0.01:
                    return 1e30
                b = _conv_full(best)
                return b.max() / (S * S)
            
            # Golden section search
            lo, hi = 0.0, max(2.0 * orig, orig + 1.0)
            gr = (1 + 5**0.5) / 2
            for _ in range(25):
                c1 = hi - (hi - lo) / gr
                c2 = lo + (hi - lo) / gr
                if sc_at(c1) < sc_at(c2):
                    hi = c2
                else:
                    lo = c1
            best[i] = (lo + hi) / 2
            newVal = sc_at(best[i])
            
            if newVal < best_sc - 1e-14:
                best_sc = newVal
                improved = True
        
        # Two-coordinate
        for _ in range(100):
            if time.monotonic() > stop_time - 5.0:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            for mi in (0.3, 0.5, 0.7, 1.4, 2.0, 0.0):
                for mj in (0.3, 0.5, 0.7, 1.4, 2.0, 0.0):
                    if mi == mj == 1.0:
                        continue
                    best[i] = oi * mi if oi > 1e-15 else 0.0
                    best[j] = oj * mj if oj > 1e-15 else 0.0
                    if best.sum() < 0.01:
                        best[i], best[j] = oi, oj
                        continue
                    sc = _score(best)
                    if sc < best_sc - 1e-15:
                        best_sc = sc
                        improved = True
            best[i], best[j] = oi, oj
        
        # Random perturbations
        if not improved:
            for _ in range(30):
                if time.monotonic() > stop_time - 5.0:
                    break
                k = int(rng.integers(n))
                old = best[k]
                for sigma in (0.01, 0.05, 0.15):
                    best[k] = max(0.0, old + rng.normal(0, sigma * max(1.0, best.mean())))
                    if best.sum() < 0.01:
                        best[k] = old
                        continue
                    sc = _score(best)
                    if sc < best_sc - 1e-15:
                        best_sc = sc
                        improved = True
                        break
                    best[k] = old
        
        if not improved:
            no_imp += 1
            if no_imp >= max_no_imp:
                break
        else:
            no_imp = 0
    
    return best, best_sc
