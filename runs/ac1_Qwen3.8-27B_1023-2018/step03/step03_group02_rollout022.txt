import math

def _get_parent():

def _convolve_fft(a, b):
    n = len(a) + len(b) - 1
    fsize = next_fast_len(n)
    Fa = rfft(a, fsize)
    Fb = rfft(b, fsize)
    return irfft(Fa * Fb, fsize)[:n]

def _conv_self(a):
    c = _convolve_fft(a, a)
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_self(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)

def _score_from_conv(a, b):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    m = float(b.max())
    if not math.isfinite(m) or m < 0:

def _build_conv_matrix(f, n):
    """Build the (2n-1) x n convolution matrix for f*g where g varies."""
    # A[k, j] = f[j] if j <= k < j+n, else 0
    # i.e., for each j, f[j] appears in rows k=j to k=j+n-1
    rows = []
    cols = []
    data = []
    for j in range(n):
        fj = f[j]
        if fj > 0:
            k_start = j
            k_end = j + n - 1
            rows.extend(range(k_start, k_end + 1))
            cols.extend([j] * n)
            data.extend([fj] * n)
    A = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    return A

def _lp_solve(f, n, M, A=None, top_k=None, time_limit=15.0):
    """Solve LP: max sum(g) s.t. f*g <= M, g >= 0"""
    if A is None:
        A = _build_conv_matrix(f, n)
    
    if top_k is not None:
        # Only use top-k constraints
        conv_f = _conv_self(f)
        idx = np.argsort(-conv_f)[:top_k]
        A_sub = A.tocsr()[idx]  # Actually, for sparse, we can do A[idx]
        b_ub = np.full(len(idx), M)
    else:
        A_sub = A.tocsr()
        b_ub = np.full(2*n-1, M)
    
    c = -np.ones(n)
    
    try:
        res = linprog(
            c, A_ub=A_sub, b_ub=b_ub,
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True, "time_limit": time_limit, "output_flag": 0},
        )
    except Exception:
        return None
    
    if res.status not in (0, 3) or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g_sum = float(g.sum())
    if g_sum < 1e-12:
        return None
    return g / g_sum

def _polish(a, stop_time, seed=99):
    """Coordinate descent polishing."""
    n = len(a)
    best = np.clip(np.asarray(a, dtype=np.float64).ravel(), 0, None).copy()
    fs = best.sum()
    if fs < 1e-10:
        return best, _score(best)
    best = best / fs
    best_sc = _score(best)
    rng = np.random.default_rng(seed)
    no_imp = 0
    max_no_imp = 4
    
    while time.monotonic() < stop_time - 1.5:
        improved = False
        
        # 1D coordinate descent
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 1.5:
                break
            orig = best[i]
            if orig > 1e-10:
                mults = (0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0, 0.0)
            else:
                mults = (0.005, 0.01, 0.05, 0.1, 0.5)
            best_val = orig
            best_val_sc = best_sc
            for mlt in mults:
                nv = orig * mlt if orig > 1e-10 else mlt
                best[i] = nv
                if best.sum() < 0.01:
                    best[i] = orig
                    continue
                sc = _score(best)
                if sc < best_val_sc - 1e-13:
                    best_val = nv
                    best_val_sc = sc
            best[i] = best_val
            if best_val_sc < best_sc - 1e-13:
                best_sc = best_val_sc
                improved = True
        
        # 2-coordinate moves
        n_pairs = min(80, int(n * 0.5))
        for _ in range(n_pairs):
            if time.monotonic() > stop_time - 1.5:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            for ratio in (0.3, 0.5, 0.7, 1.0, 1.4, 2.0):
                total = oi + oj
                if total < 1e-12:
                    continue
                best[i] = ratio * oi
                best[j] = total - best[i]
                if best[j] < 0 or best.sum() < 0.01:
                    best[i], best[j] = oi, oj
                    continue
                sc = _score(best)
                if sc < best_sc - 1e-13:
                    best_sc = sc
                    improved = True
            best[i], best[j] = oi, oj
        
        if not improved:
            no_imp += 1
            if no_imp >= max_no_imp:
                break
        else:
            no_imp = 0
    
    return best, best_sc

def _conv_self(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_self(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _conv_matrix(f, n):
    """Build (2n-1 x n) matrix such that A @ g = convolve(f, g)."""
    rows = []
    cols = []
    data = []
    for j in range(n):
        fj = f[j]
        if fj > 1e-15:
            k0 = j
            k1 = j + n - 1
            rows.extend(range(k0, k1 + 1))
            cols.extend([j] * n)
            data.extend([fj] * n)
    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _lp_solve(f, n, M, A, top_k=None, time_limit=10.0):
    if top_k is not None and top_k < 2 * n - 1:
        conv_f = _conv_self(f)
        idx = np.argsort(-conv_f)[:top_k]
        A_sub = A[idx]
        b_ub = np.full(len(idx), M)
    else:
        A_sub = A
        b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    try:
        res = linprog(
            c, A_ub=A_sub, b_ub=b_ub,
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True, "time_limit": time_limit, "output_flag": 0},
        )
    except Exception:
        return None
    if res.status not in (0, 3) or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    gs = float(g.sum())
    if gs < 1e-12:
        return None
    return g / gs

def _polish(a, stop_time, seed=99):
    n = len(a)
    best = np.clip(np.asarray(a, dtype=np.float64).ravel(), 0, None).copy()
    fs = best.sum()
    if fs < 1e-10:
        return best, _score(best)
    best = best / fs
    best_sc = _score(best)
    rng = np.random.default_rng(seed)
    no_imp = 0

    while time.monotonic() < stop_time - 1.0:
        improved = False

        # 1D coordinate descent
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 1.0:
                break
            orig = best[i]
            if orig > 1e-10:
                mults = (0.2, 0.4, 0.6, 0.8, 0.9, 1.0, 1.1, 1.25, 1.5, 2.0, 3.0, 0.0)
            else:
                mults = (0.005, 0.01, 0.03, 0.1, 0.3, 0.5)
            bv = orig
            bvs = best_sc
            for mlt in mults:
                nv = orig * mlt if orig > 1e-10 else mlt
                best[i] = nv
                if best.sum() < 0.01:
                    best[i] = orig
                    continue
                sc = _score(best)
                if sc < bvs - 1e-13:
                    bv = nv
                    bvs = sc
            best[i] = bv
            if bvs < best_sc - 1e-13:
                best_sc = bvs
                improved = True

        # 2-coordinate moves
        npairs = min(60, n // 3)
        for _ in range(npairs):
            if time.monotonic() > stop_time - 1.0:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            tot = oi + oj
            if tot < 1e-12:
                continue
            for r in (0.1, 0.3, 0.5, 0.7, 0.9, 1.0, 1.2, 1.5):
                ni = r * oi
                nj = tot - ni
                if nj < -1e-14:
                    continue
                nj = max(0.0, nj)
                best[i], best[j] = ni, nj
                if best.sum() < 0.01:
                    best[i], best[j] = oi, oj
                    continue
                sc = _score(best)
                if sc < best_sc - 1e-13:
                    best_sc = sc
                    improved = True
            best[i], best[j] = oi, oj

        if not improved:
            no_imp += 1
            if no_imp >= 4:
                break
        else:
            no_imp = 0

    return best, best_sc
