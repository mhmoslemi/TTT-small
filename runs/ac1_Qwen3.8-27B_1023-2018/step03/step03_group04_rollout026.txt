```python
def _conv_fft(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    n = len(a)
    S = float(a.sum())
    if S < 0.01:
        return math.inf
    M = float(_conv_fft(a).max())
    if not math.isfinite(M):
        return math.inf
    return 2.0 * n * M / (S * S)

def _build_A(f):
    """Build the convolution matrix A where A @ g = conv(f, g)."""
    n = len(f)
    nrows = 2 * n - 1
    A = np.zeros((nrows, n))
    for j in range(n):
        A[j:j+n, j] = f
    return A

def _lp_direction(f, M):

def _coordinate_polish(a, stop_time, seed=4242):
    """Polish with 1D and 2D coordinate moves."""
    n = len(a)
    best = np.clip(np.asarray(a, dtype=np.float64).ravel(), 0, None).copy()
    fs = best.sum()
    if fs < 1e-10:
        return best, math.inf
    best = best / fs
    best_sc = _score(best)
    rng = np.random.default_rng(seed)
    no_imp = 0
    max_no_imp = 12

    multipliers = [0.2, 0.4, 0.6, 0.8, 1.2, 1.5, 2.0, 3.0, 0.0, 5.0]

    while time.monotonic() < stop_time - 1.0:
        improved = False

        # 1-coordinate moves
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 1.0:
                break
            orig = best[i]
            best_val = orig
            best_val_sc = best_sc
            for mlt in multipliers:
                new_val = orig * mlt if orig > 1e-12 else (0.01 if mlt > 0 else 0.0)
                best[i] = new_val
                if best.sum() < 0.01:
                    best[i] = orig
                    continue
                sc = _score(best)
                if sc < best_val_sc - 1e-14:
                    best_val = new_val
                    best_val_sc = sc
            best[i] = best_val
            if best_val_sc < best_sc - 1e-14:
                best_sc = best_val_sc
                improved = True

        # 2-coordinate moves (random pairs)
        n_2coord = 100
        for _ in range(n_2coord):
            if time.monotonic() > stop_time - 1.0:
                break
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            for mi in (0.3, 0.7, 1.3, 2.0, 0.0):
                for mj in (0.3, 0.7, 1.3, 2.0, 0.0):
                    best[i] = oi * mi if oi > 1e-12 else (0.01 if mi > 0 else 0.0)
                    best[j] = oj * mj if oj > 1e-12 else (0.01 if mj > 0 else 0.0)
                    if best.sum() < 0.01:
                        best[i], best[j] = oi, oj
                        continue
                    sc = _score(best)
                    if sc < best_sc - 1e-14:
                        best_sc = sc
                        improved = True
            best[i], best[j] = oi, oj

        if not improved:
            no_imp += 1
            if no_imp >= max_no_imp:
                break
        else:
            no_imp = 0

    return best, best_sc
```

Let me think about the LP more carefully. The LP finds g* that maximizes sum(g) subject to conv(f,g) <= M. The key insight is:

- conv((1-t)f + t*g, (1-t)f + t*g) = (1-t)^2 conv(f,f) + t(1-t)(conv(f,g) + conv(g,f)) + t^2 conv(g,g)
- The cross terms conv(f,g) and conv(g,f) are bounded by M (by the LP constraint on one of them)
- If g is chosen well, the mixture has lower peak

The LP essentially finds the "dual" of the peak minimization problem. It's a valid approach.

```python
def _build_A(f):
    n = len(f)
    # A is (2n-1) x n, A[k,j] = f[k-j] if 0<=k-j<n else 0
    # This is a Toeplitz-like (actually Hankel + LP) matrix
    # Column j of A is: zeros (j of them), f, zeros (2n-2-j of them)
    # Total: j + n + (2n-2-j) = 3n-2... that's wrong.
    # A is (2n-1) rows. Column j has f[0] at row j, f[1] at row j+1, ..., f[n-1] at row j+n-1.
    # So column j: rows j to j+n-1 contain f[0] to f[n-1].
    # Other rows are 0.
    nrows = 2*n - 1
    A = np.zeros((nrows, n))
    for j in range(n):
        A[j:j+n, j] = f
    return A
```

```python
def _get_parent():

        h = globals()["height_sequence_1"]

def _conv_fft(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    S = float(a.sum())
    if S < 0.01:
        return math.inf
    M = float(_conv_fft(a).max())
    if not math.isfinite(M):
        return math.inf
    return 2.0 * n * M / (S * S)

    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):

def _lp_direction(f, M):

        iteration += 1

    return best_f, best_score


def _subgradient_refine(f0, stop_time, seed):
    n = len(f0)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel(), 0, None).copy()
    fs = f.sum()
    if fs < 1e-10:
        return f, math.inf
    f = f / fs
    best_f = f.copy()
    best_sc = _score(f)
    rng = np.random.default_rng(seed)

    lr = 0.02
    stuck = 0
    patience = 40

    for it in range(600):
        if time.monotonic() > stop_time - 2.0:
            break
        conv = _conv_fft(f)
        M = float(conv.max())

        # Top-k active constraints
        top_k = min(8, 2 * n - 1)
        idx = np.argsort(conv)[::-1][:top_k]

        # Subgradient
        subgrad = np.zeros(n, dtype=np.float64)
        for k in idx:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    if i == j:
                        subgrad[i] += 2.0 * f[i]
                    else:
                        subgrad[i] += f[j]
        subgrad /= top_k

        f_new = f - lr * subgrad
        f_new = np.clip(f_new, 0, None)
        fs_new = f_new.sum()
        if fs_new < 1e-10:
            break
        f_new = f_new / fs_new
        sc = _score(f_new)
        if sc < best_sc - 1e-13:
            best_f = f_new.copy()
            best_sc = sc
            f = f_new
            stuck = 0
            lr = min(0.08, lr * 1.03)
        else:
            stuck += 1
            lr = max(0.004, lr * 0.82)
            if stuck > patience:
                f = np.clip(best_f + rng.normal(0, 0.015, n), 0, None)
                fs = f.sum()
                f = f / fs if fs > 1e-10 else best_f.copy()
                stuck = 0
                lr = 0.02

    return best_f, best_sc


def _coordinate_polish(a, stop_time, seed):
    n = len(a)
    best = np.clip(np.asarray(a, dtype=np.float64).ravel(), 0, None).copy()
    fs = best.sum()
    if fs < 1e-10:
        return best, math.inf
    best = best / fs
    best_sc = _score(best)
    rng = np.random.default_rng(seed)
    no_imp = 0
    max_no_imp = 14

    multipliers_1d = [0.15, 0.3, 0.5, 0.7, 0.9, 1.1, 1.3, 1.6, 2.0, 3.0, 0.0, 5.0]

    while time.monotonic() < stop_time - 1.5:
        improved = False

        # 1-coordinate moves
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 1.5:
                break
            orig = best[i]
            best_val = orig
            best_val_sc = best_sc
            for mlt in multipliers_1d:
                new_val = orig * mlt if orig > 1e-12 else (0.01 if mlt > 0 else 0.0)
                best[i] = new_val
                if best.sum() < 0.01:
                    best[i] = orig
                    continue
                sc = _score(best)
                if sc < best_val_sc - 1e-14:
                    best_val = new_val
                    best_val_sc = sc
            best[i] = best_val
            if best_val_sc < best_sc - 1e-14:
                best_sc = best_val_sc
                improved = True

        # 2-coordinate moves
        n_2c = min(120, max(40, n // 3))
        for _ in range(n_2c):
            if time.monotonic() > stop_time - 1.5:
                break
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            for mi in (0.3, 0.6, 1.0, 1.5, 2.5, 0.0):
                for mj in (0.3, 0.6, 1.0, 1.5, 2.5, 0.0):
                    best[i] = oi * mi if oi > 1e-12 else (0.01 if mi > 0 else 0.0)
                    best[j] = oj * mj if oj > 1e-12 else (0.01 if mj > 0 else 0.0)
                    if best.sum() < 0.01:
                        best[i], best[j] = oi, oj
                        continue
                    sc = _score(best)
                    if sc < best_sc - 1e-14:
                        best_sc = sc
                        improved = True
            best[i], best[j] = oi, oj

        if not improved:
            no_imp += 1
            if no_imp >= max_no_imp:
                break
        else:
            no_imp = 0

    return best, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 20.0

    parent = _get_parent()
    n_p = len(parent)

    fallback = parent.copy()
    try:
        fallback_sc = float(evaluate_sequence([float(x) for x in fallback]))
    except Exception:
        fallback_sc = math.inf
    if not math.isfinite(fallback_sc):
        fallback = np.ones(384, dtype=np.float64)
        fallback_sc = float(evaluate_sequence([float(x) for x in fallback]))

    best_seq = fallback.copy()
    best_score = fallback_sc

    def _track(f):
        nonlocal best_seq, best_score
        if f is None:
            return
        f = np.asarray(f, dtype=np.float64).ravel()
        if not np.all(np.isfinite(f)):
            return
        f = np.clip(f, 0.0, 1000.0)
        if f.sum() < 0.01:
            return
        try:
            exact = float(evaluate_sequence([float(x) for x in f]))
        except Exception:
            return
        if math.isfinite(exact) and exact < best_score:
            best_seq = f.copy()
            best_score = exact

    def _run_lp(init, stop, sd):
        try:
            res, sc = _lp_optimize(init, stop, sd)
            _track(res)
        except Exception:
            pass

    # Phase 1: LP optimization from multiple starts (58% of budget)
    phase1_end = t0 + budget * 0.58
    n = n_p
    i_arr = np.arange(n, dtype=np.float64)
    rng_main = np.random.default_rng(seed)

    inits = []
    inits.append(parent.copy())
    # Raised cosine
    cos_w = np.clip(0.5 - 0.5 * np.cos(2.0 * np.pi * (i_arr + 0.5) / n), 0.01, None)
    inits.append(cos_w)
    # Dual Gaussian
    g1 = np.exp(-((i_arr - n * 0.25) / (n * 0.18)) ** 2)
    g2 = np.exp(-((i_arr - n * 0.75) / (n * 0.18)) ** 2)
    inits.append(g1 + g2 + 0.05)
    # Exponential decay both ends
    e1 = np.exp(-i_arr / (n * 0.15))
    e2 = np.exp(-(n - 1 - i_arr) / (n * 0.15))
    inits.append(e1 + e2 + 0.02)
    # Random
    inits.append(np.abs(rng_main.normal(0, 1, n)) + 0.05)
    # Parent + noise
    inits.append(np.clip(parent + rng_main.normal(0, 0.3 * max(1.0, np.mean(parent)), n), 0, None))
    # Flat center peak
    c1 = np.exp(-((i_arr - n * 0.5) / (n * 0.2)) ** 2)
    inits.append(c1 + 0.03)

    # Allocate time: first (parent) gets the most
    frac = [0.28, 0.15, 0.13, 0.11, 0.11, 0.12, 0.10]
    while len(frac) < len(inits):
        frac.append(0.08)
    total_frac = sum(frac)
    frac = [x / total_frac for x in frac]

    for idx, init in enumerate(inits):
        stop = min(phase1_end - 3.0, time.monotonic() + (phase1_end - t0) * frac[idx])
        if time.monotonic() >= stop:
            continue
        _run_lp(init, stop, seed + 1000 * (idx + 1))

    # Subgradient refinement
    if time.monotonic() < phase1_end - 8.0:
        try:
            rf, _ = _subgradient_refine(best_seq, phase1_end - 3.0, seed + 7777)
            _track(rf)
        except Exception:
            pass

    # Phase 2: Different lengths (20%)
    phase2_end = t0 + budget * 0.78
    p2_start = max(phase1_end, time.monotonic())
    other_lengths = [192, 256, 320, 448, 512, 640]
    if n_p not in other_lengths:
        other_lengths.append(n_p)
    other_lengths = sorted(set(other_lengths))
    p2_dur = max(1.0, phase2_end - p2_start)
    t_per_len = max(4.0, p2_dur / max(1, len(other_lengths)))

    for L in other_lengths:
        if time.monotonic() > phase2_end - 4.0:
            break
        stop = min(phase2_end - 3.0, time.monotonic() + t_per_len)
        if L == n_p:
            init_l = best_seq.copy()
        else:
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, L)
            init_l = np.interp(x_new, x_old, best_seq).astype(np.float64)
        _run_lp(init_l, stop, seed + 60000 + L)

    # Phase 3: Polish (remaining)
    polish_end = deadline - 6.0
    if time.monotonic() < polish_end - 20.0:
        try:
            pol, _ = _coordinate_polish(best_seq, polish_end - 8.0, seed + 31415)
            _track(pol)
        except Exception:
            pass

    # Another LP round on the polished result
    if time.monotonic() < polish_end - 15.0:
        _run_lp(best_seq, polish_end - 8.0, seed + 88888)

    # Final subgradient
    if time.monotonic() < polish_end - 10.0:
        try:
            rf, _ = _subgradient_refine(best_seq, polish_end - 5.0, seed + 55555)
            _track(rf)
        except Exception:
            pass

    # Final polish
    if time.monotonic() < deadline - 4.0:
        try:
            pol, _ = _coordinate_polish(best_seq, deadline - 3.0, seed + 27182)
            _track(pol)
        except Exception:
            pass

    # Validate and return
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            return [1.0]
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in fallback]

    return [float(x) for x in best_seq]
```
