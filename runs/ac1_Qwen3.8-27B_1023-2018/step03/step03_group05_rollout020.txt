import numpy as np
import time
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _conv(n_val, a):
    """Compute full convolution a*a using FFT."""
    n = len(a)
    nfft = next_fast_len(2*n-1)
    f = rfft(a, nfft)
    c = irfft(f*f, nfft)[:2*n-1]
    return np.maximum(c, 0)

def _score(n_val, a):
    c = _conv(n_val, a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return 2*n_val*c.max()/(s*s)

def _project_sum(v, target):
    """Project onto {x: x >= 0, sum(x) = target}."""
    v = np.maximum(v, 0)
    s = v.sum()
    if s < 1e-300:
        return np.full_like(v, target/len(v))
    return v * target / s

def _softmax_weights(c, M, temperature=0.05):
    """Compute weights concentrated on peaks."""
    T = max(M * temperature, 1e-12)
    z = (c - M) / T
    z = np.clip(z, -50, 0)
    w = np.exp(z)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[c.argmax()] = 1
    else:
        w /= s
    return w

def _grad_fft(a, w):
    """Compute gradient of sum_k w[k]*(a*a)[k] w.r.t. a."""
    n = len(a)
    m = 2*n-1
    if len(w) != m:
        w = w[:m] if len(w) > m else np.pad(w, (0, m-len(w)))
    nfft = next_fast_len(2*n-1)
    fa = rfft(a[::-1], nfft)
    fw = rfft(w, nfft)
    corr = irfft(fa * fw, nfft)
    grad = 2.0 * corr[n-1:2*n-1]
    return np.maximum(grad, 0)

def _lp_direction(a, M, deadline, rng, max_time=15):
    """Solve LP to find optimal direction."""
    n = len(a)
    if n > 600 or M <= 0 or not np.isfinite(M):
        return None
    rem = deadline - time.monotonic()
    if rem < max_time + 2:
        return None
    
    m = 2*n-1
    # Build A matrix: A[k,i] = a[k-i] if 0<=k-i<n else 0
    # This is a banded Toeplitz-like matrix
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        A[i, :i+1] = a[:i+1][::-1]
    for k in range(n, m):
        L = m - k
        s = k - n + 1
        A[k, s:s+L] = a[n-1:n-L-1:-1]
    
    # LP: max 1^T b s.t. A b <= M, b >= 0
    # Scale: A/M * b <= 1
    c_obj = -(np.ones(n) + 1e-6 * rng.standard_normal(n))
    A_s = A / M
    b_ub = np.ones(m)
    
    tl = min(max_time, max(1, rem/3))
    try:
        res = linprog(c_obj, A_ub=A_s, b_ub=b_ub,
                      bounds=[(0, None)]*n,
                      method='highs',
                      options={'time_limit': tl, 'presolve': True})
        if res.success and res.x is not None:
            b = np.maximum(res.x, 0)
            s = b.sum()
            if s > 1e-10:
                return b / s
    except Exception:
        pass
    return None

            if patience == 0 and it > 50:
                lp_every = 4
                grad_temp = max(0.02, grad_temp * 0.95)
        else:
            patience += 1

        # Determine direction
        direction = None
        improved = False

        # LP-based direction (most powerful)
        if it % lp_every == 1 and n <= 520 and rem > 20.0:
            dirb = _lp_dir(best_a, M, deadline, rng,
                           max_t=min(15.0, rem * 0.35))
            if dirb is not None:
                d = dirb - best_a
                if np.max(np.abs(d)) > 1e-10:
                    # Try multiple step sizes
                    for alpha in (1.0, 0.7, 0.4, 0.2, 0.1):
                        cand = best_a + alpha * d
                        cand = _project_simplex(cand, 1.0)
                        sc = _score(cand)
                        if sc < cur_score * (1.0 - 1e-10):
                            best_a = cand
                            improved = True
                            break

        # Gradient-based direction
        if not improved and rem > 5.0:
            # Use combined weights
            w_top = _topk_w(c, n, K=10)
            w_soft = _softmax_w(c, M, temp=grad_temp)
            w_mix = 0.4 * w_top + 0.6 * w_soft
            grad = _grad_fft(best_a, w_mix)
            if np.max(grad) > 1e-13:
                # Normalize gradient for step
                gnorm = np.linalg.norm(grad)
                if gnorm > 1e-15:
                    grad_dir = -grad / gnorm
                    # Try multiple steps
                    for eta in (0.3, 0.15, 0.08, 0.04, 0.02):
                        cand = best_a + eta * grad_dir
                        cand = _project_simplex(cand, 1.0)
                        sc = _score(cand)
                        if sc < cur_score * (1.0 - 1e-10):
                            best_a = cand
                            improved = True
                            break

        # Multiplicative update (maintains positivity naturally)
        if not improved and rem > 5.0:
            w_soft = _softmax_w(c, M, temp=0.06)
            grad = _grad_fft(best_a, w_soft)
            if np.max(grad) > 1e-13:
                gm = grad.mean()
                if gm > 1e-18:
                    for eta in (0.3, 0.6, 1.0, 1.5):
                        v = best_a * np.exp(-eta * grad / gm)
                        v = np.where(np.isfinite(v), v, 0.0)
                        cand = _project_simplex(v, 1.0)
                        sc = _score(cand)
                        if sc < cur_score * (1.0 - 1e-10):
                            best_a = cand
                            improved = True
                            break

        # Perturbation when stuck
        if not improved:
            if patience > 15:
                r = rng.random()
                if r < 0.4:
                    # Gaussian perturbation
                    cand = _project_simplex(best_a + rng.normal(0, sigma, n), 1.0)
                    sc = _score(cand)
                    if sc < cur_score:
                        best_a = cand
                        patience = 0
                elif r < 0.6:
                    # Symmetrize
                    cand = _try_sym(best_a)
                    cand = _project_simplex(cand, 1.0)
                    sc = _score(cand)
                    if sc < cur_score:
                        best_a = cand
                        patience = 0
                elif r < 0.8:
                    # Frequency smoothing
                    sig = float(rng.uniform(0.5, 1.5))
                    cand = _smooth_freq(best_a, sig)
                    sc = _score(cand)
                    if sc < cur_score:
                        best_a = cand
                        patience = 0
                else:
                    # Targeted reduction at peak lag
                    k_peak = int(c.argmax())
                    # Reduce a at positions that contribute to peak
                    # Peak at k means a[i]*a[k-i] is large
                    contrib = np.zeros(n)
                    for i in range(n):
                        ki = k_peak - i
                        if 0 <= ki < n:
                            contrib[i] = best_a[i] * best_a[ki]
                    # Reduce the most contributing elements
                    top_idx = np.argsort(contrib)[-max(3, n // 32):]
                    red = 0.01
                    cand = best_a.copy()
                    for idx in top_idx:
                        cand[idx] *= (1.0 - red)
                    cand = _project_simplex(cand, 1.0)
                    sc = _score(cand)
                    if sc < cur_score:
                        best_a = cand
                        patience = 0

                if patience > 30:
                    sigma = min(0.1, sigma * 1.3)
                    patience = 0

    # Phase 2: Final fine polish on the best
    if time.monotonic() < deadline - 5.0:
        polish_deadline = deadline
        pol_it = 0
        while time.monotonic() < polish_deadline and pol_it < 200:
            pol_it += 1
            c = _conv_full(best_a)
            M = float(c.max())
            cur_score = 2.0 * n * M
            improved_pol = False

            # Very small gradient steps
            w = _softmax_w(c, M, temp=max(0.01, grad_temp * 0.5))
            grad = _grad_fft(best_a, w)
            if np.max(grad) > 1e-14:
                gnorm = np.linalg.norm(grad)
                if gnorm > 1e-15:
                    for eta in (0.05, 0.02, 0.01, 0.005):
                        cand = best_a - (eta / gnorm) * grad
                        cand = _project_simplex(cand, 1.0)
                        sc = _score(cand)
                        if sc < cur_score * (1.0 - 1e-10):
                            best_a = cand
                            improved_pol = True
                            break

            # LP for final squeeze
            if pol_it % 5 == 0 and n <= 520 and time.monotonic() < polish_deadline - 15:
                dirb = _lp_dir(best_a, M, polish_deadline, rng, max_t=10.0)
                if dirb is not None:
                    d = dirb - best_a
                    if np.max(np.abs(d)) > 1e-12:
                        for alpha in (0.5, 0.3, 0.15, 0.05):
                            cand = best_a + alpha * d
                            cand = _project_simplex(cand, 1.0)
                            sc = _score(cand)
                            if sc < cur_score * (1.0 - 1e-10):
                                best_a = cand
                                improved_pol = True
                                break

            if not improved_pol and pol_it > 50:
                break

    # Final validation with exact evaluator
    try:
        best_a = np.asarray(best_a, dtype=np.float64).ravel()
        best_a = np.maximum(best_a, 0.0)
        best_a = _project_simplex(best_a, 1.0)
        final_score = float(evaluate_sequence([float(x) for x in best_a]))
        if np.isfinite(final_score):
            best_known = min(best_known, final_score)
    except Exception:
        final_score = float('inf')

    if not np.isfinite(best_known):
        # Fallback
        fb = np.ones(64, dtype=np.float64) / 64.0
        return [float(x) for x in fb]

    out = np.clip(best_a, 0.0, 1000.0)
    out = _project_simplex(out, 1.0)
    if out.size == 0:
        return [1.0]
    return [float(x) for x in out]
