      np.maximum(c, 0, out=c)

      n = len(f)

              infl = np.zeros(2*n-1)
              # a[k-i] is nonzero when 0 <= k-i < n, i.e., i <= k <= i+n-1
              s1 = i
              e1 = min(i+n, 2*n-1)
              if s1 < e1:
                  infl[s1:e1] += f[:e1-s1]
              # a[i-k] is nonzero when 0 <= i-k < n, i.e., i-n+1 <= k <= i
              s2 = max(0, i-n+1)
              e2 = i+1
              if s2 < e2:
                  infl[s2:e2] += f[i-s2:i+1][::-1]
              
              # 1D minimization: find x in [0, x_max] minimizing
              # 2n * max(conv + (x-v)*infl, with conv[2i] += (x-v)^2) / (S_rest+x)^2
              x_max = max(3*v + 0.5, 1.0)
