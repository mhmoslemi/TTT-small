    n = len(a)

        return np.convolve(a, a, mode='full')

    return c

    return 2.0 * n * c.max() / (s * s)

    n = len(f)

    for j in range(n):

    return A

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _conv(a):
    n = len(a)
    if n <= 256:
        c = np.convolve(a, a, mode='full')
    else:
        fsize = next_fast_len(2 * n - 1)
        F = rfft(a, fsize)
        c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(f):
    n = len(f)
    s = float(f.sum())
    if s < 0.01:
        return math.inf
    c = _conv(f)
    return 2.0 * n * float(c.max()) / (s * s)


def _lp_direction(f, time_limit=5.0):
    """Solve LP: min t s.t. (f*f')[k] <= (t+C[k])/2, sum(f')=1, f'>=0"""
    n = len(f)
    C = _conv(f)
    
    # Build convolution matrix A[k,j] = f[k-j] if 0<=k-j<n
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    
    # Variables: (f'_0, ..., f'_{n-1}, t)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A  # (f*f')[k] = sum_j A[k,j]*f'[j]
    A_ub[:, n] = -0.5  # -t/2
    b_ub = C / 2.0  # (t + C[k])/2
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq, bounds=bounds,
                      method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
    except Exception:
        return None, C
    
    if res.status != 0 or res.x is None:
        return None, C
    
    f_new = np.clip(res.x[:n], 0, None)
    s = float(f_new.sum())
    if s < 1e-10:
        return None, C
    f_new /= s
    return f_new, C
