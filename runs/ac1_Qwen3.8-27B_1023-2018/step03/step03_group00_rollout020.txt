        M = float(b.max())
        obj = 2.0 * n * M
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_m = M
            best_norm = a.copy()
            no_imp = 0
            lr = min(lr * 1.04, 0.05)
        else:
            no_imp += 1
            lr = max(lr * 0.85, 1e-7)
            if no_imp > max_no_imp:
                # Perturb to escape
                a = np.clip(a + rng.normal(0, 0.01, n), 0, None)
                a /= a.sum()
                no_imp = 0
                max_no_imp += 200
                lr = 0.003
                if no_imp > 5000:
                    break

        a_new = _subgrad_step(a, lr)
        # Verify improvement
        M_new = float(_conv_full(a_new).max())
        if 2.0 * n * M_new < obj:
            a = a_new
        # else: lr already reduced

    # Scale back to original sum
    result = best_norm * S / max(float(best_norm.sum()), 1e-15)
    return result, best_obj


def _coord_descent(a0, stop_time, seed, max_rounds=50):
    """Coordinate descent with 1D search per coordinate."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.clip(a0.copy(), 0, None)
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > stop_time - 4.0:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 4.0:
                break
            orig = a[i]
            S_other = float(a.sum()) - orig

            # Try a set of candidate values
            candidates = [orig * m for m in (0.05, 0.15, 0.3, 0.5, 0.7, 0.85, 1.1, 1.3, 1.6, 2.0, 3.0, 0.0)]
            # Also try values based on neighbors
            if i > 0:
                candidates.append(orig * 0.5 + a[i-1] * 0.5)
            if i < n - 1:
                candidates.append(orig * 0.5 + a[i+1] * 0.5)

            best_val = orig
            best_v_sc = best_sc
            for x in candidates:
                x = max(0.0, min(1000.0, x))
                a[i] = x
                if a.sum() < 0.01:
                    a[i] = orig
                    continue
                sc = _score(a)
                if sc < best_v_sc - 1e-15:
                    best_v_sc = sc
                    best_val = x
            a[i] = best_val
            if best_v_sc < best_sc - 1e-15:
                best_sc = best_v_sc
                improved = True

        if not improved:
            # Two-coordinate moves
            for _ in range(60):
                if time.monotonic() > stop_time - 4.0:
                    break
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j:
                    continue
                oi, oj = a[i], a[j]
                for mi in (0.3, 0.5, 0.7, 1.4, 1.8, 0.0):
                    for mj in (0.3, 0.5, 0.7, 1.4, 1.8, 0.0):
                        a[i] = oi * mi if oi > 1e-14 else 0.0
                        a[j] = oj * mj if oj > 1e-14 else 0.0
                        if a.sum() < 0.01:
                            a[i], a[j] = oi, oj
                            continue
                        sc = _score(a)
                        if sc < best_sc - 1e-15:
                            best_sc = sc
                            improved = True
                a[i], a[j] = oi, oj
            if not improved:
                break

    return a, best_sc


def _random_perturb(a, rng, scale=0.05):
    """Add Gaussian noise and clip."""
    n = len(a)
    a_new = np.clip(a + rng.normal(0, scale * max(1.0, a.mean()), n), 0, None)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s * a.sum()


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(20.0, float(budget_s))
    deadline = t0 + budget - 20.0

    parent = _get_parent()
    n_p = len(parent)

    # Fallback
    fallback = parent.copy()
    fallback_sc = _score(fallback)
    if not math.isfinite(fallback_sc):
        fallback = np.ones(64)
        fallback_sc = _score(fallback)

    best_seq = fallback.copy()
    best_score = fallback_sc

    def _update(seq, sc):
        nonlocal best_seq, best_score
        if seq is None or len(seq) == 0:
            return
        if not np.all(np.isfinite(seq)):
            return
        seq = np.clip(seq, 0, 1000)
        if seq.sum() < 0.01:
            return
        try:
            exact = float(evaluate_sequence([float(x) for x in seq]))
        except Exception:
            return
        if math.isfinite(exact) and exact < best_score - 1e-14:
            best_seq = seq.copy()
            best_score = exact

    # === Phase 1: LP on parent length (main optimization) ===
    n = n_p
    phase1_end = t0 + budget * 0.38
    rng_main = np.random.default_rng(seed)

    # Multiple initializations for LP
    i_arr = np.arange(n, dtype=np.float64)
    inits = [parent.copy()]

    # Smooth variants
    inits.append(0.5 * (1.0 + np.cos(2.0 * math.pi * i_arr / n)))
    # Parent with slight noise
    sig = 0.1 * max(1.0, float(np.mean(parent)))
    inits.append(np.clip(parent + rng_main.normal(0, sig, n), 0.01, 1000))
    # Bump
    center = (n - 1) / 2.0
    inits.append(np.exp(-0.5 * ((i_arr - center) / (n / 3.0)) ** 2))

    time_per_init = max(15.0, (phase1_end - t0) / len(inits))
    for idx, init in enumerate(inits):
        if time.monotonic() > phase1_end - 8.0:
            break
        stop = min(phase1_end - 4.0, time.monotonic() + time_per_init)
        try:
            res_f, res_sc = _lp_phase(init, n, stop, seed + 10000 * (idx + 1))
        except Exception:
            continue
        _update(res_f, res_sc)

    # Continue LP on best
    if time.monotonic() < phase1_end + budget * 0.15:
        p2_end = t0 + budget * 0.55
        if time.monotonic() < p2_end - 10.0:
            try:
                res_f, res_sc = _lp_phase(best_seq, len(best_seq), p2_end, seed + 77777)
            except Exception:
                res_f, res_sc = best_seq.copy(), best_score
            _update(res_f, res_sc)

    # === Phase 2: Gradient refinement ===
    if time.monotonic() < t0 + budget * 0.80:
        g_end = t0 + budget * 0.78
        try:
            gf, gsc = _grad_refine(best_seq, g_end, seed + 31415)
        except Exception:
            gf, gsc = best_seq.copy(), best_score
        _update(gf, gsc)

        # Another gradient pass from LP result
        if time.monotonic() < g_end - 5.0:
            try:
                gf2, gsc2 = _grad_refine(best_seq, t0 + budget * 0.84, seed + 27182)
            except Exception:
                gf2, gsc2 = best_seq.copy(), best_score
            _update(gf2, gsc2)

    # === Phase 3: Alternative lengths ===
    alt_end = t0 + budget * 0.82
    alt_lengths = [len(best_seq) - 8, len(best_seq) + 8, 128, 256, 320, 384, 448, 512]
    alt_lengths = [L for L in alt_lengths if 64 <= L <= 600 and time.monotonic() < alt_end - 15]
    time_per_alt = max(8.0, max(0, alt_end - time.monotonic()) / max(1, len(alt_lengths)))
    for L in alt_lengths:
        if time.monotonic() > alt_end - 5.0:
            break
        if L == len(best_seq):
            continue
        stop = min(alt_end - 3.0, time.monotonic() + time_per_alt)
        init_l = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, len(best_seq)), best_seq)
        try:
            af, asc = _lp_phase(init_l, L, stop, seed + 50000 + L)
        except Exception:
            continue
        _update(af, asc)

    # === Phase 4: Coordinate descent polishing ===
    pol_end = deadline - 5.0
    if time.monotonic() < pol_end - 10.0:
        try:
            cf, csc = _coord_descent(best_seq, pol_end, seed + 11111)
        except Exception:
            cf, csc = best_seq.copy(), best_score
        _update(cf, csc)

    # Second polish with different seed
    if time.monotonic() < deadline - 8.0:
        try:
            cf2, csc2 = _coord_descent(best_seq, deadline - 4.0, seed + 22222)
        except Exception:
            cf2, csc2 = best_seq.copy(), best_score
        _update(cf2, csc2)

    # Final gradient pass if time allows
    if time.monotonic() < deadline - 6.0:
        try:
            gf3, gsc3 = _grad_refine(best_seq, deadline - 3.0, seed + 999999, max_steps=5000)
        except Exception:
            gf3, gsc3 = best_seq.copy(), best_score
        _update(gf3, gsc3)

    # Final validation
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in fallback]

    return [float(x) for x in best_seq]
