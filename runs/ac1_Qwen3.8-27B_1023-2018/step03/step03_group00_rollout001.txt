    s = float(p.sum())
    if s < 1e-8:
        p = np.ones(416, dtype=np.float64)
    else:
        p = p / s
    return p


def _state_from(a):
    a = _prepare_array(a)
    b = _conv_full(a)
    M = float(b.max()) if b.size else math.inf
    return a, b, M


def _score_state(a, b):
    s = float(np.asarray(a).sum())
    if s < 1e-12:
        return math.inf
    L = np.asarray(b)
    if L.size == 0:
        return math.inf
    return float(2.0 * len(a) * float(L.max()) / (s * s))


def _to_list(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    if a.size == 0 or float(a.sum()) < 0.01:
        return [1.0]
    return [float(x) for x in a]


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(416, dtype=np.float64)
    if p.size == 0:
        p = np.ones(416, dtype=np.float64)
    return p


def _golden_1d(lo, hi, f, iters=22):
    if not (hi > lo):
        return f(lo), lo
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    tol = max(1e-15, 1e-12 * max(1.0, abs(lo), abs(hi)))
    for _ in range(iters):
        if hi - lo <= tol:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    x = 0.5 * (lo + hi)
    fx = f(x)
    best_f, best_x = fx, x
    if fc < best_f:
        best_f, best_x = fc, c
    if fd < best_f:
        best_f, best_x = fd, d
    return float(best_f), float(best_x)


def _pair_one(a, b, M, i, j, grid=12):
    n = int(a.size)
    L = 2 * n - 1
    p = float(a[i])
    q = float(a[j])
    r = 1.0 - p - q
    if r < 1e-10:
        return None
    hi = min(1.0, math.sqrt(max(0.0, M)))
    if hi <= 1e-14:
        return None

    idx = np.arange(n)
    Ki = np.zeros(L, dtype=np.float64)
    Ki[i : i + n] = 2.0 * a[idx]
    Ki[2 * i] = 0.0
    Kj = np.zeros(L, dtype=np.float64)
    Kj[j : j + n] = 2.0 * a[idx]
    Kj[2 * j] = 0.0
    cij = i + j
    if 0 <= cij < L:
        Ki[cij] = 0.0
        Kj[cij] = 0.0

    U = np.array(b, dtype=np.float64, copy=True)
    U -= p * Ki
    U -= q * Kj
    U[2 * i] -= p * p
    U[2 * j] -= q * q
    if i != j:
        U[cij] -= 2.0 * p * q
    np.maximum(U, 0.0, out=U)

    G = max(8, int(grid))
    xs = np.linspace(0.0, hi, G)
    X, Y = np.meshgrid(xs, xs, indexing="ij")
    valid = (X + Y) <= (1.0 - 1e-12)
    if not np.any(valid):
        return None

    C = (1.0 - X - Y) / r
    XC = X * C
    YC = Y * C
    Q = (
        U[None, None, :] * (C * C)[:, :, None]
        + Ki[None, None, :] * XC[:, :, None]
        + Kj[None, None, :] * YC[:, :, None]
    )
    Q[:, :, 2 * i] += X * X
    Q[:, :, 2 * j] += Y * Y
    if i != j:
        Q[:, :, cij] += 2.0 * X * Y

    fvals = Q.max(axis=2)
    fvals[~valid] = np.inf
    flat = int(np.argmin(fvals))
    ix, iy = np.unravel_index(flat, (G, G))
    if not np.isfinite(fvals[ix, iy]):
        return None

    x0, y0 = float(xs[ix]), float(xs[iy])
    f0 = float(fvals[ix, iy])

    if G > 2 and f0 < M:
        dx = hi / max(1, G - 1)
        lox = max(0.0, x0 - dx)
        hix = min(hi, x0 + dx)
        loy = max(0.0, y0 - dx)
        hiy = min(hi, y0 + dx)
        if hix - lox < 1e-12:
            hix = lox + max(1e-12, 1e-12 * hi)
        if hiy - loy < 1e-12:
            hiy = loy + max(1e-12, 1e-12 * hi)
        xs2 = np.linspace(lox, hix, 8)
        ys2 = np.linspace(loy, hiy, 8)
        X2, Y2 = np.meshgrid(xs2, ys2, indexing="ij")
        valid2 = (X2 + Y2) <= (1.0 - 1e-12)
        if np.any(valid2):
            C2 = (1.0 - X2 - Y2) / r
            Q2 = (
                U[None, None, :] * (C2 * C2)[:, :, None]
                + Ki[None, None, :] * (X2 * C2)[:, :, None]
                + Kj[None, None, :] * (Y2 * C2)[:, :, None]
            )
            Q2[:, :, 2 * i] += X2 * X2
            Q2[:, :, 2 * j] += Y2 * Y2
            if i != j:
                Q2[:, :, cij] += 2.0 * X2 * Y2
            f2 = Q2.max(axis=2)
            f2[~valid2] = np.inf
            flat2 = int(np.argmin(f2))
            ix2, iy2 = np.unravel_index(flat2, (8, 8))
            if np.isfinite(f2[ix2, iy2]) and f2[ix2, iy2] < f0:
                x0 = float(xs2[ix2])
                y0 = float(ys2[iy2])
                f0 = float(f2[ix2, iy2])

    if f0 >= M - 1e-14:
        return None

    old_a = a.copy()
    old_b = b.copy()
    old_M = M

    Cb = (1.0 - x0 - y0) / r
    bnew = U * (Cb * Cb) + Ki * (x0 * Cb) + Kj * (y0 * Cb)
    bnew[2 * i] += x0 * x0
    bnew[2 * j] += y0 * y0
    if i != j:
        bnew[cij] += 2.0 * x0 * y0
    np.maximum(bnew, 0.0, out=bnew)

    anew = a * Cb
    anew[i] = x0
    anew[j] = y0
    ss = float(anew.sum())
    if ss <= 1e-14:
        return None
    anew /= ss
    bfull = _conv_full(anew)
    Mnew = float(bfull.max())
    if Mnew < old_M - 1e-14:
        return True, anew, bfull, Mnew
    return False, old_a, old_b, old_M


def _choose_pair(n, b, rng):
    L = 2 * int(n) - 1
    if b.size and rng.random() < 0.7:
        top = np.argsort(b)[-min(32, L):]
        if top.size:
            k = int(top[int(rng.integers(top.size))])
            i = int(rng.integers(n))
            j = k - i
            if 0 <= j < n and j != i:
                return i, j
    i = int(rng.integers(n))
    j = int(rng.integers(n))
    while j == i:
        j = int(rng.integers(n))
    return i, j


def _pair_attempts(a, b, M, stop, rng, grid=12, max_attempts=12):
    attempts = 0
    while attempts < max_attempts and time.monotonic() < stop:
        i, j = _choose_pair(a.size, b, rng)
        one = _pair_one(a, b, M, i, j, grid=grid)
        if one is not None:
            improved, a2, b2, M2 = one
            if improved:
                return True, a2, b2, M2
        attempts += 1
    return False, a, b, M


def _coordinate_polish(
    a_in,
    b_in,
    M_in,
    stop,
    rng,
    grid=48,
    max_no=6,
    use_pair=True,
    pair_grid=12,
    pair_attempts=12,
):
    a = np.array(a_in, dtype=np.float64, copy=True)
    b = np.array(b_in, dtype=np.float64, copy=True)
    M = float(M_in)
    n = int(a.size)
    if n == 0:
        return a, b, M
    s = float(a.sum())
    if s <= 1e-12:
        a = np.ones(n, dtype=np.float64) / n
        b = _conv_full(a)
        M = float(b.max())

    L = 2 * n - 1
    idx = np.arange(n)
    best_a = a.copy()
    best_b = b.copy()
    best_M = M
    no_sweeps = 0
    updates = 0

    while time.monotonic() < stop:
        improved_sweep = False
        order = idx.copy()
        rng.shuffle(order)

        for i in order:
            if time.monotonic() >= stop:
                break
            ai = float(a[i])
            if ai >= 1.0 - 1e-13:
                continue
            d = 1.0 - ai
            if d <= 1e-13:
                continue
            hi = math.sqrt(max(0.0, M))
            if hi <= 1e-14:
                continue
            if hi > 1.0:
                hi = 1.0

            T = np.zeros(L, dtype=np.float64)
            T[i : i + n] = 2.0 * a[idx]
            center = 2 * i
            T[center] = 0.0
            U = b - ai * T
            U[center] -= ai * ai
            np.maximum(U, 0.0, out=U)

            G = max(8, int(grid))
            base = np.linspace(0.0, hi, G)
            if 1e-15 < ai < hi * (1.0 - 1e-12):
                xs = np.concatenate((base, [ai]))
                xs.sort()
            else:
                xs = base

            xg = xs[:, None]
            o = 1.0 - xg
            cf = (o * o) / (d * d)
            ct = (xg * o) / d
            Q = U[None, :] * cf + T[None, :] * ct
            Q[:, center] += xg[:, 0] * xg[:, 0]
            fvals = Q.max(axis=1)

            idxm = int(np.argmin(fvals))
            bf = float(fvals[idxm])
            bx = float(xs[idxm])

            if bf < M - 1e-14 and abs(bx - ai) > 1e-10:
                refined = False
                if len(xs) >= 3:
                    if idxm == 0:
                        lo, hh = float(xs[0]), float(xs[1])
                    elif idxm == len(xs) - 1:
                        lo, hh = float(xs[-2]), float(xs[-1])
                    else:
                        lo, hh = float(xs[idxm - 1]), float(xs[idxm + 1])
                    if hh - lo > 1e-15:
                        def fval(xx, U=U, T=T, center=center, d=d):
                            o2 = 1.0 - xx
                            cf2 = (o2 * o2) / (d * d)
                            ct2 = (xx * o2) / d
                            qq = U * cf2 + T * ct2
                            qq[center] += xx * xx
                            return float(qq.max())

                        gf, gx = _golden_1d(lo, hh, fval, iters=22)
                        if gf < bf:
                            bf, bx = gf, gx
                            refined = True

                if bf < M - 1e-14 and abs(bx - ai) > 1e-10:
                    old_a = a.copy()
                    old_b = b.copy()
                    old_M = M
                    Cb = (1.0 - bx) / d
                    bnew = U * (Cb * Cb) + T * (bx * Cb)
                    bnew[center] += bx * bx
                    np.maximum(bnew, 0.0, out=bnew)

                    anew = a * Cb
                    anew[i] = bx
                    ss = float(anew.sum())
                    if ss > 1e-14:
                        anew /= ss
                        bfull = _conv_full(anew)
                        Mnew = float(bfull.max())
                        if Mnew < old_M - 1e-14 and Mnew < best_M - 1e-14:
                            a, b, M = anew, bfull, Mnew
                            updates += 1
                            if M < best_M:
                                best_M = M
                                best_a = a.copy()
                                best_b = b.copy()
                            improved_sweep = True
                            if updates % 20 == 0:
                                bchk = _conv_full(a)
                                Mchk = float(bchk.max())
                                if Mchk < best_M:
                                    b, M = bchk, Mchk
                                    best_a = a.copy()
                                    best_b = b.copy()
                                    best_M = M
                                else:
                                    a, b, M = best_a.copy(), best_b.copy(), best_M

        if improved_sweep:
            no_sweeps = 0
            continue

        if use_pair and time.monotonic() < stop:
            pstop = min(stop, time.monotonic() + max(5.0, min(12.0, 0.05 * max(1.0, stop - time.monotonic()))))
            stripped, a2, b2, M2 = _pair_attempts(
                a, b, M, pstop, rng, grid=pair_grid, max_attempts=pair_attempts
            )
            if stripped:
                a, b, M = a2, b2, M2
                if M < best_M:
                    best_M = M
                    best_a = a.copy()
                    best_b = b.copy()
                no_sweeps = 0
                updates += 1
                continue

        no_sweeps += 1
        if no_sweeps >= max_no:
            break

    return best_a, best_b, best_M


def _lp_improve(a, b, M, stop, timeout=10):
    if linprog is None:
        return None
    n = int(a.size)
    if n > 700 or time.monotonic() >= stop:
        return None
    L = 2 * n - 1
    A = np.zeros((L, n), dtype=np.float64)
    for j in range(n):
        A[j : j + n, j] = a

    c = np.zeros(n + 1, dtype=np.float64)
    c[-1] = 1.0
    A_ub = np.hstack((A, -np.ones((L, 1), dtype=np.float64)))
    b_ub = np.zeros(L, dtype=np.float64)
    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.ones(1, dtype=np.float64)
    bounds = [(0.0, None)] * n + [(0.0, None)]

    try:
        res = linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
            options={"presolve": True, "time_limit": timeout},
        )
    except Exception:
        return None

    if res is None or getattr(res, "x", None) is None:
        return None

    g = np.asarray(res.x[:n], dtype=np.float64)
    g = np.where(np.isfinite(g), g, 0.0)
    g = np.clip(g, 0.0, None)
    sg = float(g.sum())
    if sg < 1e-12:
        return None
    g /= sg

    best_a = a.copy()
    best_b = np.array(b, dtype=np.float64, copy=True)
    best_M = float(M)

    for t in np.linspace(0.0, 1.0, 25):
        if time.monotonic() >= stop:
            break
        h = (1.0 - float(t)) * a + float(t) * g
        sh = float(h.sum())
        if sh < 1e-12:
            continue
        h /= sh
        bh = _conv_full(h)
        mh = float(bh.max())
        if mh < best_M - 1e-14:
            best_a, best_b, best_M = h, bh, mh

    if best_M < M - 1e-14:
        return best_a, best_b, best_M
    return None


def _perturb(a, sigma, rng):
    a = np.asarray(a, dtype=np.float64).ravel()
    z = np.log(np.maximum(a, 1e-15)) + rng.normal(0.0, float(sigma), a.size)
    z -= float(z.max())
    w = np.exp(z)
    s = float(w.sum())
    if s < 1e-12:
        return a.copy()
    return w / s


def _resample_seq(src, L):
    L = int(L)
    if L <= 1:
        return np.ones(L, dtype=np.float64)
    src = np.asarray(src, dtype=np.float64).ravel()
    if src.size == 1:
        x = np.linspace(0.0, 1.0, L)
    else:
        x = np.interp(np.linspace(0.0, 1.0, L), np.linspace(0.0, 1.0, src.size), src)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s < 1e-12:
        return np.ones(L, dtype=np.float64) / L
    return x / s


def _two_level(n, w=2.0 / 3.0):
    n = int(n)
    a = np.zeros(n, dtype=np.float64)
    if n <= 1:
        a[0] = 1.0
        return a
    m = max(1, n // 2)
    a[:m] = float(w) / m
    a[m:] = (1.0 - float(w)) / max(1, n - m)
    s = float(a.sum())
    if s < 1e-12:
        return np.ones(n, dtype=np.float64) / n
    return a / s


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = max(1.0, float(budget_s))
    except Exception:
        budget = 1300.0
    head = min(50.0, max(5.0, 0.05 * budget))
    hard_stop = t0 + max(1.0, budget - head)

    rng = np.random.default_rng(int(seed) % (2**31 - 1) + 1)

    parent = _get_parent()
    try:
        best_a, best_b, best_M = _state_from(parent)
    except Exception:
        best_a, best_b, best_M = _state_from(np.ones(416))

    best_score = _score_state(best_a, best_b)
    fallback_a = best_a.copy()
    fallback_list = _to_list(fallback_a)
    try:
        fallback_score = float(evaluate_sequence(fallback_list))
        if math.isfinite(fallback_score):
            best_score = min(best_score, fallback_score)
    except Exception:
        fallback_score = math.inf

    def _consider(a, b):
        nonlocal best_a, best_b, best_M, best_score
        try:
            aa = np.asarray(a, dtype=np.float64).ravel()
            bb = np.asarray(b, dtype=np.float64).ravel()
        except Exception:
            return
        if aa.size == 0 or bb.size == 0:
            return
        if not (np.all(np.isfinite(aa)) and np.all(np.isfinite(bb))):
            return
        s = float(aa.sum())
        if s < 1e-8:
            return
        aa = aa / s
        bb = bb / (s * s)
        sc = float(2.0 * aa.size * float(bb.max()))
        if not math.isfinite(sc):
            return
        if sc < best_score - 1e-12:
            best_a = aa.copy()
            best_b = bb.copy()
            best_M = float(bb.max())
            best_score = sc

    # Main dense coordinate/pair polish on the supplied parent.
    if time.monotonic() < hard_stop - 120.0:
        stop1 = min(hard_stop - 120.0, t0 + max(60.0, 0.55 * budget))
        if stop1 > t0 + 5.0:
            try:
                a1, b1, M1 = _coordinate_polish(
                    best_a, best_b, best_M, stop1, rng,
                    grid=56, max_no=8, use_pair=True,
                    pair_grid=14, pair_attempts=20,
                )
                _consider(a1, b1)
            except Exception:
                pass

    # Linear-program steering moves from the best point.
    lp_calls = 0
    while time.monotonic() < hard_stop - 140.0 and lp_calls < 3:
        stop_lp = min(hard_stop - 140.0, time.monotonic() + 25.0)
        try:
            res = _lp_improve(best_a, best_b, best_M, stop_lp, timeout=9)
        except Exception:
            res = None
        lp_calls += 1
        if res is None:
            continue
        a3, b3, M3 = res
        _consider(a3, b3)
        if time.monotonic() < hard_stop - 110.0:
            stop2 = min(hard_stop - 110.0, time.monotonic() + 45.0)
            try:
                a4, b4, M4 = _coordinate_polish(
                    a3, b3, M3, stop2, rng,
                    grid=48, max_no=4, use_pair=True,
                    pair_grid=12, pair_attempts=8,
                )
                _consider(a4, b4)
            except Exception:
                pass

    # Neighbor restarts on the current best length.
    if time.monotonic() < hard_stop - 90.0:
        for sigma in (0.05, 0.08, 0.12, 0.18, 0.28, 0.45):
            if time.monotonic() >= hard_stop - 80.0:
                break
            stop3 = min(hard_stop - 80.0, time.monotonic() + 35.0)
            try:
                start = _perturb(best_a, sigma, rng)
                a5, b5, M5 = _state_from(start)
                a6, b6, M6 = _coordinate_polish(
                    a5, b5, M5, stop3, rng,
                    grid=44, max_no=3, use_pair=True,
                    pair_grid=12, pair_attempts=6,
                )
                _consider(a6, b6)
            except Exception:
                pass

    # Careful exploration of nearby lengths.
    if time.monotonic() < hard_stop - 70.0:
        nbest = int(best_a.size)
        candidates = [
            nbest,
            nbest - 16,
            nbest + 16,
            nbest + 32,
            384, 416, 448, 480, 512,
        ]
        candidates = sorted({int(x) for x in candidates if 96 <= int(x) <= 800})
        for L in candidates:
            if time.monotonic() >= hard_stop - 60.0:
                break
            if L == nbest and time.monotonic() > t0 + 0.7 * budget:
                continue
            stop4 = min(hard_stop - 60.0, time.monotonic() + 42.0)
            try:
                inita = _resample_seq(best_a, L)
                a7, b7, M7 = _state_from(inita)
                a8, b8, M8 = _coordinate_polish(
                    a7, b7, M7, stop4, rng,
                    grid=40, max_no=3, use_pair=True,
                    pair_grid=12, pair_attempts=6,
                )
                _consider(a8, b8)
            except Exception:
                pass

    # Final polish on the best known point.
    stop_final = hard_stop - 12.0
    if time.monotonic() < stop_final - 8.0:
        try:
            a9, b9, M9 = _state_from(best_a)
            a10, b10, M10 = _coordinate_polish(
                a9, b9, M9, stop_final, rng,
                grid=56, max_no=5, use_pair=True,
                pair_grid=14, pair_attempts=14,
            )
            _consider(a10, b10)
        except Exception:
            pass

    final_list = _to_list(best_a)
    final_score = math.inf
    try:
        final_score = float(evaluate_sequence(final_list))
    except Exception:
        final_score = math.inf

    if not math.isfinite(final_score):
        return fallback_list

    if math.isfinite(fallback_score) and fallback_score < final_score:
        return fallback_list

    return final_list
