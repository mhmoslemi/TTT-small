import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog, minimize


def _clean(p):
    p = np.asarray(p, dtype=np.float64).ravel()
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    if p.size == 0:
        return np.ones(1, dtype=np.float64)
    s = float(p.sum())
    if s <= 1e-12:
        return np.ones(p.size, dtype=np.float64) / p.size
    return p / s


def _conv_self(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 1200:
        b = np.convolve(a, a)
    else:
        m = 2 * n - 1
        N = next_fast_len(m)
        F = rfft(a, N)
        b = irfft(F * F, N)[:m]
    b = np.asarray(b, dtype=np.float64)
    b = np.where(np.isfinite(b), b, 0.0)
    return np.maximum(b, 0.0)


def _score_internal(p):
    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    if n == 0:
        return math.inf
    s = float(p.sum())
    if s < 0.01:
        return math.inf
    c = _conv_self(p)
    if c.size == 0:
        return math.inf
    val = 2.0 * n * float(c.max()) / (s * s)
    return val if math.isfinite(val) else math.inf


def _resample(a, m):
    a = np.asarray(a, dtype=np.float64).ravel()
    if m <= 1:
        return _clean(a[:1] if a.size > 1 else np.ones(1))
    if a.size == 0:
        return np.ones(m, dtype=np.float64) / m
    if a.size == 1:
        return np.full(m, float(a[0]), dtype=np.float64)
    x = np.linspace(0.0, 1.0, a.size)
    xx = np.linspace(0.0, 1.0, m)
    y = np.interp(xx, x, a)
    return _clean(y)


def _sqrt_root(n):
    if n <= 0:
        return np.ones(1)
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    return _clean(a)


def _inv_power(n, alpha, reverse=False):
    if n <= 0:
        return np.ones(1)
    idx = np.arange(n, dtype=np.float64)
    if reverse:
        idx = n - 1.0 - idx
    a = np.power(idx + 1.0, -float(alpha))
    return _clean(a)


def _sym_sqrt(n):
    if n <= 0:
        return np.ones(1)
    i = np.arange(n, dtype=np.float64) + 1.0
    j = n - np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(i * np.maximum(j, 1.0))
    return _clean(a)


def _coordinate_polish(p, stop_time, seed=0, npts=None, tol=1e-13, max_sweeps=100000):

            c = _conv_self(p)

        p = p / ps

            center = 2 * i

                Snew = 1.0 + xa - v

                sc = np.full(xa.shape, np.inf, dtype=np.float64)

                xa_v = xa[valid]
                Snew_v = Snew[valid]
                t = xa_v - v
                vals = base[None, :] + t[:, None] * L[None, :]

                center_vals = center_base + xa_v * xa_v - v * v
                m_new = np.maximum(m_lin, center_vals)
                if c_out_max > -1e300:
                    m_new = np.maximum(m_new, c_out_max)
                sc[valid] = 2.0 * n * m_new / (Snew_v * Snew_v)
                return sc

            scores = eval_scores(xs)
            if not np.any(np.isfinite(scores)):
                xs = np.linspace(0.0, max(0.05, 10.0 * mean), max(24, npts // 2))
                scores = eval_scores(xs)
            if not np.any(np.isfinite(scores)):
                continue

            idx = int(np.argmin(scores))
            best_x = float(xs[idx])
            best_sc = float(scores[idx])

            step = xmax / float(npts - 1) if npts > 1 and xmax > 0 else max(10.0 * mean, 1e-6)
            if 0 < idx < len(xs) - 1:
                lo = max(0.0, xs[idx - 1] - 0.5 * step)
                hi = min(xmax, xs[idx + 1] + 0.5 * step)
            else:
                lo = max(0.0, best_x - 2.0 * step)
                hi = min(xmax, best_x + 2.0 * step)
            if hi <= lo:

            fine = np.linspace(lo, hi, refine)
            fine = np.unique(np.concatenate((fine, np.array([best_x, v, 0.0, lo, hi], dtype=np.float64))))
            fine = fine[(fine >= 0.0) & (fine <= max(xmax, v, hi) + 1e-12)]
            if fine.size > 1:
                fs = eval_scores(fine)
                if np.any(np.isfinite(fs)):

                    c = _conv_self(p)

def _lse_minimize(p, stop_time, seed=0, mode="fine"):
    p = _clean(p)
    n = p.size
    if n < 2:
        return p, _score_internal(p)

    best_p = p.copy()
    best_sc = _score_internal(p)

    m = 2 * n - 1
    full_len = 3 * n - 2
    nfft = next_fast_len(full_len)

    if mode == "rough":
        base_betas = (50, 200, 1000, 3000)
        base_iters = (45, 35, 25, 18)
    else:
        base_betas = (100, 500, 2000, 8000)
        base_iters = (70, 50, 35, 25)

    scale = max(1.0, n / 384.0)

    for beta0, miter0 in zip(base_betas, base_iters):
        if time.monotonic() > stop_time - 4.0:
            break
        beta = int(max(20.0, beta0 * scale))
        miter = int(miter0)

        y0 = np.log(np.maximum(p, 1e-300))
        y0 -= y0[0]
        stopped = {"s": False}

        def fun_g(yv):
            if stopped["s"]:
                return np.inf, np.zeros_like(yv, dtype=np.float64)
            y = np.empty(n, dtype=np.float64)
            y[0] = 0.0
            y[1:] = yv
            ym = float(y.max())
            e = np.exp(y - ym)
            Z = float(e.sum())
            if not math.isfinite(Z) or Z <= 0.0:
                return np.inf, np.zeros_like(yv, dtype=np.float64)
            pl = e / Z

            F = rfft(pl, nfft)
            c = irfft(F * F, nfft)[:m]
            c = np.maximum(c, 0.0)

            bc = beta * c
            bcmax = float(bc.max())
            bc = bc - bcmax
            w = np.exp(bc)
            wsum = float(w.sum())
            if not math.isfinite(wsum) or wsum <= 0.0:
                return np.inf, np.zeros_like(yv, dtype=np.float64)
            w = w / wsum
            val = (bcmax + math.log(wsum)) / beta

            Fr = rfft(pl[::-1], nfft)
            Fw = rfft(w, nfft)
            conv = irfft(Fr * Fw, nfft)[:full_len]
            gp = 2.0 * conv[n - 1:2 * n - 1]
            mean = float(np.dot(pl, gp))
            gyv = pl[1:] * (gp - mean)
            gyv = np.where(np.isfinite(gyv), gyv, 0.0)
            return val, gyv

        def cb(xk):
            if time.monotonic() > stop_time - 3.0:
                stopped["s"] = True
                return True
            return False

        maxit = int(min(miter, max(5, max(1.0, stop_time - time.monotonic()) / 1.5)))
        try:
            res = minimize(
                fun_g,
                y0[1:].copy(),
                jac=True,
                method="L-BFGS-B",
                options={
                    "maxiter": maxit,
                    "ftol": 1e-12,
                    "gtol": 1e-7,
                    "maxls": 30,
                    "maxcor": 10,
                    "callback": cb,
                },
            )
        except Exception:
            continue

        pnew = p.copy()
        if res.x is not None and np.all(np.isfinite(res.x)):
            y = np.empty(n, dtype=np.float64)
            y[0] = 0.0
            y[1:] = res.x
            ym = float(y.max())
            e = np.exp(y - ym)
            pnew = _clean(e)

        scnew = _score_internal(pnew)
        if math.isfinite(scnew):
            p = pnew
            if scnew < best_sc:
                best_sc = scnew
                best_p = pnew.copy()
        if stopped["s"]:
            break

    best_sc = _score_internal(best_p)
    return best_p, best_sc


def _lp_optimize(p, stop_time, seed=0):
    p = _clean(p)
    n = p.size
    if n < 2 or n > 450:
        return p, _score_internal(p)

    mlen = 2 * n - 1
    c = _conv_self(p)
    M = float(c.max())
    best_p = p.copy()
    best_sc = 2.0 * n * M
    rng = np.random.default_rng(seed)
    stall = 0

    for _ in range(80):
        if time.monotonic() > stop_time - 5.0:
            break
        if M <= 1e-15:
            break

        p_work = p + 1e-9
        p_work = p_work / p_work.sum()

        A = np.zeros((mlen, n), dtype=np.float64)
        for j in range(n):
            A[j:j + n, j] = p_work

        b_ub = np.full(mlen, M * (1.0 + 1e-8) + 1e-12, dtype=np.float64)
        c_obj = -np.ones(n, dtype=np.float64)

        lp_time = min(3.5, max(1.0, 0.2 * (stop_time - time.monotonic())))
        try:
            res = linprog(
                c_obj,
                A_ub=A,
                b_ub=b_ub,
                bounds=[(0.0, None)] * n,
                method="highs",
                options={"presolve": True, "time_limit": lp_time},
            )
        except Exception:
            break

        g = None
        if res.status == 0 and res.x is not None and np.all(np.isfinite(res.x)):
            g = np.asarray(res.x, dtype=np.float64)
            gs = float(g.sum())
            if gs > 1e-9 and np.max(g) < 1e6:
                g = g / gs

        if g is None:
            break
        if float(np.linalg.norm(g - p)) <= 1e-6:
            break

        improved = False
        for t in (0.25, 0.5, 0.75, 0.1, 0.35, 0.65, 0.9, 0.4, 0.8):
            f = (1.0 - t) * p + t * g
            f = np.clip(f, 0.0, None)
            fs = float(f.sum())
            if fs < 1e-10:
                continue
            f = f / fs
            sc = _score_internal(f)
            if math.isfinite(sc) and sc < best_sc - 1e-13:
                p = f
                best_p = f.copy()
                best_sc = sc
                improved = True

        if improved:
            stall = 0
        else:
            stall += 1
            if stall >= 4:
                break
            p = p + 0.02 * rng.normal(size=n)
            p = np.clip(p, 0.0, None)
            ps = float(p.sum())
            if ps <= 1e-10:
                p = best_p.copy()
            else:
                p = p / ps

        c = _conv_self(p)
        M = float(c.max())

    best_sc = _score_internal(best_p)
    return _clean(best_p), best_sc


def _get_parent():
    try:
        h = globals().get("height_sequence_1")
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(384, dtype=np.float64)
    p = _clean(p)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    return p


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    B = max(60.0, float(budget_s))

    parent = _get_parent()
    best_p = parent.copy()
    best_sc = math.inf

    def _consider(p):
        nonlocal best_p, best_sc
        try:
            p = np.asarray(p, dtype=np.float64).ravel()
            p = np.where(np.isfinite(p), p, 0.0)
            p = np.clip(p, 0.0, 1000.0)
        except Exception:
            return
        if p.size == 0 or float(p.sum()) < 0.01:
            return
        try:
            sc_int = _score_internal(p)
        except Exception:
            return
        if not math.isfinite(sc_int):
            return
        if sc_int > best_sc + 1e-6:
            return
        try:
            sc = float(evaluate_sequence([float(x) for x in p]))
        except Exception:
            sc = math.inf
        if math.isfinite(sc) and sc < best_sc - 1e-15:
            best_p = np.asarray(p, dtype=np.float64).copy()
            best_sc = sc

    def _polish_candidate(p, stop, seed_val, do_lp=False, lse_mode=None):
        if p is None:
            return parent.copy()
        p = _clean(p)
        if time.monotonic() > stop - 5.0:
            return p
        _consider(p)
        n = p.size
        rem = stop - time.monotonic()

        if do_lp and n <= 450 and rem > 45.0:
            lp_stop = min(stop, time.monotonic() + 0.32 * rem)
            try:
                p2, _ = _lp_optimize(p, lp_stop, seed_val)
            except Exception:
                p2 = p
            _consider(p2)
            p = p2

        if lse_mode is not None and n <= 800 and stop - time.monotonic() > 25.0:
            ratio = 0.30 if lse_mode == "rough" else 0.40
            lse_stop = min(stop, time.monotonic() + ratio * max(1.0, stop - time.monotonic()))
            try:
                p2, _ = _lse_minimize(p, lse_stop, seed_val + 1, lse_mode)
            except Exception:
                p2 = p
            _consider(p2)
            p = p2

        if stop - time.monotonic() > 10.0:
            try:
                p2, _ = _coordinate_polish(p, min(stop - 1.0, stop), seed_val + 2)
            except Exception:
                p2 = p
            _consider(p2)
            p = p2

        return p

    if B < 120.0:
        stop = t0 + max(5.0, B - 5.0)
        try:
            p, _ = _coordinate_polish(parent, stop, seed=seed + 1)
        except Exception:
            p = parent
        _consider(p)
        best_p = _clean(best_p)
        if best_p.size == 0 or float(best_p.sum()) < 0.01:
            return [1.0]
        return [float(x) for x in best_p]

    hard = t0 + B - 25.0
    _consider(best_p)
    if not math.isfinite(best_sc):
        fb = np.ones(384, dtype=np.float64) / 384.0
        _consider(fb)
        if not math.isfinite(best_sc):
            return [1.0]

    L0 = int(best_p.size)
    TARGET = 1.50274349

    quick_end = min(t0 + max(8.0, B * 0.018), hard - 120.0)
    quick_shapes = [
        _sqrt_root(L0),
        _sym_sqrt(L0),
        _inv_power(L0, 0.25),
        _inv_power(L0, 0.75, reverse=True),
        _inv_power(L0, 0.5, reverse=True),
    ]
    for arr in quick_shapes:
        if time.monotonic() >= quick_end:
            break
        _consider(arr)

    s1 = t0 + B * 0.07
    s2 = t0 + B * 0.18
    s3 = t0 + B * 0.32
    s4 = t0 + B * 0.46
    s5 = t0 + B * 0.55
    stop_phase2 = t0 + B * 0.82
    final_end = hard - 10.0

    p_work = parent.copy()
    if time.monotonic() < s1 - 8.0:
        p_work = _polish_candidate(p_work, min(s1, hard), 1000, do_lp=False, lse_mode=None)

    if time.monotonic() < s2 - 12.0:
        p_work = _polish_candidate(
            p_work,
            min(s2, hard),
            1010,
            do_lp=(L0 <= 450 and best_sc > TARGET),
            lse_mode=None,
        )

    if time.monotonic() < s3 - 18.0:
        p_work = _polish_candidate(
            p_work,
            min(s3, hard),
            1020,
            do_lp=False,
            lse_mode="fine" if L0 <= 700 else None,
        )

    if best_sc > TARGET:
        init_list = [
            _resample(best_p, L0),
            _sqrt_root(L0),
            _sym_sqrt(L0),
            _inv_power(L0, 0.25),
            _inv_power(L0, 0.75, reverse=True),
        ]
        rem_init = max(0.0, min(s4, hard) - time.monotonic())
        if rem_init > 15.0 and init_list:
            per = max(8.0, 0.78 * rem_init / float(len(init_list)))
            for k, init in enumerate(init_list):
                stop_i = min(s4, hard, time.monotonic() + per)
                if stop_i - time.monotonic() < 8.0:
                    break
                _polish_candidate(
                    init,
                    stop_i,
                    1100 + k,
                    do_lp=False,
                    lse_mode="rough" if L0 <= 512 else None,
                )

    if best_sc > TARGET:
        rng_rand = np.random.default_rng(seed + 7)
        while time.monotonic() < min(s5, hard) - 25.0:
            nR = int(best_p.size)
            gap = max(0.0, best_sc - TARGET)
            noise = 0.012 + 0.035 * min(1.0, gap / 0.05)
            q = best_p + noise * rng_rand.normal(size=nR)
            q = _clean(q)
            stop_r = min(s5, hard, time.monotonic() + 38.0)
            if stop_r - time.monotonic() < 12.0:
                break
            _polish_candidate(q, stop_r, 1200, do_lp=False, lse_mode=None)

    long_candidates = []
    for L in (512, int(round(1.5 * L0)), 640, 768, 1024):
        L = int(L)
        if 64 <= L <= 1024 and L > int(1.05 * L0):
            if L not in long_candidates:
                long_candidates.append(L)
    if B > 1000.0:
        long_candidates = long_candidates[:4]
    else:
        long_candidates = long_candidates[:3]

    if (best_sc > TARGET or B > 900.0) and time.monotonic() < stop_phase2 - 30.0:
        for idx, L in enumerate(long_candidates):
            if time.monotonic() >= min(stop_phase2, hard) - 25.0:
                break
            if int(best_p.size) > L * 1.25:
                continue
            rem = max(0.0, min(stop_phase2, hard) - time.monotonic())
            per = max(25.0, 0.82 * rem / max(1, len(long_candidates) - idx))
            stop_i = min(stop_phase2, hard, time.monotonic() + per)
            init = _resample(best_p, L)
            _polish_candidate(
                init,
                stop_i,
                1300 + idx,
                do_lp=False,
                lse_mode="rough" if L <= 640 else None,
            )

    if time.monotonic() < final_end - 100.0 and int(best_p.size) <= 600:
        lse_stop = min(final_end - 35.0, time.monotonic() + 70.0)
        try:
            p_lse, _ = _lse_minimize(best_p, lse_stop, seed + 2000, "fine")
        except Exception:
            p_lse = best_p
        _consider(p_lse)

    rng_final = np.random.default_rng(seed + 997)
    final_iter = 0
    while time.monotonic() < final_end - 25.0 and final_iter < 18:
        if final_iter == 0:
            start = best_p.copy()
        else:
            gap = max(0.0, best_sc - TARGET)
            noise = 0.008 + 0.025 * min(1.0, gap / 0.05)
            q = best_p + noise * rng_final.normal(size=int(best_p.size))
            start = _clean(q)

        stop_round = min(final_end, time.monotonic() + 80.0)
        try:
            p_round, _ = _coordinate_polish(start, stop_round, seed=seed + 5000 + final_iter)
        except Exception:
            p_round = start
        _consider(p_round)
        final_iter += 1

    best_p = _clean(best_p)
    if best_p.size == 0 or float(best_p.sum()) < 0.01:
        fb = np.ones(max(1, L0), dtype=np.float64) / max(1, L0)
        _consider(fb)
        best_p = _clean(fb)

    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_p]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        fb = np.ones(max(1, L0), dtype=np.float64) / max(1, L0)
        try:
            sc_fb = float(evaluate_sequence([float(x) for x in fb]))
        except Exception:
            sc_fb = math.inf
        if math.isfinite(sc_fb):
            return [float(x) for x in fb]
        return [1.0]

    return [float(x) for x in best_p]
