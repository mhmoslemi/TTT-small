        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n):
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_solve(f, n, M, time_limit=5.0):
    A = _build_A(f, n)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0 and res.x is not None:
            g = np.asarray(res.x, dtype=np.float64)
            gs = float(g.sum())
            if gs > 1e-14:
                return g / gs
    except Exception:
        pass
    return None


def _line_search(f, g, best_sc, t_init=0.5, max_iter=25, stop_time=None):
    t_lo, t_hi = 0.0, t_init
    best_t = 0.0
    best_fn = None
    for _ in range(max_iter):
        if stop_time and time.monotonic() > stop_time:
            break
        t = 0.5 * (t_lo + t_hi)
        fn = (1.0 - t) * f + t * g
        fn = np.maximum(fn, 0.0)
        fs = float(fn.sum())
        if fs < 1e-12:
            t_hi = t
            continue
        fn = fn / fs
        sc = _score(fn)
        if sc < best_sc - 1e-15:
            best_t = t
            best_fn = fn.copy()
            t_lo = t
        else:
            t_hi = t
        if t_hi - t_lo < 1e-7:
            break
    if best_fn is None:
        # Try a few small steps
        for t_try in (0.01, 0.05, 0.1):
            fn = (1.0 - t_try) * f + t_try * g
            fn = np.maximum(fn, 0.0)
            fs = float(fn.sum())
            if fs < 1e-12:
                continue
            fn = fn / fs
            sc = _score(fn)
            if sc < best_sc - 1e-15:
                best_t = t_try
                best_fn = fn.copy()
                break
    return best_t, best_fn


def _subgrad_direction(f, k_top=8):
    n = len(f)
    c = np.convolve(f, f, mode='full')
    top_idx = np.argsort(c)[-k_top:]
    weights = c[top_idx] - c[top_idx].min() + 1e-12
    weights /= weights.sum()
    grad = np.zeros(n)
    for w, idx in zip(weights, top_idx):
        idx = int(idx)
        j_lo = max(0, idx - n + 1)
        j_hi = min(n - 1, idx)
        if j_hi >= j_lo:
            js = np.arange(j_lo, j_hi + 1)
            grad[js] += w * 2.0 * f[idx - js]
    return grad


def _gradient_step(f, lr, k_top=8):
    grad = _subgrad_direction(f, k_top)
    s = f.sum()
    # Gradient of score = 2n*max/s^2, but we minimize max subject to sum=1
    # Descent on max in tangent space (sum=0)
    grad_tan = grad - np.mean(grad)
    f_new = f - lr * grad_tan
    f_new = np.maximum(f_new, 0.0)
    fs = f_new.sum()
    if fs > 1e-12:
        f_new = f_new / fs
    return f_new


def _lp_iteration(f0, n, stop_time, seed, patience=80):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    fs = float(f.sum())
    if fs < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / fs
    best_f = f.copy()
    best_sc = _score(f)
    no_imp = 0
    t_cur = 0.4
    it = 0

    while time.monotonic() < stop_time - 2.0:
        it += 1
        c = np.convolve(f, f, mode='full')
        M = float(c.max())
        if M < 1e-18:
            break

        # Solve LP
        g = _lp_solve(f, n, M, time_limit=3.0)
        if g is None:
            no_imp += 3
            if no_imp > patience:
                break
            # Perturb
            f = f + rng.normal(0, 0.03, n)
            f = np.maximum(f, 0)
            fs2 = f.sum()
            f = f / fs2 if fs2 > 1e-10 else best_f.copy()
            t_cur = 0.3
            continue

        dist = float(np.linalg.norm(g - f))
        if dist < 1e-7:
            # At LP fixed point, try gradient step
            lr = 0.02
            f_g = _gradient_step(f, lr)
            sc_g = _score(f_g)
            if sc_g < best_sc - 1e-14:
                f = f_g
                best_f = f.copy()
                best_sc = sc_g
                no_imp = 0
            else:
                no_imp += 1
                if no_imp > patience:
                    break
                # Perturb to escape
                perturb = rng.normal(0, 0.05, n)
                f = f + perturb
                f = np.maximum(f, 0)
                fs2 = f.sum()
                f = f / fs2 if fs2 > 1e-10 else best_f.copy()
                t_cur = 0.3
            continue

        # Line search
        res_t, f_new = _line_search(f, g, best_sc, t_init=min(0.6, t_cur * 1.1),
                                     stop_time=stop_time)
        if f_new is not None:
            sc_new = _score(f_new)
            if sc_new < best_sc - 1e-14:
                f = f_new
                best_f = f.copy()
                best_sc = sc_new
                t_cur = min(0.6, res_t * 1.2)
                no_imp = 0
            else:
                t_cur = max(0.02, t_cur * 0.5)
                no_imp += 1
        else:
            t_cur = max(0.02, t_cur * 0.5)
            no_imp += 1

        if no_imp >= patience:
            # Stronger perturbation
            f = f + rng.normal(0, 0.1, n)
            f = np.maximum(f, 0)
            fs2 = f.sum()
            f = f / fs2 if fs2 > 1e-10 else best_f.copy()
            t_cur = 0.35
            no_imp = 0

        if best_sc < 1.5028:
            break

    return best_f, best_sc


def _polish(f0, stop_time, seed):
    """Combined gradient + coordinate polish."""
    rng = np.random.default_rng(seed)
    n = len(f0)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    fs = float(f.sum())
    f = f / fs
    best_f = f.copy()
    best_sc = _score(f)
    lr = 0.015
    no_imp = 0

    while time.monotonic() < stop_time - 3.0:
        # Gradient step
        f_new = _gradient_step(f, lr, k_top=10)
        sc = _score(f_new)
        if sc < best_sc - 1e-14:
            f = f_new
            best_f = f.copy()
            best_sc = sc
            no_imp = 0
            lr = min(0.04, lr * 1.02)
        else:
            no_imp += 1
            if no_imp > 100:
                # Coordinate moves
                improved = False
                order = rng.permutation(n)
                for i in order:
                    if time.monotonic() > stop_time - 3.0:
                        break
                    orig = f[i]
                    for mlt in (0.5, 0.7, 0.85, 1.15, 1.3, 1.5):
                        f[i] = orig * mlt if orig > 1e-14 else 0.01
                        sc2 = _score(f)
                        if sc2 < best_sc - 1e-14:
                            best_sc = sc2
                            best_f = f.copy()
                            improved = True
                    if not improved:
                        f[i] = orig
                if improved:
                    no_imp = 0
                    lr = 0.01
                else:
                    # Langevin restart
                    f = f + rng.normal(0, 0.03, n)
                    f = np.maximum(f, 0)
                    fs2 = f.sum()
                    f = f / fs2 if fs2 > 1e-10 else best_f.copy()
                    no_imp = 0
                    lr = 0.02

    return best_f, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 25.0

    parent = _get_parent()
    n_p = len(parent)

    best_seq = parent.copy()
    best_score = _score(best_seq)

    def _try_update(cand):
        nonlocal best_seq, best_score
        if cand is None:
            return
        cand = np.clip(np.asarray(cand, dtype=np.float64).ravel(), 0, 1000.0)
        if cand.size == 0:
            return
        cs = cand.sum()
        if cs < 0.01:
            return
        try:
            sc = _score(cand)
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-15:
            best_seq = cand.copy()
            best_score = sc

    # Phase 1: Main LP iteration on parent length (60% of budget)
    lp_end = t0 + budget * 0.60
    n = n_p
    rng = np.random.default_rng(seed)

    inits = [parent.copy()]
    # Add a few diverse initializations
    i_arr = np.arange(n, dtype=np.float64)
    # Asymmetric ramp
    ramp = np.linspace(0.1, 1.0, n) + 0.1 * np.sin(2 * np.pi * i_arr / n)
    inits.append(np.maximum(ramp, 0.01))
    # Two-bump asymmetric
    bump = 0.5 + 0.4 * np.exp(-0.5 * ((i_arr - 0.3 * n) / (0.15 * n)) ** 2) + \
           0.3 * np.exp(-0.5 * ((i_arr - 0.75 * n) / (0.1 * n)) ** 2)
    inits.append(np.maximum(bump, 0.01))
    # Parent + structured noise
    sn = parent + 0.2 * rng.normal(0, 1, n)
    inits.append(np.maximum(sn, 0.01))

    time_per_init = max(20.0, (lp_end - t0 - 10.0) / len(inits))

    for idx, init in enumerate(inits):
        if time.monotonic() > lp_end - 5.0:
            break
        if best_score < 1.5028:
            break
        stop = min(lp_end - 4.0, time.monotonic() + time_per_init)
        try:
            res, sc = _lp_iteration(init, n, stop, seed + 10000 * (idx + 1))
            _try_update(res)
        except Exception:
            pass

    # Phase 2: Gradient polish (15%)
    polish1_end = min(deadline, t0 + budget * 0.75)
    if time.monotonic() < polish1_end - 10.0:
        try:
            res, sc = _polish(best_seq, polish1_end, seed + 99001)
            _try_update(res)
        except Exception:
            pass

    # Phase 3: Try nearby lengths (15%)
    if time.monotonic() < deadline - 30.0 and best_score > 1.5028:
        other_lengths = [n_p - 4, n_p - 2, n_p + 2, n_p + 4, n_p + 8, n_p - 8]
        other_lengths = sorted(set(l for l in other_lengths if 64 <= l <= 800))
        for L in other_lengths:
            if time.monotonic() > deadline - 30.0:
                break
            if L == n_p:
                continue
            stop = min(deadline - 28.0, time.monotonic() + 40.0)
            init_l = np.interp(np.linspace(0, 1, L),
                               np.linspace(0, 1, n_p), best_seq)
            try:
                res, sc = _lp_iteration(init_l, L, stop, seed + 50000 + L)
                _try_update(res)
            except Exception:
                pass

    # Phase 4: Final LP burst on best (10%)
    if time.monotonic() < deadline - 15.0:
        try:
            res, sc = _lp_iteration(best_seq, len(best_seq), deadline - 12.0,
                                     seed + 88001, patience=50)
            _try_update(res)
        except Exception:
            pass

    # Phase 5: Final polish
    if time.monotonic() < deadline - 8.0:
        try:
            res, sc = _polish(best_seq, deadline - 6.0, seed + 77001)
            _try_update(res)
        except Exception:
            pass

    # Final validation with exact evaluator
    final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    try:
        sc = evaluate_sequence(final_list)
        if math.isfinite(sc) and sc > 0:
            return final_list
    except Exception:
        pass

    # Fallback
    fb = [float(max(0.0, min(1000.0, x))) for x in parent]
    try:
        sc = evaluate_sequence(fb)
        if math.isfinite(sc) and sc > 0:
            return fb
    except Exception:
        pass

    return [1.0] * 384
