def _get_parent():

    except Exception:

def _conv(a):

    F = rfft(a, fs)
    return irfft(F * F, fs)[:2 * n - 1]

def _score_norm(a):

def _build_A(f, n, idx):
    """Build dense constraint matrix for LP."""
    k = len(idx)
    A = np.zeros((k, n))
    for col in range(n):
        # A[row, col] = f[idx[row] - col] if 0 <= idx[row] - col < n
        diff = idx - col
        valid = (diff >= 0) & (diff < n)
        rows = np.where(valid)[0]
        if len(rows) > 0:
            A[rows, col] = f[diff[rows]]
    return A

def _lp_direction(f, n, M, idx, timeout=3.0):
    """Solve LP: max 1'g s.t. A g <= M, g >= 0."""
    A = _build_A(f, n, idx)
    c = -np.ones(n)
    try:
        res = linprog(c, A_ub=A, b_ub=np.full(len(idx), M),
                      bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
    except Exception:
        return None

        return None

    if gs < 1e-10:
        return None
    return g / gs
