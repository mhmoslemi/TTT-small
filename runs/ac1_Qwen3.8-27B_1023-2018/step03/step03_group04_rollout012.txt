    # FFT-based convolution
    def conv_full(a):
        n = len(a)
        m = next_fast_len(2 * n - 1)
        F = rfft(a, m)
        c = irfft(F * F, m)[:2 * n - 1]
        np.maximum(c, 0, out=c)  # numerical safety
        return c
    
    def score_fast(a):
        s = a.sum()
        if s < 0.01: return math.inf
        M = conv_full(a).max()
        return 2.0 * len(a) * M / (s * s)
    
    # Smooth max score with gradient
    def smooth_obj_grad(a, t=50.0):
        n = len(a)
        s = float(a.sum())
        if s < 1e-10:
            return 1e10, np.zeros(n)
        
        conv = conv_full(a)
        c_max = conv.max()
        
        # Numerically stable logsumexp
        shifted = t * (conv - c_max)
        shifted = np.clip(shifted, -100, 100)  # prevent underflow
        exp_vals = np.exp(shifted)
        log_sum = np.log(exp_vals.sum())
        smooth_max = c_max + log_sum / t
        
        score = 2.0 * n * smooth_max / (s * s)
        
        # Softmax probabilities
        p = exp_vals / exp_vals.sum()
        
        # g_j = sum_k p_k * a_{k-j} via FFT
        L = next_fast_len(n + (2*n-1) - 1)
        F_a = rfft(a, L)
        F_p = rfft(p, L)
        g = irfft(F_a * np.conj(F_p), L)[:n]
        
        # Gradient: 4n/s^2 * (g_j - smooth_max * a_j / s)
        grad = 4.0 * n / (s * s) * (g - smooth_max * a / s)
        
        # Add small scale-fixing regularization
        reg = 1e-8 * (s - 1.0)
        grad += 2.0 * 1e-8 * reg * np.ones(n) / max(s, 1e-10)
        score += 1e-8 * reg * reg
        
        return score, grad

            bounds = [(0, 1000.0)] * n

            if res.x is not None:
                new_a = np.clip(res.x, 0, 1000)
                s_new = new_a.sum()
                if s_new > 1e-10:
                    new_a = new_a / s_new
                    sc = score_fast(new_a)
                    if sc < local_best_score - 1e-14:
                        local_best = new_a.copy()
                        local_best_score = sc
                        a = new_a.copy()
            
            # Increase temperature
            if temp < t_end:
                temp = min(t_end, temp * 1.5)
            else:
                # Temperature maxed, do a few more iterations
                temp = t_end
            
            if time.monotonic() > stop_time - 5.0:
                break
        
        return local_best, local_best_score
    
    # LP-based direction finding
    def lp_direction(f, M):
        n = len(f)
        if M < 1e-15:
            return None
        # Build constraint matrix A (2n-1) x n
        # A[k, j] = f[j] if k-j in [0, n-1] else 0
        # i.e., (f * b)[k] = sum_j A[k,j] * b[j] <= M
        A = np.zeros((2*n-1, n))
        for j in range(n):
            A[j:j+n, j] = f
        c = -np.ones(n)
        b_ub = np.full(2*n-1, M)
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub,
                          bounds=[(0, None)] * n,
                          method='highs',
                          options={"presolve": True,
                                   "dual_feasibility_tolerance": 1e-8,
                                   "primal_feasibility_tolerance": 1e-8,
                                   "time_limit": 30})
        except Exception:
            return None
        if res.status != 0 or res.x is None:
            return None
        g = np.asarray(res.x, dtype=np.float64)
        s = float(g.sum())
        if s < 1e-10:
            return None
        return g / s
