def _conv_full(a):
    n = len(a)
    if n * n < 2 * n * 4096:  # small enough for direct
        return np.convolve(a, a, mode="full")
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)
    return np.maximum(c[:2*n-1], 0.0)

def _score(f):
    n = len(f)
    s = float(f.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(f)
    m = float(np.max(b))
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)
