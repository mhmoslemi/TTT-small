import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import minimize

class _Timeout(Exception):
    pass

_fft_cache = {}

def _fft_len(m):
    m = int(m)
    if m <= 0:
        return 1
    v = _fft_cache.get(m)
    if v is None:
        v = int(next_fast_len(m))
        _fft_cache[m] = v
    return v

def _conv_same(a):
    n = int(a.size)
    if n == 0:
        return np.array([], dtype=np.float64)
    m = 2 * n - 1
    L = _fft_len(m)
    fa = rfft(a, L)
    c = irfft(fa * fa, L)[:m]
    return np.maximum(c, 0.0)

def _conv2(a, b):
    n1, n2 = int(a.size), int(b.size)
    if n1 == 0 or n2 == 0:
        return np.array([], dtype=np.float64)
    m = n1 + n2 - 1
    L = _fft_len(m)
    c = irfft(rfft(a, L) * rfft(b, L), L)[:m]
    return np.maximum(c, 0.0)

def _valid_conv(d, a):
    n = int(a.size)
    if n == 0:
        return np.array([], dtype=np.float64)
    ar = a[::-1].copy()
    c = _conv2(d, ar)
    return c[n - 1:2 * n - 1]

def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(32, dtype=np.float64) / 32.0
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not math.isfinite(s) or s < 1e-12:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s

def _score_internal(a):
    try:
        a = np.asarray(a, dtype=np.float64).ravel()
        if a.size == 0:
            return math.inf
        a = np.clip(np.where(np.isfinite(a), a, 0.0), 0.0, 1000.0)
        s = float(a.sum())
        if not math.isfinite(s) or s < 1e-12:
            return math.inf
        a = a / s
        n = int(a.size)
        c = _conv_same(a)
        m = float(c.max())
        if not math.isfinite(m) or m < 0.0:
            return math.inf

    except Exception:

        lst = [float(x) for x in np.asarray(a, dtype=np.float64).ravel()]

def _interp_to_len(parent, L):
    L = int(L)
    if L <= 0:
        return np.array([], dtype=np.float64)
    if parent is None or parent.size == 0:
        return np.ones(L, dtype=np.float64) / float(L)

def _softmax_baseline(y, b, n):
    y = np.asarray(y, dtype=np.float64).ravel()
    y = np.nan_to_num(y, nan=-30.0, posinf=30.0, neginf=-30.0)
    y = np.clip(y, -30.0, 30.0)
    ex = np.exp(y)
    Z = 1.0 + float(ex.sum())
    if not math.isfinite(Z) or Z <= 0.0:
        return np.ones(n, dtype=np.float64) / float(n)
    a = np.empty(n, dtype=np.float64)
    b = int(b)
    if b < 0 or b >= n:

    a[:b] = ex[:b] / Z
    a[b] = 1.0 / Z
    if b + 1 < n:

    s = float(a.sum())
    if s <= 0.0 or not math.isfinite(s):

    return a / s

def _best_t_quadratic(cf, cfs, cg):
    base = cf.astype(np.float64, copy=False)
    bcoef = 2.0 * (cfs - cf)
    gcoef = cf - 2.0 * cfs + cg
    ts = np.linspace(0.0, 1.0, 33)
    vals = np.max(base + bcoef * ts[:, None] + gcoef * (ts[:, None] ** 2), axis=1)
    i = int(np.argmin(vals))
    lo = ts[max(0, i - 2)]
    hi = ts[min(len(ts) - 1, i + 2)]
    if hi <= lo:

        lo = max(0.0, hi - 0.1)
    ts2 = np.linspace(lo, hi, 121)

    j = int(np.argmin(vals2))

def _lbfgs_refine(arr, eta_list, stop, maxfun_per=2000):
    f = _normalize(arr)
    n = int(f.size)
    best = f.copy()
    best_sc = _score_internal(f)
    if n < 2:
        return best, best_sc

    for eta in eta_list:
        if time.monotonic() > stop - 1.0:
            break
        c = _conv_same(f)
        M = float(c.max())
        if not math.isfinite(M) or M <= 1e-15:
            break
        beta = float(eta) / M
        b = int(np.argmax(f))
        idx = np.array([i for i in range(n) if i != b], dtype=int)
        base_val = max(float(f[b]), 1e-300)
        y0 = np.log(np.maximum(f[idx], 1e-300) / base_val)
        y0 = np.clip(np.nan_to_num(y0, nan=-30.0, posinf=30.0, neginf=-30.0), -50.0, 50.0)

        def fg(y, _beta=beta, _b=b, _idx=idx, _n=n):
            if time.monotonic() > stop:
                raise _Timeout
            y = np.asarray(y, dtype=np.float64)
            if not np.all(np.isfinite(y)):
                return 1e100, np.zeros_like(y)
            a = _softmax_baseline(y, _b, _n)
            c_val = _conv_same(a)
            maxc = float(c_val.max())
            if not math.isfinite(maxc) or maxc <= 1e-300:
                return 1e100, np.zeros(_n)
            x = _beta * (c_val - maxc)
            x = np.clip(x, -60.0, 0.0)
            w = np.exp(x)
            s_w = float(w.sum())
            if s_w <= 0.0 or not math.isfinite(s_w):
                w = np.ones_like(c_val) / float(c_val.size)
                s_w = float(c_val.size)
            fun = _beta * maxc + math.log(s_w)
            d = w / s_w
            g = (2.0 * _beta) * _valid_conv(d, a)
            if not np.all(np.isfinite(g)):
                g = np.zeros(_n, dtype=np.float64)
            dot = float(np.dot(a, g))
            rg = a[_idx] * (g[_idx] - dot)
            rg = np.where(np.isfinite(rg), rg, 0.0)
            return float(fun), rg

        try:
            res = minimize(
                fg, y0, jac=True, method="L-BFGS-B",
                options={"maxiter": 500, "maxfun": int(maxfun_per),
                         "ftol": 1e-14, "gtol": 1e-10, "maxls": 50, "iprint": 0}
            )
        except _Timeout:
            break
        except Exception:
            break

        if not hasattr(res, "x") or res.x is None:
            break
        a_new = _softmax_baseline(np.asarray(res.x, dtype=np.float64), b, n)
        sc = _score_internal(a_new)
        if math.isfinite(sc) and sc < best_sc:
            best = a_new.copy()
            best_sc = sc
            f = best
        elif getattr(res, "success", False) and math.isfinite(sc) and sc < best_sc + 1e-9:
            f = a_new.copy()

    return best, best_sc

def _subgradient_chase(a_in, n_steps=200, stop=0.0):
    a = _normalize(a_in)
    n = int(a.size)
    best = a.copy()
    best_sc = _score_internal(a)
    if n < 2 or stop <= 0:
        return best, best_sc

    step = 0
    best_step_sc = best_sc
    stall = 0
    while time.monotonic() < stop - 0.5:
        step += 1
        c = _conv_same(a)
        M = float(c.max())
        k_star = int(np.argmax(c))
        # Compute subgradient of c[k_star]
        i_arr = np.arange(n)
        j_arr = k_star - i_arr
        valid = (j_arr >= 0) & (j_arr < n)
        grad = np.zeros(n, dtype=np.float64)
        grad[valid] = a[j_arr[valid]]
        other_mask = valid & (j_arr != i_arr)
        grad[other_mask] += a[j_arr[other_mask]]
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break
        grad = grad / (gnorm * max(1.0, n) * 0.01)
        # Try multiple step sizes
        found = False
        for t in [1e-4, 3e-4, 1e-3, 3e-3, 1e-2, 3e-2, 0.1]:
            new_a = a - t * grad
            new_a = np.maximum(new_a, 0.0)
            s_new = float(new_a.sum())
            if s_new < 1e-12:
                continue
            new_a /= s_new
            sc = _score_internal(new_a)
            if math.isfinite(sc) and sc < best_sc - 1e-12:
                best = new_a.copy()
                best_sc = sc
                a = new_a
                stall = 0
                found = True
                break
        if not found:
            stall += 1
            if stall > 30:
                break
        if step >= n_steps:
            break
    return best, best_sc

        M = float(c.max())

            break
        # Compute "pressure" at each index
        # P[i] = sum_k (c[k] - M)_+ * a[k-i] (valid correlation)
        excess = np.maximum(c - M, 0.0)
        if float(excess.sum()) < 1e-15:
            break
        P = _valid_conv(excess, a)
        P = np.maximum(P, 1e-20)
        # Multiplicative update
        for eta in [0.01, 0.05, 0.1, 0.2, 0.4]:
            new_a = a / (1.0 + eta * P)
            new_a = np.maximum(new_a, 0.0)
            s_new = float(new_a.sum())
            if s_new < 1e-12:
                continue
            new_a /= s_new
            sc = _score_internal(new_a)
            if math.isfinite(sc) and sc < best_sc - 1e-12:
                best = new_a.copy()
                best_sc = sc
                a = new_a
                break
    return best, best_sc

def _coordinate_polish(a0, stop, seed=1):
    a = _normalize(a0)
    n = int(a.size)
    best = a.copy()
    best_sc = _score_internal(best)
    if n < 2:
        return best, best_sc
    rng = np.random.default_mgr(seed)
    max_passes = 4 if n <= 512 else 2

    for _ in range(max_passes):
        if time.monotonic() > stop - 0.5:
            break
        improved = False
        order = rng.permutation(n)
        default_scale = 1.0 / float(n)
        mults = (0.0, 0.5, 0.75, 0.9, 1.0, 1.1, 1.25, 1.5, 2.0, 3.0)

        for idx in order:
            if time.monotonic() > stop - 0.5:
                break
            orig = float(best[idx])
            base = orig if orig > 1e-15 else default_scale
            best_val = orig
            best_val_sc = best_sc

            vals = {orig}
            for m in mults:
                val = base * float(m)
                if 0.0 <= val <= 1000.0:
                    vals.add(val)

            for val in sorted(vals):
                best[idx] = val
                sc = _score_internal(best)
                if math.isfinite(sc) and sc < best_val_sc:
                    best_val_sc = sc
                    best_val = val

            best[idx] = best_val
            if best_val_sc < best_sc:
                best_sc = best_val_sc
                improved = True

        if not improved:
            break
    return best, best_sc

def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = max(10.0, float(budget_s))
    except Exception:
        budget = 1300.0
    hard_deadline = t0 + budget
    deadline = hard_deadline - 25.0 if budget > 200 else (hard_deadline - 10.0 if budget > 60 else hard_deadline - 2.0)

    parent = _get_parent()
    if parent is None:
        parent = _normalize(np.ones(256, dtype=np.float64))
    else:
        parent = _normalize(parent)
    n_base = int(parent.size)

    best_seq = parent.copy()
    best_score = _exact_score(best_seq.tolist())
    if not math.isfinite(best_score):
        best_score = _score_internal(best_seq)
    fallback = best_seq.copy()

    rng = np.random.default_rng(seed)

    def _try_candidate(arr):
        nonlocal best_seq, best_score
        if arr is None:
            return
        a = _normalize(arr)
        if a.size == 0:
            return
        sc_int = _score_internal(a)
        if not math.isfinite(sc_int) or sc_int >= best_score - 1e-11:
            return
        sc = _exact_score(a.tolist())
        if not math.isfinite(sc):
            sc = sc_int
        if math.isfinite(sc) and sc < best_score - 1e-12:
            best_score = sc
            best_seq = a.copy()

    def _finalize():
        nonlocal best_seq
        seq = np.clip(np.asarray(best_seq, dtype=np.float64).ravel(), 0.0, 1000.0)
        if seq.size == 0 or float(seq.sum()) < 1e-8:
            seq = np.ones(32, dtype=np.float64) / 32.0
        lst = [float(x) for x in seq]
        sc = _exact_score(lst)
        if not math.isfinite(sc):
            fb = np.clip(np.asarray(fallback, dtype=np.float64).ravel(), 0.0, 1000.0)
            if fb.size == 0 or float(fb.sum()) < 1e-8:
                fb = np.ones(32, dtype=np.float64) / 32.0
            lst2 = [float(x) for x in fb]
            sc2 = _exact_score(lst2)
            if math.isfinite(sc2):
                return lst2
        return lst

    # Very short budget
    if budget < 60.0:
        if time.monotonic() < deadline - 1.0:
            arr, _ = _coordinate_polish(best_seq, deadline - 1.0, seed=77)
            _try_candidate(arr)
        return _finalize()

    # === Phase 1: L-BFGS at base length with increasing beta ===
    p1_stop = min(deadline - 200.0, t0 + 0.35 * budget)
    cur = best_seq.copy()

    beta_schedule = [
        (30.0, 800), (100.0, 1200), (400.0, 1500), (1200.0, 2000),
        (4000.0, 2500), (12000.0, 3000), (30000.0, 3500)
    ]
    for eta, mf in beta_schedule:
        if time.monotonic() > p1_stop - 5.0:
            break
        sub_stop = min(p1_stop - 3.0, time.monotonic() + max(10.0, (p1_stop - time.monotonic()) / 2.0))
        if sub_stop - time.monotonic() < 5.0:
            break
        cur, _ = _lbfgs_refine(cur, (eta,), sub_stop, maxfun_per=mf)
        _try_candidate(cur)

    # === Phase 2: Subgradient chase ===
    p2_stop = min(deadline - 130.0, t0 + 0.45 * budget)
    if time.monotonic() < p2_stop - 5.0:
        cur, _ = _subgradient_chase(cur, n_steps=300, stop=p2_stop)
        _try_candidate(cur)

    # === Phase 3: Second round of L-BFGS at higher betas ===
    p3_stop = min(deadline - 100.0, t0 + 0.60 * budget)
    if time.monotonic() < p3_stop - 5.0:
        for eta, mf in [(60000.0, 2500), (150000.0, 2000)]:
            if time.monotonic() > p3_stop - 3.0:
                break
            cur, _ = _lbfgs_refine(cur, (eta,), min(p3_stop - 2.0, time.monotonic() + 60.0), maxfun_per=mf)
            _try_candidate(cur)

    # === Phase 4: Multiplicative + subgradient polish ===
    p4_stop = min(deadline - 70.0, t0 + 0.72 * budget)
    if time.monotonic() < p4_stop - 5.0:
        cur, _ = _multiplicative_step(cur, n_iter=40, stop=min(p4_stop - 10.0, time.monotonic() + 30.0))
        _try_candidate(cur)
        if time.monotonic() < p4_stop - 3.0:
            cur, _ = _subgradient_chase(cur, n_steps=100, stop=p4_stop)
            _try_candidate(cur)

    # === Phase 5: Try alternate lengths ===
    p5_stop = min(deadline - 50.0, t0 + 0.85 * budget)
    other_lens = [128, 384, 512, 640]
    other_lens = [L for L in other_lens if L != n_base and L <= 768]
    if other_lens and time.monotonic() < p5_stop - 20.0:
        per_time = max(20.0, (p5_stop - time.monotonic()) / len(other_lens))
        for L in other_lens[:3]:
            if time.monotonic() > p5_stop - 5.0:
                break
            arr = _interp_to_len(parent, L)
            sub_stop = min(p5_stop - 3.0, time.monotonic() + per_time)
            if sub_stop - time.monotonic() < 10.0:
                continue
            arr, _ = _lbfgs_refine(arr, (200.0, 800.0), sub_stop, maxfun_per=1000)
            _try_candidate(arr)

    # === Phase 6: Final L-BFGS at best length with very high beta ===
    cur_n = int(best_seq.size)
    p6_stop = deadline - 15.0
    if cur_n <= 1024 and time.monotonic() < p6_stop - 10.0:
        cur, _ = _lbfgs_refine(best_seq, (50000.0, 200000.0), p6_stop, maxfun_per=3000)
        _try_candidate(cur)

    # === Phase 7: Coordinate polish ===
    if time.monotonic() < p6_stop - 3.0:
        cur, _ = _coordinate_polish(best_seq, p6_stop - 1.0, seed=424242)
        _try_candidate(cur)

    # Final subgradient
    if time.monotonic() < deadline - 2.0:
        cur, _ = _subgradient_chase(best_seq, n_steps=50, stop=deadline - 1.0)
        _try_candidate(cur)

    return _finalize()
