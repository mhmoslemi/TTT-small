import numpy as np
import time
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s > 1e-300:
        return (a / s).astype(np.float64)
    return np.full(max(a.size, 1), 1.0 / max(a.size, 1), dtype=np.float64)


def _conv(a):
    n = a.size
    if n == 0:
        return np.array([0.0])
    nfft = next_fast_len(2 * n - 1)
    f = rfft(a, nfft)
    c = irfft(f * f, nfft)[:2 * n - 1]
    return np.maximum(np.where(np.isfinite(c), c, 0.0), 0.0)


def _score(a):
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return float('inf')
    c = _conv(a)
    return float(2 * n * c.max() / (s * s))


def _exact_score(a):
    try:
        return float(evaluate_sequence([float(x) for x in a]))
    except Exception:
        return float('inf')


def _build_A(f):
    n = f.size
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for k in range(m):
        jlo = max(0, k - n + 1)
        jhi = min(k, n - 1)
        if jlo <= jhi:
            A[k, jlo:jhi + 1] = f[k - jhi:k - jlo + 1][::-1]
    return A


def _solve_lp(a, M, deadline, rng):
    n = a.size
    rem = deadline - time.monotonic()
    if rem < 2.0 or n < 4:
        return None
    A = _build_A(a)
    A_ub = A / M
    b_ub = np.ones(2 * n - 1)
    tie = rng.standard_normal(n) * 1e-8
    c_obj = -np.ones(n) + tie
    tl = max(0.5, min(8.0, 0.3 * rem))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=(0, None),
                      method='highs', options={'time_limit': tl, 'presolve': True})
        if res.success:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
        if res.success:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def _softmax_weights(c, T_scale=0.03):
    M = float(c.max())
    if M <= 0 or not np.isfinite(M):
        w = np.zeros(c.size)
        w[0] = 1.0
        return w
    T = M * T_scale
    z = (c - M) / T
    z = np.clip(z, -50, 0)
    p = np.exp(z)
    s = p.sum()
    if s > 0 and np.isfinite(s):
        return p / s
    w = np.zeros(c.size)
    w[int(np.argmax(c))] = 1.0
    return w


def _weighted_grad(a, w):
    n = a.size
    m = w.size
    need = 2 * n - 1
    if m != need:
        if m > need:
            w = w[:need]
        else:
            w = np.concatenate([w, np.zeros(need - m)])
    w = np.where(np.isfinite(w), w, 0.0)
    sw = w.sum()
    if sw <= 0:
        return np.zeros(n)
    w = w / sw
    nfft = next_fast_len(3 * n - 2)
    fa = rfft(a, nfft)
    fw = rfft(w[::-1], nfft)
    cr = irfft(fa * fw, nfft)[:3 * n - 2]
    cr = np.where(np.isfinite(cr), cr, 0.0)
    idx = 2 * n - 2 - np.arange(n)
    g = 2.0 * cr[idx]
    return np.maximum(g, 0.0)


def _try_candidates(cands, best_a, best_sc):
    for cand in cands:
        if cand is None or cand.size == 0:
            continue
        cand = _normalize(cand)
        sc = _score(cand)
        if sc < best_sc:
            best_sc = sc
            best_a = cand.copy()
    return best_a, best_sc


def _refine(a, deadline, rng, n_max_lp=600):
    a = _normalize(a)
    n = a.size
    best_a = a.copy()
    best_score = _score(a)
    no_improve = 0
    max_no_improve = 25
    sigma = 0.015
    lp_count = 0

    while time.monotonic() < deadline:
        rem = deadline - time.monotonic()
        if rem < 3.0:
            break

        c = _conv(a)
        M = float(c.max())
        cur_score = 2.0 * n * M

        best_point = None
        best_point_score = cur_score

        # LP direction
        if lp_count < n_max_lp and rem > 10.0 and n <= 800:
            g = _solve_lp(a, M, deadline - 3, rng)
            lp_count += 1
            if g is not None and np.any(np.abs(g - a) > 1e-6):
                for t in (0.05, 0.1, 0.15, 0.25, 0.4, 0.6, 0.8, 1.0):
                    cand = (1 - t) * a + t * g
                    cand = _normalize(cand)
                    sc = _score(cand)
                    if sc < best_point_score:
                        best_point_score = sc
                        best_point = cand

        # Gradient step with softmax weights
        if best_point is None:
            w = _softmax_weights(c, T_scale=0.03)
            grad = _weighted_grad(a, w)
            gn = np.linalg.norm(grad)
            if gn > 1e-12:
                a_mean = a.mean()
                for eta in (0.3, 0.7, 1.2, 2.0, 3.5, 5.0):
                    if time.monotonic() >= deadline - 1:
                        break
                    cand = a - eta * a_mean * grad / gn
                    cand = _normalize(cand)
                    sc = _score(cand)
                    if sc < best_point_score:
                        best_point_score = sc
                        best_point = cand
                # Multiplicative update
                for eta in (0.5, 1.0, 2.0):
                    if time.monotonic() >= deadline - 1:
                        break
                    v = a * np.exp(-eta * grad / (a_mean + 1e-15))
                    v = np.maximum(v, 0)
                    sv = v.sum()
                    if sv > 1e-10:
                        cand = v / sv
                        sc = _score(cand)
                        if sc < best_point_score:
                            best_point_score = sc
                            best_point = cand

        if best_point is not None:
            a = best_point
            if best_point_score < best_score * (1 - 1e-9):
                best_a = a.copy()
                best_score = best_point_score
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve >= max_no_improve:
            r = rng.random()
            if r < 0.4:
                a = _normalize(a + rng.normal(0, sigma, n))
            elif r < 0.7:
                idx = rng.choice(n, size=max(1, n // 4), replace=False)
                pert = np.zeros(n)
                pert[idx] = rng.uniform(-0.05, 0.05, size=n // 4 + 1)[:len(idx)]
                a = _normalize(a + pert)
            else:
                idx = np.arange(1, n + 1, dtype=np.float64)
                alpha = float(rng.uniform(0.5, 2.0))
                a = _normalize(idx ** alpha)
            no_improve = 0
            sigma = min(0.15, sigma * 1.4)

    return best_a, best_score


def _resample(a, m):
    a = _normalize(a)
    n = a.size
    m = int(m)
    if m <= 0:
        return np.ones(1)
    if n == m:
        return a.copy()
    if m > n:
        idx = (np.arange(m, dtype=np.int64) * n) // m
        counts = np.bincount(idx, minlength=n).astype(np.float64)
        counts[counts <= 0] = 1.0
        b = a[idx] / counts[idx]
    else:
        b = np.zeros(m, dtype=np.float64)
        idx = (np.arange(n, dtype=np.int64) * m) // n
        np.add.at(b, idx, a)
        b = np.maximum(b, 0.0)
    return _normalize(b)
