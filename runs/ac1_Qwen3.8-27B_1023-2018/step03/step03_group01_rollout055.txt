import math

    fs = next_fast_len(2*n-1)
    F = rfft(a, fs)
    c = irfft(F*F, fs)[:2*n-1]

    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv(a)
    m = float(b.max())

    A = np.zeros((len(idx), n))
    for i, k in enumerate(idx):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        A[i, lo:hi] = a[k - hi + 1 : k - lo + 1 : -1]
    return A

    b = _conv(a)
    nl = len(b)
    if K >= nl:
        idx = np.arange(nl)
    else:
        idx = np.argpartition(b, -K)[-K:]

        res = linprog(-np.ones(n), A_ub=A, b_ub=bu,
                     bounds=[(0.0, None)] * n, method='highs',
                     options={'presolve': True})
    except Exception:
        return None

        return None

    if s < 1e-14:

def _line_search(f0, fd, n, best_sc, t0=0.3, deadline=None):
    """Find t in [0, t0] minimizing score((1-t)*f0 + t*fd)"""
    s0 = f0.sum()
    b0 = _conv(f0)
    M0 = b0.max()
    
    # Golden section search
    gr = (math.sqrt(5) - 1) / 2  # ~0.618

    d_ = a_ + gr * (b_ - a_)

        b = _conv(fn)
        m = b.max()
        return 2.0 * n * m / (s * s)

    bad2 = 0.1 / t0
