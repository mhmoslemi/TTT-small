import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

    except Exception:

    if p.sum() < 0.01:
        p = np.ones(384, dtype=np.float64)
    return p


def _fft_conv(a):
    n = len(a)
    sz = next_fast_len(2 * n - 1)
    F = rfft(a, sz)
    c = irfft(F * F, sz)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score_norm(a):
    """Score with sum(a)=1: 2n * max(conv(a,a))."""
    n = len(a)
    return 2.0 * n * float(_fft_conv(a).max())


def _build_A(a):
    """Build matrix A where (A*b)[k] = conv(a,b)[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        ks, ke = j, min(j + n, m)
        A[ks:ke, j] = a[0:ke - ks]
    return A


def _solve_lp(a, M):
    """Solve LP: max 1^T b s.t. A*b <= M, b >= 0."""
    n = len(a)
    if M <= 1e-15:
        return None
    A = _build_A(a)
    c_vec = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_vec, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={"presolve": True,
                               "dual_feasibility_tolerance": 1e-9,
                               "primal_feasibility_tolerance": 1e-9})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    s = float(g.sum())
    if s < 1e-10:
        return None
    return g / s


def _subgradient_step(a, conv, n, frac=0.05):
    """Compute a subgradient-based descent direction for the max of conv."""
    M = float(conv.max())
    if M < 1e-15:
        return a.copy()
    thresh = M * (1.0 - frac)
    active = np.where(conv >= thresh)[0]
    if len(active) == 0:
        return a.copy()
    g = np.zeros(n)
    for k in active:

        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        if j_lo > j_hi:
            continue
        idx = k - np.arange(j_lo, j_hi + 1)
        g[j_lo:j_hi + 1] += 2.0 * a[idx]
    g /= len(active)

        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_sc = _score_norm(a)
    t = 0.4
    t_min, t_max = 0.05, 0.8
    stall = 0
    max_stall = 80
    last_improve = 0
    it = 0

    while time.monotonic() < stop_time - 0.5 and it < max_iters:
        conv = _fft_conv(a)
        M = float(conv.max())

        b = _solve_lp(a, M)
        if b is None:
            # Fallback: subgradient
            g = _subgradient_step(a, conv, n)
            b = np.clip(a - 0.1 * g, 0, None)
            bs = b.sum()
            b = b / bs if bs > 1e-10 else a.copy()

        dist = float(np.linalg.norm(b - a))
        if dist < 1e-9:
            stall += 3
            if stall > max_stall:
                break
            a = np.clip(a + rng.normal(0, 0.02, n), 0, None)
            s2 = a.sum()
            a = a / s2 if s2 > 1e-10 else best_a.copy()
            it += 1
            continue

        # Line search with multiple candidates
        best_t = 0.0
        best_sc_trial = best_sc
        best_a_trial = None
        t_cands = [t, t * 0.5, t * 1.5, t * 0.25, t * 2.0, t * 0.75]
        t_cands = [max(t_min, min(t_max, tc)) for tc in t_cands]
        seen = set()
        for tc in t_cands:
            key = round(tc, 6)
            if key in seen:
                continue
            seen.add(key)
            a_new = (1.0 - tc) * a + tc * b
            a_new = np.clip(a_new, 0, None)
            s2 = a_new.sum()
            if s2 < 1e-10:
                continue
            a_new = a_new / s2
            sc = _score_norm(a_new)
            if sc < best_sc_trial - 1e-13:
                best_sc_trial = sc
                best_t = tc
                best_a_trial = a_new.copy()

        if best_a_trial is not None:
            a = best_a_trial
            if best_sc_trial < best_sc - 1e-13:
                best_a = a.copy()
                best_sc = best_sc_trial
                last_improve = it
                stall = 0
                t = min(t_max, t * 1.08)
            else:
                t = max(t_min, t * 0.7)
                stall += 1
        else:
            stall += 1
            t = max(t_min, t * 0.6)
            if stall >= max_stall:
                # Escape: random perturbation or reset
                if rng.random() < 0.4:
                    a = np.clip(best_a + rng.normal(0, 0.015, n), 0, None)
                else:
                    a = np.clip(a + rng.normal(0, 0.04, n), 0, None)
                s2 = a.sum()
                a = a / s2 if s2 > 1e-10 else best_a.copy()
                t = 0.3
                stall = 0

        it += 1

    return best_a, best_sc


def _coord_descent(a, stop_time, seed=7777):
    """Coordinate descent polish."""
    rng = np.random.default_rng(seed)
    n = len(a)
    a = np.clip(a.astype(np.float64), 0, None).copy()
    s = a.sum()
    if s < 1e-10:
        return a, math.inf
    a = a / s
    best_sc = _score_norm(a)
    no_imp = 0
    max_no_imp = 12
    multipliers = [0.3, 0.5, 0.7, 1.0, 1.4, 1.8, 2.5, 0.0]

    while time.monotonic() < stop_time - 0.5:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 0.5:
                break
            orig = a[i]
            best_val = orig
            best_vsc = best_sc
            for m in multipliers:
                nv = orig * m if orig > 1e-12 else (0.01 if m > 0 else 0.0)
                a[i] = nv
                s2 = a.sum()
                if s2 < 0.01:
                    a[i] = orig
                    continue
                a_n = a / s2
                sc = _score_norm(a_n)
                if sc < best_vsc - 1e-14:
                    best_vsc = sc
                    best_val = nv
            a[i] = best_val
            s2 = a.sum()
            if s2 > 1e-10:
                a = a / s2
                sc = _score_norm(a)
                if sc < best_sc - 1e-14:
                    best_sc = sc
            if best_vsc < best_sc + 1e-14:
                improved = True

        # 2-coordinate moves
        n_2coord = min(100, n // 4)
        for _ in range(n_2coord):
            if time.monotonic() > stop_time - 0.5:
                break
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            oi, oj = a[i], a[j]
            for mi in (0.5, 1.5, 0.0, 2.0, 0.7):
                for mj in (0.5, 1.5, 0.0, 2.0, 1.3):
                    a[i] = oi * mi if oi > 1e-12 else (0.01 if mi > 0 else 0.0)
                    a[j] = oj * mj if oj > 1e-12 else (0.01 if mj > 0 else 0.0)
                    s2 = a.sum()
                    if s2 < 0.01:
                        a[i], a[j] = oi, oj
                        break
                    sc = _score_norm(a / s2)
                    if sc < best_sc - 1e-14:
                        best_sc = sc
                        improved = True
            a[i], a[j] = oi, oj
            s2 = a.sum()
            if s2 > 1e-10:
                a = a / s2

        if not improved:
            no_imp += 1
            if no_imp >= max_no_imp:
                break
        else:
            no_imp = 0

    return a, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(20.0, float(budget_s))
    deadline = t0 + budget - 20.0

    parent = _get_parent()
    n_parent = len(parent)

    # Validate parent
    best_seq = parent.copy()
    try:
        best_score = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        best_score = math.inf
    if not math.isfinite(best_score):
        best_seq = np.ones(384)
        best_score = float(evaluate_sequence([float(x) for x in best_seq]))

    def _track(f):
        nonlocal best_seq, best_score
        if not np.all(np.isfinite(f)):
            return
        f = np.clip(f, 0, 1000)
        if f.sum() < 0.01:
            return
        try:
            exact = float(evaluate_sequence([float(x) for x in f]))
        except Exception:
            exact = math.inf
        if math.isfinite(exact) and exact < best_score:
            best_seq = f.copy()
            best_score = exact

    # Phase 1: LP optimization on parent length (55% budget)
    phase1_end = t0 + budget * 0.55
    n = n_parent
    rng_main = np.random.default_rng(seed)

    # Generate initializations
    inits = []
    inits.append(parent.copy())

    # Lopsided exponential
    i_arr = np.arange(n, dtype=np.float64)
    inits.append(np.exp(-i_arr / (n * 0.4)) + 0.01)
    inits.append(0.01 + np.exp(-(n - 1 - i_arr) / (n * 0.4)))

    # Asymmetric cosine
    inits.append(0.3 + 0.7 * (0.5 + 0.5 * np.cos(np.pi * i_arr / (n - 1))) ** 2)

    # Power-law decay
    inits.append(1.0 / (1.0 + i_arr / (n * 0.5)) + 0.01)

    # Random
    inits.append(np.abs(rng_main.normal(0, 1, n)) + 0.05)

    # Parent + noise
    noise_scale = 0.15 * max(1.0, np.mean(parent))
    inits.append(np.clip(parent + rng_main.normal(0, noise_scale, n), 0, None))

    # Time allocation (first gets most)
    time_alloc = [0.35, 0.12, 0.10, 0.10, 0.08, 0.10, 0.15]
    phase1_dur = phase1_end - t0

    for idx, init in enumerate(inits):
        alloc = time_alloc[idx] if idx < len(time_alloc) else 0.1
        stop = min(phase1_end - 2.0, time.monotonic() + phase1_dur * alloc)
        if time.monotonic() >= stop:
            continue
        try:
            res, sc = _lp_optimize(init, stop, seed + 1000 * (idx + 1))
            _track(res)
        except Exception:
            pass

    # Subgradient refinement on best
    if time.monotonic() < phase1_end - 5.0:
        try:
            a_sub = np.clip(best_seq, 0, None).copy()
            s = a_sub.sum()
            a_sub = a_sub / s if s > 1e-10 else a_sub
            n_best = len(a_sub)
            for it in range(500):
                if time.monotonic() > phase1_end - 2.0:
                    break
                conv = _fft_conv(a_sub)
                M = float(conv.max())
                g = _subgradient_step(a_sub, conv, n_best, frac=0.03)
                for lr in [0.05, 0.03, 0.02, 0.01, 0.005]:
                    a_new = np.clip(a_sub - lr * g, 0, None)
                    s2 = a_new.sum()
                    if s2 < 1e-10:
                        continue
                    a_new = a_new / s2
                    sc = _score_norm(a_new)
                    if sc < _score_norm(a_sub) - 1e-13:
                        a_sub = a_new
                        break
            _track(a_sub)
        except Exception:
            pass

    # Phase 2: Coordinate descent polish (15% budget)
    phase2_end = t0 + budget * 0.72
    if time.monotonic() < phase2_end - 5.0:
        try:
            polished, _ = _coord_descent(best_seq, min(phase2_end, deadline - 30.0), seed + 33333)
            _track(polished)
        except Exception:
            pass

    # Phase 3: Try another length (15% budget)
    phase3_end = t0 + budget * 0.87
    alt_lengths = [256, 512]
    if n_parent not in alt_lengths:
        alt_lengths.insert(0, n_parent)
    alt_lengths = [L for L in alt_lengths if L != len(best_seq)]

    if alt_lengths and time.monotonic() < phase3_end - 5.0:
        L = alt_lengths[0]
        phase3_dur = max(5.0, phase3_end - time.monotonic())
        init_384 = np.interp(np.linspace(0, 1, L),
                             np.linspace(0, 1, len(best_seq)), best_seq)
        try:
            res3, _ = _lp_optimize(init_384, min(phase3_end, time.monotonic() + phase3_dur), seed + 99999)
            _track(res3)
        except Exception:
            pass
        if time.monotonic() < phase3_end - 3.0:
            try:
                pol3, _ = _coord_descent(res3, phase3_end - 1.0, seed + 55555)
                _track(pol3)
            except Exception:
                pass

    # Phase 4: Final polish (remaining time)
    if time.monotonic() < deadline - 10.0:
        try:
            polished, _ = _coord_descent(best_seq, deadline - 5.0, seed + 88888)
            _track(polished)
        except Exception:
            pass

    # One more LP round if time permits
    if time.monotonic() < deadline - 15.0:
        try:
            res4, _ = _lp_optimize(best_seq, deadline - 5.0, seed + 123456)
            _track(res4)
        except Exception:
            pass
        if time.monotonic() < deadline - 5.0:
            try:
                polished, _ = _coord_descent(best_seq, deadline - 3.0, seed + 70001)
                _track(polished)
            except Exception:
                pass

    # Final validation
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        fallback = np.ones(384)
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            return [1.0]
        best_seq = fallback

    return [float(x) for x in best_seq]
