            if cs < 1e-12:

                continue
            cand = cand / cs
            new_M = float(np.convolve(cand, cand).max())

            if new_M < M * (1 - 1e-12):
                a = cand
                eta = min(eta * 1.05, 0.1)
            else:
                eta *= 0.7

            if eta < 1e-6:
                break

        return local_best

    # Log-sum-exp smoothing polish
    def lse_polish(a_init, stop_time, max_iters=200):
        nn = a_init.size
        a = a_init.copy()
        a = a / a.sum()

        local_best = a.copy()
        local_best_score = score(a)

        tau = 0.3 / nn

        def objective_and_grad(x):
            a_t = np.maximum(x, 1e-12)
            s = a_t.sum()
            a_t = a_t / s
            b = np.convolve(a_t, a_t)
            # Softmax weights
            scaled = b / tau
            max_s = scaled.max()
            w = np.exp(scaled - max_s)
            w /= w.sum()
            obj = 2.0 * nn * tau * (max_s + np.log(w.sum()))
            # Gradient: 2n * sum_k w_k * 2*a_{k-i} for each i
            # grad[i] = 4n * sum_j w[i+j] * a[j]
            grad = np.zeros(nn)
            for i in range(nn):
                k_lo = max(0, i)
                k_hi = min(2 * nn - 2, i + nn - 1)
                j_lo = k_lo - i
                j_hi = k_hi - i
                grad[i] = np.dot(w[k_lo:k_hi + 1], a_t[j_lo:j_hi + 1])
            grad *= 4.0 * nn
            # Project to tangent space (sum a = 1)
            grad -= grad.mean()
            return obj, grad

        # Use L-BFGS-B
        try:
            def obj_func(x):
                o, _ = objective_and_grad(x)
                return o

            def grad_func(x):
                _, g = objective_and_grad(x)
                return g

            res = minimize(obj_func, a, method='L-BFGS-B', jac=grad_func,
                           options={'maxiter': max_iters, 'ftol': 1e-14, 'gtol': 1e-8})
            a_res = np.maximum(res.x, 0)
            if a_res.sum() < 1e-12:
                return local_best
            a_res = a_res / a_res.sum()
            sc = score(a_res)
            if sc < local_best_score:
                local_best = a_res.copy()
                local_best_score = sc
        except Exception:
            pass

        return local_best

    # ============ MAIN OPTIMIZATION ============

    # Phase 1: Primary LP from parent
    p1_end = t0 + min(budget_s * 0.25, 300.0)
    result = lp_optimize(a0, stop_time=p1_end, max_iters=100000, seed=42)
    update_best(result)

    # Phase 2: Top-k gradient polish
    if remaining() > 80:
        p2_end = t0 + min(budget_s * 0.40, 450.0)
        result = topk_polish(best_a, stop_time=p2_end, max_iters=5000)
        update_best(result)

    # Phase 3: More LP from polished result
    if remaining() > 80:
        p3_end = t0 + min(budget_s * 0.55, 600.0)
        result = lp_optimize(best_a, stop_time=p3_end, max_iters=100000, seed=123)
        update_best(result)

    # Phase 4: LSE polish
    if remaining() > 50:
        p4_end = t0 + min(budget_s * 0.65, 700.0)
        result = lse_polish(best_a, stop_time=p4_end, max_iters=100)
        update_best(result)

    # Phase 5: Top-k polish again
    if remaining() > 40:
        p5_end = t0 + min(budget_s * 0.72, 780.0)
        result = topk_polish(best_a, stop_time=p5_end, max_iters=3000)
        update_best(result)

    # Phase 6: Try nearby lengths
    if remaining() > 60:
        base_n = best_a.size
        for dn in [-4, -2, -1, 1, 2, 4, 6, 8]:
            if remaining() < 30:
                break
            nt = base_n + dn
            if nt < 40 or nt > 500:
                continue
            x_old = np.linspace(0, 1, base_n)
            x_new = np.linspace(0, 1, nt)
            init = np.interp(x_new, x_old, best_a)
            init = np.clip(init, 0, 1000)
            if init.sum() < 0.01:
                continue
            init = init / init.sum()
            alloc = min(remaining() * 0.25, 50.0)
            try:
                result = lp_optimize(init, stop_time=time.monotonic() + alloc,
                                     max_iters=20000, seed=1000 + nt)
                update_best(result)
            except Exception:
                pass

    # Phase 7: Final LP polish
    if remaining() > 30:
        p7_end = t0 + budget_s * 0.90
        result = lp_optimize(best_a, stop_time=p7_end, max_iters=50000, seed=999)
        update_best(result)

    # Phase 8: Final top-k polish
    if remaining() > 15:
        result = topk_polish(best_a, stop_time=t0 + budget_s * 0.95, max_iters=2000)
        update_best(result)

    # Final validation
    final_arr = np.clip(best_a, 0, 1000.0)
    final_list = [float(x) for x in final_arr]
    try:
        fs = evaluate_sequence(final_list)
        if not np.isfinite(fs):
            final_list = [1.0 / 201.0] * 201
    except Exception:
        final_list = [1.0 / 201.0] * 201

    return final_list
