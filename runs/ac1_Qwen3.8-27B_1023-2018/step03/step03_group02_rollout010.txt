```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(384, dtype=np.float64)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    return p

def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    return 2.0 * n * m / (s * s)

def _lp_direction(f, n, f_sum):
    """Solve min t s.t. (f*f)_k + 2*(f*g)_k <= t, sum(g) = f_sum, g >= 0"""
    ff = _conv_full(f)  # length 2n-1
    # Build constraint matrix: for each k, sum_j 2*f[k-j]*g[j] - t <= -(f*f)_k
    # Variables: g_0, ..., g_{n-1}, t
    # A_ub shape: (2n-1, n+1)
    A = np.zeros((2*n-1, n+1))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = 2.0 * f[idx]
        A[k, n] = -1.0  # coefficient of t
    b_ub = -ff  # right-hand side
    
    # Equality: sum(g) = f_sum
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([f_sum])
    
    c = np.zeros(n+1)
    c[n] = 1.0  # minimize t
    
    res = linprog(c, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=[(0, None)]*n + [(None, None)],
                  method='highs', options={'presolve': True, 'time_limit': 10.0})
    
    if res.status not in (0, 3) or res.x is None:
        return None, None
    
    g = np.asarray(res.x[:n], dtype=np.float64)
    t_val = float(res.x[n])
    return g, t_val
```

```python
def _build_sparse_A(f, n):
    """Build sparse matrix for LP constraints.
    Row k: 2*f[k-j] for j in valid range, -1 for t column
    """
    m = 2 * n - 1
    rows = []
    cols = []
    vals = []
    
    for k in range(m):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(2.0 * f[k - j])
        # t column
        rows.append(k)
        cols.append(n)
        vals.append(-1.0)
    
    A = csc_matrix((vals, (rows, cols)), shape=(m, n + 1))
    return A

    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)

        A = _build_sparse_A(f, n)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        
        res = linprog(c_obj, A_ub=A, b_ub=-ff, A_eq=A_eq, b_eq=b_eq,
                      bounds=[(0, None)]*n + [(None, None)],
                      method='highs', options={'presolve': True, 'time_limit': 5.0})
        
        if res.status not in (0, 3) or res.x is None:

            f /= f.sum()
            b_cur = _conv_full(f)
            M = float(b_cur.max())
            continue
        
        g = np.asarray(res.x[:n], dtype=np.float64)
        g_sum = float(g.sum())
        if g_sum < 1e-10:

            if stall > max_stall:

            continue

            tau = 0.3
            continue

                best_f = f.copy()
                best_score = sc
                b_cur = _conv_full(f)
                M = float(b_cur.max())
                tau = min(1.0, tau_try * 1.1)
                stall = 0
                improved = True
                break
            elif sc < best_score:
                f = f_new
                tau = tau_try
                stall = max(0, stall - 1)
                break
        
        if not improved:
            stall += 1
            tau = max(0.1, tau * 0.7)
            if stall >= max_stall:

                strategy = iter_count % 3
                if strategy == 0:

                elif strategy == 1:

                    b_cur = _conv_full(f)

                    for i in range(n):

                    f = np.clip(f * (1 + noise), 0, None)
                f /= f.sum()
                tau = 0.4
                stall = 0

    return best_f, best_score

def _coord_descent(a, stop_time, seed=0):

    n = len(a)
    best = np.clip(np.asarray(a, dtype=np.float64).copy(), 0, None)
    fs = best.sum()
    if fs < 1e-10:
        return best, _score(best)

                sc = _score(best)

                    best_val_sc = sc
```

```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog
from scipy.sparse import csc_matrix


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(384, dtype=np.float64)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    return p


def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _lp_min_peak_direction(f, n, time_limit=8.0):
    """Solve: min t s.t. (f*f)_k + 2*(f*g)_k <= t, sum(g)=1, g>=0.
    Returns direction g (normalized to sum=1) and predicted peak t."""
    ff = _conv_full(f)
    m = 2 * n - 1
    # Build dense constraint matrix (small enough: 767 x 385 for n=384)
    A = np.zeros((m, n + 1))
    for k in range(m):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_hi >= j_lo:
            js = np.arange(j_lo, j_hi + 1)
            A[k, js] = 2.0 * f[k - js]
        A[k, n] = -1.0
    b_ub = -ff
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=[(0, None)] * n + [(None, None)],
                  method='highs', options={'presolve': True, 'time_limit': time_limit})
    if res.status not in (0, 3) or res.x is None:
        return None, None
    g = np.asarray(res.x[:n], dtype=np.float64)
    g_sum = float(g.sum())
    if g_sum < 1e-12:
        return None, None
    g = g / g_sum
    t_val = float(res.x[n])
    return g, t_val


def _lp_max_sum_direction(f, n, M, time_limit=8.0):
    """Solve: max sum(g) s.t. (f*g)_k <= M, g>=0. Returns g/sum(g)."""
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_hi >= j_lo:
            js = np.arange(j_lo, j_hi + 1)
            A[k, js] = f[k - js]
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A, b_ub=np.full(m, M),
                  bounds=[(0, None)] * n,
                  method='highs', options={'presolve': True, 'time_limit': time_limit})
    if res.status not in (0, 3) or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g_sum = float(g.sum())
    if g_sum < 1e-12:
        return None
    return g / g_sum


def _lp_optimize(f0, n, stop_time, seed=42):
    """Iterative LP-based optimization with adaptive step and perturbations."""
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel()[:n].copy(), 0, None)
    while f.size < n:
        f = np.concatenate([f, np.zeros(n - f.size)])
    f = f[:n]
    f_sum = float(f.sum())
    if f_sum < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / f_sum

    best_f = f.copy()
    best_score = _score(f)
    tau = 0.5
    stall = 0
    max_stall = 25
    iter_count = 0
    lp_variant = 0  # 0=min-peak, 1=max-sum

    while time.monotonic() < stop_time - 2.0:
        ff = _conv_full(f)
        M = float(ff.max())

        # Alternate LP formulations
        g = None
        if lp_variant == 0 or stall > 10:
            g, _ = _lp_min_peak_direction(f, n)
            if g is not None:
                lp_variant = 1
            else:
                lp_variant = 1
        if g is None:
            g = _lp_max_sum_direction(f, n, M)
            if g is not None:
                lp_variant = 0

        if g is None:
            stall += 3
            if stall > max_stall * 2:
                break
            # Perturb
            f = np.clip(f + rng.normal(0, 0.05, n), 0, None)
            fs = f.sum()
            if fs > 1e-10:
                f /= fs
            else:
                f = best_f.copy()
            continue

        dist = float(np.abs(g - f).sum())
        if dist < 1e-7:
            stall += 1
            if stall > max_stall:
                break
            f = np.clip(f + rng.normal(0, 0.04, n), 0, None)
            fs = f.sum()
            if fs > 1e-10:
                f /= fs
            else:
                f = best_f.copy()
            continue

        # Line search with multiple tau
        improved = False
        tau_candidates = sorted(set([
            min(1.0, tau), min(1.0, tau * 0.7), min(1.0, tau * 0.4),
            min(1.0, tau * 1.4), 0.1, 0.3, min(1.0, tau * 0.2)
        ]), key=lambda x: abs(x - tau))

        for tau_try in tau_candidates:
            if tau_try <= 0.01 or tau_try > 1.0:
                continue
            f_new = (1.0 - tau_try) * f + tau_try * g
            f_new = np.clip(f_new, 0, None)
            fs_new = float(f_new.sum())
            if fs_new < 1e-10:
                continue
            f_new = f_new / fs_new
            sc = _score(f_new)
            if sc < best_score - 1e-13:
                f = f_new
                best_f = f.copy()
                best_score = sc
                tau = min(1.0, tau_try * 1.15)
                stall = 0
                improved = True
                break
            elif sc < best_score:
                f = f_new
                tau = tau_try
                stall = max(0, stall - 1)
                break

        if not improved:
            stall += 1
            tau = max(0.05, tau * 0.7)
            if stall >= max_stall:
                # Strategic perturbation
                strat = iter_count % 4
                if strat == 0:
                    f = np.clip(f + rng.normal(0, 0.1, n), 0, None)
                elif strat == 1:
                    # Subgradient perturbation
                    b_cur = _conv_full(f)
                    k_star = int(np.argmax(b_cur))
                    sg = np.zeros(n)
                    for i in range(n):
                        j = k_star - i
                        sg[i] = 2.0 * (f[j] if 0 <= j < n else 0.0)
                    sg -= 2.0 * M / max(f.sum(), 1e-10)
                    f = np.clip(f - 0.1 * sg, 0, None)
                elif strat == 2:
                    # Smooth noise
                    kern = np.exp(-0.5 * (np.arange(-6, 7) / 2.2)**2)
                    kern /= kern.sum()
                    noise = np.convolve(rng.normal(0, 0.15, n), kern, 'same')
                    f = np.clip(f * (1.0 + noise), 0, None)
                else:
                    # Block swap
                    blk = n // 5
                    f_new = f.copy()
                    f_new[:blk], f_new[-blk:] = f[-blk:].copy(), f[:blk].copy()
                    f = f_new
                fs = f.sum()
                if fs > 1e-10:
                    f /= fs
                else:
                    f = best_f.copy()
                tau = 0.4
                stall = 0

        iter_count += 1

    return best_f, best_score


def _coord_descent(a, stop_time, seed=0):
    """Coordinate descent polishing with adapted search."""
    n = len(a)
    best = np.clip(np.asarray(a, dtype=np.float64).ravel(), 0, None).copy()
    fs = best.sum()
    if fs < 1e-10:
        return best, _score(best)
    best /= fs
    best_sc = _score(best)
    rng = np.random.default_rng(seed)
    no_imp = 0
    max_no_imp = 5

    while time.monotonic() < stop_time - 2.0 and no_imp < max_no_imp:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 2.0:
                break
            orig = best[i]
            bv = orig
            bv_sc = best_sc
            if orig > 1e-9:
                multis = (0.2, 0.4, 0.6, 0.8, 0.9, 1.0, 1.1, 1.25, 1.5, 2.0, 0.0)
            else:
                multis = (0.0, 0.01, 0.1, 0.5)
            for m in multis:
                nv = orig * m if orig > 1e-9 else m
                best[i] = nv
                if best.sum() < 0.01:
                    best[i] = orig
                    continue
                sc = _score(best)
                if sc < bv_sc - 1e-13:
                    bv = nv
                    bv_sc = sc
            best[i] = bv
            if bv_sc < best_sc - 1e-13:
                best_sc = bv_sc
                improved = True

        if improved:
            no_imp = 0
        else:
            no_imp += 1

        # Pair moves
        if no_imp == 0 and time.monotonic() < stop_time - 2.0:
            for _ in range(200):
                if time.monotonic() > stop_time - 2.0:
                    break
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j:
                    continue
                oi, oj = best[i], best[j]
                for ratio in (0.3, 0.5, 0.7, 1.0, 1.3, 0.0):
                    total = oi + oj
                    if total < 1e-12:
                        continue
                    best[i] = ratio * total
                    best[j] = (1.0 - ratio) * total
                    sc = _score(best)
                    if sc < best_sc - 1e-13:
                        best_sc = sc
                        improved = True
                        break
                best[i], best[j] = oi, oj

    return best, best_sc


def _subgradient_step(f, n, eta=0.05):
    """One subgradient step on the score."""
    b = _conv_full(f)
    k_star = int(np.argmax(b))
    M = float(b[k_star])
    s = float(f.sum())
    # Subgradient of max_k conv(a)_k at a = f
    sg = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            sg[i] += 2.0 * f[j]
    # Direction: -sg (decrease coordinates contributing to peak)
    f_new = np.clip(f - eta * sg, 0, None)
    fn = f_new.sum()
    if fn > 1e-10:
        f_new /= fn
    else:
        f_new = f.copy()
    return f_new
```
