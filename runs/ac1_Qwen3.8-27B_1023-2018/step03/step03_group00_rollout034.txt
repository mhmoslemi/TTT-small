def _conv_full(a):
    n = len(a)
    if n <= 512:
        return np.convolve(a, a, mode="full")
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2*n-1]
    np.maximum(c, 0.0, out=c)
    return c

    n = len(a)
    s = float(a.sum())
    if s < 0.01:

    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)

def _gradient(a, k_star):
    """Compute gradient of b[k_star] with respect to a."""
    n = len(a)
    # db[k*]/da[j] = 2 * a[k*-j] if 0 <= k*-j < n else 0
    # This is 2 * reversed_a shifted by k*
    grad = np.zeros(n)
    # a[k*-j] for j in range(n): valid when 0 <= k*-j < n, i.e., k*-n+1 <= j <= k*

    hi = min(n-1, k_star)
    if lo <= hi:
        # For j in [lo, hi]: a[k_star - j]
        indices = np.arange(lo, hi+1)
        grad[indices] = 2.0 * a[k_star - indices]
    return grad

        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            grad[idx] = 2.0 * a[k_star - idx]

        sc_grad = (2.0 * n / (s * s)) * grad - (4.0 * n * m / (s * s * s))

        a_new = a - lr * sc_grad
        a_new = np.clip(a_new, 0, None)
        if a_new.sum() < 0.01:
            lr = max(1e-9, lr * 0.5)
            continue

        new_sc = _score(a_new)
        if new_sc < sc - 1e-13:
            a = a_new
            lr = min(0.1, lr * 1.03)
        else:
            lr = max(1e-9, lr * 0.6)

    return best_a, best_sc


def _cordescent(a, stop_time, seed):
    rng = np.random.default_rng(seed)
    n = len(a)
    a = np.clip(a.astype(np.float64).copy(), 0, None)
    if a.sum() < 1e-10:
        return a, _score(a)
    best_sc = _score(a)
    no_imp = 0

    while time.monotonic() < stop_time:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time:
                break
            orig = a[i]
            if orig < 1e-14:
                continue
            best_val = orig
            best_val_sc = best_sc
            for mlt in (0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0, 0.0):
                a[i] = orig * mlt
                if a.sum() < 0.01:
                    a[i] = orig
                    continue
                sc = _score(a)
                if sc < best_val_sc - 1e-14:
                    best_val = a[i]
                    best_val_sc = sc
            a[i] = best_val
            if best_val_sc < best_sc - 1e-14:
                best_sc = best_val_sc
                improved = True

        if not improved:
            no_imp += 1
            if no_imp >= 8:
                break
        else:
            no_imp = 0

    return a, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 20.0

    parent = _get_parent()
    n_p = len(parent)

    fallback = parent.copy()
    fallback_sc = float(evaluate_sequence([float(x) for x in fallback]))
    if not math.isfinite(fallback_sc):
        fallback = np.ones(256)
        fallback_sc = float(evaluate_sequence([float(x) for x in fallback]))

    best_seq = fallback.copy()
    best_score = fallback_sc

    def _try(n, init_arr, stop, sd):
        nonlocal best_seq, best_score
        try:
            res, sc = _lp_optimize(init_arr, n, stop, sd)
        except Exception:
            return
        if res is None or len(res) == 0:
            return
        if not np.all(np.isfinite(res)):
            return
        res = np.clip(res, 0.0, 1000.0)
        if res.sum() < 0.01:
            return
        try:
            exact = float(evaluate_sequence([float(x) for x in res]))
        except Exception:
            exact = math.inf
        if math.isfinite(exact) and exact < best_score - 1e-12:
            best_seq = res.copy()
            best_score = exact

    # Phase 1: LP on parent length with multiple initializations
    n = n_p
    phase1_end = t0 + budget * 0.55

    rng_main = np.random.default_rng(seed)
    i_arr = np.arange(n, dtype=np.float64)

    inits = []
    inits.append(parent.copy())

    # Smoothed parent
    from scipy.ndimage import uniform_filter1d
    inits.append(np.clip(uniform_filter1d(parent, size=5), 0, None))

    # Raised cosine
    inits.append(0.5 * (1.0 + np.cos(2.0 * math.pi * i_arr / n)))

    # Sine window
    inits.append(np.sin(math.pi * (i_arr + 0.5) / n))

    # Two-lobe
    center = (n - 1) / 2.0
    w = n / 6.0
    inits.append(np.exp(-0.5 * ((i_arr - n * 0.25) / w) ** 2) + np.exp(-0.5 * ((i_arr - n * 0.75) / w) ** 2))

    # Parent + moderate noise
    sig = 0.1 * max(1.0, float(np.mean(parent)))
    inits.append(np.clip(parent + rng_main.normal(0, sig, n), 0, None))

    # Power-of-sine
    inits.append(np.sin(math.pi * (i_arr + 0.5) / n) ** 0.7)

    # Flat top with tapered edges
    taper = max(3, n // 20)
    flat = np.ones(n)
    for k in range(taper):
        frac = (k + 1) / (taper + 1)
        flat[k] = 0.5 * (1 + math.cos(math.pi * frac))
        flat[n - 1 - k] = 0.5 * (1 + math.cos(math.pi * frac))
    inits.append(flat)

    num_inits = min(len(inits), 7)
    time_per_init = max(15.0, (phase1_end - t0) / num_inits)

    for idx in range(num_inits):
        if time.monotonic() > phase1_end - 8.0:
            break
        if best_score < 1.50:
            break
        stop = min(phase1_end - 5.0, time.monotonic() + time_per_init)
        _try(n, inits[idx], stop, seed + 10000 * (idx + 1))

    # Phase 2: Second LP pass on current best (warm start)
    phase2_end = t0 + budget * 0.72
    if time.monotonic() < phase2_end - 20.0:
        _try(len(best_seq), best_seq.copy(), phase2_end - 5.0, seed + 99999)

    # Phase 3: Try different lengths
    phase3_end = t0 + budget * 0.85
    other_lengths = [320, 384, 400, 420, 432, 448, 464, 480, 500, 512]
    other_lengths = [L for L in other_lengths if L != n and time.monotonic() < phase3_end - 10]
    if other_lengths:
        time_per_len = max(8.0, (phase3_end - max(phase2_end, time.monotonic())) / len(other_lengths))
        for L in other_lengths:
            if time.monotonic() > phase3_end - 8.0:
                break
            if best_score < 1.50:
                break
            stop = min(phase3_end - 5.0, time.monotonic() + time_per_len)
            if abs(L - len(best_seq)) <= 8:
                init_l = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, len(best_seq)), best_seq)
            else:
                init_l = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, n), parent)
            _try(L, init_l, stop, seed + 500000 + L)

    # Phase 4: Gradient refinement on best
    phase4_end = t0 + budget * 0.93
    if time.monotonic() < phase4_end - 10.0:
        try:
            ga, gsc = _grad_refine(best_seq, phase4_end - 3.0, seed + 777777)
            if np.all(np.isfinite(ga)):
                ga = np.clip(ga, 0, 1000)
                exact = float(evaluate_sequence([float(x) for x in ga]))
                if math.isfinite(exact) and exact < best_score - 1e-12:
                    best_seq = ga
                    best_score = exact
        except Exception:
            pass

    # Phase 5: Coordinate descent polish
    polish_end = deadline - 5.0
    if time.monotonic() < polish_end - 10.0:
        try:
            ca, csc = _cordescent(best_seq, polish_end, seed + 888888)
            if np.all(np.isfinite(ca)):
                ca = np.clip(ca, 0, 1000)
                exact = float(evaluate_sequence([float(x) for x in ca]))
                if math.isfinite(exact) and exact < best_score - 1e-12:
                    best_seq = ca
                    best_score = exact
        except Exception:
            pass

    # Second coordinate descent with different seed
    if time.monotonic() < deadline - 15.0:
        try:
            ca2, csc2 = _cordescent(best_seq, deadline - 8.0, seed + 999888)
            if np.all(np.isfinite(ca2)):
                ca2 = np.clip(ca2, 0, 1000)
                exact = float(evaluate_sequence([float(x) for x in ca2]))
                if math.isfinite(exact) and exact < best_score - 1e-12:
                    best_seq = ca2
                    best_score = exact
        except Exception:
            pass

    # Final validation
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in fallback]

    return [float(x) for x in best_seq]
