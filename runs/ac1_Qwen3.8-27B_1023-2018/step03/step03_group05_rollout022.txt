    nfft = next_fast_len(n + m - 1)
    fa = rfft(a, nfft)
    fb = rfft(b, nfft)
    return irfft(fa * fb, nfft)[:n+m-1]

        a_list = [float(x) for x in a]
        return evaluate_sequence(a_list)

    n = len(a)

    z = (c - M) / T

    w = np.exp(z)
