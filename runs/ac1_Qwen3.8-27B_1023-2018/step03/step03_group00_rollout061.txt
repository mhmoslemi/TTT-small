import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(198, dtype=np.float64)

    if p.size == 0:
        p = np.ones(198, dtype=np.float64)
    return p


def _fft_conv(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return math.inf
    b = _fft_conv(a)
    m = float(np.max(b))
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _subgrad_maxconv(a):
    n = len(a)
    b = _fft_conv(a)
    m = float(np.max(b))
    eps = max(1e-15, 1e-11 * m)
    active = np.flatnonzero(b >= m - eps)
    grad = np.zeros(n)
    for k in active:
        i_lo = max(0, k - n + 1)
        i_hi = min(k, n - 1)
        if i_lo <= i_hi:
            idx = k - np.arange(i_lo, i_hi + 1)
            grad[i_lo:i_hi + 1] += 2.0 * a[idx]
    if len(active) > 0:
        grad /= len(active)
    return grad, m


def _obj_subgrad(a):
    n = len(a)
    S = float(np.sum(a))
    g_max, M = _subgrad_maxconv(a)
    f_sub = 2.0 * n * (g_max * S - 2.0 * M) / (S ** 3)
    return f_sub


def _build_A(f, n):
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_direction(f, n, M, timeout=5.0):
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.clip(res.x, 0.0, None)
    if g.sum() < 1e-15:
        return None
    return g

def _coord_descent(f0, stop_time, seed, max_epochs=50):

    best_f = f.copy()

            break
        improved = False
        order = rng.permutation(n)

                break

                continue

                    best_v = f[i]
                    best_v_sc = sc
            f[i] = best_v
            if best_v_sc < best_sc - 1e-15:
                best_sc = best_v_sc
                best_f = f.copy()
                improved = True

                        best_f = f.copy()
