import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _conv_full(a):
    n = len(a)
    if n <= 256:
        return np.convolve(a, a, mode="full")
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[: 2 * n - 1]

    s = float(a.sum())
    if s < 0.01:

    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m):

    return 2.0 * n * m / (s * s)

    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):

            lr *= 0.7
            no_improve += 1

    return best, best_sc


def _coord_polish(f, stop_time, rng):
    n = len(f)
    best = f.copy()
    best_sc = _score(best)
    no_improve = 0
    max_no = 12

    while time.monotonic() < stop_time and no_improve < max_no:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time:
                break
            orig = best[i]
            if orig < 1e-14:
                continue
            best_val = orig
            best_val_sc = best_sc
            for m in (0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.6, 2.0, 0.0):
                best[i] = orig * m if m > 0 else 0.0
                if best.sum() < 0.01:
                    best[i] = orig
                    continue
                sc = _score(best)
                if sc < best_val_sc - 1e-15:
                    best_val = best[i]
                    best_val_sc = sc
            best[i] = best_val
            if best_val_sc < best_sc - 1e-15:
                best_sc = best_val_sc
                improved = True
        if not improved:
            no_improve += 1

    return best, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 12.0

    # Get parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel().copy()
        parent = np.where(np.isfinite(parent), parent, 0.0)
        parent = np.clip(parent, 0.0, 1000.0)
    except Exception:
        parent = np.ones(416, dtype=np.float64)
    if parent.size == 0:
        parent = np.ones(416, dtype=np.float64)

    n = len(parent)
    rng = np.random.default_rng(seed)

    # Normalize parent to sum=1
    f = parent / parent.sum() if parent.sum() > 0 else np.ones(n) / n
    best_f = f.copy()
    best_score = _score(best_f)

    # === Phase 1: LP-based optimization (55% of time) ===
    phase1_end = t0 + (deadline - t0) * 0.55

    t = 0.25
    stall = 0
    max_stall = 200
    it = 0

    while time.monotonic() < phase1_end - 3.0:
        b = _conv_full(f)
        M = float(b.max())
        if M < 1e-20:
            break

        g = _lp_dir(f, M)
        if g is None:
            stall += 4
            if stall > max_stall:
                break
            f = np.clip(f + rng.normal(0, 0.05 / n, n), 0, None)
            fs = f.sum()
            f = f / fs if fs > 1e-12 else np.ones(n) / n
            continue

        g /= g.sum()

        # Check direction meaningfulness
        dn = np.linalg.norm(g - f)
        fn = np.linalg.norm(f)
        if fn > 0 and dn / fn < 1e-10:
            stall += 2
            if stall > max_stall:
                break
            # Perturb to escape
            f = np.clip(f + rng.normal(0, 0.02 / n, n), 0, None)
            f /= f.sum()
            continue

        # Line search
        best_t, best_tc = _line_search(f, g, t, 0)

        if best_tc < best_score - 1e-13:
            f_new = (1.0 - best_t) * f + best_t * g
            f_new = np.clip(f_new, 0, None)
            fs = f_new.sum()
            f = f_new / fs if fs > 1e-12 else f
            best_f = f.copy()
            best_score = best_tc
            t = best_t
            stall = 0
        else:
            stall += 1
            t = max(0.05, t * 0.75)
            if stall >= max_stall:
                # Reseed from best with perturbation
                f = np.clip(best_f + rng.normal(0, 0.03 / n, n), 0, None)
                f /= f.sum()
                t = 0.25
                stall = 0
                max_stall = int(max_stall * 1.4)

        it += 1
        if it % 50 == 0:
            remaining = deadline - time.monotonic()
            # print(f"  Phase1 it={it} score={best_score:.10f} t={t:.3f} stall={stall} rem={remaining:.0f}s")

    # === Phase 2: Second LP pass with tightened bound (25% of time) ===
    phase2_end = t0 + (deadline - t0) * 0.80
    if time.monotonic() < phase2_end - 10.0:
        f = best_f.copy()
        t = 0.15
        stall = 0
        max_stall = 150

        while time.monotonic() < phase2_end - 3.0:
            b = _conv_full(f)
            M = float(b.max())
            if M < 1e-20:
                break

            # Try slightly tighter bound
            M_tight = M * 0.9995
            g = _lp_dir(f, M_tight)
            if g is None:
                g = _lp_dir(f, M)
            if g is None:
                stall += 3
                if stall > max_stall:
                    break
                f = np.clip(f + rng.normal(0, 0.01 / n, n), 0, None)
                f /= f.sum()
                continue

            g /= g.sum()
            best_t, best_tc = _line_search(f, g, t, 0)

            if best_tc < best_score - 1e-13:
                f_new = (1.0 - best_t) * f + best_t * g
                f_new = np.clip(f_new, 0, None)
                fs = f_new.sum()
                f = f_new / fs if fs > 1e-12 else f
                best_f = f.copy()
                best_score = best_tc
                stall = 0
            else:
                stall += 1
                if stall >= max_stall:
                    f = np.clip(best_f + rng.normal(0, 0.01 / n, n), 0, None)
                    f /= f.sum()
                    stall = 0
                    max_stall += 50

    # === Phase 3: Gradient polish (10% of time) ===
    phase3_end = min(deadline - 20.0, t0 + (deadline - t0) * 0.90)
    if time.monotonic() < phase3_end - 5.0:
        try:
            f2, sc2 = _gradient_polish(best_f, phase3_end, rng)
            if sc2 < best_score - 1e-14:
                best_f = f2
                best_score = sc2
        except Exception:
            pass

    # === Phase 4: Coordinate polish (remaining time) ===
    phase4_end = deadline - 5.0
    if time.monotonic() < phase4_end - 3.0:
        try:
            f3, sc3 = _coord_polish(best_f, phase4_end, rng)
            if sc3 < best_score - 1e-14:
                best_f = f3
                best_score = sc3
        except Exception:
            pass

    # === Final: second coordinate polish if time remains ===
    if time.monotonic() < deadline - 4.0:
        try:
            f4, sc4 = _coord_polish(best_f, deadline - 3.0, np.random.default_rng(seed + 1))
            if sc4 < best_score - 1e-14:
                best_f = f4
                best_score = sc4
        except Exception:
            pass

    # Final validation with exact evaluator
    best_f = np.clip(best_f, 0.0, 1000.0)
    best_f = np.where(np.isfinite(best_f), best_f, 0.0)
    if best_f.sum() < 0.01:
        best_f = parent / parent.sum()

    final_list = [float(x) for x in best_f]
    try:
        final_sc = float(evaluate_sequence(final_list))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        final_list = [float(x) for x in parent]
        try:
            final_sc = float(evaluate_sequence(final_list))
        except Exception:
            pass
        if not math.isfinite(final_sc):
            return [1.0] * n

    return final_list
