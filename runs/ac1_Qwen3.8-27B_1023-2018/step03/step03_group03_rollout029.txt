import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _conv_full(a):
    n = len(a)
    if n <= 800:
        return np.convolve(a, a, mode="full")
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score_from_conv(conv, S, n):
    if S < 0.01:
        return math.inf
    return 2.0 * n * float(conv.max()) / (S * S)

def _score(a):
    n = len(a)
    S = float(a.sum())
    if S < 0.01:
        return math.inf
    conv = _conv_full(a)
    return _score_from_conv(conv, S, n)

def _conv_full(a):
    n = len(a)
    if n <= 1000:
        return np.convolve(a, a, mode="full")
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)

def _lp_direction(f, n, M, time_limit=5.0):
    """Find LP direction: max sum(g) s.t. (f*g)_k <= M, g >= 0"""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j+n, j] = f
    c_obj = -np.ones(n)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=np.full(2*n-1, M),
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g_sum = float(g.sum())
    if g_sum < 1e-15:
        return None
    return g / g_sum

            if S > 1e-10:

                    f = best_f.copy()
                t = 0.15
                stall = 0

    return best_f, best_score

def _conv_full(a):
    n = len(a)
    if n <= 1200:
        return np.convolve(a, a, mode="full")
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _subgradient_step(f, alpha, n_active=3):
    """One subgradient step to reduce max(f*f)."""
    n = len(f)
    S = float(f.sum())
    if S < 1e-10:
        return f
    f = f / S
    conv = _conv_full(f)
    M = float(conv.max())
    # Find active set (top n_active peaks)
    threshold = M * 0.9999
    active = np.where(conv >= threshold)[0]
    if len(active) == 0:
        active = [np.argmax(conv)]
    elif len(active) > n_active:
        top_idx = np.argsort(conv[active])[-n_active:]
        active = active[top_idx]

    # Subgradient of max(f*f)
    grad = np.zeros(n)
    for k in active:
        for i in range(n):
            # d(f*f)_k / d f_i = f[k-i] + f[i-k] (with zero extension)
            ki = k - i
            ik = i - k
            g_val = 0.0
            if 0 <= ki < n:
                g_val += f[ki]
            if 0 <= ik < n:
                g_val += f[ik]
            grad[i] += g_val / len(active)

    # Project: f_new = f - alpha * grad, then clip to [0,1] and normalize
    f_new = f - alpha * grad
    f_new = np.clip(f_new, 0, None)
    S_new = float(f_new.sum())
    if S_new < 1e-10:
        return f
    f_new /= S_new
    return f_new

            try:
                alpha = 0.01
                for sg_iter in range(30):
                    if time.monotonic() > phase2_end - 10.0:
                        break
                    f_sg = _subgradient_step(best_seq, alpha, n_active=5)
                    _try_update(f_sg)
                    if best_score < 1.5028:
                        break
                    alpha *= 0.9
            except Exception:
                pass

    # Phase 3: Try different lengths
    phase3_end = t0 + budget * 0.80
    other_lengths = [192, 256, 320, 384, 448, 512, 640]
    if n_parent not in other_lengths:
        other_lengths.append(n_parent)
    other_lengths = sorted(set(other_lengths))
    other_lengths.sort(key=lambda L: abs(L - n_parent))

    time_per_len = max(8.0, (phase3_end - time.monotonic()) / max(1, len(other_lengths)))

    for L in other_lengths:
        if time.monotonic() > phase3_end - 10.0:
            break
        if best_score < 1.5028:
            break
        stop_cd_L = min(phase3_end - 5.0, time.monotonic() + time_per_len)

        # Initialize by interpolation from best
        if len(best_seq) > 1:
            init_L = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, len(best_seq)),
                               best_seq)
        else:
            init_L = np.ones(L)
        init_L = np.clip(init_L, 0, None)
        if init_L.sum() < 1e-6:
            init_L[:] = 1.0 / L

        try:
            cd_res, _ = _coord_descend(init_L, stop_cd_L, seed=seed + 50000 + L,
                                       n_grid=90, max_sweeps=1000)
            _try_update(cd_res)
        except Exception:
            pass

    # Phase 4: Final intensive polish
    final_end = deadline - 10.0
    if time.monotonic() < final_end - 20.0:
        try:
            cd_res, _ = _coord_descend(best_seq, final_end, seed=seed + 77777,
                                       n_grid=150, max_sweeps=5000)
            _try_update(cd_res)
        except Exception:
            pass

    # Pair polish
    if time.monotonic() < final_end - 10.0:
        try:
            pp_res, _ = _pair_polish(best_seq, final_end, max_rounds=15, seed=55555)
            _try_update(pp_res)
        except Exception:
            pass

    # One more coordinate pass
    if time.monotonic() < deadline - 12.0:
        try:
            cd_res, _ = _coord_descend(best_seq, deadline - 12.0, seed=88888,
                                       n_grid=120, max_sweeps=1000)
            _try_update(cd_res)
        except Exception:
            pass

    # Final validation
    best_seq = np.clip(np.asarray(best_seq, dtype=np.float64), 0, 1000)
    best_seq = np.where(np.isfinite(best_seq), best_seq, 0.0)
    if best_seq.size == 0 or best_seq.sum() < 0.01:
        best_seq = fallback.copy()

    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in fallback]

    return [float(x) for x in best_seq]
