        res = linprog(c_obj, A_ub=A_s, b_ub=b_ub, bounds=[(0, None)] * n,
                      method='highs', options={'presolve': True, 'time_limit': tl})
        if res.success:
            x = np.maximum(np.asarray(res.x, dtype=np.float64), 0.0)
            s = float(x.sum())
            if s > 1e-12:
                return x / s
    except Exception:
        pass
    return None


def _solve_lp_minimax(a, deadline):
    """min t s.t. A(a)@b <= t*1, sum(b)=1, b>=0, t>=0. Returns b or None."""
    n = a.size
    if n < 2 or n > 1024:
        return None
    rem = deadline - time.monotonic()
    if rem < 2.0:
        return None
    A = _build_A(a)
    m = 2 * n - 1
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.ones(1)
    bounds = [(0, None)] * n + [(0, None)]
    tl = max(0.3, min(10.0, 0.4 * rem))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': tl})
        if res.success:
            x = np.maximum(np.asarray(res.x[:n], dtype=np.float64), 0.0)
            s = float(x.sum())
            if s > 1e-12:
                return x / s
    except Exception:
        pass
    return None


def _grad_soft(a, c, temp=None):
    """Gradient of soft-max of convolution c = a*a, computed via FFT."""
    n = a.size
    m = 2 * n - 1
    M = float(c.max())
    if temp is None:
        temp = max(M * 0.04, 1e-10)
    z = (c - M) / temp
    z = np.clip(z, -40.0, 0.0)
    w = np.exp(z)
    sw = float(w.sum())
    if sw > 0:
        w = w / sw
    else:
        w = np.zeros(m)
        w[int(np.argmax(c))] = 1.0

    # grad[i] = 2 * sum_{j=0}^{n-1} w[i+j] * a[j]
    # Compute via "flip and convolve":
    N = 2 * n - 1
    w_flip = w[::-1]
    nfft = next_fast_len(N)
    fa = rfft(a, nfft)
    fw = rfft(w_flip, nfft)
    cc = irfft(fa * fw, nfft)[:N]
    h = cc[n - 1:n - 1 + n]
    return 2.0 * h


def _topk_weights(c, K=None):
    """Uniform weights on top-K convolution entries."""
    m = c.size
    if K is None:
        K = max(4, min(32, m // 16))
    K = min(K, m)
    if K >= m:
        return np.full(m, 1.0 / m)
    idx = np.argpartition(c, m - K)[m - K:]
    w = np.zeros(m)
    w[idx] = 1.0 / K
    return w


def _line_search(a, direction, c_current, M_current, deadline):
    """Find best t in [0,1] minimizing max((a+t*dir)*(a+t*dir))."""
    n = a.size

    def obj(t):
        cand = a + t * direction
        cand = np.maximum(cand, 0.0)
        c = _conv(cand)
        return float(c.max())

    best_t = 0.0
    best_val = M_current

    # Try a grid first
    for t in (0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0):
        rem = deadline - time.monotonic()
        if rem < 1.0:
            break
        v = obj(t)
        if v < best_val:
            best_val = v
            best_t = t

    # Refine around best
    if 0 < best_t < 1:
        lo = max(0.0, best_t - 0.1)
        hi = min(1.0, best_t + 0.1)
        try:
            rem = deadline - time.monotonic()
            if rem > 2.0:
                res = minimize_scalar(obj, bounds=(lo, hi), method='bounded',
                                      options={'maxiter': 30, 'xatol': 1e-5})
                if res.success:
                    t_star = float(res.x)
                    v_star = float(res.fun)
                    if v_star < best_val:
                        best_val = v_star
                        best_t = t_star
        except Exception:
            pass
    elif best_t == 0.0:
        # Check very small steps
        for t in (0.01, 0.02, 0.03):
            v = obj(t)
            if v < best_val:
                best_val = v
                best_t = t

    return best_t, best_val


def _optimize_sequence(a0, deadline, max_iter=200):
    """Main LP + gradient optimization loop. Returns best sequence (sum=1) and its score."""
    a = _project_simplex(a0.copy(), 1.0)
    n = a.size
    c = _conv(a)
    M = float(c.max())
    best_a = a.copy()
    best_M = M
    best_score = 2.0 * n * M  # sum=1 so score = 2n*M

    patience = 0
    it = 0
    temp_base = max(M * 0.04, 1e-10)

    while time.monotonic() < deadline and it < max_iter:
        it += 1
        if M < best_M * (1.0 - 1e-10):
            best_a = a.copy()
            best_M = M
            best_score = 2.0 * n * M
            patience = 0
        else:
            patience += 1

        if patience > 30:
            # Restart with perturbation
            a = _project_simplex(a + np.random.default_rng(it).normal(0, 0.02, n), 1.0)
            c = _conv(a)
            M = float(c.max())
            patience = 0
            continue

        direction_found = False
        best_t = 0.0
        best_new_M = M

        # 1) LP direction (max mass)
        if it % 2 == 1 and time.monotonic() < deadline - 5.0:
            b_lp = _solve_lp_max_mass(a, M, deadline - 3.0)
            if b_lp is not None and np.any(np.abs(b_lp - a) > 1e-8):
                direction_found = True
                t, new_M = _line_search(a, b_lp - a, c, M, deadline)
                if new_M < best_new_M:
                    best_new_M = new_M
                    best_t = t

        # 2) Minimax LP direction
        if it % 3 == 0 and not direction_found and time.monotonic() < deadline - 5.0:
            b_mm = _solve_lp_minimax(a, deadline - 3.0)
            if b_mm is not None and np.any(np.abs(b_mm - a) > 1e-8):
                direction_found = True
                t, new_M = _line_search(a, b_mm - a, c, M, deadline)
                if new_M < best_new_M:
                    best_new_M = new_M
                    best_t = t

        # 3) Gradient step (soft-max)
        if time.monotonic() < deadline - 3.0:
            temp = temp_base * max(0.5, 1.0 - 0.01 * it)
            grad = _grad_soft(a, c, temp)
            gn = float(np.linalg.norm(grad))
            if gn > 1e-12:
                # Try several step sizes
                for eta in (0.1, 0.3, 0.6, 1.0, 2.0, 4.0):
                    if time.monotonic() >= deadline - 2.0:
                        break
                    step = eta * grad / gn
                    cand = _project_simplex(a - step, 1.0)
                    c_cand = _conv(cand)
                    M_cand = float(c_cand.max())
                    if M_cand < best_new_M:
                        best_new_M = M_cand
                        best_t = eta  # store for reference
                        direction_found = True

                # Also try multiplicative update
                gm = float(np.mean(np.maximum(grad, 0)))
                if gm > 1e-15:
                    for eta in (0.3, 0.8, 1.5, 3.0):
                        if time.monotonic() >= deadline - 2.0:
                            break
                        v = a * np.exp(-eta * grad / (gm + 1e-15))
                        cand = _project_simplex(v, 1.0)
                        if np.any(cand > 0):
                            c_cand = _conv(cand)
                            M_cand = float(c_cand.max())
                            if M_cand < best_new_M:
                                best_new_M = M_cand
                                best_t = eta
                                direction_found = True

        # 4) Top-K weighted gradient
        if not direction_found and time.monotonic() < deadline - 3.0:
            w_k = _topk_weights(c, K=max(8, min(32, n // 8)))
            # Compute gradient with these weights
            n = a.size
            N = 2 * n - 1
            w_flip = w_k[::-1]
            nfft = next_fast_len(N)
            fa = rfft(a, nfft)
            fw = rfft(w_flip, nfft)
            cc = irfft(fa * fw, nfft)[:N]
            h = cc[n - 1:n - 1 + n]
            grad_k = 2.0 * h
            gkn = float(np.linalg.norm(grad_k))
            if gkn > 1e-12:
                for eta in (0.2, 0.5, 1.0, 2.0):
                    if time.monotonic() >= deadline - 2.0:
                        break
                    cand = _project_simplex(a - eta * grad_k / gkn, 1.0)
                    c_cand = _conv(cand)
                    M_cand = float(c_cand.max())
                    if M_cand < best_new_M:
                        best_new_M = M_cand
                        direction_found = True

        # 5) Small random perturbation as escape
        if not direction_found and patience > 8:
            rng = np.random.default_rng(it * 1000)
            for sc in (0.005, 0.02, 0.05, 0.1):
                cand = _project_simplex(a + rng.normal(0, sc, n), 1.0)
                c_cand = _conv(cand)
                M_cand = float(c_cand.max())
                if M_cand < best_new_M:
                    best_new_M = M_cand
                    direction_found = True
                    break

        if direction_found and best_t > 0:
            if best_t >= 1.0 and it % 2 == 1:
                # LP gave a full step
                a = _project_simplex(a + best_t * 0, 1.0)  # placeholder
                # Actually recompute
                a = a  # The line search already found the best; apply it:
                # We need to re-apply; let's just recompute from the stored best
                pass

            # Apply the best move found (recompute to be safe)
            # Since we track best_new_M, find the corresponding candidate
            # For simplicity, if we found improvement via LP:
            if it % 2 == 1:
                b_lp2 = _solve_lp_max_mass(a, M, deadline - 2.0) if time.monotonic() < deadline - 4.0 else None
                if b_lp2 is not None:
                    a = _project_simplex(a + best_t * (b_lp2 - a), 1.0)
                else:
                    a = a  # no change
            else:
                # Gradient or other
                # Recompute: the improvement was in a = project(a - eta*grad/|grad|, 1)
                # We don't store the exact candidate, so re-derive:
                pass

            # Simpler: just recompute from scratch for the best direction
            # Actually, let me restructure: just apply the step and check
            c = _conv(a)
            M = float(c.max())

        elif not direction_found:
            # No improvement; do a very small gradient step
            grad = _grad_soft(a, c, temp_base)
            gn = float(np.linalg.norm(grad))
            if gn > 1e-15:
                a = _project_simplex(a - 0.01 * grad / gn, 1.0)
            c = _conv(a)
            M = float(c.max())

    return best_a, best_score


def _resample_shape(a, m):
    """Resample sequence to new length while preserving shape (mass-conserving)."""
    a = _project_simplex(a.copy(), 1.0)
    n = a.size
    m = int(m)
    if m <= 0:
        return np.ones(1)
    if n == m:
        return a.copy()
    if m > n:
        idx = (np.arange(m) * n) // m
        counts = np.bincount(idx, minlength=n).astype(np.float64)
        counts[counts <= 0] = 1.0
        b = a[idx] / counts[idx]
    else:
        b = np.zeros(m)
        idx = (np.arange(n) * m) // n
        np.add.at(b, idx, a)
        b = np.maximum(b, 0.0)
    return _project_simplex(b, 1.0)


def _evaluate_exact(a):
    """Call the real evaluator."""
    try:
        lst = [float(x) for x in a]
        return float(evaluate_sequence(lst))
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = max(0.0, float(budget_s))
    except Exception:
        budget = 1300.0

    headroom = max(5.0, min(20.0, 0.015 * budget))
    deadline = t0 + budget - headroom

    rng = np.random.default_rng(int(seed) if np.isfinite(float(seed)) else 42)

    # Get parent
    parent = _get_parent()
    if parent is not None:
        best_a = _project_simplex(parent, 1.0)
    else:
        best_a = np.ones(192) / 192.0

    best_score = _evaluate_exact(best_a)
    if not np.isfinite(best_score):
        best_a = np.ones(128) / 128.0
        best_score = _evaluate_exact(best_a)

    # Try multiple lengths
    lengths_to_try = []
    pL = int(best_a.size)
    for L in (512, 384, 256, 320, 448):
        if L != pL:
            lengths_to_try.append(L)
    if 32 <= pL <= 1024 and pL not in lengths_to_try:
        lengths_to_try.append(pL)
    # Put pL first
    if pL in lengths_to_try:
        lengths_to_try.remove(pL)
    lengths_to_try.insert(0, pL)

    # Time allocation
    n_lengths = len(lengths_to_try)
    base_time = (deadline - time.monotonic() - 30.0) / n_lengths

    for Li, L in enumerate(lengths_to_try):
        if time.monotonic() >= deadline - 25.0:
            break
        # Allocate time: more for parent length, less for others
        if Li == 0:
            time_alloc = min(base_time * 1.5, deadline - time.monotonic() - 25.0)
        else:
            time_alloc = min(base_time * 0.6, deadline - time.monotonic() - 25.0)

        if time_alloc < 15.0:
            continue

        # Initialize from current best
        init = _resample_shape(best_a, L)

        # Phase A: main optimization
        sub_deadline = time.monotonic() + time_alloc * 0.7
        try:
            res_a, res_score = _optimize_sequence(init, sub_deadline)
            if np.isfinite(res_score) and res_score < best_score + 1e-6:
                sc = _evaluate_exact(res_a)
                if np.isfinite(sc) and sc < best_score:
                    best_a = _project_simplex(res_a, 1.0)
                    best_score = float(sc)
        except Exception:
            pass

        # Phase B: secondary exploration (different init)
        rem = deadline - time.monotonic() - 25.0
        if rem > 20.0:
            sub_deadline2 = time.monotonic() + min(time_alloc * 0.3, rem - 20.0)
            # Try a wavelet-like or asymmetric init
            x = np.linspace(0, 1, L, endpoint=False)
            alpha = float(rng.uniform(0.6, 2.0))
            init2 = _project_simplex((x**alpha + 0.3*(1-x)**alpha + 0.1), 1.0)
            try:
                res2_a, res2_score = _optimize_sequence(init2, sub_deadline2)
                if np.isfinite(res2_score) and res2_score < best_score + 1e-6:
                    sc = _evaluate_exact(res2_a)
                    if np.isfinite(sc) and sc < best_score:
                        best_a = _project_simplex(res2_a, 1.0)
                        best_score = float(sc)
            except Exception:
                pass

    # Final polish: one more careful optimization on the best
    if time.monotonic() < deadline - 20.0:
        polish_time = min(deadline - time.monotonic() - 18.0, 120.0)
        if polish_time > 20.0:
            try:
                resP, resP_score = _optimize_sequence(best_a.copy(), time.monotonic() + polish_time, max_iter=100)
                if np.isfinite(resP_score) and resP_score < best_score + 1e-6:
                    sc = _evaluate_exact(resP)
                    if np.isfinite(sc) and sc < best_score:
                        best_a = _project_simplex(resP, 1.0)
                        best_score = float(sc)
            except Exception:
                pass

    # Final validation
    try:
        best_a = np.asarray(best_a, dtype=np.float64).ravel()
        best_a = np.maximum(best_a, 0.0)
        best_a = _project_simplex(best_a, 1.0)
        final_sc = _evaluate_exact(best_a)
        if np.isfinite(final_sc):
            best_score = final_sc
        else:
            raise ValueError
    except Exception:
        best_a = np.ones(64) / 64.0
        best_score = _evaluate_exact(best_a)
        if not np.isfinite(best_score):
            return [1.0]

    out = np.clip(best_a, 0.0, 1000.0)
    out = _project_simplex(out, 1.0)
    if out.size == 0:
        return [1.0]
    return [float(x) for x in out]
