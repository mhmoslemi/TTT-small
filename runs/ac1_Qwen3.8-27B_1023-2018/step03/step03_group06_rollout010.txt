def _line_search(f, g, cf, cg, cfs):
    """Find t in [0,1] minimizing max_k ((1-t)f+tg)^2_k."""
    # h_k(t) = (1-t)^2 cf_k + 2t(1-t) cfs_k + t^2 cg_k
    # = cf_k + (2*cfs_k - 2*cf_k)*t + (cg_k - 2*cfs_k + cf_k)*t^2
    
    base = cf.astype(np.float64)
    lin = 2.0 * (cfs - cf)
    quad = cf - 2.0 * cfs + cg
    
    # Evaluate on a fine grid
    ts = np.linspace(0.0, 1.0, 201)
    vals = np.max(base + lin * ts[:, None] + quad * (ts[:, None]**2), axis=1)
    
    i = np.argmin(vals)
    lo = ts[max(0, i-2)]
    hi = ts[min(len(ts)-1, i+2)]
    
    ts2 = np.linspace(lo, hi, 501)
    vals2 = np.max(base + lin * ts2[:, None] + quad * (ts2[:, None]**2), axis=1)
    j = np.argmin(vals2)
    
    return float(ts2[j]), float(vals2[j])

def _grad_smooth_max(w, a):
    """Compute gradient of smooth max c_k with respect to a.
    w: weights of length 2n-1 (sum to 1)
    a: sequence of length n
    Returns: gradient vector of length n
    """
    n = len(a)
    m = 2*n - 1
    g = np.zeros(n)
    for i in range(n):

        lo = i
        hi = min(m, i+n)
        g[i] = np.dot(w[lo:hi], a[:hi-lo])
    return 2.0 * g

def _lp_dir(f):

    n = len(f)

def _subgrad(a, t_stop, rng, max_iters=5000):
    n = len(a)
    best = a.copy()
    bs = _score(a)
    
    for it in range(max_iters):
        if time.monotonic() > t_stop: break
        
        c = _conv(a)
        cm = float(c.max())
        if cm < 1e-15: break
        
        # Top k positions
        top_k = np.argsort(c)[-3:][::-1]  # top 3 in descending order
        w = np.ones(len(top_k)) / len(top_k)
        
        # Subgradient
        s = np.zeros(n)
        for kk in top_k:
            for i in range(n):
                if 0 <= kk - i < n:
                    s[i] += 2.0 * a[kk - i]
        s /= len(top_k)
        
        # Learning rate
        smax = float(np.max(np.abs(s)))
        if smax < 1e-15: break
        lr = 0.1 * float(a.mean()) / smax
        
        # Try a few learning rates
        for lr_mult in [1.0, 0.5, 0.25, 0.1, 2.0]:
            a_new = a - lr * lr_mult * s
            a_new = np.maximum(a_new, 0.0)
            ns = float(a_new.sum())
            if ns < 1e-12: continue
            a_new /= ns
            sc = _score(a_new)
            if math.isfinite(sc) and sc < bs:
                best = a_new.copy()
                bs = sc
                a = a_new
                break
        else:
            # No improvement from subgradient; try noise
            a = a + 0.01*float(a.mean())*rng.standard_normal(n)
            a = np.maximum(a, 0.)
            a /= a.sum()
            sc = _score(a)
            if math.isfinite(sc) and sc < bs:
                best = a.copy()
                bs = sc
    
    return best, bs

import time

from scipy.fft import rfft, irfft, next_fast_len

class _TMO(Exception):
    pass

_FC = {}
def _FL(m):
    m = int(m)
    if m <= 0:
        return 1
    v = _FC.get(m)
    if v is None:
        v = int(next_fast_len(m))
        _FC[m] = v
    return v

def _conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    L = _FL(2 * n - 1)
    fa = rfft(a, L)
    c = irfft(fa * fa, L)[:2 * n - 1]
    return np.maximum(c, 0.0)

def _conv2(a, b):
    la, lb = len(a), len(b)
    if la == 0 or lb == 0:
        return np.array([0.0])
    L = _FL(la + lb - 1)
    c = irfft(rfft(a, L) * rfft(b, L), L)[:la + lb - 1]
    return np.maximum(c, 0.0)

        return math.inf
    n = len(a)
    cm = float(_conv(a).max())
    if not math.isfinite(cm):
        return math.inf
    return 2.0 * n * cm / (s * s)

    if not math.isfinite(s) or s < 1e-15:
        return np.ones(max(1, len(a)), np.float64) / max(1, len(a))
    return a / s

def _grad_conv(w, a):
    """g_a[i] = 2 * sum_j w[i+j] * a[j], where w has length 2n-1 and a has length n."""
    n = len(a)
    m = 2 * n - 1
    g = np.zeros(n)
    for i in range(n):
        hi = min(m, i + n)
        g[i] = np.dot(w[i:hi], a[:hi - i])
    return 2.0 * g
