def _proj_simplex(v, s=1.0):

        return np.full(n, s / n, dtype=np.float64)
    theta = (css[rho[-1]] - s) / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    ws = float(w.sum())
    if ws <= 1e-300:
        return np.full(n, s / n, dtype=np.float64)
    return w * (s / ws)


def _conv(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return 0.0, np.array([0.0])
    nfft = next_fast_len(2 * n - 1)
    f = rfft(a, nfft)
    c = irfft(f * f, nfft)[:2 * n - 1]
    c = np.asarray(c, dtype=np.float64)
    c = np.where(np.isfinite(c), c, 0.0)
    np.maximum(c, 0.0, out=c)
    return float(c.max()), c


def _build_A(a):
    n = a.size
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_direction(a, M, deadline):
    n = a.size
    if n < 4 or n > 1200:
        return None
    rem = deadline - time.monotonic()
    if rem < 3.0:
        return None
    A = _build_A(a)
    m = 2 * n - 1
    c_obj = -np.ones(n, dtype=np.float64)
    b_ub = np.full(m, M, dtype=np.float64)
    tl = max(1.0, min(25.0, rem * 0.4))
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0.0, None),
                      method='highs', options={'time_limit': tl, 'presolve': True})
        if res is not None and getattr(res, 'success', False):
            x = np.asarray(res.x, dtype=np.float64).ravel()
            if x.size == n and np.all(np.isfinite(x)):
                x = np.maximum(x, 0.0)
                s = float(x.sum())
                if s > 1e-12:
                    return x / s
    except Exception:
        pass
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0.0, None),
                      method='highs', options={'presolve': True})
        if res is not None and getattr(res, 'success', False):
            x = np.asarray(res.x, dtype=np.float64).ravel()
            if x.size == n and np.all(np.isfinite(x)):
                x = np.maximum(x, 0.0)
                s = float(x.sum())
                if s > 1e-12:
                    return x / s
    except Exception:
        pass
    return None


def _soft_grad(a, c, M, T):
    n = a.size
    z = (c - M) / T
    z = np.clip(z, -80.0, 0.0)
    w = np.exp(z)
    sw = float(w.sum())
    if sw < 1e-300:
        w = np.zeros(c.size, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
        sw = 1.0
    w /= sw
    A = _build_A(a)
    grad = (2.0 / T) * (A.T @ w)
    return grad, w


def _optimize_sequence(a0, deadline, rng, lp_interval=4):
    a = _norm(a0)
    n = a.size
    M, c = _conv(a)
    best_a = a.copy()
    best_M = M
    T = max(M * 0.02, 1e-6)
    stuck = 0
    it = 0

    while time.monotonic() < deadline - 5.0:
        it += 1
        M, c = _conv(a)

        if M < best_M * (1.0 - 1e-11):
            best_M = M
            best_a = a.copy()
            stuck = 0
            T = max(T * 0.97, best_M * 0.005)
        else:
            stuck += 1
            T = min(T * 1.02, best_M * 0.08)

        # Gradient step
        grad, w = _soft_grad(a, c, M, T)
        gmax = float(np.max(np.abs(grad)))
        if gmax < 1e-14:
            pass
        else:
            lr_base = T / gmax
            improved = False
            for frac in (0.4, 0.6, 0.8, 1.0, 1.4, 2.0):
                lr = frac * lr_base
                a_try = _proj_simplex(a - lr * grad)
                M_try, _ = _conv(a_try)
                if M_try < M * (1.0 - 1e-10):
                    a = a_try
                    improved = True
                    break
                if time.monotonic() > deadline - 10:
                    break
            if improved:
                continue

        # LP direction
        if (it % lp_interval == 0 or stuck > 8) and time.monotonic() < deadline - 15.0:
            b = _lp_direction(a, M, deadline)
            if b is not None and np.any(np.abs(b - a) > 1e-8):
                lp_improved = False
                for alpha in (0.15, 0.25, 0.4, 0.6, 0.8, 1.0):
                    if time.monotonic() > deadline - 10:
                        break
                    a_try = _proj_simplex(alpha * b + (1.0 - alpha) * a)
                    M_try, _ = _conv(a_try)
                    if M_try < M * (1.0 - 1e-10):
                        a = a_try
                        lp_improved = True
                        break
                if lp_improved:
                    continue

        # Multiplicative update from LP direction
        if (it % (2 * lp_interval) == 0) and time.monotonic() < deadline - 15.0:
            b = _lp_direction(a, M, deadline)
            if b is not None:
                for beta in (0.1, 0.2, 0.35):
                    if time.monotonic() > deadline - 10:
                        break
                    a_try = _norm(a * (1.0 + beta) + b * beta)
                    M_try, _ = _conv(a_try)
                    if M_try < M * (1.0 - 1e-10):
                        a = a_try
                        break

        # Stuck: perturbation
        if stuck > 25:
            r = rng.random()
            if r < 0.4:
                a = _proj_simplex(a + rng.normal(0.0, 0.015, n))
            elif r < 0.7:
                idx = rng.choice(n, size=max(3, n // 20), replace=False)
                delta = rng.normal(0.0, 0.03, size=len(idx))
                a[idx] += delta
                a = _proj_simplex(a)
            else:
                x = np.linspace(0, 1, n, endpoint=False)
                alpha = float(rng.uniform(0.3, 2.0))
                shape = (x ** alpha + 0.3) * (0.7 + 0.3 * np.cos(2 * np.pi * x * float(rng.integers(1, 5))))
                blend = float(rng.uniform(0.3, 0.7))
                a = _proj_simplex(blend * _norm(a) + (1 - blend) * _norm(shape))
            stuck = 0
            T = M * 0.04

    M, c = _conv(a)
    if M < best_M:
        best_a = a.copy()
        best_M = M
    return best_a, best_M


def _refine_final(a0, deadline, rng):
    a = _norm(a0)
    n = a.size
    M, c = _conv(a)
    best_a = a.copy()
    best_M = M
    T = max(M * 0.005, 1e-7)

    for _ in range(200):
        if time.monotonic() > deadline:
            break
        M, c = _conv(a)
        grad, w = _soft_grad(a, c, M, T)
        gmax = float(np.max(np.abs(grad)))
        if gmax < 1e-14:
            break
        lr = T / gmax
        improved = False
        for frac in (0.3, 0.5, 0.7, 1.0, 1.3):
            a_try = _proj_simplex(a - frac * lr * grad)
            M_try, _ = _conv(a_try)
            if M_try < M * (1.0 - 1e-12):
                a = a_try
                if M_try < best_M:
                    best_M = M_try
                    best_a = a.copy()
                improved = True
                break
            if time.monotonic() > deadline:
                break
        if not improved:
            T = T * 0.9
            if T < M * 0.001:
                break

    M, c = _conv(a)
    if M < best_M:
        best_a = a.copy()
        best_M = M
    return best_a, best_M


def _exact_score(a):
    try:
        lst = [float(x) for x in a]
        sc = evaluate_sequence(lst)
        return float(sc) if np.isfinite(sc) else float('inf')
    except Exception:
        return float('inf')


def _resample(a, m):
    a = _norm(a)
    n = a.size
    m = int(m)
    if m < 1:
        m = 1
    if n == m:
        return a.copy()
    if m > n:
        idx = (np.arange(m, dtype=np.int64) * n) // m
        counts = np.bincount(idx, minlength=n).astype(np.float64)
        counts[counts <= 0] = 1.0
        b = a[idx] / counts[idx]
    else:
        b = np.zeros(m, dtype=np.float64)
        idx = (np.arange(n, dtype=np.int64) * m) // n
        np.add.at(b, idx, a)
    return _norm(b)


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = max(10.0, float(budget_s))
    except Exception:
        budget = 1300.0
    headroom = max(8.0, min(25.0, 0.02 * budget))
    deadline = t0 + budget - headroom

    rng = np.random.default_rng(int(seed) if np.isfinite(float(seed)) else 42)

    parent = _get_parent()
    best_a = _norm(parent)
    best_score = _exact_score(best_a)
    if not np.isfinite(best_score):
        best_a = np.ones(256, dtype=np.float64) / 256.0
        best_score = _exact_score(best_a)
        if not np.isfinite(best_score):
            return [1.0]

    # Determine which lengths to try
    pL = int(best_a.size)
    lengths = []
    for L in (384, 512, 640, 768, 1024):
        if L not in lengths:
            lengths.append(L)
    if pL not in lengths and 64 <= pL <= 1200:
        lengths.insert(1, pL)

    # Weight lengths by preference
    len_weights = {}
    for L in lengths:
        if L == pL:
            w = 2.0
        elif L == 512:
            w = 1.8
        elif L == 384:
            w = 1.5
        elif L == 640:
            w = 1.2
        elif L == 768:
            w = 1.0
        elif L == 1024:
            w = 0.8
        else:
            w = 0.6
        len_weights[L] = w

    total_w = sum(len_weights.values())
    phase_start = time.monotonic()
    phase_end = deadline - 30.0  # reserve 30s for final polish
    phase_budget = max(10.0, phase_end - phase_start)

    for L in lengths:
        if time.monotonic() >= deadline - 35.0:
            break
        frac = len_weights[L] / total_w
        timeL = frac * phase_budget
        timeL = min(timeL, deadline - time.monotonic() - 35.0)
        if timeL < 15.0:
            continue

        init_a = _resample(best_a, L)
        loc_deadline = time.monotonic() + timeL - 5.0
        try:
            loc_a, loc_M = _optimize_sequence(init_a, loc_deadline, rng, lp_interval=3)
            sc = 2.0 * int(loc_a.size) * loc_M
            if np.isfinite(sc) and sc < best_score + 0.001:
                exact = _exact_score(loc_a)
                if np.isfinite(exact) and exact < best_score:
                    best_score = float(exact)
                    best_a = _norm(loc_a)
        except Exception:
            pass

    # Final polish on the best candidate
    if time.monotonic() < deadline - 5.0:
        pol_deadline = deadline - 3.0
        pol_time = pol_deadline - time.monotonic()
        if pol_time > 10.0:
            try:
                pol_a, pol_M = _refine_final(best_a, pol_deadline, rng)
                sc = 2.0 * int(pol_a.size) * pol_M
                if np.isfinite(sc) and sc < best_score + 0.0005:
                    exact = _exact_score(pol_a)
                    if np.isfinite(exact) and exact < best_score:
                        best_score = float(exact)
                        best_a = _norm(pol_a)
            except Exception:
                pass

    # Additional LP polish if time remains
    if time.monotonic() < deadline - 3.0:
        tri_end = deadline - 2.0
        try:
            tri_a, tri_M = _optimize_sequence(best_a, tri_end, rng, lp_interval=2)
            sc = 2.0 * int(tri_a.size) * tri_M
            if np.isfinite(sc) and sc < best_score + 0.0003:
                exact = _exact_score(tri_a)
                if np.isfinite(exact) and exact < best_score:
                    best_score = float(exact)
                    best_a = _norm(tri_a)
        except Exception:
            pass

    # Final validation
    best_a = np.asarray(best_a, dtype=np.float64).ravel()
    best_a = np.where(np.isfinite(best_a), best_a, 0.0)
    best_a = np.maximum(best_a, 0.0)
    s = float(best_a.sum())
    if s < 1e-300:
        return [1.0]
    best_a = best_a / s
    best_a = np.clip(best_a, 0.0, 1000.0)

    final_sc = _exact_score(best_a)
    if not np.isfinite(final_sc):
        # Fallback
        fb = _norm(parent)
        fb_sc = _exact_score(fb)
        if np.isfinite(fb_sc):
            return [float(x) for x in fb]
        arr = np.ones(128, dtype=np.float64) / 128.0
        return [float(x) for x in arr]

    return [float(x) for x in best_a]
