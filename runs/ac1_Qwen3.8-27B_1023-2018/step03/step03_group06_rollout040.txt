_fft_cache = {}

def _fft_len(m):
    m = int(m)
    if m <= 0:
        return 1
    v = _fft_cache.get(m)
    if v is None:
        v = int(next_fast_len(m))
        _fft_cache[m] = v
    return v

def _conv(a, b=None):
    if b is None:
        b = a
    n1, n2 = len(a), len(b)
    m = n1 + n2 - 1

    fb = fa if b is a else rfft(b, L)
    c = irfft(fa * fb, L)[:m]
    return c
