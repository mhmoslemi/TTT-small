    M = float(np.max(np.convolve(f, f, mode="full")))
    if M < 1e-15:
        return None, M
    # Build A matrix: A[k, j] = f[k-j] if 0<=k-j<=n-1 else 0
    # (f*g)[k] = sum_j f[k-j] * g[j]
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                      method="highs", options={"presolve": True, "time_limit": time_limit})
    except Exception:
        return None, M
    if res.status != 0 or res.x is None:
        return None, M
    g = np.asarray(res.x, dtype=np.float64)
    gs = float(g.sum())
    if gs < 1e-15:
        return None, M
    return g / gs, M

def _subgradient_step(f, n, step_size, stop_time):
    """One projected subgradient step for the min-max problem."""
    conv = np.convolve(f, f, mode="full")
    M = conv.max()
    if M < 1e-15:
        return f, M
    # Active set: lags within 1% of max
    tol = 0.01 * M
    active = conv >= (M - tol)
    # Subgradient: g[i] = 2 * sum_{k in active} f[k-i] * [0<=k-i<=n-1]
    g = np.zeros(n)
    for i in range(n):
        k_lo = i
        k_hi = min(i + n - 1, 2 * n - 2)
        seg = active[k_lo:k_hi + 1]
        g[i] = 2.0 * np.dot(seg, f[0:k_hi - k_lo + 1])

    # Project: subtract mean (for sum constraint), clip
    S = f.sum()
    g -= np.mean(g)  # project onto sum-preserving hyperplane
    f_new = f - step_size * g / S
    f_new = np.clip(f_new, 0, None)
    fs = f_new.sum()
    if fs < 0.01:
        return f, M
    f_new = f_new / fs * S  # rescale to maintain sum
    return f_new, M
