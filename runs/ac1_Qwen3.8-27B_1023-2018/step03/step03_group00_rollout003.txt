def _convolve_full(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    L = next_fast_len(2 * n - 1)
    A = rfft(a, L)
    c = irfft(A * A, L)[: 2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _weighted_corr(a, q):
    """r_i = sum_k q_k a_{k-i}.  q has length 2n-1."""
    a = np.asarray(a, dtype=np.float64).ravel()
    q = np.asarray(q, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    L = next_fast_len(3 * n - 2)
    A = rfft(a, L)
    Q = rfft(q[::-1], L)
    h = irfft(A * Q, L)
    K = 2 * n - 2
    seg = h[n - 1: K + 1]
    return seg[::-1].copy()


def _score_arr(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return math.inf
    a = np.where(np.isfinite(a), a, 0.0)

    a = a / s
    b = _convolve_full(a)
    m = float(b.max())

    s = float(a.sum())

            A[j:j + n, j] = f_safe

        bounds = [(0.0, None)] * n + [(0.0, None)]

            c,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,

                "presolve": True,

            vals = np.concatenate((old * np.array(mults, dtype=np.float64), base_scales))
            vals = vals[vals >= -1e-15]
            vals = np.clip(vals, 0.0, 10.0)
            if vals.size == 0:
                continue

            diff = vals - old
            newS = 1.0 + diff
            valid = newS > 1e-12
            if not valid.any():
                continue
            vals = vals[valid]
            diff = diff[valid]
            newS = newS[valid]

            start = i
            end = i + n
            outside_max = 0.0
            if start > 0:
                outside_max = max(outside_max, float(b[:start].max()))
            if end < L:
                outside_max = max(outside_max, float(b[end:].max()))

            bseg = b[start:end]
            inside = bseg[None, :] + (2.0 * diff)[:, None] * a[None, :]
            qi = 2 * i
            if 0 <= qi < L:
                idx = qi - start
                if 0 <= idx < n:
                    inside[:, idx] += vals * vals - old * old

            np.maximum(inside, 0.0, out=inside)
            max_inside = inside.max(axis=1)
            total = np.maximum(outside_max, max_inside)
            sc = (2.0 * n) * total / (newS * newS)
            sc[~np.isfinite(sc)] = math.inf

            j = int(np.argmin(sc))
            v = float(vals[j])
            if sc[j] < cur - 1e-13 and abs(v - old) > 1e-15:
                d = v - old
                b[start:end] = b[start:end] + 2.0 * d * a
                if 0 <= qi < L:
                    b[qi] += v * v - old * old
                a[i] = v

                Snew = 1.0 + d
                if Snew > 1e-12:
                    inv = 1.0 / Snew
                    a *= inv
                    b *= inv * inv
                np.maximum(b, 0.0, out=b)

                cur = 2.0 * n * float(b.max())
                if math.isfinite(cur) and cur < best_sc - 1e-13:
                    best_sc = cur
                    best_a = a.copy()
                improved = True

        if improved:
            no_imp = 0
        else:
            no_imp += 1
            if no_imp >= max_no_imp:
                break

    return best_a, best_sc


def _flex_improve(a0, stop, seed=0):
    a = _normalize(a0)
    best_a = a.copy()
    best_sc = _score_arr(a)
    a_sc = best_sc
    stop = float(stop)
    if stop <= time.monotonic() + 2.0:
        return best_a, best_sc

    try:
        if stop - time.monotonic() > 25.0:
            alloc = min(0.45 * (stop - time.monotonic()), 180.0)
            if alloc > 12.0:
                a_s, sc_s = _smooth_refine(a, time.monotonic() + alloc, seed)
                if math.isfinite(sc_s):
                    if sc_s < a_sc + 1e-3:
                        a, a_sc = a_s, sc_s
                    if sc_s < best_sc:
                        best_sc = sc_s
                        best_a = a_s.copy()

        if len(a) <= 650 and stop - time.monotonic() > 30.0:
            alloc = min(0.25 * (stop - time.monotonic()), 85.0)
            if alloc > 12.0:
                a_l, sc_l = _lp_refine(a, time.monotonic() + alloc, seed + 7)
                if math.isfinite(sc_l):
                    if sc_l < a_sc + 1e-3:
                        a, a_sc = a_l, sc_l
                    if sc_l < best_sc:
                        best_sc = sc_l
                        best_a = a_l.copy()

        if stop - time.monotonic() > 8.0:
            a_c, sc_c = _coordinate_polish(a, stop - max(2.0, 0.02 * (stop - time.monotonic())), seed + 13)
            if math.isfinite(sc_c) and sc_c < best_sc:
                best_sc = sc_c
                best_a = a_c.copy()
    except Exception:
        pass

    sc_final = _score_arr(best_a)
    if not math.isfinite(sc_final):
        best_a = a.copy()
        sc_final = _score_arr(best_a)
    if not math.isfinite(sc_final):
        best_a = a.copy()
        sc_final = _score_arr(best_a)
    return best_a, best_sc if math.isfinite(best_sc) else sc_final


def _pick_lengths(n0):
    n0 = int(max(64, n0))
    raw = [
        n0 - 4, n0 - 2, n0, n0 + 1, n0 + 2, n0 + 4, n0 + 8,
        int(round(0.75 * n0)), int(round(0.85 * n0)),
        int(round(1.10 * n0)), int(round(1.20 * n0)),
        int(round(1.35 * n0)), int(round(1.50 * n0)),
        int(round(1.75 * n0)), int(round(2.00 * n0)),
        128, 192, 256, 320, 384, 416, 448, 480, 512,
        576, 640, 768, 896, 1024,
    ]
    out = []
    seen = set()
    for L in raw:
        L = int(L)
        if 64 <= L <= 1100 and L not in seen:
            seen.add(L)
            out.append(L)

    def key(L):
        return (abs(L - n0) / float(n0), -L)

    out.sort(key=key)
    return out[:13]


def _analytic_shapes(L, rng):
    i = np.arange(L, dtype=np.float64)
    x = (i + 1.0) / float(L)
    base = np.power(x, -0.5)
    eps = 0.5 / max(1.0, float(L))
    out = []

    out.append(base.copy())
    for p in (0.25, 0.5, 1.0):
        out.append(base * np.power(1.0 - x + eps, p))
    for p in (-0.10, -0.20):
        out.append(base * np.power(1.0 - x + eps, p))
    out.append(base / (1.0 + i / L))
    out.append(base / (1.0 + 2.0 * i / L))
    out.append(np.power(x, -0.45))
    out.append(np.power(x, -0.55))
    out.append(np.power(i + 2.0, -0.5))
    out.append(np.power(i + 6.0, -0.5))
    out.append(base * np.exp(-i / L))
    out.append(base * np.exp(-2.0 * i / L))
    out.append(base * np.maximum(0.5, 1.0 + 0.3 * np.sin(2.0 * np.pi * i / L)))

    return out


def _generate_candidates(parent, stop, rng):
    cands = []
    lengths = _pick_lengths(parent.size)
    stop = float(stop)

    for L in lengths:
        if time.monotonic() > stop:
            break
        per = []

        if parent.size > 0:
            if L == parent.size:
                arr = parent.copy()
            else:
                arr = np.interp(
                    np.linspace(0.0, 1.0, L),
                    np.linspace(0.0, 1.0, parent.size),
                    parent,
                )
            arr = _normalize(arr)
            sc = _score_arr(arr)
            if math.isfinite(sc):
                per.append((sc, arr))

        for arr in _analytic_shapes(L, rng):
            try:
                arr = np.asarray(arr, dtype=np.float64).ravel()
                arr = np.where(np.isfinite(arr), arr, 0.0)
                arr = np.clip(arr, 0.0, None)
                s = float(arr.sum())
                if s <= 0.0:
                    continue
                arr = arr / s
                sc = _score_arr(arr)
                if math.isfinite(sc):
                    per.append((sc, arr))
            except Exception:
                pass

        per.sort(key=lambda z: z[0])
        for sc, arr in per[:2]:
            cands.append((sc, arr))

        if len(cands) > 140:
            break

    cands.sort(key=lambda z: z[0])
    uniq = []
    for sc, arr in cands:
        dup = False
        for _, u in uniq:
            if u.size == arr.size:
                d = float(np.dot(u, arr))
                if d > 1.0 - 1e-5:
                    dup = True
                    break
        if not dup:
            uniq.append((sc, arr))
        if len(uniq) >= 16:
            break
    return uniq


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    headroom = min(30.0, max(2.0, 0.025 * budget))
    deadline = t0 + max(5.0, budget - headroom)

    parent = _get_parent()
    best_a = parent.copy()
    best_sc = _score_arr(best_a)
    fallback_a = parent.copy()
    fallback_sc = best_sc

    if not math.isfinite(best_sc):
        best_a = np.ones(max(64, parent.size), dtype=np.float64)
        best_a = best_a / best_a.size
        best_sc = _score_arr(best_a)
        fallback_a = best_a.copy()
        fallback_sc = best_sc

    rng = np.random.default_rng(seed)

    def record(a):
        nonlocal best_a, best_sc
        try:
            a = _normalize(a)
            sc = _score_arr(a)
            if math.isfinite(sc) and sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a.copy()
        except Exception:
            pass

    # If budget is tiny, do a very short polish and return.
    if deadline < t0 + 20.0:
        try:
            a2, _ = _coordinate_polish(best_a, deadline - 2.0, seed)
            record(a2)
        except Exception:
            pass

    # Generate diverse initializations.
    gen_stop = min(deadline - 40.0, t0 + min(80.0, 0.08 * budget))
    cand_list = [(best_sc, best_a.copy())]
    try:
        gen = _generate_candidates(parent, gen_stop, rng)
        cand_list.extend(gen)
    except Exception:
        pass

    cand_list.sort(key=lambda z: z[0])
    uniq_cands = []
    for sc, arr in cand_list:
        dup = False
        for _, u in uniq_cands:
            if u.size == arr.size and float(np.dot(u, arr)) > 1.0 - 1e-5:
                dup = True
                break
        if not dup:
            uniq_cands.append((sc, arr))
        if len(uniq_cands) >= 14:
            break

    flex_stop = min(deadline - 55.0, gen_stop + min(0.70 * budget, max(0.0, deadline - gen_stop - 55.0)))
    total_flex = max(5.0, flex_stop - time.monotonic())
    weights = [0.44, 0.24, 0.14, 0.08, 0.05, 0.03, 0.03, 0.03, 0.03, 0.03]

    for idx, (sc, arr) in enumerate(uniq_cands[:10]):
        if time.monotonic() > flex_stop - 12.0:
            break
        w = weights[idx] if idx < len(weights) else 0.03
        slice_s = max(12.0, w * total_flex)
        stop_i = min(flex_stop - 8.0, time.monotonic() + slice_s)
        if stop_i - time.monotonic() < 8.0:
            break
        try:
            a2, sc2 = _flex_improve(arr, stop_i, seed + 101 + 17 * idx)
            record(a2)
        except Exception:
            try:
                a2, _ = _coordinate_polish(arr, stop_i - 3.0, seed + 101 + 17 * idx)
                record(a2)
            except Exception:
                pass

    # Extra targeted polish on the current best.
    extra_stop = deadline - 18.0
    if time.monotonic() < extra_stop - 25.0:
        slice1 = min(90.0, extra_stop - time.monotonic() - 12.0)
        if slice1 > 20.0:
            try:
                a2, _ = _coordinate_polish(best_a, time.monotonic() + slice1, seed + 31337)
                record(a2)
            except Exception:
                pass

    if len(best_a) <= 650 and time.monotonic() < extra_stop - 25.0:
        slice2 = min(70.0, extra_stop - time.monotonic() - 12.0)
        if slice2 > 15.0:
            try:
                a2, _ = _lp_refine(best_a, time.monotonic() + slice2, seed + 55555)
                record(a2)
            except Exception:
                pass

    # Small restarts around the best point.
    if time.monotonic() < deadline - 35.0:
        for k, sig in enumerate((0.05, 0.14, 0.30, 0.60)):
            if time.monotonic() > deadline - 25.0:
                break
            slice3 = min(45.0, deadline - time.monotonic() - 18.0)
            if slice3 < 12.0:
                break
            try:
                with np.errstate(over="ignore", under="ignore", invalid="ignore"):
                    ap = best_a * np.exp(rng.normal(0.0, sig, best_a.size))
                ap = _normalize(ap)
                a2, _ = _coordinate_polish(ap, time.monotonic() + slice3, seed + 700000 + 11 * (k + 1))
                record(a2)
            except Exception:
                pass

    # Final validation against the authoritative evaluator.
    eval_fn = globals().get("evaluate_sequence", None)

    def exact_score(arr):
        try:
            arr = _normalize(arr)
            lst = [float(x) for x in arr]
            if eval_fn is not None:
                v = float(eval_fn(lst))
                if math.isfinite(v):
                    return v
            v = _score_arr(arr)
            if math.isfinite(v):
                return v
        except Exception:
            pass
        return math.inf

    final_candidates = [best_a, fallback_a, parent]
    best_final = None
    best_final_sc = math.inf
    for arr in final_candidates:
        if arr is None:
            continue
        sc = exact_score(arr)
        if math.isfinite(sc) and sc < best_final_sc:
            best_final_sc = sc
            best_final = arr

    if best_final is None:
        try:
            best_final = _normalize(best_a)
        except Exception:
            best_final = np.array([1.0], dtype=np.float64)

    final_list = [float(x) for x in np.asarray(best_final, dtype=np.float64).ravel()]
    if not final_list:
        final_list = [1.0]

    # One last safety check; if exact evaluation somehow fails, return a minimal valid list.
    if eval_fn is not None:
        try:
            v = float(eval_fn(final_list))
            if not math.isfinite(v):
                fallback_list = [float(x) for x in _normalize(fallback_a).ravel()]
                v2 = float(eval_fn(fallback_list))
                if math.isfinite(v2):
                    return fallback_list
                return [1.0]
        except Exception:
            return [1.0]

    return final_list
