def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    c = np.convolve(a, a)
    return 2.0 * n * float(c.max()) / (s * s)


def _build_A(a):
    n = a.size
    m = 2 * n - 1
    k = np.arange(m)[:, None].astype(np.intp)
    j = np.arange(n)[None, :].astype(np.intp)
    idx = k - j
    mask = (idx >= 0) & (idx < n)
    A = np.where(mask, a[np.clip(idx, 0, n - 1)], 0.0)
    return A


def _solve_lp(A, b_ub, n, timeout=8.0):
    c_obj = -np.ones(n)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, 50.0)] * n,
                      method='highs',
                      options={'time_limit': timeout, 'presolve': True})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    if not np.isfinite(g).all() or g.sum() < 1e-12:
        return None
    return g


def _line_search(a, g, conv_aa, timeout_guard):
    """Fast 1D line search using precomputed convolutions."""
    conv_ag = np.convolve(a, g)
    conv_gg = np.convolve(g, g)

    def f(t):
        return ((1 - t) ** 2 * conv_aa + 2 * t * (1 - t) * conv_ag + t * t * conv_gg).max()

    # Coarse grid
    ts = np.linspace(0.0, 1.0, 33)
    vals = np.array([f(t) for t in ts])
    bi = int(np.argmin(vals))
    best_t, best_v = float(ts[bi]), float(vals[bi])

    if 0 < bi < len(ts) - 1:
        lo, hi = float(ts[max(0, bi - 1)]), float(ts[min(len(ts) - 1, bi + 1)])
    else:
        return best_t, best_v

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(50):
        if (hi - lo) < 1e-10 or time.monotonic() > timeout_guard:
            break
        t1 = hi - gr * (hi - lo)
        t2 = lo + gr * (hi - lo)
        f1, f2 = f(t1), f(t2)
        if f1 < f2:
            hi = t2
        else:
            lo = t1
    t_mid = (lo + hi) / 2.0
    v_mid = f(t_mid)
    if v_mid < best_v:
        best_t, best_v = t_mid, v_mid
    return best_t, best_v


def _subgrad_step(a, conv_aa, active, a0, g0, lr):
    """One projected subgradient step reducing max(a*a) at active lags."""
    n = len(a)
    grad = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * a[k - idx]
    grad /= len(active)
    grad -= grad.mean()
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return a, 0.0
    grad /= gn
    cand = a - lr * grad
    if cand.min() < 0:
        cand = np.maximum(cand, 0.0)
    cs = cand.sum()
    if cs < 1e-12:
        return a, 0.0
    cand /= cs
    M_new = float(np.convolve(cand, cand).max())
    return cand, M_new
