                        candidates = [orig * 0.3, orig * 0.5, orig * 0.7, orig * 1.0,
                                      orig * 1.2, orig * 1.5, 0.0]
                    else:
                        candidates = [orig * 0.3, orig * 0.5, orig * 0.7, orig * 1.0,
                                      orig * 1.3, orig * 1.7, orig * 2.5, 0.0]

                best_val = orig
                best_val_sc = best_sc
                for cv in candidates:
                    a[i] = max(0.0, cv)
                    if a.sum() < 0.01:
                        a[i] = orig
                        continue
                    sc = _score_fast(a)
                    if sc < best_val_sc - 1e-15:
                        best_val = a[i]
                        best_val_sc = sc
                a[i] = best_val
                if best_val_sc < best_sc - 1e-15:
                    best_sc = best_val_sc
                    improved = True
                    # Update b and k_max periodically
                    if best_val is not orig:
                        b = _conv_full(a)
                        M = float(b.max())
                        k_max = int(np.argmax(b))

            if not improved and pass_idx == 0:
                # Two-coordinate
                for _ in range(80):
                    if time.monotonic() > stop_time - 6.0:
                        break
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i == j:
                        continue
                    oi, oj = a[i], a[j]
                    for mi in (0.3, 0.6, 1.0, 1.5, 2.5, 0.0):
                        for mj in (0.3, 0.6, 1.0, 1.5, 2.5, 0.0):
                            if mi == 1.0 and mj == 1.0:
                                continue
                            a[i] = max(0.0, oi * mi)
                            a[j] = max(0.0, oj * mj)
                            if a.sum() < 0.01:
                                a[i], a[j] = oi, oj
                                continue
                            sc = _score_fast(a)
                            if sc < best_sc - 1e-15:
                                best_sc = sc
                                improved = True
                    a[i], a[j] = oi, oj

            if not improved:
                # Random perturbation
                for _ in range(40):
                    if time.monotonic() > stop_time - 6.0:
                        break
                    k = int(rng.integers(n))
                    pert = rng.normal(0, 0.03 * max(0.1, a.mean()))
                    old = a[k]
                    a[k] = max(0.0, old + pert)
                    if a.sum() < 0.01:
                        a[k] = old
                        continue
                    sc = _score_fast(a)
                    if sc < best_sc - 1e-15:
                        best_sc = sc
                        improved = True
                    else:
                        a[k] = old

        if not improved:
            no_imp += 1
            if no_imp >= max_no_imp:
                break
        else:
            no_imp = 0

    return a, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 20.0

    parent = _get_parent()
    n_p = len(parent)

    # Fallback
    fallback = parent.copy()
    fallback_sc = float(evaluate_sequence([float(x) for x in fallback]))
    if not math.isfinite(fallback_sc):
        fallback = np.ones(320) / 320.0
        fallback_sc = float(evaluate_sequence([float(x) for x in fallback]))

    best_seq = fallback.copy()
    best_score = fallback_sc

    def _try_lp(n, init_arr, stop, sd):
        nonlocal best_seq, best_score
        try:
            res, sc = _lp_iterate(init_arr, n, stop, sd)
        except Exception:
            return
        if res is None or len(res) == 0:
            return
        if not np.all(np.isfinite(res)):
            return
        res = np.clip(res, 0, 1000)
        if res.sum() < 0.01:
            return
        try:
            exact = float(evaluate_sequence([float(x) for x in res]))
        except Exception:
            exact = math.inf
        if math.isfinite(exact) and exact < best_score - 1e-12:
            best_seq = res.copy()
            best_score = exact

    # Phase 1: LP optimization on parent length with multiple inits
    n = n_p
    i_arr = np.arange(n, dtype=np.float64)
    rng_main = np.random.default_rng(seed)

    inits = [parent.copy()]

    # Raised cosine
    inits.append(0.5 * (1.0 + np.cos(2.0 * math.pi * i_arr / n)))

    # Multi-frequency
    sup = np.ones(n) * 0.3
    for fr in (1, 2, 3, 5, 7, 11):
        ph = rng_main.uniform(0, 2 * math.pi)
        sup += 0.15 * np.cos(2.0 * math.pi * fr * i_arr / n + ph)
    inits.append(np.abs(sup) + 0.05)

    # Random abs-normal
    inits.append(np.abs(rng_main.normal(0, 1, n)) + 0.05)

    # Parent + noise
    sig = 0.12 * max(1.0, float(np.mean(parent)))
    inits.append(np.abs(parent + rng_main.normal(0, sig, n)) + 0.01)

    # Gaussian bump
    center = (n - 1) / 2.0
    width = n / 4.0
    inits.append(np.exp(-0.5 * ((i_arr - center) / width) ** 2))

    # Two-lobe
    two = np.exp(-0.5 * ((i_arr - n * 0.3) / (n * 0.12)) ** 2) + \
          np.exp(-0.5 * ((i_arr - n * 0.7) / (n * 0.12)) ** 2)
    inits.append(two)

    # Borel-type: a[i] proportional to 1/sqrt(1+|i-center|)
    borel = 1.0 / np.sqrt(1.0 + 2.0 * np.abs(i_arr - center))
    inits.append(borel)

    # Truncated sinc
    x = i_arr - center
    sinc_val = np.sinc(x / (n * 0.3))
    inits.append(np.abs(sinc_val) + 0.02)

    phase1_end = t0 + budget * 0.55
    time_per_init = max(15.0, (phase1_end - t0) / len(inits))

    for idx, init in enumerate(inits):
        if time.monotonic() > phase1_end - 8.0:
            break
        if best_score < 1.45:
            break
        stop = min(phase1_end - 5.0, time.monotonic() + time_per_init)
        _try_lp(n, init, stop, seed + 10000 * (idx + 1))

    # Phase 2: Other lengths
    phase2_end = t0 + budget * 0.72
    other_lengths = []
    for L in [320, 352, 384, 400, 416, 432, 448, 480, 512, 544, 576, 608]:
        if L != n_p and time.monotonic() < phase2_end - 12:
            other_lengths.append(L)

    time_per_len = max(8.0, (phase2_end - max(phase1_end, t0)) / max(1, len(other_lengths)))

    for L in other_lengths:
        if time.monotonic() > phase2_end - 6.0:
            break
        if best_score < 1.45:
            break
        stop = min(phase2_end - 4.0, time.monotonic() + time_per_len)
        # Interpolate parent to new length
        init_l = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, n_p), parent)
        _try_lp(L, init_l, stop, seed + 200000 + L)

    # Phase 3: Extended LP on best (warm start)
    phase3_end = t0 + budget * 0.82
    if time.monotonic() < phase3_end - 15.0:
        stop_p3 = min(phase3_end - 4.0, time.monotonic() + (phase3_end - time.monotonic()))
        _try_lp(len(best_seq), best_seq.copy(), stop_p3, seed + 999999)

    # Phase 4: Coordinate descent polish
    polish_end = deadline - 8.0
    if time.monotonic() < polish_end - 8.0:
        try:
            best_seq, best_score = _coord_descent_polish(best_seq, polish_end, seed=4242)
        except Exception:
            pass

    # Phase 5: Second polish with different seed
    if time.monotonic() < deadline - 12.0:
        try:
            bs2, sc2 = _coord_descent_polish(best_seq, deadline - 8.0, seed=77777)
            if sc2 < best_score - 1e-12:
                best_seq = bs2
                best_score = sc2
        except Exception:
            pass

    # Final validation
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in fallback]

    if final_sc < best_score:
        best_score = final_sc

    return [float(x) for x in best_seq]
