def _conv_fft(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_fft(a)
    m = float(b.max())

    return 2.0 * n * m / (s * s)

def _build_A_sparse(f, n):

    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j+n, j] = f
    return A

def _lp_solve(f, n, M, time_limit=8.0):
    A = _build_A_sparse(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, None)] * n,
                      method="highs",
                      options={"presolve": True, "time_limit": time_limit})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    gs = float(g.sum())
    if gs < 1e-14:
        return None

def _line_search(f, g_dir, best_sc, t_init=0.3, stop_time=None):

    t_lo, t_hi = 0.0, t_init
    f_lo = f
    best_t, best_f = 0.0, f
    for it in range(20):
        if stop_time and time.monotonic() > stop_time:
            break
        t_mid = 0.5 * (t_lo + t_hi)
        fn = (1.0 - t_mid) * f + t_mid * g_dir
        np.maximum(fn, 0.0, out=fn)
        fs = float(fn.sum())
        if fs < 1e-10:
            t_hi = t_mid
            continue
        sc = _score(fn)
        if sc < best_sc - 1e-15:
            t_lo = t_mid
            best_t = t_mid
            best_f = fn.copy()
        else:
            t_hi = t_mid
        if t_hi - t_lo < 1e-4:
            break
    if best_t < 1e-5:
        return 0.0, None
    return best_t, best_f

def _coord_descent(f0, n, stop_time, seed):
    """Coordinate descent with 1 and 2 coordinate moves"""
    rng = np.random.default_rng(seed)

    f = f / f.sum()
    best_sc = _score(f)
    multipliers = (0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0, 3.0, 0.0)
    no_imp_rounds = 0
    
    while time.monotonic() < stop_time - 3.0:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 3.0:
                return f, best_sc
            orig = f[i]
            best_local = orig
            best_local_sc = best_sc
            for mlt in multipliers:
                f[i] = orig * mlt if orig > 1e-12 else (0.01 if mlt > 0 else 0.0)
                sc = _score(f)
                if sc < best_local_sc:
                    best_local_sc = sc
                    best_local = f[i]
            f[i] = best_local
            if best_local_sc < best_sc - 1e-15:
                best_sc = best_local_sc
                improved = True
        
        # 2-coordinate moves
        for _ in range(30):
            if time.monotonic() > stop_time - 3.0:
                return f, best_sc
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            oi, oj = f[i], f[j]
            for mi in (0.5, 1.5, 2.0):
                for mj in (0.5, 1.5):
                    f[i] = oi * mi if oi > 1e-12 else 0.01
                    f[j] = oj * mj if oj > 1e-12 else 0.01
                    sc = _score(f)
                    if sc < best_sc - 1e-15:
                        best_sc = sc
                        improved = True
            f[i], f[j] = oi, oj
        
        if not improved:
            no_imp_rounds += 1
            if no_imp_rounds >= 8:
                break
        else:
            no_imp_rounds = 0
    
    return f, best_sc
