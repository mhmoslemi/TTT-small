                f[i] = best_val
                best_sc = best_local
                improved = True
            else:
                f[i] = best_val

        if not improved:
            no_imp_rounds += 1
            if no_imp_rounds >= max_no_imp:
                break
        else:
            no_imp_rounds = 0

    return f.copy(), best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 20.0

    parent = _get_parent()
    n_p = len(parent)

    best_seq = parent.copy()
    best_score = _score(best_seq)

    def _update_best(candidate):
        nonlocal best_seq, best_score
        if candidate is None or len(candidate) == 0:
            return
        candidate = np.clip(np.asarray(candidate, dtype=np.float64), 0, 1000)
        if not np.all(np.isfinite(candidate)):
            return
        if candidate.sum() < 0.01:
            return
        sc = _score(candidate)
        if math.isfinite(sc) and sc < best_score - 1e-14:
            best_seq = candidate.copy()
            best_score = sc

    # Phase 1: LP optimization (45% time)
    phase1_end = t0 + budget * 0.45
    rng = np.random.default_rng(seed)
    i_arr = np.arange(n_p, dtype=np.float64)

    inits = [parent.copy()]

    sig = 0.1 * max(1.0, float(parent.mean()))
    inits.append(np.clip(parent + rng.normal(0, sig, n_p), 0, None))
    inits.append(np.clip(parent * (1.0 + 0.15 * rng.normal(0, 1, n_p)), 0, None))

    inits.append(0.5 * (1.0 + np.cos(2.0 * math.pi * i_arr / n_p)))

    sup = np.ones(n_p) * 0.25
    for fr in (1, 2, 3, 5, 7, 11):
        ph = rng.uniform(0, 2 * math.pi)
        sup += 0.12 * np.cos(2.0 * math.pi * fr * i_arr / n_p + ph)
    inits.append(np.abs(sup) + 0.03)

    center = (n_p - 1) / 2.0
    for width in (n_p * 0.2, n_p * 0.3, n_p * 0.4):
        inits.append(np.exp(-0.5 * ((i_arr - center) / width) ** 2))

    two_c = n_p * 0.3
    two_w = n_p * 0.12
    inits.append(np.exp(-0.5 * ((i_arr - two_c) / two_w) ** 2) +
                 np.exp(-0.5 * ((i_arr - (n_p - two_c - 1)) / two_w) ** 2))

    time_per_init = max(10.0, (phase1_end - t0) / len(inits))

    for idx, init in enumerate(inits):
        if time.monotonic() > phase1_end - 8.0:
            break
        stop = min(phase1_end - 5.0, time.monotonic() + time_per_init)
        if time.monotonic() > stop - 2.0:
            continue
        try:
            res, sc = _optimize_lp(init, stop, seed + 1000 * (idx + 1), peak_frac=0.05)
            _update_best(res)
        except Exception:
            pass

    # Second LP pass on best (if time allows within phase 1)
    if time.monotonic() < phase1_end - 15.0:
        stop = phase1_end - 5.0
        try:
            res, sc = _optimize_lp(best_seq, stop, seed + 888888, peak_frac=0.03)
            _update_best(res)
        except Exception:
            pass

    # Phase 2: Subgradient optimization (15% time)
    phase2_end = t0 + budget * 0.62
    if time.monotonic() < phase2_end - 10.0:
        stop = min(phase2_end - 5.0, time.monotonic() + (phase2_end - time.monotonic() - 5.0))
        try:
            res, sc = _optimize_subgrad(best_seq, stop, seed + 77777)
            _update_best(res)
        except Exception:
            pass

    # Phase 3: Coordinate descent polish (23% time)
    phase3_end = t0 + budget * 0.85
    if time.monotonic() < phase3_end - 10.0:
        stop = min(phase3_end - 5.0, time.monotonic() + (phase3_end - time.monotonic() - 5.0))
        try:
            res, sc = _polish(best_seq, stop, seed + 55555)
            _update_best(res)
        except Exception:
            pass

    # Phase 4: LP on best after polish (5% time)
    phase4_end = t0 + budget * 0.90
    if time.monotonic() < phase4_end - 10.0:
        stop = min(phase4_end - 5.0, time.monotonic() + (phase4_end - time.monotonic() - 5.0))
        try:
            res, sc = _optimize_lp(best_seq, stop, seed + 666666, peak_frac=0.02)
            _update_best(res)
        except Exception:
            pass

    # Phase 5: Try other lengths (5% time)
    phase5_end = deadline + 3.0
    if time.monotonic() < phase5_end - 20.0:
        lengths_to_try = sorted(set([n_p - 16, n_p + 16, n_p - 32, n_p + 32,
                                     n_p - 48, n_p + 48]),
                                key=lambda L: abs(L - n_p))
        for L in lengths_to_try:
            if time.monotonic() > phase5_end - 8.0:
                break
            if L < 100 or L > 800:
                continue
            x_old = np.linspace(0, 1, n_p)
            x_new = np.linspace(0, 1, L)
            init_L = np.interp(x_new, x_old, parent)
            stop = min(phase5_end - 5.0, time.monotonic() + 25.0)
            try:
                res, sc = _optimize_lp(init_L, stop, seed + L * 100)
                _update_best(res)
            except Exception:
                pass

    # Final polish
    remaining = deadline - time.monotonic()
    if remaining > 15.0:
        try:
            res, sc = _polish(best_seq, deadline - 5.0, seed + 99999)
            _update_best(res)
        except Exception:
            pass

    # Final validation with exact evaluator
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        # Fallback: try parent directly
        try:
            final_sc = float(evaluate_sequence([float(x) for x in parent]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in parent]

    return [float(x) for x in best_seq]
