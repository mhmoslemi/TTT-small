def _conv_fft(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    n = len(a)
    S = float(a.sum())
    if S < 0.01:
        return math.inf
    M = float(_conv_fft(a).max())
    if not math.isfinite(M):
        return math.inf
    return 2.0 * n * M / (S * S)

def _build_A(f):
    """Build constraint matrix for LP: (f*g)[k] <= M for k=0..2n-2"""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A

def _lp_direction(f, M):
    """Find g maximizing sum(g) s.t. f*g <= M, g >= 0"""
    n = len(f)
    if M < 1e-15:
        return None
    A = _build_A(f)
    c_vec = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)

                      method='highs',

        active = np.where(conv >= thresh)[0]
        
        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            indices = np.arange(lo, hi)
            idx = k - indices
            valid = (idx >= 0) & (idx < n)
            if np.any(valid):
                grad[indices[valid]] += 2.0 * f[idx[valid]]
        grad /= len(active)
        
        f_new = f - lr * grad
        f_new = np.clip(f_new, 0, None)
        fs_new = f_new.sum()
        if fs_new < 1e-10:
            break
        f_new = f_new / fs_new
        sc = _score(f_new)
        if sc < best_sc - 1e-12:
            best_f = f_new.copy()
            best_sc = sc
            f = f_new
            lr = min(0.1, lr * 1.02)
        else:
            lr = max(0.005, lr * 0.7)

    return best_f, best_sc

def _polish(a, stop_time, seed=4242):
    """Coordinate descent polishing."""
    n = len(a)
    best = np.clip(np.asarray(a, dtype=np.float64).ravel(), 0, None).copy()
    fs = best.sum()
    if fs < 1e-10:
        return best, math.inf
    best = best / fs
    best_sc = _score(best)
    rng = np.random.default_rng(seed)
    no_imp = 0
    max_no_imp = 5

    multipliers_1d = [0.5, 0.7, 0.85, 1.0, 1.2, 1.5, 2.0, 0.0]

    while time.monotonic() < stop_time - 1.0 and no_imp < max_no_imp:
        improved = False
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() > stop_time - 1.0:
                break
            orig = best[i]
            best_val = orig
            best_val_sc = best_sc
            for mlt in multipliers_1d:
                new_val = orig * mlt if orig > 1e-12 else (0.005 if mlt > 0 else 0.0)
                best[i] = new_val
                if best.sum() < 0.01:
                    best[i] = orig
                    continue
                sc = _score(best)
                if sc < best_val_sc - 1e-13:
                    best_val = new_val
                    best_val_sc = sc
            best[i] = best_val
            if best_val_sc < best_sc - 1e-13:
                best_sc = best_val_sc
                improved = True

        # 2-coordinate moves
        n_2coord = 100
        for _ in range(n_2coord):
            if time.monotonic() > stop_time - 1.0:
                break
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            for mi in (0.5, 1.5, 0.0):
                for mj in (0.5, 1.5, 0.0):
                    best[i] = oi * mi if oi > 1e-12 else (0.005 if mi > 0 else 0.0)
                    best[j] = oj * mj if oj > 1e-12 else (0.005 if mj > 0 else 0.0)
                    if best.sum() < 0.01:
                        best[i], best[j] = oi, oj
                        continue
                    sc = _score(best)
                    if sc < best_sc - 1e-13:
                        best_sc = sc
                        improved = True
            best[i], best[j] = oi, oj

        if not improved:
            no_imp += 1
        else:
            no_imp = 0

    return best, best_sc
