def _get_parent():

def _conv_fft(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_fft(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _subgrad_topk(a, C, top_k=15):
    """Subgradient of max(C) using top-k lags."""
    n = len(a)
    nc = len(C)
    if nc <= top_k:
        idx = np.arange(nc)
    else:
        idx = np.argpartition(C, -top_k)[-top_k:]
    Cv = C[idx]
    w = Cv - Cv.min() + 1e-8
    w /= w.sum()
    grad = np.zeros(n)
    for wi, k in zip(w, idx):
        k = int(k)
        lo1 = max(0, k - n + 1)
        hi1 = min(k, n - 1)
        if lo1 <= hi1:
            j = np.arange(lo1, hi1 + 1)
            grad[j] += wi * a[k - j]
        lo2 = max(k, 0)
        hi2 = min(k + n - 1, n - 1)
        if lo2 <= hi2:
            j = np.arange(lo2, hi2 + 1)
            grad[j] += wi * a[j - k]
    return grad

def _optimize_subgrad(a0, n, stop_time, seed, top_k=15):
    """Main optimization loop using subgradient + momentum."""
    rng = np.random.default_rng(seed)
    a = np.clip(a0.copy(), 0, None)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_sc = _score(best_a)

    eta = 5.0 / (n * n)  # initial learning rate
    beta = 0.92
    v = np.zeros(n)
    no_imp = 0
    max_no_imp = 300
    it = 0
    l2_interval = 500  # every 500 iters, use L2 gradient

    while time.monotonic() < stop_time - 1.0:
        it += 1
        C = _conv_fft(a)

        if it % l2_interval == 0:
            g = _l2_grad(a, C, n)
        else:
            g = _subgrad_topk(a, C, top_k)

        # Center gradient for sum=1 constraint
        g -= g.mean()

        # Adaptive learning rate
        norm_g = float(np.linalg.norm(g))
        if norm_g > 1e-15:
            # Scale so step is at most 0.5 in L2
            step = min(eta, 0.5 / norm_g)
        else:
            step = eta

        v = beta * v - step * g
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        sc = _score(a_new)
        if sc < best_sc - 1e-14:
            best_a = a_new.copy()
            best_sc = sc
            no_imp = 0
            a = a_new
            eta = min(0.5 / n, eta * 1.003)
        else:
            no_imp += 1
            a = a_new
            eta = max(1e-6 / n, eta * 0.997)

            if no_imp > max_no_imp:
                # Perturb and restart
                sigma = 0.1 * max(1.0 / n, float(np.std(best_a)))
                a = best_a + rng.normal(0, sigma, n)
                a = np.maximum(a, 0)
                s2 = a.sum()
                a = a / s2 if s2 > 1e-10 else best_a.copy()
                v = np.zeros(n)
                eta = 5.0 / (n * n)
                beta = rng.uniform(0.85, 0.95)
                no_imp = 0
                l2_interval = rng.integers(200, 800)

        # Gradually increase top_k
        if it % 2000 == 0:
            top_k = min(40, top_k + 3)

    return best_a, best_sc


def _coord_descent(a0, n, stop_time, seed):
    """Coordinate descent polish."""
    rng = np.random.default_rng(seed)
    a = np.clip(a0.copy(), 0, None)
    s = a.sum()
    a = a / s
    best_sc = _score(a)
    no_full_pass = 0

    while time.monotonic() < stop_time - 2.0:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 2.0:
                return a, best_sc
            orig = a[i]
            # Try several multipliers
            for mlt in (0.3, 0.5, 0.7, 1.3, 1.5, 2.0, 0.0):
                a[i] = orig * mlt if orig > 1e-15 else (0.001 if mlt > 0 else 0.0)
                sc = _score(a)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    improved = True
                    orig = a[i]
                    break
            a[i] = orig  # restore if no improvement
            # Try small perturbations
            if orig > 1e-8:
                for dt in (0.02, -0.02, 0.05, -0.05):
                    a[i] = max(0.0, orig * (1 + dt))
                    sc = _score(a)
                    if sc < best_sc - 1e-14:
                        best_sc = sc
                        improved = True
                        orig = a[i]
                        break
                a[i] = orig

        if not improved:
            no_full_pass += 1
            if no_full_pass >= 5:
                break
        else:
            no_full_pass = 0

    return a, best_sc
