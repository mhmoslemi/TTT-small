from scipy.optimize import linprog

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:

def _conv(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score_fast(a, conv=None):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf

        conv = _conv(a)

def _precompute_lp(n):
    rows = np.empty(n * n, dtype=np.int64)
    cols = np.empty(n * n, dtype=np.int64)
    idx = 0
    for k in range(2 * n - 1):
        jmin = max(0, k - n + 1)
        jmax = min(k, n - 1)
        cnt = jmax - jmin + 1
        rows[idx:idx + cnt] = k
        cols[idx:idx + cnt] = np.arange(jmin, jmax + 1)
        idx += cnt
    return rows, cols

def _lp_dir(f, n, M, rows, cols, tl=3.0):
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    A[rows, cols] = f[rows - cols]
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': tl, 'presolve': True})
        if res.status in (0, 3) and res.x is not None:
            g = np.maximum(res.x, 0.0)
            gs = g.sum()
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(384, dtype=np.float64)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    return p


def _conv(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a, conv=None):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    if conv is None:
        conv = _conv(a)
    m = float(conv.max())
    return 2.0 * n * m / (s * s)


def _precompute_lp(n):
    total = n * n
    rows = np.empty(total, dtype=np.int64)
    cols = np.empty(total, dtype=np.int64)
    idx = 0
    for k in range(2 * n - 1):
        jmin = max(0, k - n + 1)
        jmax = min(k, n - 1)
        cnt = jmax - jmin + 1
        rows[idx:idx + cnt] = k
        cols[idx:idx + cnt] = np.arange(jmin, jmax + 1)
        idx += cnt
    return rows, cols


def _lp_direction(f, n, M, rows, cols, time_limit=3.0):
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    A[rows, cols] = f[rows - cols]
    c_obj = -np.ones(n, dtype=np.float64)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, None)] * n, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status in (0, 3) and res.x is not None:
            g = np.maximum(res.x, 0.0)
            gs = float(g.sum())
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None


def _focused_lp(f, n, M, rows, cols, top_k=60, time_limit=2.0):
    conv_f = _conv(f)
    idx_sorted = np.argsort(-conv_f)[:top_k]
    A_full = np.zeros((2 * n - 1, n), dtype=np.float64)
    A_full[rows, cols] = f[rows - cols]
    A_sub = A_full[idx_sorted]
    c_obj = -np.ones(n, dtype=np.float64)
    b_sub = np.full(top_k, M, dtype=np.float64)
    try:
        res = linprog(c_obj, A_ub=A_sub, b_ub=b_sub,
                      bounds=[(0.0, None)] * n, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status in (0, 3) and res.x is not None:
            g = np.maximum(res.x, 0.0)
            gs = float(g.sum())
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None


def _line_search(f, g, best_sc, n):
    best_t = 0.0
    best_sc_t = best_sc
    ts = [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 0.9, 1.0]
    for tv in ts:
        f_try = (1.0 - tv) * f + tv * g
        np.clip(f_try, 0, None, out=f_try)
        fs = float(f_try.sum())
        if fs < 1e-10:
            continue
        f_try /= fs
        sc = _score(f_try)
        if sc < best_sc_t - 1e-15:
            best_sc_t = sc
            best_t = tv
    if best_t > 0.1 and best_t < 0.9:
        lo, hi = max(0.0, best_t - 0.08), min(1.0, best_t + 0.08)
        for _ in range(4):
            mid = 0.5 * (lo + hi)
            f_try = (1.0 - mid) * f + mid * g
            np.clip(f_try, 0, None, out=f_try)
            fs = float(f_try.sum())
            if fs < 1e-10:
                break
            f_try /= fs
            sc = _score(f_try)
            if sc < best_sc_t - 1e-15:
                best_sc_t = sc
                best_t = mid
                if sc < _score((1.0 - lo) * f + lo * g):
                    hi = mid
                else:
                    lo = mid
            else:
                lo = mid if mid > lo else hi
                hi = mid if mid < hi else lo
                lo, hi = min(lo, hi), max(lo, hi)
    return best_t, best_sc_t


def _lp_optimize(f0, n, rows, cols, stop_time, seed=42):
    rng = np.random.default_rng(seed)
    f = np.clip(f0.copy(), 0, None)
    fs = float(f.sum())
    if fs < 1e-10:
        f = np.ones(n)
    else:
        f = f / fs

    best_f = f.copy()
    best_sc = _score(f)

    t_prev = 0.25
    stall = 0
    max_stall = 70
    iters = 0
    last_improve_iter = 0

    while time.monotonic() < stop_time - 2.0:
        conv_f = _conv(f)
        M = float(conv_f.max())

        use_focused = (iters % 5 == 0)
        if use_focused:
            g = _focused_lp(f, n, M, rows, cols, top_k=50, time_limit=1.5)
        if g is None:
            g = _lp_direction(f, n, M, rows, cols, time_limit=3.0)

        if g is None:
            stall += 3
            if stall > max_stall:
                f = np.clip(f + rng.normal(0, 0.04, n), 0, None)
                fs2 = f.sum()
                if fs2 > 1e-10:
                    f /= fs2
                else:
                    f = best_f.copy()
                stall = 0
            iters += 1
            continue

        diff = np.abs(g - f)
        if diff.max() < 1e-10:
            stall += 2
            if stall > max_stall:
                f = np.clip(f + rng.normal(0, 0.04, n), 0, None)
                fs2 = f.sum()
                if fs2 > 1e-10:
                    f /= fs2
                else:
                    f = best_f.copy()
                stall = 0
            iters += 1
            continue

        best_t, best_sc_t = _line_search(f, g, best_sc, n)

        if best_t > 1e-6 and best_sc_t < best_sc - 1e-15:
            f = (1.0 - best_t) * f + best_t * g
            np.clip(f, 0, None, out=f)
            fsum = float(f.sum())
            if fsum > 1e-10:
                f /= fsum
            best_f = f.copy()
            best_sc = best_sc_t
            t_prev = best_t
            stall = 0
            last_improve_iter = iters
        else:
            stall += 1
            t_prev = max(0.05, t_prev * 0.85)
            if stall >= max_stall:
                strategy = rng.integers(0, 6)
                if strategy == 0:
                    f = np.clip(f + rng.normal(0, 0.06, n), 0, None)
                elif strategy == 1:
                    blk = n // 4
                    f2 = f.copy()
                    f2[:blk], f2[n - blk:] = f[n - blk:].copy(), f[:blk].copy()
                    f = f2
                elif strategy == 2:
                    kern = np.exp(-0.5 * (np.arange(-6, 7) / 2.5) ** 2)
                    kern /= kern.sum()
                    noise = np.convolve(rng.normal(0, 0.25, n), kern, 'same')
                    f = np.clip(f * (1.0 + noise), 0, None)
                elif strategy == 3:
                    f = np.clip(f[::-1] * 0.5 + f * 0.5 + rng.normal(0, 0.02, n), 0, None)
                elif strategy == 4:
                    # Block scale
                    bs = n // 4
                    scales = rng.uniform(0.5, 2.0, size=4)
                    for b in range(4):
                        f[b * bs:(b + 1) * bs] *= scales[b]
                    f = np.clip(f, 0, None)
                else:
                    # Sparse perturbation
                    n_pert = rng.integers(5, 20)
                    idx_pert = rng.choice(n, size=n_pert, replace=False)
                    f2 = f.copy()
                    f2[idx_pert] *= rng.uniform(0.3, 3.0, size=n_pert)
                    f = np.clip(f2, 0, None)
                fs2 = f.sum()
                if fs2 > 1e-10:
                    f /= fs2
                else:
                    f = best_f.copy()
                t_prev = 0.25
                stall = 0

        iters += 1

    return best_f, best_sc


def _coord_descent(a, stop_time, seed=123):
    n = len(a)
    rng = np.random.default_rng(seed)
    a = np.clip(a.copy(), 0, None)
    s = float(a.sum())
    if s < 1e-10:
        return a, math.inf
    conv = _conv(a)
    best_sc = 2.0 * n * conv.max() / (s * s)
    no_imp = 0
    max_no_imp = 4

    while time.monotonic() < stop_time and no_imp < max_no_imp:
        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > stop_time:
                break
            v = a[i]
            avg = s / n
            if v < 1e-10:
                candidates = [0.0, avg * 0.1, avg * 0.3, avg * 0.5, avg, avg * 1.5]
            else:
                candidates = [0.0, v * 0.1, v * 0.3, v * 0.5, v * 0.7, v * 0.9,
                              v * 1.1, v * 1.3, v * 1.7, v * 2.5, avg, avg * 2.0]

            best_cv = v
            best_local = best_sc
            best_conv_try = None

            for cv in candidates:
                cv = max(0.0, cv)
                if abs(cv - v) < 1e-14:
                    continue
                a[i] = cv
                new_s = s - v + cv
                if new_s < 0.01:
                    a[i] = v
                    continue
                new_conv = _conv(a)
                new_sc = 2.0 * n * new_conv.max() / (new_s * new_s)
                if new_sc < best_local - 1e-14:
                    best_local = new_sc
                    best_cv = cv
                    best_conv_try = new_conv
                a[i] = v

            if best_cv != v and best_conv_try is not None:
                a[i] = best_cv
                s = s - v + best_cv
                conv = best_conv_try
                best_sc = best_local
                improved = True

        if not improved:
            no_imp += 1
        else:
            no_imp = 0

    return a, best_sc


def _pair_descent(a, stop_time, seed=456):

    s = float(a.sum())

    best_sc = _score(a)

            i = int(rng.integers(n))

            if i == j:

            vi, vj = a[i], a[j]

            if total < 1e-12:

            for ratio in (0.1, 0.3, 0.5, 0.7, 0.9, 0.0, 1.0):

                a[i], a[j] = ni, nj

            if no_imp >= 2:

        else:
            no_imp = 0
