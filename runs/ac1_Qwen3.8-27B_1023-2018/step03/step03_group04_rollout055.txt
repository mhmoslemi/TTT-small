        return None
    A = _conv_mat(f)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': 15, 'mip_rel_gap': 1e-9})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    s = float(g.sum())
    if s < 1e-12:
        return None
    return g / s


def _rk_optimize(f0, stop_time, seed, max_iters=100000):
    rng = np.random.default_rng(seed)
    n = len(f0)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel(), 0, None).copy()
    S = f.sum()
    if S < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / S

    best_f = f.copy()
    best_sc = _score(f)
    t = 0.25
    stall = 0
    max_stall = 150

    for _ in range(max_iters):
        if time.monotonic() > stop_time - 1.5:
            break

        conv = _conv(f)
        M = float(conv.max())
        if M < 1e-18:
            break

        g = _lp_dir(f, M)
        if g is None:
            stall += 4
            if stall > max_stall:
                break
            f = np.clip(f + rng.normal(0, 0.06, n), 0, None)
            f = f / max(f.sum(), 1e-15)
            continue

        dist = float(np.linalg.norm(g - f))
        if dist < 1e-7:
            stall += 2
            if stall > max_stall:
                break
            f = np.clip(f + rng.normal(0, 0.04, n), 0, None)
            f = f / max(f.sum(), 1e-15)
            continue

        # Backtracking line search
        sc_now = _score(f)
        best_tc = 0.0
        best_sc_t = sc_now
        trial = None
        for tc in (t, t * 0.5, t * 1.5, t * 0.3, t * 2.0):
            tc = max(0.01, min(0.85, tc))
            fn = np.clip((1.0 - tc) * f + tc * g, 0, None)
            sn = fn.sum()
            if sn < 1e-12:
                continue
            fn = fn / sn
            s = _score(fn)
            if s < best_sc_t - 1e-15:
                best_sc_t = s
                best_tc = tc
                trial = fn

        if trial is not None:
            f = trial
            actual_sc = _score(f)
            if actual_sc < best_sc - 1e-15:
                best_f = f.copy()
                best_sc = actual_sc
                stall = 0
                t = min(0.55, t * 1.06)
            else:
                stall += 1
                t = max(0.03, t * 0.7)
        else:
            stall += 1
            t = max(0.03, t * 0.55)

        if stall > max_stall:
            # Structured restart: shift mass from peaks to valleys
            idx_hi = np.argsort(f)[-n // 4:]
            idx_lo = np.argsort(f)[:-n // 4]
            amt = 0.3 * f[rng.choice(idx_hi)]
            f[idx_hi] -= amt
            f[idx_lo] += amt / len(idx_lo)
            f = np.clip(f, 0, None)
            f = f / max(f.sum(), 1e-15)
            t = 0.25
            stall = 0

    return best_f, best_sc
