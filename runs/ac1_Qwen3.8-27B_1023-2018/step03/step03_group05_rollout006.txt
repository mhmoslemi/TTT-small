```
def project_simplex(v, target=1.0):
    n=v.size
    if n==0: return np.ones(1)*target
    v=np.where(np.isfinite,v,0)
    u=np.sort(v)[::-1]
    css=np.cumsum(u)
    idx=np.arange(1,n+1)
    cond=(css-target)/idx < u
    rho=np.nonzero(cond)[0]
    if rho.size==0: return np.full(n,target/n)
    theta=(css[rho[-1]]-target)/(rho[-1]+1)
    w=np.maximum(v-theta,0)
    s=w.sum()
    if s<=0: w=np.full(n,target/n)
    else: w*=target/s
    return w
```

```
from scipy.fft import rfft, irfft, next_fast_len
```

```python
def _safe_float_b(budget_s):
    try:
        return max(0.0, float(budget_s))
    except Exception:

def _normalize(a, target=1.0):
    arr = np.asarray(a, dtype=np.float64).ravel()
    if arr.size == 0:
        return np.ones(1, dtype=np.float64) * target
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    s = float(arr.sum())
    if s <= 1e-300:
        return np.full(arr.size, target / arr.size, dtype=np.float64)
    return (arr / s) * target


def _get_parent():
    h = globals().get("height_sequence_1", None)
    if h is not None:
        try:
            p = np.asarray(h, dtype=np.float64).ravel()
        except Exception:
            p = None
        if p is not None and p.size > 0 and np.all(np.isfinite(p)):
            p = np.clip(p, 0.0, 1000.0)
            s = float(p.sum())
            if s > 1e-300:
                return _normalize(p)
    return _normalize(np.ones(512, dtype=np.float64))


def _conv_full(a):
    arr = np.asarray(a, dtype=np.float64).ravel()
    n = arr.size
    if n == 0:
        return np.array([0.0], dtype=np.float64)
    nfft = next_fast_len(2 * n - 1)
    fa = rfft(arr, nfft)
    c = irfft(fa * fa, nfft)[:2 * n - 1]
    c = np.asarray(c, dtype=np.float64)
    c = np.where(np.isfinite(c), c, 0.0)
    np.maximum(c, 0.0, out=c)
    return c


def _weighted_grad(a, w):
    a = np.asarray(a, dtype=np.float64).ravel()
    w = np.asarray(w, dtype=np.float64).ravel()
    n = a.size
    m = 2 * n - 1
    if w.size < m:
        ww = np.zeros(m, dtype=np.float64)
        ww[:w.size] = w
        w = ww
    elif w.size > m:
        w = w[:m]
    w = np.where(np.isfinite(w), w, 0.0)
    s = float(w.sum())
    if s <= 1e-300:
        return np.zeros(n, dtype=np.float64)
    w = w / s

    nfft = next_fast_len(3 * n - 2)
    fa = rfft(a, nfft)
    fw = rfft(w[::-1], nfft)
    cr = irfft(fa * fw, nfft)[:3 * n - 2]
    idx = (2 * n - 2) - np.arange(n, dtype=np.int64)
    g = 2.0 * cr[idx]
    g = np.where(np.isfinite(g), g, 0.0)
    np.maximum(g, 0.0, out=g)
    return g


def _project_simplex(v, target=1.0):
    v = np.asarray(v, dtype=np.float64).ravel()
    n = v.size
    if n == 0:
        return np.ones(1, dtype=np.float64) * target
    v = np.where(np.isfinite(v), v, 0.0)
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    with np.errstate(divide="ignore", invalid="ignore"):
        cond = (css - target) / idx < u
    rho = np.nonzero(cond)[0]
    if rho.size == 0:
        return np.full(n, target / n, dtype=np.float64)
    r = int(rho[-1])
    theta = (css[r] - target) / (r + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = float(w.sum())
    if s <= 1e-300:
        return np.full(n, target / n, dtype=np.float64)
    return (w / s) * target


def _resample(a, m):
    a = _normalize(a)
    n = a.size
    m = int(m)
    if m <= 0:
        return np.ones(1, dtype=np.float64)
    if n == 0:
        return np.ones(m, dtype=np.float64) / m
    if n == m:
        return a.copy()
    if m > n:
        if n == 1:
            b = np.full(m, a[0], dtype=np.float64)
        else:
            old = np.linspace(0.0, 1.0, n, endpoint=True)
            new = np.linspace(0.0, 1.0, m, endpoint=True)
            b = np.interp(new, old, a)
        b = np.where(np.isfinite(b), b, 0.0)
        np.maximum(b, 0.0, out=b)
    else:
        idx = (np.arange(n, dtype=np.int64) * m) // n
        b = np.bincount(idx, minlength=m).astype(np.float64)
        b = np.where(np.isfinite(b), b, 0.0)
        np.maximum(b, 0.0, out=b)
    return _normalize(b)


def _random_smooth(n, rng, base=None):
    n = int(n)
    if n <= 0:
        return np.ones(1, dtype=np.float64)
    if n == 1:
        return np.ones(1, dtype=np.float64)
    x = np.linspace(0.0, 1.0, n, endpoint=False)
    kind = int(rng.integers(0, 8))
    if kind == 0:
        p = float(rng.uniform(0.2, 3.5))
        q = float(rng.uniform(0.2, 3.5))
        a = x ** p + (1.0 - x) ** q + 1e-3
    elif kind == 1:
        p = float(rng.uniform(0.2, 4.5))
        a = x ** p + 0.05
    elif kind == 2:
        p = float(rng.uniform(0.2, 4.5))
        a = (1.0 - x) ** p + 0.05
    elif kind == 3:
        p = float(rng.uniform(0.2, 4.0))
        q = float(rng.uniform(0.2, 4.0))
        a = (x ** p) * (1.0 - x) ** q + 0.05
    elif kind == 4:
        c = float(rng.uniform(0.15, 0.85))
        wd = float(rng.uniform(0.05, 0.35))
        a = np.exp(-((x - c) ** 2) / (2.0 * wd * wd)) + 0.05
    elif kind == 5:
        a = 1.0
        a += float(rng.uniform(0.1, 0.8)) * np.cos(2.0 * np.pi * x + float(rng.uniform(0, 2 * np.pi)))
        a += float(rng.uniform(0.1, 0.5)) * np.cos(4.0 * np.pi * x + float(rng.uniform(0, 2 * np.pi)))
    elif kind == 6:
        ramp = x - (1.0 - x)
        a = np.ones(n, dtype=np.float64) + float(rng.uniform(0.2, 0.9)) * ramp
    else:
        a = rng.uniform(0.2, 1.5, n).astype(np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    np.maximum(a, 1e-6, out=a)
    if base is not None:
        base = np.asarray(base, dtype=np.float64).ravel()
        if base.size == n and np.all(np.isfinite(base)):
            mix = float(rng.uniform(0.25, 0.65))
            a = mix * _normalize(base) + (1.0 - mix) * a
    return _normalize(a)


def _eval_candidate(v):
    if v is None:
        return None
    try:
        vv = _normalize(v)
    except Exception:
        return None
    if vv.size == 0:
        return None
    if not np.all(np.isfinite(vv)):
        return None
    try:
        cv = _conv_full(vv)
        mv = float(cv.max())
    except Exception:
        return None
    if not np.isfinite(mv) or mv < 0.0:
        return None
    return vv, cv, mv


def _weight_variants(c, n):
    c = np.asarray(c, dtype=np.float64).ravel()
    m = c.size
    if m == 0:
        return []
    M = float(c.max())
    avg = float(c.mean())
    gap = M - avg
    out = []

    unif = np.full(m, 1.0 / m, dtype=np.float64)
    out.append(unif)

    top_fracs = (0.08, 0.25) if n <= 1024 else (0.12,)
    for frac in top_fracs:
        K = int(max(4, min(m, round(m * frac))))
        if K >= m:
            idx = np.arange(m)
        else:
            idx = np.argpartition(c, m - K)[-K:]
        w = np.zeros(m, dtype=np.float64)
        w[idx] = 1.0 / float(max(1, int(idx.size)))
        out.append(w)

    if gap > 1e-15:
        T = max(1e-15, 0.12 * gap)
        z = (c - M) / T
        z = np.clip(np.where(np.isfinite(z), z, -50.0), -50.0, 0.0)
        soft = np.exp(z)
        ss = float(soft.sum())
        if ss > 0.0 and np.isfinite(ss):
            soft = soft / ss
            K = int(max(4, min(m, round(0.15 * m) if n <= 1024 else round(0.10 * m))))
            if K >= m:
                idx = np.arange(m)
            else:
                idx = np.argpartition(c, m - K)[-K:]
            top = np.zeros(m, dtype=np.float64)
            top[idx] = 1.0 / float(max(1, int(idx.size)))
            out.append(0.55 * soft + 0.45 * top)
            out.append(soft)
            tol = 0.4 * gap
            mask = c >= (M - tol)
            cnt = int(mask.sum())
            if 0 < cnt < m:
                wm = np.zeros(m, dtype=np.float64)
                wm[mask] = 1.0 / cnt
                out.append(wm)
    return out


def _random_weights(c, rng, mix=0.08):
    c = np.asarray(c, dtype=np.float64).ravel()
    m = c.size
    if m == 0:
        return None
    M = float(c.max())
    avg = float(c.mean())
    gap = M - avg
    if gap > 1e-15:
        T = max(1e-15, 0.25 * gap)
        z = (c - M) / T
        z = np.clip(np.where(np.isfinite(z), z, -50.0), -50.0, 0.0)
        base = np.exp(z)
        s = float(base.sum())
        if s <= 0.0 or not np.isfinite(s):
            base = np.full(m, 1.0 / m, dtype=np.float64)
        else:
            base = base / s
    else:
        base = np.full(m, 1.0 / m, dtype=np.float64)
    noise = rng.random(m).astype(np.float64)
    ns = float(noise.sum())
    if ns <= 0.0 or not np.isfinite(ns):
        noise = np.full(m, 1.0 / m, dtype=np.float64)
    else:
        noise /= ns
    mix = min(0.65, max(0.0, float(mix)))
    w = (1.0 - mix) * base + mix * noise
    ws = float(w.sum())
    if ws <= 0.0 or not np.isfinite(ws):
        return None
    return w / ws


def _flat_weights(c, a, kfrac=0.20):
    c = np.asarray(c, dtype=np.float64).ravel()
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    m = c.size
    if n < 2 or m < 2:
        return None

    M = float(c.max())
    avg = float(c.mean())
    gap = M - avg

    minK = int(min(m, max(4, round(0.05 * m))))
    maxK = int(min(m, max(minK, round(m * min(max(0.05, kfrac))))))
    maxK = min(maxK, 384)

    if gap > 1e-15:
        tol = max(1e-15, 0.35 * gap)
        mask = c >= (M - tol)
        cnt = int(mask.sum())
        if cnt > maxK:
            idx = np.argpartition(c, m - maxK)[-maxK:]
        elif cnt < minK:
            idx = np.argpartition(c, m - minK)[-minK:]
        else:
            idx = np.nonzero(mask)[0]
    else:
        idx = np.arange(m)

    idx = np.asarray(idx, dtype=np.int64).ravel()
    K = int(idx.size)
    if K <= 0:
        return None

    scale = max(float(np.mean(a)), 1e-300)
    B = np.zeros((n, K), dtype=np.float64)

    for t, k in enumerate(idx):
        k = int(k)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        if hi >= lo:
            j = k - np.arange(lo, hi + 1, dtype=np.int64)
            B[lo:hi + 1, t] = a[j] / scale

    A = np.empty((n + 1, K), dtype=np.float64)
    A[:n, :] = B
    A[n, :] = 1.0
    b = np.ones(n + 1, dtype=np.float64)

    try:
        res = lsq_linear(A, b, bounds=(0.0, 1.0), options={"maxiter": 300, "tol": 1e-9})
        x = np.asarray(res.x, dtype=np.float64).ravel()
        if x.size == K and np.all(np.isfinite(x)):
            x = np.maximum(x, 0.0)
            s = float(x.sum())
            if s > 1e-14:
                w = np.zeros(m, dtype=np.float64)
                w[idx] = x / s
                return w
    except Exception:
        pass

    top = np.zeros(m, dtype=np.float64)
    top[idx] = 1.0 / max(1, K)
    return top


def _lbfgs_run(a, deadline, Tfac=0.001, maxfun=80):
    a = _normalize(a)
    n = a.size
    rem = deadline - time.monotonic()
    if n < 2 or rem < 6.0:
        return a, float(_conv_full(a).max())

    c0 = _conv_full(a)
    M0 = float(c0.max())

    ref = int(np.argmax(a))
    idx = np.delete(np.arange(n, dtype=np.int64), ref)
    x0 = np.log(np.maximum(a, 1e-300) / max(float(a[ref]), 1e-300))
    x0 = np.delete(x0, ref)
    x0 = np.where(np.isfinite(x0), x0, 0.0)
    x0 = np.clip(x0, -18.0, 18.0)

    T = max(1e-12, Tfac * max(M0, 1.0 / n))
    scale = 2.0 * float(n)

    def fun(x):
        x = np.asarray(x, dtype=np.float64).ravel()
        z = np.empty(n, dtype=np.float64)
        z[ref] = 0.0
        z[idx] = x
        zmax = float(np.max(z)) if z.size else 0.0
        if not np.isfinite(zmax):
            zmax = 0.0
        z = z - zmax
        z = np.clip(z, -30.0, 30.0)
        q = np.exp(z)
        qs = float(q.sum())
        if qs <= 1e-300 or not np.isfinite(qs):
            return 1e6, np.zeros(x.shape, dtype=np.float64)
        q = q / qs
        cv = _conv_full(q)
        Mc = float(cv.max())
        if not np.isfinite(Mc):
            return 1e6, np.zeros(x.shape, dtype=np.float64)
        zz = (cv - Mc) / T
        zz = np.where(np.isfinite(zz), zz, -50.0)
        zz = np.clip(zz, -50.0, 0.0)
        ee = np.exp(zz)
        s = float(ee.sum())
        if s <= 1e-300 or not np.isfinite(s):
            s = 1.0
            ee = np.zeros(cv.size, dtype=np.float64)
            ee[0] = 1.0
        val = scale * (Mc + T * math.log(s))
        if not np.isfinite(val):
            val = 1e6
        w = ee / s
        g = _weighted_grad(q, w)
        if not np.all(np.isfinite(g)):
            g = np.zeros(n, dtype=np.float64)
        dotg = float(np.dot(q, g))
        gx = scale * q[idx] * (g[idx] - dotg)
        gx = np.where(np.isfinite(gx), gx, 0.0)
        return val, gx

    try:
        res = minimize(
            fun,
            x0,
            method="L-BFGS-B",
            jac=True,
            bounds=[(-18.0, 18.0)] * (n - 1),
            options={
                "maxfun": int(maxfun),
                "maxiter": int(max(1, int(maxfun) * 2)),
                "maxls": 20,
                "ftol": 1e-14,
                "gtol": 1e-8,
            },
        )
        x = np.asarray(res.x if getattr(res, "x", None) is not None else x0, dtype=np.float64).ravel()
    except Exception:
        return a, M0

    z = np.empty(n, dtype=np.float64)
    z[ref] = 0.0
    z[idx] = x
    zmax = float(np.max(z)) if z.size else 0.0
    if not np.isfinite(zmax):
        zmax = 0.0
    q = np.exp(np.clip(z - zmax, -30.0, 30.0))
    qs = float(q.sum())
    if qs <= 1e-300 or not np.isfinite(qs):
        return a, M0
    q = q / qs
    cv = _conv_full(q)
    Mc = float(cv.max())
    if np.isfinite(Mc) and Mc < M0 * (1.0 - 1e-12):
        return q, Mc
    return a, M0


def _lbfgs_polish(a, deadline, rng, max_total_s=60.0):
    a = _normalize(a)
    best = a.copy()
    c = _conv_full(best)
    bestM = float(c.max())
    start = time.monotonic()

    attempts = ((0.003, 60), (0.0008, 80), (0.00025, 60))
    for fac, mf in attempts:
        rem = deadline - time.monotonic()
        used = time.monotonic() - start
        if rem < 12.0 or used > max(5.0, float(max_total_s) * 0.75):
            break
        try:
            a2, M2 = _lbfgs_run(best, deadline, Tfac=fac, maxfun=mf)
        except Exception:
            a2, M2 = best, bestM
        if np.all(np.isfinite(a2)) and np.isfinite(M2) and M2 < bestM * (1.0 - 1e-12):
            best = a2
            bestM = M2
    return best, bestM


def _local_refine(a, deadline, rng, use_flat=True):
    a = _normalize(a)
    n = a.size
    if n < 2:
        return a, 1.0

    c = _conv_full(a)
    M = float(c.max())

    best_a = a.copy()
    best_M = M
    cur_a = a.copy()
    cur_c = c.copy()
    cur_M = M

    base_variants = _weight_variants(cur_c, n)
    flat_invalid = True
    it = 0
    noimp = 0

    if n <= 512:
        muls = (0.03, 0.10, 0.30, 1.00, 3.00, 8.00)
        projs = (0.20, 1.00, 4.00)
    else:
        muls = (0.05, 0.20, 0.60, 1.50, 4.00)
        projs = (0.50, 2.00)

    while time.monotonic() < deadline - 2.0:
        rem = deadline - time.monotonic()
        if rem < 3.0:
            break
        it += 1

        variants = list(base_variants)
        mix = 0.18 if noimp > 15 else 0.07
        wr = _random_weights(cur_c, rng, mix=mix)
        if wr is not None:
            variants.append(wr)

        if (
            use_flat
            and n <= 1536
            and rem > 6.0
            and (it == 1 or it % 12 == 0 or noimp >= 18 or flat_invalid)
        ):
            for kf in (0.12, 0.30):
                wf = _flat_weights(cur_c, cur_a, kfrac=kf)
                if wf is not None:
                    variants.insert(0, wf)
            flat_invalid = False

        state = [cur_a.copy(), cur_c.copy(), cur_M]

        for w in variants:
            if time.monotonic() >= deadline - 2.0:
                break
            try:
                g = _weighted_grad(cur_a, w)
            except Exception:
                continue
            if g.size != n or not np.all(np.isfinite(g)):
                continue
            gmax = float(g.max())
            if gmax <= 1e-18:
                continue
            gm = float(g.mean())
            if gm <= 1e-300 or not np.isfinite(gm):
                continue
            gs = g / gm

            for lam in muls:
                if time.monotonic() >= deadline - 2.0:
                    break
                try:
                    v = cur_a * np.exp(-lam * gs)
                except Exception:
                    v = None
                res = _eval_candidate(v)
                if res is not None and res[2] < state[2] * (1.0 - 1e-13):
                    state[0], state[1], state[2] = res

            for lam in projs:
                if time.monotonic() >= deadline - 2.0:
                    break
                try:
                    v = _project_simplex(cur_a - lam * gs)
                except Exception:
                    v = None
                res = _eval_candidate(v)
                if res is not None and res[2] < state[2] * (1.0 - 1e-13):
                    state[0], state[1], state[2] = res

        if state[2] < cur_M * (1.0 - 1e-12):
            cur_a, cur_c, cur_M = state[0], state[1], state[2]
            noimp = 0
            flat_invalid = True
            if cur_M < best_M * (1.0 - 1e-12):
                best_a = cur_a.copy()
                best_M = cur_M
        else:
            noimp += 1
            if noimp >= 18:
                pert_state = [cur_a.copy(), cur_c.copy(), cur_M]
                evaluated = [False]

                for rho in (0.05, 0.10, 0.20, 0.35, 0.55):
                    if time.monotonic() >= deadline - 2.0:
                        break
                    base = best_a if rng.random() < 0.35 else None
                    r = _random_smooth(n, rng, base=base)
                    v = _normalize((1.0 - rho) * cur_a + rho * r)
                    res = _eval_candidate(v)
                    if res is not None:
                        evaluated[0] = True
                        if res[2] < pert_state[2] * (1.0 - 1e-13):
                            pert_state[0], pert_state[1], pert_state[2] = res

                for rho in (0.05, 0.15, 0.30):
                    if time.monotonic() >= deadline - 2.0:
                        break
                    ramp = np.linspace(-1.0, 1.0, n, dtype=np.float64)
                    ramp -= float(ramp.mean())
                    v = cur_a * (1.0 + rho * ramp)
                    res = _eval_candidate(v)
                    if res is not None:
                        evaluated[0] = True
                        if res[2] < pert_state[2] * (1.0 - 1e-13):
                            pert_state[0], pert_state[1], pert_state[2] = res

                if evaluated[0]:
                    if pert_state[2] < cur_M * (1.0 - 1e-12):
                        cur_a, cur_c, cur_M = pert_state[0], pert_state[1], pert_state[2]
                        if cur_M < best_M * (1.0 - 1e-12):
                            best_a = cur_a.copy()
                            best_M = cur_M
                    elif pert_state[2] <= best_M * 1.08:
                        cur_a, cur_c, cur_M = pert_state[0], pert_state[1], pert_state[2]
                        if cur_M < best_M * (1.0 - 1e-12):
                            best_a = cur_a.copy()
                            best_M = cur_M
                noimp = 0
                base_variants = _weight_variants(cur_c, n)
                flat_invalid = True

        if noimp > 300:
            break

    rem = deadline - time.monotonic()
    if rem > 25.0:
        try:
            a2, M2 = _lbfgs_polish(cur_a, deadline, rng, max_total_s=min(70.0, max(10.0, rem - 5.0)))
            if np.all(np.isfinite(a2)) and np.isfinite(M2) and M2 < best_M * (1.0 - 1e-12):
                best_a = a2
                best_M = M2
        except Exception:
            pass

    if not np.all(np.isfinite(best_a)) or best_a.size == 0:
        best_a = _normalize(np.ones(max(2, n), dtype=np.float64))
        best_M = float(_conv_full(best_a).max())
    return _normalize(best_a), best_M


def _exact_score(a):
    try:
        a = np.asarray(a, dtype=np.float64).ravel()
        if a.size == 0 or not np.all(np.isfinite(a)):
            return float("inf")
        seq = [float(x) for x in a]
        sc = evaluate_sequence(seq)
        sc = float(sc)
        return sc if np.isfinite(sc) else float("inf")
    except Exception:
        return float("inf")


def _make_phases(budget, parent_len):
    L0 = int(parent_len) if 32 <= int(parent_len) <= 2048 else 512
    if budget < 220.0:
        return [(L0, 1.0, False)]
    if budget < 650.0:
        phases = []
        if L0 not in (512, 1024):
            phases.append((L0, 0.12, False))
        phases.extend([
            (1024, 0.42, False),
            (512, 0.36, False),
            (1024, 0.10, True),
        ])
        return phases
    phases = []
    if L0 not in (512, 1024, 1536, 2048):
        phases.append((L0, 0.08, False))
    phases.extend([
        (1024, 0.26, False),
        (512, 0.22, False),
        (1536, 0.13, False),
        (2048, 0.10, False),
        (512, 0.13, True),
        (1024, 0.10, True),
    ])
    return phases


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = _safe_float_b(budget_s)

    if budget <= 1.0:
        parent = _get_parent()
        return [float(x) for x in parent]

    head = max(8.0, min(45.0, 0.032 * budget))
    deadline = t0 + max(0.0, budget) - head
    if deadline <= t0 + 2.0:
        deadline = t0 + max(2.0, 0.92 * budget)

    try:
        rng = np.random.default_rng(int(seed) if np.isfinite(float(seed)) else 42)
    except Exception:
        rng = np.random.default_rng(42)

    parent = _get_parent()
    best_a = _normalize(parent)
    best_exact = _exact_score(best_a)
    best_fft = 2.0 * best_a.size * float(_conv_full(best_a).max())

    if not np.isfinite(best_exact):
        fb = _normalize(np.ones(256, dtype=np.float64))
        fs = _exact_score(fb)
        if np.isfinite(fs):
            best_a = fb
            best_exact = fs
            best_fft = 2.0 * fb.size * float(_conv_full(fb).max())
        else:
            return [1.0]

    phases = _make_phases(budget, best_a.size)
    total_w = 0.0
    for _, w, _ in phases:
        try:
            total_w += float(w)
        except Exception:
            total_w += 1.0
    if total_w <= 0.0:
        total_w = 1.0
    phase_budget = max(0.0, deadline - time.monotonic())

    phase_idx = 0
    while phase_idx < len(phases) and time.monotonic() < deadline - 20.0:
        L, w, do_random = phases[phase_idx]
        phase_idx += 1
        L = int(L)
        if L < 16 or L > 2048:
            continue
        timeL = (float(w) / total_w) * phase_budget
        phase_end = min(time.monotonic() + timeL, deadline - 18.0)
        if phase_end - time.monotonic() < 12.0:
            continue

        init_base = _resample(best_a, L)
        if do_random:
            init = _random_smooth(L, rng, base=init_base)
        else:
            init = init_base

        use_flat = L <= 1536
        try:
            loc_a, loc_M = _local_refine(init, phase_end, rng, use_flat=use_flat)
        except Exception:
            loc_a, loc_M = init, float(_conv_full(init).max())

        if loc_a is not None and loc_a.size > 0 and np.isfinite(loc_M):
            loc_score_fft = 2.0 * L * float(loc_M)
            if np.isfinite(loc_score_fft):
                best_fft = min(best_fft, loc_score_fft)
                if time.monotonic() < deadline - 25.0:
                    sc = _exact_score(loc_a)
                    if np.isfinite(sc) and sc < best_exact - 1e-12:
                        best_a = _normalize(loc_a)
                        best_exact = float(sc)
                        best_fft = min(best_fft, loc_score_fft)

    if time.monotonic() < deadline - 30.0:
        try:
            final_start = time.monotonic()
            final_end = min(deadline - 6.0, final_start + max(25.0, min(90.0, (deadline - final_start) * 0.75)))
            loc_a, loc_M = _local_refine(best_a, final_end, rng, use_flat=best_a.size <= 1536)
            sc = _exact_score(loc_a)
            if np.isfinite(sc) and sc < best_exact - 1e-12:
                best_a = _normalize(loc_a)
                best_exact = float(sc)
        except Exception:
            pass

    if time.monotonic() < deadline - 12.0:
        try:
            fb = _normalize(np.ones(max(16, min(512, best_a.size)), dtype=np.float64))
            fs = _exact_score(fb)
            if not np.isfinite(best_exact) and np.isfinite(fs):
                best_a = fb
                best_exact = fs
        except Exception:
            pass

    try:
        best_a = np.asarray(best_a, dtype=np.float64).ravel()
        best_a = np.where(np.isfinite(best_a), best_a, 0.0)
        best_a = np.maximum(best_a, 0.0)
        best_a = _normalize(best_a)
        fs = _exact_score(best_a)
        if np.isfinite(fs) and fs < best_exact:
            best_exact = fs
    except Exception:
        pass

    if best_a is None or best_a.size == 0 or not np.all(np.isfinite(best_a)):
        fb = _normalize(np.ones(64, dtype=np.float64))
        best_a = fb

    out = np.clip(np.asarray(best_a, dtype=np.float64).ravel(), 0.0, 1000.0)
    out = _normalize(out)
    if out.size == 0 or not np.all(np.isfinite(out)):
        out = np.ones(1, dtype=np.float64)
    return [float(x) for x in out]
```
