    f = bf / bf.sum()
    bsc = _sc(f)
    bf = f.copy()

    bf, bsc = _coord_polish(f, n, p2_end, rng, bsc, bf)

    # === Final validation ===
    fs = np.clip(bf, 0, 1000)
    try:
        fsc = float(evaluate_sequence([float(x) for x in fs]))
    except Exception:
        fsc = math.inf

    if not math.isfinite(fsc):
        fb = np.clip(par, 0, 1000)
        try:
            fb_sc = float(evaluate_sequence([float(x) for x in fb]))
        except Exception:
            fb_sc = math.inf
        if math.isfinite(fb_sc):
            return [float(x) for x in fb]
        return [1.0]

    # === Extra polish if time remains ===
    rem = hard_dl - time.monotonic()
    if rem > 15:
        f2 = fs.copy()
        s3 = f2.sum()
        if s3 > 1e-10:
            f2 /= s3
        csc = _sc(f2)
        bf2 = f2.copy()
        # Second pass of coordinate polish
        bf2, csc = _coord_polish(f2, n, hard_dl - 4.0, rng, csc, bf2)
        if csc < fsc:
            bf = bf2.copy()
            fs = np.clip(bf, 0, 1000)
            try:
                fsc = float(evaluate_sequence([float(x) for x in fs]))
            except Exception:
                fsc = math.inf
            if not math.isfinite(fsc):
                fs = np.clip(bf, 0, 1000)

    return [float(x) for x in fs]
