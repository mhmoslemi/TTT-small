import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog
from scipy.ndimage import gaussian_filter1d


def _conv(a):
    n = len(a)
    nfft = next_fast_len(2 * n - 1)
    f = rfft(a, nfft)
    c = irfft(f * f, nfft)[:2 * n - 1]
    return np.where(np.isfinite(c), c, 0.0)


def _build_A(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for k in range(n):
        A[k, :k + 1] = a[k::-1]
    for k in range(n, m):
        L = m - k
        s = k - n + 1
        A[k, s:s + L] = a[n - 1:n - L - 1:-1]
    return A


def _solve_lp(a, M, rem):
    n = len(a)
    if n < 2 or M <= 0 or not np.isfinite(M):
        return None
    if rem < 2.0:
        return None
    A = _build_A(a)
    A_s = A / M
    c_obj = -np.ones(n)
    tl = min(4.0, rem * 0.25)
    try:
        res = linprog(
            c_obj, A_ub=A_s, b_ub=np.ones(2 * n - 1),
            bounds=[(0.0, None)] * n,
            method='highs',
            options={'time_limit': tl, 'presolve': True, 'dual_feasibility_tolerance': 1e-9}
        )
        if res is not None and res.success:
            b = np.maximum(res.x, 0.0)
            sb = b.sum()
            if sb > 1e-10:
                return b / sb
    except Exception:
        pass
    return None


def _dual_weights(a, c, M):
    n = len(a)
    T = max(M * 0.03, 1e-12)
    z = (c - M) / T
    z = np.clip(z, -60.0, 0.0)
    w = np.exp(z)
    sw = w.sum()
    if sw > 0 and np.isfinite(sw):
        w = w / sw
    else:
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    return w


def _grad_weights(a, w):
    n = len(a)
    need = 2 * n - 1
    if w.size != need:
        w = w[:need] if w.size > need else np.pad(w, (0, need - w.size))
    w = np.where(np.isfinite(w), w, 0.0)
    sw = w.sum()
    if sw <= 0:
        return np.zeros(n)
    w = w / sw
    nfft = next_fast_len(3 * n - 2)
    fa = rfft(a, nfft)
    fw = rfft(w[::-1], nfft)
    cr = irfft(fa * fw, nfft)[:3 * n - 2]
    cr = np.where(np.isfinite(cr), cr, 0.0)
    idx = 2 * n - 2 - np.arange(n)
    g = 2.0 * cr[idx]
    return np.maximum(g, 0.0)
