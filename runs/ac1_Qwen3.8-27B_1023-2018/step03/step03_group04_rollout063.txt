from scipy.optimize import linprog

def _conv_fft(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score_from_conv(a, conv):

    if S < 0.01:
        return math.inf
    M = float(conv.max())
    return 2.0 * n * M / (S * S)

    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A

Hmm, this doesn't directly work because the constraints are coupled.

Actually, the LP approach IS the best response. The LP finds the g that maximizes sum(g) subject to (f*g)_k <= M for all k. This is the best response in the game where one player controls f and the other controls g.

The von Neumann minimax theorem tells us that the value of the game (max over f of min over g, or vice versa) converges. But we need f=g for the score to make sense.

import numpy as np
import time

from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _conv(a):
    n = len(a)
    fs = next_fast_len(2 * n - 1)
    F = rfft(a, fs)
    c = irfft(F * F, fs)[:2 * n - 1]
    return np.maximum(c, 0.0)

def _score1(f):
    n = len(f)
    return 2.0 * n * float(_conv(f).max())

def _conv_mat(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A

def _lp_dir(f, M, conv):
    n = len(f)
    # Active set
    thresh = M * 0.9
    active = np.where(conv >= thresh)[0]
    amax = int(np.argmax(conv))
    if amax not in active:
        active = np.append(active, amax)
    
    A = _conv_mat(f)
    c = -np.ones(n)
    try:
        res = linprog(c, A_ub=A[active, :], b_ub=np.full(len(active), M),
                      bounds=[(0, None)] * n, method='highs',
                      options={"presolve": True})
    except:
        return None
    if res.status != 0:
        return None
    g = res.x
    s = g.sum()
    if s < 1e-10:
        return None
    return g / s
