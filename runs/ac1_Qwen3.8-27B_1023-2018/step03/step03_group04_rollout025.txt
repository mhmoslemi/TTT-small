        thresh = M * 0.9999

        gap = M - M_best  # this is 0 if we're at best
        # Use a fixed fraction instead

        else:
            f_new = f_new / S_new
        
        f = f_new
        iteration += 1

import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(384, dtype=np.float64)

    return p

# Precompute FFT size
_FSIZE_N = {}
def _fsize(n):
    if n not in _FSIZE_N:
        _FSIZE_N[n] = next_fast_len(2*n-1)
    return _FSIZE_N[n]

def _conv(a):
    n = len(a)
    fs = _fsize(n)
    F = rfft(a, fs)
    c = irfft(F * F, fs)[:2*n-1]
    np.maximum(c, 0.0, out=c)
    return c

def _score_arr(a):
    n = len(a)
    S = float(a.sum())
    if S < 0.01:
        return math.inf
    M = float(_conv(a).max())
    return 2.0 * n * M / (S * S)

def _normalize(f):
    S = f.sum()
    if S < 1e-10:
        return np.ones(len(f))
    return f / S

def _lp_direction(f, M, active_frac=0.1):
    """LP to find direction g maximizing sum(g) s.t. (f*g)[k] <= M for active k."""
    n = len(f)
    conv = _conv(f)
    thresh = M * (1.0 - active_frac)
    active = np.where(conv >= thresh)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(conv))])
    
    # Build sparse constraint matrix
    # A[k, j] = f[k-j] for valid (k,j)
    A = np.zeros((len(active), n))
    for idx, k in enumerate(active):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        A[idx, js:je+1] = f[js:je+1][::-1]  # f[k-j] for j in js..je
    
    c = -np.ones(n)
    b_ub = np.full(len(active), M)
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={"time_limit": 5.0})
    except Exception:
        return None, active
    
    if res.status != 0 or res.x is None:
        return None, active
    g = np.asarray(res.x, dtype=np.float64)
    s = g.sum()
    if s < 1e-12:
        return None, active
    return g / s, active

def _optimize(f_init, stop_time, seed, max_iters=50000):
    """Main optimization loop combining LP and subgradient."""
    rng = np.random.default_rng(seed)
    n = len(f_init)
    f = np.clip(f_init.astype(np.float64).copy(), 0, None)
    f = _normalize(f)
    
    best_f = f.copy()
    best_score = _score_arr(f)
    
    t = 0.2  # step size
    stall = 0
    max_stall = 200
    
    for it in range(max_iters):
        if time.monotonic() > stop_time - 2.0:
            break
        
        conv = _conv(f)
        M = float(conv.max())
        score = 2.0 * n * M  # since sum(f) = 1
        
        if score < best_score:
            best_score = score
            best_f = f.copy()
        
        # LP direction
        g, active = _lp_direction(f, M)
        
        if g is not None:
            # Line search
            best_t = t
            best_sc = score
            cand = None
            for tc in [t, t*1.5, t*0.5, t*2.0, t*0.75]:
                tc = max(0.01, min(0.8, tc))
                f_new = np.clip((1-tc)*f + tc*g, 0, None)
                f_new = _normalize(f_new)
                sc = 2.0 * n * _conv(f_new).max()
                if sc < best_sc:
                    best_sc = sc
                    best_t = tc
                    cand = f_new
            if cand is not None:
                f = cand
                if best_sc < best_score:
                    best_score = best_sc
                    best_f = f.copy()
                    stall = 0
                t = min(0.8, t * 1.03)
            else:
                stall += 1
                t = max(0.05, t * 0.7)
        else:
            # Subgradient fallback
            thresh = M * 0.98
            active_idx = np.where(conv >= thresh)[0]
            s = np.zeros(n)
            for k in active_idx:
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                s[js:je+1] += 2.0 * f[js:je+1][::-1]
            s /= max(1, len(active_idx))
            s_norm = np.linalg.norm(s)
            if s_norm > 1e-15:
                eta = 0.01 / s_norm
                f = np.clip(f - eta * s, 0, None)
                f = _normalize(f)
            stall += 3
        
        if stall > max_stall:
            # Restart with perturbation
            f = np.clip(best_f + rng.normal(0, 0.05, n), 0, None)
            f = _normalize(f)
            t = 0.2
            stall = 0
    
    return best_f, best_score

def _polish(f, stop_time, seed=123):
    """Coordinate-wise polishing."""
    rng = np.random.default_rng(seed)
    n = len(f)
    best = np.clip(f.copy(), 0, None)
    best = _normalize(best)
    best_sc = 2.0 * n * _conv(best).max()
    
    no_imp = 0
    max_no_imp = 5
    
    while time.monotonic() < stop_time - 1.0 and no_imp < max_no_imp:
        improved = False
        order = list(range(n))
        rng.shuffle(order)
        
        for i in order:
            if time.monotonic() > stop_time - 1.0:
                break
            orig = best[i]
            best_val = orig
            best_val_sc = best_sc
            for mlt in [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0, 0.0]:
                best[i] = max(0, orig * mlt) if orig > 1e-12 else 0.0
                sc = 2.0 * n * _conv(best).max()  # sum is normalized
                if sc < best_val_sc - 1e-14:
                    best_val = best[i]
                    best_val_sc = sc
            best[i] = best_val
            if best_val_sc < best_sc - 1e-14:
                best_sc = best_val_sc
                improved = True
        
        if not improved:
            no_imp += 1
        else:
            no_imp = 0
    
    return best, best_sc
