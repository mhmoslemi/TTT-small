import time
from scipy.optimize import linprog

    deadline = t0 + budget - 25.0

        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        S = a.sum()
        if S < 0.01:
            return float('inf')
        c = np.convolve(a, a)
        return float(2.0 * a.size * c.max() / (S * S))
    
    # Get parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
    except Exception:
        parent = np.ones(201, dtype=np.float64)
    parent = np.clip(parent, 0.0, 1000.0)
    
    best_a = parent.copy()
    best_sc = _score(best_a)
    
    def _update_best(candidate):
        nonlocal best_a, best_sc
        sc = _score(candidate)
        if sc < best_sc:
            best_sc = sc
            best_a = candidate.copy()
    
    rng = np.random.default_rng(seed)
    
    # Symmetric LP optimization
    def _sym_lp_optimize(a0, stop_time, seed=0):
        """LP best-response optimization for symmetric sequences."""
        n = a0.size
        m = (n + 1) // 2  # number of free variables
        rng_local = np.random.default_rng(seed)
        
        # Start with symmetrized version
        a = 0.5 * (a0 + a0[::-1])
        a = np.maximum(a, 1e-12)
        a = a / a.sum()
        
        # Work with half-sequence
        half = a[:m].copy()
        
        pre = a.copy()  # full symmetric
        conv = np.convolve(pre, pre)
        M = conv.max()
        best_half = half.copy()
        best_M = M
        best_full = pre.copy()

            if n % 2 == 1:
                return np.concatenate([h, h[-2::-1]])
            else:
                return np.concatenate([h, h[::-1]])
        
        def _build_A(a_full):
            """Build constraint matrix A[k,j] for symmetric LP."""
            # n constraints (k=0..n-1), m variables (j=0..m-1)
            # A[k,j] = a_full[k-j] + a_full[k-(n-1-j)] (with bounds checking)
            # Special case: if n odd and j == m-1, second term is same as first
            A = np.zeros((n, m))
            for j in range(m):
                for k in range(n):
                    c = 0.0
                    idx1 = k - j
                    if 0 <= idx1 < n:
                        c += a_full[idx1]
                    if n % 2 == 1:
                        if j == m - 1:
                            pass  # middle, no second occurrence
                        else:
                            pos2 = n - 1 - j
                            idx2 = k - pos2
                            if 0 <= idx2 < n:
                                c += a_full[idx2]
                    else:
                        pos2 = n - 1 - j
                        idx2 = k - pos2
                        if 0 <= idx2 < n:
                            c += a_full[idx2]
                    A[k, j] = c
            return A

            best_alpha = 0.0
            best_val = M_cur

                h = half_cur + alpha * (half_dir - half_cur)
                h = np.maximum(h, 0.0)
                s = h.sum()
                if s < 1e-15:
                    return float('inf')
                h = h / s
                full = _expand(h)
                return float(np.convolve(full, full).max())

            val1 = _eval(1.0)
            if val1 < best_val:

                best_val = val1

                lo, hi = 0.0, 1.0

                    v = _eval(mid)

                        best_alpha = mid
                        hi = mid
                    else:
                        lo = mid
                    if hi - lo < 1e-12:

            else:

                lo, hi = 0.0, 1.0
                # First check mid=0.5
                v = _eval(0.5)

                    best_val = v
                    best_alpha = 0.5
                    lo = 0.5
                else:
                    hi = 0.5
                for _ in range(n_iter):
                    mid = 0.5 * (lo + hi)
                    v = _eval(mid)
                    if v <= best_val:
                        best_val = v
                        best_alpha = mid
                        hi = mid
                    else:
                        lo = mid
                    if hi - lo < 1e-12:
                        break

                h = half_cur + best_alpha * (half_dir - half_cur)
                h = np.maximum(h, 0.0)
                h = h / h.sum()
                return h, best_val
            return None, best_val
        
        stuck = 0
        it = 0
        while time.monotonic() < stop_time and it < 50000:
            it += 1
            if time.monotonic() > stop_time - 0.5:
                break
            
            # Compute current M
            full = _expand(half)
            conv = np.convolve(full, full)
            M = float(conv.max())
            
            if M < best_M * (1 - 1e-14):
                best_M = M
                best_half = half.copy()
                best_full = full.copy()
                stuck = 0
            else:

                half = half / half.sum()

            A = _build_A(full)

            c_obj = -np.ones(m)

                             bounds=[(0.0, None)] * m,
                             method='highs',
                             options={'time_limit': 1.0,
                                      'primal_feasibility_tolerance': 1e-10,
                                      'dual_feasibility_tolerance': 1e-10})
            except Exception:
                continue
            
            if res.status != 0 or res.x is None:
                continue
            
            g_half = np.asarray(res.x, dtype=np.float64)
            g_sum = g_half.sum()
            if g_sum <= 1e-15:

            g_half = g_half / g_sum

            if np.abs(g_half - half).max() < 1e-10:

                if stuck < 50:

                continue

            cand, val = _line_search(half, g_half, M, n_iter=30)
            if cand is not None and val < M * (1 - 1e-13):

                k_star = int(conv.argmax())
                # Gradient of (full*full)_k_star w.r.t. half
                grad = np.zeros(m)
                for j in range(m):
                    idx1 = k_star - j
                    c = 0.0
                    if 0 <= idx1 < n:
                        c += full[idx1]
                    if n % 2 == 1:
                        if j != m - 1:
                            pos2 = n - 1 - j
                            idx2 = k_star - pos2
                            if 0 <= idx2 < n:

                    else:
                        pos2 = n - 1 - j
                        idx2 = k_star - pos2
                        if 0 <= idx2 < n:
                            c += full[idx2]
                    grad[j] = c
                grad -= grad.mean()  # project to sum-0
                gnorm = np.linalg.norm(grad)
                if gnorm > 1e-12:
                    grad /= gnorm
                    for alpha in [0.02, 0.01, 0.005, 0.001]:
                        cand_h = half + alpha * grad
                        cand_h = np.maximum(cand_h, 1e-12)
                        cand_h = cand_h / cand_h.sum()
                        full_c = _expand(cand_h)
                        v = float(np.convolve(full_c, full_c).max())
                        if v < M * (1 - 1e-12):

                            break
                else:
                    half = half * np.exp(rng_local.normal(0, 0.01, m))
                    half = np.maximum(half, 1e-12)
                    half = half / half.sum()

    phase1_end = t0 + min(budget * 0.35, 450.0)
    result = _sym_lp_optimize(parent, stop_time=phase1_end, seed=1)
    _update_best(result)
    
    # Phase 2: Try a few different initializations
    inits = [parent.copy()]
    x = np.arange(n, dtype=np.float64)
    # Sine modulation
    for f in [1.0, 2.0, 3.0]:
        seq = 1.0 + 0.3 * np.cos(2 * np.pi * f * x / n)
        seq = np.maximum(seq, 0.1)
        inits.append(0.5 * parent + 0.5 * seq / seq.sum() * parent.sum())

If we write $a^T B a = \sum_{ij} B_{ij} a_i a_j$, then:
- $(a*a)_k$ has cross-terms $a_i a_j$ for $i \neq j$ (when $i + j = k$) and square terms $a_i^2$ (when $2i = k$).

                half = np.maximum(half, 1e-12)
                half /= half.sum()
                stuck = 0
                if (stuck - 120) // 120 > 20:
                    break
                continue

            g = _lp_solve(full, M * (1.0 + 1e-9))
            if g is None:
                continue

            if np.abs(g - half).max() < 5e-11:
                # Near-fixed-point: try gradient
                ks = int(np.argmax(conv))
                gr = np.zeros(m)
                for j in range(m):
                    c = 0.0
                    i1 = ks - j
                    if 0 <= i1 < n:
                        c += full[i1]
                    if n % 2 == 1:
                        if j != m - 1:
                            i2 = ks - (n - 1 - j)
                            if 0 <= i2 < n:
                                c += full[i2]
                    else:
                        i2 = ks - (n - 1 - j)
                        if 0 <= i2 < n:
                            c += full[i2]
                    gr[j] = c
                gr -= gr.mean()
                gn = np.linalg.norm(gr)
                if gn > 1e-13:
                    gr /= gn
                    for al in [0.005, 0.002, 0.001, 0.0005]:
                        ch = half + al * gr
                        ch = np.maximum(ch, 1e-12)
                        ch /= ch.sum()
                        if np.convolve(_expand(ch), _expand(ch)).max() < M * (1 - 1e-14):
                            half = ch
                            break
                stuck += 1
                if stuck > 300:
                    break
                continue

            # Line search: geometric + bisection
            best_al = 0.0
            best_v = M
            for al in [1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.1, 0.05]:
                ch = half + al * (g - half)
                ch = np.maximum(ch, 0.0)
                cs = ch.sum()
                if cs < 1e-15:
                    continue
                ch /= cs
                v = float(np.convolve(_expand(ch), _expand(ch)).max())
                if v < best_v:
                    best_v = v
                    best_al = al
            if best_al > 0:
                lo = max(0.0, best_al - 0.08)
                hi = min(1.0, best_al + 0.08)
                for _ in range(20):
                    mid = 0.5 * (lo + hi)
                    ch = half + mid * (g - half)
                    ch = np.maximum(ch, 0.0)
                    if ch.sum() < 1e-15:
                        hi = mid
                        continue
                    ch /= ch.sum()
                    v = float(np.convolve(_expand(ch), _expand(ch)).max())
                    if v <= best_v:
                        best_v = v
                        best_al = mid
                        hi = mid
                    else:
                        lo = mid
                    if hi - lo < 1e-12:
                        break
                half = half + best_al * (g - half)
                half = np.maximum(half, 1e-12)
                half /= half.sum()

        return _expand(best_h)

    # ===== Asymmetric LP optimizer =====
    def _asym_optimize(a0, stop_t, s=0):
        n = a0.size
        rng_l = np.random.default_rng(s)

        a = np.maximum(a0.astype(np.float64), 1e-12)
        a /= a.sum()

        # Precompute structure for (2n-1) x n matrix
        k_idx = np.arange(2 * n - 1)[:, None].astype(np.intp)
        j_idx = np.arange(n)[None, :].astype(np.intp)
        idx_a = k_idx - j_idx
        mask_a = (idx_a >= 0) & (idx_a < n)
        safe_a = np.clip(idx_a, 0, n - 1)
        c_obj_a = -np.ones(n)

        def _build_A(full):
            return full[safe_a] * mask_a.astype(np.float64)

        def _lp_solve(full, M_t):
            A = _build_A(full)
            try:
                res = linprog(c_obj_a, A_ub=A, b_ub=np.full(2 * n - 1, M_t),
                             bounds=[(0.0, None)] * n, method='highs',
                             options={'time_limit': 1.5,
                                      'primal_feasibility_tolerance': 1e-12,
                                      'dual_feasibility_tolerance': 1e-12})
            except Exception:
                return None
            if res.status != 0 or res.x is None:
                return None
            g = np.asarray(res.x, dtype=np.float64)
            gs = g.sum()
            if gs < 1e-15 or not np.all(np.isfinite(g)):
                return None
            return g / gs

        best_a_local = a.copy()
        best_M = np.inf
        stuck = 0
        it = 0

        while it < 100000 and time.monotonic() < stop_t - 0.5:
            it += 1
            conv = np.convolve(a, a)
            M = float(conv.max())

            if M < best_M:
                best_M = M
                best_a_local = a.copy()
                stuck = 0
            else:
                stuck += 1

            if stuck > 100:
                sc = 0.02 + 0.01 * min(stuck // 100, 10)
                a = a * np.exp(rng_l.normal(0, sc, n))
                a = np.maximum(a, 1e-12)
                a /= a.sum()
                stuck = 0
                continue

            g = _lp_solve(a, M * (1.0 + 1e-9))
            if g is None:
                continue

            if np.abs(g - a).max() < 1e-10:
                # Gradient step
                ks = int(np.argmax(conv))
                gr = np.zeros(n)
                for j in range(n):
                    i1 = ks - j
                    if 0 <= i1 < n:
                        gr[j] += a[i1]
                gr -= gr.mean()
                gn = np.linalg.norm(gr)
                if gn > 1e-13:
                    gr /= gn
                    for al in [0.005, 0.002, 0.001]:
                        ch = a + al * gr
                        ch = np.maximum(ch, 1e-12)
                        ch /= ch.sum()
                        if np.convolve(ch, ch).max() < M * (1 - 1e-14):
                            a = ch
                            break
                if stuck > 250:
                    break
                continue

            best_al = 0.0
            best_v = M
            for al in [1.0, 0.6, 0.4, 0.25, 0.15, 0.1, 0.05]:
                ch = a + al * (g - a)
                ch = np.maximum(ch, 0.0)
                if ch.sum() < 1e-15:
                    continue
                ch /= ch.sum()
                v = float(np.convolve(ch, ch).max())
                if v < best_v:
                    best_v = v
                    best_al = al
            if best_al > 0:
                lo = max(0.0, best_al - 0.08)
                hi = min(1.0, best_al + 0.08)
                for _ in range(18):
                    mid = 0.5 * (lo + hi)
                    ch = a + mid * (g - a)
                    ch = np.maximum(ch, 0.0)
                    if ch.sum() < 1e-15:
                        hi = mid
                        continue
                    ch /= ch.sum()
                    v = float(np.convolve(ch, ch).max())
                    if v <= best_v:
                        best_v = v
                        best_al = mid
                        hi = mid
                    else:
                        lo = mid
                    if hi - lo < 1e-12:
                        break
                a = a + best_al * (g - a)
                a = np.maximum(a, 1e-12)
                a /= a.sum()

        return best_a_local

    # ===== Main search phases =====

    # Phase 1: Symmetric LP from parent (primary)
    t1 = t0 + min(budget_s * 0.38, 490.0)
    _update(_sym_optimize(p, t1, s=1))

    # Phase 2: Asymmetric LP from best symmetric result
    t2 = t0 + min(budget_s * 0.62, 806.0)
    if time.monotonic() < t2 - 20:
        _update(_asym_optimize(best_a, t2, s=11))

    # Phase 3: Alternative initializations (symmetric)
    t3 = t0 + min(budget_s * 0.78, 1014.0)
    x = np.arange(n0, dtype=np.float64)
    p_norm = p / p.sum()
    inits = []
    # Cosine-modulated
    for f in [1, 2, 3, 5]:
        s = 1.0 + 0.25 * np.cos(2 * np.pi * f * x / n0)
        s = np.maximum(s, 0.05)
        inits.append(0.6 * p_norm + 0.4 * s / s.sum())
    # Gaussian
    for w in [0.2, 0.35, 0.5]:
        s = np.exp(-0.5 * ((x - n0 / 2) / (w * n0 / 2)) ** 2)
        inits.append(0.5 * p_norm + 0.5 * s / s.sum())
    # Power-law
    for pw in [0.3, 0.7, 1.5]:
        s = np.maximum(1.0 - x / (n0 - 1), 1e-6) ** pw
        inits.append(0.5 * p_norm + 0.5 * s / s.sum())

    for i, ini in enumerate(inits):
        if time.monotonic() > t3 - 25:
            break
        alloc = min(55.0, t3 - time.monotonic() - 3)
        if alloc < 10:
            break
        _update(_sym_optimize(ini, time.monotonic() + alloc, s=200 + i))

    # Phase 4: Final polish on absolute best
    if time.monotonic() < deadline - 15:
        _update(_sym_optimize(best_a, deadline - 5, s=999))

    # Phase 5: Tiny asymmetric polish
    if time.monotonic() < deadline - 5:
        _update(_asym_optimize(best_a, deadline, s=9999))

    # Final validation
    try:
        fs = float(evaluate_sequence([float(v) for v in best_a]))
    except Exception:
        fs = float('inf')

    if not np.isfinite(fs):
        return [1.0] * 32

    return [float(v) for v in best_a]
