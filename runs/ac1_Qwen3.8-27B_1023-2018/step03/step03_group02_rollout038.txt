def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

    b = _conv_full(a)

    if not math.isfinite(m) or m < 0:

    return 2.0 * n * m / (s * s)

def _build_conv_matrix(f, n):
    """Build the matrix A such that A @ g = conv(f, g) for g of length n."""
    # (f*g)_k = sum_j f[k-j] * g[j] for valid indices
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):

        start = j
        end = min(j + n, 2 * n - 1)
        A[start:end, j] = f[:end - start]
    return A

def _lp_direction(f, n, M, time_limit=5.0):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0"""
    A = _build_conv_matrix(f, n)

    try:

            c, A_ub=A, b_ub=b_ub,

        return None

        return None

    g_sum = float(g.sum())
    if g_sum < 1e-12:
        return None
    return g / g_sum  # Normalize to sum = 1

def _conv_full(a):
    n = len(a)

    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

    return A
