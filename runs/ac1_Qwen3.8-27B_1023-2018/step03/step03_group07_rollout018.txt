```python
def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1} in Euclidean norm."""
    n = v.size
    # Binary search for tau
    # sum(max(v_i - tau, 0)) = 1
    # Equivalently: sort v in decreasing order, find where cumulative sum crosses 1
    u = np.sort(v)[::-1]  # decreasing
    cumsum = np.cumsum(u)
    # Find rho: largest index where u[rho] > (cumsum[rho+1] - 1) / rho... 
    # Actually, the standard algorithm:
    # rho = max{i : u_i > (1 - sum_{j>i} 0) / i}... 
    
    # Simpler: binary search for tau
    lo = u[-1]  # minimum of v
    hi = u[0]  # maximum of v
    for _ in range(60):  # enough for double precision
        mid = (lo + hi) / 2
        s = np.sum(np.maximum(v - mid, 0))
        if s > 1:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2
    return np.maximum(v - tau, 0)
```

Actually, there's a closed-form algorithm that's O(n log n) due to sorting:

```python
def _project_simplex(v):
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u - (cssv - 1) / rho
    rho = np.max(rho[cond > 0])
    theta = (cssv[rho-1] - 1) / rho
    return np.maximum(v - theta, 0)
```

```python
def _smooth_grad(a, t=50):
    n = a.size
    c = np.convolve(a, a)
    # Softmax weights
    c_max = c.max()
    weights = np.exp(t * (c - c_max))
    weights /= weights.sum()
    grad = np.zeros(n)
    for k in range(len(c)):
        if weights[k] < 1e-15:
            continue
        m_lo = max(0, k - n + 1)
        m_hi = min(n - 1, k)
        if m_lo <= m_hi:
            m_vals = np.arange(m_lo, m_hi + 1)
            idx = k - m_vals
            grad[m_lo:m_hi+1] += 2.0 * weights[k] * a[idx]
    return grad, c, float(c.max())
```
