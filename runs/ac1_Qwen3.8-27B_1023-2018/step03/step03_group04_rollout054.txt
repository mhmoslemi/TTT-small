                    oj = f[j]
                    for mj in (0.5, 1.5, 0.0, 2.0):
                        f[j] = oj * mj if oj > 1e-12 else (0.005 if mj > 0 else 0.0)
                        if f.sum() < 0.01:
                            f[j] = oj
                            continue
                        sc = _score(f)
                        if sc < best_sc - 1e-14:
                            best_sc = sc
                    f[j] = oj if not improved else f[j]

        no_imp = 0 if improved else no_imp + 1

    return f, best_sc


def _subset_fw(f0, stop_time, seed=42):
    """FW optimization on a subset of coordinates (freeze others at 0 or small value)."""
    n = len(f0)
    f = np.clip(f0.copy(), 0, None)
    S = f.sum()
    if S < 1e-10:
        return f, math.inf
    f = f / S
    return _fw_optimize(f, stop_time, seed)


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 25.0  # 25s headroom

    parent = _get_parent()
    n = len(parent)

    # Fallback
    best_seq = parent.copy()
    try:
        best_score = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        best_score = math.inf
        best_seq = np.ones(384)
        best_score = float(evaluate_sequence([float(x) for x in best_seq]))

    def _track(f, label=""):
        nonlocal best_seq, best_score
        f = np.clip(np.asarray(f, dtype=np.float64), 0, 1000)
        if f.size == 0 or f.sum() < 0.01 or not np.all(np.isfinite(f)):
            return
        try:
            sc = float(evaluate_sequence([float(x) for x in f]))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score:
            best_seq = f.copy()
            best_score = sc

    # ===== Phase 1: Main FW from parent (50% budget) =====
    phase1_end = t0 + budget * 0.50
    try:
        res, sc = _fw_optimize(parent, phase1_end, seed)
        _track(res, "parent_fw")
    except Exception:
        pass

    # ===== Phase 2: FW from diverse inits (20% budget) =====
    phase2_end = t0 + budget * 0.70
    i_arr = np.arange(n, dtype=np.float64)
    rng = np.random.default_rng(seed + 1)
    diverse_inits = []

    # Spaced sequences (different gaps)
    for gap in (2, 3, 5):
        sp = np.zeros(n)
        sp[::gap] = 1.0
        diverse_inits.append(sp)

    # Cosine window
    diverse_inits.append(0.5 + 0.5 * np.cos(2 * np.pi * i_arr / n))

    # Bimodal Gaussian
    diverse_inits.append(
        np.exp(-((i_arr - 0.2 * n) / (0.12 * n)) ** 2) +
        np.exp(-((i_arr - 0.8 * n) / (0.12 * n)) ** 2)
    )

    # Random
    diverse_inits.append(np.abs(rng.normal(0, 1, n)) + 0.01)

    # Parent perturbed
    diverse_inits.append(np.clip(parent * rng.uniform(0.5, 1.5, n), 0, None))

    # Arithmetic progression (triangular-ish)
    diverse_inits.append(1.0 + 0.5 * i_arr / n)

    n_div = len(diverse_inits)
    time_per_div = max(2.0, (phase2_end - time.monotonic()) / max(1, n_div))

    for idx, dinit in enumerate(diverse_inits):
        if time.monotonic() > phase2_end - 3:
            break
        stop = min(phase2_end - 2, time.monotonic() + time_per_div)
        if stop <= time.monotonic():
            break
        try:
            res, sc = _fw_optimize(dinit, stop, seed + 5000 + idx * 7)
            _track(res, f"div{idx}")
        except Exception:
            pass

    # ===== Phase 3: Continue FW on best (10% budget) =====
    phase3_end = t0 + budget * 0.80
    if time.monotonic() < phase3_end - 5:
        try:
            res, sc = _fw_optimize(best_seq, phase3_end, seed + 99999)
            _track(res, "continue")
        except Exception:
            pass

    # ===== Phase 4: Coordinate descent polish (15% budget) =====
    phase4_end = t0 + budget * 0.95
    if time.monotonic() < phase4_end - 5:
        try:
            res, sc = _coord_polish(best_seq, phase4_end, seed + 222)
            _track(res, "polish1")
        except Exception:
            pass

    # Second round of FW + polish if time allows
    if time.monotonic() < deadline - 10:
        half = (deadline - 10 - time.monotonic()) / 2
        if half > 5:
            try:
                res, sc = _fw_optimize(best_seq, time.monotonic() + half, seed + 333)
                _track(res, "fw2")
            except Exception:
                pass
        if time.monotonic() < deadline - 5:
            try:
                res, sc = _coord_polish(best_seq, deadline - 3, seed + 444)
                _track(res, "polish2")
            except Exception:
                pass

    # Final validation
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        # Ultimate fallback
        fb = np.ones(384)
        try:
            fb_sc = float(evaluate_sequence([float(x) for x in fb]))
        except Exception:
            fb_sc = math.inf
        if math.isfinite(fb_sc):
            return [float(x) for x in fb]
        return [1.0]

    return [float(x) for x in best_seq]
