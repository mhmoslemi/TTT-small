    """Convolution matrix: (A @ g)[k] = (f * g)[k] for linear conv."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)

    A = _build_A(f, n)

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': time_limit})

        return None

        return None
    g = np.asarray(res.x, dtype=np.float64)

        # Cross-correlation: sum_i p_i * a_{i-j} = conv of p with reversed a
        a_rev = a[::-1]
        Fa_rev = rfft(a_rev)
        # Actually, let me just compute it directly for n=384 (it's fast enough)
        grad_a = np.zeros(n)
        for j in range(n):
            # sum_{i=j}^{min(j+n-1, 2n-2)} p[i] * a[i-j]
            i_min = j
            i_max = min(j + n - 1, 2 * n - 2)
            grad_a[j] = 2.0 * np.dot(p[i_min:i_max+1], a[0:i_max-i_min+1])
        
        # grad_w_j = 2n * (grad_a_j * a_j - a_j * dot(grad_a, a))
        dot_ga_a = np.dot(grad_a, a)
        grad_w = 2.0 * n * (grad_a * a - a * dot_ga_a)
        
        # Clamp gradient
        grad_w = np.clip(grad_w, -1e6, 1e6)
        
        return val, grad_w
    
    try:
        res = minimize(obj_grad, x0, method='L-BFGS-B', 
                      options={'maxiter': 200, 'maxfun': 500, 'gtol': 1e-8})
    except Exception:
        return None, None
    
    if res.x is None:
        return None, None
    
    a = np.exp(res.x)
    S = a.sum()
    if S < 1e-10:
        return None, None
    a = a / S
    a = np.clip(a, 0, 1000)
    sc = _score(a)
    return a, sc

def _polish(a, stop_time, seed=99):
    n = len(a)
    best = np.clip(np.asarray(a, dtype=np.float64).copy(), 0, None)
    fs = best.sum()
    if fs < 1e-10:
        best = np.ones(n) / n
    else:
        best = best / fs
    best_sc = _score(best)
    rng = np.random.default_rng(seed)
    no_imp_rounds = 0
    max_rounds = 10
    
    while time.monotonic() < stop_time - 1.0 and no_imp_rounds < max_rounds:
        improved = False
        
        # Single coordinate descent
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 1.0:
                break
            orig = best[i]
            best_i_sc = best_sc
            best_i_val = orig
            if orig > 1e-8:
                mults = (0.2, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0, 0.0)
            else:
                mults = (1e-4, 1e-3, 0.01, 0.05, 0.1, 0.5)
            for mlt in mults:
                best[i] = orig * mlt if orig > 1e-8 else (mlt / n)
                sc = _score(best)
                if sc < best_i_sc - 1e-14:
                    best_i_sc = sc
                    best_i_val = best[i]
            best[i] = best_i_val
            if best_i_sc < best_sc - 1e-14:
                best_sc = best_i_sc
                improved = True
        
        # Pair moves
        n_pairs = min(80, n)
        for _ in range(n_pairs):
            if time.monotonic() > stop_time - 1.0:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            total = oi + oj
            best_pair_sc = best_sc
            for ratio in (0.1, 0.3, 0.5, 0.7, 0.9, 0.99):
                best[i] = ratio * total
                best[j] = (1.0 - ratio) * total
                sc = _score(best)
                if sc < best_pair_sc - 1e-14:
                    best_pair_sc = sc
            if best_pair_sc < best_sc - 1e-14:
                best_sc = best_pair_sc
                improved = True
            else:
                best[i], best[j] = oi, oj
        
        if improved:
            no_imp_rounds = 0
        else:
            no_imp_rounds += 1
    
    return best, best_sc

def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard_deadline = t0 + budget - 10.0  # leave 10s buffer
    
    parent = _get_parent()
    n_parent = len(parent)
    
    # Initialize best
    best_seq = parent.copy()
    best_score = _score(best_seq)
    if not math.isfinite(best_score):
        best_seq = np.ones(n_parent) / n_parent
        best_score = _score(best_seq)
    
    n = n_parent  # primary length
    i_arr = np.arange(n, dtype=np.float64)
    
    # Generate diverse initializations
    rng_init = np.random.default_rng(seed)
    inits = [parent.copy()]
    
    # Cosine taper
    taper = 0.5 + 0.5 * np.cos(np.pi * i_arr / (n - 1))
    inits.append(taper)
    
    # Raised cosine
    inits.append(np.sin(np.pi * i_arr / (n - 1)) ** 2 + 0.01)
    
    # Multi-frequency
    sig = np.ones(n) * 0.2
    for freq in (1, 2, 3, 5, 8, 13):
        phase = rng_init.uniform(0, 2 * math.pi)
        sig += 0.3 * np.cos(2 * math.pi * freq * i_arr / n + phase)
    inits.append(np.abs(sig) + 0.005)
    
    # Smoothed random
    base = rng_init.standard_normal(n)
    kern = np.exp(-0.5 * (np.arange(-6, 7) / 2.5) ** 2)
    kern /= kern.sum()
    smoothed = np.convolve(base, kern, 'same')
    inits.append(np.abs(smoothed) + 0.003)
    
    # Parent with noise
    noise_scale = 0.1 * max(1.0, np.std(parent))
    inits.append(np.abs(parent + rng_init.normal(0, noise_scale, n)) + 0.001)
    
    # Alternating pattern
    alt = np.ones(n)
    alt[::2] *= 1.5
    alt[1::2] *= 0.7
    inits.append(alt)
    
    # Budget allocation:
    # Phase 1: LP optimization on多样 inits (60% of time)
    # Phase 2: Smooth gradient + other lengths (20% of time)
    # Phase 3: Polish (20% of time)
    
    phase1_end = t0 + budget * 0.55
    phase2_end = t0 + budget * 0.75
    phase3_end = hard_deadline - 5.0
    
    # Phase 1: LP optimization
    time_per_init = max(10.0, (phase1_end - t0) / len(inits))
    
    for idx, init in enumerate(inits):
        if time.monotonic() > phase1_end - 5.0:
            break
        if best_score < 1.503:
            break
        stop = min(phase1_end - 3.0, time.monotonic() + time_per_init)
