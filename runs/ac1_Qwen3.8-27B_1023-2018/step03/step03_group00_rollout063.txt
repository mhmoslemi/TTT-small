    B = max(10.0, float(budget_s))
    DL = t0 + B - 20.0

    par = _get_parent()
    n0 = len(par)

    best = par.copy().astype(np.float64)
    best_sc = _sc(best)

    def _upd(a, sc):
        nonlocal best, best_sc
        if (a is not None and len(a) > 0 and np.all(np.isfinite(a))
                and math.isfinite(sc) and sc < best_sc - 1e-13):
            best = a.copy()
            best_sc = sc

    def _try(f0, n, stop, sd):
        try:
            r, _ = _opt(f0, n, stop, sd)
        except Exception:
            return
        if r is None or len(r) != n:
            return
        r = np.clip(r, 0.0, 1000.0)
        if r.sum() < 0.01:
            return
        es = _sc(r)
        if math.isfinite(es):
            _upd(r, es)

    def _try_pol(f0, stop, sd):
        try:
            r, _ = _pol(f0, stop, sd)
        except Exception:
            return
        if r is None or len(r) == 0:
            return
        r = np.clip(r, 0.0, 1000.0)
        es = _sc(r)
        if math.isfinite(es):
            _upd(r, es)

    rng = np.random.default_rng(seed)
    ii = np.arange(n0, dtype=np.float64)

    # Phase 1: LP optimisation on parent length with multiple initialisations
    p1 = min(t0 + B * 0.44, DL - 260.0)
    inits = [
        par.copy(),
        0.5 * (1.0 + np.cos(2.0 * np.pi * ii / n0)),
        np.abs(par + 0.1 * max(1.0, float(par.mean())) * rng.standard_normal(n0)) + 0.01,
        np.exp(-0.5 * ((ii - (n0 - 1) / 2.0) / (n0 / 4.0)) ** 2),
    ]
    tp = max(12.0, (p1 - t0) / len(inits))
    for ix, ini in enumerate(inits):
        if time.monotonic() > p1 - 10.0:
            break
        stt = min(p1 - 5.0, time.monotonic() + tp)
        _try(ini, n0, stt, seed + 200 * (ix + 1))

    # Phase 2: warm restart on current best
    p2 = min(t0 + B * 0.58, DL - 180.0)
    if time.monotonic() < p2 - 15.0:
        _try(best, len(best), p2 - 5.0, seed + 77777)

    # Phase 3: explore other sequence lengths
    p3 = min(t0 + B * 0.74, DL - 100.0)
    for L in (384, 400, 432, 448, 480, 512, 544, 576):
        if time.monotonic() > p3 - 8.0:
            break
        if L == len(best):
            continue
        il = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, n0), par)
        if il.sum() < 1e-10:
            il = np.ones(L)
        _try(il, L, min(p3 - 5.0, time.monotonic() + 15.0), seed + 30000 + L)

    # Phase 4: second LP pass on best
    p4 = min(t0 + B * 0.86, DL - 40.0)
    if time.monotonic() < p4 - 15.0:
        _try(best, len(best), p4 - 5.0, seed + 123456)

    # Phase 5: coordinate-descent polish
    p5 = DL - 10.0
    if time.monotonic() < p5 - 10.0:
        _try_pol(best, p5, 4242)

    # Final validation with the exact evaluator
    try:
        fs = float(evaluate_sequence([float(x) for x in best]))
    except Exception:
        fs = math.inf

    if not math.isfinite(fs):
        return [1.0]

    return [float(x) for x in best]
