Actually, let me reconsider the gradient computation. I want to minimize $f(a) = \max_k (a*a)_k$ subject to $a \ge 0, \sum a = 1$.

The softmax-smoothed version: $\tilde{f}(a) = \frac{1}{\lambda} \log \sum_k e^{\lambda (a*a)_k}$

The gradient:
$\nabla_i \tilde{f} = \frac{1}{\lambda W} \sum_k \lambda e^{\lambda b_k} \cdot \frac{\partial b_k}{\partial a_i}$
$= \frac{1}{W} \sum_k w_k \cdot 2 a_{k-i} \mathbf{1}[0 \le k-i < n]$

where $w_k = e^{\lambda b_k}$ and $W = \sum_k w_k$.

For $\lambda \to \infty$, this converges to the subgradient at the maximizing lag(s).

The gradient step: $a \leftarrow a - \eta \nabla \tilde{f}$, then project onto $\{a \ge 0, \sum a = 1\}$.

For the projection onto the probability simplex, I'll use the standard algorithm (sort-based).

Let me also think about the step size. The objective is $O(1/n)$ (since $(a*a)_k \sim 1/n$ for uniform $a$). The gradient is $O(1/n)$ as well. So the step size should be $O(n)$ to make progress.

$b_k = \sum_j \frac{1}{n} \frac{1}{n} = \frac{\min(k+1, n, 2n-1-k)}{n^2}$

$\nabla_i \tilde{f} = \frac{2}{W} \sum_k w_k a_{k-i} = \frac{2}{W} \sum_k w_k \frac{1}{n}$

For large $\lambda$, $w_k$ is concentrated at $k = n-1$ (the peak), so:
$\nabla_i \tilde{f} \approx \frac{2}{n} a_{n-1-i} = \frac{2}{n^2}$

import math

    n = len(a)
    L = next_fast_len(2 * n - 1)
    F = rfft(a, L)
    c = irfft(F * F, L)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)

    s = float(a.sum())
    if s < 0.01:

    m = float(b.max())

def _build_A(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        end = min(j + n, 2 * n)
        A[j:end, j] = f[:end - j]
    return A

def _lp_dir(f, M, tl=15.0):
    n = len(f)
    A = _build_A(f)
    try:
        r = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M),
                    bounds=[(0, None)]*n, method='highs',
                    options={'presolve': True, 'time_limit': tl})
    except Exception:

    if r is None or r.x is None or r.status not in (0, 3):
        return None
    g = np.maximum(np.asarray(r.x, dtype=np.float64), 0)
    s = g.sum()
    return g / s if s > 1e-14 else None

def _proj(v, s=1.0):
    """Project onto {a>=0, sum=s}."""
    n = len(v)
    u = np.sort(v)[::-1]
    c = np.cumsum(u)
    idx = np.arange(1, n + 1)
    ok = u * idx > (c - s)
    if not ok.any():
        return np.full(n, s / n)
    r = np.max(np.nonzero(ok)[0])
    return np.maximum(v - (c[r] - s) / (r + 1), 0)

def _get_parent():

        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(384, dtype=np.float64)

def _conv(a):
    n = len(a)
    L = next_fast_len(2 * n - 1)
    F = rfft(a, L)
    c = irfft(F * F, L)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)

def _build_A(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        end = min(j + n, 2 * n)
        A[j:end, j] = f[:end - j]
    return A

def _lp_solve(f, M, tl=15.0):
    n = len(f)
    A = _build_A(f)
    try:
        r = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M),
                    bounds=[(0.0, None)]*n, method='highs',
                    options={'presolve': True, 'time_limit': tl})
    except Exception:
        return None
    if r is None or r.x is None:
        return None
    if r.status not in (0, 3):
        return None
    g = np.maximum(np.asarray(r.x, dtype=np.float64), 0.0)
    s = g.sum()
    if s < 1e-14:
        return None
    return g / s

def _proj_simplex(v, target=1.0):
    n = len(v)
    u = np.sort(v)[::-1]
    c = np.cumsum(u)
    idx = np.arange(1, n+1)
    cond = u * idx > (c - target)
    if not cond.any():
        return np.full(n, target / n)
    rho = np.max(np.nonzero(cond)[0])
    theta = (c[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _smooth_grad(a, lam):
    n = len(a)
    b = _conv(a)
    bm = b.max()
    w = np.exp(lam * (b - bm))
    W = w.sum()
    if W < 1e-300:
        return np.zeros(n)
    w /= W
    i = np.arange(n)
    ij = i[:, None] + i[None, :]
    Wmat = np.zeros((n, n))
    valid = ij < 2*n - 1
    Wmat[valid] = w[ij[valid]]
    return 2.0 * (Wmat @ a)
