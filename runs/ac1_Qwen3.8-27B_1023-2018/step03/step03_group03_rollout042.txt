    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j : j + n, j] = f
    return A


def _lp_direction(f, M, time_limit=4.0):
    n = len(f)
    A = _build_A(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(
            c, A_ub=A, b_ub=b_ub,
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True, "time_limit": time_limit},
        )
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0, None)
    gs = float(g.sum())
    if gs < 1e-15:
        return None
    return g / gs


def _smooth_max_and_grad(f, alpha):
    n = len(f)
    b = _conv_full(f)
    b_max = float(b.max())
    ab = alpha * (b - b_max)
    ab = np.clip(ab, -60, 60)
    e = np.exp(ab)
    w_sum = e.sum()
    w = e / w_sum
    smooth_val = b_max + math.log(w_sum) / alpha
    grad = 2.0 * np.correlate(w, f, mode="valid")
    return smooth_val, grad, w


def _lp_optimize(f0, stop_time, seed, max_iters=5000):
    """Main LP-based optimization loop."""
    n = len(f0)
    rng = np.random.default_rng(seed)
    f = np.clip(f0.astype(np.float64).copy(), 0, None)
    f = _normalize(f)

    best_f = f.copy()
    best_score = _score(f)
    conv = _conv_full(f)
    M = float(conv.max())

    t = 0.15
    t_min = 0.02
    t_max = 0.7
    stall = 0
    max_stall = 80
    it = 0

    while time.monotonic() < stop_time - 2.0 and it < max_iters:
        g_dir = _lp_direction(f, M, time_limit=3.0)

        if g_dir is None:
            stall += 3
            if stall > max_stall:
                # Perturb to escape
                f = np.clip(f + rng.normal(0, 0.03, n), 0, None)
                f = _normalize(f)
                t = 0.15
                stall = 0
                continue
            it += 1
            continue

        dist = float(np.linalg.norm(g_dir - f))
        if dist < 1e-8:
            stall += 5
            if stall > max_stall:
                f = np.clip(f * (1 + rng.normal(0, 0.05, n)), 0, None)
                f = _normalize(f)
                t = 0.2
                stall = 0
                continue
            it += 1
            continue

        # Bisection line search
        improved = False
        lo, hi = 0.0, 1.0
        f_new = f.copy()

        for _bis in range(18):
            t_mid = (lo + hi) / 2.0
            candidate = (1.0 - t_mid) * f + t_mid * g_dir
            candidate = np.clip(candidate, 0, None)
            cs = float(candidate.sum())
            if cs < 1e-10:
                hi = t_mid
                continue
            sc = _score(candidate)
            if sc < best_score * (1.0 - 1e-6):
                f_new = candidate.copy()
                hi = t_mid
                best_score = sc
                improved = True
            else:
                lo = t_mid

        if improved:
            f = _normalize(f_new)
            best_f = f.copy()
            conv = _conv_full(f)
            M = float(conv.max())
            t = 0.15
            stall = 0
        else:
            stall += 1
            if stall >= max_stall:
                # Try gradient escape
                try:
                    _, grad, _ = _smooth_max_and_grad(f, alpha=80.0)
                    step = 0.01
                    f_escape = np.clip(f - step * grad, 0, None)
                    f_escape = _normalize(f_escape)
                    sc_esc = _score(f_escape)
                    if sc_esc < best_score:
                        f = f_escape
                        best_f = f.copy()
                        best_score = sc_esc
                        conv = _conv_full(f)
                        M = float(conv.max())
                    else:
                        f = np.clip(f + rng.normal(0, 0.04, n), 0, None)
                        f = _normalize(f)
                except Exception:
                    f = np.clip(f + rng.normal(0, 0.05, n), 0, None)
                    f = _normalize(f)
                t = 0.2
                stall = 0

        it += 1

    return best_f, best_score


def _gradient_optimize(f0, stop_time, seed, max_iters=2000):
    """Smooth-max gradient descent with adaptive alpha and step size."""
    n = len(f0)
    rng = np.random.default_rng(seed)
    f = np.clip(f0.astype(np.float64).copy(), 0, None)
    f = _normalize(f)
    best_f = f.copy()
    best_score = _score(f)

    alpha = 50.0
    alpha_max = 200.0
    lr = 0.005
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > stop_time - 2.0:
            break

        smooth_val, grad, w = _smooth_max_and_grad(f, alpha)
        # Full gradient of score = 2n * smooth_val / S^2
        S = float(f.sum())
        grad_score = (2.0 * n / (S * S)) * grad - (4.0 * n * smooth_val / (S * S * S)) * np.ones(n)

        # Projected gradient step (project onto f >= 0)
        f_new = np.clip(f - lr * grad_score, 0, None)
        f_new = _normalize(f_new)
        sc = _score(f_new)

        if sc < best_score - 1e-12:
            f = f_new
            best_f = f.copy()
            best_score = sc
            no_improve = 0
            lr = min(lr * 1.1, 0.02)
        else:
            no_improve += 1
            lr = max(lr * 0.7, 1e-5)
            if no_improve > 50:
                alpha = min(alpha * 1.5, alpha_max)
                no_improve = 0
            if no_improve > 200:
                # Add noise
                f = np.clip(f + rng.normal(0, 0.02 * lr, n), 0, None)
                f = _normalize(f)
                lr = 0.005
                no_improve = 0

    return best_f, best_score


def _coord_polish(f0, stop_time, max_sweeps=80, seed=777):
    """Coordinate descent with golden section on each coordinate."""
    n = len(f0)
    f = np.clip(f0.astype(np.float64).copy(), 0, None)
    S = float(f.sum())
    if S < 0.01:
        return f, math.inf
    best_score = _score(f)
    rng = np.random.default_rng(seed)

    for sweep in range(max_sweeps):
        if time.monotonic() > stop_time:
            break
        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > stop_time:
                break

            v = f[i]
            # Determine search range
            x_lo = max(0.0, v * 0.1)
            x_hi = max(v * 3.0, 5.0 / n)
            x_hi = min(x_hi, 50.0 / n)
            if v < 1e-12:
                x_lo = 0.0
                x_hi = 3.0 / n

            # Golden section search
            gr = (math.sqrt(5.0) - 1.0) / 2.0
            a_g = x_lo
            b_g = x_hi
            c_g = b_g - gr * (b_g - a_g)
            d_g = a_g + gr * (b_g - a_g)

            f[i] = c_g
            sc_c = _score(f) if f.sum() >= 0.01 else math.inf
            f[i] = d_g
            sc_d = _score(f) if f.sum() >= 0.01 else math.inf

            for _gs in range(30):
                if b_g - a_g < 1e-12 * max(1.0, abs(v)):
                    break
                if sc_c < sc_d:
                    b_g = d_g
                    d_g = c_g
                    sc_d = sc_c
                    c_g = b_g - gr * (b_g - a_g)
                    if abs(c_g - d_g) < 1e-14:
                        break
                    f[i] = c_g
                    sc_c = _score(f) if f.sum() >= 0.01 else math.inf
                else:
                    a_g = c_g
                    c_g = d_g
                    sc_c = sc_d
                    d_g = a_g + gr * (b_g - a_g)
                    if abs(c_g - d_g) < 1e-14:
                        break
                    f[i] = d_g
                    sc_d = _score(f) if f.sum() >= 0.01 else math.inf

            # Pick best of c, d, original
            x_best = v
            sc_best = min(sc_c, sc_d)
            x_at_best = c_g if sc_c < sc_d else d_g
            if v > 0:
                sc_v = _score(f) if f[i] == v else (2.0 * n * _conv_full(f).max() / (f.sum()**2))
                f[i] = v
                sc_v = _score(f)
                if sc_v < sc_best:
                    x_best = v
                    sc_best = sc_v

            f[i] = x_best
            if sc_best < best_score - 1e-14:
                best_score = sc_best
                improved = True
            f[i] = x_best

        if not improved:
            break

    f = np.clip(f, 0, 1000)
    final_sc = _score(f)
    return f, final_sc


def _pair_moves(f0, stop_time, seed=9999, max_rounds=100):
    """Randomized 2-coordinate moves."""
    n = len(f0)
    f = f0.copy()
    best_score = _score(f)
    rng = np.random.default_rng(seed)

    for rnd in range(max_rounds):
        if time.monotonic() > stop_time:
            break
        improved = False
        n_trials = min(60, max(10, int((stop_time - time.monotonic()) * 20)))
        for _ in range(n_trials):
            if time.monotonic() > stop_time:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            oi, oj = f[i], f[j]
            best_pair = (oi, oj)
            best_pair_sc = best_score
            for mi in (0.0, 0.3, 0.5, 0.7, 1.0, 1.3, 1.7, 2.5):
                for mj in (0.0, 0.3, 0.5, 0.7, 1.0, 1.3, 1.7, 2.5):
                    f[i] = oi * mi if oi > 1e-12 else (0.02 / n if mi > 0 else 0.0)
                    f[j] = oj * mj if oj > 1e-12 else (0.02 / n if mj > 0 else 0.0)
                    if f.sum() < 0.01:
                        f[i], f[j] = oi, oj
                        continue
                    sc = _score(f)
                    if sc < best_pair_sc - 1e-14:
                        best_pair_sc = sc
                        best_pair = (f[i], f[j])
            f[i], f[j] = best_pair
            if best_pair_sc < best_score - 1e-14:
                best_score = best_pair_sc
                improved = True
        if not improved:
            break
    return f, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard_deadline = t0 + budget - 15.0

    parent = _get_parent()
    n_p = len(parent)

    # Fallback
    fallback = parent.copy()
    try:
        fallback_sc = _score(fallback)
    except Exception:
        fallback_sc = math.inf
    if not math.isfinite(fallback_sc):
        fallback = np.ones(384) / 384.0
        fallback_sc = _score(fallback)

    best_seq = fallback.copy()
    best_score = fallback_sc

    def _update_best(seq, sc):
        nonlocal best_seq, best_score
        try:
            exact = float(evaluate_sequence([float(x) for x in seq]))
        except Exception:
            exact = math.inf
        if math.isfinite(exact) and exact < best_score - 1e-13:
            best_seq = np.array(seq, dtype=np.float64).copy()
            best_score = exact

    def _safe_score(f):
        try:
            sc = _score(f)
            return sc if math.isfinite(sc) else math.inf
        except Exception:
            return math.inf

    # Normalize parent for optimization
    f_norm = _normalize(parent)
    n = n_p

    # === Phase 1: LP optimization on parent length ===
    phase1_end = t0 + budget * 0.45
    rng_main = np.random.default_rng(seed)

    # Multiple initializations
    inits = [f_norm.copy()]
    i_arr = np.arange(n, dtype=np.float64)
    # Cosine-shaped
    c1 = 0.5 + 0.5 * np.cos(np.pi * i_arr / (n - 1))
    inits.append(_normalize(np.clip(c1, 0, None)))
    # Two-hump
    c2 = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * i_arr / n)
    inits.append(_normalize(np.clip(c2, 0, None)))
    # Raised cosine (wider)
    c3 = 0.3 + 0.7 * np.cos(np.pi * (i_arr / (n - 1) - 0.5)) ** 2
    inits.append(_normalize(np.clip(c3, 0, None)))

    time_per_init = max(20.0, (phase1_end - t0) / len(inits))

    for idx, init in enumerate(inits):
        if time.monotonic() > phase1_end - 10.0:
            break
        stop_lp = min(phase1_end - 5.0, time.monotonic() + time_per_init)
        try:
            res, sc = _lp_optimize(init, stop_lp, seed + 1000 * (idx + 1))
        except Exception:
            res, sc = init, _safe_score(init)
        if res is None or not np.all(np.isfinite(res)) or res.sum() < 0.01:
            continue
        res = np.clip(res, 0, 1000)
        sc = _safe_score(res)
        if math.isfinite(sc) and sc < best_score:
            _update_best(res, sc)

    # === Phase 2: Gradient-based refinement ===
    phase2_end = t0 + budget * 0.60
    if time.monotonic() < phase2_end - 10.0:
        try:
            res, sc = _gradient_optimize(best_seq, phase2_end, seed=55555)
        except Exception:
            res, sc = best_seq, best_score
        if res is not None and np.all(np.isfinite(res)) and res.sum() >= 0.01:
            res = np.clip(res, 0, 1000)
            _update_best(res, _safe_score(res))

    # === Phase 3: Coordinate descent polish ===
    phase3_end = t0 + budget * 0.85
    if time.monotonic() < phase3_end - 20.0:
        try:
            res, sc = _coord_polish(best_seq, phase3_end, max_sweeps=60, seed=424242)
        except Exception:
            res, sc = best_seq, best_score
        if res is not None and np.all(np.isfinite(res)) and res.sum() >= 0.01:
            res = np.clip(res, 0, 1000)
            _update_best(res, _safe_score(res))

    # === Phase 4: Pair moves ===
    phase4_end = hard_deadline - 10.0
    if time.monotonic() < phase4_end - 5.0:
        try:
            res, sc = _pair_moves(best_seq, phase4_end, seed=88888, max_rounds=50)
        except Exception:
            res, sc = best_seq, best_score
        if res is not None and np.all(np.isfinite(res)) and res.sum() >= 0.01:
            res = np.clip(res, 0, 1000)
            _update_best(res, _safe_score(res))

    # === Phase 5: Final coordinate polish ===
    if time.monotonic() < hard_deadline - 5.0:
        try:
            res, sc = _coord_polish(best_seq, hard_deadline - 3.0, max_sweeps=30, seed=31337)
        except Exception:
            res, sc = best_seq, best_score
        if res is not None and np.all(np.isfinite(res)) and res.sum() >= 0.01:
            res = np.clip(res, 0, 1000)
            _update_best(res, _safe_score(res))

    # Final validation
    best_seq = np.clip(np.asarray(best_seq, dtype=np.float64), 0, 1000)
    best_seq = np.where(np.isfinite(best_seq), best_seq, 0.0)
    if best_seq.size == 0 or best_seq.sum() < 0.01:
        best_seq = fallback.copy()

    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in fallback]

    return [float(x) for x in best_seq]
