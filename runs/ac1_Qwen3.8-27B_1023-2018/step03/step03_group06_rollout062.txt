import math

    m = int(max(1, m))
    if m in _fft_cache:
        return _fft_cache[m]
    v = int(next_fast_len(m))

    return np.maximum(irfft(rfft(a, L) * rfft(b, L), L)[:m], 0)

    return _convolve(a, a)

    """(d * a)[j] for j = 0..n-1, where a has length n.
