    c = irfft(F * F, fsize)[:2 * n - 1]

    """Solve LP to find direction g that maximizes sum(g) subject to (f*g)_k <= M*thresh"""
    n = len(f)
    if M < 1e-15:

    A = np.zeros((2 * n - 1, n))

def _polish(a, stop_time, seed=4242):
    """Coordinate descent polish"""
    n = len(a)
    best = np.clip(np.asarray(a, dtype=np.float64).ravel(), 0, None).copy()
    fs = best.sum()
    if fs < 1e-10:
        return best, math.inf
    best = best / fs
    best_sc = _score(best)
    rng = np.random.default_rng(seed)
    no_imp = 0
    max_no_imp = 5
    
    multipliers = [0.3, 0.5, 0.7, 0.9, 1.1, 1.3, 1.7, 2.5, 4.0, 0.0]
    
    while time.monotonic() < stop_time - 1.0:
        improved = False
        
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 1.0:
                break
            orig = best[i]
            best_val = orig
            best_val_sc = best_sc
            for mlt in multipliers:
                new_val = orig * mlt if orig > 1e-14 else (0.005 if mlt > 0 else 0.0)
                best[i] = new_val
                if best.sum() < 0.01:
                    best[i] = orig
                    continue
                sc = _score(best)
                if sc < best_val_sc - 1e-14:
                    best_val = new_val
                    best_val_sc = sc
            best[i] = best_val
            if best_val_sc < best_sc - 1e-14:
                best_sc = best_val_sc
                improved = True
        
        # 2-coordinate moves
        for _ in range(60):
            if time.monotonic() > stop_time - 1.0:
                break
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            for mi in (0.5, 1.5, 0.0, 2.5, 0.3):
                for mj in (0.5, 1.5, 0.0, 2.5, 0.3):
                    best[i] = oi * mi if oi > 1e-14 else (0.005 if mi > 0 else 0.0)
                    best[j] = oj * mj if oj > 1e-14 else (0.005 if mj > 0 else 0.0)
                    if best.sum() < 0.01:
                        best[i], best[j] = oi, oj
                        continue
                    sc = _score(best)
                    if sc < best_sc - 1e-14:
                        best_sc = sc
                        improved = True
            best[i], best[j] = oi, oj
        
        if not improved:
            no_imp += 1
            if no_imp >= max_no_imp:
                break
        else:
            no_imp = 0
    
    return best, best_sc

def _lp_dir(f, M):
    n = len(f)
    if M < 1e-15:
        return None
    A = _build_A(f)
    c_vec = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_vec, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={"presolve": True})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    s = float(g.sum())
    if s < 1e-10:
        return None
    return g / s

def _lp_main(f0, stop, seed, max_iter=50000):
    rng = np.random.default_rng(seed)
    n = len(f0)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel(), 0, None).copy()
    fs = f.sum()
    if fs < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / fs

    best_f = f.copy()
    best_sc = 2.0 * n * float(_conv(f).max())  # normalized (sum=1)
    
    t = 0.25
