def _smoothed_grad(a, c, tau):
    """Compute gradient of smoothed max of c with respect to a.
    c = conv(a,a), w = softmax(c/tau)
    grad[k] = 2 * sum_j w[j] * a[j-k] (for valid j-k)
    """
    n = len(a)
    m = len(c)  # 2n-1
    # softmax with clipping for numerical stability
    c_max = float(c.max())
    if tau <= 0 or not np.isfinite(tau):
        tau = 1e-6
    z = (c - c_max) / tau
    z = np.clip(z, -50, 0)
    ez = np.exp(z)
    s = float(ez.sum())

        w[int(np.argmax(c))] = 1.0

    return grad

def _build_A_matrix(a):
    """Build the constraint matrix for the LP.
    A[k, i] = a[j] where j is the appropriate index for conv(a,b)[k] to involve b[i].
    conv(a, b)[k] = sum_i a[i'] * b[k-i'] ... 
    
    Actually: conv(a, b)[k] = sum_{i'} a[i'] * b[k - i'] for valid indices.
    So for a given k, the constraint is: sum_{i'} a[i'] * b[k-i'] <= M
    The coefficient of b[i] in row k is: a[i'] where i' = k - i, i.e., a[k-i] if 0<=k-i<n.

    return A

def _get_parent():

def _normalize(a, target=1.0):
    arr = np.asarray(a, dtype=np.float64).ravel()
    if arr.size == 0:
        return np.ones(1, dtype=np.float64) * target
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    s = float(arr.sum())
    if s > 1e-300:
        return (arr / s) * target
    return np.full(arr.size, target / arr.size, dtype=np.float64)

def _conv_full(a):
    n = len(a)
    nfft = next_fast_len(2 * n - 1)
    f = rfft(a, nfft)
    c = irfft(f * f, nfft)[:2 * n - 1]
    c = np.where(np.isfinite(c), c, 0.0)
    np.maximum(c, 0.0, out=c)
    return c

def _project_simplex(v, target=1.0):
    v = np.asarray(v, dtype=np.float64).ravel()
    n = v.size
    if n == 0:
        return np.ones(1, dtype=np.float64) * target
    v = np.where(np.isfinite(v), v, 0.0)
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    with np.errstate(divide="ignore", invalid="ignore"):
        cond = (css - target) / idx < u
    rho = np.nonzero(cond)[0]
    if rho.size == 0:
        return np.full(n, target / n, dtype=np.float64)
    theta = (css[rho[-1]] - target) / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    s = float(w.sum())
    if s <= 1e-300 or not np.isfinite(s):
        return np.full(n, target / n, dtype=np.float64)
    return (w / s) * target

def _resample(a, m):
    a = _normalize(a)
    n = a.size
    m = int(m)
    if m <= 0:
        return np.ones(1, dtype=np.float64)
    if n == m:
        return a.copy()
    if m > n:
        idx = (np.arange(m, dtype=np.int64) * n) // m
        counts = np.bincount(idx, minlength=n).astype(np.float64)
        counts[counts <= 0] = 1.0
        b = a[idx] / counts[idx]
    else:
        b = np.zeros(m, dtype=np.float64)
        idx = (np.arange(n, dtype=np.int64) * m) // n
        np.add.at(b, idx, a)
        b = np.maximum(b, 0.0)
    return _normalize(b)

def _smoothed_grad(a, c, tau):
    n = len(a)
    m = len(c)
    c_max = float(c.max())
    if tau <= 1e-15:
        tau = 1e-15
    z = (c - c_max) / tau
    z = np.clip(z, -50.0, 0.0)
    ez = np.exp(z)
    s = float(ez.sum())
    if s <= 0 or not np.isfinite(s):
        w = np.zeros(m)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ez / s
    grad = 2.0 * np.correlate(w, a, 'valid')
    return grad

def _solve_lp_min_lambda(a_fixed, deadline, rng):
    """Solve: min lambda s.t. conv(a_fixed, b)[k] <= lambda, sum(b)=1, b>=0, lambda>=0"""
    n = len(a_fixed)
    if n < 2 or n > 1200:
        return None
    if time.monotonic() >= deadline:
        return None
    m = 2 * n - 1
    A_mat = _build_A_v2(a_fixed)
    # Inequality: A_mat @ b - lambda <= 0, i.e., [A_mat, -1] @ [b, lambda] <= 0
    A_ub = np.column_stack([A_mat, -np.ones(m, dtype=np.float64)])
    b_ub = np.zeros(m, dtype=np.float64)
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    # Objective: min lambda
    c_obj = np.zeros(n + 1, dtype=np.float64)
    c_obj[n] = 1.0
    # Add tiny random perturbation to objective for b variables
    c_obj[:n] += 1e-8 * rng.standard_normal(n)
    # Bounds: b[i] >= 0, lambda >= 0
    bounds = [(0, None)] * n + [(0, None)]
    rem = deadline - time.monotonic()
    tl = max(0.3, min(8.0, 0.2 * rem))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': tl, 'presolve': True})
        if res is not None and getattr(res, 'success', False):
            b_opt = np.asarray(res.x[:n], dtype=np.float64)
            b_opt = np.maximum(b_opt, 0.0)
            s = float(b_opt.sum())
            if s > 1e-12:
                return b_opt / s
    except Exception:
        pass
    return None

def _smooth_init(n, rng, base=None):
    x = np.linspace(0.0, 1.0, n, endpoint=False)
    kind = int(rng.integers(0, 5))
    if kind == 0:
        alpha = rng.uniform(0.4, 2.5)
        a = x ** alpha
    elif kind == 1:
        alpha = rng.uniform(0.4, 2.5)
        a = (1.0 - x) ** alpha
    elif kind == 2:
        c = rng.uniform(0.25, 0.75)
        w = rng.uniform(0.1, 0.35)
        a = np.exp(-((x - c) ** 2) / (2.0 * w * w))
    elif kind == 3:
        a = np.ones(n)
        K = min(4, max(1, n // 64))
        for k in range(1, K + 1):
            amp = rng.uniform(0.05, 0.7 / k)
            ph = rng.uniform(0.0, 2.0 * np.pi)
            a += amp * np.cos(2.0 * np.pi * k * x + ph)
    else:
        a = 1.0 + 0.3 * np.sin(np.pi * x)
    a = np.maximum(a, 1e-4)
    if base is not None and len(base) == n:
        a = 0.5 * _normalize(base) + 0.5 * a
    return _normalize(a)

    patience = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        if M < best_M * (1.0 - 1e-11):
            best_a = a.copy()
            best_M = M
            patience = 0
            step = max(0.05, step * 0.995)
        else:
            patience += 1

        # Adapt smoothing

        if patience > 50:

            if r < 0.4:
                a = _project_simplex(a + rng.normal(0, 0.03, n))

            else:
                a = _smooth_init(n, rng)
                c = _conv_full(a)
                M = float(c.max())
            patience = 0
            step = 1.0

        # Gradient step with line search

        gnorm = float(np.linalg.norm(grad))

        best_cand_M = M
