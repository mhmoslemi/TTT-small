```python
    cs = np.arange(1, n+1)
    cond = u * cs > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def _cd(f, stop, max_sweeps=300, seed=0):

            m_out = 0.0

                m_out = float(np.max(conv[:ks]))
            if ke < 2*n-1:
                m_out = max(m_out, float(np.max(conv[ke:])))
            
            sp_ = 2*i
            in_r = ks <= sp_ < ke
            si = sp_ - ks if in_r else -1
            
            # Grid
            xhi = min(max(4.0*v, 0.03)+0.005, 0.4)
            ng = 45

            if not vld.any():

            bx, bo = xg[ib], ob[ib]
            
            cur = 2.0*n*float(np.max(np.maximum(np.maximum(conv[ks:ke], m_out), 0)))

                nvf = CA[None,:] + 2.0*df[:,None]*f[:nfl][None,:]
                if in_r:
                    nvf[:, si] += df*df
                maf = nvf.max(axis=1)
                mnf = np.maximum(maf, m_out)
                Sf = 1.0 - v + xf
                vf = Sf > 1e-8
                if vf.any():
                    of = np.full(len(xf), 1e30)
                    of[vf] = 2.0*n*mnf[vf]/Sf[vf]**2
                    if2 = int(np.argmin(of))
                    if of[if2] < bo:
                        bx, bo = xf[if2], of[if2]
                
                # Update
                delta = bx - v
                f[i] = bx
                conv[ks:ke] += delta * 2.0*f[:nfl]
                if in_r:
                    conv[sp_] += delta*delta
                improved = True
        
        if improved:
            stall = 0
            sc = 2.0*n*float(conv.max())
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_f = f.copy()
        else:
            stall += 1
            if stall >= 25:

    if sf > 0.01:
        sc = 2.0*n*float(_conv_sq(f).max())/sf**2
        if sc < best_sc:
            best_f, best_sc = f, sc
    return best_f, best_sc
```

```python
def _lp_step(f, n, time_limit=5.0):
    """LP-based direction finding."""
    conv = _conv_sq(f)
    M = float(conv.max())
    if M < 1e-15:
        return None
    
    # Build constraint matrix A: A[k,i] = f[k-i] if 0 <= k-i <= n-1
    # For each column i, the nonzeros are at rows i, i+1, ..., i+n-1
    # with values f[0], f[1], ..., f[n-1]
    # This is a "convolution" matrix
    
    # Dense construction (fine for n <= 1000)
    A = np.zeros((2*n-1, n))
    for i in range(n):
        A[i:i+n, i] = f
    
    c = -np.ones(n)
    try:
        res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M),
                      bounds=[(0, None)]*n, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
    except Exception:
        return None
    
    if res.status != 0 or res.x is None:
        return None
    
    g = np.asarray(res.x, dtype=np.float64)
    gs = float(g.sum())
    if gs < 1e-10:
        return None
    return g / gs
```
