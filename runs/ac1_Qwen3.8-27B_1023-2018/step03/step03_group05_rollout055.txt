def _grad_weighted(x, w):
    """Gradient of sum_k w[k] * (x*x)[k] w.r.t. x.
    w is the weight vector (length 2n-1).
    Returns gradient (length n).
    """
    n = len(x)
    m = 2*n - 1
    # Pad w to length 3n-2 for the convolution
    # (x*x)[k] depends on x[i] for i in [0,n-1]
    # grad[k,j] = x[k-j] + x[j] (simplified)
    # sum_k w[k] * x[k-j] = (x convolved with w_flip)[j] at appropriate offset
    # sum_k w[k] * x[j] = x[j] * sum_k w[k] (over valid k)
    
    # Let's compute: for each j, sum_{k: 0<=k-j<=n-1} w[k] * x[k-j]
    # = sum_{i: 0<=i<=n-1} w[i+j] * x[i]  (where i = k-j)
    # This is the cross-correlation of w with x, evaluated at offset j.
    
    # Use FFT: define W = w padded, X = x padded
    # corr[j] = sum_i w[i+j] * x[i] = sum_i w_reversed[j + (n-1) - i] * x[i]
    
    # Actually, let me just do it with a matrix multiplication for clarity,
    # since n=512 is small enough.
    
    # Or use FFT:
    nfft = next_fast_len(3*n)
    W = rfft(np.concatenate([w, np.zeros(nfft - m)]), nfft)  # w in frequency
    X = rfft(np.concatenate([x, np.zeros(nfft - n)]), nfft)  # x in frequency
    
    # We want: for each j in [0, n-1], sum_{i=0}^{n-1} w[i+j] * x[i]
    # This is (x flipped) correlated with w, at offset j.
    # x_flipped[i] = x[n-1-i]
    # sum_i w[i+j] * x[i] = sum_i w[i+j] * x_flipped[n-1-i]
    # = (x_flipped * w)[n-1+j] (linear convolution)
    
    xf = x[::-1]  # x_flipped
    nfft2 = next_fast_len(2*n - 1 + n)  # need length at least (n-1) + m
    Xf = rfft(xf, nfft2)
    W2 = rfft(w, nfft2)
    conv = irfft(Xf * W2, nfft2)
    
    # conv[k] = sum_i xf[i] * w[k-i] = sum_i x[n-1-i] * w[k-i]
    # We want sum_i w[i+j] * x[i] = sum_i w[(n-1) + j - (n-1-i)] * x[n-1-i]
    # Let m' = n-1-i, then i = n-1-m': sum_m' w[(n-1)+j-m'] * x[m'] = conv[n-1+j]
    
    term1 = conv[n-1 : n-1+n]  # term1[j] = sum_i w[i+j] * x[i]
    
    # Second term: x[j] * sum_{k: 0<=k-j<=n-1} w[k] = x[j] * sum_{k=j}^{j+n-1} w[k]
    w_cumsum = np.cumsum(np.concatenate([[0], w]))
    term2 = np.zeros(n)
    for j in range(n):
        k_lo = max(0, j)
        k_hi = min(m-1, j+n-1)
        term2[j] = w_cumsum[k_hi+1] - w_cumsum[k_lo]
    term2 *= x
    
    # Third term (correction for k=2j): -x[j] * w[2j] if 2j < m
    term3 = np.zeros(n)
    for j in range(n):
        if 2*j < m:
            term3[j] = -x[j] * w[2*j]
    
    return term1 + term2 + term3

import numpy as np
import time
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _norm1(a):
    """Normalize to nonnegative, sum=1."""
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-300:
        return np.full(a.size, 1.0/a.size)
    return a / s

    n = a.size
    nfft = next_fast_len(2*n-1)
    f = rfft(a, nfft)
    c = irfft(f*f, nfft)[:2*n-1]
    return np.maximum(c, 0.0)

    """Score from precomputed convolution (assuming sum=1)."""
    M = c.max()
    return 2.0 * n * M

def _build_A(a):
    """Build convolution matrix: A[k,j] = a[k-j] if 0<=k-j<=n-1."""
    n = a.size
    m = 2*n - 1
    A = np.zeros((m, n), dtype=np.float64)
    # Vectorized construction
    # For column j: A[k,j] = a[k-j] for k in [j, j+n-1] ∩ [0, m-1]
    for j in range(n):
        k_lo = j  # since k >= j for k-j >= 0
        k_hi = min(m-1, j + n - 1)  # k-j <= n-1 → k <= j+n-1
        A[k_lo:k_hi+1, j] = a[:k_hi+1-k_lo]
    return A

def _lp_solve(a, M, rng, deadline):
    """Solve the LP: max sum(y) s.t. A*y <= M, y >= 0."""
    n = a.size
    m = 2*n - 1
    A = _build_A(a)
    # Small random perturbation for tie-breaking
    pert = rng.standard_normal(n) * 1e-6
    c_obj = -(np.ones(n) + pert)  # minimize -sum(y) - pert^T y
    b_ub = np.full(m, M, dtype=np.float64)
    
    rem = deadline - time.monotonic()
    tl = max(0.5, min(10.0, 0.3*rem))
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0, None),
                      method='highs', options={'time_limit': tl})
        if res.success:
            y = np.maximum(res.x, 0)
            s = y.sum()
            if s > 1e-12:
                return y / s
    except:
        pass
    return None

def _lp_dual(a, M, rng, deadline):
    """Solve the dual LP: min M*sum(λ) s.t. A^T λ >= 1, λ >= 0."""
    n = a.size
    m = 2*n - 1
    A = _build_A(a)
    pert = rng.standard_normal(m) * 1e-6
    c_obj = M * (np.ones(m) + pert)
    A_low = -A.T  # A^T λ >= 1 → -A^T λ <= -1
    b_low = -np.ones(n, dtype=np.float64)
    
    rem = deadline - time.monotonic()
    tl = max(0.5, min(10.0, 0.3*rem))

                      method='highs', options={'time_limit': tl})
        if res.success:

            s = lam.sum()
            if s > 1e-12:

    # = sum_i x[i] * sum_k w[k]*x[k-i]
