def _conv(a):
    n = len(a)
    if n <= 800:
        return np.convolve(a, a, mode='full')
    from scipy.fft import rfft, irfft, next_fast_len
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)

def _score(f):

        return math.inf

def _build_A(f, n):

def _lp_solve(f, M, time_limit=3.0):
    n = len(f)
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)

    if gs < 1e-15:
