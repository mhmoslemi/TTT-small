        L = _fl(2 * n - 1)
        fa = rfft(a, L)
        return irfft(fa * fa, L)[:2 * n - 1]

    def _conv2(a, b):
        m = len(a) + len(b) - 1
        L = _fl(m)
        return irfft(rfft(a, L) * rfft(b, L), L)[:m]

    def _sc(a):
        c = _conv(a)
        return 2.0 * len(a) * float(np.max(c))

    def _ex(a):
        try:
            fn = globals()['evaluate_sequence']
            return float(fn([float(x) for x in a]))
        except Exception:
            return float('inf')

    def _norm(a):
        a = np.asarray(a, dtype=np.float64).ravel()
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return np.ones_like(a) / len(a)
        return a / s

    # Get parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(256, dtype=np.float64) / 256.0
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    if p.sum() < 1e-12:
        p = np.ones(256) / 256.0
    p = p / p.sum()
    n = len(p)

    best = p.copy()
    best_sc = _sc(best)

    def _try(a):
        nonlocal best, best_sc
        a = _norm(a)
        if a.size != n:
            return
        sc = _sc(a)
        if math.isfinite(sc) and sc < best_sc - 1e-14:
            best_sc = sc
            best = a.copy()

    # LP-based iteration: find best direction, line search
    def _lp_iter(f):
        cf = _conv(f)
        M = float(cf.max())
        if M < 1e-18:
            return f.copy()

        # Build constraint matrix A: A[k,i] = f[k-i] if 0<=k-i<n else 0
        A = np.zeros((2 * n - 1, n), dtype=np.float64)
        for i in range(n):
            A[i:i + n, i] = f

        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        c_obj = -np.ones(n, dtype=np.float64)
        bounds_lp = [(0.0, None)] * n

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds_lp,
                          method='highs', options={'presolve': True, 'time_limit': 30.0})
        except Exception:
            return f.copy()

        if res is None or res.status != 0 or res.x is None:
            return f.copy()

        g = np.maximum(res.x, 0.0)
        gs = g.sum()
        if gs < 1e-15:
            return f.copy()
        g = g / gs

        # Line search on t: minimize max_k c_k(t)
        cfs = _conv2(f, g)
        cg = _conv(g)
        a0 = cf
        a1 = 2.0 * (cfs - cf)
        a2 = cg - 2.0 * cfs + cf

        # Coarse grid
        ts = np.concatenate([np.linspace(0, 0.1, 51), np.linspace(0.1, 1.0, 151)])
        vals = np.max(a0[None, :] + a1[None, :] * ts[:, None] + a2[None, :] * ts[:, None] ** 2, axis=1)
        ib = int(np.argmin(vals))
        lo = max(0.0, float(ts[max(0, ib - 2)]))
        hi = min(1.0, float(ts[min(len(ts) - 1, ib + 2)]))
        if hi - lo < 1e-4:
            hi = min(1.0, lo + 1e-3)
            lo = max(0.0, hi - 1e-3)
        tf = np.linspace(lo, hi, 201)
        vf = np.max(a0[None, :] + a1[None, :] * tf[:, None] + a2[None, :] * tf[:, None] ** 2, axis=1)
        t_opt = float(tf[int(np.argmin(vf))])

        new = (1.0 - t_opt) * f + t_opt * g
        return _norm(new)

    # L-BFGS with smooth max (log-sum-exp)
    def _lbfgs_phase(f0, betas, stop, maxiter=300):
        f = _norm(f0)
        nloc = len(f)

        for beta in betas:
            if time.monotonic() > stop - 3.0:
                break

            def fg(y):
                if time.monotonic() > stop:
                    raise RuntimeError("timeout")
                ym = np.asarray(y, dtype=np.float64)
                if not np.all(np.isfinite(ym)):
                    return 1e100, np.zeros_like(ym)
                ym = ym - ym.max()
                ey = np.exp(ym)
                a = ey / ey.sum()
                c = _conv(a)
                bc = beta * c
                bm = bc.max()
                es = np.exp(bc - bm)
                ess = es.sum()
                if ess < 1e-300 or not math.isfinite(ess):
                    return 1e100, np.zeros_like(ym)
                obj = (bm + math.log(ess)) / beta
                w = es / ess
                wr = w[::-1]
                caw = _conv2(a, wr)
                ga = caw[nloc - 1:2 * nloc - 1][::-1]
                dot = float(np.dot(a, ga))
                gy = a * (ga - dot)
                gy = np.where(np.isfinite(gy), gy, 0.0)
                return float(obj), gy

            f_init = _norm(f)
            y0 = np.log(np.maximum(f_init, 1e-300))

            try:
                res = minimize(fg, y0, jac=True, method='L-BFGS-B',
                               options={'maxiter': maxiter, 'ftol': 1e-14,
                                        'gtol': 1e-10, 'maxls': 40})
            except Exception:
                break

            if res is not None and res.x is not None:
                y_new = np.asarray(res.x, dtype=np.float64)
                if np.all(np.isfinite(y_new)):
                    a_new = np.exp(y_new - y_new.max())
                    a_new = a_new / a_new.sum()
                    sc_new = _sc(a_new)
                    if math.isfinite(sc_new) and sc_new < _sc(f) - 1e-14:
                        f = a_new.copy()

        return f

    # Coordinate descent polish
    def _coord(a0, stop):
        a = _norm(a0).copy()
        nloc = len(a)
        mean_a = float(a.mean())
        for _pass in range(4):
            if time.monotonic() > stop:
                break
            improved = False
            for i in range(nloc):
                if time.monotonic() > stop:
                    break
                orig = float(a[i])
                base = orig if orig > 1e-14 else mean_a
                best_v, best_sc_v = orig, _sc(a)
                for mult in (0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0, 3.0):
                    v = base * mult
                    if v < 0 or v > 1000:
                        continue
                    a[i] = v
                    sc = _sc(a)
                    if sc < best_sc_v:
                        best_sc_v = sc
                        best_v = v
                a[i] = best_v
                if best_sc_v < _sc(a0) - 1e-15:
                    improved = True
            if not improved:
                break
        return a

    # Random restart helper
    def _random_init(rng, L):
        inits = [
            _norm(rng.exponential(1.0, size=L)),
            _norm(np.abs(rng.normal(0, 1, L)) + 0.01),
        ]
        x = np.ones(L)
        i = np.arange(L, dtype=np.float64)
        for m in range(1, 5):
            x += rng.uniform(0.2, 0.8) / (m + 1) * np.cos(2 * np.pi * m * i / L + rng.uniform(0, 2 * np.pi))
        inits.append(_norm(np.abs(x) + 0.02))
        return inits

    rng = np.random.default_rng(seed)

    # ===== Phase 1: LP iterations (25% of budget) =====
    p1_stop = t0 + 0.22 * (deadline - t0)
    f = best.copy()
    no_imp = 0
    while time.monotonic() < p1_stop:
        f_new = _lp_iter(f)
        sc_new = _sc(f_new)
        if math.isfinite(sc_new) and sc_new < _sc(f) - 1e-15:
            f = f_new
            _try(f)
            no_imp = 0
        else:
            no_imp += 1
            if no_imp >= 5:
                # Perturb and retry
                f = f + rng.normal(0, 0.02 * f.mean(), n)
                f = _norm(f)
                no_imp = 0
                if _sc(f) < best_sc:
                    _try(f)
            if time.monotonic() > p1_stop - 2:
                break

    # ===== Phase 2: L-BFGS smoothing (25% of budget) =====
    p2_stop = t0 + 0.47 * (deadline - t0)
    f = _lbfgs_phase(best, [25.0, 80.0, 250.0], p2_stop, maxiter=350)
    _try(f)

    # ===== Phase 3: More LP (30% of budget) =====
    p3_stop = t0 + 0.78 * (deadline - t0)
    f = best.copy()
    no_imp = 0
    while time.monotonic() < p3_stop:
        f_new = _lp_iter(f)
        sc_new = _sc(f_new)
        if math.isfinite(sc_new) and sc_new < _sc(f) - 1e-16:
            f = f_new
            _try(f)
            no_imp = 0
        else:
            no_imp += 1
            if no_imp >= 8:
                f = f + rng.normal(0, 0.005 * f.mean(), n)
                f = _norm(f)
                no_imp = 0
            if time.monotonic() > p3_stop - 2:
                break

    # ===== Phase 4: Fine L-BFGS (15% of budget) =====
    p4_stop = deadline - 25.0
    f = _lbfgs_phase(best, [400.0, 800.0, 1500.0], p4_stop, maxiter=250)
    _try(f)

    # ===== Phase 5: Coordinate descent polish =====
    fin_stop = deadline - 8.0
    f = _coord(best, fin_stop)
    _try(f)

    # One more LP burst if time
    if time.monotonic() < deadline - 5:
        f2 = _lp_iter(best)
        _try(f2)

    # ===== Final: validate and return =====
    final = _norm(best)
    final = np.clip(final, 0.0, 1000.0)
    sc_final = _ex(final)
    if not math.isfinite(sc_final) or sc_final < 0:
        fallback = _norm(p)
        final = np.clip(fallback, 0.0, 1000.0)

    return [float(x) for x in final]
