def _get_parent():

def _conv_full(a):
    n = len(a)
    if n <= 1024:
        return np.convolve(a, a, mode="full")
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[: 2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n):
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        end = min(j + n, 2 * n - 1)
        A[j:end, j] = f[: end - j]
    return A

def _coord_polish(f, stop_time, max_sweeps=60, seed=777):
    """Coordinate descent with exact 1D optimization per coordinate."""
    n = len(f)
    f = np.clip(f.astype(np.float64).copy(), 0, None)
    S = float(f.sum())
    if S < 0.01:
        return f, math.inf
    conv = _conv_full(f)
    best_score = 2.0 * n * conv.max() / (S * S)
    rng = np.random.default_rng(seed)

    for sweep in range(max_sweeps):
        if time.monotonic() > stop_time - 6.0:
            break
        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > stop_time - 6.0:
                break

            v = f[i]
            S_rest = S - v

            infl = np.zeros(2 * n - 1)
            start = i
            end = min(i + n, 2 * n - 1)
            infl[start:end] = 2.0 * f[: end - start]

            x_max = max(4.0 * v + 0.5, 1.0) if v > 1e-8 else 1.5
            x_max = min(x_max, 100.0)

            # Coarse grid (60 points)
            n_pts = 60
            x_grid = np.linspace(0.0, x_max, n_pts)
            best_x = v
            best_obj = best_score

            for x in x_grid:
                S_new = S_rest + x
                if S_new < 0.01:
                    continue
                w = x - v
                if abs(w) < 1e-14:
                    m_new = conv.max()
                else:
                    tmp = conv.copy()
                    tmp[start:end] += w * infl[start:end]
                    if 0 <= 2 * i < 2 * n - 1:
                        tmp[2 * i] += w * w
                    m_new = float(tmp.max())
                obj = 2.0 * n * m_new / (S_new * S_new)
                if obj < best_obj - 1e-15:
                    best_obj = obj
                    best_x = x

            # Fine grid
            if best_x != v:
                dw = x_max / n_pts
                x_fine = np.linspace(max(0, best_x - 4 * dw), min(x_max, best_x + 4 * dw), 40)
                for x in x_fine:
                    S_new = S_rest + x
                    if S_new < 0.01:
                        continue
                    w = x - v
                    if abs(w) < 1e-14:
                        continue
                    tmp = conv.copy()
                    tmp[start:end] += w * infl[start:end]
                    if 0 <= 2 * i < 2 * n - 1:
                        tmp[2 * i] += w * w
                    m_new = float(tmp.max())
                    obj = 2.0 * n * m_new / (S_new * S_new)
                    if obj < best_obj - 1e-15:
                        best_obj = obj
                        best_x = x

            if best_x != v and best_obj < best_score - 1e-14:
                w = best_x - v
                f[i] = best_x
                S += w
                conv[start:end] += w * infl[start:end]
                if 0 <= 2 * i < 2 * n - 1:
                    conv[2 * i] += w * w
                best_score = best_obj
                improved = True

        if not improved:
            break

    f = np.clip(f, 0, 1000)
    final_score = _score(f)
    return f, final_score


def _subgradient_fw(f, n, steps=200, stop_time=None):
    """Frank-Wolfe with subgradient of max(convolution)."""
    f = f.copy()
    if f.sum() < 1e-10:
        f = np.ones(n) / n
    f = f / f.sum()
    best_f = f.copy()
    best_score = _score(f)
    
    for it in range(steps):
        if stop_time is not None and time.monotonic() > stop_time - 5.0:
            break
        # Compute convolution and subgradient
        b = _conv_full(f)
        M = float(b.max())
        
        # Active set
        active = np.where(b >= M * (1 - 1e-9))[0]
        
        # Subgradient: subgrad[i] = sum over active k of (f[k-i] if valid + f[i-k] if valid)
        subgrad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                ki = k - i
                subgrad[i] += f[ki]
                if i != k - i:  # avoid double-counting when k-i == i
                    subgrad[i] += f[k - i] if 0 <= k - i < n else 0
            # Actually let me be more careful:
        # Redo subgradient properly
        subgrad = np.zeros(n)
        for k in active:
            for i in range(n):
                val = 0.0
                ki = k - i
                if 0 <= ki < n:
                    val += f[ki]
                ik = i - k
                if 0 <= ik < n:
                    val += f[ik]
                subgrad[i] += val
        
        # FW step: s* = e_{argmin subgrad}
        i_min = np.argmin(subgrad)
        
        # Step size: 2/(it+2)
        gamma = 2.0 / (it + 2.0)
        f_new = (1.0 - gamma) * f + gamma * (np.zeros(n) + (1.0 / n))  # s* normalized
        # Actually FW: s* is the vertex minimizing <subgrad, s> on simplex
        # That's e_{i_min}
        f_new = (1.0 - gamma) * f
        f_new[i_min] += gamma
        f_new = np.clip(f_new, 0, None)
        fs = f_new.sum()
        if fs > 1e-10:
            f_new /= fs
        
        sc = _score(f_new)
        if sc < best_score - 1e-15:
            f = f_new
            best_f = f.copy()
            best_score = sc
        else:
            f = f_new  # still use it as next iterate
    
    return best_f, best_score


def _interpolate_to_length(f, L):

    x_old = np.linspace(0, 1, len(f))

    return np.interp(x_new, x_old, f)

    if time.monotonic() < final_end - 20:
        try:
            f5, sc5 = _coord_polish(best_seq, final_end, max_sweeps=40, seed=seed + 9999)
        except Exception:
            f5, sc5 = best_seq, best_score
        if sc5 < best_score:
            best_seq = np.clip(f5, 0, 1000)
            best_score = sc5
        _try_update(best_seq, best_score)

    # === Phase 6: One more LP round if time allows ===
    if time.monotonic() < deadline - 40:
        n_best = len(best_seq)
        lp_stop6 = deadline - 25.0
        try:
            f6, sc6 = _lp_optimize(best_seq, n_best, lp_stop6, seed + 50000)
        except Exception:
            f6, sc6 = best_seq, best_score
        if sc6 < best_score:
            best_seq = np.clip(f6, 0, 1000)
            best_score = sc6
            _try_update(best_seq, best_score)
        # Quick polish
        if time.monotonic() < deadline - 15:
            try:
                f6b, sc6b = _coord_polish(best_seq, deadline - 8.0, max_sweeps=15, seed=seed + 60000)
            except Exception:
                f6b, sc6b = best_seq, best_score
            if sc6b < best_score:
                best_seq = np.clip(f6b, 0, 1000)
                best_score = sc6b
            _try_update(best_seq, best_score)

    # === Final validation ===
    best_seq = np.clip(np.asarray(best_seq, dtype=np.float64), 0, 1000)
    best_seq = np.where(np.isfinite(best_seq), best_seq, 0.0)
    if best_seq.size == 0 or best_seq.sum() < 0.01:
        best_seq = parent.copy()

    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in parent]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in parent]

    if final_sc < best_score:
        best_score = final_sc

    return [float(x) for x in best_seq]
