```
def _project_simplex(v,target=1):
    v=np.asarray(v,float).ravel()
    n=v.size
    if n==0: return np.ones(1)
    v=np.where(np.isfinite(v),v,0)
    u=np.sort(v)[::-1]
    css=np.cumsum(u)
    idx=np.arange(1,n+1,dtype=float)
    cond=(css-target)/idx < u
    rho=np.nonzero(cond)[0]
    if rho.size==0: return np.full(n,target/n)
    theta=(css[rho[-1]]-target)/(rho[-1]+1)
    w=np.maximum(v-theta,0)
    s=w.sum()
    if s<=1e-300: return np.full(n,target/n)
    return w*(target/s)
```

```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, fft, ifft, next_fast_len
from scipy.optimize import linprog


def _clean(a):
    try:
        arr = np.asarray(a, dtype=np.float64).ravel()
    except Exception:
        return np.ones(1, dtype=np.float64)
    if arr.size == 0:
        return np.ones(1, dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    return arr


def _normalize(a, target=1.0):
    arr = _clean(a)
    n = arr.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    s = float(arr.sum())
    if not math.isfinite(s) or s <= 1e-300:
        return np.full(n, float(target) / float(n), dtype=np.float64)
    return (arr / s) * float(target)


def _project_simplex(v, target=1.0):
    v = np.asarray(v, dtype=np.float64).ravel()
    n = v.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = (css - target) / idx < u
    rho = np.nonzero(cond)[0]
    if rho.size == 0:
        return np.full(n, target / float(n), dtype=np.float64)
    theta = (css[rho[-1]] - target) / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    s = float(w.sum())
    if not math.isfinite(s) or s <= 1e-300:
        return np.full(n, target / float(n), dtype=np.float64)
    return (w / s) * float(target)


def _conv_full(a):
    a = _clean(a)
    n = a.size
    if n == 0:
        return np.array([0.0], dtype=np.float64)
    nfft = next_fast_len(2 * n - 1)
    f = rfft(a, nfft)
    c = irfft(f * f, nfft)
    c = c[:2 * n - 1]
    c = np.where(np.isfinite(c), c, 0.0)
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    arr = _normalize(a, 1.0)
    if arr.size == 0:
        return float("inf")
    s = float(arr.sum())
    if not math.isfinite(s) or s < 0.01:
        return float("inf")

    eval_func = globals().get("evaluate_sequence", None)
    if eval_func is not None:
        try:
            val = eval_func([float(x) for x in arr])
            if np.isfinite(val) and val >= 0:
                return float(val)
        except Exception:
            pass

    c = _conv_full(arr)
    if c.size == 0:
        return float("inf")
    m = float(c.max())
    if not math.isfinite(m):
        return float("inf")
    den = s * s
    if den <= 0.0:
        return float("inf")
    return float(2.0 * arr.size * m / den)


def _resample(a, m):
    a = _normalize(a)
    n = a.size
    m = int(m)
    if m <= 0:
        return np.ones(1, dtype=np.float64)
    if n == 0:
        return np.ones(m, dtype=np.float64)
    if n == m:
        return a.copy()
    if m > n:
        idx = (np.arange(m, dtype=np.int64) * n) // m
        counts = np.bincount(idx, minlength=n).astype(np.float64)
        counts[counts <= 0] = 1.0
        b = a[idx] / counts[idx]
    else:
        b = np.zeros(m, dtype=np.float64)
        idx = (np.arange(n, dtype=np.int64) * m) // n
        np.add.at(b, idx, a)
        b = np.maximum(b, 0.0)
    return _normalize(b)


def _random_smooth(n, rng, base=None):
    n = int(n)
    if n <= 0:
        return np.ones(1, dtype=np.float64)
    x = np.linspace(0.0, 1.0, n, endpoint=False)
    ar = np.arange(n)
    kind = int(rng.integers(0, 8))

    if kind == 0:
        a = x ** float(rng.uniform(0.3, 2.5))
    elif kind == 1:
        a = (1.0 - x) ** float(rng.uniform(0.3, 2.5))
    elif kind == 2:
        c = float(rng.uniform(0.2, 0.8))
        wdt = float(rng.uniform(0.05, 0.30))
        a = np.exp(-((x - c) ** 2) / (2.0 * wdt * wdt)) + 0.1
    elif kind == 3:
        a = np.ones(n, dtype=np.float64)
        kcnt = int(rng.integers(1, 4))
        for k in range(kcnt):
            m = int(rng.integers(1, 5))
            amp = float(rng.uniform(0.05, 0.6 / (k + 1)))
            a += amp * np.cos(2.0 * np.pi * m * x + rng.uniform(0.0, 2.0 * np.pi))
    elif kind == 4:
        a = np.ones(n, dtype=np.float64)
        for _ in range(int(rng.integers(2, 4))):
            c = float(rng.uniform(0.1, 0.9))
            wdt = float(rng.uniform(0.04, 0.30))
            amp = float(rng.uniform(0.1, 0.8))
            a += amp * np.exp(-((x - c) ** 2) / (2.0 * wdt * wdt))
    elif kind == 5:
        a = rng.uniform(0.3, 1.5, n)
    elif kind == 6:
        beta = float(rng.uniform(0.2, 0.85))
        sign = np.where((ar % 2) == 0, 1.0, -1.0)
        a = 1.0 + beta * sign
    else:
        a = np.ones(n, dtype=np.float64)
        for m in (1, 2, 3):
            a += float(rng.uniform(0.1, 0.45)) * np.cos(
                2.0 * np.pi * m * x + rng.uniform(0.0, 2.0 * np.pi)
            )

    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 1e-5)
    if base is not None and np.asarray(base).size == n:
        a = 0.5 * _normalize(base) + 0.5 * a
    return _normalize(np.maximum(a, 0.0))


def _spectral_flat(start, rng, rho, iters=80, mix=0.35):
    p = _normalize(start)
    n = p.size
    if n < 3:
        return p
    p = p + 1e-4 * rng.standard_normal(n)
    p = np.maximum(p, 1e-12)
    p = _normalize(p)
    base = float(rho) / math.sqrt(max(n - 1, 1))
    half = n // 2
    pair_end = half if (n % 2 == 1) else half - 1
    idx = np.arange(1, pair_end + 1)

    for _ in range(int(iters)):
        F = fft(p)
        phase = np.angle(F)
        G = np.full(n, base, dtype=np.complex128) * np.exp(1j * phase)
        G[0] = 1.0 + 0j
        if idx.size:
            G[n - idx] = np.conj(G[idx])
        if n % 2 == 0:
            G[half] = float(G[half].real)
        G = np.where(np.isfinite(G), G, 0.0)
        x = ifft(G).real
        x = np.where(np.isfinite(x), x, 0.0)
        q = np.maximum(x, 0.0)
        s = float(q.sum())
        if not math.isfinite(s) or s <= 1e-15:
            q = p.copy()
        else:
            q /= s
        p = mix * p + (1.0 - mix) * q
        p = _normalize(np.maximum(p, 1e-12))
    return p


def _has_pos(w):
    try:
        return bool(np.sum(np.asarray(w, dtype=np.float64)) > 1e-12)
    except Exception:
        return False


def _active_weights(c, M, active, N):
    need = 2 * N - 1
    w = np.zeros(need, dtype=np.float64)
    if active.size == 0:
        return w
    vals = c[active]
    T = max(float(M) * 0.05, 1e-12)
    diff = (vals - M) / T
    diff = np.clip(diff, -50.0, 0.0)
    p = np.exp(diff)
    s = float(p.sum())
    if s > 0.0 and math.isfinite(s):
        soft = p / s
    else:
        soft = np.zeros(active.size, dtype=np.float64)
        soft[int(np.argmax(vals))] = 1.0

    K = min(active.size, max(8, min(64, max(8, N // 8))))
    if K < active.size:
        order = np.argpartition(vals, active.size - K)[active.size - K:]
    else:
        order = np.arange(active.size)
    top = np.zeros(active.size, dtype=np.float64)
    top[order] = 1.0 / float(K)
    uni = np.full(active.size, 1.0 / float(active.size), dtype=np.float64)

    wa = 0.75 * soft + 0.10 * uni + 0.15 * top
    sw = float(wa.sum())
    if sw > 0.0 and math.isfinite(sw):
        w[active] = wa / sw
    else:
        w[active] = uni
    return w


def _weighted_grad_fft(a, w):
    a = _clean(a)
    n = a.size
    need = 2 * n - 1
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64).ravel()
    if w.size != need:
        if w.size > need:
            w = w[:need]
        else:
            ww = np.zeros(need, dtype=np.float64)
            ww[:w.size] = w
            w = ww
    w = np.where(np.isfinite(w), w, 0.0)
    s = float(w.sum())
    if s <= 0.0 or not math.isfinite(s):
        return np.zeros(n, dtype=np.float64)
    w /= s

    nfft = next_fast_len(3 * n - 2)
    fa = rfft(a, nfft)
    fw = rfft(w[::-1], nfft)
    cr = irfft(fa * fw, nfft)
    cr = cr[:3 * n - 2]
    cr = np.where(np.isfinite(cr), cr, 0.0)
    idx = 2 * n - 2 - np.arange(n)
    g = 2.0 * cr[idx]
    np.maximum(g, 0.0, out=g)
    return g


def _build_A_rows(p, rows):
    p = _clean(p)
    N = p.size
    A = np.empty((rows.size, N), dtype=np.float64)
    for i in range(rows.size):
        k = int(rows[i])
        if k < N:
            A[i, :k + 1] = p[:k + 1][::-1]
        else:
            L = 2 * N - 1 - k
            s = k - N + 1
            A[i, s:s + L] = p[N - 1:N - L - 1:-1]
    return A


def _solve_lp_offzero(p, M, active, deadline, rng, need_dual=True):
    n = p.size
    rem = deadline - time.monotonic()
    if rem < 2.0 or M <= 1e-14 or active.size == 0 or n < 2:
        return None, None

    puse = np.asarray(p, dtype=np.float64).copy()
    puse += 1e-12
    s = float(puse.sum())
    if not math.isfinite(s) or s <= 1e-300:
        return None, None
    puse /= s

    A = _build_A_rows(puse, active)
    A = np.where(np.isfinite(A), A, 0.0)
    As = A / float(max(M, 1e-14))
    As = np.where(np.isfinite(As), As, 0.0)

    m = active.size
    dirb = None

    tl = max(0.25, min(8.0, 0.30 * rem))
    tie = rng.standard_normal(n)
    tie = (tie - tie.mean()) / (np.std(tie) + 1e-12)
    c_obj = -np.ones(n, dtype=np.float64) + 1e-6 * tie
    bounds_p = [(0.0, 1000.0)] * n
    try:
        res = linprog(
            c_obj,
            A_ub=As,
            b_ub=np.ones(m, dtype=np.float64),
            bounds=bounds_p,
            method="highs",
            options={"time_limit": float(tl), "presolve": True},
        )
        if getattr(res, "success", False):
            x = np.asarray(res.x, dtype=np.float64).ravel()
            x = np.where(np.isfinite(x), x, 0.0)
            x = np.clip(x, 0.0, 1000.0)
            sx = float(x.sum())
            if sx > 1e-12:
                dirb = x / sx
    except Exception:
        dirb = None

    wfull = None
    if need_dual and time.monotonic() < deadline - 2.0 and m > 0:
        rem = deadline - time.monotonic()
        if rem > 1.0:
            tie2 = rng.standard_normal(m)
            if m > 1:
                tie2 = (tie2 - tie2.mean()) / (np.std(tie2) + 1e-12)
            c_dual = np.ones(m, dtype=np.float64) + 1e-6 * tie2
            A_dual = -As.T
            b_dual = -np.ones(n, dtype=np.float64)
            tl2 = max(0.25, min(8.0, 0.30 * rem))
            bounds_d = [(0.0, None)] * m
            try:
                res2 = linprog(
                    c_dual,
                    A_ub=A_dual,
                    b_ub=b_dual,
                    bounds=bounds_d,
                    method="highs",
                    options={"time_limit": float(tl2), "presolve": True},
                )
                if getattr(res2, "success", False):
                    lam = np.asarray(res2.x, dtype=np.float64).ravel()
                    lam = np.where(np.isfinite(lam), lam, 0.0)
                    lam = np.maximum(lam, 0.0)
                    sl = float(lam.sum())
                    if sl > 1e-12:
                        wfull = np.zeros(2 * n - 1, dtype=np.float64)
                        wfull[active] = lam / sl
            except Exception:
                wfull = None

    return dirb, wfull


def _refine_off(start, deadline, rng, use_lp=True, cb=None):
    a = _normalize(start, 1.0)
    n = a.size
    if n < 2 or n > 2048:
        if cb is not None:
            try:
                cb(a)
            except Exception:
                pass
        return [a.copy()]

    C = n - 1
    active = np.concatenate((np.arange(0, C), np.arange(C + 1, 2 * n - 1))).astype(np.int64)

    def metrics(arr):
        c = _conv_full(arr)
        if active.size == 0:
            M = float(c.max()) if c.size else 0.0
        else:
            M = float(np.max(c[active]))
        if not math.isfinite(M) or M < 0.0:
            M = 1e-14
        M = max(M, 1e-14)
        return c, M, float(2.0 * n * M)

    def cb0(arr):
        if cb is not None:
            try:
                cb(arr)
            except Exception:
                pass

    c, M, surr = metrics(a)
    best = a.copy()
    best_surr = surr
    cb0(a)

    it = 0
    patience = 0
    sigma = 0.02

    while time.monotonic() < deadline - 4.0:
        it += 1
        rem = deadline - time.monotonic()
        if rem < 10.0:
            break
        if it % 24 == 0:
            cb0(a)

        if surr < best_surr * (1.0 - 1e-12):
            best = a.copy()
            best_surr = surr
            patience = 0
            if surr < best_surr * (1.0 - 1e-9):
                cb0(best)
        else:
            patience += 1

        if patience > 60:
            r = float(rng.random())
            if r < 0.40:
                a = _project_simplex(best + rng.normal(0.0, sigma, n))
            elif r < 0.75:
                a = 0.60 * best + 0.40 * _random_smooth(n, rng)
            else:
                a = _random_smooth(n, rng)
            c, M, surr = metrics(a)
            patience = 0
            sigma = min(0.35, sigma * 1.25)
            continue

        moved = False
        wuse = None

        if use_lp and rem > 25.0 and it % 3 == 0:
            subdeadline = time.monotonic() + max(1.0, min(10.0, 0.25 * rem))
            try:
                dirb, wlp = _solve_lp_offzero(a, M, active, subdeadline, rng, need_dual=True)
            except Exception:
                dirb, wlp = None, None

            if dirb is not None and np.all(np.isfinite(dirb)):
                for alpha in (0.02, 0.05, 0.10, 0.20, 0.35, 0.55, 0.80):
                    cand = (1.0 - alpha) * a + alpha * dirb
                    c2, M2, s2 = metrics(cand)
                    if math.isfinite(s2) and (s2 < best_surr * (1.0 - 1e-12) or s2 < surr * (1.0 - 1e-10)):
                        a = cand
                        c, M, surr = c2, M2, s2
                        moved = True
                        if s2 < best_surr * (1.0 - 1e-12):
                            best = a.copy()
                            best_surr = s2
                            cb0(best)
                        break

            if wuse is None and wlp is not None:
                wuse = wlp

        wsoft = _active_weights(c, M, active, n)
        if wuse is not None and _has_pos(wuse):
            wc = wuse + wsoft
            sc = float(wc.sum())
            if sc > 0.0 and math.isfinite(sc):
                wuse = wc / sc
            else:
                wuse = wsoft
        else:
            wuse = wsoft

        if _has_pos(wuse):
            grad = _weighted_grad_fft(a, wuse)
            gn = float(np.linalg.norm(grad, ord=1))
            if gn > 1e-14:
                gm = max(float(grad.mean()), 1e-15)
                for eta in (0.02, 0.05, 0.10, 0.20, 0.40, 0.80, 1.50):
                    e = np.exp(-eta * grad / gm)
                    e = np.clip(e, 1e-300, 1e3)
                    cand = a * e
                    cs = float(cand.sum())
                    if cs > 0.0 and math.isfinite(cs):
                        cand /= cs
                    c2, M2, s2 = metrics(cand)
                    if math.isfinite(s2) and (s2 < best_surr * (1.0 - 1e-12) or s2 < surr * (1.0 - 1e-10)):
                        a = cand
                        c, M, surr = c2, M2, s2
                        moved = True
                        if s2 < best_surr * (1.0 - 1e-12):
                            best = a.copy()
                            best_surr = s2
                            cb0(best)
                        break

                for eta in (0.02, 0.05, 0.10, 0.20, 0.40):
                    cand = _project_simplex(a - eta * grad)
                    c2, M2, s2 = metrics(cand)
                    if math.isfinite(s2) and (s2 < best_surr * (1.0 - 1e-12) or s2 < surr * (1.0 - 1e-10)):
                        a = cand
                        c, M, surr = c2, M2, s2
                        moved = True
                        if s2 < best_surr * (1.0 - 1e-12):
                            best = a.copy()
                            best_surr = s2
                            cb0(best)
                        break

        if not moved:
            cand = _project_simplex(a + rng.normal(0.0, sigma * 0.5, n))
            c2, M2, s2 = metrics(cand)
            if math.isfinite(s2):
                if s2 < best_surr:
                    best = cand.copy()
                    best_surr = s2
                    cb0(best)
                if s2 < surr * 1.05:
                    a = cand
                    c, M, surr = c2, M2, s2
                else:
                    sigma = min(0.35, sigma * 1.05)

    cb0(best)
    return [best]


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = max(0.0, float(budget_s))
    except Exception:
        budget = 1300.0

    try:
        seed_i = int(seed)
    except Exception:
        seed_i = 42
    try:
        if not math.isfinite(float(seed)):
            seed_i = 42
    except Exception:
        pass

    N = 512
    parent = None

    def quick_return_parent_or_uniform():
        h = globals().get("height_sequence_1", None)
        if h is not None:
            try:
                p = np.asarray(h, dtype=np.float64).ravel()
                if p.size > 0 and np.all(np.isfinite(p)):
                    p = np.clip(p, 0.0, 1000.0)
                    s = float(p.sum())
                    if s > 1e-300:
                        out = p * math.sqrt(2.0 * p.size) / s
                        out = np.clip(out, 0.0, 1000.0)
                        if out.size > 0 and np.all(np.isfinite(out)):
                            return [float(x) for x in out]
            except Exception:
                pass
        u = np.ones(N, dtype=np.float64) / float(N)
        u = u * math.sqrt(2.0 * N)
        return [float(x) for x in u]

    if budget < 15.0:
        return quick_return_parent_or_uniform()

    head = max(10.0, min(60.0, 0.04 * budget))
    deadline = t0 + budget - head
    if deadline <= t0 + 2.0:
        deadline = t0 + max(2.0, 0.9 * budget)

    rng = np.random.default_rng(seed_i)

    best_a = np.ones(N, dtype=np.float64) / float(N)
    best_sc = float("inf")

    def consider(a):
        nonlocal best_a, best_sc
        arr = _normalize(a, 1.0)
        if arr.size == 0:
            return
        sc = _score(arr)
        if np.isfinite(sc) and sc < best_sc - 1e-15:
            best_sc = float(sc)
            best_a = arr.copy()

    pool = []

    def consider_pool(a):
        arr = _normalize(a, 1.0)
        if arr.size > 0:
            pool.append(arr)
            consider(arr)

    h = globals().get("height_sequence_1", None)
    if h is not None:
        try:
            p = np.asarray(h, dtype=np.float64).ravel()
            if p.size > 0 and np.all(np.isfinite(p)):
                p = np.clip(p, 0.0, 1000.0)
                if float(p.sum()) > 1e-300:
                    parent = _normalize(p)
                    consider_pool(parent)
                    if parent.size != N:
                        consider_pool(_resample(parent, N))
        except Exception:
            parent = None

    base_uniform = np.ones(N, dtype=np.float64) / float(N)
    consider_pool(base_uniform)

    for L in (768, 1024):
        consider_pool(np.ones(L, dtype=np.float64) / float(L))

    x = np.linspace(0.0, 1.0, N, endpoint=False)
    ar = np.arange(N)
    for beta in (0.25, 0.50, 0.75):
        for m in (1, 2, 3):
            aa = 1.0 + beta * np.cos(2.0 * np.pi * m * x)
            consider_pool(_normalize(np.maximum(aa, 1e-6)))
        sign = np.where((ar % 2) == 0, 1.0, -1.0)
        aa = 1.0 + beta * sign
        consider_pool(_normalize(np.maximum(aa, 1e-6)))

    for _ in range(10):
        consider_pool(_random_smooth(N, rng))

    if deadline - time.monotonic() > 120.0:
        spec_starts = [base_uniform]
        if parent is not None:
            spec_starts.append(_resample(parent, N))
        spec_starts.extend(pool[:8])
        for st in spec_starts[:5]:
            for rho in (0.7, 1.0, 1.3):
                consider_pool(_spectral_flat(st, rng, rho, iters=60, mix=0.35))

    if deadline - time.monotonic() > 180.0:
        for _ in range(6):
            consider_pool(_random_smooth(N, rng, base=_resample(best_a, N)))

    off_active = np.concatenate(
        (np.arange(0, N - 1), np.arange(N, 2 * N - 1))
    ).astype(np.int64)

    def _off_surr(a):
        a = _resample(a, N)
        c = _conv_full(a)
        if off_active.size == 0:
            M = float(c.max()) if c.size else 0.0
        else:
            M = float(np.max(c[off_active]))
        if not math.isfinite(M) or M < 0.0:
            M = 1e-14
        return float(2.0 * N * M)

    starts = []
    seen = []

    def add_start(a):
        aa = _resample(a, N)
        if aa.size != N:
            return
        for bb in seen:
            try:
                if np.linalg.norm(aa - bb) < 1e-5:
                    return
            except Exception:
                pass
        seen.append(aa.copy())
        starts.append(aa.copy())

    if parent is not None:
        add_start(parent)
    add_start(base_uniform)

    scored = []
    for a in pool:
        if a.size == N:
            try:
                scored.append((_off_surr(a), a))
            except Exception:
                pass
    scored.sort(key=lambda z: z[0])
    for _, a in scored:
        add_start(a)
        if len(starts) >= 6:
            break

    if not starts:
        starts = [base_uniform.copy()]

    fracs = [0.50, 0.25, 0.15, 0.08, 0.03]
    for idx, st in enumerate(starts[:5]):
        rem = deadline - time.monotonic()
        if rem < 80.0:
            break
        frac = fracs[idx] if idx < len(fracs) else 0.03
        t = min(frac * rem, rem - 60.0)
        if t < 45.0:
            t = min(45.0, rem - 20.0)
        if t > 20.0:
            try:
                _refine_off(st, time.monotonic() + t, rng, use_lp=True, cb=consider)
            except Exception:
                pass

    if deadline - time.monotonic() > 120.0 and best_a is not None:
        t = min(140.0, deadline - time.monotonic() - 30.0)
        if t > 40.0:
            try:
                _refine_off(_resample(best_a, N), time.monotonic() + t, rng, use_lp=True, cb=consider)
            except Exception:
                pass

    final = _normalize(best_a if best_a is not None else base_uniform, 1.0)
    if final.size == 0:
        final = base_uniform.copy()

    target_sum = math.sqrt(2.0 * float(final.size))
    if not math.isfinite(target_sum) or target_sum <= 0.0:
        target_sum = 1.0
    final = _normalize(final, target_sum)
    final = np.clip(final, 0.0, 1000.0)

    fsc = _score(final)
    if not np.isfinite(fsc):
        final = base_uniform.copy() * math.sqrt(2.0 * float(N))
        fsc = _score(final)
    if not np.isfinite(fsc):
        return quick_return_parent_or_uniform()

    return [float(x) for x in final]
```
