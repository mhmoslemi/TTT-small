def _conv_fft(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_fft(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)

def _normalize(a):
    s = float(a.sum())
    if s < 1e-10:
        return np.ones(len(a)) / len(a)
    return a / s

def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

def _build_conv_matrix(f, n):
    """Build A such that A @ g = f * g (full, length 2n-1)."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j+n, j] = f
    return A

def _lp_solve(f, n, M, time_limit=5.0):
    A = _build_conv_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(
            c, A_ub=A, b_ub=b_ub,
            bounds=[(0.0, None)] * n,
            method="highs",
            options={"time_limit": time_limit, "presolve": True},
        )
    except Exception:
        return None
    if res.status not in (0, 3) or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.maximum(g, 0)
    gs = float(g.sum())
    if gs < 1e-14:
        return None
    return g / gs

            lo = t_mid

def _subgrad_loop(f0, n, stop_time, seed):

    f = _normalize(np.maximum(np.asarray(f0, dtype=np.float64).ravel(), 0))

    tau = 20.0  # temperature for log-sum-exp
    no_imp = 0

            break
        
        b = _conv_fft(f)
        bm = b.max()
        
        # Log-sum-exp weights
        shifted = tau * (b - bm)
        shifted -= shifted.max()
        weights = np.exp(shifted)
        wsum = weights.sum()
        if wsum < 1e-30:
            weights = np.zeros(2*n-1)
            weights[int(np.argmax(b))] = 1.0
        else:
            weights /= wsum
        
        # Compute gradient (top-K lags for efficiency)
        topk = min(20, 2*n-1)
        topk_idx = np.argsort(b)[-topk:][::-1]
        topk_w = weights[topk_idx]
        
        grad = np.zeros(n)
        for k, w in zip(topk_idx, topk_w):
            k = int(k)
            # d(a*a)[k]/da[j] = a[k-j] (if valid) + a[j-k] (if valid)
            lo1 = max(0, k - n + 1)
            hi1 = min(n - 1, k)
            if lo1 <= hi1:

            lo2 = max(0, k)
            hi2 = n - 1
            if lo2 <= hi2:
                js2 = np.arange(lo2, hi2 + 1)
                grad[js2] += w * f[js2 - k]
        
        # Nesterov momentum step
        momentum = beta * momentum - lr * grad
        f_new = f + momentum
        f_new = _project_simplex(f_new)
        
        sc = _score(f_new)
        if sc < best_sc - 1e-14:

                beta = 0.8

def _project_simplex(v):

    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

    for fr, amp in [(1, 0.35), (2, 0.25), (3, 0.18), (5, 0.12), (7, 0.08), (11, 0.05)]:
        mc += amp * np.cos(2 * math.pi * fr * i_arr / n_p + rng_main.uniform(0, 2 * math.pi))
    inits.append(("multicos", np.abs(mc) + 0.05))

    gauss = np.exp(-((i_arr - n_p / 2) ** 2) / (2 * (n_p * 0.12) ** 2)) + 0.03
    inits.append(("gauss", gauss))

    noise = rng_main.normal(0, 0.35 * max(1.0, np.mean(parent)), n_p)
    inits.append(("pnoise", np.abs(parent + noise) + 0.01))

    tw = 0.5 + 0.4 * np.cos(4 * math.pi * i_arr / n_p)
    inits.append(("twohump", tw))

    sp = np.abs(rng_main.normal(0, 1, n_p))
    mask = rng_main.random(n_p) < 0.25
    sp[mask] = 0.0
    inits.append(("sparse", sp + 0.02))

    # Parent with random sign flips (take abs after)
    signs = rng_main.choice([-1, 1], n_p)
    inits.append(("psign", np.abs(parent * signs[::1]) + 0.01))

    # Sinc-like
    sinc = np.sinc((i_arr - n_p / 2) / (n_p * 0.25))
    inits.append(("sinc", np.abs(sinc) + 0.05))

    # LP phase
    lp_end = t0 + budget * 0.48
    n_inits = len(inits)
    base_time = max(25.0, (lp_end - t0) / (n_inits * 1.5))

    for idx, (name, init) in enumerate(inits):
        if time.monotonic() > lp_end - 8.0:
            break
        if best_score < 1.5035:
            break

        sc_before = best_score
        stop = min(lp_end - 4.0, time.monotonic() + base_time)
        try:
            res, sc = _lp_loop(init, n_p, stop, seed + 7919 * (idx + 1))
            _try_update(res)
        except Exception:
            pass

        # If improved, continue from best for more time
        if best_score < sc_before - 1e-5 and time.monotonic() < lp_end - 10.0:
            extra_stop = min(lp_end - 4.0, time.monotonic() + base_time)
            try:
                res2, sc2 = _lp_loop(best_seq, n_p, extra_stop, seed + 31337 * (idx + 1))
                _try_update(res2)
            except Exception:
                pass

    # Try a few other lengths if time permits and we haven't beaten target
    if time.monotonic() < deadline * 0.75 and best_score > 1.5035:
        other_lengths = [320, 256, 300, 288, 350]
        for L in other_lengths:
            if time.monotonic() > deadline * 0.80:
                break
            if L == n_p:
                continue
            stop = min(deadline * 0.82, time.monotonic() + 30.0)
            init_l = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, n_p), parent)
            try:
                res, sc = _lp_loop(init_l, L, stop, seed + 50000 + L)
                _try_update(res)
            except Exception:
                pass

    # Subgradient phase
    sg_end = min(deadline, t0 + budget * 0.78)
    if time.monotonic() < sg_end - 10.0:
        try:
            res, sc = _subgrad_loop(best_seq, len(best_seq), sg_end, seed + 55501)
            _try_update(res)
        except Exception:
            pass

    # Coordinate descent
    cd_end = min(deadline, t0 + budget * 0.92)
    if time.monotonic() < cd_end - 8.0:
        try:
            res, sc = _coord_descent(best_seq, len(best_seq), cd_end, seed + 33301)
            _try_update(res)
        except Exception:
            pass

    # Final subgradient burst on best
    if time.monotonic() < deadline - 8.0:
        try:
            res, sc = _subgrad_loop(best_seq, len(best_seq), deadline - 5.0, seed + 11101)
            _try_update(res)
        except Exception:
            pass

    # Final coordinate polish
    if time.monotonic() < deadline - 4.0:
        try:
            res, sc = _coord_descent(best_seq, len(best_seq), deadline - 2.0, seed + 22201)
            _try_update(res)
        except Exception:
            pass

    # Final validation with exact evaluator
    final = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    try:
        sc = evaluate_sequence(final)
        if math.isfinite(sc) and sc > 0:
            return final
    except Exception:
        pass

    # Fallback to parent
    fb = [float(max(0.0, min(1000.0, x))) for x in parent]
    try:
        sc = evaluate_sequence(fb)
        if math.isfinite(sc) and sc > 0:
            return fb
    except Exception:
        pass

    return [1.0] * n_p
