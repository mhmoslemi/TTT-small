    """
    Compute smooth max of f*f and its gradient w.r.t. f.

    Optimize x (log-parameters) to minimize smooth max of softmax(x)^2.
    """
    n = len(x0)
    rng = np.random.default_rng(seed)
    
    def softmax(x):
        x_shifted = x - x.max()
        e = np.exp(x_shifted)
        return e / e.sum()

import numpy as np
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import minimize, linprog

def _get_parent():
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(198, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)

def _conv(a):
    n = len(a)
    fs = next_fast_len(2 * n - 1)
    F = rfft(a, fs)
    c = irfft(F * F, fs)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _smooth_max_grad(f, alpha):
    """Compute smooth max of f*f (full conv) and gradient w.r.t. f."""
    n = len(f)
    fs = next_fast_len(2 * n - 1)
    F = rfft(f, fs)
    g = irfft(F * F, fs)[:2 * n - 1]
    np.maximum(g, 0.0, out=g)

    max_g = float(g.max())
    if max_g <= 1e-30:
        return 0.0, np.zeros(n)

    shifted = alpha * (g - max_g)
    exp_g = np.exp(shifted)
    sum_exp = float(exp_g.sum())
    smooth_val = max_g + math.log(sum_exp) / alpha

    w = exp_g / sum_exp

    # Gradient: h[j] = sum_k w[k]*f[k-j] for valid indices
    # h[j] = (f * w_rev)[2n-2-j]
    w_rev = w[::-1]
    fs2 = next_fast_len(n + (2 * n - 1) - 1)
    Ff = rfft(f, fs2)
    Fwr = rfft(w_rev, fs2)
    conv_fw = irfft(Ff * Fwr, fs2)[:3 * n - 2]
    h = conv_fw[n - 1:2 * n - 1][::-1]

    grad = 2.0 * h
    return smooth_val, grad


def _simplex_project(v, s=1.0):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - s)
    if not np.any(cond):
        return np.full(n, s / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax(x):
    xm = x - x.max()
    e = np.exp(xm)
    s = e.sum()
    if s < 1e-300:
        return np.full(len(x), 1.0 / len(x))
    return e / s


def _lbfgs_optimize(f0, stop_time, seed=42, alphas=(30.0, 100.0, 300.0)):
    """Optimize via L-BFGS-B with softmax reparameterization and smooth max."""
    n = len(f0)
    f0 = np.clip(np.asarray(f0, dtype=np.float64), 0, 1000)
    s = f0.sum()
    if s < 1e-10:
        f0 = np.ones(n) / n
    else:
        f0 = f0 / s

    best = {'score': _score(f0), 'f': f0.copy()}
    rng = np.random.default_rng(seed)

    def lbfgs_run(x0, stop_t, alpha):
        timeout_flag = [False]

        def obj_grad(x):
            f = _softmax(x)
            val, g = _smooth_max_grad(f, alpha)
            fg = float(np.dot(f, g))
            grad_x = f * (g - fg)
            sc = _score(f)
            if sc < best['score'] - 1e-10:
                best['score'] = sc
                best['f'] = f.copy()
            if time.monotonic() > stop_t:
                timeout_flag[0] = True
            return val, grad_x

        def cb(x):
            if timeout_flag[0]:
                raise RuntimeError("stop")

        try:
            minimize(obj_grad, x0, jac=True, method='L-BFGS-B',
                     options={'maxiter': 50000, 'maxls': 40, 'gtol': 1e-9,
                              'factr': 1e7, 'maxfun': 50000},
                     callback=cb)
        except Exception:
            pass

    # Multi-alpha schedule
    time_left = stop_time - time.monotonic()
    splits = [0.3, 0.4, 0.3]
    cur_x = np.log(np.maximum(f0, 1e-15))
    for ai, (alpha, frac) in enumerate(zip(alphas, splits)):
        seg_stop = time.monotonic() + time_left * frac
        if time.monotonic() >= seg_stop - 2.0:
            break
        time_left2 = seg_stop - time.monotonic()
        if time_left2 < 3.0:

        lbfgs_run(cur_x, seg_stop - 1.0, alpha)
        cur_x = np.log(np.maximum(best['f'], 1e-15))
        if time.monotonic() >= stop_time - 5.0:

    return best['f'], best['score']


def _proj_grad_optimize(f0, stop_time, seed=42):

    n = len(f0)
    f = _simplex_project(np.clip(f0, 0, 1000), 1.0)

    best_f = f.copy()

    mu = 0.9
    v = np.zeros(n)
    stall = 0
    max_stall = 300
    it = 0
    alpha = 50.0

        alpha = 30.0 + 270.0 * progress

        # Nesterov
        x_y = _simplex_project(f - lr * v, 1.0)

        else:

    n = len(f0)
    f = _simplex_project(np.clip(f0, 0, 1000), 1.0)
    best_f = f.copy()
    best_sc = _score(f)

    t_step = 0.3
    stall = 0
    max_stall = 80
    it = 0

    A_rows = 2 * n - 1
    # Build A lazily

    for j in range(n):

    def rebuild_A(fv):
        A_new = np.zeros((A_rows, n))
        for j in range(n):
            A_new[j:j + n, j] = fv
        return A_new

    while time.monotonic() < stop_time - 4.0:
        if stall >= max_stall:
            # Perturbation
            strat = stall % 4
            if strat == 0:
                f = _simplex_project(f + rng.normal(0, 0.08, n), 1.0)
            elif strat == 1:

                f2 = f.copy()
                f2[:blk], f2[n-blk:] = f[n-blk:].copy(), f[:blk].copy()
                f = _simplex_project(f2, 1.0)
            elif strat == 2:

                kern /= kern.sum()

            b_cur = _conv(f)
            M = float(b_cur.max())
            t_step = 0.3
            stall = 0
            continue

        # LP: max sum(g) s.t. A@g <= M, g >= 0
        A = rebuild_A(f)
        # Focus on top constraints for speed if n large
        if n > 200:
            top_k = min(120, 2 * n - 1)
            b_full = A @ f  # = conv(f,f)
            idx = np.argsort(-b_full)[::-1]
            idx = idx[:top_k]
            A_sub = A[idx]
            b_ub_sub = np.full(top_k, M)
        else:
            A_sub = A
            b_ub_sub = np.full(A_rows, M)

        c = -np.ones(n)
        try:
            res = linprog(c, A_ub=A_sub, b_ub=b_ub_sub,
                          bounds=[(0, None)] * n,
                          method='highs',
                          options={'presolve': True, 'time_limit': 15.0})
        except Exception:
            stall += 3
            continue

        if res.status not in (0, 3) or res.x is None:
            stall += 3
            continue

        g = np.asarray(res.x, dtype=np.float64)
        g_sum = float(g.sum())
        if g_sum < 1e-12:
            stall += 3
            continue
        g = g / g_sum

        # Check if g differs from f
        dist = float(np.linalg.norm(g - f))
        if dist < 1e-8:
            stall += 3
            continue

        # Line search
        improved = False
        for _ in range(14):
            f_new = _simplex_project((1.0 - t_step) * f + t_step * g, 1.0)
            sc = _score(f_new)
            if sc < best_sc - 1e-12:
                f = f_new
                best_f = f.copy()
                best_sc = sc
                b_cur = _conv(f)
                M = float(b_cur.max())
                t_step = min(0.7, t_step * 1.06)
                improved = True
                stall = 0
                break
            else:
                t_step = max(0.03, t_step * 0.5)

        if not improved:
            stall += 1

        it += 1

    return best_f, best_sc
