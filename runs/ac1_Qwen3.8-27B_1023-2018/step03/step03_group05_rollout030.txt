    # softmax Jacobian: da/dx[i] * F_a = (a_i * (delta_ij - a_j)) * g_a[j]
    # = a_i * g_a[i] - a_i * sum_j a_j * g_a[j]
    # = a_i * (g_a[i] - a @ g_a)
    ag = a @ g_a
    g_x = a * (g_a - ag)
    return g_x
