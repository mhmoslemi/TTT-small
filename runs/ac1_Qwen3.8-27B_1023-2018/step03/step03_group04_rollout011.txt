    n = len(f)
    conv = _conv(f)

        # Soft max
        max_c = conv.max()
        exps = np.exp(beta * (conv - max_c))
        weights = exps / exps.sum()
        # grad = sum_k weights[k] * grad(f*f)[k]
        grad = np.zeros(n)
        for k in range(2*n-1):
            if weights[k] < 1e-10:
                continue
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            # d/d f_i (f*f)[k] = 2*f[k-i] for i in [lo, hi]
            # So grad_i = 2*f[k-i] for i in [lo,hi]
            for i in range(lo, hi+1):
                grad[i] += weights[k] * 2.0 * f[k-i]
        return grad, conv, max_c

        M = conv.max()
        active = np.where(conv >= M - 1e-10 * M)[0]
        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            for i in range(lo, hi+1):
                grad[i] += 2.0 * f[k-i]
        grad /= len(active)
        return grad, conv, M
