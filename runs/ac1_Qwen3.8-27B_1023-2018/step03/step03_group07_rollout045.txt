import numpy as np
import time
from scipy.optimize import linprog

def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = float(budget_s)
    deadline = t0 + budget - 5.0
    
    # Get parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
    except:
        parent = np.ones(128, dtype=np.float64) / 128.0
    
    parent = np.clip(np.asarray(parent, dtype=np.float64), 0, 1000.0)
    if parent.size == 0 or not np.isfinite(parent).all() or parent.sum() < 0.01:
        parent = np.ones(128, dtype=np.float64) / 128.0
    
    rng = np.random.default_rng(seed)
    best_a = None
    best_score = float('inf')
    
    def score_of(a):
        a = np.clip(a, 0, 1000.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        c = np.convolve(a, a)
        return 2.0 * a.size * c.max() / (s * s)
    
    def update_best(a):
        nonlocal best_a, best_score
        a = np.clip(a, 0, 1000.0)
        if a.sum() < 0.01:
            return False
        sc = score_of(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
            return True
        return False
    
    def build_A(a):
        n = a.size
        m = 2 * n - 1
        k = np.arange(m)[:, None].astype(np.int64)
        j = np.arange(n)[None, :].astype(np.int64)
        idx = k - j
        mask = (idx >= 0) & (idx < n)
        safe_idx = np.clip(idx, 0, n - 1)
        A = np.where(mask, a[safe_idx], 0.0)
        return A
    
    def solve_lp(A, M, n):
        """max sum(g) s.t. A@g <= M, g >= 0"""
        c_obj = -np.ones(n)
        b_ub = np.full(A.shape[0], M)
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                         bounds=[(0.0, None)] * n,
                         method='highs')
        except:
            return None
        if res.status != 0:
            return None
        g = np.asarray(res.x, dtype=np.float64)
        gs = g.sum()
        if gs < 1e-15 or not np.isfinite(g).all():
            return None
        return g / gs  # normalize to sum=1
    
    def linesearch(a, g, M, n):
        """Find alpha in [0,1] minimizing max(conv((1-a)·a + a·g, ...))"""
        # Coarse search
        alphas = np.linspace(0.05, 1.0, 20)
        best_alpha = 0.0
        best_val = M
        for alpha in alphas:
            cand = (1.0 - alpha) * a + alpha * g
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s < 1e-12:
                continue
            cand /= s
            val = np.convolve(cand, cand).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha == 0:
            return None, M
        
        # Golden section refinement
        if best_alpha > 0.05 and best_alpha < 1.0:
            lo = max(0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(15):
                m1 = lo + 0.382 * (hi - lo)
                m2 = lo + 0.618 * (hi - lo)
                c1 = (1 - m1) * a + m1 * g
                c1 = np.maximum(c1, 0); c1 /= c1.sum()
                v1 = np.convolve(c1, c1).max()
                c2 = (1 - m2) * a + m2 * g
                c2 = np.maximum(c2, 0); c2 /= c2.sum()
                v2 = np.convolve(c2, c2).max()
                if v1 < v2:
                    hi = m2
                else:
                    lo = m1
                if hi - lo < 1e-6:
                    break
            best_alpha = (lo + hi) / 2
            cand = (1 - best_alpha) * a + best_alpha * g
            cand = np.maximum(cand, 0)
            cand /= cand.sum()
            best_val = np.convolve(cand, cand).max()
        
        if best_val < M * (1 - 1e-10) and best_alpha > 0:
            cand = (1 - best_alpha) * a + best_alpha * g
            cand = np.maximum(cand, 0)
            cand /= cand.sum()
            return cand, best_val
        return None, M
    
    def optimize(a_init, stop_time, n_avg_perturb=100):
        """Main LP-based optimization loop."""
        a = np.clip(a_init, 0, 1000.0).copy()
        s = a.sum()
        if s < 1e-12:
            a = np.ones_like(a) / a.size
        a /= s
        n = a.size
        
        stuck = 0
        iter_count = 0
        
        while time.monotonic() < stop_time - 3.0:
            c = np.convolve(a, a)
            M = float(c.max())
            k_star = int(c.argmax())
            
            score = 2.0 * n * M  # since sum(a)=1
            
            # Check improvement
            full_score = 2.0 * n * M / 1.0  # sum=1
            if full_score < best_score * (1 - 1e-13):
                stuck = 0
            else:
                stuck += 1
            
            # Periodic perturbation
            if stuck > n_avg_perturb:
                scale = 0.02 * (1 + 0.1 * (stuck // n_avg_perturb))
                a = a * np.exp(rng.normal(0, min(scale, 0.2), n))
                a = np.maximum(a, 1e-10)
                a /= a.sum()
                stuck = 0
                continue
            
            # Build constraints and solve LP
            A = build_A(a)
            M_lp = M * (1.0 + 1e-8)  # slight relaxation
            g = solve_lp(A, M_lp, n)
            
            if g is None:
                # LP failed, try gradient
                grad = np.zeros(n)
                # Subgradient at k_star
                j_lo = max(0, k_star - n + 1)
                j_hi = min(n - 1, k_star)
                for j in range(j_lo, j_hi + 1):
                    grad[j] += 2.0 * a[k_star - j]
                grad -= grad.mean()  # project to sum-0
                gn = np.linalg.norm(grad)
                if gn > 1e-12:
                    a = a + 0.01 * grad / gn
                    a = np.maximum(a, 1e-10)
                    a /= a.sum()
                continue
            
            # Line search
            new_a, new_val = linesearch(a, g, M, n)
            if new_a is not None:
                a = new_a
            else:
                # No improvement from LP direction, small perturbation
                a = a * np.exp(rng.normal(0, 0.01, n))
                a = np.maximum(a, 1e-10)
                a /= a.sum()
            
            iter_count += 1
        
        return a
    
    # Phase 1: Optimize from parent
    a_start = parent.copy()
    a_start = np.clip(a_start, 0, 1000.0)
    a_start /= a_start.sum()
    
    phase1_end = t0 + min(budget * 0.4, 500.0)
    result = optimize(a_start, phase1_end, n_avg_perturb=80)
    update_best(result)
    
    # Phase 2: Try alternative initializations
    inits = []
    n_p = parent.size
    x = np.arange(n_p, dtype=np.float64)

    for p in [0.1, 0.3, 0.5, 1.0, 2.0, 3.0]:
        seq = np.maximum(1.0 - x / max(1, n_p - 1), 1e-8) ** p
        inits.append(seq / seq.sum())
    
    for alpha in [0.5, 1.0, 2.0, 3.0, 5.0]:
        seq = np.exp(-alpha * x / n_p)
        inits.append(seq / seq.sum())
    
    for width in [0.1, 0.2, 0.3, 0.5]:
        center = (n_p - 1) / 2.0
        seq = np.exp(-0.5 * ((x - center) / (width * n_p / 2)) ** 2)
        inits.append(seq / seq.sum())
    
    # Lopsided sequences
    for frac in [0.3, 0.5, 0.7]:
        seq = np.ones(n_p)
        seq[:int(n_p*frac)] = 2.0
        seq[int(n_p*frac):] = 0.5
        inits.append(seq / seq.sum())
    
    # Blend parent with inits
    for init in inits[:3]:
        inits.append((0.7 * a_start + 0.3 * init) / (0.7 * a_start + 0.3 * init).sum())
    
    phase2_end = t0 + min(budget * 0.65, 800.0)
    time_for_phase2 = max(0, phase2_end - time.monotonic())
    if time_for_phase2 > 20:
        per_init = time_for_phase2 / len(inits)
        per_init = max(3.0, min(per_init, 60.0))
        for idx, init in enumerate(inits):

            alloc_end = min(time.monotonic() + per_init, phase2_end - 3)
            try:

                update_best(result)
            except:
                pass

    if best_a is not None and time.monotonic() < deadline - 15:

        a_final /= a_final.sum()
