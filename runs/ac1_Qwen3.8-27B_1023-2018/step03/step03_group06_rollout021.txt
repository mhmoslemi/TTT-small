    L = _fft_len(m)
    c = irfft(rfft(a, L) * rfft(b, L), L)[:m]

            best_sc = sc_e
    
    # Phase 1: LP optimization at parent length
    n = n_parent
    lp_stop = min(deadline - 120.0, t0 + 0.45 * budget)
    arr, _ = _lp_loop(best, n, lp_stop, seed=int(seed))
    try_candidate(arr)
    
    # Phase 2: L-BFGS refinement
    lbfgs_stop = min(deadline - 80.0, time.monotonic() + 80.0)
    if lbfgs_stop - time.monotonic() > 20.0:
        arr, _ = _lbfgs_refine(best, lbfgs_stop, eta=150.0)
        try_candidate(arr)
    
    # Phase 3: More LP iterations
    lp2_stop = min(deadline - 60.0, time.monotonic() + 100.0)
    if lp2_stop - time.monotonic() > 15.0:
        arr, _ = _lp_loop(best, n, lp2_stop, seed=int(seed) + 1)
        try_candidate(arr)
    
    # Phase 4: L-BFGS with higher eta
    lbfgs2_stop = min(deadline - 40.0, time.monotonic() + 50.0)
    if lbfgs2_stop - time.monotonic() > 15.0:
        arr, _ = _lbfgs_refine(best, lbfgs2_stop, eta=400.0)
        try_candidate(arr)
    
    # Phase 5: Try alternative lengths
    cur_n = best.size
    other_lengths = [n * 2, n // 2, 512, 128]
    other_lengths = [L for L in other_lengths if 32 <= L <= 1024 and L != cur_n]
    other_lengths = sorted(set(other_lengths))[:3]
    
    other_stop = min(deadline - 30.0, time.monotonic() + 60.0)
    for L in other_lengths:
        if time.monotonic() > other_stop - 10.0:
            break
        arr = _interp(best, L)
        stopL = min(other_stop - 5.0, time.monotonic() + 30.0)
        if stopL - time.monotonic() > 10.0:
            arr, _ = _lp_loop(arr, L, stopL, seed=int(seed) + L)
            try_candidate(arr)
    
    # Phase 6: Final L-BFGS + coordinate polish at best length
    n_best = best.size
    fin_stop = deadline - 5.0
    
    lbfgs3_stop = min(fin_stop - 20.0, time.monotonic() + 40.0)
    if lbfgs3_stop - time.monotonic() > 10.0:
        arr, _ = _lbfgs_refine(best, lbfgs3_stop, eta=800.0)
        try_candidate(arr)
    
    coord_stop = min(fin_stop, time.monotonic() + 25.0)
    if coord_stop - time.monotonic() > 5.0:
        arr, _ = _coord_polish(best, coord_stop, seed=int(seed) + 999)
        try_candidate(arr)
    
    # Final validation
    seq = np.clip(np.asarray(best, dtype=np.float64).ravel(), 0.0, 1000.0)
    if seq.size == 0 or seq.sum() < 0.01:
        seq = np.ones(32) / 32.0
    lst = [float(x) for x in seq]
    sc_final = _exact(lst)
    if not math.isfinite(sc_final):
        # Fallback to parent
        lst = [float(x) for x in parent]
    
    return lst
