    n = len(a)
    m = len(b)
    fsize = next_fast_len(n + m - 1)
    F = rfft(a, fsize)

    b = _convolve(a)
    m = float(b.max())

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(384, dtype=np.float64)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    return p

def _autocorr(a):
    n = len(a)
    fs = next_fast_len(2*n-1)
    F = rfft(a, fs)
    return irfft(F*F, fs)[:2*n-1]

def _score(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return math.inf
    m = _autocorr(a).max()
    return 2.0*n*m/(s*s)

def _build_A(f, n):
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        jlo = max(0, k-n+1)
        jhi = min(n-1, k)
        lags = k - np.arange(jlo, jhi+1)
        A[k, jlo:jhi+1] = f[lags]
    return A

def _lp_direction(f, n, M):
    A = _build_A(f, n)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M),
                  bounds=[(0, None)]*n, method='highs',
                  options={'presolve': True, 'time_limit': 20.0})
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    gs = g.sum()
    if gs < 1e-14:
        return None
    return g / gs

def _line_search(f, g, t_init, best_sc):
    # Try t_init first
    fn = (1-t_init)*f + t_init*g
    fn = np.maximum(fn, 0)
    fs = fn.sum()
    if fs < 1e-10:
        return None
    fn = fn / fs
    sc = _score(fn)
    if sc < best_sc:
        return sc, fn
    # Binary search
    tlo, thi = 0.0, t_init
    for _ in range(12):
        tm = 0.5*(tlo+thi)
        fn = np.maximum((1-tm)*f + tm*g, 0)

        else:
            thi = tm

    if tlo < 1e-6:

def _subgrad_polish(f0, n, stop, seed):
    rng = np.random.default_rng(seed)
    f = np.clip(f0.copy(), 0, None)
    f = f / f.sum()
    best_f = f.copy()
    best_sc = _score(f)
    lr = 0.05
    beta = 0.8
    v = np.zeros(n)
    no_imp = 0
    
    while time.monotonic() < stop - 1.0:
        b = _autocorr(f)
        M = b.max()
        # Find top K active lags
        K = min(8, 2*n-1)
        idx = np.argpartition(b, -K)[-K:]
        w = np.exp(b[idx] - b[idx].max())  # softmax weights
        w /= w.sum()
        # Subgradient of max at active lags
        grad = np.zeros(n)
        for wi, k in zip(w, idx):
            k = int(k)
            jlo = max(0, k-n+1)
            jhi = min(n-1, k)
            lags = k - np.arange(jlo, jhi+1)
            grad[jlo:jhi+1] += wi * f[lags]
        # Gradient of score = 2n*max_b/s^2
        s = f.sum()
        grad_score = (2.0*n * (2*grad)) / (s*s) - (4.0*n*M / (s**3)) * np.ones(n)
        # Normalize (score is scale-invariant, so we project gradient)
        grad_score -= grad_score.mean()  # Remove mean component (direction of scaling)
        # Update
        v = beta*v - lr*grad_score
        fn = f + v
        fn = np.maximum(fn, 0)
        fs = fn.sum()
        if fs < 1e-10:
            fn = f.copy()
        else:
            fn = fn / fs
        sc = _score(fn)
        if sc < best_sc - 1e-14:
            f = fn
            best_f = f.copy()
            best_sc = sc
            no_imp = 0
            lr = min(0.1, lr*1.005)
        else:
            no_imp += 1
            lr = max(0.005, lr*0.98)
            if no_imp > 200:
                f = fn.copy()  # Accept anyway to escape
                no_imp = 0
                lr = 0.03
                v *= 0.1
    return best_f, best_sc
