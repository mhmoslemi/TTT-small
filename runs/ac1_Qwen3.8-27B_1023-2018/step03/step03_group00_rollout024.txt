def _get_parent():

        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        raise
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    return p

def _conv_fftf(d):
    """Fast full convolution of d with d using FFT."""
    n = len(d)
    if n <= 128:
        return np.convolve(d, d, mode='full')
    fsize = next_fast_len(2*n - 1)
    F = rfft(d, fsize)
    c = irfft(F * F, fsize)[:2*n-1]
    np.maximum(c, 0.0, out=c)
    return c

def _score(d):
    n = len(d)
    s = float(d.sum())
    if s < 0.01:
        return math.inf
    b = _conv_fftf(d)
    m = float(b.max())
    return 2.0 * n * m / (s * s)

def _build_toeplitz(f, n):
    """Build the (2n-1) x n Toeplitz matrix for (f*g) computation."""
    # (f*g)_k = sum_j f[k-j] * g_j for valid j
    # Using Toeplitz structure
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = f
    return A

def _lp_direction(f, n, M, timeout=5.0):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M, g >= 0."""
    A = _build_toeplitz(f, n)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0.0, 1000.0)
    if g.sum() < 1e-15:

    return g

import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        raise
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0 or float(p.sum()) < 0.01:
        p = np.ones(200)
    return p


def _conv2(a):
    """Full autocorrelation via FFT."""
    n = len(a)
    if n <= 96:
        return np.convolve(a, a, mode='full')
    fs = next_fast_len(2 * n - 1)
    F = rfft(a, fs)
    c = irfft(F * F, fs)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv2(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _toeplitz(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _solve_lp(f, n, M, timeout=3.0):
    A = _toeplitz(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs',
                      options={'presolve': True, 'time_limit': timeout})
    except Exception:
        return None
    if res is None or res.status != 0 or res.x is None:
        return None
    g = np.clip(res.x, 0.0, 1000.0)
    if g.sum() < 1e-15:
        return None
    return g


def _lp_loop(f0, n, stop, seed):

        if time.monotonic() > stop - 5:
            break
        b = _conv2(f)
        M = b.max()
        if M < 1e-20:
            break

        g = _solve_lp(f, n, M, timeout=3.0)
        if g is None:
            stall += 5
            if stall > max_stall:
                f = np.clip(f + rng.normal(0, 0.05 * max(1.0, f.mean()), n), 0, None)
                stall, max_stall = 0, max_stall + 50
            continue

        gs = g * (f.sum() / g.sum())
        d = np.linalg.norm(gs - f) / max(1.0, np.linalg.norm(f))
        if d < 1e-8:

            continue

        for _ in range(18):

    return best_f, best_sc


def _coord_opt(a, stop, seed):

    best = np.clip(a.copy(), 0, None)
    best_sc = _score(best)
    rng = np.random.default_rng(seed)
    mults = [0.0, 0.15, 0.3, 0.5, 0.7, 0.85, 1.15, 1.35, 1.6, 2.2, 3.5]
    lim = 0

            if best[i] < 1e-14:

            ov = best[i]

            for m in mults:

                if best.sum() < 0.01:
                    best[i] = ov
                    continue
                sc = _score(best)
                if sc < bsc - 1e-15:

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        raise ValueError("height_sequence_1 not available")
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0 or float(p.sum()) < 0.01:
        p = np.ones(200)
    return p


def _conv2(a):
    n = len(a)
    if n <= 96:
        return np.convolve(a, a, mode='full')
    fs = next_fast_len(2 * n - 1)
    F = rfft(a, fs)
    c = irfft(F * F, fs)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv2(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _toeplitz(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _solve_lp(f, n, M, timeout=3.0):
    A = _toeplitz(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs',
                      options={'presolve': True, 'time_limit': timeout,
                               'dual_feasibility_tolerance': 1e-8})
    except Exception:
        return None
    if res is None or res.status != 0 or res.x is None:
        return None
    g = np.clip(np.asarray(res.x, dtype=np.float64), 0.0, 1000.0)
    if g.sum() < 1e-15:
        return None
    return g


def _lp_loop(f0, n, stop, seed):
    rng = np.random.default_rng(seed)
    f = np.clip(f0.astype(np.float64).ravel()[:n].copy(), 0, None)
    if f.size < n:
        f = np.pad(f, (0, n - f.size))
    if f.sum() < 1e-10:
        f = np.ones(n)

    best_f, best_sc = f.copy(), _score(f)
    t, stall, max_stall = 0.35, 0, 180
    iter_count = 0

    for _ in range(3000):
        if time.monotonic() > stop - 5:
            break
        b = _conv2(f)
        M = float(b.max())
        if M < 1e-20:
            break

        g = _solve_lp(f, n, M, timeout=3.0)
        if g is None:
            stall += 5
            if stall > max_stall:
                f = np.clip(f + rng.normal(0, 0.06 * max(1.0, f.mean()), n), 0, None)
                stall, max_stall = 0, max_stall + 60
            continue

        gs = g * (f.sum() / g.sum())
        dist = float(np.linalg.norm(gs - f)) / max(1e-10, float(np.linalg.norm(f)))
        if dist < 1e-8:
            stall += 3
            if stall > 30:
                f = np.clip(f + rng.normal(0, 0.02 * max(1.0, f.mean()), n), 0, None)
                stall, t = 0, 0.35
            continue

        improved = False
        for _ in range(18):
            fn = np.clip((1 - t) * f + t * gs, 0, None)
            sc = _score(fn)
            if sc < best_sc - 1e-14:
                best_sc, best_f, f = sc, fn.copy(), fn
                improved = True
                t = min(0.85, t * 1.15)
                stall = 0
                break
            t = max(0.02, t * 0.45)

        if not improved:
            stall += 1
            if stall >= max_stall:
                f = np.clip(f + rng.normal(0, 0.04 * max(1.0, f.mean()), n), 0, None)
                t, stall = 0.35, 0
                max_stall = min(max_stall + 40, 500)
        iter_count += 1

    return best_f, best_sc


def _coord_opt(a, stop, seed):
    n = len(a)
    best = np.clip(a.astype(np.float64).copy(), 0, None)
    best_sc = _score(best)
    if not math.isfinite(best_sc):
        return best, best_sc
    rng = np.random.default_rng(seed)
    single_mults = [0.0, 0.1, 0.2, 0.35, 0.5, 0.65, 0.8, 0.9, 1.1, 1.25, 1.45, 1.8, 2.5, 4.0]
    pair_mults = [0.5, 0.75, 1.0, 1.33, 1.75]
    sweeps = 0
    no_imp_count = 0

    while time.monotonic() < stop - 5:
        sweeps += 1
        improved = False

        for i in range(n):
            if time.monotonic() > stop - 5:
                break
            ov = best[i]
            if ov < 1e-14:
                continue
            bv, bsc = ov, best_sc
            for m in single_mults:
                best[i] = ov * m
                if best.sum() < 0.01:
                    best[i] = ov
                    continue
                sc = _score(best)
                if sc < bsc - 1e-15:
                    bv, bsc = best[i], sc
            best[i] = bv
            if bsc < best_sc - 1e-15:
                best_sc, improved = bsc, True

        npairs = 80 if sweeps % 3 == 0 else 40
        for _ in range(npairs):
            if time.monotonic() > stop - 5:
                break
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            if oi < 1e-14 or oj < 1e-14:
                continue
            for mi in pair_mults:
                for mj in pair_mults:
                    if mi == 1.0 and mj == 1.0:
                        continue
                    best[i], best[j] = oi * mi, oj * mj
                    if best.sum() < 0.01:
                        best[i], best[j] = oi, oj
                        continue
                    sc = _score(best)
                    if sc < best_sc - 1e-15:
                        best_sc, improved = sc, True
            best[i], best[j] = oi, oj

        if not improved:
            no_imp_count += 1
            if no_imp_count >= 5:
                break
            for _ in range(15):
                k = int(rng.integers(n))
                old = best[k]
                best[k] = max(0.0, old + rng.normal(0, 0.03 * max(1.0, best.mean())))
                if best.sum() >= 0.01:
                    sc = _score(best)
                    if sc < best_sc:
                        best_sc, improved = sc, True
                    else:
                        best[k] = old
        else:
            no_imp_count = 0

    return best, best_sc
